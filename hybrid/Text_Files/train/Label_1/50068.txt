import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String x = sc.next();
        long sum = 0;
        int n = x.length() - 1;
        for (int i = 0; i < x.length() / 2; i++) {
            if (x.charAt(i) == '0' && x.charAt(x.length() - 1 - i) == '1') {
                sum += 1L << (n - i);
            }
        }
        System.out.println(sum % 1000000007);
    }
}
