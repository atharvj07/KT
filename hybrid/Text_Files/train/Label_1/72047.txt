def calc_value(a, n):
    while n > 0:
        temp = []
        for i in range(0, len(a), 2):
            if n % 2 == 1:
                temp.append(a[i] | a[i + 1])
            else:
                temp.append(a[i] ^ a[i + 1])
        a = temp
        n -= 1
    return a[0]

n, m = map(int, input().split())
a = list(map(int, input().split()))
for _ in range(m):
    p, b = map(int, input().split())
    a[p-1] = b
    print(calc_value(a[:], n))
