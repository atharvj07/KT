import sys

def solve():
    t = int(input())
    for _ in range(t):
        board = []
        for _ in range(8):
            board.append(input())
        q = [(i, j) for i in range(8) for j in range(8) if board[i][j] == "K"]
        if q[0][0] == q[1][0] or q[0][1] == q[1][1]:
            print("NO")
        else:
            print("YES")

solve()
