#include <iostream>
const int p = 1e9 + 7;
long long fastpow(long long x, int p) {
    if (!p) return 1LL;
    if (p == 1) return x;
    return (fastpow((x * x) % p, p / 2) * (p & 1 ? x : 1)) % p;
}
int main() {
    long long l, r, s = 0;
    std::cin >> l >> r;
    long long c = 1, p1 = 1, p2 = 2;
    while (l <= r) {
        if (l % 2) {
            long long sl = (l + 1) / 2, sr = (r + 1) / 2;
            s = (s + (fastpow(p2, sr) + p - 1) % p - fastpow(p2, sl - 1) + p) % p;
        } else {
            long long sl = l / 2, sr = r / 2;
            s = (s + (fastpow(p1, sr) + p - 1) % p - fastpow(p1, sl - 1) + p) % p;
        }
        l = (l + p1 * c + 1) / p2 * p2;
        r = r / p2 * p2;
        c *= 2;
        p1 = p2;
        p2 *= 2;
    }
    std::cout << s << std::endl;
    return 0;
}
