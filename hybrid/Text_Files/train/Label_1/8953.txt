#include<bits/stdc++.h>
using namespace std;
int main(){
    int q; cin>>q;
    while(q--){
        int n,k; cin>>n>>k;
        string s; cin>>s;
        int ans=0;
        for(int i=0;i<n-k+1;i++){
            int idx=i%k;
            if(s[i]!=s[idx] && s[idx]!='?') ++ans;
        }
        cout<<ans<<endl;
    }
    return 0;
}
