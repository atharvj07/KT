#include <iostream>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int a[n], b[n];
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    bool ok = true;
    int cnt0 = 0, cnt1 = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] == 0) {
        cnt0++;
      } else if (a[i] == 1) {
        cnt1++;
      }
    }
    for (int i = 0; i < n; i++) {
      if (b[i] == 0) {
        cnt0--;
      } else if (b[i] == 1) {
        cnt1--;
      } else {
        ok = false;
        break;
      }
    }
    if (cnt0 < 0 || cnt1 < 0 || ok == false) {
      cout << "NO\n";
    } else {
      cout << "YES\n";
    }
  }
}
