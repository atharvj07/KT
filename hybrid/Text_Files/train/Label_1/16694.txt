#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007
#define fi first
#define se second
#define pb push_back
#define mp make_pair
#define endl '\n'
const ll N = 2e5 + 10;
ll n, k, a[N], b[N], st[N], en[N], sum[N], tme = 0;
map<ll, ll> fp, fb;
map<pair<ll, ll>, ll> fs;
int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i] >> b[i];
        fp[a[i]]++, fp[b[i] + 1]--, fb[a[i]]++, fb[b[i] + 1]--;
    }
    for (auto x : fp) {
        if (x.se > 0) {
            st[++tme] = x.fi;
        } else {
            en[tme] = x.fi - 1;
        }
    }
    for (auto x : fb) {
        if (x.se > 0) {
            st[++tme] = x.fi;
        } else {
            en[tme] = x.fi - 1;
        }
    }
    sort(st + 1, st + 1 + tme);
    sort(en + 1, en + 1 + tme);
    for (int i = 1; i <= tme; ++i) {
        if (st[i] == en[i]) continue;
        fs[mp(st[i], en[i])] = en[i] - st[i] + 1;
    }
    for (auto x : fs) sum[k] = (sum[k] + x.se) % mod; // initialize sum with k
    for (int j = k - 1; j >= 1; --j) {
        for (auto x : fs) sum[j] = (sum[j] + (1ll * sum[j + 1] * x.se) % mod) % mod;
    }
    cout << sum[1] << endl;
}
