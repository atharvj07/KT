import sys
input=lambda: sys.stdin.readline().strip()
for i in range(int(input())):
    pos,color=input().split()
    pos,color=int(pos),color.strip()
    if color=='B':
        if i%2==0:print(pos,end=' ')
        else:print(-pos,end=' ')
    elif color=='R':
        if i%2==1:print(pos,end=' ')
        else:print(-pos,end=' ')
