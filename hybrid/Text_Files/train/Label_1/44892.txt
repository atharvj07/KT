import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int k = input.nextInt();

    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = input.nextInt();
    }

    Arrays.sort(a);

    int min = a[0];
    int max = a[n - 1];
    int ans = min;

    if (k >= 2) {
      ans = min + (a[1] - min) / 2;
    }

    if (k >= 3) {
      ans = min + (max - min) / 2;
    }

    System.out.println(ans);
  }
}
