def find_ate_candies(n, k):
    left, right = 0, n
    while left < right:
        mid = (left + right) // 2
        if mid * (mid + 1) // 2 - (n - mid) > k:
            right = mid
        else:
            left = mid + 1
    return n - left + 1

n, k = map(int, input().split())
print(find_ate_candies(n, k))
