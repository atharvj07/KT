#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
vector<ll> pf(ll n)
{
	vector<ll> v;
	ll k=2;
	while(n>1)
	{
			if(n%k==0)
			{
				v.push_back(k);
				while(n%k==0)
				{
					n/=k;
				}
			}
			k++;
	}
	return v;
}
int main()
{
	int t;
	cin>>t;
	while(t--)
	{
		ll n,k=0,i;
		cin>>n;
		vector<ll> v;
		v=pf(n);
		k=v.size();
		cout<<k<<endl;
		for(i=0;i<v.size();i++)
		{
			cout<<v[i]<<" ";
		}
		cout<<endl;
	}
	return 0;
}
