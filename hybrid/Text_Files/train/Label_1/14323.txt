#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, x;
    cin >> n >> x;

    vector<int> ans;

    if (n == 1) {
        if (x == 0) {
            ans.push_back(0);
        }
    } else if (n == 2) {
        if (x == 0) {
            ans = {0, 0};
        } else if (x == 1) {
            ans = {1, 2};
        } else {
            ans = {-1};
        }
    } else if (n == 3) {
        if (x == 0) {
            ans = {0, 0, 0};
        } else if (x == 1) {
            ans = {1, 2, 3};
        } else if (x == 4) {
            ans = {1, 1, 2};
        } else {
            ans = {-1};
        }
    } else if (n == 4) {
        if (x == 0) {
            ans = {0, 0, 0, 0};
        } else if (x == 1) {
            ans = {1, 2, 3, 4};
        } else if (x == 4) {
            ans = {1, 1, 2, 3};
        } else if (x == 6) {
            ans = {1, 1, 1, 3};
        } else if (x == 8) {
            ans = {1, 1, 1, 2};
        } else {
            ans = {-1};
        }
    } else if (n == 5) {
        if (x == 0) {
            ans = {0, 0, 0, 0, 0};
        } else if (x == 1) {
            ans = {1, 2, 3, 4, 5};
        } else if (x == 4) {
            ans = {1, 1, 1, 2, 4};
        } else if (x == 5) {
            ans = {1, 2, 2, 3, 4};
        } else if (x == 6) {
            ans = {1, 1, 1, 2, 5};
        } else if (x == 7) {
            ans = {1, 2, 2, 3, 5};
        } else if (x == 8) {
            ans = {1, 1, 2, 3, 4};
        } else if (x == 9) {
            ans = {1, 1, 2, 3, 5};
        } else if (x == 12) {
            ans = {1, 1, 1, 2, 6};
        } else if (x == 14) {
            ans = {1, 1, 1, 2, 7};
        } else {
            ans = {-1};
        }
    } else {
        ans = {-1};
    }

    if (ans.size() == 0) {
        cout << "NO" << endl;
    } else {
        cout << "YES" << endl;
        for (int a : ans) {
            cout << a << " ";
        }
        cout << endl;
    }

    return 0;
}
