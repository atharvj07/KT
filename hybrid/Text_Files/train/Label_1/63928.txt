n, m, i, j, a, b = map(int, input().split())
if abs(i - 1) + abs(j - m) <= a + b or abs(i - n) + abs(j - 1) <= a + b or abs(i - n) + abs(j - m) <= a + b or i - 1 <= a and j - m <= b:
    print(1)
elif i - 1 <= a and j - 1 <= b or i - n <= a and j - 1 <= b or i - n <= a and j - m <= b or i - 1 <= a and j - m <= b:
    print(2)
else:
    print("Poor Inna and pony!")
