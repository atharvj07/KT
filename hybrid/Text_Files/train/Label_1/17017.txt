#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  for (int d = -1000; d <= 1000; d++) {
    if (d == 0) { continue; }
    int cnt = 0;
    for (int x : a) { if ((x < 0 && -x % d > 0) || x > 0) { cnt++; } }
    if (cnt > n / 2) { ans = d; }
  }
  cout << ans << endl;
}
