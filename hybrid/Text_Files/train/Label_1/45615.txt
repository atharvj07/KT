import java.util.*;
import java.io.*;

class Main {
    static long b = 0;
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int q = Integer.parseInt(br.readLine());
        for (int i = 0; i < q; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int p = Integer.parseInt(st.nextToken());
            int x = 0;
            if (st.hasMoreTokens()) x = Integer.parseInt(st.nextToken());
            switch (p) {
                case 0: System.out.println(test(x)); break;
                case 1: b = set(x); break;
                case 2: b = clear(x); break;
                case 3: b = flip(x); break;
                case 4: System.out.println(all()); break;
                case 5: System.out.println(any()); break;
                case 6: System.out.println(none()); break;
                case 7: System.out.println(count()); break;
                case 8: System.out.println(val()); break;
            }
        }
    }

    static long set(int x) {
        return b |= (1L << x);
    }

    static long clear(int x) {
        return b &= ~(1L << x);
    }

    static long flip(int x) {
        return b ^= (1L << x);
    }

    static long test(int x) {
        return (b & (1L << x)) >> x;
    }
    
    static long all() {
        return b == (1L << 64) - 1 ? 1 : 0;
    }

    static long any() {
        return b != 0 ? 1 : 0;
    }

    static long none() {
        return b == 0 ? 1 : 0;
    }

    static long count() {
        long ans = 0;
        for (int i = 0; i < 64; i++) {
            ans += test(i);
        }
        return ans;
    }

    static long val() {
        return b;
    }
}
