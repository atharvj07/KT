#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a, b, c, t;
  cin >> n >> a >> b >> c >> t;
  priority_queue<long long> pq;
  long long tmp, sum = 0;
  for (int i = 0; i < n; i++) {
    cin >> tmp;
    pq.push(a - (tmp - 1) * b);
  }
  int cur = 0;
  while (cur < t && !pq.empty()) {
    sum += c * cur + pq.top();
    pq.pop();
    cur++;
  }
  while (cur < t) {
    sum += c * cur;
    cur++;
  }
  cout << sum << endl;
}
