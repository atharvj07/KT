#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int maxN = 100+5;

bool canGo(int n,int k,int l,vector<int> &d){
	vector<int> times;
	for(int i=0;i<k;++i){
		times.push_back(i);
		times.push_back(-i);
	}
	for(int i=0;i<n;++i){
		for(int t:times){
			if(d[i]+t>l)
				return 0;
		}
	}
	return 1;
}

int main(){
	int t; cin>>t;
	while(t--){
		int n,k,l; cin>>n>>k>>l;
		vector<int> d(n);
		for(int i=0;i<n;++i)
			cin>>d[i];
		cout<<(canGo(n,k,l,d)?"Yes\n":"No\n");
	}
	return 0;
}
