#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int n;
    cin >> n;
    vector<ll> a(n);
    for (ll &x : a) cin >> x;
    int one = 0;
    for (ll x : a) one += x == 1;
    if (one == 0 || one == n) { cout << "NO" << endl; return 0; }
    vector<int> ans(n);
    for (int i = 0; i < n; i++) {
        if (a[i] == 1) ans[i] = 1;
        else { ans[i] = 2; break; }
    }
    cout << "YES" << endl;
    for (int x : ans) cout << x << " ";
    cout << endl;
    return 0;
}
