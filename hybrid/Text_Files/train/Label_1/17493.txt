#include <bits/stdc++.h>
using namespace std;
typedef int ll;
const int N = 2e5 + 10;

ll n, q, m, a[N], ans[N], mp[N];

ll get(ll x) {
  return x > n ? x - n : x;
}

int main() {
  ll t, l, r;
  cin >> n >> q >> m;
  mp[0] = n;
  for (ll i = 1; i <= n; i++) {
    cin >> a[i];
    mp[i] = i;
  }
  while (q--) {
    cin >> t >> l >> r;
    if (t == 1) {
      swap(mp[r], mp[l]);
    } else {
      reverse(mp + l, mp + r + 1);
    }
  }
  for (ll i = 1; i <= m; i++) {
    cin >> l;
    cout << a[get(mp[l])] << " ";
  }
  cout << endl;
  return 0;
}
