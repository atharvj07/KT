#include <iostream>
#include <algorithm>
#include <vector>
#include <map>
using namespace std;
int main() {
    int n;
    cin >> n;
    map<string, int> m;
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        for (int j = 0; j < s.size(); j++) {
            if (s[j] == 'h') s[j] = 'k';
        }
        string t = s;
        for (int j = 0; j < s.size(); j++) {
            if (s[j] == 'u') s[j] = 'o';
        }
        m[s]++, m[t]++;
    }
    cout << m.size() << endl;
    return 0;
}
