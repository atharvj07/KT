#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  int k; cin >> k;
  int a[4][4], cnt = 0;
  for (int i = 0; i < 4; i++) {
    for (int j = 0; j < 4; j++) {
      char c; cin >> c;
      if (c != '.') {
        a[i][j] = c - '0';
        cnt++;
      }
    }
  }
  if (cnt <= 2 * k) {
    sort(&a[0][0], &a[0][0] + cnt);
    for (int i = 1; i < cnt; i++)
      if (a[i] - a[i - 1] > 1) {
        cout << "NO" << endl;
        return 0;
      }
    cout << "YES" << endl;
    return 0;
  }
  cout << "NO" << endl;
  return 0;
}
