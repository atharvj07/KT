from sys import stdin

def main():
    n, m = map(int, stdin.readline().strip().split())
    h = stdin.readline().strip()
    v = stdin.readline().strip()
    g = [[0] * (m+1) for _ in range(n+1)]
    for i in range(1, n):
        for j in range(m):
            if h[i-1] == '<' and v[j] == 'v':
                g[i][j+1] = 1
            elif h[i-1] == '>' and v[j] == '^':
                g[i][j+1] = 1
    for i in range(n):
        for j in range(1, m):
            if h[i] == '<' and v[j-1] == '^':
                g[i+1][j] = 1
            elif h[i] == '>' and v[j-1] == 'v':
                g[i+1][j] = 1
    for i in range(1, n+1):
        vis = [False] * (m+1)
        if dfs(g, (1, 1), (i, m), vis):
            print("YES")
            return
    print("NO")

def dfs(g, start, end, vis):
    if start == end:
        return True
    vis[start[1]] = True
    for i in range(start[1]+1):
        if g[start[0]][i] and not vis[i]:
            if dfs(g, (start[0], i), end, vis):
                return True
    for i in range(start[1]-1):
        if g[start[0]][i+1] and not vis[i+1]:
            if dfs(g, (start[0], i+1), end, vis):
                return True
    return False

main()
