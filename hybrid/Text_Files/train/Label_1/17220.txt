#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int n;
        cin >> n;
        vector<string> p(n);
        map<string, int> mp;
        for (int i = 0; i < n; i++)
        {
            cin >> p[i];
            if (mp.find(p[i]) != mp.end())
                mp[p[i]]++;
            else
                mp[p[i]] = 0;
        }
        int cnt = 0;
        for (auto &i : mp)
            cnt += i.second;
        cout << cnt << "\n";
        for (int i = 0; i < n; i++)
        {
            string pp = p[i];
            if (mp.find(pp) == mp.end())
            {
                cout << pp << "\n";
            }
            else if (mp[pp])
            {
                int j = 0;
                while (pp == p[i])
                    pp[j++] = (pp[j] + 1 - '0') % 10 + '0';
                cout << pp << "\n";
            }
            mp[p[i]]--;
            if (mp.find(pp) == mp.end())
                mp[pp] = 0;
            else
                mp[pp]++;
        }
    }
    return 0;
}
