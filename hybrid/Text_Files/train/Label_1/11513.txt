#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &i : a)
        cin >> i, i--;
    vector<array<int, 2>> dist(n);
    for (int i = 0; i < n; i++)
    {
        dist[i][0] = i;
        dist[i][1] = -1;
    }
    vector<int> last(n, -1);
    for (int i = 0; i < n; i++)
    {
        if (last[a[i]] == -1)
            last[a[i]] = i;
        else
        {
            if (dist[last[a[i]]][1] == -1)
                dist[last[a[i]]][1] = i;
            else
                dist[last[a[i]]][1] = max(dist[last[a[i]]][1], i);
            last[a[i]] = i;
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++)
        ans = max({ans, dist[i][1] - i, n - 1 - dist[i][0]});
    cout << ans << endl;
}
