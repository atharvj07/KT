import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] freq = new int[3];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'R') freq[0]++;
            else if (s.charAt(i) == 'G') freq[1]++;
            else freq[2]++;
        }
        ArrayList<Character> possible = new ArrayList<>();
        if (freq[0] % 2 == 1) possible.add('R');
        if (freq[1] % 2 == 1) possible.add('G');
        if (freq[2] % 2 == 1) possible.add('B');
        Collections.sort(possible);
        for (Character c : possible) System.out.print(c);
    }
}
