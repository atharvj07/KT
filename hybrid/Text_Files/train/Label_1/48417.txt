import java.util.*;
import java.io.*;

class Main {
    static int n, k, m;
    static int[] t;
    static int[] dp;
    static int[] sum;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = br.readLine().split(" ");
        n = Integer.parseInt(parts[0]);
        k = Integer.parseInt(parts[1]);
        m = Integer.parseInt(parts[2]);
        t = new int[46];
        parts = br.readLine().split(" ");
        for (int i = 1; i <= k; i++) t[i] = Integer.parseInt(parts[i - 1]);
        dp = new int[m + 1];
        sum = new int[m + 1];
        for (int i = 1; i <= k; i++) {
            for (int j = m; j >= t[i]; j--) {
                dp[j] = Math.max(dp[j], dp[j - t[i]] + 1);
            }
        }
        for (int i = 1; i <= m; i++) {
            sum[i] = sum[i - 1];
            if (dp[i] >= k) sum[i]++;
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            ans += sum[m];
            m -= k;
            if (m < 0) break;
        }
        System.out.println(ans);
    }
}
