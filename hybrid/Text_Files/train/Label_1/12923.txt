#include<bits/stdc++.h>
using namespace std;
int main(){
    int s,x1,x2,t1,t2,p,d;
    cin>>s>>x1>>x2>>t1>>t2>>p>>d;
    long long ans=1e18;
    if(d==1){
        ans=min(ans,1ll*abs(x2-x1)*t2);
        if(p>x1){
            ans=min(ans,1ll*abs(s-p)*t1+1ll*abs(x2-x1)*t2);
        }
        else{
            ans=min(ans,1ll*abs(p-x1)*t2+1ll*abs(s-x2)*t1);
        }
    }
    else{
        ans=min(ans,1ll*abs(x2-x1)*t2);
        if(p<x1){
            ans=min(ans,1ll*abs(p)*t1+1ll*abs(x2-x1)*t2);
        }
        else{
            ans=min(ans,1ll*abs(x1)*t2+1ll*(s-p)*t1);
        }
    }
    cout<<ans<<endl;
    return 0;
}
