#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    int ans = 0;
    vector<int> freq(26);
    for (int i = 0; i < n; i++) {
        freq[a[i] - 'a']++;
        freq[b[i] - 'a']--;
    }
    for (int i = 0; i < 26; i++) {
        if (freq[i] > 0) ans += freq[i];
    }
    cout << ans << endl;
    return 0;
}
