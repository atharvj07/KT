#include <iostream>
#include <vector>
using namespace std;

int main() {

    int n, a;
    cin >> n;

    vector<int> cycles;
    while (n--) {
        cin >> a;
        cycles.push_back(a);
    }

    for (int c : cycles) {
        int sum = 0;
        for (int i = 1; i * i <= c; i++) {
            if (c % i == 0) {
                sum += 2;
                if (i * i == c) sum--;
            }
        }
        if (sum % 2 == 0) {
            cout << 2 << endl;
        } else {
            cout << 1 << endl;
        }
    }

    return 0;
}
