#include<bits/stdc++.h>
using namespace std;

int main() {
    int up, down;
    cin >> up >> down;
    vector<int> stop_heights;
    if (!up) {
        stop_heights.push_back(1);
        for (int i = 0; i < down; i++) stop_heights.push_back(stop_heights[i] - 1);
    } else if (!down) {
        for (int i = 0; i < up; i++) stop_heights.push_back(i + 1);
    } else {
        int min_down_height = 2;
        int max_up_height = up;
        int curr_height = max_up_height;
        stop_heights.push_back(curr_height);
        while (down--) {
            curr_height--;
            if (curr_height < min_down_height) {
                min_down_height++;
                while (stop_heights.back() == min_down_height) {
                    min_down_height++;
                    max_up_height--;
                    curr_height = max_up_height;
                    stop_heights.push_back(curr_height);
                }
            }
            stop_heights.push_back(curr_height);
        }
    }
    sort(stop_heights.begin(), stop_heights.end());
    for (auto s : stop_heights) cout << s << " ";
}
