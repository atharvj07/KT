#include <bits/stdc++.h>
using namespace std;

int gcd(int a, int b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
typedef long long ll;
int main() {
  int n;
  cin >> n;
  vector<ll> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  ll mn = 1e18;
  pair<int, int> ans;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      ll val = gcd(a[i], a[j]);
      if (val < mn) {
        mn = val;
        ans = {i + 1, j + 1};
      }
    }
  }
  cout << ans.first << " " << ans.second << endl;
  return 0;
}
