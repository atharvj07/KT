import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;

class Main {
    private static int n, m;
    private static final int SIZE = 1000;
    private static int[][] t = new int[SIZE][SIZE];

    public static void main(String[] args) throws IOException {
        StringBuilder sb = new StringBuilder();
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        String[] parts = line.split("\\s+");
        n = Integer.parseInt(parts[0]);
        m = Integer.parseInt(parts[1]);

        for (int i = 0; i < m; i++) {
            line = br.readLine();
            parts = line.split("\\s+");
            int op = Integer.parseInt(parts[0]);
            int x0 = Integer.parseInt(parts[1]);
            int y0 = Integer.parseInt(parts[2]);
            int x1 = Integer.parseInt(parts[3]);
            int y1 = Integer.parseInt(parts[4]);

            if (op == 1) {
                sb.append(query(x0, y0, x1, y1) + "\n");
            } else if (op == 2) {
                update(x0, y0, x1, y1, Integer.parseInt(parts[5]));
            }
        }

        System.out.print(sb.toString());
    }

    private static void update(int x0, int y0, int x1, int y1, int v) {
        t[x0 - 1][y0 - 1] ^= v;
        t[x0 - 1][y1] ^= v;
        t[x1][y0 - 1] ^= v;
        t[x1][y1] ^= v;
    }

    private static int query(int x0, int y0, int x1, int y1) {
        return t[x1][y1] ^ t[x1][y0 - 1] ^ t[x0 - 1][y1] ^ t[x0 - 1][y0 - 1];
    }
}
