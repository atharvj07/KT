mod = 1000000007
def find_num_ways(n, a):
    a.sort()
    cnt = [0] * (n + 1)
    for x in a:
        cnt[x] += 1
    val = num = 1
    for x in range(1, n + 1):
        val = (val * x) % mod
        num = (num * cnt[x]) % mod
    return (val * pow(num, mod - 2, mod)) % mod
n = int(input())
a = list(map(int, input().split()))
print(find_num_ways(n, a))
