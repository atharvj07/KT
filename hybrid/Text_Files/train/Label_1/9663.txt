#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n, ans, a[100005], b[100005], u, bb, cnt;
ll c[100005];
int main()
{
    cin >> n;
    for(int i = 0; i < n; i++)
    {
        cin >> a[i] >> b[i];
        c[i] = 0;
    }
    for(int i = 0; i < n; i++)
    {
        if(a[i]%2 == 1)
        {
            bb = a[i] + 1;
            bb = bb/2;
        }
        else
        {
            bb = a[i]/2;
        }
        a[i] = bb;
    }
    for(int i = 0; i < n; i++)
    {
        u = a[i];
        cnt = 0;
        for(int j = i; j < n; j++)
        {
            if(c[j] == 0)
            {
                if(a[j] <= u)
                {
                    cnt = cnt + a[j];
                    if(cnt >= b[j])
                    {
                        u = u - (cnt - b[j]);
                        c[j] = 1;
                    }
                    if(u <= 0)
                    break;
                }
            }
        }
        ans = ans + (u*2);
    }
    cout << ans << "\n";
    return 0;
}
