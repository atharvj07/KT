import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextLong();
        }
        Arrays.sort(a);
        long mindiff = Long.MAX_VALUE;
        long[] ans = new long[n];
        for (int i = 0; i < n; i++) {
            long[] b = new long[n];
            for (int j = 0; j < n; j++) {
                b[j] = a[(i + j) % n];
            }
            long diff = 0;
            for (int j = 0; j < n - 1; j++) {
                diff += Math.abs(b[j] - b[j + 1]);
            }
            if (diff < mindiff) {
                mindiff = diff;
                ans = b;
            }
        }
        for (long l : ans) {
            System.out.print(l + " ");
        }
        System.out.println();
    }
}
