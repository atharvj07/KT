#include<bits/stdc++.h>
const int P = 1000000007;
int pow(int x, int n) {
	int res = 1;
	for (; n; n >>= 1, x = x * 1LL * x % P)
		if (n & 1)
			res = res * 1LL * x % P;
	return res;
}
int main() {
	int k;
	scanf("%d", &k);
	puts(pow(2, k) == 1 ? "0" : std::to_string((pow(2, k) - 1 + P) % P).c_str());
}
