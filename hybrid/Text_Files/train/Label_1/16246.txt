#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  if (a[0] == 0) cout << "cslnb" << endl;
  else if (a[0] == a[n - 1]) cout << "sjfnb" << endl;
  else if (((a[0] + 1) * 2) > a[n - 1]) cout << "cslnb" << endl;
  else cout << "sjfnb" << endl;
  return 0;
}
