#include <bits/stdc++.h>

using namespace std;

const int INF = 0x3f3f3f3f;
const int MAXN = 100000;

int n;
int a[MAXN][3];
int b[MAXN];
vector<int> adj[MAXN];

int dfs(int u, int p, int c) {
  b[u] = c;
  int dp[3];
  dp[0] = dp[1] = dp[2] = INF;
  for (int v : adj[u]) {
    if (v == p) {
      continue;
    }
    int ndp[3];
    for (int i = 0; i < 3; i++) {
      ndp[i] = INF;
      for (int j = 0; j < 3; j++) {
        if (i == j || dp[j] == INF) {
          continue;
        }
        ndp[i] = min(ndp[i], dp[j] + a[v][c]);
      }
    }
    dp[0] = ndp[0];
    dp[1] = ndp[1];
    dp[2] = ndp[2];
  }
  return dp[0];
}

int32_t main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 3; j++) {
      cin >> a[i][j];
    }
  }
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    u--;
    v--;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  int ans = dfs(0, -1, 0);
  if (ans == INF) {
    cout << -1 << endl;
    return 0;
  }
  dfs(0, -1, 0);
  cout << ans << endl;
  for (int i = 0; i < n; i++) {
    cout << b[i] + 1 << " ";
  }
  cout << endl;
  return 0;
}
