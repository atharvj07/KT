import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt();
        String[] walls = new String[2];
        for (int i = 0; i < 2; ++i) {
            walls[i] = in.next();
        }
        int[] pos = {0, 0};
        for (int t = 0; t < n; ++t) {
            if (t > 0) {
                ++pos[t % 2];
            }
            if (pos[0] == n || pos[1] == n) {
                System.out.println("YES");
                return;
            }
            if (pos[t % 2] + 1 < n && walls[t % 2].charAt(pos[t % 2] + 1) != 'X') {
                ++pos[t % 2];
            } else if (pos[1 - t % 2] + k < n && walls[1 - t % 2].charAt(pos[1 - t % 2] + k) != 'X') {
                pos[1 - t % 2] += k;
            } else {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("NO");
    }
}
