import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        while (t-- > 0) {
            String[] ab = br.readLine().split(" ");
            int a = Integer.parseInt(ab[0]);
            int b = Integer.parseInt(ab[1]);
            char[] mines = br.readLine().toCharArray();

            int n = mines.length;
            int[][] dp = new int[n][3];
            for (int i = 0; i < n; i++) {
                Arrays.fill(dp[i], -1);
            }

            System.out.println(rec(0, n, 0, mines, dp, a, b));
        }
    }

    public static int rec(int i, int n, int type, char[] mines, int[][] dp, int a, int b) {
        if (i == n) {
            return 0;
        }
        if (dp[i][type] != -1) {
            return dp[i][type];
        }

        int ans = Integer.MAX_VALUE;
        if (type == 0) {
            if (mines[i] == '0') {
                ans = Math.min(ans, rec(i + 1, n, 0, mines, dp, a, b));
            } else {
                ans = Math.min(ans, rec(i + 1, n, 1, mines, dp, a, b));
                ans = Math.min(ans, rec(i + 1, n, 2, mines, dp, a, b) + b);
                if (i + 1 < n) {
                    ans = Math.min(ans, rec(i + 2, n, 0, mines, dp, a, b) + a);
                }
            }
        } else if (type == 1) {
            if (mines[i] == '0') {
                ans = Math.min(ans, rec(i + 1, n, 1, mines, dp, a, b));
            } else {
                ans = Math.min(ans, rec(i + 1, n, 1, mines, dp, a, b));
                ans = Math.min(ans, rec(i + 1, n, 2, mines, dp, a, b) + b);
                if (i + 1 < n) {
                    ans = Math.min(ans, rec(i + 2, n, 0, mines, dp, a, b) + a);
                    ans = Math.min(ans, rec(i + 2, n, 1, mines, dp, a, b) + a * 2);
                }
                if (i + 2 < n) {
                    ans = Math.min(ans, rec(i + 3, n, 0, mines, dp, a, b) + a);
                }
            }
        } else if (type == 2) {
            if (mines[i] == '0') {
                ans = Math.min(ans, rec(i + 1, n, 2, mines, dp, a, b));
            } else {
                ans = Math.min(ans, rec(i + 1, n, 2, mines, dp, a, b));
                if (i + 1 < n) {
                    ans = Math.min(ans, rec(i + 2, n, 1, mines, dp, a, b) + a);
                    ans = Math.min(ans, rec(i + 2, n, 2, mines, dp, a, b) + a * 2);
                }
                if (i + 2 < n) {
                    ans = Math.min(ans, rec(i + 3, n, 0, mines, dp, a, b) + a);
                }
            }
        }

        return dp[i][type] = ans;
    }
}
