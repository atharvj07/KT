#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    ll n;
    cin >> n;
    string s[n];
    for (ll i = 0; i < n; i++) { cin >> s[i]; }
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
        for (ll j = i; j < n; j++) {
            stack<char> st;
            bool valid = true;
            string t = s[i] + s[j];
            for (ll k = 0; k < t.size(); k++) {
                char c = t[k];
                if (c == '(') st.push(c);
                else {
                    if (st.empty()) { valid = false; } else st.pop();
                }
            }
            if (!st.empty()) valid = false;
            if (valid) ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
