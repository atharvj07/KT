#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0, cur = 0, l = 0, r = 0;
  for (int i = 0; i < n; i++) {
    if (!a[i]) continue;
    if (a[i] > r) {
      ans++;
      l = i;
      cur = 1;
    } else if (a[i] > cur) {
      cur++;
    }
    r = max(r, a[i]);
  }
  cout << ans << endl;
  return 0;
}
