#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1000005;
vector<pair<int,int> > v[MAXN];
int n, m, a[MAXN];
bool solve() {
    for (int i = 0; i <= n * m; i++) v[i].clear();
    for (int i = 1; i <= n * m; i++) {
        a[i] = i - 1;
        for (int j = i; j <= n * m; j++) {
            int dx = j - i;
            int dy = j / m - i / m;
            if (i + dx <= n * m) v[i + dx].push_back({j, dy + 1});
            if (i - dx >= 1) v[i - dx].push_back({j, -dy - 1});
        }
    }
    vector<int> order;
    queue<int> q;
    q.push(0);
    while (!q.empty()) {
        int cur = q.front(); q.pop();
        order.push_back(cur);
        for (int i = 0; i < v[cur].size(); i++) {
            int nn = v[cur][i].first;
            if (a[nn] > a[cur]) {
                a[nn] = a[cur];
                q.push(nn);
            }
        }
    }
    reverse(order.begin(), order.end());
    for (int i = 0; i < order.size(); i++) {
        if (a[order[i]] != order[i]) {
            return false;
        }
    }
    for (int i = 1; i <= n * m; i++) {
        if (a[i] == 0) {
            cout << i / m + 1 << ' ' << i % m + 1 << endl;
            for (int j = 0; j < v[i].size(); j++) {
                int nn = v[i][j].first;
                int dd = v[i][j].second;
                if (dd == 0) {
                    cout << nn / m + 1 << ' ' << nn % m + 1;
                    break;
                }
            }
            return true;
        }
    }
    return false;
}
int main() {
    int t; cin >> t;
    if (t == 1) {
        cout << "-1" << endl;
        return 0;
    }
    n = sqrt(t);
    m = t / n;
    int flag = 0;
    while (n * m < t) {
        if (!flag) m += 1, flag = 1;
        else n += 1;
    }
    if (n * m != t) {
        cout << "-1" << endl;
        return 0;
    }
    for (int i = 1; i <= t; i++) {
        int x; cin >> x;
        if (x == 0) {
            cout << "-1" << endl;
            return 0;
        }
        if (x >= t) {
            cout << "-1" << endl;
            return 0;
        }
    }
    cout << n << " " << m << endl;
    bool ok = solve();
    if (!ok) cout << "-1" << endl;
    return 0;
}
