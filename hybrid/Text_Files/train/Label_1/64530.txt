import math
def ll(a):
    return int(a) if a>0 else int(a)-1
def rr(a):
    return int(a) if a>0 else int(a)+1
def is_prime(n):
    if n<=3:
        return n>1
    if(n%2==0 or n%3==0):
        return False
    i=5
    while(i*i<=n):
        if(n%i==0 or n%(i+2)==0):
            return False
        i+=6
    return True
def is_palindromic(n):
    return str(n)==str(n)[::-1]
def func(p,q):
    p*=210
    a,b=p,q
    for i in range(3,int(math.sqrt(p))+1,2):
        if p%i==0:
            while p%i==0:
                p//=i
            if i%4==1:
                a//=i
                b*=i
    if p>1:
        if p%4==1:
            a//=p
            b*=p
    return (a,b)
p,q=map(int,input().split())
a,b=func(p,q)
for i in range(10000):
    if is_prime(i):
        if a*rr(i)//b<=is_palindromic(i):
            print(i-1)
            return
    elif is_palindromic(i):
        if a*ll(i)//b>=is_prime(i):
            print(i)
            return
print('Palindromic tree is better than splay tree')
