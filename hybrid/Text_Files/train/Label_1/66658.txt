import sys

def solve(n, next):
    if max(next) > n:
        return -1
    p = [i for i in range(1, n + 1)]
    for i in range(n):
        if next[i] != -1:
            for j in range(i + 1, n):
                if p[j] > p[i]:
                    p[i], p[j] = p[j], p[i]
                    break
    for i in range(n):
        if next[i] != -1 and next[i] != p[i]:
            return -1
    return p

t = int(input())
for _ in range(t):
    n = int(input())
    next = list(map(int, input().split()))
    res = solve(n, next)
    print(*res) if res != -1 else print(-1)
