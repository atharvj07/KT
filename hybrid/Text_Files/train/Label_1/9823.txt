#include <iostream>
#include <vector>

int main() {
  int n, m, h;
  std::cin >> n >> m >> h;
  std::vector<int> u(n);
  for (int &v : u) { std::cin >> v; }
  std::vector<int> c1(m), c2(m);
  for (int i = 0; i < m; i++) {
    std::cin >> c1[i] >> c2[i];
    c1[i]--, c2[i]--;
  }
  std::vector<int> cnt(h);
  for (int i = 0; i < n; i++) { cnt[u[i]]++; }
  for (int i = 0; i < m; i++) {
    if (cnt[u[c1[i]]] == 1 && cnt[u[c2[i]]] == 1) {
      std::cout << 1 << '\n' << c1[i] + 1 << '\n';
      return 0;
    }
    if (cnt[u[c2[i]]] == 1 && cnt[u[c1[i]]] > 1) {
      std::cout << 1 << '\n' << c2[i] + 1 << '\n';
      return 0;
    }
    if (cnt[u[c1[i]]] == 1 && cnt[u[c2[i]]] > 1) {
      std::cout << 1 << '\n' << c1[i] + 1 << '\n';
      return 0;
    }
  }
  std::vector<int> ans;
  for (int i = 0; i < n; i++) {
    if (cnt[u[i]] > 1 && !ans.empty()) {
      std::cout << ans.size() << '\n';
      for (int a : ans) std::cout << a + 1 << ' ';
      std::cout << '\n';
      return 0;
    }
    ans.push_back(i);
  }
  std::cout << ans.size() << '\n';
  for (int a : ans) std::cout << a + 1 << ' ';
  std::cout << '\n';
  return 0;
}
