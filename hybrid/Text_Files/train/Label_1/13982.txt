#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

pair<int, int> get_min(const vector<pair<int, int>> &l, const vector<pair<int, int>> &r, int n) {
    int min_len = n;
    int pos = 0;
    for (int i = 0; i < (int)l.size(); i++) {
        for (int j = 0; j < (int)r.size(); j++) {
            if (l[i].first < r[j].first) {
                int len = n - (r[j].second - l[i].first + 1);
                if (l[i].first == r[j].second) {
                    len--;
                }
                if (len < min_len) {
                    min_len = len;
                    pos = i;
                }
            }
        }
    }
    return {pos, min_len};
}

int main() {
    int n;
    cin >> n;
    vector<string> w(n);
    for (int i = 0; i < n; i++) cin >> w[i];
    vector<pair<int, int>> l, r;
    for (int i = 0; i < n; i++) {
        for (int j = i; j < n; j++) {
            bool ok = true;
            for (int k = 0; k <= j - i && ok; k++) {
                ok &= w[i + k] == w[j + k];
            }
            if (ok) {
                l.emplace_back(i, j);
            }
            ok = true;
            for (int k = 0; k <= j - i && ok; k++) {
                ok &= w[j + k] == w[j - k];
            }
            if (ok && i != j) {
                r.emplace_back(i, j);
            }
        }
    }
    sort(l.begin(), l.end());
    sort(r.begin(), r.end());
    pair<int, int> ans = get_min(l, r, n);
    if (ans.second == n) {
        ans = get_min(r, l, n);
    }
    cout << ans.second << endl;
    return 0;
}
