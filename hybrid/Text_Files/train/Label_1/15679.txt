#include <iostream>
#include <string>
#include <cstdio>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    if (k > 13 * n || k < 0) {
        cout << "-1" << endl;
    } else {
        string ans;
        int mov = k / n;
        int rem = k % n;
        char c;
        for (int i = 0; i < n; i++) {
            c = s[i] + mov;
            if (c > 'z')
                c -= 26;
            if (rem > 0) {
                if (i < rem) {
                    c--;
                }
                rem--;
            }
            ans += c;
        }
        cout << ans << endl;
    }
}
