import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long p = in.nextLong();
        if (p == 0) {
            long ans = 0;
            while (n > 0) {
                ans += n >> 1;
                n >>= 1;
            }
            System.out.println(ans);
            return;
        }
        long ans = 0;
        long cur = 1;
        while (n > 0) {
            if (n >= cur + p) {
                n -= cur + p;
                ans++;
                cur <<= 1;
            } else {
                cur = 1;
                ans++;
            }
        }
        System.out.println(ans);
    }
}
