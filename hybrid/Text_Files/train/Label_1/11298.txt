#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_map<int, int> cnt;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    cnt[x]++;
    if (x != 0) ans += cnt[x] - 1;
  }
  if (ans % 2 == 0) cout << ans / 2 << endl;
  else cout << -1 << endl;
  return 0;
}
