import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        try (Scanner sc = new Scanner(System.in)) {
            int n = sc.nextInt();
            int[] d = new int[n];
            for (int i = 0; i < n; i++) {
                d[i] = sc.nextInt();
            }
            int maxRight = d[n - 1];
            boolean canReturn = false;
            for (int i = n - 2; i >= 0; i--) {
                if (d[i] >= maxRight + 10) {
                    maxRight = d[i];
                    canReturn = true;
                }
            }
            if (maxRight >= d[0]) {
                System.out.println("yes");
            } else if (canReturn) {
                System.out.println("yes");
            } else {
                System.out.println("no");
            }
        }
    }
}
