#include <bits/stdc++.h>
using namespace std;
const int MM = 1e6 + 5;
const int INF = 1e9;
int n, m, s, b, k, h, v1[MM], v2[MM], dist[MM], pv[MM], prv[MM];
bool vis[MM];
vector<int> adj[MM];
vector<pair<int, int>> edges;

void dijkstra(int s, int arr[]) {
    memset(vis, 0, sizeof(vis));
    memset(dist, INF, sizeof(dist));
    dist[s] = 0;
    for (int i = 0; i < n; i++) {
        int mn = INF, idx = -1;
        for (int j = 1; j <= n; j++) {
            if (!vis[j] && dist[j] < mn) {
                mn = dist[j], idx = j;
            }
        }
        if (idx == -1) break;
        vis[idx] = 1;
        for (int u : adj[idx]) {
            if (dist[idx] + 1 < dist[u]) {
                dist[u] = dist[idx] + 1;
                pv[u] = idx;
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        arr[i] = dist[i];
    }
}

int mex(vector<int>& a) {
    map<int, int> cnt;
    for (int x : a) cnt[x] = 1;
    int ans = 0;
    while (cnt[ans]) ans++;
    return ans;
}

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    int u, v;
    for (int i = 0; i < m; i++) {
        cin >> u >> v;
        edges.push_back({u, v});
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    cin >> s >> b >> k >> h;
    for (int i = 0; i < s; i++) {
        cin >> v1[i] >> v2[i];
    }
    for (int i = 0; i < b; i++) {
        cin >> prv[i] >> pv[i];
    }
    dijkstra(1, v1);
    dijkstra(n, v2);
    int ans = INF;
    for (int i = 0; i < 20; i++) {
        vector<int> dist;
        for (int j = 0; j < s; j++) {
            if (v1[j] + v2[j] > prv[i]) dist.push_back(v1[j] + v2[j]);
        }
        int mn = mex(dist);
        int cost = 0;
        for (auto& e : edges) {
            u = e.first, v = e.second;
            if (v1[u - 1] + v2[v - 1] + 1 <= prv[i] && v1[v - 1] + v2[u - 1] + 1 <= prv[i]) continue;
            cost += mn;
        }
        cost += mn * h;
        cost += mn * k * b;
        ans = min(ans, cost);
    }
    cout << ans << '\n';
}
