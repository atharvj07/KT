#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int n, m;
    cin >> n >> m;
    vector<ll> a(n + 1), b(m + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= m; i++) cin >> b[i];
    sort(a.begin(), a.end()), sort(b.begin(), b.end());
    if (a[1] < 0)
        reverse(b.begin(), b.end());
    cout << a[1] * b[1] << endl;
    return 0;
}
