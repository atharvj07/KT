import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int na = sc.nextInt();
        int nb = sc.nextInt();
        int w = sc.nextInt();
        int[] wa = new int[na];
        int[] ca = new int[na];
        int[] wb = new int[nb];
        int[] cb = new int[nb];
        for (int i = 0; i < na; i++) {
            wa[i] = sc.nextInt();
            ca[i] = sc.nextInt();
        }
        for (int i = 0; i < nb; i++) {
            wb[i] = sc.nextInt();
            cb[i] = sc.nextInt();
        }
        int[][] dp = new int[na + 1][nb + 1];
        for (int i = 0; i < na + 1; i++) {
            for (int j = 0; j < nb + 1; j++) {
                dp[i][j] = 1000000000;
            }
        }
        dp[0][0] = 0;
        for (int i = 0; i < na; i++) {
            for (int j = 0; j < nb; j++) {
                dp[i + 1][j] = Math.min(dp[i + 1][j], dp[i][j] + ca[i]);
                dp[i][j + 1] = Math.min(dp[i][j + 1], dp[i][j] + cb[j]);
                int max = Math.max(ca[i], cb[j]);
                int min = Math.min(ca[i], cb[j]);
                dp[i + 1][j + 1] = Math.min(dp[i + 1][j + 1], dp[i][j] + max - min);
            }
        }
        int ans = 1000000000;
        for (int i = 0; i < na + 1; i++) {
            for (int j = 0; j < nb + 1; j++) {
                int ma = 0;
                for (int k = 0; k < i; k++) ma = Math.max(ma, wa[k]);
                int mb = 0;
                for (int k = 0; k < j; k++) mb = Math.max(mb, wb[k]);
                int max = Math.max(ma, mb);
                int min = Math.min(ma, mb);
                if (ma <= w && mb <= w) ans = Math.min(ans, dp[i][j] + max - min);
            }
        }
        System.out.println(ans);
    }
}
