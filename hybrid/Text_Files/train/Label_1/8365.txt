#include<bits/stdc++.h>
using namespace std;
int main(){
    int p,q,l,r,A,B,ans=0;
    cin>>p>>q>>l>>r;
    vector<pair<int,int>>v1,v2;
    for(int i=0;i<p;i++){
        cin>>A>>B;
        v1.push_back({A,B});
    }
    for(int i=0;i<q;i++){
        cin>>A>>B;
        v2.push_back({A,B});
    }
    for(int i=l;i<=r;i++){
        int c=0;
        for(int j=0;j<p;j++){
            if(i>=v1[j].first&&i<=v1[j].second){
                c=1;
                break;
            }
        }
        if(c){
            for(int j=0;j<q;j++){
                if(i+v2[j].first>=v2[j].second){
                    ans++;
                    break;
                }
            }
        }
    }
    cout<<ans<<endl;
}
