import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long a[] = new long[n];
        long b[] = new long[n];
        long sum1 = 0, sum2 = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
            sum1 += a[i];
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextLong();
            sum2 += b[i];
        }
        if (sum1 > sum2)
            System.out.println("NO");
        else
            System.out.println("YES");
    }
}
