import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = s.charAt(i) - 'a';
        }
        int[] c = new int[n];
        int[] cnt = new int[26];
        for (int i = 0; i < n; i++) {
            cnt[p[i]]++;
        }
        int m = 0;
        for (int i = 0; i < 26; i++) {
            m = Math.max(m, cnt[i]);
        }
        int[] ans = new int[n];
        int cur = 1;
        for (int i = 0; i < n; i++) {
            if (cnt[p[i]] > 1) {
                c[i] = cur++;
                cnt[p[i]]--;
            }
        }
        for (int i = 0; i < n; i++) {
            if (c[i] == 0) {
                c[i] = cur++;
            }
        }
        System.out.println(cur - 1);
        for (int i : c) {
            System.out.print(i + " ");
        }
        System.out.println();
    }
}
