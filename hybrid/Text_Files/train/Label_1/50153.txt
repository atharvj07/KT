import java.util.Scanner;

public class SmallestPositiveIntegerWithExactlyNDivisors {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        // Find the sqrt of n, as the number of divisors is at most sqrt(n)
        int sqrtN = (int) Math.sqrt(n);

        // Initialize the smallest positive integer with n divisors
        long smallestInteger = 1;

        // Iterate from 2 to sqrt(n) to find the divisors
        for (int i = 2; i <= sqrtN; i++) {
            if (n % i == 0) {
                // If i is a divisor, add i and n/i to the smallest integer
                smallestInteger *= i;

                // If i is not equal to n/i (i.e., i is not a perfect square), add n/i as well
                if (i != n / i) {
                    smallestInteger *= n / i;
                }
            }
        }

        // Print the smallest positive integer with n divisors
        System.out.println(smallestInteger);
    }
}
