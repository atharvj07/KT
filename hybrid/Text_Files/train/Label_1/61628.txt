n = int(input())
c = list(map(int, input().split()))

def count_regular_subsequences(c):
    total = 0
    for i in range(0, n, 2):
        open_brackets = 0
        for j in range(i, n, 2):
            open_brackets += c[j]
            close_brackets = 0
            for k in range(j + 1, n, 2):
                close_brackets += c[k]
                if close_brackets == open_brackets:
                    total += 1
                elif close_brackets > open_brackets:
                    break
    return total

print(count_regular_subsequences(c))
