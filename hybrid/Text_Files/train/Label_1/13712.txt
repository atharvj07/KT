#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n{};
  cin >> n;
  vector<int> v;
  v.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  if (n < 3) {
    for (auto& x : v) {
      cout << x << " ";
    }
    cout << endl;
    return 0;
  }
  int t = 0;
  int m;
  for (int i = 1; i < n; i++) {
    if ((i % 2 == 0 && v[i] < v[i - 1]) || (i % 2 == 1 && v[i] > v[i - 1])) {
      t++;
      m = i;
    }
  }
  if (t == 1) {
    for (int i = m - 1; i >= 0; i--) {
      cout<<v[i]<<" ";
    }
    for (int i = m; i < n; i++) {
      cout<<v[i]<<" ";
    }
    cout << endl;
    return 0;
  }
  cout << "Impossible" << endl;
  return 0;
}
