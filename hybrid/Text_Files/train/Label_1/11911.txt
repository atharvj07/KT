#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define pb push_back
#define ll long long
#define mod 1000000007
bool cmpx(const pair<pair<int, int>, int>&a, const pair<pair<int, int>, int>&b) { return a.first.first < b.first.first; }
bool cmpy(const pair<pair<int, int>, int>&a, const pair<pair<int, int>, int>&b) { return a.first.second < b.first.second; }
bool cmpz(const pair<pair<int, int>, int>&a, const pair<pair<int, int>, int>&b) { return a.second < b.second; }
int main() {
	int n;
	cin >> n;
	vector<pair<pair<int, int>, int>> p(n);
	for (int i = 0;i < n;i++) {
		int x, y, z;
		cin >> x >> y >> z;
		p[i] = {{x, y}, z};
	}
	int ans = n / 2;
	sort(p.begin(), p.end(), cmpx);
	for (int i = 0;i < n;i++) {
		if (i % 2 == 0) {
			for (int j = i + 1;j < n;j++) {
				if ((p[i].first.second != p[j].first.second) && (p[i].second != p[j].second)) {
					cout << i + 1 << " " << j + 1 << endl;
					break;
				}
			}
		}
	}
	sort(p.begin(), p.end(), cmpy);
	for (int i = 0;i < n;i++) {
		if (i % 2 == 0) {
			for (int j = i + 1;j < n;j++) {
				if ((p[i].first.first != p[j].first.first) && (p[i].second != p[j].second)) {
					cout << i + 1 << " " << j + 1 << endl;
					break;
				}
			}
		}
	}
	sort(p.begin(), p.end(), cmpz);
	for (int i = 0;i < n;i++) {
		if (i % 2 == 0) {
			for (int j = i + 1;j < n;j++) {
				if ((p[i].first.second != p[j].first.second) && (p[i].first.first != p[j].first.first)) {
					cout << i + 1 << " " << j + 1 << endl;
					break;
				}
			}
		}
	}

	return 0;
}
