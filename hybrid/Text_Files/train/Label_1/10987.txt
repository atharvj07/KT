#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int N = 1000001;
const int INF = 1e9;

struct Interval {
	int a, b;
	int id;
};
typedef long long ll;

vector<Interval> intervals;
vector<bool> used;

int32_t main() {
	ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
	int n; cin >> n;
	while (n--) {
		int t; cin >> t;
		if (t == 1) {
			int a, b; cin >> a >> b;
			if (a > b) swap(a, b);
			if (intervals.empty()) {
				intervals.push_back({a, b});
				used.push_back(true);
				continue;
			}
			vector<Interval> tmp;
			for (size_t i = 0; i < intervals.size(); ++i) {
				if (used[i] == false) {
					tmp.push_back(intervals[i]);
					continue;
				}
				if (intervals[i].b < a || b < intervals[i].a) {
					tmp.push_back(intervals[i]);
					continue;
				}
				a = min(a, intervals[i].a);
				b = max(b, intervals[i].b);
			}
			tmp.push_back({a, b});
			intervals = tmp;
			used.assign(tmp.size(), true);
		}
		else {
			int i, j; cin >> i >> j;
			--i, --j;
			cout << (i == j ? "YES\n" : "NO\n");
		}
	}
}
