#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n + 2, 1);
  int ans = k * 2 - 2 + n - 1;
  for (int i = 1; i <= n; ++i) {
    int j = k + i - 1;
    if (j <= n) {
      a[j]++;
      if (a[j] == 1) ans--;
      if (a[j] == 2) ans++;
    }
  }
  for (int i = 1; i <= n; ++i) {
    int j = k - i + 1;
    if (j >= 1) {
      a[j]++;
      if (a[j] == 1) ans--;
      if (a[j] == 2) ans++;
    }
  }
  cout << ans << '\n';
  return 0;
}
