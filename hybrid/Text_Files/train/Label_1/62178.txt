from sys import stdin
def calsum(n,p,a,x,y):
    rem=[]
    ad=[]
    for i in range(0,p-1):
        if(a[i]==1):
            rem.append(i)
    for i in range(p-1,len(a)):
        if(a[i]==0 and (i-p+1)%k==0):
            ad.append(i)
    rem.sort()
    ad.sort()
    ans=0
    while(len(rem)!=0 and len(ad)!=0):
        if(rem[0]<ad[0]):
            ans+=y
            rem.pop(0)
        else:
            ans+=x
            ad.pop(0)
    while(len(rem)!=0):
        ans+=y
        rem.pop(0)
    while(len(ad)!=0):
        ans+=x
        ad.pop(0)
    return ans
for i in range(int(stdin.readline())):
    n,p,k=map(int,stdin.readline().split())
    a=list(map(int,stdin.readline()))
    x,y=map(int,stdin.readline().split())
    print(calsum(n,p,a,x,y))
