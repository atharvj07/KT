#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
    long long int n, a_i, moves = 0;
    vector<bool> is_square;
    cin >> n;
    for(long long int i = 0; i < n; i++){
        cin >> a_i;
        is_square.push_back(sqrt(a_i) == int(sqrt(a_i)));
    }
    long long int even = 0, odd = 0;
    for(bool i : is_square){
        if(i) even++;
        else odd++;
    }
    if(even == n / 2){
        moves = 0;
    }else{
        moves = min(even, odd);
    }
    cout << moves << endl;
    return 0;
}
