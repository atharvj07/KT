#include<bits/stdc++.h>

using namespace std;
using ll = long long;
const int mod = 998244353;

int main() {
    int n; cin >> n;
    vector<ll> a(n);
    for (auto &x : a) cin >> x;
    sort(a.begin(), a.end());
    vector<int> dp(n); // number of valid renumerations ending at this index
    unordered_map<ll, int> cnt; // number of occurences of each element before this index
    dp[0] = 1;
    cnt[a[0]] = 1;
    for (int i = 1; i < n; i++) {
        if (a[i] == a[i - 1]) { dp[i] = dp[i - 1]; cnt[a[i]]++; }
        else {
            dp[i] = 1ll * dp[i - 1] * (cnt[a[i]] + 1) % mod;
            cnt[a[i]] = 1;
        }
    }
    cout << dp[n - 1] << endl;
}
