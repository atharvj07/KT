#include <iostream>
#include <queue>
#include <vector>
using namespace std;
typedef long long ll;

const int maxN = 5e5 + 5;

vector<pair<ll, ll>> videos;
ll servers[maxN];
ll elapsed = 0;
priority_queue<ll, vector<ll>, greater<ll>> freeServers;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  ll n, k;
  cin >> n >> k;
  for (int i = 0; i < k; i++) freeServers.push(i);
  for (ll i = 0; i < n; i++) {
    ll s, m;
    cin >> s >> m;
    videos.push_back({s, m});
  }
  for (auto video : videos) {
    while (!freeServers.empty() && freeServers.top() < video.first) {
      freeServers.pop();
      elapsed++;
    }

    if (freeServers.empty()) {
      elapsed = video.first;
    }
    servers[freeServers.top()] = elapsed + video.second;
    freeServers.pop();
  }
  for (int i = 0; i < n; i++) cout << servers[i] << '\n';
  return 0;
}
