#include<bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    int min_val = INT_MAX, max_val = INT_MIN;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      min_val = min(min_val, x);
      max_val = max(max_val, x);
    }
    if (max_val - min_val > 2 * k)
      cout << -1 << "\n";
    else
      cout << (min_val + max_val) / 2 << "\n";
  }
}
