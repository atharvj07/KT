#include <iostream>
#include <vector>
#include <utility>
using namespace std;

long long solve(const vector<int> &a, const string &s) {
    int n = s.size();
    vector<pair<int, int>> pref(n);
    pref[0] = {a[s[0] - 'a'], 1};
    for (int i = 1; i < n; i++) {
        pref[i] = {pref[i - 1].first + a[s[i] - 'a'],
                       pref[i - 1].second + (s[i] == s[i - 1])};
    }
    long long ans = 0;
    for (int i = 1; i < n - 1; i++) {
        if (s[i] == s[i - 1] && pref[i - 1].first == pref[i].first) {
            ans += pref[n - 1].second - pref[i].second;
        }
    }
    return ans;
}
int main() {
    vector<int> a(26);
    for (int &x : a) cin >> x;
    string s;
    cin >> s;
    cout << solve(a, s) << endl;
    return 0;
}
