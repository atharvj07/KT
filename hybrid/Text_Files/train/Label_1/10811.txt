#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const int MAXN = 2e5 + 5;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, m;
    cin >> n >> k >> m;
    vector<int> a(n), b(m);
    for (int i = 0; i < n; ++i) a[i] = i + 1;
    for (int i = 0; i < m; ++i) cin >> b[i];
    int l = 0, r = 0;
    for (int i = 0; i < m; ++i) {
      while (r + 1 < n && a[r + 1] < b[i]) ++r;
      if (r + 1 < n && a[r + 1] == b[i]) ++l;
      else {
        cout << "NO\n";
        goto nxt;
      }
    }
    if ((r - l + 1) % k != 0) {
      cout << "NO\n";
      goto nxt;
    }
    cout << "YES\n";
  nxt:;
  }
  return 0;
}
