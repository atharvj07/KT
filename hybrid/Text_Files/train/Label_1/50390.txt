import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int c = sc.nextInt();
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            ans = (ans + powerMod(i + c - 1, i, 1000000003)) % 1000000003;
        }
        System.out.println(ans);
    }

    static long powerMod(long a, long b, long m) {
        if (b == 0) {
            return 1;
        }
        if (b == 1) {
            return a;
        }
        long half = powerMod(a, b / 2, m);
        long half2 = half * half;
        half2 %= m;
        return b % 2 == 1 ? (half2 * a) % m : half2;
    }
}
