import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        String s = sc.next();
        int[] freq = new int[m];
        for (char c : s.toCharArray()) {
            freq[c - 'a']++;
        }
        Arrays.sort(freq);
        int slowness = 0;
        for (int i = 1; i < m; i++) {
            slowness += freq[i];
        }
        System.out.println(slowness);
    }
}
