#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> a(n, vector<int>(m));
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) cin >> a[i][j];
  int ans = 0;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      if (a[i][j] == 1) ans++;
      while (a[i][j] % 2 == 0) a[i][j] /= 2, ans++;
      while (a[i][j] % 3 == 0) a[i][j] /= 3, ans++;
      if (a[i][j] > 1) ans++;
    }
  cout << ans << endl;
  return 0;
}
