n, m = map(int, input().split())
a = list(map(int, input().split()))

distinct_counts = [0] * n
distinct_elements = set()
for i in range(n-1, -1, -1):
    distinct_elements.add(a[i])
    distinct_counts[i] = len(distinct_elements)

for _ in range(m):
    l = int(input())
    print(distinct_counts[l-1])
