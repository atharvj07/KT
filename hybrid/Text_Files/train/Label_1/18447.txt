#include <iostream>
#include <vector>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n, q;
        std::cin >> n >> q;
        std::vector<int> a(n);
        for (int i = 0; i < n; i++) {
            char c;
            std::cin >> c;
            a[i] = c == '+' ? 1 : -1;
        }
        std::vector<int> sum(n);
        sum[0] = a[0];
        for (int i = 1; i < n; i++) {
            sum[i] = sum[i - 1] + a[i];
        }
        while (q--) {
            int l, r;
            std::cin >> l >> r;
            l--;
            int ans = 0;
            if (l == 0) {
                ans = sum[r - 1] == 0 ? 0 : 1;
            } else {
                if (sum[l - 1] == sum[r - 1]) {
                    ans = 0;
                } else if (sum[l - 1] == sum[r - 1] + a[l - 1]) {
                    ans = 1;
                } else {
                    ans = 2;
                }
            }
            std::cout << ans << std::endl;
        }
    }
}
