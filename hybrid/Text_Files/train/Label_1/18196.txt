#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;

struct ufds {
  vector<int> p, r, s;
  int count;
  ufds(int n): p(n), r(n), s(n), count(n) {
    for (int i = 0; i < n; i++) p[i] = i;
  }
  int find(int x) {
    if (x != p[x]) p[x] = find(p[x]);
    return p[x];
  }
  int size(int x) {
    return s[find(x)];
  }
  void unite(int a, int b, int sum) {
    a = find(a), b = find(b);
    if (a != b) {
      if (r[a] < r[b]) {
        p[a] = b;
        s[b] += s[a];
        s[a] = sum;
      }
      else {
        p[b] = a;
        s[a] += s[b];
        s[b] = sum;
        if (r[a] == r[b]) r[a]++;
      }
      count--;
    }
  }
};

int main() {
  ios_base::sync_with_stdio(false); cin.tie(nullptr); cout.tie(nullptr);
  int t; cin >> t;
  while (t --> 0) {
    int n, m; cin >> n >> m;
    vector<ll> v(n), t(n);
    for (ll &x: v) cin >> x;
    for (ll &x: t) cin >> x;
    ufds d(n);
    while (m --> 0) {
      int a, b; cin >> a >> b; a--, b--;
      d.unite(a, b, v[a] + v[b]);
    }
    bool ok = true;
    for (int i = 0; i < n; i++) {
      int root = d.find(i);
      if (v[i] + d.s[root] != t[i]) ok = false;
    }
    cout << (ok ? "YES\n" : "NO\n");
  }
}
