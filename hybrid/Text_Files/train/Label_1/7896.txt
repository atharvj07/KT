#include <bits/stdc++.h>

using namespace std;

#define vi(x) vector<x>
#define pb push_back
#define all(x) x.begin(), x.end()

using ll = long long;
constexpr ll inf = 1e18;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);

  int n;
  cin >> n;

  vi(ll) v(n), t(n);
  for (int i{}; i < n; cin >> v[i++])
    ;
  for (int i{}; i < n; cin >> t[i++])
    ;

  vi(ll) ans(n);

  stack<tuple<ll, ll>> st;  // (t_index, v)

  for (int day{1}, td{0}; day <= n; day++, td++) {
    while (!st.empty()) {
      if (get<0>(st.top()) <= t[td]) {
        ans[td] += get<1>(st.top());
        st.pop();
      } else
        break;
    }
    st.push({t[td], v[day - 1]});
    if (!st.empty() && get<1>(st.top()) <= t[td]) {
      ans[td] += get<1>(st.top());
      st.pop();
    }
  }

  for (auto x : ans) cout << x << ' ';
  cout << '\n';

  return 0;
}
