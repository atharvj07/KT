import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int d = 0;
        int positiveCount = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] > 0) {
                positiveCount++;
            }
        }

        if (positiveCount >= (n + 1) / 2) {
            d = 1;
        } else {
            for (int i = -1000; i <= 1000; i++) {
                positiveCount = 0;
                for (int j = 0; j < n; j++) {
                    if ((a[j] / i) > 0) {
                        positiveCount++;
                    }
                }
                if (positiveCount >= (n + 1) / 2) {
                    d = i;
                    break;
                }
            }
        }

        System.out.println(d);
    }
}
