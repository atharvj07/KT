#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;
const ll MOD = 1e9 + 7;

int main() {
  ll n, q;
  cin >> n >> q;
  vector<char> s(n);
  for (ll i = 0; i < n; i++) { cin >> s[i]; }
  for (ll i = 0; i < q; i++) {
    ll pos;
    char c;
    cin >> pos >> c;
    s[pos - 1] = c;
    ll ans = 0;
    for (ll j = 0; j < n - 2; j++) {
      if (s[j] == 'a' && s[j + 1] == 'b' && s[j + 2] == 'c') {
        ans++;
        j += 2;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
