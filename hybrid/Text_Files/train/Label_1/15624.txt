#include <iostream>
#include <vector>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n + 1);
  for (int i = 0; i <= n; i++) {
    if (!std::cin.get()) return 0;
    char c;
    std::cin >> c;
    if (c == '?') {
      a[i] = -1;
    } else {
      std::cin >> a[i];
    }
  }
  if (k == 0) {
    if (a[0] == 0) {
      std::cout << "Yes";
    } else {
      std::cout << "No";
    }
    return 0;
  }
  if (a[n] == 0) {
    std::cout << "Yes";
    return 0;
  }
  if (n == 1 && a[1] == 0) {
    std::cout << "Yes";
    return 0;
  }
  if (a[n - 1] == 0 && a[n] == -1) {
    std::cout << "Yes";
    return 0;
  }
  std::cout << "No";
  return 0;
}
