t = int(input())
for _ in range(t):
    n, m, k = map(int, input().split())
    a = list(map(int, input().split()))
    k = min(k, m - 1)
    ans = 0
    for i in range(k + 1):
        min_val = float('inf')
        for j in range(m - k):
            min_val = min(min_val, max(a[i + j], a[i + j + n - m]))
        ans = max(ans, min_val)
    print(ans)
