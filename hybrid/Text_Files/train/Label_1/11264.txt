#include <bits/stdc++.h>
 
using namespace std;
#define ll long long
 
const int NM = 5005;
 
int a[NM], b[NM], c[NM], n, m, k;
vector<vector<int>> g(NM);
vector<bool> vis(NM);
int dp[NM][NM];
 
int dfs(int node, int army) {
    if (army < 0) return -1e9;
    if (node == n + 1) return 0;
    if (dp[node][army] != -1) return dp[node][army];
    int ans = 0;
    for (auto u : g[node]) {
        if (u <= node) continue;
        ans = max(ans, dfs(u, army) + c[u]);
    }
    ans = max(ans, dfs(node + 1, army - a[node] + b[node]) + c[node]); 
    if (army >= a[node]) ans = max(ans, dfs(node + 1, army - a[node]) + c[node]);
    return dp[node][army] = ans;
}
 
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    memset(dp, -1, sizeof dp);
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i] >> c[i];
    }
    for (int i = 1; i <= m; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
    }
    int ans = dfs(1, k);
    if (ans < 0) cout << -1 << endl;
    else cout << ans << endl;
    return 0;
}
