import sys
n = int(input())
events = []
for _ in range(2*n):
    events.append(input())
exist = set()
show = []
for e in events:
    if e[0] == '+':
        show.append(int(e[1:]))
        exist.add(int(e[1:]))
    else:
        price = int(e[1:])
        if price not in exist:
            print('NO')
            sys.exit()
        exist.remove(price)
        while show and show[-1] > price:
            print('NO')
            sys.exit()
        show.pop()
print('YES')
print(*reversed(show))
