#include <iostream>
#include <map>
#include <vector>
using namespace std;

int a[150001], cnt[1000001];

int main() {
  int n;
  cin >> n;
  map<int, vector<int>> b;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    b[a[i]].push_back(i);
    cnt[a[i]]++;
  }
  vector<int> ans;
  for (int i = 0; i < n; i++) {
    if (cnt[a[i]] < 2) {
      ans.push_back(a[i]);
    } else {
      cnt[a[i]]--;
      b[a[i]].erase(b[a[i]].begin());
      ans.push_back(a[i] * 2);
      i = b[a[i]][0];
    }
  }
  cout << ans.size() << '\n';
  for (int x : ans) cout << x << ' ';
  return 0;
}
