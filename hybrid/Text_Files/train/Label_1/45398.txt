import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int R0 = sc.nextInt();
            int W0 = sc.nextInt();
            int C = sc.nextInt();
            int R = sc.nextInt();
            if (R0 == 0 && W0 == 0 && C == 0 && R == 0) {
                break;
            }
            int X = (int) Math.max(0, (C * W0 - R0) / (double) R);
            System.out.println(X);
        }
        sc.close();
    }
}
