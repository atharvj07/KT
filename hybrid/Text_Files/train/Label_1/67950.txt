n=int(input())
a=list(map(int,input().split()))
for i in range(1,n):
    x=1
    y=0
    flag=0
    for j in range(i,n):
        if x<=0 or x>n:
            flag=1
            break
        else:
            x+=a[x-1]
            y+=a[x-1]
            x-=a[x-1]
            y+=a[x-1]
    if not flag:
        print(-1)
    else:
        print(y)
