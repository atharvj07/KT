import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int q = 0; q < t; q++) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }
            int l = -1;
            int r = -1;
            boolean found = false;
            for (int i = 0; i < n; i++) {
                for (int j = i; j < n; j++) {
                    int max = a[i];
                    int min = a[i];
                    for (int k = i + 1; k <= j; k++) {
                        max = Math.max(max, a[k]);
                        min = Math.min(min, a[k]);
                    }
                    if (max - min >= j - i + 1) {
                        l = i + 1;
                        r = j + 1;
                        found = true;
                        break;
                    }
                }
                if (found) {
                    break;
                }
            }
            if (found) {
                System.out.println("YES");
                System.out.println(l + " " + r);
            } else {
                System.out.println("NO");
            }
        }
    }
}
