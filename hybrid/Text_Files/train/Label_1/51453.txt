import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        int n = Integer.parseInt(tokenizer.nextToken());
        int k = Integer.parseInt(tokenizer.nextToken());
        List<Integer> b = new ArrayList<>();
        tokenizer = new StringTokenizer(reader.readLine());
        for (int i = 0; i < n; i++) {
            b.add(Integer.parseInt(tokenizer.nextToken()));
        }
        int q = Integer.parseInt(reader.readLine());
        SegmentTree tree = new SegmentTree(n * k, b);
        for (int i = 0; i < q; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            if (tokenizer.nextToken().equals("2")) {
                int l = Integer.parseInt(tokenizer.nextToken()) - 1;
                int r = Integer.parseInt(tokenizer.nextToken()) - 1;
                System.out.println(tree.query(l, r));
            } else {
                int l = Integer.parseInt(tokenizer.nextToken()) - 1;
                int r = Integer.parseInt(tokenizer.nextToken()) - 1;
                int x = Integer.parseInt(tokenizer.nextToken());
                tree.update(l, r, x);
            }
        }
    }

    private static class SegmentTree {

        private int[] tree;
        private int size;

        SegmentTree(int size, List<Integer> b) {
            this.size = size;
            tree = new int[size * 4];
            buildTree(1, 0, size - 1, b);
        }

        private void buildTree(int node, int start, int end, List<Integer> b) {
            if (start == end) {
                tree[node] = b.get(start % b.size());
            } else {
                int mid = (start + end) / 2;
                buildTree(2 * node, start, mid, b);
                buildTree(2 * node + 1, mid + 1, end, b);
                tree[node] = Math.min(tree[2 * node], tree[2 * node + 1]);
            }
        }

        public void update(int left, int right, int value) {
            update(1, 0, size - 1, left, right, value);
        }

        private void update(int node, int start, int end, int left, int right, int value) {
            if (left > right) {
                return;
            }
            if (start == left && end == right) {
                tree[node] = value;
                return;
            }
            int mid = (start + end) / 2;
            update(2 * node, start, mid, left, Math.min(right, mid), value);
            update(2 * node + 1, mid + 1, end, Math.max(left, mid + 1), right, value);
            tree[node] = Math.min(tree[2 * node], tree[2 * node + 1]);
        }

        public int query(int left, int right) {
            return query(1, 0, size - 1, left, right);
        }

        private int query(int node, int start, int end, int left, int right) {
            if (left > right) {
                return Integer.MAX_VALUE;
            }
            if (start == left && end == right) {
                return tree[node];
            }
            int mid = (start + end) / 2;
            int leftValue = query(2 * node, start, mid, left, Math.min(right, mid));
            int rightValue = query(2 * node + 1, mid + 1, end, Math.max(left, mid + 1), right);
            return Math.min(leftValue, rightValue);
        }
    }
}
