import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int[] dp = new int[n];
        int[] pre = new int[n];
        dp[0] = 1;
        pre[0] = -1;
        int max = 1;
        int maxInd = 0;
        for (int i = 1; i < n; i++) {
            dp[i] = 1;
            pre[i] = -1;
            for (int j = 0; j < i; j++) {
                if (arr[j] < arr[i] && dp[i] < dp[j] + 1) {
                    dp[i] = dp[j] + 1;
                    pre[i] = j;
                }
            }
            if (dp[i] > max) {
                max = dp[i];
                maxInd = i;
            }
        }
        StringBuilder sb = new StringBuilder();
        while (maxInd != -1) {
            if (maxInd == pre[maxInd]) {
                sb.append('L');
            } else {
                sb.append('R');
            }
            maxInd = pre[maxInd];
        }
        sb.reverse();
        System.out.println(max);
        System.out.println(sb.toString());
    }
}
