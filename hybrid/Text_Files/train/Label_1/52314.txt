import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[] p1 = new int[n + 1];
        int[] p2 = new int[n + 1];
        int[] b1 = new int[n + 1];
        int[] b2 = new int[n + 1];
        int t1 = 0;
        int t2 = 0;
        for (int i = 0; i < m; i++) {
            String action = sc.next();
            int team = sc.nextInt();
            if (action.equals("p")) {
                if (team == 1) {
                    p1[t1++] = s[i];
                } else {
                    p2[t2++] = s[i];
                }
            } else {
                if (team == 1) {
                    b1[t1++] = s[i];
                } else {
                    b2[t2++] = s[i];
                }
            }
        }
        Arrays.sort(p1);
        Arrays.sort(p2);
        Arrays.sort(b1);
        Arrays.sort(b2);
        int sum1 = 0;
        int sum2 = 0;
        for (int i = 0; i < t1; i++) {
            sum1 += p1[i];
        }
        for (int i = 0; i < t2; i++) {
            sum2 += p2[i];
        }
        for (int i = 0; i < t1; i++) {
            if (i < t2) {
                sum1 += b2[i];
            }
        }
        for (int i = 0; i < t2; i++) {
            if (i < t1) {
                sum2 += b1[i];
            }
        }
        System.out.println(sum1 - sum2);
    }
}
