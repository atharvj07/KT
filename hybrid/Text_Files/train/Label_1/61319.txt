w, l = map(int, input().split())
a = list(map(int, input().split()))

# Create a sliding window of size l
min_stones = [0] * (w-1)
min_stones[0] = a[0]

for i in range(1, w-1):
    min_stones[i] = min_stones[i-1] + a[i]
    if i >= l:
        min_stones[i] -= a[i-l]

max_frogs = min_stones[l-1]
for i in range(l, w-1):
    max_frogs = min(max_frogs, min_stones[i])

print(max_frogs)
