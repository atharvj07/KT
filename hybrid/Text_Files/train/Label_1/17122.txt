#include <bits/stdc++.h>
using namespace std;
string q,ans[27];
int main() {
  int k;
  cin >> k >> q;
  if (k > q.size()) {
    cout << "NO";
    return 0;
  }
  cout << "YES"
       << "\n";
  int idx = 0;
  for (int i = 0; i < q.size(); i++) {
    if (ans[q[i] - 'a'].empty()) {
      if (idx == k) {
        cout << "NO";
        return 0;
      }
      ans[q[i] - 'a'] = q.substr(i);
      idx++;
    }
  }
  for (int i = 0; i < k; i++) cout << ans[i] << "\n";
  return 0;
}
