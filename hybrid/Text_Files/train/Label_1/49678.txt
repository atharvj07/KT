// package whateveryouwant;

import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            String p = scanner.next();
            String h = scanner.next();
            String p_ = "";
            boolean found = false;
            for (int j = 0; j < h.length(); j++) {
                if (p.indexOf(h.charAt(j)) < 0) {
                    p_ += h.charAt(j);
                } else {
                    if (!found) {
                        found = true;
                        p_ = h.charAt(j) + p_;
                    } else {
                        p_ += h.charAt(j);
                    }
                }
            }
            if (p.length() != p_.length()) {
                System.out.println("NO");
            } else if (p.equals(p_)) {
                System.out.println("YES");
            } else {
                int[] count = new int[26];
                for (int j = 0; j < p.length(); j++) {
                    count[p.charAt(j) - 'a']++;
                    count[p_.charAt(j) - 'a']--;
                }
                boolean valid = true;
                for (int j = 0; j < 26; j++) {
                    if (count[j] != 0) {
                        valid = false;
                        break;
                    }
                }
                if (valid) {
                    System.out.println("YES");
                } else {
                    System.out.println("NO");
                }
            }
        }
    }
}
