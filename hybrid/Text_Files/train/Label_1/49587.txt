import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        ArrayList<Interval> intervals = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int t1 = Integer.parseInt(st.nextToken());
            int x1 = Integer.parseInt(st.nextToken());
            int y1 = Integer.parseInt(st.nextToken());
            int t2 = Integer.parseInt(st.nextToken());
            int x2 = Integer.parseInt(st.nextToken());
            int y2 = Integer.parseInt(st.nextToken());
            intervals.add(new Interval(t1, t2, x1, y1, x2, y2));
        }
        intervals.sort(Comparator.comparing(Interval::getT1));
        int ans = 1;
        int cnt = 1;
        for (int i = 1; i < n; i++) {
            Interval prev = intervals.get(i - 1);
            Interval curr = intervals.get(i);
            if (prev.intersects(curr)) {
                cnt++;
            } else {
                cnt = 1;
            }
            ans = Math.max(ans, cnt);
        }
        System.out.println(ans);
    }

    static class Interval {
        int t1, t2, x1, y1, x2, y2;

        Interval(int t1, int t2, int x1, int y1, int x2, int y2) {
            this.t1 = t1;
            this.t2 = t2;
            this.x1 = x1;
            this.y1 = y1;
            this.x2 = x2;
            this.y2 = y2;
        }

        boolean intersects(Interval other) {
            if (this.t2 < other.t1 || other.t2 < this.t1) {
                return false;
            }
            if (this.x1 == this.x2 && this.y1 == this.y2) {
                return other.x1 == this.x1 && other.y1 == this.y1;
            }
            if (other.x1 == other.x2 && other.y1 == other.y2) {
                return this.x1 == other.x1 && this.y1 == other.y1;
            }
            double slope1 = (double) (this.y2 - this.y1) / (double) (this.x2 - this.x1);
            double intercept1 = this.y1 - slope1 * this.x1;
            double slope2 = (double) (other.y2 - other.y1) / (double) (other.x2 - other.x1);
            double intercept2 = other.y1 - slope2 * other.x1;
            if (slope1 == slope2 && intercept1 == intercept2) {
                return true;
            }
            double x = (intercept2 - intercept1) / (slope1 - slope2);
            double y = slope1 * x + intercept1;
            return x >= Math.min(this.x1, this.x2) && x <= Math.max(this.x1, this.x2) && x >= Math.min(other.x1, other.x2) && x <= Math.max(other.x1, other.x2) && y >= Math.min(this.y1, this.y2) && y <= Math.max(this.y1, this.y2) && y >= Math.min(other.y1, other.y2) && y <= Math.max(other.y1, other.y2);
        }

        int getT1() {
            return this.t1;
        }
    }
}
