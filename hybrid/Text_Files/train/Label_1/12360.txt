#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int MAX = (1001);
ll dp[MAX][MAX][2];
ll mod = 1000000007;
ll f(ll n, ll k, ll d, bool  atleast_once){
	if(n==0 && atleast_once) return 1;
	if(n<0 || k<0) return 0;
	
	ll &res = dp[n][k][atleast_once];
	if(res != -1) return res;
	res = 0;
	ll op1,op2;
	if(atleast_once==1){
		op1 = f(n-1,k-1,d,true);
		op2 = f(n-d,k-1,d,true);
	}
	else{
		op1 = f(n-1,k-1,d,false);
		op2 = f(n-d,k-1,d,true);
	}
	res = (op1%mod + op2%mod)%mod;
	return res;
}
int main(){
	ll n,k,d; cin>>n>>k>>d;
	memset(dp,-1,sizeof dp);
	cout<<f(n,k,d,false)%mod;
	return 0;
}
