#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  if (a == b) {
    cout << -1 << endl;
    return 0;
  }
  vector<int> p(n + 1);
  for (int i = 1; i <= n; ++i) {
    p[i] = i;
  }
  int cnt_a = 0, cnt_b = 0;
  for (int i = 1; i <= n; ++i) {
    if (cnt_a == cnt_b) {
      if (a != b) {
        swap(p[i], p[p[i]]);
        ++cnt_a;
      }
    } else {
      swap(p[i], p[p[i]]);
      ++cnt_b;
    }
  }
  for (int i = 1; i <= n; ++i) {
    cout << p[i] << " ";
  }
  cout << endl;
  return 0;
}
