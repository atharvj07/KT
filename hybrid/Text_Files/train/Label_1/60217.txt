def solve(n):
    for i in range(2, int(n**0.5)+1):
        if n % i == 0:
            factors = []
            while n % i == 0:
                factors.append(i)
                n //= i
            if n > 1:
                factors.append(n)
            return len(factors), factors
    return 1, [n]

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    k, sequence = solve(n)
    print(k)
    print(*sequence)
