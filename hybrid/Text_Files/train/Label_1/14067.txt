#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(2 * n);
    unordered_map<int, int> mp;
    for (int &x : a) {
        mp[x] ^= 1;
        cin >> x;
    }
    if (mp.size() == 1) {
        cout << -1;
    } else {
        auto it = mp.begin();
        cout << it->first;
        while (it != mp.end()) {
            it++;
            cout << ' ' << it->first;
        }
    }
}
