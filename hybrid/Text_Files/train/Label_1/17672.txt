#include <iostream>
#include <string>
#include <unordered_map>
using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_map<string, int> cheeses;
  for (int i = 0; i < n; i++) {
    string cheese, type;
    cin >> cheese >> type;
    if (type == "soft") cheeses[cheese]++;
  }
  cout << cheeses.size() << endl;
  return 0;
}
