import java.util.Scanner;

public class Main {
    private static final long MOD = 1000000007L;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long d = sc.nextLong();
        long ans = n * (n - 1) / 2;
        ans = ans * (d - 1) % MOD;
        System.out.println(ans);
        sc.close();
    }
}
