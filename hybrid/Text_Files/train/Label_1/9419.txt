#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int x1, y1, z1, x2, y2, z2;
    cin >> x1 >> y1 >> z1 >> x2 >> y2 >> z2;
    long long ans = 0;
    if (x1 < x2) ans += 2 * min(y1, z2);
    if (y1 < y2) ans += 2 * min(x1, z2);
    if (z1 < z2) ans += 2 * min(x1, y2);
    cout << ans << endl;
  }
  return 0;
}
