import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), I = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        sc.close();

        Arrays.sort(arr);

        long ans = Long.MAX_VALUE;
        for (int l = 0; l < n; l++) {
            for (int r = n - 1; r >= 0; r--) {
                if (l == r) {
                    continue;
                }
                long change = 0;
                for (int i = 0; i < n; i++) {
                    if (arr[i] < arr[l]) {
                        change += arr[l] - arr[i];
                    } else if (arr[i] > arr[r]) {
                        change += arr[i] - arr[r];
                    }
                }
                if (change <= (long) I * 8) {
                    ans = Math.min(ans, change);
                    break;
                }
            }
        }

        System.out.println(ans);
    }
}
