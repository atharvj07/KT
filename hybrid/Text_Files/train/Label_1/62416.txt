from sys import stdin

readline = stdin.readline

for i in range(int(readline())):
  r, c = map(int, readline().split())
  x = [['A' * c] for i in range(r)]
  for j in range(r):
    x[j] = list(readline()[:-1])
  if 'P' not in x[0] and 'P' not in [i[-1] for i in x]:
    print(0)
    continue
  for j in range(c):
    if 'P' in [i[j] for i in x]:
      print(1)
      break
  else:
    print('MORTAL')
