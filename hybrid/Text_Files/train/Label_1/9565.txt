#include <iostream>
#include <vector>

using namespace std;

const int MOD = 1e9 + 7;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  vector<int> cnt(n);
  for (int i = 0; i < n; ++i) {
    cnt[i] = s[i] == s[(i + 1) % n];
  }

  int ans = 1;
  for (int i = 0; i < n; ++i) {
    ans = 1LL * ans * (cnt[i] + 1) % MOD;
  }

  cout << ans << endl;
  return 0;
}
