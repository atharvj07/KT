#include <bits/stdc++.h>

using namespace std;

const int MAXN = 55;

int Grid[MAXN][MAXN];

int main() {
    int N;
    cin >> N;
    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++) {
            cin >> Grid[r][c];
        }
    }
    bool ok = true;
    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++) {
            if (Grid[r][c] == 1) {
                continue;
            }
            int sum = 0;
            for (int c2 = 0; c2 < N; c2++) {
                sum += (Grid[r][c2] != 1);
            }
            for (int r2 = 0; r2 < N; r2++) {
                sum += (Grid[r2][c] != 1);
            }
            if (sum < Grid[r][c]) {
                ok = false;
            }
        }
    }
    cout << (ok ? "Yes" : "No") << endl;
    return 0;
}
