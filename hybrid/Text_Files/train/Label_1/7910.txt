#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
using ll = long long;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> v(n);
        for (int i = 0; i < n; i++) cin >> v[i];
        ll swaps = 0;
        int l = 1, r = 1;
        for (int i = 0; i < n; i++) {
            if (v[i] != i + 1) {
                if (v[i] < i + 1) r = i + 1;
                l = i + 1;
                while (r < n && v[r] == r + 1) r++;
                reverse(v.begin() + l - 1, v.begin() + r);
                swaps += r - l + 1;
                l = r + 1;
            }
        }
        cout << swaps << endl;
    }
    return 0;
}
