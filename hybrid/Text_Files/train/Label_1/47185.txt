import java.util.Scanner;

public class Main {

    static long[] solve(long[] a, long x) {
        long min = 0, max = 0;
        for (long _a : a) {
            min += (_a + x - 1) / x;
            max += _a / x;
        }
        return new long[]{min, max};
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long x = sc.nextInt();
            long[] a = new long[n];
            for (int i = 0; i < n; i++) a[i] = sc.nextLong();
            long[] ans = solve(a, x);
            System.out.println(ans[0] + " " + ans[1]);
        }
        sc.close();
    }
}
