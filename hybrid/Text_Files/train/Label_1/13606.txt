#include <bits/stdc++.h>
using namespace std;

int main() {
    string s;
    cin >> s;
    int cnt[26] = {};
    for (char &c : s) { cnt[c - 'a']++; }
    bool ans = false;
    for (int i = 0; i < 26; i++) {
        for (int j = 0; j < 26; j++) {
            if (i == j) continue;
            if (cnt[i] > 0 && cnt[j] > 0) { ans = true; }
        }
    }
    cout << (ans ? "Yes" : "No") << endl;
}
