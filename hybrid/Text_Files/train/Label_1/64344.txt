import sys
input = sys.stdin.readline
mod = 10**9+7
n, m = map(int, input().split())
g = [[] for i in range(n+1)]
for i in range(m):
    a, b, c = map(int, input().split())
    g[a].append([b,c])
    g[b].append([a,c])

def dfs(node, p=-1, l=True):
    sm = 1
    for u, c in g[node]:
        if u == p: continue
        if l and c: dfs(u, node, False)
        elif l and not c: sm = 0
        elif not l and c: sm *= dfs(u, node, True)
        elif not l and not c: sm *= dfs(u, node, False)
    return sm % mod

print(dfs(1))
