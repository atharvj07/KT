#include <bits/stdc++.h>

using namespace std;

const int N = 500;

int n, k;
long long a[N], b[N];
long long red, blue;

int main() {
    cin >> n >> k;
    for (int i = 0; i < n; ++i) {
        cin >> a[i] >> b[i];
        red += a[i];
        blue += b[i];
    }
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        ans += min(a[i] / k, b[i] / k);
    }
    ans += red / k;
    ans += blue / k;
    cout << ans << endl;
    return 0;
}
