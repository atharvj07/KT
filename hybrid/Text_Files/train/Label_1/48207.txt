import java.util.Scanner;

public class Main {
public static void main(String[] args) {
	Scanner scanner = new Scanner(System.in);
	int k = scanner.nextInt(), n = scanner.nextInt(), mod = 998244353;
	long[][] dp = new long[k * 2 + 1][n + 1];
	dp[0][0] = 1;
	for (int i = 1; i <= k * 2; i++) {
		for (int j = 0; j <= n; j++) {
			for (int l = 1; l <= k && j + l <= n; l++) {
				if (i - l >= 1) dp[i][j + l] = (dp[i][j + l] + dp[i - l][j]) % mod;
			}
		}
	}
	for (int i = 2; i <= k * 2; i++) System.out.println(dp[i][n]);
}
}
