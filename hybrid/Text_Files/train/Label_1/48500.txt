import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] heights = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                heights[i][j] = sc.nextInt();
            }
        }

        int[][] ans = new int[n][m];
        for (int i = 0; i < n; i++) {
            int[] arr = new int[m + 1];
            for (int j = 0; j < m; j++) {
                arr[j + 1] = heights[i][j];
            }
            Arrays.sort(arr);
            int cnt = 1;
            for (int j = 0; j < m; j++) {
                if (heights[i][j] == arr[cnt]) {
                    ans[i][j] = cnt++;
                }
            }
        }

        for (int j = 0; j < m; j++) {
            int[] arr = new int[n + 1];
            for (int i = 0; i < n; i++) {
                arr[i + 1] = heights[i][j];
            }
            Arrays.sort(arr);
            int cnt = 1;
            for (int i = 0; i < n; i++) {
                if (heights[i][j] == arr[cnt]) {
                    ans[i][j] = cnt++;
                }
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(ans[i][j] + " ");
            }
            System.out.println();
        }
    }
}
