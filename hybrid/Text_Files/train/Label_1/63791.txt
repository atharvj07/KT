import sys

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        arr = list(map(int, input().split()))
        mn = min(arr)
        if mn > n:
            print(-1)
        else:
            ans = 0
            for i in arr:
                if i % mn != 0:
                    ans += 1
            print(ans)

solve()
