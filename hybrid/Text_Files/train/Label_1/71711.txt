import numpy as np
n = int(input())
board = np.array([list(input()) for i in range(n)])
for i in range(n):
    for j in range(n):
        if board[i][j] == "#" and (i == 0 or j == 0 or i == n - 1 or j == n - 1):
            print("NO")
            exit()
        elif board[i][j] == "." and (i == 0 or j == 0 or i == n - 1 or j == n - 1):
            board[i][j] = "#"
            if i > 0 and board[i - 1][j] == ".":
                board[i - 1][j] = "#"
            if i < n - 1 and board[i + 1][j] == ".":
                board[i + 1][j] = "#"
            if j > 0 and board[i][j - 1] == ".":
                board[i][j - 1] = "#"
            if j < n - 1 and board[i][j + 1] == ".":
                board[i][j + 1] = "#"
            # check
            flag = True
            for k in range(n):
                for l in range(n):
                    if board[k][l] == ".":
                        flag = False
            if flag:
                print("YES")
                exit()
            board[i][j] = "."
            if i > 0 and board[i - 1][j] == "#":
                board[i - 1][j] = "."
            if i < n - 1 and board[i + 1][j] == "#":
                board[i + 1][j] = "."
            if j > 0 and board[i][j - 1] == "#":
                board[i][j - 1] = "."
            if j < n - 1 and board[i][j + 1] == "#":
                board[i][j + 1] = "."
print("NO")
