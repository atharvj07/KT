#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    string s;
    cin >> s;
    int cnt[10] = {0};
    for (const auto& c : s) cnt[c - '0']++;
    int ans = 0;
    if ((cnt[6] + cnt[9]) % 2 == 0)
        ans += (cnt[6] + cnt[9]) / 2;
    else
        ans += (cnt[6] + cnt[9] - 1) / 2;
    if ((cnt[5] + cnt[2]) % 2 == 0)
        ans += (cnt[5] + cnt[2]) / 2;
    else
        ans += (cnt[5] + cnt[2] - 1) / 2;
    ans += min(cnt[0], cnt[1]);
    ans += min(cnt[3], cnt[4]);
    ans += min(cnt[7], cnt[8]);
    cout << ans << endl;
    return 0;
}
