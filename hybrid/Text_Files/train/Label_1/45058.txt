import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    static int n;
    static ArrayList<Integer>[] adj;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        adj = new ArrayList[n + 1];
        int[] degree = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 1; i <= n; i++) {
            int p = sc.nextInt();
            adj[p].add(i);
            adj[i].add(p);
            degree[p]++;
            degree[i]++;
        }
        boolean valid = true;
        for (int i = 1; i <= n; i++) {
            if (degree[i] % 2 == 1) {
                valid = false;
                break;
            }
        }
        if (!valid) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            boolean[] visited = new boolean[n + 1];
            dfs(1, visited, 0);
        }
    }

    static void dfs(int u, boolean[] visited, int depth) {
        visited[u] = true;
        System.out.println(u);
        for (int v : adj[u]) {
            if (!visited[v]) {
                dfs(v, visited, depth + 1);
            }
        }
    }
}
