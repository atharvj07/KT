import java.util.Scanner;

public class JigsawArrangement {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int N = scanner.nextInt();
        int H = scanner.nextInt();

        int[] A = new int[N];
        int[] B = new int[N];
        int[] C = new int[N];
        int[] D = new int[N];

        for (int i = 0; i < N; i++) {
            A[i] = scanner.nextInt();
            B[i] = scanner.nextInt();
            C[i] = scanner.nextInt();
            D[i] = scanner.nextInt();
        }

        boolean possible = true;

        // Check if the total height of the left and right parts is less than or equal to H for each piece.
        for (int i = 0; i < N; i++) {
            if (A[i] + B[i] > H) {
                possible = false;
                break;
            }
        }

        // Check if the pieces can be arranged in a non-overlapping manner.
        if (possible) {
            int[] leftHeights = new int[H + 1];
            int[] rightHeights = new int[H + 1];

            for (int i = 0; i < N; i++) {
                // Check if the left part of the current piece can be placed on the table.
                if (leftHeights[A[i] + C[i]] > 0) {
                    possible = false;
                    break;
                }

                // Update the left heights array.
                for (int j = 0; j <= A[i] + C[i]; j++) {
                    leftHeights[j]++;
                }

                // Check if the right part of the current piece can be placed on the table.
                if (rightHeights[B[i] + D[i]] > 0) {
                    possible = false;
                    break;
                }

                // Update the right heights array.
                for (int j = 0; j <= B[i] + D[i]; j++) {
                    rightHeights[j]++;
                }
            }
        }

        if (possible) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
