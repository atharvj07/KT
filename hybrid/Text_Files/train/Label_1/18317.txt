#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int getMaxCliqueSize(vector<int>& v, int n)
{
    vector<int> dp(n, 1);
    int maxLength = 1;
    for (int i = 1; i < n; i++)
    {
        for (int j = 0; j < i; j++)
        {
            if (v[i] % v[j] == 0 || v[j] % v[i] == 0)
                dp[i] = max(dp[i], dp[j] + 1);
        }
        maxLength = max(maxLength, dp[i]);
    }
    return maxLength;
}

int main()
{
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++)
        cin >> v[i];
    cout << getMaxCliqueSize(v, n) << endl;
    return 0;
}
