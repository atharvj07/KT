from collections import defaultdict, deque
n = int(input())
a = list(map(int,input().split()))
g = defaultdict(list)
for x in a:
    g[x] = []
for i in range(n-1):
    if a[i] > a[i+1]:
        g[a[i]].append(a[i+1])
        g[a[i+1]].append(a[i])
        a[i], a[i+1] = a[i+1], a[i]
vis = [False] * (n+1)
ans = 0
for e in a:
    if vis[e]: continue
    q = deque()
    q.append(e)
    sz = 0
    while q:
        e = q.popleft()
        if vis[e]: continue
        vis[e] = True
        sz += 1
        for i in g[e]: q.append(i)
    ans = max(ans,sz)
print(ans)
