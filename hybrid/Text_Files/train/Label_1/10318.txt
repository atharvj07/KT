#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  string s;
  cin >> s;
  int cnt[26] = {0};
  for (char c : s) cnt[c - 'a']++;
  int odd = 0;
  for (int i = 0; i < 26; i++) {
    odd += cnt[i] % 2;
  }
  if (odd > 1)
    cout << "Second" << endl;
  else
    cout << "First" << endl;
  return 0;
}
