#include <iostream>
#include <vector>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> ans(n, 0);
  for (int i = 0; i < n; i++) {
    if (s[i] == '<') {
      if (i == 0) {
        ans[i] = 1;
      } else if (s[i - 1] == '>') {
        ans[i] = 1;
      }
    } else if (s[i] == '>') {
      if (i == n - 1) {
        ans[i] = 1;
      } else if (s[i + 1] == '<') {
        ans[i] = 1;
      }
    }
  }
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    cnt += ans[i];
  }
  cout << cnt << endl;
  return 0;
}
