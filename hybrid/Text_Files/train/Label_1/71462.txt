import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
k = list(map(int, input().split()))

# Construct dp table
dp = [[[-1] * (sum(k) + 1) for _ in range(m + 1)] for _ in range(n + 1)]

# Base case
dp[0][0][0] = 0

# Calculate dp table
for i in range(n):
    for j in range(m):
        for s in range(sum(k)):
            # Try to include current element
            if a[i] == j + 1 and s + k[j] <= sum(k):
                dp[i + 1][j + 1][s + k[j]] = max(dp[i + 1][j + 1][s + k[j]], dp[i][j][s] + 1)
            # Try to exclude current element
            dp[i + 1][j][s] = max(dp[i + 1][j][s], dp[i][j][s])

# Check if it is possible
if dp[n][m][sum(k)] == sum(k):
    print("YES")
else:
    print("NO")
