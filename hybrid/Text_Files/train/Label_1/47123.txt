import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == a[Math.min(i + k - 1, n - 1)]) cnt++;
            else cnt = 1;
            if (cnt == k) {
                i = i + k - 1;
                cnt = 0;
            }
        }
        System.out.println(n - cnt);
    }
}
