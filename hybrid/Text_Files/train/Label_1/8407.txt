#include <iostream>

using namespace std;

int main() {
  long long p, y;
  cin >> p >> y;
  long long ans = -1;
  for (long long i = p; i <= y; i++) {
    bool flag = true;
    for (long long j = 2; j * j <= i; j++) {
      if (i % j == 0) {
        flag = false;
        break;
      }
    }
    if (flag) {
      ans = i;
    }
  }
  cout << ans << endl;
  return 0;
}
