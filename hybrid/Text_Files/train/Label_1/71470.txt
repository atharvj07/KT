def max_gain(n, m, a):
    dp1 = [[0 for _ in range(m)] for _ in range(n)]
    dp2 = [[0 for _ in range(m)] for _ in range(n)]
    dp3 = [[0 for _ in range(m)] for _ in range(n)]
    dp4 = [[0 for _ in range(m)] for _ in range(n)]
    
    for i in range(n):
        for j in range(m):
            if i == 0 and j == 0:
                dp1[i][j] = a[i][j]
            elif i == 0:
                dp1[i][j] = dp1[i][j - 1] + a[i][j]
            elif j == 0:
                dp1[i][j] = dp1[i - 1][j] + a[i][j]
            else:
                dp1[i][j] = max(dp1[i - 1][j], dp1[i][j - 1]) + a[i][j]
    
    for i in range(n - 1, -1, -1):
        for j in range(m - 1, -1, -1):
            if i == n - 1 and j == m - 1:
                dp2[i][j] = a[i][j]
            elif i == n - 1:
                dp2[i][j] = dp2[i][j + 1] + a[i][j]
            elif j == m - 1:
                dp2[i][j] = dp2[i + 1][j] + a[i][j]
            else:
                dp2[i][j] = max(dp2[i + 1][j], dp2[i][j + 1]) + a[i][j]
    
    for i in range(n - 1, -1, -1):
        for j in range(m):
            if i == n - 1 and j == 0:
                dp3[i][j] = a[i][j]
            elif i == n - 1:
                dp3[i][j] = dp3[i][j - 1] + a[i][j]
            elif j == 0:
                dp3[i][j] = dp3[i + 1][j] + a[i][j]
            else:
                dp3[i][j] = max(dp3[i + 1][j], dp3[i][j - 1]) + a[i][j]
    
    for i in range(n):
        for j in range(m - 1, -1, -1):
            if i == 0 and j == m - 1:
                dp4[i][j] = a[i][j]
            elif i == 0:
                dp4[i][j] = dp4[i][j + 1] + a[i][j]
            elif j == m - 1:
                dp4[i][j] = dp4[i - 1][j] + a[i][j]
            else:
                dp4[i][j] = max(dp4[i - 1][j], dp4[i][j + 1]) + a[i][j]
    
    ans = 0
    for i in range(1, n - 1):
        for j in range(1, m - 1):
            ans = max(ans, dp1[i - 1][j] + dp2[i + 1][j] + dp3[i][j - 1] + dp4[i][j + 1],
                         dp1[i][j - 1] + dp2[i][j + 1] + dp3[i + 1][j] + dp4[i - 1][j])
    
    return ans

n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
print(max_gain(n, m, a))
