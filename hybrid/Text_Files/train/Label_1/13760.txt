#include <vector>
#include <iostream>
int main() {
    int n, m, x, y, ans = 0, c = 0, r = 0;
    std::vector<bool> col(100001, false), row(100001, false);
    std::cin >> n >> m;
    for (int i = 0; i < m; i++) {
        std::cin >> x >> y;
        if (!col[x] && !row[y]) {
            c++;
            r++;
        } else if (!col[x]) {
            c++;
        } else if (!row[y]) {
            r++;
        }
        col[x] = row[y] = true;
        ans += n * n - c * n - r * n + c * r;
        std::cout << ans << '\n';
    }
}
