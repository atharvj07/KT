#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    while (cin >> n) {
        if (n % 4 != 1) return 0;
        for (int i = 1, j = n; i <= n * n; i++) {
            int x, y;
            if (i <= n * n / 2) {
                if (i % 2 == 1) x = n - i % n, y = i % n;
                else x = i % n, y = n - i % n;
            } else {
                if (i % 2 == 0) x = i % n, y = n - i % n;
                else x = n - i % n, y = i % n;
            }
            cout << x * n + y << ' ';
            if (j-- == 0) j = n, cout << '\n';
        }
    }
}
