from collections import deque
n=int(input())
a=[0]*(n+1);a[1]=1
vis=[0]*(n+1);vis[1]=1
q=deque([2]);q2=deque()
while q:
 i=q.popleft()
 j=2
 while j*i<=n:
  if vis[j*i]==0:
   vis[j*i]=1;q.append(j*i)
   if j>1: q2.append(j*i)
  j+=1
while q2:
 i=q2.popleft()
 a[i]=n
for i in range(2,n+1):
 if a[i]==0: a[i]=1
print(*a[2:])
