#include <bits/stdc++.h>
#define endl '\n'

using namespace std;

typedef long long ll;

const int N = 1e5 + 5;

int n, m, h;
int u[N];
unordered_map<int, vector<int>> adj;
vector<pair<int, int>> ans;

int main() {
    cin >> n >> m >> h;
    for (int i = 1; i <= n; ++i) {
        cin >> u[i];
        adj[u[i]].push_back(i);
    }
    for (int i = 1; i <= m; ++i) {
        int a, b;
        cin >> a >> b;
        if (u[a] == u[b]) {
            cout << 1 << endl << a << endl;
            return 0;
        }
    }
    int cnt = 0;
    for (int i = 0; i < h; ++i) {
        int sz = adj[i].size();
        if (sz > 0) {
            if (cnt == 0) {
                for (int j = 0; j < sz; ++j) {
                    int cur = adj[i][j];
                    if (ans.empty() || ans.back().second != cur) {
                        ++cnt;
                        ans.push_back({i, cur});
                        if (cnt == 2) break;
                    }
                }
            }
        }
    }
    cout << ans.size() << endl;
    for (auto p : ans) cout << p.second << ' ';
    return 0;
}
