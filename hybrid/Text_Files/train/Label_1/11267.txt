#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[200001];
int main() {
    ll n;
    cin >> n;
    for (ll i = 1; i <= n; i++) cin >> a[i];
    priority_queue<ll, vector<ll>, greater<ll> > q;
    for (ll i = 1; i <= n; i++) q.push(a[i]);
    while (q.size() > 1) {
        ll u = q.top();
        q.pop();
        ll v = q.top();
        q.pop();
        q.push(u + v);
    }
    cout << q.top() << endl;
}
