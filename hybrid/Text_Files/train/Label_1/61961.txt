import sys

n = int(input())
s = input()
x, y = map(int, input().split())
l, r = 0, n - 1
while l < r:
    if x == 0 and y == 0: break
    if s[l] == s[r]:
        if s[l] == 'U': y -= 1
        elif s[l] == 'D': y += 1
        elif s[l] == 'L': x += 1
        else: x -= 1
        l += 1
        r -= 1
    else: break
if x == 0 and y == 0: print(r - l + 1)
else: print(-1)
