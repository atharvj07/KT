#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct data {
    string type;
    int val;
    int cnt;
};

bool comp(data a, data b) {
    return a.val > b.val;
}

int main() {
    int n;
    cin >> n;
    vector<data> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].type >> v[i].val;
    }
    for (int i = 0; i < n; i++) {
        if (v[i].type == "11") {
            v[i].cnt = 2;
        } else if (v[i].type == "10") {
            v[i].cnt = 1;
        } else if (v[i].type == "01") {
            v[i].cnt = 1;
        } else {
            v[i].cnt = 0;
        }
    }
    sort(v.begin(), v.end(), comp);
    int a = 0, b = 0, sum = 0;
    for (int i = 0; i < n; i++) {
        if (a >= n / 2 && b >= n / 2) {
            break;
        }
        if (v[i].cnt == 2) {
            a++;
            b++;
            sum += v[i].val;
        } else if (a < n / 2 && v[i].cnt == 1) {
            a++;
            sum += v[i].val;
        } else if (b < n / 2 && v[i].cnt == 1) {
            b++;
            sum += v[i].val;
        }
    }
    if (a >= n / 2 && b >= n / 2) {
        cout << sum << endl;
    } else {
        cout << 0 << endl;
    }
    return 0;
}
