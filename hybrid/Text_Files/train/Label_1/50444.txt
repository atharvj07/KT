import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int[] D = new int[N];
        for (int i = 0; i < N; i++) {
            D[i] = scanner.nextInt();
        }
        boolean[] canReach = new boolean[N];
        boolean canReturn = false;
        canReach[0] = true;
        for (int i = 0; i < N; i++) {
            if (!canReach[i]) continue;
            if (i == N - 1) canReturn = true;
            for (int j = i + 1; j <= i + D[i] && j < N; j++) {
                canReach[j] = true;
            }
        }
        System.out.println(canReturn && canReach[N - 1] ? "yes" : "no");
    }
}
