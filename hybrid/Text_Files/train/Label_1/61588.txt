n = int(input())
a = [int(x) for x in input().split()]
ans = []
if n % 2 == 0:
    for i in range(0, n, 2):
        if a[i] >= a[i + 1]:
            ans.append(a[i])
        else:
            ans = ["Impossible"]
            break
    for i in range(1, n, 2):
        if a[i] <= a[i + 1]:
            ans.append(a[i])
        else:
            ans = ["Impossible"]
            break
else:
    for i in range(0, n - 1, 2):
        if a[i] >= a[i + 1]:
            ans.append(a[i])
        else:
            ans = ["Impossible"]
            break
    for i in range(1, n, 2):
        if a[i] <= a[i - 1]:
            ans.append(a[i])
        else:
            ans = ["Impossible"]
            break
print(*ans)
