import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n,k = map(int,input().split())
    a = list(map(int,input().split()))
    g = [[]for i in range(n+1)]
    for _ in range(n-1):
        u,v = map(int,input().split())
        g[u].append(v)
        g[v].append(u)
    c = 0
    vis = [False] * (n+1)
    ans = True
    def dfs(u,x,p):
        nonlocal c
        nonlocal ans
        vis[u]=True
        c^=a[u-1]
        for v in g[u]:
            if vis[v] == False:
               dfs(v,a[u-1],u)
        if c!=x:
            ans=False
    dfs(1,a[0],0)
    if ans:
        print("YES")
    else:
        print("NO")
