#include<iostream>
#include<algorithm>
#include<vector>
using namespace std;
int main() {
  int n, k, ans = 0;
  cin >> n >> k;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) cin >> a[i], ans = max(ans, b[i] / a[i]);
  sort(a.begin(), a.end()), sort(b.begin(), b.end());
  while (k >= a.front() && ans * n + 1 <= b.back() / a.front() + ans) {
    k -= a.front(), ans++, a.erase(a.begin());
  }
  cout << (k >= 0 ? ans : -1) << '\n';
  return 0;
}
