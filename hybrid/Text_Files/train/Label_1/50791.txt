import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), x = sc.nextInt();
        int[][] a = new int[N][];
        for (int i = 0; i < N; i++) {
            a[i] = new int[2 * i + 1];
            Arrays.fill(a[i], -1);
        }
        a[0][0] = x;
        for (int i = 1; i < N; i++) {
            for (int j = 0; j < 2 * i + 1; j++) {
                if (a[i][j] != -1) {
                    int m = (j > 0) ? a[i][j - 1] : -1;
                    int c = a[i][j];
                    int r = (j < 2 * i) ? a[i][j + 1] : -1;
                    for (int d = m + 1; d <= r; d++) {
                        if (d == c) {
                            continue;
                        }
                        if (a[i - 1][j / 2] == -1) {
                            a[i - 1][j / 2] = d;
                        } else if (a[i - 1][j / 2] == d) {
                            if (a[i - 1][(j / 2) - 1] == -1) {
                                a[i - 1][(j / 2) - 1] = d;
                            } else if (a[i - 1][(j / 2) + 1] == -1) {
                                a[i - 1][(j / 2) + 1] = d;
                            }
                        }
                    }
                }
            }
        }
        if (a[N - 1][0] != -1) {
            System.out.println("Yes");
            for (int i = 0; i < N; i++) {
                for (int j = 0; j < 2 * i + 1; j++) {
                    if (a[i][j] != -1) {
                        System.out.println(a[i][j]);
                    }
                }
            }
        } else {
            System.out.println("No");
        }
    }
}
