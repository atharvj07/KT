import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int d = sc.nextInt();
        
        int step=0;
        while(Math.abs(c-b)<d && Math.abs(c-a)<d){
            if(c>b){
                c--;
                step++;
            }else if(c<b){
                c++;
                step++;
            }else{
                if(a>b){
                    b++;
                    step++;
                }else if(a<b){
                    a++;
                    step++;
                }
            }
        }
        
        System.out.println(step);
    }
}
