#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    if (k < 0 || k > 25 * n) {
        cout << "-1" << endl;
        return 0;
    }
    for (int i = 0; i < n; i++) {
        s[i] = (s[i] - 'a' + k) % 26 + 'a';
        k -= min(s[i] - 'a', 26 - (s[i] - 'a'));
    }
    if (k) { cout << "-1" << endl; } else { cout << s << endl; }
    return 0;
}
