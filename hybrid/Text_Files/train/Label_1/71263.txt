import sys
input = sys.stdin.readline

n, m = map(int, input().split())
g = [set() for _ in range(n+1)]
for _ in range(m):
  a, b = map(int, input().split())
  g[a].add(b)
  g[b].add(a)

ok = False
for i in range(1, n+1):
  for j in g[i]:
    for k in g[j]:
      if k in g[i]:
        ok = True
        print(sum(len(g[x])-3 for x in [i, j, k]))
        exit()

print(-1)
