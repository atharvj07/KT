import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Deque;
import java.util.ArrayDeque;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        char[] chars = br.readLine().toCharArray();
        Deque<Integer> stack = new ArrayDeque<>();
        int n = 0;
        for (char c: chars)
            if (c == '(') ++n;
            else if (n > 0) {
                --n;
                continue;
            } else stack.add(stack.isEmpty() ? 1 : stack.getLast() + 2);

        StringBuilder sb = new StringBuilder();
        for (char c: chars)
            if (c == '(') {
                if (stack.isEmpty() || stack.getLast() > 2) sb.append('(');
                else sb.append(')');
            } else {
                if (!stack.isEmpty()) {
                    if (stack.getLast() == 1) sb.append('(');
                    else sb.append(')');
                    stack.removeLast();
                } else sb.append(')');
            }
        StringBuilder res = sb;
        while (!stack.isEmpty()) {
            sb = new StringBuilder();
            for (char c: res.toString().toCharArray())
                if (c == '(') {
                    if (stack.getLast() == 1) sb.append('(');
                    else sb.append(')');
                    stack.removeLast();
                } else sb.append(')');
            if (sb.length() < res.length() || sb.length() == res.length() && sb.compareTo(res) < 0) res = sb;
        }
        System.out.println(res);
    }
}
