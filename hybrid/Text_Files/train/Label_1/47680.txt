import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        sc.nextLine();
        int[][] a = new int[n][];
        for (int i = 0; i < n; i++) {
            String[] line = sc.nextLine().split(" ");
            a[i] = new int[line.length];
            for (int j = 0; j < line.length; j++) {
                a[i][j] = Integer.parseInt(line[j]);
            }
        }
        int[][] preMaxRow = new int[n][m];
        int[][] preMaxCol = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                preMaxRow[i][j] = a[i][j];
                preMaxCol[i][j] = a[i][j];
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 1; j < m; j++) {
                preMaxRow[i][j] = Math.max(preMaxRow[i][j], preMaxRow[i][j - 1]);
            }
        }
        for (int j = 0; j < m; j++) {
            for (int i = 1; i < n; i++) {
                preMaxCol[i][j] = Math.max(preMaxCol[i][j], preMaxCol[i - 1][j]);
            }
        }
        int maxVal = 0;
        int x = -1;
        int y = -1;
        for (int i = k - 1; i < n - k; i++) {
            for (int j = k - 1; j < m - k; j++) {
                int curMax = Math.min(preMaxRow[i][j], preMaxCol[i][j]);
                if (maxVal < curMax) {
                    maxVal = curMax;
                    x = i + 1;
                    y = j + 1;
                }
            }
        }
        System.out.println(x + " " + y);
    }
}
