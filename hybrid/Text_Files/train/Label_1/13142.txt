#include <bits/stdc++.h>
using namespace std;
bool comp(int a,int b){return a>b;}
int main(){
    int n,l;
    cin>>n>>l;
    vector<int> v(n);
    for(int i=0;i<n;++i){
        cin>>v[i];
        v[i]=v[i]*(int)pow(2,i);
    }
    sort(v.begin(),v.end(),comp);
    int sum=0,ans=0;
    for(int i=0;i<n;++i){
        if(sum+v[i]<=l){
            sum+=v[i];
            ans+=v[i]/((int)pow(2,i));
        }else{
            ans+=(l-sum)/((int)pow(2,i));
            break;
        }
    }
    cout<<ans<<endl;
}
