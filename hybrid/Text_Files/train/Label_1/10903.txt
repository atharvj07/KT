#include<bits/stdc++.h>
#define mod 1000000007
#pragma GCC optimize ("Ofast")
using namespace std;
int n,m,dp[2005][2005];
char s[2005][2005];
int32_t main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0),cout.tie(0);
    cin>>n>>m;
    for(int i=1;i<=n;++i)
        for(int j=1;j<=m;++j){
            cin>>s[i][j];
        }
    dp[1][1]=1;
    for(int i=1;i<=n;++i){
        for(int j=1;j<=m;++j){
            if(s[i][j]=='.'){
                dp[i][j]=((dp[i-1][j]+dp[i][j-1])%mod);
            }
            else{
                if(j+1<=m&&s[i][j+1]=='.') dp[i][j+1]=(dp[i][j+1]+dp[i][j])%mod;
                if(i+1<=n&&s[i+1][j]=='.') dp[i+1][j]=(dp[i+1][j]+dp[i][j])%mod;
            }
        }
    }
    cout<<dp[n][m]<<endl;
}
