#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define pb push_back

int main() {
  int n;
  cin >> n;
  ll ans = 0, R = 0;
  for (int i = 0; i < n; i++) {
    ll l;
    cin >> l;
    if (i < R) continue;
    ans++;
    R = i + l;
  }
  cout << ans << endl;
  return 0;
}
