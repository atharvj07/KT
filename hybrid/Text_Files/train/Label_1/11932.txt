#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  int ans = -1;
  for (int i = 1; i < n; i++) {
    int d = (a[i] - a[i - 1]) / 2;
    if (d * 2 == a[i] - a[i - 1] && (a[i] + d >= 1)) {
      if (ans == -1)
        ans = d;
      else
        ans = min(ans, d);
    }
  }
  cout << ans << endl;
  return 0;
}
