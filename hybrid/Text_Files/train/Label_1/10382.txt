#include <bits/stdc++.h>
using namespace std;
#define int long long
const int maxn=1e5+5;
int a[maxn];
main(){
 int n;
 cin>>n;
 for(int i=1; i<=n; i++)cin>>a[i];
 int mn=1e9;
 int ans=-1;
 for(int j=1; j<=n; j++){
  if(a[j]==0)continue;
  int sum=0, l=j, r=j;
  while(l>=1||r<=n){
   sum+=a[l];
   sum+=a[r];
   l--;
   r++;
   if(sum==0)continue;
   if(sum%2==0)ans=min(ans, l+r-2-j);
   else ans=min(ans, l+r-j-1);
  }
 }
 if(ans!=-1e9)cout<<ans<<endl;
 else cout<<-1<<endl;
 return 0;
}
