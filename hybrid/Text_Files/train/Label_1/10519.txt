#include <bits/stdc++.h>
#define rep(i, a, b) for(int i = a; i <= b; i++)
using namespace std;

bool done[110][110], ok;
int a[110][110], b[110][110];
int main() {
    int m, n;
    cin >> m >> n;
    rep(i, 1, m) rep(j, 1, n) cin >> b[i][j];
    rep(i, 1, m) {
        bool t = 0;
        rep(j, 1, n) {
            t |= b[i][j];
            done[i][j] = 1;
        }
        rep(j, 1, n) if(!t) a[i][j] = b[i][j];
    }
    rep(j, 1, n) {
        bool t = 0;
        rep(i, 1, m) {
            t |= b[i][j];
            done[i][j] = 1;
        }
        rep(i, 1, m) if(!t) a[i][j] = b[i][j];
    }
    rep(i, 1, m) rep(j, 1, n) if(!done[i][j]) {
        if(a[i][j] == 1) {
            if((b[i][j] != 1) || (j == 1 && i == 1)) {ok = 0; break;}
            a[i][j] = 0;
            bool got = 0;
            rep(k, 1, n) if(b[i][k] == 0) {a[i][j] = 1; got = 1; break;}
            if(!got) {ok = 0; break;}
            rep(k, 1, m) if(b[k][j] == 0) {a[i][j] = 1; got = 1; break;}
            if(!got) {ok = 0; break;}
        }
    }
    cout << (ok ? "YES\n" : "NO\n");
    if(!ok) return 0;
    rep(i, 1, m) {rep(j, 1, n) cout << a[i][j] << ' '; cout << '\n';}
}
