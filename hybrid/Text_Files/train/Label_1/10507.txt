#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5+5;
int32_t main() {
    int t; cin >> t; while (t--) {
        int n; cin >> n;
        int q[MAXN] = {0};
        for (int i=0; i<n; ++i) cin >> q[i];
        int ans[MAXN] = {0};
        if (q[0] == q[1]) {
            cout << "-1\n"; continue;
        }
        for (int i=0; i<n-1; ++i) ans[i] = q[i+1]-q[i];
        if (ans[0] > 1) {
            cout << "-1\n"; continue;
        }
        ans[n-1] = n;
        for (int i=n-2; i>=0; --i) {
            int curr = ans[i+1];
            if (!curr) {
                if (ans[i] != 1) {
                    cout << "-1\n"; continue;
                }
                curr = 2;
            }
            ans[i] = curr-1;
        }
        for (int i=0; i<n; ++i) cout << ans[i] << " ";
        cout << '\n';
    }
    return 0;
}
