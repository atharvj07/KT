#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int a[n][m], b[n][m];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> a[i][j];
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> b[i][j];
    }
  }
  bool isSame = true;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (a[i][j] != b[i][j]) {
        isSame = false;
        break;
      }
    }
    if (!isSame) break;
  }
  if (isSame) {
    cout << "YES" << endl;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      swap(a[i][j], a[j][i]);
    }
  }
  isSame = true;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (a[i][j] != b[i][j]) {
        isSame = false;
        break;
      }
    }
    if (!isSame) break;
  }
  if (isSame) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
}
