import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            Arrays.sort(a);
            long sum = 0;
            for (int j = 0; j < n; j++) {
                sum += a[j];
            }
            long avg = (sum + n - 1) / n;
            System.out.println(avg);
        }
        sc.close();
    }
}
