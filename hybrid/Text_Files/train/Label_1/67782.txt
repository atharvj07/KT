import sys
from collections import Counter

n = int(input())
s = [input() for i in range(n)]
mp = Counter()

for t in s:
    mp[t] += 1

ans = 0

for t in mp:
    if t == t[::-1]:
        ans += mp[t] * (mp[t] - 1) // 2

    else:
        rev = t[::-1]
        if rev in mp:
            ans += mp[t] * mp[rev]

print(ans)
