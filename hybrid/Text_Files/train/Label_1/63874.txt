n, m, k = map(int, input().split())
arr = [[] for _ in range(n)]
mem = [False for _ in range(k + 1)]
for i in range(n):
    arr[i] = [int(x) for x in input().split()]
ans = [0] * n
locked = []
for j in range(m):
    core = []
    for i in range(n):
        if arr[i][j] != 0 and not mem[arr[i][j]]:
            core.append(i)
        if len(core) > 1:
            for x in core:
                if x not in locked:
                    locked.append(x)
                    ans[x] = j + 1
            for x in core:
                mem[arr[x][j]] = True
print(*ans)
