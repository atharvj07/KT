#include<bits/stdc++.h>
using namespace std;
typedef unsigned long long ll;
const int N=1e5+10;
int n;
ll b[N];
int32_t main() {
    cin>>n;
    for(int i=1; i<=n; i++)cin>>b[i];
    map<ll,int>mp;
    for(int i=1; i<=n; i++)mp[b[i]]++;
    for(int i=1; i<=n; i++){
        if(mp[b[i]]!=1)return cout<<"No",0;
        b[i]=b[i]-i;
    }
    sort(b+1,b+1+n);
    cout<<"Yes\n";
    for(int i=1; i<=n; i++)cout<<b[i]<<" ";
}
