#include <algorithm>
#include <iostream>
#include <string>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<std::string> v(n);
  for (std::string &s : v) std::cin >> s;
  sort(v.begin(), v.end(), [&](const std::string &l, const std::string &r) {
    return l + r < r + l;
  });
  for (const auto &s : v) std::cout << s;
  std::cout << '\n';
  return 0;
}
