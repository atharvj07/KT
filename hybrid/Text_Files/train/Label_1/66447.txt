n=int(input())
t=[tuple(map(int,input().split()))for i in range(n)]
t.sort(reverse=True)
f,s,p,q=[0]*n,[0]*n,[0]*n,[0]*n
for i in range(n):
 f[i]+=t[i][0]-t[i][1]
for i in range(1,n):
 f[i]+=f[i-1]
for i in range(n-1,-1,-1):
 s[i]+=t[i][0]-t[i][1]
for i in range(n-2,-1,-1):
 s[i]+=s[i+1]
for i in range(n):
 for j in range(i,n):
  if i==j:
   p[j]+=t[i][1]
   q[j]+=t[i][1]
  elif f[j-1]-q[j-1]>=t[i][0]:
   p[j]+=t[i][1]
   q[j]+=t[i][1]-t[i][0]
print(p[n-1]%1000000007)
