#include <iostream>
#include <cstdio>
#include <vector>
using namespace std;

struct ent{
    int l, r;
};

int main(){
    int n, x, i, j, t, ans = 0, now = 0;
    vector<ent> v;
    scanf("%d %d", &n, &x);
    for(i = 0; i < n; i++){
        ent t;
        scanf("%d %d", &t.l, &t.r);
        v.push_back(t);
    }
    for(i = 0; i < n; i++){
        if(v[i].l > now)
            ans += v[i].l - now, now = v[i].l;
        now = max(now, v[i].r);
    }
    printf("%d\n", ans);
    return 0;
}
