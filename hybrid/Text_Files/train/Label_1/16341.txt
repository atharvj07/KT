#include <iostream>
#include <vector>

int main() {
  int n, s, t;
  std::cin >> n >> s >> t;

  std::vector<int> perm(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> perm[i];
  }

  std::vector<int> pos(n + 1, -1);
  for (int i = 0; i < n; ++i) {
    pos[perm[i]] = i;
  }

  if (pos[s] == -1 || pos[t] == -1) {
    std::cout << -1 << std::endl;
    return 0;
  }

  int ans = 0;
  while (pos[s] != pos[t]) {
    s = perm[s];
    ++ans;
  }

  std::cout << ans << std::endl;
  return 0;
}
