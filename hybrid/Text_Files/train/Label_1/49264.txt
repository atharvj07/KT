import java.util.*;

class Main {
  static class App implements Comparable<App> {
    int idx, a, b;

    App(int idx, int a, int b) {
      this.idx = idx;
      this.a = a;
      this.b = b;
    }

    @Override
    public int compareTo(App other) {
      return (this.a - this.b) - (other.a - other.b);
    }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    StringBuilder sb = new StringBuilder();
    int t = sc.nextInt();

    while (t-- > 0) {
      int n = sc.nextInt();
      int m = sc.nextInt();
      App[] apps = new App[n];
      int ans = -1;

      for (int i = 0; i < n; i++) {
        int a = sc.nextInt();
        int b = sc.nextInt();
        apps[i] = new App(i + 1, a, b);
      }

      Arrays.sort(apps);
      long suma = 0;

      for (App app : apps) {
        if (suma + app.a >= m) {
          ans = 0;

          for (App sortedApp : apps) {
            if (suma >= m) break;
            if (sortedApp.idx <= app.idx) continue;
            suma += sortedApp.a;
            ans += sortedApp.b;
          }
          break;
        }
        suma += app.a;
      }

      sb.append(ans).append('\n');
    }
    System.out.print(sb);
  }
}
