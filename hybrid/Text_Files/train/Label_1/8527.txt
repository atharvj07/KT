#include<bits/stdc++.h>
using namespace std;
int main(){
    int a,b,c,m,f=0,l=0,sum=0;
    cin>>a>>b>>c>>m;
    map<string,int>v1,v2,v3;
    int k=0;
    while(m--){
        int x; string s;
        cin>>x>>s;
        if(s=="USB"){
            if(v1[s]<a) {v1[s]++;l+=x;k++;}
            else if(c&&v3[s]<c) {v3[s]++;sum+=x;k++;}
        }
        if(s=="PS/2"){
            if(v2[s]<b) {v2[s]++;f+=x;k++;}
            else if(c&&v3[s]<c) {v3[s]++;sum+=x;k++;}
        }
    }
    cout<<k<<" "<<f+l+sum<<endl;
}
