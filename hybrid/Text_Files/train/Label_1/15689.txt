#include <iostream>
#include <vector>
using namespace std;

const long long INF = 1e18;

long long dfs(int v, int p, const vector<vector<pair<int, int>>>& g, vector<bool>& seen, long long sum){
    seen[v] = true;
    long long ret = INF;
    for(const auto& edge: g[v]){
        int to = edge.first, w = edge.second;
        if(seen[to]) continue;
        ret = min(ret, dfs(to, v, g, seen, sum + w));
    }
    if(p != -1)
        ret = min(ret, sum + 2 * (g[v][0].second == g[p][0].second ? g[v][1].second : g[v][0].second) - g[v][0].second - g[p][0].second);
    return ret;
}

int main(){
    int n, x, y;
    cin >> n >> x >> y;
    vector<vector<pair<int, int>>> g(n);
    for(int i = 0; i < n - 1; i++){
        int u, v;
        cin >> u >> v;
        u--, v--;
        g[u].emplace_back(v, x);
        g[v].emplace_back(u, x);
    }
    vector<long long> sum(n);
    for(int i = 0; i < n; i++){
        for(auto& edge: g[i])
            sum[i] += edge.second;
    }
    vector<bool> seen(n, false);
    cout << dfs(0, -1, g, seen, 0) << endl;
    return 0;
}
