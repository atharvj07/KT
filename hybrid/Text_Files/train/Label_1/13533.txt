#include <bits/stdc++.h>

using namespace std;

int main() {
  long long a, b, x, y;
  cin >> a >> b >> x >> y;
  long long gcd = __gcd(x, y);
  long long min_w = (a / x) * x, min_h = (b / y) * y;
  if (min_w > a || min_h > b) {
    cout << 0 << endl;
  } else {
    cout << (a / gcd) * (b / gcd) << endl;
  }
  return 0;
}
