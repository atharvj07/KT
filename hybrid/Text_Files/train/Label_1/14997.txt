#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;cin >> n;
    int r[n+1];
    long long int sum=0; int count=0; int mx=0;
    for(int i=1;i<=n;i++)
    {
        cin >> r[i];
        sum+=r[i];
        if(sum > 100*i)
        {
            sum-=r[i-count];
            count++;
        }
        mx=max(mx,i-count);
    }
    cout << mx;
}
