n, m = map(int, input().split())
s = input()
t = input()
if '*' in s:
    if n-1 > m or s.count('*') > 1:
        print('NO')
    else:
        t = t[:s.index('*')] + t[s.index('*')+1:]
        if s.strip('*') == t:
            print('YES')
        else:
            print('NO')
else:
    if s == t:
        print('YES')
    else:
        print('NO')
