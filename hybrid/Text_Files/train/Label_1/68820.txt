n, m = map(int, input().split())
colors = list(map(int, input().split()))

edges = [set() for _ in range(n)]
for _ in range(m):
    a, b = map(int, input().split())
    edges[a-1].add(b-1)
    edges[b-1].add(a-1)

color_neighbors = {}
for i in range(n):
    color = colors[i]
    if color not in color_neighbors:
        color_neighbors[color] = set()
    for neighbor in edges[i]:
        if colors[neighbor] != color:
            color_neighbors[color].add(colors[neighbor])

max_diversity = 0
result_color = 0
for color, neighbors in color_neighbors.items():
    if len(neighbors) > max_diversity:
        max_diversity = len(neighbors)
        result_color = color
    elif len(neighbors) == max_diversity and color < result_color:
        result_color = color

print(result_color)
