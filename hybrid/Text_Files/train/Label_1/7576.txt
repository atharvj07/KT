#include <bits/stdc++.h>
using namespace std;
using ll=long long;
int main() {
    ll n;
    cin>>n;
    unordered_map<int, ll> m;
    vector<pair<int, int>> v(n);
    for(ll i=0; i<n; i++){
        cin>>v[i].first>>v[i].second;
    }
    for(ll i=0; i<n; i++){
        m[v[i].first]++;
        m[v[i].second]++;
    }
    for(ll i=0; i<n; i++){
        if(m[v[i].first]==1) cout<<n-1<<" ";
        else cout<<n-m[v[i].first]<<" ";
        if(m[v[i].second]==1) cout<<n-1<<endl;
        else cout<<n-m[v[i].second]<<endl;
    }
    return 0;
}
