import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        sc.close();
        int n = s.length();
        int even = 0;
        int odd = 0;
        for (int i = 0; i < n; i++) {
            boolean[] arr = new boolean[n];
            for (int j = i; j < n; j++) {
                if (arr[j]) {
                    continue;
                }
                int l = j;
                int r = j;
                while (r < n && s.charAt(l) == s.charAt(r)) {
                    r++;
                }
                int len = r - l;
                if (len % 2 == 0) {
                    even++;
                }
                if (len % 2 == 1) {
                    odd++;
                }
                for (int k = l; k < r; k++) {
                    arr[k] = true;
                }
            }
        }
        System.out.println(even + " " + odd);
    }
}
