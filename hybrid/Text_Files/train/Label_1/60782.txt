from sys import stdin, setrecursionlimit
input=stdin.readline
setrecursionlimit(1<<20)

n,a=int(input()),[int(x) for x in input().split()]
xor=0
for i in range(61):
    on=0
    for v in a:
        if v&(1<<i): on+=1
    xor^=(1<<i)*((on*(on-1))&1)
if xor==0:ans=n*(n+1)//2-1
else:ans=0
for i in range(n):
    if a[i]==xor:ans+=i+1
print(ans)
