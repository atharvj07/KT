import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nkq = br.readLine().split(" ");
        int n = Integer.parseInt(nkq[0]);
        int k = Integer.parseInt(nkq[1]);
        int q = Integer.parseInt(nkq[2]);

        long[] friends = new long[n];
        for (int i = 0; i < n; i++) {
            friends[i] = Long.parseLong(br.readLine());
        }

        Arrays.sort(friends);

        int[] displayed = new int[k];
        Arrays.fill(displayed, -1);

        for (int i = 0; i < q; i++) {
            String[] query = br.readLine().split(" ");
            int type = Integer.parseInt(query[0]);
            int id = Integer.parseInt(query[1]) - 1;
            if (type == 1) {
                for (int j = 0; j < k; j++) {
                    if (displayed[j] == -1 || friends[id] > friends[displayed[j]]) {
                        displayed[j] = id;
                        break;
                    }
                }
            } else {
                for (int j = 0; j < k; j++) {
                    if (displayed[j] == id) {
                        System.out.println("YES");
                        break;
                    } else if (displayed[j] == -1) {
                        System.out.println("NO");
                        break;
                    }
                }
            }
        }
    }
}
