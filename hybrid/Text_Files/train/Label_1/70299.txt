H, n = map(int, input().split())
d = list(map(int, input().split()))

total_damage = sum(d)
hp = H
for i in range(n):
    hp += d[i]
    if hp <= 0:
        print(i + 1)
        break
else:
    if total_damage >= 0:
        print(-1)
    else:
        rounds = (H - 1) // -total_damage
        hp = H + rounds * total_damage
        minutes = rounds * n
        for i in range(n):
            hp += d[i]
            minutes += 1
            if hp <= 0:
                print(minutes)
                break
