#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        long long x, n, ans;
        cin>>x>>n;
        long long pos = n * (n + 1) / 2;
        if(pos % 2 == 0){
            ans = x - (pos / 2);
        }else{
            ans = x + (pos / 2 + 1);
        }
        cout<<ans<<endl;
    }
    return 0;
}
