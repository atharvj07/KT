#include<bits/stdc++.h>
using namespace std;
int main(){
    long long n, a, b, sum = 0; cin >> n >> a >> b;
    vector<long long> hole;
    for(int i = 0; i < n; i++) {long long x; cin >> x; hole.push_back(x), sum += x;}
    sort(hole.begin(), hole.end());
    long long lt = 0, rt = n - 1, ans = 0;
    while(lt <= rt){
        if(a * hole[lt] / sum >= b){
            ans = rt, rt--;
        }else lt++;
    }
    cout << n - 1 - ans << endl;
}
