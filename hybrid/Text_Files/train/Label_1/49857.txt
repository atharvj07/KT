import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int H = sc.nextInt();
        int W = sc.nextInt();
        int X = sc.nextInt();
        int Y = sc.nextInt();
        if (H * W % 2 == X + Y % 2) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
