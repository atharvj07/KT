#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, k;
    cin >> n >> m >> k;
    vector<long long> h(n);
    for (int i = 0; i < n; i++) cin >> h[i];
    sort(h.rbegin(), h.rend());
    if (m < (h[n - 2] - h[n - 1] > k)) {
      cout << "NO" << endl;
    } else
      cout << "YES" << endl;
  }
  return 0;
}
