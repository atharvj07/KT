import collections

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    mnleft = [0]*n
    mnleft[0] = a[0]
    for i in range(1, n):
        mnleft[i] = min(mnleft[i-1], a[i])

    mnright = [0]*n
    mnright[n-1] = a[n-1]
    for i in range(n-2, -1, -1):
        mnright[i] = min(mnright[i+1], a[i])

    inv = 0
    for i in range(n-1):
        inv += max(0, mnleft[i] - mnright[i+1])

    print(inv)
