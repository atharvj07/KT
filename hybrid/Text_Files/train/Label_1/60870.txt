import sys

n, s, k = map(int, input().split())
candy = list(map(int, input().split()))
color = list(input())
if candy[s - 1] < k:
    print(-1)
    sys.exit()
ans = 0
need = k
last = color[s - 1]
for i in range(s - 1, n):
    if color[i] != last:
        if candy[i] >= need:
            ans += i - (s - 1)
            need = 0
            last = color[i]
        else:
            need -= candy[i]
    if need == 0:
        break
if need > 0:
    print(-1)
else:
    print(ans)
