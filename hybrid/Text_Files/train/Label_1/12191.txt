#include <iostream>
#include <vector>

#define ll long long

using namespace std;

const int nmsz = 1e5 + 10;
const ll mod = 1e9 + 7;

struct Point {
    ll x, y, z;
    Point() {}
    Point(ll _x, ll _y, ll _z) : x(_x), y(_y), z(_z) {}
};

vector<Point> a, b;

ll cross(const Point& a, const Point& b, const Point& c) {
    return (b.x - a.x) * (c.y - a.y) - (c.x - a.x) * (b.y - a.y);
}

int main() {
    int n, m;
    cin >> n;
    for (int i = 0; i < n; i++) {
        ll x, y, z;
        cin >> x >> y >> z;
        a.push_back(Point(x, y, z));
    }
    cin >> m;
    for (int i = 0; i < m; i++) {
        ll x, y, z;
        cin >> x >> y >> z;
        b.push_back(Point(x, y, z));
    }
    bool ok = 0;
    for (int i = 0; i < n; i++) {
        ll up = 0, down = 0;
        for (int j = 0; j < m; j++) {
            ll cr = cross(a[i], a[(i + 1) % n], b[j]);
            if (cr != 0) {
                if (cr > 0)
                    up++;
                else
                    down++;
            }
        }
        if (!(up == 0 && down == 0))
            ok = 1;
    }
    for (int i = 0; i < m; i++) {
        ll up = 0, down = 0;
        for (int j = 0; j < n; j++) {
            ll cr = cross(b[i], b[(i + 1) % m], a[j]);
            if (cr != 0) {
                if (cr > 0)
                    up++;
                else
                    down++;
            }
        }
        if (!(up == 0 && down == 0))
            ok = 1;
    }
    if (ok)
        cout << "YES\n";
    else
        cout << "NO\n";
}
