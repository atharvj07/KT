import sys
read = sys.stdin.readline

n = int(read().strip())
a = read().strip()
ans = 0
stack = 0
for c in a:
    if c == '(':
        stack += 1
    else:
        if stack == 0:
            ans += 1
        else:
            stack -= 1
            ans += 2
if stack > 0:
    print(-1)
else:
    print(ans)
