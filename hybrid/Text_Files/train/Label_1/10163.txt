#include <iostream>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;

  int l = k, r = k;
  for (int i = 0; i < n; i++) {
    if (i + 1 < l) {
      cout << "LEFT" << endl;
      l--;
    } else if (i + 1 > r) {
      cout << "RIGHT" << endl;
      r++;
    }
    cout << "PRINT " << s[i] << endl;
  }

  return 0;
}
