#include <iostream>
#include <vector>
#include <utility>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (auto& x : v) {
    cin >> x.first >> x.second;
  }
  sort(v.begin(), v.end());
  int p1 = 0, p2 = 0, r = 0;
  for (int i = 0; i < n; ++i) {
    while (p1 != n && v[p1].first < -r) ++p1;
    while (p2 != n && v[p2].first <= r) ++p2;
    if (p1 == n && p2 == n) break;
    if (p1 != n && p2 != n) {
      r = v[p1].first;
      ++p1;
      r = max(r, -v[p2].first);
      ++p2;
    } else if (p1 != n) {
      r = v[p1].first;
      ++p1;
    } else {
      r = -v[p2].first;
      ++p2;
    }
  }
  int ans = 0;
  p1 = 0, p2 = 0;
  for (int i = 0; i < n; ++i) {
    while (p1 != n && v[p1].first < -r) ++p1;
    while (p2 != n && v[p2].first <= r) ++p2;
    if (p1 == n && p2 == n) break;
    if (p1 != n && p2 != n) {
      ans += v[p1].second;
      r = v[p1].first;
      ++p1;
      ans += v[p2].second;
      r = max(r, -v[p2].first);
      ++p2;
    } else if (p1 != n) {
      ans += v[p1].second;
      r = v[p1].first;
      ++p1;
    } else {
      ans += v[p2].second;
      r = -v[p2].first;
      ++p2;
    }
  }
  cout << ans << endl;
  return 0;
}
