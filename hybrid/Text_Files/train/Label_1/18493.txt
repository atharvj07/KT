#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  while (true) {
    bool changed = false;
    for (int i = 0; i < s.size() - 1; i++) {
      if (s[i] == 'y' && s[i + 1] == 'x') {
        swap(s[i], s[i + 1]);
        changed = true;
        break;
      } else if (s[i] == 'x' && s[i + 1] == 'y') {
        s.erase(i, 2);
        changed = true;
        break;
      }
    }
    if (!changed) break;
  }
  cout << s << endl;
  return 0;
}
