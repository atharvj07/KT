#include <bits/stdc++.h>
using namespace std;
typedef pair<long long, long long> pi;
#define fi first
#define se second
const long long INF = 2e18;

int main() {
    long long t;
    cin >> t;
    while (t--) {
        long long n, m, a, b;
        cin >> n >> m >> a >> b;
        vector<long long> s(m);
        for (long long i = 0; i < m; i++) cin >> s[i];
        sort(s.begin(), s.end());
        priority_queue<pi, vector<pi>, greater<pi>> pq;
        pq.push({-1, a - 1});
        pq.push({-1, a + 1});
        long long ans = 0;
        while (!pq.empty()) {
            long long p = pq.top().fi, x = pq.top().se, t = 0;
            pq.pop();
            if (x == b) break;
            if (!p) ans++;
            while (!pq.empty() && pq.top().fi <= t) {
                p = pq.top().fi, x = pq.top().se;
                pq.pop();
                if (!p || x == b) break;
                if (!p) ans++;
            }
            if (s.size() && s[0] <= t + 1 && (p || !pq.size() || pq.top().fi > t + 1)) {
                pq.push({t + s[0], x});
                s.erase(s.begin());
            }
            if (x + 1 <= n) pq.push({p + 1, x + 1});
            if (x - 1 >= 1) pq.push({p + 1, x - 1});
            t++;
        }
        cout << ans << "\n";
    }
}
