mod = 10**9 + 7
for _ in range(int(input())):
    n, m, k = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort()
    cnt = [0] * n
    for i in a:
        cnt[i - 1] += 1
    if m == 1 or k == 0:
        ans = 1
        for i in cnt:
            ans = ans * (i + 1) % mod
        print(ans - 1)
    else:
        l, r = 0, k
        ans = 0
        while r < n:
            cur = 0
            for i in range(l, r + 1):
                cur += cnt[i] * (cnt[i] - 1) // 2
                cur %= mod
            ans += cur * (cur + 1) // 2 * cnt[r + 1] % mod
            l += 1
            r += 1
        print(ans)
