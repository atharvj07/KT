import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class ElectricalGrid {

    static class Node {
        int id;
        List<Node> children;

        public Node(int id) {
            this.id = id;
            this.children = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int a = scanner.nextInt();
        Node mainGridRoot = buildTree(scanner, a);
        int[] mainGridConnections = new int[n];
        for (int i = 0; i < n; i++) {
            mainGridConnections[i] = scanner.nextInt();
        }
        int b = scanner.nextInt();
        Node reserveGridRoot = buildTree(scanner, b);
        int[] reserveGridConnections = new int[n];
        for (int i = 0; i < n; i++) {
            reserveGridConnections[i] = scanner.nextInt();
        }

        int maxCut = 0;
        for (int i = 0; i < n; i++) {
            // Try cutting the wire from the device to the main grid
            int cutFromMain = canCut(mainGridRoot, mainGridConnections[i], reserveGridRoot, reserveGridConnections[i]);
            // Try cutting the wire from the device to the reserve grid
            int cutFromReserve = canCut(reserveGridRoot, reserveGridConnections[i], mainGridRoot, mainGridConnections[i]);
            // Choose the option that cuts more wires
            maxCut = Math.max(maxCut, Math.max(cutFromMain, cutFromReserve));
        }

        System.out.println(maxCut);
    }

    // Builds a tree from the given input
    private static Node buildTree(Scanner scanner, int nodesCount) {
        Node root = new Node(1);
        for (int i = 1; i < nodesCount; i++) {
            int parent = scanner.nextInt();
            Node child = new Node(i + 1);
            root.children.add(child);
            buildTreeRecursive(root, child, parent);
        }
        return root;
    }

    // Recursive helper for building the tree
    private static void buildTreeRecursive(Node parent, Node child, int targetParent) {
        if (parent.id == targetParent) {
            return;
        }
        for (Node node : parent.children) {
            buildTreeRecursive(node, child, targetParent);
        }
    }

    // Checks if a wire can be cut from the given node in the main grid to the given node in the reserve grid
    // and returns the number of wires that can be cut
    private static int canCut(Node mainGridRoot, int mainGridNode, Node reserveGridRoot, int reserveGridNode) {
        int mainPathLength = findPathLength(mainGridRoot, mainGridNode);
        int reservePathLength = findPathLength(reserveGridRoot, reserveGridNode);
        return Math.min(mainPathLength - 1, reservePathLength - 1);
    }

    // Finds the length of the path from the root of the tree to the given node
    private static int findPathLength(Node root, int targetNode) {
        if (root.id == targetNode) {
            return 1;
        }
        for (Node child : root.children) {
            int pathLength = findPathLength(child, targetNode);
            if (pathLength > 0) {
                return pathLength + 1;
            }
        }
        return 0;
    }
}
