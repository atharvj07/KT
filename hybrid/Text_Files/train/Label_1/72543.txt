t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    found = False
    for i in range(n):
        for j in range(i+2, n):
            if a[i] == a[j]:
                found = True
                break
        if found:
            break
    if found:
        print("YES")
    else:
        print("NO")
