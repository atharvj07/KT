#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
#define int long long
#define fast ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL)
int32_t main() {
    fast;
    int n, x; cin >> n >> x;
    vector<int> colors(n);
    for (int i = 0; i < n; i++) cin >> colors[i];
    int cur = 0, ans;
    map<int, int> mp;
    set<int> col;
    for (int i = 0; i < n; i++) {
        if (i == n - 1 || colors[i] != colors[i + 1]) {
            mp[colors[i]] += cur;
            cur = 0;
        }
        cur += (colors[i] == x ? -1 : 1);
        if (cur <= 0) col.insert(colors[i]);
    }
    for (int i = 1; i <= 1e6; i++) {
        if (col.find(i) == col.end()) {
            ans = i;
            break;
        }
    }
    if (mp[ans] != 0) ans = -1;
    cout << ans << endl;
}
