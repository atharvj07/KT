import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), g = sc.nextInt(), r = sc.nextInt();
        long[] l = new long[n + 2];
        for (int i = 1; i <= n + 1; i++) {
            l[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        long[] t = new long[q];
        for (int i = 0; i < q; i++) {
            t[i] = sc.nextInt();
        }
        for (long ti : t) {
            long time = l[1] + ti;
            for (int i = 2; i <= n + 1; i++) {
                long tmp = (time + g) / (g + r) * (g + r);
                if (tmp + g >= time) {
                    time = tmp;
                }
                time += l[i];
            }
            System.out.println(time);
        }
    }
}
