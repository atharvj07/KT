#include <cstdio>
#include <vector>
using namespace std;
const int MXN = 2e5+5, MXE = 4e5+5;

int n, t, k, ct, dx, a[MXN], b[MXN], st[MXE], en[MXE];
bool vis[MXN], mk[MXN];
vector<int> g[MXN];

void dfs(int u, int p)
{
	if (vis[u]) return;
	vis[u] = true;
	if (g[u].size() == 1) mk[u] = true;
	for (int i=0; i<(int)g[u].size(); i++)
		if (g[u][i] != p)
			dfs(g[u][i], u);
}

int main()
{
	scanf("%d %d %d", &n, &t, &k);
	for (int i=0; i<t; i++)
		scanf("%d", &a[i]);
	for (int i=0; i<n; i++)
	{
		scanf("%d", &b[i]);
		if (b[i]) ct++;
		else dx++;
	}
	if (ct != k || dx != n-1-k) return (puts("-1"), 0);
	int l=1, r=t, cen=0;
	while (l<=r)
	{
		int mid = l+(r-l)/2;
		cen = 0;
		for (int i=0; i<mid; i++) cen += a[i];
		if (cen <= n-1-k) r = mid-1;
		else l = mid+1;
	}
	for (int i=0; i<cen; i++)
	{
		for (int j=0; j<a[i]; j++)
		{
			g[i+2].push_back(n-ct);
			g[n-ct].push_back(i+2);
			st[dx] = i+2, en[dx] = n-ct, dx--;
		}
		ct--;
	}
	for (int i=0; i<n-1-dx-k; i++)
	{
		g[i+2].push_back(n-k+i);
		g[n-k+i].push_back(i+2);
		st[dx] = i+2, en[dx] = n-k+i, dx--;
	}
	for (int i=0; i<k; i++)
	{
		for (int j=0; j<n; j++)
			if (!vis[j] && !mk[j])
			{
				g[i+2].push_back(j+1);
				g[j+1].push_back(i+2);
				st[dx] = i+2, en[dx] = j+1, dx--;
				dfs(j+1, i+2);
			}
	}
	printf("%d\n", n);
	for (int i=0; i<n-1; i++)
		printf("%d %d\n", st[i], en[i]);
	return 0;
}
