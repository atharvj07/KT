#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m, q;
  cin >> n >> m;
  vector<int> a(n + 1), b(m + 1), ans(q);
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= m; i++) cin >> b[i];
  cin >> q;
  for (int i = 0; i < q; i++) {
    int tp, x, y;
    cin >> tp >> x >> y;
    if (tp == 1) {
      a[x] = y;
    } else {
      b[x] = y;
    }
    vector<int> bb = b;
    sort(bb.begin() + 1, bb.end());
    sort(a.begin(), a.end());
    int j = 1;
    for (int i = 1; i <= m; i++) {
      if (j <= n && a[j] <= bb[i]) j++;
    }
    if (j > n) ans[i] = -1;
    else ans[i] = a[j];
  }
  for (int i = 0; i < q; i++) { cout << ans[i] << endl; }
  return 0;
}
