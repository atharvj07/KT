import java.util.*;
import java.math.BigInteger;

public class Main {

    private static final int MOD = 1_000_000_007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x = in.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = in.nextInt();
        }
        long[][] dp = new long[x + 1][n];
        dp[0][0] = 1;
        for (int i = 1; i <= x; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if (i - d[k] >= 0) {
                        dp[i][j] = (dp[i][j] + dp[i - d[k]][k]) % MOD;
                    }
                }
            }
        }
        long ans = 0;
        for (int i = 0; i <= x; i++) {
            ans = (ans + dp[i][0]) % MOD;
        }
        System.out.println(ans);
    }
}
