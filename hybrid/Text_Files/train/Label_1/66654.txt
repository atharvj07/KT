from collections import deque

def dfs(i, p):
    if i == n:
        yield p
    else:
        for j in range(n):
            if used[j] or next[i] != -1 and next[i] <= j + 1:
                continue
            used[j] = True
            p.append(j + 1)
            for q in dfs(i + 1, p):
                yield q
            used[j] = False
            p.pop()

t = int(input())
for _ in range(t):
    n = int(input())
    next = list(map(int, input().split()))
    used = [False] * n
    for p in dfs(0, []):
        print(*p)
        break
    else:
        print(-1)
