import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int max = 0;
        for (int i = 0; i < n; i++) {
            max = Math.max(max, a[i]);
        }
        int k = max;
        while (true) {
            int awruk = 0;
            int elodreip = 0;
            for (int i = 0; i < n; i++) {
                awruk += k - a[i];
                elodreip += a[i];
            }
            if (awruk > elodreip) {
                break;
            }
            k++;
        }
        System.out.println(k);
    }
}
