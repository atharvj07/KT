#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
const int maxn = 2e5 + 5;

ll a[maxn], b[maxn];

ll f(ll n, ll m) {
    ll ans = 0;
    ll sm = 0;
    sort(a, a + n);
    sort(b, b + m);
    for (ll i = n - 1, j = m - 1; i >= 0 && j >= 0; i--) {
        if (a[i] > b[j]) {
            sm += a[i];
            if (sm > b[j]) {
                sm -= b[j];
                ans += sm;
                sm = 0;
            }
        } else {
            break;
        }
        j--;
    }
    sort(a, a + n, greater<ll>());
    sort(b, b + m, greater<ll>());
    for (ll i = n - 1, j = m - 1; i >= 0 && j >= 0; i--) {
        if (a[i] < b[j]) {
            if (sm > 0) {
                sm -= b[j];
                ans += b[j];
                j--;
                if (sm == 0) j++;
            } else {
                sm += a[i];
                if (sm > 0) ans += sm, sm = 0;
            }
        } else {
            break;
        }
    }
    return ans + sm;
}
int main() {
    ll n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < m; i++) cin >> b[i];
    cout << f(n, m);
}
