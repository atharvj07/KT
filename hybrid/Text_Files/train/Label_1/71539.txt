import math

def rotate(x, y, angle):
    radians = math.radians(angle)
    cos_val = math.cos(radians)
    sin_val = math.sin(radians)
    x_new = x * cos_val + y * sin_val
    y_new = -x * sin_val + y * cos_val
    return x_new, y_new

n, m = map(int, input().split())
x, y = n, 0
angle = 0

for _ in range(m):
    op_type, segment, value = map(int, input().split())
    if op_type == 1:
        if segment == n:
            x += value * math.cos(math.radians(angle))
            y += value * math.sin(math.radians(angle))
        else:
            x += value * math.cos(math.radians(angle)) * (n - segment) / n
            y += value * math.sin(math.radians(angle)) * (n - segment) / n
    else:
        angle += value
        angle %= 360
    print(f"{x:0.10f} {y:0.10f}")
