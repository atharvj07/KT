#include <iostream>
#include <vector>
#include <queue>
#include <tuple>
#include <cstring>
using namespace std;
typedef long long ll;

const ll n = 8;
ll dx[] = {-2, -2, 2, 2};
ll dy[] = {2, -2, -2, 2};

int main() {
  ll t;
  while (cin >> t) {
    while (t--) {
      vector<vector<char>> m(n, vector<char>(n));
      for (ll x = 0; x < n; x++) {
        for (ll y = 0; y < n; y++) cin >> m[x][y];
      }
      ll sx = 0, sy = 0, sxx = 0, syy = 0;
      for (ll x = 0; x < n; x++) {
        for (ll y = 0; y < n; y++) {
          if (m[x][y] == 'K') {
            if (!sx && !sy) {
              sx = x;
              sy = y;
            } else {
              sxx = x;
              syy = y;
            }
          }
        }
      }
      vector<vector<vector<bool>>> vis(n, vector<vector<bool>>(n, vector<bool>(2)));
      queue<tuple<ll, ll, ll>> q;
      q.push(make_tuple(sx, sy, 0));
      bool f = false;
      while (!q.empty()) {
        ll x, y, k;
        tie(x, y, k) = q.front();
        q.pop();
        if (x == sxx && y == syy && k) {
          f = true;
          break;
        }
        if (vis[x][y][k]) continue;
        vis[x][y][k] = true;
        for (ll i = 0; i < 4; i++) {
          ll nx = x + dx[i], ny = y + dy[i];
          if (nx >= 0 && nx < n && ny >= 0 && ny < n && m[nx][ny] != '#')
            q.push(make_tuple(nx, ny, k ^ 1));
        }
      }
      cout << (f ? "YES" : "NO") << endl;
    }
  }
  return 0;
}
