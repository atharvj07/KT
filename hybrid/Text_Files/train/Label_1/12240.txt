#include <iostream>
#include <vector>
using namespace std;

int main() {
  int s;
  cin >> s;
  vector<string> v = {"", "one", "two", "three", "four", "five", "six", "seven",
                      "eight", "nine", "ten", "eleven", "twelve", "thirteen",
                      "fourteen", "fifteen", "sixteen", "seventeen", "eighteen",
                      "nineteen", "twenty"};
  vector<string> w = {"", "", "twenty", "thirty", "forty", "fifty", "sixty",
                      "seventy", "eighty", "ninety"};
  if (s == 0) {
    cout << "zero";
  } else if (s < 20) {
    cout << v[s];
  } else {
    cout << w[s / 10] << "-" << v[s % 10];
  }
  return 0;
}
