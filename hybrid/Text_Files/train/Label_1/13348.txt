#include<bits/stdc++.h>
using namespace std;
#define p pair<long long,long long>
vector<p>a;
set<long long>check;
int main()
{
    long long n,l,x,y,i;
    cin>>n;
    for(i=1; i<=n; i++)
    {
        cin>>l;
        a.push_back({l,i});
    }
    for(i=1; i<=n; i++)
    {
        cin>>l;
        if(l>a[i-1].first)
            return cout<<"NO",0;
        a[i-1].first-=l;
        if(a[i-1].second!=i)
            check.insert(a[i-1].second);
    }
    for(i=1; i<=n-1; i++)
    {
        cin>>x>>y;
        if(check.find(y)!=check.end())
            return cout<<"NO",0;
        swap(a[x-1],a[y-1]);
        check.insert(y);
    }
    cout<<"YES";
}
