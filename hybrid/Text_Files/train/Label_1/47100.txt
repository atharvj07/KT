import java.util.*;

class Main {
    static boolean[] visited;
    static int[][] relations;
    static int[] rabbitIds;
    static int numOfRabbits;
    static StringBuilder output = new StringBuilder();

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        numOfRabbits = input.nextInt();
        int numOfEvents = input.nextInt();
        visited = new boolean[numOfRabbits];
        relations = new int[numOfRabbits][numOfRabbits];
        input.nextLine();

        for (int i = 0; i < numOfEvents; i++) {
            String[] event = input.nextLine().split(" ");
            if (event[0].equals("1")) {
                relations[Integer.parseInt(event[1])][Integer.parseInt(event[2])] = 1;
                relations[Integer.parseInt(event[2])][Integer.parseInt(event[1])] = 1;
            } else if (event[0].equals("2")) {
                relations[Integer.parseInt(event[1])][Integer.parseInt(event[2])] = 0;
                relations[Integer.parseInt(event[2])][Integer.parseInt(event[1])] = 0;
            } else if (event[0].equals("3")) {
                Arrays.fill(visited, false);
                rabbitIds = new int[numOfRabbits];
                int rabbitId = 1;

                for (int j = 0; j < numOfRabbits; j++) {
                    if (!visited[j]) {
                        dfs(j, rabbitId++);
                    }
                }

                if (rabbitIds[Integer.parseInt(event[1])] == rabbitIds[Integer.parseInt(event[2])]) {
                    output.append("YES\n");
                } else {
                    output.append("NO\n");
                }
            }
        }

        System.out.print(output);
    }

    static void dfs(int node, int id) {
        visited[node] = true;
        rabbitIds[node] = id;

        for (int i = 0; i < numOfRabbits; i++) {
            if (relations[node][i] == 1 && !visited[i]) {
                dfs(i, id);
            }
        }
    }
}
