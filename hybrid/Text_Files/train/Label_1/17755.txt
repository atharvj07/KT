#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

const int N = 2e5 + 5;
int n, m, x, k, y, a[N], b[N], p[N], pos[N], mana;

int main() {
    cin >> n >> m >> x >> k >> y;
    for (int i = 1; i <= n; i++) cin >> a[i], p[a[i]] = i;
    for (int i = 1; i <= m; i++) cin >> b[i], pos[b[i]] = i;
    for (int i = 1; i <= n; i++) {
        mana += (pos[a[i]] > pos[b[i]] ? x : 0);
        if (pos[a[i]] > pos[b[i]] - k) {
            if (pos[a[i]] != pos[b[i]] - 1) return cout << -1, 0;
            if (a[i] < b[i]) mana += y;
        } else
            mana += y;
    }
    cout << mana << endl;
}
