#include <iostream>
#include <algorithm>
#include <cmath>

using namespace std;

int main() {
    int n, x;
    cin >> n;
    int a[n + 1];
    for (int i = 1; i <= n; ++i) {
        cin >> x;
        if (sqrt(x) == int(sqrt(x))) a[i] = -1;
        else a[i] = x;
    }
    sort(a + 1, a + n + 1);
    cout << a[n];
    return 0;
}
