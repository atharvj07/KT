import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    String s = sc.next();
    int[][] dp = new int[n + 1][n + 1];
    for (int i = n - 1; i >= 0; i--) {
      for (int j = i; j < n; j++) {
        if (i == j) {
          dp[i][j] = a[i] >= 0 ? a[i] : 0;
        } else if (s.charAt(i) == s.charAt(j)) {
          dp[i][j] = Math.max(a[i] + dp[i + 1][j - 1], Math.max(dp[i][j - 1], dp[i + 1][j]));
        } else {
          dp[i][j] = Math.max(dp[i][j - 1], dp[i + 1][j]);
        }
      }
    }
    System.out.println(dp[0][n - 1]);
  }
}
