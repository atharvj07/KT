#include <bits/stdc++.h>

using namespace std;

const int MAX_N = 1e5+5;
int root, n, k;
vector<int> adj[MAX_N], colors;
vector<bool> visited;
int a[MAX_N];

void dfs(int u, int c) {
    if (visited[u]) return;
    visited[u] = true;
    colors[u] = c;
    for (auto v : adj[u]) {
        if (!visited[v])
            dfs(v, c);
    }
}

bool solve() {
    visited.clear();
    visited.resize(n+1, false);
    colors.clear();
    colors.resize(n+1, -1);
    
    int c = 0;
    for (int i = 1; i <= n; i++) {
        if (!visited[i]) {
            dfs(i, c++);
            if (c > 1 || c == k)
                return false;
        }
    }
    return true;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> k;
        for (int i = 1; i <= n; i++) cin >> a[i];
        for (int i = 1; i < n; i++) {
            int u, v;
            cin >> u >> v;
            adj[u].push_back(v);
            adj[v].push_back(u);
        }
        if (solve())
            cout << "YES\n";
        else
            cout << "NO\n";
        for (int i = 1; i <= n; i++)
            adj[i].clear();
    }
    return 0;
}
