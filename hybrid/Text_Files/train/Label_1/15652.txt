#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int a, b, c, d;
    cin >> a >> b >> c >> d;

    // Check if it's impossible
    if (a + b + c + d < 3 || abs(a - c) > 1 || abs(b - d) > 1) {
        cout << "NO" << endl;
        return 0;
    }

    // Construct the sequence
    cout << "YES" << endl;
    int seq[a + b + c + d];
    int i = 0;
    
    while (a-- > 0) seq[i++] = 0;
    while (b-- > 0) seq[i++] = 1;
    while (c-- > 0) seq[i++] = 2;
    while (d-- > 0) seq[i++] = 3;

    // Sort the sequence to make it beautiful
    sort(seq, seq + i);

    // Print the sequence
    for (int j = 0; j < i; j++) {
        cout << seq[j] << ' ';
    }
    cout << endl;


    return 0;
}
