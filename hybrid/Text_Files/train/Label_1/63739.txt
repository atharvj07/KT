n,q=map(int,input().split())
a=list(map(int,input().split()))
cnt=0
for i in range(1,n):
    if a[i]!=a[i-1]:cnt+=1
print(cnt)
for i in range(q):
    pos,val=map(int,input().split())
    pos-=1
    if pos==0 or pos==n-1:
        print(cnt)
        continue
    if a[pos-1]==a[pos+1] and a[pos]!=val:
        cnt+=1
    elif a[pos-1]!=a[pos+1] and a[pos]==val:
        cnt-=1
    a[pos]=val
    print(cnt)
