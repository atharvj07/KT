#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int M=51;
int n,a[M][M],p[M];
bool vis[M];

int main(){
    cin>>n;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            cin>>a[i][j];
        }
    }
    for(int i=1;i<=n;i++){
        p[i]=0;
        vis[i]=0;
    }
    for(int i=1;i<=n;i++){
        int v=-1;
        for(int j=1;j<=n;j++){
            if(a[i][j]>0 && p[i]==0 && p[j]==0 && !vis[j]){
                v=j;
            }
        }
        vis[v]=1;
        p[v]=i;
    }
    for(int i=1;i<=n;i++){
        if(p[i]==0){
            for(int j=1;j<=n;j++){
                if(a[i][j]>0){
                    p[i]=j;
                    vis[j]=1;
                }
            }
        }
    }
    for(int i=1;i<=n;i++){
        cout<<p[i]<<" ";
    }
    cout<<endl;
    return 0;
}
