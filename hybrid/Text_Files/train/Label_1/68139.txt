import sys
input = sys.stdin.readline

n, q = map(int, input().split())
a = list(map(int, input().split()))
lisp = [0] * n
for i in range(n - 1):
    lisp[i] = abs(a[i + 1] - a[i])
for j in range(1, n): lisp[j] = max(lisp[j], lisp[j - 1])
for i in range(q):
    l, r = map(int, input().split())
    l, r = l - 1, r - 1
    if l == 0: ans = lisp[r]
    else: ans = lisp[r] - lisp[l - 1]
    print(ans)
