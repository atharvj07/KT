#include <cstdio>

int main() {
    int t;
    scanf("%d", &t);
    while (t--) {
        int l, r, q;
        bool ok = true;
        scanf("%d %d %d", &l, &r, &q);
        if (l > r || l > q || q - r > 1 || q < 0 || (l == 0 && r == 0 && q != 0))
            ok = false;
        puts(ok ? "YES" : "NO");
    }
    return 0;
}
