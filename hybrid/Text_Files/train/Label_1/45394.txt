import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n1 = sc.nextInt();
        int m1 = sc.nextInt();
        int[][] a = new int[n1][m1];
        for (int i = 0; i < n1; i++) {
            String s = sc.next();
            for (int j = 0; j < m1; j++) {
                a[i][j] = s.charAt(j) - '0';
            }
        }
        int n2 = sc.nextInt();
        int m2 = sc.nextInt();
        int[][] b = new int[n2][m2];
        for (int i = 0; i < n2; i++) {
            String s = sc.next();
            for (int j = 0; j < m2; j++) {
                b[i][j] = s.charAt(j) - '0';
            }
        }
        int ansX = 0, ansY = 0;
        int maxOverlap = 0;
        for (int x = -109; x <= 109; x++) {
            for (int y = -109; y <= 109; y++) {
                int overlap = 0;
                for (int i = 0; i < n1; i++) {
                    for (int j = 0; j < m1; j++) {
                        int i2 = i + x, j2 = j + y;
                        if (i2 >= 0 && i2 < n2 && j2 >= 0 && j2 < m2) {
                            overlap += a[i][j] * b[i2][j2];
                        }
                    }
                }
                if (maxOverlap < overlap) {
                    maxOverlap = overlap;
                    ansX = x;
                    ansY = y;
                }
            }
        }
        System.out.println(ansX + " " + ansY);
    }
}
