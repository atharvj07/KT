import queue

n = int(input())
tasks = [list(map(int, input().split())) for _ in range(n)]
q = queue.Queue()
time_last = max_size = 0

for t, c in tasks:
    while not q.empty() and q.queue[0] + 1 < t:
        q.get()
    q.put(t + c)
    max_size = max(max_size, q.qsize())

if q:
    time_last = q.queue[-1]

print(time_last, max_size)
