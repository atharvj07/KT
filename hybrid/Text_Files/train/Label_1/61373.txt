n, m, k = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]

score = 0
replacements = 0

for j in range(m):
    for i in range(n):
        if a[i][j] == 1:
            ones_count = sum(a[l][j] for l in range(i, min(i + k, n)))
            if ones_count < k:
                replacements += 1
                score += ones_count - 1
            else:
                score += k
            break

print(score, replacements)
