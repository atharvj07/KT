import java.util.Arrays;
import java.util.Scanner;

class Main {
    static int maxH;
    static int maxW;

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int a = s.nextInt();
        int b = s.nextInt();
        int h = s.nextInt(), w = s.nextInt(), n = s.nextInt();
        maxH = 0;
        maxW = 0;
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = s.nextInt();
        }
        Arrays.sort(arr);
        if (h >= a && w >= b) {
            System.out.println(0);
            return;
        }
        int ans = -1;
        int p = n - 1;
        while (p >= 0) {
            if (w >= b) {
                ans = p + 1;
                break;
            }
            int height = 0;
            for (int i = p; i >= 0; i--) {
                if (height >= a) {
                    ans = p - i + 1;
                    break;
                }
                if (Math.abs(height - (h * arr[i])) < Math.abs(height - maxH)) {
                    maxH = h * arr[i];
                    height = maxH;
                }
            }
            p--;
        }
        if (ans == -1) {
            if (w < b) {
                p = n - 1;
                while (p >= 0) {
                    if (h >= a) {
                        ans = p + 1;
                        break;
                    }
                    int width = 0;
                    for (int i = p; i >= 0; i--) {
                        if (width >= b) {
                            ans = p - i + 1;
                            break;
                        }
                        if (Math.abs(width - (w * arr[i])) < Math.abs(width - maxW)) {
                            maxW = w * arr[i];
                            width = maxW;
                        }
                    }
                    p--;
                }
            }
        }
        System.out.println(ans);
    }
}
