#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, a;
  cin >> n >> a;
  vector<int> t(n);
  for (int i = 0; i < n; i++) { cin >> t[i]; }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (i == a - 1) {
      continue;
    }
    if (t[a - 1] >= (n - 1) * t[i]) { ans++; }
  }
  cout << ans << endl;
  return 0;
}
