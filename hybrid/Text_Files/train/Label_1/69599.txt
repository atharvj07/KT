import sys

# read the number of bombs
n = int(input())

# read the coordinates of the bombs
bombs = []
for _ in range(n):
    x, y = map(int, input().split())
    bombs.append((x, y))

# sort the bombs by their distance to the origin
bombs.sort(key=lambda bomb: bomb[0] ** 2 + bomb[1] ** 2)

# the operations to be performed
operations = []

# initialize the current position of the robot
x, y = 0, 0

# destroy all the bombs
while bombs:
    # get the closest bomb
    bomb = bombs.pop(0)

    # move to the bomb
    if bomb[0] > x:
        operations.extend([(1, bomb[0] - x, "R")])
        x = bomb[0]
    elif bomb[0] < x:
        operations.extend([(1, x - bomb[0], "L")])
        x = bomb[0]
    if bomb[1] > y:
        operations.extend([(1, bomb[1] - y, "U")])
        y = bomb[1]
    elif bomb[1] < y:
        operations.extend([(1, y - bomb[1], "D")])
        y = bomb[1]

    # pick up the bomb
    operations.append((2,))

    # return to the origin
    if x != 0 or y != 0:
        operations.extend([(1, abs(x), "L" if x < 0 else "R")])
        operations.extend([(1, abs(y), "D" if y < 0 else "U")])
        x, y = 0, 0

    # destroy the bomb
    operations.append((3,))

# output the number of operations and the operations
print(len(operations))
for operation in operations:
    print(*operation)
