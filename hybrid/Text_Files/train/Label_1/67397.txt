from sys import stdin
mod = 1000000007
s = stdin.readline().strip()
n = len(s)
dp = [[-1] * 3 for i in range(n + 2)]
dp[n][0] = dp[n][1] = dp[n][2] = 1
for i in range(n - 1, -1, -1):
    if s[i] == '?':
        dp[i][0] = (dp[i + 1][1] + dp[i + 1][0]) % mod
        dp[i][1] = (dp[i + 1][1] + dp[i + 1][2] + dp[i + 1][0]) % mod
        dp[i][2] = (dp[i + 1][2] + dp[i + 1][0]) % mod
    elif s[i] == '0':
        dp[i][0] = dp[i + 1][0]
        dp[i][1] = 0
        dp[i][2] = 0
    elif s[i] == '1':
        dp[i][0] = 0
        dp[i][1] = dp[i + 1][0] + dp[i + 1][1]
        dp[i][2] = 0
    elif s[i] == '2':
        dp[i][0] = 0
        dp[i][1] = dp[i + 1][0] + dp[i + 1][1] + dp[i + 1][2]
        dp[i][2] = 0
print(dp[0][0])
