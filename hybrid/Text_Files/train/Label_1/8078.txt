#include <bits/stdc++.h>
using namespace std;
const int N = 2050;
char s[N], p[N];
int n, m, f[N][N];
void solve() {
    cin >> s >> p;
    n = strlen(s);
    m = strlen(p);
    memset(f, 0, sizeof f);
    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j + i < n; j++) {
            f[i][j] = max(f[i + 1][j], f[i][j + 1]);
            if (s[i + j] == p[j]) f[i][j] = max(f[i][j], (j + m <= n ? f[i + m][j + 1] + 1 : 0));
        }
    }
    for (int i = 0; i <= n; i++) cout << f[0][i] << " ";
    cout << "\n";
}
int main() {
    solve();
    return 0;
}
