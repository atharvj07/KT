#include<bits/stdc++.h>
#include <ext/pb_ds/assoc_container.hpp>
using namespace __gnu_pbds;
using namespace std;
#define ar array
#define ll long long
gp_hash_table<ll, ll> dp;
int32_t main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    ll q; cin >> q;
    while(q--) {
        ll k; cin >> k;
        if(dp.find(k) != dp.end()) {
            cout << dp[k] << "\n";
            continue;
        }
        ll pos = 0, dgt = 1;
        while(pos + dgt * 9 * dgt < k) {
            pos += dgt * 9 * dgt;
            dgt++;
        }
        ll rem = k - pos - 1;
        ll idx = rem / dgt;
        ll s_pos = idx * dgt + 1;
        ll ans = s_pos + (rem % dgt);
        dp[k] = ans % 10;
        cout << ans % 10 << "\n";
    }
}
