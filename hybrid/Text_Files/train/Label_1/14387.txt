#include<bits/stdc++.h>
using namespace std;
#define reg register
#define il inline
#define fi first
#define se second
#define pb push_back
typedef long long ll;
typedef unsigned long long ull;
typedef long double ld;
typedef pair<int, int> pi;
typedef pair<ll, ll> pll;
const int N = 1e3 + 10;
int n, m, p;
ll s[N];
char G[N][N];
bool vis[N][N];
int dr[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
int dc[8] = {-1, 0, 1, -1, 1, -1, 0, 1};
struct node {
    int x, y, s;
    node() {}
    node(int _x, int _y, int _s):x(_x), y(_y), s(_s) {}
};
queue<node> q;
int main() {
    cin >> n >> m >> p;
    for (reg int i = 1; i <= p; ++i)  cin >> s[i];
    for (reg int i = 1; i <= n; ++i) {
        for (reg int j = 1; j <= m; ++j) {
            cin >> G[i][j];
            if (isdigit(G[i][j])) vis[i][j] = true;
        }
    }
    while (true) {
        int f = false;
        for (reg int i = 1; i <= n; ++i) {
            for (reg int j = 1; j <= m; ++j) {
                if (vis[i][j]) {
                    for (reg int k = 0; k < 8; ++k) {
                        int ni = i + dr[k], nj = j + dc[k];
                        if (ni < 1 || ni > n || nj < 1 || nj > m || vis[ni][nj]) continue;
                        int pi = G[i][j] - '0';
                        if (s[pi] >= abs(ni - i) + abs(nj - j)) {
                            vis[ni][nj] = true;
                            q.push(node(ni, nj, s[pi] - abs(ni - i) - abs(nj - j)));
                            f = true;
                        }
                    }
                }
            }
        }
        if (!f) break;
        while (!q.empty()) {
            node t = q.front();
            q.pop();
            for (reg int k = 0; k < 8; ++k) {
                int ni = t.x + dr[k], nj = t.y + dc[k];
                if (ni < 1 || ni > n || nj < 1 || nj > m || vis[ni][nj]) continue;
                int pi = G[t.x][t.y] - '0';
                if (t.s >= abs(ni - t.x) + abs(nj - t.y)) {
                    vis[ni][nj] = true;
                    q.push(node(ni, nj, t.s - abs(ni - t.x) - abs(nj - t.y)));
                    f = true;
                }
            }
        }
    }
    int ans[N] = {};
    for (reg int i = 1; i <= n; ++i) {
        for (reg int j = 1; j <= m; ++j) {
            if (vis[i][j]) ++ans[G[i][j] - '0'];
        }
    }
    for (reg int i = 1; i <= p; ++i)  cout << ans[i] << " ";
    return 0;
}
