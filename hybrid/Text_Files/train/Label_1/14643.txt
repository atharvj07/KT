#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

bool dfs(int node, int prev, int len, vector<vector<int>> &adj, vector<int> &path, int &ans, int k){
    path.push_back(node);
    if(len > k && node == path[0]){
        ans = len;
        return true;
    }
    for(auto next: adj[node]){
        if(next == prev) continue;
        if(dfs(next, node, len + 1, adj, path, ans, k)) return true;
    }
    path.pop_back();
    return false;
}

int main(){
    int n, m, k; cin >> n >> m >> k;
    vector<vector<int>> adj(n + 1);
    for(int i = 0; i < m; i++){
        int a, b; cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    int ans = 0;
    vector<int> path;
    for(int i = 1; i <= n; i++){
        if(dfs(i, i, 0, adj, path, ans, k)) break;
    }
    cout << ans << endl;
    for(int i = 0; i < ans; i++){
        cout << path[i] << " ";
    }
    cout << endl;
    return 0;
}
