#include <iostream>
#include <vector>
using namespace std;
int n, k, ans, d[55][5005], m;
vector<int> a;
int main() {
  cin >> n >> k;
  a.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    a[i] /= 50;
  }
  d[0][0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = k; j >= 0; j--) {
      for (int x = 0; x <= 1; x++) {
        if (j + a[i] * x <= k && d[i][j]) {
          d[i + 1][j + a[i] * x] = 1;
        }
      }
    }
    if (d[i + 1][k]) {
      m = i + 1;
      break;
    }
  }
  ans = d[m][k];
  if (ans == 0) {
    cout << -1 << endl << 0 << endl;
    return 0;
  }
  for (int i = m, j = k; i >= 1; i--) {
    int f = 0;
    for (int x = 0; x <= 1; x++) {
      if (d[i][j] && d[i - 1][j - x * a[i - 1]] && !f) {
        f = 1;
        ans = ans * 2 % 1000000007;
      }
    }
    if (a[i - 1] > 0) j -= a[i - 1];
  }
  cout << m << endl << ans << endl;
  return 0;
}
