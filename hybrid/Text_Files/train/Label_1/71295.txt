import math
for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    q = int(input())
    for i in range(n):
        l = [1]
        for j in range(n):
            if i == j:
                continue
            g = math.gcd(a[i], a[j])
            l.append(a[i] // g * a[j] // g)
        a[i] = max(l)
    for i in range(q):
        for j in range(n):
            a[j] = max(a[j], max(a))
        print(a.count(a[0]))
