def nearly_full_subsequences(a):
    n = len(a)
    s = sum(a)
    dp = [[0] * (s + 1) for _ in range(n + 1)]
    dp[n][0] = 1
    for i in range(n - 1, -1, -1):
        for j in range(s + 1):
            dp[i][j] = dp[i + 1][j]
            if j >= a[i]:
                dp[i][j] += dp[i + 1][j - a[i]]
    return dp[0][s - 1]

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    print(nearly_full_subsequences(a))
