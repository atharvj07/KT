#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const int mod = 1e9 + 7;

int main() {
  int n;
  cin >> n;
  vector<ll> b(n + 1), a(n + 1), x(n), k(n);
  for (int i = 1; i <= n; i++) cin >> b[i];
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 0; i < n - 1; i++) {
    cin >> x[i] >> k[i];
  }
  for (int i = n - 1; i >= 1; i--) {
    ll d = a[x[i - 1] + 1] - b[x[i - 1] + 1];
    if (d > 0) {
      b[i] -= k[i - 1] * d;
      b[x[i - 1] + 1] += d;
    }
  }
  for (int i = 1; i <= n; i++) {
    if (b[i] < a[i]) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
