#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> sg(n);
  for (int i = 0; i < n; i++) {
    cin >> sg[i].first >> sg[i].second;
  }
  long long total = 0;
  vector<int> res(n);
  for (int i = 0; i < n; i++) {
    int x = max(0, min(sg[i].first + 1, sg[i].second) - sg[i].first);
    if (i > 0 && abs(res[i - 1] + x - sg[i - 1].first - sg[i - 1].second) > 1) {
      cout << -1 << endl;
      return 0;
    }
    total += x;
    res[i] = sg[i].first + x;
  }
  cout << total << endl;
  for (int i = 0; i < n; i++) {
    cout << res[i] << " ";
  }
  cout << endl;
  return 0;
}
