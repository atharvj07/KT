#include<bits/stdc++.h>
#define mod 1000000007
using namespace std;
int n, m, f[2010][2010][2], mul[2010];
int main() {
    cin >> n >> m;
    string s;
    cin >> s;
    f[m][0][0] = 1;
    mul[0] = 1;
    for(int i = 1; i <= n; i++) mul[i] = 1LL * mul[i-1] * 2 % mod;
    for(int i = m-1; i >= 0; i--) {
        for(int j = 0; j <= n-m; j++) {
            for(int k = 0; k < 2; k++) {
                int v = k;
                if(s[i] == '(') v^=1;
                f[i][j][k] = (f[i+1][j][k] + 1LL * f[i+1][j+1][v] * mul[j] % mod) % mod;
            }
        }
    }
    cout << f[0][0][0] << endl;
}
