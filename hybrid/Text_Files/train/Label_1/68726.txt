from collections import deque

def solve(n, k, a, b):
    if a > k + 1 and b > k + 1:
        return 'NO'
    q = deque()
    ans = []
    for i in range(n):
        if len(q) == k:
            cur = q.popleft()
            if cur == 'G':
                b -= 1
            else:
                a -= 1
        if a == 0:
            q.append('B')
            ans.append('B')
            b -= 1
        elif b == 0:
            q.append('G')
            ans.append('G')
            a -= 1
        elif a > b:
            q.append('G')
            ans.append('G')
            a -= 1
        else:
            q.append('B')
            ans.append('B')
            b -= 1
    return ''.join(ans)

n, k, a, b = map(int, input().split())
print(solve(n, k, a, b))
