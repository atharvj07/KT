import java.io.*;
import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n + 1];
        int[] id = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = in.nextInt();
            id[p[i]] = i;
        }
        int[] c = new int[n + 1];
        in.close();
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (id[j] < id[i]) {
                    c[i]++;
                    c[j]--;
                }
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (c[i] == 0) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
