t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    unique_elements = set(a)
    if k == 1:
        print(n)
        print(' '.join(map(str, a)))
    elif len(unique_elements) > k:
        print(-1)
    else:
        beautiful_array = list(unique_elements) + [min(unique_elements)] * (k - len(unique_elements))
        m = n * len(beautiful_array)
        result = beautiful_array * n
        print(m)
        print(' '.join(map(str, result)))
