def min_filters_needed(n, m, k, sockets):
    sockets.sort(reverse=True)
    available_sockets = k
    used_filters = 0
    
    for i in range(n):
        if m <= available_sockets:
            return used_filters
        available_sockets += sockets[i] - 1
        used_filters += 1
        
    if m <= available_sockets:
        return used_filters
    else:
        return -1

n, m, k = map(int, input().split())
sockets = list(map(int, input().split()))
print(min_filters_needed(n, m, k, sockets))
