import sys

n, l, r, x = map(int, input().split())
c = sorted([int(c) for c in input().split()])
ans = 0

# Iterate over all possible pairs of difficulties
for i in range(n - 1):
    for j in range(i + 1, n):
        # Check if the pair of difficulties is valid
        if c[j] - c[i] >= x and l <= c[i] + c[j] <= r:
            # Count the number of ways to choose the remaining problems
            k = n - 2
            ans += 2 ** k

print(ans)
