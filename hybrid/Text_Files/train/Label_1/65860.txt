from functools import reduce

mod = 998244353
n, m, al = map(int, input().split())
b = [int(x) for x in input().split()]

ans = pow(al, n, mod)
sb = [0] * 33
for x in b:
    for j in range(32):
        if x & (1 << j):
            sb[j] += 1

for j in range(32):
    x = sb[j]
    y = n - 2 * x
    if x & 1:
        ans -= (pow(al - 1, y, mod) * pow(al, 2 * x - 1, mod)) % mod
    else:
        ans -= (pow(al - 1, y, mod) * pow(al, 2 * x, mod)) % mod
    ans %= mod
print(ans)
