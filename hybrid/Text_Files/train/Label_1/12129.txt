#include<iostream>
#include<vector>
#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,x;
    cin>>n;
    vector<int> a(n),b(n);
    for(int i=0;i<n;i++){
        cin>>x;
        a[i]=x;
        b[x-1]=i;
    }
    int j,c=1;
    for(int i=n-1;i>=0;i--){
        j=b[i];
        while(j>0&&a[j-1]>a[j]){
            swap(a[j-1],a[j]);
            j--;
        }
        b[a[j]-1]=j;
    }
    for(int i=0;i<n;i++){
        cout<<a[i]<<" ";
    }
    cout<<endl;
    return 0;
}
