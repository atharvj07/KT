n, k = map(int, input().split())
sizes = list(map(int, input().split()))

# Calculate the minimum size of a box needed
def min_box_size(n, k, sizes):
    # Single cowbells in each box
    if n <= k:
        return max(sizes)
    
    # Calculate the pairs and single cowbells
    pairs_needed = n - k
    min_size = 0
    for i in range(pairs_needed):
        min_size = max(min_size, sizes[i] + sizes[2 * pairs_needed - i - 1])
    
    if n > 2 * pairs_needed:
        min_size = max(min_size, sizes[2 * pairs_needed])
    
    return min_size

print(min_box_size(n, k, sizes))
