import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.Map;
import java.util.StringTokenizer;

class Main {
    static int[] parent;
    static int[] type;
    static int[] hasSpecial;
    static int[] hasPart;

    private static void init(int n) {
        parent = new int[n + 1];
        type = new int[n + 1];
        hasSpecial = new int[n + 1];
        hasPart = new int[n + 1];
    }

    private static void dfs(int node) {
        if (type[node] == 0) {
            if (hasSpecial[parent[node]] == 0) {
                hasSpecial[parent[node]] = 1;
                hasSpecial[node] = 1;
            }
        } else if (type[node] == 1) {
            if (hasPart[parent[node]] == 0) {
                hasPart[parent[node]] = 1;
                hasPart[node] = 1;
            }
        }

        for (int i = node + 1; i < parent.length; i++) {
            if (parent[i] == node) {
                dfs(i);
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;
        int n = Integer.parseInt(br.readLine());
        init(n);
        for (int i = 1; i <= n; i++) {
            st = new StringTokenizer(br.readLine());
            parent[i] = Integer.parseInt(st.nextToken());
            type[i] = Integer.parseInt(st.nextToken());
        }

        dfs(1);

        int q = Integer.parseInt(br.readLine());
        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());

            if (type == 1) {
                if (hasSpecial[u] == 1 && hasSpecial[v] == 1) {
                    System.out.println("YES");
                } else {
                    System.out.println("NO");
                }
            } else if (type == 2) {
                if (hasPart[u] == 1 && hasPart[v] == 1) {
                    System.out.println("YES");
                } else {
                    System.out.println("NO");
                }
            }
        }
    }
}
