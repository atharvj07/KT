#include<bits/stdc++.h>
using namespace std;
int main() {
  long long n, a[200005], sum = 0;
  cin >> n;
  int flg = 1;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    if (flg)
      sum += a[i];
    else
      sum -= a[i];
    flg = flg ^ 1;
  }
  if (sum < 0)
    sum += 1000000007;
  cout << sum % 1000000007 << endl;
}
