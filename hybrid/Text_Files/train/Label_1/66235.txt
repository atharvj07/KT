import sys

def main():
    t=int(sys.stdin.readline())
    for _ in range(t):
        n,k=map(int,sys.stdin.readline().split())
        l=list(map(int,sys.stdin.readline().split()))
        l.sort()
        r=list(map(int,sys.stdin.readline().split()))
        ans=0
        for i in range(n):
            if l[i]==r[i]:
                ans+=1
                l[i]+=k
            elif l[i]+k>=r[i]:
                ans+=1
        print(ans)

main()
