#include <iostream>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  if (k == 1) {
    cout << -1;
    return 0;
  }
  int deg[n + 1] = {}, adj[n + 1][n + 1] = {};
  for (int i = 0; i < k; i++) {
    int a, b;
    cin >> a >> b;
    adj[a][(a + 1) % n] = 1, adj[(a + n - 1) % n][a] = 1;
    deg[a]++, deg[(a + 1) % n]++;
  }
  for (int i = 1; i <= n; i++)
    if (deg[i] == 1) {
      cout << -1;
      return 0;
    }
  cout << 1;
  return 0;
}
