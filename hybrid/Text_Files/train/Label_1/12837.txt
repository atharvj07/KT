#include <algorithm>
#include <cstdio>
#include <vector>
using namespace std;
typedef long long ll;
vector<int> v;
bool vis[10000001];
int main() {
  int n, i, l, r, ans = 1e9, sz, cnt, I;
  scanf("%d%d", &n, &I);
  I *= 8;
  for (i = 1; i <= n; i++) {
    int x;
    scanf("%d", &x);
    v.push_back(x);
    vis[x] = 1;
  }
  sort(v.begin(), v.end());
  v.resize(unique(v.begin(), v.end()) - v.begin());
  sz = v.size();
  for (l = 0; l < sz; l++) {
    cnt = 0;
    for (r = l; r < sz; r++) {
      while (r + 1 < sz && v[r] + 1 == v[r + 1]) r++, cnt++;
      if (2 * (ll)r - cnt <= I) ans = min(ans, cnt);
      cnt++;
    }
  }
  printf("%d", ans);
  return 0;
}
