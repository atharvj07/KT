for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    if sorted(a) != a or a.count(0) != a.count(1):
        print(-1)
    else:
        p = [1]
        for i in range(n):
            if a[i] == 1:
                p.append(n+1)
            p.append(i+2)
        print(*p)
