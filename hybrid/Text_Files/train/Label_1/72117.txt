l = int(input())
x = 0
overflow = False
for _ in range(l):
    cmd = input().strip()
    if cmd == 'add':
        x += 1
        if x == 2**32:
            overflow = True
    elif cmd.startswith('for '):
        n = int(cmd.split(' ')[1])
        for _ in range(n):
            if overflow:
                break
            cmd = input().strip()
            if cmd == 'add':
                x += 1
                if x == 2**32:
                    overflow = True
            elif cmd.startswith('for '):
                n = int(cmd.split(' ')[1])
                for _ in range(n):
                    if overflow:
                        break
                    cmd = input().strip()
                    if cmd == 'add':
                        x += 1
                        if x == 2**32:
                            overflow = True
            elif cmd == 'end':
                break
    elif cmd == 'end':
        break

if overflow:
    print('OVERFLOW!!!')
else:
    print(x)
