#include <vector>
#include <cstdio>
#include <queue>

using namespace std;

int main() {
  int n;
  scanf("%d", &n);
  vector<int> crush(n);
  for (int i = 0; i < n; i++) {
    scanf("%d", &crush[i]);
  }
  bool found = false;
  for (int t = 1;; t++) {
    queue<int> q;
    vector<bool> vis(n);
    for (int i = 0; i < n; i++) {
      q.push(i);
      while (!q.empty()) {
        int u = q.front();
        q.pop();
        if (vis[u]) continue;
        vis[u] = true;
        if (u == i) {
          found = true;
          printf("%d\n", t);
          break;
        }
        if (t > 1) q.push(crush[u]);
        else u = crush[u];
      }
      if (found) break;
    }
    if (found) break;
  }
  if (!found) printf("-1\n");
  return 0;
}
