import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        for (int i = 0; i < t; i++) {
            long n = Long.parseLong(br.readLine());
            boolean[] vis = new boolean[(int) n + 1];
            int[] ans = new int[(int) n];
            Arrays.fill(ans, -1);
            ArrayList<Integer> path = new ArrayList<>();
            boolean flag = dfs(n, 1, vis, ans, path);
            if (!flag) {
                System.out.println(-1);
            } else {
                System.out.println(path.size());
                for (Integer x : path) {
                    System.out.print(x + " ");
                }
                System.out.println();
            }
        }
    }

    private static boolean dfs(long n, long m, boolean[] vis, int[] ans, ArrayList<Integer> path) {
        if (m == n) {
            ans[path.size() - 1] = 0;
            return true;
        }
        if (m > n || vis[(int) m]) {
            return false;
        }
        vis[(int) m] = true;
        path.add(0);
        if (dfs(n, m * 2, vis, ans, path)) {
            return true;
        }
        path.remove(path.size() - 1);
        path.add(1);
        if (dfs(n, m * 2 + 1, vis, ans, path)) {
            return true;
        }
        vis[(int) m] = false;
        path.remove(path.size() - 1);
        return false;
    }
}
