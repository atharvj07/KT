import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int area = 0;
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) {
                area += h * (w - x[i]);
            } else {
                area += h * x[i];
            }
        }
        System.out.println(area);
    }
}
