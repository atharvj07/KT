#include <bits/stdc++.h>

using namespace std;

int n, m, a[100], k[100], F[100][100][100];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= m; i++) cin >> k[i];
    for (int i = 0; i <= n; i++)
        for (int j = 0; j <= m; j++)
            for (int cnt = 0; cnt <= n; cnt++) F[i][j][cnt] = -1;
    F[0][0][0] = 0;
    for (int i = 1; i <= n; i++)
        for (int j = 0; j <= m; j++)
            for (int cnt = 0; cnt <= n; cnt++) {
                F[i][j][cnt] = F[i - 1][j][cnt];
                if (a[i] == j) {
                    if (F[i - 1][j - 1][cnt] >= 0) F[i][j][cnt] = max(F[i][j][cnt], F[i - 1][j - 1][cnt] + 1);
                    if (cnt > 0 && F[i - 1][j][cnt - 1] >= 0) F[i][j][cnt] = max(F[i][j][cnt], F[i - 1][j][cnt - 1] + 1);
                }
            }
    bool ok = 1;
    for (int i = 1; i <= m; i++) {
        if (F[n][i][k[i]] < 0) {
            ok = 0;
            break;
        }
    }
    cout << (ok ? "YES" : "NO") << endl;
    return 0;
}
