import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int[] sticks = new int[4];
        for (int i = 0; i < 4; i++) {
            sticks[i] = input.nextInt();
        }
        // Sort the sticks in ascending order
        java.util.Arrays.sort(sticks);
        // Check if it is possible to construct a non-degenerate triangle
        if (sticks[0] + sticks[1] > sticks[2]) {
            System.out.println("TRIANGLE");
        } else if (sticks[0] + sticks[1] == sticks[2]) {
            System.out.println("SEGMENT");
        } else {
            System.out.println("IMPOSSIBLE");
        }
    }
}
