import sys
from math import ceil

input = sys.stdin.readline

t = int(input())

for _ in range(t):
    n = int(input())
    times = [list(map(int, input().split())) for _ in range(n)]
    t = list(map(int, input().split()))
    
    tt = [times[i][0] for i in range(n)]
    t = [times[i][1] for i in range(n)]

    for i in range(n - 1):
        x = ceil((t[i] - tt[i]) / 2)
        tt[i + 1] = max(tt[i] + t[i] + x, tt[i + 1])

    print(tt[n - 1])
