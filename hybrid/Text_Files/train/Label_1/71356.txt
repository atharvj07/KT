from sys import stdin;

n, m = list(map(int, stdin.readline().strip().split()))
p = list(map(int, stdin.readline().strip().split()))
a = [[]]*(m+1)
for i in range(m):
    x, y = list(map(int, stdin.readline().strip().split()))
    a[i] = x, y
ans = (1.0*n*(n-1)) / 4
for j in range(m):
    x, y = a[j]
    if p[x-1] > p[y-1]:
        ans -= 1.0
print(ans)
