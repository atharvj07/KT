#include <bits/stdc++.h>
using namespace std;
struct event { int a, b; };
int main() {
  int n, p, m;
  cin >> n >> p >> m;
  vector<event> e(n + 2);
  e[0].b = -p;
  for (int i = 1; i <= n; ++i) cin >> e[i].a >> e[i].b;
  e[n + 1].a = m + 1;
  int neg = 0;
  long long ans = 0;
  for (int i = 0; i <= n; ++i) {
    neg += e[i].b;
    if (neg < 0) ans += e[i + 1].a - e[i].a;
  }
  cout << ans << endl;
}
