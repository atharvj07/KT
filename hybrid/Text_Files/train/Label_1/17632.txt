#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main() {
    int t; cin >> t;
    while (t--) {
        int n, l, r; cin >> n >> l >> r;
        cout << ((l + r) % 2 ? (l + r) / 2 + n * ((l + r) / 2 - 1) : (l + r - 2) / 2 + n * (l + r - 2)) % (n * (n - 1)) + 1 << endl;
    }
}
