n, m = map(int, input().split())
ji = []
ci = []
for i in range(n):
    t, s = input().split()
    ji.append([t, int(s)])
for i in range(m):
    s = int(input())
    ci.append(s)
ans = 0
ji.sort(key=lambda x: x[1], reverse=True)
ci.sort(reverse=True)
for t, s in ji:
    if not ci:
        break
    if t == 'ATK' and ci[-1] < s:
        ci.pop()
    elif t == 'DEF' and ci[-1] <= s:
        ci.pop()
    else:
        ans += s - ci.pop()
print(ans)
