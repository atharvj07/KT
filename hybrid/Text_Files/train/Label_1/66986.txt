for t in range(int(input())):
    n, T = map(int, input().split())
    a = [*map(int, input().split())]
    a.sort()
    p = [0] * n
    for i in range(n):
        l, r = i, n - 1
        ind = -1
        while l <= r:
            mid = (l + r) // 2
            if a[mid] <= T - a[i]:
                ind = mid
                l = mid + 1
            else:
                r = mid - 1
        if ind != -1:
            p[ind] = 1
    print(*p)
