#include<bits/stdc++.h>
using namespace std;
#define ll long long
pair<ll, ll> arr[200001];
ll ans = 0;

int main(){
    int n;
    cin >> n;
    for(int i = 1; i <= n; i++){
        cin >> arr[i].first;
        arr[i].second = i;
    }
    sort(arr + 1, arr + 1 + n);
    for(int i = 1, j = 1; i <= n; i++){
        while(arr[i].first - arr[j].first > 1 && j <= n) j++;
        if(arr[i].first - arr[j].first > 1){
            continue;
        }
        else{
            ans += arr[i].second - j;
        }
    }
    cout << ans;
    return 0;
}
