#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> boss(n + 1);
    vector<vector<int>> sub(n + 1);
    map<int, int> doc;
    int cnt = 0;
    while (m--) {
        int t, x, y;
        cin >> t >> x;
        if (t == 1) {
            cin >> y;
            sub[y].push_back(x);
            boss[x] = y;
        } else if (t == 2) {
            doc[x] = ++cnt;
        } else {
            cin >> y;
            int p = x;
            while (p) {
                if (doc[p] == y) {
                    cout << "YES" << endl;
                    return 0;
                }
                p = boss[p];
            }
            cout << "NO" << endl;
        }
    }
    return 0;
}
