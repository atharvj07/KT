#include<bits/stdc++.h>

using namespace std;

typedef long long ll;

int main() {
    ll n, a, b, k, ans = 0, ten = 1e9 + 9;
    cin >> n >> a >> b >> k;
    string tmp;
    cin >> tmp;
    map<ll, ll> mp;
    mp.clear();
    ll pw1 = 1, pw2 = 1;
    for (int i = 0; i < k; i++) {
        if (tmp[i] == '+') pw1 = (pw1 * a) % ten, pw2 = (pw2 * b) % ten, ans = (ans + pw1 - pw2 + ten) % ten;
        else pw1 = (pw1 * a) % ten, pw2 = (pw2 * b) % ten, ans = (ans - pw1 + pw2 + ten) % ten;
    }
    ll cycle = (n + 1) / k;
    ans = (ans * cycle) % ten;
    pw1 = 1, pw2 = 1;
    ll rem = (n + 1) % k;
    for (int i = 0; i < rem; i++) {
        if (tmp[i] == '+') pw1 = (pw1 * a) % ten, pw2 = (pw2 * b) % ten, ans = (ans + pw1 - pw2 + ten) % ten;
        else pw1 = (pw1 * a) % ten, pw2 = (pw2 * b) % ten, ans = (ans - pw1 + pw2 + ten) % ten;
    }
    cout << ans << endl;
    return 0;
}
