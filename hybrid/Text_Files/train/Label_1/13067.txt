#include<bits/stdc++.h>
using namespace std;
int n,a[101],f[101];
int main(){
    cin>>n;
    for(int i=1;i<=n;++i)cin>>a[i];
    sort(a+1,a+1+n);
    int ans=1;
    for(int i=1;i<=n;++i)if(a[i]!=a[i-1])f[++ans]=0;
    for(int i=1;i<=n;++i)++f[lower_bound(a+1,a+1+n,a[i])-a];
    cout<<ans<<endl;
    return 0;
}
