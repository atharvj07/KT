#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> v(k);
    for (int i = 0; i < k; i++) { cin >> v[i]; }
    int ans = 0;
    int mx = 0, mn = 1e9;
    for (int i = 0; i < k; i++) {
      ans = max(ans, (v[i] - mn + 1) / 2);
      ans = max(ans, (mx - v[i] + 1) / 2);
      mx = max(mx, v[i]);
      mn = min(mn, v[i]);
    }
    cout << ans << "\n";
  }
  return 0;
}
