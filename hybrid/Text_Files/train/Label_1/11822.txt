#include <bits/stdc++.h>
using namespace std;
int main() {
  int tt;
  cin >> tt;
  while (tt--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    sort(a.begin(), a.end());
    long long ans = 0;
    for (int i = 0; i + 1 < n; ++i) { ans += a[i + 1] - a[i]; }
    cout << ans << endl;
  }
  return 0;
}
