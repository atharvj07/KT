import sys
input = sys.stdin.readline

T = int(input())
for _ in range(T):
    n = int(input())
    a = list(map(int, input().split()))
    l = list(map(int, input().split()))
    unlocked = [i for i in range(n) if l[i] == 0]
    # print(unlocked, a, l)
    a_u = [a[i] for i in unlocked]
    a_u.sort()
    j = 0
    for i in unlocked:
        a[i] = a_u[j]
        j += 1
    print(*a)
