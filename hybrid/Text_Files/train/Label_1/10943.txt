#include <vector>
#include <iostream>

using namespace std;

struct Node {
    int prev, next;
};

int main() {
    int n;
    cin >> n;
    vector<Node> nodes(n + 1);
    for (int i = 1; i <= n; i++) {
        int l, r;
        cin >> l >> r;
        nodes[i].prev = l;
        nodes[i].next = r;
    }

    vector<int> cycle;
    int curr = 1;
    while (curr != 0) {
        cycle.push_back(curr);
        curr = nodes[curr].next;
    }

    int cycle_len = cycle.size();
    int cycle_start = cycle[0];

    int head = 0;
    for (int i = 1; i <= n; i++) {
        if (nodes[i].prev == 0) {
            head = i;
            break;
        }
    }

    if (head != 0) {
        nodes[head].prev = nodes[cycle[cycle_len - 1]].next;
        nodes[nodes[cycle[cycle_len - 1]].next].next = head;
    }

    for (int i = 1; i <= n; i++) {
        cout << nodes[i].prev << ' ' << nodes[i].next << endl;
    }

    return 0;
}
