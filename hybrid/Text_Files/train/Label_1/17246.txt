#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> P;

int main() {
    int n;
    cin >> n;

    double ans = 0;
    P p0 = make_pair(0, 0);
    P p1, p2;

    for (int i = 0; i < n; i++) {
        cin >> p1.first >> p1.second;
        if (i != 0) {
            double p1xabs = abs(p1.first);
            double p1yabs = abs(p1.second);
            double p2xabs = abs(p2.first);
            double p2yabs = abs(p2.second);

            // cross product
            double t1 = p1xabs * p2yabs - p1yabs * p2xabs;
            double t2 = p2xabs * p0.second - p2yabs * p0.first;
            double t3 = p0.first * p1yabs - p0.second * p1xabs;
            ans += t1 + t2 + t3;
        }
        p2 = p1;
    }
    ans = abs(ans);
    ans = ans * 180.0 / 3.14159265359;

    cout << fixed << setprecision(10) << ans << endl;

    return 0;
}
