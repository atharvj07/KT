#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2005;
int n, h, l, r, a[maxn];
bool good(int t) { return l <= t && t <= r; }
int dp[maxn][2];
int solve(int i, bool sl) {
  if (i == n) return 0;
  int &ret = dp[i][sl];
  if (~ret) return ret;
  ret = 0;
  if (!sl) {
    ret = solve(i + 1, true) + good(a[i]);
    if (a[i] > 1) ret = max(ret, solve(i + 1, false) + good(a[i] - 1));
  } else {
    if (a[i] + h - 1 <= r) ret = solve(i + 1, true) + good(a[i] + h - 1);
    if (a[i] + h - 1 <= r && a[i] > 1)
      ret = max(ret, solve(i + 1, false) + good(a[i] + h - 2));
    if (a[i] > 1) ret = max(ret, solve(i + 1, false) + good(a[i] - 1));
  }
  return ret;
}
int main() {
  memset(dp, -1, sizeof dp);
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n >> h >> l >> r;
  for (int i = 0; i < n; i++) cin >> a[i];
  cout << solve(0, false) << "\n";
}
