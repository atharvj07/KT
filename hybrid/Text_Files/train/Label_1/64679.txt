q = int(input())
for _ in range(q):
    l, r = map(int, input().split())
    if l % 2 == 0:
        if r % 2 == 0:
            print((r // 2) - ((l - 1) // 2))
        else:
            print(-((r + 1) // 2) + ((l - 1) // 2))
    else:
        if r % 2 == 0:
            print(-((r // 2) - (l // 2)))
        else:
            print(((r + 1) // 2) - (l // 2))
