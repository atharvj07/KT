#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define mod 1000000007

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.length();
    bool flag = true;
    for (int i = 1; i < n; i++) {
      if (s[i] != s[i - 1]) {
        flag = false;
        break;
      }
    }
    cout << (flag ? "YES" : "NO") << endl;
  }
}
