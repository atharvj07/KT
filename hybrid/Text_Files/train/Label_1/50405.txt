import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), ai[] = new int[n+2];
        for (int i = 1; i <= n; i++) ai[i] = in.nextInt();
        Arrays.sort(ai);
        for (int i = n; i >= 1; i--) System.out.println(ai[i]);
    }
}
