import sys
 
def bfs(adj, n, seq):
    q = [1]
    used = [False] * (n + 1)
    used[1] = True
    i = 0
    while q and i < n:
        v = q.pop(0)
        if seq[i] != v:
            return "No"
        for u in adj[v]:
            if not used[u]:
                q.append(u)
                used[u] = True
        i += 1
    return "Yes"
 
input = sys.stdin.readline
n = int(input())
adj = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)
seq = list(map(int, input().split()))
print(bfs(adj, n, seq))
