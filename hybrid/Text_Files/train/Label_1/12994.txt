#include <bits/stdc++.h>
#include <ext/pb_ds/assoc_container.hpp>
#include <ext/pb_ds/tree_policy.hpp>

using namespace std;
using namespace __gnu_pbds;

using ll = long long;
using ld = long double;

const int mod = 1e9 + 7;

ll dp[55][55][55][2][2];
int cc[55];
int n, p;

ll add(ll a, ll b) {
    return (a + b) % mod;
}

ll mul(ll a, ll b) {
    return (a * b) % mod;
}

ll binpow(ll a, ll b) {
    if (b == 0) return 1;
    if (b == 1) return a;
    if (b % 2) return mul(binpow(a, b - 1), a);
    return binpow(mul(a, a), b / 2);
}

ll solve(int pos, int lst, int rem, bool parity, bool painted) {
    if (pos == n + 1) {
        if (rem == 0 && parity == p) return 1;
        return 0;
    }
    ll &ret = dp[pos][lst][rem][parity][painted];
    if (ret != -1) return ret;
    ret = 0;
    if (cc[pos] != -1) {
        int nparity = parity;
        if (cc[pos] == 0) nparity ^= 1;
        ret = solve(pos + 1, lst, rem, nparity, true);
        return ret;
    }
    ret = add(ret, solve(pos + 1, pos, rem - 1, parity, painted)); // paint black
    ret = add(ret, solve(pos + 1, pos, rem, parity ^ 1, painted)); // paint white
    for (int i = lst + 1; i <= pos; i++) {
        ret = add(ret, solve(pos + 1, i, rem - 1, parity, true)); // add arrow
    }
    return ret;
}

int main() {
    memset(dp, -1, sizeof dp);
    cin >> n >> p;
    for (int i = 1; i <= n; i++)
        cin >> cc[i];
    cout << solve(1, 0, n, false, false) << endl;
    return 0;
}
