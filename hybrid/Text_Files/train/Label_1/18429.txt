#include<bits/stdc++.h>
using namespace std;

typedef long long LL;

const int N = 1005;
const int INF = 1e9;

int n, a[N], b[N], c[N], d[N];
pair<LL, int> p[N];

int main(){
    cin >> n;
    for(int i = 1; i <= n; i ++){
        cin >> a[i] >> b[i] >> c[i] >> d[i];
        p[i].first = a[i] + b[i] + c[i] + d[i];
        p[i].second = i;
    }
    sort(p + 1, p + n + 1);
    reverse(p + 1, p + n + 1);
    for(int i = 1; i <= n; i ++){
        if(p[i].second == 1){
            cout << i << endl;
            return 0;
        }
    }
}
