n, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
b = [0] * (len(a) + 1)
for i in range(k):
    cur = i
    while cur < len(a):
        b[cur] += a[cur]
        cur += k
cur = 0
ans = 0
for i in range(len(a)):
    for j in range(cur, b[i + 1]):
        if j % 100 == i + 1:
            ans += 1
    cur = b[i + 1]
print(ans)
