import java.util.LinkedList;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int K = sc.nextInt();
        List<Integer> dq = new LinkedList<>();
        int[] values = new int[N + 1];
        for (int i = 1; i <= N; i++) {
            values[i] = sc.nextInt();
        }
        int[] pref = new int[N + 1];
        pref[0] = values[1];
        for (int i = 1; i < N; i++) {
            pref[i] = pref[i - 1] + values[i + 1];
        }
        long[] dp = new long[K + 2];
        for (int i = 1; i <= K; i++) {
            for (int j = 0; j < N; j++) {
                if (i - (j + 1) >= 0) {
                    dp[i] = Math.max(dp[i], dp[i - (j + 1)] + pref[j]);
                }
            }
        }
        System.out.println(dp[K]);
    }
}
