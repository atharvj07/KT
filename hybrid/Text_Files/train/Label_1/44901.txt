import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

public class Main {
    private static final long MOD = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        long n = Long.parseLong(line[0]), x = Long.parseLong(line[1]), y = Long.parseLong(line[2]);
        long[] c = new long[(int) n], w = new long[(int) n];
        for (int i = 0; i < n; i++) {
            line = br.readLine().split(" ");
            c[i] = Long.parseLong(line[0]);
            w[i] = Long.parseLong(line[1]);
        }

        Map<Long, Long> weightByColor = new HashMap<>();
        for (int i = 0; i < n; i++) {
            weightByColor.put(c[i], weightByColor.getOrDefault(c[i], 0L) + w[i]);
        }

        long[] colors = new long[weightByColor.size()];
        int idx = 0;
        for (Long color : weightByColor.keySet()) {
            colors[idx++] = color;
        }

        int m = colors.length;
        long[][] dp = new long[m][3];
        Arrays.fill(dp[0], 1);
        for (int i = 1; i < m; i++) {
            for (int j = 0; j < 3; j++) {
                dp[i][j] = dp[i - 1][j];
                if (j < 2 && weightByColor.get(colors[i - 1]) + weightByColor.get(colors[i]) <= x) {
                    dp[i][j] += dp[i - 1][j + 1];
                    dp[i][j] %= MOD;
                }
                if (j > 0 && weightByColor.get(colors[i - 1]) + weightByColor.get(colors[i]) <= y) {
                    dp[i][j] += dp[i - 1][j - 1];
                    dp[i][j] %= MOD;
                }
            }
        }

        System.out.println(dp[m - 1][1]);
    }
}
