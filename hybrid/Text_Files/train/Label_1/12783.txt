#include <bits/stdc++.h>

using i64 = long long;

const int MOD = 1e9 + 7;
const int MAXN = 1e9;

int main() {
    i64 k;
    std::cin >> k;
    i64 pwr = 1;
    i64 ans = 0;
    for (i64 i = 1; i <= k + 1; ++i) {
        ans += pwr;
        pwr *= 2;
        pwr %= MOD;
        ans %= MOD;
    }
    std::cout << ans << std::endl;
}
