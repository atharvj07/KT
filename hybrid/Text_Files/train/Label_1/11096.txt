#include <bits/stdc++.h>
using namespace std;
int n, m, x, y, a[6000];
char opt, v[6000];
int get(string s) {
  int res = 0;
  reverse(s.begin(), s.end());
  for (int i = 0; i < s.size(); i++)
    if (s[i] == '1') res += 1 << i;
  return res;
}
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> v[i];
    cin >> opt;
    if (cin.peek() != '\n') {
      cin >> x;
      a[i] = x;
    } else
      a[i] = '?';
  }
  int mx = 0, mn = (1 << m) - 1, tmp;
  for (int i = 1; i <= n; i++) {
    if (a[i] != '?') {
      if (opt == '&')
        mn &= a[i], mx &= a[i];
      else if (opt == '|')
        mn |= a[i], mx |= a[i];
      else if (opt == '^')
        mn ^= a[i], mx ^= a[i];
    } else {
      if (opt == '&')
        mn &= y;
      else if (opt == '|')
        mx |= x;
      else if (opt == '^')
        tmp = x ^ y, mn ^= tmp, mx ^= tmp;
    }
    x = mn, y = mx;
  }
  bitset<1000> M(mn);
  string s = M.to_string();
  reverse(s.begin(), s.end());
  s = s.substr(1000 - m, m);
  cout << s << endl;
  M = mx;
  s = M.to_string();
  reverse(s.begin(), s.end());
  s = s.substr(1000 - m, m);
  cout << s << endl;
}
