import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();

        int operations = 0;
        while (N > 0) {
            if (checkPowerOfSix(N)) {
                N -= 6;
            } else if (checkPowerOfNine(N)) {
                N -= 9;
            } else {
                N -= 1;
            }
            operations++;
        }

        System.out.println(operations);
    }

    private static boolean checkPowerOfSix(int N) {
        return N > 0 && (N % 6 == 0) && (N / 6 % 6 == 0);
    }

    private static boolean checkPowerOfNine(int N) {
        return N > 0 && (N % 9 == 0) && (N / 9 % 9 == 0);
    }
}
