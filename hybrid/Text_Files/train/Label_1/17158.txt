#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    ll n, m;
    cin >> n >> m;
    ll a[m], ans = LLONG_MAX;
    for (int i = 0; i < m; i++) { cin >> a[i]; }
    for (int i = 0; i < m; i++) {
      ll sum = 0, cnt = 0;
      for (int j = 0; j < m; j++) {
        if (i == j)
          continue;
        else if (a[j] > a[i])
          sum += a[j];
        else
          cnt++;
      }
      if (sum + a[i] >= n) { ans = min(ans, cnt); }
    }
    if (ans == LLONG_MAX)
      ans = -1;
    cout << ans << "\n";
  }
  return 0;
}
