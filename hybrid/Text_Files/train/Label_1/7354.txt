#include <bits/stdc++.h>
using namespace std;

void solve() {
  int n;
  cin >> n;
  vector<pair<int, int>> points(n);
  for (auto &p : points) cin >> p.first >> p.second;
  sort(points.begin(), points.end());
  string ans = "R";
  for (int i = 0; i < n; i++) {
    while (points[i].second > ans.size() - 1)
      ans += 'U';
    for (int j = ans.size(); j < points[i].first; j++) ans += 'R';
  }
  cout << (ans.size() == points.back().second + points.back().first ? "YES" : "NO")
       << '\n';
  if (ans.size() == points.back().second + points.back().first) cout << ans << '\n';
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int tests;
  cin >> tests;
  while (tests--) solve();
  return 0;
}
