#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;
int n, m, ta, tb, k, a[MAXN], b[MAXN], aa[MAXN], bb[MAXN];
int main() {
  cin >> n >> m >> ta >> tb >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= m; i++) cin >> b[i];
  merge(a + 1, a + n + 1, b + 1, b + m + 1, aa + 1);
  long long l = 0, r = 1e18, ans = -1;
  while (l <= r) {
    long long mid = (l + r) / 2;
    bool ok = false;
    long long last = -1e18;
    int cnt = 0;

    for (int i = 1, j = 1; i <= n; i++) {
      while (j <= m && b[j] + tb < aa[i] + ta) j++;
      if (j <= m) {
        last = aa[i] + ta;
        cnt++;
        if (cnt > k) break;
      }
    }
    if (cnt <= k) {
      ok = true;
      ans = mid;
      l = mid + 1;
    } else
      r = mid - 1;
  }
  cout << ans << endl;
  return 0;
}
