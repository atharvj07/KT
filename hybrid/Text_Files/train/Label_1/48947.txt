import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt() - 1;
        int s = sc.nextInt() - 1;
        int[] a = new int[n];
        int[] q = new int[n]; // queue
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int l = 0, r = 0; // pointers
        q[l++] = s;
        a[s] = -1;
        while (l <= r) {
            int u = q[l++];
            for (int i = 0; i < n; i++) {
                if (a[i] != -1 && a[i] == u) {
                    q[l++] = i;
                    a[i] = -1;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            if (a[i] >= 0) {
                r++;
            }
        }
        System.out.println(r);
    }
}
