#include <iostream>
#include <vector>
#include <algorithm>
struct str{int p,s;};
int main(){
  int n;std::cin>>n;std::vector<str> v(2*n-2);
  for(auto& it:v){std::string x;std::cin>>x;if(x.back()=='a')it.p=1;else it.s=1;}
  std::sort(v.begin(),v.end(),[](str a,str b){return a.p+a.s<b.p+b.s;});
  for(auto a:v)if(a.p)std::cout<<"P";else std::cout<<"S";
}
