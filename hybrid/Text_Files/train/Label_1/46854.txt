import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static boolean sieve[] = new boolean[2000001];

    static void Sieve() {
        Arrays.fill(sieve, true);
        sieve[0] = sieve[1] = false;
        for (int p = 2; p * p <= 2000000; ++p) {
            if (sieve[p]) {
                for (int i = p * p; i <= 2000000; i += p) {
                    sieve[i] = false;
                }
            }
        }
    }

    static class Move {
        int x, y;
        boolean small;

        Move(int x, int y, boolean small) {
            this.x = x;
            this.y = y;
            this.small = small;
        }
    }

    static ArrayList<Move> moves = new ArrayList<>();

    static void GenerateMoves(int f) {
        moves.clear();
        for (int i = 2; i <= 2000000; ++i) {
            if (sieve[i] && i != f) {
                moves.add(new Move(i, 1, true));
                if (i % 2 == 0) {
                    moves.add(new Move(i, i / 2, false));
                }
            }
        }
    }

    static boolean sim(int[] pos[], boolean Alice) {
        int n = pos.length;
        int a = 0, b = 1;
        if (!Alice) {
            a = 1;
            b = 0;
        }
        boolean vis[] = new boolean[2000001];
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < 3; ++j) {
                if (pos[i][j] >= 0) vis[pos[i][j]] = true;
            }
        }
        vis[0] = true;
        for (var move : moves) {
            if (vis[move.y]) continue;
            for (int i = 0; i < n; ++i) {
                int x = pos[i][a] + move.x * (Alice ? 1 : -1);
                if (x >= 0 && !vis[x] && (move.small || x != pos[i][b])) {
                    return true;
                }
            }
        }
        return false;
    }

    public static void main(String[] args) {
        Sieve();
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int f = in.nextInt();
        GenerateMoves(f);
        int[][] pos = new int[n][3];
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < 3; ++j) {
                pos[i][j] = in.nextInt();
            }
        }
        String ans = (sim(pos, true) ? "Alice" : "Bob");
        ans += '\n' + (sim(pos, false) ? "Alice" : "Bob");
        System.out.println(ans);
    }
}
