import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = input.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = line.charAt(j);
            }
        }
        int[][] north = new int[n][m];
        int[][] south = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '#' && (i == 0 || south[i - 1][j] == 0) && (j == 0 || south[i][j - 1] == 0)) {
                    south[i][j] = 1;
                }
                else if (grid[i][j] == '.' && (i == 0 || north[i - 1][j] == 0) && (j == 0 || north[i][j - 1] == 0)) {
                    north[i][j] = 1;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (south[i][j] == 0) {
                    System.out.println(-1);
                    return;
                }
            }
        }
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (north[i][j] == 0 && grid[i][j] == '#') {
                    count++;
                }
            }
        }
        System.out.println(count);
    }
}
