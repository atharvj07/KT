#include<bits/stdc++.h>
using namespace std;
int n,x,a[105];
int main()
{
    cin>>n>>x;
    for(int i=0; i<n; i++)cin>>a[i];
    bool f=1;
    for(int i=0; i<=x; i++)
    {
        int j;
        for(j=0; j<n; j++)if(a[j]==i)break;
        if(j==n){f=0;break;}
    }
    if(f)cout<<0<<endl;
    else cout<<1<<endl;
}
