def find_path_in_grid(s):
    from collections import Counter

    # Count occurrences of each letter
    letter_counts = Counter(s)

    # Check for the letter that occurs 3 times
    for letter, count in letter_counts.items():
        if count == 3:
            triple_letter = letter
            break
    else:
        return "Impossible"

    # Find positions of the triple letter
    positions = [i for i, letter in enumerate(s) if letter == triple_letter]

    # Check if the positions of the triple letter allow forming a valid path
    if positions[1] - positions[0] > 4 or positions[2] - positions[1] > 4:
        return "Impossible"

    # Initial grid setup
    grid = [[''] * 13 for _ in range(2)]

    # Place the triple letter and its immediate neighbors in the grid
    grid[0][6] = s[positions[1] - 1]
    grid[0][7] = triple_letter
    grid[0][8] = s[positions[1] + 1]

    # Fill in the remaining letters
    left_index = 6
    right_index = 8
    for i in range(len(s)):
        if i == positions[1] - 1 or i == positions[1] or i == positions[1] + 1:
            continue
        if left_index > 0:
            left_index -= 1
            grid[left_index // 13][left_index % 13] = s[i]
        else:
            right_index += 1
            grid[right_index // 13][right_index % 13] = s[i]

    # Convert the grid to the desired output format
    return '\n'.join(''.join(row) for row in grid)

# Read input string
s = input().strip()

# Find and print the resulting grid or "Impossible"
print(find_path_in_grid(s))
