#include <bits/stdc++.h>

#define MAXN 100005
#define INF 1e9

using namespace std;

int n, a[MAXN], c[MAXN], p[MAXN], vis[MAXN], ans;
vector<pair<int, int>> adj[MAXN];

void dfs(int u, int par, int sum) {
  vis[u] = 1;
  if (adj[u].size() == 1 && u != 1) {
    ans++;
    return;
  }
  for (auto &i : adj[u]) {
    int v = i.first, w = i.second;
    if (v == par) continue;
    if (sum + w > a[v] || vis[v])
      ans++;
    else
      dfs(v, u, sum + w);
  }
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 2; i <= n; i++) {
    cin >> p[i] >> c[i];
    adj[i].push_back({p[i], c[i]});
    adj[p[i]].push_back({i, c[i]});
  }
  dfs(1, 1, 0);
  cout << ans << endl;
  return 0;
}
