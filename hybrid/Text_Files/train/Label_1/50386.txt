import java.util.*;
import java.io.*;

class Main {
    static int n, m;
    static int[] p, size;
    static boolean[] vis;
    static ArrayList<Integer>[] adj, ans;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter out = new BufferedWriter(new OutputStreamWriter(System.out));
        StringTokenizer tokenizer = new StringTokenizer(in.readLine());
        n = Integer.parseInt(tokenizer.nextToken());
        m = Integer.parseInt(tokenizer.nextToken());

        p = new int[n + 1];
        size = new int[n + 1];
        vis = new boolean[n + 1];
        adj = new ArrayList[n + 1];
        ans = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
            ans[i] = new ArrayList<>();
        }
        for (int i = 0; i <= n; i++) {
            p[i] = i;
            size[i] = 1;
        }

        int u, v;
        for (int i = 0; i < m; i++) {
            tokenizer = new StringTokenizer(in.readLine());
            u = Integer.parseInt(tokenizer.nextToken());
            v = Integer.parseInt(tokenizer.nextToken());
            adj[u].add(v);
            adj[v].add(u);
        }
        if (m % 2 == 1) {
            out.write("No solution\n");
            out.close();
            return;
        }

        for (int i = 1; i <= n; i++) {
            if (!vis[i]) dfs(i, 0);
        }

        for (List<Integer> arr : ans) {
            if (arr.size() >= 3) {
                out.write("No solution\n");
                out.close();
                return;
            }
        }

        for (int i = 1; i <= n; i++) {
            if (ans[i].size() == 2)
                for (int ver : ans[i]) {
                    out.write(i + " " + ver + " " + find(3 - i - ver) + "\n");
                }
        }
        out.close();
    }

    static void dfs(int node, int par) {
        if (vis[node]) return;
        vis[node] = true;
        ans[node].add(par);
        for (int ver : adj[node]) {
            if (ver == par) continue;
            dfs(ver, node);
        }
    }

    static int find(int node) {
        if (p[node] == node) return node;
        return p[node] = find(p[node]);
    }
}
