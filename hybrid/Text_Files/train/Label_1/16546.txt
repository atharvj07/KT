#include <iostream>

int main() {
	int n;
	std::cin >> n;
	long long s[n + 1];
	for (int i = n - 1; i >= 2; i--) {
		long long a;
		std::cin >> a;
		s[i - 1] = s[i] + a;
	}
	for (int i = 1; i < n; i++) {
		if (i + s[i] > n || s[i] <= 0)
			std::cout << -1 << '\n';
		else
			std::cout << s[i] + 2 * s[i + s[i]] << '\n';
	}
}
