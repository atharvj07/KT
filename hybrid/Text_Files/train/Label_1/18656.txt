#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    bool flag = true;
    for (int i = 0; i < n; i++) {
      if (b[i] < a[i]) flag = false;
      else
        a[i] = b[i];
    }
    cout << ((flag) ? "YES\n" : "NO\n");
  }
  return 0;
}
