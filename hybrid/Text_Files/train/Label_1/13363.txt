#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> a(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> a[i][j];
    }
  }

  for (int i = 0; i < n; i++) {
    vector<int> b(a[i]);
    b.erase(unique(b.begin(), b.end()), b.end());
    for (int j = 0; j < m; j++) {
      cout << b[lower_bound(b.begin(), b.end(), a[i][j]) - b.begin()] << " ";
    }
    cout << endl;
  }

  return 0;
}
