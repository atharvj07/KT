#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int zeroes = 0, ones = 0;
  for (int i = 0; i < n;) {
    int j = i;
    if (j == n) break;
    int c = 0;
    while (j < n && cin >> c) {
      if ((c == 1 && zeroes > 0) || (c == 0 && ones > 0)) {
        cout << "NO" << endl;
        return 0;
      }
      j++;
      if (c == 0) zeroes++;
      if (c == 1) ones++;
    }
    if (zeroes != ones) {
      cout << "NO" << endl;
      return 0;
    }
    zeroes = 0, ones = 0;
    i = j;
  }
  cout << "YES" << endl;
  return 0;
}
