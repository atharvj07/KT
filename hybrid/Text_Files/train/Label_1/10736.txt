#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10, inf = 1e9;
int arr[N], fr[N];
int main() {
  int n, k, p;
  cin >> n >> k;
  vector<pair<int, int>> v;
  unordered_map<int, int> cnt;
  for (int i = 0; i < n; ++i) {
    cin >> arr[i], cnt[arr[i]]++;
  }
  for (auto it : cnt) v.push_back(it);
  sort(v.rbegin(), v.rend());
  for (int i = 0; i < v.size() && k; ++i) {
    int x = min(k, cnt[v[i].first]);
    arr[p] = v[i].first;
    fr[v[i].first] = x;
    p += x, k -= x;
  }
  int cntt = 0;
  for (int i = 0; i < n; ++i) {
    if (fr[arr[i]]) {
      cout << arr[i] << " ";
      cntt++;
      fr[arr[i]]--;
      if (cntt == k) break;
    }
  }
  return 0;
}
