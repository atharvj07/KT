#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  int t; cin >> t;
  while(t--) {
    int n, k, d; cin >> n >> k >> d;
    vector<int> ar(n);
    for(int i = 0; i < n; i++) cin >> ar[i];
    int cnt = 0, ans = k;
    bool ok = 0;
    unordered_map<int, int> mp;
    for(int i = 0, j = 0; i < n; i++) {
      mp[ar[i]]++;
      if(mp[ar[i]] == 1) cnt++;
      while(cnt == k) {
        ans = min(ans, i - j + 1);
        if(ans == d) {
          ok = 1;
          break;
        }
        mp[ar[j]]--;
        if(mp[ar[j]] == 0) cnt--;
        j++;
      }
      if(ok) break;
    }
    cout << ans << endl;
  }
  return 0;
}
