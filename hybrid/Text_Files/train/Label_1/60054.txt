n, m = map(int, input().split())
order = []

# Fill the window seats first
for i in range(1, min(m+1, 2*n+1)):
    if i <= n:
        order.append(i)
    else:
        order.append(i + n)

# Fill the non-window seats
for i in range(1, m-2*n+1):
    order.insert((i-1)*2, n+i)

print(' '.join(map(str, order)))
