import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String t = in.nextLine();
        int n = t.length();
        for (int i = 0; i < n - 1; i++) {
            if (t.charAt(i) == t.charAt(i + 1) && t.charAt(i) != 'a') {
                System.out.println(t.substring(0, i + 1));
                return;
            }
        }
        System.out.println(":(");
    }
}
