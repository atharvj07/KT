from collections import defaultdict

n, m, q = map(int, input().split())
s = input()
t = input()
pref = [defaultdict(int) for _ in range(n + 1)]
for i in range(n):
    pref[i + 1] = pref[i].copy()
    pref[i + 1][s[i]] += 1
for i in range(q):
    l, r = map(int, input().split())
    cnt = pref[r][t[0]]
    if len(t) > 1:
        cnt -= pref[l - 1][t[0]]
    for j in range(1, len(t)):
        cnt -= pref[r][t[j]]
        cnt += pref[l - 1][t[j]]
    print(cnt)
