import sys

N, M = map(int, input().split())
C = int(input())
hotels = [list(map(int, input().split())) for _ in range(C)]
H = int(input())
restaurants = [list(map(int, input().split())) for _ in range(H)]

ans = int(1e9)
idx = 0

for x, y in restaurants:
    cur = 0
    for a, b in hotels:
        cur = max(cur, abs(x - a) + abs(y - b))
    if cur < ans:
        ans = cur
        idx = x, y

print(ans)
print(idx[0] if idx else 'None')
