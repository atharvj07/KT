#include <bits/stdc++.h>
#define ll long long
#define INF 1 << 30

using namespace std;

const int N = 100005;
vector<int>primes; // store prime numbers

bool vis[N];

int main() {
    for (int i = 2; i < N; i++) {
        if (!vis[i]) { // i is a prime
            primes.push_back(i);
            for (int j = i + i; j < N; j += i) vis[j] = true; // mark multiples of i
        }
    }

    int n, x;
    cin >> n;
    map<int, int>good; // store the frequency of good integers
    for (int i = 0; i < n; i++) {
        cin >> x;
        good[x]++;
    }

    int ans = 0;
    for (auto it : good) {
        if (it.first == 1) continue;
        int cur = good[it.first];
        for (int i = 0; primes[i] <= it.first; i++) {
            if (it.first % primes[i] == 0) {
                cur = min(cur, good[it.first / primes[i]]);
            }
        }
        ans += cur;
    }

    cout << ans << '\n';
    return 0;
}
