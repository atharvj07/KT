#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> a(n);
  for (auto &x : a) { cin >> x; }
  vector<int> b = a;
  sort(b.begin(), b.end());
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] != b[i]) cnt++;
  }
  cout << cnt / 2 << "\n";
  return 0;
}
