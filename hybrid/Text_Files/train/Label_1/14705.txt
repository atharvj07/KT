#include <vector>
#include <queue>
#include <iostream>

template <typename T>
std::vector<std::vector<T>> floyd_warshall(const std::vector<std::vector<T>>& g) {
    size_t n = g.size();
    std::vector<std::vector<T>> dist = g;
    for (size_t k = 0; k < n; ++k)
        for (size_t i = 0; i < n; ++i)
            for (size_t j = 0; j < n; ++j)
                dist[i][j] = std::min(dist[i][j], dist[i][k] + dist[k][j]);
    return dist;
}

int main() {
    size_t n;
    std::cin >> n;
    std::vector<int> deg(n);
    for (size_t i = 0; i < n; ++i) {
        int a;
        std::cin >> a;
        deg[i] = a;
    }
    std::vector<std::vector<int>> g(n, std::vector<int>(n, 1e9));
    for (int i = 0; i < n; ++i) {
        g[i][i] = 0;
        for (int j = i+1; j < n; ++j) {
            if (deg[i]==1 && deg[j]==1) continue;
            g[i][j] = g[j][i] = 1;
            deg[i]--;
            deg[j]--;
        }
    }
    int cnt = 0;
    for (size_t i = 0; i < n; ++i)
        for (size_t j = 0; j < n; ++j)
            if (g[i][j]<1e9)
                cnt++;
    if (cnt!=n) {
        std::cout << "NO" << std::endl;
        return 0;
    }
    int mx = 0;
    auto dist = floyd_warshall(g);
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            if (i==j || dist[i][j]==1e9)
                continue;
            mx = std::max(mx, int(dist[i][j]));
        }
    }
    std::cout << "YES " << mx << std::endl << cnt << std::endl;
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            if (i==j || g[i][j]==1e9)
                continue;
            std::cout << i+1 << ' ' << j+1 << std::endl;
        }
    }
    return 0;
}
