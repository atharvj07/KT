#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        int a[n], b[n];
        for (int i = 0; i < n; i++) cin >> a[i];
        for (int i = 0; i < n; i++) cin >> b[i];
        sort(b, b + n);
        bool ans = true;
        for (int i = 0; i < n; i++) {
            if (a[i] + b[i] > x) {
                ans = false;
                break;
            }
        }
        cout << (ans ? "Yes" : "No") << endl;
    }
    return 0;
}
