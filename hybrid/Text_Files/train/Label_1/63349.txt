import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    g = [[]]
    for i in range(n):
        g.append([])
    for i in range(n - 1):
        u, v = map(int, input().split())
        g[u].append(v)
        g[v].append(u)
    vis = [0] * (n + 1)
    sz = [0] * (n + 1)
    ans = 0

    def dfs(cur, pre):
        nonlocal ans
        sz[cur] = 1
        for x in g[cur]:
            if not vis[x] and x != pre:
                dfs(x, cur)
                sz[cur] += sz[x]
                ans = max(ans, sz[x])
    dfs(1, 1)
    print(max(n - 1 - ans, ans))
