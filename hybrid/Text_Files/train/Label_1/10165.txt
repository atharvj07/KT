#include <iostream>

using namespace std;

long long powmod(long long x, long long pow, long long mod) {
    long long res = 1;
    while (pow) {
        if (pow & 1) res = res * x % mod;
        x = x * x % mod;
        pow /= 2;
    }
    return res;
}

int main() {
    long long n, mod = 1e9 + 7;
    cin >> n;
    cout << powmod(n, n / 2 - 1, mod) << endl;
    return 0;
}
