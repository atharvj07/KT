n, m = map(int, input().split())
a = [[int(x) for x in input().split()] for i in range(n)]

ans = 0
for i in range(n):
    for j in range(m):
        if a[i][j] == 0:
            f = False
            if i > 0 and a[i - 1][j] == 1:  f = True
            if i < n - 1 and a[i + 1][j] == 1:  f = True
            if j > 0 and a[i][j - 1] == 1:  f = True
            if j < m - 1 and a[i][j + 1] == 1:  f = True
            if f: ans += 1
print(ans)
