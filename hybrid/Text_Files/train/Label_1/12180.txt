#include <bits/stdc++.h>
using namespace std;

using ll = long long;
const int N = 200007;

int n;
ll a[N], ans;
vector<int> g[N];

int32_t main() {
    cin >> n;
    for (int i = 1; i <= n; ++i)
        cin >> a[i];
    sort(a + 1, a + 1 + n);
    int u, v;
    for (int i = 1; i < n; ++i) {
        u = lower_bound(a + 1, a + 1 + n, a[i] ^ a[i + 1]) - a;
        if (u == i + 1)
            continue;
        if (g[i].empty() || g[i].back() != u) {
            ++ans;
            g[i].push_back(u);
            g[u].push_back(i);
        }
    }
    cout << ans << endl;
}
