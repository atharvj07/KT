#include <iostream>
#include <string>
#include <stack>
using namespace std;
int main() {
  int n, m, p;
  cin >> n >> m >> p;
  string s;
  cin >> s;
  stack<char> st;
  for (int i = 0; i < s.size(); i++) {
    if (st.empty()) {
      st.push(s[i]);
      continue;
    }
    char c = st.top();
    if (c == '(' && s[i] == ')') {
      st.pop();
    } else {
      st.push(s[i]);
    }
  }
  string ans;
  for (int i = 0, j = 0; i < s.size(); i++) {
    if (s[i] == '(') {
      j++;
      if (j == p) {
        ans += '(';
        continue;
      }
    } else {
      j--;
      if (j == p - 1) {
        ans += ')';
        continue;
      }
    }
  }
  while (!st.empty()) {
    ans += st.top();
    st.pop();
  }
  cout << ans;
  return 0;
}
