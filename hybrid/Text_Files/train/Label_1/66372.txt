import sys
input = sys.stdin.readline

n, p = map(int, input().split())
l = []
for i in range(n):
    li, ri = map(int, input().split())
    l.append((li, ri))
 
ans = 0
for i in range(n):
    li, ri = l[i]
    lj, rj = l[(i + 1) % n]
    ans += (ri - li + 1) * (rj - lj + 1) / (ri * rj + 1)
    ans %= p
print(ans)
