#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, t;
  cin >> n >> t;
  vector<int> ti(n);
  for (int i = 0; i < n; i++) cin >> ti[i];
  int ans = 0, cur = 0;
  for (int i = 0; i < n; i++) {
    if (cur + 1 >= ti[i]) {
      cur++;
      ans++;
    } else if (cur + 1 + t >= ti[i]) {
      cur++;
      t -= ti[i] - cur;
    } else
      break;
  }
  cout << ans << endl;
  return 0;
}
