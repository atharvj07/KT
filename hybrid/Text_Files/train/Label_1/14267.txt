#include <cstdio>
#include <cmath>
using namespace std;

int n;
double a[4001];
int main() {
  scanf("%d", &n);
  for (int i = 0; i < 2 * n; i++) scanf("%lf", &a[i]);
  double c = 0.0;
  for (int i = 0; i < 2 * n; i += 2) c += floor(a[i] + 0.5) - a[i];
  for (int i = 1; i < 2 * n; i += 2) c += ceil(a[i] - 0.5) - a[i];
  printf("%.3lf",fabs(c));
  return 0;
}
