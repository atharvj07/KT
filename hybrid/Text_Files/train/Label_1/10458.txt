#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n_r, n_g, n_b;
    cin >> n_r >> n_g >> n_b;
    vector<int> r(n_r), g(n_g), b(n_b);
    for (int i = 0; i < n_r; i++) cin >> r[i];
    for (int i = 0; i < n_g; i++) cin >> g[i];
    for (int i = 0; i < n_b; i++) cin >> b[i];
    sort(r.begin(), r.end());
    sort(g.begin(), g.end());
    sort(b.begin(), b.end());
    long long mr = -1, mg = -1, mb = -1;
    if (n_r >= 1) mr = r[r.size() / 2];
    if (n_g >= 1) mg = g[g.size() / 2];
    if (n_b >= 1) mb = b[b.size() / 2];
    cout << (mr - mg) * (mr - mg) + (mg - mb) * (mg - mb) + (mb - mr) * (mb - mr)
         << endl;
  }
}
