import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        List<Long> len = Stream.of(br.readLine().split(" "))
                .mapToLong(Long::parseLong).boxed().collect(Collectors.toList());
        List<Long> cost = Stream.of(br.readLine().split(" "))
                .mapToLong(Long::parseLong).boxed().collect(Collectors.toList());
        len.sort(Collections.reverseOrder());
        cost.sort(Collections.reverseOrder());

        long totalCost = 0;
        long minCost = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            totalCost += cost.get(i);
            minCost = Math.min(minCost, totalCost + len.get(i));
        }
        System.out.println(totalCost >= minCost ? -1 : minCost);
    }
}
