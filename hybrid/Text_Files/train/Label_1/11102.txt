#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> times(n);
    for (int i = 0; i < n; i++) {
        int h, m;
        char c;
        cin >> h >> c >> m;
        times[i] = h * 60 + m;
    }
    sort(times.begin(), times.end());
    int max_sleep = 0;
    for (int i = 0; i < n - 1; i++) {
        max_sleep = max(max_sleep, times[i + 1] - times[i] - 1);
    }
    max_sleep = max(max_sleep, 24 * 60 - times[n - 1] + times[0]);
    int h = max_sleep / 60;
    int m = max_sleep % 60;
    cout << h << ":" << m << endl;
    return 0;
}
