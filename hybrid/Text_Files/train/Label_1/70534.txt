n,k,a,b=map(int,input().split())
l=len(bin(2**n)[2:])
arr=[0 for _ in range(0,l+1)]
for x in map(int,input().split()):arr[x-1]=1
p=[[0 for _ in range(0,l+1)]for _ in range(0,l+1)]
for i in range(l-1,-1,-1):
    for j in range(l-1,-1,-1):
        p[i][j]=min((a+b*arr[k]*len(range(i,j+1)))if j-i+1>=2 else 1e9,p[i][j+1],p[i+1][j])
print(p[0][0])
