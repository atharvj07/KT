import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n + 1], b = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextLong();
        }
        for (int i = 1; i <= n; i++) {
            b[i] = sc.nextLong();
        }
        // dp[i][0] = the minimum number of operations to change a[i-1] to b[i-1] without changing a[i]
        // dp[i][1] = the minimum number of operations to change a[i-1] to b[i-1] and a[i] to b[i]
        long[][] dp = new long[n + 1][2];
        for (int i = 1; i <= n; i++) {
            dp[i][0] = Math.abs(b[i - 1] - a[i - 1]);
            dp[i][1] = dp[i][0] + a[i] - b[i];
        }
        for (int i = 2; i <= n; i++) {
            dp[i][0] = Math.min(dp[i][0], dp[i - 1][0] + Math.abs(b[i - 1] - a[i - 1]));
            dp[i][1] = Math.min(dp[i][1], dp[i - 1][0] + a[i] - b[i]);
            dp[i][1] = Math.min(dp[i][1], dp[i - 1][1] + Math.abs(b[i] - a[i]));
        }
        if (dp[n][1] == dp[n][0]) System.out.println(-1);
        else System.out.println(dp[n][1]);
    }
}
