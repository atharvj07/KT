import heapq

n, m, t = map(int, input().split())
adj = [[] for _ in range(n+1)]
for _ in range(m):
    a, b, w = map(int, input().split())
    adj[a].append((w, b))

q = [(0, 1, [1])]
ans = []
while q:
    time, node, path = heapq.heappop(q)
    if node == n:
        if len(path) > len(ans): ans = path
        continue
    if time > t: continue
    for w, next in adj[node]:
        heapq.heappush(q, (time+w, next, path+[next]))
print(len(ans))
print(*ans)
