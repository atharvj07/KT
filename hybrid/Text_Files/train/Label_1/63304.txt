n, k = map(int, input().split())
a, b, c, d = map(int, input().split())

if n == 4 or k < n + 1:
    print(-1)
else:
    remaining_cities = [i for i in range(1, n + 1) if i not in [a, b, c, d]]
    path1 = [a] + remaining_cities + [b]
    path2 = [c] + remaining_cities[::-1] + [d]
    path1 = [a, c] + remaining_cities + [d, b]
    path2 = [c, a] + remaining_cities[::-1] + [b, d]
    print(' '.join(map(str, path1)))
    print(' '.join(map(str, path2)))
