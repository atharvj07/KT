#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int a[100001], p[100001];
int main() {
  int n, i, mx = 0, pos = 0;
  cin >> n;
  for (i = 0; i < n; i++) {
    cin >> a[i];
    if (p[a[i]] == 0) {
      p[a[i]] = 1;
      mx++;
      if (mx > 1 && a[i] > a[pos]) pos = i;
    }
  }
  cout << mx << endl;
  while (pos >= 0) {
    cout << a[pos] << " ";
    pos--;
  }
  cout << endl;
  return 0;
}
