import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] tiles = new int[n][4];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 4; j++) {
                tiles[i][j] = sc.nextInt();
            }
        }
        sc.close();

        // Sort tiles by xLower
        Arrays.sort(tiles, (a, b) -> a[0] - b[0]);

        int ans = 0;
        int curLow = Integer.MIN_VALUE;
        int curHigh = Integer.MIN_VALUE;
        for (int[] tile : tiles) {
            if (tile[0] <= curHigh) {
                curHigh = Math.max(curHigh, tile[2]);
            } else {
                ans++;
                curLow = tile[0];
                curHigh = tile[2];
            }
        }

        System.out.println(ans);
    }
}
