#include <iostream>
#include <vector>

const int MOD = 1000000007;
int t[65][65][6];

int main() {
    int k;
    std::cin >> k;
    t[1][1][1] = t[1][0][0] = t[1][1][0] = 6;
    for (int i = 2; i <= k; i++) {
        for (int j = 0; j <= i; j++) {
            for (int l = 0; l < 6; l++) {
                if (j) t[i][j][l] = ((t[i - 1][j - 1][l] + t[i - 1][j - 1][(l + 5) % 6]) % MOD + t[i - 1][j][l]) % MOD;
                if (2 * j + 1 <= i) t[i][j][l] = (t[i][j][l] + t[i][j + 1][l]) % MOD;
            }
        }
    }
    std::cout << t[k][0][0] << std::endl;
}
