import java.util.Scanner;

public class JapaneseCrossword {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        String row = scanner.next();

        int groups = 0;
        int currentGroupSize = 0;

        for (int i = 0; i < n; i++) {
            if (row.charAt(i) == 'B') {
                currentGroupSize++;
            } else {
                if (currentGroupSize > 0) {
                    groups++;
                }
                currentGroupSize = 0;
            }
        }

        if (currentGroupSize > 0) {
            groups++;
        }

        System.out.println(groups);

        int[] groupSizes = new int[groups];
        int groupIndex = 0;

        for (int i = 0; i < n; i++) {
            if (row.charAt(i) == 'B') {
                currentGroupSize++;
            } else {
                if (currentGroupSize > 0) {
                    groupSizes[groupIndex++] = currentGroupSize;
                }
                currentGroupSize = 0;
            }
        }

        if (currentGroupSize > 0) {
            groupSizes[groupIndex] = currentGroupSize;
        }

        for (int groupSize : groupSizes) {
            System.out.print(groupSize + " ");
        }
    }
}
