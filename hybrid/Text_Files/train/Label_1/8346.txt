#include <bits/stdc++.h>
using namespace std;
const int N = 5005;
struct node {
    int s;
    double x;
    bool operator<(const node& a) const { return x < a.x; }
};
int main() {
    int n, m;
    cin >> n >> m;
    vector<node> v(n);
    for (auto& i : v) cin >> i.s >> i.x;
    sort(v.begin(), v.end());
    int ans = 0;
    for (int i = m - 1; i < n; i++) {
        if (v[i].s > v[i - m + 1].s) ans++;
    }
    cout << ans << endl;
    return 0;
}
