import math

n, k, A, B = map(int, input().split())

avengers = list(map(int, input().split()))

mx = int(math.log2(2**n)) + 1

power = [0 for x in range(mx + 1)]

for i in range(1, mx + 1):
    power[i] = min(power[i - 1] * 2 + A, power[i - 1] + B * k * (2**(i - 1)))

cur = 2**n
ans = 0
while cur > 0:
    p = 0
    while p < mx and (2**(p + 1)) <= cur:
        p += 1
    if p in avengers:
        ans += power[p]
        k -= 1

    cur = int(cur / 2)

print(ans)
