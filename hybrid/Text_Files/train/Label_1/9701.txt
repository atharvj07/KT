#include <bits/stdc++.h>
#define pii pair<long long, long long>
#define mp make_pair
using namespace std;

long long n, b;
vector<long long> bvals;
vector<pii> gaps;
long long ans;

int main() {
  cin >> n;
  bvals.resize(n + 1);
  for (long long i = 1; i <= n; i++) {
    cin >> b;
    bvals[i] = b;
  }
  for (long long i = 2, gap; i <= n; i++) {
    gap = bvals[i] - bvals[i - 1];
    if (gap > 0) gaps.push_back(mp(gap, i));
  }
  sort(gaps.begin(), gaps.end());
  for (pii gap : gaps) {
    ans += bvals[gap.second];
  }
  cout << ans << "\n";
  return 0;
}
