#include <iostream>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int x, y, z;
    cin >> x >> y >> z;
    if (y > x) swap(x, y);
    if (z > x) swap(x, z);
    if (z == x && y == x) cout << "YES\n" << x << ' ' << y << ' ' << z << endl;
    else if (z + y >= x) cout << "YES\n" << z << ' ' << y << ' ' << x << endl;
    else cout << "NO\n";
  }
}
