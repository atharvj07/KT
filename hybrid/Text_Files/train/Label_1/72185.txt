from collections import defaultdict
import sys
input = sys.stdin.readline

for t in range(int(input())):
    input()
    adj = defaultdict(list)
    n, m = map(int, input().split())
    for i in range(m):
        a, b = map(int, input().split())
        adj[a].append(b)

    vis = [1] * n  # 1 -> unvisited, 2 -> finished, 3 -> in stack
    ans = [0] * n

    for node in range(1, n + 1):
        def dfs(u):
            if vis[u - 1] == 2:
                return
            if vis[u - 1] == 3:
                ans[node - 1] = -1
                return

            vis[u - 1] = 3
            for v in adj[u]:
                dfs(v)
            vis[u - 1] = 2
            if node != u:
                if ans[u - 1] == -1:
                    ans[node - 1] = -1
                elif ans[u - 1] < ans[node - 1] or ans[node - 1] == 0:
                    ans[node - 1] = ans[u - 1] + 1
        dfs(node)

    print(*ans)
