#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, s;
  std::cin >> n >> s;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }
  std::sort(a.begin(), a.end());
  if (a[0] * (n - 1) <= s) {
    std::cout << "YES" << std::endl;
  } else {
    std::cout << "NO" << std::endl;
  }
  return 0;
}
