#include <iostream>
#include <vector>

using namespace std;

int main() {
  long long x, y, m;
  cin >> x >> y >> m;
  if (m <= max(x, y))
    cout << 0 << endl;
  else {
    vector<vector<int>> dp(2, vector<int>(m + 1, m + 1));
    dp[0][max(x, y)] = 0;
    dp[1][max(x, y)] = 0;
    for (int i = max(x, y) - 1; i >= 0; i--) {
      for (int j = 0; j <= m; j++) {
        dp[0][j] = m + 1;
        dp[1][j] = m + 1;
      }
      for (int j = 0; j <= m; j++) {
        if (j + x <= m) dp[0][j + x] = min(dp[0][j + x], dp[0][j] + 1);
        if (j + y <= m) dp[0][j + y] = min(dp[0][j + y], dp[0][j] + 1);
        if (j + x <= m) dp[1][j + x] = min(dp[1][j + x], dp[1][j] + 1);
        if (j + y <= m) dp[1][j + y] = min(dp[1][j + y], dp[1][j] + 1);
      }
    }
    int ans = m + 1;
    for (int i = m; i <= m; i++) {
      ans = min(ans, dp[0][i]);
      ans = min(ans, dp[1][i]);
    }
    if (ans != m + 1)
      cout << ans << endl;
    else
      cout << -1 << endl;
  }
}
