import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] y = new int[3];
        int[] m = new int[3];
        int[] p = new int[3];
        for (int i = 0; i < 3; i++) {
            y[i] = sc.nextInt();
        }
        for (int i = 0; i < 3; i++) {
            m[i] = sc.nextInt();
        }
        for (int i = 0; i < 3; i++) {
            p[i] = sc.nextInt();
        }
        sc.close();
        int ans = 0;
        while (true) {
            int dy = Math.max(0, m[1] - y[2]);
            int dx = Math.max(0, y[1] - m[2]);
            if (y[0] - dx <= 0 && m[0] - dy > 0) {
                break;
            } else {
                int hy = Math.max(0, (int) Math.ceil((double) (m[0] - dy) / dx));
                int hp = Math.max(0, (int) Math.ceil((double) (y[0] - dx) / dy));
                ans += Math.min(hy * p[0], hp * p[0]);
                y[0] = y[0] - dx * hp;
                m[0] = m[0] - dy * hy;
                y[2] += hp;
                m[2] += hy;
            }
        }
        System.out.println(ans);
    }
}
