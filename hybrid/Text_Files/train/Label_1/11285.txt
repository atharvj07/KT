#include <bits/stdc++.h>
using namespace std;

const int MAXN = 110;

struct Item {
    int t, d, p;
    bool operator<(const Item &other) const {
        return t + d < other.t + other.d;
    }
};

int n;
Item items[MAXN];

int solve() {
    sort(items, items + n);
    int dp[n];
    for (int i = 0; i < n; i++) dp[i] = items[i].p;
    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (items[i].t >= items[j].d) continue;
            dp[i] = max(dp[i], dp[j] + items[i].p);
        }
    }
    int mx = *max_element(dp, dp + n);
    vector<int> ans;
    for (int i = n - 1; i >= 0; i--) {
        if (dp[i] == mx) {
            ans.push_back(i);
            mx -= items[i].p;
        }
    }
    reverse(ans.begin(), ans.end());
    cout << dp[n - 1] << endl << ans.size() << endl;
    for (int i : ans) cout << i + 1 << " ";
    return 0;
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> items[i].t >> items[i].d >> items[i].p;
    return solve();
}
