#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    long long n, m, a, b;
    vector<long long> stairs, ans;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a;
        stairs.push_back(a);
    }
    cin >> m;
    while (m--) {
        cin >> a >> b;
        ans.push_back(upper_bound(stairs.begin(), stairs.begin() + a, b) - stairs.begin());
    }
    for (auto i : ans) { cout << stairs[i - 1] << endl; }
    return 0;
}
