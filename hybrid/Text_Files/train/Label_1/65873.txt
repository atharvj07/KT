q = int(input())
a = [0] * 500001
sums = {}

for _ in range(q):
    t, x, y = map(int, input().split())
    if t == 1:
        a[x] += y
        for k in sums.keys():
            if x % k == y % k:
                sums[k][y % k] += y
    else:
        if x not in sums:
            sums[x] = [0] * x
            for i in range(1, 500001):
                sums[x][i % x] += a[i]
        print(sums[x][y])
