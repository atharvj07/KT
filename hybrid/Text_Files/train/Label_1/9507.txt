#include<iostream>
using namespace std;
char G[55][55];
int n;
int T[55][55];
int dx[]={-1,1,0,0};
int dy[]={0,0,-1,1};
inline bool check(int x,int y){
    for(int i=x;i<x+3;i++){
        for(int j=y;j<y+3;j++){
            if(G[i][j]=='#')return 0;
        }
    }
    for(int i=x+1;i<x+3;i++){
        if(G[i][y+1]=='#')return 0;
    }
    for(int i=x;i<x+2;i++){
        if(G[i][y+2]=='#')return 0;
    }
    return 1;
}
inline void fill(int x,int y){
    for(int i=x;i<x+3;i++){
        for(int j=y;j<y+3;j++){
            G[i][j]='#';
        }
    }
    for(int i=x+1;i<x+3;i++){
        if(G[i][y+1]=='#')G[i][y+1]='.';
    }
    for(int i=x;i<x+3;i++){
        if(G[i][y+2]=='#')G[i][y+2]='.';
    }
}
int main(){
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>G[i];
    }
    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++){
            if(G[i][j]=='.'){
                if(check(i,j)){
                    T[i][j]=1;
                    fill(i,j);
                }
            }
        }
    }
    bool have=0;
    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++){
            if(G[i][j]=='.'){
                have=1;
                break;
            }
        }
    }
    if(have)cout<<"NO\n";
    else cout<<"YES\n";
    return 0;
}
