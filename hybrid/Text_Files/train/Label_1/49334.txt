import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] a = new int[4];
        int[] b = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = in.nextInt();
            b[i] = in.nextInt();
        }
        int[] o = new int[4];
        o[0] = 1; o[1] = 2; o[2] = 3; o[3] = 4;
        int[] t1 = {a[o[0] - 1], b[o[1] - 1]};
        int[] t2 = {a[o[2] - 1], b[o[3] - 1]};
        int[] w = new int[2];
        if (t1[0] > t2[1] && t1[1] > t2[0]) {
            w[0] = 1;
        }
        if (t2[0] > t1[1] && t2[1] > t1[0]) {
            w[1] = 1;
        }
        int cnt = w[0] + w[1];
        if (cnt == 0) {
            System.out.println("Draw");
        } else if (cnt == 1) {
            System.out.println(w[0] == 1 ? "Team 1" : "Team 2");
        } else {
            System.out.println("Draw");
        }
    }
}
