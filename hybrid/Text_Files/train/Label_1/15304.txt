#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[100001], b[100001], flag[100001];
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) cin >> b[i];
    memset(flag, 0, sizeof(flag));
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (a[i] < a[j] && b[i] > b[j]) {
                flag[i] = flag[j] = 1;
            }
        }
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (flag[i] == 1) ans++;
    }
    cout << ans << "\n";
    return 0;
}
