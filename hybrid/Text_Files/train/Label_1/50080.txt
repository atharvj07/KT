import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long[] A = new long[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextLong();
        }

        long total = 0;
        for (long a : A) {
            total += a;
        }

        long ans = 0;
        long[] coins = {1, 10, 100, 1000, 10000, 100000, 1000000, 10000000, 100000000, 1000000000, 10000000000L, 100000000000L, 1000000000000L, 10000000000000L, 100000000000000L, 1000000000000000L, 10000000000000000L, 100000000000000000L, 1000000000000000000L};

        for (int i = coins.length - 1; i >= 0; i--) {
            while (total >= coins[i]) {
                total -= coins[i];
                ans++;
            }
        }

        System.out.println(ans);
    }
}
