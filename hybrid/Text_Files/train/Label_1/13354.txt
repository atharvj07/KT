#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n,m;
        cin>>n>>m;
        int a[n],b[m],dp[n+1][m+1],ans[n];
        for(int i=0;i<n;i++) cin>>a[i];
        for(int i=0;i<m;i++) cin>>b[i];
        for(int i=0;i<=n;i++){
            for(int j=0;j<=m;j++){
                if(!i||!j){
                    dp[i][j]=0; continue;
                }
                if(a[i-1]==b[j-1]) dp[i][j]=dp[i-1][j-1]+1;
                else dp[i][j]=max(dp[i][j-1],dp[i-1][j]);
            }
        }
        if(!dp[n][m]){
            cout<<"NO\n";
            continue;
        }
        cout<<"YES\n";
        int i=n,j=m,k=dp[n][m]-1;
        while(i>0&&j>0){
            if(a[i-1]==b[j-1]) ans[k--]=a[i-1],i--,j--;
            else if(dp[i][j]==dp[i-1][j]) i--;
            else j--;
        }
        cout<<dp[n][m]<<endl;
        for(int i=0;i<dp[n][m];i++) cout<<ans[i]<<" ";
        cout<<endl;
    }
    return 0;
}
