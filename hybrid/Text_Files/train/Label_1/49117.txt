import java.util.Scanner;

public class JapaneseCrossword {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String row = scanner.next();

        int k = 0;
        int currentCount = 0;
        for (int i = 0; i < n; i++) {
            if (row.charAt(i) == 'B') {
                currentCount++;
            } else {
                if (currentCount > 0) {
                    k++;
                }
                currentCount = 0;
            }
        }
        if (currentCount > 0) {
            k++;
        }

        System.out.println(k);
        if (k > 0) {
            int[] groups = new int[k];
            k = 0;
            currentCount = 0;
            for (int i = 0; i < n; i++) {
                if (row.charAt(i) == 'B') {
                    currentCount++;
                } else {
                    if (currentCount > 0) {
                        groups[k++] = currentCount;
                    }
                    currentCount = 0;
                }
            }
            if (currentCount > 0) {
                groups[k] = currentCount;
            }
            for (int group : groups) {
                System.out.print(group + " ");
            }
        }
    }
}
