from collections import defaultdict
from math import gcd

n = int(input())
a = [int(i) for i in input().split()]
g = defaultdict(list)
for _ in range(n - 1):
    x, y = tuple(int(i) for i in input().split())
    g[x].append(y)
    g[y].append(x)
ans = [0] * n
q = [a[0]]
vis = [False] * (n + 1)
vis[0] = True
ans[0] = a[0]
while q:
    u = q.pop(0)
    vis[u] = False
    for v in g.get(u, []):
        if not vis[v]:
            vis[v] = True
            q.append(v)
            ans[v] = gcd(ans[u], a[v - 1])
for i in ans:
    print(i, end=' ')
