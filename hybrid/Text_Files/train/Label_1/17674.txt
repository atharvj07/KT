#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int n, x, a, b;
    cin >> n >> x >> a >> b;
    if (a > b) swap(a, b);
    int d = b - a;
    if (d % 2 == 0)
      d--;
    if (d > x)
      d = x;
    else if (d <= 0)
      d = 1;
    if (d == 0) {
      d = 1;
      if (a + 1 <= n) a++;
      if (b - 1 >= 1) b--;
    } else if (d % 2 == 0)
      d++;
    cout << d << "\n";
  }
  return 0;
}
