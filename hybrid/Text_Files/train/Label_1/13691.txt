#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
int main() {
    ll n, q; cin >> n >> q;
    vector<ll> a(n);
    for (ll& x : a) { cin >> x; }
    vector<ll> k(q);
    for (ll& x : k) { cin >> x; }
    ll cnt_alive = 0, cur = 0;
    for (ll i = 1; i <= q; i++) {
        while (cur < n && (cnt_alive + max(1ll, (k[i] - (cur ? a[cur] : LLONG_MAX)))) < a[cur]) cnt_alive++, cur++;
        cout << cnt_alive << '\n';
    }
}
