import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] s = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            s[i] = in.nextInt();
        }
        int m = in.nextInt();
        int[] p1 = new int[n + 1];
        int[] p2 = new int[n + 1];
        for (int i = 0; i < m; i++) {
            String action = in.next();
            int team = in.nextInt();
            if (action.equals("p")) {
                if (team == 1) {
                    p1[i] = s[i];
                } else {
                    p2[i] = s[i];
                }
            }
        }
        Arrays.sort(p1);
        Arrays.sort(p2);
        int sum1 = 0;
        int sum2 = 0;
        for (int i = 0; i < m; i++) {
            if (p1[i] > 0) {
                sum1 += p1[i];
            }
            if (p2[i] > 0) {
                sum2 += p2[i];
            }
        }
        System.out.println(sum1 - sum2);
    }
}
