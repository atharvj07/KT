#include <iostream>
using namespace std;
int main() {
  string a, b;
  cin >> a >> b;
  int d = 0, count[2] = {};
  for (char c : a) count[c - '0']++;
  for (char c : b.substr(0, a.size())) count[c - '0']--;
  d += count[0] * count[0] + count[1] * count[1];
  for (size_t i = a.size(); i < b.size(); i++) {
    count[b[i - a.size()] - '0']++;
    count[b[i] - '0']--;
    d += count[0] * count[0] + count[1] * count[1];
  }
  cout << d << endl;
}
