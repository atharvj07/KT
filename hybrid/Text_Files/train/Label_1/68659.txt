import sys

input = sys.stdin.readlines()

n,m = map(int,input[0].split())
matrix=[list(map(int,x.split())) for x in input[1:n+1]]
ans=[0]*int(input[n+1])

for x in range(n+2,len(input)):
    l,r=map(int,input[x].split())
    ans[x-n-2]="No"
    for y in range(m):
        flag=True
        for z in range(l,r):
            if matrix[z-1][y]>matrix[z][y]: flag=False
        if flag:
            ans[x-n-2]="Yes"
            break

print("\n".join(ans))
