n = int(input())
a = list(map(int, input().split()))

a = list(set(a))  # Remove duplicates to deal with strictly increasing subsequences
a.sort()  # Sort to ensure increasing order

# Use a set to keep track of XOR results
xor_results = set([0])

# Generate all possible increasing subsequences and their XOR
for val in a:
    new_results = set()
    for result in xor_results:
        new_results.add(result ^ val)
    xor_results.update(new_results)

# Prepare and print the output
xor_results = sorted(list(xor_results))
print(len(xor_results))
print(' '.join(map(str, xor_results)))
