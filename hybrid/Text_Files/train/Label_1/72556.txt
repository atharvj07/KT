n, u = map(int, input().split())
e = [int(x) for x in input().split()]
mx = 0
for i in range(n - 2):
    for j in range(i + 1, n - 1):
        for k in range(j + 1, n):
            if e[k] - e[i] <= u:
                mx = max(mx, (e[k] - e[j]) / (e[k] - e[i]))
if mx != 0:
    print(mx)
else:
    print(-1)
