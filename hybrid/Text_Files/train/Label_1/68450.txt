def calculate_max_beauty(ribbon, turns):
    from collections import Counter
    count = Counter(ribbon)
    max_freq = max(count.values())
    if len(count) == 1 and turns == 1:  # Special case when all characters are the same and only 1 turn
        return max_freq - 1
    return min(len(ribbon), max_freq + turns)

n = int(input().strip())
ribbons = [input().strip() for _ in range(3)]

beauties = [calculate_max_beauty(ribbon, n) for ribbon in ribbons]

if beauties.count(max(beauties)) > 1:
    print("Draw")
else:
    winner = ["Kuro", "Shiro", "Katie"][beauties.index(max(beauties))]
    print(winner)
