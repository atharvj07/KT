a, t_a = map(int, input().split())
b, t_b = map(int, input().split())
h, m = map(int, input().split(':'))

simion_start = h * 60 + m
simion_end = simion_start + t_a

buses_met = 0

for start_time in range(300, 24*60, b):  # from 05:00 AM to 11:59 PM
    end_time = start_time + t_b
    if end_time > simion_start and start_time < simion_end:
        buses_met += 1

print(buses_met)
