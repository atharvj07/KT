from collections import defaultdict, deque
def moves(n, s, E):
    L = [[] for _ in range(n)]
    for i in range(n):
        for j in range(i+1,n):
            if E[i][j]==E[s[0]-1][s[1]-1]: L[i].append(j); L[j].append(i)
    d = defaultdict(list)
    for i in range(n):
        if i not in s:
            q, L[i] = deque([i]), L[i][:]
            while q:
                u = q.popleft()
                for e in L[u]:
                    if e not in d:
                        d[e] = [u]
                        q.append(e)
                    else: d[e].append(u)
    ans = []
    for e in sorted(d, key=lambda x:len(d[x])):
        for u in d[e]:
            if u not in s:
                ans.append([u+1,s[s.index(e)+1]+1])
                s[s.index(e)], s[s.index(u)]=s[s.index(e)+1],s[s.index(e)]
                break
    print(len(ans));
    for ans1 in ans: print(*ans1)
    return

n=int(input())
s=[int(x)-1 for x in input().split()]
E=[list(input()) for _ in range(n)]
moves(n,s,E)
