n, k = map(int, input().split())
a = list(map(int, input().split()))

a.sort()
left, right = 0, 0
current_count = 1
min_moves = float('inf')

while right < n:
    while right + 1 < n and a[right] == a[right + 1]:
        current_count += 1
        right += 1
    if current_count >= k:
        min_moves = 0
        break
    if right + 1 < n:
        increase = (a[right + 1] - a[right]) * current_count
        decrease = (a[right] - a[left]) * (k - current_count)
        min_moves = min(min_moves, increase + decrease)
    current_count += 1
    right += 1
    while current_count > k:
        current_count -= 1
        left += 1

print(min_moves)
