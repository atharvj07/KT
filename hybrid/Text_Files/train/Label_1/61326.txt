t = int(input().strip())
for _ in range(t):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort(reverse=True)
    teams = 0
    members = 0
    for skill in a:
        members += 1
        if members * skill >= x:
            teams += 1
            members = 0
    print(teams)
