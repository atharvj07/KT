#include<bits/stdc++.h>
using namespace std;
const int maxn=1e5+5,maxk=15;
typedef long long ll;
int n,k,a[maxn],c[maxk],res;
ll cnt[maxk][2];
int main(){
    cin>>n>>k;
    for(int i=0;i<n;i++) scanf("%d",&a[i]);
    for(int i=0;i<n;i++){
        ll mask=0,tmp=a[i];
        for(int j=0;j<k;j++){
            if(tmp&1) cnt[j][mask]++,tmp^=1<<j;
            else mask^=1<<j;
        }
    }
    for(int i=0;i<n;i++){
        ll mask=0,tmp=a[i];
        for(int j=0;j<k;j++){
            if(!(tmp&1)) res+=cnt[j][mask],tmp^=1<<j;
            else mask^=1<<j;
        }
    }
    cout<<res<<endl;
    return 0;
}
