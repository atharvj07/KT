#include <bits/stdc++.h>
using namespace std;
long long qpow(long long a, long long b, long long p) {
  long long ans = 1;
  while (b) {
    if (b & 1) ans = ans * a % p;
    a = a * a % p;
    b >>= 1;
  }
  return ans;
}
int main() {
  long long n, b;
  cin >> n >> b;
  long long ans = 0;
  for (long long i = 5; i <= n; i *= 5) ans += n / i;
  for (long long i = 2; i <= n; i *= 2) ans -= n / i;
  cout << ans << endl;
  return 0;
}
