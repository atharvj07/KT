def can_make_interesting_problemset():
    n, k = map(int, input().split())
    problems = [input().strip() for _ in range(n)]
    unique_problems = set(problems)
    
    for problem_set in unique_problems:
        can_select = True
        for team in range(k):
            known_problems = 0
            for problem in problem_set:
                if problem[team] == '1':
                    known_problems += 1
            if known_problems > len(problem_set) / 2:
                can_select = False
                break
        if can_select:
            return "YES"
    return "NO"

print(can_make_interesting_problemset())
