#include <iostream>
#include <cstdio>
#include <cstring>

using namespace std;

char a[505][505];
int dx[] = {-1, 0, 1, 0}, dy[] = {0, 1, 0, -1};
int n, m, k;
bool vis[505][505];

void dfs(int x, int y, int& cnt)
{
    cnt++;
    vis[x][y] = true;
    for (int i = 0; i < 4; i++)
    {
        int tx = x + dx[i], ty = y + dy[i];
        if (tx >= 0 && tx < n && ty >= 0 && ty < m && a[tx][ty] != '#' && !vis[tx][ty])
            dfs(tx, ty, cnt);
    }
}

int main()
{
    cin >> n >> m >> k;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            cin >> a[i][j];
    int cnt = 0;
    dfs(0, 0, cnt);
    if (cnt != n * m)
        return cout << "NO", 0;
    int cnt2 = 0;
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            if (cnt2 == k)
                break;
            if (a[i][j] == '.' && (i == 0 || i == n - 1 || j == 0 || j == m - 1))
            {
                cnt2++;
                a[i][j] = 'X';
            }
        }
        if (cnt2 == k)
            break;
    }
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
            cout << a[i][j];
        cout << endl;
    }
    return 0;
}
