import java.util.PriorityQueue;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int[][] g = new int[n][n];
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            a--;
            b--;
            g[a][b] = g[b][a] = 1;
        }
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        pq.add(0);
        int[] dist = new int[n];
        for (int i = 0; i < n; i++) {
            dist[i] = Integer.MAX_VALUE;
        }
        dist[0] = p[0];
        while (!pq.isEmpty()) {
            int u = pq.poll();
            for (int i = 0; i < n; i++) {
                if (g[u][i] == 1 && dist[i] > dist[u] + p[i]) {
                    dist[i] = dist[u] + p[i];
                    pq.add(i);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (dist[i] != Integer.MAX_VALUE) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
