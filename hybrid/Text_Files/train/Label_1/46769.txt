import java.util.Scanner;

public class InnopolisConstruction {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] hills = new int[n];
        int[] houses = new int[n];
        for (int i = 0; i < n; i++) {
            hills[i] = scanner.nextInt();
            houses[i] = 0;
        }

        int mid = (n + 1) / 2;
        int[] minTime = new int[mid];

        // Calculate minimum time for each number of houses
        for (int i = 1; i <= mid; i++) {
            int currentMinTime = 0;
            int j = 0;
            int k = n - 1;

            while (j < k) {
                if (hills[j] > hills[k]) {
                    currentMinTime += hills[j] - i;
                    j++;
                } else if (hills[j] < hills[k]) {
                    currentMinTime += hills[k] - i;
                    k--;
                } else {
                    currentMinTime += Math.max(0, hills[j] - i);
                    j++;
                    k--;
                }
            }

            if (j == k) {
                currentMinTime += Math.max(0, hills[j] - i);
            }

            minTime[i - 1] = currentMinTime;
        }

        // Print the results
        for (int time : minTime) {
            System.out.print(time + " ");
        }
    }
}
