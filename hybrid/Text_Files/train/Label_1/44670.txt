import java.util.Scanner;

public class Main {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[4 * n + 1];
        int[] y = new int[4 * n + 1];
        int cnt = 0;
        for (int i = 0; i < 4 * n + 1; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
            if (x[i] == 0 || x[i] == 2 || y[i] == 0 || y[i] == 2) {
                cnt++;
            }
        }
        for (int i = 0; i < 4 * n + 1; i++) {
            if (x[i] != 0 && x[i] != 2 && y[i] != 0 && y[i] != 2) {
                System.out.println(x[i] + " " + y[i]);
                break;
            }
        }
    }
}
