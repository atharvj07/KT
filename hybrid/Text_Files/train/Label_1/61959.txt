from collections import defaultdict

n = int(input())
s = input()
x, y = map(int, input().split())

sumx, sumy = 0, 0
msumx, msumy = 0, 0
ans = 1e9
pre = defaultdict(int)

for i in range(n):
    sumx += (1 if s[i] == "R" else -1 if s[i] == "L" else 0)
    sumy += (1 if s[i] == "U" else -1 if s[i] == "D" else 0)
    msumx = max(msumx, sumx)
    msumy = max(msumy, sumy)
    if sumx == x and sumy == y:
        ans = min(ans, i + 1)
    if sumx - msumx == x and sumy - msumy == y:
        if (sumx - msumx, sumy - msumy) not in pre:
            pre[sumx - msumx, sumy - msumy] = i
        else:
            ans = min(ans, i - pre[sumx - msumx, sumy - msumy])

if ans == 1e9:
    print(-1)
else:
    print(ans)
