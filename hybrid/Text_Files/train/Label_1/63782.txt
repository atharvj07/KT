from collections import deque

n = int(input())
h = list(map(int, input().split()))

def min_jumps(h):
    dp = [float('inf')] * n
    dp[0] = 0
    queue = deque([0])
    
    while queue:
        i = queue.popleft()
        for j in range(i + 1, n):
            if h[j] > min(h[i], h[j-1]) and h[j] < max(h[i], h[j-1]):
                break
            if dp[j] > dp[i] + 1:
                dp[j] = dp[i] + 1
                queue.append(j)
    return dp[-1]

print(min_jumps(h))
