t = int(input().strip())
for _ in range(t):
    n, m, a, b = map(int, input().strip().split())

    if n * a != m * b:
        print("NO")
        continue

    print("YES")
    matrix = [[0 for _ in range(m)] for _ in range(n)]
    for i in range(n):
        start = (i * a) % m
        for j in range(a):
            matrix[i][(start + j) % m] = 1
    
    for row in matrix:
        print(''.join(map(str, row)))
