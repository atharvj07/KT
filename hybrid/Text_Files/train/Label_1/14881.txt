#include <iostream>
#include <vector>
using namespace std;
int main() {
  string S;
  cin >> S;
  int n = S.size();
  vector<int> l(n, 0), r(n, 0);
  for (int i = 0; i < n; i++) l[i] = i ? l[i - 1] : 0;
  for (int i = n - 1; i >= 0; i--) {
    r[i] = i != n - 1 ? r[i + 1] : 0;
    if (i + 1 < n && S[i] == S[i + 1] && S[i + 1] == S[i + 2]) { r[i] += 1; }
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (l[i] && r[i]) {
      ans += l[i] * (long long)r[i];
    }
  }
  cout << ans << endl;
}
