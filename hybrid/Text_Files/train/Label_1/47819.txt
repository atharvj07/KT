import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
  public static void main(String[] args) throws Exception {
    BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(reader.readLine());
    for (int i = 0; i < n; i++) {
      StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
      int p0x = Integer.parseInt(tokenizer.nextToken());
      int p0y = Integer.parseInt(tokenizer.nextToken());
      int p1x = Integer.parseInt(tokenizer.nextToken());
      int p1y = Integer.parseInt(tokenizer.nextToken());
      int p2x = Integer.parseInt(tokenizer.nextToken());
      int p2y = Integer.parseInt(tokenizer.nextToken());
      int p3x = Integer.parseInt(tokenizer.nextToken());
      int p3y = Integer.parseInt(tokenizer.nextToken());
      long s1x1 = Math.min(p0x, p1x);
      long s1x2 = Math.max(p0x, p1x);
      long s1y1 = Math.min(p0y, p1y);
      long s1y2 = Math.max(p0y, p1y);
      long s2x1 = Math.min(p2x, p3x);
      long s2x2 = Math.max(p2x, p3x);
      long s2y1 = Math.min(p2y, p3y);
      long s2y2 = Math.max(p2y, p3y);
      System.out.println((s1x1 < s2x2 && s2x1 < s1x2 && s1y1 < s2y2 && s2y1 < s1y2) ? 1 : 0);
    }
  }
}
