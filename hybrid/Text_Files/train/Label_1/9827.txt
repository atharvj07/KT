#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  vector<int> p(q), l(q), r(q);
  for (int i = 0; i < q; i++) {
    int t;
    cin >> t;
    if (t == 1) {
      cin >> p[i];
    } else {
      cin >> l[i] >> r[i];
    }
  }
  int width = n;
  for (int i = q - 1; i >= 0; i--) {
    if (l[i] != 0 || r[i] != width) {
      cout << width - l[i] - (width - r[i]) << '\n';
    }
    if (p[i] != 0) { width = p[i]; }
  }
  return 0;
}
