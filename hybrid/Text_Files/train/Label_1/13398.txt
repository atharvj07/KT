#include <iostream>

const int maxn = 1e5+10;

int a[maxn];
int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n, k;
        std::cin >> n >> k;
        if(k > (1LL<<(n-1)) - 1) { puts("-1"); continue; }
        for(int i=0; i<n; ++i) a[i] = i + 1;
        k--;
        for(int i=n; i>=1; --i) {
            if((1LL<<(i-1)) <= k) {
                std::swap(a[i], a[i-1]);
                k -= 1LL<<(i-1);
            }
        }
        for(int i=0; i<n; ++i) std::cout << a[i] << ' ';
        std::cout << '\n';
    }
    return 0;
}
