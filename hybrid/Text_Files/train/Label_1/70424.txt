from collections import Counter

s = input()
n = len(s)
b = [0] * n
d = {}
l = [[-1] * 2 for i in range(n)]
for i in range(n):
    if s[i] == '(':
        b[i] = -1
    if i > 0:
        b[i] += b[i - 1]
for i in range(n):
    if b[i] not in d:
        d[b[i]] = []
    d[b[i]].append(i)
    l[i][0] = d[b[i]][-2] if len(d[b[i]]) > 1 else -1
    l[i][1] = d[b[i]][-1]
l[n - 1][0] = l[n - 1][1]
for i in range(n - 2, -1, -1):
    if l[i][0] == -1:
        l[i][0] = l[i + 1][0]
    if l[i][1] == -1:
        l[i][1] = l[i + 1][1]
m = int(input())
for i in range(m):
    li, ri = map(int, input().split())
    li -= 1
    ri -= 1
    if b[li] == 0:
        print(0)
    else:
        ans = 2 * (ri - max(li, l[ri][0]) + 1)
        if b[li] < 0 and li > 0:
            ans += b[li - 1]
        print(ans)
