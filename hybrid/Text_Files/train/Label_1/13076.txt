#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s, t;
    cin >> s >> t;
    vector<int> dif;
    for (int i = 0; i < n; i++) {
        if (s[i] != t[i]) dif.push_back(i);
    }
    int m = dif.size();
    int swaps = m;
    int pos[2] = {-1, -1};
    if (m >= 2) {
        swaps = 1;
        pos[0] = dif[0];
        pos[1] = dif[1];
    }
    cout << swaps << endl;
    cout << pos[0] + 1 << ' ' << pos[1] + 1 << endl;
}
