#include<bits/stdc++.h>
using namespace std;
const int mx=2e5+5;
typedef long long ll;
ll ar[mx],cost[mx],dp[mx],n;
int main(){
	cin>>n;
	for(int i=0;i<n;i++){
		cin>>ar[i];
	}
	ll ans=0;
	for(int i=0;i<n;i++){
		if(ar[i]){
			cost[i]=ar[i]*(ar[i]+1)/2;
			ans+=cost[i];
			ar[i]=0;
			ar[i+1]+=ar[i];
		}
	}
	dp[n]=0;
	for(int i=n-1;i>=0;i--){
		dp[i]=dp[i+1]+cost[i];
		if(i+2<n) dp[i]=min(dp[i],dp[i+2]);
	}
	cout<<ans+dp[0]<<endl;
	return 0;
}
