#include <iostream>
#include <vector>
#include <climits>

using namespace std;

int main() {
    int n, min_flips = INT_MAX;
    cin >> n;
    vector<int> front(n), back(n);
    vector<pair<int, int>> pairs(2 * n);
    for (int i = 0; i < n; i++) {
        cin >> front[i] >> back[i];
        pairs[front[i] - 1] = {front[i], back[i]};
        pairs[back[i] - 1] = {back[i], front[i]};
    }
    for (int i = 0; i < 2 * n; i++) {
        if (i != pairs[i].first - 1) min_flips = min(min_flips, 1);
        else if (i != pairs[i].second - 1) min_flips = min(min_flips, 2);
    }
    min_flips == INT_MAX ? cout << "-1" : cout << min_flips;
    return 0;
}
