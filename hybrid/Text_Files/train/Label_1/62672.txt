import sys
n, k = map(int, input().split())
s = input()
a = [int(x) for x in s]
p = 10 ** 9 + 7
dp = [[0] * n for _ in range(k+1)]
for i in range(n):
    dp[0][i] = a[i]
for j in range(1, k+1):
    for i in range(n):
        if j+1 <= i:
            dp[j][i] = (dp[j][i] + dp[j-1][i-j-1] * a[i]) % p
        if i > 0:
            dp[j][i] = (dp[j][i] + dp[j][i - 1]) % p
print(dp[k][n-1] % p)
