from math import gcd

def find_average_distance(n, destinations):
    destinations.sort()
    total_distance = sum((destinations[i] - destinations[i - 1]) * i * (n - i) for i in range(1, n))
    numerator = total_distance * 2 + sum(destinations)
    denominator = n
    common_divisor = gcd(numerator, denominator)
    return numerator // common_divisor, denominator // common_divisor

n = int(input())
destinations = list(map(int, input().split()))
numerator, denominator = find_average_distance(n, destinations)
print(numerator, denominator)
