#include <bits/stdc++.h>
#define int long long
using namespace std;

const int MAXN = 2e5 + 10, MAXQ = 2e5 + 10;

int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, a = 0, b = 0;
    cin >> n >> k;
    vector<int> S(MAXN, 0), occ(MAXN, 0);
    set<int> active;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      S[x]++;
      occ[x]++;
      active.insert(x);
    }
    while (k--) {
      b = *prev(active.end());
      int p = 0;
      for (int i = 0; i < MAXN; i++) {
        if (occ[i] == 0) {
          if (p == 0) {
            a = i;
          }
          p++;
        }
      }
      int c = (a + b) / 2 + (a + b) % 2;
      active.insert(c);
      S[c]++;
      occ[c]++;
    }
    cout << active.size() << "\n";
  }
}
