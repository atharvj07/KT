import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {

    static class Student implements Comparable<Student> {
        long r, w;

        public Student(long r, long w) {
            this.r = r;
            this.w = w;
        }

        @Override
        public int compareTo(Student other) {
            return Long.compare(r * other.w, w * other.r);
        }
    }

    static int n, c, d;
    static Student[] students;
    static List<int[]> goodTeams;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        c = Integer.parseInt(line[1]);
        d = Integer.parseInt(line[2]);
        students = new Student[n];
        for (int i = 0; i < n; i++) {
            line = br.readLine().split(" ");
            students[i] = new Student(Long.parseLong(line[0]) - c, Long.parseLong(line[1]) - d);
        }
        Arrays.sort(students);
        goodTeams = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (students[i].compareTo(students[j]) > 0 &&
                            students[j].compareTo(students[k]) > 0 &&
                            students[k].compareTo(students[i]) > 0) {
                        int[] team = {i + 1, j + 1, k + 1};
                        Arrays.sort(team);
                        goodTeams.add(team);
                    }
                }
            }
        }
        System.out.println(goodTeams.size());
    }
}
