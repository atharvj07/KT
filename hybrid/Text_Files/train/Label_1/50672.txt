import java.util.*;
import java.lang.*;

class Main {
	static int[] f = new int[100005];
	static int[] g = new int[100005];
	static int[] h = new int[100005];
	static int[][] t = new int[1005][1005];
	static int tcol;
	static boolean cycle;
	static int n;
	static int first;
	static int second;

	static void dfs(int u) {
		int i;
		for (i = 1; i <= n; i++) {
			if (t[u][i] != 0) {
				if (t[u][i] == 1) {
					t[u][i] = 2;
					dfs(i);
				}
				if (t[u][i] == 2) {
					cycle = true;
				}
				if (t[u][i] == 3) {
					if (!cycle) {
						first = u;
						second = i;
					}
					t[u][i] = 4;
					t[i][u] = 4;
					dfs(i);
				}
			}
		}
	}

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		n = sc.nextInt();
		int i;
		int b;
		for (i = 1; i <= n; i++) {
			g[i] = 0;
			h[i] = 0;
			f[i] = sc.nextInt();
		}

		for (i = 1; i <= n; i++) {
			if (g[i] == 0) {
				tcol++;
				g[i] = tcol;
				t[tcol][f[i]] = 1;
				h[f[i]] = tcol;
				dfs(i);
			}
		}
		if (cycle) {
			System.out.println(-1);
		} else {
			if (!cycle && first != 0) {
				List<Integer> vt = new ArrayList<Integer>();
				int v1 = first, v2 = second;
				while (v1 != v2) {
					vt.add(v1);
					v1 = f[v1];
				}
				vt.add(v2);
				System.out.println(vt.size());
				for (i = vt.size() - 1; i >= 0; i--) {
					System.out.print(vt.get(i) + " ");
					g[vt.get(i)] = vt.size() - i;
				}
				System.out.println();
				for (i = 1; i <= vt.size(); i++) {
					h[i] = vt.get(i - 1);
					System.out.print(h[i] + " ");
				}
			} else {
				System.out.println(1);
				for (i = 1; i <= n; i++) {
					System.out.print(1 + " ");
				}
				System.out.println();
				for (i = 1; i <= n; i++) {
					System.out.print(f[i] + " ");
				}
			}
		}
	}
}
