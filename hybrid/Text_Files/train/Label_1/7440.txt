#include <bits/stdc++.h>
using namespace std;
typedef long long int64;
int main() {
  int64 q;
  cin >> q;
  const int64 MOD = 1e9 + 7;
  while (q--) {
    int64 n, f;
    cin >> n >> f;
    if (f == 1) {
      cout << 1 << endl;
    } else if (f == 2) {
      cout << n - 1 << endl;
    } else if (f == n) {
      cout << 1 << endl;
    } else {
      int64 ans = f * (n - f) % MOD;
      ans = (ans * (n - 1)) % MOD;
      cout << ans << endl;
    }
  }
  return 0;
}
