#include <iostream>
#include <string>
using namespace std;
string s;
bool vis[300001];
int ans,n,m;
int main(){
    int t;
    cin>>t;
    for(int i=1;i<=t;++i){
        cin>>n>>s;
        if(n%2==1){
            cout<<n<<endl;
            continue;
        }
        m=(n+1)/2;
        for(int i=0;i<n;++i)vis[i]=0;
        for(int i=0,j;i<n;++i){
            j=i;
            while(!vis[j]){
                vis[j]=1;
                if(s[j]=='>')j=(j+1)%n;
                else j=(j-1+n)%n;
            }
        }
        ans=0;
        for(int i=0;i<n;++i)if(!vis[i])ans++;
        cout<<ans+(n%2)*2<<endl;
    }
}
