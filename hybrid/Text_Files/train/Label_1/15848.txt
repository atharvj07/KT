#include <iostream>
#include <vector>
using namespace std;
int main() {
  int have[11], w;
  string s;
  cin >> s >> w;
  for (int i = 0; i <= 9; i++) {
    have[i] = s[i] == '1';
  }
  if (w <= 1 || (w == 2 && have[1])) {
    cout << "YES\n";
    for (int i = 0; i < w; i++) { cout << 1 << ' '; }
    cout << endl;
    return 0;
  }
  if (have[0] && w % 2 == 0) {
    cout << "YES\n";
    for (int i = 0; i < w; i++) { cout << 0 << ' '; }
    cout << endl;
    return 0;
  }
  if (have[1] && w % 2 != 0) {
    cout << "YES\n";
    cout << 1 << ' ';
    for (int i = 0; i < w - 1; i++) { cout << 0 << ' '; }
    cout << endl;
    return 0;
  }
  for (int i = 2; i <= 9; i++) {
    if (have[i]) {
      cout << "YES\n";
      w -= i;
      cout << i << ' ';
      while (w >= i) {
        cout << 0 << ' ';
        w -= i;
      }
      if (w > 0) { cout << w << ' '; }
      cout << endl;
      return 0;
    }
  }
  cout << "NO\n";
  return 0;
}
