import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        sc.nextLine();
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            grid[i] = sc.nextLine().toCharArray();
        }
        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = -1;
            }
        }
        Queue<Integer> q = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] >= '1' && grid[i][j] <= '3') {
                    q.add(i * m + j);
                    dist[i][j] = 0;
                }
            }
        }
        int[] dx = {1, -1, 0, 0};
        int[] dy = {0, 0, 1, -1};
        while (!q.isEmpty()) {
            int u = q.poll();
            int x = u / m;
            int y = u % m;
            for (int k = 0; k < 4; k++) {
                int nx = x + dx[k];
                int ny = y + dy[k];
                if (nx >= 0 && nx < n && ny >= 0 && ny < m) {
                    if (dist[nx][ny] == -1 && grid[nx][ny] != '#') {
                        dist[nx][ny] = dist[x][y] + 1;
                        q.add(nx * m + ny);
                    }
                }
            }
        }
        boolean reachable = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] != '#' && dist[i][j] == -1) {
                    reachable = false;
                }
            }
        }
        int ans = 0;
        if (reachable) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (grid[i][j] == '.' && dist[i][j] > 0) {
                        ans++;
                    }
                }
            }
        } else {
            ans = -1;
        }
        System.out.println(ans);
    }
}
