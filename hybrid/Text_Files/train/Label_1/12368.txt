#include<bits/stdc++.h>
using namespace std;
#define mod 998244353
typedef long long ll;
int main(){
    ll n,k;
    cin>>n>>k;
    ll a[n+1];
    for(ll i=0;i<n;i++)cin>>a[i];
    sort(a,a+n);
    ll ans=0;
    for(ll i=0;i<n-k+1;i++){
        ans=(ans%mod+(a[i+k-1]-a[i])%mod)%mod;
    }
    cout<<ans<<endl;
}
