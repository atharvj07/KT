#include <iostream>
#include <vector>
#include <climits>
#include <algorithm>
using namespace std;

int main() {
    long long n, sx, sy;
    cin >> n >> sx >> sy;
    vector<pair<long long, long long>> students(n);
    for (long long i = 0; i < n; i++) {
        cin >> students[i].first >> students[i].second;
    }
    sort(students.begin(), students.end());
    long long l1 = LLONG_MAX, l2 = -LLONG_MAX, l3 = LLONG_MAX, l4 = -LLONG_MAX;
    for (long long i = 0; i < n; i++) {
        l1 = min(l1, students[i].first - sx);
        l2 = max(l2, students[i].first - sx);
        l3 = min(l3, students[i].second - sy);
        l4 = max(l4, students[i].second - sy);
    }
    long long maxCnt = 0, px, py;
    for (long long i = 0; i < n; i++) {
        long long x = students[i].first - sx;
        long long y = students[i].second - sy;
        if (l1 <= x && x <= l2 && l3 <= y && y <= l4) {
            maxCnt++;
            px = students[i].first;
            py = students[i].second;
        }
    }
    cout << maxCnt << endl;
    cout << px << " " << py << endl;
    return 0;
}
