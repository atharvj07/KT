#include <iostream>
#include <string>

using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  int n = s.length();
  string p;
  for (int i = 0; i < n; i++) {
    if (s[i] == t[i])
      p += s[i];
    else
      p += (s[i] + t[i]) % 2 + '0';
  }
  if (p == s || p == t)
    cout << p << endl;
  else
    cout << "impossible" << endl;
  return 0;
}
