import java.util.Scanner;

class Main {

    public static void main(String[] a) {
        Scanner sc = new Scanner(System.in);
        long a1 = sc.nextLong();
        long b1 = sc.nextLong();
        long k = sc.nextLong();
        long ans = (b1 - a1) / k + (a1 % k == 0 ? 1 : 0);
        System.out.println(ans);
    }
}
