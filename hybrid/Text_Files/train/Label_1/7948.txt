#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long l, r;
        cin >> l >> r;
        if (l % 2 == 0) {
            cout << l << " " << l + 1 << "\n";
            continue;
        }
        if (l == 1) {
            cout << "-1 -1\n";
            continue;
        }
        if (2 * l <= r) {
            cout << l << " " << 2 * l << "\n";
        } else {
            cout << "-1 -1\n";
        }
    }
}
