import java.util.Scanner;

public class Main {
    private static final char[] DIR = new char[]{'U', 'D', 'L', 'R'};
    private static int w, h, n;
    private static int x1, y1, x2, y2;
    private static int[][][] step;
    private static int[][] pos;
    private static int[] px, py;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        w = sc.nextInt();
        h = sc.nextInt();
        n = sc.nextInt();
        x1 = sc.nextInt();
        y1 = sc.nextInt();
        x2 = sc.nextInt();
        y2 = sc.nextInt();
        step = new int[w + 2][h + 2][2];
        pos = new int[n][2];
        px = new int[n];
        py = new int[n];
        for (int i = 0; i < n; i++) {
            pos[i][0] = sc.nextInt();
            pos[i][1] = sc.nextInt();
            px[i] = pos[i][0];
            py[i] = pos[i][1];
            String s = sc.next();
            int t = sc.nextInt();
            for (int j = 0; j < t; j++) {
                for (char c : s.toCharArray()) {
                    go(i, c);
                }
            }
        }
        solve();
        int min = Integer.MAX_VALUE;
        for (int i = 0; i <= w + 1; i++) {
            min = Math.min(min, step[i][0][0] + step[i][h + 1][0]);
            min = Math.min(min, step[0][i][0] + step[w + 1][i][0]);
            min = Math.min(min, step[i][0][1] + step[i][h + 1][1]);
            min = Math.min(min, step[0][i][1] + step[w + 1][i][1]);
        }
        System.out.println(min);
    }

    private static void go(int i, char c) {
        switch (c) {
            case 'U':
                if (py[i] > 0) {
                    step[px[i]][py[i]][0]++;
                    py[i]--;
                    step[px[i]][py[i]][0]--;
                }
                break;
            case 'D':
                if (py[i] < h - 1) {
                    step[px[i]][py[i]][0]++;
                    py[i]++;
                    step[px[i]][py[i]][0]--;
                }
                break;
            case 'L':
                if (px[i] > 0) {
                    step[px[i]][py[i]][1]++;
                    px[i]--;
                    step[px[i]][py[i]][1]--;
                }
                break;
            case 'R':
                if (px[i] < w - 1) {
                    step[px[i]][py[i]][1]++;
                    px[i]++;
                    step[px[i]][py[i]][1]--;
                }
                break;
        }
    }

    private static void solve() {
        for (int i = 1; i <= w; i++) {
            for (int j = 1; j <= h; j++) {
                step[i][j][0] = step[i - 1][j][0] + step[i][j - 1][0] - step[i - 1][j - 1][0];
                step[i][j][1] = step[i - 1][j][1] + step[i][j - 1][1] - step[i - 1][j - 1][1];
                if (i >= x1 && i <= x2 && j >= y1 && j <= y2) {
                    step[i][j][0] += step[i][y1 - 1][0] - step[i][y2][0];
                    step[i][j][1] += step[x1 - 1][j][1] - step[x2][j][1];
                }
            }
        }
    }
}
