import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] arr = new int[n + 1][m + 1];
        int[] min = new int[n + 1];
        int[] min2 = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                arr[i][j] = sc.nextInt();
                if (arr[i][j] < min[i]) {
                    min2[i] = min[i];
                    min[i] = arr[i][j];
                } else if (min[i] != min2[i] && arr[i][j] < min2[i]) {
                    min2[i] = arr[i][j];
                }
            }
        }
        int ind1 = 0;
        int ind2 = 0;
        int max = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (max < Math.max(min[i], min[j])) {
                    max = Math.max(min[i], min[j]);
                    if (min[j] > min[i]) {
                        ind1 = i;
                        ind2 = j;
                    } else {
                        ind1 = j;
                        ind2 = i;
                    }
                } else if (max == Math.max(min[i], min[j])) {
                    if (min2[i] > min[j]) {
                        ind1 = i;
                        ind2 = j;
                    } else if (min2[j] > min[i]) {
                        ind1 = j;
                        ind2 = i;
                    }
                }
            }
        }
        System.out.println(ind1 + " " + ind2);
    }
}
