#include <cstdio>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    scanf(" %d %d ", &n, &k);
    vector<int> a(k), b(k);
    for (int i = 0; i < k; i++) scanf(" %d ", &a[i]), b[i] = a[i];
    sort(a.begin(), a.end());
    for (int i = 0; i < k; i++) {
        int p1 = n - (lower_bound(a.begin(), a.end(), a[i]) - a.begin()) % n;
        for (int j = 0; j < k; j++)
            if (b[j] == a[i] && p1 > 0) printf("%d ", b[j]), b[j] = 2e9, p1--;
        for (int j = n * n; j >= 1; j--)
            if (p1 > 0 && b[i] != j) printf("%d ", j), p1--;
    }
    return 0;
}
