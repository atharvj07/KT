import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a1 = scanner.nextInt();
        int a2 = scanner.nextInt();

        int minutes = 0;
        while (a1 > 0 && a2 > 0) {
            if (a1 <= a2) {
                a1++;
                a2 -= 2;
            } else {
                a1 -= 2;
                a2++;
            }
            minutes++;
        }

        System.out.println(minutes);
    }
}
