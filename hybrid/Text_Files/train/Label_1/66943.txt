from collections import deque
n, m, k = map(int, input().split())
x = [1e9] + list(map(int, input().split())) + [1e9]
y = [1e9] + list(map(int, input().split())) + [1e9]
dx = [1, 0, -1, 0]
dy = [0, 1, 0, -1]

dist = [[-1]*(m+2) for i in range(n+2)]
q = deque()
q.append((1, 1))
dist[1][1] = 0
while q:
    cx, cy = q.popleft()
    for i in range(4):
        nx, ny = cx+dx[i], cy+dy[i]
        if dist[nx][ny] == -1:
            dist[nx][ny] = dist[cx][cy]+1
            q.append((nx, ny))
            if (nx, ny) in zip(x,y):
                print(dist[nx][ny]-1)
            elif nx == 0 or nx == n+1 or ny == 0 or ny == m+1:
                break
print(*[-1]*(k - len(x)+1))
