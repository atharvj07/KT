def solve_chess_tournament():
    t = int(input().strip())
    for _ in range(t):
        n = int(input().strip())
        s = input().strip()

        if '2' in s and s.count('2') <= 2:
            print("NO")
        else:
            print("YES")
            for i in range(n):
                for j in range(n):
                    if i == j:
                        print("X", end="")
                    elif s[i] == '1' or s[j] == '1':
                        print("=", end="")
                    elif (s[i] == '2' and s[j] == '2') and ((i < j and j - i <= n // 2) or (i > j and i - j > n // 2)):
                        print("+", end="")
                    elif s[i] == '2' and s[j] == '2':
                        print("-", end="")
                    else:
                        print("=", end="")
                print()
                
solve_chess_tournament()
