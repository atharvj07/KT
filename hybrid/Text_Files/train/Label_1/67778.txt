from collections import defaultdict, Counter
class Solution:
    def __init__(self):
        self.dp = {}
    def isPalindrome(self, s):
        if s in self.dp: return self.dp[s]
        n = len(s)
        self.dp[s] = s==s[::-1]
        return self.dp[s]
    def solve(self, arr):
        cnt = Counter()
        for w in arr:
            cnt[w] += 1
        ans = 0
        for w in arr:
            ws = set(w)
            lw = len(w)
            if self.isPalindrome(w):
                ans += cnt[w] * (cnt[w] - 1) // 2
                continue
            rev = w[::-1]
            if rev in cnt:
                ans += cnt[w] * cnt[rev]
            for c in ws:
                if c != w[0]:
                    pre = w[1:]
                    if self.isPalindrome(pre) and rev.endswith(pre):
                        ans += cnt[w] * cnt[rev[:-len(pre)]]
                if c != w[-1]:
                    suf = w[:-1]
                    if self.isPalindrome(suf) and rev.startswith(suf):
                        ans += cnt[w] * cnt[rev[len(suf):]]
        return ans
n = int(input())
arr = []
for _ in range(n):
    arr.append(input())
print(Solution().solve(arr))
