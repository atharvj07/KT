#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int mxN = 3e5;
const ll MOD = 1e9 + 7;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> a;
    vector<int> id(2 * n + 1);
    for (int i = 0; i < n; ++i) {
        int x, y;
        cin >> x >> y;
        a.emplace_back(make_pair(x, y));
        id[x] = id[y] = i;
    }
    sort(a.begin(), a.end());
    vector<int> ans;
    int last = -1;
    bool flag = true;
    for (auto x : a) {
        int i = id[x.first];
        int j = id[x.second];
        if (i > j)
            swap(i, j);
        if (flag) {
            if (i != last) {
                ans.push_back(i);
                last = i;
                flag = false;
            }
        } else {
            if (j != last) {
                ans.push_back(j);
                last = j;
                flag = true;
            }
        }
    }
    cout << ans.size() << endl;
    for (auto x : ans) cout << x + 1 << ' ';
    cout << endl;
}
