n, m = map(int, input().split())
d = [[] for i in range(n+1)]
s = [0] * (n+1)
for i in range(m):
    a, b = map(int, input().split())
    d[a].append(b)
    s[a] ^= 1
    s[b+1] ^= 1

ans = []
for i in range(n+1):
    if i != n and s[i] == 1:
        ans.append(1)
    elif i != n and s[i] == 0:
        ans.append(0)
    else:
        if d[i] == []:
            ans.append(0)
        else:
            ans.append(1)
print("".join(map(str, ans)))
