#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b, c;
    cin >> a >> b >> c;
    if (a % 2 == 0 || b % 2 == 0 || c % 2 == 0)
      cout << "YES" << endl;
    else {
      if (a == b || b == c || c == a)
        cout << "YES" << endl;
      else
        cout << "NO" << endl;
    }
  }
}
