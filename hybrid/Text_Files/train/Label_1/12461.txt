#include <iostream>
#include <string>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    int cnt_end = 0, cnt_res = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (s[i] == ')')
        cnt_end++;
      else
        break;
    }
    for (int i = 0; i < n; i++) {
      if (s[i] != ')')
        cnt_res++;
    }
    if (cnt_end > cnt_res)
      std::cout << "Yes\n";
    else
      std::cout << "No\n";
  }
  return 0;
}
