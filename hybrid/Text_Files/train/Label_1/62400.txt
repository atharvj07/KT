def get_min_replacements(s, q):
    l = [0] * len(s)
    r = [0] * len(s)
    for i in range(len(s)):
        if s[i] == 'a':
            if i - 1 >= 0:
                l[i] = l[i - 1]
            r[i] = i
        elif s[i] == 'b':
            if i - 1 >= 0:
                r[i] = r[i - 1]
            l[i] = i
        else:
            if i - 1 >= 0:
                l[i] = l[i - 1]
                r[i] = r[i - 1]
    for _ in range(q):
        pos, c = input().split()
        pos = int(pos) - 1
        if c == 'a':
            if pos - 1 >= 0:
                l[pos] = l[pos - 1]
            r[pos] = pos
        elif c == 'b':
            if pos - 1 >= 0:
                r[pos] = r[pos - 1]
            l[pos] = pos
        else:
            if pos - 1 >= 0:
                l[pos] = l[pos - 1]
                r[pos] = r[pos - 1]
        if l[pos] <= r[pos]:
            print(0)
        elif r[pos] < len(s) - 1 and s[r[pos] + 1] == 'c':
            print(1)
        elif l[pos] > 0 and s[l[pos] - 1] == 'a':
            print(1)
        else:
            print(2)

n, q = map(int, input().split())
s = input()
get_min_replacements(s, q)
