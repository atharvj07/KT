#include <iostream>
#include <vector>
using namespace std;

inline int64_t absDiff(int64_t a, int64_t b) { return a > b ? a - b : b - a; }
const int64_t MOD = 1e9 + 7;
const int kN = 1e5 + 5;
int64_t dp[kN], a[kN];

int main() {
  int n, q;
  cin >> n >> q;
  for (int i = 1; i <= n; ++i) cin >> a[i];
  for (int i = n; i >= 1; --i) {
    dp[i] = (a[i + 1] - a[i]) * (i < n ? dp[i + 1] : 0) % MOD;
  }
  while (q--) {
    int l, r;
    cin >> l >> r;
    cout << (dp[l] - dp[r + 1] + MOD) % MOD << '\n';
  }
  return 0;
}
