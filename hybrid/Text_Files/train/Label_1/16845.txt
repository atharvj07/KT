#include <bits/stdc++.h>
using namespace std;
const int N=500,M=1e9;
char g[N+10][N+10];
int main() {
  int n,m,k,q,t[N+10][N+10];
  cin >> n >> m >> k >> q;
  memset(t,0x3f,sizeof(t));
  for(int i=1;i<=q;++i) {
    int x,y,tt;
    cin >> x >> y >> tt;
    g[x][y]=1;
    t[x][y]=tt;
  }
  int ans=M;
  for(int i=1;i<=n;++i) {
    for(int j=1;j<=m;++j) {
      if(g[i][j]) {
        for(int r=0;r<k;++r) {
          for(int c=0;c<k;++c) {
            if(i+r<=n && j+c<=m) {
              if(g[i+r][j+c]) ans=min(ans,t[i+r][j+c]);
            }
          }
        }
      }
    }
  }
  if(ans>=M) ans=-1;
  cout << ans << '\n';
}
