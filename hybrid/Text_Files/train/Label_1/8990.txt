#include <cstdio>
#include <algorithm>
#include <vector>
using namespace std;
typedef long long ll;
vector<int> adj[200001];
int main() {
  int n; ll a[200001];
  scanf("%d%lld",&n,&a[1]);
  for(int i = 2; i <= n; i++) {
    scanf("%lld",&a[i]);
    int j;
    while(a[i]--) {
      scanf("%d",&j);
      adj[i].push_back(j);
    }
  }
  vector<int> ans;
  int deg[200001] = {};
  int in[200001] = {};
  for(int i = 1; i <= n; i++) { deg[i] = adj[i].size(); }
  for(int i = 1; i <= n; i++) {
    for(int j = 0; j < adj[i].size(); j++) { in[adj[i][j]]++; }
  }
  while(ans.size() != n) {
    int p = -1;
    for(int i = 1; i <= n; i++) {
      if(deg[i] == 0 && in[i] == 0) {
        p = i;
        break;
      }
    }
    if(p == -1) { printf("Impossible"); return 0; }
    ans.push_back(p); deg[p] = -1;
    for(int i = 0; i < adj[p].size(); i++) {
      in[adj[p][i]]--; deg[adj[p][i]]--;
    }
  }
  for(int i = 0; i < ans.size(); i++) { printf("%d ",ans[i]); }
  return 0;
}
