import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        sc.nextLine();
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = sc.nextLine();
            for (int j = 0; j < m; j++) {
                grid[i][j] = line.charAt(j);
            }
        }
        int[][][] dist = new int[4][n][m];
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < n; j++) {
                Arrays.fill(dist[i][j], Integer.MAX_VALUE);
            }
        }
        int ans = -1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] != '.' && grid[i][j] != '#') {
                    boolean[][] vis = new boolean[n][m];
                    Queue<int[]> q = new LinkedList<>();
                    q.add(new int[]{i, j, grid[i][j] - '0'});
                    vis[i][j] = true;
                    dist[grid[i][j] - '0'][i][j] = 0;
                    while (!q.isEmpty()) {
                        int[] cur = q.poll();
                        if (grid[cur[0]][cur[1]] == '.') {
                            ans = Math.min(ans == -1 ? Integer.MAX_VALUE : ans, dist[cur[2]][cur[0]][cur[1]]);
                        }
                        for (int[] dir : new int[][]{{0, 1}, {0, -1}, {1, 0}, {-1, 0}}) {
                            int nr = cur[0] + dir[0];
                            int nc = cur[1] + dir[1];
                            if (nr >= 0 && nr < n && nc >= 0 && nc < m && !vis[nr][nc] && grid[nr][nc] != '#') {
                                vis[nr][nc] = true;
                                dist[cur[2]][nr][nc] = dist[cur[2]][cur[0]][cur[1]] + 1;
                                q.add(new int[]{nr, nc, cur[2]});
                            }
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
