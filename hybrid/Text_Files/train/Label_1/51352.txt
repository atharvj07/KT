import java.util.Scanner;

public class AntonAndArrays {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] b = new int[n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            // Find the smallest power of 2 that is greater than or equal to b[i]
            int power = 1;
            while (power < b[i]) {
                power *= 2;
            }

            // If the power is greater than c[i], then there is no solution
            if (power > c[i]) {
                System.out.println(-1);
                return;
            }

            // Set a[i] to the power
            a[i] = power;
        }

        // Print the array a
        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
    }
}
