#include <cstdio>
#define mod 1000000007

long long fastpow(long long a, long long p) {
  long long ret = 1;
  for (; p; p >>= 1) {
    if (p & 1) ret = ret * a % mod;
    a = a * a % mod;
  }
  return ret;
}
int main() {
  long long x, ans = 1;
  int n;
  scanf("%d %lld", &n, &x);
  for (long long a; n--;) {
    scanf("%lld", &a);
    ans = ans * fastpow(x, a) % mod;
  }
  printf("%lld", ans);
  return 0;
}
