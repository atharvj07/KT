#include <bits/stdc++.h>

using namespace std;

int n;
int a[1001][4];

bool cmp(int x, int y) {
  int sumX = a[x][0] + a[x][1] + a[x][2] + a[x][3];
  int sumY = a[y][0] + a[y][1] + a[y][2] + a[y][3];
  if (sumX != sumY)
    return sumX > sumY;
  return x < y;
}

main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < 4; j++) {
      cin >> a[i][j];
    }
  }
  vector<int> v(n + 1);
  iota(v.begin(), v.end(), 0);
  sort(v.begin() + 1, v.end(), cmp);
  for (int i = 1; i <= n; i++) {
    if (v[i] == 1) {
      cout << i << endl;
      break;
    }
  }
}
