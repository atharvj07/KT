import java.io.*;
import java.util.*;

public class Main {
    static final int INF = 1 << 28;
    static int n, m, max = 0, ans = -1;
    static boolean[] marked;
    static int[] dist, color;
    static List<Edge> edges;

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        color = new int[n + 1];
        dist = new int[n + 1];
        marked = new boolean[n + 1];
        edges = new ArrayList<>();
        for (int i = 1; i <= m; i++) {
            int a = sc.nextInt(), b = sc.nextInt(), t = sc.nextInt();
            edges.add(new Edge(a, b, t));
        }

        for (int i = 1; i <= n; i++) {
            if (!marked[i]) {
                dfs1(i, 0);
                dfs2(i, 0);
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            sb.append(color[i] ^ 1);
        }

        System.out.println(ans == -1 ? -1 : max);
        System.out.println(sb.toString());
    }

    public static void dfs1(int s, int c) {
        marked[s] = true;
        dist[s] = c;
        if (c > max) {
            max = c;
            ans = s;
        }

        for (Edge e : edges) {
            if (e.from == s) {
                if (e.type == c & !marked[e.to]) {
                    dfs1(e.to, c);
                } else if (e.type != c & !marked[e.to]) {
                    dfs1(e.to, e.type);
                }
            }
        }
    }

    public static void dfs2(int s, int c) {
        color[s] = c;
        for (Edge e : edges) {
            if (e.from == s) {
                if (e.type == c & !marked[e.to]) {
                    dfs2(e.to, c);
                } else if (e.type != c & !marked[e.to]) {
                    dfs2(e.to, e.type);
                }
            }
        }
    }

    public static class Edge {
        public int from;
        public int to;
        public int type;

        public Edge(int from, int to, int type) {
            this.from = from;
            this.to = to;
            this.type = type;
        }
    }
}
