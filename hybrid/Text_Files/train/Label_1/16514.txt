#include <iostream>
#include <vector>

using namespace std;

int main() {
  long long n, k, ans = 0;
  cin >> n >> k;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  long long sum = 0, prod = 1;
  for (int l = 0, r = 0; r < n; r++) {
    sum += a[r];
    prod *= a[r];
    while (l <= r && prod > sum * k) {
      sum -= a[l];
      prod /= a[l];
      l++;
    }
    if (prod == sum * k) ans++;
  }
  cout << ans << endl;
  return 0;
}
