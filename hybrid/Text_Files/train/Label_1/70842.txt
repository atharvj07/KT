import sys
input = sys.stdin.readline

n = int(input())
arr = list(map(int, input().split()))
cnt = {}
ans = []
for x in arr:
    if x not in cnt: cnt[x] = 1
    else: cnt[x] += 1
for x in cnt:
    while cnt[x] > 1:
        cnt[x] -= 2
        ans.append([x, x])
if len(ans) == 0:
    print(-1)
else:
    print(len(ans))
    for x, y in ans:
        print(x, y)
