from math import inf

s = input()
l = list(s)
n = len(l)

dp = [[0] * 10 for _ in range(n + 1)]

for i in range(n - 1, -1, -1):
    for prv in range(10):
        for dig in range(10):
            t = int(''.join(l[i:]))
            cost = 0
            x = (t + dig * (10 ** (n - i - 1))) % 25
            if prv == 0 and dig == 0 and i != 0:
                cost = 0
            elif prv == 0 and dig != 0:
                cost = 1
            elif x in [0, 10, 15]:
                cost = 1
            elif x in [5, 20]:
                cost = 2
            else:
                cost = inf
            dp[i][prv] = min(dp[i][prv], cost + dp[i + 1][dig])
            t = int(''.join(l[i:]))
            x = (t + (10 * dig + prv) * (10 ** (n - i - 1))) % 25
            if prv == 0 and dig == 0 and i != 0:
                cost = 0
            elif prv == 0 and dig != 0:
                cost = 1
            elif x in [0, 10, 15]:
                cost = 1
            elif x in [5, 20]:
                cost = 2
            else:
                cost = inf
            dp[i][prv] = min(dp[i][prv], cost + dp[i + 1][dig])
        for dig in range(10):
            t = int(''.join(l[i:]))
            x = (t + (prv * (10 ** (n - i - 1))) + dig) % 25
            if prv == 0 and dig == 0 and i != 0:
                cost = 0
            elif prv == 0 and dig != 0:
                cost = 1
            elif x in [0, 10, 15]:
                cost = 1
            elif x in [5, 20]:
                cost = 2
            else:
                cost = inf
            dp[i][prv] = min(dp[i][prv], cost + dp[i + 1][dig])

ans = dp[0][0]
if ans >= inf:
    print(-1)
else:
    print(ans)
