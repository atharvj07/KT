import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[] a = Arrays.stream(in.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] b = Arrays.stream(in.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        long[] preA = new long[n];
        long[] preB = new long[n];
        for (int i = 0; i < n; i++) {
            preA[i] = i == 0 ? a[i] : preA[i - 1] + a[i];
            preB[i] = i == 0 ? b[i] : preB[i - 1] + b[i];
        }
        long sumA = preA[n - 1];
        long sumB = preB[n - 1];
        if (sumA == sumB) {
            System.out.println("0");
        } else if ((sumA - sumB) % 3 != 0) {
            System.out.println("-1");
        } else {
            long c = (sumA - sumB) / 3;
            long ans = 0;
            long ca = 0;
            long cb = 0;
            for (int i = 0; i < n; i++) {
                ca += a[i];
                cb += b[i];
                long ra = ca - c;
                long rb = cb - c;
                long rc = sumA - ra - ca;
                if (ra >= 0 && rb >= 0 && rc >= 0 && ra + rb + rc == sumA && check(preA, preB, i, ra, rb, rc)) {
                    if (rc == 0) ans = Math.min(ans, i);
                    else ans = i + 1;
                }
            }
            System.out.println(ans == 0 ? "-1" : ans);
        }
    }

    private static boolean check(long[] pre, long[] pre2, int index, long ra, long rb, long rc) {
        return (index == 0 ? 0 : pre[index - 1]) + ra == pre2[index] &&
                pre[index] + rb == pre2[index + 1] &&
                (index == pre.length - 1 ? 0 : pre[pre.length - 1]) + rc == pre2[0];
    }
}
