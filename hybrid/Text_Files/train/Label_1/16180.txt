#include <bits/stdc++.h>
using namespace std;

pair<long long, char> a[200005], b[200005];

int n, m;

long long check() {
  long long L = 0, R = 0;
  long long ans = 0;
  while (L < n && R < m) {
    if (a[L].second == b[R].second) {
      if (a[L].first < b[R].first)
        L++;
      else if (a[L].first > b[R].first)
        R++;
      else {
        R++;
        ans++;
      }
    } else
      R++;
  }
  return ans;
}

int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++) cin >> a[i].first >> a[i].second;
  for (int j = 0; j < m; j++) cin >> b[j].first >> b[j].second;
  cout << check() << endl;
  return 0;
}
