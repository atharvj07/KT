#include <bits/stdc++.h>
using namespace std;
#define int long long
int pre[1000001];
main() {
  int n, x;
  cin >> n >> x;
  int a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    pre[a[i]]++;
  }
  for (int i = 2; i <= x; i++) pre[i] += pre[i - 1];
  int ans = 0;
  for (int l = 1; l <= x; l++) {
    for (int r = l; r <= x; r++) {
      ans += pre[r] * (pre[r] - (pre[r - 1] - pre[l - 1]));
    }
  }
  cout << ans;
}
