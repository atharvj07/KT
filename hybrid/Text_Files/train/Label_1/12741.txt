#include <bits/stdc++.h>
using namespace std; typedef pair<int,int> pi; 

int main(){
  int n; cin >> n;
  vector<pi> t(n);
  for (auto &[l,r] : t) cin >> l >> r;
  sort(t.begin(),t.end());
  priority_queue<int,vector<int>,greater<int>> pq;
  for (auto& [l,r] : t){
    while (!pq.empty() && pq.top() <= l) pq.pop();
    pq.push(r);
    if (pq.size() > 2) { cout << "NO"; return 0; }
  }
  cout << "YES"; return 0;
}
