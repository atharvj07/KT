n, m, k = map(int, input().split())
a = [['0' * (k + 2)] for _ in range(n + 2)]
for i in range(1, n + 1):
    for j in range(1, m + 1):
        line = '0' + input().replace('0', ' ').replace('1', '1 ') + '0'
        a[i][j] = '0' + line + '0'
ans, vis = 0, [[[0] * (k + 2) for _ in range(m + 2)] for _ in range(n + 2)]
for x in range(1, n + 1):
    for y in range(1, m + 1):
        for z in range(1, k + 1):
            if a[x][y][z] == '1':
                # print(x, y, z)
                if not vis[x][y][z]:
                    vis[x][y][z] = 1
                    # print(x + 1, y, z)
                    if a[x + 1][y][z] == '1' and not vis[x + 1][y][z]:
                        ans += 1
                        # print(x, y + 1, z)
                    if a[x][y + 1][z] == '1' and not vis[x][y + 1][z]:
                        ans += 1
                        # print(x, y, z + 1)
                    if a[x][y][z + 1] == '1' and not vis[x][y][z + 1]:
                        ans += 1
print(ans)
