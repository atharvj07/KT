import java.util.*;
import java.math.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            BigInteger p = sc.nextBigInteger(), q = sc.nextBigInteger();
            System.out.println(p.divide(p.gcd(q)));
        }
    }
}
