#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, d;
  cin >> n >> d;
  vector<int> s(n), p(n);
  for (int i = 0; i < n; ++i) {
    cin >> s[i];
  }
  for (int i = 0; i < n; ++i) {
    cin >> p[i];
  }
  int score_d = s[d - 1] + p[0];
  int cnt = 1;
  for (int i = 1; i < n; ++i) {
    if (score_d > s[i - 1] + p[i]) {
      ++cnt;
    } else {
      break;
    }
  }
  cout << cnt << endl;
  return 0;
}
