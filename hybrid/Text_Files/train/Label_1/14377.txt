#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;
const ll INF = 1e18;

ll solve(ll x, ll y, ll z) {
    ll ans1, ans2, balance = 0;
    if(x > y) {
        balance = (x - y + z - 1) / z;
        ans1 = (x - balance) / z;
        ans2 = (y + balance) / z;
    } else {
        balance = (y - x + z - 1) / z;
        ans2 = (y - balance) / z;
        ans1 = (x + balance) / z;
    }
    return ans1 + ans2;
}

int main() {
    ll x, y, z;
    cin >> x >> y >> z;
    cout << solve(x, y, z) << " " << abs(x - y) / z << endl;
    return 0;
}
