#include<iostream>
#include<vector>
#include<string>
using namespace std;
int dx[4]={1,-1,0,0};
int dy[4]={0,0,1,-1};
vector<vector<int>> v,ans;
vector<vector<bool>> vis;
void dfs(int x,int y,int now){
    vis[x][y]=true;
    ans[x][y]=now%10;
    for (int i=0;i<4;i++){
        int nx=x+dx[i];
        int ny=y+dy[i];
        if ((nx>=0) && (nx<v.size()) && (ny>=0) && (ny<v[0].size()))
            if (v[nx][ny]==0 and vis[nx][ny]==false)
                dfs(nx,ny,now+1);
    }
}
int main() {
    int n,m;
    cin >> n >> m;
    v.resize(n,vector<int>(m));
    ans.resize(n,vector<int>(m));
    vis.resize(n,vector<bool>(m,false));
    for (int x=0;x<n;x++){
        string s;
        cin >> s;
        for (int y=0;y<m;y++)
            if (s[y]=='.') v[x][y]=0;
            else v[x][y]=1,ans[x][y]=-1;
    }
    int cnt=0;
    for (int x=0;x<n;x++){
        for (int y=0;y<m;y++){
            if (v[x][y]==1 and vis[x][y]==false){
                dfs(x,y,++cnt);
            }
        }
    }
    for (int x=0;x<n;x++){
        for (int y=0;y<m;y++)
            if (v[x][y]==1) cout << ans[x][y];
            else cout << ".";
        cout << endl;
    }
    return 0;
}
