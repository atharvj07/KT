import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        long n = sc.nextLong();
        long m = sc.nextLong();
        long a = sc.nextLong();
        long b = sc.nextLong();

        long minBurles = 0;

        if (n % m != 0) {
            minBurles = Math.min(n * a, (m - n % m) * b);
        }

        System.out.println(minBurles);
    }
}
