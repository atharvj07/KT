import sys

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
a.sort()
a = [a[0]] + a
l = 0
r = 0
ans = 0
for i in range(1, n+1):
    l = i
    if i == n:
        r += 1
        if r - l + 1 > ans:
            ans = r - l + 1
            ansx = l - 1
            ansy = r - l
    else:
        while r+1 <= n and a[r+1] == a[i]:
            r += 1
        r += 1
        if r - l + 1 > ans:
            ans = r - l + 1
            ansx = l - 1
            ansy = r - l
r = ansx
l = -1
for i in range(ansy):
    l += 1
    while r+1 <= n and a[r+1] == a[r]:
        r += 1
    print(*a[l: r+1])
    r += 1
