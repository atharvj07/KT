from operator import and_, or_, xor
from functools import reduce

n, m = map(int, input().split())
vars = {}
for i in range(n):
    k, *v = input().split()
    ops = vars.keys() & set(v)
    if v[0].isdigit():
        vars[k] = int(v[0], 2)
    elif len(ops) == len(v) - 1:
        x, op, y = v
        vars[k] = reduce({
            "AND": and_,
            "OR": or_,
            "XOR": xor
        }[op], map(lambda x: vars[x] if x in ops else int(x, 2), v))
    else:
        vars[k] = '?'
        vals = [vars[x] for x in ops] + [int(x, 2) for x in v if x.isdigit()]
        if 'XOR' in v:
            vars[k] = reduce(xor, vals)
            vars[k] -= 2 * reduce(and_, vals)
            vars[k] += reduce(or_, vals)
        elif 'OR' in v:
            vars[k] = 2 * reduce(or_, vals) - reduce(and_, vals)
        else:
            vars[k] = reduce(and_, vals)
ans1 = ans2 = 0
for x in vars:
    if vars[x] == '?':
        ans1 = min(ans1, 2 << (m - 1))
        ans2 = max(ans2, (2 << (m - 1)) - 1)
    else:
        ans1 = max(ans1, vars[x])
        ans2 = min(ans2, vars[x])
print(bin(ans1)[2:].zfill(m))
print(bin(ans2)[2:].zfill(m))
