import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;
import java.util.PriorityQueue;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int k = Integer.parseInt(input[1]);
        int q = Integer.parseInt(input[2]);
        int[] relations = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        PriorityQueue<Integer> onlineFriends = new PriorityQueue<>(Comparator.reverseOrder());
        boolean[] online = new boolean[n];
        for (int i = 0; i < q; i++) {
            input = br.readLine().split(" ");
            int type = Integer.parseInt(input[0]);
            int id = Integer.parseInt(input[1]) - 1;
            if (type == 1) {
                onlineFriends.offer(relations[id]);
                online[id] = true;
            } else {
                boolean displayed = false;
                for (int j = 0; j < k && onlineFriends.size() > 0; j++) {
                    int relation = onlineFriends.poll();
                    if (relations[id] == relation) {
                        displayed = true;
                        onlineFriends.offer(relation);
                        break;
                    } else online[relations[id] - 1] = false;
                }
                System.out.println(displayed ? "YES" : "NO");
            }
        }
    }
}
