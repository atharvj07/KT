import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int l = -1, r = -1;
        for (int i = 0; i < n - 1; i++) {
            if (a[i] > a[i + 1]) {
                if (l == -1) l = i;
                r = i + 1;
            }
        }
        if (l == -1) System.out.println("YES");
        else if (l == 0 || r == n - 1) System.out.println("YES");
        else if (a[l - 1] <= a[r] && a[l] <= a[r + 1]) System.out.println("YES");
        else System.out.println("NO");
    }
}
