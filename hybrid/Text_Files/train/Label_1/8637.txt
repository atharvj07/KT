#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, x;
  cin >> n >> x;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  long long ans = 0, pref = 0;
  for (int i = 0; i < n; i++) {
    pref += a[i];
    if (pref < 0) pref = 0;
    ans = max(ans, pref);
  }

  long long suf = 0;
  for (int i = n - 1; i >= 0; i--) {
    suf += a[i] * x;
    if (suf < 0) suf = 0;
    ans = max(ans, suf);
  }

  cout << ans << endl;
  return 0;
}
