from sys import stdin
input=stdin.readline
for _ in range(int(input())):
    n,k=map(int,input().split())
    x=list(map(int,input().split()))
    y=list(map(int,input().split()))
    x.sort()
    y.sort()
    #print(x,y)
    z=[1]*n
    ans=0
    for i in range(n):
        low=i
        high=n-1
        mid=-1
        while low<=high:
            m=(low+high)//2
            if y[m]<=y[i]+k:
                mid=m
                low=m+1
            else:
                high=m-1
        #print(i,mid)
        if mid!=-1:
            ans+=mid-i+1
            for j in range(i,mid+1):
                z[j]=0
    #print(z)
    low=0
    high=n-1
    for i in range(n-1,-1,-1):
        if z[i]==0:
            continue
        mid=-1
        while low<=high:
            m=(low+high)//2
            if x[m]>=x[i]-k:
                mid=m
                low=m+1
            else:
                high=m-1
        #print(i,mid)
        if mid!=-1:
            ans+=i-mid+1
            for j in range(mid,i+1):
                z[j]=0
    print(ans)
