#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    unordered_map<int, int> m;
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        long long x = (long long)a[i] * (long long)a[i];
        for (int j = 2; j <= k; j++) {
            if (x % j == 0) {
                ans += m[x / j];
            }
        }
        m[a[i]]++;
    }
    cout << ans << endl;
    return 0;
}
