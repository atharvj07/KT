#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  string s;
  cin >> s;
  vector<pair<char, char>> changes(m);
  for (int i = 0; i < m; i++) {
    char x, y;
    cin >> x >> y;
    changes[i] = {x, y};
  }
  for (int j = 0; j < m; j++) {
    for (int i = 0; i < n; i++) {
      if (s[i] == changes[j].first) {
        s[i] = changes[j].second;
      } else if (s[i] == changes[j].second) {
        s[i] = changes[j].first;
      }
    }
  }
  cout << s << endl;
  return 0;
}
