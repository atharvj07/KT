#include<bits/stdc++.h>
using namespace std;

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, q;
    cin >> n >> q;
    vector<int> a(n);
    for(int i = 0; i < n; i++) cin >> a[i];
    while(q--){
        char s;
        int x;
        cin >> s >> x;
        for(int i = 0; i < n; i++) a[i] = (s == '<' && a[i] < x || s == '>' && a[i] > x) ? -a[i] : a[i];
    }
    for(int i = 0; i < n; i++) cout << a[i] << " ";
    return 0;
}
