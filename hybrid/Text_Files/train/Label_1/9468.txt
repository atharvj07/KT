#include<bits/stdc++.h>
using namespace std;

int32_t main() {
  int n, l = 0, r = 0;
  cin >> n;
  vector<pair<int, int>> v(n), ans;
  for (int i = 0; i < n; i++) {
    int t;
    cin >> t;
    v[i] = {t, i};
  }
  sort(v.begin(), v.end());

  int i = 0, j = n - 1;
  while (i < j) {
    if (v[i].first < v[i + 1].first) {
      l++;
      ans.push_back({v[i].second, v[i].first});
      i++;
    } else if (v[j].first > v[j - 1].first) {
      r++;
      ans.push_back({v[j].second, v[j].first});
      j--;
    } else {
      cout << "NO" << endl;
      return 0;
    }
  }

  if (l == 0 || r == 0)
    cout << "NO" << endl;
  else {
    cout << "YES" << endl;
    cout << l << endl;
    for (int i = 0; i < l; i++) cout << ans[i].second << " ";
    cout << endl;
    cout << r << endl;
    for (int i = l; i < l + r; i++) cout << ans[i].second << " ";
    cout << endl;
  }
  return 0;
}
