#include<bits/stdc++.h>
#define EPS 1e-6
using namespace std;
int n, m;
double p[1005][1005];
int a[1005];
int main()
{
    cin >> n >> m;
    for(int i = 1; i <= n; i++)
    {
        cin >> a[i];
    }
    for(int i = 1; i <= n; i++)
    {
        for(int j = i + 1; j <= n; j++)
        {
            p[i][j] = p[j][i] = 1.0 / 2.0;
        }
    }
    for(int i = 1; i <= m; i++)
    {
        int x, y;
        cin >> x >> y;
        for(int j = 1; j <= n; j++)
        {
            if(a[x] < a[j])
            {
                p[a[x]][a[j]] *= 1 - p[x][y];
            }
            if(a[y] < a[j])
            {
                p[a[y]][a[j]] *= 1 - p[x][y];
            }
            swap(a[x], a[y]);
        }
    }
    double ans = 0;
    for(int i = 1; i <= n; i++)
    {
        for(int j = i + 1; j <= n; j++)
        {
            ans += p[i][j];
        }
    }
    cout << fixed << setprecision(9) << ans << endl;
}
