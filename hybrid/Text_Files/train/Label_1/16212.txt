#include <cstdio>
#include <cstring>
#include <unordered_map>

struct Record {
  char name[11];
  int count;
  char phones[200][11];
};

int main() {
  int n;
  scanf("%d", &n);

  std::unordered_map<std::string, Record> friends;
  std::unordered_map<std::string, int> suffixes;
  for (int i = 0; i < n; ++i) {
    Record r;
    scanf("%s %d ", r.name, &r.count);
    for (int j = 0; j < r.count; ++j) {
      char phone[11];
      scanf("%s ", phone);
      if (suffixes.count(phone) == 0) {
        strcpy(r.phones[j], phone);
      }
    }

    friends[r.name] = r;
    for (int j = 0; j < r.count; ++j) {
      int length = strlen(r.phones[j]);
      suffixes[std::string(r.phones[j] + length - 2)]++;
    }
  }

  printf("%d\n", (int)friends.size());
  for (auto &f : friends) {
    printf("%s %d", f.second.name, f.second.count);
    for (int j = 0; j < f.second.count; ++j) {
      printf(" %s", f.second.phones[j]);
    }
    printf("\n");
  }
  return 0;
}
