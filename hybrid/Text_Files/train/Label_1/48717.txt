import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            String a = in.next();
            char[] arr = a.toCharArray();
            boolean swap = false;
            for (int i = 0; i < arr.length - 1; i++) {
                if ((arr[i] - '0') % 2 != (arr[i + 1] - '0') % 2) {
                    swap = true;
                    char temp = arr[i];
                    arr[i] = arr[i + 1];
                    arr[i + 1] = temp;
                    break;
                }
            }
            if (swap) {
                while (arr[0] == '0' && arr.length > 1) {
                    char[] newArr = new char[arr.length - 1];
                    for (int i = 1; i < arr.length; i++) {
                        newArr[i - 1] = arr[i];
                    }
                    arr = newArr;
                }
                for (char c : arr) {
                    System.out.print(c);
                }
            } else {
                System.out.print(a);
            }
            System.out.println();
        }
    }
}
