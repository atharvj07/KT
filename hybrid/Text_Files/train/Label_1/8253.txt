#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int sk[n];
  map<int, int> f;
  for (int i = 0; i < n; ++i) {
    cin >> sk[i];
    f[sk[i]]++;
  }
  sort(sk, sk + n);
  int ans = 0;
  for (int i = 0; i < n;) {
    int cnt = 1;
    int j = i + 1;
    while (j < n && sk[j] == sk[i]) {
      j++;
      cnt++;
    }
    cnt /= 2;
    if (cnt == n / 2) {
      break;
    }
    if (cnt * 2 < f[sk[i]]) {
      int t = f[sk[i]] - cnt * 2;
      ans += t;
      f[sk[i]] -= t;
    } else {
      if (i != j) {
        int t = cnt * 2;
        ans += t;
        f[sk[i]] -= t;
        f[sk[j]] -= t;
      }
    }
    i = j;
  }
  cout << ans << endl;
  return 0;
}
