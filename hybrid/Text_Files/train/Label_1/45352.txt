import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int[] A = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextInt();
        }
        int MEX = 1;
        while (true) {
            boolean found = false;
            for (int i = 0; i < N; i++) {
                if (A[i] % MEX == 0) {
                    found = true;
                    break;
                }
            }
            if (!found) {
                break;
            }
            MEX++;
        }
        long ans = (M - 1) / MEX + 1;
        System.out.println(ans);
    }
}
