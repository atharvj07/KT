#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> p(n + 1);
    for (int i = 1; i <= n; ++i) cin >> p[i];
    vector<int> ans(n + 1);
    for (int i = 1; i <= n; ++i) {
        vector<bool> vis(n + 1);
        int x = i;
        while (!vis[x]) {
            vis[x] = true;
            x = p[x];
        }
        ans[i] = x;
    }
    for (int i = 1; i <= n; ++i) cout << ans[i] << " ";
}
