import java.util.Arrays;
import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    int M = sc.nextInt();
    long[] A = new long[N];
    for (int i = 0; i < N; i++) {
      A[i] = sc.nextInt();
    }
    long[] cost = new long[N];
    for (int i = 0; i < N; i++) {
      cost[i] = A[i];
    }
    Arrays.sort(A);
    int M2 = Math.min(M, N / 2);
    for (int i = 0; i < M2; i++) {
      cost[i / 2] /= 2;
    }
    long ans = 0;
    for (int i = 0; i < N; i++) {
      ans += cost[i];
    }
    System.out.println(ans);
  }
}
