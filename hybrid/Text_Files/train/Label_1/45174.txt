import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[][] dist = new long[n + 1][n + 1];
        for (long[] dd : dist)
            Arrays.fill(dd, Long.MAX_VALUE);
        for (int i = 0; i <= n; i++)
            dist[i][i] = 0;
        for (int i = 0; i < m; i++) {
            int p = sc.nextInt(), q = sc.nextInt(), c = sc.nextInt();
            dist[p][q] = Math.min(dist[p][q], c);
            dist[q][p] = Math.min(dist[q][p], c);
        }
        for (int k = 1; k <= n; k++)
            for (int i = 1; i <= n; i++)
                for (int j = 1; j <= n; j++)
                    dist[i][j] = Math.min(dist[i][j], dist[i][k] + dist[k][j]);
        long ans = dist[1][n];
        System.out.println(ans == Long.MAX_VALUE ? -1 : ans);
    }
}
