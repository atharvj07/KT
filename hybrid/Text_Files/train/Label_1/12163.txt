#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 998244353;
int qpow(int a, int b) {
  if (!b) return 1;
  int t = qpow(a, b / 2);
  t = t * t % mod;
  if (b & 1) t = t * a % mod;
  return t;
}
int fac[200005], invfac[200005];
void init() {
  fac[0] = 1;
  for (int i = 1; i <= 200000; i++) fac[i] = i * fac[i - 1] % mod;
  invfac[200000] = qpow(fac[200000], mod - 2);
  for (int i = 199999; i >= 0; i--) invfac[i] = invfac[i + 1] * (i + 1) % mod;
}
int c(int a, int b) { return fac[a] * invfac[b] % mod * invfac[a - b] % mod; }
signed main() {
  init();
  int n, k;
  cin >> n >> k;
  vector<int> h(n);
  for (int i = 0; i < n; i++) cin >> h[i];
  vector<int> cnt(k + 1);
  for (int x : h) cnt[x]++;
  vector<int> sum(k + 1);
  for (int i = 1; i <= k; i++) sum[i] = sum[i - 1] + cnt[i];
  int ans = 0;
  vector<int> can(k + 1);
  for (int now = 1; now <= k; now++) {
    int tmp = (sum[k] - sum[now - 1] - cnt[now] + n) % n;
    ans = (ans + (tmp + 1) * c(n, tmp) % mod * fac[sum[k]] % mod * invfac[sum[k] - cnt[now] - 1] % mod * invfac[cnt[now]]) % mod;
    can[now] = ans;
  }
  for (int i = 0; i < n; i++) {
    int s = sum[h[i] - 1], t = sum[k] - s - cnt[h[i]];
    if (s && t < n) {
      int tmp = (t + 1) * c(n, t) % mod * fac[sum[k]] % mod * invfac[s] % mod * invfac[sum[k] - s - cnt[h[i]] - 1] % mod;
      ans = (ans + (can[h[i]] - tmp + mod) % mod) % mod;
    }
  }
  cout << ans << endl;
  return 0;
}
