#include <iostream>
#include <cmath>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        int ans = 0;
        while (n > 0) {
            if (n % k == 0) {
                n /= k;
                ans++;
            } else {
                n--;
                ans++;
            }
        }
        cout << ans << endl;
    }
}
