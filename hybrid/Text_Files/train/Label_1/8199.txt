#include <bits/stdc++.h>
using namespace std;

const int N = 105;

int n, m;
char gri[N][N];
vector<pair<int, int>> pos;
int x[N], y[N], sz[N], vis[N], ans;

inline bool valid(int ni, int nj) {
    if(ni < 1 || ni > n || nj < 1 || nj > m) return false;
    if(gri[ni][nj] == '.') return false;
    for(int i = 0; i < ans; i++) {
        int nx = x[i], ny = y[i], s = sz[i];
        if(abs(ni - nx) <= s && abs(nj - ny) <= s) return false;
    }
    return true;
}

void solve(int p) {
    if(p == pos.size()) {
        cout << ans << endl;
        for(int i = 0; i < ans; i++)
            cout << x[i] << ' ' << y[i] << ' ' << sz[i] << endl;
        exit(0);
    }
    int ni = pos[p].first, nj = pos[p].second;
    for(int len = 1; valid(ni, nj); len++) {
        x[ans] = ni, y[ans] = nj, sz[ans] = len;
        ans++, solve(p + 1), ans--;
    }
}

int main() {
    cin >> n >> m;
    string s;
    for(int i = 1; i <= n; i++) {
        cin >> s;
        for(int j = 1; j <= m; j++) {
            gri[i][j] = s[j - 1];
            if(s[j - 1] == '*') pos.push_back({i, j});
        }
    }
    solve(0);
    cout << -1 << endl;
    return 0;
}
