#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, t;
    cin >> n >> t;
    vector<long long> t_i(n);
    for (int i = 0; i < n; i++) {
        cin >> t_i[i];
    }
    int cnt = 0;
    long long cur = 0;
    for (int i = 0; i < n; i++) {
        if (cur + 1 >= t_i[i]) {
            cnt++;
            cur++;
        } else if (cur + 1 + t_i[i] - 1 <= t) {
            cnt++;
            cur = t_i[i];
        } else {
            break;
        }
    }
    cout << cnt << endl;
    return 0;
}
