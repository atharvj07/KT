#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n; std::cin >> n;
  std::vector<std::pair<int, int>> scores(n);
  for (auto& score : scores) std::cin >> score.first >> score.second;
  int diff = 0, max_diff = 0, ans = 0;
  for (const auto& score : scores) {
    diff += score.first - score.second;
    if (diff > max_diff) {
      max_diff = diff;
      ans = 1;
    } else if (diff < max_diff) ans = 0;
    else ++ans;
  }
  std::cout << ans << std::endl;
  return 0;
}
