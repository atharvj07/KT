MOD = 10**9 + 7
n, m = map(int, input().split())
s = input()
dp_left = [0] * (m + 1)
dp_right = [0] * (m + 1)
p, q = 0, 0
for i in range(m):
    dp_left[i + 1] = dp_left[i]
    if s[i] == '(':
        p += 1
    else:
        if p:
            p -= 1
            dp_left[i + 1] += 1
for i in range(m - 1, -1, -1):
    dp_right[i] = dp_right[i + 1]
    if s[i] == ')':
        q += 1
    else:
        if q:
            q -= 1
            dp_right[i] += 1
ans = (dp_left[m] + q * (dp_left[m] + dp_right[m]) % MOD) % MOD
print(ans)
