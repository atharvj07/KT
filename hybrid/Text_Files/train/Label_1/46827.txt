import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String s = scanner.next();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (i + 1 < n && s.charAt(i) != s.charAt(i + 1)) {
                ans += 2;
                i++;
            } else {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
