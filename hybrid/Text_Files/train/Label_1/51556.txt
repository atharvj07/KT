import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt();
        int k = sc.nextInt();
        if (d > n - 1 || d > 2 * (k - 1)) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        int[] deg = new int[n + 1];
        for (int i = 2; i <= n; i++) {
            deg[i] = 1;
        }
        int a = 1;
        int b = 2;
        deg[1] = 2;
        deg[2] = 1;
        int len = 1;
        while (len < d) {
            if (a == 1) {
                for (int i = 3; i <= n; i++) {
                    if (deg[i] < k) {
                        deg[1]++;
                        deg[i]++;
                        System.out.println(1 + " " + i);
                        break;
                    }
                }
                b = a;
                a++;
            } else {
                for (int i = a + 1; i <= n; i++) {
                    if (deg[i] < k) {
                        deg[b]++;
                        deg[i]++;
                        System.out.println(b + " " + i);
                        break;
                    }
                }
                a = b;
                b++;
            }
            len++;
        }
    }
}
