n, m, k = map(int, input().split())
w = list(map(int, input().split()))
games = [tuple(map(int, input().split())) for _ in range(m)]

# Initialize the stadium for each game as 0 (not played)
stadiums_for_games = [0] * m

# Initialize games played in each stadium by each team
s = [[0]*k for _ in range(n)]

for game_index, (a, b) in enumerate(games):
    max_earning = -1
    best_stadium = 0

    # Find the best stadium for the current game
    for stadium_index in range(k):
        # Calculate potential earnings
        earning = w[a-1] + w[b-1]

        # Check if playing the game in this stadium would violate the rule
        sa, sb = s[a-1][stadium_index], s[b-1][stadium_index]
        next_sa, next_sb = sa + 1, sb + 1

        # Calculate the minimum and maximum games played in a stadium by team a and b
        min_a, max_a = min(s[a-1]), max(next_sa, max(s[a-1]))
        min_b, max_b = min(s[b-1]), max(next_sb, max(s[b-1]))

        # Check if the rule is violated for either team
        if max_a - min_a <= 2 and max_b - min_b <= 2:
            # If the rule is not violated and the earning is higher, update the best stadium
            if earning > max_earning:
                max_earning = earning
                best_stadium = stadium_index + 1

    # Assign the best stadium for the game
    stadiums_for_games[game_index] = best_stadium
    if best_stadium > 0:
        # Update games played in the chosen stadium by each team
        s[a-1][best_stadium-1] += 1
        s[b-1][best_stadium-1] += 1

# Output the stadium for each game
for stadium in stadiums_for_games:
    print(stadium)
