#include <bits/stdc++.h>
using namespace std;
pair<long long,long long>a[10000];
long long d[1001];
bool c[1001];
int main(){
    long long n,m,i,j,l,r,ans=0,sum=0;
    cin>>n>>m;
    for(i=1;i<=n;i++){
        cin>>a[i].first;
    }
    for(i=1;i<=m;i++){
        cin>>a[i].second>>l;
        if(l){
            d[a[i].second]+=a[i].first;
            c[a[i].second]=1;
        }
    }
    for(i=1;i<=n;i++){
        if(c[i])continue;
        for(j=1;j<=1000;j++){
            if(d[j]){
                ans+=max(d[j],a[i].first)/d[j]*(j+1);
                d[j]-=max(d[j],a[i].first);
            }
            else{
                ans+=j;
                break;
            }
        }
    }
    cout<<ans<<endl;
}
