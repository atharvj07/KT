#include <bits/stdc++.h>
#define ll long long
using namespace std;

ll dif(ll a, ll b) {
    if (a > b)
        return a - b;
    else
        return b - a;
}
int main() {
    ll u, v;
    cin >> u >> v;
    if ((u + v) % 2 == 0 && (u || v)) {
        ll ans = v / 2;
        if (u)
            cout << "1\n" << ans + u << endl;
        else
            cout << "0\n" << endl;
    } else
        cout << "-1\n";
}
