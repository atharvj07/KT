from collections import defaultdict
from queue import Queue

n = int(input())
adj = defaultdict(list)
for x, y in [map(int, input().split()) for _ in range(n - 1)]:
    adj[x].append(y)
    adj[y].append(x)

q = Queue()
q.put((1, -1))
seen = [False] * (n + 1)
seen[1] = True
ans = []
while not q.empty():
    x, p = q.get()
    for y in adj[x]:
        if not seen[y]:
            seen[y] = True
            q.put((y, x))
            ans.append((x, y))

if len(ans) == n - 1:
    print('YES')
    for x, y in ans:
        print(x, y)
else:
    print('NO')
