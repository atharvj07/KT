def make_equal(n, r):
    R = max(r)
    t = 0
    ans = []
    for i in range(n):
        if r[i] < R:
            t += 1
            ans.append('1' * (n - i) + '0' * i)
    return R, t, ans

n = int(input())
r = list(map(int, input().split()))
R, t, ans = make_equal(n, r)
print(R)
print(t)
for s in ans:
    print(s)
