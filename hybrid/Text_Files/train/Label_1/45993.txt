import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String x = sc.next();
        int d = sc.nextInt();

        StringBuilder ans = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (d > 0 && x.charAt(i) == '0') {
                ans.append('1');
                d--;
            } else {
                ans.append('0');
            }
        }

        int cnt = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (x.charAt(i) == '1' && ans.charAt(i) == '0') {
                cnt++;
            }
        }

        if (cnt > d) {
            ans.setCharAt(n - cnt, '1');
            d--;
        }

        while (d > 0) {
            for (int i = n - 1; i >= 0; i--) {
                if (ans.charAt(i) == '0') {
                    ans.setCharAt(i, '1');
                    d--;
                    break;
                }
            }
        }

        System.out.println(ans.toString());
    }
}
