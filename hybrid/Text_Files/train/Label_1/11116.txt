#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a;
  cin >> n >> a;
  int arr[n];
  for (int i = 0; i < n; i++) { cin >> arr[i]; }
  int l = 0, r = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    r += arr[i];
    if (i + a < n) r += arr[i + a];
    if (i - a >= 0) l += arr[i - a];
    if (l + r == 1) ans++;
  }
  cout << ans << endl;
  return 0;
}
