n = int(input())
sequence = list(map(int, input().split()))

# Count the frequency of each number in the sequence
frequency = [0] * 8
for num in sequence:
    frequency[num] += 1

# Check for possible combinations
groups = []
while frequency[1] > 0 and frequency[2] > 0 and frequency[4] > 0:
    groups.append([1, 2, 4])
    frequency[1] -= 1
    frequency[2] -= 1
    frequency[4] -= 1

while frequency[1] > 0 and frequency[2] > 0 and frequency[6] > 0:
    groups.append([1, 2, 6])
    frequency[1] -= 1
    frequency[2] -= 1
    frequency[6] -= 1

while frequency[1] > 0 and frequency[3] > 0 and frequency[6] > 0:
    groups.append([1, 3, 6])
    frequency[1] -= 1
    frequency[3] -= 1
    frequency[6] -= 1

# If any element is left ungrouped, print -1
if sum(frequency) > 0:
    print(-1)
else:
    for group in groups:
        print(*group)
