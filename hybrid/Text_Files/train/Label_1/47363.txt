import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), m = sc.nextInt(), p = 0;
    int[] a = new int[n], b = new int[m], sum = new int[n + 1];
    for (int i = 0; i < n; i++) a[i] = sc.nextInt();
    for (int i = 0; i < m; i++) b[i] = sc.nextInt();
    sum[0] = a[0];
    for (int i = 1; i < n; i++) {
      sum[i] = sum[i - 1] + a[i];
      if (sum[i] < 0) {
        sum[i] = 0;
        p = i + 1;
      }
    }
    for (int x : b) {
      int l = 0, r = n - 1, q = -1;
      while (l <= r) {
        int mid = (l + r) / 2;
        if (sum[mid] + x >= 0) {
          q = mid;
          l = mid + 1;
        } else {
          r = mid - 1;
        }
      }
      System.out.println(q == -1 ? p : n - q);
    }
  }
}
