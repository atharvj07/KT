#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>
using namespace std;
int main() {
    int n;
    cin >> n;
    if (n == 1) {
        cout << "1 1\n";
        return 0;
    }
    vector<int> a(n), freq(1e6 + 4, 0);
    unordered_map<int, pair<int, int>> freq_pos;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        freq[a[i]]++;
        if (freq_pos[a[i]].first == 0) {
            freq_pos[a[i]] = make_pair(i + 1, i + 1);
        } else
            freq_pos[a[i]].second = i + 1;
    }
    int freq_max = *max_element(freq.begin(), freq.end());
    for (int i = 1; i <= 1e6 + 3; i++) {
        if (freq[i] == freq_max) {
            cout << freq_pos[i].first << " " << freq_pos[i].second << "\n";
            return 0;
        }
    }
}
