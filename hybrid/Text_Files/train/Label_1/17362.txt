#include <bits/stdc++.h>
using namespace std;
vector<int> v;
int main() {
  int n, q;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    v.push_back(x);
  }
  sort(v.begin(), v.end());
  cin >> q;
  for (int i = 0; i < q; i++) {
    int m;
    cin >> m;
    int ans = upper_bound(v.begin(), v.end(), m) - v.begin();
    cout << ans << "\n";
  }
  return 0;
}
