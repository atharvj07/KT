n = int(input())
shows = []
for _ in range(n):
    l, r = map(int, input().split())
    shows.append((l, r, 1))  # 1 for start
    shows.append((r, l, -1))  # -1 for end

shows.sort()
tv_count = 0
for _, _, delta in shows:
    tv_count += delta
    if tv_count > 2:
        print("NO")
        break
else:
    print("YES")
