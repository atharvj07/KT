import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int p = in.nextInt();
    int[] time = new int[n];
    for (int i = 0; i < n; i++) {
      time[i] = in.nextInt();
    }
    int[] finish = new int[n];
    int lastFinish = 0;
    for (int i = 0; i < n; i++) {
      finish[i] = Math.max(time[i], lastFinish) + p;
      lastFinish = finish[i];
    }
    for (int i = 0; i < n; i++) {
      System.out.print(finish[i] + " ");
    }
    System.out.println();
  }
}
