#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
const int mod=1000000007;
const int N=3e5+5;
int n,m,q[N],a[N],b[N],c[N],d[N];

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin>>m;
    for(int i=1; i<=m; i++) cin>>q[i];
    cin>>n;
    for(int i=1; i<=n; i++) {
       cin>>a[i];
       b[i]=a[i];
    }
    sort(b+1,b+n+1);
    for(int i=1; i<=m; i++) {
        int t=0;
        for(int j=1; j<=n; j++) {
            if(t>=q[i]) break;
            c[i]+=min(a[j],b[t+1]);
            t++;
        }
        c[i]+=b[1]*min(2,q[i]-t);
    }
    int ans=1e9;
    for(int i=1; i<=m; i++) ans=min(ans,c[i]);
    cout<<ans<<endl;
}
