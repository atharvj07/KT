import java.io.*;
import java.util.*;

class Main {
    static char[][] keyboard;
    static int n, m, x;
    static boolean[][] canUseOneHand;
    static int[] dx = {-1, 0, 1, 0};
    static int[] dy = {0, 1, 0, -1};

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        x = Integer.parseInt(st.nextToken());
        keyboard = new char[n][m];
        canUseOneHand = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            String line = br.readLine();
            for (int j = 0; j < m; j++) {
                keyboard[i][j] = line.charAt(j);
            }
        }
        st = new StringTokenizer(br.readLine());
        int q = Integer.parseInt(st.nextToken());
        String text = br.readLine();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (keyboard[i][j] != 'S') {
                    canUseOneHand[i][j] = true;
                } else {
                    for (int k = 0; k < 4; k++) {
                        int nx = i + dx[k];
                        int ny = j + dy[k];
                        if (nx >= 0 && nx < n && ny >= 0 && ny < m && keyboard[nx][ny] != 'S') {
                            canUseOneHand[i][j] = true;
                        }
                    }
                }
            }
        }
        int cnt = 0;
        for (int i = 0; i < q; i++) {
            char c = text.charAt(i);
            boolean found = false;
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < m; k++) {
                    if (canUseOneHand[j][k] && (Character.toLowerCase(c) == keyboard[j][k])) {
                        found = true;
                        break;
                    }
                }
            }
            if (!found) {
                cnt++;
            }
        }
        if (cnt == 0) {
            System.out.println(0);
            return;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!canUseOneHand[i][j]) {
                    int dist = Integer.MAX_VALUE;
                    for (int k = 0; k < 4; k++) {
                        int nx = i + dx[k];
                        int ny = j + dy[k];
                        if (nx >= 0 && nx < n && ny >= 0 && ny < m && canUseOneHand[nx][ny]) {
                            dist = Math.min(dist, dx[k] * dx[k] + dy[k] * dy[k]);
                        }
                    }
                    if (dist > x * x) {
                        System.out.println(-1);
                        return;
                    }
                }
            }
        }
        System.out.println(cnt);
    }
}
