#include <iostream>

using namespace std;

int main() {
    int n, k, c[105][105];
    cin >> n >> k;
    for (int i = 0; i < n * k; i++) {
        int x;
        cin >> x;
        c[x][0]++;
        c[x][c[x][0]] = i + 1;
    }
    for (int i = 1; i <= n; i++) {
        cout << c[i][1] << ' ' << c[i][k] << endl;
    }
    return 0;
}
