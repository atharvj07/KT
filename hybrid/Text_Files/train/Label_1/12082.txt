#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[10];
int main() {
    for(ll i=1;i<=6;i++) cin >> a[i];
    ll n; cin >> n;
    ll b[n+3];
    for(ll i=1;i<=n;i++) {
        cin >> b[i];
        b[i]-=a[1];
    }
    sort(b+1,b+n+1);
    ll mn = LONG_LONG_MAX;
    for(ll i=0;i<n;i++) {
        mn=min(mn,b[i+1]-b[i]);
    }
    cout << mn << endl;
}
