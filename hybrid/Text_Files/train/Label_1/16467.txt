#include <iostream>
#include <vector>
#include <cmath>

struct Vec {
    int x, y;
    double angle;
};

int main() {
    int n;
    std::cin >> n;
    std::vector<Vec> v(n);
    for (int i = 0; i < n; i++) {
        std::cin >> v[i].x >> v[i].y;
        v[i].angle = atan2(v[i].y, v[i].x);
    }
    double min_angle = 2 * std::acos(-1);
    int a = 0, b = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            double angle = std::abs(v[i].angle - v[j].angle);
            if (angle > std::acos(-1)) {
                angle = 2 * std::acos(-1) - angle;
            }
            if (angle < min_angle) {
                min_angle = angle;
                a = i + 1;
                b = j + 1;
            }
        }
    }
    std::cout << a << " " << b << std::endl;
    return 0;
}
