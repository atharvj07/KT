#include<bits/stdc++.h>
#define pii pair<int,int>
using namespace std;
int n, k, fa[200001], sz[200001], ans[200001], col[200001], prv[200001], lastans = 0, mxsz[200001];
vector<pii> g[200001];
pii getrt(int u, int f) {
	pii ret = pii(u, 0);
	for(pii v : g[u]) {
		if(v.first == f) continue;
		pii tmp = getrt(v.first, u);
		ret = max(ret, tmp);
		sz[u] = max(sz[u], sz[v.first]);
	}
	sz[u] = max(sz[u], n - sz[u]);
	return ret;
}
void solve(int u, int f, int c) {
	col[u] = c;
	prv[u] = f;
	for(pii v : g[u]) {
		if(v.first == f) continue;
		if(col[v.first] == col[u]) {
			lastans++;
			solve(v.first, u, c);
		}
		else solve(v.first, u, c);
	}
}
bool cmp(pii a, pii b) {
	return a.second > b.second;
}
int main() {
	cin >> n >> k;
	for(int i = 1; i < n; i++) {
		int x, y; cin >> x >> y;
		g[x].push_back(pii(y, i));
		g[y].push_back(pii(x, i));
	}
	pii rt = getrt(1, 0);
	rt = getrt(rt.first, 0);
	int nowc = 0;
	int last = -1;
	for(pii v : g[rt.first]) {
		if(v.first == fa[rt.first]) continue;
		solve(v.first, rt.first, ++nowc);
		ans[v.second] = nowc;
		ans[prv[v.first]] = nowc;
		if(sz[prv[v.first]] <= sz[v.first]) {
			fa[prv[v.first]] = v.first;
			last = prv[v.first];
		}
		else fa[v.first] = prv[v.first];
	}
	int cur = rt.first;
	while(cur != last) {
		ans[prv[cur]] = nowc + 1;
		fa[prv[cur]] = last;
		cur = prv[cur];
	}
	vector<pii> mx;
	for(int i = 1; i <= n; i++) {
		mx.push_back(pii(sz[i], i));
	}
	sort(mx.begin(), mx.end(), cmp);
	while(lastans > k) {
		int cl = col[mx[mxsz[nowc]].second];
		mxsz[nowc++] = mx[lastans - k - 1].second;
		solve(mx[mxsz[nowc - 1]].second, last, nowc);
		for(int i = mx[mxsz[nowc - 1]].second; i != fa[i]; i = fa[i]) {
			ans[prv[i]] = nowc;
		}
	}
	cout << nowc << '\n';
	for(int i = 1; i < n; i++) cout << ans[i] << ' ';
	cout << '\n';
}
