import java.util.*;

public class Main {
    static long[][] f;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        f = new long[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                f[i][j] = -1;
            }
        }
        for (int i = 0; i < m; i++) {
            int xi = sc.nextInt() - 1;
            int yi = sc.nextInt() - 1;
            System.out.println(dp(xi, yi, a));
        }
    }

    public static long dp(int xi, int yi, int[] a) {
        if (f[xi][yi] != -1) {
            return f[xi][yi];
        }
        if (xi == 0) {
            f[xi][yi] = a[yi];
            return f[xi][yi];
        }
        if (xi > yi) {
            f[xi][yi] = 0;
            return f[xi][yi];
        }
        f[xi][yi] = Math.min(dp(xi - 1, yi, a), dp(xi - 1, yi - 1, a)) + a[yi];
        return f[xi][yi];
    }
}
