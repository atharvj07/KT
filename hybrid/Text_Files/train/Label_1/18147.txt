#include <iostream>
#include <cmath>
using namespace std;
int main() {
    double x1, y1, r1, x2, y2, r2;
    cin >> x1 >> y1 >> r1 >> x2 >> y2 >> r2;
    double d = sqrt((x2 - x1) * (x2 - x1) + (y2 - y1) * (y2 - y1));
    if (d > r1 + r2) cout << "0.00000000000000000000";
    else if (d <= abs(r2 - r1)) cout << min(r1 * r1, r2 * r2) * acos(-1);
    else {
        double ang1 = acos((r1 * r1 + d * d - r2 * r2) / (2 * r1 * d));
        double ang2 = acos((r2 * r2 + d * d - r1 * r1) / (2 * r2 * d));
        cout << r1 * r1 * (ang1 - sin(ang1) * cos(ang1)) + r2 * r2 * (ang2 - sin(ang2) * cos(ang2));
    }
    return 0;
}
