#include <bits/stdc++.h>
using namespace std;
int n,m,x[11],y[11],z[11];
int main(){
    cin>>n>>m;
    for(int i=0;i<n;i++){
        cin>>x[i];
    }
    for(int i=0;i<m;i++){
        cin>>y[i];
        z[y[i]]=1;
    }
    for(int i=n-1;i>=0;i--){
        if(z[x[i]]){
            for(int j=i-1;j>=0;j--){
                if(x[j]==x[i]){
                    i=j;
                    break;
                }
                else if(!z[x[j]]){
                    break;
                }
            }
            cout<<x[i]<<" ";
        }
    }
    return 0;
}
