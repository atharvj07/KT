#include <iostream>

int main() {
  int n, x, y, c;
  std::cin >> n >> x >> y >> c;
  if (n * n <= c)
    std::cout << 0 << std::endl;
  else if (c > n)
    std::cout << 1 << std::endl;
  else if (c == 1)
    std::cout << n * n - 1 << std::endl;
  else
    std::cout << n * n - c << std::endl;
  return 0;
}
