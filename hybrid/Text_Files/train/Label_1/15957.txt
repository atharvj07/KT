#include <bits/stdc++.h>
using namespace std;

const int maxn = 2e5 + 10;
int n, c, a[maxn], pre[maxn];
long long dp[maxn][2];

int main() {
  cin >> n >> c;
  for (int i = 1; i <= n; i++) cin >> a[i];
  sort(a + 1, a + n + 1);
  pre[0] = 0;
  for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + a[i];
  for (int i = 2; i <= n; i++)
    for (int j = 0; j < c; j++) {
      dp[i][j] = 1e18;
      dp[i][j] = min(dp[i][j], dp[i - 1][j] + a[i]);
      for (int z = i / c * c; z >= max(j, i - c + 1); z--)
        dp[i][j] = min(dp[i][j], dp[z - 1][j] + pre[i] - pre[z - 1]);
    }
  cout << dp[n][0];
  return 0;
}
