import math

n, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
x = []
for i in range(len(a)):
    if i + 1 < len(a) and a[i] == a[i + 1]:
        continue
    x.append([a[i], i])
mn = 0
for i in range(len(x)):
    if i + k - 1 < len(x):
        mn = max(mn, math.ceil(math.log2(x[i + k - 1][0] / x[i][0])))
print(mn)
