#include <iostream>
using namespace std;
typedef pair<int, int> pii;
const int MAXN = 55;
char board[MAXN][MAXN];
pii gap[MAXN * MAXN];
int gsz, ans[MAXN * MAXN];
int dx[] = {1, 0, -1, 0};
int dy[] = {0, 1, 0, -1};
int isgood() {
  for (int i = 0; i < gsz; ++i) {
    pii p = gap[i];
    if (board[p.first - 1][p.second] == '#' && board[p.first + 1][p.second] == '#' && board[p.first][p.second - 1] == '#' && board[p.first][p.second + 1] == '#') {
      ans[i] = 0;
    } else if (board[p.first - 1][p.second] == '#' && board[p.first + 1][p.second] == '#' && board[p.first][p.second - 1] == '#' && board[p.first][p.second + 1] == '.') {
      ans[i] = 1;
    } else if (board[p.first - 1][p.second] == '#' && board[p.first + 1][p.second] == '#' && board[p.first][p.second - 1] == '.' && board[p.first][p.second + 1] == '#') {
      ans[i] = 2;
    } else if (board[p.first - 1][p.second] == '#' && board[p.first + 1][p.second] == '.' && board[p.first][p.second - 1] == '#' && board[p.first][p.second + 1] == '#') {
      ans[i] = 3;
    } else if (board[p.first - 1][p.second] == '.' && board[p.first + 1][p.second] == '#' && board[p.first][p.second - 1] == '#' && board[p.first][p.second + 1] == '#') {
      ans[i] = 4;
    } else {
      ans[i] = -1;
      return 0;
    }
  }
  return 1;
}
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= n; ++j) {
      cin >> board[i][j];
      if (board[i][j] == '.') gap[gsz++] = {i, j};
    }
  }
  if (isgood()) {
    cout << "YES" << endl;
    for (int i = 0; i < gsz; ++i) { cout << ans[i] << " "; }
    cout << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
