import sys

input = sys.stdin.readline

n, m = map(int, input().split())
x = [0] * (n+1)
y = [0] * (n+1)
for i in range(1, n+1):
    x[i], y[i] = map(int, input().split())

ban = [[]]
for _ in range(m):
    u, v = map(int, input().split())
    ban += [[u, v]]

g = [[]]
for i in range(n+1):
    g += [[0] * (n+1)]

for i in range(1, n+1):
    for j in range(i+1, n+1):
        if [i, j] not in ban and [j, i] not in ban:
            g[i] += [j]
            g[j] += [i]

vis = [False] * (n+1)

def dfs(node, score):
    vis[node] = True
    ret = score
    for u in g[node]:
        if not vis[u]:
            ret += dfs(u, x[u] if score == y[node] else y[u])
    return ret

ans = []
for i in range(1, n+1):
    if not vis[i]:
        ans += [dfs(i, x[i])]

for v in ans:
    print(v)
