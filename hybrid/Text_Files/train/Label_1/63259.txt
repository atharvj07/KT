n=int(input())
a=list(map(int,input().split()))
fre=[0]*2500005
for i in a: fre[i]+=1
ans=0
for i in a:
    for j in a:
        if i==j:
            if fre[i]==1:
                ans+=0
            elif fre[i]==2:
                ans+=1
            else:
                ans+=2
        else:
            if fre[i]==fre[j]==1:
                ans+=0
            elif fre[i]>1 and fre[j]>1:
                ans+=fre[i]*fre[j]
                if fre[i]==fre[j]:
                    ans-=fre[i]
            else:
                ans+=fre[i]*fre[j]
print('YES'if ans else 'NO')
if ans:
    for i in a:
        for j in a:
            if i==j:
                if fre[i]==1:
                    continue
                if fre[i]==2:
                    print(a.index(i)+1,a.index(i)+1,a.index(i)+1,a.index(i)+1)
                    exit()
                if fre[i]>2:
                    print(a.index(i)+1,a.index(i)+1,a.index(i)+1,a.index(i)+2)
                    exit()
            else:
                if i+j==a[a.index(j)+1]:
                    print(a.index(i)+1,a.index(j)+1,a.index(j)+1,a.index(j)+2)
                    exit()
                if i+j==a[a.index(j)-1]:
                    print(a.index(i)+1,a.index(j)-1,a.index(j)-1,a.index(j)+1)
                    exit()
