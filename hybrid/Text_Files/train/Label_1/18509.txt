#include<bits/stdc++.h>
#define FOR(i,a,b) for (int i = (a), _b = (b); i <= _b; i++)
#define FORD(i,b,a) for (int i = (b), _a = (a); i >= _a; i--)
#define REP(i,n) for (int i = 0, _n = (n); i < _n; i++)
#define FORE(i, v) for (__typeof__((v).begin()) i = (v).begin(); i != (v).end(); i++)
#define ALL(v) (v).begin(), (v).end()
#define SZ(v) ((int) (v).size())
#define PB push_back
#define MP make_pair
#define F first
#define S second
#define X first
#define Y second
#define maxn 200011
using namespace std;

typedef long long ll;
typedef pair<ll, ll> ii;

int n, m, q, a[maxn], deg[maxn], go[maxn], ans[maxn];
vector<int> vt[maxn];

void BFS(int id) {
    queue<int> qu;
    go[id] = 0;
    qu.push(id);
    while (!qu.empty()) {
        int u = qu.front(); qu.pop();
        ans[id] += go[u];
        REP(i, SZ(vt[u])) {
            int v = vt[u][i];
            if (go[v] == -1) {
                if (a[u] > a[v]) go[v] = 1;
                else go[v] = 0;
                qu.push(v);
            }
        }
    }
}

int main() {
    ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    cin >> n >> m;
    FOR(i, 1, n) a[i] = i;
    FOR(i, 1, m) {
        int u, v; cin >> u >> v;
        vt[u].PB(v); vt[v].PB(u);
        deg[u]++; deg[v]++;
    }
    cin >> q;
    while (q--) {
        int type;
        cin >> type;
        if (type == 1) {
            int u, v;
            cin >> u >> v;
            vt[u].PB(v); vt[v].PB(u);
            deg[u]++; deg[v]++;
        } else if (type == 2) {
            int u, v;
            cin >> u >> v;
            REP(i, SZ(vt[u])) {
                if (vt[u][i] == v) {vt[u].erase(vt[u].begin() + i); break;}
            }
            REP(i, SZ(vt[v])) {
                if (vt[v][i] == u) {vt[v].erase(vt[v].begin() + i); break;}
            }
            deg[u]--; deg[v]--;
        } else {
            int cnt = 0;
            FOR(i, 1, n) {
                if (deg[i] == 0) continue;
                FOR(j, 1, n) go[j] = -1;
                BFS(i);
                cnt += ans[i];
                REP(j, SZ(vt[i])) {
                    deg[vt[i][j]]--;
                    vt[vt[i][j]].erase(find(ALL(vt[vt[i][j]]), i));
                }
                vt[i].clear(); deg[i] = 0;
                //cout << cnt << " ";
            }
            cout << n - cnt << endl;
        }
    }
}
