import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static int n, m;
    static int[][] map, d, cnt;

    static int[] dx = {0, 0, 1, -1};
    static int[] dy = {1, -1, 0, 0};

    static class Point {
        int x, y;

        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static boolean isInside(int x, int y) {
        return 1 <= x && x <= n && 1 <= y && y <= m;
    }

    static int bfs(int sx, int sy, int ex, int ey) {
        int ans = -1;
        Queue<Point> q = new LinkedList<>();

        q.add(new Point(sx, sy));
        d[sx][sy] = 1;

        while (!q.isEmpty()) {
            Point now = q.poll();
            int x = now.x;
            int y = now.y;

            if (x == ex && y == ey) {
                ans = d[x][y] - 1;
                break;
            }

            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];

                if (isInside(nx, ny) && map[nx][ny] != 1 && d[nx][ny] == 0) {
                    q.add(new Point(nx, ny));
                    d[nx][ny] = d[x][y] + 1;
                }
            }
        }

        return ans;
    }

    static int solve() {
        int group = 0;
        int[] ax = new int[5];
        int[] ay = new int[5];

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (map[i][j] == 2 || map[i][j] == 3) {
                    ax[++group] = i;
                    ay[group] = j;
                }
            }
        }

        int ans = 0;
        for (int i = 1; i < group; i++) {
            for (int j = i + 1; j <= group; j++) {
                for (int a = 1; a <= n; a++) {
                    for (int b = 1; b <= m; b++) {
                        d[a][b] = 0;
                    }
                }

                int len1 = bfs(ax[i], ay[i], ax[j], ay[j]);
                cnt = new int[n + 1][m + 1];
                for (int a = 1; a <= n; a++) {
                    for (int b = 1; b <= m; b++) {
                        cnt[a][b] = 0;
                    }
                }

                int len2 = bfs(ax[j], ay[j], ax[i], ay[i]);

                if (len1 == -1 || len2 == -1) {
                    return 0;
                }

                ans += Math.min(len1, len2);
            }
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            n = sc.nextInt();
            m = sc.nextInt();
            if (n == 0 && m == 0) {
                break;
            }

            map = new int[n + 1][m + 1];
            d = new int[n + 1][m + 1];

            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= m; j++) {
                    map[i][j] = sc.nextInt();
                }
            }

            System.out.println(solve());
        }
    }
}
