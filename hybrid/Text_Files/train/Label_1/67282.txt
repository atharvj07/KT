import sys

n, k = [int(x) for x in sys.stdin.readline().split()]
a = [int(x) for x in sys.stdin.readline().split()]
b = [int(x) for x in sys.stdin.readline().split()]
c = [0] * (k+1)
d = [0] * (k+1)

for i in range(n):
    c[a[i]] += 1
    d[a[i]] = min(d[a[i]], b[i])

need = 0
ans = 0
for i in range(1, k+1):
    if c[i] == 0:
        need += 1
    else:
        if need > 0:
            ans += min(d[i], need * b[i])
            need = 0
        d[i] = min(d[i], b[i])

print(ans)
