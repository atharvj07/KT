import sys

a, b, c = map(int, input().split())
m = int(input())
price = [[int(input().split()[0]), input().split()[1]] for _ in range(m)]
price.sort(key=lambda x: x[0])
ans = 0
cost = 0
for val, tp in price:
    if (a and tp == 'USB') or (b and tp == 'PS/2') or (c and (tp == 'USB' or tp == 'PS/2')):
        ans += 1
        cost += val
        if tp == 'USB':
            a -= 1
        else:
            b -= 1
        c -= 1
print(ans, cost)
