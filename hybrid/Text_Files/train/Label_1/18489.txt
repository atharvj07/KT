#include<iostream>
#include<vector>
using namespace std;
const int N=2e5+5;
int n,par[N],val[N],dp[N][2],ans;
vector<pair<int,int>>g[N];
void dfs(int u){
    for(const auto &[v,w]:g[u]){
        if(v!=par[u]){
            par[v]=u;
            int p=(val[u]^w);
            val[v]=p;
            dfs(v);
        }
    }
}
void dfs2(int u){
    dp[u][val[u]]=1;
    for(const auto &[v,w]:g[u]){
        if(v!=par[u]){
            dfs2(v);
            for(int i=0;i<2;i++){
                for(int j=0;j<2;j++){
                    if((i|j)!=val[u]){
                        dp[u][i|j]+=dp[v][i]*dp[v][j];
                    }
                }
            }
        }
    }
    for(int i=0;i<2;i++){
        for(int j=0;j<2;j++){
            if(i==j){
                ans+=dp[u][i]*dp[u][j]/2;
            }
            else{
                ans+=dp[u][i]*dp[u][j];
            }
        }
    }
}
int main(){
    cin>>n;
    for(int i=2;i<=n;i++){
        int u,v,w;
        cin>>u>>v>>w;
        g[u].push_back({v,w});
        g[v].push_back({u,w});
    }
    dfs(1);
    dfs2(1);
    cout<<ans<<endl;
    return 0;
}
