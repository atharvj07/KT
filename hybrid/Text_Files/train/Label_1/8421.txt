#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

int const mod = 5;
int const order = 4;

int powmod(int base, long long p) {
    if (p == 0) return 1;
    if (p == 1) return base;
    int t = powmod(base, p/2);
    t = (1LL * t * t) % mod;
    if (p&1) t = (1LL * t * base) % mod;
    return t;
}

int main() {
    long long n; cin >> n;
    int ans = 0;
    for (int i = 0; i < order; ++i) {
        ans += powmod(i+1, n);
        ans %= mod;
    }
    cout << ans << endl;
}
