import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int m = input.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = input.nextInt();
        }
        int max = 0, count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int ratio = a[i] / b[j];
                if (a[i] % b[j] == 0) {
                    if (ratio > max) {
                        max = ratio;
                        count = 1;
                    } else if (ratio == max) {
                        count++;
                    }
                }
            }
        }
        System.out.println(count);
    }
}
