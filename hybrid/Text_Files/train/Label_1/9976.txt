#include <iostream>
using namespace std;
int main() {
  int s, v1, v2, t1, t2;
  cin >> s >> v1 >> v2 >> t1 >> t2;
  int t = t1 + t2 + (s - 1) * max(v1, v2);
  if (t1 + (s - 1) * v1 == t2 + (s - 1) * v2)
    cout << "Friendship";
  else if (t1 + (s - 1) * v1 < t2 + (s - 1) * v2)
    cout << "First";
  else
    cout << "Second";
  return 0;
}
