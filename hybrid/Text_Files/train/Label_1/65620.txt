n, k = map(int, input().split())
data = [list(input()) for _ in range(n)]
ans = (-1, -1)
mx = -1

for i in range(n):
    for j in range(n):
        if data[i][j] == '#':
            continue

        cnt = 0
        flag = False

        if i + k - 1 < n:
            for t in range(i, i + k):
                if data[t][j] == '.':
                    cnt += 1
                else:
                    flag = True
                    break
        if cnt == k and not flag:
            ans = (i + 1, j + 1)
            mx = k

        cnt = 0
        flag = False

        if j + k - 1 < n:
            for t in range(j, j + k):
                if data[i][t] == '.':
                    cnt += 1
                else:
                    flag = True
                    break
        if cnt == k and not flag:
            ans = (i + 1, j + 1)
            mx = k

print(*ans)
