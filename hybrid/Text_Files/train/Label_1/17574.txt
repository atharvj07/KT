#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int mod = 998244353;
const int maxn = 2003;

int a[maxn][maxn];
ll dp[4][maxn][maxn];

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    string s;
    cin >> s;
    for (int j = 1; j <= m; j++) { a[i][j] = s[j - 1] == '#'; }
  }

  dp[1][1][2] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 2; j <= m; j++) {
      if (a[i][j] || a[i][j - 1]) continue;

      dp[0][i][j] = (dp[0][i - 1][j] + dp[0][i][j - 1] - dp[0][i - 1][j - 1] +
                    mod) %
                   mod;
      dp[1][i][j] = (dp[1][i - 1][j - 1] + dp[0][i][j] + dp[2][i - 1][j - 1] -
                    dp[2][i][j - 1] + mod) %
                   mod;
      dp[2][i][j] = (dp[1][i - 1][j - 1] + dp[2][i - 1][j] + dp[3][i - 1][j - 1] -
                    dp[3][i][j - 1] + mod) %
                   mod;
      dp[3][i][j] = (dp[2][i - 1][j - 1] + dp[3][i - 1][j] - dp[3][i - 1][j - 1] +
                    mod) %
                   mod;
    }
  }

  ll ans = (dp[0][n][m] + dp[1][n][m] + dp[2][n][m] + dp[3][n][m]) % mod;
  cout << ans << endl;
  return 0;
}
