#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ll n;
    cin >> n;
    ll ans = 0;
    for (ll i = 2; i * i <= n; i++) {
        if (n % i == 0) { ans = max(ans, i); }
        if (i * i != n && n % (n / i) == 0) { ans = max(ans, n / i); }
    }
    ans = max(ans, n);
    cout << ans << endl;
    return 0;
}
