#include <iostream>
#include <unordered_map>
using namespace std;

int main() {
    string s, t;
    cin >> s >> t;
    unordered_map<char, char> map;
    int cnt0 = 0, cnt1 = 0;
    for (char c : s) {
        if (c == '0') cnt0++;
        if (c == '1') cnt1++;
    }
    for (int i = 0; i < t.size(); i++) {
        if (map.count(t[i])) {
            if (map[t[i]] == '0' && s[i] == '1') {
                cout << 0 << endl;
                return 0;
            }
            if (map[t[i]] == '1' && s[i] == '0') {
                cout << 0 << endl;
                return 0;
            }
        } else {
            if (cnt0 > 0) {
                map[t[i]] = '0';
                cnt0--;
            } else if (cnt1 > 0) {
                map[t[i]] = '1';
                cnt1--;
            } else {
                cout << 0 << endl;
                return 0;
            }
        }
    }
    if (cnt0 > 0 || cnt1 > 0) {
        cout << 0 << endl;
        return 0;
    }
    cout << (1LL << cnt0) * (1LL << cnt1) << endl;
    return 0;
}
