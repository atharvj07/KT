q = int(input())
for _ in range(q):
    n = int(input())
    boards = [list(map(int, input().split())) for _ in range(n)]
    dp = [[0, 0, 0] for _ in range(n)]
    
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = float('inf')
            for k in range(3):
                if boards[i-1][0] + k != boards[i][0] + j:
                    dp[i][j] = min(dp[i][j], dp[i-1][k] + j * boards[i][1])
    
    print(min(dp[-1]))
