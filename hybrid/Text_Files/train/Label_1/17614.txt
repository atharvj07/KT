#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<int>> rows(n, vector<int>(m));
        vector<vector<int>> cols(m, vector<int>(n));
        vector<int> r(n * m), c(n * m);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                cin >> rows[i][j];
                rows[i][j]--;
                r[rows[i][j]] = i;
            }
        }
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                cin >> cols[i][j];
                cols[i][j]--;
                c[cols[i][j]] = i;
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int x = rows[i][j];
                rows[i][j] = cols[c[x]][r[x]];
            }
            for (int j = 0; j < m; j++) rows[i][j]++;
            for (int j = 0; j < m; j++) cout << rows[i][j] << ' ';
            cout << '\n';
        }
    }
    return 0;
}
