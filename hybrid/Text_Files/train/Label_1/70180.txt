n,m=map(int,input().split())
g=[[0]*m for i in range(n)]
for i in range(n):
	g[i]=[int(j) for j in input().split()]
ans=n*m
for i in range(n):
	for j in range(m):
		if g[i][j]>0:
			ans-=1
			for k in range(n):
				if g[k][j]>0:
					ans-=1
			for k in range(m):
				if g[i][k]>0:
					ans-=1
print(ans)
