import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            long sum = 0;
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
                sum += a[j];
            }
            long need = sum / n;
            boolean found = false;
            for (int j = 0; j < n && !found; j++) {
                if (a[j] <= need) {
                    need = a[j];
                    found = true;
                }
            }
            for (int j = 0; j < n; j++) {
                need = Math.max(need, (sum - a[j]) / (n - 1));
            }
            System.out.println(need);
        }
    }
}
