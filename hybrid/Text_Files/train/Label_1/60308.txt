def min_length_after_abbreviation(n, text):
    words = text.split()
    max_saved = 0

    for i in range(n):
        for j in range(i + 1, n + 1):
            segment = words[i:j]
            segment_str = ' '.join(segment)
            abbreviation = ''.join(word[0].upper() for word in segment)
            saved = 0
            k = text.find(segment_str)
            while k != -1:
                saved += len(segment_str) - len(abbreviation)
                k = text.find(segment_str, k + len(segment_str))
            if saved > max_saved and text.count(segment_str) > 1:
                max_saved = saved

    return len(text) - max_saved

n = int(input().strip())
text = input().strip()
print(min_length_after_abbreviation(n, text))
