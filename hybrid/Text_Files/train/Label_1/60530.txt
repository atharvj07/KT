from sys import stdin
input = stdin.readline

for _ in range(int(input())):
    s = input().strip()
    bal = 0
    ans = 0
    for ch in s:
        if ch == '(' or ch == '[':
            bal += 1
        elif ch == ')':
            if bal > 0:
                bal -= 1
                ans += 1
        elif ch == ']':
            if bal > 1:
                bal -= 1
                ans += 1
    print(ans)
