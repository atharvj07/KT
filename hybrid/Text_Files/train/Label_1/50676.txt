import java.util.Scanner;

public class AdjacentNumbersMatrix {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();

            // Check if a valid matrix is possible
            if (n % 2 == 0 || n == 1) {
                System.out.println(-1);
                continue;
            }

            // Generate the matrix
            int[][] matrix = new int[n][n];
            int count = 1;

            // Fill the first row and column
            for (int i = 0; i < n; i++) {
                matrix[0][i] = count++;
            }
            for (int i = 1; i < n; i++) {
                matrix[i][0] = count++;
            }

            // Fill the remaining cells in a spiral pattern
            int row = n - 1;
            int col = n - 1;
            while (count <= n * n) {
                // Move right
                while (col > 0 && matrix[row][col - 1] == 0) {
                    matrix[row][col--] = count++;
                }

                // Move down
                while (row < n - 1 && matrix[row + 1][col] == 0) {
                    matrix[row++][col] = count++;
                }

                // Move left
                while (col < n - 1 && matrix[row][col + 1] == 0) {
                    matrix[row][col++] = count++;
                }

                // Move up
                while (row > 0 && matrix[row - 1][col] == 0) {
                    matrix[row--][col] = count++;
                }
            }

            // Print the matrix
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    System.out.print(matrix[i][j] + " ");
                }
                System.out.println();
            }
        }

        scanner.close();
    }
}
