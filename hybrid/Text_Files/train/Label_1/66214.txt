import sys

t = int(sys.stdin.readline())

for i in range(t):
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    inc = 1
    ans = 0
    for j in range(1, n):
        if a[j] < a[j - 1]:
            inc = 0
        if a[j] > a[j - 1]:
            inc = 1
        ans += inc
    print(ans)
