n = int(input())
a = list(map(int, input().split()))

# First operation: Make all elements equal to n * their value
print(1, n)
for i in a:
    print(i * (n - 1), end=' ')
print()

# Second operation: Make elements from 1 to n-1 equal to 0
if n > 1:
    print(1, n-1)
    for i in range(n-1):
        print(-a[i] * (n - 1), end=' ')
    print()

# Third operation: Make the nth element equal to 0
print(n, n)
print(-a[n-1] * n)
