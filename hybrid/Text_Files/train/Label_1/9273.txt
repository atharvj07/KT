#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  bool pal = true;
  for (int i = 0; i < n / 2; i++) {
    if (s[i] != s[n - i - 1]) {
      pal = false;
      break;
    }
  }
  cout << (pal ? "TAK" : "NIE") << endl;
  return 0;
}
