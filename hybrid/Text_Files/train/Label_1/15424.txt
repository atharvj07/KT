#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    int sum = 0;
    for (int i : a) sum += i;

    if (sum % 2 == 0) {
        cout << "-1" << endl;
    } else {
        int sum1 = 0;
        int count = 0;
        for (int i = 0; i < n; i++) {
            sum1 += a[i];
            count++;
            if (sum1 > sum / 2) {
                cout << count << endl;
                for (int j = 0; j < count; j++) cout << j + 1 << " ";
                cout << endl;
                return 0;
            }
        }
    }
}
