#include <iostream>
#include <cmath>
#include <string>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        if (k == 1) {
            cout << n << endl;
            continue;
        }
        long long x = n;
        int cnt = 0;
        string s = to_string(n);
        for (char c : s) {
            if (c != '0') {
                cnt++;
            }
        }
        if (cnt <= k) {
            cout << n << endl;
            continue;
        }
        // replace 0 in sequence with next digit
        int i = 0;
        while (cnt > k) {
            if (s[i] == '0') {
                bool found = false;
                int j = i + 1;
                while (j < s.size() && !found) {
                    if (s[j] != '0') {
                        s[i] = s[j];
                        cnt--;
                        found = true;
                    }
                    j++;
                }
                if (!found) {
                    // add new digit after last digit
                    s += string(1, char('0' + k + 1));
                    cnt--;
                }
            }
            i++;
        }
        x = stoll(s);
        cout << x << endl;
    }
    return 0;
}
