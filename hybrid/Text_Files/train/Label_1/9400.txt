#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long p, a, b, c;
    cin >> p >> a >> b >> c;
    long long t1 = (p + a - 1) / a;
    long long t2 = (p + b - 1) / b;
    long long t3 = (p + c - 1) / c;
    long long m = min({t1, t2, t3});
    cout << a * m + p - a * t1 << endl;
  }
  return 0;
}
