#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
vector<long long> g[100005], x, s;
vector<int> level;
int dfs(int v, int p) {
    level[v] = level[p] + 1;
    s[v] = x[v];
    for (auto u : g[v]) {
        if (u != p) {
            s[v] = (1LL * s[v] * dfs(u, v)) % mod;
        }
    }
    return s[v];
}
int main() {
    int n;
    cin >> n;
    level.resize(n + 1, 0);
    s.resize(n + 1);
    x.resize(n + 1);
    for (int i = 0; i < n; i++) {
        cin >> x[i + 1];
    }
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        g[a].push_back(b);
        g[b].push_back(a);
    }
    dfs(1, -1);
    long long ans = 0;
    for (int i = 1; i <= n; i++) {
        ans = (ans + (1LL * x[i] * s[i]) % mod) % mod;
    }
    cout << ans << endl;
    return 0;
}
