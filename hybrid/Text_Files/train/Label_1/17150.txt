#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> k(n);
  for (int &x : k) {
    cin >> x;
  }
  vector<vector<int>> m(n);
  for (int i = 0; i < n; i++) {
    m[i].resize(k[i] + 1);
    for (int j = 1; j <= k[i]; j++) {
      cin >> m[i][j];
    }
  }
  int ans = 2e9;
  for (int i = 0; i < n; i++) {
    int sum = 0;
    for (int j = 1; j <= k[i]; j++) {
      sum += m[i][j];
    }
    ans = min(ans, sum * 5 + sum * 15);
  }
  cout << ans << '\n';
  return 0;
}
