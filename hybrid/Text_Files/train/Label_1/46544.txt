import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    static int n, u, v;
    static List<List<Integer>> graph;
    static int[] distToTakahashi;
    static int[] distToAoki;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        u = Integer.parseInt(line[1]);
        v = Integer.parseInt(line[2]);
        graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            line = br.readLine().split(" ");
            int a = Integer.parseInt(line[0]);
            int b = Integer.parseInt(line[1]);
            graph.get(a).add(b);
            graph.get(b).add(a);
        }
        distToTakahashi = new int[n + 1];
        distToAoki = new int[n + 1];
        dfs(u, -1, 0);
        dfs(v, -1, 0);
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (distToTakahashi[i] > distToAoki[i]) {
                ans = Math.max(ans, distToTakahashi[i] - distToAoki[i]);
            }
        }
        System.out.println(ans);
    }

    static void dfs(int now, int parent, int dist) {
        if (distToTakahashi[now] != 0 && distToAoki[now] != 0) {
            return;
        }
        if (now == u) {
            distToTakahashi[now] = dist;
        }
        if (now == v) {
            distToAoki[now] = dist;
        }
        for (int next : graph.get(now)) {
            if (next == parent) {
                continue;
            }
            dfs(next, now, dist + 1);
        }
    }
}
