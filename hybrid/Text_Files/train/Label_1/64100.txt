import sys, heapq

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    h = []
    cnt = 0
    for _ in range(n):
        t, x = map(int, input().split())
        heapq.heappush(h, (x - t, t, x))
    while h:
        t, a, b = heapq.heappop(h)
        if a <= t <= b:
            cnt += 1
    print(cnt)
