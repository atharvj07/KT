import math

def volume_of_pyramids(l_3, l_4, l_5):
    volume_triangular = (math.sqrt(2)/12) * l_3**3
    volume_quadrangular = (l_4**3)/3
    volume_pentagonal = (5*(5+2*math.sqrt(5))/12) * l_5**3 / math.tan(math.radians(54))
    return volume_triangular + volume_quadrangular + volume_pentagonal

l_3, l_4, l_5 = map(int, input().split())
print(volume_of_pyramids(l_3, l_4, l_5))
