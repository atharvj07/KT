import sys
input = sys.stdin.readline

for t in range(int(input())):
    input()
    n, k = map(int,input().split())
    mines = {}
    for _ in range(n):
        x, y, t = map(int,input().split())
        mines[(x,y)] = t
    ans = 1e9
    for x, y in mines:
        if k == 0:
            ans = min(ans,mines[(x,y)])
        if (x-k, y) in mines: ans = min(ans,mines[(x-k,y)]-1)
        if (x+k, y) in mines: ans = min(ans,mines[(x+k,y)]-1)
        if (x, y-k) in mines: ans = min(ans,mines[(x,y-k)]-1)
        if (x, y+k) in mines: ans = min(ans,mines[(x,y+k)]-1)
    print(max(0,ans))
