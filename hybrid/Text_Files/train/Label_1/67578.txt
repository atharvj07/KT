n, m = map(int, input().split())
screen = [input() for _ in range(n)]

min_row, min_col, max_row, max_col = n, m, 0, 0
for i in range(n):
    for j in range(m):
        if screen[i][j] == 'w':
            min_row = min(min_row, i)
            min_col = min(min_col, j)
            max_row = max(max_row, i)
            max_col = max(max_col, j)

if max_row - min_row != max_col - min_col:
    print(-1)
else:
    frame_size = max_row - min_row + 1
    if frame_size == 1:
        for row in screen:
            print(row)
    else:
        valid = True
        for i in range(min_row, max_row + 1):
            for j in range(min_col, max_col + 1):
                if i in [min_row, max_row] or j in [min_col, max_col]:
                    if screen[i][j] == '.':
                        valid = False
                        break
                else:
                    if screen[i][j] == 'w':
                        valid = False
                        break
            if not valid:
                break
        
        if not valid:
            print(-1)
        else:
            for i in range(n):
                for j in range(m):
                    if i in [min_row, max_row] and min_col <= j <= max_col:
                        print('+' if screen[i][j] == '.' else 'w', end='')
                    elif j in [min_col, max_col] and min_row <= i <= max_row:
                        print('+' if screen[i][j] == '.' else 'w', end='')
                    else:
                        print(screen[i][j], end='')
                print()
