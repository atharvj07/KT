import math

n, x1, y1, x2, y2 = map(int, input().split())
flowers = [list(map(int, input().split())) for _ in range(n)]

d = [(x - x1)**2 + (y - y1)**2 for x, y in flowers]
d.sort()
a = [(x - x2)**2 + (y - y2)**2 for x, y in flowers]
a.sort()
ans = 10**15
for i in range(len(d)):
    for j in range(len(a)):
        if d[i] <= a[j]:
            ans = min(ans, d[i] + a[j])
print(ans)
