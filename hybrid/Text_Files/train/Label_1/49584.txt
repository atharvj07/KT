import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Integer, Integer> times = new HashMap<>();
        for (int i = 0; i < n; i++) {
            int t1 = sc.nextInt();
            int x1 = sc.nextInt();
            int y1 = sc.nextInt();
            int t2 = sc.nextInt();
            int x2 = sc.nextInt();
            int y2 = sc.nextInt();
            int dx = x2 - x1;
            int dy = y2 - y1;
            int dt = t2 - t1;
            int gcd = gcd(dx, dy);
            times.put(gcd, times.getOrDefault(gcd, 0) + 1);
        }
        int max = 0;
        for (int time : times.values()) {
            max = Math.max(max, time);
        }
        System.out.println(max);
    }

    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
