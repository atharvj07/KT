#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<string> p(n);
    for (int i = 0; i < n; i++) cin >> p[i];
    sort(p.begin(), p.end());
    int k = 0;
    for (int i = 1; i < n; i++) {
      if (p[i] == p[i - 1]) k++;
      else
        for (int j = 0; j < 4; j++)
          if (p[i][j] == p[i - 1][j]) { p[i][j]++; break; }
    }
    cout << k << endl;
    for (int i = 0; i < n; i++) cout << p[i] << endl;
  }
}
