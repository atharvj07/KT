#include <iostream>
#include <vector>
int main() {
  int q;
  std::cin >> q;
  while (q--) {
    std::string s, t;
    std::cin >> s >> t;
    int cnt = 0;
    for (size_t i = 0; i < s.size(); i++) {
      if (s[i] != t[i] && cnt % 2 == 0) { cnt++; }
      if (s[i] == t[i] && cnt % 2 == 1) { cnt++; }
    }
    if (cnt % 2 != 0 || (cnt == 0 && s < t)) {
      std::cout << "NO" << std::endl;
      continue;
    }
    std::cout << "YES" << std::endl;
  }
}
