#include <iostream>
#include <map>
using namespace std;
typedef long long ll;

map<ll, int> idx;

int main() {
  int m; cin >> m;
  for (int i = 0, x; i < m; ++i) {
    cin >> x;
    map<ll, int>::iterator it = idx.find(x);
    if (it != idx.end()) {
      cout << 1 << ' ' << it->second << '\n';
    } else {
      idx.insert(make_pair(x, i));
      idx.insert(make_pair(x^1, i));
      for (it = idx.begin(); it != idx.end(); ) {
        if (it->first != x && !(it->first&x)) {
          cout << 2 << ' ' << it->second << ' ' << i << '\n';
          idx.insert(make_pair(x^it->first, i));
          break;
        } else ++it;
      }
      if (it == idx.end()) cout << 0 << '\n';
    }
  }
  return 0;
}
