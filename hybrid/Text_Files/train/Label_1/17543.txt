#include <bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> adj(n);
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      int a;
      cin >> a;
      if (a == 1) {
        adj[i].push_back(j);
      }
    }
  }
  vector<int> deg(n);
  for (int i = 0; i < n; ++i) {
    deg[i] = adj[i].size();
  }
  int even = 0;
  for (int i = 0; i < n; ++i) {
    if (deg[i] % 2 == 0) {
      even++;
    }
  }
  if (even < 3) {
    cout << 0 << endl;
    return 0;
  }
  long long ans = 1;
  for (int i = 1; i < n - 1; ++i) {
    ans = ans * i % mod;
  }
  cout << ans << endl;
  return 0;
}
