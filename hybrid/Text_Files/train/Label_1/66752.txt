n = int(input())
min1, max1 = map(int, input().split())
min2, max2 = map(int, input().split())
min3, max3 = map(int, input().split())

ans = [0] * 3
a = n - min2 - min3
if a >= max1:
    ans[0] = max1
    ans[1] = min2
    ans[2] = min3
elif a > 0:
    ans[0] = a
    ans[1] = min2
    ans[2] = n - min2 - a
else:
    if max1 - min1 < max2 - min2:
        ans[0] = n - min2 - min3
        ans[1] = min2
        ans[2] = min3
    else:
        ans[0] = min1
        ans[1] = n - min1 - min3
        ans[2] = min3
print(*ans)
