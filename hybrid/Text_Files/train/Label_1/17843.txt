#include<bits/stdc++.h>
using namespace std;
#define int long long
const int N = 2e5 + 10;
vector<int>g[N];
int32_t main()
{
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        int x, y;
        cin >> x >> y;
        g[x].push_back(y);
        g[y].push_back(x);
    }
    int vis[N] = {};
    vector<int>p;
    function<void(int)> dfs = [&](int node) {
        vis[node] = 1;
        p.push_back(node);
        for (int var : g[node])
        {
            if (vis[var])continue;
            dfs(var);
        }
    };
    for (int i = 1; i <= n; i++)
    {
        if (vis[i])continue;
        dfs(i);
    }
    for (int var : p)cout << var << " ";
    cout << endl;
}
