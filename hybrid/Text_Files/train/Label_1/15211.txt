#include <bits/stdc++.h>

using namespace std;

const int maxn = 1010;
char mp[maxn][maxn];
int n, m;
int vis[maxn][maxn];
int dx[] = {0, 0, 1, -1}, dy[] = {1, -1, 0, 0};

int inrange(int x, int y) { return x >= 1 && x <= n && y >= 1 && y <= m; }

int dfs(int x, int y) {
  if (vis[x][y]) return 0;
  vis[x][y] = 1;
  int ret = 1;
  for (int i = 0; i < 4; i++) {
    int nx = x + dx[i], ny = y + dy[i];
    if (inrange(nx, ny) && mp[nx][ny] == mp[x][y] + 1) ret += dfs(nx, ny);
  }
  return ret;
}

int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) cin >> mp[i][j];
  int ans = 0;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++)
      if (mp[i][j] == 'D') ans = max(ans, dfs(i, j));
  if (ans == 0)
    cout << "Poor Dima!" << endl;
  else if (ans > 1)
    cout << "Poor Inna!" << endl;
  else
    cout << ans << endl;
  return 0;
}
