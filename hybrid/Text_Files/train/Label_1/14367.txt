#include <cstdio>

int d[19];

int main() {
    int t; scanf("%d", &t);
    while (t--) {
        int n, k; scanf("%d%d", &n, &k);
        for (int i = 0; i < 19; i++) d[i] = 0;
        for (int x = n; x; x /= 10) d[x % 10]++;
        while (k < 0 || d[0] > 0 || (d[0] == 0 && n == 0)) {
            int p = 0;
            for (int i = 0; i < 10; i++) {
                if (!d[i]) {
                    d[i]++;
                    p = i;
                    break;
                }
            }
            if (d[0] > 0 || (d[0] == 0 && n == 0)) {
                n = n * 10 + p;
            } else {
                n = n + p - n % 10;
            }
            d[0]--;
            k++;
        }
        printf("%d\n", n);
    }
    return 0;
}
