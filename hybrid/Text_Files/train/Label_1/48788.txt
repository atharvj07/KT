import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class SubnetMask {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        List<String> addresses = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            addresses.add(scanner.nextLine());
        }

        int[][] ipAddresses = new int[n][4];
        for (int i = 0; i < n; i++) {
            ipAddresses[i] = Arrays.stream(addresses.get(i).split("\\.")).mapToInt(Integer::parseInt).toArray();
        }

        int[] subnetMask = findSubnetMask(ipAddresses, k);
        if (subnetMask != null) {
            System.out.println(String.format("%d.%d.%d.%d", subnetMask[0], subnetMask[1], subnetMask[2], subnetMask[3]));
        } else {
            System.out.println("-1");
        }
    }

    private static int[] findSubnetMask(int[][] ipAddresses, int k) {
        int[] minSubnetMask = new int[4];
        Arrays.fill(minSubnetMask, 255);

        for (int i = 0; i < 32; i++) {
            int[] subnetMask = new int[4];
            for (int j = 0; j < 4; j++) {
                subnetMask[j] = 255 - (1 << i);
            }

            int distinctNetworks = countDistinctNetworks(ipAddresses, subnetMask);
            if (distinctNetworks == k) {
                return subnetMask;
            } else if (distinctNetworks < k) {
                break;
            } else {
                for (int j = 0; j < 4; j++) {
                    minSubnetMask[j] = Math.min(minSubnetMask[j], subnetMask[j]);
                }
            }
        }

        return countDistinctNetworks(ipAddresses, minSubnetMask) == k ? minSubnetMask : null;
    }

    private static int countDistinctNetworks(int[][] ipAddresses, int[] subnetMask) {
        int distinctNetworks = 0;
        for (int[] ipAddress : ipAddresses) {
            int networkAddress = getNetworkAddress(ipAddress, subnetMask);
            distinctNetworks = Math.max(distinctNetworks, networkAddress + 1);
        }
        return distinctNetworks;
    }

    private static int getNetworkAddress(int[] ipAddress, int[] subnetMask) {
        int networkAddress = 0;
        for (int i = 0; i < 4; i++) {
            networkAddress |= (ipAddress[i] & subnetMask[i]) << (8 * (3 - i));
        }
        return networkAddress;
    }
}
