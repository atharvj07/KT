#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n, a, b; cin >> n >> a >> b;
        if (a > (n - 1) / 2 || b > (n - 1) / 2) { cout << -1 << endl; continue; }
        vector<int> p(n);
        for (int i = 0; i < n; i++) p[i] = i + 1;
        int mx = 1, mn = n;
        for (int i = 0; i < a; i++) {
            swap(p[mx + i], p[mx + i + 1]);
            mx++;
        }
        for (int i = 0; i < b; i++) {
            swap(p[mn - 1 - i], p[mn - 2 - i]);
            mn--;
        }
        for (int i = 0; i < n; i++) cout << p[i] << ' ';
        cout << endl;
    }
}
