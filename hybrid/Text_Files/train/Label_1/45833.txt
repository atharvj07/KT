import java.util.*;
import java.io.*;

class Main {
    static class Pair implements Comparable<Pair> {
        int index;
        int pos;

        Pair(int index, int pos) {
            this.index = index;
            this.pos = pos;
        }

        @Override
        public int compareTo(Pair other) {
            return Integer.compare(this.pos, other.pos);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String[] s = br.readLine().split(" ");
        String[] t = br.readLine().split(" ");
        int[] start = new int[n], target = new int[n];
        for (int i = 0; i < n; i++) {
            start[i] = Integer.parseInt(s[i]);
            target[i] = Integer.parseInt(t[i]);
        }

        Map<Integer, List<Integer>> posToIndices = new HashMap<>();
        for (int i = 0; i < n; i++) {
            posToIndices.computeIfAbsent(start[i], k -> new ArrayList<>()).add(i);
        }

        Pair[] pairs = new Pair[n];
        for (int i = 0; i < n; i++) {
            pairs[i] = new Pair(i, target[i]);
        }
        Arrays.sort(pairs);

        StringBuilder sb = new StringBuilder();
        int moves = 0;
        for (Pair pair : pairs) {
            int index = pair.index;
            int oldPos = start[index];
            int newPos = pair.pos;
            if (oldPos == newPos) {
                continue;
            }
            List<Integer> indices = posToIndices.computeIfAbsent(newPos, k -> new ArrayList<>());
            int targetIndex = indices.isEmpty() ? -1 : indices.remove(indices.size() - 1);
            if (targetIndex == -1) {
                sb.append("NO").append("\n");
                System.out.print(sb);
                return;
            }
            start[index] = newPos;
            start[targetIndex] = oldPos;
            sb.append(index + 1).append(" ").append(targetIndex + 1).append(" ").append((newPos - oldPos) / 2).append("\n");
            moves++;
        }
        sb.insert(0, "YES\n" + moves + "\n");
        System.out.print(sb);
    }
}
