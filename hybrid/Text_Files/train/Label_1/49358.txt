import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] deg = new int[n + 1];
        int b = 0;
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            char c = sc.next().charAt(0);
            if (c == 'R') {
                deg[u]++;
                deg[v]++;
            } else {
                b++;
            }
        }
        if (n - b % 2 == 1) {
            System.out.println(-1);
            return;
        }
        List<Integer> ans = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (deg[i] % 2 == 1) {
                ans.add(i);
            }
        }
        if (ans.size() % 2 == 1) {
            ans.add(ans.get(0));
        }
        System.out.println(ans.size() / 2);
        for (int i = 0; i < ans.size(); i += 2) {
            System.out.print(ans.get(i) + " ");
        }
        System.out.println();
    }
}
