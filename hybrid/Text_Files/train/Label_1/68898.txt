n, p = map(int, input().split())
s = input()
i, j = n - 1, n - 1
while i >= 0 and j >= 0 and s[j] == s[i]:
    j -= 1
if i < 0:
    print("NO")
else:
    for x in range(ord(s[i]) + 1, ord('a') + p):
        t = s[:i] + chr(x) + chr(x) + s[j + 1:]
        if t > s:
            print(t)
            exit()
    print("NO")
