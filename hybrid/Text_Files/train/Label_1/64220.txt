import sys
n,m = map(int, sys.stdin.readline().split())
g = [[] for i in range(n+1)]
for i in range(m):
    a,b = map(int,sys.stdin.readline().split())
    g[a].append(b)
    g[b].append(a)
d = [0]*7
ans = 0
for i in range(1,n+1):
    k = []
    for j in g[i]:
        if j>i: k.append(d[j])
    k.sort()
    for j in k:
        if d[i]==j: break
        d[i]+=1
        ans+=1
print(ans)
