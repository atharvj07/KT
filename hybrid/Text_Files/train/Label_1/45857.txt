import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] x = new int[n];
        int[] y = new int[n];
        for (int i = 0; i < n; i++) {
            String[] s = br.readLine().split(" ");
            x[i] = Integer.parseInt(s[0]);
            y[i] = Integer.parseInt(s[1]);
        }
        Set<Integer> sx = new HashSet<>();
        Set<Integer> sy = new HashSet<>();
        for (int i = 0; i < n; i++) {
            sx.add(x[i]);
            sy.add(y[i]);
        }
        int[] xs = new int[sx.size()];
        int[] ys = new int[sy.size()];
        int idx = 0;
        for (int num : sx) {
            xs[idx++] = num;
        }
        idx = 0;
        for (int num : sy) {
            ys[idx++] = num;
        }
        Arrays.sort(xs);
        Arrays.sort(ys);
        List<Integer> ansX = new ArrayList<>();
        List<Integer> ansY = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            ansX.add(x[i]);
            ansY.add(y[i]);
        }
        for (int i = 0; i < xs.length; i++) {
            for (int j = 0; j < ys.length; j++) {
                if (!sx.contains(xs[i]) || !sy.contains(ys[j])) {
                    ansX.add(xs[i]);
                    ansY.add(ys[j]);
                }
            }
        }
        System.out.println(ansX.size());
        for (int i = 0; i < ansX.size(); i++) {
            System.out.println(ansX.get(i) + " " + ansY.get(i));
        }
    }
}
