n, x = map(int, input().split())
if x == 0 and n != 1:
    print("NO")
elif x > 0 and (n & 1 == x & 1):
    print("YES")
    arr = [0] * (n)
    for i in range(0, n - 1, 2):
        arr[i] = i + 1
        arr[i + 1] = x ^ i + 1
    if n & 1:
        arr[-1] = x
    print(*arr)
else:
    print("NO")
