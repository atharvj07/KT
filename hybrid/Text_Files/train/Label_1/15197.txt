#include <bits/stdc++.h>
using namespace std;

const long long INF = 1e18;
int main() {
    int n, m;
    cin >> n >> m;
    vector<int> k(n);
    for (int &x : k) cin >> x;
    unordered_map<int, vector<int>> offers;
    for (int i = 1; i <= m; ++i) {
        int x, y;
        cin >> x >> y;
        offers[y].push_back(x);
    }
    vector<long long> dist(n + 1, INF);
    dist[0] = 0;
    for (int _ = 0; _ <= 1; ++_) {
        for (int i = 1; i <= n; ++i) {
            for (int x : offers[i]) {
                dist[i] = min(dist[i], 1LL * x);
            }
        }
        for (int i = 1; i <= n; ++i) {
            long long j = 0;
            for (int cnt = 0; cnt < k[i - 1]; ++cnt) {
                dist[i] = min(dist[i], dist[j] + 2LL * (cnt + 1));
                j = min(j + 1, dist[i]);
            }
        }
    }
    for (int i = 1; i <= n; ++i) {
        if (k[i - 1] == 0) continue;
        cout << dist[i] << endl;
        return 0;
    }
}
