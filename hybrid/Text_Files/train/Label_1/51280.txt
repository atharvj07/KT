import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), m = scanner.nextInt();
        int[] a = new int[n], b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        int[] sortedA = a.clone();
        Arrays.sort(sortedA);
        int[] sortedB = b.clone();
        Arrays.sort(sortedB);
        int minX = -1;
        for (int i = 0; i < n; i++) {
            int diff = (sortedB[i] - sortedA[i] + m) % m;
            minX = minX == -1 ? diff : Math.min(minX, diff);
        }
        System.out.println(minX);
    }
}
