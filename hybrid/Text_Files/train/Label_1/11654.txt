#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int open = 0;
    for (char c : s) {
      if (c == '(') open++;
      if (c == ')') open--;
      if (open < 0) break;
    }
    cout << (open == 0 ? "YES" : "NO") << "\n";
  }
}
