#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, a, b;
    cin >> n >> a >> b;

    vector<bool> arthur(n + 1), alexander(n + 1);
    for (int i = 0; i < a; i++) {
        int x;
        cin >> x;
        arthur[x] = true;
    }
    for (int i = 0; i < b; i++) {
        int x;
        cin >> x;
        alexander[x] = true;
    }

    for (int i = 1; i <= n; i++) {
        if (arthur[i]) {
            cout << 1;
        } else if (alexander[i]) {
            cout << 2;
        }
    }

    return 0;
}
