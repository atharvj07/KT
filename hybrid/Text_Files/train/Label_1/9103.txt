#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  vector<pair<int, int>> points(3);
  for (auto &point : points) { cin >> point.first >> point.second; }

  sort(points.begin(), points.end());

  int ans = 0;
  if (points[0].second != points[1].second) ans++;
  if (points[1].second != points[2].second) ans++;
  if (points[0].first != points[1].first) ans++;
  if (points[1].first != points[2].first) ans++;

  cout << ans << endl;
  return 0;
}
