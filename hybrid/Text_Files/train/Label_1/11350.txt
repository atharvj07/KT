#include <iostream>
#include <set>
using namespace std;
int main() {
    set<string> gemNames = {"Power", "Time", "Space", "Soul", "Reality", "Mind"};
    set<string> presentGems;
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        string color;
        cin >> color;
        if (color == "purple") presentGems.insert("Power");
        else if (color == "green") presentGems.insert("Time");
        else if (color == "blue") presentGems.insert("Space");
        else if (color == "orange") presentGems.insert("Soul");
        else if (color == "red") presentGems.insert("Reality");
        else if (color == "yellow") presentGems.insert("Mind");
    }
    cout << gemNames.size() - presentGems.size() << endl;
    for (string gemName : gemNames) {
        if (presentGems.find(gemName) == presentGems.end()) {
            cout << gemName << endl;
        }
    }
    return 0;
}
