#include <iostream>
#include <utility>

using namespace std;

struct Point {
  int x, y;
};

Point vec(Point a, Point b) {
  return {b.x - a.x, b.y - a.y};
}

int dot(Point a, Point b) {
  return a.x * b.x + a.y * b.y;
}

int cross(Point a, Point b) {
  return a.x * b.y - a.y * b.x;
}

Point p[4], q;

int main() {
  for (int i = 0; i < 4; i++) cin >> p[i].x >> p[i].y;
  q = vec(p[1], p[0]);
  for (int i = 2; i < 4; i++)
    if (dot(q, vec(p[i], p[0])) != 0 || cross(vec(p[i], p[0]), vec(p[i], p[1])) != 0) {
      cout << -1 << endl;
      return 0;
    }
  for (int i = 2; i < 4; i++)
    p[i] = Point{p[1].x + vec(p[0], p[i]).x, p[1].y + vec(p[0], p[i]).y};
  for (int i = 0; i < 4; i++)
    if (i % 2 == 0) swap(p[i].x, p[i].y);
  for (int i = 0; i < 4; i++) cout << p[i].x << ' ' << p[i].y << ' ';
  return 0;
}
