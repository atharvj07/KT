import sys
input = sys.stdin.readline

for i in range(int(input())):
    n, k = map(int, input().split())
    if k > n - 2 or (k == n - 1 and n == 2) or (k == n - 2 and n % 2 == 0):
        print(-1)
        continue
    x = [n - i for i in range(k + 1)]
    for i in range(n - k - 1):
        x.append(i + 1)
    print(*x)
