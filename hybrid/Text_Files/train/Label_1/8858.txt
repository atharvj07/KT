#include <iostream>
#include <vector>

int main() {
  int ans = 0;
  std::vector<std::vector<int>> a(5, std::vector<int>(5));
  for (int i = 0; i < 5; ++i) {
    for (int j = 0; j < 5; ++j) {
      std::cin >> a[i][j];
      ans += a[i][j] == 1 ? std::abs(i - 2) + std::abs(j - 2) : 0;
    }
  }
  std::cout << ans << std::endl;
}
