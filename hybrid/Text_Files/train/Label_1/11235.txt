#include <cstdio>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  scanf("%d", &t);
  for (int cas = 1; cas <= t; cas++) {
    int n;
    scanf("%d", &n);
    vector<int> a(n), b;
    for (int i = 0; i < n; i++) {
      scanf("%d", &a[i]);
    }
    sort(a.begin(), a.end());
    int cnt = 0, lst = -1;
    for (int i = 0; i < n; i++) {
      if (a[i] == lst) cnt++;
      else {
        for (int j = 0; j < cnt; j++) b.push_back(lst);
        lst = a[i];
        cnt = 1;
      }
    }
    for (int j = 0; j < cnt; j++) b.push_back(lst);
    for (int i = 0; i < n; i++) printf("%d ", b[i]);
    printf("\n");
  }
  return 0;
}
