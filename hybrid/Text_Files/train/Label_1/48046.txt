import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        BigInteger p = scanner.nextBigInteger();
        int k = scanner.nextInt();
        if (p.mod(BigInteger.valueOf(k)).equals(BigInteger.ZERO)) {
            System.out.println(-1);
            return;
        }
        List<BigInteger> coefficients = new ArrayList<>();
        for (int i = 0; i <= k * 10; i++) {
            coefficients.add(BigInteger.ZERO);
        }
        coefficients.set(0, p);
        for (int i = 0; i < coefficients.size(); i++) {
            if (coefficients.get(i).equals(BigInteger.ZERO)) {
                continue;
            }
            BigInteger coefficient = coefficients.get(i);
            for (int j = i + 1; j < coefficients.size(); j++) {
                BigInteger oldCoefficient = coefficients.get(j);
                BigInteger newCoefficient = oldCoefficient.subtract(coefficient.multiply(BigInteger.valueOf(k - i)));
                if (newCoefficient.compareTo(BigInteger.ZERO) < 0) {
                    coefficients.set(j, oldCoefficient);
                    break;
                }
                coefficients.set(j, newCoefficient);
            }
        }
        int degree = coefficients.size() - 1;
        while (degree >= 0 && coefficients.get(degree).equals(BigInteger.ZERO)) {
            degree--;
        }
        if (degree < 0) {
            System.out.println(-1);
            return;
        }
        System.out.println(degree + 1);
        for (int i = 0; i < degree; i++) {
            System.out.print(coefficients.get(i) + " ");
        }
        System.out.println(coefficients.get(degree));
    }
}
