
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.BitSet;

class Main {
    public static void main(String[] args) throws IOException {
        InputStreamReader r = new InputStreamReader(System.in);
        BufferedReader buf = new BufferedReader(r);
        String str[] = buf.readLine().split(" ");
        int n = Integer.parseInt(str[0]);
        int m = Integer.parseInt(str[1]);
        BitSet bitSet = new BitSet(n + 1);
        for (int i = 0; i < m; i++) {
            str = buf.readLine().split(" ");
            bitSet.set(Integer.parseInt(str[0]));
            bitSet.set(Integer.parseInt(str[1]));
        }
        int ans = (bitSet.cardinality() * (bitSet.cardinality() - 1) * (bitSet.cardinality() - 2)) / 6;
        ans += (((n - bitSet.cardinality()) * (n - bitSet.cardinality() - 1) * (n - bitSet.cardinality() - 2)) / 6);
        System.out.print(ans);
    }
}
