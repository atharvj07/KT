#include <bits/stdc++.h>
using namespace std;

const int mod = 998244353, inf = 0x3f3f3f3f;

struct edge {
    int to, w;
    edge() {}
    edge(int _to, int _w) : to(_to), w(_w) {}
};

int main() {
    int n, ans = inf;
    cin >> n;
    vector<vector<edge>> g(n);
    vector<array<int, 3>> p(n);
    for (int i = 0; i < n; ++i) {
        cin >> p[i][0] >> p[i][1] >> p[i][2];
    }
    for (int i = 0; i < n - 1; ++i) {
        int a, b;
        cin >> a >> b, --a, --b;
        g[a].emplace_back(b, p[b][0]);
        g[b].emplace_back(a, p[a][0]);
    }
    function<void(int, int, int)> dfs = [&](int u, int par, int c) {
        array<array<int, 2>, 2> dp = {0};
        for (int i = 0; i < 2; ++i) dp[0][i] = p[u][i] != c ? inf : 0;
        for (const auto& e : g[u]) {
            if (e.to != par) {
                dfs(e.to, u, c);
                array<array<int, 2>, 2> ndp = {0};
                for (int i = 0; i < 2; ++i) {
                    for (int j = 0; j < 2; ++j) {
                        ndp[i][j] = min(ndp[i][j], dp[i][j] + e.w + p[e.to][j] != c ? inf : 0);
                    }
                }
                dp = ndp;
            }
        }
        ans = min(ans, dp[1][p[u][2]]);
    };
    if (p[0][2] == p[0][1]) {
        cout << 0 << endl;
        return 0;
    }
    dfs(0, -1, p[0][2]);
    if (ans != inf) cout << ans << endl;
    else
        cout << -1 << endl;
    return 0;
}
