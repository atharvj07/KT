import sys
n=int(sys.stdin.readline())
a=[int(x) for x in sys.stdin.readline().split()]
e=[[] for x in range(n)]
for x in range(n-1):
 e1,e2=map(int,sys.stdin.readline().split())
 e[e1-1].append(e2-1)
 e[e2-1].append(e1-1)
st,ans,x=[0],0,0
st[0]=0
for x in range(n):
 if len(e[st[-1]])==1:st.append(e[st[-1]][0])
 b=[0]*n
 for x in range(n):
  if not b[x]:
   st,ans1,x1=[st[-1]],0,x
   while x1!=st[0]:
    b[x1]=1;ans1+=a[x1];x1=st[0]
    if len(e[x1])==1:st[0]=e[x1][0]
    else:st.pop(0)
   ans=max(ans,ans1)
print(ans)
