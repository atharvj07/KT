#include <iostream>
#include <string>
#include <vector>
#include <map>

using namespace std;

int main() {
  int n;
  cin >> n;

  map<string, vector<string>> phonebook;

  for (int i = 0; i < n; i++) {
    string name;
    cin >> name;

    int num_phones;
    cin >> num_phones;

    for (int j = 0; j < num_phones; j++) {
      string phone_number;
      cin >> phone_number;

      // Check if the number is a suffix of any existing number
      bool is_suffix = false;
      for (string existing_number : phonebook[name]) {
        if (existing_number.size() >= phone_number.size() &&
          existing_number.substr(existing_number.size() - phone_number.size()) == phone_number) {
          is_suffix = true;
          break;
        }
      }

      if (!is_suffix) {
        phonebook[name].push_back(phone_number);
      }
    }
  }

  cout << phonebook.size() << endl;
  for (auto it : phonebook) {
    cout << it.first << " " << it.second.size() << " ";
    for (string phone_number : it.second) {
      cout << phone_number << " ";
    }
    cout << endl;
  }

  return 0;
}
