#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll INF = 1e18;
ll m, ans;
map<ll, ll> cnt;

int main() {
  cin >> m;
  for (int i = 1; i <= m; i++) {
    ll x;
    cin >> x;
    cnt[x]++;
  }
  for (auto &i : cnt) {
    ans += i.second / 2;
    i.second %= 2;
  }
  ll now = 0;
  for (auto &i : cnt) {
    if (i.second == 1) {
      if (now == 0) now = 1;
      else now *= 2;
      ans += now;
    }
  }
  cout << ans << "\n";
  return 0;
}
