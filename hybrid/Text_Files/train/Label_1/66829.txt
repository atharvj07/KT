T = int(input())
for _ in range(T):
    s = input()
    x, y = 0, 0
    min_x, max_x, min_y, max_y = 0, 0, 0, 0
    for c in s:
        if c == 'W':
            y -= 1
        elif c == 'S':
            y += 1
        elif c == 'A':
            x -= 1
        elif c == 'D':
            x += 1
        min_x = min(min_x, x)
        max_x = max(max_x, x)
        min_y = min(min_y, y)
        max_y = max(max_y, y)

    width = max_x - min_x + 1
    height = max_y - min_y + 1
    area = width * height

    if width > 2:
        # Check if removing one column decreases the area
        x, y = 0, 0
        column_counts = {}
        for c in s:
            if c == 'A':
                x -= 1
            elif c == 'D':
                x += 1
            column_counts[x] = column_counts.get(x, 0) + 1
        if max(column_counts.values()) > 1:
            area = min(area, (width - 1) * height)
    
    if height > 2:
        # Check if removing one row decreases the area
        x, y = 0, 0
        row_counts = {}
        for c in s:
            if c == 'W':
                y -= 1
            elif c == 'S':
                y += 1
            row_counts[y] = row_counts.get(y, 0) + 1
        if max(row_counts.values()) > 1:
            area = min(area, width * (height - 1))
    
    print(area)
