N,T=map(int,input().split())
A=[]
for i in range(N):
    a,b=map(int,input().split())
    A.append((a,b,-1))
A.sort()
for i in range(N-1):
    if A[i][1]==A[i+1][1] and A[i][2]==-1:
        A[i][2]=0
        A[i+1][2]=0
for i in range(N):
    if A[i][2]==-1:
        A[i][2]=1
mn=[0]*(T+1)
mn[0]=1
for a,b,c in A:
    for i in range(T-a,-1,-1):
        mn[i+a]=(mn[i+a]+mn[i]*c)%1000000007
print(mn[T])
