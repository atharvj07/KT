#include <iostream>
#include <cstring>
#include <string>
using namespace std;
int len;
bool used[52], vis[52];
char str[52];
int t1, t2;
string ans, st;
void dfs(int deep, int t1, int t2) {
  if (vis[deep]) {
    return;
  }
  vis[deep] = true;
  str[deep] = ans[deep];
  if (ans[deep] == '|') {
    if (t2 != 0) {
      return;
    }
    if (t1 == 0) {
      return;
    }
    if (deep == len - 1) {
      cout << str << '\n';
      exit(0);
    }
    dfs(deep + 1, 0, t1);
    return;
  }
  dfs(deep + 1, t1 + 1, t2);
  if (deep > 0 && ans[deep - 1] != '|') {
    dfs(deep + 1, t1, t2 + 1);
  }
  return;
}
int main() {
  string s, t;
  cin >> s >> t;
  len = s.length();
  for (int i = 0; i < len; i++) {
    if (s[i] != '|') {
      st += s[i];
    }
  }
  for (int i = 0; i < (int)t.length(); i++) {
    if (!used[t[i] - 'A']) {
      st += t[i];
    }
    used[t[i] - 'A'] = true;
  }
  len = st.length();
  for (int i = 0; i < len; i++) {
    for (int j = 0; j < len; j++) {
      if (st[j] >= st[i]) {
        swap(st[j], st[j - 1]);
      }
    }
  }
  for (int i = 0; i < len; i++) {
    if (s[i] == '|') {
      ans += '|';
      t2 = i + 1;
      t1 = i;
    } else {
      if (i < t1) {
        ans += '-';
      } else if (i < t2) {
        ans += '+';
      }
    }
  }
  for (int i = t2; i < len; i++) {
    if (ans[i - 1] != '|') {
      t2 = i;
      t1 = i;
      ans += '|';
    }
  }
  dfs(0, 0, 0);
  cout << "Impossible\n";
  return 0;
}
