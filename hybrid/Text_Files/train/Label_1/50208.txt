import java.io.BufferedReader;
import java.io.InputStreamReader;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] arg = br.readLine().split(" ");
        int n = Integer.parseInt(arg[0]);
        int m = Integer.parseInt(arg[1]);
        int k = Integer.parseInt(arg[2]);
        int q = Integer.parseInt(arg[3]);
        int[][] rook = new int[k][2];
        for (int i = 0; i < k; i++) {
            arg = br.readLine().split(" ");
            rook[i][0] = Integer.parseInt(arg[0]) - 1;
            rook[i][1] = Integer.parseInt(arg[1]) - 1;
        }
        int[][] site = new int[q][4];
        for (int i = 0; i < q; i++) {
            arg = br.readLine().split(" ");
            site[i][0] = Integer.parseInt(arg[0]) - 1;
            site[i][1] = Integer.parseInt(arg[1]) - 1;
            site[i][2] = Integer.parseInt(arg[2]) - 1;
            site[i][3] = Integer.parseInt(arg[3]) - 1;
        }
        for (int j = 0; j < q; j++) {
            int x1 = site[j][0];
            int y1 = site[j][1];
            int x2 = site[j][2];
            int y2 = site[j][3];
            boolean flag = true;
            for (int i = x1; i <= x2; i++) {
                for (int t = y1; t <= y2; t++) {
                    if (i == x1 || i == x2 || t == y1 || t == y2) {
                        boolean f = false;
                        for (int[] p : rook) {
                            if (p[0] == i || p[1] == t) {
                                f = true;
                                break;
                            }
                        }
                        if (!f) {
                            flag = false;
                            break;
                        }
                    }
                }
            }
            System.out.println(flag ? "YES" : "NO");
        }
    }
}
