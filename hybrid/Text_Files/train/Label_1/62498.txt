import sys
readline = sys.stdin.readline

def solve():
    _ = int(readline())
    for _ in range(_):
        s = readline()
        t = readline()
        if len(s) == len(t):
            ans = 'YES' if s == t else 'NO'
        elif len(s) > len(t):
            ans = 'NO'
        else:
            c = s[-1]
            ans = 'NO' if t.count(c) < len(s) else 'YES'
        print(ans)

solve()
