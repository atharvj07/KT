#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n; cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int Alice = 0, Bob = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) continue;
    if(!Alice) {
      Bob += a[i] / 2;
      Alice = a[i] % 2;
    } else {
      if (a[i] % 2) Alice = 0;
    }
  }
  cout << (Alice ? "Alice" : "Bob") << endl;
  return 0;
}
