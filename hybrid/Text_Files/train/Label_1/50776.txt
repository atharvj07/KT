import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int[] f = new int[2001];
        for (int i = 0; i < 2001; i++) {
            if (i % k != 0) {
                f[i] = i;
            } else {
                f[i] = f[i / k];
            }
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = f[a[i]];
        }
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (b[i] + b[j] == 1) {
                    System.out.println("YES");
                    System.out.println(a[i] + " " + a[j]);
                    return;
                }
            }
        }
        System.out.println("NO");
    }
}
