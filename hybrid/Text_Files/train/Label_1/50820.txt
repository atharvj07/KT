import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n;
        while ((n = input.nextInt()) != 0) {
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                String s = input.next();
                if (s.equals("x")) {
                    a[i] = -1;
                } else {
                    a[i] = Integer.parseInt(s);
                }
            }
            boolean unique = true;
            int value = -1;
            for (int i = 1; i < n - 1; i += 2) {
                if (a[i] == -1) {
                    if (a[i - 1] < a[i + 1]) {
                        if (unique) {
                            value = a[i - 1] + 1;
                            unique = false;
                        } else if (value != a[i - 1] + 1) {
                            unique = false;
                        }
                    } else if (a[i - 1] > a[i + 1]) {
                        if (unique) {
                            value = a[i + 1] - 1;
                            unique = false;
                        } else if (value != a[i + 1] - 1) {
                            unique = false;
                        }
                    }
                }
            }
            if (!unique) {
                System.out.println("ambiguous");
            } else if (value == -1) {
                System.out.println("none");
            } else {
                System.out.println(value);
            }
        }
    }
}
