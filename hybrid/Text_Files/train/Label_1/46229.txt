import java.util.Scanner;

public class NikolayAndRooms {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            scanner.nextLine(); // Consume newline
            String stairs = scanner.nextLine();

            int maxRooms = 2 * n - stairs.length(); // Maximum rooms on each floor + rooms connected by stairs

            // Check for additional rooms reachable due to stairs
            for (int j = 0; j < stairs.length(); j++) {
                if (stairs.charAt(j) == '1') {
                    maxRooms++;
                }
            }

            System.out.println(maxRooms);
        }
    }
}
