import java.math.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextInt();
        long m = sc.nextInt();
        long mod = (long) 1e9 + 7;
        long ans = BigInteger.valueOf(2).modPow(BigInteger.valueOf(n * m), BigInteger.valueOf(mod)).longValue();
        System.out.println(ans);
    }
}
