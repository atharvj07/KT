from collections import Counter

n, m = map(int, input().split())
a = list(map(int, input().split()))

c = Counter(a)
ans = 0

for i in c:
    rem = i % m
    ans += max(0, c[rem] - n // m)

print(ans)

b = []
for i in c:
    rem = i % m
    if c[rem] > n // m:
        b.extend([i] * (n // m))
    else:
        b.extend([i] * c[rem])

print(*b)
