import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

public class Main {
    private static final String OPENING_BRACKET = "(";
    private static final String CLOSING_BRACKET = ")";

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String sequence = reader.readLine();
        List<String> correctSequences = getCorrectSequences(sequence);
        System.out.println(correctSequences.stream()
                .min(Comparator.comparingInt(String::length))
                .orElse(""));
    }

    private static List<String> getCorrectSequences(String sequence) {
        List<String> correctSequences = new ArrayList<>();
        int balance = 0;
        StringBuilder builder = new StringBuilder();
        for (char character : sequence.toCharArray()) {
            if (character == OPENING_BRACKET.charAt(0)) {
                balance++;
                builder.append(character);
            } else if (balance > 0) {
                balance--;
                builder.append(character);
            }
        }
        if (balance == 0) {
            correctSequences.add(builder.toString());
            return correctSequences;
        }
        for (int i = 0; i < balance; i++) {
            builder.append(CLOSING_BRACKET);
        }
        StringBuilder cyclicBuilder = new StringBuilder(builder);
        for (int i = 0; i < builder.length(); i++) {
            char firstCharacter = cyclicBuilder.charAt(0);
            cyclicBuilder.deleteCharAt(0);
            cyclicBuilder.append(firstCharacter);
            correctSequences.add(cyclicBuilder.toString());
        }
        return correctSequences;
    }
}
