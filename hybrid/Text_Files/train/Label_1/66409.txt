n = int(input())
a = list(map(int, input().split()))

max_a = max(a)
min_a = min(a)

if max_a == min_a:
    print(0)
else:
    mid = (max_a + min_a) // 2
    if all(x == mid or x == max_a or x == min_a for x in a):
        if (max_a - min_a) % 2 == 0:
            print((max_a - min_a) // 2)
        else:
            print(max_a - min_a)
    else:
        print(-1)
