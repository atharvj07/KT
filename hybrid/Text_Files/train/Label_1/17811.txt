#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1e9+7;
#define all(v) v.begin(),v.end()
#define pb push_back
#define endl "\n"
vector<int>par(401);
unordered_map<string,int>mp;
int findpar(int x){
    if(par[x]==x) return x;
    return par[x]=findpar(par[x]);
}
void unite(int a,int b){
    a=findpar(a);
    b=findpar(b);
    par[b]=a;
}

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL); cout.tie(NULL);
    int n;cin>>n;
    for(int i=1;i<=400;i++) par[i]=i;
    for(int i=1;i<=n;i++){
        string s;cin>>s;
        string t=s;
        for(int j=0;j<s.length();j++){
            if(s[j]=='u') t[j]='o';
            else if(s[j]=='h') t[j]='k';
        }
        if(!mp[s]) mp[s]=i;
        if(!mp[t]) mp[t]=i;
        unite(mp[s],mp[t]);
    }
    set<int>st;
    for(auto x:mp){
        st.insert(findpar(x.second));
    }
    cout<<st.size()<<endl;
    return 0;
}
