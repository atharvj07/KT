from sys import stdin
input = stdin.readline

for _ in range(int(input())):
    n = int(input())
    p = map(int, input().split())
    a, x = map(int, input().split())
    b, y = map(int, input().split())
    k = int(input())
    p = list(sorted(p))
    ans = -1
    tot = 0
    cur = []
    for i in range(n - 1, -1, -1):
        if ans != -1: break
        tot += p[i]
        cur.append(round(p[i] * x / 100))
        if not i % a: cur.append(round(p[i] * y / 100))
        cur.sort(reverse = True)
        if tot >= k:
            tmp = 0
            for c in cur: tmp += c
            if tmp >= k:
                ans = n - i
                break
    print(ans)
