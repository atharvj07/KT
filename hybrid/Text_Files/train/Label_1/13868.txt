#include <iostream>
#include <vector>
#include <map>
#include <set>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<string> abs;
    map<string, int> ys;
    for (int i = 0; i < n; i++) {
        string a;
        cin >> a;
        abs.push_back(a);
        string y = a.substr(a.find("'") + 1);
        if (y.length() <= 2) ys[y]++;
        else ys[y.substr(0, 2)]++;
    }
    map<string, set<int>> y2y;
    for (auto [y, cnt] : ys) {
        if (cnt > 1) {
            int j = 0;
            while (j < y.length() - 1 && ys[y.substr(0, j + 1)] > 1) j++;
            y2y[y.substr(0, j + 1)].insert(cnt);
        }
    }
    for (int i = 0; i < n; i++) {
        string a = abs[i];
        string y = a.substr(a.find("'") + 1);
        if (y2y.count(y)) {
            while (y2y.count(y) && y2y[y].count(ys[y]));
            y2y[y].insert(ys[y]);
            cout << (y.length() > 2 ? y.substr(0, 2) : y) << '0' << y.substr(y.length() > 2) << endl;
        } else {
            cout << (y.length() > 2 ? y.substr(0, 2) : y) << y.substr(y.length() > 2) << endl;
        }
    }
    return 0;
}
