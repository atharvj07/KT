from math import pi, sqrt,acos

def dis(p,q):
    return sqrt((p[0]-q[0])**2+(p[1]-q[1])**2)

def area(r):
    return pi*r**2

x1,y1,r1=map(int,input().split())
x2,y2,r2=map(int,input().split())

d=dis((x1,y1),(x2,y2))

if d>r1+r2:
    print(0)
elif d+r1<=r2 or d+r2<=r1:
    print(area(min(r1,r2)))
else:
    a=(d**2+r1**2-r2**2)/(2*d)
    h=sqrt(r1**2-a**2)
    interArea=area(h)
    interArea+=r2**2*acos((d**2+r2**2-r1**2)/(2*d*r2))
    interArea-=r1**2*acos((d**2+r1**2-r2**2)/(2*d*r1))
    print(interArea)
