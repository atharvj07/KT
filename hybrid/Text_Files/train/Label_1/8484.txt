#include <iostream>
#include <cstdio>
#include <vector>
#define ll long long

using namespace std;

int main() {
    ll n;
    cin >> n;
    vector<ll> v(n), dp(n, 1e12);
    for (int i = 0; i < n; i++)
        cin >> v[i];
    ll cur = v[0];
    dp[0] = cur;
    for (int i = 1; i < n; i++) {
        if (cur + 2 > v[i])
            dp[i] = min(dp[i - 1], cur);
        else
            dp[i] = min(dp[i - 1], cur + 1), cur++;
    }
    for (int i = 0; i < n; i++) {
        v[i] -= dp[i];
        cout << v[i] << " ";
    }
    cout << endl;
}
