import sys
from collections import defaultdict

n = int(input())
p = list(map(int, input().split()))
s = list(map(int, input().split()))
pos = defaultdict(list)
for i in range(n): pos[p[i]].append(i)
ops = []
ans = 0
for i in range(n):
    if p[i] == s[i]: continue
    j = pos[s[i]].pop()
    ans += abs(i - j)
    p[i], p[j] = p[j], p[i]
    ops.append([i + 1, j + 1])
print(ans)
print(len(ops))
for x in ops: print(*x)
