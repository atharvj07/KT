import sys

input = lambda: sys.stdin.readline()

def mex(arr):
    sm = set()
    for v in arr: sm.add(v)
    ans = 0
    while ans in sm: ans += 1
    return ans

n = int(input())
for _ in range(n):
    m = int(input())
    arr = list(map(int,input().split()))
    ans = []
    for i in range(101):
        while arr.count(i):
            ans.append(i)
            arr.pop(arr.index(i))
    print(*ans)
