import java.util.Scanner;

public class VasyaAndRectangles {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        char[][] grid = new char[n][m];

        for (int i = 0; i < n; i++) {
            String row = scanner.next();
            grid[i] = row.toCharArray();
        }

        boolean hasTwoFrames = false;
        int x1 = -1, y1 = -1, x2 = -1, y2 = -1;
        int x3 = -1, y3 = -1, x4 = -1, y4 = -1;

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '#' && (i == 0 || j == 0 || grid[i - 1][j] == '.' || grid[i][j - 1] == '.')) {
                    if (!hasTwoFrames) {
                        x1 = i;
                        y1 = j;
                    } else {
                        x3 = i;
                        y3 = j;
                    }
                    hasTwoFrames = true;
                } else if (grid[i][j] == '#' && hasTwoFrames) {
                    x2 = i;
                    y2 = j;
                } else if (grid[i][j] == '#' && !hasTwoFrames) {
                    x4 = i;
                    y4 = j;
                }
            }
        }

        if (hasTwoFrames && x1 != -1 && y1 != -1 && x2 != -1 && y2 != -1 && x3 != -1 && y3 != -1 && x4 != -1 && y4 != -1) {
            System.out.println("YES");
            System.out.println(x1 + 1 + " " + y1 + 1 + " " + x2 + 1 + " " + y2 + 1);
            System.out.println(x3 + 1 + " " + y3 + 1 + " " + x4 + 1 + " " + y4 + 1);
        } else {
            System.out.println("NO");
        }
    }
}
