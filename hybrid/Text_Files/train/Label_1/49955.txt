import java.util.Scanner;

public class UnimodalArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] array = new int[n];
        for (int i = 0; i < n; i++) {
            array[i] = scanner.nextInt();
        }

        if (isUnimodal(array)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isUnimodal(int[] array) {
        int increasingStart = 0;
        int increasingEnd = -1;
        int decreasingStart = array.length - 1;
        int decreasingEnd = -1;

        for (int i = 0; i < array.length - 1; i++) {
            if (array[i] < array[i + 1]) {
                if (increasingEnd == -1) {
                    increasingStart = i;
                }
                increasingEnd = i + 1;
            } else {
                if (decreasingEnd == -1) {
                    decreasingStart = i + 1;
                }
                decreasingEnd = i + 1;
            }
        }

        if (increasingEnd == -1 || decreasingEnd == -1) {
            return true;
        }

        return increasingEnd == decreasingStart;
    }
}
