#include <bits/stdc++.h>
using namespace std;
#define SZ(x) ((int)x.size())

const int mxN = 1e5;
int sub[mxN];
char vis[mxN];
vector<int> adj[mxN];
vector<int> del;

void dfs(int u, int p) {
  sub[u] = 1;
  vis[u] = 1;
  for (int v : adj[u]) {
    if (v == p) continue;
    if (vis[v]) {
      del.push_back(v);
      continue;
    }
    dfs(v, u);
    sub[u] += sub[v];
  }
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    for (int i = 0; i < n; i++) adj[i].clear();
    for (int i = 0; i < n; i++) cin >> vis[i];
    for (int i = 1; i < n; i++) {
      int u, v;
      cin >> u >> v;
      --u, --v;
      adj[u].push_back(v);
      adj[v].push_back(u);
    }
    del.clear();
    dfs(0, 0);
    int ans = -1;
    for (int x : del) {
      if (sub[x] > k - 1) {
        ans = x;
        break;
      }
    }
    if (ans == -1) cout << "YES" << '\n';
    else
      cout << "NO" << '\n';
  }
  return 0;
}
