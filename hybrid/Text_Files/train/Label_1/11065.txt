#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    if (n == 1) {
        cout << "YES"
             << "\n0";
        return 0;
    }
    for (int i = 0; i < n; i++)
        if (a[i] == 0) {
            for (int j = n - 1; j >= 0; j--)
                if (a[j] == 1) {
                    string ans = "";
                    for (int k = 0; k < i; k++) ans += "1->";
                    ans += "(";
                    for (int k = i; k <= j; k++) ans += to_string(a[k]) + "->";
                    ans = ans.substr(0, ans.size() - 2) + ")";
                    for (int k = j + 1; k < n; k++)
                        if (a[k] == 1 && k + 1 < n && a[k + 1] == 0) ans += "1->0";
                    cout << "YES"
                         << "\n"
                         << ans;
                    return 0;
                }
        }
    cout << "NO";
    return 0;
}
