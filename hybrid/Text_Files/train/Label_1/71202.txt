n = int(input())
x = list(map(int, input().split()))

min_x = min(x)
max_x = max(x)

# Count occurrences of each possible value
counts = {i: x.count(i) for i in range(min_x, max_x+1)}

# Calculate average
average = sum(x) / n

# Generate Anya's measurements
anya_measurements = []

# If all Kirill's measurements are the same, Anya's should be the same
if min_x == max_x:
    anya_measurements = x
# If the difference between max and min is 1, fill with the average rounded to nearest integer
elif max_x - min_x == 1:
    anya_measurements = [round(average)] * n
# If the difference between max and min is 2
else:
    # Strategy: fill with the middle value as much as possible to minimize equal measurements
    middle = min_x + 1
    if counts[min_x] % 2 == 0:
        anya_measurements = [middle] * n
    else:
        anya_measurements = [middle] * (n-1) + [min_x if average < middle else max_x]

print(len(set(anya_measurements) & set(x)))
print(' '.join(map(str, anya_measurements)))
