n = int(input())
grid = [list(input()) for _ in range(n)]
ans = 0
for i in range(n):
    for j in range(n):
        if grid[i][j] == 'C':
            for k in range(n):
                ans += (grid[i][k] == 'C')
            for k in range(n):
                ans += (grid[k][j] == 'C')
            ans -= 1
print(ans//2)
