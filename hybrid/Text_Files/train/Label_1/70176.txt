for t in range(int(input())):
    n = int(input())
    s = input()
    ok = True
    for i in range(n):
        if s[i] != s[n - i - 1]:
            if s[i] + s[n - i - 1] in 'abzyxw':
                ok = ok and (ord(s[i]) < ord(s[n - i - 1]))
            else:
                ok = False
    print(['NO', 'YES'][ok])
