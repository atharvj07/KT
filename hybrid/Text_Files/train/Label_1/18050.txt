#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int p = 1e9 + 7, N = 1e6 + 10;
int n, a, b, c[N], dp[N][3], cost[3][3];
int go(int pos, int prv) {
    if (~dp[pos][prv]) return dp[pos][prv];
    if (pos > n / 2) return dp[pos][prv] = 0;
    dp[pos][prv] = 1e9;
    if (!prv) {
        dp[pos][prv] = min({dp[pos][prv], go(pos + 1, 0) + cost[prv][c[pos]], go(pos + 1, 1) + cost[prv][c[pos]] + a, go(pos + 1, 2) + cost[prv][c[pos]] + b});
    } else {
        dp[pos][prv] = min({dp[pos][prv], go(pos + 1, 1) + cost[prv][c[pos]], go(pos + 1, 2) + cost[prv][c[pos]] + a});
    }
    return dp[pos][prv];
}
int32_t main() {
    cin >> n >> a >> b;
    for (int i = 1; i <= n; ++i) { cin >> c[i]; }
    cost[0][0] = 0;
    cost[1][0] = cost[0][1] = 1e9;
    cost[1][1] = cost[2][2] = 0;
    cost[1][2] = cost[2][1] = a;
    cost[2][0] = cost[0][2] = b;
    memset(dp, -1, sizeof dp);
    int ans = go(1, 0);
    if (ans == 1e9) cout << -1 << '\n';
    else cout << ans << '\n';
    return 0;
}
