import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int min = 1, max = n;
            for (int j = 0; j < 2 * (int) (Math.ceil(Math.log(n) / Math.log(3))); j++) {
                if (min == max) {
                    break;
                }
                int mid = (min + max) / 2;
                System.out.println("? " + min + " " + mid);
                System.out.flush();
                String ans = sc.nextLine();
                if (ans.equals("<")) {
                    max = mid - 1;
                } else if (ans.equals("=")) {
                    max = mid;
                }
                System.out.println("? " + mid + " " + max);
                System.out.flush();
                ans = sc.nextLine();
                if (ans.equals(">")) {
                    min = mid + 1;
                } else if (ans.equals("=")) {
                    min = mid;
                }
            }
            System.out.println("! " + min + " " + max);
            System.out.flush();
        }
        sc.close();
    }
}
