import java.util.*;
import java.io.*;

class Main {
  static final long INF = 1L << 60;
  static List<List<Query>> graph;
  static long[] dist;
  static int[] prev;

  static class Query implements Comparable<Query> {
    int to;
    long time;
    Query(int to, long time) {
      this.to = to;
      this.time = time;
    }
    @Override
    public int compareTo(Query other) {
      return Long.compare(this.time, other.time);
    }
  }

  static void relax(int from, int to, long time) {
    if (dist[to] > dist[from] + time) {
      dist[to] = dist[from] + time;
      prev[to] = from;
    }
  }

  static void dijkstra(int n, int start, int end) {
    dist = new long[n + 1];
    prev = new int[n + 1];
    for (int i = 0; i < dist.length; i++) {
      dist[i] = INF;
    }
    dist[start] = 0;
    PriorityQueue<Query> queue = new PriorityQueue<>();
    queue.add(new Query(start, 0));
    while (!queue.isEmpty()) {
      Query current = queue.poll();
      if (dist[current.to] < current.time) {
        continue;
      }
      for (Query next : graph.get(current.to)) {
        relax(current.to, next.to, next.time);
        queue.add(new Query(next.to, dist[next.to]));
      }
    }
  }

  static List<Integer> getRoute(int start, int end) {
    List<Integer> route = new ArrayList<>();
    while (end != 0) {
      route.add(end);
      end = prev[end];
    }
    Collections.reverse(route);
    return route;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    String[] line = in.readLine().split(" ");
    int n = Integer.parseInt(line[0]);
    int m = Integer.parseInt(line[1]);
    long t = Long.parseLong(line[2]);
    graph = new ArrayList<>();
    for (int i = 0; i <= n; i++) {
      graph.add(new ArrayList<>());
    }
    for (int i = 0; i < m; i++) {
      String[] road = in.readLine().split(" ");
      int from = Integer.parseInt(road[0]);
      int to = Integer.parseInt(road[1]);
      long time = Long.parseLong(road[2]);
      graph.get(from).add(new Query(to, time));
    }
    dijkstra(n, 1, n);
    if (dist[n] <= t) {
      List<Integer> route = getRoute(1, n);
      System.out.println(route.size());
      for (Integer node : route) {
        System.out.print(node + " ");
      }
    }
    else {
      System.out.println(-1);
    }
  }
}
