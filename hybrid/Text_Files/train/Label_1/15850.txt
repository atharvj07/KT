#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s >> s;
    int m;
    cin >> m;
    if (s[9] == '0' && m > 1) {
        cout << "NO" << endl;
    } else if (m == 2 && s[9] == '0' && s[7] == '0' && s[5] == '0') {
        cout << "NO" << endl;
    } else if (s[9] == '0' && s[7] == '0' && s[5] == '0') {
        cout << "NO" << endl;
    } else if (s[9] == '0' && s[7] == '0' && s[5] == '0' && s[3] == '0') {
        cout << "NO" << endl;
    } else if (s[9] == '0' && s[7] == '0' && s[5] == '0' && s[3] == '0' && s[1] == '0') {
        cout << "NO" << endl;
    } else if (s[9] == '0' && s[7] == '0' && s[5] == '0' && s[3] == '0' && s[1] == '0' && s[0] == '0') {
        cout << "NO" << endl;
    } else {
        cout << "YES" << endl;
        for (int i = 9; i >= 0; i--) {
            if (s[i] == '1') {
                while (i > 0) {
                    cout << i + 1 << " ";
                    i -= 2;
                }
                if (i == 0) cout << 1 << " ";
            }
        }
    }
}
