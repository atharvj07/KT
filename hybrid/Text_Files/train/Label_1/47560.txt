import java.util.*;
class Main {
    static final long mod = 1_000_000_007;
    static long[] f = new long[(int)2e6 + 10];
    static long[] invf = new long[(int)2e6 + 10];
    static long pow(long x, long p) {
        x %= mod;
        if (p == 0) return 1;
        if (p == 1) return x;
        return pow(x * x % mod, p / 2) * (p % 2 == 0 ? 1 : x % mod);
    }
    static long inv(long v) {
        return pow(v % mod, mod - 2);
    }
    static long ncr(int n, int r) {
        if (r > n) return 0;
        if (r == n) return 1;
        return f[n] * inv(f[n - r]) % mod * inv(f[r]) % mod;
    }
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        f[0] = invf[0] = 1;
        for (int i = 1; i < f.length; i++) {
            f[i] = i * f[i - 1];
            f[i] %= mod;
            invf[i] = inv(f[i]);
        }
        if (n % 2 == 1) {
            System.out.println(ncr(n - 1, (n - 1) / 2));
        } else {
            System.out.println((ncr(n - 1, n / 2) + ncr(n - 1, n / 2 - 1)) % mod);
        }
    }
}
