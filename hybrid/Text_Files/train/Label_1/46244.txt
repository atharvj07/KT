import java.util.Scanner;

public class BullsAndCows {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        int[][] guesses = new int[n][2];
        int[][] digits = new int[10][2];

        for (int i = 0; i < n; i++) {
            String[] line = scanner.nextLine().split(" ");
            int guess = Integer.parseInt(line[0]);
            int bulls = Integer.parseInt(line[1]);
            int cows = Integer.parseInt(line[2]);
            guesses[i] = new int[]{guess, bulls + cows};

            for (int j = 0; j < 4; j++) {
                int digit = guess / (int) Math.pow(10, 3 - j) % 10;
                digits[digit][0]++;
                if (j < bulls) {
                    digits[digit][1]++;
                }
            }
        }

        int[] sought = new int[4];
        boolean valid = true;

        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 10; j++) {
                if (digits[j][0] == guesses[i][1] && digits[j][1] == i) {
                    sought[i] = j;
                    break;
                }
            }
            if (sought[i] == 0) {
                valid = false;
                break;
            }
        }

        if (valid) {
            for (int digit : sought) {
                System.out.print(digit);
            }
        } else {
            System.out.println("Need more data");
        }
    }
}
