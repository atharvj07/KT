import java.util.Scanner;

public class UniqueSubstring {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        char[] s = new char[n];

        for (int i = 0; i < n; i++) {
            s[i] = (i % k == 0) ? '1' : '0';
        }

        System.out.println(new String(s));
    }
}
