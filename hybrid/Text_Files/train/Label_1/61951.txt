def count_triangles(A, B, C, D):
    count = 0
    for z in range(C, D+1):
        x_min = max(A, z-C+1)
        x_max = min(B, z-1)
        if x_max >= x_min:
            count += (x_max - x_min + 1) * (x_max - x_min + 2) // 2
    return count

A, B, C, D = map(int, input().split())
print(count_triangles(A, B, C, D))
