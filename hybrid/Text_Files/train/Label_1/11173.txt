#include <bits/stdc++.h>
using namespace std;
#define int long long
typedef pair<int,int> pii;
const int maxN = 2e5+5;
int bridge[maxN];
vector<int> a[maxN];
int32_t main() {
    int n,m;
    cin >> n >> m;
    for(int i = 1; i <= n; i++) {
        int l,r;
        cin >> l >> r;
        a[r-1].emplace_back(l);
    }
    priority_queue<pii> pq;
    for(int i = 0; i <= 1e18; i++) {
        for(int x : a[i]) {
            pq.push({x,i+1});
        }
        if(pq.empty()) {
            cout << "No" << endl;
            return 0;
        }
        int l = pq.top().first;
        pq.pop();
        if(l+1 != i) {
            cout << "No" << endl;
            return 0;
        }
        else {
            bridge[i] = pq.top().second;
            pq.pop();
        }
    }
    cout << "Yes" << endl;
    for(int i = 1; i <= 1e18; i++) {
        if(bridge[i]) {
            cout << bridge[i] << " ";
        }
    }
    cout << endl;
    return 0;
}
