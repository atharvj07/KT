import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    xs, ys, us, vs = [[] for _ in range(4)]
    for _ in range(n):
        x, y, u, v = map(int, input().split())
        xs.append(x)
        ys.append(y)
        us.append(u)
        vs.append(v)
    ans = 0
    for i in range(n):
        a = (us[i]-xs[i])/(ys[i]-vs[i])
        for j in range(i+1, n):
            b = (us[j]-xs[j])/(ys[j]-vs[j])
            if a == b or a*b == -1:
                ans += 1
    print(ans)
