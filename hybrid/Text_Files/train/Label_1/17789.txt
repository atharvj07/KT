#include <bits/stdc++.h>
using namespace std;

const int mod=1e9+7;
typedef long long ll;

int n;
vector <int> adj[200005],val(200005);
vector <bool> vis(200005);

int mn=INT_MAX,mx=INT_MIN;

void dfs(int nd,int p,int x){
	vis[nd]=true;
	if(adj[nd].size()==1 && nd!=1){
		if(x==0)return;
		val[nd]=x;
		mn=min(mn,x);
		mx=max(mx,x);
	}
	for(auto nbr:adj[nd]){
		if(nbr==p)continue;
		if(vis[nbr]==false){
			dfs(nbr,nd,x+1);
		}
	}
}

int main(){
	cin>>n;
	for(int i=1;i<n;i++){
		int a,b;
		cin>>a>>b;
		adj[a].push_back(b);
		adj[b].push_back(a);
	}
	dfs(1,1,0);
	cout<<mx<<" "<<mn<<endl;
	return 0;
}
