typedef long long ll;
const long long mod=1e9+7;
const int mxN=1e6+5;
vector<bool>is_prime(mxN,1);
vector<ll>prime;
void seive(){
    is_prime[0]=is_prime[1]=0;
    for(ll i=2;i*i<mxN;i++){
        if(is_prime[i]){
            for(ll j=i*i;j<mxN;j+=i){
                is_prime[j]=0;
            }
        }
    }
     for(ll i=2;i<mxN;i++){
        if(is_prime[i])prime.push_back(i);
    }
}
ll gcd(ll a,ll b){
    if(b==0)return a;
    return gcd(b,a%b);
}
ll power(ll base, ll p) {
    if (p == 0) return 1LL;
    if (p == 1) return base;
    ll ans = power(base, p / 2);
    ans = (ans * ans) % mod;
    if ((p & 1) == 1) ans = (ans * base) % mod;
    return ans % mod;
}
ll mulmod(ll a, ll b, ll mod){
    a = a % mod;
    return (((a * b) % mod) + mod) % mod;
}
ll addmod(ll a, ll b, ll mod){
    a = a % mod;
    b = b % mod;
    return (((a + b) % mod) + mod) % mod;
}
