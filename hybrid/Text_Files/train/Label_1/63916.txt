import sys
n=int(input())
g=[[]for i in range(n+1)]
for i in range(n-1):
 a,b=map(int,input().split())
 g[a].append(b)
 g[b].append(a)
o=list(map(int,input().split()))
q=[1]
vis=[1]*n+[-1]
x=1
while q:
 if q[0]==o[x-1]and x<=n:
  x+=1
  y=q[0]
  vis[y-1]=y
  q=q[1:]+g[y]
 else:
  print('No')
  sys.exit()
print('Yes')
