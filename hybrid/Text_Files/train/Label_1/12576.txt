#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k;
    cin >> n >> k;
    vector<long long> arr(n);
    for (int i = 0; i < n; i++) { cin >> arr[i]; }
    long long ans = 0;
    for (int i = 0; i < n; i++) {
      if (arr[i] % k != 0) {
        long long dif = k - arr[i] % k;
        ans += dif;
        arr[i] += dif;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
