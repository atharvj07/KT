import sys

mod = 998244353
input = sys.stdin.readline

n, k = map(int, input().split())
p = list(map(int, input().split()))

dp = [[[-1] * 3 for _ in range(k+1)] for _ in range(n+1)]
def solve(idx, seg, mx):
    if idx==n:
        if seg==k: return 1
        else: return 0
    if dp[idx][seg][mx]!=-1: return dp[idx][seg][mx]
    ret = 0
    if mx == 0:
        ret = (ret + solve(idx+1, seg+1, p[idx]%mod)) % mod
    else:
        if p[idx] < mx:
            ret = (ret + solve(idx+1, seg, mx)) % mod
        else:
            ret = (ret + solve(idx+1, seg, p[idx]%mod)) % mod
            ret = (ret + solve(idx+1, seg+1, p[idx]%mod)) % mod

    dp[idx][seg][mx] = ret
    return ret

print((solve(0, 0, 0)), (solve(0, 0, 0)%mod))
