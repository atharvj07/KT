import java.util.*;

class Main {
    static class Solution {
        public char[][] grid;
        public char[][] pattern;
        public int[][] appearance;
        public int n, m, r, c;

        public void input() {
            Scanner in = new Scanner(System.in);
            n = in.nextInt();
            m = in.nextInt();
            grid = new char[n][m];
            in.nextLine();
            for (int i = 0; i < n; i++) {
                grid[i] = in.nextLine().toCharArray();
            }
            r = in.nextInt();
            c = in.nextInt();
            pattern = new char[r][c];
            in.nextLine();
            for (int i = 0; i < r; i++) {
                pattern[i] = in.nextLine().toCharArray();
            }
            in.close();
        }

        public boolean isAppearance(int x, int y) {
            for (int i = 0; i < r; i++) {
                for (int j = 0; j < c; j++) {
                    if (pattern[i][j] == '?' || pattern[i][j] == grid[(x + i) % n][(y + j) % m]) {
                        continue;
                    }
                    return false;
                }
            }
            return true;
        }

        public void printAppearance() {
            appearance = new int[n][m];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (isAppearance(i, j)) {
                        appearance[i][j] = 1;
                    }
                }
            }
            for (int[] row : appearance) {
                for (int v : row) {
                    System.out.print(v);
                }
                System.out.println();
            }
        }

        public void solve() {
            input();
            printAppearance();
        }
    }

    public static void main(String[] args) {
        new Solution().solve();
    }
}
