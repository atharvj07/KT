from array import array

t = int(input())
for _ in range(t):
    n, u, v = map(int, input().split())
    a = array('i', map(int, input().split()))
    p = []
    for i in range(n - 1, -1, -1):
        j = a[i] - 1
        p.append(j)
        if i < n - 1:
            if p[-1] != p[-2]:
                u += min(u, v)
    print(u)
