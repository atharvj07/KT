#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main(){
    int a;
    cin >> a;
    vector<int> ans={a%10, (a/10)%10, (a/100)%10, (a/1000)%10, (a/10000)%10, (a/100000)%10};
    int cnt=0;
    for(auto x:ans){
        cnt+=x==8;
    }
    if(cnt>0) cout<<1<<endl;
    else cout<<10<<endl;
    return 0;
}
