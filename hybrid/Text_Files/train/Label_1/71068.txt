MOD = 1000000007

def calculate_sum(l, r):
    def sum_of_ap(first_term, common_diff, n):
        return n * (2 * first_term + (n - 1) * common_diff) // 2

    def find_sum(l, r):
        even_sum, odd_sum = 0, 0
        even_count, odd_count = 0, 0
        stage, num = 1, 1
        while num < l:
            if stage % 2 == 0:
                num += 2**(stage-1)
                even_count += 2**(stage-1)
            else:
                num += 2**(stage-1)
                odd_count += 2**(stage-1)
            stage += 1
        num -= 2**(stage-1)
        if stage % 2 == 0:
            even_count -= 2**(stage-1)
        else:
            odd_count -= 2**(stage-1)
        remaining = r - l + 1
        while remaining > 0:
            if stage % 2 == 0:
                current_count = min(2**(stage-1), remaining)
                even_sum += sum_of_ap(even_count*2 + 2, 2, current_count)
                even_count += current_count
                remaining -= current_count
            else:
                current_count = min(2**(stage-1), remaining)
                odd_sum += sum_of_ap(odd_count*2 + 1, 2, current_count)
                odd_count += current_count
                remaining -= current_count
            stage += 1
        return (even_sum + odd_sum) % MOD

    return find_sum(l, r)

l, r = map(int, input().split())
print(calculate_sum(l, r))
