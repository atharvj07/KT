#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<long long> c(n);
    for (int i = 0; i < n; i++) { std::cin >> c[i]; }
    long long res = 0;
    long long l = 0, r = 0;
    for (int i = 0; i < n; i++) {
        if (c[i] % 2 == 0) r += c[i];
        else l += c[i];
        if (l == r) res += r + 1;
        else if (r > l) l = r = c[i];
    }
    std::cout << res << std::endl;
    return 0;
}
