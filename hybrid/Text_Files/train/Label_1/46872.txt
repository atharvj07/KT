import java.util.Scanner;
import java.util.TreeMap;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        TreeMap<Integer, Integer> populationMap = new TreeMap<>();
        for (int i = 0; i < n; i++) {
            int birth = in.nextInt();
            int death = in.nextInt();
            populationMap.put(birth, populationMap.getOrDefault(birth, 0) + 1);
            populationMap.put(death, populationMap.getOrDefault(death, 0) - 1);
        }
        int maxPopulation = 0;
        int yearOfMaxPopulation = 0;
        int currentPopulation = 0;
        for (int year : populationMap.keySet()) {
            currentPopulation += populationMap.get(year);
            if (currentPopulation > maxPopulation) {
                maxPopulation = currentPopulation;
                yearOfMaxPopulation = year;
            }
        }
        System.out.println(yearOfMaxPopulation + " " + maxPopulation);
    }
}
