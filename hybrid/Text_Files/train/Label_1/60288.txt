import sys

rd = sys.stdin.readline

t = int(rd())
for _ in range(t):
    n, k = map(int, rd().split())
    a, b = rd().strip(), rd().strip()
    ans = 'YES' if a == b else 'NO'
    can = True
    for i in range(n - 1):
        if a[i] == 'z' and b[i] != 'z':
            can = False
            break
        if a[i] > b[i]:
            can = False
            break
        if a[i] != b[i] and a[i + 1] != b[i + 1] and (b[i] != 'z' or b[i + 1] != 'z'):
            can = False
            break
    if not can or a[n - 1] != b[n - 1]:
        ans = 'NO'
    print(ans)
