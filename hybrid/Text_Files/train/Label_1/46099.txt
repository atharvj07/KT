import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        int[] b = new int[m];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 0; i < m; i++) {
            b[i] = scanner.nextInt();
        }
        int cnt = 0;
        for (int i = 1; i <= k; i++) {
            if (a[(i - 1) % n] != b[(i - 1) % m]) {
                cnt++;
            }
            if (cnt == k) {
                System.out.println(i);
                break;
            }
        }
    }
}
