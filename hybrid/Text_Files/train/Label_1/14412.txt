#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int base=1e7;
int p[9]={1,10,100,1000,10000,100000,1000000,10000000,100000000};
int pow_mod(int a,int x,int p){
    int ans=1;
    while(x){
        if(x&1) ans=1ll*ans*a%p;
        a=1ll*a*a%p;
        x>>=1;
    }
    return ans;
}
ll get(ll k){
    int l=0,r=8,ans=-1,num=0;
    while(l<=r){
        int mid=(l+r)>>1;
        if((ll)p[mid]*2-1>=k){
            ans=mid;
            r=mid-1;
        }else{
            l=mid+1;
        }
    }
    num=k/(2*ans+1);
    ll tmp=k%(2*ans+1);
    if(tmp==0) tmp=(2*ans+1);
    if(tmp<=ans+1){
        int res=pow_mod(num,tmp-1,base);
        return (tmp+9*res)%10;
    }else{
        return (tmp+ans+9*(num-1))%10;
    }
}
main(){
    int q; cin>>q;
    while(q--){
        ll k; cin>>k;
        cout<<get(k)<<endl;
    }
}
