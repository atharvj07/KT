import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        input.nextLine();
        while (t-- > 0) {
            String x = input.nextLine();
            String y = input.nextLine();
            int n = x.length(), m = y.length();
            int k = 0;
            boolean same = true;
            for (int i = 0; i < m && same; i++) {
                if (x.charAt(n - 1 - i) - '0' > y.charAt(m - 1 - i) - '0') {
                    same = false;
                } else if (x.charAt(n - 1 - i) - '0' < y.charAt(m - 1 - i) - '0') {
                    k = i + 1;
                    same = true;
                }
            }
            if (same) {
                for (int i = m; i < n && same; i++) {
                    if (x.charAt(n - 1 - i) == '1') {
                        k = i + 1;
                    } else {
                        same = false;
                    }
                }
            }
            System.out.println(k);
        }
    }
}
