#include <iostream>
#include <vector>
#include <set>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    set<int> s;
    for (auto &x : v) {
        if (!s.count(x.second)) {
            s.insert(x.first);
        }
    }
    cout << s.size() << endl;
    return 0;
}
