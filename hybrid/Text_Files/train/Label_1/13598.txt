#include <bits/stdc++.h>

using namespace std;

#define ll long long
#define rep(i, n) for (int i = 0; i < n; ++i)
#define rev(i, n) for (int i = n - 1; i >= 0; --i)

ll fastpow(ll a, ll n) {
    if (n == 0) return 1ll;
    else if (n == 1ll) return a;
    ll hr = fastpow(a, n / 2ll);
    if (n % 2ll == 1ll) return hr * hr * a;
    else return hr * hr;
}

int main() {
    ll x, y, l, r;
    cin >> x >> y >> l >> r;
    ll maxp = 0ll, minp = LONG_LONG_MAX;
    for (ll a = 0ll; fastpow(x, a) <= r; ++a) {
        for (ll b = 0ll; fastpow(y, b) <= r; ++b) {
            ll ans = fastpow(x, a) + fastpow(y, b);
            if (ans >= l && ans <= r) {
                maxp = max(maxp, ans);
                minp = min(minp, ans);
            }
        }
    }
    if (maxp == 0ll && minp == LONG_LONG_MAX) cout << 0;
    else cout << maxp - minp + 1ll;
    return 0;
}
