import java.io.PrintWriter;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        int d = in.nextInt(), s = in.nextInt();
        int n = 0;
        while (s >= 10) {
            if ((s % 9) <= d) {
                n *= 10;
                n += s % 9;
                s -= s % 9;
            } else {
                n *= 10;
                n += d;
                s -= d;
            }
        }
        if (s != 0 && s <= d) {
            n *= 10;
            n += s;
        } else {
            n = -1;
        }
        out.println(n);
        out.close();
    }
}
