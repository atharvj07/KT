import sys
input = sys.stdin.readline

n = int(input())
points = [tuple(map(int, input().split())) for _ in range(n)]
removed = [False] * n
for a in range(n):
    for b in range(n):
        if a == b or removed[a] or removed[b]: continue
        flag = True
        x1, y1, z1 = points[a]
        x2, y2, z2 = points[b]
        for c in range(n):
            if c != a and c != b and not removed[c]:
                x3, y3, z3 = points[c]
                if min(x1, x2) <= x3 <= max(x1, x2) and min(y1, y2) <= y3 <= max(y1, y2) and min(z1, z2) <= z3 <= max(z1, z2):
                    flag = False
                    break
        if flag:
            print(a+1, b+1)
            removed[a] = True
            removed[b] = True
            break
