#include<iostream>
#include<cstdio>
#include<cstdlib>
#include<cmath>
#include<cstring>
#include<algorithm>
#include<climits>
#include<vector>
#include<queue>
#include<deque>
#include<set>
#include<map>
#include<stack>
using namespace std;
typedef long long ll;
const int maxn = 5000+10;
char s[maxn];
int dp[maxn][maxn];
ll cal[maxn];
int main()
{
    memset(dp,-1,sizeof(dp));
    int n;
    int ans,mid,l,r;
    scanf(" %s ",s+1);
    n = strlen(s+1);
    for(int i=1;i<=n;i++)
    {
        cal[i] = s[i]-'0';
        if(i&1) cal[i]+=cal[i-1]*10;
    }
    for(int i=1;i<=n;i++)
    {
        if(i&1)
        {
            dp[i][i] = cal[i];
        }
        else
        {
            if(s[i]=='+')
            {
                dp[i-1][i] = cal[i-1]+cal[i];
            }
            if(s[i]=='*')
            {
                dp[i-1][i] = cal[i-1]*cal[i];
            }
        }
    }
    for(int len=3;len<=n;len++)
    {
        for(l=1;l+len-1<=n;l++)
        {
            r = l+len-1;
            for(mid=l+1;mid<r;mid++)
            {
                if(s[mid]=='+')
                {
                    dp[l][r] = max(dp[l][r],dp[l][mid]+dp[mid+1][r]);
                }
                else
                {
                    dp[l][r] = max(dp[l][r],dp[l][mid]*dp[mid+1][r]);
                }
            }
        }
    }
    ans = dp[1][n];
    printf("%d\n",ans);
    return 0;
}
