#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n; cin>>n;
    long long arr[n];
    for(int i=0; i<n; i++) cin>>arr[i];
    sort(arr, arr+n);
    int count = 1;
    int k = 1;
    bool f = true;
    long long ans = 1;
    for(int i=1; i<n; i++) {
        ans = (arr[i]-arr[i-1]);
        if(ans<=0) {
            f = false;
            break;
        }
        if(ans==k) count++;
        else k++;
    }
    if(count==1) {
        for(int i=1; i<=1000000000; i++) {
            int tmp = i/count;
            long long aa = tmp*(count-1) + 1;
            int bb = arr[0]-aa + 1;
            if(aa==arr[0] && bb==arr[count-1] && tmp>=1 && bb>=1) {
                f = true;
                ans = tmp;
                k = bb;
                break;
            }
        }
    }
    if(f) cout<<"YES"<<endl<<count<<" "<<k<<endl;
    else cout<<"NO"<<endl;
    return 0;
}
