t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    taps = list(map(int, input().split()))
    max_distance = 0
    for i in range(1, n + 1):
        distance = min([abs(i - tap) for tap in taps]) + 1
        max_distance = max(max_distance, distance)
    print(max_distance)
