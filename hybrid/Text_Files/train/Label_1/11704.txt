#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, k, d;
    std::cin >> n >> k >> d;
    std::vector<int> v(n);
    for (int& x : v) std::cin >> x;
    std::sort(v.begin(), v.end());
    int ans = 0, j = 0;
    for (int i = 0; i < n; i++) {
      if (j < i && v[j] + d - 1 < v[i]) j = i;
      ans = std::max(ans, v[i] - v[j] + 1);
    }
    std::cout << ans << std::endl;
  }
  return 0;
}
