#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k;cin>>n>>k;
    unordered_map<int,int> occ;
    vector<int> ans;
    for(int i=1;i<=n;++i){
        int x;cin>>x;
        if(occ[x]==0){++occ[x];ans.push_back(i);}
        if((int)ans.size()==k){
            cout<<"YES\n";
            for(auto c:ans) cout<<c<<" ";
            return 0;
        }
    }
    cout<<"NO\n";
    return 0;
}
