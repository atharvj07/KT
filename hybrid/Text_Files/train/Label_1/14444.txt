#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int t, s, n;
  cin >> t;
  while (t--) {
    cin >> s >> n;
    vector<int> arr(n, 0);
    int rem = s % n;
    for (int i = n - 1; i >= 0; i--) {
      if (i > 0 || rem == 0)
        arr[i] = 10;
      else
        arr[i] = rem;
      rem -= arr[i];
    }
    for (int i = 0; i < n; i++) cout << arr[i] << " ";
    cout << "\n";
  }
  return 0;
}
