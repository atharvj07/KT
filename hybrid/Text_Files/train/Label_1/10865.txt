#include <bits/stdc++.h>
using namespace std;
int main() {
    int q;
    cin >> q;
    while (q--) {
        string s, t;
        cin >> s >> t;
        int sp = 0, tp = 0, ls = s.length(), lt = t.length();
        while (sp < ls && tp < lt)
            if (s[sp] == t[tp])
                sp++, tp++;
            else
                sp = 0, tp = 0;
        if (tp < lt)
            cout << -1 << endl;
        else {
            int gc = __gcd(ls, lt), gcs = ls / gc, gct = lt / gc;
            int lcs = min(ls, lt);
            string ans = "";
            for (int i = 0; i < gcs; i++) ans += s;
            for (int i = 0; i < gct; i++) ans += t;
            if (ans.length() == lcs)
                cout << ans << endl;
            else
                cout << -1 << endl;
        }
    }
}
