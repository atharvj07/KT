#include <bits/stdc++.h>
#define int long long
using namespace std;
const int MOD = 1000000007;
signed main() {
  int n, m;
  cin >> n >> m;
  unordered_map<int, int> mp;
  for (int i = 0; i < m; i++) {
    int x;
    cin >> x;
    mp[x]++;
  }
  vector<long long> dp(n + 2, 0);
  dp[0] = dp[1] = 1;
  for (int i = 2; i <= n; i++) {
    for (int j = 0; j < i; j++) {
      dp[i] += ((dp[j] * dp[i - j - 1]) % MOD);
      dp[i] %= MOD;
    }
  }
  long long ans = 0;
  for (auto &x : mp) {
    ans += ((dp[x.second] * dp[n - x.second]) % MOD);
    ans %= MOD;
  }
  cout << ans << '\n';
  return 0;
}
