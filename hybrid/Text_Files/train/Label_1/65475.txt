n = int(input())
a = list(map(int, input().split()))

def can_complete_wall(n, a):
    max_height = max(a)
    for i in range(n - 1):
        if a[i] == a[i + 1]:
            a[i] += 1
            a[i + 1] += 1
    return all(height == max_height for height in a)

print("YES" if can_complete_wall(n, a) else "NO")
