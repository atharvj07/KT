import java.util.Scanner;

public class Main {
    public static int findOccurrences(String t, String s) {
        int occurrences = 0;
        int i = 0, j = 0;
        while (i < s.length() && j < t.length()) {
            if (s.charAt(i) == t.charAt(j)) {
                i++;
                j++;
            } else {
                i = 0;
                if (s.charAt(i) == t.charAt(j)) {
                    i++;
                    j++;
                } else {
                    j++;
                }
            }
            if (i == s.length()) {
                occurrences++;
            }
        }
        return occurrences;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        String t = "";
        for (int i = 0; i < n; i++) {
            String part = scanner.next();
            int length = Integer.parseInt(part.substring(0, part.indexOf('-')));
            char letter = part.charAt(part.indexOf('-') + 1);
            for (int j = 0; j < length; j++) {
                t += letter;
            }
        }
        String s = "";
        for (int i = 0; i < m; i++) {
            String part = scanner.next();
            int length = Integer.parseInt(part.substring(0, part.indexOf('-')));
            char letter = part.charAt(part.indexOf('-') + 1);
            for (int j = 0; j < length; j++) {
                s += letter;
            }
        }
        int occurrences = findOccurrences(t, s);
        System.out.println(occurrences);
    }
}
