import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        long b = in.nextLong();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        int max = n - 1;
        if (a.length > 1) {
            max = 0;
        }
        for (int i = n - 1; i >= 0; i--) {
            if (b >= a[i]) {
                b -= a[i];
                k--;
                max = i;
            }
            if (k == 0) {
                break;
            }
        }
        System.out.println(n - max);
    }
}
