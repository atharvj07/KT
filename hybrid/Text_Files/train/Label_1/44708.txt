import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long sum = (n + 2) * n / 2;
        long x = (long) Math.sqrt(1 + 8 * sum) - 1;
        x /= 2;
        long ans = x * (x + 1) / 2;
        System.out.println(n - ans + 1);
    }
}
