import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] A = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextInt();
        }
        int Q = sc.nextInt();
        int[][] BC = new int[Q][2];
        for (int i = 0; i < Q; i++) {
            BC[i][0] = sc.nextInt();
            BC[i][1] = sc.nextInt();
        }
        long[] S = new long[Q];
        for (int i = 0; i < N; i++) {
            S[0] += A[i];
        }
        for (int i = 1; i < Q; i++) {
            S[i] = S[i - 1];
            for (int j = 0; j < N; j++) {
                if (A[j] == BC[i - 1][0]) {
                    S[i] += BC[i][1] - BC[i - 1][0];
                    A[j] = BC[i][1];
                }
            }
        }
        for (long s : S) {
            System.out.println(s);
        }
    }
}
