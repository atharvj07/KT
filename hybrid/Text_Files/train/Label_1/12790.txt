#include <bits/stdc++.h>
using namespace std;
const int MAXN = 501, MAXK = 501;
int dp[MAXN][MAXK], n, k, coins[MAXN];

int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> coins[i];
  dp[0][0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= k; j++) {
      dp[i][j] = dp[i - 1][j];
      if (j >= coins[i]) dp[i][j] |= dp[i - 1][j - coins[i]];
    }
  }
  vector<int> ans;
  for (int j = 0; j <= k; j++) if (dp[n][j]) ans.push_back(j);
  cout << ans.size() << '\n';
  for (int x : ans) cout << x << ' ';
  cout << '\n';
}
