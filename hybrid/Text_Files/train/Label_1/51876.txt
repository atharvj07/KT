import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }

            long[] dp = new long[n];
            dp[n - 1] = a[n - 1];
            for (int i = n - 2; i >= 0; i--) {
                dp[i] = Math.max(a[i] + dp[(i + a[i]) % n], dp[i + 1]);
            }

            System.out.println(dp[0]);
        }
    }
}
