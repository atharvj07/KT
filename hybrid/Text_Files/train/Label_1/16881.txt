#include<bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n; string s; cin >> s;
    vector<int> last(26, -1);
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (last[s[i] - 'a'] == -1) ans++;
        last[s[i] - 'a'] = i;
    }
    cout << ans << endl;
    return 0;
}
