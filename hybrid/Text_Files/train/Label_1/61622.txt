import sys
input = sys.stdin.readline
inf = 1000000001

n, k = map(int, input().split())
x = [0] * n
y = [0] * n
for i in range(n):
    a, b, c, d = map(int, input().split())
    x[i] = (a + c) / 2
    y[i] = (b + d) / 2
x.sort()
y.sort()
if k > 0:
    ans = inf
    for i in range(k):
        ans = min(ans, x[i + n - k] - x[i] + y[i + n - k] - y[i])
else:
    ans = x[n - 1] - x[0] + y[n - 1] - y[0]
print(ans * ans)
