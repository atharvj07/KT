#include <bits/stdc++.h>
using namespace std;
char s[100001],ans[100001];
int main(){
    int n,k;
    cin>>n>>k;
    cin>>s;
    bool f=0;
    if(k>=25)f=1;
    else{
        for(int i=0;i<n;i++){
            if(k%2==1 && s[i]=='z'){
                f=1;
                break;
            }
            k--;
            k-=2;
        }
        if(k!=0)f=1;
    }
    if(f==1)cout<<"-1"<<endl;
    else{
        int cnt=0;
        for(int i=0;i<n;i++){
            if(k%2==1 && s[i]=='z'){
                ans[i]='a';
                k--;
                cnt++;
                continue;
            }
            if(s[i]+k/2>'z'){
                ans[i]=s[i]+k/2-'z';
                cnt+=(k/2+26);
                k=0;
                continue;
            }
            ans[i]=s[i]+k/2;
            k=0;
        }
        for(int i=0;i<n;i++){
            if(k==0)break;
            if(s[i]=='z' && cnt>0){
                ans[i]='a';
                cnt--;
                k--;
                continue;
            }
            if(ans[i]=='z'){
                ans[i]='a';
                k-=2;
                continue;
            }
            ans[i]=ans[i]+1;
            k--;
        }
        for(int i=0;i<n;i++)cout<<ans[i];
        cout<<endl;
    }
}
