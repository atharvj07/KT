import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            int n = in.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = in.nextInt();
            }
            int[] cnt = new int[101];
            for (int j : a) {
                cnt[j]++;
            }
            int ans = 0;
            for (int j = 0; j < n; j++) {
                if (cnt[a[j]] == 1) {
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
