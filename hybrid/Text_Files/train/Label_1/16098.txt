#include <iostream>
#include <vector>
using namespace std;

typedef long long ll;
const int MAXN = 1e9;

int a, b, c;
vector<ll> ans;

ll sum(ll n) {
  ll ans = 0;
  while (n) {
    ans += n % 10;
    n /= 10;
  }
  return ans;
}

void solve(ll l, ll r) {
  if (l > r) return;
  ll mid = (l + r) / 2;
  ll now = b * sum(mid) + c;
  if (now > mid)
    solve(mid + 1, r);
  else if (now == mid) {
    ans.push_back(mid);
    solve(l, mid - 1);
    solve(mid + 1, r);
  } else
    solve(l, mid - 1);
}

int main() {
  cin >> a >> b >> c;
  solve(1, MAXN);
  cout << ans.size() << '\n';
  for (auto it : ans) cout << it << ' ';
  cout << '\n';
}
