#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a.begin(), a.end());
        long long ans = 1LL * a[0] * a[n - 1];
        for (int i = 1, j = n - 2; i < j; i++, j--) {
            ans = max(ans, 1LL * a[i] * a[j]);
        }
        cout << ans << "\n";
    }
    return 0;
}
