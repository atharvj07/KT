#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int N = 3e5 + 10;
int b[N];
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> b[i];
    vector<ll> v(n);
    ll sum = 0;
    for (int i = 0; i < n; i++) {
        sum += b[i];
        v[i] = sum;
    }
    ll ans = LLONG_MAX;
    for (int i = 0; i <= n; i++) {
        ll foo = v[i - 1];
        if (i != 0) foo -= (1LL * i * (i - 1)) / 2;
        ll bar = v[n - 1] - foo;
        if (i != n) bar -= (1LL * (n - i) * (n - i - 1)) / 2;
        ans = min(ans, foo + bar);
    }
    cout << ans << endl;
    return 0;
}
