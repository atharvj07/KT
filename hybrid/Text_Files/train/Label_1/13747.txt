#include <bits/stdc++.h>

using namespace std;

int main() {
  string t;
  cin >> t;
  int n = t.size();
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (t[i] == 'a') cnt++;
  }
  if (cnt * 2 == n) {
    for (int i = 0; i < cnt; i++) cout << "a";
  } else {
    cout << ":(";
  }
  cout << endl;
  return 0;
}
