x = input()
y = input()
n, m = len(x), len(y)
L = [[0] * (m + 1) for _ in range(n + 1)]
for i in range(n):
    for j in range(m):
        if x[i] == y[j]:
            L[i + 1][j + 1] = L[i][j] + 1
        else:
            L[i + 1][j + 1] = max(L[i][j + 1], L[i + 1][j])
print(-1 if L[n][m] == n else max(n, m) - L[n][m])
