import java.util.Scanner;

public class VasyaAndRectangles {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        char[][] paper = new char[n][m];

        // Read the paper
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                paper[i][j] = line.charAt(j);
            }
        }

        boolean hasTwoFrames = false;
        int x1 = -1, y1 = -1, x2 = -1, y2 = -1;
        int x3 = -1, y3 = -1, x4 = -1, y4 = -1;

        // Find the first frame
        for (int i = 0; i < n - 2; i++) {
            for (int j = 0; j < m - 2; j++) {
                if (paper[i][j] == '#' && paper[i][j + 2] == '#' && paper[i + 2][j] == '#' && paper[i + 2][j + 2] == '#') {
                    hasTwoFrames = true;
                    x1 = i + 1;
                    y1 = j + 1;
                    x2 = i + 2;
                    y2 = j + 2;
                    break;
                }
            }
            if (hasTwoFrames) {
                break;
            }
        }

        // Find the second frame
        if (hasTwoFrames) {
            for (int i = 0; i < n - 2; i++) {
                for (int j = 0; j < m - 2; j++) {
                    if (paper[i][j] == '#' && paper[i][j + 2] == '#' && paper[i + 2][j] == '#' && paper[i + 2][j + 2] == '#') {
                        x3 = i + 1;
                        y3 = j + 1;
                        x4 = i + 2;
                        y4 = j + 2;
                        break;
                    }
                }
                if (x3 != -1) {
                    break;
                }
            }
        }

        // Check if the paper has two frames and nothing besides them
        if (hasTwoFrames && x3 != -1) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (paper[i][j] == '#' && (i < x1 || i > x2 || j < y1 || j > y2) && (i < x3 || i > x4 || j < y3 || j > y4)) {
                        hasTwoFrames = false;
                        break;
                    }
                }
                if (!hasTwoFrames) {
                    break;
                }
            }
        }

        if (hasTwoFrames) {
            System.out.println("YES");
            System.out.println(x1 + " " + y1 + " " + x2 + " " + y2);
            System.out.println(x3 + " " + y3 + " " + x4 + " " + y4);
        } else {
            System.out.println("NO");
        }
    }
}
