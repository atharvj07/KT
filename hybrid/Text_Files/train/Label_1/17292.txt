#include <bits/stdc++.h>
using namespace std;
struct cir {
  double x, y, r;
  cir(double x, double y, double r) : x(x), y(y), r(r) {}
};
int n, x, y, r;
int main() {
  cin >> n;
  vector<cir> c;
  while (n--) { cin >> x >> y >> r, c.push_back(cir(x, y, r)); }
  double mx = -1e9, my = -1e9, mxr = 0, myr = 0, flg = 0;
  for (auto cx : c) {
    if (cx.x - cx.r <= mx) {
      flg = 1;
      break;
    }
    mx = max(mx, cx.x + cx.r);
  }
  for (auto cx : c) {
    if (cx.x + cx.r >= mxr) {
      mxr = cx.x + cx.r;
    }
    if (cx.y - cx.r <= my) {
      flg = 1;
      break;
    }
    my = max(my, cx.y + cx.r);
  }
  for (auto cx : c) {
    if (cx.y + cx.r >= myr) {
      myr = cx.y + cx.r;
    }
  }
  if (!flg) {
    if (mxr >= mx && my <= 0) mxr = mx + 1e9;
    if (myr >= my && mx <= 0) myr = my + 1e9;
  }
  // cout << mx << ' ' << my << ' ' << mxr << ' ' << myr << endl;
  int ans = 4 + (mxr > mx) + (myr > my);
  cout << ans << endl;
}
