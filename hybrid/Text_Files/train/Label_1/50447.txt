import java.util.Scanner;
import java.util.List;
import java.util.ArrayList;

public class Main {
    private static final int MAX_CAMP = 300005;
    private static List<Integer> dist = new ArrayList<>();
    private static int[] left = new int[MAX_CAMP];
    private static int[] right = new int[MAX_CAMP];

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        for (int i = 0; i < n; i++) {
            dist.add(in.nextInt());
        }
        for (int i = 1; i <= n; i++) {
            if (0 <= i - dist.get(i - 1) && i - dist.get(i - 1) <= n) {
                left[i - dist.get(i - 1)] = Math.max(left[i - dist.get(i - 1)], i);
            }
            if (0 <= i + dist.get(i - 1) && i + dist.get(i - 1) <= n) {
                right[i + dist.get(i - 1)] = Math.max(right[i + dist.get(i - 1)], i);
            }
        }
        for (int i = n; i >= 1; i--) {
            if (i <= left[i]) {
                left[i] = left[left[i]];
            }
            if (right[i] <= i) {
                right[i] = right[right[i]];
            }
        }
        int flg = 0;
        for (int i = 1; i <= n; i++) {
            if (left[i] == n && right[i] == 1) {
                flg = 1;
                break;
            }
        }
        System.out.println(flg == 1 ? "yes" : "no");
    }
}
