#include<bits/stdc++.h>
using namespace std;
const int siz = 100005;
struct data {
    string name;
    int score, reg;
    data() {}
    data(string _name, int _score, int _reg) {
        name = _name, score = _score, reg = _reg;
    }
    bool operator<(const data &p)const {
        return score > p.score;
    }
};
data arr[siz];
int main() {
    int n, m;
    cin >> n >> m;
    for(int i = 0; i < n; i++) {
        string s; int x, y;
        cin >> s >> x >> y;
        arr[i] = data(s, y, x);
    }
    sort(arr, arr + n);
    for(int i = 0, cnt = 0; i < n; i++) {
        if(cnt == 2) {
            i++, cnt = 1;
            cout << "?" << "\n";
            continue;
        }
        if(arr[i].reg == arr[i + 1].reg)cnt++;
        else {
            cout << arr[i].name << " " << arr[i + 1].name << "\n";
            i++, cnt = 0;
        }
    }
    return 0;
}
