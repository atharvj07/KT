import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int[] b = new int[n];
        int[] c = new int[n];
        int j = 0;
        for (int i = 0; i < n; i++) {
            if (i == 0 || a[i] != a[i - 1]) {
                b[j] = a[i];
                c[j] = 1;
                j++;
            } else {
                c[j - 1]++;
            }
        }
        int ans = 0;
        int t = 0;
        for (int i = 0; i < j; i++) {
            if (t + b[i] <= k) {
                t += b[i];
                ans += c[i];
            } else {
                int r = (k - t) % b[i];
                ans += (k - t) / b[i];
                if (r > 0) {
                    ans++;
                }
                break;
            }
        }
        System.out.println(ans);
    }
}
