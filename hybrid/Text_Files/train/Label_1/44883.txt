import java.util.Scanner;

class Main {
    static int m, n, ans = -1;
    static char[][] a = new char[55][55];
    static boolean[][] used = new boolean[55][55];

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < m; j++)
                a[i][j] = s.charAt(j);
        }
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++) {
                if (used[i][j])
                    continue;
                dfs(i, j);
                ans = ans == -1 ? 1 : ans + 1;
            }
        System.out.println(ans);
    }

    static int[] dx = { 0, 0, 1, -1 }, dy = { 1, -1, 0, 0 };

    static void dfs(int i, int j) {
        used[i][j] = true;
        for (int d = 0; d < 4; d++) {
            int ii = i + dx[d];
            int jj = j + dy[d];
            if (ii >= 0 && ii < n && jj >= 0 && jj < m && used[ii][jj] == false && a[ii][jj] == '#')
                dfs(ii, jj);
        }
    }
}
