#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 998244353
ll power(ll x, ll y) {
    if (y == 0) return 1;
    ll p = power(x, y / 2) % mod;
    p = (p * p) % mod;
    if (y % 2 == 1) p = (p * x) % mod;
    return p;
}
int main() {
    ll n, k; cin >> n >> k;
    cout << (power(n, k) - power(n - k, k) + mod) % mod << endl;
    return 0;
}
