def can_make_equal(n, a):
    if sum(a) % n != 0:
        return ("NO",)
    target = sum(a) // n
    operations = []
    for i in range(1, n):
        remainder = (i + 1 - a[i] % (i + 1)) % (i + 1)
        operations.append((1, i + 1, remainder))
        a[0] -= remainder
        a[i] += remainder
        operations.append((i + 1, 1, a[i] // (i + 1)))
        a[0] += a[i]
        a[i] = 0
    if a[0] != target * n:
        return ("NO",)
    operations.append((1, 2, target))
    return ("YES", len(operations)) + tuple(operations)

n = int(input().strip())
a = list(map(int, input().strip().split()))

result = can_make_equal(n, a)
print(result[0])
if result[0] == "YES":
    print(result[1])
    for op in result[2:]:
        print(*op)
