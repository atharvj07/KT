import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), r = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int l = Math.max(0, i - r), r2 = Math.min(n - 1, i + r);
            int cnt = k / (r2 - l + 1);
            k -= cnt * (r2 - l + 1);
            a[i] += cnt;
        }
        for (int i = 0; i < n; i++) {
            ans = Math.max(ans, a[i]);
        }
        System.out.println(ans);
    }
}
