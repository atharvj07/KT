import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long weight[] = new long[n];
        long ans = n * 1000000007;
        for (int i = 0; i < n; i++) {
            weight[i] = sc.nextInt();
            if (weight[i] > k) {
                ans = -1;
            }
        }
        if (ans != -1) {
            if (n == 1) {
                System.out.println(1);
                System.out.println(1);
                return;
            }
            Arrays.sort(weight);
            if (weight[0] == weight[n - 1]) {
                ans = 1;
            } else {
                int cnt = 0;
                long sum = 0;
                for (int i = n - 1; i >= 0; i--) {
                    if (sum + weight[i] <= k) {
                        sum += weight[i];
                        cnt++;
                    } else {
                        break;
                    }
                }
                ans = 1;
                for (int i = 0; i < cnt - 1; i++) {
                    ans = ans * 2 % 1000000007;
                }
                ans = ans * cnt % 1000000007;
            }
        }
        if (ans == -1) {
            System.out.println(-1);
            System.out.println(0);
        } else {
            System.out.println(ans);
        }
    }
}
