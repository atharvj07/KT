#include <bits/stdc++.h>
using namespace std;
int lcs(string a, string b) {
  int n = a.size(), m = b.size();
  int l[n + 1][m + 1];
  for (int i = n; i >= 0; i--) {
    for (int j = m; j >= 0; j--) {
      if (i == n || j == m)
        l[i][j] = 0;
      else if (a[i] == b[j])
        l[i][j] = 1 + l[i + 1][j + 1];
      else
        l[i][j] = max(l[i + 1][j], l[i][j + 1]);
    }
  }
  return l[0][0];
}
int main() {
  string a, b;
  cin >> a >> b;
  cout << a.size() - lcs(a, b) << endl;
  return 0;
}
