import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), M = sc.nextInt();
        int[] w = new int[N];
        int[] b = new int[M];
        for (int i = 0; i < N; i++) w[i] = sc.nextInt();
        for (int i = 0; i < M; i++) b[i] = sc.nextInt() - 1;
        long ans = 0;
        long sum = 0;
        TreeSet<Integer> ts = new TreeSet<>();
        for (int i = 0; i < N; i++) {
            ts.add(i);
            sum += w[i];
        }
        for (int i = 0; i < M; i++) {
            int x = b[i];
            long t = sum - (long)w[x];
            ts.remove(x);
            ans += t;
            Integer u = ts.lower(x);
            if (u != null) ans -= (long)w[u];
            u = ts.higher(x);
            if (u != null) ans += (long)w[u];
        }
        System.out.println(ans);
    }
}
