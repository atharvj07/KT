import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int a = input.nextInt();
        int b = input.nextInt();
        int f = input.nextInt();
        int k = input.nextInt();
        if ((f - 1) / b + ((a - f) - 1) / b * 2 >= k) {
            System.out.println((k - 1) / 2 + 1);
        } else {
            System.out.println(-1);
        }
    }
}
