#include <bits/stdc++.h>
using namespace std;
int n, k, ans[(int)2e3];
int main() {
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < 7; j++) {
      if (s[j] == '0') ans[i] |= 1 << j;
    }
  }
  if (k != __builtin_popcount(ans[0])) {
    cout << -1;
    return 0;
  }
  for (int i = 1; i < n; i++) {
    ans[i] &= ans[i - 1];
    if (k != __builtin_popcount(ans[i])) {
      cout << -1;
      return 0;
    }
  }
  for (int i = 9; i >= 0; i--) {
    int cnt = __builtin_popcount(i);
    if (cnt <= k) {
      for (int j = 0; j < n; j++) {
        if ((ans[j] & i) == i) { cout << i; }
      }
      return 0;
    }
  }
}
