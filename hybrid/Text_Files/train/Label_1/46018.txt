import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextLong();
        }
        long a1 = b[0];
        int[] p = new int[n];
        for (int i = 1; i < n; i++) {
            p[i] = i;
            a1 ^= b[i];
        }
        if (a1 <= 0) {
            System.out.println("No");
            return;
        }
        long[] a = new long[n];
        a[0] = a1;
        for (int i = 1; i < n; i++) {
            a[i] = a[i - 1] ^ b[i];
        }
        int[] q = new int[n];
        q[0] = 0;
        for (int i = 1; i < n; i++) {
            q[i] = i;
        }
        for (int i = 1; i < n; i++) {
            for (int j = i; j > 0; j--) {
                if (a[q[j]] < a[q[j - 1]]) {
                    int tmp = q[j];
                    q[j] = q[j - 1];
                    q[j - 1] = tmp;
                } else {
                    break;
                }
            }
        }
        System.out.println("Yes");
        for (int i = 0; i < n; i++) {
            System.out.print(b[q[i]] + " ");
        }
        System.out.println();
    }
}
