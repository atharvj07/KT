#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int cal(long long x) {
    vector<int> a;
    while (x) {
        a.emplace_back(x%10);
        x/=10;
    }
    reverse(a.begin(),a.end());
    while (!a.empty() && !a.back()) a.pop_back();
    reverse(a.begin(),a.end());
    if (!a.empty()) {
        long long ret=0;
        for (auto &p:a) {
            ret*=10;
            ret+=p;
        }
        return ret;
    }
    return 0;
}
int32_t main() {
    int t;
    cin>>t;
    while (t--) {
        long long x;
        cin>>x;
        cout<<cal(cal(cal(x)))<<'\n';
    }
}
