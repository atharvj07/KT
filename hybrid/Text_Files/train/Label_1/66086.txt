n = int(input())
ans = []
if n % 2 == 0:
    ans.append(n - 1)
    if n % 4 == 0:
        ans.append(2)
    else:
        for i in range(3, n, 2):
            if ans[-1] * i % n == 1:
                ans.append(i)
    ans.reverse()
else:
    for i in range(1, n, 2):
        if ans[-1] * i % n == 1:
            ans.append(i)
    ans.reverse()
print(len(ans))
print(*ans)
