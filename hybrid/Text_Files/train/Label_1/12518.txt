#include<bits/stdc++.h>
using namespace std;
bool has_vowel(char c, char d){
    if(c=='n') return true;
    return d=='a'||d=='e'||d=='i'||d=='o'||d=='u';
}
int main(){
    string s; cin>>s;
    for(int i=0; i<s.size(); i++){
        if((s[i]>='a' && s[i]<='z') && has_vowel(s[i], (i==s.size()-1?'n':s[i+1]))==false){
            cout<<"NO";
            return 0;
        }
    }
    cout<<"YES";
    return 0;
}
