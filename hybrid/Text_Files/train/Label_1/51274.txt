import java.util.*;

public class Main {
    static char[][] t;
    static int n, m;
    static int[][] dp;

    public static boolean check(int i, int j) {
        if (i == 1 && j == 1) return true;
        return (i > 1 && check(i - 1, j)) || (j > 1 && check(i, j - 1));
    }

    public static int func(int sum, boolean first) {
        if (sum < 0 || sum > 0 && first) return -1;
        if (sum > 0 && !first) return 1;
        if (dp[sum + m][n] != -2) return dp[sum + m][n];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (check(i, j)) {
                    char ch = t[i][j];
                    if (first) {
                        if (ch == 'a') {
                            int ret = func(sum + 1, !first);
                            if (ret != -1) {
                                dp[sum + m][n] = ret;
                                return ret;
                            }
                        }
                    } else {
                        if (ch == 'b') {
                            int ret = func(sum - 1, !first);
                            if (ret != -1) {
                                dp[sum + m][n] = ret;
                                return ret;
                            }
                        }
                    }
                }
            }
        }
        dp[sum + m][n] = -1;
        return -1;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        t = new char[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            String s = in.next();
            for (int j = 1; j <= n; j++) {
                t[i][j] = s.charAt(j - 1);
            }
        }
        m = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (check(i, j)) m++;
            }
        }
        dp = new int[2 * m + 1][n + 1];
        for (int i = 0; i < 2 * m + 1; i++) {
            for (int j = 0; j <= n; j++) {
                dp[i][j] = -2;
            }
        }
        int res = func(0, true);
        if (res == -1) {
            System.out.println("DRAW");
        } else if (res == 1) {
            System.out.println("FIRST");
        } else {
            System.out.println("SECOND");
        }
    }
}
