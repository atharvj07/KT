#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;
const int MAXN = 100005;
int a[MAXN], n, a1, a2, a3;
ll ans;

int main() {
  cin >> n >> a1 >> a2 >> a3;
  for (int i = 0; i < n; ++i) cin >> a[i];
  sort(a, a + n);
  ans = a3 * n + 1ll * (a[n / 2 - 1] - a[0]) * a1 + 1ll * (a[n - 1] - a[n / 2]) * a2;
  cout << ans << endl;
  return 0;
}
