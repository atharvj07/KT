#include <bits/stdc++.h>

using namespace std;

int apps, events;
vector<int> nots;
int unread = 0;

int main() {
  cin >> apps >> events;
  nots.resize(apps + 1);
  for (int i = 0, tp, x, t; i < events; i++) {
    cin >> tp >> x;
    if (tp == 1) {
      nots[x]++;
      unread++;
    } else if (tp == 2) {
      unread -= nots[x];
      nots[x] = 0;
    } else {
      unread = max(unread - t, 0);
    }
    cout << unread << endl;
  }
  return 0;
}
