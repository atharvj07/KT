import sys
input = sys.stdin.readline

m = int(input())
h1, a1 = map(int, input().split())
x1, y1 = map(int, input().split())
h2, a2 = map(int, input().split())
x2, y2 = map(int, input().split())
if h1 == a1 or h2 == a2:
    print(-1)
    sys.exit()
lst = [(1, x1, y1), (m-1, x2, y2)]
c1 = 0
c2 = 0
flag = True
for l, x, y in lst:
    if flag:
        if (h1 * x + y) % m == a1:
            flag = False
            c1 += 1
        else:
            c1 += 1
            h1 = (h1 * x + y) % m
    else:
        if (h2 * x + y) % m == a2:
            break
        else:
            c2 += 1
            h2 = (h2 * x + y) % m
if flag:
    print(-1)
else:
    print(c1 + c2)
