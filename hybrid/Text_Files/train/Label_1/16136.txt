#include <map>
#include <vector>
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<long long>> a(n, vector<long long>(m));
  vector<string> s(n);
  vector<map<char, long long>> pos(n);
  for (int i = 0; i < n; i++) {
    string t;
    cin >> t;
    s[i] = t;
    for (int j = 0; j < m; j++) {
      pos[i][t[j]]++;
      cin >> a[i][j];
    }
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    long long mn = 1e9;
    for (int j = 0; j < m; j++) {
      if (pos[i][s[i][j]] == 1) {
        mn = min(mn, a[i][j]);
      }
    }
    if (mn != 1e9)
      ans += mn;
  }
  cout << ans << endl;
  return 0;
}
