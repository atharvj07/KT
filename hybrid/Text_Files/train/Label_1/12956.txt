#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int t; cin >> t;
    while (t--) {
        int n, x, ans = 1; cin >> n >> x;
        vector<int> a(n);
        for (int& i : a) cin >> i;
        sort(a.begin(), a.end());
        for (int i : a) {
            if (i <= ans) ans++;
            else break;
        }
        if (ans + x >= a.back()) ans = a.back() + 1;
        cout << ans << endl;
    }
}
