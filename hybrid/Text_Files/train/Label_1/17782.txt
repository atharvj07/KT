#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<double> p(n);
    for (double &x : p) { cin >> x; }
    double l = 0, r = 0;
    for (const double &x : p) { l = max(l, x); }
    for (const double &x : p) { r += (1 - x); }
    r = 1 - r;
    cout.precision(15);
    cout << max(l, r) << endl;
}
