#include <iostream>
#include <vector>

using namespace std;

vector<bool> digits(10);

bool is_fair(long long n) {
    while (n > 0) {
        long long d = n % 10;
        n /= 10;
        if (d == 0 || n % d != 0) return false;
    }
    return true;
}

long long find_next_fair(long long n) {
    long long x = n;
    while (!is_fair(x)) {
        bool incd = false;
        for (long long i = 9; i >= 0; i--) {
            if (digits[i]) {
                if (!incd) incd = true;
                else digits[i] = false;
            } else {
                digits[i] = true;
                incd = false;
                break;
            }
        }
        if (!incd) {
            long long l = x;
            long long r = l * 10 + 9;
            while (l <= r) {
                long long m = (l + r) / 2;
                if (is_fair(m)) x = m, r = m - 1;
                else l = m + 1;
            }
        }
        x++;
    }
    return x;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n;
        cin >> n;
        cout << find_next_fair(n) << endl;
    }
    return 0;
}
