#include <bits/stdc++.h>
#define MN 1001
using namespace std;
int n, m, a[MN][MN];
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { for (int j = 1; j <= m; j++) { cin >> a[i][j]; } }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            vector<int> x;
            x.push_back(a[i][j]);
            if (i > 1) x.push_back(a[i - 1][j]);
            if (j > 1) x.push_back(a[i][j - 1]);
            if (i < n) x.push_back(a[i + 1][j]);
            if (j < m) x.push_back(a[i][j + 1]);
            sort(x.begin(), x.end());
            int cnt = 0;
            for (int k = 0; k < x.size(); k++) {
                if (k == 0 || x[k] != x[k - 1]) cnt++;
            }
            cout << cnt << " ";
        }
        cout << endl;
    }
    return 0;
}
