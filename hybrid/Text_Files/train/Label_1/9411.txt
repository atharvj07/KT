#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> votes(m, vector<int>(n));
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            cin >> votes[i][j];
        }
    }
    vector<pair<int, int>> city_winners(m);
    for (int i = 0; i < m; i++) {
        int max_votes = 0, winner = 0;
        for (int j = 0; j < n; j++) {
            if (votes[i][j] > max_votes) {
                max_votes = votes[i][j];
                winner = j;
            } else if (votes[i][j] == max_votes) {
                winner = min(winner, j);
            }
        }
        city_winners[i] = {max_votes, winner};
    }
    int max_wins = 0, winner = 0;
    for (int i = 0; i < m; i++) {
        if (city_winners[i].first > max_wins) {
            max_wins = city_winners[i].first;
            winner = city_winners[i].second;
        } else if (city_winners[i].first == max_wins) {
            winner = min(winner, city_winners[i].second);
        }
    }
    cout << winner + 1 << endl;
    return 0;
}
