n = int(input())
a = [int(x) for x in input().split()]
dp = [[None] * 3 for _ in range(n + 1)]

def solve(i, p):
    if i == n:
        return 0

    if not dp[i][p]:
        if p == 1:
            dp[i][p] = solve(i + 1, 1)
        else:
            dp[i][p] = max(solve(i + k, 1) + sum(a[i:i + k]) for k in range(2, n - i + 1))

    return dp[i][p]

print(solve(0, 1))
