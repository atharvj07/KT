import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] y = new int[3], m = new int[3], p = new int[3];
        for (int i = 0; i < y.length; i++) {
            y[i] = scanner.nextInt();
        }
        for (int i = 0; i < m.length; i++) {
            m[i] = scanner.nextInt();
        }
        for (int i = 0; i < p.length; i++) {
            p[i] = scanner.nextInt();
        }
        System.out.println(Math.max(0, (m[0] - 1 + p[2]) / y[2]) * p[0] + Math.max(0, (y[0] - 1 + p[1]) / m[1]) * p[1]);
    }
}
