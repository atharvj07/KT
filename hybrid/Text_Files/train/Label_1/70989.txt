import sys

def solve():
    t = int(input())
    for _ in range(t):
        n, m = map(int, input().split())
        a = list(map(int, input().split()))
        a.sort()
        cur = 0
        ok = True
        for ai in a:
            cur += ai
            if cur / (n - a.index(ai)) > m:
                ok = False
                break
        print("YES" if ok else "NO")

solve()
