#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll mod = 1000000007;
const ll inf = 1e18;
ll n, d;
int main()
{
    cin >> n >> d;
    ll sz = 0;
    while (sz * (sz + 1) / 2 < n)
        sz++;
    ll a[sz];
    cout << sz << endl;
    for (ll i = 0; i < sz; i++)
    {
        a[i] = i * d + 1;
        cout << a[i] << ' ';
    }
    cout << endl;
    return 0;
}
