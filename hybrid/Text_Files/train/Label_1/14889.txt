#include <iostream>
#include <iomanip>
using namespace std;
int main(){
    int n;
    cin >> n;
    for (int i = 0; i < n; i++){
        for (int j = 0; j < n; j++){
            cout << ((i == (n - 1) / 2 || j == (n - 1) / 2 || i + j == (n - 1) / 2 || j - i == (n - 1) / 2) ? "D" : "*");
        }
        cout << '\n';
    }
    return 0;
}
