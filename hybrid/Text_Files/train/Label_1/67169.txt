import sys

n = int(input())
trips = [int(input()) for _ in range(n)]

dp = [0] * (n + 1)
tickets = [20, 50, 120]
times = [1, 90, 1440]

for i in range(1, n + 1):
    dp[i] = dp[i - 1] + tickets[0]
    for j in range(1, 3):
        t = trips[i - 1] - times[j] + 1
        if t >= 0:
            dp[i] = min(dp[i], dp[t] + tickets[j])

print(*[dp[i] - dp[i - 1] for i in range(1, n + 1)])
