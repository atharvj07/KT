import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner stdin = new Scanner(System.in);
        long n = stdin.nextLong();
        long m = stdin.nextLong();
        long[][] a = new long[2][(int) (n + 1)];
        for (int i = 1; i <= n; i++) {
            a[0][i] = a[0][i - 1] + stdin.nextLong();
            a[1][i] = (a[0][i] % m + m) % m;
        }
        long ans = 0;
        for (int l = 1; l <= n; l++) {
            for (int r = l; r <= n; r++) {
                if ((a[1][r] - a[1][l - 1]) % m == 0) ans++;
            }
        }
        System.out.println(ans);
    }
}
