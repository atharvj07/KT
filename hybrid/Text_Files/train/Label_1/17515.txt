#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<ll> a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        ll sum = 0;
        for (auto x : a) { sum += x; }
        if (sum <= 0) {
            cout << "NO\n";
            continue;
        }
        ll mx = LLONG_MIN;
        for (int i = 0; i < n; i++) { mx = max(mx, a[i]); }
        if (sum > mx) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
    return 0;
}
