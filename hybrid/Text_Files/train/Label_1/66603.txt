n = int(input().strip())
rectangles = [list(map(int, input().strip().split())) for _ in range(n)]

# Initially, set the maximum possible height to a very large number.
max_height = 10**9

for w, h in rectangles:
    # For each rectangle, choose the orientation that gives the maximum height
    # that is still not greater than the maximum height so far.
    if max(w, h) <= max_height:
        max_height = max(w, h)
    elif min(w, h) <= max_height:
        max_height = min(w, h)
    else:
        print('NO')
        break
else:
    print('YES')
