#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
#define endl '\n'

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    ll n, m;
    cin >> n >> m;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    ll ans = 1;
    for (ll i = 0; i < n; i++) { ans = (ans * (a[i] - a[0])) % m; }
    cout << ans << endl;
    return 0;
}
