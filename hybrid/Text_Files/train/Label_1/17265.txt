#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  if (n == 1) {
    cout << "No" << endl;
    return 0;
  }
  cout << "Yes" << endl;
  vector<int> a, b, c;
  if (n > 2) {
    for (int i = 2; i <= n; i += 2) a.push_back(i);
    for (int i = 1; i <= n; i += 2) b.push_back(i);
  } else {
    a.push_back(2);
    b.push_back(1);
  }
  cout << a.size() << ' ';
  for (int x : a) cout << x << ' ';
  cout << endl << b.size() << ' ';
  for (int x : b) cout << x << ' ';
  cout << endl;
}
