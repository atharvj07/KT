#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m;cin>>n>>m;
    vector<vector<int>>cost(n,vector<int>(m,0));
    vector<string>vec(n);
    for(int i=0;i<n;i++)cin>>vec[i];
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            cin>>cost[i][j];
        }
    }
    int ans=0;
    for(int j=0;j<m;j++){
        vector<int>freq(26,0);
        int ch=-1,pos=-1;
        for(int i=0;i<n;i++){
            int x=vec[i][j]-'a';
            if(freq[x]>0){
                ch=x;
                pos=i;
                break;
            }
            freq[x]++;
        }
        if(ch!=-1){
            for(int i=0;i<n;i++){
                if(vec[i][j]==vec[pos][j])continue;
                ans+=cost[i][j];
            }
        }
    }
    cout<<ans<<endl;
    return 0;
}
