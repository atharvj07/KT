import java.math.BigInteger;
import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[][] ranges = new long[n][2];
        for (int i = 0; i < n; i++) {
            ranges[i][0] = scanner.nextLong();
            ranges[i][1] = scanner.nextLong();
        }
        // Sort by birth year
        Arrays.sort(ranges, (a, b) -> Long.compare(a[0], b[0]));
        // Track the maximum number of alive individuals and the corresponding year
        long maxAlive = -1, maxYear = -1;
        // Initialize count with the first person's birth year
        long count = 1;
        // Initialize the current year to the first person's birth year
        long year = ranges[0][0];
        // Iterate through the ranges
        for (int i = 1; i < n; i++) {
            // Update the current year to the year of the next person's birth
            year = ranges[i][0];
            // Subtract those who died before this year
            while (year >= ranges[i - 1][1]) {
                count--;
                i++;
                if (i == n) {
                    break;
                }
                year = ranges[i][0];
            }
            // Add those who were born this year
            count++;
            // Update the maximum number of alive individuals and the corresponding year if necessary
            if (count > maxAlive) {
                maxAlive = count;
                maxYear = year;
            }
        }
        System.out.println(maxYear + " " + maxAlive);
    }
}
