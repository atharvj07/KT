import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] cnt = new int[2];
        while (true) {
            int n = sc.nextInt();
            if (n == 0) break;
            String[] s = new String[n];
            for (int i = 0; i < n; i++) s[i] = sc.next();
            for (int i = 0; i < n; i++) {
                if (s[i].startsWith("l") && s[i].endsWith("u")) cnt[0]++;
                if (s[i].startsWith("r") && s[i].endsWith("d")) cnt[0]++;
                if (s[i].startsWith("l") && s[i].endsWith("d")) cnt[1]++;
                if (s[i].startsWith("r") && s[i].endsWith("u")) cnt[1]++;
            }
            System.out.println(Math.min(cnt[0], cnt[1]));
            cnt[0] = cnt[1] = 0;
        }
    }
}
