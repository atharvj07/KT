#include <bits/stdc++.h>
using namespace std;

const int N = 140582;
long long a[N], b[N];
long long egcd(long long a, long long b, long long &x, long long &y) {
  if (b == 0) {
    x = 1;
    y = 0;
    return a;
  }
  long long g = egcd(b, a % b, y, x);
  y -= (a / b) * x;
  return g;
}
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> b[i];
  long long x, y;
  for (int i = 0; i < n; i++) {
    if (i) b[i] = egcd(a[i - 1], b[i], x, y);
    if (b[i]) {
      cout << "NO\n";
      return 0;
    }
    a[i] = 1LL * i * a[i - 1];
  }
  cout << "YES\n";
  for (int i = 0; i < n; i++) cout << a[i] << " ";
  return 0;
}
