#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  int ans = 1e9;
  for (int i = m - 1; i >= 0; i--) {
    if (!a[i] || a[i] > k) break;
    ans = min(ans, (m - i - 1) * 10);
    k -= a[i];
  }
  for (int i = m + 1; i < n; i++) {
    if (!a[i] || a[i] > k) break;
    ans = min(ans, (i - m) * 10);
    k -= a[i];
  }
  cout << ans << endl;
  return 0;
}
