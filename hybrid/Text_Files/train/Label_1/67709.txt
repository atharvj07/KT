from collections import Counter
from sys import stdin

n, m = map(int, stdin.readline().split())
maze = []
for _ in range(n):
    maze.append(stdin.readline().strip())
start = end = None
for i in range(n):
    for j in range(m):
        if maze[i][j] == 'S':
            start = (i, j)
        elif maze[i][j] == 'E':
            end = (i, j)
dirs = stdin.readline().strip()
h, w = -1, 1  # bottom, top
l, r = 0, 2  # left, right
for row in maze:
    for ch in row:
        if ch == '#':
            for i in (h, w, l, r):
                dirs = dirs.replace(str(i), '')
        elif ch == 'E':
            for i in (h, l):
                dirs = dirs.replace(str(i), '')
        elif ch == 'S':
            for i in (w, r):
                dirs = dirs.replace(str(i), '')
dirs = Counter(dirs)
print(dirs[str(h)] * dirs[str(l)] * dirs[str(w)] * dirs[str(r)])
