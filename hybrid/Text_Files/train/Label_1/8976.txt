#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 2e5+1;
int n;
char s[N];
bool ok;
int main(){
    cin>>n>>s;
    int ans = 0;
    for(int i=0, a=0, b=0; i<n; i++){
        if(s[i]=='a')a++;
        else b++;
        if(i%2==0 && a!=b){
            if(!ok){
                ok = 1;
                ans = i+1;
                for(int j=0; j<ans; j++){
                    if(s[j]=='a')s[j] = 'b';
                    else s[j] = 'a';
                }
            }
        }
    }
    cout<<ans<<endl<<s<<endl;
    return 0;
}
