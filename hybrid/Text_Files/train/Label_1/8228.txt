#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, s, k;
        cin >> n >> s >> k;
        set<long long> closed;
        for (long long i = 0; i < k; i++) {
            long long a;
            cin >> a;
            closed.insert(a);
        }
        cout << min(abs(s - 1), abs(s - n)) << endl;
    }
}
