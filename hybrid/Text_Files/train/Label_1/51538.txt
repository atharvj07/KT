import java.util.*;

public class Main {

    static class Cell {
        int x, y, t, p;

        public Cell(int x, int y, int t, int p) {
            this.x = x;
            this.y = y;
            this.t = t;
            this.p = p;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        char[][] map = new char[n][m];
        Cell start = null;
        List<Cell> treasures = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                map[i][j] = s.charAt(j);
                if (map[i][j] == 'S') {
                    start = new Cell(i, j, 0, 0);
                } else if (map[i][j] >= '1' && map[i][j] <= '8') {
                    treasures.add(new Cell(i, j, map[i][j] - '0', 0));
                }
            }
        }
        int[] tresP = new int[treasures.size()];
        for (int i = 0; i < treasures.size(); i++) {
            tresP[i] = sc.nextInt();
        }
        int[][][] dp = new int[n][m][1 << treasures.size()];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                Arrays.fill(dp[i][j], -1);
            }
        }
        int ans = dfs(start, 0, 0, map, dp, tresP);
        System.out.println(ans);
    }

    public static int dfs(Cell cell, int mask, int steps, char[][] map, int[][][] dp, int[] tresP) {
        if (dp[cell.x][cell.y][mask] != -1) {
            return dp[cell.x][cell.y][mask];
        }
        int ans = 0;
        if (mask == (1 << tresP.length) - 1) {
            ans = Math.max(ans, cell.p - steps);
            dp[cell.x][cell.y][mask] = ans;
            return ans;
        }
        for (int i = 0; i < tresP.length; i++) {
            if ((mask & (1 << i)) == 0) {
                Cell tres = getTreasure(cell, i, map);
                if (tres != null) {
                    ans = Math.max(ans, dfs(tres, mask | (1 << i), steps + 1, map, dp, tresP) + tresP[i]);
                }
            }
        }
        dp[cell.x][cell.y][mask] = ans;
        return ans;
    }

    public static boolean isInside(Cell cell, List<Cell> path) {
        int cnt = 0;
        for (int i = 0; i < path.size() - 1; i++) {
            Cell a = path.get(i), b = path.get(i + 1);
            if (a.x != b.x) {
                if (a.x < b.x) {
                    if (cell.x >= a.x && cell.x <= b.x && cell.y == a.y) {
                        cnt++;
                    }
                } else {
                    if (cell.x >= b.x && cell.x <= a.x && cell.y == a.y) {
                        cnt++;
                    }
                }
            } else {
                if (a.y < b.y) {
                    if (cell.y >= a.y && cell.y <= b.y && cell.x == a.x) {
                        cnt++;
                    }
                } else {
                    if (cell.y >= b.y && cell.y <= a.y && cell.x == a.x) {
                        cnt++;
                    }
                }
            }
        }
        return cnt % 2 == 1;
    }

    public static boolean isValid(Cell cell, char[][] map) {
        return cell.x >= 0 && cell.x < map.length && cell.y >= 0 && cell.y < map[0].length && map[cell.x][cell.y] != '#' && map[cell.x][cell.y] != 'B';
    }

    public static Cell getTreasure(Cell cell, int tres, char[][] map) {
        int dx[] = {1, -1, 0, 0};
        int dy[] = {0, 0, 1, -1};
        for (int i = 0; i < 4; i++) {
            int nx = cell.x + dx[i], ny = cell.y + dy[i];
            if (isValid(new Cell(nx, ny, 0, 0), map) && map[nx][ny] == tres + '0') {
                return new Cell(nx, ny, 0, 0);
            }
        }
        return null;
    }
}
