import sys

xors = [0] * (300001)
cnt = [0] * (300001)
pos0 = {}
n = int(input())
a = list(map(int, input().split()))
ans = 0
for i in range(n - 1, -1, -1):
  for j in range(29, -1, -1):
    if (1 << j) & a[i]:
      if j not in pos0: pos0[j] = n
      pos0[j] = min(pos0[j], i)
  xors[i] = xors[i + 1]
  for j in range(29, -1, -1):
    if (1 << j) & a[i]:
      xors[i] |= xors[pos0[j]]
      cnt[j] += 1
  for j in range(29, -1, -1):
    if (1 << j) & xors[i]:
      if cnt[j] > 1:
        ans += cnt[j] * (cnt[j] - 1) // 2
print(ans)
