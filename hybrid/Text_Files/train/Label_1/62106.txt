t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().split())
    s = input().strip()
    prefix_sums = [0] * (k + 1)
    for i in range(k):
        count = 0
        for j in range(i, n, k):
            if s[j] == '1':
                count += 1
        prefix_sums[i + 1] = prefix_sums[i] + count
    
    ans = float('inf')
    for i in range(k):
        on = off = 0
        for j in range(i, n, k):
            if s[j] == '1':
                on += 1
            else:
                off += 1
            # Calculate minimum moves for this pattern
            ans = min(ans, prefix_sums[k] - (on - off))
    
    print(ans)
