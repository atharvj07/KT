#include <cstdio>
const int N = 500;
bool a[N], b[N];

int main() {
    int t, n;
    scanf("%d", &t);
    while (t--) {
        scanf("%d", &n);
        for (int i = 0; i < n; i++) {
            scanf("%d", &a[i]);
            scanf("%d", &b[i]);
        }
        int p = 0, q = n - 1;
        while (p < q) {
            while (p < q && a[p] == a[p + 1] && b[p] == b[p + 1]) ++p;
            while (p < q && a[q] == a[q - 1] && b[q] == b[q - 1]) --q;
            if (p < q && b[p] != b[q]) {
                ++p;
                --q;
            } else {
                break;
            }
        }
        puts(p < q ? "No" : "Yes");
    }
    return 0;
}
