import java.util.*;

public class Main {
    static int[] ancestor;
    static Map<Integer, Set<String>> sons;
    static Map<Integer, String> names;
    static int[] ans;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        ancestor = new int[n + 1];
        sons = new HashMap<>();
        names = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            names.put(i, sc.next());
            int p = sc.nextInt();
            ancestor[i] = p;
            if (p != 0) {
                sons.putIfAbsent(p, new HashSet<>());
            }
        }
        int m = sc.nextInt();
        ans = new int[m];
        for (int i = 0; i < m; i++) {
            int v = sc.nextInt();
            int k = sc.nextInt();
            int cur = v;
            for (int j = 1; j < k; j++) {
                cur = ancestor[cur];
            }
            ans[i] = sons.getOrDefault(cur, new HashSet<>()).size();
        }
        for (int v : ans) {
            System.out.print(v + " ");
        }
        System.out.println();
    }
}
