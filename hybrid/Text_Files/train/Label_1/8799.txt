#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    while (k--) {
        string t = s;
        t[0] = (s[n - 1] == s[1] || s[1] == s[2] ? s[0] : (s[0] == 'B' ? 'W' : 'B'));
        for (ll i = 1; i < n; ++i) {
            t[i] = (s[i - 1] == s[i + 1] || s[i - 1] == s[(i + 1) % n] ? s[i] : (s[i] == 'B' ? 'W' : 'B'));
        }
        s = t;
    }
    cout << s << endl;
}
