t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    k = list(map(int, input().split()))
    c = list(map(int, input().split()))

    k.sort(reverse=True)
    j = 0
    total_cost = 0
    for i in range(n):
        if j < k[i] and j < m:
            total_cost += c[j]
            j += 1
        else:
            total_cost += c[k[i] - 1]
    print(total_cost)
