#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll check(string s) {
    ll ret = 0, sum = 0;
    for (auto &c : s) {
        if (c == '(') ++sum;
        if (sum > 0) {
            if (c == '?') ret += sum - 1;
            if (c == ')') --sum;
        }
    } return ret;
}
int main() {
    string s;
    cin >> s;
    cout << check(s) << endl;
}
