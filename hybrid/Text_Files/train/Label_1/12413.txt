#include <bits/stdc++.h>
using namespace std;

const int N = 22;
const int NN = 22*22;

int ar[NN], nr, nc;
bool vis[NN];

int ans[N][N];

bool palindrome() {
    for (int i = 0; i < nr; i++) {
        for (int j = 0; j < nc; j++) {
            if (ans[i][j] != ans[nr - i - 1][nc - j - 1])
                return false;
        }
    }

    return true;
}

void print_matrix() {
    for (int i = 0; i < nr; i++) {
        for (int j = 0; j < nc; j++)
            cout << ans[i][j] << ' ';
        cout << '\n';
    }
}

bool solve(int idx, int r, int c) {
    if (idx == nr * nc + 1)
        return palindrome();
    for (int i = 1; i <= nr * nc; i++) {
        if (!vis[i]) {
            ans[r][c] = ar[i];
            vis[i] = true;
            bool pos = solve(idx + 1, r + (idx % nc == 0), (idx % nc + 1) % nc);
            vis[i] = false;
            if (pos)
                return pos;
        }
    }
    return false;
}

int main() {
    cin >> nr;
    int nn = nr*nr;
    for (int i = 1; i <= nn; i++)
        cin >> ar[i];
    sort(ar + 1, ar + nn + 1);

    for (int i = 2; i <= nn; i++)
        if (ar[i] == ar[i - 1])
            nc++;
    nc++;
    nr = nn / nc;
    solve(1, 0, 0);
    if (palindrome()) {
        cout << "YES\n";
        print_matrix();
    } else
        cout << "NO\n";
    return 0;
}
