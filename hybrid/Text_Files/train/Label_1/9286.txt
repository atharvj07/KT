#include<bits/stdc++.h>

using namespace std;

char a[2005][2005];
int dp[2005][2005];

int main(){
    int n,m,k;
    cin>>n>>m>>k;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            cin>>a[i][j];
        }
    }
    int ans=0;
    for(int i=1;i<=n;i++){
        int cnt=0;
        for(int j=1;j<=m;j++){
            if(a[i][j]=='.') cnt++;
            else cnt=0;
            if(cnt>=k) ans++;
        }
    }

    for(int j=1;j<=m;j++){
        int cnt=0;
        for(int i=1;i<=n;i++){
            if(a[i][j]=='.') cnt++;
            else cnt=0;
            if(cnt>=k) ans++;
        }
    }
    cout<<ans<<endl;
}
