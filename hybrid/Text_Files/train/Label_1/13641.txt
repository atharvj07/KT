#include <bits/stdc++.h>
using namespace std;

long long gcd(long long a, long long b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}

long long lcm(long long a, long long b) {
  return (a / gcd(a, b)) * b;
}

int main() {
  long long l, r, x, y;
  cin >> l >> r >> x >> y;
  long long cnt = 0;
  for (long long i = l; i <= r; i++) {
    for (long long j = i; j <= r; j++) {
      if (i * j % gcd(i, j) == x && i * j / gcd(i, j) == y) cnt++;
    }
  }
  cout << cnt << endl;
  return 0;
}
