n,m=(int(x)for x in input().split())
a = [['.']*m for i in range(n)]
for i in range (n):
    b = input()
    for j in range(m):
        a[i][j] = b[j]
ans = 0
for i in range(n):
    for j in range(m):
        if i==0 or j==0 or i==n-1 or j==m-1:
            continue
        if a[i][j] == '.':
            continue
        if a[i][0]==a[i][m-1]=='.' and a[0][j]==a[n-1][j]=='.':
            ans = 1
            print('YES')
            print(i+1,j+1)
            exit(0)
print('NO')
