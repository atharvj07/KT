#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, k, a2;
    cin >> n >> k;
    vector<long long> a(n + 1);
    for (int i = 0; i <= n; i++) { cin >> a[i]; }
    a2 = a[2];
    if (a2 > 0 && 2 * a[0] <= k)
        k = 2 * a[0];
    else if (a2 < 0 && 2 * a[0] >= -k)
        k = 2 * a[0];
    else
        k = 0;
    cout << k / 2 + 1 << endl;
}
