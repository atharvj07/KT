#include <bits/stdc++.h>
#define int long long
using namespace std;
typedef pair<int, int> pi;
typedef vector<int> vi;
const int mod = 1e9 + 7;
const int mxN = 1e5;
pi dp[mxN][3];

pi dfs(int i, int prv) {
  pi &res = dp[i][prv];
  if (res.first != -1) return res;
  res.first = 1;
  res.second = 1;
  if (i == 0) return res;
  res.first = res.second = 0;
  pi a = dfs(i - 1, 1);
  pi b = dfs(i - 1, 2);
  if (prv != 1) {
    res.first += a.first + a.second;
    res.second += a.first + a.second;
  }
  if (prv != 2) {
    res.first += b.first + b.second;
    res.second += b.first + b.second;
  }
  res.first %= mod;
  res.second %= mod;
  return res;
}

int32_t main() {
  int n;
  memset(dp, -1, sizeof dp);
  cin >> n;
  pi ans = dfs(n, 0);
  cout << (ans.first + ans.second - 2 + mod) % mod << '\n';
}
