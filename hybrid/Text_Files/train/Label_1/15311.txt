#include <iostream>

using namespace std;

int main()
{
    int n;
    cin >> n;
    long long* arr = new long long[n];
    for (int i = 0; i < n; i++)
    {
        cin >> arr[i];
    }

    int ans = 0;
    long long sum = 0, mx = 0;
    for (int i = 0; i < n; i++)
    {
        sum += arr[i];
        if (sum < mx)
        {
            sum = mx;
        }
        if (sum >= 0)
        {
            ans++;
        }
        mx = max(mx, sum);
    }

    cout << ans << endl;

    return 0;
}
