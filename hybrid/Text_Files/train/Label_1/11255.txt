#include <iostream>
#include <set>
using namespace std;

int main() {
    int n, y;
    cin >> n;
    set<int> s;
    while (n--) { cin >> y; s.insert(y); }
    int a = *s.begin(), b = -1;
    for (int x : s) {
        if (a == x) {
            ++a;
            continue;
        }
        if (b == -1 || b - a > x - b) { b = x; a = *s.begin(); }
    }
    cout << b - (b - a) / 2 << endl;
    return 0;
}
