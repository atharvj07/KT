import java.util.*;
import java.io.*;

class Main {
    static long[] getNumbers( int m, int l ) {
        long[] result = new long[m + 1];
        result[0] = 0;
        for (int i = 1; i <= m; i++) {
            result[i] = 10 * result[i - 1] + (i);
        }
        long STEP_2 = result[m];
        result[m] = result[m - 1];
        STEP_2 -= result[m - 1];
        for (int i = m; i >= 2; i--) {
            long diff = STEP_2 * (l - 1);
            l--;
            STEP_2 -= diff;
            STEP_2 -= result[i - 1];
            result[i - 1] = STEP_2 + result[i - 1];
        }
        if (l == 1) {
            result[0] = result[0] + STEP_2;
        }
        return result;
    }

    public static void main( String[] args ) {
        Scanner scanner = new Scanner( System.in );
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        String s = scanner.next();
        int[] nums = new int[s.length()];
        for (int i = 0; i < s.length(); i++) {
            nums[i] = s.charAt( i ) - '0';
        }
        long MOD = 1000000007;
        if (k > n - 1) {
            System.out.println( 0 );
            return;
        }
        long[][] sums = new long[k + 1][n + 1];
        for (int i = 0; i <= k; i++) {
            for (int j = 0; j <= n; j++) {
                sums[i][j] = -1;
            }
        }
        long[] digits = getNumbers( n - 1, n - k - 1 );
        long answer = 0;
        for (int div = 0; div < n - k - 1; div++) {
            answer += (nums[div] * digits[div + 1]);
            answer %= MOD;
        }
        for (int i = n - k - 1; i < n; i++) {
            answer += nums[i];
            answer %= MOD;
        }
        System.out.println( answer );
    }
}
