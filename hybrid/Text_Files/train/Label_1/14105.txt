#include <bits/stdc++.h>
using namespace std;
const int maxn = 100;
int n, m;
int G[maxn][maxn];
vector<int> p[maxn];
vector<pair<int, int>> v[maxn];
int main() {
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        a--, b--;
        G[a][b] = G[b][a] = 1;
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (G[i][j]) {
                p[i].push_back(j);
                p[j].push_back(i);
            }
        }
    }
    int cur = 1;
    while (cur <= n) {
        for (int i = 0; i < n; i++) {
            if (v[i].empty()) {
                v[i].push_back({1, cur});
                for (int j : p[i]) {
                    v[j].push_back({1, cur});
                }
                cur++;
            }
        }
    }
    for (int i = 0; i < n; i++) {
        cout << v[i].size() << endl;
        for (auto x : v[i]) {
            cout << x.first << " " << x.second << endl;
        }
    }
    return 0;
}
