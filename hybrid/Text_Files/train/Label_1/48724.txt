import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        String ac = sc.next();
        String af = sc.next();
        int c = lcm(a, b);
        if ((c % a == 0 && af.substring(0, c / a).equals(ac)) || (c % b == 0 && ac.substring(0, c / b).equals(af)))
            System.out.println(c);
        else
            System.out.println(-1);
    }

    public static int gcd(int a, int b) {
        if (b == 0)
            return a;
        return gcd(b, a % b);
    }

    public static int lcm(int a, int b) {
        return a * b / gcd(a, b);
    }
}
