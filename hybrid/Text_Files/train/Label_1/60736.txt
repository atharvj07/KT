import sys
input = sys.stdin.readline
N, X = map(int,input().split())
lst = []
for _ in range(N):
    l, r, cost = map(int,input().split())
    if r-l+1 == X: lst.append([cost,l,r])
lst.sort()
ans = 1e9
L = []
for j in range(len(lst)):
    cost, l, r = lst[j]
    L.append([cost, l, r])
    for i in range(j):
        cost1, l1, r1 = L[i]
        if l1 > r or l > r1:
            ans = min(ans, cost+cost1)
if ans != 1e9: print(ans)
else: print(-1)
