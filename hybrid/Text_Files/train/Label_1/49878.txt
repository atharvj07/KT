import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int ans = n - 1;
        boolean[] vis = new boolean[n + 1];
        for (int i = 0; i < n; i++) {
            if (!vis[a[i]]) {
                ans--;
                vis[a[i]] = true;
            }
        }

        if (ans == 0) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }
}
