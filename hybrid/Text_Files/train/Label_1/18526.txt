#include <bits/stdc++.h>
using namespace std;
int main() {
  int tc;
  cin >> tc;
  while (tc--) {
    int n;
    cin >> n;
    array<int, 200001> f;
    fill(f.begin(), f.end(), 0);
    vector<int> v(n);
    int mx = 0;
    for (int i = 0; i < n; ++i) {
      cin >> v[i];
      f[v[i]]++;
      mx = max(mx, f[v[i]]);
    }
    int ans = 0;
    for (int i = 1; i <= n; ++i) {
      if (f[i] > 0) {
        if (mx == f[i])
          ans += f[i] / 2;
        else
          ans += min(2 * f[i], mx / 2);
      }
      ans = min(ans, n / 2);
    }
    cout << ans << endl;
  }
  return 0;
}
