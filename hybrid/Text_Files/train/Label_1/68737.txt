import sys
input = sys.stdin.readline

n,m = map(int,input().split())
a = [[] for i in range(n)]
for i in range(n-1):
    u,v = map(int,input().split())
    u-=1;v-=1
    a[u].append(v)
    a[v].append(u)
b = list(map(int,input().split()))
for i in range(m): b[i]-=1
ans = []
for c in b:
    x = []
    q = [(c,0)]
    vis = [0]*n
    vis[c] = 1
    while q:
        u,d = q.pop(0)
        x.append((d,u))
        for v in a[u]:
            if vis[v] == 0:
                q.append((v,d+1))
                vis[v] = 1
    x.sort()
    ans.append(x)
d = -1;t = n+1;c = -1
for i in range(m):
    for j in range(len(ans[i])):
        if ans[i][j][0] < t:
            t = ans[i][j][0]
            c = ans[i][j][1]
            d = (i+1)
print(c+1)
print(t)
