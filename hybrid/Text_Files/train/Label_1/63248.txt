import sys
input = sys.stdin.readline

n, k, x = map(int, input().split())
arr = list(map(int, input().split()))

# Sort the array
arr.sort()

# Check if it is possible to make all elements negative
neg = True
for e in arr:
    if e % x != 0:
        neg = False
        break

# If it is possible, make all elements negative
if neg:
    for i in range(n):
        arr[i] = -arr[i]

# Apply the operations
for i in range(min(k, n)):
    if arr[i] % x != 0:
        arr[i] -= x

# Print the result
print(*arr)
