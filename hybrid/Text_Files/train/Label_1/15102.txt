#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn=2e5+5;
const ll mod=1e9+7;
#define pb push_back
int n;
struct node
{
    int e,link;
    map<char,int> go;
};
node t[2*maxn];
int len[2*maxn],p[2*maxn];
char s[maxn];
void build_suffix_trie()
{
    int last=0;
    len[0]=0;
    for(int i=0;i<n;i++)
    {
        int c=s[i]-'a';
        int cur=last;
        while(cur&&!t[cur].go.count(c))
        {
            cur=t[cur].link;
        }
        if(!t[cur].go.count(c))
        {
            t[cur].go[c]=++last;
            len[last]=len[cur]+1;
            p[last]=cur;
            int u=cur;
            while(u&&p[u]!=u)
            {
                t[u].e+=len[u]-len[t[u].link];
                u=p[u];
            }
        }
        cur=t[cur].go[c];
        t[cur].e++;
        last=cur;
    }
}
#define fi first
#define se second
pair<pair<int,int>,pair<int,int> > f[maxn];
int32_t main()
{
    ios_base::sync_with_stdio(false),cin.tie(NULL);
    cin>>n;
    for(int i=0;i<n;i++)
    {
        cin>>(&s[0]);
        int m=strlen(s);
        for(int j=0;j<m;j++)
        {
            f[i].fi.se++;
            f[i].se.se+=len[j];
            f[i].fi.fi=max(f[i].fi.fi,len[j]);
            f[i].se.fi=max(f[i].se.fi,t[j].e);
        }
    }
    for(int i=0;i<n;i++)
    {
        cin>>(&s[0]);
        int m=strlen(s);
        for(int j=0;j<m;j++)
        {
            f[i+n].se.se+=len[j];
            f[i+n].se.fi=max(f[i+n].se.fi,t[j].e);
        }
    }
    pair<int,int> ans[maxn];
    int cur=0;
    for(int i=n-1;i>=0;i--)
    {
        vector<pair<int,pair<int,int> > > temp;
        for(int j=0;j<n;j++)
        {
            if(f[j].fi.fi+cur>=f[i+n].se.fi)
            {
                temp.pb({f[j].fi.se+f[i+n].se.se,
                         {j,i+n}});
            }
        }
        sort(temp.begin(),temp.end());
        ans[i]=temp[0].se;
        cur+=f[temp[0].se.fi].fi.fi;
    }
    cout<<cur<<"\n";
    for(int i=0;i<n;i++)
    {
        cout<<ans[i].fi+1<<" "<<ans[i].se-n+1<<"\n";
    }
}
