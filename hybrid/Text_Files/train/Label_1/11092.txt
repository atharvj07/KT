#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, a, m, x;
  cin >> n >> k >> a >> m;
  vector<bool> was(n + 1);
  for (int i = 0; i < m; i++) {
    cin >> x;
    if (was[x - a + 1] || was[x]) {
      cout << i + 1 << endl;
      return 0;
    }
    was[x] = 1;
  }
  cout << -1 << endl;
  return 0;
}
