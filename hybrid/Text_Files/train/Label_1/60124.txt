a, b = map(int, input().split())
c, d = map(int, input().split())

for i in range(1, 10001):
    if b + a * (i - 1) == d + c * (i - 1):
        print(b + a * (i - 1))
        break
    if b + a * (i - 1) > d + c * (i - 1):
        for j in range(1, 10001):
            if b + a * (i - 1) == d + c * (j - 1):
                print(b + a * (i - 1))
                exit(0)
        break
else:
    print(-1)
