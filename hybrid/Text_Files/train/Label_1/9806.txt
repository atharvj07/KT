#include <iostream>
#include <cstring>
#include <vector>
#include <sstream>
#include <ctime>
#include <cmath>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  string s;
  getline(cin, s);
  vector<pair<int, string>> logs;
  while (getline(cin, s)) {
    istringstream ss(s.substr(0, s.find(':')));
    int time = 0;
    for (int i = 0; i < 3; i++) {
      int t;
      ss >> t;
      time = time * 100 + t;
    }
    logs.push_back({time, s.substr(s.find(':') + 1)});
  }
  int cur = 0;
  for (int i = 0; i < min((int)logs.size(), n); i++) {
    cur += logs[i].second.size() + 1;
  }
  int ans = -1;
  for (int i = n; i < (int)logs.size(); i++) {
    cur -= logs[i - n].second.size() + 1;
    cur += logs[i].second.size() + 1;
    if (cur >= m) {
      ans = logs[i - n + 1].first;
      break;
    }
  }
  if (ans == -1) {
    cout << -1 << endl;
  } else {
    int h = ans / 10000, m = ans % 10000 / 100, s = ans % 100;
    cout << ans / 100000000 << '-' << h / 1000000 % 10 << h / 100000 % 10 << h / 10000 % 10 << h / 1000 % 10 << h / 100 % 10 << h / 10 % 10 << h % 10 << '-' << m / 10 % 10 << m % 10 << '-' << s / 10 % 10 << s % 10 << endl;
  }
  return 0;
}
