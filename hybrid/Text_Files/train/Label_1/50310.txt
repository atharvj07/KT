import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        double k = in.nextDouble();
        double d = in.nextDouble();
        double t = in.nextDouble();
        if (d <= k) {
            System.out.println(t);
        } else {
            System.out.println((t * k + (d - k) * 2 * t) / (k + d));
        }
    }
}
