import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[][] a = new int[n][m];
        int[] deg = new int[m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j) - '0';
                deg[j] += a[i][j];
            }
        }
        boolean ans = false;
        for (int i = 0; i < n; i++) {
            boolean ok = true;
            for (int j = 0; j < m; j++) {
                if (a[i][j] == 1 && deg[j] == 1) {
                    ok = false;
                    break;
                }
            }
            ans |= ok;
        }
        System.out.println(ans ? "YES" : "NO");
    }
}
