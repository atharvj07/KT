#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;
int f[N], minp[N];

vector<int> get(int n) {
  vector<int> ret;
  for (int i = 2; i <= n; i++) {
    if (minp[i] == i) {
      while (n % i == 0) {
        n /= i;
        ret.push_back(i);
      }
    }
  }
  if (n > 1) ret.push_back(n);
  return ret;
}

int main() {
  f[1] = 0;
  for (int i = 2; i < N; i++) {
    if (minp[i] == 0) {
      for (int j = i; j < N; j += i) minp[j] = i;
    }
    f[i] = f[i - 1];
    if (minp[i] == i) f[i]++;
  }
  int n;
  cin >> n;
  cout << f[n] << endl;
  auto v = get(n);
  for (auto x : v) cout << x << " ";
  return 0;
}
