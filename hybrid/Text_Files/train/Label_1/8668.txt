#include <bits/stdc++.h>
using namespace std;
int32_t main() {
    int n;
    cin >> n;
    vector<int64_t> v(n);
    for (auto &i : v) cin >> i;
    int64_t ans = 0, dec = 0, inc = 0;
    for (int i = 1; i < n; i++) {
        if (v[i] < v[i - 1]) dec += v[i - 1] - v[i];
        else inc += v[i] - v[i - 1];
    }
    ans = max(inc, dec);
    cout << ans << '\n';
    return 0;
}
