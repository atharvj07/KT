from collections import defaultdict
n, m = map(int, input().split())
p = list(map(int, input().split()))
a = defaultdict(list)
for i in range(m):
    x, y = map(int, input().split())
    a[x].append(y)
    a[y].append(x)
s = set()
ans = 0
for j in p:
    if j not in s:
        l = [j]
        while l:
            i = l.pop()
            if i in s: continue
            s.add(i)
            ans += i - 1 - len(a[i])
            for k in a[i]:
                if k not in s: l.append(k)
print(ans)
