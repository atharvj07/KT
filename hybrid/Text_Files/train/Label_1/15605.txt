#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> arr(n);
    for (int i = 0; i < n; i++) {
        std::cin >> arr[i];
    }

    sort(arr.begin(), arr.end());

    for (int i = 0; i < n - 2; i++) {
        if (arr[i + 1] - arr[i] > 2 || arr[i + 2] - arr[i] > 2) {
            std::cout << "NO";
            return 0;
        }
    }

    std::cout << "YES";
    return 0;
}
