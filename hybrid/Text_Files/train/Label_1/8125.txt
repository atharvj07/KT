#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    if (a[i] < 3) {
      ans += 3 - a[i];
    }
  }
  if (ans % 4 == 0) {
    std::cout << ans / 4;
  } else {
    std::cout << "-1";
  }
  return 0;
}
