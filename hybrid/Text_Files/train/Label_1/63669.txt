n, m = map(int, input().split())
a = list(map(int, input().split()))
se = []
for i in range(m):
  l, r = map(int, input().split())
  se.append((l, r))
a.sort()
se.sort(key = lambda x : x[1])
d = 0
b = []
for l, r in se:
  while a and a[-1] >= r:
    a.pop()
  if not a or a[-1] < l:
    continue
  d += a.pop() - l + 1
  b.append((l, r))
print(d)
print(len(b))
for l, r in b:
  print(l, r)
