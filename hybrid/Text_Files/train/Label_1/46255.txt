import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    PriorityQueue<Integer> pq = new PriorityQueue<>();
    for (int i = 0; i < n; i++) {
      int p = sc.nextInt();
      int c = sc.nextInt();
      if (p != 1) pq.add(c);
    }
    int ans = 0;
    while (!pq.isEmpty() && pq.peek() <= ans) pq.poll();
    while (!pq.isEmpty()) {
      ans += pq.poll();
      n -= 2;
    }
    System.out.println(ans);
  }
}
