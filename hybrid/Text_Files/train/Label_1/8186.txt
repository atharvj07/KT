#include <cstdio>
#include <vector>
using namespace std;

struct Node {
    int parent;
    int depth;
    int subTreeSize;
    int isLeaf;
};

const int MAX_N = 200000;
vector<Node> nodes;
vector<vector<int>> graph;
int n;

int addEdge(int u, int v) {
    graph[u].emplace_back(v);
    graph[v].emplace_back(u);
}

void dfs1(int node, int depth, int parent) {
    nodes[node].parent = parent;
    nodes[node].depth = depth;
    nodes[node].subTreeSize = 1;
    nodes[node].isLeaf = true;

    for (int u : graph[node]) {
        if (parent != u) {
            dfs1(u, depth + 1, node);
            nodes[node].isLeaf = false;
            nodes[node].subTreeSize += nodes[u].subTreeSize;
        }
    }
}

void dfs2(int node, int parent) {
    for (int u : graph[node]) {
        if (u != parent) {
            nodes[u].depth = nodes[node].depth + 1;
            if (nodes[node].isLeaf)
                nodes[node].depth++;
            nodes[u].subTreeSize = n - nodes[u].subTreeSize;
            dfs2(u, node);
        }
    }
}

long long sum;

void dfs3(int node, int parent) {
    sum += 1LL * nodes[node].depth * (n - nodes[node].subTreeSize);
    for (int u : graph[node]) {
        if (u != parent) {
            dfs3(u, node);
        }
    }
}

int main() {
    scanf("%d", &n);
    nodes.resize(n + 1);
    graph.resize(n + 1);

    for (int i = 0; i < n - 1; i++) {
        int u, v;
        scanf("%d%d", &u, &v);
        addEdge(u, v);
    }

    dfs1(1, 0, -1);
    dfs2(1, -1);
    dfs3(1, -1);
    printf("%lld\n", sum);
    return 0;
}
