import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] ink = new long[n + 1];
        for (int i = 1; i <= n; i++) ink[i] = sc.nextLong();
        long days = (n + 5) / 7;
        long minInk = Long.MAX_VALUE;
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            long curInk = ink[i] - days;
            if (curInk < minInk) {
                minInk = curInk;
                ans = i;
            }
        }
        System.out.println(ans);
    }
}
