import sys
input = sys.stdin.readline
p, q, l, r = map(int, input().rstrip().split())
a = []
b = []
for _ in range(p):
    x, y = map(int, input().rstrip().split())
    a.append(x)
    b.append(y)
c = []
d = []
for _ in range(q):
    x, y = map(int, input().rstrip().split())
    c.append(x)
    d.append(y)
ans = 0
for i in range(l, r+1):
    f = 0
    for j in range(p):
        if i >= a[j] and i <= b[j]:
            f = 1
            break
    for j in range(q):
        if i >= c[j] and i <= d[j]:
            f = 1
            break
    if f: ans += 1
print(ans)
