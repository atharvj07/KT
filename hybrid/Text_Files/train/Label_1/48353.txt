import java.util.Scanner;

public class Main {

    static final long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long ans;
        if (n == 0) {
            ans = 1;
        } else if (n == 1) {
            ans = 2;
        } else {
            if (n % 4 == 0) {
                n--;
            }
            if (n % 4 == 1) {
                ans = (n + 4) * (n + 3) / 4;
            } else if (n % 4 == 2) {
                ans = n * (n + 1) / 2;
            } else {
                ans = n * (n + 2) / 4;
            }
        }
        System.out.println(ans % MOD);
    }
}
