#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        vector<int> ans(n);
        int sum = 0;
        for (int i = n - 1; i >= 0; i--) {
            sum += a[i];
            if (sum > 0) ans[i] = 1, sum--;
        }
        for (int i = 0; i < n; i++) { cout << ans[i] << ' '; }
        cout << '\n';
    }
    return 0;
}
