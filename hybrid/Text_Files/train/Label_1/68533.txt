ticket = input().strip()
first_half_sum = sum(map(int, ticket[:3]))
second_half_sum = sum(map(int, ticket[3:]))
diff = abs(first_half_sum - second_half_sum)

if diff == 0:
    print(0)
else:
    changes = 0
    for digit in sorted(ticket[:3] + ticket[3:], key=int, reverse=True):
        if diff <= 0:
            break
        diff -= 9 - int(digit)
        changes += 1
    print(changes)
