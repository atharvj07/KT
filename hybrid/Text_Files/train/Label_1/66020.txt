for _ in range(int(input())):
    n, d = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort()
    ans = "NO"
    for l in range(n):
        for r in range(n):
            if l == r:
                continue
            if a[l] + a[r] <= d:
                ans = "YES"
                break
        if ans == "YES":
            break
    print(ans)
