#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<string> board(n);
    for (int i = 0; i < n; i++) cin >> board[i];
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (board[i][j] == '.') {
                if (j >= 1 && board[i][j - 1] != board[i][j])
                    cout << (board[i][j - 1] == 'B' ? 'W' : 'B');
                else if (i >= 1 && board[i - 1][j] != board[i][j])
                    cout << (board[i - 1][j] == 'B' ? 'W' : 'B');
                else if (j < m - 1 && board[i][j + 1] != board[i][j])
                    cout << (board[i][j + 1] == 'B' ? 'W' : 'B');
                else if (i < n - 1 && board[i + 1][j] != board[i][j])
                    cout << (board[i + 1][j] == 'B' ? 'W' : 'B');
                else if (j >= 2 && board[i][j - 2] != board[i][j])
                    cout << (board[i][j - 2] == 'B' ? 'W' : 'B');
                else if (i >= 2 && board[i - 2][j] != board[i][j])
                    cout << (board[i - 2][j] == 'B' ? 'W' : 'B');
                else if (j < m - 2 && board[i][j + 2] != board[i][j])
                    cout << (board[i][j + 2] == 'B' ? 'W' : 'B');
                else if (i < n - 2 && board[i + 2][j] != board[i][j])
                    cout << (board[i + 2][j] == 'B' ? 'W' : 'B');
                else if (j >= 3 && board[i][j - 3] != board[i][j])
                    cout << (board[i][j - 3] == 'B' ? 'W' : 'B');
                else if (i >= 3 && board[i - 3][j] != board[i][j])
                    cout << (board[i - 3][j] == 'B' ? 'W' : 'B');
                else if (j < m - 3 && board[i][j + 3] != board[i][j])
                    cout << (board[i][j + 3] == 'B' ? 'W' : 'B');
                else if (i < n - 3 && board[i + 3][j] != board[i][j])
                    cout << (board[i + 3][j] == 'B' ? 'W' : 'B');
            } else
                cout << '-';
        }
        cout << '\n';
    }
    return 0;
}
