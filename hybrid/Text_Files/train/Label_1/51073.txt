import java.util.*;

public class Main {
    static int[] x, y;
    static int n;
    static long mod = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        x = new int[n];
        y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
        }
        long ans = cut(0, n - 1, 0, 0);
        System.out.println(ans);
    }

    static long cut(int l, int r, int L, int R) {
        if (l > r) return 0;
        int best = -1;
        for (int i = l; i <= r; i++) {
            if (inTriangle(L, R, i)) best = i;
        }
        if (best == -1) return 1;
        long ans = 0;
        ans += cut(l, best - 1, L, best);
        ans += cut(best + 1, r, best, R);
        return ans % mod;
    }

    static boolean inTriangle(int L, int R, int i) {
        if ((cross(x[i], y[i], x[L], y[L], x[R], y[R]) > 0) || (cross(x[L], y[L], x[R], y[R], x[i], y[i]) > 0))
            return false;
        if ((cross(x[R], y[R], x[i], y[i], x[L], y[L]) > 0) || (cross(x[i], y[i], x[L], y[L], x[R], y[R]) > 0))
            return false;
        return true;
    }

    static long cross(int x1, int y1, int x2, int y2, int x3, int y3) {
        return ((long) x1 * (y2 - y3) + (long) x2 * (y3 - y1) + (long) x3 * (y1 - y2));
    }
}
