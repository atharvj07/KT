import queue
n, k = map(int, input().split())
p = list(map(int, input().split()))
q = queue.PriorityQueue()
for i in range(n):
    q.put((int(p[i]), -i))
lst = []
ans, cnt = 0, 0
while not q.empty():
    power, idx = q.get()
    lst.append(-idx)
    cnt += 1
    if power > ans:
        ans, cnt = power, 1
    if cnt == k:
        break
print(ans)
