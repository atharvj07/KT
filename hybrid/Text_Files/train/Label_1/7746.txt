#include <bits/stdc++.h>
using namespace std;
int n, t, a[30005], v[30005];
int main() {
  cin >> n >> t;
  for (int i = 0; i <= n; i++) v[i] = -1;
  for (int i = 1; i < n; i++) {
    cin >> a[i];
    if (i + a[i] <= n) v[i + a[i]] = max(v[i + a[i]], i);
  }
  cout << ((v[t] == -1) ? "NO" : "YES") << endl;
}
