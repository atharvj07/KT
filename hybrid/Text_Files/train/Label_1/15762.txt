#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;
const int inf = 1e9;
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n, m;
  cin >> n >> m;
  vector<int> a(n + 1);
  for (int i = 1; i <= n; i++) cin >> a[i];
  vector<int> f(n + 1);
  for (int i = 1; i <= n; i++) f[i] = a[i] * i;
  sort(a.begin(), a.end());
  const int mxn = 2e5;
  const int M = mxn + 5;
  vector<int> fw(M);
  auto upd = [&](int p, int v) {
    for (; p < M; p += p & -p) fw[p] = min(fw[p], v);
  };
  auto get = [&](int p) {
    int ans = inf;
    for (; p; p -= p & -p) ans = min(ans, fw[p]);
    return ans;
  };
  for (int i = m + 1; i <= n; i++) {
    fw[i] = f[i];
    int r = i - m;
    upd(r, f[i - m]);
  }
  for (int i = m; i >= 1; i--) {
    int r = i - m + 1;
    f[i] += get(r);
    upd(i, f[i]);
  }
  for (int i = 1; i <= n; i++) cout << f[i] << " ";
  cout << "\n";
  return 0;
}
