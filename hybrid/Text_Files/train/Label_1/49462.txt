import java.util.Scanner;

public class Main {
    static final int MAX_K = 2;
    static final int MAX_N = 100000;

    static long[] length = new long[MAX_N];
    static long[] width = new long[MAX_N];
    static long[] height = new long[MAX_N];
    static long[] volume = new long[MAX_N];

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        for (int i = 0; i < n; i++) {
            length[i] = scanner.nextInt();
            width[i] = scanner.nextInt();
            height[i] = scanner.nextInt();
            volume[i] = length[i] * width[i] * height[i];
        }

        long maxVolume = 0;
        int index1 = -1;
        int index2 = -1;

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (length[i] == length[j] && width[i] == width[j]) {
                    if (height[i] + height[j] > maxVolume) {
                        maxVolume = height[i] + height[j];
                        index1 = i;
                        index2 = j;
                    }
                } else if (length[i] == length[j] && height[i] == height[j]) {
                    if (width[i] + width[j] > maxVolume) {
                        maxVolume = width[i] + width[j];
                        index1 = i;
                        index2 = j;
                    }
                } else if (width[i] == width[j] && height[i] == height[j]) {
                    if (length[i] + length[j] > maxVolume) {
                        maxVolume = length[i] + length[j];
                        index1 = i;
                        index2 = j;
                    }
                }
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (length[i] == length[j] && width[i] == width[j] && height[i] == height[j]) {
                    if (volume[i] + volume[j] > maxVolume) {
                        maxVolume = volume[i] + volume[j];
                        index1 = i;
                        index2 = j;
                    }
                }
            }
        }

        if (index1 != -1 && index2 != -1) {
            System.out.println(2);
            System.out.println(index1 + 1);
            System.out.println(index2 + 1);
        } else {
            long maxValue = 0;
            int maxIndex = -1;

            for (int i = 0; i < n; i++) {
                if (volume[i] > maxValue) {
                    maxValue = volume[i];
                    maxIndex = i;
                }
            }

            System.out.println(1);
            System.out.println(maxIndex + 1);
        }
    }
}
