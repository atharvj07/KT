#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  int c = 0, b = 0, ba = 0;
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    if (i % 3 == 0)
      c += a;
    else if (i % 3 == 1)
      b += a;
    else
      ba += a;
  }
  vector<int> ans = {c, b, ba};
  auto m = max_element(ans.begin(), ans.end());
  if (*m == c)
    cout << "chest";
  else if (*m == b)
    cout << "biceps";
  else
    cout << "back";
}
