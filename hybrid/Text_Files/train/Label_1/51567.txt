import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long k = sc.nextLong();
        long p = sc.nextLong();
        long[] queries = new long[(int) p];
        for (int i = 0; i < p; i++) {
            queries[i] = sc.nextLong() - 1;
        }
        long[] ans = solve(n, k, queries);
        for (long a : ans) {
            System.out.print(a == 1 ? "X" : ".");
        }
    }

    public static long[] solve(long n, long k, long[] queries) {
        long[] ans = new long[(int) n];
        long[] count = new long[(int) n];
        for (long q : queries) {
            count[(int) q]++;
        }
        long t = k;
        for (int i = 0; i < n; i++) {
            if (count[i] > 0) {
                t--;
                ans[i] = 1;
                if (t == 0) {
                    break;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            if (t > 0) {
                ans[i] = 1;
                t--;
            }
        }
        return ans;
    }
}
