#include<bits/stdc++.h>
using namespace std;
int n,i,j,k,l,vis[100001];
string a[100001],b[100001];
int main(){
  cin>>n>>k;
  for(i=0;i<n;i++){
    cin>>a[i];
    b[i]=a[i]+'$';
  }
  sort(b,b+n);
  k--;
  for(i=0;i<n;i++){
    l=a[i].length();
    int p=lower_bound(b+i,b+n,a[i]+'#')-b;
    if(!vis[p]){
      j++;
      vis[p]=1;
    }
  }
  if((k-j)%2==1)printf("First\n");
  else printf("Second\n");
  return 0;
}
