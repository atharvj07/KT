#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
  ll a, b;
  cin >> a >> b;
  ll gcd = 1;
  for (ll i = 2; i <= 12; i++) {
    if (a % i == 0 && b % i == 0) {
      gcd *= i;
      a /= i;
      b /= i;
      i--;
    } else if (a % i == 0) {
      while (a % i == 0) {
        gcd *= i;
        a /= i;
      }
    } else if (b % i == 0) {
      while (b % i == 0) {
        gcd *= i;
        b /= i;
      }
    }
  }
  cout << gcd << endl;
  return 0;
}
