#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
#define f first
#define s second
#define all(v) (v).begin(), (v).end()
#define pb push_back
const int oo = (int)1e9;
int dx[] = {1, 0, -1, 0}, dy[] = {0, 1, 0, -1};
const string NO = "NO", YES = "YES";

struct Package {
    ll x, y;
    bool operator<(const Package& rhs) const {
        return (x == rhs.x) ? y < rhs.y : x < rhs.x;
    }
    friend istream& operator>>(istream& in, Package& p) {
        in >> p.x >> p.y;
        return in;
    }
};

int32_t main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<Package> packages(n);
        for (auto& pkg : packages) {
            cin >> pkg;
        }
        sort(all(packages));
        ll cur_x = 0, cur_y = 0;
        string path = "";
        bool is_impossible = 0;
        for (const auto& pkg : packages) {
            while (cur_x < pkg.x) {
                cur_x++;
                path += 'R';
            }
            while (cur_y < pkg.y) {
                cur_y++;
                path += 'U';
            }
            if (cur_x != pkg.x || cur_y != pkg.y) {
                is_impossible = 1;
                break;
            }
        }
        cout << (is_impossible ? NO : YES) << '\n';
        if (!is_impossible)
            cout << path << '\n';
    }
    return 0;
}
