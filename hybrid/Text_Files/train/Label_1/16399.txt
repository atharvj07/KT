#include <iostream>
#include <vector>
#include <limits>

using namespace std;

typedef long long ll;

struct Edge {
    ll to, cost;
    Edge(ll to, ll cost) : to(to), cost(cost) {}
};

const ll INF = numeric_limits<ll>::max() / 2;

int main() {
    ll n, m;
    cin >> n >> m;
    vector<ll> a(n);
    for (ll& ai : a) cin >> ai;
    
    vector<vector<Edge>> adj(n);
    for (ll i = 0; i < m; i++) {
        ll x, y, w;
        cin >> x >> y >> w;
        x--, y--;
        adj[x].emplace_back(y, w);
        adj[y].emplace_back(x, w);
    }
    
    vector<ll> dist(n, INF);
    vector<bool> vis(n, false);
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
        if (!vis[i]) {
            vis[i] = true;
            ll tot = 0;
            for (const Edge& e : adj[i]) {
                if (!vis[e.to]) {
                    tot += min(e.cost, a[e.to]);
                }
            }
            ans += min(tot, a[i]);
        }
    }
    cout << ans << endl;
    return 0;
}
