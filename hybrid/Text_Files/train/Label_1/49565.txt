import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Set<Integer>[] wins = new HashSet[n + 1];
        for (int i = 1; i <= n; i++) {
            wins[i] = new HashSet<>();
        }
        for (int i = 0; i < n * (n - 1) / 2 - 1; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            wins[x].add(y);
        }
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (!wins[i].contains(j) && !wins[j].contains(i)) {
                    System.out.println(i + " " + j);
                    return;
                }
            }
        }
    }
}
