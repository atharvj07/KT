#include<bits/stdc++.h>
using namespace std;
#define int long long
#define endl "\n"
#define IOS ios_base::sync_with_stdio(false);cin.tie(NULL);
#define mod 998244353

int fpow(int base, int p) {
	if (p == 0) return 1;
	if (p == 1) return base;
	int ret = fpow(base, p / 2);
	ret = (1LL * ret * ret) % mod;
	if (p % 2) ret = (1LL * ret * base) % mod;
	return ret;
}

int n, f[200005][2], pref[200005], val[200005];
int32_t main() {
	IOS;
	int i, j, x;
	cin >> n;
	map<int, int> mp;
	for (i = 1; i <= n; i++) {
		cin >> x;
		val[i] = x;
		if (!mp.count(x)) mp[x] = mp.size();
		x = mp[x];
		pref[i] = pref[i - 1];
		if (f[x][0] == 0) f[x][0] = i, pref[i]++;
		f[x][1] = i;
	}
	f[0][1] = n + 1;
	int ans = 1;
	for (i = 1; i <= mp.size(); i++) {
		int st = f[i][0], en = f[i][1];
		ans = (ans * fpow(2, pref[en] - pref[st])) % mod;
	}
	cout << ans << endl;
	return 0;
}
