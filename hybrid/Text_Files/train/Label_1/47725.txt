import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int planets = input.nextInt();
        double payload = input.nextDouble();
        int[] takeoff = new int[planets];
        int[] landing = new int[planets];
        for (int i = 0; i < planets; i++) takeoff[i] = input.nextInt();
        for (int i = 0; i < planets; i++) landing[i] = input.nextInt();

        double fuel = 0;
        for (int i = 1; i < planets - 1; i++) fuel += (payload + fuel) / takeoff[i];
        for (int i = planets - 1; i >= 1; i--) fuel += (payload + fuel) / landing[i];

        System.out.println((fuel < 1e9) ? String.format("%.6f", fuel) : -1);
    }
}
