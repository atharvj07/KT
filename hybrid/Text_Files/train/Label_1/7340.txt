#include <iostream>
#include <stack>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    stack<char> st;
    int ans = 0;
    for (char c : s) {
      if (c == '(' || c == '[') {
        st.push(c);
      } else {
        if (!st.empty()) {
          if ((c == ')' && st.top() == '(') || (c == ']' && st.top() == '[')) {
            st.pop();
            ans++;
          }
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
