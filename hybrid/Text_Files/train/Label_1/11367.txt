#include<bits/stdc++.h>
using namespace std;
#define maxn 200020
#define mod 1000003
#define LL long long
int n,c;
LL fastpow(int base, int p){
    if(p==0)return 1;
    LL ret=fastpow(base,p/2);
    ret*=ret;ret%=mod;
    if(p&1)ret*=base,ret%=mod;
    return ret;
}
vector<LL>dp(maxn);
int main(){
    dp[0]=1,dp[1]=1,dp[2]=2;
    for(int i=3; i<maxn; i++)dp[i]=(dp[i-1]+dp[i-2])%mod;
    for(int i=0; i<maxn; i++)dp[i]=fastpow(dp[i],c);dp[0]=0;
    for(int i=1; i<maxn; i++)dp[i]=(dp[i]+dp[i-1])%mod;
    int test;
    cin>>test;
    while(test--){
        scanf("%d %d",&n,&c);
        if(n==1){printf("1\n");continue;}
        printf("%lld\n",(dp[n]-dp[n-2]+mod)%mod);
    }
    return 0;
}
