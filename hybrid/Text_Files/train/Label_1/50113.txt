import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int r = 0, d = 0;
    while (scanner.hasNextLine()) {
      int[] a = Arrays.stream(scanner.nextLine().split(",")).mapToInt(Integer::parseInt).toArray();
      if (a[0] == a[1])
        d++;
      else if (a[0] + a[1] == a[2])
        r++;
    }
    System.out.println(r + "\n" + d);
  }
}
