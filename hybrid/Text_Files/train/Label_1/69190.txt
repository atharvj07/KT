n, m = map(int, input().split())
A = [list(map(int, input().split())) for _ in range(n)]
B = [list(map(int, input().split())) for _ in range(n)]

for i in range(n):
    for j in range(m):
        if A[i][j] != B[i][j]:
            if (i, j) not in [(x, y) for x in range(n) for y in range(m) if A[x][y] == B[i][j] and B[x][y] == A[i][j]]:
                print("NO")
                exit()
print("YES")
