import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] h = new int[6];
        int[] w = new int[6];
        for (int i = 0; i < 6; i++) {
            h[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }
        Arrays.sort(h);
        Arrays.sort(w);
        if (h[0] == h[1] && h[0] == h[2] && w[0] == w[1] && w[0] == w[2]) {
            System.out.println("yes");
        } else if (h[0] == h[1] && h[0] == h[3] && w[0] == w[1] && w[0] == w[4]) {
            System.out.println("yes");
        } else if (h[0] == h[1] && h[0] == h[4] && w[0] == w[1] && w[0] == w[5]) {
            System.out.println("yes");
        } else if (h[0] == h[2] && h[0] == h[3] && w[0] == w[2] && w[0] == w[6]) {
            System.out.println("yes");
        } else if (h[0] == h[2] && h[0] == h[4] && w[0] == w[2] && w[0] == w[5]) {
            System.out.println("yes");
        } else if (h[0] == h[3] && h[0] == h[4] && w[0] == w[3] && w[0] == w[5]) {
            System.out.println("yes");
        } else if (h[1] == h[2] && h[1] == h[3] && w[1] == w[2] && w[1] == w[6]) {
            System.out.println("yes");
        } else if (h[1] == h[2] && h[1] == h[4] && w[1] == w[2] && w[1] == w[5]) {
            System.out.println("yes");
        } else if (h[1] == h[3] && h[1] == h[4] && w[1] == w[3] && w[1] == w[5]) {
            System.out.println("yes");
        } else if (h[2] == h[3] && h[2] == h[4] && w[2] == w[3] && w[2] == w[5]) {
            System.out.println("yes");
        } else {
            System.out.println("no");
        }
    }
}
