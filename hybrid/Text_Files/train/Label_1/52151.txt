import java.util.*;
import java.io.*;

class Main {
    static class Vertex {
        int parent;
        int c;
        ArrayList<Integer> adj = new ArrayList<>();
        
        public Vertex(int parent, int c) {
            this.parent = parent;
            this.c = c;
        }
    }
    
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        Vertex[] vertices = new Vertex[n + 1];
        for (int i = 1; i <= n; i++) {
            String[] line = br.readLine().split(" ");
            vertices[i] = new Vertex(Integer.parseInt(line[0]), Integer.parseInt(line[1]));
        }
        
        for (int i = 1; i <= n; i++) {
            int p = vertices[i].parent;
            if (p > 0) vertices[p].adj.add(i);
        }
        
        if (!checkTree(vertices)) {
            System.out.println("NO");
            return;
        }
        
        int[] res = dfs(1, 1, vertices);
        if (res == null) {
            System.out.println("NO");
            return;
        }
        
        System.out.println("YES");
        for (int i : res) System.out.print(i + " ");
        System.out.println();
    }
    
    static int[] dfs(int cur, int val, Vertex[] vertices) {
        int[] res = new int[vertices.length];
        res[cur] = val;
        for (int next : vertices[cur].adj) {
            int[] r = dfs(next, val + 1, vertices);
            if (r == null) return null;
            
            res = merge(res, r);
            if (res == null) return null;
        }
        
        if (vertices[cur].c != res[cur] - 1) return null;
        return res;
    }
    
    static int[] merge(int[] a, int[] b) {
        if (a[0] == 0 || b[0] == 0) return null;
        
        for (int i = 1; i < a.length; i++) {
            if (a[i] != b[i]) return null;
        }
        
        return a;
    }
    
    static boolean checkTree(Vertex[] vertices) {
        int[] cnt = new int[vertices.length];
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        cnt[1] = 1;
        
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (int next : vertices[cur].adj) {
                if (cnt[next] > 0) return false;
                
                cnt[next] = cnt[cur] + 1;
                q.add(next);
            }
        }
        
        for (int i = 1; i < cnt.length; i++) {
            if (cnt[i] == 0) return false;
        }
        
        return true;
    }
}
