#include<bits/stdc++.h>
using namespace std;
#define ll long long int

int main()
{
    int n, m, k, s;
    cin >> n >> m >> k >> s;
    vector<vector<ll>> v(n, vector<ll>(m));
    vector<int> a(s);

    for(int i = 0; i < n; i++){
        for(int j = 0; j < m; j++){
            cin >> v[i][j];
        }
    }

    for(int i = 0; i < s; i++) cin >> a[i];

    vector<pair<ll, ll>> ans;
    for(int i = 0; i < s; i++){
        for(int j = 0; j < n; j++){
            for(int l = 0; l < m; l++){
                if(v[j][l] == a[i]){
                    ans.push_back({j + 1, l + 1});
                }
            }
        }
    }

    int max_cost = 0;
    for(int i = 1; i < s; i++){
        int cost = abs(ans[i].first - ans[i - 1].first) + abs(ans[i].second - ans[i - 1].second);
        max_cost = max(max_cost, cost);
    }

    cout << max_cost << endl;

    return 0;
}
