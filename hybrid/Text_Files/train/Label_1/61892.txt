# Sides of the square parallel to the coordinate axes.
x, y, a, b, c, d, e, f = map(int, input().split())
a -= x
b -= y
c -= x
d -= y
e -= x
f -= y

# Sides of the square at 45 degrees.
x1, y1, a1, b1, c1, d1, e1, f1 = map(int, input().split())
a1 -= x1
b1 -= y1
c1 -= x1
d1 -= y1
e1 -= x1
f1 -= y1

# Check if squares intersect.
if a <= a1 <= e or x <= e1 <= c or b <= b1 <= d or y <= d1 <= f or a <= e1 <= e or x <= c1 <= c or b <= d1 <= d or y <= f1 <= f or a1 <= a <= e1 or x1 <= c <= e or b1 <= b <= d1 or y1 <= f <= f or a1 <= e <= e1 or x1 <= c <= c1 or b1 <= d <= d1 or y1 <= f <= f1:
    print("YES")
else:
    print("NO")
