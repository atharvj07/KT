#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (auto& p : v) cin >> p.first >> p.second;
    sort(v.begin(), v.end(), [](const auto& p1, const auto& p2) { return p1.second < p2.second; });
    int ans = 1, last = v[0].second;
    for (int i = 1; i < n; i++) {
        if (v[i].first >= last) {
            ans++;
            last = v[i].second;
        }
    }
    cout << ans << endl;
    return 0;
}
