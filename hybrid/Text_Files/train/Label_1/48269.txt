import java.util.*;
import java.io.*;

public class Main{
    public static void main(String[]  args){
        Scanner sc = new Scanner(System.in);
        long l = sc.nextLong();
        long r = sc.nextLong();
        System.out.println((r-l+1) - (int)Math.floor(Math.log10(r) + 1) + (int)Math.floor(Math.log10(l) + 1));
    }
}
