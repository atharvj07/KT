n, p, q, r = map(int, input().split())
a = list(map(int, input().split()))
sum = 0
ans = -2e9
for i in range(n):
    sum += a[i] * p
    for j in range(i, n):
        sum += a[j] * q
        for k in range(j, n):
            sum += a[k] * r
            ans = max(ans, sum)
        sum -= a[j] * r
    sum -= a[i] * p
print(ans)
