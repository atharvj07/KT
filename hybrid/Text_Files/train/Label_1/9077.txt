using namespace std;
using ll = long long;

template <typename T>
inline void print(const vector<T> &arr) {
  for (const auto &i : arr) cout << i << ' ';
  cout << '\n';
}

struct ipv4 {
  uint32_t ip;
  uint32_t mask;
  ipv4() : ip(0), mask(0) {}
  ipv4(uint32_t ip, uint32_t mask) : ip(ip), mask(mask) {}
  ipv4(const string &s) {
    size_t found = s.find('/');
    if (found != string::npos) {
      ip = stoi(s.substr(0, found));
      mask = 32 - stoi(s.substr(found + 1));
    } else {
      ip = stoi(s);
      mask = 32;
    }
  }
  bool operator<(const ipv4 &other) const { return ip < other.ip; }
};

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<ipv4> wlist, blist;
  for (int i = 0; i < n; i++) {
    string subnet;
    cin >> subnet;
    ipv4 ipv(subnet);
    if (subnet[0] == '-') {
      blist.emplace_back(ipv);
    } else {
      wlist.emplace_back(ipv);
    }
  }
  sort(wlist.begin(), wlist.end());
  vector<ipv4> opt;
  uint32_t prev_end = 0;
  for (auto &ip : blist) {
    uint32_t start = ip.ip & (uint32_t)(~0 << ip.mask);
    if (start > prev_end) {
      for (auto it = lower_bound(wlist.begin(), wlist.end(), ipv4(start, 32));
           it != wlist.end(); it++) {
        uint32_t end = it->ip | ((1 << it->mask) - 1);
        if (start <= end) {
          prev_end = end;
          start = end + 1;
          it = lower_bound(it, wlist.end(), ipv4(start, 32));
          if (it == wlist.end()) break;
        }
      }
      if (start > prev_end) {
        opt.emplace_back(ip);
        prev_end = start | ((1 << ip.mask) - 1);
      }
    }
  }
  if (opt.size() == blist.size()) {
    cout << opt.size() << '\n';
    for (auto &i : opt) cout << i.ip << '/' << 32 - i.mask << '\n';
  } else {
    cout << -1 << '\n';
  }
  return 0;
}
