#include <vector>
#include <iostream>

using namespace std;

vector<int> g[30001];
bool visited[30001];

int main() {
  int n, t;
  cin >> n >> t;
  for (int i = 0; i < n - 1; i++) {
    int a;
    cin >> a;
    g[i + 1].push_back(i + 1 + a);
  }
  visited[1] = true;
  for (int i = 1; i < n; i++) {
    if (!visited[i]) continue;
    for (int j : g[i]) {
      visited[j] = true;
    }
  }
  if (visited[t]) cout << "YES" << endl;
  else cout << "NO" << endl;
  return 0;
}
