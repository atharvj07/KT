class Main {

        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), tot = 0, bonus = 0, ans = 0;
        while (n-- > 0) {
            int a = in.nextInt();
            if (bonus >= a) {
                bonus -= a;
