#include <bits/stdc++.h>
#define ll long long
#define vi vector<ll>
#define pii pair<ll, ll>
#define mp make_pair
#define pb push_back
#define MOD 998244353
using namespace std;

ll fastpow(ll x, ll y) {
    ll ans = 1;
    while (y) {
        if (y % 2) {
            ans = (ans * x) % MOD;
        }
        y /= 2;
        x = (x * x) % MOD;
    }
    return ans;
}

int main() {
    ll n;
    cin >> n;
    map<ll, ll> m;
    ll sum_p = 0;
    for (ll i = 0; i < n; i++) {
        ll x;
        cin >> x;
        m[x]++;
        sum_p += x;
    }
    if (n == 1) {
        return cout << 2 << endl, 0;
    }
    ll com = sum_p * fastpow(100 - sum_p, MOD - 2);
    com %= MOD;
    cout << (com + fastpow(100, MOD - 2) - sum_p % MOD * fastpow(100, MOD - 2) % MOD + MOD) %
             MOD
         << endl;
    return 0;
}
