n = int(input())
snacks = list(map(int, input().split()))

waiting = [False] * (n + 1)
next_snack = n

for snack in snacks:
    waiting[snack] = True
    if snack == next_snack:
        while waiting[next_snack]:
            print(next_snack, end=' ')
            waiting[next_snack] = False
            next_snack -= 1
        print()
    else:
        print()
