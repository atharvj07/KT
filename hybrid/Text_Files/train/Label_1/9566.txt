#include <iostream>
#include <utility>
#include <vector>

using namespace std;

typedef long long ll;
typedef pair<ll, ll> pll;

const ll mod = 1e9 + 7;
const int maxN = 2e5 + 5;
int n;
int failure[maxN];
char str[maxN];
vector<pll> pi[4];

void calcFailure() {
  failure[0] = 0;
  for (int i = 1, j = 0; i < n; ++i) {
    if (str[i] == str[j])
      failure[i] = ++j;
    else if (j > 0) {
      j = failure[j - 1];
      --i;
    }
  }
}

void calcPI() {
  for (int i = 0; i < 4; ++i) pi[i].resize(n);

  for (int i = 0, j = 0; i < n; ++i) {
    while (j > 0 && str[i] != str[(i + j) % n]) j = failure[j - 1];
    if (str[i] == str[(i + j) % n]) ++j;
    if (j == i + 1) {
      int c;
      if (i - j + 1 < n)
        c = str[j];
      else
        c = str[j % (i - j + 1)];
      switch (c) {
        case 'A':
          pi[0][i] = {j, (i - j + 1) % n};
          break;
        case 'C':
          pi[1][i] = {j, (i - j + 1) % n};
          break;
        case 'G':
          pi[2][i] = {j, (i - j + 1) % n};
          break;
        case 'T':
          pi[3][i] = {j, (i - j + 1) % n};
          break;
      }
    }
  }
}

int main() {
  cin >> n >> str;

  calcFailure();
  calcPI();

  ll ans = 1;

  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < 4; ++j) {
      if (!pi[j][i].first) continue;
      ans += n - pi[j][i].second;
      ans %= mod;
    }
  }

  cout << ans << endl;
  return 0;
}
