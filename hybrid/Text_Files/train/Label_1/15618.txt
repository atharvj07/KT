#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  while (n--) {
    string s;
    getline(cin, s);
    if (s.size() >= 6 && s.substr(s.size() - 6) == "lala.")
      cout << "Freda's" << endl;
    else if (s.size() >= 5 && s.substr(0, 5) == "miao.")
      cout << "Rainbow's" << endl;
    else
      cout << "OMG>.< I don't know!" << endl;
  }
  return 0;
}
