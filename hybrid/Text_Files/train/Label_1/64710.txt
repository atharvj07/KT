MOD = 998244353

n, k = map(int, input().split())
p = list(map(int, input().split()))

# Find the k largest elements and their sum
k_largest = sorted(p, reverse=True)[:k]
max_partition_value = sum(k_largest)

# Count the number of ways to partition
indices = sorted([p.index(x) for x in k_largest])
ways = 1
for i in range(1, k):
    ways *= indices[i] - indices[i-1]
    ways %= MOD

print(max_partition_value, ways)
