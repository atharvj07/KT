import sys
def gcd(a,b):
    while(b):
        a,b=b,a%b
    return a
n=int(input())
des={}
ss={}
for i in range(n):
    des[i],ss[i]=map(int,input().split())
m=0
for i in range(n):
    for j in range(des[i]):
        m+=1
        mn=gcd(ss[i],ss[j])
        if mn!=ss[i] and mn!=ss[j]:
            print(i,j);
            ss[i],ss[j]=0,0
print(m//2)
