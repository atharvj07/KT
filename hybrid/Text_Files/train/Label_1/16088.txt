#include <iostream>

int main() {
    int n; std::cin >> n;
    std::string s; std::cin >> s;
    int ans = 1;
    for (int i = 0, j = 0; i < n; i++) {
        if (s[i] == s[j]) continue;
        ans = std::max(ans, i - j + 1);
        j = i;
    }
    std::cout << ans + 1 << std::endl;
    return 0;
}
