#include<iostream>
#include<algorithm>
using namespace std;
long long a[105],c[105];
main(){
    long long n,i,l,r,b[105],k;
    cin>>n;
    for(i=1;i<=n;i++){
        cin>>a[i];
        b[i]=a[i];
    }
    cin>>l>>r;
    sort(a+1,a+1+n);
    for(i=1;i<=n;i++){
        c[a[i]]+=b[i];
    }
    for(i=1;i<=a[n];i++){
        if(c[i]>=l&&c[i]<=r){
            cout<<i;
            return 0;
        }
    }
    cout<<0;
    return 0;
}
