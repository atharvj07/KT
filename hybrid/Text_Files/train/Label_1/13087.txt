#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <cmath>

using namespace std;

int main() {
    int a, b, n;
    cin >> a >> b >> n;
    vector<vector<long long>> dp(n + 1, vector<long long>(2, 0));
    dp[0][0] = 1;
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j < 2; ++j) {
            dp[i][j] = (dp[i - 1][j] * (b - a)) % 1000000007;
            if (j == 0) {
                dp[i][j] = (dp[i][j] + dp[i - 1][1]) % 1000000007;
            }
        }
    }
    cout << dp[n][1] << endl;
    return 0;
}
