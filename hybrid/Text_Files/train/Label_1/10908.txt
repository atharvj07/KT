#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define vll vector<ll>
#define pb push_back
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        bool check[200001] = {false};
        int n, x, winner = -1, mn = INT_MAX;
        cin >> n;
        for (int i = 0; i < n; i++)
        {
            cin >> x;
            if (check[x] == false)
            {
                check[x] = true;
                if (x < mn)
                {
                    mn = x;
                    winner = i + 1;
                }
            }
        }
        cout << winner << endl;
    }
    return 0;
}
