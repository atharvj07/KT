n, m = map(int, input().split())
restricted = set(tuple(sorted(map(int, input().split()))) for _ in range(m))

# Finding a city that is not in any restricted pair to make it a central hub
for i in range(1, n + 1):
    if all(i not in pair for pair in restricted):
        central_city = i
        break

print(n - 1)
for i in range(1, n + 1):
    if i != central_city:
        print(central_city, i)
