def fastpow(a, n):
    if n == 0:
        return 1
    if n == 1:
        return a

    if n % 2 == 0:
        return (fastpow(a, n // 2) ** 2) % 5
    else:
        return ((fastpow(a, n // 2) ** 2) * a) % 5


n = int(input())
print((fastpow(1, n) + fastpow(2, n) + fastpow(3, n) + fastpow(4, n)) % 5)
