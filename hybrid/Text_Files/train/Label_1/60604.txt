import sys

input = sys.stdin.readline
mod = int(1e9) + 7

def pow(a, b):
    if b == 0: return 1
    if b == 1: return a

    val = pow(a, b // 2)
    val = (val * val) % mod

    if b % 2 != 0:
        val = (val * a) % mod

    return val

n, h = map(int, input().split())
a = list(map(int, input().split()))
f = [0] * 2001

for x in a:
    f[x] += 1

ans = 0
for i in range(h, 2001):
    if f[i] == 0: continue
    tot = 0
    for j in range(h, i + 1):
        tot += f[j]
    ans += pow(2, tot) - 1
    ans %= mod

print(ans)
