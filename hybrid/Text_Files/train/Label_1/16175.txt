#include <bits/stdc++.h>
using namespace std;
#define ll int64_t

ll find_pow(ll x) {
  ll pw = 0;
  while (x > 1) {
    x /= 2;
    ++pw;
  }
  return pw;
}
int solve(vector<ll> &a, ll b) {
  vector<ll> pw(21, 0);
  for (ll x : a) {
    ++pw[find_pow(x)];
  }
  ll ans = 0;
  for (ll i = 0; i < 21; ++i) {
    if ((b >> i) & 1) {
      if (pw[i])
        --pw[i];
      else
        return -1;
      ++ans;
    }
  }
  return ans;
}
int main() {
  int n, q;
  cin >> n >> q;
  vector<ll> a(n);
  for (ll &x : a) cin >> x;
  for (int i = 0; i < q; ++i) {
    ll b;
    cin >> b;
    cout << solve(a, b) << '\n';
  }
  return 0;
}
