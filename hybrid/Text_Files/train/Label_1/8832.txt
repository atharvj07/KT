#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        int ans = 0;
        while (n > 0) {
            ans += n % 10;
            n /= 10;
        }
        if ((ans + (ans + k) * k / 2) > n)
            cout << -1 << endl;
        else
            cout << ans << endl;
    }
    return 0;
}
