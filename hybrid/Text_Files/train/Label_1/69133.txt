from sys import stdin
read = lambda:stdin.readline().strip()
mod = 998244353
n = int(read())
a = [read() for _ in range(n)]
dp = [[[0,0] for _ in range(2)] for _ in range(n)]
dp[0][0][0] = 1
dp[0][1][0] = 0
dp[0][0][1] = 0
dp[0][1][1] = 1
for i in range(1,n):
    for c, e in enumerate(['W','B']):
        for j in range(2):
            if a[i][c] == '?' or a[i][c] == e:
                dp[i][j][c] = dp[i - 1][j][1 - c]
            dp[i][j][c] %= mod
print(dp[n - 1][0][0])
