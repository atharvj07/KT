for _ in range(int(input())):
    n = int(input())
    need = []
    for _ in range(n):
        m, p = map(int, input().split())
        need.append((p, m))
    need.sort()
    ans = 0
    cover = 0
    for p, m in need:
        if cover + m >= n - 1: break
        cover += m
        ans += p
    print(ans)
