#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
typedef long long ll;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n, k;
    cin >> n >> k;
    vector<ll> a(n);
    for (auto &i : a) cin >> i;
    sort(a.begin(), a.end());
    if (k < a[n / 2] - a[(n - 1) / 2]) {
      cout << -1 << '\n';
      continue;
    }
    ll b = a[n / 2] + k;
    ll ans = b;
    for (int i = n / 2 - 1; i >= 0; i--) {
      ans = min(ans, b + a[(n + 1) / 2] - a[i]);
    }
    cout << ans << '\n';
  }
  return 0;
}
