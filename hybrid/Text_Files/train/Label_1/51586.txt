import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int prizeFirstPlace = 300000;
        int prizeSecondPlace = 200000;
        int prizeThirdPlace = 100000;
        int bonusFirstPlaceBoth = 400000;

        int totalPrize = 0;
        if (x == 1 || y == 1) {
            totalPrize += bonusFirstPlaceBoth;
        }
        if (x <= 3) {
            totalPrize += prizeFirstPlace - (x - 1) * 100000;
        }
        if (y <= 3) {
            totalPrize += prizeFirstPlace - (y - 1) * 100000;
        }

        System.out.println(totalPrize);
    }
}
