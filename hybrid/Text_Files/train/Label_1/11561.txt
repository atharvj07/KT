#include <bits/stdc++.h>
#include <vector>
using namespace std;

const int MAXN = 5005;
const int MOD = 1e9 + 7;
vector<int> v, ps;
vector<vector<int>> dp;
int n, a[MAXN];

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i], v.push_back(a[i]);
    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());
    for (int i : v) {
        for (int j : v) ps.push_back(i * j);
    }
    sort(ps.begin(), ps.end());
    ps.erase(unique(ps.begin(), ps.end()), ps.end());
    for (int p : ps) {
        dp.push_back(vector<int> (n + 1, 0));
        for (int i = 0; i < n; i++) {
            if (i) dp.back()[i] = dp.back()[i - 1];
            if (a[i] == 1) {
                dp.back()[i]++;
                continue;
            }
            dp.back()[i] += (binary_search(ps.begin(), ps.end(), a[i] * a[i]));
        }
    }
    for (int i = 0; i < n; i++) {
        int res = dp[0][i];
        for (int j = 1; j < dp.size(); j++) {
            res += (binary_search(ps.begin(), ps.end(), a[i] * ps[j])) * dp[j][i];
            res %= MOD;
        }
        cout << res << " ";
    }
}
