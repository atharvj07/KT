MOD = 998244353

def comb(n, k, fac, inv_fac):
    if n < k or k < 0:
        return 0
    return fac[n] * inv_fac[k] % MOD * inv_fac[n-k] % MOD

def solve(a, b, c):
    fac = [1] * (max(a, b, c) + 1)
    for i in range(2, max(a, b, c) + 1):
        fac[i] = fac[i-1] * i % MOD

    inv_fac = [1] * (max(a, b, c) + 1)
    inv_fac[max(a, b, c)] = pow(fac[max(a, b, c)], MOD-2, MOD)
    for i in range(max(a, b, c) - 1, 0, -1):
        inv_fac[i] = inv_fac[i+1] * (i+1) % MOD

    def count_ways(x, y):
        ways = 0
        for i in range(min(x, y) + 1):
            ways = (ways + comb(x, i, fac, inv_fac) * comb(y, i, fac, inv_fac) % MOD * fac[i]) % MOD
        return ways

    return count_ways(a, b) * count_ways(b, c) % MOD * count_ways(a, c) % MOD

a, b, c = map(int, input().split())
print(solve(a, b, c))
