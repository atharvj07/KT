def min_cost_to_complete_delivery(d, n, m, gas_stations):
    gas_stations.sort()
    gas_stations.append((d, 0))
    current_pos, current_fuel, total_cost = 0, n, 0
    stations_stack = []
    
    for station in gas_stations:
        distance_to_station = station[0] - current_pos
        while stations_stack and (current_fuel < distance_to_station or station[1] < stations_stack[-1][1]):
            if current_fuel < distance_to_station:
                fuel_needed = distance_to_station - current_fuel
                while fuel_needed > 0:
                    if not stations_stack:
                        return -1
                    cheapest_station = stations_stack.pop()
                    fuel_to_buy = min(fuel_needed, n - current_fuel)
                    total_cost += fuel_to_buy * cheapest_station[1]
                    current_fuel += fuel_to_buy
                    fuel_needed -= fuel_to_buy
            if stations_stack and station[1] < stations_stack[-1][1]:
                stations_stack.pop()
            else:
                break
        if station[1] == 0 or not stations_stack or station[1] < stations_stack[-1][1]:
            stations_stack.append(station)
        current_fuel -= distance_to_station
        current_pos = station[0]
    
    return total_cost

d, n, m = map(int, input().split())
gas_stations = [tuple(map(int, input().split())) for _ in range(m)]
print(min_cost_to_complete_delivery(d, n, m, gas_stations))
