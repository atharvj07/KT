import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int cnt = 0;
        while (h >= a || w >= b) {
            if (h < a) {
                int tmp = a;
                a = b;
                b = tmp;
            }
            cnt += h / a * (w / b);
            h %= a;
        }
        System.out.println(h * w - cnt * a * b);
    }
}
