n, r = map(int, input().split())
points = [list(map(int, input().split())) for _ in range(n)]

def coverable_points(x0, y0, r, points):
    return sum(1 for x, y in points if abs(x - x0) + abs(y - y0) <= r)

def max_coverable_points(n, r, points):
    max_points = 0
    for x, y in points:
        for dx, dy in [(0, r), (r, 0), (0, -r), (-r, 0)]:
            max_points = max(max_points, coverable_points(x + dx, y + dy, r, points))
    return max_points

print(max_coverable_points(n, r, points))
