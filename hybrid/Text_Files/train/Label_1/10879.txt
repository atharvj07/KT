#include<bits/stdc++.h>
#define ll long long
using namespace std;
#define lcm(a, b) (a * (b/__gcd(a, b)))
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    ll n;
    cin >> n;
    ll t = n;
    vector<ll> v(n);
    for (int i = 0; i < n; i++) { v[i] = i + 1; }
    while (t--) {
        ll g = 0;
        for (int i = 0; i < n; i++) {
            if (v[i] != -1) {
                if (g == 0) { g = v[i]; }
                else { g = __gcd(g, v[i]); }
            }
        }
        cout << g << " ";
        ll mxindex = -1;
        for (int i = 0; i < n; i++) {
            if (v[i] != -1) {
                if (mxindex == -1) { mxindex = i; }
                else {
                    if (v[i] > v[mxindex]) { mxindex = i; }
                }
            }
        }
        v[mxindex] = -1;
    }
    return 0;
}
