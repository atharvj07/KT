import sys
input=sys.stdin.readline
n=int(input())
m=[int(x) for x in input().split()]
l=[0 for _ in range(n+2)]
r=[0 for _ in range(n+2)]
for i in range(1,n):
    l[i]=l[i-1]
    if m[i-1]>m[i]:
        l[i]=i
for i in range(n-1,0,-1):
    r[i]=r[i+1]
    if m[i]>m[i+1]:
        r[i]=i
ans=[0 for _ in range(n)]
for i in range(n):
    ans[i]=min(m[i],m[l[i]],m[r[i]])
print(*ans)
