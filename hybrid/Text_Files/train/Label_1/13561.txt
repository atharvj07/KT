#include <iostream>
#include <vector>

using namespace std;

int main() {
  int q, x;
  cin >> q >> x;
  vector<int> v;
  vector<int> f(1000001, 0);
  for (int i = 0; i < q; i++) {
    int y;
    cin >> y;
    v.push_back(y);
    f[y] = 1;
    int mex = 0;
    while (f[mex]) { mex++; }
    int ans = mex;
    for (int j = 0; j < (int)v.size(); j++) {
      int z = v[j] + x;
      while (z >= 0 && f[z]) { z += x; }
      ans = max(ans, z);
      z = v[j] - x;
      while (z >= 0 && f[z]) { z -= x; }
      ans = max(ans, z);
      if (ans >= 1000000) { break; }
    }
    cout << ans << endl;
  }
  return 0;
}
