#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, s;
  cin >> n >> s;
  vector<pair<int, int>> times(n);
  for (auto& [h, m] : times) cin >> h >> m;
  int t = 0;
  for (int i = 0; i < n; i++) {
    if (times[i].first * 60 + times[i].second - t >= s) {
      cout << t / 60 << ' ' << t % 60 << endl;
      return 0;
    }
    t = times[i].first * 60 + times[i].second + 1;
  }
  cout << times[n - 1].first + 1 << ' ' << times[n - 1].second << endl;
  return 0;
}
