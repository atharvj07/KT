#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

int main() {
    int k, n;
    cin >> k >> n;
    vector<pair<string, int>> strs(k);
    for (int i = 0; i < k; ++i) {
        string s;
        int a;
        cin >> s >> a;
        strs[i] = {s, a};
    }
    sort(strs.begin(), strs.end(), [](const pair<string, int>& a, const pair<string, int>& b) {
        return a.first != b.first ? a.first < b.first : a.second > b.second;
    });
    int ans = 0;
    for (const auto& [s, a] : strs) {
        string r = s;
        reverse(r.begin(), r.end());
        if (s == r) {
            ans += max(a, 0);
        } else {
            ans += max(a, a * 2);
        }
    }
    cout << ans << endl;
    return 0;
}
