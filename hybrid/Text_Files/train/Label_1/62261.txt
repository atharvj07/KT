from sys import stdin

n, m, mod = map(int, stdin.readline().split())
rows = [stdin.readline() for _ in range(m)]
ans = 1
for i in range(m):
    cnt = 0
    for j in range(n):
        if rows[i][j] == '1': cnt += 1
    if cnt != 2:
        ans = 0
        break
for i in range(n):
    cnt = 0
    for j in range(m):
        if rows[j][i] == '1': cnt += 1
    if cnt != 2:
        ans = 0
        break
if ans:
    for _ in range(n - m):
        ans = ans * 2 % mod
print(ans)
