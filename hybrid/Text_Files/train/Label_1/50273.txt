import java.util.*;
import java.io.*;

public class Main {
    static final long MAX_VAL = (long)2e18;
    static final long LONG_MAX = Long.MAX_VALUE;

    public static void main(String[] args) throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        long[] taxes = new long[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; ++i) {
            taxes[i] = Long.parseLong(st.nextToken());
        }
        ArrayList<Integer>[] adj = new ArrayList[n];
        for (int i = 0; i < n; ++i) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; ++i) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken()) - 1;
            int b = Integer.parseInt(st.nextToken()) - 1;
            adj[a].add(b);
        }

        long[] dp = new long[n];
        Arrays.fill(dp, LONG_MAX);
        boolean[] vis = new boolean[n];

        if (!dfs(adj, taxes, dp, vis, 0, 0)) {
            System.out.println("LOSE");
            return;
        }
        System.out.println("WIN");
        for (long tax : dp) {
            System.out.print(tax + " ");
        }
    }

    static boolean dfs(ArrayList<Integer>[] adj, long[] taxes, long[] dp, boolean[] vis, int v, long mn) {
        //if we've already visited this node, then we can skip it
        if (vis[v]) {
            return true;
        }

        vis[v] = true;
        boolean canWin = false;
        dp[v] = taxes[v];

        for (int u: adj[v]) {
            //if we can move from node v to node u
            if (dp[v] <= taxes[u]) {
                //dp[u] will be set to the minimum of it's current value and dp[v] + 1
                dp[u] = Math.min(dp[u], dp[v] + 1);
                canWin |= dfs(adj, taxes, dp, vis, u, mn);
            }
        }

        //if we can move from node v to all of it's neighbors such that dp[v] <= taxes[u], for all neighbors u,
        //then we can win
        return canWin || (dp[v] <= mn);
    }
}
