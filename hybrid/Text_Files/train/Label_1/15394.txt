#include <bits/stdc++.h>
using namespace std;
#define pb push_back
const int N = 1e5 + 5;
vector<int> adj[N];
int sz[N], nn, ans[N], vis[N];
void dfs(int node, int prev) {
    sz[node] = 1;
    for (auto x : adj[node]) {
        if (x == prev) continue;
        dfs(x, node);
        sz[node] += sz[x];
    }
    ans[++nn] = node;
}
int32_t main() {
    int n, x, y;
    cin >> n;
    for (int i = 1; i < n; i++) {
        cin >> x >> y;
        adj[x].pb(y);
        adj[y].pb(x);
    }
    dfs(1, 0);
    for (int i = n; i >= 1; i--) cout << ans[i] << " ";
    return 0;
}
