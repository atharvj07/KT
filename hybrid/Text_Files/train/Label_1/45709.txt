import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int K = sc.nextInt();
        int S = sc.nextInt();
        int[] A = new int[N];
        int sum = 0;
        for (int i = 0; i < N; i++) {
            if (K > 0) {
                int t = Math.min(S - sum, 1000000000);
                A[i] = t;
                sum += t;
                K--;
            } else {
                A[i] = S - sum;
            }
        }
        for (int a : A) {
            System.out.print(a + " ");
        }
        System.out.println();
    }
}
