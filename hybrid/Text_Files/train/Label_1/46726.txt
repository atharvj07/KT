import java.util.Arrays;
import java.util.Scanner;

public class AlarmClocks {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] alarms = new int[n];
        for (int i = 0; i < n; i++) {
            alarms[i] = scanner.nextInt();
        }

        Arrays.sort(alarms);

        int minAlarmsToTurnOff = 0;
        int consecutiveAlarms = 0;
        for (int i = 0; i < n; i++) {
            if (i > 0 && alarms[i] == alarms[i - 1] + 1) {
                consecutiveAlarms++;
            } else {
                consecutiveAlarms = 1;
            }
            if (consecutiveAlarms >= k) {
                minAlarmsToTurnOff++;
                consecutiveAlarms -= k - 1;
            }
        }

        System.out.println(minAlarmsToTurnOff);
    }
}
