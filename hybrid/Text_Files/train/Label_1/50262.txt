import java.util.ArrayList;
import java.util.Scanner;

public class Main {

    private static ArrayList<Sticker> stickers;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (true) {
            int n = Integer.parseInt(sc.nextLine());
            if (n == 0) break;

            stickers = new ArrayList<>(n);

            for (int i = 0; i < n; ++i) {
                String[] coords = sc.nextLine().split(",");
                double x = Double.parseDouble(coords[0]);
                double y = Double.parseDouble(coords[1]);
                stickers.add(new Sticker(x, y));
            }

            int maxOverlap = 0;
            for (Sticker sticker1 : stickers) {
                int overlapCount = 0;
                for (Sticker sticker2 : stickers) {
                    if (sticker1 != sticker2 && sticker1.overlaps(sticker2)) {
                        overlapCount++;
                    }
                }
                maxOverlap = Math.max(maxOverlap, overlapCount);
            }

            System.out.println(maxOverlap);
        }

        sc.close();
    }

    private static class Sticker {

        private double x;
        private double y;

        public Sticker(double x, double y) {
            this.x = x;
            this.y = y;
        }

        public boolean overlaps(Sticker other) {
            double distance = Math.sqrt(Math.pow(this.x - other.x, 2) + Math.pow(this.y - other.y, 2));
            return distance <= 2.0;
        }
    }
}
