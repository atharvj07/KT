#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=1e9+7;
int main(){
    ll n;
    cin>>n;
    vector<ll>c(n);
    for(auto &it:c) cin>>it;
    ll l=0,ans=0,open=0;
    for(ll r=0;r<n;r++){
        open+=c[r]%2==1?1:-1;
        if(open>=0) ans+=1;
        else{
            while(open<0) open-=c[l]%2==1?1:-1,l++;
        }
    }
    cout<<ans;
}
