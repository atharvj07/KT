import java.util.*;

public class Main {
    static int N;
    static int[][] w;
    static int[][] p;
    static boolean[] vis;
    static int[] cycle;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        w = new int[N+1][N+1];
        p = new int[N+1][N+1];
        vis = new boolean[N+1];
        cycle = new int[N+1];

        int l = 1 << (N-1);
        for (int i = 1; i <= N; i++) {
            for (int j = 1; j <= (N-1)/2; j++) {
                w[i][N-j] = l;
                p[i][N-j] = i;
                w[i][j] = w[i][N-j];
                p[i][j] = p[i][N-j];
                l *= 2;
            }
        }

        dfs(1, 1);

        for (int i = 1; i <= N; i++) {
            System.out.print(w[i][1]);
            for (int j = 2; j <= N; j++) {
                System.out.print(" " + w[i][j]);
            }
            System.out.println();
        }
    }

    static void dfs(int cur, int prev) {
        vis[cur] = true;
        cycle[prev] = cur;
        int next = 0;
        for (int i = 1; i <= N; i++) {
            if (w[cur][i] != 0 && !vis[i]) {
                next = i;
                break;
            }
        }
        if (next != 0) {
            dfs(next, cur);
        } else {
            int cnt = 0;
            for (int i = 1; i <= N; i++) {
                if (cycle[i] != 0) {
                    cnt++;
                }
            }
            if (cnt == N) {
                int min = -1;
                for (int i = 1; i <= N; i++) {
                    for (int j = i+1; j <= N; j++) {
                        if (w[i][j] < min || min == -1) {
                            min = w[i][j];
                        }
                    }
                }
                for (int i = 1; i <= N; i++) {
                    for (int j = i+1; j <= N; j++) {
                        if (w[i][j] == min) {
                            w[i][j] *= 2;
                            w[j][i] *= 2;
                        }
                    }
                }
            }
        }
        vis[cur] = false;
        cycle[prev] = 0;
    }
}
