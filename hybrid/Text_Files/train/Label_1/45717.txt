import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[m];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            b[i] = sc.nextInt();
        }
        boolean possible = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if ((a[i] ^ b[j]) % 2 == 1) {
                    possible = false;
                }
            }
        }
        if (possible) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    System.out.print((a[i] ^ b[j]) / 2 + " ");
                }
                System.out.println();
            }
        } else {
            System.out.println("NO");
        }
    }
}
