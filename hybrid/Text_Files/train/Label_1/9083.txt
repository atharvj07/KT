#include <iostream>
#include <string>
using namespace std;
int main() {
  int t;
  string a, b, c;
  cin >> t;
  while (t--) {
    cin >> a >> b >> c;
    int cnt[26] = {0};
    for (char x : a) cnt[x - 'a']++;
    for (char x : b) cnt[x - 'a']--;
    for (char x : c) cnt[x - 'a']++;
    bool f = true;
    for (int i = 0; i < 26; i++)
      if (cnt[i] != 0) f = false;
    if (f)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
