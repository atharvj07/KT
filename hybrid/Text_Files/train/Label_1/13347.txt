#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define faster ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL)

int main() {
    faster;
    ll n, b[100001], a[100001], x[100001], k[100001];
    cin >> n;
    for(int i = 1; i <= n; i++) cin >> b[i], cin >> a[i];
    for(int i = 2; i <= n; i++){ cin >> x[i] >> k[i]; x[i]++, k[i]++; }
    ll sum = 0;
    for(int i = 1; i <= n; i++) sum += b[i] - a[i];
    if(sum >= 0) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
