n = int(input())
b = list(map(int, input().split()))

difference_to_beauty = {}

for i in range(n):
    difference = i - b[i]
    if difference in difference_to_beauty:
        difference_to_beauty[difference] += b[i]
    else:
        difference_to_beauty[difference] = b[i]

print(max(difference_to_beauty.values()))
