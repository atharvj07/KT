import sys
input=sys.stdin.readline

n,m,k=map(int,input().split())
g=[[] for i in range(n+1)]
vis=[False for i in range(m+1)]
cur=0
ans=[0 for i in range(n+1)]
for i in range(1,m+1):
    x,y=map(int,input().split())
    g[x].append(y)
    g[y].append(x)
    if len(g[x])>=k:
        vis[i]=True
        cur+=1
        ans[i]=cur
    if len(g[y])>=k:
        vis[i]=True
        cur+=1
        ans[i]=cur
for i in range(1,m+1):
    if not vis[i]:
        x=g[cur][0]
        for y in g[cur]:
            if y!=x and len(g[y])>=k:
                cur=y
                break
        ans[i]=cur
    print(ans[i])
