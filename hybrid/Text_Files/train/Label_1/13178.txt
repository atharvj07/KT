#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define pb push_back

const ll mod = 1e9 + 7;
const char nl = '\n';

void solve() {
    int n; cin >> n;
    vi a(n);
    for (int& ai : a) cin >> ai;
    int mn = *min_element(a.begin(), a.end());
    int mx = *max_element(a.begin(), a.end());
    int pt = 0;
    for (int i = mn; i <= mx; ++i) {
        for (int j = 0; j < (i == mx || i > mn); ++j) {
            a[pt++] = i;
            if (pt == n) {
                cout << n - j << nl;
                for (int ai : a) cout << ai << " ";
                cout << nl;
                return;
            }
        }
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int t = 1;
    while (t--) solve();
    return 0;
}
