#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define mod 1000000007

ll binpow(ll a, ll p) {
  ll res = 1;
  while (p) {
    if (p & 1) res = res * a % mod;
    a = a * a % mod;
    p >>= 1;
  }
  return res;
}
ll fact[505], invfact[505];

ll nCr(ll n, ll r) {
  return (fact[n] * invfact[r] % mod) * invfact[n - r] % mod;
}

int main() {
  ll n, m, b, ans = 0;
  cin >> n >> m >> b;
  fact[0] = invfact[0] = 1;
  for (ll i = 1; i <= 500; i++) {
    fact[i] = fact[i - 1] * i % mod;
    invfact[i] = binpow(fact[i], mod - 2);
  }
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];
  for (ll i = 0; i <= b; i++) {
    ll sum = 0;
    for (ll j = 0; j < n; j++) sum += a[j];
    if (sum <= i) ans = (ans + nCr(m - 1, i - sum)) % mod;
  }
  cout << ans << endl;
  return 0;
}
