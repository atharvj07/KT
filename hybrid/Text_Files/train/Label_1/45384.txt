import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        try {
            String[] firstLine = reader.readLine().split(" ");
            int t = Integer.valueOf(firstLine[0]);
            int n = Integer.valueOf(firstLine[1]);

            for (int i = 0; i < t; i++) {
                String[] query = reader.readLine().split(" ");
                int a = Integer.valueOf(query[0]);
                int b = Integer.valueOf(query[1]);
                System.out.println(solve(n, a, b));
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private static int solve(int n, int a, int b) {
        if (n == 0) {
            return 0;
        }

        if (n == 1) {
            return 1;
        }

        int m = (int) Math.pow(2, n);
        int mid = m / 2;

        if (a <= mid && b <= mid) {
            return solve(n - 1, a, b);
        } else if (a > mid && b > mid) {
            return solve(n - 1, a - mid, b - mid);
        } else {
            return solve(n - 2, a, b) + 1;
        }
    }
}
