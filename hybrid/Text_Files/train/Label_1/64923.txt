n, m = map(int, input().split())

subjects = [[] for _ in range(m)]

for _ in range(n):
    s, r = map(int, input().split())
    subjects[s-1].append(r)

for i in range(m):
    subjects[i].sort(reverse=True)

max_sum = 0

for i in range(1, n+1):
    current_sum = 0
    for subject in subjects:
        sum_i = sum([subject[j] for j in range(min(len(subject), i)) if subject[j] > 0])
        if sum_i > 0:
            current_sum += sum_i
    max_sum = max(max_sum, current_sum)

print(max_sum)
