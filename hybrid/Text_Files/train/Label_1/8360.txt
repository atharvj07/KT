#include <iostream>
#include <vector>
#include <queue>
using namespace std;

struct guest {
  int a, b;
  bool operator<(const guest& g) const {
    if (a != g.a) return a < g.a;
    return b < g.b;
  }
};

vector<vector<int> > graph;
vector<bool> visited;
int n, k;

int bfs(int start) {
  priority_queue<guest> q;
  q.push({0, 0});
  int res = 0;
  while (!q.empty()) {
    guest c = q.top();
    q.pop();
    if (visited[c.a] && visited[c.b]) continue;
    if (c.a == c.b) {
      if (!visited[c.a]) {
        visited[c.a] = true;
        res++;
      }
      continue;
    }
    if (!visited[c.a]) {
      visited[c.a] = true;
      q.push({c.a, c.b});
      res++;
    }
    if (!visited[c.b]) {
      visited[c.b] = true;
      q.push({c.b, c.a});
      res++;
    }
  }
  return res;
}

int main() {
  cin >> n >> k;
  graph.assign(n + 1, vector<int>());
  visited.assign(n + 1, false);
  for (int i = 0; i < k; i++) {
    int a, b;
    cin >> a >> b;
    graph[a].push_back(b);
    graph[b].push_back(a);
  }
  int ans = k - bfs(1);
  cout << ans << endl;
  return 0;
}
