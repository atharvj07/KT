#include <iostream>
#include <vector>

using namespace std;

vector<vector<pair<int, int>>> v;
string s;

int main() {
  int n, m;
  cin >> n >> m;
  cin >> s;
  v.resize(n + 1);
  for (int i = 0; i < m; i++) {
    int l, r, c1, c2;
    cin >> l >> r >> c1 >> c2;
    v[l].push_back({r, c2});
    if (r + 1 <= n) v[r + 1].push_back({l - 1, c1});
  }
  for (int i = 1; i <= n; i++) {
    for (auto &p : v[i]) {
      if (s[i - 1] == p.second) s[i - 1] = p.first;
    }
    cout << s[i - 1];
  }
  return 0;
}
