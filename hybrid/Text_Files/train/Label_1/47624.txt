import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int n = s.length(), ans = 0;
        for (int i = 0; i < n; i++) {
            int cnt = 0;
            for (int j = i; j < n; j++) {
                if (s.charAt(j) == '(') cnt++;
                else cnt--;
                if (cnt == 0 && i != 0 && j != n - 1) ans++;
            }
        }
        System.out.println(ans);
    }
}
