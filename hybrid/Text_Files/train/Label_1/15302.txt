#include <bits/stdc++.h>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> en(n), ex(n);
  for (int i = 0; i < n; i++) {
    std::cin >> en[i];
    en[i]--;
  }
  for (int i = 0; i < n; i++) {
    std::cin >> ex[i];
    ex[i]--;
  }
  std::vector<int> ans(n), where(n);
  for (int i = 0; i < n; i++) {
    where[en[i]] = i;
  }
  for (int i = 0; i < n; i++) {
    if (where[i] > where[ex[i]]) {
      ans[ex[i]] = 1;
    }
  }
  int res = 0;
  for (int i = 0; i < n; i++) { res += ans[i]; }
  std::cout << res << '\n';
  return 0;
}
