#include <bits/stdc++.h>

using namespace std;

#define int long long

int32_t main() {
    int tt;
    cin >> tt;
    while (tt--) {
        int n,d;
        cin >> n >> d;
        int a[n], mn = LLONG_MAX;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            mn = min(mn, a[i]);
        }
        if(mn <= d)
            puts("YES");
        else
            puts("NO");
    }
}
