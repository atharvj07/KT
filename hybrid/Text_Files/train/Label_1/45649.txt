import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int count = 0;
        for (int i = 1; i <= n; i++) {
            if (isBinary(i)) {
                count++;
            }
        }
        System.out.println(count);
    }

    private static boolean isBinary(int n) {
        while (n > 0) {
            if (n % 10 != 0 && n % 10 != 1) {
                return false;
            }
            n /= 10;
        }
        return true;
    }
}
