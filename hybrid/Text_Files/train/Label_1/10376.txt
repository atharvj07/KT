#include <bits/stdc++.h>
using namespace std;
typedef pair<int,int> pii;
const int mod = 1e9+7;
const int INF = 0x3f3f3f3f;
#define int long long
#define endl "\n"
#define all(x) (x).begin(), (x).end()
#define sz(x) (int)(x).size()
#define pb push_back
int sz, cp;
vector<int> dp;
int32_t main() {
    cin.tie(0)->sync_with_stdio(0);
    cin >> sz >> cp;
    dp.resize(sz+1);
    vector<int> a(sz);
    for(int& x : a) cin >> x;
    for(int i = 1; i <= sz; i++) {
        int cur = dp[i-1];
        for(int j = i-1; j > 0; j--) {
            if(a[j-1] == a[i-1]) {
                cur = min(cur, dp[j-1]+1);
            }
        }
        if(cur == dp[i-1]) cur++;
        dp[i] = cur;
    }
    cout << dp[sz] << endl;
    return 0;
}
