#include <iostream>
#include <vector>
#include <queue>
#include <limits>

using namespace std;

struct Edge {
    int to;
    long long weight;
};

typedef vector<vector<Edge>> Graph;

const long long INF = numeric_limits<long long>::max();

void dijkstra(const Graph& graph, vector<long long>& dist, int start) {
    dist.assign(graph.size(), INF);
    dist[start] = 0;

    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
    pq.push({0, start});

    while (!pq.empty()) {
        long long current_dist = pq.top().first;
        int current_vertex = pq.top().second;
        pq.pop();

        if (current_dist > dist[current_vertex]) {
            continue;
        }

        for (const Edge& edge : graph[current_vertex]) {
            if (dist[current_vertex] + edge.weight < dist[edge.to]) {
                dist[edge.to] = dist[current_vertex] + edge.weight;
                pq.push({dist[edge.to], edge.to});
            }
        }
    }
}

int main() {
    int n;
    cin >> n;

    Graph graph(n);
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            int weight;
            cin >> weight;
            graph[i].push_back({j, weight});
            graph[j].push_back({i, weight});
        }
    }

    vector<int> deleted_vertices;
    for (int i = 0; i < n; ++i) {
        int vertex;
        cin >> vertex;
        --vertex;
        deleted_vertices.push_back(vertex);
    }

    for (int i = 0; i < n; ++i) {
        vector<long long> dist;
        dijkstra(graph, dist, i);

        long long sum = 0;
        for (int j = 0; j < n; ++j) {
            if (i == j) {
                continue;
            }
            sum += dist[j];
        }

        cout << sum << " ";

        for (int j = 0; j < n; ++j) {
            graph[i][j] = graph[j][i] = {deleted_vertices[i], INF};
        }
    }

    cout << endl;

    return 0;
}
