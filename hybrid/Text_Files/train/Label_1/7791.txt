#include <iostream>
using namespace std;

int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        long long a, b;
        cin >> a >> b;
        if (b < a || (a & b) != a)
        {
            cout << -1 << '\n';
        }
        else
        {
            int ans = 0;
            while (b != a)
            {
                if (b % 8 == 0)
                {
                    b /= 8, ++ans;
                }
                else if (b % 4 == 0)
                {
                    b /= 4, ++ans;
                }
                else if (b % 2 == 0)
                {
                    b /= 2, ++ans;
                }
                else
                {
                    cout << -1 << '\n';
                    return 0;
                }
            }
            cout << ans << '\n';
        }
    }
    return 0;
}
