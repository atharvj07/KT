#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    int n, m, a;
    cin >> n >> m >> a;
    vector<ll> b(n), p(m);
    for (auto &bi : b)
        cin >> bi;
    for (auto &pi : p)
        cin >> pi;
    sort(b.rbegin(), b.rend()), sort(p.begin(), p.end());
    ll ans1 = 0, ans2 = 0;
    for (int i = 0; i < min(n, m); i++) {
        ll budget = a / 2;
        if (budget + b[i] >= p[i]) {
            ans1++, ans2 += b[i] + p[i] - budget;
            a -= 2 * p[i];
        }
    }
    cout << ans1 << ' ' << ans2 << '\n';
}
