import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int h = in.nextInt();
        int max = 0;
        for (int i = 0; i < n; i++) {
            max = Math.max(max, in.nextInt());
        }
        System.out.println(Math.min(n, max <= h ? n : 2 * n - (max + h - 1) / h));
    }
}
