#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<bool>> a(n, vector<bool>(m));
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      int x;
      cin >> x;
      a[i][j] = x;
    }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (!a[i][j]) {
        bool up = false, down = false, left = false, right = false;
        if (i > 0 && a[i - 1][j]) up = true;
        if (i < n - 1 && a[i + 1][j]) down = true;
        if (j > 0 && a[i][j - 1]) left = true;
        if (j < m - 1 && a[i][j + 1]) right = true;
        if (up || down || left || right) ans++;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
