#include <bits/stdc++.h>
#define vi vector<int>
#define pb push_back
using namespace std;

map<char, char> m{{'(', ')'}, {'<', '>'}, {'[', ']'}, {'{', '}'}};
map<char, int> c;

int main()
{
    string s; cin >> s;
    stack<char> st;
    for (char &ch : s) {
        if (m.count(ch)) {
            st.push(ch);
            ++c[ch];
        } else {
            if (st.empty() || m[st.top()] != ch) {
                cout << "Impossible" << '\n';
                return 0;
            } else {
                st.pop();
                --c[m[ch]];
            }
        }
    }
    while (!st.empty()) {
        if (c[st.top()] == 0) {
            cout << "Impossible" << '\n';
            return 0;
        }
        st.pop();
    }
    cout << (int)s.size() - 2 * st.size() << '\n';
    return 0;
}
