import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            int d = sc.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }
            int[] last = new int[k + 1];
            Arrays.fill(last, -1);
            int ans = 0;
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                if (i - last[a[i]] >= d) {
                    cnt--;
                }
                last[a[i]] = i;
                cnt++;
                ans = Math.max(ans, cnt);
            }
            System.out.println(k - ans);
        }
    }
}
