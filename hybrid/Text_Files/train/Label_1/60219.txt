import sys
#from collections import Counter

def decomposition(n, a):
	if n == 1:
		return a[::-1]
	for d in range(2, int(n**.5) + 1):
		if n % d == 0:
			return decomposition(n // d, [d] + a)
			#return decomposition(d, a + [n // d])
	return [n] + a

for t in range(int(sys.stdin.readline().strip())):
	n = int(sys.stdin.readline().strip())
	a = decomposition(n, [])
	print(len(a))
	print(*a)
