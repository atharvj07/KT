#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  vector<int> b, c, d;
  int s1 = 0, s2 = 0, s3 = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] < 0) {
      b.push_back(a[i]);
      s1++;
    } else if (a[i] > 0) {
      c.push_back(a[i]);
      s2++;
    } else {
      d.push_back(a[i]);
      s3++;
    }
  }
  cout << s1 << " ";
  for (int x : b) cout << x << " ";
  cout << "\n" << s2 << " ";
  for (int x : c) cout << x << " ";
  cout << "\n" << s3 << " ";
  for (int x : d) cout << x << " ";
}
