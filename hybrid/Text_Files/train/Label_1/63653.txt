mod = 10**9 + 7
n, m = map(int, input().split())
a = list(map(int, input().split()))
dp = [0] * (1<<n)
dp[0] = 1
for i in range((1<<n)-1):
    last = -1
    for j in range(n):
        if i&(1<<j):
            if last != -1 and last != j-1 and last != j+1:
                break
            last = j
    else:
        for j in range(n):
            if not i&(1<<j):
                dp[i|(1<<j)] += dp[i]
                dp[i|(1<<j)] %= mod
print(dp[(1<<n)-1])
