#include <iostream>
#include <string>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int x = 0, y = 0;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'U') {
      y++;
    } else if (s[i] == 'D') {
      y--;
    } else if (s[i] == 'L') {
      x--;
    } else if (s[i] == 'R') {
      x++;
    }
    if (x == 0 && y == 0) {
      ans = i + 1;
    }
  }
  cout << ans << endl;
}
