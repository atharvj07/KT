#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> p(n);
  if (k == 1) {
    for (int i = 0; i < n; ++i) cout << i + 1 << " ";
  } else {
    int d = n / k;
    int q = n % k;
    if (d == 1) {
      for (int i = 0; i < q; ++i) p[i] = i + d + 1;
      for (int i = q; i < n; ++i) p[i] = i - q;
    } else {
      for (int i = 0; i < q; ++i) p[i] = d * (i + 1);
      for (int i = q; i < n; ++i) p[i] = d * (i - q) + 1;
    }
    for (int i = 0; i < n; ++i) cout << p[i] << " ";
  }
  cout << endl;
  return 0;
}
