#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (auto &ai : a) cin >> ai, --ai;
  int ans = 0, cnt = 0;
  bool same = true;
  for (int i = 0; i < n; ++i) {
    if (i && a[i - 1] == a[i]) same = true;
    if (!i || a[i - 1] != a[i]) { same = false, cnt = 0; }
    ++cnt;
    if (!same) ans += cnt - 1;
  }
  cout << ans << '\n';
}
