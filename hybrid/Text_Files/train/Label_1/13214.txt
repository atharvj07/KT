#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  cin >> T;
  while (T--) {
    int n, x;
    cin >> n >> x;
    int a[n], mx = 0;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) {
      if (a[i] >= x) mx = max(mx, a[i]);
    }
    if (mx == 0) {
      cout << 0 << endl;
      continue;
    }
    int ans = mx - x, cnt = 0, f = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] > x) cnt++;
    }
    if (cnt < ans) {
      cout << -1 << endl;
      continue;
    }
    cout << ans << endl;
  }
  return 0;
}
