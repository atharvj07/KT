n = int(input())
a = list(map(int, input().split()))

unique_elements = {}
for i, val in enumerate(a):
    unique_elements[val] = i

ordered_unique_elements = sorted(unique_elements.items(), key=lambda x: x[1])

print(len(ordered_unique_elements))
print(' '.join(str(elem[0]) for elem in ordered_unique_elements))
