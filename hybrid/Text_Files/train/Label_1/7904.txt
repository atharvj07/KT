#include<bits/stdc++.h>
#define int long long
#define mem(ar,val) memset(ar,val,sizeof(ar))
#define pb(x) push_back(x)
using namespace std;
int p[200],suf[200],s[200];
int32_t main(){
    string q;
    mem(s,0);
    for(int i=0;i<26;++i){
        cin>>s[i];
    }
    cin>>q;
    int n=q.size();
    for(int i=0;i<n;++i){
        p[q[i]-'a']+=i;
    }
    for(int i=n-1;i>=0;--i){
        suf[q[i]-'a']+=n-1-i;
    }
    int ans=0;
    for(int i=0;i<n;++i){
        for(int j=0;j<26;++j){
            if(s[j]==0 or i==0)continue;
            int x=p[j]+suf[j]-i-i+1;
            if(x%s[j]==0)ans+=x/s[j]-1;
        }
    }
    cout<<ans<<endl;
}
