import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = s.length(), l = 0, r = n - 1, ans = n;
        int[] res = new int[n];
        for (int i = 0; i < n; ++i) {
            if (s.charAt(i) == 'R') {
                res[r--] = i + 1;
                if (l > r) {
                    --ans;
                    l = r = i + 1;
                }
            } else {
                res[l++] = i + 1;
                if (r < l) {
                    --ans;
                    r = l = i + 1;
                }
            }
        }
        System.out.println(ans);
        for (int val : res) System.out.print(val + " ");
        System.out.println();
    }
}
