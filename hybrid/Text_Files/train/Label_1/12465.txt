#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> x(n), y(n);
  for (int i = 0; i < n; i++) cin >> x[i];
  for (int i = 0; i < n; i++) cin >> y[i];
  int lastX = 0, lastY = 0;
  int xCnt = 0, yCnt = 0;
  for (int i = 0; i < n; i++) {
    if (xCnt + x[i] <= k) xCnt += x[i];
    else { lastX = xCnt; xCnt = x[i]; }
    if (yCnt + y[i] <= k) yCnt += y[i];
    else { lastY = yCnt; yCnt = y[i]; }
    if (xCnt > k || yCnt > k) { cout << "NO" << endl; return 0; }
  }
  if (xCnt + lastX > k || yCnt + lastY > k) cout << "NO" << endl;
  else cout << "YES" << endl;
  return 0;
}
