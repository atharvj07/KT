import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;

public class Main {
    public static void main(String[] args) throws NumberFormatException, IOException {
        BufferedReader stdin = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(stdin.readLine());

        int[][] want = new int[n][3];
        int[][] test = new int[n][3];
        for (int i = 0; i < n; i++) {
            String[] line = stdin.readLine().split(" ");
            want[i][0] = Integer.parseInt(line[0]);
            want[i][1] = Integer.parseInt(line[1]);
            want[i][2] = Integer.parseInt(line[2]);
        }
        for (int i = 0; i < n; i++) {
            String[] line = stdin.readLine().split(" ");
            test[i][0] = Integer.parseInt(line[0]);
            test[i][1] = Integer.parseInt(line[1]);
            test[i][2] = Integer.parseInt(line[2]);
        }

        boolean[][] canDeploy = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(canDeploy[i], false);
        }
        for (int i = 0; i < n; i++) {
            if (test[i][0] == 1) {
                canDeploy[0][i] = true;
            }
        }

        Queue<Integer> q = new LinkedList<>();
        q.add(0);
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (int i = 0; i < n; i++) {
                if (canDeploy[cur][i] && !canDeploy[0][i]) {
                    q.add(i);
                    canDeploy[0][i] = true;
                }
            }
        }

        boolean canDeployAll = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                if (want[i][j] == 1 && !canDeploy[0][i]) {
                    canDeployAll = false;
                    break;
                }
            }
        }

        if (!canDeployAll) {
            System.out.println("Impossible");
            return;
        }

        System.out.println("Possible");
        boolean[] ct = new boolean[n];
        for (int i = 0; i < n; i++) {
            ct[i] = test[i][1] == 0 && test[i][2] == 0;
        }

        int ctCount = 0;
        for (int i = 0; i < n; i++) {
            if (ct[i]) {
                ctCount++;
            }
        }

        System.out.println(Arrays.toString(ct).replaceAll("[\\[\\]]", "").replace(", ", " "));
        System.out.println(ctCount);

        int cdCount = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (canDeploy[i][j] && (ct[i] || ct[j])) {
                    System.out.println(i + 1 + " " + (j + 1));
                    cdCount++;
                }
            }
        }
    }
}
