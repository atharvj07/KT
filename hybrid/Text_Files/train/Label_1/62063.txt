def max_photos(n, a, b, T, photos):
    if photos[0] == 'w':
        T -= b
    if T < 1:
        return 0
    T -= 1
    left, right, max_photos = 0, n - 1, 1
    total_time, extra_time = T, [0]
    for i in range(1, n):
        cost = a + 1 + (b if photos[i] == 'w' else 0)
        if total_time >= cost:
            total_time -= cost
            max_photos += 1
            extra_time.append(extra_time[-1] + cost - a)
        else:
            break
    best = max_photos
    for i in range(max_photos, 0, -1):
        needed_time = extra_time[i] + (i + 1) * a
        while right - left > i and needed_time + (a + 1 + (b if photos[right] == 'w' else 0)) <= T:
            needed_time += a + 1 + (b if photos[right] == 'w' else 0)
            right -= 1
        best = max(best, right - left + i)
        if left + 1 < n:
            needed_time -= a + 1 + (b if photos[left + 1] == 'w' else 0)
            left += 1
    return best

n, a, b, T = map(int, input().split())
photos = input().strip()
print(max_photos(n, a, b, T, photos))
