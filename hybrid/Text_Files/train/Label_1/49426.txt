import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        char[] arr = in.next().toCharArray();
        int cnt = 0;
        int last = -1;
        for (char c : arr) {
            if (c != last) {
                cnt++;
            }
            last = c;
        }
        System.out.println(cnt - 1);
    }
}
