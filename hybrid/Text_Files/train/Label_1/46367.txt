import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j) - '0';
            }
        }
        boolean[] used = new boolean[n];
        boolean[] on = new boolean[m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == 1 && !on[j]) {
                    used[i] = true;
                    for (int k = 0; k < m; k++) {
                        if (a[i][k] == 1) {
                            on[k] = true;
                        }
                    }
                    break;
                }
            }
        }
        boolean ok = true;
        for (int i = 0; i < m; i++) {
            if (!on[i]) {
                ok = false;
                break;
            }
        }
        System.out.println(ok ? "YES" : "NO");
    }
}
