#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    adj[u].push_back(v), adj[v].push_back(u);
  }
  vector<bool> vis(n);
  set<int> a, b;
  for (int i = 0; i < n; i++) {
    if (vis[i]) continue;
    vis[i] = true;
    int cnt = 1;
    a.insert(i);
    vector<int> q = {i};
    while (!q.empty()) {
      int u = q.back();
      q.pop_back();
      for (int v : adj[u]) {
        if (!vis[v]) {
          vis[v] = true;
          cnt++;
          a.insert(v);
          q.push_back(v);
        }
      }
    }
    if (cnt * 2 > n) {
      cout << -1 << endl;
      return 0;
    }
    for (int j = 0; j < n; j++) {
      if (vis[j]) continue;
      vis[j] = true;
      int cnt = 1;
      b.insert(j);
      vector<int> q = {j};
      while (!q.empty()) {
        int u = q.back();
        q.pop_back();
        for (int v : adj[u]) {
          if (!vis[v]) {
            vis[v] = true;
            cnt++;
            b.insert(v);
            q.push_back(v);
          }
        }
      }
      if (cnt * 2 > n) {
        cout << -1 << endl;
        return 0;
      }
    }
    break;
  }
  cout << a.size() << endl;
  for (int i : a) { cout << i + 1 << " "; }
  cout << endl;
  cout << b.size() << endl;
  for (int i : b) { cout << i + 1 << " "; }
  cout << endl;
  return 0;
}
