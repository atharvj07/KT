n, k = map(int, input().split())
if (k * (n - 1) & 1) or (n - 1) * k > n * (n - 1) / 2:
    print(-1)
else:
    m = n * (n - 1) / 2
    print(int(m))
    for i in range(2, n + 1):
        for j in range(1, i):
            if ((i - j) % 2 == 0 or k % 2 == 0) and k > 0:
                k -= 1
                print(j, i)
