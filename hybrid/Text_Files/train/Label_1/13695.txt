#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, q, k, res;
    long long a, cur;
    cin >> n >> q;
    vector<long long> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    vector<long long> l(q);
    for (int i = 0; i < q; i++) cin >> l[i];
    for (int i = 0, j = 0; i < q; i++) {
        for (; j < n; j++) {
            cur += l[i];
            if (cur > v[j]) res = j;
            else break;
        }
        res++;
        if (res == 1 && i != 0) res = 0;
        cout << res << "\n";
        j = 0;
        cur = 0;
    }
}
