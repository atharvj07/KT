import sys
input = sys.stdin.readline

t = int(input())
for i in range(t):
    a, b = map(int, input().split())
    if not a or not b:
        print(max(a,b), 0)
    elif a==b:
        print(a, 0)
    else:
        g = a
        small = min(a,b)
        a %= small
        while a:
            g, a = a, g%a
        print(g, small//g-1)
