import sys
input=sys.stdin.readline
n,x,y=map(int,input().split())
g=[[] for _ in range(n+1)]
for _ in range(n-1):
    a,b=map(int,input().split())
    g[a].append(b)
    g[b].append(a)
d=[0]*(n+1)
def dfs(c,p):
    for t in g[c]:
        if t!=p:
            dfs(t,c)
            d[c]+=d[t]
    if c==x:
        d[c]+=1
dfs(x,x)
ans=0
for t in g[x]:
    ans+=d[t]*(n-1-d[t])
for t in g[y]:
    ans+=d[t]*(n-1-d[t]-d[y]+d[t])
print(ans)
