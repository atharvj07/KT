#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];

  sort(a.begin(), a.end());

  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (i + 1 < n && a[i] == a[i + 1]) i++;
    ans += a[i];
  }

  cout << ans << endl;
  return 0;
}
