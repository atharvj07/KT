import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.next();
        int n = s.length();
        int[] cnt = new int[n];
        for (int i = 0; i < n; i++) {
            cnt[i] = 1;
            for (int j = i + 1; j < n; j++) {
                if (s.charAt(i) == s.charAt(j)) {
                    cnt[i]++;
                } else {
                    break;
                }
            }
        }
        double ans = 0;
        for (int i = 0; i < n; i++) {
            ans += 1.0 / cnt[i];
        }
        System.out.println(ans / n);
    }
}
