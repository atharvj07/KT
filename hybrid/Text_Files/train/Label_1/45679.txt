import java.util.*;
import java.io.*;

class Main {
    static final int MOD = 1000000007;

    static long[][] pref;
    static long[] hash;
    static char[] s;

    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        s = ("a" + sc.nextLine()).toCharArray();
        hash = new long[s.length];
        long p = 31;
        hash[0] = s[0] - 'a' + 1;
        for (int i = 1; i < s.length; i++) {
            hash[i] = ((hash[i - 1] * p) % MOD + s[i] - 'a' + 1) % MOD;
        }

        long[][] pow = new long[s.length][];
        for (int i = 0; i < s.length; i++) {
            pow[i] = new long[s.length];
            pow[i][0] = 1;
        }
        for (int i = 1; i < s.length; i++) {
            for (int j = 1; j < s.length; j++) {
                pow[i][j] = (pow[i][j - 1] * p) % MOD;
            }
        }

        pref = new long[n + 1][s.length];
        for (int i = 0; i < s.length; i++) {
            pref[1][i] = hash[i];
        }
        for (int i = 2; i <= n; i++) {
            long cur = pref[i - 1][s.length - 1];
            for (int j = s.length - 2; j >= 0; j--) {
                cur = (cur - hash[j] * pow[s.length - j - 1][i - 1] % MOD + MOD) % MOD;
                pref[i][j] = (pref[i - 1][j] + cur) % MOD;
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < m; i++) {
            char[] q = sc.nextLine().toCharArray();
            long val = 0;
            p = 31;
            for (int j = q.length - 1; j >= 0; j--) {
                val = (val * p + q[j] - 'a' + 1) % MOD;
                if (j > 0) {
                    p = (p * p) % MOD;
                }
            }
            int idx = -1;
            for (int j = 0; j + q.length - 1 < s.length; j++) {
                long cur = pref[Math.min(n, j + q.length)][j];
                if (cur == val) {
                    idx = j;
                    break;
                }
            }
            sb.append(idx < 0 ? 0 : pref[n][idx] + MOD - pref[n][idx + q.length - 1])
                .append('\n');
        }
        System.out.print(sb);
    }
}
