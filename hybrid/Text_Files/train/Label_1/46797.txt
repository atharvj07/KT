import java.util.Scanner;

public class Main {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), l = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }

        int[] dp = new int[n + 1];
        dp[n] = 1;
        for (int i = n - 1; i >= 1; i--) {
            dp[i] = dp[i + 1];
            if (a[i] == a[i + 1] - 1) {
                dp[i] += dp[i + 1];
                if (dp[i] >= MOD) dp[i] -= MOD;
            }
        }

        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i] != a[i - 1] + 1) ans += dp[i];
            if (ans >= MOD) ans -= MOD;
        }

        System.out.println(ans);
    }
}
