from sys import stdin

n, m = map(int, stdin.readline().split())
a = [list(map(int, stdin.readline().split())) for _ in range(n)]
impossible = False
for i in range(1, n - 1):
    for j in range(1, m - 1):
        if a[i][j] == 0:
            if a[i][j + 1] != 0 and a[i][j - 1] != 0 and a[i + 1][j] != 0 and a[i - 1][j] != 0:
                impossible = True
            if a[i + 1][j] == 0:
                a[i + 1][j] = a[i][j - 1] + a[i][j + 1] - a[i][j]
            if a[i - 1][j] == 0:
                a[i - 1][j] = a[i][j + 1] + a[i][j - 1] - a[i][j]
            if a[i][j + 1] == 0:
                a[i][j + 1] = a[i + 1][j] + a[i - 1][j] - a[i][j]
            if a[i][j - 1] == 0:
                a[i][j - 1] = a[i + 1][j] + a[i - 1][j] - a[i][j]
if impossible:
    print(-1)
else:
    print(sum(sum(row) for row in a))
