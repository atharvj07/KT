import sys
input = sys.stdin.readline

n, m = list(map(int, input().split()))
a = [[int(x) for x in input().split()] for _ in range(n)]
k = int(input())
for l, r in [list(map(int, input().split())) for _ in range(k)]:
    l -= 1
    r -= 1
    for j in range(m):
        f = 1
        for i in range(l, r):
            if a[i][j] > a[i+1][j]:
                f = 0
                break
        if f:
            print("Yes")
            break
    else: print("No")
