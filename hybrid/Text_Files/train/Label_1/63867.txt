n=int(input())
a=list(map(int,input().split()))
mod=998244353
a.insert(0,0)
dp=[[[0]*3 for j in range(n+2)] for i in range(n+2)]

dp[n][n][0]=dp[n][n][1]=dp[n][n][2]=1
for i in range(n-1,-1,-1):
    for j in range(i+1,n+1):
        dp[i][j][0]=(dp[i][j][0]+dp[i+1][j][0]+dp[i+1][j][1]+dp[i+1][j][2])%mod
        dp[i][j][1]=(dp[i][j][1]+dp[i+1][j][2])%mod
        dp[i][j][2]=(dp[i][j][2]+dp[i+1][j][1])%mod
        dp[i][j][1]=(dp[i][j][1]+a[j]-a[i])%mod
        dp[i][j][2]=(dp[i][j][2]+a[j]-a[i])%mod

print((dp[1][n][0]+dp[1][n][1]+dp[1][n][2])*pow(2,n-1,mod)%mod)
