#include <iostream>
#include <queue>
#include <vector>
using namespace std;

int main() {
  long long n, m, a, b, i, x;
  priority_queue<long double> q;
  cin >> n >> m;
  while (n--) {
    cin >> a >> b;
    q.push(a * 1.0 / b);
  }
  for (i = 0, x = 0; i < m; i++) {
    x += (long long)(q.top() * 1000000 + 0.5);
    q.pop();
    q.push(q.top() / 2);
  }
  cout << (long double)x / 1000000 << endl;
  return 0;
}
