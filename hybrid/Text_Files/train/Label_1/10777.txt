#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<pair<int, int>> segs(n);
        for (int i = 0; i < n; i++) {
            int l, r;
            cin >> l >> r;
            segs[i] = {l, r};
        }
        sort(segs.begin(), segs.end());
        int ans = segs[0].second - segs[0].first;
        for (int i = 1; i < n; i++) {
            ans = min(ans, segs[i].second - segs[i - 1].first);
        }
        cout << ans << endl;
    }
    return 0;
}
