#include<bits/stdc++.h>
using namespace std;
int a[1001][1001], n, c1, c2;
bool vis[1001];
void dfs(int c){
    if(vis[c])return;
    vis[c] = true;
    for(int i=1; i<=n; i++){
        if(i==c||!a[c][i])continue;
        dfs(i);
    }
}
int main(){
    cin >> n >> c1 >> c2;
    if(c1==c2){
        cout << "NO" << endl;
        return 0;
    }
    int cnt=0;
    if(c1==1){
        if(n==2){
            cout << "YES" << endl << "01" << endl << "10" << endl;
            return 0;
        }
        int k=n/2;
        while(k--){
            a[cnt+1][cnt+2]=a[cnt+2][cnt+1]=1;
            cnt+=2;
        }
        if(n%2){
            a[n][1]=a[1][n]=1;
        }
    }
    else if(c1==n){
        int k = n/2;
        while(k--){
            a[cnt+1][cnt+2]=a[cnt+2][cnt+1]=1;
            cnt+=2;
        }
        if(n%2){
            a[n][1]=a[1][n]=1;
        }
    }
    else{
        int k = n/2;
        while(k--){
            a[cnt+1][cnt+2]=a[cnt+2][cnt+1]=1;
            cnt+=2;
        }
        if(n%2){
            a[n][1]=a[1][n]=1;
        }
        k = (n-c1)/2;
        cnt=n;
        while(k--){
            while(vis[cnt]==true){
                cnt--;
            }
            a[cnt][n]=a[n][cnt]=1;
            dfs(cnt);
        }
    }
    cout << "YES" << endl;
    for(int i=1; i<=n; i++){
        for(int j=1; j<=n; j++){
            cout << a[i][j];
        }
        cout << endl;
    }
    return 0;
}
