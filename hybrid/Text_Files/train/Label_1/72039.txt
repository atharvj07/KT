import sys

t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    i = 0
    j = 1
    k = 2
    while i != k and k != n-1:
        if a[k] - a[i] > a[j] - a[i] or a[k] - a[j] > a[j] - a[i]:
            i += 1
            j += 1
            k += 1
        else:
            break
    if i == k or k == n-1:
        print(-1)
    else:
        print(*[i+1, j+1, k+1])
