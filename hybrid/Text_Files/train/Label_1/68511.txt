from itertools import permutations

def can_balance(k, boxes):
    # Calculate the sum of all numbers in all boxes
    total_sum = sum(sum(box) for box in boxes)
    
    # If total sum is not divisible by k, it's not possible to balance
    if total_sum % k != 0:
        return "No"
    
    # Target sum for each box
    target_sum = total_sum // k
    
    # Prepare list for storing solutions
    solutions = []
    
    # Generate all permutations of box indices
    for perm in permutations(range(k), k):
        moves = []
        success = True
        for i, to_box in enumerate(perm):
            # Calculate current and target sums for the box
            current_sum = sum(boxes[i])
            to_sum = sum(boxes[to_box])
            
            # Check if there is a number that can be moved
            found = False
            for num in boxes[i]:
                if current_sum - num + to_sum == target_sum and to_sum + num - current_sum == target_sum:
                    found = True
                    moves.append((num, to_box + 1))
                    # Update the sums as if the move was made
                    to_sum += num
                    current_sum -= num
                    break
            if not found:
                success = False
                break
        
        if success:
            return ["Yes"] + moves

    return ["No"]

# Input reading
k = int(input().strip())
boxes = [list(map(int, input().strip().split()))[1:] for _ in range(k)]

# Solve
result = can_balance(k, boxes)

# Output
if result == "No":
    print(result)
else:
    print(result[0])
    for move in result[1:]:
        print(*move)
