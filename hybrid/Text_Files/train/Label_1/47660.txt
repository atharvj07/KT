import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        int[][] stu = new int[n][2];
        int[][] chk = new int[m][2];
        for (int i = 0; i < n; i++) {
            line = br.readLine().split(" ");
            stu[i][0] = Integer.parseInt(line[0]);
            stu[i][1] = Integer.parseInt(line[1]);
        }
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            chk[i][0] = Integer.parseInt(line[0]);
            chk[i][1] = Integer.parseInt(line[1]);
        }
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            int min = Integer.MAX_VALUE;
            int idx = -1;
            for (int j = 0; j < m; j++) {
                int d = Math.abs(stu[i][0] - chk[j][0]) + Math.abs(stu[i][1] - chk[j][1]);
                if (d < min) {
                    min = d;
                    idx = j;
                } else if (d == min) {
                    idx = Math.min(idx, j);
                }
            }
            ans[i] = idx + 1;
        }
        for (int x : ans) {
            System.out.println(x);
        }
    }
}
