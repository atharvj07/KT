import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n;
        while ((n = sc.nextInt()) != 0) {
            int cnt = 0;
            while (n != 1) {
                n = (n % 2 == 0) ? n / 2 : n * 3 + 1;
                cnt++;
                if (n > 1000000) {
                    cnt = -1;
                    break;
                }
            }
            System.out.println(cnt);
        }
    }
}
