#include <bits/stdc++.h>
#define ll long long
using namespace std;

string a, b;
int main() {
  cin >> a >> b;
  reverse(a.begin(), a.end());
  reverse(b.begin(), b.end());
  if (a.size() < b.size()) {
    swap(a, b);
  }
  while (!b.empty() && a[a.size() - 1] == b[b.size() - 1]) {
    a.pop_back();
    b.pop_back();
  }
  reverse(a.begin(), a.end());
  reverse(b.begin(), b.end());
  while (!a.empty() && a[0] == '0') {
    a.erase(0, 1);
  }
  while (!b.empty() && b[0] == '0') {
    b.erase(0, 1);
  }
  if (a.empty() || b.empty()) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl << a << endl << b << endl;
  }
  return 0;
}
