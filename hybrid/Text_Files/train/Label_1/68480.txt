from itertools import permutations

def find_divisible_by_7(a):
    # Extract digits 1, 6, 8, 9 and the rest of the digits
    special_digits = ['1', '6', '8', '9']
    other_digits = [digit for digit in a if digit not in special_digits]
    special_permutations = permutations(special_digits)
    
    # Try to find a permutation of the special digits that forms a number divisible by 7
    for perm in special_permutations:
        for i in range(len(other_digits) + 1):
            # Insert the permutation into the other digits at different positions
            temp_digits = other_digits[:i] + list(perm) + other_digits[i:]
            temp_number = int(''.join(temp_digits))
            if temp_number % 7 == 0:
                return temp_number
    return 0

# Input
a = input().strip()

# Output
print(find_divisible_by_7(a))
