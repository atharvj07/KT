import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int k = sc.nextInt();
        int ans = 0;
        for (long i = 1; i <= n; i++) {
            String s = String.valueOf(i);
            int cnt = 0;
            for (char c : s.toCharArray()) {
                if (c != '0') {
                    cnt++;
                }
            }
            if (cnt == k) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
