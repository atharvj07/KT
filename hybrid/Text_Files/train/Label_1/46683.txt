import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();
        int[] l = new int[n];
        int[] r = new int[n];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'L') {
                l[i]++;
            } else {
                r[i]++;
            }
        }
        for (int i = 1; i < n; i++) {
            l[i] += l[i - 1];
            r[i] += r[i - 1];
        }
        int max = 0;
        for (int i = 0; i <= k; i++) {
            int happy = l[i] + r[n - 1] - r[i];
            if (i > 0) {
                happy -= l[i - 1];
            }
            max = Math.max(max, happy);
        }
        System.out.println(max);
    }
}
