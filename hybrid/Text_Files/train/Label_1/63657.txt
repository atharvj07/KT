for _ in range(int(input())):
    a, b = map(int, input().split())
    ans = ([0, 1][a > b], [1, 0][a > b])
    c = min(a, b)
    if a == c and b != c:
        print(ans[0], ans[1] + c - 1)
    elif a != c and b == c:
        print(ans[0] + c - 1, ans[1])
    else:
        print(ans[0] + c - 1, ans[1] + c - 1)
