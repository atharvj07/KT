import sys
 
n = int(sys.stdin.readline())
k = list(map(int, sys.stdin.readline().split()))
t = []
for i in range(n):
    t.append([int(j) for j in sys.stdin.readline().split()])
 
ans = 99999999999
for i in range(n):
    x = 0
    for j in range(k[i]):
        x += t[i][j] * 5
    x += 15 * k[i]
    ans = min(ans, x)
print(ans)
