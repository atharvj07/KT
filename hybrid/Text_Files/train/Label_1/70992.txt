from collections import defaultdict
for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    d = defaultdict(list)
    for i in range(n):
        d[a[i]].append(i)
    c = 1
    ans = [0] * n
    for i in sorted(d):
        for j in d[i]:
            ans[j] = c
        c += 1
    print(c - 1)
    print(*ans)
