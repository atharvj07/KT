#include <bits/stdc++.h>
using namespace std;
char a[2001][2001];
int vis[2001][2001];
int dx[4] = {1, -1, 0, 0};
int dy[4] = {0, 0, 1, -1};
int n, m;
bool valid(int x, int y) {
  if (x < 1 || x > n || y < 1 || y > m)
    return false;
  if (vis[x][y])
    return false;
  return true;
}
void dfs(int x, int y, int size) {
  vis[x][y] = 1;
  if (size == 1) {
    for (int i = 0; i < 4; i++) {
      int nx = x + dx[i], ny = y + dy[i];
      if (valid(nx, ny) && a[nx][ny] == 'w')
        dfs(nx, ny, size);
    }
  } else {
    for (int i = 0; i < 4; i++) {
      int nx = x + dx[i], ny = y + dy[i];
      if (valid(nx, ny) && a[nx][ny] == '.')
        dfs(nx, ny, size - 1);
    }
  }
}
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cin >> a[i][j];
    }
  }
  int ans = -1;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (a[i][j] == 'w') {
        memset(vis, 0, sizeof vis);
        dfs(i, j, max(n, m));
        ans = 1;
        for (int k = 1; k <= n; k++) {
          for (int l = 1; l <= m; l++) {
            if (a[k][l] == 'w' && !vis[k][l]) {
              ans = -1;
              break;
            }
          }
        }
        for (int k = 1; k <= n; k++) {
          for (int l = 1; l <= m; l++) {
            if (vis[k][l] && a[k][l] == '.') {
              a[k][l] = '+';
            }
          }
        }
      }
    }
  }
  if (ans == -1)
    cout << "-1" << endl;
  else {
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= m; j++) {
        cout << a[i][j];
      }
      cout << endl;
    }
  }
  return 0;
}
