#include <bits/stdc++.h>
using namespace std;
#define ll long long

ll a, d[300005], c[300005], dp[300005];
int main() {
  ll n;
  cin >> n >> a;
  for (ll i = 1; i <= n; i++) {
    cin >> d[i] >> c[i];
    dp[i] = -1e18;
  }
  dp[0] = 0;
  ll mx = 0;
  for (int i = 1; i <= n; i++) {
    dp[i] = max(dp[i], dp[i - 1]);
    dp[i] = max(dp[i], mx - (d[i] - d[i - 1]) * (d[i] - d[i - 1]) + a);
    mx = max(mx, dp[i] + (d[i] - d[i - 1]) * (d[i] - d[i - 1]) - a);
  }
  cout << *max_element(dp, dp + n + 1);
  return 0;
}
