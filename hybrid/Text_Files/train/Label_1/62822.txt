from collections import defaultdict
from heapq import heapify, heappush, heappop

n, m = map(int, input().split())
v = list(map(int, input().split()))
d = defaultdict(list)
for _ in range(m):
    x, y = map(int, input().split())
    d[x].append(y)
    d[y].append(x)
    
q = []
heapify(q)
heappush(q, (v[0], 1))
vis = [False] * (n + 1)
vis[1] = True
res = 0

while q:
    v, x = heappop(q)
    if x == n:
        break
    res += v
    for y in d[x]:
        if vis[y] == True:
            continue
        vis[y] = True
        heappush(q, (v + v[y], y))

print(res)
