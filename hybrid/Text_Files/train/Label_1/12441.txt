#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod=998244353;
int qpow(int a,int b){
    int ans=1;
    for(;b;b>>=1){
        if(b&1)ans=(ans*a)%mod;
        a=(a*a)%mod;
    }
    return ans;
}
int32_t main() {
    int n;
    cin>>n;
    int ans=1;
    for(int i=1;i<=n;i++)ans=(ans*i)%mod;
    ans=(ans*qpow(2,mod-2))%mod;
    cout<<ans<<endl;
    return 0;
}
