import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class Warehouse {

    private static class Shelf {
        private List<String> boxes;

        public Shelf(int width) {
            boxes = new ArrayList<>(width);
            for (int i = 0; i < width; i++) {
                boxes.add(null);
            }
        }

        public boolean addBox(String id) {
            for (int i = 0; i < boxes.size(); i++) {
                if (boxes.get(i) == null) {
                    boxes.set(i, id);
                    return true;
                }
            }
            return false;
        }

        public String removeBox(String id) {
            for (int i = 0; i < boxes.size(); i++) {
                if (boxes.get(i).equals(id)) {
                    String removedBox = boxes.get(i);
                    boxes.set(i, null);
                    return removedBox;
                }
            }
            return null;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(); // Number of shelves
        int m = scanner.nextInt(); // Width of each shelf
        int k = scanner.nextInt(); // Number of operations

        List<Shelf> shelves = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            shelves.add(new Shelf(m));
        }

        Map<String, Integer[]> boxLocations = new HashMap<>();

        for (int i = 0; i < k; i++) {
            String operation = scanner.next();
            if (operation.equals("+1")) {
                int shelfIndex = scanner.nextInt() - 1;
                int sectionIndex = scanner.nextInt() - 1;
                String id = scanner.next();

                if (!shelves.get(shelfIndex).addBox(id)) {
                    for (int j = shelfIndex + 1; j < n; j++) {
                        if (shelves.get(j).addBox(id)) {
                            boxLocations.put(id, new Integer[]{j + 1, sectionIndex + 1});
                            break;
                        }
                    }
                } else {
                    boxLocations.put(id, new Integer[]{shelfIndex + 1, sectionIndex + 1});
                }
            } else if (operation.equals("-1")) {
                String id = scanner.next();
                Integer[] location = boxLocations.remove(id);
                if (location != null) {
                    System.out.println(location[0] + " " + location[1]);
                } else {
                    System.out.println("-1 -1");
                }
            }
        }
    }
}
