#include <cstdio>
#include <cstring>
#include <algorithm>
#include <vector>

int n, m, s, a[100005], b[100005], c[100005], res[100005], id[100005], vis[100005], sum;
struct node
{
    int x, y, z;
};
std::vector<node> ans;

bool ok(int val)
{
    memset(vis, 0, sizeof(vis));
    sum = 0;
    for (int i = 1; i <= m; i++)
    {
        res[i] = -1;
        if (b[val] < a[i])
            continue;
        for (int j = 1; j <= n; j++)
            if (b[j] >= a[i] && !vis[j] && sum + c[j] <= s)
            {
                res[i] = j;
                vis[j] = 1;
                sum += c[j];
                break;
            }
    }
    for (int i = 1; i <= m; i++)
        if (res[i] == -1)
            return false;
    return true;
}

int main()
{
    scanf("%d%d%d", &n, &m, &s);
    for (int i = 1; i <= m; i++)
        scanf("%d", &a[i]);
    for (int i = 1; i <= n; i++)
        scanf("%d", &b[i]);
    for (int i = 1; i <= n; i++)
        scanf("%d", &c[i]);
    int l = 1, r = n, sol = -1;
    while (l <= r)
    {
        int mid = (l + r) / 2;
        if (ok(mid))
        {
            sol = mid;
            l = mid + 1;
        }
        else
            r = mid - 1;
    }
    if (sol == -1)
        printf("NO\n");
    else
    {
        ok(sol);
        printf("YES\n");
        for (int i = 1; i <= m; i++)
            printf("%d ", res[i]);
        printf("\n");
    }
    return 0;
}
