#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  long long sum = 0;
  long long mx = 0;
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    sum += a;
    mx = max(mx, 1ll * a);
  }
  cout << max(0ll, mx * (k - 1) - (sum - mx)) << endl;
  return 0;
}
