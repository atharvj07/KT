import sys

input = sys.stdin.readline
n, k = map(int, input().split())
tree = [[] for _ in range(n + 1)]
a = list(map(int, input().split()))
for _ in range(n - 1):
    u, v = map(int, input().split())
    tree[u].append(v)
    tree[v].append(u)

dp = [[-1] * 3 for _ in range(n + 1)]


def dfs(u, p):
    dp[u][0] = a[u - 1]
    dp[u][1] = 0
    dp[u][2] = 0

    for v in tree[u]:
        if v == p:
            continue
        dfs(v, u)

        dp[u][0] += max(dp[v][1], dp[v][2])
        dp[u][2] += dp[v][1]

    # Try taking u and its children that maximizes the weight
    # Try not taking u and take its children
    dp[u][1] = dp[u][0]
    dp[u][2] = max(dp[u][0], dp[u][2])

    for v in tree[u]:
        if v == p:
            continue
        # Try taking u and taking one of v's subtree
        dp[u][1] = max(dp[u][1], dp[u][0] - max(dp[v][1], dp[v][2]) + dp[v][2])

    return dp[u]


ans = dfs(1, 1)
print(max(ans[1], ans[2]))
