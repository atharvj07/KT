import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    int n = sc.nextInt();
    int[] a = new int[n];
    int min = Integer.MAX_VALUE;
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
      min = Math.min(min, a[i]);
    }

    int count = 0;
    while (min % 2 == 0) {
      count++;
      min /= 2;
      for (int i = 0; i < n; i++) {
        a[i] /= 2;
      }
    }

    System.out.println(count);
  }
}
