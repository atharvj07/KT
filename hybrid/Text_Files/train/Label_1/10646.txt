#include <bits/stdc++.h>
#define MOD 998244353
#define ll long long
#define fi first
#define se second
typedef std::pair<int, int> pii;
const int N = 2e5 + 5;
std::vector<int> a;
int fact[N];
void init() {
    fact[0] = 1;
    for (int i = 1; i < N; ++i)
        fact[i] = (1ll * fact[i - 1] * i) % MOD;
}
int qpow(int a, int b) {
    int ans = 1;
    while (b) {
        if (b & 1) ans = (1ll * ans * a) % MOD;
        a = (1ll * a * a) % MOD;
        b >>= 1;
    }
    return ans;
}
int inv(int a) { return qpow(a, MOD - 2); }
int calc(int n) {
    if (a[n] > 1) return 0;
    return fact[n] - 1 < 0 ? fact[n] - 1 + MOD : fact[n] - 1;
}
int main() {
    init();
    int t;
    std::cin >> t;
    while (t--) {
        a.clear();
        int n;
        std::cin >> n;
        a.emplace_back(0);
        for (int i = 0; i < n; ++i) {
            int x;
            std::cin >> x;
            if (x & 1) x += MOD - 1;
            x >>= 1;
            a.emplace_back(x);
        }
        std::sort(a.rbegin(), a.rend());
        int ans = fact[n];
        for (int i = 1; i <= n; ++i)
            ans = (1ll * ans * inv(calc(i))) % MOD;
        std::cout << (ans + MOD) % MOD << '\n';
    }
    return 0;
}
