#include <vector>
#include <iostream>
#include <queue>
using namespace std;
int n;
struct edge {
    int to, c;
    edge(int to, int c) : to(to), c(c) {}
};
vector<vector<edge>> adj;
vector<int> d;
void dfs(int v, int p, int _d) {
    d[v] = _d;
    for (auto e : adj[v]) {
        int to = e.to, c = e.c;
        if (to == p) continue;
        dfs(to, v, _d + c);
    }
}
int main() {
    cin >> n;
    adj.resize(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int a, b, c; cin >> a >> b >> c;
        adj[a].push_back(edge(b, c));
        adj[b].push_back(edge(a, c));
    }
    d.resize(n + 1);
    dfs(1, 1, 0);
    int mx = 0, leaf = 0;
    for (int i = 1; i <= n; i++) {
        if (adj[i].size() == 1) {
            if (d[i] > mx) {
                mx = d[i];
                leaf = i;
            }
        }
    }
    d.clear();
    d.resize(n + 1);
    dfs(leaf, leaf, 0);
    mx = 0;
    for (int i = 1; i <= n; i++) {
        mx = max(mx, d[i]);
    }
    cout << mx + 1 << endl;
    return 0;
}
