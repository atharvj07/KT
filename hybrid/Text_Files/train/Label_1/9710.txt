#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
const int mxN = 1e5 + 10;
int freq[mxN];
int main() {
    string s; cin >> s;
    int n = s.size();
    for (int i = 0; i < n; i++) { freq[s[i]]++; }
    int k = n;
    for (int i = 0; i < 128; i++) {
        if (freq[i] == n) { k = 1; }
        if (freq[i] >= n - 1) { k = 2; }
    }
    cout << k << endl;
}
