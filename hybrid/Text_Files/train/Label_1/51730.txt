import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        String[] strings = new String[n];
        for (int i = 0; i < n; i++) {
            strings[i] = sc.nextLine();
        }
        String product = "";
        for (int i = 0; i < n; i++) {
            product += strings[i];
            for (int j = 0; j < strings[i].length(); j++) {
                product += strings[i].charAt(j);
            }
        }
        int longestBeauty = 0;
        char currentChar = product.charAt(0);
        int currentBeauty = 1;
        for (int i = 1; i < product.length(); i++) {
            if (product.charAt(i) == currentChar) {
                currentBeauty++;
            } else {
                longestBeauty = Math.max(longestBeauty, currentBeauty);
                currentBeauty = 1;
                currentChar = product.charAt(i);
            }
        }
        longestBeauty = Math.max(longestBeauty, currentBeauty);
        System.out.println(longestBeauty);
    }
}
