import java.util.*;
public class Main {
    static int n, m;
    static int[][] grid;
    static boolean[][] visited;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        grid = new int[n][m];
        visited = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = sc.nextInt();
            }
        }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!visited[i][j]) {
                    if (dfs(i, j, grid[i][j]) != -1)
                        cnt++;
                }
            }
        }
        System.out.println(cnt);
    }
    static int dfs(int i, int j, int l) {
        if (!inBound(i, j)) return -1;
        if (visited[i][j] || grid[i][j] < l) return -1;
        visited[i][j] = true;
        int ret = grid[i][j];
        ret = Math.max(ret, dfs(i + 1, j, l));
        ret = Math.max(ret, dfs(i - 1, j, l));
        ret = Math.max(ret, dfs(i, j + 1, l));
        ret = Math.max(ret, dfs(i, j - 1, l));
        return ret;
    }
    static boolean inBound(int i, int j) {
        return i >= 0 && i < n && j >= 0 && j < m;
    }
}
