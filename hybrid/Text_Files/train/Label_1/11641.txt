#include<bits/stdc++.h>
using namespace std;
int main() {
  int t; cin >> t;
  while (t--) {
    int n, k; cin >> n >> k; 
    int lmin, lmax; cin >> lmin; lmax = lmin;
    for(int i = 1; i < n; ++i) {
      int h; cin >> h;
      lmin = max(lmin - h, 0);
      lmax = max(lmax, h);
    }
    if (lmax - lmin <= k) cout << "YES\n";
    else cout << "NO\n";
  }
}
