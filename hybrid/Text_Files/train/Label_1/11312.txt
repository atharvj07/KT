#include <bits/stdc++.h>
using namespace std;
int main() {
  int l;
  cin >> l;
  string n;
  cin >> n;
  int ans = atoi(n.substr(0, l / 2).c_str()) + atoi(n.substr(l / 2, l - l / 2).c_str());
  for (int i = 1; i < l - 1; i++) {
    int x = atoi(n.substr(0, i).c_str());
    int y = atoi(n.substr(i, l - i).c_str());
    ans = min(ans, x + y);
  }
  cout << ans << endl;
}
