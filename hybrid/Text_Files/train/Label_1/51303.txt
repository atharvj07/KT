import java.util.Scanner;

public class LiegesOfLegendre {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        String winner = solve(n, k, a);
        System.out.println(winner);
    }

    private static String solve(int n, int k, int[] a) {
        boolean[] visited = new boolean[n];
        int[] count = new int[n];
        for (int i = 0; i < n; i++) {
            count[i] = a[i];
        }

        while (true) {
            int index = -1;
            for (int i = 0; i < n; i++) {
                if (!visited[i] && count[i] > 0) {
                    index = i;
                    break;
                }
            }

            if (index == -1) {
                break;
            }

            visited[index] = true;
            count[index]--;

            if (count[index] > 0 && count[index] % 2 == 0) {
                int newCount = count[index] / 2;
                count[index] = 0;
                for (int i = 0; i < k; i++) {
                    visited[n + i] = false;
                    count[n + i] = newCount;
                }
                n += k;
            }
        }

        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += count[i];
        }

        return sum % 2 == 0 ? "Nicky" : "Kevin";
    }
}
