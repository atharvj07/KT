import java.util.Scanner;

public class MagicSubarrays {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        int magicalSubarrays = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                if (isMagical(arr, i, j)) {
                    magicalSubarrays++;
                }
            }
        }

        System.out.println(magicalSubarrays);
    }

    private static boolean isMagical(int[] arr, int i, int j) {
        int min = arr[i];
        int max = arr[i];
        for (int k = i + 1; k <= j; k++) {
            min = Math.min(min, arr[k]);
            max = Math.max(max, arr[k]);
        }
        return min == max;
    }
}
