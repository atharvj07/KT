#include<iostream>
#include<vector>
using namespace std;
int main() {
    int n, x;
    cin >> n >> x;
    vector<int> v(n);
    for(int i = 0; i < n; i++) {
        char c;
        int a;
        cin >> c >> a;
        v[i] = (c == '+' ? a : -a);
    }
    int distress = 0, sum = x;
    for(auto i : v) {
        if(i > 0) {
            sum += i;
        }else if(sum + i >= 0) {
            sum += i;
        }else {
            sum = 0;
            distress++;
        }
    }
    cout << sum << ' ' << distress << endl;
    return 0;
}
