import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] arr = sc.next().toCharArray();
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 'X') cnt++;
        }
        int ans = Math.abs(cnt - n / 2);
        System.out.println(ans);
        for (int i = 0; i < n; i++) {
            if (cnt < n / 2) {
                if (arr[i] == 'x') {
                    arr[i] = 'X';
                    cnt++;
                }
            } else {
                if (arr[i] == 'X') {
                    arr[i] = 'x';
                    cnt--;
                }
            }
        }
        System.out.println(new String(arr));
    }
}
