n=int(input())
s=[c for c in input()]
a=s.count('a')
b=s.count('b')
x=min(a,b)
if x==a:y=b-x
else:y=a-x
print(y)
for i in range(n):
    if y<=0:break
    if s[i]=='a' and b>a:
        s[i]='b'
        b-=1
        y-=1
    elif s[i]=='b' and a>b:
        s[i]='a'
        a-=1
        y-=1
print(''.join(s))
