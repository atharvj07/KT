import sys
input = sys.stdin.readline

n, m = list(map(int, input().split()))

graph = [[] for _ in range(n+1)]
for _ in range(m):
    a, b = list(map(int, input().split()))
    graph[a].append(b)
    graph[b].append(a)

s1, t1, l1 = list(map(int, input().split()))
s2, t2, l2 = list(map(int, input().split()))

visited = [False] * (n+1)
def bfs(s, l):
    queue = [(s, 0)]
    while queue:
        node, dist = queue.pop(0)
        if dist > l:
            continue
        if node==t1 or node==t2:
            return True
        if visited[node]:
            continue
        visited[node] = True
        for next in graph[node]:
            queue.append((next, dist+1))
    return False

visited = [False] * (n+1)
max_destroyed = 0
for u in range(1, n+1):
    for v in graph[u]:
        if v > u:
            graph[u].remove(v)
            graph[v].remove(u)
            visited = [False] * (n+1)
            if not bfs(s1, l1) or not bfs(s2, l2):
                graph[u].append(v)
                graph[v].append(u)
            else:
                max_destroyed += 1
print(max_destroyed)
