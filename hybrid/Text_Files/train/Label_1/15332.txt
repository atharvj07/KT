#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n + 1), b(n + 1);
  for (int i = 1; i <= n; ++i) {
    cin >> p[i];
  }
  for (int i = 1; i <= n; ++i) {
    cin >> b[i];
    b[i] = b[i] ^ (p[i] < p[i + 1]);
  }
  int ans = 0;
  for (int i = 1; i <= n; ++i) {
    ans += b[i] ^ (p[i] < p[i + 1]);
  }
  cout << ans << endl;
  return 0;
}
