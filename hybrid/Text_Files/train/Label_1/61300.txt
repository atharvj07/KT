import sys

N, M = map(int, input().split())
G = [set() for _ in range(N + 1)]  # set of neighbors
for _ in range(M):
    u, v = map(int, sys.stdin.readline().split())
    G[u].add(v)
    G[v].add(u)
t, w = N // 2, 1
for i in range(1, N + 1):
    if len(G[i]) % 2 == 1:
        t = 1
        w = 0
        break
    w *= len(G[i]) // 2
print(t, w)
