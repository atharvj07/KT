import heapq
d, n, m = map(int, input().split())
hq = []
cur, cost = n, 0
for i in range(m):
    x, p = map(int, input().split())
    if x > cur:
        if len(hq) and p < hq[0]:
            cost += hq[0] * (x - cur)
            cur = x
        else:
            break
    heapq.heappush(hq, p)
if cur >= d:
    print(cost)
else:
    print(-1)
