#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mp make_pair
#define pb push_back
#define pi pair<int,int>
int n,a[100001],sz[100001],ans[100001];
vector<int> adj[100001];
void dfs(int u,int p){
    sz[u]=1;
    for(auto v:adj[u]){
        if(v==p) continue;
        dfs(v,u);
        sz[u]+=sz[v];
    }
}
void dfs2(int u,int p,int color){
    ans[u]=color;
    int mx=0;
    int big=-1;
    for(auto v:adj[u]){
        if(v==p) continue;
        if(sz[v]>mx){
            mx=sz[v];
            big=v;
        }
    }
    int col=color;
    for(auto v:adj[u]){
        if(v==p||v==big) continue;
        dfs2(v,u,++col);
    }
    if(big!=-1)
        dfs2(big,u,col);
}
int main(){
    cin>>n;
    for(int i=2;i<=n;i++){
        cin>>a[i];
        adj[i].pb(a[i]);
        adj[a[i]].pb(i);
    }
    dfs(1,1);
    dfs2(1,1,1);
    for(int i=1;i<=n;i++){
        cout<<ans[i]<<" ";
    }
}
