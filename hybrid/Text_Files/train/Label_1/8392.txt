#include <iostream>
using namespace std;

int n, a[200000][3], ans;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i][0] >> a[i][1] >> a[i][2];
  for (int i = 0; i < n; i++)
    for (int j = i + 1; j < n; j++) {
      int w = a[i][0] * a[j][2] + a[j][0] * a[i][2],
          h = a[j][1] * a[i][2] + a[i][1] * a[j][2];
      if ((w - a[j][0]) % a[i][0] == 0 && (h - a[j][1]) % a[i][1] == 0) ans++;
    }
  cout << ans << endl;
}
