import sys
n=int(input())
s=input()
f=[0]*4
for c in s:
    if c=='?':continue
    f[ord(c)-ord('A')]+=1
fin=sum(f)//4
for i in range(4):
    if fin>f[i]:
        print('===')
        sys.exit()
t=''
for x in s:
    if x=='?':
        for i in range(4):
            if f[i]<fin:
                f[i]+=1
                t+=chr(i+ord('A'))
                break
    else:
        t+=x
print(t)
