#include <bits/stdc++.h>
#include <vector>
using namespace std;
const int MAXN = 305;
int n, p[MAXN], ans[MAXN], b[MAXN], A[MAXN][MAXN];
bool v[MAXN];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> p[i];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      cin >> A[i][j];
    }
  }
  int now = 1, t;
  for (int i = 1; i <= n; i++) {
    if (!v[i]) {
      t = i, v[i] = true;
      while (p[now] != i) {
        v[p[now]] = true;
        b[p[now]] = b[now];
        t = now, now = p[now];
      }
      while (now != t) {
        ans[t] = p[now];
        t = now, now = p[now];
      }
      ans[t] = i;
      now++;
    }
  }
  for (int i = 1; i <= n; i++) cout << ans[i] << " ";
  cout << endl;
  return 0;
}
