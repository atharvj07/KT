#include <iostream>
#include <vector>

using namespace std;

int main() {
  long long n, q;
  cin >> n >> q;
  vector<long long> arr(n);
  for (long long i = 0; i < n; i++) {
    arr[i] = i + 1;
  }
  for (long long i = 0; i < q; i++) {
    long long x;
    cin >> x;
    x = (x + 1) / 2;
    while (arr[x - 1] == 0) x++;
    cout << arr[x - 1] << endl;
  }
  return 0;
}
