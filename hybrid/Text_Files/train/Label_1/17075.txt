#include <iostream>
#include <vector>
using namespace std;

const int mod = 1e9 + 7;
vector<vector<int>> fact(17);
vector<int> inv(17);

int power(int a, int b) {
    int res = 1;
    while (b > 0) {
        if (b % 2 == 1) {
            res = 1ll * res * a % mod;
        }
        a = 1ll * a * a % mod;
        b /= 2;
    }
    return res;
}

void precomp() {
    for (int i = 0; i <= 16; i++) {
        int f = 1;
        for (int j = 2; j <= i; j++) {
            f = 1ll * f * j % mod;
        }
        fact[i].push_back(f);
    }
    inv[1] = 1;
    for (int i = 2; i <= 16; i++) {
        inv[i] = 1ll * (mod - mod / i) * inv[mod % i] % mod;
    }
}

int comb(int n, int k) {
    if (k < 0 || k > n) {
        return 0;
    }
    return 1ll * fact[n][k] * inv[k] % mod * inv[n - k] % mod;
}

int main() {
    precomp();
    int n;
    cin >> n;
    if (n % 2 == 1) {
        cout << 0 << endl;
        return 0;
    }
    int ans = 1ll * comb(n / 2, n / 2) % mod * comb(n - 1, n / 2) % mod * power(2, n / 2 - 1) % mod;
    cout << ans << endl;
    return 0;
}
