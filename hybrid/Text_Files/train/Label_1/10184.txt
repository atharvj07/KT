#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int MAXN = 1e6 + 5;

vector<ll> F(MAXN);

void sieve() {
    F[1] = 1;
    for (ll i = 2; i < MAXN; i++) {
        if (!F[i]) {
            for (ll j = i; j < MAXN; j += i) {
                if (!F[j]) F[j] = i;
            }
        }
    }
}

int main() {
    sieve();
    int t; cin >> t;
    while (t--) {
        ll n, k; cin >> n >> k;
        while (k--) {
            n += F[n];
        }
        cout << n << '\n';
    }
    return 0;
}
