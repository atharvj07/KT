import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int x = sc.nextInt();
            int y = sc.nextInt();
            int[][] a = new int[n][m];
            for (int i = 0; i < n; i++) {
                String s = sc.next();
                for (int j = 0; j < m; j++) {
                    if (s.charAt(j) == '*')
                        a[i][j] = 1;
                }
            }
            int ans = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (a[i][j] == 0) {
                        if (j + 1 < m && a[i][j + 1] == 0) {
                            ans += y;
                            j++;
                        } else {
                            ans += x;
                        }
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
