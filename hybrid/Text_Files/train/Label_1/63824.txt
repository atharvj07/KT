n, m, k = map(int, input().split())
droids = [list(map(int, input().split())) for _ in range(n)]

def max_droids_destroyed(n, m, k, droids):
    left, right, max_length = 0, 0, 0
    shots_needed = [0] * m
    current_shots = [0] * m
    while right < n:
        for i in range(m):
            current_shots[i] += droids[right][i]
        if sum(current_shots) <= k:
            if right - left + 1 > max_length:
                max_length = right - left + 1
                shots_needed = current_shots.copy()
            right += 1
        else:
            for i in range(m):
                current_shots[i] -= droids[left][i]
            left += 1
    return shots_needed

print(*max_droids_destroyed(n, m, k, droids))
