n, m = map(int, input().split())
a = []
for i in range(n):
    a.append(list(input()))
for i in range(n):
    for j in range(m):
        c0 = c1 = True
        for k in range(n):
            c0 &= a[k][j] == '.'
        for k in range(m):
            c1 &= a[i][k] == '.'
        if c0 or c1:
            print('YES')
            print(i+1, j+1)
            exit(0)
print('NO')
