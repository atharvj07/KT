#include <iostream>
#include <algorithm>
#include <vector>
#include <map>

using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  int n = s.length();
  map<char, int> freq;
  for (char c : t) freq[c]++;
  int yay = 0, whoops = 0;
  for (int i = 0; i < n; i++) {
    if (freq[s[i]] > 0) {
      if (s[i] == t[i]) yay++;
      else whoops++;
      freq[s[i]]--;
    }
  }
  cout << yay << " " << whoops << endl;
  return 0;
}
