#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define rep(i,j,k) for(int i=j;i<k;++i)
#define erp(i,j,k) for(int i=j;i>k;--i)
#define pb push_back
#define mp make_pair
#define all(x) (x).begin(),(x).end()
#define mod 1000000007
#define N 1000050
 
 
int32_t main()
{
	int n,a,b;
	cin>>n>>a>>b;
	if(a==b)
	{
		cout<<-1;
		return 0;
	}
	vector<int>ans;
	if(a>b){
		int temp=a;
		a=b;
		b=temp;
	}
	for(int i=n;i>b;--i) ans.pb(i);
	int cnt=a;
	for(int i=1;i<=n;++i)
	{
		if(i!=a && i!=b)
		{
			if(cnt>a)
			{
				ans.pb(i);
				cnt--;
			}
		}
	}
	ans.pb(a);
	ans.pb(b);
	cnt=n;
	for(int i=1;i<=n;++i)
	{
		if(i!=a && i!=b)
		{
			if(cnt>a)
			{
				ans.pb(i);
				cnt--;
			}
		}
	}
	
	for(int i:ans) cout<<i<<" ";
	return 0;
}
