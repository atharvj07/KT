#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pi;
#define endl "\n"
#define fastio ios::sync_with_stdio(0); cin.tie(0);

const int mxN = 2e5 + 5;
const int mod = 998244353;
ll n, m, k, a[mxN], dp[2105];
pi offers[mxN];

int main() {
	fastio;
	cin >> n >> m >> k;
	for (ll i = 1; i <= n; ++i) cin >> a[i];
	sort(a + 1, a + n + 1);
	for (ll i = 1; i <= m; ++i) cin >> offers[i].first >> offers[i].second;
	for (ll i = 0; i <= k; ++i) dp[i] = 1e18;
	dp[0] = 0;
	for (ll i = 1; i <= k; ++i) {
		for (ll j = 0; j <= n; ++j) {
			for (ll l = 1; l <= m; ++l) {
				if (offers[l].first <= i)
					dp[i] = min(dp[i], dp[i - offers[l].first] + a[max(1ll, j - offers[l].second + 1)]);
				else
					break;
			}
			dp[i] = min(dp[i], dp[i - 1] + a[j + 1]);
		}
	}
	cout << dp[k] << '\n';
	return 0;
}
