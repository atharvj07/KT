#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[100005], sum = 0, l = 0, r = 0;
    cin >> n;
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    for (int i = 0; i < n; ++i) { sum += a[i]; }
    for (int i = 0; i < n; ++i) {
        sum -= a[i];
        l += a[i];
        if (l == sum) { cout << "YES" << endl; }
    }
    cout << "NO" << endl;
    return 0;
}
