import numpy as np
def solve(a):
    a = np.array(a)
    for i in range(len(a)):
        if not np.all(np.bincount(a[i]) == 1): return "NO"
    u = np.unique(a)
    if len(u) != 8: return "NO"
    return "YES\n" + "\n".join(map(str, a))
print(solve([list(map(int, input().split())) for _ in range(8)]))
