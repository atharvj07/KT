#include <iostream>
#include <cmath>
using namespace std;
int main() {
  long long n, m, x, y, vx, vy;
  cin >> n >> m >> x >> y >> vx >> vy;
  if ((vx > 0 && vy > 0 && x == n) || (vx < 0 && vy > 0 && x == 0) ||
      (vx < 0 && vy < 0 && y == 0) || (vx > 0 && vy < 0 && y == m))
    cout << x << ' ' << y << '\n';
  else {
    double angle = atan2(vy, vx);
    if (angle >= 0) {
      while (true) {
        if (x == 0 && y == 0) {
          cout << "0 0\n";
          return 0;
        } else if (x == n && y == m) {
          cout << n << ' ' << m << '\n';
          return 0;
        }
        if (x == n)
          angle = M_PI - angle;
        else if (y == m)
          angle = -angle;
        else if (x == 0)
          angle = M_PI - angle;
        else if (y == 0)
          angle = -angle;
        if (angle >= -M_PI / 8 && angle <= M_PI / 8)
          y += vy;
        else if (angle >= -3 * M_PI / 8 && angle <= 3 * M_PI / 8)
          x += vx;
        else
          y += vy, x += vx;
      }
    } else {
      angle += 2 * M_PI;
      while (true) {
        if (x == 0 && y == 0) {
          cout << "0 0\n";
          return 0;
        } else if (x == n && y == m) {
          cout << n << ' ' << m << '\n';
          return 0;
        }
        if (x == n)
          angle = M_PI - angle;
        else if (y == m)
          angle = -angle;
        else if (x == 0)
          angle = M_PI - angle;
        else if (y == 0)
          angle = -angle;
        if (angle >= -M_PI / 8 && angle <= M_PI / 8)
          y += vy;
        else if (angle >= -3 * M_PI / 8 && angle <= 3 * M_PI / 8)
          x += vx;
        else
          y += vy, x += vx;
      }
    }
  }
  cout << "-1\n";
  return 0;
}
