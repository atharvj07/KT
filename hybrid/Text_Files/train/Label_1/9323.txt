#include <iostream>
#include <iomanip>

using namespace std;

const int MAXN = 100;
char board[MAXN][MAXN];

int n, m;

bool valid(int x, int y) {
    if (x < 0 || x >= n || y < 0 || y >= m) return false;
    return board[x][y] == '.';
}

void dfs(int x, int y, char c) {
    if (!valid(x, y)) return;
    board[x][y] = c;
    dfs(x + 1, y, c == 'W' ? 'B' : 'W');
    dfs(x, y + 1, c == 'W' ? 'B' : 'W');
}

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> board[i];
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (board[i][j] == '.') dfs(i, j, 'W');
        }
    }
    for (int i = 0; i < n; i++) {
        cout << board[i] << endl;
    }
    return 0;
}
