#include <bits/stdc++.h>
int main() {
  int n;
  std::cin >> n;
  std::string str;
  std::cin >> str;
  int l = 0, r = 0, u = 0, d = 0;
  for (char c : str) {
    if (c == 'L')
      l--;
    else if (c == 'R')
      r++;
    else if (c == 'U')
      u++;
    else if (c == 'D')
      d--;
  }
  std::cout << std::min({std::abs(l) + std::abs(r), std::abs(u) + std::abs(d)}) << std::endl;
  return 0;
}
