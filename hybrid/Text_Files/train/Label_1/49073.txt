import java.util.*;
public class Main {
    static List<List<Integer>> g1, g2;
    static int[] color;
    static boolean ok = true;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        color = new int[n + 1];
        g1 = new ArrayList<>();
        g2 = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            g1.add(new ArrayList<>());
            g2.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            g1.get(u).add(v);
            g2.get(v).add(u);
        }
        for (int i = 1; i <= n; i++) {
            if (color[i] == 0) {
                ok &= dfs(i, g1, 1);
                ok &= dfs(i, g2, 2);
            }
        }
        if (!ok) {
            System.out.println(-1);
            return;
        }
        System.out.println(2);
    }
    static boolean dfs(int u, List<List<Integer>> g, int c) {
        color[u] = c;
        for (int v : g.get(u)) {
            if (color[v] == 0) {
                if (!dfs(v, g, c)) {
                    return false;
                }
            } else if (color[v] == c) {
                return false;
            }
        }
        return true;
    }
}
