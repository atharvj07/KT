#include<bits/stdc++.h>
using namespace std;
int main(){
    int a,b,c,d;
    cin>>a>>b>>c>>d;
    if((a==c && abs(b-d)==1) || (b==d && abs(a-c)==1))
        cout<<a<<" "<<1-b<<" "<<c<<" "<<1-d<<endl;
    else if((a==c && abs(b-d)==3) || (b==d && abs(a-c)==3))
        cout<<a<<" "<<3-b<<" "<<c<<" "<<3-d<<endl;
    else
        cout<<-1<<endl;
    return 0;
}
