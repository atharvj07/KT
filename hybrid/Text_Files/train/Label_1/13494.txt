#include <iostream>
using namespace std;
int main() {
  int n, q;
  cin >> n >> q;
  string s;
  cin >> s;
  int l[n], r[n], dp[n];
  // initilization for longest valid prefix
  l[0] = 0;
  r[0] = (s[0] != 'a' ? 0 : 1);
  dp[0] = 1;
  char prv, prvprv;
  prv = s[0];
  prvprv = '#';
  // build r and l arrays
  for (int i = 1; i < n; i++) {
    l[i] = l[i - 1];
    r[i] = r[i - 1];
    prvprv = prv;
    prv = s[i];
    if (prv == 'a') {
      if (prvprv == '#')
        l[i]++;
      if (r[i - 1] > 0) l[i]++;
    } else if (prv == 'b') {
      if (prvprv == 'a') l[i]++;
      if (r[i - 1] > 0) r[i]++;
    } else if (prv == 'c') {
      if (prvprv == 'b') r[i]++;
      if (l[i - 1] > 0) dp[i] = 1;
    }
    if (l[i] > 0 && r[i] > 0) dp[i] = 1;
  }
  // dp
  while (q--) {
    int p;
    char c;
    cin >> p >> c;
    p--;
    prv = s[p];
    prvprv = (p > 0 ? s[p - 1] : '#');
    if (c == 'a') {
      if (prvprv == '#')
        l[p]++;
      if (r[p - 1] > 0) l[p]++;
    } else if (c == 'b') {
      if (prvprv == 'a') l[p]++;
      if (r[p - 1] > 0) r[p]++;
    } else if (c == 'c') {
      if (prvprv == 'b') r[p]++;
      if (l[p - 1] > 0) dp[p] = 1;
    }
    if (l[p] > 0 && r[p] > 0) dp[p] = 1;
    if (dp[p])
      cout << 0 << endl;
    else {
      // find right a
      int x = p;
      while (x + 1 < n && s[x + 1] != 'a') x++;
      // find left c
      int y = p;
      while (y - 1 >= 0 && s[y - 1] != 'c') y--;
      if (x == n || y == -1)
        cout << 1 << endl;
      else
        cout << 2 << endl;
    }
    s[p] = c;
  }
  return 0;
}
