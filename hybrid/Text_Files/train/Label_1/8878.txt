#include<bits/stdc++.h>
using namespace std;
int n;
int check(int x) {
    if(x&1==1) return 2;
    else if(x%4==0) return 1;
    else return 2;
}
int main() {
    cin >> n;
    int ans = 2;
    while(n--) {
        int x;
        cin >> x;
        ans = check(ans+x);
        cout << ans << endl;
    }
    return 0;
}
