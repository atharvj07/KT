from collections import deque

def add_edges(n, edges):
    graph = [[] for _ in range(n + 1)]
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    # BFS to find the distance of each vertex from vertex 1
    distance = [-1] * (n + 1)
    distance[1] = 0
    queue = deque([1])
    while queue:
        vertex = queue.popleft()
        for neighbor in graph[vertex]:
            if distance[neighbor] == -1:
                distance[neighbor] = distance[vertex] + 1
                queue.append(neighbor)
    
    # Count vertices that are at distance 2,
    # as they are directly connected to vertices at distance 1
    count = 0
    visited = [False] * (n + 1)
    for vertex in range(2, n + 1):
        if distance[vertex] == 2:
            for neighbor in graph[vertex]:
                if distance[neighbor] == 1 and not visited[neighbor]:
                    visited[neighbor] = True
                    count += 1
                    break
    
    # The answer is the number of vertices at distance 3,
    # as each such vertex requires a new edge to be connected to vertex 1
    # minus the count of vertices at distance 2 that are not connected to vertices at distance 1
    return max(0, len([v for v in range(2, n + 1) if distance[v] == 3]) - count)

n = int(input().strip())
edges = [list(map(int, input().strip().split())) for _ in range(n - 1)]
print(add_edges(n, edges))
