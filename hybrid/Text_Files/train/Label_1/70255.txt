def count_even_mismatches(a, b):
    len_a, len_b = len(a), len(b)
    mismatches = sum(a[i] != b[i] for i in range(len_b))
    even_mismatches = 0
    if mismatches % 2 == 0:
        even_mismatches += 1
    for i in range(1, len_a - len_b + 1):
        if a[i-1] != b[0]:
            mismatches -= 1
        if a[i+len_b-1] != b[-1]:
            mismatches += 1
        if mismatches % 2 == 0:
            even_mismatches += 1
    return even_mismatches

a = input().strip()
b = input().strip()
print(count_even_mismatches(a, b))
