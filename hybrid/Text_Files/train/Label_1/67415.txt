class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

n, m, k, q = map(int, input().split())
time = [[-1 for _ in range(m+1)] for _ in range(n+1)]
broken = set()

for i in range(q):
    x, y, t = map(int, input().split())
    time[x][y] = t
    broken.add(Point(x, y))

ans = -1
for i in range(1, n-k+2):
    for j in range(1, m-k+2):
        flag = True
        for x in range(i, i+k):
            for y in range(j, j+k):
                if time[x][y] == -1 or Point(x, y) not in broken:
                    flag = False
                    break
            if not flag:
                break
        if flag:
            cur = time[i][j]
            for x in range(i, i+k):
                for y in range(j, j+k):
                    cur = min(cur, time[x][y])
            ans = max(ans, cur)

print(ans)
