#include <vector>
#include <iostream>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<bool> light(n + 1, 1);
    vector<int> ans(n + 1, n + 1);
    while (m--) {
        int x;
        cin >> x;
        for (int i = x; i <= n; i++) {
            if (light[i]) {
                light[i] = 0;
                ans[i] = x;
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        cout << ans[i] << " ";
    }
    return 0;
}
