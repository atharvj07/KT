#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll M = 1e15;

ll F[105][105][105]; // dp[L][W][U] = number of ways to win with L losers, W winners, and the highest possible level of loser is U

int main() {
    int t; cin >> t;
    while (t--) {
        int n; ll ans = 0; cin >> n;
        vector<int> a(n);
        for (int& x : a) cin >> x;
        ll mx = *max_element(a.begin(), a.end());
        F[0][1][100] = 1;
        for (int L = 0; L <= n; L++) {
            for (int W = 0; W + L <= n; W++) {
                for (int U = 1; U <= mx; U++) {
                    (F[L + 1][W][U] += F[L][W][U]) %= M;
                    if (U + 1 <= mx) (F[L][W + 1][U + 1] += F[L][W][U]) %= M;
                    (F[L][W][U + 1] += F[L][W][U] * (mx - U)) %= M;
                }
                ans = (ans + F[L][W][mx]) % M;
            }
        }
        cout << ans << '\n';
    }
}
