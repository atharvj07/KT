#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) cin >> p[i];
  int ans = 0;
  string s;
  for (int i = 0; i < n; i++) {
    int c = 0;
    cin >> s;
    for (int j = 0; j < s.size(); j++) {
      if (s[j] == 'a' || s[j] == 'e' || s[j] == 'i' || s[j] == 'o' ||
          s[j] == 'u' || s[j] == 'y') {
        c++;
        if (j + 1 < s.size() && (s[j + 1] == 'a' || s[j + 1] == 'e' ||
                                 s[j + 1] == 'i' || s[j + 1] == 'o' ||
                                 s[j + 1] == 'u' || s[j + 1] == 'y'))
          c--;
      }
    }
    if (c) ans += c;
  }
  cout << (ans == p[0] ? "YES" : "NO") << endl;
}
