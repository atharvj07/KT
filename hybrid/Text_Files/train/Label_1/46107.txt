import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.LinkedList;

class Main {
    private static BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    private static int n;
    private static char[][] board;
    private static LinkedList<Cell> cells;
    private static boolean[][] visited;
    private static int[] dx = {0, 1, -1, 0}, dy = {1, 0, 0, -1};

    public static void main(String[] args) throws IOException {
        n = Integer.parseInt(in.readLine());
        board = new char[n + 2][n + 2];
        cells = new LinkedList<>();
        for (int i = 1; i <= n; i++) {
            String row = in.readLine();
            for (int j = 1; j <= n; j++) {
                board[i][j] = row.charAt(j - 1);
                if (board[i][j] == '.') cells.add(new Cell(i, j));
            }
        }

        while (!cells.isEmpty()) {
            Cell start = cells.poll();
            visited = new boolean[n + 2][n + 2];
            if (cover(start)) break;
        }

        System.out.println(cells.isEmpty() ? "YES" : "NO");
    }

    private static boolean cover(Cell start) {
        visited[start.i][start.j] = true;
        for (int k = 0; k < 4; k++) {
            int x1 = start.i + dx[k], y1 = start.j + dy[k];
            int x2 = start.i + 2 * dx[k], y2 = start.j + 2 * dy[k];
            if (x1 >= 1 && x1 <= n && y1 >= 1 && y1 <= n && !visited[x1][y1] && board[x1][y1] == '.' &&
                x2 >= 1 && x2 <= n && y2 >= 1 && y2 <= n && !visited[x2][y2] && board[x2][y2] == '.' &&
                cover(new Cell(x1, y1)) && cover(new Cell(x2, y2))) return true;
        }
        return false;
    }

    private static class Cell {
        int i, j;

        Cell(int i, int j) {
            this.i = i;
            this.j = j;
        }
    }
}
