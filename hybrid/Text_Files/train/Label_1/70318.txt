from bisect import bisect_left

for _ in range(int(input())):
    n = int(input())
    lefts = []
    rights = []
    for _ in range(n):
        l, r = map(int, input().split())
        lefts.append(l)
        rights.append(r)
    lefts.sort()
    rights.sort()
    res = rights[0] - lefts[-1]
    for i in range(n):
        idxl = bisect_left(lefts, rights[i])
        idxr = bisect_left(rights, lefts[i])
        res = min(res, rights[idxl - 1] - lefts[i])
        res = min(res, rights[i] - lefts[idxr])
    print(res)
