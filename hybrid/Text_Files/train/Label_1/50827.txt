import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n;
        while ((n = sc.nextInt()) != 0) {
            int count = 0;
            boolean[] state = new boolean[2];
            for (int i = 0; i < n; i++) {
                String s = sc.next();
                if (s.charAt(0) == 'l') {
                    state[0] = !state[0];
                } else {
                    state[1] = !state[1];
                }
                if (state[0] && state[1]) {
                    count++;
                    state[0] = false;
                    state[1] = false;
                }
            }
            System.out.println(count);
        }
    }
}
