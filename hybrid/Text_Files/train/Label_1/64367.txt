import sys
import bisect

in_str = sys.stdin.readlines()
m, k, n, s = map(int, in_str[0].split())
lia = list(map(int, in_str[1].split()))
sch = list(map(int, in_str[2].split()))
sch_cnt = {typ: cnt for typ, cnt in zip(sch, [1] * s)}

for i in range(len(lia)):
    if lia[i] in sch:
        sch_cnt[lia[i]] -= 1
        if sch_cnt[lia[i]] < 0:
            print(-1)
            exit(0)

lia_cnt = {typ: 0 for typ, cnt in sch_cnt.items()}

for i in range(len(lia)):
    if lia[i] in sch:
        lia_cnt[lia[i]] += 1
    else:
        break

a_rem = []
for typ, cnt in lia_cnt.items():
    dif = cnt - sch_cnt[typ]
    if dif > 0:
        a_rem.extend([i for i in range(lia.index(typ) + 1, len(lia)) if lia[i] == typ][:dif])

print(len(a_rem), end="\n")
print(*a_rem)
