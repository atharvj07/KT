#include <bits/stdc++.h>

using namespace std;

const int maxn = 5e5 + 7;
const int maxm = 5e5 + 7;
typedef pair<int, int> pii;
map < int, int > mp;
vector <int> ans[maxm];
bool vis[maxm], mark[maxm];

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        mp[x] = i;
    }
    for (int i = 2; i <= n; i++) {
        if (!vis[i]) {
            mark[i] = true;
            for (int j = i + i; j <= n; j += i) {
                vis[j] = true;
            }
            for (int j = i; j <= n; j += i) {
                if (mp[j] < mp[j + i]) {
                    ans[i].push_back(mp[j]);
                    ans[i].push_back(mp[j + i]);
                }
            }
        }
    }
    int anss = 0;
    for (int i = 2; i <= n; i++) {
        if (mark[i]) {
            anss += ans[i].size();
        }
    }
    cout << anss << endl;
    for (int i = 2; i <= n; i++) {
        if (mark[i]) {
            for (int j = 0; j < ans[i].size(); j += 2) {
                cout << ans[i][j] << " " << ans[i][j + 1] << endl;
            }
        }
    }
    return 0;
}
