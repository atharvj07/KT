from collections import defaultdict
n=int(input())
a=list(map(int,input().split()))
g=defaultdict(list)
for _ in range(n-1):
    u,v=map(int,input().split())
    g[u].append(v)
    g[v].append(u)
ans=dict()
def dfs(v,par):
    ans[v]=2*a[v-1]-1
    for u in g[v]:
        if u!=par:
            dfs(u,v)
            ans[v]=max(ans[u]+2*a[v-1]-1,ans[v])
dfs(1,0)
for i in range(1,n+1):
    print(ans[i])
