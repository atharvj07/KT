r, g, b = map(int, input().split())

# Make as many mixed bouquets as possible
mixed = min(r, g, b)

# Update the numbers after making mixed bouquets
r -= mixed
g -= mixed
b -= mixed

# Make single color bouquets from the remaining flowers
single_color = (r // 3) + (g // 3) + (b // 3)

# Check if making two more mixed bouquets by sacrificing some single color bouquets is beneficial
rest = r % 3 + g % 3 + b % 3
if rest == 4 and mixed > 0:
    single_color += 1
elif rest == 5:
    single_color += 2

print(mixed + single_color)
