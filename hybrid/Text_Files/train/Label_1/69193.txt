import sys
def nm(l):return [int(x) for x in l.split()]
n,m=nm(sys.stdin.readline())
a=[[x for x in nm(sys.stdin.readline())] for i in range(n)]
b=[[x for x in nm(sys.stdin.readline())] for i in range(n)]
for i in range(n):
 if sorted(a[i])!=sorted(b[i]):print('NO');exit()
for i in range(m):
 if sorted(a[j][i] for j in range(n))!=sorted(b[j][i] for j in range(n)):print('NO');exit()
print('YES')
