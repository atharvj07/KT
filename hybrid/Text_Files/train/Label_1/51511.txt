import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Stack;

public class Main {
  private static int N, K, maxDepth1, maxDepth2;
  private static int maxDiameter;
  private static ArrayList<Integer>[] adj;
  private static boolean[] visited;
  private static int[] parent;
  private static Queue<Integer> leaves;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] s = br.readLine().split(" ");
    N = Integer.parseInt(s[0]);
    K = Integer.parseInt(s[1]);
    adj = new ArrayList[N + 1];
    visited = new boolean[N + 1];
    Arrays.fill(visited, false);
    parent = new int[N + 1];
    Arrays.fill(parent, 0);
    for (int i = 0; i <= N; i++) {
      adj[i] = new ArrayList<>();
    }
    for (int i = 0; i < N - 1; i++) {
      s = br.readLine().split(" ");
      int u = Integer.parseInt(s[0]);
      int v = Integer.parseInt(s[1]);
      adj[u].add(v);
      adj[v].add(u);
    }
    leaves = new LinkedList<>();
    int farthestNode = findFarthestNode(1, 0);
    int secondFarthestNode = findFarthestNode(farthestNode, 0);
    maxDiameter = getDiameter(farthestNode, secondFarthestNode);

    if (maxDiameter <= K) {
      System.out.println(0);
      return;
    }

    // Find all leaves
    visited = new boolean[N + 1];
    Arrays.fill(visited, false);
    leaves = new LinkedList<>();
    getLeaves(1, 0);

    int removedNodes = 0;
    while (!leaves.isEmpty()) {
      int leaf = leaves.poll();
      int leafParent = parent[leaf];
      if (maxDiameter <= K) {
        break;
      }

      maxDiameter = Math.max(getDiameter(leafParent, secondFarthestNode), getDiameter(leaf, farthestNode));
      removedNodes++;
      visited[leaf] = true;
      visited[leafParent] = true;
      for (int neighbor : adj[leaf]) {
        if (!visited[neighbor]) {
          getLeaves(neighbor, leaf);
        }
      }
    }

    System.out.println(removedNodes);
  }

  private static void getLeaves(int node, int par) {
    parent[node] = par;
    visited[node] = true;
    if (adj[node].size() == 1) {
      leaves.add(node);
    }
    for (int neighbor : adj[node]) {
      if (!visited[neighbor]) {
        getLeaves(neighbor, node);
      }
    }
  }

  private static int findFarthestNode(int node, int par) {
    maxDepth1 = 0;
    visited = new boolean[N + 1];
    Arrays.fill(visited, false);
    dfs(node, par, 0);
    return maxDepth1;
  }

  private static void dfs(int node, int par, int depth) {
    visited[node] = true;
    if (depth > maxDepth1) {
      maxDepth1 = depth;
    }
    for (int neighbor : adj[node]) {
      if (!visited[neighbor]) {
        dfs(neighbor, node, depth + 1);
      }
    }
  }

  private static int getDiameter(int node1, int node2) {
    Stack<Integer> path = new Stack<>();
    visited = new boolean[N + 1];
    Arrays.fill(visited, false);
    findPath(node1, node2, path);
    return path.size() - 1;
  }

  private static void findPath(int node, int target, Stack<Integer> path) {
    visited[node] = true;
    path.push(node);
    if (node == target) {
      return;
    }
    for (int neighbor : adj[node]) {
      if (!visited[neighbor]) {
        findPath(neighbor, target, path);
      }
    }
  }
}
