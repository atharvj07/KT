import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int s = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (i == s) continue;
            int cnt = 0;
            for (int j = i; j != s; j = a[j]) {
                cnt++;
            }
            if (cnt != a[i]) ans++;
        }
        System.out.println(ans);
    }
}
