#include <bits/stdc++.h>
using namespace std;
long long n, m, ans;
int main() {
  cin >> n >> m;
  while (n != m) {
    ans++;
    if (n % 2 == 0 && n * 2 <= m)
      n *= 2;
    else if (n % 3 == 0 && n * 3 <= m)
      n *= 3;
    else
      return cout << -1, 0;
  }
  cout << ans << endl;
  return 0;
}
