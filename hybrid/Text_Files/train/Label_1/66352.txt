t = int(input())
for _ in range(t):
    n, s, k = map(int, input().split())
    closed = set(map(int, input().split()))
    for i in range(n):
        if s - i > 0 and (s - i) not in closed:
            print(i)
            break
        if s + i <= n and (s + i) not in closed:
            print(i)
            break
