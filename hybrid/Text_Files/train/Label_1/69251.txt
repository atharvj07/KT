import sys
input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for i in range(n+1)]
for i in range(m):
    x, y = map(int, input().split())
    adj[x].append(y)
    adj[y].append(x)
q = int(input())
for i in range(q):
    op = list(map(int, input().split()))
    if op[0] == 1:
        x, y = op[1], op[2]
        adj[x].append(y)
        adj[y].append(x)
    elif op[0] == 2:
        x, y = op[1], op[2]
        adj[x].remove(y)
        adj[y].remove(x)
    else:
        vis = [False] * (n+1)
        for i in range(1, n+1):
            if len(adj[i]) == 0: continue
            flag = False
            for j in adj[i]:
                if i < j: flag = True
            if not flag: vis[i] = True
        while True:
            flag = False
            for i in range(1, n+1):
                if vis[i]:
                    for j in adj[i]:
                        if not vis[j]:
                            vis[j] = True
                            flag = True
            if not flag: break
        ans = 0
        for i in range(1, n+1):
            if not vis[i]: ans += 1
        print(ans)
