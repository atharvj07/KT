#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  int n;
  cin >> n;
  vector<ll> a(n);
  for (auto &x : a) cin >> x;
  vector<ll> dp(n);
  ll ans = LLONG_MIN;
  if (n == 1) {
    cout << a[0] << endl;
    return 0;
  }
  dp[1] = a[0];
  for (int i = 2; i <= n; i++) {
    dp[i] = max(dp[i - 1] + a[i - 1], a[i - 1]);
    ans = max(dp[i], ans);
  }
  cout << ans << endl;
  return 0;
}
