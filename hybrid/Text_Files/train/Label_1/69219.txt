from sys import stdin

def solve(s, x):
    if 2 * x + 1 > len(s):
        return -1
    w = ['0'] * len(s)
    for i in range(x, len(s) - x):
        if s[i] == '1':
            w[i] = '1'
            w[i - x] = w[i + x] = '1'
    if '0' not in w[x:len(s) - x]:
        return -1
    return ''.join(w)

for _ in range(int(stdin.readline())):
    s = stdin.readline()
    x = int(stdin.readline())
    print(solve(s, x))
