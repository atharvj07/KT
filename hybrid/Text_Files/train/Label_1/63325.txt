def solve(n, d, b, a):
    def count_mismatches(rooms, direction):
        mismatches, total, target = 0, 0, 0
        for i in range(len(rooms)):
            total += rooms[i]
            while total >= b and target < len(rooms) // 2:
                total -= b
                target += 1
            if direction * (target - i) > d:
                mismatches += 1
        return mismatches

    left_rooms = a[:len(a)//2 + len(a) % 2]
    right_rooms = a[len(a)//2 + len(a) % 2:][::-1]

    left_mismatches = count_mismatches(left_rooms, 1)
    right_mismatches = count_mismatches(right_rooms, -1)

    return max(left_mismatches, right_mismatches)

n, d, b = map(int, input().split())
a = list(map(int, input().split()))
print(solve(n, d, b, a))
