#include<bits/stdc++.h>
using namespace std;
int main(){
    int t; cin>>t;
    while(t--){
        int n, k; cin>>n>>k;
        for(int i=0;; i++){
            int x=i, sum=0;
            for(int j=0; j<=k; j++){
                sum+=x%10;
                x/=10;
            }
            if(sum==n){
                cout<<i<<endl;
                break;
            }
            if(x==0){
                cout<<"-1"<<endl;
                break;
            }
        }
    }
    return 0;
}
