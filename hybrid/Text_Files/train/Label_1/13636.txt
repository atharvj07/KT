#include<bits/stdc++.h>
using namespace std ;
int n, a[300010], sum[300010], f[300010], id[300010], ans[300010], cnt ;
int main() {
	cin >> n ;
	for(int i = 1 ; i <= n ; i ++ ) cin >> a[i] ;
	for(int i = 1 ; i <= n ; i ++ ) sum[i] = sum[i - 1] + a[i] ;
	for(int i = 1 ; i <= n ; i ++ ) f[i] = f[i - 1], id[i] = id[i - 1] ;
	for(int i = 2 ; i < n ; i ++ ) {
		if(a[i] == a[i + 1] && f[i] < sum[i]) {
			f[i + 1] = sum[i], id[i + 1] = i ;
		} else f[i + 1] = f[i], id[i + 1] = id[i] ;
	}
	int cur = id[n] ;
	while(cur) ans[++ cnt] = cur, cur = id[cur - 1] ;
	cout << f[n] << ' ' << cnt << endl ;
	for(int i = cnt ; i >= 1 ; i -- ) cout << ans[i] << ' ' ;
}
