from typing import Dict, List

def solve(n: int, m: int, filenames: List[str], delete: List[int]) -> str:
    length = len(filenames[0])
    pattern = ['?' for _ in range(length)]
    delete_set = set(delete)
    for i in range(n):
        if i+1 in delete_set: continue
        for j in range(length):
            if filenames[i][j] != pattern[j]:
                if pattern[j] != '?' and filenames[i][j] != pattern[j]:
                    return "No"
                pattern[j] = '?'
    return "Yes\n" + ''.join(pattern)

def main():
    n, m = map(int, input().split())
    filenames = [input() for _ in range(n)]
    delete = list(map(int, input().split()))
    print(solve(n, m, filenames, delete))

main()
