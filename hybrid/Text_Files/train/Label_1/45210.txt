import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int r = in.nextInt();
        int x = in.nextInt();
        int y = in.nextInt();
        int x2 = in.nextInt();
        int y2 = in.nextInt();

        // Check if the destination point is inside the circle with radius r.
        if (Math.pow(x2 - x, 2) + Math.pow(y2 - y, 2) > Math.pow(r, 2)) {
            System.out.println(-1);
            return;
        }

        // Compute the angle of rotation.
        double angle = Math.atan2(y2 - y, x2 - x);

        // Compute the number of steps required.
        int steps = (int) Math.ceil(Math.abs(angle) / Math.acos(-1));

        System.out.println(steps);
    }
}
