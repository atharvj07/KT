#include <bits/stdc++.h>

using namespace std;
const int N = 2e5 + 10;
vector<int>adj[N];
vector<int>dp(N, -1);
int vis[N];

int DFS(int node, int par = -1) {
    if (dp[node] != -1)return dp[node];
    dp[node] = 0;
    for (int child : adj[node]) {
        if (child != par) {
            dp[node] = max(dp[node], DFS(child, node) + vis[node]);
        }
    }
    return dp[node];
}

int main() {
    int n, m;
    cin >> n >> m;
    long long sum = 0;
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        sum += x;
        vis[i] = x;
    }
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int s;
    cin >> s;
    cout << sum - DFS(s) << '\n';
}
