import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] t = new int[5];
        for (int i = 0; i < 5; i++) {
            t[i] = in.nextInt();
        }
        in.close();
        Arrays.sort(t);
        int sum = t[0] + t[1] + t[2];
        if (t[3] == t[4]) {
            sum += t[3];
        } else {
            sum += t[3] + t[4] - t[2];
        }
        System.out.println(sum);
    }
}
