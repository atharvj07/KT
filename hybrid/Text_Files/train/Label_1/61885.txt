import heapq

n = int(input())
a = list(map(int, input().split()))
a = [0] + a

k = 1
while k * n <= 1e9+10:
    h = []
    ans = 0
    for i in a:
        heapq.heappush(h, i)
        if len(h) > k:
            if h[0] > i:
                ans += 1
            heapq.heappop(h)
    print(ans, end=" ")
    k *= 2
