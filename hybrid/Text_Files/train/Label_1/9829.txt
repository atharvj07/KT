#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxN = 1e5 + 1;
ll n, q, cur, l[maxN], r[maxN];
int main() {
    cin >> n >> q;
    cur = n;
    while (q--) {
        int type;
        cin >> type;
        if (type == 1) {
            int pos;
            cin >> pos;
            l[cur] = pos;
            r[cur] = cur - pos;
            cur = pos;
        } else {
            int x, y;
            cin >> x >> y;
            cout << max(0ll, max(max(r[x] - x, y - l[y]), r[y] - x) - max(y - l[x], r[x] - y)) << "\n";
        }
    }
    return 0;
}
