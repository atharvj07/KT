import sys

n = int(input())

pos = [[int(x) for x in sys.stdin.readline().split()] for _ in range(n)]

xs, ys, zs = [list(i) for i in zip(*pos)]
xs.sort()
ys.sort()
zs.sort()

xmid = (xs[n // 2] + xs[(n // 2) - 1]) / 2
ymid = (ys[n // 2] + ys[(n // 2) - 1]) / 2
zmid = (zs[n // 2] + zs[(n // 2) - 1]) / 2

l = []
for p in pos:
    if p[0] < xmid:
        l.append(p)

for p in pos:
    if p[0] > xmid:
        l.append(p)

for p in pos:
    if p[1] < ymid:
        l.append(p)

for p in pos:
    if p[1] > ymid:
        l.append(p)

for p in pos:
    if p[2] < zmid:
        l.append(p)

for p in pos:
    if p[2] > zmid:
        l.append(p)

ans = []
for i in range(0, n, 2):
    ans.append([pos.index(l[i]) + 1, pos.index(l[i + 1]) + 1])

for p in ans:
    print(' '.join(map(str, p)))
