import sys
input = sys.stdin.readline

def dfs(u, p, flow, bee, forbidden):
    if u == bee:
        bee = -1
    if u == flow:
        flow = -1
    if flow != -1 and bee != -1 and flow > bee:
        forbidden[u] = True
    for v in adj[u]:
        if v != p:
            dfs(v, u, flow, bee, forbidden)

n, flow, bee = map(int, input().split())
adj = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

forbidden = [False] * (n+1)
dfs(1, 1, flow, bee, forbidden)

ans = 0
for i in range(1, n+1):
    for j in range(i+1, n+1):
        if not forbidden[i] and not forbidden[j]:
            ans += 1
print(ans)
