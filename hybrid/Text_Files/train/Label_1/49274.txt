import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.StringTokenizer;

public class Main {
    static final int MAX_SIZE = 100000;
    static boolean[] isPrime = new boolean[MAX_SIZE + 1];

    static {
        Arrays.fill(isPrime, true);
        isPrime[0] = isPrime[1] = false;
        for (int i = 2; i * i <= MAX_SIZE; i++) {
            if (isPrime[i]) {
                for (int j = i * i; j <= MAX_SIZE; j += i) {
                    isPrime[j] = false;
                }
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        int[] ages = new int[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            ages[i] = Integer.parseInt(st.nextToken());
        }

        Arrays.sort(ages);

        if (!isPossible(ages, n)) {
            System.out.println("Impossible");
            return;
        }

        List<Integer>[] adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (isPrime[ages[i] + ages[j]]) {
                    adj[i].add(j);
                    adj[j].add(i);
                }
            }
        }

        List<Integer> cycle = new ArrayList<>();
        boolean[] visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (visited[i]) {
                continue;
            }

            dfs(i, i, adj, visited, cycle);
        }

        if (cycle.size() < 3) {
            System.out.println("Impossible");
            return;
        }

        int m = (cycle.size() - 1) / 2;
        System.out.println(m);

        for (int i = 0; i < m; i++) {
            System.out.print(cycle.size() + " ");
            for (int j = 0; j < cycle.size(); j++) {
                System.out.print((cycle.get(j) + 1) + " ");
            }

            System.out.println();

            cycle.remove(0);
        }
    }

    static boolean isPossible(int[] ages, int n) {
        int minSum = ages[0] + ages[1];
        int maxSum = ages[n - 2] + ages[n - 1];

        if (!isPrime[minSum] || !isPrime[maxSum]) {
            return false;
        }

        int[] freq = new int[MAX_SIZE + 1];
        for (int age : ages) {
            freq[age]++;
        }

        if (freq[2] % 2 == 1) {
            return false;
        }

        int oddCount = 0;
        for (int i = 3; i <= MAX_SIZE; i += 2) {
            if (isPrime[i] && freq[i] % 2 == 1) {
                oddCount++;
            }
        }

        return oddCount <= 1;
    }

    static void dfs(int cur, int start, List<Integer>[] adj, boolean[] visited, List<Integer> cycle) {
        visited[cur] = true;
        cycle.add(cur);

        for (int next : adj[cur]) {
            if (!visited[next]) {
                dfs(next, start, adj, visited, cycle);
            } else if (next == start) {
                cycle.add(cur);
            }
        }
    }
}
