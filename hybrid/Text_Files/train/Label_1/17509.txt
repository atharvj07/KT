#include <iostream>

using namespace std;

bool is_square(string &s) {
    int n = s.length();
    for (int i = 1; i <= n / 2; i++) {
        bool ok = true;
        for (int j = 0; j < i; j++) {
            if (s[j] != s[j + i]) {
                ok = false;
                break;
            }
        }
        if (ok) return true;
    }
    return false;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        if (is_square(s)) cout << "YES\n";
        else cout << "NO\n";
    }
    return 0;
}
