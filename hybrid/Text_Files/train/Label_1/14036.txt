#include <bits/stdc++.h>

using namespace std;

int main() {
    int v;
    cin >> v;
    for (int i = 0; i < v; i++) {
        vector<int> freq(2001, 0);
        for (int j = 0; j < 250; j++) {
            int x;
            cin >> x;
            freq[x]++;
        }
        int uniform_max = *max_element(freq.begin(), freq.begin() + 2 * 1000 + 1);
        if (uniform_max > 250 / (2 * 1000 + 1))
            cout << "uniform\n";
        else
            cout << "poisson\n";
    }
    return 0;
}
