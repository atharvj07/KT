import sys
t=int(input())
for _ in range(t):
    n=int(input())
    a=[int(x) for x in input().split()]
    if a==sorted(a):
        print(0)
        continue
    elif n<6 or n%4==1:
        print(-1)
        continue
    else:
        if a[0]==n:
            ans=[1,3,1]
            a=[a[2],n,a[1],a[3],a[4]]
        else:
            ans=[1,3,1,3]
            a=[n,a[2],a[1],a[3],a[4]]
        for i in range(n-5,1,-2):
            ans.extend([i,i+2,i])
            a=a[i+2:i:-1]+a[i+2:]
        print(len(ans))
        print(*ans)
