for test in range(int(input())):
    n, m = map(int, input().split())
    a = input()
    for _ in range(m):
        b = '0' + a + '0'
        c = ''
        for i in range(1, n+1):
            if (b[i+1] == b[i-1]) or (a[i] == '1'):
                c += '1'
            else:
                c += '0'
        a = c
    print(a)
