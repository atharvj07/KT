import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] towers = new int[n];
        int maxTower = 0;
        for (int i = 0; i < n; i++) {
            towers[i] = input.nextInt();
            maxTower = Math.max(maxTower, towers[i]);
        }
        int operations = 0;
        while (maxTower > 0) {
            for (int i = 0; i < n; i++) {
                if (towers[i] > 0) {
                    towers[i]--;
                }
            }
            operations++;
            maxTower = 0;
            for (int i = 0; i < n; i++) {
                maxTower = Math.max(maxTower, towers[i]);
            }   
        }
        System.out.println(operations);
    }
}
