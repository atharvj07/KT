import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int result = -1;
        for (int i = 1; i <= n; i++) {
            if (i % 2 == 0) {
                if ((n - i) % m == 0) {
                    result = i;
                    break;
                }
            } else {
                if ((n - i) % m == 1) {
                    result = i;
                    break;
                }
            }
        }
        System.out.println(result);
    }
}
