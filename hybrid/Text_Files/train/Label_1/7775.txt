#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    ll n, m, a, d, x, ans = 0, t = -1e18, c = 0;
    cin >> n >> m >> a >> d;
    priority_queue<ll, vector<ll>, greater<ll>> q;
    for (ll i = 0; i < m; i++) { cin >> x; q.push(x); }
    for (ll i = 1; i <= n; i++) {
        x = i * a;
        if (x > t) ans++, t = x + d, c = 0;
        q.push(x);
    }
    while (!q.empty()) {
        x = q.top();
        q.pop();
        if (x <= t) {
            if (++c == 1) ans++;
        } else {
            t = x + d;
            c = 1;
        }
    }
    cout << ans << "\n";
    return 0;
}
