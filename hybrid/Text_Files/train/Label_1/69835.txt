from sys import stdin
import string
inp = stdin.readlines()[1:];
eng = sorted(string.ascii_lowercase)
for s in inp:
    t = list(s.strip())
    for i in range(len(t)):
        if i%2 == 0:  j = min(filter(lambda c: c>t[i], eng))
        else:         j = max(filter(lambda c: c<t[i], eng))
        t[i] = j
    print(''.join(t))
