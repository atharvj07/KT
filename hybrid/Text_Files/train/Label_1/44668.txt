import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] x = new int[4 * n + 1];
        int[] y = new int[4 * n + 1];
        int x0 = 0, y0 = 0;
        for (int i = 0; i < 4 * n + 1; i++) {
            x[i] = in.nextInt();
            y[i] = in.nextInt();
        }
        for (int i = 0; i < 4 * n + 1; i++) {
            if (x[i] > 0 && x[i] < n && y[i] > 0 && y[i] < n) {
                x0 = x[i];
                y0 = y[i];
            }
        }
        System.out.println(x0 + " " + y0);
    }
}
