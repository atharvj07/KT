import sys
t = int(sys.stdin.readline().rstrip())
for _ in range(t):
    n = int(sys.stdin.readline().rstrip())
    a = [int(x) for x in sys.stdin.readline().rstrip().split()]
    alice, bob = 0, 0
    cnt = 0
    alice_turn = True
    while a:
        if alice_turn:
            sa = 0
            i = 0
            while i < len(a) and sa <= bob:
                sa += a[i]
                i += 1
            alice += sa
            a = a[i:]
        else:
            sb = 0
            i = len(a) - 1
            while i >= 0 and sb <= alice:
                sb += a[i]
                i -= 1
            bob += sb
            a = a[:i+1]
        alice_turn = not alice_turn
        cnt += 1
    print(cnt, alice, bob)
