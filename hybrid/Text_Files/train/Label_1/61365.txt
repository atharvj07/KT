def min_time_to_home(n, m, sx, sy, fx, fy, locations):
    # Add starting and finishing points as portals for simplification
    locations.append((sx, sy))
    locations.append((fx, fy))
    
    # Calculate the direct distance as one of the options
    min_time = abs(sx - fx) + abs(sy - fy)
    
    # Sort locations by x and y to find the closest portal in x or y direction
    locations.sort(key=lambda x: x[0])  # Sort by x-coordinate
    xs = [abs(sx - x) + abs(fx - x) for x, _ in locations]
    min_time = min(min_time, min(xs) + abs(sy - fy))
    
    locations.sort(key=lambda x: x[1])  # Sort by y-coordinate
    ys = [abs(sy - y) + abs(fy - y) for _, y in locations]
    min_time = min(min_time, min(ys) + abs(sx - fx))
    
    return min_time

# Input reading
n, m = map(int, input().split())
sx, sy, fx, fy = map(int, input().split())
locations = [tuple(map(int, input().split())) for _ in range(m)]

# Output the minimum time required to get home
print(min_time_to_home(n, m, sx, sy, fx, fy, locations))
