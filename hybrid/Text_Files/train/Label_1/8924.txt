#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[100005], b[100005], sum = 0, Max = 0;
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i], sum += a[i], Max = max(Max, b[i]); }
    if (sum < 0 || sum > Max || sum & 1)
        cout << "NO" << endl;
    else
        cout << "YES" << endl;
    return 0;
}
