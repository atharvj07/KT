#include <bits/stdc++.h>
using namespace std;
#define MAXN 2005
struct data { long long f; long long c; long long r; };
vector<data> v;
long long dp[MAXN][MAXN], ans;

long long go(long long idx, long long rem) {
  if (rem < 0) { return LLONG_MAX; }
  if (idx == v.size()) {
    if (rem != 0) { return LLONG_MAX; }
    return 0;
  }
  if (dp[idx][rem] != -1) { return dp[idx][rem]; }
  long long &ret = dp[idx][rem] = LLONG_MAX;
  ret = min(ret, go(idx + 1, rem));
  if (rem >= v[idx].c) {
    ret = min(ret, go(idx + 1, rem - v[idx].c) + v[idx].f + v[idx].r);
  }
  return ret;
}

int main() {
  ans = LLONG_MAX;
  long long n, k;
  cin >> n >> k;
  memset(dp, -1, sizeof(dp));
  for (long long i = 0; i < n; i++) {
    long long x;
    cin >> x;
    if (x <= k) {
      v.push_back({x, 0, 0});
    } else {
      v.push_back({x - k, x - k, k});
    }
  }
  long long res = go(0, k);
  if (res == LLONG_MAX) { res = -1; }
  cout << res << endl;
  return 0;
}
