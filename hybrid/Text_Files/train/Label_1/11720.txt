#include <algorithm>
#include <vector>
#include <iostream>
using namespace std;
#define ll long long

ll n, m;
vector<ll> a;
ll get_min_moves() {
    sort(a.begin(), a.end());
    ll moves = 0;
    ll mini = a[0] % m, maxi = a[n - 1] % m, cnt = (a[n - 1] - a[0]) / m;
    if (mini == 0) {
        moves = (n / m - 1) * cnt;
        mini = m;
    }
    if (maxi == 0) {
        moves = (n / m - 1) * cnt;
        maxi = m;
    }
    if (mini == maxi) return moves;
    for (ll i = 0; i < n; i++) {
        if (a[i] % m == mini && moves <= (n / m - 1) * cnt) {
            ll delta = m - mini, extra = max<ll>(0, (n / m - 1) * cnt - moves);
            if (a[i] + delta <= a[i + extra]) {
                moves += delta;
                a[i] += delta;
                mini = m;
                if (maxi - mini > cnt) maxi -= cnt;
            }
            else break;
        }
        if (a[i] % m == maxi && moves <= (n / m - 1) * cnt) {
            if (a[i] - cnt >= a[i - 1] || a[i] - cnt >= 0) {
                moves += cnt;
                a[i] -= cnt;
                maxi -= cnt;
                if (maxi - mini > cnt) mini += cnt;
            }
            else break;
        }
    }
    mini = a[0] % m, maxi = a[n - 1] % m;
    if (maxi - mini > cnt) return -1;
    return moves;
}
int main() {
    cin >> n >> m;
    a.resize(n);
    for (ll &i : a) cin >> i;
    ll moves = get_min_moves();
    cout << moves << '\n';
    if (moves == -1) return 0;
    for (ll &i : a) cout << i << ' ';
    cout << '\n';
    return 0;
}
