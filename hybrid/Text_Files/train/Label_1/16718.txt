#include <iostream>
#include <vector>
using namespace std;

int dx[4] = {-2, -2, 2, 2};
int dy[4] = {2, -2, 2, -2};
int ax, ay, bx, by;
string a, str;

int main() {
    int t;
    cin >> t;
    getchar();
    while (t--) {
        vector<vector<char>> board(8);
        int cntgood = 0, cnt = 0;
        bool found = false;
        for (int i = 0; i < 8; i++) {
            getline(cin, a);
            for (int j = 0; j < 8; j++) {
                board[i].push_back(a[j]);
                if (a[j] == 'K') {
                    if (cnt == 0)
                        ax = i, ay = j;
                    else
                        bx = i, by = j;
                    cnt++;
                }
                if (a[j] != '#') cntgood++;
            }
        }
        if (cntgood == 0) {
            cout << "NO" << endl;
            continue;
        }

        for (int x = 0; x < 4 && !found; x++) {
            int nx = ax + dx[x], ny = ay + dy[x];
            if (nx < 0 || nx >= 8 || ny < 0 || ny >= 8 ||
                board[nx][ny] == '#')
                continue;
            for (int y = 0; y < 4; y++) {
                int px = bx + dx[y], py = by + dy[y];
                if (px < 0 || px >= 8 || py < 0 || py >= 8 ||
                    board[px][py] == '#')
                    continue;
                if (px == nx && py == ny) {
                    found = true;
                    break;
                }
            }
        }
        if (found)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
