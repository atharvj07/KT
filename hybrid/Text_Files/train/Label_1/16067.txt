#include<bits/stdc++.h>
using namespace std;
#define ll long long
struct node{
    ll x,y;
}a[200005];
ll n,ans;
int main(){
    cin>>n;
    for(ll i=1;i<=n;i++)a[i].x=0,a[i].y=0;
    for(ll i=1;i<=n;i++)cin>>a[i].x;
    sort(a+1,a+1+n,[&](node x,node y){
        return x.x<y.x;
    });
    ll one=n-1,two=-1;
    for(ll i=1;i<=n;i++){
        if(i==1)continue;
        if(a[i].x==a[i-1].x&&a[i].x==0){
            two=i;
            one--;
        }
        a[i].y=a[i].x*(a[i-1].x);
    }
    for(ll i=1;i<=n;i++){
        if(a[i].x==0&&i!=n)cout<<"1 "<<i<<" "<<i+1<<endl;
        a[i].x=a[i].y;
    }
    a[n].x=a[n].y=0;
    sort(a+1,a+1+n,[&](node x,node y){
        return x.x>y.x;
    });
    for(ll i=1;i<=n;i++){
        if(a[i].x==0)continue;
        if(two!=-1){
            cout<<"1 "<<two<<" "<<i<<endl;
            two=-1;
        }
        else{
            cout<<"1 "<<i<<" "<<i+1<<endl;
            one--;
        }
        a[i].x=a[i].y=0;
    }
    if(one)cout<<"2 "<<n<<endl;
}
