#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define pb push_back
#define pii pair<int, int>
#define mp make_pair
#define f first
#define s second
#define mii map<int, int>

const int N = 1e5+10;

int n, p;
string s;
vi vec[N];
bool vis[N];

int main(){
	cin >> n >> p;
	cin >> s;
	for(int i=1; i<=n; i++){
		vec[s[i-1]-'a'].pb(i);
	}
	int res = n;
	for(int i=0; i<26; i++){
		for(int j=0; j<vec[i].size(); j++){
			int x = vec[i][j];
			if(vis[x]) continue;
			if(x > p){
				res = min(n - x + p - 1, res);
			}else{
				res = min(p - x + n - 1, res);
			}
			int y = vec[i][(j+1)%vec[i].size()];
			if(x > p){
				res = min(abs(x-y) + n - p, res);
			}else{
				res = min(abs(x-y) + p - 1, res);
			}
			vis[x] = 1;
			vis[y] = 1;
		}
	}
	cout << res << endl;
	return 0;
}
