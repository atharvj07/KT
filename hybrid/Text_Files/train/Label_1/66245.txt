n, v = map(int, input().split())
a = []
b = []
for i in range(n):
    aa, bb = map(int, input().split())
    a.append(aa)
    b.append(bb)
a.sort()
ans, l, r = 0, 0, 0
for i in a:
    if r + b[l] <= i:
        ans += min(v, r - l + b[l])
        l += 1
    r = max(r, i + 1)
ans += min(v, r - l + b[l])
print(ans)
