from math import factorial

def nth_permutation(n, k):
    sequence = list(range(1, n + 1))
    permutation = []
    k -= 1
    while n > 0:
        n -= 1
        fact = factorial(n)
        index = k // fact
        k %= fact
        permutation.append(sequence.pop(index))
    return permutation

def good_permutations(n, k):
    if n == 1:
        return [-1] if k > 1 else [1]
    if n % 2 == 1:
        return [-1]

    half_n = n // 2
    if k > factorial(half_n):
        return [-1]

    perm = nth_permutation(half_n, k)
    result = []
    for x in perm:
        result.append(x * 2 - 1)
        result.append(x * 2)
    return result

def main():
    t = int(input().strip())
    for _ in range(t):
        n, k = map(int, input().strip().split())
        print(*good_permutations(n, k))

main()
