#include <bits/stdc++.h>
using namespace std ;
typedef long long ll ;

int main(){
    ll n, k, ans=-1 ;
    cin >> n >> k ;
    vector<ll> a(n+1) ;
    for (int i=1; i<=n; i++){ cin >> a[i] ; a[i] += a[i-1] ; }
    if (a[n] < k ){ cout << ans << endl ; return 0 ; }
    for (int i=1; i<=n; i++){
        if (a[i]/8 >= k || k <= a[n]-a[i] ){ ans = i ; break ; }
    }
    cout << ans << endl ;
    return 0 ;
}
