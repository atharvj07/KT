t = int(input().strip())
for _ in range(t):
    s = input().strip()
    count_round = count_square = 0
    for char in s:
        if char == '(':
            count_round += 1
        elif char == ')' and count_round > 0:
            count_round -= 1
        elif char == '[':
            count_square += 1
        elif char == ']' and count_square > 0:
            count_square -= 1
    print(min(count_round, count_round) + min(count_square, count_square))
