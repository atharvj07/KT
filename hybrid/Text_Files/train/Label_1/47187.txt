import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int t = cin.nextInt();
        while (t-- > 0) {
            int n = cin.nextInt(), x = cin.nextInt();
            long[] a = new long[n];
            for (int i = 0; i < n; i++) a[i] = cin.nextLong();
            Arrays.sort(a);
            long minBeauty = (a[0] + x - 1) / x, maxBeauty = 0;
            for (long v : a) maxBeauty += (v + x - 1) / x;
            System.out.println(minBeauty + " " + maxBeauty);
        }
    }
}
