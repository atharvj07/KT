n = int(input())
l = [0] * (n + 1)
r = [0] * (n + 1)

for i in range(1, n + 1):
    l[i], r[i] = map(int, input().split())

# Find heads of all lists
heads = []
for i in range(1, n + 1):
    if l[i] == 0:
        heads.append(i)

# Connect all lists sequentially
for i in range(len(heads) - 1):
    r[heads[i]] = heads[i + 1]
    l[heads[i + 1]] = heads[i]

# Output the result
for i in range(1, n + 1):
    print(l[i], r[i])
