import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();
            String t = sc.next();
            boolean flag = true;
            for (int i = 0; i < n; i++) {
                if (s.charAt(i) != t.charAt(i)) {
                    if (s.charAt(i) == t.charAt((i + 1) % n) && s.charAt((i + 1) % n) == t.charAt(i)) {
                        flag = true;
                    } else {
                        flag = false;
                        break;
                    }
                }
            }
            System.out.println(flag ? "YES" : "NO");
        }
    }
}
