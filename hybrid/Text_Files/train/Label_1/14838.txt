#include <iostream>
#include <vector>
using namespace std;
const int MX = 2e5 + 5;
int a[MX], p[MX];
vector<pair<int, int>> ans[MX];
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    for (int i = 0; i < n + 2; i++) {
      a[i] = i;
      p[i] = i;
      ans[i].clear();
    }
    for (int i = n + 1; i > 1; i--) {
      int j = a[i] - 1;
      ans[j].push_back({i, p[i]});
      a[p[i] - 1] = i;
      p[i] = p[i] - 1;
    }
    cout << ans[1].size() << '\n';
    for (auto x : ans[1]) { cout << x.first << ' ' << x.second << '\n'; }
  }
  return 0;
}
