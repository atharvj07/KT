import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long a = sc.nextInt();
            long b = sc.nextInt();
            long k = sc.nextInt();
            System.out.println(a * k >= b && b * k >= a ? "YES" : "NO");
        }
    }
}
