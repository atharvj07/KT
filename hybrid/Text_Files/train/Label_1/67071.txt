from sys import stdin
n = int(stdin.readline())
a = [int(x)-1 for x in stdin.readline().split()]
g = [[]]
for _ in range(n):
    g.append([])
for i in range(n-1):
    if(a[i] > a[i+1]):
        g[a[i]].append(a[i+1])
        g[a[i+1]].append(a[i])
        a[i],a[i+1] = a[i+1],a[i]
vis = [False] * n
ans = 0
for i in range(n):
    if(not vis[i]):
        sz = 0
        q = [i]
        while(q):
            u = q.pop()
            if(vis[u]):
                continue
            vis[u] = True
            sz += 1
            for v in g[u]:
                if(not vis[v]):
                    q.append(v)
        ans = max(ans,sz)
print(ans)
