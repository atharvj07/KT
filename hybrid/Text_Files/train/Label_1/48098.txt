import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long n = input.nextLong();
        int k = input.nextInt();
        input.close();

        int w = 0;
        long last = n % 10L;
        while (n > 0L) {
            long cur = n % 10L;

            if ((cur < last && (k > 1 || last % k > cur % k)) || (cur == 0L && last > 0L && n > 10L))
                w++;
            else
                last = cur;

            n /= 10L;
        }

        System.out.println(w);
    }
}
