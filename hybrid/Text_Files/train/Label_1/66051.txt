t = int(input())
for _ in range(t):
    a, b = map(int, input().split())
    if b == 0:
        print(1)
    elif a > b:
        if a == b + 1:
            print(a)
        else:
            print(a - 1)
    else:
        if b == 1:
            print(2)
        else:
            if b % 2 == 0:
                if a <= b // 2:
                    print(2)
                else:
                    print(3)
            else:
                if a <= (b + 1) // 2:
                    print(2)
                else:
                    print(3)
