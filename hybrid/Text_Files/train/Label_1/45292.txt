import java.util.*;
import java.io.*;
public class Main {
    final static int maxn = 1000100;
    int p[], q[],Ans[];
    int n, m;
    class Query {
        int l1, r1, l2, r2;
    }
    Query[] queries;
    void solve() {
        int x = 0;
        for (int l = 0; l < m; l++) {
            int a = queries[l].l1, b = queries[l].r1, c = queries[l].l2, d = queries[l].r2;
            queries[l].l1 = Math.min(f(a, x), f(b, x)) + 1;
            queries[l].r1 = Math.max(f(a, x), f(b, x)) + 1;
            queries[l].l2 = Math.min(f(c, x), f(d, x)) + 1;
            queries[l].r2 = Math.max(f(c, x), f(d, x)) + 1;
            x = Ans[l];
        }
        for (int l = 0; l < m; l++)
            System.out.println(Ans[l]);
    }
    int f(int x, int add) {
        x--;
        x += add;
        x %= n;
        return (x + n) % n;
    }
    void input() {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        p = new int[maxn]; q = new int[maxn];
        Ans = new int[maxn];
        queries = new Query[maxn];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
            q[i] = in.nextInt();
        }
        m = in.nextInt();
        for (int i = 0; i < m; i++) {
            queries[i] = new Query();
            queries[i].l1 = in.nextInt();
            queries[i].r1 = in.nextInt();
            queries[i].l2 = in.nextInt();
            queries[i].r2 = in.nextInt();
        }
    }
    int count(int l1, int r1, int l2, int r2) {
        int ans = 0;
        for (int i = l1; i <= r1; i++) {
            for (int j = l2; j <= r2; j++)
                if (p[i - 1] == q[j - 1]) ans++;
        }
        return ans;
    }
    void prepare() {
        for (int i = 0; i < m; i++) {
            Ans[i] = count(queries[i].l1, queries[i].r1, queries[i].l2, queries[i].r2);
        }
    }
    public static void main(String[] args) {
        Main m = new Main();
        m.input();
        m.prepare();
        m.solve();
    }
}
