import java.util.Scanner;
import java.util.Arrays;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int s = sc.nextInt();
    int[] slices = new int[n];
    int[] happiness1 = new int[n];
    int[] happiness2 = new int[n];
    for (int i = 0; i < n; i++) {
      slices[i] = sc.nextInt();
      happiness1[i] = sc.nextInt();
      happiness2[i] = sc.nextInt();
    }

    Arrays.sort(slices);
    Arrays.sort(happiness1);
    Arrays.sort(happiness2);

    long ans = 0;
    int idx1 = 0;
    int idx2 = n - 1;
    for (int i = 0; i < n; i++) {
      if (happiness1[idx1] > happiness2[idx2]) {
        ans += happiness1[idx1] * slices[i];
        idx1++;
      } else {
        ans += happiness2[idx2] * slices[i];
        idx2--;
      }
    }

    System.out.println(ans);
  }
}
