#include <iostream>
#include <limits>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  long long sum = 0, mn = numeric_limits<long long>::max();
  for (int &x : a) {
    cin >> x;
    mn = min(mn, (long long)x);
  }
  for (int &x : a) {
    sum += (x - mn);
  }
  cout << (sum % n + n) % n << endl;
  cout << mn << " " << (sum / n) << endl;
}
