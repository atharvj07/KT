import java.util.*;
import java.lang.*;

class Main {
	public static void main (String[] args) throws java.lang.Exception {
		Scanner scanner = new Scanner(System.in);
		int t = scanner.nextInt();
		while (t-- > 0) {
			int n = scanner.nextInt();
			int[] a = new int[n];
			for (int i = 0; i < n; i++) a[i] = scanner.nextInt();
			
			boolean[] alive = new boolean[n];
			Arrays.fill(alive, true);
			
			for (int i = 0; i < n - 1; i++) {
				int p1 = (int) (Math.random() * n);
				while (!alive[p1]) p1 = (int) (Math.random() * n);
				int p2 = (int) (Math.random() * n);
				while (!alive[p2] || p1 == p2) p2 = (int) (Math.random() * n);
				
				if (a[p1] > a[p2]) alive[p2] = false;
				else alive[p1] = false;
				
				if (a[p1] == a[p2]) {
					if (Math.random() < 0.5) alive[p1] = false;
					else alive[p2] = false;
				} else if (a[p1] > a[p2]) {
					a[p1] += a[p2];
				} else {
					a[p2] += a[p1];
				}
			}
			
			int cnt = 0;
			for (int i = 0; i < n; i++) if (alive[i]) cnt++;
			System.out.println(cnt);
			for (int i = 0; i < n; i++) if (alive[i]) System.out.print((i + 1) + " ");
			System.out.println();
		}
	}
}
