for i in range(int(input())):
    n = int(input()); a = [0]; b = a.copy()
    for j in range(n):
        ai, bi = map(int,input().split())
        a.append(ai); b.append(bi)
    tm = list(map(int,input().split()))
    cur=b[0]+tm[0];
    for j in range(1,n):
        if cur<b[j]: cur=b[j]
        cur+= tm[j]
        cur+= int((b[j]-a[j])/2+0.5)
    print(cur)
