n, k = map(int, input().split())
a = [int(x) for x in input().split()]
z = 0
j = 0
for i in range(n):
    z += a[i] - 1
    if z <= k:
        j = i
z = 0
for i in range(n):
    if i <= j:
        a[i] = 1
    else:
        if z < k and a[i] == 0:
            a[i] = 1
            z += 1
print(z + 1)
print(*a)
