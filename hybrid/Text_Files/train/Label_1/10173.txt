#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, a, ans = 0, sum = 0;
    cin >> n;
    for (int i = 0; i < n; i++) {
      cin >> a;
      if (a % 3) ans++;
      sum += a % 3;
    }
    cout << ans + min(sum % 3, 1) << endl;
  }
  return 0;
}
