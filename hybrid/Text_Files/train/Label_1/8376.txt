#include<bits/stdc++.h>
int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, m, a, b, c;
    bool flag;
    std::vector<int> mo(100002, 0), m2(100002, 0), m3(100002, 0);
    std::cin >> n >> m;
    for (int i = 1; i <= n; i++) { std::cin >> a; mo[a]++; }
    for (int i = 1; i <= n; i++) { std::cin >> a; m2[a]++; }
    for (int i = 0; i < m; i++) { std::cin >> a; m3[a]++; }
    flag = true;
    for (int i = 1; i <= n; i++) {
      if (m2[i] > mo[i] + m3[i]) { flag = false; }
    }
    for (int i = 1; i <= n; i++) {
      if (mo[i] + m3[i] == 0) { flag = false; }
    }
    if (flag)
      std::cout << "YES\n";
    else
      std::cout << "NO\n";
    if (flag) {
      for (int i = 1; i <= n; i++) {
        if (mo[m2[i]] + m3[m2[i]] > m2[i]) {
          m3[m2[i]]++, m2[m2[i]]--;
        }
      }
      for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
          if (m3[j] > 0 && m2[i] < mo[i] + m3[i]) {
            std::cout << i << " ", m3[j]--, m2[i]++;
          }
        }
      }
      std::cout << "\n";
    }
  }
}
