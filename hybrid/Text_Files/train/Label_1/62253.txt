n = int(input())
s = input()

def make_good_string(s):
    result = []
    for char in s:
        if len(result) % 2 == 0 or result[-1] != char:
            result.append(char)
    if len(result) % 2:
        result.pop()
    return ''.join(result)

good_string = make_good_string(s)
print(n - len(good_string))
print(good_string)
