#include <iostream>

int main() {
  int n = 0, k = 0;
  std::cin >> n >> k;
  int a[n];
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }
  int cycle_len = n * (n + 1) / 2;
  long long c = k / cycle_len;
  long long r = k % cycle_len;
  if (r < n) {
    std::cout << a[r] << std::endl;
  } else {
    long long j = r % n;
    long long p = r / n;
    std::cout << a[j] << std::endl;
  }
  return 0;
}
