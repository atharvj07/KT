import sys

n, k = map(int, sys.stdin.readline().split())
dist = [int(x) for x in sys.stdin.readline().split()]
fuel = [int(x) for x in sys.stdin.readline().split()]
cur_fuel = fuel[0]
time = 0
for i in range(n - 1):
    if i != 0 and cur_fuel <= 0:
        time += k
        cur_fuel = fuel[i]
    time += dist[i]
    cur_fuel -= dist[i]
    cur_fuel += fuel[i]

print(time)
