import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] cups = new int[n];
        int total = 0;
        for (int i = 0; i < n; i++) {
            cups[i] = sc.nextInt();
            total += cups[i];
        }
        int diff = 0;
        int a = 0, b = 0;
        for (int i = 0; i < n; i++) {
            int d = total / n - cups[i];
            if (d > 0 && d <= cups[i]) {
                if (diff == 0) {
                    diff = d;
                    a = i + 1;
                } else if (diff != d) {
                    System.out.println("Unrecoverable configuration.");
                    return;
                }
            } else if (d < 0 || d > cups[i]) {
                System.out.println("Unrecoverable configuration.");
                return;
            }
        }
        if (diff != 0) {
            for (int i = 0; i < n; i++) {
                if (cups[i] == total / n) {
                    b = i + 1;
                    break;
                }
            }
            System.out.println(diff + " ml. from cup #" + a + " to cup #" + b + ".");
        } else {
            System.out.println("Exemplary pages.");
        }
    }
}
