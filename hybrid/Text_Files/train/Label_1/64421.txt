import sys

m, b = map(int, sys.stdin.readline().split())

if m == 1:
    print(b ** 2)
else:
    l = 1
    r = b
    max_bananas = 0
    while l <= r:
        mid = (l + r) // 2
        bananas = mid + (-mid / m + b) * m
        if bananas > max_bananas:
            max_bananas = bananas
            l = mid + 1
        else:
            r = mid - 1
    print(max_bananas)
