import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        Map<Integer, Integer> map = new HashMap<>();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] arr = new int[n];
            for (int j = 0; j < n; j++) {
                int num = 1;
                while (map.containsKey(num)) num++;
                arr[j] = num;
                map.put(num, j);
            }
            for (int j = 0; j < n; j++) System.out.print(arr[j] + " ");
            map.clear();  
            System.out.println();
        }
    }
}
