n, k = map(int, input().split())
A = list(map(int, input().split()))
ans = 0

for i in range(n):
    s = str(A[i])
    ss = 0
    for x in s:
        ss = (ss * 10 + int(x)) % k
    
    for j in range(n):
        if i == j: continue
        st = str(A[j])
        t = 0
        for x in st:
            t = (t * 10 + int(x)) % k
        ans += (ss + t) % k == 0

print(ans)
