n, A, cf, cm, m = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
cur = m * (cf / cm)
ans = []
for i in range(n):
    if a[i] < A:
        aa = a[i]
        a[i] = min(A, a[i] + cur)
        cur -= a[i] - aa
        if cur < 0:
            cur = 0
            break
    ans.append(a[i])
print(cf * ans.count(A) + cm * min(ans))
for j in ans:
    print(j, end = ' ')
