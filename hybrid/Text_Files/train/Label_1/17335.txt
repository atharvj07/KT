#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> a(2, vector<int>(n - 1));
    vector<int> b(n);
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < n - 1; j++) {
            cin >> a[i][j];
        }
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i];
    }
    int ans = 0;
    for (int i = 0; i < n - 1; i++) {
        ans += a[0][i];
    }
    ans += b[0];
    for (int i = 0; i < n - 1; i++) {
        ans += a[1][i];
    }
    ans += b[n - 1];
    cout << ans << endl;
    return 0;
}
