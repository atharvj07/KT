import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] f = new int[n+1];
        for (int i = 1; i <= n; i++) {
            f[i] = in.nextInt();
        }
        int[] g = new int[n+1];
        int[] h = new int[n+1];
        int[] cnt = new int[n+1];
        for (int i = 1; i <= n; i++) {
            cnt[f[i]]++;
        }
        int m = 0;
        for (int i = 1; i <= n; i++) {
            if (cnt[i] > 0) {
                g[i] = ++m;
            }
        }
        for (int i = 1; i <= n; i++) {
            h[g[f[i]]] = i;
        }
        if (m > 0) {
            System.out.println(m);
            for (int i = 1; i <= n; i++) {
                System.out.print(g[i] + " ");
            }
            System.out.println();
            for (int i = 1; i <= m; i++) {
                System.out.print(h[i] + " ");
            }
            System.out.println();
            return;
        }
        System.out.println(-1);
    }
}
