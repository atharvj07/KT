#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int n, m, a, b;
ll g0, x, y, z;
const int MAXN = 3005;
ll f[MAXN][MAXN], p[MAXN];

int main() {
    cin >> n >> m >> a >> b;
    cin >> g0 >> x >> y >> z;
    p[0] = g0;
    for (int i = 1; i <= n * m; i++) {
        p[i] = (p[i - 1] * x + y) % z;
    }
    f[n + 1][m + 1] = 0;
    for (int i = n; i >= 1; i--) {
        for (int j = m; j >= 1; j--) {
            f[i][j] = p[(i - 1) * m + j - 1] + f[i + 1][j] + f[i][j + 1] - f[i + 1][j + 1];
        }
    }
    ll ans = 0;
    for (int i = 1; i <= n - a + 1; i++) {
        for (int j = 1; j <= m - b + 1; j++) {
            ans += f[i][j] - f[i + a][j] - f[i][j + b] + f[i + a][j + b];
        }
    }
    cout << ans;
    return 0;
}
