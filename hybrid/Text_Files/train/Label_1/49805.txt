import java.util.*;

public class Main {
    static final int MOD = 1000000007;

    static int n, q, t;
    static int[] a;
    static int[][] dp;
    static boolean[] vis;

    static int solve(int i, int rem) {
        if (i == n) {
            return rem == 0 ? 1 : 0;
        }
        if (dp[i][rem] != -1) return dp[i][rem];
        int ret = solve(i + 1, rem);
        if (rem >= a[i]) ret = (ret + solve(i + 1, rem - a[i])) % MOD;
        dp[i][rem] = ret;
        return ret;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        q = sc.nextInt();
        t = sc.nextInt();
        a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        dp = new int[n][t + 1];
        for (int i = 0; i < n; i++) Arrays.fill(dp[i], -1);
        vis = new boolean[n];
        for (int i = 0; i < q; i++) {
            int b = sc.nextInt() - 1;
            int c = sc.nextInt() - 1;
            vis[b] = true;
            vis[c] = true;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (!vis[i]) {
                ans = (ans + solve(i + 1, t)) % MOD;
            }
        }
        System.out.println(ans);
        sc.close();
    }
}
