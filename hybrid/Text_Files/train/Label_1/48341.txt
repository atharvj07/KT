import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] points = new int[n];
        for (int i = 0; i < n; i++) {
            points[i] = sc.nextInt();
        }
        String s = sc.next();
        int[][] dp = new int[n][n];
        for (int i = 0; i < n; i++) {
            dp[i][i] = points[i];
        }
        for (int len = 2; len <= n; len++) {
            for (int i = 0; i <= n - len; i++) {
                int j = i + len - 1;
                if (s.charAt(i) == s.charAt(j) && (len == 2 || dp[i + 1][j - 1] > 0)) {
                    dp[i][j] = Math.max(dp[i][j], dp[i + 1][j - 1] + points[j]);
                    if (i + 1 <= j - 1) {
                        dp[i][j] = Math.max(dp[i][j], dp[i + 1][j] + points[i]);
                    }
                }
            }
        }
        int max = 0;
        for (int[] row : dp) {
            for (int val : row) {
                max = Math.max(max, val);
            }
        }
        System.out.println(max);
    }
}
