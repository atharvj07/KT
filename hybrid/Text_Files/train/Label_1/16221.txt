#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, t;
    cin >> n >> t;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<vector<int>> dp(n + 1, vector<int>(301, 0));
    int ans = 0;
    for (int i = 0; i <= n; i++) {
        for (int j = 1; j <= 300; j++) {
            int pre = i - 1;
            if (i == 0) pre = n;
            if (j - a[i % n] >= 1) dp[i][j] = max(dp[i][j], dp[pre][j - a[i % n]] + 1);
            dp[i][j] = max(dp[i][j], dp[pre][j]);
            ans = max(ans, dp[i][j]);
        }
    }
    cout << ans << endl;
    return 0;
}
