#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < 4; i++) {
        int a[4];
        for (int j = 0; j < 4; j++) cin >> a[j];
        sort(a, a + 4);
        if (n >= a[2] + a[3])
            cout << i + 1 << " " << a[2] << " " << a[3] << endl, exit(0);
    }
    puts("-1");
}
