import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long d = sc.nextLong();
        long n = sc.nextLong();
        int m = sc.nextInt();

        long[] a = new long[m];
        long[] b = new long[m];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextLong();
            b[i] = sc.nextLong();
        }

        PriorityQueue<Long> pq = new PriorityQueue<>();

        long ans = 0;
        long pos = 0;
        for (int i = 0; i < m; i++) {
            while (!pq.isEmpty() && pos >= pq.peek()) {
                ans -= pq.poll();
            }
            if (pos >= d) {
                System.out.println(ans);
                return;
            }
            if (pos + n < a[i]) {
                System.out.println(-1);
                return;
            }
            pq.add(a[i]);
            ans += b[i];
            pos = a[i];
        }

        while (!pq.isEmpty() && pos >= pq.peek()) {
            ans -= pq.poll();
        }
        if (pos >= d) {
            System.out.println(ans);
        } else {
            System.out.println(-1);
        }
    }
}
