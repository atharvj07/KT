import math
n = int(input())
nums, ans = [int(i) for i in input().split()], []
for k in range(1, n):
    count = 0
    for i in range(1, math.ceil(n/k)):
        for j in range(k * i - k + 1, k * i + 1):
            if j > n: 
                break
            if nums[j] < nums[(j-1)//k]: 
                count += 1
    ans.append(count)
print(*ans)
