#include <iostream>
using namespace std;
int main() {
    int y;
    cin >> y;
    y++;
    for (; y < 100000; y++) {
        if (((y % 4 == 0) && (y % 100 != 0)) || (y % 400 == 0)) {
            if (y % 7 == 5) break;
        } else if (y % 7 == 2) break;
    }
    cout << y << endl;
    return 0;
}
