#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define ll long long

vector<int> adj[200001];
int n, m, k, a[200001], dis[200001], vis[200001], sp[200001];
pair<int, int> edges[200001];

void dfs(int x) {
    vis[x] = 1;
    for (auto u : adj[x]) {
        if (!vis[u]) {
            if (sp[x] && sp[u]) { dis[u] = dis[x] + 1; }
            dfs(u);
        }
    }
}

int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= k; i++) { cin >> a[i]; }
    for (int i = 1; i <= m; i++) {
        int x, y;
        cin >> x >> y;
        adj[x].pb(y);
        adj[y].pb(x);
        edges[i] = {x, y};
    }
    for (int i = 1; i <= n; i++) { sort(adj[i].begin(), adj[i].end()); }
    for (int i = 1; i <= k; i++) { sp[a[i]] = 1; }
    int cnt = 1;
    int dia = 0;
    int cur = 1;
    dfs(1);
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) {
            dfs(i);
            cnt++;
        }
    }
    for (int i = 1; i <= n; i++) { dis[i] = 0; }
    for (int i = 1; i <= n; i++) { vis[i] = 0; }
    for (int i = 1; i <= n; i++) {
        if (dis[i] > dis[cur]) { cur = i; }
    }
    dfs(cur);
    for (int i = 1; i <= n; i++) { dia = max(dia, dis[i]); }
    int far = cur;
    for (int i = 1; i <= n; i++) { dis[i] = 0; }
    for (int i = 1; i <= n; i++) { vis[i] = 0; }
    dfs(far);
    for (int i = 1; i <= n; i++) { dia = max(dia, dis[i]); }
    int ans = dia;
    for (int i = 1; i <= m; i++) {
        int x = edges[i].first;
        int y = edges[i].second;
        if (sp[x] && sp[y]) {
            sp[x] = 0;
            sp[y] = 0;
            for (int j = 1; j <= n; j++) { dis[j] = 0; }
            for (int j = 1; j <= n; j++) { vis[j] = 0; }
            dfs(1);
            for (int j = 1; j <= n; j++) {
                if (!vis[j]) {
                    dfs(j);
                    cnt++;
                }
            }
            for (int j = 1; j <= n; j++) { dis[j] = 0; }
            for (int j = 1; j <= n; j++) { vis[j] = 0; }
            for (int j = 1; j <= n; j++) {
                if (dis[j] > dis[cur]) { cur = j; }
            }
            dfs(cur);
            for (int j = 1; j <= n; j++) { dia = max(dia, dis[j]); }
            far = cur;
            for (int j = 1; j <= n; j++) { dis[j] = 0; }
            for (int j = 1; j <= n; j++) { vis[j] = 0; }
            dfs(far);
            for (int j = 1; j <= n; j++) { dia = max(dia, dis[j]); }
            ans = max(ans, dia);
            sp[x] = 1;
            sp[y] = 1;
        }
    }
    cout << ans << endl;
}
