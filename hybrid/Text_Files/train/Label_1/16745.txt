#include<bits/stdc++.h>
using namespace std;
int main(){
    int n, a[1005];
    cin >> n;
    for(int i = 0; i < n; i++) cin >> a[i];
    for(int i = 0; i < n; i++) a[i] = (a[i] - i + n) % n;
    for(int i = 0; i < n; i++) if(a[i] != i){
        cout << "No" << endl;
        return 0;
    }
    cout << "Yes" << endl;
}
