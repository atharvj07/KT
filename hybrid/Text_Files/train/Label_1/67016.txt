import sys
import math

n, k = map(int, sys.stdin.readline().split())

if k == 2:
    print(3)
    print('3 2 2 1 2 2 1 3 1 1')
    exit()

print(2)

colors = []
for i in range(1, n):
    for j in range(i+k, n+1):
        colors.append(1 if (j-i)%2==1 else 2)

for i in range(n-k):
    for j in range(i+1, i+k):
        colors.append(1 if (j-i)%2==1 else 2)

print(' '.join(map(str, colors)))
