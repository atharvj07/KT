#include <bits/stdc++.h>
using namespace std;
int a[105], n;
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        int p = 1, odd = 0;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            if (a[i] & 1)
                odd++;
            else
                p *= a[i];
        }
        if (odd > 1 || p != 1)
            cout << "YES\n";
        else
            cout << "NO\n";
    }
    return 0;
}
