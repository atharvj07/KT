n = int(input())
and_mask = 1023
or_mask = 0
xor_mask = 0

for _ in range(n):
    op, x = input().split()
    x = int(x)
    if op == "&":
        and_mask &= x
    elif op == "|":
        or_mask |= x
    elif op == "^":
        xor_mask ^= x

# Simplifying the operations
and_mask &= 1023
or_mask &= 1023
xor_mask &= 1023

res = []
if and_mask != 1023:
    res.append(f"& {and_mask}")
if or_mask != 0:
    res.append(f"| {or_mask}")
if xor_mask != 0:
    res.append(f"^ {xor_mask}")

print(len(res))
for r in res:
    print(r)
