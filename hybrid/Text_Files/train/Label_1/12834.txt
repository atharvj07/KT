#include <cmath>
#include <iostream>
using namespace std;
typedef long long ll;

int main() {
  ll a, b, c, x1, y1, x2, y2;
  cin >> a >> b >> c >> x1 >> y1 >> x2 >> y2;
  ll dx = x2 - x1, dy = y2 - y1;
  ll x = x1, y = y1;
  ll ans = abs(dx) + abs(dy);
  if (a * x1 + b * y1 + c != 0) {
    double t = static_cast<double>(-(a * x1 + b * y1 + c)) / (a * dx + b * dy);
    if (t >= 0 && t <= 1) {
      ll xx = round(x1 + t * dx), yy = round(y1 + t * dy);
      ans = min(ans, abs(x1 - xx) + abs(y1 - yy) + abs(xx - x2) + abs(yy - y2));
    }
  }
  if (a * x2 + b * y2 + c != 0) {
    double t = static_cast<double>(-(a * x2 + b * y2 + c)) / (a * dx + b * dy);
    if (t >= 0 && t <= 1) {
      ll xx = round(x1 + t * dx), yy = round(y1 + t * dy);
      ans = min(ans, abs(x1 - xx) + abs(y1 - yy) + abs(xx - x2) + abs(yy - y2));
    }
  }
  cout << ans << "\n";
  return 0;
}
