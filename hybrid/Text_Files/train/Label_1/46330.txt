import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);

    long n = in.nextLong();
    long x = in.nextInt();

    if (n % 2 == 0) {
      System.out.println(x);
    } else {
      System.out.println((x + 1) % 3);
    }
  }
}
