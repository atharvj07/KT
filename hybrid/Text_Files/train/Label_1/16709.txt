#include <iostream>
int main() {
    char s[11];
    int cnt[10] = {};
    std::cin >> s;
    for (char &c : s) ++cnt[c - '0'];
    for (int i = 0; i < 10; i++)
        if (cnt[i] != cnt[(i + 1) % 10]) {
            std::cout << "No" << std::endl;
            return 0;
        }
    std::cout << "Yes" << std::endl;
    return 0;
}
