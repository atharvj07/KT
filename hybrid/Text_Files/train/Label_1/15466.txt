#include <bits/stdc++.h>
using namespace std;
const int MX=3e5+123, LG=21;
int n, a[MX], ans, dp[LG][MX], p2[LG];
int get(int l, int r){
    int x=0;
    for(int j=LG-1; j>=0; j--)
        if(l+(1<<j)-1<=r)
            x^=dp[j][l+(1<<j)-1], l+=(1<<j);
    return x;
}
int main(){
    cin>>n;
    for(int i=1; i<=n; i++) cin>>a[i];
    p2[0]=1;
    for(int i=1; i<=LG; i++) p2[i]=p2[i-1]<<1;
    for(int i=1; i<=n; i++)
        for(int j=1; j<=LG; j++)
            if(i+p2[j]-1<=n)
                dp[j][i]=dp[j-1][i]^dp[j-1][i+p2[j-1]];
    for(int i=1; i<=n; i++)
        dp[0][i]=a[i];
    for(int l=1; l<=n; l++){
        int x=0;
        for(int r=l; r<=n; r++){
            x^=a[r];
            int y=get(l, r-1);
            if(x==y) ans++;
        }
    }
    cout<<ans<<'\n';
}
