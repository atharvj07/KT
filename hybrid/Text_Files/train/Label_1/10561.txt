#include <iostream>
#include <vector>

int main() {
  int n, m, q;
  std::cin >> n >> m >> q;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) { std::cin >> a[i]; }
  std::vector<int> cnt(m);
  for (auto ai : a) cnt[ai - 1]++;
  std::vector<int> order(m);
  for (int i = 0; i < m; i++) order[i] = i + 1;
  for (int i = 0; i < m; i++) {
    for (int j = i + 1; j < m; j++) {
      if (cnt[order[i] - 1] > cnt[order[j] - 1]) {
        std::swap(order[i], order[j]);
      } else if (cnt[order[i] - 1] == cnt[order[j] - 1]) {
        if (order[i] > order[j]) {
          std::swap(order[i], order[j]);
        }
      }
    }
  }
  for (int i = 0; i < q; i++) {
    int k;
    std::cin >> k;
    int x = (k - n - 1) % m;
    std::cout << order[x] << std::endl;
  }
  return 0;
}
