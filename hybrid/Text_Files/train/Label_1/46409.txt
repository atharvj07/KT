import java.util.*;

class Main {
    static int[] dist, path, p;
    static boolean[] used;

    static void dijkstra(int[] dist, int n, int c, long[] sum) {
        PriorityQueue<Integer> pq = new PriorityQueue<>((a, b) -> dist[a] - dist[b]);
        pq.add(c);
        dist[c] = 0;
        for (int i = 1; i <= n; i++) {
            if (i == c) {
                continue;
            }
            dist[i] = Integer.MAX_VALUE;
        }
        while (!pq.isEmpty()) {
            int cur = pq.poll();
            if (used[cur]) {
                continue;
            }
            used[cur] = true;
            for (int[] e : graph[cur]) {
                int newDist = dist[cur] + e[1];
                int u = e[0];
                if (newDist < dist[u]) {
                    dist[u] = newDist;
                    path[u] = cur;
                    pq.add(u);
                }
            }
        }
        long sum_ = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] <= c) {
                sum_ += p[i];
            }
        }
        sum[0] = sum_;
    }

    static void dfs(int v, long t, long[] tavas, long[] nafas) {
        if (used[v]) {
            return;
        }
        used[v] = true;
        tavas[0] += p[v];
        nafas[0] += p[v];
        for (int[] e : graph[v]) {
            if (dist[v] + e[1] <= t) {
                nafas[0] -= p[e[0]];
                dfs(e[0], t, tavas, nafas);
                nafas[0] += p[e[0]];
            } else {
                tavas[0] += p[e[0]];
                dfs(e[0], t, tavas, nafas);
                tavas[0] -= p[e[0]];
            }
        }
    }

    static List<int[]>[] graph;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        long[] sum = new long[1];
        dist = new int[n + 1];
        p = new int[n + 1];
        path = new int[n + 1];
        used = new boolean[n + 1];
        graph = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }
        int s = in.nextInt();
        int t = in.nextInt();
        for (int i = 1; i <= n; i++) {
            p[i] = in.nextInt();
        }
        for (int i = 0; i < m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            int w = in.nextInt();
            graph[u].add(new int[]{v, w});
            graph[v].add(new int[]{u, w});
        }
        dijkstra(dist, n, s, sum);
        used = new boolean[n + 1];
        long[] tavas = new long[1];
        long[] nafas = new long[1];
        dfs(s, sum[0], tavas, nafas);
        used = new boolean[n + 1];
        dijkstra(dist, n, t, sum);
        long sum_ = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] <= sum[0]) {
                sum_ += p[i];
            }
        }
        dfs(t, sum[0], tavas, nafas);
        if (tavas[0] > sum_) {
            System.out.println("Break a heart");
        } else if (tavas[0] < sum_) {
            System.out.println("Cry");
        } else {
            System.out.println("Flowers");
        }
    }
}
