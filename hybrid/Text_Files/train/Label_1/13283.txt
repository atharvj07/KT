#include <iostream>
using namespace std;
typedef long long ll;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int T;
  cin >> T;
  while (T--) {
    ll A;
    cin >> A;
    if (A == 2) {
      cout << 3 << '\n';
    } else if (A % 2) {
      cout << (A + 1) / 2 << '\n';
    } else {
      cout << A / 2 << '\n';
    }
  }
}
