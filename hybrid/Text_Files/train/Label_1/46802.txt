import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        double t = sc.nextInt();
        double[] a = new double[n];
        double[] b = new double[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        for (int i = 0; i < n; i++) b[i] = sc.nextInt();
        double l = 0;
        double r = 1e9;
        for (int i = 0; i < 100; i++) {
            double m = (l + r) / 2;
            double sum = 0;
            for (int j = 0; j < n; j++) {
                sum += a[j] * b[j] / (t + m * b[j]);
            }
            if (sum > t) l = m;
            else r = m;
        }
        double sum = 0;
        for (int j = 0; j < n; j++) {
            sum += a[j] * b[j] / (t + r * b[j]);
        }
        if (Math.abs(sum - t) < 1e-6) System.out.println(r);
        else System.out.println(0);
    }
}
