n, k = map(int, input().split())
a = list(map(int, input().split()))

a.sort()

i, j = 0, n - 1
while i < j and k > 0:
    if i+1 < n-j:  # More numbers on the left side
        diff = a[i+1] - a[i]
        if k >= diff * (i+1):
            k -= diff * (i+1)
            i += 1
        else:
            a[i] += k // (i+1)
            k = 0
    else:  # More numbers on the right side or equal
        diff = a[j] - a[j-1]
        if k >= diff * (n-j):
            k -= diff * (n-j)
            j -= 1
        else:
            a[j] -= k // (n-j)
            k = 0

print(a[j] - a[i])
