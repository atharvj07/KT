#include <bits/stdc++.h>
using namespace std;
int a[200005], b[200005];
int main() {
  int n, q;
  cin >> n >> q;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < q; i++) cin >> b[i];
  for (int i = 0; i < q; i++) {
    int ans = -1;
    for (int j = 0; j < n; j++) {
      if (b[i] % a[j] == 0) {
        ans = 0;
        break;
      }
    }
    if (ans == 0) {
      int cnt = 0;
      while (b[i]) {
        for (int j = 0; j < n; j++) {
          if (b[i] % a[j] == 0) {
            cnt++;
            b[i] /= a[j];
            break;
          }
        }
      }
      ans = cnt;
    }
    cout << ans << endl;
  }
  return 0;
}
