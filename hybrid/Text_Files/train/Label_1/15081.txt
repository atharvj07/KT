#include <bits/stdc++.h>
#define ll long long

using namespace std;

const ll mod = 1e9+7;
const ll nMod = mod*mod;

ll pw[4010];

ll power(ll b, ll p) {
    if (p == 0) return 1;
    if (p == 1) return b;
    if (p % 2 == 0) {
        ll ret = power(b, p/2) % mod;
        return ((ret * ret) % mod);
    } else {
        return (b * power(b, p-1)) % mod;
    }
}

int main() {
    pw[0] = 1;
    for (ll i = 1; i < 4010; i++) {
        pw[i] = (pw[i-1] * 2) % mod;
    }
    ll n;
    cin >> n;
    ll res = (pw[n] - pw[n-1] + nMod) % mod;
    cout << res << endl;
    return 0;
}
