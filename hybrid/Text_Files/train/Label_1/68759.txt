import sys
input = sys.stdin.readline

n, a = map(int, input().split())
d, c = [0] * n, [0] * n
for i in range(n):
    d[i], c[i] = map(int, input().split())

dp = [0] * n
gaps = [0] * n
gaps[0] = 0
for i in range(1, n):
    gaps[i] = (d[i] - d[i - 1])**2
dp[n - 1] = a - c[n - 1]
for i in range(n - 2, -1, -1):
    dp[i] = max(dp[i + 1], a - c[i] - gaps[i + 1])

print(max(dp))
