n = int(input())
a = list(map(int, input().split()))
adj = [[] for i in range(n+1)]
for _ in range(n-1):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)
ans = [0]*n

def dfs(x, p, c):
    ans[x-1] = c+a[x-1]
    for u in adj[x]:
        if u!=p: dfs(u, x, c+a[x-1]-2*a[u-1])

dfs(1, 1, 0)
print(*ans)
