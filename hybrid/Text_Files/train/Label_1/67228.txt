def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        seg = []
        for _ in range(n):
            l, r = map(int, input().split())
            seg.append((l, r))
        seg.sort(key=lambda x: (x[0], -x[1]))
        ans = [-1] * n
        last = -1
        for i in range(n):
            l, r = seg[i]
            if l > last:
                last = r
                ans[i] = 1
        if -1 in ans:
            for i in range(n):
                if ans[i] == -1:
                    ans[i] = 2
        if len(set(ans)) == 2:
            print(*ans)
        else:
            print(-1)

solve()
