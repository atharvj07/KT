#include <iostream>

int res, b;

int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}

int main() {
    std::cin >> b;
    for (int i = 1; i * i <= b; i++) {
        if (b % i == 0) {
            res++;
            if (i * i != b) res++;
        }
    }
    std::cout << res << std::endl;
    return 0;
}
