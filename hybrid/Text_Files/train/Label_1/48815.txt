import java.util.*;

public class Main {
    private static final Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        while (sc.hasNextLine()) {
            String enc = sc.nextLine();

            int max = -1;
            String res = null;
            for (int shift = -26; shift <= 26; shift++) {
                String dec = decode(enc, shift);
                int lf = 0;
                for (String w : new String[]{"the", "this", "that"}) {
                    if (dec.contains(w)) {
                        if (++lf > 1) {
                            break;
                        }
                    }
                }
                if (lf > max) {
                    max = lf;
                    res = dec;
                }
            }

            System.out.println(res);
        }
    }

    private static String decode(String enc, int shift) {
        char[] chars = enc.toCharArray();
        for (int i = 0; i < chars.length; i++) {
            if (chars[i] >= 'a' && chars[i] <= 'z') {
                chars[i] = (char) ('z' - (26 + shift - (chars[i] - 'a')) % 26);
            }
        }
        return String.valueOf(chars);
    }
}
