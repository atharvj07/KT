#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 998244353;

ll pw(ll a, ll n) {
    if (n == 0) return 1;
    ll r = pw(a, n / 2);
    r = (r * r) % mod;
    if (n & 1) return (r * a) % mod;
    return r % mod;
}

int main() {
    ll n, k;
    cin >> n >> k;
    cout << (pw(n, k) - pw(n - 1, k) + mod) % mod << endl;
}
