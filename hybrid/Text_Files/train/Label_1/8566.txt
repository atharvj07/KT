#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int N = 3e5+10;
vector<ll> a(N), b(N);

int main() {
    ll n, ans = 0;
    cin >> n;
    for(int i = 0; i < n; i++) {
        cin >> a[i] >> b[i];
        if(a[i] > b[i]) continue;
        ans += a[i];
    }
    sort(a.begin(), a.begin()+n, greater<ll>());
    sort(b.begin(), b.begin()+n, greater<ll>());
    for(int i = 0; i < n; i++) {
        if(ans < b[i]) continue;
        ans -= b[i];
        ++ans;
    }
    cout << ans << endl;
    return 0;
}
