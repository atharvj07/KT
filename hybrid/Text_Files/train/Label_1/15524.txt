#include <bits/stdc++.h>
#define ll long long
using namespace std;
typedef pair<ll, ll> pi;
#define vi(x) vector<x>
#define pb push_back
const ll mod = 1e9 + 7;
const char nl = '\n';
int main() {
    ll n, a, b;
    cin >> n >> a >> b;
    vi(pi) v(n);
    for (ll i = 0; i < n; i++) {
        ll x, vx, vy;
        cin >> x >> vx >> vy;
        v[i] = {vx - vy * a, vy + vx * a};
    }
    sort(v.begin(), v.end());
    ll ans = 0, cur = 0;
    for (auto p : v) {
        cur += p.first;
        ans = max(ans, cur - p.second);
    }
    cout << max(ans, 0LL) << nl;
    return 0;
}
