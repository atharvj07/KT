#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m, k, s;
  cin >> n >> m >> k >> s;
  vector<vector<int>> a(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> a[i][j];
    }
  }
  vector<int> q(s);
  for (int i = 0; i < s; i++) {
    cin >> q[i];
  }
  vector<vector<int>> dp(s, vector<int>(k + 1, -1));
  dp[0][q[0]] = 0;
  for (int i = 1; i < s; i++) {
    for (int j = 1; j <= k; j++) {
      if (dp[i - 1][j] != -1) {
        for (int l = 0; l < n; l++) {
          for (int r = 0; r < m; r++) {
            if (a[l][r] == j) {
              dp[i][q[i]] = max(dp[i][q[i]], dp[i - 1][j] + abs(l - (i - 1) / m) + abs(r - (i - 1) % m));
            }
          }
        }
      }
    }
  }
  int ans = 0;
  for (int i = 1; i <= k; i++) {
    ans = max(ans, dp[s - 1][i]);
  }
  cout << ans << endl;
  return 0;
}
