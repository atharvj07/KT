#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n), vis(n);
  for (int x, i = 0; i < n; i++) cin >> x, b[i] = x ? x - 1 : -1;
  for (int x, i = 0; i < n; i++) cin >> x, a[i] = x ? x - 1 : -1;
  int s = -1, ans = 0;
  for (int i = 0; i < n; i++) {
    if (!~b[i]) s = i;
    if (!~a[i]) vis[i] = 1, ans++;
  }
  while (~s) {
    vis[s] = 1;
    s = b[s];
    ans++;
  }
  cout << ans << endl;
  for (int i = 1; i <= n; i++)
    if (vis[i - 1]) cout << i << " ";
}
