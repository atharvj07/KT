#include <vector>
#include <iostream>
#include <set>

using namespace std;

const int M = 1e9 + 7;
typedef long long ll;

vector<ll> powm;

void pre_compute(int n) {
    powm.resize(n + 1);
    powm[0] = 1;
    for(int i = 1; i <= n; i++) {
        powm[i] = (2 * powm[i - 1]) % M;
    }
}

ll add(ll a, ll b) {
    return (a + b) % M;
}

ll mul(ll a, ll b) {
    return (a * b) % M;
}

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> out(n);
    for(int i = 1; i <= k; i++) out[i - 1] = i + 1;
    for(int i = k + 1; i <= n; i++) out[i - 1] = -1;
    pre_compute(k * 2 + 1);
    ll ans = 0, inv = powm[k * 2];
    set<int> taken;
    for(int i = k; i < n; i++) {
        if(out[i - k] > 0) {
            int j = i - k;
            if(out[j] > out[i] && !taken.count(out[j])) taken.insert(out[j]), out[j] = out[i];
            else {
                ans = add(ans, mul(mul(powm[2], mul(powm[i - j - k], inv)), powm[n - (i + k)]));
            }
        }
    }
    taken.clear();
    inv = powm[k];
    for(int i = k + 1; i <= 2 * k; i++) {
        if(out[i - k] > 0) {
            int j = i - k;
            if(out[j] > out[i] && !taken.count(out[j])) taken.insert(out[j]), out[j] = out[i];
            else {
                ans = add(ans, mul(mul(powm[k * 2 - j], powm[n - (i + k)]), mul(inv, inv)));
            }
        }
    }
    bool first = true;
    for(int i : out) {
        if(i == -1) {
            if(first) {
                ans = add(ans, powm[n - 1]);
                first = false;
            }
            continue;
        }
        first = false;
        if(i <= k) {
            ans = add(ans, mul(powm[2], powm[n - 2]));
        } else {
            ans = add(ans, mul(powm[k * 2], powm[n - 2]));
        }
    }
    cout << ans << "\n";
    return 0;
}
