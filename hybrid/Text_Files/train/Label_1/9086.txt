#include <iostream>
#include <vector>
using namespace std;

int main() {
  int N, max_cnt = 0, winner = 0, past_winner = 0;
  cin >> N;
  vector<int> likes(N + 1, 0);
  for (int i = 1; i <= N; i++) {
    int a;
    cin >> a;
    likes[a]++;
    if (likes[a] > max_cnt) {
      max_cnt = likes[a];
      winner = a;
      past_winner = winner;
    } else if (likes[a] == max_cnt && a < past_winner) {
      past_winner = winner;
      winner = a;
    }
  }
  cout << winner << endl;
  return 0;
}
