#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  bool ok = true;
  int a = 0, b = 1, c = 2;
  for (int i = 0; i < n && ok; i++) {
    if (v[i] == a) {
      if (i % 2 == 0)
        ok = false;
      else
        c = b, b = a, a = c;
    } else if (v[i] == b) {
      if (i % 2 == 1)
        ok = false;
      else
        c = a, a = b, b = c;
    } else if (v[i] == c) {
      if (i % 2 == 0)
        ok = false;
      else
        b = c, c = a, a = b;
    }
  }
  cout << (ok ? "YES" : "NO") << endl;
  return 0;
}
