import bisect
for _ in range(int(input())):
    n,q=map(int,input().split())
    a=list(map(int,input().split()))
    b=[a[0]]
    for x in sorted(a[1:]):
        i=bisect.bisect_left(b,x)
        b=b[:i]+[x]+b[i:]
    for _ in range(q):
        print(["No","Yes"][bisect.bisect_left(b,int(input()))&1])
