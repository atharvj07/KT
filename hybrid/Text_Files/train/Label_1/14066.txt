#include <bits/stdc++.h>
using namespace std;
vector<int> ans;
int main() {
  int n;
  cin >> n;
  ans.resize(n << 1);
  vector<pair<int, int>> val;
  for (int i = 0; i < (n << 1); ++i) {
    int x;
    cin >> x;
    val.push_back({x, i});
  }
  sort(val.begin(), val.end());
  long long psum = 0, ssum = 0;
  int i = 0, j = (n << 1) - 1;
  bool ok = false;
  while (i < j) {
    psum += val[i].first;
    ssum += val[j].first;
    if (psum != ssum) {
      ok = true;
      break;
    }
    ++i;
    --j;
  }
  if (!ok) {
    cout << "-1" << endl;
    return 0;
  }
  for (int i = 0; i < (n << 1); ++i) { ans[val[i].second] = val[i].first; }
  for (int x : ans) cout << x << ' ';
  cout << endl;
  return 0;
}
