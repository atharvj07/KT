#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n + 1, 0);
  int cnt = 0;
  for (int i = 2; i <= n; i++) {
    if (!a[i]) {
      a[i] = ++cnt;
      for (int j = i * 2; j <= n; j += i) { a[j] = cnt; }
    }
  }
  for (int i = 2; i <= n; i++) cout << a[i] << " ";
  cout << "\n";
  return 0;
}
