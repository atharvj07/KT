import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Walk {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int t = scanner.nextInt();
        int q = scanner.nextInt();

        List<House> houses = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int a = scanner.nextInt();
            int d = scanner.nextInt();
            houses.add(new House(a, d));
        }

        List<Integer> importantPeople = new ArrayList<>();
        for (int i = 0; i < q; i++) {
            importantPeople.add(scanner.nextInt());
        }

        for (int i = 0; i < q; i++) {
            int houseIndex = importantPeople.get(i) - 1;
            House house = houses.get(houseIndex);
            int position = house.getPositionAfterTSeconds(t);
            System.out.println(position);
        }
    }

    private static class House {
        private final int position;
        private final int direction;

        public House(int position, int direction) {
            this.position = position;
            this.direction = direction;
        }

        public int getPositionAfterTSeconds(int t) {
            return position + t * direction;
        }
    }
}
