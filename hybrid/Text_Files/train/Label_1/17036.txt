#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    char w[n];
    for (int i = 0; i < n; i++) {
      if (s[i] == s[i + n - 1])
        w[i] = s[i];
      else
        w[i] = '0';
    }
    w[n] = '\0';
    cout << w << endl;
  }
  return 0;
}
