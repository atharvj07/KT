class Solution:
    def solve(self, n, m, k):
        a = [[0] * (m+2) for _ in range(n+2)]
        for _ in range(k):
            r, c = map(int, input().split())
            if a[r][c] == a[r][c+1] == a[r+1][c] == a[r+1][c+1] == 1: return _+1
            a[r][c] = 1
        return 0
print(Solution().solve(*map(int, input().split())))
