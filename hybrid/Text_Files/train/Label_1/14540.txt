#include <iostream>
#include <string>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        std::string s;
        std::cin >> s;
        int cnt = 0, ans = 0;
        for (char c : s) {
            if (c == '0')
                ++cnt;
            else if (cnt) {
                ++ans;
                cnt = 0;
            }
        }
        std::cout << ans << '\n';
    }
    return 0;
}
