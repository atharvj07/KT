n, w = map(int, input().split())
a = list(map(int, input().split()))

half_fills = [(i + 1) // 2 for i in a]
remaining_tea = w - sum(half_fills)
if remaining_tea < 0:
    print("-1")
else:
    order = sorted(range(n), key=lambda x: a[x], reverse=True)
    filled = [half for half in half_fills]
    for i in order:
        can_fill = min(a[i] - filled[i], remaining_tea)
        filled[i] += can_fill
        remaining_tea -= can_fill
        if remaining_tea == 0:
            break
    if remaining_tea > 0:
        print("-1")
    else:
        print(' '.join(map(str, filled)))
