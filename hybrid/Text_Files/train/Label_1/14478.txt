#include <cstdio>

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n, k, l;
    scanf("%d %d %d", &n, &k, &l);
    int a[n];
    for (int i = 0; i < n; i++) {
      scanf("%d", &a[i]);
    }
    bool ok = true;
    for (int j = k - 1; j < n * k && ok; j += k * 2 - 2) {
      for (int i = 0; i < n; i++) {
        if (a[i] + (j / k % (k * 2) >= k ? (k * 2 - 1 - (j / k % (k * 2)))
                                             : j / k % (k * 2))
          > l) {
          ok = false;
          break;
        }
      }
    }
    puts(ok ? "Yes" : "No");
  }
}
