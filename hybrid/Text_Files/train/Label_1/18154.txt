#include <bits/stdc++.h>
using namespace std;

int a[100000], x[100000], l[100000], r[100000];

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    x[i] = i;
  }
  sort(x, x + n, [&](int i, int j) { return a[i] < a[j]; });
  for (int i = 0; i < n; ++i) {
    int j = i + 1;
    while (j < n && a[x[j]] < a[x[i]] * 2) ++j;
    r[x[i]] = j - 1;
  }
  for (int i = n - 1; i >= 0; --i) {
    int j = i - 1;
    while (j >= 0 && a[x[j]] < a[x[i]] * 2) --j;
    l[x[i]] = j + 1;
  }
  int ans = 0, idx = -1;
  for (int i = 0; i < n; ++i) {
    if (l[i] && r[i] && r[i] - l[i] + 1 > ans) {
      ans = r[i] - l[i] + 1;
      idx = i;
    }
  }
  cout << ans << endl;
  for (int i = l[idx]; i <= r[idx]; ++i) cout << a[x[i]] << ' ';
  cout << endl;
  return 0;
}
