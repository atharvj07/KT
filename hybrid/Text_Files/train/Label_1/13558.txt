#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAX = (int)1e5;

ll a[MAX + 5], ans = 1;
ll pre[MAX + 5];
ll dp[MAX + 5][3];

int main() {
  int n, s, f;
  cin >> n >> s >> f;
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  for (int i = 0; i <= n; i++) {
    pre[i] = a[i] + (i ? pre[i - 1] : 0);
  }

  for (int i = s; i <= f; i++) {
    if (i > 1) {
      dp[i][1] = max(dp[i][1], dp[i - 1][0] + pre[i - 1]);
      dp[i][2] = max(dp[i][2], dp[i - 1][1] + pre[i - 1]);
    }
    dp[i][0] = max({dp[i][0], dp[i - 1][0], dp[i - 1][1] + pre[i - 1], dp[i - 1][2] + pre[i - 1]});
  }

  for (int i = s; i <= f; i++) {
    if (dp[i][0] > dp[ans][0]) { ans = i; }
  }

  cout << ans << endl;
}
