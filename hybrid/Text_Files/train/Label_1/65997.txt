import sys

n, k, p = map(int, input().split())
a = sorted(map(int, input().split()))
if (sum(a) % 2 == 1):
    if p == k:
        print("YES")
        print(*[(1, y) for y in a])
        sys.exit()
    else:
        print("NO")
        sys.exit()
if p == k:
    if sum(a) % 2 == 0:
        print("YES")
        print(*[(1, y) for y in a])
        sys.exit()
    else:
        print("NO")
        sys.exit()
if k - p < 1:
    print("NO")
    sys.exit()
i = 0
odd = []
while i < n:
    if a[i] % 2 == 1:
        odd.append(a[i])
        i += 1
    else:
        break
j = len(odd)
if j < k - p:
    print("NO")
    sys.exit()
if j == k - p and sum(odd) % 2 == 1:
    print("NO")
    sys.exit()
if j == k - p and sum(odd) % 2 == 0:
    even = [odd]
    odd = []
    for y in a[i:]:
        if y % 2 == 0:
            odd.append(y)
        else:
            even.append([(1, y)])
    if len(odd) % 2 == 0:
        odd.extend(even[0])
        even = even[1:]
    for y in odd:
        even.append([(1, y)])
    print("YES")
    print("\n".join([" ".join(map(str, y)) for y in even]))
    sys.exit()
add = a[i:]
odd.sort()
i = 0
while i < len(add):
    if len(odd) == 0:
        break
    if add[i] % 2 == 1:
        odd.append(add[i])
        add = add[:i] + add[i+1:]
        i -= 1
    i += 1
odd.sort()
add.sort()
j = len(odd)
if j < k - p:
    print("NO")
    sys.exit()
if j > k - p:
    print("NO")
    sys.exit()
odd = odd[:k-p]
even = []
for y in odd:
    add.remove(y)
for y in add:
    if y % 2 == 0:
        even.append(y)
for y in odd:
    even.append([(1, y)])
if len(even) % 2 == 0:
    even.extend(add[0])
    add = add[1:]
for y in add:
    even.append([(1, y)])
print("YES")
print("\n".join([" ".join(map(str, y)) for y in even]))
