n, m = map(int, input().strip().split())
pic = [list(input().strip()) for _ in range(n)]
#   horizontal check
cnt = 0
for i in range(n - 1):
    for j in range(m - 1):
        if set(pic[i][j:j+2] + pic[i+1][j:j+2]) == set("face"): cnt += 1
#   vertical check
for i in range(n - 1):
    for j in range(m):
        if set(pic[i][j] + pic[i+1][j]) in set(["fa", "ce"]):
            if set(pic[i][j+1] + pic[i+1][j+1]) in set(["fa", "ce"]):
                cnt += 1
print(cnt)
