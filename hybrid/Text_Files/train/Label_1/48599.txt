import java.util.Scanner;

public class Jigsaw {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int h = scanner.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        int[] c = new int[n];
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
            c[i] = scanner.nextInt();
            d[i] = scanner.nextInt();
        }

        if (isPossible(n, h, a, b, c, d)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    static boolean isPossible(int n, int h, int[] a, int[] b, int[] c, int[] d) {
        int[] heights = new int[n + 1];
        for (int i = 0; i < n; i++) {
            heights[i] = h - a[i] - b[i];
            heights[i + 1] = h;
        }

        int[] widths = new int[n + 2];
        widths[0] = 0;
        for (int i = 0; i < n; i++) {
            widths[i + 1] = widths[i] + a[i] + b[i] + 2;
        }

        for (int i = 0; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (widths[j] - widths[i] > 100) {
                    continue;
                }

                int heightDifference = heights[i] - heights[j];

                if (heightDifference == c[j - 1]) {
                    if (d[i] >= heightDifference) {
                        continue;
                    }
                } else if (heightDifference == -d[i]) {
                    if (c[j - 1] >= -heightDifference) {
                        continue;
                    }
                } else {
                    return false;
                }
            }
        }

        return true;
    }
}
