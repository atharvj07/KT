n,k=map(int,input().split())
a=[int(x) for x in input().split()]
b=[int(x) for x in input().split()]
f=[-1]*10
for i in range(n//k):a[i]=10**(k-1)+(a[i]-1)*(10**k) - (9-b[i])
f[0]=1
for v in a:
 for i in range(9,0,-1):
  if(f[i]==-1):continue
  f[i]=(f[i]*(10-i))%1000000007
  f[(i+v)%10]=(f[(i+v)%10]+f[i])%1000000007
print(f[0])
