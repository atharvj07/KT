#include <bits/stdc++.h>
 
using namespace std;
 
typedef long long ll;
 
const int N = 1e5 + 5;
 
ll a[N];
 
int32_t main() {
  int h;
  cin >> h;
  for (int i = 0; i <= h; ++i) {
    cin >> a[i];
  }
  if (h <= 2 || a[h] > 1) {
    cout << "ambiguous\n";
    cout << a[0] << ' ' << 1 << ' ';
    for (int i = 2; i <= h; ++i) {
      cout << i << ' ';
    }
    cout << '\n';
    cout << a[0] << ' ' << 0 << ' ';
    for (int i = 2; i <= h; ++i) {
      cout << a[0] + 1 << ' ';
    }
    cout << '\n';
  } else {
    cout << "perfect\n";
  }
  return 0;
}
