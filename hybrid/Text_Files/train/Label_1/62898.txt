import sys
input = sys.stdin.readline

def solve():
    n, k = map(int, input().split())
    h = list(map(int, input().split()))
    flag = True
    for i in range(1, n):
        if h[i] > h[i - 1] + k - 1:
            flag = False
            break
        elif h[i] < h[i - 1]:
            h[i] = h[i - 1]
    if not flag:
        print("NO")
        return
    for i in range(n - 1, 0, -1):
        if h[i] < h[i - 1]:
            h[i] = h[i - 1]
    print("YES")

t = int(input())
for _ in range(t):
    solve()
