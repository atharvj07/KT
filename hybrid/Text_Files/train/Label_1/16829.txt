#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, d, m;
    cin >> n >> d >> m;
    long long a[n];
    for (int i = 0; i < n; ++i) cin >> a[i];
    sort(a, a + n);
    reverse(a, a + n);
    long long ans = 0;
    for (int p = 0; p < n;) {
        long long sum = 0;
        for (int i = p + d; i < n && i < p + d + d; ++i) sum += a[i];
        if (a[p] > m) {
            p += d;
        } else {
            ans += a[p];
            ans += sum;
            p += d + d;
        }
    }
    cout << ans << endl;
    return 0;
}
