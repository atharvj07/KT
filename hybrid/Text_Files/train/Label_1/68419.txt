import sys
input = sys.stdin.readline

def fun():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        sa, sv = 0, 0
        for i in range(1, n-1):
            if a[i]>a[i-1] and a[i]>a[i+1]:
                sa += 1
            elif a[i]<a[i-1] and a[i]<a[i+1]:
                sv += 1
        ans = min(sa, sv)
        print(ans)

fun()
