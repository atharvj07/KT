import java.util.Scanner;

public class Main {

        Scanner sc = new Scanner(System.in);
        int a1 = sc.nextInt(), b1 = sc.nextInt();
        int a2 = sc.nextInt(), b2 = sc.nextInt();
        int a3 = sc.nextInt(), b3 = sc.nextInt();
