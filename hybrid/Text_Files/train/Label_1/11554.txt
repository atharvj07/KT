#include<bits/stdc++.h>
#define mp make_pair 
using namespace std;
const int N = 3e5 + 5;
bool vis[N];
int sz[N], par[N];
int find_par(int x) {
	return par[x] == x ? x : par[x] = find_par(par[x]);
}
void join(int a, int b) {
	a = find_par(a), b = find_par(b);
	if (a == b)return;
	if (sz[a] < sz[b])swap(a, b);
	par[b] = a; sz[a] += sz[b];
}
int main() {
	int n, m, k; cin >> n >> m >> k;
	for (int i = 1; i <= n; i++) {
		vis[i] = 1;
		sz[i] = 1;
		par[i] = i;
	}
	for (int i = 1; i <= m; i++) {
		int h; cin >> h;
		vis[h] = 0;
	}
	set<pair<int, int>>s;
	for (int i = 0; i < k; i++) {
		int u, v; cin >> u >> v;
		if (!vis[u] || !vis[v])continue;
		if (s.count(mp(u, v)))continue;
		if (s.count(mp(v, u)))s.erase(mp(v, u));
		else s.insert(mp(u, v));
	}
	for (auto p : s) {
		int x = find_par(p.first);
		int y = find_par(p.second);
		if (x != y)join(x, y);
	}
	int ans = -1;
	for (int i = 1; i <= n; i++) {
		if (!vis[i])continue;
		int x = find_par(i);
		if (ans == -1)ans = x;
		else if (ans != x)ans = -1;
	}
	cout << ans << '\n';
}
