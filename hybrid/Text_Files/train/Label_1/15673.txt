#include <iostream>
#include <vector>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<int> buttons(m);
  int light_status[101] = {};
  for (int i = 0; i < m; i++) {
    std::cin >> buttons[i];
    for (int j = buttons[i]; j <= n; j++) {
      if (light_status[j] == 0) {
        light_status[j] = i + 1;
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    std::cout << light_status[i] << " ";
  }
  std::cout << std::endl;
  return 0;
}
