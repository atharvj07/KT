import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        if (k >= n - 1 && k <= 2 * n - 2 && (k - n + 1) % 2 == 0) {
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = n - i;
            }
            int cnt = 0;
            for (int i = 0; i < n - 1; i++) {
                if (a[i] > a[i + 1]) {
                    cnt++;
                }
            }
            if (cnt == k) {
                for (int i : a) {
                    System.out.print(i + " ");
                }
            } else {
                System.out.println(-1);
            }
        } else {
            System.out.println(-1);
        }
    }
}
