n = int(input())
p = list(map(int, input().split()))
ans = []
for i in range(n-1, -1, -1):
    if p[i] != i+1:
        for j in range(i+1, n):
            if p[j] == i+1:
                ans.append((j+1, i+1))
                p[j] = p[i]
                p[i] = i+1
                break
len_ans = len(ans)
print(len_ans)
for x in ans:
    print(*x)
