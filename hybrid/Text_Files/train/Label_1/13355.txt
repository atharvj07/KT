#include <bits/stdc++.h>
using namespace std;

void solve() {
  int n, m, k, last, j, x;
  cin >> n >> m;
  vector<int> a(n), b(m);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < m; i++) {
    cin >> b[i];
  }
  for (k = 1; k <= min(n, m); k++) {
    last = 1 << 30;
    j = 0;
    for (int i = 0; i < n; i++) {
      while (j < m && b[j] < a[i]) j++;
      x = a[i];
      if (j < m && x == b[j]) {
        if (x < last) {
          last = x;
          if (k == 1 || last != j) {
            cout << "YES" << endl;
            cout << 1 << " " << x << endl;
            return;
          }
        }
        j++;
      }
    }
  }
  if (k == min(n, m) + 1) {
    cout << "NO" << endl;
  }
}

int main() {
  int t;
  cin >> t;
  while (t--) { solve(); }
  return 0;
}
