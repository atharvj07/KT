import java.io.*;
import java.util.*;

public class Main {
  static class Pair {
    int length;
    char character;

    public Pair(int length, char character) {
      this.length = length;
      this.character = character;
    }
  }

  static int solve(int tn, int sn, List<Pair> t, List<Pair> s) {
    int p1 = 0, p2 = 0, ans = 0;
    while (p1 < tn && p2 < sn) {
      Pair block1 = t.get(p1);
      Pair block2 = s.get(p2);
      if (block1.character == block2.character) {
        if (block1.length >= block2.length) {
          p1 += 1;
          p2 += 1;
          ans += block1.length - block2.length + 1;
        } else {
          p2 += 1;
        }
      } else {
        p1 += 1;
        p2 = 0;
      }
    }
    return ans;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] tsn = br.readLine().split(" ");
    int tn = Integer.parseInt(tsn[0]);
    int sn = Integer.parseInt(tsn[1]);
    List<Pair> t = new ArrayList<>();
    String[] blocks = br.readLine().split(" ");
    for (int i = 0; i < tn; i++) {
      String[] block = blocks[i].split("-");
      int length = Integer.parseInt(block[0]);
      char character = block[1].charAt(0);
      t.add(new Pair(length, character));
    }
    List<Pair> s = new ArrayList<>();
    blocks = br.readLine().split(" ");
    for (int i = 0; i < sn; i++) {
      String[] block = blocks[i].split("-");
      int length = Integer.parseInt(block[0]);
      char character = block[1].charAt(0);
      s.add(new Pair(length, character));
    }
    System.out.println(solve(tn, sn, t, s));
  }
}
