import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T1 = sc.nextInt();
        int T2 = sc.nextInt();
        int R1 = sc.nextInt();
        int R2 = sc.nextInt();

        if (R1 != -1 && R2 != -1) {
            if (R1 > R2) {
                System.out.println("Alice");
            } else if (R1 < R2) {
                System.out.println("Bob");
            } else {
                if (T1 < T2) {
                    System.out.println("Alice");
                } else if (T1 > T2) {
                    System.out.println("Bob");
                } else {
                    System.out.println("Draw");
                }
            }
        } else if (R1 != -1) {
            System.out.println("Alice");
        } else if (R2 != -1) {
            System.out.println("Bob");
        } else {
            if (T1 < T2) {
                System.out.println("Alice");
            } else if (T1 > T2) {
                System.out.println("Bob");
            } else {
                System.out.println("Draw");
            }
        }
    }
}
