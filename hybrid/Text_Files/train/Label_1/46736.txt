import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        double p = in.nextDouble();
        int t = in.nextInt();
        double[] dp = new double[t+1];
        for (int i = 1; i <= t; i++)
            dp[i] = (1 - p) * dp[i - 1] + p * (1 + dp[i - 1]);
        System.out.println(dp[t]);
    }
}
