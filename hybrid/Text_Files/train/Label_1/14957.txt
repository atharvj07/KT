#include <algorithm>
#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  pair<int, int> va[n];
  pair<int, int> vb[n];
  for (int i = 0; i < n; i++) {
    int sa_i, ea_i, sb_i, eb_i;
    cin >> sa_i >> ea_i >> sb_i >> eb_i;
    va[i] = {sa_i, ea_i};
    vb[i] = {sb_i, eb_i};
  }
  sort(va, va + n);
  sort(vb, vb + n);
  for (int i = 0; i < n; i++) {
    if (va[i].second < vb[i].first || va[i].first > vb[i].second) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
