from sys import stdin
n, S = map(int, stdin.readline().split())
sl, al, bl = [], [], []
for _ in range(n):
    s, a, b = map(int, stdin.readline().split())
    sl.append(s)
    al.append(a)
    bl.append(b)
sl.sort()
al.sort()
bl.sort()
res = 0
for i in range(n):
    res += max(al[i], bl[i]) * sl[i]
print(res)
