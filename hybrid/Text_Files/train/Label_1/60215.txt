def child_choice():
    choices = [input().split('.', 1)[1] for _ in range(4)]
    lengths = [len(choice) for choice in choices]

    great_choices = []
    for i, length in enumerate(lengths):
        if all(length * 2 <= other for j, other in enumerate(lengths) if i != j) or all(length >= other * 2 for j, other in enumerate(lengths) if i != j):
            great_choices.append(i)
    
    if len(great_choices) == 1:
        return chr(ord('A') + great_choices[0])
    else:
        return 'C'

print(child_choice())
