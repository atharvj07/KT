import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class SupersonicRocket {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        List<Point> engine1 = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            line = reader.readLine().split(" ");
            int x = Integer.parseInt(line[0]);
            int y = Integer.parseInt(line[1]);
            engine1.add(new Point(x, y));
        }

        List<Point> engine2 = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int x = Integer.parseInt(line[0]);
            int y = Integer.parseInt(line[1]);
            engine2.add(new Point(x, y));
        }

        boolean safe = isSafe(engine1, engine2);
        System.out.println(safe ? "YES" : "NO");
    }

    private static boolean isSafe(List<Point> engine1, List<Point> engine2) {
        Set<Point> combinedField = new HashSet<>();
        for (Point p1 : engine1) {
            for (Point p2 : engine2) {
                for (double k = 0; k < 1; k += 0.01) {
                    combinedField.add(new Point((int) (p1.x * k + p2.x * (1 - k)), (int) (p1.y * k + p2.y * (1 - k))));
                }
            }
        }

        for (Point p1 : engine1) {
            Set<Point> newField = new HashSet<>();
            for (Point p2 : engine2) {
                for (double k = 0; k < 1; k += 0.01) {
                    newField.add(new Point((int) (p1.x * k + p2.x * (1 - k)), (int) (p1.y * k + p2.y * (1 - k))));
                }
            }
            if (!combinedField.equals(newField)) {
                return false;
            }
        }

        for (Point p2 : engine2) {
            Set<Point> newField = new HashSet<>();
            for (Point p1 : engine1) {
                for (double k = 0; k < 1; k += 0.01) {
                    newField.add(new Point((int) (p1.x * k + p2.x * (1 - k)), (int) (p1.y * k + p2.y * (1 - k))));
                }
            }
            if (!combinedField.equals(newField)) {
                return false;
            }
        }

        return true;
    }

    private static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
