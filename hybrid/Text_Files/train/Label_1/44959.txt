import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt(), b = sc.nextInt(), ans = 0;
        for(int base = 10; base <= 1000; base++) {
            int res = a + b, len = 0;
            while(res > 0) {
                res /= base;
                len++;
            }
            ans = Math.max(ans, len);
        }
        System.out.println(ans);
    }
}
