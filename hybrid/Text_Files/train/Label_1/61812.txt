def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def min_moves(n, chocolates):
    total_chocolates = sum(chocolates)
    if total_chocolates == 1:
        return -1

    def check(k):
        moves, current_sum, mod_sum = 0, 0, 0
        for chocolate in chocolates:
            current_sum += chocolate
            mod_sum += current_sum % k
            moves += min(mod_sum, k - mod_sum)
        return moves

    answer = float('inf')
    for i in range(2, int(total_chocolates**0.5) + 1):
        if total_chocolates % i == 0:
            answer = min(answer, check(i))
            while total_chocolates % i == 0:
                total_chocolates //= i
    if total_chocolates > 1:
        answer = min(answer, check(total_chocolates))

    return answer if answer != float('inf') else -1

n = int(input().strip())
chocolates = list(map(int, input().strip().split()))

print(min_moves(n, chocolates))
