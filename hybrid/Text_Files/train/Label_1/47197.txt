import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt();
        int[] unread = new int[n];
        for (int i = 0; i < q; i++) {
            int type = sc.nextInt();
            int x = sc.nextInt();
            if (type == 1) {
                unread[x - 1]++;
            } else if (type == 2) {
                unread[x - 1] = 0;
            } else {
                for (int j = 0; j < x; j++) {
                    unread[j] = 0;
                }
            }
            int count = 0;
            for (int j = 0; j < n; j++) {
                count += unread[j];
            }
            System.out.println(count);
        }
    }
}
