#include <iostream>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, a, b, c, d;
    std::cin >> n >> a >> b >> c >> d;
    bool flag = true;
    int minX = a - b;
    if (n * minX > c + d) flag = false;
    int maxX = a + b;
    if (n * maxX < c - d) flag = false;
    if (flag) std::cout << "Yes" << std::endl;
    else std::cout << "No" << std::endl;
  }
  return 0;
}
