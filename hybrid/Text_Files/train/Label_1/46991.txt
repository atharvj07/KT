import java.util.Scanner;

public class BoxPacking {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] objects = new int[n];
        for (int i = 0; i < n; i++) {
            objects[i] = scanner.nextInt();
        }

        int maxObjects = 0;
        int currentBoxSize = k;
        int currentObjectIndex = 0;

        while (currentObjectIndex < n && currentBoxSize >= objects[currentObjectIndex]) {
            maxObjects++;
            currentBoxSize -= objects[currentObjectIndex];
            currentObjectIndex++;

            if (currentBoxSize == 0 && currentObjectIndex < n) {
                m--;
                currentBoxSize = k;
            }
        }

        System.out.println(maxObjects);
    }
}
