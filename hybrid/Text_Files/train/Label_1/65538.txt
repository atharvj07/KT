from sys import stdin

input = stdin.readline

n = int(input())
arr = list(map(int, input().split()))
s = [input().strip() for i in range(n)]
a = sorted(s)
sa = sorted(s, key=lambda x: x[::-1])
ans = 0
for i in range(n):
    if a[i] != s[i]: ans += arr[i]
    if sa[i] != s[i]: ans += arr[i]
    if a[i] != sa[i]: ans = -1; break
if ans == -1: print(ans)
else: print(ans // 2)
