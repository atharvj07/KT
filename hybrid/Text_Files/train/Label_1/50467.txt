import java.util.*;

public class DynamicArray {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        ArrayList<Integer>[] arr = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            arr[i] = new ArrayList<>();
        }
        for (int i = 0; i < q; i++) {
            int type = sc.nextInt();
            int t = sc.nextInt();
            if (type == 0) {
                int x = sc.nextInt();
                arr[t].add(x);
            } else if (type == 1) {
                if (arr[t].isEmpty()) {
                    System.out.println();
                } else {
                    for (int j = 0; j < arr[t].size(); j++) {
                        System.out.print(arr[t].get(j) + " ");
                    }
                    System.out.println();
                }
            } else if (type == 2) {
                arr[t].clear();
            }
        }
    }
}
