#include <bits/stdc++.h>
using namespace std;
using ll=long long;
int main() {
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int n; cin >> n;
    vector<int> a(n);
    for (auto &i: a) cin >> i;
    map<int, int> m;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (m.count(a[i])) ans = max(ans, i - m[a[i]]);
        m[a[i]] = i;
    }
    cout << ans << '\n';
}
