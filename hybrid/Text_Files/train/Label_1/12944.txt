#include<bits/stdc++.h>
using namespace std;
#define ll long long
void u(){cout<<"NO"<<endl;exit(0);}
int main(){
  ll n,m,k;cin>>n>>m>>k;
  if(2ll*n*m-2ll*n-2ll*m<k)u();
  cout<<"YES"<<endl;
  ll a=k,b=(4ll*n*m-2ll*n-2ll*m)/4ll;
  if(a%b)u();
  ll t=a/b;
  cout<<t<<endl;
  for(ll i=0;i<t;i++){
    cout<<b<<" R"<<endl;
    cout<<b<<" D"<<endl;
    cout<<(b-1)<<" L"<<endl;
    cout<<(b+1)<<" R"<<endl;
  }
}
