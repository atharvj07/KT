n, q = map(int, input().split())
arr = list(map(int, input().split()))
queries = [list(map(int, input().split())) for _ in range(q)]

# Count the frequency of each index
freq = [0] * (n + 1)
for l, r in queries:
    freq[l-1] += 1
    freq[r] -= 1

# Calculate prefix sum of frequencies to get actual frequencies
for i in range(1, n):
    freq[i] += freq[i-1]

# Sort the array and frequencies in non-ascending order
arr.sort(reverse=True)
freq.sort(reverse=True)

# Calculate the maximum sum
max_sum = 0
for i in range(n):
    max_sum += arr[i] * freq[i]

print(max_sum)
