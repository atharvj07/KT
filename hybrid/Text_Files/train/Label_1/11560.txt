#include <bits/stdc++.h>
#define ll long long
#define ld long double
#define vi vector<int>
#define pi pair<int, int>
using namespace std;
const int mod = 1e9 + 7;
bool is_sqr(int x){
    int i = sqrt(x);
    return i * i == x;
}
int main(){
    int n, x;
    cin>>n;
    vi a(n);
    for(auto &i : a) cin>>i;
    vector<bool> L(n), R(n);
    L[0] = is_sqr(a[0]);
    R[n - 1] = is_sqr(a[n - 1]);
    for(int i = 1; i < n; i++){
        L[i] = L[i - 1] && is_sqr(a[i - 1] * a[i]);
    }
    for(int i = n - 2; i >= 0; i--){
        R[i] = R[i + 1] && is_sqr(a[i + 1] * a[i]);
    }
    vector<vector<int>> dp(n, vector<int>(n, 0));
    for(int i = 0; i < n; i++){
        for(int j = i; j < n; j++){
            if(! (L[i] && R[j]) ) continue;
            if(i == j) dp[i][j] = 1;
            else if(i + 1 == j){
                int l = a[i] * a[j];
                dp[i][j] = l <= 0? 1 : 0;
            }else{
                for(int k = i; k < j; k++){
                    dp[i][j] += dp[i][k] + dp[k + 1][j];
                    dp[i][j] %= mod;
                }
            }
        }
    }
    for(int k = 1; k <= n; k++){
        int ans = 0;
        for(int i = 0; i < n; i++){
            for(int j = i; j < n; j++){
                ans += dp[i][j] == k;
                ans %= mod;
            }
        }
        cout<<ans<<" ";
    }
    return 0;
}
