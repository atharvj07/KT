def main():
    n, m = [int(x) for x in input().split()]
    horiz = input()
    vert = input()
    graph = [[0 for _ in range(m)] for _ in range(n)]
    for i, d in enumerate(horiz):
        if d == '>':
            for j in range(m):
                graph[i][j] |= (1 << (j + 1))
        else:
            for j in range(m - 1, -1, -1):
                graph[i][j] |= (1 << j)
    for j, d in enumerate(vert):
        if d == 'v':
            for i in range(n):
                graph[i][j] |= (1 << (i + 1))
        else:
            for i in range(n - 1, -1, -1):
                graph[i][j] |= (1 << i)

    visited = [[False for _ in range(m)] for _ in range(n)]
    q = [(0, 0)]
    while q:
        i, j = q.pop(0)
        if visited[i][j]:
            continue
        visited[i][j] = True
        if i > 0 and graph[i - 1][j] & (1 << j):
            q.append((i - 1, j))
        if i < n - 1 and graph[i][j] & (1 << (j + 1)):
            q.append((i + 1, j))
        if j > 0 and graph[i][j - 1] & (1 << i):
            q.append((i, j - 1))
        if j < m - 1 and graph[i][j] & (1 << (i + 1)):
            q.append((i, j + 1))

    for i in range(n):
        for j in range(m):
            if not visited[i][j]:
                print("NO")
                return
    print("YES")

main()
