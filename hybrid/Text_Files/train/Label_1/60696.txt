from itertools import permutations

s = input()
p, m = map(int, input().split())

op = '? + -'[:p + m]
v = [int(c) for c in s if c.isdigit()]
res = 0
for ops in permutations(op):
    val = v[0]
    for i in range(len(v) - 1):
        if ops[i] == '+':
            val += v[i + 1]
        else:
            val -= v[i + 1]
    res = max(res, val)
print(res)
