#include <bits/stdc++.h>
#define pb push_back
#define all(c) c.begin(),c.end()
using namespace std;
map<char,char>m={{'[',']'},{'(',')'},{'<','>'},{'{','}'}};
int main(){
	int bal[5]={0};
	int n,l=0;
	string s;
	cin>>s,n=s.size();
	for(char x:s){
		if(x=='['||x=='{'||x=='('||x=='<')bal[0]++;
		else if(bal[0]){
			if(x==m[s[l]])bal[0]--,l++;
			else bal[x=='<'?1:(x=='{'?2:(x=='('?3:4))]=1;
		}
		else bal[x=='<'?1:(x=='{'?2:(x=='('?3:4))]=1;
	}
	for(int x:bal)if(x)n+=x-1;
	cout<<(n<=0?"0":"Impossible");
	return 0;
}
