def count_regions(n, circles):
    # Count of intersections
    intersections = 0
    for i in range(n):
        for j in range(i+1, n):
            d = ((circles[i][0] - circles[j][0])**2 + (circles[i][1] - circles[j][1])**2)**0.5
            if d < circles[i][2] + circles[j][2] and d > abs(circles[i][2] - circles[j][2]):
                intersections += 2
    # Formula to calculate regions
    regions = 1 + n + intersections
    return regions

n = int(input())
circles = [list(map(int, input().split())) for _ in range(n)]
print(count_regions(n, circles))
