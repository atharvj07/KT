#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  string s, t;
  cin >> s >> t;
  vector<int> dp(m + 1);
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = m; j >= 0; --j) {
      if (s[i] == t[j]) {
        dp[j + 1] = max(dp[j + 1], dp[j] + 1);
        if (dp[j + 1] > ans && (j + 1) % k == 0) ans = dp[j + 1];
      }
    }
  }
  cout << ans * k << endl;
  return 0;
}
