const int N = 2e5 + 5;
vector<int> a(N);

int solve(int n) {
    int ans = 0, i = 1, j = 1;
    while (j <= n) {
        while (a[j] == i) i++, j++;
        if (j > n) break;
        ans++, j++;
    }
    return ans;
}
