#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
    int n, m;
    cin >> n >> m;
    int sx, sy, fx, fy;
    cin >> sx >> sy >> fx >> fy;
    int x[m], y[m];
    for (int i = 0; i < m; i++)
        cin >> x[i] >> y[i];
    int ans = abs(sx - fx) + abs(sy - fy);
    for (int i = 0; i < m; i++) {
        if (sx == x[i])
            ans = min(ans, abs(sy - y[i]) + abs(fx - x[i]));
        if (sy == y[i])
            ans = min(ans, abs(sx - x[i]) + abs(fy - y[i]));
    }
    cout << ans << endl;
    return 0;
}
