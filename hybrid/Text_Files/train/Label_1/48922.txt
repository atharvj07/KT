import java.util.Scanner;

public class UniversityTime {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        scanner.nextLine();

        int[][] timetable = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.nextLine();
            for (int j = 0; j < m; j++) {
                timetable[i][j] = line.charAt(j) - '0';
            }
        }

        int minHours = Integer.MAX_VALUE;
        for (int skippedLessons = 0; skippedLessons <= k; skippedLessons++) {
            int[][] skippedLessonsMask = new int[n][m];
            int hoursSpent = 0;
            for (int i = 0; i < n; i++) {
                int firstLesson = -1;
                int lastLesson = -1;
                for (int j = 0; j < m; j++) {
                    if (timetable[i][j] == 1) {
                        if (firstLesson == -1) {
                            firstLesson = j;
                        }
                        lastLesson = j;
                    }
                }

                if (firstLesson != -1 && lastLesson != -1) {
                    int lessonsToSkip = Math.min(skippedLessons, lastLesson - firstLesson + 1);
                    skippedLessons -= lessonsToSkip;
                    hoursSpent += lastLesson - firstLesson + 1 - lessonsToSkip;
                    for (int j = firstLesson; j <= lastLesson; j++) {
                        if (skippedLessonsMask[i][j] == 0) {
                            skippedLessonsMask[i][j] = 1;
                            lessonsToSkip--;
                        }
                    }
                }
            }

            minHours = Math.min(minHours, hoursSpent);
        }

        System.out.println(minHours);
    }
}
