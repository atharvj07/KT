from collections import Counter
n,m=map(int,input().split())
a=[input()for i in range(n)]
b=[a[int(i)-1]for i in input().split()]
for i in range(n):
    for j in range(max(map(len,a))):
        d=Counter()
        for k in range(n):
            if j<len(a[k]):
                if k==i:d['*']+=1
                elif a[i][j]!=a[k][j]:d['?']+=1
                elif a[i][j]=='.':d['.']+=1
                else:break
        if all(v<=1 for v in d.values()):
            t=[a[i][:j]+'*'*(max(map(len,a))-j)for j in range(max(map(len,a)))]
            if all(t[k]==b[k]for k in range(n)):
                print('Yes');print(*t);exit()
print('No')
