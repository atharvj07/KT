n, m = map(int, input().split())
items = []
for _ in range(n):
    items.append(tuple(map(int, input().split())))

items.sort(key=lambda x: x[1]/x[0], reverse=True)

ans = 0
for weight, cost in items:
    if m >= weight:
        m -= weight
        ans += cost
    else:
        ans += m * cost / weight
        break

print(ans)
