import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class EducationSystemReform {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        Subject[] subjects = new Subject[m];
        for (int i = 0; i < m; i++) {
            int ai = scanner.nextInt();
            int bi = scanner.nextInt();
            int ci = scanner.nextInt();
            subjects[i] = new Subject(ai, bi, ci);
        }

        Arrays.sort(subjects, Comparator.comparingInt(Subject::getComplexity));

        boolean isValid = true;
        long totalExercises = 0;
        for (int i = 0; i < n; i++) {
            Subject subject = subjects[i];
            if (subject.getComplexity() > i + 1) {
                isValid = false;
                break;
            }
            long exercises = subject.getExercises(k, i);
            totalExercises += exercises;
        }

        if (!isValid || totalExercises > Math.pow(10, 16)) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                Subject subject = subjects[i];
                long exercises = subject.getExercises(k, i);
                System.out.println(subject.getIndex() + 1 + " " + exercises);
            }
        }
    }

    private static class Subject {
        private final int ai;
        private final int bi;
        private final int ci;

        public Subject(int ai, int bi, int ci) {
            this.ai = ai;
            this.bi = bi;
            this.ci = ci;
        }

        public int getComplexity() {
            return ci;
        }

        public int getIndex() {
            return ai;
        }

        public long getExercises(int k, int day) {
            if (day == 0) {
                return ai;
            } else if (ai == bi) {
                return k * getExercises(k, day - 1);
            } else {
                return k + getExercises(k, day - 1);
            }
        }
    }
}
