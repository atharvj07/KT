import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] years = new int[2 * n];
        for (int i = 0; i < n; i++) {
            years[2 * i] = input.nextInt();
            years[2 * i + 1] = input.nextInt();
        }
        Arrays.sort(years);

        int maxYear = 0;
        int maxCount = 0;
        int currentYear = 0;
        int currentCount = 0;
        for (int year : years) {
            if (year == currentYear) {
                currentCount++;
            } else {
                currentYear = year;
                currentCount = 1;
            }
            if (currentCount > maxCount) {
                maxYear = currentYear;
                maxCount = currentCount;
            }
        }
        System.out.println(maxYear + " " + maxCount);
    }
}
