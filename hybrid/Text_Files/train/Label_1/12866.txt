#include <bits/stdc++.h>
using namespace std;

struct point{
	int x;
	int y;
};

struct edge{
	int x;
	int y;
	int w;
};

const int M = 2e5+5;
const int INF = 1e9+7;

int main() {
	int t;
	cin >> t;
	while(t--){
		int n, k;
		cin >> n >> k;
		vector<vector<pair<int,int>>> adj(n+1,vector<pair<int,int>>());
		vector<int> friends(k+1);
		for(int i=1; i<=k; i++){
			cin >> friends[i];
		}
		for(int i=1; i<n; i++){
			int x, y;
			cin >> x >> y;
			adj[x].push_back({y,0});
			adj[y].push_back({x,0});
		}
		queue<point> q;
		vector<int> dis(n+1,INF),par(n+1,-1);
		q.push({1,0});
		dis[1] = 0;
		par[1] = -1;
		while(!q.empty()){
			point u = q.front();
			q.pop();
			for(auto p : adj[u.x]){
				int v = p.first;
				int w = p.second;
				if(dis[v] > dis[u.x] + w){
					dis[v] = dis[u.x] + w;
					q.push({v, dis[v]});
					par[v] = u.x;
				}
			}
		}
		bool flag = true;
		for(int i=1; i<=k; i++){
			if(dis[friends[i]] == 1){
				flag = false;
				break;
			}
		}
		if(flag) cout << "YES" << endl;
		else cout << "NO" << endl;
	}
	return 0;
}
