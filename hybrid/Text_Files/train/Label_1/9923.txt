#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007
#define vec vector
#define pb push_back
const ll N = 2e5+6;
ll n,m,q;
vector<pair<ll,ll>>e[N];
bitset<N>vis;
void dfs(ll x)
{
    vis[x]=1;
    for(auto nbr:e[x])
    if(!vis[nbr.first]) dfs(nbr.first);
}
int main()
{
    cin>>n>>m>>q;
    while(q--)
    {
        ll x,y;
        cin>>x>>y;
        e[m*(x-1)+y].push_back({m*(x-1)+y+1,1});
        e[m*(x-1)+y+1].push_back({m*(x-1)+y,1});
        e[m*(x-1)+y].push_back({m*x+y,1});
        e[m*x+y].push_back({m*(x-1)+y,1});
    }
    ll cnt=0;
    for(ll i=1;i<=n*m;++i)
    if(!vis[i])
    {
        dfs(i);
        cnt++;
    }
    cout<<cnt<<endl;
}
