#include<bits/stdc++.h>
typedef long long ll;
const int N=2e5+11,K=30;
int n,deg[N],xorseg[N],xorseg2[N];
struct edge{int to,xorn;} edges[N];
ll num_edges;
int main(){
	scanf("%d",&n);
	for(int i=0;i<n;++i){
		scanf("%d%d",&deg[i],&xorseg[i]);
		if(!deg[i]) xorseg2[i]=xorseg[i];
	}
	for(int i=0;i<n;++i){
		if(!deg[i]) continue;
		int xorn=xorseg[i];
		for(int j=i+1;j<n;++j){
			if(!deg[j]) continue;
			xorn^=xorseg[j];
			edges[num_edges++]={j,xorn};
			edges[num_edges++]={i,xorn};
			xorseg[i]^=xorseg2[j]; xorseg[j]^=xorseg2[i];
			--deg[i],--deg[j];
			if(!deg[i] && !xorseg[i]) xorseg2[i]=xorseg[i];
			if(!deg[j] && !xorseg[j]) xorseg2[j]=xorseg[j];
		}
	}
	printf("%lld\n",num_edges);
	for(int i=0;i<num_edges;++i) printf("%d %d\n",edges[i].to,edges[i].xorn);
	return 0;
}
