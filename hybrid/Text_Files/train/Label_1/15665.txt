#include <ext/pb_ds/assoc_container.hpp>
#include <bits/stdc++.h>
using namespace std;
using namespace __gnu_pbds;

typedef tree<int, null_type, less<int>, rb_tree_tag, tree_order_statistics_node_update> ordered_set;

const int inf = 1 << 30, MOD = 1e9 + 7;

int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    int n;
    cin >> n;
    vector<int> p(n);
    for (int &x : p) cin >> x;

    sort(p.begin(), p.end());

    int ans = 1;
    int cur = p[0];
    vector<int> v = {cur};
    for (int i = 1; i < n; i++)
    {
        if ((p[i] - cur)&(p[i] - cur - 1) == 0)
        {
            ans++;
            cur = p[i];
            v.push_back(p[i]);
        }
    }

    cout << ans << endl;
    for (int x : v) cout << x << ' ';
    cout << endl;
}
