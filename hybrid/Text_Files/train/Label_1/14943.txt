#include <bits/stdc++.h>

using namespace std;

struct query { int l, r, d; };

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (auto &x : a) cin >> x;
  vector<query> q(m);
  for (auto &x : q) cin >> x.l >> x.r >> x.d;
  vector<query> qq(k);
  for (auto &x : qq) cin >> x.l >> x.r;
  vector<int> add(m + 1);
  for (int i = 0; i < k; i++) {
    add[qq[i].l] += q[qq[i].l - 1].d;
    add[qq[i].r + 1] -= q[qq[i].r].d;
  }
  for (int i = 1; i < m + 1; i++) add[i] += add[i - 1];
  for (int i = 0; i < n; i++) a[i] += add[i + 1];
  for (auto x : a) cout << x << ' ';
  return 0;
}
