from bisect import bisect_right as cbr

n1, n2 = map(int,input().split())
b1 = sorted(map(int,input().split()))
b2 = sorted(map(int,input().split()))
i = 0
ans = 0
sum1 = sum2 = 0
for a in b1:
    sum1+=a
    while i < len(b2) and sum2 + b2[i] <= sum1:
        sum2+=b2[i]
        i+=1
    if sum1 == sum2: ans+=1
print(ans)
