#include<bits/stdc++.h>
using namespace std;
typedef pair<double,double> pi;
#define ff first
#define ss second
int main(){
    int n;
    cin>>n;
    vector<pi>v(n);
    for(int i=0;i<n;i++){
        double a,b;
        cin>>a>>b;
        v[i]={a,b};
    }
    sort(v.rbegin(),v.rend());
    double ans=v[0].ff;
    for(int i=1;i<n;i++){
        ans+=v[i].ff*(1-ans);
    }
    cout<<fixed<<setprecision(12)<<ans<<endl;
}
