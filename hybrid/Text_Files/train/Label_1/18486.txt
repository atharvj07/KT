#include <vector>
#include <cstdio>
#include <algorithm>
using namespace std;
struct node { vector<int> son; int type, val; };
vector<node> t;
int ans, a[300001]; bool vis[300001];
void dfs(int p)
{
    vis[p]=true;
    t[p].val=a[p];
    for(int i=0;i<t[p].son.size();i++)
    {
        if(vis[t[p].son[i]]) continue;
        dfs(t[p].son[i]);
        if(t[p].type==0) t[p].val=min(t[p].val,t[t[p].son[i]].val);
        else t[p].val=max(t[p].val,t[t[p].son[i]].val);
    }
    ans=max(ans,t[p].val);
}
int main()
{
    int n;
    scanf("%d",&n);
    t.resize(n+1);
    for(int i=1;i<=n;i++) scanf("%d",&t[i].type);
    for(int i=2;i<=n;i++) { int p; scanf("%d",&p); t[p].son.push_back(i); }
    int k=0;
    for(int i=1;i<=n;i++) if(t[i].son.size()==0) { a[++k]=i; vis[i]=true; }
    sort(a+1,a+1+k);
    dfs(1);
    printf("%d\n",ans);
    return 0;
}
