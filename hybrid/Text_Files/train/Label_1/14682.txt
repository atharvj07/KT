#define pb push_back
#define ll long long

using namespace std;
using namespace __gnu_pbds;
typedef tree<int, null_type, less_equal<int>, rb_tree_tag, tree_order_statistics_node_update> ordered_set;
const int M = 1e5 + 5, P = 1e9 + 7, N = 1e5 + 5;
int v[M], p;
int main()
{
    int n, m;
    cin >> n >> p >> m;
    priority_queue<int> pq;
    while (n--)
    {
        int day, money;
        cin >> day >> money;
        pq.push(day - money);
    }

    int ans = 0;
    while (pq.size())
    {

        int days = pq.top();
        pq.pop();
        if (days <= m)
        {
            ans++;
            days += p;
            pq.push(days);
        }
    }
    cout << ans << '\n';
    return 0;
}
