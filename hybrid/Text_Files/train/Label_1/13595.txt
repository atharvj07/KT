#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

void solve() {
  ll x, y, l, r;
  cin >> x >> y >> l >> r;
  ll ans = r - l + 1;
  for (ll a = 0; a <= 60; a++) {
    for (ll b = 0; b <= 60; b++) {
      ll sum = x * x * a + y * y * b;
      if (sum >= l && sum <= r) ans = min(ans, sum - l);
    }
  }
  cout << ans << endl;
}

int main() {
  solve();
  return 0;
}
