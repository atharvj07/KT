from collections import defaultdict

n = int(input())
seg = []
for _ in range(n):
    a, b, c, d = map(int, input().split())
    seg.append([(a, b), (c, d)])

pt = defaultdict(lambda: 0)
for x1, y1 in seg[0]:
    for x2, y2 in seg:
        if x1 != x2: pt[(x2, y1)] += 1
        if y1 != y2: pt[(x1, y2)] += 1
for x, y in seg: pt[(x, y)] += 1

print(len(pt))
