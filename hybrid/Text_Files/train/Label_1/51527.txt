import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            char[] s = sc.next().toCharArray();
            int[] state = new int[n];
            for (int i = 0; i < n; i++) {
                if (s[i] == '<') {
                    state[i] = -1;
                } else if (s[i] == '>') {
                    state[i] = 1;
                }
            }
            int[] pre = new int[n];
            int[] suf = new int[n];
            for (int i = 0; i < n; i++) {
                if (i == 0) {
                    pre[i] = state[i];
                } else {
                    pre[i] = pre[i - 1] + state[i];
                }
                if (i == n - 1) {
                    suf[i] = state[i];
                } else {
                    suf[i] = suf[i + 1] + state[i];
                }
            }
            int ans = 0;
            for (int i = 0; i < n; i++) {
                if (pre[i] == 0 && suf[i] == 0) {
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
