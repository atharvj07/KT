#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, t;
    cin >> n >> t;
    vector<pair<int, int>> buses(n);
    for (auto &bus : buses) cin >> bus.first >> bus.second;
    int ans = -1;
    for (int i = 0; i < n; i++) {
        int cur = buses[i].first;
        while (cur <= t) {
            if (cur == t) ans = i + 1;
            cur += buses[i].second;
        }
    }
    cout << ans << endl;
}
