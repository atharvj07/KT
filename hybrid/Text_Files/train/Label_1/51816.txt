import java.util.*;

public class Main {
    public static void main(String[] args) {
        try (Scanner sc = new Scanner(System.in)) {
            int n = sc.nextInt();
            int[] b = new int[n];
            for (int i = 0; i < n; i++) b[i] = sc.nextInt();
            int[] dp = new int[n];
            dp[0] = b[0];
            int ans = b[0];
            for (int i = 1; i < n; i++) {
                dp[i] = b[i];
                for (int j = 0; j < i; j++) {
                    if (b[i] - b[j] == i - j) {
                        dp[i] = Math.max(dp[i], dp[j] + b[i]);
                    }
                }
                ans = Math.max(ans, dp[i]);
            }
            System.out.print(ans);
        }
    }
}
