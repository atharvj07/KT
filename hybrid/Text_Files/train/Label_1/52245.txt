import java.util.*;
import java.lang.*;

class Main{
    static int n, m, s1[][], s2[][], bar[];

    public static void main(String[] ags){
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        s1 = new int[n][n];
        s2 = new int[n][n];
        bar = new int[n];
        for(int i=0; i<n; i++){
            String str = in.next();
            for(int j=0; j<n; j++) s1[i][j] = str.charAt(j)-'0';
            str = in.next();
            for(int j=0; j<n; j++) s2[i][j] = str.charAt(j)-'0';
        }
        String str = in.next();
        for(int i=0; i<n; i++) bar[i] = str.charAt(i)-'0';
        if(check()){
            System.out.println(-1); return;
        }
        m = 0;
        dfs(0);
        System.out.println(m);
        for(int i=0; i<m; i++) System.out.println(dir[i]);
    }

    static boolean check(){
        int b=0, w=0;
        for(int i=0; i<n; i++){
            for(int j=0; j<n; j++){
                if(s1[i][j]==bar[j] && s1[i][j]==s2[i][j]) return true;
                if(s1[i][j]==bar[j]) w++; else b++;
                if(s2[i][j]==bar[j]) w--; else b++;
            }
            if(w<0 || b<0) return true;
            w=b=0;
        }
        for(int j=0; j<n; j++){
            for(int i=0; i<n; i++){
                if(s1[i][j]==bar[i] && s1[i][j]==s2[i][j]) return true;
                if(s1[i][j]==bar[i]) w++; else b++;
                if(s2[i][j]==bar[i]) w--; else b++;
            }
            if(w<0 || b<0) return true;
            w=b=0;
        }
        return false;
    }

    static void dfs(int x){
        if(x==n*n){
            return;
        }
        int i=x/n,j=x%n;
        for(int k=0; k<n; k++){
            if(s1[i][j]==s2[i][j]) continue;
            if(s1[i][j]==bar[k]){
                s1[i][j]=s2[i][j];
                dir[m++]="row "+i;
                dfs(x+1);
                m--;
                s1[i][j]=1-s2[i][j];
            }
            else{
                s1[i][j]=s2[i][j];
                dir[m++]="col "+j;
                dfs(x+1);
                m--;
                s1[i][j]=1-s2[i][j];
            }
        }
    }

    static String[] dir = new String[100005];
}
