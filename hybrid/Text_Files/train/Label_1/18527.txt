#include <cstdio>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int T;
    scanf("%d", &T);
    while (T--) {
        int n;
        scanf("%d", &n);
        vector<int> a(n);
        for (int &x: a)
            scanf("%d", &x);
        sort(a.begin(), a.end());
        int mx = 0;
        for (int i = 0, j = n - 1; i < j;) {
            if (a[i] == a[j])
                mx++, i++, j--;
            else
                j--;
        }
        printf("%d\n", mx);
    }
    return 0;
}
