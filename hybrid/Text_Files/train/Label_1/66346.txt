from sys import stdin 

n = int(stdin.readline())
a = [[int(x) for x in stdin.readline().split()] for i in range(n)]
a.sort(key=lambda x: (x[1],-x[2]))
ans = 0
for m,s in a[1:]:
    if s!=a[0][1]:
        ans+=(s-a[0][1])
        for i in a[0][2:]:
            i+=s-a[0][1]
    a[0][1]=s
    a[0][2].extend(a[0][2:a[0][2].index(a[0][1]-1):-1])
    del a[0][2:a[0][2].index(a[0][1]-1):-1]
    a[0][2]+=a[1][2:]
    del a[1]
print(ans)
