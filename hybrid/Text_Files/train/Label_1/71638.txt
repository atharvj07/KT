t = int(input())
for _ in range(t):
    s = input()
    s = s.replace('?','a')
    for i in range(1,len(s)):
        if s[i] == s[i-1]:
            s = s.replace('?','b',1)
            if s[i] == s[i-1]:
                s = s.replace('?','c',1)
                break
    if s[-1] == s[-2]: s = '-1'
    print(s)
