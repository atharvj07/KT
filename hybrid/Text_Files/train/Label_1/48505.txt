import java.util.*;
public class Main {
    static int n, m;
    static List<Integer>[] friend;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        friend = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) friend[i] = new ArrayList<>();
        for (int i = 1; i <= m; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            friend[a].add(b);
            friend[b].add(a);
        }
        int ans = 2 * n;
        int[] path = new int[n + 1];
        int[] cnt = new int[n + 1];
        for (int i = n; i >= 1; i--) {
            cnt[i] = 1;
            for (int j : friend[i]) {
                if (j < i && cnt[j] == 0) {
                    boolean have = false;
                    for (int k : friend[j]) {
                        if (k > i && cnt[k] > 0) {
                            path[i] = j;
                            cnt[i] = cnt[k] + 1;
                            have = true;
                            break;
                        }
                    }
                    if (!have) {
                        for (int k : friend[j]) {
                            if (k > j) {
                                path[i] = k;
                                cnt[i] = cnt[k] + 1;
                                break;
                            }
                        }
                    }
                }
            }
            ans = Math.min(ans, cnt[i]);
        }
        System.out.println(ans - 1);
        int now = 1;
        for (int i = 2; i <= ans; i++) {
            System.out.print(path[now] + " ");
            now = path[now];
        }
    }
}
