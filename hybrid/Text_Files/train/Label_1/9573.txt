#include <bits/stdc++.h>
using namespace std;
const int N = 100005;
int n;
long long a[N];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    for (int i = 1; i <= 3; i++) {
        long long sum = accumulate(a + 1, a + n + 1, 0LL);
        if (sum % n == 0) {
            cout << "1 " << n << endl;
            cout << sum / n << endl;
        } else {
            int l = 1, r = n;
            while (l < r) {
                if (a[l] - a[r] <= sum % n) {
                    cout << l << " " << r << endl;
                    cout << (sum % n - a[l] + a[r]) / (r - l + 1) << endl;
                    break;
                }
                sum -= a[l];
                sum += a[r];
                l++;
                r--;
            }
        }
        for (int j = 1; j <= n; j++) { a[j] -= sum / n; }
    }
}
