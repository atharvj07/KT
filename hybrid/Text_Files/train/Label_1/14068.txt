#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int tasks, subtasks, time;
    cin >> tasks >> subtasks >> time;
    vector<int> durations(subtasks);
    for (int i = 0; i < subtasks; i++) {
        cin >> durations[i];
    }
    sort(durations.begin(), durations.end());
    int points = 0;
    for (int i = 0; i < subtasks; i++) {
        if (time >= durations[i]) {
            points++;
            time -= durations[i];
        } else {
            break;
        }
    }
    points += (time / subtasks) * subtasks;
    cout << points << endl;
    return 0;
}
