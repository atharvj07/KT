#include <bits/stdc++.h>
using namespace std;

const int LIM = 3e5 + 5;

int n, a[LIM], u[LIM], v[LIM], w[LIM], ans;
vector<pair<int, int>> adj[LIM];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i < n; i++) {
    cin >> u[i] >> v[i];
    w[i] = (u[i] < v[i] ? u[i] : v[i]);
    adj[u[i]].push_back({v[i], i});
    adj[v[i]].push_back({u[i], i});
  }
  priority_queue<pair<int, int>> pq;
  for (int i = 1; i <= n; i++) pq.push({-a[i], i});
  while (!pq.empty()) {
    int cnt = 0, pu = pq.top().second;
    ans = max(ans, -pq.top().first);
    pq.pop();
    vector<int> nw;
    for (auto vi : adj[pu]) {
      int vu = vi.first;
      int id = vi.second;
      if (a[vu] == -1) {
        cnt++;
      } else {
        nw.push_back(id);
      }
    }
    for (int i = 0; i < nw.size(); i++) {
      int id = nw[i];
      int vu = u[id] + v[id] - pu;
      a[vu] = -1;
      pq.push({-a[vu], vu});
    }
    if (!cnt) break;
  }
  cout << ans << endl;
  return 0;
}
