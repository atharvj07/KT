import sys
input = sys.stdin.readline

n, x = map(int, input().split())
edge = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    edge[a].append(b)
    edge[b].append(a)

vis = [False] * (n+1)
dis = [-1] * (n+1)

def dfs(now, p, d):
    vis[now] = True
    dis[now] = d
    for nxt in edge[now]:
        if vis[nxt] == False:
            dfs(nxt, now, d+1)

dfs(x, 0, 0)
mx = 0

for i in range(1, n+1):
    if vis[i] == False:
        dfs(i, 0, 0)
    mx = max(mx, dis[i])

print(mx+1)
