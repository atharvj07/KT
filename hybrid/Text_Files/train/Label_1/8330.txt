#include <bits/stdc++.h>
using namespace std;

int row[105][105], col[105][105], ans[105];

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++) {
            cin >> row[i][j];
            col[j][i] = row[i][j];
        }
    ans[1] = 1;
    for (int i = 2; i <= n; i++) {
        int x = 0;
        for (int j = 1; j <= n; j++) x |= row[i][j];
        int y = 0;
        for (int j = 1; j <= n; j++) y |= col[i][j];
        if (x == y) ans[i] = x;
        else ans[i] = x | y;
    }
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    cout << endl;
    return 0;
}
