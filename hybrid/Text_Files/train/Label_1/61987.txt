def dfs(i, color, path, adj):
    if i == len(adj) - 1:
        return 1 if path.count(0) == path.count(1) else 0
    ans = 0
    for j in range(i + 1, len(adj)):
        if adj[i][j] and (color[j] != -1 and color[j] != color or color[j] == -1):
            ans += dfs(j, color, path | {color[j]}, adj)
    return ans

n, p = map(int, input().split())
c = [-1] + list(map(int, input().split()))
adj = [[0] * (n + 1) for _ in range(n + 1)]
for i in range(n):
    for j in range(i + 1, n):
        if c[i] != 0 and c[j] != 0 and c[i] != c[j]:
            adj[i][j] = adj[j][i] = 1
print(dfs(0, c, set(), adj))
