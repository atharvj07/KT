#include<bits/stdc++.h>
using namespace std;
int const N = 100;
int p[N], a[N][N];
int main() {
    int n, m, k;
    cin >> n >> m >> k;
    for(int i = 0; i < k; ++i) cin >> p[i];
    for(int i = 0; i < n; ++i) {
        for(int j = 0; j < m; ++j) cin >> a[i][j];
    }
    int res = 0;
    bool vis[N] = {};
    for(int i = 0; i < n; ++i) {
        for(int j = 0; j < m; ++j) {
            if(vis[a[i][j] - 1]) continue;
            res += p[a[i][j] - 1];
            for(int ii = a[i][j] - 1; ii > 0; --ii) {
                p[ii] = p[ii - 1];
                vis[ii] = vis[ii - 1];
            }
            p[0] = a[i][j];
            vis[0] = true;
        }
    }
    cout << res << endl;
}
