def lds(s):
	n = len(s)
	dp = [1] * n
	inc = [i for i in range(n) if s[i] > s[i-1]]
	for i in inc:
		for j in range(i):
			if s[i] > s[j]:
				dp[i] = max(dp[i], dp[j] + 1)
	return max(dp)

def get_zeroes(s):
	return s.count('0')

def solve(s):
	n = len(s)
	t = ''
	for i in range(n):
		if s[i] == '0':
			t += '0'
		else:
			if lds(s[:i+1]) + get_zeroes(s[i+1:]) == lds(s[:i]) + get_zeroes(s[i+1:]):
				t += '0'
			else:
				t += '1'
	return t

s = input()
print(solve(s))
