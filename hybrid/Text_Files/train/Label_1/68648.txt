import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, q = map(int, input().split())
    a = [-1 if s == "-" else 1 for s in input()]
    for _ in range(q):
        l, r = map(int, input().split())
        l -= 1
        r -= 1
        print(max(0, a[l:r + 1].count(1) - a[l:r + 1].count(-1)))
