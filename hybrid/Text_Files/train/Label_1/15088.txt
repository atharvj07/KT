#include <bits/stdc++.h>
using namespace std;

const int maxn = 5001;
vector<bool> painted(maxn, false);

int main() {
  int n, q;
  cin >> n >> q;
  for (int i = 0; i < q; i++) {
    int l, r;
    cin >> l >> r;
    for (int j = l; j <= r; j++) {
      painted[j] = true;
    }
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (painted[i]) ans++;
  }
  cout << ans << endl;
}
