def solve(s, n):
    # Initialize the result string to the original and set the best k to 1
    best_str = s
    best_k = 1
    
    # Iterate over all possible values of k
    for k in range(1, n + 1):
        # Generate the modified string for the current k
        modified_str = s[:k][::-1] + s[k:]
        for i in range(1, n - k + 1):
            modified_str = modified_str[:i] + modified_str[i:i+k][::-1] + modified_str[i+k:]
        
        # Update the result if a lexicographically smaller string is found
        if modified_str < best_str:
            best_str = modified_str
            best_k = k
    
    return best_str, best_k

# Read the number of test cases
t = int(input())
for _ in range(t):
    # Read the length of the string and the string itself for each test case
    n = int(input())
    s = input()
    
    # Solve the problem and output the result
    result_str, result_k = solve(s, n)
    print(result_str)
    print(result_k)
