#include <bits/stdc++.h>
using namespace std;
string z,t,p;
int main(){
    cin >> z >> t >> p;
    int n=z.size(), m=t.size(), l=p.size();
    int dp[n+5][m+5][l+5];
    for(int i=n; i>=1; i--){
        for(int j=m; j>=1; j--){
            for(int k=l; k>=1; k--){
                if(z[i-1]==t[j-1] && z[i-1]!=p[k-1]){
                    dp[i][j][k]=max(dp[i][j][k], 1+dp[i-1][j-1][k]);
                }
                dp[i][j][k]=max(dp[i][j][k], dp[i-1][j][k]);
                dp[i][j][k]=max(dp[i][j][k], dp[i][j-1][k]);
                dp[i][j][k]=max(dp[i][j][k], dp[i][j][k-1]);
            }
        }
    }
    if(dp[n][m][l]==0) cout << 0 << endl;
    else{
        string ans="";
        int i=n, j=m, k=l;
        while(i>0 && j>0){
            if(z[i-1]==t[j-1] && z[i-1]!=p[k-1] && dp[i][j][k]==dp[i-1][j-1][k]+1){
                ans+=z[i-1];
                i--;
                j--;
            }
            else if(dp[i][j][k]==dp[i-1][j][k]) i--;
            else j--;
        }
        reverse(ans.begin(), ans.end());
        cout << ans << endl;
    }
}
