#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll n, q, a[100005], ans, p1 = 1e9, p;
    cin >> n;
    for (ll i = 0; i < n; i++) cin >> a[i];
    cin >> q;
    while (q--) {
        ll l, r;
        cin >> l >> r;
        l--, r--;
        ans = 0;
        while (l < r) {
            bool ch = 0;
            p = p1;
            for (ll i = l; i <= r; i++) {
                if ((i - l) % 2 == 0 && !ch) {
                    p = a[i], ch = 1;
                    continue;
                }
                p = (p + a[i]) % 10;
                if (p >= 10) ans++, ch = 0, p /= 10;
            }
            l = (l + 2) / 2, r = (r - 1) / 2;
        }
        cout << ans << endl;
    }
}
