#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  bool foundQ = false;
  for (char c : s) {
    if (c == 'Q') {
      foundQ = true;
      break;
    }
  }
  if (foundQ) {
    cout << "Yes" << endl;
  } else {
    cout << "No" << endl;
  }
  return 0;
}
