#include <bits/stdc++.h>
using namespace std;

const int maxn = 100010;
int n, k, A[maxn], D[maxn], E[maxn], dp[maxn][110];
vector<int> adj[maxn];

int solve(int cur, int bond) {
  if (bond < 0) return -1e9;
  if (cur == n) return 0;
  int &ans = dp[cur][bond];
  if (ans != -1) return ans;
  ans = solve(cur + 1, bond);
  ans = max(ans, solve(E[cur], bond - 1) + A[cur] - D[cur]);
  for (int nxt : adj[cur])
    ans = max(ans, solve(nxt, bond) + A[cur] - D[cur]);
  return ans;
}

int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> A[i];
  for (int i = 1; i <= n; i++) cin >> D[i];
  for (int i = 1; i < n; i++) { cin >> E[i]; adj[i].push_back(E[i]); }
  memset(dp, -1, sizeof(dp));
  cout << solve(1, k) << '\n';
  return 0;
}
