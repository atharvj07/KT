import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        boolean ok = true;
        for (int i = 1; i < n; i++) {
            if (a[i - 1] != (a[i] ^ a[i - 1])) {
                ok = false;
                break;
            }
        }
        if (a[n - 1] != (a[0] ^ a[n - 1])) {
            ok = false;
        }
        System.out.println(ok ? "Yes" : "No");
    }
}
