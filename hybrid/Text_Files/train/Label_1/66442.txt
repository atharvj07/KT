n,m=map(int,input().split())
r,c=map(int,input().split())
x,y=map(int,input().split())
g=[input() for i in range(n)]
g[r-1]=g[r-1][:c-1]+'*'+g[r-1][c:]
vis=[[False]*m for i in range(n)]
vis[r-1][c-1]=True
q=[[r-1,c-1]]
for i,j in q:
    if i+1<n and not vis[i+1][j] and g[i+1][j]!='*':
        vis[i+1][j]=True
        q+=[[i+1,j]]
    if i-1>=0 and not vis[i-1][j] and g[i-1][j]!='*':
        vis[i-1][j]=True
        q+=[[i-1,j]]
    if j+1<m and not vis[i][j+1] and g[i][j+1]!='*':
        vis[i][j+1]=True
        q+=[[i,j+1]]
    if j-1>=0 and not vis[i][j-1] and g[i][j-1]!='*':
        if j-1>=c-1 or x>0:
            x-=1
            vis[i][j-1]=True
            q+=[[i,j-1]]
        elif y>0:
            y-=1
            vis[i][j-1]=True
            q+=[[i,j-1]]
cnt=0
for i in range(n):
    for j in range(m):
        if vis[i][j]:
            cnt+=1
print(cnt)
