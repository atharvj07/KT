n = int(input())
a = [int(x) for x in input().split()]
m = int(input())
b = [int(x) for x in input().split()]
c = [int(x) for x in input().split()]
cnt = [0 for x in range(1000000)]
for x in a:
    cnt[x] += 1
mx = 0
ans = 0
ans2 = 0
for i in range(len(b)):
    tmp = cnt[b[i]] + cnt[c[i]]
    if tmp > mx:
        mx = tmp
        ans = i + 1
        ans2 = tmp
    elif tmp == mx:
        ans2 = min(ans2, tmp)
print(ans)
