import sys
n,a = map(int,input().split())
d,c = [],[]
for i in range(n):
    x,y = map(int,input().split())
    d.append(x)
    c.append(y)
g = []
for i in range(n):
    if i>0: g.append((d[i]-d[i-1])**2)
    else: g.append(0)
ps = [0]*n
ps[n-1] = a-c[n-1]+g[n-1]
for i in range(n-2,-1,-1): ps[i] = max(ps[i+1],a-c[i]+g[i])
for i in range(n-2,-1,-1):
    if i==0: vv = 0
    else: vv = ps[i-1]
    ps[i] = max(ps[i],vv+a-c[i]+g[i])
print(max(ps))
