a, b = map(int, input().split())
n = 1
while a > 0:
    if a < n:
        n -= 1
        break
    a -= n**2
    n += 1
print(n - 1)
print(*[i**2 for i in range(n - 1, 0, -1)])
m = 1
while b > 0:
    if b < m:
        m -= 1
        break
    b -= m**2
    m += 1
print(m - 1)
print(*[i**2 for i in range(m - 1, 0, -1)])
