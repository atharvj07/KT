import java.util.*;
class Main {
    public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int fibo[] = new int[1005];
		fibo[1] = 1; fibo[2] = 1;
		for(int i = 3; i <= 1000; i++)	fibo[i] = fibo[i - 1] + fibo[i - 2];
		int n = sc.nextInt();
		for(int i = 1; i <= n; i++)
			System.out.print(fibo[i] % 2 == 1 ? 'O' : 'o');
		System.out.println();
	}
}
