import java.util.*;
import java.io.*;
class Main {
  static int n, p; static char[] c; static String next;
  static boolean isPalin(int l, int r) {
    for (int i = l; i <= (r + l) / 2; i++) if (c[i] != c[r - (i - l)]) return false;
    return true;
  }
  static boolean lex(char[] a, int l, int r) {
    for (int i = l; i <= r; i++) if (a[i] < c[i]) return true;
    return false;
  }
  static void dfs(int idx) {
    if (idx == n) { if (lex(c, 0, n - 1)) { next = String.valueOf(c); return; } }
    else for (char ch = 'a'; ch <= 'a' + p - 1; ch++) {
      if (idx > 0 && ch == c[idx - 1]) if (idx == 1 || !isPalin(idx - 2, idx)) continue;
      c[idx] = ch; dfs(idx + 1);
      if (next != null) return;
    }
  }
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt(); p = sc.nextInt(); c = sc.next().toCharArray();
    dfs(0);
    System.out.println(next == null ? "NO" : next);
  }
}
