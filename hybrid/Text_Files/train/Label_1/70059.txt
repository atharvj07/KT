n, k = map(int, input().split())
s = input()

s = sorted(set(s))  # Get unique stages sorted alphabetically
weight = 0
count = 0
prev = None

for c in s:
    if count == k:
        break
    if prev is None or ord(c) - ord(prev) > 1:
        weight += ord(c) - 96  # ASCII 'a' is 97
        prev = c
        count += 1

if count == k:
    print(weight)
else:
    print(-1)
