#include <bits/stdc++.h>

using namespace std;

const int N = 105;
char a[N];

int main() {
  int n;
  scanf("%s", a + 1);
  n = strlen(a + 1);
  int f[3] = {0, 0, 0};
  for (int i = 1; i <= n; i++) {
    if (a[i] == '.') continue;
    f[a[i] - 'A']++;
  }
  int h[3];
  for (int c = 0; c < 3; c++) h[c] = f[c];
  for (int i = 1; i <= n; i++) {
    if (a[i] == '.') continue;
    h[(a[i] - 'A' + 1) % 3]++;
    h[(a[i] - 'A' - 1 + 3) % 3]++;
  }
  for (int c = 0; c < 3; c++) {
    if (h[c] >= n) {
      puts("Yes");
      return 0;
    }
  }
  puts("No");
  return 0;
}
