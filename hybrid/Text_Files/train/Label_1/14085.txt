#include<bits/stdc++.h>
using namespace std;
int m,a[100005],b[100005],vis[100005],ans=0,l=0,r=0;
int main(){
    cin>>m;
    for(int i=1;i<=m;++i){
        cin>>a[i];
        if(!vis[a[i]]){b[++l]=a[i];vis[a[i]]=1;}
    }
    sort(b+1,b+1+l);
    for(int i=1;i<=l;++i){
        if(i==1||b[i]-b[i-1]>1){ans++;r=i;}
    }
    cout<<ans<<endl;
    for(int i=r,j=ans;j>=1;--i,--j){
        for(int k=1;k<=m;++k){
            if(a[k]==b[i]){cout<<a[k]<<" ";break;}
        }
    }
    return 0;
}
