import sys

def get_char_rank(ch):
    return ord(ch) - ord('a')

def get_ugly_pairs(s):
    n = len(s)
    ugly_pairs = []
    for i in range(n-1):
        if abs(get_char_rank(s[i]) - get_char_rank(s[i+1])) == 1:
            ugly_pairs.append((i, i+1))
    return ugly_pairs

def remove_ugly_pairs(s, ugly_pairs):
    n = len(s)
    t = []
    vis = [False] * n
    for i in range(n):
        if vis[i]:
            continue
        p1, p2 = -1, -1
        for p in ugly_pairs:
            if p[0] == i or p[1] == i:
                if p1 == -1:
                    p1 = p[0]
                    p2 = p[1]
                else:
                    p1 = min(p1, p[0])
                    p2 = max(p2, p[1])
        if p1 == -1:
            t.append(s[i])
            vis[i] = True
        else:
            for j in range(p1, p2+1):
                t.append(s[j])
                vis[j] = True
    return ''.join(t)

t = int(sys.stdin.readline())
for _ in range(t):
    s = sys.stdin.readline().strip()
    ugly_pairs = get_ugly_pairs(s)
    if len(ugly_pairs) > 0:
        ans = remove_ugly_pairs(s, ugly_pairs)
        print(ans)
    else:
        print(s)
