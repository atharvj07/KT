#include <bits/stdc++.h>
using namespace std;
long long dp[102][100002];
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> v(n);
  vector<int> va(m);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  for (int i = 0; i < m; i++) { cin >> va[i]; }
  int mx = 0;
  for (int i = 0; i < m; i++) { mx = max(mx, v[va[i] - 1]); }
  for (int i = 0; i <= n; i++) {
    for (int j = 0; j <= mx; j++) { dp[i][j] = -1e18; }
  }
  dp[0][0] = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= mx; j++) {
      dp[i][j] = dp[i - 1][j];
      if (j >= v[i - 1]) {
        dp[i][j] = max(dp[i][j], dp[i - 1][j - v[i - 1]] + v[i - 1]);
      }
      if (binary_search(va.begin(), va.end(), i)) {
        if (j + v[i - 1] <= mx) {
          dp[i][j] = max(dp[i][j], dp[i - 1][j] + v[i - 1]);
        }
      }
    }
  }
  long long ans = 0;
  for (int i = 0; i <= mx; i++) { ans = max(ans, dp[n][i]); }
  cout << ans << endl;
}
