#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int mod = 1e9 + 7;

int32_t main() {
    ll a, b, k, t;
    cin >> a >> b >> k >> t;
    ll ans = 0;
    for (ll x = -k; x <= k; x++) {
        ll diff = x + a - b;
        if (diff > 0) {
            ans += (1LL << t) - (1LL << max(0LL, t - (2 * k + 1 - abs(diff)))) + mod;
            ans %= mod;
        }
    }
    cout << ans << endl;
    return 0;
}
