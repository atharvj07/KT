#include <bits/stdc++.h>
 
using namespace std;
 
int main() {
    int n; cin >> n;
    int a, b;
    long long sum_a = 0, sum_b = 0;
 
    for (int i = 0; i < n; i++) {
        cin >> a >> b;
        sum_a += a;
        sum_b += b;
    }
    (sum_a <= sum_b/2) ? cout << "YES" : cout << "NO";
    return 0;
}
