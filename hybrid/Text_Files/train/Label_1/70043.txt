from collections import defaultdict

def dfs(graph, start):
    visited, stack = [], [start]
    while stack:
        vertex = stack.pop()
        if vertex not in visited:
            visited.append(vertex)
            stack.extend(sorted(graph[vertex], reverse=True))
    return visited

def solve(n, q, superiors, queries):
    graph = defaultdict(list)
    for i, p in enumerate(superiors, 2):
        graph[p].append(i)
    
    ans = []
    for u, k in queries:
        order = dfs(graph, u)
        ans.append(order[k-1] if k <= len(order) else -1)
    
    return ans

# Input reading and output printing
n, q = map(int, input().split())
superiors = list(map(int, input().split()))
queries = [tuple(map(int, input().split())) for _ in range(q)]

ans = solve(n, q, superiors, queries)
for a in ans:
    print(a)
