n = int(input())
children = [list(map(int, input().split())) for _ in range(n)]

treated_children = []
i = 0

while i < len(children):
    v, d, p = children[i]
    treated_children.append(i + 1)
    additional_cry = 0
    
    for j in range(i + 1, len(children)):
        if j == i + 1:
            children[j][2] -= v
        else:
            if v > 0:
                children[j][2] -= v
                v -= 1
            children[j][2] -= additional_cry
        
        if children[j][2] < 0:
            additional_cry += children[j][1]
            children[j] = 'crying'

    children = [child for child in children if child != 'crying']
    i += 1

print(len(treated_children))
print(*treated_children)
