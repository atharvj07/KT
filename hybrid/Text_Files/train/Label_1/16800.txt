#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }
  std::vector<int> b(n);
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (a[j] > a[i]) {
        b[j]++;
      } else {
        b[i]++;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    std::cout << b[i] + 1 << " ";
  }
  std::cout << "\n";
  return 0;
}
