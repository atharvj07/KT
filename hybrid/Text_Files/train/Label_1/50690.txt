import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        BigInteger x = in.nextBigInteger(), y = in.nextBigInteger(), z = in.nextBigInteger();
        System.out.println(x.divide(z).add(y.divide(z)) + " " + x.min(y).mod(z));
    }
}
