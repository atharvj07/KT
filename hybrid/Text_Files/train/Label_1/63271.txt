import sys

mod = 10**9 + 7

n = int(input())
s = input()

l = 0
r = 0
ct = 0
ans = 0

for i in range(n):
    if s[i] != '?':
        if s[i] == 'b':
            l -= ct
            r -= ct
            ct = 0
        if s[i] == 'c':
            r = max(r, ct + 1)
    else:
        ct += 1

ans = (2 * ans) % mod
ans = (ans + l * r) % mod

print(ans)
