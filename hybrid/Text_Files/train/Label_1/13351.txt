#include<bits/stdc++.h>
#define ll long long int
using namespace std;
int32_t main() {
    ll t;
    cin >> t;
    while (t--) {
        ll n, m;
        cin >> n >> m;
        vector<ll> a(n + 1), b(m + 1), dp(m + 2, 0);
        for (ll i = 1; i <= n; i++) {
            cin >> a[i];
        }
        for (ll j = 1; j <= m; j++) {
            cin >> b[j];
        }
        for (ll i = n; i >= 1; i--) {
            for (ll j = m; j >= 1; j--) {
                if (a[i] == b[j]) {
                    dp[j] = dp[j + 1] + 1;
                } else {
                    dp[j] = max(dp[j], dp[j + 1]);
                }
            }
        }
        if (dp[1] > 0) {
            cout << "YES" << endl;
            cout << dp[1] << endl;
            ll cnt = dp[1];
            for (ll i = 1; i <= m; i++) {
                if (dp[i + 1] == dp[i]) {
                    continue;
                }
                for (ll j = n; j >= 1; j--) {
                    if (a[j] == b[i]) {
                        cnt--;
                        cout << a[j] << " ";
                        break;
                    }
                }
                if (cnt == 0) {
                    break;
                }
            }
            cout << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
