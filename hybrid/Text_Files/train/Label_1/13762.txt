#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
    int n, m;
    cin >> n >> m;
    vector<int> r(n), c(n);
    while (m--) {
        int x, y;
        cin >> x >> y;
        r[x - 1] = 1;
        c[y - 1] = 1;
        cout << (n * n - r[x - 1]) * (n * n - c[y - 1]) << " ";
    }
}
