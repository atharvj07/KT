import java.util.Arrays;
import java.util.Scanner;

public class Main {
    private static final int INF = Integer.MAX_VALUE;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while(t-- > 0){
            int n = sc.nextInt();
            int x = sc.nextInt();
            int[] a = new int[n];
            for(int i = 0; i < n; i++){
                a[i] = sc.nextInt();
            }
            Arrays.sort(a);
            int ans = 0;
            int cur = x;
            for(int i = n - 1; i >= 0; i--){
                if(a[i] == cur) continue;
                if(a[i] < cur) cur = a[i];
                ans++;
            }
            System.out.println(ans);
        }
    }
}
