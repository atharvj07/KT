#include <iostream>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int j = k;
    for (int i = 0; i < n; i++) {
        if (j == k) {
            cout << "PRINT " << s[i] << endl;
        } else {
            if (j < k) {
                for (int x = 0; x < k - j; x++) { cout << "RIGHT" << endl; }
                j = k;
                cout << "PRINT " << s[i] << endl;
            }
            if (j > k) {
                for (int x = 0; x < j - k; x++) { cout << "LEFT" << endl; }
                j = k;
                cout << "PRINT " << s[i] << endl;
            }
        }
        if (i + 1 < n) { k++; }
    }
    return 0;
}
