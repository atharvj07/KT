#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<ll> a(n);
  for (auto &it : a) cin >> it;
  k--;
  multiset<ll> b;
  ll ans = -1000000000000000000LL;
  for (int i = 0; i < n - 1; i++) {
    b.insert(a[i]);
    if (b.size() == k) {
      ans = max(ans, *b.begin());
      b.erase(b.begin());
    }
  }
  cout << max(ans, a[n - 1]) << endl;
}
