import sys
from math import sqrt

input = sys.stdin.readline

ax, ay, bx, by, tx, ty = map(int, input().split())
n = int(input())
bottles = [list(map(int, input().split())) for i in range(n)]

def dist(a, b):
    return sqrt((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2)

dp = [[[[1e9] * 3 for j in range(3)] for i in range(3)] for k in range(n + 1)]
dp[0][0][0][0] = 0

for i in range(n):
    x, y = bottles[i]
    for b1 in range(3):
        for b2 in range(3):
            for a1 in range(3):
                for a2 in range(3):
                    dp[i + 1][b1][b2][a2] = min(
                        dp[i + 1][b1][b2][a2],
                        dp[i][b1][b2][a1] + dist([(ax if a1 == 0 else bx if a1 == 1 else tx), (ay if a1 == 0 else by if a1 == 1 else ty)], [x, y]),
                        dp[i][b1][b2][a2] + dist([(ax if a2 == 0 else bx if a2 == 1 else tx), (ay if a2 == 0 else by if a2 == 1 else ty)], [x, y]),
                        dp[i][b1][b2][a1] + dist([(bx if b1 == 0 else tx), (by if b1 == 0 else ty)], [x, y]),
                        dp[i][b1][b2][a2] + dist([(bx if b2 == 0 else tx), (by if b2 == 0 else ty)], [x, y]),
                    )

print(dp[n][2][2][2])
