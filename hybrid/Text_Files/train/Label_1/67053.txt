n = int(input())
p = list(map(int, input().split()))
d = {}
for i in range(0, n):
    if p[i] in d:
        d[p[i]].append(i)
    else:
        d[p[i]] = [i]
ans = n
t = []
for i in range(0, n):
    if len(d[p[i]]) == 1:
        ans += n - 1
        t.append(i)
if len(t) > 1:
    ans += n - 1
print(ans)
