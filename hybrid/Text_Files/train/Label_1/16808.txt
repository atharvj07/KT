#include <bits/stdc++.h>
using namespace std;
int main() {
    int m;
    string s;
    cin >> m >> s;
    vector<bool> v(s.length() + 1, false);
    for (int i = 0, j; i < s.length(); i++) {
        if (i == 0 || s[i] != s[i - 1]) j = i;
        if (i + m - 1 < s.length() && s[i] == s[i + m - 1]) {
            for (int k = j; k <= i; k++) v[k] = true;
        }
    }
    vector<char> ans;
    for (int i = 0; i < s.length(); i++) {
        if (v[i]) ans.push_back(s[i]);
    }
    sort(ans.begin(), ans.end());
    for (auto c : ans) cout << c;
    cout << endl;
}
