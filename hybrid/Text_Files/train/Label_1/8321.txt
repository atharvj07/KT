#include <bits/stdc++.h>
using namespace std;

int main() {
  int r, d, n;
  cin >> r >> d;
  cin >> n;
  long long ans = 0;
  while (n--) {
    int x, y, ri;
    cin >> x >> y >> ri;
    if ((long long)x * x + (long long)y * y <= (long long)(r - d) * (r - d)) {
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
