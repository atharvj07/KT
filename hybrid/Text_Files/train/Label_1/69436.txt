n, s = int(input()), input()
r, l = [], []
for i in range(1, n + 1):
    if s[i - 1] == 'R': r.append(i)
    else: l.append(i)
p, m = [], len(r) if len(r) > len(l) else len(l)
for i in range(m):
    if r: p.append([r.pop(0), l.pop(0)])
    else: break
for i in range(m, len(r) + len(l) - 1, 2):
    p.append([r.pop(0), l.pop(0)])
    p.append([l.pop(0), r.pop(0)])
for i in p: print(*i)
