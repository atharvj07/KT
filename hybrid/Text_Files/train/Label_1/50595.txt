import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt(), k = sc.nextInt();
            String s = sc.next();
            int[] count = new int[26];
            for (int i = 0; i < n; i++) {
                count[s.charAt(i) - 'a']++;
            }
            boolean found = false;
            for (int i = 0; i < 26; i++) {
                if (count[i] % k != 0) {
                    char c = (char) (i + 'a');
                    int need = (k - count[i] % k) % k;
                    for (int j = n - 1; j >= 0; j--) {
                        if (s.charAt(j) > c) {
                            count[s.charAt(j) - 'a']--;
                            count[i]++;
                            s = s.substring(0, j) + c + s.substring(j + 1);
                            found = true;
                            break;
                        }
                    }
                    if (found) {
                        break;
                    }
                }
            }
            for (int i = 0; i < 26; i++) {
                while (count[i] % k != 0) {
                    count[i]++;
                    s += (char) (i + 'a');
                }
            }
            if (s.length() == n) {
                System.out.println(s);
            } else {
                System.out.println("-1");
            }
        }
    }
}
