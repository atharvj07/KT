#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MAXN = 2e5+5;
ll n, k, h[MAXN];

int main() {
    cin>>n>>k;
    for(int i = 0; i < n; ++i) cin>>h[i];
    sort(h, h+n);
    int L = 0, R = n-1, l = 0, r = 0;
    while(L <= R) {
        if(h[L] + h[R] <= k) {
            l += h[L] + h[R] - 1;
            L++, R--;
        }
        else {
            r += h[R] - 1;
            R--;
        }
    }
    cout<<l+r<<endl;
}
