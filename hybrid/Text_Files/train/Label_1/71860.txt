from collections import defaultdict
from math import gcd
from sys import stdin

def prime_factors(n):
    factors = defaultdict(int)
    d = 2
    while d * d <= n:
        while (n % d) == 0:
            factors[d] += 1
            n //= d
        d += 1
    if n > 1:
       factors[n] += 1
    return factors

def normalize_factors(factors, k):
    return tuple(sorted((p, e % k) for p, e in factors.items() if e % k != 0))

n, k = map(int, stdin.readline().split())
a = list(map(int, stdin.readline().split()))

factor_counts = defaultdict(int)
for number in a:
    factors = prime_factors(number)
    normalized = normalize_factors(factors, k)
    factor_counts[normalized] += 1

result = 0
for factors, count in factor_counts.items():
    if all(e == 0 for _, e in factors):
        result += count * (count - 1) // 2
    else:
        inv_factors = tuple((p, (k - e) % k) for p, e in factors)
        if inv_factors in factor_counts:
            result += count * factor_counts[inv_factors]

# To avoid double counting
result //= 2

print(result)
