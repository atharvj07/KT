n = int(input())
a = [int(x) for x in input().split()]
a.append(0)
dp = [[[1000000000] * 3 for _ in range(n + 2)] for _ in range(n + 2)]
dp[0][1][1] = 0
for i in range(1, n + 1):
    for j in range(1, n + 2):
        for k in range(1, 3):
            if a[i] > 0:
                dp[i][j][k] = min(dp[i][j][k], dp[i - 1][j][k])
            if a[i] <= j - k:
                dp[i][j][k] = min(dp[i][j][k], dp[i - 1][j - a[i]][k] + 1)
            if i + 1 <= n:
                dp[i][j][k] = min(dp[i][j][k], dp[i][j][k - 1] + 1)
print(dp[n][n][1])
