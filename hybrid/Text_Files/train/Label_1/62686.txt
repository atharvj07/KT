import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
freq = [0] * 100001

for x in a:
    freq[x] += 1

mod = 1000000007

def fastpow(x, p):
    if p == 0:
        return 1
    if p == 1:
        return x

    val = fastpow(x, int(p / 2)) % mod
    val = (val * val) % mod

    if p % 2 != 0:
        val = (val * x) % mod

    return val % mod

num = 1
den = 1
for x in range(2, 100001):
    if freq[x] > 0:
        num = (num * fastpow(x, freq[x])) % mod
        den = (den * fastpow(x - 1, freq[x])) % mod

print((num * fastpow(den, mod - 2)) % mod)
