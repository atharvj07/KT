n = int(input())
players = []
for _ in range(n):
    data = list(map(int, input().split()))
    m_i = data[0]
    a_i = sorted(data[1:])
    players.append(a_i)

def can_win(players):
    sorted_players = sorted(players, key=lambda x: (len(x), x))
    for i in range(len(sorted_players)-1):
        if all(a >= b for a, b in zip(sorted_players[i+1], sorted_players[i])):
            return "NO"
    return "YES"

results = [can_win(players[:i] + players[i+1:]) for i in range(n)]
for result in results:
    print(result)
