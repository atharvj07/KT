n,m,x,y=map(int,input().split())
lines=[]
for i in range(1,n+1):
    for j in range(1,m+1):
        if i==x and j==y:
            for k in range(1,n+1):
                lines.append([k,j])
            for k in range(1,m+1):
                lines.append([i,k])
        else:
            lines.append([i,j])
            
for l in lines:
    print(*l)
