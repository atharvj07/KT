#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 5;
int n, a[MAXN];

int main() {
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> a[i];
  sort(a, a + n);
  long long sum = 0;
  for (int i = 1; i < n; ++i) {
    sum += max(0, a[i - 1] - a[i] + 1);
    a[i] = max(a[i], a[i - 1] - 1);
  }
  cout << sum / (a[n - 1] - a[0]) + 1 << " " << a[n - 1] - a[0] << "\n";
  return 0;
}
