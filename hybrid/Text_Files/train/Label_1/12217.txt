#include <bits/stdc++.h>

const long long MOD = 1e6;

int main() {
  long long n;
  std::cin >> n;
  std::vector<std::pair<long long, long long>> v(n);
  for (int i = 0; i < n; i++) {
    std::cin >> v[i].first >> v[i].second;
    v[i].first %= MOD;
    v[i].second %= MOD;
  }

  std::vector<int> ans(n);
  long long x = 0, y = 0;
  for (int i = 0; i < n; i++) {
    if (std::abs(x + v[i].first) + std::abs(y + v[i].second) <= MOD) {
      ans[i] = 1;
      x += v[i].first;
      y += v[i].second;
    } else {
      ans[i] = -1;
      x -= v[i].first;
      y -= v[i].second;
    }
  }

  for (int &i : ans) std::cout << i << " ";
  std::cout << std::endl;
}
