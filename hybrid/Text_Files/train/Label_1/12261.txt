#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int m, n;
  cin >> m;
  vector<int> q(m);
  for (int i = 0; i < m; i++) { cin >> q[i]; }
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  sort(q.begin(), q.end());
  int ans = 0;
  int j = 0;
  for (int i = 0; i < n; i++) {
    if (j < m && q[j] <= n - i) {
      ans += a[i] + min(a[i], a[i + 1]);
      i += q[j] - 1;
      j++;
    } else {
      ans += a[i];
    }
  }
  cout << ans << endl;
  return 0;
}
