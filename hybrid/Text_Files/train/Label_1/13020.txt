#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, t;
  cin >> n >> t;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    int s, d;
    cin >> s >> d;
    while (s <= t) s += d;
    v[i] = {s, d};
  }
  int ans = 0;
  for (int i = 1; i < n; i++) {
    if (v[i].first < v[ans].first) ans = i;
  }
  cout << ans + 1 << endl;
  return 0;
}
