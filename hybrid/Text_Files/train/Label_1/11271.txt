#include <iostream>
#include <vector>
using namespace std;
const int MAXN = 200000;
long long a[MAXN];

vector<long long> merge(vector<long long> v1, vector<long long> v2) {
  vector<long long> ans;
  int n1 = v1.size(), n2 = v2.size(), i = 0, j = 0;
  while (i < n1 && j < n2) {
    if (v1[i] > v2[j]) {
      ans.push_back(v1[i]);
      i++;
    } else {
      ans.push_back(v2[j]);
      j++;
    }
  }
  while (i < n1) {
    ans.push_back(v1[i]);
    i++;
  }
  while (j < n2) {
    ans.push_back(v2[j]);
    j++;
  }
  return ans;
}

vector<long long> mergeSort(vector<long long> arr) {
  if (arr.size() <= 1) return arr;
  int n = arr.size();
  vector<long long> left = mergeSort(vector<long long>(arr.begin(), arr.begin() + n / 2));
  vector<long long> right = mergeSort(vector<long long>(arr.begin() + n / 2, arr.end()));
  return merge(left, right);
}

int main() {
  int n;
  cin >> n;
  long long sum = 0;
  for (int i = 0; i < n; i++) cin >> a[i], sum += a[i];
  vector<long long> arr(a, a + n);
  arr = mergeSort(arr);
  int i = 0, j = n - 1;
  while (i < j) {
    sum += 2 * arr[i] * arr[j];
    i++;
    j--;
  }
  cout << sum << endl;
  return 0;
}
