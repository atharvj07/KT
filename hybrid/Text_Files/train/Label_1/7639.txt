int n,m;
int main() {
    int t; cin>>t;
    while(t--){
        cin>>n;
        string s; cin>>s;
        int lst = -1;
        int ans = 0;
        for (int i=0; i<n; i++){
            if (s[i]==s[lst]){
                i--;
                lst--;
                ans--;
            }
            else{
                ans++;
                lst = i;
            }
        }
        cout<<ans<<endl;
    }
}
