MOD = 10**9 + 7

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(k):
    nodes = 2**k - 1
    leaf_nodes = 2**(k-1)
    internal_nodes = nodes - leaf_nodes
    total_ways = 1

    # For each internal node, we have 4 ways to assign colors such that children are of different colors.
    total_ways *= power(4, internal_nodes, MOD)
    total_ways %= MOD

    # For the leaf nodes, we have 6 choices for the first node and 4 choices for the rest.
    total_ways *= 6 * power(4, leaf_nodes-1, MOD)
    total_ways %= MOD

    return total_ways

k = int(input())
print(solve(k))
