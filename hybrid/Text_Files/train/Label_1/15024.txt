#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
const int maxn = 3e5 + 5;
ll n, a, d[maxn], c[maxn], dp[maxn], dif[maxn];
int main() {
    cin >> n >> a;
    for (ll i = 1; i <= n; i++) cin >> d[i] >> c[i];
    for (ll i = 1; i <= n; i++) dif[i] = (d[i + 1] - d[i]) * (d[i + 1] - d[i]);
    for (ll i = 1; i <= n; i++) {
        dp[i] = max(dp[i - 1], dp[i - 2] + a - c[i] - dif[i]);
    }
    cout << dp[n] << endl;
    return 0;
}
