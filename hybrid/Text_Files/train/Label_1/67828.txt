n = int(input().strip())
x = list(map(int, input().strip().split()))
v = list(map(int, input().strip().split()))

points = sorted(zip(x, v), key=lambda p: (p[0], p[1]))

total_distance = 0
for i in range(n):
    for j in range(i + 1, n):
        if points[i][1] <= points[j][1]:
            total_distance += points[j][0] - points[i][0]
        else:
            intersect_time = (points[i][0] - points[j][0]) / (points[j][1] - points[i][1])
            new_i_pos = points[i][0] + points[i][1] * intersect_time
            new_j_pos = points[j][0] + points[j][1] * intersect_time
            total_distance += max(0, new_j_pos - new_i_pos)

print(total_distance)
