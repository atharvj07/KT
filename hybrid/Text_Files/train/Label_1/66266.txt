import sys
n, m = map(int, sys.stdin.readline().split())
a = [list(sys.stdin.readline().strip()) for i in range(n)]
x = [[0 for i in range(m+2)] for j in range(n+2)]
for i in range(1, n+1):
    for j in range(1, m+1):
        if a[i-1][j-1] == '.': b = 0
        else: b = int(a[i-1][j-1])
        x[i][j] = b + x[i+1][j] + x[i-1][j] + x[i][j+1] + x[i][j-1] + x[i+1][j+1] + x[i-1][j-1] + x[i+1][j-1] + x[i-1][j+1]
        if a[i-1][j-1] != '.' and b != x[i][j] - x[i][j-1] - x[i][j+1]:
            print('NO')
            sys.exit()
print('YES')
