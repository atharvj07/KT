#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (int)(n); ++i)

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  rep(i, n) cin >> a[i];
  int l = -1e9, r = 1e9;
  rep(i, n) {
    l = max(l, a[i]);
    r = min(r, a[i]);
  }
  if (l == r) {
    cout << 1 << '\n';
    cout << l << ' ' << 0 << '\n';
    return 0;
  }
  int p = 0, cnt = 0, x = 0;
  bool f = 0;
  rep(i, n) {
    if (a[i] < l || a[i] > r) {
      continue;
    }
    if (cnt == 0) {
      cnt = 1;
      x = a[i];
      f = 1;
    } else if (a[i] - x == p) {
      cnt++;
    } else {
      l = a[i];
      cnt = 1;
      x = a[i];
    }
    p = a[i] - x;
  }
  if (f) {
    cout << 1 << '\n';
    cout << x << ' ' << p << '\n';
  } else {
    cout << 0 << '\n';
  }
  return 0;
}
