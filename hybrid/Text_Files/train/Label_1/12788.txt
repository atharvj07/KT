#include <bits/stdc++.h>

using namespace std;

string s;
    int n;
    vector<int> pos[3];
int main() {
    cin >> s;
    n = s.size();
    for (int i = 0; i < n; i++) pos[s[i]-'0'].push_back(i);
    for (int i = 2; i >= 0; i--) {
      while (pos[i].size() >= 2) {
        s[pos[i].back()] = i + '0';
        pos[i].pop_back();
      }
      for (int j = i-1; j >= 0; j--) {
        while (pos[i].size() && pos[j].size()) {
          s[pos[i].back()] = i + '0';
          s[pos[j].back()] = j + '0';
          pos[i].pop_back();
          pos[j].pop_back();
        }
      }
    }
    cout << s << endl;
    return 0;
}
