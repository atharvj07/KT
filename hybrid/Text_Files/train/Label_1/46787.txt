import java.util.*;

public class Main {
  static int res;

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    List<Integer> arr = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      arr.add(in.nextInt());
    }
    Collections.sort(arr);
    int common = 0;
    for (int i = 1; i < n; i++) {
      common = gcd(common, arr.get(i) - arr.get(i - 1));
      if (common == 1) {
        break;
      }
    }
    if (common != 1) {
      System.out.println(-1);
      return;
    }
    res = 0;
    for (int val : arr) {
      if (val % common != 0) {
        res = -1;
        break;
      }
      res += val / common - 1;
    }
    System.out.println(res);
  }

  public static int gcd(int a, int b) {
    if (b == 0) {
      return a;
    }
    return gcd(b, a % b);
  }
}
