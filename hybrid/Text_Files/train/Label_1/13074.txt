#include <bits/stdc++.h>
using namespace std;
#define int long long
struct node{
    int pos,val;
};
bool operator<(node a,node b){
    return a.val>b.val;
}
int32_t main(){
    int t; cin>>t;
    while(t--){
        int h,n; cin>>h>>n;
        vector<node> v;
        for(int i=0,var;i<n;i++){
            cin>>var; v.push_back({var,i});
        }
        sort(v.begin(),v.end());
        int cur=h;
        int mag=0;
        for(int i=0;i<n;i++){
            int dif=cur-v[i].pos;
            if(dif>3){
                mag+=dif/3-1;
                cur-=(dif/3-1)*3;
            }
            cur--;
        }
        if(cur>2) mag+=cur-2;
        cout<<mag<<endl;
    }
    return 0;
}
