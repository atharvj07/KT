import sys
input = sys.stdin.readline
n, m = map(int,input().split())
a = list(map(int,input().split()))
a = [0]*m + a + [0]*m
ans = []
gl = []
gr = []
for i in range(1, n+1):
    if not a[a[i]]:
        a[a[i]] = i; continue
    if a[a[i]]>n: gr.append(i); continue
    gl.append(i)
    a[a[i]] = 0
L = len(gl)
for i in range(L):
    ans.append([gl[i],gr[i]])
print(L)
for i in ans: print(*i)
