t = int(input())
for _ in range(t):
    px, py = map(int, input().split())
    s = input()
    rx = s.count('R') - s.count('L')
    uy = s.count('U') - s.count('D')
    
    if (px <= rx and py <= uy) or (px >= -rx and py >= -uy):
        if (px >= 0 and py >= 0 and rx >= px and uy >= py) or (px <= 0 and py <= 0 and -rx <= px and -uy <= py):
            print("YES")
        else:
            if px >= 0 and rx < px:
                print("NO")
            elif px <= 0 and -rx > px:
                print("NO")
            elif py >= 0 and uy < py:
                print("NO")
            elif py <= 0 and -uy > py:
                print("NO")
            else:
                print("YES")
    else:
        print("NO")
