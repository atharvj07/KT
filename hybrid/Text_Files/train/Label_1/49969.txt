import java.util.*;

public class Main {
    private static List<Point> points;
    private static int n;

    private static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        points = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            points.add(new Point(x, y));
        }

        double sum = 0;
        for (int i = 0; i < n; i++) {
            int next = (i + 1) % n;
            sum += Math.atan2(points.get(next).y - points.get(i).y,
                    points.get(next).x - points.get(i).x);
        }

        System.out.println(Math.abs(sum) < Math.PI ? "1" : "0");
    }
}
