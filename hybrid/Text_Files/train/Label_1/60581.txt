n = int(input())
p = list(map(int, input().split()))

# Create a chessboard initialized with 0
chessboard = [[0 for _ in range(n)] for _ in range(n)]

# Function to fill the region for a given cell
def fill_region(x, y, val):
    if x < 0 or x >= n or y < 0 or y >= n or chessboard[x][y] != 0:
        return 0
    chessboard[x][y] = val
    return 1 + fill_region(x+1, y, val) + fill_region(x, y-1, val)

# Fill regions starting from the diagonal cells
for i in range(n):
    if fill_region(i, p[i]-1, p[i]) != p[i]:
        print(-1)
        exit(0)

# Print the filled chessboard
for i in range(n):
    print(*chessboard[i][:i+1])
