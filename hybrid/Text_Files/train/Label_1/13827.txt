#include<bits/stdc++.h>
using namespace std;
const int mxN = 1e6+10;
int p[mxN];
int find(int x){
    return p[x]==x? x : p[x]=find(p[x]);
}
int main(){
    ios_base::sync_with_stdio(0);
    for(int i=0; i<mxN; i++){
        p[i]=i;
    }
    int x, y, z;
    set<int> st;
    map<int, int> mp;
    for(int i=0; i<8; i++){
        cin>>x>>y>>z;
        st.insert(x), st.insert(y), st.insert(z);
        if(mp.count(x^y)){
            int par = find(mp[x^y]);
            int cpar = find(z);
            if(par==cpar){
                cout<<"NO\n";
                return 0;
            }
            p[par] = cpar;
        }
        else{
            mp[x^y] = z;
        }
    }
    if(st.size()!=8){
        cout<<"NO\n";
        return 0;
    }
    for(auto x : st){
        int par = find(x);
        cout<<x-par<<" "<<y-par<<" "<<z-par<<"\n";
    }
}
