#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int mod = 1e9 + 7, N = 21;
int n, m, k;
ll dp[N][N][1 << 10];
int g[N][N];

ll solve(int x, int y, int cur = 0) {
    if (dp[x][y][cur] >= 0) return dp[x][y][cur];
    if (x == n && y == m) { ll ret = (cur == k); return dp[x][y][cur] = ret; }
    dp[x][y][cur] = 0;
    if (x < n) (dp[x][y][cur] += solve(x + 1, y, cur ^ g[x + 1][y])) %= mod;
    if (y < m) (dp[x][y][cur] += solve(x, y + 1, cur ^ g[x][y + 1])) %= mod;
    return dp[x][y][cur];
}

int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> g[i][j];
        }
    }
    memset(dp, -1, sizeof dp);
    cout << solve(1, 1) << "\n";
}
