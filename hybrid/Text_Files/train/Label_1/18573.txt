#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll dp[55][3][3];
ll func(int pos, int zero, int pre, int n) {
  if (pos == n) return 1;
  ll &ret = dp[pos][zero][pre];
  if (~ret) return ret;
  ret = 0;
  if (zero == 0) ret += func(pos + 1, 0, 0, n);
  if (pre == 0) ret += func(pos + 1, 1, 1, n);
  ret += func(pos + 1, 1, 0, n);
  return ret;
}
int main() {
  ll n;
  cin >> n;
  memset(dp, -1, sizeof dp);
  cout << func(0, 0, 0, n) << "\n";
  return 0;
}
