from collections import Counter
from sys import stdin
input = stdin.readline

n = int(input())
s = input().rstrip()
bs = list(map(int, input().rstrip().split()))
cs = Counter(s)
b = 0
for k, v in sorted(cs.items()):
    v2 = v // 2
    b += (v - v2) * bs[ord(k) - ord('a')]
    cs[k] = v2
m = n // 2
for i in range(m):
    l = r = len(s) - i - 1
    for k in 'zyxwvutsrqponmlkjihgfedcba':
        if cs[k]:
            if bs[ord(k) - ord('a')] > bs[ord(s[l]) - ord('a')]:
                cs[k] -= 1
                cs[s[l]] += 1
                b += bs[ord(k) - ord('a')] - bs[ord(s[l]) - ord('a')]
                s = s[:l] + k + s[l+1:]
            else:
                cs[k] -= 1
                cs[s[r]] += 1
                b += bs[ord(k) - ord('a')] - bs[ord(s[r]) - ord('a')]
                s = s[:r] + k + s[r+1:]
            break
print(b)
