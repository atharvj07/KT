n = int(input())
a = list(map(int, input().split()))
mx, y = 0, 0
for x in a:
    if y > mx: mx = y
    y -= x
    if y > mx: mx = y
    y += x
x, y = 0, 0
for i in reversed(range(len(a))):
    x += a[i]
    if y > mx: mx = y
    y -= a[i]
    if y > mx: mx = y
    y += a[i]
    print((' ' * (x - 1 if x > 1 else 0)) + '/' + (' ' * (a[i] - 1)) + '\\')
