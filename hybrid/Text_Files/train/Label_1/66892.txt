n, m, k = map(int, input().split())
a = list(map(int, input().split()))
dp = [[0] * (n+1) for _ in range(m+1)]
sum = [0] * (n+1)
for i in range(1, n+1):
    sum[i] = sum[i-1] + a[i-1]
for r in range(1, n+1):
    for l in range(max(r - m, 0), r):
        dp[l % m+1][r] = max(dp[l % m+1][r], dp[l % m+1][l] + sum[r] - sum[l] - k*((r-l+1+m-1)//m))
print(max(dp[0][n], *sum))
