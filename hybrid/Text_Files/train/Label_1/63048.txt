def can_flow(arr1, arr2, n):
    for i in range(n):
        if arr1[i] == 1 and (arr2[i - 1] == 0 or arr2[i + 1] == 0):
            return "NO"
        if arr2[i] == 1 and (arr1[i - 1] == 0 or arr1[i + 1] == 0):
            return "NO"
        if arr1[i] == 2 and (arr2[i - 1] != 0 or arr2[i + 1] != 0):
            return "NO"
        if arr2[i] == 2 and (arr1[i - 1] != 0 or arr1[i + 1] != 0):
            return "NO"
        if arr1[i] == 4 and (arr2[i] != 5 and arr2[i] != 4):
            return "NO"
        if arr1[i] == 5 and (arr2[i] != 4 and arr2[i] != 5):
            return "NO"
        if arr2[i] == 3 and (arr1[i] != 4 and arr1[i] != 5):
            return "NO"
        if arr2[i] == 6 and (arr1[i] != 4 and arr1[i] != 5):
            return "NO"
    return "YES"


q = int(input())
for _ in range(q):
    n = int(input())
    arr1 = [int(x) for x in input()]
    arr2 = [int(x) for x in input()]
    print(can_flow(arr1, arr2, n))
