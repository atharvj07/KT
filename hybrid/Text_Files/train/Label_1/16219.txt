#include<bits/stdc++.h>
typedef long long ll;
const ll mod=1e9+7;
using namespace std;
ll pw[100001];
main(){
    int n,k;
    cin>>n>>k;
    string s;
    cin>>s;
    pw[0]=1;
    for(int i=1;i<=100000;i++)pw[i]=(pw[i-1]*10)%mod;
    ll ans=0;
    for(int i=0;i<n;i++){
        int num=s[i]-'0';
        if(i){
            int num2=s[i-1]-'0';
            if(k){
                ans=(ans+(((num2%mod)*(num%mod))%mod)*(pw[k-1]%mod))%mod;
            }
            k--;
        }
        ans=(ans+(num%mod))%mod;
    }
    cout<<ans<<endl;
}
