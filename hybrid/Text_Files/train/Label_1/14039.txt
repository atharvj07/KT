#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<double> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<int> b(n);
    double sum = 0;
    for (int i = 0; i < n - 1; i++) {
        double x = a[i] - floor(a[i]);
        sum += x;
        b[i] = floor(a[i]);
    }
    b[n - 1] = ceil(a[n - 1] - sum);
    for (int i = 0; i < n; i++) {
        cout << b[i] << "\n";
    }
    return 0;
}
