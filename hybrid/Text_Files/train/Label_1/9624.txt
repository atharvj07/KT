#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int MX = 2e5 + 5;
const ll INF = 1e18;
ll n, m, d, a[MX], dp[MX], days, i;

int main() {
    cin >> n >> m >> d;
    for (i = 1; i <= n; i++) cin >> a[i];
    memset(dp, d, sizeof dp);
    dp[0] = 0;
    for (i = 1; i <= n; i++)
        for (int j = max(a[i] - d, 0LL); j >= 0; j--) dp[a[i]] = min(dp[a[i]], dp[j] + 1);
    for (i = 1; i <= m; i++)
        days += (dp[i] < d);
    cout << days << '\n';
    for (i = 1; i <= n; i++)
        cout << lower_bound(dp, dp + m + 1, dp[a[i]]) - dp << ' ';
    cout << endl;
}
