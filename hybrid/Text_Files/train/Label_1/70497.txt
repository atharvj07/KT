import sys

n, k = map(int, sys.stdin.readline().split())
L = []
while n>0:
    msk = 0
    for i in range(0, 63):
        if (n & (1 << i)):
            msk = i
            break
    L.append(msk)
    n -= (1 << msk)
while len(L)<k: L.append(0)
if sum(L)==n:
    L.sort(reverse=True)
    print("Yes")
    for i in L:
        print(i, end=' ')
    print()
else:
    print("No")
