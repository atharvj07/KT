#include <bits/stdc++.h>

using namespace std;

#define pb push_back
#define endl '\n'
#define f first
#define s second

typedef long long ll;
typedef pair<int,int> pii;
const int maxn = 2e5+1;
const int inf = 1e9;

int n, m, k, a[maxn], vis[maxn], d[maxn];
vector<int> adj[maxn], adjs[maxn];

int dijk(int s, int t) {
    for (int i = 1; i <= n; i++) {
        d[i] = inf;
    }
    priority_queue<pii, vector<pii>, greater<pii>> pq;
    pq.push({0, s});
    d[s] = 0;
    while (!pq.empty()) {
        int u = pq.top().s;
        int w = pq.top().f;
        pq.pop();
        if (u == t) {
            return w;
        }
        if (d[u] != w) {
            continue;
        }
        for (int i = 0; i < adjs[u].size(); i++) {
            int v = adjs[u][i];
            if (d[v] > d[u] + 1) {
                d[v] = d[u] + 1;
                pq.push({d[v], v});
            }
        }
    }
    return -1;
}

int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= k; i++) {
        cin >> a[i];
        vis[a[i]] = 1;
    }
    for (int i = 1; i <= m; i++) {
        int x, y;
        cin >> x >> y;
        adj[x].pb(y);
        adj[y].pb(x);
        if (vis[x] && vis[y]) {
            adjs[x].pb(y);
            adjs[y].pb(x);
        }
    }
    int ans = -1;
    for (int i = 1; i <= k; i++) {
        for (int j = i+1; j <= k; j++) {
            if (!vis[a[i]] || !vis[a[j]]) {
                continue;
            }
            adjs[a[i]].pb(a[j]);
            adjs[a[j]].pb(a[i]);
            int res = dijk(1, n);
            if (res == -1) {
                res = 0;
            }
            ans = max(ans, res);
            adjs[a[i]].pop_back();
            adjs[a[j]].pop_back();
        }
    }
    cout << ans << endl;
    return 0;
}
