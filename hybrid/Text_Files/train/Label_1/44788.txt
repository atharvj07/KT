import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long cScore = 0, jScore = 0;
        while (n-- > 0) {
            int size = sc.nextInt();
            PriorityQueue<Integer> pq = new PriorityQueue<>((a, b) -> b - a);
            for (int i = 0; i < size; i++) pq.add(sc.nextInt());
            while (!pq.isEmpty()) {
                if (pq.size() % 2 == 1) cScore += pq.poll();
                else jScore += pq.poll();
            }
        }
        System.out.println(cScore + " " + jScore);
    }
}
