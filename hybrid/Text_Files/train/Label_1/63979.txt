import sys

n, q = map(int, sys.stdin.readline().split())
grid = [[0] * (n+2) for _ in range(3)]
for i in range(q):
    r, c = map(int, sys.stdin.readline().split())
    grid[r][c] ^= 1
    ans = 'Yes'
    for i in range(1, n+1):
        if grid[1][i] or grid[2][i-1] or grid[2][i] or grid[1][i+1]:
            ans = 'No'
            break
    print(ans)
