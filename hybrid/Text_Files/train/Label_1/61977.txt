import sys
n=int(sys.stdin.readline())
a=[int(x,16) for x in sys.stdin.read().split()]
b0=[x%16 for x in a];c0=set(b0)
b1=[(x>>4)%16 for x in a];c1=set(b1)
c=set(b0)^set(b1)
ans=4
for i in range(5,1<<4):
	t=set()
	for j in c:t.add(j)
	for j in c:t.add(j<<4)
	for j in c0:t.add(j<<8)
	for j in c1:t.add(j<<8)
	for j in range(4):
		for k in c:t.add(j<<12|k)
		for k in c0:t.add(j<<12|k)
		for k in c1:t.add(j<<12|k)
	for j in range(16):
		for k in c:t.add(j<<16|k)
	if len(t)==i:
		c=t
		ans=i
print(ans//4)
