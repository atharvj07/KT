import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int[] l = new int[2];
        for (int i = 0; i < n; i++) {
            x[i] = (i % 2 == 0 ? -x[i] : x[i]);
            l[i % 2] += x[i];
        }
        System.out.println(h * (Math.abs(l[0]) + Math.abs(l[1]) + w));
    }
}
