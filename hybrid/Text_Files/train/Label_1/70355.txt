def can_form_permutation(n, a, colors):
    blue_min, red_max = 1, n
    for i in range(n):
        if colors[i] == 'B':
            if a[i] < blue_min:
                return "NO"
            blue_min = max(blue_min, a[i])
        else:
            if a[i] > red_max:
                return "NO"
            red_max = min(red_max, a[i])
    return "YES"

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    colors = input().strip()
    print(can_form_permutation(n, a, colors))
