from collections import Counter

def max_joy(n, k, cards, favs, joys):
    card_count = Counter(cards)
    fav_count = Counter(favs)
    joy_levels = [0] * (k + 1)
    dp = [[0] * (n * k + 1) for _ in range(n + 1)]
    
    for i in range(1, k + 1):
        joy_levels[i] = joys[i - 1]
    
    for f in fav_count:
        f_count = min(card_count[f], k)
        for i in range(n, 0, -1):
            for j in range(n * k, f_count - 1, -1):
                for x in range(1, f_count + 1):
                    if j - x >= 0:
                        dp[i][j] = max(dp[i][j], dp[i - 1][j - x] + joy_levels[x])
    
    return dp[n][n * k]

n, k = map(int, input().split())
cards = list(map(int, input().split()))
favs = list(map(int, input().split()))
joys = list(map(int, input().split()))

print(max_joy(n, k, cards, favs, joys))
