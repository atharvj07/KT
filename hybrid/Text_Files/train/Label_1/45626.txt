import java.util.*;

class Main {
    static final int MAX_SIZE = 50;
    static final int MAX_VALUE = 100;
    static final int NA = -1;

    static int H, W, h, w;
    static int[][] A, B, C;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        H = sc.nextInt();
        W = sc.nextInt();
        A = new int[H + 1][W + 1];
        B = new int[H + 1][W + 1];
        for (int i = 1; i <= H; i++) {
            for (int j = 1; j <= W; j++) {
                A[i][j] = sc.nextInt();
            }
        }
        for (int i = 1; i <= H; i++) {
            for (int j = 1; j <= W; j++) {
                B[i][j] = sc.nextInt();
            }
        }
        h = sc.nextInt();
        w = sc.nextInt();
        C = new int[h + 1][w + 1];
        for (int i = 1; i <= h; i++) {
            for (int j = 1; j <= w; j++) {
                C[i][j] = sc.nextInt();
            }
        }
        sc.close();

        int ans = NA;
        for (int y = 1; y + h - 1 <= H; y++) {
            for (int x = 1; x + w - 1 <= W; x++) {
                int sum = 0;
                boolean ok = true;
                for (int i = y; i <= y + h - 1; i++) {
                    for (int j = x; j <= x + w - 1; j++) {
                        if (B[i][j] != C[i - y + 1][j - x + 1]) {
                            ok = false;
                            break;
                        } else if (B[i][j] == 1) {
                            sum += A[i][j];
                        }
                    }
                    if (!ok) {
                        break;
                    }
                }
                if (ok) {
                    ans = Math.max(ans, sum);
                }
            }
        }
        System.out.println(ans == NA ? "NA" : ans);
    }
}
