import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken()); // number of tournaments
        int k = Integer.parseInt(st.nextToken()); // number of kinds of sport

        // create an array to store the powers of sportsmen in each kind of sport
        int[][] powers = new int[n][k];

        // read the powers of sportsmen in each kind of sport
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            for (int j = 0; j < k; j++) {
                powers[i][j] = Integer.parseInt(st.nextToken());
            }
        }

        // create an array to store the number of contenders who can win for each tournament
        int[] winners = new int[n];

        for (int i = 0; i < n; i++) {
            // find the maximum power in each kind of sport
            int[] maxPowers = new int[k];
            for (int j = 0; j < k; j++) {
                maxPowers[j] = powers[i][j];
                for (int l = 0; l < n; l++) {
                    if (l == i) {
                        continue;
                    }
                    maxPowers[j] = Math.max(maxPowers[j], powers[l][j]);
                }
            }

            // count the number of contenders who can win
            int count = 0;
            for (int j = 0; j < k; j++) {
                if (powers[i][j] == maxPowers[j]) {
                    count++;
                }
            }
            winners[i] = count;
        }

        // print the number of contenders who can win for each tournament
        for (int i = 0; i < n; i++) {
            System.out.println(winners[i]);
        }
    }
}
