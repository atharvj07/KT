m = int(input())
q = [int(i) for i in input().split()]
n = int(input())
a = [int(i) for i in input().split()]
a.sort()
best = 1e9
for i in range(m):
    sum = 0
    x = 0
    for j in range(q[i]):
        sum += a[x]
        x += 1
    for j in range(2):
        if x < n:
            sum += a[x]
            x += 1
    best = min(best, sum)
if n <= 2: best = min(best, sum(a[:n]))
print(best)
