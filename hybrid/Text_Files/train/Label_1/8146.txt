#include<cstdio>
#include<vector>
#include<iostream>
#include<deque>
#include<algorithm>
using namespace std;
typedef long long ll;
int main() {
  char s[1024];
  scanf("%s", s);
  deque<ll> d;
  for (int i = 0; s[i]; ++i) {
    if (s[i] == '(' || s[i] == '[') {
      d.push_back(-1);
    } else if (s[i] == ')') {
      ll x = d.front();
      if (x >= 10) {d.clear(); break;}
      d.pop_front();
      if (d.empty()) {d.push_front(x); continue;}
      ll y = d.front();
      if (y < 0 || y + x >= 10) {d.clear(); break;}
      d.pop_front();
      d.push_front(y + x);
    } else if (s[i] == ']') {
      ll x = d.back();
      if (x >= 10) {d.clear(); break;}
      d.pop_back();
      if (d.empty()) {d.push_back(x); continue;}
      ll y = d.back();
      if (y < 0 || y + x >= 10) {d.clear(); break;}
      d.pop_back();
      d.push_back(y + x);
    } else {
      ll x = s[i] - '0';
      if (x >= 10) {d.clear(); break;}
      d.push_back(x);
    }
  }
  if (d.size() == 1 && d.front() == 0) {cout << 0 << endl;}
  else {
    while (!d.empty() && d.front() == 0) d.pop_front();
    if (d.empty()) {cout << "false" << endl;}
    else {
      string r;
      for (ll x : d) r += char('0' + x);
      cout << r << endl;
    }
  }
  return 0;
}
