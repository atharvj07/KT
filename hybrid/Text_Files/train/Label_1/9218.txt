#include<bits/stdc++.h>
using namespace std;
int main()
{
string s;
cin>>s;
int n=s.size();
int lps[n+1];

for(int i=0;i<=n;i++)lps[i]=0;

int j=0,i=1;
while(i<n)
{
    if(s[i]==s[j])
    {
        j++;
        lps[i]=j;
        i++;
    }
    else
    {
        if(j!=0)j=lps[j-1];
        else
        {
            lps[i]=0;
            i++;
        }
    }
}
vector<int>ans;
i=0;j=n;
while(j>0)
{
    if(lps[j]==0)
    {
        ans.push_back(0);
        j--;
    }
    else
    {
        if(s[j-1]==s[lps[j]-1])
        {
            ans.push_back(0);
            j--;
        }
        else
        {
            ans.push_back(1);
            j=lps[j]-1;
        }
    }
}
reverse(ans.begin(),ans.end());
for(int i=0;i<n;i++)cout<<ans[i]<<" ";
cout<<endl;
return 0;
}
