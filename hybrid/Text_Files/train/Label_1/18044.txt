#include<bits/stdc++.h>
using namespace std;
#define x first
#define y second
typedef long long ll;
int T,R,B;
pair<ll,ll>pt[15];
bool colinear(pair<ll,ll> p,pair<ll,ll> q,pair<ll,ll> r){
    return ((q.x-p.x)*(r.y-q.y)==(q.y-p.y)*(r.x-q.x));
}
bool intersect(pair<ll,ll> p,pair<ll,ll> q,pair<ll,ll> r,pair<ll,ll> s){
    int x1=p.x,y1=p.y,x2=q.x,y2=q.y,x3=r.x,y3=r.y,x4=s.x,y4=s.y;
    return (((x1<x4 && x4<x2) || (x2<x4 && x4<x1)) && ((y1<y4 && y4<y2) || (y2<y4 && y4<y1))) && (((x3<x1 && x1<x4) || (x4<x1 && x1<x3)) && ((y3<y1 && y1<y4) || (y4<y1 && y1<y3)));
}
int main(){
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    cin>>R>>B;
    for(int i=0;i<R;i++){
        cin>>pt[i].x>>pt[i].y;
    }
    for(int i=R;i<R+B;i++){
        cin>>pt[i].x>>pt[i].y;
    }
    bool fl=0;
    for(int i=0;i<R+B;i++){
        for(int j=0;j<i;j++){
            for(int k=i+1;k<R+B;k++){
                for(int l=0;l<k;l++){
                    if(i==j || i==k || i==l || j==k || j==l || k==l) continue;
                    if(colinear(pt[i],pt[j],pt[k]) && colinear(pt[i],pt[j],pt[l])){
                        if(intersect(pt[i],pt[j],pt[k],pt[l])){
                            fl=1;
                        }
                    }
                    else if(intersect(pt[i],pt[j],pt[k],pt[l])){
                        fl=1;
                    }
                }
            }
        }
    }
    if(fl) cout<<"No";
    else cout<<"Yes";
}
