#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1e9 + 7;
int n, m;
string s, p;
ll dp[100005];
int main() {
    cin >> s >> p;
    n = s.size();
    m = p.size();
    dp[m - 1] = 1;
    ll inv[n + 5];
    inv[1] = 1;
    for (int i = 2; i <= n; i++) inv[i] = mod - mod / i * (ll)inv[mod % i] % mod;
    ll pw[n + 5];
    pw[0] = 1;
    for (int i = 1; i <= n; i++) pw[i] = pw[i - 1] * 26 % mod;
    vector<int> h(n + 5);
    vector<int> p_h(m + 5);
    ll cur = 0;
    ll p_cur = 0;
    for (int i = 0; i < m; i++) {
        p_cur = p_cur * 26 + p[i] - 'a';
        p_h[i + 1] = p_cur;
    }
    for (int i = n - 1; i >= 0; i--) {
        cur = cur * 26 + s[i] - 'a';
        h[i] = cur;
    }
    for (int i = n - m; i >= 0; i--) {
        dp[i] += dp[i + 1];
        if (h[i] == p_h[m]) dp[i] = (dp[i] + dp[i + m]) % mod;
    }
    dp[0] = dp[0] * inv[m] % mod;
    for (int i = 1; i < n - m + 1; i++) {
        dp[i] = dp[i] * inv[m] % mod;
        dp[i] += dp[i - 1];
        dp[i] = dp[i] % mod;
    }
    cout << dp[n - m] << '\n';
    return 0;
}
