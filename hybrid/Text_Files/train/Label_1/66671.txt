n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))

i, j = 0, 0
length = 0
while i < n and j < m:
    sum_a, sum_b = a[i], b[j]
    while i < n and j < m and sum_a != sum_b:
        if sum_a < sum_b:
            i += 1
            if i < n:
                sum_a += a[i]
        else:
            j += 1
            if j < m:
                sum_b += b[j]
    
    if sum_a == sum_b:
        length += 1
        i += 1
        j += 1

if i < n or j < m:
    print(-1)
else:
    print(length)
