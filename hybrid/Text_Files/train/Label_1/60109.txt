from math import gcd
from functools import reduce

def find_gcd(list):
    x = reduce(gcd, list)
    return x

n = int(input())
strengths = list(map(int, input().split()))

max_pokemon = 1
for i in range(n):
    for j in range(i+1, n+1):
        if find_gcd(strengths[i:j]) > 1:
            max_pokemon = max(max_pokemon, j-i)

print(max_pokemon)
