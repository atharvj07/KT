n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

# Cumulative sum to know end positions of rooms in dormitories
cumulative = [0] * n
cumulative[0] = a[0]
for i in range(1, n):
    cumulative[i] = cumulative[i-1] + a[i]

# Function to find dormitory for given room number
def find_dormitory(room):
    low, high = 0, n-1
    while low <= high:
        mid = (low + high) // 2
        if cumulative[mid] < room:
            low = mid + 1
        else:
            high = mid - 1
    return low

for room in b:
    dormitory = find_dormitory(room)
    if dormitory == 0:
        print(dormitory + 1, room)
    else:
        print(dormitory + 1, room - cumulative[dormitory-1])
