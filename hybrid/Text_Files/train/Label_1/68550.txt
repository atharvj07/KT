for _ in range(int(input())):
    n = int(input())
    universities = list(map(int, input().split()))
    skills = list(map(int, input().split()))

    uni_skills = {}
    for uni, skill in zip(universities, skills):
        if uni in uni_skills:
            uni_skills[uni].append(skill)
        else:
            uni_skills[uni] = [skill]

    for uni in uni_skills:
        uni_skills[uni].sort(reverse=True)
        prefix_sum = [0]
        for skill in uni_skills[uni]:
            prefix_sum.append(prefix_sum[-1] + skill)
        uni_skills[uni] = prefix_sum

    strengths = [0] * n
    for uni in uni_skills:
        for k in range(1, len(uni_skills[uni])):
            full_teams = (len(uni_skills[uni]) - 1) // k
            strengths[k-1] += uni_skills[uni][full_teams * k]

    print(" ".join(map(str, strengths)))
