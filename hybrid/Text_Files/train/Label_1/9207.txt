#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
const int maxn = 2e5 + 5;

int n, k, par[maxn], depth[maxn], dis[maxn];
vector<int> adj[maxn];

int dfs(int u, int p){
    par[u] = p;
    depth[u] = depth[p] + 1;
    for(int v : adj[u]){
        if(v == p) continue;
        dis[v] = dis[u] + 1;
        dfs(v, u);
    }
}

int main(){
    cin >> n >> k;
    for(int i=0; i<n-1; i++){
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    dfs(1, 0);
    long long ans = 0, cnt = 0;
    for(int i=2; i<=n; i++){
        if(dis[i] > k){
            cnt += n - i;
            ans += 1ll * (n - i) * i;
        }
    }
    for(int i=2; i<=n; i++){
        if(dis[i] > k){
            long long x = cnt - (n - i);
            cnt = x;
            ans += 1ll * x * (i - 1);
        }
    }
    cout << ans << "\n";
}
