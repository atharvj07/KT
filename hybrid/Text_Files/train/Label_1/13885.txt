#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int m = 1;
  for (int i = 1, j = 1; i < n; i++) {
    if (a[i] <= a[i - 1]) j = i + 1;
    m = max(m, i - j + 2);
  }
  for (int i = 1; i < n; i++) {
    if (i - 1 == 0 || a[i] > a[i - 2]) {
      int j = i + 1;
      while (j < n && a[j] > a[j - 1]) j++;
      m = max(m, j - i);
    }
  }
  cout << m << endl;
  return 0;
}
