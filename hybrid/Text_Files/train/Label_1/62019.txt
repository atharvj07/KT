for _ in range(int(input())):
    p = input()
    q = ''
    for i in reversed(p):
        if i=='0' and q=='':
            continue
        q+=i
    if q=='':
        print(1)
    else:
        p = q[::-1]
        q = ''
        for i in reversed(p):
            if i=='0' and q=='':
                continue
            q+=i
        print(1+(int(q)>1))
