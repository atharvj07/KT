#include <cstdio>
#include <iostream>
#include <string>
#include <stack>
#include <algorithm>
using namespace std;

int main() {
	int n;
	cin >> n;
	if (n == 1) {
		cout << "D" << endl;
		return 0;
	}
	for (int i = 0; i < n / 2; i++) {
		for (int j = 0; j < n / 2 - i; j++) {
			cout << '*';
		}
		for (int j = 0; j < 2 * i + 1; j++) {
			cout << 'D';
		}
		for (int j = 0; j < n / 2 - i; j++) {
			cout << '*';
		}
		cout << endl;
	}
	for (int i = n / 2; i < n; i++) {
		for (int j = 0; j < i - n / 2; j++) {
			cout << '*';
		}
		for (int j = 0; j < n - 2 * (i - n / 2); j++) {
			cout << 'D';
		}
		for (int j = 0; j < i - n / 2; j++) {
			cout << '*';
		}
		cout << endl;
	}
	return 0;
}
