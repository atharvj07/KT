import bisect

r = list(map(int, input().split()))
c, p = zip(*[map(int, input().split()) for _ in range(r[0])])
k = int(input())
t = list(map(int, input().split()))
t.sort()

a = sorted(zip(p, c, range(r[0])))
re = 0
ans = []

for p, c, i in a:
    idx = bisect.bisect_left(t, c)
    if idx == len(t):
        continue
    re += p
    t[idx] -= c
    ans.append((i + 1, idx + 1))

print(len(ans), re)
print(*ans, sep="\n")
