#include <bits/stdc++.h>
using namespace std;

string solve(string s) {
  int n = s.size();
  for (int i = 0; i < n - 1; i++) {
    if (s[i] + 1 == s[i + 1]) {
      swap(s[i], s[i + 1]);
      i++;
    }
  }
  return s;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    string ans = solve(s);
    if (ans == s)
      cout << "No answer" << endl;
    else
      cout << ans << endl;
  }
  return 0;
}
