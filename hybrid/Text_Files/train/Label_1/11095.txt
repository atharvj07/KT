#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll INF = 1e18L + 5;
ll mod = 1e9 + 7;
#define rep(i, a, b) for (int i = a; i < b; i++)
#define all(a) (a).begin(), (a).end()
#define nl "\n"

ll ceil_div(ll a, ll b) {
  return (a + b - 1) / b;
}
ll pwr(ll a, ll b) {
  ll ans = 1;
  while (b) {
    if (b & 1) ans = (ans * a) % mod;
    a = (a * a) % mod;
    b >>= 1;
  }
  return ans % mod;
}
bool cmp(pair<ll, ll> &a, pair<ll, ll> &b) { return a.second < b.second; }
void solve() {
  int n;
  cin >> n;
  vector<int> p(n), s(n), q(n, 0), idx(n, 0);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
    idx[p[i] - 1] = i;
  }
  string ss;
  cin >> ss;
  for (int i = 0; i < n; i++) {
    s[i] = (ss[i] == '1');
  }
  int cnt = 1;
  for (int i = 0; i < n; i++) {
    if (!s[i]) {
      q[idx[i]] = cnt++;
    }
  }
  for (int i = 0; i < n; i++) {
    if (s[i]) {
      q[idx[i]] = cnt++;
    }
  }
  for (int x : q) cout << x << " ";
  cout << nl;
}
int main() {
  int tests;
  cin >> tests;
  while (tests--) solve();
  return 0;
}
