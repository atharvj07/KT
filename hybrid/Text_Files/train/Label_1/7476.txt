#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll M = (1LL<<15);
const ll N = 5005;
ll dp[N], a[N], cnt[M], freq[N];
int main() {
    ll n;
    cin >> n;
    for (ll i = 1; i <= n; i++) cin >> a[i], ++freq[a[i]];
    for (ll x = 1; x <= 5000; x++) {
        if (freq[x] > 0)
            for (ll i = 0; i < M; i++)
                cnt[i ^ x] += freq[x] - (i == x);
    }
    dp[0] = dp[1] = 0;
    for (ll i = 2; i <= n; i++) { dp[i] = max(dp[i - 1], dp[i - 2] + cnt[a[i]]); }
    cout << dp[n] << '\n';
    return 0;
}
