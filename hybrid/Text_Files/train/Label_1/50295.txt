import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int d = sc.nextInt();
        if (a + b + c + d <= 1) {
            System.out.println("NO");
            return;
        }
        if (a == 0 && b == 0 && c == 0) {
            System.out.println("NO");
            return;
        }
        if (d > 0 && (b == 0 || a == 0)) {
            System.out.println("NO");
            return;
        }
        StringBuilder sb = new StringBuilder();
        sb.append("YES\n");
        while (a > 0) {
            sb.append('0').append(' ');
            a--;
        }
        while (b > 0) {
            sb.append('1').append(' ');
            b--;
        }
        while (c > 0) {
            sb.append('2').append(' ');
            c--;
        }
        while (d > 0) {
            sb.append('3').append(' ');
            d--;
        }
        System.out.println(sb.toString());
    }
}
