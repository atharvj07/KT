#include <iostream>
#include <cmath>
using namespace std;
long long llabs(long long a) { return a < 0ll ? -a : a; }
int main() {
  long long r, d, x, y, c, r2, a;
  cin >> r >> d;
  r -= d;
  cin >> c;
  while (c--) {
    cin >> x >> y >> r2;
    a = x * x + y * y;
    a = a >= (r - r2) * (r - r2) && a <= (r + r2) * (r + r2);
    a = a || (llabs(x - r) <= r2 && a >= (r - r2) * (r - r2));
    a = a || (llabs(x + r) <= r2 && a >= (r - r2) * (r - r2));
    a = a || (llabs(y - r) <= r2 && a >= (r - r2) * (r - r2));
    a = a || (llabs(y + r) <= r2 && a >= (r - r2) * (r - r2));
    cout << a;
  }
  return 0;
}
