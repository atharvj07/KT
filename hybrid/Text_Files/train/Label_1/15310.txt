#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ll n, a[200005], sum = 0;
    cin >> n;
    for (ll i = 1; i <= n; i++) { cin >> a[i]; }
    ll cnt = 0;
    for (ll i = 1; i <= n; i++) {
        sum += a[i];
        if (sum < 0) {
            sum = 0;
            continue;
        }
        cnt++;
    }
    cout << cnt;
    return 0;
}
