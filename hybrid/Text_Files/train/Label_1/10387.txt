#include <iostream>
#include <map>
#include <vector>

using namespace std;

int main()
{
    map<char, int> suit;
    map<char, int> rank;
    vector<string> hand(5);
    string table;
    cin >> table;
    suit[table[1]]++;
    rank[table[0]]++;
    for (int i = 0; i < 5; i++)
    {
        string s;
        cin >> s;
        suit[s[1]]++;
        rank[s[0]]++;
        hand[i] = s;
    }
    for (int i = 0; i < 5; i++)
    {
        if (suit[table[1]] && suit[hand[i][1]])
        {
            cout << "YES";
            return 0;
        }
        if (rank[table[0]] && rank[hand[i][0]])
        {
            cout << "YES";
            return 0;
        }
    }
    cout << "NO";
    return 0;
}
