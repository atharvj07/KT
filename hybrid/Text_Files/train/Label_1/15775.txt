#include<bits/stdc++.h>
using namespace std;
bool leap (int y) {
    return (((y % 4 == 0) && (y % 100 != 0)) || (y % 400 == 0));
}
int main() {
    int y;
    cin>>y;
    int i=y+1;
    while(1) {
        if(6==((i + (leap(i) ? 2 : 1) + (i - 1) / 100 - (i - 1) / 400 + (i - 1) / 400 - (i - 1) / 100) % 7)) {
            cout<< i<<endl;
            return 0;
        }
        i++;
    }
}
