#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int v, a[10];
  cin >> v;
  for (int i = 1; i <= 9; i++) cin >> a[i];

  vector<pair<int, int>> p;
  for (int i = 1; i <= 9; i++) p.push_back({a[i], i});
  sort(p.begin(), p.end());

  string ans = "";
  int left = v;
  for (int i = p.size() - 1; i >= 0; i--) {
    int digit = p[i].second;
    int cost = p[i].first;
    if (left >= cost) {
      ans += (char)(digit + '0');
      left -= cost;
    }
  }
  if (ans == "") cout << -1;
  else cout << ans;
  cout << endl;
  return 0;
}
