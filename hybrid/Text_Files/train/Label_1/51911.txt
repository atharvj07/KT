import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] lines = new int[n][3];
        for (int i = 0; i < n; i++) {
            lines[i][0] = sc.nextInt();
            lines[i][1] = sc.nextInt();
            lines[i][2] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    int a1 = lines[i][0], b1 = lines[i][1], c1 = lines[i][2];
                    int a2 = lines[j][0], b2 = lines[j][1], c2 = lines[j][2];
                    int a3 = lines[k][0], b3 = lines[k][1], c3 = lines[k][2];
                    int x = (b2 * c3 - b3 * c2) * a1 + (b3 * c1 - b1 * c3) * a2 + (b1 * c2 - b2 * c1) * a3;
                    int y = (a2 * c3 - a3 * c2) * b1 + (a3 * c1 - a1 * c3) * b2 + (a1 * c2 - a2 * c1) * b3;
                    int z = (a2 * b3 - a3 * b2) * c1 + (a3 * b1 - a1 * b3) * c2 + (a1 * b2 - a2 * b1) * c3;
                    if (x * x + y * y - z * z == 0) ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
