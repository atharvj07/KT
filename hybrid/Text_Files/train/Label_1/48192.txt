import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int y = sc.nextInt();
            int max = 0;
            int maxBank = 0;
            for (int i = 1; i <= n; i++) {
                int b = sc.nextInt();
                int r = sc.nextInt();
                int t = sc.nextInt();
                double p = 100;
                if (t == 1) {
                    p += (r * y);
                } else {
                    for (int j = 0; j < y; j++) {
                        p *= (1 + r / 100.0);
                    }
                }
                if (p > max) {
                    max = (int) p;
                    maxBank = b;
                }
            }
            System.out.println(maxBank);
        }
    }
}
