#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    bool flag = true;
    for (int i = 1; i < n; i++) {
      if (a[i] < a[i - 1]) {
        if (i + 1 < n && a[i + 1] < a[i - 1]) {
          flag = false;
          break;
        }
      }
    }
    cout << (flag ? "YES" : "NO") << endl;
  }
  return 0;
}
