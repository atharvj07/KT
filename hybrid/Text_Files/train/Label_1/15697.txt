#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
    }
    bool sorted = true;
    for (int i = 1; i < n; i++) {
      if (a[i] < a[i - 1]) {
        sorted = false;
        break;
      }
    }
    if (sorted) {
      cout << "YES" << endl;
    } else {
      int ans = n * (n - 1) / 2 - 1;
      if (ans <= 0) {
        cout << "NO" << endl;
      } else {
        cout << "YES" << endl;
      }
    }
  }
  return 0;
}
