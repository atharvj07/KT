import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] inv = new int[n];
        for (int i = 0; i < n; i++) {
            inv[i] = sc.nextInt();
        }
        Arrays.sort(inv);
        int[] ans = new int[n];
        int cur = 1;
        for (int i = 0; i < n; i++) {
            if (inv[i] != cur) {
                ans[i] = cur++;
            } else {
                cur++;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
