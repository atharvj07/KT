#include <iostream>
#include <string>
using namespace std;
int main() {
  string a, b;
  cin >> a >> b;
  int ans = 0;
  for (int i = 0; i + b.size() <= a.size(); i++) {
    int cnt = 0;
    for (int j = 0; j < b.size(); j++) {
      if (a[i + j] != b[j]) cnt++;
    }
    if (cnt % 2 == 0) ans++;
  }
  cout << ans << endl;
}
