def mpow(a, b):
    if b == 0:
        return 1
    if b == 1:
        return a
    p = mpow(a, b // 2) % mod
    p = p * p % mod
    if b % 2:
        p = p * a % mod
    return p

mod = 998244353
n, m = map(int, input().split())
board = [list(input()) for i in range(n)]
row = []
for i in board:
    cnt = 0
    for c in i:
        if c == "*":
            cnt = 0
        else:
            cnt += 1
        row.append(cnt)
col = []
for i in range(m):
    cnt = 0
    for j in range(n):
        if board[j][i] == "*":
            cnt = 0
        else:
            cnt += 1
        col.append(cnt)
sum = 0
for x in row:
    sum += (mpow(2, x) - 1 + mod) % mod
for x in col:
    sum += (mpow(2, x) - 1 + mod) % mod
print(sum % mod)
