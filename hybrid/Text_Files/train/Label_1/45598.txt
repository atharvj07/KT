import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int[] arr = new int[2];
    for (int i = 0; i < arr.length; i++) {
      arr[i] = sc.nextInt();
    }

    int[] res = solve(arr);
    System.out.println(res[0]);
    StringBuilder sb = new StringBuilder();
    for (int i = 0; i < arr.length; i++) {
      if (res[i] == 1) {
        sb.append('o');
      } else {
        sb.append('x');
      }
    }
    System.out.println(sb);
  }

  static int[] solve(int[] arr) {
    int a = arr[0];
    int b = arr[1];
    if (a == 0) {
      return new int[] {-1 * b * b, b, a};
    }
    if (b == 0) {
      return new int[] {a * a, a, b};
    }
    int[] ans = new int[3];
    ans[0] = Math.max(a * a, -1 * b * b);
    if (ans[0] == a * a) {
      ans[1] = a;
      ans[2] = b;
    } else {
      ans[1] = b;
      ans[2] = a;
    }
    return ans;
  }
}
