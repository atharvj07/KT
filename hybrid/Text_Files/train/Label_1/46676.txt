import java.util.Scanner;

public class DormDryer {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int[][] intervals = new int[n][2];
        for (int i = 0; i < n; i++) {
            intervals[i][0] = scanner.nextInt();
            intervals[i][1] = scanner.nextInt();
        }

        int alexeyStart = intervals[0][0];
        int alexeyEnd = intervals[0][1];

        int availableLength = alexeyEnd - alexeyStart;

        for (int i = 1; i < n; i++) {
            int start = intervals[i][0];
            int end = intervals[i][1];

            if (start <= alexeyStart && end >= alexeyEnd) {
                availableLength = 0;
                break;
            } else if (start <= alexeyStart && end < alexeyEnd) {
                availableLength -= end - alexeyStart;
            } else if (start > alexeyStart && end >= alexeyEnd) {
                availableLength -= alexeyEnd - start;
            } else if (start > alexeyStart && end < alexeyEnd) {
                availableLength -= end - start;
            }
        }

        System.out.println(availableLength);
    }
}
