#include<bits/stdc++.h>
using namespace std;
const int maxn=505;
int n,a[maxn],b[maxn];
vector<int>l[15],ans[maxn];
int main(){
    int x,y,z;
    cin >> n;
    for (int i=0;i<n;++i) {
        cin >> a[i];
        l[a[i]].push_back(i+1);
    }
    for (int i=0;i<n;++i) {
        x=i+1, y=a[i], z=n-i;
        if(l[y].size()<z) {
            cout << "-1" << endl;
            return 0;
        }
        b[x]=l[y][z-1], l[y].erase(l[y].begin()+z-1);
        ans[a[i]].push_back(b[x]);
    }
    for (int i=1;i<=n;++i) {
        for (int j=0;j<ans[i].size();++j) cout << ans[i][j] << " ";
        cout << endl;
    }
    return 0;
}
