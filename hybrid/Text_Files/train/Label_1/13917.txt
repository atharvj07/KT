#include <bits/stdc++.h>
using namespace std;

int p[100001];

int main() {
  int n;
  cin >> n;
  vector<vector<int>> g(n);
  for (int i = 0; i < n - 1; i++) {
    int a, b;
    cin >> a >> b;
    a--, b--;
    g[a].push_back(b);
    g[b].push_back(a);
  }

  int l = 0;
  bool vis[n] = {0};
  function<void(int, int)> dfs = [&](int cur, int fa) {
    vis[cur] = true;
    for (const auto& to : g[cur]) {
      if (vis[to]) continue;
      p[to] = p[cur];
      dfs(to, cur);
      p[to]++;
      l = max(l, p[to]);
    }
  };
  dfs(0, 0);

  vector<vector<int>> ed(n);
  for (int i = 0; i < n; i++) {
    for (const auto& to : g[i]) {
      if (i < to) {
        ed[p[i]].push_back(p[to]);
        ed[p[to]].push_back(p[i]);
      }
    }
  }

  vector<bool> used(n);
  function<int(int, int)> dfs2 = [&](int cur, int fa) {
    for (const auto& x : ed[cur]) {
      if (used[x]) continue;
      used[x] = true;
      return x;
    }
    assert(false);
  };

  for (int i = 0; i < l; i++) {
    for (const auto& x : ed[i]) {
      if (used[x]) continue;
      used[x] = true;
      cout << dfs2(x, -1) << ' ';
    }
  }
  cout << '\n';
}
