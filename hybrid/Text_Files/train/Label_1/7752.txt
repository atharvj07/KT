#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    int mx = 0;
    for (int i = 1; i < n; i++) mx = max(mx, a[i] - a[i - 1]);
    for (int i = 1; i < n; i++) if (a[i] - a[i - 1] == mx) {
        cout << (mx + 1) / 2 << endl;
        cout << a[i - 1] << ' ' << mx << endl;
        return 0;
    }
    return 0;
}
