#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> cnt(n + 1);
  while (k--) {
    int m;
    cin >> m;
    while (m--) {
      int x;
      cin >> x;
      cnt[x]++;
    }
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    ans += cnt[i] > 1 ? cnt[i] - 1 : 0;
  }
  cout << ans << endl;
  return 0;
}
