import java.util.Scanner;

public class Main {
  public static void main(String[] args) throws Exception {
    Scanner sc = new Scanner(System.in);
    int[] days = {0, 31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30};
    String[] weeks = {"Thursday", "Friday", "Saturday", "Sunday", "Monday", "Tuesday", "Wednesday"};
    while (true) {
      int m = sc.nextInt(), d = sc.nextInt();
      if (m == 0 && d == 0) break;
      int daysBefore = 0;
      for (int i = 1; i < m; i++) {
        daysBefore += days[i];
      }
      daysBefore += d - 1;
      System.out.println(weeks[daysBefore % 7]);
    }
  }
}
