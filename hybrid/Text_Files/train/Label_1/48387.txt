import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
    static class TShirt implements Comparable<TShirt> {
        int price;
        int quality;

        public TShirt(int price, int quality) {
            this.price = price;
            this.quality = quality;
        }

        public int compareTo(TShirt other) {
            return this.quality - other.quality;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());

        TShirt[] tshirts = new TShirt[n];
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            tshirts[i] = new TShirt(Integer.parseInt(line[0]), Integer.parseInt(line[1]));
        }

        Arrays.sort(tshirts);

        int k = Integer.parseInt(br.readLine());

        int[] budgets = new int[k];
        String[] line = br.readLine().split(" ");
        for (int i = 0; i < k; i++) {
            budgets[i] = Integer.parseInt(line[i]);
        }

        for (int i = 0; i < k; i++) {
            int budget = budgets[i];
            int count = 0;
            for (int j = 0; j < n; j++) {
                if (budget >= tshirts[j].price) {
                    budget -= tshirts[j].price;
                    count++;
                } else {
                    break;
                }
            }
            System.out.print(count + " ");
        }
        System.out.println();
    }
}
