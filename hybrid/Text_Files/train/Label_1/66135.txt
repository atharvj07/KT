def min_moves(k1, k2, k3, a1, a2, a3):
    n = k1 + k2 + k3
    pos = [0] * (n + 1)
    for i, a in enumerate(a1, start=1):
        pos[a] = i
    for i, a in enumerate(a2, start=1):
        pos[a] = i + k1
    for i, a in enumerate(a3, start=1):
        pos[a] = i + k1 + k2
    
    dp = [[float('inf')] * 4 for _ in range(n + 1)]
    dp[0][0] = 0
    
    for i in range(1, n + 1):
        dp[i][0] = dp[i-1][0] + (pos[i] > k1)
        dp[i][1] = min(dp[i-1][0], dp[i-1][1]) + (pos[i] <= k1 or pos[i] > k1 + k2)
        dp[i][2] = min(dp[i-1][0], dp[i-1][1], dp[i-1][2]) + (pos[i] <= k1 + k2)
        dp[i][3] = min(dp[i-1][0], dp[i-1][1], dp[i-1][2], dp[i-1][3])
    
    print(dp[n][3])

k1, k2, k3 = map(int, input().split())
a1 = list(map(int, input().split()))
a2 = list(map(int, input().split()))
a3 = list(map(int, input().split()))
min_moves(k1, k2, k3, a1, a2, a3)
