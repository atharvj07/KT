#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  long long pre = 0, suf = 0;
  for (int i = 0; i < n - 1; i++) suf += a[i];
  for (int i = 0; i < n; i++) {
    pre += a[i];
    suf -= a[i];
    if (pre == suf) {
      cout << "YES" << endl;
      return 0;
    }
  }
  cout << "NO" << endl;
}
