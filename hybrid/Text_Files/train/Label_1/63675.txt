def candies_eaten(n):
    left, right = 1, n
    while left < right:
        k = (left + right) // 2
        eaten, remain = 0, n
        while remain > 0:
            vasya_eats = min(k, remain)
            remain -= vasya_eats
            eaten += vasya_eats
            petya_eats = remain // 10
            remain -= petya_eats
            if eaten >= n // 2:
                break
        if eaten >= n // 2:
            right = k
        else:
            left = k + 1
    return left

n = int(input().strip())
print(candies_eaten(n))
