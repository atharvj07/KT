n, m = map(int, input().split())
a = [list(map(int, input().split())) for i in range(n)]
c = [[0] * m for i in range(n)]
ans = 0
for i in range(n):
    for j in range(m):
        if i == 0 or j == 0:
            c[i][j] = 1
        elif a[i][j] != a[i - 1][j] and a[i][j] != a[i][j - 1]:
            c[i][j] = 1
        ans += c[i][j]
print(ans)
