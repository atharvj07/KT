#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    unordered_map<int, int> m;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        if (x != 0) {
            if (m[x] == 0) {
                m[x] = 1;
                ans++;
            } else {
                cout << "-1" << endl;
                return 0;
            }
        }
    }
    cout << ans / 2 << endl;
    return 0;
}
