import sys
input=sys.stdin.readline

n, m, k=map(int, input().split())
s=input()
t=input()
dp=[[[-1]*11 for _ in range(m+1)] for _ in range(n+1)]

def solve(i, j, rem):
    if i==n and j==m and rem==0:
        return 0
    if i==n or j==m or rem==0:
        return -1e9
    if dp[i][j][rem]!=-1:
        return dp[i][j][rem]
    dp[i][j][rem]=max(solve(i+1, j, rem), solve(i, j+1, rem))
    if s[i]==t[j]:
        dp[i][j][rem]=max(dp[i][j][rem], solve(i+1, j+1, rem-1)+1)
    return dp[i][j][rem]

ans=solve(0, 0, k)
if ans==-1e9:
    print(0)
else:
    print(ans)
