import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int q = input.nextInt();
        for (int i = 0; i < q; i++) {
            int n = input.nextInt();
            String s = input.next();
            String t = input.next();
            boolean possible = true;
            for (int j = 0; j < n; j++) {
                if (s.charAt(j) != t.charAt(j)) {
                    if ((s.charAt(j) == 'a' && t.charAt(j) == 'b') || (s.charAt(j) == 'b' && t.charAt(j) == 'a')) {
                        possible = true;
                    } else {
                        possible = false;
                        break;
                    }
                }
            }
            if (possible) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
