for _ in range(int(input())):
    n = int(input())
    sl = [list(map(int, input().split())) for i in range(n)]
    tm = list(map(int, input().split()))
    now, ans = sl[0][0], sl[0][1]
    for i in range(1, n):
        a, b = sl[i]
        if (b - now) < tm[i]:
            ans += tm[i] + now - b
        ans += a - max(now, b - tm[i])
        now = b
    print(ans)
