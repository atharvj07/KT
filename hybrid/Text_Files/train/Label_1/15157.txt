#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for(int i=0; i<n; i++)

struct student {
  int id, a, b;
  bool operator<(const student& o) const {
    return a + b < o.a + o.b;
  }
};

int main() {
  int n, p, s, a, b; cin >> n >> p >> s;
  student stu[n]; rep(i, n) { cin >> a >> b; stu[i]={i+1, a, b}; }
  sort(stu, stu+n);
  cout << stu[p+s-1].a + stu[p+s-1].b << endl;
  rep(i, p) cout << stu[i].id << " ";
  cout << endl;
  rep(i, s) cout << stu[p+i].id << " ";
  cout << endl;
  return 0;
}
