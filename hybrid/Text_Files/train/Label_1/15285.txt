#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int MAXN = 2e5 + 5;
struct node{
    ll val;
    int id;
    node(){}
    node(ll x, int y){
        val = x; id = y;
    }
};
node seg[4 * MAXN];
ll ar[MAXN];
int n, m;
void add(int l, int r, int id, int s, int e, ll x){
    if(e < l || s > r) return;
    if(s <= l && r <= e){
        seg[id].val += x;
        return;
    }
    int mid = (l + r) / 2;
    add(l, mid, id * 2, s, e, x);
    add(mid + 1, r, id * 2 + 1, s, e, x);
    seg[id].val = max(seg[id * 2].val, seg[id * 2 + 1].val);
}
node query(int l, int r, int id, int s){
    if(s < l || s > r || l > r) return node(LLONG_MIN, -1);
    if(s <= l && r <= s) return seg[id];
    int mid = (l + r) / 2;
    node a = query(l, mid, id * 2, s);
    node b = query(mid + 1, r, id * 2 + 1, s);
    if(a.val == b.val) return a.id < b.id ? a : b;
    return a.val > b.val ? a : b;
}

int main(){
    cin >> n;
    for(int i = 1; i <= n; i++){
        cin >> ar[i];
        add(1, n, 1, i, i, ar[i]);
    }
    cin >> m;
    for(int i = 0; i < m; i++){
        int k, pos;
        cin >> k >> pos;
        node tmp = query(1, n, 1, pos);
        cout << tmp.val << '\n';
        add(1, n, 1, tmp.id, tmp.id, -ar[tmp.id]);
    }
    return 0;
}
