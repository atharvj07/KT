#include <iostream>

using namespace std;

int main() {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    double Misha = max((3.0 * a) / 10, a - (0.004 * a) * c);
    double Vasya = max((3.0 * b) / 10, b - (0.004 * b) * d);
    if (Misha > Vasya)
        cout << "Misha";
    else if (Vasya > Misha)
        cout << "Vasya";
    else
        cout << "Tie";
    return 0;
}
