#include <iostream>
using namespace std;
const int MAX = 1e5 + 1, INF = 1e9 + 1;

int n, p, a[MAX], dp[MAX];

int solve() {
  if (p >= n) return 1;
  int &res = dp[p];
  if (res != -1) return res;
  res = 0;
  for (int i = min(n, p + a[p] * p); i >= p; i--)
    res = max(res, solve() + i - p);
  return res;
}

int main() {
  cin >> n >> p;
  for (int i = 1; i <= n; i++) cin >> a[i];
  fill(dp, dp + n + 1, -1);
  cout << solve() << endl;
}
