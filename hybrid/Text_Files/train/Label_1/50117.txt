import java.util.*;
import java.math.*;

class Solution {
    static final int MOD = 1000000007;
    
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String expression = scanner.nextLine();
        int n = scanner.nextInt();
        int[][] values = new int[n][5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                values[i][j] = scanner.nextInt();
            }
        }
        scanner.close();
        int result = countWays(expression, values);
        System.out.println(result);
    }
    
    static int countWays(String expression, int[][] values) {
        int[][] dp = new int[expression.length() + 1][1 << values.length];
        dp[0][0] = 1;
        for (int i = 1; i <= expression.length(); i++) {
            char c = expression.charAt(i - 1);
            if (Character.isLetter(c)) {
                for (int j = 0; j < (1 << values.length); j++) {
                    for (int k = 0; k < values.length; k++) {
                        if ((j & (1 << k)) != 0) {
                            int newJ = j & ~(1 << k);
                            if (c == 'A' || c == 'a' && values[k][0] == 1 || c == 'B' || c == 'b' && values[k][1] == 1
                                    || c == 'C' || c == 'c' && values[k][2] == 1 || c == 'D' || c == 'd' && values[k][3] == 1) {
                                dp[i][newJ] = (dp[i][newJ] + dp[i - 1][j]) % MOD;
                            }
                        }
                    }
                }
            } else if (c == '?') {
                for (int j = 0; j < (1 << values.length); j++) {
                    for (int k = 0; k < values.length; k++) {
                        if ((j & (1 << k)) == 0) {
                            int newJ = j | (1 << k);
                            dp[i][newJ] = (dp[i][newJ] + dp[i - 1][j]) % MOD;
                        }
                    }
                }
            } else if (c == '&') {
                for (int j = 0; j < (1 << values.length); j++) {
                    for (int k = 0; k < values.length; k++) {
                        if ((j & (1 << k)) != 0) {
                            int newJ = j & ~(1 << k);
                            if (values[k][4] == 0) {
                                dp[i][newJ] = (dp[i][newJ] + dp[i - 1][j]) % MOD;
                            }
                        }
                    }
                }
            } else if (c == '|') {
                for (int j = 0; j < (1 << values.length); j++) {
                    for (int k = 0; k < values.length; k++) {
                        if ((j & (1 << k)) == 0) {
                            int newJ = j | (1 << k);
                            if (values[k][4] == 1) {
                                dp[i][newJ] = (dp[i][newJ] + dp[i - 1][j]) % MOD;
                            }
                        }
                    }
                }
            }
        }
        return dp[expression.length()][0];
    }
}
