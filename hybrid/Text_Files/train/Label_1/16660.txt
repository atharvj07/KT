#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define vi vector<int>

int main() {
    int n, sum = 0;
    cin >> n;
    vi a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    for (int i = 0; i < n; i++) {
        int x = min(a[i], b[i] / 2);
        int y = b[i] - x;
        sum += x * y;
    }
    cout << sum << endl;
    return 0;
}
