import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int k = Integer.parseInt(input[1]);
        int[] a = new int[n];
        String[] aInput = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(aInput[i]);
        }
        System.out.println(maxTotalValue(a, n, k));
    }

    private static int maxTotalValue(int[] a, int n, int k) {
        int[] lastPos = new int[n + 1];
        Arrays.fill(lastPos, -1);
        int[] dp = new int[n + 1];
        int[] maxType = new int[n + 1];
        int maxValue = 0;
        for (int i = 0; i < n; i++) {
            if (lastPos[a[i]] != -1) {
                dp[i + 1] = Math.max(dp[i + 1], dp[lastPos[a[i]]] + 1);
            } else {
                dp[i + 1] = dp[i] + 1;
            }
            lastPos[a[i]] = i;
            maxType[i + 1] = Math.max(maxType[i], dp[i + 1]);
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i + j >= n) {
                    break;
                }
                maxValue = Math.max(maxValue, maxType[i] + maxType[j + i + 1]);
            }
        }
        if (maxValue <= k) {
            return maxValue;
        }
        int[] boxes = new int[k];
        int boxIndex = 0;
        int remaining = n;
        for (int i = n - 1; i >= 0; i--) {
            if (remaining <= 0) {
                break;
            }
            if (maxType[i] > 0 && remaining - maxType[i] >= 0) {
                boxes[boxIndex] = maxType[i];
                boxIndex++;
                remaining -= maxType[i];
            }
        }
        Arrays.sort(boxes);
        int sum = 0;
        for (int i = 0; i < k; i++) {
            sum += boxes[i];
        }
        return sum;
    }
}
