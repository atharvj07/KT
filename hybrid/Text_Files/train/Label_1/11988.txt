#include<bits/stdc++.h>
using namespace std;
int main(){
	int t,n;
	long long int a[200000];
	cin>>t;
	while(t--){
		cin>>n;
		for(int i=0;i<n;i++){
			cin>>a[i];
		}
		deque<long long int> dq;
		bool rev=false;
		int ans=0;
		for(int i=0;i<n;i++){
			if(rev){
				while(!dq.empty() && a[i]<=dq.front()){
					ans++;
					dq.pop_front();
				}
				dq.push_front(a[i]);
			}
			else{
				while(!dq.empty() && a[i]<=dq.back()){
					ans++;
					dq.pop_back();
				}
				dq.push_back(a[i]);
			}
			rev=!rev;
		}
		cout<<ans<<endl;
	}
	return 0;
}
