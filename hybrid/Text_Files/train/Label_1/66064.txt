def s(x):
    return sum([int(d) for d in str(x)])

def find_solutions(a, b, c):
    solutions = []
    for x in range(1, 82):  # s(x) can be at most 9*9=81 for x<10^9
        n = b * (x ** a) + c
        if n > 0 and n < 10**9 and s(n) == x:
            solutions.append(n)
    solutions.sort()
    return solutions

a, b, c = map(int, input().split())
solutions = find_solutions(a, b, c)
print(len(solutions))
if len(solutions) > 0:
    print(' '.join(map(str, solutions)))
