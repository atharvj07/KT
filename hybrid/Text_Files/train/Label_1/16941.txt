#include <iostream>

using namespace std;

int main() {
  int a[24];
  for (int i = 0; i < 24; i++) { cin >> a[i]; }
  bool ok = true;
  for (int i = 0; i < 6; i++) {
    int cnt = 0;
    for (int j = 0; j < 4; j++) {
      cnt += (a[i * 4 + j] == a[i * 4]);
    }
    ok &= (cnt == 1 || cnt == 4);
  }
  cout << (ok ? "YES" : "NO") << endl;
  return 0;
}
