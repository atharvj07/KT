#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  int f1 = -1, f2 = -1, len = 0;
  for (int i = 0; i < n; i++) {
    if (f1 == -1 || a[i] == f1 + 1) {
      f1 = a[i];
      len++;
    } else if (f2 == -1 || a[i] == f2 + 2) {
      f2 = a[i];
      len++;
    }
  }
  cout << len << endl;
  return 0;
}
