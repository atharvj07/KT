def solve():
    t = int(input())
    for _ in range(t):
        n, m = [int(x) for x in input().split()]
        tiles = []
        for _ in range(n):
            a, b = [int(x) for x in input().split()]
            c, d = [int(x) for x in input().split()]
            tiles.append([a, b, c, d])
        L = n
        if n % 2 == 1 and m % 2 == 1:
            L += 1
        if L > m:
            print("NO")
            continue
        x = m // 2
        y = m % 2
        if L == m:
            print("YES")
            continue
        a = L + y
        b = m - a
        if a % 2 == b % 2 and a <= m and a >= 1:
            print("YES")
            continue
        print("NO")


solve()
