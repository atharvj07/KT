#include <iostream>
#include <vector>
using namespace std;
int main() {
    string s, t;
    getline(cin, s);
    getline(cin, t);
    int yay = 0, whoops = 0;
    for (int i = 0; i < s.size(); i++) {
        bool yay_ = false, whoops_ = false;
        for (int j = 0; j < t.size(); j++) {
            if (s[i] == t[j]) {
                if (isupper(s[i]) == isupper(t[j])) {
                    yay_ = true;
                    whoops_ = false;
                    break;
                } else {
                    yay_ = false;
                    whoops_ = true;
                    break;
                }
            }
        }
        if (yay_) yay++;
        if (whoops_) whoops++;
    }
    cout << yay << ' ' << whoops;
}
