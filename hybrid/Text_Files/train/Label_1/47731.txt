import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int[] times = new int[n];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < i; j++) {
                times[j] += n - i;
            }
            times[i] = 1;
        }
        for (int i = 0; i < n; i++) {
            k -= times[i];
            if (k <= 0) {
                System.out.println(arr[i]);
                break;
            }
        }
    }
}
