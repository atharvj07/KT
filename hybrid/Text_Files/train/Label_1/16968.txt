#include <array>
#include <algorithm>
#include <cstring>
#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 100;

long long a[MAXN];

struct OptSubseq {
  long long v = 0;
  OptSubseq* next = nullptr;
};

OptSubseq* dp[MAXN + 1][MAXN + 1];

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> a[i];

  int m;
  cin >> m;

  memset(dp, 0, sizeof(dp));

  for (int i = 0; i <= n; ++i) {
    for (int j = 0; j <= n; ++j) {
      OptSubseq* cur = nullptr;
      if (dp[i + 1][j]) cur = dp[i + 1][j];
      if (!cur || a[i] > cur->v) {
        cur = new OptSubseq();
        cur->v = a[i];
        cur->next = dp[i + 1][j];
        dp[i + 1][j] = cur;
      }
      if (j > 0) {
        if (dp[i][j - 1]) cur = dp[i][j - 1];
        if (!cur || a[i] > cur->v) {
          cur = new OptSubseq();
          cur->v = a[i];
          cur->next = dp[i][j - 1];
          dp[i][j] = cur;
        }
      }
    }
  }

  for (int i = 0; i < m; ++i) {
    int k, p;
    cin >> k >> p;
    p = min(p, k);
    vector<OptSubseq*> v;
    for (OptSubseq* s = dp[k][p]; s; s = s->next) v.push_back(s);
    cout << v[v.size() - 1]->v << endl;
  }
  return 0;
}
