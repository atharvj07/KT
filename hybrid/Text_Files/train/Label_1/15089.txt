#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  vector<pair<int, int>> v(q);
  for (int i = 0; i < q; i++) { cin >> v[i].first >> v[i].second; }
  sort(v.begin(), v.end());
  int ans = 0, r = 0, cnt = 0;
  for (int l = 1; l <= n; l++) {
    while (r < q && v[r].first <= l) {
      if (v[r].first <= l && v[r].second >= l) cnt++;
      r++;
    }
    if (cnt >= q - 2) ans += n - l + 1;
    cnt = 0;
  }
  cout << ans << endl;
  return 0;
}
