from collections import defaultdict

n, k, a, b, q = map(int, input().split())
orders = defaultdict(int)
for _ in range(q):
    t, *x = map(int, input().split())
    if t == 1:
        orders[x[0]] += x[1]
    else:
        cur = sum(min(b, orders[i]) for i in range(x[0], x[0] + n - k + 1)) + sum(min(a, orders[i]) for i in range(x[0] + n - k + 1, n + 1))
        print(cur)
