#include<bits/stdc++.h>
using namespace std;
int main(){
  int n; cin >> n;
  vector<string>bus(n);
  for(int i = 0; i < n; ++i) cin >> bus[i];
  bool ok = false;
  for(int i = 0; i < n; ++i){
    for(int j = 0; j < 4; j += 2){
      if(bus[i][j] == 'O' && bus[i][j + 1] == 'O'){
        bus[i][j] = bus[i][j + 1] = '+';
        ok = true;
        break;
      }
    }
    if(ok) break;
  }
  if(ok) cout << "YES\n";
  else cout << "NO\n";
  for(int i = 0; i < n; ++i) cout << bus[i] << "\n";
}
