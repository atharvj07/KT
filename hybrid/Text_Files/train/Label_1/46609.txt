import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        for (int t = 0; t < T; t++) {
            int f = sc.nextInt();
            int[] a = new int[8];
            int[] b = new int[8];
            for (int i = 0; i < 8; i++) a[i] = sc.nextInt();
            for (int i = 0; i < 8; i++) b[i] = sc.nextInt();
            boolean winAlice = false;  
            boolean winBob = false;
            boolean deal = true;  
            for (int i = 0; i < 8; i++) {
                if (f == 0) {  
                    for (int j = 0; j < 8; j++) {
                        if (a[i] != 0 && b[j] != 0 && (a[i] * b[j]) % 5 != 0) {
                            winAlice = true;
                            f = 1;  
                            a[i] = (a[i] + b[j]) % 5;
                            b[j] = 0;
                            break;
                        }
                    }
                } else {
                    for (int j = 0; j < 8; j++) {
                        if (b[i] != 0 && a[j] != 0 && (b[i] * a[j]) % 5 != 0) {
                            winBob = true;
                            f = 0;  
                            b[i] = (b[i] + a[j]) % 5;
                            a[j] = 0;
                            break;
                        }
                    }
                }
            }
            for (int i = 0; i < 8; i++) {
                if (a[i] != 0 || b[i] != 0) {
                    deal = false;
                    break;
                }
            }
            if (winAlice) System.out.println("Alice");
            else if (winBob) System.out.println("Bob");
            else if (deal) System.out.println("Deal");
            else System.out.println("Deal");
        }
    }
}
