#include <iostream>
#include <unordered_set>
using namespace std;

int main() {
  string s;
  int n;
  cin >> s >> n;
  unordered_set<string> words;
  for (int i = 0; i < n; i++) {
    string w;
    cin >> w;
    words.insert(w);
  }
  for (int i = 0; i < s.size(); i++) {
    if (words.find(s.substr(i, 2)) != words.end()) {
      cout << "YES" << endl;
      return 0;
    }
  }
  cout << "NO" << endl;
  return 0;
}
