t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    matrix = [list(map(int, input().split())) for _ in range(n)]
    changes = 0
    for i in range((n + 1) // 2):
        for j in range((m + 1) // 2):
            opposite_i, opposite_j = n - i - 1, m - j - 1
            values = [matrix[i][j]]
            if (i != opposite_i):
                values.append(matrix[opposite_i][j])
            if (j != opposite_j):
                values.append(matrix[i][opposite_j])
            if (i != opposite_i and j != opposite_j):
                values.append(matrix[opposite_i][opposite_j])
            changes += min(values.count(0), values.count(1))
    print(changes)
