#include <bits/stdc++.h>
using namespace std;

typedef long long LL;

const LL mod = 1e9 + 7;

#define endl '\n'
#define all(a) a.begin(), a.end()
#define rall(a) a.rbegin(), a.rend()
#define pb push_back
#define mp make_pair
#define f first
#define s second

map<LL, set<LL>> p;
map<LL, LL> dp;

int main() {
    LL m, k;
    cin >> m >> k;

    for (LL i = 0; i < m; i++) {
        LL a, b;
        cin >> a >> b;
        p[a].insert(b);
        p[b].insert(a);
    }

    for (auto it = p.begin(); it != p.end(); it++) {
        LL id = it->f;
        set<LL> f = it->s;

        for (auto it = f.begin(); it != f.end(); it++) {
            dp[id] += p[*it].size();
        }

        dp[id] -= p[id].size();
        dp[id] /= 2;

        if (k * dp[id] < 100 * (LL)f.size()) {
            dp[id] = 0;
        }
    }

    vector<pair<LL, LL>> v;

    for (auto it = dp.begin(); it != dp.end(); it++) {
        v.pb({it->s, it->f});
    }

    sort(all(v));

    for (auto it = v.begin(); it != v.end(); it++) {
        set<LL> tmp;
        LL id = it->s;
        LL cnt = it->f;

        for (auto it = p[id].begin(); it != p[id].end(); it++) {
            if (dp[*it] > 0) {
                tmp.insert(*it);
            }
        }

        cout << id << ": " << tmp.size() << " ";

        for (auto it = tmp.begin(); it != tmp.end(); it++) {
            cout << *it << " ";
        }

        cout << endl;
    }

    return 0;
}
