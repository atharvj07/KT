#include <bits/stdc++.h>
using namespace std;
#define LL long long
const LL mod = 1e9 + 7;
const int maxn = 1e5 + 5;
vector<int> adj[maxn];
LL x[maxn], ans[maxn];
int par[maxn];

LL gcd(LL a, LL b) { return !b ? a : gcd(b, a % b); }
void dfs(int u, int p) {
    par[u] = p;
    for (auto v : adj[u]) {
        if (v == p) continue;
        dfs(v, u);
        ans[u] = (ans[u] + ans[v] + gcd(x[u], x[v])) % mod;
    }
}
int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> x[i]; }
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, 1);
    LL tot = 0;
    for (int i = 1; i <= n; i++) { tot = (tot + ans[i]) % mod; }
    cout << tot << endl;
}
