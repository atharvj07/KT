n=int(input())
a=[None]*n
b=[None]*n
h=[None]*n
for i in range(n):
    a[i],b[i],h[i]=map(int,input().split())
order=sorted(range(n),key=lambda i:h[i],reverse=True)
ans,curouter,curinner=0,1e9,1e9
for ring in order:
    if a[ring]<curinner and b[ring]<curouter:
        ans+=h[ring]
        curouter=b[ring]
        curinner=a[ring]
print(ans)
