#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mxn = 205;
int dp[mxn][mxn], a[mxn];
int main() {
    int n, k, x;
    cin >> n >> k >> x;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 0; i <= n; i++)
        for (int j = 0; j <= x; j++)
            dp[i][j] = -1e9;
    dp[n + 1][x] = 0;
    for (int i = n; i >= 1; i--) {
        for (int j = x; j >= 0; j--) {
            dp[i][j] = max(dp[i][j], dp[i + k][j]);
            dp[i][j] = max(dp[i][j], dp[i + k][j + 1] + a[i]);
        }
    }
    cout << (dp[1][0] >= 0 ? dp[1][0] : -1) << "\n";
    return 0;
}
