#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<ll>
const ll mod = 998244353;
const ll MX = 3e5 + 1;

int main() {
    ll n, m;
    cin >> n >> m;
    vi h(n);
    vector<pair<ll, ll>> data;
    for (ll i = 0; i < m; i++) {
        ll day, hei;
        cin >> day >> hei;
        day--, h[day] = hei;
        data.push_back({day, hei});
    }
    h[0] = h[n - 1] = 1e9;
    multiset<ll> heights;
    heights.insert(h[0]);
    bool flg = 1;
    for (auto day : data) {
        ll x = day.first, y = day.second;
        auto it = heights.lower_bound(y);
        if (it == heights.begin() || *it != y) {
            if (*prev(it) + 1 != y) {
                flg = 0;
                break;
            }
        } else {
            if (*next(it) != y) {
                flg = 0;
                break;
            }
        }
        heights.insert(y);
        heights.erase(heights.find(h[x]));
        h[x] = y;
    }
    ll mx = 0;
    for (ll i = 0; i < n; i++) mx = max(mx, h[i]);
    if (!flg)
        cout << "IMPOSSIBLE" << "\n";
    else
        cout << mx << "\n";
    return 0;
}
