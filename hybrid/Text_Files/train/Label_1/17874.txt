#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> groups(n);
  for (int i = 0; i < n; i++) {
    std::cin >> groups[i];
  }

  int ans = 0;
  for (int i = 0; i < n;) {
    if (groups[i] == 1) {
      if (i + 2 < n && groups[i + 1] == 1 && groups[i + 2] == 1) {
        ans++;
        i += 3;
      } else {
        i++;
      }
    } else {
      if (i + 1 < n && groups[i + 1] == 2) {
        ans++;
        i += 2;
      } else {
        i++;
      }
    }
  }

  std::cout << ans << std::endl;

  return 0;
}
