n = int(input().strip())
a = list(map(int, input().strip().split()))

coins = 0
negative_count = 0
zero_count = 0

for number in a:
    if number < 0:
        coins += (-1 - number)
        negative_count += 1
    elif number > 0:
        coins += (number - 1)
    else: # number == 0
        coins += 1
        zero_count += 1

if negative_count % 2 != 0 and zero_count == 0:
    coins += 2

print(coins)
