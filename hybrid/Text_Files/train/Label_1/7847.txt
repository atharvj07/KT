#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> s(n);
  for (int i = 0; i < n; ++i) cin >> s[i];
  int mx = s.back();
  int ans = 0;
  for (int i = n - 1, cnt = 0; i >= 0; --i) {
    int rem = k - cnt;
    if (rem * 2 >= i + 1) {
      ans = max(ans, s[i] + mx);
    } else {
      int add = max(1, (i + 1 - 2 * rem) / 2);
      ans = max(ans, s[i + add - 1] + s[min(i + add, n - 1)]);
      cnt += add;
      i -= add - 1;
    }
  }
  cout << ans << endl;
  return 0;
}
