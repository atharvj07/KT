#include <iostream>

using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        string s; cin >> s;
        int ans = 0;
        for (int i = 0; i + 1 < n; i++) {
            ans += (s[i] != s[i + 1]);
        }
        cout << ans / 2 << endl;
    }
    return 0;
}
