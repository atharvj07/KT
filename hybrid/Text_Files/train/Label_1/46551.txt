import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = reader.readLine().split(" ");
        int n = Integer.parseInt(nk[0]), k = Integer.parseInt(nk[1]);
        int[] a = new int[n];
        String[] aInput = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) a[i] = Integer.parseInt(aInput[i]);

        int[] z = new int[3 * n + 2];
        z[n + k] = z[n + k + 1] = z[n + k + 2] = -1;
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] > 0) {
                z[i + k + 2] = z[i + k + 1] = z[i + k];
                z[i + k + 1] += 1;
            } else {
                z[i + k + 2] = Math.max(z[i + k + 1], z[i + k + 2]);
                z[i + k + 1] = z[i + k] + 1;
            }
        }

        int res = 0;
        for (int i = 0; i < n + k + 1; i++)
            res = Math.max(res, z[i]);

        System.out.println(res);

        for (int i = 0, p = n + k; i < n; i++) {
            if (a[i] > 0) {
                p--;
                System.out.print(1 + " ");
            } else if (p > 0 && (z[p + 2] < z[p + 1] || (z[p + 2] == z[p + 1] && z[p + 1] == res))) {
                p--;
                System.out.print(1 + " ");
            } else System.out.print(0 + " ");
        }
    }
}
