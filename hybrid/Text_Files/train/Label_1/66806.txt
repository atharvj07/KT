n = int(input())
a = []
for _ in range(n):
    a.append([int(i) for i in input().split()[1:]])
b = [0] * n
c = [[0] * 1000001 for _ in range(n)]
for i, x in enumerate(a):
    for y in x:
        c[i][y] += 1
        if not b[i] and y != 0:
            b[i] = y
for i in range(n):
    for j in range(n):
        if i == j:
            for y in range(b[i], 1000001):
                if c[i][y]:
                    break
            for x in range(max(b[i], 1), y):
                if c[i][x]:
                    b[i] = x
                    break
for i in range(n):
    for j in range(n):
        if i != j and c[i][b[j]] and b[i] < b[j]:
            b[i] = b[j]
print(sum([x != 0 for x in b]))
