#include <bits/stdc++.h>
using namespace std;

void input (int &n, int &k, string &s) {
    cin >> n >> k;
    cin >> s;
}

void solve (int n, int k, string s) {
    for (int i = 0; i < n - 1 && k; i++) {
        if (s[i] > s[i + 1]) {
            swap (s[i], s[i + 1]);
            k--;
        }
    }
    cout << s << '\n';
}

int main () {
    int q;
    cin >> q;
    while (q--) {
        int n, k;
        string s;
        input (n, k, s);
        solve (n, k, s);
    }
    return 0;
}
