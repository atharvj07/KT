#include <bits/stdc++.h>
#define ll long long
#define pii pair<ll, ll>
using namespace std;

const int mxN = 1e6+1;
ll f[mxN];

int main() {
    ll n; cin >> n;
    vector<ll> q;
    for (ll i = 0; i < n; i++) {
        q.push_back(i+1);
        f[i+1] = i+1;
    }
    ll p = n;

    while (p) {
        ll g = 0;
        for (auto x : q) {
            g = __gcd(g, x);
        }
        cout << g << ' ';
        ll idx = -1;
        for (ll i = 0; i < q.size(); i++) {
            if (q[i]%g==0) {
                idx = i; break;
            }
        }
        swap(q[idx], q.back());
        q.pop_back();
        p--;
    }
}
