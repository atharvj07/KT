#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, l, r, ql, qr;
  cin >> n >> l >> r >> ql >> qr;
  vector<int> w(n);
  for (int i = 0; i < n; i++) { cin >> w[i]; }
  if (l * ql > r * qr) swap(l, r), swap(ql, qr);
  long long ans = 0;
  for (int i = n - 1; i >= 0; i -= 2) ans += r * w[i];
  for (int i = 0; i < n; i += 2) ans += l * w[i];
  cout << ans << endl;
  return 0;
}
