#include <bits/stdc++.h>

using namespace std;

long long n, k, ans, l, r, m;
long long wave[2001][3];

int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; ++i) {
    cin >> l >> r >> m;
    wave[i][0] = l;
    wave[i][1] = r;
    wave[i][2] = m;
  }
  l = 1e9;
  for (int i = n; i >= 1; --i) {
    m = wave[i][2];
    r = wave[i][1];
    if (m > k) {
      cout << -1;
      return 0;
    }
    l = min(l, r - m + 1);
    ans += (l - 1 + k) / k * k;
    l = r + 1;
  }
  cout << ans;
  return 0;
}
