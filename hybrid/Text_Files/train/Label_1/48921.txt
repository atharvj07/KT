import java.util.Scanner;

public class IvanAtTheUniversity {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] timetable = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                timetable[i][j] = line.charAt(j) - '0';
            }
        }

        int[][] dp = new int[n + 1][k + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = Integer.MAX_VALUE;
            }
        }

        for (int j = 0; j <= k; j++) {
            dp[1][j] = countHours(timetable[0], j);
        }

        for (int i = 2; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                for (int l = 0; l <= j; l++) {
                    dp[i][j] = Math.min(dp[i][j], dp[i - 1][j - l] + countHours(timetable[i - 1], l));
                }
            }
        }

        System.out.println(dp[n][k]);
    }

    private static int countHours(int[] day, int skip) {
        int hours = 0;
        int skipped = 0;
        for (int i = 0; i < day.length; i++) {
            if (day[i] == 1) {
                if (skipped == skip) {
                    hours++;
                } else {
                    skipped++;
                }
            }
        }
        return hours;
    }
}
