import java.util.Scanner;

public class Main {

  static class Player {
    int defense, attack;

    Player(int defense, int attack) {
      this.defense = defense;
      this.attack = attack;
    }
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);

    Player[] team1 = new Player[2];
    for (int i = 0; i < 2; i++) {
      team1[i] = new Player(scanner.nextInt(), scanner.nextInt());
    }

    Player[] team2 = new Player[2];
    for (int i = 0; i < 2; i++) {
      team2[i] = new Player(scanner.nextInt(), scanner.nextInt());
    }

    if (team1[1].attack > team2[0].defense
        && team1[0].defense > team2[1].attack) {
      System.out.println("Team 1");
    } else if (team1[0].attack > team2[1].defense
        && team1[1].defense > team2[0].attack) {
      System.out.println("Team 2");
    } else {
      System.out.println("Draw");
    }
  }
}
