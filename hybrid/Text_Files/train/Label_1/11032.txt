#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n, r;
    std::cin >> n >> r;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    std::sort(a.begin(), a.end());
    int ans = 0;
    int last = -1e9;
    for (int i = 0; i < n; i++) {
        if (a[i] == 1) {
            last = i;
        }
        if (i - last > r - 1) {
            ans++;
            last = i;
        }
    }
    if (last == -1e9) {
        std::cout << -1 << std::endl;
    } else {
        std::cout << ans << std::endl;
    }
    return 0;
}
