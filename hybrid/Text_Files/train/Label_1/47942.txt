import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int[] min = new int[n];
        int[] minPos = new int[n];
        Arrays.fill(min, Integer.MAX_VALUE);
        for (int i = n - 1; i >= 0; i--) {
            if (i + k < n) {
                min[i] = Math.min(min[i], min[i + k]);
                minPos[i] = Math.min(minPos[i], minPos[i + k]);
            }
            if (p[i] < min[i]) {
                min[i] = p[i];
                minPos[i] = i;
            }
        }
        for (int i = 0; i < n; i++) {
            if (i + k < n && minPos[i] == i + k && p[i] > min[i]) {
                p[i] = min[i];
                p[i + k] = p[i] + 1;
                break;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.println(p[i]);
        }
    }
}
