import sys
from collections import defaultdict

def add_edge(adj, u, v):
    adj[u].add(v)
    adj[v].add(u)

def dfs(adj, u, p, x, y, dep):
    if u != p:
        if dep&1:
            x[u] = x[p]
            y[u] = y[p] + 1
        else:
            x[u] = x[p] + 1
            y[u] = y[p]
    for v in adj[u]:
        if v != p:
            dfs(adj, v, u, x, y, dep+1)

n = int(sys.stdin.readline())
adj = defaultdict(set)
for i in range(n-1):
    u, v = map(int, sys.stdin.readline().split())
    add_edge(adj, u, v)

x = [0] * (n+1)
y = [0] * (n+1)
dfs(adj, 1, 1, x, y, 0)

print("YES")
for i in range(1, n+1):
    print(x[i], y[i])
