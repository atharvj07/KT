#include <iostream>
typedef long long ll;
const ll mod = int(1e9) + 7;
ll pow10[10];
int main() {
  int n, k;
  std::cin >> n >> k;
  pow10[0] = 1;
  for (int i = 1; i <= k; i++)
    pow10[i] = pow10[i - 1] * 10 % mod;
  int a[n], b[n];
  for (int i = 0; i < n; i++) std::cin >> a[i];
  for (int i = 0; i < n; i++) std::cin >> b[i];
  ll ans = 0;
  for (int i = 0; i < n;) {
    int num = 0;
    bool ok = true;
    for (int j = 0; j < k; j++) {
      int now;
      std::cin >> now;
      if (j == 0 && now == b[i / k]) ok = false;
      num = num * 10 + now;
    }
    ans += ok && (num % a[i / k] == 0);
    i += k;
  }
  std::cout << ans % mod << '\n';
  return 0;
}
