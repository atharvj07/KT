import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int h = scanner.nextInt();
        int[] a = new int[h + 1];
        for (int i = 1; i <= h; i++) {
            a[i] = scanner.nextInt();
        }
        buildMaxHeap(a, h);
        for (int i = 1; i <= h; i++) {
            System.out.print(a[i] + " ");
        }
    }

    public static void buildMaxHeap(int[] a, int h) {
        for (int i = h / 2; i >= 1; i--) {
            maxHeapify(a, i, h);
        }
    }

    public static void maxHeapify(int[] a, int i, int h) {
        int l = 2 * i;
        int r = 2 * i + 1;
        int largest = i;
        if (l <= h && a[l] > a[i]) {
            largest = l;
        }
        if (r <= h && a[r] > a[largest]) {
            largest = r;
        }
        if (largest != i) {
            int temp = a[i];
            a[i] = a[largest];
            a[largest] = temp;
            maxHeapify(a, largest, h);
        }
    }
}
