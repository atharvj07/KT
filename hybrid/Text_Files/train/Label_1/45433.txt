import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int s = scanner.nextInt();
        int[][] parcels = new int[n][5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                parcels[i][j] = scanner.nextInt();
            }
        }

        int[][] dp = new int[2 * n + 1][s + 1];
        for (int i = 0; i <= 2 * n; i++) {
            Arrays.fill(dp[i], -1);
        }

        int answer = solve(parcels, dp, 0, s);
        System.out.println(answer);
    }

    private static int solve(int[][] parcels, int[][] dp, int idx, int strength) {
        if (idx == 2 * parcels.length) {
            return 0;
        }

        if (dp[idx][strength] != -1) {
            return dp[idx][strength];
        }

        int answer = solve(parcels, dp, idx + 1, strength);
        int weight = parcels[idx / 2][2];
        int value = parcels[idx / 2][4];
        if (strength - weight >= 0 && weight <= parcels[idx / 2][3]) {
            answer = Math.max(answer, solve(parcels, dp, idx + 1, strength - weight) + value);
        }

        dp[idx][strength] = answer;
        return answer;
    }
}
