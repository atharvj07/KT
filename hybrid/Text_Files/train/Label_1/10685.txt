#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<long long> a(n);
    for (long long &aa : a) cin >> aa;
    long long ans = 1e9;
    for (int i = 0; i <= k; i++) {
        ans = min(ans, *min_element(a.begin(), a.end(), [k, i](const long long &x, const long long &y) { return (x + k - i) % (y + k - i) < (y + k - i) % (x + k - i); }));
    }
    cout << ans << endl;
    return 0;
}
