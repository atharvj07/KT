from bisect import bisect_right
n = int(input())
X = []
W = []
for _ in range(n):
  x, w = map(int, input().split())
  X.append(x)
  W.append(w)
X2 = sorted(zip(X, W))

ms = 0
for i in range(n):
  x, w = X2[i]
  l = bisect_right(X, x - w - 1)
  r = bisect_right(X, x + w)
  ms = max(ms, r - l + 1)
print(ms)
