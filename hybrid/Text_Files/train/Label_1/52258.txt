import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);

    int n = in.nextInt(), ans = n, steps = 0;
    while (ans % 2 == 0 || ans % 3 == 0) {
      if (ans % 2 == 0) {
        ans /= 2;
      } else {
        ans /= 3;
      }
      steps++;
    }
    System.out.println(ans + " " + steps);
  }
}
