#include <iostream>
#include <string>

using namespace std;

int main() {
  string x;
  cin >> x;

  int len = x.length();
  int exp = 0;
  int ntz = 0;
  bool dec = false;

  for (int i = 0; i < len; i++) {
    if (x[i] == '.') {
      dec = true;
      ntz = 0;
    } else if (x[i] != '0' || dec) {
      if (dec) exp--;
      ntz = 0;
    } else {
      ntz++;
    }
  }

  if (ntz == len) exp = 0;
  x = x.substr(0, len - ntz);

  if (x[0] == '.') {
    x = x.substr(1, len - 1 - ntz);
    exp--;
  } else if (x[0] == '0') {
    for (int i = 1; i < len - ntz; i++) {
      if (x[i] != '0') {
        x = x.substr(i, len - ntz - i);
        exp -= i;
        break;
      }
    }
  }

  if (x.back() == '.') x.pop_back();

  if (exp == 0) cout << x << endl;
  else cout << x << "E" << exp << endl;

  return 0;
}
