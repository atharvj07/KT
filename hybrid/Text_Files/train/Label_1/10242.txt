#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int n;
    while (cin >> n) {
        string s;
        cin >> s;
        for (int i = 0; i < n; i++) {
            if (i + 1 < n && s[i] > s[i + 1]) {
                cout << s.substr(0, i) + s.substr(i + 1) << endl;
                return 0;
            }
        }
        s.pop_back();
        cout << s << endl;
    }
}
