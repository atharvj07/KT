#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  int max_coders = 0;
  vector<vector<char>> board(n, vector<char>(n, '.'));
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      bool is_safe = true;
      if (i > 0 && board[i - 1][j] == 'C') is_safe = false;
      if (i < n - 1 && board[i + 1][j] == 'C') is_safe = false;
      if (j > 0 && board[i][j - 1] == 'C') is_safe = false;
      if (j < n - 1 && board[i][j + 1] == 'C') is_safe = false;
      if (is_safe) {
        board[i][j] = 'C';
        max_coders++;
      }
    }
  }
  cout << max_coders << endl;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      cout << board[i][j];
    }
    cout << endl;
  }
  return 0;
}
