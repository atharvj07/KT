import sys
inp = sys.stdin.readlines()
n, m, k = [int(x) for x in inp[0].split()]
a = [[int(x) for x in line.split()] for line in inp[1:]]
for p in range(m):
    s = []
    for i in range(n):
        if a[i][p]:
            s.append(i)
    if len(s) < n:
        t = [k, s[0]]
        for i in range(len(s) - 1):
            t[0] -= min(k, s[i + 1] - s[i])
            if t[0] <= 0:
                break
            t = [t[0], s[i + 1]]
        result = [0] * m
        result[p] = t[0]
    else:
        result = [0] * m
    print(*result)
