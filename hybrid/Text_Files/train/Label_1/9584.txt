#include <cstdio>
#include <vector>
using namespace std;

int main() {
  int n, k;
  scanf("%d %d", &n, &k);
  vector<int> a(n);
  for (int &x : a) scanf("%d", &x);
  int m = n / k;
  for (int i = 1; i < m; i++)
    for (int j = 0; j < k; j++) if (a[j] != a[j + i * k]) m = i;
  int cnt = 0;
  for (int i = 0; i < n; i++) if (a[i % k] != a[i]) cnt++;
  printf("%d\n", cnt * m);
  return 0;
}
