#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>

int main() {
    long long n, k, s, m, t = 0;
    std::cin >> n >> k;
    std::priority_queue<long long> q;
    std::vector<long long> ans(n);
    for (long long i = 0; i < n; i++) {
        std::cin >> s >> m;
        while (!q.empty() && q.top() <= s) q.pop();
        if (q.size() < k) {
            q.push(s + m);
            ans[i] = s + m;
        } else {
            t = q.top();
            q.pop();
            q.push(t + m);
            ans[i] = t + m;
        }
    }
    for (long long a : ans) std::cout << a << std::endl;
    return 0;
}
