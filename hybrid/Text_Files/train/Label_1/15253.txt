#include <iostream>
#include <vector>
using namespace std;
int main()
{
    int n;
    cin >> n;
    long long int sum = 0;
    vector<long long int> v(n);
    for (int i = 0; i < n; i++)
    {
        cin >> v[i];
        sum += v[i];
    }
    if (sum % 2 == 1)
    {
        cout << "NO" << endl;
        return 0;
    }
    long long int suml = 0, sumr = sum;
    for (int i = 0; i < n; i++)
    {
        suml += v[i];
        sumr -= v[i];
        if (suml == sumr)
        {
            cout << "YES" << endl;
            return 0;
        }
    }
    cout << "NO" << endl;
}
