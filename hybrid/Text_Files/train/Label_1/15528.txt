#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 5e5+5,K = 505;
int a[N],sum[K][N];
int main(){
    int q,t,x,y,c;
    cin >> q;
    while(q--){
        cin >> t >> x >> y;
        c = x%K;
        if(t == 1) a[x] += y;
        else {
            int ans = sum[c][x];
            if(y) ans -= sum[c][y-1];
            cout << ans << '\n';
        }
        sum[c][x] += a[x];
    }
}
