import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[][] p = new int[m][2];
        for (int i = 0; i < m; i++) {
            p[i][0] = sc.nextInt();
            p[i][1] = sc.nextInt();
        }

        for (int i = 0; i < m; i++) {
            int sum = 0;
            for (int j = 0; j < n; j++) {
                if (j + 1 == p[i][0]) {
                    sum += p[i][1];
                } else {
                    sum += t[j];
                }
            }
            System.out.println(sum);
        }
    }
}
