import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    private static int[] solve(int h, int w, int[] a, int[] b) {
        int[] dp = new int[h + 1];
        Arrays.fill(dp, -1);
        dp[0] = 0;
        for (int i = 0; i < h; i++) {
            int xa = a[i];
            int xb = b[i];
            for (int j = 0; j < w; j++) {
                if (xa <= j && j < xb) {
                    dp[i + 1] = -1;
                }
                if (dp[i] >= 0) {
                    if (dp[i + 1] == -1 || dp[i + 1] > dp[i] + 1) {
                        dp[i + 1] = dp[i] + 1;
                    }
                    if (j + 1 < w && (dp[i + 1] == -1 || dp[i + 1] > dp[i] + 1)) {
                        dp[i + 1] = dp[i] + 1;
                    }
                }
            }
        }
        int[] res = new int[h];
        for (int i = 0; i < h; i++) {
            res[i] = dp[i + 1];
        }
        return res;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        int h = Integer.parseInt(s[0]);
        int w = Integer.parseInt(s[1]);
        int[] a = new int[h];
        int[] b = new int[h];
        for (int i = 0; i < h; i++) {
            s = br.readLine().split(" ");
            a[i] = Integer.parseInt(s[0]);
            b[i] = Integer.parseInt(s[1]);
        }
        int[] res = solve(h, w, a, b);
        for (int i = 0; i < h; i++) {
            if (res[i] == -1) {
                System.out.println(-1);
            } else {
                System.out.println(res[i]);
            }
        }
    }
}
