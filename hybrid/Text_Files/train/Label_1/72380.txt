import bisect, sys

input = sys.stdin.readline
n, q = map(int, input().split())
L = []; server = []
for x in range(q):
    t, k, d = map(int, input().split())
    L += [[t, k, d, x]]
L = sorted(L)
for t, k, d, i in L:
    while server and server[0] < t: server.pop(0)
    if len(server) >= k:
        res = 0; server.sort()
        for x in range(k): res += server[x]
        for x in range(k): server[x] = t + d
    else: print(-1); continue
    print(res)
