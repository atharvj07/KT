import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for(int i = 0; i < n; i++) a[i] = sc.nextInt();
        int[] p = new int[n];
        for(int i = 0; i < n; i++) p[a[i] - 1] = i;
        int ans = 0;
        for(int i = 1; i < n; i++) ans += Math.abs(p[i] - p[i - 1]);
        System.out.println(ans);
    }
}
