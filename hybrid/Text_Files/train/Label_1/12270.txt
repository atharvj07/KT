#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int p = 1e9 + 7;
ll f[2005][2005];
void add(ll &a, ll b) {
    a = (a + b) % p;
}
int main() {
    int n, k;
    cin >> n;
    f[0][0] = 1;
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j <= 2 * i; ++j)
            for (int k = 0; k + j <= 2 * i; ++k) { add(f[i][j + k], f[i - 1][j] * f[i - 1][k] % p); }
    }
    ll ans = 0;
    for (int i = 0; i <= 2 * n; ++i) { add(ans, f[n][i]); }
    cout << ans << endl;
    return 0;
}
