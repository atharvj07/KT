import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        String exp = null, msg = null;
        for (int i = 0; i < n; i++) {
            String line = sc.nextLine().trim();
            if (line.startsWith("throw")) {
                exp = line.substring(6, line.length() - 1);
            } else if (line.startsWith("catch")) {
                msg = line.substring(6, line.length() - 1);
                if (exp != null && exp.equals(msg.substring(0, msg.indexOf(',')))) {
                    System.out.println(msg.substring(msg.indexOf(',') + 2, msg.length() - 1));
                    return;
                }
            }
        }
        System.out.println("Unhandled Exception");
    }
}
