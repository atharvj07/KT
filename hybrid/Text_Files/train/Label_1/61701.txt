def main():
    n = int(input())
    if n % 4 == 3:
        print('No solution')
        return
    a = [[0 for j in range(n)] for i in range(n)]
    row = [2 * i + 1 for i in range(n)]
    col = [row[n - 1 - i] for i in range(n)]
    for i in range(n):
        for j in range(n):
            if i == j:
                a[i][j] = (row[i] + col[j]) // 2
            else:
                a[i][j] = n * n + 1 - a[i][j]
    for line in a:
        print(*line)

main()
