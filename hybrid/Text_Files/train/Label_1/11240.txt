#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  vector<string> v;
  for (int i = 0; i < n; i++) {
    cin >> s;
    v.push_back(s);
  }
  vector<int> dp(n + 1, 0);
  dp[0] = 0;
  for (int i = 1; i <= n; i++) {
    dp[i] = dp[i - 1];
    for (int j = 0; j < v[i - 1].size(); j++) {
      if (v[i - 1][j] == 's' && (j == 0 || v[i - 1][j - 1] == 'h')) {
        dp[i] = max(dp[i], dp[j] + (int)v[i - 1].size() - 1 - j);
      }
    }
  }
  cout << dp[n] << endl;
  return 0;
}
