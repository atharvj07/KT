#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 1e9 + 7;
const int MAXN = 5e5 + 5;
int a[MAXN];
ll dp[MAXN];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) {
    dp[i] = a[i];
    if (i - 1 >= 0) { dp[i] = max(dp[i], dp[i - 1] - a[i]); }
    if (i + 1 < n) { dp[i] = max(dp[i], dp[i + 1] - a[i]); }
  }
  ll ans = -1e18;
  for (int i = 0; i < n; i++) { ans = max(ans, dp[i]); }
  cout << ans << endl;
  return 0;
}
