import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int row = h / a;
        int col = w / b;
        int area = h * w - row * col * a * b;
        System.out.println(area);
    }
}
