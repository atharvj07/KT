from sys import stdin
n, m = map(int, stdin.readline().split())
grid = [list(stdin.readline().strip()) for _ in range(n)]
for i in range(n):
    for j in range(m):
        if grid[i][j] == 'B':
            if i > 0 and j > 0 and grid[i - 1][j] == 'B' and grid[i][j - 1] == 'B':
                print("NO")
                exit()
            if i + 1 < n and j > 0 and grid[i + 1][j] == 'B' and grid[i][j - 1] == 'B':
                print("NO")
                exit()
            if i > 0 and j + 1 < m and grid[i - 1][j] == 'B' and grid[i][j + 1] == 'B':
                print("NO")
                exit()
            if i + 1 < n and j + 1 < m and grid[i + 1][j] == 'B' and grid[i][j + 1] == 'B':
                print("NO")
                exit()
print("YES")
