from math import ceil, floor

for _ in range(int(input())):
    n, m, k = map(int, input().split())
    for _ in range(k):
        print("\n".join([" ".join(map(str, [ceil(n / m) if i % m == 0 else floor(n / m), *range(1 + (i % m) * ceil(n / m), 1 + (i % m + 1) * ceil(n / m))])) for i in range(m)]))
