import math

def solve(n, k):
    # base cases
    if n == 1:
        return [1]
    elif n == 2:
        if k == 1:
            return [1, 2]
        else:
            return [2, 1]
    
    # calculate number of good permutations
    num_good_permutations = math.factorial(n) // 2
    
    # check if k is greater than number of good permutations
    if k > num_good_permutations:
        return -1
    
    # build permutation
    permutation = []
    visited = [False for i in range(n + 1)]
    
    # start with 1
    permutation.append(1)
    visited[1] = True
    
    # iterate through remaining elements
    i = 2
    while len(permutation) < n:
        # find next unvisited element reachable from current element
        next_element = i
        while visited[next_element] or i == 1:
            next_element += 1
            if next_element > n:
                next_element = 1
        
        # add next element to permutation
        permutation.append(next_element)
        visited[next_element] = True
        i = next_element + 1
    
    # shift permutation so that smallest element is first
    min_element = min(permutation)
    min_index = permutation.index(min_element)
    permutation = permutation[min_index:] + permutation[:min_index]
    
    return permutation

for _ in range(int(input())):
    n, k = map(int, input().split())
    result = solve(n, k)
    if result == -1:
        print(result)
    else:
        print(*result)
