from collections import defaultdict

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

n = int(input())
a = list(map(int, input().split()))
g = defaultdict(list)
for _ in range(n - 1):
    x, y = map(int, input().split())
    g[x].append(y)
    g[y].append(x)

vis = [False] * (n + 1)
ans = [0] * (n + 1)

def dfs(v, p):
    ans[v] = a[v - 1]
    for u in g[v]:
        if u == p:
            continue
        dfs(u, v)
        ans[v] = gcd(ans[v], ans[u])

def dfs2(v, p):
    for u in g[v]:
        if u == p:
            continue
        ans[u] = gcd(ans[v], a[u - 1] if ans[u] == 1 else ans[u])
        dfs2(u, v)

dfs(1, 1)
dfs2(1, 1)
print(*ans[1:])
