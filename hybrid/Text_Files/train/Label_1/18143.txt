#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);

  int n, k;
  cin >> n >> k;

  vector<int> a(n), b;
  vector<bool> in_b(n + 1, false);
  for (int &x : a) cin >> x;
  if (k) {
    b.resize(k);
    for (int &x : b) {
      cin >> x;
      in_b[x] = true;
    }
  }

  vector<int> nxt(n + 1, -1);
  for (int i = n - 2; i >= 0; i--) {
    if (!in_b[i + 1]) { nxt[i] = i + 1; }
    else {
      nxt[i] = nxt[i + 1];
    }
  }

  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (nxt[i] == -1) continue;
    if (a[i] < a[nxt[i]]) continue;

    int dif = a[nxt[i]] - a[i];
    a[i] = a[nxt[i]] - 1;
    ans += dif;
  }

  for (int i = 0; i < n - 1; i++) {
    if (a[i] >= a[i + 1]) {
      cout << -1 << endl;
      return 0;
    }
  }

  cout << ans << endl;

  return 0;
}
