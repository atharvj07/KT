#include <iostream>
#include <vector>
#include <string>
using namespace std;
string op(string s, int l, int r, char c1, char c2) {
  int n = s.size();
  for (int i = l - 1; i < r; i++)
    if (s[i] == c1) s[i] = c2;
  return s;
}
int main() {
  int n, m;
  cin >> n >> m;
  string s;
  cin >> s;
  for (int i = 0; i < m; i++) {
    int l, r;
    char c1, c2;
    cin >> l >> r >> c1 >> c2;
    s = op(s, l, r, c1, c2);
  }
  cout << s;
  return 0;
}
