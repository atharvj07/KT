#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int cnt = k;
  for (int i = 0; i < n; i++) {
    int add = m - cnt;
    if (add <= a[i]) {
      cnt += a[i] - add;
    }
    if (cnt >= m) {
      cout << i + 1 << endl;
      return 0;
    }
  }
  cout << -1 << endl;
  return 0;
}
