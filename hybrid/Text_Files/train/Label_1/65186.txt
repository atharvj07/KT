class TimeTravelingMultiset:
    def __init__(self):
        self.history = []  # Stores (time, value, operation)

    def add(self, time, x):
        self.history.append((time, x, 1))

    def remove(self, time, x):
        self.history.append((time, x, -1))

    def count(self, time, x):
        count = 0
        for h_time, h_x, op in self.history:
            if h_time <= time and h_x == x:
                count += op
        return count

n = int(input())
multiset = TimeTravelingMultiset()

for _ in range(n):
    query = list(map(int, input().split()))
    if query[0] == 1:
        multiset.add(query[1], query[2])
    elif query[0] == 2:
        multiset.remove(query[1], query[2])
    else:
        print(multiset.count(query[1], query[2]))
