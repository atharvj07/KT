#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    if (!(n & 1)) {
        int k = 0;
        for (int i = 1; i < n; i += 2) {
            if (s[i] == s[i - 1]) {
                k++;
            }
        }
        cout << k << '\n';
        for (int i = 0; i < n; i += 2) {
            cout << s[i];
        }
        cout << '\n';
    } else {
        int k = 0;
        for (int i = 0; i < n; i += 2) {
            if (i + 1 < n && s[i] == s[i + 1]) {
                k++;
            }
            if (i > 0 && s[i] == s[i - 1]) {
                k++;
            }
        }
        cout << k << '\n';
        for (int i = 0; i < n; i += 2) {
            if (i + 1 < n && s[i] == s[i + 1]) {
                continue;
            }
            cout << s[i];
        }
        cout << '\n';
    }
}
