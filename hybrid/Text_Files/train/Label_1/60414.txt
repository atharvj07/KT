from collections import deque
from sys import setrecursionlimit

setrecursionlimit(10 ** 6)

n = int(input())
par = [0] * (n + 1)
g = [[]] * (n + 1)
for i in range(1, n):
    a, b = map(int, input().split())
    g[a].append(b)
    par[b] = a

leaf = 0
for i in range(1, n + 1):
    if len(g[i]) == 0:
        leaf += 1

num = [0] * (leaf + 1)
ind = [0] * (n + 1)
c = 1

q = deque()
q.append(1)
while len(q) != 0:
    u = q.popleft()
    ind[u] = c
    num[c] = u
    c += 1
    for v in g[u]:
        q.append(v)

print(num[leaf], num[1])
