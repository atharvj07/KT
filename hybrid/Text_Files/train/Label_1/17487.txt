#include<bits/stdc++.h>
using namespace std;
int n; string s;
int main(){
    cin>>s,n=s.size();
    for(int i=0;i<n;i+=2){
        if(s[i+1]=='+') for(int j=0;j<n;j+=2){
            if(j==i) continue;
            int a=0,b=0,c=0;
            if(i-2>=0) a=s[i-2]-'0';
            if(i+2<n) b=s[i+2]-'0';
            if(j-2>=0) c=s[j-2]-'0';
            s[i+1]='*',s[j+1]='+';
            int ans=a*(s[i]-'0')+c*(s[j]-'0')+b;
            s[i+1]='+',s[j+1]='*';
            cout<<ans<<endl;
            return 0;
        }
    }
    cout<<0<<endl;
}
