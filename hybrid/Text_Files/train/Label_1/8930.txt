#include<bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;
int n, a[N];
map<int, int> num;

int main() {
    int t; cin >> t;
    while(t--) {
        cin >> n;
        for(int i = 1; i <= n; i++)   {
            cin >> a[i];
            num[a[i]]++;
        }
        int ans = INT_MAX;
        int c = 0;
        for(int i = 1; i <= n; i++) {
            for(long long x = a[i] * 2; x <= 2e5; x += a[i]) {
                if(num[x] > 0) c++;
            }
            ans = min(ans, c);
            c = 0;
            if(num[a[i]] > 1) {
                ans = min(ans, (int)(num[a[i]] - 1));
                num[a[i]] = 1;
            }
            else num.erase(a[i]);
        }
        for(auto &tmp : num) {
            if(tmp.second > 0) ans = min(ans, tmp.second);
        }
        cout << ans << '\n';
        num.clear();
    }
    return 0;
}
