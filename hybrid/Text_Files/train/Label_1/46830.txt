import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    String s = sc.next();
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (i + 1 < n && s.charAt(i) != s.charAt(i + 1)) {
        cnt += 2;
        i++;
      }
    }
    System.out.println(n - cnt);
  }
}
