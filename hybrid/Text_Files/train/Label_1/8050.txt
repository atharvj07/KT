#include <iostream>
using namespace std;
long long int solve(long long int h, long long int n) {
  if (n == 1) {
    return 1;
  }
  long long int ans = 1 + solve(h - 1, (n + 1) / 2);
  if ((n + 1) % 2 == 0) {
    ans -= 2;
  }
  return ans;
}
int main() {
  long long int h, n;
  cin >> h >> n;
  cout << solve(h, n) << endl;
  return 0;
}
