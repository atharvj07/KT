#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    sort(s.begin(), s.end());
    if (k > n * (n + 1) / 2) {
        cout << -1 << endl;
        return 0;
    }
    int ans = 0;
    for (int i = 0, j = 0; i < n;) {
        if (k == j) break;
        while (i < n && s[i] == s[j]) i++;
        int cnt = i - j;
        if (cnt > k - j) cnt = k - j;
        ans += cnt * (n - cnt);
        j += cnt;
    }
    cout << ans << endl;
}
