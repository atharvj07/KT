#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  for (int i = 0; i < n; i++) {
    if (a[i] + b[i] & 1) {
      cout << "No" << endl;
      return 0;
    }
  }
  for (int i = 0; i < n; i++) {
    if ((b[i] - a[i]) % 2 == 0 || b[i] == a[i]) continue;
    for (int j = i + 1; j < n; j++) {
      if ((b[i] - a[i]) == (a[j] - b[j])) {
        a[i]++;
        a[j]--;
        b[i]--;
        b[j]++;
        break;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (a[i] != b[i]) {
      cout << "No" << endl;
      return 0;
    }
  }
  cout << "Yes" << endl;
  return 0;
}
