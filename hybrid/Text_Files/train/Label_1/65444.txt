import sys

input = sys.stdin.readline

for t in range(int(input())):
    n, x = map(int, input().split())
    d = [0] * n
    h = [0] * n
    for i in range(n):
        d[i], h[i] = map(int, input().split())
    ans = -1
    for mask in range(1 << n):
        cur = x
        for i in range(n):
            if (mask >> i) & 1:
                cur = (cur - d[i] + h[i])
                if cur <= 0:
                    break
        if cur <= 0:
            cnt = bin(mask).count("1")
            if ans == -1 or ans > cnt:
                ans = cnt
    print(ans)
