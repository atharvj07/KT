#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, a, b;
  cin >> n >> a >> b;
  vector<long long> v(n);
  for (auto &i : v) cin >> i;
  for (long long i = 0; i < n; i++) {
    long long y = a * v[i] / b;
    y = y - v[i] * y / a;
    cout << v[i] - y << " ";
  }
  cout << endl;
}
