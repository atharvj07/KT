n, k, m = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
add = [0] * n
ans = 0
for i in range(n - 1, -1, -1):
    if m <= 0:
        break
    need = k - add[i]
    if need <= m:
        ans += need
        m -= need
        add[i] += need
    else:
        ans += m
        m = 0
ans += sum(a)
print(ans / n)
