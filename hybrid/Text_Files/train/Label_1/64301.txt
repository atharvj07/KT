import bisect

n = int(input())
a = list(map(int, input().split()))

inc_seq = []
dec_seq = []

inc_idx = 0
for i in range(n):
    idx = bisect.bisect_left(inc_seq, a[i])
    if idx == len(inc_seq):
        inc_seq.append(a[i])
        inc_idx = idx + 1
    elif inc_seq[idx] == a[i]:
        continue
    else:
        inc_seq[idx] = a[i]

dec_seq = [a[-1]]
dec_idx = 0
for i in range(n - 1, -1, -1):
    idx = bisect.bisect_right(dec_seq, a[i])
    if idx == 0:
        dec_seq.insert(0, a[i])
        dec_idx = 0
    elif dec_seq[idx - 1] == a[i]:
        continue
    else:
        dec_seq.insert(idx, a[i])

if len(inc_seq) + len(dec_seq) == n and set(inc_seq + dec_seq) == set(a):
    print("YES")
    print(inc_idx)
    print(*inc_seq)
    print(len(dec_seq))
    print(*dec_seq)
else:
    print("NO")
