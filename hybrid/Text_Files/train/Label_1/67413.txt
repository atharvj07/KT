n, k = map(int, input().split())
ratings = list(map(int, input().split()))
excluded = []

while True:
    exclude = False
    for i in range(1, len(ratings)):
        d_i = sum(ratings[j] * j - (len(ratings) - i) * ratings[i] for j in range(i))
        if d_i < k:
            excluded.append(n - len(ratings) + i + 1)
            ratings.pop(i)
            exclude = True
            break
    if not exclude:
        break

print("\n".join(map(str, excluded)))
