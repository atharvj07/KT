import java.util.ArrayList;
import java.util.Scanner;
import java.util.Arrays;
import java.util.List;

class Main {
    static List<String> words = Arrays.asList("the", "this", "that");

    public static void main(String[] args) {
        final Scanner scanner = new Scanner(System.in);
        while (scanner.hasNext()) {
            String encrypted = scanner.nextLine();
            for (int i = 1; i < 26; i++) {
                String decrypted = decrypt(i, encrypted);
                if (containsAny(decrypted, words)) {
                    System.out.println(decrypted);
                    break;
                }
            }
        }
        scanner.close();
    }

    static boolean containsAny(String str, List<String> words) {
        for (String word : words) {
            if (str.contains(word)) {
                return true;
            }
        }
        return false;
    }

    static String decrypt(int shift, String encrypted) {
        char[] chars = encrypted.toCharArray();
        for (int i = 0; i < chars.length; i++) {
            char c = chars[i];
            if (c >= 'a' && c <= 'z') {
                chars[i] = (char) ((c - 'a' - shift + 26) % 26 + 'a');
            }
        }
        return new String(chars);
    }
}
