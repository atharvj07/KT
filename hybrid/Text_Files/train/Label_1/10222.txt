#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long p, q;
        cin >> p >> q;
        long long l = 1, r = p, g = 1;
        while (l <= r) {
            long long mid = (l + r) / 2;
            if (p % mid == 0 && mid % q) {
                g = mid;
                l = mid + 1;
            } else {
                r = mid - 1;
            }
        }
        cout << g << '\n';
    }
    return 0;
}
