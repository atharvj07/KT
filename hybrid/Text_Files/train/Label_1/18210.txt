#include <iostream>
#include <vector>

using namespace std;

int main() {
  int h1, a1, c1, h2, a2;
  cin >> h1 >> a1 >> c1 >> h2 >> a2;
  if (h2 <= a1) {
    cout << 1 << endl;
    cout << "STRIKE" << endl;
    return 0;
  }
  int phases = (h2 + a2 - 1) / a1;
  cout << phases << endl;
  for (int i = 0; i < phases; i++) {
    if (i == phases - 1 || h1 - a2 <= c1 - a1) {
      cout << "STRIKE" << endl;
      h2 -= a1;
    } else {
      cout << "HEAL" << endl;
      h1 = min(h1 + c1, h1 + a1 - a2);
    }
  }
  return 0;
}
