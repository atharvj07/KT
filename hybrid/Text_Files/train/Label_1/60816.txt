n = int(input())
d = list(map(int,input().split()))
prefix = [0]*n
prefix[0] = d[0]
for i in range(1,n):
    prefix[i] = prefix[i-1] + d[i]
suffix = [0]*n
suffix[n-1] = d[n-1]
for i in range(n-2,-1,-1):
    suffix[i] = suffix[i+1] + d[i]
#
ans = 0
for i in range(n-1):
    #
    l = i+1
    r = n-1
    pos = -1
    while l<=r:
        m = (l+r)//2
        if prefix[m] - (prefix[i] if i>0 else 0)>=suffix[m+1]:
            pos = m
            l = m+1
        else:
            r = m-1
    #
    if pos==-1:
        continue
    #
    if (prefix[pos] - (prefix[i] if i>0 else 0))==(suffix[pos+1]):
        ans = max(ans,prefix[pos] - (prefix[i] if i>0 else 0))
#
print(ans)
