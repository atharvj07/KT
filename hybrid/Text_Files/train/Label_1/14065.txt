#include <bits/stdc++.h>
#define ll long long
using namespace std;
int n, a[2005];
int main() {
  cin >> n;
  for (int i = 1; i <= 2 * n; i++) cin >> a[i];
  for (int i = n + 1; i <= 2 * n; i++) {
    if (a[i] != a[i - n]) {
      for (int j = 1; j <= n; j++) cout << a[j] << ' ';
      for (int j = n + 1; j <= 2 * n; j++) if (a[j] != a[j - n]) cout << a[j] << ' ';
      return 0;
    }
  }
  cout << -1;
}
