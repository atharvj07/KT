#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll n, a, b, c;
    cin >> n >> a >> b >> c;
    if (n % 4 == 0) cout << 0 << endl;
    else if (n % 4 == 1) cout << min(a, min(3 * c, b + c)) << endl;
    else if (n % 4 == 2) cout << min(2 * a, min(c, 2 * b)) << endl;
    else cout << min(a + c, 3 * b) << endl;
    return 0;
}
