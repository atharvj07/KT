import sys
input = sys.stdin.readline

n = int(input())
xs = []
ws = []
for i in range(n):
    x, w = map(int, input().split())
    xs.append(x)
    ws.append(w)

xs.sort()
ws.sort()

ans = 0
for i in range(n):
    l = i
    r = n - 1
    while l < r:
        mid = (l + r) // 2
        if xs[mid] - xs[i] >= ws[i] + ws[mid]:
            l = mid + 1
        else:
            r = mid
    if xs[l] - xs[i] >= ws[i] + ws[l]:
        ans = max(ans, l - i + 1)

print(ans + 1)
