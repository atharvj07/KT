import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int H = sc.nextInt();
        int W = sc.nextInt();
        int[] x = new int[N];
        for (int i = 0; i < N; i++) {
            x[i] = sc.nextInt();
        }

        int sum = 0;
        for (int i = 0; i < N; i++) {
            if (i % 2 == 0) {
                sum += W - x[i];
            } else {
                sum += x[i] - 1;
            }
        }
        System.out.println(sum * H);
    }
}
