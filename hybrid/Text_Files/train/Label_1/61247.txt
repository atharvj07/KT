import sys

MAX_N = 2 * 10**5

for t in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]

    s = sum(a)
    if s == n - 1:
        print("NO")
        continue

    if s % 2 == 0:
        # Can swap adjacent piles
        print("YES")
    elif s % 4 == 1:
        # Can swap piles with odd number of stones
        for i in range(n):
            if (a[i] + a[(i + 1) % n]) % 2 == 1:
                print("YES")
                break
        else:
            print("NO")
    else:
        # Cannot remove all stones
        print("NO")
