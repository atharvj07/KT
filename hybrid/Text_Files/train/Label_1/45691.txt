import java.util.Arrays;
import java.util.Scanner;

public class VovaAndPoints {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int p = scanner.nextInt();
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int[] marks = new int[k];
        for (int i = 0; i < k; i++) {
            marks[i] = scanner.nextInt();
        }

        Arrays.sort(marks);

        int remainingMarksSum = x - Arrays.stream(marks).sum();
        int remainingMarksCount = n - k;
        int minMarkAllowed = (remainingMarksSum + remainingMarksCount - 1) / remainingMarksCount;
        int maxMarkAllowed = (remainingMarksSum + remainingMarksCount - 1) / remainingMarksCount;

        if (maxMarkAllowed < y || k == n) {
            System.out.println("-1");
            return;
        }

        int[] remainingMarks = new int[remainingMarksCount];
        for (int i = 0; i < remainingMarksCount; i++) {
            if (i < remainingMarksCount - (maxMarkAllowed - y)) {
                remainingMarks[i] = maxMarkAllowed;
            } else {
                remainingMarks[i] = minMarkAllowed;
            }
        }

        Arrays.sort(remainingMarks);
        for (int mark : remainingMarks) {
            System.out.print(mark + " ");
        }
    }
}
