import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int size = Integer.parseInt(scanner.nextLine());
        String[] board = new String[size];
        for (int i = 0; i < size; i++) {
            board[i] = scanner.nextLine();
        }

        char[][] grid = new char[size + 2][size + 2];
        for (int i = 0; i < size; i++) {
            grid[i + 1] = ("." + board[i] + ".").toCharArray();
        }

        int freeCells = 0;
        for (char[] row : grid) {
            for (char cell : row) {
                if (cell == '.') freeCells++;
            }
        }

        if (freeCells % 5 != 0) {
            System.out.println("NO");
            return;
        }

        int[][][] dp = new int[size + 1][size + 1][freeCells / 5 + 1];
        dp[0][0][0] = 1;

        int[] dx = new int[]{-1, 1, 0, 0};
        int[] dy = new int[]{0, 0, -1, 1};
        for (int i = 0; i <= size; i++) {
            for (int j = 0; j <= size; j++) {
                for (int k = 0; k <= freeCells / 5; k++) {
                    if (dp[i][j][k] == 0) continue;

                    for (int l = 0; l < 4; l++) {
                        int x = i + dx[l];
                        int y = j + dy[l];
                        if (x >= 0 && x <= size && y >= 0 && y <= size) {
                            if (grid[x][y] == '.') {
                                int[][] newGrid = new int[size + 2][size + 2];
                                for (int m = 0; m <= size; m++) {
                                    for (int n = 0; n <= size; n++) {
                                        newGrid[m][n] = grid[m][n];
                                    }
                                }
                                newGrid[x][y] = '#';

                                int[] newPos = Arrays.copyOf(new int[]{x, y, k + 1}, 3);
                                if (dp[x][y][k + 1] == 0) {
                                    dp[x][y][k + 1] = Arrays.hashCode(newPos);
                                }
                            }
                        }
                    }
                }
            }
        }

        System.out.println(dp[size][size][freeCells / 5] == 0 ? "NO" : "YES");
    }
}
