#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mp make_pair
#define pb push_back
#define F first
#define S second
#define vi vector<int>
#define vl vector<ll>
ll mod = 1e9 + 7;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<int> ans;
    for (int k = 1; k <= n; k++) {
      bool flag = true;
      for (int i = 0; i + k - 1 < n; i++) {
        ll g = a[i];
        for (int j = i + 1; j < i + k; j++) {
          g &= a[j];
        }
        bool c = false;
        for (int j = i; j < i + k; j++) {
          if (a[j] == g && !c) {
            c = true;
            continue;
          }
          if (a[j] < g || (a[j] - g) % k) { flag = false; }
        }
      }
      if (flag) { ans.pb(k); }
    }
    for (vector<int>::iterator it = ans.begin(); it != ans.end(); it++) {
      cout << *it << " ";
    }
    cout << "\n";
  }
  return 0;
}
