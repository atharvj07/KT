import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        int n = in.nextInt();
        int m = in.nextInt();

        long[] skills = new long[k];
        for (int i = 0; i < k; i++) {
            skills[i] = in.nextLong();
        }

        Comparator<Improvement> comp = new Comparator<Improvement>() {
            @Override
            public int compare(Improvement a, Improvement b) {
                long va = a.toApply(skills);
                long vb = b.toApply(skills);
                return Long.compare(vb, va);
            }
        };

        PriorityQueue<Improvement> pq = new PriorityQueue<>(comp);
        for (int i = 0; i < n; i++) {
            int type = in.nextInt();
            int skill = in.nextInt() - 1;
            int val = in.nextInt();
            pq.add(new Improvement(type, skill, val));
        }

        List<Integer> ans = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            Improvement improvement = pq.poll();
            ans.add(improvement.index);
            skills[improvement.skill] = improvement.toApply(skills);
        }

        System.out.println(ans.size());
        for (int i = 0; i < ans.size(); i++) {
            System.out.print(ans.get(i) + " ");
        }
        System.out.println();
    }

    static class Improvement {
        int type;
        int skill;
        int val;
        int index;

        public Improvement(int type, int skill, int val) {
            this.type = type;
            this.skill = skill;
            this.val = val;
            this.index = index;
        }

        public long toApply(long[] skills) {
            long skill = skills[this.skill];
            switch (type) {
                case 1:
                    return val;
                case 2:
                    return skill + val;
                case 3:
                    return skill * val;
            }
            return -1;
        }
    }
}
