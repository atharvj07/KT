import java.util.*;

public class Main {
    static final int INF = 1000000000;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int c_l = sc.nextInt();
        int c_e = sc.nextInt();
        int v = sc.nextInt();
        int[] l = new int[c_l];
        for (int i = 0; i < c_l; i++) {
            l[i] = sc.nextInt();
        }
        int[] e = new int[c_e];
        for (int i = 0; i < c_e; i++) {
            e[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        int[][] d = new int[n + 1][m + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                d[i][j] = INF;
            }
        }
        d[1][1] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (j > 1) d[i][j] = Math.min(d[i][j], d[i][j - 1] + 1);
                if (i > 1) d[i][j] = Math.min(d[i][j], d[i - 1][j] + 1);
                if (Arrays.binarySearch(l, j) >= 0) d[i][j] = Math.min(d[i][j], d[i - 1][j] + 1);
                if (Arrays.binarySearch(e, j) >= 0) d[i][j] = Math.min(d[i][j], d[i - v][j]);
            }
        }
        for (int i = 0; i < q; i++) {
            int x1 = sc.nextInt();
            int y1 = sc.nextInt();
            int x2 = sc.nextInt();
            int y2 = sc.nextInt();
            System.out.println(d[x2][y2] - d[x1][y1]);
        }
    }
}
