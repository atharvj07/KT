#include <vector>
#include <limits>
#include <iostream>
#include <utility>
#include <algorithm>

using namespace std;

int main() {
    long t;
    cin >> t;
    while(t--) {
        long n, x;
        cin >> n >> x;
        vector<pair<long, long>> blows(n);
        for (long i = 0; i < n; i++) {
            cin >> blows[i].first >> blows[i].second;
        }
        sort(blows.begin(), blows.end(), [](const auto& a, const auto& b) {
                return a.first < b.first;
        });
        long ans = numeric_limits<long>::max(), curX = x;
        for (long i = 0; i < n; i++) {
            long d = blows[i].first, h = blows[i].second;
            if (min(d, curX) <= 0) {
                continue;
            }
            curX = min(d, curX) - h;
            ans = min(ans, 1 + (curX + d - 1) / d);
        }
        if (ans == numeric_limits<long>::max()) {
            ans = -1;
        }
        cout << ans << endl;
    }
    return 0;
}
