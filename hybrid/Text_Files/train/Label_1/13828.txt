#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1000000007;

int main() {
    int n;
    cin >> n;
    vector<ll> a(n), p(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 1; i < n; i++) { cin >> p[i]; p[i]--; }
    vector<vector<int>> g(n);
    for (int i = 1; i < n; i++) { g[p[i]].push_back(i); }
    vector<int> sz(n), depth(n);
    function<void(int, int)> dfs1 = [&](int u, int pr) {
        sz[u] = 1;
        depth[u] = depth[pr] + 1;
        for (int v : g[u]) {
            if (v != pr) { dfs1(v, u); sz[u] += sz[v]; }
        }
    };
    dfs1(1, 1);
    vector<int> dp(n);
    function<void(int, int)> dfs2 = [&](int u, int pr) {
        int odd = a[u];
        for (int v : g[u]) {
            if (v != pr) { dfs2(v, u); odd += dp[v]; }
        }
        if (depth[u] % 2 == 0)
            dp[u] = (a[u] + odd) % 2;
        else
            dp[u] = (a[u] - odd) % 2;
    };
    dfs2(1, 1);
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j : g[i]) {
            if (i == 1 || j == 1 || sz[i] == n || sz[j] == n)
                ans += dp[1] != dp[i] && dp[1] != dp[j];
        }
    }
    cout << ans / 2 << '\n';
    return 0;
}
