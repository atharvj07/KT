import sys

input = sys.stdin.readline

n, k = map(int, input().split())
a = []
for _ in range(n):
    l, r = map(int, input().split())
    a.append((l, r))
a.sort(key=lambda x: x[1])
ans = 0
for i in range(n):
    l, r = a[i]
    if (l + r) % k == 0:
        continue
    if (r + 1) % k == 0:
        ans += 1
        continue
    if (l - 1) % k == 0:
        ans += 1
        continue
    ans += 2
print(ans)
