import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Circle> circles = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int r = scanner.nextInt();
            circles.add(new Circle(x, r));
        }
        int m = scanner.nextInt();
        List<Point> points = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            points.add(new Point(x, y));
        }
        int hitCount = 0;
        List<Integer> hitIndices = new ArrayList<>();
        for (Circle circle : circles) {
            int hitIndex = -1;
            for (int i = 0; i < points.size(); i++) {
                Point point = points.get(i);
                if (circle.isInside(point)) {
                    hitIndex = i + 1;
                    break;
                }
            }
            hitIndices.add(hitIndex);
            if (hitIndex != -1) {
                hitCount++;
            }
        }
        System.out.println(hitCount);
        for (int hitIndex : hitIndices) {
            System.out.print(hitIndex + " ");
        }
    }

    private static class Circle {
        private int x;
        private int r;

        public Circle(int x, int r) {
            this.x = x;
            this.r = r;
        }

        public boolean isInside(Point point) {
            return (point.x - x) * (point.x - x) + point.y * point.y <= r * r;
        }
    }

    private static class Point {
        private int x;
        private int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
