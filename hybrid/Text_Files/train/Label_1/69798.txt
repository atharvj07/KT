import bisect

n, s = map(int, input().split())
orders = []
for _ in range(n):
    d, p, q = input().split()
    orders.append((d, int(p), int(q)))
orders.sort(key=lambda o: (o[0], -o[1]))

buy = []
sell = []
for d, p, q in orders:
    if d == 'B':
        i = bisect.bisect_left(buy, (p, -q))
        buy.insert(i, (p, -q))
    else:
        i = bisect.bisect_left(sell, (p, q))
        sell.insert(i, (p, q))

for p, q in sell[:s]:
    print('S', p, q)
for p, q in buy[:s]:
    print('B', p, -q)
