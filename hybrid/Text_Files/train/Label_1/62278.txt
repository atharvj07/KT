import heapq as hq
import sys

n, m, k = map(int, input().split())
g = [[] for _ in range(n+1)]
for _ in range(m):
    x, y, w = map(int, input().split())
    g[x].append((y,w))
    g[y].append((x,w))
vis = [False for _ in range(n+1)]
ans = []
pq = []
hq.heappush(pq, (0,1))
while pq and len(ans) < k:
    d, u = hq.heappop(pq)
    if vis[u]:
        continue
    vis[u] = True
    ans.append(d)
    for v, w in g[u]:
        if not vis[v]:
            hq.heappush(pq, (d+w,v))
print(ans[-1])
