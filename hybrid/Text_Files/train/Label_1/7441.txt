#include <bits/stdc++.h>
using namespace std;
#define ll long long int
const int mod = 1e9 + 7;

int power(ll x, ll y) {
    ll res = 1;
    while (y > 0) {
        if (y & 1)
            res = (res * x) % mod;
        y = y >> 1;
        x = (x * x) % mod;
    }
    return res;
}

int main() {
    ll t;
    cin >> t;
    while (t--) {
        ll n, f;
        cin >> n >> f;
        ll ans = power(f, n - 1) % mod;
        cout << ans << endl;
    }
    return 0;
}
