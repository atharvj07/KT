#include <cstdio>
#include <vector>
#define int long long
using namespace std;
const int maxn = 2e5 + 10;
int n, a[maxn], k;

signed main() {
    scanf("%lld", &k);
    while (k--) {
        scanf("%lld", &n);
        vector<int> freq(n + 1), pre(n + 1);
        for (int i = 1; i <= n; ++i) scanf("%lld", &a[i]), ++freq[a[i]];
        for (int i = 1; i <= n; ++i) pre[i] = pre[i - 1] + freq[i];
        int ans = 1e18;
        for (int i = 1; i <= n; ++i) {
            int cnt = pre[i] - pre[i - freq[a[i]]];
            for (int j = freq[a[i]] + 1; j <= n; ++j)
                if (cnt + freq[j] > n / 2) {
                    ans = min(ans, j - i + 1);
                    break;
                }
        }
        printf("%lld\n", ans == 1e18 ? -1 : ans);
    }
}
