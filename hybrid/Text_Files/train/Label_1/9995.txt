#include <bits/stdc++.h>
using namespace std;
struct node {int t, v[2];};
node a[1000001];
int out, vis[1000001];

int dfs(int u) {
    if (vis[u] == 2) return out;
    vis[u] = 2;
    if (a[u].t == 0) return a[u].v[0];
    int tmp = dfs(a[u].v[0]);
    if (a[u].t == 4) tmp = (!tmp);
    else tmp = (tmp ? 1 : 0);
    if (a[u].t != 4) tmp = (tmp ? (dfs(a[u].v[1]) ? 1 : 0) : (dfs(a[u].v[1]) ? 0 : 1));
    vis[u] = 1;
    if (u == 1) out = tmp;
    return tmp;
}

int main() {
    ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int n, u, v, x; cin >> n;
    for (int i=1; i<=n; ++i) {
        string s; cin >> s;
        if (s == "IN") {
            cin >> x;
            a[i] = {0, {x, -1}};
        }
        else if (s == "NOT") {
            cin >> u;
            a[i] = {4, {u, -1}};
        }
        else {
            cin >> u >> v;
            if (s == "AND") a[i] = {1, {u, v}};
            else if (s == "OR") a[i] = {2, {u, v}};
            else if (s == "XOR") a[i] = {3, {u, v}};
        }
    }
    dfs(1);
    for (int i=2; i<=n; ++i) {
        if (a[i].t == 0) a[i].v[0] = (!a[i].v[0]), vis[i] = 0;
        dfs(1);
        cout << out;
        if (a[i].t == 0) a[i].v[0] = (!a[i].v[0]), vis[i] = 0;
    }
    return 0;
}
