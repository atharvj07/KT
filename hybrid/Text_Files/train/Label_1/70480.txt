from collections import Counter
n = int(input())
init = list(map(int, input().split()))+[0]
cnt = Counter(init)
ans = 0
for i in range(n-1, 0, -1):
    if init[i] > init[i-1]:
        cnt[init[i]] -= 1
        cnt[init[i-1]] += 1
        init[i] = init[i-1]
        ans += 1
        if cnt[init[i]] >= init[i]:
            break
print(ans)
