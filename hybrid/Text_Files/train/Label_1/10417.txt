#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1000;

int n, m;
vector<int> adj[MAXN + 1];
bool vis[MAXN + 1];
vector<int> path;
int diameter;

void dfs(int u, int p) {
    vis[u] = true;
    path.push_back(u);
    for (int v : adj[u]) {
        if (v == p) continue;
        if (vis[v]) {
            diameter = max(diameter, (int)path.size());
            continue;
        }
        dfs(v, u);
    }
    path.pop_back();
}

int main() {
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) {
            path.clear();
            dfs(i, -1);
        }
    }
    int u = path[0], v = path[diameter / 2];
    if (diameter % 2 == 0) {
        v = path[diameter / 2 - 1];
    }
    cout << diameter << endl;
    for (int i = 1; i <= n; i++) {
        if (i == u || i == v) continue;
        cout << u << ' ' << i << endl;
    }
    return 0;
}
