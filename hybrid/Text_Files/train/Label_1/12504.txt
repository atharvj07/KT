#include <bits/stdc++.h>
using namespace std;
int n, q, col[100005];
string s;
bool ok(int i) {
  int cnt[3] = {0, 0, 0};
  int last[3] = {-1, -1, -1};
  for (int j = i; j <= n; ++j) {
    if (col[j] != 0) {
      if (last[col[j]] + 1 == j) continue;
      return false;
    }
    int p = lower_bound(last + 1, last + 4, j) - last - 1;
    cnt[p]++;
    if (cnt[p] > 1) return false;
    last[p] = j;
  }
  return true;
}
int main() {
  cin >> n >> q;
  cin >> s;
  while (q--) {
    string t;
    cin >> t;
    if (t[0] == '+') {
      int i;
      char c;
      cin >> i >> c;
      col[n + 1] = i;
      s += c;
      n++;
    } else {
      int i;
      cin >> i;
      col[n] = 0;
      s.pop_back();
      n--;
    }
    cout << (ok(n - 1 + (t[0] == '-')) ? "YES\n" : "NO\n");
  }
}
