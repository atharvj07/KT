#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, x, y;
  cin >> n >> x >> y;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  int ans = 0;
  for (int i = 0; i < n; ++i)
    if (a[i] <= x) ++ans;
  else if (a[i] <= x + y) ++ans, a[i] = 0;
  cout << ans << endl;
  return 0;
}
