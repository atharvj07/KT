import heapq
from collections import defaultdict

n, m, k = map(int, input().split())
arr = list(map(int, input().split()))
d = [0] * k
g = defaultdict(list)
for i in range(m):
    a, b, w = map(int, input().split())
    g[a].append((w, b))
    g[b].append((w, a))
for i in range(k):
    dist = [10 ** 18] * (n + 1)
    dist[arr[i]] = 0
    q = [(0, arr[i])]
    while q:
        w, u = heapq.heappop(q)
        if dist[u] < w: continue
        for e, v in g[u]:
            if dist[v] > w + e:
                dist[v] = w + e
                heapq.heappush(q, (dist[v], v))
    for j in range(k):
        d[i] = max(d[i], dist[arr[j]])
print(*d)
