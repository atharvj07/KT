#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (s[i - 1] == '>' && (i == n || s[i] == '<')) {
            ans++;
        } else if (s[i - 1] == '<' && (i == 1 || s[i - 2] == '>')) {
            ans++;
        }
    }
    cout << ans;
    return 0;
}
