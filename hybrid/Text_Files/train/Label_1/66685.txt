
from sys import stdin,stdout

def I(): return int(stdin.readline())
def L(): return map(int,stdin.readline().split())

n,m,k=L()
W=list(L())
A=[[] for i in range(n+1)]
for i in range(1,m+1):
    a,b=L()
    A[a].append((i,b))
    A[b].append((i,a))
S=[0 for i in range(k+1)]
L=[(i,W[i-1]) for i in range(1,n+1)]
L.sort(key=lambda x:x[1],reverse=True)
t=[0 for i in range(m+1)]
for p,w in L:
    cnt=[0 for i in range(k+1)]
    aa=set()
    for game,opp in A[p]:
        if t[game]: continue
        cur=S[t[game]]
        if cur>0:cnt[cur]-=1
        cur+=1
        if cur>0:cnt[cur]+=1
        if cur>2:
            for x in aa: S[x]-=1
            aa.clear()
            cnt=[0 for i in range(k+1)]
            cnt[cur]+=1
        aa.add(t[game])
        if abs(max(cnt)-min(cnt))<=2:
            S[t[game]]=cur
            t[game]=t[opp]=S[t[game]]
stdout.write(' '.join([str(x) for x in t[1:]]))
