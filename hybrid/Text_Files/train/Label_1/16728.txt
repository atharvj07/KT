#include<bits/stdc++.h>
using namespace std;
int32_t main() {
	int n; cin >> n; string s; cin >> s; int ans = 0;
	for (int i = 0, x = 0, y = 0; i < n; i++) {
		if (s[i] == 'U') { if (y < 0) { ans++; y++; } else y++; }
		if (s[i] == 'D') { if (y > 0) { ans++; y--; } else y--; }
		if (s[i] == 'L') { if (x > 0) { ans++; x--; } else x--; }
		if (s[i] == 'R') { if (x < 0) { ans++; x++; } else x++; }
	}
	cout << ans << endl;
}
