from sys import stdin

n, k = map(int, stdin.readline().split())
matrix = [list(stdin.readline().strip()) for _ in range(n)]

ans = ['a'] * (2 * n - 1)
for i in range(n):
    for j in range(n - i):
        if k > 0 and matrix[i + j][i] > ans[i + j]:
            ans[i + j] = matrix[i + j][i]
            k -= 1
print(''.join(ans))
