import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        System.out.println((n -phi(n)));
    }

    static long phi(long n) {
        long result =n;
        for (long i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                result -= result / i;
                while (n % i == 0) {
                    n /= i;
                }
            }
        }
        if (n > 1) {
            result -= result / n;
        }
        return result;
    }
}
