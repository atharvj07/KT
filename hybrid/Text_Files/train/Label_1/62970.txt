n = int(input())
a = list(map(int, input().split()))

sequence = [4, 8, 15, 16, 23, 42]
count = [0] * 6

for num in a:
    if num == 4:
        count[0] += 1
    else:
        for i in range(5):
            if num == sequence[i + 1] and count[i] > count[i + 1]:
                count[i + 1] += 1
                break

print(n - 6 * count[-1])
