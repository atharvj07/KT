n = int(input())
s = input()
mx = 0
cnt = 0
ans = 0
L = []
for i in s:
    if i == '(':
        L.append(0)
        if ans < cnt:
            ans = cnt
        cnt = 0
    elif i == ')':
        mx = max(L.pop(), mx)
        if ans < cnt:
            ans = cnt
        cnt = 0
    else:
        cnt += 1
else:
    L.append(0)
    if ans < cnt:
        ans = cnt
    cnt = 0
    mx = max(L.pop(), mx)
print(mx, ans)
