MOD = 10**9 + 7

def comb(n, k, fact, inv_fact):
    return fact[n] * inv_fact[k] % MOD * inv_fact[n-k] % MOD

n, m = map(int, input().split())
s = input()

open_brackets = s.count('(')
close_brackets = s.count(')')

min_open = max_open = balance = 0
for c in s:
    if c == '(':
        balance += 1
    else:
        balance -= 1
    min_open = min(min_open, balance)

max_open = open_brackets - min_open
min_close = close_brackets + min_open
max_close = n // 2

if max_open > max_close or max_close < 0:
    print(0)
else:
    fact = [1] * (n+1)
    inv_fact = [1] * (n+1)
    for i in range(2, n+1):
        fact[i] = fact[i-1] * i % MOD
    inv_fact[n] = pow(fact[n], MOD-2, MOD)
    for i in range(n-1, 1, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % MOD
    
    ans = 0
    for i in range(max_open, max_close+1):
        if n - m - i + balance >= 0:
            ans += comb(i - min_open, max_open - min_open, fact, inv_fact) * comb(max_close - i + balance, n - m - i + balance, fact, inv_fact)
            ans %= MOD
    print(ans)
