#include <bits/stdc++.h>
using namespace std;

const int maxn = 10;

int n, k1, k2, a[2][maxn], ans;

int main() {
	cin >> n;
	cin >> k1;
	for (int i = 1; i <= k1; i++) cin >> a[0][i];
	cin >> k2;
	for (int i = 1; i <= k2; i++) cin >> a[1][i];
	int i = 0;
	while (k1 && k2) {
		i++;
		if (a[0][k1] > a[1][k2]) {
			k2--;
			a[0][++k1] = a[1][k2];
			a[0][++k1] = a[0][k1];
		} else {
			k1--;
			a[1][++k2] = a[0][k1];
			a[1][++k2] = a[1][k2];
		}
	}
	if (k1) {
		ans = i;
		cout << ans << ' ' << 1 << endl;
	} else if (k2) {
		ans = i;
		cout << ans << ' ' << 2 << endl;
	} else {
		cout << -1 << endl;
	}
	return 0;
}
