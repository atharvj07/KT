import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;
import java.util.Set;
import java.util.TreeSet;

public class Main {
    static int[][] dir = {{0, 1}, {1, 0}};

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        Set<Integer> banned = new TreeSet<>();
        for (int i = 0; i < m; i++) banned.add(scanner.nextInt() * 1000 + scanner.nextInt());
        Queue<Integer> q = new LinkedList<>();
        for (int i = 1; i <= n; i++) {
            if (!banned.contains(i * 1000 + 1) && !banned.contains(i * 1000 + n)) {
                q.add(i * 1000 + 1);
                q.add(i * 1000 + n);
            }
            if (!banned.contains(1 * 1000 + i) && !banned.contains(n * 1000 + i)) {
                q.add(1 * 1000 + i);
                q.add(n * 1000 + i);
            }
        }
        Set<Integer> s = new TreeSet<>();
        while (!q.isEmpty()) {
            int v = q.poll();
            if (s.contains(v)) continue;
            s.add(v);
        }
        System.out.println(s.size() / 2);
    }
}
