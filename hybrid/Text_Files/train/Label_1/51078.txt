import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        List<int[]>[] adj = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int l = sc.nextInt();
            adj[a].add(new int[] { b, l });
            adj[b].add(new int[] { a, l });
        }
        List<Integer> malls = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            malls.add(sc.nextInt());
        }
        int minDist = Integer.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            int dist = bfs(adj, i, malls);
            minDist = Math.min(minDist, dist);
        }
        System.out.println(minDist);
    }

    public static int bfs(List<int[]>[] adj, int start, List<Integer> malls) {
        Queue<int[]> q = new LinkedList<>();
        q.add(new int[] { start, 0 });
        boolean[] visited = new boolean[adj.length];
        while (!q.isEmpty()) {
            int[] curr = q.poll();
            int node = curr[0];
            int dist = curr[1];
            if (visited[node]) {
                continue;
            }
            visited[node] = true;
            if (malls.contains(node)) {
                return dist;
            }
            for (int[] nbr : adj[node]) {
                q.add(new int[] { nbr[0], dist + nbr[1] });
            }
        }
        return Integer.MAX_VALUE;
    }
}
