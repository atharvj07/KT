import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = cin.nextInt();
        }
        int m = cin.nextInt();
        for (int i = 0; i < m; i++) {
            int x = cin.nextInt() - 1;
            int y = cin.nextInt() - 1;
            for (int j = y; j < a[x]; j++) {
                if (x - 1 >= 0) {
                    a[x - 1]++;
                }
                if (x + 1 < n) {
                    a[x + 1]++;
                }
            }
            a[x] -= y;
        }
        for (int i = 0; i < n; i++) {
            System.out.println(a[i]);
        }
    }
}
