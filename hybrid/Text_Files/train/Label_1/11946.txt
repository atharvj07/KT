#include <bits/stdc++.h>

using namespace std;
typedef pair<int, int> pii;

bool valid(vector<int>& a, vector<int>& b, int n) {
    vector<pii> q;
    for (int i = 0; i < n; ++i) {
        if (a[i] != b[i]) {
            q.push_back({a[i], b[i]});
        }
    }
    sort(q.begin(), q.end());
    int m = q.size();
    if (m % 2 == 1) return false;
    for (int i = 0; i < m; i += 2) {
        if (q[i].second != q[i + 1].first) return false;
    }
    return true;
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n), b(n);
        for (int& x : a) cin >> x;
        for (int& x : b) cin >> x;
        cout << (valid(a, b, n) ? "Yes" : "No") << '\n';
    }
    return 0;
}
