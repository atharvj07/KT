#include<bits/stdc++.h>
using namespace std;

const int maxn=2e5+5;

struct seg{
    long long l,r;
}s[maxn];

typedef long long ll;
int n,t;
ll ans;
int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin>>t;
    while(t--){
        cin>>n;
        map<pair<ll,ll>,int> mp;
        for(int i=1;i<=n;i++){
            cin>>s[i].l>>s[i].r;
            mp[make_pair(s[i].l,s[i].r)]++;
        }
        for(auto x:mp)  ans=max(ans,(ll)x.second);
        cout<<ans<<endl;
        ans=0;
    }
    return 0;
}
