#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_map<int, int> cnt;
  for (int a, i = 0; i < n; ++i) {
    cin >> a, ++cnt[a];
  }
  priority_queue<pair<int, int>> pq;
  for (auto &c : cnt) pq.emplace(c.second, c.first);
  int ans = 0;

  while (!pq.empty()) {
    int c, t;
    tie(c, t) = pq.top(), pq.pop();

    if (c <= ans) break;
    if (ans <= 0) {
      ans += c;
    } else
      ans *= 2, c -= ans + 1;
    if (c > 0) pq.emplace(c, t);
  }
  cout << ans << endl;
  return 0;
}
