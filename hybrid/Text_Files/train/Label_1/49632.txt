import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Permutations {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        List<List<Integer>> permutations = permute(n);
        for (List<Integer> permutation : permutations) {
            for (int num : permutation) {
                System.out.print(num + " ");
            }
            System.out.println();
        }
    }

    private static List<List<Integer>> permute(int n) {
        List<List<Integer>> permutations = new ArrayList<>();
        if (n == 1) {
            List<Integer> basePermutation = new ArrayList<>();
            basePermutation.add(1);
            permutations.add(basePermutation);
            return permutations;
        }

        List<List<Integer>> subPermutations = permute(n - 1);
        for (List<Integer> subPermutation : subPermutations) {
            for (int i = 0; i < n; i++) {
                List<Integer> permutation = new ArrayList<>(subPermutation);
                permutation.add(i, n);
                permutations.add(permutation);
            }
        }

        return permutations;
    }
}
