import java.util.*;

class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long n = sc.nextInt();
    long k = sc.nextInt();
    System.out.println(pow(k, n));
  }

  private static long pow(long a, long b) {
    long res = a;
    for (long i = 0; i < b - 1; i++) {
      res *= a;
      res %= 1000000007;
    }
    return res;
  }
}
