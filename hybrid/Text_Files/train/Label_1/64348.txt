MOD = 1000000007

def find_set(parent, x):
    if parent[x] != x:
        parent[x] = find_set(parent, parent[x])
    return parent[x]

def union_sets(parent, rank, x, y):
    x = find_set(parent, x)
    y = find_set(parent, y)
    if x != y:
        if rank[x] < rank[y]:
            x, y = y, x
        parent[y] = x
        if rank[x] == rank[y]:
            rank[x] += 1

n, m = map(int, input().split())
parent = [i for i in range(2*n)]
rank = [0] * (2*n)

for _ in range(m):
    a, b, c = map(int, input().split())
    a -= 1
    b -= 1
    if c == 1:
        union_sets(parent, rank, a, b)
        union_sets(parent, rank, a+n, b+n)
    else:
        union_sets(parent, rank, a, b+n)
        union_sets(parent, rank, b, a+n)

for i in range(n):
    if find_set(parent, i) == find_set(parent, i+n):
        print(0)
        break
else:
    ans = 1
    for _ in range(n//2):
        ans = ans * 2 % MOD
    print(ans)
