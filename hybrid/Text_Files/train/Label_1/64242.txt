n = int(input())
a = [int(x) for x in input().split()]
a.sort()

blocks = []
for i in range(0, n, 3):
    if a[i] >= a[i + 1] or a[i + 1] >= a[i + 2] or a[i] * a[i + 2] != a[i + 1] * a[i + 1]:
        print(-1)
        exit()
    blocks.append([a[i], a[i + 1], a[i + 2]])

for b in blocks:
    print(*b)
