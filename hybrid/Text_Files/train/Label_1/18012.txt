#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int maxn = 1e3+5;
const ll INF = 1e9;
int n, m, x[maxn], ti, ans, cnt; ll a[maxn], st[maxn], en[maxn];
int main() {
    cin >> n >> m;
    for(int i = 1; i <= n; i++) cin >> a[i];
    for(int i = 1; i <= m; i++) cin >> st[i] >> en[i];
    sort(st+1, st+1+m); sort(en+1, en+1+m);
    for(int i = 1; i <= n; i++) {
        ti += a[i];
        bool f = false;
        while(cnt < m && ti >= st[cnt+1]) cnt++;
        if(cnt < m && ti <= en[cnt+1]) f = true;
        if(!f) { ans = -1; break;}
    }
    cout << ans;
    return 0;
}
