n, k = map(int, input().split())
skills = list(map(int, input().split()))

improvements_needed = [10 - skill % 10 if skill % 10 != 0 else 0 for skill in skills]
improvements_needed.sort()

total_rating = sum(skill // 10 for skill in skills)

for improvement in improvements_needed:
    if k >= improvement:
        k -= improvement
        total_rating += 1
    else:
        break

total_rating += k // 10
if total_rating > n * 10:
    total_rating = n * 10

print(total_rating)
