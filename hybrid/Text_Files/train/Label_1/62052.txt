import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = [1, 2, 3]
    ans = 0
    a.sort()
    for i in a:
        while i > 0:
            if i >= b[-1]:
                i -= b[-1]
                ans += 1
            else:
                b.pop()
    print(ans)
