import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
even = []
odd = []
ex = []
for i in a:
    if i % 2 == 0:
        even.append(i)
    else:
        odd.append(i)
if len(even) != len(odd):
    print(-1)
else:
    for i in range(len(even)):
        ex.append(odd[i])
        ex.append(even[i])
    print(len(ex) // 2)
    print(*ex)
