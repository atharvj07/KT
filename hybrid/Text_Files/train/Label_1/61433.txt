def calculate_force(n, A, cf, cm, m, skills):
    skills.sort()
    total_cost = sum(A - skill for skill in skills)
    if total_cost <= m:
        return cf * n + cm * A, [A] * n
    
    skills.append(A)
    prefix_sum = [0]
    for skill in skills:
        prefix_sum.append(prefix_sum[-1] + skill)
    
    max_force = 0
    best_combination = []
    for i in range(n + 1):
        perfected = n - i
        available_money = m - (n - i) * (A - skills[i])
        if available_money < 0:
            break
        low = 0
        high = skills[i]
        
        while low < high:
            mid = (low + high + 1) // 2
            required_money = mid * (i + 1) - prefix_sum[i + 1]
            if required_money <= available_money:
                low = mid
            else:
                high = mid - 1
                
        min_skill = low
        current_force = cf * perfected + cm * min_skill
        
        if current_force > max_force:
            max_force = current_force
            best_combination = [max(min_skill, skill) for skill in skills[:i]] + [A] * (n - i)
    
    return max_force, best_combination[:-1]

n, A, cf, cm, m = map(int, input().split())
skills = list(map(int, input().split()))
max_force, best_combination = calculate_force(n, A, cf, cm, m, skills)
print(max_force)
print(' '.join(map(str, best_combination)))
