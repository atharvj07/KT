import java.util.*;

class Main {
  static final int INF = 1 << 29;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    int W = sc.nextInt();
    int H = sc.nextInt();
    int[][] pos = new int[N][2];
    for (int i = 0; i < N; i++) {
      pos[i][0] = sc.nextInt() - 1;
      pos[i][1] = sc.nextInt() - 1;
    }

    int[][] dist = new int[N][N];
    for (int i = 0; i < N; i++) {
      for (int j = 0; j < N; j++) {
        dist[i][j] = INF;
      }
    }

    for (int i = 0; i < N; i++) {
      int[] dx = {0, 0, 1, -1};
      int[] dy = {1, -1, 0, 0};
      Queue<Integer> que = new LinkedList<>();
      que.add(i);
      dist[i][i] = 0;
      while (!que.isEmpty()) {
        int now = que.poll();
        for (int dir = 0; dir < 4; dir++) {
          int nx = pos[now][0] + dx[dir];
          int ny = pos[now][1] + dy[dir];
          if (0 <= nx && nx < W && 0 <= ny && ny < H) {
            for (int j = 0; j < N; j++) {
              if (pos[j][0] == nx && pos[j][1] == ny) {
                if (dist[now][j] == INF) {
                  que.add(j);
                  dist[now][j] = dist[now][i] + 1;
                }
              }
            }
          }
        }
      }
    }

    int ans = INF;
    for (int i = 0; i < N; i++) {
      for (int j = i + 1; j < N; j++) {
        ans = Math.min(ans, dist[i][j]);
      }
    }
    System.out.println(ans);
  }
}
