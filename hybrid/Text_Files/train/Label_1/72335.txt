import string

s = input()
alphabets = set(string.ascii_uppercase)
occurrences = [0] * 26
for ch in s:
    if ch != '?':
        occurrences[ord(ch) - ord('A')] += 1

nice = True
for occurrence in occurrences:
    if not (occurrence == 0 or occurrence == 1):
        nice = False

if not nice or s.count('?') < 26 - occurrences.count(1):
    print(-1)
else:
    result = []
    for ch in s:
        if ch == '?':
            for i in range(26):
                if occurrences[i] == 0:
                    occurrences[i] = 1
                    result.append(chr(ord('A') + i))
                    break
        else:
            result.append(ch)
    print(''.join(result))
