#include<bits/stdc++.h>
#define int long long
using namespace std;

int b[604], c[604], r[604][604];
int n, m;

int32_t main(){
    cin>>n>>m;
    for(int i=1; i<=n; i++) cin>>b[i];
    for(int i=1; i<=n; i++){
        for(int j=0; j<=m; j++){
            if(j+b[i]<=m){
                c[j+b[i]]=1;
                r[i][j+b[i]]=1;
            }
            if(!r[i-1][j]) r[i][j]=1;
        }
        for(int j=0; j<=m; j++) c[j]|=r[i][j];
    }
    int q=0;
    for(int i=0; i<=m; i++) if(c[i]) q++;
    cout<<q<<endl;
    for(int i=0; i<=m; i++) if(c[i]) cout<<i<<' ';
}
