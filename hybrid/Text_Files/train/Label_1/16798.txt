#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,a[100];
    cin>>n;
    for(int i=1;i<=n;i++) cin>>a[i];
    vector<int> ans;
    int cnt=0;
    for(int i=n;i>=1;i--){
        cnt+=a[i];
        ans.push_back(cnt);
    }
    reverse(ans.begin(), ans.end());
    for(int x: ans) cout<<x<<" ";
    cout<<"\n";
    return 0;
}
