#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> c(n), a(m);
  for (int i = 0; i < n; i++) cin >> c[i];
  for (int j = 0; j < m; j++) cin >> a[j];
  int ans = 0;
  for (int i = 0, j = 0; i < n; i++) {
    if (j < m && a[j] >= c[i]) {
      j++;
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
