import heapq
n,m=map(int,input().split())
d={i:[] for i in range(1,n+1)}
for i in range(m):
    x,y,z=map(int,input().split())
    d[x].append((y,z))
    d[y].append((x,z))
dst=[[1000000000]*(n+1) for i in range(n+1)]
vis=[False]*(n+1)
q=[]
dst[1][1]=0
heapq.heappush(q,(0,1))
ans=[]
while q:
    p,pos=heapq.heappop(q)
    if vis[pos]==True:continue
    vis[pos]=True
    if pos==n:break
    for i in d[pos]:
        if vis[i[0]]==True:continue
        w=0
        if i[1]==0:w=1
        if dst[1][pos]+w<dst[1][i[0]]:
            dst[1][i[0]]=dst[1][pos]+w
            heapq.heappush(q,(dst[1][i[0]],i[0]))
            if i[1]==0:ans.append((pos,i[0],1))
            if i[1]==1:ans.append((pos,i[0],0))
print(len(ans))
for i in ans:
    print(*i)
