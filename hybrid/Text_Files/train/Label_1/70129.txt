from queue import Queue

def knights_meet(board):
    q = Queue()
    for i in range(8):
        for j in range(8):
            if board[i][j] == 'K':
                q.put((i, j, 0))
    while not q.empty():
        x, y, z = q.get()
        if x < 0 or x >= 8 or y < 0 or y >= 8:
            continue
        if board[x][y] == '#' and z == 0:
            continue
        if board[x][y] == '#' and z == 1:
            continue
        if board[x][y] == 'K' and z == 1:
            return True
        if z == 0:
            q.put((x-2, y+2, z))
            q.put((x-2, y-2, z))
            q.put((x+2, y+2, z))
            q.put((x+2, y-2, z))
            board[x][y] = '#'
        if z == 1:
            q.put((x-1, y+2, z+1))
            q.put((x-1, y-2, z+1))
            q.put((x+1, y+2, z+1))
            q.put((x+1, y-2, z+1))
            board[x][y] = '#'
    return False

for _ in range(int(input())):
    board = [list(input()) for _ in range(8)]
    print("YES" if knights_meet(board) else "NO")
