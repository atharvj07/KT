#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> c(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> c[i][j]; }
  }
  int ans = -1;
  for (int i = 0; i < n; i++) {
    int x = *min_element(c[i].begin(), c[i].end());
    ans = ans == -1 ? x : max(x, ans);
  }
  cout << ans << endl;
  return 0;
}
