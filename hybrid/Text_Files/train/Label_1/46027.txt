import java.io.BufferedInputStream;
import java.util.Arrays;
import java.util.Scanner;

class Main {
    static long[] a, b;

    public static void main(String[] args) {
        Scanner sc = new Scanner(new BufferedInputStream(System.in));
        int n = sc.nextInt();
        a = new long[n];
        b = new long[n];
        a[0] = sc.nextLong();
        b[0] = a[0];
        for (int i = 1; i < n; i++) {
            long x = sc.nextLong();
            b[i] = x;
            a[i] = a[i - 1] ^ x;
        }
        long[] c = a.clone();
        Arrays.sort(c);
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            if (c[i] != a[i]) {
                ok = false;
                break;
            }
        }
        if (!ok) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            for (long x : b) {
                System.out.print(x + " ");
            }
            System.out.println();
        }
    }
}
