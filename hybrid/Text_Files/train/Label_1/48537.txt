import java.util.Scanner;
import java.util.Arrays;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    int N = sc.nextInt();
    int K = sc.nextInt();
    int[] A = new int[N];
    for (int i = 0; i < N; i++) {
      A[i] = sc.nextInt();
    }

    Arrays.sort(A);

    int min = N;
    for (int i = 0; i < N - K; i++) {
      int diff = A[i + K - 1] - A[i];
      min = Math.min(min, diff);
    }

    System.out.println(min);
  }
}
