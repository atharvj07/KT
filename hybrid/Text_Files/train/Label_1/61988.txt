from sys import stdin
n, p = map(int, stdin.readline().split())
c = list(map(int, stdin.readline().split()))

mod = 10**9 + 7
dp = [[[[-1] * 2 for _ in range(2)] for _ in range(2)] for _ in range(n + 1)]

def solve(i, painted, odd, last):
  if i == n:
    if painted == n and odd == p:
      return 1
    return 0

  if dp[i][painted][odd][last] != -1:
    return dp[i][painted][odd][last]

  ret = 0
  if c[i] == -1:
    for color in range(2):
      ret += solve(i + 1, painted + (color == 1), odd ^ (color == last), color)
  else:
    ret += solve(i + 1, painted + (c[i] == 1), odd ^ (c[i] == last), c[i])

  dp[i][painted][odd][last] = ret % mod
  return ret

print(solve(0, 0, 0, -1))
