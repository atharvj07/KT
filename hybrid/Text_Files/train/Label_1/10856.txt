#include<bits/stdc++.h>
using namespace std;
#define int long long
#define pb push_back
#define mp make_pair
const int mod=1e9+7;
int power(int x,int y){int ret=1;while(y){if(y&1)ret=ret*x;x=x*x;y>>=1;}return ret;}
main()
{
    int n,k;
    cin>>n>>k;
    vector<int>v(n),ans(n,0);
    unordered_map<int,int>m;
    for(int i=0;i<n;i++)
    {
        cin>>v[i];
        if(m.find(v[i])!=m.end())ans[i]=m[v[i]];
        int p=k,q=2;
        while(p>=q)
        {
            if(p==q)
            {
                ans[i]+=m[power(v[i],q)];
            }
            else
            {
                ans[i]+=m[power(v[i],p)]+m[power(v[i],q)];
            }
            q++;
            p--;
        }
        m[v[i]]++;
    }
    int sum=0;
    for(int i=0;i<n;i++)
    {
        sum+=ans[i];
    }
    cout<<sum<<endl;
}
