import sys
input = sys.stdin.readline

def solve(n, s, t):
    if s == t: return 0
    mis = [0 for i in range(26)]
    for c in s: mis[ord(c) - ord('a')] += 1
    for c in t: mis[ord(c) - ord('a')] -= 1
    if any(mis): return -1
    return abs(n // 2 - s.count(t[n // 2]))

q = int(input())
for _ in range(q):
    n = int(input())
    s = input().strip()
    t = input().strip()
    print(solve(n, s, t))
