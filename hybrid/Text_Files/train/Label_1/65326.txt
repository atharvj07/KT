def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def next_prime(n):
    while True:
        n += 1
        if is_prime(n):
            return n

n, m = map(int, input().split())
matrix = [list(map(int, input().split())) for _ in range(n)]

min_moves = float('inf')

for i in range(n):
    moves = 0
    for j in range(m):
        next_p = next_prime(matrix[i][j])
        moves += next_p - matrix[i][j]
    min_moves = min(min_moves, moves)

for j in range(m):
    moves = 0
    for i in range(n):
        next_p = next_prime(matrix[i][j])
        moves += next_p - matrix[i][j]
    min_moves = min(min_moves, moves)

print(min_moves)
