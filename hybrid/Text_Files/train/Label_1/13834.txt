#include <iostream>

using namespace std;

int n, k;
char a[100001];
void flip(int l, int r) { for (int i = l; i <= r; i++) a[i] = a[i] == '0' ? '1' : '0'; }
bool check() {
    if (k == 1) return 0;
    int pre = 0;
    for (int i = 1; i <= n; i++) {
        if (a[i] != a[i - 1]) pre = i;
    }
    if (pre == 0 || pre + k - 1 > n) return 0;
    for (int i = pre; i + k - 1 <= n; i++) {
        if (i == pre) flip(i, i + k - 1);
        else flip(i, i + k - 2);
        if (check()) return 1;
        flip(i, i + k - 1);
    }
    return 0;
}

int main() {
    cin >> n >> k;
    cin >> a;
    if (n > 2 * k + 10) {
        cout << "once again" << endl;
        return 0;
    }
    if (check()) {
        cout << "tokitsukaze" << endl;
    } else {
        cout << "quailty" << endl;
    }
    return 0;
}
