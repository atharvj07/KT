#include <bits/stdc++.h>

using namespace std;

const int maxn =125;
int a[maxn];
int n;

// d means whether the current number is even, the next number is odd
// and whether the complexity can be reduced
int f[maxn][2];

int main()
{
    int ans = 0;
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
        if (a[i] == 0)
        {
            a[i] = a[i - 1] + 1;
        }
    }
    for (int j = 1; j <= 1; j++)
        f[1][j] = 1;
    for (int i = 2; i <= n; i++)
        for (int j = 0; j <= 1; j++)
        {
            if (j == 0 && a[i] % 2 != a[i - 1] % 2)
            {
                f[i][j] = 1;
            }
            else if (j == 1 && a[i] % 2 == a[i - 1] % 2)
            {
                f[i][j] = 1;
            }
            else
            {
                f[i][j] = f[i - 1][!j];
            }
            ans = max(ans, f[i][j]);
        }
    cout << f[n][0] << endl;
    return 0;
}
