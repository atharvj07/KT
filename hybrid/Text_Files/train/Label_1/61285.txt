import sys

def f(s):
    ans = 0
    for i in range(len(s)):
        num = 0
        for j in range(i, len(s)):
            num = (num << 1) + int(s[j])
            if num == j - i + 1: ans += 1
    return ans

t = int(sys.stdin.readline())
for _ in range(t):
    s = sys.stdin.readline().strip()
    print(f(s))
