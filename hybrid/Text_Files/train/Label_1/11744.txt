#include <bits/stdc++.h>

using namespace std;

int main() {
  long long a;
  string s;
  cin >> a >> s;
  int ans = 0;
  for (int i = 0; i < s.size(); i++)
    for (int j = i + 1; j < s.size(); j++) {
      long long p = (s[i] - '0') * (s[j] - '0');
      if (p == a) ans++;
    }
  cout << ans << endl;
  return 0;
}
