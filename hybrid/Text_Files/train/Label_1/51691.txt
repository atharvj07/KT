import java.util.Scanner;

public class PashasGame {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        boolean[][] field = new boolean[n][m];
        int moveNumber = 0;

        for (int i = 0; i < k; i++) {
            int row = scanner.nextInt() - 1;
            int col = scanner.nextInt() - 1;

            field[row][col] = !field[row][col]; // Toggle the color of the pixel

            if (checkLoss(field, row, col)) {
                moveNumber = i + 1;
                break;
            }
        }

        System.out.println(moveNumber);
    }

    static boolean checkLoss(boolean[][] field, int row, int col) {
        // Check all four possible 2x2 squares around the colored pixel
        return (row > 0 && col > 0 && field[row - 1][col - 1] && field[row][col - 1] && field[row - 1][col]) ||
               (row > 0 && col < field[0].length - 1 && field[row - 1][col + 1] && field[row][col + 1] && field[row - 1][col]) ||
               (row < field.length - 1 && col > 0 && field[row + 1][col - 1] && field[row][col - 1] && field[row + 1][col]) ||
               (row < field.length - 1 && col < field[0].length - 1 && field[row + 1][col + 1] && field[row][col + 1] && field[row + 1][col]);
    }
}
