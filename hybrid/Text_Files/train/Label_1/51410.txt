import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] weights = new int[n];
        for (int i = 0; i < n; i++) {
            weights[i] = scanner.nextInt();
        }

        int[][] dp = new int[n + 1][k + 1];
        for (int i = 0; i < n + 1; i++) {
            for (int j = 0; j < k + 1; j++) {
                dp[i][j] = -1;
            }
        }

        dp[0][0] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                if (dp[i - 1][j] != -1) {
                    dp[i][j] = dp[i - 1][j];
                }
                if (j - weights[i - 1] >= 0 && dp[i - 1][j - weights[i - 1]] != -1) {
                    dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - weights[i - 1]] + 1);
                }
            }
        }

        int minRides = dp[n][k];
        if (minRides == -1) {
            System.out.println(-1);
            System.out.println(0);
        } else {
            int numWays = 0;
            for (int i = 0; i <= k; i++) {
                if (dp[n][i] == minRides) {
                    numWays++;
                }
            }

            System.out.println(minRides);
            System.out.println(numWays % 1000000007);
        }
    }
}
