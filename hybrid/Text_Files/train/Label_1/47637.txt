import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[][] robots = new int[n][3];
        for (int i = 0; i < n; i++) {
            robots[i][0] = sc.nextInt();
            robots[i][1] = sc.nextInt();
            robots[i][2] = sc.nextInt();
        }
        Arrays.sort(robots, (a, b) -> a[0] - b[0]);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (robots[j][0] - robots[i][1] > robots[i][0]) break;
                if (Math.abs(robots[i][2] - robots[j][2]) <= k) ans++;
            }
        }
        System.out.println(ans);
    }
}
