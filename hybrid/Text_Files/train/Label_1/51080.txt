import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();

        // Create graph
        Map<Integer, List<Edge>> graph = new HashMap<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int l = sc.nextInt();
            graph.computeIfAbsent(a, x -> new ArrayList<>()).add(new Edge(b, l));
            graph.computeIfAbsent(b, x -> new ArrayList<>()).add(new Edge(a, l));
        }

        // Create array of distances to nearest shopping mall
        int[] dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);

        // Create queue to store towns to visit
        Queue<Integer> queue = new LinkedList<>();

        // Add towns with shopping malls to queue
        for (int i = 0; i < k; i++) {
            int s = sc.nextInt();
            queue.add(s);
            dist[s] = 0;
        }

        // Perform BFS to find shortest distances
        while (!queue.isEmpty()) {
            int v = queue.poll();
            for (Edge e : graph.getOrDefault(v, new ArrayList<>())) {
                int w = e.to;
                int w_dist = e.len;
                if (dist[v] + w_dist < dist[w]) {
                    dist[w] = dist[v] + w_dist;
                    queue.add(w);
                }
            }
        }

        // Find maximum distance to nearest shopping mall
        int maxDist = 0;
        for (int i = 1; i <= n; i++) {
            maxDist = Math.max(maxDist, dist[i]);
        }

        // Output maximum distance
        System.out.println(maxDist);
    }

    private static class Edge {
        int to;
        int len;

        public Edge(int to, int len) {
            this.to = to;
            this.len = len;
        }
    }
}
