#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    map<int, int> fre;
    for (int i = 0; i < m; i++) {
        int x;
        cin >> x;
        fre[x]++;
    }
    int ans = 0;
    for (auto it : fre) { ans = max(ans, it.second); }
    cout << (ans >= n ? ans / n : 0) << endl;
    return 0;
}
