#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef vector<int> vi;
#define pb push_back

int n;
vi a;
ll d[100001];

int main() {
    cin >> n;
    a.resize(n);
    for (int &x : a) cin >> x;
    for (int i = n - 1; i >= 0; i--) {
        d[i] = a[i];
        if (i + 1 < n) d[i] += d[i + 1];
    }
    ll ans = LLONG_MIN;
    for (int i = 0; i < n; i++) {
        for (int j = i; j < n; j++) {
            ll sum = d[i];
            if (i > 0) sum -= d[i];
            if (j + 1 < n) sum -= d[j + 1];
            ans = max(ans, sum);
        }
    }
    cout << ans << endl;
    return 0;
}
