from collections import defaultdict

def solve_segments(n, k, segments):
    events = []
    for i, (l, r) in enumerate(segments):
        events.append((l, -1, i + 1))  # Start of segment
        events.append((r, 1, i + 1))   # End of segment
    events.sort()

    active_segments = defaultdict(int)
    coverage = 0
    to_remove = []

    for point, event_type, index in events:
        if event_type == -1:  # Start of segment
            coverage += 1
            active_segments[index] += 1
            if coverage > k:
                to_remove.append(index)
                coverage -= 1
                active_segments[index] -= 1
        else:  # End of segment
            if active_segments[index] > 0:
                coverage -= 1
                active_segments[index] -= 1

    return len(to_remove), to_remove

# Reading from standard input and writing to standard output
n, k = map(int, input().split())
segments = [list(map(int, input().split())) for _ in range(n)]

m, removed_segments = solve_segments(n, k, segments)
print(m)
if m > 0:
    print(' '.join(map(str, removed_segments)))
