#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll MAXN = 1e2 + 5, INF = 1e9 + 7;
ll n, ans[MAXN], a[MAXN];
bool cmp(ll x, ll y) { return a[x] < a[y]; }

int main() {
  cin >> n;
  iota(ans + 1, ans + n + 1, 1);
  for (ll i = 1; i <= n; i++) cin >> a[i];
  sort(ans + 1, ans + n + 1, cmp);
  for (ll i = 1; i <= n; i++) cout << a[ans[i]] << " ";
  return 0;
}
