def is_prime(n):
    if n < 2:
        return False
    if n % 2 == 0 and n != 2:
        return False
    i = 3
    while i * i <= n:
        if n % i == 0:
            return False
        i += 2
    return True

n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
ans = 0

for i in range(n):
    for j in range(m):
        if is_prime(a[i][j]):
            ans += 1
if ans == n or ans == m:
    print(0)
else:
    print(1)
