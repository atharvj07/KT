#include <cstdio>

long long mpow(long long a, long long n) {
    long long res = 1LL;
    while (n > 0) {
        if (n & 1)
            res = (res * a) % 1000000007;
        n >>= 1;
        a = (a * a) % 1000000007;
    }
    return res;
}

int main() {
    long long n, m, k;
    scanf("%lld %lld %lld", &n, &m, &k);
    if (!n || !m)
        printf("0\n");
    else
        printf("%lld\n", ((mpow(2LL, n) - 1 + 1000000007) % 1000000007) * ((mpow(2LL, m) - 1 + 1000000007) % 1000000007) % 1000000007);
    return 0;
}
