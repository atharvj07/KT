#include <bits/stdc++.h>
using namespace std;
map<int,int>f,g;
int main(){
    int n,m;cin>>n>>m;
    int a[n+1],b[m+1];
    for(int i=1;i<=n;++i)cin>>a[i],f[a[i]]++;
    for(int i=1;i<=m;++i)cin>>b[i],g[b[i]]++;
    if(f!=g){cout<<"Impossible";return 0;}
    cout<<"Possible\n";
    for(int i=1;i<=m;++i)for(int j=1;j<=n;++j)if(f[j]==1&&a[j]==b[i])
        cout<<j<<" ";
    return 0;
}
