#include <iostream>

using namespace std;

int main() {
  unsigned long long m;
  cin >> m;

  for (unsigned long long n = 3;; n++) {
    unsigned long long v = 1;

    for (int i = 0; i < 4; i++) {
      if (v > m) {
        cout << -1 << endl;
        return 0;
      }
      if (v == m) {
        cout << n << endl;
        return 0;
      }
      v *= n;
    }
  }
}
