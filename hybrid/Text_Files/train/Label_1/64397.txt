n, m = map(int, input().split())
words = {}
for _ in range(m):
    a, b = input().split()
    words[a] = words.get(a, []) + [b]
    words[b] = words.get(b, []) + [a]
lecture = input().split()
for word in lecture:
    l = len(word)
    s = words[word]
    if len(s[0]) < l:
        word = s[0]
    print(word, end=' ')
