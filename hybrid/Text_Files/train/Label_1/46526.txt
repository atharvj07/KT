import java.util.*;

class Main {
    static final int MAXP = 1000;
    static int numHuts;
    static int numRoads;
    static int[][] roads;
    static boolean[] used;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        numHuts = scanner.nextInt();
        numRoads = scanner.nextInt();
        scanner.nextLine();
        roads = new int[numRoads][3];
        used = new boolean[numRoads];
        for (int i = 0; i < numRoads; i++) {
            String[] line = scanner.nextLine().split(" ");
            roads[i][0] = Integer.parseInt(line[0]) - 1;
            roads[i][1] = Integer.parseInt(line[1]) - 1;
            roads[i][2] = line[2].equals("S") ? 0 : 1;
        }
        int result = findSolution();
        if (result == -1) {
            System.out.println(-1);
        } else {
            System.out.println(result);
            ArrayList<Integer> answer = new ArrayList<>();
            for (int i = 0; i < numRoads; i++) {
                if (used[i]) {
                    answer.add(i + 1);
                }
            }
            StringBuilder output = new StringBuilder();
            for (int number : answer) {
                output.append(number).append(" ");
            }
            output.setLength(output.length() - 1);
            System.out.println(output);
        }
    }

    private static int findSolution() {
        int[][] graph = new int[numHuts][numHuts];
        for (int[] road : roads) {
            graph[road[0]][road[1]] = 1;
            graph[road[1]][road[0]] = 1;
        }
        boolean[] visited = new boolean[numHuts];
        int[] count = new int[2];
        for (int i = 0; i < numHuts; i++) {
            if (!visited[i]) {
                if (dfs(i, -1, graph, visited, used, count) == -1) {
                    return -1;
                }
            }
        }
        if (count[0] == count[1]) {
            return count[0];
        }
        return -1;
    }

    private static int dfs(int current, int prev, int[][] graph, boolean[] visited, boolean[] used, int[] count) {
        if (visited[current]) {
            return -1;
        }
        visited[current] = true;
        for (int i = 0; i < numHuts; i++) {
            if (graph[current][i] == 1) {
                if (i == prev) {
                    continue;
                }
                int roadIndex = findRoadIndex(current, i);
                if (roadIndex == -1) {
                    return -1;
                }
                used[roadIndex] = true;
                count[roads[roadIndex][2]]++;
                if (dfs(i, current, graph, visited, used, count) == -1) {
                    return -1;
                }
            }
        }
        return 1;
    }

    private static int findRoadIndex(int a, int b) {
        for (int i = 0; i < numRoads; i++) {
            if ((roads[i][0] == a && roads[i][1] == b) || (roads[i][0] == b && roads[i][1] == a)) {
                return i;
            }
        }
        return -1;
    }
}
