#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, p, q, r, a[100001];
  cin >> n >> p >> q >> r;
  for (int i = 0; i < n; i++) cin >> a[i];
  pair<pair<int, int>, int> best = {{-1e9, 1e9}, -1e9};
  for (int i = 0; i < n; i++) {
    best = max(best, {{-a[i], a[i]}, a[i] * (p + q + r)});
  }
  for (int i = 0; i < n; i++) {
    best = max(best, {{a[i], -a[i]}, a[i] * (p + q - r)});
    best = max(best, {{a[i], -a[i]}, a[i] * (p - q + r)});
    best = max(best, {{a[i], -a[i]}, a[i] * (-p + q + r)});
  }
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      int prod = (a[i] + a[j]) * p + (a[j] - a[i]) * q + r * a[j];
      best = max(best, {{min(a[i], a[j]), max(a[i], a[j])}, prod});
    }
  }
  cout << best.second << '\n';
}
