#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size(), ans = 0;
  vector<int> cnt(n);
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == 'r') cnt[i] = (i < n - 1 ? cnt[i + 1] : 0);
    if (s[i] == 'a') cnt[i] += (i < n - 2 ? cnt[i + 2] : 0);
    if (s[i] == 'e') cnt[i] += (i < n - 3 ? cnt[i + 3] : 0);
    if (s[i] == 'b') cnt[i] += (i < n - 4 ? cnt[i + 4] : 1);
    ans += cnt[i];
  }
  cout << ans << endl;
  return 0;
}
