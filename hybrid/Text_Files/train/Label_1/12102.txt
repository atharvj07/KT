#include <bits/stdc++.h>
using namespace std;
struct node {
  int val, pos;
  bool operator<(const node &t) const { return val < t.val; }
};
const int maxn = 2e5 + 10;
int n, m, p;
node a[maxn];
vector<int> b;
bool solve(int x) {
  vector<node> c;
  for (int i = x; i <= n; i += p) { c.push_back(a[i]); }
  sort(c.begin(), c.end());
  sort(b.begin(), b.end());
  if (c.size() != b.size()) return false;
  for (int i = 0; i < c.size(); ++i) {
    if (c[i].val != b[i]) return false;
  }
  return true;
}
int main() {
  cin >> n >> m >> p;
  for (int i = 1; i <= n; ++i) { cin >> a[i].val; }
  for (int i = 1; i <= n; ++i) { a[i].pos = i; }
  for (int i = 1; i <= m; ++i) {
    int x;
    cin >> x;
    b.push_back(x);
  }
  sort(a + 1, a + 1 + n);
  int ans = 0;
  vector<int> v;
  for (int i = 1; i <= n; ++i) {
    if (solve(i)) ans++, v.push_back(i);
  }
  cout << ans << endl;
  for (auto it : v) cout << it << ' ';
  cout << endl;
}
