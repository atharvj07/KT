#include <iostream>
const int segs[10]={2,4,5,4,5,6,6,3,7,6};
int main(){
  int t;
  std::cin >> t;
  while(t--){
    int n;
    std::cin >> n;
    int ans = 0;
    for(int i = 9; i >= 0; --i){
      if(n >= segs[i]){
        n -= segs[i];
        ans = ans * 10 + i;
      }
    }
    std::cout << ans << "\n";
  }
}
