public_key = input().strip()
a, b = map(int, input().split())

prefix_mod = [0] * len(public_key)
suffix_mod = [0] * len(public_key)

mod = 0
for i in range(len(public_key)):
    mod = (mod * 10 + int(public_key[i])) % a
    prefix_mod[i] = mod

mod = 0
power_ten_mod_b = 1
for i in range(len(public_key) - 1, -1, -1):
    mod = (int(public_key[i]) * power_ten_mod_b + mod) % b
    suffix_mod[i] = mod
    power_ten_mod_b = (power_ten_mod_b * 10) % b

for i in range(len(public_key) - 1):
    if prefix_mod[i] == 0 and suffix_mod[i + 1] == 0 and public_key[i + 1] != '0':
        print("YES")
        print(public_key[:i+1])
        print(public_key[i+1:])
        break
else:
    print("NO")
