import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
pos = [0] * n
gest = 0
for i in range(n):
    pos[a[i] - 1] = (i // k + 1)
for i in b:
    gest += pos[i - 1]
    for j in range(n):
        if a[j] == i:
            a[j], a[j - 1] = a[j - 1], a[j]
            pos[i - 1] -= 1
            pos[a[j - 1] - 1] += 1
            break
print(gest)
