import sys
input = sys.stdin.readline

n = int(input())

xy = []
for i in range(n):
    x,y,z = map(int, input().split())
    xy.append([x,y])

xy.sort()

ans = []

for i in range(n//2):
    if xy[i][0] == xy[i+n//2][0] and xy[i][1] == xy[i+n//2][1]:
        ans.append([i+1,i+n//2+1])
    else:
        ans.append([i+1,n-i])

for a,b in ans:
    print(a,b)
