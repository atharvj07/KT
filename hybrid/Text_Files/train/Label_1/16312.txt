#include <bits/stdc++.h>
#define ll long long

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, s;
    cin >> n >> s;
    vector<pair<int, int>> v;
    for (int i = 0; i < n; i++) {
      int a, b;
      cin >> a >> b;
      v.push_back({a, b});
    }
    sort(v.begin(), v.end(), [](const pair<int, int> &x, const pair<int, int> &y) {
      return x.second > y.second;
    });
    int ans = 0;
    for (auto &p : v) {
      if (s - p.second >= 0) {
        ans = p.second;
        s -= p.second;
      } else {
        ans = s + p.first;
        break;
      }
    }
    cout << ans << "\n";
  }
  return 0;
}
