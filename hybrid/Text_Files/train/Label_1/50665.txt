import java.util.*;

public class Main {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int m = sc.nextInt();
		Map<Integer, PriorityQueue<Integer>> map = new HashMap<>();
		for (int i = 0; i < m; i++) {
			int p = sc.nextInt();
			int y = sc.nextInt();
			if (!map.containsKey(p)) {
				map.put(p, new PriorityQueue<>());
			}
			map.get(p).offer(y);
		}
		for (int i = 1; i <= n; i++) {
			if (!map.containsKey(i)) {
				System.out.println("00000" + i + "000001");
				continue;
			}
			PriorityQueue<Integer> pq = map.get(i);
			int count = 1;
			while (!pq.isEmpty()) {
				int y = pq.poll();
				String s = String.format("%06d%06d", i, count);
				System.out.println(s);
				count++;
			}
		}
	}
}
