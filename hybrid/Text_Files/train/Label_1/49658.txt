import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scan.nextInt();
        }
        long days = n * 6;
        long ans = -1;
        for (int i = 0; i < n; i++) {
            long time = (a[i] + days - 1) / days;
            if (i == 0 || time < ans) {
                ans = time;
            }
        }
        System.out.println(ans);
    }
}
