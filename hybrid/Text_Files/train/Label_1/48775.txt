import java.util.*;
import java.io.*;

public class Main {

    static class UnionFind {
        int par[];
        int rank[];
        int cnt;

        public UnionFind(int n) {
            this.par = new int[n];
            this.rank = new int[n];

            // initialisation
            for (int i = 0; i < n; i++) {
                this.par[i] = i;
                this.rank[i] = 0;
            }

            this.cnt = n;
        }

        public int find(int x) {
            if (par[x] == x) return x;
            else return par[x] = find(par[x]);
        }

        public void unite(int x, int y) {
            x = find(x);
            y = find(y);
            if (x == y) return;

            if (rank[x] < rank[y]) {
                par[x] = y;
            } else {
                if (rank[x] == rank[y]) rank[x]++;
                par[y] = x;
            }
            cnt--;
        }

        public boolean same(int x, int y) {
            return find(x) == find(y);
        }

        public int count() {
            return cnt;
        }
    }

    static class Ingredient implements Comparable<Ingredient> {
        String name;
        int price;

        public Ingredient(String name, int price) {
            this.name = name;
            this.price = price;
        }

        @Override
        public int compareTo(Ingredient other) {
            return this.price - other.price;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader stdin = new BufferedReader(new InputStreamReader(System.in));

        // input N
        int N = Integer.parseInt(stdin.readLine());
        List<Ingredient> ingredients = new ArrayList<>();

        // input & init ingredients
        for (int i = 0; i < N; i++) {
            String[] line = stdin.readLine().split(" ");
            ingredients.add(new Ingredient(line[0], Integer.parseInt(line[1])));
        }

        // sort ingredients by price
        Collections.sort(ingredients);

        // input M
        int M = Integer.parseInt(stdin.readLine());
        UnionFind uf = new UnionFind(N);

        for (int i = 0; i < M; i++) {
            String[] line = stdin.readLine().split(" ");
            String a = line[0], b = line[1];
            int idxA = -1, idxB = -1;

            for (int j = 0; j < N; j++) {
                Ingredient x = ingredients.get(j);
                if (x.name.equals(a)) idxA = j;
                else if (x.name.equals(b)) idxB = j;
            }

            if (idxA != -1 && idxB != -1) {
                uf.unite(idxA, idxB);
            }
        }

        // output
        int ans = 0;
        for (int i = 0; i < N; i++) {
            if (uf.find(i) == i) ans += ingredients.get(i).price;
        }
        System.out.println(ans);
    }
}
