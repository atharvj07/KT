import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] t = new int[n], c = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = scanner.nextInt();
            c[i] = scanner.nextInt();
        }
        Arrays.sort(t);
        Arrays.sort(c);
        long sum = 0;
        for (int i = 0; i < n; i++) {
            if (t[i] == 0) {
                sum += c[i];
            } else {
                sum += c[i] - t[i];
            }
        }
        System.out.println(sum);
    }
}
