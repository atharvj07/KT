#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, x;
    cin >> n;
    vector<int> v(n / 2);
    for (int i = 0; i < n / 2; i++) cin >> v[i];
    sort(v.begin(), v.end());
    int ans = 0;
    for (int i = 0; i < n / 2; i++) {
        ans += min(v[i] - 1, n - v[i]);
    }
    cout << ans << endl;
    return 0;
}
