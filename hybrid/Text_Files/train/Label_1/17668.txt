#include <bits/stdc++.h>
using namespace std;
const int MAXN = 3e5 + 5;
const long long INF = 1e12;
int n, h[MAXN], l[MAXN], r[MAXN];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> h[i]; }
  l[n + 1] = INF;
  for (int i = n; i >= 1; i--) {
    if (h[i] < h[i - 1]) {
      l[i] = 0;
    } else {
      l[i] = l[i + 1] + 1;
    }
  }
  for (int i = 1; i <= n; i++) {
    if (h[i] > h[i + 1])
      r[i] = INF;
    else
      r[i] = r[i + 1] + 1;
  }
  int ans = 0;
  for (int i = 1; i < n; i++) {
    if (l[i + 1] > 0) {
      if (l[i + 1] < r[i])
        ans += l[i + 1];
      else
        ans += l[i] + 1;
    }
  }
  cout << ans << endl;
}
