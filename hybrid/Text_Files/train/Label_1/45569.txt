import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        int f = in.nextInt();
        int k = in.nextInt();
        int journeys = 0;
        int refuels = 0;
        while (journeys < k) {
            if (journeys % 2 == 0) {
                if (f - b <= 0) {
                    refuels++;
                }
                if (f + b >= a) {
                    journeys++;
                }
            } else {
                if (a - f <= 0) {
                    refuels++;
                }
                if (f - b >= 0) {
                    journeys++;
                }
            }
        }
        if (journeys == k) {
            System.out.println(refuels);
        } else {
            System.out.println(-1);
        }
    }
}
