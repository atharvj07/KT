from collections import deque
import sys

input = sys.stdin.readline

n = int(input())
points = []
for _ in range(n):
    points.append(tuple(int(x) for x in input().split()))
points.sort()

Q = deque(points)
while len(Q) > 1:
    a = Q.popleft()
    b = Q.pop()
    print(f"{points.index(a) + 1} {points.index(b) + 1}")
