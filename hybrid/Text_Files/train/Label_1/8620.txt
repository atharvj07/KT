#include <iostream>
#include <vector>

using namespace std;

int main() {
  long long H, n;
  cin >> H >> n;
  vector<long long> d(n);
  for (int i = 0; i < n; i++) {
    cin >> d[i];
  }

  long long current_hp = H;
  for (int i = 0; i < n; i++) {
    current_hp += d[i];
    if (current_hp <= 0) {
      cout << i + 1 << endl;
      return 0;
    }
  }

  cout << -1 << endl;
  return 0;
}
