n, m, k = map(int, input().split())
sheet = [[0] * m for _ in range(n)]

for _ in range(k):
    operation, idx, color = map(int, input().split())
    if operation == 1:
        for j in range(m):
            sheet[idx-1][j] = color
    else:
        for i in range(n):
            sheet[i][idx-1] = color

for row in sheet:
    print(' '.join(map(str, row)))
