#include <bits/stdc++.h>
using namespace std;
int main() {
  int test;
  cin >> test;
  while (test--) {
    string s;
    cin >> s;
    for (char c : s) {
      if (c == 'R')
        cout << 'P';
      else if (c == 'P')
        cout << 'S';
      else
        cout << 'R';
    }
    cout << endl;
  }
}
