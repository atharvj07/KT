import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextLong();
        }
        long a0 = 0;
        for (int i = 1; i <= n; i++) {
            a0 = a0 ^ a[i];
        }
        for (int i = 1; i <= n; i++) {
            System.out.print((a0 ^ a[i]) + " ");
        }
        System.out.println();
    }
}
