import java.util.Scanner;

public class ObjectPacking {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int maxObjects = 0;
        int currentBoxSize = k;
        int remainingObjects = n;

        for (int i = 0; i < n; i++) {
            if (a[i] > currentBoxSize) {
                m--;
                currentBoxSize = k;
            }

            if (a[i] <= currentBoxSize) {
                currentBoxSize -= a[i];
                maxObjects++;
                remainingObjects--;
            }

            if (m == 0) {
                break;
            }
        }

        // Remove objects from the beginning until the remaining objects can be packed
        while (remainingObjects > 0 && maxObjects > 0) {
            remainingObjects--;
            maxObjects--;
            currentBoxSize += a[maxObjects];
            if (currentBoxSize >= k) {
                m--;
                currentBoxSize = k;
            }
        }

        System.out.println(maxObjects);
    }
}
