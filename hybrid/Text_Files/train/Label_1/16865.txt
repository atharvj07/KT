#include <bits/stdc++.h>
using namespace std;
int main(){
    int t, n, x, count, ans;
    double sum;
    cin >> t;
    while (t--){
        cin >> n;
        sum = 0;
        count = 0;
        for (int i = 0; i < n; i++){
            cin >> x;
            sum += (double)x;
        }
        ans = (int)(sum / n);
        cout << n - ans << endl;
    }
    return 0;
}
