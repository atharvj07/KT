#include <bits/stdc++.h>
using namespace std;
int main() {
  long long a, b, f, k;
  cin >> a >> b >> f >> k;
  if ((a - f) % b == 0 && a < b * (k + 1)) {
    cout << -1;
    return 0;
  }
  if (f > a - f)
    a -= f;
  else
    a = a - f - b;
  if (a % b == 0)
    a /= b;
  else
    a = a / b + 1;
  k -= a;
  cout << k;
}
