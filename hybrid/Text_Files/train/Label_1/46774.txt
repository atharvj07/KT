import java.util.*;
import java.io.*;

public class Main {
  static int n, m;
  static int[] a;
  static int[][] b;
  static int[][] c = new int[101][101];
  static int[][][] d = new int[101][101][2]; // [l][r][x] = for [l, r], num of flowers is x

  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
    StringTokenizer st = new StringTokenizer(br.readLine());
    n = Integer.parseInt(st.nextToken());
    m = Integer.parseInt(st.nextToken());
    a = new int[n + 1];
    st = new StringTokenizer(br.readLine());
    for (int i = 1; i <= n; i++) {
      a[i] = Integer.parseInt(st.nextToken());
    }
    b = new int[m][2];
    for (int i = 0; i < m; i++) {
      st = new StringTokenizer(br.readLine());
      b[i][0] = Integer.parseInt(st.nextToken());
      b[i][1] = Integer.parseInt(st.nextToken());
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = i; j <= n; j++) {
        c[i][j] = c[i - 1][j - 1] + a[j];
      }
    }
    for (int l = 0; l < m; l++) {
      d[b[l][0]][b[l][1]][0] += 1;
    }
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= n; j++) {
        for (int k = 0; k < 2; k++) {
          d[i][j][k] += d[i - 1][j][k] + d[i][j - 1][k] - d[i - 1][j - 1][k];
        }
      }
    }
    for (int i = 1; i <= n; i++) {
      for (int j = i; j <= n; j++) {
        int num = d[i][j][1] - d[i - 1][j][1] - d[i][j - 1][1] + d[i - 1][j - 1][1];
        ans = Math.max(ans, c[i][j] * num);
      }
    }
    bw.write(ans + "\n");
    bw.flush();
  }
}
