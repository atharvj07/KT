#include <iostream>
#include <algorithm>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, x;
    std::cin >> n >> x;
    int a[n], b[n];
    for (int i = 0; i < n; i++) {
      std::cin >> a[i];
    }
    for (int i = 0; i < n; i++) {
      std::cin >> b[i];
    }
    std::sort(a, a + n);
    std::sort(b, b + n);
    bool flag = true;
    for (int i = 0; i < n; i++) {
      if (a[i] + b[n - i - 1] > x) {
        flag = false;
      }
    }
    std::cout << (flag ? "Yes" : "No") << std::endl;
  }
  return 0;
}
