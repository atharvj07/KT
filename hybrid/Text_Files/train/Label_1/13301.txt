#include <iostream>

using namespace std;

int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    if (n % 2) {
      cout << n - 3 << " " << 3 << endl;
    } else {
      cout << n / 2 << " " << n / 2 << endl;
    }
  }
}
