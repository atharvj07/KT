def fun(l, r, seq):
  count = [0] * 10
  for i in range(l - 1, r):
    if seq[i].isdigit():
      x = int(seq[i])
      count[x] += 1
      seq[i] = str(int(seq[i]) - 1)
      while seq[i] == '0':
        seq = seq[:i] + seq[i + 1:]
        count[0] += 1
        i -= 1
  return count

n, q = map(int, input().split())
seq = input()
for _ in range(q):
  l, r = map(int, input().split())
  print(*fun(l, r, list(seq)))
