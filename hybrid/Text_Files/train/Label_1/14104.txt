#include <bits/stdc++.h>
#define MAX_N 100
#define MAX_M 1000

struct Edge {
    int u, v;
};

std::vector<int> color[MAX_N + 1];
std::vector<Edge> edges;
int n, m;

bool dfs(int u, int c, std::vector<int>& vis) {
    vis[u] = c;
    bool connected = true;
    for (int i = 0; i < color[u].size(); i++) {
        int v = color[u][i];
        if (vis[v] == -1) {
            connected &= dfs(v, c, vis);
        } else if (vis[v] != c) {
            connected = false;
        }
    }
    return connected;
}

int main() {
    std::cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        std::cin >> u >> v;
        edges.push_back({u, v});
        color[u].push_back(v);
        color[v].push_back(u);
    }

    std::vector<int> vis(n + 1, -1);
    int c = 0;
    for (int i = 1; i <= n; i++) {
        if (vis[i] == -1) {
            c++;
            if (!dfs(i, c, vis)) {
                std::cout << -1 << std::endl;
                return 0;
            }
        }
    }

    std::vector<int> cnt(c + 1, 0);
    for (int i = 1; i <= n; i++) {
        cnt[vis[i]]++;
    }

    int k = 0;
    for (int i = 1; i <= c; i++) {
        k += cnt[i];
        std::cout << cnt[i] << std::endl;
        for (int j = 0; j < cnt[i]; j++) {
            std::cout << 1 << " " << k << std::endl;
        }
    }

    return 0;
}
