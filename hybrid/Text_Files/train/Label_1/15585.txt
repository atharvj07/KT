#include <bits/stdc++.h>
using namespace std;

long long m;
vector<pair<long long, long long>> v[4];

bool comp(pair<long long, long long> a, pair<long long, long long> b) {
  return a.first < b.first;
}

long long f(int n, long long h, long long l, long long k) {
  multiset<long long> s;
  long long c = LLONG_MAX, sum = 0, ans = LLONG_MAX, trains = n;
  for (int i = 0; i < 4; i++) {
    sort(v[i].begin(), v[i].end(), comp);
    for (auto a : v[i]) {
      long long b = a.first * l + a.second;
      if (b >= sum + k) {
        s.insert(b - sum);
        sum = b;
      }
    }
    while (!s.empty() && *s.begin() < sum) {
      trains--;
      sum -= *s.begin();
      s.erase(s.begin());
      ans = min(ans, c + sum);
    }
    c = min(c, sum);
  }
  return n - trains;
}

int main() {
  long long n, h, k;
  cin >> n >> h >> m >> k;
  m /= 2;
  for (int i = 0; i < n; i++) {
    long long x, y;
    cin >> x >> y;
    v[x % 4].push_back({x, y});
  }
  long long ans = f(n, h, m, k);
  cout << ans << " 0" << endl;
  for (auto a : v[0]) {
    if (f(n, h, m, k) < ans)
      cout << a.first << " " << a.second << endl;
    else
      ans = f(n, h, m, k);
  }
  return 0;
}
