import sys

n = int(input())
a = list(map(int, input().split()))
l = 0
r = 0
for i in range(n):
    if a[i] == 0:
        l += 1
        if l == r and l > 0:
            print(i + 1)
            sys.exit()
    else:
        r += 1
        if r > l and r > 0:
            print(i + 1)
            sys.exit()
