import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt();
        long[][] videos = new long[n][2];
        for (int i = 0; i < n; i++) {
            videos[i][0] = input.nextInt();
            videos[i][1] = input.nextInt();
        }
        long[][] channels = new long[m][3];
        for (int i = 0; i < m; i++) {
            channels[i][0] = input.nextInt();
            channels[i][1] = input.nextInt();
            channels[i][2] = input.nextInt();
        }
        long max = 0;
        long i = -1, j = -1;
        for (long[] video : videos) {
            for (long[] channel : channels) {
                long start = Math.max(channel[0], video[0]);
                long end = Math.min(channel[1], video[1]);
                long time = Math.max(0, end - start);
                if (time > max) {
                    max = time;
                    i = video[0];
                    j = channel[0];
                }
            }
        }
        System.out.println(max);
        if (max > 0) {
            int videoIndex = -1, channelIndex = -1;
            for (int a = 0; a < n; a++) {
                if (videos[a][0] == i) {
                    videoIndex = a + 1;
                }
            }
            for (int b = 0; b < m; b++) {
                if (channels[b][0] == j) {
                    channelIndex = b + 1;
                }
            }
            System.out.println(videoIndex + " " + channelIndex);
        }
    }
}
