#include <bits/stdc++.h>
using namespace std;
long long a[200005],b[200005],ce[200005],ans[200005];
int main() {
  long long i,j,n;
  cin>>n;
  for(i=1; i<=n; i++) {
    cin>>a[i];
    ce[a[i]]=i;
  }
  for(i=1; i<=n; i++) {
    cin>>b[i];
  }
  for(i=1; i<=n; i++) {
    ans[i]=ce[b[i]]-i;
    for(j=ce[b[i]]; j<i; j++) ce[a[j]]--;
  }
  for(i=1; i<=n; i++) {
    cout<<ans[i]+1<<" ";
  }
  return 0;
}
