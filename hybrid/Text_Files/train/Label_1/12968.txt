#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e3 + 5;
ll a[N], b[N], n, m, ans[N][2], inv[N][N];
int main() {
    cin >> n;
    for (ll i = 1; i <= n; i++) {
        cin >> a[i];
        b[i] = a[i];
    }
    sort(b + 1, b + n + 1);
    for (ll i = 1; i <= n; i++) {
        for (ll j = 1; j <= n; j++) {
            if (b[i] == a[j]) inv[i][j] = 1;
        }
    }
    for (ll i = 1; i <= n; i++) {
        for (ll j = i + 1; j <= n; j++) {
            if (inv[i][j] && inv[j][i]) { ans[m][0] = i; ans[m][1] = j; m++; }
        }
    }
    if (m == 0)
        cout << -1;
    else {
        cout << m << endl;
        for (ll i = 0; i < m; i++) cout << ans[i][0] << " " << ans[i][1] << endl;
    }
}
