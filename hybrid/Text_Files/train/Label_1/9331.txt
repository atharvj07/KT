#include <cstdio>
#include <cstring>
#include <vector>
using namespace std;
char s[55], t[55];
int main() {
  int k, n;
  scanf("%d", &k);
  for (int ks = 1; ks <= k; ks++) {
    scanf("%d%s%s", &n, s + 1, t + 1);
    vector<pair<int, int>> ans;
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= n; j++) {
        if (s[i] == t[j] && s[i] != t[i]) {
          ans.push_back({i, j});
          swap(s[i], t[j]);
          break;
        }
      }
    }
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= n; j++) {
        if (s[i] == t[j] && s[i] != t[i]) {
          ans.push_back({i, j});
          swap(s[i], t[j]);
          break;
        }
      }
    }
    for (int i = 1; i <= n; i++) {
      if (t[i] == s[n]) {
        ans.push_back({n, i});
        swap(s[n], t[i]);
      }
    }
    if (strcmp(s + 1, t + 1) == 0) {
      printf("Yes\n");
      printf("%d\n", ans.size());
      for (auto &x : ans) printf("%d %d\n", x.first, x.second);
    } else {
      printf("No\n");
    }
  }
}
