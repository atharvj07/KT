#include <bits/stdc++.h>
using namespace std;

#define ll long long

int main() {
  ll dp[1 << 5][5] = {}, g[5][5];
  for (int i = 0; i < 5; ++i)
    for (int j = 0; j < 5; ++j) cin >> g[i][j];
  for (int i = 1; i < 1 << 5; ++i)
    for (int j = 0; j < 5; ++j)
      if (i & (1 << j))
        for (int k = 0; k < 5; ++k)
          if (i & (1 << k) && j != k) dp[i][j] = max(dp[i][j], dp[i & ~(1 << j)][k] + g[j][k]);
  cout << *max_element(dp[(1 << 5) - 1], dp[(1 << 5) - 1] + 5) << endl;
  return 0;
}
