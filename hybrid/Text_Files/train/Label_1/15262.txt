#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<string> grid(n);
        for (int i = 0; i < n; i++) {
            cin >> grid[i];
        }
        bool valid = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '.') {
                    int numR = 0, numW = 0;
                    if (i > 0 && grid[i - 1][j] == 'R') numR++;
                    if (i < n - 1 && grid[i + 1][j] == 'R') numR++;
                    if (j > 0 && grid[i][j - 1] == 'R') numR++;
                    if (j < m - 1 && grid[i][j + 1] == 'R') numR++;
                    if (i > 0 && grid[i - 1][j] == 'W') numW++;
                    if (i < n - 1 && grid[i + 1][j] == 'W') numW++;
                    if (j > 0 && grid[i][j - 1] == 'W') numW++;
                    if (j < m - 1 && grid[i][j + 1] == 'W') numW++;
                    if (numR == 0 || numW == 0) valid = false;
                }
            }
        }
        if (valid) {
            cout << "YES" << endl;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (grid[i][j] == '.') {
                        if (rand() % 2 == 0) grid[i][j] = 'R';
                        else grid[i][j] = 'W';
                    }
                }
            }
            for (int i = 0; i < n; i++) cout << grid[i] << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
