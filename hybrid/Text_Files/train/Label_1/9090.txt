#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> doors(n);
  for (int i = 0; i < n; i++) { std::cin >> doors[i]; }
  int l = 0, r = 0, k = 0;
  for (int i = 0; i < n; i++) {
    if (doors[i] == 0)
      l++;
    else
      r++;
    if (l > 0 && r > 0) {
      k = i + 1;
      break;
    }
  }
  std::cout << k << '\n';
  return 0;
}
