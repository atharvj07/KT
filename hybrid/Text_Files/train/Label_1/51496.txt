import java.util.Scanner;

public class OddSum {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
            }

            boolean isOddSumPossible = isOddSumPossible(a);
            System.out.println(isOddSumPossible ? "YES" : "NO");
        }
    }

    static boolean isOddSumPossible(int[] a) {
        int oddCount = 0;
        for (int element : a) {
            if (element % 2 == 1) {
                oddCount++;
            }
        }

        return oddCount > 0 && oddCount % 2 == 1;
    }
}
