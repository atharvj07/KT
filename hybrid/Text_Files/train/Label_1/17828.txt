#include <bits/stdc++.h>
using namespace std;
map<int,int> vis;
int a[55],b[55];
int32_t main(){
  ios_base::sync_with_stdio(0);cin.tie(0);
  int t;cin>>t;
  while(t--){
    int n;cin>>n;
    for(int i=0;i<n;i++){
      cin>>a[i];
      vis[a[i]]++;
    }
    sort(a,a+n);
    int sum=0;
    int flg=0;
    for(int i=0;i<n;i++){
      if(i&1)sum-=a[i];
      else sum+=a[i];
      if(sum==0){
        flg=1;break;
      }
    }
    if(!flg){
      for(int i=0;i<n;i++){
        if(vis[a[i]]){
          b[i]=a[i];vis[a[i]]--;
        }else{
          b[i]=a[n-1];a[n-1]-=1000000;
        }
      }
    }else{
      cout<<"NO"<<endl;
      continue;
    }
    cout<<"YES"<<endl;
    for(int i=0;i<n;i++){
      cout<<b[i]<<" ";
    }
    cout<<endl;
    vis.clear();
  }
}
