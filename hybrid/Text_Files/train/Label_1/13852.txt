#include <iostream>
#include <vector>
using namespace std;
int main() {
  string s, t;
  cin >> s >> t;
  int n = s.size();
  int m = t.size();
  vector<int> pos(m);
  for (int i = 0, j = 0; i < n && j < m; i++) {
    if (s[i] == t[j]) pos[j++] = i;
  }
  int ans = n - m;
  for (int i = 1; i < m; i++) {
    ans = max(ans, pos[i] - pos[i - 1] - 1);
  }
  cout << ans << endl;
  return 0;
}
