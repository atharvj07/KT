#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0, lst = a[0];
  for (int i = 1; i < n; i++) {
    if (a[i] != lst) { ans++; }
    lst = a[i];
  }
  cout << ans << endl;
  while (q--) {
    int idx, val;
    cin >> idx >> val;
    idx--;
    if (val != a[idx]) {
      if (idx > 0 && a[idx] == a[idx - 1]) { ans--; }
      if (idx < n - 1 && a[idx] == a[idx + 1]) { ans--; }
      a[idx] = val;
      if (idx > 0 && a[idx] == a[idx - 1]) { ans++; }
      if (idx < n - 1 && a[idx] == a[idx + 1]) { ans++; }
    }
    cout << ans << endl;
  }
  return 0;
}
