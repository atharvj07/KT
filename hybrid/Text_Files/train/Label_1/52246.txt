import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] a = new int[n][n];
        int[][] b = new int[n][n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < n; j++) {
                a[i][j] = s.charAt(j) - '0';
            }
        }
        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < n; j++) {
                b[i][j] = s.charAt(j) - '0';
            }
        }
        String s = in.next();
        for (int i = 0; i < n; i++) {
            c[i] = s.charAt(i) - '0';
        }
        int cnt = 0;
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            int sum = 0;
            for (int j = 0; j < n; j++) {
                sum += a[i][j];
            }
            if (sum % 2 != b[i][n - 1]) ok = false;
            cnt += b[i][n - 1] - sum % 2;
        }
        for (int j = 0; j < n; j++) {
            int sum = 0;
            for (int i = 0; i < n; i++) {
                sum += a[i][j];
            }
            if (sum % 2 != c[j]) ok = false;
            cnt += c[j] - sum % 2;
        }
        if (!ok) {
            System.out.println(-1);
            return;
        }
        System.out.println(cnt);
        for (int i = 0; i < n; i++) {
            if (b[i][n - 1] == 1) System.out.println("row " + i);
            if (c[i] == 1) System.out.println("col " + i);
        }
    }
}
