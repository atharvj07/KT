#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  int cnt = 1;
  int ans = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] == a[i - 1])
      cnt++;
    else {
      ans += cnt > 1;
      cnt = 1;
    }
  }
  ans += cnt > 1;
  cout << ans << '\n';
  return 0;
}
