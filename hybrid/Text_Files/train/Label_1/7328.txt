#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
string s;
int n, cnt[30];
int main() {
    cin >> n >> s;
    for (int i = 0; i < n; i++) cnt[s[i] - 'a']++;
    int ans = 0;
    for (int i = 0; i < 26; i++) ans += cnt[i] % 2;
    if (n % 2 == 0 && ans == 0) ans--;
    else if (n % 2 == 1 && ans <= 1) ans = 0;
    cout << ans << endl;
}
