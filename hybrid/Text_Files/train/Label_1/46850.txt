import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int s = sc.nextInt(), b = sc.nextInt();
        int[] a = new int[s];
        for (int i = 0; i < s; i++) {
            a[i] = sc.nextInt();
        }
        int[][] d = new int[b][2];
        for (int i = 0; i < b; i++) {
            d[i][0] = sc.nextInt();
            d[i][1] = sc.nextInt();
        }
        Arrays.sort(a);
        Arrays.sort(d, Comparator.comparingInt(o -> o[0]));
        long[] ans = new long[s];
        for (int i = 0, j = 0; i < s; i++) {
            while (j < b && a[i] >= d[j][0]) {
                ans[i] += d[j][1];
                j++;
            }
        }
        for (long x : ans) {
            System.out.println(x);
        }
    }
}
