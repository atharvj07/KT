import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    long a[] = new long[n];
    long b[] = new long[n];
    long c = 0; // total capacity
    long v = 0; // total remaining volume
    for (int i = 0; i < n; i++) {
      a[i] = input.nextLong();
      b[i] = input.nextLong();
      c += b[i];
      v += a[i];
    }
    if (v > c) System.out.println("NO");
    else System.out.println("YES");
  }
}
