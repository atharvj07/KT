import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] coordinates = new int[4];
        for (int i = 0; i < 4; i++) {
            coordinates[i] = sc.nextInt();
        }
        int numSets = Math.abs((coordinates[0] - coordinates[2]));
        int separation = Math.abs(coordinates[1] - coordinates[3]);
        int numPartitions = numSets * 4 + separation + 1;
        System.out.println(numPartitions);
    }
}
