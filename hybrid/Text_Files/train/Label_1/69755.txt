def min_strokes(n, heights):
    def paint(start, end, base_height):
        if start > end:
            return 0
        min_height = min(heights[start:end+1])
        vertical_strokes = min_height - base_height
        i = start
        strokes = 0
        while i <= end:
            if heights[i] == min_height:
                i += 1
                continue
            start_segment = i
            while i <= end and heights[i] > min_height:
                i += 1
            end_segment = i - 1
            strokes += paint(start_segment, end_segment, min_height)
        return min(end - start + 1, strokes + vertical_strokes)
    
    return paint(0, n-1, 0)

n = int(input().strip())
heights = list(map(int, input().strip().split()))
print(min_strokes(n, heights))
