#include<bits/stdc++.h>
using namespace std;
int main()
{
    string s;
    cin>>s;
    int n=s.size();
    long long sum=0;
    for(int i=0;i<n;i++)
    {
        if(sum+s[i]-'0'>s[i]-'0')
        {
            sum=sum+s[i]-'0';
        }
        else
        {
            long long ans=0;
            for(int j=0;j<i;j++)
            {
                ans=ans*10+(s[j]-'0');
            }
            long long anss=0;
            for(int j=i;j<n;j++)
            {
                anss=anss*10+(s[j]-'0');
            }
            cout<<ans<<"+"<<anss<<"="<<ans+anss<<endl;
            return 0;
        }
    }
    sum=sum+s[n-1]-'0';
    cout<<0<<"+"<<sum<<"="<<sum<<endl;
    return 0;
}
