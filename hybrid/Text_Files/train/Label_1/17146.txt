#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  int t;
  cin >> t;
  int rem[10];
  vector<int> ans;
  while (t--) {
    ll n;
    cin >> n;
    bool ok = false;
    for (int i = 0; n > 0; i++) {
      if (i == 1) ok = true;
      rem[i] = n % 10;
      if (rem[i] != 0) {
        if (ok) ans.push_back(rem[i]);
        ok = true;
      }
      n /= 10;
    }
    int sum = 0;
    reverse(ans.begin(), ans.end());
    for (auto x : ans) sum += x;
    cout << sum << endl;
    ans.clear();
  }
  return 0;
}
