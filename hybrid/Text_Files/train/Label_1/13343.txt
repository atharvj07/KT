#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vll vector<ll>

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vll a(n), b(m);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        for (int i = 0; i < m; i++) { cin >> b[i]; }
        sort(a.begin(), a.end());
        sort(b.begin(), b.end());
        ll cost = 0;
        for (int i = 0; i < n; i++) {
            int idx = lower_bound(b.begin(), b.end(), a[i]) - b.begin();
            cost += min(idx ? b[idx - 1] : b[0], b[idx]);
        }
        cout << cost << '\n';
    }
    return 0;
}
