n = int(input())
board = [input() for _ in range(n)]

def can_draw_crosses(n, board):
    for i in range(1, n-1):
        for j in range(1, n-1):
            if board[i][j] == '#' and board[i-1][j] == '#' and board[i+1][j] == '#' and board[i][j-1] == '#' and board[i][j+1] == '#':
                board[i] = board[i][:j] + '.' + board[i][j+1:]
                board[i-1] = board[i-1][:j] + '.' + board[i-1][j+1:]
                board[i+1] = board[i+1][:j] + '.' + board[i+1][j+1:]
                board[i][j-1] = '.'
                board[i][j+1] = '.'
                board[i-1] = list(board[i-1])
                board[i+1] = list(board[i+1])
                board[i] = list(board[i])
                board[i][j-1] = '.'
                board[i][j+1] = '.'
                board[i-1][j] = '.'
                board[i+1][j] = '.'
                board[i-1] = ''.join(board[i-1])
                board[i+1] = ''.join(board[i+1])
                board[i] = ''.join(board[i])

    for i in range(n):
        for j in range(n):
            if board[i][j] == '#':
                return "NO"
    return "YES"

print(can_draw_crosses(n, board))
