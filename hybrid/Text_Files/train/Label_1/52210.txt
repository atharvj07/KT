import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            char[] c = sc.next().toCharArray();
            int n = c.length;
            int[] last = new int[26];
            Arrays.fill(last, -1);
            for (int i = 0; i < n; i++) {
                if (last[c[i] - 'a'] != -1) {
                    c[last[c[i] - 'a']] = '*';
                }
                last[c[i] - 'a'] = i;
            }
            StringBuilder ans = new StringBuilder();
            for (char x : c) {
                if (x != '*') {
                    ans.append(x);
                }
            }
            System.out.println(ans);
        }
    }
}
