import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))

# Check for contradictions
for i in range(1, n):
    if a[i] == a[i - 1]:
        print("NO")
        exit()

# Find the longest strictly increasing and decreasing subsequences
inc = []
dec = []
for x in a:
    if not inc or x > inc[-1]:
        inc.append(x)
    if not dec or x < dec[-1]:
        dec.append(x)

# Split the sequence into increasing and decreasing subsequences
if len(inc) + len(dec) == n:
    print("YES")
    print(len(inc))
    print(*inc)
    print(len(dec))
    print(*dec)
else:
    print("NO")
