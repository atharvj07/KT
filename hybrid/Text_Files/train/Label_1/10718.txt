#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n), v(n + 1);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        v[a[i]] = i;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) if (v[i] != i - 1) ans++;
    cout << ans / 2 << endl;
    return 0;
}
