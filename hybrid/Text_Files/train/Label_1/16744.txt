#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  bool r = 1;
  int t = 0;
  for (int i = 0; i < n; i++) {
    r &= a[i] == t;
    t = (t + 1) % n;
  }
  cout << (r ? "Yes" : "No") << endl;
}
