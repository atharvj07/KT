n = int(input())
s = input()
m = int(input())

def min_replacements(n, s, m):
    max_beauty = 0
    min_replacement = 0

    for start in range(0, m):
        current_replacement = 0
        occurrences = 0
        for i in range(start, n, m):
            expected = 'a' if (i - start) % 2 == 0 else 'b'
            if s[i] == '?':
                current_replacement += 1
            elif s[i] != expected:
                if occurrences > max_beauty:
                    max_beauty = occurrences
                    min_replacement = current_replacement
                elif occurrences == max_beauty:
                    min_replacement = min(min_replacement, current_replacement)
                occurrences = 0
                current_replacement = 0
                continue
            occurrences += 1

        if occurrences > max_beauty:
            max_beauty = occurrences
            min_replacement = current_replacement
        elif occurrences == max_beauty:
            min_replacement = min(min_replacement, current_replacement)

    return min_replacement

print(min_replacements(n, s, m))
