import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.BufferedWriter;
import java.io.OutputStreamWriter;
import java.util.Arrays;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter out = new BufferedWriter(new OutputStreamWriter(System.out));
        int[] vals = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int n = vals[0], k = vals[1], q = vals[2];
        int[] tis = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        FenwickTree ft = new FenwickTree(n);
        for (int i = 0; i < n; i++) ft.update(i + 1, tis[i]);
        for (int i = 0; i < q; i++) {
            vals = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            if (vals[0] == 1) {
                ft.update(vals[1], -ft.get(vals[1]));
            } else {
                out.write(ft.query(Math.min(k, vals[1])) >= tis[vals[1] - 1] ? "YES\n" : "NO\n");
            }
        }
        out.flush();
        out.close();
        br.close();
    }

    private static class FenwickTree {
        int[] ft;
        int n;

        FenwickTree(int n) {
            this.n = n;
            ft = new int[n + 1];
        }

        int get(int i) {
            int sum = 0;
            while (i > 0) {
                sum += ft[i];
                i -= (i & -i);
            }
            return sum;
        }

        void update(int i, int delta) {
            while (i <= n) {
                ft[i] += delta;
                i += (i & -i);
            }
        }

        int query(int k) {
            int sum = 0, idx = 0;
            for (int bitWidth = n / 2; bitWidth > 0; bitWidth /= 2) {
                idx += bitWidth;
                if (idx <= n) {
                    if (get(idx) >= k) {
                        sum = idx;
                        idx -= bitWidth;
                    }
                }
            }
            return get(sum + 1);
        }
    }
}
