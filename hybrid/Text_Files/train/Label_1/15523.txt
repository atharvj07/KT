#include <iostream>
#include <vector>
#include <set>

using namespace std;

struct ghost {
  long long x, vx, vy;
  ghost(long long x, long long vx, long long vy) : x(x), vx(vx), vy(vy) {}
};

int main() {
  long long n, a, b;
  cin >> n >> a >> b;
  vector<ghost> ghosts;
  set<pair<long long, long long>> seen;
  for (long long i = 0; i < n; i++) {
    long long x, vx, vy;
    cin >> x >> vx >> vy;
    ghosts.push_back(ghost(x, vx, vy));
    seen.insert({x, a * x + b});
  }
  long long gx = 0;
  for (long long i = 0; i < n; i++) {
    for (long long j = i + 1; j < n; j++) {
      long long det = ghosts[i].vx * ghosts[j].vy - ghosts[i].vy * ghosts[j].vx;
      if (det == 0) continue;
      long long x = (ghosts[j].vy * (ghosts[i].x - ghosts[j].x) -
                     ghosts[i].vy * (ghosts[j].x - ghosts[i].x)) /
                    det;
      long long y = (ghosts[i].vx * (ghosts[j].x - ghosts[i].x) -
                     ghosts[j].vx * (ghosts[i].x - ghosts[j].x)) /
                    det;
      if (seen.count({x, y}) == 0) {
        gx += 2;
        seen.insert({x, y});
      }
    }
  }
  cout << gx << endl;
  return 0;
}
