#include <bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int n;
        cin >> n;
        vector<int> p(2 * n);
        for (int i = 0; i < 2 * n; i++) cin >> p[i];
        int l1 = 0, l2 = 0, r1 = 2 * n - 1, r2 = 2 * n - 1;
        bool fl = 0;
        for (int i = 0; i < 2 * n; i++)
        {
            if (p[i] <= n)
            {
                if (l1 > l2)
                {
                    fl = 1;
                    break;
                }
                l1++;
                l2++;
            }
            else
            {
                if (r1 < r2)
                {
                    fl = 1;
                    break;
                }
                r1--;
                r2--;
            }
        }
        if (fl)
            cout << "NO" << endl;
        else
            cout << "YES" << endl;
    }
    return 0;
}
