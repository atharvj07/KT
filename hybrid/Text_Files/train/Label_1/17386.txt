#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int n;
    cin >> n;
    vector<pair<int, int>> arrival_departures(n);
    vector<int> extra_times(n);
    for (int j = 0; j < n; j++) {
      int a, b;
      cin >> a >> b;
      arrival_departures[j] = {a, b};
    }
    for (int j = 0; j < n; j++) { cin >> extra_times[j]; }

    int current_time = 0;
    for (int j = 0; j < n; j++) {
      int time_at_station =
          arrival_departures[j].first + extra_times[j] - current_time;
      int min_stay = (arrival_departures[j].second - arrival_departures[j].first) /
                      2;
      if (time_at_station < min_stay) {
        current_time += min_stay - time_at_station;
      }
      current_time = arrival_departures[j].second;
    }
    cout << current_time << endl;
  }
  return 0;
}
