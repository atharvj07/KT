#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

vector<int> adj[51];
int dp[51][51];
int ans[51];

int dfs(int node, int sum) {
  if (dp[node][sum] != -1) return dp[node][sum];
  if (sum == 0) return dp[node][sum] = 1;
  for (int i = 0; i < adj[node].size(); i++) {
    int u = adj[node][i];
    if (sum - u >= 0) {
      if (dfs(u, sum - u)) {
        ans[sum] = node;
        return dp[node][sum] = 1;
      }
    }
  }
  return dp[node][sum] = 0;
}

int main() {
  int t;
  cin >> t;
  for (int i = 0; i <= 50; i++) {
    for (int j = 0; j <= 50; j++) {
      dp[i][j] = -1;
    }
  }
  for (int i = 1; i <= 9; i++) { adj[i].push_back(0); }
  for (int i = 10; i <= 50; i++) {
    int num = i;
    while (num > 0) {
      int digit = num % 10;
      adj[i].push_back(digit);
      num /= 10;
    }
  }
  while (t--) {
    int x;
    cin >> x;
    if (dfs(x, x)) {
      int num = 0;
      while (x > 0) {
        num *= 10;
        num += ans[x];
        x -= ans[x];
      }
      cout << num << endl;
    } else {
      cout << -1 << endl;
    }
  }
  return 0;
}
