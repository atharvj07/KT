#include <iostream>
using namespace std;
#define ll long long

int main() {
  ll n, p;
  cin >> n >> p;

  if (p < 0) {
    if (n % 2 == p % 2) {
      if (n >= p) {
        cout << 1 << endl;
        return 0;
      }
    } else
      n++;
  }

  ll ans = 0;
  while (n > 0) {
    ll tmp = 1;
    while (tmp <= n)
      tmp *= 2;
    tmp /= 2;
    n -= tmp + p;
    ans++;
  }

  if (n == 0)
    cout << ans << endl;
  else
    cout << -1 << endl;

  return 0;
}
