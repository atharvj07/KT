#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, ans = 0;
  cin >> n;
  map<char, int> mp;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < s.size(); j++) {
      mp[s[j]]++;
    }
  }
  int l = 0, a = 2;
  for (char i = 'j'; i >= 'a'; i--) {
    if (l == mp[i]) {
      a--;
      continue;
    }
    l = mp[i];
    ans += mp[i] * a;
    a++;
  }
  cout << ans << endl;
}
