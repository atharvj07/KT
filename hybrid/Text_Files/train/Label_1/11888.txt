#include <bits/stdc++.h>

using namespace std;

const char* dd="LRUD";
const int di[]={0,0,-1,1};
const int dj[]={1,-1,0,0};
int n,dist[25];

int main(){
	cin>>n;
	string s;
	cin>>s;
	for(int i=0;i<n;++i){
		for(int k=0;k<4;++k)
			if(s[i]==dd[k])
				dist[k]+=di[k],dist[3-k]+=dj[k];
	}
	cout<<max({abs(dist[0]),abs(dist[1]),abs(dist[2]),abs(dist[3])})<<endl;
}
