#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  vector<char> t;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'e' || s[i] == 'o') {
      if (i + 1 < n && s[i] == s[i + 1]) {
        t.push_back(s[i]);
        i++;
      } else
        t.push_back(s[i]);
    } else
      t.push_back(s[i]);
  }

  int m = t.size();
  for (int i = 1; i < m; i++) {
    if (t[i] == t[i - 1])
      t.erase(t.begin() + i), i--, m--;
  }

  for (int i = 0; i < m; i++) cout << t[i];
  cout << endl;

  return 0;
}
