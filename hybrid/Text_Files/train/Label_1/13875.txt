#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int k = 0, cnt = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] < 0) cnt++;
    if (cnt >= 3) { k++; cnt = 1; }
  }
  if (cnt) k++;
  cout << k << '\n';
  cnt = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] < 0) cnt++;
    if (cnt >= 3) { cout << i - cnt << ' '; cnt = 1; }
  }
  if (cnt) cout << n - cnt;
}
