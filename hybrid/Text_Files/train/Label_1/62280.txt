from collections import defaultdict, deque
import heapq
n, m, k = map(int, input().split())
g = defaultdict(list)
for _ in range(m):
    a, b, c = map(int, input().split())
    g[a].append((b, c))
    g[b].append((a, c))
vis = [False] * (n + 1)
dist = [float('inf')] * (n + 1)
q = [(0, 1)]
ans = []
while q:
    d, u = heapq.heappop(q)
    if vis[u]: continue
    vis[u] = True
    ans.append(d)
    if len(ans) == k: break
    for v, w in g[u]:
        if not vis[v]:
            heapq.heappush(q, (d + w, v))
print(ans[-1])
