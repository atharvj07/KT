import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
adj = [[] for i in range(n)]
for i in range(n-1):
    if a[i] > a[i+1]:
        adj[a[i]-1].append(a[i+1]-1)
        adj[a[i+1]-1].append(a[i]-1)
        a[i], a[i+1] = a[i+1], a[i]

vis = [False for i in range(n)]
ans = 0
for i in range(n):
    if vis[i]: continue
    q = [i]
    vis[i] = True
    ans += 1
    while q:
        x = q.pop()
        for y in adj[x]:
            if not vis[y]:
                vis[y] = True
                q.append(y)

print(ans)
