#include <bits/stdc++.h>
using namespace std;
int n, a[200005], b[200005];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  b[n + 1] = n + 2;
  for (int i = n; i >= 1; i--) {
    if (a[i] > a[b[i + 1]])
      b[i] = b[i + 1];
    else
      b[i] = i + 1;
  }
  int ans = 1;
  for (int i = 1; i <= n; i++) {
    ans = max(ans, b[i] - i);
  }
  cout << ans << endl;
  return 0;
}
