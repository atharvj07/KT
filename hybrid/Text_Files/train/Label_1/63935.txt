import sys

n, q, c = map(int, sys.stdin.readline().split())
stars = [[int(x) for x in sys.stdin.readline().split()] for _ in range(n)]
stars.sort(key=lambda x: (x[2], x[1]))

views = [[int(x) for x in sys.stdin.readline().split()] for _ in range(q)]

for view in views:
    t, x1, y1, x2, y2 = view
    ans = 0
    l, r = -1, -1

    for x, y, s in stars:
        if s + t > c:
            continue
        if y1 <= y <= y2 and x1 <= x <= x2:
            ans += min(s + t, c)
        elif y == y1 and x1 <= x <= x2:
            if r == -1 or x > r:
                r = x
                l = x1 if l == -1 else min(l, x1)
        elif y == y2 and x1 <= x <= x2:
            if l == -1 or x < l:
                l = x
                r = x2 if r == -1 else max(r, x2)

    if l != -1:
        ans += min(c, c - l + r - t + 1) * (y2 - y1)
    print(ans % (10**9 + 7))
