import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long p = sc.nextLong();
            long q = sc.nextLong();
            if (p % q != 0)
                System.out.println(p);
            else
                System.out.println(p / q * q);
        }
    }
}
