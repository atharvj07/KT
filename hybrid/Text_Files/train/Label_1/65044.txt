n, h, m = map(int, input().split())
dp = [[-1]*(h + 1) for i in range(n + 1)]
restrictions = []
for i in range(m):
    l, r, x = map(int, input().split())
    restrictions.append((l, r, x))

def get(i, height):
    if i == n + 1:
        return 0

    if dp[i][height] != -1:
        return dp[i][height]

    ans = -1e9
    for j in range(height + 1):
        if j == h:
            continue
        valid = True
        for l, r, x in restrictions:
            if l <= i <= r:
                if j > x:
                    valid = False
                    break
        if valid:
            ans = max(ans, get(i + 1, j) + j**2)

    dp[i][height] = ans
    return ans

print(get(1, 0))
