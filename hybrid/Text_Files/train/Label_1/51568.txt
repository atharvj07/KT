import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int k = sc.nextInt();
        int p = sc.nextInt();
        int[] slots = new int[n];

        for (int i = 0; i < p; i++) {
            int xi = sc.nextInt();
            slots[xi - 1] = 1;
        }

        Arrays.sort(slots);

        int bullets = k;
        for (int i = 0; i < n && bullets > 0; i++) {
            if (slots[i] == 0) {
                slots[i] = 1;
                bullets--;
            }
        }

        for (int slot : slots) {
            System.out.print(slot == 0 ? "." : "X");
        }
    }
}
