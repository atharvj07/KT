#include <bits/stdc++.h>
using namespace std;
#define ll long long

vector<ll>par;
vector<vector<ll> >adj;
ll findpar(ll x){
    if(x==par[x] or par[x]==-1)
        return x;
    return par[x]=findpar(par[x]);
}
int main() {
    ll n,m; cin>>n>>m;
    par.resize(n+1,-1);
    adj.resize(n+1);
    while(m--){
        ll t; cin>>t;
        if(t==1){
            ll x,y; cin>>x>>y;
            ll px=findpar(x);
            par[px]=findpar(y);
        }
        if(t==2){
            ll x; cin>>x;
            ll px=findpar(x);
            adj[px].push_back(x);
        }
        if(t==3){
            ll x,y; cin>>x>>y;
            ll px=findpar(x);
            if(adj[px].empty())
                cout<<"NO"<<endl;
            else{
                cout<<"YES"<<endl;
                adj[px].erase(std::remove(adj[px].begin(), adj[px].end(), y), adj[px].end());
            }
        }
    }
    return 0;
}
