import java.util.PriorityQueue;
import java.util.Scanner;

public class BeruTaxi {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int n = scanner.nextInt();
        PriorityQueue<Taxi> taxis = new PriorityQueue<>((t1, t2) -> Double.compare(t1.time, t2.time));
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            int v = scanner.nextInt();
            taxis.offer(new Taxi(x, y, v, a, b));
        }
        System.out.println(taxis.peek().time);
    }

    private static class Taxi {
        int x, y, v, a, b;
        double time;

        Taxi(int x, int y, int v, int a, int b) {
            this.x = x;
            this.y = y;
            this.v = v;
            this.a = a;
            this.b = b;
            this.time = Math.sqrt(Math.pow(x - a, 2) + Math.pow(y - b, 2)) / v;
        }
    }
}
