import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        char[] t = s.toCharArray();
        int j = 0;
        for (int i = 0; i < n; i++)
            if (i == n - 1 || (s.charAt(i) != s.charAt(i + 1) || s.charAt(i) == 'e' && s.charAt(i + 1) == 'e' || s.charAt(i) == 'o' && s.charAt(i + 1) == 'o'))
                t[j++] = s.charAt(i);
        System.out.println(new String(t, 0, j));
    }
}
