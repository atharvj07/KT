#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, l;
    cin >> n >> k >> l;
    vector<int> d(n);
    for (int &i : d) cin >> i;
    int p[2 * k];
    for (int i = 0; i < 2 * k; i++) p[i] = i % k;
    bool drown = false;
    for (int i = 0; i <= n && !drown; i++) {
      for (int j = 0; i <= n && j < 2 * k && !drown; j++) drown |= d[i - 1] + p[j] > l;
    }
    cout << (drown ? "No" : "Yes") << endl;
  }
}
