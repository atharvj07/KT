#include <bits/stdc++.h>
using namespace std;
const int N = 101, M = 1e5 + 1;
struct star {
    int x, y, s;
};
star stars[M];
int main() {
    int n, q, c;
    cin >> n >> q >> c;
    for (int i = 1; i <= n; i++) {
        cin >> stars[i].x >> stars[i].y >> stars[i].s;
    }
    while (q--) {
        int t, x1, y1, x2, y2, ans = 0;
        cin >> t >> x1 >> y1 >> x2 >> y2;
        for (int i = 1; i <= n; i++) {
            int x = stars[i].x, y = stars[i].y, s = stars[i].s;
            s += t;
            if (s > c)
                s = c;
            if (x >= x1 && x <= x2 && y >= y1 && y <= y2)
                ans += s;
        }
        cout << ans << endl;
    }
    return 0;
}
