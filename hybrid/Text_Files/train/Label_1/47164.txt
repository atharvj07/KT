import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.next();
        int n = s.length();
        int[] cnt = new int[26];
        for (char ch : s.toCharArray()) {
            cnt[ch - 'a']++;
        }
        int mxCnt = 0;
        for (int i = 0; i < 26; i++) {
            mxCnt = Math.max(mxCnt, cnt[i]);
        }
        double ans = 1.0 * mxCnt / n;
        System.out.println(ans);
    }
}
