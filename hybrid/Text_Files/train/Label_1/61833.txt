l,w,x,y,vx,vy = map(int,input().split())
if vx==0 and (y==0 or y==w): print(x,y)
elif vy==0 and (x==0 or x==l): print(x,y)
elif l-x == w-y:
    if (vx>0 and vy>0 and y!=w) or (vx<0 and vy<0 and y!=0): print(l,w)
    else: print(-1)
elif x == y:
    if (vx<0 and vy>0 and x!=0) or (vx>0 and vy<0 and x!=l): print(0,0)
    else: print(-1)
else: print(-1)
