inline int read() {
    int x=0,f=1;
    char ch=getchar();
    while (ch<'0'||ch>'9') {
        if (ch=='-') f=-1;
        ch=getchar();
    }
    while (ch>='0'&&ch<='9') {
        x=x*10+ch-'0';
        ch=getchar();
    }
    return f*x;
}
int main() {
    int k,n,ans=0,f=0;
    string s[N];
    int a[N];
    k=read();n=read();
    rep(i,1,k) {
        cin>>s[i];
        a[i]=read();
        f|=a[i]+a[i]>0;
    }
    rep(i,1,k) {
        bool pd=1,pd1=1;
        rep(j,1,n) {
            if (s[i][j]!=s[i][n-j+1]) pd=0;
            if (s[i][j]!=s[i-1][n-j+1]) pd1=0;
        }
        if (pd) ans+=a[i];
        else if (pd1) ans+=max(a[i],a[i-1]);
    }
    if (!f) ans=max(ans,0);
    cout<<ans<<endl;
    return 0;
}
