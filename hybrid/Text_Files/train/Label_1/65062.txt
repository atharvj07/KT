n=int(input())
a=list(map(int,input().split()))
ai,aj=[],[]
for i in range(n):
    if a[i]==a[0]: ai.append(i+1)
    if a[i]==a[-1]: aj.append(i+1)
si=sum(a)
if si%2==0:
    si-=a[0]
si//=2
for i in range(n):
    if si<=0: break
    if a[i]>a[0]:
        si-=a[i]-a[0]
        a[i]=a[0]
        print(i+1,end=" ")
    elif a[i]<a[0]:
        si-=a[0]-a[i]
        a[i]=a[0]

print(sum(a),len(a)-n)
