import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long a = sc.nextLong();
        long b = sc.nextLong();
        long p = sc.nextLong();
        long q = sc.nextLong();
        long min = (m + a - 1) / a;
        if ((p * min + q * ((m + b - 1) / b)) >= m)
            System.out.println(1);
        else
            System.out.println(0);
    }
}
