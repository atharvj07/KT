#include <cstdio>
#include <iostream>
int main() {
  int n;
  std::cin >> n;
  if (n % 10 == 0) {
    std::cout << n;
    return 0;
  }
  if (n > 0)
    std::cout << n % 10;
  else if (n < 0 && -n % 10 > -n / 10)
    std::cout << -n % 10;
  else if (n < 0 && -n % 10 < -n / 10)
    std::cout << -n / 10;
  return 0;
}
