#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n,x;
    cin >> n >> x;
    int a[n];
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int sum = 0;
    for (int i = 0; i < n; i++) sum += a[i];
    if (sum == x) {
        sort(a, a + n);
        int last = -1;
        for (int i = 0; i < n; i++) {
            if (a[i] == last) {
                cout << "NO" << endl;
                return 0;
            }
            last = a[i];
        }
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
