n = int(input())
room = [input().strip() for _ in range(n)]

row_frequency = {}
max_clean_rows = 0

for row in room:
    if row in row_frequency:
        row_frequency[row] += 1
    else:
        row_frequency[row] = 1
    max_clean_rows = max(max_clean_rows, row_frequency[row])

print(max_clean_rows)
