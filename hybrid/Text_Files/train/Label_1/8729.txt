#include <bits/stdc++.h>
#define int long long
using namespace std;
vector<int> v, w;
int ans, n, m, dp[202][1<<9];
int go(int pos, int ors) {
  if (pos == n) return 0;
  if (dp[pos][ors] != -1) { return dp[pos][ors]; }
  int ret = 1e18;
  for (int i = 0; i < m; i++) {
    int tmp = v[pos] & w[i];
    if ((ors | tmp) > ret) continue;
    ret = min(ret, go(pos + 1, ors | tmp));
  }
  return dp[pos][ors] = ret;
}
signed main() {
  cin >> n >> m;
  v.resize(n), w.resize(m);
  for (int i = 0; i < n; i++) cin >> v[i];
  for (int i = 0; i < m; i++) cin >> w[i];
  memset(dp, -1, sizeof(dp));
  ans = go(0, 0);
  cout << ans << "\n";
  return 0;
}
