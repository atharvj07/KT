import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Scanner;
import java.util.Set;

public class SupersonicRocket {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Point> engine1 = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            engine1.add(new Point(scanner.nextInt(), scanner.nextInt()));
        }

        List<Point> engine2 = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            engine2.add(new Point(scanner.nextInt(), scanner.nextInt()));
        }

        if (isSafe(engine1, engine2)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isSafe(List<Point> engine1, List<Point> engine2) {
        Set<Point> powerField = new HashSet<>();
        for (Point p1 : engine1) {
            for (Point p2 : engine2) {
                powerField.add(p1.combine(p2));
            }
        }

        for (Point p1 : engine1) {
            Set<Point> newPowerField = new HashSet<>();
            for (Point p2 : engine2) {
                newPowerField.add(p1.combine(p2));
            }
            if (!newPowerField.containsAll(powerField)) {
                return false;
            }
        }

        for (Point p2 : engine2) {
            Set<Point> newPowerField = new HashSet<>();
            for (Point p1 : engine1) {
                newPowerField.add(p1.combine(p2));
            }
            if (!newPowerField.containsAll(powerField)) {
                return false;
            }
        }

        return true;
    }

    private static class Point {
        private final int x;
        private final int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }

        public Point combine(Point other) {
            return new Point((x + other.x) / 2, (y + other.y) / 2);
        }

        @Override
        public int hashCode() {
            return x * 31 + y;
        }

        @Override
        public boolean equals(Object obj) {
            if (obj instanceof Point) {
                Point other = (Point) obj;
                return x == other.x && y == other.y;
            }
            return false;
        }
    }
}
