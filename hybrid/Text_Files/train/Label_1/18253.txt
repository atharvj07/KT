#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  map<int, pair<int, int>> mp;
  while (n--) {
    string s;
    int x;
    cin >> s >> x >> s;
    if (s[0] == 'Y') {
      if (s[1] == '>') mp[x] = pair<int, int>(0, 1);
      if (s[1] == '<') mp[x] = pair<int, int>(-1, 0);
      if (s[1] == '=') mp[x] = pair<int, int>(0, 0);
    } else {
      if (s[1] == '>') mp[x] = pair<int, int>(1, 0);
      if (s[1] == '<') mp[x] = pair<int, int>(0, 1);
      if (s[1] == '=') mp[x] = pair<int, int>(1, 1);
    }
  }
  int ans = 0;
  int r = 0;
  bool flag = true, flg = false;
  for (auto it : mp) {
    int x = it.first;
    pair<int, int> p = it.second;
    if (p.first == 0 && !p.second) { ans = x; flg = true; }
    if (p.second) {
      if (x <= r) {
        flag = false;
        break;
      } else
        r = x;
    } else {
      if (x >= r) {
        flag = false;
        break;
      } else
        r = x;
    }
  }
  if (!flag || !flg)
    cout << "Impossible";
  else
    cout << ans;
  return 0;
}
