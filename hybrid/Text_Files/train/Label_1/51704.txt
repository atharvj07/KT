import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n + 1];
        for (int i = 1; i <= n; ++i) {
            arr[in.nextInt()] = i;
        }

        ArrayList<Operation> operations = new ArrayList<>();
        int[] rev = new int[n + 1];
        int lst = 0, lstj = 0;
        for (int i = 1; i <= n; ++i) {
            if (arr[i] == i) {
                continue;
            }

            int j = i;
            int len = 0;
            while (arr[j] != i) {
                lst = j;
                j = arr[j];
                ++len;
            }

            operations.add(new Operation(lstj, len, n - lst - len));

            rev[i] = lst;
            rev[j] = lst;
            while (arr[lst] != i) {
                int t = arr[lst];
                arr[lst] = j;
                j = lst;
                lst = t;
            }

            lstj = lst;
        }

        System.out.println(operations.size());
        for (Operation op : operations) {
            System.out.println(op.a + " " + op.b + " " + op.c);
        }
    }
}

class Operation {
    int a, b, c;

    public Operation(int a, int b, int c) {
        this.a = a;
        this.b = b;
        this.c = c;
    }
}
