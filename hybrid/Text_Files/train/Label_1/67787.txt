def find_lis(arr):
    n = len(arr)
    lis = [1] * n
    for i in range(1, n):
        for j in range(i):
            if arr[i] > arr[j]:
                lis[i] = max(lis[i], lis[j] + 1)
    return max(lis)

def can_split_to_xor_zero(arr):
    xor_sum = find_lis(arr)
    for i in range(1, len(arr)):
        lhs_lis = find_lis(arr[:i])
        rhs_lis = find_lis(arr[i:])
        if xor_sum == (lhs_lis ^ rhs_lis):
            return True
    return False

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    arr = list(map(int, input().strip().split()))
    # Check if it's possible
    if n % 2 == 0 or can_split_to_xor_zero(arr):
        print("YES")
    else:
        print("NO")
