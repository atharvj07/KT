#include <algorithm>
#include <bitset>
#include <vector>
#include <iostream>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<bitset<20>> soldiers(m + 1);
  for (int i = 0; i <= m; ++i) {
    long long x;
    cin >> x;
    soldiers[i] = bitset<20>(x);
  }
  int cnt = 0;
  for (int i = 0; i < m; ++i) {
    cnt += (soldiers[m] ^ soldiers[i]).count() <= k;
  }
  cout << cnt << endl;
  return 0;
}
