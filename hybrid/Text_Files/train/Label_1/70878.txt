n, q = map(int, input().split())
unread = [0] * (n + 1)
for t in range(q):
    t, *args = map(int, input().split())
    if t == 1:
        unread[args[0]] += 1
    elif t == 2:
        unread[args[0]] = 0
    else:
        unread = [max(u - (t >= i) for i, u in enumerate(unread))]
    print(sum(unread))
