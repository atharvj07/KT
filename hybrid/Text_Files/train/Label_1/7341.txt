#include <bits/stdc++.h>

#define ll long long
#define pb push_back
#define vi vector<int>
#define ii pair<int,int>
#define ff first
#define ss second
#define endl "\n"
const int N = 1e5;
using namespace std;

int main() {
	int t;
	cin >> t;
	while(t--) {
		string s;
		cin >> s;
		stack<char> st;
		int ans = 0;
		bool ok = true;
		for(char c: s) {
			if(c == '(' || c == '[') {
				st.push(c);
			} else {
				if(st.empty()) {
					ok = false;
				} else if((c == ')' && st.top() == '(') || (c == ']' && st.top() == '[')) {
					st.pop();
					ans++;
				} else {
					ok = false;
				}
			}
		}
		ans += st.empty() && ok;
		cout << ans << endl;
	}
	return 0;
}
