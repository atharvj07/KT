t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().split()))
    # Count frequency of each number
    freq = {}
    for num in a:
        if num in freq:
            freq[num] += 1
        else:
            freq[num] = 1
    # Sort numbers by their value, but put numbers with frequency > 1 at the end
    unique_sorted = sorted(freq)
    rest = [num for num in unique_sorted if freq[num] > 1]
    result = unique_sorted[:]
    for num in rest:
        result += [num] * (freq[num] - 1)
    print(' '.join(map(str, result)))
