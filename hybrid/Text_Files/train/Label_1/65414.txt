import sys
sys.setrecursionlimit(2000)
def dfs(v:int, p:int):
    vis[v] = True
    for to in graph[v]:
        if not vis[to]:
            if dfs(to, v):
                return True
        elif to!=p:
            return True
    return False
n,m = map(int,input().split())
graph = [[] for i in range(n+1)]
for i in range(m):
    a,b=map(int,input().split())
    graph[a].append(b)
vis = [False]*(n+1)
print('YES' if dfs(1,-1) else 'NO')
