#include<bits/stdc++.h>
using namespace std;

struct Event
{
	int type,d,m;
};

int main()
{
	int n,i,req=0;
	cin>>n;

	vector<Event>ev;

	for(i=0;i<n;++i)
	{
		int d,m,p,t;
		cin>>m>>d>>p>>t;

		ev.push_back({0,d,m-t});
		ev.push_back({1,d+t,m});
		req=max(req,p);
	}

	sort(ev.begin(),ev.end(),[](Event a,Event b)->bool{return a.d<b.d || (a.d==b.d && a.type<b.type);});

	int ans=0,cur=0;

	for(auto v: ev)
	{
		if(v.type==0)
			cur++;
		else
			cur--;

		ans=max(ans,cur);
	}

	cout<<max(ans,req)<<endl;

	return 0;
}
