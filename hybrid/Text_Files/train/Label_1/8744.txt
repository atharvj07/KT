#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, k, m;
    cin >> n >> k >> m;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    long long ans = n * m;
    for (int i = 0; i < n; ++i) {
        int j = (i + k - 1) % n;
        ans -= (a[i] == a[j]);
    }
    cout << ans << endl;
    return 0;
}
