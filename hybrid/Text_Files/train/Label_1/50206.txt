import java.util.Scanner;

public class CandyShop {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        long[] allowances = new long[n];
        for (int i = 0; i < n; i++) {
            allowances[i] = scanner.nextLong();
        }

        long maxCandies = 0;
        for (int i = 1; i <= m; i++) {
            long candiesToSell = i;
            long remainingKids = n;
            for (long allowance : allowances) {
                long candiesThisKidCanBuy = allowance / i;
                candiesToSell -= candiesThisKidCanBuy;
                remainingKids--;
                if (candiesToSell <= 0) {
                    break;
                }
            }
            if (candiesToSell <= 0) {
                maxCandies = Math.max(maxCandies, i * remainingKids);
            }
        }

        System.out.println(maxCandies);
    }
}
