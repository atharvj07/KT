def can_fit(lines, words, max_width):
    current_line = 0
    current_width = 0
    for word in words:
        parts = word.split('-')
        for i, part in enumerate(parts):
            if current_width + len(part) <= max_width:
                current_width += len(part)
            else:
                current_line += 1
                current_width = len(part)
                if current_line >= lines:
                    return False
            if i < len(parts) - 1:  # If there are more parts, account for the hyphen
                if current_width + 1 <= max_width:
                    current_width += 1  # Hyphen fits in the current line
                else:
                    current_line += 1  # Move hyphen to the next line
                    current_width = 1  # Reset width because of new line
                    if current_line >= lines:
                        return False
            if i < len(parts) - 1 or (i == len(parts) - 1 and word != words[-1]):
                if current_width + 1 <= max_width:  # Check if there's space for a space or moving to the next word
                    current_width += 1
                else:
                    current_line += 1
                    current_width = 0
                    if current_line >= lines:
                        return False
    return True

def find_min_width(k, ad_text):
    words = ad_text.split(' ')
    left, right = 1, len(ad_text)
    result = right
    while left <= right:
        mid = (left + right) // 2
        if can_fit(k, words, mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1
    return result

k = int(input().strip())
ad_text = input().strip()
print(find_min_width(k, ad_text))
