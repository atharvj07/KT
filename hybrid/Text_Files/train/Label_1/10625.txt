#include<bits/stdc++.h>
using namespace std;

#define vt vector
#define pb push_back
#define all(c) (c).begin(),(c).end()
#define f first
#define s second
#define vi vector<int>
#define vvi vector<vi>
#define pii pair<int,int>
#define unmap unordered_map
#define unset unordered_set
#define mod 1000000007
#define inf 1e18
#define nline '\n'

void solve(){
    int n,m,k;
    cin>>n>>m>>k;
    vi a[m];
    for(int i=0;i<n;i++){
        int j;
        for(j=0;j<m;j++){
            int x;
            cin>>x;
            a[j].pb(x);
        }
    }

    while(k--){
        int x,y;
        cin>>x>>y;
        x--;
        y--;
        for(int i=0;i<m;i++){
            if(i==y or !a[i][x]){
                continue;
            }
            a[i][x]++;
        }
    }

    for(int i=0;i<m;i++){
        for(int j=0;j<n;j++){
            cout<<a[i][j]<<" ";
        }
    }
    cout<<nline;
}

int32_t main(){
    ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t=1;
    // cin>>t;
    while(t--) solve();
    return 0;
}
