#include<vector>
#include<iostream>
#include<algorithm>
using namespace std;
int main(){
    int t;cin>>t;
    while(t--){
        int n;cin>>n;
        vector<pair<int,int>>a(n);
        for(int i=0;i<n;i++){
            int p;cin>>p;
            a[i].first=p;
        }
        for(int i=0;i<n;i++){
            int p;cin>>p;
            a[i].second=p;
        }
        sort(a.begin(),a.end());
        int ans=1;
        for(int i=1;i<n;i++){
            if(a[i].first<a[i-1].first){
                if(a[i].second!=a[i-1].second){
                    continue;
                }
                else{
                    ans=0;
                    break;
                }
            }
        }
        if(ans) cout<<"Yes"<<endl;
        else cout<<"No"<<endl;
    }
}
