#include <bits/stdc++.h>

using namespace std;

int main() {
  int k;
  cin >> k;
  vector<vector<int>> a(k);
  for (int i = 0; i < k; ++i) {
    int n;
    cin >> n;
    a[i].resize(n);
    int sum = 0;
    for (int j = 0; j < n; ++j) {
      cin >> a[i][j];
      sum += a[i][j];
    }
    if (sum % n != 0) {
      cout << "No" << endl;
      return 0;
    }
  }
  int total = 0;
  for (const auto& ai : a) {
    total += ai[0];
  }
  if (total % k != 0) {
    cout << "No" << endl;
    return 0;
  }
  int valToInsert = total / k;
  cout << "Yes" << endl;
  for (int i = 1; i < k; ++i) {
    cout << a[i][0] - valToInsert << " " << 1 << endl;
    a[0].push_back(a[i][0]);
    a[i].pop_back();
  }
  return 0;
}
