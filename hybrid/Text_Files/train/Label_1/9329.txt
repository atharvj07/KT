#include <bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false); cin.tie(0); cout.tie(0);

int32_t main() {
    IOS;
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        string s, t; cin >> s >> t;
        vector<pair<int, int>> ans;
        int diff = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] != t[i]) {
                diff++;
                for (int j = i + 1; j < n; j++) {
                    if (s[j] == s[i] && t[j] == t[i]) {
                        ans.push_back({i + 1, j + 1}), swap(s[i], t[j]), diff--;
                    } else if (t[j] == s[i] && s[j] == t[i]) {
                        ans.push_back({i + 1, j + 1}), swap(s[j], t[i]), diff--;
                    }
                    if (!diff) break;
                }
            }
        }
        if (diff) cout << "No\n";
        else {
            cout << "Yes\n";
            cout << ans.size() << '\n';
            for (auto p : ans) {
                cout << p.first << ' ' << p.second << '\n';
            }
        }
    }
    return 0;
}
