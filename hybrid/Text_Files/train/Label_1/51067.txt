import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = reader.readLine().split(" ");
        int n = Integer.parseInt(line1[0]), d = Integer.parseInt(line1[1]) - 1;
        int[] points = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] awards = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int sum = points[d] + awards[0];
        int rank = 1;
        for (int i = 0; i < n; i++) {
            if (i == d) continue;
            if (points[i] + awards[rank] > sum) {
                rank++;
            }
        }
        System.out.println(rank);
    }
}
