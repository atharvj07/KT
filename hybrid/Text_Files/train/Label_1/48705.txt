import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt();
        int x = input.nextInt();

        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }

        Arrays.sort(arr);

        int median = arr[(n - 1) / 2];

        if (x <= median) {
            int min = Integer.MAX_VALUE;
            for (int i = 0; i <= n; i++) {
                if (i % 2 == 1) {
                    if (i < n && arr[i] < x) {
                        min = Math.min(min, x - arr[i]);
                    }
                } else {
                    if (i > 0 && arr[i - 1] > x) {
                        min = Math.min(min, arr[i - 1] - x);
                    }
                }
            }
            if (min == Integer.MAX_VALUE) {
                min = -1;
            }
            System.out.println(min);
        } else {
            int min = Integer.MAX_VALUE;
            for (int i = 0; i <= n; i++) {
                if (i % 2 == 0) {
                    if (i < n && arr[i] < x) {
                        min = Math.min(min, x - arr[i]);
                    }
                } else {
                    if (i > 0 && arr[i - 1] > x) {
                        min = Math.min(min, arr[i - 1] - x);
                    }
                }
            }
            if (min == Integer.MAX_VALUE) {
                min = -1;
            }
            System.out.println(min);
        }
    }
}
