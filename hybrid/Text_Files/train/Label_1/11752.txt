#include <iostream>
#include <vector>
#include <unordered_map>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        unordered_map<long long, int> freq;
        long long sum = 0;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            if (sum == 1) ans++;
            ans += freq[sum - 1];
            freq[sum]++;
        }
        cout << ans << endl;
    }
    return 0;
}
