import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class JOI {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int N = Integer.parseInt(line[0]);
        int T = Integer.parseInt(line[1]);
        int Q = Integer.parseInt(line[2]);

        int[] houses = new int[N];
        int[] directions = new int[N];
        for (int i = 0; i < N; i++) {
            line = br.readLine().split(" ");
            houses[i] = Integer.parseInt(line[0]);
            directions[i] = Integer.parseInt(line[1]);
        }

        int[] importantPeople = new int[Q];
        for (int i = 0; i < Q; i++) {
            importantPeople[i] = Integer.parseInt(br.readLine()) - 1;
        }

        Arrays.sort(houses);

        for (int i = 0; i < Q; i++) {
            int houseIndex = importantPeople[i];
            int housePosition = houses[houseIndex];
            int direction = directions[houseIndex];

            int positionAfterT = housePosition + direction * T;
            System.out.println(positionAfterT);
        }
    }
}
