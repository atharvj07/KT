import java.util.Scanner;

public class Caracal {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long H = scanner.nextLong();

        long count = 0;
        while (H > 0) {
            if (H == 1) {
                count++;
                break;
            }
            count++;
            H = (H - 1) / 2;
        }

        System.out.println(count);
    }
}
