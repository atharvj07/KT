n = int(input())
a = [int(x) for x in input().split()]
m = int(input())
b = [0] * n
for i in range(n):
    b[a[i] - 1] = i
inv = 0
for i in range(n):
    for j in range(i):
        if a[i] < a[j]:
            inv += 1
odd = inv % 2
for i in range(m):
    l, r = [int(x) for x in input().split()]
    l -= 1
    r -= 1
    inv -= (((b[l] - l) * (b[r] - r) - (b[r] - l) * (b[l] - r)) % 2)
    for i in range(l, r + 1):
        for j in range(l, i):
            if a[i] < a[j]:
                inv += 1
        for j in range(i + 1, r + 1):
            if a[i] > a[j]:
                inv += 1
    l, r = r, l
    a[l:r + 1] = reversed(a[l:r + 1])
    for i in range(l, r + 1):
        b[a[i] - 1] = i
    odd = inv % 2
    print(["even", "odd"][odd])
