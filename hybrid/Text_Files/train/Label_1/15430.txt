#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct Problem {
    int a, t;
};

bool cmp(const Problem& p1, const Problem& p2) {
    return p1.t * p1.a < p2.t * p2.a;
}

int main() {
    int n, T;
    cin >> n >> T;
    vector<Problem> problems(n);
    for (int i = 0; i < n; i++) {
        cin >> problems[i].a >> problems[i].t;
    }
    sort(problems.begin(), problems.end(), cmp);
    int maxScore = 0;
    int time = 0;
    vector<int> ans;
    for (int i = 0; i < n; i++) {
        if (time + problems[i].t <= T) {
            time += problems[i].t;
            maxScore++;
            ans.push_back(i + 1);
        }
    }
    cout << maxScore << endl;
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
