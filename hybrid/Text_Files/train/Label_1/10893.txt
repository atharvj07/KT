#include<iostream>
#include<vector>
using namespace std;
const int size = 1e5 + 10;
int main() {
	int n, m, k;
	cin >> n >> m >> k;
	vector<int>cnt1(size, 0), cnt2(size, 0);
	for (int i = 0; i < n; i++) {
		int a; cin >> a;
		cnt1[a]++;
	}
	for (int i = 0; i < m; i++) {
		int a; cin >> a;
		cnt2[a]++;
	}
	bool flag = 0;
	for (int i = 1; i <= k; i++) {
		if (cnt1[i] != 0 && cnt2[i] == 0) {
			flag = 1;
		}
	}
	if (flag) {
		cout << "YES" << endl;
	}
	else {
		cout << "NO" << endl;
	}
	return 0;
}
