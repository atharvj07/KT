#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pi;

#define F first
#define S second
const ll MM = 2e5 + 5;
const ll mod = 1e9 + 7;
#define pb push_back
#define IOS ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);

ll n, m, x, y, c, k;
ll arr[MM];
vector<ll> adj[MM];
set<ll> ans;

void dfs(ll u, ll p) {
    ans.insert(u);
    for (auto v : adj[u]) {
        if (v == p) continue;
        dfs(v, u);
    }
}

int main() {
    IOS;
    cin >> n;
    for (ll i = 1; i <= n - 1; i++) {
        cin >> x >> y;
        adj[x].pb(y);
        adj[y].pb(x);
    }
    dfs(1, 0);
    for (ll i = 1; i <= n; i++) {
        if (!ans.count(i)) cout << -1 << " ";
        else cout << i << " ";
    }
    return 0;
}
