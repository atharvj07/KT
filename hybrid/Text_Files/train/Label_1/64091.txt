import sys
n,m=map(int,input().split())
h=[[-1,-1] for i in range(100005)]
ans,ansdist=1000000000000000000,1000000000000000000
for i in range(int(input())):
    x,y=map(int,input().split())
    if h[x][y]==-1:
        h[x][y]=0
    h[x][y]+=1
    ansdist=min(ansdist,y)
for i in range(int(input())):
    x,y=map(int,input().split())
    if h[x][y]>0:
        ans=min(ans,y-ansdist)
print(ans)
for i in range(1,m+1):
    for j in range(1,n+1):
        if h[j][i]>0 and y-ansdist==i-ansdist:
            print(i)
            sys.exit()
