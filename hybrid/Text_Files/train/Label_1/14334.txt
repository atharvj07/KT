#include <bits/stdc++.h>

using namespace std;
bool ok(long long n, int k) {
    if (k & 1) {
        if (n % 2 == 0)
            return false;
        else if (n < k)
            return false;
    }
    if (n % 2 == 1)
        return n > k;
    return n > 2 * k;
}
void solve() {
    long long n;
    int k;
    cin >> n >> k;
    if (!ok(n, k)) {
        cout << "NO" << endl;
        return;
    }
    cout << "YES";
    for (int i = 0; i < k - 1; i++)
        cout << (n % 2 == 0 ? (n / 2) : 1) << ' ';
    cout << n - (n % 2 == 0 ? (n / 2) * (k - 1) : (n - 1));
    cout << endl;
}
int main() {
    int t;
    cin >> t;
    while (t--) { solve(); }
}
