n, m = list(map(int, input().split()))
nums = [0] * (m + 1)
cost = [0] * (m + 1)
for i in range(m):
    a, b = list(map(int, input().split()))
    nums[i + 1] = a
    cost[i + 1] = b
dp = [0] * (n + 1)
for i in range(1, n + 1):
    a = {}
    for j in range(m + 1):
        if nums[j] <= i:
            a[nums[j]] = min(a.get(nums[j], 1e9), cost[j])
    if i > 1:
        dp[i] = min(dp[i - 1], dp[i - 2])
    for k in a:
        if k + 1 == i or k - 1 == i:
            dp[i] = min(dp[i], a[k])
        dp[i] += a[k]
print(dp[n])
