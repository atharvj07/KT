n, m = map(int, input().split())
a = list(map(int, input().split()))
prev = [-1] * (m + 1)
res = []
l = 1
cur = 0
for i in range(n):
    cur = cur // a[i] * a[i] + a[i]
    for j in range(m, -1, -1):
        if (cur >> j) & 1:
            if prev[j] != -1:
                if len(res) < i - prev[j]:
                    res = a[prev[j] + 1: i + 1]
                    l = cur
            prev[j] = i
print(l, len(res))
print(*res)
