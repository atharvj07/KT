import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        char[] s = scanner.next().toCharArray();
        int d = scanner.nextInt();
        int[] b0 = new int[n];
        int[] b1 = new int[n];
        int[] b2 = new int[n];
        for (int i = n - 1; i >= 0; i--) {
            b0[i] = s[i] - '0';
            b1[i] = b0[i];
            b2[i] = b0[i];
        }
        for (int i = 0; i < n; i++) {
            if (d > 0) {
                b2[i] = b1[i] ^ 1;
                d--;
            } else {
                break;
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            if (b0[i] == 0 && b1[i] == 0 && b2[i] == 0) {
                continue;
            }
            if (b0[i] == 1) {
                System.out.print(0);
            } else {
                if (b1[i] == 1) {
                    System.out.print(1);
                } else {
                    System.out.print(b2[i]);
                }
            }
        }
    }
}
