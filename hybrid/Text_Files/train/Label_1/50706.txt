import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextLong();
        }
        Arrays.sort(a);
        Arrays.sort(b);
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 0 || b[i] == 0) {
                ans++;
            } else {
                break;
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] == 0 || b[i] == 0) {
                ans++;
            } else {
                break;
            }
        }
        System.out.println(ans);
    }
}
