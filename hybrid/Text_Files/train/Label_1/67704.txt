def valid(r, c, a):
    if r < 0 or r >= n or c < 0 or c >= m: return False
    return a[r][c] != '#'

n, m = map(int, input().split())
a = [list(input()) for _ in range(n)]
s = input()

r, c, d = 0, 0, 0
for i in s:
    if i != d:
        d = i
        r += [0, -1, 0, 1][int(i)]
        c += [1, 0, -1, 0][int(i)]
    if not valid(r, c, a):
        print(0)
        exit()
    if a[r][c] == 'E':
        print(1)
        exit()
print(0)
