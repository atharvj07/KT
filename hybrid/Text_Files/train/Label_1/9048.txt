#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n), t(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> t[i];
    }
    int ans = 0;
    for (int i = 0; i < n - k + 1; i++) {
        int sum = 0;
        for (int j = i; j < i + k; j++) {
            sum += a[j] * t[j];
        }
        ans = max(ans, sum);
    }
    cout << ans << endl;
    return 0;
}
