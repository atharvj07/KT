import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t1 = sc.nextInt();
        int t2 = sc.nextInt();
        int x1 = sc.nextInt();
        int x2 = sc.nextInt();
        int t0 = sc.nextInt();
        int y2 = (t0 - t1) * x1 / (t2 - t1);
        int y1 = x1 - y2;
        if (y2 > x2) {
            y2 = x2;
            y1 = 0;
        }
        System.out.println(y1 + " " + y2);
    }
}
