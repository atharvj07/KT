import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class FindSubnetMask {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        List<Integer> ipAddresses = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String ipAddress = scanner.nextLine();
            ipAddresses.add(convertIpAddressToInteger(ipAddress));
        }

        List<Integer> subnetMasks = findSubnetMasks(ipAddresses, k);

        if (subnetMasks.isEmpty()) {
            System.out.println(-1);
        } else {
            int minOnesSubnetMask = findSubnetMaskWithLeastOnes(subnetMasks);
            String subnetMask = convertIntegerToIpAddress(minOnesSubnetMask);
            System.out.println(subnetMask);
        }
    }

    // Converts an IP address string to an integer representation.
    private static int convertIpAddressToInteger(String ipAddress) {
        String[] octets = ipAddress.split("\\.");
        int result = 0;
        for (int i = 0; i < 4; i++) {
            result += Integer.parseInt(octets[i]) << (8 * (3 - i));
        }
        return result;
    }

    // Converts an integer representation of an IP address to a string.
    private static String convertIntegerToIpAddress(int ipAddress) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 4; i++) {
            sb.append((ipAddress >> (8 * (3 - i))) & 0xFF).append(".");
        }
        sb.deleteCharAt(sb.length() - 1); // Remove trailing dot
        return sb.toString();
    }

    // Finds all possible subnet masks that divide the given IP addresses into the specified number of networks.
    private static List<Integer> findSubnetMasks(List<Integer> ipAddresses, int k) {
        List<Integer> subnetMasks = new ArrayList<>();
        int maxSubnetMask = 0xFFFFFFFF; // 255.255.255.255 in integer representation

        // Iterate over possible subnet mask values from most specific (255.255.255.255) to least specific (0.0.0.0)
        for (int subnetMask = maxSubnetMask; subnetMask >= 0; subnetMask--) {
            int networkCount = 0;

            // Check if the current subnet mask divides the IP addresses into k networks
            for (int ipAddress : ipAddresses) {
                int networkAddress = ipAddress & subnetMask; // Perform bitwise AND to get the network address

                // If a new network address is encountered, increment the network count
                if (!subnetMasks.contains(networkAddress)) {
                    subnetMasks.add(networkAddress);
                    networkCount++;
                }
            }

            // If the current subnet mask divides the IP addresses into exactly k networks, add it to the result list
            if (networkCount == k) {
                return subnetMasks;
            }
        }

        // No valid subnet mask found
        return new ArrayList<>();
    }

    // Finds the subnet mask with the least number of 1 bits from the given list.
    private static int findSubnetMaskWithLeastOnes(List<Integer> subnetMasks) {
        int minOnesSubnetMask = subnetMasks.get(0);
        int minOnesCount = Integer.bitCount(minOnesSubnetMask);

        for (int subnetMask : subnetMasks) {
            int onesCount = Integer.bitCount(subnetMask);
            if (onesCount < minOnesCount) {
                minOnesSubnetMask = subnetMask;
                minOnesCount = onesCount;
            }
        }

        return minOnesSubnetMask;
    }
}
