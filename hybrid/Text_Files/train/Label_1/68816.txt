def interesting_submissions(n, k, a):
    times = []
    for i in range(n):
        for j in range(a[i]):
            times.append((j + 1) * 100 / a[i])
    times.sort()
    processes = [0] * k
    interesting = 0
    finished = 0
    for time in times:
        min_process = min(processes)
        if min_process < time:
            finished += 1
            for i in range(k):
                if processes[i] == min_process:
                    if finished <= n:
                        processes[i] += a[finished - 1]
                    break
        if round(time) == round(100 * finished / n):
            interesting += 1
    return interesting - finished

n, k = map(int, input().split())
a = list(map(int, input().split()))
print(interesting_submissions(n, k, a))
