#include <bits/stdc++.h>

using namespace std;

string s;
int n, k, dp[105][105], a[105];

int f(int pos, int cnt) {
	if (cnt == k) return 0;
	if (pos == n) return -1e9;
	if (dp[pos][cnt] != -1) return dp[pos][cnt];
	int ret = max(f(pos + 1, cnt), f(pos + 1, cnt + 1) + a[cnt] - a[pos]);
	return dp[pos][cnt] = ret;
}

int main() {
	memset(dp, -1, sizeof dp);
	cin >> n >> k;
	cin >> s;
	for (int i = n - 1, j = 0; i >= 0; --i, ++j) {
		a[j] = a[j + 1];
		if (s[i] == 'a') a[j]++;
	}
	int p = f(0, 0);
	if (p <= 0) cout << -1 << endl;
	else cout << n * k - p << endl;
	return 0;
}
