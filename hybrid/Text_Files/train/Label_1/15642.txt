#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin >> t;
    while(t--){
        int n;
        cin >> n;
        string a, b;
        cin >> a >> b;
        int c = 0;
        for(int i = 0; i < n; ++i){
            if(a[i] != b[i]){
                ++c;
                if(c > 2) break;
            }
        }
        if(c <= 2 && c > 0) cout << "Yes\n";
        else cout << "No\n";
    }
    return 0;
}
