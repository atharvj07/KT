#include<bits/stdc++.h>
using namespace std;

#define mod 1000000007

vector<vector<vector<int>>> memo;
vector<int> fre;
vector<vector<pair<int, int>>> songs;

int solve(int mask, int idx, int time) {
    int &cur = memo[mask][idx][time];
    if(~cur)
        return cur;
    if(idx == songs.size()) {
        cur = (time == 0);
        return cur;
    }
    cur = 0;
    for(int i = 0; i < 3; i++) {
        if(mask & (1 << i))
            continue;
        if(fre[i] > 0)
            cur = (cur + solve(mask | (1 << i), idx + 1, time - songs[idx][i].first) % mod) % mod;
        cur = (cur + solve(mask, idx + 1, time - songs[idx][i].first) % mod) % mod;
    }
    return cur;
}

int main() {
    int n, t;
    cin >> n >> t;
    songs.resize(n);
    memo.resize(1 << 3, vector<vector<int>>(n + 1, vector<int>(t + 1, -1)));
    fre.resize(3, 0);
    for(int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        fre[b - 1]++;
        songs[i] = {{a, b}};
    }
    cout << solve(0, 0, t) << '\n';
    return 0;
}
