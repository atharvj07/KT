import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args){
    Scanner cin = new Scanner(System.in);
    long n = cin.nextLong();
    long r = 1;
    if(n % 2 == 0){
      n /= 2; r *= 2;
      while(n % 2 == 0)n /= 2;
    }
    for(long i = 3; i * i <= n; i += 2){
      if(n % i == 0){
        n /= i; r *= i - 1;
        while(n % i == 0){
          n /= i; r *= i;
        }
      }
    }
    if(n > 1){
      r *= n - 1;
    }
    System.out.println(r);
  }
}
