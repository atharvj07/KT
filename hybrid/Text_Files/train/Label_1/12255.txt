#include<bits/stdc++.h>
using namespace std;
const int N = 2e5+5;
int a[N];
int main() {
    int n, k; cin >> n >> k;
    string s; cin >> s;
    a[n+2] = 0;
    for(int i = n+1; i >= 1; --i) {
        a[i] = a[i+1];
        if(s[i-1] == s[i]) a[i] = i;
    }
    string t; cin >> t;
    int idx = 0;
    long long ans = 0;
    for(int i = 1; i <= n; ++i) {
        while(idx < t.size() && t[idx] != s[i]) ++idx;
        if(idx == t.size()) break;
        ++idx;
        ans += i - max(a[i], i-idx);
    }
    cout << ans << endl;
}
