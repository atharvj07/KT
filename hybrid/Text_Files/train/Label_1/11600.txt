#include <iostream>

typedef long long ll;

int main() {
    ll a, d, b;
    char c;
    std::cin >> a >> c >> d >> c >> b;

    a *= 10LL << b;
    ll gcd = 1LL, x = a, bx = d;
    while ((bx >>= 1)) {
        x >>= 1;
        if (bx & 1) gcd *= 2;
    }
    if(x%gcd==0) x/=gcd;
    std::cout << x << '\n';
    return 0;
}
