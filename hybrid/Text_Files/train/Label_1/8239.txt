#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int N = 1e5 + 5;
const ll oo = 1e18;

int n, m;
ll a[N];
ll b[N];

ll dp[N][2];

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    for (int i = 1; i <= m; i++) { cin >> a[i] >> b[i]; }
    for (int i = 1; i <= n; i++) {
      for (int prv = 0; prv <= 1; prv++) {
        dp[i][prv] = -oo;
      }
    }
    dp[0][0] = 0;
    for (int i = 1; i <= n; i++) {
      dp[i][0] = max(dp[i][0], dp[i - 1][0]);
      for (int j = 1; j <= m; j++) {
        dp[i][1] = max(dp[i][1], dp[i - 1][0] + a[j]);
        dp[i][1] = max(dp[i][1], dp[i - 1][1] + b[j]);
      }
    }
    cout << dp[n][1] << endl;
  }
  return 0;
}
