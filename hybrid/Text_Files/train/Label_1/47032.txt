import java.util.Scanner;

public class Lemmings {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] shields = new int[n];
        for (int i = 0; i < n; i++) {
            shields[i] = scanner.nextInt();
        }
        int[] protection = new int[(n * (n - 1)) / 2 + 1];
        for (int k = 0; k <= (n * (n - 1)) / 2; k++) {
            protection[k] = calculateProtection(shields, k);
        }
        for (int k = 0; k <= (n * (n - 1)) / 2; k++) {
            System.out.print(protection[k] + " ");
        }
    }

    private static int calculateProtection(int[] shields, int k) {
        int[] left = new int[shields.length];
        int[] right = new int[shields.length];
        for (int i = 1; i < shields.length; i++) {
            if (shields[i - 1] == 1) {
                left[i] = left[i - 1] + 1;
            } else {
                left[i] = left[i - 1];
            }
        }
        for (int i = shields.length - 2; i >= 0; i--) {
            if (shields[i + 1] == 1) {
                right[i] = right[i + 1] + 1;
            } else {
                right[i] = right[i + 1];
            }
        }
        int protection = 0;
        for (int i = 0; i < shields.length; i++) {
            if (shields[i] == 0) {
                protection += Math.min(left[i], right[i]);
            }
        }
        return Math.min(protection, k);
    }
}
