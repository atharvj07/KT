#include <iostream>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  int a[n];
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  while (m--) {
    int l, r;
    cin >> l >> r;
    --l, --r;  // 0-based indexing
    bool up = true, down = true;
    for (int i = l; i + 1 <= r; i++) {
      up &= a[i] <= a[i + 1];
      down &= a[i] >= a[i + 1];
    }
    cout << (up || down ? "Yes" : "No") << endl;
  }
  return 0;
}
