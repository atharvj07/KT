#include <bits/stdc++.h>

using namespace std;

const int MX = 2e5 + 10;
int freq[MX], f[MX];
int main() {
  int tt;
  cin >> tt;
  while (tt--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    int X = 0, cnt = 0;
    bool ok = false;
    for (int x : a) {
      if (!f[x]) f[x] = 1, cnt++;
      freq[x]++;
      if (freq[x] > 1) ok = true;
      if (freq[x] == cnt) X = x;
    }
    if (ok)
      cout << 1 << '\n';
    else
      cout << (cnt > 1 ? 2 : 0) << '\n';
    memset(freq, 0, sizeof freq);
    memset(f, 0, sizeof f);
  }
}
