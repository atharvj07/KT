#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 3e5 + 5;
const ll inf = 1e18;
int n, k;
ll a[maxn];

ll solve() {
  vector<ll> pre(n);
  pre[0] = a[0];
  for (int i = 1; i < n; i++) { pre[i] = pre[i - 1] + a[i]; }
  ll l = 1, r = k - 1, ans = -inf;
  while (r < n) {
    ll sum = pre[r] - (l > 1 ? pre[l - 2] : 0);
    ll cost = sum * k;
    ans = max(ans, cost);
    l++, r++;
  }
  return ans;
}
int main() {
  cin >> n >> k;
  for (int i = 0; i < n; i++) cin >> a[i];
  cout << solve() << endl;
  return 0;
}
