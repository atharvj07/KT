import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        long[] a = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextInt();
        }
        List<Improvements> improvements = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int tj = scanner.nextInt();
            int ij = scanner.nextInt();
            int bj = scanner.nextInt();
            improvements.add(new Improvements(tj, ij, bj));
        }
        Collections.sort(improvements, (o1, o2) -> {
            if (o1.tj == o2.tj && o1.ij == o2.ij) {
                return o1.bj - o2.bj;
            } else if (o1.tj == o2.tj) {
                return o1.ij - o2.ij;
            }
            return o1.tj - o2.tj;
        });
        List<Integer> result = new ArrayList<>();
        for (Improvements improvement : improvements) {
            if (m <= 0) {
                break;
            }
            if (improvement.tj == 1) {
                a[improvement.ij - 1] = improvement.bj;
                result.add(improvement.index + 1);
                m--;
            } else if (improvement.tj == 2) {
                a[improvement.ij - 1] += improvement.bj;
                result.add(improvement.index + 1);
                m--;
            } else {
                a[improvement.ij - 1] *= improvement.bj;
                result.add(improvement.index + 1);
                m--;
            }
        }
        System.out.println(result.size());
        for (Integer integer : result) {
            System.out.print(integer + " ");
        }
    }

    static class Improvements {
        int tj;
        int ij;
        int bj;
        int index;

        public Improvements(int tj, int ij, int bj) {
            this.tj = tj;
            this.ij = ij;
            this.bj = bj;
        }

        public int getTj() {
            return tj;
        }

        public void setTj(int tj) {
            this.tj = tj;
        }

        public int getIj() {
            return ij;
        }

        public void setIj(int ij) {
            this.ij = ij;
        }

        public int getBj() {
            return bj;
        }

        public void setBj(int bj) {
            this.bj = bj;
        }

        public int getIndex() {
            return index;
        }

        public void setIndex(int index) {
            this.index = index;
        }
    }
}
