#include<bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
long long a[MAXN];
int main() {
    int n, s, l;
    cin >> n >> s >> l;
    for(int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    int L = 0, R = 0, ans = n + 1;
    while(R < n) {
        if(a[R] - a[L] <= s) {
            ans = min(ans, R - L + 1);
            R++;
        }
        else L++;
    }
    if(ans > n) ans = -1;
    cout << ans << endl;
    return 0;
}
