import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Integer>[] adj = new List[n];
        for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
        while (scanner.hasNext()) {
            int u = scanner.nextInt(), k = scanner.nextInt();
            for (int i = 0; i < k; i++) {
                int v = scanner.nextInt(), c = scanner.nextInt();
                adj[u].add(v);
                adj[u].add(c);
            }
        }
        dijsktra(0, n, adj);
    }

    private static void dijsktra(int start, int n, List<Integer>[] adj) {
        int[] dist = new int[n];
        Arrays.fill(dist, Integer.MAX_VALUE);
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        dist[start] = 0;
        pq.add(start);
        while (!pq.isEmpty()) {
            int u = pq.poll();
            for (int i = 1; i < adj[u].size(); i += 2) {
                int v = adj[u].get(i - 1);
                int c = adj[u].get(i);
                if (dist[u] + c < dist[v]) {
                    dist[v] = dist[u] + c;
                    pq.add(v);
                }
            }
        }
        for (int i = 0; i < n; i++) System.out.println(i + " " + dist[i]);
    }
}
