#include <bits/stdc++.h>

const int mxN = 8;
int indeg[mxN] = {0};
int deg[mxN][mxN] = {0};
int ans;

inline void floyd_warshall(int n) {
  for (int k = 1; k <= n; ++k)
    for (int i = 1; i <= n; ++i)
      for (int j = 1; j <= n; ++j) { deg[i][j] = deg[i][j] || (deg[i][k] && deg[k][j]); }
}

void dfs_visit(int u, int id, int s, int d) {
  if (2 * s + d > ans)
    ans = 2 * s + d;
  for (int v = 1; v <= 7; ++v)
    if (u != v && deg[u][v] && indeg[v] == 0) {
      indeg[v] = 1;
      dfs_visit(v, ++id, s + (id == s + 1), d + (id == d));
      indeg[v] = 0;
    }
}

int main() {
  freopen("input.txt", "r", stdin);
  int n, m;
  std::cin >> n >> m;
  while (m--) {
    int u, v;
    std::cin >> u >> v;
    ++deg[u][v], ++deg[v][u];
  }
  floyd_warshall(n);
  ans = 0;
  dfs_visit(1, 1, 0, 0);
  std::cout << ans;
  return 0;
}
