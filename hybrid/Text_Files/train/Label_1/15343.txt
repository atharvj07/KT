#include <cstdio>
#include <cstring>
#define ll long long
using namespace std;

ll a[200],b[200];

ll gcd(ll a,ll b)
{
	if(b==0) return a;
	return gcd(b,a%b);
}

int main()
{
	char ca[201] = { 0 };

	int i,j,la,lb,k,ans;

	scanf("%s %s", ca, ca + 101);

	la = strlen(ca); lb = strlen(ca + 101);

	for(j=1; j<=la; ++j) a[la-j] = ca[j-1]-'0';
	for(j=1; j<=lb; ++j) b[lb-j] = ca[100+j-1]-'0';

	ans = 0;
	k = 1;
	for(i=199; i>=0; --i){
		if(!a[i] && !b[i])
			ans = gcd(ans,k);
		else if(a[i]==b[i])
			k = k*10%ans;
		else if(a[i]+1==b[i])
			ans = gcd(ans,k*10);
	}

	printf("%d\n", ans);

	return 0;
}
