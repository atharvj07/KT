from sys import stdin

n = int(stdin.readline())
di = [[-1, 0] for i in range(n)]
for i in range(n):
    a, b = map(int, stdin.readline().split())
    di[i] = [b, a]
di.sort(key = lambda x: x[1])
dp = [1 for i in range(n)]
for i in range(1, n):
    for j in range(i):
        if di[i][1] <= di[j][0]:
            dp[i] = max(dp[i], dp[j] + 1)
cur = 0
di.sort(reverse = True, key = lambda x: x[1])
for i in range(n):
    for j in range(i):
        if di[i][1] <= di[j][0]:
            cur = max(cur, dp[i] + dp[j])
mx = 0
ans = 0
for i in dp:
    if i == cur:
        ans += 1
    if i == mx: ans += 1
    else: mx = i

print(ans)
