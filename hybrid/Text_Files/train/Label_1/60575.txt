from sys import stdin, stderr

tc = int(stdin.readline())
for i in range(tc):
    n = int(stdin.readline())
    C,T = map(float, stdin.readline().split())
    EPS = 1e-3
    a = [[int(j) for j in stdin.readline().split()] for i in range(n)]
    a.sort(key=lambda x: x[1]/x[0], reverse=True)

    def solve(t):
        s, T_now = 1.0, t
        ret = 0
        for x, p in a:
            if s*x > T_now:
                break
            T_now -= s*x
            s *= 0.9;
            ret += p
        return ret

    l, r, mx = 0, T, 0
    while r-l>EPS:
        m = (l+r)/2.0
        tmp = solve(m)
        if tmp>mx:
            mx = tmp
            l = m
        else:
            r = m
    print(mx)
