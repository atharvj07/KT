import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), d = sc.nextInt();
        if(n > k * d) {
            System.out.println(-1);
            return;
        }
        int[][] a = new int[d][n];
        int idx = 0;
        for(int i = 0; i < d; ++i) {
            for(int j = 0; j < n; ++j) {
                idx = idx % k + 1;
                a[i][j] = idx;
            }
        }
        for(int i = 0; i < d; ++i) {
            for(int j = 0; j < n; ++j) {
                System.out.print(a[i][j] + " ");
            }
            System.out.println();
        }
    }
}
