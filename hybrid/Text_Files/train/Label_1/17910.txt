#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int fmt;
    string time;
    cin >> fmt >> time;
    int h = (time[0]-'0')*10+(time[1]-'0');
    int m = (time[3]-'0')*10+(time[4]-'0');
    if (fmt == 12) {
        if (h == 0) h = 12;
        else if (h > 12) h -= 12;
    } else if (h > 23) h-=24;
    h = min(h, 23);
    cout << (h/10 > 0?to_string(h):"0"+to_string(h)) << ":" << (m/10 > 0?to_string(m):"0"+to_string(m));
    return 0;
}
