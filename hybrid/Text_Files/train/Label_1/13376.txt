#include <bits/stdc++.h>
using namespace std;
const int MaxN = 2e5 + 10;
int p[MaxN], u[MaxN];
int main() {
  int n;
  cin >> n;
  int a[n], b[n];
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  for (int i = 0; i < n; i++) { p[a[i]] = i; }
  for (int i = 0; i < n; i++) {
    if (i > 0) { cout << ' '; }
    for (int j = b[i]; j >= 1; j--) {
      if (!u[a[j - 1]]) {
        cout << p[a[j - 1]] + 1;
        u[a[j - 1]] = true;
        break;
      }
    }
  }
  cout << '\n';
  return 0;
}
