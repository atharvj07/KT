import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = scanner.nextInt();
        }
        int[] sorted = x.clone();
        Arrays.sort(sorted);
        int s = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (x[i] == sorted[j]) {
                    s += Math.min(i, n - 1 - i);
                    break;
                }
            }
        }
        System.out.println(s);
    }
}
