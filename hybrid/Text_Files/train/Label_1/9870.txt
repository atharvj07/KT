#include <bits/stdc++.h>
using namespace std;

const int N = 140582;

vector<long long> a;
bool f;

long long gcd(long long x, long long y) {
  while (y) {
    long long tmp = x % y;
    x = y;
    y = tmp;
  }
  return x;
}

void solve() {
  int n;
  cin >> n;
  vector<long long> b(n);
  for (auto& x : b) cin >> x;
  vector<int> cnt(N);
  for (int i = 0; i < n; i++) {
    cnt[b[i]]++;
    if (cnt[b[i]] >= 2) f = 1;
  }
  long long lcm = 0;
  for (int i = 0; i < N; i++) {
    if (cnt[i] == 0) continue;
    if (lcm == 0)
      lcm = i;
    else
      lcm = gcd(lcm, 1LL * i);
  }
  if (f == 0) {
    for (int i = 0; i < n; i++) a.push_back(lcm);
    cout << "YES" << endl;
    for (int i = 0; i < n; i++) cout << a[i] << " ";
    cout << endl;
  } else
    cout << "NO" << endl;
}

int main() {
  f = 0;
  solve();
  return 0;
}
