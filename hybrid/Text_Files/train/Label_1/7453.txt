#include <bits/stdc++.h>

using namespace std;

typedef uint64_t ll;

int main() {
    int m, d;
    cin >> m >> d;
    string a, b;
    cin >> a >> b;
    int na = a.size(), nb = b.size();
    int ans = 0;
    function<void(int, int, ll)> dfs = [&](int pos, int lead, ll val) {
        if (pos == na) {
            if ((val % m == 0) && (d == 0 || (pos % 2) + 1 == d)) {
                ans++;
                ans %= 1000000007;
            }
            return;
        }
        int min_dig = 0, max_dig = b[pos] - '0';
        if (pos == 0) max_dig = 9;
        if (pos == 0 || lead) min_dig = a[pos] - '0';
        for (int dig = min_dig; dig <= max_dig; dig++) {
            int new_lead = lead;
            if (dig == d) new_lead ^= 1;
            dfs(pos + 1, new_lead, val * 10 + dig);
        }
    };
    dfs(0, 0, 0);
    cout << ans << '\n';
    return 0;
}
