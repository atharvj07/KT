#include <iostream>
#include <vector>
#include <algorithm>

struct Segment {
  int l, r;
  Segment(int ll, int rr) {
    l = ll;
    r = rr;
  }
  bool operator<(const Segment &o) const { return l < o.l; }
};

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<Segment> x, y;
  for (int i = 0; i < n; ++i) {
    int x1, y1, x2, y2;
    std::cin >> x1 >> y1 >> x2 >> y2;
    x.push_back(Segment(x1, x2));
    y.push_back(Segment(y1, y2));
  }
  sort(x.begin(), x.end());
  sort(y.begin(), y.end());
  long long ans = 1e18;
  int j = 0;
  for (int i = 0; i + k < n; ++i) {
    while (j + 1 < n && x[i + k].r > x[j + 1].l) { ++j; }
    if (i + k == n) {
      ans = std::min(ans, 1ll * (x[i].r - x[i].l) * (y[j].r - y[j].l));
      break;
    }
    ans = std::min(ans, 1ll * (x[i + k].r - x[i].l) * (y[j].r - y[j].l));
    if (x[i + k].r >= x[j].r) { ++j; }
  }
  std::cout << ans << '\n';
}
