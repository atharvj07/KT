#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pi;

const ll MOD = 998244353;
const int MAXN = 200005;
int n, k, p[MAXN], segMax[MAXN];

struct DSU {
    int par[MAXN], sz[MAXN];
    void init(int n) {
        for (int i = 1; i <= n; ++i) par[i] = i, sz[i] = 1;
    }
    int find(int x) {
        while (x != par[x]) x = par[x] = par[par[x]];
        return x;
    }
    int mrg(int a, int b) {
        a = find(a), b = find(b);
        if (a == b) return a;
        if (sz[a] < sz[b]) swap(a, b);
        sz[a] += sz[b], par[b] = a;
        return a;
    }
} dsu;

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) cin >> p[i];
    dsu.init(n + 1);
    for (int i = n; i >= 1; --i) {
        segMax[dsu.find(i)] = max(segMax[dsu.find(i)], p[i]);
        if (i > 1) dsu.mrg(i, i - 1);
    }
    ll mx = 0, cnt = 1;
    for (int i = 1; i <= k; ++i) mx += segMax[i];
    for (int i = k + 1; i <= n; ++i) {
        if (segMax[dsu.find(i)] == segMax[dsu.find(k)]) cnt = (cnt * dsu.sz[dsu.find(i)]) % MOD;
    }
    cout << mx << " " << cnt << "\n";
    return 0;
}
