#include <iostream>
#include <cmath>
#include <cstdio>

int main() {
    double a, b;
    std::string s;
    std::cin >> a >> s >> b;
    std::cout << std::fixed << std::showpoint;
    if (static_cast<int>(floor(b)) == b)
        std::cout << static_cast<long long>(a * pow(10, static_cast<int>(b))) << '\n';
    else
        std::cout << a * pow(10, static_cast<int>(b)) << '\n';
    return 0;
}
