import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int m = scanner.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
            }
            int[] x = new int[m];
            for (int j = 0; j < m; j++) {
                x[j] = scanner.nextInt();
            }
            for (int j = 0; j < m; j++) {
                long sum = 0;
                boolean found = false;
                for (int k = 0; k < n; k++) {
                    sum += a[k];
                    if (sum >= x[j]) {
                        found = true;
                        System.out.print(k + 1 + " ");
                        break;
                    }
                }
                if (!found) {
                    System.out.print(-1 + " ");
                }
            }
            System.out.println();
        }
    }
}
