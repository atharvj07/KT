import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Integer, Set<Integer>> graph = new HashMap<>();
        for (int i = 0; i < n - 1; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            graph.computeIfAbsent(x, k -> new HashSet<>()).add(y);
            graph.computeIfAbsent(y, k -> new HashSet<>()).add(x);
        }
        List<Integer> order = new ArrayList<>();
        Set<Integer> visited = new HashSet<>();
        for (int i = 1; i <= n; i++) {
            if (!visited.contains(i)) {
                dfs(i, -1, graph, order, visited);
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(order.get(i) + " ");
        }
    }

    private static void dfs(int u, int p, Map<Integer, Set<Integer>> graph, List<Integer> order, Set<Integer> visited) {
        order.add(u);
        visited.add(u);
        for (int v : graph.getOrDefault(u, new HashSet<>())) {
            if (v != p) {
                dfs(v, u, graph, order, visited);
            }
        }
    }
}
