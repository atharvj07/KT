r, c, n, k = map(int, input().split())
l = []
for i in range(n):
    l.append(tuple(map(int, input().split())))
ans = 0

for i in range(1 << n):
    x = 0
    y = []
    for t in range(n):
        if i >> t & 1:
            x += 1
            y.append(l[t][1])
    if x < k:
        continue
    y.sort()
    flag = True
    for t in range(1, len(y)):
        if y[t] - y[t - 1] > 1:
            flag = False
            break
    if flag:
        ans += (1 << (r - x)) - 1

print(ans)
