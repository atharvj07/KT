from collections import Counter
for t in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    c = Counter(a)
    for x in a:
        need = x
        for y in a:
            if y != x and c[x % y] == 0:
                need = min(need, x % y)
        for y in a:
            if y != x and x % y == need:
                print(x, y)
