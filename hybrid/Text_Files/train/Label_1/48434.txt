import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int count = 0;
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < n; j++) {
                int xor = a[i] ^ a[j];
                int cnt = 0;
                for (int l = 0; l < 31; l++) {
                    if ((xor & (1 << l)) != 0) {
                        cnt++;
                    }
                }
                if (cnt == k) {
                    count++;
                }
            }
        }
        System.out.println(count);
    }
}
