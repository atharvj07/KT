#include<bits/stdc++.h>
using namespace std;
int n,m,ans,vis[1005],dep[1005],par[1005],sz[1005];
vector<int>g[1005];
void dfs(int u){
    vis[u]=1;
    for(int i=0;i<g[u].size();i++){
        int v=g[u][i];
        if(!vis[v]){
            dep[v]=dep[u]+1;
            par[v]=u;
            dfs(v);
        }
    }
}
int main(){
    cin>>n>>m;
    int mx=0,u=0,v=0;
    for(int i=1;i<=m;i++){
        int x,y;
        cin>>x>>y;
        g[x].push_back(y);
        g[y].push_back(x);
    }
    dfs(1);
    for(int i=1;i<=n;i++){
        if(dep[i]>mx){
            mx=dep[i];
            v=i;
        }
    }
    dep[v]=0;
    par[v]=0;
    dfs(v);
    for(int i=1;i<=n;i++){
        if(dep[i]>mx){
            mx=dep[i];
            u=i;
        }
    }
    int a=u,b=v;
    while(a!=b){
        sz[a]++;
        a=par[a];
    }
    sz[a]++;
    for(int i=1;i<=n;i++){
        if(i!=u && i!=v){
            if(sz[i]%2){
                cout<<u<<' '<<i<<endl;
            }
        }
    }
    ans=(mx+1)/2;
    cout<<ans<<endl;
    return 0;
}
