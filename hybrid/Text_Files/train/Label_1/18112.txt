#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> beauty(n);
  for (int i = 0; i < n; i++) cin >> beauty[i];
  string color;
  cin >> color;
  vector<vector<long long>> dp(n + 1, vector<long long>(3, LLONG_MIN));
  dp[0][0] = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < k; j++) {
      for (int l = 0; l < 3; l++) {
        dp[i + 1][l] = max(dp[i + 1][l], dp[i][l]);
        if (color[i] != 'W' || l != 0)
          dp[i + 1][l] = max(dp[i + 1][l], dp[i][l] + beauty[i]);
        if (color[i] != 'O' || l != 1)
          dp[i + 1][l] = max(dp[i + 1][l], dp[i][l] + beauty[i]);
        if (color[i] != 'R' || l != 2)
          dp[i + 1][l] = max(dp[i + 1][l], dp[i][l] + beauty[i]);
      }
    }
  }
  long long ans = LLONG_MIN;
  for (int i = 0; i < 3; i++) ans = max(ans, dp[n][i]);
  if (ans == LLONG_MIN) cout << -1 << '\n';
  else cout << ans << '\n';
  return 0;
}
