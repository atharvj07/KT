#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  string s, t;
  cin >> s >> t;

  bool hasStar = false;
  int i = 0, j = 0;
  while (i < n && j < m) {
    if (s[i] == '*') {
      hasStar = true;
      i++;
      continue;
    }
    if (s[i] != t[j]) {
      if (hasStar) {
        i++;
        continue;
      } else {
        cout << "NO" << endl;
        return 0;
      }
    }
    i++;
    j++;
  }
  while (i < n) {
    if (s[i] == '*') i++;
    else {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
