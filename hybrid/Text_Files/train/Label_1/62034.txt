import sys

n, m, k = map(int, input().split())
c = list(map(int, input().split()))
p = [list(map(int, input().split())) for _ in range(n)]
dp = [[[-1] * (k + 1) for _ in range(m + 1)] for _ in range(n + 1)]

def dfs(i, cnt, cur):
    if i == n:
        if cnt == k: return 0
        return -1
    if dp[i][cnt][cur] != -1: return dp[i][cnt][cur]
    dp[i][cnt][cur] = -1
    if c[i] != 0:
        if cur == c[i]:
            res = dfs(i + 1, cnt, c[i])
            if res != -1: dp[i][cnt][cur] = res
        return dp[i][cnt][cur]
    for j in range(1, m + 1):
        if j == cur:
            res = dfs(i + 1, cnt, j)
            if res != -1: dp[i][cnt][cur] = min(dp[i][cnt][cur], res)
        else:
            res = dfs(i + 1, cnt + 1, j)
            if res != -1: dp[i][cnt][cur] = min(dp[i][cnt][cur], p[i][j - 1] + res)
    return dp[i][cnt][cur]

res = dfs(0, 0, 0)
sys.stdout.write('-1\n' if res == -1 else str(res) + '\n')
