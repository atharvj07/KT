#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 1000, mod = 998244353;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s[maxn];
        map<string, int> mp;
        for (int i = 0; i < n; i++) {
            cin >> s[i];
            mp[s[i]]++;
        }
        if (mp.size() == 1)
            cout << "YES" << "\n";
        else
            cout << "NO" << "\n";
    }
}
