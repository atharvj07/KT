#include <iostream>
#include <vector>
#include <string>

using namespace std;

typedef long long i64;

int main() {
    i64 n;
    // Input n
    cin >> n;
    vector<string> s(n);
    // Input n strings
    for (i64 i = 0; i < n; i++) {
        cin >> s[i];
    }
    // Compute the concatenation of all strings
    string concat = "";
    for (i64 i = 0; i < n; i++) {
        concat += s[i];
    }
    // Compute the beauty of the concatenated string
    i64 beauty = 0;
    i64 count = 1;
    for (i64 i = 1; i < concat.length(); i++) {
        if (concat[i] == concat[i - 1]) {
            count++;
        } else {
            beauty = max(beauty, count);
            count = 1;
        }
    }
    beauty = max(beauty, count);
    // Output the beauty
    cout << beauty << endl;
    return 0;
}
