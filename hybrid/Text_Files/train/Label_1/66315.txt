import sys

input = sys.stdin.readline
n, m = map(int, input().split())
candy = [(-1, -1)] * (m + 1)
for i in range(m):
    x, y = map(int, input().split())
    candy[i + 1] = (x, y)

mn = [n**2] * (n + 1)
mn[1] = 0
for i in range(1, m + 1):
    mn[candy[i][1]] = min(mn[candy[i][1]], mn[candy[i][0]] + 1)

    for j in range(1, n + 1):
        if mn[j] != n**2:
            mn[j] += 1

    for j in range(candy[i][0] + 1, n + 1):
        if mn[j] != n**2:
            mn[j] += 1

    mn[1] = min(mn[1], mn[n] + 1)

print(*mn[1:])
