import java.util.*;

class Main {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt();
        int[] coins = new int[n];
        for (int i = 0; i < n; i++) coins[i] = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int num = sc.nextInt(), ans = 0, j = 0;
            if (num == 0) ans = 1;
            while (num > 0) {
                if (num % 2 != 0) {
                    if (j < n && coins[j] <= num) {
                        ans++;
                        num -= coins[j];
                    } else {
                        ans = -1;
                        break;
                    }
                } else {
                    if (j < n && coins[j] <= num / 2) j++;
                    else {
                        ans = -1;
                        break;
                    }
                }
                num /= 2;
            }
            System.out.println(ans);
        }
    }
}
