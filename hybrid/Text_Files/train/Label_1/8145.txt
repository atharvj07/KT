#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n, m, k, a[200005], l = 1, r = 1, ans = 2e9;
    cin >> n >> m >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    while (r <= n) {
        if (a[r] - a[l - 1] <= m - 1) {
            ans = min(ans, n - r + 1);
            r++;
        } else l++;
    }
    cout << ans;
}
