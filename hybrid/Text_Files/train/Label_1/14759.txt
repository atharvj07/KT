#include<bits/stdc++.h>
using namespace std;
#define int long long
main(){
    int n; cin>>n;
    vector<int> a(n);
    for(int &x : a) cin>>x;
    sort(a.begin(),a.end());
    if(n==1){
        cout<<-1<<'\n';
        return 0;
    }
    int d=a[1]-a[0];
    int cnt=0;
    for(int i=2;i<n;++i) if(a[i]-a[i-1]!=d) ++cnt;
    cout<<(cnt==1?a[cnt]+d:0)<<'\n';
    if(cnt==1){
        for(int i=-11212121212ll;i<11212121212ll;++i){
            if(i==a[cnt]+d) continue;
            vector<int> b=a;
            b.push_back(i);
            sort(b.begin(),b.end());
            bool flag=true;
            for(int i=1;i<n;++i){
                if(b[i]-b[i-1]!=d) flag=false;
            }
            if(flag) cout<<i<<' ';
        }
    }
}
