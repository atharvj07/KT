def find_permutation(n, segments):
    adjacency_list = {i: set() for i in range(1, n + 1)}
    for segment in segments:
        for i in range(len(segment) - 1):
            adjacency_list[segment[i]].add(segment[i + 1])
            adjacency_list[segment[i + 1]].add(segment[i])

    # Find the starting point, which should only have one connection
    start = next(v for v, adj in adjacency_list.items() if len(adj) == 1)

    # Reconstruct the permutation
    permutation = [start]
    while len(permutation) < n:
        last = permutation[-1]
        next_val = (adjacency_list[last] - set(permutation)).pop()
        permutation.append(next_val)
    
    return permutation

def solve(t, test_cases):
    result = []
    for i in range(t):
        n = test_cases[i][0]
        segments = test_cases[i][1]
        result.append(find_permutation(n, segments))
    return result

# Parse input
t = int(input().strip())
test_cases = []
for _ in range(t):
    n = int(input().strip())
    segments = [list(map(int, input().strip().split()))[1:] for _ in range(n-1)]
    test_cases.append((n, segments))

# Solve and output
for case_result in solve(t, test_cases):
    print(' '.join(map(str, case_result)))
