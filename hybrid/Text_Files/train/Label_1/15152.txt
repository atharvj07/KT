#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1000;
char a[MAXN + 5];
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> a + 1;
    int n = strlen(a + 1);
    bool found = false;
    for (int i = 1; i < n; i++) {
      if (a[i] != a[i + 1]) {
        found = true;
        break;
      }
    }
    if (!found)
      cout << -1 << endl;
    else {
      sort(a + 1, a + n + 1);
      cout << a + 1 << endl;
    }
  }
  return 0;
}
