import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[][] arr = new int[n][m];
        int[] best = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = in.nextInt();
            }
        }
        for (int i = 0; i < n; i++) {
            int[] dp = new int[m + 1];
            for (int j = 0; j < m; j++) {
                for (int k = 0; k <= j; k++) {
                    if ((j - k) % 2 == 0 || dp[k] <= dp[k + 1]) {
                        dp[k + 1] = Math.max(dp[k + 1], dp[k] + arr[i][j]);
                    }
                }
            }
            best[i] = dp[m];
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            ans += best[i];
        }
        System.out.println(ans);
    }
}
