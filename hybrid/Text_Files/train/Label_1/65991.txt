import sys

n = int(input())
ls = []
for _ in range(n):
  a, b, c, d = map(int, input().split())
  ls += [(a, b, c, d), (c, d, a, b)]

ls.sort()
cur = []
for i in range(len(ls)):
  lst, rst = ls[i][1:3]
  if i == 0:
    cur += [ls[i]]
  elif cur[-1][1] <= lst:
    cur += [ls[i]]
  elif cur[-1][1] > lst and cur[-1][3] <= rst:
    cur[-1] = (ls[i][0], ls[i][3], ls[i][0], ls[i][3])
  else:
    print("NO")
    sys.exit()

print("YES")
