#include <bits/stdc++.h>
using namespace std;
const int MAXN = 5e5 + 5;
int a[MAXN], b[MAXN], need[MAXN];
int main() {
  int m, k, n, s;
  cin >> m >> k >> n >> s;
  for (int i = 0; i < m; i++) cin >> a[i];
  for (int i = 0; i < s; i++) cin >> b[i], need[b[i]]++;
  int j = 0;
  for (int i = 0; i < m; i++) {
    while (j < s && need[a[i]]) {
      need[a[i]]--;
      j++;
    }
    if (j == s) {
      cout << m - i << endl;
      for (; i < m; i++) cout << i + 1 << " ";
      return 0;
    }
  }
  cout << "-1" << endl;
  return 0;
}
