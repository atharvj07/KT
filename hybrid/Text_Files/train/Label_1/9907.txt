#include <cstdio>
#include <cstring>
#include <vector>
using namespace std;

vector<int>adj[1005];
int visit[1005],cnt;

void dfs(int s){
    if(visit[s])return;
    visit[s]=1;
    cnt++;
    for(int i=0;i<adj[s].size();i++){
        dfs(adj[s][i]);
    }
}

int main(){
    int n,m,x,y;
    scanf("%d%d",&n,&m);
    for(int i=1;i<=m;i++){
        scanf("%d%d",&x,&y);
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    int flag=0,com=0;
    for(int i=1;i<=n;){
        if(!visit[i]){
            cnt=0;
            dfs(i);
            if(cnt==n){
                com=1;
            }
            else flag=1;
            i=i+cnt;
        }
        else i++;
    }
    if(!flag && com)printf("yes\n");
    else printf("no\n");
    return 0;
}
