import java.util.*;
import java.lang.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int A = sc.nextInt();
        int B = sc.nextInt();
        if (Math.min(A + 1, B + 1) == Math.max(A + 1, B + 1)) {
            System.out.println("Impossible");
            return;
        }
        System.out.println("Possible");
    }
}
