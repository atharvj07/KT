#include<bits/stdc++.h>
using namespace std;
int main() {
  int t; cin >> t;
  while(t--) {
    int n, l, r, k; cin >> n >> l >> r >> k;
    vector<int> p(n);
    for(int i = 0; i < n; i++) { cin >> p[i]; }
    sort(p.begin(), p.end());
    int cnt = 0;
    for(int i = 0; i < n; i++) {
      if(p[i] < l || p[i] > r) {
        continue;
      }
      else {
        k -= p[i];
        cnt++;
        if(k < 0) {
          cnt--;
          break;
        }
      }
    }
    cout << cnt << endl;
  }
  return 0;
}
