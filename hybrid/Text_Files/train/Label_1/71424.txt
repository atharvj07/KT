def max_weapons(t, test_cases):
    results = []
    for i in range(t):
        p, f = test_cases[i][0]
        cnt_s, cnt_w = test_cases[i][1]
        s, w = test_cases[i][2]
        
        if s > w:
            s, w = w, s
            cnt_s, cnt_w = cnt_w, cnt_s
        
        max_weapons = 0
        for swords_taken in range(cnt_s + 1):
            if swords_taken * s <= p:
                remaining_p = p - swords_taken * s
                war_axes_taken_p = min(cnt_w, remaining_p // w)
                
                remaining_s = cnt_s - swords_taken
                remaining_w = cnt_w - war_axes_taken_p
                
                swords_taken_f = min(remaining_s, f // s)
                remaining_f = f - swords_taken_f * s
                war_axes_taken_f = min(remaining_w, remaining_f // w)
                
                max_weapons = max(max_weapons, swords_taken + war_axes_taken_p + swords_taken_f + war_axes_taken_f)
        results.append(max_weapons)
    return results

# Reading number of test cases
t = int(input().strip())

# List to hold all test cases
test_cases = []

for _ in range(t):
    # Reading capacities
    p, f = map(int, input().strip().split())
    # Reading counts of swords and war axes
    cnt_s, cnt_w = map(int, input().strip().split())
    # Reading weights of swords and war axes
    s, w = map(int, input().strip().split())
    test_cases.append([(p, f), (cnt_s, cnt_w), (s, w)])

# Calculating and printing results
results = max_weapons(t, test_cases)
for result in results:
    print(result)
