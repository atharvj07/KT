m, s = map(int, input().split())
if s > m * 9 or (s == 0 and m > 1):
    print(-1, -1)
else:
    smallest = '1' + '0' * (m - 1)
    largest = '9' * m
    sm = s - int(smallest)
    lm = s - int(largest)
    i = -1
    while sm > 0:
        if smallest[i] != '9':
            smallest = smallest[:i] + str(int(smallest[i]) + 1) + smallest[i + 1:]
            sm -= 1
            i = -1
        i -= 1
    i = -1
    while lm > 0:
        if largest[i] != '0':
            largest = largest[:i] + str(int(largest[i]) - 1) + largest[i + 1:]
            lm -= 1
            i = -1
        i -= 1
    print(smallest, largest)
