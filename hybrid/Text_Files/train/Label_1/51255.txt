import java.util.*;

class Main {
    static int n, m;
    static int[][] a, b;
    static int[][] ops;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        a = new int[n][m];
        b = new int[n][m];
        ops = new int[n * m][3];

        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                a[i][j] = sc.nextInt();
        int cnt = 0;
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++) {
                if (a[i][j] == 1 && b[i][j] == 0) {
                    ops[cnt][0] = i;
                    ops[cnt][1] = j;
                    ops[cnt++][2] = 1;
                    b[i][j] = b[i][j + 1] = b[i + 1][j] = b[i + 1][j + 1] = 1;
                }
            }
        if (Arrays.deepEquals(a, b)) {
            System.out.println(cnt);
            for (int i = 0; i < cnt; i++)
                System.out.println(ops[i][0] + " " + ops[i][1]);
        } else
            System.out.println(-1);
    }
}
