#include <bits/stdc++.h>
using namespace std;
#define N 5005

int32_t main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  vector<int> a[N];
  map<int, int> w[N];
  int32_t n, m, p[N], c[N], d, k[N];
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> p[i];
  }
  for (int i = 1; i <= n; i++) {
    cin >> c[i];
    w[c[i]][p[i]]++;
    a[c[i]].push_back(p[i]);
  }
  cin >> d;
  for (int i = 1; i <= d; i++) {
    cin >> k[i];
  }
  while (d--) {
    map<int, int> mp;
    for (auto v : a[c[k[d + 1]]]) { mp[v]++; }
    for (auto v : w[c[k[d + 1]]]) {
      mp[v.first] -= v.second;
    }
    while (mp.size() && !mp.begin()->second) { mp.erase(mp.begin()); }
    cout << (mp.empty() ? 0 : mp.begin()->first) << "\n";
    w[c[k[d + 1]]][p[k[d + 1]]]--;
    for (auto &v : a[c[k[d + 1]]]) {
      if (v == p[k[d + 1]]) continue;
      w[c[k[d + 1]]][v]++;
    }
    a[c[k[d + 1]]].erase(remove(a[c[k[d + 1]]].begin(), a[c[k[d + 1]]].end(), p[k[d + 1]]), a[c[k[d + 1]]].end());
  }
}
