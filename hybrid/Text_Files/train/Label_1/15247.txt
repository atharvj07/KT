#include <algorithm>
#include <cstdio>
#include <cstring>
using namespace std;

#define re register

template <typename T>
inline void read(T &x) {
  x = 0;
  char s = (char)getchar();
  bool f = false;
  while (!(s >= '0' && s <= '9')) {
    if (s == '-') {
      f = true;
    }
    s = (char)getchar();
  }
  while (s >= '0' && s <= '9') {
    x = (x << 1) + (x << 3) + s - '0';
    s = (char)getchar();
  }
  if (f) {
    x = (~x) + 1;
  }
}

const int N = 2e5 + 5;

struct node {
  int val, pre;
} t[N];
int rt[N];

inline void insert(int &x, int pre, int val) {
  x = ++*rt;
  t[x].pre = pre;
  t[x].val = val;
}

inline int query(int a, int b, int val) {
  if (a == b) {
    return t[a].val;
  }
  int mid = (a + b) >> 1;
  if (t[a].val < val) {
    return query(mid + 1, b, val - t[a].val) + t[a].pre;
  } else {
    return query(a, mid, val);
  }
}

int main() {
  int n;
  read(n);
  for (re int i = 1; i <= n; i++) {
    int x;
    read(x);
    insert(rt[i], rt[i - 1], x);
  }
  for (re int i = 1; i <= n; i++) {
    int ans = query(1, *rt, i);
    printf("%d ", ans);
  }
  return 0;
}
