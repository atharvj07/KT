import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int x = input.nextInt();
        int m = input.nextInt();
        int y = input.nextInt();
        if (n != m) {
            System.out.println((long) (n * 4 + m * 4));
        } else {
            if (x == y)
                System.out.println((long) (n * 4 - 4));
            else
                System.out.println((long) (n * 4 + m * 4 - 4));
        }
    }
}
