import java.util.Scanner;
import java.util.Set;
import java.util.HashSet;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int greatestCommonDivisor = 0;
        Set<Integer> set = new HashSet<>();

        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            set.add(a - b);
        }

        for (int value : set) {
            greatestCommonDivisor = gcd(greatestCommonDivisor, value);
        }

        System.out.println((greatestCommonDivisor > 1) ? "Yes" : "No");
    }

    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
