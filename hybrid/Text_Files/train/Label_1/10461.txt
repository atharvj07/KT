#include <bits/stdc++.h>
#define dbg(x) cout << #x << " = " << x << endl
#define ll long long
#define ii pair<int, int>
#define pb push_back
#define vi vector<int>
using namespace std;

struct DSU {
    vi p;
    DSU(int n) {
        p.resize(n + 1);
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) { return x == p[x] ? x : (p[x] = find(p[x])); }
    void unite(int a, int b) { p[find(b)] = find(a); }
};

int main() {
    int n, m, s, t;
    cin >> n >> m >> s >> t;
    DSU dsu(n);
    for (int i = 1; i <= m; ++i) {
        int u, v;
        cin >> u >> v;
        dsu.unite(u, v);
    }
    set<int> ss, st;
    for (int i = 1; i <= n; ++i) {
        ss.insert(dsu.find(s));
        st.insert(dsu.find(t));
    }
    cout << ss.size() * st.size() - 1 << endl;
    return 0;
}
