#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define vll vector<ll>
#define all(x) x.begin(), x.end()
const ll md = 1e9 + 7;

int main() {
    ll n;
    cin >> n;
    vll s(n), d(n);
    for (auto &x : s) cin >> x;
    for (auto &x : d) cin >> x;
    ll cur = 1e9;
    vll res(n);
    for (ll i = 0; i < n; i++) {
        if (cur <= s[i]) {
            res[i] = s[i];
            cur = s[i] + d[i];
        } else {
            res[i] = cur;
            cur += d[i];
        }
        cout << res[i] << " " << res[i] + d[i] - 1 << endl;
    }
    return 0;
}
