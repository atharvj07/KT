import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.Scanner;

class Pair implements Comparable<Pair> {
    int weight;
    int cute;

    public Pair(int weight, int cute) {
        this.weight = weight;
        this.cute = cute;
    }

    @Override
    public int compareTo(Pair o) {
        return this.weight - o.weight;
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int na = scanner.nextInt();
        int nb = scanner.nextInt();
        int w = scanner.nextInt();

        ArrayList<Pair> maleCats = new ArrayList<>();
        ArrayList<Pair> femaleCats = new ArrayList<>();

        for (int i = 0; i < na; i++) {
            int weight = scanner.nextInt();
            int cute = scanner.nextInt();

            maleCats.add(new Pair(weight, cute));
        }

        for (int i = 0; i < nb; i++) {
            int weight = scanner.nextInt();
            int cute = scanner.nextInt();

            femaleCats.add(new Pair(weight, cute));
        }

        Collections.sort(maleCats);
        Collections.sort(femaleCats);

        int minUncute = Integer.MAX_VALUE;
        int minWeight = Integer.MAX_VALUE;

        for (int i = 0; i < maleCats.size(); i++) {
            for (int j = 0; j < femaleCats.size(); j++) {
                int weight = maleCats.get(i).weight + femaleCats.get(j).weight;

                if (weight > w) {
                    continue;
                }

                int cuteMax = Math.max(maleCats.get(i).cute, femaleCats.get(j).cute);
                int cuteMin = Math.min(maleCats.get(i).cute, femaleCats.get(j).cute);

                int uncute = Math.max(weight - w, cuteMax - cuteMin);

                if (uncute < minUncute) {
                    minUncute = uncute;
                    minWeight = weight;
                } else if (uncute == minUncute) {
                    minWeight = Math.min(minWeight, weight);
                }
            }
        }

        System.out.println(minUncute);
    }
}
