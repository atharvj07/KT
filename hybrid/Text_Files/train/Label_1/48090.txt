import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        sc.close();
        int w = 10 - n % 10 % k;
        if (n % 10 == 0) w = 10;
        if (w == 10) w = 0;
        System.out.println(w);
    }
}
