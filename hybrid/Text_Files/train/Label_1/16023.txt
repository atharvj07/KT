#include<bits/stdc++.h>
using namespace std;
int a[30],b[30],ans[30];
int main()
{
    int n,m,i,j,k,x,vis1[10]={0},vis2[10]={0};
    cin>>n>>m;
    for(i=0;i<n;i++)
    {
        cin>>a[i];
        vis1[a[i]]=1;
    }
    for(i=0;i<m;i++){
        cin>>b[i];
        vis2[b[i]]=1;
    }
    for(i=1;;i++){
        x=i;k=0;
        while(x>0){
            ans[++k]=x%10;
            x/=10;
        }
        j=1;
        for(x=k;x>=1;x--){
            if(vis1[ans[x]] && vis2[ans[j]])j++;
        }
        if(j>1){
            x=0;
            for(j=k;j>=1;j--){
                x*=10;
                x+=ans[j];
            }
            cout<<x<<endl;
            return 0;
        }
    }
}
