def can_place_weights(available_weights, m):
    for i in range(1, 11):
        if available_weights[i-1] == '1':
            if place_weight(i, i, 1, m, available_weights, [i]):
                return True
    return False

def place_weight(current_weight, diff, count, m, available_weights, sequence):
    if count == m:
        print("YES")
        print(" ".join(map(str, sequence)))
        return True
    for i in range(1, 11):
        if available_weights[i-1] == '1' and i != current_weight and i > diff:
            if place_weight(i, i-diff, count+1, m, available_weights, sequence + [i]):
                return True
    return False

available_weights = input().strip()
m = int(input().strip())

if not can_place_weights(available_weights, m):
    print("NO")
