from collections import defaultdict
import sys
input = sys.stdin.readline

m, init = list(map(str,input().split())), input().split()
n = int(input())
d = defaultdict(list)
for i in range(n):
    x, y = map(str,input().split())
    d[x].append(y)
r = ['r','R']
for i in d:
    if any(j in i for j in r):
        for j in d[i]:
            d[j].append(i)
uniq = []
for i in init:
    if i not in uniq: uniq.append(i)
for i in uniq:
    if any(j in i for j in r):
        for j in d[i]:
            if j not in uniq: uniq.append(j)

ret = ''.join(init)
for i in uniq:
    if any(j in i for j in r):
        ret = ret.replace(i,d[i][0])
print(ret.count('r')+ret.count('R'), len(ret))
