#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, a[N];
vector<int> vec;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1);
    vec.push_back(a[1]);
    for (int i = 2; i <= n; i++) {
        if (a[i] - vec.back() > 1) break;
        vec.push_back(a[i]);
    }
    int m = vec.size();
    cout << m << endl;
    for (int i = 0; i < m; i++) cout << vec[i] << " ";
    return 0;
}
