#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<string> v(n);
    string c = "ABC";
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (i + k - 1 < n && s[i] == 'Y') {
            v[i] = c[cnt];
            v[i + k - 1] = c[cnt];
            cnt++;
        } else
            v[i] = "AN";
    }
    for (int i = 0; i < n; i++) {
        if (v[i] == "AN") {
            for (char ch = 'A'; ch <= 'Z'; ch++) {
                bool f = 1;
                for (int j = i; j < i + k && j < n; j++) {
                    if (v[j] == string(1, ch)) {
                        f = 0;
                        break;
                    }
                }
                if (f) {
                    v[i] = string(1, ch);
                    break;
                }
            }
        }
    }
    for (int i = 0; i < n; i++) {
        if (v[i] == "AN") {
            for (char ch = 'a'; ch <= 'j'; ch++) {
                bool f = 1;
                for (int j = i; j < i + k && j < n; j++) {
                    if (v[j] == string(1, ch)) {
                        f = 0;
                        break;
                    }
                }
                if (f) {
                    v[i] = string(1, ch);
                    break;
                }
            }
        }
    }
    for (int i = 0; i < n; i++) {
        cout << v[i] << " ";
    }
    cout << endl;
    return 0;
}
