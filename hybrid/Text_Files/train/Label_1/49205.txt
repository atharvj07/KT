import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {

    static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static int H, W;
    static char[][] maze;
    static int[][] dist;
    static int[] dx = {0, 1, 0, -1};
    static int[] dy = {1, 0, -1, 0};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        H = sc.nextInt();
        W = sc.nextInt();
        maze = new char[H][W];
        dist = new int[H][W];
        for (int i = 0; i < H; i++) {
            String line = sc.next();
            for (int j = 0; j < W; j++) {
                maze[i][j] = line.charAt(j);
            }
        }

        int maxDist = 0;
        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                if (maze[i][j] == '.') {
                    bfs(i, j);
                    for (int k = 0; k < H; k++) {
                        for (int l = 0; l < W; l++) {
                            if (maze[k][l] == '.' && dist[k][l] > maxDist) {
                                maxDist = dist[k][l];
                            }
                        }
                    }
                }
            }
        }

        System.out.println(maxDist);
    }

    static void bfs(int sx, int sy) {
        Queue<Point> que = new LinkedList<>();
        que.add(new Point(sx, sy));
        dist[sx][sy] = 0;
        while (!que.isEmpty()) {
            Point p = que.poll();
            for (int i = 0; i < 4; i++) {
                int nx = p.x + dx[i];
                int ny = p.y + dy[i];
                if (0 <= nx && nx < H && 0 <= ny && ny < W && maze[nx][ny] == '.' && dist[nx][ny] == 0) {
                    que.add(new Point(nx, ny));
                    dist[nx][ny] = dist[p.x][p.y] + 1;
                }
            }
        }
    }
}
