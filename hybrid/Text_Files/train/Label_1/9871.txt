#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int n;
        cin >> n;
        int freq[n + 1] = {};
        int a[n];
        int mx = 0;
        for (int i = 0; i < n; i++)
        {
            cin >> a[i];
            freq[a[i]]++;
            mx = max(mx, freq[a[i]]);
        }
        if (mx > n / 2)
            cout << -1 << endl;
        else
            cout << n - 2 * mx << endl;
    }
    return 0;
}
