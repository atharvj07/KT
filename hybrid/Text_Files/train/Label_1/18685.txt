#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<long long> arr(n), ans(n);
    for (int i = 0;i < n;i++) cin >> arr[i];
    ans = arr;
    long long maxi = 0;
    for (int i = 0;i < n;i++) {
        maxi = max(maxi, arr[i]);
    }
    long long mini = 1e9;
    for (int i = 0;i < n;i++) {
        mini = min(mini, arr[i]);
    }
    if (maxi >= 0 && mini >= 0) {
        long long anss = 0;
        for (int i = 0;i < n;i++) {
            anss += (maxi - arr[i]);
        }
        cout << anss << endl;
    }
    else if (maxi < 0 && mini < 0) {
        long long anss = 1e9;
        for (int i = 0;i < n;i++) {
            anss = min(anss, abs(maxi - arr[i]) + abs(mini - arr[i]));
        }
        cout << anss << endl;
    }
    else if (maxi >= 0) {
        long long anss = 1e9;
        for (int i = 0;i < n;i++) {
            anss = min(anss, (maxi - arr[i]));
        }
        cout << anss << endl;
    }
    else {
        long long anss = 1e9;
        for (int i = 0;i < n;i++) {
            anss = min(anss, (abs(maxi) + arr[i]));
        }
        cout << anss << endl;
    }
}
