for _ in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    p = [0] * n
    vis = [False] * (n + 1)
    for x in a:
        if vis[x]: continue
        vis[x] = True
        for i in range(n):
            if p[i] == 0:
                p[i] = x
                break
            if p[i] == x:
                p[i + 1] = x
                break
    print(*p)
