n = int(input())
a = [int(x) for x in input().split()]
a = [x for x in a if x != 0]
n = len(a)
dp = [[0] * 4 for _ in range(n)]
for l in range(n - 1, -1, -1):
    for r in range(l, n):
        if l == r:
            dp[l][1] = a[l]
            dp[l][2] = a[l]
            dp[l][3] = a[l]
            continue
        dp[l][1] = max(dp[l + 1][1], dp[l + 1][2] + a[l])
        for k in range(2, r - l + 2):
            dp[l][k] = max(dp[l][k - 1], dp[l + k - 1][k - 1] + sum(a[l:l + k]))
        dp[l][3] = max(dp[l][3], dp[r][2] + a[l])
print(dp[0][3])
