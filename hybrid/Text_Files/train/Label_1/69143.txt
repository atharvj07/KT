s = input()
cnt = 0
t = 0
for c in s:
    if c == 'B' or c == 'b':
        if t == 0: t = 1
        elif t == 4: t = 5
        elif t == 6: t = 7
    elif c == 'u' or c == 'U':
        if t == 1: t = 2
    elif c == 'l' or c == 'L':
        if t == 2: t = 3
    elif c == 'a' or c == 'A':
        if t == 3: t = 4
    elif c == 's' or c == 'S':
        if t == 5: t = 6
    elif c == 'a' or c == 'A':
        if t == 7: t = 8
    if t == 8:
        t = 0
        cnt += 1
print(cnt)
