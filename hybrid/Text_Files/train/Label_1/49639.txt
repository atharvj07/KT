import java.util.*;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int R = sc.nextInt();
        int L = sc.nextInt();

        Team[] teams = new Team[N + 1];
        for (int i = 1; i <= N; i++) {
            teams[i] = new Team(i);
        }

        for (int i = 0; i < R; i++) {
            int d = sc.nextInt();
            int t = sc.nextInt();
            int x = sc.nextInt();

            teams[d].addScore(t, x);
        }

        Team longestTeam = teams[1];
        for (int i = 2; i <= N; i++) {
            if (teams[i].getLongestTime() > longestTeam.getLongestTime()) {
                longestTeam = teams[i];
            } else if (teams[i].getLongestTime() == longestTeam.getLongestTime() && teams[i].getId() < longestTeam.getId()) {
                longestTeam = teams[i];
            }
        }

        System.out.println(longestTeam.getId());
    }

    static class Team {
        private int id;
        private List<Event> events;

        public Team(int id) {
            this.id = id;
            this.events = new ArrayList<>();
        }

        public int getId() {
            return id;
        }

        public void addScore(int time, int score) {
            events.add(new Event(time, score));
        }

        public int getLongestTime() {
            int longestTime = 0;
            int currentScore = 0;
            int startTime = 0;

            for (Event event : events) {
                if (currentScore == 0) {
                    startTime = event.getTime();
                }
                currentScore += event.getScore();
                if (currentScore == 0) {
                    longestTime = Math.max(longestTime, event.getTime() - startTime);
                }
            }

            return longestTime;
        }
    }

    static class Event {
        private int time;
        private int score;

        public Event(int time, int score) {
            this.time = time;
            this.score = score;
        }

        public int getTime() {
            return time;
        }

        public int getScore() {
            return score;
        }
    }
}
