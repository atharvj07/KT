import sys
input = sys.stdin.readline

n = int(input())
arr = [list(map(int, input().split())) for _ in range(n)]
cnt = 0

for i in range(n):
    mi = 1e9
    for j in range(len(arr[i])):
        mi = min(mi, arr[i][j])
        for k in range(i, n):
            for l in range(len(arr[k])):
                if arr[k][l] > mi:
                    cnt += 1

print(cnt)
