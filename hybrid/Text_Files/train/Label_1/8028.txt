#include<bits/stdc++.h>
using namespace std;
long long k,a,b;
long long gcd(long long a,long long b){
    if(b == 0)return a;
    return gcd(b,a%b);
}
int main(){
    cin>>k>>a>>b;
    long long lcm = k * (a/k);
    if(lcm < a)lcm += k;
    long long ans = b / k - (a-1) / k;
    if(lcm > b)ans--;
    cout<<ans<<endl;
}
