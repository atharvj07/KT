n, k = map(int, input().split())
s = input()

balance = 0
answer = []

for char in s:
    if char == '(' and balance < k // 2:
        answer.append(char)
        balance += 1
    elif char == ')' and balance > 0:
        answer.append(char)
        balance -= 1
        k -= 2

print(''.join(answer))
