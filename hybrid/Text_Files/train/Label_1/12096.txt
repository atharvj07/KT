#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v;
  for (int i = 0; i < n; i++) {
    int a, b;
    cin >> a >> b;
    v.push_back({a, b});
  }
  sort(v.begin(), v.end());
  int mxh = 0, ans = 0;
  for (auto x : v) {
    mxh = max(mxh, x.second);
    ans += x.first * mxh;
  }
  cout << ans << endl;
  return 0;
}
