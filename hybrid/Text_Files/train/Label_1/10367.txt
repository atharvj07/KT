#include <bits/stdc++.h>
using namespace std;
#define MAXN 300005
vector<int> e[MAXN];
vector<int> l(MAXN);
vector<map<int, int>> mem;
int n, m;
int dfs(int x, int p) {
  if (mem[x].count(p)) { return mem[x][p]; }
  map<int, int> &v = mem[x];
  v[p] = 1;
  for (int u : e[x]) {
    if (u == p) { continue; }
    int nv = dfs(u, x);
    if (nv == -1) { return v[p] = -1; }
    v[p] = max(v[p], v[u]);
    if (v[p] >= 26) { return v[p] = -1; }
    v[p] += nv;
  }
  v[p] += l[x] == l[p];
  return v[p];
}
int main() {
  cin >> n >> m;
  mem.resize(n + 1);
  for (int i = 1; i <= n; i++) {
    char c;
    cin >> c;
    l[i] = c - 'a';
  }
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    e[u].push_back(v);
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    int t = dfs(i, -1);
    ans = max(ans, t);
  }
  cout << (ans == -1 ? -1 : ans) << endl;
  return 0;
}
