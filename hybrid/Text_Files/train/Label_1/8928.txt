#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;

void min_remove(vector<int>& v) {
    sort(v.begin(), v.end());
    int n = v.size(), ans = 0;
    for (int i = 1; i < n; i++) {
        if (v[i] % v[i - 1] && v[i - 1] % v[i]) ans++;
    }
    cout << ans << endl;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> v(n);
        for (int i = 0; i < n; i++) cin >> v[i];
        min_remove(v);
    }
    return 0;
}
