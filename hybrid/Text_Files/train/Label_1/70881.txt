n, q = map(int, input().split())

unread_notifications = 0
app_notifications = [0] * (n + 1)
all_notifications = []

for _ in range(q):
    event = list(map(int, input().split()))
    if event[0] == 1:
        x = event[1]
        unread_notifications += 1
        app_notifications[x] += 1
        all_notifications.append(x)

    elif event[0] == 2:
        x = event[1]
        unread_notifications -= app_notifications[x]
        app_notifications[x] = 0

    else:
        t = event[1]
        for i in range(t):
            if i < len(all_notifications):
                app_id = all_notifications[i]
                if app_notifications[app_id] > 0:
                    app_notifications[app_id] -= 1
                    unread_notifications -= 1
        all_notifications = all_notifications[t:]

    print(unread_notifications)
