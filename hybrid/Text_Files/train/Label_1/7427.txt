#include<bits/stdc++.h>
using namespace std;
vector<int>primes;
const int maxn = 100005;
int vis[maxn];
void get_prime(int n){
	for(int i = 2; i * i <= n; ++i){
		if(vis[i] == 0){
			primes.push_back(i);
			for(int j = i * i; j <= n; j += i){
				vis[j] = 1;
			}
		}
	}
	return ;
}
int main() {
	int n;
	cin >> n;
	get_prime(n);
	int ans = 0;
	for(int x : primes){
		if(n % x == 0){
			ans++;
			while(n % x == 0){
				n /= x;
			}
		}
	}
	if(n > 1)ans++;
	cout << ans << endl;
	for(int x : primes){
		if(n % x == 0){
			while(n % x == 0){
				cout << x << " ";
				n /= x;
			}
		}
	}
	if(n > 1)cout << n << " ";
	cout << endl;
	return 0;
}
