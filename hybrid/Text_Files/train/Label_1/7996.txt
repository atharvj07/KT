#include <cstdio>
#include <vector>
#include <bitset>

int main() {
    long long n, k;
    std::vector<std::bitset<32>> ips;
    std::scanf("%lld %lld", &n, &k);
    while (n--) {
        long long a, b, c, d;
        std::scanf("%lld.%lld.%lld.%lld", &a, &b, &c, &d);
        ips.push_back(std::bitset<32>(a << 24 | b << 16 | c << 8 | d));
    }
    while (k--) {
        std::bitset<32> mask;
        long long count = 0;
        long long min = 42;
        for (long long i = 31; i >= 0; i--) {
            bool same = true;
            for (auto &ip : ips) if (ip[i] != ips[0][i]) same = false;
            if (!same) {
                mask[i] = 1;
                count++;
                if (count >= min) break;
            }
        }
        if (count >= min) mask.reset();
        if (mask.any()) {
            for (long long i = 0; i < 4; i++) {
                std::printf("%lld", mask[i * 8] << 7 | mask[i * 8 + 1] << 6 | mask[i * 8 + 2] << 5 | mask[i * 8 + 3] << 4 | mask[i * 8 + 4] << 3 |
                            mask[i * 8 + 5] << 2 | mask[i * 8 + 6] << 1 | mask[i * 8 + 7]);
                if (i < 3) std::printf(".");
            }
            std::printf("\n");
            return 0;
        }
    }
    std::printf("-1\n");
    return 0;
}
