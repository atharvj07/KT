#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll const MOD = 998244353;
ll bin_exp(ll base, ll exp) {
    if (exp == 0) return 1ll;
    if (exp == 1) return base;
    if (exp % 2 == 1) { return (bin_exp(base, exp - 1) * base) % MOD; }
    ll ret = bin_exp(base, exp / 2);
    ret = (ret % MOD * ret % MOD) % MOD;
    return ret;
}
int main() {
    ll ans = 0ll;
    ll n, m; cin >> n >> m;
    string a, b; cin >> a >> b;
    while (b[m - 1] == '0') m--;
    reverse(a.begin(), a.end());
    reverse(b.begin(), b.end());
    for (ll i = 0; i < m; i++) {
        if (b[i] == '1' && a[i] == '1') ans = (ans % MOD + bin_exp(2, (n - i - 1))) % MOD;
        if (i == m - 1) break;
        b.pop_back();
    }
    cout << ans << '\n';
}
