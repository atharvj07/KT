#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define vi(x) vector<x>
const int M = 1e5 + 5;
const ll inf = 1e18;
int n, k, l, m, a[M];
ll dp[M][2];
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cin >> n >> k >> l;
  m = n * k;
  for (int i = 0; i < m; i++) cin >> a[i];
  sort(a, a + m);
  for (int j = 0; j < k; j++) dp[0][j] = -inf;
  dp[0][0] = 0;
  for (int i = 1; i <= m; i++) {
    for (int j = 0; j < k; j++) {
      int op = j;
      dp[i][j] = max(dp[i - 1][j], dp[i][j]);
      if (i >= k) { op -= (ll)l * (op > 0); }
      ll r = a[i - 1];
      if (j != 0) {
        op++;
        if (i >= k) { op -= (ll)l * (op > 0); }
        r = max(r, a[i - 1] - (ll)l * op);
        dp[i][j] = max(dp[i - 1][op] + r, dp[i][j]);
      }
    }
  }
  cout << dp[m][0] << '\n';
  return 0;
}
