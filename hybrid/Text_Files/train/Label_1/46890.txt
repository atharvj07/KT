import java.io.*;
import java.util.*;

public class Main {
    static boolean[][] adj;
    static boolean[] flag;
    static int[] ans;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        int t = Integer.parseInt(br.readLine());
        nextLine:
        while (t-- > 0) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int n = Integer.parseInt(st.nextToken());
            int m = Integer.parseInt(st.nextToken());
            int k = Integer.parseInt(st.nextToken());
            adj = new boolean[n + 1][n + 1];
            flag = new boolean[n + 1];
            ans = new int[n + 1];
            while (m-- > 0) {
                st = new StringTokenizer(br.readLine());
                int a = Integer.parseInt(st.nextToken());
                int b = Integer.parseInt(st.nextToken());
                adj[a][b] = adj[b][a] = true;
            }
            if (k == 1) {
                bw.write("1\n" + n + "\n");
                while (n-- > 0) {
                    ans[n] = n + 1;
                }
                continue nextLine;
            }
            int cnt = 0;
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++) {
                    if (flag[j]) continue;
                    if (!adj[i][j]) continue;
                    ans[cnt++] = j;
                    flag[j] = true;
                    break;
                }
                if (cnt == k) {
                    bw.write("1\n" + cnt + "\n");
                    for (int j = 0; j < cnt; j++) {
                        bw.write(ans[j] + " ");
                    }
                    continue nextLine;
                }
            }
            flag = new boolean[n + 1];
            for (int i = 1; i <= n; i++) {
                if (flag[i]) continue;
                int cnt2 = (adj[i][i] ? 1 : 0);
                for (int j = 1; j <= n; j++) {
                    if (i == j) continue;
                    cnt2 += (adj[i][j] ? 1 : 0);
                    if (flag[j] || !adj[i][j]) continue;
                    ans[cnt++] = j;
                    flag[j] = true;
                }
                if (cnt2 >= k) {
                    bw.write("1\n" + cnt + "\n");
                    for (int j = 0; j < cnt; j++) {
                        bw.write(ans[j] + " ");
                    }
                    continue nextLine;
                }
            }
            bw.write("-1\n");
        }
        bw.flush();
        bw.close();
    }
}
