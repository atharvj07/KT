import java.util.*;
import java.io.*;

class Main {
  static int n, rt;
  static List<Integer>[] g;
  static int[] p, c;
  static long[] res;

  static boolean solve(long[] a) {
    for (int i = 1; i <= n; i++) {
      if (p[i] != 0)
        a[i] = a[p[i]] + c[p[i]] - c[i] + 1;
      if (a[i] < 1 || a[i] > 1e9)
        return false;
    }
    return true;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    n = Integer.parseInt(br.readLine());
    g = new ArrayList[n + 1];
    for (int i = 0; i <= n; i++)
      g[i] = new ArrayList<>();
    p = new int[n + 1];
    c = new int[n + 1];
    res = new long[n + 1];
    for (int i = 1; i <= n; i++) {
      StringTokenizer st = new StringTokenizer(br.readLine());
      p[i] = Integer.parseInt(st.nextToken());
      c[i] = Integer.parseInt(st.nextToken());
      if (p[i] == 0)
        rt = i;
      else {
        g[p[i]].add(i);
        g[i].add(p[i]);
      }
    }
    dfs1(rt, -1);
    if (dfs2(rt, -1)) {
      System.out.println("YES");
      for (int i = 1; i <= n; i++)
        System.out.print(res[i] + " ");
      System.out.println();
    } else
      System.out.println("NO");
  }

  static void dfs1(int u, int par) {
    for (int v : g[u]) {
      if (v == par)
        continue;
      dfs1(v, u);
      c[u] += c[v];
    }
  }

  static boolean dfs2(int u, int par) {
    boolean flag = true;
    for (int v : g[u]) {
      if (v == par)
        continue;
      res[v] = res[u] + c[u] - c[v] - 1;
      flag &= dfs2(v, u);
    }
    return flag & solve(res);
  }
}
