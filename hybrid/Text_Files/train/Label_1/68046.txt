n=int(input())
s=input()
sb=s+s
ans=n
for i in range(1,len(s)):
  if(s[i]==s[i-1]):
    ans=min(ans,i+1+int((n-i)/len(s))+(n-i%len(s)!=0))
    ans=min(ans,i+int(n/len(s))+(n%len(s)!=0))
    ans=min(ans,i+int(n/len(sb))+(n%len(sb)!=0))
    ans=min(ans,i+1+int((n-i)/len(sb))+(n-i%len(sb)!=0))
print(ans)
