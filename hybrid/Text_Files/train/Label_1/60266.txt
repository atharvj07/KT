n = int(input())
is_rated = False
is_unrated = False
prev_a = float('inf')

for i in range(n):
    a, b = map(int, input().split())
    if a != b:
        is_rated = True
        break
    if a <= prev_a:
        prev_a = a
    else:
        is_unrated = True

if is_rated:
    print("rated")
elif is_unrated:
    print("unrated")
else:
    print("maybe")
