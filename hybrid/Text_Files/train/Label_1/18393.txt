#include <bits/stdc++.h>

using namespace std;

#define endl '\n'
#define all(s) s.begin(), s.end()
#define pb push_back

typedef long long ll;

struct ring {
    ll b, a, h;
    bool operator<(const ring &r) const {
        return b < r.b || (b == r.b && a > r.a);
    }
};

const ll MAXN = 100005;
vector<ring> rings;

ll dp[MAXN];

int main() {
    ll n;
    cin >> n;
    rings.resize(n);
    for (ll i = 0; i < n; i++) {
        cin >> rings[i].a >> rings[i].b >> rings[i].h;
    }
    sort(rings.begin(), rings.end());
    for (ll i = 0; i < n; i++) {
        dp[i] = rings[i].h;
        for (ll j = i - 1; j >= 0; j--) {
            if (rings[j].b > rings[i].a) {
                dp[i] = max(dp[i], dp[j] + rings[i].h);
            }
        }
    }
    ll mx = 0;
    for (ll i = 0; i < n; i++) {
        mx = max(mx, dp[i]);
    }
    cout << mx;
    return 0;
}
