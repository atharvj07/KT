mo, ma = [], []
for n in input().split(' '): mo.append(int(n))
for n in input().split(' '): ma.append(int(n))
for n in range(len(mo)):
    if mo[n] < ma[n]:
        x = mo[n]
        y = ma[n]
        mo.remove(x)
        ma.remove(y)
        mo.append(x)
        ma.append(y)
flicks = 0
for n in range(len(mo)):
    if mo[n] < ma[n]: flicks += 1
print(flicks, len(mo)-flicks)
