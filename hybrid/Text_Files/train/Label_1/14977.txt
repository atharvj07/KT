#include <bits/stdc++.h>
using namespace std;
using ll=long long;
#define int ll

int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &el : a) cin >> el;
    vector<int> p(n + 1), suf(n + 1);
    p[1] = a[0];
    for (int i = 2; i <= n; i++) p[i] = a[i - 1] + p[i - 1];
    for (int i = n; i >= 1; i--) suf[i] = a[i - 1] + suf[i + 1];
    int ans = LLONG_MIN;
    for (int i = 1; i <= n; i++) {
        ans = max(ans, p[i] + suf[i + 1]);
    }
    cout << ans;
    return 0;
}
