n, m = map(int, input().split())
ban = [(int(x), int(y)) for x, y in [input().split() for _ in range(m)]]
G = [[] for _ in range(n + 1)]
for a, b in ban:
    G[a] += [b]
    G[b] += [a]

ans = []
for i in range(1, n + 1):
    tmp = [j for j in range(1, n + 1) if j != i and j not in G[i]]
    if len(tmp) > 0:
        G[i] += [tmp[0]]
        G[tmp[0]] += [i]
        ans += [(i, tmp[0])]
        m -= 1
    if m == 0: break

print(len(ans))
for a, b in ans:
    print(a, b)
