import sys
from collections import defaultdict

n=int(input())
g=defaultdict(list)
for _ in range(n-1):
    a,b=map(int,sys.stdin.readline().split())
    g[a].append(b)
    g[b].append(a)
q=[1]
vis=[0]*n
odd=[]
while q:
    u=q.pop()
    if vis[u-1]:
        if len(g[u])%2:
            odd.append(u)
            continue
    else:
        vis[u-1]=1
    for v in g[u]:
        if not vis[v-1]:
            q.append(v)
if len(odd)<2:
    print("No")
else:
    print("Yes")
    print(len(odd)//2)
    for i in range(0,len(odd),2):
        print(odd[i],odd[i+1])
