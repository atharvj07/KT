#include <iostream>
#include <unordered_map>
#include <vector>
#include <algorithm>

bool solve() {
  std::string p, h;
  std::cin >> p >> h;
  if (p.size() * 2 + 1 > h.size()) {
    return false;
  }
  std::unordered_map<char, int> pr, hr;
  for (char c : p) { ++pr[c]; }
  h.erase(std::remove(h.begin(), h.end(), ' '), h.end());
  for (char c : h) { ++hr[c]; }
  if (pr != hr) {
    return false;
  }
  if (p.size() == h.size()) {
    return true;
  }
  int pos = 0;
  for (char c : h) {
    if (pos < p.size() && c == p[pos]) {
      ++pos;
      continue;
    }
    if (pos == p.size()) {
      return true;
    }
  }
  return false;
}

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    if (solve()) {
      std::cout << "YES" << std::endl;
    } else {
      std::cout << "NO" << std::endl;
    }
  }
  return 0;
}
