import math
n=int(input())
x=[0]*n
y=[0]*n
r=[0]*n
for i in range(n):
    x[i],y[i],r[i]=map(int,input().split())
region=0
for i in range(n):
    for j in range(i+1,n):
        d=math.sqrt((x[i]-x[j])**2+(y[i]-y[j])**2)
        if d>r[i]+r[j]:
            region+=2
        elif d==r[i]+r[j]:
            region+=1
region+=1
print(region)
