#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int n = s.length(), ans = 0;
  vector<bool> f(n), g(n);
  for (int i = n - 1; i >= 0; --i) {
    int j = (i != 0);
    f[i] = ((s[i] - '0') % 3 + g[i] + (f[i + 1] % 3) * 10) % 3 == 0;
    g[i] = ((s[i] - '0') % 3 + f[i + 1] + (g[i + 1] % 3) * 10) % 3 == 0;
    ans += f[i];
  }
  cout << ans << endl;
  return 0;
}
