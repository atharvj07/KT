import java.util.ArrayDeque;
import java.util.Queue;
import java.util.Scanner;

public class JaberMissions {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] grid = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = scanner.nextInt();
            }
        }

        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int r1 = scanner.nextInt() - 1;
            int c1 = scanner.nextInt() - 1;
            int r2 = scanner.nextInt() - 1;
            int c2 = scanner.nextInt() - 1;

            int minTime = findMinTime(grid, r1, c1, r2, c2);
            System.out.println(minTime);
        }
    }

    private static int findMinTime(int[][] grid, int r1, int c1, int r2, int c2) {
        int n = grid.length;
        int m = grid[0].length;

        boolean[][] visited = new boolean[n][m];

        Queue<int[]> queue = new ArrayDeque<>();
        queue.offer(new int[]{r1, c1, 0});
        visited[r1][c1] = true;

        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            int row = current[0];
            int col = current[1];
            int time = current[2];

            if (row == r2 && col == c2) {
                return time;
            }

            int[][] directions = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
            for (int[] direction : directions) {
                int newRow = row + direction[0];
                int newCol = col + direction[1];

                if (newRow >= 0 && newRow < n && newCol >= 0 && newCol < m && !visited[newRow][newCol]) {
                    if (grid[newRow][newCol] == grid[row][col] || grid[newRow][newCol] == grid[r2][c2]) {
                        visited[newRow][newCol] = true;
                        queue.offer(new int[]{newRow, newCol, time});
                    } else {
                        visited[newRow][newCol] = true;
                        queue.offer(new int[]{newRow, newCol, time + 1});
                    }
                }
            }
        }

        return -1;
    }
}
