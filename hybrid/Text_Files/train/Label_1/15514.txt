#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n - 1), b(n - 1);
  for (int i = 0; i < n - 1; i++) cin >> a[i];
  for (int i = 0; i < n - 1; i++) cin >> b[i];
  cout << (a[0] & b[0] == 0 ? "YES\n" : "NO\n");
  if (a[0] & b[0] == 0) {
    cout << '0';
    for (int i = 0; i < n - 1; i++) cout << ' ' << (a[i] ^ b[i]);
    cout << '\n';
  }
  return 0;
}
