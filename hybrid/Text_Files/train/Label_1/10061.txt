#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> c(n);
    for (int &i : c) cin >> i;
    vector<bool> capital(n);
    for (int i; cin >> i;) capital[i - 1] = true;
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        ans += c[i] * (c[(i + 1) % n] + capital[i] * c[(i - 1 + n) % n]);
    }
    cout << ans << '\n';
}
