import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            int n = in.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = in.nextInt();
            }
            boolean can = true;
            for (int j = 1; j < n; j++) {
                if (a[j] != a[j - 1]) {
                    can = false;
                    break;
                }
            }
            if (can) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
            }
        }
    }
}
