#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    int dp[n][10];
    int a[n][6];
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 6; j++) {
            cin >> a[i][j];
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 10; j++) {
            dp[i][j] = -1;
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 1; j < 10; j++) {
            for (int k = 0; k < 6; k++) {
                if (a[i][k] == j) {
                    if (i == 0) {
                        dp[i][j] = 1;
                    } else if (dp[i - 1][j - 1] != -1) {
                        dp[i][j] = dp[i - 1][j - 1] + 1;
                        ans = max(ans, dp[i][j]);
                    }
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
