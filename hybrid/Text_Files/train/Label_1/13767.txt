#include <iostream>
using namespace std;

int main() {
  int n;
  cin >> n;
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    string s = to_string(i);
    bool ok = 1;
    for (auto &c : s) {
      if (c != '0' && c != '4' && c != '7') ok = 0;
    }
    ans += ok;
  }
  cout << ans << endl;
  return 0;
}
