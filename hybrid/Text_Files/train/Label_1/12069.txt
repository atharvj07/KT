#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
struct edge {
    ll v, w;
};
vector<vector<edge>> ar;
vector<ll> ans;
vector<bool> vis;
void dfs(ll start, ll p = -1, ll d = 0) {
    vis[start] = 1;
    ans.push_back(start);
    for (auto& cur : ar[start]) {
        ll u = cur.v, w = cur.w;
        if (!vis[u]) dfs(u, start, d + 1);
    }
    if (p != -1) ans.push_back(p);
}
int main() {
    ll n, q;
    cin >> n >> q;
    ar.resize(n + 1);
    for (ll i = 2; i <= n; i++) {
        ll p;
        cin >> p;
        ar[p].push_back({i, 0});
        ar[i].push_back({p, 0});
    }
    for (ll i = 0; i <= n; i++) vis.push_back(0);
    dfs(1);
    while (q--) {
        ll u, k;
        cin >> u >> k;
        if (k > ans.size())
            cout << -1 << "\n";
        else
            cout << ans[k - 1] << "\n";
    }
    return 0;
}
