#include<bits/stdc++.h>
using namespace std;
int main() {
    int x;
    cin >> x;
    while (x--) {
        int n, p;
        cin >> n >> p;
        int sub = 2 * n + p - n * (n - 1) / 2;
        vector<vector<int>> graph(n + 1);
        for (int i = 1; i <= (n / 2); i++) {
            for (int j = i + 1; j <= n; j++) {
                graph[i].push_back(j);
                sub--;
                if (!sub) break;
            }
            if (!sub) break;
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < graph[i].size(); j++)
                cout << i << " " << graph[i][j] << endl;
        }
    }
}
