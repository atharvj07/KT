import sys
input = sys.stdin.readline

def solve():
    h, q = map(int, input().split())
    ans = [[] for i in range(h+1)]
    for i in range(q):
        i, l, r, t = map(int, input().split())
        ans[i].append((l, r, t))
    
    def check(node):
        if node > (1<<h)-1: return True
        flag1, flag2 = False, False
        for l, r, t in ans[node]:
            if t:
                if node%2 == 0: flag1 |= check(node//2*2+1)
                else: flag2 |= check(node//2*2)
        if node%2 == 0 and flag1 and flag2: return False
        elif node%2 == 1 and (flag1 or flag2): return False
        else: return True
    
    if check(1): return "Game cheated!"
    ex = 1
    for i in range(2, h+1):
        if not check(ex*2): ex = ex*2+1
    if check(ex): return ex
    else: return "Data not sufficient!"

print(solve())
