#include <iostream>
#include <vector>
#include <queue>

int main() {
    int n, m, k;
    std::cin >> n >> m >> k;
    if (2 * k > n) {
        std::cout << n << "\n";
        for (int i = 1; i <= n; i++) {
            std::cout << i << " ";
        }
        std::cout << "\n";
        return 0;
    }
    std::vector<std::vector<int>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int a, b;
        std::cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    std::queue<std::pair<int, int>> q;
    std::vector<bool> vis(n + 1, false);
    for (int i = 1; i <= n; i++) {
        if (adj[i].size() < k) {
            q.push({i, i});
            vis[i] = true;
        }
    }
    while (!q.empty()) {
        int u = q.front().first;
        int p = q.front().second;
        q.pop();
        for (auto v : adj[u]) {
            if (v == p) continue;
            if (vis[v]) {
                std::vector<int> cycle;
                while (u != v) {
                    cycle.push_back(u);
                    u = p;
                    if (vis[u] == 1) {
                        p = q.front().second;
                        u = q.front().first;
                        q.pop();
                    } else {
                        p = u;
                    }
                }
                cycle.push_back(u);
                std::cout << cycle.size() << "\n";
                for (auto x : cycle) {
                    std::cout << x << " ";
                }
                std::cout << "\n";
                return 0;
            } else {
                vis[v] = 1;
                q.push({v, u});
            }
        }
        vis[u] = 2;
    }
    return 0;
}
