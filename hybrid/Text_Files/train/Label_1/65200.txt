import collections

n = int(input())
l = list(map(int, input().split()))
d = collections.Counter(l)
t = min(d[1], min(d[2], d[3]))
print(t)
for i in range(t):
    print(" ".join([str(li+1) for li, v in enumerate(l) if v==1][i]), " ".join([str(li+1) for li, v in enumerate(l) if v==2][i]), " ".join([str(li+1) for li, v in enumerate(l) if v==3][i]))
