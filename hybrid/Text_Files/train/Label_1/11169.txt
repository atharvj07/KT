#include <iostream>
#include <vector>
#include <cstring>
using namespace std;
int a[100], pref[100];
int main() {
    int n; cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        pref[i] = a[i] ^ pref[i - 1];
    }
    int mx = 0;
    for (int l = 1; l <= n; l++) {
        for (int r = l; r <= n; r++) {
            mx = max(mx, pref[r] ^ pref[l - 1]);
        }
    }
    cout << mx << endl;
}
