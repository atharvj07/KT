n = int(input())
sequences = [list(map(int, input().split()))[1:] for _ in range(n)]

has_ascent = [False] * n
min_vals = []
max_vals = []

for i, seq in enumerate(sequences):
    min_val = seq[0]
    max_val = seq[0]
    for num in seq:
        if num > min_val:
            has_ascent[i] = True
            break
        min_val = min(min_val, num)
        max_val = max(max_val, num)
    if not has_ascent[i]:
        min_vals.append(min_val)
        max_vals.append(max_val)

min_vals.sort()
max_vals.sort()

count = 0
for max_val in max_vals:
    count += len(min_vals) - (bisect.bisect_left(min_vals, max_val + 1))

total_pairs = n ** 2
ascent_pairs = total_pairs - count

print(ascent_pairs)
