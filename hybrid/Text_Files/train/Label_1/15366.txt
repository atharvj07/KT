#include<bits/stdc++.h>
using namespace std;
int main(){
    int t; cin >> t;
    while(t--){
        int n, k; cin >> n >> k;
        vector<int> v(n);
        for(int &i : v) cin >> i;
        sort(v.begin(), v.end());
        if(v[0] == v[n - 1]) cout << "yes" << '\n';
        else cout << "no" << '\n';
    }
    return 0;
}
