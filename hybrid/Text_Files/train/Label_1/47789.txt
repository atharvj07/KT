import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] ch = sc.next().toCharArray();
        int ans = 0;
        int x = 0;
        int y = 0;
        for (char c : ch) {
            if (c == 'X') {
                x++;
            } else {
                y++;
            }
        }
        ans = Math.abs(x - y) / 2;
        System.out.println(ans);
        for (int i = 0; i < n; i++) {
            if (x == y) {
                break;
            }
            if (x > y) {
                if (ch[i] == 'X') {
                    ch[i] = 'x';
                    x--;
                }
            } else {
                if (ch[i] == 'x') {
                    ch[i] = 'X';
                    y--;
                }
            }
        }
        System.out.println(ch);
    }
}
