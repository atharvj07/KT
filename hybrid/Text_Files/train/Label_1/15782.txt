#include <iostream>

int main() {
    int n; std::cin >> n;
    std::string seq; std::cin >> seq;
    bool first = true, ok = true;
    for (char c : seq) {
        if (c == '0')
            if (!first)
                ok &= false;
        else
            first = false;
    }
    std::cout << (ok ? "Yes" : "No") << std::endl;
    return 0;
}
