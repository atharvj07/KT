#include <bits/stdc++.h>
using namespace std;
#define all(x) x.begin(), x.end()
#define ll long long
const ll mod = 1e9 + 7;
 
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> pf(n + 1);
    for (int i = 0; i < n; i++) {
        int x = a[i];
        int g = 0;
        for (int j = 2; j * j <= x; j++) {
            int c = 0;
            while (x % j == 0) {
                x /= j;
                c++;
            }
            g = __gcd(g, c % 2);
        }
        if (x > 1) g = __gcd(g, 1);
        pf[i + 1] = g;
    }
    for (int i = 1; i <= n; i++) pf[i] = __gcd(pf[i], pf[i - 1]);
    vector<int> ans(n + 1);
    for (int i = 0; i < n; i++) {
        for (int j = i; j < n; j++) {
            int g = pf[j + 1];
            if (i > 0) g = __gcd(g, pf[i]);
            if (g == 1) ans[j - i + 1]++;
        }
    }
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    cout << "\n";
}
