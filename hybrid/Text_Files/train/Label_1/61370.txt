import sys

rl = sys.stdin.readline

n, m, k = map(int, rl().split())
a = [list(map(int, rl().split())) for _ in range(n)]
ans = 0  # Maximum possible score
mn = 10**9  # Minimum number of replacements

for j in range(m):
    start = -1
    for i in range(n):
        if a[i][j] == 1:
            ans += 1
            if start == -1:  # Finding topmost 1 in the column
                start = i
            if start + k <= i:  # Out of range elements
                break
            ans += sum(a[start + 1:i + 1])
            mn = min(mn, i - start)

print(ans, mn)
