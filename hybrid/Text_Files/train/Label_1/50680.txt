import java.util.Scanner;

public class Main {

    static int getMaxSatisfaction(int limit, int n, int m, int x, String[] secNames, int[] secEmploys, int[] secSatisfactions, String[] names, int[] employs, int[] satisfactions) {
        int[][][] dp = new int[n + 1][m + 1][limit + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                for (int k = 0; k <= limit; k++) {
                    dp[i][j][k] = dp[i - 1][j][k];
                    if (k >= secEmploys[i - 1]) {
                        dp[i][j][k] = Math.max(dp[i][j][k], dp[i - 1][j][k - secEmploys[i - 1]] + secSatisfactions[i - 1]);
                    }
                    if (j > 0 && k >= employs[j - 1]) {
                        dp[i][j][k] = Math.max(dp[i][j][k], dp[i][j - 1][k - employs[j - 1]] + satisfactions[j - 1]);
                    }
                }
            }
        }
        int maxSatisfaction = 0;
        for (int j = x; j <= m; j++) {
            maxSatisfaction = Math.max(maxSatisfaction, dp[n][j][limit]);
        }
        return maxSatisfaction;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int limit = scanner.nextInt();
            int n = scanner.nextInt();
            int m = scanner.nextInt();
            int x = scanner.nextInt();
            if (limit == 0 && n == 0 && m == 0 && x == 0) {
                break;
            }
            String[] secNames = new String[n];
            int[] secEmploys = new int[n];
            int[] secSatisfactions = new int[n];
            for (int i = 0; i < n; i++) {
                secNames[i] = scanner.next();
                secEmploys[i] = scanner.nextInt();
                secSatisfactions[i] = scanner.nextInt();
            }
            String[] names = new String[m];
            int[] employs = new int[m];
            int[] satisfactions = new int[m];
            for (int i = 0; i < m; i++) {
                names[i] = scanner.next();
                employs[i] = scanner.nextInt();
                satisfactions[i] = scanner.nextInt();
            }
            int maxSatisfaction = getMaxSatisfaction(limit, n, m, x, secNames, secEmploys, secSatisfactions, names, employs, satisfactions);
            System.out.println(maxSatisfaction);
        }
        scanner.close();
    }
}
