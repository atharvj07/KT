#include <bits/stdc++.h>
using namespace std;

const int MAXN = 20;
int n, m, p[MAXN][MAXN];
int row[MAXN], col[MAXN];

int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      cin >> p[i][j];
      row[i] |= 1 << (p[i][j] - 1);
      col[j] |= 1 << (p[i][j] - 1);
    }
  bool ok = true;
  for (int i = 0; i < n; i++)
    if (row[i] != (1 << m) - 1) ok = false;
  for (int i = 0; i < m; i++)
    if (col[i] != (1 << m) - 1) ok = false;
  cout << (ok ? "YES" : "NO") << endl;
}
