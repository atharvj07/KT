// Import the necessary libraries.
import java.util.*;
import java.io.*;

// Create the Main class.
public class Main {
    // Create the main method.
    public static void main(String[] args) throws IOException {
        // Create a Scanner object to read input from the standard input stream.
        Scanner scanner = new Scanner(System.in);

        // Read the number of cities and roads from the input.
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        // Create an array to store the taxes in each city.
        long[] taxes = new long[n];

        // Read the taxes in each city from the input.
        for (int i = 0; i < n; i++) {
            taxes[i] = scanner.nextLong();
        }

        // Create a graph to store the roads.
        Map<Integer, List<Integer>> graph = new HashMap<>();

        // Read the roads from the input.
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            List<Integer> neighbors = graph.getOrDefault(u, new ArrayList<>());
            neighbors.add(v);
            graph.put(u, neighbors);
        }

        // Create a queue to store the cities that have not been visited yet.
        Queue<Integer> queue = new LinkedList<>();
        queue.add(1);

        // Create a set to store the cities that have been visited.
        Set<Integer> visited = new HashSet<>();
        visited.add(1);

        // Create an array to store the taxes in each city after the convention.
        long[] newTaxes = new long[n];

        // While the queue is not empty, continue the BFS.
        while (!queue.isEmpty()) {
            // Remove the first city from the queue.
            int city = queue.remove();

            // Update the tax in the city.
            newTaxes[city - 1] = 0;

            // For each neighbor of the city, add it to the queue if it has not been visited yet.
            for (int neighbor : graph.getOrDefault(city, List.of())) {
                if (!visited.contains(neighbor)) {
                    queue.add(neighbor);
                    visited.add(neighbor);
                }
            }
        }

        // Check if the Wrong Answer Party can win the election.
        boolean canWin = true;
        for (int i = 0; i < n; i++) {
            if (taxes[i] > newTaxes[i]) {
                canWin = false;
                break;
            }
        }

        // Print the result.
        if (canWin) {
            System.out.println("WIN");
            for (int i = 0; i < n; i++) {
                System.out.print(newTaxes[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("LOSE");
        }
    }
}
