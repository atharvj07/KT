n, m = map(int, input().split())
f = dict()
for i in map(int, input().split()):
    f[i] = f.get(i, 0) + 1
bs = [int(x) for x in input().split()]
a = [-1] * m
used = set()
for i in range(m):
    if bs[i] in f and f[bs[i]] > 0 and bs[i] not in used:
        f[bs[i]] -= 1
        a[i] = bs[i]
        used.add(bs[i])
if -1 in a:
    print('Impossible')
else:
    if len(set(bs)) != len(bs):
        print('Ambiguity')
    else:
        print('Possible')
        print(*a)
