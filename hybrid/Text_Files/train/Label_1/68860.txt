n = int(input())
a = [int(input()) for i in range(n)]
b = [0] * 101
for x in a:
    b[x] += 1
ans = []
for i in range(1, 101):
    if b[i]:
        b[i] -= 1
        ans.append(i)
        break
for i in range(1, 101):
    if b[i]:
        b[i] -= 1
        ans.append(i)
        break
if sum(b) > 0:
    print("NO")
else:
    print("YES")
    print(*ans)
