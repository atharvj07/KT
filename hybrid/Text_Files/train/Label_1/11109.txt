#include <iostream>
#include <cstdio>
#include <vector>
using namespace std;

long long k, a, b;

int main() {
  cin >> k >> a >> b;
  vector<vector<int>> al(4, vector<int>(4));
  for (int i = 1; i < 4; i++) {
    for (int j = 1; j < 4; j++) { cin >> al[i][j]; }
  }
  vector<vector<int>> bob(4, vector<int>(4));
  for (int i = 1; i < 4; i++) {
    for (int j = 1; j < 4; j++) { cin >> bob[i][j]; }
  }
  long long Alice = 0, Bob = 0;
  for (long long i = 0; i < k; i++) {
    if (a == b) {
    } else if (a == 1 && b == 2) {
      Bob++;
    } else if (a == 2 && b == 3) {
      Bob++;
    } else if (a == 3 && b == 1) {
      Bob++;
    } else if (a == 1 && b == 3) {
      Alice++;
    } else if (a == 2 && b == 1) {
      Alice++;
    } else if (a == 3 && b == 2) {
      Alice++;
    }
    a = al[a][b];
    b = bob[b][a];
  }
  cout << Alice << " " << Bob << endl;
  return 0;
}
