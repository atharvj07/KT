import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class BullsAndCows {

    public static String guessNumber(String[][] guesses) {
        Set<String> possibleNumbers = new HashSet<>();
        for (int i = 0; i <= 9999; i++) {
            String number = String.format("%04d", i);
            if (isValidNumber(number)) {
                possibleNumbers.add(number);
            }
        }

        for (String[] guess : guesses) {
            String experimentalNumber = guess[0];
            int bulls = Integer.parseInt(guess[1]);
            int cows = Integer.parseInt(guess[2]);
            possibleNumbers = filterNumbers(possibleNumbers, experimentalNumber, bulls, cows);
        }

        if (possibleNumbers.size() == 1) {
            return possibleNumbers.iterator().next();
        } else if (possibleNumbers.isEmpty()) {
            return "Incorrect data";
        } else {
            return "Need more data";
        }
    }

    private static boolean isValidNumber(String number) {
        return number.length() == 4 && number.charAt(0) != '0' && new HashSet<>(List.of(number.split(""))).size() == 4;
    }

    private static Set<String> filterNumbers(Set<String> possibleNumbers, String experimentalNumber, int bulls, int cows) {
        Set<String> filteredNumbers = new HashSet<>();
        for (String number : possibleNumbers) {
            if (checkBullsAndCows(number, experimentalNumber) == bulls + cows) {
                filteredNumbers.add(number);
            }
        }
        return filteredNumbers;
    }

    private static int checkBullsAndCows(String number1, String number2) {
        int bulls = 0;
        int cows = 0;
        for (int i = 0; i < 4; i++) {
            if (number1.charAt(i) == number2.charAt(i)) {
                bulls++;
            } else if (number2.contains(String.valueOf(number1.charAt(i)))) {
                cows++;
            }
        }
        return bulls + cows;
    }

    public static void main(String[] args) {
        String[][] guesses = {
                {"1263", "1", "2"},
                {"8103", "2", "1"}
        };
        String result = guessNumber(guesses);
        System.out.println(result);
    }
}
