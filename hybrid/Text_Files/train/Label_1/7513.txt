#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
int n, k;
int cost[MAXN], p[MAXN], c[MAXN];
int main() {
  cin >> n >> k;
  string s;
  cin >> s;
  s = ' ' + s;
  int sum = 0;
  for (int i = 1; i <= n; i++) {
    if (i > 1) cost[i] = cost[i - 1];
    sum += ('a' <= s[i] && s[i] <= 'z');
    if (s[i] == s[i - 1]) cost[i]++;
    p[i] = i - 1 - cost[i];
    if (i > 1) c[i] = c[p[i]] + 1;
  }
  if ((sum + c[n]) < k) {
    cout << -1 << endl;
    return 0;
  }
  int ans = 0;
  for (int i = n; i >= 1; i--) {
    if (c[i] + sum >= k) {
      ans += n - i;
      sum -= c[i];
    }
  }
  cout << ans << endl;
}
