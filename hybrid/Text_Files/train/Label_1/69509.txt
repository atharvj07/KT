import sys
def main():
    n = int(sys.stdin.readline())
    if n == 1 or n == 2:
        print("YES")
        print(1)
        return

    if n % 2 == 0:
        print("NO")
        return

    print("YES")
    p = [1, 3]
    for i in range(5, n + 1, 2):
        p.append(i)

    print(" ".join(map(str, p)))

main()
