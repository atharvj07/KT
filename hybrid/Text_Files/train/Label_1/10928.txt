#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, h, l, r; cin >> n >> h >> l >> r;
  vector<int> a(n); for (int& x : a) cin >> x;

  sort(a.begin(), a.end());
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] - 1 >= l) ans++;
    if (a[i] >= r) break;
  }
  cout << ans << endl;
  return 0;
}
