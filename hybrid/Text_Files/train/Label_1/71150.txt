def count_possible_integers(s):
    if s == '0':
        return 1
    if 'X' in s:
        if s[0] == 'X' and len(set(s) - {'X'}) == 0:
            return 0
        if s[0] == 'X':
            possible_ends = ['00', '25', '50', '75']
            count = 0
            for end in possible_ends:
                if s[-2:] == end or (s[-2] == 'X' and s[-1] == end[1]) or (s[-1] == 'X' and s[-2] == end[0]) or (s[-2:] == 'XX'):
                    count += 1
            if s[0] == 'X':
                return 9 * count
            return count
        else:
            possible_ends = ['00', '25', '50', '75']
            count = 0
            for end in possible_ends:
                if s[-2:] == end or (s[-2] == 'X' and s[-1] == end[1]) or (s[-1] == 'X' and s[-2] == end[0]) or (s[-2:] == 'XX'):
                    count += 1
            return count
    else:
        count = 0
        for i in range(10):
            for j in range(10):
                new_s = s.replace('_', str(i))
                if (new_s.endswith('00') or new_s.endswith('25') or new_s.endswith('50') or new_s.endswith('75')) and new_s[0] != '0':
                    count += 1
                    break
        return count

s = input().strip()
print(count_possible_integers(s))
