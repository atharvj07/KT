import sys

def solve(s):
    n = len(s)
    if n == 2:
        return False
    if s[0] < s[1]:
        return True
    for i in range(1, n):
        if s[i] < s[i - 1]:
            return False
    return True

for line in sys.stdin:
    if line.strip() == "":
        break
    q = int(line)
    for i in range(q):
        n = int(input())
        s = input()
        if solve(s):
            print("YES")
            print(n - 1)
            for i in range(n - 1):
                print(s[i], end="")
            print()
        else:
            print("NO")
