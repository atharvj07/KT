def can_remove_all_stones(t, test_cases):
    for i in range(t):
        n = test_cases[i][0]
        a = test_cases[i][1]
        prefix_sum = [0] * (n + 1)
        for j in range(n):
            prefix_sum[j + 1] = prefix_sum[j] + a[j]
        total_sum = prefix_sum[n]
        
        if total_sum % 2 == 1:
            print("NO")
            continue
        
        half_sum = total_sum // 2
        possible = False
        for j in range(1, n):
            if prefix_sum[j] == half_sum:
                possible = True
                break
            elif prefix_sum[j] > half_sum:
                diff = prefix_sum[j] - half_sum
                if (j > 1 and a[j - 2] >= diff) or (j < n and a[j] >= diff):
                    possible = True
                    break
        
        if possible:
            print("YES")
        else:
            print("NO")

t = int(input())
test_cases = []
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    test_cases.append((n, a))

can_remove_all_stones(t, test_cases)
