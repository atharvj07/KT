import java.util.Scanner;

class Main {

    private static boolean isProper(int[][] net) {
        // Check if opposite faces sum to 7
        for (int i = 0; i < 5; i++) {
            if (net[i][0] + net[i][4] != 7) {
                return false;
            }
            if (net[0][i] + net[4][i] != 7) {
                return false;
            }
        }

        // Check if any faces have duplicate numbers
        for (int i = 0; i < 5; i++) {
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    for (int l = 0; l < 5; l++) {
                        if (i != k || j != l) {
                            if (net[i][j] == net[k][l]) {
                                return false;
                            }
                        }
                    }
                }
            }
        }

        // Check if the net forms a cube
        boolean[][] visited = new boolean[5][5];
        int count = 0;
        for (int i = 0; i < 5; i++) {
            for (int j = 0; j < 5; j++) {
                if (net[i][j] != 0 && !visited[i][j]) {
                    dfs(net, visited, i, j);
                    count++;
                }
            }
        }
        return count == 1;
    }

    private static void dfs(int[][] net, boolean[][] visited, int i, int j) {
        visited[i][j] = true;
        if (i > 0 && net[i - 1][j] != 0 && !visited[i - 1][j]) {
            dfs(net, visited, i - 1, j);
        }
        if (j > 0 && net[i][j - 1] != 0 && !visited[i][j - 1]) {
            dfs(net, visited, i, j - 1);
        }
        if (i < 4 && net[i + 1][j] != 0 && !visited[i + 1][j]) {
            dfs(net, visited, i + 1, j);
        }
        if (j < 4 && net[i][j + 1] != 0 && !visited[i][j + 1]) {
            dfs(net, visited, i, j + 1);
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        for (int i = 0; i < n; i++) {
            int[][] net = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    net[j][k] = scanner.nextInt();
                }
            }
            System.out.println(isProper(net));
        }
    }
}
