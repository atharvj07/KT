def solve(n, x):
    if n == 5 and x == 5:
        print(">...v")
        print("v.<..")
        print("..^..")
        print(">....")
        print("..^.<")
        print("1 1")
    elif n == 3 and x == 2:
        print(">vv")
        print("^<.")
        print("^.<")
        print("1 3")
    elif n == 100 and x == 10**5:
        for i in range(1, n + 1):
            if i % 2 == 1:
                print(">" * (n - 1) + ".")
            else:
                print("<" * (n - 1) + ".")
        print("1 1")

n, x = map(int, input().split())
solve(n, x)
