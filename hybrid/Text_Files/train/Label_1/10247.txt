#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    while (n--) {
        string s;
        cin >> s;
        int l = s.size();
        for (int i = 1; i < l; i++) {
            if (s[i] - s[i - 1] != 1) {
                cout << "No" << endl;
                goto nxt;
            }
        }
        cout << "Yes" << endl;
    nxt:;
    }
    return 0;
}
