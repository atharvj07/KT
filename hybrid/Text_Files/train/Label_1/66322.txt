from typing import List
from functools import reduce

n = int(input())
balls = list(map(int, input().split()))
balls.sort()

def ways_to_win(a: int, b: int, c: int, n: int) -> int:
  if b > a:
    return 0
  elif a == 0 or b == n:
    return (1 << n) - 1
  ans = 0
  for i in range(b, n):
    if i != c:
      ans += ways_to_win(i, i + 1, c, n)
  return ans

ans = ways_to_win(balls[1], balls[2], balls[0], n) / (1 << n)
print(ans)
