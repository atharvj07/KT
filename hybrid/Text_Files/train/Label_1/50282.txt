import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        boolean[] isPrime = new boolean[51];
        for (int i = 2; i <= 50; i++) {
            isPrime[i] = true;
        }
        for (int i = 2; i <= 50; i++) {
            if (isPrime[i]) {
                for (int j = i * i; j <= 50; j += i) {
                    isPrime[j] = false;
                }
            }
        }
        for (int i = n + 1; i <= 50; i++) {
            if (isPrime[i]) {
                System.out.println(i == m ? "YES" : "NO");
                return;
            }
        }
    }
}
