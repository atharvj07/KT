import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] c = new int[m + 1];
        for (int i = 0; i < n; i++) {
            int p = sc.nextInt(), v = sc.nextInt();
            c[p] = Math.min(c[p], v);
        }
        int[] a = c.clone();
        Arrays.sort(a);
        long ans = 0;
        for (int i = m; i > 1; i--) {
            ans += a[i - 1];
            for (int j = m; j > 1 && c[j] > 0; j--) {
                c[j - 1] = Math.min(c[j - 1], c[j]);
                c[j] = 0;
            }
        }
        System.out.println(ans);
    }
}
