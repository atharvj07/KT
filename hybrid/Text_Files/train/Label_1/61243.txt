for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    if sum(a) & 1:
        print("NO")
        continue
    if a[-1] & 1:
        a[-1], a[-2] = a[-2], a[-1]
    for i in range(n - 1):
        if a[i] & 1:
            a[i], a[i + 1] = a[i + 1], a[i]
            break
    if a[0] & 1:
        print("NO")
    else:
        print("YES")
