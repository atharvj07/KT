#include<bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  int r1, c1, r2, c2;
  cin >> r1 >> c1 >> r2 >> c2;
  --r1, --r2;
  if (r1 != r2) {
    cout << abs(r1 - r2) + abs(c1 - 1) + abs(c2 - 1);
    return 0;
  }
  cout << abs(c1 - c2);
}
