n = int(input())
s = input()
d = [[0 for i in range(n+2)] for j in range(n+2)]
d[n+1][n+1] = 0
for i in range(n, 0, -1):
    for j in range(n, 0, -1):
        if s[i-1] == 'U':
            d[i][j] = 1 + d[i+1][j]
        else:
            d[i][j] = 1 + d[i-1][j]
for i in range(2, n+1):
    if d[1][i] != 0:
        d[1][i] = d[1][i-1] + 1
    if d[i][1] != 0:
        d[i][1] = d[i-1][1] + 1
for i in range(2, n+1):
    for j in range(2, n+1):
        if d[i][j] != 0:
            d[i][j] = min(d[i][j], d[i-1][j] + 1, d[i][j-1] + 1)
for i in range(1, n+1):
    if d[i][1] == 0:
        print(-1)
    else:
        print(d[i][1])
