#include <bits/stdc++.h>
using namespace std;
int n, sum[105];
vector<int> ans;
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    sum[i] = sum[i - 1] + x;
  }
  if (sum[n] % 2 == 0) {
    cout << 0 << "\n";
    return 0;
  }
  int p = n + 1;
  for (int i = n; i; i--) {
    if (sum[i - 1] > sum[n] - sum[i - 1]) {
      p = i;
    }
  }
  cout << n - p + 1 << "\n";
  for (int i = p; i <= n; i++) ans.push_back(i);
  for (int i = 0; i < ans.size(); i++) cout << ans[i] << " ";
  return 0;
}
