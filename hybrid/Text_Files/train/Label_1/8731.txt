#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<string> a(n);
        for (string &s : a) {
            cin >> s;
        }
        vector<string> b;
        for (int i = 0; i < (1 << m); i++) {
            string s;
            for (int j = 0; j < m; j++) {
                s += (i & (1 << j)) ? '1' : '0';
            }
            bool flag = false;
            for (string &x : a) {
                if (x == s) {
                    flag = true;
                    break;
                }
            }
            if (!flag) {
                b.push_back(s);
            }
        }
        sort(b.begin(), b.end());
        cout << b[(b.size() - 1) / 2] << endl;
    }
    return 0;
}
