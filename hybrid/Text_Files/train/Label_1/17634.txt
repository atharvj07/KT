#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> pii;
typedef pair<pii, int> piii;
#define fi first
#define se second

int main() {
    int n;
    cin >> n;
    vector<pii> soda(n);
    for (int i = 0; i < n; i++) {
        cin >> soda[i].fi;
    }
    for (int i = 0; i < n; i++) {
        cin >> soda[i].se;
    }
    sort(soda.begin(), soda.end());
    int k = 0, t = 0, cur = 0;
    priority_queue<piii, vector<piii>, greater<piii> > pq;
    for (int i = 0; i < n; i++) {
        cur += soda[i].fi;
        pq.push({{soda[i].fi, soda[i].se}, i});
        while (!pq.empty() && cur >= pq.top().fi.se) {
            k++;
            t += pq.top().fi.fi;
            cur -= pq.top().fi.se;
            pq.pop();
        }
    }
    cout << k << ' ' << t << '\n';
    return 0;
}
