import sys
n = int(input())
H = list(map(int, input().split()))
H.append(-1)
res = 0
for i in range(n):
    if H[i] == H[i + 1]:
        continue
    l, r = i, i
    while l >= 0 and H[l] >= H[i]:
        l -= 1
    while r < n and H[r] >= H[i]:
        r += 1
    res += (l + 1) * (r - i)
    res %= 1000000007
print(res)
