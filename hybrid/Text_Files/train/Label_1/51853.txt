import java.util.Arrays;
import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int r = sc.nextInt();
    Edge[] edges = new Edge[m];
    for (int i = 0; i < m; i++) {
      int s = sc.nextInt();
      int t = sc.nextInt();
      int w = sc.nextInt();
      edges[i] = new Edge(s, t, w);
    }
    int[] dist = new int[n];
    Arrays.fill(dist, Integer.MAX_VALUE);
    dist[r] = 0;
    for (int i = 0; i < n - 1; i++) {
      for (Edge e : edges) {
        if (dist[e.s] < Integer.MAX_VALUE && dist[e.t] > dist[e.s] + e.w) {
          dist[e.t] = dist[e.s] + e.w;
        }
      }
    }
    long sum = 0;
    for (int d : dist) {
      if (d < Integer.MAX_VALUE) {
        sum += d;
      }
    }
    System.out.println(sum);
  }

  static class Edge {
    int s, t, w;

    Edge(int s, int t, int w) {
      this.s = s;
      this.t = t;
      this.w = w;
    }
  }
}
