#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin >> n;
    bool changed = false, inversion = false;
    while(n--){
        int old_rating, new_rating;
        cin >> old_rating >> new_rating;
        changed |= old_rating != new_rating;
        inversion |= new_rating < old_rating;
    }
    if(inversion && changed) cout << "rated" << endl;
    else if(!inversion) cout << "unrated" << endl;
    else cout << "maybe" << endl;
    return 0;
}
