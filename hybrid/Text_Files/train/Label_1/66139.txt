n, p = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
ans = []
for x in range(1, n + 1):
    cnt = 0
    for ai in a:
        if x >= ai:
            cnt += 1
        else:
            break
    if cnt * (n - cnt) % p != 0:
        ans.append(x)
print(len(ans))
print(' '.join(map(str, ans)))
