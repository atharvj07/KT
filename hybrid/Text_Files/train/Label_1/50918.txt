import java.util.*;
public class Main {
    static int mod = 1000000007;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt();
        long[] w = new long[n];
        for (int i = 0; i < n; i++) {
            w[i] = in.nextInt();
        }
        Arrays.sort(w);
        if (w[0] == 100 && k < 100) {
            System.out.println(-1);
            System.out.println(0);
            return;
        }
        if (w[n - 1] > k) {
            System.out.println(-1);
            System.out.println(0);
            return;
        }
        long t = 0, ans = 0, ways = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (t + w[i] <= k) {
                t += w[i];
                ways = (ways * 2) % mod;
            } else {
                ans++;
                t = w[i];
                ways = 1;
            }
        }
        System.out.println(ans);
        System.out.println(ways);
    }
}
