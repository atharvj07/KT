#include <iostream>
#include <cstring>
using namespace std;
string a[20000];
int main() {
    string s;
    getline(cin, s);
    int l = s.length();
    bool flag = 0, ans = 1;
    int now = 0;
    for (int i = 0; i < l; i++) {
        if (s[i] != ' ') {
            if (!flag) {
                flag = 1;
                ans = s[i] == '"';
                if (!ans) a[now].push_back(s[i]);
            } else {
                if (!ans) {
                    if (s[i] == '"')
                        flag = 0;
                    else
                        a[now].push_back(s[i]);
                } else {
                    a[now].push_back(s[i]);
                    if (s[i] == '"') flag = 0;
                }
            }
        } else
            flag = 0;
        if (!flag && !a[now].empty()) now++;
    }
    for (int i = 0; i <= now; i++) cout << "<" << a[i] << ">\n";
    return 0;
}
