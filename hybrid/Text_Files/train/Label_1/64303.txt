from collections import Counter

def solve_sequence_problem(n, a):
    a_sorted = sorted(a)
    inc, dec = [], []
    counter_a = Counter(a)
    for i in range(n):
        if i > 0 and a_sorted[i] == a_sorted[i-1]:
            if counter_a[a_sorted[i]] > 1:
                counter_a[a_sorted[i]] -= 2
                dec.append(a_sorted[i])
                inc.insert(0, a_sorted[i])
            else:
                return "NO"
        elif i == 0 or a_sorted[i] > a_sorted[i-1]:
            counter_a[a_sorted[i]] -= 1
            dec.append(a_sorted[i])
        else:
            return "NO"
    if len(inc) + len(dec) == n:
        return f"YES\n{len(inc)}\n{' '.join(map(str, inc))}\n{len(dec)}\n{' '.join(map(str, dec[::-1]))}"
    else:
        return "NO"

n = int(input().strip())
a = list(map(int, input().strip().split()))
print(solve_sequence_problem(n, a))
