import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] input = reader.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int q = Integer.parseInt(input[1]);

        ArrayList<Stack> stacks = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            stacks.add(new Stack());
        }

        for (int i = 0; i < q; i++) {
            String[] query = reader.readLine().split(" ");
            int type = Integer.parseInt(query[0]);
            int stackIndex = Integer.parseInt(query[1]);

            if (type == 0) {
                int value = Integer.parseInt(query[2]);
                stacks.get(stackIndex).push(value);
            } else if (type == 1) {
                Integer top = stacks.get(stackIndex).top();
                if (top != null) {
                    System.out.println(top);
                }
            } else if (type == 2) {
                stacks.get(stackIndex).pop();
            }
        }
    }

    private static class Stack {
        private int[] elements;
        private int size;

        public Stack() {
            elements = new int[10];
            size = 0;
        }

        public void push(int value) {
            if (size == elements.length) {
                elements = Arrays.copyOf(elements, elements.length * 2);
            }

            elements[size++] = value;
        }

        public Integer top() {
            if (size == 0) {
                return null;
            }

            return elements[size - 1];
        }

        public void pop() {
            if (size > 0) {
                size--;
            }
        }
    }
}
