from collections import defaultdict
n=int(input())
crush=list(map(int,input().split()))
seen=[False]*n
cnt=[0]*n
ok=True
for i in range(n):
    x=i
    while not seen[x]:
        seen[x]=True
        x=crush[x]-1
        cnt[x]+=1
        if cnt[x]>1:
            ok=False
    if x!=i:
        ok=False
if ok:
    print(max(cnt))
else:
    print(-1)
