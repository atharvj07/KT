import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        char[] s = br.readLine().toCharArray();
        long ans = 0;
        int[] l = new int[n];
        int[] r = new int[n];
        Stack<Integer> st = new Stack<>();
        // left boundary index for currently processed character
        for (int i = 0; i < n; i++) {
            while (!st.isEmpty() && s[st.peek()] != s[i]) {
                st.pop();
            }
            l[i] = st.isEmpty() ? 0 : st.peek() + 1;
            st.push(i);
        }
        st.clear();
        // right boundary index for currently processed character
        for (int i = n - 1; i >= 0; i--) {
            while (!st.isEmpty() && s[st.peek()] != s[i]) {
                st.pop();
            }
            r[i] = st.isEmpty() ? n - 1 : st.peek() - 1;
            st.push(i);
        }
        for (int i = 0; i < n; i++) {
            // update ans for all substrings that start within [l[i], i] and end within [i, r[i]]
            ans += (long) (i - l[i] + 1) * (r[i] - i + 1);
        }
        System.out.println(ans);
    }
}
