#include <iostream>
#include <cstring>

using namespace std;

string board[300];
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> board[i]; }

    bool flag = false;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (board[i] != board[j]) flag = true;
        }
    }
    for (int i = 0; i < n; i++) {
        if (board[0][i] != board[i][i] || board[0][i] != board[n - 1 - i][i]) {
            flag = true;
        } else if (board[n - 1][i] != board[i][n - 1 - i] ||
                   board[n - 1][i] != board[0][i]) {
            flag = true;
        }
    }
    cout << (flag ? "NO" : "YES");
    return 0;
}
