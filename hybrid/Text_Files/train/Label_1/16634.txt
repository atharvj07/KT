#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll n, a, b;
    cin >> n >> a >> b;
    vector<ll> x(n);
    for (ll &i : x) cin >> i;
    vector<ll> ans(n);
    for (ll i = 0, mx = (a + b - 1) / b, lst = 0; i < n; i++) {
        ll t = (x[i] + lst * b - 1) / (a + b);
        ans[i] = x[i] - t * a;
        lst = t;
        cout << ans[i] << " ";
    }
}
