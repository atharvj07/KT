#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, sum = 0;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; sum += a[i]; }
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 0; i < n - 1; i++) { ans += a[n - 1] - a[i]; }
    cout << ans << endl;
}
