f = list(map(int, input().split()))
s = list(input())
t = list(s)
n = len(s)
ans = [0] * n
l, r = 0, 0
mx = 0
for i in range(n):
    if s[i] == t[i]: continue
    elif s[i] > t[i]:
        for j in range(i, n):
            if s[j] <= t[j]: break
            ans[j] = 1
            mx = j
    else:
        l = i
        r = i
        while l > 0 and r + 1 < n and s[l - 1] <= t[l - 1] and s[r + 1] >= t[r + 1]:
            l -= 1
            r += 1
        if l == 0 and r == n - 1:
            for j in range(n):
                s[j] = t[j]
                ans[j] = 1
            mx = n - 1
            break
        for j in range(l, r + 1):
            ans[j] = 1
        mx = r
    for j in range(n):
        s[j] = str(f[int(s[j])])
for i in range(n):
    if ans[i] == 0:
        s[i] = t[i]
print(''.join(s))
