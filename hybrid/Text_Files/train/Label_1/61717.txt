import sys

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
edges = []
for _ in range(n - 1):
    u, v = map(int, input().split())
    edges.append((u, v))
graph = [[] for _ in range(n + 1)]
for u, v in edges:
    graph[u].append(v)
    graph[v].append(u)

vis = [False] * (n + 1)
res = [0] * (n + 1)


def dfs(node, par):
    vis[node] = True
    cnt = a[node - 1]
    for u in graph[node]:
        if vis[u] or u == par:
            continue
        dfs(u, node)
        cnt += res[u]
    res[node] = cnt - sum(a) + a[node - 1] * n


dfs(1, 1)
for i in res[1:]:
    print(i)
