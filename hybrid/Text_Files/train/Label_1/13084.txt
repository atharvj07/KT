#include <bits/stdc++.h>
using namespace std;

#define ll long long

map<ll, ll> sum;
stack<ll> st;

int main() {
  ll n;
  cin >> n;
  ll arr[n];
  for (ll i = 0; i < n; i++) {
    cin >> arr[i];
  }
  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    while (!st.empty() && arr[st.top()] < arr[i]) {
      ans += sum[st.top()];
      ll idx = st.top();
      st.pop();
      if (!st.empty()) {
        ans += sum[st.top()];
        sum[st.top()] += sum[idx];
        st.push(idx);
      }
    }
    sum[i] = arr[i];
    st.push(i);
  }
  cout << ans << endl;
  return 0;
}
