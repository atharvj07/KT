#include <cstdio>
#include <cstring>
#include <set>
using namespace std;
char a[1000], b[200];
const int maxs = 100000;
bool bkt[maxs];
int main() {
  set<int> s;
  int n;
  scanf(" %d", &n);
  s.clear();
  memset(bkt, 0, sizeof(bkt));
  for (int i = 0; i < n; ++i) {
    scanf(" %s", a);
    for (int j = 0; a[j] != 0; ++j) {
      b[j] = (a[j] | ' ');
    }
    for (int j = 0; b[j] != 0; ++j) {
      if (bkt[((int)b[j]) + maxs / 2]) {
        puts("YES");
        s.insert(i);
        break;
      }
      bkt[((int)b[j]) + maxs / 2] = 1;
    }
    if (s.find(i) == s.end()) puts("NO");
  }
}
