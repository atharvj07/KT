#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int dp[1005][1005];

int main() {
  int n, k;
  cin >> n >> k;
  vector<vector<int>> p(k, vector<int>(n));
  for (int i = 0; i < k; i++) {
    for (int j = 0; j < n; j++) {
      cin >> p[i][j];
    }
  }
  int ans = 0;
  for (int i = 0; i < k; i++) {
    for (int j = 0; j < k; j++) {
      if (i == j)
        dp[i][j] = 1;
      else {
        for (int l = 0; l < n; l++) {
          for (int m = 0; m < n; m++) {
            if (p[i][l] == p[j][m]) {
              dp[i][j] = max(dp[i][j], dp[i][m] + 1);
            }
          }
        }
      }
      ans = max(ans, dp[i][j]);
    }
  }
  cout << ans << endl;
  return 0;
}
