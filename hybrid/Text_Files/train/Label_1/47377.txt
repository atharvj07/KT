import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class GridSolution {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int[] p = new int[a - 1];
        for (int i = 0; i < a - 1; i++) {
            p[i] = scanner.nextInt();
        }
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = scanner.nextInt();
        }
        int b = scanner.nextInt();
        int[] q = new int[b - 1];
        for (int i = 0; i < b - 1; i++) {
            q[i] = scanner.nextInt();
        }
        int[] y = new int[n];
        for (int i = 0; i < n; i++) {
            y[i] = scanner.nextInt();
        }

        List<Integer>[] mainGrid = buildGrid(a, p);
        List<Integer>[] reserveGrid = buildGrid(b, q);

        int maxWiresToRemove = 0;
        for (int i = 0; i < n; i++) {
            int mainGridPath = findPathLength(mainGrid, 1, x[i]);
            int reserveGridPath = findPathLength(reserveGrid, 1, y[i]);
            maxWiresToRemove += Math.max(mainGridPath, reserveGridPath) - 1;
        }

        System.out.println(maxWiresToRemove);
    }

    private static List<Integer>[] buildGrid(int n, int[] p) {
        List<Integer>[] grid = new List[n];
        for (int i = 0; i < n; i++) {
            grid[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            grid[p[i] - 1].add(i + 2);
        }
        return grid;
    }

    private static int findPathLength(List<Integer>[] grid, int start, int end) {
        int[] pathLength = new int[grid.length];
        Arrays.fill(pathLength, -1);
        dfs(grid, start, pathLength, 0);
        return pathLength[end - 1];
    }

    private static void dfs(List<Integer>[] grid, int node, int[] pathLength, int currentLength) {
        pathLength[node - 1] = currentLength;
        for (int child : grid[node - 1]) {
            if (pathLength[child - 1] == -1) {
                dfs(grid, child, pathLength, currentLength + 1);
            }
        }
    }
}
