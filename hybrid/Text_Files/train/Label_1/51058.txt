import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] cups = new int[n];
        for (int i = 0; i < n; i++) {
            cups[i] = sc.nextInt();
        }
        int sum = 0;
        for (int cup : cups) {
            sum += cup;
        }
        if (sum % n != 0) {
            System.out.println("Unrecoverable configuration.");
        } else {
            int avg = sum / n;
            int from = -1, to = -1, diff = -1;
            for (int i = 0; i < n; i++) {
                if (cups[i] < avg) {
                    from = i + 1;
                    diff = avg - cups[i];
                } else if (cups[i] > avg) {
                    to = i + 1;
                    if (diff == -1 || diff > cups[i] - avg) {
                        diff = cups[i] - avg;
                    }
                }
            }
            if (from == -1 || to == -1 || diff == -1) {
                System.out.println("Exemplary pages.");
            } else {
                System.out.println(diff + " ml. from cup #" + from + " to cup #" + to + ".");
            }
        }
    }
}
