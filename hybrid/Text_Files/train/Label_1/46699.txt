import java.util.*;
import static java.lang.Math.*;

class Main {
    static final int MOD = (int)1e9 + 7;
    static int n, dp[][][];
    static String s;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        n = input.nextInt();
        s = input.next();
        dp = new int[n + 2][n + 2][2];
        for (int[][] ldp : dp) {
            for (int[] rdp : ldp) {
                Arrays.fill(rdp, -1);
            }
        }
        System.out.println(go(1, n, 1));
    }

    static int go(int l, int r, int pre) {
        if (dp[l][r][pre] != -1) {
            return dp[l][r][pre];
        }
        int ans = 0;
        if (l > r) {
            return dp[l][r][pre] = 1;
        }
        if (s.charAt(l - 1) == '0' && s.charAt(r - 1) == '0') {
            return dp[l][r][pre] = 0;
        }
        if (pre == 0) {
            ans = (ans + go(l, r - 1, (s.charAt(r - 1) - '0') % 2)) % MOD;
        }
        if (pre == 1) {
            if (s.charAt(l - 1) != s.charAt(r - 1)) {
                ans = (ans + go(l + 1, r - 1, (s.charAt(l - 1) - '0') % 2)) % MOD;
            }
            ans = (ans + go(l, r - 1, pre)) % MOD;
            ans = (ans + go(l + 1, r, pre)) % MOD;
        }
        return dp[l][r][pre] = ans;
    }
}
