#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
  int q;
  cin >> q;
  while (q--) {
    int n, k;
    cin >> n >> k;
    if (k * 1ll * (k + 1) / 2 > n || (n - (k + 1) * 1ll * k / 2) % 2)
      cout << "NO\n";
    else {
      cout << "YES\n";
      int sum = n + 1;
      for (int i = 1; i < k; i++) { cout << sum << " "; sum += 2; }
      cout << sum << "\n";
    }
  }
}
