t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    answer = n
    for i in range(1, int(n**0.5) + 1):
        if n % i == 0:
            if i <= k:
                answer = min(answer, n // i)
            if n // i <= k:
                answer = min(answer, i)
    print(answer)
