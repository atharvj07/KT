N, A, B = map(int, input().split())
if N % (abs(A - B) + 1):
    print(-1)
else:
    X = [0] * N
    L1 = []
    L2 = []
    L3 = []
    for i in range(N):
        if (i + B) % (A - B) == 0:
            L1.append(i + 1)
        elif (i + A) % (B - A) == 0:
            L2.append(i + 1)
        else:
            L3.append(i + 1)
    ans = []
    while L2 and L3:
        ans.append(L2.pop())
        ans.append(L3.pop())
    if L3:
        k = len(L3) // (A - B)
        for i in range(k):
            for j in range(A - B):
                ans.append(L3.pop())
            if L2:
                ans.append(L2.pop())
        if L3:
            print(-1)
            exit()
    while L1 and L3:
        ans.append(L1.pop())
        ans.append(L3.pop())
    if L1:
        k = len(L1) // (B - A)
        for i in range(k):
            for j in range(B - A):
                ans.append(L1.pop())
            if L2:
                ans.append(L2.pop())
        if L1:
            print(-1)
            exit()
    while L1 and L2:
        ans.append(L1.pop())
        ans.append(L2.pop())
    if L2:
        k = len(L2) // (A - B)
        for i in range(k):
            for j in range(A - B):
                ans.append(L2.pop())
            if L1:
                ans.append(L1.pop())
        if L2:
            print(-1)
            exit()
    ans.extend(L1)
    ans.extend(L2)
    ans.extend(L3)
    print(*ans)
