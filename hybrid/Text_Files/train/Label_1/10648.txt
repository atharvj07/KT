#include <bits/stdc++.h>
using namespace std;

const int mod = 998244353;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<int> odd_cnt(n);
    for (int i = 0; i < n; i++) { odd_cnt[i] = (a[i] % 2); }
    if (accumulate(odd_cnt.begin(), odd_cnt.end(), 0) > 1) {
      cout << 0 << "\n";
      continue;
    }
    long long ans = 1;
    for (auto x : a) { ans = ans * x % mod; }
    cout << ans << "\n";
  }
  return 0;
}
