#include <iostream>
#include <map>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    map<int, vector<int>> indices;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        indices[a[i]].push_back(i);
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            int sum = a[i] + a[j];
            if (indices[sum].size() >= 2) {
                cout << "YES" << endl;
                cout << i + 1 << " " << j + 1 << " ";
                for (int idx : indices[sum]) {
                    if (idx != i && idx != j) {
                        cout << idx + 1 << " ";
                        break;
                    }
                }
                for (int idx : indices[sum]) {
                    if (idx != i && idx != j) {
                        cout << idx + 1 << endl;
                        return 0;
                    }
                }
            }
        }
    }
    cout << "NO" << endl;
    return 0;
}
