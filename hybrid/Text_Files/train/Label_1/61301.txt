from collections import defaultdict
def dfs(x, p, g, col):
    col[x] = 1
    for y in g[x]:
        if col[y] == 0:
            if dfs(y, x, g, col):
                return True
        elif y != p:
            return True
    return False
n, m = map(int, input().split())
g = defaultdict(list)
for i in range(m):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)
t, w = n, 0
col = [0] * (n+1)
for i in range(1, n+1):
    if col[i] == 0:
        if dfs(i, -1, g, col):
            t = 1
            w = 1
            break
        else:
            t -= 2
            w += 1
print(t, w)
