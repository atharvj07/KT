#include <iostream>
#include <climits>
using namespace std;
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = INT_MAX;
  for (int i = 0; i < n; i++) {
    if (a[i] > 0 && a[i] <= k) ans = min(ans, abs(i - m) * 10);
  }
  cout << ans << endl;
  return 0;
}
