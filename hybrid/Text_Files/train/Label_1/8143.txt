#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

vector<int> a;
int pre[1000005];

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  a.resize(n);
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  int ans = n;
  for (int i = 0; i < n; ++i) {
    if (i > 0) {
      pre[a[i]] = pre[a[i - 1]];
    }
    ++pre[a[i]];
    if (i >= m - 1) {
      int cnt = 0;
      for (int j = a[i] - m + 1; j <= a[i]; ++j) {
        cnt += pre[j];
      }
      if (cnt >= k) {
        ans = min(ans, i);
      }
    }
  }
  cout << ans << endl;
  return 0;
}
