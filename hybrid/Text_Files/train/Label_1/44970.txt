import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            int minDis = n;
            for (int j = 0; j < n; j++) {
                if (a[j] == b[i]) {
                    minDis = Math.min(minDis, Math.abs(j - i));
                }
            }
            ans[i] = minDis;
        }
        for (int i : ans) {
            System.out.println(i);
        }
    }
}
