import java.util.*;
public class Main{
    public static void main(String[] args) {
        Scanner in=new Scanner(System.in);
        int n=in.nextInt(),m=in.nextInt(),k=in.nextInt();
        char[][]map=new char[n][m];
        int[][]cnt=new int[n][m];
        boolean[][]vis=new boolean[n][m];
        for(int i=0;i<n;i++)map[i]=in.next().toCharArray();
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if((j==0||map[i][j-1]=='.')&&(map[i][j]=='R'||map[i][j]=='U')){
                    int nowx=i,nowy=j;
                    while(nowx>=0&&nowy<m){
                        if(map[nowx][nowy]=='.')cnt[i][j]++;
                        else{
                            vis[nowx][nowy]=true;
                            break;
                        }
                        if(map[nowx][nowy]=='U')nowx--;
                        else nowy++;
                    }
                }
                if((i==0||map[i-1][j]=='.')&&(map[i][j]=='L'||map[i][j]=='D')){
                    int nowx=i,nowy=j;
                    while(nowx<n&&nowy>=0){
                        if(map[nowx][nowy]=='.')cnt[i][j]++;
                        else{
                            vis[nowx][nowy]=true;
                            break;
                        }
                        if(map[nowx][nowy]=='L')nowy--;
                        else nowx++;
                    }
                }
            }
        }
        for(int i=0;i<k;i++){
            for(int j=0;j<n;j++){
                for(int l=0;l<m;l++){
                    if(vis[j][l]&&cnt[j][l]>0)cnt[j][l]--;
                }
            }
        }
        for(int i=0;i<m;i++){
            System.out.print(cnt[0][i]+" ");
        }
    }
}
