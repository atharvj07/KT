import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        long[] a = new long[n + 1];
        for (int i = 0; i < m; i++) {
            int t = in.nextInt();
            int l = in.nextInt();
            int r = in.nextInt();
            if (t == 2) {
                int v = in.nextInt();
                for (int j = l; j <= r; j++) a[j] = Math.max(a[j], v);
            } else {
                int d = in.nextInt();
                for (int j = l; j <= r; j++) a[j] += d;
            }
        }
        System.out.println("YES");
        for (int i = 1; i <= n; i++) System.out.print(a[i] + " ");
    }
}
