#include <bits/stdc++.h>
using i64 = long long;
using namespace std;
#define rep(i, a, b) for (int i = (a); i < (b); ++i)

struct Run {
  int l, r;
  int flp;

  int num() const { return r - l + 1; }
};

int main() {
  string s, t;
  cin >> s >> t;
  int ns = s.size(), nt = t.size();
  if (nt % ns) {
    cout << -1 << endl;
    return 0;
  }
  vector<vector<int>> dp(nt + 1, vector<int>(ns + 1, 1e9));
  dp[0][0] = 0;
  rep(i, 0, nt) rep(j, 0, ns) if (dp[i][j] < 1e9) {
    rep(k, j, ns) {
      int nj = (k + 1) % ns;
      if (s[k] == t[i]) dp[i + 1][nj] = min(dp[i + 1][nj], dp[i][j] + 1);
    }
  }

  if (dp[nt][0] >= 1e9) {
    cout << -1 << endl;
  } else {
    cout << dp[nt][0] << endl;
    int i = nt, j = 0;
    vector<Run> ans;
    while (i) {
      int prv = dp[i][j] - 1;
      rep(k, j, ns) {
        int nj = (k + 1) % ns;
        if (dp[i][nj] == prv && s[k] == t[i - 1]) {
          int len = k - j + 1;
          if (len % 2) {
            ans.push_back({j, k, 1});
          } else {
            ans.push_back({k, j, 0});
          }
          i -= len;
          j = nj;
          break;
        }
      }
    }
    reverse(ans.begin(), ans.end());
    for (const auto &run : ans)
      cout << run.l + 1 << ' ' << run.r + 1 << endl;
  }
  return 0;
}
