def find_swaps(arr):
    swaps = []
    for i in range(len(arr)):
        for j in range(i+1, len(arr)):
            if arr[i] > arr[j]:
                swaps.append((i+1, j+1))
    arr_sorted = sorted(arr)
    for swap in swaps:
        u, v = swap
        arr[u-1], arr[v-1] = arr[v-1], arr[u-1]
        if arr == arr_sorted:
            return swaps[:swaps.index(swap)+1]
    return -1

n = int(input())
arr = list(map(int, input().split()))

swaps = find_swaps(arr)
if swaps == -1:
    print(-1)
else:
    print(len(swaps))
    for swap in swaps:
        print(swap[0], swap[1])
