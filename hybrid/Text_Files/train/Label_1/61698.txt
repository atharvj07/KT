def Create(n):
    f = [[0] * n for i in range(n)]
    diag = [1, n**2]
    for i in range(n):
        f[i][i] = diag[0]
        diag[0] += 2
    for i in range(n):
        f[i][n-i-1] = diag[1]
        diag[1] -= 2
    return f
def f(n):
    a = Create(n)
    l = n**2
    for i in range(n):
        for j in range(n):
            if not a[i][j]:
                for k in range(1, l+1):
                    if k not in a[i] and k not in [a[t][j] for t in range(n)]:
                        a[i][j] = k
                        break
    for i in range(n):
        print(*a[i])
n = int(input())
f(n)
