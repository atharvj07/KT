#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; ++i)
        std::cin >> a[i];
    std::sort(a.begin(), a.end(), std::greater<int>());
    int ans = 0, dir = 0;  // dir = 0 means vertical, dir = 1 means horizontal
    for (int i = 0; i < n; ++i) {
        if ((i % 2 == 0 && dir == 0) || (i % 2 == 1 && dir == 1))
            ans += a[i] * a[i];
        else
            dir ^= 1;
    }
    std::cout << ans << "\n";
}
