import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[][] t = new int[3][3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                t[i][j] = in.nextInt();
            }
        }
        int n = in.nextInt();
        System.out.println(solve(t, n));
    }

    public static int solve(int[][] t, int n) {
        int[][] dp = new int[3][n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < 3; j++) {
                dp[j][i] = Integer.MAX_VALUE;
                for (int k = 0; k < 3; k++) {
                    if (j == k) {
                        continue;
                    }
                    dp[j][i] = Math.min(dp[j][i], dp[k][i - 1] + t[k][j]);
                }
            }
        }
        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < 3; i++) {
            ans = Math.min(ans, dp[i][n]);
        }
        return ans;
    }
}
