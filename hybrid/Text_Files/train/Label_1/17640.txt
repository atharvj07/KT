#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> sizes(n);
  for (int i = 0; i < n; i++) {
    cin >> sizes[i];
  }
  sort(sizes.begin(), sizes.end());
  int visible = 0;
  int current = 0;
  for (int i = 0; i < n; i++) {
    if (sizes[i] >= current) {
      current = sizes[i] * 2;
      visible++;
    }
  }
  cout << visible << endl;
  return 0;
}
