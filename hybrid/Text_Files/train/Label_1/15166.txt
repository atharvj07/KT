#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int y;
  cin >> y;
  y++;
  while(true) {
    vector<int> digits;
    int temp = y;
    while(temp > 0) {
      digits.push_back(temp % 10);
      temp /= 10;
    }
    sort(digits.begin(), digits.end());
    bool distinct = true;
    for(int i = 0; i < digits.size() - 1; i++) {
      if(digits[i] == digits[i + 1]) {
        distinct = false;
        break;
      }
    }
    if(distinct) {
      cout << y << endl;
      return 0;
    }
    y++;
  }
  return 0;
}
