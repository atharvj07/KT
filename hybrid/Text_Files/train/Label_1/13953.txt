#include <bits/stdc++.h>
using namespace std;

const int mod = 1e9+7;
const int maxN = 2e5 + 10;
int n, k;
long long dp[maxN][10];
int w[maxN];

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++)
        cin >> w[i];
    dp[0][0] = 1;
    for (int i = 1; i <= n; i++) {
        for (int sum = k - 1; sum >= 0; sum--) {
            for (int p = 0; p < k; p++) {
                // put in cell
                dp[i][sum + 1] = (dp[i][sum + 1] + dp[i - 1][sum]) % mod;
                // put in current cell
                dp[i][sum] = (dp[i][sum] + 1LL * w[i] * dp[i - 1][sum] % mod) % mod;
            }
        }
    }
    cout << dp[n][k] << '\n';
    return 0;
}
