t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a, b, c = map(int, input().strip().split())
    s = input().strip()

    wins_needed = (n + 1) // 2
    wins = 0

    alice_sequence = [''] * n
    used = [False] * n

    # Calculate wins and prepare Alice's sequence
    for i, hand in enumerate(s):
        if hand == 'R' and b > 0:
            b -= 1
            wins += 1
            alice_sequence[i] = 'P'
            used[i] = True
        elif hand == 'P' and c > 0:
            c -= 1
            wins += 1
            alice_sequence[i] = 'S'
            used[i] = True
        elif hand == 'S' and a > 0:
            a -= 1
            wins += 1
            alice_sequence[i] = 'R'
            used[i] = True

    if wins < wins_needed:
        print("NO")
    else:
        print("YES")
        for i in range(n):
            if not used[i]:
                if a > 0:
                    alice_sequence[i] = 'R'
                    a -= 1
                elif b > 0:
                    alice_sequence[i] = 'P'
                    b -= 1
                elif c > 0:
                    alice_sequence[i] = 'S'
                    c -= 1
        print(''.join(alice_sequence))
