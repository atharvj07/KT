import sys
input = sys.stdin.readline
n = int(input())
*p, = [tuple(map(int, input().split())) for i in range(n)]
for i in range(n):
    x1, y1 = p[i]
    for j in range(n):
        if i == j: continue
        x2, y2 = p[j]
        if (x1 * y2 - x2 * y1) * (x1 * p[0][1] - x1 * p[-1][1] + y1 * p[-1][0] - y1 * p[0][0]) < 0:
            print('Yes')
            exit()
print('No')
