a, b, c = map(int, input().split())
if a + b + c == 2:
    print(0, 0, 0)
elif a + b + c == 3:
    if a == 1:
        print(0, 1, 0)
    elif b == 1:
        print(0, 0, 1)
    else:
        print(1, 0, 0)
elif a + b + c == 4:
    if a == 2:
        print(0, 2, 0)
    elif b == 2:
        print(0, 0, 2)
    else:
        print(1, 1, 0)
else:
    print("Impossible")
