n=int(input())
c=[[int(x) for x in input().split()] for _ in range(n)]
ans=0
def _is_possible(x):
    if x==0:return False
    q=[]
    for i in range(n):
        for j in range(6):
            if c[i][j]==x:
                q.append((i,j))
    if len(q)==0:return False
    vis_c=[False for _ in range(n)]
    vis_f=[False for _ in range(6)]
    def dfs(msk):
        if msk==(1<<n)-1:
            return True
        for k in q:
            i,j=k
            if vis_c[i] or vis_f[j]:continue
            vis_c[i],vis_f[j]=True,True
            if dfs(msk|(1<<i)):
                return True
            vis_c[i],vis_f[j]=False,False
        return False
    return dfs(0)
for i in range(9,-1,-1):
    if _is_possible(i):
        ans=i
        break
print(ans)
