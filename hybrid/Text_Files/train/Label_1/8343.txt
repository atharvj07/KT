#include <bits/stdc++.h>
#define int long long
using namespace std;
const int MAXN = 1e3 + 10;
int w[MAXN], dp[MAXN][MAXN], n, k;
string s;
int func(int pos,int taken){
    if(pos == n){
        if(taken == k) return 0;
        return -1e17;
    }
    int &ret = dp[pos][taken];
    if(ret != -1) return ret;
    ret = max(0LL,func(pos+1,taken)+w[s[pos]-'a']*(pos+1));
    if(taken + 1 <= k) ret = max(ret,func(pos,taken+1)+w[s[pos]-'a']*(pos+1));
    for(char c = 'a'; c<='z'; c++){
        if(taken + 1 <= k) ret = max(ret,func(pos,taken+1)+w[c-'a']*(pos+1));
    }
    return ret;
}
int32_t main(){
    cin>>s>>k;
    n = s.size();
    memset(dp,-1,sizeof dp);
    for(int i = 0; i<26; i++) cin>>w[i];
    cout<<func(0,0)<<endl;
    return 0;
}
