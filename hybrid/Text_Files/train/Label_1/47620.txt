import java.util.*;

public class FoolRoads {
    static class Road {
        int u, v;

        Road(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Road> roads = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            roads.add(new Road(u, v));
        }
        int k = scanner.nextInt();
        List<Pair> pairs = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            pairs.add(new Pair(a, b));
        }

        int[] roadCounts = new int[n - 1];
        for (Pair pair : pairs) {
            int pathStart = pair.a;
            int pathEnd = pair.b;
            for (Road road : roads) {
                if ((road.u == pathStart && road.v == pathEnd) || (road.u == pathEnd && road.v == pathStart)) {
                    roadCounts[roads.indexOf(road)]++;
                    break;
                }
            }
        }

        for (int count : roadCounts) {
            System.out.print(count + " ");
        }
    }

    static class Pair {
        int a, b;

        Pair(int a, int b) {
            this.a = a;
            this.b = b;
        }
    }
}
