#include <bits/stdc++.h>
using namespace std;
int solve() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  int p = n - 1, ans = 0;
  while (p >= 0) {
    if (a[p] + a[0] > k) break;
    p--;
    ans++;
  }
  return ans;
}
int main() {
  int t;
  cin >> t;
  while (t--) cout << solve() << '\n';
}
