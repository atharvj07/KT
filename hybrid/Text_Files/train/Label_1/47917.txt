import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long[] xor = new long[n + 1];
        for (int i = 0; i < n; i++) {
            xor[i + 1] = xor[i] ^ a[i];
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (xor[j] == xor[i]) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
