#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int main() {
  int n;
  cin >> n;
  int ans = 0;
  if (n < 3) {
    cout << 0 << endl;
    return 0;
  }
  if (n & 1) {
    ans += (n + 1) / 2;
  } else {
    ans += n / 2;
  }
  ans = ans % mod;
  int mult = 2;
  for (int i = n / 2 + 1; i <= n; i++) {
    ans = (ans + mult) % mod;
    mult = (2LL * mult) % mod;
  }
  cout << ans << endl;
}
