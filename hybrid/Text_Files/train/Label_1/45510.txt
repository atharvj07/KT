import java.util.*;

public class Main {
    static int n, m;
    static char[][] grid;
    static int[][] pop;
    static int[][][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        grid = new char[n][m];
        pop = new int[n][m];
        dp = new int[n][m][3];
        for (int i = 0; i < n; i++) {
            grid[i] = sc.next().toCharArray();
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                pop[i][j] = (grid[i][j] == '#') ? -1 : 0;
            }
        }
        int ans = 0;
        int x = 0, y = 0, t = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                for (int k = 0; k < 3; k++) {
                    dp[i][j][k] = -1;
                    if (valid(i, j) && (k == 0 || isBlue(i, j))) {
                        dp[i][j][k] = Math.max(dp[i][j][k], dp(i + 1, j, k) + pop[i][j]);
                        dp[i][j][k] = Math.max(dp[i][j][k], dp(i, j + 1, k) + pop[i][j]);
                        if (k == 0 || isRed(i, j)) {
                            dp[i][j][k] = Math.max(dp[i][j][k], dp(i + 1, j, 1) + 2 * pop[i][j]);
                            dp[i][j][k] = Math.max(dp[i][j][k], dp(i, j + 1, 1) + 2 * pop[i][j]);
                            dp[i][j][k] = Math.max(dp[i][j][k], dp(i + 1, j, 2) + 2 * pop[i][j]);
                            dp[i][j][k] = Math.max(dp[i][j][k], dp(i, j + 1, 2) + 2 * pop[i][j]);
                        }
                    }
                    if (dp[i][j][k] > ans) {
                        ans = dp[i][j][k];
                        x = i;
                        y = j;
                        t = k;
                    }
                }
            }
        }
        System.out.println(ans);
        while (x != 0 || y != 0 || t != 0) {
            int nx = x - (dp[x - 1][y][t] == ans - pop[x - 1][y] ? 1 : 0);
            int ny = y - (dp[x][y - 1][t] == ans - pop[x][y - 1] ? 1 : 0);
            if (t == 0 || isBlue(nx, ny)) {
                if (dp[nx][ny][0] == ans - pop[nx][ny]) {
                    System.out.println("B " + (nx + 1) + " " + (ny + 1));
                    ans -= pop[nx][ny];
                    x = nx;
                    y = ny;
                    t = 0;
                    continue;
                }
                if (dp[nx][ny][1] == ans - 2 * pop[nx][ny]) {
                    System.out.println("R " + (nx + 1) + " " + (ny + 1));
                    ans -= 2 * pop[nx][ny];
                    x = nx;
                    y = ny;
                    t = 1;
                    continue;
                }
                if (dp[nx][ny][2] == ans - 2 * pop[nx][ny]) {
                    System.out.println("R " + (nx + 1) + " " + (ny + 1));
                    ans -= 2 * pop[nx][ny];
                    x = nx;
                    y = ny;
                    t = 2;
                    continue;
                }
            }
            if (t == 0 || isRed(nx, ny)) {
                if (dp[nx][ny][1] == ans - 2 * pop[nx][ny]) {
                    System.out.println("R " + (nx + 1) + " " + (ny + 1));
                    ans -= 2 * pop[nx][ny];
                    x = nx;
                    y = ny;
                    t = 1;
                    continue;
                }
                if (dp[nx][ny][2] == ans - 2 * pop[nx][ny]) {
                    System.out.println("R " + (nx + 1) + " " + (ny + 1));
                    ans -= 2 * pop[nx][ny];
                    x = nx;
                    y = ny;
                    t = 2;
                    continue;
                }
            }
            System.out.println("D " + (x + 1) + " " + (y + 1));
            x += nx - x;
            y += ny - y;
            t = 0;
        }
    }

    static boolean valid(int x, int y) {
        return x >= 0 && x < n && y >= 0 && y < m;
    }

    static int dp(int x, int y, int t) {
        if (!valid(x, y)) {
            return 0;
        }
        if (dp[x][y][t] != -1) {
            return dp[x][y][t];
        }
        int ans = 0;
        if (valid(x, y) && (t == 0 || isBlue(x, y))) {
            ans = Math.max(ans, dp(x + 1, y, t) + pop[x][y]);
            ans = Math.max(ans, dp(x, y + 1, t) + pop[x][y]);
            if (t == 0 || isRed(x, y)) {
                ans = Math.max(ans, dp(x + 1, y, 1) + 2 * pop[x][y]);
                ans = Math.max(ans, dp(x, y + 1, 1) + 2 * pop[x][y]);
                ans = Math.max(ans, dp(x + 1, y, 2) + 2 * pop[x][y]);
                ans = Math.max(ans, dp(x, y + 1, 2) + 2 * pop[x][y]);
            }
        }
        return dp[x][y][t] = ans;
    }

    static boolean isBlue(int x, int y) {
        return valid(x - 1, y) && grid[x - 1][y] == '.' || valid(x + 1, y) && grid[x + 1][y] == '.' || valid(x, y - 1) && grid[x][y - 1] == '.' || valid(x, y + 1) && grid[x][y + 1] == '.';
    }

    static boolean isRed(int x, int y) {
        return isBlue(x, y) && (valid(x - 2, y) && grid[x - 2][y] == '.' || valid(x + 2, y) && grid[x + 2][y] == '.' || valid(x, y - 2) && grid[x][y - 2] == '.' || valid(x, y + 2) && grid[x][y + 2] == '.');
    }
}
