def calculate_difference(level1, level2, n, m):
    diff = 0
    for i in range(n):
        for j in range(m):
            if level1[i][j] != level2[i][j]:
                diff += 1
    return diff

def find_min_traffic(n, m, k, w, levels):
    cost = [[0 for _ in range(k)] for _ in range(k)]
    for i in range(k):
        for j in range(i + 1, k):
            cost[i][j] = cost[j][i] = calculate_difference(levels[i], levels[j], n, m) * w
    
    dp = [float('inf')] * k
    dp[0] = n * m
    parent = [-1] * k
    for i in range(1, k):
        dp[i] = n * m
        for j in range(i):
            if dp[j] + cost[j][i] < dp[i]:
                dp[i] = dp[j] + cost[j][i]
                parent[i] = j
    
    min_cost = sum(n*m if x == n*m else cost[parent[i]][i] for i, x in enumerate(dp))
    seq = []
    used = [False] * k
    for i in range(k - 1, -1, -1):
        if not used[i]:
            cur = i
            while cur != -1:
                seq.append((cur + 1, 0 if parent[cur] == -1 else parent[cur] + 1))
                used[cur] = True
                cur = parent[cur]
    seq.reverse()
    
    return min_cost, seq

# Input reading
n, m, k, w = map(int, input().split())
levels = [[input() for _ in range(n)] for _ in range(k)]

min_traffic, sequence = find_min_traffic(n, m, k, w, levels)

print(min_traffic)
for x, y in sequence:
    print(x, y)
