import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long l = sc.nextLong();
    long r = sc.nextLong();
    long ans = r - l + 1;
    for (long i = 0; (1 << i) < 10; i++) {
      if (((l >> i) & 1) == 0 && ((r >> i) & 1) != 0) {
        ans--;
      }
    }
    System.out.println(ans);
  }
}
