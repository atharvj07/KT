for _ in range(int(input())):
    n,k=[int(x) for x in input().split()]
    if k==1:
        print("YES",n)
    elif k>1 and n%2==0 and n>=k and (n-k)%2==0:
        print("YES",k//2,k//2,(n-k))
    elif n%2==0 and k>1:
        print("NO")
    elif k>1 and n%2 and n>=k+1 and (n-k-1)%2==0:
        print("YES",1,k//2,k//2,(n-k-1))
    elif n%2 and k>1:
        print("NO")
