#include<bits/stdc++.h>
using namespace std;
int32_t main() {
    int n, p, q, r;
    cin >> n >> p >> q >> r;
    vector<long long> a(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        a[i] += a[i - 1];
    }
    long long ans = LLONG_MIN;
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j++) {
            ans = max(ans, p * a[i - 1] + q * (a[j] - a[i - 1]) + r * (a[n] - a[j]));
        }
    }
    cout << ans << "\n";
    return 0;
}
