#include <vector>
#include <iostream>
#include <unordered_map>

int main() {
  int q, x;
  std::cin >> q >> x;
  std::vector<int> v;
  std::unordered_map<int, int> m;
  for (int i = 0; i < q; ++i) {
    int y;
    std::cin >> y;
    v.push_back(y);
    m[y]++;
  }
  for (int y : v) {
    m[y]--;
    int mex = 0;
    while (m.count(mex)) mex++;
    int v1 = y - mex, v2 = y + x - mex;
    if (v1 >= 0 && !m.count(v1))
      v1 = -1;
    if (v2 >= 0 && !m.count(v2))
      v2 = -1;
    int t = std::max(v1, v2);
    if (t >= 0) {
      m[t]++;
      y = t;
    }
    m[y]++;
    std::cout << mex << std::endl;
  }
  return 0;
}
