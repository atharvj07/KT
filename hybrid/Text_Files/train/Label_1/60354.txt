import sys

n = int(sys.stdin.readline())
arr = list(map(int, sys.stdin.readline().split()))
odd_idx, even_idx = [], []
for i in range(2 * n):
    if arr[i] % 2 == 1:
        odd_idx.append(i)
    else:
        even_idx.append(i)
ans = len(odd_idx) // 2 + len(even_idx) // 2
if len(odd_idx) % 2 == 1 or len(even_idx) % 2 == 1:
    ans += 1
if ans > n:
    ans = -1
print(ans)
