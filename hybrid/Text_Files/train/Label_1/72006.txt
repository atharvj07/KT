from collections import deque

for _ in range(int(input())):
    s, n = map(int, input().split())
    res = deque()
    while n > 0:
        if n == 1:
            res.appendleft(s)
            n -= 1
        elif n == 2:
            if s - 1 >= 10:
                res.appendleft(s - 1)
                res.appendleft(9)
            else:
                res.appendleft(s)
                res.appendleft(0)
            s -= 10
            n -= 2
        else:
            t = (s - 10*(n-2)) // (n-1)
            if t < 10:
                res.appendleft(t)
                s -= t
                n -= 1
            else:
                res.appendleft(10)
                s -= 10
                n -= 1
    print(*res)
