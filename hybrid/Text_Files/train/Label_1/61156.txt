for _ in range(int(input())):
    n, p, k = map(int, input().split())
    a = sorted(map(int, input().split()))
    ans = 0
    i = n - 1
    while i >= 0 and p >= 0:
        ans += 1
        p -= a[i]
        i -= 1
        if i >= 0 and p >= a[i]:
            ans += 1
            p -= a[i]
            i -= k - 1
    print(ans)
