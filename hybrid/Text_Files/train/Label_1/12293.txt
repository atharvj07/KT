#include <iostream>
#include <vector>
using namespace std;

int main() {
  int k;
  cin >> k;
  if (k == 2)
    cout << "NO";
  else {
    cout << "YES\n";
    int n = 2 * k + 2, m = 3 * k + 2;
    cout << n << " " << m << "\n";
    for (int i = 2; i <= k + 1; i++) cout << 1 << " " << i << "\n";
    for (int i = k + 2; i <= 2 * k + 1; i++) cout << i << " " << i + 1 << "\n";
    cout << 1 << " " << n << "\n";
    cout << k + 2 << " " << n << "\n";
    cout << 2 * k + 2 << " " << n << "\n";
  }
  return 0;
}
