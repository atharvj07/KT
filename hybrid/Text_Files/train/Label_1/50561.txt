import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] circles = new int[n][3];
        for (int i = 0; i < n; i++) {
            circles[i][0] = scanner.nextInt();
            circles[i][1] = scanner.nextInt();
            circles[i][2] = scanner.nextInt();
        }
        int regions = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (areCirclesIntersecting(circles[i], circles[j])) {
                    regions++;
                }
            }
        }
        System.out.println(regions + n + 1);
    }

    public static boolean areCirclesIntersecting(int[] circle1, int[] circle2) {
        int dx = circle1[0] - circle2[0];
        int dy = circle1[1] - circle2[1];
        int distance = dx * dx + dy * dy;
        int radiusSum = circle1[2] + circle2[2];
        return distance < radiusSum * radiusSum;
    }
}
