#include <cstdio>

int main() {
    unsigned long long x = 0, n, ans = 1e9;
    int r = 0;
    char s[30];
    for (int i = 0; i < 100010; i++) {
        scanf("%s", s);
        if (s[0] == 'a') {
            if (x + ans * r < x) {
                r = -1;
            }
            x += ans * r;
        } else if (s[0] == 'f') {
            scanf("%llu", &n);
            r++;
            ans /= n;
        } else {
            r--;
            ans *= r;
        }
        if (r == -1) {
            printf("OVERFLOW!!!");
            return 0;
        }
    }
    printf("%llu", x);
    return 0;
}
