#include <bits/stdc++.h>
using namespace std;
int m, n;
string s;

int main() {
    int z = 0;
    cin >> s;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] == '|') z++;
        else {
            if (s[i] == '+') m = z;
            else if (s[i] == '=') n = z;
        }
    }
    if (m == n)
        cout << "Impossible";
    else {
        if (m < n) s[m + n] = '|', s[m] = '+';
        else
            s[m + n] = '+', s[m] = '|';
        cout << s;
    }
    return 0;
}
