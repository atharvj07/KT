t = int(input())
for _ in range(t):
    n, m, k = map(int, input().split())
    h = list(map(int, input().split()))
    possible = True
    for i in range(n - 1):
        if h[i] >= h[i + 1] - k:
            m += min(h[i] - (h[i + 1] - k), h[i])
        else:
            needed = (h[i + 1] - k) - h[i]
            if m >= needed:
                m -= needed
            else:
                possible = False
                break
    print("YES" if possible else "NO")
