#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int n, m, k;
vector<vector<int>>g;
int main() {
    cin >> n >> m >> k;
    g.resize(n + 1, vector<int>(m + 1));
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> g[i][j];
        }
    }
    int ans = 0, cnt = 0;
    for (int j = 1; j <= m; j++) {
        int l = -1;
        for (int i = 1; i <= n; i++) {
            if (i < n) {
                int t = i + 1;
                if (g[t][j] == 1) {
                    if (l != -1)g[l][j] = 0, cnt++;
                    l = i, ans++;
                }
                continue;
            }
            if (g[i][j])l = i, ans++;
            for (int p = min(i + k, n); p > i; p--) {
                ans += (g[p][j] > 0);
            }
        }
    }
    cout << ans << " " << cnt << '\n';
    return 0;
}
