import sys

n, m, k = map(int, sys.stdin.readline().strip().split())
a = list(map(int, sys.stdin.readline().strip().split()))
b = list(map(int, sys.stdin.readline().strip().split()))
ans = 0
for i in b:
    ind = a.index(i)
    ans += ind // k + (ind % k != 0)
    a.pop(ind)
    a.insert(0, i)
print(ans)
