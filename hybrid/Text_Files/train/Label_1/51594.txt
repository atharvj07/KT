import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        // Graph representation
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int x = Integer.parseInt(line[0]);
            int y = Integer.parseInt(line[1]);
            graph.get(x).add(y);
            graph.get(y).add(x);
        }

        long[] danger = new long[n + 1];
        boolean[] visited = new boolean[n + 1];

        long maxDanger = 1;

        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                long currDanger = dfs(i, graph, danger, visited, 1);
                maxDanger = Math.max(maxDanger, currDanger);
            }
        }

        System.out.println(maxDanger);
    }

    static long dfs(int node, List<List<Integer>> graph, long[] danger, boolean[] visited, long currDanger) {
        visited[node] = true;
        danger[node] = currDanger;

        for (int neighbor : graph.get(node)) {
            if (!visited[neighbor]) {
                currDanger = dfs(neighbor, graph, danger, visited, currDanger * 2);
            }
        }

        return currDanger;
    }
}
