import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        if (k < n) {
            System.out.println(-1);
            return;
        }
        int[] a = new int[n + 1];
        for (int i = 0; i < n; i++) {
            a[i + 1] = i + 1;
        }

        int sum = 0;
        for (int i = 1; i <= n; i++) {
            int j = k - i;
            if (j < i || j > n) {
                break;
            }
            sum += Math.max(a[i], a[j]);
        }

        if (sum > k) {
            System.out.println(-1);
            return;
        }

        System.out.println(sum);
        int[] p = new int[n + 1];
        int[] q = new int[n + 1];
        Arrays.fill(p, 0);
        Arrays.fill(q, 0);
        int ai = 1;
        int bi = n;
        for (int i = 1; i <= n; i++) {
            int j = k - i;
            if (j < i || j > n) {
                break;
            }
            if (a[i] > a[j]) {
                p[i] = a[i];
                q[j] = a[j];
                ai++;
            } else {
                p[j] = a[j];
                q[i] = a[i];
                bi--;
            }
        }

        for (int i = 1; i <= n; i++) {
            if (p[i] == 0) {
                p[i] = a[ai++];
            }
            if (q[i] == 0) {
                q[i] = a[bi--];
            }
        }

        for (int i = 1; i <= n; i++) {
            System.out.print(p[i] + " ");
        }
        System.out.println();
        for (int i = 1; i <= n; i++) {
            System.out.print(q[i] + " ");
        }
        System.out.println();
    }
}
