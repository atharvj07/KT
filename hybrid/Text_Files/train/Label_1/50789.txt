import java.util.Scanner;

public class Main {
    static int[] solve(int n, int x) {
        if (x == 1 || x == 2 * n - 1) return null;

        int num = (x + 1) / 2;
        int[] ans = new int[2 * n - 1];
        ans[n - 1] = num;

        int now = 1;
        for (int i = n - 2; i >= 0; i--) {
            int m = (ans[(i + 1) * 2] + ans[(i + 1) * 2 + 1] + ans[i * 2 + 1]) / 2;
            if ((m & 1) == 1) m++;
            if (now < m) ans[i * 2] = now++;
            else if (now > m) ans[i * 2 + 1] = now++;
            else ans[i * 2] = ans[i * 2 + 1] = now++;
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int[] ans = solve(n, x);
        if (ans == null) System.out.println("No");
        else {
            System.out.println("Yes");
            for (int v : ans) System.out.println(v);
        }
    }
}
