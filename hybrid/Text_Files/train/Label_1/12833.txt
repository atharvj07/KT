#include <bits/stdc++.h>
using namespace std;
int n, m, k, a[110][110], ans, sum, mn = 1e9;
int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) {
      cin >> a[i][j];
      if (a[i][j]) { sum++; }
    }
  for (int j = 1; j <= m; j++) {
    for (int i = 1; i <= n; i++) {
      if (a[i][j]) {
        ans += min(k, n - i + 1);
        while (a[i][j] && i <= n) {
          a[i][j] = 0;
          i++;
        }
        break;
      }
    }
  }
  cout << ans << " " << sum - ans << endl;
}
