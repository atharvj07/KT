#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.rbegin(), a.rend());
    long long ans = 0;
    int x = 0, y = 0;
    for (int i = 0; i < n; i += 2) {
        if (i + 1 < n) {
            ans += (y + a[i + 1]) * (y + a[i + 1]);
            y += a[i + 1];
        }
        ans += (x + a[i]) * (x + a[i]);
        x += a[i];
    }
    cout << ans << endl;
    return 0;
}
