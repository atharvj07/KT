import sys
N = int(input())
grid = [[int(ch) for ch in l.strip()] for l in sys.stdin]
possible = (N+1)//2
for i in range(N):
    for j in range(N):
        if grid[i][j]>2: possible-=1
        if i+1<N and j+1<N:
            if grid[i][j]+grid[i+1][j+1]+grid[i+1][j]+grid[i][j+1]>3:
                possible-=1
for j in range(N):
    c=0
    for i in range(N):
        if grid[i][j]: c+=1
    if c==1: possible-=1
for i in range(N):
    c=0
    for j in range(N):
        if grid[i][j]: c+=1
    if c==1: possible-=1
if possible>0: print("Yes")
else: print("No")
