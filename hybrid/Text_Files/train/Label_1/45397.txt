import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n1 = scanner.nextInt();
        int m1 = scanner.nextInt();
        scanner.nextLine();
        int[][] a = new int[n1][m1];
        for (int i = 0; i < n1; i++) {
            String line = scanner.nextLine();
            for (int j = 0; j < m1; j++) {
                a[i][j] = line.charAt(j) - '0';
            }
        }
        int n2 = scanner.nextInt();
        int m2 = scanner.nextInt();
        scanner.nextLine();
        int[][] b = new int[n2][m2];
        for (int i = 0; i < n2; i++) {
            String line = scanner.nextLine();
            for (int j = 0; j < m2; j++) {
                b[i][j] = line.charAt(j) - '0';
            }
        }
        long maxOverlap = 0;
        long x = 0;
        long y = 0;
        for (int i = -n2 + 1; i <= n1 - 1; i++) {
            for (int j = -m2 + 1; j <= m1 - 1; j++) {
                long overlap = 0;
                for (int k = 0; k < n2; k++) {
                    for (int l = 0; l < m2; l++) {
                        if ((k + i >= 0) && (k + i < n1) && (l + j >= 0) && (l + j < m1)) {
                            overlap += a[k + i][l + j] * b[k][l];
                        }
                    }
                }
                if (overlap > maxOverlap) {
                    maxOverlap = overlap;
                    x = i;
                    y = j;
                }
            }
        }
        System.out.println(x + " " + y);
    }
}
