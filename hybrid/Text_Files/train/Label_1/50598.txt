import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
    private static final int MAXN = 100005;
    private static long[] tree = new long[MAXN];

    private static void update(int i, long val) {
        tree[i] += val;
        i = (i + 1) / 2;
        while (i > 0) {
            tree[i] = Math.max(tree[i * 2], tree[i * 2 + 1]);
            i /= 2;
        }
    }

    private static long get(int l, int r) {
        long ans = 0;
        for (l = (l + MAXN - 1) / 2, r = (r + MAXN) / 2; l <= r; l = (l + 1) / 2, r = (r - 1) / 2) {
            ans = Math.max(ans, tree[l]);
            ans = Math.max(ans, tree[r]);
        }
        return ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        long[] a = new long[n];
        StringTokenizer tokenizer = new StringTokenizer(in.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Long.parseLong(tokenizer.nextToken());
        }
        int m = Integer.parseInt(in.readLine());
        while (m-- > 0) {
            tokenizer = new StringTokenizer(in.readLine());
            int type = Integer.parseInt(tokenizer.nextToken());
            if (type == 0) {
                int i = Integer.parseInt(tokenizer.nextToken()) - 1;
                long val = Long.parseLong(tokenizer.nextToken());
                update(i, val - a[i]);
                a[i] = val;
            } else {
                int l = Integer.parseInt(tokenizer.nextToken()) - 1;
                int r = Integer.parseInt(tokenizer.nextToken()) - 1;
                int k = Integer.parseInt(tokenizer.nextToken());
                long[] dp = new long[k + 2];
                dp[0] = 0;
                dp[1] = Math.max(a[l], 0);
                for (int i = l + 1; i <= r; i++) {
                    dp[k + 1] = Math.max(dp[k + 1], dp[k] + a[i]);
                    for (int j = k; j > 0; j--) {
                        dp[j] = Math.max(dp[j], dp[j - 1] + a[i]);
                    }
                }
                long ans = 0;
                for (int i = 0; i <= k; i++) {
                    ans = Math.max(ans, dp[i]);
                }
                System.out.println(ans);
            }
        }
    }
}
