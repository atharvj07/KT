#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<ll> a(n);
        unordered_map<ll, ll> cnt;
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        vector<pair<ll, ll>> ans;
        for (auto x : a) {
            for (ll y = 1; y * y <= x; y++) {
                if (x % y == 0) {
                    if (!cnt[y] && !cnt[x / y]) {
                        cnt[y] = cnt[x / y] = 1;
                        ans.emplace_back(x, y);
                        if (ans.size() == n / 2) break;
                    }
                }
            }
        }
        for (auto p : ans) cout << p.first << " " << p.second << "\n";
    }
    return 0;
}
