n, m, k = map(int, input().split())
adj = [[] for _ in range(n+1)]
for _ in range(m):
    x, y, w = map(int, input().split())
    adj[x].append([y, w])
    adj[y].append([x, w])

vis = [False for _ in range(n+1)]
vis[1] = True
q = [(0, 1)]
dist = [10**18 for _ in range(n+1)]
dist[1] = 0
while q:
    d, u = q.pop(0)
    for v, w in adj[u]:
        if not vis[v]:
            vis[v] = True
            q.append((d+w, v))
            dist[v] = d+w

edges = []
for u in range(1, n+1):
    for v, w in adj[u]:
        if dist[v] == dist[u] + w:
            edges.append((u, v, w))
edges.sort(key=lambda x: x[2])

ans = []
cnt = 0
for u, v, w in edges:
    if cnt == k:
        break
    cnt += 1
    ans.append(w)

print(cnt)
print(' '.join(map(str, ans)))
