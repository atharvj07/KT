import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        boolean hasLinear = false;
        for (int i = 0; i < n; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            if (a + b == 1) {
                hasLinear = true;
            }
        }
        System.out.println(hasLinear ? "Ani" : "Borna");
    }
}
