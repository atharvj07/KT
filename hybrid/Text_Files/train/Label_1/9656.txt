#include<bits/stdc++.h>
using namespace std;
const int maxN = 2e5 + 5;
int n, m, a[maxN];
int main() {
    cin >> n >> m;
    int p = 1, cur = n;
    for (int i = 1; i <= m; i++) {
        int x; cin >> x;
        while (a[p] != 0) p++;
        while (a[cur] == 0) cur--;
        a[p] = a[cur]; a[cur] = 0;
        if (p == x) continue;
        a[x] = a[p]; a[p] = 0;
        if (p < x) cur--;
    }
    for (int i = 1; i <= n; i++) cout << a[i] << endl;
    return 0;
}
