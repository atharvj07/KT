n = int(input())
trees = [list(map(int, input().split())) for _ in range(n)]

count = 0
last_position = float('-inf')

for i in range(n):
    if i == 0:
        # Always can fell the first tree to the left
        count += 1
        last_position = trees[i][0]
    elif i == n - 1 or trees[i][0] - trees[i][1] > last_position:
        # Fell to the left if it does not overlap or it's the last tree
        count += 1
        last_position = trees[i][0]
    elif trees[i][0] + trees[i][1] < trees[i + 1][0]:
        # Fell to the right if it doesn't overlap with the next tree
        count += 1
        last_position = trees[i][0] + trees[i][1]
    else:
        last_position = trees[i][0]

print(count)
