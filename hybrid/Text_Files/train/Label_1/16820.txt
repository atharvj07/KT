#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[100];
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  if (n == 2 && a[0] > a[1])
    return cout << 0, 0;
  bool f = 0;
  int sum = 0, ans[100], ai = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] > a[0]) f = 1;
    sum += a[i];
    if (sum > a[0] + a[0] / 2) {
      ai = i;
      break;
    }
  }
  if (f == 0)
    return cout << a[0] + a[0] / 2, 0;
  if (a[ai] + a[ai] / 2 > a[0])
    return cout << 0, 0;
  int tot = sum + a[ai] + a[ai] / 2;
  if (tot > a[0] + a[0] / 2) {
    ans[0] = 1;
    ans[1] = ai + 1;
    cout << 2 << '\n';
    cout << ans[0] << ' ' << ans[1] << '\n';
  } else
    cout << 0 << '\n';
  return 0;
}
