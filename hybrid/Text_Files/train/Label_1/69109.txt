n = int(input())
s = input()
a = list(map(int, input().split()))
rm = [0] * (n + 2)
for i in range(n - 1, -1, -1):
    if s[i] != 'r': rm[i] = rm[i + 1]
    else: rm[i] = rm[i + 1] + a[i]
ans = 1e9
for i in range(n):
    if i > 0: ans = min(ans, rm[i] + a[i])
    else: ans = min(ans, rm[i])
print(ans)
