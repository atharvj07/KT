#include <cstdio>

using ll = long long;

int main() {
    int t;
    scanf("%d", &t);
    while (t--) {
        ll n;
        scanf("%lld", &n);

        int ans = 0;
        for (ll i = n / 5; i >= 1; i--) {
            ll tmp = n - 5 * i;
            if (tmp % 3 == 0) {
                ans++;
            }
        }
        printf("%d\n", ans);
    }
    return 0;
}
