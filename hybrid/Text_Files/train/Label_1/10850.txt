#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const int MOD = 10007;

int main() {
    int n;
    cin >> n;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    ll ans = 0;
    for (int i = 0; i < n; i++) { ans = (ans + a[i] * a[n - 1 - i]) % MOD; }
    cout << ans << '\n';
    return 0;
}
