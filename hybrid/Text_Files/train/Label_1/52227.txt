import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        int[] cnt = new int[6];
        for (int x : a) {
            cnt[x]++;
        }
        for (int x : b) {
            cnt[x]--;
        }
        int ans = 0;
        for (int i = 1; i <= 5; i++) {
            if (cnt[i] < 0) {
                ans += Math.abs(cnt[i]);
            }
        }
        if (ans % 2 == 0) {
            System.out.println(ans / 2);
        } else {
            System.out.println(-1);
        }
    }
}
