def sum_of_digits(x):
    return sum([int(digit) for digit in str(x)])

def min_moves(n, s):
    if sum_of_digits(n) <= s:
        return 0
    moves, multiplier = 0, 1
    for digit in reversed(str(n)):
        increment = (10 - int(digit)) % 10
        if sum_of_digits(n + increment * multiplier) > s:
            moves += increment * multiplier
            n += increment * multiplier
        else:
            return moves + increment * multiplier
        multiplier *= 10
    return moves

t = int(input())
for _ in range(t):
    n, s = map(int, input().split())
    print(min_moves(n, s))
