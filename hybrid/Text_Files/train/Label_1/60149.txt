import sys
input = sys.stdin.readline

n,m,k = map(int,input().split())
a,b,c = [list(map(int,input().split())) for _ in range(n)]
port = [[0]*n for _ in range(n)]
for _ in range(m):
    x,y = map(int,input().split())
    x -= 1; y -= 1
    port[x][y] = 1
dp = [[[-1]*(k+1) for _ in range(n)] for _ in range(n)]

def dfs(now,prev,army):
    if now == n-1:
        return c[now] if army >= a[now] else -1
    if dp[now][prev][army] != -1: return dp[now][prev][army]
    ret = 0
    if army >= a[now]:
        ret = max(ret,dfs(now+1,now,army+b[now]-1)+c[now])
        for i in range(now):
            if port[now][i] and i != prev:
                ret = max(ret,dfs(now+1,i,army-1))
    dp[now][prev][army] = ret
    return ret

print(dfs(0,-1,k))
