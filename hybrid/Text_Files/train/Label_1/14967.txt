#include<bits/stdc++.h>
using namespace std;
int main() {
    long long x, y, n, a=0, b=1;
    cin >> x >> y >> n;
    double dif=1e9;
    for(int i=1; i<=n; i++) {
        long long num = x*i/y;
        if(abs((double)x/y - (double)num/i)<dif) {
            dif = abs((double)x/y - (double)num/i);
            a = num;
            b = i;
        }
        else if(abs((double)x/y-(double)num/i)==dif) {
            if(i<b) {
                a=num;
                b=i;
            }
            else if(i==b) {
                if(num<a) {
                    a=num;
                    b=i;
                }
            }
        }
    }
    cout << a << "/" << b << endl;
    return 0;
}
