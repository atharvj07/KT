import sys

lines = []
for i in range(8):
    a, b, c = map(int, input().split())
    lines.append((a, b, c))

lines.sort()

a = lines[0]
b = lines[1]
c = lines[2]
d = lines[3]
e = lines[4]
f = lines[5]
g = lines[6]
h = lines[7]

if (a[0] == b[0] == c[0] == d[0] and
        e[0] == f[0] == g[0] == h[0] and
        a[1] == e[1] == f[1] == g[1] and
        b[1] == c[1] == d[1] == h[1] and
        a[2] == b[2] == e[2] == f[2] and
        c[2] == d[2] == g[2] == h[2]):
    print('YES')
    print(*a)
    print(*b)
    print(*c)
    print(*d)
    print(*e)
    print(*f)
    print(*g)
    print(*h)
else:
    print('NO')
