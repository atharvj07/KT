def max_length_sum(n, m, k, s, t):
    dp = [[[0 for _ in range(k + 1)] for _ in range(m + 1)] for _ in range(n + 1)]
    for i in range(n + 1):
        for j in range(m + 1):
            for l in range(1, k + 1):
                dp[i][j][l] = max(dp[i][j][l], dp[i-1][j][l], dp[i][j-1][l])
                if i and j and s[i-1] == t[j-1]:
                    dp[i][j][l] = max(dp[i][j][l], dp[i-1][j-1][l-1] + 1)
    return max(max(row[k] for row in level) for level in dp)

n, m, k = map(int, input().split())
s = input()
t = input()
print(max_length_sum(n, m, k, s, t))
