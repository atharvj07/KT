#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 2005, oo = 1e9;
int n, m, k, a[N][N], nx[N][N], ny[N][N], cnt[N][N];
int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) {
      char c;
      cin >> c;
      if (i == 1)
        cnt[i][j] = a[i][j] = 0;
      else if (c == '.')
        cnt[i][j] = a[i][j] = 0;
      else if (c == 'L')
        cnt[i][j] = a[i][j] = 1, nx[i][j] = i, ny[i][j] = j - 1;
      else if (c == 'R')
        cnt[i][j] = a[i][j] = 1, nx[i][j] = i, ny[i][j] = j + 1;
      else if (c == 'U')
        cnt[i][j] = a[i][j] = 1, nx[i][j] = i - 1, ny[i][j] = j;
      else if (c == 'D')
        cnt[i][j] = a[i][j] = 1, nx[i][j] = i + 1, ny[i][j] = j;
    }
  for (int step = 1, i = 2; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cnt[i][j] = cnt[i - 1][j] + a[i][j];
    }
    for (int j = 1; j <= m; j++) {
      int x = nx[i][j], y = ny[i][j];
      if (x >= 2 && y >= 1 && y <= m) { cnt[x][y] += a[i][j]; }
    }
  }
  for (int j = 1; j <= m; j++) {
    cout << cnt[n][j] << " ";
  }
}
