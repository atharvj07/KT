#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n + 1, vector<int>());
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        for (int j = 0; j < i; j++) {
            if (a % adj[j][0] == 0 || adj[j][0] % a == 0) {
                adj[i].push_back(adj[j][0]);
                adj[j].push_back(a);
            }
        }
        adj[i].push_back(a);
    }
    int mx = 0;
    for (int i = 0; i < n; i++) {
        mx = max(mx, (int)adj[i].size());
    }
    cout << mx << endl;
    return 0;
}
