#include <cstdio>
#include <vector>
using namespace std;

const int N = 1005;
int n, ans, vis[N][N];
char grid[N][N];

vector<pair<int, int>> moves = {{-1, 0}, {0, 1}, {1, 0}, {0, -1}};

bool valid(int x, int y) {
    return (x >= 0 && x < n && y >= 0 && y < n && !vis[x][y]);
}

void dfs(int x, int y) {
    vis[x][y] = 1;
    ans++;
    for (auto move : moves) {
        int nx = x + move.first;
        int ny = y + move.second;
        if (valid(nx, ny)) {
            dfs(nx, ny);
        }
    }
}

int main() {
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            grid[i][j] = '.';
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (!vis[i][j]) {
                dfs(i, j);
                for (int k = 0; k < ans; k++) {
                    grid[i][j] = 'C';
                }
            }
        }
    }
    printf("%d\n", ans);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            printf("%c", grid[i][j]);
        }
        printf("\n");
    }
    return 0;
}
