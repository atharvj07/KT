#include<vector>
#include<cmath>
#include<iostream>
using namespace std;
int main() {
    int i, t = 0, tem = 0;
    string s;
    cin >> s;
    vector<int> v(s.size() + 5, 0);
    for(i = 0; i < (int)s.size(); i++) {
        if(s[i] == '^') {
            t = i;
            continue;
        }
        if(s[i] >= '0' && s[i] <= '9') {
            v[i - t]++;
        }
    }
    for(i = 0; i < (int)s.size(); i++) {
        if(i > t)
            v[i] = v[i] * (i - t);
        else
            v[i] = v[i] * (t - i);
        tem += v[i];
    }
    if(tem == 0)
        cout << "balance";
    else if(tem > 0)
        cout << "right";
    else
        cout << "left";
    return 0;
}
