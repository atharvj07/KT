for tc in range(int(input())):
    n = int(input())
    b = list(map(int, input().split()))[::-1]
    a = [0] * n
    l = r = n // 2
    for i in range(n):
        if i % 2 == 0:
            a[l] = b[i]
            l -= 1
        else:
            a[r] = b[i]
            r += 1
    print(" ".join(map(str, a)))
