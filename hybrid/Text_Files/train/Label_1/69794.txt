import heapq

n, s = map(int, input().split())
orders = []
for _ in range(n):
    d, p, q = input().split()
    p = int(p)
    q = int(q)
    orders.append((d, p, q))

orders.sort(key = lambda x: (x[1], x[0]))

sell = []
buy = []

for d, p, q in orders:
    if d == 'S':
        heapq.heappush(sell, (p, q))
    else:
        heapq.heappush(buy, (-p, q))

for _ in range(min(s, len(sell))):
    p, q = heapq.heappop(sell)
    print(f"S {p} {q}")

for _ in range(min(s, len(buy))):
    p, q = heapq.heappop(buy)
    print(f"B {abs(p)} {q}")
