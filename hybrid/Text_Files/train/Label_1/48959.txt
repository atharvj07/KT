import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), q = sc.nextInt();
        int[] t = new int[n + 1];
        for (int i = 1; i <= n; i++) t[i] = sc.nextInt();
        int[] p = new int[n + 1];
        PriorityQueue<Integer> pq = new PriorityQueue<>((a, b) -> t[a] == t[b] ? a - b : t[b] - t[a]);
        while (q-- > 0) {
            int type = sc.nextInt(), id = sc.nextInt();
            if (type == 1) {
                p[id] = 1;
                pq.add(id);
            } else {
                while (!pq.isEmpty() && p[pq.peek()] == 0) pq.poll();
                if (pq.isEmpty()) System.out.println("NO");
                else System.out.println(pq.size() >= k && p[id] == 1 ? "YES" : "NO");
            }
        }
    }
}
