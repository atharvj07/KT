import java.util.*;
import java.io.*;

class Main {
    int N;
    char[] s;
    int[] cnt;

    void solve() {
        N = s.length;
        int ans = 0;
        for (int i = 0; i < 26; i++) {
            int c = 0;
            for (int j = 0; j < N; j++) {
                if (s[j] - 'a' == i) {
                    c++;
                }
            }
            ans += c / 2;
            cnt[i] = c % 2;
        }
        int odd = 0;
        for (int i = 0; i < 26; i++) {
            if (cnt[i] == 1) {
                odd++;
            }
        }
        if (odd > 0) {
            ans++;
        }
        System.out.println(ans);
    }

    void run() {
        Scanner sc = new Scanner(System.in);
        s = sc.nextLine().toCharArray();
        cnt = new int[26];
        solve();
    }

    public static void main(String[] args) {
        new Main().run();
    }
}
