def count_pairs(n):
    s = str(n * (n + 1) // 2)
    l = len(s)
    ans = 0
    for i in range(1, l):
        if s[-i] != '9':
            if i == 1:
                ans = (n % 10) - 1
            else:
                prefix = int(s[:-i]) - 1
                ans = (prefix + 1) * 10 - int(s[-i])
            break
    if ans == 0:
        ans = n - 1
    return ans

n = int(input())
print(count_pairs(n))
