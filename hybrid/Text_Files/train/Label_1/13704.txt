#include <vector>
#include <iostream>
int main() {
  int n, sum = 0, last;
  std::vector<int> v;
  std::cin >> n;
  for(int i = 0; i < n; i++){
    int x;
    std::cin >> x;
    v.push_back(x);
  }
  for(int i = 1; i < n; i++){
    sum += std::abs(v[i] - v[i - 1]);
  }
  std::cout << sum << '\n';
}
