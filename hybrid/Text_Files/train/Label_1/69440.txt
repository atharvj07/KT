n = int(input())
temperatures = list(map(int, input().split()))

def predict_temperature(n, temperatures):
    if n == 2:
        return temperatures[1] * 2 - temperatures[0]
    else:
        common_difference = temperatures[1] - temperatures[0]
        for i in range(2, n):
            if temperatures[i] - temperatures[i-1] != common_difference:
                return temperatures[-1]
        return temperatures[-1] + common_difference

print(predict_temperature(n, temperatures))
