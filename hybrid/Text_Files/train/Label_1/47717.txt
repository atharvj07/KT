import java.util.Scanner;
import java.util.TreeSet;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int b = sc.nextInt();
        int[] x = new int[n];
        int[] k = new int[n];
        int[][] a = new int[n][];
        int[][] f = new int[n + 1][m + 1];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            k[i] = sc.nextInt();
            a[i] = new int[sc.nextInt()];
            for (int j = 0; j < a[i].length; j++) {
                a[i][j] = sc.nextInt() - 1;
            }
        }
        for (int i = n; i >= 0; i--) {
            for (int j = 0; j <= m; j++) {
                f[i][j] = 1 << 30;
                if (i < n) {
                    for (int l = 0; l <= j; l++) {
                        TreeSet<Integer> id = new TreeSet();
                        for (int ii = 0; ii < a[i].length; ii++) {
                            if (a[i][ii] < l) {
                                id.add(a[i][ii]);
                            }
                        }
                        if (id.size() >= k[i]) {
                            f[i][j] = Math.min(f[i][j], f[i + 1][j - id.size()]);
                        }
                    }
                    f[i][j] = Math.min(f[i][j], f[i + 1][j] + x[i]);
                }
            }
        }
        System.out.println(f[0][m] >= 1 << 30 ? -1 : f[0][m] + b * k[0]);
    }
}
