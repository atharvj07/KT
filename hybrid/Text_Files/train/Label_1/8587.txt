#include <bits/stdc++.h>
using namespace std;
#define pb push_back

template <typename T>
vector<T> operator-(vector<T>& a, vector<T>& b) {
	vector<T> c;
	for (T& t: a) {
		bool found = false;
		for (T& t2: b)
			if (t==t2) {
				found = true;
				break;
			}
		if (!found) c.pb(t);
	}
	return c;
}
int main() {
	int n;
	cin >> n;
	vector<vector<int>> g(n+1);
	for (int i = 0; i < 2*n; i++) {
		int a, b;
		cin >> a >> b;
		g[a].pb(b);
		g[b].pb(a);
	}
	queue<int> q;
	for (int i = 1; i <= n; i++)
		if (g[i].size() == 1) q.push(i);
	vector<int> res;
	while (!q.empty()) {
		int cur = q.front();
		q.pop();
		res.pb(cur);
		int next = g[cur][0];
		g[cur].clear();
		g[next].erase(g[next].begin());
		if (g[next].size() == 1) q.push(next);
	}
	if (res.size() == n) {
		for (int t: res)
			cout << t << ' ';
		cout << '\n';
	}
	else cout << -1 << '\n';
}
