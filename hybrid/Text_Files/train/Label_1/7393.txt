#include <cstdio>
#include <list>

int main() {
    int T;
    scanf("%d", &T);
    while (T--) {
        int n;
        scanf("%d", &n);
        std::list<long long> monster;
        for (int i = 0; i < n; i++) {
            long long x, y;
            scanf("%lld%lld", &x, &y);
            monster.push_back(x);
            monster.push_back(y);
        }
        long long ans = 0, A, B, C, D;
        auto it = monster.begin();
        A = *it;
        ++it;
        B = *it;
        while (A > 0) {
            ++ans;
            A -= 1;
            C = B;
            D = *next(it);
            if (it == monster.end())
                it = monster.begin();
            else
                ++it;
            if (B <= 0) {
                if (C + D > 0)
                    B = D;
                else
                    B = -1;
            } else
                B -= 1;
        }
        printf("%lld\n", ans);
    }
    return 0;
}
