#include<bits/stdc++.h>
using namespace std;
int main(){
    int n; cin>>n;
    vector<long long> a(n);
    for(int i=0;i<n;i++){cin>>a[i];}
    long long v=1;
    int ans=0;
    for(int i=n-1;i>=0;i--){
        if(v<=a[i]){
            v<<=1;
            ans++;
        }
    }
    cout<<ans<<endl;
}
