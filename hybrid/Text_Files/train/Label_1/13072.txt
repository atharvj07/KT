#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long
const int mxn = 2e5 + 15;
const ll mod = 1e9 + 7;

int main() {
    int q;
    cin >> q;
    while (q--) {
        ll h, n;
        cin >> h >> n;
        vi v(h + 15, 0);
        for (int i = 1; i <= n; i++) {
            int x;
            cin >> x;
            v[x] = 1;
        }
        int ans = 0;
        for (ll i = h; i >= 2; i--) {
            if (v[i - 1] || v[i - 2]) continue;
            v[i] = 0;
            ans++;
        }
        cout << ans << '\n';
    }
}
