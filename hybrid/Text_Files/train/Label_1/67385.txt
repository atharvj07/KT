for _ in range(int(input())):
    n,m,k=map(int,input().split())
    a=list(map(int,input().split()))
    ans="YES"
    for i in range(n-1):
        if a[i]>a[i+1]+k:
            ans="NO"
            break
        if a[i+1]-a[i]>k and m==0:
            ans="NO"
            break
        m+=a[i]-a[i+1]
        m=max(0,m)
    print(ans)
