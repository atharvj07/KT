import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        // n is the number of points
        int n = sc.nextInt();
        // Create an array of points
        Point[] points = new Point[n];
        // Read the coordinates of the points
        for (int i = 0; i < n; i++) {
            points[i] = new Point(sc.nextInt(), sc.nextInt());
        }
        // Sort the points by their y-coordinates
        Arrays.sort(points, Comparator.comparingInt(Point::getY));
        // Find the leftmost point
        Point leftmost = points[0];
        // Find the rightmost point
        Point rightmost = points[n - 1];
        // Create a stack to store the points on the convex hull
        Stack<Point> stack = new Stack<>();
        // Add the leftmost point to the stack
        stack.push(leftmost);
        // Add the rightmost point to the stack
        stack.push(rightmost);
        // Iterate over the remaining points
        for (int i = 1; i < n - 1; i++) {
            // Get the current point
            Point p = points[i];
            // While the stack has at least two points and the current point is not a left turn
            while (stack.size() >= 2) {
                // Get the top two points on the stack
                Point p2 = stack.pop();
                Point p1 = stack.peek();
                // If the current point is a right turn, add it to the stack
                if (isRightTurn(p1, p2, p)) {
                    stack.push(p2);
                    stack.push(p);
                    break;
                }
            }
        }
        // Print the number of points on the convex hull
        System.out.println(stack.size());
        // Print the points on the convex hull
        for (Point p : stack) {
            System.out.println(p.getX() + " " + p.getY());
        }
    }

    // Check if the point p is a right turn from the line defined by p1 and p2
    public static boolean isRightTurn(Point p1, Point p2, Point p) {
        return (p2.getX() - p1.getX()) * (p.getY() - p2.getY()) - (p2.getY() - p1.getY()) * (p.getX() - p2.getX()) > 0;
    }

    // Point class
    public static class Point {
        private int x;
        private int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }

        public int getX() {
            return x;
        }

        public int getY() {
            return y;
        }
    }
}
