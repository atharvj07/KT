#include <bits/stdc++.h>
using namespace std;

const int MAXN = 110;

int n, k;
string s;
int dp[MAXN][MAXN][MAXN];

int solve(int pos, int rem, int cur) {
    int &ret = dp[pos][rem][cur];
    if (ret != -1) return ret;
    if (pos == n) {
        return rem == 0 ? 0 : -1;
    }
    int ans = -1;
    if (rem > 0)
        ans = solve(pos + 1, rem - 1, cur);
    if (ans == -1)
        ans = solve(pos + 1, rem, cur);
    if (ans != -1)
        ans += n - pos - cur;
    if (s[pos] == s[cur])
        ans = max(ans, solve(pos + 1, rem, cur + 1));
    ret = ans;
    return ans;
}

int main() {
    cin >> n >> k;
    cin >> s;
    memset(dp, -1, sizeof(dp));
    cout << solve(0, k, 0) << endl;
    return 0;
}
