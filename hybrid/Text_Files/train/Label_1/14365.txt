#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int GCD(int a, int b) {
    if (b == 0) {
        return a;
    }
    return GCD(b, a % b);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int g = a[0];
    for (int i = 1; i < n; i++) {
        g = GCD(g, a[i]);
    }
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i] / g;
    }
    cout << sum * g << endl;
    return 0;
}
