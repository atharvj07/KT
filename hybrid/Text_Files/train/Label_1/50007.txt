import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int k = sc.nextInt(), n = sc.nextInt(), maxb = sc.nextInt(), t = sc.nextInt();
    int[] ans = new int[k];
    for (int i = 0; i < k; i++) {
      int[] a = new int[n * t];
      for (int j = 0; j < n; j++) {
        int bi = sc.nextInt();
        for (int ti = 0; ti < t; ti++) {
          a[j * t + ti] = bi % maxb;
          bi /= maxb;
        }
      }
      ans[i] = lcs(a);
    }
    for (int i = 0; i < k; i++) {
      System.out.println(ans[i]);
    }
    sc.close();
  }

  public static int lcs(int[] a) {
    int[] dp = new int[a.length];
    Arrays.fill(dp, 1);
    for (int i = 1; i < a.length; i++) {
      for (int j = 0; j < i; j++) {
        if (a[j] < a[i]) {
          dp[i] = Math.max(dp[i], dp[j] + 1);
        }
      }
    }
    int max = 0;
    for (int i = 0; i < a.length; i++) {
      max = Math.max(max, dp[i]);
    }
    return max;
  }
}
