#include <iostream>
#include <string>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.size();
        int cnt1 = 0, cnt2 = 0;
        for (int i = 0; i < n; i++) {
            if (i + 1 < n && s[i] == 'a' && s[i + 1] == 'b')
                cnt1++;
            if (i + 1 < n && s[i] == 'b' && s[i + 1] == 'a')
                cnt2++;
        }
        int ans = abs(cnt1 - cnt2);
        if (cnt1 < cnt2)
            for (int i = 0; i < n; i++) {
                if (i + 1 < n && s[i] == 'b' && s[i + 1] == 'b') {
                    s[i + 1] = 'a';
                    ans--;
                    break;
                }
            }
        else if (cnt2 < cnt1)
            for (int i = 0; i < n; i++) {
                if (i + 1 < n && s[i] == 'a' && s[i + 1] == 'a') {
                    s[i + 1] = 'b';
                    ans--;
                    break;
                }
            }
        cout << s << '\n';
    }
    return 0;
}
