#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while(t--) {
        int n;
        cin >> n;
        vector<pair<int,int>> vec(n);
        for(auto& x: vec) cin >> x.first >> x.second;
        sort(vec.begin(), vec.end(), [](auto& a, auto& b){
            return a.second < b.second;
        });
        int ans = 0;
        int r = vec[0].second;
        for(int i = 1; i < n; i++) {
            if(vec[i].first <= r) ans++;
            else r = vec[i].second;
        }
        cout << ans << "\n";
    }
}
