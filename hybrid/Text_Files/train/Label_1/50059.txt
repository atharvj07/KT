import java.util.*;
import java.lang.*;

class Main {
    static ArrayList<Integer>[] g;
    static int[] subtreeCnt, subtreeHash;
    static int[] subtreeSize;
    static int n, ans, maxCnt;

    static void dfs(int u, int p, int hash) {
        subtreeHash[u] = hash;
        subtreeSize[u] = 1;
        for (int v : g[u]) {
            if (subtreeHash[v] == -1) {
                dfs(v, u, u);
                subtreeSize[u] += subtreeSize[v];
                subtreeHash[u] = subtreeHash[u] ^ subtreeCnt[v];
            }
        }
        subtreeCnt[u]++;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        g = new ArrayList[n + 1];
        subtreeCnt = new int[n + 1];
        subtreeHash = new int[n + 1];
        subtreeSize = new int[n + 1];
        for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
        for (int i = 1; i < n; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            g[a].add(b);
            g[b].add(a);
        }
        for (int i = 1; i <= n; i++) subtreeHash[i] = -1;
        dfs(1, 1, 1);
        for (int i = 1; i <= n; i++) {
            if (subtreeCnt[i] > maxCnt || (subtreeCnt[i] == maxCnt && subtreeSize[i] < subtreeSize[ans])) {
                ans = i;
                maxCnt = subtreeCnt[i];
            }
        }
        System.out.println(ans);
    }
}
