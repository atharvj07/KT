from collections import defaultdict

s,p=input(),input()
z=[0]*len(p)
a=[defaultdict(int) for _ in range(len(s)+1)]
i,j,x,anscur=0,0,0,0
while i<len(p):
    if p[i]==p[j]:
        z[i]=j
        j+=1
        i+=1
    elif j>0:j=z[j-1]
    else:i+=1
i,j,x,anscur=0,0,0,0
for i in range(len(s)):
    if s[i]==p[j]:
        a[x+1][j]=a[x][j-1]+1
        j+=1
        if j==len(p):x,j=x+1,0
    elif j>0:j=z[j-1]
for x in range(1,len(s)+1):
    for j in range(1,len(p)):anscur=max(anscur,a[x][j])
    print(anscur,end=' ')
