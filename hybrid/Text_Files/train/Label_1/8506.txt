#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e18;
ll u, v, t;
int main() {
    cin >> u >> v;
    if (v <= u)
        return cout << -1, 0;
    t = (v + u) / 2;
    v -= t;
    if (!t || (t & v))
        cout << -1;
    else
        cout << 1 << '\n' << t << '\n';
}
