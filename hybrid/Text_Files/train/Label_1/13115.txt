#include<bits/stdc++.h>
using namespace std;
#define rep(i,a,n) for(int i=(a);i<(n);i++)
#define per(i,a,n) for(int i=(n)-1;i>=(a);i--)
#define fi first
#define se second
#define pb push_back
#define mp make_pair
typedef pair<int,int> pi;
typedef vector<int> vi;
typedef vector<pi> vpi;
typedef long long ll;

int main(){
  int n;cin>>n;
  vi a(n);
  rep(i,0,n)cin>>a[i];
  int ans=0;
  rep(i,0,n-1){
    int mi=i;
    rep(j,i+1,n)if(a[j]<a[mi])mi=j;
    ans+=mi-i;
    reverse(a.begin()+i,a.begin()+mi+1);
  }
  bool f=1;
  rep(i,0,n-1)if(a[i]>a[i+1])f=0;
  if(f)cout<<ans<<endl;
  else cout<<-1<<endl;
}
