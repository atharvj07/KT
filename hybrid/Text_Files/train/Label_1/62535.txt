import heapq as hq
n, k = map(int, input().split())
grid = []
pos = []
for i in range(n):
    l = list(input())
    for j in range(n):
        if l[j] != 'a':
            hq.heappush(pos, (l[j], i, j))
    grid.append(l)
for i in range(min(k, len(pos))):
    v, r, c = hq.heappop(pos)
    grid[r][c] = 'a'
for r in grid: print(''.join(r))
