import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = scanner.nextInt();
        }
        if (a[0] != b[0]) {
            System.out.println("NO");
            return;
        }
        int[][] c = new int[n][m];
        for (int i = 0; i < n; i++) {
            c[i][0] = a[i] ^ b[0];
        }
        for (int i = 0; i < m; i++) {
            c[0][i] = b[i] ^ a[0];
        }
        for (int i = 1; i < n; i++) {
            for (int j = 1; j < m; j++) {
                c[i][j] = (c[i - 1][j] ^ a[i]) ^ (c[i][j - 1] ^ b[j]);
            }
        }
        System.out.println("YES");
        for (int[] row : c) {
            for (int cell : row) {
                System.out.print(cell + " ");
            }
            System.out.println();
        }
    }
}
