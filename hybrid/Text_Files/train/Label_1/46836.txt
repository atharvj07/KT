import java.util.Scanner;

class Main {
    static final Scanner in = new Scanner(System.in);
    static char[] chs;

    public static void main(String[] __) {
        int n = in.nextInt();
        chs = in.next().toCharArray();
        int i = 0, j, ans = 0;
        while (i < n) {
            if (chs[i] == '0') {
                i++;
            } else {
                j = i + 1;
                while (j < n && chs[i] == chs[j]) {
                    j++;
                }
                ans += j - i;
                i = j;
            }
        }
        System.out.println(ans);
    }
}
