#include <bits/stdc++.h>
using namespace std;
int main() {
    string s; cin >> s;
    bool ok = false;
    for (int i = 0; i < s.size(); ++i) {
        ok |= s[i] != '.';
        if (s[i] != '.' && s[i] != s[(i + 1) % s.size()] && s[i] != s[(i - 1 + s.size()) % s.size()]) {
            cout << "Yes" << endl; return 0;
        }
    }
    cout << "No" << endl;
    return 0;
}
