from collections import defaultdict
from sys import stdin
n,k=map(int,stdin.readline().split())
a=list(map(int,stdin.readline().split()))
q=defaultdict(list)
for _ in range(k):
    q[stdin.readline().strip().split()].append(_)

mn=sorted({i:_[1] for i,_ in enumerate(zip(a,q))},key=lambda x:x[1])
ans=[0]*n
for i,j in mn:
    for l in j:
        k=l+1
        while k<len(mn) and mn[k][1]==j:
            ans[mn[k][0]]+=1
            k+=1
print(*ans)
