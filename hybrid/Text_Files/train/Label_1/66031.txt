n, k = map(int, input().split())
a = list(map(int, input().split()))

cost = 0
library = []

for i in range(n):
    if a[i] in library:
        continue
    if len(library) < k:
        library.append(a[i])
        cost += 1
    else:
        furthest, idx = 0, -1
        for j in range(len(library)):
            if library[j] not in a[i+1:]:
                idx = j
                break
            else:
                furthest_pos = a[i+1:].index(library[j])
                if furthest_pos > furthest:
                    furthest, idx = furthest_pos, j
        library.pop(idx)
        library.append(a[i])
        cost += 1

print(cost)
