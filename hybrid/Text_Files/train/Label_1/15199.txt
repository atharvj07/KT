#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e3 + 5;
int n, m, sale[maxn][maxn], ans, pre[maxn];
bool vis[maxn];
vector<int> ord;
int main()
{
    cin >> n >> m;
    memset(sale, -1, sizeof(sale));
    memset(vis, false, sizeof(vis));
    vector<int> a(n + 5), cnt(n + 5);
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    for (int i = 1; i <= m; i++)
    {
        int d, t;
        cin >> d >> t;
        sale[d][t] = 0;
    }
    for (int i = 1; i <= n; i++)
    {
        int mn = 2e9;
        int pos;
        for (int j = 1; j <= 1000; j++)
        {
            if (sale[j][i] != -1 && sale[j][i] < mn)
            {
                mn = sale[j][i];
                pos = j;
            }
        }
        if (mn == 2e9)
            mn = 2;
        cnt[i] = a[i];
        sale[pos][i] = mn;
        pre[i] = pos;
    }
    while (!ord.empty())
        ord.pop_back();
    for (int i = 1; i <= n; i++)
    {
        if (cnt[i] == 0)
            ord.push_back(i);
    }
    while (!ord.empty())
    {
        int num = ord.back();
        ord.pop_back();
        if (vis[num])
            continue;
        ans = max(ans, pre[num]);
        vis[num] = true;
        cnt[num] = max(0, cnt[num] - 1);
        int sum = 0;
        for (int i = 1; i <= n; i++)
            sum += cnt[i];
        if (sum == 0)
            break;
        for (int i = 1; i <= n; i++)
        {
            if (cnt[i] == 0)
                ord.push_back(i);
        }
    }
    cout << ans << endl;
    return 0;
}
