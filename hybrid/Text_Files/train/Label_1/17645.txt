#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        ans += 1LL * a[i] * i;
    }
    cout << ans << endl;
    for (int i = 0; i < n; i++) {
        cout << lower_bound(a.begin(), a.end(), a[i]) - a.begin() + 1 << " ";
    }
    cout << endl;
    return 0;
}
