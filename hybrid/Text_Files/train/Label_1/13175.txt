#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    map<int, int> mp;
    for (int x : a) {
        mp[x]++;
        if (mp[x] == 2) {
            cout << 0 << endl;
            return 0;
        }
    }
    vector<pair<int, int>> v;
    for (auto it : mp) {
        v.push_back({it.first, it.second});
    }
    sort(v.begin(), v.end());
    vector<int> b;
    int ans = 0;
    for (int i = 0; i < v.size(); i++) {
        int val = v[i].first;
        int cnt = v[i].second;
        while (cnt--) {
            if (b.empty() || b.back() != val) {
                b.push_back(val);
                ans++;
            }
            val++;
        }
    }
    cout << ans << endl;
    for (int x : b) cout << x << ' ';
    cout << '\n';
}
