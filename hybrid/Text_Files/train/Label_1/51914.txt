import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] lines = new long[n * 3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                lines[i * 3 + j] = in.nextInt();
            }
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    boolean ok = true;
                    long x1 = 0, y1 = 0, x2 = 0, y2 = 0, x3 = 0, y3 = 0;
                    for (int p = 0; p < 3; p++) {
                        if (lines[i * 3 + p] == 0 && lines[j * 3 + p] == 0 && lines[k * 3 + p] == 0) {
                            ok = false;
                            break;
                        }
                        if (lines[i * 3 + p] != 0) {
                            if (p == 0) x1 = -lines[i * 3 + 2] / lines[i * 3 + 0];
                            if (p == 1) y1 = -lines[i * 3 + 2] / lines[i * 3 + 1];
                        }
                        if (lines[j * 3 + p] != 0) {
                            if (p == 0) x2 = -lines[j * 3 + 2] / lines[j * 3 + 0];
                            if (p == 1) y2 = -lines[j * 3 + 2] / lines[j * 3 + 1];
                        }
                        if (lines[k * 3 + p] != 0) {
                            if (p == 0) x3 = -lines[k * 3 + 2] / lines[k * 3 + 0];
                            if (p == 1) y3 = -lines[k * 3 + 2] / lines[k * 3 + 1];
                        }
                    }
                    ok &= (x1 != 0 || y1 != 0) && (x2 != 0 || y2 != 0) && (x3 != 0 || y3 != 0);
                    if (ok) {
                        long a = x2 - x1, b = y2 - y1, c = x3 - x1, d = y3 - y1, e = a * (x1 + x2) + b * (y1 + y2), f = c * (x1 + x3) + d * (y1 + y3);
                        long dx = b * f - d * e, dy = d * c - b * a;
                        long g = dx * dx + dy * dy - a * a - b * b - c * c - d * d;
                        long x = -dx / g, y = -dy / g;
                        x1 -= x;
                        y1 -= y;
                        x2 -= x;
                        y2 -= y;
                        x3 -= x;
                        y3 -= y;
                        ok &= x1 * x1 + y1 * y1 == x2 * x2 + y2 * y2 && x2 * x2 + y2 * y2 == x3 * x3 + y3 * y3;
                    }
                    if (ok) ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
