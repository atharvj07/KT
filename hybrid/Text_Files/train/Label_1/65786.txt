from collections import defaultdict
import heapq

n, m = map(int, input().split())
a = list(map(int, input().split()))
edges = []
for _ in range(m):
    x, y, w = map(int, input().split())
    edges.append((w, x, y))

a.sort()
edges.sort()

par = [i for i in range(n+1)]
sz = [1 for i in range(n+1)]

def findpar(x):
    if par[x] == x:
        return x
    return findpar(par[x])

def unite(a, b):
    rootA = findpar(a)
    rootB = findpar(b)
    if rootA != rootB:
        if sz[rootA] > sz[rootB]:
            par[rootB] = rootA
            sz[rootA] += sz[rootB]
        else:
            par[rootA] = rootB
            sz[rootB] += sz[rootA]

ans = 0
for w, x, y in edges:
    if findpar(x) != findpar(y):
        unite(x, y)
        ans += w

for i in range(1, n):
    if findpar(i) != findpar(i+1):
        ans += a[i]

print(ans)
