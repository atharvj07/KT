import java.util.*;
 
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int x = sc.nextInt();
            int[] a = new int[n];
            int mn = 0, mx = 0;
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
                mn += (int) Math.ceil((double) a[i] / x);
                mx += a[i];
            }
            System.out.println(mn + " " + (int) Math.ceil((double) mx / x));
        }
    }
}
