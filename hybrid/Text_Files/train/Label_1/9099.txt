#include <iostream>
#define ll long long
using namespace std;
ll x[3], y[3], ans = 1e9;

int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  for (int i = 0; i < 3; i++) { cin >> x[i] >> y[i]; }
  if (x[0] == x[1]) {
    if (x[1] == x[2]) ans = 2;
    else {
      if (y[0] == y[2] || y[1] == y[2]) ans = 2;
      else ans = 3;
    }
  } else {
    if (y[0] == y[1]) {
      if (y[1] == y[2]) ans = 2;
      else {
        if (x[0] == x[2] || x[1] == x[2]) ans = 2;
        else ans = 3;
      }
    } else {
      if (x[2] == x[0] || x[2] == x[1] || y[2] == y[0] || y[2] == y[1])
        ans = 3;
      else ans = 4;
    }
  }
  cout << ans;
}
