#include <bits/stdc++.h>
using namespace std;
#define int long long
#define mod 998244353
int32_t main() {
    int n, ans = 0;
    cin >> n;
    vector<int> a(n);
    vector<bool> vis(n + 5);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (a[i] != -1) vis[a[i]] = 1;
    }
    int cnt = 0, pre = -1;
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) {
            while (pre != -1) {
                ans += abs(i - pre - 1) * cnt;
                pre--;
            }
            cnt = 0;
            pre = i;
        }
        cnt++;
    }
    while (pre != -1) {
        ans += abs(n - pre - 1) * cnt;
        pre--;
    }
    int inv = (ans % mod) * (n * (n - 1) / 2) % mod;
    int x = inv % mod;
    int y = (n * (n - 1) / 2) % mod;
    for (int i = 1; i <= n; i++) {
        x *= i;
        x %= mod;
    }
    y = __gcd(x, y);
    x = x / y;
    y /= y;
    x %= mod;
    y %= mod;
    cout << (x * y % mod) % mod << endl;
    return 0;
}
