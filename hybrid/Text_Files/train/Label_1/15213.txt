#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define fastio ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);

int n, m, ans = 0;
vector<vector<char>> arr;
vector<vector<int>> vis;

bool valid(int x, int y) {
    return (x >= 0 && x < n && y >= 0 && y < m && arr[x][y] != '.');
}

void dfs(int x, int y, int d, int cnt) {
    if(d == 4) {
        ans = max(ans, cnt);
        return;
    }
    for(int i = -1; i <= 1; i++) {
        for(int j = -1; j <= 1; j++) {
            if(abs(i) == abs(j)) continue;
            int nx = x + i, ny = y + j;
            if(valid(nx, ny) && !vis[nx][ny] && arr[nx][ny] == "DIM"[d]) {
                vis[nx][ny] = 1;
                dfs(nx, ny, d + 1, cnt + 1);
                vis[nx][ny] = 0;
            }
        }
    }
}

int main() {
    fastio;
    cin >> n >> m;
    arr.resize(n, vector<char>(m));
    vis.resize(n, vector<int>(m));
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            cin >> arr[i][j];
        }
    }
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            if(arr[i][j] == 'D') {
                vis[i][j] = 1;
                dfs(i, j, 0, 0);
                vis[i][j] = 0;
            }
        }
    }
    if(ans) cout << ans << endl;
    else cout << "Poor Dima!" << endl;
    return 0;
}
