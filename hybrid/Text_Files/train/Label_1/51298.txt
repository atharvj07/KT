import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int[] freq = new int[n + 1];
        for (int i = 0; i < n; i++) {
            freq[a[i]]++;
        }
        int maxFreq = 0;
        for (int i = 1; i <= n; i++) {
            maxFreq = Math.max(maxFreq, freq[i]);
        }
        int ans = 0;
        int curFreq = 0;
        int left = 0;
        int right = 0;
        while (right < n) {
            curFreq += (freq[a[right]] == maxFreq ? 1 : 0);
            while (curFreq > 1) {
                curFreq -= (freq[a[left]] == maxFreq ? 1 : 0);
                left++;
            }
            ans = Math.max(ans, right - left + 1);
            right++;
        }
        System.out.println(ans);
    }
}
