import sys

def solve(n, p):
    vis = [False] * (n + 1)
    ans = [0] * (n + 1)
    for i in range(1, n + 1):
        if not vis[i]:
            cnt = 0
            j = i
            while not vis[j]:
                cnt += 1
                vis[j] = True
                j = p[j - 1]
            for k in range(cnt):
                ans[j] = cnt
                j = p[j - 1]
    return ans[1:]

for _ in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    print(*solve(n, p))
