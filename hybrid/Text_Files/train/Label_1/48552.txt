import java.util.*;

class Main {

    private static final int MAX = 100;
    private static int[] e = new int[MAX];

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        for (int i = 0; i < 12; i++) {
            e[i] = in.nextInt();
        }
        if (canMakeRectangularParallelepiped()) {
            System.out.println("yes");
        } else {
            System.out.println("no");
        }
    }

    private static boolean canMakeRectangularParallelepiped() {
        Arrays.sort(e);
        int[] a = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = e[i];
        }
        for (int i = 0; i < 3; i++) {
            for (int j = i + 1; j < 4; j++) {
                if (e[4 + i] != a[j]) {
                    return false;
                }
            }
        }
        return true;
    }
}
