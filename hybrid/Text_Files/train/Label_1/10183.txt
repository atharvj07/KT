#include<bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int &x: a) cin >> x;
        vector<int> adj[n + 2];
        for (int i = 1; i <= n; i++) {
            if (i != n) adj[i].push_back(i + 1);
            if (a[i - 1] == 0) adj[i].push_back(n + 1);
            else adj[n + 1].push_back(i);
        }
        vector<int> vis(n + 2, 0);
        vector<int> ans;
        function<void(int)> dfs = [&](int u) {
            vis[u] = 1;
            ans.push_back(u);
            for (int v: adj[u])
                if (!vis[v]) dfs(v);
        };
        dfs(1);
        if (ans.size() == n + 1) {
            for (int x: ans) cout << x << ' ';
            cout << '\n';
        } else cout << -1 << '\n';
    }
    return 0;
}
