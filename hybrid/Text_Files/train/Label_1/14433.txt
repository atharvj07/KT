#include <bits/stdc++.h>
using namespace std;
int n;
string s;
int main() {
    cin >> n >> s;
    for (int i = 1; i < n; i++) {
        if (s[i] == '0' && s[i - 1] == '1')
            swap(s[i], s[i - 1]);
    }
    string t = "";
    for (int i = 0; i < n; i++) {
        if (s[i] == '1') t += '1';
        if (i < n - 1 && s[i] == '1' && s[i + 1] == '1') {
            t += '0';
            i++;
        }
    }
    cout << t;
    return 0;
}
