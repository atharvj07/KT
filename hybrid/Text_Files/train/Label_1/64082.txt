n = int(input())
p = list(map(int,input().split()))
l = []; i = n+1; t = 1

for x in p:
    if x == i:
        l += [(i,i+1)]; i += 2

for x in reversed(p):
    if x == i:
        l += [(i,i-1)]; i -= 2

for x in p:
    if x > t and x < i:
        if t == p[x-1]:
            if t+1 == p[x+1]:
                l += [(x,x+1)]; t += 1
            else:
                l += [(x,x-1)]; t += 1
        else:
            if t+1 == p[x-1]:
                l += [(x+1,x)]; t += 1
            else:
                l += [(x-1,x)]; t += 1

print(len(l))
for x,y in l: print(x,y)
