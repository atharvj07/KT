#include <bits/stdc++.h>
using namespace std;
struct point{int x,y;};
bool check(vector<point>& p, vector<point>& s, int a, int b)
{
    double ax = s[a].x - p[a].x, ay = s[a].y - p[a].y;
    double bx = s[b].x - p[b].x, by = s[b].y - p[b].y;
    if((ax * by - ay * bx)==0) return false;
    return ((ax * bx + ay * by)<0 || (ax * bx+ay*by)>0);
}
int main()
{
    int R, B,x,y; cin>>R>>B; vector<point> p(R+1), s(B+1);
    for(int i=0; i<=R; i++) {cin>>x>>y; p[i].x=x; p[i].y=y;}
    for(int i=0; i<=B; i++) {cin>>x>>y; s[i].x=x; s[i].y=y;}
    bool ok= true;
    for(int i=0; i<=R; i++){
        for(int j=0; j<=B; j++){
            if(p[i].x==s[j].x&&p[i].y==s[j].y) continue;
            for(int k=i+1; k<=R; k++){
                if(p[i].x==p[k].x&&p[i].y==p[k].y) continue;
                if(check(p,s,i,j) &&check(p,s,k,j)) {ok=false;}
            }
        }
    }
    cout<<(ok?"Yes":"No")<<endl;
    return 0;
}
