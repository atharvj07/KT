import java.util.Scanner;
import java.util.HashSet;

public class RageOfEmpires {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        HashSet<Integer> army = new HashSet<>();

        for (int i = 0; i < q; i++) {
            String[] event = scanner.nextLine().split(" ");
            int type = Integer.parseInt(event[0]);
            int personality = Integer.parseInt(event[1]);

            switch (type) {
                case 1:
                    army.add(personality);
                    break;
                case 2:
                    army.remove(personality);
                    break;
                case 3:
                    int leadership = Integer.parseInt(event[2]);
                    int count = 0;
                    for (int p : army) {
                        if ((p ^ personality) > leadership) {
                            count++;
                        }
                    }
                    System.out.println(count);
                    break;
            }
        }
    }
}
