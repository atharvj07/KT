import sys
import math

n = int(input())
a = [list(map(int, input().split())) for _ in range(n)]

up = sum(aa[0] for aa in a)
down = sum(aa[1] for aa in a)
up = (up + 1) // 2
down = (down + 1) // 2

ans = up + down
div = 0
for aa in a:
    x, y = aa
    if (x + y) & 1:
        div = 1

if up != down and not div:
    ans = -1
print(ans)
