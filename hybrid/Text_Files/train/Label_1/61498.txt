import sys
input = sys.stdin.readline

def solve():
    mod = 998244353
    n, m = map(int, input().strip().split())
    grid = [list(input()) for _ in range(n)]
    dp = [[[[-1] * 4 for _ in range(m+1)] for _ in range(n+1)] for _ in range(1<<m)]
    
    def get_best(mask, r, c, dir):
        if r == n or c == m:
            return 0
        if dp[mask][r][c][dir] != -1:
            return dp[mask][r][c][dir]
        res = 0
        if grid[r][c] == 'o':
            if dir == 0:
                if (mask & (1 << c)) == 0:
                    res = max(res, get_best(mask, r+1, c, 0) + 1)
                if c + 1 < m and (mask & (1 << (c+1))) == 0:
                    res = max(res, get_best(mask, r+1, c+1, 0) + 1)
            elif dir == 1:
                if (mask & (1 << c)) == 0:
                    res = max(res, get_best(mask, r, c, 0) + 1)
                if r + 1 < n and (mask & (1 << c)) == 0:
                    res = max(res, get_best(mask, r+1, c, 0) + 1)
            elif dir == 2:
                if (mask & (1 << c)) == 0:
                    res = max(res, get_best(mask | (1 << c), r, c, 1) + 1)
            elif dir == 3:
                if r + 1 < n and (mask & (1 << c)) == 0:
                    res = max(res, get_best(mask | (1 << c), r+1, c, 1) + 1)
        res = max(res, get_best(mask, r + (dir == 0), c + (dir == 1), dir))
        dp[mask][r][c][dir] = res
        return res
    
    result = get_best(0, 0, 0, 0)
    print(result)

solve()
