import java.util.Scanner;

public class MoneyWithdrawal {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();

        int operations = 0;
        while (N > 0) {
            // Find the largest power of 6 that is less than or equal to N.
            int powerOf6 = 0;
            while (Math.pow(6, powerOf6) <= N) {
                powerOf6++;
            }
            powerOf6--;

            // Find the largest power of 9 that is less than or equal to N.
            int powerOf9 = 0;
            while (Math.pow(9, powerOf9) <= N) {
                powerOf9++;
            }
            powerOf9--;

            // Withdraw the larger amount.
            if (Math.pow(6, powerOf6) > Math.pow(9, powerOf9)) {
                N -= Math.pow(6, powerOf6);
            } else {
                N -= Math.pow(9, powerOf9);
            }

            operations++;
        }

        System.out.println(operations);
    }
}
