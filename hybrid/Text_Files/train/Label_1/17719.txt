#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'R') {
      cnt++;
      if (i + 1 < n && s[i + 1] == 'B') {
        cnt++;
        i++;
      }
    }
  }
  cout << cnt << endl;
  return 0;
}
