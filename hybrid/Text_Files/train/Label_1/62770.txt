def solve():
    for _ in range(int(input())):
        n = int(input())
        a = list(map(int, input().split()))
        sa = sorted(a)
        ev = [x for x in sa if not x%2]
        od = [x for x in sa if x%2]
        if sum(ev) > sum(od):
            print("Alice")
        elif sum(od) > sum(ev):
            print("Bob")
        else:
            print("Tie")

solve()
