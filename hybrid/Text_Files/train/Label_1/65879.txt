n, s, t = map(int, input().split())
p = list(map(int, input().split()))

visited = [False] * (n + 1)
vis = [0] * (n + 1)
ans = 1e9
vis[s] = 1

for i in range(1, n + 1):
    ns = p[vis[s] - 1]
    vis[s] += 1
    if ns == t:
        ans = min(ans, vis[s] - 1)
    if visited[ns]:
        ans = 0
    visited[ns] = True
    s = ns

if ans == 1e9:
    ans = -1
print(ans)
