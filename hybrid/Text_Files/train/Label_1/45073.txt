import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int[] A = new int[N];
        int[] B = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = scanner.nextInt();
        }
        for (int i = 0; i < N; i++) {
            B[i] = scanner.nextInt();
        }
        scanner.close();

        int[][] diff = new int[N][N];
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                diff[i][j] = B[(i + j) % N] - A[i];
            }
        }

        int minOps = Integer.MAX_VALUE;
        for (int i = 0; i < N; i++) {
            int sum = 0;
            for (int j = 0; j < N; j++) {
                sum += diff[i][j];
            }
            if (sum % 3 == 0) {
                int ops = sum / 3;
                if (ops >= 0 && ops < minOps) {
                    minOps = ops;
                }
            }
        }

        if (minOps == Integer.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(minOps);
        }
    }
}
