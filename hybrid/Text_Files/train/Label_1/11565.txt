#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v1(n), v2(n);
  for (int i = 0; i < n; i++) cin >> v1[i];
  for (int i = 0; i < n; i++) cin >> v2[i];
  sort(v1.begin(), v1.end());
  sort(v2.begin(), v2.end());
  int sum = 0;
  for (int i = 0; i < n; i++) {
    if (i & 1)
      sum += v2[i];
    else
      sum += v1[i];
  }
  cout << sum << endl;
  return 0;
}
