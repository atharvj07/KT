#include <iostream>
#include <vector>
using namespace std;

int dfs1(int nod, vector<int> adj[], vector<int> &viz) {
    viz[nod]=1;
    int ok=0;
    for (auto x:adj[nod])
        if (!viz[x])
            ok|=dfs1(x,adj,viz);
        else if (viz[x]==1)
            ok=2;
    viz[nod]=2;
    return ok;
}

int main() {
    int t; cin>>t;
    while(t--) {
        int n,m; cin>>n>>m;
        vector<int> adj[n+1];
        for (int i=0; i<m; i++) {
            int x,y; cin>>x>>y;
            adj[x].push_back(y);
        }
        vector<int> viz(n+1,0);
        for (int i=1; i<=n; i++)
            if (!viz[i]) {
                if (dfs1(i,adj,viz)==2) {
                    for (int j=1; j<=n; j++)
                        cout<<"-1 ";
                    cout<<endl;
                    goto end;
                }
            }
        for (int i=1; i<=n; i++) {
            if (viz[i]==0) cout<<"0 ";
            else if (viz[i]==1) cout<<"1 ";
            else cout<<"2 ";
        }
        cout<<endl;
        end:;
    }
}
