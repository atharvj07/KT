import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        char[] s = br.readLine().toCharArray();
        char[] t = br.readLine().toCharArray();
        int[] cntS = new int[2]; // count of a, b in s
        int[] cntT = new int[2]; // count of a, b in t
        List<int[]> diffs = new ArrayList<>(); // record different indices
        for (int i = 0; i < n; i++) {
            if (s[i] != t[i]) {
                diffs.add(new int[]{i, s[i], t[i]});
                if (s[i] == 'a') {
                    cntS[0]++;
                } else {
                    cntS[1]++;
                }
                if (t[i] == 'a') {
                    cntT[0]++;
                } else {
                    cntT[1]++;
                }
            }
        }
        if (!Arrays.equals(cntS, cntT)) {
            System.out.println(-1);
        } else {
            int ans = diffs.size() / 2;
            System.out.println(ans);
            for (int i = 0; i < ans; i++) {
                System.out.println(diffs.get(i)[0] + 1 + " " + diffs.get(diffs.size() - i - 1)[0] + 1);
            }
        }
    }
}
