#include <bits/stdc++.h>

using namespace std;

bool is_palindrome(vector<int>& v, int l, int r) {
  while(l < r) {
    if(v[l++] != v[r--])
      return false;
  }
  return true;
}

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for(int i = 0; i < n; ++i) cin >> v[i];

  int ans = n;
  for(int i = 0; i < n; ++i) {
    for(int j = i + 1; j < n; ++j) {
      if(is_palindrome(v, i, j)) ans = min(ans, i + n - 1 - j);
    }
  }
  cout << ans << '\n';

  return 0;
}
