#include <bits/stdc++.h>
#define MOD 1000000007

using namespace std;
int32_t main() {
  int n, h;
  cin >> n >> h;
  vector<int> arr(n);
  for (int i = 0; i < n; i++) { cin >> arr[i]; }
  int dp[n + 1][h + 1];
  memset(dp, 0, sizeof(dp));
  dp[0][0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= h; j++) {
      dp[i][j] = dp[i - 1][j];
      if (j >= arr[i - 1]) { dp[i][j] = (dp[i][j] + dp[i - 1][j - arr[i - 1]]) % MOD; }
    }
  }
  cout << dp[n][h] << endl;
}
