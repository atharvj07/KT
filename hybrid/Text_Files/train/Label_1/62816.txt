for _ in range(int(input())):
    a, b, p = map(int, input().split())
    s = input()
    n = len(s)
    mn = n
    for i in range(n):
        if i > 0:
            p -= min(a, b)
        if p < 0:
            break
        if s[i] == s[n - 1]:
            continue
        if s[n - 1] == 'A':
            if s[i] == 'A':
                p += a
            else:
                p -= b
        else:
            if s[i] == 'B':
                p += b
            else:
                p -= a
        if p >= 0:
            mn = i + 1
    print(mn)
