#include <bits/stdc++.h>
using namespace std;
int n, a[200001], ans[200001];
int main() {
  int t;
  cin >> t;
  while (t--) {
    memset(ans, 0, sizeof(ans));
    cin >> n;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      for (int j = max(0, i - a[i] + 1); j <= i; j++) ans[j] = 1;
    }
    for (int i = 0; i < n; i++) { cout << ans[i] << " "; }
    cout << '\n';
  }
  return 0;
}
