#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<vector<int>> a(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) cin >> a[i][j];
  }
  vector<int> cnt(n);
  for (int i = 0; i < k; i++) {
    int x, y;
    cin >> x >> y;
    x--;
    y--;
    for (int j = 0; j < n; j++) {
      if (a[j][y] && j != x) cnt[j]++;
    }
  }
  for (int i = 0; i < n; i++) cout << cnt[i] << " ";
  cout << "\n";
  return 0;
}
