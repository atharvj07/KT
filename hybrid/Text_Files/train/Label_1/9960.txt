#include<iostream>
#include<vector>
using namespace std;

int main()
{
    int n, i, j;
    cin >> n;
    vector<int> L, R;
    string s;
    cin >> s;
    for(i = 0; i < n; i++)
    {
        if(s[i] == 'L')
        {
            L.push_back(i+1);
        }
        else
        {
            R.push_back(i+1);
        }
    }
    i = 0, j = 0;
    while(i < L.size() && j < R.size())
    {
        cout << L[i] << " " << R[j] << endl;
        i++;
        j++;
    }
    return 0;
}
