#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n, m, k[1001], need[1001];
    cin >> n >> m;
    int ans = 0;
    for (int i = 0; i < n; i++)
    {
        cin >> k[i];
        need[i] = k[i] + 1;
    }
    bool ch;
    while (1)
    {
        ch = 0;
        ans++;
        for (int i = 0; i < n; i++)
        {
            if (need[i] <= 0)
                continue;
            need[i] -= 2;
            ch = 1;
        }
        for (int i = 0; i < m; i++)
        {
            int p, q;
            cin >> p >> q;
            p--;
            q--;
            if (need[q] > 0 and p == ans)
                need[q]--;
        }
        if (ch == 0)
            break;
    }
    cout << ans << endl;
    return 0;
}
