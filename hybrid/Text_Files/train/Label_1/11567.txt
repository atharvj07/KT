#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int& x : a) cin >> x;
    for (int& x : b) cin >> x;
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    ll sum = 0;
    for (int i = 0, j = n - 1; i < n; i++, j--) {
        sum += max(a[i], b[j]);
        if (i + 1 < n) sum += max(a[i + 1], b[j]);
    }
    cout << sum << '\n';
}
