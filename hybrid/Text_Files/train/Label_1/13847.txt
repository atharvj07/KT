#include<bits/stdc++.h>
using namespace std;
int n,a[110];
int main(){
	string s;
	cin>>s;
	n=s.size();
	for(int i=0;i<n;i++){
		if(s[i]=='Q') a[i]=1;
		else if(s[i]=='A') a[i]=a[i-1]+1;
		else a[i]=a[i-1];
	}
	int ans=0;
	for(int i=0;i<n;i++){
		if(s[i]=='Q') ans+=a[i]*(a[n-1]-a[i]);
	}
	cout<<ans<<'\n';
	return 0;
}
