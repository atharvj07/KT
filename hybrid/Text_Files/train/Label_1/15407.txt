#include <cstdio>

typedef long long ll;
const int mod = (int)1e6;

int main() {
  int q, m, x = 0, y = 0, t = 0, h = 0, ans = 0;
  ll mana;
  scanf("%d %lld", &q, &mana);
  while (q--) {
    int k;
    scanf("%d", &k);
    if (k == 1) {
      scanf("%d %d", &x, &y);
      x = (x + t) % mod + 1;
      y = (y + h) % mod + 1;
    } else {
      scanf("%d %d", &t, &h);
      t = (t + ans) % mod + 1;
      h = (h + ans) % mod + 1;
      ans = (ll)t * x <= mana && (ll)t * y <= mana;
      printf("%s\n", ans ? "YES" : "NO");
    }
  }
  return 0;
}
