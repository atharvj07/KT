#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x;
  cin >> n >> x;
  int sum = 0;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int num;
    cin >> num;
    sum += num;
    if (num < 0) ans++;
  }
  if (sum == 0) {
    cout << 0;
    return 0;
  }
  if (sum > 0) {
    if (sum % x == 0) {
      cout << ans;
      return 0;
    } else {
      cout << ans + 1;
      return 0;
    }
  } else {
    int temp = ceil((-1 * sum) / x);
    if (ans >= temp) {
      cout << ans - temp;
      return 0;
    } else {
      cout << -1;
      return 0;
    }
  }
}
