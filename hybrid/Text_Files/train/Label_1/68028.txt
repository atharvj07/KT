import math
n, x0, y0 = map(int, input().split())
points = [[int(x) for x in input().split()] for _ in range(n)]
INF = math.inf
ans = INF
for i in range(n):
    xi, yi = points[i]
    if xi == x0:
        continue
    k = (yi - y0) / (xi - x0)
    b = y0 - k * x0
    cnt = 1
    for j in range(n):
        xj, yj = points[j]
        if xi == xj:
            if xj == x0:
                cnt += 1
        elif (yj - y0) / (xj - x0) == k:
            cnt += 1
    ans = min(ans, n - cnt)
print(ans)
