import java.util.*;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        BigInteger I = BigInteger.valueOf(sc.nextInt());
        int[] a = new int[n];
        int max = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            max = Math.max(max, a[i]);
        }
        BigInteger k = BigInteger.valueOf(max).divide(I.multiply(BigInteger.valueOf(8)));
        k = k.add(BigInteger.ONE);
        k = k.pow(2);
        System.out.println(k.subtract(BigInteger.valueOf(max)).intValue());
    }
}
