#include <iostream>
#include <vector>

int main() {
  std::string s;
  std::cin >> s;

  std::vector<int> cnt(s.size() + 1, 0);

  for (int i = s.size() - 1; i >= 0; i--) {
    cnt[i] += cnt[i + 1];

    if (i + 3 < s.size() && s.substr(i, 4) == "bear") cnt[i]++;
  }

  long long ans = 0;
  for (int i = 0; i < s.size(); i++) ans += cnt[i];

  std::cout << ans << std::endl;

  return 0;
}
