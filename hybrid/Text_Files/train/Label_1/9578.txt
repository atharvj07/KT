#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<long long, int>> requests(n);
  for (auto& p : requests) cin >> p.first >> p.second;
  sort(requests.begin(), requests.end());
  vector<long long> starts(n, -1), ends(n, -1);
  for (int i = 0, start = 1; i < n; ++i) {
    const auto& [s, d] = requests[i];
    if (s <= start) {
      starts[i] = start;
      ends[i] = start + d - 1;
      start += d;
    } else {
      starts[i] = s;
      ends[i] = s + d - 1;
    }
  }
  for (int i = 0; i < n; ++i) cout << starts[i] << ' ' << ends[i] << '\n';
  return 0;
}
