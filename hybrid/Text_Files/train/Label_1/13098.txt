#include <cmath>
#include <cstdio>
#include <vector>

using namespace std;

int n , s;

double get_ans(const vector<double> &x , const vector<double> &y , const vector<double> &k) {
    double l = 0 , r = 1e9;
    int flag = 0;
    while (fabs(r - l) > 1e-6) {
        double mid = (l + r) / 2;
        double tot = 0;
        for (int i = 0;i < n;i++) {
            if (x[i] * x[i] + y[i] * y[i] <= mid * mid) {
                tot += k[i];
            }
        }
        if (s + tot >= 1e6) {
            r = mid;
            flag = 1;
        }
        else {
            l = mid;
        }
    }
    if (flag) return r;
    else return -1;
}

int main() {
    scanf("%d %d" , &n , &s);
    vector<double> x(n + 5) , y(n + 5) , k(n + 5);
    for (int i = 0;i < n;i++) {
        scanf("%lf %lf %lf" , &x[i] , &y[i] , &k[i]);
    }
    double ans = get_ans(x , y , k);
    if (ans == -1) {
        printf("-1\n");
    }
    else {
        printf("%.6f\n" , ans);
    }
    return 0;
}
