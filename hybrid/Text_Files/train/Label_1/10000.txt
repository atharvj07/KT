#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2e5+1;

long long arr1[MAXN], arr2[MAXN], a[MAXN], b[MAXN];
long long n;
int main() {
  cin >> n;
  for (long long i = 1; i <= n; i++)
    cin >> a[i], cin >> b[i], arr1[i] = a[i] - b[i];
  long long ans = 0;
  sort(arr1 + 1, arr1 + n + 1);
  for (long long i = 1; i <= n; i++) {
    long long l = i;
    long long r = n;
    long long best = -1;
    while (l <= r) {
      long long mid = (l + r) / 2;
      if (arr1[mid] >= -arr1[i]) {
        best = mid;
        l = mid + 1;
      } else {
        r = mid - 1;
      }
    }
    if (best != -1)
      ans += (n - best);
  }
  cout << ans << endl;
}
