#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n), b(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        for (int i = 0; i < n; i++) cin >> b[i];
        vector<pair<int, pair<int, int>>> v(n);
        for (int i = 0; i < n; i++) {
            v[i].first = a[i];
            v[i].second.first = b[i];
            v[i].second.second = i + 1;
        }
        vector<int> ans(n);
        sort(v.begin(), v.end());
        for (int i = 0, j = n - 1; i < j; i++, j--) {
            ans[v[i].second.second - 1] = ans[v[j].second.second - 1] = 1;
        }
        for (int i = 0; i < n; i++) cout << ans[i];
        cout << '\n';
    }
    return 0;
}
