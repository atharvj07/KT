#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 5e5 + 5;
int a[MAXN], b[MAXN], t[MAXN];
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int m, k, n, s;
  cin >> m >> k >> n >> s;
  for (int i = 0; i < m; i++) cin >> a[i];
  for (int i = 0; i < s; i++) { cin >> b[i]; }
  int c[k + 5];
  for (int i = 0; i <= k; i++) c[i] = 0;
  for (int i = 0; i < m; i++) c[a[i]]++;
  int rem = 0;
  for (int i = 0; i < s; i++) {
    if (c[b[i]] == 0) { rem = -1; }
    c[b[i]]--;
  }
  for (int i = 0; i < m; i++) {
    c[a[i]]++;
    for (int j = 0; j < s; j++) {
      if (c[b[j]] <= 0) { rem = i + 1; }
      c[b[j]]--;
    }
  }
  if (rem == -1)
    cout << -1;
  else {
    int cnt = 0;
    cout << rem << "\n";
    for (int i = 0; i < m; i++) {
      if (i < rem) {
        cout << i + 1 << " ";
        t[i] = 1;
      }
      for (int j = 0; j < s; j++) {
        if (a[i] == b[j]) { t[i] = 1; }
      }
    }
    for (int i = 0; i < m; i++) {
      if (t[i] == 0) {
        cout << i + 1 << " ";
        cnt++;
      }
      if (cnt == n - 1) break;
    }
    cout << "\n";
  }
  return 0;
}
