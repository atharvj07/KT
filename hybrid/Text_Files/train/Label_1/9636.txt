#include <bits/stdc++.h>
using namespace std;
int p[200005], suf[200005];
int main() {
  string s;
  cin >> s;
  int n = s.size();
  for (int i = n - 1, sum = 0; i >= 0; i--) {
    suf[i] = sum;
    sum = (sum + s[i] - '0') % 3;
  }
  map<int, int> m;
  for (int i = 0, sum = 0; i < n; i++) {
    sum = (sum + s[i] - '0') % 3;
    if (!i)
      p[i] = sum;
    else
      p[i] = p[i - 1] + sum;
    m[p[i]]++;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int need = (3 - ((s[i] - '0') % 3)) % 3;
    ans += m[p[i] + need];
    if ((p[i] + need) % 3 == 0) ans--;
    if (need == 0 && suf[i] == 0) ans--;
    if (need == 0 && i) {
      need = (3 - ((s[i - 1] - '0') % 3)) % 3;
      if (p[i - 1] + need == p[i]) ans--;
    }
  }
  cout << ans << endl;
  return 0;
}
