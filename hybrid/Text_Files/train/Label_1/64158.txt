import sys

n,m,k=map(int,input().split())
e=[]
for _ in range(m):
    e.append(list(map(int,input().split())))
e.sort()
d=[[i==1 for i in range(m)] for j in range(n)]
vis=[0]*n
for i,j in e:
    vis[i-1]=vis[j-1]=1
    if i==1:
        d[j-1]=[False]*m
    if j==1:
        d[i-1]=[False]*m
    d[i-1][e.index([i,j])]=d[j-1][e.index([i,j])]=True
totrep=[[0]*m for i in range(k)]
t=0
for i in range(1,n):
    if not vis[i]:
        continue
    cur=[False]*m
    for j in range(m):
        if d[i][j]==1:
            cur[j]=True
    if totrep.count(cur)==0:
        for j in range(m):
            if cur[j]==1:
                d[1][j]=d[1][j] and d[i][j]
        totrep[t]=cur
        t+=1
        if t==k:
            break
for i in range(t):
    for j in range(m):
        if totrep[i][j]==True:
            sys.stdout.write("1")
        else:
            sys.stdout.write("0")
    sys.stdout.write("\n")
