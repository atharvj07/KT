#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  std::ios::sync_with_stdio(false);
  int q;
  std::cin >> q;
  while (q--) {
    int n;
    std::cin >> n;
    std::vector<int> p(n);
    std::vector<std::pair<int, int>> ops;
    for (int i = 0; i < n; ++i) {
      std::cin >> p[i];
      p[i]--;  // switch to 0-based indexing
    }
    for (int i = 0; i + 1 < n; ++i) {
      int min_val = p[i];
      int min_idx = i;
      for (int j = i + 1; j < n; ++j) {
        if (p[j] < min_val) {
          min_val = p[j];
          min_idx = j;
        }
      }
      if (min_idx != i) {
        ops.push_back({i, min_idx});
        std::swap(p[i], p[min_idx]);
      }
    }
    std::cout << p[0] + 1;
    for (int i = 1; i < n; ++i) {
      std::cout << " " << p[i] + 1;
    }
    std::cout << "\n";
  }
  return 0;
}
