import java.util.*;
import java.io.*;

public class Main {

  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(in.readLine());
    int[] strength = new int[n];
    StringTokenizer st = new StringTokenizer(in.readLine());
    for (int i = 0; i < n; i++) {
      strength[i] = Integer.parseInt(st.nextToken());
    }
    int m = Integer.parseInt(in.readLine());
    boolean[] banned = new boolean[n];
    boolean[] picked = new boolean[n];
    int[] team1 = new int[n], team2 = new int[n];
    for (int i = 0; i < m; i++) {
      st = new StringTokenizer(in.readLine());
      char action = st.nextToken().charAt(0);
      int team = Integer.parseInt(st.nextToken()) - 1;
      if (action == 'p') {
        for (int j = 0; j < n; j++) {
          if (!banned[j] && !picked[j]) {
            if (team == 0) {
              team1[j] = strength[j];
            } else {
              team2[j] = strength[j];
            }
            picked[j] = true;
            break;
          }
        }
      } else {
        for (int j = 0; j < n; j++) {
          if (!banned[j] && !picked[j]) {
            banned[j] = true;
            break;
          }
        }
      }
    }
    Arrays.sort(team1);
    Arrays.sort(team2);
    int sum1 = 0, sum2 = 0;
    for (int i = 0; i < n; i++) {
      sum1 += team1[i];
      sum2 += team2[i];
    }
    System.out.println(sum1 - sum2);
  }
}
