n = int(input())
a = list(map(int, input().split()))

max_odd_sum = float('-inf')
total_sum = 0
min_pos_odd = float('inf')

for i in a:
    if i > 0:
        total_sum += i
    if i % 2 != 0:
        max_odd_sum = max(max_odd_sum, i)
        if i > 0:
            min_pos_odd = min(min_pos_odd, i)

if total_sum % 2 == 0:
    total_sum -= min_pos_odd

print(total_sum)
