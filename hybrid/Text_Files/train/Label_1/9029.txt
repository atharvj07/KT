#include <iostream>
#include <algorithm>
#include <vector>
#include <cstring>
using namespace std;

const int MAXN=2e5+1;
int n, f[11], ans[MAXN], dp[MAXN];
char a[MAXN];

int main() {
    cin >> n >> a >> f[1];
    for (int i=2; i<=9; i++) cin >> f[i];
    for (int i=n; i>=1; i--) {
        ans[i] = a[i]-'0';
        for (int j=i; j+1<=n && a[j+1]>=a[j]; j++)
            dp[i] = max(dp[i], dp[j+1] + (f[ans[j]] > ans[j]));
        ans[i] = max(ans[i], ans[i+1]);
        dp[i] = max(dp[i], dp[i+1]);
    }
    for (int i=1; i<=n; i++) ans[i] = max(ans[i], ans[i-1]);
    for (int i=1; i<=n; i++) ans[i] = max(ans[i], dp[i] + ans[i+dp[i]]);
    for (int i=1; i<=n; i++) ans[i] = f[ans[i]];
    int flg = 1;
    for (int i=1; i<=n; i++) {
        if (!flg && ans[i]) cout << ans[i];
        else if (ans[i]) {
            cout << ans[i];
            flg = 0;
        }
    }
    if (flg) cout << 0;
    cout << endl;
}
