#include <bits/stdc++.h>
#define ll long long

using namespace std;

ll gcd(ll a, ll b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    ll a, b;
    cin >> a >> b;
    if (a == 0 || b == 0)
      cout << 0 << " " << 0 << endl;
    else {
      ll g = gcd(a, b);
      cout << g << " " << min(a % g, b % g) << endl;
    }
  }
  return 0;
}
