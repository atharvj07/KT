#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> v1(n), v2(m);
  for (int i = 0; i < n; i++) cin >> v1[i];
  for (int i = 0; i < m; i++) cin >> v2[i];
  sort(v1.begin(), v1.end());
  sort(v2.begin(), v2.end());
  int ans = 0;
  for (int i = 0, j = 0; i < n || j < m;) {
    if (i < n && j < m) {
      if (v1[i] == v2[j]) {
        ans += 2;
        i++;
        j++;
      } else if (v1[i] < v2[j]) {
        i++;
      } else {
        j++;
      }
    } else if (i == n) {
      j++;
    } else {
      i++;
    }
  }
  cout << ans << endl;
  return 0;
}
