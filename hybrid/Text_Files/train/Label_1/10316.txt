#include <iostream>
#include <vector>
using namespace std;

typedef long long ll;

ll solve(const vector<ll>&p, ll k) {
  k = (k + 99) / 100;
  ll ans = 0, sum = 0;
  for(int i = 0; i < p.size(); i++) {
    if(p[i] >= sum) continue;
    ll d = (sum + k - 1) / k - sum;
    d = min(d, sum - p[i]);
    sum += d;
    ans += d;
  }
  return ans;
}

int main() {
  int t; cin >> t;
  while(t--) {
    int n, k; cin >> n >> k;
    vector<ll>p(n);
    for(auto &x : p) cin >> x;
    cout << solve(p, k) << endl;
  }
  return 0;
}
