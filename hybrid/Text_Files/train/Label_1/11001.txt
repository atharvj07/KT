#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> x(n), y(m);
  for (int i = 0; i < n; i++) { cin >> x[i]; }
  for (int i = 0; i < m; i++) { cin >> y[i]; }
  sort(x.begin(), x.end());
  sort(y.rbegin(), y.rend());
  int ans = 0;
  int cur = 0;
  for (int i = 0, j = 0; i < n;) {
    if (x[i] + cur <= y[j]) {
      cur += x[i++];
      ans++;
    } else {
      cur -= y[j++];
    }
  }
  cout << ans << endl;
  return 0;
}
