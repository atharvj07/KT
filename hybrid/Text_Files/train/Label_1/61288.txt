MOD = 998244353

n, k = map(int, input().split())
a = list(map(int, input().split()))

if n == 1:
    print(k if a[0] == -1 else 1)
else:
    dp = [[0] * 2 for _ in range(2)]
    if a[0] == -1:
        dp[1][0] = k
        dp[0][1] = 0
    else:
        dp[1][0] = 1
        dp[0][1] = 0

    for i in range(1, n):
        new_dp = [[0] * 2 for _ in range(2)]
        if a[i] == -1:
            new_dp[0][0] = (dp[1][0] * (k - 1) + dp[0][0] * (k - 2)) % MOD
            new_dp[0][1] = dp[1][0]
            new_dp[1][0] = (dp[1][0] * k) % MOD
            new_dp[1][1] = dp[0][1]
        else:
            if a[i] != a[i - 1] or a[i - 1] == -1:
                new_dp[0][0] = dp[0][0]
                new_dp[1][0] = dp[1][0]
            new_dp[0][1] = dp[1][0]
            new_dp[1][1] = dp[0][1]
        dp = new_dp

    print(sum(dp[1]) % MOD)
