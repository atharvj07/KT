#include <vector>
#include <cstdio>

using namespace std;

#define min(x, y) ((x) < (y) ? (x) : (y))
#define abs(x) ((x) >= 0 ? (x) : -(x))

struct state {
    int x, y, len;
    state(int x, int y, int len) {
        this->x = x;
        this->y = y;
        this->len = len;
    }
    state move(char c) {
        if (c == 'U') return state(x, y + 1, len + 1);
        if (c == 'D') return state(x, y - 1, len + 1);
        if (c == 'L') return state(x - 1, y, len + 1);
        if (c == 'R') return state(x + 1, y, len + 1);
        return *this;
    }
};

const int SIZE = 5e5;
vector<state> dp(SIZE, state(0, 0, 0));
vector<char> trace(SIZE);

int main() {
    int n; scanf("%d", &n);
    char c; scanf(" %c", &c);
    for (int i = 0; i < n; i++) {
        scanf(" %c", &c);
        dp[i + 1] = dp[i].move(c);
        trace[i + 1] = c;
    }
    int minlen = n;
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (dp[i].x == dp[j].x && dp[i].y == dp[j].y) {
                minlen = min(minlen, j - i);
                break;
            }
        }
    }
    printf("%d\n", minlen);
    return 0;
}
