#include <cstdio>
#include <cmath>
#include <vector>
using namespace std;

int main() {
  int n, k, m, a;
  scanf("%d %d %d %d", &n, &k, &m, &a);
  vector<pair<int, int>> d(n);
  vector<int> c(n);
  vector<bool> chan(n);
  for (int i = 0; i < a; ++i) {
    int x;
    scanf("%d", &x);
    ++c[x - 1];
    if (!chan[x - 1]) {
      chan[x - 1] = 1;
      ++d[x - 1].second;
    }
    d[x - 1].first = i;
  }
  for (int i = 0; i < n; ++i) {
    int r = 1;
    if (c[i] + (m - a) >= k) r = 1;
    else if (c[i] + (m - a) > 0) r = 2;
    else r = 3;
    if (r == 3) printf("3 ");
    else {
      bool t = 1;
      int j;
      for (j = 0; j < n; ++j) {
        if (c[j] > c[i] && (j + 1 == i || (c[j] == c[i] && d[j] > d[i]))) t = 0;
      }
      for (j = 0; j < n; ++j) {
        if (c[j] + (m - a) >= k && (j + 1 == i || (c[j] + (m - a) == c[i] + (m - a) && d[j] < d[i])))
          t = 0;
      }
      printf("%d ", 1 + t);
    }
  }
  return 0;
}
