import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        java.util.Scanner scan = new java.util.Scanner(System.in);
        int t = scan.nextInt();
        for (int i = 0; i < t; ++i) {
            int n = scan.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; ++j) a[j] = scan.nextInt();
            Arrays.sort(a);
            if (n == 2 || a[0] == a[1]) System.out.println("YES");
            else System.out.println("NO");
        }
    }
}
