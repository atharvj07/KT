#include <bits/stdc++.h>
using namespace std;
#define SZ(x) ((int)x.size())
const int N = 1e5 + 1, M = 3;
int n, m, f[N];
bool p[N][M][10], ok[M][10];
array<int, 10> a[M], b;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> f[i];
    for (int j = 0; j < f[i]; j++) {
      int x;
      cin >> x;
      p[i][0][x] = 1;
    }
  }
  for (int i = 0; i < m; i++) {
    cin >> a[i][0] >> a[i][1];
    for (int j = 0; j < a[i][1]; j++) {
      cin >> a[i][j + 2];
      p[i + m][1][a[i][j + 2]] = 1;
    }
  }
  int j1 = 0, j2 = 0, res = 0;
  for (int i = 0; i < n + m; i++) {
    for (int j = 0; j < n + m; j++) {
      if (i != j) {
        int c = 0;
        for (int k = 1; k <= 9; k++) ok[0][k] = ok[1][k] = 0;
        for (int k = 1; k <= 9; k++) {
          if (!p[i][0][k]) ok[0][k] = 1;
          if (!p[j][1][k]) ok[1][k] = 1;
        }
        for (int k = 1; k <= 9; k++) {
          if (ok[0][k] && ok[1][k]) c++;
        }
        if (c > res) {
          res = c;
          j1 = i;
          j2 = j;
        }
      }
    }
  }
  cout << j1 + 1 << " " << j2 + 1 << endl;
}
