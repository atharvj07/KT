import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong(), k = in.nextInt();
        int ans = 0;
        while (n % 10 != 0 && n % 10 != k) {
            n /= 10;
            ans++;
        }
        System.out.println(ans);
    }
}
