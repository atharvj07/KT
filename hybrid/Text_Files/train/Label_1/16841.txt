#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;
const int maxn=500+10;
int n,m,k,q;
pair<int,int> grid[maxn][maxn];
int main(){
  cin>>n>>m>>k>>q;
  for(int i=0;i<q;i++){
    int x,y,t;
    cin>>x>>y>>t;
    if(grid[x][y]<make_pair(1,t))grid[x][y]=make_pair(1,t);
  }
  int ans=-1;
  for(int i=1;i<=n;i++){
    for(int j=1;j<=m;j++){
      if(grid[i][j].first==1&&grid[i-1][j].first==1&&grid[i][j-1].first==1&&grid[i-1][j-1].first==1){
        if(ans==-1)ans=grid[i][j].second;
        else ans=min(ans,grid[i][j].second);
      }
    }
  }
  cout<<ans<<endl;
}
