import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String a = in.nextLine();
        String b = in.nextLine();
        int n = a.length();
        int ops = 0;
        for (int i = 0; i < n; i++) {
            if (a.charAt(i) != b.charAt(i)) {
                if (a.charAt(i) == '7') {
                    ops += 1;
                } else {
                    ops += 2;
                }
            }
        }
        System.out.println(ops);
    }
}
