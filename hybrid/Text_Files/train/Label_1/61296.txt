s=input()
t=input()
if s==t:
    print('0')
else:
    x=[]
    y=[]
    for i in range(len(s)):
        if s[i]!=t[i]:
            x.append(s[i])
            y.append(t[i])
            if len(x)==2 and x[0]==y[1] and x[1]==y[0]:
                x=[]
                y=[]
    if len(x)==0:
        print(len(set(s)))
        for i in range(0,len(s),2):
            print(s[i],s[i+1])
    else: print('-1')
