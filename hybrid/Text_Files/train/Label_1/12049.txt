#include <bits/stdc++.h>

using namespace std;

vector<int> a;
bool possible[700][700];
bool vis[700];

bool dfs(int l, int r) {
    if (l == r) {
        return true;
    }
    for (int i = l + 1; i <= r; i++) {
        if (!vis[i] && __gcd(a[l], a[i]) > 1) {
            vis[i] = true;
            if (dfs(l + 1, i - 1) && dfs(i, r)) {
                possible[l][r] = true;
                return true;
            }
            vis[i] = false;
        }
    }
    return false;
}

int main() {
    int n;
    cin >> n;
    a.resize(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    if (dfs(1, n)) {
        cout << "Yes" << endl;
    } else {
        cout << "No" << endl;
    }
    return 0;
}
