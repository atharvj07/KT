import java.util.Scanner;

class Main {
    static boolean[][] banned;
    static int n, m;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        banned = new boolean[n + 1][n + 1];
        for (int i = 0; i < m; i++)
            banned[in.nextInt()][in.nextInt()] = true;
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (!banned[1][i] && !banned[n][i])
                ans++;
            if (!banned[i][1] && !banned[i][n])
                ans++;
        }
        System.out.println(ans);
    }
}
