#include <bits/stdc++.h>
#define int long long

using namespace std;

const int mod = 1e9 + 7;

int32_t main() {
  int n, h;
  cin >> n >> h;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> dp(h + 1);
  dp[0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = h; j >= 0; j--) {
      dp[j + a[i]] = (dp[j + a[i]] + dp[j]) % mod;
    }
  }
  cout << dp[h] << endl;
}
