t = int(input())
for _ in range(t):
    n, c = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    days = 0
    savings = 0
    for i in range(n):
        if savings >= c:
            break
        if i == n-1:
            days += (c - savings + a[i] - 1) // a[i]
            break
        days_to_promotion = (b[i] - savings + a[i] - 1) // a[i] if savings < b[i] else 0
        savings += days_to_promotion * a[i] - b[i]
        days += days_to_promotion + 1
        if savings >= c:
            break
        min_days_to_goal = (c - savings + a[i+1] - 1) // a[i+1]
        if i == n-2 or min_days_to_goal < (b[i+1] - savings + a[i+1] - 1) // a[i+1]:
            days += min_days_to_goal
            break
    print(days)
