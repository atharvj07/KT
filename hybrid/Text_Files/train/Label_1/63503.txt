t = int(input())
for _ in range(t):
    n, m, rb, cb, rd, cd = map(int, input().split())
    dr, dc = 1, 1
    time = 0
    while True:
        if rb == rd or cb == cd:
            print(time)
            break
        if rb + dr > n or rb + dr < 1:
            dr = -dr
        if cb + dc > m or cb + dc < 1:
            dc = -dc
        rb += dr
        cb += dc
        time += 1
