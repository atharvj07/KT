#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;

    std::vector<long long> b(n);
    for (auto &el : b) std::cin >> el;

    std::vector<long long> a(n);
    a[0] = b[0];
    for (int i = 1; i < n; ++i) a[i] = a[i - 1] + b[i];

    for (auto &el : a) std::cout << el << ' ';

    return 0;
}
