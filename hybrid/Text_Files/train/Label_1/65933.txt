from collections import defaultdict
MOD = 998244353
n = int(input())
a = [int(x) for x in input().split()]
d = defaultdict(int)
s = 0
for x in a:
    s += 1
    d[x] += 1
ans = 1
for x in d.values():
    ans *= x+1
    ans %= MOD
ans -= 1
ans %= MOD
for x in d.values():
    x -= 1
    ans *= pow(x+1, MOD-2, MOD)
    ans %= MOD
print(ans)
