#include <bits/stdc++.h>
const int mod = 1e9+7;
int main() {
  long long n, x;
  std::cin >> n >> x;
  std::vector<long long> a(n);
  for (auto &ai : a) std::cin >> ai;
  long long s = 1, t = 0;
  for (auto &ai : a) {
    s = (s * x % mod + mod) % mod;
    t = (t + ai) % mod;
  }
  long long dv = std::__gcd(s, t);
  std::cout << dv % mod;
  return 0;
}
