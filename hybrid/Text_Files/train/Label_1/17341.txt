#include <bits/stdc++.h>
using namespace std;
#define MAX 100000
vector<int> v;
int32_t main() {
    int n, q, x;
    char s;
    cin >> n >> q;
    vector<pair<char, int>> vec;
    for (int i = 0; i < n; i++) {
        cin >> x;
        v.push_back(x);
    }
    for (int i = 0; i < q; i++) {
        cin >> s >> x;
        vec.push_back(make_pair(s, x));
    }
    for (auto x : vec) {
        if (x.first == '>') {
            for (int j = 0; j < n; j++) {
                if (v[j] > x.second) v[j] = -v[j];
            }
        } else {
            for (int j = 0; j < n; j++) {
                if (v[j] < x.second) v[j] = -v[j];
            }
        }
    }
    for (int x : v) { cout << x << " "; }
}
