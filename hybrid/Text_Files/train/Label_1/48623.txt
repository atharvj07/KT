import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.BitSet;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        BitSet byteCities = new BitSet();
        int[] cityCoordinates = new int[n];
        int[] previous = new int[n];
        int[] next = new int[n];
        for (int i = 0; i < n; i++) {
            String[] city = br.readLine().split(" ");
            cityCoordinates[i] = Integer.parseInt(city[0]);
            if (city[1].charAt(0) == 'B') {
                byteCities.set(i);
                if (i > 0) {
                    previous[i] = i - 1;
                    next[i - 1] = i;
                }
            } else {
                if (i > 0) {
                    next[i - 1] = i;
                }
            }
        }
        int cost = 0;
        for (int i = 0; i < n; i++) {
            if (byteCities.get(i)) {
                if (previous[i] != -1 && !byteCities.get(previous[i])) {
                    cost += cityCoordinates[i] - cityCoordinates[previous[i]];
                }
                if (next[i] != n && !byteCities.get(next[i])) {
                    cost += cityCoordinates[next[i]] - cityCoordinates[i];
                }
            } else {
                if (i > 0 && !byteCities.get(i - 1)) {
                    cost += cityCoordinates[i] - cityCoordinates[i - 1];
                }
                if (i + 1 < n && !byteCities.get(i + 1)) {
                    cost += cityCoordinates[i + 1] - cityCoordinates[i];
                }
            }
        }
        System.out.println(cost);
    }
}
