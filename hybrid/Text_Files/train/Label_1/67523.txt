def gcd(a, b):
    while b:
        a, b = b, a % b
    return a
n = int(input())
a = list(map(int, input().split()))
ans = []
for i in range(1, n + 1):
    cur = a[0]
    ok = True
    for j in range(1, n):
        if cur > a[j]:
            ok = False
            break
        cur += i
        cur -= a[j]
        cur = cur % i
    if ok:
        ans.append(i)
print(len(ans))
print(*ans)
