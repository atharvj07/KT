import java.util.*;

class Main {
    private final int MOD = 1000000007;
    private int[][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        System.out.println(new Main().solve(n, k, a));
    }

    private int solve(int n, int k, int[] a) {
        dp = new int[n + 1][k + 1];
        for (int[] row : dp) {
            Arrays.fill(row, -1);
        }
        return dfs(n, k, a, 0);
    }

    private int dfs(int n, int k, int[] a, int idx) {
        if (idx == n) {
            return k == 0 ? 1 : 0;
        }
        if (dp[idx][k] != -1) {
            return dp[idx][k];
        }
        int ans = 0;
        for (int i = 0; i <= Math.min(a[idx], k); i++) {
            ans = (ans + dfs(n, k - i, a, idx + 1)) % MOD;
        }
        return dp[idx][k] = ans;
    }
}
