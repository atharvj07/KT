#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> v(n), l(n), r(n), q;
    for (auto &x: v) cin >> x;
    for (int i = 1; i < n - 1; i++) l[i] += max(0, v[i - 1] + v[i + 1] + v[i]);
    for (int i = n - 2; i > 0; i--) r[i] += max(0, v[i - 1] + v[i + 1] + v[i]);
    for (int i = 1; i < n - 1; i++) {
        int sum = v[i - 1] + v[i] + v[i + 1];
        if (l[i] && r[i]) {
            l[i] = r[i] = sum;
            q.emplace_back(i);
        }
    }
    int idx = max_element(l.begin(), l.end()) - l.begin();
    cout << l[idx] + r[idx] << ' ' << q.size() / 2 << '\n';
    for (int i = 0; i < q.size(); i += 2) cout << q[i] << ' ' << q[i + 1] << '\n';
}
