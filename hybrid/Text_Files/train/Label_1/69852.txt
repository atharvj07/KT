import sys
input=sys.stdin.readline

for _ in range(int(input())):
    n,l,r=map(int,input().split())
    c=list(map(int,input().split()))
    c.sort()
    ans=0
    i=0;j=n-1
    while i<j:
        if c[i]==c[i+1]:
            i+=2
        elif c[j]==c[j-1]:
            j-=2
        elif c[i]==c[j]:
            ans+=2
            i+=1;j-=1
        else:
            ans+=1
            i+=1
    print(ans)
