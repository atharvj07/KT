import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long k = sc.nextLong();
        System.out.println((k % n == 0 ? n : k % n) + " " + ((k / n) % 2 == 0 ? (k / n) % m + 1 : m - (k / n) % m));
    }
}
