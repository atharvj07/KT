#include <bits/stdc++.h>
#define ll long long

using namespace std;

const int N = 2e5 + 5;
int n;
vector<int> g[N];
int sz[N], dep[N], ans, par[N], p[N];
bool vis[N];

void dfs_sz(int u, int pr) {
    sz[u] = 1;
    for (int v : g[u]) {
        if (v != pr) {
            dfs_sz(v, u);
            sz[u] += sz[v];
        }
    }
}

void dfs_par(int u, int pr) {
    if (pr == -1) par[u] = 0;
    else par[u] = pr;
    for (int v : g[u]) {
        if (v != pr) dfs_par(v, u);
    }
}

void dfs_dis(int u, int pr, int d) {
    dep[u] = d;
    for (int v : g[u]) {
        if (v != pr) dfs_dis(v, u, d + 1);
    }
}

void dfs1(int u, int pr) {
    vis[u] = true;
    for (int v : g[u]) {
        if (v != pr) {
            if (!vis[v]) dfs1(v, u);
            else if (dep[u] > dep[v] && !p[v]) {
                ans++;
                p[u] = p[v] = 1;
            }
        }
    }
}

int solve() {
    cin >> n;
    for (int i = 1; i < n; ++i) {
        int x, y;
        cin >> x >> y;
        g[x].push_back(y);
        g[y].push_back(x);
    }
    dfs_sz(1, -1);
    dfs_par(1, -1);
    dfs_dis(1, -1, 0);
    dfs1(1, -1);
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int ans = solve();
    cout << ans << '\n';
    return 0;
}
