#include <cstdio>
#include <iostream>
#include <vector>
#include <cstring>

using namespace std;

typedef pair<int, long long> pii;

const int MAXN = 2e5 + 5;
vector<pii> g[MAXN];
int a[MAXN];
int path[MAXN];
int n;
int done;

void dfs_path(int x)
{
    if(g[x].size() == 0)
	{
        done = x;
        return;
	}
    for(int i = 0; i < g[x].size(); i++)
	{
        int v = g[x][i].first;
        if(path[v] == 0)
		{
            path[v] = x;
            dfs_path(v);
            return;
		}
    }
}

int main()
{
    int p;
    scanf("%d", &n);
    for(int i = 2; i <= n; i++)
	{
        scanf("%d", &p);
        g[p].push_back(pii(i, 0));
    }
    for(int i = 1; i <= n; i++)
	{
        scanf("%d", &a[i]);
    }

    dfs_path(1);

    long long ans = 0;
    while(done > 1)
	{
        ans += a[done];
        done = path[done];
    }
    cout << ans << endl;
    return 0;
}
