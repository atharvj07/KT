#include <bits/stdc++.h>
#define ll long long
using namespace std;

ll n, a[200005], b[200005], ans[200005], cnt, x, y;
vector<pair<ll, ll>> v;

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i] >> b[i];
    v.push_back(make_pair(a[i], i));
    v.push_back(make_pair(b[i], i));
  }
  sort(v.begin(), v.end());
  for (int i = 1; i < 2 * n; i++) {
    if (v[i - 1].first != v[i].first) {
      if (cnt % 2 == 0) {
        ans[++cnt] = v[i - 1].second;
      } else {
        x = i;
      }
    } else {
      if (cnt % 2 == 1) {
        ans[++cnt] = v[i].second;
        y = i;
      } else {
        x = i;
      }
    }
  }
  cout << cnt << endl;
  for (int i = 1; i <= cnt; i++) {
    cout << ans[i] << " ";
  }
  cout << endl;
  return 0;
}
