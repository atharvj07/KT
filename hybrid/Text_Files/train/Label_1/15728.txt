#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<ll> v(n);
        for (int i = 0; i < n; i++) cin >> v[i];
        sort(v.rbegin(), v.rend());
        int ans = 1;
        for (int i = 1; i < n; i++) {
            if (v[i - 1] - v[i] >= v[0]) ans++;
            else break;
        }
        cout << ans << endl;
    }
}
