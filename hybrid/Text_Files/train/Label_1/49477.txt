import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String s = input.next();
        int min = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '-') {
                min--;
            } else {
                min++;
            }
        }
        System.out.println(min);
    }
}
