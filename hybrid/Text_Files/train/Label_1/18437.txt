#include <bits/stdc++.h>
using i64 = long long;
int main() {
  i64 n, k, x;
  std::cin >> n >> k >> x;
  std::vector<i64> a(n);
  for (auto& p : a) { std::cin >> p; }
  std::sort(a.begin(), a.end());
  i64 ans = 1;
  i64 temp = x;
  for (i64 i = 1; i < n; ++i) {
    if (a[i] - a[i - 1] > x) {
      ++ans;
      temp = x;
    } else {
      temp -= a[i] - a[i - 1];
    }
  }
  std::cout << ans << "\n";
  return 0;
}
