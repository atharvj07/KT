import java.util.Arrays;
import java.util.Scanner;

public class CokeParty {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextInt();
        }

        Arrays.sort(a);

        int minLiters = -1;
        for (int i = 0; i < k; i++) {
            int targetLiters = (int) Math.ceil((double) n * a[i] / 1000);
            if (targetLiters > 0) {
                minLiters = targetLiters;
                break;
            }
        }

        if (minLiters == -1) {
            System.out.println(-1);
        } else {
            System.out.println(minLiters);
        }
    }
}
