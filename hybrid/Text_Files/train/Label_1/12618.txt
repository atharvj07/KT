#include <iostream>
#include <algorithm>
using namespace std;
pair<int,int>a[100001];
int main()
{
	int n;
	cin >> n;
	for(int i = 0; i < n; i++) cin >> a[i].first >> a[i].second;
	sort(a,a+n);
	int mxr = 0, ans = -1;
	for(int i = 0; i < n; i++)
	{
		if(a[i].first > mxr) ans = -1;
		if(a[i].second > mxr) mxr = a[i].second, ans = i + 1;
	}
	cout << ans << endl;
	return 0;
}
