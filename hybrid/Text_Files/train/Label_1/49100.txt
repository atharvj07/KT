import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
        }

        int max = arr[n - 1];
        int ans = 0;
        for (int i = n - 2; i >= 0; i--) {
            if (max < arr[i]) {
                ans = -1;
                break;
            }
            if (arr[i] > arr[i + 1]) {
                ans++;
            }
            max = Math.max(max, arr[i]);
        }

        System.out.println(ans);
    }
}
