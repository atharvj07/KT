n,m=map(int,input().split());a=[[0]*m for i in range(n)];mod=10**9+7;dp=[[0]*m for i in range(n)]
for i in range(n):a[i]=input()
dp[0][0]=1
for i in range(n):
    for j in range(m):
        if i+1<n and a[i+1][j]==".": dp[i+1][j]+=dp[i][j]%mod
        if j+1<m and a[i][j+1]==".": dp[i][j+1]+=dp[i][j]%mod
        if a[i][j]=="." and i+1<n and a[i+1][j]=="R": dp[i+1][j]+=dp[i][j]%mod
        if a[i][j]=="." and j+1<m and a[i][j+1]=="R": dp[i][j+1]+=dp[i][j]%mod
print(dp[n-1][m-1]%mod)
