#include <iostream>

using namespace std;

int main() {
  char start, end;
  int n;
  cin >> start >> end >> n;
  if (n % 4 == 1) {
    if (start == 'v' && end == '>') cout << "cw";
    else if (start == '^' && end == '<') cout << "ccw";
    else cout << "undefined";
  } else if (n % 4 == 3) {
    if (start == 'v' && end == '<') cout << "cw";
    else if (start == '^' && end == '>') cout << "ccw";
    else cout << "undefined";
  } else cout << "undefined";
  return 0;
}
