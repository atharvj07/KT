import sys
import heapq

input = sys.stdin.readline

n = int(input())
v = list(map(int, input().split()))

m = int(input())
q = []
for _ in range(m):
    t, l, r = map(int, input().split())
    q.append((t, l - 1, r))

v_sorted = sorted(v)
for i in range(n):
    if v[i] != v_sorted[i]:
        heapq.heappush(q, (2, i, v_sorted[i]))

ans = [0] * m
for i, (t, l, r) in enumerate(q):
    if t == 1:
        ans[i] = sum(v[l:r])
    else:
        while q and q[0][0] == 2 and q[0][1] <= r:
            _, j, vj = heapq.heappop(q)
            v[j] = vj
        ans[i] = sum(v[l:r])

for a in ans:
    print(a)
