from math import factorial

def perm(n, k):
    cnt = 0
    res = []
    vis = [False] * (n+1)
    ls = []
    for i in range(1, n+1):
        if not vis[i]:
            vis[i] = True
            ls.append([i])
            c = n-1
            j = i
            while c > 0:
                j = j + i
                if j > n:
                    j = j - n
                if not vis[j]:
                    vis[j] = True
                    ls[-1].append(j)
                    c = c - 1
            ls[-1].sort()
            cnt = cnt + 1
        if cnt == k:
            res = ls[-1]
            break
    return res

for i in range(int(input())):
    n, k = map(int, input().split())
    if factorial(n) < k:
        print(-1)
    else:
        tmp = perm(n, k)
        print(*tmp)
