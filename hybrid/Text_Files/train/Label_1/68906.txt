from heapq import *
from math import gcd
for t in range(int(input())):
    n,p=map(int,input().split())
    a = [int(x) for x in input().split()]
    g=[[] for i in range(n)]
    for i in range(n-1):
        for j in range(i+1,n):
            x=a[i]
            for k in range(i+1,j+1):
                x=gcd(x,a[k])
            if x==min(a[i:j+1]): g[i].append([j,min(a[i:j+1])])
            if j==i+1: g[i].append([j,p])
    x=[p]
    vis=[]
    for i in range(n):
        if i in vis: continue
        m=float('inf')
        q=[]
        for edg in g[i]:
            heappush(q,[edg[1],edg[0]])
        vis.append(i)
        while q:
            w,u=heappop(q)
            if u in vis: continue
            if w<m: m=w
            vis.append(u)
            for edg in g[u]:
                heappush(q,[edg[1],edg[0]])
        x.append(m)
    print(min(x))
