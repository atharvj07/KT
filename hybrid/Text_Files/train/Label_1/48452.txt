import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] teams = new int[k];
        for (int i = 0; i < k; i++) {
            teams[i] = sc.nextInt();
        }
        Arrays.sort(teams);
        int ans = 0;
        for (int i = 0; i < k; i++) {
            int team = teams[i];
            int pos = 1 << n;
            while (pos > 1) {
                pos >>= 1;
                if ((team & pos) == 0) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
