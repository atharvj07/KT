#include <cstdio>
#include <cstring>
#include <iostream>
#include <vector>
using namespace std;

const int maxN = 2e5 + 10;

char s[maxN], prv[maxN];
int n, k;

vector<int> e[maxN];

void process()
{
    int pos, cnt, tmp;
    for (int j = 0; j < k; j++)
    {
        for (int i = 1; i <= n; i++)
        {
            if (s[i] == 'W')
                cnt = 0;
            else
                cnt = 1;
            pos = i - 1;
            if (pos == 0)
                pos = n;
            if (s[pos] == 'W')
                cnt++;
            else
                cnt--;
            pos = i + 1;
            if (pos == n + 1)
                pos = 1;
            if (s[pos] == 'W')
                cnt++;
            else
                cnt--;
            if (cnt > 0)
                prv[i] = 'W';
            else
                prv[i] = 'B';
        }
        for (int i = 1; i <= n; i++)
            s[i] = prv[i];
    }
}

int main()
{
    scanf("%d%d%s", &n, &k, s + 1);
    prv[n + 1] = '_';
    process();
    printf("%s", s + 1);
    return 0;
}
