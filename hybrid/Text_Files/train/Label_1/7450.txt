#include <bits/stdc++.h>

const int MOD = 1e9 + 7;

int fastpow(int64_t base, int64_t exp) {
  if (exp == 0) { return 1; }
  int64_t half = fastpow(base, exp / 2);
  if (exp % 2 == 0) { return half * half % MOD; }
  return half * half % MOD * base % MOD;
}

int main() {
  int m, d, a, b;
  std::cin >> m >> d >> a >> b;
  int n = b - a + 1;
  int pw = fastpow(10, n);
  int coef = (fastpow(pw - 1, MOD - 2) + MOD) % MOD;
  int ans = (((b / m - (a - 1) / m) % MOD) * pw % MOD) * coef % MOD;
  int count = 0;
  int mask = 1 << d;
  for (int i = 0; i < n; i++) {
    if ((a >> i & mask) && ((n - i) % 2 == 0)) { count++; }
  }
  ans += (int64_t)count * pw % MOD;
  ans %= MOD;
  std::cout << ans << '\n';
  return 0;
}
