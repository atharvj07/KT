n = int(input())
a = [int(i) for i in input().split()]
mod = 998244353
dp = [[0] * 2 for _ in range(n+1)]
dp[n][0] = 1
for i in range(n-1,-1,-1):
    for j in range(2):
        if a[i] == -1:
            dp[i][j] = (dp[i+1][0] + dp[i+1][1]) % mod
        elif a[i] == 0:
            dp[i][j] = dp[i+1][j]
        elif j == 0:
            dp[i][j] = (dp[i+1][0] + dp[i+1][1] * (a[i] == 1)) % mod
        else:
            dp[i][j] = dp[i+1][j] * (a[i] == a[i-1])
print(dp[0][0])
