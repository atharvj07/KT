from queue import Queue
import sys

input = sys.stdin.readline

n = int(input())
p = list(map(int, input().split()))
a = [list(input().strip()) for _ in range(n)]

q = Queue()
ans = []
seen = [0] * (n + 1)

for i in range(1, n + 1):
    if not seen[i]:
        q.put(i)
        while not q.empty():
            cur = q.get()
            ans.append(cur)
            seen[cur] = 1
            for i in range(1, n + 1):
                if a[cur - 1][i - 1] == '1' and not seen[i]:
                    q.put(i)

print(*ans)
