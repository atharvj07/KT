#include<bits/stdc++.h>
using namespace std;
const int N = 100005;
int dp[2][N];
int main() {
    int n, S;
    cin >> n >> S;
    dp[0][0] = 0;
    for(int i = 1;i <= n;i++) {
        dp[0][i] = N;
        dp[1][i] = N;
    }
    for(int i = 1;i <= S;i++) {
        dp[1][i] = N;
        for(int j = 1;j <= n;j++) {
            if(i - j >= 0) {
                dp[1][i] = min(dp[1][i], dp[0][i - j]);
            }
        }
        dp[1][i] += 1;
        dp[0][i] = dp[1][i];
    }
    cout << dp[0][S];
    return 0;
}
