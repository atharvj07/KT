#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ll n, k, ans = 0, a[200001];
    cin >> n >> k;
    map<ll, ll> m;
    for (int i = 0; i < n; i++){
        cin >> a[i];
        m[a[i] % k]++;
    }
    for (int i = 0; i < n; i++){
        if (a[i] % k == 0) ans += m[0] - 1;
        else{
            if (m.count(k - a[i] % k)){
                ans += m[k - a[i] % k];
                m[k - a[i] % k]--;
                m[a[i] % k]--;
            }
        }
    }
    cout << ans << endl;
    return 0;
    }
