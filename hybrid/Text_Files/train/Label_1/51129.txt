import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long[] a = new long[N];

        for (int i = 0; i < N; i++) {
            a[i] = sc.nextLong();
        }

        // a[i] XOR x = x
        long x = a[0];

        for (int i = 1; i < N; i++) {
            x ^= a[i];
        }

        for (int i = 0; i < N; i++) {
            System.out.print((x ^ a[i]) + " ");
        }

        System.out.println();

        sc.close();
    }
}
