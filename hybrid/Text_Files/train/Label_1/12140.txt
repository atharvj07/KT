#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n % 2 != 1) {
      cout << -1 << endl;
    } else {
      int a[n][n];
      int cnt = 1;
      for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
          if ((i + j) % 2 == 0) {
            a[i][j] = cnt++;
          } else {
            a[i][j] = cnt + n - 1;
            cnt++;
          }
        }
      }
      for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
          cout << a[i][j] << " ";
        }
        cout << endl;
      }
    }
  }
  return 0;
}
