#include<bits/stdc++.h>
using namespace std;
bool rem[100];
string s;
int n, m;
 
int main(){
    cin >> s;
    reverse(s.begin(), s.end());
    n = s.size();
    for(int i = 0; i < n; i++){
        if(s[i] == '0') continue;
        m = i;
        break;
    }
    if(s[m] == '8'){
        cout << "YES\n0";
        return 0;
    }
    if(s[m] == '4' || s[m] == '9'){
        s[m + 1] = '4';
        for(int i = m + 2; i < n; i++){
            s[i] = '0';
        }
        reverse(s.begin(), s.end());
        cout << "YES\n" << s << '\n';
        return 0;
    }
    if(s[m] == '2' || s[m] == '3'){
        s[m + 1] = '8';
        for(int i = m + 2; i < n; i++){
            s[i] = '0';
        }
        reverse(s.begin(), s.end());
        cout << "YES\n" << s << '\n';
        return 0;
    }
    if(s[m] == '1'){
        for(int i = m + 1; i < n; i++){
            if(s[i] == '8' || s[i] == '4' || s[i] == '9'){
                s[m + 1] = s[i];
                s[i] = '0';
                break;
            }
        }
        if(s[m + 1] == '0'){
            s[m + 1] = '4';
            for(int i = m + 2; i < n; i++){
                s[i] = '0';
            } 
        }
        reverse(s.begin(), s.end());
        cout << "YES\n" << s << '\n';
        return 0;
    }
    cout << "NO\n";
    return 0;
}
