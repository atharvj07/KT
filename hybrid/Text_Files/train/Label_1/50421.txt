import java.util.Scanner;
import java.util.Stack;
import java.util.HashMap;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int q = sc.nextInt();
    Stack<Integer>[] stacks = (Stack<Integer>[])new Stack[n];
    for(int i = 0; i < n; i++) {
      stacks[i] = new Stack<Integer>();
    }
    HashMap<Integer, Integer> tops = new HashMap<Integer, Integer>();
    while(q-- > 0) {
      int op = sc.nextInt();
      int t = sc.nextInt();
      if(op == 0) {
        int x = sc.nextInt();
        stacks[t].push(x);
        tops.put(t, x);
      }
      else if(op == 1) {
        if(stacks[t].size() > 0) {
          System.out.println(tops.get(t));
        }
      }
      else if(op == 2) {
        if(stacks[t].size() > 0) {
          stacks[t].pop();
          tops.remove(t);
          if(stacks[t].size() > 0) {
            tops.put(t, stacks[t].peek());
          }
        }
      }
    }
  }
}
