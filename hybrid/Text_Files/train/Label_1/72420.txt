import math

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

for _ in range(int(input())):
    n = int(input())
    while True:
        sum = 0
        for d in str(n):
            sum += int(d)
        g = gcd(n, sum)
        if g > 1:
            print(n)
            break
        n += 1
