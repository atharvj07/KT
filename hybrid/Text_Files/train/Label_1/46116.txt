import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        char[][] board = new char[n][n];
        int freeCount = 0;
        for (int i = 0; i < n; i++) {
            String line = br.readLine();
            for (int j = 0; j < n; j++) {
                board[i][j] = line.charAt(j);
                if (board[i][j] == '.') {
                    freeCount++;
                }
            }
        }
        if (freeCount % 5 != 0) {
            System.out.println("NO");
        } else {
            int pieces = freeCount / 5;
            boolean[][] used = new boolean[n][n];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (!used[i][j] && board[i][j] == '.') {
                        if (tryPlacePiece(board, used, i, j, pieces)) {
                            System.out.println("YES");
                            return;
                        }
                    }
                }
            }
            System.out.println("NO");
        }
    }

    private static boolean tryPlacePiece(char[][] board, boolean[][] used, int i, int j, int pieces) {
        if (pieces == 0) {
            return true;
        }
        if (i < 0 || i >= board.length || j < 0 || j >= board[0].length || used[i][j] || board[i][j] == '#') {
            return false;
        }
        used[i][j] = true;
        if (tryPlacePiece(board, used, i + 1, j, pieces) ||
                tryPlacePiece(board, used, i - 1, j, pieces) ||
                tryPlacePiece(board, used, i, j + 1, pieces) ||
                tryPlacePiece(board, used, i, j - 1, pieces)) {
            return true;
        }
        used[i][j] = false;
        return false;
    }
}
