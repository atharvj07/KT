#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = n;
  for (int i = 1; i * i <= s.size(); i++) {
    if (s.size() % i == 0) {
      string t = s.substr(0, i);
      bool good = true;
      for (int j = i; j < s.size(); j += i) {
        if (s.substr(j, i) != t) {
          good = false;
          break;
        }
      }
      if (good) ans = min(ans, n / i + 1);
    }
  }
  cout << ans << endl;
  return 0;
}
