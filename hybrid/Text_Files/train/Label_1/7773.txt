#include <iostream>

using namespace std;

static const size_t MAX_N = 1e9;
static const size_t MAX_M = 1e5;
static const size_t MAX_A = 1e9;
static const size_t MAX_D = 1e18;

long long n, m, a, d;
long long t[MAX_M];

int main() {
    cin >> n >> m >> a >> d;
    for (size_t i = 0; i < m; i++) cin >> t[i];
    long long last_time = 0;
    long long cnt = 0;
    for (size_t i = 0; i <= n; i++) {
        long long time = i * a;
        if (time < last_time) continue;
        cnt++;
        last_time = time + d;
        while (m > 0 && t[m - 1] >= time && t[m - 1] < last_time) {
            m--;
            last_time = time + d;
        }
    }
    cout << cnt << endl;
    return 0;
}
