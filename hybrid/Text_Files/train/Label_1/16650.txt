#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int INF = 2e9;

int n, m, ans, cnt;
vector<vector<pair<int, int>>> g;
vector<pair<pair<int, int>, int>> edges;
vector<bool> vis;

bool dfs(int v, int p, int cnt) {
    vis[v] = true;
    for (auto next : g[v]) {
        if (next.first == p) continue;
        if (vis[next.first]) {
            if (cnt - next.second < 0) return true;
        } else if (dfs(next.first, v, cnt - next.second)) return true;
    }
    return false;
}

int main() {
    cin >> n >> m;
    g.resize(n + 1);
    vis.assign(n + 1, false);
    for (int i = 1; i < m; i++) {
        int u, v, c; cin >> u >> v >> c;
        g[u].emplace_back(v, c);
        g[v].emplace_back(u, c);
        edges.emplace_back(make_pair(u, v), c);
    }
    for (int i = 1; i <= n; i++) {
        if (!vis[i] && dfs(i, i, 0)) {
            for (auto edge : edges) {
                int u = edge.first.first, v = edge.first.second, c = edge.second;
                if (vis[u] && !vis[v]) ans -= c;
                if (vis[v] && !vis[u]) ans -= c;
            }
            cnt++;
        }
    }
    cout << ans << " " << cnt << endl;
    for (auto edge : edges) {
        int u = edge.first.first, v = edge.first.second, c = edge.second;
        if (vis[u] && !vis[v]) cout << c << " ";
        if (vis[v] && !vis[u]) cout << c << " ";
    }
    cout << endl;
    return 0;
}
