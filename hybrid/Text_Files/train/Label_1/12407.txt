#include <iostream>
#include <set>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  set<string> photobooks;
  for (int i = 0; i <= s.size(); i++) {
    for (char c = 'a'; c <= 'z'; c++) {
      string photobook = s.substr(0, i) + c + s.substr(i);
      photobooks.insert(photobook);
    }
  }
  cout << photobooks.size() << endl;
}
