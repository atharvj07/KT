#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int n; cin >> n;
    vector<pair<int, int>> v(n + 1);
    for (int i = 0; i <= n; i++) { cin >> v[i].first >> v[i].second; }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        int x1 = v[i - 1].first, y1 = v[i - 1].second;
        int x2 = v[i].first, y2 = v[i].second;
        if (x1 < x2 && y1 > y2) { ans += (y1 - y2 - 1); }
        if (x2 < x1 && y1 < y2) { ans += (y2 - y1 - 1); }
        if (y1 < y2 && x1 < x2) { ans += (x2 - x1 - 1); }
        if (y2 < y1 && x1 > x2) { ans += (x1 - x2 - 1); }
    }
    cout << ans << '\n';
}
