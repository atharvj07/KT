#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<pair<int, int>> moves = {{0, 1}, {0, -1}, {-1, 0}, {1, 0}};
  int x = 0, y = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 4; j++) {
      if (s[i] == 'U' && moves[j].second == 1) {
        x += moves[j].first;
        y += moves[j].second;
        ans = i + 1;
        break;
      } else if (s[i] == 'D' && moves[j].second == -1) {
        x += moves[j].first;
        y += moves[j].second;
        ans = i + 1;
        break;
      } else if (s[i] == 'L' && moves[j].first == -1) {
        x += moves[j].first;
        y += moves[j].second;
        ans = i + 1;
        break;
      } else if (s[i] == 'R' && moves[j].first == 1) {
        x += moves[j].first;
        y += moves[j].second;
        ans = i + 1;
        break;
      }
    }
    if (x == 0 && y == 0) {
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
