#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll c[100005],t[100005],v[100005];
int main()
{
    ll n,m,i,j,s=0;
    cin>>n>>m;
    for(i=1;i<=n;i++)
    {
        cin>>c[i]>>t[i];
        s+=c[i]*t[i];
    }
    for(i=1;i<=n;i++)
    {
        for(j=1;j<=c[i];j++)
        {
            if(s>=v[1])
            {
                cout<<i<<endl;
                v[1]++;
            }
            s-=t[i];
        }
    }
}
