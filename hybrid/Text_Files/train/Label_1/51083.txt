import java.util.*;

public class Main {
    static List<Edge>[] adj;
    static int[] dist;
    static int N, M, K;
    static Set<Integer> shops;

    public static class Edge {
        int to, w;
        Edge(int to, int w) {
            this.to = to;
            this.w = w;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        K = sc.nextInt();

        adj = new List[N + 1];
        for (int i = 1; i <= N; i++) adj[i] = new ArrayList<>();

        for (int i = 0; i < M; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int w = sc.nextInt();
            adj[a].add(new Edge(b, w));
            adj[b].add(new Edge(a, w));
        }

        shops = new HashSet<>();
        for (int i = 0; i < K; i++) shops.add(sc.nextInt());

        dist = new int[N + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);

        dijkstra();

        int maxDist = 0;
        for (int i = 1; i <= N; i++) {
            if (!shops.contains(i)) maxDist = Math.max(maxDist, dist[i]);
        }

        System.out.println(maxDist);
    }

    public static void dijkstra() {
        PriorityQueue<Edge> pq = new PriorityQueue<>((a, b) -> a.w - b.w);

        for (int shop : shops) {
            pq.add(new Edge(shop, 0));
            dist[shop] = 0;
        }

        while (!pq.isEmpty()) {
            Edge e = pq.poll();

            if (dist[e.to] < e.w) continue;

            for (Edge next : adj[e.to]) {
                int w = e.w + next.w;
                if (dist[next.to] > w) {
                    dist[next.to] = w;
                    pq.add(new Edge(next.to, w));
                }
            }
        }
    }
}
