#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  string s;
  cin >> n >> s;
  stack<char> stk;
  for (auto &c : s) {
    if (!stk.empty() && c == stk.top()) stk.pop();
    else stk.push(c);
  }
  cout << stk.size() << endl;
  while (!stk.empty()) { cout << stk.top(); stk.pop(); }
  return 0;
}
