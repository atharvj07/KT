n, m = map(int, input().split())
a = list(map(int, input().split()))
segments = [tuple(map(int, input().split())) for _ in range(m)]

# Initialize the difference array and the operation count array
difference = [0] * (n + 1)
operation_count = [0] * (n + 1)

# Apply each segment as an operation and update the difference and operation count arrays
for i, (l, r) in enumerate(segments, start=1):
    difference[l - 1] += 1
    difference[r] -= 1
    operation_count[l - 1] += 1
    operation_count[r] -= i / m  # Decreasing fractionally to keep track of the segment index

# Calculate the prefix sums to get the final values and the chosen segments
for i in range(1, n):
    difference[i] += difference[i - 1]
    operation_count[i] += operation_count[i - 1]

# Calculate the array after applying all possible operations
b = [a[i] - difference[i] for i in range(n)]

# Find the maximum difference and the segments to achieve it
max_diff = max(b) - min(b)
chosen_segments = sorted(set(int((m + 1) * (operation_count[i] % 1)) for i in range(n) if operation_count[i] % 1 != 0))

print(max_diff)
print(len(chosen_segments))
print(*chosen_segments)
