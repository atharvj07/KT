#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 100;
vector<int>g[maxn];
vector<int>a(maxn);
int dis[maxn];
bool vis[maxn];
int n;
bool dfs1(int u, int fa)
{
    bool flag = true;
    for (int v : g[u])
    {
        if (v == fa)continue;
        if (!dfs1(v, u))flag = false;
    }
    if (flag)
    {
        if (a[u] < 0)
        {
            dis[u] = dis[fa] + a[u];
        }
        else
        {
            dis[u] = a[u];
        }
    }
    else
    {
        dis[u] = maxn;
    }
    return flag;
}
bool dfs2(int u, int fa)
{
    bool ans = false;
    for (int v : g[u])
    {
        if (v == fa)continue;
        bool ok = true;
        if (dis[v] == maxn)ok = false;
        if (dis[v] > a[u])ok = false;
        ok &= dfs2(v, u);
        if (!ok)ans = true;
        if (vis[v] && ok)ans = true;
    }
    vis[u] = true;
    if (ans)
    {
        return false;
    }
    else
    {
        return true;
    }
}
int main()
{
    cin >> n;
    for (int i = 1;i <= n;i++)
    {
        cin >> a[i];
    }
    for (int i = 2;i <= n;i++)
    {
        int p, c;
        cin >> p >> c;
        g[p].push_back(i);
        g[i].push_back(p);
    }
    dfs1(1, 0);
    int ans = dfs2(1, 0);
    cout << ans << endl;
    return 0;
}
