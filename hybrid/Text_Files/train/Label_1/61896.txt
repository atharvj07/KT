def det(a, b, c, d):
    return a * d - b * c
ax, ay, bx, by, cx, cy, dx, dy = map(int, input().split())
ex, ey, fx, fy, gx, gy, hx, hy = map(int, input().split())
isx = det(ax, bx, ex, fx) * det(ay, by, ey, fy) > 0 and det(cx, dx, ex, fx) * det(cy, dy, ey, fy) > 0 and det(ax, bx, cx, dx) * det(ay, by, cy, dy) > 0
isy = det(ex, fx, gx, hx) * det(ey, fy, gy, hy) > 0 and det(ax, bx, gx, hx) * det(ay, by, gy, hy) > 0 and det(ex, fx, ax, cx) * det(ey, fy, ay, cy) > 0
print("YES" if isx != isy else "NO")
