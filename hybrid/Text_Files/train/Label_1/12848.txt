#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            ans = max(ans, (a[i] - a[j]) / (j - i));
            ans = max(ans, (a[j] - a[i]) / (j - i));
        }
    }
    cout << ans << endl;
    return 0;
}
