#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mod = 1e9 + 7;
int main() {
    ll n, k;
    while (cin >> n >> k) {
        vector<ll> a(n);
        for (int i = 0; i < n; ++i) cin >> a[i];
        ll cnt = k + 1, cur = 0;
        for (int i = 0; i < n; ++i) {
            cur += (a[i] == a[(i - 1 + n) % n]);
            if ((i - k + 1) >= 0) {
                cur -= (a[i - k + 1] == a[i]);
            }
            if (cur != cnt) {
                if (cur + (k - (a[i - k + 1] == a[i])) == cnt) {
                    cur += k + 1;
                    cnt++;
                } else
                    break;
            }
        }
        cout << cnt << '\n';
    }
}
