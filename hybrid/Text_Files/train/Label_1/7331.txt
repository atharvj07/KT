#include <iostream>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin >> n;
    string s;
    cin >> s;
    int rev = 0;
    for (int i = 0; i < n / 2; i++) {
        if (s[i] != s[n - i - 1]) rev++;
    }
    cout << rev << endl;
    return 0;
}
