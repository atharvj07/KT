import java.util.Arrays;
import java.util.BitSet;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        BitSet a = new BitSet(n + 1), b = new BitSet(n + 1), c = new BitSet(n + 1);
        int x = sc.nextInt();
        while (x-- > 0) {
            a.set(sc.nextInt());
        }
        int y = sc.nextInt();
        while (y-- > 0) {
            b.set(sc.nextInt());
        }
        int z = sc.nextInt();
        while (z-- > 0) {
            c.set(sc.nextInt());
        }
        BitSet d = new BitSet(n + 1), e = new BitSet(n + 1);
        d.or(a);
        d.flip(0, n + 1);
        d.and(c);
        e.or(b);
        e.and(c);
        BitSet f = new BitSet(n + 1);
        f.or(d);
        f.or(e);
        System.out.println(f.cardinality());
    }
}
