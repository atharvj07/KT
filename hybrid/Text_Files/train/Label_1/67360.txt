n = int(input())
a = list(map(int, input().split()))

def solve(n, a):
    result = []
    for i in range(n):
        max_coolness = a[i]
        tracks_count = 1
        for j in range(1, n):
            current_track = a[(i + j) % n]
            if current_track < max_coolness / 2:
                break
            tracks_count += 1
            max_coolness = max(max_coolness, current_track)
        if tracks_count < n:
            result.append(tracks_count)
        else:
            max_coolness_double = max(a) * 2
            all_tracks = a + a
            idx = all_tracks.index(max_coolness)
            if max_coolness_double in all_tracks[idx + 1: idx + n]:
                result.append(-1)
            else:
                result.append(n)
    return result

print(*solve(n, a))
