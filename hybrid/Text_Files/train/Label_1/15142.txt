#include<bits/stdc++.h>
using namespace std;
typedef long long LL;
#define pb push_back
const int p=1e9+7;

int main() {
    string s;
    cin >> s;
    unordered_map<char,int> fre;
    for (char x:s) fre[x]++;
    LL ans=1;
    for (auto x:fre)
        if (x.first!='9')
            ans=ans*x.second%p;
    cout << ans << endl;
    return 0;
}
