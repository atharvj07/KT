import java.util.*;

public class Main {
    static Edge[] edges;
    static int[] danger;
    static int n, m;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        edges = new Edge[m];
        danger = new int[n+1];
        for (int i = 0; i < m; i++) {
            edges[i] = new Edge(sc.nextInt(), sc.nextInt());
        }
        Arrays.sort(edges, (a, b) -> Integer.compare(a.y, b.y));
        int maxDanger = 1;
        for (Edge e : edges) {
            int u = e.x, v = e.y;
            danger[v] = Math.max(danger[v], danger[u] * 2);
            maxDanger = Math.max(maxDanger, danger[v]);
        }
        System.out.println(maxDanger);
    }
}

class Edge {
    int x, y;

    public Edge(int x, int y) {
        this.x = x;
        this.y = y;
    }
}
