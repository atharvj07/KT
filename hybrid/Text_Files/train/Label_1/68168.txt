import string

n = int(input())
s = input()
t = input()

if s == t: ans = "NO"
elif s[1] == t[0]: ans = "NO"
elif [s, t] == ["ab", "ca"]: ans = "NO"
elif [s, t] == ["ca", "ab"]: ans = "NO"
else:
    ans = "YES\n" + string.ascii_lowercase[string.ascii_lowercase.index(s[0]):string.ascii_lowercase.index(s[1])] * n

print(ans)
