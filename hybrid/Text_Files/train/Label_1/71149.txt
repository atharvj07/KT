s=input()
i,j,k=s.count('X'),s.count('_'),s[-2:]=="25"
ans=0
if k==0 or s=="0":ans=1
elif i==0:
    if not s[0]=="0":
        ans+=9
    if j==0:
        if not s[0]=="0":
            ans+=9
    elif j==1:
        if s[0]=="0" or s[1]=="0":ans-=1
        ans+=9*8
    else:ans+=9*8*8
elif i==1:
    if j==0:
        if s[0]=="0" or s!=s[::-1]:ans-=1
        ans+=9*9
    elif j==1:
        if s[0]=="0" or s[1]=="0":ans-=1
        if s[0]==s[2]:ans+=9*8
        if s[0]!=s[2]:ans+=9*9
    else:ans+=9*8*8
else:
    if j==0:
        if s!=s[::-1]:ans-=1
        ans+=9*9
    elif j==1:
        if s[0]==s[2]:ans+=9*8
        if s[0]!=s[2]:ans+=9*9
    else:ans+=9*8*8
print(max(ans,0))
