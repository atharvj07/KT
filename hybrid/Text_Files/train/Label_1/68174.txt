import sys

n = int(input())
a = [int(x) for x in sys.stdin.readline().split()]
m = int(input())
b = [int(x) for x in sys.stdin.readline().split()]

sa = sum(a)
sb = sum(b)
a.sort()
b.sort()

v = abs(sa - sb)
ss = []

if n != m:
    print(v)
    print(0)
else:
    for i in range(n):
        v = min(v, abs(sa - sb + 2 * (b[i] - a[i])))
        ss.append((a[i], b[i]))

    if v == 0:
        print(0)
        print(0)
    else:
        print(v)
        print(1)
        for i in range(n):
            if abs(sa - sb + 2 * (b[i] - a[i])) == v:
                print(ss[i][0], ss[i][1])
                break
