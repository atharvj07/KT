n = int(input())
points = list(map(int, input().split()))

def check_intersections(points):
    for i in range(n - 1):
        for j in range(i + 2, n - 1):
            if (points[i] < points[j] < points[i + 1] and points[i + 1] < points[j + 1]) or (points[i] > points[j] > points[i + 1] and points[i + 1] > points[j + 1]):
                return "yes"
    return "no"

print(check_intersections(points))
