#include <iostream>
#include <vector>
using namespace std;
int main() {
  int m;
  cin >> m;
  vector<int> c(m);
  for (int i = 0; i < m; i++) cin >> c[i];
  int x, y;
  cin >> x >> y;
  for (int i = m - 1; i >= 0; i--) {
    int sum = 0;
    for (int j = i; j < m; j++) sum += c[j];
    if (x <= sum && sum <= y) {
      cout << i + 1 << endl;
      return 0;
    }
  }
  cout << 0 << endl;
  return 0;
}
