n, k = map(int, input().split())
h = list(map(int, input().split()))

h.sort(reverse=True)

cuts = 0
total_cubes_removed = 0
current_height = h[0]

for i in range(1, n):
    if h[i] < current_height:
        cubes_to_remove = (current_height - h[i]) * i
        if total_cubes_removed + cubes_to_remove > k:
            cuts += 1
            total_cubes_removed = 0
        total_cubes_removed += cubes_to_remove
        current_height = h[i]

if total_cubes_removed > 0:
    cuts += 1

print(cuts)
