#include <iostream>
using namespace std;
int main() {
  long long n;
  cin >> n;
  if (n % 2 == 0) {
    long long a = n * n / 4 - 1, b = n - 1;
    cout << a << ' ' << b << endl;
  } else {
    long long d = n * n - 1;
    long long x = (1 + d) / 2, y = (d - 1) / 2;
    if (x % 2 == 0) swap(x, y);
    if (x * x + y * y == n * n) cout << x << ' ' << y << endl;
    else
      cout << -1 << endl;
  }
  return 0;
}
