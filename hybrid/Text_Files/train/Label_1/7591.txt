#include <iostream>

using namespace std;
const long long mod = 998244353;
long long quickpow(long long base, long long p) {
  long long ans = 1;
  while (p) {
    if (p & 1) { ans = ans * base % mod; }
    base = base * base % mod;
    p >>= 1;
  }
  return ans;
}
void solve() {
  long long n;
  cin >> n;
  long long ans = ((n * (n + 1) / 2) % mod) * ((n * (n + 1) / 2) % mod) % mod;
  ans = ans * quickpow(n * n, mod - 2) % mod;
  cout << ans << endl;
}
int main() { solve(); }
