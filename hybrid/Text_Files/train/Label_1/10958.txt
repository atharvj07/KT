#include <iostream>
#define ll long long
int main() {
    ll a;
    std::cin >> a;
    ll ans = 0, cur = 0, cnt = 0;
    while (true) {
        if (cnt == 0) cur = a;
        while (cur % 10 == 9 && cur != 0) {
            cnt++;
            cur /= 10;
        }
        if (cur == 0) break;
        while (cur % 10 != 9) {
            ans += cur % 10;
            cur /= 10;
        }
        cur += cnt;
        ans += cnt;
        cnt = 0;
    }
    std::cout << ans << std::endl;
    return 0;
}
