n=int(input())
s=input()
x={'A':0,'C':0,'G':0,'T':0}
for j in s:
    if j!='?':
        x[j]+=1
c=0
for j in x:
    c+=x[j]%4
if c>n%4:
    print('===')
else:
    for i in range(n):
        if s[i]=='?':
            for j in x:
                if x[j]==min(x.values()):
                    x[j]+=1
                    s=s[:i]+j+s[i+1:]
                    break
    print(s)
