import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextLong();
        }

        boolean isGP = true;
        for (int i = 1; i < n; i++) {
            if (arr[i] * arr[i - 1] <= 0) {
                isGP = false;
                break;
            }
        }

        if (isGP) {
            System.out.println(0);
            return;
        }

        boolean canBecomeGP = false;
        for (int i = 0; i < n; i++) {
            long[] newArr = new long[n - 1];
            for (int j = 0; j < i; j++) {
                newArr[j] = arr[j];
            }
            for (int j = i + 1; j < n; j++) {
                newArr[j - 1] = arr[j];
            }

            boolean isNewArrGP = true;
            for (int j = 1; j < n - 1; j++) {
                if (newArr[j] * newArr[j - 1] <= 0) {
                    isNewArrGP = false;
                    break;
                }
            }

            if (isNewArrGP) {
                canBecomeGP = true;
                break;
            }
        }

        if (canBecomeGP) {
            System.out.println(1);
        } else {
            System.out.println(2);
        }
    }
}
