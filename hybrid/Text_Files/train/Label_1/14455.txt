#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100 + 5;
int n, a[MAXN], dp[MAXN][MAXN];
string s;

int solve(int l, int r) {
  if (l > r) return 0;
  int &res = dp[l][r];
  if (res != -1) return res;
  res = 0;
  if (l == r) return res = a[r];
  for (int i = l; i <= r; i++) {
    if (s[i] == s[l]) { res = max(res, solve(l + 1, i) + solve(i + 1, r) + a[i - l + 1]); }
  }
  return res;
}

int main() {
  cin >> n >> s;
  for (int i = 0; i < n; i++) cin >> a[i];
  memset(dp, -1, sizeof(dp));
  cout << solve(0, n - 1) << "\n";
  return 0;
}
