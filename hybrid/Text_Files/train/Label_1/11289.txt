#include <cstdio>
int main() {
    int n;
    scanf("%d\n", &n);
    char c[n + 1];
    int d[n + 1];
    scanf("%s\n", c);
    for (int i = 0; i < n; i++) scanf("%d", &d[i]);
    int x = 0;
    for (int i = 0; i < n; i++)
        if ((x = (c[i] == '<' ? x - d[i] : x + d[i])) < 0 || x > n) {
            puts("FINITE");
            return 0;
        }
    puts("INFINITE");
    return 0;
}
