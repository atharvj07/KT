#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> in(n + 1), out(n + 1);
  for (int i = 0; i < n; ++i) {
    cin >> in[i + 1];
    cin >> out[i + 1];
  }
  vector<int> pos_in(n + 1), pos_out(n + 1);
  for (int i = 1; i <= n; ++i) { pos_in[in[i]] = i; }
  for (int i = 1; i <= n; ++i) { pos_out[out[i]] = i; }
  vector<int> ans(n + 1, 0);
  for (int i = 1; i <= n; ++i) {
    int j = i;
    while (pos_out[i] > pos_out[j]) { ans[out[j]] = 1; ++j; }
  }
  int cnt = 0;
  for (int i = 1; i <= n; ++i) { cnt += ans[i]; }
  cout << cnt;
}
