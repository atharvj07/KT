import java.util.Scanner;

public class TriangleConstruction {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int c = scanner.nextInt();

        int[] sides = {a, b, c};
        int minutes = 0;

        while (!isValidTriangle(sides)) {
            int shortestSide = findShortestSide(sides);
            sides[shortestSide]++;
            minutes++;
        }

        System.out.println(minutes);
    }

    private static boolean isValidTriangle(int[] sides) {
        for (int i = 0; i < sides.length; i++) {
            int a = sides[i];
            int b = sides[(i + 1) % 3];
            int c = sides[(i + 2) % 3];

            if (a + b <= c || a + c <= b || b + c <= a) {
                return false;
            }
        }
        return true;
    }

    private static int findShortestSide(int[] sides) {
        int shortestSide = 0;
        for (int i = 1; i < sides.length; i++) {
            if (sides[i] < sides[shortestSide]) {
                shortestSide = i;
            }
        }
        return shortestSide;
    }
}
