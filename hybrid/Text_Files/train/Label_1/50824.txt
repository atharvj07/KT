import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0)
                break;
            int cnt = 0;
            boolean floor = true;
            for (int i = 0; i < n; i++) {
                String s = sc.next();
                char c = s.charAt(0);
                if (c == 'l' && floor) {
                    cnt++;
                    floor = false;
                } else if (c == 'r' && floor) {
                    cnt++;
                    floor = false;
                } else if (c == 'l' && !floor) {
                    floor = true;
                } else if (c == 'r' && !floor) {
                    floor = true;
                }
            }
            System.out.println(cnt);
        }
    }
}
