n, w = map(int, input().split())
arrays = [list(map(int, input().split()))[1:] for _ in range(n)]

# Preprocessing step: Calculate prefix sums for each array to speed up sum calculations
prefix_sums = []
for array in arrays:
    prefix_sum = [0]
    for num in array:
        prefix_sum.append(prefix_sum[-1] + num)
    prefix_sums.append(prefix_sum)

# Initialize the result list with 0s
result = [0] * w

for i in range(n):
    l = len(arrays[i])
    for start in range(w - l + 1):
        for j in range(l):
            result[start + j] = max(result[start + j], prefix_sums[i][j + 1] - prefix_sums[i][j] + result[start + j])

print(*result)
