def cook_time(k, d, t):
    on_time = k
    total_cycle = k + max(0, d - k)
    cooked_per_cycle = on_time + (total_cycle - on_time) / 2
    full_cycles = t // cooked_per_cycle
    remaining_cook_time = t - full_cycles * cooked_per_cycle
    total_time = full_cycles * total_cycle

    if remaining_cook_time <= on_time:
        total_time += remaining_cook_time
    else:
        total_time += on_time + 2 * (remaining_cook_time - on_time)
    return total_time

k, d, t = map(int, input().split())
print(cook_time(k, d, t))
