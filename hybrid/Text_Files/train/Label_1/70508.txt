import sys

for t in range(int(input())):
    input()
    k, n, m = map(int, input().split())
    a = [int(x) for x in input().split()]
    b = [int(x) for x in input().split()]
    cur = b[0]
    added = 0
    res = []

    for x in a + b:
        if x == 0:
            added += 1
            res.append(x)
        else:
            if x <= k + added:
                res.append(x)
            else:
                res = []
                break
    print(' '.join(map(str, res)) if res else -1)
