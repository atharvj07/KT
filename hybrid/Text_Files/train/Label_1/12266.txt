#include <bits/stdc++.h>
#define ll long long
#define vi vector<int>
#define vll vector<ll>
#define vb vector<bool>
#define vs vector<string>
#define rep(i, a, b) for(int i = a; i < b; i++)
#define repr(i, a, b) for(int i = a; i >= b; i--)
#define pb push_back
#define all(a) a.begin(), a.end()
#define rall(a) a.rbegin(), a.rend()
const ll mod = 1000000007LL;
using namespace std;

const int mxN = 1e5+5;
vi adj[mxN];
int dist[mxN];

int bfs(int a, int b) {
    memset(dist, -1, sizeof dist);
    queue<int> q;
    q.push(a);
    dist[a] = 0;
    while(!q.empty()) {
        int u = q.front(); q.pop();
        for(int v: adj[u]) {
            if(dist[v] == -1) {
                dist[v] = dist[u] + 1;
                if(v == b) return dist[b];
                q.push(v);
            }
        }
    }
    assert(0);
}

int main() {
    int n, m, k; cin >> n >> m >> k;
    while(m--) {
        int u, v, w; cin >> u >> v >> w;
        adj[u].pb(v);
        adj[v].pb(u);
    }
    priority_queue<array<int, 2>> pq;
    while(k--) {
        int x; cin >> x;
        pq.push({-dist[x], -x});
    }
    if(pq.empty()) {
        cout << "-1" << '\n';
        return 0;
    }
    array<int, 2> ans = pq.top(); pq.pop();
    cout << -ans[0] << '\n';
    return 0;
}
