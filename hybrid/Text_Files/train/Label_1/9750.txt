#include <iostream>

using namespace std;

const int MAXN = 1e9 + 10;

int n, ans[MAXN];

int main() {
  cin >> n;
  for (int i = 1; i <= 1000000; i++) {
    int tmp = i;
    int sum = 0;
    while (tmp) {
      sum += tmp % 10;
      tmp /= 10;
    }
    if (i + sum == n) ans[++ans[0]] = i;
  }
  cout << ans[0] << '\n';
  for (int i = 1; i <= ans[0]; i++) cout << ans[i] << '\n';
  return 0;
}
