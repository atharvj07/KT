#include<bits/stdc++.h>
using namespace std;
int main(){
	int n,sa=0,sb=0;
	cin>>n;
	int a[1001];
	for(int i=0;i<n;i++) cin>>a[i];
	int l=0,r=n-1;
	while(l<=r){
		if(a[l]>a[r]){
			sa+=a[l++];
		}else{
			sa+=a[r--];
		}
		if(l<=r){
			if(a[l]>a[r]){
				sb+=a[l++];
			}else{
				sb+=a[r--];
			}
		}
	}
	cout<<sa<<" "<<sb;
}
