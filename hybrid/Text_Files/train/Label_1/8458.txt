#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MX = 105;
int n, a[MX], b[MX];

int32_t main() {
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    queue<int> q;
    for (int i = n; i > 0; i--) {
        if (a[i] > 0) q.push(i);
        b[i] = b[i + 1] + a[i];
    }
    int ans = 0;
    while (q.size()) {
        int mx = q.front();
        int now = b[max(1, mx - 1)];
        int cnt = 0;
        for (int i = mx; i; i >>= 1) {
            if (q.front() == i) now += a[i], cnt++;
            if (now >= b[i]) break;
        }
        if (cnt == 0) ans = -1;
        for (int i = 0; i < cnt; i++)
            q.pop();
        ans++;
    }
    cout << ans;
    return 0;
}
