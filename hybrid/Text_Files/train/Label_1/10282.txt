#include<bits/stdc++.h>
#define ll long long
#define maxn 200005
using namespace std;
ll n,s[maxn],a[maxn],ans;
int main(){
    cin>>n;
    for(int i=1;i<=n;i++)cin>>a[i];
    sort(a+1,a+n+1);
    reverse(a+1,a+n+1);
    for(int i=1;i<=n;i++)s[i]=s[i-1]+a[i];
    for(int k=n;k>=1;k--){
        int p=lower_bound(a+1,a+n+1,s[k]/3)-a;
        while(p&&s[k]%3){
            s[k]-=a[p];
            ans+=a[p];
            p--;
        }
        if(s[k]%3==0)ans+=s[k]/3;
    }
    cout<<ans<<endl;
    return 0;
}
