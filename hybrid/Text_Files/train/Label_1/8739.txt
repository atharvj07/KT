#include <bits/stdc++.h>
using namespace std;
#define int long long
typedef long double ld;

typedef pair<int,int> pii;
#define F first
#define S second
vector<pii> queries;
int n, b;
int get(int x) {
	int lo = 0, hi = queries.size() - 1;
	while(lo <= hi) {
		int mid = (lo + hi) / 2;
		if(queries[mid].F < x)
			lo = mid + 1;
		else	
			hi = mid - 1;
	}
	if(lo >= queries.size() || queries[lo].F != x)
		return -1;
	return lo;
}

signed main() {
	cin >> n >> b;
	for(int i = 0; i < n; i++) {
		int a, b;
		cin >> a >> b;
		queries.push_back({a, b});
	}
	vector<int> ans(n, -1);
	int last = 0;
	vector<pii> q;
	for(auto query : queries) {
		while(!q.empty() && q.front().S <= query.F) {
			q.erase(q.begin());
		}
		if(q.size() < b) {
			if(last <= query.F)
				last = query.F;
			last += query.S;
			q.push_back({last, last + query.S});
			ans[get(query.F)] = last;
		}
	}
	for(auto x : ans)
		cout << x << " ";
	cout << endl;
	return 0;
}
