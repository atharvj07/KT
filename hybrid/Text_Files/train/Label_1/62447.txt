n = int(input())
piles = list(map(int, input().split()))

def moves_to_square(x):
    root = int(x**0.5)
    if root**2 == x:
        return 0
    return min(x - root**2, (root + 1)**2 - x)

def moves_to_non_square(x):
    if x == 0:
        return 2
    root = int(x**0.5)
    if root**2 == x:
        return 1
    return 0

square_moves = []
non_square_moves = []

for pile in piles:
    if moves_to_square(pile) == 0:
        non_square_moves.append(moves_to_non_square(pile))
    else:
        square_moves.append(moves_to_square(pile))

square_moves.sort()
non_square_moves.sort()

half_n = n // 2
square_moves_sum = sum(square_moves[:half_n])
non_square_moves_sum = sum(non_square_moves[:half_n])

print(max(square_moves_sum, non_square_moves_sum))
