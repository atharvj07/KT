t = int(input())
for _ in range(t):
    n, p, k = map(int, input().split())
    prices = list(map(int, input().split()))
    prices.sort()
    # dp[i] stores the minimum cost to buy i goods
    dp = [0] * (n + 1)
    for i in range(1, n + 1):
        if i == 1:
            dp[i] = prices[i - 1]
        elif i == 2:
            dp[i] = min(prices[i - 1], dp[i - 1] + prices[i - 1])
        else:
            dp[i] = min(dp[i - 1] + prices[i - 1], dp[i - 2] + prices[i - 1])
    max_goods = 0
    for i in range(1, n + 1):
        if dp[i] <= p:
            max_goods = i
    print(max_goods)
