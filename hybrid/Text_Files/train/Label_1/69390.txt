import sys
n, x = map(int, input().split())
if x == 0:
    print("YES")
    for i in range(n):
        print(0, end = " ")
    print("")
    sys.exit()
elif n == 1:
    print("YES")
    print(x)
    sys.exit()
if x % 2 == 1:
    if n % 2 == 0:
        print('NO')
        sys.exit()
    else:
        print('YES')
        print(0, end = " ")
        for i in range(n - 1):
            print(1, end = " ")
        print('')
        sys.exit()
p = 2
ans = []
while p <= 1000000:
    if x % p == 0:
        ans.append(p)
        x /= p
        while x % p == 0:
            ans.append(p)
            x /= p
    p += 1
if len(ans) > n:
    print('NO')
else:
    print('YES')
    for x in ans:
        print(x, end = " ")
    print('')
