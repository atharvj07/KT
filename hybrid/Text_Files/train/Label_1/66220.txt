from collections import deque

def bfs(adj, start):
    n = len(adj)
    dist = [-1] * n
    dist[start] = 0
    q = deque([start])
    while q:
        v = q.popleft()
        for u in adj[v]:
            if dist[u] == -1:
                dist[u] = dist[v] + 1
                q.append(u)
    return dist

n, x = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(n-1):
    a, b = map(int, input().split())
    adj[a-1].append(b-1)
    adj[b-1].append(a-1)

dist_from_1 = bfs(adj, 0)
dist_from_x = bfs(adj, x-1)

max_dist = 0
for d1, dx in zip(dist_from_1, dist_from_x):
    if d1 <= dx:
        max_dist = max(max_dist, dx)

print(max_dist * 2)
