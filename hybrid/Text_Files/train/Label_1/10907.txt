#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> seen(n + 1, 0);
    vector<bool> won(n + 1, false);
    int minUnique = -1;
    for (int i = 0; i < n; i++) {
      int a;
      cin >> a;
      if (!seen[a]) {
        seen[a] = 1;
        won[a] = true;
        if (minUnique == -1 || a < minUnique) {
          minUnique = a;
        }
      } else {
        won[a] = false;
      }
    }
    if (minUnique == -1) {
      cout << -1 << endl;
    } else {
      for (int i = 1; i <= n; i++) {
        if (won[i]) {
          cout << i << endl;
          break;
        }
      }
    }
  }
  return 0;
}
