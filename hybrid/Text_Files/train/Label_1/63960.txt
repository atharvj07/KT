n, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
for i in range(k):
    mn = a[0]
    for j in range(n):
        if a[j] == 0:
            continue
        if a[j] < mn:
            mn = a[j]
    if mn == 0:
        print(0)
    else:
        print(mn)
        for j in range(n):
            if a[j] > 0:
                a[j] -= mn
