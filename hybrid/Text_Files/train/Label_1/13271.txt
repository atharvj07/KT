#include<bits/stdc++.h>
#define ll long long
#define vi vector<int>
#define pii pair<int,int>
#define mp make_pair
#define pb push_back
#define fi first
#define se second
#define FOR(i,n) for(int i=0;i<n;i++)
using namespace std;
int n,a[100001],p[100001],ok[100001],cnt[100001];
vi pr;
int main()
{
	cin>>n;
	FOR(i,n){
		cin>>a[i];
		if(!ok[a[i]])ok[a[i]]=1,pr.pb(a[i]);
		cnt[a[i]]++;
	}
	cout<<"0"<<endl;
	FOR(i,pr.size()){
		int x=pr[i],chk=0;
		FOR(j,a[i]){
			if(a[j]>x)++chk;
		}
		if(chk%2!=cnt[x]%2)continue;
		FOR(j,a[i]){
			if(a[j]>x)cout<<a[i]<<" "<<a[j]<<endl;		
			swap(a[i],a[j]);
		}
	}
}
