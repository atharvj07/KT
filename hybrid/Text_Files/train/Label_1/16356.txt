#include <bits/stdc++.h>
using namespace std;
const int MXN = 1001, INF = 1e9;
char a[MXN][MXN];
int dp[MXN][MXN], b[MXN];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (a[i][j] == '<')
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + 1);
            else if (a[i][j] == '=')
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j]);
        }
    }
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            if (a[j][i] == '>')
                dp[j + 1][i + 1] = max(dp[j + 1][i + 1], dp[j][i] + 1);
            else if (a[j][i] == '=')
                dp[j + 1][i + 1] = max(dp[j + 1][i + 1], dp[j][i]);
        }
    }
    if (dp[n][m] == n * m) {
        cout << "Yes" << endl;
        for (int i = 0; i < n; i++) {
            int mn = INF;
            for (int j = 0; j < m; j++) {
                if (a[i][j] == '<')
                    mn = min(mn, dp[i][j] + 1);
                else if (a[i][j] == '=')
                    mn = min(mn, dp[i][j]);
            }
            cout << mn << ' ';
        }
        cout << endl;
        for (int i = 0; i < m; i++) {
            int mx = -INF;
            for (int j = 0; j < n; j++) {
                if (a[j][i] == '>')
                    mx = max(mx, dp[j][i] + 1);
                else if (a[j][i] == '=')
                    mx = max(mx, dp[j][i]);
            }
            cout << mx << ' ';
        }
        cout << endl;
    } else
        cout << "No" << endl;
}
