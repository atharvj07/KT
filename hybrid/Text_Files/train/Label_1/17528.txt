using namespace std;
#include <bits/stdc++.h>
const int inf = 1<<30;
int t, n, m, a, l, r, x, y, seg[2010], dp[2010][2010];
vector<pair<int,int>>rain,umbrella;
int main() {
  cin>>t>>n>>m;
  for(int i=0;i<n;i++) {
    cin>>l>>r;
    rain.push_back({l,r});
  }
  for(int i=0;i<m;i++) {
    cin>>x>>y;
    umbrella.push_back({x,y});
  }
  if(a%2==0 && rain.size()%2==1) {
    return cout<<-1<<'\n',0;
  }
  dp[0][0] = 0;
  for(int i=1;i<=a;i++) dp[i][0] = inf;
  for(int j=1;j<=2000;j++) {
    for(int i=0;i<=a;i++) {
      dp[i][j] = dp[i][j-1];
      if(i==0) continue;
      int b = lower_bound(umbrella.begin(),umbrella.end(),make_pair(i,inf))-umbrella.begin();
      for(;b>=0;b--) {
        if(umbrella[b].first==i) {
          if(seg[i]==0 || seg[i]==j) dp[i][j] = min(dp[i][j],dp[i-1][j]+umbrella[b].second);
          seg[i+1] = j;
        }
      }
      b = lower_bound(rain.begin(),rain.end(),make_pair(i,inf))-rain.begin();
      if(b<n && rain[b].first==i) {
        dp[i][j] = inf;
        seg[i+1] = 0;
        continue;
      }
      if(seg[i]>0) seg[i]++;
      if(seg[i]>0 && seg[i]!=j) dp[i][j] = inf;
    }
  }
  cout<<(dp[a][m]==inf?-1:dp[a][m])<<'\n';
}
