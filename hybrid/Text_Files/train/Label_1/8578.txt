#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> del;
  for (int i = 1; i < n; i += 2) {
    if (i + 1 < n && s[i] == s[i + 1]) { del.push_back(i); }
  }
  cout << del.size() << '\n';
  for (int i = 0; i < n; i++) {
    if (find(del.begin(), del.end(), i) == del.end()) { cout << s[i]; }
  }
  cout << '\n';
  return 0;
}
