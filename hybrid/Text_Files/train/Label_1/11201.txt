#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int MAXN = 5e5 + 5;
pair<ll, ll> a[MAXN];

int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> a[i].first >> a[i].second;
  }
  sort(a, a + n);
  int cnt = 0, end = -1;
  for (int i = 0; i < n; ++i) {
    if (a[i].first >= end) {
      cnt++;
      end = a[i].second;
    }
  }
  cout << cnt << endl;
  return 0;
}
