n = int(input())
a = list(input())
cnt1, cnt2 = 0, 0

for x in a:
    if x == 'a':
        cnt1 += 1
    else:
        cnt2 += 1

if cnt1 == cnt2:
    print(0)
    print(''.join(a))
else:
    if cnt1 > cnt2:
        for i in range(cnt1 - cnt2):
            ind = a.index('a')
            a[ind] = 'b'
    else:
        for i in range(cnt2 - cnt1):
            ind = a.index('b')
            a[ind] = 'a'
    print(cnt1 - cnt2)
    print(''.join(a))
