import sys
input = sys.stdin.readline

def solve():
    s = []
    for q in range(int(input())):
        query = list(map(int, input().split()))
        if query[0] == 1:
            s.append(query[1])
            s.sort()
        else:
            mn, mx = s[0], s[-1]
            mean = (mn+mx)/2
            print(mx-mean)

solve()
