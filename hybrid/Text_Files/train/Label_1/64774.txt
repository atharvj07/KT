n,m=map(int,input().split());rows=[[row for row in input()] for i in range(n)]
def dp(i,j,dir,cnt):
 global ans
 if i==n-1 and j==m-1 and cnt==2:ans+=1;return
 if i==n-1:dir=1
 if j==m-1:dir=2
 if dir==0:
  if rows[i][j+1]=='.':dp(i,j+1,0,cnt)
  if rows[i+1][j]=='.':dp(i+1,j,1,cnt+1)
 if dir==1:
  if rows[i+1][j]=='.':dp(i+1,j,1,cnt)
  if rows[i][j+1]=='.' and cnt!=2:dp(i,j+1,0,cnt+1)
  if rows[i][j-1]=='.':dp(i,j-1,2,cnt+1)
 if dir==2:
  if rows[i][j-1]=='.':dp(i,j-1,2,cnt)
  if rows[i+1][j]=='.':dp(i+1,j,1,cnt+1)
  if rows[i][j+1]=='.' and cnt!=2:dp(i,j+1,0,cnt+1)
ans=0
dp(0,1,0,0)
print(ans)
