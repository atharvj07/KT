import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = Integer.parseInt(scanner.nextLine());
        String s = scanner.nextLine();
        int min = 0;
        for (int i = 0; i < n; i++) {
            min += (s.charAt(i) == '+' ? 1 : -1);
        }
        System.out.println(min);
    }
}
