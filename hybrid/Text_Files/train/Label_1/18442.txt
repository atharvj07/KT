#include <iostream>
#include <vector>

using namespace std;

int perm[1000];
int inv[1000];
long long binpow(long long a, long long b, long long mod) {
  long long ans = 1;
  for (; b > 0; b >>= 1, a = a * a % mod) {
    if (b % 2) {
      ans = ans * a % mod;
    }
  }
  return ans;
}
const int N = 502;
int c[N][N], dp[N][N][4];
int main() {
  int n, m, mod;
  cin >> n >> m >> mod;
  vector<string> kek(m);
  for (int i = 0; i < m; ++i) {
    cin >> kek[i];
  }
  for (int i = 0; i < N; ++i) {
    c[i][0] = c[i][i] = 1;
    for (int j = 1; j < i; ++j) {
      c[i][j] = (c[i - 1][j - 1] + c[i - 1][j]) % mod;
    }
  }
  for (int i = 1; i <= n; ++i) {
    perm[i] = binpow(i, mod - 2, mod);
    inv[i] = perm[i];
    inv[i + i] = (mod - inv[i]) % mod;
  }
  dp[m][1][1] = 1;
  dp[m][1][2] = dp[m][1][3] = 0;
  for (int i = m - 1; i >= 0; --i) {
    for (int used = 1; used <= 3; ++used) {
      for (int j = 0; j <= 2; ++j) {
        dp[i][j][used] = 0;
        if (kek[i][j] == '1') {
          dp[i][j][used] = (dp[i][j][used] + dp[i][j + 1][2]) % mod;
        }
        dp[i][j][used] %= mod;
        if (kek[i][j] == '0' && used != 1) {
          dp[i][j][used] = (dp[i][j][used] + dp[i][j + 1][1]) % mod;
          dp[i][j + 1][1] = (dp[i][j + 1][1] +
                               dp[i][j + 1][2] * perm[j] % mod * inv[j + 1] %
                                   mod) %
                              mod;
        }
        dp[i][j][used] %= mod;
      }
    }
  }
  cout << dp[0][0][1] << "\n";
  return 0;
}
