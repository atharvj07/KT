#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2e5 + 10;

long long a[MAXN];
long long dp[MAXN][20][2];

vector<long long> pos[MAXN];

long long f(int idx, int k, bool locked, vector<long long> &v) {
    if (k < 0) return -1e18;
    if (idx == -1) return 0;
    long long &ans = dp[idx][k][locked];
    if (ans != -1) return ans;
    ans = f(idx - 1, k, locked, v);
    if (!locked && idx <= k - 1) {
        v.push_back(a[idx]);
        ans = max(ans, f(idx - 1, k - 1, true, v) + a[idx]);
        v.pop_back();
    }
    return ans;
}

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    int m;
    cin >> m;
    while (m--) {
        int k, pos;
        cin >> k >> pos;
        pos--;
        vector<long long> v;
        memset(dp, -1, sizeof dp);
        long long ans = f(n - 1, k, false, v);
        cout << v[pos] << endl;
    }
    return 0;
}
