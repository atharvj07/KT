import sys

def solve():
    n = int(input())
    a = list(map(int, input().split()))

    dp = [1] * n
    for i in range(1, n):
        if a[i] == a[i - 1] or a[i - 1] != -1 and a[i] != -1:
            dp[i] = dp[i - 1]
        else:
            dp[i] = dp[i - 1] + 1

    return dp[n - 1]


print(solve())
