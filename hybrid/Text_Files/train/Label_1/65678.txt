def optimize_essay(m, essay, n, synonyms):
    from collections import defaultdict

    # Convert essay to lowercase and split into words
    essay_words = essay.lower().split()

    # Synonyms processing
    graph = defaultdict(list)
    for x, y in synonyms:
        x, y = x.lower(), y.lower()
        graph[x].append(y)

    # Function to find the best synonym (least 'r' and shortest)
    def find_best(word):
        best_word = word
        best_r_count = word.count('r')
        best_length = len(word)
        explored = {word}
        stack = [word]
        
        while stack:
            current_word = stack.pop()
            for synonym in graph[current_word]:
                if synonym not in explored:
                    explored.add(synonym)
                    stack.append(synonym)
                    r_count = synonym.count('r')
                    if r_count < best_r_count or (r_count == best_r_count and len(synonym) < best_length):
                        best_word, best_r_count, best_length = synonym, r_count, len(synonym)
        return best_word

    # Optimize the essay
    optimized_essay = [find_best(word) for word in essay_words]

    # Count 'R's and length
    r_count = sum(word.count('r') for word in optimized_essay)
    length = sum(len(word) for word in optimized_essay)

    return r_count, length

# Read input
m = int(input())
essay = input()
n = int(input())
synonyms = [input().split() for _ in range(n)]

# Process and print output
result = optimize_essay(m, essay, n, synonyms)
print(result[0], result[1])
