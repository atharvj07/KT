import java.util.Scanner;

public class BeruTaxi {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int n = scanner.nextInt();
        double minTime = Double.MAX_VALUE;

        for (int i = 0; i < n; i++) {
            int xi = scanner.nextInt();
            int yi = scanner.nextInt();
            int vi = scanner.nextInt();

            // Calculate the distance between Vasiliy and the i-th taxi.
            double distance = Math.sqrt(Math.pow(xi - a, 2) + Math.pow(yi - b, 2));

            // Calculate the time it takes for the i-th taxi to reach Vasiliy.
            double time = distance / vi;

            // Update the minimum time if necessary.
            minTime = Math.min(minTime, time);
        }

        System.out.println(minTime);
    }
}
