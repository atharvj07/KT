def min_cost(n, m, k):
    if k == n*m or k == 0:
        return 0
    if k > n*m:
        return float('inf')
    cost = float('inf')
    # Vertical break
    for i in range(1, n):
        for j in range(k//i+1):
            if k-i*j <= m and k-i*j >= 0:
                cost = min(cost, m**2 + min_cost(i, m, j*i), m**2 + min_cost(n-i, m, k-j*i))
    # Horizontal break
    for i in range(1, m):
        for j in range(k//i+1):
            if k-i*j <= n and k-i*j >= 0:
                cost = min(cost, n**2 + min_cost(n, i, j*i), n**2 + min_cost(n, m-i, k-j*i))
    return cost

t = int(input().strip())
for _ in range(t):
    n, m, k = map(int, input().strip().split())
    print(min_cost(n, m, k))
