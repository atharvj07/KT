#include <iostream>
#include <vector>

int main() {
    int n, k;
    std::cin >> n >> k;

    std::vector<std::vector<int>> adj(n + 1);
    for (int i = 1; i < n; i++) {
        int u, v;
        std::cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    int center = -1;
    int deg = 0;
    for (int i = 1; i <= n; i++) {
        if (adj[i].size() > deg) {
            center = i;
            deg = adj[i].size();
        }
    }

    if (deg < 3) {
        std::cout << "No" << std::endl;
        return 0;
    }

    int leaves = 0;
    for (int i = 1; i <= n; i++) {
        if (adj[i].size() == 1) {
            leaves++;
        }
    }

    int hedgehogs = leaves / (k - 1) + 1;
    if (leaves % (k - 1) == 0 && hedgehogs == deg) {
        std::cout << "Yes" << std::endl;
    } else {
        std::cout << "No" << std::endl;
    }

    return 0;
}
