#include <bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;
const char nl = '\n';
typedef long long ll;
vector<int> adj[200001];

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    string s;
    cin >> n >> s;
    vector<int> cnt(26), pos(n, -1);
    for (int i = 0; i < n; i++) {
        cnt[s[i] - 'a']++;
        if (pos[s[i] - 'a'] == -1) pos[s[i] - 'a'] = i;
    }
    int distinct = 0;
    for (int i = 0; i < 26; i++) {
        distinct += cnt[i] > 0;
    }
    cout << distinct << nl;
    vector<int> color(n);
    int cur = 1;
    for (int i = 0; i < 26; i++) {
        for (int j = pos[i]; j < n; j++) {
            if (s[j] - 'a' == i) {
                color[j] = cur;
                cur++;
            }
        }
    }
    for (int i : color) cout << i << " ";
    cout << nl;
    return 0;
}
