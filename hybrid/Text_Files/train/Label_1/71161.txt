n, a, b = map(int, input().split())
if a == 1:
    print("YES")
    for i in range(n):
        s = "0" * n
        s = s[:i] + "1" + s[i+1:]
        print(s)
    exit()
if a == 2 and b == 2 and n >= 4:
    print("YES")
    print("0110")
    print("1011")
    print("1101")
    print("0110")
    exit()
print("NO")
