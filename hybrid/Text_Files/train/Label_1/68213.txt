n = int(input().strip())
pattern = input().strip()

eevee_evolutions = ["vaporeon", "jolteon", "flareon", "espeon", "umbreon", "leafeon", "glaceon", "sylveon"]

for evolution in eevee_evolutions:
    if len(evolution) == n:
        match = True
        for p, e in zip(pattern, evolution):
            if p != '.' and p != e:
                match = False
                break
        if match:
            print(evolution)
            break
