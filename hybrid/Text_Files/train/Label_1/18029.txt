#include <iostream>
#include <vector>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<int> seg(m + 1, 0);
    for (int i = 0; i < n; i++) {
        int l, r;
        std::cin >> l >> r;
        seg[l]++;
        seg[r + 1]--;
    }
    int cnt = 0;
    for (int i = 1; i <= m; i++) {
        seg[i] += seg[i - 1];
        if (seg[i] == 0) cnt++;
    }
    std::cout << cnt << std::endl;
    for (int i = 1; i <= m; i++) {
        if (seg[i] == 0) std::cout << i << ' ';
    }
    return 0;
}
