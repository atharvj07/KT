import java.util.*;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int[] p = new int[3001], c = new int[3001];
		int n = sc.nextInt(), m = sc.nextInt();
		for (int i = 1; i <= n; i++) {
			p[i] = sc.nextInt();
			c[i] = sc.nextInt();
		}
		Arrays.sort(c, 1, n + 1);
		int sum = 0;
		int cnt = 0;
		for (int i = n; i >= 1; i--) {
			if (p[i] != 1) {
				cnt++;
				sum += c[i];
				if (cnt > 0 && cnt >= sum)
					break;
			}
		}
		System.out.println(sum);
	}
}
