#include <bits/stdc++.h>
#define ll long long
using namespace std;

int main() {
    ll k;
    cin >> k;
    vector<ll> a(k);
    for (ll i = 0; i < k; i++) { cin >> a[i]; }
    ll ans = 1, sum = 0;
    for (const ll& x : a) {
        sum += x - 1;
        ans *= x;
        ans %= 1000000007;
    }
    ll p = 1ll * ans * sum, q = ans;
    p %= 1000000007;
    q %= 1000000007;
    cout << p << "/" << q << endl;
}
