n = int(input())
a = []
ans = 0
for i in range(n):
    a.append(input())
for i in range(n):
    for j in range(n):
        if a[i][j] == 'C':
            for k in range(n):
                if a[k][j] == 'C':
                    ans += 1
                if a[i][k] == 'C':
                    ans += 1
print(ans//2)
