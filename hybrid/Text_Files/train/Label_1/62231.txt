n, m = map(int, input().split())
s = [[int(i) for i in input().split()] for _ in range(n)]
ans = 0
for i in range(n):
    for j in range(m):
        if s[i][j] == 0:
            for dx, dy in zip([-1, 1, 0, 0], [0, 0, -1, 1]):
                ii, jj = i + dx, j + dy
                if 0 <= ii < n and 0 <= jj < m and s[ii][jj] == 1:
                    ans += 1
print(ans)
