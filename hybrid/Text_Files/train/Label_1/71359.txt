import math

n, m = map(int, input().split())
p = list(map(int, input().split()))
for _ in range(m):
    a, b = map(int, input().split())
    a -= 1
    b -= 1
    p[a], p[b] = p[b], p[a]

ans = 0
for i in range(n):
    for j in range(i + 1, n):
        if p[i] > p[j]:
            ans += 0.5 ** (j - i)

print(ans)
