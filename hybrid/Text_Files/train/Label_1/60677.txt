from sys import stdin

n = int(stdin.readline())
s = sorted([(a*b*c, a, b, c) for a, b, c in [map(int, stdin.readline().split()) for _ in range(n)]])

if s[-1][0] > s[-2][0] * 2:
    print(1)
    print(s[-1][1], end="")
else:
    a, b, c, d = s[-1][1], s[-1][2], s[-1][3], s[-2][1]
    if a == b and c == d or a == c and b == d:
        print(1)
        print(s[-1][1], end="")
    elif b == c and c == d:
        print(2)
        print(s[-1][1], s[-2][1], end="")
    else:
        print(1)
        print(s[-1][1], end="")
