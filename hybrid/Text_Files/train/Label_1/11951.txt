#include <bits/stdc++.h>
using namespace std;
const int maxn = 3e5 + 10;
string s;
int main() {
    int n;
    cin >> n >> s;
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == '?') {
            s[i] = ')';
            if (i + 1 < n && s[i + 1] == '(') {
                s[i] = '(';
            }
        }
    }
    stack<char> st;
    for (int i = 0; i < n; i++) {
        if (s[i] == '(') {
            st.push(s[i]);
        } else {
            if (st.empty()) {
                cout << ":(";
                return 0;
            }
            st.pop();
        }
    }
    if (!st.empty()) {
        cout << ":(";
        return 0;
    }
    cout << s;
    return 0;
}
