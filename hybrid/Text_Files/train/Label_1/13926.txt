#include <iostream>
#include <vector>

int main() {
  int n, k, x;
  std::cin >> n >> k >> x;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) { std::cin >> a[i]; }
  int ans = 0;
  for (int i = 0; i < k && i < n; ++i) { ans += x; }
  for (int i = k; i < n; ++i) { ans += a[i]; }
  std::cout << ans << std::endl;
  return 0;
}
