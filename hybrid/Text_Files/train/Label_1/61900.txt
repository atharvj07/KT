import bisect
n,m,ta,tb,k = map(int,input().split())
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
dp=[[1<<32,-1]]
for i in range(min(n,k+1)):
    dp+=[[1<<32,-1] for j in range(min(m,k-i+1))]
    for j in range(min(m,k-i+1)):
        if j>0: dp[-1][j]=min(dp[-1][j],dp[-1][j-1])
        if i<n: dp[-1][j]=min(dp[-1][j],a[i]+ta+tb)
        if j>0 and i>0:
            l,r=bisect.bisect_right(b,a[i]+ta),len(b)
            if l>0: dp[-1][j]=min(dp[-1][j],dp[i-1][j-1]+b[l-1]+tb)
            if l<len(b): dp[-1][j]=min(dp[-1][j],dp[i-1][j-1]+b[l]+tb)
print(-1 if dp[-1][k]==1<<32 else dp[-1][k])
