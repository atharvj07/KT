import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        long n = in.nextLong();
        boolean wins = false;
        for (long i = a + b; i < n; i *= 2) {
            wins = !wins;
        }
        System.out.println(wins ? "Stas" : "Masha");
    }
}
