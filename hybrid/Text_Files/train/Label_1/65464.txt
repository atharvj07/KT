import queue
import sys

inqueue = queue.Queue()

n, b = map(int,input().strip().split(' '))
t = 0
for i in range(n):
    ta, d = map(int,input().strip().split(' '))
    if t<ta:t=ta
    while not inqueue.empty() and inqueue.queue[0]<=t: t=inqueue.get()+1
    if inqueue.qsize()<b: inqueue.put(t+d-1)
    else: print(-1); continue
    t=t+d
 
while not inqueue.empty(): print(inqueue.get()+1,end=' ')
