#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e6 + 10;
bitset<maxn>isPrime;
void sieve() {
  isPrime.set();
  isPrime[0] = isPrime[1] = 0;
  for (long long p = 2; p < maxn; p++) {
    if (!isPrime[p]) continue;
    for (long long i = p + p; i < maxn; i += p) isPrime[i] = 0;
  }
}
int32_t main() {
  sieve();
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  vector<int> ans;
  for (int i = 0; i < n; i++) {
    bool valid = 1;
    for (int j : ans) {
      if (!isPrime[a[i] + j]) {
        valid = 0;
        break;
      }
    }
    if (valid) ans.push_back(a[i]);
  }
  cout << ans.size() << endl;
  for (int x : ans) cout << x << " ";
}
