import java.util.*;

public class Main {
    static List<List<Integer>> flowers = new ArrayList<>();
    static int[] moods;
    static int[][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        moods = new int[n + 1];
        for (int i = 1; i <= n; i++) moods[i] = sc.nextInt();
        dp = new int[n + 5][1 << 10];
        for (int i = 0; i < m; i++) {
            List<Integer> interval = new ArrayList<>();
            interval.add(sc.nextInt());
            interval.add(sc.nextInt());
            flowers.add(interval);
        }
        int ans = dfs(1, 0);
        System.out.println(ans);
    }

    static int dfs(int idx, int mask) {
        if (idx == flowers.size()) return 0;
        if (dp[idx][mask] != 0) return dp[idx][mask];
        int op1 = 0, op2 = 0;
        List<Integer> interval = flowers.get(idx);
        if ((mask >> (interval.get(1) - interval.get(0))) % 2 == 0) {
            op1 = dfs(idx + 1, mask);
            for (int i = interval.get(0); i <= interval.get(1); i++) op1 += moods[i];
        }
        op2 = dfs(idx + 1, mask | ((1 << (interval.get(1) - interval.get(0)))));
        return dp[idx][mask] = Math.max(op1, op2);
    }
}
