#include<iostream>
#include<algorithm>
#include<vector>
#include<stack>
using namespace std;

vector<int> adj[200005];
stack<int> dfs_stack;
int vis[200005], col[200005], depth[200005], col_count, min_path[200005], max_path[200005];

void dfs(int node, int par){
    vis[node]=1;
    dfs_stack.push(node);
    col[node]=col_count;
    depth[node]=depth[par]+1;
    for(int child: adj[node]){
        if(vis[child]==0){
            dfs(child,node);
        }
    }
}
int main(){
    int t;
    cin>>t;
    while(t--){
        for(int i=0;i<200005;i++){
            adj[i].clear();
            col[i]=-1;
            vis[i]=0;
            min_path[i]=1e9;
            max_path[i]=-1;
        }
        col_count=0;
        int n,m,a,b;
        cin>>n>>m>>a>>b;
        for(int i=0;i<m;i++){
            int u,v;
            cin>>u>>v;
            adj[u].push_back(v);
            adj[v].push_back(u);
        }
        dfs(1,1);
        while(dfs_stack.size()>0){
            int top=dfs_stack.top();
            dfs_stack.pop();
            if(top==a || top==b){
                max_path[top]=depth[top];
                min_path[top]=depth[top];
            }
            else{
                max_path[top]=max({max_path[adj[top][0]],max_path[adj[top][1]]});
                min_path[top]=min({min_path[adj[top][0]],min_path[adj[top][1]]});
            }
        }
        int ans=0;
        for(int i=1;i<=n;i++){
            if(i==a || i==b) continue;
            if(min_path[i]<max_path[i]){
                ans+=n-1-col[i];
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
