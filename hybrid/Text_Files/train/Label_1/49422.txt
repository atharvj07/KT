import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        // Read input parameters
        String[] nk = br.readLine().split(" ");
        int n = Integer.parseInt(nk[0]);
        int k = Integer.parseInt(nk[1]);

        // Read input stripe
        String stripe = br.readLine();

        // Initialize array to store repainted colors
        int[] repainted = new int[n];

        // Initialize number of repainted cells
        int repaints = 0;

        // Initialize repainted color
        int color = 1;

        // Iterate over each cell in the stripe
        for (int i = 0; i < n; i++) {
            // If current cell color is the same as previous cell color
            if (i > 0 && stripe.charAt(i) == stripe.charAt(i - 1)) {
                // Repaint current cell with the next color
                repainted[i] = color++;
                // Increment number of repainted cells
                repaints++;
            } else {
                // Keep current cell color
                repainted[i] = stripe.charAt(i) - 'A' + 1;
            }
        }

        // Output number of repainted cells
        System.out.println(repaints);

        // Output repainted stripe
        for (int i = 0; i < n; i++) {
            System.out.print((char) (repainted[i] + 'A' - 1));
        }
        System.out.println();
    }
}
