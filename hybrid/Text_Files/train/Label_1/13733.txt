#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    map<int, int> m;
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        ans += m[a[i]];
        m[a[i]]++;
    }
    cout << ans << endl;
    return 0;
}
