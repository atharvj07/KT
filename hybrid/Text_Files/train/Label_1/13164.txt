#include <iostream>
#include <cstdio>
#include <vector>
#include <numeric>

using namespace std;

bool check(vector<int> v, int x)
{
    int s = 0;
    for(int i=0; i<v.size(); i++)
    {
        s += v[i];
        if(s==x)
            return 1;
    }
    return 0;
}

int main()
{
    int t; cin >> t;
    while(t--)
    {
        int n, x; cin >> n >> x;
        string s; cin >> s;
        vector<int>v;
        for(int i=0; i<n; i++)
        {
            int a; a = s[i] - '0';
            v.push_back(2*a - 1);
        }
        for(int i=1; i<n; i++)
        {
            v[i] += v[i-1];
        }
        int ans = 0;
        for(int i=0; i<n; i++)
        {
            for(int j=i; j<n; j++)
            {
                if(j==i)
                {
                    if(v[j]==x)
                        ans++;
                }
                else
                {
                    if(check(v, x+v[i]-v[j]))
                        ans++;
                }
            }
        }
        if(x==0)
            ans++;
        if(ans == -1)
            cout << "-1\n";
        else
            cout << ans << "\n";
    }
}
