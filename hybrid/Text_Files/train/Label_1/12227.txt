#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N = 1e8;

ll n, m, a[N], b[N];
vector<pair<ll, ll>> v;

int32_t main() {
	cin >> n >> m;
	for (int i = 0; i < m; i++) {
		ll x, y;
		cin >> x >> y;
		v.push_back(make_pair(x, y));
	}
	ll mx = -1;
	for (int i = 0; i < v.size(); i++) {
		if (i > 0 && v[i].first - v[i - 1].first > 1) {
			cout << "IMPOSSIBLE" << endl;
			return 0;
		}
		if (i == 0) a[v[i].first] = v[i].second;
		else {
			ll x = v[i - 1].first, y = v[i - 1].second;
			a[v[i].first] = max(a[x], y) + (v[i].first - x);
		}
		mx = mx > a[v[i].first] ? mx : a[v[i].first];
	}
	cout << mx << endl;
	return 0;
}
