from collections import Counter

for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    a = Counter(a)
    even, odd = 0, 0
    for v in a.values():
        if v % 2 == 1: odd, v = odd +1, v%2
        else: even, v = even +1, v%2
    if odd % 2 == 1: odd -= 1
    if odd == 0: print("Tie")
    elif even == 0: print("Bob")
    elif odd: print("Alice")
    else: print("Bob")
