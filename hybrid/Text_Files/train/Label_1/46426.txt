import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int t = input.nextInt();
    while (t-- > 0) {
      int n = input.nextInt();
      int[] a = new int[n];
      int[] b = new int[n];
      for (int i = 0; i < n; i++) {
        a[i] = input.nextInt();
      }
      for (int i = 0; i < n; i++) {
        b[i] = input.nextInt();
      }
      boolean valid = true;
      for (int i = 0; i < n; i++) {
        if (a[i] != b[(n + i - 1) % n]) {
          valid = false;
          break;
        }
      }
      System.out.println(valid ? "Yes" : "No");
    }
  }
}
