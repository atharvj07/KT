#include <iostream>
#include <string>
#include <algorithm>

using namespace std;

int main() {
    string s, t;
    cin >> s >> t;
    int n = s.size(), m = t.size();
    int ans = n + m;
    for (int i = 0, j = 0; i < n && j < m; i++, j++) {
        if (s[i] == t[j]) ans -= 2;
    }
    cout << ans << endl;
    return 0;
}
