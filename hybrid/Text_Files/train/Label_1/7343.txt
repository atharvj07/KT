#include <bits/stdc++.h>
using namespace std;
#define mod 1000000007LL
typedef long long ll;
const ll N = 1e7 + 5;
bool vis[N];
vector<ll> ans;
void sieve() {
  for (ll i = 1; i < N; i++) {
    ll x = i, rev = 0;
    while (x > 0) {
      rev = rev * 10 + x % 10;
      x /= 10;
    }
    if (i == rev) {
      if (__builtin_popcountll(i) % 2 == 0) ans.push_back(i);
      vis[i] = 1;
    }
  }
}
int main() {
  ll k, p;
  cin >> k >> p;
  sieve();
  ll sum = 0;
  for (ll i = 0; i < k && i < ans.size(); i++) {
    sum = (sum + ans[i] + p) % p;
  }
  cout << sum << endl;
  return 0;
}
