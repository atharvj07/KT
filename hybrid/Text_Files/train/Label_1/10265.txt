#include <bits/stdc++.h>
using namespace std;

const int mxN = 2005;
int up[mxN], down[mxN], n, k;

int main(){
    cin >> n >> k;
    vector< int > v(n);
    for(auto &x : v) cin >> x;
    sort(v.begin(), v.end());
    int tp = 0, td = 0;
    for(auto x : v){
        if(td + x <= tp){
            td += x;
            down[td]++;
        }
        else{
            tp += x;
            up[tp]++;
        }
    }
    int mx = max(td, tp);
    int ans = 2*mx;
    for(int i = 1; i < mx; i++){
        ans = min(ans, max(down[i], up[i]) + i + i + mx - min(down[i], up[i]));
    }
    cout << ans << endl;
}
