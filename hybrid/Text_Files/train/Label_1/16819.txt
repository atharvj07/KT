#include<bits/stdc++.h>
using namespace std;
int a[105];
vector<int> v;
int main(){
    int n;
    cin>>n;
    int sm=0;
    for(int i=1; i<=n; i++){
        cin>>a[i];
        sm+=a[i];
    }
    if(a[1]*2>sm){
        cout<<"0\n";
        return 0;
    }
    int fl=0;
    sm/=2;
    for(int i=2; i<=n; i++){
        if(a[i]<=sm && a[i]*2<=a[1]){
            fl=1;
            v.push_back(i);
        }
    }
    if(!fl){
        cout<<"0\n";
        return 0;
    }
    cout<<v.size()+1<<'\n';
    cout<<"1 ";
    for(int x: v)    cout<<x<<' ';
    cout<<'\n';
    return 0;
}
