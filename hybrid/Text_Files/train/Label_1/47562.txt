import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();
        String[] strings = new String[n];
        for (int i = 0; i < n; i++) {
            strings[i] = scanner.nextLine();
        }
        int q = scanner.nextInt();
        scanner.nextLine();
        for (int i = 0; i < q; i++) {
            int k = scanner.nextInt();
            String permutation = scanner.nextLine();
            List<String> sorted = new ArrayList<>(Arrays.asList(strings));
            Collections.sort(sorted, (a, b) -> {
                for (int j = 0; j < Math.min(a.length(), b.length()); j++) {
                    int indexA = permutation.indexOf(a.charAt(j));
                    int indexB = permutation.indexOf(b.charAt(j));
                    if (indexA != indexB) {
                        return indexA - indexB;
                    }
                }
                return a.length() - b.length();
            });
            int index = sorted.indexOf(strings[k - 1]);
            System.out.println(index + 1);
        }
    }
}
