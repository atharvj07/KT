from collections import deque

n, m, d = map(int, input().split())
edges = []
for i in range(m):
    a, b = map(int, input().split())
    edges.append((a, b))

adj = [[] for i in range(n + 1)]
for a, b in edges:
    adj[a].append(b)
    adj[b].append(a)

vis = [False] * (n + 1)
par = [-1] * (n + 1)
deg = [0] * (n + 1)

q = deque([1])
vis[1] = True
while q:
    u = q.popleft()
    for v in adj[u]:
        if not vis[v]:
            vis[v] = True
            q.append(v)
            par[v] = u
            deg[u] += 1

if deg[1] == d:
    print("YES")
    for i in range(2, n + 1):
        print(par[i], i)
else:
    print("NO")
