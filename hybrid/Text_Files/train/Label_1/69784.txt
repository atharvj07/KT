def find_overlap(N, M, map1, map2):
    for i in range(N - M + 1):
        for j in range(N - M + 1):
            match = True
            for k in range(M):
                if map1[i + k][j:j + M] != map2[k]:
                    match = False
                    break
            if match:
                return i + 1, j + 1
    return -1, -1  # This case should not happen based on problem statement

N, M = map(int, input().split())
map1 = [input().strip() for _ in range(N)]
map2 = [''.join(row) for row in zip(*[input().strip() for _ in range(M)])]
i, j = find_overlap(N, M, map1, map2)
print(i, j)
