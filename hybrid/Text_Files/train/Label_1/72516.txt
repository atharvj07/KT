from collections import Counter

def solve():
    t = int(input())
    for _ in range(t):
        n, k = map(int, input().split())
        n1 = str(n)
        cnt = Counter(n1)
        if len(cnt) <= k:
            print(n)
        else:
            i = 0
            while len(cnt) > k:
                if n1[i] != '9':
                    n1 = n1[:i] + '9' + n1[i + 1:]
                    cnt = Counter(n1)
                else:
                    i += 1
            print(n1)

solve()
