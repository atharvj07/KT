#include <iostream>

using namespace std;

const int NMAX = 65000;
int n, v[NMAX];
long long dp[NMAX];

int main() {
    cin >> n;
    for (int i = 0; i < n; ++i) { cin >> v[i]; }
    dp[n] = 0;
    for (int i = n - 1; i >= 0; --i) {
        dp[i] = dp[i + 1];
        if ((v[i] & 1) == 0) { dp[i] += dp[i + 1] + 1; }
        if (i + 1 < n && (v[i] % 10 * 10 + v[i + 1] & 1) == 0) { dp[i] += dp[i + 2] + 1; }
    }
    cout << dp[0] << "\n";
    return 0;
}
