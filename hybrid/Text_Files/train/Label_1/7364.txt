#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using vi = vector<int>;
const ll mod = 1e9 + 7;

int main() {
  int n, h;
  cin >> n >> h;
  vi a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  sort(a.begin(), a.end());

  unordered_map<int, int> cnt;
  for (int i = 0; i < n; ++i) {
    if (i == 0 || a[i] != a[i - 1]) {
      cnt[a[i]]++;
    }
  }

  int mx = 0;
  for (const auto &p : cnt) { mx = max(mx, p.second); }

  ll ans = 1;
  for (int i = 1; i <= h - a[0]; ++i) { ans = (ans * mx) % mod; }
  cout << ans << '\n';

  return 0;
}
