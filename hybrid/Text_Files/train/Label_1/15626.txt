#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long n;
  cin >> n;
  vector<long long> res;
  for (long long k = 0;; k++) {
    long long v = (k + 1) * k / 2;
    if (v == n)
      res.push_back(k + 1);
    if (v > n)
      break;
  }
  if (!res.empty()) {
    for (long long e : res)
      cout << e << endl;
  } else
    cout << -1 << endl;
  return 0;
}
