import sys

n = int(input())
rings = []
for _ in range(n):
    a, b, h = map(int, input().split())
    rings.append((a, b, h))
rings.sort(key=lambda ring: (ring[1], ring[0]))
dp = [0] * n
dp[0] = rings[0][2]
for i in range(1, n):
    for j in range(i):
        if rings[i][0] > rings[j][1]:
            dp[i] = max(dp[i], dp[j] + rings[i][2])
    dp[i] = max(dp[i], rings[i][2])
print(max(dp))
