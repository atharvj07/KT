import math

def rotate_point(cx, cy, angle, px, py):
    # Rotating a point (px, py) around point (cx, cy) by angle
    s = math.sin(angle)
    c = math.cos(angle)

    # Translate point back to origin
    px -= cx
    py -= cy

    # Rotate point
    xnew = px * c - py * s
    ynew = px * s + py * c

    # Translate point back
    px = xnew + cx
    py = ynew + cy
    return px, py

def solve(n, q, vertices, queries):
    # Initial pins at vertices 1 and 2
    pin1 = 0
    pin2 = 1
    for query in queries:
        if query[0] == 1:
            f, t = query[1]-1, query[2]-1
            if pin1 == f:
                pin1 = t
            else:
                pin2 = t
            # Assuming the polygon rotates around pin2
            # Calculating angle of rotation based on the position of the new and old pin
            # Since it's a strict convex polygon, we assume a simple rotation without flipping
            # This is a simplification and may not reflect real physical behavior
            angle = math.atan2(vertices[pin2][1] - vertices[pin1][1], vertices[pin2][0] - vertices[pin1][0]) - math.pi
            for i in range(n):
                vertices[i] = rotate_point(vertices[pin2][0], vertices[pin2][1], angle, *vertices[i])
        else:
            v = query[1]-1
            print(f"{vertices[v][0]:.10f} {vertices[v][1]:.10f}")

# Example usage

n, q = map(int, input().split())
vertices = [tuple(map(int, input().split())) for _ in range(n)]
queries = [tuple(map(int, input().split())) for _ in range(q)]

solve(n, q, vertices, queries)
