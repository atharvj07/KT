import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int[] l = new int[N];
        int[] r = new int[N];
        for (int i = 0; i < N; i++) {
            l[i] = sc.nextInt();
            r[i] = sc.nextInt();
        }

        int[] max = new int[M];
        for (int d = 1; d <= M; d++) {
            int cnt = 0;
            for (int i = 0; i < N; i++) {
                if (l[i] <= d && d <= r[i]) {
                    cnt++;
                }
            }
            max[d - 1] = cnt;
        }

        for (int i = 0; i < M; i++) {
            System.out.println(max[i]);
        }
    }
}
