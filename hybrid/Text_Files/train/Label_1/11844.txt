using namespace std;
typedef long long ll;
const double EPS = 1e-9;
const int inf = 1e9;
struct node{
  int v,d;
};
typedef pair <int,int> pi;
int n,k,t;
int a[210000];
int ans[210000];
int dist[210000];
vector <pi> adj[210000];
queue <node> q;
bool flag;
int find_path(int u,int p,int d){
  ans[u] = p;
  dist[u] = d;
  for(auto i:adj[u]){
    int v = i.se;
    if(v==p) continue;
    int w = i.fi;
    if(find_path(v,u,d+w)) return true;
    if(flag) return true;
  }
  return false;
}
int main(){
  ios_base::sync_with_stdio(false); cin.tie(0);
  cin>>n>>t>>k;
  a[t] = 1;
  int sum = 0;
  REP(i,1,t) cin>>a[i],sum+=a[i];
  if(a[t]<=k || sum!=n-1){
    cout<<-1<<endl;
    return 0;
  }
  a[t] = k+1;
  REP(i,1,t) if(a[i]==0){
    a[i] = n-sum-k;
    break;
  }
  int j=0;
  for(int i=t-1;i>=1;i--){
    REP(ind,0,a[i]) j+=2,adj[j].pb(mkp(1,j+1)),adj[j+1].pb(mkp(1,j));
    if(a[i]==0) continue;
    int p = j+1;
    REP(ind,0,a[i-1]){
      j+=2,adj[p].pb(mkp(1,j)),adj[j].pb(mkp(1,p));
      p=j;
    }
  }
  q.push(node{1,-1});
  int dead=0,ct=0;
  while(!q.empty()){
    node u = q.front(); q.pop();
    if(sz(adj[u.v])==1 && u.v!=1){
      dead+=u.d==1;
      ct+=u.d==0;
    }
    for(auto i:adj[u.v]){
      int v = i.se;
      int w = i.fi;
      if(v==u.v) continue;
      q.push(node{v,u.d+w});
    }
  }
  if(dead!=k){
    cout<<-1<<endl;
    return 0;
  }
  if(find_path(1,-1,0)) flag=true;
  int ct_1=0,ct_0=0;
  REP(i,1,n+1){
    if(dist[i]==1 && i!=1) ct_1++;
    if(dist[i]==0 && i!=1) ct_0++;
  }
  if(ct_1!=k){
    cout<<-1<<endl;
    return 0;
  }
  cout<<n<<endl;
  REP(i,1,n){
    int u = i;
    int cnt=0;
    while(u!=1){
      u = ans[u];
      if(dist[u]==1) cnt++;
    }
    if(cnt==1) continue;
    cout<<i<<" "<<ans[i]<<endl;
  }
  return 0;
}
