public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();
        while (t-- > 0) {
            long[] arr = new long[3];
            arr[0] = sc.nextInt();
            arr[1] = sc.nextInt();
            arr[2] = sc.nextInt();

            Arrays.sort(arr);
            if (arr[1] == arr[2]) {
                System.out.println("YES");
                System.out.println(arr[0] + " " + arr[1] + " " + arr[1]);
            } else {
                System.out.println("NO");
            }
        }
    }
}
