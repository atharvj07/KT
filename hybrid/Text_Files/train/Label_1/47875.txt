import java.util.Scanner;

public class Main {

    private static int[] a;
    private static int n;
    private static boolean[] lucky;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        a = new int[n + 1];
        lucky = new boolean[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 1; i <= n; i++) {
            if (a[i] == 4 || a[i] == 7) {
                lucky[i] = true;
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (!lucky[i]) {
                for (int j = i + 1; j <= n; j++) {
                    if (!lucky[j]) {
                        ans++;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
