import sys

day1 = sys.stdin.readline().strip()
day2 = sys.stdin.readline().strip()

if day1 == day2:
  print("YES")
elif (day1 == "monday" and day2 == "tuesday") or (day1 == "tuesday" and day2 == "monday"):
  print("YES")
elif (day1 == "wednesday" and day2 == "thursday") or (day1 == "thursday" and day2 == "wednesday"):
  print("YES")
elif (day1 == "friday" and day2 == "saturday") or (day1 == "saturday" and day2 == "friday"):
  print("YES")
elif (day1 == "sunday" and day2 == "monday") or (day1 == "monday" and day2 == "sunday"):
  print("YES")
else:
  print("NO")
