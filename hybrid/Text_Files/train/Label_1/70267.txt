t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    if k == 3:
        if n % 2 == 0:
            if n % 4 == 0:
                print(n//2, n//4, n//4)
            else:
                print(2, (n-2)//2, (n-2)//2)
        else:
            print(1, (n-1)//2, (n-1)//2)
    else:
        ans = [1] * (k-3) + [2, (n-k+1)//2, (n-k+1)//2]
        print(*ans)
