#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  stack<pair<char, int>> st;
  for (int i = n - 1; i >= 0; i--) {
    if (st.empty()) {
      st.push({s[i], 0});
    } else if (st.top().second + 1 == s[i] - 'a') {
      st.top().second++;
      if (st.top().second == (n + 1) / 2) {
        st.pop();
      }
    } else {
      st.push({s[i], 0});
    }
  }
  string ans = "";
  while (!st.empty()) {
    ans += st.top().first;
    st.pop();
  }
  reverse(ans.begin(), ans.end());
  cout << ans << "\n";
}
