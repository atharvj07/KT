import sys
n, k, s, t = map(int, input().split())
cars = [(a, b) for a, b in (map(int, input().split()) for _ in range(n))]
gas = sorted(list(set(map(int, input().split()))))
best = -1
for p, v in cars:
  nk = min(k, (s-1)//v)
  nj = (s-1)%v
  a1 = (nj+1)//2 * 2
  a2 = (nj+1+v)//2
  tc = t-a1-a2
  if tc >= 0 and tc%2==0 and (len(gas)>=nk or (nk-len(gas))%2==0):
    best = p if best==-1 else min(best, p)
print(best)
