#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        int n;
        string s;
        cin >> n >> s;

        vector<int> ans(n);
        for (int i = 1; i < n; i++) {
            if (s[i] != s[i - 1]) {
                if (s[i] == 'R') {
                    ans[i] = 1;
                } else {
                    ans[i - 1] = 1;
                }
            }
        }

        if (s[n - 1] != s[0]) {
            if (s[n - 1] == 'R') {
                ans[n - 1] = 1;
            } else {
                ans[0] = 1;
            }
        }

        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += ans[i];
        }

        cout << sum << endl;
    }
}
