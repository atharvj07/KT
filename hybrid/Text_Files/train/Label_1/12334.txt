#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (n); ++i)

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  rep(i, n) cin >> a[i];
  vector<pair<int, int>> seg(m);
  rep(i, m) { cin >> seg[i].first >> seg[i].second; }
  sort(seg.begin(), seg.end(), [](const pair<int, int> &l, const pair<int, int> &r) {
    return l.first != r.first ? l.first < r.first : l.second < r.second;
  });
  vector<bool> used(m, false);
  vector<int> res;
  vector<int> ans;
  int ansd = 0;
  int s = 0, t = 0;
  rep(i, m) {
    if (s >= t) {
      s = seg[i].first;
      t = seg[i].second;
      res.clear();
      res.push_back(i);
      used[i] = true;
    } else if (seg[i].first <= t) {
      res.push_back(i);
      t = seg[i].second;
      used[i] = true;
    }
    if (a[t] - a[s - 1] > ansd) {
      ans = res;
      ansd = a[t] - a[s - 1];
    }
  }
  cout << ansd << endl;
  cout << ans.size() << endl;
  rep(i, ans.size()) { cout << ans[i] + 1 << endl; }
}
