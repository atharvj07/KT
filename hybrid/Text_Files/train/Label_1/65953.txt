def is_swap_possible(s):
    open_1 = 0
    open_0 = 0
    for c in s:
        if c == '1':
            if open_0 > 0:
                return False
            open_1 += 1
        else:
            if open_1 > 0:
                open_1 -= 1
            else:
                open_0 += 1
    return open_1 == 0 and open_0 == 0


for _ in range(int(input())):
    n = int(input())
    s = input()
    if is_swap_possible(s):
        print("YES")
        print("(" * n, ")", sep="")
        print(")" * n, "(", sep="")
    else:
        print("NO")
