#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> a(n), b(n + 1);
  cin >> a[0];
  b[0] = a[0];
  for (int i = 1; i < n; i++) {
    cin >> a[i];
    b[0] += a[i];
    b[i + 1] = b[0];
  }
  const ll INF = 1e18;
  ll i = 0;
  while (b[i + 1] < k) i++;
  cout << i << endl;
}
