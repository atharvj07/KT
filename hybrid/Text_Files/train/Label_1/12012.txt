#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 1;
int a[N], b[N];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  int ans = 1, gap = 0, lst = -1;
  for (int i = 1; i <= n; i++) {
    if (a[i] <= lst)
      gap = i, b[gap]++;
    else
      lst = a[i];
    ans = max(ans, b[gap] + 1);
  }
  cout << ans << endl;
  return 0;
}
