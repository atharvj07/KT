#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int n, s;
    cin >> n >> s;
    vector<tuple<int, int, int>> arr(n);
    for (int i = 0; i < n; i++) {
        int x, a, b;
        cin >> x >> a >> b;
        arr[i] = {x, a, b};
    }
    sort(arr.begin(), arr.end(), [](auto &a, auto &b) {
        return get<1>(a) + get<2>(a) > get<1>(b) + get<2>(b);
    });
    int cnt = 0, sum = 0;
    for (int i = 0; i < n; i++) {
        if (cnt + get<0>(arr[i]) <= s) {
            cnt += get<0>(arr[i]);
            sum += get<0>(arr[i]) * (get<1>(arr[i]) + get<2>(arr[i]));
        } else {
            sum += (s - cnt) * (get<1>(arr[i]) + get<2>(arr[i]));
            cnt = s;
            break;
        }
    }
    cout << sum << endl;
}
