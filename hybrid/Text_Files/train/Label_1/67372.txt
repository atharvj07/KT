import sys

n = int(input())
sums = [int(input()) for _ in range(n)]

a = [1] * n
for i in range(1, n):
    for d in range(1, 10):
        if sums[i] >= d and sums[i] - d > sums[i - 1]:
            a[i] = int(''.join([str(x) for x in range(a[i - 1] + 1, a[i - 1] + 1 + d)]))
            sums[i] -= d
            break

for x in a:
    print(x)
