from collections import defaultdict

def find_cycle(vertex, graph, visited, rec_stack):
    visited[vertex] = True
    rec_stack[vertex] = True
    for neighbour in graph[vertex]:
        if not visited[neighbour]:
            if find_cycle(neighbour, graph, visited, rec_stack):
                return True
        elif rec_stack[neighbour]:
            return True
    rec_stack[vertex] = False
    return False

def is_cyclic(graph, n):
    visited = [False] * n
    rec_stack = [False] * n
    for node in range(n):
        if not visited[node]:
            if find_cycle(node, graph, visited, rec_stack):
                return True
    return False

def can_make_acyclic(graph, n):
    for u in list(graph):
        for v in list(graph[u]):
            graph[u].remove(v)
            if not is_cyclic(graph, n):
                return "YES"
            graph[u].add(v)
    return "NO"

n, m = map(int, input().split())
graph = defaultdict(set)
for _ in range(m):
    u, v = map(int, input().split())
    graph[u-1].add(v-1)

print(can_make_acyclic(graph, n))
