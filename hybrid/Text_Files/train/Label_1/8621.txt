#include <iostream>

int main() {
    long long H, n;
    std::cin >> H >> n;
    long long d[n];
    for (long long i = 0; i < n; i++) { std::cin >> d[i]; }
    long long s = 0;
    for (long long i = 0; i < n; i++) { s += d[i]; }
    if (s <= 0) {
        std::cout << -1 << std::endl;
        return 0;
    }
    std::cout << (H + s - 1) / s << std::endl;
    return 0;
}
