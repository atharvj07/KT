p, q, l, r = map(int, input().split())
a = [list(map(int, input().split())) for i in range(p)]
c = [list(map(int, input().split())) for i in range(q)]
ans = []
for i in range(l, r + 1):
    x = 0
    for j in range(q):
        if i + c[j][0] <= c[j][1]:
            x = 1
            break
    y = 0
    for j in range(p):
        if a[j][0] <= i + c[j][0] <= a[j][1]:
            y = 1
            break
    if x + y == 2:
        ans.append(i)
print(len(ans))
