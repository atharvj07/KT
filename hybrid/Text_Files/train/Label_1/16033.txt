#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll n, m, k;
string s, t;
vector<int> f(1010), g(1010);

int main() {
  cin >> n >> m >> k;
  cin >> s >> t;
  for (int i = 1; i <= n; ++i) {
    f[i] = f[i - 1] + (s[i - 1] == s[i - 1 % k]);
  }
  for (int j = 1; j <= n; ++j) {
    g[j] = g[j - 1] + (s[j - 1] == t[j - 1]);
  }
  ll ans = 0;
  for (int i = 0; i + k <= n; ++i) {
    ll cur = f[i + k] - f[i];
    int j = i + k - 1;
    while (j < n && g[j + 1] > g[i]) cur++, j++;
    ans = max(ans, cur);
  }
  cout << ans << endl;
  return 0;
}
