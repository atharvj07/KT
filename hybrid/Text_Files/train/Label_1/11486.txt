#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
  string s, t;
  int k;
  cin >> s >> k;
  if (s.size() < k) {
    cout << "Impossible";
    return 0;
  }
  vector<char> v;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '?' || s[i] == '*')
      continue;
    v.push_back(s[i]);
    if (s[i] == s[i + 1] && (s[i + 2] == '?' || s[i + 2] == '*')) {
      i += 2;
    } else if (s[i + 1] == '?' || s[i + 1] == '*') {
      i++;
    }
  }
  if (v.size() != k) {
    cout << "Impossible";
    return 0;
  }
  for (int i = 0; i < v.size(); i++) {
    t += v[i];
  }
  cout << t;
  return 0;
}
