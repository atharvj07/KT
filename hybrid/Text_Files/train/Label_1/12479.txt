#include <bits/stdc++.h>
using namespace std;
int cost_to_char(char c) {
  static unordered_map<char, int> costs;
  if (costs.count(c) == 0) {
    costs[c] = min(c - 'a', 'z' - c + 1);
  }
  return costs[c];
}
int main() {
  string s;
  cin >> s;
  int ans = cost_to_char(s[0]);
  for (int i = 1; i < s.size(); i++) {
    ans += min(cost_to_char(s[i]) + 1, cost_to_char(s[i - 1]) + 1);
  }
  cout << ans << endl;
  return 0;
}
