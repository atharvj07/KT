#include<bits/stdc++.h>
using namespace std;
#define nl '\n'
#define ll long long

struct node{
    int x, y, c;
    node(int _x, int _y, int _c){
        x = _x;
        y = _y;
        c = _c;
    }
};

char grid[1005][1005];
int vis[1005][1005];
int dx[] = {0, 0, 1, -1};
int dy[] = {1, -1, 0, 0};
int n, m;

bool isvalid(int x, int y){
    if(x >= 0 && x < n && y >= 0 && y < m && !vis[x][y] && grid[x][y] != '#'){
        return true;
    }
    return false;
}

int bfs(){
    queue<node>q;
    for(int i = 0; i < n; i++){
        for(int j = 0; j < m; j++){
            if(grid[i][j] >= '1' && grid[i][j] <= '3'){
                q.push(node(i, j, grid[i][j]));
                vis[i][j] = 1;
            }
        }
    }
    int ans = 0;
    while(!q.empty()){
        int sz = q.size();
        while(sz--){
            node top = q.front(); q.pop();
            for(int i = 0; i < 4; i++){
                int nx = top.x + dx[i];
                int ny = top.y + dy[i];
                if(isvalid(nx, ny)){
                    if(top.c >= '1' && top.c <= '3' && grid[nx][ny] == '.'){
                        ans++;
                        grid[nx][ny] = top.c;
                    }
                    vis[nx][ny] = 1;
                    q.push(node(nx, ny, grid[nx][ny]));
                }
            }
        }
    }
    return ans;
}

int main(){
    cin >> n >> m;
    for(int i = 0; i < n; i++){
        for(int j = 0; j < m; j++){
            cin >> grid[i][j];
        }
    }
    int ans = bfs();
    cout << ans << nl;
    return 0;
}
