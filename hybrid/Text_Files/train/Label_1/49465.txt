import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    List<long[]> stones = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      long[] stone = new long[3];
      stone[0] = in.nextInt();
      stone[1] = in.nextInt();
      stone[2] = in.nextInt();
      Arrays.sort(stone);
      stones.add(stone);
    }

    Collections.sort(stones, (a, b) -> Long.compare(a[0] * a[1] * a[2], b[0] * b[1] * b[2]));

    if (stones.get(0)[0] == stones.get(0)[1] && stones.get(0)[1] == stones.get(0)[2]) {
      System.out.println(1);
      System.out.println(1);
    } else if (stones.get(n - 1)[0] == stones.get(n - 1)[1] || stones.get(n - 1)[1] == stones.get(n - 1)[2]) {
      System.out.println(1);
      System.out.println(n);
    } else {
      System.out.println(2);
      System.out.println(n - 1 + " " + n);
    }
  }
}
