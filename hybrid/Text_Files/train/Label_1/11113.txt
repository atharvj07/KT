#include <iostream>
#include <vector>

int main() {
  long long k, a, b;
  std::cin >> k >> a >> b;
  std::vector<std::vector<int>> alice(4, std::vector<int>(4, 0));
  std::vector<std::vector<int>> bob(4, std::vector<int>(4, 0));
  for (int i = 1; i <= 3; ++i) {
    for (int j = 1; j <= 3; ++j) { std::cin >> alice[i][j]; }
  }
  for (int i = 1; i <= 3; ++i) {
    for (int j = 1; j <= 3; ++j) { std::cin >> bob[i][j]; }
  }
  a = alice[a][b];
  b = bob[b][a];
  int alice_score = 0, bob_score = 0;
  while (k--) {
    if (a == b) {
      ++k;
      continue;
    }
    if ((a == 1 && b == 3) || (a == 2 && b == 1) || (a == 3 && b == 2)) {
      ++alice_score;
    } else {
      ++bob_score;
    }
    a = alice[a][b];
    b = bob[b][a];
  }
  std::cout << alice_score << " " << bob_score << std::endl;
  return 0;
}
