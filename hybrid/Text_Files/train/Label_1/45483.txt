import java.util.*;

public class VasyaAndPark {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt(); // Number of glades
        int m = scanner.nextInt(); // Number of trails

        List<Integer>[] graph = new List[n + 1]; // Adjacency list for the graph
        for (int i = 1; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }

        // Read the trails and build the graph
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            graph[x].add(y);
            graph[y].add(x);
        }

        // Perform a depth-first search (DFS) to check if the walk is possible
        boolean[] visited = new boolean[n + 1];
        boolean possible = dfs(1, 1, graph, visited);

        // If the walk is possible, print 0
        if (possible) {
            System.out.println(0);
        } else {
            // If the walk is impossible, find the minimum number of trails to add
            int minTrailsToAdd = findMinimumTrailsToAdd(graph);
            System.out.println(minTrailsToAdd);
        }
    }

    // DFS function to check if the walk is possible
    private static boolean dfs(int currentGlade, int startGlade, List<Integer>[] graph, boolean[] visited) {
        visited[currentGlade] = true;

        for (int nextGlade : graph[currentGlade]) {
            if (!visited[nextGlade]) {
                if (dfs(nextGlade, startGlade, graph, visited)) {
                    return true;
                }
            } else if (nextGlade != startGlade) {
                return true; // Found a cycle
            }
        }

        return currentGlade == startGlade && isAllVisited(visited);
    }

    // Check if all glades are visited
    private static boolean isAllVisited(boolean[] visited) {
        for (int i = 1; i < visited.length; i++) {
            if (!visited[i]) {
                return false;
            }
        }
        return true;
    }

    // Find the minimum number of trails to add to make the walk possible
    private static int findMinimumTrailsToAdd(List<Integer>[] graph) {
        int[] degree = new int[graph.length];
        for (int i = 1; i < graph.length; i++) {
            degree[i] = graph[i].size();
        }

        int oddDegreeCount = 0;
        for (int i = 1; i < graph.length; i++) {
            if (degree[i] % 2 != 0) {
                oddDegreeCount++;
            }
        }

        // The minimum number of trails to add is equal to half of the number of odd-degree glades
        return oddDegreeCount / 2;
    }
}
