from collections import Counter

# Read input
n = int(input())
languages_known = list(map(int, input().split()))
m = int(input())
audio_languages = list(map(int, input().split()))
subtitles_languages = list(map(int, input().split()))

# Count how many scientists know each language
language_count = Counter(languages_known)

# Calculate the number of very pleased and almost pleased scientists for each movie
pleased_counts = []
for i in range(m):
    very_pleased = language_count[audio_languages[i]]
    almost_pleased = language_count[subtitles_languages[i]]
    pleased_counts.append((very_pleased, almost_pleased, i + 1))

# Sort to find the movie with maximum very pleased and then maximum almost pleased scientists
best_movie = max(pleased_counts)

# Output the index of the chosen movie
print(best_movie[2])
