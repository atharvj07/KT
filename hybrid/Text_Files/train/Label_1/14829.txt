#include<bits/stdc++.h>
using namespace std;
int main(){
    long long d,n,m,i,c,l,x,p,ans=LLONG_MAX,k;
    cin>>d>>n>>m;
    vector<pair<long long,long long>>v;
    while(m--){
        cin>>x>>p;
        v.push_back({p,x});
    }
    sort(v.begin(),v.end());
    for(auto z: v){
        if(n>=d){
            cout<<ans<<endl;
            return 0;
        }
        if(z.second<=n){
            c+=z.first;
            n=d;
            ans=min(ans,c);
        }
        else{
            k=(z.second-n)/d;
            c+=k*z.first;
            n+=k*d;
            if(n+d-z.second>=d){
                n=d;
            }
            else{
                n+=d-z.second;
                c+=z.first;
            }
            ans=min(ans,c);
        }
    }
    if(n<d){
        cout<<-1<<endl;
        return 0;
    }
    cout<<ans<<endl;
    return 0;
}
