import java.util.Scanner;

public class WormEvolution {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] lengths = new int[n];

        for (int i = 0; i < n; i++) {
            lengths[i] = scanner.nextInt();
        }

        boolean found = false;

        // Iterate over all possible pairs of worms
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                // Check if the length of the third worm equals the sum of the first two
                for (int k = 0; k < n; k++) {
                    if (k != i && k != j && lengths[k] == lengths[i] + lengths[j]) {
                        System.out.println(i + 1 + " " + (j + 1) + " " + (k + 1));
                        found = true;
                        break;
                    }
                }

                if (found) {
                    break;
                }
            }

            if (found) {
                break;
            }
        }

        if (!found) {
            System.out.println(-1);
        }

        scanner.close();
    }
}
