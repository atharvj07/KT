import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int x = input.nextInt();
        int y = input.nextInt();

        System.out.println(Math.max(0, (n * y - x * 100 + 99) / 100));
    }
}
