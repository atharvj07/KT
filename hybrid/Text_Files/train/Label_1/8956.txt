#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long

const int maxN = 2e3 + 5;
int n, arr[maxN];

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> arr[i];
  int ans = 0;
  for (int i = 1; i < n; ++i) {
    if (arr[i] < arr[i - 1]) {
      ans++;
      arr[i] *= -1;
    }
  }
  cout << ans << '\n';
  return 0;
}
