#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
int n, m, p[MAXN], cnt;
vector<int> adj[MAXN];
void DFS(int u) {
    for (auto v : adj[u]) {
        if (p[v] != p[u]) {
            p[v] = p[u];
            DFS(v);
        }
    }
}
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) p[i] = i;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
    }
    for (int i = 1; i <= n; i++) {
        if (p[i] == i) {
            DFS(i);
            cnt++;
        }
    }
    cout << cnt - 1 << endl;
}
