#include<vector>
#include<iostream>
int main(){
    std::string a,b;
    std::cin>>a>>b;
    int ans=a.size()-b.size();
    std::vector<int>l,r;
    int cl=0,cr=0;
    for(int j=0;j<b.size();j++){
        for(int i=cl;i<a.size();i++){
            if(a[i]==b[j]){
                l.push_back(i);
                cl=i+1;
                break;
            }
        }
    }
    for(int j=b.size()-1;j>=0;j--){
        for(int i=cr;i>=0;i--){
            if(a[i]==b[j]){
                r.push_back(i);
                cr=i-1;
                break;
            }
        }
    }
    if(l.size()!=b.size()){
        std::cout<<"-1";
        return 0;
    }
    int sz=l.size();
    for(int i=0;i<sz-1;i++){
        ans-=r[i+1]-l[i]-1;
    }
    std::cout<<ans;
    return 0;
}
