#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll s, x;
int main() {
   cin >> s >> x;
   if(x == 0 && s == 0) cout << 1 << endl;
   else if(x == 0 && s != 0) cout << 0 << endl;
   else if((s - x) % 2 == 0 && (s - x) >= 0 && 0 <= (s + x) / 2) cout << 1 << endl;
   else cout << 0 << endl;
}
