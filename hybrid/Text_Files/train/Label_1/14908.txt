#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string a, b;
  cin >> a >> b;
  vector<pair<int, int>> ua, ub;
  pair<int, int> na(0, 1), nb(0, 1);
  for (int i = 0; i < n - 1; i++) {
    if (a[i] == 'N')
      na.second++;
    else if (a[i] == 'E')
      na.first++;
    else if (a[i] == 'S')
      na.second--;
    else
      na.first--;
    if (b[i] == 'N')
      nb.second++;
    else if (b[i] == 'E')
      nb.first++;
    else if (b[i] == 'S')
      nb.second--;
    else
      nb.first--;
    ua.push_back(na), ub.push_back(nb);
  }
  int ans = 0;
  for (int i = 0; i < n - 1; i++)
    if (ua[i] == na && ub[i] == nb) ans = i + 1;
  if (na == ua[n - 2] && nb == ub[n - 2] && ans)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
