#include <cstdio>
#include <vector>
#include <iostream>
#define MAXN 200200
using namespace std;
int n, lmax[MAXN], rmin[MAXN], a[MAXN], b[MAXN];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) { cin >> b[i]; }
  lmax[1] = a[1];
  for (int i = 2; i <= n; i++) { lmax[i] = max(a[i], lmax[i - 1]); }
  rmin[n] = b[n];
  for (int i = n - 1; i >= 1; i--) { rmin[i] = min(b[i], rmin[i + 1]); }
  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = i; j <= n; j++) {
      if (lmax[j] == rmin[i]) { ans++; }
    }
  }
  cout << ans << endl;
  return 0;
}
