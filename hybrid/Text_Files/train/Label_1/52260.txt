import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long res = n, op = 0;
        while (!isPrime(res)) {
            while (n % 2 == 0) {
                n /= 2;
                op++;
            }
            for (long i = 3; i * i <= n; i += 2) {
                while (n % i == 0) {
                    n /= i;
                    op++;
                }
            }
            if (n > 2) {
                op++;
                n = 1;
            }
            res = (long) Math.sqrt(res);
            op++;
        }
        System.out.println(res + " " + op);
    }
    private static boolean isPrime(long n) {
        if (n < 2) {
            return false;
        }
        for (long i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                return false;
            }
        }
        return true;
    }
}
