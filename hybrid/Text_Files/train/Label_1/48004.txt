import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;
import java.util.stream.Collectors;

class Main {
    private static final int[][] DIRECTIONS = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};

    public static void main(String[] args) throws IOException {
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(System.in))) {
            StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
            int rows = Integer.parseInt(tokenizer.nextToken());
            int columns = Integer.parseInt(tokenizer.nextToken());
            char[][] grid = new char[rows][columns];
            for (int i = 0; i < rows; i++) {
                grid[i] = reader.readLine().toCharArray();
            }
            boolean[][] visited = new boolean[rows][columns];
            int maxScore = 0;
            for (int i = 0; i < rows; i++) {
                for (int j = 0; j < columns; j++) {
                    if (grid[i][j] != '.' && !visited[i][j]) {
                        int score = bfs(grid, visited, i, j);
                        maxScore = Math.max(maxScore, score);
                    }
                }
            }
            System.out.println(maxScore);
        }
    }

    private static int bfs(char[][] grid, boolean[][] visited, int row, int column) {
        Queue<int[]> queue = new LinkedList<>();
        queue.add(new int[]{row, column});
        char color = grid[row][column];
        int score = 0;
        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            if (visited[current[0]][current[1]]) {
                continue;
            }
            visited[current[0]][current[1]] = true;
            score++;
            for (int[] direction : DIRECTIONS) {
                int nextRow = current[0] + direction[0];
                int nextColumn = current[1] + direction[1];
                if (nextRow >= 0 && nextRow < grid.length && nextColumn >= 0 && nextColumn < grid[0].length && grid[nextRow][nextColumn] == color && !visited[nextRow][nextColumn]) {
                    queue.add(new int[]{nextRow, nextColumn});
                }
            }
        }
        return score;
    }
}
