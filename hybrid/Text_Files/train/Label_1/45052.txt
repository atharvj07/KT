import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        for (int i = 0; i < d; i++) {
            int n = sc.nextInt();
            char[][] a = new char[n][n];
            int x = 0, y = n - 1, t = 1;
            while (t <= n * n) {
                for (int j = y; j >= x; j--) {
                    a[j][x] = '#';
                    t++;
                }
                for (int j = x + 1; j <= y; j++) {
                    a[x][j] = '#';
                    t++;
                }
                for (int j = x + 1; j <= y; j++) {
                    a[j][y] = '#';
                    t++;
                }
                for (int j = y - 1; j >= x; j--) {
                    a[y][j] = '#';
                    t++;
                }
                x++;
                y--;
            }
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    System.out.print(a[j][k]);
                }
                System.out.println();
            }
            System.out.println();
        }
    }
}
