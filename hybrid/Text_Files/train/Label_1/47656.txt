import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int d = sc.nextInt();
        int[][] arr = new int[d][n];
        if (k < d) {
            System.out.println(-1);
            return;
        }
        for (int i = 0; i < d; i++) {
            for (int j = 0; j < n; j++) {
                arr[i][j] = (j % k) + 1;
            }
        }
        for (int[] a : arr) {
            System.out.println(Arrays.toString(a).replaceAll("\\[|\\]|,", ""));
        }
    }
}
