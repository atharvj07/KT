import java.util.Scanner;

class Main {
    static final int MOD = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int S = sc.nextInt();
        int[] A = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextInt();
        }

        int[][] dp = new int[N + 1][S + 1];
        dp[0][0] = 1;
        for (int i = 0; i < N; i++) {
            for (int j = 0; j <= S; j++) {
                dp[i + 1][j] = dp[i][j];
                if (j + A[i] <= S) {
                    dp[i + 1][j + A[i]] += dp[i][j];
                    dp[i + 1][j + A[i]] %= MOD;
                }
            }
        }

        long ans = 0;
        for (int i = 0; i < N; i++) {
            for (int j = i + 1; j <= N; j++) {
                ans += dp[i][S];
                ans %= MOD;
            }
        }

        System.out.println(ans);
    }
}
