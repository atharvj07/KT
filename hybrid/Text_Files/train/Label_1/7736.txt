#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int mod=1e9+7;
int pref[1000005]={};
int main(){
    int n,x;
    cin>>n>>x;
    for(int i=0;i<n;i++){
        int a;cin>>a;pref[a]++;
    }
    int cnt=0;
    for(int l=1;l<=x;l++){
        int lsum=0,rsum=pref[x];
        for(int r=l;r<=x;r++){
            lsum+=pref[r-1];
            rsum-=pref[r];
            if(lsum<=rsum)cnt++;
        }
    }
    cout<<cnt<<endl;
}
