for _ in range(int(input())):
    n = int(input())
    vis = [0] * 10000
    a = []
    for i in range(n):
        x = int(input())
        a.append(x)
        vis[x] += 1
    k = 0
    for i in range(n):
        f = 0
        for j in range(10):
            if i == 0:
                if vis[1000 * a[i] + j] == 0:
                    k += 1
                    a[i] = 1000 * a[i] + j
                    vis[a[i]] += 1
                    f = 1
                    break
            else:
                if vis[1000 * a[i] + j] == 0 and 1000 * a[i] + j not in a:
                    k += 1
                    a[i] = 1000 * a[i] + j
                    vis[a[i]] += 1
                    f = 1
                    break
        for j in range(10):
            if i == 0:
                if vis[100 * a[i] + j] == 0:
                    k += 1
                    a[i] = 100 * a[i] + j
                    vis[a[i]] += 1
                    f = 1
                    break
            else:
                if vis[100 * a[i] + j] == 0 and 100 * a[i] + j not in a:
                    k += 1
                    a[i] = 100 * a[i] + j
                    vis[a[i]] += 1
                    f = 1
                    break
        for j in range(10):
            if i == 0:
                if vis[10 * a[i] + j] == 0:
                    k += 1
                    a[i] = 10 * a[i] + j
                    vis[a[i]] += 1
                    f = 1
                    break
            else:
                if vis[10 * a[i] + j] == 0 and 10 * a[i] + j not in a:
                    k += 1
                    a[i] = 10 * a[i] + j
                    vis[a[i]] += 1
                    f = 1
                    break
        for j in range(10):
            if vis[a[i] * 10 + j] == 0:
                k += 1
                a[i] = a[i] * 10 + j
                vis[a[i]] += 1
                f = 1
                break
        if f == 0:
            print(-1)
            break

    print(k)
    for i in a:
        print(i)
