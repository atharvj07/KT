#include <bits/stdc++.h>
#define int long long
using namespace std;
const int md = 1e9 + 7;
set<int> ans;
signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    int n;
    cin >> n;
    vector<int> ar(n + 1);
    for (int i = 0; i <= n; i++) cin >> ar[i];
    for (int i = 1; i <= n; i++) {
        int p = ar[i];
        p -= ar[i - 1];
        int j = i;
        while (j <= n) {
            // cerr << p << ' ';
            if (p < 0 || ar[j] - p < 0) break;
            if (ar[j] - p == ar[j] % i) ans.insert(i), j += i;
            else break;
        }
    }
    cout << ans.size() << '\n';
    for (auto it : ans) cout << it << ' ';
    cout << '\n';
}
