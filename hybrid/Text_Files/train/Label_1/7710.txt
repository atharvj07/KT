#include<bits/stdc++.h>
using namespace std;
int vis[105][105];
int a[105][105];
int dx[]={1,-1,0,0};
int dy[]={0,0,1,-1};
int f,r,c,k;
void bfs(int x,int y){
    queue<pair<int,int>>q;
    q.push(make_pair(x,y));
    vis[x][y]=1;
    while(!q.empty()){
        pair<int,int>t=q.front();
        q.pop();
        for(int i=0;i<4;i++){
            int tx=t.first+dx[i];
            int ty=t.second+dy[i];
            if(tx<0||tx>=f||ty<0||ty>=r)continue;
            if(a[tx][ty]==a[x][y]&&vis[tx][ty]==0){
                vis[tx][ty]=1;
                q.push(make_pair(tx,ty));
            }
        }
    }
}
int check(){
    int sum=0,cnt=0,mn=INT_MAX;
    for(int j=0;j<r;j++){
        for(int i=0;i<f;i++){
            if(vis[i][j]==0){
                sum++;
                bfs(i,j);
                cnt=0;
                for(int ij=0;ij<r;ij++){
                    for(int ii=0;ii<f;ii++){
                        if(vis[ii][ij]==1){
                            cnt++;
                            vis[ii][ij]=0;
                        }
                        if(a[ii][ij]==0)continue;
                        if(a[ii][ij]!=a[i][j])return -1;
                    }
                }
                mn=min(mn,cnt);
            }
        }
    }
    k=k-sum;
    if(k<0)return -1;
    return mn;
}
int main()
{
    cin>>f>>r>>k;
    for(int i=0;i<f;i++){
        for(int j=0;j<r;j++){
            cin>>a[i][j];
        }
    }
    int ans=check();
    cout<<ans<<endl;
    return 0;
}
