#include <iostream>
#include <vector>
#include <sstream>
#include <string>
using namespace std;

int main() {
    int n;
    cin >> n;
    while (n--) {
        string ip;
        cin >> ip;
        vector<string> parts;
        stringstream ss(ip);
        string part;
        while (getline(ss, part, ':')) {
            if (part == "::") {
                int cnt = 0;
                for (int i = 0; i < parts.size(); i++) {
                    if (parts[i] == "0") cnt++;
                }
                for (int i = 0; i < 8 - parts.size() - cnt; i++) {
                    parts.push_back("0");
                }
                break;
            }
            parts.push_back(part);
        }
        for (int i = 0; i < parts.size(); i++) {
            while (parts[i].size() < 4) {
                parts[i] = '0' + parts[i];
            }
        }
        for (int i = 0; i < parts.size(); i++) {
            cout << parts[i];
            if (i != parts.size() - 1) cout << ':';
        }
        cout << '\n';
    }
    return 0;
}
