import java.util.*;

public class Main {
    static final int INF = Integer.MAX_VALUE;
    static Map<String, Integer> costMap;
    static Map<String, String> changeMap;
    static int[] memo;
    static String[] a;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        costMap = new HashMap<>();
        changeMap = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            int c = sc.nextInt();
            costMap.put(s, c);
        }
        int m = sc.nextInt();
        for (int i = 0; i < m; i++) {
            String s = sc.next();
            String t = sc.next();
            changeMap.put(s, t);
        }
        a = new String[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.next();
        }
        memo = new int[1 << n];
        Arrays.fill(memo, INF);
        int res = dfs(0, 0);
        System.out.println(res == INF ? -1 : res);
    }

    static int dfs(int s, int t) {
        if (s == (1 << a.length) - 1) {
            return 0;
        }
        if (memo[s] != INF) {
            return memo[s];
        }
        int min = INF;
        for (int i = 0; i < a.length; i++) {
            if ((s & (1 << i)) == 0) {
                min = Math.min(min, costMap.get(a[i]) + dfs(s | (1 << i), t | (1 << i)));
                if (!changeMap.containsKey(a[i])) {
                    continue;
                }
                for (int j = 0; j < a.length; j++) {
                    if ((t & (1 << j)) != 0) {
                        min = Math.min(min, dfs(s | (1 << i), t | (1 << i)) + costMap.get(changeMap.get(a[i])));
                    }
                }
            }
        }
        return memo[s] = min;
    }
}
