#include <iostream>
#include <string>
#include <algorithm>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        std::string s;
        std::cin >> s;
        std::string res;
        std::sort(s.begin(), s.end());
        for (int i = 0; i < (int)s.size(); i++) {
            if (i + 1 == (int)s.size() || s[i] != s[i + 1]) {
                res += s[i];
            }
            i++;
        }
        std::cout << res << std::endl;
    }
    return 0;
}
