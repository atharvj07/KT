import java.util.*;
import java.io.*;

public class Main {
    final static int N = 2505;
    final static int M = 5005;
    static int[] s1 = new int[N];
    static int[] s2 = new int[N];
    static int[] t = new int[M];
    static int n, m;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int k = Integer.parseInt(st.nextToken());
        n = Integer.parseInt(st.nextToken());
        m = 2 * n * k;
        for (int i = 0; i < k; i++) {
            String str = br.readLine();
            for (int j = 0; j < n; j++) {
                s1[j] = str.charAt(j) - 'a';
                s2[j] = s1[j];
            }
            Arrays.sort(s1, 0, n);
            for (int j = 0; j < n; j++) {
                t[i * n + j] = s1[j];
                t[(i + 1) * n - 1 - j] = s2[j];
            }
        }
        Arrays.sort(t, 0, m);
        for (int i = 0; i < m; i++) {
            if (i < m / 2 && t[i] != t[m - 1 - i]) {
                System.out.println(-1);
                return;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print((char) (t[i] + 'a'));
        }
    }
}
