#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int frq[N];
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    fill(frq, frq + N, 0);
    for (int i = 0, x; i < n; i++) {
      cin >> x;
      frq[x]++;
    }
    int ans = 0;
    for (int i = 0; i < N; i++) {
      if (frq[i] & 1) ans++;
    }
    cout << ans << '\n';
  }
}
