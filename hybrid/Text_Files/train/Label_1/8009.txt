#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == -1) continue;
        int mn = a[i];
        for (int j = i+1; j < n; j++) {
            if (a[j] == -1) continue;
            if (a[j] % mn != 0) {
                mn = -1;
                break;
            }
        }
        if (mn != -1) ans++;
        for (int j = i; j < n; j++) {
            if (a[j] == -1) continue;
            if (a[j] % mn == 0) a[j] = -1;
        }
    }
    cout << ans << endl;
    return 0;
}
