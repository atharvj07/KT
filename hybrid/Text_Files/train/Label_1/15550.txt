#include <cstdio>
#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  int anton = 0, danik = 0;
  for (string s; cin >> s; n--)
    s[0] == 'A' ? anton++ : danik++;
  return (anton > danik ? cout << "Anton" : anton < danik ? cout << "Danik"
                                                         : cout << "Friendship"),
         0;
}
