#include <bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            int b; cin >> b;
            if (i != j) a[i] |= b;
        }
    }
    for (int i = 0; i < n; i++) cout << a[i] << " ";
    return 0;
}
