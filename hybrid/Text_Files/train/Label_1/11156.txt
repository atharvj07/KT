#include<bits/stdc++.h>
using namespace std;
#define i64 long long int
const int mod = 1e9+7;
const int N = 2e5+10;
i64 h[N];
int main(){
    i64 n, m, x, y, ans=0;
    cin>>n>>m;
    while(m--){
        cin>>x>>y;
        if(x>y) swap(x, y);
        ++h[x];
        --h[y];
    }
    for(i64 i=1; i<=n; i++){
        h[i]+=h[i-1];
        if(h[i]!=1) ans=1;
    }
    if(!ans) cout<<"Yes"<<endl;
    else cout<<"No"<<endl;
    return 0;
}
