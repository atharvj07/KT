#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;

int field[501][501], pre[4][501][501];
string s;

void check(int x, int y, int dir) {
    if (x > 0 && field[x - 1][y] != 1) {
        if (field[x][y] == 0) pre[dir][x][y] = 1;
        pre[dir][x - 1][y] += pre[dir][x][y];
    }
    if (x < 500 && field[x + 1][y] != 1) {
        if (field[x][y] == 0) pre[dir][x][y] = 1;
        pre[dir][x + 1][y] += pre[dir][x][y];
    }
    if (y > 0 && field[x][y - 1] != 1) {
        if (field[x][y] == 0) pre[dir][x][y] = 1;
        pre[dir][x][y - 1] += pre[dir][x][y];
    }
    if (y < 500 && field[x][y + 1] != 1) {
        if (field[x][y] == 0) pre[dir][x][y] = 1;
        pre[dir][x][y + 1] += pre[dir][x][y];
    }
}

int main() {
    int x, y, x0, y0;
    cin >> x >> y >> x0 >> y0;
    cin >> s;
    for (int i = 0; i < s.length(); i++) {
        if (s[i] == 'U')
            field[x0 - 1][y0] = 1;
        else if (s[i] == 'D')
            field[x0 + 1][y0] = 1;
        else if (s[i] == 'L')
            field[x0][y0 - 1] = 1;
        else
            field[x0][y0 + 1] = 1;

        check(x0, y0, 0);
        check(x0 - 1, y0, 1);
        check(x0 + 1, y0, 2);
        check(x0, y0 - 1, 3);
        check(x0, y0 + 1, 3);

        if (s[i] == 'U')
            x0--;
        else if (s[i] == 'D')
            x0++;
        else if (s[i] == 'L')
            y0--;
        else
            y0++;
    }

    for (int l = s.length() - 1; l >= 0; l--) {
        for (int i = 0; i < x; i++)
            for (int j = 0; j < y; j++)
                pre[l % 4][i][j] += pre[(l + 1) % 4][i][j];
    }

    for (int l = 0; l <= s.length(); l++) cout << pre[l % 4][x0 - 1][y0 - 1] << " ";

    return 0;
}
