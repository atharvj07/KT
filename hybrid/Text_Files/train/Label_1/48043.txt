import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

class Main{
	public static void main(String[] args){
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int[] arr = new int[n];
		for(int i = 0; i < n; ++i){
			arr[i] = sc.nextInt();
		}
        int sum = 0;
		Arrays.sort(arr);
		double median;
        if(n % 2 == 0){
            median = ((double)arr[n / 2] + (double)arr[(n / 2) - 1]) / 2.0;
        }else{
            median = arr[n / 2];
        }
        for(int el : arr){
            sum += el;
        }
        double mean = (double)sum / n;
		double maxSkewness = -Double.MAX_VALUE;
		ArrayList<Integer> ans = new ArrayList<>();
		for(int i = 0; i < n; ++i){
			for(int j = i; j < n; ++j){
				double skewness = mean - ((double)arr[j] + (double)arr[i]) / 2.0;
				if(skewness > maxSkewness){
					maxSkewness = skewness;
					ans.clear();
					for(int k = i; k <= j; ++k){
						ans.add(arr[k]);
					}
				}
			}
		}
		System.out.println(ans.size());
		for(int el : ans){
			System.out.print(el + " ");
		}
		System.out.println();
	}
}
