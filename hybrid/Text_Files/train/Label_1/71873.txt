import sys
input = sys.stdin.readline

N = int(input().strip())
points = set()
for _ in range(N):
    x, y = map(int, input().strip().split())
    points.add((x, y))

ans = 0
for (x1, y1) in points:
    for (x2, y2) in points - {(x1, y1)}:
        for (x3, y3) in points - {(x1, y1), (x2, y2)}:
            if (x1 - x2) * (y3 - y2) == (x3 - x2) * (y1 - y2):
                ans += 1

print(ans)
