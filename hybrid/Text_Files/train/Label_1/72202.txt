import sys

def solve():
    n = int(input())
    w = list(map(int, input().split()))
    adj = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u, v, c = map(int, input().split())
        adj[u].append((v, c))
        adj[v].append((u, c))
    
    def dfs(u, p, fuel):
        ret = fuel + w[u - 1]
        for v, c in adj[u]:
            if v == p:
                continue
            ret = max(ret, dfs(v, u, min(fuel + w[u - 1], c)))
        return ret
    
    print(dfs(1, 1, 0))

solve()
