#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(m);
    for (int i = 0; i < m; ++i) cin >> a[i];
    sort(a.begin(), a.end());
    int cur = a[0], cnt = 0, ans = 0;
    for (int i = 0; i < m; ++i) {
        if (cur == a[i]) cnt++;
        else {
            ans += min(cnt, n);
            cur = a[i];
            cnt = 1;
        }
    }
    ans += min(cnt, n);
    cout << ans << endl;
    return 0;
}
