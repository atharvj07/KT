import java.util.*;

public class Main {
    static int[] m, s = new int[100];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        m = new int[n];
        for (int i = 0; i < n; i++) {
            String t = sc.next();
            m[i] = sc.nextInt();
            for (char c : t.toCharArray()) {
                s[c - '0']++;
            }
        }
        Arrays.sort(s);
        Arrays.sort(m);
        int ans = 0;
        for (int i = 9, j = n - 1; i >= 0; i--) {
            while (j >= 0 && s[i] > 0) {
                if (s[i] <= m[j]) {
                    ans += (s[i] + 1) * j;
                    s[i] = 0;
                } else {
                    ans += (m[j] + 1) * j;
                    s[i] -= m[j];
                }
                j--;
            }
            if (s[i] > 0) {
                ans += (s[i] + 1) * (k - j - 1);
                s[i] = 0;
            }
        }
        System.out.println(ans);
    }
}
