import java.util.*;

class Main {
    static int[][] edge;
    static int[][] edge2;
    static int[] deg;
    static int[] deg2;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int E = sc.nextInt();
        int S = sc.nextInt();
        int T = sc.nextInt();
        int R = sc.nextInt();
        edge = new int[N][N];
        edge2 = new int[N][N];
        deg = new int[N];
        deg2 = new int[N];
        for (int i = 0; i < M; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            edge[a][b] = 1;
            edge[b][a] = 1;
            deg[a]++;
            deg[b]++;
        }
        for (int i = 0; i < E; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            edge2[a][b] = 1;
            edge2[b][a] = 1;
            deg2[a]++;
            deg2[b]++;
        }
        int[][] dis = new int[N][R + 1];
        for (int i = 0; i < N; i++) {
            Arrays.fill(dis[i], -1);
        }
        Queue<Integer> que = new LinkedList<Integer>();
        que.add(S);
        dis[S][0] = 0;
        while (!que.isEmpty()) {
            int v = que.poll();
            for (int i = 0; i < N; i++) {
                if (edge[v][i] == 1) {
                    if (dis[i][0] == -1) {
                        dis[i][0] = dis[v][0] + 1;
                        que.add(i);
                    }
                    for (int r = 1; r <= R; r++) {
                        if (dis[v][r] != -1 && dis[v][r] + 1 < dis[i][r]) {
                            dis[i][r] = dis[v][r] + 1;
                            que.add(i);
                        }
                    }
                }
            }
        }
        int ans = -1;
        for (int r = 0; r <= R; r++) {
            if (dis[T][r] != -1) {
                if (ans == -1) {
                    ans = dis[T][r];
                } else {
                    ans = Math.min(ans, dis[T][r]);
                }
            }
        }
        if (ans == -1) {
            System.out.println(-1);
            return;
        }
        int[][] dis2 = new int[N][R + 1];
        for (int i = 0; i < N; i++) {
            Arrays.fill(dis2[i], -1);
        }
        que.clear();
        que.add(S);
        dis2[S][0] = 0;
        while (!que.isEmpty()) {
            int v = que.poll();
            for (int i = 0; i < N; i++) {
                if (edge2[v][i] == 1) {
                    if (dis2[i][0] == -1) {
                        dis2[i][0] = dis2[v][0] + 1;
                        que.add(i);
                    }
                    for (int r = 1; r <= R; r++) {
                        if (dis2[v][r] != -1 && dis2[v][r] + 1 < dis2[i][r]) {
                            dis2[i][r] = dis2[v][r] + 1;
                            que.add(i);
                        }
                    }
                }
            }
        }
        for (int r = 0; r <= R; r++) {
            if (dis[S][r] != -1 && dis2[T][R - r] != -1) {
                if (ans == -1) {
                    ans = dis[S][r] + dis2[T][R - r];
                } else {
                    ans = Math.min(ans, dis[S][r] + dis2[T][R - r]);
                }
            }
        }
        System.out.println(ans);
    }
}
