#include<bits/stdc++.h>
using namespace std;
int main() {
  int t; cin >> t; while (t--) {
    int n; cin >> n; string s; cin >> s;
    for (int i = 0, j = 0; i < n; i++, j += 2) {
      if (j + 1 < s.size() && s[j] == s[j + 1]) cout << s[j];
      else cout << (!stoi(s.substr(j, 2)));
    } cout << endl;
  }
}
