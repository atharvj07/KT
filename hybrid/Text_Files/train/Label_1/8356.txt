#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll maxN = 100001;
vector<ll> g[maxN];
unordered_map<ll,ll>deg;
int main(){
    ll n,k;cin>>n>>k;
    while(k--){
        ll x,y;cin>>x>>y;
        g[x].push_back(y);
        g[y].push_back(x);
        deg[x]++;
        deg[y]++;
    }
    vector<ll> ans;
    for(int i=1;i<=n;++i){
        if(deg[i]%2) ans.push_back(i);
    }
    cout<<ans.size()/2<<'\n';
    return 0;
}
