t = int(input())
for _ in range(t):
    a, b, c, d = map(int, input().split())
    x, y, x1, y1, x2, y2 = map(int, input().split())
    
    possible = True
    if x - a + b < x1 or x - a + b > x2:
        possible = False
    if y - c + d < y1 or y - c + d > y2:
        possible = False
    if x1 == x2 and (a > 0 or b > 0):
        possible = False
    if y1 == y2 and (c > 0 or d > 0):
        possible = False
    
    print("YES" if possible else "NO")
