import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        TreeSet<Long> set = new TreeSet<>();
        for (int i = 0; i < n; i++) {
            set.add(in.nextLong());
        }
        Iterator<Long> iter = set.iterator();
        long prev = iter.next();
        boolean ok = true;
        while (iter.hasNext()) {
            long curr = iter.next();
            if (ok) {
                ok = (prev * 3 == curr);
            } else {
                ok = (curr % prev == 0 && curr / prev != 3);
            }
            prev = curr;
        }
        if (!ok) {
            System.out.println("NO");
            return;
        }
        iter = set.iterator();
        while (iter.hasNext()) {
            System.out.print(iter.next() + " ");
        }
        System.out.println();
    }
}
