n, m = map(int, input().split())
a = list(map(int, input().split()))
g = [[] for i in range(n)]
for i in range(m):
    x, y = map(int, input().split())
    x, y = x - 1, y - 1
    g[x].append(y)
    g[y].append(x)

ans = 0
vis = [0] * n
for i in range(n):
    if vis[i]: continue
    vis[i] = 1
    
    q = [i]
    odd = a[i] % 2
    while q:
        x = q.pop(0)
        for j in g[x]:
            if vis[j]: continue
            vis[j] = 1
            if a[j] % 2 != odd: continue
            q.append(j)
    ans += 1
print(ans)
