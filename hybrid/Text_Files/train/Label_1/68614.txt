n = int(input())
s = input()
cnt = s.count('1') - s.count('0')
ans = 0
pref = [0] * n
if s[0] == '1': pref[0] = 1
for i in range(1, n):
	pref[i] = pref[i - 1]
	if s[i] == '1': pref[i] += 1
for i in range(n):
	l = i
	r = n - 1
	pos = -1
	while l <= r:
		mid = (l + r) // 2
		cur = pref[mid]
		if i > 0: cur -= pref[i - 1]
		if cur == cnt:
			pos = mid
			l = mid + 1
		else: r = mid - 1
	if pos != -1: ans = max(ans, pos - i + 1)
print(ans)
