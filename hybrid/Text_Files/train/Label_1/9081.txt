#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string a, b, c;
    cin >> a >> b >> c;
    int cnta = 0, cntb = 0;
    for (int i = 0; i < a.size(); i++) {
      if (a[i] != c[i]) {
        cnta++;
      }
      if (b[i] != c[i]) {
        cntb++;
      }
    }
    if (cnta % 2 == cntb % 2) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
  return 0;
}
