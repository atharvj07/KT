#include <bits/stdc++.h>
using namespace std;

#define f first
#define s second

int main() {
    int n, m, k, q;
    cin >> n >> m >> k >> q;
    vector<pair<pair<int, int>, int>> v(q);
    for (int i = 0; i < q; i++) {
        int x, y, t;
        cin >> x >> y >> t;
        v[i] = {{x, y}, t};
    }
    sort(v.begin(), v.end(),
         [](const pair<pair<int, int>, int> &a, const pair<pair<int, int>, int> &b) {
             return a.s < b.s;
         });
    int ans = -1;
    for (int i = 0; i < q; i++) {
        int x = v[i].f.f, y = v[i].f.s, t = v[i].s;
        bool ok = true;
        for (int j = x; j < x + k && ok; j++) {
            for (int h = y; h < y + k && ok; h++) {
                ok &= (j >= 1 && j <= n && h >= 1 && h <= m);
            }
        }
        if (ok) {
            ans = t;
            break;
        }
    }
    cout << ans << endl;
}
