import sys, math
n = int(input())
a = [int(x) for x in input().split()]
assert(len(a) == n)
aa = 0
for i in range(2):
    aa += a[min(a)]
    a.pop(min(a))
bb = min(a)
res = 0
for ip in range(n-3):
    for iq in range(ip+1,n-2):
        for ir in range(iq+1,n-1):
            t = a[ip] + a[iq] + a[ir]
            if t > bb + aa: res += 1
res /= math.comb(n,3)
sys.stdout.write('%.12f' % res)
