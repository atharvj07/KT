#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long n;
    cin>>n;
    const long long mod=1e9+7;
    long long x=1,ans=0;
    for(long long i=1;i<=n;i++)
    {
        if(i%10==0||i%10==6||i%10==8)continue;
        ans=ans+x;
        if(ans>=mod)ans=ans-mod;
        x=x*10%mod;
    }
    cout<<ans<<endl;
    return 0;
}
