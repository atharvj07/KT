import java.util.*;
import java.io.*;

public class Main {
	static int N, M, D[];
	static List<Integer> adj[];
	static boolean visited[];

	public static void main(String[] args) throws Exception {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		N = Integer.parseInt(st.nextToken());
		M = Integer.parseInt(st.nextToken());
		adj = new List[N + 1];
		for (int i = 0; i <= N; i++) adj[i] = new ArrayList<>();
		for (int i = 0; i < M; i++) {
			st = new StringTokenizer(br.readLine());
			int a = Integer.parseInt(st.nextToken());
			int b = Integer.parseInt(st.nextToken());
			adj[a].add(b);
			adj[b].add(a);
		}
		D = new int[N + 1];
		Arrays.fill(D, -1);
		System.out.println(check() ? "Yes" : "No");
	}

	private static boolean check() {
		boolean ok = true;
		for (int i = 1; i <= N; i++) {
			if (D[i] == -1) ok &= dfs(i, i, 0);
		}
		return ok;
	}

	private static boolean dfs(int node, int start, int depth) {
		D[node] = depth;
		for (int next : adj[node]) {
			if (D[next] == -1) {
				if (!dfs(next, start, depth + 1)) return false;
			} else {
				if (Math.abs(D[next] - D[node]) != 1 && (D[next] - D[node] + N) % (N - 1) != 1)
					return false;
			}
		}
		return (D[node] - D[start] + N) % (N - 1) == 0 ? true : false;
	}
}
