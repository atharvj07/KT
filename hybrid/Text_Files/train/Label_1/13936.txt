#include <bits/stdc++.h>
#define rep(i,a,N) for (int i=a;i<N;++i)
#define rrep(i,N,a) for (int i=N;i>=a;--i)
using namespace std;
typedef long long ll;
typedef pair<ll, ll> p;

int main() {
    int n,m,q;
    cin>>n>>m>>q;
    vector<int> p(n+m+1,0);
    map<int,int> mp,mp2;
    rep(i,0,n) {
        int v;
        cin>>v;
        p[v]=i+1;
        mp[v]++;
    }
    rep(i,0,m) {
        int v;
        cin>>v;
        mp2[v]++;
    }
    rep(i,0,q) {
        int l,r;
        cin>>l>>r;
        bool ok=true;
        if (mp2[0]>0) ok=true;
        else {
            rep(j,1,n+1) {
                if (!(mp2[j]&&mp[j])) ok=false;
            }
        }
        if (!ok) cout<<0;
        else {
            cout<<1;
        }
    }
    cout<<endl;
    return 0;
}
