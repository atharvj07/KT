import sys
n = int(input())
a = [list(map(int, input().split())) for _ in range(n)]
for i in a:
    i.sort()
balls = [False for _ in range(100)]
for i in a:
    for j in i:
        balls[j - 1] = True
    if not any(balls):
        print("YES")
    else:
        balls = [False for _ in range(100)]
        print("NO")
