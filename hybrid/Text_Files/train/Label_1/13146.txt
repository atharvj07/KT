#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, L;
    cin >> n >> L;
    vector <long long> c(n);
    for (int i = 0; i < n; i++)
        cin >> c[i];

    long long sum = 0, ans = 1e18;
    for (int i = 0; i < n; i++) {
        int k = 0;
        while (sum + (1LL << k) - 1 <= L) {
            sum += (1LL << k) - 1;
            k++;
        }
        ans = min(ans, c[i] * k);
    }
    cout << ans << endl;
    return 0;
}
