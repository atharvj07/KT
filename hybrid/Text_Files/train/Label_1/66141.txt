p = []
n, k = input().split()
n = int(n)
k = int(k)
a = list(map(int, input().split()))
a = sorted(a)
for i in range(n):
    g = 1
    for j in range(n):
        if i == j:
            continue
        g = g * (a[i] + 1 - a[j])
        g = g % k
    p.append(g)
ans = []
for i in range(n + 1):
    g = 1
    for j in range(n):
        g = g * (i + 1 - a[j])
        g = g % k
    if g % k != 0:
        ans.append(i)
print(len(ans))
for i in ans:
    print(i, end=' ')
