import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        if (k == 0) {
            for (int i = 0; i < n; i++) {
                a[i] = i + 1;
            }
        } else if (k == 1) {
            if (n < 2) {
                System.out.println(-1);
                return;
            }
            for (int i = 0; i < n; i++) {
                if (i % 2 == 0) {
                    a[i] = i + 2;
                } else {
                    a[i] = i + 1;
                }
            }
        } else {
            if (n < 3) {
                System.out.println(-1);
                return;
            }
            int j = 2;
            for (int i = 0; i < n; i++) {
                if (i % 3 == 0) {
                    a[i] = j;
                    j++;
                } else {
                    a[i] = i + 1;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
        System.out.println();
    }
}
