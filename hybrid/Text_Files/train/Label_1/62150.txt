for _ in range(int(input())):
    n, k, z = map(int, input().split())
    a = list(map(int, input().split()))
    dp = [[[-1] * (z + 1) for _ in range(k + 1)] for _ in range(n + 1)]

    def solve(idx, moves, moves_left):
        if idx == n:
            return 0

        if dp[idx][moves][moves_left] != -1:
            return dp[idx][moves][moves_left]

        ans = 0
        if moves > 0:
            ans = max(ans, solve(idx + 1, moves - 1, moves_left) + a[idx])

        if moves_left > 0:
            if idx > 1:
                ans = max(ans, solve(idx - 1, moves - 1, moves_left - 1) + a[idx - 1])

        dp[idx][moves][moves_left] = ans
        return ans

    print(solve(1, k, z))
