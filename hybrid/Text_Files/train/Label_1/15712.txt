#include <bits/stdc++.h>
using namespace std;
int n,m,deg[155];
vector<int>poly[155];
void solve(vector<int>&a,vector<int>&b)
{
    if(b.empty())
    {
        for(int i:a) cout<<i<<" ";
        return;
    }
    int l1=a.size(),l2=b.size();
    reverse(a.begin(),a.end());
    for(int i=l1;i<l2;i++) a.push_back(0);
    for(int i=0;i<min(l1,l2);i++) b[i]=-b[i];
    reverse(a.begin(),a.end());
    reverse(b.begin(),b.end());
    while(l2<=l1)
    {
        for(int i=0;i<l2;i++)
        {
            int now=l1-l2+i;
            a[now]+=b[i],deg[now]=max(deg[now],deg[l1-l2+i]+deg[i]);
        }
        while(a.size()>1&&a.back()==0) a.pop_back();
        l1=a.size(),l2=b.size();
    }
    solve(b,a);
}
int main()
{
    cin>>n;
    if(n==1)
    {
        cout<<"1"<<endl<<"0 1"<<endl<<"0"<<endl<<"1"<<endl;
        return 0;
    }
    if(n==2)
    {
        cout<<"2"<<endl<<"-1 0 1"<<endl<<"1"<<endl<<"0 1"<<endl;
        return 0;
    }
    m=n-2;
    vector<int>a(m+1),b(m);
    for(int i=1;i<=m;i++)
    {
        a[i]=rand()%2;
        b[i-1]=rand()%2;
        if(i&1) a[i]*=-1;
        if(i&1) b[i-1]*=-1;
    }
    solve(a,b);
    return 0;
}
