import sys

def solve():
    n = int(input())
    a = []
    b = []
    for i in range(n):
        x, y = map(int, input().split())
        a.append(x)
        b.append(y)
    if n % 2 == 1:
        print(-1)
        return
    for i in range(n):
        if a[i] % 2 == b[i] % 2:
            print(-1)
            return
    for i in range(n):
        if i % 2 == 0:
            print(1, 2)
        else:
            print(2, 1)

solve()
