#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    unordered_map<string, int> cnt;
    vector<string> s(n);
    for (int i = 0; i < n; i++) {
      cin >> s[i];
      cnt[s[i]]++;
    }
    sort(s.begin(), s.end());
    int mx = 0;
    for (auto &p : cnt) mx = max(mx, p.second);
    if (mx > (n + 1) / 2 || s[0] != s[n - 1])
      cout << "NO" << endl;
    else
      cout << "YES" << endl;
  }
  return 0;
}
