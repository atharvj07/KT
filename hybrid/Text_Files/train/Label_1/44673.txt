import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long sum = 0;
        for (int i = 0; i < n; i++) {
            long a = sc.nextLong();
            sum += a;
        }
        System.out.println((sum % 2 == 1) ? "First" : "Second");
    }
}
