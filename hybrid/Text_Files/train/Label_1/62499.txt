from collections import defaultdict

for _ in range(int(input())):
    s, t = input(), input()
    if len(t) > 2*len(s) - 1:
        print('NO')
        continue
    d = defaultdict(int)
    for i in range(len(s)):
        d[s[i]] += 1
    for c in t:
        d[c] -= 1
        if d[c] < 0:
            break
    else:
        print('YES')
        continue
    print('NO')
