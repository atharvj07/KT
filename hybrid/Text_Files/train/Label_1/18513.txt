#include <iostream>
#include <vector>
#include <algorithm>

#define loop(i, a, b) for (int i = a; i < b; i++)
using namespace std;

typedef long long ll;

struct node {
    int parent;
    ll power;
    vector<int> friends;
    vector<int> _friends;

    node() {
        parent = 0;
        power = 0;
    }
};

const int MX = 2e5 + 5;
node arr[MX];
ll ans = 0;

int get(int x) {
    if (arr[x].parent == x) return x;
    else return arr[x].parent = get(arr[x].parent);
}

void _union(int a, int b) {
    a = get(a);
    b = get(b);
    if (a != b) {
        if (arr[a].power < arr[b].power) {
            swap(a, b);
        }
        arr[b].parent = a;
        ans += arr[b].power;
    }
}

int main() {
    int n, m, q;
    cin >> n >> m;
    loop(i, 0, m) {
        int u, v;
        cin >> u >> v;
        u--;
        v--;
        arr[u].friends.push_back(v);
        arr[u]._friends.push_back(v);
        arr[v].friends.push_back(u);
        arr[v]._friends.push_back(u);
    }
    loop(i, 0, n) {
        arr[i].parent = i;
        arr[i].power = i + 1;
    }

    cin >> q;
    while (q--) {
        int type;
        cin >> type;
        if (type == 1) {
            int u, v;
            cin >> u >> v;
            u--;
            v--;
            arr[u].friends.push_back(v);
            arr[u]._friends.push_back(v);
            arr[v].friends.push_back(u);
            arr[v]._friends.push_back(u);
        } else if (type == 2) {
            int u, v;
            cin >> u >> v;
            u--;
            v--;
            auto it = find(arr[u].friends.begin(), arr[u].friends.end(), v);
            arr[u].friends.erase(it);
            it = find(arr[v].friends.begin(), arr[v].friends.end(), u);
            arr[v].friends.erase(it);
        } else {
            ans = 0;
            loop(i, 0, n) {
                if (arr[i].friends.size()) {
                    int f = 1;
                    for (auto x : arr[i].friends) {
                        if (get(i) == get(x)) {
                            f = 0;
                            break;
                        }
                        if (arr[i].power < arr[x].power) {
                            f = 0;
                            break;
                        }
                    }
                    if (f) {
                        _union(i, i);
                    }
                }
            }
            ll res = n - ans;
            cout << res << endl;
            loop(i, 0, n) {
                arr[i].parent = i;
                arr[i].friends = arr[i]._friends;
            }
        }
    }
}
