n,k=map(int,input().split())
if k==0:
	print(*range(1,n+1))
elif k==n:
	if n&1:
		print(*[2*i for i in range(1,n+1,2)],end=' ')
		print(*[2*i+1 for i in range(1,n+1,2)],end=' ')
	else:
		print(*range(2,n+1,2),end=' ')
		print(*range(1,n+1,2),end=' ')
elif n-k&1:
	print(*[2*i for i in range(1,k+1,2)],end=' ')
	print(2*k+1,' ',end='')
	print(*[2*i+1 for i in range(k+1,n+1,2)],end=' ')
	print(*[2*i for i in range(k+2,n+1,2)],end=' ')
else:
	print(*[2*i for i in range(1,k+1,2)],end=' ')
	print(2*k,' ',end='')
	print(*[2*i+1 for i in range(k+1,n+1,2)],end=' ')
	print(*[2*i for i in range(k+1,n+1,2)],end=' ')
