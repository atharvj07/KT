#include <bits/stdc++.h>

using namespace std;

const int dx[] = {-1, 0, 1, 0};
const int dy[] = {0, 1, 0, -1};

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> g(n);
  for (int i = 0; i < n; i++) cin >> g[i];
  for (int x = 0; x < n; x++) {
    for (int y = 0; y < m; y++) {
      for (int dir = 0; dir < 4; dir++) {
        int nx = x + dx[dir];
        int ny = y + dy[dir];
        if (nx < 0 || nx >= n || ny < 0 || ny >= m || g[x][y] != g[nx][ny]) continue;
        int cnt = 0;
        int s = 0;
        for (int k = 0; k < 4; k++) {
          for (int i = 0; i <= cnt; i++) {
            int tx = x + i * dx[k];
            int ty = y + i * dy[k];
            if (tx < 0 || tx >= n || ty < 0 || ty >= m || g[tx][ty] != g[s][s]) {
              cnt = 0;
              break;
            }
            cnt++;
          }
          if (cnt >= 3) {
            cout << "Yes" << endl;
            return 0;
          }
        }
      }
    }
  }
  cout << "No" << endl;
  return 0;
}
