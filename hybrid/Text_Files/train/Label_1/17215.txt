#include<bits/stdc++.h>
using namespace std;
const long long MOD=1e9+7;
int main(){
  long long n,m;
  cin>>n>>m;
  long long dp[n+1][2];
  memset(dp,0,sizeof(dp));
  dp[0][0]=1;
  dp[0][1]=1;
  for(int j=0;j<=1;j++){
    for(int i=1;i<=n;i++){
      dp[i][j]=(dp[i-1][0]+dp[i-1][1])%MOD;
      if(i>1){
        dp[i][j]=(dp[i][j]-dp[i-2][j]+MOD)%MOD;
      }
    }
  }
  long long ans=dp[n][0];
  ans=(ans*m)%MOD;
  ans=(ans*m)%MOD;
  cout<<ans<<endl;
}
