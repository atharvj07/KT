#include<bits/stdc++.h>
using namespace std;
#define vi(x) vector<x>
#define pb push_back
const int mod = 1e9 + 7;
const int N = 2e5 + 5;

int n;
vi(int) price;
vi(string) s;
int dp[1 << 3];
int func(int mask) {
	if (mask == 7)return 0;
	int &ret = dp[mask];
	if (ret != -1)return ret;
	ret = 1e9;
	for (int i = 0; i < n; i++) {
		int nmask = mask;
		for (char &ch : s[i])if (ch == 'A')nmask |= 1;
		else if (ch == 'B')nmask |= 2;
		else nmask |= 4;
		ret = min(ret, func(nmask) + price[i]);
	}
	return ret;
}
int main() {
	cin >> n;
	price.resize(n); s.resize(n);
	for (int i = 0; i < n; i++) {
		cin >> price[i] >> s[i];
	}
	memset(dp, -1, sizeof dp);
	int ans = func(0);
	if (ans == 1e9)ans = -1;
	cout << ans << endl;
}
