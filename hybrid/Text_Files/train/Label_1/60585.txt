import math
n,m=map(int,input().split())
p=[list(map(int,input().split()))for i in range(n)]
def dot(a,b):return a[0]*b[0]+a[1]*b[1]
def cross(a,b):return a[0]*b[1]-a[1]*b[0]
def dis(p):return math.sqrt(dot(p,p))
for i in range(m):
    x,y,x1,y1=map(int,input().split())
    a=[x1-x,y1-y]
    b=[p[(i+1)%n][0]-p[i][0],p[(i+1)%n][1]-p[i][1]]
    d=cross(a,b)/dis(a)*dis(b)
    if (cross(a,[x-x1,y-y1])<=0 and cross(b,[x-x1,y-y1])>=0 or
        cross(a,[x-x1,y-y1])>=0 and cross(b,[x-x1,y-y1])<=0)and \
        d>=0 and d<=dis(b):
        print(math.sqrt(max(0,dis([x-p[i][0],y-p[i][1]])**2-d**2)))
    else:print(0)
