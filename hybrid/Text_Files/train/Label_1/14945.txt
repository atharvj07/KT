#include <iostream>
#include <vector>

using namespace std;
typedef long long ll;
const int MAXN = 1e5 + 10;
const int MAXM = 1e5 + 10;

ll arr[MAXN];
ll op[MAXM][3];
ll qr[MAXM][2];

ll tree[MAXN];

void add(int idx, ll val){
    while(idx < MAXN){
        tree[idx] += val;
        idx += (idx & -idx);
    }
}

ll get(int idx){
    ll ans = 0;
    while(idx > 0){
        ans += tree[idx];
        idx -= (idx & -idx);
    }
    return ans;
}

int main(){
    ll n, m, k;
    cin >> n >> m >> k;
    for(int i = 1; i <= n; i++){
        cin >> arr[i];
    }
    for(int i = 0; i < m; i++){
        cin >> op[i][0] >> op[i][1] >> op[i][2];
    }
    for(int i = 0; i < k; i++){
        cin >> qr[i][0] >> qr[i][1];
        add(qr[i][0], op[qr[i][0] - 1][2]);
        if(qr[i][1] < m){
            add(qr[i][1] + 1, -op[qr[i][1]][2]);
        }
    }
    for(int i = 1; i <= n; i++){
        cout << (arr[i] + get(i)) << " ";
    }
    cout << endl;
    return 0;
}
