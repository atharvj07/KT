#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
struct gadget{
    int t,c;
};
vector <ll> a,b;
vector <gadget> g;
ll burles,n,m,k;
ll days[200005];
bool vis[200005];
bool cmp(gadget x,gadget y)
{
    return x.c<y.c;
}
int main()
{
    cin>>n>>m>>k>>burles;
    a.resize(n);
    b.resize(n);
    for(int i=0;i<n;i++)
        cin>>a[i];
    for(int i=0;i<n;i++)
        cin>>b[i];
    for(int i=0;i<m;i++)
    {
        gadget x;
        cin>>x.t>>x.c;
        g.push_back(x);
    }
    sort(g.begin(),g.end(),cmp);
    ll l=0,r=n-1,ans,day=0,f=0;
    while(l<=r)
    {
        ll mid=(l+r)/2;
        for(int i=0;i<=m;i++)
            days[i]=0;
        for(int i=0;i<=m;i++)
            vis[i]=false;
        ll bl=burles;
        for(int i=1;i<=m;i++)
        {
            if(g[i-1].t==1)
            {
                if(bl>=a[mid]*g[i-1].c)
                {
                    if(day==0)
                        day=mid+1;
                    vis[i]=true;
                    days[i]=mid+1;
                    bl-=a[mid]*g[i-1].c;
                }
            }
            else
            {
                if(bl>=b[mid]*g[i-1].c)
                {
                    if(day==0)
                        day=mid+1;
                    vis[i]=true;
                    days[i]=mid+1;
                    bl-=b[mid]*g[i-1].c;
                }
            }
            if(bl<=0)
                break;
        }
        ll cnt=0;
        for(int i=1;i<=m;i++)
            cnt+=vis[i];
        if(cnt>=k&&f==0)
        {
            ans=mid+1;
            f=1;
        }
        if(cnt>=k)
            r=mid-1;
        else
            l=mid+1;
    }
    if(f==0)
    {
        cout<<-1<<endl;
        return 0;
    }
    cout<<ans<<endl;
    ll cnt=0;
    for(int i=1;i<=m;i++)
    {
        if(vis[i])
        {
            cout<<i<<' '<<days[i]<<endl;
            cnt++;
        }
        if(cnt==k)
            break;
    }
    return 0;
}
