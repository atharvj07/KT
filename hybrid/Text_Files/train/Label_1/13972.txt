#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, k;
    cin >> n >> k;
    long long a[k];
    for (long long i = 0; i < k; i++) cin >> a[i];
    long long ans = LLONG_MIN, ind = -1;
    for (long long i = 0; i < k; i++) {
        if (n / a[i] > ans) {
            ind = i;
            ans = n / a[i];
        }
    }
    cout << ind + 1 << " " << ans << endl;
}
