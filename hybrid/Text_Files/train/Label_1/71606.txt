w, h = map(int, input().split())
grid = [input() for _ in range(h)]

t_grid = [["." for _ in range(2 * w)] for _ in range(2 * h)]

for i in range(h):
    for j in range(w):
        t_grid[2 * i][2 * j] = grid[i][j]
        t_grid[2 * i][2 * j + 1] = grid[i][j]
        t_grid[2 * i + 1][2 * j] = grid[i][j]
        t_grid[2 * i + 1][2 * j + 1] = grid[i][j]

for t_r in t_grid:
    print("".join(t_r))
