import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        long f = Long.parseLong(st.nextToken()), T = Long.parseLong(st.nextToken()), t0 = Long.parseLong(st.nextToken());
        st = new StringTokenizer(in.readLine());
        long a1 = Long.parseLong(st.nextToken()), t1 = Long.parseLong(st.nextToken()), p1 = Long.parseLong(st.nextToken());
        st = new StringTokenizer(in.readLine());
        long a2 = Long.parseLong(st.nextToken()), t2 = Long.parseLong(st.nextToken()), p2 = Long.parseLong(st.nextToken());

        long ans = (f + t0 - 1) / t0 * p1;
        long b1 = (t1 - t0) / (a1 * t0 - t1);
        long b2 = (t2 - t0) / (a2 * t0 - t2);
        for (long l = 0, r = f; l <= r;) {
            long mid = (l + r) / 2, cur = (f - mid) * p1;
            if (mid % a1 != 0)
                cur += p1 * (1 + (a1 - mid % a1) / a1);
            if (mid % a2 != 0)
                cur += p2 * (1 + (a2 - mid % a2) / a2);
            long rem = b1 * mid + b2 * (f - mid);
            long cost = (rem + t0 - 1) / t0 * p1;
            if (cost < cur) {
                r = mid - 1;
                ans = Math.min(ans, cost);
            } else {
                l = mid + 1;
            }
        }
        System.out.println(ans != (f + t0 - 1) / t0 * p1 ? ans : -1);
    }
}
