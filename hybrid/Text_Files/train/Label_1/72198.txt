from collections import defaultdict, deque

n = int(input())
w = [int(x) for x in input().split()]
adj = defaultdict(list)
for _ in range(n - 1):
    u, v, c = map(int, input().split())
    adj[u].append((v, c))
    adj[v].append((u, c))

vis = [False] * (n + 1)
par = [-1] * (n + 1)
q = deque([1])
while q:
    v = q.popleft()
    vis[v] = True
    for u, c in adj[v]:
        if not vis[u]:
            q.append(u)
            par[u] = v

mx_gas = [0] * (n + 1)
mx_gas[1] = w[0]
for v in range(2, n + 1):
    mx_gas[v] = mx_gas[par[v]] + w[v - 1] - adj[v][0][1]

ans = 0
for v in range(1, n + 1):
    u = par[v]
    if u != -1:
        c = adj[u][0][1]
        mx_gas[v] = mx_gas[u] + w[v - 1] - c
        ans = max(ans, mx_gas[v])
    else:
        ans = max(ans, mx_gas[v])

print(ans)
