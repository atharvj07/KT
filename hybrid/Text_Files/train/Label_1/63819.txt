import sys
input = sys.stdin.readline
n, m, k = map(int,input().split())
adj = [[]for _ in range(1005)]
vis = [[-1]*1005 for _ in range(1005)]
pos = [-1,-1]
for i in range(n):
    s = input()
    for j in range(m):
        if s[j] == 'X':
            pos = [i,j]
        elif s[j] == '.':
            adj[i].append([j,0])
        else:
            adj[i].append([j,-1])
ans = 'X'
def dfs(x,y,d,cnt):
    global ans
    if x == pos[0] and y == pos[1] and cnt == k and d < ans:
        ans = d
        return
    if cnt == k or x < 0 or y < 0 or x >= n or y >= m or adj[x][y][1] == -1 or vis[x][y] == d:
        return
    vis[x][y] = d
    dfs(x+1,y,d+'D',cnt+1)
    dfs(x-1,y,d+'U',cnt+1)
    dfs(x,y+1,d+'R',cnt+1)
    dfs(x,y-1,d+'L',cnt+1)
dfs(pos[0],pos[1],'',0)
print(ans if ans != 'X' else "IMPOSSIBLE")
