#include<bits/stdc++.h>
#define vi(x) vector<x>
#define pb push_back
using namespace std;
const int mod = 1e9 + 7;
const char nl = '\n';
typedef long long ll;
int main(){
	int n,p,i,ans=0;
	string s;
	cin>>n>>p;
	cin>>s;
	s=s+s;
	for(i=0;i<n;++i){
		if(s[i]==s[i+p])continue;
		else{
			if(s[i+1]==s[i+p]){
				++i;++ans;
			}
			else if(s[i+p-1]==s[i+p]){
				++i;
				ans+=2;
			}
			else{
				ans+=2;	
			}
		}
	}
	cout<<ans<<nl;
	return 0;
}
