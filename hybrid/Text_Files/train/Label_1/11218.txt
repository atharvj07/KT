#include <iostream>
#include <string>
#include <vector>

int main() {
  std::vector<std::string> days = {"monday", "tuesday", "wednesday", "thursday", "friday", "saturday", "sunday"};
  std::string first, second;
  std::cin >> first >> second;
  int first_idx = -1, second_idx = -1;
  for (int i = 0; i < 7; ++i) {
    if (days[i] == first) { first_idx = i; }
    if (days[i] == second) { second_idx = i; }
  }
  if (first_idx == -1 || second_idx == -1) {
    std::cout << "NO" << std::endl;
    return 0;
  }
  int diff = (second_idx - first_idx + 7) % 7;
  if (diff == 1 || diff == 3) {
    std::cout << "YES" << std::endl;
  } else {
    std::cout << "NO" << std::endl;
  }
  return 0;
}
