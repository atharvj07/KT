#include <bits/stdc++.h>
using namespace std;

int main() {
  bool ab = false, ba = false;
  string s;
  cin >> s;
  for (char c : s) {
    if (c == 'A') {
      if (ba) {
        cout << "YES" << endl;
        return 0;
      }
      ab = true;
    }
    if (c == 'B') {
      if (ab) {
        cout << "YES" << endl;
        return 0;
      }
      ba = true;
    }
  }
  cout << "NO" << endl;
  return 0;
}
