#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const int N = 2e5 + 10;

int n;
int a[N];
vector <int> ans;

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++)
    cin >> a[i];
  
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    cnt += (a[i] > 0 ? 1 : -1);
    if (cnt < 0) {
      cout << -1 << '\n';
      return 0;
    }
    if (cnt == 0) {
      ans.push_back(i);
    }
  }
  if (cnt > 0) {
    cout << -1 << '\n';
    return 0;
  }

  int sz = ans.size();
  cout << sz << '\n';
  for (int i = 0; i < sz; i++) {
    cout << ans[i] - (i == 0 ? 0 : ans[i - 1]) << ' ';
  }
  cout << '\n';
}
