#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n, x, sum = 0;
    cin >> n;
    vector<long long> a(n);
    for (long long i = 0; i < n; i++) {
        cin >> a[i];
        sum += a[i];
    }
    sort(a.begin(), a.end());
    for (long long i = 0; i < n - 1; i++) {
        sum -= a[i];
    }
    cout << sum << endl;
    return 0;
}
