#include <iostream>
#include <cstring>
using namespace std;

char grid[55][55];
int n, m, has_red, has_white;

bool check() {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (grid[i][j] == '.') {
                has_red = has_white = 0;
                if (i > 0 && grid[i - 1][j] == 'R') has_red = 1;
                if (i + 1 < n && grid[i + 1][j] == 'R') has_red = 1;
                if (j > 0 && grid[i][j - 1] == 'R') has_red = 1;
                if (j + 1 < m && grid[i][j + 1] == 'R') has_red = 1;
                if (i > 0 && grid[i - 1][j] == 'W') has_white = 1;
                if (i + 1 < n && grid[i + 1][j] == 'W') has_white = 1;
                if (j > 0 && grid[i][j - 1] == 'W') has_white = 1;
                if (j + 1 < m && grid[i][j + 1] == 'W') has_white = 1;
                if (has_red == has_white) return false;
            }
        }
    }
    return true;
}

void dfs(int x, int y) {
    if (x < 0 || x > n - 1 || y < 0 || y > m - 1 || grid[x][y] != '.') return;
    if (has_red == 0) grid[x][y] = 'R';
    else grid[x][y] = 'W';
    dfs(x + 1, y);
    dfs(x, y + 1);
    dfs(x - 1, y);
    dfs(x, y - 1);
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> m;
        for (int i = 0; i < n; i++) cin >> grid[i];
        if (!check()) cout << "NO" << endl;
        else {
            dfs(0, 0);
            cout << "YES" << endl;
            for (int i = 0; i < n; i++) cout << grid[i] << endl;
        }
    }
    return 0;
}
