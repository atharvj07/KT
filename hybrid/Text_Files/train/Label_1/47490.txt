import java.util.Scanner;
import java.util.TreeSet;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt(), n = in.nextInt();
        TreeSet<Long> kx = new TreeSet<>(), ky = new TreeSet<>(), mx = new TreeSet<>(), my = new TreeSet<>();
        for (int i = 0; i < k; i++) {
            kx.add(in.nextLong());
            ky.add(in.nextLong());
        }
        for (int i = 0; i < n; i++) {
            mx.add(in.nextLong());
            my.add(in.nextLong());
        }
        long res = 0;
        for (long x : kx) {
            res += mx.higher(x) == null ? 0 : 1;
            res += mx.lower(x) == null ? 0 : 1;
        }
        for (long y : ky) {
            res += my.higher(y) == null ? 0 : 1;
            res += my.lower(y) == null ? 0 : 1;
        }
        System.out.println(res);
        in.close();
    }
}
