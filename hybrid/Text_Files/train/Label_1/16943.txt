#include <bits/stdc++.h>
#include <vector>
using namespace std;
int main() {
    vector<int> a(24);
    for (int i = 0; i < 24; ++i) { cin >> a[i]; }
    int c[7] = {};
    for (int i = 0; i < 24; ++i) { c[a[i]]++; }
    int cnt = 0;
    for (int i = 1; i < 7; ++i) { if (c[i] % 4 == 0) { cnt++; } }
    if (cnt == 6) {
        cout << "YES";
    } else {
        cout << "NO";
    }
    return 0;
}
