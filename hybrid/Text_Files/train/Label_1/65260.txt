def next_power_of_2(x):
    p = 1
    while p < x:
        p <<= 1
    return p

def solve(x):
    operations = []
    while True:
        if x & (x + 1) == 0:  # If x is already a perfect longcat
            break
        next_p = next_power_of_2(x + 1)
        mask = next_p - 1
        diff = x ^ mask
        highest_diff_bit = len(bin(diff)) - 3
        if highest_diff_bit >= 0:
            operations.append(highest_diff_bit + 1)
            x ^= ((1 << (highest_diff_bit + 1)) - 1)
        if x & (x + 1) == 0:
            break
        x += 1
        operations.append(0)  # Placeholder for addition operations, not needed in output
    return operations

x = int(input().strip())
operations = solve(x)
print(len(operations) // 2 + len(operations) % 2)
for op in operations:
    if op != 0:
        print(op, end=" ")
