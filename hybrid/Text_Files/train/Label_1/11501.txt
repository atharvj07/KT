#include <iostream>
#include <vector>
using namespace std;

int main() {
    string s;
    cin >> s;
    int k;
    cin >> k;
    int n = s.size();
    if (n % k != 0) {
        cout << "NO";
        return 0;
    }
    int len = n / k;
    vector<bool> is_palindrome(len, true);
    for (int i = 0; i < len; i++) {
        for (int j = i; j < n; j += len) {
            if (s[j] != s[j + len - 1]) {
                is_palindrome[i] = false;
            }
        }
    }
    bool is_valid = true;
    for (int i = 0; i < len; i++) {
        if (!is_palindrome[i]) {
            is_valid = false;
            break;
        }
    }
    if (is_valid) {
        cout << "YES";
    } else {
        cout << "NO";
    }
    return 0;
}
