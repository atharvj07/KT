#include <iostream>

using namespace std;

const int MAXN = 100005;

int n;
int a[MAXN], b[MAXN];

int main() {
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  for (int i = 0; i < n; ++i) {
    cin >> b[i];
  }

  int ans = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = i + 1; j < n; ++j) {
      if ((a[i] | b[i]) != (a[j] | b[j]) && (a[i] | b[j]) != (a[j] | b[i])) {
        ++ans;
      }
    }
  }

  cout << ans << endl;

  return 0;
}
