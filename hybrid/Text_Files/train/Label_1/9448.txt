#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin >> t;
    while(t--) {
        long long a, b;
        cin >> a >> b;
        long long x = min(a, b);
        long long y = max(a, b);
        if(y - x > 1 || y == 0) {
            cout << x << endl;
        }
        else {
            cout << x + 1 << endl;
        }
    }
}
