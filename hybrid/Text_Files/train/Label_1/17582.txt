#include <vector>
#include <iostream>
using namespace std;

int main() {
    int q;
    cin >> q;
    vector<int> arr;
    vector<int> cnt;
    while (q--) {
        int t, x, y;
        cin >> t >> x;
        if (t == 1) {
            arr.push_back(x);
            cnt.push_back(1);
        } else {
            cin >> y;
            for (int i = 0; i < arr.size(); i++) {
                if (arr[i] == x) {
                    cnt[i] += y - x;
                    arr[i] = y;
                }
            }
        }
    }
    for (int i = 0; i < arr.size(); i++) {
        for (int j = 0; j < cnt[i]; j++) {
            cout << arr[i] << " ";
        }
    }
    return 0;
}
