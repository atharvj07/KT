#include <cstdio>
#include <algorithm>
using namespace std;

int main() {
    int t;
    scanf("%d", &t);
    while(t--) {
        int n, m, k;
        scanf("%d %d %d", &n, &m, &k);
        int mx, mn;
        mx = max(n, m);
        mn = min(n, m);
        int hc, vc;
        int rem;
        rem = mn * mx;
        hc = k;
        vc = mn * mx / 2 - k;
        if(vc > mx || vc < 0 || hc < 0 || (mn % 2 == 1 && k > 0))
            printf("NO\n");
        else
            printf("YES\n");
    }
    return 0;
}
