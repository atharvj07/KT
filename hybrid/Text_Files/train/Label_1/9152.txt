#include <bits/stdc++.h>

#define ll long long
#define maxn 300010
#define MOD 998244353

using namespace std;

ll ans;
int n, m, cnt_[maxn];
char g[maxn];

void dfs(int col, int sum, int cnt) {
    if (col == m + 1) {
        ans = (ans + (1LL * sum * cnt) % MOD) % MOD;
        return;
    }
    dfs(col + 1, sum, cnt);
    if (cnt_[col] != 0) {
        dfs(col + 1, sum + cnt_[col], cnt + cnt_[col]);
    }
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> g;
        for (int j = 1; j <= m; j++) {
            if (g[j] == 'o') {
                cnt_[j]++;
            }
        }
    }
    dfs(1, 0, 0);
    cout << ans << endl;
    return 0;
}
