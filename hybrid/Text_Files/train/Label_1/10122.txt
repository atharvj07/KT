#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> m(n);
  for (int i = 0; i < n; ++i) {
    cin >> m[i];
  }
  sort(m.begin(), m.end());
  int ans = 0;
  while (m[(n - 1) / 2] != m[0]) {
    m[(n - 1) / 2]--;
    m[0]++;
    ans++;
  }
  cout << ans << '\n';
  return 0;
}
