n=input()
msg=False
if int(n[-1])%2==0:
    for i in range(len(n)-1):
        if int(n[i])%2==0 and int(n[i+1])%2:
            n=n[:i]+n[i+1]+n[i]+n[i+2:]
            msg=True
            break
    if not msg:
        for i in range(len(n)-1):
            if int(n[i])%2 and int(n[i+1])%2==0 and int(n[i+1])>int(n[i]):
                n=n[:i]+n[i+1]+n[i]+n[i+2:]
                msg=True
                break
if not msg: print(-1)
else: print(n)
