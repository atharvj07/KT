#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int mod = 1e9 + 7;

int main() {
    int n;
    cin >> n;
    vector<int> p(n);
    for (int &x : p) {
        cin >> x;
    }
    int l = -1, r = -1, cnt = 0;
    for (int x : p) {
        if (x == -1) {
            if (l == -1)
                l = r = cnt;
            cnt++;
        }
    }
    int inv = 0;
    for (int i = l + 1; i < r; i++) {
        for (int j = i + 1; j <= r; j++) {
            if (p[i] > p[j]) inv++;
        }
    }
    int ans = 1;
    bool ok = true;
    for (int i = l + 1; i < r; i++) {
        ans = 1LL * ans * i % mod;
        if (p[i] < p[l] || p[i] > p[r]) ok = false;
    }
    if (l > 0 && r != n - 1 && !ok) ans = 0;
    for (int t = 0; t < inv; t++) {
        ans = 1LL * ans * 2 % mod;
    }
    cout << ans << '\n';
    return 0;
}
