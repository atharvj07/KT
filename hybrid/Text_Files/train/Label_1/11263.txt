#include <bits/stdc++.h>

using namespace std;

#define INF 1e18

typedef long long ll;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int n, m, k;
  cin >> n >> m >> k;
  vector<tuple<ll, ll, ll>> g(n);
  for (int i = 0; i < n; ++i) {
    ll a, b, c;
    cin >> a >> b >> c;
    g[i] = make_tuple(a, b, c);
  }
  vector<pair<int, int>> edges(m);
  for (int i = 0; i < m; ++i) {
    cin >> edges[i].first >> edges[i].second;
  }
  sort(edges.begin(), edges.end());
  vector<vector<pair<int, ll>>> graph(n);
  for (const auto& [u, v] : edges) {
    graph[u - 1].emplace_back(v - 1, 0);
  }
  vector<ll> dp(1 << n, -INF);
  dp[0] = 0;
  for (int i = 0; i < (1 << n); ++i) {
    if (dp[i] == -INF) continue;
    for (int j = 0; j < n; ++j) {
      if (i & (1 << j)) continue;
      const auto& [a, b, c] = g[j];
      if (dp[i | (1 << j)] < dp[i] + max(0LL, c - a) + b) {
        dp[i | (1 << j)] = dp[i] + max(0LL, c - a) + b;
        for (const auto& [to, w] : graph[j]) {
          dp[i | (1 << j)] = max(dp[i | (1 << j)], dp[i | (1 << j) | (1 << to)] + w);
        }
      }
    }
  }
  cout << (dp[(1 << n) - 1] != -INF ? dp[(1 << n) - 1] : -1) << endl;
  return 0;
}
