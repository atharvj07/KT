#include <iostream>
#include <algorithm>

using namespace std;
#define FOR(i,a,b) for(int i = (a); i < (b); i++)

int main() {
    int g[5][5];
    FOR(i,0,5){
        FOR(j,0,5){
            cin >> g[i][j];
        }
    }

    int ans = 0;
    bool order[5] = {0,};
    int arr[5];

    FOR(i,1,6){
        order[i-1] = 1;
        FOR(j,0,5){
            FOR(k,0,5){
                if(order[j] && order[k] && j != k){
                    if(j%2==0 && k%2==0)
                        ans += g[j/2][k/2];
                    if(j%2==1 && k%2==1)
                        ans += g[j/2+1][k/2+1];
                }
            }
        }
        order[i-1] = 0;
    }
    cout << ans << endl;

    return 0;
}
