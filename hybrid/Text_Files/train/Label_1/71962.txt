def maximize_number(a, k):
    a = list(str(a))
    n = len(a)
    for i in range(n):
        if k <= 0:
            break
        pos = i
        for j in range(i+1, min(n, i+k+1)):
            if a[j] > a[pos]:
                pos = j
        k -= (pos - i)
        while pos > i:
            a[pos], a[pos-1] = a[pos-1], a[pos]
            pos -= 1
    return ''.join(a)

a, k = map(int, input().split())
print(maximize_number(a, k))
