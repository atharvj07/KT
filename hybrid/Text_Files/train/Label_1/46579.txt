import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int sum = 0;
        int day = -1;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            if (sum * 2 >= sum && day == -1) {
                day = i + 1;
            }
        }
        System.out.println(day);
    }
}
