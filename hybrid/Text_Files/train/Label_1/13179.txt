#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	ll n,mi=1e9,ma=-1e9;
	cin>>n;
	unordered_map<ll,ll>mp;
	for(ll i=0;i<n;++i) {
		ll x;cin>>x;
		++mp[x];
		mi=min(mi,x);
		ma=max(ma,x);
	}
	ll sum=0;
	for(auto i:mp) {
		sum+=i.first*i.second;
	}
	sum/=(ll)n;
	ll ans=n;
	for(auto i:mp) {
		if(i.first<sum) {
			ans-=min(i.second,mp[sum]);
			mp[sum]-=min(i.second,mp[sum]);
		}
		else if(i.first>sum) {
			ans-=min(i.second,mp[sum]);
			mp[sum]+=min(i.second,mp[sum]);
		}
	}
	cout<<ans<<'\n';
	for(ll i=mi;i<=ma;++i) {
		while(mp[i]--) cout<<i<<' ';
	}
	cout<<'\n';
	return 0;
}
