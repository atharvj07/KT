import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int round = 1;
        while (a != b) {
            if (a > b) {
                a = (a + 1) / 2;
            } else {
                b = (b + 1) / 2;
            }
            round++;
        }
        System.out.println(round == n ? "Final!" : round);
    }
}
