#include <bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    string s, ans; cin >> s;
    for (int i = 0; i < n; i++) {
        ans += s[i];
        unordered_map<char, int> mp;
        for (char c: ans) mp[c]++;
        int cnt = 0;
        for (auto p: mp) cnt = max(cnt, p.second);
        if (cnt <= ans.size()/2) {
            cout << "YES" << endl;
            cout << ans << endl;
            return 0;
        }
    }
    cout << "NO" << endl;
    return 0;
}
