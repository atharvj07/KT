#include<bits/stdc++.h>
using namespace std;
#define endl "\n"
typedef long long ll;
const int M=1e5+5;
int m;
string s,ans="";
int main()
{
    cin>>m;
    cin>>s;
    int n=s.size();
    s+="#";
    int Fr[M]={0},vis[M]={0};
    for(int i=0;i<n;i++){
        Fr[s[i]]++;
    }
    int last=-1;
    for(int i=0;i<n;i++)
    {
        if(Fr[s[i]]>0 && vis[s[i]]==0)
        {
            while(last!=-1 && Fr[s[last]]>0 && s[last]>s[i] && Fr[s[i]]>0)
            {
                Fr[s[last]]--;
                ans.pop_back();
                vis[s[last]]=0;
                last--;
            }
            ans+=s[i];
            vis[s[i]]=1;
            last++;
        }
        else{
            Fr[s[i]]--;
        }
        if(last-i+1>=m)
        {
            Fr[s[i]]++;
            vis[s[i]]=0;
            last--;
            ans.pop_back();
        }
    }
    cout<<ans<<endl;
    return 0;
}
