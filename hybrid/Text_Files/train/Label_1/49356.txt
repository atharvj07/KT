import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] expertMixture = new int[3];
        for (int i = 0; i < 3; i++) {
            expertMixture[i] = scanner.nextInt();
        }
        int numChanges = scanner.nextInt();
        Map<Double, List<Bottle>> bottles = new HashMap<>();
        for (int i = 0; i < numChanges; i++) {
            String command = scanner.next();
            if (command.equals("A")) {
                int[] bottle = new int[3];
                for (int j = 0; j < 3; j++) {
                    bottle[j] = scanner.nextInt();
                }
                double ratio = (double) bottle[0] / bottle[1] / bottle[2];
                bottles.computeIfAbsent(ratio, k -> new ArrayList<>()).add(new Bottle(i, bottle));
            } else if (command.equals("R")) {
                int toRemove = scanner.nextInt();
                for (Map.Entry<Double, List<Bottle>> entry : bottles.entrySet()) {
                    entry.getValue().removeIf(bottle -> bottle.num == toRemove);
                }
            }
            double expertRatio = (double) expertMixture[0] / expertMixture[1] / expertMixture[2];
            int result = 0;
            if (bottles.containsKey(expertRatio)) {
                result = 1;
            } else {
                for (Map.Entry<Double, List<Bottle>> entry1 : bottles.entrySet()) {
                    for (Map.Entry<Double, List<Bottle>> entry2 : bottles.entrySet()) {
                        if (entry1 != entry2) {
                            double ratio1 = entry1.getKey();
                            double ratio2 = entry2.getKey();
                            if (ratio1 * ratio2 == expertRatio) {
                                result = 2;
                                break;
                            }
                        }
                    }
                    if (result > 0) {
                        break;
                    }
                }
            }
            System.out.println(result);
        }
    }

    private static class Bottle {
        int num;
        int[] contents;

        public Bottle(int num, int[] contents) {
            this.num = num;
            this.contents = contents;
        }
    }
}
