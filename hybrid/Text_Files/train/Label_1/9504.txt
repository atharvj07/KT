#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> board(n);
  for (int i = 0; i < n; i++) { cin >> board[i]; }

  bool possible = true;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (board[i][j] == '#') { continue; }
      if (i + 2 < n && j + 2 < n && board[i + 1][j] == board[i + 2][j] &&
          board[i + 2][j] == board[i + 1][j + 1] &&
          board[i + 1][j + 1] == board[i + 1][j + 2] &&
          board[i][j] == board[i + 1][j]) {
        board[i + 1][j] = board[i + 2][j] = board[i + 1][j + 1] =
            board[i + 1][j + 2] = board[i][j] = '#';
      } else if (i + 2 < n && j - 2 >= 0 && board[i + 1][j] == board[i + 2][j] &&
                 board[i + 2][j] == board[i + 1][j - 1] &&
                 board[i + 1][j - 1] == board[i + 1][j - 2] &&
                 board[i][j] == board[i + 1][j]) {
        board[i + 1][j] = board[i + 2][j] = board[i + 1][j - 1] =
            board[i + 1][j - 2] = board[i][j] = '#';
      } else if (j + 2 < n && i - 2 >= 0 && board[i][j] == board[i][j + 1] &&
                 board[i][j + 1] == board[i - 1][j + 1] &&
                 board[i - 1][j + 1] == board[i - 1][j + 2] &&
                 board[i][j] == board[i - 1][j + 1]) {
        board[i][j] = board[i][j + 1] = board[i - 1][j + 1] =
            board[i - 1][j + 2] = board[i][j] = '#';
      } else if (j - 2 >= 0 && i - 2 >= 0 && board[i][j] == board[i][j - 1] &&
                 board[i][j - 1] == board[i - 1][j - 1] &&
                 board[i - 1][j - 1] == board[i - 1][j - 2] &&
                 board[i][j] == board[i - 1][j - 1]) {
        board[i][j] = board[i][j - 1] = board[i - 1][j - 1] =
            board[i - 1][j - 2] = board[i][j] = '#';
      } else {
        possible = false;
        break;
      }
    }
  }

  if (possible) {
    cout << "YES";
  } else {
    cout << "NO";
  }
  cout << endl;

  return 0;
}
