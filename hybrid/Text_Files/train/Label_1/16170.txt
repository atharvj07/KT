#include<bits/stdc++.h>
using namespace std;
struct T { int l, r; } p[2005];
int a[2005], b[2005];
int main() {
    int n, ans = 0; cin >> n;
    for (int i = 1; i <= n; i++) {
        int x1, y1, x2, y2; cin >> x1 >> y1 >> x2 >> y2;
        if (x1 > x2) swap(x1, x2), swap(y1, y2);
        a[i] = x2, b[i] = y2;
        p[i].l = y1, p[i].r = y2;
    }
    sort(a + 1, a + n + 1), sort(b + 1, b + n + 1);
    sort(p + 1, p + n + 1, [&](T x, T y) {if (x.l == y.l) return x.r > y.r; return x.l < y.l; });
    int cnt = 0, l, r;
    for (int i = 1; i <= n; i++) {
        l = lower_bound(a + 1, a + n + 1, p[i].l) - a;
        r = upper_bound(a + 1, a + n + 1, p[i].r) - a - 1;
        if (cnt == r - l + 1) continue;
        cnt = r - l + 1;
        ans += (upper_bound(b + 1, b + n + 1, p[i].r) - lower_bound(b + 1, b + n + 1, p[i].l));
    }
    cout << ans << endl;
    return 0;
}
