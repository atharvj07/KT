import java.util.Scanner;
import java.math.BigInteger;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long n = sc.nextLong();
    long l = 1;
    long r = 1;
    while (r < n) {
      l = r + 1;
      r += l;
    }
    System.out.println(n - l + 1);
  }
}
