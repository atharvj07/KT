#include <cstdio>
int main() {
  int n, m, p = 0, v = 0;
  scanf("%d %d", &n, &m);
  if (n < m) {
    p += n - 1;
    m -= n;
    v += m / 2;
    m %= 2;
    if (m) p++, v++;
  } else {
    v += m - 1;
    n -= m;
    p += n / 2;
    n %= 2;
    if (n) p++, v++;
  }
  printf("%d %d", p, v);
  return 0;
}
