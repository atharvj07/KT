#include <bits/stdc++.h>
using namespace std; using ll = long long;
const int maxN = 1e5, maxM = 1e5;
vector<int> adj[maxN];
int proc[maxN];
int main() {
    int N, M; cin >> N >> M;
    for(int i=0; i<N; ++i) {
        cin >> proc[i];
    }
    for(int i=0; i<M; ++i) {
        int a, b; cin >> a >> b;
        adj[b].push_back(a);
    }
    queue<int> q;
    for(int i=0; i<N; ++i) {
        if(proc[i] == 0) q.push(i);
    }
    int ans = 0;
    while(!q.empty()) {
        ++ans;
        vector<int> cur;
        while(!q.empty()) {
            int u = q.front(); q.pop();
            cur.push_back(u);
            for(auto v: adj[u]) {
                if(--proc[v] == 0) {
                    q.push(v);
                }
            }
        }
        for(auto x: cur) {
            for(auto v: adj[x]) {
                if(proc[v] != 0) {
                    proc[v] = -1;
                }
            }
        }
    }
    cout << ans << endl;
}
