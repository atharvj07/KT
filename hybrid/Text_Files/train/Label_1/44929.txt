import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        Set<Integer> set = new HashSet<>();
        for (int i = 0; i < m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            int d = v - u;
            if (d < 0) d += n;
            set.add(d);
        }
        int[] g = new int[n];
        for (int i : set) {
            g[i]++;
        }
        for (int i = 0; i < n; i++) {
            if (g[i] == m || g[(n - i) % n] == m) {
                System.out.println("Yes");
                return;
            }
        }
        System.out.println("No");
    }
}
