import sys

def solve():
    n, k = map(int, input().split())
    p = list(map(int, input().split()))
    q = list(map(int, input().split()))
    s = ['a'] * n
    taken = [False] * 26
    for i in p:
        j = q.index(i)
        if taken[ord(s[j]) - ord('a')]:
            print('NO')
            return
        s[j] = chr(ord('a') + min(k - taken.count(True), 26 - j))
        taken[ord(s[j]) - ord('a')] = True
    print('YES')
    print(''.join(s))

solve()
