#include <bits/stdc++.h>
using namespace std;

const int maxN = 2e5 + 5;
int a[maxN];
long long dp[maxN][2];

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  dp[n + 1][0] = dp[n + 1][1] = 0;
  for (int i = n; i >= 1; i--) {
    for (int j = 0; j < 2; j++) {
      dp[i][j] = 1e18;
      if (a[i] == j)
        dp[i][j] = min(dp[i][j], dp[i + 1][!j]);
      dp[i][j] = min(dp[i][j], dp[i + 1][j] + a[i]);
    }
  }
  cout << dp[1][0] << endl;
  return 0;
}
