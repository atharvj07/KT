def fastpow(x: int, p: int) -> int:
    if p == 0:
        return 1
    if p == 1:
        return x

    val = fastpow(x, int(p / 2))
    val = (val * val) % (10 ** 9 + 7)

    if p % 2 != 0:
        val = (val * x) % (10 ** 9 + 7)

    return val

n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
mod = 10 ** 9 + 7

dp = [[[-1] * 10 for _ in range(k + 1)] for _ in range(n + 1)]

def solve(idx: int, cnt: int, last: int) -> int:
    if idx == n:
        if cnt == k:
            return 1
        return 0

    ret = dp[idx][cnt][last]
    if ret != -1:
        return ret

    st = 0
    if cnt == 0:
        st = 1

    ret = 0
    for i in range(st, 10):
        if i == b[int(idx / k)]:
            continue
        if (i * fastpow(10, k - cnt % k - 1) % mod + last) % a[int(idx / k)] == 0:
            ret += solve(idx + 1, cnt + 1, i)
            ret %= mod

    dp[idx][cnt][last] = ret
    return ret

print(solve(0, 0, 0))
