def max_army_strength(a):
    peaks = []
    for i in range(len(a)):
        if (i == 0 or a[i] > a[i-1]) and (i == len(a)-1 or a[i] > a[i+1]):
            peaks.append(a[i])
        elif (i == 0 or a[i] < a[i-1]) and (i == len(a)-1 or a[i] < a[i+1]):
            peaks.append(-a[i])
    if not peaks:
        return 0
    if peaks[0] < 0:
        peaks = peaks[1:]
    if not peaks:
        return 0
    if peaks[-1] < 0:
        peaks = peaks[:-1]
    return sum(peaks)

def solve():
    t = int(input().strip())
    for _ in range(t):
        n, q = map(int, input().split())
        a = list(map(int, input().split()))
        print(max_army_strength(a))
        for __ in range(q):
            l, r = map(int, input().split())
            a[l-1], a[r-1] = a[r-1], a[l-1]
            print(max_army_strength(a))

solve()
