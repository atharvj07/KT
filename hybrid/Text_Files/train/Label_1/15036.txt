#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=1e5,M=998244353;
int f[N];
bool isnotp[N],gs[N+5];
int ex[N],e2[N],e3[N],Fac[N],invF[N];
int p,n;
//************************
inline ll fpow(ll a,ll b){ ll ans=1;for(;b;b>>=1){ if(b&1) ans=ans*a%M;a=a*a%M;}return ans;}
inline int getfactor(int x){
    int ans=0,tmp=x;
    for(int i=2;i*i<=tmp;i++){
        if(tmp%i==0){
            int cnt=0;
            while(tmp%i==0){ tmp/=i;cnt++;}
            ans+=cnt;
        }
    }
    if(tmp>1) ans++;
    return ans;
}
//************************
int main(){
    int tmp;
    ios::sync_with_stdio(false);
    cin>>n>>p;
    for(int i=0;i<=n;i++){
        cin>>tmp;
        tmp=tmp%p;
        gs[tmp]=true;
    }
    //**********Fact and Invfact********
    Fac[0]=1;
    for(int i=1;i<=n;i++) Fac[i]=Fac[i-1]*i%p;
    invF[n]=fpow(Fac[n],p-2);
    for(int i=n-1;i>=1;i--) invF[i]=invF[i+1]*(i+1)%p;
    //***************isnotp*******************
    isnotp[0]=isnotp[1]=true;
    isnotp[2]=false;
    for(int i=4;i<=n;i+=2) isnotp[i]=true;
    for(int i=3;i*i<=n;i+=2){
        if(!isnotp[i]){
            for(int j=i*i;j<=n;j+=2*i){
                isnotp[j]=true;
            }
        }
    }
    //*****initialize*******
    for(int i=0;i<=1e5;i++){
        if(!isnotp[i]) {
            ex[i]=getfactor(i);
            if(ex[i]==1){
                if(i%2==0) e2[i]=ex[i];
                if(i%3==0) e3[i]=ex[i];
            }
        }
    }
    //****************
    for(int i=0;i<=n;i++){
        if(!isnotp[i]) {
            if(i%2==0) e2[i]+=ex[i/2];
            if(i%3==0) e3[i]+=ex[i/3];
        }
    }
    //*********f*********
    f[0]=1;
    for(int i=1;i<=n-1;i++){
        f[i]=Fac[i];
        if(i%2==0) f[i]=f[i]*invF[i/2]%p;
        if(i%3==0) f[i]=f[i]*invF[i/3]%p;
    }
    f[n]=-1;
    if(p%2==0) f[n]=f[n]*invF[n/2]%p;
    if(p%3==0) f[n]=f[n]*invF[n/3]%p;
    //********ex**************
    for(int i=0;i<=n;i++){
        if(gs[i]){
            e2[i]=max(e2[i],e2[i-1]);
            e3[i]=max(e3[i],e3[i-1]);
        }else{
            e2[i]=e2[i-1];
            e3[i]=e3[i-1];
        }
    }
    //*****check g***********
    int ans=0;
    for(int i=0;i<=n;i++){
        if(e2[i]<e3[i]){
            if(f[i]!=0) ans++;
        }
    }
    cout<<ans<<endl;
    for(int i=0;i<=n;i++){
        if(e2[i]<e3[i]){
            if(f[i]!=0) cout<<i<<" ";
        }
    }
    return 0;
}
