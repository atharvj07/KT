n = int(input())
w = [0 for _ in range(n)]
h = [0 for _ in range(n)]
for i in range(n):
    w[i], h[i] = map(int, input().split())
for i in range(n):
    sum_width = 0
    max_height = 0
    for j in range(n):
        if i == j:
            continue
        sum_width += w[j]
        max_height = max(max_height, h[j])
    print(sum_width * max_height)
