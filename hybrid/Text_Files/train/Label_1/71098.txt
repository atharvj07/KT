from collections import deque

def bfs(graph, start, n, k):
    distances = [-1] * (n + 1)
    queue = deque([start])
    distances[start] = 0
    while queue:
        vertex = queue.popleft()
        for neighbor in graph[vertex]:
            if distances[neighbor] == -1:
                distances[neighbor] = distances[vertex] + 1
                queue.append(neighbor)
    return [((dist - 1) // k) + 1 for dist in distances if dist != -1]

n, k = map(int, input().split())
graph = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

result = 0
for i in range(1, n + 1):
    dists = bfs(graph, i, n, k)
    result += sum(dists) - 1  # Subtract 1 because bfs includes the start node itself

print(result)
