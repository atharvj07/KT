#include<iostream>
#include<vector>
#define MOD 1000000007
using namespace std;
int main()
{
    int n,k,p10[10],a[100010],b[100010];
    cin>>n>>k;
    for(int i=0; i<n/k; i++)
        cin>>a[i];
    for(int i=0; i<n/k; i++)
        cin>>b[i];
    p10[0]=1;
    for(int i=1; i<=9; i++)
        p10[i]=p10[i-1]*10;
    long long int ans=0;
    for(int i=0; i<n/k; i++)
    {
        long long int s=0;
        for(int j=i*k; j<(i+1)*k; j++)
        {
            s*=10;
            s+=b[i];
        }
        s++;
        ans=(ans+(s/a[i]))%MOD;
    }
    cout<<ans<<endl;
}
