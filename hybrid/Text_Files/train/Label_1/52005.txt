import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[][] w = new int[N][N];
        List<Integer> roads = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            w[i][i] = 0;
            for (int j = i + 1; j < N; j++) {
                roads.add(i * N + j);
            }
        }
        Collections.shuffle(roads);
        int[] ans = new int[N * (N - 1) / 2];
        for (int i = 0; i < N * (N - 1) / 2; i++) {
            int r = roads.get(i);
            w[r / N][r % N] = ans[i] = (int) (Math.random() * 1000000000) + 1;
            w[r % N][r / N] = ans[i];
        }
        for (int[] a : w) {
            for (int b : a) {
                System.out.print(b + " ");
            }
            System.out.println();
        }
    }
}
