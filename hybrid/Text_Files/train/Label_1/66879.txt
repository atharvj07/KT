h = input()
a = input()
n = int(input())
d = {}
for i in range(n):
    t, s, m, c = input().split()
    t = int(t)
    if s == 'h' and c == 'y':
        if h+str(m) in d: d[h+str(m)].append(t)
        else: d[h+str(m)] = [t]
    elif s == 'h' and c == 'r':
        print(h, m, t)
    elif s == 'a' and c == 'y':
        if a+str(m) in d: d[a+str(m)].append(t)
        else: d[a+str(m)] = [t]
    elif s == 'a' and c == 'r':
        print(a, m, t)
    if s+str(m) in d and len(d[s+str(m)]) == 2:
        print(h if s == 'h' else a, m, d[s+str(m)][1])
        del d[s+str(m)]
