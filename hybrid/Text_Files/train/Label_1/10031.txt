#include <iostream>
#include <cmath>

int main() {
    int n;
    std::cin >> n;
    int a[n];
    for (int i = 0; i < n; ++i) {
        std::cin >> a[i];
    }
    int max_not_perfect_square = -1e9;
    for (int i = 0; i < n; ++i) {
        int sqrt_a = std::sqrt(a[i]);
        if (sqrt_a * sqrt_a != a[i]) {
            max_not_perfect_square = std::max(max_not_perfect_square, a[i]);
        }
    }
    std::cout << max_not_perfect_square << std::endl;
    return 0;
}
