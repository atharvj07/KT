n = int(input())
cards = [list(map(int, input().split()))[1:] for _ in range(n)]
nums = [0] * 101
for c in cards:
    for num in c:
        nums[num] += 1

for c in cards:
    can = True
    for num in c:
        if nums[num] == 1:
            can = False
    if can:
        print("YES")
    else:
        print("NO")
