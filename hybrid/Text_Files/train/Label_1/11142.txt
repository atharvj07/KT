#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, d;
    cin >> n >> d;
    string s;
    cin >> s;
    int dp[n] = {0};
    for (int i = 1; i < n; i++) {
        if (!s[i]) continue;
        int mn = INT_MAX;
        for (int j = max(0, i - d); j < i; j++) {
            if (dp[j] < INT_MAX) mn = min(mn, dp[j]);
        }
        if (mn == INT_MAX) continue;
        dp[i] = mn + 1;
    }
    if (dp[n - 1] == 0)
        cout << -1;
    else
        cout << dp[n - 1];
    return 0;
}
