#include <bits/stdc++.h>
using namespace std;
const int maxn=3e5+10;
vector<int> g[55];
int a[maxn],ans[maxn],x[55];
int main() {
    int n,q;cin>>n>>q;
    for(int i=1;i<=n;i++) {cin>>a[i];g[a[i]].push_back(i);}
    for(int i=1;i<=q;i++){
        int u;cin>>u;
        ans[i]=g[u][x[u]];
        x[u]++;
    }
    for(int i=1;i<=q;i++) cout<<ans[i]<<'\n';
    return 0;
}
