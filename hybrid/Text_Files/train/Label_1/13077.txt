#include <algorithm>
#include <vector>
#include <functional>
#include <iostream>
#include <climits>

using namespace std;

int solve(string a, string b) {
  int len = a.size();
  int minhd = INT_MAX, mini = -1, minj = -1;

  for (int i = 0; i < len; ++i) {
    for (int j = 0; j < len; ++j) {
      if (i == j) continue;
      int hd = 0;
      for (int k = 0; k < len; ++k) {
        hd += (a[k] == b[k] || (k == i && a[k] == b[j]) || (k == j && a[k] == b[i])) ? 0 : 1;
      }
      if (hd < minhd) {
        minhd = hd;
        mini = i+1;
        minj = j+1;
      }
    }
  }
  cout << minhd << '\n';
  if (minhd < len) cout << mini << ' ' << minj << '\n';
  else cout << "-1 -1\n";
  return 0;
}

int main() {
  int len;
  cin >> len;
  string a, b;
  cin >> a >> b;
  solve(a, b);
  return 0;
}
