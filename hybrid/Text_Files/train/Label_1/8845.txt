#include <bits/stdc++.h>

using namespace std;

const int maxn = 1e5 + 10;
char s[maxn];
int n, q, l, r, k;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> s >> q;
  n = strlen(s);
  while (q--) {
    cin >> l >> r >> k;
    reverse(s + l - 1, s + r);
    reverse(s + r - k, s + r);
    reverse(s + l - 1, s + r);
  }
  cout << s << '\n';
  return 0;
}
