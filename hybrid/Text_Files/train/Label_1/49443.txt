import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();
        int m = scanner.nextInt();

        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = a + i * d;
        }

        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt() - 1;
            int z = scanner.nextInt() - 1;

            if (x == 0) {
                reverse(arr, y, z);
            } else if (x == 1) {
                increment(arr, y, z);
            } else {
                halve(arr, y, z);
            }
        }

        int k = scanner.nextInt() - 1;
        System.out.println(arr[k]);
        scanner.close();
    }

    private static void reverse(int[] arr, int y, int z) {
        while (y < z) {
            int temp = arr[y];
            arr[y] = arr[z];
            arr[z] = temp;
            y++;
            z--;
        }
    }

    private static void increment(int[] arr, int y, int z) {
        for (int i = y; i <= z; i++) {
            arr[i]++;
        }
    }

    private static void halve(int[] arr, int y, int z) {
        for (int i = y; i <= z; i++) {
            arr[i] /= 2;
        }
    }
}
