n, k = map(int, input().split())
c = list(map(int, input().split()))
capitals = list(map(int, input().split()))

total = 0
sum_c = sum(c)

for i in range(n):
    if (i + 1) in capitals:
        total += c[i] * (sum_c - c[i])
    else:
        if i == 0:
            total += c[i] * c[n-1]
        else:
            total += c[i] * c[i-1]

for i in range(k - 1):
    for j in range(i + 1, k):
        total -= c[capitals[i] - 1] * c[capitals[j] - 1]

print(total)
