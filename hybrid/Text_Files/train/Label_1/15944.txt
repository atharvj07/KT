#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k, x;
  cin >> n >> k >> x;
  deque<long long> arr(n);
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  sort(arr.rbegin(), arr.rend());
  while (k > 0) {
    if (arr[0] < x) {
      arr[0] *= x;
      k--;
    } else
      break;
  }
  long long ans = arr[0];
  for (int i = 1; i < n; i++) {
    ans |= arr[i];
  }
  cout << ans << endl;
  return 0;
}
