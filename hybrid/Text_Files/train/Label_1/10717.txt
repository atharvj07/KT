#include <iostream>
#include <vector>
#include <queue>

using namespace std;

struct cell {
    int x, y, v;
};

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<int>> a(n, vector<int>(m));
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                cin >> a[i][j];
        bool win = true;
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                if (a[i][j] > 0)
                    win = false;
        cout << (win ? "Jeel" : "Ashish") << endl;
    }
    return 0;
}
