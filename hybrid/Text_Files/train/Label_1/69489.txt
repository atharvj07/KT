for _ in range(int(input())):
  n, m = map(int, input().split())
  a = [[0] * (m + 2) for _ in range(n + 2)]
  for i in range(n):
    x = list(map(int, input().split())) + [0, 0]
    a[i+1][1:m+1] = x
  ans = 'Vivek'
  for i in range(1, n+1):
    for j in range(1, m+1):
      if a[i][j] == 0:
        if sum(a[i]) + sum(x[j] for x in a) == 1:
          ans = 'Ashish'
  print(ans)
