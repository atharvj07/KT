#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, k, m; cin >> n >> k >> m;
  vector<int> t(k); for (int i = 0; i < k; i++) cin >> t[i];
  sort(t.begin(), t.end());
  int ans = 0, sum = 0, i = 0;
  while (sum + t[i] <= m and i < k) sum += t[i++];
  ans += i;
  if (i == k) ans += (m - sum) / t[k - 1];
  cout << ans << endl;
  return 0;
}
