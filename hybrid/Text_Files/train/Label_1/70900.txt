n, k = map(int, input().split())
x = [int(i) for i in input().split()]
y = [int(i) for i in input().split()]

if max(x) > k or max(y) > k:
    print("NO")
    exit()

i = 0
while i < n:
    j = i
    while j < n and x[j] <= k:
        j += 1
    if j - i > k:
        print("NO")
        exit()
    i = j

i = 0
while i < n:
    j = i
    while j < n and y[j] <= k:
        j += 1
    if j - i > k:
        print("NO")
        exit()
    i = j

print("YES")
