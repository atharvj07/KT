n, k = map(int, input().split())
if k > n:
    print(-1)
else:
    ans = [0] * n
    cnt = 0
    for i in range(2, n + 1):
        if cnt >= k:
            break
        if n % i == 0:
            ans[n // i - 1] = i
            cnt += 1
    for i in range(n):
        if ans[i] == 0:
            ans[i] = i + 1
    print(*ans)
