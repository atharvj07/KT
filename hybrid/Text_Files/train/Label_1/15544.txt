#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<ll, ll> pi;
#define F first
#define S second
const ll MM = 1e5 + 5;
vector<pi> adj[MM];
ll n, m, k, dist[MM], mx[MM];

void dijkstra(ll st) {
    priority_queue<pi, vector<pi>, greater<pi>> pq;
    pq.push({0, st});
    memset(dist, 0x3f, sizeof dist);
    dist[st] = 0;
    while (!pq.empty()) {
        ll u = pq.top().S;
        ll w = pq.top().F;
        pq.pop();
        if (dist[u] != w) continue;
        for (auto &e : adj[u]) {
            ll v = e.F;
            ll wv = e.S;
            if (dist[v] > dist[u] + wv) {
                dist[v] = dist[u] + wv;
                pq.push({dist[v], v});
            }
        }
    }
}

int main() {
    cin >> n >> m >> k;
    vector<ll> sp(k);
    for (auto &it : sp) cin >> it;
    while (m--) {
        ll u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    for (auto &it : sp) dijkstra(it);
    for (auto &it : sp) {
        for (auto &e : adj[it]) {
            mx[it] = max(mx[it], dist[e.F]);
        }
    }
    for (auto &it : sp) cout << mx[it] << " ";
}
