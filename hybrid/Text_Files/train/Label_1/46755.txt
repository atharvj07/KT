import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] pieces = new int[n / 2];
        for (int i = 0; i < n / 2; i++) {
            pieces[i] = sc.nextInt();
        }

        int ans = n;
        for (int i = 0; i < n / 2; i++) {
            int left = pieces[i] - 1;
            int right = n - pieces[i];
            ans = Math.min(ans, left + right);
        }

        System.out.println(ans);
    }
}
