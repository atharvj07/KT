#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<bool> color(n);
  bool can = true;
  for (int i = 0; i < n - 1; i++) {
    if (s[i] > s[i + 1]) {
      color[i] = true;
    } else if (s[i] == s[i + 1]) {
      can = false;
    }
  }
  if (!can) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    for (int i = 0; i < n; i++) {
      cout << (color[i] ? '1' : '0');
    }
    cout << endl;
  }
  return 0;
}
