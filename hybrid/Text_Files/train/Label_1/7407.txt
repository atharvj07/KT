#include <bits/stdc++.h>
using namespace std;
const int N = 1e3,L = 55;
int n,m,cnt[30];
char s[L];
string a[N];
int main(){
    cin>>n>>s>>m;
    for(int i=0;i<m;i++){
        cin>>a[i];
        for(int j=0;j<n;j++)if(s[j]!='*')cnt[a[i][j]-'a']++;
    }
    for(int i=0;i<n;i++)if(s[i]=='*'){
        for(char c='a';c<='z';c++){
            bool ok = true;
            for(int j=0;j<m;j++){
                if(a[j][i]==c&&cnt[c-'a']==0){
                    ok = false;
                    break;
                }
            }
            if(ok){
                cout<<1<<endl;
                return 0;
            }
        }
    }
    cout<<0<<endl;
    return 0;
}
