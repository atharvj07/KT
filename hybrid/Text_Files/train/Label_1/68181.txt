from collections import deque

s=input()
l=len(s)
opn=0
a=[]
for i in range(l):
	if s[i]=='(':
		opn+=1
	elif opn==0:
		a.append(-1)
	else:
		opn-=1
		a.append(0)

if opn!=0:
	print(-1)
else:
	cl=0
	for i in range(l-1,-1,-1):
		if s[i]==')':
			cl+=1
		elif cl==0:
			a[i]=-1
		else:
			cl-=1
			a[i]=cl

	for x in a:
		print(max(x,0))
