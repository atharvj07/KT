#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    sort(v.begin(), v.end());
    long long ans = 0;
    for (int i = 0; i < n; i += 4) {
        if (i + 3 < n && v[i] == v[i + 1] && v[i + 2] == v[i + 3]) {
            ans += (long long)v[i] * v[i + 2];
        }
    }
    cout << ans << endl;
    return 0;
}
