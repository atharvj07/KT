MOD = 998244353

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(n, m, board):
    w = sum(row.count('o') for row in board)
    dp = [0] * (max(n,m) + 1)
    dp[1] = 1
    for i in range(2, max(n,m) + 1):
        dp[i] = (dp[i-1] + dp[i-2]) % MOD
    ans = 0
    for i in range(n):
        cnt = 0
        for j in range(m):
            if board[i][j] == 'o':
                cnt += 1
            else:
                ans = (ans + dp[cnt] * power(2, w - cnt, MOD)) % MOD
                cnt = 0
        ans = (ans + dp[cnt] * power(2, w - cnt, MOD)) % MOD
    for j in range(m):
        cnt = 0
        for i in range(n):
            if board[i][j] == 'o':
                cnt += 1
            else:
                ans = (ans + dp[cnt] * power(2, w - cnt, MOD)) % MOD
                cnt = 0
        ans = (ans + dp[cnt] * power(2, w - cnt, MOD)) % MOD
    return ans

n, m = map(int, input().split())
board = [input().strip() for _ in range(n)]
print(solve(n, m, board))
