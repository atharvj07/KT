#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    long long ans = 0;
    for (int i = 0; i < (n + 1) / 2; i++) { ans += a[i] * (n - 1 - i) - a[n - 1 - i] * i; }
    cout << ans << '\n';
  }
  return 0;
}
