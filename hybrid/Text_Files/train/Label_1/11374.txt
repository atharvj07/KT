#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;
vector<int> graph[MAXN];

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        long long mx = a[0] + 1;
        int ans = 1;
        for (int i = 1; i < n; i++) {
            if (a[i] >= mx) {
                ans++;
                mx = a[i] + 1;
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
