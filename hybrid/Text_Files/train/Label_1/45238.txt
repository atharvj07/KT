import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] d = new int[n];
        String s = sc.next();
        for (int i = 0; i < n; i++) {
            d[i] = s.charAt(i) - '0';
        }
        for (int i = 0; i < k; i++) {
            for (int j = 1; j < n - 1; j++) {
                if (d[j] == 4 && d[j + 1] == 7) {
                    d[j] = d[j + 1] = (j % 2 == 0) ? 7 : 4;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(d[i]);
        }
    }
}
