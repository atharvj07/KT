int main(){
  int n,p,q,k=0;
  cin >> n >> p >> q;
  string s;
  cin >> s;
  if(s.size() % (p+q) || p*q != s.size()) cout << -1;
  else{
    for(int i = 0; i < s.size(); i++){
      if(k % 2 == 0) cout << s.substr(i,p) << "\n", i += p - 1;
      else cout << s.substr(i,q) << "\n", i += q - 1;
      k++;
    }
  }
}
