import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int[] b = new int[2 * n];
        b[0] = 1;
        b[1] = 2;
        for (int i = 2; i < 2 * n; i++) {
            if (a[i] - a[i - 1] <= 10) {
                b[i] = b[i - 1];
            } else {
                b[i] = b[i - 1] ^ 3;
            }
        }
        System.out.println((n + 1) * n / 2);
        for (int i = 0; i < 2 * n; i++) {
            System.out.print(b[i] + " ");
        }
    }
}
