#include<bits/stdc++.h>
using namespace std;
int main() {
 int t;
 cin>>t;
 while(t--){
  string s;
  cin>>s;
  int n=s.size();
  int a[27]={0};
  for(int i=0;i<n;i++){
    a[s[i]-'a']++;
  }
  for(int i=0;i<26;i++){
    if(a[i]>1){
      cout<<"NO\n";
      break;
    }
  }
  if(a[0]>0)cout<<"NO\n";
  else cout<<"YES\n";
 }
}
