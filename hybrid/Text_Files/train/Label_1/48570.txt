import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    int m = sc.nextInt();
    long[] b = new long[m];
    for (int i = 0; i < m; i++) {
      b[i] = sc.nextInt();
    }
    long l = 0, r = 0, al = a[0], ar = a[n - 1], bl = b[0], br = b[m - 1];
    for (int i = 1; i < n; i++) {
      l = Math.max(l, a[i] - a[i - 1]);
      al = Math.min(al, a[i]);
      ar = Math.max(ar, a[i]);
    }
    for (int i = 1; i < m; i++) {
      r = Math.max(r, b[i] - b[i - 1]);
      bl = Math.min(bl, b[i]);
      br = Math.max(br, b[i]);
    }
    if (al == bl && ar == br) {
      System.out.println(Math.max(n, m) - Math.max(l, r));
    } else {
      System.out.println(-1);
    }
  }
}
