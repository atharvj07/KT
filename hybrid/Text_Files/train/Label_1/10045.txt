#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

ll beauty(vector<vector<ll>> &a, ll n, ll x, ll y) {
  if (n == 0) return a[x][y];
  ll mx = 0;
  for (ll i = x; i < x + (1 << n); ++i)
    for (ll j = y; j < y + (1 << n); ++j) mx = max(mx, a[i][j]);
  ll ret = mx;
  ret += beauty(a, n - 1, x, y);
  ret += beauty(a, n - 1, x, y + (1 << (n - 1)));
  ret += beauty(a, n - 1, x + (1 << (n - 1)), y);
  ret += beauty(a, n - 1, x + (1 << (n - 1)), y + (1 << (n - 1)));
  return ret;
}

int main() {
  ll n;
  cin >> n;
  ll k = n;
  n = 0;
  while (k) {
    n += (k & 1);
    k >>= 1;
  }
  vector<ll> v(1 << (2 * n));
  for (ll i = 0; i < (1 << (2 * n)); ++i) cin >> v[i];
  ll sz = 1 << n;
  vector<vector<ll>> a(sz, vector<ll>(sz));
  for (ll i = 0; i < (1 << (2 * n)); ++i) {
    ll x = i / (1 << n);
    ll y = i % (1 << n);
    a[x][y] = v[i];
  }
  cout << beauty(a, n, 0, 0) << '\n';
  return 0;
}
