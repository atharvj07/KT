#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const ll inf = 1e9;
const int maxn = 1e5 + 5;
vector<pair<int, int>>adj[maxn];
int par[maxn];
vector<int>res;
int findpar(int node) {
    if (node == par[node]) return node;
    return par[node] = findpar(par[node]);
}
void unite(int a, int b) {
    a = findpar(a);
    b = findpar(b);
    if (a != b) {
        par[b] = a;
    }
}
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int n, m; cin >> n >> m;
    for (int i = 0; i <= n; i++)par[i] = i;
    int ans = 0, k = 0;
    while (m--) {
        int a, b, w; cin >> a >> b >> w;
        if (findpar(a) == findpar(b)) {
            ans += w;
            res.push_back(m + 1);
            unite(a, b ^ 1);
            k++;
        }
    }
    cout << ans << ' ' << k << endl;
    for (int x : res)cout << x << ' ';
    cout << endl;
    return 0;
}
