#include<bits/stdc++.h>
using namespace std;typedef long long ll;
ll f(string a){
    ll n=a.size();
    ll pre[n+1]={0},suf[n+1]={0},dp[n+4]={0};
    for(ll i=0;i<n;i++){
        if(i>0)pre[i]=pre[i-1];
        if(a[i]=='9')pre[i]++;
    }
    for(ll i=n-1;i>=0;i--){
        if(i<n-1)suf[i]=suf[i+1];
        if(a[i]=='9')suf[i]++;
    }
    for(ll i=n-1;i>=0;i--){
        if(i<n-1)dp[i]=dp[i+1];
        if(a[i]==a[i+1]&&a[i]-'0'+a[i+1]-'0'==9)dp[i]++;
        else if(a[i]-'0'+suf[i+1]>=9){
            ll res=pre[i-1];
            res=res+suf[i+1]-(a[i]-'0'+suf[i+1]==9);
            res=res+dp[i+1]+1;
            dp[i]=max(dp[i],res);
        }
    }
    return dp[0];
}
int main()
{
    string a;cin>>a;cout<<f(a)<<endl;
}
