#include <iostream>
#include <string>

using namespace std;

bool equivalent(string a, string b) {
  int n = a.size();
  if (a == b) return true;
  for (int i = 1; i < n; i++) {
    string a1 = a.substr(0, i);
    string a2 = a.substr(i, n - i);
    string b1 = b.substr(0, i);
    string b2 = b.substr(i, n - i);
    if ((equivalent(a1, b1) && equivalent(a2, b2)) || (equivalent(a1, b2) && equivalent(a2, b1))) return true;
  }
  return false;
}

int main() {
  string a, b;
  cin >> a >> b;
  cout << (equivalent(a, b) ? "YES" : "NO") << endl;
  return 0;
}
