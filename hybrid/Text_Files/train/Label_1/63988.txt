n, w = map(int, input().split())
cups = [int(i) for i in input().split()]
cups.sort(reverse=True)
tea = [0]*n
rem = w
for i in range(n):
    tea[i] = cups[i]//2
    rem -= tea[i]
    if rem < 0:
        print(-1)
        exit()
if rem > 0:
    for i in range(n):
        tea[i] += 1
        rem -= 1
        if rem == 0:
            break
print(*tea)
