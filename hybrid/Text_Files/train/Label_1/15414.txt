#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int mx = *max_element(a.begin(), a.end());
    while (true) {
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] < mx) cnt += mx - a[i];
        }
        if (cnt > 0) break;
        mx++;
    }
    cout << mx << endl;
    return 0;
}
