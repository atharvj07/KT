import java.util.*;
import java.lang.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int l = in.nextInt();
        int a = in.nextInt();
        int[] t = new int[n];
        int[] l1 = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = in.nextInt();
            l1[i] = in.nextInt();
        }
        int p = 0;
        int ans = 0;
        int i = 0;
        for (; i < n; i++) {
            p = Math.max(p, t[i]);
            if (p + l1[i] <= l) {
                p = p + l1[i] + a;
                ans++;
            } else {
                break;
            }
        }
        System.out.println(ans);
    }
}
