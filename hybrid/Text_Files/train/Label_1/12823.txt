#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n + 1);
  for (int i = 1; i <= n; i++) { cin >> p[i]; }
  vector<int> ans(n + 1);
  for (int i = 1; i <= n; i++) {
    int j = i;
    while (ans[j] == 0) {
      ans[j] = i;
      j = p[j];
    }
    if (ans[j] == i) { ans[j] = -1; }
  }
  for (int i = 1; i <= n; i++) {
    if (ans[i] == -1) {
      ans[i] = i;
    }
    cout << ans[i] << " ";
  }
  cout << endl;
  return 0;
}
