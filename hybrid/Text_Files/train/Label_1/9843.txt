#include<bits/stdc++.h>
using namespace std;

int n, m, k;
vector<int> c;
vector<vector<int>> p;
vector<int> vis;

int main() {
    ios_base::sync_with_stdio(false);

    cin >> n >> m >> k;
    c.assign(n + 1, 0), p.assign(n + 1, vector<int>(m + 1, 0));
    vis.assign(n + 1, 0);

    for (int i = 1; i <= n; i++) {
        cin >> c[i];
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> p[i][j];
        }
    }

    int R = 0;
    for (int l = 1, r = 1; r <= n; r++) {
        if (!c[r]) {
            R++;
        }
        while (R > k) {
            if (!c[l]) {
                R--;
            }
            l++;
        }
        if (R == k) {
            int ans = 0;
            for (int i = l; i <= r; i++) {
                if (!c[i]) {
                    ans += p[i][c[i]];
                }
            }
            if (!vis[ans]) {
                vis[ans] = 1;
                cout << ans << endl;
                return 0;
            }
        }
    }
    cout << -1 << endl;
    return 0;
}
