#include <iostream>
#include <sstream>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s = to_string(n);
  s.push_back(s[0]), s.push_back(s[2]), s.push_back(s[4]);
  s[1] = s[3], s[2] = s[4], s[4] = '\0';
  stringstream ss(s);
  long long ans;
  ss >> ans;
  ans = ans * ans * ans * ans * ans;
  s = to_string(ans);
  cout << s.substr(s.length() - 5, 5);
}
