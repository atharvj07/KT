def lowbit(x):
    return x & -x

def solve(sum, limit):
    elements = [(lowbit(i), i) for i in range(1, limit + 1)]
    elements.sort(reverse=True)
    result = []
    for val, idx in elements:
        if sum >= val:
            sum -= val
            result.append(idx)
        if sum == 0:
            break
    if sum > 0:
        return [-1]
    else:
        return [len(result)] + result

sum, limit = map(int, input().split())
result = solve(sum, limit)
for item in result:
    print(item)
