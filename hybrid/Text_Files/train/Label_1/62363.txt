n, k = map(int, input().split())
skills = list(map(int, input().split()))

team = [0] * n
order = sorted(range(n), key=lambda x: skills[x], reverse=True)
coach = 1

for i in order:
    if team[i] == 0:
        team[i] = coach
        for j in range(1, k + 1):
            if i - j >= 0 and team[i - j] == 0:
                team[i - j] = coach
            if i + j < n and team[i + j] == 0:
                team[i + j] = coach
        coach = 2 if coach == 1 else 1

print("".join(map(str, team)))
