#include <iostream>
using namespace std;
int f(int n) {
    for (int i = 2; i * i <= n; ++i) {
        if (n % i == 0) return i;
    }
    return n;
}
int main() {
    int t;
    cin >> t;
    for (int _ = 0; _ < t; ++_) {
        int n, k;
        cin >> n >> k;
        for (int i = 0; i < k; ++i) {
            n += f(n);
        }
        cout << n << endl;
    }
}
