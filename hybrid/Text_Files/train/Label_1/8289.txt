#include <iostream>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  for (int i = 0; i < n; i++) {
    for (char c = '0'; c <= '9'; c++) {
      if (i == 0 && c == '0') continue;
      if (s[i] > c && k > 0) {
        s[i] = c;
        k--;
        break;
      }
    }
  }
  while (k--) s[n - 1 - k] = '0';
  cout << s << endl;
  return 0;
}
