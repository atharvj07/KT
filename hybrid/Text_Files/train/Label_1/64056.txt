import sys

mod = 1000000007
n = int(sys.stdin.readline())
a = [int(x) for x in sys.stdin.readline().split()]
g = [1 for x in a]

for i in range(n):
    for j in range(i):
        if a[i] % a[j] == 0:
            g[i] = max(g[i], g[j] + 1)

strength = 0

for i in range(n):
    strength = (strength + g[i] * a[i]) % mod

print(strength)
