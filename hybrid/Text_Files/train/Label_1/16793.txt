#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll p[1000005], inv[1000005];
const ll N = 1e9 + 7;
ll bigmod(ll n, ll p) {
    if (p == 0) return 1;
    if (p % 2 == 0) {
        ll val = bigmod(n, p / 2) % N;
        return (val * val) % N;
    }
    return (n % N * bigmod(n, p - 1) % N) % N;
}
ll modinv(ll n) { return bigmod(n, N - 2); }
ll fact(ll n) {
    if (n == 0) return 1;
    return (n % N * fact(n - 1) % N) % N;
}
void pre() {
    p[0] = 1;
    inv[0] = 1;
    for (ll i = 1; i < 1000005; i++) {
        p[i] = (p[i - 1] % N * i % N) % N;
        inv[i] = modinv(p[i]);
    }
}
ll ncr(ll n, ll r) {
    if (n < r) return 0;
    ll ans = (p[n] % N * inv[n - r] % N) % N;
    ans = (ans % N * inv[r] % N) % N;
    return ans;
}
int main() {
    pre();
    ll n, k;
    cin >> n >> k;
    vector<ll> v(n);
    for (ll i = 0; i < n; i++) {
        cin >> v[i];
    }
    ll cnt = 0;
    for (ll i = 0; i < n; i++) {
        if (v[i] == i) cnt++;
    }
    ll ans = ncr(n + k - cnt - 1, n - cnt - 1);
    cout << ans % N << endl;
    return 0;
}
