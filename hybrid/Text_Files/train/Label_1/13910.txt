#include <queue>
#include <vector>
#include <cstdio>
#include <algorithm>

using namespace std;

int main() {
  int n;
  long long s = 0;
  priority_queue<long long> q;
  scanf("%d", &n);
  for(int i = 0; i < n; i++) {
    long long a;
    scanf("%lld", &a);
    s += a;
    q.push(-a);
  }
  double avg = s / (double) n;
  vector<double> ans(n, avg);
  while(q.size() > 1) {
    long long a = -q.top(); q.pop();
    long long b = -q.top(); q.pop();
    int cnt = 1;
    for(int i = 0; i < n; i++) {
      if(ans[i] == a) {
        ans[i] = (a * cnt + b) / (double) (cnt + 1);
        cnt++;
      }
    }
    s -= a - (a * cnt + b) / (double) (cnt + 1);
    avg = s / (double) n;
    q.push(-(a * cnt + b) / (double) (cnt + 1));
  }
  for(int i = 0; i < n; i++) printf("%.9lf\n", ans[i]);
  return 0;
}
