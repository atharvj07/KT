n = int(input())
a = input()
f = list(map(int, input().split()))

i = 0
result = ''
while i < n:
    if f[int(a[i])-1] > int(a[i]):
        while i < n and f[int(a[i])-1] > int(a[i]):
            result += str(f[int(a[i])-1])
            i += 1
    else:
        result += a[i]
        i += 1
print(result)
