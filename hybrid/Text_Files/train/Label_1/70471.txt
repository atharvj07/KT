import sys

input = sys.stdin.readline

n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(4)]

ans = []
for x in range(1, m + 1):
    for i in range(1, n + 1):
        if a[0][i] == x or a[3][i] == x:
            for j in range(1, n + 1):
                if a[2][j] == x and (i, j) != (1, i) and (i, j) != (4, i):
                    ans.append((x, 2, j))
                    ans.append((x, 3, i))
                    ans.append((x, 1, i))
                    break
        if a[1][i] == x or a[2][i] == x:
            for j in range(1, n + 1):
                if a[0][j] == x and (i, j) != (1, i) and (i, j) != (4, i):
                    ans.append((x, 1, i))
                    ans.append((x, 4, i))
                    ans.append((x, 3, j))
                    break

if len(ans) > 20000:
    print(-1)
else:
    print(len(ans))
    for i in ans:
        print(*i)
