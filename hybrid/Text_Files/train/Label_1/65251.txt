import sys

def main():
    t = int(sys.stdin.readline())
    for _ in range(t):
        n = int(sys.stdin.readline())
        a = list(sys.stdin.readline().strip())
        if 't' in a:
            a.remove('t')
            a.append('t')
        if 'r' in a:
            a.remove('r')
            a.append('r')
        if 'y' in a:
            a.remove('y')
            a.append('y')
        if 'g' in a:
            a.remove('g')
            a.append('g')
        if 'u' in a:
            a.remove('u')
            a.append('u')
        if 'b' in a:
            a.remove('b')
            a.append('b')
        print(''.join(a))

main()
