import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, k = map(int, input().split())
    x = [int(x) for x in input().split()]
    y = [int(y) for y in input().split()]

    x.sort()
    y.sort()

    ans = n
    for i in range(n-1):
        if x[i+1] - x[i] > k:
            ans -= 1

    print(ans)
