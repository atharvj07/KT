#include <array>
#include <iostream>
int main() {
    int n;
    std::cin >> n;
    std::array<int, 100> a;
    for (size_t i = 0; i < n; i++) { std::cin >> a[i]; }
    for (int d = -1000; d <= 1000; d++) {
        if (d == 0) continue;
        int cnt = 0;
        for (size_t i = 0; i < n; i++) {
            if (a[i] * 1.0 / d > 0) cnt++;
        }
        if (cnt >= (n + 1) / 2) {
            std::cout << d << "\n";
            return 0;
        }
    }
    std::cout << 0 << "\n";
    return 0;
}
