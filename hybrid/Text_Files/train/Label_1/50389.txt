import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        boolean[] visited = new boolean[n + 1];
        Map<Integer, List<Integer>> graph = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            graph.put(i, new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int u = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);
            graph.get(u).add(v);
            graph.get(v).add(u);
        }
        if (m % 2 != 0 || hasOddDegreeVertex(graph)) {
            System.out.println("No solution");
            return;
        }
        List<List<Integer>> paths = new ArrayList<>();
        while (m > 0) {
            int start = findUnvisitedVertex(visited);
            if (start == -1) {
                break;
            }
            Queue<Integer> queue = new LinkedList<>();
            queue.offer(start);
            visited[start] = true;
            while (!queue.isEmpty()) {
                int current = queue.poll();
                int next = findUnvisitedNeighbor(current, graph, visited);
                if (next == -1) {
                    continue;
                }
                List<Integer> path = new ArrayList<>();
                path.add(current);
                path.add(next);
                paths.add(path);
                visited[next] = true;
                m -= 2;
            }
        }
        if (m == 0) {
            for (List<Integer> path : paths) {
                System.out.println(path.get(0) + " " + path.get(1) + " " + path.get(2));
            }
        } else {
            System.out.println("No solution");
        }
    }

    private static int findUnvisitedVertex(boolean[] visited) {
        for (int i = 1; i < visited.length; i++) {
            if (!visited[i]) {
                return i;
            }
        }
        return -1;
    }

    private static int findUnvisitedNeighbor(int vertex, Map<Integer, List<Integer>> graph, boolean[] visited) {
        for (int neighbor : graph.get(vertex)) {
            if (!visited[neighbor]) {
                return neighbor;
            }
        }
        return -1;
    }

    private static boolean hasOddDegreeVertex(Map<Integer, List<Integer>> graph) {
        for (int vertex : graph.keySet()) {
            if (graph.get(vertex).size() % 2 != 0) {
                return true;
            }
        }
        return false;
    }
}
