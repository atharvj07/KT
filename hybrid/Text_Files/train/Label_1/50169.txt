import java.util.*;
import java.io.*;

class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n + 1];
        int[] b = new int[n + 1];
        for (int i = 1; i <= n; i++) a[i] = in.nextInt();
        System.arraycopy(a, 0, b, 0, n + 1);
        Arrays.sort(b);
        boolean increasing = true;
        boolean decreasing = true;
        for (int i = 1; i <= n; i++) {
            if (a[i] > b[i]) increasing = false;
            if (a[i] < b[i]) decreasing = false;
        }
        if (increasing || decreasing) System.out.println(-1);
        else {
            for (int i = 1; i <= n; i++) {
                if (a[i] != b[i]) {
                    System.out.println(i + " " + (Arrays.binarySearch(b, a[i]) + 1));
                    return;
                }
            }
        }
    }
}
