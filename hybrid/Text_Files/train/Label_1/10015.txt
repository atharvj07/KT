#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> p(n);
  for (int i = 0; i < n; ++i) std::cin >> p[i];
  if (p[0] == p[n - 1]) {
    std::cout << p[0] << std::endl;
  } else if (p[0] < p[n - 1]) {
    std::cout << p[1] << std::endl;
  } else {
    std::cout << p[n - 2] << std::endl;
  }
}
