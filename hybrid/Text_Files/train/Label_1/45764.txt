import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), g = sc.nextInt(), r = sc.nextInt();
        long[] l = new long[n + 2];
        for (int i = 1; i <= n + 1; i++) l[i] = sc.nextLong();
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            long t = sc.nextLong();
            long ti = ((t + g - 1) / (g + r)) * (g + r);
            for (int j = 1; j <= n + 1; j++) {
                ti += l[j];
                if (ti % (g + r) >= g) ti += g + r - ti % (g + r);
            }
            System.out.println(ti);
        }
    }
}
