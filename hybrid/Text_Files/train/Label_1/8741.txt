#include <iostream>
#include <vector>
#include <algorithm>

struct Query {
    long long time;
    long long duration;
};

bool operator<(const Query& a, const Query& b) {
    return a.time < b.time;
}

int main() {
    long long n, b;
    std::cin >> n >> b;

    std::vector<Query> queries(n);
    for (long long i = 0; i < n; i++) {
        std::cin >> queries[i].time >> queries[i].duration;
    }

    std::sort(queries.begin(), queries.end());

    long long current_time = 0;
    long long end_time = 0;
    std::vector<long long> results(n);
    for (long long i = 0; i < n; i++) {
        if (current_time <= queries[i].time) {
            current_time = queries[i].time;
        }
        end_time = std::max(end_time, current_time + queries[i].duration);
        if (i + 1 < n && end_time > queries[i + 1].time) {
            results[i] = -1;
            continue;
        }
        results[i] = end_time;
        current_time = end_time;
    }

    for (long long i = 0; i < n; i++) {
        std::cout << results[i] << "\n";
    }

    return 0;
}
