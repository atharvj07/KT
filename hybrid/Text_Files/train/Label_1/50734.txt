import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[][] arr = new long[n][4];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 4; j++) {
                arr[i][j] = sc.nextLong();
            }
        }
        long[] dp = new long[n];
        for (int i = 0; i < n; i++) {
            dp[i] = 1;
        }
        for (int i = 0; i < n; i++) {
            int maxPos = -1;
            long max = 0;
            for (int j = 0; j < i; j++) {
                if (arr[j][2] < arr[i][0] && arr[j][3] < arr[i][1]) {
                    if (dp[j] > max) {
                        max = dp[j];
                        maxPos = j;
                    }
                }
            }
            if (maxPos != -1) {
                dp[i] = dp[maxPos] + 1;
            }
        }
        long ans = 0;
        for (long x : dp) {
            ans = Math.max(ans, x);
        }
        System.out.println(ans);
    }
}
