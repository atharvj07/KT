import java.util.Scanner;
import java.util.LinkedList;

class Main {
    private static final long MOD = 1000000007L;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        LinkedList<Edge> edges = new LinkedList<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            int c = sc.nextInt();
            edges.add(new Edge(a, b, c));
        }
        long ans = dfs(n, edges, 0, 0, new boolean[n], new boolean[n], new boolean[n]);
        System.out.println(ans);
    }

    private static long dfs(int n, LinkedList<Edge> edges, int a, int b, boolean[] va, boolean[] vb, boolean[] vc) {
        for (Edge e : edges) {
            if (va[e.a] && vb[e.b]) {
                if (vc[e.c] == e.w) {
                    continue;
                } else {
                    return 0;
                }
            }
        }
        if (a == n - 1 && b == n - 1) {
            return 1;
        }
        long ans = 0;
        va[a] = true;
        vb[b] = true;
        vc[a] = true;
        vc[b] = true;
        for (Edge e : edges) {
            if (va[e.a] && vb[e.b]) {
                if (vc[e.c]) {
                    continue;
                } else {
                    ans += dfs(n, edges, e.c, b, va, vb, vc);
                    ans %= MOD;
                }
            } else if (va[e.a] && vc[e.b]) {
                if (vb[e.c]) {
                    continue;
                } else {
                    ans += dfs(n, edges, e.c, b, va, vb, vc);
                    ans %= MOD;
                }
            } else if (vb[e.a] && vc[e.b]) {
                if (va[e.c]) {
                    continue;
                } else {
                    ans += dfs(n, edges, e.c, b, va, vb, vc);
                    ans %= MOD;
                }
            }
        }
        va[a] = false;
        vb[b] = false;
        vc[a] = false;
        vc[b] = false;
        return ans;
    }

    private static class Edge {
        int a;
        int b;
        int c;
        boolean w;

        public Edge(int a, int b, int c) {
            this.a = a;
            this.b = b;
            this.c = c;
            this.w = c == 1;
        }
    }
}
