import sys

def read_int():
    return int(sys.stdin.readline())

def read_ints():
    return [int(x) for x in sys.stdin.readline().split()]

n = read_int()
s = set(read_ints())
for x in range(32):  # 2^32 is the maximum possible size of the permutation
    mask = (1 << x) - 1
    if all((x ^ (x >> 1)) in s for x in range(1, mask)):
        print(x)
        print(*[0] + [x for x in range(1, mask) if (x ^ (x >> 1)) in s])
        break
