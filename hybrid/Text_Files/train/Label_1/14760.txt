#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll n;
    cin >> n;
    if (n == 1) {
        cout << "-1";
        return 0;
    }
    vector<ll> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    ll d = 0;
    for (int i = 0; i + 1 < n; i++) {
        ll nd = a[i + 1] - a[i];
        if (d == 0) {
            d = nd;
        } else if (d != nd) {
            cout << "0";
            return 0;
        }
    }
    ll l = a[0] - d, r = a[n - 1] + d;
    ll sum = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i];
    }
    if ((sum % n) % d == 0) {
        ll p = (sum / n) % d;
        cout << 1 << "\n"
             << p;
    } else {
        cout << 2 << "\n"
             << l << " " << r;
    }
    return 0;
}
