from collections import defaultdict
from heapq import *
def dijkstra(start, edges, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    while pq:
        _, u = heappop(pq)
        if u == n:
            return dist[n]
        for v, w in edges[u]:
            new_dist = dist[u] + w
            if dist[v] > new_dist:
                dist[v] = new_dist
                heappush(pq, (new_dist, v))

n, m, k = map(int, input().split())
a = set(map(int, input().split()))
edges = defaultdict(list)
for _ in range(m):
    x, y = map(int, input().split())
    edges[x].append((y, 1))
    edges[y].append((x, 1))
max_dist = 0
for i in a:
    for j in a:
        if i != j:
            if i not in edges[j]:
                edges[i].append((j, 1))
                edges[j].append((i, 1))
            dist = dijkstra(1, edges, n)
            max_dist = max(max_dist, dist)
            edges[i].pop()
            edges[j].pop()
print(max_dist)
