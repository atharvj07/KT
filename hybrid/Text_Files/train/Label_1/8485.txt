#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> h(n);
  for (int i = 0; i < n; i++) cin >> h[i];
  for (int i = n - 2; i >= 0; i--)
    while (h[i] + 2 <= h[i + 1]) h[i]++, h[i + 1]--;
  for (int i = 0; i < n; i++) cout << h[i] << ' ';
  return 0;
}
