#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> grid(n);
  for (auto &row : grid) cin >> row;
  int ans = 0;
  for (int j = 0; j < n; ++j) {
    int cnt = 0;
    bool allDirty = true;
    for (int i = 0; i < n; ++i) {
      cnt += grid[i][j] == '1';
      allDirty &= grid[i][j] == '0';
    }
    if (allDirty || cnt > n / 2) ans++;
  }
  cout << ans << '\n';
  return 0;
}
