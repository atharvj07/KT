import sys
for t in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    ans = 0
    while a and a[0] != a[-1]:
        b = a[0]
        for i in range(n):
            if a[i] != b:
                break
        a[i] += b
        a.pop(0)
        ans += 1
    print(ans)
