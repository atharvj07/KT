#include <iostream>
#include <vector>

using namespace std;

int n, m;
char s[1005][1005];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) cin >> s[i] + 1;
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= m; ++j) {
            if (s[i][j] == '.' && s[i - 1][j] == '*' && s[i][j - 1] == '*' && s[i][j + 1] == '*' && s[i + 1][j] == '*') {
                s[i][j] = '*';
                s[i - 1][j] = '.';
                s[i][j - 1] = '.';
                s[i][j + 1] = '.';
                s[i + 1][j] = '.';
            }
        }
    int ans = 0;
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= m; ++j)
            if (s[i][j] == '*') ans++;
    if (ans == n * m) {
        cout << ans << endl;
        for (int i = 1; i <= n; ++i)
            for (int j = 1; j <= m; ++j)
                if (s[i][j] == '*') cout << i << ' ' << j << ' ' << 1 << endl;
    } else
        cout << "-1\n";
    return 0;
}
