import sys
import array
input = sys.stdin.readline

def solve():
    n = int(input())
    a = list(map(int, input().split()))
    ans = [0]*n
    q = []
    for i in range(n):
        if a[i]==-1: q.append(i)
        else: q.insert(a[i]-1,i)
    for i in range(n):
        ans[i] = q.pop(0)+1
    print(*ans)

t = int(input())
for _ in range(t): solve()
