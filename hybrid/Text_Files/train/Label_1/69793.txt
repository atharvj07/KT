def evaluate(node, value=None):
    if nodes[node][0] == "IN":
        if value is not None:
            return value
        return nodes[node][1]
    if nodes[node][0] == "NOT":
        return 1 - evaluate(nodes[node][1])
    val1 = evaluate(nodes[node][1])
    val2 = evaluate(nodes[node][2])
    if nodes[node][0] == "AND":
        return val1 & val2
    if nodes[node][0] == "OR":
        return val1 | val2
    return val1 ^ val2

def dfs(node, influence):
    if nodes[node][0] == "IN":
        influences[node] = influence
        return
    if nodes[node][0] == "NOT":
        dfs(nodes[node][1], influence)
        return
    val1 = evaluate(nodes[node][1])
    val2 = evaluate(nodes[node][2])
    if nodes[node][0] == "AND":
        dfs(nodes[node][1], influence & val2)
        dfs(nodes[node][2], influence & val1)
    elif nodes[node][0] == "OR":
        dfs(nodes[node][1], influence | val2)
        dfs(nodes[node][2], influence | val1)
    else:  # XOR
        dfs(nodes[node][1], influence)
        dfs(nodes[node][2], influence)

n = int(input().strip())
nodes = {}
inputs = []
for i in range(1, n + 1):
    data = input().strip().split()
    if data[0] == "IN":
        nodes[i] = (data[0], int(data[1]))
        inputs.append(i)
    else:
        nodes[i] = (data[0], int(data[1]), int(data[2]))

original_output = evaluate(1)
influences = {}
dfs(1, 1)

result = []
for inp in inputs:
    if original_output == evaluate(1, 1 - nodes[inp][1]):
        result.append('0' if influences[inp] else '1')
    else:
        result.append('1' if influences[inp] else '0')

print("".join(result))
