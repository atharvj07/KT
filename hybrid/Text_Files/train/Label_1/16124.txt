#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int mod = 1e9 + 7;

ll n;
vector<ll> a;

ll ans() {
  ll ans = a[0];
  bool f = true;
  for (ll i = 1; i < n; i++) {
    if (f)
      ans += a[i];
    else
      ans -= a[i];
    ans = (ans % mod + mod) % mod;
    f = !f;
  }
  return ans;
}

int main() {
  cin >> n;
  a.resize(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  cout << ans() << '\n';
  return 0;
}
