import java.util.*;
public class Main {
   public static void main(String[] args) {
     Scanner input = new Scanner(System.in);
     int a = input.nextInt();
     int b = input.nextInt();
     int c = input.nextInt();
     int d = input.nextInt();
     input.close();
     List<Integer> list = Arrays.asList(a,b,c,d);
     Collections.sort(list);
     int min = list.get(0);
     int max = list.get(3);
     int[] arr = new int[4];
     arr[0] = list.get(1);
     arr[1] = list.get(2);
     arr[2] = max;
     arr[3] = min;
     boolean isPossible = false, isSegment = false;
     for(int i=0; i<4; i++){
        if(arr[i]+arr[(i+1)%4]+arr[(i+2)%4]>max) isPossible = true;
        if(arr[i]+arr[(i+1)%4]>max) isSegment = true;
     }
     if(isPossible) System.out.println("TRIANGLE");
     else if(isSegment) System.out.println("SEGMENT");
     else System.out.println("IMPOSSIBLE");
   }
}
