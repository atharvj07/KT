n = int(input().strip())
cows = list(map(int, input().strip().split()))

left_facing = cows.count(0)
right_facing = 0
lost_milk = 0

for cow in cows:
    if cow == 0:  # cow facing left
        left_facing -= 1
    else:  # cow facing right
        lost_milk += left_facing
        right_facing += 1

print(lost_milk)
