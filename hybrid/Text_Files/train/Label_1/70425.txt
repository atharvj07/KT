from collections import deque

s = input()
n = len(s)
a = [0] * n
st = deque()
for i in range(n):
    if s[i] == '(':
        st.append(i)
    else:
        if st:
            a[st.pop()] = a[i] = 1
m = int(input())
for _ in range(m):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    print(max(a[l:r + 1]))
