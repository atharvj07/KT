#include <iostream>
using namespace std;
int main() {
  int v, ans = 0;
  cin >> v;
  if (v % 4 == 0) {
    if (v <= 99)
      ans = 2;
  }
  if (v % 4 == 2) { ans = 0; }
  if (v % 4 == 3) { ans = 1; }
  char cat;
  switch (v % 4 + ans) {
  case 0:
    cat = 'D';
    break;
  case 1:
    cat = 'A';
    break;
  case 2:
    cat = 'C';
    break;
  case 3:
    cat = 'B';
    break;
  }
  cout << ans << " " << cat << endl;
  return 0;
}
