n, a, b = map(int, input().split())
if n == 1:
    print("YES\n0")
elif a == 1:
    print("NO")
elif a == n:
    for i in range(n):
        print("0" * (i + 1) + "1" * (n - i - 1))
else:
    for j in range(1, n):
        if not (n - j - b <= a <= n - j):
            print("NO")
            exit()
        print("0" * j + "1" * (a - n + j) + "0" * (b - a + n - j) + "1" * (n - b - j))
