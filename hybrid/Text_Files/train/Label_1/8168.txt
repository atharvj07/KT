#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
int n,m,w,a[N],dp[N];
int main(){
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);
    cin>>n>>m>>w;
    for(int i=1;i<=n;i++) cin>>a[i];
    sort(a+1,a+n+1);
    for(int i=1;i<=n;i++){
        dp[i]=max(dp[i],dp[max(i-w+1,0)]+a[i]);
        dp[i]=max(dp[i],a[i]);
    }
    cout<<dp[n]<<endl;
}
