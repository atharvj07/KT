import sys


n = int(input())
a = list(map(int, input().split()))

inc = []
dec = []
for i in range(n):
    if not inc or a[i] >= inc[-1]:
        inc.append(a[i])
    elif not dec or a[i] <= dec[-1]:
        dec.append(a[i])
    else:
        print("NO")
        sys.exit()

res = [0] * n
for i in inc:
    res[a.index(i)] = 0
    a.remove(i)
for i in dec:
    res[a.index(i)] = 1
    a.remove(i)

print("YES")
print(*res)
