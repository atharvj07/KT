#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    int ans = 0;
    for (int i = 1, j = n; i <= j;) {
        int mi = min(a[i], a[j]);
        a[i] -= mi, a[j] -= mi;
        if (!a[i]) i++;
        if (!a[j]) j--;
        ans++;
    }
    cout << ans - 1 << endl;
    return 0;
}
