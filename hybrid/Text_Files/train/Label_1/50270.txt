import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[m];
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt() - 1;
            b[i] = sc.nextInt() - 1;
        }
        int[] ans = new int[n];
        for (int start = 0; start < n; start++) {
            int time = 0;
            int cur = start;
            boolean[] taken = new boolean[m];
            for (int i = 0; i < m; i++) {
                if (taken[i]) continue;
                while (cur != a[i]) {
                    cur = (cur + 1) % n;
                    time++;
                }
                taken[i] = true;
                while (cur != b[i]) {
                    cur = (cur + 1) % n;
                    time++;
                }
            }
            ans[start] = time;
        }
        for (int x : ans) System.out.print(x + " ");
    }
}
