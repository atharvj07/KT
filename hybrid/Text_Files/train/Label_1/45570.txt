import java.util.*;
import java.io.*;

class Main {
  static int n, m, s, b, k;
  static ArrayList<ArrayList<Integer>> adj;
  static int[] shipX, shipA, shipF, shipP;
  static int[] baseX, baseD, baseG;
  static boolean[] taken;
  static int[][] dp;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    m = sc.nextInt();
    adj = new ArrayList<>();
    for (int i = 0; i <= n; i++) adj.add(new ArrayList<>());
    for (int i = 0; i < m; i++) {
      int u = sc.nextInt();
      int v = sc.nextInt();
      adj.get(u).add(v);
      adj.get(v).add(u);
    }
    s = sc.nextInt();
    b = sc.nextInt();
    k = sc.nextInt();
    shipX = new int[s + 1];
    shipA = new int[s + 1];
    shipF = new int[s + 1];
    shipP = new int[s + 1];
    taken = new boolean[s + 1];
    for (int i = 1; i <= s; i++) {
      shipX[i] = sc.nextInt();
      shipA[i] = sc.nextInt();
      shipF[i] = sc.nextInt();
      shipP[i] = sc.nextInt();
    }
    baseX = new int[b + 1];
    baseD = new int[b + 1];
    baseG = new int[b + 1];
    for (int i = 1; i <= b; i++) {
      baseX[i] = sc.nextInt();
      baseD[i] = sc.nextInt();
      baseG[i] = sc.nextInt();
    }
    int[] dep = new int[k + 1];
    for (int i = 1; i <= k; i++) dep[i] = sc.nextInt();
    Arrays.sort(dep, 1, k + 1);
    for (int i = 1; i <= k; i++) {
      if (!taken[dep[i]]) {
        dfs(dep[i]);
      }
    }
    dp = new int[s + 1][1 << s];
    for (int i = 0; i <= s; i++) {
      for (int j = 0; j < (1 << s); j++) {
        dp[i][j] = -1;
      }
    }
    System.out.println(solve(s, 0));
  }

  public static int bfs(int a, int b) {
    Queue<Integer> q = new LinkedList<>();
    int[] dist = new int[n + 1];
    Arrays.fill(dist, -1);
    q.offer(a);
    dist[a] = 0;
    while (!q.isEmpty()) {
      int cur = q.poll();
      if (cur == b) return dist[cur];
      for (int next : adj.get(cur)) {
        if (dist[next] == -1) {
          dist[next] = dist[cur] + 1;
          q.offer(next);
        }
      }
    }
    return -1;
  }

  public static void dfs(int v) {
    taken[v] = true;
    for (int next : adj.get(shipX[v])) {
      if (!taken[next]) {
        dfs(next);
      }
    }
  }

  public static int solve(int idx, int mask) {
    if (idx == 0) return 0;
    if (dp[idx][mask] != -1) return dp[idx][mask];
    int ans = solve(idx - 1, mask);
    for (int i = 1; i <= b; i++) {
      if (((mask >> (i - 1)) & 1) == 0 && shipA[idx] >= baseD[i] && shipF[idx] >= bfs(shipX[idx], baseX[i])) {
        ans = Math.max(ans, solve(idx - 1, mask | (1 << (i - 1))) + baseG[i] - shipP[idx]);
      }
    }
    return dp[idx][mask] = ans;
  }
}
