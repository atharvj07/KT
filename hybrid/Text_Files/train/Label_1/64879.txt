from collections import deque
for _ in range(int(input())):
    n = int(input())
    s = str(input())
    q = deque([s[0]])
    for c in s[1:]:
        if c > q[-1]:
            q.append(c)
    if len(q) == 1:
        print("NO")
    else:
        print("YES")
        print(len(q))
        print(*q)
