#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) { std::cin >> a[i]; }

    std::vector<int> b(n);
    std::vector<int> mex(n + 1, -1);
    int k = 0;
    for (int i = 0; i < n; i++) {
      if (mex[a[i]] == -1) { mex[a[i]] = k++; }
      b[i] = mex[a[i]];
    }

    for (int i = 0; i < n; i++) { std::cout << b[i] << ' '; }
    std::cout << '\n';
  }
}
