n, m = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(m)]

# Checking if it's possible - it's only possible if the graph is a tree or has exactly n edges (forming a cycle)
if m != n - 1 and m != n:
    print("NO")
else:
    print("YES")
    for edge in edges:
        # Printing 0 if the first vertex is smaller than the second, 1 otherwise
        print(0 if edge[0] < edge[1] else 1, end="")
    print()
