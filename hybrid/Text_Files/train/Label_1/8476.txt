#include <bits/stdc++.h>
using namespace std;
#define nl '\n'

int a[100005];
int b[200005];
int p[100005];
int main() {
    int n, m;
    cin >> m;
    int sz = 0;
    for (int i = 1; i <= m; i++) {
        int t, x, y, z;
        cin >> t;
        if (t == 1) {
            cin >> x;
            a[++sz] = x;
            p[sz] = sz;
            b[sz] = 1;
        } else {
            cin >> x >> y;
            for (int j = 1; j <= x; j++) {
                a[++sz] = a[p[sz - j + 1]];
                p[sz] = p[sz - j + 1];
                b[sz] = b[sz - j + 1];
            }
            for (int j = 1; j <= x; j++) b[sz] *= y;
            p[sz] = p[sz - x + 1];
        }
    }
    cin >> n;
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        cout << a[x] << " ";
    }
    return 0;
}
