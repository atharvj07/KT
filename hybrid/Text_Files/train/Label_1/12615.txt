#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  pair<int, int> segments[n];
  for (int i = 0; i < n; i++) {
    cin >> segments[i].first >> segments[i].second;
  }
  sort(segments, segments + n);
  int l = segments[0].first, r = segments[0].second;
  bool ans = true;
  for (int i = 1; i < n; i++) {
    if (segments[i].first <= r) {
      r = max(r, segments[i].second);
    } else {
      ans = false;
    }
  }
  cout << (ans ? 1 : -1) << endl;
  return 0;
}
