import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Map<String, Integer> scores = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String name = scanner.next();
            int score = scanner.nextInt();
            scores.put(name, scores.getOrDefault(name, 0) + score);
        }
        List<String> names = new ArrayList<>(scores.keySet());
        Collections.sort(names, (a, b) -> scores.get(b) - scores.get(a));
        int m = names.size();
        System.out.println(m);
        for (String name : names) {
            int rank = (int) Math.ceil(100.0 * names.indexOf(name) / m);
            String category = "pro";
            if (rank > 1) {
                category = "hardcore";
                if (rank > 10) {
                    category = "average";
                    if (rank > 20) {
                        category = "random";
                        if (rank > 50) {
                            category = "noob";
                        }
                    }
                }
            }
            System.out.println(name + " " + category);
        }
    }
}
