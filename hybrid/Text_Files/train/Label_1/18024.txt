#include <bits/stdc++.h>
using namespace std;
#define int long long
const int maxn = 5000 + 4;
array<int, maxn> a, c, cc;
signed main() {
    int n, m, d, x;
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) { cin >> a[i], cc[a[i]]++; }
    for (int i = 1; i <= n; ++i) { cin >> c[i], cc[c[i]]++; }
    cin >> d;
    while (d--) {
        cin >> x, c[x] = 0, cc[a[x]]--;
        int mx = 0;
        for (int i = 1; i <= n; ++i) { if (cc[i]) mx = i; }
        cout << mx << endl;
    }
}
