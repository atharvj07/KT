#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> sherlock(n);
  vector<int> moriarty(n);
  for (int i = 0; i < n; i++) { cin >> sherlock[i]; }
  for (int i = 0; i < n; i++) { cin >> moriarty[i]; }

  vector<int> sherlockCopy = sherlock;
  int moriartyFlicksMin = 0;
  for (int i = 0; i < n; i++) {
    if (sherlock[i] > moriarty[i]) {
      moriartyFlicksMin++;
      int sherlockDigitIndex = -1;
      for (int j = i + 1; j < n; j++) {
        if (sherlock[j] < sherlock[i]) {
          sherlockDigitIndex = j;
          break;
        }
      }
      if (sherlockDigitIndex != -1) {
        sherlock[sherlockDigitIndex] = sherlock[i];
        sherlock[i] = moriarty[i];
      }
    }
  }

  vector<int> sherlockCopy2 = sherlockCopy;
  int moriartyFlicksMax = 0;
  for (int i = 0; i < n; i++) {
    if (sherlockCopy2[i] < moriarty[i]) {
      moriartyFlicksMax++;
      int moriartyDigitIndex = -1;
      for (int j = i + 1; j < n; j++) {
        if (moriarty[j] > moriarty[i]) {
          moriartyDigitIndex = j;
          break;
        }
      }
      if (moriartyDigitIndex != -1) {
        moriarty[moriartyDigitIndex] = moriarty[i];
        moriarty[i] = sherlockCopy2[i];
      }
    }
  }

  cout << moriartyFlicksMin << endl;
  cout << moriartyFlicksMax << endl;

  return 0;
}
