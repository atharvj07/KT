def solve(n, m):
    if n < m:
        a, b = n, m
        transpose = True
    else:
        a, b = m, n
        transpose = False

    res = [[''] * b for _ in range(a)]
    color = ord('A')

    for i in range(a):
        for j in range(b):
            if res[i][j]:
                continue
            if i + 1 < a and j + 1 < b and not res[i + 1][j] and not res[i][j + 1] and not res[i + 1][j + 1]:
                res[i][j] = res[i + 1][j] = res[i][j + 1] = res[i + 1][j + 1] = chr(color)
                color += 1
            else:
                res[i][j] = chr(color)
                color += 1
            if color > ord('Z'):
                color = ord('A')

    if transpose:
        res = list(zip(*res))

    for row in res:
        print(''.join(row))

n, m = map(int, input().split())
solve(n, m)
