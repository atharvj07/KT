#include <bits/stdc++.h>
using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);

  int n;
  cin >> n;
  string a, b;
  cin >> a >> b;
  int ans = 0, flip = INT_MAX;
  for (int i = 0; i < n; i++) {
    if (a[i] != b[i]) {
      if (a[i] == '1') flip = min(flip, i);
      ans++;
    }
  }
  if (flip != INT_MAX) ans = min(ans, (n - 1 - flip) + 1);
  cout << ans << endl;
  return 0;
}
