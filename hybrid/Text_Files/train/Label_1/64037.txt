def minDays(monsters, heroes):
    monsters.sort()
    heroes.sort(key = lambda x: x[0])
    used = [False] * len(heroes)
    cur = 0
    days = 0
    for monster in monsters:
      found = False
      for i in range(cur, len(heroes)):
        if not used[i] and heroes[i][0] >= monster:
          if heroes[i][1] == 1:
            used[i] = True
          heroes[i][1] -= 1
          found = True
          cur = i + 1
          break
      if not found:
        return -1
      days += 1
    return days
        

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    m = int(input())
    heroes = []
    for _ in range(m):
      p, s = list(map(int, input().split()))
      heroes.append([p, s])
    print(minDays(a, heroes))
