#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (n); ++i)
using namespace std;
using ll = long long;

int main() {
  int n, m;
  cin >> n >> m;
  vector<array<int, 2>> friends(n);
  rep(i, n) {
    int f;
    cin >> f;
    friends[i][0] = f;
    rep(j, f) { int b;
      cin >> b;
      friends[i][1] |= (1 << (b - 1)); }
  }
  vector<array<int, 10>> pizzas(m);
  rep(i, m) {
    cin >> pizzas[i][0];
    int r;
    cin >> r;
    rep(j, r) { int a;
      cin >> a;
      pizzas[i][a] = 1; }
  }
  int ans = 0, ans_pizza[2];
  rep(i, m) { rep(j, i) {
      int cnt = __builtin_popcount(friends[0][1] & (pizzas[i][1] | pizzas[j][1]));
      rep(k, n) {
        if ((cnt < friends[k][0]) && ((pizzas[i][1] | pizzas[j][1]) & friends[k][1]))
          cnt++;
      }
      if (cnt > ans) {
        ans = cnt;
        ans_pizza[0] = i + 1;
        ans_pizza[1] = j + 1;
      }
    } }
  cout << ans_pizza[0] << ' ' << ans_pizza[1] << '\n';
}
