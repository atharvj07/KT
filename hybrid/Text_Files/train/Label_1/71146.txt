s = input()
if s == '0': print(1); exit()
if len(s) != 8:
    if s == '0_25': print(0); exit()
    x = s.count('X')
    print(x * (x-1) * 10 ** (4-x))
else:
    l = []
    for c in s:
        if c == '_': l.append(5)
        elif c == 'X': l.append(10)
        else: l.append(int(c))
    mul = 1
    ans = 0
    for i in l[::-1]:
        for j in range(i):
            ans += mul * j
        mul *= i
    print(ans)
