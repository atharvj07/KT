from sys import stdin

input = stdin.readline
t = int(input())

for _ in range(t):
    n = int(input())
    a = [int(x) for x in input().strip().split()]
    ans = 0
    st = [a[0]]
    for x in a[1:]:
      if x >= st[-1]:
        st.append(x)
      else:
        ans += 1
        st.insert(0, x)
    print(ans)
