#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    vector<vector<int>> dp(n + 1, vector<int>(k + 1, 0));
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= k; j++) {
            dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - 1] + a[i - 1]);
        }
    }

    cout << dp[n][k] << endl;

    vector<int> ans;
    int i = n, j = k;
    while (i > 0 && j > 0) {
        if (dp[i][j] == dp[i - 1][j]) {
            i--;
        } else {
            ans.push_back(i);
            i--;
            j--;
        }
    }

    reverse(ans.begin(), ans.end());

    for (int x : ans) cout << x - (find(a.begin(), a.end(), x) - a.begin()) + 1 << " ";
    cout << endl;

    return 0;
}
