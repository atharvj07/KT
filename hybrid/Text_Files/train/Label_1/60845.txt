MOD = 998244353

def comb_mod(n, k, mod):
    if k < 0 or k > n:
        return 0
    p = 1
    for i in range(1, min(k, n - k) + 1):
        p = p * (n - i + 1) * pow(i, mod - 2, mod) % mod
    return p

def solve(n, k, arr):
    arr.sort()
    answer = 0
    for i in range(n):
        for j in range(i + 1, n):
            diff = arr[j] - arr[i]
            count = comb_mod(j - i - 1, k - 2, MOD)
            answer = (answer + diff * count) % MOD
    return answer

n, k = map(int, input().split())
arr = list(map(int, input().split()))
print(solve(n, k, arr))
