#include <iostream>
#include <map>
#include <queue>
#include <vector>

using namespace std;

typedef long long ll;

int main() {
  int T;
  cin >> T;
  while (T--) {
    int n;
    cin >> n;
    vector<int> p(2 * n);
    for (int i = 0; i < 2 * n; i++) {
      cin >> p[i];
    }

    map<int, int> cnt;
    for (const int x : p) { cnt[x]++; }

    int ans = 1;
    for (const auto& [val, num] : cnt) {
      if (num > 1) ans = 0;
    }

    if (ans)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }

  return 0;
}
