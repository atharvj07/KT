import sys

def main():
    n = int(input())
    x1, y1, x2, y2 = [0] * 4
    xl = []
    xr = []
    yl = []
    yu = []
    for i in range(n):
        x1, y1, x2, y2 = map(int, input().split())
        xl.append(x1)
        xr.append(x2)
        yl.append(y1)
        yu.append(y2)
    xl.sort()
    xr.sort()
    yl.sort()
    yu.sort()
    print(xl[n//2], yl[n//2])

main()
