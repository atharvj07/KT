#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
typedef string str;

int main() {
    ll n;
    cin >> n;
    str s;
    cin >> s;
    ll ans = 0;
    for (ll i = 0; i+1 < n; i+=2) {
        if (s[i] == s[i+1]) {
            ++ans;
            s.erase(i+1, 1);
            --i;
            --n;
        }
    }
    cout << ans << endl;
    cout << s;
    return 0;
}
