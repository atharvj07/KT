import java.util.*;
import java.io.*;

public class Main{
  public static void main(String[] args) {
    try(BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
      String[] ln = br.readLine().split(" ");
      int N = Integer.parseInt(ln[0]);
      int S = Integer.parseInt(ln[1]);
      int[] A = new int[N];
      ln = br.readLine().split(" ");
      for(int i = 0; i < N; i++) {
        A[i] = Integer.parseInt(ln[i]);
      }
      long cnt = 0;
      for(int i = 0; i < N; i++) {
        int j = i-1, sum = 0;
        int mod = (int)1e9 + 7;
        while(j >= 0) {
          if(sum <= S) {
            if(sum + A[j] > S) {
              sum += A[j]; break;
            }
            sum += A[j];
            cnt += 1;
          }
          else break;
          j--;
        }
      }
      System.out.println(cnt);
    }
    catch(Exception e){
      e.printStackTrace();
    }
  }
}
