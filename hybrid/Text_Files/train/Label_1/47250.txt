import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }

        int buy = -1;
        int sell = -1;
        int profit = 0;
        for (int i = 0; i < n; i++) {
            if (buy == -1 || p[i] < p[buy]) {
                buy = i;
            } else if (sell == -1 || p[i] > p[sell]) {
                sell = i;
                profit = Math.max(profit, p[sell] - p[buy]);
            }
        }

        System.out.println(profit);
    }
}
