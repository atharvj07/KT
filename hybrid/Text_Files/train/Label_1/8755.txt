#include <bits/stdc++.h>
#define int int64_t
using namespace std;
const int inf = 0x3f3f3f3f3f3f3f3f;

vector<int> adj[200005];
int ind[200005], ord[200005], type[200005];
int in[2], out[2];
int dfs1(int u, int t) {
  ind[u] = ++*in;
  type[u] = t;
  for (int v : adj[u])
    if (ind[v] == 0) dfs1(v, t);
  ord[u] = ++*out;
}

void dfs2(int u, int c) {
  for (int v : adj[u])
    if (type[v] != type[u])
      c = 0;
    else if (ind[v] < ind[u] && ord[v] > ord[u]) dfs2(v, c);
  if (c)
    out[type[u]] = 0;
}

int32_t main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++) cin >> type[i];
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    adj[b].push_back(a);
  }
  in[0] = in[1] = out[0] = out[1] = 0;
  for (int i = 0; i < n; i++)
    if (ind[i] == 0) dfs1(i, i);
  for (int i = 0; i < n; i++)
    dfs2(i, 1);
  cout << out[0] + out[1] << endl;
}
