#include <bits/stdc++.h>
using namespace std;
const int MAXN = 3e5 + 5;
int n, x, y;
vector<int> adj[MAXN];
bool vis[MAXN], vis2[MAXN];
int dfs(int u, int p = -1) {
  int ret = 1;
  vis[u] = true;
  for (int v : adj[u]) {
    if (v == p) continue;
    if (vis[v]) {
      vis2[u] = true;
      continue;
    }
    ret += dfs(v, u);
  }
  return ret;
}
int dfs2(int u, int p = -1) {
  int ret = 1;
  vis[u] = true;
  for (int v : adj[u]) {
    if (v == p) continue;
    if (!vis[v]) {
      ret += dfs2(v, u);
      continue;
    }
    if (!vis2[v]) ret += dfs2(v, u);
  }
  return ret;
}
int main() {
  cin >> n >> x >> y;
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs(x);
  memset(vis, false, sizeof vis);
  cout << dfs2(y) - 1 << endl;
  return 0;
}
