#include <bits/stdc++.h>
using namespace std;

const int MAXN = 5e5 + 10;
const int dx[] = {0, 0, 1, -1};
const int dy[] = {1, -1, 0, 0};
int n;
vector<pair<int, int>>r[MAXN];

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cin >> n;
    vector<pair<pair<int, int>, pair<int, int>>> v;
    for (int i = 0; i < n; i++) {
        int x1, y1, x2, y2; cin >> x1 >> y1 >> x2 >> y2;
        v.push_back({{x1, y1}, {x2, y2}});
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            if (v[i].first.first > v[j].second.first or v[i].first.second > v[j].second.second or
                v[i].second.first < v[j].first.first or v[i].second.second < v[j].first.second) continue;
            r[i].push_back({j, 0});
            r[j].push_back({i, 0});
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            if (v[i].first.first >= v[j].second.first or v[i].first.second >= v[j].second.second or
                v[i].second.first <= v[j].first.first or v[i].second.second <= v[j].first.second) continue;
            r[i].push_back({j, 1});
            r[j].push_back({i, 1});
        }
    }
    vector<int> color(n, 0);
    queue<int> q;
    q.push(0);
    color[0] = 1;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (auto x: r[u]) {
            int v = x.first, t = x.second;
            if (!color[v]) {
                color[v] = color[u] + !t;
                if (color[v] == color[u] and t == 0) {
                    cout << "NO" << endl;
                    return 0;
                }
                q.push(v);
            } else {
                if (color[v] == color[u] and t == 0) {
                    cout << "NO" << endl;
                    return 0;
                }
                if (color[v] != color[u] + !t) {
                    cout << "NO" << endl;
                    return 0;
                }
            }
        }
    }
    cout << "YES" << endl;
    for (int i = 0; i < n; i++) {
        cout << color[i] << endl;
    }
    return 0;
}
