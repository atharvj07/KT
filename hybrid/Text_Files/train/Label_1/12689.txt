#include <bits/stdc++.h>
using namespace std;
int n,q,k;
vector<int> a[2];
int main(){
	cin >> n;
	a[0].resize(n);
	a[1].resize(n);
	for (int i=0;i<2;i++) for (int j=0;j<n;j++) cin >> a[i][j];
	cin >> k;
	if (n<k||a[0][0]!=a[1][0]||a[0][n-1]!=a[1][k-1]){
		cout << "NO";
		return 0;
	}
	vector<int> w(n+1);
	int j=0;
	for (int i=n-1;i>=0;i--){
		while (a[0][i]<a[1][j]) j++;
		if (a[0][i]!=a[1][j]) break;
		w[i]=w[i+1]+j-i;
	}
	if (w[0]>0){
		cout << "NO";
		return 0;
	}
	vector<pair<int,int>> v;
	for (int i=0,j=0;i<k;i++){
		while (a[0][j]<a[1][i]) j++;
		if (a[0][j]==a[1][i]){
			if (w[j+1]>0){
				cout << "NO";
				return 0;
			}
			for (int l=j;l<n;l++) w[l]-=j-l;
			j++;
			if (j!=n){
				if (a[0][j-1]==a[0][j]){
					if (w[j]!=0){
						cout << "NO";
						return 0;
					}
					v.emplace_back(j,1);
					for (int l=j;l<n;l++) w[l]-=j-l;
				}
				else v.emplace_back(j,-1);
			}
		}
	}
	cout << "YES" << endl;
	for (auto x:v) cout << x.first << ' ' << (x.second==1?"L":"R") << endl;
	return 0;
}
