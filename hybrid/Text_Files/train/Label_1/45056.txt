import java.io.*;
import java.util.*;

class Main {

    static int[] parent;
    static int[] degree;
    static int[] ans;
    static int a;

    static void dfs(int u, int p) {
        degree[u] = 1;
        parent[u] = p;
        for (int v : edges[u]) {
            if (v == p) continue;
            dfs(v, u);
            degree[u] += degree[v];
        }
    }

    static void solve() {
        int n = nextInt();
        edges = new ArrayList[n + 1];
        parent = new int[n + 1];
        degree = new int[n + 1];
        ans = new int[n + 1];
        for (int i = 0; i <= n; i++) edges[i] = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            int p = nextInt();
            edges[i].add(p);
            edges[p].add(i);
        }
        dfs(1, 1);
        boolean possible = true;
        for (int i = 1; i <= n; i++) {
            if (degree[i] % 2 == 0) continue;
            if (degree[i] == 1 && parent[i] != 1) continue;
            possible = false;
        }
        if (!possible) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        for (int i = 1; i <= n; i++) {
            if (degree[i] % 2 == 0) {
                ans[a++] = i;
                remove(i, parent[i]);
            }
        }
        for (int i = 1; i <= n; i++) {
            if (degree[i] == 1 && parent[i] != 1) {
                ans[a++] = i;
                remove(i, parent[i]);
            }
        }
        for (int i = 1; i <= n; i++) {
            if (degree[i] != 0) ans[a++] = i;
        }
        for (int i = 0; i < n - 1; i++) {
            System.out.println(ans[i]);
        }
    }

    static void remove(int u, int p) {
        degree[u] = 0;
        edges[u].remove((Integer) p);
        edges[p].remove((Integer) u);
    }

    static List<Integer>[] edges;

    static int nextInt() {
        return sc.nextInt();
    }

    static long nextLong() {
        return sc.nextLong();
    }

    static double nextDouble() {
        return sc.nextDouble();
    }

    static String next() {
        return sc.next();
    }

    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        solve();
    }

}
