#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const ll INF = LONG_LONG_MAX;

ll a, b, c, n;
vector<ll> x;
vector<ll> dp;

int main() {
  cin >> a >> b >> c >> n;
  x.resize(n + 1);
  dp.resize(n + 1);
  for (int i = 1; i <= n; i++) cin >> x[i];

  for (int i = 1; i <= n; i++) {
    dp[i] = max({dp[i - 1], dp[i], (x[i] >= b && x[i] <= c ? 0 : x[i])});
  }
  cout << dp[n] << endl;
  return 0;
}
