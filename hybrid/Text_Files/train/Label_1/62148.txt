import sys

x, d = map(int, sys.stdin.readline().strip().split())
if d >= 3:
    print(1)
    print(d)
    sys.exit(0)
if d == 2:
    print(2)
    print(2, 3)
    sys.exit(0)
if d == 1:
    if x == 2:
        print(2)
        print(1, 2)
        sys.exit(0)
    if x > 2:
        print(2)
        print(1, 10**18)
        sys.exit(0)
    if x == 1:
        print(-1)
        sys.exit(0)
if d == 0:
    if x > 1:
        print(2)
        print(1, 10**18)
        sys.exit(0)
    if x == 1:
        print(1)
        print(1)
        sys.exit(0)
