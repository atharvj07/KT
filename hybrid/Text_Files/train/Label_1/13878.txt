#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int k = 0, cnt = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] < 0) cnt++;
        if (i == n - 1 || cnt >= 3 || (a[i] >= 0 && cnt > 0)) {
            cout << k + 1 << " ";
            k++;
            cnt = 0;
        }
    }
    cout << endl;
    cnt = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] < 0) cnt++;
        if (i == n - 1 || cnt >= 3 || (a[i] >= 0 && cnt > 0)) {
            cout << i - cnt + 1 << " ";
            k++;
            cnt = 0;
        }
    }
    cout << endl;
    return 0;
}
