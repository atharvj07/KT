#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const long long MOD = 1000000007;

int main() {
    int n;
    cin >> n;
    vector<long long> x(n);
    for (long long& xi : x) cin >> xi;
    sort(x.begin(), x.end());
    long long sum = 0;
    for (int i = 0; i < n - 1; ++i) {
        sum = (sum + (x[i + 1] - x[i]) * (n - i - 1)) % MOD;
    }
    cout << sum << endl;
    return 0;
}
