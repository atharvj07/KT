import copy, sys
input = sys.stdin.readline

r, c = map(int, input().split())
a = [list(map(int, input().split())) for i in range(r)]

Max = 0
for i in range(r):
    for j in range(c):
        Max = max(Max, a[i][j])

Max += 1
dp = copy.deepcopy(a)
for i in range(r):
    for j in range(c):
        if a[i][j] == 0: a[i][j] = Max

for i in range(r):
    for j in range(1, c):
        if a[i][j-1] >= a[i][j]:
            print(-1); quit()

for j in range(c):
    for i in range(1, r):
        if a[i-1][j] >= a[i][j]:
            print(-1); quit()

for i in range(r):
    for j in range(c):
        dp[i][j] = a[i][j]
        if i > 0: dp[i][j] = max(dp[i][j], dp[i-1][j]+1)
        if j > 0: dp[i][j] = max(dp[i][j], dp[i][j-1]+1)

ans = 0
for i in range(r):
    for j in range(c):
        ans += dp[i][j]

print(ans)
