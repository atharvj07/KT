import java.util.Arrays;
import java.util.Scanner;

public class TeaParty {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int w = scanner.nextInt();
        int[] a = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            a[i] = scanner.nextInt();
        }
        scanner.close();

        Arrays.sort(a);

        double maxWater = 0;
        int girlIndex = 0;
        int boyIndex = n - 1;
        while (girlIndex < n && boyIndex >= 0) {
            if (a[girlIndex] * 2 <= a[boyIndex]) {
                maxWater += a[girlIndex] * 2;
                girlIndex++;
                boyIndex--;
            } else {
                maxWater += a[boyIndex];
                boyIndex--;
            }
        }

        double waterPerGirl = maxWater / n;
        if (waterPerGirl * 2 <= w) {
            maxWater += waterPerGirl * n;
        } else {
            maxWater += w / 2.0 * n;
        }

        System.out.println(maxWater);
    }
}
