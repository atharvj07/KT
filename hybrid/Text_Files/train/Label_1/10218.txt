#include <bits/stdc++.h>

using namespace std;

int n, m, k;

int d[1001];
int s[1001];

int main() {
  cin >> m >> k;
  for (int i = 1; i <= m; ++i) { cin >> d[i]; }
  for (int i = 1; i <= m; ++i) { cin >> s[i]; }

  int cnt = 0;
  int t = 0;

  for (int i = 1; i <= m;) {
    if (cnt + d[i] > s[i]) {
      t += k;
      cnt = s[i];
    }
    t += d[i];
    cnt -= d[i];
    cnt += s[++i];
  }

  cout << t << endl;

  return 0;
}
