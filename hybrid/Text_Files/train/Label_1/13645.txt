#include <iostream>
#include <climits>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int mn = INT_MAX;
    int cur = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '-') cur--;
        else cur++;
        mn = min(mn, cur);
    }
    cout << mn << endl;
    return 0;
}
