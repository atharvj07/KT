#include <iostream>

using std::cin;
using std::cout;

int main() {
  int n, k;
  cin >> n >> k;

  int cnt = 0;
  for (int i = 0; i < n; i++) {
    int num;
    cin >> num;

    int lucky = 0;
    while (num > 0) {
      int x = num % 10;
      lucky += (x == 4 || x == 7);
      num /= 10;
    }

    cnt += (lucky <= k);
  }

  cout << cnt << '\n';

  return 0;
}
