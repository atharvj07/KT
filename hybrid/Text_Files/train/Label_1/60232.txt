import math

def solve():
    t = int(input())
    for _ in range(t):
        n, k = map(int, input().split())
        a = list(map(int, input().split()))
        ans = math.inf
        for i in range(n):
            cur = (k - a[i] % k) % k
            ans = min(ans, cur + i)
        print(ans)

solve()
