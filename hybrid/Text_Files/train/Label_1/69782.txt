import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = [list(input().rstrip()) for i in range(n)]
b = [list(input().rstrip()) for j in range(m)]
for i in range(n - m + 1):
    for j in range(n - m + 1):
        if a[i:i + m] == list(map(list, zip(*b[j:j + m]))):
            print(i + 1, j + 1)
            break
