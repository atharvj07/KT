#include <iostream>
#include <algorithm>
#include <string>
#include <vector>
#include <stack>

using namespace std;

const int N = 1e5 + 1;
vector<int> nxt[N][3];
int lnk[N];
char s[N];

int main() {
  int n;
  cin >> n >> s + 1;
  for (int i = n; i >= 1; i--) {
    int p = lnk[i];
    for (int j = 0; j < 3; j++)
      for (int k : nxt[p][j])
        if (i + j != k) nxt[i][j].push_back(k);
    lnk[i] = i - 1;
    if (i < n && s[i] == s[i + 1]) lnk[i] = lnk[i + 1];
  }
  stack<int> stk;
  for (int i = 1; i <= n; i++) {
    if (i + 2 <= n && s[i] == s[i + 1] && s[i] == s[i + 2]) {
      while (!stk.empty() && stk.top() <= i) stk.pop();
      stk.push(i + 2);
    }
    if (i + 1 <= n && s[i] == s[i + 1]) {
      while (!stk.empty() && stk.top() <= i) stk.pop();
      stk.push(i + 1);
    }
  }
  while (!stk.empty()) {
    int j = stk.top();
    stk.pop();
    for (int k = 0; k < 3; k++) {
      for (int p : nxt[j][k]) nxt[lnk[j]][k].push_back(p);
      nxt[j][k].clear();
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < 3; j++)
      for (int k : nxt[i][j])
        if (i + j != k) nxt[i][j].push_back(k);
  }
  vector<string> res;
  for (int i = 1; i <= n; i++) {
    if (s[i] == s[i + 1] && s[i] == s[i + 2]) {
      int p = i;
      while (p < n && s[p] == s[p + 1] && s[p] == s[p + 2]) p++;
      res.push_back(string(s + i, s + i + 3));
      i = p - 1;
    } else if (s[i] == s[i + 1]) {
      int p = i;
      while (p < n && s[p] == s[p + 1]) p++;
      res.push_back(string(s + i, s + i + 2));
      i = p - 1;
    }
  }
  sort(res.begin(), res.end());
  cout << res.size() << '\n';
  for (string t : res) cout << t << '\n';
  return 0;
}
