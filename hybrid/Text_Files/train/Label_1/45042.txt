import java.util.Scanner;

public class SecondPriceAuction {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read the number of bidders
        int n = scanner.nextInt();

        // Read the bids
        int[] bids = new int[n];
        for (int i = 0; i < n; i++) {
            bids[i] = scanner.nextInt();
        }

        // Find the winner and the second-highest bid
        int winnerIndex = 0;
        int secondHighestBid = Integer.MIN_VALUE;
        for (int i = 0; i < n; i++) {
            if (bids[i] > bids[winnerIndex]) {
                winnerIndex = i;
            } else if (bids[i] > secondHighestBid) {
                secondHighestBid = bids[i];
            }
        }

        // Print the result
        System.out.println((winnerIndex + 1) + " " + secondHighestBid);
    }
}
