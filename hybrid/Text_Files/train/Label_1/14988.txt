#include <iostream>
using namespace std;
int main() {
  string s1, s2;
  cin >> s1 >> s2;
  if (s1 == s2)
    cout << "YES";
  else if (s1.size() % 2 == 0) {
    string s1f = s1.substr(0, s1.size() / 2),
           s1s = s1.substr(s1.size() / 2), s2f = s2.substr(0, s2.size() / 2),
           s2s = s2.substr(s2.size() / 2);
    if ((s1f == s2f && s1s == s2s) || (s1f == s2s && s1s == s2f))
      cout << "YES";
    else
      cout << "NO";
  } else
    cout << "NO";
  return 0;
}
