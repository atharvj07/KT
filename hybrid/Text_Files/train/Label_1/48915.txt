import java.util.*;

public class Main {
  static int n;
  static ArrayList<Integer>[] g;
  static int[] sz;
  static int[] dp;
  static void dfs(int u, int p) {
    dp[u] = 1;
    sz[u] = 1;
    for(int v : g[u]) {
      if(v != p) {
        dfs(v, u);
        sz[u] += sz[v];
        dp[u] = Math.max(dp[u], 1 + dp[v]);
      }
    }
  }
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    n = in.nextInt();
    g = new ArrayList[n + 1];
    sz = new int[n + 1];
    dp = new int[n + 1];
    for(int i = 0; i <= n; i++) g[i] = new ArrayList<>();

    for(int i = 1; i < n; i++) {
      int x = in.nextInt();
      int y = in.nextInt();
      g[x].add(y);
      g[y].add(x);
    }
    dfs(1, 1);
    System.out.println(dp[1]);
  }
}
