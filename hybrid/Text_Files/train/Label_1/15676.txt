#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int inc = k / n % 26;
    int dec = k / n - inc;
    string ans = "";
    for (char c : s) {
        ans += c + dec;
        if (dec > 0) dec--;
    }
    for (char c : s) {
        ans += c + inc;
        if (inc > 0) inc--;
    }
    if (k % n != 0) ans = "-1";
    cout << ans << endl;
    return 0;
}
