#include <iostream>
#include <cmath>
#include <vector>
#include <cstdlib>
using namespace std;

struct Point { double x, y; };
struct Circle { Point p; double r; };

int n;
Circle c[3];

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        double x, y, r; cin >> x >> y >> r;
        c[i].p.x = x;  c[i].p.y = y;  c[i].r = r;
    }
    int ans = 1, i, j;
    for (i = 0; i < n; i++) for (j = i + 1; j < n; j++) {
        double dx = c[i].p.x - c[j].p.x, dy = c[i].p.y - c[j].p.y;
        double dist2 = dx * dx + dy * dy;
        double rsum = c[i].r + c[j].r;
        if (dist2 > rsum * rsum) ans++;
        else if (dist2 == rsum * rsum) {
            double cross = dx * (c[j].p.y - c[i].p.y) - dy * (c[j].p.x - c[i].p.x);
            if (cross != 0) ans++;
            else ans += 2;
        }
    }
    cout << ans;
    return 0;
}
