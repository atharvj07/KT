import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] x1 = new int[n + 1];
        int[] y1 = new int[n + 1];
        int[] x2 = new int[m + 1];
        int[] y2 = new int[m + 1];
        for (int i = 1; i <= n; i++) {
            x1[i] = sc.nextInt();
            y1[i] = sc.nextInt();
        }
        for (int i = 1; i <= m; i++) {
            x2[i] = sc.nextInt();
            y2[i] = sc.nextInt();
        }
        int[] ans = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            int mind = Integer.MAX_VALUE;
            int idx = -1;
            for (int j = 1; j <= m; j++) {
                int d = Math.abs(x1[i] - x2[j]) + Math.abs(y1[i] - y2[j]);
                if (d < mind) {
                    mind = d;
                    idx = j;
                } else if (d == mind) {
                    idx = Math.min(idx, j);
                }
            }
            ans[i] = idx;
        }
        for (int i = 1; i <= n; i++) {
            System.out.println(ans[i]);
        }
    }
}
