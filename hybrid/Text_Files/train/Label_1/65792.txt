import heapq
import sys

input = sys.stdin.readline
heap = []

n, m = map(int, input().split())
costs = list(map(int, input().split()))

for i in range(m):
    a, b, w = map(int, input().split())
    a -= 1
    b -= 1
    heapq.heappush(heap, (w, a, b))

for i in range(n):
    heapq.heappush(heap, (costs[i], i, i))

par = list(range(n))
ans = 0

def find_par(x):
    if par[x] != x:
        par[x] = find_par(par[x])
    return par[x]
def unite(a, b):
    a = find_par(a)
    b = find_par(b)
    if a != b:
        if costs[a] < costs[b]:
            par[b] = a
        else:
            par[a] = b

while heap:
    w, u, v = heapq.heappop(heap)
    if find_par(u) != find_par(v):
        unite(u, v)
        ans += w

print(ans)
