import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int n = s.length();
        for (int k = 0; k < n; k++) {
            if (k == 0 || s.charAt(k) > s.charAt(k - 1))
                System.out.println("Mike");
            else
                System.out.println("Ann");
        }
    }
}
