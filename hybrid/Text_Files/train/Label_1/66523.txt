from collections import Counter

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        p = list(map(int, input().split()))
        cnt = Counter(p)
        for i in range(n - 1):
            if cnt[p[i]] + cnt[p[i + 1]] >= 2:
                p[i], p[i + 1] = p[i + 1], p[i]
                break
        print(' '.join(map(str, p)))

solve()
