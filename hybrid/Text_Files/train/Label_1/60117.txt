from sys import stdin

n, k = map(int, stdin.readline().split())

def make(n, k):
    if k == 0:
        return [i for i in range(1, 2*n + 1)]
    if k == 1:
        return [2*i for i in range(1, n + 1)] + [2*i - 1 for i in range(n, 2*n)]
    if k == n-1:
        return [2*i - 1 for i in range(1, n + 1)] + [2*i for i in range(n, 2*n)]
    if n % 2:
        L = make(n//2, k-n//2)
        R = make(n-n//2, k)
        return L + [2*i+n%2 for i in R]
    L = make(n//2, k-n//2)
    R = make(n-n//2, k)
    return [2*i for i in L] + [2*i+1 for i in R]

a = make(n, k)
for ai in a:
    print(ai, end=" ")
