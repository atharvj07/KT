#include<bits/stdc++.h>
using namespace std;
int main(){
    vector<array<int,3>> v(8);
    for(auto &i:v){
        for(auto &j:i){ cin>>j; }
    }
    int dx[]={0,0,0,0,1,1,1,1};
    int dy[]={0,1,0,-1,0,1,0,-1};
    int dz[]={1,0,-1,0,1,0,-1,0};
    int sx=0,sy=0,sz=0;
    for(auto i:v){
        if(i[0]==0&&i[1]==0&&i[2]==0){
            continue;
        }
        if(sx==0&&sy==0&&sz==0){
            sx=i[0],sy=i[1],sz=i[2];
        }else{
            for(int j=0;j<8;++j){
                if(i[0]==sx+dx[j]&&i[1]==sy+dy[j]&&i[2]==sz+dz[j]){
                    continue;
                }else{
                    goto notcube;
                }
            }
        }
    }
    cout<<"YES"<<endl;
    for(auto i:v){
        for(int j=0;j<8;++j){
            if(i[0]==sx+dx[j]&&i[1]==sy+dy[j]&&i[2]==sz+dz[j]){
                cout<<sx<<" "<<sy<<" "<<sz<<endl;
                sx+=dx[j],sy+=dy[j],sz+=dz[j];
                break;
            }
        }
    }
    return 0;
notcube:
    cout<<"NO"<<endl;
    return 0;
}
