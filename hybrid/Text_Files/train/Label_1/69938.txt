n = int(input())
a = input()
b = [[1, 0] for i in range(n)]
for i in range(n):
    for j in range(n):
        if a[i] == a[j] and i != j:
            b[j][0] = 0
for i in range(1, n):
    if b[i][0] == 1:
        for j in range(0, i):
            if b[j][0] == 1:
                b[j][1] += 1
for i in range(1, n):
    if b[i][0] == 1 and b[i][1] % 2 == 0:
        b[i][0] = 0
if b[0][0] == 1:
    print("D")
else:
    print("R")
