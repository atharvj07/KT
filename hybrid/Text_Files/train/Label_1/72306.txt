n, m = map(int, input().split())
t = sorted([input() for _ in range(n)])
s = [input() for _ in range(n)]
mn = m
for i in range(1, m):
    c = 0
    for j in range(n):
        if s[j][i] < s[j][i - 1]: c = 1
    if s[0][i] < t[0][i]: c = 1
    for j in range(1, n):
        if s[j][i] < t[j][i - 1]: c = 1
    mn = min(mn, c)
print(mn)
