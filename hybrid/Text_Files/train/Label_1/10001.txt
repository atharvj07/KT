#include<bits/stdc++.h>
using namespace std;
int32_t main() {
    int n; cin >> n;
    vector<int64_t>a(n), b(n);
    for(auto& x: a) cin >> x;
    for(auto& x: b) cin >> x;
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    long long ans = 0;
    for (int i = 0, j = n - 1; i < n; i++) {
        while (j >= 0 && a[i] + a[j] > b[i] + b[j]) j--;
        ans += n - j - 1;
    }
    cout << ans << endl;
    return 0;
}
