#include <iostream>
using namespace std;
int main() {
    int t; cin >> t;
    while (t--) {
        int x, y, p, q; cin >> x >> y >> p >> q;
        if (p * y == q * x) cout << 0 << endl;
        else if (p * y < q * x) cout << -1 << endl;
        else {
            if ((q * x) % p != 0) cout << -1 << endl;
            else cout << (q * x) / p - x << endl;
        }
    }
    return 0;
}
