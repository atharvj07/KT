#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<int> c(m, 0);
    for (int i = 0; i < n; i++) {
        c[a[i] % m]++;
    }
    int ans = 0;
    for (int i = 0; i < m; i++) {
        if (c[i] > n / m) {
            ans += c[i] - n / m;
            c[(i + 1) % m] += c[i] - n / m;
            c[i] = n / m;
        }
    }
    cout << ans << endl;
    for (int i = 0; i < n; i++) {
        cout << a[i] << " ";
        a[i] += ans;
        a[i] = (a[i] % m + m) % m;
    }
    cout << endl;
    return 0;
}
