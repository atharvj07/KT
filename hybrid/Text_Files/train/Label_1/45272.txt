import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static class Edge {
        int to, cost;

        public Edge(int to, int cost) {
            this.to = to;
            this.cost = cost;
        }
    }

    static class Node {
        int val;
        List<Edge> edges;

        public Node() {
            this.val = 0;
            this.edges = new ArrayList<>();
        }
    }

    final static int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        Node[] nodes = new Node[N + 1];
        for (int i = 1; i <= N; i++) {
            nodes[i] = new Node();
        }
        int[] parents = new int[N + 1];
        for (int i = 2; i <= N; i++) {
            int a = scanner.nextInt();
            nodes[a].edges.add(new Edge(i, 0));
            nodes[i].edges.add(new Edge(a, 0));
            parents[i] = a;
        }
        int Q = scanner.nextInt();
        for (int i = 0; i < Q; i++) {
            int type = scanner.nextInt();
            if (type == 1) {
                int v = scanner.nextInt();
                int x = scanner.nextInt();
                int k = scanner.nextInt();
                dfs1(nodes, v, parents, x, k, 0);
            } else {
                int v = scanner.nextInt();
                System.out.println(nodes[v].val);
            }
        }
    }

    private static void dfs1(Node[] nodes, int v, int[] parents, int x, int k, int depth) {
        nodes[v].val = (nodes[v].val + x) % MOD;
        for (Edge edge : nodes[v].edges) {
            if (parents[v] == edge.to) {
                continue;
            }
            dfs1(nodes, edge.to, parents, x, k, depth + 1);
        }
        nodes[v].val = (nodes[v].val - (depth * k)) % MOD;
        if (nodes[v].val < 0) {
            nodes[v].val += MOD;
        }
    }
}
