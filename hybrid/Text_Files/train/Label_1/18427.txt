#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 1005;
int n, a[N][4], sum[N];
pair<int, int> ans[N];
bool cmp(const pair<int, int> &x, const pair<int, int> &y) {
    if (x.first != y.first) return x.first > y.first;
    return x.second < y.second;
}
int32_t main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < 4; j++) { cin >> a[i][j]; }
        sum[i] += a[i][0] + a[i][1] + a[i][2] + a[i][3];
        ans[i] = {sum[i], i};
    }
    sort(ans + 1, ans + 1 + n, cmp);
    for (int i = 1; i <= n; i++) {
        if (ans[i].second == 1) cout << i;
    }
    return 0;
}
