import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n + 1];
        String[] s = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i + 1] = Integer.parseInt(s[i]);
        }
        int[] l = new int[n + 1];
        int[] r = new int[n + 1];
        Set<Long> s1 = new HashSet<>();
        for (int i = 1; i <= n; i++) {
            r[a[i]] = i;
            if (l[a[i]] != 0) {
                s1.add(((long) i + (long) l[a[i]]) * 1000000000L + (long) (a[i]));
            }
            l[a[i]] = i;
        }
        Map<Long, Set<Integer>> s2 = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            r[a[i]] = i;
            for (int j = i + 1; j <= n && (a[j] > a[i]); j++) {
                if (l[a[j]] != 0) {
                    s2.putIfAbsent(((long) i + (long) j) * 1000000000L + (long) (a[j]), new HashSet<>());
                    s2.get(((long) i + (long) j) * 1000000000L + (long) (a[j])).add(a[i]);
                }
            }
            l[a[i]] = i;
        }
        long ans = s1.size() + s2.size();
        for (Set<Integer> j : s2.values()) {
            ans += (long) j.size() * (long) (j.size() - 1) / 2;
        }
        System.out.println(ans);
    }
}
