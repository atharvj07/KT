#include<bits/stdc++.h>
#pragma GCC optimize("Ofast")
using namespace std;
typedef long long ll;
#define endl '\n'
#define fio ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0)
const int N = 1e5 + 9, M = 5 + 9, K = 1e9 + 9;
int n, m, k;
int a[N][M];
int ans[M];

void input() {
    cin >> n >> m >> k;
    for(int i = 0;i < n;i++) {
        for(int j = 0;j < m;j++) {
            cin >> a[i][j];
        }
    }
}

void solve() {
    int del[m];
    for(int j = 0;j < m;j++) {
        del[j] = 0;
        int tot = 0;
        for(int i = 0;i < n;i++) {
            if(!a[i][j]) continue;
            tot++;
            if(tot > n / 2) {
                del[j]++;
            }
        }
    }

    int mx = 0;
    for(int j = 0;j < m;j++) {
        mx = max(mx, del[j]);
    }

    for(int j = 0;j < m;j++) {
        ans[j] = del[j];
        k -= ans[j];
    }

    for(int j = 0;j < m;j++) {
        if(k <= 0) break;
        while(k >= ans[j]) {
            ans[j]++;
            k--;
        }
    }
}

void output() {
    for(int j = 0;j < m;j++) {
        cout << ans[j] << ' ';
    }
    cout << endl;
}

int main() {
    fio;
    input();
    solve();
    output();
    return 0;
}
