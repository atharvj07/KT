def min_moves(n):
    moves = 0
    while n != 1:
        if n % 2 == 0:
            n //= 2
        elif n % 3 == 0:
            n = 2 * n // 3
        elif n % 5 == 0:
            n = 4 * n // 5
        else:
            return -1
        moves += 1
    return moves

q = int(input().strip())
for _ in range(q):
    n = int(input().strip())
    print(min_moves(n))
