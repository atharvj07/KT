import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] a = new int[n];
        Map<Integer, Integer> indexA = new HashMap<>();
        for (int i = 0; i < n; i++) {
            a[i] = scan.nextInt();
            indexA.put(a[i], i);
        }
        int[] b = new int[n];
        Map<Integer, Integer> indexB = new HashMap<>();
        for (int i = 0; i < n; i++) {
            b[i] = scan.nextInt();
            indexB.put(b[i], i);
        }
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[indexA.get(b[i])] = i;
        }
        System.out.println(n - 1);
        for (int i = 0; i < n - 1; i++) {
            if (i != p[i]) {
                System.out.println((i + 1) + " " + (p[i] + 1));
                int tmp = p[i];
                p[i] = p[tmp];
                p[tmp] = tmp;
            }
        }
    }
}
