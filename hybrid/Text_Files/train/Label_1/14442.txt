#include <iostream>
#include <vector>
using namespace std;

int check_kth(vector<int>& a, int k) {
    int n = a.size(), p = a[k];
    for (int i = 1; i < n; i++) {
        if (i != k && a[i] >= p) return -1;
    }
    return k;
}

int solve() {
    int n; cin >> n;
    vector<int> a(n);
    for (int& x : a) cin >> x;
    int j = -1;
    for (int i = 0; i < n; i++) {
        if (!~j) j = i;
        if (a[i] > a[j]) j = i;
    }
    if (j == -1) return -1;
    return check_kth(a, j - 1) == -1 ? check_kth(a, j + 1) : j;
}

int main() {
    ios_base::sync_with_stdio(false), cin.tie(nullptr);
    int t; cin >> t;
    while (t--) {
        cout << solve() << endl;
    }
    return 0;
}
