import java.util.Scanner;

public class ValentinAndElectricShock {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();

        char[] selectedLetters = new char[26];
        for (int i = 0; i < 26; i++) {
            selectedLetters[i] = ' ';
        }

        int countExcessiveShocks = 0;
        for (int i = 0; i < n; i++) {
            String action = scanner.nextLine();
            char firstChar = action.charAt(0);
            if (firstChar == '.') {
                String word = action.substring(2);
                for (char c : word.toCharArray()) {
                    selectedLetters[c - 'a'] = ' ';
                }
            } else if (firstChar == '!') {
                String word = action.substring(2);
                for (char c : word.toCharArray()) {
                    if (selectedLetters[c - 'a'] == ' ') {
                        selectedLetters[c - 'a'] = c;
                    } else {
                        countExcessiveShocks++;
                    }
                }
            } else if (firstChar == '?') {
                char guess = action.charAt(2);
                if (selectedLetters[guess - 'a'] == guess) {
                    break;
                } else {
                    countExcessiveShocks++;
                }
            }
        }

        System.out.println(countExcessiveShocks);
    }
}
