def min_presses_to_palindrome(n, p, s):
    s = list(s)
    total_moves = 0
    if p > n // 2:
        p = n - p + 1
    left, right = p - 1, p - 1
    for i in range(p - 1, n // 2):
        if s[i] != s[n - i - 1]:
            diff = abs(ord(s[i]) - ord(s[n - i - 1]))
            total_moves += min(diff, 26 - diff)
            right = i
    for i in range(p - 1, -1, -1):
        if s[i] != s[n - i - 1]:
            diff = abs(ord(s[i]) - ord(s[n - i - 1]))
            total_moves += min(diff, 26 - diff)
            left = i
    if total_moves > 0:
        total_moves += min(abs(left - (p - 1)) + abs(right - left), abs(right - (p - 1)) + abs(right - left))
    return total_moves

n, p = map(int, input().split())
s = input().strip()
print(min_presses_to_palindrome(n, p, s))
