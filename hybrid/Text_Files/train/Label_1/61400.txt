n, m = map(int, input().split())
a = list(map(int, input().split()))
edges = [list(map(int, input().split())) for _ in range(m)]
edges.sort(key=lambda e: e[2])
u = [-1] * (n + 1)
sz = [1] * (n + 1)
ans = 0
for v, u, w in edges:
    if a[v - 1] < 0 and a[u - 1] < 0:
        continue
    rv, ru = find(v), find(u)
    if rv != ru:
        ans += w
        if sz[rv] > sz[ru]:
            u[ru] = rv
            sz[rv] += sz[ru]
        else:
            u[rv] = ru
            sz[ru] += sz[rv]
        a[rv] = max(a[rv], a[ru])

def find(x):
    if u[x] != x:
        u[x] = find(u[x])
    return u[x]
print(ans)
