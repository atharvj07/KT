#include <cstdio>
#include <algorithm>
using namespace std;

struct contest{
    int p1, p2, w;
    bool operator <(const contest &b)const{
        return w<b.w;
    }
}contestants[802];

int p[401], w[401], pa[401];

int findp(int i){
    if(i==p[i]) return i;
    return p[i]=findp(p[i]);
}

int main(){
    int n, m, i, j, t, c=2;
    scanf("%d", &n);
    for(i=1; i<=2*n; i++){
        scanf("%d", &w[i]);
        p[i]=i;
    }
    for(i=1; i<2*n; i++){
        for(j=i+1; j<=2*n; j++){
            contestants[c].p1=i;
            contestants[c].p2=j;
            contestants[c].w=w[i]+w[j];
            c++;
        }
    }
    sort(contestants+2, contestants+c);
    for(i=2; i<c; i++){ // start from 2 because 1st pair will always be 1 and 2
        int u=findp(contestants[i].p1);
        int v=findp(contestants[i].p2);
        if(u!=v){
            p[u]=v;
            pa[contestants[i].p1]=contestants[i].p2;
            pa[contestants[i].p2]=contestants[i].p1;
        }
    }
    for(i=1; i<=2*n; i++) printf("%d ", pa[i]);
    printf("\n");
    return 0;
}
