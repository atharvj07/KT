#include<bits/stdc++.h>
using namespace std;
#define ll long long
int t,n;
ll s[100001];
int main(){
    cin>>t;
    while(t--){
        cin>>n;
        for(int i=1;i<=n;i++) cin>>s[i];
        int ans=1;
        for(int i=2;i<=n;i++){
            if(s[i-1]<s[i] and (s[i]%s[i-1]==0 or s[i-1]%s[i]==0)) ans++;
        }
        cout<<ans<<endl;
    }
}
