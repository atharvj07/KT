#include <iostream>
using namespace std;

int mod = 1e9 + 7;
long long pow(long long b, long long p) {
    if (p == 0) { return 1; }
    if (p == 1) { return b; }
    long long temp = pow(b, p / 2);
    temp = (temp * temp) % mod;
    if (p % 2) { temp = (temp * b) % mod; }
    return temp;
}
int main() {
    long long n, m;
    cin >> n >> m;
    long long ans = (pow(m, (n + 1) / 2) - 1) % mod;
    ans = (ans * pow((m - 1), mod - 2)) % mod;
    cout << ans << endl;
    return 0;
}
