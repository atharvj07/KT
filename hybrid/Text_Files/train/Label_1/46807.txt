import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
        String[] line = input.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int k = Integer.parseInt(line[1]);
        long s = Long.parseLong(line[2]);
        long t = Long.parseLong(line[3]);
        long[] car = new long[n];
        for (int i = 0; i < n; i++) {
            line = input.readLine().split(" ");
            car[i] = Long.parseLong(line[0]) * (s + 1) + Long.parseLong(line[1]);
        }
        line = input.readLine().split(" ");
        long[] gas = new long[k];
        for (int i = 0; i < k; i++) {
            gas[i] = Long.parseLong(line[i]);
        }
        long best = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            long cur = car[i];
            long fuel = car[i];
            for (int j = 0; j < k; j++) {
                if (gas[j] <= s && s - gas[j] <= fuel) {
                    fuel = car[i];
                }
                fuel -= s / 2;
                cur += (s + 1) / 2;
            }
            if (fuel >= s) {
                cur -= s;
            }
            best = Math.min(cur, best);
        }
        if (best <= t) {
            System.out.println(best);
        } else {
            System.out.println(-1);
        }
    }
}
