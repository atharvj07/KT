import java.util.Scanner;

public class RyokosMemoryNote {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }

        int minPages = solve(n, a);
        System.out.println(minPages);
    }

    private static int solve(int n, int[] a) {
        int minPages = 0;
        int lastPage = 1;

        for (int i = 0; i < a.length; i++) {
            int pagesToTurn = Math.abs(a[i] - lastPage);
            minPages += pagesToTurn;
            lastPage = a[i];
        }

        // Find the optimal page to merge
        int bestMergePage = 0;
        int bestMergeSaving = 0;

        for (int i = 1; i < n; i++) {
            int mergeSaving = 0;
            for (int j = 0; j < a.length; j++) {
                int pagesToTurnBefore = Math.abs(a[j] - lastPage);
                int pagesToTurnAfter = Math.abs(a[j] - i);
                mergeSaving += Math.min(pagesToTurnBefore, pagesToTurnAfter);
            }

            if (mergeSaving > bestMergeSaving) {
                bestMergePage = i;
                bestMergeSaving = mergeSaving;
            }
        }

        // Apply the merge
        if (bestMergeSaving > 0) {
            minPages -= bestMergeSaving;
        }

        return minPages;
    }
}
