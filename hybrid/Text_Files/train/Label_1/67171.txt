from bisect import bisect

n = int(input())
t = [int(input()) for _ in range(n)]
one = 20
time_one = 90
cost_one = 50
time_day = 1440
cost_day = 120
cost = [one]
times = [0]
for ti in t:
    best_cost = 1e9
    idx = bisect(times, ti - time_one)
    if idx > 0:
        best_cost = min(best_cost, cost[idx - 1] + cost_one)
    idx = bisect(times, ti - time_day)
    if idx > 0:
        best_cost = min(best_cost, cost[idx - 1] + cost_day)
    best_cost = min(best_cost, cost[-1] + one)
    cost.append(best_cost)
    times.append(ti)
    print(best_cost - cost[-2])
