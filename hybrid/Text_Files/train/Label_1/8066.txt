#include <iostream>
#include <vector>
#define mod 1000000007

using namespace std;
typedef vector<int> vi;

int dx[]={-1,0,1,0};
int dy[]={0,1,0,-1};
int n,m;
vector<string> arr;
vector<vi> dp;

bool valid(int x,int y)
{
  return (x>=0&&x<n&&y>=0&&y<m&&arr[x][y]!='#');
}

int func(int ind,int x,int y)
{
  if(ind==arr.size())
  {
    return (x==n-1&&y==m-1);
  }

  if(dp[x][y]!=-1)
    return dp[x][y];

  int &ret=dp[x][y];
  ret=0;
  
  for(int i=0;i<4;i++)
  {
    if(i==ind)
    {
      int p=func(ind+1,x+dx[i],y+dy[i]);
      ret+=p;
      ret%=mod;
    }
  }

  return ret;
}

int main()
{
  int t;
  string s;
  cin>>n>>m;
  arr.resize(n);
  for(int i=0;i<n;i++)
    cin>>arr[i];
  cin>>s;

  int x,y;
  for(int i=0;i<n;i++)
    for(int j=0;j<m;j++)
    {
      if(arr[i][j]=='S')
      {
        x=i;
        y=j;
      }
    }

  dp.assign(n,vi(m,-1));
  cout<<func(0,x,y)<<endl;
}
