#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int mx = *max_element(a.begin(), a.end()), mn = *min_element(a.begin(), a.end());
    if ((mx - mn) % 2 == 1) {
        cout << -1 << endl;
    } else {
        cout << (mx - mn) / 2 << endl;
    }
    return 0;
}
