import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int y = sc.nextInt();
        int w = sc.nextInt();
        if (y < w) {
            System.out.println("0/1");
        } else if (y > w) {
            System.out.println("1/1");
        } else {
            System.out.println("1/2");
        }
    }
}
