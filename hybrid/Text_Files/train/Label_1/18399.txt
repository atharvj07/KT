#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, m, L = 0;
  cin >> n >> m;
  vector<int> a(n), l(n), r(n);
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    l[i] = i ? __gcd(l[i - 1], a[i]) : a[i];
  }
  for (int i = n - 1; i >= 0; --i) {
    r[i] = i < n - 1 ? __gcd(r[i + 1], a[i]) : a[i];
  }
  int st = -1, end = -1, L2 = 0;
  for (int i = 0; i < n; ++i) {
    long long ml = 1LL * l[i] * r[i];
    if (ml <= m) {
      if (L2 < l[i]) {
        L2 = l[i];
        st = i;
        end = i;
      }
      if (l[i] == L2 && i - st > end - st) {
        end = i;
      }
    }
  }
  cout << l[st] << " ";
  if (st == -1) {
    cout << 0 << "\n";
  } else {
    cout << end - st + 1 << "\n";
    for (int i = st; i <= end; ++i) {
      cout << i + 1 << " ";
    }
    cout << "\n";
  }
  return 0;
}
