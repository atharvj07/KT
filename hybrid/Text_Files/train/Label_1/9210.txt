#include <iostream>
#include <set>
#include <algorithm>
#include <vector>
using namespace std;
typedef long long ll;
const int maxN = 2e5+1;
int n, k;
vector<int> arr[maxN];
ll dist[maxN], ans;
set<int> path;
void dfs(int x, int parent) {
    ans += dist[x];
    path.insert(dist[x]);
    for (int i : arr[x]) {
        if (i != parent) {
            if (dist[i] == 0) {
                dist[i] = dist[x]+1;
                dfs(i, x);
            } else
                path.insert(dist[i]);
        }
    }
}
int main() {
    int a, b;
    cin >> n >> k;
    for (int i = 1; i < n; i++) {
        cin >> a >> b;
        arr[a].push_back(b);
        arr[b].push_back(a);
    }
    dfs(1, 1);
    cout << ans;
    return 0;
}
