import java.util.*;
import java.io.*;
import static java.lang.Math.*;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int m = cin.nextInt();
        int k = cin.nextInt();
        int[][] dp = new int[n][m];
        int[][] arr = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = cin.nextInt();
            }
        }
        for (int i = 0; i < k; i++) {
            dp[0][i] = arr[0][i];
            if (i > 0)
                dp[0][i] += dp[0][i - 1];
        }
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int cur = arr[i][j];
                if (j >= k) {
                    cur += dp[i][j - k];
                }
                if (i > 1) {
                    cur += dp[i - 1][m - 1];
                    if (j >= k)
                        cur -= dp[i - 1][j - k];
                }
                dp[i][j] = max(dp[i][j], cur);
            }
        }
        System.out.println(dp[n - 1][m - 1]);
    }
}
