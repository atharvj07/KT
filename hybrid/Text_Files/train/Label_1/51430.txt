import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        String s = sc.next();
        sc.close();

        int[] x = new int[n + 1];
        for (int i = 1; i <= n; i++)
            x[i] = (s.charAt(i - 1) == 'X' ? -1 : (s.charAt(i - 1) == 'B' ? 1 : 0));
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++)
            p[i] = p[i - 1] + x[i];

        long ans = 0, sum = 0;
        for (int i = k, j; i <= n; i++) {
            j = i - k + 1;
            sum += x[j] == -1 ? 1 : 0;
            if (p[i] - p[j - 1] == k)
                ans += sum;
        }
        System.out.println(ans % 1000000007);
    }
}
