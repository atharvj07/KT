import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    long[] x = new long[n];
    long[] p = new long[m];
    for (int i = 0; i < n; i++) x[i] = sc.nextLong();
    for (int i = 0; i < m; i++) p[i] = sc.nextLong();
    long first = x[0];
    boolean found = false;
    for (long pj : p) {
      long y = first % pj == 0 ? first : first + (pj - (first % pj));
      boolean ok = true;
      for (long xj : x) {
        if ((xj - y) % pj != 0) {
          ok = false;
          break;
        }
      }
      if (ok) {
        System.out.println("YES");
        System.out.println(y + " " + (Arrays.binarySearch(p, pj) + 1));
        found = true;
        break;
      }
    }
    if (!found) System.out.println("NO");
  }
}
