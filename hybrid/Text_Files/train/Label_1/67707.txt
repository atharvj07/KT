def valid_mapping(maze, s, x, y, ex, ey):
    cnt = 0
    for i in range(4):
        nx, ny = x + dirs[i][0], y + dirs[i][1]
        if 0 <= nx < len(maze) and 0 <= ny < len(maze[0]) and  maze[nx][ny] != '#' and (nx, ny) != (x, y):
            if nx == ex and ny == ey:
                cnt += 1
            else:
                cnt += valid_mapping(maze, s, nx, ny, ex, ey)
    return cnt

inp = input().split()
n, m = int(inp[0]), int(inp[1])
maze = [list(input()) for i in range(n)]

dirs = [(1, 0), (0, -1), (-1, 0), (0, 1)]

for i in range(n):
    for j in range(m):
        if maze[i][j] == 'S':
            sx, sy = i, j
        elif maze[i][j] == 'E':
            ex, ey = i, j

print(valid_mapping(maze, input(), sx, sy, ex, ey))
