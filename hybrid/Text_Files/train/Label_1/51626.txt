import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = in.nextInt();
        }
        in.close();
        List<Pair> list = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            list.add(new Pair(a[i], b[i]));
        }
        Collections.sort(list);
        int processors = 0;
        double power = 0;
        boolean blocked = false;
        int lastProcessor = 100;
        for (Pair pair : list) {
            int p = pair.second;
            int pw = pair.first;
            if (blocked && lastProcessor == p) {
                continue;
            }
            if (processors + p <= 2) {
                processors += p;
                power += pw;
                if (p == 1) {
                    lastProcessor = 1;
                } else {
                    lastProcessor = 100;
                }
            } else {
                if (processors == 0) {
                    processors = p;
                    power = pw;
                    if (p == 1) {
                        lastProcessor = 1;
                    } else {
                        lastProcessor = 100;
                    }
                } else {
                    if (power / processors >= pw) {
                        double diff = power / processors - pw;
                        int newP = (int) Math.ceil(diff / pw);
                        if (processors + newP <= 2) {
                            processors += newP;
                            power += newP * pw;
                            if (newP == 1) {
                                lastProcessor = 1;
                            } else {
                                lastProcessor = 100;
                            }
                            blocked = true;
                            continue;
                        }
                    }
                    blocked = true;
                    lastProcessor = p;
                }
            }
        }
        System.out.println(Math.round(power / processors * 1000));
    }
}

class Pair implements Comparable<Pair> {
    int first;
    int second;

    public Pair(int first, int second) {
        this.first = first;
        this.second = second;
    }

    @Override
    public int compareTo(Pair o) {
        if (o.first != first) {
            return first - o.first;
        }
        return o.second - second;
    }
}
