#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < n; i++)
#define all(v) v.begin(), v.end()
#define pb push_back
typedef long long ll;
int n;
vector<ll> a, b;

int main() {
  int T;
  cin >> T;
  while (T--) {
    cin >> n;
    a.resize(n), b.resize(n);
    rep(i, n) cin >> a[i];
    rep(i, n) cin >> b[i];
    sort(all(a)), sort(all(b));
    rep(i, n) cout << a[i] << ' ';
    cout << endl;
    rep(i, n) cout << b[i] << ' ';
    cout << endl;
  }
  return 0;
}
