#include <bits/stdc++.h>
using namespace std;
#define rep(i,a,b) for(int i=a,i##_E=b;i<i##_E;++i)
const int N=50,M=N*N,mo=998244353;

inline char gc() {
        static char buf[M],*p1=buf,*p2=buf;
        return p1==p2&&(p2=(p1=buf)+fread(buf,1,M,stdin),p1==p2)?EOF:*p1++;
}

inline int mid(int n) { return n-n&1; }

inline int read() {
        int x=0,f=1;char ch=gc();
        for(;!isdigit(ch);ch=gc()) if(ch=='-') f=-1;
        for(;isdigit(ch);ch=gc()) x=x*10+ch-'0';
        return x*f;
}

int n,ans,pre[N][N];
char s[N+2];

int main() {
        ios::sync_with_stdio(0);
        cin.tie(0);
        for(int T=read();T--;) {
                n=read(),fill(pre[0],pre[n]+n*n,0);
                cin>>s;
                rep(i,0,n) if(s[i]=='?') ans=1;
                else rep(j,0,n) {
                        if(j==mid(n)) {
                                if(i&1) {
                                        pre[i][j]=(pre[i-1][j]+pre[i][j-1])%mo;
                                        if(s[i]=='a') pre[i][j]=(pre[i][j]+pre[i-1][j-1])%mo;
                                }
                                else {
                                        pre[i][j]=(pre[i-1][j]+pre[i][j-1])%mo;
                                        if(s[i]=='b') pre[i][j]=(pre[i][j]+pre[i-1][j-1])%mo;
                                }
                        }
                        else {
                                pre[i][j]=(pre[i-1][j]+pre[i][j-1])%mo;
                                if(s[i]==s[j]) pre[i][j]=(pre[i][j]+pre[i-1][j-1])%mo;
                        }
                }
                cout<<(pre[n-1][mid(n)]==1?"Yes":"No")<<endl;
                if(ans) {
                        rep(i,0,n) {
                                if(s[i]=='?') {
                                        if(i&1) s[i]=(pre[i][mid(n)]&1)?'a':'c';
                                        else s[i]=(pre[i][mid(n)]&1)?'b':'d';
                                }
                        }
                        cout<<s<<endl;
                }
        }
        return 0;
}
