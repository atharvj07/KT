#include <bits/stdc++.h>
#define endl "\n"
using namespace std;
vector<int>V;
int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int w,h,n;
	cin>>w>>h>>n;
	V.push_back(w*h);
	while(n--)
	{
		char s;
		int x;
		cin>>s>>x;
		if(s=='H')
		{
			int j=0;
			for(int i=0;i<V.size();i++)
			{
				if(V[i]>h-x)
				{
					V[j]=V[i]-min(V[i],h-x);
					j++;
				}
				V[j]=min(V[i],h-x);
				j++;
			}
			sort(V.rbegin(),V.rend());
		}
		else
		{
			int j=0;
			for(int i=0;i<V.size();i++)
			{
				if(V[i]>w-x)
				{
					V[j]=V[i]-min(V[i],w-x);
					j++;
				}
				V[j]=min(V[i],w-x);
				j++;
			}
			sort(V.rbegin(),V.rend());
		}
		cout<<V[0]<<endl;
	}

	return 0;
}
