#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 10;
int cnt[MAXN], a[MAXN];
long long ans[MAXN];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= m; i++) {
        int l;
        cin >> l;
        ans[i] = cnt[a[l]];
        for (int j = l; j <= n; j++) {
            cnt[a[j]]++;
            ans[i] = max(ans[i], 1ll * cnt[a[j]]);
        }
    }
    for (int i = 1; i <= m; i++) cout << ans[i] << '\n';
    return 0;
}
