#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 2005;
ll dp[maxn][maxn];
int a[maxn];
int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) {
        dp[i][i] = 1;
    }
    for (int len = 1; len < n; len++) {
        for (int l = 1; l + len <= n; l++) {
            int r = l + len;
            if (a[l] <= a[r]) {
                dp[l][r] = max(dp[l][r], dp[l + 1][r] + 1);
            }
            if (a[l] <= a[r - 1]) {
                dp[l][r] = max(dp[l][r], dp[l][r - 1] + 1);
            }
            if (a[r] >= a[l - 1]) {
                dp[l][r] = max(dp[l][r], dp[l - 1][r - 1] + 1);
            }
            if (a[r] >= a[l]) {
                dp[l][r] = max(dp[l][r], dp[l][r - 1] + 1);
            }
        }
    }
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) { ans = max(ans, dp[i][j]); }
    }
    cout << ans << endl;
    return 0;
}
