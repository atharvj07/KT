#include<bits/stdc++.h>

using namespace std;
typedef long long ll;

const int MAXN = 1000;
int N, M, K;
vector<int> adj[MAXN+1];
bool vis[MAXN+1];
int c[MAXN+1];
int deg[MAXN+1];

bool can_add(int u, int v){
    if(vis[u] && vis[v]) return false;
    for(const int &adj_v: adj[u]) if(vis[adj_v]) return false;
    return true;
}

int main(){
    cin>>N>>M>>K;
    for(int i=0; i<K; ++i){
        cin>>c[i];
        vis[c[i]] = true;
    }
    for(int i=0; i<M; ++i){
        int u, v;
        cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
        deg[u]++; deg[v]++;
    }

    int ans = 0;
    for(int i=1; i<=N; ++i){
        for(int j=i+1; j<=N; ++j){
            if(can_add(i, j)){
                adj[i].push_back(j);
                adj[j].push_back(i);
                deg[i]++; deg[j]++;
                ans++;
            }
        }
    }

    cout<<ans<<endl;
}
