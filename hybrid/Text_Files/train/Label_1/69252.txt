from sys import stdin
input = stdin.readline
n, m = map(int, input().split())
friends = {i: list() for i in range(1, n+1)}
for i in range(m):
    a, b = map(int, input().split())
    friends[a].append(b)
    friends[b].append(a)
q = int(input())
process = [[], [], []]
for i in range(q):
    x = input().split()
    process[int(x[0]) - 1].append(x)
for add in process[0]:
    a, b = map(int, add[1:])
    friends[a].append(b)
    friends[b].append(a)
for delte in process[1]:
    a, b = map(int, delte[1:])
    friends[a].remove(b)
    friends[b].remove(a)
can_delete = [False for i in range(n+1)]
for vul in process[2]:
    vulnerable = [True for i in range(n+1)]
    for i in range(1, n+1):
        if not friends[i]:
            vulnerable[i] = False
        else:
            for j in friends[i]:
                if i > j:
                    vulnerable[i] = False
                    break
    for i in range(1, n+1):
        if vulnerable[i]:
            can_delete[i] = True
    count = 0
    for i in range(1, n+1):
        if not can_delete[i]:
            count += 1
    print(count)
    for i in range(1, n+1):
        if can_delete[i]:
            for j in friends[i]:
                friends[j].remove(i)
            friends[i] = []
            can_delete[i] = False
