def main():
    n = int(input())
    for i in range(n):
        s = input().split()
        sign = s[0]
        x = int(s[1])
        ans = s[2]
        if sign == '>':
            if ans == 'N':
                return 'Impossible'
        elif sign == '<':
            if ans == 'Y':
                return 'Impossible'
        elif sign == '>=':
            if ans == 'N' and x > 0:
                return 'Impossible'
        elif sign == '<=':
            if ans == 'N' and x < 0:
                return 'Impossible'
    return 0

print(main())
