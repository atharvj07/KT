n, k = map(int, input().split())
a = sorted([(p+1, e) for p, e in (map(int, input().split()) for _ in range(n))])
ans = -1
for i, (points, e) in enumerate(a):
    if i < len(a) - 1 and points == a[i + 1][0]: continue
    if n - i <= k - 1:
        ans = sum(x for _, x in a[i:])
        break
print(ans)
