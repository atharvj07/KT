#include <iostream>
#include <vector>
#include <queue>
#include <utility>
#include <algorithm>

using namespace std;

#define ll long long
#define pb push_back
#define mp make_pair

const int inf = 1e9;

int main() {
    ll n, m;
    cin >> n >> m;
    vector<pair<ll, ll> > g[n + 1];
    for (ll i = 0; i < m; i++) {
        ll u, v, w;
        cin >> u >> v >> w;
        g[u].pb(mp(w, v));
        g[v].pb(mp(w, u));
    }
    vector<ll> dist(n + 1, inf), p(n + 1, -1);
    dist[1] = 0;
    priority_queue<pair<ll, ll> > q;
    q.push(mp(0, 1));
    while (!q.empty()) {
        ll d = -q.top().first;
        ll v = q.top().second;
        q.pop();
        if (d > dist[v]) continue;
        for (auto u : g[v]) {
            if (d + u.first < dist[u.second]) {
                dist[u.second] = d + u.first;
                p[u.second] = v;
                q.push(mp(-dist[u.second], u.second));
            }
        }
    }
    vector<bool> vis(n + 1);
    ll ans = 0;
    for (ll i = 2; i <= n; i++) {
        if (!vis[i]) {
            ll v = i;
            while (v != -1) {
                vis[v] = true;
                if (dist[v] < dist[p[v]]) ans++;
                v = p[v];
            }
        }
    }
    cout << ans << endl;
    return 0;
}
