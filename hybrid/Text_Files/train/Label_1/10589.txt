#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

long long min_fee(vector<long long> &a, long long x, long long f) {
    vector<long long> cost(a.size(), 0);
    sort(a.begin(), a.end());
    a.push_back(0);
    long long cur = 0, cnt = 0, sum = 0, ans = 0;
    for (long long i=a.size()-1; i>=0; i--) {
        cur += a[i];
        cnt++;
        if (cur > x) {
            sum += cnt / 2 * f;
            ans += cnt / 2 * f;
            cost[i] += cnt % 2 * f;
            cur = cur % x;
            cnt = 1;
        }
    }
    for (auto &y: cost) ans += y;
    return ans;
}

int main() {
    long long n, x, f;
    scanf("%lld", &n);
    vector<long long> a(n, 0);
    for (long long i=0; i<n; i++) scanf("%lld", &a[i]);
    scanf("%lld %lld", &x, &f);
    printf("%lld\n", min_fee(a, x, f));
}
