#include <algorithm>
#include <cstdint>
#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<uint64_t> l(n);
    for (uint64_t& x : l) std::cin >> x;
    std::string s;
    std::cin >> s;
    std::vector<std::pair<char, uint64_t>> a;
    char cur = 'G';
    uint64_t sum = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] != cur) {
            a.emplace_back(cur, sum);
            sum = 0;
            cur = s[i];
        }
        sum += l[i];
    }
    a.emplace_back(cur, sum);
    int ans = 0;
    int stam = 0;
    for (const auto& p : a) {
        if (p.first == 'G') {
            ans += 5 * p.second + stam;
            stam += p.second;
        } else if (p.first == 'W') {
            if (p.second > stam) {
                ans += 3 * (p.second - stam);
                stam = 0;
            } else {
                ans += 3 * p.second;
                stam -= p.second;
            }
        } else {
            if (p.second > stam) {
                ans += std::max(3 * (p.second - stam), 1 * p.second);
                stam = 0;
            } else {
                ans += 1 * p.second;
                stam -= p.second;
            }
        }
    }
    std::cout << ans << std::endl;
}
