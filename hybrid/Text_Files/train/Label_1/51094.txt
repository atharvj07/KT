import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int n = s.length();
        int[] p = new int[n];
        int[] ans = new int[n];
        int bk = 0, lst = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'R') {
                p[i] = -1;
            } else {
                p[i] = 1;
                lst = i;
            }
        }
        for (int i = 0; i < n; i++) {
            if (p[i] == 0) {
                if (lst != 0) {
                    bk++;
                    ans[i] = lst + 1;
                    p[lst] = p[i] = -p[lst];
                } else {
                    ans[i] = i + 1;
                }
            } else {
                ans[i] = i + 1;
                lst = i;
            }
        }
        System.out.println(bk);
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
