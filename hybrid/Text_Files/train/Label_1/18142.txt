#include <bits/stdc++.h>
using namespace std;

int main() {
  long long n, k;
  cin >> n >> k;
  vector<long long> a(n), b;
  unordered_set<long long> s;
  for (long long i = 0; i < n; i++) { cin >> a[i]; }
  if (k) {
    for (long long i = 0; i < k; i++) {
      long long x;
      cin >> x;
      b.push_back(x);
      s.insert(x);
    }
  }
  vector<long long> ma(n, 0);
  long long ans = 0;
  for (long long i = n - 2; i >= 0; i--) {
    if (s.count(i + 1)) continue;
    ma[i] = max(a[i], a[i + 1]);
    if (ma[i] > a[i]) ans++;
  }
  if (is_sorted(a.rbegin(), a.rend())) {
    cout << ans << endl;
  } else {
    cout << -1 << endl;
  }
}
