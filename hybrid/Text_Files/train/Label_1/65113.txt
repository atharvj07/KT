def min_time_all_dishes_home(t, test_cases):
    results = []
    for i in range(t):
        n = test_cases[i][0]
        a = test_cases[i][1]
        b = test_cases[i][2]
        
        # Combine the a and b arrays into a single array of tuples for easier manipulation
        dishes = sorted([(a[i], b[i]) for i in range(n)], key=lambda x: max(x[0], x[1]), reverse=True)
        
        total_time = 0
        pick_up_time = 0
        for dish in dishes:
            delivery_time, pick_up_time_i = dish
            pick_up_time += pick_up_time_i
            total_time = max(total_time, min(pick_up_time, delivery_time))
        
        results.append(max(total_time, min(delivery_time, pick_up_time)))
    return results

# Example of usage
t = int(input().strip())
test_cases = []
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    b = list(map(int, input().strip().split()))
    test_cases.append((n, a, b))
    
results = min_time_all_dishes_home(t, test_cases)
for result in results:
    print(result)
