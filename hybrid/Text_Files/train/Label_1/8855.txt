#include <bits/stdc++.h>
using namespace std;
int arr[6][6];
int main() {
  int x = 0, y = 0;
  for (int i = 1; i <= 5; i++) {
    for (int j = 1; j <= 5; j++) {
      cin >> arr[i][j];
      if (arr[i][j]) x = i, y = j;
    }
  }
  int ans = abs(x - 3) + abs(y - 3);
  cout << ans << endl;
  return 0;
}
