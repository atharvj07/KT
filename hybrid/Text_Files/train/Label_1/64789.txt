import sys
from collections import deque

def ccw(p1, p2, p3):
    return (p2[0] - p1[0]) * (p3[1] - p1[1]) - (p2[1] - p1[1]) * (p3[0] - p1[0])

def intersect(p1, p2, p3, p4):
    return ccw(p1, p2, p3) * ccw(p1, p2, p4) <= 0 and ccw(p3, p4, p1) * ccw(p3, p4, p2) <= 0

r, b = map(int, input().split())
spaceships = [tuple(map(int, input().split())) for _ in range(r)]
bases = [tuple(map(int, input().split())) for _ in range(b)]
g = [[] for _ in range(r + b + 1)]

for i in range(r):
    for j in range(b):
        if not any(intersect(spaceships[i], bases[j], p1, p2) for p1, p2 in zip(spaceships, bases) if p1 != spaceships[i] and p2 != bases[j]):
            g[i + 1] += [j + r + 1]
            g[j + r + 1] += [i + 1]

match = [0] * (r + b + 1)
visited = [False] * (r + b + 1)

def dfs(x):
    if visited[x]:
        return False
    visited[x] = True
    for y in g[x]:
        if match[y] == 0 or dfs(match[y]):
            match[y] = x
            match[x] = y
            return True
    return False

ans = 0
for i in range(1, r + 1):
    if dfs(i):
        ans += 1

print('Yes' if ans == r else 'No')
