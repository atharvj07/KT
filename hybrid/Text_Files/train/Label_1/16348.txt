#include <bits/stdc++.h>
using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  int n = a.size();
  vector<int> ca(26, 0), cb(26, 0);
  for (char c : a) ca[c - 'a']++;
  for (char c : b) cb[c - 'a']++;
  string ans = "";
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 26; j++) {
      if (ca[j] && (cb[j] == 0 || j < cb[j] - ca[j])) {
        ans += j + 'a';
        ca[j]--;
        break;
      }
    }
    for (int j = 25; j >= 0; j--) {
      if (cb[j] && (ca[j] == 0 || j > ca[j] - cb[j])) {
        ans += j + 'a';
        cb[j]--;
        break;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
