import java.util.*;

public class Main {

    static char[][] table;
    static int n;
    static int[][][] dp;
    static final int UNKNOWN = -1;
    static final int FIRST_WINS = 1;
    static final int SECOND_WINS = 2;
    static final int DRAW = 0;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        table = new char[n][n];
        for (int i = 0; i < n; i++) {
            String line = sc.next();
            for (int j = 0; j < n; j++) {
                table[i][j] = line.charAt(j);
            }
        }
        dp = new int[n][n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < 2; k++) {
                    dp[i][j][k] = UNKNOWN;
                }
            }
        }
        int result = solve(0, 0, 0);
        switch (result) {
            case FIRST_WINS:
                System.out.println("FIRST");
                break;
            case SECOND_WINS:
                System.out.println("SECOND");
                break;
            case DRAW:
                System.out.println("DRAW");
                break;
        }
    }

    static int solve(int i, int j, int turn) {
        if (i == n - 1 && j == n - 1) {
            return DRAW;
        }
        if (dp[i][j][turn] != UNKNOWN) {
            return dp[i][j][turn];
        }
        int result = UNKNOWN;
        if (turn == 0) {
            // First player's turn
            if (i + 1 < n && table[i + 1][j] >= table[i][j]) {
                result = Math.max(result, solve(i + 1, j, 1 - turn));
            }
            if (j + 1 < n && table[i][j + 1] >= table[i][j]) {
                result = Math.max(result, solve(i, j + 1, 1 - turn));
            }
        } else {
            // Second player's turn
            if (i + 1 < n && table[i + 1][j] >= table[i][j]) {
                result = Math.min(result, solve(i + 1, j, 1 - turn));
            }
            if (j + 1 < n && table[i][j + 1] >= table[i][j]) {
                result = Math.min(result, solve(i, j + 1, 1 - turn));
            }
        }
        if (result == UNKNOWN) {
            result = DRAW;
        }
        dp[i][j][turn] = result;
        return result;
    }
}
