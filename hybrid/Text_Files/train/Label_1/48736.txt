import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        boolean found = false;
        for (int d = -1000; d <= 1000 && !found; d++) {
            if (d == 0) continue;
            int pos = 0;
            for (int num : arr) {
                if (num / d > 0) pos++;
            }
            if (pos >= (n + 1) / 2) {
                System.out.println(d);
                found = true;
            }
        }
        if (!found) {
            System.out.println(0);
        }
    }
}
