#include <bits/stdc++.h>
using namespace std;

#define ll long long

int main() {
    ll n, k, d;
    cin >> n >> k >> d;
    vector<ll> a(n);
    for (ll &i : a) cin >> i;
    sort(a.begin(), a.end());
    bool ok = true;
    for (ll i = 0, j = 0; i < n; i = j) {
        while (j < n && a[j] <= a[i] + d) j++;
        if (j - i < k) {
            ok = false;
            break;
        }
    }
    cout << (ok ? "YES" : "NO") << endl;
}
