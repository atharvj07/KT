import sys

n, m, k = map(int, input().split())
grid = [list(map(int, input().split())) for _ in range(n)]

dp = [[[None] * (1<<16)  for _ in range(m)] for _ in range(n)]

def dfs(i, j, cur):
    if i == n or j == m:
        return 0 if cur == k else 1
    if dp[i][j][cur] is not None:
        return dp[i][j][cur]
    res = dfs(i+1, j, cur^grid[i][j])
    res += dfs(i, j+1, cur^grid[i][j])
    dp[i][j][cur] = res
    return res

print(dfs(0, 0, 0))
