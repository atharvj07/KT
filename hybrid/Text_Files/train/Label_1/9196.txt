#include <bits/stdc++.h>
using namespace std;
constexpr int MAX = 2e5 + 10;
int n,  k, a, b, q, d[MAX], ord[MAX], pre[MAX], suf[MAX], pos[MAX], que[MAX];
int main() {
  cin >> n >> k >> a >> b >> q;
  while (q--) {
    int t, p;
    cin >> t >> p;
    if (t == 1)
      ord[p] += p, d[p] = 1;
    else
      que[p] = 1;
  }
  for (int i = 1; i <= n; ++i) pre[i] = pre[i - 1] + ord[i], pos[i] = pos[i - 1] + d[i];
  for (int i = n; i; --i) suf[i] = suf[i + 1] + ord[i];
  for (int p = 1; p <= n - k + 1; ++p) {
    int l = p, r = p + k - 1, ans = pos[l - 1] + suf[r + 1];
    if (r <= n) ans += min(a * (r - l + 1), suf[l]);
    if (l > 1) ans += min(b * k, pre[l - 1]);
    if (que[p]) cout << ans << endl;
  }
}
