#include <cstdio>
#include <cstring>
#include <algorithm>

int main() {
    int n;
    scanf(" %d ", &n);
    char s[105];
    scanf(" %s ", s + 1);
    for (int i = 1; i < n; i++) {
        if (s[i] == '1' && s[i] == s[i + 1]) {
            s[i + 1] = '0';
        }
    }
    while (n > 1 && s[n] == '0') --n;
    s[n + 1] = 0;
    printf("%s\n", s + 1);
}
