import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[][] a = new int[n][m];
    int[][] b = new int[n][m];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        a[i][j] = sc.nextInt();
      }
    }
    List<String> ans = new ArrayList<>();
    boolean ok = true;
    for (int i = 0; i < n - 1; i++) {
      for (int j = 0; j < m - 1; j++) {
        if (a[i][j] + a[i + 1][j] + a[i][j + 1] + a[i + 1][j + 1] == 3) {
          ans.add((i + 1) + " " + (j + 1));
          b[i][j] = b[i + 1][j] = b[i][j + 1] = b[i + 1][j + 1] = 1;
        }
      }
    }
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (a[i][j] != b[i][j]) ok = false;
      }
    }
    System.out.println(ok ? ans.size() : -1);
    for (String s : ans) System.out.println(s);
  }
}
