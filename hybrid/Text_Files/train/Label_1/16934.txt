#include <bits/stdc++.h>

using namespace std;

int32_t main() {
  int t;
  cin >> t;

  while (t--) {
    int n, m, a, b;
    cin >> n >> m >> a >> b;
    vector<vector<int>> adj(n + 1);

    for (int i = 0; i < m; i++) {
      int x, y;
      cin >> x >> y;
      adj[x].push_back(y);
      adj[y].push_back(x);
    }

    vector<int> dist(n + 1, -1);
    queue<int> q;

    q.push(a);
    dist[a] = 0;

    while (!q.empty()) {
      int u = q.front();
      q.pop();

      for (int v : adj[u]) {
        if (dist[v] == -1) {
          dist[v] = dist[u] + 1;
          q.push(v);
        }
      }
    }

    int cnt = 0;

    for (int i = 1; i <= n; i++) {
      if (dist[i] == dist[b]) cnt++;
    }

    cout << cnt * (cnt - 1) / 2 << endl;
  }
}
