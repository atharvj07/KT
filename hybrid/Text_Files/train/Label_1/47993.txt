import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int[] nm = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int n = nm[0], m = nm[1];
        int[] a = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int s = Integer.parseInt(br.readLine());
        long[] q = new long[s + 1];
        int[][] qi = new int[s + 1][2];
        for (int i = 0; i < s; i++) {
            int[] in = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            qi[i] = in;
            q[i + 1] = Math.max(q[i], in[2] - a[in[0] - 1]);
        }
        long[] p = new long[n];
        for (int i = s; i > 0; i--) {
            for (int j = qi[i - 1][1]; j >= qi[i - 1][0]; j--) {
                p[j - 1] = Math.max(p[j - 1], q[i - 1] - a[j - 1]);
            }
        }
        for (int i = 1; i < n; i++) {
            q[1] = Math.max(q[1], p[i]);
        }
        // for (int i = 0; i <= s; i++) System.out.println(p[i]);
        Arrays.stream(q).limit(s).forEachOrdered(System.out::println);
    }
}
