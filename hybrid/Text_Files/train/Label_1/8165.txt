#include <bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int q, d;
        cin >> q >> d;
        vector<int> a(q);
        for (int i = 0; i < q; i++) cin >> a[i];
        for (int i = 0; i < q; i++) {
            int num = a[i];
            bool ok = false;
            while (num > 0) {
                if (num % 10 == d) ok = true;
                num /= 10;
            }
            if (ok) cout << "YES" << endl;
            else cout << "NO" << endl;
        }
    }
    return 0;
}
