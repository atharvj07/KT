#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    scanf("%d",&t);
    while(t--) {
        int n;
        scanf("%d",&n);
        vector<int> v(n);
        for(int i = 0; i < n; ++i) scanf("%d",&v[i]);
        sort(v.begin(),v.end());
        int ans = v.back();
        for(int i = 0; i < n - 1; ++i) ans = max(ans,v[i+1] - v[i]);
        printf("%d\n",ans);
    }
    return 0;
}
