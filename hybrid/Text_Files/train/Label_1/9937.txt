#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  string s;
  cin >> s;
  int ans = 0;
  vector<int> cnt(26);
  for (char c : s) cnt[c - 'a']++;
  int need[10] = {0, 0, 0, 1, 0, 1, 0, 0, 1, 0};
  while (cnt['n'] >= need[0] && cnt['i'] >= need[1] && cnt['n'] >= need[2] && cnt['e'] >= need[3] &&
         cnt['t'] >= need[4] && cnt['e'] >= need[5] && cnt['e'] >= need[6] && cnt['n'] >= need[7]) {
    ans++;
    for (int i = 0; i < 10; i++) cnt[need[i] + 'a']--;
  }
  cout << ans << endl;
  return 0;
}
