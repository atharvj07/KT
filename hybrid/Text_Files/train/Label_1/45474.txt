import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt(), k = sc.nextInt();
            System.out.println(Math.abs(k - n) / 2 + (Math.abs(k - n) % 2 != 0 ? 1 : 0));
        }
        sc.close();
    }
}
