from sys import stdin
read = lambda: stdin.readline().strip()

n, t = map(int, read().split())
s = read()
p = [1] * (26)
for c in range(ord('a'), ord('z') + 1): chr(c); p[c - ord('a')] = pow(2, c - ord('a'))

def f(l, r):
    if l >= r:
        return (r == l) * p[ord(s[l]) - ord('a')]

    sum = 0
    for m in range(l, r):
        if m != l: sum -= f(l, m)
        sum += f(m + 1, r)
        if sum == t: return sum
    return sum

print(('Yes', 'No')[f(0, n - 1) != t])
