#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

ll n, m, l, r, a, b, c, d;
ll ans;

int main() {
    cin >> a >> b >> c >> d >> l >> r;
    if (d > b)
        swap(a, c), swap(b, d), swap(l, r);
    if (l < c)
        ans += min(r, b) - max(l, c) + 1;
    if (b >= c) {
        if (d > b - c || a - c <= 0)
            ans += 1;
        else
            ans += (b - c) / (a - c) + 1;
    }
    cout << ans << endl;
    return 0;
}
