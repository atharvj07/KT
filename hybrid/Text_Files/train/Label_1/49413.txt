import java.util.*;
import java.io.*;


class Main {

    static int[] up;
    static int[] sz;
    static boolean[] mark;
    static ArrayList<Integer>[] g;
    static int n;

    public static void main(String[] args) throws IOException {
        BufferedReader r = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(r.readLine());
        g = new ArrayList[n];
        sz = new int[n];
        up = new int[n];
        Arrays.fill(up, -1);
        for (int i = 0; i < n; i++) g[i] = new ArrayList<>();;
        StringTokenizer t;
        for (int i = 1; i < n; i++) {
            t = new StringTokenizer(r.readLine());
            int a = Integer.parseInt(t.nextToken()) - 1;
            int b = Integer.parseInt(t.nextToken()) - 1;
            g[a].add(b);
            g[b].add(a);
        }
        mark = new boolean[n];
        dfs(0, -1, 0, sz);
        for (int i = 0; i < n; i++) mark[i] = false;

        int root = -1;
        for (int i = 0; i < n; i++) {
            if (sz[i] * 2 > n) {
                root = i;
                break;
            }
        }

        mark[root] = true;
        dfs2(root, -1, 0, up);

        for (int i = 0; i < n; i++) {
            mark[i] = false;
            up[i] = -1;
        }
        dfs2(root, -1, 0, up);

        int cnt = 0;
        LinkedList<Integer> q = new LinkedList<>();
        q.add(root);
        while (!q.isEmpty()) {
            int v = q.remove();
            for (int u : g[v]) {
                if (!mark[u]) {
                    if ((n - sz[v] - sz[u]) * 2 <= n) {
                        cnt++;
                    }
                    mark[u] = true;
                    q.add(u);
                }
            }
        }

        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (i == root) {
                if (cnt > 0) ans++;
                continue;
            }
            if (sz[i] * 2 > n) continue;
            if (up[i] != -1 && !mark[up[i]]) ans++;
        }

        for (int i = 0; i < n; i++) {
            System.out.print((ans > 0 ? 1 : 0) + " ");
        }
        System.out.println();
    }


    static void dfs(int v, int p, int dep, int[] sz) {
        Main.sz[v] = 1;
        for (int u : g[v]) {
            if (u == p || mark[u]) continue;
            dfs(u, v, dep + 1, sz);
            Main.sz[v] += Main.sz[u];
        }
    }

    static void dfs2(int v, int p, int dep, int[] up) {
        for (int u : g[v]) {
            if (u == p || mark[u]) continue;
            up[u] = v;
            dfs2(u, v, dep + 1, up);
        }
    }
}
