n, k, x = map(int, input().split())
a = list(map(int, input().split()))

# Calculate the number of times we need to add/subtract x to make a number zero or negative
operations_required = [max(0, (abs(ai) + x - 1) // x) for ai in a]

# Sort indices by the operations required in descending order
indices_sorted = sorted(range(n), key=lambda i: operations_required[i], reverse=True)

for i in indices_sorted:
    if k <= 0:
        break
    # Calculate the maximum operations that can be applied to the current element
    max_ops = min(k, operations_required[i])
    # Apply operations
    if a[i] > 0:
        a[i] -= x * max_ops
    else:
        a[i] += x * max_ops
    # Decrease the remaining operations
    k -= max_ops

print(*a)
