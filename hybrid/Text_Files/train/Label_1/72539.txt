n,m=map(int,input().split())
s=[input() for i in range(n)]
a=list(map(int,input().split()))
for i in range(m):
  mx=0
  c=0
  for j in range(n):
    if s[j][i]==chr(ord('A')+mx):
      c+=1
    elif s[j][i]!=chr(ord('A')+mx) and a[i]>a[mx]:
      c=1
      mx=j
  a[mx]=max(a[mx],a[i])
  mx=0
  for j in range(n):
    if s[j][i]==chr(ord('A')+mx):
      a[mx]=0
      break
    elif s[j][i]!=chr(ord('A')+mx) and a[i]>a[mx]:
      a[mx]=0
      mx=j
      break
print(sum(a))
