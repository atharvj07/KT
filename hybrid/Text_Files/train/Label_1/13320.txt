#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int mxn = 1e5 + 5;
int n, f, k[mxn], l[mxn];
ll dp[mxn][2];

ll fun(int pos, bool flg) {
    if (pos == n + 1) return 0;
    ll &ret = dp[pos][flg];
    if (ret != -1) return ret;
    ret = fun(pos + 1, flg);
    ll p = 0;
    if (flg) p = k[pos];
    else {
        if (k[pos] >= l[pos]) p = l[pos];
        else p = k[pos];
    }
    ret = max(ret, fun(pos + 1, !flg) + p);
    return ret;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cin >> n >> f;
    for (int i = 1; i <= n; i++) cin >> k[i] >> l[i];
    memset(dp, -1, sizeof(dp));
    cout << fun(1, 1) << endl;
}
