#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 1; i <= n; ++i)
typedef long long ll;
const long long mod = 1e9 + 7;
int main() {
  int a, b, x;
  cin >> a >> b >> x;
  string ans;
  while (a > 0) {
    ans += '0';
    --a;
    if (b > 0 && x > 0) {
      ans += '1';
      --b;
      --x;
    }
  }
  while (b > 0) {
    ans += '1';
    --b;
  }
  cout << ans << endl;
  return 0;
}
