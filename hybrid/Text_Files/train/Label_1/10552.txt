#include<iostream>
using namespace std;

int main()
{
    int n,m;
    cin>>n>>m;
    int a[n][m];
    for(int i=0;i<n;i++)
    {
        for(int j=0;j<m;j++)
        {
            cin>>a[i][j];
        }
    }
    int cnt=0;
    for(int i=0;i<n;i++)
    {
     int prv=-1;
     for(int j=0;j<m;j++)
     {
        if(a[i][j]==prv)continue;
        cnt++;
        prv=a[i][j];
     }
    }
    for(int i=0;i<m;i++)
    {
        int prv=-1;
        for(int j=0;j<n;j++)
        {
            if(a[j][i]==prv)continue;
            cnt++;
            prv=a[j][i];
        }
    }
    cout<<cnt<<endl;
    return 0;
}
