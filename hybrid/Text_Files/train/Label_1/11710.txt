#include<bits/stdc++.h>
using namespace std ;
#define ll long long 
ll m , n , l , r , mid , ans ;
ll f(ll x , ll n){
    return (x - 1) * (x - 1) <= n ;
}
ll bs(ll n){
    l = 0 ;
    while (l <= r)
    {
        mid = (l + r) / 2 ;
        if (f(mid , n))
        {
            ans = mid ;
            l = mid + 1 ;
        }
        else
        {
            r = mid - 1 ;
        }
    }
    return ans ;
}
int main(){
    ios_base :: sync_with_stdio(false) ;
    cin.tie(NULL) ;
    cin >> n ;
    m = n ;
    r = 1e5 ;
    ll res = bs(n) ;
    ll powr_2 = (1ll << res) - 1 ;
    ll gd = powr_2 * powr_2 ;
    cout << gd << endl ;
}
