#include <bits/stdc++.h>

typedef long long ll;
typedef unsigned long long ull;
typedef long double ld;

#define endl '\n'

using namespace std;

const ll maxn = 1e6 + 1;

ll n, m, bit[maxn], a[maxn];

void update(ll x, ll val) {
	for (; x <= n; x += (x & (-x))) bit[x] = __gcd(bit[x], val);
}

ll get(ll x) {
	ll ans = 1;
	for (; x > 0; x -= (x & (-x))) ans = __gcd(ans, bit[x]);
	return ans;
}

void solve() {
	cin >> n >> m;
	for (int i = 1; i <= n; i++) {
		cin >> a[i];
		update(i, a[i]);
	}
	ll l, ans = 1, ans_len = 0;
	for (int i = n; i >= 1 && l <= m; i--) {
		ll now = get(i);
		if (now <= m) {
			if (i - now + 1 > ans_len) {
				ans_len = i - now + 1;
				l = now;
				ans = i;
			}
		}
		update(i, 0);
	}
	cout << l << ' ' << ans_len << endl;
	for (int i = ans - ans_len + 1; i <= ans; i++) {
		cout << i << ' ';
	}
	cout << endl;
}

int main() {
	ios_base::sync_with_stdio(0);
	cin.tie(0);
	cout.tie(0);

	solve();

	return 0;
}
