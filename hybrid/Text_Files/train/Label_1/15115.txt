#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x, y;
  cin >> n >> x >> y;
  vector<int> a(n);
  for (int& ai : a) cin >> ai;
  sort(a.begin(), a.end());
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] <= x) {
      a[i] = 0;
      ans++;
    } else
      break;
  }
  for (int i = n - 1; i >= 0; i--) {
    if (a[i] > 0) {
      if (a[i] % (x + y) == 0 && a[i] / (x + y) > 0) {
        if (a[i] / (x + y) % 2 == 1) ans++;
      } else if (a[i] % (x + y) > x)
        ans++;
    }
  }
  cout << ans << endl;
}
