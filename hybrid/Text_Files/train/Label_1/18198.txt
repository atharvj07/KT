#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
const ll mod = 1e9 + 7;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> adj(n + 1);
    vector<ll> v(n + 1), t(n + 1);
    for (int i = 1; i <= n; i++) cin >> v[i];
    for (int i = 1; i <= n; i++) cin >> t[i];
    for (int i = 0; i < m; i++) {
      int x, y;
      cin >> x >> y;
      adj[x].push_back(y);
      adj[y].push_back(x);
    }
    bool ok = true;
    ll sum = 0;
    for (int i = 1; i <= n; i++) sum += v[i] - t[i];
    if (sum % n != 0) ok = false;
    else {
      ll add = sum / n;
      for (int i = 1; i <= n; i++) {
        if (v[i] - add != t[i]) ok = false;
      }
    }
    if (ok)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
