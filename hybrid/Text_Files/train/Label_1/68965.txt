import sys

n, m = map(int, input().split())
similar = [[] for i in range(n+1)]
for _ in range(m):
    u, v = map(int, input().split())
    similar[u].append(v)
    similar[v].append(u)

dp = [[-1]*(1<<n) for i in range(n+1)]
def solve(i, mask):
    if i == n+1:
        return 1
    if dp[i][mask] != -1:
        return dp[i][mask]
    dp[i][mask] = 0
    for j in similar[i]:
        if (mask & (1<<j)) > 0:
            continue
        dp[i][mask] += solve(i+1, mask|(1<<i)|(1<<j))
    return dp[i][mask]

print(solve(1, 0))
