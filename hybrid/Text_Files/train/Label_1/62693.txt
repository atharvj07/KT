import math

def calculate_angle(x, y):
    return math.degrees(math.atan2(y, x)) % 360

n = int(input())
angles = []

for _ in range(n):
    x, y = map(int, input().split())
    angles.append(calculate_angle(x, y))

angles.sort()

max_angle_diff = angles[0] + 360 - angles[-1]
for i in range(1, n):
    max_angle_diff = max(max_angle_diff, angles[i] - angles[i-1])

print(360 - max_angle_diff)
