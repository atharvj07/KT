import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] arr = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = sc.nextInt();
            }
        }
        int[] dp = new int[m + 1];
        for (int i = 0; i < n; i++) {
            Arrays.sort(arr[i]);
            int[] tmp = new int[m + 1];
            for (int j = 1; j <= m; j++) {
                for (int k = j; k <= m; k++) {
                    tmp[j] = Math.max(tmp[j], dp[k] + arr[i][j - 1]);
                }
            }
            dp = tmp;
        }
        System.out.println(dp[1]);
    }
}
