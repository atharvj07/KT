s = input().strip()
n = len(s)
result = 0

for l in range(n):
    for r in range(l, n):
        found = False
        for x in range(l, r):
            for k in range(1, (r - x) // 2 + 1):
                if x + 2*k <= r and s[x] == s[x+k] == s[x+2*k]:
                    found = True
                    break
            if found:
                break
        if found:
            result += 1

print(result)
