#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main(){
    int n,k;
    cin>>n>>k;
    if(k==1) cout<<5<<endl<<"1 2 3 5"<<endl;
    else{
        int t=182*n;
        cout<<t<<endl;
        for(int i=1;i<=n;i++){
            cout<<k<<' '<<k*2<<' '<<k*4<<' '<<t<<endl;
            t-=182;
        }
    }
    return 0;
}
