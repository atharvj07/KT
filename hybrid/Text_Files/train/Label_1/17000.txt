#include<bits/stdc++.h>
using namespace std;
const int maxn = 1e4 + 10;
char expr[maxn], tmp[maxn];
int d[maxn], l, r, mx;

int solve(int l, int r, int p, int m) {
    if(l > r) return 0;
    int ans = INT_MIN;
    if(l == r) return d[l];
    else if(l + 1 == r) {
        if(p > 0) return d[l] + d[r];
        else return d[l] - d[r];
    } 
    else {
        for(int i = l; i <= r; i++) {
            if(tmp[i] == '+') {
                ans = max({ans, solve(l, i - 1, p - 1, m) + solve(i + 1, r, p, m), solve(l, i - 1, p, m) - solve(i + 1, r, p, m)});
            } else {
                ans = max({ans, solve(l, i - 1, p, m + 1) + solve(i + 1, r, p, m), solve(l, i - 1, p, m) - solve(i + 1, r, p, m)});
            }
        }
        return ans;
    }
}

int main() {
    scanf("%s", expr);
    int len = strlen(expr);
    scanf("%d %d", &l, &r);
    int cnt = 0;
    for(int i = 0; i < len; i++) {
        if(expr[i] == '?' || expr[i] == '+') tmp[cnt++] = expr[i];
    }
    for(int i = 0; i < len; i++) {
        if(expr[i] >= '0' && expr[i] <= '9') d[cnt++] = expr[i] - '0';
    }
    printf("%d\n", solve(0, cnt - 1, l, r));
    return 0;
}
