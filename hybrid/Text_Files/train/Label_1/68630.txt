n, m, k = map(int, input().split())
alice_fish = list(map(int, input().split()))
bob_fish = list(map(int, input().split()))

max_alice_fish = max(alice_fish)
max_bob_fish = max(bob_fish)

if max_alice_fish > max_bob_fish:
    print("YES")
else:
    alice_fish_count = {i: alice_fish.count(i) for i in set(alice_fish)}
    bob_fish_count = {i: bob_fish.count(i) for i in set(bob_fish)}

    possible = False
    for fish_type in sorted(alice_fish_count.keys(), reverse=True):
        if fish_type not in bob_fish_count or alice_fish_count[fish_type] > bob_fish_count[fish_type]:
            possible = True
            break

    print("YES" if possible else "NO")
