#include <iostream>
#include <array>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  std::array<std::array<int, 2>, 4> remaining;
  for (int i = 0; i < n; ++i) {
    std::cin >> remaining[i][0] >> remaining[i][1];
  }
  std::sort(remaining.begin(), remaining.begin() + n,
            [](const auto &a, const auto &b) { return a[0] < b[0]; });
  std::sort(remaining.begin(), remaining.begin() + n,
            [](const auto &a, const auto &b) { return a[1] < b[1]; });
  if (remaining[1][0] == remaining[2][0] || remaining[1][1] == remaining[2][1]) {
    std::cout << -1;
    return 0;
  }
  std::cout << (remaining[1][0] - remaining[0][0]) *
                   (remaining[1][1] - remaining[0][1]);
  return 0;
}
