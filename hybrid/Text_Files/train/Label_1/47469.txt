import java.util.*;

public class Main {

    private static final long MOD = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int c = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        int[][] dp = new int[n][c + 1];
        dp[n - 1][arr[n - 1]] = 1;
        for (int i = n - 2; i >= 0; i--) {
            for (int j = 1; j <= c; j++) {
                dp[i][j] = (dp[i + 1][j] + ((j == arr[i]) ? 1 : 0)) % (int) MOD;
            }
        }

        long[] p = new long[n];
        p[0] = 1;
        for (int i = 1; i < n; i++) {
            p[i] = (2 * p[i - 1]) % MOD;
        }

        long[] s = new long[n + 1];
        for (int i = 0; i < n; i++) {
            s[i + 1] = (s[i] + p[i] * dp[i][c]) % MOD;
        }

        for (int i = 0; i <= n; i++) {
            System.out.print(s[i] + " ");
        }
        System.out.println();
    }
}
