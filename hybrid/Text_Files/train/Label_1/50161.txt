import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class JOIWalk {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int t = scanner.nextInt();
        int q = scanner.nextInt();

        List<Person> people = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int ai = scanner.nextInt();
            int di = scanner.nextInt();
            people.add(new Person(ai, di));
        }

        List<Integer> importantPeople = new ArrayList<>();
        for (int i = 0; i < q; i++) {
            int xi = scanner.nextInt();
            importantPeople.add(xi);
        }

        for (int xi : importantPeople) {
            int position = people.get(xi - 1).getPosition(t);
            System.out.println(position);
        }
    }

    private static class Person {
        private int ai;
        private int di;

        public Person(int ai, int di) {
            this.ai = ai;
            this.di = di;
        }

        public int getPosition(int t) {
            if (di == 1) {
                return ai + t;
            } else {
                return ai - t;
            }
        }
    }
}
