#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for(auto &p : a) {
        cin >> p;
    }
    int ans = 0, unread = 0;
    for(int i = 0; i < n; i++) {
        if(a[i]) unread++;
        else continue;
        int j = i + 1, k = i - 1, cnt = 0;
        while(j < n && a[j] == 0) {
            cnt++;
            j++;
        }
        while(k >= 0 && a[k] == 0) {
            cnt++;
            k--;
        }
        if(a[i]) {
            if(unread == 1) break;
            ans += min(cnt, 1);
        }
        unread--;
    }
    cout << ans << endl;
    return 0;
}
