def solve():
    q = int(input())
    s, t = "What are you doing at the end of the world? Are you busy? Will you save us?",\
          "What are you doing while sending "
    for i in range(q):
        n, k = map(int, input().split())
        if n == 0:
            print('.' if k > len(s) else s[k - 1])
        else:
            print('.' if k > len(t) + n * len(s) else t[k - 1 - n * len(s):k - n * len(s)] + s[k - 1 - n * len(s) - len(t):k - 1 - n * len(s)])
solve()
