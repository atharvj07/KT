#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> answers(n);
  for (int i = 0; i < n; i++) {
    cin >> answers[i];
  }
  vector<int> points(m);
  for (int i = 0; i < m; i++) {
    cin >> points[i];
  }
  int maxScore = 0;
  for (int i = 0; i < m; i++) {
    int count[5] = {0, 0, 0, 0, 0};
    for (int j = 0; j < n; j++) {
      count[answers[j][i] - 'A']++;
    }
    int maxCount = 0;
    for (int j = 0; j < 5; j++) {
      maxCount = max(maxCount, count[j]);
    }
    maxScore += points[i] * maxCount;
  }
  cout << maxScore << endl;
  return 0;
}
