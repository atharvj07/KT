#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll MOD = 1e9 + 7;
ll n, a[100005], ans[100005];
int main() {
  cin >> n;
  for (ll i = 1; i <= n; i++) cin >> a[i];
  for (ll i = 1; i <= n; i++) {
    ll cnt = 0;
    for (ll j = 1; j <= i; j++)
      if (a[j] % 2) cnt++;
    if (cnt % 2 == 0)
      ans[i] = 1;
    else
      ans[i] = 2;
    a[i] = (a[i] + 1) / 2;
  }
  for (ll i = 1; i <= n; i++)
    cout << ans[i] << endl;
  return 0;
}
