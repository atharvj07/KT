import sys

n = int(input())
g = [[]]
for i in range(n + 1):
    g.append([])
for i in range(n - 1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

ans = 0
for i in range(1, n + 1):
    for j in g[i]:
        for k in g[j]:
            if k > i:
                ans += 1
print(ans)
