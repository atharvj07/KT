#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
int n, m, a, b[MAXN], p[MAXN], f[MAXN], prv[MAXN];

int main() {
  cin >> n >> m >> a;
  for (int i = 1; i <= n; i++) cin >> b[i];
  for (int i = 1; i <= m; i++) cin >> p[i];
  sort(p + 1, p + m + 1);
  for (int i = 2; i <= m; i++) {
    p[i] += p[i - 1];
  }
  for (int i = 1; i <= n; i++) {
    int lo = 1, hi = m, ans = 0;
    while (lo <= hi) {
      int mid = (lo + hi) / 2;
      if (p[mid] <= b[i] + a) {
        ans = mid;
        lo = mid + 1;
      } else {
        hi = mid - 1;
      }
    }
    f[i] = ans, prv[i] = ans ? prv[ans] : 0;
    prv[ans] = i;
  }
  int r = 0, s = 0;
  for (int i = 1; i <= n; i++) {
    r = max(r, f[i]);
    if (f[i]) s += b[i] + a - p[f[i]];
  }
  cout << r << " " << s << endl;
  return 0;
}
