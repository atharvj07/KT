#include<bits/stdc++.h>
using namespace std;
int32_t main()
{
 int t;
 cin >> t;
 while(t--)
 {
  int64_t n;
  cin >> n;
  int count = 0;
  while(n > 0)
  {
    if(n % 2 == 0)
    {
      n /= 2;
      count++;
    }
    else
    {
      n -= 1;
      count++;
    }
  }
  cout << count << endl;
 }
}
