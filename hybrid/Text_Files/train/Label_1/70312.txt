n, m = map(int, input().split())
a = [input() for _ in range(n)]
mi = min(n, m)
ans = 0
for i in range(mi):
    for j in range(mi):
        if sum([a[k][j] == 'B' for k in range(i, n)]) < j + 1 or sum([a[i][k] == 'B' for k in range(j, m)]) < i + 1:
            continue
        c = 0
        for k in range(i, n):
            for l in range(j, m):
                c += (a[k][l] != 'B')
        ans = max(ans, c)
if ans != 0:
    print(ans)
else:
    print(-1)
