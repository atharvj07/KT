#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
int main() {
    int n;
    cin >> n;
    vector<int> s(n), t(n);
    for(int i=0; i<n; i++) cin >> s[i];
    for(int i=0; i<n; i++) cin >> t[i];
    int i=0,j=0,ans=0;
    for(i=0; i<n; i++) {
        if(t[j]==s[i]) j++;
        if(j==n) {
            ans=i;
            break;
        }
    }
    while(i<n) ans++, i++;
    cout << n-ans << endl;
}
