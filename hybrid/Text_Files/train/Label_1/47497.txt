import java.util.*;
class Main {
    public static void main(String[] x) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt(), c = 0;
            int[] per = new int[n];
            for (int i = n; i >= 1; i--) {
                if (c + 1 <= i) { per[n - i] = c + 1; c++; }
                else per[n - i] = i;
            }
            for (int i = 0; i < n; i++) System.out.print(per[i] + " ");
            System.out.println();
        }
    }
}
