x, y, n = map(int, input().split())
a, b = x // y, 1
k = y // x
for i in range(1, n + 1):
    if i > y or (i + k - 1) // k > n:
        continue
    if abs(x * (i + k - 1) // k - y * i) < abs(x * b - y * a) or (abs(x * (i + k - 1) // k - y * i) == abs(x * b - y * a) and i + k - 1 < b):
        a, b = i, i + k - 1
print(f'{a}/{b}')
