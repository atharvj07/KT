#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int mxN = 1010;
ll dp[mxN][mxN], a[mxN][mxN];

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
    }
    dp[1][1] = a[1][1];
    dp[n][1] = a[n][1];
    for (int j = 2; j <= m; j++) { dp[1][j] = dp[1][j - 1] + a[1][j]; }
    for (int i = 2; i <= n; i++) { dp[i][1] = dp[i - 1][1] + a[i][1]; }
    for (int i = 2; i <= n; i++) {
        for (int j = 2; j <= m; j++) {
            dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]) + a[i][j];
        }
    }
    for (int i = n - 1; i >= 1; i--) {
        for (int j = m - 1; j >= 1; j--) {
            dp[i][j] = max(dp[i + 1][j], dp[i][j + 1]) + a[i][j];
        }
    }
    ll mx = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) { mx = max(mx, dp[i][j]); }
    }
    cout << mx << endl;
    return 0;
}
