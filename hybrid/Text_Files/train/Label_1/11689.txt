#include <iostream>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;
    vector<long long> b(7);
    while (t--) {
        for (int i = 0; i < 7; i++) {
            cin >> b[i];
        }
        cout << b[0] << ' ' << b[3] - b[0] << ' ' << b[6] - b[3] << '\n';
    }
}
