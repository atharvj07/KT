#include <iostream>
#include <vector>
using namespace std;

int main() {
  int k;
  cin >> k;
  vector<vector<int>> nums(k);
  vector<int> sum(k, 0);
  for (int i = 0; i < k; i++) {
    int n;
    cin >> n;
    nums[i].resize(n);
    for (int j = 0; j < n; j++) {
      cin >> nums[i][j];
      sum[i] += nums[i][j];
    }
  }
  bool found = false;
  for (int i = 0; i < k; i++) {
    for (int j = 0; j < k; j++) {
      if (i == j) continue;
      for (int x = 0; x < nums[i].size(); x++) {
        for (int y = 0; y < nums[j].size(); y++) {
          if (sum[i] - nums[i][x] == sum[j] - nums[j][y]) {
            cout << "YES" << endl;
            cout << i + 1 << " " << x + 1 << endl;
            cout << j + 1 << " " << y + 1 << endl;
            found = true;
            break;
          }
        }
        if (found) break;
      }
      if (found) break;
    }
    if (found) break;
  }
  if (!found) cout << "NO" << endl;
}
