#include<bits/stdc++.h>
using namespace std;
int a[2001],b[2001],c[2001],d[2001],n;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cin>>n;
    for(int i=1;i<=2*n;i++)
    {
        cin>>a[i];
        b[a[i]]=i;
    }
    for(int i=1;i<=2*n;i++)
    {
        cin>>c[i];
        d[c[i]]=i;
    }
    int ans=0;
    for(int i=1;i<=2*n;i++)
        if(b[i]!=d[i])
            ans++;
    if(ans&1)
        ans=-1;
    cout<<ans/2<<endl;
    return 0;
}
