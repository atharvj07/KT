#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ii pair<int,int>
#define vii vector<ii>
#define pb push_back
#define fi first
#define se second
const int N=1e6+5;
const int mod=1e9+7;
int n,m,k,ans;
string s,t;
int pr[N],suf[N];
int solve(string &s){
	int n=(int)s.size(),ans=0;
	for(int i=0,j=0;i<n;){
		if(s[i]==s[j]){
			j++;
			if(j==n)ans++;
		}else{
			i=j;
			j=0;
		}
	}
	return ans;
}
int main(){
	cin>>s>>t;
	n=(int)s.size();
	m=(int)t.size();
	s=" "+s;
	t=" "+t;
	pr[0]=1;
	for(int i=1;i<=m;i++){
		if(s[i]==t[i])pr[i]=pr[i-1]+1;
		else pr[i]=pr[i-1];
	}
	suf[m+1]=1;
	for(int i=m;i>=1;i--){
		if(s[i]==t[i])suf[i]=suf[i+1]+1;
		else suf[i]=suf[i+1];
	}
	for(int i=1;i<=m;i++){
		if(pr[i-1]+suf[i+1]==n)ans++;
	}
	cout<<ans<<endl;
	return 0;
}
