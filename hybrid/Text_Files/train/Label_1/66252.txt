n, m = map(int, input().split())
p = list(map(int, input().split()))

# Record which pupils can swap places
can_swap = set()
for _ in range(m):
    u, v = map(int, input().split())
    can_swap.add((u, v))

# Count how many pupils Nastya can move forward by swapping
move_forward = 0
for i in range(n-2, -1, -1):
    # If the current pupil can swap with the next one (initially Nastya), increase count
    if (p[i], p[i+1]) in can_swap or p[i+1] == n:
        move_forward += 1
        p[i], p[i+1] = p[i+1], p[i]  # Perform the swap
    else:
        break  # Can't move further

print(move_forward)
