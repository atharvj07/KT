#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        long long ans = 0;
        for (long long i = 2, j = 9, cnt = 0; i <= n + 1; i++, j--) {
            ans = ans * 10 + j;
            cnt++;
            if (cnt == i - 1) {
                cnt = 0;
                j = min(j + 1, 9LL);
            }
        }
        cout << ans << endl;
    }
    return 0;
}
