#include <bits/stdc++.h>
using namespace std;
const int maxN = 300001;
int n;
int pos[maxN], col[maxN];
long long sol, cnt[3];

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> pos[i] >> col[i];
    cnt[col[i] - 'G']++;
  }
  sort(pos, pos + n);
  for (int i = 0; i < n; i++) {
    cnt[col[i] - 'G']--;
    if (cnt[2] == 0 && cnt[1] == 0) {
      sol += pos[i] - pos[i - 1];
    }
  }
  cout << sol << endl;
  return 0;
}
