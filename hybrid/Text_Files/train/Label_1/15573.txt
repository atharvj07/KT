#include<bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    int a[n], cnt = 0, sum = 0;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (a[i] > 0)
            cnt++;
        sum += a[i];
    }
    int x = 0;
    for (int i = 0; i < n; i++) {
        if (sum - a[i] > 0 && (sum - a[i]) % cnt == 0)
            x++;
    }
    cout << x + 1 << "\n";
    return 0;
}
