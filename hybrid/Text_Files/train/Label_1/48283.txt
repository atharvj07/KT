import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    int[] v = new int[n];
    for (int i = 0; i < n; i++) {
      v[i] = sc.nextInt();
    }
    int ans = 0;
    int sum = 0;
    int l = 0;
    int r = n - 1;
    for (int i = 0; i < k; i++) {
      if (l > r) {
        break;
      }
      if (Math.abs(v[l]) > Math.abs(v[r])) {
        if (v[l] > 0) {
          sum += v[l];
          ans = Math.max(ans, sum);
        } else {
          if (sum > 0) {
            sum = 0;
          } else {
            sum += v[l];
            ans = Math.max(ans, sum);
          }
        }
        l++;
      } else {
        if (v[r] > 0) {
          sum += v[r];
          ans = Math.max(ans, sum);
        } else {
          if (sum > 0) {
            sum = 0;
          } else {
            sum += v[r];
            ans = Math.max(ans, sum);
          }
        }
        r--;
      }
    }
    System.out.println(ans);
  }
}
