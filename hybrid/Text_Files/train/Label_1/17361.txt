#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> prices(n);
  for (int& x : prices) cin >> x;
  int q;
  cin >> q;
  vector<int> money(q);
  for (int& x : money) cin >> x;
  for (int i = 0; i < q; i++) {
    int ans = 0;
    for (int j = 0; j < n; j++) {
      if (money[i] >= prices[j]) ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
