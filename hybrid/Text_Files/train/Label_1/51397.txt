import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {
  public static void main(String[] args) {
    int[][] grid;
    int h, w, n, x1, y1, x2, y2;
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    try {
      String[] str = br.readLine().split(" ");
      w = Integer.parseInt(str[0]);
      h = Integer.parseInt(str[1]);
      n = Integer.parseInt(str[2]);
      grid = new int[h + 2][w + 2];
      str = br.readLine().split(" ");
      x1 = Integer.parseInt(str[0]);
      y1 = Integer.parseInt(str[1]);
      x2 = Integer.parseInt(str[2]);
      y2 = Integer.parseInt(str[3]);
      for (int i = 0; i < n; i++) {
        str = br.readLine().split(" ");
        int x = Integer.parseInt(str[0]);
        int y = Integer.parseInt(str[1]);
        int t = Integer.parseInt(br.readLine());
        String s = br.readLine();
        move(x, y, t, s, grid);
      }
      if (y1 == y2) {
        for (int i = Math.min(x1, x2) + 1; i <= Math.max(x1, x2); i++) grid[y1][i]++;
      }
      if (x1 == x2) {
        for (int j = Math.min(y1, y2) + 1; j <= Math.max(y1, y2); j++) grid[j][x1]++;
      }
      int min = Integer.MAX_VALUE;
      for (int i = 0; i < h + 2; i++)
        for (int j = 0; j < w + 2; j++) min = Math.min(min, grid[i][j]);
      System.out.println(min);
    } catch (IOException e) {
      e.printStackTrace();
    }
  }

  public static void move(int x, int y, int t, String s, int[][] grid) {
    x++;
    y++;
    while (t-- > 0)
      for (char ch : s.toCharArray()) {
        if (ch == 'U' && y > 0) y--;
        if (ch == 'D' && y < grid.length - 1) y++;
        if (ch == 'L' && x > 0) x--;
        if (ch == 'R' && x < grid[y].length - 1) x++;
        grid[y][x]++;
      }
  }
}
