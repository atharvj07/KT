import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    vis = [False] * n
    for i in range(n):
        j = i
        cnt = 0
        while not vis[j]:
            vis[j] = True
            cnt += 1
            j = (j + a[j % n] + n) % n
        if cnt != n:
            print("NO")
            break
    else:
        print("YES")
