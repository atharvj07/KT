n = int(input())
tree = [None] * (n + 1)
children = [[] for _ in range(n + 1)]
respect = [0] * (n + 1)

for i in range(1, n + 1):
    p, c = map(int, input().split())
    tree[i] = (p, c)
    respect[i] = c
    if p != -1:
        children[p].append(i)

to_delete = []
for i in range(1, n + 1):
    if respect[i] == 1:  # If the node doesn't respect its parent
        all_children_disrespect = all(respect[child] == 1 for child in children[i])
        if all_children_disrespect:
            to_delete.append(i)

if not to_delete:
    print(-1)
else:
    print(' '.join(map(str, sorted(to_delete))))
