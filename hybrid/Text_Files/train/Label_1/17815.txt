#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  long long sum = 0;
  for (int i = 0; i < n - 1; i++) {
    cout << sum + a[i] << ' ';
    sum += a[i];
  }
  cout << sum + a[n - 1] << endl;
  return 0;
}
