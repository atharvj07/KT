n=int(input())
l=[]
for i in range(n):l.append(list(map(int,input().split())))
l.sort(key=lambda x:x[0])
for i in l:
    if not n:break
    if i[0]<=l[0][0]-1:
        l[0][0]=i[0]
        l.sort(key=lambda x:x[0])
    elif i[0]==l[0][0]:
        if i[1]<l[0][1]:
            l.remove(i)
    else:
        l[0][0]+=l[0][1]
        n-=1
print(l[0][0]-1)
