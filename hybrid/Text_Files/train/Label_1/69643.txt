s=input();t=input()
if len(t)==1:print(0);exit()
ans=0
for i in range(len(t)):
    r0=t[i]
    for j in range(len(t)):
        if j==i or t[j]==r0:continue
        r1=t[j]
        ok=1
        for k in range(len(s)):
            if s[k]=='0':
                if k+len(r0)>len(t) or t[k:k+len(r0)]!=r0:ok=0;break
            elif s[k]=='1':
                if k+len(r1)>len(t) or t[k:k+len(r1)]!=r1:ok=0;break
        ans+=ok
print(ans)
