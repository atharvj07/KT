import sys

for _ in range(int(input())):
    n = int(input())
    if n % 2 != 0:
        print(-1)
    else:
        x = n // 2
        if x % 3 == 0:
            a = x // 3
            b = c = 0
        else:
            a = x // 3
            c = 1
            b = (x - 3 - c * 5) // 5
        print(a, b, c)
