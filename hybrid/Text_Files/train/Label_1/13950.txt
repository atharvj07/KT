#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
const int maxn = 2e5 + 5;
ll fact[maxn], finv[maxn];

ll powmod(ll x, ll p) {
    if (p == 0) return 1;
    if (p == 1) return x;
    ll v = powmod(x, p / 2) % mod;
    v = (v * v) % mod;
    if (p % 2 != 0) v = (v * x) % mod;
    return v;
}

void precal() {
    fact[0] = fact[1] = finv[0] = finv[1] = 1;
    for (int i = 2; i < maxn; i++) {
        fact[i] = (fact[i - 1] * i) % mod;
        finv[i] = powmod(fact[i], mod - 2);
    }
}

ll nck(int n, int k) {
    if (n < k || k < 0 || n < 0) return 0;
    return ((fact[n] * finv[n - k]) % mod * finv[k]) % mod;
}

int main() {
    precal();
    int n, k;
    cin >> n >> k;
    vector<ll> w(n);
    for (int i = 0; i < n; i++) cin >> w[i];
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        ll v = (w[i] * nck(n - 1, k - 1)) % mod;
        ans += v;
        ans = ans % mod;
    }
    cout << ans << "\n";
    return 0;
}
