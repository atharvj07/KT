#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int n, a, b, v[4]; ll ans = LLONG_MAX;
int main(){
    cin >> n >> a;
    for(int j, k = 180 / n; k <= 360; k += 360 / n)
        for(int i = 2; i <= n; i++)
            for(j = 1; j < i; j++)
                if(abs(k - a) < ans){
                    ans = abs(k - a);
                    v[1] = i; v[2] = j; v[3] = (j + 1) % n + 1;
                }
    for(int i = 1; i <= 3; i++) cout << v[i] << ' ';
}
