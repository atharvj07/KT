#include <iostream>
using namespace std;

string a[301];
int main() {
  int n, i, j;
  char c;
  cin >> n;
  for (i = 0; i < n; i++) { cin >> a[i]; }
  c = a[0][0];
  for (i = 0; i < n && c == a[i][i] && c == a[i][n - i - 1]; i++) {}
  if (i != n) {
    cout << "NO"
         << "\n";
    return 0;
  };
  for (i = 0; i < n; i++)
    for (j = 0; j < n; j++)
      if (i != j && i != n - j - 1 && c != a[i][j]) {
        cout << "NO"
             << "\n";
        return 0;
      };
  cout << "YES"
       << "\n";
  return 0;
}
