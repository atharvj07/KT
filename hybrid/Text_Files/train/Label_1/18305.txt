#include <iostream>
#include <string>
#include <set>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        set<string> s;
        for (int i = 0; i < n; i++) {
            string str;
            cin >> str;
            s.insert(str);
        }
        cout << (s.size() == 1 ? "YES" : "NO") << endl;
    }
    return 0;
}
