#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> ids(n);
  for (ll i = 0; i < n; i++) { cin >> ids[i]; }
  ll ans;
  ll cur = 0;
  for (ll i = 0; i <= n; i++) {
    if (cur < k && cur + i + 1 <= k)
      ans = ids[i - 1];
    cur += i + 1;
  }
  cout << ans << endl;
  return 0;
}
