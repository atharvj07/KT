for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    if n%3==0:
        c0 = a.count(0)
        c1 = a.count(1)
        c2 = a.count(2)
        mn = min(c0,c1,c2)
        c0-=mn
        c1-=mn
        c2-=mn
        ans = (2*mn)
        x = min(c1,c2)
        c1-=x
        c2-=x
        ans+=(c1+c2)
        print(ans)
