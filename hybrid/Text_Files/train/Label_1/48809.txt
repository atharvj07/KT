import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    static class Node {
        int num;
        List<Node> children;
        boolean[] used;
        int from;

        public Node(int num) {
            this.num = num;
            this.children = new ArrayList<>();
            this.used = new boolean[num];
        }

        public void addChild(Node child) {
            this.children.add(child);
        }

        public void removeChild(int childNum) {
            for (int i = 0; i < children.size(); i++) {
                if (children.get(i).num == childNum) {
                    children.remove(i);
                    return;
                }
            }
        }

        public int findMinFree() {
            for (int i = 0; i < used.length; i++) {
                if (!used[i]) {
                    used[i] = true;
                    return i;
                }
            }
            return used.length;
        }

        public void printPath(int currentLabel) {
            System.out.print(currentLabel + " ");
            if (from != 0) {
                children.get(from - 1).printPath(used[currentLabel] ? currentLabel + 1 : currentLabel);
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        Node[] nodes = new Node[n];
        for (int i = 0; i < n; i++) {
            nodes[i] = new Node(i + 1);
        }
        for (int i = 0; i < n - 1; i++) {
            String[] line = br.readLine().split(" ");
            int u = Integer.parseInt(line[0]) - 1;
            int v = Integer.parseInt(line[1]) - 1;
            nodes[u].addChild(nodes[v]);
            nodes[v].addChild(nodes[u]);
        }
        int[] labels = new int[n - 1];
        Map<Integer, Integer> labelToIndex = new HashMap<>();
        labelDFS(nodes[0], 0, labels, labelToIndex);
        for (int label : labels) {
            System.out.println(label);
        }
    }

    public static void labelDFS(Node node, int from, int[] labels, Map<Integer, Integer> labelToIndex) {
        int label = node.findMinFree();
        labels[from] = label;
        labelToIndex.put(label, from);
        for (int i = 0; i < node.children.size(); i++) {
            if (i != from) {
                node.children.get(i).from = from;
                labelDFS(node.children.get(i), i, labels, labelToIndex);
            }
        }
    }
}
