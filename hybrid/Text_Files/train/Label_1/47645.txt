import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = in.readLine().split(" ");
        int n = Integer.parseInt(nk[0]), k = Integer.parseInt(nk[1]);
        int[][] robots = new int[n][3];
        for (int i = 0; i < n; i++) {
            String[] robotInfo = in.readLine().split(" ");
            int x = Integer.parseInt(robotInfo[0]), r = Integer.parseInt(robotInfo[1]), q = Integer.parseInt(robotInfo[2]);
            robots[i][0] = x;
            robots[i][1] = r;
            robots[i][2] = q;
        }

        Arrays.sort(robots, (a, b) -> a[0] - b[0]);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int x1 = robots[i][0], x2 = robots[i][0] + robots[i][1];
            for (int j = i + 1; j < n; j++) {
                if (x2 < robots[j][0] - robots[j][1]) {
                    break;
                }
                if (x1 > robots[j][0] + robots[j][1]) {
                    continue;
                }
                if (Math.abs(robots[i][2] - robots[j][2]) <= k) {
                    ans++;
                }
            }
        }

        System.out.println(ans);
    }
}
