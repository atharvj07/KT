from collections import defaultdict

n, m = map(int, input().split())
a = list(map(int, input().split()))
dct = defaultdict(int)
for i in range(n - 1, -1, -1):
    dct[a[i]] += 1
for i in range(m):
    x = int(input())
    print(len(dct))
    dct[a[x - 1]] -= 1
    if dct[a[x - 1]] == 0:
        del dct[a[x - 1]]
