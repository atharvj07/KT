#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;
typedef long long ll;
const ll B = 1e9;

int main() {
  int n, m;
  cin >> n >> m;
  vector<ll> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<pair<int, int>> seg(m);
  for (auto& [l, r] : seg) { cin >> l >> r; }
  sort(seg.begin(), seg.end(), [&a](const auto& p1, const auto& p2) {
    return a[p1.second] - a[p1.first] > a[p2.second] - a[p2.first];
  });
  ll ans = 0;
  vector<int> ansseg;
  for (const auto& [l, r] : seg) {
    ll x = max({a[l - 1], a[r], 0LL});
    ll y = min({a[l - 1], a[r], 0LL});
    if (x - y > ans) {
      ans = x - y;
      ansseg = {l, r};
    }
  }
  if (m >= 2) {
    for (const auto& [l1, r1] : seg) {
      for (const auto& [l2, r2] : seg) {
        if (l1 == l2 && r1 == r2) continue;
        ll x = max({a[l1 - 1], a[r1], a[l2 - 1], a[r2], 0LL});
        ll y = min({a[l1 - 1], a[r1], a[l2 - 1], a[r2], 0LL});
        if (x - y > ans) {
          ans = x - y;
          ansseg = {l1, r1, l2, r2};
        }
      }
    }
  }
  if (m >= 3) {
    for (const auto& [l1, r1] : seg) {
      for (const auto& [l2, r2] : seg) {
        for (const auto& [l3, r3] : seg) {
          if (l1 == l3 || l1 == r3) continue;
          if (r1 == l3 || r1 == r3) continue;
          if (l2 == l3 || l2 == r3) continue;
          if (r2 == l3 || r2 == r3) continue;
          ll x = max({a[l1 - 1], a[r1], a[l2 - 1], a[r2], a[l3 - 1], a[r3],
                      0LL});
          ll y = min({a[l1 - 1], a[r1], a[l2 - 1], a[r2], a[l3 - 1], a[r3],
                      0LL});
          if (x - y > ans) {
            ans = x - y;
            ansseg = {l1, r1, l2, r2, l3, r3};
          }
        }
      }
    }
  }

  cout << ans << endl;
  cout << ansseg.size() << endl;
  for (int segidx : ansseg) cout << segidx << " ";
  cout << endl;

  return 0;
}
