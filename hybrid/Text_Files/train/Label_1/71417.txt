n, m = map(int, input().split())
a = list(map(int, input().split()))
g = [[], []]
for i in range(m):
    x, y = map(int, input().split())
    g[y % 2].append((x - 1, y - 1))
ma = 0
for i in [0, 1]:
    for x, y in g[i]:
        while a[x] % 2 == 0 and a[y] % 2 == 0:
            ma += 1
            a[x] //= 2
            a[y] //= 2
print(ma)
