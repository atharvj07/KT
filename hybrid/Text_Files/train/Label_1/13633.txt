#include <bits/stdc++.h>
using namespace std;
#define int long long
main() {
    int n, m, x, y;
    cin >> n >> m;
    vector<int> p(n), a(n + 1), dp(n + 1, -1e18);
    for (int i = 0; i < n; ++i) cin >> p[i], a[p[i]] = i;
    dp[n] = 0;
    while (m--) { cin >> x >> y; swap(a[p[x - 1]], a[p[y - 1]]); }
    for (int i = n - 1; i >= 0; --i)
        for (int j = i + 1; j <= n; ++j)
            dp[i] = max(dp[i], 1 + dp[j] + abs(a[j] - i));
    int cur = 0, last = -1;
    for (int i = 0; i < n; ++i) {
        for (int j = cur + 1; j <= n; ++j) {
            if (i + dp[j] > last) cur = j, last = i + dp[j];
        }
        cout << p[cur - 1] << ' ';
    }
    cout << endl;
}
