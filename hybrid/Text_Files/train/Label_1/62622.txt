from sys import stdin, stdout
f = lambda a, b: (int(a) - 1) // b
n, m = map(int, stdin.readline().split())
v = [f(x, 10**9) for x in map(int, stdin.readline().split())]
h = [(f(x1, 10**9), f(x2, 10**9), y) for x1, x2, y in map(int, stdin.readline().split()) for x in range(x1, x2 + 1)]
v.sort()
h.sort()
l1 = len(v)
l2 = len(h)
a1 = [10**9] * l2
a2 = [0] * l2
a3 = [0] * l1
i1 = i2 = i3 = 0
j = 10**9
for x, y, z in h:
    if x == 1:
        if y < j:
            j = y
        if z > 0:
            a1[i2] = z
        if z < 10**9:
            a2[i2] = z
        i2 += 1
while i3 < l1:
    if v[i3] > j:
        break
    a3[i3] = v[i3]
    i3 += 1
for x in a3:
    if x > 0 and x <= j:
        print(1)
        exit()
k = 10**9
for i in range(l2 - 1, -1, -1):
    if a2[i] < k:
        break
    if a1[i] >= k or i1 < l1 and a3[i1] < a2[i]:
        i1 += 1
        continue
    print(1)
    exit()
    k = a2[i]
    i1 += 1
print(0)
