#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

#define endl '\n'
#define dbg(x) cout<<#x<<"="<<x<<endl;
#define pb push_back
#define fi first
#define se second

vector <pair<ll, ll> >fac;
const ll md=1e9+7;

ll fastpow(ll a, ll b)
{
    ll res=1;
    while(b)
    {
        if(b&1) res=(res*a)%md;
        a=(a*a)%md;
        b>>=1;
    }
    return res;
}

void factorize(ll x)
{
    for(ll i=2; i*i<=x; i++)
    {
        if(!(x%i))
        {
            ll cnt=0;
            while(!(x%i)) x/=i, cnt++;
            fac.pb({i, cnt});
        }
    }
    if(x>1) fac.pb({x, 1});
}

ll func(ll id, ll acc, vector <ll> &ans)
{
    if(id==fac.size())
    {
        ans.pb(acc);
        return 1;
    }
    ll sum=0;
    for(ll i=0; i<=fac[id].se; i++)
    {
        sum+=func(id+1, acc*fastpow(fac[id].fi, i), ans);
        sum%=md;
    }
    return sum;
}

int main()
{
    ll n;
    cin>>n;
    ll x, ans=0;
    for(ll i=1; i<=n; i++) cin>>x, factorize(x);
    vector <ll> v;
    cout<<func(0, 1, v)<<endl;
    return 0;
}
