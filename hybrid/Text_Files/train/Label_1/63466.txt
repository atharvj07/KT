n = int(input())
a = list(map(int, input().split()))

folders = []
losses = 0
folder_size = 0

for profit in a:
    folder_size += 1
    if profit < 0:
        losses += 1
        if losses == 3:
            folders.append(folder_size - 1)
            folder_size = 1
            losses = 1
folders.append(folder_size)

print(len(folders))
print(*folders)
