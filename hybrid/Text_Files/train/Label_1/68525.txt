import sys

w = list(map(int, sys.stdin.readline().strip()))
m = int(sys.stdin.readline().strip())
if w[0] + w[1] + w[2] == 0 or m & 1 and w[0] == w[1] == 0 or m > 9 and w.count(1) < (m + 1) // 2:
    print('NO')
else:
    ans = []
    w.sort(reverse = True)
    t = 0
    for i in range(len(w)):
        while w[i] > 0 and t < m // 2:
            t += 1
            ans.append(i + 1)
            w[i] -= 1
    for i in range(len(w)):
        while w[i] > 0 and t < m:
            t += 1
            ans.append(i + 1)
            w[i] -= 1
    print('YES')
    print(*ans)
