#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<pair<int, int>> v;
    for (int i = 0; i < n; i++) {
        int f, t;
        cin >> f >> t;
        v.push_back({f, t});
    }
    
    long long ans = -2e9;
    for (auto& x : v) {
        if (x.second <= k) {
            ans = max(ans, (long long)x.first);
        } else {
            ans = max(ans, (long long)x.first - x.second + k);
        }
    }
    cout << ans << endl;
    return 0;
}
