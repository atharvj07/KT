mod = 998244353

n=int(input())
a=input().split()
a=list(map(lambda x:x.zfill(len(max(a))),a))

ans=[0]*len(a[0])

for i in range(0, len(a)):
    for j in range(0, len(a)):
        num1=list(a[i])
        num2=list(a[j])
        
        for k in range(0, len(a[i])):
            ans[k]+=int(num1[k]+num2[k])
            ans[k]%=mod
print(sum(ans)%mod)
