import java.util.*;
import java.io.*;

public class Main {
    private static void solve() {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long D = sc.nextLong();
        long[] d = new long[N + 1];
        for (int i = 1; i <= N; i++) {
            d[i] = sc.nextLong();
        }
        int Q = sc.nextInt();
        long[] q = new long[Q + 1];
        for (int i = 1; i <= Q; i++) {
            q[i] = sc.nextLong();
        }

        long sum = 0L;
        boolean[] reachable = new boolean[Q + 1];
        for (int i = 1; i <= N; i++) {
            if (sum + d[i] >= D) {
                for (int j = 1; j <= Q; j++) {
                    if (q[j] == i) {
                        if (sum + d[i] == D) {
                            reachable[j] = true;
                        }
                        break;
                    }
                }
                break;
            }
            sum += d[i];
        }

        for (int j = 1; j <= Q; j++) {
            if (reachable[j]) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }

    public static void main(String[] args) {
        solve();
    }
}
