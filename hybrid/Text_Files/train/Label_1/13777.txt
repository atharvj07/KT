#include <iostream>
#include <map>
#include <vector>
#include <string>
#include <algorithm>

int main() {
  std::ios_base::sync_with_stdio(false);
  int n, m; // length of strings s and t
  std::cin >> n >> m;
  std::string s, t; // strings s and t
  std::cin >> s >> t;
  std::map<char, int> t_map; // character count in t
  std::vector<int> diff;       // positions where s != t
  for (int i = 0; i < m; i++) {
    ++t_map[t[i]];
  }
  for (int i = 0; i < n; i++) {
    --t_map[s[i]];
    if (t_map[s[i]] < 0) diff.push_back(i + 1);
  }
  std::cout << diff.size() << '\n';
  for (int i : diff) std::cout << i << ' ';
  std::cout << '\n';
  return 0;
}
