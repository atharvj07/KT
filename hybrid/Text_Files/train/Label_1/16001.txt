#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }

  sort(a.begin(), a.end());

  int m = 0;
  int sum = 0;
  for (int i = 0; i < n; ++i) {
    if (sum + a[i] <= a[n - 1]) {
      m++;
      sum += a[i];
    }
  }

  if (sum == a[n - 1]) {
    cout << m << endl;
  } else {
    cout << -1 << endl;
  }

  return 0;
}
