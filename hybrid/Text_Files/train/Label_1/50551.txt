import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.next();
        int n = s.length();
        // dp[i] = minimum length of compressed version of s[0, i]
        int[] dp = new int[n + 1];
        // dp[0] = 0
        dp[0] = 0;
        // dp[1] = 1 (since the minimum length of compressed version of a single character is 1)
        dp[1] = 1;
        // for each i from 2 to n
        for (int i = 2; i <= n; i++) {
            // dp[i] = minimum length of compressed version of s[0, i]
            dp[i] = dp[i - 1] + 1; // initialize dp[i] with the minimum length of compressed version of s[0, i - 1] plus 1
            // for each j from 1 to i - 1
            for (int j = 1; j < i; j++) {
                // if s[0, j] is equal to s[i - j, i]
                if (s.substring(0, j).equals(s.substring(i - j, i))) {
                    // dp[i] = minimum(dp[i], dp[i - j] + (i - j + 1) / j)
                    dp[i] = Math.min(dp[i], dp[i - j] + (i - j + 1) / j);
                }
            }
        }
        // print the minimum length of compressed version of s
        System.out.println(dp[n]);
    }
}
