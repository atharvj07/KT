#include<iostream>
#include<vector>
#include<cstdio>
using namespace std;
vector<vector<bool>>a;
vector<vector<int>>ans;
int mx=0, n, k;
void f(int i, int j, int cnt, bool d){
	if(cnt>mx){
		mx=cnt;
		ans.clear();
	}
	if(cnt==mx)ans.push_back({i, j});
	if(d){
		if(j<k-1){
			if(a[i][j+1])f(i, j+1, cnt, true);
		}
		else{
			if(i+1<n && a[i+1][j])f(i+1, j, cnt+1, false);
		}
	}
	else{
		if(i<k-1){
			if(a[i+1][j])f(i+1, j, cnt, false);
		}
		else{
			 if(j+1<n && a[i][j+1])f(i, j+1, cnt+1, true);
		}
	}
}
int main(){
	int tmp=0;
	cin>>n>>k;
	a.assign(n, vector<bool>(n));
	ans.assign(n, vector<int>(n, 0));
	for(int i=0; i<n; i++){
		for(int j=0; j<n; j++){
			char x=getchar();
			if(x=='\n')x=getchar();
			a[i][j]=(x=='.');
		}
	}
	for(int i=0; i<n; i++){
		for(int j=0; j<n; j++){
			if(a[i][j]){
				f(i, j, 1, false);
				f(i, j, 1, true);
			}
		}
	}
	cout<<ans[0][0]+1<<' '<<ans[0][1]+1<<endl;
}
