#include <cmath>
#include <cstdio>
#include <vector>
using namespace std;
int main() {
  int x[3], y[3], r[3];
  for (int i = 0; i < 3; i++) { scanf("%d%d%d", &x[i], &y[i], &r[i]); }
  int ans = 4;
  for (int i = 0; i < 3; i++) {
    for (int j = i + 1; j < 3; j++) {
      double d = hypot(x[i] - x[j], y[i] - y[j]);
      if (d > r[i] + r[j]) ans -= 2;
      if (d == r[i] + r[j]) ans -= 1;
      if (d + r[i] < r[j]) ans += 2;
      if (d + r[j] < r[i]) ans += 2;
    }
  }
  printf("%d\n", ans);
  return 0;
}
