#include <bits/stdc++.h>
using namespace std;

void solve() {
    int d;
    cin >> d;
    if (d == 1)
        cout << 6 << endl;
    else if (d == 2)
        cout << 15 << endl;
    else
        cout << (d + 1) * (d + 2) << endl;
}

int main() {
    int t;
    cin >> t;
    while (t--) { solve(); }
}
