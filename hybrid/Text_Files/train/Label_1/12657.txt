#include <cstdio>
#include <utility>
#include <vector>
using namespace std;

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n, m, a, b;
    scanf("%d %d %d %d", &n, &m, &a, &b);
    if (a * n != b * m || a > m || b > n) {
      printf("NO\n");
      continue;
    }
    printf("YES\n");
    vector<vector<int>> grid(n, vector<int>(m, 0));
    int ca = 0, cb = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < a; j++) {
        if (cb < b) {
          grid[i][ca++] = 1;
          cb++;
        }
      }
      ca = 0;
      cb = 0;
    }
    for (const auto &r : grid) {
      for (const int &x : r) { printf("%d", x); }
      printf("\n");
    }
  }
  return 0;
}
