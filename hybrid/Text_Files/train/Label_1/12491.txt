#include <bits/stdc++.h>
using namespace std;
long long a[100001], b[100001];
int main() {
  int n, r, avg;
  cin >> n >> r >> avg;
  for (int i = 0; i < n; i++) { cin >> a[i] >> b[i]; }
  priority_queue<pair<int, int>> pq;
  long long sum = 0, cnt = 0;
  for (int i = 0; i < n; i++) {
    pq.push({(r - a[i]) / b[i], i});
    sum += a[i];
  }
  while (!pq.empty()) {
    long long tmp = pq.top().first;
    int ind = pq.top().second;
    pq.pop();
    if (sum + tmp * b[ind] >= avg * n) {
      cnt += tmp;
      break;
    } else {
      cnt += tmp;
      sum += tmp * b[ind];
      a[ind] += tmp * b[ind];
      pq.push({(r - a[ind]) / b[ind], ind});
    }
  }
  cout << cnt << endl;
  return 0;
}
