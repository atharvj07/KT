#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        if (n % 4 == 0) {
            cout << "YES\n";
            for (int i = 1; i <= n; i++) cout << 2 * (i % 2) + i % 2 << ' ';
        } else
            cout << "NO\n";
    }
    return 0;
}
