import sys

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    if len(set(a)) == n:
        print(0)
    else:
        if len(set(a)) > 3:
            print(-1)
        else:
            if a[0] == a[1] and a[2] == a[3]:
                print(-1)
            else:
                if a[0] == a[1]:
                    a[1], a[2] = a[2], a[1]
                elif a[1] == a[2]:
                    a[1], a[3] = a[3], a[1]
                print(1)
