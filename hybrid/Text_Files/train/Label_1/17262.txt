#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, a, b;
        cin >> n >> a >> b;
        string s;
        cin >> s;
        int score = 0;
        for (int i = 0; i < n; i++) {
            if (i > 0 && s[i] != s[i - 1]) {
                score += (s[i] == '0' ? a : b);
            }
        }
        cout << score << "\n";
    }
    return 0;
}
