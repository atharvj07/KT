import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        double[][] d = new double[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                d[i][j] = Double.MAX_VALUE;
            }
        }
        for (int i = 0; i < n; i++) {
            d[i][i] = 0;
        }
        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (d[i][j] > d[i][k] + d[k][j] + (m + d[i][k]) / a[k] + (m + d[k][j]) / b[k]) {
                        d[i][j] = d[i][k] + d[k][j] + (m + d[i][k]) / a[k] + (m + d[k][j]) / b[k];
                    }
                }
            }
        }
        if (d[0][n - 1] == Double.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(d[0][n - 1]);
        }
    }
}
