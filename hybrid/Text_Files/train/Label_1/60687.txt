n, m = map(int, input().split())
table = [input() for _ in range(n)]

for i in range(n):
    for j in range(m):
        if table[i][j] == 'B':
            top_left = (i, j)
            break
    else:
        continue
    break

for i in range(n-1, -1, -1):
    for j in range(m-1, -1, -1):
        if table[i][j] == 'B':
            bottom_right = (i, j)
            break
    else:
        continue
    break

center_r = (top_left[0] + bottom_right[0]) // 2 + 1
center_c = (top_left[1] + bottom_right[1]) // 2 + 1

print(center_r, center_c)
