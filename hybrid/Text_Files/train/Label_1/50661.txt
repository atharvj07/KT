import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        ArrayList<Integer>[] cities = new ArrayList[N + 1];
        for (int i = 0; i <= N; i++) {
            cities[i] = new ArrayList<>();
        }
        for (int i = 0; i < M; i++) {
            int p = sc.nextInt();
            int y = sc.nextInt();
            cities[p].add(y);
        }
        for (int p = 1; p <= N; p++) {
            Collections.sort(cities[p]);
            for (int i = 0; i < cities[p].size(); i++) {
                String id = String.format("%06d%06d", p, i + 1);
                System.out.println(id);
            }
        }
    }
}
