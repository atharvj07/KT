#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, x, sum = 0;
    cin >> n >> x;
    vector<long long> v(n);
    for (long long i = 0; i < n; i++) {
      cin >> v[i];
      sum += v[i];
    }
    for (long long i = 0; i < n; i++) {
      if (v[i] % x == 0) {
        sum += v[i] / x * x;
      } else {
        cout << sum << endl;
        goto out;
      }
    }
    cout << sum << endl;
  out:;
  }
  return 0;
}
