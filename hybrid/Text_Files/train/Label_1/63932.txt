def find_fourth_point():
    points = [tuple(map(int, input().split())) for _ in range(3)]
    possible_points = set()

    for i in range(3):
        for j in range(i + 1, 3):
            dx = points[i][0] - points[j][0]
            dy = points[i][1] - points[j][1]
            for k in range(3):
                if k != i and k != j:
                    possible_points.add((points[k][0] + dx, points[k][1] + dy))
                    possible_points.add((points[k][0] - dx, points[k][1] - dy))

    print(len(possible_points))
    for point in possible_points:
        print(*point)

find_fourth_point()
