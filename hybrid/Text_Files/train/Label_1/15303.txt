#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> a(n + 2);
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    a[x] = {i, 0};
  }
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    a[x].second = i;
  }
  int ans = 0;
  for (int i = 2; i <= n; i++) {
    if (a[i].first > a[i - 1].second) ans++;
  }
  cout << ans << endl;
  return 0;
}
