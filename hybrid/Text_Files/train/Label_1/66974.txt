def count_paths(n, m, k, grid):
    dp = [[{} for _ in range(m)] for _ in range(n)]
    dp[0][0][grid[0][0]] = 1

    for i in range(n):
        for j in range(m):
            if i == 0 and j == 0:
                continue
            current_dp = dp[i][j]
            if i > 0:
                for xor_val, count in dp[i-1][j].items():
                    new_xor = xor_val ^ grid[i][j]
                    if new_xor in current_dp:
                        current_dp[new_xor] += count
                    else:
                        current_dp[new_xor] = count
            if j > 0:
                for xor_val, count in dp[i][j-1].items():
                    new_xor = xor_val ^ grid[i][j]
                    if new_xor in current_dp:
                        current_dp[new_xor] += count
                    else:
                        current_dp[new_xor] = count

    return dp[n-1][m-1].get(k, 0)

def solve():
    n, m, k = map(int, input().split())
    grid = [list(map(int, input().split())) for _ in range(n)]
    print(count_paths(n, m, k, grid))

solve()
