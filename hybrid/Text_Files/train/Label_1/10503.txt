#include <cstdio>
#include <algorithm>
#include <cmath>

using namespace std;

int n, a[5], b[5];
double mean, med, rng;

bool solve() {
    sort(a, a + n);
    mean = (a[0] + a[n - 1]) / 2.0;
    med = a[n / 2];
    rng = a[n - 1] - a[0];
    bool ok = true;
    if (mean < med || mean > med + 1e-9)
        ok = false;
    if (fabs(mean - rng) > 1e-9)
        ok = false;
    return ok;
}

int main() {
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);
    if (solve()) {
        puts("YES");
        for (int i = 0; i < 4 - n; i++)
            printf("%d\n", (int)round(mean));
    } else
        puts("NO");
    return 0;
}
