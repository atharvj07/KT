#include <bits/stdc++.h>
using namespace std;
void solve_problem() {
    int n, m, d;
    cin >> n >> m;
    int p[n + 1], c[n + 1];
    map<int, set<int>> clubs;
    for (int i = 1; i <= n; i++) {
        cin >> p[i];
    }
    for (int i = 1; i <= n; i++) {
        cin >> c[i];
        clubs[c[i]].insert(i);
    }
    cin >> d;
    for (int i = 1; i <= d; i++) {
        int k;
        cin >> k;
        clubs[c[k]].erase(k);
        int ans = 0;
        for (int j = 1; j <= m; j++) {
            if (clubs[j].size()) {
                int x = *clubs[j].begin();
                ans = max(ans, p[x]);
            }
        }
        cout << ans << "\n";
    }
}
int main() { solve_problem(); }
