#include <vector>
#include <algorithm>
#include <iostream>

using namespace std;

long long solve() {
    long long n; cin >> n;
    vector<long long> a(n);
    for (long long i = 0; i < n; ++i) cin >> a[i];
    sort(a.begin(), a.end());
    long long res = 0;
    for (long long i = 0; i + 1 < n; ++i) {
        if (a[i] == -1) continue;
        res += a[i];
        a[++i] = -1;
    }
    return res;
}

int main() {
    cout << solve() << endl;
    return 0;
}
