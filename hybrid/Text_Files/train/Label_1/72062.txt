n = int(input())
a = list(map(int, input().split()))
a.sort(reverse=True)

length = 0
current_max = a[0]

for i in range(n):
    if a[i] < current_max:
        if a[i] == 0:
            break
        length += a[i]
        current_max = a[i]
    else:
        if current_max > 0:
            length += current_max
            current_max -= 1

print(length)
