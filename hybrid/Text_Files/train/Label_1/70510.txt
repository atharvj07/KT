def solve():
    t = int(input())
    for tc in range(t):
        input()
        k, n, m = map(int, input().split())
        a = list(map(int, input().split()))
        b = list(map(int, input().split()))

        ans = [0] * (n + m)
        i = j = 0
        for p in range(n + m):
            if a[i] <= b[j]:
                ans[p] = a[i]
                i += 1
                if i == n: break
            elif b[j] != 0:
                ans[p] = b[j]
                j += 1
                if j == m: break
        if i != n or j != m: print(-1)
        else: print(*ans)

solve()
