#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x, y, ans = 0;
  cin >> n;
  cin >> x >> y;
  pair<int, int> pos = {x, y};
  while (cin >> x >> y) {
    if (pos.first == x && pos.second == y) break;
    if (((pos.first == x && pos.second < y) ||
         (pos.second == y && pos.first < x)) &&
        (x - y) % 2 == 1)
      ans++;
    pos = {x, y};
  }
  cout << ans << '\n';
}
