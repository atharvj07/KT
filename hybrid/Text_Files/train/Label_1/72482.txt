import math

n = int(input())
circles = []
for i in range(n):
    x, y, r = map(int, input().split())
    circles.append((x, y, r))

def is_intersect(c1, c2):
    x1, y1, r1 = c1
    x2, y2, r2 = c2
    d = math.sqrt((x1 - x2)**2 + (y1 - y2)**2)
    return d <= r1 + r2

cnt = 0
for i in range(n):
    for j in range(i + 1, n):
        if is_intersect(circles[i], circles[j]):
            cnt += 1

print(cnt + 1 + n)
