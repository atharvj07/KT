n, h, m = map(int,input().split())
a = [(0,0)]*n
for i in range(m):
	l, r, x = map(int,input().split())
	l-=1
	r-=1
	a[l] = (a[l][0],max(a[l][1],x))
	a[r] = (max(a[r][0],x),a[r][1])
p = [0]*n
for i in range(n):
	a[i] = (max(a[i][0],a[i-1][1]),a[i][1])
	p[i] = a[i][0]*a[i][0]
	if i: p[i]+=p[i-1]
print(p[n-1])
