#include<bits/stdc++.h>
#define tr(v,it) for(auto it=v.begin();it!=v.end();it++)
#define ll long long int
#define re register
#define vi vector<int>
#define vvi vector<vi>
#define pii pair<int,int>
#define vpii vector<pii>
#define mp make_pair
#define pb push_back
#define F first
#define S second
#define endl "\n"
const int mod=1e9+7;
const int N=1e6+5;
using namespace std;
int dx[]={0,0,1,-1}; // UDLR
int dy[]={1,-1,0,0}; // UDLR

int n,m,sol;
char c[1010][1010];
bool vis[1010][1010];
queue<pii> q;

void bfs()
{
    while(!q.empty())
    {
        int x=q.front().F;
        int y=q.front().S;
        q.pop();
        sol++;
        for(re int i=0;i<4;i++)
        {
            int xx=x+dx[i];
            int yy=y+dy[i];
            if(xx>=1 && xx<=n && yy>=1 && yy<=m)
            {
                if(!vis[xx][yy] && (c[xx][yy]!='.' && c[xx][yy]!='#'))
                {
                    vis[xx][yy]=1;
                    q.push(mp(xx,yy));
                }
            }
        }
    }
}

int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    int i,j;
    cin>>n>>m;
    for(i=1;i<=n;i++)
        for(j=1;j<=m;j++)
        {
            cin>>c[i][j];
            if(c[i][j]!='.' && c[i][j]!='#')
            {
                sol=1;
                vis[i][j]=1;
                q.push(mp(i,j));
            }
        }
    if(sol==1) bfs();
    sol==1 ? cout<<sol-1<<endl : cout<<-1<<endl;
    return 0;
}
