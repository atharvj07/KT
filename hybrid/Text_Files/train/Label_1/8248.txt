#include<bits/stdc++.h>
#define ll long long
using namespace std;
const ll mod=998244353;
ll pw(ll a,ll b){
    ll r=1;
    while(b){
        if(b%2)r=r*a%mod;
        a=a*a%mod;
        b/=2;
    }
    return r%mod;
}
const ll N = 1e5+10;
int main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    ll n,q,a[N],b[N],c[N],d[N];
    cin>>n;
    for(int i=1;i<=n;++i){
        cin>>a[i];
        b[a[i]]++;
    }
    for(int j=1;j<=1e5;++j){
        if(!b[j])continue;
        if(b[j]&1)c[1]+=b[j]/2;
        else c[1]+=(b[j]/2),c[2]+=(b[j]%2);
    }
    cin>>q;
    while(q--){
        char x;
        ll y;
        cin>>x>>y;
        b[y]--;
        if(c[1]){
            if(b[y]==0){
                c[1]--;
                c[2]+=2;
            }
            else{
                if(b[y]%2==0)c[1]--,c[2]+=2;
                else c[2]++;
            }
        }
        else{
            if(b[y]%2==1)c[2]--;
            else c[2]-=2;
        }
        if(c[1])cout<<"YES"<<'\n';
        else{
            if((c[2]%4==0 and (c[2]-4)>=0) or (c[2]-2)>=0)cout<<"YES"<<'\n';
            else cout<<"NO"<<'\n';
        }
