#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  bool vis[10] = {false};
  int ans = 1;
  for (int i = 0; i < n; i++) {
    int d = s[i] - '0';
    if (i == 0 && d == 0) {
      ans = 0;
      break;
    }
    if (!vis[d]) ans *= d;
    vis[d] = true;
  }
  cout << ans << endl;
  return 0;
}
