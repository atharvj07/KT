import java.util.Scanner;
import java.util.Arrays;

public class Main {
    static int[][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[][] a = new int[n][m];
            dp = new int[n + 1][m + 1];
            for (int i = 0; i <= n; i++) {
                Arrays.fill(dp[i], -1);
            }
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    a[i][j] = sc.nextInt();
                }
            }
            if (solve(a, n, m) == 1) {
                System.out.println("Ashish");
            } else {
                System.out.println("Jeel");
            }
        }
    }

    public static int solve(int[][] a, int n, int m) {
        if (dp[n][m] != -1) {
            return dp[n][m];
        }
        if (n == 1 && m == 1) {
            return dp[n][m] = a[n - 1][m - 1] > 0 ? 1 : 2;
        }
        int ans = 2;
        if (a[n - 1][m - 1] > 0) {
            ans = 1;
            for (int i = n - 1; i >= 1; i--) {
                if (a[i - 1][m - 1] == 0) {
                    break;
                }
                ans = 2 - solve(a, i, m);
                if (ans == 1) {
                    break;
                }
            }
            for (int j = m - 1; j >= 1; j--) {
                if (a[n - 1][j - 1] == 0) {
                    break;
                }
                ans = 2 - solve(a, n, j);
                if (ans == 1) {
                    break;
                }
            }
        }
        return dp[n][m] = ans;
    }
}
