#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;
int n, ans[N];
bool leaf[N];
vector<int> g[N];

void dfs(int u, int p) {
    leaf[u] = true;
    for (auto v : g[u]) {
        if (v == p) continue;
        leaf[u] = false;
        dfs(v, u);
    }
}

void dfs2(int u, int p, int color) {
    ans[u] = 1;
    for (auto v : g[u]) {
        if (v == p) continue;
        int cur = leaf[v] ? color : color + 1;
        dfs2(v, u, cur);
        ans[u] = max(ans[u], ans[v]);
    }
}

int main() {
    cin >> n;
    for (int i = 2; i <= n; i++) {
        int p;
        cin >> p;
        g[i].push_back(p);
        g[p].push_back(i);
    }
    dfs(1, 1);
    dfs2(1, 1, 0);
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    return 0;
}
