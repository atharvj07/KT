def max_importance(n, m, k, castles, portals):
    castles = [(0, 0, 0)] + castles  # Add a dummy castle at the beginning for easier indexing
    dp = [[-1] * (k + 1) for _ in range(n + 1)]  # DP table, dp[i][j] = max importance if i castles are taken with j warriors
    dp[0][k] = 0  # Base case: 0 castles taken with k warriors has 0 importance

    for i in range(1, n + 1):
        req, hire, imp = castles[i]
        for j in range(k + 1):
            if j >= req and dp[i - 1][j - req] != -1:  # If can take current castle
                dp[i][j + hire] = max(dp[i][j + hire], dp[i - 1][j - req])  # Without defending this castle
                
                # If decide to defend current castle
                dp[i][j + hire - 1] = max(dp[i][j + hire - 1], dp[i - 1][j - req] + imp)
    
    # Apply portals, which can only increase the importance
    for _ in range(n):  # In the worst case, we may need to propagate updates n times
        for u, v in portals:
            for j in range(k + 1):
                if dp[u][j] != -1:  # If it's possible to reach u with j warriors left
                    dp[v][j] = max(dp[v][j], dp[u][j] + castles[v][2])  # Update the importance of v

    return max(dp[n])  # The maximum importance achievable

# Input reading
n, m, k = map(int, input().split())
castles = [tuple(map(int, input().split())) for _ in range(n)]
portals = [tuple(map(int, input().split())) for _ in range(m)]

print(max_importance(n, m, k, castles, portals))
