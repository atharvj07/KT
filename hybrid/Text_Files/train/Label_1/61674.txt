t = int(input())
for _ in range(t):
    n, k, d = map(int, input().split())
    s = input()
    ans = k
    for j in range(k):
        l = -1
        cnt = 0
        for i in range(n):
            if s[i] == str(j+1):
                if l==-1:
                    l = i
                cnt+=1
                if cnt>=d:
                    ans = min(ans, j+1)
                    break
                else:
                    l=i-d+1
        cnt = 0
        r=-1
        for i in range(n):
            if s[i] == str(j+1):
                if r==-1:
                    r = i
                cnt+=1
                if cnt>=d:
                    ans = min(ans, j+1)
                    break
                else:
                    r=i+1-d
    print(ans)
