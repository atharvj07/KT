import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] sum = new int[n];
        sum[n - 1] = a[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            sum[i] = sum[i + 1] * a[i];
        }
        int neg = 0, pos = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                if (sum[i] <= 0) {
                    neg++;
                } else {
                    pos++;
                }
            }
        }
        System.out.println(neg + " " + pos);
    }
}
