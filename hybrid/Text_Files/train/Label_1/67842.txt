def can_find_special_boxes(n, candies):
    candies.sort()
    if n == 0:
        # 0 boxes: we can always form 1,1,3,3
        return True, [1, 1, 3, 3]
    elif n == 1:
        # 1 box: we can form 1,1,3,3 or any set with the same mean, median, and range.
        return True, [candies[0], candies[0], 3*candies[0] - candies[0], 3*candies[0] - candies[0]]
    elif n == 2:
        # 2 boxes: mean = median = range implies a symmetric set around the median.
        if candies[1] - candies[0] == 1:
            return True, [candies[0] - 1, candies[1] + 1]
        elif candies[0] == candies[1]:
            return True, [candies[0] + 2, candies[1] + 2]
        else:
            return False, []
    elif n == 3:
        # 3 boxes: need to check if we can form a set with equal mean, median, and range.
        # For 3 boxes, two must be equal and the third should form a symmetric set with a new number.
        if candies[1] - candies[0] == candies[2] - candies[1]:
            return True, [2*candies[1] - candies[0] - candies[2]]
        elif candies[0] == candies[1]:
            return True, [2*candies[2] - candies[0]]
        elif candies[1] == candies[2]:
            return True, [2*candies[0] - candies[2]]
        else:
            return False, []
    else:
        # 4 boxes: check if they already satisfy the condition.
        if (candies[3] - candies[0] == (candies[1] + candies[2]) / 2 == (candies[0] + candies[1] + candies[2] + candies[3]) / 4):
            return True, []
        else:
            return False, []

n = int(input())
candies = [int(input()) for _ in range(n)]
exists, missing_boxes = can_find_special_boxes(n, candies)
if exists:
    print("YES")
    for box in missing_boxes:
        print(box)
else:
    print("NO")
