import java.util.*;
class Main {
  public static void main(String[] args){
    Scanner sc = new Scanner(System.in);
    for(int t = sc.nextInt(); t-- > 0;){
      int n = sc.nextInt();
      int[] arr = new int[n];
      int[] min = new int[n+2];
      int[] max = new int[n+2];
      for(int i = 0; i < n; ++i){
        arr[i] = sc.nextInt();
        min[i+1] = Math.min(arr[i], min[i]);
        max[i+1] = Math.max(arr[i], max[i]);
      }
      boolean exists = false;
      for(int i = 0; i < n; ++i){
        for(int j = i+1; j < n+2; ++j){
          if(max[j] - min[i] >= j - i){
            System.out.println("YES");
            System.out.println((i+1) + " " + j);
            exists = true;
            break;
          }
        }
        if(exists) break;
      }
      if(!exists) System.out.println("NO");
    }
  }
}
