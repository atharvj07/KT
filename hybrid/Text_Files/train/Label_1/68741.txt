import math

n, k = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
s = sum(math.log(y) for y in a)
i, e = 0, math.log(k)
ans = 0
for j in range(n):
    while i < j and s - math.log(a[i]) < e:
        i += 1
    if s - math.log(a[j]) >= e:
        ans += n - j - (i == j)
    s -= math.log(a[j])
print(ans)
