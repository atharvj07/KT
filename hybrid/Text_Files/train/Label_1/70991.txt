def solve_carousel(n, types):
    colors = [1] * n
    for i in range(1, n):
        if types[i] != types[i - 1]:
            colors[i] = 3 - colors[i - 1]
    if types[0] != types[-1] and colors[0] == colors[-1]:
        found = False
        for i in range(1, n):
            if types[i] == types[i - 1]:
                for j in range(i, n):
                    colors[j] = 3 - colors[j]
                found = True
                break
        if not found:
            colors[-1] = 3
            return 3, colors
    return 2 if types[0] != types[-1] else 1, colors

q = int(input().strip())
for _ in range(q):
    n = int(input().strip())
    types = list(map(int, input().strip().split()))
    k, colors = solve_carousel(n, types)
    print(k)
    print(" ".join(map(str, colors)))
