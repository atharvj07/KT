import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        while (t-- > 0) {
            int n = Integer.parseInt(br.readLine());
            int[] a = parseIntArray(br.readLine());
            Map<String, Integer> dp = new HashMap<>();
            System.out.println(solve(a, 0, dp));
        }
    }

    private static int solve(int[] a, int idx, Map<String, Integer> dp) {
        if (idx == a.length) return 1;
        if (dp.containsKey(Arrays.toString(a))) return dp.get(Arrays.toString(a));
        int ans = 0;
        for (int i = idx + 1; i < a.length; i++) {
            if (a[idx] < a[i]) {
                a[i]++;
                ans += solve(a, idx + 1, dp);
                a[i]--;
            } else if (a[idx] > a[i]) {
                a[idx]++;
                ans += solve(a, idx + 1, dp);
                a[idx]--;
            }
        }
        dp.put(Arrays.toString(a), ans);
        return ans;
    }

    private static int[] parseIntArray(String s) {
        String[] arr = s.split(" ");
        int[] ret = new int[arr.length];
        for (int i = 0; i < arr.length; i++) {
            ret[i] = Integer.parseInt(arr[i]);
        }
        return ret;
    }
}
