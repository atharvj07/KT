#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, l, r, ql, qr;
  cin >> n >> l >> r >> ql >> qr;
  vector<int> w(n + 1);
  for (int i = 1; i <= n; i++) { cin >> w[i]; }
  long long dp[n + 1][2];
  dp[n + 1][0] = dp[n + 1][1] = 0;
  for (int i = n; i >= 1; i--) {
    dp[i][0] = min(dp[i + 1][0] + w[i] * l + ql, dp[i + 1][1] + w[i] * r + qr);
    dp[i][1] = min(dp[i + 1][1] + w[i] * r + qr, dp[i + 1][0] + w[i] * l + ql);
  }
  cout << dp[1][0] << endl;
}
