#include <iostream>
#include <vector>

using namespace std;

int n;
vector<pair<int, int>> bishops;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    cin >> n;
    bishops.resize(n);
    for (auto &bishop : bishops) {
        cin >> bishop.first >> bishop.second;
    }

    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (bishops[i].first + bishops[i].second == bishops[j].first + bishops[j].second ||
                bishops[i].first - bishops[i].second == bishops[j].first - bishops[j].second) {
                ans++;
            }
        }
    }

    cout << ans << endl;

    return 0;
}
