n, m, a = map(int, input().split())
b = list(map(int, input().split()))
p = list(map(int, input().split()))

b.sort()
p.sort()

def is_possible(x):
    total = a
    for i in range(x):
        if b[i] < p[i]:
            total -= p[i] - b[i]
    return total >= 0

l, r = 0, min(n, m) + 1
while r - l > 1:
    mid = (l + r) // 2
    if is_possible(mid):
        l = mid
    else:
        r = mid

def min_personal_money(x):
    total = a
    personal = 0
    for i in range(x):
        if b[i] < p[i]:
            total -= p[i] - b[i]
            personal += b[i]
        else:
            personal += p[i]
    return personal

print(l, min_personal_money(l))
