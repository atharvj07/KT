import java.util.Scanner;

public class Main {
    private static int mod = 100000000;
    private static int[][][] dp = new int[101][101][2];
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n1 = s.nextInt();
        int n2 = s.nextInt();
        int k1 = s.nextInt();
        int k2 = s.nextInt();
        int ans = go(n1, n2, 0, k1, k2);
        System.out.println(ans);
    }

    private static int go(int n1, int n2, int state, int k1, int k2) {
        if(n1 == 0 && n2 == 0) return 1;
        if(dp[n1][n2][state] != 0) return dp[n1][n2][state];
        int ans = 0;
        if(state == 0) {
            if(n1 > 0) ans += go(n1 - 1, n2, 1, k1, k2);
            if(n2 > 0) ans += go(n1, n2 - 1, 0, k1, k2);
        } else {
            if(n1 > 0) ans += go(n1 - 1, n2, 1, k1, k2);
            if(n2 > 0 && k2 > 0) ans += go(n1, n2 - 1, 0, k1, k2 - 1);
        }
        dp[n1][n2][state] = ans % mod;
        return dp[n1][n2][state];
    }
}
