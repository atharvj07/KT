t = int(input().strip())
for _ in range(t):
    n, x = map(int, input().strip().split())
    a = list(map(int, input().strip().split()))

    operations = 0
    impossible = False
    for i in range(1, n):
        if a[i-1] > a[i]:
            if x > a[i]:
                impossible = True
                break
            else:
                while i > 0 and a[i-1] > x:
                    a[i-1], x = x, a[i-1]
                    operations += 1
                    i -= 1
    if impossible or any(a[i] > a[i+1] for i in range(n-1)):
        print(-1)
    else:
        print(operations)
