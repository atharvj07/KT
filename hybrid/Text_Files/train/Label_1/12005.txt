#include <bits/stdc++.h>

const int MN = 1e5 + 5;

std::vector<std::pair<int, int>> edges[MN];
std::bitset<MN> vis;

int dfs(int node) {
  vis[node] = true;
  int sum = 0;
  for (const auto& edge : edges[node]) {
    int u, w;
    std::tie(u, w) = edge;
    if (!vis[u]) {
      sum += w + dfs(u);
    }
  }
  return sum;
}

int main() {
  int n, m;
  std::cin >> n >> m;
  while (m--) {
    int u, v;
    std::cin >> u >> v;
    edges[u].emplace_back(v, 1);
    edges[v].emplace_back(u, 1);
  }
  std::cout << dfs(1) << std::endl;
}
