import java.util.*;
class Main {
   public static void main(String[] args) {
      Scanner input = new Scanner(System.in);
      int n = input.nextInt(), x = input.nextInt();
      int[] arr = new int[n];
      for (int i = 0; i < n; i++) arr[i] = input.nextInt();
      Arrays.sort(arr);
      int median = (n + 1)/2 - 1, small = 0;
      for (int i = 0; i < n; i++) {
         if (arr[i] < x) small++;
         else break;
      }
      int ans = Math.max(0, x - arr[median] + (median - small));
      System.out.println(ans);
   }
}
