#include <iostream>
#include <unordered_set>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  unordered_set<int> positions;
  positions.insert(0);
  int x = 0;
  for (char c : s) {
    if (c == 'L')
      --x;
    else
      ++x;
    positions.insert(x);
  }
  cout << positions.size() << endl;
  return 0;
}
