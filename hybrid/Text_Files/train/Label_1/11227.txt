#include <iostream>
#include <cmath>

using namespace std;

const double eps = 1e-6;

double sum, ans, cnt;

int main() {
  int n;
  cin >> n;
  while (n--) {
    int t;
    cin >> t;
    if (t == 1) {
      int a, x;
      cin >> a >> x;
      sum += a * x;
      cnt += a;
    } else if (t == 2) {
      int k;
      cin >> k;
      sum += k;
      cnt += 1;
    } else {
      sum -= ans;
      cnt -= 1;
      if (fabs(sum) < eps)
        ans = 0;
      else
        ans = sum / cnt;
    }
    cout.precision(6);
    cout << fixed << ans << '\n';
  }
  return 0;
}
