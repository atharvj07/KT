#include <bits/stdc++.h>
using namespace std;

int n;
vector<long long> a;

int main() {
  cin >> n;
  a.resize(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  sort(a.rbegin(), a.rend());
  long long ans = a[0];
  for (int i = 1; i < n; ++i) {
    if (a[i] < 0) ans -= a[i];
    else ans += a[i];
    if (ans > 0) {
      for (int j = i + 1; j < n; ++j) {
        cout << "1 " << i + 1 << " " << j + 1 << "\n";
        a[j] = -1e9;
      }
      break;
    }
  }
  a[0] = ans;
  for (int i = 1; i < n; ++i) {
    if (a[i] != -1e9) cout << "2 " << i + 1 << "\n";
  }
}
