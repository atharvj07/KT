#include<bits/stdc++.h>
using namespace std;
#define ll long long int
const ll mod=1000000007;
ll fact(ll n){
    ll i,ans=1;
    for(i=2;i<=n;i++){
        ans=(ans*i)%mod;
    }
    return ans;
}
ll bigmod(ll a,ll b){
    ll ans=1;
    while(b>0){
        if(b%2==1)
        ans=(ans*a)%mod;
        a=(a*a)%mod;
        b/=2;
    }
    return ans;
}
ll ncr(ll n, ll r){
    ll ans;
    ans=(fact(n)*bigmod((fact(r)*fact(n-r))%mod,mod-2))%mod;
    return ans;
}
int main (){
    ll n,k,i,j,ans;
    cin>>n>>k;
    if(k==1){
        cout<<1;
        return 0;
    }
    if(n-k<=0){
        cout<<0;
        return 0;
    }
    ans=(fact(n-k)*ncr(n,n-k))%mod;
    cout<<ans;
}
