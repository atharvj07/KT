#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int q; cin >> q;
    for (int z = 0; z < q; z++) {
        int n, k; cin >> n >> k;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        long long mn = *min_element(a.begin(), a.end()), mx = *max_element(a.begin(), a.end());
        if (mx - mn > 2LL * k) cout << -1 << "\n";
        else {
            long long ans = (mx + mn + 1LL) / 2LL;
            if ((mx - ans) <= k && (ans - mn) <= k) cout << ans << "\n";
            else cout << -1 << "\n";
        }
    }
}
