import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        while (t-- > 0) {
            long n = sc.nextLong();
            long k = sc.nextLong();

            long a = (k - 1) % n + 1;
            long b = (1 + a + k - 1) % n;
            b = b == 0 ? n : b;

            System.out.println(b);
        }
    }
}
