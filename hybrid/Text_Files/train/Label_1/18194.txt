#include <bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;

int par[N], sz[N];

int findPar(int u) {
  if (par[u] == u) return u;
  return par[u] = findPar(par[u]);
}

void unite(int u, int v) {
  int a = findPar(u);
  int b = findPar(v);
  if (a != b) {
    if (sz[a] < sz[b]) swap(a, b);
    par[b] = a;
    sz[a] += sz[b];
  }
}

int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<int> v(n), t(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    for (int i = 0; i < n; i++) cin >> t[i];
    vector<pair<int, int>> edges;
    for (int i = 0; i < m; i++) {
      int u, w;
      cin >> u >> w;
      u--, w--;
      edges.push_back({u, w});
    }
    for (int i = 0; i < n; i++) par[i] = i, sz[i] = 1;
    bool ok = true;
    int sum = 0;
    for (auto x : edges) {
      int u = x.first, w = x.second;
      if (v[u] != v[w]) ok = false;
      if (t[u] != t[w]) ok = false;
      sum += (t[u] - v[u]);
    }
    if (sum % n) ok = false;
    if (!ok) {
      cout << "NO\n";
      continue;
    }
    int target = sum / n;
    for (auto x : edges) {
      int u = x.first, w = x.second;
      unite(u, w);
    }
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (par[i] == i) {
        if (v[i] != target) cnt++;
      }
    }
    if (cnt > 1) ok = false;
    if (ok)
      cout << "YES\n";
    else
      cout << "NO\n";
  }
  return 0;
}
