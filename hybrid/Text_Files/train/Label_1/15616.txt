#include <iostream>
#include <string>

using namespace std;

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        if (s.find("miao.") == 0 && s.rfind("lala.") == s.size() - 6) {
            cout << "OMG>.< I don't know!" << endl;
        } else if (s.rfind("lala.") == s.size() - 6) {
            cout << "Freda's" << endl;
        } else if (s.find("miao.") == 0) {
            cout << "Rainbow's" << endl;
        } else {
            cout << "OMG>.< I don't know!" << endl;
        }
    }
    return 0;
}
