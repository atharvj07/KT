import java.util.Scanner;

public class SmallestIntegerWithNdivisors {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        // Find the smallest integer with n divisors
        int smallestInteger = 1;
        for (int i = 2; i * i <= n; i++) {
            int count = 0;
            while (n % i == 0) {
                n /= i;
                count++;
            }
            if (count > 0) {
                smallestInteger *= Math.pow(i, count + 1);
            }
        }
        if (n > 1) {
            smallestInteger *= n + 1;
        }

        // Print the result
        System.out.println(smallestInteger);
    }
}
