import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), d = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] ans = new int[n];
        int day = 1;
        int last = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] - last >= d) {
                ans[i] = day;
                last = a[i];
            } else {
                day++;
                ans[i] = day;
                last = a[i];
            }
        }
        System.out.println(day);
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
