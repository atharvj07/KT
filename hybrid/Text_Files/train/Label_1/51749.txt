import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int n = s.length();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = (s.charAt(i) == '1' ? 1 : (s.charAt(i) == '0' ? -1 : 0));
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = (i > 0 ? b[i - 1] : 0) + a[i];
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                ans = Math.max(ans, Math.abs(b[j] - (i > 0 ? b[i - 1] : 0)));
            }
        }
        System.out.println(ans);
    }
}
