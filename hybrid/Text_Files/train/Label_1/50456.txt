import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        boolean left = true, right = true;
        int cur = 0;
        while (cur < n && left) {
            int max = arr[cur];
            int next = -1;
            for (int i = cur + 1; i <= cur + max; i++) {
                if (i < n && arr[i] >= max) {
                    next = i;
                }
            }
            if (next == -1) {
                left = false;
            } else {
                cur = next;
            }
        }

        cur = n - 1;
        while (cur >= 0 && right) {
            int max = arr[cur];
            int next = -1;
            for (int i = cur - 1; i >= cur - max; i--) {
                if (i >= 0 && arr[i] >= max) {
                    next = i;
                }
            }
            if (next == -1) {
                right = false;
            } else {
                cur = next;
            }
        }

        if (left && right) {
            System.out.println("yes");
        } else {
            System.out.println("no");
        }
    }
}
