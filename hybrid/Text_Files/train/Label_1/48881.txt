import java.util.*;
import java.io.*;

public class Main {
    static ArrayList<ArrayList<Integer>>Adj = new ArrayList<>();
    static int n, m, s, t, dist[];
    static boolean vis[];
    static Queue<Integer>q;
    static void addEdge(int u, int v)
    {
        Adj.get(u).add(v);
        Adj.get(v).add(u);
    }
    static void BFS(int s, int t)
    {
        dist[s] = 0;
        q.add(s);
        while(!q.isEmpty())
        {
            int v = q.poll();
            for(int u:Adj.get(v))
            {
                if(dist[u] == -1)
                {
                    dist[u] = dist[v] + 1;
                    q.add(u);
                }
            }
        }
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        s = sc.nextInt() - 1;
        t = sc.nextInt() - 1;
        q = new LinkedList<>();
        dist = new int[n];
        vis = new boolean[n];
        Arrays.fill(dist, -1);
        for(int i = 0; i < n; i++)
            Adj.add(new ArrayList<>());
        for(int i = 0; i < m; i++)
        {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            addEdge(u, v);
        }
        BFS(s, t);
        int ans = 0;
        for(int i = 0; i < n; i++)
            for(int j = i + 1; j < n; j++)
                if(dist[i] != -1 && dist[j] != -1 && dist[i] == dist[j] && !Adj.get(i).contains(j))
                    ans++;
        System.out.println(ans);
    }
}
