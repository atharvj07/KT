#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (auto &ai : a) cin >> ai;
    sort(a.begin(), a.end());
    vector<int> dp(n, -1);
    long long sum = 0, ans;
    int cnt = 0;
    for (int i = n - 2; i >= 0; --i) {
        dp[i] = i;
        if (i + 1 < n) dp[i] = dp[i + 1];
        if ((i + 1 < n) && (a[i + 1] - a[i] <= 2)) dp[i] = dp[i + 1] - 1;
        sum += dp[i];
        if (dp[i] >= 0) ++cnt;
    }
    if (dp[0] >= 0) ans = a[1] + (a[1] - a[0]) * dp[0];
    if (cnt == n)
        cout << -1
             << "\n";
    else {
        cout << cnt << "\n";
        for (int i = 0; i < n; ++i) {
            if (dp[i] < 0) continue;
            cout << ans - (a[1] - a[0]) * dp[i] << " ";
        }
    }
}
