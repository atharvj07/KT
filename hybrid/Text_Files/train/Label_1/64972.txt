from collections import defaultdict
import bisect
t = int(input())
for i in range(t):
    n = int(input())
    x = list(map(int, input().split()))
    y = []
    for i in x:
        y.append(i)
    y.sort()
    z = defaultdict(int)
    for i in y:
        z[i] += 1
    ans = 0
    for i in z:
        if z[i] % 2 == 1:
            ans += 1
    print(ans)
