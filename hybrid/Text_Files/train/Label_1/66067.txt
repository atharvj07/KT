from sys import stdin
def solve():
    n,m=[int(x)for x in stdin.readline().split()]
    A=[0]
    G=[[]]
    vl=[int(x) for x in stdin.readline().split()]
    for i in range(n):
        A.append([vl[i],0])
        G.append([])
    for i in range(m):
        a,b,c=map(int,stdin.readline().split())
        A[a][1]+=c
        A[b][1]+=c
        G[a].append(b)
        G[b].append(a)
    seen=[False]*n
    def dfs(v):
        if seen[v]:
            return
        seen[v]=True
        for u in G[v]:
            if not seen[u]:
                dfs(u)
                A[v][0]+=A[u][0]
                A[v][1]+=A[u][1]
    dfs(1)
    mx=0
    for i in range(1,n+1):
        if A[i][1]==0:
            continue
        mx=max(mx,A[i][0]/A[i][1])
    return mx
print("{0:.12f}".format(solve()))
