#include <bits/stdc++.h>
using namespace std;

const int MAXN = 101;
const int INF = 1e9 + 1;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    int ans = INF;
    for (int x = 1; x <= n; x++) {
        int cost = 0;
        for (int i = 1; i <= n; i++) {
            cost += abs(x - i) * a[i - 1];
        }
        ans = min(ans, cost);
    }
    cout << ans << '\n';
    return 0;
}
