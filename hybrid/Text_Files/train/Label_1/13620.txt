#include <algorithm>
#include <cstdio>
using namespace std;
#define ll long long
const ll inf = 1e18;
ll mi[500001];
ll ma[500001];
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n, q;
    scanf("%d %d", &n, &q);
    for (int i = 0; i < n; i++) {
      scanf("%lld", &mi[i]);
      ma[i] = mi[i];
    }
    for (int i = 1; i < n; i++) {
      mi[i] = min(mi[i], mi[i - 1]);
      ma[i] = max(ma[i], ma[i - 1]);
    }
    for (int i = 0; i < q; i++) {
      ll tgt;
      scanf("%lld", &tgt);
      while (n > 1) {
        ll hi = ma[n - 1];
        ll lo = mi[n - 1];
        ll mid = (hi + lo) >> 1;
        ll left = inf;
        if (lo <= mid) left = mi[n - 2] >= lo ? mi[n - 2] : mid;
        ll rem = tgt - left;
        if (rem > 0 && rem <= hi) {
          ma[n - 1] = mid;
          n--;
          tgt = rem;
        } else
          break;
      }
      puts(tgt == 0 ? "Yes" : "No");
    }
  }
}
