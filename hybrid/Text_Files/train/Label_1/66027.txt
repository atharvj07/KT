import queue

n, k = map(int, input().split())
books = list(map(int, input().split()))

lib = queue.Queue()
cost = 0

for book in books:
    if book in lib.queue:
        lib.get(block=False)
    else:
        if lib.qsize() == k:
            lib.get(block=False)
        cost += 1
    lib.put(book)

print(cost)
