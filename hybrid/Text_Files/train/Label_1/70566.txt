N = int(input().strip())
A = list(map(int, input().strip().split()))

left, right = 0, N-1
last_num = -1
turns = 0

while left <= right:
    if A[left] > last_num and (A[right] <= last_num or A[left] < A[right]):
        last_num = A[left]
        left += 1
    elif A[right] > last_num:
        last_num = A[right]
        right -= 1
    else:
        break
    turns += 1

print("Alice" if turns % 2 == 1 else "Bob")
