import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[] b = a.clone();
        int cnt = 0;
        while (true) {
            cnt++;
            boolean changed = false;
            for (int i = 1; i < n - 1; i++) {
                int m = (a[i - 1] + a[i] + a[i + 1]) / 2;
                if (b[i] != m) {
                    b[i] = m;
                    changed = true;
                }
            }
            if (!changed) {
                break;
            }
            a = b.clone();
        }

        if (cnt == 1) {
            System.out.println(-1);
        } else {
            System.out.println(cnt - 1);
            for (int i = 0; i < n; i++) {
                System.out.print(b[i] + " ");
            }
            System.out.println();
        }
    }
}
