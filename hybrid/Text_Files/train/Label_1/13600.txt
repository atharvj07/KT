#include <cstdio>
#include <algorithm>
#include <cmath>

int main() {
  static const int N = 105, inf = 1e9;
  char s[N];
  int n[3], p[3], cnt, i, j, x = inf;
  scanf("%s%d%d%d%d%d%d%d", s, n, n + 1, n + 2, p, p + 1, p + 2, &cnt);
  for (i = 0; s[i]; i++) {
    for (j = 0; j < 3; j++) {
      if (s[i] == s[0] + j)
        x = std::min(x, n[j] / (i + 1));
    }
  }
  int ans = 0;
  for (j = 0; j < 3; j++) {
    ans = std::max(ans, std::min(cnt / p[j], x * n[j]));
  }
  printf("%d", ans);
}
