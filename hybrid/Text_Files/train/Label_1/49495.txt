import java.util.Arrays;
import java.util.Scanner;

class Main {
 public static void main(String[] args) {
  Scanner in = new Scanner(System.in);
  int d = in.nextInt(), n = in.nextInt(), m = in.nextInt();
  int[] p = new int[m + 2];
  int[] dist = new int[m + 2];
  dist[0] = 0;  // DIST TO 1ST GAS STATION
  for (int i = 1; i <= m; i++) {
   p[i] = in.nextInt();
   dist[i] = in.nextInt();
  }
  dist[m + 1] = d;
  p[m + 1] = 0;
  int best = -1;
  for (int i = 0; i <= m; i++) {
   int min = p[i];
   for (int j = i + 1; j <= m + 1; j++) {
    int distToNext = dist[j] - dist[i];
    if (distToNext > n) break;
    min = Math.min(min, p[j]);
   }
   if (min == 0) break;
   if (best != -1) best = Math.min(best, min);
   else best = min;
  }
  System.out.println(best);
 }
}
