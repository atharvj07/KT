#include <iostream>
#include <vector>

int main() {
  std::string recipe;
  std::cin >> recipe;
  int lst[3] = {0, 0, 0};
  for (auto &c : recipe) {
    if (c == 'B')
      lst[0]++;
    else if (c == 'S')
      lst[1]++;
    else
      lst[2]++;
  }
  int n[3], p[3];
  for (int i = 0; i < 3; i++) std::cin >> n[i];
  for (int i = 0; i < 3; i++) std::cin >> p[i];
  long long ans = 1e18;
  int r;
  std::cin >> r;
  for (int i = 0; i < 3; i++) {
    if (lst[i] && ((long long)n[i] + r / p[i]) / lst[i] < ans)
      ans = ((long long)n[i] + r / p[i]) / lst[i];
  }
  if (ans == 1e18)
    std::cout << 0 << std::endl;
  else
    std::cout << ans << std::endl;
  return 0;
}
