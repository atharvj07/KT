import java.util.*;
import java.io.*;

public class Main {
    static int n;
    static long[] jumps;
    static long[] cost;

    static int minCost() {
        jumps = Arrays.stream(jumps).sorted().distinct().toArray();
        int ans = 0;
        for (int i = 0; i < jumps.length; i++) {
            if (jumps[i] > ans + 1) return -1;
            while (jumps[i] > ans) {
                ans += jumps[i];
                cost[i]++;
            }
        }
        long sum = 0;
        for (int i = 0; i < n; i++) sum += cost[i] * jumps[i];
        return (int) sum;
    }

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        jumps = new long[n];
        cost = new long[n];
        for (int i = 0; i < n; i++) jumps[i] = sc.nextLong();
        for (int i = 0; i < n; i++) cost[i] = sc.nextLong();
        System.out.println(minCost());
    }
}
