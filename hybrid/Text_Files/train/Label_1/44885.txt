import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        long[] x = new long[n];
        long[] y = new long[m];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextLong();
        }
        for (int i = 0; i < m; i++) {
            y[i] = sc.nextLong();
        }

        Arrays.sort(x);
        Arrays.sort(y);

        long ans = 0;
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = 0; k < m - 1; k++) {
                    for (int l = k + 1; l < m; l++) {
                        ans += (x[j] - x[i]) * (y[l] - y[k]);
                        ans %= 1000000007;
                    }
                }
            }
        }

        System.out.println(ans);
    }
}
