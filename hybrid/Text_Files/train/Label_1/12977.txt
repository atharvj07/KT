#include <cstdio>
#include <vector>
using namespace std;

int main() {
    vector<int> v(6);
    for (int i = 0; i < 6; i++) {
        scanf("%1d", &v[i]);
    }
    int ans = 0;
    for (int i = 0, sum1 = 0, sum2 = 0; i < 6; i++) {
        sum1 += v[i];
        sum2 += v[5 - i];
        if (sum1 > sum2) swap(v[i], v[5 - i]), ans++;
    }
    printf("%d\n", ans);
}
