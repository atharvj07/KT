#include <bits/stdc++.h>
using namespace std;

int main() {
  string s;
  int m;
  cin >> s >> m;
  if (m <= 1) {
    cout << "YES\n";
    return 0;
  }
  if (s[10] == '0' || s[1] == '0' || (s[1] == '1' && s[9] == '1' && m > 3)) {
    cout << "NO\n";
    return 0;
  }
  cout << "YES\n";
  if (m > 2) {
    cout << 8 << " " << 10 << " ";
    m -= 2;
  }
  while (m--) {
    if (s[10] == '1')
      cout << 10, m--;
    if (m >= 2 && s[9] == '1')
      cout << 9 << " " << 9 << " ", m -= 2;
    else if (s[1] == '1')
      cout << 1, m--;
    if (m >= 2 && s[2] == '1')
      cout << 2 << " " << 2 << " ", m -= 2;
    else if (s[3] == '1')
      cout << 3, m--;
    if (m >= 2 && s[4] == '1')
      cout << 4 << " " << 4 << " ", m -= 2;
    else if (s[5] == '1')
      cout << 5, m--;
    if (m >= 2 && s[6] == '1')
      cout << 6 << " " << 6 << " ", m -= 2;
    else if (s[7] == '1')
      cout << 7, m--;
    if (m >= 2 && s[8] == '1')
      cout << 8 << " " << 8 << " ", m -= 2;
  }
  cout << "\n";
  return 0;
}
