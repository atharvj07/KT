r, c, n, k = map(int, input().split())
a = [[0] * c for i in range(r)]
for i in range(n):
    x, y = map(int, input().split())
    x -= 1
    y -= 1
    a[x][y] = 1
ans = 0
for x0 in range(r):
    for x1 in range(x0, r):
        for y0 in range(c):
            for y1 in range(y0, c):
                sum = 0
                for i in range(x0, x1 + 1):
                    for j in range(y0, y1 + 1):
                        sum += a[i][j]
                if sum >= k:
                    ans += 1
print(ans)
