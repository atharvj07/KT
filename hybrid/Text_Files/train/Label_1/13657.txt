#include <iostream>
#include <vector>
#include <algorithm>
#include<map>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    map<int,int>mp;
    for (int i = 0; i < n; i++) {
        cin >> v[i];
        mp[v[i]]++;
    }
    int ans=0,cnt=0;
    for(int i=0;i<n;i++){
        cnt-=mp[v[i]]-1;
        ans+=cnt;
        mp[v[i]]--;
        cnt++;
    }
    cout<<ans<<endl;
    return 0;
}
