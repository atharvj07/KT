import java.util.Arrays;
import java.util.Scanner;
import java.util.Set;
import java.util.TreeSet;

public class Main {
    private static final long MOD = 998244353;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        long[] intervals = new long[n * 2];
        for (int i = 0; i < n; i++) {
            intervals[i * 2] = input.nextInt();
            intervals[i * 2 + 1] = input.nextInt();
        }
        Arrays.sort(intervals);
        Set<Integer> activeIntervals = new TreeSet<>();
        long ans = 0;
        for (long p : intervals) {
            int type = activeIntervals.contains((int) p) ? -1 : 1;
            if (type == 1 && activeIntervals.size() == k - 1) {
                ans = (ans + type) % MOD;
            }
            activeIntervals.add((int) p);
            if (activeIntervals.size() > k) {
                activeIntervals.remove((int) p);
            }
        }
        System.out.println(ans);
    }
}
