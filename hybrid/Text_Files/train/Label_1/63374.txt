def dfs(x, y, px, py, color, visited, board):
    if visited[x][y]:
        return True
    visited[x][y] = True
    directions = [(0, 1), (0, -1), (1, 0), (-1, 0)]
    for dx, dy in directions:
        nx, ny = x + dx, y + dy
        if nx >= 0 and nx < len(board) and ny >= 0 and ny < len(board[0]) and (nx != px or ny != py) and board[nx][ny] == color:
            if dfs(nx, ny, x, y, color, visited, board):
                return True
    return False

def has_cycle(board):
    n, m = len(board), len(board[0])
    visited = [[False for _ in range(m)] for _ in range(n)]
    for i in range(n):
        for j in range(m):
            if not visited[i][j]:
                if dfs(i, j, -1, -1, board[i][j], visited, board):
                    return "Yes"
    return "No"

n, m = map(int, input().split())
board = [input().strip() for _ in range(n)]
print(has_cycle(board))
