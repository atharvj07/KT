import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int[][] grid = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    grid[j][k] = sc.nextInt();
                }
            }
            System.out.println(isValid(grid));
        }
    }

    private static boolean isValid(int[][] grid) {
        // Check if the sum of opposite faces is 7
        for (int i = 0; i < 5; i++) {
            if (grid[i][0] + grid[i][4] != 7 || grid[0][i] + grid[4][i] != 7) {
                return false;
            }
        }

        // Check if there are any duplicate numbers on the faces
        Set<Integer> numbers = new HashSet<>();
        for (int i = 0; i < 5; i++) {
            for (int j = 0; j < 5; j++) {
                if (numbers.contains(grid[i][j])) {
                    return false;
                }
                numbers.add(grid[i][j]);
            }
        }

        return true;
    }
}
