import java.util.*;

public class AtomPuzzleSolver {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int N = scanner.nextInt();
        int K = scanner.nextInt();

        int[] A = new int[N];
        int[] D = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = scanner.nextInt();
        }
        for (int i = 0; i < N; i++) {
            D[i] = scanner.nextInt();
        }

        // Initialize bonds and their costs
        int[] E = new int[N];
        for (int i = 0; i < N - 1; i++) {
            E[i] = i + 1;
        }
        int[] bondCost = new int[N];
        for (int i = 0; i < N; i++) {
            bondCost[i] = D[i] - A[i];
        }

        // Sort bonds by cost in descending order
        Arrays.sort(bondCost);

        // Change bonds to minimize cost
        int totalBondCost = 0;
        for (int i = 0; i < K; i++) {
            totalBondCost += bondCost[i];
        }

        // Calculate maximum energy
        int maxEnergy = 0;
        for (int i = 0; i < N; i++) {
            if (D[i] - totalBondCost <= A[i]) {
                maxEnergy += A[i] - (D[i] - totalBondCost);
            }
        }

        System.out.println(maxEnergy);
    }
}
