n=input()
l=list(n);ans=[]
f=0
for i in range(len(l)):
    for j in range(i+1,len(l)):
        if int(l[i])<int(l[j]):
            l[i],l[j]=l[j],l[i]
            s=''.join(l)
            ans.append(s)
            l[i],l[j]=l[j],l[i]
            f=1
        elif int(l[i])==int(l[j]):
            continue
        else:
            break
if f==0 or int(n)%2==0:print(-1)
else:print(max(ans))
