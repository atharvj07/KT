import java.io.*;
import java.util.*;

class Main {
  public static void main(String[] args) throws Exception {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    String t = in.readLine(), s = "";

    for (int i = 0; i < t.length(); i++) {
      if (t.charAt(i) != 'a') {
        s += t.charAt(i);
      }
    }

    for (int i = 0; i < t.length(); i++) {
      if (i + s.length() < t.length() && t.charAt(i) != t.charAt(i + s.length())) {
        System.out.println(":(");
        return;
      }
    }

    System.out.println(s);
  }
}
