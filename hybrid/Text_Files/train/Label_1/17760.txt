#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> left(n), right(n);
  for (int i = 0; i < n; i++) { cin >> left[i] >> right[i]; }
  int t = 0;
  for (int i = 0; i < n; i++) {
    if (left[i] != right[i]) t++;
  }
  cout << (t + 1) / 2 << endl;
}
