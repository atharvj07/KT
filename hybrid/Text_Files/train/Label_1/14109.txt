#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define int ll
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    sort(a.begin(), a.end());
    int ans = x / a[n - 1];
    x %= a[n - 1];
    ans += x / a[n - 2];
    x %= a[n - 2];
    ans += x > 0;
    cout << ans << endl;
  }
  return 0;
}
