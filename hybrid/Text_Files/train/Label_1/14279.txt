#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int& x : a) { std::cin >> x; }
  int sum = 0;
  for (int x : a) { sum += x; }
  if (sum % 2) { std::cout << -1 << std::endl; return 0; }
  int b = sum / 2;
  std::sort(a.begin(), a.end());
  int idx = 0;
  while (idx < n && a[idx] <= b) { sum -= a[idx]; idx++; }
  if (idx != n) { std::cout << b - sum << std::endl; }
  idx = n - 1;
  while (idx >= 0 && a[idx] > b) { sum -= a[idx]; idx--; }
  if (idx != -1) { std::cout << b - sum << std::endl; }
}
