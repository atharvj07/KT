#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<string> words(n);
  for (int i = 0; i < n; i++) { cin >> words[i]; }
  string res = words[0];
  for (int i = 1; i < n; i++) {
    int j = 0, k, m = words[i].length();
    for (k = res.length(); k >= 0; k--) {
      if (j < m && res[k] == words[i][j]) { j++; } else { break; }
    }
    res += words[i].substr(j);
  }
  cout << res << endl;
  return 0;
}
