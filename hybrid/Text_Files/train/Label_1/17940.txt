#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int main(){
    int t;
    cin >> t;
    while(t--){
        int n;
        cin >> n;
        vector<int> a(n);
        unordered_set<int> s;
        for(int &x : a) {
            cin >> x;
            s.insert(x);
        }
        for(int i = 0; i < n; ++i) {
            for(int j = 0; j < n; ++j) {
                if(i == j) continue;
                int x = (a[i] % a[j]);
                if(!s.count(x)) {
                    cout << a[i] << ' ' << a[j] << '\n';
                }
            }
        }
    }
}
