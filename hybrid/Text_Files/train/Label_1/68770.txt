import sys
input = sys.stdin.readline

def main():
    n, m = map(int, input().split())
    a = []
    for _ in range(n):
        a.append(list(map(int, input().split())))
    # dp[i][j][0] = 最小値
    # dp[i][j][1] = 最大値
    dp = [[[-1] * 3 for _ in range(m)] for _ in range(n)]
    def dfs(i, j, prv):
        if i == n or j == m:
            return 0
        if dp[i][j][prv] != -1:
            return dp[i][j][prv]
        res = 0
        if a[i][j] == 0:
            mn = 1
            if prv == 1:
                mn = a[i][j - 1] + 1
            for k in range(mn, 8001):
                if (prv == 0 and (i == 0 or k > a[i - 1][j])) or (prv == 1 and k > a[i][j - 1]) or (prv == 2 and k > a[i][j + 1]):
                    continue
                if j == m - 1 or k < a[i][j + 1]:
                    res = max(res, k + dfs(i, j + 1, 1))
                if i == n - 1 or k < a[i + 1][j]:
                    res = max(res, k + dfs(i + 1, j, 2))
                if prv == 0:
                    res = max(res, k + dfs(i + 1, j, 0))
        else:
            if prv == 0 or a[i][j] > a[i - 1][j]:
                res = max(res, dfs(i + 1, j, 0))
            if prv == 1 or a[i][j] > a[i][j - 1]:
                res = max(res, dfs(i, j + 1, 1))
            if j == m - 1 or a[i][j] < a[i][j + 1]:
                res = max(res, dfs(i, j + 1, 1))
            if i == n - 1 or a[i][j] < a[i + 1][j]:
                res = max(res, dfs(i + 1, j, 2))
        dp[i][j][prv] = res
        return res
    ans = dfs(0, 0, 0)
    print(ans if ans > 0 else -1)

main()
