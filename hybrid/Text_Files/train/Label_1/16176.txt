#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int n, q, a[maxn], b[maxn];
bool vis[31];
int main() {
    cin >> n >> q;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        for (int j = 0; j < 31; j++) {
            if (a[i] & (1 << j)) { vis[j] = 1; }
        }
    }
    for (int i = 1; i <= q; i++) {
        cin >> b[i];
        int ans = 0;
        for (int j = 0; j < 31; j++) {
            if (b[i] & (1 << j)) {
                if (!vis[j]) { ans = -1; break; }
                ans++;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
