#include <iostream>
#include <vector>

int main() {
  long long l, r;
  std::cin >> l >> r;
  long long L = l, R = r;
  for (long long i = 2; i * i <= r; ++i) {
    if (l % i == 0) {
      while ((l % i == 0) && (l <= r)) {
        l = l / i;
      }
      L = L / i;
      if (l % i == 0) {
        L = L / i;
        break;
      }
    }
    if (r % i == 0) {
      while ((r % i == 0) && (l <= r)) {
        r = r / i;
      }
      R = R / i;
      if (r % i == 0) {
        R = R / i;
        break;
      }
    }
  }
  if (L == 1 && R == 1) {
    std::cout << 0 << std::endl;
  } else if (L * 2 <= R) {
    std::cout << 1 + (R - L * 2) / 3 << std::endl;
  } else {
    std::cout << R - L << std::endl;
  }
  return 0;
}
