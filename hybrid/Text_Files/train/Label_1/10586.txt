#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    long long sum = 0;
    for (int i = 0; i < n; i++) sum += a[i];
    sort(a.begin(), a.end());

    long long cost = 0;
    for (int i = 0, j = n - 1; i < j;) {
      if (-a[i] <= a[j]) {
        cost -= a[i];
        sum += a[i];
        i++;
      } else {
        cost += a[j];
        sum += a[j];
        j--;
      }
    }

    cout << cost << '\n';
  }
  return 0;
}
