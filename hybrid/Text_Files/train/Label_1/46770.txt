import java.util.Scanner;

public class InnopolisHills {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] hills = new int[n];
        for (int i = 0; i < n; i++) {
            hills[i] = scanner.nextInt();
        }

        int[] minHours = new int[n / 2 + 1];
        minHours[0] = 0;

        for (int i = 1; i <= n / 2; i++) {
            int min = Integer.MAX_VALUE;
            for (int j = 0; j < i; j++) {
                int requiredHeight = hills[j] - (i - 1);
                if (requiredHeight > 0) {
                    min = Math.min(min, minHours[j] + requiredHeight);
                }
            }
            minHours[i] = min;
        }

        for (int i = 0; i <= n / 2; i++) {
            System.out.print(minHours[i] + " ");
        }
    }
}
