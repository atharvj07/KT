#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    bool consistent = true;
    for (int i = 0; i < s.size(); i++) {
      if (s[i] == 'E') {
        if (s[(i + 1) % s.size()] == 'N') {
          consistent = false;
          break;
        }
      }
    }
    if (consistent) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
  return 0;
}
