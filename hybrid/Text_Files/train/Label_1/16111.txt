#include <bits/stdc++.h>

using namespace std;

struct Query {
  int x1, y1, x2, y2;
};

const int inf = 1e9;
const int max_n = 1e8 + 7;

int n, m;
vector<int> stairs, elevators;
vector<Query> queries;

int main() {
  cin >> n >> m;
  int c_l, c_e, v;
  cin >> c_l >> c_e >> v;
  for (int i = 0; i < c_l; ++i) {
    int l;
    cin >> l;
    stairs.push_back(l);
  }
  for (int i = 0; i < c_e; ++i) {
    int e;
    cin >> e;
    elevators.push_back(e);
  }
  int q;
  cin >> q;
  for (int i = 0; i < q; ++i) {
    int x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2;
    queries.push_back({x1, y1, x2, y2});
  }
  vector<int> ans(q);
  for (int i = 0; i < q; ++i) {
    auto query = queries[i];
    int x1 = query.x1, y1 = query.y1, x2 = query.x2, y2 = query.y2;
    if (x1 == x2) {
      ans[i] = abs(y1 - y2);
      continue;
    }
    int min_elevator = max_n, max_elevator = -max_n;
    for (auto elevator : elevators) {
      min_elevator = min(min_elevator, elevator);
      max_elevator = max(max_elevator, elevator);
    }
    int dist_to_x1_stairs = inf, dist_to_x2_stairs = inf;
    for (auto stair : stairs) {
      dist_to_x1_stairs = min(dist_to_x1_stairs, abs(stair - y1));
      dist_to_x2_stairs = min(dist_to_x2_stairs, abs(stair - y2));
    }
    if (y1 > max_elevator) {
      dist_to_x1_stairs = min(dist_to_x1_stairs, y1 - max_elevator + 1);
    }
    if (y2 > max_elevator) {
      dist_to_x2_stairs = min(dist_to_x2_stairs, y2 - max_elevator + 1);
    }
    if (y1 < min_elevator) {
      dist_to_x1_stairs = min(dist_to_x1_stairs, min_elevator - y1 + 1);
    }
    if (y2 < min_elevator) {
      dist_to_x2_stairs = min(dist_to_x2_stairs, min_elevator - y2 + 1);
    }
    int ans1 = dist_to_x1_stairs + 1 + abs(x1 - x2) + dist_to_x2_stairs;
    int ans2 = min(abs(y1 - min_elevator), abs(y1 - max_elevator)) + 1 + min(abs(x1 - x2) * v, n - 1) + min(abs(y2 - min_elevator), abs(y2 - max_elevator)) + 1;
    ans[i] = min(ans1, ans2);
  }
  for (auto answer : ans) {
    cout << answer << '\n';
  }
  return 0;
}
