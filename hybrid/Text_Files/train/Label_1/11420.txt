#include <bits/stdc++.h>
#define nl '\n'
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<bool>> grid(n + 1, vector<bool>(m + 1));
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        while (x--) {
            int y;
            cin >> y;
            grid[i][y] = true;
        }
    }
    for (int i = 1; i <= m; i++) {
        bool ok = false;
        for (int j = 1; j <= n; j++) {
            ok |= grid[j][i];
        }
        if (!ok) {
            cout << "NO" << nl;
            return 0;
        }
    }
    cout << "YES" << nl;
    return 0;
}
