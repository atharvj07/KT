#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int main() {
    int n, k; cin >> n >> k;
    vector<int> c(k * n), f(n), h(k);
    for (int& x : c) cin >> x;
    for (int& x : f) cin >> x;
    for (int& x : h) cin >> x;
    sort(c.begin(), c.end(), greater<int>());
    sort(f.begin(), f.end());
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int j = 0;
        while (j < k && c[j] != f[i]) j++;
        if (j == k) continue;
        int cnt = 0;
        while (j < k && c[j] == f[i]) ans += h[cnt++], j++;
    }
    cout << ans << endl;
    return 0;
}
