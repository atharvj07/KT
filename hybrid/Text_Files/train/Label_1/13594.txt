#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int main() {
  long long x, y, l, r;
  cin >> x >> y >> l >> r;

  long long ans = 0;
  for (long long i = l; i <= r; i++) {
    bool flag = false;
    for (long long j = 0; j <= 62; j++) {
      for (long long k = 0; k <= 62; k++) {
        if (i == pow(x, j) + pow(y, k)) {
          flag = true;
          break;
        }
      }
      if (flag) break;
    }
    if (!flag) ans++;
  }

  cout << ans << endl;

  return 0;
}
