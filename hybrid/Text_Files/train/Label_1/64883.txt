def solve_queries(queries):
    for n, s in queries:
        if s[0] >= s[1]:
            print("NO")
        else:
            print("YES")
            print(2)
            print(s[0], s[1:])

q = int(input().strip())
queries = []
for _ in range(q):
    n = int(input().strip())
    s = input().strip()
    queries.append((n, s))
