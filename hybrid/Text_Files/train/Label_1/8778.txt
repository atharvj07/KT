#include <iostream>
using namespace std;
#define int long long
#define endl "\n"
const int mxN = 1e5 + 5;
int a[mxN];
int f[mxN];
int32_t main() {
  cin.tie(0);
  ios_base::sync_with_stdio(false);
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
      f[a[i]]++;
    }
    for (int i = 0, j = n; j >= 0; j--) {
      if (!f[j]) {
        i = j;
      }
      f[j + 1] += f[j];
    }
    int num = n;
    for (int j = 0; j <= n; j++) {
      if (f[j] > 0) {
        if (j != 0) { num += min(f[j - 1], k); }
        k -= min(f[j - 1], k);
        if (k == 0) { break; }
      }
    }
    if (k > 0) { num++; }
    cout << num << endl;
    for (int i = 0; i < mxN; i++) f[i] = 0;
  }
}
