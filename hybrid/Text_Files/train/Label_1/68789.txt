import sys, itertools
input = sys.stdin.readline
n, m = map(int, input().split())
a = list(map(int, input().split()))
b = []
for i in range(n):
    if a[i] == 1: b.append(i)
for _ in range(m):
    l, r = map(int, input().split())
    if b.count(l - 1) & 1 ^ b.count(r) & 1: print(1)
    else: print(0)
