import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] range = new int[n][2];
        for (int i = 0; i < n; i++) {
            range[i][0] = sc.nextInt();
            range[i][1] = sc.nextInt();
        }
        Arrays.sort(range, (a, b) -> a[0] - b[0]);
        int max = 0;
        int end = 0;
        for (int[] r : range) {
            if (r[0] >= end) {
                end = r[1];
                max++;
            }
        }
        System.out.println(max);
    }
}
