#include<iostream>
#include<string>
#include<map>
using namespace std;
int main(){
    int t;
    cin>>t;
    map<pair<int,int>,int> visited;
    while(t--){
        int time=0;
        int x=0,y=0;
        string s;
        cin>>s;
        for(char c:s){
            if(c=='N') ++y;
            if(c=='S') --y;
            if(c=='E') ++x;
            if(c=='W') --x;
            if(visited[make_pair(x,y)]) time+=1;
            else time+=5,visited[make_pair(x,y)]=1;
        }
        cout<<time<<endl;
    }
}
