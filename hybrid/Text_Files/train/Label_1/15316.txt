#include <iostream>
#include <string>
using namespace std;
int main() {
  int n;
  string s;
  cin >> n >> s;
  int ans1 = 0, ans2 = 0;
  int cnt = 0;
  int mx = 0;
  for (char c : s) {
    if (cnt == 0) {
      if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')) {
        mx++;
      } else {
        ans1 = max(ans1, mx);
        mx = 0;
      }
    } else {
      if (c == '(')
        cnt++;
      else if (c == ')')
        cnt--;
      else if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z'))
        ans2++;
    }
    if (c == '(') cnt++;
    if (c == ')') cnt--;
  }
  ans1 = max(ans1, mx);
  cout << ans1 << ' ' << ans2 << endl;
  return 0;
}
