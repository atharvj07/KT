#include <bits/stdc++.h>
using namespace std;

const int N = 100005;
struct Point { int x, y; };
Point p[N];
int w[N], n, L[N], R[N]; // L[i] = Leftmost point with special value = w[i] = R[i]

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> p[i].x >> p[i].y; }
  for (int i = 1; i <= n; i++) { cin >> w[i]; }

  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (p[j].x >= p[i].x && p[j].y >= p[i].y && p[j].x - p[j].y == w[i]) {
        L[i] = j;
        break;
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (p[j].x >= p[i].x && p[j].y >= p[i].y && p[j].x - p[j].y == w[i]) {
        R[i] = j;
        break;
      }
    }
  }
  vector<Point> ans;
  for (int i = 1; i <= n; i++) {
    if (L[i] > R[i]) {
      cout << "NO\n";
      return 0;
    }
    ans.push_back(p[L[i]]);
  }
  cout << "YES\n";
  for (auto x : ans) { cout << x.x << ' ' << x.y << '\n'; }
  return 0;
}
