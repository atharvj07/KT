#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
typedef pair<int, int> pii;
#define all(x) (x).begin(), (x).end()
#define llint long long int
#define fastIO ios::sync_with_stdio(false); cin.tie(0); cout.tie(0);
llint mod = 1e9 + 7;
llint g0, x, y, z;
int main() {
    llint n, m, a, b;
    cin >> n >> m >> a >> b;
    cin >> g0 >> x >> y >> z;
    vector<vector<int>> dp(n + 5, vector<int>(m + 5, 0));
    dp[1][1] = g0;
    for(int i = 1; i <= n; i++)
        for(int j = 1; j <= m; j++) {
            if(i == 1 && j == 1) continue;
            dp[i][j] = ((1ll * dp[i - 1][j] * x) % z + y) % z;
        }
    llint ans = 0;
    for(int i = 1; i <= n - a + 1; i++) {
        for(int j = 1; j <= m - b + 1; j++) {
            int mini = dp[i][j];
            for(int k = i; k <= i + a - 1; k++)
                for(int l = j; l <= j + b - 1; l++)
                    mini = min(mini, dp[k][l]);
            ans += mini;
        }
    }
    cout << ans << "\n";
    return 0;
}
