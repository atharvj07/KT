import java.util.*;

public class Main {
  static int n, m, s, t;
  static List<Integer>[] adj;
  static int[] dist;
  static boolean[] vis;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    m = sc.nextInt();
    s = sc.nextInt() - 1;
    t = sc.nextInt() - 1;
    adj = new List[n];
    for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
    for (int i = 0; i < m; i++) {
      int u = sc.nextInt() - 1;
      int v = sc.nextInt() - 1;
      adj[u].add(v);
      adj[v].add(u);
    }
    dist = new int[n];
    vis = new boolean[n];
    bfs(s);
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (vis[i] && dist[i] == dist[t]) ans++;
    }
    for (int i = 0; i < n; i++) {
      if (!vis[i]) ans += i != s && i != t ? 1 : 0;
    }
    System.out.println(ans);
  }

  static void bfs(int u) {
    Queue<Integer> q = new LinkedList<>();
    q.add(u);
    vis[u] = true;
    while (!q.isEmpty()) {
      int v = q.poll();
      for (int next : adj[v]) {
        if (!vis[next]) {
          vis[next] = true;
          dist[next] = dist[v] + 1;
          q.add(next);
        }
      }
    }
  }
}
