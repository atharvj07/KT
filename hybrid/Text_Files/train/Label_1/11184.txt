#include <iostream>
using namespace std;
long long n, mex, l, r, M = 1e9 + 7, ans[2], a[21];
long long power(long long x, long long y){
    if (y == 0)return 1;
    if (y == 1)return x % M;
    if (y&1)return (x * power(x, y - 1)) % M;
    else return power((x * x) % M, y / 2) % M;
}
int main(){
    cin >> n >> l >> r;
    mex = 1;
    for (long long i = 1; i <= r; i++)a[i] = power(2, i - 1);
    for (long long i = 1; i <= n; i++)
        if (i <= l)ans[1] += 1, ans[0]++;
        else{
            ans[1] += a[r];
            ans[0] += mex;
            mex++;
        }
    cout << ans[0] << ' ' << ans[1] << '\n';
    return 0;
}
