#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<string> passwords(n);
  for (int i = 0; i < n; i++) { cin >> passwords[i]; }
  string correctPassword;
  cin >> correctPassword;
  sort(passwords.begin(), passwords.end());
  int bestCase = 1, worstCase = 1;
  for (int i = 0; i < n; i++) {
    if (passwords[i] == correctPassword) { break; }
    bestCase++;
    worstCase += 5;
  }
  cout << bestCase << " " << worstCase << endl;
  return 0;
}
