#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int a, b, c;
    cin >> a >> b >> c;
    vector<int> x(a), y(b), z(c);
    for (int i = 0; i < a; i++) cin >> x[i];
    for (int i = 0; i < b; i++) cin >> y[i];
    for (int i = 0; i < c; i++) cin >> z[i];
    sort(z.begin(), z.end());
    int ans = 0, l = 0, r = c - 1;
    for (int i = 0; i < a; i++) {
        while (l < r && z[l] != x[i]) l++;
        if (l >= r) break;
        ans++, l++;
    }
    l = 0, r = c - 1;
    for (int i = 0; i < b; i++) {
        while (l < r && z[r] != y[i]) r--;
        if (r <= l) break;
        ans++, r--;
    }
    cout << ans << endl;
    return 0;
}
