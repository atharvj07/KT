import java.util.*;

class Main {
    static class Node {
        int val = 0;
        int cnt;
        boolean visited = false;
        List<Node> children = new ArrayList<>();
    }

    static int solve(Node root) {
        if (root.visited)
            return root.val;
        root.visited = true;
        int max = root.val;
        for (Node n : root.children) {
            max = Math.max(max, solve(n) + root.val);
            root.val += n.val;
        }
        return max;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Node[] nodes = new Node[n];
        for (int i = 0; i < n; i++) {
            nodes[i] = new Node();
            nodes[i].val = sc.nextInt();
        }
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            nodes[u].children.add(nodes[v]);
            nodes[v].children.add(nodes[u]);
        }
        int[] res = new int[n];
        for (int i = 0; i < n; i++) {
            res[i] = solve(nodes[i]);
        }
        for (int i = 0; i < n; i++) {
            System.out.print(res[i] + " ");
        }
    }
}
