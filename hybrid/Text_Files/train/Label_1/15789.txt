#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int sum = (s[n - 1] - '0') % 2 == 0;
  for (int i = n - 2; i >= 0; i--) {
    int d = s[i] - '0';
    sum += (d % 2 == 0) * ((d % 4 == 0) + 1);
    sum += (i + 1) * (d % 2 == 0 && d != 0);
  }
  cout << sum << endl;
  return 0;
}
