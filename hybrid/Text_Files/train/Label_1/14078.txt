#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> grid(n);
  for (int i = 0; i < n; i++) cin >> grid[i];
  int ans = 0;
  for (int i = 0; i < n - 2; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] != grid[i + 1][j] && grid[i + 1][j] != grid[i + 2][j])
        ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
