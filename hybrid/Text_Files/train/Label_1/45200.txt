import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] cubes = new int[n][6];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 6; j++) {
                cubes[i][j] = scanner.nextInt();
            }
        }
        int max = 0;
        for (int i = 1; i <= 9; i++) {
            if (canMake(cubes, i)) {
                max = i;
            } else {
                break;
            }
        }
        System.out.println(max);
    }

    public static boolean canMake(int[][] cubes, int x) {
        boolean[] vis = new boolean[10];
        return dfs(cubes, x, vis, 0);
    }

    public static boolean dfs(int[][] cubes, int x, boolean[] vis, int cur) {
        if (cur == x) {
            return true;
        }
        if (cur > x) {
            return false;
        }
        for (int[] cube : cubes) {
            for (int i = 0; i < 6; i++) {
                int digit = cube[i];
                if (!vis[digit] && (cur == 0 || digit != 0)) {
                    vis[digit] = true;
                    if (dfs(cubes, x, vis, cur * 10 + digit)) {
                        return true;
                    }
                    vis[digit] = false;
                }
            }
        }
        return false;
    }
}
