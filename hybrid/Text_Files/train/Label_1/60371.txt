n, h = map(int, input().split())
a = list(map(int, input().split()))

a_sorted = sorted(a)
if n == 2 or h == 0:
    print(0)
    print("1 " * n)
else:
    min_diff = a_sorted[1] - a_sorted[0]
    second_min_diff = a_sorted[-1] - a_sorted[-2] if n > 2 else min_diff
    if h < min_diff:
        print(min_diff)
        partition = ["1" if x <= a_sorted[0] or x >= a_sorted[-1] else "2" for x in a]
    else:
        if h < second_min_diff:
            print(second_min_diff)
        else:
            print(h)
        partition = ["2" for _ in a]
    print(" ".join(partition))
