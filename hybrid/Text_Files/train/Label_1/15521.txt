#include <bits/stdc++.h>
#define all(v) v.begin(), v.end()
using namespace std;
typedef long long ll;
typedef pair <ll, ll> pi;
typedef vector <ll> vi;
typedef vector <pi> vpi;

const ll mod = 1e9 + 7, inf = 1e18 + 5, MX = 2e5 + 5;

vi adj[MX];
vi S;

void dfs(int v, int p){
	S.push_back(v);
	for(int u : adj[v]){
		if(u == p) continue;
		dfs(u, v);
	}
}

int main(){
	int n;
	cin >> n;
	vpi V(n);
	for(auto &[x, y] : V) cin >> x >> y, --x, --y;

	for(int i = 0; i < n; i++){
		int j = i;
		while(V[j].second != i) j = V[j].second;
		if(i == j) continue;
		adj[i].push_back(j);
		adj[j].push_back(i);
	}

	int R = -1;
	for(int i = 0; i < n; i++) if(adj[i].empty()){
		dfs(i, -1);
		if(S.size() & 1) R = S.back();
		break;
	}

	vi ans;
	if(R != -1) ans.push_back(R+1);
	for(int v : S){
		if(v == R) continue;
		if(adj[v].size() >= 2){
			ans.push_back(v+1);
			ans.push_back(adj[v][0]+1);
			break;
		}
	}

	cout << ans.size() << '\n';
	for(int v : ans) cout << v << ' ';
}
