#include<iostream>
#include<vector>
using namespace std;
vector<int> reduce(int a){
    vector<int> v;
    for(int i=1;i<=a;i++){
        if(a%i == 0){
            v.push_back(i);
        }
    }
    return v;
}
int main(){
    int n;cin>>n;
    vector<int> a(n+1);
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    for(int i=1;i<=n;i++){
        vector<int> v = reduce(a[i]);
        int mx = 0;
        for(int x : v){
            int s = x;
            for(int j=1;j<=n;j++){
                if(a[j]%s == 0){
                    s = max(s,a[j]);
                }
            }
            mx = max(mx,s);
        }
        a[i] = mx;
    }
    int ans = 0;
    for(int i=1;i<=n;i++){
        ans += a[i];
    }
    cout<<ans<<'\n';
    return 0;
}
