#include <iostream>
#include <string>
#include <vector>
#include <map>
#include <algorithm>
#include <cstring>

using namespace std;

const int MOD = 1000000007;
string s;
int n;
vector<pair<char, string>> q;
map<char, int> p;

int main() {
  cin >> s >> n;
  for (int i = 0; i < n; i++) {
    char c;
    string t;
    cin >> c >> t;
    q.push_back(make_pair(c, t));
  }
  for (int i = 0; i < s.size(); i++) {
    p[s[i]]++;
  }
  for (int i = 0; i < q.size(); i++) {
    p[q[i].first] -= p[q[i].first];
    for (int j = 0; j < q[i].second.size(); j++) {
      p[q[i].second[j]]++;
    }
    string t = "";
    for (int j = 0; j < p.size(); j++) {
      if (p[j] > 0) {
        for (int k = 0; k < p[j]; k++) {
          t += j + '0';
        }
      }
    }
    s = t;
  }
  int ans = 0;
  for (int i = 0; i < s.size(); i++) {
    ans = (ans * 10 + (s[i] - '0')) % MOD;
  }
  cout << ans << endl;
  return 0;
}
