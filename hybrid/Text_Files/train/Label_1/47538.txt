import java.util.Scanner;

public class Main {
    private static char[][] map;
    private static int[] dx = {0, 1, 0, -1};
    private static int[] dy = {1, 0, -1, 0};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            if (n == 0 && k == 0) break;
            map = new char[n][n];
            int res = dfs(0, 0, 0, 0, k);
            if (res == 0) System.out.println("No");
            else {
                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < n; j++) System.out.print(map[i][j]);
                    System.out.println();
                }
            }
            System.out.println();
        }
    }

    private static int dfs(int x, int y, int dir, int cnt, int k) {
        if (cnt == k) return 1;
        if (x < 0 || x >= map.length || y < 0 || y >= map.length || map[x][y] != '.') return 0;
        map[x][y] = 'E';
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[(dir + i) % 4];
            int ny = y + dy[(dir + i) % 4];
            if (dfs(nx, ny, (dir + i) % 4, cnt + 1, k) == 1) return 1;
        }
        map[x][y] = '.';
        return 0;
    }
}
