import sys
input = sys.stdin.readline
n, m = map(int,input().split())
a = [[0]*m for i in range(n)]
dp = [[0]*m for i in range(n+1)]
for i in range(n):
    x,y = map(int,input().split())
    x-=1
    a[i][x]=y
for i in range(n-1,-1,-1):
    for j in range(m):
        dp[i][j] = max(dp[i+1][j],dp[i+1][(j+1)%m]+a[i][j])
print(dp[0][0])
