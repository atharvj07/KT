#include<iostream>
#include<queue>
#include<vector>
#include<algorithm>
#define x first
#define y second
using namespace std;
const int N = 205, M = 205;
int n, m, k, sx[N], sy[N], fx[N], fy[N], dx[] = {0, 1, -1, 0}, dy[] = {1, 0, 0, -1}, dist[N][M][4];
bool vis[N][M][4];
struct node{
    int x, y, dir;
    node(int x, int y, int dir){this->x = x, this->y = y, this->dir = dir;}
};
void bfs(int x, int y, int dir){
    queue<node> q;
    q.push(node(x, y, dir));
    vis[x][y][dir] = 1;
    dist[x][y][dir] = 0;
    while(!q.empty()){
        node cur = q.front();
        q.pop();
        for(int i = 1 ; i <= 2 ; i++){
            int nx = cur.x + dx[cur.dir] * i, ny = cur.y + dy[cur.dir] * i;
            if(nx >= 1 and nx <= n and ny >= 1 and ny <= m and !vis[nx][ny][cur.dir]){
                vis[nx][ny][cur.dir] = 1;
                dist[nx][ny][cur.dir] = dist[cur.x][cur.y][cur.dir] + 1;
                q.push(node(nx, ny, cur.dir));
            }
        }
        for(int i = 0 ; i < 4 ; i++){
            if(i != cur.dir){
                if(!vis[x][y][i]){
                    vis[x][y][i] = 1;
                    dist[x][y][i] = dist[cur.x][cur.y][cur.dir] + 1;
                    q.push(node(x, y, i));
                }
                else if(dist[x][y][i] > dist[cur.x][cur.y][cur.dir] + 1){
                    dist[x][y][i] = dist[cur.x][cur.y][cur.dir] + 1;
                    q.push(node(x, y, i));
                }
            }
        }
    }
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cin >> n >> m >> k;
    for(int i = 0 ; i < k ; i++){
        cin >> sx[i] >> sy[i] >> fx[i] >> fy[i];
    }
    bfs(sx[0], sy[0], 3);
    for(int i = 1 ; i < k ; i++){
        if(dist[fx[i]][fy[i]][3] == -1){
            cout << -1 << '\n';
            return 0;
        }
    }
    int ans = 0;
    for(int i = 0 ; i < k ; i++){
        ans += dist[fx[i]][fy[i]][3];
    }
    cout << ans << '\n';
    vector<pair<int, int>> way;
    for(int i = k - 1 ; i >= 0 ; i--){
        int nx = fx[i], ny = fy[i];
        while(nx != sx[i] or ny != sy[i]){
            for(int j = 0 ; j < 4 ; j++){
                if(dist[nx][ny][j] < dist[nx][ny][3]){
                    way.push_back({j, dist[nx][ny][j] - dist[nx][ny][3]});
                    if(j == 0) ny--;
                    else if(j == 1) ny++;
                    else if(j == 2) nx++;
                    else if(j == 3) nx--;
                }
            }
            nx = fx[i], ny = fy[i];
        }
        while(!way.empty()){
            int dir = way.back().first, cnt = way.back().second;
            if(dir == 0) cout << 'L';
            else if(dir == 1) cout << 'R';
            else if(dir == 2) cout << 'D';
            else cout << 'U';
            cnt--;
            way.pop_back();
            while(cnt--){
                cout << dir;
            }
        }
    }
    cout << '\n';
    return 0;
}
