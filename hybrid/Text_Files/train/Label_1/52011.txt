import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        for (int i = 1; i < k; i++) {
            for (int j = 1; j < k; j++) {
                System.out.print(i * j + " ");
            }
            System.out.println();
        }
    }
}
