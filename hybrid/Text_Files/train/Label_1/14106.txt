#include <bits/stdc++.h>
#define ll long long

using namespace std;

const ll mod = 1e9 + 7;

struct DSU {
    ll n;
    vector<ll> p, sz;
    DSU(ll _n) : n(_n) {
        p.resize(n);
        sz.resize(n);
        iota(p.begin(), p.end(), 0);
        fill(sz.begin(), sz.end(), 1);
    }
    ll find(ll x) {
        if (p[x] != x) return p[x] = find(p[x]);
        return x;
    }
    void unite(ll x, ll y) {
        x = find(x);
        y = find(y);
        if (x != y) {
            if (sz[x] < sz[y]) swap(x, y);
            p[y] = x;
            sz[x] += sz[y];
        }
    }
};

int main() {
    ll n, m;
    cin >> n >> m;
    unordered_map<ll, vector<ll>> adj;
    DSU dsu(n);
    for (ll i = 0; i < m; i++) {
        ll a, b;
        cin >> a >> b;
        a--, b--;
        adj[a].push_back(b);
        adj[b].push_back(a);
        dsu.unite(a, b);
    }
    map<ll, vector<pair<ll, ll>>> comps;
    for (ll i = 0; i < n; i++) {
        ll p = dsu.find(i);
        comps[p].push_back({i, -1});
    }
    ll cnt = 0;
    for (auto &[k, v] : comps) {
        cnt += v.size();
        if (cnt > 5000) {
            cout << -1 << endl;
            return 0;
        }
        ll last = 1e9;
        for (ll i = 0; i < v.size(); i++) {
            ll cur = i % 2;
            if (cur == last) cur++;
            last = cur;
            v[i].second = cur;
        }
    }
    for (auto &[k, v] : comps) {
        cout << v.size() << endl;
        for (auto &[x, y] : v) {
            cout << 1 + (y == -1 ? rand() % 1000000000 : y) << ' ' << 1 + x << endl;
        }
    }
    return 0;
}
