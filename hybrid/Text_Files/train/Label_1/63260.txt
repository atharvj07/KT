import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))

d = {}
for i, x in enumerate(a):
    for j in range(i):
        if x + a[j] in d:
            print("YES")
            print(*d[x + a[j]], i+1, j+1)
            exit()
        d[x + a[j]] = [i+1, j+1]
print("NO")
