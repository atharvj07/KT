#include <bits/stdc++.h>
using namespace std;
int main(){
    int n, m, c = 0; cin >> n >> m;
    vector<int>a(n), b(m);
    for(auto &i:a) cin >> i, c += i;
    for(auto &i:b) cin >> i, c += i;
    for(int i = 0; c && i < n; i++){
      for(int j = 0; c && j < m; j++){
        if(a[i] < b[j]) b[j] -= a[i], a[i] = 0, c -= b[j];
        else a[i] -= b[j], b[j] = 0, c -= a[i];
      }
    }
    if(!c) cout << c;
    else cout << -1;
}
