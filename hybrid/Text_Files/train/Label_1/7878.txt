#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> t(n);
  for (int i = 0; i < n; i++) cin >> t[i];
  deque<pair<int, int>> d;
  for (int i = 0; i < n; i++) {
    while (!d.empty() && d.back().first <= t[i]) d.pop_back();
    if (!d.empty()) d.back().second++;
    d.push_back({t[i], 1});
    int sum = 0;
    while (!d.empty() && sum + d.front().first <= m) { sum += d.front().first; d.pop_front(); }
    cout << i - (d.empty() ? i : d.front().second) << " ";
  }
}
