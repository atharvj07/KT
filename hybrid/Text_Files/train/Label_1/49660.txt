import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }
        int ans = 0, mx = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (i % 7 != 0) {
                arr[i] -= 7 - i % 7;
                if (arr[i] < mx) {
                    ans = i + 1;
                    break;
                }
                mx = arr[i];
            }
        }
        input.close();
        System.out.println(ans);
    }
}
