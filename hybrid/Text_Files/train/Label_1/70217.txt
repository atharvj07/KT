n,m,sx,sy=map(int,input().split())
grid=[]
for i in range(n):
    l=[0 for i in range(m)]
    grid.append(l)
grid[sx-1][sy-1]=1
p=[[0,1],[1,0],[-1,0],[0,-1]]
x1,y1=sx-1,sy-1
for k in range(n*m-1):
    for i in range(4):
        x=x1+p[i][0]
        y=y1+p[i][1]
        if 0<=x<n and 0<=y<m and not grid[x][y]:
            grid[x][y]=1
            x1=x
            y1=y
            print(x+1,y+1)
            break
