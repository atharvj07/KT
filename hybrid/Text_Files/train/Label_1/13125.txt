#include <iostream>
#include <vector>

int main() {
    int q; std::cin >> q;
    while (q--) {
        int n; std::cin >> n;
        std::vector<std::vector<int>> grid(3, std::vector<int>(n));
        for (int i = 0; i < 3; ++i) {
            std::string s; std::cin >> s;
            for (int j = 0; j < n; ++j)
                grid[i][j] = s[j] - '0';
        }
        bool ok = true;
        for (int j = 0; j < n; ++j)
            ok &= grid[0][j] == 1 || grid[0][j] == 2 || grid[2][j] == 5 || grid[2][j] == 6;
        for (int i = 1; i < 3; ++i)
            for (int j = 0; j < n; ++j) {
                if (!ok) break;
                int a = grid[i][j], b = grid[3 - i][j];
                ok &= (a == 1 || a == 2 || a == 3 || a == 4) &&
                    (b == 1 || b == 2 || b == 5 || b == 6);
            }
        std::cout << (ok ? "YES\n" : "NO\n");
    }
    return 0;
}
