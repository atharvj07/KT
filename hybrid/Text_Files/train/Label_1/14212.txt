#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 1e5 + 5;
int n, m;
struct node { int s, r; } a[maxn];
int c[maxn];
ll ans;
int main() {
	cin >> n >> m;
	for (int i = 1; i <= n; i++) {
		cin >> a[i].s >> a[i].r;
		c[a[i].s]++;
		ans += a[i].r;
	}
	sort(a + 1, a + n + 1, [&](node x, node y) { return x.r > y.r; });
	int pos = 1, tmp = 0;
	for (int i = 1; i <= m; i++) {
		int now = a[pos].r;
		while (pos <= n && a[pos].r == now) {
			tmp++;
			pos++;
		}
		tmp = min(tmp, c[i]);
		if (tmp == 0) break;
		ans -= 1ll * (tmp - 1) * now;
		tmp = 0;
	}
	if (pos > n) cout << ans << endl;
	else cout << 0 << endl;
	return 0;
}
