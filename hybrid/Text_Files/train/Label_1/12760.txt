#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main() {
    ll n;
    cin >> n;
    vector<int> a(n);
    int zero_idx = -1;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (a[i] == 0) zero_idx = i;
    }
    sort(a.begin(), a.end());
    ll sum = 0;
    for (int i = 0; i < n; i++) sum += a[i];
    if (sum != (n - 1) * n / 2) {
        cout << -1 << endl;
        return 0;
    }
    int min_a = a[0], max_a = a[n - 1];
    int row = min_a, col = max_a;
    int t = a[zero_idx];
    if (t - min_a < max_a - t) {
        row -= t - min_a;
        col -= max_a - t;
    } else {
        row -= max_a - t;
        col -= t - min_a;
    }
    cout << row << " " << col << "\n" << zero_idx / col + 1 << " " << zero_idx % col + 1 << endl;
    return 0;
}
