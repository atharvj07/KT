#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int v;
    cin >> v;

    vector<int> a(9);
    for (int i = 0; i < 9; ++i) {
        cin >> a[i];
    }

    sort(a.begin(), a.end());

    int ans = 0;
    for (int i = 8; i >= 0; --i) {
        if (v < a[i]) {
            break;
        }

        v -= a[i];
        ++ans;
    }

    if (ans == 0) {
        cout << -1 << endl;
    } else {
        string s;
        for (int i = 0; i < ans; ++i) {
            s += '0' + (i + 1);
        }

        reverse(s.begin(), s.end());

        cout << s << endl;
    }

    return 0;
}
