#include <bits/stdc++.h>
using namespace std;

const int maxN = 2e5 + 5;

int n;
long long arr[maxN];

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> arr[i]; }
    sort(arr, arr + n);
    long long mid = arr[(n - 1) / 2];
    long long ans = mid * mid - arr[0] * arr[0];
    long long l = 0, r = n - 1;
    while (l < r) {
        if (arr[l] * arr[l] == mid * mid)
            l++;
        else if (arr[r] * arr[r] == mid * mid)
            r--;
        else {
            long long suml = mid - arr[l];
            long long sumr = arr[r] - mid;
            if (suml <= sumr) {
                l++;
                ans += suml;
            } else {
                r--;
                ans += sumr;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
