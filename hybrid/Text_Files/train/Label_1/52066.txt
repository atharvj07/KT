import java.util.*;
import java.math.BigInteger;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String l = sc.next();
        String r = sc.next();
        BigInteger r2 = new BigInteger(r, 2);
        BigInteger l2 = new BigInteger(l, 2);
        BigInteger ans = r2.xor(r2.shiftRight(1));
        if (l2.equals(BigInteger.ZERO)) {
            ans = ans.xor(new BigInteger("1", 2).shiftLeft(n - 1));
        } else {
            ans = ans.xor(r2.xor(l2.subtract(BigInteger.ONE)).shiftRight(1));
        }
        System.out.println(ans.toString(2));
    }
}
