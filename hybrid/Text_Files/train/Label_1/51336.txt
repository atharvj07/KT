import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int l = sc.nextInt();
        int[] k = new int[n];
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            k[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        Arrays.sort(k);
        Arrays.sort(s);
        boolean flag = true;
        for (int i = 0; i < n; i++) {
            if (k[i] != s[i]) {
                flag = false;
                break;
            }
        }
        System.out.println(flag ? "YES" : "NO");
    }
}
