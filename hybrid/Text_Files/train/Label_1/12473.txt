#include <bits/stdc++.h>
using namespace std;
int main()
{
    string s;
    cin >> s;
    int n = s.size();
    if(s == "0" || s=="1" || s=="2" || s=="4" || s=="6" || s=="8" || s=="10" || s=="12" || s=="14" || s=="16" )
    {
        cout << "NO\n";
    }
    else
    {
        cout << "YES\n";
        s.erase(s.begin() + n - 1);
        cout << s << '\n';
    }
    return 0;
}
