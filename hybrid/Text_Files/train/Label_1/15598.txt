#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
int main() {
  int n;
  cin >> n;
  vector<int> v{1};
  for (int i = 1; i < n; i++) {
    if (v.back() == 1)
      v.push_back(1);
    else {
      v.back()--;
      v.push_back(1);
    }
    while (v.size() > 1 && v.back() == v[v.size() - 2]) {
      v.pop_back();
      v[v.size() - 1]++;
    }
  }
  for (auto i : v) cout << i << " ";
  cout << endl;
  return 0;
}
