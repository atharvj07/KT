import sys
from collections import defaultdict

for t in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    d = defaultdict(list)
    for i in range(n):
        d[a[i]].append(i+1)
    if len(set(a)) == 1:
        print("NO")
    else:
        print("YES")
        for i in range(1,n):
            sys.stdout.write(' '.join(map(str,[d[i][0], d[i+1][0]]))+"\n")
            d[a[d[i][0]-1]].extend(d[a[d[i+1][0]-1]])
            d[a[d[i+1][0]-1]] = []
