import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = sc.nextInt();
        Set<String> bs = new HashSet<>();
        for (int i = 0; i < n; i++) {
            bs.add(sc.nextLine());
        }
        int len = 0, pos = 0, start = 0;
        for (int i = 0, j = 0; i < s.length(); i++) {
            while (j < s.length() && !bs.contains(s.substring(j, i + 1))) {
                j++;
            }
            if (j - i > len) {
                len = j - i;
                pos = i;
            }
            if (j == i) {
                j++;
            }
        }
        System.out.println(len + " " + pos);
    }
}
