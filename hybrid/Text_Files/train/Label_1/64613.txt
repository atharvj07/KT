n = int(input())
dct = {}
for _ in range(n):
    w = input()
    l = len(w)
    if len(set(w)) <= 2:
        if l not in dct: dct[l] = set()
        dct[l].add(w)
l = list(dct.keys())
l.sort(reverse=True)
ans = 0
for x in l:
    if not dct[x]: continue
    t = dct[x].pop()
    ans += len(t)
    if not dct[x]: del dct[x]
print(ans)
