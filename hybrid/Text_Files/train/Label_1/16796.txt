#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 998244353;
const int inf = LLONG_MAX;
int pwr[200050], inv[200050];
int inv2;
int n, k, res = 1;
vector<int> p;
void pre() {
  pwr[0] = 1, inv[0] = 1;
  for (int i = 1; i < 200050; i++) {
    pwr[i] = (pwr[i - 1] * 26) % mod;
    inv[i] = (inv[i - 1] * inv2) % mod;
  }
}
void multiply(int &x, int y) { x = (x * y) % mod; }
int get_ans(int a, int b) {
  if (b < 0) return 1;
  int tmp1 = pwr[a], tmp2 = inv[b];
  multiply(tmp1, tmp2);
  return tmp1;
}
signed main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cin >> n >> k, inv2 = (mod + 1) / 2;
  pre();
  for (int i = 0, a, b; i < n; i++) {
    cin >> a, b = i;
    res = (((res * get_ans(a, n - i - 1)) % mod) * get_ans(b, k - a - 1)) %
           mod;
  }
  cout << res << "\n";
  return 0;
}
