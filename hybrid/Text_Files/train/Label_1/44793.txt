import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] w = new int[2 * n + 1];
        for (int i = 0; i < 2 * n; i++) {
            w[i] = sc.nextInt();
        }
        Arrays.sort(w);
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            ans += w[2 * i + 1] - w[2 * i];
        }
        System.out.println(ans);
    }
}
