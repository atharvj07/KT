import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        int n = scanner.nextInt();
        int[][] stones = new int[k][2];
        int[][] monsters = new int[n][2];
        for (int i = 0; i < k; i++) {
            stones[i][0] = scanner.nextInt();
            stones[i][1] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            monsters[i][0] = scanner.nextInt();
            monsters[i][1] = scanner.nextInt();
        }
        Arrays.sort(stones, (a, b) -> {
            if (a[0] == b[0]) {
                return a[1] - b[1];
            }
            return a[0] - b[0];
        });
        Arrays.sort(monsters, (a, b) -> {
            if (a[0] == b[0]) {
                return a[1] - b[1];
            }
            return a[0] - b[0];
        });
        long answer = 0;
        for (int[] monster : monsters) {
            if (binarySearch(stones, monster[0]) >= 0) {
                int index = binarySearch(stones, monster[0]);
                if (stones[index][1] >= monster[1]) {
                    answer++;
                }
            }
            if (binarySearch(stones, monster[0]) == -1) {
                int index = -binarySearch(stones, monster[0]) - 1;
                if (index < stones.length && stones[index][1] >= monster[1]) {
                    answer++;
                }
            }
        }
        System.out.println(answer);
    }

    public static int binarySearch(int[][] arr, int target) {
        int left = 0;
        int right = arr.length - 1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (arr[mid][0] == target) {
                return mid;
            } else if (arr[mid][0] < target) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        return -left - 1;
    }
}
