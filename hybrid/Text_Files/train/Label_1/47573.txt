import java.util.Scanner;

public class Main {

    private static final int MOD = 1_000_000_007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        int[] dp = new int[n + 1];
        dp[0] = 1;

        int[] broken = new int[m];
        for (int i = 0; i < m; i++) {
            broken[i] = sc.nextInt();
        }

        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < m; j++) {
                if (i == broken[j]) {
                    dp[i] = 0;
                    break;
                }
            }
            if (dp[i] == 0) {
                continue;
            }
            dp[i] += dp[i - 1];
            dp[i] %= MOD;
            if (i - 2 >= 0) {
                dp[i] += dp[i - 2];
                dp[i] %= MOD;
            }
        }

        System.out.println(dp[n]);
    }
}
