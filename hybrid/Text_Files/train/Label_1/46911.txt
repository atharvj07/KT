import java.util.Scanner;

public class LexicographicallySmallestTriple {

    static int[] sequence = {1, 2, 3, 4, 8, 12, 5, 10, 15};

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            System.out.println(findNthElement(n));
        }
    }

    static int findNthElement(int n) {
        if (n <= sequence.length) {
            return sequence[n - 1];
        }

        int lastElement = sequence[sequence.length - 1];
        int diff = n - sequence.length;

        return lastElement + (diff / 3) * 3;
    }
}
