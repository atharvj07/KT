#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int g = a[0];
    for (int i = 1; i < n; i++) {
        g = gcd(g, a[i]);
    }
    if (g == 1) {
        cout << "NO" << endl;
        return 0;
    }
    int p = 1;
    while (p * p <= g) {
        if (g % p == 0) {
            vector<int> b1, b2;
            for (int i = 0; i < n; i++) {
                if (a[i] % p == 0) {
                    b1.push_back(a[i]);
                } else if (a[i] % (g / p) == 0) {
                    b2.push_back(a[i]);
                }
            }
            if (b1.size() && b2.size()) {
                cout << "YES" << endl;
                for (int i = 0; i < n; i++) {
                    if (a[i] % p == 0) {
                        cout << 1 << " ";
                    } else {
                        cout << 2 << " ";
                    }
                }
                return 0;
            }
        }
        p++;
    }
    cout << "NO" << endl;
    return 0;
}
