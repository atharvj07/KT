n, m, d = map(int, input().split())
p = [(c, w) for c, w in sorted([(int(c), int(w)) for c, w in [input().split() for _ in range(n)]], reverse=1)]
i = [(c, w) for c, w in sorted([(int(c), int(w)) for c, w in [input().split() for _ in range(m)]], reverse=1)]
cur_d = res = 0
for c, w in p:
    if cur_d + w <= d:
        cur_d += w
        res += c
    else:
        break
cur_d = 0
for c, w in i:
    if cur_d + w <= d:
        cur_d += w
        res += c
    else:
        break
print(res)
