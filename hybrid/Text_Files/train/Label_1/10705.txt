#include <iostream>
#include <algorithm>
using namespace std;

int n, k;
int ans[11];

int gcd(int a, int b) {
    if (b == 0) return a;
    return gcd(b, a % b);
}

int main() {
    cin >> n >> k;
    if (n % k != 0) {
        cout << -1 << endl;
        return 0;
    }
    n /= k;
    for (int i = k - 1; i >= 0; i--) {
        int x = min(n, i + 1);
        n -= x;
        x = gcd(x, n);
        ans[i] = x;
    }
    for (int i = 0; i < k; i++) {
        cout << ans[i] << ' ';
    }
    cout << endl;
    return 0;
}
