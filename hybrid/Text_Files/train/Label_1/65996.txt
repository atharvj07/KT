def solve(n, k, p, arr):
    even, odd = [], []
    for num in arr:
        if num % 2 == 0:
            even.append(num)
        else:
            odd.append(num)
    
    if len(odd) < k - p or (len(odd) - (k - p)) % 2 != 0 or len(even) + (len(odd) - (k - p)) // 2 < p:
        return "NO"
    
    result = []
    for i in range(p - 1):
        if even:
            result.append([1, even.pop()])
        else:
            result.append([2, odd.pop(), odd.pop()])

    for i in range(k - p - 1):
        result.append([1, odd.pop()])

    last_part = even + odd
    result.append([len(last_part)] + last_part)
    
    output = "YES\n"
    for part in result:
        output += " ".join(map(str, part)) + "\n"
    return output.strip()

n, k, p = map(int, input().split())
arr = list(map(int, input().split()))
print(solve(n, k, p, arr))
