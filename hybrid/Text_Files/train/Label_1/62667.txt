n = int(input())
a = list(map(int, input().split()))

MOD = 998244353
dp = [[[0 for j in range(2)] for k in range(2)] for i in range(n + 1)]
dp[n][1][0] = dp[n][1][1] = 1
for i in range(n - 1, -1, -1):
    for lng in range(2):
        for prv in range(2):
            dp[i][lng][prv] = dp[i + 1][lng][prv]
            if a[i] == 0 or (prv == 1 and a[i] == 1):
                dp[i][lng][prv] = (dp[i][lng][prv] + dp[i + 1][lng + 1][1]) % MOD
            if lng == 1 and (a[i] == -1 or (prv == 1 and a[i] == 0)):
                dp[i][lng][prv] = (dp[i][lng][prv] + dp[i + 1][0][0]) % MOD
print(dp[0][0][0])
