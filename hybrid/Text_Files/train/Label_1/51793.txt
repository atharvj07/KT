import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.nextLine();
        int n = s.length();
        int[] stack = new int[n];
        int top = 0;
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '(') {
                stack[top++] = i;
            } else {
                if (top > 0) {
                    top--;
                    if (stack[top] == i - 1) { // consecutive
                        sb.append("()");
                    }
                } else {
                    sb.append('(');
                    stack[top++] = i;
                }
            }
        }
        while (top > 0) {
            sb.append(')');
            top--;
        }
        System.out.println(sb);
    }
}
