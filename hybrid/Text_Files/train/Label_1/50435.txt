import java.util.*;
import java.lang.*;

public class Main {
  static final int inf = (int) 1e9;

  public static void main(String[] args) {
    Scanner stdin = new Scanner(System.in);
    int n = stdin.nextInt();
    int m = stdin.nextInt();
    int ans = dfs(0, n, m);
    System.out.println(ans >= inf ? -1 : ans);
  }

  static int dfs(int cur, int n, int m) {
    if (cur == n) return 0;
    if (cur > n) return inf;
    return Math.min(dfs(cur + 1, n, m), dfs(cur + 2, n, m)) + 1;
  }
}
