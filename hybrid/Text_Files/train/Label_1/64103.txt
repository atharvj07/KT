import bisect
for _ in range(int(input())):
    arr = []
    suc = 0
    for _ in range(int(input())):
        t, x = map(int, input().split())
        i = bisect.bisect_right(arr, [t, -1])
        if i == len(arr) or arr[i][0] != t:
            arr.insert(i, [t, x])
        if arr[i][1] == x:
            suc += 1
    print(suc)
