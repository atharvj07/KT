def find_swapped_keys(s, t):
    # Dictionary to keep track of the swaps
    swaps = {}
    
    # Iterate through both strings simultaneously
    for i in range(len(s)):
        # If the characters are the same, continue
        if s[i] == t[i]:
            continue
        # If we already encountered one of the characters, verify it's a valid swap
        elif s[i] in swaps:
            if swaps[s[i]] != t[i]:
                return -1
        elif t[i] in swaps:
            if swaps[t[i]] != s[i]:
                return -1
        else:
            # Otherwise, add the swap to our dictionary
            swaps[s[i]] = t[i]
            swaps[t[i]] = s[i]
            
    # Filter out duplicate swaps
    filtered_swaps = list(set([(min(k, v), max(k, v)) for k, v in swaps.items()]))
    
    # If there are any one-sided swaps, it's invalid
    if any(swaps[k] != k for k in swaps.keys()):
        # Print the number of swaps and each swap
        print(len(filtered_swaps))
        for swap in sorted(filtered_swaps):
            print(swap[0], swap[1])
    else:
        print(-1)

# Reading input
s = input().strip()
t = input().strip()
find_swapped_keys(s, t)
