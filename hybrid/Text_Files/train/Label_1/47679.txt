import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int m = input.nextInt();
    int k = input.nextInt();
    int[][] dp = new int[n][m];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        dp[i][j] = input.nextInt();
      }
    }
    int max = 0;
    int a = 0;
    int b = 0;
    for (int i = k - 1; i < n - k + 1; i++) {
      for (int j = k - 1; j < m - k + 1; j++) {
        int sum = 0;
        for (int x = i - k + 1; x <= i; x++) {
          for (int y = j - k + 1; y <= j; y++) {
            sum += dp[x][y];
          }
        }
        if (sum > max) {
          max = sum;
          a = i + 1;
          b = j + 1;
        }
      }
    }
    System.out.println(a + " " + b);
  }
}
