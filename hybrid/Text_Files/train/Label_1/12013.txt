#include<bits/stdc++.h>
using namespace std;
#define ll long long
int n;
ll a[100005];
int main()
{
    cin>>n;
    for(int i=0;i<n;++i)cin>>a[i];
    int ans=1,be=0,en=0,f=0,g=0;
    for(int i=1;i<n;++i)
    {
        if(a[i-1]>=a[i])
        {
            if(f==0)
            {
                f=1;
                be=i-1;
                g=1;
            }
            else if(g==0)
            {
                g=1;
                be=i-1;
            }
            en=i;
        }
        else
        {
            ans=max(ans,en-be+1);
            f=0;
            g=0;
        }
    }
    ans=max(ans,en-be+1);
    cout<<ans<<endl;
    return 0;
}
