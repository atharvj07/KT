import java.util.*;
import java.io.*;

public class BerlandRoads {

    static class Edge {
        int u, v;

        Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static int n, k;
    static List<Edge>[] graph;
    static int[] pathCount;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        graph = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            String[] line = br.readLine().split(" ");
            int u = Integer.parseInt(line[0]) - 1;
            int v = Integer.parseInt(line[1]) - 1;
            graph[u].add(new Edge(u, v));
            graph[v].add(new Edge(v, u));
        }
        k = Integer.parseInt(br.readLine());
        pathCount = new int[n - 1];
        for (int i = 0; i < k; i++) {
            String[] line = br.readLine().split(" ");
            int start = Integer.parseInt(line[0]) - 1;
            int end = Integer.parseInt(line[1]) - 1;
            dfs(start, end, -1);
        }
        for (int i = 0; i < n - 1; i++) {
            System.out.print(pathCount[i] + " ");
        }
    }

    static void dfs(int u, int v, int parent) {
        for (Edge edge : graph[u]) {
            int next = edge.v;
            if (next != parent) {
                if (next == v) {
                    pathCount[edge.u < edge.v ? edge.u : edge.v]++;
                } else {
                    dfs(next, v, u);
                }
            }
        }
    }
}
