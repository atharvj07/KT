import sys

n = int(input())
adj = [[] for i in range(n+1)]
for i in range(2, n+1):
    adj[int(input())].append(i)

visited = [False] * (n+1)
sub = [0] * (n+1)
dp = [0] * (n+1)

def dfs(u, cnt):
    visited[u] = True
    sub[u] = 1
    dp[u] = cnt
    for v in adj[u]:
        if not visited[v]:
            dfs(v, cnt+1)
            sub[u] += sub[v]
            dp[u] += dp[v] + sub[v]

dfs(1, 0)
ans = [0] * (n+1)
for i in range(n, 0, -1):
    ans[i] = dp[1] - dp[i] + 1
    for v in adj[i]:
        ans[i] = min(ans[i], ans[v])
for i in range(1, n+1):
    print(ans[i], end=" ")
