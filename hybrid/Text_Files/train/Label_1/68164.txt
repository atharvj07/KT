import sys

n = int(input())
s, t = input(), input()
chars = ['a', 'b', 'c']
for i in range(3):
    for j in range(3):
        for k in range(3):
            if i == j or j == k or i == k or chars[i] + chars[j] == s or chars[j] + chars[k] == s or chars[i] + chars[k] == s or chars[i] + chars[j] == t or chars[j] + chars[k] == t or chars[i] + chars[k] == t:
                continue
            print("YES")
            print(chars[i] * n + chars[j] * n + chars[k] * n)
            sys.exit()
print("NO")
