#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
char s[maxn];
int main() {
    int n;
    cin >> n;
    cin >> s + 1;
    vector<int> nd(n + 1);
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (s[i] == '(')
            nd[i] = nd[i - 1] + 1;
        else
            nd[i] = nd[i - 1] - 1;
        ans = max(ans, nd[i]);
    }
    string R = "", B = "";
    int bal = 0;
    for (int i = 1; i <= n; i++) {
        if (nd[i] != ans)
            B += s[i];
        else {
            if (s[i] == '(')
                bal++;
            else
                bal--;
            if (bal)
                B += s[i];
            else {
                R += s[i];
            }
        }
    }
    string t = "";
    bal = 0;
    for (int i = 0; i < n; i++) {
        if (R[i] == '(')
            bal++;
        else
            bal--;
        if (bal)
            t += "0";
        else
            t += "1";
    }
    cout << t << endl;
    return 0;
}
