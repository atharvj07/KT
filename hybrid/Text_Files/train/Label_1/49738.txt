import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        char[] ch = s.toCharArray();
        int n = s.length();
        int count = 1;
        char[] unique = new char[26];
        unique[0] = ch[0];
        for (int i = 1; i < n; i++) {
            boolean flag = false;
            for (int j = 0; j < count; j++) {
                if (ch[i] == unique[j]) {
                    flag = true;
                    break;
                }
            }
            if (!flag) {
                unique[count++] = ch[i];
            }
        }
        if (count <= 26) System.out.println("YES");
        else System.out.println("NO");
    }
}
