#include <bits/stdc++.h>
using namespace std;
#define int long long

int n, cnt = 0, ans;
string s;
vector<int> v;

int32_t main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> s;
        int bal = 0;
        for (char j : s) {
            bal += (j == '(');
            bal -= (j == ')');
            cnt += (bal == 0);
        }
        v.push_back(bal);
    }
    for (int i = 0; i < n; i++) {
        int bal = v[i];
        for (int j = i; j < n; j++) {
            bal += v[j];
            ans += (bal == 0);
        }
    }
    cout << ans + cnt;
}
