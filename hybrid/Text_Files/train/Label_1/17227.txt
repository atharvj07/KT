#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<string> s(n - k + 1);
    for (int i = 0; i < n - k + 1; i++) cin >> s[i];
    string pre;
    vector<string> ans(n);
    for (int i = 0; i < n - k + 1; i++) {
        if (s[i] == "YES") {
            if (pre == "")
                pre = "A";
            ans[i + k - 1] = pre;
        } else {
            if (pre == "")
                pre = "A";
            else
                pre[0]++;
        }
    }
    for (int i = 0; i < k - 1; i++) {
        if (ans[i] == "") {
            ans[i] = "A";
            for (int j = i + 1; j < k; j++) {
                if (ans[j] == ans[i]) ans[j][0]++;
            }
        }
    }
    for (int i = k; i < n; i++) {
        if (ans[i] == "") {
            ans[i] = "A";
            for (int j = i + 1; j < n; j++) {
                if (ans[j] == ans[i]) ans[j][0]++;
            }
        }
    }
    for (int i = 0; i < n; i++) cout << ans[i] << " ";
    cout << endl;
}
