#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<vector<int>> dp1(n, vector<int>(2, 0));
  vector<vector<int>> dp2(n, vector<int>(2, 0));
  dp1[0][0] = 1;
  dp1[0][1] = 1;
  for (int i = 1; i < n; i++) {
    if (s[i] == '1') {
      dp1[i][0] = dp1[i - 1][0] + 1;
      dp1[i][1] = max(dp1[i - 1][0], dp1[i - 1][1]);
    } else {
      dp1[i][0] = max(dp1[i - 1][0], dp1[i - 1][1]);
      dp1[i][1] = dp1[i - 1][1] + 1;
    }
  }
  dp2[n - 1][0] = 1;
  dp2[n - 1][1] = 1;
  for (int i = n - 2; i >= 0; i--) {
    if (s[i] == '1') {
      dp2[i][0] = max(dp2[i + 1][0], dp2[i + 1][1]) + 1;
      dp2[i][1] = dp2[i + 1][1];
    } else {
      dp2[i][0] = dp2[i + 1][0];
      dp2[i][1] = max(dp2[i + 1][0], dp2[i + 1][1]) + 1;
    }
  }
  string ans = "";
  int i = 0, j = 0;
  while (i < n && j < n) {
    if (s[i] == '1') {
      if (dp1[i][0] == dp2[i][0]) {
        ans += '0';
        i++;
      } else {
        ans += '1';
        j++;
      }
    } else {
      if (dp1[i][1] == dp2[i][1]) {
        ans += '0';
        i++;
      } else {
        ans += '1';
        j++;
      }
    }
  }
  cout << ans << endl;
}
