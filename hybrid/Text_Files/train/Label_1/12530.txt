#include<bits/stdc++.h>
#include<algorithm>
using namespace std;
int main(){
    int n,a[1005];
    cin>>n;
    if(n==1){
        cout<<"0\n";
        return 0;
    }
    a[1]=2,a[2]=3,a[3]=4;
    if(n==2){
        a[3]=5;
    }
    if(n==3){
        a[3]=6;
    }
    if(n==4){
        a[3]=5;
        a[4]=6;
    }
    int l=1;
    while(a[l]!=n){
        l++;
    }
    cout<<l<<endl;
    for(int i=1;i<=l;i++){
        cout<<a[i]<<" ";
    }
    return 0;
}
