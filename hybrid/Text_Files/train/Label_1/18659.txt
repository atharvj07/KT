#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    int sum_a = 0, sum_b = 0;
    vector<int> delta(n);
    for (int i = 0; i < n; i++) {
      delta[i] = b[i] - a[i];
      sum_a += a[i];
      sum_b += b[i];
    }
    bool flag = true;
    if (sum_a != sum_b) flag = false;
    int csum = 0;
    for (int i = 0; i < n && flag; i++) {
      csum += delta[i];
      if (csum < 0) flag = false;
    }
    cout << (flag ? "YES" : "NO") << endl;
  }
  return 0;
}
