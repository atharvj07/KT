import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        int[] countA = new int[6];
        int[] countB = new int[6];
        for (int i = 0; i < n; i++) {
            countA[a[i]]++;
            countB[b[i]]++;
        }
        int ans = 0;
        for (int i = 1; i <= 5; i++) {
            ans += Math.abs(countA[i] - countB[i]) / 2;
        }
        if (ans % 2 == 0) {
            System.out.println(ans / 2);
        } else {
            System.out.println(-1);
        }
    }
}
