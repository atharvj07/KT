#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;

ll power(ll base, ll e) {
    ll res = 1;
    while (e > 0) {
        if (e % 2 == 1) {
            res *= base;
            res %= mod;
        }
        base *= base;
        base %= mod;
        e /= 2;
    }
    return res;
}

int main() {
    ll n, t;
    cin >> n >> t;
    vector<ll> time(n);
    for (int i = 0; i < n; i++) { cin >> time[i]; }
    ll sum = 0;
    ll p = 1;
    for (int i = 0; i < n; i++) {
        sum += time[i];
        if (sum > t) { break; }
        p = (p * 2) % mod;
    }
    if (sum <= t) { p = (p + mod - 1) % mod; }
    ll q = power(2, n);
    cout << (p * power(q, mod - 2) % mod) << '\n';
    return 0;
}
