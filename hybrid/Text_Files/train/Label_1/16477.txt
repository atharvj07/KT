#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::string s;
  std::cin >> s;
  std::vector<char> t;
  for (char c : s) {
    if (!t.empty() && t.back() == c && c == 'U') continue;
    if (!t.empty() && t.back() == 'R' && c == 'U') t.pop_back();
    else t.push_back(c);
  }
  std::cout << t.size() << std::endl;
}
