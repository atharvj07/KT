#include <bits/stdc++.h>
using namespace std;
#define int long long
const int maxn = 2e5 + 5;
vector<int> g[maxn];
bool vis[maxn];
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        for (int i = 1; i <= k; i++) {
            int x;
            cin >> x;
            g[x].push_back(i);
        }
        int ans = 0;
        queue<int> q;
        for (int i = 1; i <= n; i++) {
            if (!vis[i]) {
                bool have = 0;
                for (auto& x : g[i]) have |= (x > ans);
                if (have) {
                    q.push(i);
                    vis[i] = 1;
                }
            }
        }
        while (!q.empty()) {
            ans++;
            int sz = q.size();
            while (sz--) {
                int node = q.front();
                q.pop();
                for (auto& x : g[node]) {
                    if (!vis[x - ans] && x - ans > 0) {
                        q.push(x - ans);
                        vis[x - ans] = 1;
                    }
                    if (!vis[x + ans] && x + ans <= n) {
                        q.push(x + ans);
                        vis[x + ans] = 1;
                    }
                }
            }
        }
        cout << ans << endl;
        for (int i = 1; i <= n; i++) {
            vis[i] = 0, g[i].clear();
        }
    }
}
