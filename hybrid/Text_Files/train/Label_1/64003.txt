from sys import stdin

h, w = list(map(int, stdin.readline().split()))
g = [list(stdin.readline()) for _ in range(h)]
q = int(stdin.readline())
for i in range(h):
    for j in range(w):
        if g[i][j] == '.':
            g[i][j] = [0, 0]
        else:
            g[i][j] = [-1, -1]

for i in range(h):
    for j in range(1, w):
        if g[i][j][0] != -1 and g[i][j - 1][0] != -1:
            g[i][j][0] = g[i][j - 1][0] + 1
            g[i][j - 1][0] = g[i][j][0]

for i in range(h):
    for j in range(w - 1):
        if not (g[i][j][1] == -1 or g[i][j + 1][1] == -1):
            g[i][j][1] = g[i][j + 1][1] + 1
            g[i][j + 1][1] = g[i][j][1]

for _ in range(q):
    r1, c1, r2, c2 = list(map(int, stdin.readline().split()))
    r1 -= 1
    c1 -= 1
    r2 -= 1
    c2 -= 1
    res = g[r2][c2][1]
    if c1 > 0:
        res -= g[r2][c1 - 1][1]
    if g[r1][c1][0] > 0:
        res += g[r1][c1][0]
    if r1 > 0 and g[r1 - 1][c1][0] > 0:
        res -= g[r1 - 1][c1][0]
    print(res)
