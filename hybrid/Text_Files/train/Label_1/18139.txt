#include<bits/stdc++.h>
using namespace std;
#define vi(x) vector<x>
#define pb push_back
const int mod = 1e9 + 7;
const int N = 2e5 + 5;

int n, k, ans;
vi(int) a(N), b;

int main() {
	cin >> n >> k;
	for (int i = 0; i < n; i++)cin >> a[i];
	if (k) {
		int x;
		for (int i = 0; i < k; i++)cin >> x, b.pb(x);
		sort(b.begin(), b.end());
		for (int i = 0; i < b.size(); i++) {
			if (b[i] <= i + 1) {
				b.clear();
				break;
			}
		}
	}
	if (b.size() == 0) {
		int cnt = 0;
		for (int i = 0; i < n - 1; i++)if (a[i] >= a[i + 1])cnt++;
		cout << cnt << '\n';
	}
	else cout << -1 << '\n';
	return 0;
}
