#include<bits/stdc++.h>
using namespace std;
const int N = 55;
struct P{
    int x, y;
    P(int _x = 0, int _y = 0) : x(_x), y(_y) {}
    bool operator < (const P &rhs) const {
        if(x != rhs.x) return x < rhs.x;
        return y < rhs.y;
    }
};
map<P, int> mp;
int main(){
    int n; cin >> n;
    vector<P> p(n);
    for(int i = 0; i < n; ++i) cin >> p[i].x >> p[i].y, mp[p[i]] = 1;
    sort(p.begin(), p.end());
    int ans = 0;
    for(int i = 0; i < n; ++i){
        for(int j = i + 1; j < n; ++j){
            int dx = p[i].x - p[j].x, dy = p[i].y - p[j].y;
            if(dx == 0){
                if(mp.count(P(p[i].x, p[i].y + dy)) and mp.count(P(p[j].x, p[j].y + dy))){
                    ans += 2;
                }
                continue;
            }
            if(dy == 0){
                if(mp.count(P(p[i].x + dx, p[i].y)) and mp.count(P(p[j].x + dx, p[j].y))){
                    ans += 2;
                }
                continue;
            }
            int k = p[i].y - p[j].y, b = p[i].x - k * p[i].y / dy;
            if(mp.count(P(k, b + dx)) and mp.count(P(k, b))){
                ans += 2;
            }
        }
    }
    cout << ans << endl;
}
