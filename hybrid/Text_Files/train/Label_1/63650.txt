from sys import stdin
n, m = map(int, stdin.readline().split())
a = list(map(int, stdin.readline().split()))
b = [0] * (n + 2)
for x in a: b[x] = 1
ans = 1
flag = 1
for i in range(1, n + 1):
  if not b[i]:
    if not b[i - 1] and not b[i + 1]:
      flag = 0
      break
    if b[i - 1]:
      b[i + 1] = 1
    else: b[i - 1] = 1
print(flag * (2 ** (m - 1)) % (10 ** 9 + 7))
