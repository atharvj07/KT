#include <iostream>
#include <vector>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n, k;
        std::cin >> n >> k;
        if (k > (n - 1) / 2 || k == n || (n == 1 && k == 1)) {
            std::cout << "-1" << std::endl;
            continue;
        }
        std::vector<int> a(n);
        if (k == (n - 1) / 2) {
            for (int i = 0; i < n; i += 2) {
                a[i] = i + 1;
                a[i + 1] = n - i;
            }
        } else {
            a[0] = 1;
            for (int i = 1; i < n - 1; i += 2) {
                a[i] = n - i;
                a[i + 1] = i + 1;
            }
            if (k % 2 == 0) a[n - 1] = n - k / 2 - 1;
            else a[n - 1] = n - k / 2;
        }
        for (auto &x : a) std::cout << x << " ";
        std::cout << std::endl;
    }
}
