import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        int s = sc.nextInt();
        if (d > 9 * s || (d != 9 || s != 1)) {
            System.out.println(-1);
            return;
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 9; i >= 0; i--) {
            int cnt = Math.min(s, (d - i) / 9);
            for (int j = 0; j < cnt; j++) {
                sb.append(i);
            }
            s -= cnt;
        }
        System.out.println(sb.toString());
    }
}
