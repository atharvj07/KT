#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,p,k;
    cin>>n>>p>>k;
    for(int i=p-k;i<=p+k;i++){
        if(i<=0||i>n) continue;
        if(i==p) cout<<"("<<i<<") ";
        else cout<<i<<" ";
    }
    if(p-k>1) cout<<"<< ";
    if(p+k<n) cout<<" >>";
    cout<<endl;
    return 0;
}
