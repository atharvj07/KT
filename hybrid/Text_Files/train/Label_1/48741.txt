import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int ans = 0;
        for (int d = -1000; d <= 1000; d++) {
            if (d == 0) continue;
            int cnt = 0;
            for (int x : arr) {
                cnt += (x / d > 0) ? 1 : 0;
            }
            if (cnt >= (n + 1) / 2) {
                ans = d;
                break;
            }
        }
        System.out.println(ans);
    }
}
