#include <bits/stdc++.h>
using namespace std;
const int MOD = 998244353;
#define ll long long
ll pw(ll a, ll b) {
  ll ans = 1;
  while (b) {
    if (b & 1) { ans = ans * a % MOD; }
    a = a * a % MOD;
    b >>= 1;
  }
  return ans;
}
const int MX = 1000030;
ll fact[MX];
ll invfact[MX];
ll inv[MX];
void init() {
  fact[0] = 1;
  for (ll i = 1; i < MX; i++) { fact[i] = fact[i - 1] * i % MOD; }
  inv[1] = 1;
  for (ll i = 2; i < MX; i++) { inv[i] = (MOD - (MOD / i)) * inv[MOD % i] % MOD; }
  invfact[0] = 1;
  for (ll i = 1; i < MX; i++) { invfact[i] = invfact[i - 1] * inv[i] % MOD; }
}
ll C(ll n, ll k) { return fact[n] * invfact[n - k] % MOD * invfact[k] % MOD; }
int main() {
  init();
  ll n;
  cin >> n;
  ll ans = (fact[n] * C(n * n, n - 1)) % MOD;
  cout << ans << "\n";
}
