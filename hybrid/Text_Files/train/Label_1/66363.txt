import sys
input = sys.stdin.readline

n, m = map(int, input().split())
t = ["0"] + [t for l, t in zip(*[iter(input().split())]*2)]
s = ["0"] + [s for l, s in zip(*[iter(input().split())]*2)]
i, j, ans = 0, 1, 0
while i + len(s) <= len(t):
    if s[j] == t[i]:
        if j == len(s) - 1:
            ans += 1
            i -= len(s) - 1
        j += 1
        i += 1
    else:
        i += 1
        j = 1
print(ans)
