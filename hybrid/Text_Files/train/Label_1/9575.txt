#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define nl "\n"

int n;
ll ans[3][2],x[100005];
ll a[100005],tot=0;

int main() {
    cin>>n;
    int neg=0;
    for(int i=1; i<=n; i++){
        cin>>a[i];
        tot+=a[i];
        if(a[i]<0)neg=1;
    }
    if(!neg){
    ans[1][0]=1,ans[1][1]=n;
    ans[2][0]=1,ans[2][1]=n;
    ans[3][0]=1,ans[3][1]=n;
    if(tot%n!=0){
        cout<<-1<<nl;
        return 0;
    }
    ll val=tot/n;
    for(int i=1; i<=n; i++)
        x[i]=x[i-1]+(a[i]-val);
    ll mn=0;
    for(int i=1; i<=n; i++){
        if(x[i]<mn){
            ans[1][0]=i,ans[1][1]=i;
            mn=x[i];
        }
    }
    ll mx=0;
    for(int i=n; i>=1; i--){
        if(x[i]>mx){
            ans[2][0]=i,ans[2][1]=i;
            mx=x[i];
        }
    }
    if(mx-mn>2*val){
        cout<<-1<<nl;
        return 0;
    }
    ll dif=val-(mx-mn);
    if(dif>=0){
        ans[3][0]=ans[1][0],ans[3][1]=ans[2][0];
    }
    else{
        ans[3][0]=ans[2][1]+1,ans[3][1]=ans[2][1]+(dif/3)+1;
    }
    for(int i=1; i<=3; i++){
        cout<<ans[i][0]<<" "<<ans[i][1]<<nl;
        ll px=x[ans[i][0]-1];
        for(int j=ans[i][0]; j<=ans[i][1]; j++){
            cout<<(a[j]-val-px)/((ans[i][1]-ans[i][0]+1))<<" ";
            px=a[j]-val;
        }
        cout<<nl;
    }
    }
    else{
        cout<<-1<<nl;
    }
    return 0;
}
