n, m, k = map(int, input().split())
park = [input() for _ in range(n)]

result = [0] * m

for row in range(1, n):
    for col in range(m):
        if park[row][col] == 'U' and row % 2 == 1:
            result[col] += 1
        elif park[row][col] == 'L' and col - row >= 0:
            result[col - row] += 1
        elif park[row][col] == 'R' and col + row < m:
            result[col + row] += 1

print(' '.join(map(str, result)))
