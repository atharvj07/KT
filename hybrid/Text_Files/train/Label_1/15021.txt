#include <bits/stdc++.h>
using namespace std;
int main() {
  int n; cin >> n;
  int a[n], b[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  map<int, int> cnta, cntb;
  for (int i = 0; i < n; i++) cnta[a[i]]++, cntb[b[i]]++;
  int cnt = 0;
  for (int i = 1; i <= 5; i++) cnt += abs(cnta[i] - cntb[i]) / 2;
  if (cnt % 2) cout << -1 << endl;
  else cout << cnt / 2 << endl;
  return 0;
}
