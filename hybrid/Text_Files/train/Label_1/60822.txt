n, t = map(int, input().split())
s1 = input()
s2 = input()
if s1 == s2:
    if t == 0:
        print(s1)
    else:
        print(-1)
else:
    diff = []
    for i in range(n):
        if s1[i] != s2[i]:
            diff.append(i)
    if len(diff) != t:
        print(-1)
    else:
        s3 = ''
        for i in range(n):
            if i in diff:
                if s1[i] == s2[i]:
                    s3 += 'a'
                else:
                    s3 += 'z'
            else:
                s3 += s1[i]
        print(s3)
