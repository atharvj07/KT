#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, s, k;
    cin >> n >> s >> k;
    vector<bool> closed(n + 1);
    for (int i = 0; i < k; i++) {
      long long a;
      cin >> a;
      closed[a] = true;
    }
    long long ans = n - s;
    for (long long i = s + 1; i <= n; i++) {
      if (!closed[i]) {
        ans = min(ans, i - s - 1);
      }
    }
    for (long long i = s - 1; i >= 1; i--) {
      if (!closed[i]) {
        ans = min(ans, s - i - 1);
      }
    }
    cout << ans << "\n";
  }
}
