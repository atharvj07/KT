#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, c1, c2;
  cin >> n >> c1 >> c2;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int j = i + 1;
    while (j < n && a[j] == 0) { j++; }
    if (j < n) {
      ans += (c1 + c2 * (j - i) * (j - i));
      i = j;
    }
  }
  cout << ans << endl;
  return 0;
}
