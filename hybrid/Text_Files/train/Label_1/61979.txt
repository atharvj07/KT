from sys import stdin

n = int(stdin.readline())
a = []

for _ in range(n):
    a.append(stdin.readline().strip())

a = [i[::-1] for i in a][::-1]

n = len(a[0])

for d in range(n, 0, -1):
    if n % d == 0 and all(a[i][j] == a[i % d][j % d] for i in range(n) for j in range(n)):
        print(d)
        break
