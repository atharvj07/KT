n, a, b = map(int, input().split())
s = []
for i in range(n):
    x, y = map(int, input().split())
    s.append((x, y))
    s.append((y, x))
s.sort(reverse=True)
ans = 0
for i in range(n):
    for j in range(i+1, n):
        xi, yi = s[i]
        xj, yj = s[j]
        if xi <= a and xj <= a and yi + yj <= b:
            ans = max(ans, xi * yi + xj * yj)
print(ans)
