n = int(input())
a = [int(input()) for _ in range(n)]
d = {}
for x in a:
  d[x] = d.get(x, 0) + 1
cnt = 0
for x in d:
  if d[x] % 2:
    cnt += 1
if cnt == 1 or cnt == 0 and n % 4 == 0:
  print("YES")
  for x in d:
    if d[x] % 2:
      print(x, sum(a) - x)
      break
    elif d[x] >= n / 2:
      print(x, sum(a) - x)
      break
else:
  print("NO")
