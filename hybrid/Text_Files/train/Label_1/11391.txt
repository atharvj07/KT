#include <iostream>
#include <vector>

using namespace std;

bool can_sort(vector<int>& v) {
  for (int i = 0; i < v.size() - 1; i++) {
    if (v[i] == v[i + 1]) {
      return false;
    }
  }
  return true;
}

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  if (can_sort(v)) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
