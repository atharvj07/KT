#include <iostream>
#include <vector>
using namespace std;

long long gcd(long long a, long long b) {
    while (b > 0) {
        a %= b;
        swap(a, b);
    }
    return a;
}

long long f(long long a, long long b) {
    if (b == 0) {
        return 0;
    }
    return 1 + f(a, b - gcd(a, b));
}

int main() {
    long long x, y;
    cin >> x >> y;
    cout << f(x, y) << endl;
    return 0;
}
