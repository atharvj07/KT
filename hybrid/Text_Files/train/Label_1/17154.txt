#include <bits/stdc++.h>
#define int long long
using namespace std;

const int maxn = 1e5 + 5;
int n, a[maxn], dist[maxn], ans[maxn];
vector<int> adj[maxn];
set<int> dom[maxn];

void dfs(int v, int p) {
  ans[v] = a[v];
  for (int u : adj[v]) {
    if (u == p) continue;
    dfs(u, v);
    set<int>& cur = dom[v];
    set<int>& pre = dom[u];
    if (pre.size() > cur.size()) swap(pre, cur);
    for (int col : pre) {
      if (cur.find(col) != cur.end()) {
        ans[v] += col;
        continue;
      }
      if (dist[col] < dist[v]) cur.insert(col);
    }
    for (int col : cur) dist[col] = v;
  }
  dom[v].insert(a[v]); dist[a[v]] = v;
}

int32_t main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) { cin >> a[i]; }
  for (int i = 2; i <= n; ++i) {
    int x, y;
    cin >> x >> y;
    adj[x].push_back(y);
    adj[y].push_back(x);
  }
  dfs(1, 1);
  for (int i = 1; i <= n; ++i) cout << ans[i] << " ";
  cout << endl;
}
