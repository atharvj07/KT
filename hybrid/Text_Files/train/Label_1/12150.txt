#include <iostream>
#include <vector>
using namespace std;
vector<int> a;
int main() {
  int n;
  cin >> n;
  a.resize(n + 1);
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
  }
  int sum = 0;
  for (int i = 1; i <= n; ++i) sum += a[i];
  sum /= 2;
  vector<vector<pair<int, int>>> opt(n + 1, vector<pair<int, int>>(sum + 1));
  for (int i = 1; i <= n; ++i) {
    for (int j = 0; j <= sum; ++j) {
      if (j >= a[i])
        opt[i][j] = max(opt[i][j], opt[i - 1][j - a[i]]);
      if (j + a[i] <= sum) {
        opt[i][j] = max(opt[i][j], opt[i - 1][j + a[i]]);
        if (j + a[i] == sum) opt[i][j].second = i;
      }
      opt[i][j] = max(opt[i][j], make_pair(opt[i - 1][j].first, 0));
    }
  }
  vector<int> ans;
  int i = n, j = sum;
  while (i > 0) {
    if (opt[i][j].second) {
      ans.push_back(opt[i][j].second);
      j -= a[opt[i][j].second];
    }
    i--;
  }
  cout << ans.size() << endl;
  for (int i = 0; i < ans.size(); ++i) cout << ans[i] << " ";
  return 0;
}
