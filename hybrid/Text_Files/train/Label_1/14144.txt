#include <iostream>
#include <vector>

void go(std::string s, std::vector<std::vector<int>>& v) {
  int n = s.length();
  std::vector<int> good;
  std::vector<int> bad;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0')
      bad.push_back(i);
    else
      good.push_back(i);
  }
  for (int i = 0; i < bad.size(); i++) {
    int j = i + 1;
    int l = 0;
    std::vector<int> indices;
    indices.push_back(bad[i]);
    while (j < bad.size() && j - i <= good.size()) {
      if (l % 2 == 0 && s[bad[j]] == '0') {
        indices.push_back(bad[j]);
        j++;
        l++;
      } else if (l % 2 == 1 && s[good[l]] == '1') {
        indices.push_back(good[l]);
        j++;
        l++;
      } else
        break;
    }
    if (l % 2 == 1 && j - i == good.size() + 1) v.push_back(indices);
  }
}

int main() {
  std::string s;
  std::cin >> s;
  std::vector<std::vector<int>> v;
  go(s, v);
  if (v.empty()) {
    std::cout << -1 << std::endl;
  } else {
    std::cout << v.size() << std::endl;
    for (auto indices : v) {
      std::cout << indices.size() << ' ';
      for (int i : indices) std::cout << i + 1 << ' ';
      std::cout << '\n';
    }
  }
  return 0;
}
