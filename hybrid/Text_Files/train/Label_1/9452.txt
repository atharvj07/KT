#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<char>> v(n, vector<char>(m));
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            cin >> v[i][j];
        }
    }
    int leftmost = m;
    int rightmost = -1;
    int top = n;
    int bottom = -1;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (v[i][j] == 'X') {
                if (j < leftmost) leftmost = j;
                if (j > rightmost) rightmost = j;
                if (i < top) top = i;
                if (i > bottom) bottom = i;
            }
        }
    }
    int width = rightmost - leftmost + 1;
    int height = bottom - top + 1;
    if (width % 2 == 0 || height % 2 == 0) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
