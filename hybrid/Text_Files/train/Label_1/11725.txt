#include <bits/stdc++.h>
using namespace std;
const int maxN = 10, maxM = 10, maxK = 1000, maxW = 1000;
int main() {
    int n, m, k, w;
    cin >> n >> m >> k >> w;
    vector<string> g(k);
    for (int i = 0; i < k; i++) { cin >> g[i]; }
    vector<int> ans(k, 0);
    for (int i = 1; i < k; i++) {
        int sum = 0;
        for (int j = 0; j < n; j++) {
            for (int k = 0; k < m; k++) sum += (g[i - 1][j * m + k] != g[i][j * m + k]);
        }
        int ansW = sum * w;
        ans[i] = -1;
        int ansVal = maxN * maxM;
        for (int j = i - 1; j >= 0; j--) {
            int dif = 0;
            for (int u = 0; u < n; u++) {
                for (int v = 0; v < m; v++) dif += (g[j][u * m + v] != g[i][u * m + v]);
            }
            if (dif < ansVal && dif * w < ansW) {
                ans[i] = j + 1;
                ansVal = dif;
            }
        }
        ansW = n * m;
        if (ans[i] == -1 || ansW < ansVal * w) {
            ans[i] = 0;
            ansVal = ansW;
        }
    }
    int sum = n * m * k;
    for (int i = 1; i < k; i++) { sum -= ans[i] * w; }
    cout << sum << endl;
    for (int i = 0; i < k; i++) cout << i + 1 << ' ' << ans[i] << endl;
    return 0;
}
