#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int k, n;
  std::cin >> k >> n;
  std::vector<int> p(k), v(n);
  for (int& x : p) std::cin >> x;
  for (int& x : v) std::cin >> x;
  std::sort(v.begin(), v.end());
  std::sort(p.begin(), p.end());
  if (n == 0) std::cout << k + 1 << std::endl;
  else if (k == n) {
    if (p.back() != v.front() || *p.begin() != *v.begin())
      std::cout << 0 << std::endl;
    else
      std::cout << 1 << std::endl;
  } else {
    int i = 1, ans = 0;
    for (int j = 0; j < n; ++j) {
      while (i < k && p[i] < v[j]) ++i;
      if (p[i] == v[j]) ++ans;
    }
    std::cout << ans << std::endl;
  }
  return 0;
}
