#include <iostream>
#include <vector>
const int MAXN = 2e5 + 5;
using namespace std;

int main() {
    int n, a;
    vector<int> seq;
    cin >> n;
    seq.reserve(n);
    for (int i = 0, a; i < n - 1; i++) {
        cin >> a;
        seq.emplace_back(a);
    }
    vector<long long> val(n, -1);
    for (int i = n - 2, cur = 0, y = 0; i >= 0; i--) {
        y += seq[i];
        if (cur + seq[i] <= i || cur + seq[i] > n) {
            val[i] = -1;
            continue;
        }
        cur += seq[i];
        if ((cur - seq[i] >= 0) && (cur - seq[i] <= i)) {
            if (val[cur - seq[i]] == -1) {
                val[i] = -1;
            } else
                val[i] = val[cur - seq[i]] + y;
        } else {
            val[i] = -1;
        }
    }
    for (int i = 0; i < n - 1; i++) {
        cout << val[i] << '\n';
    }
    return 0;
}
