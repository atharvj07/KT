#include <bits/stdc++.h>
using namespace std;
const int MAXN = 200005;
typedef long long ll;
typedef pair<int, int> pr;
int n, m, ta, tb, k, a[MAXN], b[MAXN];
ll ans = LLONG_MAX;
struct cmp {
  bool operator()(int a, int b) {
    if (::a[a] != ::a[b]) return ::a[a] < ::a[b];
    return b < a;
  }
};
vector<int> pos[MAXN];
int main() {
  cin >> n >> m >> ta >> tb >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= m; i++) cin >> b[i];
  for (int i = 1; i <= n; i++) pos[a[i] + ta].push_back(i);
  for (int j = 1; j <= m; j++) {
    if (pos[b[j]].empty()) continue;
    int x = pos[b[j]][0];
    while (x <= n && k) {
      k--;
      x = *upper_bound(pos[b[j]].begin(), pos[b[j]].end(), x, cmp());
    }
    if (x > n)
      ans = min(ans, 1ll * b[j] + tb);
  }
  if (ans == LLONG_MAX)
    cout << -1 << endl;
  else
    cout << ans << endl;
  return 0;
}
