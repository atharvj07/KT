#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> digits;
  for (int i = 9; i >= 1; i--) {
    while (n >= i) {
      n -= i;
      digits.push_back(i);
    }
  }
  cout << digits.size() << endl;
  for (auto d : digits) cout << d << " ";
  cout << endl;
  return 0;
}
