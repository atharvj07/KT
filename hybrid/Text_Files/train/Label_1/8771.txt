#include <iostream>
#include <algorithm>
#include <vector>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        std::vector<int> s(4);
        for (int i = 0; i < 4; i++) {
            std::cin >> s[i];
        }
        std::sort(s.begin(), s.end());
        std::cout << (s[2] == s[1] && s[3] == s[2] ? "YES" : "NO") << "\n";
    }
    return 0;
}
