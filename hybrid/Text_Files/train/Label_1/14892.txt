#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
vi adj[5001];
int c[5001],vis[5001];
int n,ans;
void dfs(int x,int pre){
    if(vis[x]) return;
    vis[x]=1;
    for(int v:adj[x]){
        if(c[x]==c[v]) dfs(v,x);
    }
}
int main(){
    int x;
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>x;
        c[i]=x;
    }
    for(int i=1;i<=n;i++){
        for(int j=i+1;j<=n;j++){
            if(c[i]==c[j]){
                adj[i].push_back(j);
                adj[j].push_back(i);
            }
        }
    }
    for(int i=1;i<=n;i++){
        if(!vis[i]){
            dfs(i,i);
            ans++;
        }
    }
    cout<<ans;
    return 0;
}
