n, m, s, f = map(int, input().split())
t = list(map(int, input().split()) for _ in range(m))
ans = []
p = s
for t_i, l_i, r_i in t:
    if p >= l_i and p <= r_i:  # spy is being watched
        continue
    if p < f:
        while p < r_i:
            p += 1
            ans.append('R')
    else:
        while p > l_i:
            p -= 1
            ans.append('L')
    if p == f:
        break
print(''.join(ans))
