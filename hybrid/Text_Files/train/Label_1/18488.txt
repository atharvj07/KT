#include <iostream>
#include <vector>
#include <queue>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<vector<int>> childs(n);
  for (int i = 2; i < n; i++) {
    int fi;
    cin >> fi;
    childs[fi].push_back(i);
  }
  vector<int> ops(n), visit(n);
  for (int i = 0; i < n; i++) cin >> ops[i];
  queue<int> que;
  que.push(1);
  visit[1] = 1;
  while (!que.empty()) {
    int f = que.front();
    que.pop();
    int ch = 0, mn = 1e9, mx = 0;
    for (int c : childs[f]) {
      que.push(c);
      visit[c] = 1;
      ch++;
      if (ops[c] == 0) {
        if (visit[c]) mx = max(mx, ops[c]);
        mn = min(mn, ops[c]);
      } else {
        if (visit[c]) mn = min(mn, ops[c]);
        mx = max(mx, ops[c]);
      }
    }
    if (ch == 0)
      ops[f] = 0;
    else {
      if (ops[f] == 0)
        ops[f] = mn;
      else
        ops[f] = mx;
    }
  }
  cout << ops[1] << endl;
  return 0;
}
