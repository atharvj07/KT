#include <algorithm>
#include <iostream>
#include <vector>

int main() {
    int n, k;
    std::cin >> n >> k;
    std::vector<int> v(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> v[i];
    }
    std::sort(v.begin(), v.end());
    int res = 1e9;
    for (int i = 0; i + k - 1 < n; ++i) {
        res = std::min(res, v[i + k - 1] - v[i]);
    }
    std::cout << res << std::endl;
    return 0;
}
