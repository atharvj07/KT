#define fast ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);
#include<bits/stdc++.h>
using namespace std;
const long long int mod = 1e9 + 7;
int main() {
    fast;
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int a[n][5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                cin >> a[i][j];
            }
        }
        int check = 0;
        for (int i = 0; i < 5; i++) {
            int sum = 0;
            for (int j = 0; j < n; j++) {
                sum += a[j][i];
            }
            if (sum > n / 2)check = 1;
        }
        if (check == 1)cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
