#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int MAXN = 1e5 + 5;
vector<ll> a;
ll b[MAXN], c[MAXN];
int main() {
    ll n, q, m, t, x, y, z;
    cin >> m;
    for (int i = 1; i <= m; i++) {
        cin >> t;
        if (t == 1) {
            cin >> x;
            a.push_back(x);
        } else {
            cin >> x >> y;
            b[a.size() - x] += y;
            c[a.size() - x + 1] -= y;
        }
    }
    cin >> n;
    ll p[n];
    for (int i = 0; i < n; i++) { cin >> p[i]; }
    ll sum = 0;
    for (int i = 0; i < a.size(); i++) {
        sum += b[i];
        if (p[0] > (i + 1)) continue;
        for (int j = 0; j < n; j++) {
            if (i + 1 >= p[j]) p[j] += sum;
        }
    }
    for (int i = 0; i < n; i++) cout << a[(p[i] - 1)] << ' ';
    return 0;
}
