import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    static class Edge {
        int from;
        int to;
        Edge rev;

        public Edge(int from, int to) {
            this.from = from;
            this.to = to;
        }
    }

    static class Event {
        int city;
        Edge edge;

        public Event(int city, Edge edge) {
            this.city = city;
            this.edge = edge;
        }
    }

    static int n, m, e, s, t, r;
    static List<Edge> edges;
    static List<Event> events;
    static int[][] dp;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input1 = br.readLine().split(" ");
        n = Integer.parseInt(input1[0]);
        m = Integer.parseInt(input1[1]);
        e = Integer.parseInt(input1[2]);
        s = Integer.parseInt(input1[3]);
        t = Integer.parseInt(input1[4]);
        r = Integer.parseInt(input1[5]);

        edges = new ArrayList<>();
        events = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            String[] input = br.readLine().split(" ");
            int from = Integer.parseInt(input[0]);
            int to = Integer.parseInt(input[1]);
            Edge edge = new Edge(from, to);
            edge.rev = new Edge(to, from);
            edges.add(edge);
        }
        for (int i = 0; i < e; i++) {
            String[] input = br.readLine().split(" ");
            int city = Integer.parseInt(input[0]);
            int from = Integer.parseInt(input[1]);
            int to = Integer.parseInt(input[2]);
            Edge edge = new Edge(from, to);
            edge.rev = new Edge(to, from);
            events.add(new Event(city, edge));
        }

        dp = new int[n][r + 1];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], -1);
        }
        int ans = dfs(s, 0);
        System.out.println(ans);
    }

    static int dfs(int city, int cnt) {
        if (cnt > r) {
            return -1;
        }
        if (city == t) {
            return cnt;
        }
        if (dp[city][cnt] != -1) {
            return dp[city][cnt];
        }
        int ans = -1;
        for (Edge edge : edges) {
            if (edge.from == city) {
                int next = dfs(edge.to, cnt + 1);
                if (next != -1) {
                    ans = Math.max(ans, next);
                }
            }
        }
        for (Event event : events) {
            if (event.city == city) {
                int next = dfs(event.edge.from, cnt + 1);
                if (next != -1) {
                    ans = Math.max(ans, next);
                }
                next = dfs(event.edge.to, cnt + 1);
                if (next != -1) {
                    ans = Math.max(ans, next);
                }
            }
        }
        dp[city][cnt] = ans;
        return ans;
    }
}
