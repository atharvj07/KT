from collections import defaultdict

k = int(input())
boxes = defaultdict(list)

for i in range(k):
    n, *a = map(int, input().split())
    boxes[n].extend(sorted(a))

# print('boxes', boxes)
su = 0
for l in boxes.values():
    su += l[0]

if not su % k == 0:
    print('No')
else:
    ans = []
    t = su // k
    for l in boxes.values():
        if len(l) > 1:
            ans.append((l[0], l.pop()))
        else:
            ans.append((l[0], t))

    print("Yes")
    for a in ans:
        print(*a)
