#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    int ans = 1e9, tot = 0;
    vector<int> a;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        tot += x;
        a.push_back(x);
    }
    for (int i = 0; i <= n - (k + 1) / 2; i++) {
        ans = min(ans, a[i] + max(0, a[i + k - 1] - a[i + (k - 1) / 2]));
    }
    cout << ans << endl;
    return 0;
}
