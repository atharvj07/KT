import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }
        int cnt = 0;
        boolean prev = false;
        for (int i = 1; i < n; i++) {
            if (arr[i] == -1) {
                continue;
            }
            if (prev) {
                if (arr[i] != arr[i - 1] + (arr[i - 2] - arr[i - 1])) {
                    cnt++;
                }
            } else {
                prev = true;
            }
        }
        System.out.println(cnt);
    }
}
