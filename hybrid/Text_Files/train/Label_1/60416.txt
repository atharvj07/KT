import sys

t = int(input())

for _ in range(t):
    n = int(input())
    p = [int(i) for i in input().split()]
    c = [int(i) for i in input().split()]

    last = {}
    for i in range(n):
        last[c[i]] = i

    ans = float('inf')
    vis = [False] * (n + 1)
    for i in range(1, n + 1):
        if vis[i]:
            continue

        k = i
        cycle_len = 0
        while not vis[k]:
            vis[k] = True
            cycle_len += 1
            k = p[k - 1]

        if cycle_len == n:
            ans = 1
            break

        if last[c[k - 1]] in last:
            ans = min(ans, cycle_len + last[c[k - 1]])

    print(ans)
