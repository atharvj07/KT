#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=1e9+7;
ll quick_power(ll num,ll po){
    ll re=1;
    while(po){
        if(po%2)re=(re*num)%mod;
        num=(num*num)%mod;
        po/=2;
    }
    return re;
}
int main(){
    ll n,x,ans=1;
    cin>>n;
    unordered_map<ll,ll>m;
    for(int i=0;i<n;i++){
        cin>>x;
        m[x]++;
    }
    for(auto it:m){
        ans=(ans*it.second)%mod;
        ans=(ans*quick_power(2,it.second-1))%mod;
    }
    cout<<ans<<endl;
    return 0;
}
