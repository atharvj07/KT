import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) break;

            int[] scores = new int[n];
            for (int i = 0; i < m; i++) {
                for (int j = 0; j < n; j++) {
                    scores[j] += sc.nextInt();
                }
            }

            int maxScore = 0;
            for (int score : scores) {
                maxScore = Math.max(maxScore, score);
            }

            System.out.println(maxScore);
        }
    }
}
