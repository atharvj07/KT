#include<bits/stdc++.h>
using namespace std;
#define Mod 1000000007
const int N=5005;
 
int f[N][N];
 
int main() {
    int n;
    cin>>n;
    f[n][1]=1;
    f[n][0]=1;
    for(int i=n-1;i>=1;i--) {
        char x;
        cin>>x;
        if(x=='f') {
            for(int j=i;j<=n;j++) {
                f[i][1]=(f[i][1]+f[j][0])%Mod;
            }
            for(int j=i+1;j<=n;j++) {
                f[i][j]=(f[i][j]+f[j][j-1])%Mod;
            }
        }
        else {
            for(int j=i;j<=n;j++) {
                f[i][j]=(f[i][j]+f[j][j])%Mod;
            }
        }
    }
    cout<<f[1][1]<<endl;
}
