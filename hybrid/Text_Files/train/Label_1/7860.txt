#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, m, sum = 0;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i], sum += a[i];
    sort(a.rbegin(), a.rend());
    cout << max(a[0] + m / n, sum / n + (m + n - 1) / n) << ' '
         << a[0] + m << endl;
}
