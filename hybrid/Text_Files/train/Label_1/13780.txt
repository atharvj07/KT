#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<long long> a(n), b(m);
    for (long long &x : a) cin >> x;
    for (long long &x : b) cin >> x;
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    cout << a.back() * b.front();
}
