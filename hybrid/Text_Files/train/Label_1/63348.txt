import sys
input = sys.stdin.readline

q = int(input())
for _ in range(q):
    n = int(input())
    adj = [[] for _ in range(n)]
    for _ in range(n - 1):
        a, b = map(int, input().split())
        a, b = a - 1, b - 1
        adj[a].append(b)
        adj[b].append(a)
    dep = [-1] * n
    vis = [False] * n
    def dfs(u, p, d):
        vis[u] = True
        dep[u] = d
        for v in adj[u]:
            if v == p:
                continue
            if not vis[v]:
                dfs(v, u, d + 1)
    dfs(0, 0, 0)
    ans = 0
    for i in range(n):
        for j in adj[i]:
            if i < j and dep[i] % 2 != dep[j] % 2:
                ans += 1
    print(ans + 1)
