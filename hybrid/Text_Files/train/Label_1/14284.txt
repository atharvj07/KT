#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> l(n), r(n);
  l[0] = a[0];
  r[n - 1] = a[n - 1];
  for (int i = 1; i < n; i++) l[i] = max(l[i - 1], a[i]);
  for (int i = n - 2; i >= 0; i--) r[i] = max(r[i + 1], a[i]);
  for (int i = 0; i < n; i++) {
    if (i % 2 == 0)
      cout << max(l[i], r[i + 1]) << " ";
    else
      cout << min(l[i], r[i + 1]) << " ";
  }
}
