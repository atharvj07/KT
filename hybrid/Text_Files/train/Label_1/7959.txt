#include <bits/stdc++.h>
using namespace std;

int main() {
    int q; cin >> q;
    while (q--) {
        int n; cin >> n;
        vector<int> v(4 * n);
        for (int &x: v) cin >> x;
        sort(v.rbegin(), v.rend());
        bool ok = 1;
        for (int i = 0; i < 4*n; i += 4) {
            if (v[i] != v[i+1] || v[i] != v[i+2] || v[i] != v[i+3]) ok = 0;
        }
        cout << (ok ? "YES" : "NO") << endl;
    }
}
