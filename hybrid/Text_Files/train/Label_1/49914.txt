import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int t = scanner.nextInt();
    String s = scanner.next();
    char[] a = s.toCharArray();
    for (int i = 0; i < t; i++) {
      for (int j = 0; j < n - 1; j++) {
        if (a[j] == 'B' && a[j + 1] == 'G') {
          char temp = a[j];
          a[j] = a[j + 1];
          a[j + 1] = temp;
        }
      }
    }
    System.out.println(new String(a));
  }
}
