n, k = map(int, input().split())
s = input().split()
ans = []
for i in range(n - k + 1):
    if s[i] == "NO":
        if i == 0 or ans[i - 1][0] != 'A':
            ans.append('A')
        else:
            c = ord(ans[i - 1][0]) + 1
            while c <= ord('Z'):
                if all(x[0] != chr(c) for x in ans):
                    break
                c += 1
            ans.append(chr(c))
    else:
        ans.append(ans[i - 1])
print(*ans)
