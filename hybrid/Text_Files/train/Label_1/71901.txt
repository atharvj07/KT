from bisect import bisect_right

n = int(input())
*a, = map(int, input().split())
*b, = map(int, input().split())

# calculate prefix sum for a
pref = [0]
for i in a:
    pref.append(pref[-1] + (b[i - 1] == i))

# calculate suffix sum for b
suff = [0] * (n + 1)
for i in range(n - 1, -1, -1):
    suff[i] = suff[i + 1] + (a[b[i] - 1] == i)

ans = 0
for i in range(n):
    # calculate the number of matching pairs by cyclically shifting b
    shift = bisect_right(a, b[i])
    ans = max(ans, pref[shift] + suff[i + 1] + 1)
print(ans)
