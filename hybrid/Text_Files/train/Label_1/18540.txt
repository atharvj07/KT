#include <iostream>
#include <sstream>
#include <string>
using namespace std;
int main() {
  string a, b;
  getline(cin, a);
  getline(cin, b);
  stringstream aa(a), bb(b);
  int y1, mo1, d1, y2, mo2, d2;
  char d;
  aa >> y1 >> d >> mo1 >> d >> d1;
  bb >> y2 >> d >> mo2 >> d >> d2;
  long long ans = 0, y = 0;
  ans += (y2 - y1) * 365;
  for (y = y1; y < y2; ++y) {
    if ((y % 4) == 0 && (y % 100) != 0 || (y % 400) == 0) { ++ans; }
  }
  if (((y2 % 4) == 0 && (y2 % 100) != 0 || (y2 % 400) == 0) && mo2 > 2) {
    ++ans;
  }
  ans -= ((y1 % 4) == 0 && ((y1 % 100)) != 0 || ((y1 % 400)) == 0);
  for (int i = mo1 + 1; i < 13; ++i) {
    if (i == 2 && ((y1 % 4) == 0 && (y1 % 100) != 0 || (y1 % 400) == 0)) {
      ans -= 1;
      continue;
    }
    if (i == 2) {
      ans -= 2;
      continue;
    }
    if (i < 8) {
      if (i % 2 == 0) ans -= 3;
      if ((i + 1) % 2 == 0) ans -= 3;
    } else if (i % 2 == 0)
      ans -= 3;
  }
  if (mo2 == 2) ans -= d2 - 1;
  if (mo1 == 1) ans += d1 - 1;
  if (mo1 == 2) ans += d1;
  if (mo2 > 2) ans += 31 - d2;
  if (mo2 < mo1) ans += 31;
  if (mo1 < 8 && mo1 % 2 == 0 && mo2 > mo1) ans += 3;
  if (mo1 < 8 && (mo1 + 1) % 2 == 0 && mo2 > mo1) ans += 3;
  if (mo1 > 7 && mo2 > mo1) ans += 30;
  cout << ans;
  return 0;
}
