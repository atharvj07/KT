import java.util.*;

class Main {
    static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] weights = new int[n];
        for (int i = 0; i < n; i++) {
            weights[i] = sc.nextInt();
        }
        if (!canCross(weights, k)) {
            System.out.println(-1);
            System.out.println(0);
            return;
        }
        int rides = minRides(weights, k);
        System.out.println(rides);
        System.out.println(countWays(weights, k, rides) % MOD);
    }

    private static boolean canCross(int[] weights, int k) {
        int total = 0;
        for (int w : weights) {
            total += w;
        }
        return total <= k;
    }

    private static int minRides(int[] weights, int k) {
        int rides = 0, total = 0;
        for (int w : weights) {
            if (total + w <= k) {
                total += w;
            } else {
                rides++;
                total = w;
            }
        }
        return rides + 1;
    }

    private static long countWays(int[] weights, int k, int rides) {
        return countWays(weights, k, rides, 0, 0);
    }

    private static long countWays(int[] weights, int k, int rides, int idx, int total) {
        if (rides == 0) {
            return total == 0 ? 1 : 0;
        }
        if (idx == weights.length) {
            return total == 0 ? 1 : 0;
        }
        long ways = 0;
        if (total + weights[idx] <= k) {
            ways += countWays(weights, k, rides, idx + 1, total + weights[idx]);
            ways %= MOD;
        }
        ways += countWays(weights, k, rides, idx + 1, total);
        ways %= MOD;
        return ways;
    }
}
