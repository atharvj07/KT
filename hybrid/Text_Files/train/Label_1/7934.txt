#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int MOD = 1000000007;
const ll LLINF = 4e18;

#define rep(i, n) for (ll i = 0; i < (ll)(n); i++)
#define mod(x) ((x) % MOD + MOD) % MOD
#define all(x) (x).begin(), (x).end()
#define rall(x) (x).rbegin(), (x).rend()

struct Node {
  int p;
  bool violate;
};

int main() {
  int n;
  cin >> n;

  vector<int> a(n);
  rep(i, n) cin >> a[i];

  vector<Node> node(n);
  rep(i, n) node[i] = {0, false};

  vector<int> ans(n - 1);

  rep(k, n - 1) {
    int cnt = 0;
    for (int i = n - 1; i >= 0; i--) {
      int p = (i + k) / (k + 1);
      if (p != 0 && a[i] < a[p - 1]) cnt++;
      node[i] = {p, a[i] < a[p - 1]};
    }
    ans[k] = cnt;
  }

  rep(i, n - 1) cout << ans[i] << " ";

  return 0;
}
