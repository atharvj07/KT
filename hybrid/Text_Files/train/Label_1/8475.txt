#include <iostream>

using namespace std;

const int MAXN = 100005;

int main() {
    int m, n;
    cin >> m >> n;
    long long a[MAXN];
    for (int i = 0; i < m; i++) {
        int type;
        cin >> type;
        if (type == 1) {
            long long x;
            cin >> x;
            a[i] = x;
        } else {
            int l, c;
            cin >> l >> c;
            for (int j = 0; j < c; j++) {
                for (int k = 0; k < l; k++) {
                    a[i + j * l + k] = a[k];
                }
            }
            i += c * l - 1;
        }
    }
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        cout << a[x - 1] << " ";
    }
    cout << endl;
    return 0;
}
