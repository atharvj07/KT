def can_print_s_with_t(s, t):
    i, j = 0, 0
    while i < len(s) and j < len(t):
        if s[i] != t[j]:
            return "NO"
        # Move j to match the stretch of the current character in t
        while i < len(s) and j < len(t) and s[i] == t[j]:
            j += 1
        i += 1
    # If i has reached the end of s, j must also have reached the end of t
    if i == len(s) and j == len(t):
        return "YES"
    else:
        return "NO"

n = int(input())
for _ in range(n):
    s = input()
    t = input()
    print(can_print_s_with_t(s, t))
