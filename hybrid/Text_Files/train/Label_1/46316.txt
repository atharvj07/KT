import java.util.*;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        List<Integer> p = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            p.add(in.nextInt() - 1);
        }
        List<Integer> pref = new ArrayList<>();
        long sum = 0;
        for (int i = 0; i < n; i++) {
            pref.add((int) (sum = sum + (n - p.get(i))));
        }
        int min = pref.stream().min(Integer::compare).get();
        int k = 0;
        for (int i = 0; i < n; i++) {
            int cur = pref.get(i) - min;
            if (cur == 0) {
                k = i;
                break;
            }
        }
        System.out.println(min + " " + k);
    }
}
