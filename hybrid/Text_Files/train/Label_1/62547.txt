n = int(input())
scores, ids = [], []
for i in range(n):
    a, b, c, d = [int(x) for x in input().split()]
    scores.append(a + b + c + d)
    ids.append(i + 1)
rank = [1] * n
for i in range(n):
    for j in range(n):
        if scores[i] < scores[j]:
            rank[i] += 1
        elif scores[i] == scores[j] and ids[i] > ids[j]:
            rank[i] += 1
print(rank.index(min(rank)) + 1)
