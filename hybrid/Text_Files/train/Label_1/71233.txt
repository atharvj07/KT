s=input()
t=input()
n=len(s)
k=len(t)
t='$'+t+'#'
lps=[0]*len(t)
i=1
j=0
while i<len(t):
    if t[i]==t[j]:
        j+=1
        lps[i]=j
        i+=1
    else:
        if j==0:
            lps[i]=0
            i+=1
        else:
            j=lps[j-1]
oc=[0]*len(t)
oc[len(t)-1]=1
for i in range(len(t)-2,-1,-1):
    oc[i]=oc[i+1]
    if t[i]==t[lps[i]]: oc[i]+=oc[i+lps[i]]
ans=0
for i in range(n,k-n+1):
    if s==t[i-n:i]: ans+=oc[i+n]
print(ans)
