#include <bits/stdc++.h>
using namespace std;
int a[111], n;
bool ok() {
    for (int i = 1; i < n; i++)
        if (a[i] > a[i + 1]) return false;
    return true;
}
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    while (!ok()) {
        for (int i = 1; i + 1 <= n; i += 2)
            if (a[i] > a[i + 1]) swap(a[i], a[i + 1]), cout << i << " " << i + 1 << endl;
    }
    return 0;
}
