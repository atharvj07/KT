s = input()
t = input()
n = len(s)
a = [0]*26
b = [0]*26
for i in range(n):
    a[ord(s[i])-ord('a')]+=1
    b[ord(t[i])-ord('a')]+=1
for i in range(n+1, len(t)):
    b[ord(t[i])-ord('a')]+=1
res = [[0]*26 for _ in range(2)]
for i in range(26):
    res[0][i] = min(a[i], b[i])
    res[1][i] = a[i]-res[0][i]
    b[i]-=res[0][i]
yay = whoops = 0
for i in range(n):
    if t[i]==s[i]: yay+=1
    if t[i].lower()==s[i].lower(): whoops+=1
    a[ord(s[i])-ord('a')]-=1
    b[ord(t[i])-ord('a')]-=1
    sb = 0
    for j in range(26):
        sb+=min(res[1][j]-b[j], a[j])
        whoops+=min(res[1][j]-b[j], a[j])
    yay+=sb
print(yay, whoops)
