n = int(input())
a = list(map(int, input().split()))
odd = [False] * n
even = [False] * n
if a[0] % 2 == 1:
    odd[0] = True
if a[0] % 2 == 0:
    even[0] = True
for i in range(1, n):
    if a[i] % 2 == 1:
        odd[i] = True
        even[i] = even[i - 1]
    else:
        odd[i] = odd[i - 1]
        even[i] = True
        if odd[i - 1]:
            even[i] = True
ans = 0
for i in range(n):
    if odd[i]:
        ans = max(ans, a[i])
    else:
        if even[i - 1]:
            ans = max(ans, a[i])
print(ans)
