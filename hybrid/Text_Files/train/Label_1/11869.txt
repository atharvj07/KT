#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<string> s(n);
  int xs, xe, ys, ye;
  for (int i = 0; i < n; i++) {
    cin >> s[i];
    for (int j = 0; j < m; j++) {
      if (s[i][j] == 'w') {
        xs = min(xs, i);
        xe = max(xe, i);
        ys = min(ys, j);
        ye = max(ye, j);
      }
    }
  }
  if (xs == ye || xs == xe || xe - xs != ye - ys) {
    cout << -1;
    return 0;
  }
  xe++;
  ye++;
  for (int i = xs; i < xe; i++) { s[i][ys] = '+'; }
  for (int j = ys + 1; j < ye; j++) { s[xs][j] = '+'; }
  for (int i = xs + 1; i < xe; i++) { s[i][ye - 1] = '+'; }
  for (int j = ys; j < ye - 1; j++) { s[xe - 1][j] = '+'; }
  for (int i = 0; i < n; i++) cout << s[i] << '\n';
}
