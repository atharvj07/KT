#include <bits/stdc++.h>
#define REP(i, x, n) for (int i = x; i < n; i++)
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        long long a[n];
        REP(i, 0, n) { cin >> a[i]; }
        sort(a, a + n);
        long long sum = 0;
        REP(i, 0, n) { sum += a[i] * 1.0 / (i + 1); }
        cout << (sum == m ? "YES" : "NO") << endl;
    }
}
