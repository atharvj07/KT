#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int n, x;
ll a[300005], f[300005], g[300005];
int main() {
    cin >> n >> x;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    f[1] = a[1];
    for (int i = 2; i <= n; ++i) f[i] = max(f[i - 1], 0LL) + a[i];
    g[n] = x * a[n];
    for (int i = n - 1; i >= 1; --i) g[i] = max(x * a[i], g[i + 1]);
    ll ans = 0;
    for (int i = 1; i <= n; ++i) ans = max(ans, max(f[i - 1], g[i + 1]));
    cout << ans << endl;
    return 0;
}
