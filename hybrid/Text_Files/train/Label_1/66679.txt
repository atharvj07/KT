t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    pin_codes = [input().strip() for _ in range(n)]
    changes = 0
    updated_codes = set()
    for code in pin_codes:
        original_code = code
        while code in updated_codes:
            code = list(code)
            code[0] = str((int(code[0]) + 1) % 10)
            code = ''.join(code)
            changes += 1
        updated_codes.add(code)
        print(code)
    print(changes, end='\n' if _ < t - 1 else '')
