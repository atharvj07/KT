import sys

n = int(input())
min1, max1 = map(int, input().split())
min2, max2 = map(int, input().split())
min3, max3 = map(int, input().split())

f = [0] * 3
s = [0] * 3

for i in range(3):
    f[i] = max1 - min1
    s[i] = max2 - min2
    if f[i] > max1:
        f[i] = max1
    if f[i] < min1:
        f[i] = min1
    if s[i] > max2:
        s[i] = max2
    if s[i] < min2:
        s[i] = min2

for i in range(3):
    if n - f[i] - s[i] > max3:
        t = n - f[i] - s[i] - max3
        if f[i] > t:
            f[i] -= t
        elif s[i] > t:
            s[i] -= t
        else:
            f[i] -= t

print(*f)
