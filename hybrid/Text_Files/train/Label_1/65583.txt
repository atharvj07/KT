MOD = 1000000009

def pow_mod(x, y, mod):
    res = 1
    while y > 0:
        if y % 2:
            res = (res * x) % mod
        x = (x * x) % mod
        y //= 2
    return res

def solve(n, m, k):
    if m < k:
        return m
    full_cycles = m // k
    rem = m % k
    total_score = (pow_mod(2, full_cycles, MOD) * (k-1) + rem) % MOD
    total_score = (total_score + pow_mod(2, full_cycles, MOD) - 1) % MOD
    return total_score

n, m, k = map(int, input().split())
print(solve(n, m, k))
