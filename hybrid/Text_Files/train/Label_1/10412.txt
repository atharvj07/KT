#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<pair<int, int>> books;
  for (int i = 0; i < n; i++) {
    int t, w;
    cin >> t >> w;
    books.push_back({t, w});
  }

  sort(books.begin(), books.end());

  int vertical_thickness = 0;
  int horizontal_thickness = 0;
  for (int i = 0; i < n; i++) {
    if (books[i].first == 1) {
      vertical_thickness += books[i].second;
    } else {
      if (horizontal_thickness + books[i].second <= vertical_thickness) {
        horizontal_thickness += books[i].second;
      } else {
        vertical_thickness += books[i].second;
        horizontal_thickness = 0;
      }
    }
  }

  cout << vertical_thickness << endl;

  return 0;
}
