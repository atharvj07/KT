import sys

n = int(input())
g = [list(map(int, input().split())) for _ in range(n)]
d = [[1e9] * n for _ in range(n)]

for u in range(n):
    d[u][u] = 0
    for v in range(u + 1, n):
        if g[u][v]: d[u][v] = g[u][v]
        d[v][u] = d[u][v]

for k in range(n):
    for u in range(n):
        for v in range(n):
            if d[u][k] + d[k][v] < d[u][v]: d[u][v] = d[u][k] + d[k][v]

print(max(max(row) for row in d if row != [1e9] * n))
