from collections import deque

def bfs(start, graph, color):
    queue = deque([start])
    visited = [False] * len(graph)
    visited[start] = True
    moves = []

    while queue:
        vertex = queue.popleft()
        for neighbor, edge_color in graph[vertex]:
            if visited[neighbor]:
                continue
            visited[neighbor] = True
            if edge_color != color:
                moves.append(neighbor + 1)
                for next_neighbor, _ in graph[neighbor]:
                    graph[next_neighbor] = [(v, 'R' if c == 'B' else 'B') for v, c in graph[next_neighbor]]
            else:
                queue.append(neighbor)

    for i in range(len(graph)):
        for _, c in graph[i]:
            if c != color:
                return False, []

    return True, moves

def solve_graph_coloring(n, m, edges):
    graph = [[] for _ in range(n)]
    for u, v, c in edges:
        graph[u-1].append((v-1, c))
        graph[v-1].append((u-1, c))

    for color in ['R', 'B']:
        possible, moves = bfs(0, [list(adj) for adj in graph], color)
        if possible:
            return [len(moves)] + moves

    return [-1]

n, m = map(int, input().split())
edges = [list(input().split())[:2] + [input().split()[2]] for _ in range(m)]
result = solve_graph_coloring(n, m, edges)
print(' '.join(map(str, result)))
