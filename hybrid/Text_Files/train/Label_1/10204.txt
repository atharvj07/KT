#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n,d,pos;
    cin >> n >> d;
    pos = d;
    vector<long long> s(n), p(n);
    for (int i=0; i<n; i++)
        cin >> s[i];
    for (int i=0; i<n; i++)
        cin >> p[i];
    s[d-1] += p[n-1];
    sort(s.begin(), s.end(), greater<long long>());
    d--;
    while (d!=-1 && s[d]==s[d+1])
        d--;
    cout << d+1 << endl;
    return 0;
}
