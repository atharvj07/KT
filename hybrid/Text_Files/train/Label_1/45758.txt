import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] a = new long[n + 1];
        long[][] s = new long[n + 1][n + 2];
        boolean ok = true;
        for (int i = 0; i < m; i++) {
            int t = sc.nextInt();
            int l = sc.nextInt();
            int r = sc.nextInt();
            if (t == 1) {
                int d = sc.nextInt();
                a[l - 1] += d;
                a[r] -= d;
                s[l - 1][r] += d;
            } else {
                long mi = sc.nextInt();
                long mx = 0;
                for (int j = l - 1; j < r; j++) {
                    mx = Math.max(mx, s[j][r] - s[j][l - 1]);
                }
                if (mi != mx) {
                    ok = false;
                }
            }
        }
        if (!ok) {
            System.out.println("NO");
            return;
        }
        long cur = 0;
        for (int i = 0; i < n; i++) {
            cur += a[i];
            s[i][i + 1] = cur;
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            System.out.print(s[i][i + 1] + " ");
        }
    }
}
