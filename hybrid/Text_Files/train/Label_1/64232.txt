import sys

def solve():
    n, k, d = [int(x) for x in input().split()]
    if n > k * d:
        print(-1)
        return

    buses = []
    for i in range(d):
        buses.append([0] * n)

    students = [i for i in range(n)]
    for day in range(d):
        for i in range(n):
            buses[day][i] = students[i % k]
        students = [buses[day][i] for i in range(n)]

    for i in range(d):
        print(' '.join([str(buses[i][j] + 1) for j in range(n)]))

solve()
