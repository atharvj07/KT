#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll a[50], b[50], c[50], d[50], pw[50];
int32_t main() {
  ll n;
  string s;
  cin >> n >> s;
  pw[0] = 1;
  for (ll i = 1; i <= n; i++) { pw[i] = pw[i - 1] * 2; }
  ll m = s.size();
  for (ll i = 0; i < m; i++) {
    a[i] = pw[i];
    if (s[i] == '1') b[i] = 1;
  }
  ll cur = 0;
  for (ll i = m - 1; i >= 0; i--) {
    c[i] = cur;
    if (s[i] == '1') cur += pw[n - i];
  }
  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    cur = 0;
    for (ll j = 0; j < m; j++) { d[j] = 0; }
    for (ll j = i; j < i + m; j++) {
      ll idx = j % n;
      d[idx] += a[j - i];
      if (s[j - i] == '1') cur += a[j - i];
    }
    cur -= c[i % n];
    for (ll j = i + 1; j < i + n; j++) {
      ll idx = j % n;
      ans += b[idx] * cur;
      cur -= d[idx];
    }
  }
  cout << ans << endl;
  return 0;
}
