n = int(input())
strings = [input() for _ in range(n)]

counts = [(s.count('s'), s.count('h')) for s in strings]
counts.sort(key=lambda x: (x[0], -x[1]))

total_s = sum(count[0] for count in counts)
max_noise = 0
for s, h in counts:
    total_s -= s
    max_noise += s * total_s
    max_noise += s * h

print(max_noise)
