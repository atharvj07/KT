#include<iostream>
#include<vector>
using namespace std;
int n,m,ans,l[200001],r[200001],c[200001];
vector<int>v[200001];
void dfs(int x,int y){
	if(l[x])return;
	l[x]=1,r[y]=1;
	int t=0;
	for(int i=0;i<v[x].size();++i)if(!r[v[x][i]])t=v[x][i];
	if(t>0)dfs(t,x);
}
int main(){
	cin>>n;
	for(int i=1;i<=n-1;++i){
		int a,b;cin>>a>>b;
		v[a].push_back(b),v[b].push_back(a);
	}
	for(int i=1;i<=n;++i)if(!l[i])++ans,dfs(i,i),++c[ans];
	cout<<ans<<'\n';
	for(int i=1;i<=ans;++i){
		cout<<c[i];
		for(int j=1;j<=n;++j)if(c[i]==r[j])cout<<' '<<j;
		cout<<'\n';
	}
	return 0;
}
