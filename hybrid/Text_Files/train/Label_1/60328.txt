def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def next_prime(n):
    while True:
        n += 1
        if is_prime(n):
            return n

t = int(input().strip())
for _ in range(t):
    d = int(input().strip())
    first_prime = next_prime(d)
    second_prime = next_prime(first_prime + d - 1)
    print(first_prime * second_prime)
