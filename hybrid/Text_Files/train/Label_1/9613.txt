#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= 4; i++) {
        int a, b, c, d;
        cin >> a >> b >> c >> d;
        int cost = min(a + c, b + d);
        if (cost <= n) {
            cout << i << ' ' << a << ' ' << b << endl;
            return 0;
        }
    }
    cout << -1 << endl;
    return 0;
}
