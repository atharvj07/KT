import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) throws Exception {
        long[] c = new long[200001];
        int[] a = new int[200001];
        boolean[] f = new boolean[200001];
        boolean[] u = new boolean[200001];
        long ans = 0;
        int n;

        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;

        n = Integer.parseInt(in.readLine());
        st = new StringTokenizer(in.readLine());
        for (int i = 1; i <= n; ++i) {
            c[i] = Integer.parseInt(st.nextToken());
        }
        st = new StringTokenizer(in.readLine());
        for (int i = 1; i <= n; ++i) {
            a[i] = Integer.parseInt(st.nextToken());
            u[i] = true;
        }
        for (int i = 1; i <= n; ++i) {
            if (u[i]) {
                long Min = Long.MAX_VALUE;
                int x = i;
                while (true) {
                    if (f[x]) {
                        break;
                    }
                    Min = Math.min(Min, c[x]);
                    f[x] = true;
                    x = a[x];
                }
                ans += Min;
            }
            if (u[a[i]]) {
                u[a[i]] = false;
                for (int j = a[i];; j = a[j]) {
                    u[j] = false;
                    if (j == i) {
                        break;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
