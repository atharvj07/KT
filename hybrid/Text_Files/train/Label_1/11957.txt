#include <iostream>

using namespace std;

int main() {
  int n, k, m, t;
  cin >> n >> k >> m >> t;
  while (t--) {
    int op, x;
    cin >> op >> x;
    if (op) {
      if (x < k)
        k++;
      if (n != m)
        n++;
    } else {
      if (x == k - 1)
        k--;
      if (n != 2)
        n--;
    }
    cout << n << " " << k << endl;
  }
}
