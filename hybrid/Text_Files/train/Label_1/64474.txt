import sys

n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]

def bfs(sr, sc):
    q = [(sr, sc)]
    vis[sr][sc] = True
    while q:
        r, c = q.pop(0)
        for dr, dc in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
            nr = r + dr
            nc = c + dc
            if 0 <= nr < n and 0 <= nc < m and grid[nr][nc] == 'B' and not vis[nr][nc]:
                q.append((nr, nc))
                vis[nr][nc] = True

vis = [[False] * m for _ in range(n)]
for i in range(n):
    for j in range(m):
        if grid[i][j] == 'B':
            bfs(i, j)
            break
    else:
        continue
    break

ok = True
for i in range(n):
    for j in range(m):
        if grid[i][j] == 'B' and not vis[i][j]:
            ok = False
            break
    if not ok:
        break

print('YES' if ok else 'NO')
