#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 1; i < n - 1; i++) {
        if (a[i] > a[0] && a[i] < a[n - 1]) {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
