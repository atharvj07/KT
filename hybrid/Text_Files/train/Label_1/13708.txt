#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int& ai : a) { cin >> ai; }
  bool impossible = false;
  for (int i = 2; i < n; i += 2) {
    if (a[i] < a[i - 1]) { impossible = true; }
  }
  for (int i = 1; i < n; i += 2) {
    if (a[i] > a[i - 1]) { impossible = true; }
  }
  if (impossible) {
    cout << "Impossible" << endl;
    return 0;
  }
  vector<int> b(n);
  b[0] = a[0];
  for (int i = 1; i < n; i++) {
    b[i] = (i & 1) ? max(a[i], b[i - 1] - 1) : min(a[i], b[i - 1] + 1);
  }
  for (int bi : b) { cout << bi << " "; }
  cout << endl;
  return 0;
}
