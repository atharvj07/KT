#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
typedef pair<ll, ll> pi;
typedef pair<pi, ll> pii;

const ll INF = 1e18;
const ll MAX_N = 1e5;

ll n, m, s;
ll a[MAX_N], b[MAX_N], c[MAX_N];
vector<pi> adj[MAX_N];
ll dis[MAX_N];

bool dijkstra() {
  for (ll i = 1; i <= n; i++) { dis[i] = INF; }
  dis[1] = 0;
  priority_queue<pi, vector<pi>, greater<pi>> pq;
  pq.push({0, 1});
  while (!pq.empty()) {
    ll d, u;
    tie(d, u) = pq.top();
    pq.pop();
    if (d > dis[u]) continue;
    for (const auto& [v, w] : adj[u]) {
      if (d + w < dis[v]) {
        dis[v] = d + w;
        pq.push({dis[v], v});
      }
    }
  }
  return dis[n] != INF;
}

int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> m >> s;
  for (ll i = 1; i <= m; i++) cin >> a[i];
  for (ll i = 1; i <= n; i++) cin >> b[i];
  for (ll i = 1; i <= n; i++) cin >> c[i];
  for (ll i = 1; i <= m; i++) {
    for (ll j = 1; j <= n; j++) {
      if (b[j] >= a[i]) adj[j].push_back({n + i, 1});
    }
  }
  for (ll i = 1; i <= n; i++) adj[n + i].push_back({n, c[i]});
  if (!dijkstra()) cout << "NO" << endl;
  else {
    cout << "YES" << endl;
    vector<ll> ans;
    ll u = n;
    while (u != 1) {
      for (const auto& [v, w] : adj[u]) {
        if (dis[u] == dis[v] + w) {
          if (v <= n) ans.push_back(v - n);
          u = v;
          break;
        }
      }
    }
    reverse(ans.begin(), ans.end());
    for (const auto& x : ans) cout << x << " ";
    cout << endl;
  }
  return 0;
}
