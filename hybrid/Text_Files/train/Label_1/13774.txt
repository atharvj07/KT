#include <vector>
#include <string>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);

  int n, m;
  cin >> n >> m;

  string s, t;
  cin >> s >> t;

  vector<int> ans;
  for (int i = 0; i < n; i++) {
    if (s[i] != t[i]) {
      ans.push_back(i + 1);
    }
  }

  cout << ans.size() << endl;
  for (int x : ans) {
    cout << x << " ";
  }
  cout << endl;

  return 0;
}
