import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class OgudarOlokWarehouse {

    private static final int EMPTY = 0;
    private static final int OCCUPIED = 1;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] warehouse = new int[n][m];
        Map<String, Integer[]> boxLocation = new HashMap<>();

        for (int i = 0; i < k; i++) {
            String operation = scanner.next();
            if (operation.equals("+1")) {
                int shelf = scanner.nextInt() - 1;
                int section = scanner.nextInt() - 1;
                String id = scanner.next();
                addBox(warehouse, boxLocation, shelf, section, id);
            } else if (operation.equals("-1")) {
                String id = scanner.next();
                removeBox(warehouse, boxLocation, id);
            }
        }
    }

    private static void addBox(int[][] warehouse, Map<String, Integer[]> boxLocation, int shelf, int section, String id) {
        if (warehouse[shelf][section] == EMPTY) {
            warehouse[shelf][section] = OCCUPIED;
            boxLocation.put(id, new Integer[]{shelf, section});
        } else {
            for (int i = section + 1; i < warehouse[shelf].length; i++) {
                if (warehouse[shelf][i] == EMPTY) {
                    warehouse[shelf][i] = OCCUPIED;
                    boxLocation.put(id, new Integer[]{shelf, i});
                    return;
                }
            }
            for (int i = shelf + 1; i < warehouse.length; i++) {
                for (int j = 0; j < warehouse[i].length; j++) {
                    if (warehouse[i][j] == EMPTY) {
                        warehouse[i][j] = OCCUPIED;
                        boxLocation.put(id, new Integer[]{i, j});
                        return;
                    }
                }
            }
        }
    }

    private static void removeBox(int[][] warehouse, Map<String, Integer[]> boxLocation, String id) {
        Integer[] location = boxLocation.get(id);
        if (location != null) {
            warehouse[location[0]][location[1]] = EMPTY;
            boxLocation.remove(id);
            System.out.println(location[0] + 1 + " " + (location[1] + 1));
        } else {
            System.out.println("-1 -1");
        }
    }
}
