from collections import Counter
n=input()
a=dict(Counter(input().split()))
b=dict(Counter(input().split()))
diff=[]
for i in range(1,6):
    if a.get(str(i),0)!=b.get(str(i),0):
        diff.append(abs(a.get(str(i),0)-b.get(str(i),0)))
if sum(diff)%2==0:
    print(sum(diff)//2)
elif sum(diff)%4==0:
    print(sum(diff)//4)
else:
    print(-1)
