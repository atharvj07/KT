#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        string s;
        cin >> n >> s;
        int l = 0, r = n - 1, lp = -1;
        string ans(n, '1');
        for (int i = 0; i < n; ++i) {
            if (s[i] > s[r]) {
                ans[i] = '2';
                r = i;
            } else if (ans[i] == '1' && s[i] > s[l]) {
                lp = i;
                ans[i] = '2';
            } else if (ans[i] == '1' && s[i] < s[l]) {
                if (lp != -1) {
                    ans[lp] = '1';
                    lp = -1;
                }
                l = i;
                if (l >= r) r = i;
            }
        }
        if (lp != -1) ans[lp] = '1';
        string nw = "";
        for (int i = 0; i < n; ++i) {
            if (ans[i] == '1') nw += s[i];
        }
        for (int i = 0; i < n; ++i) {
            if (ans[i] == '2') nw += s[i];
        }
        bool f = 1;
        for (int i = 0; i < n - 1; ++i) {
            if (nw[i] > nw[i + 1]) {
                f = 0;
                break;
            }
        }
        if (f) {
            cout << ans << '\n';
        } else {
            cout << '-' << '\n';
        }
    }
}
