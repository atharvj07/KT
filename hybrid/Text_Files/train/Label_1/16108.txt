#include <algorithm>
#include <cstdio>
#include <limits>

const int Maxn = 100000000;
const int Mod = int(1e9) + 7;

inline int read() {
  int x;
  scanf("%d", &x);
  return x;
}

int n, m, c_l, c_e, v;
int l[500000], e[500000];

void solve() {
  n = read(), m = read();
  c_l = read(), c_e = read();
  v = read();
  for (int i = 0; i < c_l; ++i) l[i] = read();
  for (int i = 0; i < c_e; ++i) e[i] = read();
  for (int i = 0; i < c_l; ++i) l[i] = (l[i] - 1) * n + 1;
  for (int i = 0; i < c_e; ++i) e[i] = (e[i] - 1) * n;
  int q = read();
  while (q--) {
    int x1 = read(), y1 = read(), x2 = read(), y2 = read();
    int ans = Maxn;
    int floyd = n + m + 2;
    for (int i = 0; i < floyd; ++i)
      for (int j = 0; j < floyd; ++j) {
        int dist = Maxn;
        if (i == j) dist = 0;
        if (i)
          dist = std::min(dist, n);
        if (j)
          dist = std::min(dist, n);
        if (i % n == j % n)
          dist = std::min(dist, m);
        if (i % n + 1 == j % n && j < c_l)
          dist = std::min(dist, l[j]);
        if (i % n == j % n + 1 && j + 1 < c_l)
          dist = std::min(dist, l[j + 1]);
        if (i % n == j % n)
          dist = std::min(dist, m);
        if (i % n == j % n + v)
          dist = std::min(dist, e[j / n]);
        if (i % n + v == j % n)
          dist = std::min(dist, e[j / n + 1]);
        if (dist < Maxn)
          ans = std::min(ans, dist + 1);
      }
    printf("%d\n", ans);
  }
}

int main() { solve(); }
