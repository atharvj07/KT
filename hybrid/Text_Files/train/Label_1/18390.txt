#include <bits/stdc++.h>
using namespace std;

struct Ring
{
    int a, b, h;
    Ring() {}
    Ring(int a, int b, int h) : a(a), b(b), h(h) {}
};

bool cmp(const Ring &a, const Ring &b)
{
    return a.b > b.b;
}

int main()
{
    int n;
    cin >> n;
    vector<Ring> rings;
    for (int i = 0; i < n; i++)
    {
        int a, b, h;
        cin >> a >> b >> h;
        rings.push_back(Ring(a, b, h));
    }
    sort(rings.begin(), rings.end(), cmp);
    int ans = 0;
    map<int, int> dp;
    for (int i = 0; i < n; i++)
    {
        int cur = rings[i].h;
        for (int j = rings[i].b; j >= 1; j--)
        {
            if (dp.count(j) && rings[i].a > dp[j])
                cur = max(cur, rings[i].h + dp[j]);
        }
        dp[rings[i].b] = max(dp[rings[i].b], cur);
        ans = max(ans, cur);
    }
    cout << ans << endl;
    return 0;
}
