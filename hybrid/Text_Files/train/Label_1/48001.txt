import java.util.Scanner;

public class DistanceSum {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        // Precompute factorials and inverses for efficient modular arithmetic
        int[] factorials = new int[n + 1];
        int[] inverses = new int[n + 1];
        factorials[0] = 1;
        inverses[0] = 1;
        for (int i = 1; i <= n; i++) {
            factorials[i] = (int) ((long) factorials[i - 1] * i % MOD);
            inverses[i] = modInverse(factorials[i], MOD);
        }

        // Calculate the sum of shortest distances using the formula
        long sum = 0;
        for (int i = 1; i <= n - 1; i++) {
            for (int j = i + 1; j <= n; j++) {
                sum = (sum + (long) i * (n - j) * choose(n - 1, j - 1, factorials, inverses)) % MOD;
            }
        }

        System.out.println(sum);
    }

    // Modular inverse using Fermat's Little Theorem
    private static int modInverse(int a, int m) {
        return (int) Math.pow(a, m - 2) % m;
    }

    // Binomial coefficient calculation using modular arithmetic
    private static int choose(int n, int k, int[] factorials, int[] inverses) {
        return (int) ((long) factorials[n] * inverses[k] % MOD * inverses[n - k] % MOD);
    }

    private static final int MOD = 1000000007;
}
