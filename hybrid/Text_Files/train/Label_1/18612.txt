#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>

using namespace std;

typedef long long ll;
const ll INF = 1e18;

int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> a(k), b(k);
  for (ll i = 0; i < k; i++) {
    cin >> a[i] >> b[i];
    a[i]--;
  }
  vector<vector<pair<ll, ll>>> adj(n);
  for (ll i = 0; i < k; i++) {
    adj[a[i]].push_back({b[i], i});
  }
  vector<ll> dist(n, INF);
  priority_queue<pair<ll, ll>, vector<pair<ll, ll>>, greater<pair<ll, ll>>> pq;
  for (ll i = 0; i < k; i++) {
    if (b[i] >= 2) {
      pq.push({0, i});
    }
  }
  while (!pq.empty()) {
    ll d = pq.top().first;
    ll idx = pq.top().second;
    pq.pop();
    if (d != dist[idx]) continue;
    for (auto p : adj[idx]) {
      ll w = p.first;
      ll i = p.second;
      if (dist[a[i]] > d + 1) {
        dist[a[i]] = d + 1;
        pq.push({d + 1, i});
      }
    }
  }
  bool good = false;
  for (ll i = 0; i < n; i++) {
    if (dist[i] < INF) {
      good = true;
    }
  }
  if (good) cout << 1 << endl;
  else cout << -1 << endl;
  return 0;
}
