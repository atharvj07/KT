import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int xp0 = sc.nextInt(), yp0 = sc.nextInt(), xp1 = sc.nextInt(), yp1 = sc.nextInt();
            int xp2 = sc.nextInt(), yp2 = sc.nextInt(), xp3 = sc.nextInt(), yp3 = sc.nextInt();
            if (xp0 > xp3 || xp2 > xp1)
                System.out.println(0);
            else if (yp0 < yp1)
                System.out.println((yp2 > yp0 && yp3 > yp0) || (yp2 < yp1 && yp3 < yp1) ? 1 : 0);
            else
                System.out.println((yp1 > yp2 && yp0 > yp2) || (yp1 < yp3 && yp0 < yp3) ? 1 : 0);
        }
    }
}
