#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n, h;
        cin >> n >> h;
        vector<ll> arr(n);
        for (ll i = 0; i < n; i++)
            cin >> arr[i];
        ll k = h, prv = -1;
        for (const auto& x : arr) {
            if (x - prv >= k) {
                prv = x;
                k = h;
                continue;
            }
            k -= (prv + k - x % k) / k;
            prv = x;
        }
        cout << k << endl;
    }
    return 0;
}
