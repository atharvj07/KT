#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
int main() {
	int n;
	cin>>n;
	vector<int>a(n);
	for(int i=0;i<n;i++)cin>>a[i];
	vector<int> b = a;
	sort(b.begin(),b.end());
	int c = 0;
	for(int i=0;i<(n+1)/2;i++) b[i]=a[c++];
	for(int i=(n+1)/2;i<n;i++) b[i]=a[c++];
	for(int i=0;i<n;i++) cout<<b[i]<<' ';
	cout<<endl;
	return 0;
}
