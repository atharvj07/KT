import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long N = sc.nextLong();
        long M = sc.nextLong();
        long[] A = new long[(int) N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextLong();
        }

        PriorityQueue<Long> pq = new PriorityQueue<>();
        for (long a : A) {
            pq.offer(a);
        }

        long ans = 0;
        while (!pq.isEmpty()) {
            long a = pq.poll();
            if (M > 0) {
                a /= 2;
                M--;
            }
            ans += a;
        }

        System.out.println(ans);
    }
}
