#include <bits/stdc++.h>
using namespace std;
int n,x,k;
vector<long long> a;
int main(){
	cin>>n>>k>>x;
	a.resize(n);
	for(int i=0;i<n;i++) cin>>a[i];
	sort(a.begin(),a.end());
	int ans=1;
	for(int i=0,j=0;i<n;i++){
		ans+=int(a[i]-a[j]>x);
		while(i<n&&a[i]-a[j]<=x) i++;
		j=i;
	}
	if(k>0){
		cout<<ans-1;
		return 0;
	}
	cout<<ans;
}
