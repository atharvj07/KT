import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int K = sc.nextInt();
        Map<Integer, Set<Integer>> friendships = new HashMap<>();
        Map<Integer, Set<Integer>> blockships = new HashMap<>();
        for (int i = 0; i < M; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            friendships.computeIfAbsent(a, x -> new HashSet<>()).add(b);
            friendships.computeIfAbsent(b, x -> new HashSet<>()).add(a);
        }
        for (int i = 0; i < K; i++) {
            int c = sc.nextInt();
            int d = sc.nextInt();
            blockships.computeIfAbsent(c, x -> new HashSet<>()).add(d);
            blockships.computeIfAbsent(d, x -> new HashSet<>()).add(c);
        }
        int[] ans = new int[N + 1];
        for (int i = 1; i <= N; i++) {
            for (int j = 1; j <= N; j++) {
                if (i == j || friendships.containsKey(i) && friendships.get(i).contains(j) || blockships.containsKey(i) && blockships.get(i).contains(j)) {
                    continue;
                }
                Queue<Integer> q = new ArrayDeque<>();
                q.add(i);
                boolean[] visited = new boolean[N + 1];
                visited[i] = true;
                while (!q.isEmpty()) {
                    int u = q.poll();
                    if (u == j) {
                        ans[i]++;
                        break;
                    }
                    if (friendships.containsKey(u)) {
                        for (int v : friendships.get(u)) {
                            if (!visited[v]) {
                                q.add(v);
                                visited[v] = true;
                            }
                        }
                    }
                }
            }
        }
        for (int i = 1; i <= N; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
