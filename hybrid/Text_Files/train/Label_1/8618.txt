#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> f(n + 1);
    f[0] = 0;
    f[1] = 1;
    for (int i = 2; i <= n; ++i) {
        f[i] = f[i - 1] + 1;
        for (int j = 1; j * 2 <= i; ++j) {
            if (s.substr(0, j) == s.substr(i - j, j)) {
                f[i] = min(f[i], f[i - j] + 1);
            }
        }
    }
    cout << f[n] << endl;
    return 0;
}
