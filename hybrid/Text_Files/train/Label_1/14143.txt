#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    int n = s.size();
    if (s[0] != '0' || s[n - 1] != '0') {
        cout << -1 << endl;
        return 0;
    }
    vector<int> ans;
    int cnt = 0, l = 1;
    for (int i = 0; i < n; i++) {
        if (s[i] == '0')
            cnt++;
        else
            cnt--;
        if (cnt == 0) {
            ans.push_back(l);
            l = i + 2;
        }
    }
    cout << ans.size() << endl;
    for (auto a : ans) {
        cout << (a + 1) / 2 << " ";
        for (int i = 1; i <= (a + 1) / 2; i++) cout << l + i - 1 << " ";
        l += a;
        cout << endl;
    }
    return 0;
}
