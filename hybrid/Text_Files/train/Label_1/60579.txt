import sys
sys.setrecursionlimit(10000)

n = int(input())
p = list(map(int, input().split()))
ans = [[0] * n for _ in range(n)]
visited = [[False] * n for _ in range(n)]
di = [0, -1, 1, 0]
dj = [-1, 0, 0, 1]

def dfs(i, j, v):
    visited[i][j] = True
    ans[i][j] = v
    for d in range(4):
        ii = i + di[d]
        jj = j + dj[d]
        if 0 <= ii < n and 0 <= jj < n and not visited[ii][jj] and p[ii] == v and p[jj] == v:
            dfs(ii, jj, v)

for i in range(n):
    for j in range(n):
        if i >= j and not visited[i][j]:
            dfs(i, j, p[i])

if ans[n - 1][n - 1] == 0:
    print(-1)
else:
    for row in ans:
        print(*row)
