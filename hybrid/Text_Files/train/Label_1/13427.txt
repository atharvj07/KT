#include <iostream>
#include <vector>
#include <string>
using namespace std;
#define ll long long

int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll n;
    cin >> n;
    string s = to_string(n);
    int ok = 0;
    while (!ok) {
      ok = 1;
      for (auto &&i : s) {
        if (i != '0' && n % (i - '0')) ok = 0;
      }
      if (!ok) {
        s[s.size() - 1]++;
        for (int i = s.size() - 1; i > 0; i--) {
          if (s[i] > '9') {
            s[i] = '0';
            s[i - 1]++;
          }
        }
      }
    }
    cout << s << "\n";
  }
}
