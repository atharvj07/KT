import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[a[i] - 1] = i;
        }
        List<Pair> list = new ArrayList<>();
        int swaps = 0;
        for (int i = 0; i < n; i++) {
            int j = p[i];
            if (i != j) {
                swaps++;
                list.add(new Pair(i + 1, j + 1));
                int tmp = a[i];
                a[i] = a[j];
                a[j] = tmp;
                p[a[i] - 1] = i;
                p[a[j] - 1] = j;
            }
        }
        if (swaps == 0) {
            System.out.println(0);
        } else {
            System.out.println(swaps);
            for (Pair pair : list) {
                System.out.println(pair);
            }
        }
    }

    static class Pair {
        int u, v;

        public Pair(int u, int v) {
            this.u = u;
            this.v = v;
        }

        @Override
        public String toString() {
            return u + " " + v;
        }
    }
}
