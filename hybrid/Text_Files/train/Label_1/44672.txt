import java.util.*;

public class Main {

    public static void main(String args[]) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        long sum = 0;
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
            sum += a[i];
        }
        scanner.close();
        if (sum % 2 == 1) {
            System.out.println("First");
        } else {
            System.out.println("Second");
        }
    }
}
