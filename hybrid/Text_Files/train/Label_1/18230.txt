#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, m, cnt[105] = {}, sum[105] = {}, tmp[105] = {};
    cin >> n >> m;
    for (int i = 0, a; i < n; i++) {
        cin >> a;
        cnt[a]++;
    }
    for (int i = 1; i <= m; i++) { cin >> tmp[i]; }
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= m; j++) { sum[i] += tmp[j]; }
        if (cnt[i] < tmp[i] || sum[i] < cnt[i]) {
            cout << "NO";
            return 0;
        }
    }
    cout << "YES";
    return 0;
}
