#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;
int main() {
    int n;
    cin >> n;
    unordered_map<int, vector<int>> m;
    for (int i = 0; i < n; i++) {
        int r;
        cin >> r;
        vector<int> v(r);
        for (int j = 0; j < r; j++) {
            cin >> v[j];
            m[v[j]];
        }
        for (int j = 0; j < r; j++) m[v[j]].push_back(i);
    }
    for (auto & p : m) {
        if (p.second.size() == n) {
            cout << p.first << " ";
        }
    }
    return 0;
}
