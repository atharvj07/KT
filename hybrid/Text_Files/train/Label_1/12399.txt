#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<char, ll> pi;
int main() {
    ll n, k, ans = 0;
    map<string, pi> mp;
    string temp;
    cin >> k >> n;
    while (k--) {
        cin >> temp;
        ll a;
        cin >> a;
        string reversed = temp;
        reverse(reversed.begin(), reversed.end());
        if (temp == reversed)
            ans += max(a, mp[temp].second);
        else
            ans += max({0LL, a, mp[temp].second, mp[reversed].second});
        mp[temp].first = temp[0];
        mp[temp].second = max(mp[temp].second, a);
    }
    cout << ans << endl;
}
