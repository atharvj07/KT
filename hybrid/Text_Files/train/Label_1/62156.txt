import sys
input = sys.stdin.readline

n = int(input())
s = input().strip()
m = len(s)
msk = 0
for j in range(m):
    msk |= 1 << j

pw = [1]
for i in range(n):
    pw.append(pw[-1] * 2)

ans = 0
for j in range(m):
    for i in range(n):
        if (pw[i] & msk) == pw[j]:
            ans += 1

print(ans)
