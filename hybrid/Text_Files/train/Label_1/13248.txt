#include <bits/stdc++.h>
using namespace std;
int n, ans[305];
bitset<305> a[305];
void gen(int x){
    if(x>n){
        for(int i=1;i<=n;++i) cout<<ans[i]<<" ";
        cout<<'\n';
        exit(0);
    }
    vector<int> v;
    int c=0;
    for(int i=1;i<=n;++i){
        if(!a[x][i]) continue;
        if(!c){
            c=1;
            ans[x]=i;
        }
        else{
            v.push_back(i);
        }
    }
    for(auto &x:v) swap(ans[x],ans[c++]);
    gen(x+1);
    for(auto &x:v) swap(ans[x],ans[c++]);
    gen(x+1);
}
int main(){
    cin>>n;
    for(int i=1;i<=n;++i) cin>>ans[i];
    for(int i=1;i<=n;++i){
        string s; cin>>s;
        for(int j=1;j<=n;++j) if(s[j-1]=='1') a[i].set(j);
    }
    gen(1);
    return 0;
}
