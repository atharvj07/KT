#include <bits/stdc++.h>
using namespace std;
int main(){
    int n,k;
    cin>>n>>k;
    if(n>k||n&1) return cout<<"NO",0;
    cout<<"YES\n";
    vector<int> b(n),g(n);
    for(int i=0;i<n;i+=2){
        b[i]=i/2+1,g[i]=i/2+2;
    }
    for(int i=1;i<n;i+=2){
        b[i]=i/2+2,g[i]=i/2+1;
    }
    for(int x:b) cout<<x<<' ';
    cout<<'\n';
    for(int x:g) cout<<x<<' ';
}
