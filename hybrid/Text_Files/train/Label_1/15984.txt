#include <bits/stdc++.h>
#define ll long long
#define vi vector<int>
#define vll vector<ll>
#define pb push_back
using namespace std;
int main() {
  int n;
  cin >> n;
  vi a(n), b(n);
  for (int &i : a) cin >> i;
  for (int &i : b) cin >> i;
  sort(a.rbegin(), a.rend());
  sort(b.rbegin(), b.rend());
  ll sum = 0;
  ll cnt = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i];
    cnt += b[i];
  }
  ll l = 0, r = 1e9 + 1, ans = 1e9;
  while (l <= r) {
    ll mid = (l + r) / 2;
    if (sum <= mid * cnt)
      r = mid - 1, ans = mid;
    else
      l = mid + 1;
  }
  cout << ans * 1000 << '\n';
  return 0;
}
