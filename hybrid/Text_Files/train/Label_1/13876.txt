#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int k = 0, cnt = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] < 0) cnt++;
        if (cnt >= 3) {
            k++;
            cnt = 0;
        }
    }
    if (cnt > 0) k++;
    cout << k << endl;
    cnt = 0;
    for (int i = 0; i < n; i++) {
        cnt++;
        if (cnt == 1) cout << cnt << ' ';
        if (a[i] < 0) cnt = 0;
        if (cnt >= 3) {
            cnt = 0;
            cout << cnt << ' ';
        }
    }
    if (cnt > 0) cout << cnt << endl;
    return 0;
}
