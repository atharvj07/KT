def parse_comments(feed):
    stack = []
    current = ""
    num = ""
    depth = 0
    max_depth = 0
    comments_at_depth = {}
    
    for char in feed:
        if char == ",":
            if num == "":
                if current != "":
                    depth = len(stack) + 1
                    max_depth = max(max_depth, depth)
                    if depth not in comments_at_depth:
                        comments_at_depth[depth] = []
                    comments_at_depth[depth].append(current)
                    current = ""
            else:
                num = int(num)
                if num == 0:
                    while stack and stack[-1][1] == 1:
                        stack.pop()
                        depth -= 1
                    if stack:
                        stack[-1][1] -= 1
                else:
                    stack.append([current, num])
                    current = ""
                num = ""
        elif char.isdigit():
            num += char
        else:
            current += char
    
    print(max_depth)
    for depth in range(1, max_depth + 1):
        print(" ".join(comments_at_depth.get(depth, [])))

input_feed = input()
parse_comments(input_feed)
