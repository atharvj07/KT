import java.io.*;
import java.util.*;
import java.math.*;

public class Main {

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int d = in.nextInt(), n = in.nextInt(), m = in.nextInt();
    int[] p = new int[m], x = new int[m];
    for (int i = 0; i < m; i++) {
      x[i] = in.nextInt();
      p[i] = in.nextInt();
    }
    Arrays.sort(x); Arrays.sort(p);
    boolean ok = true;
    long ans = 0, cur = 0, cnt = n;
    for (int i = 0; i < m; i++) {
      if (cur < x[i]) {
        if (cnt * p[i] < ans) {
          ok = false;
          break;
        }
        long req = (x[i] - cur) / n;
        if ((x[i] - cur) % n > 0) req++;
        ans += p[i] * req;
        cnt -= req;
        cur = x[i];
      }
      cur += cnt;
    }
    if (ok && cur >= d) System.out.println(ans);
    else System.out.println(-1);
  }

}
