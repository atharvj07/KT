#include <iostream>
#include <vector>
using namespace std;

#define ll long long

int main() {
  ll n, m, k;
  cin >> n >> m >> k;
  vector<ll> p(m);
  for (ll i = 0; i < m; i++) { cin >> p[i]; }
  ll pp = 0, ans = 0;
  for (ll i = 0; i < m; i++) {
    while (pp < p[i] && pp + k < n) { pp += k; }
    if (pp >= p[i]) { ans++; }
    pp++;
  }
  cout << ans << endl;
  return 0;
}
