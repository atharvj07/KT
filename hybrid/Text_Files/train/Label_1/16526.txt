#include<bits/stdc++.h>
#define pb push_back
#define mp make_pair
#define all(x) (x).begin(), (x).end()
#define sz(x) (int)(x).size()
const int maxn = 4001;
using namespace std;
typedef long long LL;
typedef pair<int, int> PII;
const int Max = 1000000007;
int n, m, ans[maxn];
vector<int> adj[maxn];

void read_graph() {
	cin >> n >> m;
	for (int i = 0, u, v; i < m; i++) {
		cin >> u >> v;
		adj[u].pb(v), adj[v].pb(u);
	}
}

int bf(int s) {
	memset(ans, -1, sizeof ans);
	queue<int> q;
	q.push(s);
	int Step = 1;
	ans[s] = 0;
	while (!q.empty()) {
		int u = q.front();
		q.pop();
		for (int i = 0, v; i < adj[u].size(); i++) {
			v = adj[u][i];
			if (ans[v] == -1) {
				q.push(v);
				ans[v] = ans[u] + 1;
			}
			else if (ans[v] == ans[u]) return -1;
		}
		Step++;
	}
	return ans[s];
}

int validate() {
	vector<int> Arr;
	for (int i = 1; i <= n; i++) if (ans[i] >= 0) Arr.pb(i);
	if (Arr.size() != 3) return -1;
	int frt = Arr[0], sec = Arr[1], thd = Arr[2];
	if ((adj[frt][0] != sec and adj[frt][0] != thd) or
		(adj[sec][0] != frt and adj[sec][0] != thd) or
		(adj[thd][0] != frt and adj[thd][0] != sec)) return -1;
	return ans[1] + ans[2] + ans[3];
}

int main() {
	ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	read_graph();
	int res = bf(1);
	if (res != -1) {
		res = validate();
	}
	cout << res << "\n";
	return 0;
}
