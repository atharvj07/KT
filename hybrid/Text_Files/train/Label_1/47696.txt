import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long t = sc.nextInt();
        int[] x = new int[n], t_i = new int[n], p = new int[n], l = new int[n];
        for (int i = 0; i < n; i++)
            x[i] = sc.nextInt();
        for (int i = 0; i < n; i++)
            t_i[i] = sc.nextInt();
        for (int i = 1; i < n; i++) {
            p[i] = sc.nextInt();
            l[i] = sc.nextInt();
        }
        long[] dp = new long[n];
        dp[0] = x[0];
        for (int i = 1; i < n; i++) {
            if (t_i[i] <= l[i])
                dp[i] = Math.max(dp[i], dp[p[i]] + x[i]);
            dp[p[i]] = Math.max(dp[p[i]], dp[i] + x[p[i]]);
        }
        long ans = 0;
        for (int i = 0; i < n; i++)
            ans = Math.max(ans, dp[i]);
        System.out.println(ans);
    }
}
