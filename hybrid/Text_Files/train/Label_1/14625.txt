#define _CRT_SECURE_NO_WARNINGS
#include <algorithm>
#include <cstdio>
#include <cstring>

using namespace std;

typedef long long i64;
typedef pair<i64, i64> pi64;
typedef pair<pi64, i64> pi64pi64;

const i64 mod = 1e9 + 7;
const i64 maxn = 363304;
const i64 maxnum = 308983066;
i64 powmod(i64 a, i64 b) { i64 r = 1; for (; b; b >>= 1) { if (b & 1)r = (r*a) % mod; a = (a*a) % mod; }return r; }

struct data {
    i64 price, prv, nt;
    bool op, del;
    data() : price(0), prv(-1), nt(-1), op(false), del(false) {}
};

i64 n, id = 1, f[maxn + 1], invf[maxn + 1];
map<i64, i64> mp;
data a[maxn + 1];

bool cmp(i64 x, i64 y) {
    return a[x].price < a[y].price;
}

void preprocess() {
    f[0] = f[1] = invf[0] = invf[1] = 1;
    for (i64 i = 2; i <= maxn; ++i) {
        f[i] = i*f[i - 1];
        invf[i] = powmod(f[i], mod - 2);
        f[i] %= mod;
        invf[i] %= mod;
    }
}

i64 comb(i64 n, i64 m) {
    i64 x = f[n];
    x = x*invf[m];
    x %= mod;
    x = x*invf[n - m];
    x %= mod;
    return x;
}

int main() {
    preprocess();

    cin >> n;
    for (i64 i = 1; i <= n; i++) {
        string str;
        cin >> str >> a[i].price;
        if (str == "ADD")
            mp[a[i].price] = i;
        else {
            if (mp.count(a[i].price) == 0) {
                printf("%lld\n", 0);
                return 0;
            }
            a[mp[a[i].price]].op = false;
            a[mp[a[i].price]].del = true;
        }
    }

    set<i64> s;
    for (i64 i = 1; i <= n; i++)
        if (!a[i].del)
            s.insert(i);

    i64 ans = 1;
    while (!s.empty()) {
        vector<i64> v;
        for (set<i64>::iterator it = s.begin(); it != s.end();) {
            if (!a[*it].op)
                it = s.erase(it);
            else {
                v.push_back(*it);
                it++;
            }
        }
        if (v.size() <= 1) {
            ans = 0;
            break;
        }
        sort(v.begin(), v.end(), cmp);
        ans = ans*comb(v.size(), v.size() - 1);
        ans %= mod;
        for (i64 i = 0; i < v.size() - 1; i++) {
            a[v[i]].op = false;
            a[v[i]].prv = v[i + 1];
            a[v[i + 1]].op = true;
            a[v[i + 1]].nt = v[i];
        }
    }
    cout << ans;

    return 0;
}
