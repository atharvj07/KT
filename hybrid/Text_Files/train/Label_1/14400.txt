#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n < 14) {
      cout << "NO" << endl;
      continue;
    }
    n -= 10;
    if (n & 1) {
      cout << "NO" << endl;
    } else {
      cout << "YES" << endl;
      cout << 6 << " " << 7 << " " << n / 2 - 1 << " ";
      if (n % 4 == 0) {
        cout << n / 4 << endl;
      } else {
        cout << n / 4 + 1 << endl;
      }
    }
  }
  return 0;
}
