from collections import Counter
 
mod=998244353
 
n,k=map(int,input().split())
l=[0]*n
r=[0]*n
 
for i in range(n):
    x,y=map(int,input().split())
    l[i]=x-1
    r[i]=y
 
dic={}
for x in l+r:
    dic[x]=0
 
cnt=Counter(l+r)
 
ans=1
for x in dic:
    ans*=pow(cnt[x],cnt[x])
    ans%=mod
 
for x in dic:
    if cnt[x]>=k:
        ans*=pow(cnt[x]-k+1,cnt[x]-k+1)
        ans%=mod
 
print(ans)
