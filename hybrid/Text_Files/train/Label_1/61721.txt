import sys

input = sys.stdin.readline

n = int(input())
arr = list(map(int, input().split()))
graph = [[] for _ in range(n+1)]
for _ in range(n-1):
    u, v = map(int, input().split())
    graph[u].append(v)
    graph[v].append(u)

def dfs(node, par):
    sub[node] = arr[node-1]
    for u in graph[node]:
        if u == par: continue
        dfs(u, node)
        sub[node] += sub[u]

def ans(node, par):
    res[node] = sub[node]
    for u in graph[node]:
        if u == par: continue
        res[node] = max(res[node], sub[node] - 2*sub[u] + res[u])
        ans(u, node)

sub = [0] * (n+1)
res = [0] * (n+1)
dfs(1, 1)
ans(1, 1)
for i in range(1, n+1):
    print(res[i])
