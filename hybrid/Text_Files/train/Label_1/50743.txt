import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        char[] arr = s.toCharArray();
        int n = s.length();
        int[] lps = new int[n + 1];
        lps[0] = 0;
        lps[1] = 0;
        for (int i = 2; i <= n; i++) {
            int j = lps[i - 1];
            while (j > 0 && arr[j] != arr[i - 1]) {
                j = lps[j];
            }
            if (arr[j] == arr[i - 1]) {
                j++;
            }
            lps[i] = j;
        }
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (lps[i] > 0 && 2 * lps[i] <= i + 1) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
