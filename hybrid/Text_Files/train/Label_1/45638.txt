import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        int[] left = new int[n];
        int[] right = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt() - 1;
        }
        for (int i = 0; i < n; i++) {
            if (i == 0) {
                left[i] = -1;
            } else {
                left[i] = i - 1;
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            if (i == n - 1) {
                right[i] = -1;
            } else {
                right[i] = i + 1;
            }
        }
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            if (p[i] != i) {
                ans[i]++;
                if (left[i] != -1 && p[left[i]] < p[i]) {
                    ans[i]++;
                }
                if (right[i] != -1 && p[right[i]] < p[i]) {
                    ans[i]++;
                }
            }
        }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (ans[i] == 3) {
                cnt++;
            }
        }
        System.out.println(cnt);
    }
}
