#include <cstdio>
using namespace std;
typedef long long ll;
ll const mod = (ll)(1e9) + 7;
int main() {
  ll n, m, k;
  scanf("%lld %lld %lld", &n, &m, &k);
  if (n == 1 || m == 1) {
    printf("%d\n", k == -1);
  } else {
    ll ans = 2;
    if (k != -1)
      ans = (ans * 2) % mod;
    printf("%lld\n", ans);
  }
  return 0;
}
