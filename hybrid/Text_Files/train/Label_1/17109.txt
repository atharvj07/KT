#include <cstdint>
#include <cstdio>

template <typename T, typename U = long long int>
T fastpow(T base, U expt) {
    T ret = 1;
    while (expt > 0) {
        if (expt % 2 == 0) {
            base = base * base;
            expt /= 2;
        } else {
            ret = ret * base;
            expt -= 1;
        }
    }
    return ret;
}

int main() {
    long long int n, m, q;
    std::scanf("%lld %lld %lld", &n, &m, &q);

    int64_t maxN = fastpow<int64_t>(n, 18);
    n = maxN;
    int64_t maxM = fastpow<int64_t>(m, 18);
    m = maxM;

    while (q--) {
        int sx, sy, ex, ey;
        std::scanf("%d %d %d %d", &sx, &sy, &ex, &ey);

        bool ans = (sx != ex) || ((sx == 1 && 1 <= sy && sy <= n) || (sx == 2 && 1 <= sy && sy <= m)) ||
                    ((ex == 1 && 1 <= ey && ey <= n) || (ex == 2 && 1 <= ey && ey <= m));
        std::puts(ans ? "YES" : "NO");
    }
    return 0;
}
