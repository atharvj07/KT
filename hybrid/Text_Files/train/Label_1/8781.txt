#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  int t;
  cin >> t;
  while (t--) {
    ll n, k, a;
    cin >> n >> k;
    set<ll> s;
    for (ll i = 0; i < n; i++) {
      cin >> a;
      s.insert(a);
    }
    for (; k--;) {
      a = *s.begin();
      ll b = *s.rbegin(), c = (a + b + 1) / 2;
      if (!s.count(c)) s.insert(c);
    }
    cout << s.size() << '\n';
  }
  return 0;
}
