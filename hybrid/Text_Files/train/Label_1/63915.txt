from collections import deque, defaultdict

def is_valid_bfs(n, edges, sequence):
    graph = defaultdict(list)
    for x, y in edges:
        graph[x].append(y)
        graph[y].append(x)
    
    for key in graph:
        graph[key].sort()
        
    visited = [False] * (n + 1)
    queue = deque([1])
    visited[1] = True
    idx = 0
    
    while queue:
        cur_vertex = queue.popleft()
        if sequence[idx] != cur_vertex:
            return "No"
        idx += 1
        
        for neighbor in graph[cur_vertex]:
            if not visited[neighbor]:
                visited[neighbor] = True
                queue.append(neighbor)
                
    return "Yes"

n = int(input().strip())
edges = [list(map(int, input().strip().split())) for _ in range(n - 1)]
sequence = list(map(int, input().strip().split()))

print(is_valid_bfs(n, edges, sequence))
