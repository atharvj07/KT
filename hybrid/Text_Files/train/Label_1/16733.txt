#include <bits/stdc++.h>

using namespace std;

const int mxn = 14;

int main() {
  int stones[mxn];
  for (int i = 0; i < mxn; i++) {
    cin >> stones[i];
  }
  int ans = 0;
  for (int i = 0; i < mxn; i++) {
    if (stones[i] > 0) {
      int taken = stones[i];
      stones[i] = 0;
      int j = i + 1;
      while (taken > 0) {
        j %= mxn;
        stones[j]++;
        j++;
        taken--;
      }
      int score = 0;
      for (int k = 0; k < mxn; k++) {
        if (stones[k] % 2 == 0) {
          score += stones[k];
          stones[k] = 0;
        }
      }
      ans = max(ans, score);
    }
  }
  cout << ans << endl;
}
