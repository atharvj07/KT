#include <bits/stdc++.h>
#define int long long
using namespace std;
int minD(int x) {
  int ans = 9;
  while (x > 0) {
    if (x % 10 < ans) ans = x % 10;
    x /= 10;
  }
  return ans;
}
int maxD(int x) {
  int ans = 0;
  while (x > 0) {
    if (x % 10 > ans) ans = x % 10;
    x /= 10;
  }
  return ans;
}
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int a, k;
    cin >> a >> k;
    for (int i = 1; i < k; i++) {
      a += minD(a) * maxD(a);
    }
    cout << a << "\n";
  }
}
