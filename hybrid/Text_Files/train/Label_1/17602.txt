#include<bits/stdc++.h>
using namespace std;
const int N = 100 + 10, M = 100 + 10;
int n, m, q, par[N], col[N], ans[M];
vector<pair<int, int>>adj[N], vec[N];
bool vis[N];

int f(int x) { return par[x] == x ? x : par[x] = f(par[x]); }

void dfs(int u, int p) {
    vis[u] = 1, par[u] = u, col[u] = p;
    for (auto e: adj[u]) {
        int v, w = e.second;
        if (vis[v = e.first]) {
            if (col[u] != col[v]) ans[w] = 1;
        } else {
            dfs(v, p);
            par[f(v)] = f(u);
        }
    }
}

int main() {
    cin >> n >> m;
    for (int i = 0, u, v, w; i < m; ++i) {
        cin >> u >> v >> w;
        adj[u].push_back({v, w}), adj[v].push_back({u, w});
        vec[w].push_back({u, v});
    }
    for (int i = 1; i <= m; ++i) {
        if (!ans[i]) {
            for (auto e: vec[i]) {
                int u, v;
                tie(u, v) = e;
                if (!vis[u]) dfs(u, i);
                if (!vis[v]) dfs(v, i);
            }
        }
    }
    cin >> q;
    for (int i = 0, u, v; i < q; ++i) {
        cin >> u >> v;
        cout << ans[f(u) == f(v)] << endl;
    }
    return 0;
}
