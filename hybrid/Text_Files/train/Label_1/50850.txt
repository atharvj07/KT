import java.util.*;

class Main {

    static final int MOD = (int) 1e9 + 7;
    static long[][] dp, sum;

    static long pow(long b, long e) {
        if (e == 0) return 1;
        long h = pow(b, e / 2);
        h = (h * h) % MOD;
        if (e % 2 == 1) h = (h * b) % MOD;
        return h;
    }

    static long go(int i, int l, int r, String s, int[] a) {
        if (i == r + 1) return 1;
        if (dp[i][l] != -1) return dp[i][l];
        long ans = 0;
        for (int j = i; j <= r; j++) {
            if (j - i + 1 > a[s.charAt(j) - 'a']) break;
            if (j == r || sum[j + 1][r] <= a[s.charAt(j) - 'a']) {
                ans = (ans + go(j + 1, j + 1, r, s, a)) % MOD;
            }
        }
        return dp[i][l] = ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] a = new int[26];
        for (int i = 0; i < 26; i++) {
            a[i] = sc.nextInt();
        }
        dp = new long[n + 5][n + 5];
        sum = new long[n + 5][n + 5];
        for (long[] d : dp) Arrays.fill(d, -1);
        for (int l = n - 1; l >= 0; l--) {
            for (int r = l; r < n; r++) {
                if (l == r) sum[l][r] = 1;
                else sum[l][r] = (sum[l][r - 1] + 1) % MOD;
            }
        }
        long ans = go(0, 0, n - 1, s, a);
        int mx = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                if (j - i + 1 <= a[s.charAt(j) - 'a']) {
                    mx = Math.max(mx, j - i + 1);
                }
            }
        }
        int mn = (int) pow(2, n);
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                if (j - i + 1 <= a[s.charAt(j) - 'a']) {
                    mn = Math.min(mn, 1 + (int) pow(2, j - i + 1) - 1);
                }
            }
        }
        System.out.println(ans);
        System.out.println(mx);
        System.out.println(mn);
    }
}
