import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long m = in.nextInt(), n = in.nextInt(), ans = 1;
        while (n > 0) {
            if ((n & 1) == 1) {
                ans = (ans * m) % 1000000007;
            }
            m = (m * m) % 1000000007;
            n >>= 1;
        }
        System.out.println(ans);
    }
}
