from sys import stdin,stdout

n,q,c=map(int,stdin.readline().split())
stars=[[*map(int,stdin.readline().split())]for i in range(n)]
events=[[*map(int,stdin.readline().split())]for i in range(q)]

s=[[]for i in range(c+1)]
ans=[0 for i in range(q)]

for x,y,z in stars:
    s[z].append((x,y))

for t,x1,y1,x2,y2 in events:
    for v in range(c+1):
        for x,y in s[v]:
            if x1 <= x <= x2 and y1 <= y <= y2:
                ans[t]+=1

print(*ans,sep='\n')
