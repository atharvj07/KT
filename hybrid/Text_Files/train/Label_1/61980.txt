import sys

n = int(sys.stdin.readline())
row = []
for l in sys.stdin:
    for i in l.strip():
        v = int(i, 16)
        row.append(bin(v)[2:].zfill(4))
    if len(row) >= n**2: break
grid = [row[i:i+n] for i in range(0, len(row), n)]

for x in range(n, 0, -1):
    if n % x == 0:
        v = grid[0][0]
        for i in range(n//x):
            for j in range(n//x):
                if v != grid[i][j]:
                    break
            else:
                continue
            break
        else:
            print(x)
            break
