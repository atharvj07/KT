t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    first, last = -1, -1
    for i in range(n):
        if a[i] == 1:
            if first == -1:
                first = i
            last = i
    gap_count = 0
    for i in range(first, last + 1):
        if a[i] == 0:
            gap_count += 1
    print(gap_count)
