#include<bits/stdc++.h>
#define ll long long
#define plli pair<ll,ll>
using namespace std;
int main(){
	ll n,q; cin>>n>>q;
	vector<plli> blocks,pos;
	ll ans=0;
	bool block[3][100010]={0};
	while(q--){
		ll x,y; cin>>x>>y;
		if(block[x][y]) block[x][y]=0;
		else{
			if(x==1) pos.emplace_back(y,n);
			else pos.emplace_back(1,y);
			block[x][y]=1;
		}
		ll ok=0;
		for(auto &i: pos){
			if(block[1][i.first] && block[2][i.second]){
				ok=1;
				break;
			}
		}
		if(!ok) cout<<"Yes\n";
		else cout<<"No\n";
	}
	return 0;
}
