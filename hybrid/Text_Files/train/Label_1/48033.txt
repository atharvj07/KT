import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        int n, m;
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();

        for (int i = 0; i < n + m; i++) {
            if (i % 2 == 0 && n > 0) {
                System.out.print('B');
                n--;
            } else if (m > 0) {
                System.out.print('G');
                m--;
            }
        }

        System.out.println();
    }
}
