#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<string> files(n);
    for (int i = 0; i < n; ++i) { cin >> files[i]; }
    vector<int> toDel(m);
    for (int i = 0; i < m; ++i) {
        cin >> toDel[i];
        --toDel[i];
    }
    string pattern = files[toDel[0]];
    for (int i = 1; i < m; ++i) {
        for (int j = 0; j < min(pattern.size(), files[toDel[i]].size()); ++j) {
            if (pattern[j] != files[toDel[i]][j] && pattern[j] != '?') { pattern[j] = '?'; }
        }
    }
    if (pattern.size() == 100) {
        cout << "No" << endl;
        return 0;
    }
    for (int i = 0; i < n; ++i) {
        if (files[i].size() != pattern.size()) {
            cout << "No" << endl;
            return 0;
        }
        bool ok = true;
        for (int j = 0; j < files[i].size(); ++j) {
            if (pattern[j] != '?' && pattern[j] != files[i][j]) { ok = false; }
        }
        if (!ok) {
            cout << "No" << endl;
            return 0;
        }
    }
    cout << "Yes" << endl << pattern << endl;
}
