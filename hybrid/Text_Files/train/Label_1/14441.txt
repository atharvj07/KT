#include <bits/stdc++.h>
using namespace std;

void solve() {
  int n;
  cin >> n;
  vector<int> a(n + 2);
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) {
    if (a[i] > a[i - 1] && a[i] > a[i + 1]) {
      cout << i << endl;
      return;
    }
  }
  cout << -1 << endl;
}
int main() {
  int t;
  cin >> t;
  while (t--) { solve(); }
}
