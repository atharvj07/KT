import sys

def is_possible(n: int, m: int, v: list[int], t: list[int], edges: list[tuple[int, int]]) -> str:
    """
    Checks if it's possible to achieve the target values for each node in the graph by performing a series of operations.

    Args:
    n: The number of nodes in the graph.
    m: The number of edges in the graph.
    v: The list of initial values of the nodes.
    t: The list of target values of the nodes.
    edges: The list of edges in the graph.

    Returns:
    "YES" if it's possible to achieve the target values for each node, "NO" otherwise.
    """
    # Create a graph from the given data.
    graph = [[] for _ in range(n+1)]
    for edge in edges:
        graph[edge[0]].append(edge[1])
        graph[edge[1]].append(edge[0])

    # Perform a DFS to check if the graph is connected.
    visited = [False] * (n+1)
    def dfs(node: int):
        visited[node] = True
        for neighbor in graph[node]:
            if not visited[neighbor]:
                dfs(neighbor)

    dfs(1)

    # Check if all nodes are reachable from node 1.
    if not all(visited):
        return "NO"

    # Check if the sum of the differences between the initial values and the target values is 0.
    diff_sum = 0
    for i in range(1, n+1):
        diff_sum += (t[i] - v[i])

    if diff_sum != 0:
        return "NO"

    # Find the maximum and minimum differences between the initial values and the target values.
    max_diff = -sys.maxsize
    min_diff = sys.maxsize
    for i in range(1, n+1):
        diff = t[i] - v[i]
        max_diff = max(max_diff, diff)
        min_diff = min(min_diff, diff)

    # Check if the maximum and minimum differences are of opposite signs.
    if max_diff * min_diff < 0:
        return "NO"

    # Return "YES" if it's possible to achieve the target values for each node, "NO" otherwise.
    return "YES"

t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    v = list(map(int, input().split()))
    t = list(map(int, input().split()))
    edges = []
    for _ in range(m):
        a, b = map(int, input().split())
        edges.append((a, b))
    print(is_possible(n, m, v, t, edges))
