#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2000, MAXM = 2000;
int dp[MAXN][MAXM];
int main() {
    int n, m, k;
    cin >> n >> m >> k;
    int dx[] = {0, 0, 1, -1};
    int dy[] = {1, -1, 0, 0};
    char a[MAXN][MAXM];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++) cin >> a[i][j];
    for (int i = n - 2; i >= 0; i--) {
        for (int j = 0; j < m; j++) {
            dp[i][j] = 1e9;
            for (int k = 0; k < 4; k++) {
                int cur_i = i + dx[k], cur_j = j + dy[k];
                if (cur_i < n && cur_i >= 0 && cur_j < m && cur_j >= 0)
                    dp[i][j] = min(dp[i][j], 1 + dp[cur_i][cur_j]);
            }
            if (a[i][j] != '.') dp[i][j] = 0;
        }
    }
    for (int i = 0; i < m; i++) cout << dp[0][i] << " ";
    cout << endl;
    return 0;
}
