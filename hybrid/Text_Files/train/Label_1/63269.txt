n=int(input())
s=input()
ans=0
cnt=[0]*3
for c in s:
    if c!='?':
        cnt[ord(c)-ord('a')]+=1
for i in range(3):
    for j in range(3):
        for k in range(3):
            if i!=j and j!=k and i!=k:
                ans=(ans+cnt[i]%1000000007*cnt[j]%1000000007*cnt[k]%1000000007)%1000000007
print(ans)
