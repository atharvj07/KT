#include <cstdio>
#include <vector>
#include <cmath>
#include <algorithm>

const double PI = std::acos(-1.0);
const double eps = 1e-6;
using namespace std;

int main() {
    int n;
    scanf("%d", &n);

    vector<double> r(n), h(n);
    for (int i = 0; i < n; i++) {
        scanf("%lf %lf", &r[i], &h[i]);
    }

    vector<double> dp(n, 0.0);
    for (int i = 0; i < n; i++) {
        dp[i] = PI * r[i] * r[i] * h[i];
        for (int j = 0; j < i; j++) {
            if (r[j] < r[i]) {
                dp[i] = max(dp[i], dp[j] + PI * r[i] * r[i] * h[i]);
            }
        }
    }

    double ans = *max_element(dp.begin(), dp.end());
    printf("%.10f\n", ans);

    return 0;
}
