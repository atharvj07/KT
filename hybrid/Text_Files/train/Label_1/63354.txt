n = int(input())
lst = list(map(int, input().split()))
m = int(input())
g = [[] for _ in range(n+1)]
for _ in range(m):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)
    
st = [1]
vis = [0]*(n+1)
ans = 10**18
mn = 10**18
ways = 0
while st:
    t = st.pop(0)
    vis[t] = 1
    if ans > lst[t-1]:
        ans = lst[t-1]
        ways = 1
    elif ans == lst[t-1]:
        ways += 1
        
    for nxt in g[t]:
        if not vis[nxt]:
            st.append(nxt)
        
print(ans, ways)
