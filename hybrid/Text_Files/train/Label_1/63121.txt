n = int(input())
salaries = [list(map(int, input().split())) for _ in range(n)]
unique_salaries = sorted({salaries[i][j] for i in range(n) for j in range(n) if i != j})

# Map each unique salary to a new employee number starting from n+1
salary_to_employee = {sal: idx + n + 1 for idx, sal in enumerate(unique_salaries)}

# Initialize the list of employees with their salaries, starting with lower-level employees
employees = [salaries[i][i] for i in range(n)] + unique_salaries

# Find the head of the company (the employee with the highest salary)
head = salary_to_employee[unique_salaries[-1]]

# Initialize the structure of the company
company_structure = []

for i in range(n):
    for j in range(i + 1, n):
        # For each pair of lower-level employees, find their direct supervisor
        direct_supervisor_salary = salaries[i][j]
        direct_supervisor = salary_to_employee[direct_supervisor_salary]

        # Add the pair (lower-level employee, supervisor) to the company structure
        company_structure.append((i + 1, direct_supervisor))
        company_structure.append((j + 1, direct_supervisor))

# Remove duplicates from the company structure by converting it to a set of tuples
company_structure = list(set(company_structure))

print(len(employees))
print(' '.join(map(str, employees)))
print(head)
for v, u in company_structure:
    print(v, u)
