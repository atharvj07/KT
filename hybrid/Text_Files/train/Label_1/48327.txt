import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] win = new int[n + 1];
        int[][] power = new int[k][n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < k; j++) {
                power[j][i] = sc.nextInt();
            }
        }
        for (int i = n; i > 0; i--) {
            int winner = -1;
            for (int j = 0; j < k; j++) {
                if (winner == -1 || power[j][i] > power[j][winner]) {
                    winner = i;
                }
            }
            win[i] = winner;
            for (int j = 0; j < k; j++) {
                if (power[j][i] > power[j][winner]) {
                    power[j][winner] = power[j][i];
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            System.out.println(win[i]);
        }
    }
}
