#include <bits/stdc++.h>
using namespace std;
long double n, m, x, d, ans, t, a, b;
int main() {
  cin >> n >> m;
  while (m--) {
    cin >> x >> d;
    t += x * n + d * (n * (n + 1) / 2);
  }
  ans = t / (n * n);
  if (ans < 1e-9)
    a = 0;
  else
    a = (int)ans;
  b = ans - a;
  if (b < 1e-9)
    cout << (long long)a << ".000000000000000" << endl;
  else
    cout << fixed << setprecision(15) << ans << endl;
}
