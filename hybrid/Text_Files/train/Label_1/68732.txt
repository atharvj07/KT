t = int(input())
for _ in range(t):
    n, m, x, y = map(int, input().split())
    total_cost = 0
    for i in range(n):
        row = input()
        j = 0
        while j < m:
            if row[j] == '.':
                # Check if we can use 1x2 tile and it's cheaper or not
                if j+1 < m and row[j+1] == '.' and y < 2*x:
                    total_cost += y
                    j += 2
                else:
                    total_cost += x
                    j += 1
            else:
                j += 1
    print(total_cost)
