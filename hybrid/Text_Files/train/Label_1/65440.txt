inf = 10**9
for _ in range(int(input())):
    n, x = map(int, input().split())
    t = [(int(d),int(h)) for d, h in [input().split() for _ in range(n)]]
    t.sort(key=lambda x:x[0]-x[1])
    ans = inf
    cur = 0
    for d, h in t:
        if x<=0: break
        y = (x+d-1)//d
        cur+=y
        x-=y*d-h
        ans = min(ans, cur)
    if x<=0: print(ans)
    else: print(-1)
