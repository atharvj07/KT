import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    long a = input.nextLong();
    long b = input.nextLong();

    if (a == b) {
      System.out.println(0);
      return;
    }

    long operations = 0;
    while (a != b) {
      if (a > b) {
        if (a % 2 == 0) {
          a /= 2;
          operations++;
        } else if (a % 3 == 0) {
          a = 2 * a / 3;
          operations++;
        } else if (a % 5 == 0) {
          a = 4 * a / 5;
          operations++;
        } else {
          System.out.println(-1);
          return;
        }
      } else {
        if (b % 2 == 0) {
          b /= 2;
          operations++;
        } else if (b % 3 == 0) {
          b = 2 * b / 3;
          operations++;
        } else if (b % 5 == 0) {
          b = 4 * b / 5;
          operations++;
        } else {
          System.out.println(-1);
          return;
        }
      }
    }

    System.out.println(operations);
  }
}
