#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  int odd = 0, sum = 0;
  for (int i = 0; i < n; ++i) {
    odd += a[i] % 2;
    sum += a[i];
  }
  if (sum % 2 || !odd) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  return 0;
}
