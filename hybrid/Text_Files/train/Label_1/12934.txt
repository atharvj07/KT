#include <cstdio>
#include <algorithm>
using namespace std;

#define mod 1000000007

int n,k,m;
long long a[100010],f[100010];

int main(){
	scanf("%d%d%d",&n,&k,&m);
	for(int i=0;i<n;++i)scanf("%lld",&a[i]),++f[a[i]%m];
	sort(f+1,f+m+1);
	if(k<=f[1] || 1ll*f[1]*2>=k){
		printf("Yes\n");
		for(int i=0;i<k;++i)printf("%lld ",a[i]);
	}else if(m%2==1 && 1ll*f[1]*2==k+1 && f[m/2]>=1){
		printf("Yes\n");
		for(int i=0;i<f[1];++i)printf("%lld ",a[i]);
		for(int i=m/2;i!=0 && f[i]>=1;--i)printf("%lld ",i);
	}else if(m%2==0 && 1ll*f[1]*2==k+1 && f[m/2]>=2){
		printf("Yes\n");
		for(int i=0;i<f[1];++i)printf("%lld ",a[i]);
		for(int i=m/2;i!=0 && f[i]>=2;--i)printf("%lld ",i);
	}else printf("No");
}
