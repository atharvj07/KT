#include <iostream>
#include <vector>
#include <climits>

using namespace std;

typedef long long ll;

int main() {
    ll n, m; cin >> n >> m;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
    }

    ll ans = 0;
    for (ll i = 1; i < n; i++) {
        ans = max(ans, a[i] - a[i - 1]);
    }

    for (ll i = 0; i <= n; i++) {
        ll l = (i == 0 ? 0ll : a[i - 1]);
        ll r = (i == n ? m : a[i]);
        ans = max(ans, (r - l) / 2 + (r - l) % 2);
    }

    cout << ans << endl;

    return 0;
}
