#include <iostream>
#include <vector>
#include <cstring>
using namespace std;

int main() {
    int n, m; cin >> n >> m;
    vector<bool> a(m + 1, true);
    while (n--) {
        int l, r; cin >> l >> r;
        for (int i = l; i <= r; ++i) a[i] = false;
    }
    int ans = 0;
    for (int i = 1; i <= m; ++i) if (a[i]) ++ans;
    cout << ans << endl;
    for (int i = 1; i <= m; ++i) if (a[i]) cout << i << ' ';
    cout << endl;
    return 0;
}
