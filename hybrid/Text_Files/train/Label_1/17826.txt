#include <bits/stdc++.h>
using namespace std;

const int MX = 50;

int n, a[MX], b[MX], pre[MX];
bool vis[MX];

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        sort(a, a + n);
        for (int i = 0; i < n; i++) { vis[i] = 0; }
        int sum = 0;
        bool ok = 1;
        for (int i = 0; i < n; i++) {
            if (!vis[i]) {
                int cnt = 0;
                for (int j = i; j < n; j++) {
                    if (a[j] == a[i]) {
                        vis[j] = 1;
                        cnt++;
                    }
                }
                sum += cnt;
                if (sum == 0) {
                    ok = 0;
                    break;
                }
                b[i] = a[i];
                pre[i + 1] = pre[i] + cnt;
            }
        }
        if (!ok) {
            cout << "NO" << endl;
            continue;
        }
        cout << "YES" << endl;
        for (int i = 0; i < n; i++) {
            for (int j = pre[i]; j < pre[i + 1]; j++) { cout << b[i] << " "; }
        }
        cout << endl;
    }
}
