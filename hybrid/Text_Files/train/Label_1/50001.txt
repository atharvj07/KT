import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

    String[] split = br.readLine().split(" ");
    int k = Integer.parseInt(split[0]), n = Integer.parseInt(split[1]), t = Integer.parseInt(split[3]), maxb = Integer.parseInt(split[2]);

    int[][] b = new int[k][n];
    for (int i = 0; i < k; i++) {
      String[] strs = br.readLine().split(" ");
      for (int j = 0; j < n; j++) {
        b[i][j] = Integer.parseInt(strs[j]);
      }
    }

    int[] res = new int[k];
    for (int idx = 0; idx < k; idx++) {
      int[] length = new int[n * t];
      for (int i = 0; i < n * t; i++) {
        length[i] = 1;
      }

      for (int j = 1; j < n * t; j++) {
        for (int i = 0; i < j; i++) {
          if (b[idx][i % n] < b[idx][j % n] && length[j] < length[i] + 1) {
            length[j] = length[i] + 1;
          }
        }
      }

      int ans = 0;
      for (int x : length) {
        ans = Math.max(ans, x);
      }
      res[idx] = ans;
    }

    for (int x : res) {
      System.out.println(x);
    }
  }
}
