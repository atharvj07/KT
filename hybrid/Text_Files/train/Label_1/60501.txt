from collections import Counter
n, m = map(int, input().split())
a = list(map(int, input().split()))
c = Counter(a)
mn = n // m
ans = 0
for r in range(1, m):
    if r not in c:
        ans += mn
    elif c[r] < mn:
        ans += mn - c[r]
print(ans)
for i in range(n):
    r = a[i] % m
    if mn == c[r] or r == 0 and c[0] > mn:
        print(a[i], end=' ')
    else:
        a[i] += m - r
        c[r] -= 1
        c[r % m] += 1
        print(a[i], end=' ')
