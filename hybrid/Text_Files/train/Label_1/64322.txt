import sys

x, y, x0, y0 = map(int, sys.stdin.readline().split())
s = sys.stdin.readline()

grid = [[False]*y for _ in range(x)]
grid[x0-1][y0-1] = True

ans = [0]*(len(s)+1)
cx, cy = x0-1, y0-1
for i, c in enumerate(s):
    if c=='U':
        cx -= 1
        if cx < 0 or grid[cx][cy]:
            ans[i+1] += ans[i]
            break
    elif c=='D':
        cx += 1
        if cx >= x or grid[cx][cy]:
            ans[i+1] += ans[i]
            break
    elif c=='L':
        cy -= 1
        if cy < 0 or grid[cx][cy]:
            ans[i+1] += ans[i]
            break
    elif c=='R':
        cy += 1
        if cy >= y or grid[cx][cy]:
            ans[i+1] += ans[i]
            break

    grid[cx][cy] = True
    ans[i+1] += ans[i]+1

print(*ans)
