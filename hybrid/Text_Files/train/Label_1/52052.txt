import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String singular = scanner.nextLine();
        String plural;
        if (singular.endsWith("s")) {
            plural = singular + "es";
        } else {
            plural = singular + "s";
        }
        System.out.println(plural);
    }
}
