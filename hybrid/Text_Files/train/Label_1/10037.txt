#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll b[100005];
ll f[100005];
ll n,d,m,s;
int main(){
    cin>>n>>d;
    for(ll i=0;i<n;i++){
        cin>>m>>s;
        b[i]=m;
        f[i]=s;
    }
    sort(b,b+n);
    ll ans=0,j=0,k=0;
    while(j<n&&k<n){
        if(b[k]-b[j]<=d){
            ans+=f[k];
            k++;
        }
        else{
            j++;
        }
    }
    cout<<ans<<endl;
    return 0;
}
