import sys

input = sys.stdin.readline

n,m = map(int,input().split())
a = list(map(int,input().split()))
b = list(map(int,input().split()))
q = int(input())
ab = []
for i in range(q):
    x,y,z = map(int,input().split())
    ab.append([x,y,z])

aa = sorted(list(set(a)))
idx=[0]*n
dis=[False]*n
ans=[0]*(q+1)
for i in range(q):
    x,y,z = ab[i]
    if x==1:
        a[y-1]=z
        aa = sorted(list(set(a)))
        for j in range(n):
            if a[j]==aa[-1] and not dis[j]:
                idx[j]=m-1
                dis[j]=True
                break
    else:
        b[y-1]=z
        idx[0]=max(idx[0],y-1)
        for j in range(n):
            if not dis[j]:
                for k in range(idx[j],m):
                    if z<b[k]:
                        idx[j]=k-1
                        break
                if idx[j]==m-1:
                    dis[j]=True
    ans[i+1]=aa[-1] if any(not t for t in dis) else -1
    print(ans[i+1])
print(ans[q])
