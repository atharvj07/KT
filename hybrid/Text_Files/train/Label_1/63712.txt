def dist(s, t):
    ans = 0
    for a, b in zip(s, t):
        ans += abs(ord(a) - ord(b))
    return ans
n, k = map(int, input().split())
s = input()
if k < n - 1 or k > 25 * n:
    print(-1)
elif k == n - 1:
    for i in range(n):
        if s[i] != 'z':
            s = s[:i] + chr(ord(s[i]) + 1) + s[i + 1:]
            break
    print(s)
else:
    for i in reversed(range(n)):
        if k >= 25 - ord('z') - ord(s[i]):
            k -= 25 - ord('z') - ord(s[i])
            s = s[:i] + chr(ord(s[i]) + 1) + s[i + 1:]
        elif k > 0:
            k -= ord('z') - ord(s[i])
            s = s[:i] + 'a' + s[i + 1:]
    print(s)
