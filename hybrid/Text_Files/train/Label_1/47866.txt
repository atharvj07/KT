import java.util.Scanner;

public class Main {
    static int n;
    static int[] arr;
    static int[] ans;
    static boolean[] vis;
    static int[][] g;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        arr = new int[n + 1];
        ans = new int[n + 1];
        vis = new boolean[n + 1];
        g = new int[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            arr[i] = sc.nextInt();
        }
        for (int i = 1; i < n; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            g[u][v] = g[v][u] = 1;
        }
        dfs(1, 1);
        for (int i = 1; i <= n; i++) {
            System.out.print(ans[i] + " ");
        }
    }

    public static void dfs(int cur, int fa) {
        vis[cur] = true;
        int sum = arr[cur];
        for (int i = 1; i <= n; i++) {
            if (g[cur][i] == 1 && !vis[i]) {
                dfs(i, cur);
                sum += ans[i];
            }
        }
        ans[cur] = Math.max(sum, -sum);
    }
}
