#include <iostream>
#include <vector>

int main() {
  int q;
  std::cin >> q;
  std::vector<int> arr;
  std::vector<int> last_occ(500002, -1);
  while (q--) {
    int type;
    std::cin >> type;
    if (type == 1) {
      int x;
      std::cin >> x;
      arr.push_back(x);
      last_occ[x] = arr.size() - 1;
    } else {
      int x, y;
      std::cin >> x >> y;
      for (int i = last_occ[x]; i >= 0; i--) {
        arr[i] = y;
        last_occ[x] = i;
      }
    }
  }
  for (auto x : arr) std::cout << x << ' ';
  return 0;
}
