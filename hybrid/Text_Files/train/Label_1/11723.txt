#include <bits/stdc++.h>
#define MAXIMUM 11
using namespace std;
int n, m, k, w;
char grid[MAXIMUM][MAXIMUM];
int main() {
  cin >> n >> m >> k >> w;
  for (int i = 1; i <= k; i++) {
    for (int j = 1; j <= n; j++) {
      for (int l = 1; l <= m; l++) { cin >> grid[j][l]; }
      cin.ignore();
    }
    if (i == 1) {
      cout << n * m * w << endl;
      cout << 1 << " " << 0 << endl;
    } else {
      int minAns = 1e9, ans = 0;
      for (int j = 1; j <= n; j++) {
        for (int l = 1; l <= m; l++) {
          if (grid[i][l] != grid[i - 1][l]) minAns = min(minAns, 1);
        }
      }
      if (minAns == 1e9) minAns = 0;
      cout << minAns * w << endl;
      cout << i << " " << i - 1 << endl;
    }
  }
}
