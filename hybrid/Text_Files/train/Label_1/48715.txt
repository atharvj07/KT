import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		int tests = s.nextInt();
		for (int t = 0; t < tests; t++) {
			char[] a = s.next().toCharArray();
			int[] odd = new int[a.length];
			int[] even = new int[a.length];
			int o = 0;
			int e = 0;
			for (int i = a.length - 1; i >= 0; i--) {
				int d = a[i] - '0';
				if (d % 2 == 0) {
					even[i] = e;
					e++;
				} else {
					odd[i] = o;
					o++;
				}
			}
			for (int i = 0; i < a.length; i++) {
				int d = a[i] - '0';
				if (d % 2 == 0) {
					if (odd[i] > 0) {
						int min = odd[i] - 1;
						if (min < e) {
							a[i] = (char)('0' + d);
							a[i + min] = '0';
							e--;
							o--;
						}
					}
				} else {
					if (even[i] > 0) {
						int min = even[i] - 1;
						if (min < o) {
							a[i] = (char)('0' + d);
							a[i + min] = '0';
							e--;
							o--;
						}
					}
				}
			}
			int leading = 0;
			while (leading < a.length && a[leading] == '0') {
				leading++;
			}
			for (int i = leading; i < a.length; i++) {
				System.out.print(a[i]);
			}
			System.out.println();
		}
	}
}
