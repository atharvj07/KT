import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] b = new int[n];
            for (int i = 0; i < n; i++) {
                b[i] = sc.nextInt();
            }
            int[] a = new int[n];
            int idx = 0;
            int l = 0, r = n - 1;
            boolean left = true;
            for (int i = 0; i < n; i++) {
                if (left) {
                    a[l++] = b[idx++];
                } else {
                    a[r--] = b[idx++];
                }
                left = !left;
            }
            for (int i = 0; i < n; i++) {
                System.out.print(a[i] + " ");
            }
            System.out.println();
        }
    }
}
