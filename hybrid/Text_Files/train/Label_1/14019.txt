#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 1e5 + 2;
int n, k, sz[MAXN];
vector<int> adj[MAXN];

bool dfs(int u, int p, int d) {
  if (d > k && (int)adj[u].size() > 1) return false;
  if (d == k + 1 && (int)adj[u].size() != (int)adj[p].size() + 1) return false;
  sz[u] = 1;
  for (int v : adj[u]) {
    if (v != p && !dfs(v, u, d + 1)) return false;
    sz[u] += sz[v];
  }
  return true;
}

int main() {
  cin >> n >> k;
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  bool valid = dfs(1, 1, 0);
  int deg = 0, vtx = 0;
  for (int i = 1; i <= n; i++) {
    if ((int)adj[i].size() > 1) { deg++, vtx = i; }
  }
  if (deg != 1 || k == 0 || (k == 1 && deg != 0) || (k > 1 && deg != 2) || (k == 1 && sz[vtx] < 3))
    valid = false;
  cout << (valid ? "Yes" : "No") << '\n';
}
