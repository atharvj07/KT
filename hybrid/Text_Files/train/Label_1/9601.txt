#include <bits/stdc++.h>
const int mod = 998244353;
int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) std::cin >> a[i];
  std::vector<long long> dp(n + 1);
  dp[n] = 1;
  for (int i = n - 1; i >= 0; i--) {
    for (int j = 1; j <= a[i]; j++) {
      if (i + 1 < n && j == a[i + 1]) continue;
      dp[i] = (dp[i] + dp[i + 1]) % mod;
    }
  }
  std::cout << dp[0] << '\n';
}
