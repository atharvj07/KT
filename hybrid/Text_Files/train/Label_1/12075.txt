#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    vector<int> copy_a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; copy_a[i] = a[i]; }
    sort(a.begin(), a.end());
    int ans = 0;
    while (a != copy_a) {
      copy_a = a;
      for (int i = n - 1; i >= 0; i--) {
        int x = a[i];
        vector<int> left, right;
        for (int j = 0; j < n; j++) {
          if (a[j] > x) right.push_back(a[j]);
          else left.push_back(a[j]);
        }
        a.clear();
        for (int j = 0; j < left.size(); j++) a.push_back(left[j]);
        for (int j = 0; j < right.size(); j++) a.push_back(right[j]);
      }
      ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
