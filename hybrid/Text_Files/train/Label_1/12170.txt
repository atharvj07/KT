#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long n;
  cin >> n;
  vector<long long> a(n);
  for (long long i = 0; i < n; i++) { cin >> a[i]; }
  long long res = 0;
  for (long long i = 0; i < n; i++) {
    long long cur = 0;
    for (long long j = 0; j < n; j++) {
      if (a[i] % a[j] == 0) { cur += a[j]; }
    }
    res = max(res, cur);
  }
  cout << res << endl;
  return 0;
}
