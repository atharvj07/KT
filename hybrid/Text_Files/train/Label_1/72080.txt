import heapq
n=int(input())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
vis=[0]*n
h=[]
heapq.heappush(h,(-a[0]-b[0],0))
while h:
    cur,pos=heapq.heappop(h)
    if pos==0:
        print(-cur)
        break
    vis[pos]=1
    for i in range(pos+1,min(n,pos+a[pos]+1)):
        if not vis[i] and pos+b[pos]<=i:
            heapq.heappush(h,(-a[i]-b[i],i))
else:print(-1)
