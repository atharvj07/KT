#include<bits/stdc++.h>
using namespace std;
long long ans(long long n, long long k, long long *a, long long *b){
  vector<pair<long long, long long>>v;
  for(int i=0;i<n;i++){
    v.push_back({a[i],b[i]});
  }
  sort(v.begin(),v.end());
  long long sum_a=0, sum_b = 0;
  for(int i=0;i<n;i++){
    long long x = v[i].first;
    long long y = v[i].second;
    sum_a += x;
    sum_b += y;
    if(sum_a>k || (sum_b+1)*(n-i) > k) return i;
  }
  return n;
}
int main(){
  long long n, k;
  cin>>n>>k;
  long long a[n], b[n];
  for(int i=0;i<n;i++){
    cin>>a[i];
  }
  for(int i=0;i<n;i++){
    cin>>b[i];
  }
  cout<<ans(n, k, a, b)<<endl;
  return 0; 
}
