s=input()
a=[0]*len(s)
b=[0]*len(s)
ans1=[0]*(len(s)+2)
ans2=[0]*(len(s)+2)
for i in range(len(s)-1,-1,-1):
    if i==len(s)-1 or s[i]!=s[i+1]:a[i]=1
    else:a[i]=a[i+1]+1
for i in range(len(s)-1,-1,-1):
    if i==len(s)-1 or s[i]!=s[i+1]:b[i]=1
    else:b[i]=b[i+1]+1
for i in range(len(s)):
    if s[i]==s[i-1]:
        ans1[i+1]=a[i]-1
    else:ans1[i+1]=a[i]
ans1[len(s)]=a[len(s)-1]
for i in range(len(s)):
    if s[i]==s[i-1]:
        ans2[i+1]=b[i]-1
    else:ans2[i+1]=b[i]
ans2[len(s)]=b[len(s)-1]
print(*ans1[1:],*ans2[1:])
