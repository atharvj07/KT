#include <bits/stdc++.h>

using namespace std;

int main() {
    long long n, m, k;
    cin >> n >> m >> k;
    vector<long long> p(m);
    for (long long i = 0; i < m; i++) cin >> p[i];
    long long ans = 0;
    for (long long i = 0, j = 0; i < n; i++) {
        if (j < m && p[j] == i + 1) {
            j++;
            ans++;
            while (i % k && i < n) i++;
        }
    }
    cout << ans << endl;
    return 0;
}
