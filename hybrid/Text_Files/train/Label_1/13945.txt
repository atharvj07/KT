#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> pos(k + 1);
  for (int i = 1; i <= k; i++) {
    int x;
    cin >> x;
    pos[x] = i;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int sum = 0;
    set<int> s;
    for (int j = 0; j < m; j++) {
      int x;
      cin >> x;
      if (s.count(x) == 0) {
        sum += pos[x];
        s.insert(x);
      }
      pos[x] = 1;
      for (int l = 2; l <= k; l++) {
        if (pos[l] < pos[l - 1]) pos[l]++;
      }
    }
    ans += sum;
  }
  cout << ans << endl;
  return 0;
}
