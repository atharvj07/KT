t = int(input())
for _ in range(t):
    x, y = map(int, input().split())
    c = list(map(int, input().split()))

    # Convert the problem into minimal steps in three directions
    # (right, right-up, left-up) instead of six.
    # This simplification is possible due to the hexagonal grid structure.
    # Adjust x, y to account for the hexagonal grid's coordinate system.
    if x >= 0 and y >= 0:
        ans = x * c[0] + y * c[1] - min(x, y) * min(c[0]+c[1], c[2])
    elif x < 0 and y < 0:
        ans = -x * c[3] - y * c[4] + min(x, y) * min(c[3]+c[4], c[5])
    elif x >= 0 and y < 0:
        ans = x * c[0] - y * c[4] - min(0, x+y) * min(c[0]+c[4], c[5], c[3])
    else:  # x < 0 and y >= 0
        ans = -x * c[3] + y * c[1] - min(0, x+y) * min(c[3]+c[1], c[2], c[0])

    print(ans)
