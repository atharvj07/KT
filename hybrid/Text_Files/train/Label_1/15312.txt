#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &el : a) { cin >> el; }
  int health = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    health += a[i];
    if (health < 0) {
      ans = i;
      health = 0;
    }
  }
  cout << n - ans << endl;
  return 0;
}
