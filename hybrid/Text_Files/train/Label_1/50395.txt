import java.util.*;
import java.io.*;

public class Main {

    static int mod = 1000000007;

    static int[] dp;
    static ArrayList<ArrayList<Integer>> adj;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();

        dp = new int[n];
        Arrays.fill(dp, -1);
        adj = new ArrayList<ArrayList<Integer>>();
        for (int i = 0; i < n; i++) {
            adj.add(new ArrayList<Integer>());
        }

        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            adj.get(u).add(v);
            adj.get(v).add(u);
        }

        int ans = dfs(0, -1, k);
        System.out.println(ans);
    }

    static int dfs(int node, int parent, int k) {
        if (dp[node] != -1) {
            return dp[node];
        }

        long ans = 1;
        for (int child : adj.get(node)) {
            if (child != parent) {
                if (k == 0) {
                    ans = ans * 2 % mod;
                } else {
                    long ways = 1 + dfs(child, node, k - 1);
                    ans = ans * ways % mod;
                }
            }
        }

        return dp[node] = (int) ans;
    }
}
