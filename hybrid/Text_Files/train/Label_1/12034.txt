#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll mod = 1e9 + 7;
int main() {
  ll n, m, ans = 1;
  cin >> n >> m;
  vector<unordered_map<int, int>> nums(n);
  for (ll i = 0; i < n; i++) {
    ll x;
    cin >> x;
    unordered_map<int, int> cnt;
    while (x--) {
      ll y;
      cin >> y;
      cnt[y]++;
    }
    nums[i] = cnt;
  }
  vector<int> cnt(m + 1);
  for (ll i = 0; i < n; i++)
    for (auto p : nums[i]) cnt[p.first] += p.second;
  for (ll i = 1; i <= m; i++) ans = (ans * cnt[i]) % mod;
  cout << ans << '\n';
  return 0;
}
