#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n), c(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  for (int i = 0; i < n; i++) { cin >> c[i]; }

  vector<int> dp(n + 1);
  // dp[i] = maximum total joy if first i hares are fed
  dp[0] = 0;
  for (int i = 1; i <= n; i++) {
    dp[i] = max({dp[i - 1] + a[i - 1], dp[i - 2] + b[i - 1], dp[i - 3] + c[i - 1]});
  }

  cout << dp[n] << endl;

  return 0;
}
