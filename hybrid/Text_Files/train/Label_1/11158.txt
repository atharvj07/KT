#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  set<int> x;
  set<pair<int, int>> y;
  int ans = 0;
  while (n--) {
    int a;
    cin >> a;
    x.insert(a);
    ans++;
  }
  while (m--) {
    int a, b, c;
    cin >> a >> b >> c;
    for (auto it : x) {
      if (a <= it && it <= b) {
        ans++;
        break;
      }
    }
    y.insert({a, c});
    y.insert({b, c});
  }
  cout << ans << endl;
}
