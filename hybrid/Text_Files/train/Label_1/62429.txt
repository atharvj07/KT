import sys

def solve(n, k, p):
    ans = 0
    for i in range(1, n):
        if 100 * p[i] > k * (sum(p[:i + 1]) - p[i]):
            ans += 100 * p[i] - k * (sum(p[:i + 1]) - p[i])
            p[i] = k * (sum(p[:i + 1]) - p[i]) // 100
    return ans

t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    p = list(map(int, input().split()))
    print(solve(n, k, p))
