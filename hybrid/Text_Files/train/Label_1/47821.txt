import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        while (q-- > 0) {
            int xp0 = scanner.nextInt(), yp0 = scanner.nextInt(), xp1 = scanner.nextInt(), yp1 = scanner.nextInt();
            int xp2 = scanner.nextInt(), yp2 = scanner.nextInt(), xp3 = scanner.nextInt(), yp3 = scanner.nextInt();
            System.out.println(isIntersect(xp0, yp0, xp1, yp1, xp2, yp2, xp3, yp3) ? 1 : 0);
        }
    }

    private static boolean isIntersect(int xp0, int yp0, int xp1, int yp1, int xp2, int yp2, int xp3, int yp3) {
        int a1 = yp1 - yp0, b1 = xp0 - xp1, c1 = a1 * xp0 + b1 * yp0;
        int a2 = yp3 - yp2, b2 = xp2 - xp3, c2 = a2 * xp2 + b2 * yp2;
        int a3 = a1 * xp2 + b1 * yp2 - c1, a4 = a2 * xp0 + b2 * yp0 - c2;
        return a3 * a4 <= 0 && a1 * xp3 + b1 * yp3 - c1 <= 0 && a2 * xp1 + b2 * yp1 - c2 <= 0;
    }
}
