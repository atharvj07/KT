n, k = map(int, input().split())
s = list(input())
a = list(map(int, s))
f = True
for i in range(1, len(a)):
    if i % k != 0:
        a[i] = a[i-1]
    else: f = False
if f:
    print(n)
    print(''.join(map(str, a)))
else:
    t = a[k-1:]
    for i in range(1, len(t)):
        if t[i] < t[i-1]:
            t[i-1] += 1
            for j in range(i, len(t)):
                t[j] = t[i-1]
            break
    t[k-1] += 1
    a = s[:k-1] + t
    while len(a) < n:
        a.append(a[-1])
    print(len(a))
    print(''.join(map(str, a)))
