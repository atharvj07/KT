import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long k = sc.nextLong();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        long ans = 0;
        int cnt = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (cnt + arr[i] >= k) {
                ans = k - cnt - 1;
                break;
            }
            cnt += arr[i];
        }
        if (cnt < k) {
            ans = -1;
        }
        System.out.println(ans);
    }
}
