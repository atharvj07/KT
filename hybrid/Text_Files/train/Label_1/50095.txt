import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    private static final String PUSH_STACK = "pushStack";
    private static final String POP_STACK = "popStack";
    private static final String PUSH_QUEUE = "pushQueue";
    private static final String POP_QUEUE = "popQueue";
    private static final String PUSH_FRONT = "pushFront";
    private static final String POP_FRONT = "popFront";
    private static final String PUSH_BACK = "pushBack";
    private static final String POP_BACK = "popBack";

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        List<Integer> commands = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            commands.add(Integer.parseInt(reader.readLine()));
        }

        List<String> result = solve(commands);
        for (String s : result) {
            System.out.println(s);
        }
    }

    private static List<String> solve(List<Integer> commands) {
        List<Stack> stacks = new ArrayList<>();
        List<String> result = new ArrayList<>();
        int nextStack = 0;
        for (Integer command : commands) {
            if (command == 0) {
                int count = 0;
                List<String> popCommands = new ArrayList<>();
                for (int i = 0; i < 3; i++) {
                    if (!stacks.isEmpty()) {
                        count++;
                        Stack stack = stacks.remove(0);
                        popCommands.add(stack.pop());
                    }
                }
                result.add(count + " " + String.join(" ", popCommands));
            } else {
                while (stacks.size() <= nextStack) {
                    stacks.add(new Stack());
                }
                Stack stack = stacks.get(nextStack++);
                if (stack.isEmpty()) {
                    result.add(PUSH_STACK);
                } else if (stack.peek().equals(PUSH_QUEUE)) {
                    result.add(PUSH_BACK);
                } else {
                    result.add(PUSH_FRONT);
                }
                stack.push(command == 1 ? PUSH_STACK : command == 2 ? PUSH_QUEUE : PUSH_FRONT);
            }
        }
        return result;
    }

    private static class Stack {
        private final List<String> elements = new ArrayList<>();

        public void push(String element) {
            elements.add(element);
        }

        public String pop() {
            return elements.remove(elements.size() - 1);
        }

        public String peek() {
            return elements.get(elements.size() - 1);
        }

        public boolean isEmpty() {
            return elements.isEmpty();
        }
    }
}
