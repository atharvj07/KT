#include <bits/stdc++.h>
using namespace std;

const int N = 200005;
int n, k;
vector<int> ans;

int a[N], pos[N];
int q[N], qn;

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        pos[a[i]] = 0;
    }
    for (int i = 1; i <= n; ++i) {
        if (pos[a[i]] > 0) {
            for (int j = pos[a[i]]; j > 1; --j) q[j] = q[j - 1];
            q[1] = a[i];
        } else {
            ++qn;
            if (qn > k) {
                pos[q[qn]] = 0;
                for (int j = qn; j > 1; --j) q[j] = q[j - 1];
            }
            q[1] = a[i];
            pos[a[i]] = qn;
        }
    }
    int m = 0;
    for (int i = 1; i <= qn; ++i) if (pos[q[i]] > 0) ans.push_back(q[i]), ++m;
    cout << m << endl;
    for (int x : ans) cout << x << " ";
    cout << endl;
    return 0;
}
