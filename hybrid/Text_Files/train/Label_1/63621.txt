import math

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def lcm(a, b):
    return (a * b) // gcd(a, b)

n = int(input())
a_list = []
b_list = []
for i in range(n):
    a, b = map(int, input().split())
    a_list.append(a)
    b_list.append(b)
g = a_list[0]
for i in range(1, n):
    g = gcd(g, a_list[i])
    g = gcd(g, b_list[i])
lcm_list = []
for i in range(n):
    lcm_list.append(lcm(g, a_list[i]))
    lcm_list.append(lcm(g, b_list[i]))
lcm_list.sort()
x = lcm_list[0]
for i in range(1, len(lcm_list)):
    if x != lcm_list[i]:
        x = lcm_list[i]
        break
if x == g:
    x = -1
print(x)
