#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    long long k = 0;
    while (true) {
      string s = to_string(1LL << k);
      if (s.find(to_string(a[i])) != string::npos) {
        cout << k << endl;
        break;
      }
      k++;
    }
  }
  return 0;
}
