#include<bits/stdc++.h>
using namespace std;
string a[101];
int dp[101][101][2];
int func(int l,int r,int f){
    if(l>r) return 1;
    int &ret=dp[l][r][f];
    if(ret!=-1) return ret;
    ret=0;
    if(f==0){
        if(a[l]==a[r]){
            if(l+1<=r-1) ret=max(ret,1+func(l+1,r-1,f));
            else if(l==r) ret=max(ret,1+func(l+1,r-1,1));
            else if(l+1==r) ret=max(ret,1+func(l+1,r-1,0));
        }
        ret=max(ret,func(l+1,r,0));
        ret=max(ret,func(l,r-1,0));
    }
    else{
        ret=max(ret,func(l+1,r,1));
        ret=max(ret,func(l,r-1,1));
    }
    return ret;
}
int main(){
    int n,i,ans,l,r;
    cin>>n;
    for(i=1,l=0;i<=n;++i){
        cin>>r;
        while(r--){
            a[++l]=i+'a'-1;
        }
    }
    memset(dp,-1,sizeof dp);
    ans=func(1,l,0);
    cout<<ans<<endl;
    memset(dp,-1,sizeof dp);
    func(1,l,0);
    for(i=1;i<=l;++i){
        cout<<a[i];
    }
    cout<<endl;
    return 0;
}
