#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
vector<ll> v;
int main() {
  ll n, ans = 0;
  cin >> n;
  for(ll i = 0, x; i < n; i++) cin >> x, v.push_back(x);
  for(ll p1 = 0, p2 = 0; p1 < n; ) {
    if(v[p1] * 2 < v[p2]) p2++;
    else p2 = p1 + 1;
    ans = max(ans, p2 - p1 + 1);
    p1++;
  }
  cout << ans << endl;
}
