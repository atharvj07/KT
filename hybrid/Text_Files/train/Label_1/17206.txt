#include <bits/stdc++.h>
using namespace std;

const int MAXN = 200005;
int arr[MAXN];
bool ex[MAXN];

int main() {
    int n; cin >> n;
    for (int i = 0; i < n; i++) cin >> arr[i];
    map<int, vector<int>> mp;
    for (int i = 0; i < n; i++) {
        mp[arr[i]].push_back(i);
        ex[arr[i]] = 1;
    }
    bool flag = 0;
    for (auto a : mp) {
        if (a.second.size() >= 4) {
            cout << "YES\n";
            for (int i = 0; i < 4; i++) cout << a.second[i] + 1 << " ";
            flag = 1;
            break;
        }
    }
    sort(arr, arr + n);
    for (int i = 0; i < n - 3; i++) {
        if (arr[i] + arr[i + 3] < arr[i + 1] + arr[i + 2]) {
            if (ex[arr[i + 3] + arr[i + 1] - arr[i]]) {
                cout << "YES\n";
                for (int j : mp[arr[i]]) cout << j + 1 << " ";
                cout << mp[arr[i + 3] + arr[i + 1] - arr[i]][0] + 1 << " ";
                cout << mp[arr[i + 2] + arr[i + 1] - arr[i]][0] + 1 << "\n";
                flag = 1;
                break;
            }
        }
    }
    if (!flag) cout << "NO\n";
}
