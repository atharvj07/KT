#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int N = 150005;

int n, a[N], b[N];

ll gcd(ll x, ll y) { return !y ? x : gcd(y, x % y); }

int main() {
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> a[i] >> b[i];
  vector<ll> g;
  for (int i = 0; i < n; ++i) {
    ll x = gcd(a[i], b[i]);
    if (x) g.push_back(x);
  }
  sort(g.begin(), g.end());
  g.resize(unique(g.begin(), g.end()) - g.begin());
  ll ans = 0;
  for (const auto& x : g) {
    ll tmp = 0;
    for (int i = 0; i < n; ++i) {
      if (a[i] % x == 0 || b[i] % x == 0) tmp |= (1 << i);
    }
    if (tmp == (1 << n) - 1) ans = x;
  }
  cout << (ans ? ans : -1) << '\n';
  return 0;
}
