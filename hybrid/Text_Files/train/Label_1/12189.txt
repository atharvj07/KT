#include <bits/stdc++.h>
using namespace std;
inline int sgn(int x) { return x < 0 ? -1 : x > 0; }
int n, m, v1, v2, v3, v4, a[100002][3], b[100002][3];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++)
    cin >> a[i][0] >> a[i][1] >> a[i][2];
  cin >> m;
  for (int i = 1; i <= m; i++)
    cin >> b[i][0] >> b[i][1] >> b[i][2];
  for (int i = 1; i <= n; i++) {
    v1 = a[i][0] - a[i % n + 1][0], v2 = a[i][1] - a[i % n + 1][1],
    v3 = a[i][2] - a[i % n + 1][2];
    for (int j = 1; j <= m; j++) {
      v4 = b[j][0] - b[j % m + 1][0], v1 *= v4, v3 -= v4;
      v4 = b[j][1] - b[j % m + 1][1], v2 *= v4, v3 -= v4;
      v4 = b[j][2] - b[j % m + 1][2], v2 *= v4, v1 -= v4;
    }
    if (v1 || v2 || v3) { cout << "YES" << endl; return 0; }
  }
  for (int i = 1; i <= m; i++) {
    v1 = b[i][0] - b[i % m + 1][0], v2 = b[i][1] - b[i % m + 1][1],
    v3 = b[i][2] - b[i % m + 1][2];
    for (int j = 1; j <= n; j++) {
      v4 = a[j][0] - a[j % n + 1][0], v1 *= v4, v3 -= v4;
      v4 = a[j][1] - a[j % n + 1][1], v2 *= v4, v3 -= v4;
      v4 = a[j][2] - a[j % n + 1][2], v2 *= v4, v1 -= v4;
    }
    if (v1 || v2 || v3) { cout << "YES" << endl; return 0; }
  }
  cout << "NO" << endl;
}
