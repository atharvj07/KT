#include <iostream>
#include <string>

using namespace std;

int main() {
    int n; cin >> n;
    string s; cin >> s;
    bool f = false;
    for (int i = 0; i < n - 1; ++i) {
        if (s[i] > s[i + 1]) {
            cout << "YES\n" << i + 1 << ' ' << i + 2 << '\n';
            f = true;
            break;
        }
    }
    if (!f) cout << "NO\n";
}
