import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        System.out.println(modPow(2, m * (n - 1) - 2, 1000000007));
    }

    private static long modPow(long x, long n, int mod) {
        if (n == 0)
            return 1;
        long val = modPow(x, n / 2, mod);
        val = val * val % mod;
        if (n % 2 == 1)
            val = val * x % mod;
        return val;
    }
}
