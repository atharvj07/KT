import sys

r,c=map(int,input().split())
ms,mc=r,c
a=[list(input().strip()) for x in range(r)]
b=list(zip(*[list(input().strip()) for x in range(c)]))

def chk(x, y):
    for i in range(min(r,c)):
        for j in range(min(r,c)):
            if a[x+i][y+j]!=b[x+i][y+j]: return False
    return True

for i in range(r-c+1):
    for j in range(r-c+1):
        if chk(i,j):
            ms,mc=i+1,j+1

print(ms,mc)
