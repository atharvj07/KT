#include <iostream>

using ll = long long;

const int mod = 1e9 + 7;

ll binpow(ll a, ll n) {
    ll ret = 1;
    while (n > 0) {
        if (n & 1) ret = ret * a % mod;
        a = a * a % mod;
        n >>= 1;
    }
    return ret;
}

ll inva(ll a) { return binpow(a, mod - 2); }

ll cntOnes(ll n) {
    ll ret = 0;
    while (n) {
        ret += (n & 1);
        n >>= 1;
    }
    return ret;
}

int main() {
    ll n;
    std::cin >> n;
    int k;
    std::cin >> k;
    int ones = cntOnes(n);
    if (k > ones) {
        std::cout << "0\n";
        return 0;
    } else if (ones == k) {
        ll invOnes = inva(ones);
        ll invRest = inva(ones - k);
        ll ans = invOnes * invRest % mod;
        ans = (ans + mod) % mod;
        std::cout << ans << "\n";
        return 0;
    } else {
        ll pw = binpow(2, ones - k);
        ll ans = pw % mod;
        ans = (ans + mod) % mod;
        std::cout << ans << "\n";
        return 0;
    }
}
