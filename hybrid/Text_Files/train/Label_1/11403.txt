#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k,p;
    cin>>n>>k>>p;
    vector<int>a(n),b(k);
    for(int i=0;i<n;i++)cin>>a[i];
    for(int i=0;i<k;i++)cin>>b[i];
    sort(a.begin(),a.end());
    sort(b.begin(),b.end());
    int i=0,j=0,ans=0;
    while(i<n && j<k){
        if(a[i]==b[j]){
            j++;
        }
        else if(a[i]<b[j]){
            ans+=(p-a[i]);
            i++;
        }
        else{
            ans+=(b[j]-a[i]);
            j++;
        }
    }
    while(i<n){
        ans+=(p-a[i]);
        i++;
    }
    cout<<ans<<endl;
    return 0;
}
