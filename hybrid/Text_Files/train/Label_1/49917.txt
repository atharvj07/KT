import java.util.Scanner;

public class Main {
    static int n, m;
    static int[] deg;
    static Edge[] edges;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        deg = new int[n + 1];
        edges = new Edge[m];
        for (int i = 0; i < m; i++) {
            int u = in.nextInt() - 1;
            int v = in.nextInt() - 1;
            int w = in.nextInt();
            edges[i] = new Edge(u, v, w);
            deg[v]++;
        }
        int ans = 0;
        boolean[] vis = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (deg[i] == 0) {
                ans = Math.max(ans, dfs(i, vis, 0));
            }
        }
        System.out.println(ans);
    }

    static int dfs(int u, boolean[] vis, int w) {
        vis[u] = true;
        int ans = 1;
        for (Edge e : edges) {
            if (e.u == u && !vis[e.v] && e.w > w) {
                ans = Math.max(ans, 1 + dfs(e.v, vis, e.w));
            }
        }
        vis[u] = false;
        return ans;
    }

    static class Edge {
        int u, v, w;

        public Edge(int u, int v, int w) {
            this.u = u;
            this.v = v;
            this.w = w;
        }
    }
}
