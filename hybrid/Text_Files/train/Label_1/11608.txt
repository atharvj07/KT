#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    for (int i = 0; i < n; i++) {
      bool ok = true;
      for (int j = 0; j < n; j++) {
        if (i != j) {
          if (a[i] < a[j] && b[i] < b[j]) {
            ok = false;
            break;
          }
        }
      }
      (ok ? cout << '1' : cout << '0');
    }
    cout << '\n';
  }
  return 0;
}
