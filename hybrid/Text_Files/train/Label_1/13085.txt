#include <bits/stdc++.h>

const int MOD = 1e9 + 7;

int add(int a, int b) {
  return (a + b) % MOD;
}

int mul(int a, int b) {
  return (a * 1LL * b) % MOD;
}

int pow(int a, int b) {
  int res = 1;
  while (b) {
    if (b & 1) res = mul(res, a);
    a = mul(a, a);
    b >>= 1;
  }
  return res;
}

constexpr int MAXN = 1000001;
int inv[MAXN];

void pre() {
  inv[1] = 1;
  for (int i = 2; i < MAXN; ++i) {
    inv[i] = mul(MOD - (MOD / i), inv[MOD % i]);
  }
}

int main() {
  pre();
  int a, b, n;
  std::cin >> a >> b >> n;
  if (a == b) {
    std::cout << pow(2, n - 1) << '\n';
    return 0;
  }
  int x = add(b - a, MOD - 1);
  int y = pow(x, n - 1);
  std::cout << mul(y, x) << '\n';
}
