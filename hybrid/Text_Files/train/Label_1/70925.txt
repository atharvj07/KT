n, m, k = map(int, input().split())
p = [int(x) for x in input().split()]
dp = [[0] * (k + 1) for _ in range(n + 1)]
for i in range(1, n + 1):
    for j in range(k + 1):
        for l in range(i - m, i):
            if l >= 0:
                dp[i][j] = max(dp[i][j], dp[l][j - 1] + sum(p[l:i]))
print(dp[n][k])
