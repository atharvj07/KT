#include <iostream>
#include <string>
#include <vector>

using namespace std;

const int MAX_CMD = 127;
const int MAX_N = 1024;

int main() {
  string golorp;
  cin >> golorp;

  int n, i;
  vector<int> seq;

  // skip '?'
  i = 1;

  while (i < golorp.size() && golorp[i] != '(') i++;
  i++;

  // skip '('
  n = 0;
  while (i < golorp.size() && golorp[i] != ')') {
    switch (golorp[i]) {
      case '*':
        n = n < 1 ? MAX_CMD : n - 1;
        break;
      case '_':
        n = 0;
        break;
      default:
        n = (n + 1) % 10;
        break;
    }
    i++;
  }

  // skip ')'
  i++;

  if (n == 0) {
    cout << "false";
    return 0;
  }

  while (i < golorp.size()) {
    switch (golorp[i]) {
      case '-':
        seq.push_back(9 - (n - 1));
        break;
      case '+':
        if (n < 9) {
          seq.push_back(n + 1);
          n = n + 1;
        } else {
          seq.push_back(0);
          n = 0;
        }
        break;
      case '<':
        seq.push_back(n);
        break;
      case '>':
        if (n > 0) {
          seq.push_back(n - 1);
          n = n - 1;
        } else {
          seq.push_back(9);
          n = 9;
        }
        break;
      case ',':
        seq.push_back(0);
        n = 0;
        break;
      case '.':
        cout << "false";
        return 0;
        break;
    }
    i++;
  }

  for (i = 0; i < seq.size(); i++) {
    cout << seq[i];
  }

  return 0;
}
