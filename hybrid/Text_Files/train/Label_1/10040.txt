#include <iostream>
#include <cstdio>
#include <algorithm>
using namespace std;
const int MAXN = 1e5 + 5, MAXM = 1e9 + 5;
long long n, d, m[MAXN], s[MAXN], ms[MAXN], fs[MAXN], ans;
bool cmp(int a, int b) {
  return ms[a] < ms[b];
}
int main() {
  cin >> n >> d;
  for (int i = 1; i <= n; i++) {
    cin >> m[i] >> s[i];
    ms[i] = m[i] + d;
    fs[i] = s[i];
  }
  sort(ms + 1, ms + n + 1);
  sort(fs + 1, fs + n + 1, cmp);
  int l = 1, r = n;
  while (l <= r) {
    int mid = (l + r) >> 1;
    int x = fs[mid] + fs[mid + 1] - s[mid + 1];
    int sum = 0, i = n;
    while (i && m[i] > ms[mid])
      sum += s[i], i--;
    if (sum > x)
      l = mid + 1;
    else
      ans = x, r = mid - 1;
  }
  cout << ans << endl;
  return 0;
}
