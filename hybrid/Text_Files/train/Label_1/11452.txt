#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define ld long double
#define endl '\n'
#define io ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0)
#define all(v) v.begin(),v.end()
#define rep(i,a,b) for(ll i = a;i <= b;i++)
int main(){
    io;
    ll n;cin>>n;
    vector<ll>v(n),b(n);
    for(ll i=0;i<n;i++)cin>>v[i],b[i] = v[i];
    sort(b.begin(),b.end());
    for(ll i=1;i<=n-2;i++){
        vector<ll>a;
        for(ll j=0;j<n;j++)if(j+1!=i)a.push_back(v[j]);
        sort(a.begin(),a.end());
        ll diff = a[1]-a[0],f=0;
        for(ll j=1;j<a.size();j++){
            if(a[j]!=a[j-1]+diff){f=1;break;}
        }
        if(f==0){cout<<i;return 0;}
    }
    cout<<-1;
    return 0;
}
