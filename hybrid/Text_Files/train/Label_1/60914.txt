import math

def calculate_beauty(arr, x):
    return sum(math.ceil(i / x) for i in arr)

t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    arr = list(map(int, input().split()))
    
    max_beauty = calculate_beauty(arr, x)
    
    total_sum = sum(arr)
    min_beauty = math.ceil(total_sum / x)
    
    print(min_beauty, max_beauty)
