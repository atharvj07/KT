import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        int sum = 0;
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
            sum += p[i];
        }
        int total = sum / 2;
        int ans = 0;
        int curr = 0;
        for (int i = 0; i < n; i++) {
            curr += p[i];
            ans++;
            if (curr > total) {
                break;
            }
        }
        System.out.println(ans);
    }
}
