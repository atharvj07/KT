import java.util.Scanner;

public class EqualizeStrings {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String a = scanner.next();
        String b = scanner.next();

        int cost = 0;
        for (int i = 0; i < n; i++) {
            if (a.charAt(i) != b.charAt(i)) {
                // If the bits are different, check if it's cheaper to swap or flip
                if (i > 0 && a.charAt(i - 1) != b.charAt(i - 1)) {
                    cost += 1; // Flip the previous bit
                } else {
                    cost += Math.abs(i - (n - 1 - i)); // Swap with the last different bit
                }
            }
        }

        System.out.println(cost);
    }
}
