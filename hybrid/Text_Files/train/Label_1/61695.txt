from itertools import product

def min_hints(cards):
    colors = {card[0] for card in cards}
    values = {card[1] for card in cards}
    min_hints = float('inf')

    for color_hints in product([0, 1], repeat=len(colors)):
        for value_hints in product([0, 1], repeat=len(values)):
            hinted_colors = {color for color, hint in zip(colors, color_hints) if hint}
            hinted_values = {value for value, hint in zip(values, value_hints) if hint}

            if all(card[0] in hinted_colors or card[1] in hinted_values for card in cards):
                min_hints = min(min_hints, sum(color_hints) + sum(value_hints))

    return min_hints

n = int(input())
cards = input().split()
print(min_hints(cards))
