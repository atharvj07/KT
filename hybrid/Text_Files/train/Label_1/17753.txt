#include<bits/stdc++.h>
using namespace std;

const int N = 2e5+10;

int n,m,x,k,y;
vector<int>a,b;
int ans = 1e9;

bool ok(int i, int j) {
    if(a[i] != b[j]) return false;
    if(i+1 <= n && j+1 <= m) return ok(i+1, j+1);
    if(i+1 > n || j+1 > m) return true;
}

void go(int pos, int cost) {
    if(pos > n) {
        if(ok(1,1)) ans = min(ans, cost);
        return;
    }

    // skip
    go(pos+1, cost);

    // fireball
    if(pos+k-1 <= n) {
        int need = 0;
        for(int i = pos; i < pos+k; i++) need += (b[i] != a[i]);
        if(need <= 1) go(pos+k, cost+need*x);
    }

    // berserk
    if(pos+1 <= n) {
        go(pos+2, cost+y);
    }
}

int main() {
    cin >> n >> m;
    cin >> x >> k >> y;
    a.resize(n+1); b.resize(m+1);
    for(int i = 1; i <= n; i++) { cin >> a[i]; }
    for(int i = 1; i <= m; i++) { cin >> b[i]; }

    go(1,0);

    if(ans == 1e9) ans = -1;
    cout << ans << endl;

    return 0;
}
