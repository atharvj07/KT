import sys

for line in sys.stdin:
    n = int(line.split()[0])
    a = list(map(int, line.split()[1:]))
    cnt, ans = {}, 0
    for x in a:
        if x in cnt:
            if cnt[x] == 1:
                ans -= 1
            else:
                ans += 1
        else:
            ans += 1
        cnt[x] = cnt.get(x, 0) + 1
    print(ans)
