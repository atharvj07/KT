#include <iostream>
#include <vector>
int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<std::string> map(n);
  for (auto &a : map) {
    std::cin >> a;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (map[i][j] == '.' && i != 0 && i != n - 1 && j != 0 && j != m - 1) {
        if ((map[i + 1][j] == '.' && map[i][j + 1] == '#') ||
            (map[i - 1][j] == '.' && map[i][j - 1] == '#') ||
            (map[i][j + 1] == '.' && map[i][j - 1] == '#'))
          ans++;
      }
    }
  }
  std::cout << ans / 2 << std::endl;
  return 0;
}
