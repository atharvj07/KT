import sys

input=sys.stdin.readline
n,m,t=map(int,input().split())
a=[0]
for _ in range(n):
    tmp=[max(0,int(i)-t)for i in input().split(':')]
    tmp=[tmp[0]*3600+tmp[1]*60+tmp[2]]
    if tmp!=a[-1]:a+=[tmp[0]]
idLst=[0]*len(a)
for i in range(m,len(a)):
    for j in range(i-m,i):
        if idLst[i]==0:
            break
        if a[i]==a[j]+t and idLst[j]!=0:
            idLst[i]=idLst[j]
if idLst.count(0)!=0:
    print('No solution')
else:
    user=1
    print(idLst.count(0))
    for i in idLst[1:]:
        if i==0:
            while 1 in idLst:
                idLst[idLst.index(1)]=user
                user+=1
            print(user)
        else:
            print(i)
