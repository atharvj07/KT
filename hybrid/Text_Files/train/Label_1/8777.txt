#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
int main() {
    int n;
    cin >> n;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    if (a[n - 1] - a[n - 2] < 0) {
        cout << "NO" << endl;
        return 0;
    }
    ll delta = a[n - 1] - a[n - 2];
    ll x = 0, y = 0;
    x = (ll)ceil(sqrtl(delta));
    y = delta / x;
    if (x * y == delta && x <= 1e9 && y <= 1e9) {
        cout << "YES" << endl;
        cout << x << ' ' << y << endl;
    } else
        cout << "NO" << endl;
    return 0;
}
