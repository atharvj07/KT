#include <bits/stdc++.h>
using namespace std;

const int N = 1000 + 1;
int a[N], b[N], c[N];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { c[i] = i; }
    for (int i = 1; i <= m; i++) {
        cin >> a[i] >> b[i];
        c[b[i]] = max(c[b[i]], c[a[i] - 1] + 1);
    }
    int nw = c[n];
    for (int i = n; i >= 1; i--) {
        if (c[i] == nw) {
            a[nw] = i;
            nw--;
        }
    }
    for (int i = 1; i <= n; i++) {
        if (a[i] % 2 == 1) {
            cout << '1';
        } else {
            cout << '0';
        }
    }
    cout << endl;
    return 0;
}
