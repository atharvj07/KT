import sys
input = sys.stdin.readline

n = int(input())
arr = list(map(int, input().split()))
q = int(input())
log = []
for i in range(q):
    log.append(list(map(int, input().split())))

bal = [0] * n

for event in log:
    if event[0] == 1:
        bal[event[1] - 1] = event[2]
    else:
        thres = event[1]
        for i in range(n):
            if bal[i] < thres:
                bal[i] = thres

for i in bal:
    print(i)
