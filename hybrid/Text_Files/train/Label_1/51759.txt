import java.util.Scanner;

class Main {
  private static boolean isLucky(int hour, int minute) {
    return String.valueOf(hour).contains("7") || String.valueOf(minute).contains("7");
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int x = scanner.nextInt();
    int hour = scanner.nextInt();
    int minute = scanner.nextInt();

    int count = 0;
    int time = (hour * 60 + minute) - (x * count);
    while (!isLucky(time / 60, time % 60)) {
      count++;
      time = (hour * 60 + minute) - (x * count);
    }

    System.out.println(count);
  }
}
