from collections import deque, defaultdict
from string import ascii_lowercase

def bfs(n, u, colors, stones):
    q, dist, par = deque(), [-1] * (n+1), [-1] * (n+1)
    q.append(u)
    dist[u] = 0
    while q:
        v = q.popleft()
        for i in range(1, n+1):
            if i != v and colors[v][i] == colors[u][i]:
                if i not in stones:
                    if dist[i] == -1:
                        dist[i] = dist[v] + 1
                        par[i] = v
                        q.append(i)
                elif dist[i] == dist[v] + 1:
                    return par, dist[i]
    return None, -1

n = int(input())
a, b, c = map(int, input().split())
colors = [list(input()) for _ in range(n)]
stones = {a, b, c}

ans = []
while stones != {1, 2, 3}:
    par, dist = bfs(n, stones.pop(), colors, stones)
    if dist == -1: break
    while par[stones[-1]] != stones[0]: stones.add(par[stones[-1]])
    ans.append([stones.pop(), par[stones[-1]]])

print(len(ans))
for a, b in ans: print(a, b)
