#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    if (a == b) {
      cout << "YES" << endl;
    } else {
      int sumA = 0, sumB = 0;
      for (int i = 0; i < n; i++) {
        sumA += a[i];
        sumB += b[i];
      }
      int diff = sumB - sumA;
      if (diff % n == 0 && diff > 0) {
        cout << "YES" << endl;
      } else {
        cout << "NO" << endl;
      }
    }
  }
  return 0;
}
