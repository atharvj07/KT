#include <bits/stdc++.h>
using namespace std;
int ch[300000], mark[300000];
string s, t;
int main() {
  cin >> s >> t;
  int q;
  cin >> q;
  while (q--) {
    int aa, bb, cc, dd;
    cin >> aa >> bb >> cc >> dd;
    int cur = cc;
    int fl = 0;
    for (int i = aa; i <= bb; i++) {
      if (t[cur] == s[i]) {
        cur++;
      }
      if (cur > dd) {
        fl = 1;
        break;
      }
    }
    if (fl == 0 && cur == dd + 1)
      cout << "1";
    else
      cout << "0";
  }
  cout << endl;
}
