#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> b(n);
    for (int& bi : b) cin >> bi;
    vector<int> a(2 * n, 0);
    bool flag = false;
    for (int i = 0; i < n; i++) {
      if (b[i] == 2 * n) {
        flag = true;
        break;
      }
      a[2 * i] = b[i];
      a[2 * i + 1] = 2 * n;
    }
    if (flag) {
      cout << -1 << "\n";
      continue;
    }
    for (int i = 1; i < 2 * n; i++)
      if (a[i] == 0) {
        int j = i - 1;
        while (j >= 0 && a[j] != 0) j--;
        int p = 0;
        while (p < b[j / 2] && a[i] != 0) p++;
        a[i] = p + 1;
      }
    for (int ai : a) cout << ai << " ";
    cout << "\n";
  }
}
