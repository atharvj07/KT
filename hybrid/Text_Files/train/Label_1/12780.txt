#include <iostream>
#include <vector>

#define MOD 1000000007

using namespace std;

int main() {
  long long k;
  cin >> k;
  vector<long long> dp(k + 1, 0);
  dp[0] = 1;
  for (long long i = 1; i <= k; i++) {
    for (long long j = 0; j < i; j++) {
      dp[i] = (dp[i] + dp[j] * dp[i - j - 1]) % MOD;
    }
  }
  cout << dp[k] << endl;
  return 0;
}
