#include <iostream>

using namespace std;

typedef long long LL;

int gcd(LL a, LL b)
{
	return b ? gcd(b, a % b) : a;
}

int main()
{
    int x, y, a, b;
    cin >> x >> y >> a >> b;
    if (a == b) {
        if ((a + x - 1) % x == 0 && (a + y - 1) % y == 0) {
            cout << 1 << endl;
        } else {
            cout << 0 << endl;
        }
    } else {
        int g = gcd(x, y);
        int L = (b - a + 1) / g;
        int S = (((a + g - 1) / g) % (y / g) + L - 1) % L + 1;
        cout << (S <= L ? L - S + 1 : 0) << endl;
    }
    return 0;
}
