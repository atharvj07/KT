#include<bits/stdc++.h>
using namespace std;

int a[200010];
int b[200010];
int freq[200010];

vector<int> ina,inb;

int main(){
    int n;
    cin >> n;
    for(int i = 0; i < n; i++) cin >> a[i];
    bool flag = true;
    int Max = 0;
    for(int i = 0; i < n; i++){
        freq[a[i]]++;
        if(freq[a[i]] > 1) flag = false;
        if(a[i] > Max) Max = a[i];
    }
    int cnt = 0;
    for(int i = 0; i < n; i++){
        if(a[i] == Max) cnt++;
    }
    if(cnt > 1) flag = false;
    sort(a,a+n);
    int id = 0;
    for(int i = 1; i <= 2*n; i++){
        if(freq[i] == 0){
            b[id] = i;
            id++;
        }
    }
    int m = n - cnt;
    int inc = 0;
    for(int i = 0; i < id; i++){
        if(inc < m){
            ina.push_back(b[i]);
            inc++;
        }
        else inb.push_back(b[i]);
    }
    if(!flag) cout << "NO" << endl;
    else{
        cout << "YES" << endl;
        cout << ina.size() << endl;
        for(int i = 0; i < ina.size(); i++) cout << ina[i] << ' ';
        cout << endl;
        cout << inb.size() << endl;
        for(int i = 0; i < inb.size(); i++) cout << inb[i] << ' ';
        cout << endl;
    }
}
