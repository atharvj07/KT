#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct item {
    int t, d, p;
};

int main() {
    int n;
    cin >> n;
    vector<item> items(n);
    for (int i = 0; i < n; i++) {
        cin >> items[i].t >> items[i].d >> items[i].p;
    }
    sort(items.begin(), items.end(), [](const item& a, const item& b) { return a.d < b.d; });
    int sum = 0, m = 0;
    vector<int> ids;
    for (int i = 0; i < n; i++) {
        if (items[i].t <= items[i].d) {
            sum += items[i].p;
            m++;
            ids.push_back(i + 1);
        }
    }
    cout << sum << endl << m << endl;
    for (int id : ids) { cout << id << ' '; }
    cout << endl;
    return 0;
}
