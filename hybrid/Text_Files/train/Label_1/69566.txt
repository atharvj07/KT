def max_weights_revealed(n, weights):
    from collections import Counter

    def can_form_subset(dp, total, count):
        for i in range(len(dp) - 1, -1, -1):
            if dp[i] and i + total < len(dp) and dp[i + total] >= count:
                return True
        return False

    weights.sort()
    counter = Counter(weights)
    unique_weights = list(counter.keys())

    if len(unique_weights) == 1:
        return n

    dp = [0] * (sum(weights) + 1)
    dp[0] = float('inf')

    for weight in weights:
        for j in range(len(dp) - weight - 1, -1, -1):
            if dp[j] > 0:
                dp[j + weight] = max(dp[j + weight], dp[j] + 1)

    max_reveal = 1
    for i in range(len(unique_weights)):
        for j in range(i + 1, len(unique_weights)):
            total = unique_weights[i] + unique_weights[j]
            if can_form_subset(dp, total, counter[unique_weights[i]] + counter[unique_weights[j]]):
                max_reveal = max(max_reveal, counter[unique_weights[i]] + counter[unique_weights[j]])

    for weight, count in counter.items():
        if count > 1:
            max_reveal = max(max_reveal, count)

    return max_reveal

n = int(input().strip())
weights = list(map(int, input().strip().split()))
print(max_weights_revealed(n, weights))
