def valid_moves(x, y):
    return [(x+2, y+2), (x+2, y-2), (x-2, y+2), (x-2, y-2)]

def check_meet(x1, y1, x2, y2, bad):
    q = [(x1, y1)]
    vis = set()
    while q:
        x, y = q.pop(0)
        if (x, y) in vis: continue
        vis.add((x, y))
        if (x, y) == (x2, y2): return True
        for a, b in valid_moves(x, y):
            if 0 <= a < 8 and 0 <= b < 8 and (a, b) not in bad:
                q.append((a, b))
    return False

for _ in range(int(input())):
    bad = set()
    board = [input() for _ in range(8)]
    for x in range(8):
        for y in range(8):
            if board[x][y] == '#': bad.add((x, y))
    
    kx1, ky1, kx2, ky2 = -1, -1, -1, -1
    for x in range(8):
        for y in range(8):
            if board[x][y] == 'K':
                if kx1 == -1:
                    kx1, ky1 = x, y
                else:
                    kx2, ky2 = x, y
    
    print(['NO', 'YES'][check_meet(kx1, ky1, kx2, ky2, bad)])
