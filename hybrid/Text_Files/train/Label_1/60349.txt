import sys
input = sys.stdin.readline

n, m, k = map(int,input().split())
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
pos = [0]*100001

for i in range(n):
    pos[a[i]] = i+1

ans = 0

cur = 1
for x in b:
    ny = (pos[x]-1)//k+1
    ans += abs(cur-ny) + 1
    cur = ny
    pos[x] -= 1
    a[pos[x]], a[pos[x]+1] = a[pos[x]+1], a[pos[x]]
    pos[a[pos[x] + 1]] = pos[x] + 1
    pos[x] += 1

print(ans)
