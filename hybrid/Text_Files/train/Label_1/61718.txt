from collections import defaultdict
n=int(input())
a=list(map(int,input().split()))
g=defaultdict(list)
for i in range(n-1):
    x,y=map(int,input().split())
    g[x].append(y)
    g[y].append(x)
def dfs(x,p,w,b):
    if p==x:
        if a[x-1]==1:w=1
        else:b=1
    else:
        if a[x-1]==1:w+=1
        else:b+=1
    mx=w-b
    for nx in g[x]:
        if nx!=p:
            nw,nb=dfs(nx,x,w,b)
            mx=max(mx,nw-nb)
    return w,b,mx
ans=[0]*n
for i in range(1,n+1):
    dfs(i,i,0,0)
    print(ans[i-1])
