#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define vi vector<int>
#define vvi vector<vi>
int n, k;
vvi g;
vi p;
struct Node{
    int best, ans;
}dp[1001][1001];
Node Merge(Node a, Node b){
    if(a.best > b.best) return a;
    if(a.best < b.best) return b;
    return (a.ans < b.ans ? a : b);
}
Node dfs(int u, int prv){
    dp[u][u] = {0, u};
    for(auto v : g[u]){
        if(v == prv) continue;
        Node ans;
        ans = dfs(v, u);
        if(p[ans.ans] < p[u]) ans = Merge(ans, dfs(ans.ans, u));
        for(auto j : g[u]){
            if(j == v || j == prv) continue;
            Node tmp = Merge(dfs(j, u), ans);
            if(tmp.best > ans.best) ans = tmp;
        }
        dp[u][v] = ans;
        ans.best++;
        ans.ans = v;
        dp[u][u] = Merge(dp[u][u], ans);
    }
    return dp[u][prv];
}
int main(){
    cin >> n;
    g.resize(n + 1);
    p.resize(n + 1);
    for(int i = 1; i <= n; i++) cin >> p[i];
    for(int i = 0; i < n - 1; i++){
        int a, b;
        cin >> a >> b;
        g[a].pb(b);
        g[b].pb(a);
    }
    Node ans = dfs(1, 1);
    if(ans.best != n - 1) cout << "NO" << endl;
    else{
        cout << "YES" << endl;
        int u = ans.ans, v = 1;
        for(int i = 1; i < n; i++){
            if(dp[u][v].ans == v){
                int tmp = v;
                v = u;
                u = tmp;
            }
            cout << u << ' ' << v << endl;
            v = dp[u][v].ans;
        }
    }
}
