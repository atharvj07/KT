#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int MXN = 1e5 + 5;
const ll MXL = 1e9 + 5;
int n;
ll l[MXN], r[MXN];

int main() {
    cin >> n;
    for(int i = 0; i < n; i++) {
        cin >> l[i] >> r[i];
    }
    ll  maxn = 0, sum = 0;
    for(int i = 0; i < n; i++) {
        maxn = max({maxn, l[i], r[i]});
        sum += l[i] + r[i];
    }
    ll ans = maxn + max(sum, maxn);
    cout << max(ans, 1LL) << endl;
    return 0;
}
