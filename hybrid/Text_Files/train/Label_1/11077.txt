#include <bits/stdc++.h>
#define MAXN 505
using namespace std;
int n, m, a[MAXN][MAXN], x, y, f1[MAXN][MAXN][2], f2[MAXN][MAXN][2];
char c;
int q;
int main()
{
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
        {
            cin >> c;
            if (c == '#')
                a[i][j] = 1;
        }
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
        {
            if (!a[i][j])
            {
                //f1[i][j][0|1] = so cach dat domino theo chieu ngang|doc o o(i,j)
                //f2[i][j][0|1] = so cach dat domino theo chieu ngang|doc ket thuc o o(i,j)
                f1[i][j][0] = f1[i][j - 1][0] + (i == 1 || a[i - 1][j]);
                f1[i][j][1] = f1[i - 1][j][1] + (j == 1 || a[i][j - 1]);
                f2[i][j][0] = f2[i][j - 1][0] + (i == n || a[i + 1][j]);
                f2[i][j][1] = f2[i - 1][j][1] + (j == m || a[i][j + 1]);
            }
        }
    cin >> q;
    while (q--)
    {
        cin >> x >> y >> n >> m;
        cout << (f1[n][m][0] - f1[n][y - 1][0] - f1[x - 1][m][0] + f1[x - 1][y - 1][0]) + (f1[n][m][1] - f1[x - 1][m][1] - f1[n][y - 1][1] + f1[x - 1][y - 1][1]) + (f2[n][m][0] - f2[n][x - 1][0] - f2[y - 1][m][0] + f2[x - 1][y - 1][0]) + (f2[n][m][1] - f2[n][y - 1][1] - f2[x - 1][m][1] + f2[x - 1][y - 1][1]) << endl;
    }
    return 0;
}
