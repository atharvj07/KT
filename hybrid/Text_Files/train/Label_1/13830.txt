#include <bits/stdc++.h>

using namespace std;

#define endl '\n'

typedef long long ll;
typedef pair<int, int> pii;
typedef vector<int> vi;
typedef vector<ll> vll;
typedef vector<pii> vpii;

#define FOR(i, a, b) for (int i = (a); i < (b); ++i)
#define ROF(i, a, b) for (int i = (a); i >= (b); --i)
#define F0R(i, n) FOR(i, 0, n)
#define R0F(i, n) ROF(i, n, 0)

const int mxN = 1e5 + 5;
bool vis[mxN], mark[mxN];
int par[mxN], a[mxN], dep[mxN], sz[mxN], mxsz[mxN];
vi g[mxN];

void dfs1(int u) {
    vis[u] = 1;
    sz[u] = 1;
    for (int &v : g[u]) {
        if (!vis[v]) {
            dfs1(v);
            sz[u] += sz[v];
            mxsz[u] = max(mxsz[u], sz[v]);
        }
    }
}

void dfs2(int u) {
    vis[u] = 1;
    int mx1 = -1, mx2 = -1, id1 = -1, id2 = -1;
    for (int &v : g[u]) {
        if (!vis[v]) {
            if (sz[v] > mx1) {
                mx2 = mx1;
                mx1 = sz[v];
                id2 = id1;
                id1 = v;
            } else if (sz[v] > mx2) {
                mx2 = sz[v];
                id2 = v;
            }
        }
    }

    for (int &v : g[u]) {
        if (!vis[v]) {
            if (mx1 + mx2 + 1 < sz[u] || mx1 + mx2 + 1 < sz[u] - sz[v]) {
                mark[u] = 1;
            }
            dfs2(v);
        }
    }
    if (!mark[u] && mx1 > sz[u] - mx1) {
        mark[u] = 1;
    }
    if (!mark[u] && (id1 != -1 && mark[id1]) || (id2 != -1 && mark[id2])) {
        mark[u] = 1;
    }
}

void dfs3(int u) {
    vis[u] = 1;
    for (int &v : g[u]) {
        if (!vis[v]) {
            if (par[u] != v) {
                par[v] = u;
            }
            dep[v] = dep[u] + 1;
            dfs3(v);
        }
    }
}

int main() {
    int n;
    cin >> n;
    F0R(i, n) cin >> a[i];
    F0R(i, n - 1) {
        int p;
        cin >> p;
        g[p].push_back(i + 1);
        g[i + 1].push_back(p);
    }

    dfs1(1);
    F0R(i, n) vis[i] = 0;
    dfs2(1);
    F0R(i, n) vis[i] = 0;
    dfs3(1);

    ll ans = 0;
    F0R(i, n) {
        if (par[i] != 0 && dep[par[i]] % 2 == dep[i] % 2) {
            ans += a[i];
        }
    }

    F0R(i, n) {
        if (!mark[i]) {
            F0R(j, n) {
                if (!mark[j] && i != j) {
                    if ((dep[i] % 2 == dep[j] % 2) && (a[i] + a[j] > ans)) {
                        ++ans;
                    }
                }
            }
        }
    }

    cout << ans << endl;
    return 0;
}
