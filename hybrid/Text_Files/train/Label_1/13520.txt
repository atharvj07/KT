#include <bits/stdc++.h>
using namespace std;
#define ll long long
int32_t main()
{
    ll n, m;
    cin >> n >> m;
    vector<ll> p(n);
    for (auto &x : p)
        cin >> x;
    vector<vector<ll>> adj(n + 1);
    for (ll i = 0; i < m; i++)
    {
        ll a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    vector<bool> vis(n + 1, false);
    ll ans = (n * (n + 1)) / 2LL;
    function<void(ll)> dfs = [&](ll node) {
        if (vis[node])
            return;
        vis[node] = true;
        ll sz = adj[node].size();
        ans -= sz;
        for (auto x : adj[node])
            dfs(x);
    };
    for (ll i = 1; i <= n; i++)
    {
        if (!vis[i])
            dfs(p[i - 1]);
    }
    cout << ans << endl;
    return 0;
}
