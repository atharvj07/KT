#include <string>
#include <vector>
#include <limits>
#include <iostream>

int main() {
    std::string s; std::getline(std::cin, s);
    std::string mask; std::getline(std::cin, mask);
    int k; std::cin >> k;
    size_t n = s.size();
    std::vector<int> sum(n + 1);
    for (size_t i = 0; i < n; ++i) sum[i + 1] = (mask[s[i] - 'a'] == '0' ? 1 : 0) + sum[i];
    long long ans = 0;
    for (size_t i = 0; i < n; ++i) {
        int l = i, r = (int)n - 1;
        while (l <= r) {
            int mid = (l + r) / 2;
            int cnt = sum[mid + 1] - sum[i];
            if (cnt <= k) {
                ans += mid - i + 1;
                l = mid + 1;
            } else {
                r = mid - 1;
            }
        }
    }
    std::cout << ans << std::endl;
    return EXIT_SUCCESS;
}
