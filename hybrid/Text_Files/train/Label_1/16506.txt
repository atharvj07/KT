#include <bits/stdc++.h>
using namespace std;

vector<set<int>> G;
vector<bool> vis;
vector<vector<int>> ans;

void dfs(int u) {
    vis[u] = true;
    for (int v : G[u]) {
        if (!vis[v]) {
            ans.back().push_back(G[u].size() + G[v].size() - 1);
            dfs(v);
        }
    }
}

int main() {
    int n;
    cin >> n;
    G.resize(n + 1);
    vis.resize(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        G[u].insert(v);
        G[v].insert(u);
    }

    for (int i = 1; i <= n; i++) {
        if (!vis[i]) {
            ans.push_back(vector<int>());
            dfs(i);
        }
    }

    cout << ans.size() << endl;
    for (auto& v : ans) {
        cout << v.size() << " ";
        for (int x : v) { cout << x << " "; }
        cout << endl;
    }
}
