s = input().strip()
balance = 0
hash_counts = []
for c in s:
    if c == '(':
        balance += 1
    elif c == ')':
        balance -= 1
    elif c == '#':
        hash_counts.append(1)
        balance -= 1
    if balance < 0:
        print(-1)
        break
else:
    n = len(hash_counts)
    hash_counts[-1] += balance
    balance_sum = sum(hash_counts)
    if balance_sum != 0:
        for i in range(len(s) - 1, -1, -1):
            if s[i] == '(':
                balance_sum -= 1
            elif s[i] == ')':
                balance_sum += 1
            if balance_sum < 0:
                print(-1)
                break
        else:
            for count in hash_counts:
                print(count)
    else:
        print(-1)
