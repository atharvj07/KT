N = int(input())
edges = [list(map(int, input().split())) for _ in range(N)]

def floyd_warshall(matrix):
    n = len(matrix)
    dist = [[float('inf')] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for i in range(n):
        for j in range(n):
            dist[i][j] = matrix[i][j]
    for k in range(n):
        for i in range(n):
            for j in range(n):
                dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])
    return dist

distances = floyd_warshall(edges)
max_distance = 0
for i in range(N):
    for j in range(N):
        max_distance = max(max_distance, distances[i][j])
print(max_distance)
