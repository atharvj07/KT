import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        String s = sc.next();
        int[] a = new int[10];
        int[][] ans = new int[q][10];
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            for (int j = 0; j < 10; j++) {
                a[j] = 0;
            }
            for (int j = l; j <= r; j++) {
                if (s.charAt(j) >= '0' && s.charAt(j) <= '9') {
                    int d = s.charAt(j) - '0';
                    a[d]++;
                }
            }
            for (int j = l; j <= r; j++) {
                if (s.charAt(j) >= '0' && s.charAt(j) <= '9') {
                    int d = s.charAt(j) - '0';
                    a[d]--;
                    if (a[d] == 0) {
                        ans[i][d]++;
                        for (int k = j + 1; k <= r; k++) {
                            if (s.charAt(k) == '<') {
                                j = k;
                                break;
                            } else if (s.charAt(k) == '>') {
                                break;
                            }
                        }
                    }
                } else if (s.charAt(j) == '<') {
                    for (int k = j - 1; k >= l; k--) {
                        if (s.charAt(k) == '>') {
                            j = k;
                            break;
                        }
                    }
                } else if (s.charAt(j) == '>') {
                    for (int k = j + 1; k <= r; k++) {
                        if (s.charAt(k) == '<') {
                            j = k;
                            break;
                        }
                    }
                }
            }
        }
        for (int i = 0; i < q; i++) {
            for (int j = 0; j < 10; j++) {
                System.out.print(ans[i][j] + " ");
            }
            System.out.println();
        }
    }
}
