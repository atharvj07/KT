#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    long long n, m, d;
    cin >> n >> m >> d;
    vector<long long> a(n);
    for (long long i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    long long ans = 0;
    long long cur = 0;
    for (long long i = 0; i < n; i++) {
        if (a[i] >= cur) {
            cur = a[i] + d;
            ans++;
        }
    }
    cout << ans << endl;
    for (long long i = 0; i < n; i++) {
        long long l = -1, r = ans;
        while (r - l > 1) {
            long long mid = (l + r) / 2;
            if (a[i] >= (mid * d)) {
                l = mid;
            } else {
                r = mid;
            }
        }
        cout << l + 1 << ' ';
    }
    cout << endl;
}
