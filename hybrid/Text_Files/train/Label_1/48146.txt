import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        long[] A = new long[N];
        for (int i = 0; i < N; i++) A[i] = sc.nextInt();

        long cost = 0;
        for (long a : A) {
            cost += Math.ceil(a / Math.pow(2, M));
        }
        System.out.println(cost);
    }
}
