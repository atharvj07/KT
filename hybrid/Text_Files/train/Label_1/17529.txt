#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2001;

int a, n, m;
int l[MAXN], r[MAXN];
int x[MAXN], p[MAXN];
int ans[MAXN];
int f[MAXN][MAXN];

int main() {
  cin >> a >> n >> m;
  for (int i = 0; i < n; i++) { cin >> l[i] >> r[i]; }
  for (int i = 0; i < m; i++) { cin >> x[i] >> p[i]; }
  memset(ans, 0x3f, sizeof(ans));
  ans[0] = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (x[j] >= l[i] && x[j] < r[i]) {
        for (int k = l[i]; k <= r[i]; k++) { f[k][j] = p[j]; }
      }
    }
  }
  for (int i = 1; i <= a; i++) {
    for (int j = 0; j < m; j++) {
      if (f[i - 1][j] < 1e9) {
        for (int k = 0; k < m; k++) {
          f[i][k] = min(f[i][k], f[i - 1][j] + p[k]);
        }
      }
    }
  }
  for (int i = 0; i < m; i++) { ans[x[i]] = min(ans[x[i]], f[a][i]); }
  if (ans[a] >= 1e9) {
    cout << "-1" << endl;
    return 0;
  }
  cout << ans[a] << endl;
  return 0;
}
