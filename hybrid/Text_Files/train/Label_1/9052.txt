#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 998244353;
int pow(int x, int y) {
    int ans = 1;
    while (y) {
        if (y % 2 == 0) {
            x = x * x % mod;
            y /= 2;
        } else {
            ans = ans * x % mod;
            y--;
        }
    }
    return ans % mod;
}
int32_t main() {
    int n, k, len;
    cin >> n >> k >> len;
    vector<int> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i]; }
    int l = 0, r = 0, cnt = 0;
    while (r < n) {
        if (v[r] == -1) {
            cnt++;
        }
        while (cnt > 0 && r - l + 1 > len) {
            if (v[l] == -1) {
                cnt--;
            }
            l++;
        }
        r++;
    }
    cout << pow(k, cnt) << endl;
    return 0;
}
