import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
grid = [[int(x) for x in input().split()] for _ in range(n)]
dp = [[[-1]*(1<<11) for _ in range(m+1)] for _ in range(n+1)]

def solve(x, y, mask):
    if x == n and y == m:
        return 1 if mask == k else 0
    if dp[x][y][mask] != -1:
        return dp[x][y][mask]
    ret = solve(x+1, y, mask^grid[x][y]) + solve(x, y+1, mask^grid[x][y])
    dp[x][y][mask] = ret
    return ret

print(solve(1, 1, 0))
