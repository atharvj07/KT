#include <bits/stdc++.h>
using namespace std;
#define int long long
typedef long double ld;
int p[1000005]={0};
int32_t main()
{
    int n,k; cin>>n>>k;
    int dp[n][2]; dp[0][0]=dp[0][1]=1;
    set<int>s;
    for(int i=0;i<n;i++)
    {
        int x; cin>>x;
        s.insert(x);
    }
    for(auto &x:s) p[x]=1; 
    int ans=1;
    for(auto &x:s)
    {
        dp[x][0]=1;
        dp[x][1]=0;
        if(k%x==0 and p[k/x]==1) dp[x][0]=0;
        else dp[x][1]=1;
        if(dp[x][0]==1 or dp[x][1]==1) ans++;
        p[x]=0;
    }
    cout<<ans<<endl;
    return 0;
}
