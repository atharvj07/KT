import sys
input = sys.stdin.readline

n = int(input())
a = [0] * n
b = [0] * n
for i in range(n):
    t, ai, bi = input().split()
    t, ai, bi = t[0], int(ai), int(bi)
    a[i] = ai
    b[i] = bi

c = 0
for day in range(1, 367):
    mf = 0
    mm = 0
    for i in range(n):
        if a[i] <= day <= b[i]:
            if t == "F": mf += 1
            if t == "M": mm += 1
    if mf == mm and mf > 0: c = mf + mm

print(c)
