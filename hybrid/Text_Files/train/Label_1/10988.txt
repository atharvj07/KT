#include <bits/stdc++.h>
using namespace std;

struct Line {
  int l, r;
  Line(int l, int r) : l(l), r(r) {}
  bool operator<(const Line& other) const {
    return l < other.l;
  }
};

vector<Line> scanline;
set<Line> lines;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int type;
    cin >> type;
    if (type == 1) {
      int l, r;
      cin >> l >> r;
      scanline.emplace_back(l, r);
    } else {
      int a, b;
      cin >> a >> b;
      --a, --b;
      const auto& la = scanline[a], &lb = scanline[b];
      if (la.l < lb.l && lb.l < la.r)
        puts("YES");
      else
        puts("NO");
    }
  }
  return 0;
}
