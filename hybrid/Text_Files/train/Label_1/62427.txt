from sys import stdin
input = stdin.readline

def solve():
    n, k = map(int, input().split())
    p = list(map(int, input().split()))
    inc = []
    cur = 0
    for pp in p:
        inc.append(pp + cur)
        cur += pp
    #
    tot = 0
    for i in range(n):
        delta = max(0, int((inc[i] * k) / 100) - p[i])
        inc[i] += delta
        tot += delta
    #
    return tot

t = int(input())
for _ in range(t):
    print(solve())
