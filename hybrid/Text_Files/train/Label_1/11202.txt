#include <bits/stdc++.h>
using namespace std;

struct ITree {
    int n;
    vector<int64_t> t;

    ITree() : n() {}
    ITree(int n) : n(n) { t.resize(2 * n); }

    void set(int p, int64_t v) { set(1, 0, n - 1, p, v); }

    void set(int i, int l, int r, int p, int64_t v) {
        if (l == r) { t[i] = v; return; }
        int m = (l + r) >> 1;
        if (p <= m) set(2 * i, l, m, p, v);
        else set(2 * i + 1, m + 1, r, p, v);
        t[i] = max(t[2 * i], t[2 * i + 1]);
    }

    int64_t query(int l, int r) { return query(1, 0, n - 1, l, r); }

    int64_t query(int i, int l, int r, int a, int b) {
        if (a > r || b < l) return 0;
        if (l >= a && r <= b) return t[i];
        int m = (l + r) >> 1;
        return max(query(2 * i, l, m, a, b), query(2 * i + 1, m + 1, r, a, b));
    }
};

int main() {
    int n; cin >> n;
    vector<pair<int64_t, int64_t>> a(n);
    for (auto &x : a) cin >> x.first >> x.second;

    sort(a.begin(), a.end(),
         [](const pair<int64_t, int64_t> &x, const pair<int64_t, int64_t> &y) { return x.second < y.second; });

    ITree t(n);
    int res = 0;
    for (auto x : a) {
        int64_t lo = t.query(0, x.first - 1);
        if (lo == 0) {
            res++;
            t.set(x.first, lo + 1);
        }
    }

    cout << res << endl;

    return 0;
}
