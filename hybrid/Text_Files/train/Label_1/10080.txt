#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> image(n);
  for (int i = 0; i < n; ++i) {
    cin >> image[i];
  }

  int min_flips = n * m;
  for (int k = 2; k <= min(n, m); ++k) {
    int flips = 0;
    for (int i = 0; i < n; ++i) {
      for (int j = 0; j < m; ++j) {
        if (i % k != 0 && j % k != 0 && image[i][j] != image[i % k][j % k]) {
          ++flips;
        }
      }
    }
    min_flips = min(min_flips, flips);
  }

  cout << min_flips << endl;

  return 0;
}
