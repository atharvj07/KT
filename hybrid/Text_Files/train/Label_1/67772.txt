t = int(input())
for _ in range(t):
    n = int(input())
    p = list(map(int, input().split()))
    s = input()
    
    disliked = []
    liked = []
    for i in range(n):
        if s[i] == '0':
            disliked.append(p[i])
        else:
            liked.append(p[i])
    
    disliked.sort()
    liked.sort()
    
    q = [0] * n
    for i in range(n):
        if s[i] == '0':
            q[i] = disliked.pop(0)
        else:
            q[i] = liked.pop(0)
    
    print(' '.join(map(str, q)))
