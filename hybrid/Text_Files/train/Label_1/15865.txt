#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
const ll mod = 1e9 + 7;
const int MAXN = 3e5 + 5;

ll a[MAXN];
ll pre[MAXN];
int main() {
    ll n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + a[i];
    ll l = 0, r = 0;
    ll mx_sum = 0, cur_sum = 0;
    while (r <= n) {
        cur_sum += a[r] * r;
        cur_sum -= a[r] * l;
        while (l <= r && cur_sum < 0) {
            cur_sum -= a[l] * l;
            cur_sum += a[l] * (l + 1);
            l++;
        }
        mx_sum = max(mx_sum, cur_sum);
        r++;
    }
    cout << mx_sum << endl;
}
