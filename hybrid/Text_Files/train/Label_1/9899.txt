#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b;
    cin >> a >> b;
    string s;
    cin >> s;
    int n = s.size();
    vector<int> dp(n + 1, 1e9);
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = 0; j < i; j++) {
        if (s[j] == '0') { dp[i] = min(dp[i], dp[j] + b); }
        if (s[j] == '1') { dp[i] = min(dp[i], dp[j] + a); }
      }
    }
    cout << dp[n] << "\n";
  }
  return 0;
}
