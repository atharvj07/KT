#include <iostream>
#include <vector>
#include <cstring>
#include <utility>
using namespace std;

int main() {
    int n, m, k, s;
    cin >> n >> m >> k >> s;
    vector<vector<int> > values(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> values[i][j];
        }
    }
    vector<int> song(s);
    for (int i = 0; i < s; i++) {
        cin >> song[i];
    }
    vector<pair<int, int> > dp(s + 1);
    for (int i = 0; i < s; i++) {
        dp[i + 1] = dp[i];
        for (int j = 0; j < n; j++) {
            for (int l = 0; l < m; l++) {
                if (values[j][l] == song[i]) {
                    dp[i + 1] = max(dp[i + 1], make_pair(dp[i].first + abs(dp[i].second - l), j));
                }
            }
        }
    }
    cout << dp[s].first << endl;
    return 0;
}
