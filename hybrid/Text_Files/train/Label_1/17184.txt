#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  ll t;
  cin >> t;
  while (t--) {
    ll n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    ll moves = k - ((s[0] == '1') * (n - 1));
    int i = 1;
    while (moves && i < n) {
      if (s[i] == '0' && s[i - 1] == '1') {
        swap(s[i], s[i - 1]), moves--;
      }
      i++;
    }
    cout << s << "\n";
  }
  return 0;
}
