#include <bits/stdc++.h>
#include <ext/pb_ds/assoc_container.hpp>
#include <ext/pb_ds/tree_policy.hpp>
using namespace __gnu_pbds;
using namespace std;

typedef long long ll;
typedef long double ld;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;

int main() {
    ll m, k; cin >> m >> k;
    if (m > (1LL << k) - 1) {
        cout << -1 << endl;
        return 0;
    }
    ll ans = (1LL << (k - 1)) - 1;
    if (m == 0) {
        cout << ans << endl;
        return 0;
    }
    for (ll i = 1; i < 1000000; i++) {
        ll cnt = 0;
        for (ll j = ans + 1; j <= ans + i; j++) {
            int t = j;
            while (t > 0) {
                cnt += t % 2;
                t /= 2;
            }
        }
        if (cnt == m) {
            cout << ans + i << endl;
            return 0;
        }
    }
    cout << -1 << endl;
}
