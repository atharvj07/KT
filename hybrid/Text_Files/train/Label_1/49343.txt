import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] skills = new int[4][2];
        for (int i = 0; i < 4; i++) {
            skills[i][0] = scanner.nextInt();
            skills[i][1] = scanner.nextInt();
        }
        int[] defense = new int[2];
        int[] attack = new int[2];
        for (int i = 0; i < 4; i++) {
            if (i < 2) {
                defense[0] = Math.max(defense[0], skills[i][0]);
                attack[0] = Math.max(attack[0], skills[i][1]);
            } else {
                defense[1] = Math.max(defense[1], skills[i][0]);
                attack[1] = Math.max(attack[1], skills[i][1]);
            }
        }
        if (defense[0] > attack[1] && attack[0] > defense[1]) {
            System.out.println("Team 1");
        } else if (defense[1] > attack[0] && attack[1] > defense[0]) {
            System.out.println("Team 2");
        } else {
            System.out.println("Draw");
        }
    }
}
