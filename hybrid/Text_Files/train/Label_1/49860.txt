import java.util.*;
import java.io.*;

public class Main {
    private static final int MAX_COORD = 1000000000;

    public static void main(String[] args) throws IOException {
        InputStreamReader isr = new InputStreamReader(System.in);
        BufferedReader br = new BufferedReader(isr);

        // Read n: the number of convex polygons
        int n = Integer.parseInt(br.readLine());

        // Create an array of polygons
        Polygon[] polygons = new Polygon[n];
        for (int i = 0; i < n; i++) {
            // Read the number of vertices in the polygon
            int k = Integer.parseInt(br.readLine());

            // Create a new polygon with k vertices
            polygons[i] = new Polygon(k);

            // Read the coordinates of the polygon vertices 2
            for (int j = 0; j < k; j++) {
                String[] coords = br.readLine().split(" ");
                int x = Integer.parseInt(coords[0]);
                int y = Integer.parseInt(coords[1]);
                polygons[i].vertices[j] = new Point(x, y);
            }
        }

        // Read q: the number of Minkowski sum queries
        int q = Integer.parseInt(br.readLine());

        // For each query, calculate and print the number of vertices in the Minkowski sum
        for (int i = 0; i < q; i++) {
            String[] query = br.readLine().split(" ");
            int l = Integer.parseInt(query[0]);
            int r = Integer.parseInt(query[1]);
            Polygon sum = polygons[l - 1];
            for (int j = l + 1; j <= r; j++) {
                sum = sum.minkowskiSum(polygons[j - 1]);
            }
            System.out.println(sum.vertices.length);
        }
    }
}

class Point {
    public int x;
    public int y;

    public Point(int x, int y) {
        this.x = x;
        this.y = y;
    }
}

class Polygon {
    public Point[] vertices;

    public Polygon(int n) {
        vertices = new Point[n];
    }

    public Polygon minkowskiSum(Polygon other) {
        int n = vertices.length;
        int m = other.vertices.length;
        Polygon sum = new Polygon(n + m);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int x = vertices[i].x + other.vertices[j].x;
                int y = vertices[i].y + other.vertices[j].y;
                sum.vertices[i + j] = new Point(x, y);
            }
        }
        return sum;
    }
}
