t = int(input())
for _ in range(t):
    n, k = map(int, input().strip().split())
    dist = [n + 1] * (n + 1)
    for i in map(int, input().strip().split()):
        dist[i] = 1
    for i in range(2, n + 1):
        for j in range(i - 2, i + 2):
            if dist[i] > dist[j] + 1:
                dist[i] = dist[j] + 1
    print(dist[n])
