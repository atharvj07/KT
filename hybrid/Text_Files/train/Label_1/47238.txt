import java.util.Scanner;

public class RegularBracketSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        for (int i = 0; i < t; i++) {
            String s = scanner.nextLine();
            int moves = 0;

            while (true) {
                int openingIndex = findOpeningBracket(s);
                if (openingIndex == -1) {
                    break;
                }

                int closingIndex = findClosingBracket(s, openingIndex);
                if (closingIndex == -1) {
                    break;
                }

                s = s.substring(0, openingIndex) + s.substring(closingIndex + 1);
                moves++;
            }

            System.out.println(moves);
        }
    }

    private static int findOpeningBracket(String s) {
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == '(' || c == '[') {
                return i;
            }
        }
        return -1;
    }

    private static int findClosingBracket(String s, int openingIndex) {
        char openingBracket = s.charAt(openingIndex);
        char closingBracket = openingBracket == '(' ? ')' : ']';

        int balance = 0;
        for (int i = openingIndex + 1; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == openingBracket) {
                balance++;
            } else if (c == closingBracket) {
                balance--;
                if (balance == 0) {
                    return i;
                }
            }
        }
        return -1;
    }
}
