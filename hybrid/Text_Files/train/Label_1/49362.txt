import java.util.Scanner;

public class Main {
    static int n, m, ans = -1;
    static char[][] map;
    static boolean[][] vis;

    static void dfs(int x, int y) {
        if (x < 0 || x >= n || y < 0 || y >= m || vis[x][y]) {
            return;
        }
        vis[x][y] = true;
        if (map[x][y] != '.' && map[x][y] != '#') {
            ans = 0;
            return;
        }
        if (ans == -1) {
            ans = 1;
        }
        dfs(x + 1, y);
        dfs(x - 1, y);
        dfs(x, y + 1);
        dfs(x, y - 1);
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        n = input.nextInt();
        m = input.nextInt();
        map = new char[n][m];
        vis = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            map[i] = input.next().toCharArray();
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!vis[i][j] && map[i][j] != '#') {
                    dfs(i, j);
                }
            }
        }
        System.out.println(ans);
    }
}
