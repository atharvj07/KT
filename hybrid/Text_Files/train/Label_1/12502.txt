#include <iostream>
#include <string>
#include <vector>
#include <tuple>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  string s;
  cin >> s;
  vector<vector<tuple<int, char>>> evs(3);
  for (int i = 0; i < q; i++) {
    int t;
    cin >> t;
    if (t > 0) {
      char c;
      cin >> c;
      evs[t - 1].emplace_back(i, c);
    } else {
      evs[2 - t].pop_back();
    }
  }
  vector<string> patterns(3);
  vector<int> last(3, -1);
  for (int i = 0; i < q; i++) {
    for (int j = 0; j < 3; j++) {
      auto &ev = evs[j];
      if (last[j] >= int(ev.size()) - 1) {
        continue;
      }
      if (get<0>(ev[++last[j]]) == i) {
        patterns[j] += get<1>(ev[last[j]]);
      }
    }
    bool ok = true;
    for (int j = 0; j < 3; j++) {
      int l = -1, r = -1;
      for (int k = 0; k < n; k++) {
        if (patterns[j].size() <= k) {
          break;
        }
        if (s[k] == patterns[j][k]) {
          if (l == -1) l = k;
          r = k;
        }
      }
      if (l != -1 && (j == 0 || r < last[j - 1]) && (j == 2 || l > last[j + 1])) {
        continue;
      }
      ok = false;
    }
    if (ok) {
      cout << "YES" << '\n';
    } else {
      cout << "NO" << '\n';
    }
  }
  return 0;
}
