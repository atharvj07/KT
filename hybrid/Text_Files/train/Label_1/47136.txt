import java.util.*;
import java.io.*;

public class Main {
    static int[] arr;
    static Map<Integer, Integer> map;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        arr = new int[n+1];
        map = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            arr[i] = in.nextInt();
            map.put(arr[i], i);
        }
        for (int i = 1; i <= n; i++) {
            if (!map.containsKey(i)) {
                int pos = map.get(arr[i]);
                arr[pos] = i;
                map.put(i, pos);
            }
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(arr[i] + " ");
        }
        System.out.println();
    }
}
