#include <bits/stdc++.h>

using namespace std;

int main() {
  int x[3], y[3];
  for (int i = 0; i < 3; i++) cin >> x[i] >> y[i];
  int s = *min_element(x, x + 3) + *min_element(y, y + 3);
  if (max(*max_element(x, x + 3), *max_element(y, y + 3)) > s) s = -1;
  cout << s << endl;
  if (s == -1) return 0;
  vector<string> ans(s, string(s, '.'));
  for (int j = 0; j < 3; j++) {
    int i = 3 - 1 - j;
    if (x[i] > y[i]) swap(x[i], y[i]);
    int xnow = 0;
    while (y[i] > 0) {
      for (int jj = xnow; jj < xnow + x[i]; jj++) {
        for (int kk = 0; kk < y[i]; kk++) {
          ans[kk][jj] = 'A' + i;
        }
      }
      xnow += x[i];
      y[i] -= s - xnow;
      xnow %= s;
    }
  }
  for (const string &i : ans) cout << i << endl;
}
