def find_min_i(n, k, A):
    if any(x >= k for x in A):
        return 0
    total, i = sum(A), 0
    while True:
        A = [sum(A[:j + 1]) for j in range(n + i)]
        if any(x >= k for x in A):
            return i + 1
        total = total * 2 - A[0]
        if total >= k:
            return i + 1
        i += 1

n, k = map(int, input().split())
A = list(map(int, input().split()))
print(find_min_i(n, k, A))
