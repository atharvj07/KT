#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<string> v(n);
    for (auto &i : v) cin >> i;
    sort(v.rbegin(), v.rend());
    int ans = 0;
    for (auto &i : v) {
        int x = 0;
        for (int j = 0; j < 7; j++) {
            if (i[j] == '1') x |= 1 << j;
        }
        if (x == 0) continue;
        if (k < 0) break;
        for (int j = 0; j < 7; j++) {
            if (x & (1 << j)) {
                x ^= 1 << j;
                k--;
                if (x == 0) break;
            }
        }
        ans = ans * 10 + x;
    }
    if (k != 0) ans = -1;
    cout << ans << endl;
}
