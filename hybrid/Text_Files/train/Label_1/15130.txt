#include <bits/stdc++.h>
using namespace std;

void print(bool ok) {
  cout << (ok ? "YES" : "NO") << '\n';
}

const int N = 2e5 + 5;
int n;
vector<vector<int>> graph, rgraph;
vector<char> path;

bool dfs(int u, int prev) {
  int m = (int)graph[u].size() + (int)rgraph[u].size();
  if (m % 2 == 1) return false;
  for (auto v : graph[u]) {
    if (v != prev) {
      path.push_back('a' + graph[u].size() - 1);
      if (!dfs(v, u)) return path.pop_back(), false;
      if (rgraph[v].size() % 2 == 1) path.pop_back(), path.push_back('z' - rgraph[v].size());
      else path.pop_back();
    }
  }
  for (auto v : rgraph[u]) {
    if (v != prev) {
      path.push_back('z' - rgraph[u].size() + 1);
      if (!dfs(v, u)) return path.pop_back(), false;
      path.pop_back(), path.push_back('a' + graph[v].size());
    }
  }
  return true;
}

int main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  int m, x, y;
  char c;
  cin >> n >> m;
  graph.resize(n + 1), rgraph.resize(n + 1);
  while (m--) {
    cin >> c;
    if (c == '+') {
      cin >> x >> y >> c;
      graph[x].push_back(y), rgraph[y].push_back(x);
    } else if (c == '-') {
      cin >> x >> y;
      for (int i = 0; i < graph[x].size(); i++) {
        if (graph[x][i] == y) {
          graph[x].erase(graph[x].begin() + i);
          for (int j = 0; j < rgraph[y].size(); j++) {
            if (rgraph[y][j] == x) {
              rgraph[y].erase(rgraph[y].begin() + j);
              break;
            }
          }
          break;
        }
      }
    } else {
      cin >> x;
      path.clear();
      bool ok = dfs(1, 0);
      if (!ok) print(false);
      else {
        print(true);
        while (!path.empty()) {
          cout << path.back();
          path.pop_back();
        }
        cout << '\n';
      }
    }
  }
  return 0;
}
