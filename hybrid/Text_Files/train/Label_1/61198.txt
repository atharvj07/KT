import sys
from bisect import bisect_left, bisect_right

input = sys.stdin.readline

n = int(input())
v = sorted(list(map(int, input().split())))
u = sorted([(x, i) for i, x in enumerate(v)])
m = int(input())

for _ in range(m):
    t, l, r = map(int, input().split())
    l -= 1
    r -= 1
    if t == 1:
        print(sum(v[l:r + 1]))
    else:
        a, b = u[l][1], u[r][1]
        print(sum(v[a:b + 1]))
