import sys

inp = sys.stdin.readlines()

n = int(inp[0])
l = list(map(int, inp[1].split()))

ans = 0
for i in range(n):
    if l[i] > 0: continue
    cnt = 0
    for j in range(n):
        if i == j or l[j] == -1: continue
        if l[j] < l[i]: cnt += 1
    ans += i * cnt

mod = 998244353
print(ans % mod)
