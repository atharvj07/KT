import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int V = in.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = in.nextInt();
        }
        in.close();
        double x = 1e9;
        for (int i = 0; i < n; i++) {
            if (a[i] > 0) {
                x = Math.min(x, (double) b[i] / a[i]);
            }
        }
        double ans = Math.min(x, V);
        System.out.println(ans);
    }
}
