#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<long long, long long>> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i].first >> v[i].second; }
  sort(v.begin(), v.end());
  long long sum = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    ans += sum + v[i].second * (n - i - 1);
    sum += v[i].first;
  }
  cout << ans << endl;
  return 0;
}
