#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
const int MAXM = 3002;
int n, m, p[MAXM], c[MAXM], cnt[MAXM];

inline bool cmp(const int &a, const int &b) { return p[a] < p[b]; }

int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> p[i] >> c[i];

  for (int i = 1; i <= n; i++) {
    cnt[p[i]]++;
    m = max(m, p[i]);
  }

  vector<int> tochange, untouched;
  for (int i = 1; i <= n; i++) {
    if (p[i] != 1)
      tochange.push_back(i);
    else
      untouched.push_back(i);
  }

  if (2 * cnt[1] > n) {
    cout << "0\n";
    return 0;
  }

  sort(tochange.begin(), tochange.end(), cmp);
  int ans = 0, sum = cnt[1], nt = 0;
  for (int i = 0; i < tochange.size(); i++) {
    ans += c[tochange[i]];
    sum++;
    nt++;
    if (2 * sum > n) break;
  }

  cout << ans << endl;
  return 0;
}
