#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  string a;
  cin >> a;
  vector<int> b(a.length());
  for (int i = 0; i < a.length(); i++) {
    b[i] = (a[i] - '0');
  }
  int l = 0, r = 0;
  for (int i = 0; i < a.length(); i++) {
    if (b[i] == 1) l = i;
  }
  for (int i = a.length() - 1; i >= 0; i--) {
    if (b[i] == 1) r = i;
  }
  int ans;
  if (r - l <= 1) {
    ans = 1;
  } else {
    if (l < a.length() - 1 - r) {
      ans = r;
    } else {
      ans = l;
    }
  }
  b.erase(b.begin() + ans);
  for (int x : b) { cout << x; }
  cout << "\n";
  return 0;
}
