n=int(input())
a=sorted(map(int,input().split()))
if len(set(a))!=n:
    for i in a:
        if a.count(i)>1:
            a.remove(i)
            break
ans=[a[0]]
b=[i for i in a if i>a[0]]
while b:
    if b[0]<ans[-1]:
        b.pop(0)
        continue
    if b[-1]>ans[-1]:
        ans.append(b[-1])
        b.pop()
    else:
        if b[0]<b[-1]:
            ans.append(b[0])
            b.pop(0)
        else:
            ans.append(b[-1])
            b.pop(-1)
print(len(ans))
for i in ans:
    if i==a[0]:
        print('L',end='')
        a.remove(a[0])
    else:
        print('R',end='')
        a.remove(i)
