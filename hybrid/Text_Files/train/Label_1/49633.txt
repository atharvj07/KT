import java.util.*;
import java.io.*;

class Main {

    static int[] a;
    static int[] used;
    static int[] ans;
    static int n;
    static int least = Integer.MAX_VALUE;

    static void backtrack(int sum, int cnt) {
        if (cnt == n) {
            if (sum < least) {
                least = sum;
                for (int i = 0; i < n; i++) {
                    ans[i] = used[i];
                }
            }
            return;
        }
        for (int i = 0; i < n; i++) {
            if (used[i] == 0) {
                used[i] = cnt + 1;
                backtrack(sum + (a[i] * cnt + 1), cnt + 1);
                used[i] = 0;
            }
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter pw = new PrintWriter(System.out);
        n = Integer.parseInt(br.readLine());
        a = new int[n];
        used = new int[n];
        ans = new int[n];
        String[] s = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(s[i]);
        }
        backtrack(0, 0);
        pw.println(least);
        for (int i = 0; i < n; i++) {
            pw.print(ans[i] + " ");
        }
        pw.println();
        pw.close();
    }
}
