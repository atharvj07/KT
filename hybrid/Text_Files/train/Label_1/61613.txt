import sys

t = int(input())

for _ in range(t):
    n = int(input())
    b = list(map(int, input().split()))
    a = [0] * (2 * n)
    for i in range(n):
        a[2 * i] = b[i]
        a[2 * i + 1] = min(a[2 * i], b[(i + 1) % n])
    if any(a[i] > a[i + 1] for i in range(2 * n - 1)):
        print(-1)
    else:
        print(*a)
    sys.stdout.flush()
