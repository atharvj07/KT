#include <bits/stdc++.h>
using namespace std;
int main() {
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<pair<int, int>> v(n);
        for (int i = 0; i < n; i++) {
            int a, b;
            cin >> a >> b;
            v[i] = {a, b};
        }
        long long ans = 0;
        for (int i = 1; i < n; i++) {
            if (v[i].first == v[i - 1].first) {
                int diff = v[i].second - v[i - 1].second;
                if (diff > 0) ans += diff;
                else {
                    v[i] = {v[i].first + 1, v[i].second};
                    ans += v[i].second;
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
