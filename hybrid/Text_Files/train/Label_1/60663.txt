from itertools import product

r, c, n, k = map(int, input().split())
v = [[(i == 0 and j == 0) for j in range(c)] for i in range(r)]
for i in range(n):
    x, y = map(int, input().split())
    x -= 1
    y -= 1
    v[x][y] = True

ans = 0
for i in range(r):
    for j in range(c):
        for xx in range(i, r):
            for yy in range(j, c):
                if sum(sum(row[j:yy + 1]) for row in v[i:xx + 1]) >= k:
                    ans += 1
print(ans)
