n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
x = int(input())
ans = 0
for i in range(n):
    for j in range(m):
        c, d = max(x // a[i], 1), max(x // b[j], 1)
        x1, x2 = i, i + c - 1
        y1, y2 = j, j + d - 1
        if x1 <= x2 and y1 <= y2 and a[i] * b[j] * (x2 - x1 + 1) * (y2 - y1 + 1) <= x:
            ans = max(ans, (x2 - x1 + 1) * (y2 - y1 + 1))
print(ans)
