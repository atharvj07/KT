import java.util.*;

public class Main {
  static int n, m;
  static int[] deg = new int[100001];
  static ArrayList<Integer>[] adj = new ArrayList[100001];
  static ArrayList<Integer> ans = new ArrayList<>();
  static int[] vis = new int[100001];

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    m = sc.nextInt();

    for (int i = 1; i <= n; i++) {
      adj[i] = new ArrayList<>();
    }

    for (int i = 0; i < m; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();

      adj[a].add(b);
      adj[b].add(a);

      deg[a]++;
      deg[b]++;
    }

    for (int i = 1; i <= n; i++) {
      if (deg[i] == 2) {
        dfs(i, -1);
      }
    }

    System.out.println(ans.size());
    for (int x : ans) {
      System.out.print(x + " ");
    }
  }

  static void dfs(int x, int par) {
    vis[x] = 1;

    for (int u : adj[x]) {
      if (u != par) {
        if (vis[u] == 1) {
          ans.add(u);
          ans.add(x);
        } else {
          dfs(u, x);
        }
      }
    }
  }
}
