import sys

input = sys.stdin.readline

k = int(input())
lst = []
for _ in range(k):
    n = int(input())
    a = list(map(int, input().split()))
    lst.append((n, a))

for i in range(k):
    for j in range(k):
        if i == j:
            continue
        a, b = lst[i]
        c, d = lst[j]
        for x in range(a):
            for y in range(c):
                if sum(a) - b[x] == sum(c) - d[y]:
                    print("YES")
                    print(i + 1, x + 1)
                    print(j + 1, y + 1)
                    exit()

print("NO")
