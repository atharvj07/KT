import sys

n = int(input())
a = [[int(x) for x in input().split()] for _ in range(n)]

b = [-1] * n
for i in range(n):
    for j in range(n):
        if i == j: continue
        if a[i][1] == a[j][1] ^ 1:
            b[i], b[j] = j, i
            break
        elif a[i][1] == a[j][1]:
            b[i], b[j] = j, i

edges = 0
for i in range(n):
    if b[i] == -1: continue
    edges += 1
    print(i, b[i])
print(edges)
