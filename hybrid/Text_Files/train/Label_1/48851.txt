import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }
            boolean allSame = true;
            int xor = a[0];
            a[0] = 0;
            for (int i = 1; i < n; i++) {
                int x = a[i];
                a[i] = 0;
                if (i == n - 1) {
                    xor ^= x;
                } else if (xor != x) {
                    allSame = false;
                }
            }
            if (allSame) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
        sc.close();
    }
}
