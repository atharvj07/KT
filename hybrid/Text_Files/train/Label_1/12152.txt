#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(m);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < m; i++) cin >> b[i];
  sort(a.begin(), a.end());
  sort(b.rbegin(), b.rend());
  long long ans = LLONG_MAX;
  long long sum = 0;
  if (a[0] > b[0]) {
    ans = 0;
  } else {
    for (int i = 0; i < m; i++) {
      if (a[0] <= b[i]) {
        sum = b[i] - a[0];
        ans = min(ans, sum);
      }
    }
  }
  cout << ans << endl;
  return 0;
}
