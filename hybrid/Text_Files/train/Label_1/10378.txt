#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll md = 998244353;
int32_t main() {
    int n;
    cin >> n;
    string a[n + 1];
    int l = 0;
    ll sum = 0;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        l = max(l, (int)a[i].size());
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            int x = (int)a[i].size(), y = (int)a[j].size();
            int p = 0, q = 0;
            string ans = "";
            while (p != x || q != y) {
                if (p == x)
                    ans += a[j][q++];
                else if (q == y)
                    ans += a[i][p++];
                else {
                    if ((x - p) >= (y - q)) {
                        ans += a[i][p++];
                    } else {
                        ans += a[j][q++];
                    }
                }
            }
            ll cur = 0;
            int f = 1;
            for (int k = (int)ans.size() - 1; k >= 0; k--) {
                cur = cur + (ll)(ans[k] - '0') * f;
                cur %= md;
                f *= 10;
                f %= md;
            }
            sum += cur;
            sum %= md;
        }
    }
    cout << sum << endl;
}
