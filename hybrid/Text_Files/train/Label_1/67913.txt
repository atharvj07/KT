import queue
n,m,p=map(int,input().split())
s=[int(a) for a in input().split()]
grid=[list(input()) for i in range(n)]
nr=[1,-1,0,0]
nc=[0,0,1,-1]

def valid_cell(u,v):
    if u>=0 and u<n and v>=0 and v<m and grid[u][v]!='#' and (grid[u][v]=='.') :
        return True
    return False

def bfs(u,v,i):
    que=queue.Queue()
    que.put((u,v,1))
    while not que.empty():
        uu,vv,count=que.get()
        if count<=s[i-1]:
            grid[uu][vv]=str(i)
            for i in range(4):
                r,c=uu+nr[i],vv+nc[i]
                if valid_cell(r,c):
                    que.put((r,c,count+1))

for i in range(n):
    for j in range(m):
        if grid[i][j].isdigit():
            bfs(i,j,int(grid[i][j]))
    
for i in range(p):
    cnt=0
    for a in range(n):
        for b in range(m):
            if grid[a][b]==str(i+1):
                cnt+=1
    print(cnt,end=' ')
