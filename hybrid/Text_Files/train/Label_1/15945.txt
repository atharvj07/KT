#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(k);
  for (int i = 0; i < k; i++) { cin >> a[i]; }
  int m = n * k;
  vector<bool> used(m + 1);
  for (int i = 0; i < k; i++) { used[a[i]] = true; }
  int pos = 1;
  for (int i = 0; i < k; i++) {
    for (int j = 1; j <= m; j++) {
      if (used[j]) { cout << j << " "; } else {
        used[j] = true;
        a[i] = j;
      }
    }
    cout << endl;
  }
}
