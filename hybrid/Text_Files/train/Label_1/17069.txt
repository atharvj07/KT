#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) {
    a[i] = a[i] % 4;
  }
  for (int i = 0; i < n; i++) {
    int x = a[i];
    a[i] = 4 - a[i];
    int j = n - i - 1;
    a[j] = (a[j] + x) % 4;
  }
  bool win = true;
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) win = false;
  }
  if (win)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
}
