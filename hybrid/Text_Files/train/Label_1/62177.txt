import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, m = map(int, input().split())
    p = list(map(int, input().split()))
    ans = []
    for i in range(n):
        if p[i] != p[(i-m) % n]:
            break
        ans.append((i-m) % n)
    print(len(ans))
    print(*ans)
