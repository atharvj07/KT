import sys
input = sys.stdin.readline

n, k = map(int, input().split())
l, r = [], []
for _ in range(n):
    _l, _r = map(int, input().split())
    l.append(_l)
    r.append(_r)
L, R = -1e9, 1e9
ans = 0
for i in range(k):
    if l[i] > L: L = l[i]
    if r[i] < R: R = r[i]
if L > R: ans = 0
else:
    for i in range(n):
        if L <= l[i] and r[i] <= R: ans += 1
print(ans)
for i in range(k):
    if l[i] == L and r[i] == R:
        print(i + 1, end=' ')
