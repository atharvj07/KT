import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        List<Integer>[] g = new List[n];
        for (int i = 0; i < n; i++) g[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            g[a].add(b);
            g[b].add(a);
        }
        int[] cnt = new int[n];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '1') cnt[i]++;
        }
        int[][] dp = new int[n][1 << 11];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], -1);
        }
        int res = dfs(0, 1 << 0, g, cnt, dp, s);
        System.out.println(res == -1 ? -1 : res / 2);
    }

    private static int dfs(int cur, int mask, List<Integer>[] g, int[] cnt, int[][] dp, String s) {
        if (dp[cur][mask] != -1) return dp[cur][mask];
        dp[cur][mask] = Integer.MAX_VALUE;
        int sum = cnt[cur];
        for (int adj : g[cur]) {
            if ((mask & (1 << adj)) != 0) {
                sum += cnt[adj];
            }
        }
        if (sum == 0) {
            dp[cur][mask] = 0;
            return 0;
        }
        for (int adj : g[cur]) {
            if ((mask & (1 << adj)) != 0) {
                int nmask = mask ^ (1 << adj);
                dp[cur][mask] = Math.min(dp[cur][mask], dfs(adj, nmask, g, cnt, dp, s) + 2);
            }
        }
        if (s.charAt(cur) == '1') {
            for (int i = 0; i < g[cur].size(); i++) {
                int adj = g[cur].get(i);
                if ((mask & (1 << adj)) == 0) {
                    int nmask = mask | (1 << adj);
                    dp[cur][mask] = Math.min(dp[cur][mask], dfs(adj, nmask, g, cnt, dp, s) + 1);
                }
            }
        }
        return dp[cur][mask];
    }
}
