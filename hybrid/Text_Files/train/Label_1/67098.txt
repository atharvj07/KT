n, m = map(int, input().split())
L = [input() for _ in range(n)]
p = [[[(0, 0)] * m for _ in range(m)] for _ in range(n)]
for i in range(n):
    a = L[i]
    for j in range(m):
        for k in range(m - 1, -1, -1):
            if j == k:
                p[i][j][k] = (j, k)
            elif a[j] == a[k]:
                p[i][j][k] = p[i][j + 1][k - 1] if j + 1 <= k - 1 else (j, k)
            else:
                p[i][j][k] = max(p[i][j + 1][k], p[i][j][k - 1], key=lambda x: x[1] - x[0])
c = 0
for i in range(n):
    l, r = p[i][0][m - 1]
    if r - l + 1 > c:
        c = r - l + 1
        t = L[i][l:r + 1]
for i in range(1, n):
    for j in range(m):
        for k in range(m - 1, -1, -1):
            l1, r1 = p[i][j][k]
            l2, r2 = p[i - 1][m - 1 - r1][m - 1 - l1]
            if r2 - l2 + 1 + r1 - l1 + 1 > c:
                c = r2 - l2 + 1 + r1 - l1 + 1
                t = L[i][l1:r1 + 1] + L[i - 1][l2:r2 + 1]
print(c)
if c > 0:
    print(t)
