#include<bits/stdc++.h>
#define ll long long
#define N 100005
using namespace std;
char eq[105];
int main() {
    cin>>eq;
    int len=strlen(eq),x=0,y=0,z=0;
    for(int i=0; i<len; i++) if(eq[i]=='|') x++;
    for(int i=0; i<len; i++) if(eq[i]=='+') y=i;
    for(int i=0; i<len; i++) if(eq[i]=='=') z=i;
    if(x==(y-1) && z-y==(x+1)) cout<<eq<<endl;
    else if(z-y==(x+1)) {
        x=eq[x];
        eq[x]=eq[y-1];
        eq[y-1]=x;
        cout<<eq<<endl;
    }
    else cout<<"Impossible"<<endl;
}
