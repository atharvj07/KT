#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll func(ll w,ll m,ll k)
{
    ll sum(0),dig(0),temp(m),ans(0);
    while(temp)
        dig++,temp/=10;
    while(sum<=w)
    {
        sum+=dig*k;
        if(sum<=w)
            ans++;
        else
            break;
        dig++;
        m++;
    }
    return ans;
}
int main()
{
    ll w,m,k;
    cin>>w>>m>>k;
    cout<<func(w,m,k)<<endl;
    return 0;
}
