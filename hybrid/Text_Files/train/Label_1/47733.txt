import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long[][] ab = new long[(int) n][2];
        for (int i = 0; i < n; i++) {
            ab[i][0] = sc.nextLong();
            ab[i][1] = sc.nextLong();
        }
        sc.close();

        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += (ab[i][0] - 1) / ab[i][1];
        }
        System.out.println(ans);
    }
}
