#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  getline(cin, s);
  int cnt = 0;
  char b[7] = "Bulbas";
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == 'B') {
      if (i + 5 <= s.size() && s.substr(i, 6) == b) {
        cnt++;
        i += 5;
        continue;
      }
    }
    if (s[i] == 'b') {
      if (i + 5 <= s.size() && s[i + 1] - 'a' >= 0 && s[i + 1] - 'a' <= 25 && s[i + 2] - 'a' >= 0 && s[i + 2] - 'a' <= 25 &&
          s[i + 3] - 'a' >= 0 && s[i + 3] - 'a' <= 25 && s[i + 4] - 'a' >= 0 && s[i + 4] - 'a' <= 25 && s[i + 5] == 'r') {
        cnt++;
        i += 5;
        continue;
      }
    }
  }
  cout << cnt << endl;
  return 0;
}
