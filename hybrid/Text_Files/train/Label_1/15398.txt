#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<int> v(n), inv(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
        inv[v[i]] = i;
    }
    int sm = n;
    for (int i = 0; i < m; i++) {
        if (inv[i] < sm) {
            sm = inv[i];
        }
    }
    cout << m - 1 - sm << endl;
    return 0;
}
