#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            a[i]--;
        }
        if (k > n) {
            cout << -1 << '\n';
            continue;
        }
        vector<int> pos(n, -1);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (pos[a[i]] != -1) {
                ans = max(ans, pos[a[i]] - i);
            }
            pos[a[i]] = i;
        }
        cout << ans << '\n';
    }
    return 0;
}
