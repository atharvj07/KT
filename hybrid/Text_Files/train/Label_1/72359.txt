import queue

n = int(input())
tasks = []
for i in range(n):
    tasks.append(list(map(int, input().split())))

q = queue.Queue()
last = 0
max_q = 0

for t, c in tasks:
    while last < t and not q.empty():
        q.get()
        last += 1
    if last < t:
        last = t
    for _ in range(c):
        q.put(t)
        last += 1
    max_q = max(max_q, q.qsize())

print(last, max_q)
