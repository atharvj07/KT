#include <iostream>
#include <vector>
#include <queue>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, k;
        cin >> n >> m >> k;
        vector<int> h(n);
        for (int i = 0; i < n; i++) {
            cin >> h[i];
        }
        if (h[0] == h[n - 1]) {
            cout << "YES" << endl;
            continue;
        }
        vector<int> dist(n, -1);
        queue<int> q;
        q.push(0);
        dist[0] = m;
        while (!q.empty()) {
            int cur = q.front();
            q.pop();
            for (int i = 1; i <= k; i++) {
                int next = cur + i;
                if (next < n && dist[next] == -1) {
                    dist[next] = dist[cur] + 1;
                    if (h[cur] + dist[cur] - 1 >= h[next]) {
                        q.push(next);
                    }
                }
            }
        }
        if (dist[n - 1] != -1) {
            cout << "YES" << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
