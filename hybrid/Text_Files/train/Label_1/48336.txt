import java.util.*;
import java.io.*;

public class Main {
    static int n, k, ans[];
    static int[][] a;
    static boolean[] b;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        String s = sc.next();
        ans = new int[n + 1];
        a = new int[k + 1][];
        b = new boolean[n + 1];
        for (int i = 1; i <= k; i++) {
            int c = sc.nextInt();
            a[i] = new int[c];
            for (int j = 0; j < c; j++) {
                a[i][j] = sc.nextInt();
                if (s.charAt(a[i][j] - 1) == '0') b[a[i][j]] = true;
            }
        }
        dfs(1, 0);
        for (int i = 1; i <= n; i++) {
            System.out.println(ans[i]);
        }
    }

    static void dfs(int x, int dep) {
        if (x > n) return;
        ans[x] = dep;
        for (int i = 1; i <= k; i++) {
            boolean fl = true;
            for (int j = 0; j < a[i].length; j++) {
                if (a[i][j] <= x && b[a[i][j]]) {
                    fl = false;
                    break;
                }
            }
            if (fl) dfs(x + 1, dep + 1);
        }
    }
}
