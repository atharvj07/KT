for _ in range(int(input())):
    n, k = map(int, input().split())
    if (k % 2 == 0 and n % 2 == 1) or (k % 2 == 1 and n % 2 == 0) or k > n:
        print("NO")
    else:
        print("YES")
        ans = [0]*k
        if n % 2 == 0:
            n //= 2
            k //= 2
            for i in range(k-1):
                ans[i] = 1
            if k % 2 == 0:
                ans[k-1] = n - k + 2
            else:
                ans[k-1] = n - k + 1
        else:
            n //= 2
            k -= 1
            ans[0] = n - k + 1
            for i in range(1, k):
                ans[i] = 1
        print(*ans)
