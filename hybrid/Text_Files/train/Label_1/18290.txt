#include<bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<ll, ll> pll;

const int maxN = 2e5+5;

ll n, m, a[maxN], lazy[maxN<<2], f[maxN];

#define lc (rt<<1)
#define rc (rt<<1|1)
#define mid ((l+r)>>1)

void build(ll rt, ll l, ll r)
{
    lazy[rt] = 0;
    if(l == r)
    {
        f[l] = 1;
        return;
    }
    build(lc, l, mid);
    build(rc, mid+1, r);
}

void pushdown(ll rt, ll l, ll r)
{
    if(lazy[rt] == 0)
        return;
    a[l] += lazy[rt];
    a[r] += lazy[rt];
    if(l != r)
    {
        lazy[lc] += lazy[rt];
        lazy[rc] += lazy[rt];
    }
    lazy[rt] = 0;
}

void upd(ll rt, ll l, ll r, ll ql, ll qr, pll val)
{
    if(l > qr || r < ql)
        return;
    if(ql <= l && r <= qr)
    {
        if(val.first == 1)
        {
            a[l] = val.second;
            if(l != r)
            {
                lazy[lc] = lazy[rc] = 0;
                f[l] = 1, f[r] = 1;
            }
        }
        else if(val.first == 3)
        {
            lazy[rt] += val.second;
        }
        return;
    }
    pushdown(rt, l, r);
    upd(lc, l, mid, ql, qr, val);
    upd(rc, mid+1, r, ql, qr, val);
}

ll query(ll rt, ll l, ll r, ll ql, ll qr)
{
    if(l > qr || r < ql)
        return 0;
    if(ql <= l && r <= qr)
    {
        ll ans = 0;
        for(ll i = l; i <= r; i++)
            ans = (ans + (f[i]*a[i])%1000000000)%1000000000;
        return ans;
    }
    pushdown(rt, l, r);
    return (query(lc, l, mid, ql, qr) + query(rc, mid+1, r, ql, qr))%1000000000;
}

int main()
{
    cin >> n >> m;
    for(ll i = 1; i <= n; i++)
        cin >> a[i], f[i] = 1;
    build(1, 1, n);
    for(ll i = 1; i <= m; i++)
    {
        ll type, l, r, x, v, d;
        cin >> type;
        if(type == 1)
            cin >> x >> v, upd(1, 1, n, x, x, {type, v});
        else if(type == 2)
            cin >> l >> r, cout << query(1, 1, n, l, r) << endl;
        else if(type == 3)
        {
            cin >> l >> r >> d;
            pll val = {type, d};
            upd(1, 1, n, l, r, val);
        }
    }
    return 0;
}
