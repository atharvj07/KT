#include<bits/stdc++.h>
using namespace std;

const int maxn = 2e5 + 5, mod = 1e9 + 7;
int n, fa[maxn], a[maxn], b[maxn], c[maxn];
vector<int> G[maxn];

int get(int u) { return fa[u] == u ? u : fa[u] = get(fa[u]); }

void join(int u, int v) { fa[get(u)] = get(v); }

struct BIT {
    int n;
    vector<int> bit;
    void init(int _n) { n = _n; bit.assign(n, 0); }
    void update(int idx, int val) { for (; idx <= n; idx += idx & -idx) bit[idx] += val; }
    int query(int idx) { int ret = 0; for (; idx > 0; idx -= idx & -idx) ret += bit[idx]; return ret; }
    int query(int l, int r) { return query(r) - (l > 0 ? query(l - 1) : 0); }
} bit;

int dfs(int u, int p, int root) {
    int sum = 0;
    for (int v : G[u]) {
        if (v == p) continue;
        sum = (sum + dfs(v, u, root)) % mod;
    }
    if (b[u] == c[u]) {
        bit.update(u, a[u]);
        sum = (sum + a[u]) % mod;
    }
    else if (b[root] == c[root]) {
        int tmp = bit.query(u);
        sum = (sum + 1ll * a[u] * tmp % mod) % mod;
        bit.update(u, -a[u]);
    }
    return sum;
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i] >> c[i];
        fa[i] = i;
    }
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        G[u].emplace_back(v);
        G[v].emplace_back(u);
        join(u, v);
    }
    bit.init(n);
    int root = get(1);
    int ans = dfs(1, 1, root);
    if (c[root] == 0) {
        ans = -1;
    }
    cout << ans << '\n';
    return 0;
}
