#include <cstdio>

using ll = long long;

const int maxl = 2e5 + 10;

int n, k, q;
ll l[maxl], r[maxl];
int deg[maxl];

int main() {
	scanf("%d%d%d", &n, &k, &q);
	for (int i = 1; i <= n; ++i) {
		scanf("%lld%lld", &l[i], &r[i]);
		for (int j = l[i]; j <= r[i]; ++j) { deg[j]++; }
	}
	for (int a, b, i = 1; i <= q; ++i) {
		scanf("%d%d", &a, &b);
		int ans = 0;
		for (int j = a; j <= b; ++j) { ans += deg[j] >= k; }
		printf("%d\n", ans);
	}
	return 0;
}
