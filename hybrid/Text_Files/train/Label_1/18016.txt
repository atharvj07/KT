#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  string t = "abc";
  int mn = 1e9;
  int cnta = 0, cntb = 0, cntc = 0;
  for (char c : s) {
    cnta += c == 'a', cntb += c == 'b', cntc += c == 'c';
    mn = min({mn, cnta, cntb});
  }
  cout << (cnta >= mn && cntb >= mn && cntc == mn ? "YES" : "NO") << endl;
  return 0;
}
