import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int p = sc.nextInt();
        if (n > 0 && p < 0 && (-p) % 2 != 0) {
            System.out.println(-1);
            return;
        }
        long ans = 0;
        while (n > 0) {
            if ((n & 1) == 1) {
                n += p;
            }
            if (n % 2 != 0) {
                System.out.println(-1);
                return;
            }
            n >>= 1;
            ans++;
        }
        System.out.println(ans);
    }
}
