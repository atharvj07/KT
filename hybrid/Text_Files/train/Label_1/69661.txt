q = int(input())
for _ in range(q):
    n, m, k = map(int, input().split())
    if max(n, m) > k:
        print(-1)
    else:
        diff = abs(n - m)
        if diff % 2 == 0:
            if k % 2 == (n + m) % 2:
                print(k - diff)
            else:
                print(k - 2)
        else:
            print(k - 1)
