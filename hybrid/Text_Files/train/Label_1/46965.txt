import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int[] events = new int[n];
    for (int i = 0; i < n; i++) {
      events[i] = scanner.nextInt();
    }
    int count = 0;
    int free = 0;
    for (int i = 0; i < n; i++) {
      if (events[i] == -1) {
        if (free == 0) {
          count++;
        } else {
          free--;
        }
      } else {
        free += events[i];
      }
    }
    System.out.println(count);
  }
}
