#include <iostream>
#include <vector>

using namespace std;
#define ll long long

const int MAXN = 1e5 + 1;
const int INF = 1e9;

int main() {
    int n, m, k;
    cin >> n >> m >> k;

    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }

    vector<int> b(m);
    for (int i = 0; i < m; i++) { cin >> b[i]; }

    vector<int> pos(n + 1);
    for (int i = 0; i < n; i++) { pos[a[i]] = i; }

    int cur = 1;
    int ans = 0;
    for (int i = 0; i < m; i++) {
        if (pos[b[i]] <= cur) {
            ans += pos[b[i]];
        } else {
            int x = (pos[b[i]] - cur) / k;
            ans += (pos[b[i]] - cur) % k + k * x;
        }
        pos[b[i]] = cur++;
    }

    cout << ans << endl;
}
