#include<iostream>
using namespace std;
int main(){
    long long int n;
    cin >> n;
    long long int x;
    while (n > 0){
        x = n % 10;
        n /= 10;
        if (x == 0) cout << "O-|-OOOO" << endl;
        if (x == 1) cout << "O-|OOO-O" << endl;
        if (x == 2) cout << "O-|OO-OO" << endl;
        if (x == 3) cout << "-O|OOO-O" << endl;
        if (x == 4) cout << "-O|OO-OO" << endl;
        if (x == 5) cout << "-|OOOOO" << endl;
        if (x == 6) cout << "-|OOO-O" << endl;
        if (x == 7) cout << "O-|O-OOO" << endl;
        if (x == 8) cout << "O--|OOO" << endl;
        if (x == 9) cout << "O--|O-OO" << endl;
    }
    return 0;
}
