t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    max_a = max(a)
    k = 0
    while True:
        x = a[-1]
        if x == max_a:
            break
        left = [i for i in a if i <= x]
        right = [i for i in a if i > x]
        a = left + right
        k += 1
    print(k)
