#include<bits/stdc++.h>
using namespace std;
const int mxN=2e5+5;
int a[mxN];
int sum[mxN];
int main() {
    int n, m;
    cin>>n>>m;
    for(int i=0; i<n; ++i) {
        cin>>a[i];
    }
    int sm=0;
    for(int i=0; i<n; ++i) {
        sm+=a[i];
        sum[i]=sm;
    }
    while(m--) {
        int l, r;
        cin>>l>>r;
        --l, --r;
        cout<<(l?sum[r]-sum[l-1]:sum[r])?0:1<<'\n';
    }
}
