#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n), t(n);
  for (int i = 0; i < n; ++i) { std::cin >> a[i]; }
  for (int i = 0; i < n; ++i) { std::cin >> t[i]; }
  std::vector<int> dp(n);
  dp[k - 1] = t[k - 1] * a[k - 1];
  for (int i = k; i < n; ++i) {
    dp[i] = std::max(dp[i - 1] + t[i] * a[i], t[i] * a[i]);
  }
  int ans = 0;
  for (int i = 0; i < k - 1; ++i) {
    int tmp = 0;
    for (int j = i; j < i + k; ++j) {
      tmp += t[j] * a[j];
    }
    ans = std::max(ans, tmp + dp[n - 1 - (i + k - 1)]);
  }
  std::cout << ans << std::endl;
  return 0;
}
