import sys
from collections import deque,defaultdict
input = sys.stdin.readline

n,m=map(int,input().split())
sx,sy,fx,fy=map(int,input().split())
lo=[tuple(map(int,input().split())) for i in range(m)]
st=(sx,sy),0
en=(fx,fy)
g=defaultdict(list)
for x,y in lo:
 for dx in (-1,0,1):
  for dy in (-1,0,1):
   if 0<=x+dx<=n and 0<=y+dy<=n:
    g[(x,y)].append((x+dx,y+dy))
vis=set()
q=deque([st])
while q:
 cur,d=q.popleft()
 if cur==en: print(d); break
 if cur in vis: continue
 vis.add(cur)
 for nxt in g[cur]:
  q.append((nxt,d+1))
else: print(-1)
