#include <iostream>
using namespace std;
int n, k, m, t;
int main()
{
    cin >> n >> k >> m >> t;
    while (t--)
    {
        int op, v;
        cin >> op >> v;
        if (op == 0)
        {
            if (v < k)
                k--;
            n--;
        }
        else
        {
            n++;
            if (v <= k)
                k++;
        }
        cout << n << " " << k << endl;
    }
    return 0;
}
