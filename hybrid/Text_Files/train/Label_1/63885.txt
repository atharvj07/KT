import sys
input = sys.stdin.readline

n = int(input())
points = []
for _ in range(n):
    x,y = map(int, input().split())
    points.append((x,y))

ans = 'NO'
for i in range(n):
    for j in range(i+1,n):
        x1,y1 = points[i]
        x2,y2 = points[j]
        if (x2-x1)*(x2-x1) + (y2-y1)*(y2-y1) != 0:
            ans = 'YES'
            break
    if ans=='YES':
        break
print(ans)
