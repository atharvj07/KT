import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int k = (int) Math.sqrt(n * m + 1);
        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.println(i + " " + (n - 1 - i));
        }
    }
}
