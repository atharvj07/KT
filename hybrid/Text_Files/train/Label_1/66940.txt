s=input()
t=input()
cnt=0
ans=""
last=-1
for i in s:
    if i=="?":
        if cnt==0: ans+="a"
        else: 
            ans+=t[cnt-1]
            ans+=i
            cnt+=1
    else:
        ans+=i
        if i==t[cnt]: cnt+=1
        if cnt==len(t):
            last=i
            ans=ans[:-len(t)]
            cnt=0
if last!=-1:
    for i in range(len(ans)-1,-1,-1):
        if ans[i]==last:
            ans=ans[:i+1]
            break
print(ans)
