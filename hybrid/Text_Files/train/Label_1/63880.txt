s, f, t = map(int, input().split())
n = int(input())
wait = [0] * n
tt = [0] * n
lis = map(int, input().split())
for i, el in enumerate(lis):
    if i == 0:
        wait[0] = max(0, el - s)
        tt[0] = wait[0] + t
        continue
    tt[i] = max(tt[i - 1] + t, el + wait[i - 1])
    wait[i] = max(0, el - (tt[i - 1] - wait[i - 1]))
ans = 0
for w in wait:
    if w == 0:
        ans += 1
print (f - ans * t)
