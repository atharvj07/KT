import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int boxes = 0;
        int maxObjects = 0;
        int remainingCapacity = k;
        for (int i = 0; i < n; i++) {
            if (a[i] > remainingCapacity) {
                boxes++;
                remainingCapacity = k;
            }
            if (boxes == m) {
                break;
            }
            remainingCapacity -= a[i];
            maxObjects++;
        }

        System.out.println(maxObjects);
    }
}
