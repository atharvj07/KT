#include<bits/stdc++.h>
using namespace std;
typedef pair<int, int> pi;
const int MAXN=5005;
int n, color[MAXN], component[MAXN], ans;
int find(int x) {return x==component[x]?x:component[x]=find(component[x]);}
void merge(int x, int y) {
    x=find(x), y=find(y);
    if(x!=y) --ans, component[y]=x;
}
int main() {
    cin>>n;
    for(int i=1;i<=n;++i) cin>>color[i];
    for(int i=1;i<=n;++i) component[i]=i;
    ans=n;
    for(int i=1;i<=n;++i) if(i==1||color[i]!=color[i-1]) merge(i, i-1);
    cout<<ans<<'\n';
}
