from collections import Counter

def solve():
    for _ in range(int(input())):
        a, b = input(), input()
        c = Counter(a)
        d, e, f = b[0], b[1], b[2]
        for i in sorted(c.keys()):
            if c[i] > c[d] + c[e] + c[f] or (c[i] == c[d] + c[e] + c[f] and i < d):
                c[d] -= 1; c[e] -= 1; c[f] -= 1; c[i] += 1; break
        for i in sorted(c.keys()):
            print(i * c[i], end='')
        print()

solve()
