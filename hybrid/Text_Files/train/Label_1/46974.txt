import java.util.Scanner;
import java.util.LinkedList;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    LinkedList<Integer> recruits = new LinkedList<>();
    int crimes = 0;
    for (int i = 0; i < n; i++) {
      int e = sc.nextInt();
      if (e == -1) {
        if (recruits.size() > 0) {
          recruits.removeLast();
        } else {
          crimes++;
        }
      } else {
        recruits.addFirst(e);
      }
    }
    System.out.println(crimes);
  }
}
