import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    static HashMap<Integer, List<Integer>> graph;
    static long[] taxes;
    static boolean[] visited;
    static long[] dp;
    static int n, m;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken()); // number of cities
        m = Integer.parseInt(st.nextToken()); // number of roads

        taxes = new long[n + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            taxes[i] = Long.parseLong(st.nextToken()); // amount of taxes paid in the i-th city
        }

        graph = new HashMap<>();
        for (int i = 0; i <= n; i++) {
            graph.put(i, new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            graph.get(a).add(b);
        }

        dp = new long[n + 1];
        Arrays.fill(dp, -1);

        visited = new boolean[n + 1];
        long ans = dfs(1);

        if (ans == Long.MAX_VALUE) {
            System.out.println("LOSE");
        } else {
            System.out.println("WIN");
            dfs2(1, ans);
        }
    }

    static long dfs(int node) {
        if (visited[node]) {
            return dp[node];
        }

        visited[node] = true;

        long min = Long.MAX_VALUE;
        for (int neighbor : graph.get(node)) {
            long val = dfs(neighbor);
            if (val != Long.MAX_VALUE) {
                min = Math.min(min, val + 1);
            }
        }

        dp[node] = min;
        return min;
    }

    static void dfs2(int node, long ans) {
        visited[node] = true;

        for (int neighbor : graph.get(node)) {
            if (!visited[neighbor] && dfs(neighbor) + 1 == ans) {
                taxes[node] = 0;
                dfs2(neighbor, ans - 1);
                return;
            }
        }

        taxes[node] = ans;
    }
}
