n = int(input().strip())
y = input().strip()

def count_suitable_k(n, y):
    def gcd(a, b):
        while b:
            a, b = b, a % b
        return a

    # Convert the bitstring into a list of zeroes and ones
    y_list = [int(bit) for bit in y]

    # Count the number of ones in the bitstring
    count_ones = sum(y_list)

    # If there are no ones or the string is all ones, the result is n - 1
    if count_ones == 0 or count_ones == n:
        return n - 1

    # Calculate the greatest common divisor (gcd) of n and the number of ones
    gcd_result = gcd(n, count_ones)

    # The result is the gcd minus 1, since we cannot have k = 0
    return gcd_result

print(count_suitable_k(n, y))
