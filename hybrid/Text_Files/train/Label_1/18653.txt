#include <cstdio>

typedef long long ll;

int main() {
    ll ans = 0;
    int x = 0, y = 0;
    char ch;

    while ((ch = getchar()) != '\n') {
        if (ch == 'U') y++;
        else if (ch == 'D') y--;
        else if (ch == 'R') x++;
        else if (ch == 'L') x--;
        else {
            puts("-1");
            return 0;
        }
        if (x == 0 && y == 0) { ans = 0; continue; }
        if (x > 0 && y == 0) {
            ans += x / 2;
            x = x % 2;
        }
        if (y > 0) {
            if (x == 0) {
                ans += y;
                y = 0;
            } else {
                ans += y / 2;
                y = y % 2;
            }
        }
    }
    if (x != 0 || y != 0) ans = -1;
    printf("%lld\n", ans);
    return 0;
}
