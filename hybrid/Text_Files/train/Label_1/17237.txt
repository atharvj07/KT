#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  vector<vector<int> > dp(n, vector<int>(k, 0));
  dp[0][a[0] % k] = 1;
  for (int i = 1; i < n; i++) {
    for (int j = 0; j < k; j++) {
      dp[i][(j + a[i] % k + k) % k] += dp[i - 1][j];
    }
  }
  cout << dp[n - 1][0];
}
