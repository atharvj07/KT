n = int(input().strip())
w = list(map(int, input().strip().split()))
seq = input().strip()

rows = [(w[i], i + 1) for i in range(n)]
rows.sort()

occupied = []
index = 0
result = []

for s in seq:
    if s == '0':
        result.append(rows[index][1])
        occupied.append(rows[index])
        index += 1
    else:
        result.append(occupied[-1][1])
        occupied.pop()
        
print(" ".join(map(str, result)))
