#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int tpow = 9, ppow = 60, maxp = 1e9 + 7;
int a[300010], q[1<<tpow], sum[300010], sqsum[300010], dp[1<<tpow][300010];
int main() {
    int n, p, k;
    cin >> n >> p >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a+n);
    for (int i = 0; i < n; i++) {
        sum[i] = a[i];
        sqsum[i] = a[i] * a[i] % p;
        if (i) {
            sum[i] += sum[i-1];
            sqsum[i] += sqsum[i-1];
        }
    }
    const int t = 1<<tpow;
    for (int i = 0; i < t; i++) {
        for (int j = 0; j < n; j++) dp[i][j] = maxp;
    }
    dp[0][0] = 0;
    for (int msk = 0; msk < t; msk++) {
        for (int i = 0; i < n; i++) {
            dp[msk][i+1] = min(dp[msk][i+1], dp[msk][i]);
            dp[msk|q[a[i]%t]][i+1] = min(dp[msk|q[a[i]%t]][i+1], dp[msk][i] + (a[i]%p) * (a[i]%p) % p + k);
        }
    }
    for (int i = 0; i < t; i++) q[i] = t;
    for (int i = 0; i < p; i++) q[i] = i>>tpow;
    int ans = maxp;
    for (int msk = 0; msk < t; msk++) {
        ans = min(ans, dp[msk][n]);
    }
    if (ans == maxp) ans = -1;
    cout << ans << "\n";
    return 0;
}
