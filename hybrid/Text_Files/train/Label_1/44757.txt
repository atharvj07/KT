import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        Arrays.sort(a);
        int l = 0, r = n - 1, ans = 0;
        while (l < r && k > 0) {
            ans += a[r] - a[l];
            l++;
            r--;
            k--;
        }
        System.out.println(ans);
    }
}
