#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
int n, q;
vector<long long int> a, b, t;
long long int bit[MAXN], ans[MAXN];
void update(int idx, long long int val) {
  while (idx <= n) {
    bit[idx] += val;
    idx += (idx & -idx);
  }
}
long long int query(int idx) {
  long long int sum = 0LL;
  while (idx > 0) {
    sum += bit[idx];
    idx -= (idx & -idx);
  }
  return sum;
}
int main() {
  cin >> n >> q;
  a.resize(n + 1);
  b.resize(n + 1);
  for (int i = 1; i <= n; ++i) { cin >> a[i]; }
  for (int i = 1; i <= n; ++i) { cin >> b[i]; }
  t.resize(2 * n + 5);
  for (int i = 1; i <= n; ++i) {
    t[i] = a[i] - b[i];
    update(i, t[i]);
  }
  for (int i = n + 1; i <= 2 * n; ++i) { t[i] = 2LL; }
  for (int i = 1; i <= n; ++i) {
    ans[i] = query(i);
    update(i, -2LL);
    ans[i] += query(n) - query(i);
  }
  while (q--) {
    int l, r;
    cin >> l >> r;
    if (ans[r] - ans[l - 1] <= 0) {
      cout << -1 << endl;
      continue;
    }
    if ((ans[r] - ans[l - 1]) % 2 == 0)
      cout << (ans[r] - ans[l - 1]) / 2 << endl;
    else
      cout << (ans[r] - ans[l - 1]) / 2 + 1 << endl;
  }
  return 0;
}
