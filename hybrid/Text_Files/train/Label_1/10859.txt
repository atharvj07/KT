#include <bits/stdc++.h>
using namespace std;

unordered_map<int, int> fre;

int main() {
  int n, k; cin >> n >> k;
  for (int i = 0; i < n; ++i) {
    int x; cin >> x; fre[x]++;
  }
  long long ans = 0;
  for (const auto& [x, f] : fre) {
    if (x == 0) continue;
    ans += 1ll * f * (f - 1) / 2;
    if (k % 2 == 0 && (1ll << (k / 2)) == x) ans -= 1ll * f * (f - 1) / 2;
  }
  cout << ans << endl;
}
