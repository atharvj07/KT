def valid_directions(maze, instructions, sx, sy, ex, ey):
    valid_count = 0
    directions = [(0, 1), (0, -1), (-1, 0), (1, 0)]  # Right, Left, Up, Down

    # Try all permutations of directions
    from itertools import permutations
    for perm in permutations(directions):
        x, y = sx, sy
        for i in instructions:
            dx, dy = perm[int(i)]
            x += dx
            y += dy
            if x < 0 or y < 0 or x >= len(maze) or y >= len(maze[0]) or maze[x][y] == '#':
                break
            if x == ex and y == ey:
                valid_count += 1
                break
    return valid_count

n, m = map(int, input().split())
maze = [list(input().strip()) for _ in range(n)]
instructions = input().strip()

# Find start and end positions
for i in range(n):
    for j in range(m):
        if maze[i][j] == 'S':
            sx, sy = i, j
        elif maze[i][j] == 'E':
            ex, ey = i, j

print(valid_directions(maze, instructions, sx, sy, ex, ey))
