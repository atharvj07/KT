#include<bits/stdc++.h>
using namespace std;
int main() {
  long long x;
  cin >> x;
  long long k = x - 1, i = 0;
  for (i = 1; i * i <= x; i++) {
    if (!(x % i)) k += ((x / i) - i);
  }
  cout << k << '\n';
  i = 0;
  for (i = 1; i * i <= x; i++) {
    if (!(x % i)) {
      cout << i << ' ' << x / i << '\n';
      if (i != x / i) cout << x / i << ' ' << i << '\n';
    }
  }
}
