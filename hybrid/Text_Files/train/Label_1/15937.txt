#include <unordered_map>
#include <vector>
#include <iostream>
using namespace std;

int main() {
  int m;
  unordered_map<long long, vector<int>> hash;
  cin >> m;
  vector<long long> arr(m);
  for (int i = 0; i < m; ++i) {
    long long val;
    cin >> val;
    arr[i] = val;
    if (hash.count(val))
      hash[val].push_back(i);
    else
      hash[val] = {i};
  }
  for (auto a : arr) {
    bool found = false;
    for (int i = 0; i < 63; ++i) {
      long long val = (1ll << i);
      if (hash.count(val ^ a)) {
        found = true;
        for (int idx : hash[val ^ a]) {
          if (idx == i) continue;
          cout << 2 << ' ' << idx << ' ' << i << '\n';
          break;
        }
        break;
      }
      for (int j = i + 1; j < 63; ++j) {
        long long val = (1ll << i) | (1ll << j);
        if (hash.count(val ^ a) && i != j && (1ll << j) != a) {
          found = true;
          for (int idx : hash[val ^ a]) {
            if (idx == i || idx == j) continue;
            cout << 3 << ' ' << idx << ' ' << i << ' ' << j << '\n';
            break;
          }
          break;
        }
      }
    }
    if (!found) cout << 0 << '\n';
  }
  return 0;
}
