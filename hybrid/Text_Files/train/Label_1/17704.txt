#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  if (k > (n * n + 1) / 2) {
    cout << "-1";
    return 0;
  }
  vector<vector<int>> m(n, vector<int>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < i; j++) {
      if (k > 0) {
        m[i][j] = m[j][i] = 1;
        k--;
      }
    }
    if (k > 0) {
      m[i][i] = 1;
      k--;
    }
  }
  for (auto r : m) {
    for (auto c : r) {
      cout << c;
    }
    cout << endl;
  }
  return 0;
}
