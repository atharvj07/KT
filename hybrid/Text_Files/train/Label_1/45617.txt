import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int min = m / n;
        for (int i = 0; i < n; i++) {
            min = Math.max(min, m / n + a[i]);
        }
        int max = m;
        for (int i = 0; i < n; i++) {
            max = Math.max(max, m + a[i]);
        }
        System.out.println(min + " " + max);
    }
}
