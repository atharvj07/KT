#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
#define mp make_pair
const int N = 3e5 + 5;
int n, k;
ll l[N], r[N];
pair<ll, int> P[N];
int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; ++i)
        cin >> l[i] >> r[i], P[i] = mp(l[i], i);
    sort(P + 1, P + 1 + n);
    vector<int> V;
    for (int i = 1; i <= n; ++i)
        if (k == 0)
            break;
        else {
            if (i == 1 || P[i].first > P[i - 1].first)
                V.pb(P[i].second), k--;
        }
    cout << r[V[0]] - l[V[0]] + 1 << endl;
    for (int x : V)
        cout << x << " ";
    cout << endl;
}
