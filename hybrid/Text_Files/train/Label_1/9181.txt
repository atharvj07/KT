#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int sieveN = 3e5 + 10;
ll prime[sieveN];
int spf[sieveN];

void sieve() {
  int pcnt = 0;
  for (ll i = 2; i < sieveN; ++i) {
    if (!spf[i])
      prime[++pcnt] = i, spf[i] = i;
    for (int j = 1; j <= pcnt && i * prime[j] < sieveN; ++j) {
      spf[i * prime[j]] = prime[j];
      if (!(i % prime[j])) break;
    }
  }
}

bool gcd1(ll a, ll b) {
  while (b) {
    if (!(a % b) or !(b % a)) return false;
    a %= b, swap(a, b);
  }
  return true;
}

int main() {
  ll l, r;
  cin >> l >> r;
  sieve();
  if (l == r) {
    cout << "NO";
    return 0;
  }
  ll p = l + l % 2;
  if (p <= r) {
    cout << "YES" << endl;
    for (ll j = 0; j <= r - l; j += 2) cout << p + j << " " << p + j + 1 << endl;
    return 0;
  }
  bool not_found = true;
  for (ll j = l; j <= r; j++) {
    if ((r - j + 1) % 2 != 0) {
      if (gcd1(j, r - j + 1)) {
        cout << "YES" << endl;
        for (ll k = 0; k <= r - l; k += 2) {
          if (k == r - j + 1) continue;
          cout << j << " " << k << endl;
        }
        cout << r - j + 1 << " " << j << endl;
        return 0;
      }
      not_found = false;
    }
  }
  if (!not_found)
    cout << "NO";
  else
    cout << "YES" << endl;
  return 0;
}
