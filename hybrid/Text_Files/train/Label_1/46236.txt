import java.util.Scanner;

public class MoneyWithdrawal {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();

        int operations = 0;
        while (N > 0) {
            int maxPowerOf6 = 0;
            int maxPowerOf9 = 0;

            for (int i = 1; i <= 15; i++) {
                if (Math.pow(6, i) <= N) {
                    maxPowerOf6 = i;
                }
                if (Math.pow(9, i) <= N) {
                    maxPowerOf9 = i;
                }
            }

            int maxWithdrawal = Math.max((int) Math.pow(6, maxPowerOf6), (int) Math.pow(9, maxPowerOf9));
            N -= maxWithdrawal;
            operations++;
        }

        System.out.println(operations);
    }
}
