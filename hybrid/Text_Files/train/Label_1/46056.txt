import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[][] g = new int[n][n];
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt() - 1, b = sc.nextInt() - 1;
            g[a][b] = g[b][a] = 1;
        }
        int[] d = new int[n];
        Queue<Integer> q = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '1') {
                q.add(i);
                d[i] = 0;
            } else {
                d[i] = -1;
            }
        }
        while (!q.isEmpty()) {
            int v = q.poll();
            for (int i = 0; i < n; i++) {
                if (g[v][i] == 1 && d[i] == -1) {
                    d[i] = d[v] + 1;
                    q.add(i);
                }
            }
        }
        int max = 0;
        for (int i = 0; i < n; i++) {
            max = Math.max(max, d[i]);
        }
        if (max == -1) {
            System.out.println(-1);
        } else {
            System.out.println(max);
        }
    }
}
