import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[][] en = new int[n + 1][4];

        for (int i = 0; i < n + 1; i++) {
            String[] s = in.readLine().split(" ");
            for (int j = 0; j < 4; j++)
                en[i][j] = Integer.parseInt(s[j]);
        }

        Arrays.sort(en, (a, b) -> b[3] - a[3]);
        long ans = 0;

        for (int i = 1; i < n + 1; i++) {
            int res = Math.max(en[0][2] - en[i][1], 0);
            en[0][0] -= res;
            if (en[0][0] <= 0) {
                System.out.println(-1);
                return;
            }
            ans += res;
        }

        System.out.println(ans);
    }
}
