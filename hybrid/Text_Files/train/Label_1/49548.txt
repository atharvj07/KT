import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] stones = new long[1 << n];
        for (int i = 0; i < stones.length; i++) {
            stones[i] = in.nextInt();
        }

        int l = 1 << n;
        while (l > 1) {
            for (int i = 0; i < l; i += 2) {
                stones[i / 2] = Math.max(stones[i], stones[i + 1]);
                stones[i / 2] -= Math.min(stones[i], stones[i + 1]);
            }
            l /= 2;
        }

        System.out.println(stones[0]);
    }
}
