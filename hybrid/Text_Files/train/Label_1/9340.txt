#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  vector<int> f(s.size()), g(s.size()), h(s.size());
  for (int i = s.size() - 1; ~i; --i) {
    if (s[i] == '?' || s[i] == t[t.size() - 1]) f[i] = (i ? f[i + 1] : 0) + (s[i] == t[t.size() - 1]);
    if (s[i] == t[t.size() - 2]) g[i] = (i + 1 < s.size() ? g[i + 1] : 0) + (s[i] == t[t.size() - 2]);
    if (s[i] == t[0]) h[i] = (i ? h[i + 1] : 0) + (s[i] == t[0]);
  }
  int ans = 0;
  for (int i = 0; i + t.size() - 1 < s.size(); ++i) {
    if (s[i] == '?' || s[i] == t[0]) ans = max(ans, h[i + t.size() - 1]);
    if (ans && i + t.size() - 1 < s.size() && (s[i + t.size() - 1] == '?' || s[i + t.size() - 1] == t[t.size() - 1])) ans += f[i + t.size() - 1];
    if (ans >= 2 && i + t.size() - 2 < s.size() && (s[i + t.size() - 2] == '?' || s[i + t.size() - 2] == t[t.size() - 2])) ans += g[i + t.size() - 2];
  }
  cout << ans << endl;
  return 0;
}
