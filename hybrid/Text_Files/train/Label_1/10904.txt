#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define M 1000000007
int n,m,dp[2001][2001],dir[2001][2001];
char c[2001][2001];
int go(int x,int y)
{
    if(dp[x][y]!=-1) return dp[x][y];
    dp[x][y]=0;
    if(x==n && y==m) return dp[x][y]=1;
    if(c[x][y]=='R') return dp[x][y]=0;
    if(x+1<=n) dp[x][y]=go(x+1,y),dir[x][y]=0;
    if(y+1<=m) {dp[x][y]=(dp[x][y]+go(x,y+1))%M,dir[x][y]=1;}
    return dp[x][y];
}
int main() {
    cin>>n>>m;
    for(int i=1; i<=n; i++) {
        for(int j=1; j<=m; j++) {
            cin>>c[i][j];
        }
    }
    memset(dp,-1,sizeof(dp));
    cout<<go(1,1)<<endl;
	return 0;
}
