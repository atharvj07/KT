#include <bits/stdc++.h>
using namespace std;
const int MX = 2e5 + 5;
vector<int> g[MX];
int ans[MX];
void dfs(int u, int p, int c){
    ans[u] = c;
    for(int v : g[u]){
        if(v != p){
            dfs(v, u, c + 1);
        }
    }
}
int main(){
    int n, k; cin >> n >> k;
    for(int i = 0; i < n - 1; i++){
        int a, b; cin >> a >> b;
        g[a].push_back(b);
        g[b].push_back(a);
    }
    dfs(1, 1, 1);
    int mx = 0;
    for(int i = 1; i <= n; i++){
        mx = max(mx, ans[i]);
    }
    cout << mx << endl;
    for(int i = 0; i < n - 1; i++){
        cout << ans[i + 1] << ' ';
    }
}
