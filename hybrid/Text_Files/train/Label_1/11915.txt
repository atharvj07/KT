#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<int, int> pi;
typedef pair<ll, ll> pl;
typedef vector<int> vi;
typedef vector<ll> vl;
typedef vector<pi> vpi;
typedef vector<pl> vpl;
typedef vector<vi> vvi;
typedef vector<vl> vvl;

#define all(x) (x).begin(), (x).end()

void nope() {
  cout << -1 << '\n';
  exit(0);
}

int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin.exceptions(cin.failbit);

  int n;
  cin >> n;
  map<string, int> m;
  map<int, map<char, vi>> mm;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    m[s]++;
    int c = 0;
    char last = '#';
    for (char t : s) {
      if (t == 'a' || t == 'e' || t == 'i' || t == 'o' || t == 'u') {
        c++;
        last = t;
      }
    }
    mm[c][last].push_back(i);
  }
  vector<pi> v;
  for (auto& pp : mm) {
    int c = pp.first;
    for (auto p : pp.second) {
      auto& v1 = p.second;
      if (v1.size() >= 4) {
        v.push_back({v1[0], v1[v1.size() - 1]});
      }
    }
  }
  cout << v.size() / 2 << '\n';
  for (auto& x : v) {
    int a = x.first, b = x.second;
    cout << a + 1 << ' ' << b + 1 << '\n';
    cout << b + 1 << ' ' << a + 1 << '\n';
  }
  return 0;
}
