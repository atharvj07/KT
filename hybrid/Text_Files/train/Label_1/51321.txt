import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int min_num = solve(scanner.nextInt(), scanner.nextInt());
        System.out.println(min_num);
    }
    static int solve(int d, int s) {
        int n = 0;
        while (s > 0) {
            n++;
            int tmp = (s >= 9) ? 9 : s;
            if (n % d == 0) {
                s -= tmp;
            } else {
                s -= Math.min(s, d - n % d);
            }
        }
        if (n % d == 0) return n;
        return -1;
    }
}
