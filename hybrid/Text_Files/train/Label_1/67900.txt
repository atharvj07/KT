def find_set(x, parent):
    if x != parent[x]:
        parent[x] = find_set(parent[x], parent)
    return parent[x]

def union_sets(a, b, parent, rank):
    a = find_set(a, parent)
    b = find_set(b, parent)
    if a != b:
        if rank[a] < rank[b]:
            a, b = b, a
        parent[b] = a
        if rank[a] == rank[b]:
            rank[a] += 1

n = int(input())
cities = [tuple(map(int, input().split())) for _ in range(n)]
c = list(map(int, input().split()))
k = list(map(int, input().split()))

edges = []
for i in range(n):
    for j in range(i + 1, n):
        cost = (abs(cities[i][0] - cities[j][0]) + abs(cities[i][1] - cities[j][1])) * (k[i] + k[j])
        edges.append((cost, i, j))

edges.sort()

parent = list(range(n))
rank = [0] * n
power_stations = []
connections = []
total_cost = 0

for cost, ci, cj in edges:
    if find_set(ci, parent) != find_set(cj, parent):
        if c[ci] + c[cj] > cost:
            total_cost += cost
            connections.append((ci + 1, cj + 1))
            union_sets(ci, cj, parent, rank)
        else:
            if c[ci] < c[cj] and ci not in power_stations:
                if find_set(ci, parent) == ci:
                    total_cost += c[ci]
                    power_stations.append(ci + 1)
                    union_sets(ci, cj, parent, rank)
            if c[cj] <= c[ci] and cj not in power_stations:
                if find_set(cj, parent) == cj:
                    total_cost += c[cj]
                    power_stations.append(cj + 1)
                    union_sets(ci, cj, parent, rank)

for i in range(n):
    if find_set(i, parent) == i and i + 1 not in power_stations:
        total_cost += c[i]
        power_stations.append(i + 1)

print(total_cost)
print(len(power_stations))
print(" ".join(map(str, power_stations)))
print(len(connections))
for connection in connections:
    print(*connection)
