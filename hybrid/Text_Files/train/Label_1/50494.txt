import java.util.*;

public class Main {
    static char[][] keyboard;
    static boolean[][] shiftKeys;
    static char[] text;

    static int distance(int x1, int y1, int x2, int y2) {
        int dx = x1 - x2;
        int dy = y1 - y2;
        return dx * dx + dy * dy;
    }

    static int solve(int n, int m, int x) {
        shiftKeys = new boolean[n][m];
        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                char c = keyboard[i][j];
                shiftKeys[i][j] = c == 'S';
                dist[i][j] = Integer.MAX_VALUE;
            }
        }

        Queue<int[]> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (shiftKeys[i][j]) {
                    queue.add(new int[]{i, j});
                    dist[i][j] = 0;
                }
            }
        }

        while (!queue.isEmpty()) {
            int[] cur = queue.poll();
            int x1 = cur[0];
            int y1 = cur[1];
            for (int i = x1 - 1; i <= x1 + 1; i++) {
                for (int j = y1 - 1; j <= y1 + 1; j++) {
                    if (i >= 0 && j >= 0 && i < n && j < m) {
                        if (distance(x1, y1, i, j) <= x) {
                            if (dist[i][j] > dist[x1][y1] + 1) {
                                dist[i][j] = dist[x1][y1] + 1;
                                queue.add(new int[]{i, j});
                            }
                        }
                    }
                }
            }
        }

        int ans = 0;
        for (int i = 0; i < text.length; i++) {
            char c = text[i];
            boolean found = false;
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < m; k++) {
                    if (keyboard[j][k] == c || (shiftKeys[j][k] && Character.toUpperCase(keyboard[j][k]) == c)) {
                        if (dist[j][k] <= x) {
                            found = true;
                            break;
                        }
                    }
                }
            }
            if (!found) {
                ans++;
            }
        }

        return ans == 0 ? -1 : ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int x = sc.nextInt();
        sc.nextLine();
        keyboard = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = sc.nextLine();
            for (int j = 0; j < m; j++) {
                keyboard[i][j] = line.charAt(j);
            }
        }
        int q = sc.nextInt();
        sc.nextLine();
        text = sc.nextLine().toCharArray();
        int result = solve(n, m, x);
        System.out.println(result);
    }
}
