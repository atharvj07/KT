#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int main() {
    int n, m; cin >> n >> m;
    string s; cin >> s;
    vector<vector<int>> dp(n, vector<int>(m, 0));
    dp[n - 1][s[n - 1] - 'a'] = 1;
    for(int i = n - 2; i >= 0; i--) {
        for(int j = 0; j < m; j++) {
            for(int k = 0; k < m; k++) {
                if(s[i] - 'a' == j || k == s[i] - 'a') continue;
                dp[i][j] = (1LL * dp[i][j] + dp[i + 1][k]) % mod;
            }
        }
    }
    int ans = 0;
    for(int i = 0; i < m; i++) ans = (1LL * ans + dp[0][i]) % mod;
    cout << ans << endl;
    return 0;
}
