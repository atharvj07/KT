import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] t = new int[n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = input.nextInt();
            c[i] = input.nextInt();
        }
        int pay = 0;
        int free = 0;
        for (int i = 0; i < n; i++) {
            if (t[i] == 0) {
                free += c[i];
            } else {
                free = Math.max(0, free - 1);
                pay += c[i] + free;
            }
        }
        System.out.println(pay);
    }
}
