import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n - 1];
        int[] b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        if (n == 2) {
            System.out.println("YES");
            System.out.println(a[0] + " " + b[0]);
            return;
        }
        boolean found = false;
        for (int i = 1; i < n; i++) {
            for (int j = i; j < n; j++) {
                if (a[i - 1] < b[i - 1]) {
                    if ((a[i - 1] & b[j - 1]) != a[i - 1]) {
                        found = true;
                        break;
                    }
                } else if (a[i - 1] > b[i - 1]) {
                    if ((a[i - 1] & b[j - 1]) != b[i - 1]) {
                        found = true;
                        break;
                    }
                } else {
                    if ((a[i - 1] & b[j - 1]) != 0) {
                        found = true;
                        break;
                    }
                }
            }
            if (found) {
                break;
            }
        }
        if (found) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            int[] t = new int[n];
            t[0] = a[0];
            t[1] = b[0];
            for (int i = 2; i < n; i++) {
                t[i] = a[i - 1] | b[i - 1];
            }
            for (int i = 0; i < n; i++) {
                System.out.print(t[i] + " ");
            }
            System.out.println();
        }
    }
}
