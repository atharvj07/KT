#include<bits/stdc++.h>
using namespace std;
int T,n,k,N=62510;
bool f[62510];
int main(){
    cin>>n>>k;
    for(register int i=2;i<=n;++i)f[i]=1;
    for(register int i=2;i*i<=n;++i){
        if(f[i]){
            for(register int j=i*i;j<=n;j+=i){
                f[j]=0;
            }
        }
    }
    int s=sqrt(n),ans=0;
    for(register int i=2;i<s;++i){
        if(f[i]){
            if(n/i>=k)ans=max({ans,n/i,i});
        }
    }
    cout<<ans;
    return 0;
}
