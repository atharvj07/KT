#include <vector>
#include <iostream>

int main() {
    int t;
    std::cin >> t;
    
    while (t--) {
        int n, s;
        std::cin >> n >> s;
        
        std::vector<int> a(n);
        for (auto& x : a) {
            std::cin >> x;
        }
        
        int skip = 0;
        int mx = 0;
        int sum = 0;
        
        for (int i = 1; i <= n; i++) {
            sum += a[i - 1];
            
            if (sum <= s) {
                mx = std::max(mx, i);
            }
            else if (i - 1 > mx) {
                skip = i;
                mx = i - 1;
            }
        }
        
        std::cout << skip << "\n";
    }
    
    return 0;
}
