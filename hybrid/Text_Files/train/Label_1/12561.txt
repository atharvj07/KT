#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, v;
  cin >> n >> v;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0, cur = 0;
  for (int i = 0; i < n; i++) {
    if (i - cur + 1 <= v)
      ans += a[i] * (i - cur + 1);
    else {
      ans += a[i] * v;
      cur = i;
    }
  }
  cout << ans << endl;
  return 0;
}
