import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T1 = sc.nextInt();
        int T2 = sc.nextInt();
        int R1 = sc.nextInt();
        int R2 = sc.nextInt();

        if (R1 == -1 && R2 == -1) {
            System.out.println(T1 < T2 ? "Alice" : T1 > T2 ? "Bob" : "Draw");
        } else if (R1 != -1 && R2 != -1) {
            System.out.println(R1 > R2 ? "Alice" : R1 < R2 ? "Bob" : "Draw");
        } else {
            System.out.println(R1 != -1 ? "Alice" : "Bob");
        }
    }
}
