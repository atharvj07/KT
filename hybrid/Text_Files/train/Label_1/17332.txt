#include <cstdio>
#include <vector>
using namespace std;

int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  vector<int> a[n];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      int x;
      scanf("%d", &x);
      a[i].push_back(x);
    }
  }
  int ans = 0;
  int c[500];
  for (int i = 0; i < n; i++) {
    int x = 0;
    for (int j = 0; j < m; j++) {
      x |= a[i][j];
    }
    if (x == 0) {
      puts("NIE");
      return 0;
    }
    ans |= x;
    c[i] = 0;
    for (int j = 0; j < m; j++) {
      if ((x & a[i][j]) != 0) {
        c[i] = j + 1;
        break;
      }
    }
  }
  if (ans == 0) {
    puts("NIE");
    return 0;
  }
  puts("TAK");
  for (int i = 0; i < n; i++) {
    printf("%d ", c[i]);
  }
  puts("");
  return 0;
}
