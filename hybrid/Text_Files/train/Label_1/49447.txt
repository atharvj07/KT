import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            String s = sc.next();
            int n = s.length();
            int ones = 0, twos = 0;
            for (int i = 0; i < n; i++) {
                ones += s.charAt(i) == 'o' ? 1 : 0;
                twos += s.charAt(i) == 't' ? 1 : 0;
            }
            int min = Math.min(ones, twos);
            System.out.println(min);
            for (int i = 0; i < min; i++) {
                System.out.print((ones > twos ? "o" : "t") + " ");
            }
            System.out.println();
        }
    }
}
