import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<Integer>[] graph = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int[] edge = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            graph[edge[0]].add(edge[1]);
            graph[edge[1]].add(edge[0]);
        }
        int[] labels = new int[n - 1];
        dfs(1, -1, labels, graph, new boolean[n + 1], 0);
        System.out.println(Arrays.stream(labels).mapToObj(String::valueOf).collect(Collectors.joining(" ")));
    }

    public static void dfs(int node, int parent, int[] labels, List<Integer>[] graph, boolean[] used, int label) {
        used[node] = true;
        for (int neighbor : graph[node]) {
            if (!used[neighbor]) {
                dfs(neighbor, node, labels, graph, used, label + 1);
                labels[label] = label;
            }
        }
    }
}
