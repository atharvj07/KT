#include <bits/stdc++.h>

#define int long long
using namespace std;

const int mod = 998244353;
int dp[1005][2]; // dp[i][0] is the number of good subsequences ending at a_i where a_i = 0, dp[i][1] is the number of good subsequences ending at a_i where a_i > 0
int a[1005];

int32_t main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  dp[1][0] = 1; // Base case: the empty subsequence is a good subsequence
  for (int i = 2; i <= n; i++) {
    dp[i][0] = (dp[i - 1][0] + dp[i - 1][1]) % mod; // If a_i = 0, then we can either extend the previous good subsequence (if it ends with a_i - 1 = 0) or start a new good subsequence with a_i = 0
    if (a[i] > 0) {
      dp[i][1] = (dp[i - 1][0] + dp[i - 1][1]) % mod; // If a_i > 0, then we can either extend the previous good subsequence (if it ends with a_i - 1 = 0) or start a new good subsequence with a_i = a_i
    }
  }
  cout << dp[n][1] << '\n';
  return 0;
}
