for _ in range(int(input())):
    n, m, k = map(int, input().split())
    p = n//m
    r = n%m
    pl = list(range(1, n+1))
    for i in range(k):
        for j in range(m):
            if j<r: s = p+1
            else: s = p
            print(s, end=" ")
            for x in range(s): print(pl.pop(0), end=" ")
            print()
        pl.extend(pl)
