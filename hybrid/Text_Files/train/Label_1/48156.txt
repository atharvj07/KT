import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), M = sc.nextInt();
        Integer[] A = new Integer[N+1];
        for(int i = 0; i < N; i++) {
            A[i] = sc.nextInt();
        }
        Arrays.sort(A, 0, N, (a, b) -> {return b.compareTo(a);});
        long sum = 0;
        for(int i = 0; i < N; i++) {
            long a = A[i];
            if(M <= 0) {
                sum += a;
                continue;
            }
            while(a > 0) {
                a = a >> 1;
                M--;
            }
            sum += a;
        }
        System.out.println(sum);
    }
}
