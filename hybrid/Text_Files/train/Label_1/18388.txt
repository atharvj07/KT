#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 2e5 + 5;
struct data {
    int w, id;
};
int n, tw, sum;
vector<data> v;
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> tw;
        sum = 0;
        v.clear();
        for (int i = 1, a; i <= n; i++) {
            cin >> a;
            if (a > tw) continue;
            v.push_back({a, i});
            sum += a;
        }
        n = v.size();
        sort(v.begin(), v.end(), [&](data a, data b) { return a.w > b.w; });
        bool ok = 0;
        if (sum < (tw + 1) / 2) { cout << "-1\n"; continue; }
        cout << n << '\n';
        sum = 0;
        for (data x : v) {
            if (sum + x.w <= tw) {
                cout << x.id << ' ';
                sum += x.w;
            }
        }
        cout << '\n';
    }
    return 0;
}
