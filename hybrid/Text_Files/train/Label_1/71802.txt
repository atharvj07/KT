import sys

n=int(input());cur=[0]*(n+1);L=[[] for _ in range(n+1)];res=[];ans=0
for i in range(n):
    a,b=map(int,sys.stdin.readline().split())
    L[a].append(b)
    cur[a]+=1
    cur[b]+=1
for i in range(1,n+1):
    if cur[i]==1:ans+=1;res.append(i)
    for j in L[i]:
        cur[j]-=1
print(ans);print(*res)
