import sys

m = int(input().strip())
q = [int(i) for i in input().strip().split(' ')]
n = int(input().strip())
a = [int(i) for i in input().strip().split(' ')]
a.sort()

l = 0
r = m-1
best_q = -1
while l <= r:
    mid = (l+r) // 2
    if q[mid] <= n:
        l = mid + 1
        best_q = q[mid]
    else:
        r = mid - 1

if best_q == -1:
    print(0)
else:
    ans = 0
    for i in range(best_q - 1):
        if i < len(a):
            ans += a[i]
    ans += max(a[-1-min(2, n - best_q + 1):], 0)
    print(ans)
