n, k = map(int, input().split())
mag = []
for i in range(n):
    mag.append(list(map(int, input().split())))
mag.sort()
for i in mag:
    i.sort()
w = mag[k][1] - mag[0][0]
h = mag[k][3] - mag[0][2]
for i in range(k+1, n):
    w = min(w, mag[i][1] - mag[k][0])
    h = min(h, mag[i][3] - mag[k][2])
print(w * h)
