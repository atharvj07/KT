import sys, heapq
input = sys.stdin.readline

n, m = map(int, input().split())
edges = []
for i in range(m):
    a, b, c = map(int, input().split())
    edges.append((a, b, c))

graph = [[] for _ in range(n+1)]
for a, b, c in edges:
    graph[a].append((b,c))
    graph[b].append((a,c))

# dijkstra to find shortest path and determine affected roads
dist = [1e9]*(n+1)
vis = [False]*(n+1)
road_st = {}
pq = [(0,1)]
while pq:
    cnt, node = heapq.heappop(pq)
    if node == n:
        break
    if vis[node]:
        continue
    vis[node] = True
    for nxt, st in graph[node]:
        if not vis[nxt]:
            new_cnt = cnt
            if st:
                new_cnt += 1
            else:
                new_cnt -= 1
            road_st[(node,nxt)] = new_cnt != cnt
            if dist[nxt] > new_cnt:
                dist[nxt] = new_cnt
                heapq.heappush(pq, (new_cnt,nxt))
# find the affected roads
affected_roads = []
for edge in edges:
    a, b, c = edge
    if road_st[(a,b)]:
        affected_roads.append((a,b,road_st[(a,b)]))

print(len(affected_roads))
for road in affected_roads:
    a, b, st = road
    print(a, b, int(st))
