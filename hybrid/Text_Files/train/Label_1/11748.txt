#include <iostream>
#include <vector>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        int sum = 0;
        for (int &x : a) {
            cin >> x;
            sum += x;
        }
        vector<vector<int>> dp(n + 1, vector<int>(sum + 1));
        dp[0][0] = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= sum; j++) {
                dp[i + 1][j] += dp[i][j];
                if (j + a[i] <= sum) dp[i + 1][j + a[i]] += dp[i][j];
            }
        }
        cout << dp[n][sum - 1] << '\n';
    }
    return 0;
}
