#include<bits/stdc++.h>
using namespace std;
int sum,ans;
int32_t main(){
    int n;
    cin>>n;
    for(int i=0;i<n;i++){
        int m,mx=0,mn=1000000000;
        cin>>m;
        for(int j=0;j<m;j++){
            int p;
            cin>>p;
            mx=max(p,mx);
            mn=min(p,mn);
        }
         sum+=mx;
         ans+=mn;
    }
    cout<<sum<<" "<<ans<<endl;
    return 0;
}
