#include <bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;
int a[N], b[N], m, n, f[N];
vector<pair<int, int>> event[N];
int get(int x) { return f[x] == x ? x : f[x] = get(f[x]); }
void join(int x, int y) { f[get(x)] = get(y); }

int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= m; i++) {
    int t, r;
    cin >> t >> r;
    event[r].push_back(make_pair(i, t));
  }
  iota(f, f + 2 * m + 5, 0);
  for (int r = 1; r <= n; r++) {
    for (auto it : event[r]) {
      int i = it.first, t = it.second;
      if (t == 1)
        join(i, i + 1);
      else
        join(i, i - 1);
    }
    int p = get(r);
    b[p] = a[r];
  }
  for (int r = 1; r <= n; r++) { cout << b[get(r)] << ' '; }
  cout << endl;
  return 0;
}
