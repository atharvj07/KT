import java.util.Scanner;

public class AntonAndArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] b = new int[n];
        int[] c = new int[n];
        int[] a = new int[n];

        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        boolean possible = true;
        for (int i = 0; i < n; i++) {
            int x = b[i] | c[i];
            if (x < b[i]) {
                possible = false;
                break;
            }
            a[i] = x;
        }

        if (possible) {
            for (int i : a) {
                System.out.print(i + " ");
            }
        } else {
            System.out.println(-1);
        }
    }
}
