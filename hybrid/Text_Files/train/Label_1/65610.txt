import sys
input = sys.stdin.readline

n = int(input())
s = []
for _ in range(n):
    s.append(input()[:-1])

ans = 0
for i in range(n):
    for j in range(i + 1, n):
        if s[i] == s[j]:
            ans += 2
        elif s[i] == '(':
            c1, c2 = 0, 0
            for k in s[j]:
                c1 += (k == '(')
                c2 -= (k == ')')
            if c1 > 0 and c2 == 0:
                ans += 1
        else:
            c1, c2 = 0, 0
            for k in s[j]:
                c2 -= (k == '(')
                c1 += (k == ')')
            if c1 > 0 and c2 == 0:
                ans += 1
print(ans)
