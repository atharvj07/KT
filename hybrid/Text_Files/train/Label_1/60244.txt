import sys

def solve():
    n = int(input())
    a = list(map(int, input().split()))
    p = []
    for x in a:
        if x not in p:
            p.append(x)
    q = []
    for x in a:
        if x in p:
            q.append(x)
            p.remove(x)
        else:
            q.append(x)
    print(*q)

for _ in range(int(input())): solve()
