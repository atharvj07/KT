import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for(int i=0; i<n; i++) a[i]= sc.nextInt();
        int sum=0;
        for(int i=0; i<n; i++){
            sum+=a[i];
            if(sum * 2 >= (i+1) * (i+2)) {
                System.out.println(i+1);
                return;
            }
        }
        System.out.println(-1);
    }
}
