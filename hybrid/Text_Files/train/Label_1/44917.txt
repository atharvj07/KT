import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), r = sc.nextInt(), ans = 0;
        long[][] ducks = new long[n][2];
        for (int i = 0; i < n; ++i) {
            ducks[i][0] = sc.nextLong();
            ducks[i][1] = sc.nextLong();
        }
        Arrays.sort(ducks, (a, b) -> a[1] == b[1] ? Long.compare(a[0], b[0]) : Long.compare(a[1], b[1]));
        long last = 0;
        for (long[] duck: ducks) {
            if (duck[1] - r >= last) {
                ans += 1;
                last = duck[1];
            }
        }
        System.out.print(ans);
    }
}
