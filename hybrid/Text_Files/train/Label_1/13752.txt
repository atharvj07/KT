#include <iostream>
using namespace std;

int main() {
  int n, x, y, ta, xa = 0, xb = 0;
  cin >> n;
  while (n--) {
    cin >> ta >> x >> y;
    if (ta == 1)
      xa += x;
    else
      xb += x;
  }
  cout << (xa > 50 ? "LIVE" : "DEAD") << endl;
  cout << (xb > 50 ? "LIVE" : "DEAD") << endl;
  return 0;
}
