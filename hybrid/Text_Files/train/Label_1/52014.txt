import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        for (int i = 1; i < k; i++) {
            for (int j = 1; j < k; j++) {
                int p = i * j;
                String s = Integer.toString(p, k);
                for (char c : s.toCharArray()) {
                    System.out.print(Character.getNumericValue(c));
                }
                System.out.print(" ");
            }
            System.out.println();
        }
    }
}
