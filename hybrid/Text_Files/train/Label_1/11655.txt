#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
const int MAXN = 5000 + 5;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int a[MAXN];
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            cin >> a[i];
            ans += (a[i] > 1);
        }
        cout << ans << endl;
    }
    return 0;
}
