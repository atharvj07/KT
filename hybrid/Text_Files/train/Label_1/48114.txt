import java.util.*;
import java.io.*;
import java.math.BigInteger;
import static java.lang.Math.*;

class Main {
    static long mod = 998244353;
    static long inv(long x) { return pow(x, mod - 2, mod); }
    static long pow(long x, long e, long m) {
        if (e == 0) return 1;
        if (e % 2 == 1) return x * pow(x, e - 1, m) % m;
        x = pow(x, e / 2, m) % m;
        return x * x % m;
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[] dist = new long[n], sum = new long[n + 1], ans = new long[n + 1];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i] += pow(sc.nextInt(), 2, mod);
            }
            dist[i] %= mod;
        }
        long S = 0;
        for (int i = 0, j = 0; i <= n; i++) {
            while (j < n && dist[j] <= i) {
                S += sum[i - (int)(dist[j])] % mod;
                S %= mod;
                j++;
            }
            ans[i] = (mod + S % mod * i % mod % mod) % mod;
            sum[i] = (mod + ans[i] % mod - (i > 0 ? ans[i - 1] : 0) % mod) % mod;
        }
        System.out.println(ans[n] % mod * inv(n) % mod);
    }
}
