import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    int[] v = new int[n];
    for (int i = 0; i < n; i++) {
      v[i] = sc.nextInt();
    }
    int[] dp = new int[k + 1];
    if (v[0] > 0) dp[1] = v[0];
    for (int i = 2; i <= k; i++) {
      for (int j = 0; j < n; j++) {
        if (i - j - 1 < 0) break;
        int sum = dp[i - j - 1];
        for (int l = j; l < n; l++) {
          sum += v[l];
          dp[i] = Math.max(dp[i], sum);
        }
      }
    }
    System.out.println(dp[k]);
  }
}
