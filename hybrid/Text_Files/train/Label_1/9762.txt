#include <iostream>
#include <string>
#include <map>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    string s;
    cin >> s;
    map<char, char> r;
    while (m--) {
        char x, y;
        cin >> x >> y;
        r[x] = y, r[y] = x;
    }
    for (int i = 0; i < n; i++) {
        cout << r[s[i]];
    }
    cout << endl;
    return 0;
}
