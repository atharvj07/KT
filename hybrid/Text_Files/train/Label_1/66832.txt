n = int(input())
a = input()
b = input()

def min_preprocess_moves(n, a, b):
    a_counts = [0] * 26
    b_counts = [0] * 26
    # Count occurrences of each character in both strings
    for i in range(n):
        a_counts[ord(a[i]) - ord('a')] += 1
        b_counts[ord(b[i]) - ord('a')] += 1
    # Check if there is a character that is in one string but not in the other
    for i in range(26):
        if (a_counts[i] + b_counts[i]) % 2 != 0:
            return -1
    # Calculate the minimum number of preprocess moves required
    preprocess_moves = 0
    for i in range(n // 2):
        # If characters at the symmetric positions are not equal
        if a[i] != b[n - i - 1]:
            # If swapping within the same string doesn't help, a preprocess move is needed
            if a[i] != b[i] and a[n - i - 1] != b[n - i - 1]:
                preprocess_moves += 1
    return preprocess_moves

print(min_preprocess_moves(n, a, b))
