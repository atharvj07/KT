from collections import Counter

def count_bulbasaurs(text):
    count = Counter(text)
    required_counts = Counter("Bulbasaur")
    bulbasaurs = float('inf')
    for char in required_counts:
        if char in count:
            bulbasaurs = min(bulbasaurs, count[char] // required_counts[char])
        else:
            return 0
    return bulbasaurs

# Reading input
text = input().strip()

# Printing the result
print(count_bulbasaurs(text))
