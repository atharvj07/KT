#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, sum = 0, ans = 0;
    cin >> n >> m;
    int arr[n];
    for (int i = 0; i < n; i++) { cin >> arr[i]; }
    for (int i = 0; i < n; i++) {
        if (sum + arr[i] > m) {
            ans++;
            sum = arr[i];
        } else
            sum += arr[i];
    }
    if (sum) ans++;
    cout << ans << endl;
    return 0;
}
