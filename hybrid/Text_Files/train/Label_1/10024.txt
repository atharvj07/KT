#include <iostream>
#include <vector>
#include <map>
#include <algorithm>

using namespace std;

int main() {
  int m, n;
  cin >> m;
  string essay;
  cin >> essay;
  cin >> n;
  map<string, vector<string>> dict;
  for (int i = 0; i < n; i++) {
    string a, b;
    cin >> a >> b;
    dict[a].push_back(b);
    dict[b].push_back(a);
  }
  vector<string> words;
  for (int i = 0; i < essay.size(); i++) {
    int start = i;
    while (i + 1 < essay.size() && essay[i + 1] != ' ') i++;
    words.push_back(essay.substr(start, i - start + 1));
    i++;
  }
  int ans1 = 1e9, ans2 = 1e9;
  vector<string> ans;
  for (string& word : words) {
    transform(word.begin(), word.end(), word.begin(), ::tolower);
    if (count(word.begin(), word.end(), 'r') < ans1) {
      ans1 = count(word.begin(), word.end(), 'r');
      ans2 = word.size();
      ans = {word};
    } else if (count(word.begin(), word.end(), 'r') == ans1) {
      if (word.size() < ans2) {
        ans2 = word.size();
        ans = {word};
      }
    }
    for (string& synonym : dict[word]) {
      transform(synonym.begin(), synonym.end(), synonym.begin(), ::tolower);
      if (count(synonym.begin(), synonym.end(), 'r') < ans1) {
        ans1 = count(synonym.begin(), synonym.end(), 'r');
        ans2 = synonym.size();
        ans = {synonym};
      } else if (count(synonym.begin(), synonym.end(), 'r') == ans1) {
        if (synonym.size() < ans2) {
          ans2 = synonym.size();
          ans = {synonym};
        }
      }
    }
  }
  cout << ans1 << ' ' << ans2 << endl;
  return 0;
}
