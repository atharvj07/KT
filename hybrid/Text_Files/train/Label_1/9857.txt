#include <iostream>
#include <vector>

using namespace std;

const long long INF = 1e18;
typedef long long ll;

int main() {
    int n;
    cin >> n;
    vector<ll> a(n);
    vector<int> p(n, -1);
    vector<ll> c(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 1; i < n; i++) {
        cin >> p[i] >> c[i];
        p[i]--;
    }
    vector<ll> dp(n, INF);
    vector<ll> best(n, INF);
    dp[0] = 0;
    for (int i = 1; i < n; i++) {
        if (a[i] >= 0) {
            best[i] = dp[p[i]];
        }
        if (best[p[i]] != INF) {
            dp[i] = best[p[i]] + c[i] + a[p[i]];
        }
        if (dp[i] >= a[i]) {
            dp[i] = INF;
        }
        if (p[p[i]] != -1) {
            if (dp[p[p[i]]] != INF) {
                best[i] = min(best[i], dp[p[p[i]]] + c[p[i]] + c[i] + a[p[p[i]]]);
            }
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (dp[i] == INF) {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
