#include<bits/stdc++.h>
using namespace std ;
const int N= 2e5+1 ;
vector<int> graph[N] ;
int color[N] ;
int ans ;
void dfs(int node, int par, int c){
    if(c!=color[node]) ans++ ;
    color[node]=c ;
    for(auto it : graph[node] ){
        if(it==par) continue ;
        dfs(it,node,c) ;
    }
}
int main(){
    int n, i, u, v ;
    cin>>n ;
    for(i=0;i<n;i++) cin>>color[i] ;
    for(i=0;i<n-1;i++){
        cin>>u>>v ;
        graph[u].push_back(v) ;
        graph[v].push_back(u) ;
    }
    ans=0 ;
    dfs(1,0,0) ;
    cout<<ans<<endl ;
    return 0 ;
}
