#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, x, y;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) {
    cin >> x;
    a[i] = x;
  }
  for (int i = 0; i < n; i++) {
    cin >> y;
    b[i] = y;
  }
  for (int i = 0; i < n; i++) {
    if (a[i] > b[i] + 1 || b[i] > a[i] + 1) {
      cout << "No" << endl;
      return 0;
    }
  }
  cout << "Yes" << endl;
  return 0;
}
