n, k = map(int, input().split())
a = list(map(int, input().split()))
odd = [a[i] for i in range(0, n, 2)]
even = [a[i] for i in range(1, n, 2)]
odd.sort(reverse=True)
even.sort(reverse=True)
if k & 1 == 0:
    k //= 2
    print(min(odd[:k] + even[:k]))
else:
    k = (k - 1) // 2
    print(min(odd[:k] + even[:k + 1]))
