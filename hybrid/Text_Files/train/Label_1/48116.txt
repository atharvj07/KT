import java.util.*;
import java.math.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();

        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = in.nextInt() - 1;
            }
        }

        long num = 0;
        long den = 1;
        for (int i = 0; i < n; i++) {
            long[] cur = count(dist, i);
            num = num * cur[0] / __gcd(num, cur[0]);
            num += cur[1];
            den *= cur[0];
        }

        long g = __gcd(num, den);
        num /= g;
        den /= g;

        if (den < 0) {
            num = -num;
            den = -den;
        }

        long mod = 998244353L;
        long p = pow(den % mod, mod - 2, mod);
        long ans = num * p % mod;
        if (ans < 0) ans += mod;
        System.out.println(ans);
    }

    static long[] count(int[][] dist, int city) {
        long num = 1;
        long den = 1;
        for (int i = 0; i < dist.length; i++) {
            if (i != city) {
                int d = dist[i][0];
                num *= d;
                den *= d + 1;
            }
        }

        long g = __gcd(num, den);
        num /= g;
        den /= g;

        return new long[] { num, den };
    }

    static long __gcd(long a, long b) {
        if (b == 0) return a;
        return __gcd(b, a % b);
    }

    static long pow(long base, long exp, long mod) {
        if (exp == 0) return 1;
        if (exp % 2 == 0) {
            long half = pow(base, exp / 2, mod);
            return half * half % mod;
        }
        return base * pow(base, exp - 1, mod) % mod;
    }
}
