#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
int main() {
  ll n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  map<char, vector<ll>> mp;
  for (ll i = 0; i < n; i++)
    mp[s[i]].pb(i);
  ll ans = 0;
  ll cnt = 0;
  for (auto &i : mp) {
    vector<ll> v = i.second;
    for (ll j = 0; j < v.size(); j++) {
      if (cnt + 1 <= k)
        ans += v[j], cnt++;
      else
        break;
    }
  }
  if (cnt == k)
    cout << ans << endl;
  else
    cout << -1 << endl;
  return 0;
}
