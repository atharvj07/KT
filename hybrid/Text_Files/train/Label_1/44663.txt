import java.util.HashSet;
import java.util.Scanner;

public class ConstructSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        HashSet<Integer> forbiddenProducts = new HashSet<>();
        for (int i = 0; i < n; i++) {
            forbiddenProducts.add(scanner.nextInt());
        }

        int k = 0;
        int currentProduct = 1;
        for (int i = 0; i < m; i++) {
            if (!forbiddenProducts.contains(currentProduct)) {
                k++;
                currentProduct = (currentProduct * i) % m;
            }
        }

        System.out.println(k);

        currentProduct = 1;
        for (int i = 0; i < m; i++) {
            if (!forbiddenProducts.contains(currentProduct)) {
                System.out.print(i + " ");
                currentProduct = (currentProduct * i) % m;
            }
        }
    }
}
