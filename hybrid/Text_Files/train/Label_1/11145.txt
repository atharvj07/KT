#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, d;
  cin >> n >> d;
  string s;
  cin >> s;
  vector<int> dp(n + 1, 100000000);
  dp[1] = 0;
  for (int i = 1; i <= n; i++) {
    if (s[i] == '0') continue;
    for (int j = 1; j <= d; j++) {
      if (i + j <= n) dp[i + j] = min(dp[i + j], dp[i] + 1);
    }
  }
  if (dp[n] == 100000000) cout << -1 << endl;
  else cout << dp[n] << endl;
  return 0;
}
