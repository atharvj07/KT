#include <iostream>

using namespace std;

int main() {
  long long n, x;
  cin >> n >> x;
  if (n == 1) {
    cout << "YES\n" << x << endl;
  } else if (n == 2) {
    if (x == 0)
      cout << "YES\n0 0" << endl;
    else if (x == 1)
      cout << "YES\n0 1" << endl;
    else
      cout << "NO" << endl;
  } else if (x == 0) {
    for (long long i = 0; i < n; i++) cout << 0 << " ";
    cout << endl;
  } else if (x == 1) {
    cout << "YES\n1 ";
    for (long long i = 0; i < n - 1; i++) cout << 0 << " ";
    cout << endl;
  } else if (x == 2) {
    cout << "YES\n2 0 ";
    for (long long i = 0; i < n - 2; i++) cout << 0 << " ";
    cout << endl;
  } else if (x == 3) {
    if (n >= 3)
      cout << "YES\n3 1 0";
    else
      cout << "NO";
    for (long long i = 3; i < n; i++) cout << " 0";
    cout << endl;
  } else if (x == 4) {
    if (n >= 3)
      cout << "YES\n4 2 0";
    else
      cout << "NO";
    for (long long i = 3; i < n; i++) cout << " 0";
    cout << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
