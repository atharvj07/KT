#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll t, x;
    cin >> t;
    while (t--) {
        cin >> x;
        ll cube = round(cbrt(x));
        if (cube * cube * cube == x)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
}
