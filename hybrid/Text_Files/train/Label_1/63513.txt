def solve():
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    e = [[] for _ in range(n+1)]
    for _ in range(n-1):
        u, v = map(int, input().split())
        e[u].append(v)
        e[v].append(u)
    
    ans = "NO"
    vis = [False] * (n+1)
    par = [0] * (n+1)
    sub = [0] * (n+1)

    def dfs(node, p, x):
        vis[node] = True
        par[node] = p
        sub[node] = a[node-1]
        for u in e[node]:
            if u != p:
                dfs(u, node, x)
                sub[node] ^= sub[u]
        if node == root and sub[node] != x:
            nonlocal ans
            ans = "YES"
    
    for i in range(1, n+1):
        if not vis[i]:
            root = i
            dfs(root, 0, a[i-1])
    
    if ans == "NO":
        for i in range(1, n+1):
            if not vis[i]:
                root = i
                p = -1
                for u in e[i]:
                    if sub[u] == a[i-1]:
                        p = u
                if p == -1:
                    continue
                dfs(p, i, a[i-1])
                if ans == "YES":
                    break
    
    print(ans)

for _ in range(int(input())):
    solve()
