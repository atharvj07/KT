#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
const int INF = 1 << 30;
int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int& i : v) cin >> i;
    if (n == 2) {
        cout << "1 2\n";
        return 0;
    }
    sort(v.begin(), v.end());
    int g = v[0];
    for (int i = 1; i < n; ++i) g = __gcd(g, v[i]);
    if (g > 1) {
        for (int i = 0; i < n; ++i) if (v[i] % g == 0) { cout << "1 "; break; }
        for (int i = 0; i < n; ++i) if (v[i] % g == 0 && v[i] != v[i - 1]) { cout << i + 1 << '\n'; break; }
    } else {
        cout << "1 " << n << '\n';
    }
    return 0;
}
