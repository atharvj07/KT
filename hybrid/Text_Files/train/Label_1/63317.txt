import sys

x1, y1 = map(int, input().split())
x2, y2 = map(int, input().split())
n = int(input())
s = input()

dx = [0, 0, -1, 1]
dy = [1, -1, 0, 0]

res = -1
cur_x, cur_y, dir = x1, y1, 0
appeared = {(cur_x, cur_y): []}
for i in range(len(s) + 1):
    nxt_x, nxt_y = cur_x + dx[dir], cur_y + dy[dir]
    if (nxt_x, nxt_y) == (x2, y2):
        res = i
        break
    if (nxt_x, nxt_y) in appeared and i in appeared[(nxt_x, nxt_y)]:
        break
    else:
        appeared[(nxt_x, nxt_y)] = appeared.get((nxt_x, nxt_y), []) + [i]
    if i < len(s):
        if s[i] == 'L':
            dir = (dir + 3) % 4
        elif s[i] == 'R':
            dir = (dir + 1) % 4

print(res)
