import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class RailroadProblem {

    private static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    private static class Line {
        Point start, end;
        int owner, type; // 1: elevated, 0: underground

        public Line(Point start, Point end, int owner, int type) {
            this.start = start;
            this.end = end;
            this.owner = owner;
            this.type = type;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int datasets = scanner.nextInt();

        for (int i = 0; i < datasets; i++) {
            Point a = new Point(scanner.nextInt(), scanner.nextInt());
            Point b = new Point(scanner.nextInt(), scanner.nextInt());
            int n = scanner.nextInt();

            List<Line> lines = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                Point start = new Point(scanner.nextInt(), scanner.nextInt());
                Point end = new Point(scanner.nextInt(), scanner.nextInt());
                int owner = scanner.nextInt();
                int type = scanner.nextInt();
                lines.add(new Line(start, end, owner, type));
            }

            int minDoors = solve(a, b, lines);
            System.out.println(minDoors);
        }
    }

    private static int solve(Point a, Point b, List<Line> lines) {
        int minDoors = 0;

        // Check for intersections with existing lines
        for (Line line : lines) {
            if (intersects(a, b, line.start, line.end)) {
                // If the existing line is owned by the same company, no door is needed
                if (line.owner == 1) {
                    continue;
                }

                // If the existing line is elevated, the new line must be underground at the intersection
                if (line.type == 1) {
                    minDoors++;
                }
            }
        }

        // Check for intersections with the endpoints of existing lines
        for (Line line : lines) {
            if (intersects(a, b, line.start) || intersects(a, b, line.end)) {
                // If the existing line is owned by the same company, no door is needed
                if (line.owner == 1) {
                    continue;
                }

                // If the existing line is elevated, the new line must be underground at the intersection
                if (line.type == 1) {
                    minDoors++;
                }
            }
        }

        return minDoors;
    }

    // Method to check if two line segments intersect
    private static boolean intersects(Point p1, Point p2, Point p3, Point p4) {
        double denominator = (p4.y - p3.y) * (p2.x - p1.x) - (p4.x - p3.x) * (p2.y - p1.y);

        // If denominator is 0, the lines are parallel
        if (denominator == 0) {
            return false;
        }

        double ua = ((p4.x - p3.x) * (p1.y - p3.y) - (p4.y - p3.y) * (p1.x - p3.x)) / denominator;
        double ub = ((p2.x - p1.x) * (p1.y - p3.y) - (p2.y - p1.y) * (p1.x - p3.x)) / denominator;

        // If ua and ub are between 0 and 1, the lines intersect
        return ua >= 0 && ua <= 1 && ub >= 0 && ub <= 1;
    }

    // Method to check if two points are the same
    private static boolean intersects(Point p1, Point p2, Point p) {
        return p1.x == p.x && p1.y == p.y || p2.x == p.x && p2.y == p.y;
    }
}
