#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
bool isprime(ll n)
{
    if (n == 1)
        return false;
    for (ll i = 2; i <= sqrt(n); ++i)
    {
        if (n % i == 0)
            return false;
    }
    return true;
}
int main()
{
    ll t;
    cin >> t;
    while (t--)
    {
        ll k, n;
        cin >> k >> n;
        string s = to_string(n);
        for (int i = 0; i < k - 1; i++)
        {
            string t = s.substr(i + 1);
            if (!isprime(stol(t)))
            {
                cout << t.size() << endl
                     << t << endl;
                break;
            }
        }
    }
    return 0;
}
