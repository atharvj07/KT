#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string a, b;
  cin >> a >> b;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] != b[i]) {
      if (a[i] == '0')
        ans += (i % 2 == 0);
      else
        ans += (i % 2 == 1);
    }
  }
  cout << ans << endl;
  return 0;
}
