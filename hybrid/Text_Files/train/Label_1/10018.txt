#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
using namespace std;
const int mxN = 5e5 + 5;
int n, x, y, h, w;
vector<pair<int, int>> a;
int main() {
	cin >> n;
	int mx1 = 0, mx2 = 0;
	while (n--) {
		char op; cin >> op;
		if (op == '+') {
			cin >> x >> y;
			mx1 = max(mx1, x);
			mx2 = max(mx2, y);
		}
		else {
			cin >> h >> w;
			cout << (max(h, w) >= mx1 && max(h, w) >= mx2 ? "YES" : "NO") << '\n';
		}
	}
	return 0;
}
