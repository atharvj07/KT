#include <bits/stdc++.h>
using namespace std;

const long long INF = 9e18;

int main() {
  int n;
  cin >> n;
  vector<long long> x(n);
  for (int i = 0; i < n; i++) { cin >> x[i]; }
  map<long long, long long> last;
  long long ans_ = 0;
  long long pow2 = 2;
  vector<long long> ans;
  for (int i = 0; i < n; i++) {
    long long best = 0;
    for (int pow = 0; pow < 64; pow++) {
      if (last.count(x[i] - pow2)) best = max(best, last[x[i] - pow2]);
      if (last.count(x[i] + pow2)) best = max(best, last[x[i] + pow2]);
    }
    if (best == 0) ans.push_back(x[i]);
    last[x[i]] = best + 1;
    ans_ = max(ans_, best + 1);
    pow2 *= 2;
  }
  cout << ans_ << endl;
  for (auto &p : ans) cout << p << " ";
  return 0;
}
