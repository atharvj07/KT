#include <iostream>
#include <string>
using namespace std;
const int MAXN = 100001;
int n,ans;
string s[MAXN];
int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> s[i];
        s[i] += s[i - 1];
    }
    int cnt = 0;
    char pre = '%';
    for (int i = 1; i < s[n].size(); ++i) {
        if (s[n][i] == pre) {
            cnt++;
        }
        else {
            ans = max(ans,cnt);
            cnt = 1;
        }
        pre = s[n][i];
    }
    ans = max(ans,cnt);
    cout << ans << endl;
    return 0;
}
