#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> cake(n);
  for (int i = 0; i < n; i++) { cin >> cake[i]; }
  int ans = 0;
  for (int r = 0; r < n; r++) {
    for (int c = 0; c < n; c++) {
      if (cake[r][c] == 'C') {
        ans++;
        for (int j = 0; j < n; j++) {
          if (cake[r][j] == 'C') ans++;
          if (cake[j][c] == 'C') ans++;
        }
      }
    }
  }
  ans /= 2;
  cout << ans << endl;
}
