#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> dp(n + 1, -1);
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
      int x = i, sum = 0;
      while (x) {
        sum += x % 10;
        x /= 10;
      }
      dp[i] = sum;
    }
    int ans = -1;
    for (int i = 0; i <= n - k; i++) {
      int sum = 0;
      for (int j = i; j <= i + k; j++) sum += dp[j];
      if (sum == n) {
        ans = i;
        break;
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
