#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n), b(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) cin >> b[i];
    sort(a.begin(), a.end());
    sort(b.rbegin(), b.rend());
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        ans += a[i] > 0 && b[i] > 0;
        ans += a[i] < 0 && b[i] < 0;
    }
    cout << ans << endl;
    return 0;
}
