for _ in range(int(input())):
    t=[0]*200000
    n,k,m=map(int,input().split())
    a=list(map(int,input().split()))
    for i in a:
        t[i]=1
    mn=a[0]
    for i in a:
        mn=min(i,mn)
    b=True
    for i in range(mn,n+1):
        if(k%2==1):
            if(t[i]==1 and t[i+k>>1]==0):
                b=False
                break
        else:
            if(t[i]==1 and (t[i+k//2]==0 or t[i+k//2+1]==0)):
                b=False
                break
    if(b):
        print('YES')
    else:
        print('NO')
