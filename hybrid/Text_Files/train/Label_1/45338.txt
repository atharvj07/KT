import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n + 1];
        int[] pos = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = sc.nextInt();
            pos[p[i]] = i;
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (p[i] == i) continue;
            ans++;
            int j = pos[i];
            int t = p[j];
            pos[t] = j;
            p[j] = t;
            pos[i] = i;
            p[i] = i;
        }
        System.out.println(ans);
    }
}
