#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int mod = 1e9 + 7;
const int N = 1e6 + 7;
ll dp[N][4];

int n, a[N], fact[N], invFact[N];

ll fastpow(ll base, ll p) {
  if (p == 0) return 1;
  if (p == 1) return base;
  ll val = fastpow(base, p / 2);
  val = (val * val) % mod;
  if (p % 2 != 0) val = (val * base) % mod;
  return val;
}

void preprocess() {
  fact[0] = 1;
  for (int i = 1; i < N; i++) fact[i] = (1LL * fact[i - 1] * i) % mod;
  invFact[N - 1] = fastpow(fact[N - 1], mod - 2);
  for (int i = N - 2; i >= 0; i--) invFact[i] = (1LL * invFact[i + 1] * (i + 1)) % mod;
}

ll nCr(int a, int b) {
  if (a < b) return 0;
  return (1LL * fact[a] * invFact[b] % mod) % invFact[a - b];
}

ll solve(int pos, int swaps) {
  if (pos == n) {
    if (swaps % 2 == 0) return 1;
    else return 0;
  }
  ll &ret = dp[pos][swaps];
  if (ret != -1) return ret;
  ret = 0;
  ret += solve(pos + 1, swaps);
  ret %= mod;
  if (a[pos] >= 1) {
    ret += (nCr(n - 1 - pos, swaps) * solve(pos + 1, swaps + 1)) % mod;
    ret %= mod;
  }
  if (a[pos] >= 2) {
    ret += (nCr(n - 1 - pos, swaps - 1) * solve(pos + 1, swaps + 1)) % mod;
    ret %= mod;
  }
  return ret;
}

int main() {
  memset(dp, -1, sizeof dp);
  preprocess();
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  cout << solve(0, 0) << endl;
  return 0;
}
