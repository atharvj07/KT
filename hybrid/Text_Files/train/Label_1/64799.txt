n, k, m = map(int, input().split())
powers = list(map(int, input().split()))
powers.sort()

max_avg = sum(powers) / n

for i in range(min(n - 1, m)):
    remove_lowest = (sum(powers[i + 1:]) + min(m - i, k * (n - i - 1))) / (n - i - 1)
    max_avg = max(max_avg, remove_lowest)

print("{:.20f}".format(max_avg))
