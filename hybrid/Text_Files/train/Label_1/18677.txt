#include <iostream>
#include <cmath>
using namespace std;

int main() {
    long long n;
    cin >> n;
    int cnt = 0;
    for (long long i = 2; i * i <= n; i++) {
        while (n % i == 0) {
            n /= i;
            cnt++;
        }
    }
    if (n > 1) cnt++;
    cout << cnt << endl;
    return 0;
}
