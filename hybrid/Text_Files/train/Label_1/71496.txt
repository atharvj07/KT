from collections import Counter

def solve(n, m, days):
    max_plays = (m + 1) // 2
    selection = []
    counts = Counter()

    for day in days:
        chosen = None
        for friend in day:
            if counts[friend] < max_plays:
                chosen = friend
                break
        if chosen is None:
            return "NO"
        counts[chosen] += 1
        selection.append(str(chosen))

    return "YES\n" + " ".join(selection)

t = int(input().strip())
for _ in range(t):
    n, m = map(int, input().split())
    days = [list(map(int, input().split()))[1:] for _ in range(m)]
    print(solve(n, m, days))
