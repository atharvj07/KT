from collections import defaultdict, namedtuple
import heapq

# Read input
n, m, s, t = map(int, input().split())
edges = [namedtuple('Edge', 'a b w')(*map(int, input().split())) for _ in range(m)]

# Build graph
graph = defaultdict(list)
for edge in edges:
    graph[edge.a].append((edge.b, edge.w))

# Dijkstra to find minimum distance from s to t
dist = [float('inf') for _ in range(n+1)]
dist[s] = 0
pq = [(0, s)]
while pq:
    d, u = heapq.heappop(pq)
    if u == t:
        break
    if d > dist[u]:
        continue
    for v, w in graph[u]:
        if d + w < dist[v]:
            dist[v] = d + w
            heapq.heappush(pq, (d+w, v))

# Check if each edge is on the shortest path from s to t
for edge in edges:
    if dist[edge.a] + edge.w == dist[edge.b]:
        print('YES')
    else:
        if dist[edge.b] - edge.w > 0:
            print(f'CAN {dist[edge.b] - edge.w}')
        else:
            print('NO')
