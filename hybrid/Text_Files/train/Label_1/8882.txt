#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

int main(){
    int n,l,v1,v2,k;
    cin>>n>>l>>v1>>v2>>k;
    if(v1>=v2){
        cout<<(double)l/v1<<endl;
        return 0;
    }
    int trips=ceil((double)n/k);
    double tb=l/v1,tb2=0;
    double n2=n;
    for(int i=1;i<=trips;++i){
        double x=ceil(n2/k);
        tb2+=(l/v2)*(x);
        n2=n2-x*k;
        tb+=l/v1;
    }
    double ans=min(tb,tb2);
    printf("%.10lf\n",ans);
    return 0;
}
