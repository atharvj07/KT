int32_t main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);cout.tie(0);
    int n,c1=0,c2=0,m=-1e9,ans=0;
    cin>>n;
    int a[n+100];
    for(int i=1; i<=n; i++)
    {
        cin>>a[i];
    }
    for(int i=1; i<=n; i++)
    {
        if(a[i]==1)
            c1++;
        else
            break;
    }
    for(int i=n; i>=1; i--)
    {
        if(a[i]==1)
            c2++;
        else
            break;
    }
    for(int i=1; i<=n; i++)
    {
        if(a[i]==1)
            continue;
        else
        {
            c1--;
            c2++;
            if(c1>c2)
            {
                ans++;
            }
        }
    }
    cout<<ans+c2<<endl;
    return 0;
}
