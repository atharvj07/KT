import java.util.Scanner;

public class MagnetGroups {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        int groups = 1; // Start with one group for the first magnet
        char prevPole = scanner.nextLine().charAt(0); // Read and store the first magnet's pole

        for (int i = 1; i < n; i++) {
            char currPole = scanner.nextLine().charAt(0);
            if (currPole != prevPole) {
                groups++; // Different poles, new group
            }
            prevPole = currPole; // Update previous pole for next comparison
        }

        System.out.println(groups);
    }
}
