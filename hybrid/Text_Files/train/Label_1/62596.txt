import re

for _ in range(int(input())):
    n = int(input())
    s = input()
    if s.count('?') == 0 or '?' not in s[:6] or '?' not in s[-6:]:
        print('No')
    elif re.match(r'^ab.*ba.*ba$', s) or re.match(r'^ba.*ab.*ba$', s):
        print('Yes')
        s = re.sub(r'\?', 'c', s)
        print(s)
    else:
        print('No')
