n, p = map(int, input().split())
s = input()
pals = []
for l in range(2, n+1):
    for r in range(n - l + 1):
        t = s[r:r+l]
        if t == t[::-1]:
            pals.append((r, l))
pals.sort(reverse=True)
ok = 1
for x, y in pals:
    if s[(x+y-1)%n] != s[x]:
        t = list(s)
        t[(x+y-1)%n] = s[x]
        t[x] = s[(x+y-1)%n]
        print(''.join(t))
        ok = 0
        break
if ok:
    print('NO')
