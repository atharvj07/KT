#include <bits/stdc++.h>
using namespace std;
#define rep(i, a, b) for (int i = (a); i < (b); i++)

using ll = long long;
const ll inf = 1LL << 60;

struct edge {
    int to, id; ll cst;
    edge *rev;
    edge() : to(0), id(0), cst(0), rev(nullptr) {}
    edge(int t, ll c) : to(t), cst(c), rev(nullptr) {}
};

vector<edge*> g[100010];

struct dijkstra {
    ll dst[100010];
    vector<int> ord;
    void run(int s) {
        rep(i, 0, 100010) dst[i] = inf;
        priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll, int>>> pq;
        dst[s] = 0;
        pq.push({0, s});
        while (!pq.empty()) {
            int v = pq.top().second; pq.pop();
            ord.push_back(v);
            for (edge* e : g[v]) {
                int u = e->to;
                if (dst[v] + e->cst < dst[u]) {
                    dst[u] = dst[v] + e->cst;
                    pq.push({dst[u], u});
                }
            }
        }
    }
};

dijkstra dk;

int main() {
    int n, m, k; cin >> n >> m >> k;
    vector<int> sp(k);
    rep(i, 0, k) { cin >> sp[i]; sp[i]--; }
    rep(i, 0, m) {
        int a, b, c; cin >> a >> b >> c;
        a--; b--;
        edge* ab = new edge(b, c);
        edge* ba = new edge(a, c);
        ab->rev = ba;
        ba->rev = ab;
        g[a].push_back(ab);
        g[b].push_back(ba);
    }
    for (int j : sp) dk.run(j);
    for (int j : sp) {
        int id = -1;
        for (int i : dk.ord) if (sp[j] == i) { id = i; break; }
        if (id == -1) cout << -1 << " ";
        else cout << dk.dst[id] << " ";
    }
    cout << endl;
}
