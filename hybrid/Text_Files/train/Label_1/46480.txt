import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int numTests = in.nextInt();

        for (int test = 0; test < numTests; test++) {
            boolean[][] cubes = new boolean[5][5];
            for (int i = 0; i < 5; i++) {
                for (int j = 0; j < 5; j++) {
                    cubes[i][j] = in.nextInt() != 0;
                }
            }

            if (!isProper(cubes)) {
                System.out.println("false");
                continue;
            }

            Set<Integer> faceSums = new HashSet<>();
            for (int i = 0; i <= 2; i++) {
                for (int j = i + 2; j < 5; j++) {
                    int sum = 0;
                    for (int k = 0; k < 5; k++) {
                        if (cubes[i][k] || cubes[j][k]) {
                            sum++;
                        }
                    }
                    faceSums.add(sum);
                }
            }

            if (faceSums.size() != 3 || !faceSums.contains(7)) {
                System.out.println("false");
            } else {
                System.out.println("true");
            }
        }
    }

    public static boolean isProper(boolean[][] cubes) {
        int total = 0;
        for (boolean[] row : cubes) {
            for (boolean cell : row) {
                if (cell) {
                    total++;
                }
            }
        }

        if (total != 24) {
            return false;
        }

        for (int i = 0; i < 5; i++) {
            if (cubes[i][i] || cubes[i][4 - i] || cubes[2][i] || cubes[2][4 - i]) {
                return false;
            }
        }

        return true;
    }
}
