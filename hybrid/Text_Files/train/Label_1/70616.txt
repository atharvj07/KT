n, k, x = map(int, input().split())
a = list(map(int, input().split()))

# Precompute powers of x to avoid recalculating them
powers_of_x = [1] * (k+1)
for i in range(1, k+1):
    powers_of_x[i] = powers_of_x[i-1] * x

# Precompute prefix and suffix OR arrays
prefix_or = [0] * n
suffix_or = [0] * n

prefix_or[0] = a[0]
for i in range(1, n):
    prefix_or[i] = prefix_or[i-1] | a[i]

suffix_or[n-1] = a[n-1]
for i in range(n-2, -1, -1):
    suffix_or[i] = suffix_or[i+1] | a[i]

# Calculate the maximum OR value
max_or = 0
for i in range(n):
    # Calculate OR value if we multiply a[i] by x^k
    before = prefix_or[i-1] if i > 0 else 0
    after = suffix_or[i+1] if i < n-1 else 0
    current = a[i] * powers_of_x[k]
    max_or = max(max_or, before | current | after)

print(max_or)
