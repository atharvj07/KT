#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int sum = 0;
        int min_ops = 0;
        while (n--) {
            int x;
            cin >> x;
            sum += x;
        }
        min_ops = max(0, 1 - sum);
        cout << min_ops << endl;
    }
}
