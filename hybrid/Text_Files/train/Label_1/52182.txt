import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<List<int[]>> cards = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            List<int[]> turn = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                int c = sc.nextInt();
                int d = sc.nextInt();
                turn.add(new int[]{c, d});
            }
            cards.add(turn);
        }
        sc.close();
        int totalDamage = 0;
        for (List<int[]> turn : cards) {
            Collections.sort(turn, (a, b) -> a[1] - b[1]);
            int cost = 0;
            int cnt = 0;
            for (int[] card : turn) {
                if (cost + card[0] <= 3) {
                    totalDamage += card[1];
                    cost += card[0];
                    cnt++;
                }
            }
            if (cnt % 10 < 2) {
                totalDamage += Math.max(0, turn.get(cnt - 1)[1]);
            }
        }
        System.out.println(totalDamage);
    }
}
