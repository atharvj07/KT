#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    vector<int> max_pre(n), min_pre(n);
    max_pre[0] = a[0];
    min_pre[0] = a[0];
    for (int i = 1; i < n; ++i) {
      max_pre[i] = max(max_pre[i - 1], a[i]);
      min_pre[i] = min(min_pre[i - 1], a[i]);
    }
    long long ans = 0;
    for (int i = 0; i < n - 1; ++i) { ans = max(ans, 1LL * max_pre[i] * min_pre[i + 1]); }
    cout << ans << "\n";
  }
  return 0;
}
