from array import array
n,m=map(int,input().split())
l=list(map(int,input().split()))
s=[[i] for i in range(m)]
for i in range(m):
 r,l=map(lambda x:x-1,input().split())
 for j in range(l,r):s[j].append(i)
l=sorted(l)
d=array('i',l)
for i in s:
 for j in range(len(i)):
  k=i[j]
  if d[l[k]]<l[k]:d[l[k]]=l[k]
  if d[l[k]+1]>l[k]:d[l[k]+1]=l[k]
print(max(d)-min(d),len([i for i in d if i!=l[0]]))
print(len([i for i in d if i!=l[0]]))
print(*[i for i in s if len([j for j in i if d[l[j]]!=l[0]])>0])
