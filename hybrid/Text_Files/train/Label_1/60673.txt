def max_prettiness(q, queries):
    results = []
    for n, arr in queries:
        arr.sort(reverse=True)
        max_pretty = arr[0]

        for i in range(n):
            for j in range(i+1, n):
                if arr[i] % arr[j] != 0:
                    max_pretty = max(max_pretty, arr[i] + arr[j])
                    for k in range(j+1, n):
                        if arr[i] % arr[k] != 0 and arr[j] % arr[k] != 0:
                            max_pretty = max(max_pretty, arr[i] + arr[j] + arr[k])
                            break
                    break
        results.append(max_pretty)
    return results

q = int(input())
queries = []
for _ in range(q):
    n = int(input())
    arr = list(map(int, input().split()))
    queries.append((n, arr))

results = max_prettiness(q, queries)
for result in results:
    print(result)
