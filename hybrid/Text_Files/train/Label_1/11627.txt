#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  int l = 0, o = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '-')
      l++;
    else
      o++;
  }
  if (l % (o - 1))
    cout << "NO" << endl;
  else
    cout << "YES" << endl;
  return 0;
}
