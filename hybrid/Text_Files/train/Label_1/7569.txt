#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;
    int a[m]={0};
    int j=1,k=m,l=n,u=2*n-1;
    for(int i=0;i<m;i++){
        if(i/2<n){
            if(i%2==0)
                a[i]=j++;
            else
                a[i]=l++;
        }
        else{
            if(i%2==0)
                a[i]=u--;
            else
                a[i]=k--;
        }
    }
    for(int i=0;i<m;i++)
        cout<<a[i]<<" ";
    cout<<endl;
    return 0;
}
