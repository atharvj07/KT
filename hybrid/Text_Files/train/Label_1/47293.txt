import java.util.*;

public class Main {
    static List<Node>[] graph;
    static int[] color, dist;
    static Node INF = new Node(0, 0);

    static class Node {
        int v, w;

        Node(int v, int w) {
            this.v = v;
            this.w = w;
        }

        public Node() {
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (!(o instanceof Node)) return false;

            Node node = (Node) o;

            return v == node.v;
        }

        @Override
        public int hashCode() {
            return v;
        }
    }

    static int dijkstra(int start, int end) {
        Queue<Node> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a.w));
        pq.offer(new Node(start, 0));
        dist[start] = 0;
        while (!pq.isEmpty()) {
            Node cur = pq.poll();
            if (cur.w != dist[cur.v]) continue;
            if (cur.v == end) return cur.w;
            for (Node next : graph[cur.v]) {
                int nw = cur.w + (color[cur.v] != color[next.v] ? 1 : 0);
                if (nw < dist[next.v]) {
                    pq.offer(new Node(next.v, nw));
                    dist[next.v] = nw;
                }
            }
        }
        return -1;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt(), m = sc.nextInt();
        graph = new List[n + 1];
        for (int i = 0; i <= n; i++) graph[i] = new ArrayList<>();
        color = new int[n + 1];
        dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);

        for (int i = 0; i < m; i++) {
            int u = sc.nextInt(), v = sc.nextInt(), c = sc.nextInt();
            graph[u].add(new Node(v, c));
            graph[v].add(new Node(u, c));
        }

        for (int i = 1; i <= n; i++) {
            if (graph[i].isEmpty()) {
                System.out.println(-1);
                System.out.println("1".repeat(n));
                return;
            }
        }

        int mx = 0;
        for (int i = 1; i <= n; i++) {
            if (color[i] == 0) {
                for (Node next : graph[i]) color[next.v] = 1;
            } else {
                for (Node next : graph[i]) color[next.v] = 0;
            }
            mx = Math.max(mx, dijkstra(1, n));
            color[i] = 0;
        }

        if (mx == -1) {
            System.out.println(-1);
            System.out.println("1".repeat(n));
        } else {
            System.out.println(mx);
            for (int i = 1; i <= n; i++) System.out.print(color[i]);
            System.out.println();
        }
    }
}
