import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);
        dp[0] = 0;
        for (int i = 1; i <= n; i++) {
            if (i - 1 >= 0 && dp[i - 1] != -1 && (dp[i - 1] + 1) % m == 0) {
                dp[i] = dp[i - 1] + 1;
            }
            if (i - 2 >= 0 && dp[i - 2] != -1 && (dp[i - 2] + 1) % m == 0) {
                dp[i] = dp[i - 2] + 1;
            }
        }
        System.out.println(dp[n] == -1 ? -1 : dp[n]);
    }
}
