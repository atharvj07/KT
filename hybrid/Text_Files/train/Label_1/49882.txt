import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = br.readLine().split(" ");
        int n = Integer.parseInt(nk[0]);
        int k = Integer.parseInt(nk[1]);
        int[] arr = new int[n];
        String[] a = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            arr[i] = Integer.parseInt(a[i]);
        }
        Map<Integer, Integer> map = new HashMap<>();
        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);
        System.out.println(rec(arr, map, 0, dp, k));
    }

    public static int rec(int[] arr, Map<Integer, Integer> map, int idx, int[] dp, int k) {
        if (idx == arr.length) {
            if (map.size() == 0) return 0;
            int ans = 0;
            for (int key : map.keySet()) {
                ans += (map.get(key) + 1) * (map.get(key)) / 2;
            }
            return ans;
        }
        if (dp[idx] != -1) return dp[idx];
        int x = rec(arr, map, idx + 1, dp, k);
        int y = 0;
        if (k > 0) {
            if(map.containsKey(arr[idx])) {
                map.put(arr[idx], map.get(arr[idx]) + 1);
            } else {
                map.put(arr[idx], 1);
            }
            y = rec(arr, map, idx + 1, dp, k - 1);
            map.put(arr[idx], map.get(arr[idx]) - 1);
            if (map.get(arr[idx]) == 0) {
                map.remove(arr[idx]);
            }
        }
        return dp[idx] = Math.max(x, y);
    }
}
