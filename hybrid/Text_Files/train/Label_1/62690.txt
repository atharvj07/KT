import math
n = int(input())
a = []
for i in range(n):
    x, y = map(int, input().split())
    a.append((math.atan2(y, x), math.hypot(x, y)))
a.sort()
ans = -1
for i in range(len(a)):
    for j in range(i + 1, len(a)):
        min_angle = math.degrees(a[j][0] - a[i][0])
        if min_angle > ans:
            ans = min_angle
        if min_angle >= 180:
            break
print(ans)
