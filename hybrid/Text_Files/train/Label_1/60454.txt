import sys
input=sys.stdin.readline
n,a,b=map(int,input().split())
lst=[list(map(int,input().split())) for i in range(n)]
for i in lst:
    i[0],i[1]=sorted(i)
lst.sort()
j=0
ans=0
while j<n and a:
    a-=1
    ans+=lst[j][0]
    lst[j][0]*=2
    lst[j][0],lst[j][1]=sorted(lst[j])
    j+=1
lst.sort()
k=n-1
while k>=0 and b:
    b-=1
    ans+=lst[k][0]
    lst[k][0],lst[k][1]=sorted(lst[k])
    k-=1
lst.sort()
for i in lst:
    ans+=i[1]
print(ans)
