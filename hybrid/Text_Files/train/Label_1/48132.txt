import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[sc.nextInt()];
        int k = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int q = sc.nextInt();
        sc.nextLine();
        while (q-- > 0) {
            String[] s = sc.nextLine().split(" ");
            if (s[0].equals("1")) {
                arr[Integer.parseInt(s[1]) - 1] += Integer.parseInt(s[2]);
            } else {
                System.out.println(getAns(Arrays.copyOfRange(arr, Integer.parseInt(s[1]) - 1, Integer.parseInt(s[1]) - 1 + k), a, b));
            }
        }
    }

    private static int getAns(int[] arr, int a, int b) {
        int ans = 0, pr = 0;
        for (int i = 0; i < arr.length; ++i) {
            pr = pr + Math.min(a, arr[i] + pr - b);
            ans = Math.max(ans, pr);
        }
        return ans;
    }
}
