#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> cnt(5001);
  vector<int> seg(n, -1);
  int ans = 0, p = -1;
  for (int l = 0, r = 0; r < n; r++) {
    cnt[a[r]]++;
    if (cnt[a[r]] == 1) {
      seg[r] = p;
      p = r;
      ans ^= a[r];
    } else {
      int j = l;
      while (j < r && cnt[a[r]] > 1) {
        cnt[a[j]]--;
        if (cnt[a[j]] == 1) {
          seg[j] = p;
        }
        j++;
      }
      l = j;
      p = seg[l - 1];
      ans ^= a[r];
    }
  }
  cout << ans << endl;
}
