from collections import Counter

x, y, x0, y0 = map(int, input().split())
s = input()

c = Counter([(1 if s[i] == 'U' else -1 if s[i] == 'D' else 0) for i in range(len(s))])
ans = [0 for i in range(0, len(s) + 1)]
for i in range(1, len(s) + 1):
    y1 = y0 + c[i]
    if (0 <= y1 < y):
        ans[i] += ans[i - 1]
    if (c[i] == 0):
        ans[i] += x * y - x0 * y
print(*ans)
