import sys
def dfs1(u, p):
    vis[u] = True
    for v in adj[u]:
        if v == p:
            continue
        if vis[v] == True:
            return True
        if dfs1(v, u) == True:
            return True
    return False

def dfs2(u, p):
    vis[u] = True
    for v in adj[u]:
        if v == p:
            continue
        if vis[v] == True:
            return True
        if dfs2(v, u) == False:
            return False
    return True

n, m = map(int, sys.stdin.readline().split())
adj = [[] for i in range(n + 1)]
for i in range(m):
    u, v = map(int, sys.stdin.readline().split())
    adj[u].append(v)
vis = [False for i in range(n + 1)]
ans = 'NO'
for i in range(1, n + 1):
    if vis[i] == True:
        continue
    if dfs1(i, -1) == True:
        ans = 'YES'
        break
    vis = [False for i in range(n + 1)]
    if dfs2(i, -1) == False:
        ans = 'YES'
        break
print(ans)
