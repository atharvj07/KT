#include <bits/stdc++.h>

using namespace std;

bool win(int n, vector<int>& a) {
    int mex = 0;
    sort(a.begin(), a.end());
    for (int i = 0; i < n; i++) {
        if (a[i] != mex) break;
        mex++;
    }
    return mex % 2 != n % 2;
}

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    cout << (win(n, a) ? "Alice" : "Bob") << endl;
    return 0;
}
