t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    h = list(map(int, input().split()))
    for _ in range(k):
        for i in range(n - 1):
            if h[i] < h[i + 1]:
                h[i] += 1
                break
        else:
            print(-1)
            break
    else:
        print(i + 1)
