#include <vector>
#include <deque>
#include <algorithm>
#include <iostream>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n;
        std::cin >> n;
        std::vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            std::cin >> a[i];
        }

        std::deque<int> b = {a[0]};
        for (int i = 1; i < n; ++i) {
            if (a[i] >= b.front()) {
                b.push_front(a[i]);
            } else {
                b.push_back(a[i]);
            }
        }

        int ans = 0;
        for (int i = 0; i < n - 1; ++i) {
            if (b[i] > b[i + 1]) {
                ++ans;
            }
        }

        std::cout << ans << std::endl;
    }
    return 0;
}
