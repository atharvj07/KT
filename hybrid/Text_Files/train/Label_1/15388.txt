#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
const int MAXN = 1 << 20;
int c[MAXN], N, L[MAXN];
long long ans;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> N;
    for (int i = 0; i < N; i++) {
        cin >> L[i];
        c[L[i]]++;
    }
    sort(L, L + N);
    for (int i = MAXN - 1; i >= 2; i--)
        if (c[i])
            if (c[i] >= 4)
                c[i - 1] += c[i] / 4, ans += (long long)(i - 1) * (long long)(i - 1) * (c[i] - c[i] % 4);
            else
                c[i - 1] += c[i], ans += (long long)(i - 1) * (long long)(i - 1) * c[i];
    cout << ans << "\n";
    return 0;
}
