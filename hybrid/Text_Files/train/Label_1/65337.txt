n = int(input())
a = list(map(int, input().split()))
m = int(input())
for i in range(m):
    x, y = map(int, input().split())
    y -= 1
    for j in range(y):
        a[x - 1] -= 1
        a[x] += 1
    for j in range(y + 1, len(a)):
        a[x - 1] -= 1
        a[x - 2] += 1
for i in a:
    print(i)
