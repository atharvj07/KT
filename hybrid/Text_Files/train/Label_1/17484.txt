#include <bits/stdc++.h>
using namespace std;
#define int long long
const int maxn = 2e5 + 5;
int n, k, a[maxn], b[maxn];
main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    int sum = 0;
    for (int i = 1; i <= n; i++) {
        int x = a[i];
        b[i] = x;
        sum -= (x - 1) * (x - 1);
        sum += x * x;
    }
    sort(a + 1, a + n + 1);
    int med = a[(n + 1) / 2];
    for (int i = 1; i <= n; i++) {
        if (k == 0) break;
        if (b[i] != med) {
            k--;
            a[i]++;
            sum += 2 * (med - b[i]);
            b[i] = med;
        }
    }
    sort(a + 1, a + n + 1);
    cout << a[(n + 1) / 2] << endl;
}
