def solve_petya_problem(n, m, k, start_positions, final_positions):
    # Since we need to ensure every chip visits its final position at least once
    # and considering the limit of operations, a strategy that works is to first
    # move all chips to one corner and then sweep the entire board.

    # Moving all chips to top-left corner (1,1)
    # This can be achieved by first moving all chips up (U) n-1 times and then left (L) m-1 times.
    operations = "U"*(n-1) + "L"*(m-1)

    # Now, we sweep the entire board to ensure every chip visits its final position.
    # We do this by moving right and left while going down row by row.
    for row in range(n):
        if row > 0:
            operations += "D"
        operations += "R"*(m-1) if row % 2 == 0 else "L"*(m-1)

    return len(operations), operations

# Reading input
n, m, k = map(int, input().split())
start_positions = [tuple(map(int, input().split())) for _ in range(k)]
final_positions = [tuple(map(int, input().split())) for _ in range(k)]

# Solve
op_count, operations = solve_petya_problem(n, m, k, start_positions, final_positions)

# Output
print(op_count)
print(operations)
