#include<iostream>
#include<set>
using namespace std;const int N=1e9;int n,m;typedef pair<int,int> pi;set<pi> exclude;long long ans;int main(){cin>>n>>m;while(m--){int l1,b1,r1,t1;cin>>l1>>b1>>r1>>t1;for(long long i=l1;i<=r1;i++)exclude.insert({i,b1}),exclude.insert({i,t1});}for(long long i=1;i<=n;i++){auto it=exclude.lower_bound({i,0});if(it==exclude.end()||it->first!=i)ans+=(n-i+1);}cout<<ans<<endl;return 0;}
