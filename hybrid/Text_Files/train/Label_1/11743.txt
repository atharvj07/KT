#include <bits/stdc++.h>

using namespace std;

int get_ans(const string &s, long long a) {
    int ans = 0;
    const int n = s.size();
    vector<long long> pref(n + 1);
    for (int i = 0; i < n; ++i) {
        pref[i + 1] = pref[i] * 10 + s[i] - '0';
    }
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) {
            ans += (pref[j] * pref[i - 1] == a);
        }
    }
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) {
            for (int k = i; k <= n; ++k) {
                long long cur = pref[k] * pref[i - 1] * pref[j - 1] - pref[k] * pref[j - 1] - pref[k - 1] * pref[i - 1] + pref[i - 1];
                ans += cur == a;
            }
        }
    }
    return ans;
}

int main() {
    long long a;
    cin >> a;
    string s;
    cin >> s;
    cout << get_ans(s, a) << '\n';
}
