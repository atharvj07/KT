#include <bits/stdc++.h>
#define ll long long
#define pb push_back
using namespace std;
const int maxn = 1e5 + 5;
ll b[maxn], a[maxn], p[maxn];
int main() {
    int n; cin >> n;
    for (int i = 1; i <= n; i++) cin >> b[i];
    for (int i = n, j = 1; i >= 1; i--) {
        if (j > 1) p[i] = b[i];
        a[i] = a[i + 1] ^ b[i]; j++;
    }
    sort(a + 1, a + n + 1);
    sort(p + 1, p + n + 1);
    for (int i = 1; i <= n; i++) {
        if (a[i] != p[i]) { cout << "No" << endl; return 0; }
    }
    cout << "Yes" << endl;
    for (int i = 1; i <= n; i++) cout << b[i] << ' ';
    cout << endl;
    return 0;
}
