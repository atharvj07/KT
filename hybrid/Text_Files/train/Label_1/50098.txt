import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Stack<Integer> stack = new Stack<>();
        Queue<Integer> queue = new ArrayDeque<>();
        ArrayDeque<Integer> deque = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            int t = in.nextInt();
            if (t > 0) {
                System.out.println(push(t, stack, queue, deque));
            } else {
                System.out.println(pop(stack, queue, deque));
            }
        }
    }

    public static String push(int t, Stack<Integer> stack, Queue<Integer> queue, ArrayDeque<Integer> deque) {
        if (stack.isEmpty()) {
            stack.push(t);
            return "pushStack";
        } else if (queue.isEmpty()) {
            queue.offer(t);
            return "pushQueue";
        } else if (deque.isEmpty()) {
            deque.offerLast(t);
            return "pushBack";
        } else {
            deque.offerFirst(t);
            return "pushFront";
        }
    }

    public static String pop(Stack<Integer> stack, Queue<Integer> queue, ArrayDeque<Integer> deque) {
        int size = 0;
        if (!stack.isEmpty()) {
            size++;
        }
        if (!queue.isEmpty()) {
            size++;
        }
        if (!deque.isEmpty()) {
            size++;
        }
        System.out.print(size + " ");
        if (!stack.isEmpty()) {
            System.out.print("popStack ");
        }
        if (!queue.isEmpty()) {
            System.out.print("popQueue ");
        }
        if (!deque.isEmpty()) {
            if (deque.size() > 1) {
                System.out.print("popBack ");
            } else {
                System.out.print("popFront ");
            }
        }
        stack.clear();
        queue.clear();
        deque.clear();
        System.out.println();
        return "";
    }
}
