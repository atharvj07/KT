t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    possible = True
    has_pos = has_neg = False
    
    for i in range(n):
        if a[i] < b[i]:
            if not has_pos:
                possible = False
                break
        elif a[i] > b[i]:
            if not has_neg:
                possible = False
                break
        if a[i] == 1:
            has_pos = True
        elif a[i] == -1:
            has_neg = True
    
    if possible:
        print("YES")
    else:
        print("NO")
