#include <bits/stdc++.h>

using namespace std;

int main() {
  string s;
  int t;
  cin >> t;
  while (t--) {
    cin >> s;
    int sz = s.size();
    bool ok = 1;
    string t = "abcdefghijklmnopqrstuvwxyz";
    for (int i = 0; i < sz - 1; i++) {
      ok &= s[i] != s[i + 1] || t.find(s[i]) + 1 != t.find(s[i + 1]);
    }
    if (ok) {
      cout << s << "\n";
    } else {
      cout << "No answer\n";
    }
  }
  return 0;
}
