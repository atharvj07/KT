from collections import deque

n = int(input())

adj = [[]]
for i in range(n - 1):
    x, y = map(lambda z: int(z) - 1, input().split())
    adj[x].append(y)
    adj[y].append(x)

a = list(map(lambda z: int(z) - 1, input().split()))
used = [False] * n
q = deque([0])
used[0] = True
i = 0
while i < n and q:
    v = q.popleft()
    if v != a[i]:
        print("No")
        exit()
    i += 1
    for u in adj[v]:
        if not used[u]:
            q.append(u)
            used[u] = True
if i == n:
    print("Yes")
else:
    print("No")
