#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 2e5 + 10;
int a[N], mx[N];
int main() {
    int n;
    scanf("%d", &n);
    for (int i = 1; i <= n; ++i) scanf("%d", a + i);
    deque<int> q;
    for (int i = 1; i <= n; ++i) {
        while (!q.empty() && a[q.back()] >= a[i]) q.pop_back();
        q.push_back(i);
        while (!q.empty() && q.front() <= i - n) q.pop_front();
        mx[i] = a[q.front()];
    }
    q.clear();
    for (int i = n; i >= 1; --i) {
        while (!q.empty() && a[q.back()] >= a[i]) q.pop_back();
        q.push_back(i);
        while (!q.empty() && q.front() <= i - n) q.pop_front();
        mx[i] = max(mx[i], a[q.front()]);
    }
    for (int i = 1; i <= n; ++i) printf("%d ", mx[i]);
    puts("");
    return 0;
}
