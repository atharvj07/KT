#include <cstdio>
#include <vector>

int main() {
    int n, L, a;
    scanf("%d%d%d", &n, &L, &a);
    std::vector<std::pair<int, int>> v(n);
    for (int i = 0; i < n; ++i) { scanf("%d%d", &v[i].first, &v[i].second); }
    int ans = 0, now = 0;
    for (int i = 0; i < n; ++i) {
        if (v[i].first < now) continue;
        if (now + v[i].second <= v[i].first + a) ++ans, now = v[i].first + v[i].second;
    }
    printf("%d", ans);
    return 0;
}
