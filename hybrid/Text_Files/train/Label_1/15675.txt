#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<int> lights(n + 1);
  for (int i = 1; i <= n; i++) lights[i] = i;

  for (int i = 0; i < m; i++) {
    int b;
    cin >> b;
    for (int j = b; j <= n; j++) {
      if (lights[j] > 0) {
        lights[j] = b;
        break;
      }
    }
  }

  for (int i = 1; i <= n; i++) cout << lights[i] << " ";
  cout << endl;

  return 0;
}
