#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
const long long INF = 2e9;
typedef pair<long long, int> PII;
vector<PII> t, w;
int main() {
    ios::sync_with_stdio(0);
    int n;
    cin >> n;
    for (int i = 0; i < n; ++i) {
        int a, b;
        cin >> a >> b;
        if (a == 1)
            t.push_back({b, 1});
        else
            w.push_back({b, 0});
    }
    sort(t.begin(), t.end());
    sort(w.begin(), w.end());
    int ans = 0;
    reverse(w.begin(), w.end());
    long long sum = 0;
    for (int i = 0; i < t.size(); ++i) {
        sum += t[i].first;
        while (!w.empty() && w.back().first <= sum && ans + w.back().second < n) {
            sum -= w.back().first;
            ans += w.back().second;
            w.pop_back();
        }
    }
    cout << sum << "\n";
}
