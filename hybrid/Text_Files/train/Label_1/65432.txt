n, m = map(int, input().split(" "))
w = [0]
c = [0]
for i in range(n):
    wi, ci = map(int, input().split(" "))
    w.append(wi)
    c.append(ci)
dp = [[0] * (m + 1) for _ in range(n + 1)]
for i in range(1, n + 1):
    for j in range(1, m + 1):
        if w[i] <= j:
            dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - w[i]] + c[i])
        else:
            dp[i][j] = dp[i - 1][j]
print(dp[n][m])
