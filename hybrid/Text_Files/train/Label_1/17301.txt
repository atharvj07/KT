#include <bits/stdc++.h>
using namespace std;
using ll=long long;
int main() {
    ll n;
    cin >> n;
    map<ll, pair<ll, ll>> a;
    for (ll i = 0; i < n; i++) {
        ll x, y;
        cin >> x >> y;
        a[x].first = y;
        a[y].second = x;
    }
    bool ok = true;
    for (ll i = 1; i <= 2 * n; i++) {
        if (a[i].first > a[i].second) {
            ok = false;
            break;
        }
    }
    if (ok) {
        cout << 0 << "\n";
    } else {
        ll ans = 0;
        for (ll i = 1; i <= 2 * n; i++) {
            if (a[i].first > a[i].second) { ans++; }
        }
        ans /= 2;
        cout << ans << "\n";
    }
    return 0;
}
