#include <iostream>
#include <vector>

std::vector<bool> to_binary(long long n) {
  std::vector<bool> result;
  while (n > 0) {
    result.push_back(n % 2);
    n /= 2;
  }
  return result;
}

int main() {
  long long m, k;
  std::cin >> m >> k;
  long long lb = 1, ub = 1e18;
  while (lb < ub) {
    long long mid = (lb + ub) / 2;
    std::vector<bool> binary = to_binary(mid + 1);
    long long count = 0;
    for (bool bit : binary) {
      if (bit) count++;
    }
    if (count == k) lb = mid + 1;
    else ub = mid;
  }
  std::cout << lb - 1 << std::endl;
}
