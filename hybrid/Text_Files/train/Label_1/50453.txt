import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] distances = new int[n];
        for (int i = 0; i < n; i++) {
            distances[i] = scanner.nextInt();
        }
        System.out.println(isRoundTripPossible(distances) ? "yes" : "no");
    }

    private static boolean isRoundTripPossible(int[] distances) {
        int maxRight = distances[0];
        int maxLeft = distances[distances.length - 1];
        int currentRight = distances[0];
        int currentLeft = distances[distances.length - 1];
        for (int i = 1; i < distances.length - 1; i++) {
            if (currentRight < i) {
                return false;
            }
            maxRight = Math.max(maxRight, currentRight + distances[i]);
            currentRight = Math.max(currentRight - 1, i - distances[i]);
            if (currentLeft < distances.length - 1 - i) {
                return false;
            }
            maxLeft = Math.max(maxLeft, currentLeft + distances[distances.length - 1 - i]);
            currentLeft = Math.max(currentLeft - 1, distances.length - 1 - i - distances[distances.length - 1 - i]);
            if (currentRight >= maxLeft) {
                return true;
            }
        }
        return false;
    }
}
