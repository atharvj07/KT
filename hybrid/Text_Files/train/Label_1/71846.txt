import sys

for t in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    if any(a[i] > a[i-1] for i in range(1,n)):
        print(-1)
    elif all(i==0 for i in a):
        print(0)
    elif a[1] != a[0]:
        print(a[1],a[0])
    elif n == 2:
        print(-1)
    else:
        m = a[0]-a[2]
        if all(a[i]-a[i-1]==m for i in range(2,n)):
            print(m,a[1]%m)
        else:
            print(-1)
