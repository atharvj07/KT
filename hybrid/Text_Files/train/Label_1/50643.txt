import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        System.out.println(solve(n, h, w, x));
    }

    public static int solve(int n, int h, int w, int[] x) {
        int ans = h * w * n;
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) {
                ans -= x[i] * h;
            } else {
                ans += x[i] * h;
            }
        }
        return ans;
    }
}
