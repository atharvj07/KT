#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod=1e9+7;
int32_t main()
{
    int a,b;
    cin>>a>>b;
    if(a!=1)
    {
        int ans=(a+1)*a/2%mod;
        ans=(ans*b)%mod;
        if(a&1)
            ans=(ans+a/2)%mod;
        else
            ans=(ans+a/2-1)%mod;
        cout<<ans<<'\n';
    }
    else
        cout<<0<<'\n';
}
