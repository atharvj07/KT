#include<iostream>
#include<unordered_map>

using namespace std;

typedef long long ll;

int main(){
    unordered_map<ll,ll> mp;
    int n,x,a;
    cin >> n >> x;
    ll sum = 0;
    for(int i=0;i<n;++i){
        cin >> a;
        mp[a]++;
        sum+=a;
    }
    int ans = 0;
    if(sum==0)
        cout << 0;
    else if(sum > 0){
        if(mp.find(-1)!=mp.end()){
            ans = (sum+1)/2;
            cout << ans;
        }
        else if(mp[-sum]>0){
            mp[-sum]--;
            ans = (sum)/2+1;
            cout << ans;
        }
        else{
            ans = sum+1;
            cout << ans;
        }
    }
    else{
        if(mp.find(1)!=mp.end()){
            ans = (abs(sum)+1)/2;
            cout << ans;
        }
        else if(mp[abs(sum)]>0){
            mp[abs(sum)]--;
            ans = (abs(sum))/2+1;
            cout << ans;
        }
        else{
            ans = abs(sum)+1;
            cout << ans;
        }
    }
}
