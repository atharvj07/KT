MOD = 1000000007

def power(base, exponent):
    result = 1
    while exponent > 0:
        if exponent % 2 == 1:
            result = (result * base) % MOD
        base = (base * base) % MOD
        exponent = exponent // 2
    return result

def solve(n, m, p, positions):
    lp = len(p)
    for i in range(1, m):
        if positions[i] < positions[i - 1] + lp:
            if p[:positions[i] - positions[i - 1]] != p[positions[i - 1] + lp - positions[i]:]:
                return 0
    free_chars = n - lp * m
    for i in range(1, m):
        free_chars += max(0, positions[i] - positions[i - 1] - lp)
    return power(26, free_chars)

n, m = map(int, input().split())
p = input()
positions = list(map(int, input().split()))
print(solve(n, m, p, positions))
