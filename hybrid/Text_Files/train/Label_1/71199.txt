import sys

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
a.sort()
ans = []
cnt = 0
i = 0
j = n-1
while i <= j:
    if a[j] == a[i]:
        ans.append(a[i])
        cnt += 1
        i += 1
        j -= 1
    elif a[j] < a[i]:
        i += 1
    else:
        j -= 1
print(cnt)
for x in ans:
    print(x, end=' ')
for x in a:
    if cnt == 0:
        break
    if x in ans:
        cnt -= 1
        continue
    print(x, end=' ')
