#include <iostream>
#include <algorithm>
#include <vector>
#include <stack>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> h(n), b(n);
    for (int i = 0; i < n; i++) cin >> h[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    vector<int> l(n), r(n);
    stack<int> s;
    for (int i = 0; i < n; i++) {
        while (!s.empty() && h[s.top()] > h[i]) s.pop();
        l[i] = s.empty() ? -1 : s.top();
        s.push(i);
    }
    s = stack<int>();
    for (int i = n - 1; i >= 0; i--) {
        while (!s.empty() && h[s.top()] > h[i]) s.pop();
        r[i] = s.empty() ? n : s.top();
        s.push(i);
    }
    long long ans = 0;
    for (int i = 0; i < n; i++) ans += h[i] * (r[i] - l[i] - 1) * b[i];
    cout << ans << '\n';
    return 0;
}
