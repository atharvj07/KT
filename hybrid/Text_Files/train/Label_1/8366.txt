#include <iostream>
#include <vector>
using namespace std;

int n, a, b, k;
vector<int> h;

int get_points() {
  int p = 0;
  bool skipped = false;
  for (int i = 0; i < n; i++) {
    if (h[i] <= a) {
      p++;
      continue;
    }
    if (!skipped && h[i] - a >= b && k) {
      k--;
      h[i] -= a;
      skipped = true;
    } else if (h[i] - b >= a) {
      h[i] -= b;
      skipped = false;
    } else {
      return p;
    }
  }
  return p;
}

int main() {
  cin >> n >> a >> b >> k;
  h.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> h[i];
  }
  cout << get_points() << endl;
  return 0;
}
