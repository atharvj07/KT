from collections import defaultdict

n = int(input())
g = defaultdict(list)
for i in range(2, n + 1):
    g[int(input())].append(i)
col = [-1] * (n + 1)
ans = [0] * (n + 1)

def dfs(v, p):
    sub = 1
    col[v] = 0
    for u in g[v]:
        if u != p:
            sub += dfs(u, v)
            if col[u] == col[v]:
                col[v] += 1
    ans[sub] = max(ans[sub], col[v] + 1)
    return sub

dfs(1, 1)
print(*ans[1:])
