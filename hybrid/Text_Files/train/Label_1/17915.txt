#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m, q, cnt = 0;
  cin >> n >> m;
  string s[200005], ss, t, x, y;
  for (int i = 0; i < n; i++) {
    cin >> s[i];
  }
  for (int i = 0; i < m; i++) {
    cnt = 0;
    cin >> ss;
    for (int j = 0; j < n; j++) {
      x = ss, y = s[j];
      q = 0;
      for (int k = 0; k < ss.size(); k++) {
        if (x[k] != y[k]) q++;
      }
      if (q == 1) {
        cnt = 1;
        break;
      }
    }
    if (cnt == 1)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
