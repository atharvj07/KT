import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt(), x = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            int ans = 0;
            boolean[] vis = new boolean[n + 1];
            Arrays.sort(arr);
            for (int i = n - 1; i >= 0; i--) {
                if (x == arr[i] && !vis[i]) {
                    vis[i] = true;
                    ans++;
                    for (int j = i - 1; j >= 0; j--) {
                        if (vis[j] || arr[j] != x) break;
                        vis[j] = true;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
