def factorial(x):
    if x == 0:
        return 1
    return x * factorial(x - 1)

def ways_to_place_pennants(n):
    bug_pennants = 5
    feature_pennants = 3
    return (factorial(n + bug_pennants - 1) // (factorial(bug_pennants) * factorial(n - 1))) * \
           (factorial(n + feature_pennants - 1) // (factorial(feature_pennants) * factorial(n - 1)))

n = int(input().strip())
print(ways_to_place_pennants(n))
