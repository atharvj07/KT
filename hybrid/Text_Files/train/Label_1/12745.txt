#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;

int l[N], r[N], pos[N];

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> l[i] >> r[i];

  stack<int> s;
  for (int i = n - 1; i >= 0; i--) {
    while (!s.empty() && r[s.top()] >= l[i]) s.pop();
    if (!s.empty() && r[s.top()] > r[i]) {
      cout << "NO" << endl;
      return 0;
    }
    s.push(i);
  }
  cout << "YES" << endl;
}
