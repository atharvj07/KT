#include <bits/stdc++.h>
#define MAXN 5000
#define MAXM 100
int n,m,color[MAXN+1],visited[MAXN+1],chk[MAXM+1],Lcolor[MAXN+1],Rcolor[MAXN+1];
int mx;
struct par{
	int x,y;
};
std::vector<int>adjacent[MAXM+1];
void dfs(int x,int rem){
	visited[x]=1;
	int y,tmp;
	tmp=color[x];
	for(int i=0;i<adjacent[tmp].size();i++){
		y=adjacent[tmp][i];
		if(!visited[y]){
			if(color[y]==tmp)rem--;
			if(color[y]!=tmp || (color[y]==tmp && rem))chk[tmp]=1,dfs(y,rem);
		}
	}
}
void dfs2(int x,int l,int r,int rem){
	visited[x]=1;
	int y;
	Lcolor[x]=l,Rcolor[x]=r;
	for(int i=0;i<adjacent[color[x]].size();i++){
		y=adjacent[color[x]][i];
		if(!visited[y]){
			if(color[y]==color[x])rem--;
			if(chk[color[x]] && rem){
				dfs2(y,r,l,rem);
			}else dfs2(y,l,r,rem);
		}
	}
}
int main(){
	std::cin>>n>>m;
	for(int i=1;i<=n;i++)std::cin>>color[i];
	for(int i=1;i<=n;i++){
		int x=color[i];
		for(int j=i+1;j<=n;j++){
			if(color[j]==x){
				adjacent[x].push_back(j);
				adjacent[j].push_back(i);
			}
		}
	}
	int tmp=0;
	for(int i=1;i<=m;i++){
		std::memset(visited,0,sizeof visited);
		dfs(i,n-1);
	}
	for(int i=1;i<=m;i++){
		if(chk[i])tmp++;
	}
	mx=tmp;
	std::cout<<mx<<'\n';
	for(int i=1;i<=m;i++){
		if(chk[i]){
			std::memset(visited,0,sizeof visited);
			dfs2(i,1,1,n-2);
		}
	}
	for(int i=1;i<=n;i++)std::cout<<Lcolor[i]<<' '<<Rcolor[i]<<'\n';
	return 0;
}
