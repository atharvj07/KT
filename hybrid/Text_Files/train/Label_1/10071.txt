#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    int ans = 0;
    while (n > 1) {
      ans++;
      if (!(n & 1)) {
        n /= 2;
      } else {
        bool found = false;
        for (long long i = 3; i * i <= n; i += 2) {
          if (!(n % i)) {
            n /= i;
            found = true;
            break;
          }
        }
        if (!found) n--;
      }
    }
    cout << ans << "\n";
  }
  return 0;
}
