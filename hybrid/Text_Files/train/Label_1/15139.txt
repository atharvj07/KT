#include <bits/stdc++.h>

using namespace std;

int main() {
    uint64_t a;
    cin >> a;
    int sum = 0;
    if (a > 99) {
        if (a >= 100) sum++;
        a /= 10;
        if ((a % 10 + a / 10 % 10) % 9 == 0) sum++;
    } else if (a >= 10) {
        sum++;
    }
    cout << sum + 1 << endl;
}
