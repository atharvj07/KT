st = input().split()
n, m, k = int(st[0]), int(st[1]), int(st[2])
a = [[int(j) for j in input().split()] for i in range(n)]
d = {}

for i in range(n):
    d[i] = 0
for i in range(k):
    st = input().split()
    x, y = int(st[0]) - 1, int(st[1]) - 1
    for j in range(n):
        if j == x:
            continue
        if a[j][y] == 1:
            d[j] += 1

for i in d.values():
    print(i, end=' ')
