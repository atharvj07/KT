#include <bits/stdc++.h>
using namespace std;
#define int long long

vector<int> v(100001);
void solve() {
    int n, q;
    cin >> n >> q;
    for (int i = 1; i <= n; i++) cin >> v[i];
    while (q--) {
        int x, y, cnt = 0, s;
        cin >> s;
        for (int i = 1; i <= n; i++) { y += v[i]; }
        if (s == y) {
            cout << "Yes" << endl;
            continue;
        }
        if (s > y) {
            cout << "No" << endl;
            continue;
        }
        x = (y - s) / 2;
        for (int i = 1; i <= n; i++) {
            if (s >= v[i]) {
                s -= v[i];
            } else {
                cnt++;
            }
        }
        if (x <= (n - cnt)) cout << "Yes" << endl;
        else
            cout << "No" << endl;
    }
}
int32_t main() {
    int T;
    cin >> T;
    while (T--) { solve(); }
}
