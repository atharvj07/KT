import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        while (true) {
            int an = input.nextInt();
            int bn = input.nextInt();
            int r = input.nextInt();
            if (an == 0 && bn == 0 && r == 0) break;
            int[][] ap = new int[an][2];
            int[][] bp = new int[bn][2];
            for (int i = 0; i < an; i++) {
                ap[i][0] = input.nextInt();
                ap[i][1] = input.nextInt();
            }
            for (int i = 0; i < bn; i++) {
                bp[i][0] = input.nextInt();
                bp[i][1] = input.nextInt();
            }
            int sum = 0;
            for (int[] a : ap) {
                int cnt = 0;
                for (int[] b : bp) {
                    if (Math.sqrt(Math.pow(a[0] - b[0], 2) + Math.pow(a[1] - b[1], 2)) <= 2 * r) cnt++;
                }
                sum += cnt;
            }
            System.out.println(sum);
        }
        input.close();
    }
}
