#include<bits/stdc++.h>

using namespace std;

int main() {
    int t; cin >> t;
    while(t--) {
        long long a, b; cin >> a >> b;
        long long x = 0, ans = 0;
        for(int i = 1; x <= b; i++) {
            x = i;
            for(int j = 1; x*j + j <= b; j++) {
                if(x*j + i + j + x == i*10 + j) ans++;
                x = i*10 + j;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
