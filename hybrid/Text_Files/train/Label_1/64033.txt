import sys

n, p, k = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
a.sort()
ans = 0
for i in range(n):
    l, r = i + 1, n - 1
    while l <= r:
        m = (l + r) // 2
        if (a[i] + a[m]) * (a[i] ** 2 + a[m] ** 2) % p == k:
            ans += m - i
            l = m + 1
        else:
            r = m - 1
print(ans)
