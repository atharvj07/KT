#include<iostream>
#include<vector>
#include<queue>
using namespace std;
#define ll long long
const int N = 2e5 + 5;
ll n, m, c[N], cost = 0;
vector<ll>adj[N];
priority_queue<ll, vector<ll>, greater<ll>>pq;

int main() {
    cin >> n >> m;
    for (ll i = 1; i <= n; i++) {
        cin >> c[i];
        pq.push(c[i]);
    }
    for (ll i = 1; i <= m; i++) {
        ll x, y; cin >> x >> y;
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    while (!pq.empty()) {
        ll v = pq.top(); pq.pop();
        if (cost > v)break;
        cost += v;
        for (ll u : adj[v]) {
            pq.push(c[u]);
            c[u] = 0;
        }
    }
    cout << cost << endl;
    return 0;
}
