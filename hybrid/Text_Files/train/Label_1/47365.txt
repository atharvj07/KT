import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = sc.nextInt();
        }
        int[] ans = new int[m];
        for (int i = 0; i < m; i++) {
            int cur = b[i];
            int cnt = 0;
            for (int j = 0; j < n; j++) {
                if (cur < 0) {
                    cnt++;
                }
                cur += a[j];
            }
            ans[i] = cnt;
        }
        for (int i = 0; i < m; i++) {
            System.out.println(ans[i]);
        }
    }
}
