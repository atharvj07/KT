import java.util.*;

public class Main {
    static final int M = 2002;
    static final int W = 16;
    static int n, k;
    static int[] a = new int[W];
    static int[] f = new int[M];
    static List<Pair> trace = new ArrayList<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt(); k = sc.nextInt();
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        sc.close();
        for (int x = 1; x < M; x++) {
            if (x % k != 0) f[x] = x;
            else f[x] = f[x / k];
        }

        for (int i = 0; i < n; i++) {
            int x = a[i];
            if (f[x] == 1) continue;
            for (int j = i + 1; j < n; j++) {
                int y = a[j];
                if (f[x] * f[y] == 1) {
                    trace.add(new Pair(x, y));
                    a[i] = f[x + y];
                    a[j] = 0;
                    break;
                }
            }
        }

        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                System.out.println("YES");
                for (Pair p : trace) {
                    System.out.printf("%d %d\n", p.x, p.y);
                }
                return;
            }
        }
        System.out.println("NO");
    }
}

class Pair {
    int x, y;
    public Pair(int x, int y) {
        this.x = x;
        this.y = y;
    }
}
