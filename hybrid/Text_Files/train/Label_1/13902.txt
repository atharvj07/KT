#include <bits/stdc++.h>
using namespace std;
int test,n,a[200005];
int main(){
    cin >> test;
    while(test--){
        cin >> n;
        int sum=0;
        int x=-1;
        for(int i=1;i<=n;i++){
            cin >> a[i];
            if(a[i]==x){
                sum++;
            }
            else{
                if(x==-1) x=a[i];
            }
        }
        cout << n-sum-1 << endl;
    }
    return 0;
}
