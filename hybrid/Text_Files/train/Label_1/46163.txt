import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
        String[] line = input.readLine().split(" ");
        int n = Integer.parseInt(line[0]), m = Integer.parseInt(line[1]);
        List<List<Edge>> adjList = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adjList.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            line = input.readLine().split(" ");
            int a = Integer.parseInt(line[0]), b = Integer.parseInt(line[1]), w1 = Integer.parseInt(line[2]), w2 = Integer.parseInt(line[3]);
            adjList.get(a).add(new Edge(b, w1));
            adjList.get(b).add(new Edge(a, w2));
        }
        System.out.println(findMinCycle(adjList, n));
    }

    private static int findMinCycle(List<List<Edge>> adjList, int n) {
        int[] dist = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            dist[i] = Integer.MAX_VALUE;
        }
        dist[1] = 0;
        int minCycle = Integer.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            for (Edge edge : adjList.get(i)) {
                int to = edge.to, w = edge.weight;
                if (dist[to] > dist[i] + w) {
                    dist[to] = dist[i] + w;
                    if (i != 1 && i == to) {
                        minCycle = Math.min(minCycle, dist[i]);
                    }
                }
            }
        }
        return minCycle == Integer.MAX_VALUE ? 0 : minCycle;
    }

    private static class Edge {
        int to, weight;

        public Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }
}
