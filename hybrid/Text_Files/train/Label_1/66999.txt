t = int(input())
while t > 0:
    n, k = [int(x) for x in input().split(" ")]
    s = input()
    t -= 1
    if k > s.count("1", s.find("1")) + 1:
        print(-1)
    else:
        if s[0] != "1":
            print(s.count("0", 0, s.find("1")) + 1)
        else:
            print(s.count("0", s.find("1")))
