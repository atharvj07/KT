#include<bits/stdc++.h>
using namespace std;

int a,b;

int main(){
	cin>>a>>b;
	if (a>b || (b-a)%2){
		cout<<"NO\n";
		return 0;
	}
	cout<<"YES\n";
	vector<int> v;
	v.push_back(a);
	while(a<b){
		if (b%2){
			v.push_back(b);
			b--;
			a=b/2;
		}
		else{
			v.push_back(b);
			b/=2;
		}
	}
	cout<<v.size()<<endl;
	for(auto val:v)cout<<val<<" ";
	return 0;
}
