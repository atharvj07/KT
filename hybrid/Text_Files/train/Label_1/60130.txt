import sys
import math

n, p, q = map(int, sys.stdin.readline().split())
arr = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]
arr.sort(key=lambda x: x[1]/x[0], reverse=True)
ans = 0
while p>0 or q>0:
    if p>0 and q>0:
        if arr[0][0]/arr[0][1]>arr[1][0]/arr[1][1]:
            if p-arr[0][0]>0:
                t = min(q/arr[0][1], 1)
                p -= t*arr[0][0]
                q -= t*arr[0][1]
                ans += t
            else:
                t = p/arr[0][0]
                p -= t*arr[0][0]
                q -= t*arr[0][1]
                ans += t
        else:
            if q-arr[1][1]>0:
                t = min(p/arr[1][0], 1)
                p -= t*arr[1][0]
                q -= t*arr[1][1]
                ans += t
            else:
                t = q/arr[1][1]
                p -= t*arr[1][0]
                q -= t*arr[1][1]
                ans += t
    else:
        if p>0:
            if p-arr[0][0]>0:
                t = p/arr[0][0]
                p -= t*arr[0][0]
                q -= t*arr[0][1]
                ans += t
            else:
                t = 1
                p -= t*arr[0][0]
                q -= t*arr[0][1]
                ans += t
        else:
            if q-arr[1][1]>0:
                t = q/arr[1][1]
                p -= t*arr[1][0]
                q -= t*arr[1][1]
                ans += t
            else:
                t = 1
                p -= t*arr[1][0]
                q -= t*arr[1][1]
                ans += t
print(ans)
