def can_form_abacaba(n, s):
    def count_abacaba(s):
        return sum(s[i:i+7] == "abacaba" for i in range(len(s)-6))
    
    for i in range(n-6):
        if all(s[i+j] in ["?", "abacaba"[j]] for j in range(7)):
            temp = s[:i] + "abacaba" + s[i+7:]
            temp = temp.replace('?', 'z')
            if count_abacaba(temp) == 1:
                return "Yes\n" + temp
    return "No"

T = int(input().strip())
for _ in range(T):
    n = int(input().strip())
    s = input().strip()
    print(can_form_abacaba(n, s))
