#include<bits/stdc++.h>
using namespace std;
 
int main()
{
    int t; cin >> t;
    while(t--)
    {
        int n, m, sum = 0; cin >> n >> m;
        vector<int> v(n);
        for(int &x : v) {cin >> x; sum += x;}
        cout << sum / n + min(m - sum / n, m - v[0]) << '\n';
    }
    return 0;
}
