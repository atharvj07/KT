n = int(input())

max_digit = max([int(d) for d in str(n)])
quasi_binaries = []

for _ in range(max_digit):
    quasi = 0
    temp_n = n
    factor = 1
    while temp_n > 0:
        if temp_n % 10 > 0:
            quasi += factor
        temp_n //= 10
        factor *= 10
    quasi_binaries.append(quasi)
    n -= quasi

print(len(quasi_binaries))
print(' '.join(map(str, quasi_binaries)))
