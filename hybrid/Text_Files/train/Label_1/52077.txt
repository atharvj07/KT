import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        String l = sc.next();
        String r = sc.next();
        for (int i = 0; i < n; i++) {
            a[i] = l.charAt(i) - '0';
            b[i] = r.charAt(i) - '0';
        }
        int[] pref = new int[n];
        for (int i = 0; i < n; i++) {
            if (i == 0) {
                pref[i] = a[i];
            } else {
                pref[i] = pref[i - 1] ^ a[i];
            }
        }
        int ans = 0;
        for (int i = n - 1; i >= 0; i--) {
            int j = i;
            while (j >= 0 && pref[i] != pref[j]) {
                j--;
            }
            if (j >= 0) {
                ans = Math.max(ans, pref[i] ^ b[j]);
            }
        }
        StringBuilder sb = new StringBuilder();
        for (int i = n - 1; i >= 0; i--) {
            sb.append(ans & 1);
            ans >>= 1;
        }
        System.out.println(sb.toString());
    }
}
