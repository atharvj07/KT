#include <iostream>
#include <iomanip>
using namespace std;
int main() {
  int H, M, S, T1, T2;
  cin >> H >> M >> S >> T1 >> T2;
  double a = M / 5.0 + S / 300.0;
  if (T1 <= T2) {
    if (a <= T1 || a >= T2) cout << "NO";
    else cout << "YES";
  } else {
    if (a >= T2 && a <= T1) cout << "NO";
    else cout << "YES";
  }
}
