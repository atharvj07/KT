import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }

        long sum = 0;
        for (long ai : a) {
            sum += ai;
        }

        long half = sum / 2;
        long curr = 0;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            curr += a[i];
            if (curr >= half) {
                ans = i + 1;
                break;
            }
        }

        System.out.println(ans);
    }
}
