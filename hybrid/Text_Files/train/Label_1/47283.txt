import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int n = s.length();
        for (int k = 0; k < n; k++) {
            if (s.charAt(k) < s.charAt(n - k - 1)) System.out.println("Mike");
            else System.out.println("Ann");
        }
    }
}
