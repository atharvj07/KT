import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] b = new int[n];
        for (int i = 0; i < n; i++) b[i] = sc.nextInt();
        int ans = b[0];
        int cur = 0;
        for (int i = 1; i < n; i++) {
            if (cur < b[i] - b[i - 1]) {
                cur = b[i] - b[i - 1];
                ans += cur;
            }
        }
        System.out.println(ans);
    }
}
