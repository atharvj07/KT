#include <bits/stdc++.h>
using namespace std;
#define pb push_back
const int MAXN = 5e1 + 10;

int n, a[MAXN];

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        int sum = 0;
        bool f1 = false, f2 = false;
        for (int i = 1; i <= n; i++) {
            char ch;
            cin >> ch;
            if (ch == 'a')
                a[i] = 1;
            else
                a[i] = -1;
            sum += a[i];
            if (sum == 0)
                f1 = true;
            if (sum == -n)
                f2 = true;
        }
        if (f1)
            cout << 1 << ' ' << n << '\n';
        else if (f2 && sum > 0)
            cout << 1 << ' ' << n - sum << '\n';
        else if (sum == 0)
            cout << 1 << ' ' << n << '\n';
        else
            cout << -1 << ' ' << -1 << '\n';
    }
}
