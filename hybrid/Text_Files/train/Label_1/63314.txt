from queue import Queue

x1, y1 = map(int, input().split())
x2, y2 = map(int, input().split())
n = int(input())
s = input()
dx = [1, -1, 0, 0]
dy = [0, 0, -1, 1]
dist = [[[(x2 - x1) * (y2 - y1) + 1 for k in range(4)] for j in range(n + 1)] for i in range(n + 1)]
dist[n][n][:] = [0 for k in range(4)]
q = Queue()
q.put((n, n, (x2 - x1) * (y2 - y1) + 1))
while not q.empty():
    cst, ans, mn = q.get()
    if ans == 0:
        print(cst)
        exit(0)
    for i in range(4):
        cx = x1 + dx[i] * (ans % n)
        cy = y1 + dy[i] * (ans % n)
        if 0 <= cx <= x2 and 0 <= cy <= y2 and dist[ans % n][ans][i] > dist[(ans + 1) % n][ans + 1][i] + (x2 - cx) * (y2 - cy):
            dist[ans % n][ans][i] = dist[(ans + 1) % n][ans + 1][i] + (x2 - cx) * (y2 - cy)
            q.put(((ans + 1) % n, ans + 1, dist[ans % n][ans][i]))
print(-1)
