#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
typedef long long ll;
int n, m, ta, tb, k;
ll a[N], b[N];
ll dp[N][20]; // dp[i][j]: max arrival time at C after taking i-th flight from A and canceling j flights
int main() {
    ios_base::sync_with_stdio(0);
    cin >> n >> m >> ta >> tb >> k;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= m; i++) { cin >> b[i]; }
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= k; j++) { dp[i][j] = -1; }
    }
    dp[0][0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= k; j++) {
            if (dp[i - 1][j] >= 0) { dp[i][j] = max(dp[i][j], dp[i - 1][j]); }
            int idx = lower_bound(b + 1, b + m + 1, a[i] + ta) - b;
            if (idx <= m) {
                if (j > 0) { dp[i][j] = max(dp[i][j], dp[i - 1][j - 1]); }
                if (dp[i - 1][j] >= 0) { dp[i][j] = max(dp[i][j], b[idx] + tb); }
            }
        }
    }
    ll ans = -1;
    for (int i = 0; i <= k; i++) { ans = max(ans, dp[n][i]); }
    cout << ans << endl;
    return 0;
}
