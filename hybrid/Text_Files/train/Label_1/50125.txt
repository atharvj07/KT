import java.util.Scanner;

public class CaracalAndMonster {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long monsterHealth = scanner.nextLong();

        long attacks = 0;
        while (monsterHealth > 0) {
            attacks++;
            if (monsterHealth == 1) {
                monsterHealth--;
            } else {
                monsterHealth = (monsterHealth - 1) / 2;
            }
        }

        System.out.println(attacks);
    }
}
