import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (true) {
            int n = in.nextInt();
            if (n == 0) {
                break;
            }
            int cnt = 0;
            for (int i = 1; i <= n; i++) {
                int sum = i;
                for (int j = i + 1; j <= n; j++) {
                    sum += j;
                    if (sum == n) {
                        cnt++;
                    } else if (sum > n) {
                        break;
                    }
                }
            }
            System.out.println(cnt);
        }
        in.close();
    }
}
