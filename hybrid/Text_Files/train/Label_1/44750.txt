import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long c = scanner.nextLong();
        long hr = scanner.nextLong();
        long hb = scanner.nextLong();
        long wr = scanner.nextLong();
        long wb = scanner.nextLong();
        long maxr = c / wr;
        long maxb = c / wb;
        long max = 0;
        for (long i = 0; i <= maxr; i++) {
            long j = (c - i * wr) / wb;
            if (j <= maxb) {
                max = Math.max(max, i * hr + j * hb);
            }
        }
        System.out.println(max);
    }
}
