import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<Integer> weights = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            weights.add(sc.nextInt());
        }
        sc.close();
        boolean isPossible = false;
        int sum1 = 0, sum2 = 0;
        for (int i = 0; i < weights.size(); i++) {
            if (sum1 == sum2) {
                sum1 += weights.get(i);
                isPossible = true;
            } else if (sum1 < sum2) {
                sum1 += weights.get(i);
                isPossible = true;
            } else {
                sum2 += weights.get(i);
                isPossible = true;
            }
        }
        System.out.println(isPossible ? "YES" : "NO");
    }
}
