#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int n, m, i, x[55], y[55];
    ll ans = 1e18;
    cin >> n >> m;
    for (i = 0; i < n; i++) cin >> x[i];
    for (i = 0; i < m; i++) cin >> y[i];
    sort(x, x + n);
    sort(y, y + m, greater<int>());
    for (i = n - 1; i >= 0; --i) ans = min(ans, 1LL * x[i] * y[i]);
    cout << ans << endl;
}
