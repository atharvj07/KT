#include<bits/stdc++.h>
using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    for (int i = 0; i + 1 < n; i++) {
        for (int j = i + 1; j + 1 < n; j++) {
            long long a = stol(s.substr(0, i + 1));
            long long b = stol(s.substr(i + 1, j - i));
            long long c = stol(s.substr(j + 1, n - j - 1));
            if (a + b == c) {
                cout << s.substr(0, i + 1) << "+" << s.substr(i + 1, j - i) << "=" << s.substr(j + 1, n - j - 1) << endl;
                return 0;
            }
        }
    }
    return 0;
}
