#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const ll MM = 1e9;
ll n, x, y;
vector<ll> adj[200005];

ll dfs(ll u, ll p) {
    ll sum = 0;
    for (ll v : adj[u]) {
        if (v == p) continue;
        sum += dfs(v, u) + 1;
    }
    return min(sum, n - 1 - sum);
}

int main() {
    cin >> n >> x >> y;
    for (int i = 1; i < n; i++) {
        ll u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    ll ans = LLONG_MAX;
    for (ll i = 1; i <= n; i++) {
        ll tot = dfs(i, 0);
        ans = min(ans, tot * x + (n - 1 - tot) * y);
    }
    cout << ans << endl;
}
