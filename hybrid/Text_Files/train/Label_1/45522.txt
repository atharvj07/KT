import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int xv = sc.nextInt(), yv = sc.nextInt();
        int xp = sc.nextInt(), yp = sc.nextInt();
        int xw1 = sc.nextInt(), yw1 = sc.nextInt(), xw2 = sc.nextInt(), yw2 = sc.nextInt();
        int xm1 = sc.nextInt(), ym1 = sc.nextInt(), xm2 = sc.nextInt(), ym2 = sc.nextInt();
        sc.close();
        System.out.println(canSee(xv, yv, xp, yp, xw1, yw1, xw2, yw2, xm1, ym1, xm2, ym2) ? "YES" : "NO");
    }

    static double getDistance(int x1, int y1, int x2, int y2) {
        return Math.sqrt((x2 - x1) * (x2 - x1) + (y2 - y1) * (y2 - y1));
    }

    static boolean canSee(int xv, int yv, int xp, int yp, int xw1, int yw1, int xw2, int yw2, int xm1, int ym1, int xm2,
            int ym2) {
        if (isSegmentIntersect(xv, yv, xp, yp, xw1, yw1, xw2, yw2)) {
            return false;
        }
        if (isSegmentIntersect(xv, yv, xp, yp, xm1, ym1, xm2, ym2)) {
            return false;
        }
        double dist = getDistance(xv, yv, xm1, ym1);
        double dist1 = getDistance(xp, yp, xm1, ym1);
        double dist2 = getDistance(xp, yp, xm2, ym2);
        if (dist1 > dist && dist2 > dist) {
            return false;
        }
        if (dist1 == dist && dist2 == dist) {
            return true;
        }
        double dist3 = getDistance(xv, yv, xp, yp);
        if (dist1 == dist && dist3 < dist) {
            return false;
        }
        if (dist2 == dist && dist3 < dist) {
            return false;
        }
        if (areCollinear(xv, yv, xm1, ym1, xp, yp)) {
            return true;
        }
        if (areCollinear(xv, yv, xm2, ym2, xp, yp)) {
            return true;
        }
        return false;
    }

    static boolean isSegmentIntersect(int x1, int y1, int x2, int y2, int x3, int y3, int x4, int y4) {
        int o1 = orientation(x1, y1, x2, y2, x3, y3);
        int o2 = orientation(x1, y1, x2, y2, x4, y4);
        int o3 = orientation(x3, y3, x4, y4, x1, y1);
        int o4 = orientation(x3, y3, x4, y4, x2, y2);
        if (o1 != o2 && o3 != o4) {
            return true;
        }
        if (o1 == 0 && onSegment(x1, y1, x3, y3, x2, y2)) {
            return true;
        }
        if (o2 == 0 && onSegment(x1, y1, x4, y4, x2, y2)) {
            return true;
        }
        if (o3 == 0 && onSegment(x3, y3, x1, y1, x4, y4)) {
            return true;
        }
        if (o4 == 0 && onSegment(x3, y3, x2, y2, x4, y4)) {
            return true;
        }
        return false;
    }

    static int orientation(int x1, int y1, int x2, int y2, int x3, int y3) {
        int val = (y2 - y1) * (x3 - x2) - (y3 - y2) * (x2 - x1);
        if (val == 0) {
            return 0;
        }
        return (val > 0) ? 1 : -1;
    }

    static boolean onSegment(int x1, int y1, int x2, int y2, int x3, int y3) {
        if (x2 <= Math.max(x1, x3) && x2 >= Math.min(x1, x3) && y2 <= Math.max(y1, y3) && y2 >= Math.min(y1, y3)) {
            return true;
        }
        return false;
    }

    static boolean areCollinear(int x1, int y1, int x2, int y2, int x3, int y3) {
        return (y2 - y1) * (x3 - x2) == (y3 - y2) * (x2 - x1);
    }
}
