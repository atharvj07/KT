import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long c = sc.nextLong();
    long[] x = new long[n];
    long[] v = new long[n];
    for (int i = 0; i < n; i++) {
      x[i] = sc.nextLong();
      v[i] = sc.nextLong();
    }
    Arrays.sort(x);
    long max = 0;
    for (int i = 0; i < n; i++) {
      long dist = (x[i] + c - x[0]) % c;
      max = Math.max(max, v[i] - dist);
    }
    System.out.println(max);
  }
}
