import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] deg = new int[n + 1];
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            deg[a]++;
            deg[b]++;
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            ans = Math.max(ans, deg[i]);
        }
        System.out.println(ans);
    }
}
