#include <iostream>
#include <vector>
using namespace std;

int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }

int main() {
  int n, k;
  cin >> n >> k;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  long long g = a[0];
  for (int i = 1; i < n; i++) { g = gcd(g, a[i]); }
  cout << g << endl;
  return 0;
}
