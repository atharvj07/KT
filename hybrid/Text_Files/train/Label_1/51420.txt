import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int t = scan.nextInt();
        while(t-- > 0) {
            long n = scan.nextLong();
            long k = scan.nextLong();
            long ans = (k-1) * n + 1;
            System.out.println(ans);
        }
    }
}
