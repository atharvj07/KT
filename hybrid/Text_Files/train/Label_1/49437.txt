import java.util.Scanner;

public class ArithmeticProgression {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();
        int m = scanner.nextInt();

        int[] seq = new int[n];
        for (int i = 0; i < n; i++) {
            seq[i] = a + i * d;
        }

        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt() - 1;
            int z = scanner.nextInt() - 1;

            switch (x) {
                case 0:
                    reverse(seq, y, z);
                    break;
                case 1:
                    increment(seq, y, z);
                    break;
                case 2:
                    halve(seq, y, z);
                    break;
            }
        }

        int k = scanner.nextInt() - 1;
        System.out.println(seq[k]);
    }

    private static void reverse(int[] seq, int y, int z) {
        for (int i = y, j = z; i < j; i++, j--) {
            int temp = seq[i];
            seq[i] = seq[j];
            seq[j] = temp;
        }
    }

    private static void increment(int[] seq, int y, int z) {
        for (int i = y; i <= z; i++) {
            seq[i]++;
        }
    }

    private static void halve(int[] seq, int y, int z) {
        for (int i = y; i <= z; i++) {
            seq[i] /= 2;
        }
    }
}
