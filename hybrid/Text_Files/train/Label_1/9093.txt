#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (auto &i : a) cin >> i;
  int ans = 0, l = 0, r = 0;
  for (auto &i : a) {
    if (i == 0) ++l;
    if (i == 1) ++r;
    if (l > 0 && r > 0) {
      ans = 1;
      break;
    }
  }
  if (ans == 0) {
    for (int i = 0; i < n; ++i) {
      if (a[i] == 0) ++l;
      if (a[i] == 1) ++r;
      if (l > 0 && r > 0) {
        ans = i + 1;
        break;
      }
    }
  }
  cout << ans << '\n';
}
