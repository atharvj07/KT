import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long a = sc.nextLong();
            long b = sc.nextLong();
            long k = sc.nextLong();
            if ((a / ((long) Math.pow(2, k))) == b) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
