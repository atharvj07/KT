N, K, A, B = map(int, input().split())
ava = list(map(int, input().split()))
ans = 0
while len(ava) > 1:
    ava = list(filter(lambda x: x % 2, ava))
    ans += (A if not ava else B * len(ava) * 2**N)
    ava = list(map(lambda x: x // 2, ava))
    N -= 1
print(ans if not ava else B * len(ava) * 2**N)
