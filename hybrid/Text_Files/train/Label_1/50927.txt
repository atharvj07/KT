import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int[] nk = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] a = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] min = new int[nk[1]];
        int temp = nk[1];
        for (int i = 0; i < nk[0]; i++) {
            if (temp == 1) {
                min[nk[1] - 1] = Math.min(min[nk[1] - 1], a[i]);
            } else {
                min[nk[1] - temp] = Math.min(min[nk[1] - temp], a[i]);
                temp--;
            }
        }
        System.out.println(Arrays.stream(min).max().getAsInt());
    }
}
