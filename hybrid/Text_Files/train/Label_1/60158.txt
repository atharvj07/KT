import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    x = int(input())
    if x < 10:
        print(x)
    else:
        ans = []
        while x > 9:
            ans.append(9)
            x -= 9
        ans.append(x)
        ans = ans[::-1]
        for i in ans:
            print(i, end='')
        print()
