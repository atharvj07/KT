#include <bits/stdc++.h>

using namespace std;

#define x first
#define y second

bool chk(vector<pair<pair<int, int>, int>> V, vector<pair<pair<int, int>, int>> W) {
    int n = (int)V.size();
    vector<pair<int, int>> vec(n);
    for (int i = 0; i < n; i++) { vec[i] = {V[i].x.x, W[i].x.x}; }
    sort(vec.begin(), vec.end());
    for (int i = 0; i < n; i++) {
        if (vec[i].x != V[i].x.x || vec[i].y != W[i].x.x) return 0;
    }
    return 1;
}

int main() {
    int n, m;
    cin >> n >> m;
    vector<pair<pair<int, int>, int>> V(m);
    for (int i = 0; i < m; i++) {
        int tt, a, b;
        cin >> tt;
        if (tt == 1) {
            cin >> a >> b >> V[i].y;
            V[i].x = {a, b};
        } else {
            cin >> a >> b >> V[i].x.y;
            V[i].x = {a, b};
        }
    }
    vector<pair<pair<int, int>, int>> W(m);
    for (int i = 0; i < m; i++) {
        int tt, a, b;
        cin >> tt;
        if (tt == 1) {
            cin >> a >> b >> W[i].x.y;
            W[i].x = {a, b};
        } else {
            cin >> a >> b >> W[i].y;
            W[i].x = {a, b};
        }
    }
    if (chk(V, W)) {
        cout << "YES\n";
        for (int i = 0; i < n; i++) { cout << i + 1 << " "; }
    } else {
        cout << "NO\n";
    }
}
