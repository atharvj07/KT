#include<iostream>
#include<vector>
#include<algorithm>

#define ll long long

using namespace std;

const ll INF = 2e18;

struct Edge{
    ll to;
    ll w;
    Edge(ll to_, ll w_): to(to_), w(w_) {}
};

int main() {
    ll n, c;
    cin >> n >> c;
    vector<ll> a(n - 1), b(n - 1);
    for(ll i = 0; i < n - 1; i++) {
        cin >> a[i];
    }
    for(ll i = 0; i < n - 1; i++) {
        cin >> b[i];
    }
    vector<vector<Edge>> G(n);
    for(ll i = 0; i < n - 1; i++) {
        G[i].push_back(Edge(i + 1, a[i]));
        G[i].push_back(Edge(i + 1, c + b[i]));
    }
    vector<ll> d(n, INF);
    d[0] = 0;
    for(ll i = 0; i < n; i++) {
        for(ll j = 0; j < n; j++) {
            for(Edge e: G[j]) {
                if(d[j] + e.w < d[e.to]) {
                    d[e.to] = d[j] + e.w;
                }
            }
        }
    }
    for(ll i = 0; i < n; i++) {
        cout << d[i] << " ";
    }
    cout << endl;
    return 0;
}
