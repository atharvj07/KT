#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> friends(n);
  for (auto &friend_ : friends) cin >> friend_.first >> friend_.second;
  sort(begin(friends), end(friends));
  int width = 0, height = 0;
  for (auto &friend_ : friends) {
    height = max(height, max(friend_.first, friend_.second));
    width += min(friend_.first, friend_.second);
  }
  cout << width * height << endl;
  return 0;
}
