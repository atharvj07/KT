import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
        }
        int count = 0;
        int free = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == -1) {
                if (free == 0) {
                    count++;
                } else {
                    free--;
                }
            } else {
                free += arr[i];
            }
        }
        System.out.println(count);
    }
}
