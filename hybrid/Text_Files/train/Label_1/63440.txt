import sys
input = sys.stdin.readline
n, m, k = map(int, input().split())
a = sorted(list(map(int, input().split())))
g = [[] for _ in range(n+1)]
for _ in range(m):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)

d = [-1] * (n+1)
vis = [False] * (n+1)
def dfs(u, p):
    vis[u] = True
    d[u] = d[p] + 1
    for v in g[u]:
        if not vis[v]:
            dfs(v, u)

dfs(1, 1)
idx = -1
mx = 0
for i in a:
    if d[i] > mx:
        idx, mx = i, d[i]

vis = [False] * (n+1)
dfs(idx, idx)
print(mx)
