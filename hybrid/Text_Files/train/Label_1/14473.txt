#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  int ans = 1, end = a[0];
  for (int i = 1; i < n; i++) {
    if (a[i] == end + 1)
      end = a[i], ans++;
  }
  cout << ans << endl;
  for (int i = 0; i < n; i++) {
    if (a[i] == end - ans + 1) {
      cout << i + 1 << ' ';
      end--;
    }
  }
}
