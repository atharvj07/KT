#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll MAXN = 105;

ll n;
ll dp[MAXN][2][MAXN];
ll sum[MAXN][MAXN];
ll arr[MAXN][MAXN];

ll f(ll idx, ll turn, ll lst) {
  if (idx == n + 1) return 0;
  if (dp[idx][turn][lst] != -1) return dp[idx][turn][lst];

  ll ans = 0;
  if (turn == 0) {
    if (idx == lst)
      ans = max(ans, f(idx + 1, turn ^ 1, lst) + sum[idx][arr[idx][1]]);
    else
      for (ll j = 1; j <= arr[idx][0]; j++)
        ans = max(ans, f(idx + 1, turn ^ 1, idx) + sum[idx][arr[idx][j]]);
  } else {
    if (idx == lst)
      ans = max(ans, f(idx + 1, turn ^ 1, lst) + sum[idx][arr[idx][arr[idx][0]]]);
    else
      for (ll j = arr[idx][0]; j >= 1; j--)
        ans = max(ans, f(idx + 1, turn ^ 1, idx) + sum[idx][arr[idx][j]]);
  }
  return dp[idx][turn][lst] = ans;
}

int main() {
  cin >> n;
  memset(dp, -1, sizeof(dp));
  for (ll i = 1; i <= n; i++) {
    cin >> arr[i][0];
    for (ll j = 1; j <= arr[i][0]; j++) {
      cin >> arr[i][j];
      sum[i][arr[i][j]] = sum[i - 1][arr[i][j]] + arr[i][j];
    }
  }

  ll ans1 = f(1, 0, n + 1);
  memset(dp, -1, sizeof(dp));
  ll ans2 = f(1, 1, n + 1);
  cout << ans1 << " " << ans2 << endl;

  return 0;
}
