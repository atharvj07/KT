#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main(int argc, const char * argv[]) {
    int n, k; ll a[100000], p[100000];
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    ll ans = -2e9;
    p[0] = a[0];
    for (int i = 1; i < n; i++) p[i] = min(p[i-1], a[i]);
    for (int i = 0; i < n - k + 1; i++) {
        ans = max(ans, p[i]);
    }
    cout << ans << endl;
    return 0;
}
