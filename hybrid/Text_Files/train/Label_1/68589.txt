n = int(input())
weights = list(map(int, input().split()))

weights.sort()

team = set()

for weight in weights:
    if weight - 1 > 0 and (weight - 1) not in team:
        team.add(weight - 1)
    elif weight not in team:
        team.add(weight)
    elif weight + 1 not in team:
        team.add(weight + 1)

print(len(team))
