import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int t = scanner.nextInt();

    for (int i = 0; i < t; i++) {
      int n = scanner.nextInt();
      int[] b = new int[n];
      for (int j = 0; j < n; j++) {
        b[j] = scanner.nextInt();
      }

      int[] a = new int[n];
      a[0] = b[0];
      int leftIndex = 1;
      int rightIndex = n - 1;

      for (int j = 1; j < n; j++) {
        if (j % 2 == 1) {
          a[leftIndex] = b[j];
          leftIndex++;
        } else {
          a[rightIndex] = b[j];
          rightIndex--;
        }
      }

      for (int j = 0; j < n; j++) {
        System.out.print(a[j] + " ");
      }

      System.out.println();
    }
  }
}
