import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        while(t-->0){
            long a=sc.nextLong(),b=sc.nextLong(),k=sc.nextLong();
            if(a==b)System.out.println("YES");
            else if(k==0)System.out.println("NO");
            else{
                if((a*b)<0||((a>0&&b<0)||(a<0&&b>0)))System.out.println("NO");
                else if(a==0||b==0)System.out.println("NO");
                else if((a*a==b)||(b*b==a))System.out.println("YES");
                else System.out.println("NO");
            }
        }
    }
}
