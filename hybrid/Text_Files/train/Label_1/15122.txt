#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 1000001;
vector<int> dp(MAXN);

int main() {
  int t;
  cin >> t;
  while (t--) {
    int x;
    cin >> x;
    if (dp[x] != 0) {
      cout << dp[x] << endl;
      continue;
    }
    if (x % 2 == 0) {
      x /= 2;
    } else {
      if (dp[x - 1] == 0) dp[x - 1] = dp[(x - 1) / 2] + 2;
      if (dp[x + 1] == 0) dp[x + 1] = dp[(x + 1) / 2] + 2;
      x++;
    }
    while (x > 1) {
      if (x % 2 == 0)
        x /= 2;
      else
        x++;
      dp[x]++;
    }
    cout << dp[x] << endl;
  }
  return 0;
}
