#include <bits/stdc++.h>
using namespace std;
int a[100001], ans[100001];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  int l = 1, r = n, d = 0, idx = 0, cnt = 0;
  while (l <= r) {
    while (a[l] < 0 && l < n) l++;
    if (l == n) break;
    while (a[r] > 0) r--;
    if (l > r) continue;
    if (abs(a[l]) == abs(a[r])) {
      d++;
      cnt += 2;
      ans[++idx] = cnt;
      l++, r--;
    } else
      r--;
  }
  if (cnt < n)
    cout << -1 << endl;
  else {
    cout << d << endl;
    for (int i = 1; i <= idx; i++) cout << ans[i] << " ";
    cout << endl;
  }
  return 0;
}
