
#include <vector>
#include <iostream>
using namespace std;
int main() {
  int q; cin >> q;
  while (q--) {
    int n; cin >> n;
    vector<int> a(n), ans(n + 1);
    for (int& b:a) { cin >> b; }
    for (int i = 1; i <= n; i++) {
      ans[i] = a[i - 1];
      for (int j = 0; j < i - 1; j++) {
        if (a[i - 1] % a[j] == 0) continue;
        if (a[j] % a[i - 1] == 0) continue;
        ans[i] = max(ans[i], ans[j] + a[i - 1]);
      }
    }
    cout << ans[n] << '\n';
  }
  return 0;
}
