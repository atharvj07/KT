import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    long n = input.nextLong();
    int k = input.nextInt();
    if (k == 2) {
      System.out.println(n == 1 ? 0 : -1);
    } else if (k == 3) {
      System.out.println(n % 2 == 0 ? n / 2 : (n + 1) / 2);
    } else {
      System.out.println(n % 3 == 0 ? n / 3 : (n + 2) / 3);
    }
  }
}
