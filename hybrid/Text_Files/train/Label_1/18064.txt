#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

struct Frac {
    long long num, den;
    bool operator<(const Frac& o) const {
        return num * o.den < o.num * den;
    }
    bool operator==(const Frac& o) const {
        return num * o.den == o.num * den;
    }
};

int main() {
    int n, x0, y0;
    cin >> n >> x0 >> y0;
    vector<Frac> P;
    for (int i = 0; i < n; i++) {
        int x, y;
        cin >> x >> y;
        if (x == x0) {
            x = x0+1;
        }
        P.push_back({y - y0, x - x0});
    }
    sort(P.begin(), P.end());
    P.push_back({0LL, 1LL});
    int ans = 0;
    Frac last = P[0];
    for (auto p : P) {
        if (p == last) continue;
        ans++;
        last = p;
    }
    cout << ans << endl;
    return 0;
}
