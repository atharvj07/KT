import java.util.*;

class Main {
    static class Car {
        int num, lane, time;

        Car(int num, int lane, int time) {
            this.num = num;
            this.lane = lane;
            this.time = time;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] lanes = new int[n + 1];
        Queue<Car> q = new LinkedList<>();
        for (int i = 0; i < m; i++) {
            int type = sc.nextInt();
            if (type == 0) {
                int lane = sc.nextInt();
                while (!q.isEmpty() && q.peek().lane == lane) {
                    System.out.println(q.poll().num);
                }
                lanes[lane]--;
            } else {
                int num = sc.nextInt();
                int lane = 1;
                for (int j = 2; j <= n; j++) {
                    if (lanes[j] < lanes[lane]) {
                        lane = j;
                    }
                }
                lanes[lane]++;
                q.offer(new Car(num, lane, i));
            }
        }
        sc.close();
    }
}
