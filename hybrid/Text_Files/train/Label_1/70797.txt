def min_hours_to_obtain_armor_and_weapon(n, m, q, synergies):
    armor_synergy = [0] * (n + 1)
    weapon_synergy = [0] * (m + 1)
  
    for a, b in synergies:
        armor_synergy[a] = max(armor_synergy[a], b)
        weapon_synergy[b] = max(weapon_synergy[b], a)
  
    max_armor_with_synergy = max_weapon_with_synergy = 0
  
    for i in range(1, n + 1):
        max_armor_with_synergy = max(max_armor_with_synergy, armor_synergy[i])
        armor_synergy[i] = max_armor_with_synergy
  
    for i in range(1, m + 1):
        max_weapon_with_synergy = max(max_weapon_with_synergy, weapon_synergy[i])
        weapon_synergy[i] = max_weapon_with_synergy
  
    hours = 0
    armor, weapon = 1, 1
  
    while armor < n or weapon < m:
        if armor_synergy[armor] > weapon or weapon_synergy[weapon] > armor:
            next_armor = min(n, max(armor_synergy[armor], weapon_synergy[weapon]))
            next_weapon = min(m, max(armor_synergy[armor], weapon_synergy[weapon]))
        else:
            next_armor = min(n, armor + 1)
            next_weapon = min(m, weapon + 1)
  
        armor, weapon = next_armor, next_weapon
        hours += 1
  
    return hours

n, m = map(int, input().split())
q = int(input())
synergies = [tuple(map(int, input().split())) for _ in range(q)]

print(min_hours_to_obtain_armor_and_weapon(n, m, q, synergies))
