s = input()
first_bracket = s.find('[')
last_bracket = s.rfind(']')
if first_bracket == -1 or last_bracket == -1 or first_bracket >= last_bracket:
    print(-1)
else:
    first_colon = s.find(':', first_bracket, last_bracket)
    last_colon = s.rfind(':', first_bracket, last_bracket)
    if first_colon == -1 or last_colon == -1 or first_colon >= last_colon:
        print(-1)
    else:
        vertical_lines = s.count('|', first_colon, last_colon)
        print(4 + vertical_lines)
