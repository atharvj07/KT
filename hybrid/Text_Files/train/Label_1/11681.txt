#include<bits/stdc++.h>
using namespace std;
int d[1001][1001];
main()
{
	int n,k;
	cin>>n>>k;
	if(k>n-1 || n*k%2) {cout<<-1<<endl; return 0;}
	int m=n*k;
	int i,j;
	for(i=1; i<=n; i++)
		for(j=1; j<=n; j++)
			if(i==j) continue;
			else if(d[i][j]<k)
			{
				d[i][j]++;
				d[j][i]++;
				cout<<i<<" "<<j<<endl;
			}
}
