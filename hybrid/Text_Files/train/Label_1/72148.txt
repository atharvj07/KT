n = int(input())
old_nums = list(map(int, input().split()))
old_to_new = {0: 0}
occupied = set()
for i, old_num in enumerate(old_nums):
    if old_num in occupied:
        old_to_new[old_num] = 0
    else:
        old_to_new[old_num] = i + 1
        occupied.add(old_num)
for num in old_nums:
    print(old_to_new[num], end=' ')
