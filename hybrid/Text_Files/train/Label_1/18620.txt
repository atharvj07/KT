#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=998244353;
void solve() {
    ll n,k;
    cin>>n>>k;
    if (k>n) {
        cout<<0<<endl;
        return;
    }
    ll ans=(1LL<<(n-1))%mod;
    ans=(ans*((k*(k+1))%mod))%mod;
    cout<<ans<<endl;
}
int main() {
    solve();
    return 0;
}
