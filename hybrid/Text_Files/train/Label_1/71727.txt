import math
n = int(input())
def dist(x, y):
    return math.hypot(x[0]-y[0], x[1]-y[1])
def edge_dist(i):
    return dist(lst[(i-1)%n], lst[i])
def extend(i, d):
    c = lst[i][:]
    e = lst[(i-1)%n][:]
    f = lst[(i+1)%n][:]
    ed = edge_dist(i)
    if (d / ed) * (d / ed) <= ((e[0]-c[0]) * (f[1]-e[1]) - (e[0]-f[0]) * (f[1]-c[1])) / (edge_dist((i-1)%n) * edge_dist((i+1)%n)):
        return c[0]+(d/ed) * (f[1]-e[1]), c[1]-(d/ed) * (f[0]-e[0])
    else:
        return c[0], c[1]
lst = []
for i in range(n):
    x, y = map(int, input().split())
    lst.append((x, y))
ans = 0
for i in range(n):
    l = 0
    r = edge_dist(i)
    while r - l > 1e-10:
        m = (l+r)/2
        tmp = []
        for j in range(n):
            tmp.append(extend(j, m))
        flag = True
        for j in range(n):
            for k in range(j+2, j+n):
                k%=n
                if (tmp[j][0]*tmp[k][1] - tmp[j][0]*tmp[(k+1)%n][1] - tmp[k][0]*tmp[j][1] + tmp[k][0]*tmp[(k+1)%n][1] + tmp[j][0]*tmp[(k+1)%n][1] - tmp[(k+1)%n][0]*tmp[j][1]) / (tmp[j][0]-tmp[(k+1)%n][0]) < tmp[k][1]-tmp[(k+1)%n][1]:
                    flag = False
                    break
            if not flag:
                break
        if not flag:
            r = m
        else:
            l = m
            ans = m
print(ans)
