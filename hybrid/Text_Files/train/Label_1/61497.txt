import sys
input = sys.stdin.readline

n = int(input())
adj = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    x, y = map(int, input().split())
    adj[x].append(y)
    adj[y].append(x)

vis = [False] * (n + 1)
par = [0] * (n + 1)
sz = [0] * (n + 1)
dep = [0] * (n + 1)
mn_lst = [0] * (n + 1)

def dfs1(u: int, p: int, d: int) -> int:
    vis[u] = True
    par[u] = p
    dep[u] = d
    sz[u] = 1

    for v in adj[u]:
        if vis[v] == False:
            sz[u] += dfs1(v, u, d + 1)

    return sz[u]

def dfs2(u: int, p: int) -> int:
    for v in adj[u]:
        if v != p:
            mn_lst[v] = min(mn_lst[u], sz[v])
            dfs2(v, u)

    return 0

dfs1(1, 0, 0)
for i in range(1, n + 1):
    mn_lst[i] = n

dfs2(1, 0)

mn_ans = 1

for i in range(1, n + 1):
    if mn_lst[i] > dep[par[i]]:
        mn_ans = max(mn_ans, dep[i] + 1)

print(mn_ans)
