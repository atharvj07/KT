from collections import defaultdict
n, k = map(int, input().split())
a = list(map(int, input().split()))
G = defaultdict(list)
for _ in range(n - 1):
    u, v = map(int, input().split())
    G[u].append(v)
    G[v].append(u)
dp = [[0] * 2 for _ in range(n + 1)]
def dfs(u, p):
    dp[u][1] = a[u - 1]
    for v in G[u]:
        if v != p:
            dfs(v, u)
            dp[u][0] += max(dp[v][0], dp[v][1])
            dp[u][1] += dp[v][0]
dfs(1, 1)
print(max(dp[1]))
