#include <bits/stdc++.h>
using namespace std;

const int LIM = 2e5 + 5e4 + 10;
const long long MOD = 1e9 + 7;

int n;
vector<int> g[LIM], r[LIM];
int a[LIM];
int ans;

void dfs(int u, int p, int v, int c) {
    ans = max(ans, c);
    for (auto w : g[u]) {
        if (w == p) continue;
        dfs(w, u, v, c + a[w]);
    }
}

int main() {
    cin >> n;
    for (int i = 2; i <= n; i++) {
        int p; cin >> p;
        g[i].push_back(p);
        r[p].push_back(i);
    }
    for (int i = 1; i <= n; i++) cin >> a[i];
    ans = 0;
    for (int i = 1; i <= n; i++) {
        if (r[i].empty()) {
            dfs(i, i, i, a[i]);
        }
    }
    cout << ans << endl;
    return 0;
}
