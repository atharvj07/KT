n, k, M, D = map(int, input().split())

max_candies = 0
for times in range(1, D + 1):
    low = 1
    high = M
    while low <= high:
        x = (low + high) // 2
        total_rounds = (times - 1) * k + 1
        if total_rounds * x <= n // times:
            max_candies = max(max_candies, x * times)
            low = x + 1
        else:
            high = x - 1

print(max_candies)
