#include<bits/stdc++.h>
using namespace std;
int main() {
    int q;
    cin>>q;
    while(q--) {
        int n;
        cin>>n;
        string s;
        cin>>s;
        int i=0;
        vector<string> ans;
        string last="";
        while(i<s.length()) {
            if(last=="" || last[last.length()-1]<=s[i]) {
                last+=s[i];
                i++;
            }
            else {
                ans.push_back(last);
                last=s[i];
                i++;
            }
        }
        if(last!="")ans.push_back(last);
        if(ans.size()<2)cout<<"NO\n";
        else {
            cout<<"YES\n";
            cout<<ans.size()<<"\n";
            for(auto x: ans)cout<<x<<" ";
            cout<<"\n";
        }
    }
}
