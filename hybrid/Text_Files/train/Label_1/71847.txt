def solve(arr):
    n = len(arr)
    diffs = [0] * (n - 1)
    for i in range(1, n):
        diffs[i-1] = arr[i] - arr[i-1]
    
    # Calculate gcd of differences
    gcd_diff = diffs[0]
    for diff in diffs[1:]:
        gcd_diff = gcd(gcd_diff, diff)
    
    if gcd_diff == 0:
        return 0, 0
    
    # Check for m
    m = max(abs(gcd_diff), max(arr) + 1)
    c = abs(gcd_diff)
    return m, c

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    if n == 1:
        print("0")
    else:
        m, c = solve(a)
        if m == 0:
            print("0")
        elif c >= m or c == 0:
            print("-1")
        else:
            print(m, c)
