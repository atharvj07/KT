n, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
ans = []
for i in range(k):
    if n > a[i]:
        if (n - a[i]) % a[i] == 0:
            ans.append((n - a[i]) // a[i])
        else:
            ans.append((n - a[i]) // a[i] + 1)
        cur = (n - a[i]) // a[i]
        n -= cur * a[i]
        k -= cur
    else:
        if n % a[i] == 0:
            ans.append(n // a[i])
        else:
            ans.append(n // a[i] + 1)
        break
if n == 0:
    print(sum(ans))
else:
    print(-1)
