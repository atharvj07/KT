n,m=map(int,input().split())
vis=[]
for i in range(m):
    x,y=map(int,input().split())
    vis.append((x-1,y-1))
ans=['0']*n
for l,r in vis:
    cnt=0
    for i in range(l,r+1):
        cnt+=(ans[i]=='0')
    if cnt*cnt > (r-l+1-cnt)*(r-l+1-cnt):
        for i in range(l,r+1):
            ans[i]='1'
print(''.join(ans))
