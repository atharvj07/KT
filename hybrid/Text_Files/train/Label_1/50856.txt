import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] dp = new long[n + 1];
        dp[1] = 1;
        for (int i = 1; i <= n; i++) {
            dp[i] = (dp[i] + dp[i - 1] * (i + 1)) % 1000000007;
        }
        System.out.println(dp[n]);
    }
}
