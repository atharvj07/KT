#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
typedef pair<int, int> pii;

int main() {
  int n;
  cin >> n;
  vector<pii> v;
  for (int i = 0; i < n; ++i) {
    int k;
    cin >> k;
    int sum = 0;
    for (int j = 0; j < k; ++j) {
      int a;
      cin >> a;
      sum += a;
    }
    v.push_back({sum, k});
  }
  sort(v.begin(), v.end());
  cout << v[0].first * 5 + v[0].second * 15 << endl;
}
