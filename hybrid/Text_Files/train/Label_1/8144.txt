#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
unordered_map<ll, ll> mp;
int main() {
    ll n, m, k;
    cin >> n >> m >> k;
    for (ll a, i = 0; i < n; ++i) {
        cin >> a, mp[a] = i + 1;
    }
    ll cnt = 0, l = 1, ans = n;
    for (ll r = 1; r <= 1e6; ++r) {
        if (mp.find(r) != mp.end()) cnt += (l == mp[r]);
        while (cnt >= k) cnt -= (l == mp[l]), l++;
        if (r - l + 1 == m) ans = min(ans, n - cnt);
    }
    cout << ans << '\n';
    return 0;
}
