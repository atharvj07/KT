#include <cstdio>
#include <algorithm>
#include <map>
using namespace std;
const int maxn = 2e5 + 5;
typedef long long ll;
inline ll qread() {
    ll ans = 0;
    char ch = getchar();
    while (!isdigit(ch)) ch = getchar();
    while (isdigit(ch)) {
        ans = ans * 10 + ch - '0';
        ch = getchar();
    }
    return ans;
}
const ll mod = 998244353;
int n, ans, a[maxn], b[maxn], c[maxn];
int main() {
    n = qread();
    for (int i = 1; i <= 2 * n; i++)
        a[i] = qread(), b[a[i]] = i;
    for (int i = 1; i <= 2 * n; i += 2) {
        c[i / 2 + 1] = abs(b[a[i]] - b[a[i + 1]]);
        ans += c[i / 2 + 1] % 2;
    }
    printf("%d\n", ans);
    return 0;
}
