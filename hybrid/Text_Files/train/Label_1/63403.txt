from collections import Counter, deque
n = int(input())
c = Counter()
for x in map(int,input().split()):
    c[x]+=1
q = deque()
for x,y in c.items():
    if x==1:
        continue
    if y>x:
        print('NO')
        break
    else:
        q.append((x,y))
while q:
    x,y = q.popleft()
    if x==2:
        if y%2==1:
            print('NO')
            break
        y//=2
        if c[y]==0:
            c[1]+=y
            q.append((1,y))
    else:
        y-=1
        if y==0:
            print('NO')
            break
        if c[x-1]==0:
            c[x-1]+=y
            q.append((x-1,y))
else:
    print('YES')
