import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    arr = [list(map(int, input().split())) for _ in range(n)]
    arr.sort()
    a = [i[0] for i in arr]
    b = [i[1] for i in arr]
    ca = [0] * (n + 1)
    cb = [0] * (n + 1)
    for x in a:
        ca[x] += 1
    for x in b:
        cb[x] += 1
    ans = (sum(ca) * (sum(ca) - 1))
    ans += (sum(cb) * (sum(cb) - 1))
    for x in a:
        ans -= (ca[x] * (ca[x] - 1))
    for x in b:
        ans -= (cb[x] * (cb[x] - 1))
    print(ans)
