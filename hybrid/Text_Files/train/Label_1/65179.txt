import sys

n = int(sys.stdin.readline())
p = [int(x) for x in sys.stdin.readline().split()]
s = [int(x) for x in sys.stdin.readline().split()]
pos = [0 for x in range(n+1)]
for i in range(n):
    pos[p[i]] = i
ans = 0
k = 0
for i in range(n):
    if p[i] != s[i]:
        j = pos[s[i]]
        ans += abs(i - j)
        p[i], p[j] = p[j], p[i]
        pos[p[i]] = i
        pos[p[j]] = j
        k += 1
print(ans)
print(k)
for i in range(n):
    if p[i] != s[i]:
        j = pos[s[i]]
        print(i+1, j+1)
        p[i], p[j] = p[j], p[i]
        pos[p[i]] = i
        pos[p[j]] = j
