#include<bits/stdc++.h>
using namespace std;
int n, x, h[2005], m[2005], t[2005], p[2005][2005], v[2][2005];
int main() {
    cin >> n >> x;
    for (int i = 1; i <= n; i++) {
        cin >> t[i] >> h[i] >> m[i];
    }
    for (int i = n; i >= 1; i--) {
        for (int j = 0; j <= x + 10; j++) {
            int k = t[i] ^ 1;
            v[t[i]][j] = max({v[t[i]][j], p[i + 1][j], p[i + 1][j + m[i]]});
            v[k][j + m[i]] = max(v[k][j + m[i]], p[i + 1][j] + m[i]);
            p[i][j] = v[t[i]][j];
            p[i][j + m[i]] = max(p[i][j + m[i]], v[k][j + m[i]]);
        }
    }
    cout << *max_element(p[1], p[1] + x + 11);
}
