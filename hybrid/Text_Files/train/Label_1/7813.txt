#include <algorithm>
#include <cstdio>
#include <iostream>
#include <utility>
#include <vector>

void solve() {
    int n, k;
    std::cin >> n >> k;
    if (k < 3) {
        std::cout << "NO" << std::endl;
    } else {
        std::cout << "YES" << std::endl;
        int cur = 1;
        std::vector<int> b(n), g(n);
        for (int i = 0; i < n; ++i) {
            b[i] = cur % k;
            g[i] = cur % k + 1;
            cur++;
        }
        for (int i = 0; i < n; ++i) {
            if (b[i] == 0) b[i] = k;
            if (g[i] == 0) g[i] = k;
            std::cout << b[i] << " " << g[i] << std::endl;
        }
    }
}

int main() {
    solve();
    return 0;
}
