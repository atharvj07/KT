import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int n = s.length();
        int[] dp = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            dp[i] = dp[i - 1] + 1;
            for (int j = 0; j < i; j++) {
                if (s.substring(j, i).equals(s.substring(i - (i - j), i))) {
                    dp[i] = Math.min(dp[i], dp[j] + 1 + i - j - (i - j) / s.substring(j, i).length());
                }
            }
        }
        System.out.println(dp[n]);
    }
}
