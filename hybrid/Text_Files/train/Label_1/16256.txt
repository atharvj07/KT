#include <bits/stdc++.h>
using namespace std;
long long n, A, cf, cm, m, a[100010];
int main() {
  cin >> n >> A >> cf >> cm >> m;
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  long long sum = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == A) sum += cf;
    if (m == 0) break;
    if (a[i] < A) {
      long long cost = A - a[i];
      if (cost <= m) {
        sum += cf;
        m -= cost;
        a[i] += cost;
      }
    }
  }
  long long force = sum + cm * a[0];
  cout << force << endl;
  for (int i = 0; i < n; i++) cout << a[i] << " ";
  return 0;
}
