#include <iostream>
#include <string>

int main() {
    long long k;
    std::string n;
    std::cin >> k >> n;

    int cnt = 0;
    long long sum = 0;
    for (int i = 0; i < n.size(); i++) {
        sum += n[i] - '0';
        if (sum < k) {
            cnt++;
            sum = n[i] - '0';
        }
    }

    std::cout << cnt << std::endl;

    return 0;
}
