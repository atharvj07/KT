#include <bits/stdc++.h>
using namespace std;
const int maxn=3e5+5;
typedef long long ll;
ll n,q,a[maxn],beauty;
inline ll gcd(ll a,ll b){
  if(b==0)
    return a;
  return gcd(b,a%b);
}
inline ll lcm(ll a,ll b){
  return a*b/gcd(a,b);
}
inline ll chk(ll x,ll y){
  ll g=gcd(x,y);
  ll t=lcm(x,y)/g;
  ll p=0;
  for(ll i=2;i<=sqrt(t);i++){
    if(t%i==0){
      p+=1;
      while(t%i==0)
        t/=i;
    }
  }
  if(t>1)
    p+=1;
  if(p<=1)
    return 1;
  return 0;
}
int main(){
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int t;
  cin>>t;
  while(t--){
    beauty=0;
    cin>>n;
    for(int i=0;i<n;++i){
      cin>>a[i];
      if(i){
        beauty=max(beauty,chk(a[i],a[i-1]));
      }
    }
    cin>>q;
    while(q--){
      ll x;
      cin>>x;
      if(x%2==0)
        cout<<n<<endl;
      else
        cout<<beauty<<endl;
    }
  }
  return 0;
}
