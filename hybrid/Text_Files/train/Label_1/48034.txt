import java.util.Scanner;

 class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int m = scanner.nextInt();
    for (int i = 0; i < n; i++) {
      if (i % 2 == 0) {
        System.out.print("G");
      } else {
        System.out.print("B");
      }
    }
    for (int i = 0; i < m; i++) {
      if (i % 2 == 0) {
        System.out.print("B");
      } else {
        System.out.print("G");
      }
    }
    scanner.close();
  }
}
