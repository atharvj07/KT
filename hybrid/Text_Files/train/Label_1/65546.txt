import re
good = input()
pattern = input()
n = int(input())
bad = set(ch for ch in 'abcdefghijklmnopqrstuvwxyz' if ch not in good)
for i in range(n):
    query = input()
    if re.match(pattern.replace('?', good).replace('*', '[' + ''.join(bad) + ']*'), query):
        print('YES')
    else:
        print('NO')
