from collections import deque

def solve_dishes(n, m, impressions):
    graph = [[] for _ in range(n + m)]
    in_degree = [0] * (n + m)
    value = [1] * (n + m)

    for i in range(n):
        for j, impression in enumerate(impressions[i]):
            if impression == "=":
                graph[i].append(n + j)
                graph[n + j].append(i)
            elif impression == "<":
                graph[i].append(n + j)
                in_degree[n + j] += 1
            else:  # impression == ">"
                graph[n + j].append(i)
                in_degree[i] += 1

    queue = deque([i for i in range(n + m) if in_degree[i] == 0])

    while queue:
        node = queue.popleft()
        for neighbor in graph[node]:
            value[neighbor] = max(value[neighbor], value[node] + (neighbor not in graph[node]))
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    if any(degree > 0 for degree in in_degree):
        return "No\n"
    else:
        return "Yes\n" + " ".join(map(str, value[:n])) + "\n" + " ".join(map(str, value[n:])) + "\n"

# Reading inputs
n, m = map(int, input().split())
impressions = [input() for _ in range(n)]

# Solve and print the result
print(solve_dishes(n, m, impressions))
