import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;
import java.util.Scanner;

class Job implements Comparable<Job> {
    public int reward;
    public int days;

    public Job(int reward, int days) {
        this.reward = reward;
        this.days = days;
    }

    @Override
    public int compareTo(Job other) {
        if (this.days == other.days) {
            return this.reward - other.reward;
        } else {
            return this.days - other.days;
        }
    }
}

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Job> jobs = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int days = scanner.nextInt();
            int reward = scanner.nextInt();
            jobs.add(new Job(reward, days));
        }

        Collections.sort(jobs, Comparator.reverseOrder());

        int maxReward = 0;
        int remainingDays = m;
        for (Job job : jobs) {
            if (remainingDays >= job.days) {
                maxReward += job.reward;
                remainingDays -= job.days;
            } else {
                break;
            }
        }

        System.out.println(maxReward);
    }
}
