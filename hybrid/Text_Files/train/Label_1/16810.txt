#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main()
{
  int m;
  string s;
  cin >> m >> s;
  vector<int> ans;
  for (int i = 0; i < (int)s.length() - m + 1; i++) {
    if (s[i] == s[i + m - 1]) {
      ans.push_back(i);
    }
  }
  for (int i = 0; i < (int)s.length() - m + 1; i++) {
    if (s[i] < s[i + m - 1]) {
      ans.push_back(i);
      break;
    }
  }
  string anss = s.substr(ans[0], m);
  for (int i = 1; i < (int)ans.size(); i++) {
    anss += s.substr(ans[i], m);
  }
  sort(anss.begin(), anss.end());
  cout << anss << endl;
}
