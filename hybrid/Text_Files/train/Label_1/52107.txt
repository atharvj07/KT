import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = sc.nextInt();
        Set<String> set = new HashSet<>();
        for (int i = 0; i < n; i++) {
            set.add(sc.nextLine());
        }
        int maxLen = 0;
        int start = 0;
        int end = 0;
        int[] dp = new int[s.length() + 1];
        dp[0] = 0;
        for (int i = 1; i <= s.length(); i++) {
            dp[i] = dp[i - 1];
            for (int j = i; j >= 1; j--) {
                String sub = s.substring(j - 1, i);
                if (!set.contains(sub)) {
                    dp[i] = Math.max(dp[i], dp[j - 1] + i - j + 1);
                }
            }
            if (dp[i] > maxLen) {
                maxLen = dp[i];
                end = i;
            }
        }
        for (int i = end - maxLen; i < end; i++) {
            start = i;
        }
        System.out.println(maxLen + " " + start);
    }
}
