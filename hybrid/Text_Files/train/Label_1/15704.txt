#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int n, d;
  cin >> n >> d;
  vector<int> x(n);
  for (int i = 0; i < n; ++i) cin >> x[i];
  sort(x.begin(), x.end());
  int ans = 0;
  for (int i = 0; i < n - 1; ++i) {
    if (x[i + 1] - x[i] > d) {
      ++ans;
      ++i;
    }
  }
  cout << ans << endl;
  return 0;
}
