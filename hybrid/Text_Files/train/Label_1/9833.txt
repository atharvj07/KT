#include <bits/stdc++.h>
#define endl "\n"
using namespace std;
int main() {
    string s;
    cin >> s;
    int n = s.size();
    int m;
    cin >> m;
    while (m--) {
        int l, r;
        cin >> l >> r;
        if (r - l < 2) {
            cout << "YES" << endl;
            continue;
        }
        int x, y, z;
        x = y = z = 0;
        for (int i = l - 1; i <= r - 1; i++) {
            if (s[i] == 'x')
                x++;
            else if (s[i] == 'y')
                y++;
            else
                z++;
        }
        if (x && y && z)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
