import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class BlackWhiteTree {

    static class Node {
        int id;
        List<Node> children;

        Node(int id) {
            this.id = id;
            this.children = new ArrayList<>();
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            nodes[i] = new Node(i);
        }

        for (int i = 0; i < n - 1; i++) {
            line = br.readLine().split(" ");
            int x = Integer.parseInt(line[0]);
            int y = Integer.parseInt(line[1]);
            nodes[x].children.add(nodes[y]);
            nodes[y].children.add(nodes[x]);
        }

        long[] dp = new long[n + 1];
        dfs(nodes[1], null, dp, m);

        for (int i = 1; i <= n; i++) {
            System.out.println(dp[i]);
        }
    }

    private static void dfs(Node node, Node parent, long[] dp, int m) {
        long allWhite = 1;
        long allBlack = 1;
        for (Node child : node.children) {
            if (child != parent) {
                dfs(child, node, dp, m);
                allWhite = (allWhite * (dp[child.id] + 1)) % m;
                allBlack = (allBlack * dp[child.id]) % m;
            }
        }
        dp[node.id] = (allWhite + allBlack) % m;
    }
}
