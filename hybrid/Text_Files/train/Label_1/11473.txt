#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, s;
        cin >> n >> s;
        vector<int> a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        int ans = 0;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            if (sum + a[i] > s && sum + a[ans] <= s) { ans = i; }
            sum += a[i];
        }
        cout << ans << '\n';
    }
    return 0;
}
