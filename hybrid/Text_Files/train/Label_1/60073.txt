n, k, m = map(int, input().split())

def count_numbers(n, k, m):
    dp = [[0 for i in range(2)] for j in range(n+1)]
    dp[0][0] = 1
    
    for i in range(1, n+1):
        for j in range(10):
            if i == 1 and j == 0: continue
            if j % k == 0:
                dp[i][1] = (dp[i][1] + dp[i-1][0] + dp[i-1][1]) % m
            else:
                dp[i][1] = (dp[i][1] + dp[i-1][1]) % m
                dp[i][0] = (dp[i][0] + dp[i-1][0]) % m
    
    return (dp[n][1]) % m

print(count_numbers(n, k, m))
