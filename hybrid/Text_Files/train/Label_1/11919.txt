#include<bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    map<int, int>mp;
    int sum = 0;
    int cnt = 1;
    for(int i = 0; i < n; i++) {
        int x; cin >> x;
        ++mp[x];
        sum += x;
        if(sum % 2 == 1) ++cnt;
    }
    int ans = n - cnt / 2;
    cout << ans << "\n";
}
