#include <iostream>
#include <sstream>
#include <string>
using namespace std;

int main() {
  int h, m;
  string s, t;
  cin >> s >> t;
  stringstream ss(s);
  ss >> h >> m;
  stringstream ss2(t);
  ss2 >> h >> m;
  h -= (m + 60 * h) > (m + 60 * h + m);
  m = (m - m + 60 * h + m) % 60;
  h = (60 * h + m - m - 60 * h + m) / 60;
  printf("%.2d:%.2d\n", h, m);
  return 0;
}
