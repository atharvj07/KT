#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  set<string> titles;
  for (int i = 0; i < n; ++i) {
    string title;
    cin >> title;
    titles.insert(title);
  }
  for (int i = 1; ; ++i) {
    string title(i, 'a');
    bool original = true;
    for (auto &t : titles) {
      if (t.find(title) != string::npos) {
        original = false;
        break;
      }
    }
    if (original) {
      cout << title << endl;
      return 0;
    }
  }
  return 0;
}
