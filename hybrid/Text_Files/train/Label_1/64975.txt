import sys

t = int(sys.stdin.readline())


def cross_product(p1, p2, p3):
    x1, y1 = p1
    x2, y2 = p2
    x3, y3 = p3
    return (x2 - x1) * (y3 - y2) - (y2 - y1) * (x3 - x2)


for _ in range(t):
    n = int(sys.stdin.readline())
    people = []
    for i in range(n):
        x, y, u, v = map(int, sys.stdin.readline().split())
        people.append(((x, y), (u, v)))

    ans = 0

    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            a1, b1 = people[i]
            a2, b2 = people[j]
            if cross_product(a1, a2, b1) * cross_product(a1, a2, b2) > 0:
                ans += 1

    print(ans)
