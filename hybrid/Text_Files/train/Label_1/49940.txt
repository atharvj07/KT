import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        boolean[][] burnt = new boolean[n + 1][m + 1];
        int[][] dist = new int[n + 1][m + 1];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(burnt[i], false);
            Arrays.fill(dist[i], Integer.MAX_VALUE);
        }
        int k = input.nextInt();
        for (int i = 0; i < k; i++) {
            int x = input.nextInt();
            int y = input.nextInt();
            burnt[x][y] = true;
            dist[x][y] = 0;
        }
        Queue<int[]> q = new LinkedList<>();
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (burnt[i][j]) {
                    q.add(new int[]{i, j});
                }
            }
        }
        while (!q.isEmpty()) {
            int[] cur = q.poll();
            int x = cur[0];
            int y = cur[1];
            if (x > 1 && dist[x - 1][y] > dist[x][y] + 1) {
                dist[x - 1][y] = dist[x][y] + 1;
                q.add(new int[]{x - 1, y});
            }
            if (x < n && dist[x + 1][y] > dist[x][y] + 1) {
                dist[x + 1][y] = dist[x][y] + 1;
                q.add(new int[]{x + 1, y});
            }
            if (y > 1 && dist[x][y - 1] > dist[x][y] + 1) {
                dist[x][y - 1] = dist[x][y] + 1;
                q.add(new int[]{x, y - 1});
            }
            if (y < m && dist[x][y + 1] > dist[x][y] + 1) {
                dist[x][y + 1] = dist[x][y] + 1;
                q.add(new int[]{x, y + 1});
            }
        }
        int maxDist = 0;
        int x = 0;
        int y = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (dist[i][j] > maxDist) {
                    maxDist = dist[i][j];
                    x = i;
                    y = j;
                }
            }
        }
        System.out.println(x + " " + y);
    }
}
