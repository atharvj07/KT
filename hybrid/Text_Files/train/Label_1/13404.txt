#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<std::string> passwords(n);
  for (int i = 0; i < n; i++) {
    std::cin >> passwords[i];
  }
  std::string target;
  std::cin >> target;
  std::sort(passwords.begin(), passwords.end(), [](const auto &a, const auto &b) { return a.size() < b.size(); });
  int best = 0, worst = 0;
  int p = 0, c = 0;
  for (const auto &password : passwords) {
    c++;
    if (password == target) {
      p++;
      best += (p <= k) ? 1 : 6;
      worst += c;
      break;
    }
    if (c == k) {
      c = 0;
      worst += 5;
    }
  }
  std::cout << best << " " << worst << std::endl;
  return 0;
}
