import java.util.*;
import java.io.*;

public class Main {
    static int[] a;
    static int[][] f, g;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            a = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                a[i] = sc.nextInt();
            }
            f = new int[n + 1][101];
            g = new int[n + 1][101];
            for (int i = 1; i <= 100; i++) {
                f[n][i] = (a[n] % i == 0) ? a[n] / i : 1000000000;
            }
            for (int i = n - 1; i >= 1; i--) {
                for (int j = 1; j <= 100; j++) {
                    f[i][j] = 1000000000;
                    for (int x = 0; x <= a[i]; x++) {
                        if (j - x >= 1) {
                            f[i][j] = Math.min(f[i][j], f[i + 1][j - x]);
                        }
                        if (i + 1 < n && j - x >= 1) {
                            f[i][j] = Math.min(f[i][j], f[i + 1][j - x] + 1);
                        }
                        if (x > 0) {
                            f[i][j] = Math.min(f[i][j], f[i][j] + 1);
                        }
                    }
                }
            }
            int ans = 1000000000;
            for (int i = 1; i <= 100; i++) {
                if (f[1][i] <= k) {
                    ans = Math.min(ans, i);
                }
            }
            System.out.println(ans == 1000000000 ? -1 : ans);
        }
    }
}
