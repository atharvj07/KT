#include <iostream>

using namespace std;

int main() {
  long long n;
  cin >> n;
  string dir;
  cin >> dir;
  long long dist[n];
  for (long long i = 0; i < n; i++) { cin >> dist[i]; }
  if (dir[0] == '<') { cout << "FINITE" << endl; } else if (dir[n - 1] == '>') {
    cout << "FINITE" << endl;
  } else {
    cout << "INFINITE" << endl;
  }
  return 0;
}
