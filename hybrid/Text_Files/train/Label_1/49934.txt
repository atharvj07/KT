import java.util.*;

public class Main {
    static int[] color, degree;
    static ArrayList<Integer>[] adj;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        color = new int[n + 1];
        degree = new int[n + 1];
        adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++)
            adj[i] = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            color[i] = in.nextInt();
        }
        for (int i = 0; i < m; i++) {
            int a = in.nextInt(), b = in.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        int ans = 0, ansColor = 0;
        for (int i = 1; i <= n; i++) {
            HashSet<Integer> hs = new HashSet<>();
            for (int nbr : adj[i]) {
                hs.add(color[nbr]);
            }
            if (hs.size() > ans) {
                ans = hs.size();
                ansColor = color[i];
            } else if (hs.size() == ans) {
                ansColor = Math.min(ansColor, color[i]);
            }
        }
        System.out.println(ansColor);
    }
}
