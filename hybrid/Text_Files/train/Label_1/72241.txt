n,d=map(int,input().split())
a=[0]*n
for i in range(n):
    a[i]=list(map(int,input().split()))
a.sort()
ans,fr=-1<<30,-1<<30
sm=0
for i in range(n):
    if i and a[i][0]<=a[i-1][0]+d:continue
    tmp=a[i][1]
    for j in range(i+1,n):
        if a[j][0]<=a[i][0]+d:tmp+=a[j][1]
        else:break
    if tmp>fr:
        fr=tmp
        ans=a[i][0]
print(fr)
