#include<bits/stdc++.h>
using namespace std;
int main() {
  long long n, x, c = 0, ans = 0;
  cin >> n >> x;
  while (n--) {
    char ch;
    long long d;
    cin >> ch >> d;
    if (ch == '+') {
      x += d;
      c++;
    } else {
      if (x - d >= 0) {
        x -= d;
      } else {
        ans++;
      }
    }
  }
  cout << x << " " << ans;
}
