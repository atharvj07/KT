import sys

n = int(sys.stdin.readline())
a = [int(x) for x in sys.stdin.readline().split()]
p = [int(x) for x in sys.stdin.readline().split()]
k = a.count(0)

def dfs(v):
    if a[v] == 0:
        return 1
    L = []
    for u in range(v + 1, n + 1):
        if p[u] == v:
            L.append(dfs(u))
    if a[v] == 1:
        return max(L) + 1
    else:
        return min(L) + 1

print(dfs(0))
