#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int m;
        cin >> m;
        vector<vector<int>> a(2, vector<int>(m, 0));
        for (int i = 0; i < 2; i++) {
            for (int j = 0; j < m; j++) { cin >> a[i][j]; }
        }
        vector<vector<int>> dp(2, vector<int>(m, 0));
        for (int j = m - 2; j >= 0; j--) { dp[0][j] = dp[1][j + 1] + a[0][j]; }
        for (int i = 1; i < 2; i++) {
            for (int j = m - 2; j >= 0; j--) {
                dp[i][j] = max(dp[i - 1][j + 1], dp[i][j + 1]) + a[i][j];
            }
        }
        cout << dp[1][0] << endl;
    }
    return 0;
}
