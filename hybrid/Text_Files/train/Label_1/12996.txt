#include <bits/stdc++.h>

#define int long long

const int mod = 1e9 + 7;
const int maxn = 52;

int32_t main() {
  int n, p;
  std::cin >> n >> p;
  std::vector<int> c(n);
  for (int i = 0; i < n; ++i) std::cin >> c[i];
  std::vector<std::vector<int>> dp(n, std::vector<int>(2, 0));
  if (!c[0]) {
    if (p) {
      dp[0][1] = 1;
    } else {
      dp[0][1] = 1;
      dp[0][0] = 1;
    }
  } else {
    dp[0][c[0] + 1] = 1;
  }
  for (int i = 1; i < n; ++i) {
    for (int las = 0; las < 2; ++las) {
      for (int j = i; j < n; ++j) {
        if (!c[j]) {
          if (las) {
            dp[j][1] += dp[i][0];
            dp[j][1] %= mod;
          } else {
            dp[j][1] += dp[i][0];
            dp[j][1] %= mod;
            dp[j][0] += dp[i][1];
            dp[j][0] %= mod;
          }
        } else {
          if (las + c[j] != 1) continue;
          dp[j][c[j] + 1] += dp[i][las];
          dp[j][c[j] + 1] %= mod;
        }
      }
    }
  }
  std::cout << dp[n - 1][p] << "\n";
}
