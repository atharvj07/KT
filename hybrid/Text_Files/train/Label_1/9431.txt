#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9+7;
const int maxn = 2e5+5;
ll a[maxn],b[maxn];
int main() {
    ll n,k;
    cin>>n>>k;
    for(ll i=0; i<n; i++) cin>>a[i];
    for(ll i=0; i<n; i++) cin>>b[i];
    sort(a,a+n);
    sort(b,b+n);
    ll ans=0;
    for(ll i=0; i<k; i++) ans+=a[i];
    for(ll i=k; i<n; i++) ans+=min(a[i],b[i]);
    cout<<ans<<'\n';
}
