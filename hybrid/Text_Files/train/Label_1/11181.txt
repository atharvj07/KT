#include <bits/stdc++.h>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n);
  for (int& x : a) {
    std::cin >> x;
  }
  std::sort(a.begin(), a.end(), std::greater<>());
  int ans = (k - 1) * n - std::accumulate(a.begin(), a.end(), 0);
  if (ans > 0) {
    std::cout << ans << '\n';
  } else {
    std::cout << 0 << '\n';
  }
  return 0;
}
