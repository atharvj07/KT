n, k = map(int, input().split())
h = list(map(int, input().split()))
m = bin(k)[2:]
pow = [1] * len(m)
for i in range(1, len(m)):
    pow[i] = pow[i - 1] * 2
msk = []
for i in range(len(m)):
    if m[i] == '0':
        msk.append(0)
    else:
        msk.append(pow[i] - 1)
ans = 0
for l in range(len(msk)):
    cur = []
    msk[l] = 0
    for i in range(n):
        cur.append(h[i])
        h[i] -= 1
        h[i] &= msk[i % len(msk)]
    cur.append(cur[0])
    cur.reverse()
    add = 1
    for i in range(n):
        if cur[i] != cur[i + 1]:
            add = add * (cur[i + 1] + 1) % 998244353
    ans += add
ans = ans % 998244353
print(ans)
