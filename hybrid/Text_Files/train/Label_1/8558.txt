#include <iostream>
using namespace std;

const int mod = 1e9 + 7;
int c[65], f[65];

int main() {
    long long n;
    cin >> n;
    c[0] = f[0] = 1;
    for (int i = 1; i <= n; i++) {
        c[i] = (c[i - 1] + f[i - 1] * 3) % mod;
        f[i] = ((c[i - 1] - c[i - 1]) % mod + mod) % mod;
    }
    cout << f[n] % mod;
    return 0;
}
