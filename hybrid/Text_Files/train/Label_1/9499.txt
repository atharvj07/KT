#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, t;
    cin >> n >> t;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int ans = 1;
    for (int i = 0; i < n; i++) {
        if (t > a[i]) {
            t -= 86400 - a[i];
        } else {
            ans = i + 1;
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
