import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[] x = new int[m];
    int[] y = new int[m];
    int[] z = new int[m];
    for (int i = 0; i < m; i++) {
      x[i] = sc.nextInt() - 1;
      y[i] = sc.nextInt() - 1;
      z[i] = sc.nextInt();
    }
    int[] ans = new int[n];
    Arrays.fill(ans, -1);
    int[] deg = new int[n];
    for (int i = 0; i < m; i++) {
      deg[x[i]]++;
      deg[y[i]]++;
    }
    Queue<Integer> q = new LinkedList<>();
    for (int i = 0; i < n; i++) {
      if (deg[i] == 1) q.add(i);
    }
    int cost = 0;
    while (!q.isEmpty()) {
      int now = q.poll();
      cost++;
      if (ans[now] == -1) ans[now] = z[0] - ans[y[0]];
      for (int i = 0; i < m; i++) {
        if (x[i] == now || y[i] == now) {
          int to = x[i] + y[i] - now;
          if (ans[to] == -1) {
            ans[to] = z[i] - ans[now];
            deg[to]--;
            if (deg[to] == 1) q.add(to);
          }
        }
      }
    }
    System.out.println(cost);
  }
}
