#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<string> strs(n);
    for (int i = 0; i < n; i++) {
        cin >> strs[i];
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int s = 0, h = 0;
        for (int j = 0; j < strs[i].size(); j++) {
            if (strs[i][j] == 's') s++;
            else if (strs[i][j] == 'h') h++;
        }
        ans += s * h;
    }
    cout << ans << endl;
    return 0;
}
