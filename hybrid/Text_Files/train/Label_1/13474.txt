#include <iostream>
#include <vector>
#define ll long long

using namespace std;

int main() {
  int t;
  ll n;
  ll a, b;
  vector<pair<ll, ll>> v;

  cin >> t;
  while (t--) {
    cin >> n;
    ll ans = 0;
    v.clear();
    for (int i = 0; i < n; i++) {
      cin >> a >> b;
      v.push_back({a, b});
    }
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if (v[i].first <= v[j].first && v[i].second == v[j].second) { ans++; }
      }
    }
    cout << ans << endl;
  }

  return 0;
}
