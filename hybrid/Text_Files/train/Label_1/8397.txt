#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    vector<int> ans;
    int g = 0;
    for (int x : a) g = __gcd(g, x);
    for (int i = 1; i * i <= g; i++) {
      if (g % i == 0) {
        if (g / i != i) ans.push_back(g / i);
        ans.push_back(i);
      }
    }
    sort(ans.begin(), ans.end());
    ans.erase(unique(ans.begin(), ans.end()), ans.end());
    for (int x : ans) cout << x << " ";
    cout << "\n";
  }
}
