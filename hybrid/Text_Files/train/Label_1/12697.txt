#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

bool possible(vector<int> a, int n, int sum) {
  sort(a.begin(), a.end());
  int l = 0, r = n - 1;
  while (l < r) {
    if (a[l] + a[r] > sum) return false;
    l++, r--;
  }
  return true;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    int sum = 0;
    for (int &x : a) {
      cin >> x;
      sum += x;
    }
    if (!possible(a, n, sum)) {
      cout << -1 << endl;
      continue;
    }
    sort(a.begin(), a.end());
    for (int i = 0; i < n - 1; i++) {
      if (a[i] == a[i + 1]) cout << a[i] << " ";
      else cout << a[i] / 2 << " ";
    }
    cout << a.back() << endl;
  }
}
