n = int(input())
a = list(map(int, input().split()))

# Function to calculate the result for each k
def solve(n, a):
    # For k = 0, the game is played optimally so we calculate the result of the game
    dp = [0] * n
    for length in range(2, n + 1, 2):
        for i in range(n - length + 1):
            j = i + length - 1
            dp[i] = max(min(dp[i + 1], a[j]), min(dp[i], a[i]))

    # Results for when k is even, both players play optimally
    even_results = [max(a)]
    for i in range(2, n, 2):
        even_results.append(max(even_results[-1], dp[i]))

    # Results for when k is odd, Oleg has made one move already
    odd_results = [max(a[0], a[-1])]
    for i in range(1, n - 1, 2):
        odd_results.append(max(odd_results[-1], dp[i]))

    # Combining the results
    results = []
    for i in range(n):
        if i % 2 == 0:
            results.append(even_results[i // 2])
        else:
            results.append(odd_results[i // 2])

    return results

# Printing the results
print(*solve(n, a))
