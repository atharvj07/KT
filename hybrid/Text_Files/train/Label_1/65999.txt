import sys

def partition(lst, k, p):
    lst.sort(reverse=True)
    prefix = [0] * len(lst)
    prefix[0] = lst[0]
    for i in range(1, len(lst)):
        prefix[i] = prefix[i - 1] + lst[i]
    if prefix[len(lst) - 1] % 2 == 0:
        return False
    dp = [[-1] * (k + 1) for _ in range(len(lst) + 1)]
    def solve(i, j):
        if i == len(lst):
            if j == k:
                return True
            return False
        if dp[i][j] != -1:
            return dp[i][j]
        ans = False
        if (prefix[i] - (prefix[len(lst) - 1] - prefix[i])) % 2 == 1:
            ans |= solve(i + 1, j + 1)
        elif j > 0:
            ans |= solve(i + 1, j)
        dp[i][j] = ans
        return ans
    if solve(0, 0):
        print("YES")
        i = 0
        j = 0
        while i < len(lst):
            if (prefix[i] - (prefix[len(lst) - 1] - prefix[i])) % 2 == 1:
                print(i + 1, *lst[j:i + 1])
                j = i + 1
            i += 1
    else:
        print("NO")

n, k, p = map(int, sys.stdin.readline().split())
lst = list(map(int, sys.stdin.readline().split()))
partition(lst, k, p)
