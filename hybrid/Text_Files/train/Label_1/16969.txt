#include <iostream>
#include <string>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        int l = -1, r = -1;
        int cnt_a = 0, cnt_b = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == 'a') {
                cnt_a++;
            } else {
                cnt_b++;
            }
            if (cnt_a == cnt_b && (l == -1 || i - l > r - l)) {
                l = i - cnt_a + 1;
                r = i;
            }
        }
        cout << l << ' ' << r << endl;
    }
    return 0;
}
