#include <iostream>
#include <cmath>

using namespace std;

 int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k, d1, d2;
    cin >> n >> k >> d1 >> d2;
    if (d1 + d2 > k || (n - k) % 3 != 0 || abs(d1 - d2) > 1)
      cout << "no" << endl;
    else
      cout << "yes" << endl;
  }
  return 0;
}
