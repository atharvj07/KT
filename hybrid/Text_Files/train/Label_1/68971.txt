n = int(input().strip())
trophies = input().strip()

gold_counts = [0] * (n + 1)
total_gold = 0

for i in range(n):
    gold_counts[i + 1] = gold_counts[i] + (trophies[i] == 'G')
    total_gold += trophies[i] == 'G'

max_beauty = 0
for i in range(1, n + 1):
    if trophies[i - 1] == 'S':
        left = i - 1 - gold_counts[i - 1] + gold_counts[0]
        right = gold_counts[n] - gold_counts[i]
        max_beauty = max(max_beauty, gold_counts[i - 1] + right + (left + right < total_gold))
        
if total_gold == 0:
    print(0)
else:
    print(max(max_beauty, gold_counts[n]))
