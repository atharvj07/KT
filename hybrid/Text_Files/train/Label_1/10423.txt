#include<bits/stdc++.h>
using namespace std;

int main() {
    long long n;
    cin >> n;
    map<int, vector<int>> price;
    for (int i = 0; i < n;i++) {
        int x, y;
        cin >> x >> y;
        price[1000 - x].push_back(i);
    }
    string ans = "";
    long long sumA = 0, sumB = 0;
    for (auto i : price) {
        if (i.second.size() % 2 == 1) {
            ans += (sumA > sumB ? "A" : "G");
            (sumA > sumB ? sumA : sumB) += 1000 - i.first;
            continue;
        }
        for (int j = 0;j < i.second.size() / 2;j++) {
            ans += "A";
            sumA += 1000 - i.first;
        }
        for (int j = i.second.size() / 2;j < i.second.size();j++) {
            ans += "G";
            sumB += 1000 - i.first;
        }
    }
    if (abs(sumA - sumB) > 500) ans = "-1";
    cout << ans;
}
