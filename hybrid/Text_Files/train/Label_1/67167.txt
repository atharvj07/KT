import sys

n = int(input())
a = [0] * (n - 1)
b = [0] * (n - 1)
for i in range(n - 1):
    a[i], b[i] = map(int, input().split())

deg = [0] * (n + 1)
for i in range(n - 1):
    deg[a[i]] += 1
    deg[b[i]] += 1

if max(deg) > 2:
    print('NO')
    sys.exit()

for i in range(1, n + 1):
    if deg[i] == 0:
        print('NO')
        sys.exit()

    if deg[i] == n - 1:
        root = i
        break

edges = []
for i in range(n - 1):
    if a[i] == root or b[i] == root:
        edges.append((root, a[i] + b[i] - root))
    else:
        edges.append((a[i], b[i]))

print('YES')
for e in edges:
    print(*e)
