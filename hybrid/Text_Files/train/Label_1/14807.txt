#include <iostream>
#include <vector>
#include <limits.h>
using namespace std;
const int INF = INT_MAX;
int main() {
    int n1, n2, n3, n4, m1, m2, m3;
    cin >> n1 >> n2 >> n3 >> n4;
    vector<int> a(n1 + 1), b(n2 + 1), c(n3 + 1), d(n4 + 1);
    for (int i = 1; i <= n1; ++i) cin >> a[i];
    for (int i = 1; i <= n2; ++i) cin >> b[i];
    for (int i = 1; i <= n3; ++i) cin >> c[i];
    for (int i = 1; i <= n4; ++i) cin >> d[i];
    cin >> m1;
    vector<pair<int, int>> bad1(m1);
    for (int i = 0; i < m1; ++i) { cin >> bad1[i].first >> bad1[i].second; }
    cin >> m2;
    vector<pair<int, int>> bad2(m2);
    for (int i = 0; i < m2; ++i) { cin >> bad2[i].first >> bad2[i].second; }
    cin >> m3;
    vector<pair<int, int>> bad3(m3);
    for (int i = 0; i < m3; ++i) { cin >> bad3[i].first >> bad3[i].second; }
    vector<vector<int>> dp(n1 + 1, vector<int>(n2 + 1, INF));
    dp[0][0] = 0;
    for (int i = 0; i <= n1; ++i) {
        for (int j = 0; j <= n2; ++j) {
            if (i) {
                dp[i][j] = min(dp[i][j], dp[i - 1][j] + a[i]);
                if (j) {
                    bool ok = true;
                    for (pair<int, int> p : bad1)
                        if ((p.first == i && p.second == j) || (p.first == j && p.second == i)) { ok = false; break; }
                    if (ok) dp[i][j] = min(dp[i][j], dp[i - 1][j - 1] + a[i] + b[j]);
                }
            }
            if (j) {
                dp[i][j] = min(dp[i][j], dp[i][j - 1] + b[j]);
            }
        }
    }
    int ans = INF;
    for (int j = 0; j <= n2; ++j) {
        for (int k = 0; k <= n3; ++k) {
            bool ok = true;
            for (pair<int, int> p : bad2)
                if (p.first == j && p.second == k) { ok = false; break; }
            if (ok) ans = min(ans, dp[n1][j] + b[j] + c[k]);
        }
    }
    if (ans == INF) { cout << -1 << endl; return 0; }
    for (int j = 0; j <= n2; ++j) {
        for (int k = 0; k <= n3; ++k) {
            for (int h = 0; h <= n4; ++h) {
                bool ok = true;
                for (pair<int, int> p : bad2)
                    if ((p.first == j && p.second == k) || (p.first == k && p.second == j)) { ok = false; break; }
                for (pair<int, int> p : bad3)
                    if ((p.first == k && p.second == h) || (p.first == h && p.second == k)) { ok = false; break; }
                if (ok) ans = min(ans, dp[n1][j] + b[j] + c[k] + d[h]);
            }
        }
    }
    cout << ans << endl;
}
