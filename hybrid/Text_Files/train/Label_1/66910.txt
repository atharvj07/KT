import math
for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    b = list(map(int,input().split()))
    x = 0
    y = 0
    for i in range(n):
        x += a[i] - b[i]
        if a[i] > b[i]: y += a[i] - b[i]
    if x == 0 or (x > 0 and y % x == 0): print('YES')
    else: print('NO')
