#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mxN = 1e6 + 10, mod = 1e9 + 7;
vector<int> frq(mxN), add(mxN);
int32_t main() {
    int n, q, k;
    cin >> n >> q >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i], frq[a[i]]++;
    int ans = 0;
    for (int i = 1; i <= k; i++) add[i] = add[i - 1] + frq[i];
    while (q--) {
        int l, r;
        cin >> l >> r;
        l--, r--;
        ans = (add[r] - (l > 0 ? add[l - 1] : 0));
        if (a[l] == a[r])
            ans--;
        else
            ans += 2;
        ans = max(ans, 0ll);
        cout << ans << "\n";
    }
    return 0;
}
