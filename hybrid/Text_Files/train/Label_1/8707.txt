#include <iostream>
#include <cstring>

using namespace std;

int main() {
  int n;
  cin >> n;
  char a[n + 1], b[n + 1];
  cin >> a >> b;
  int cnt[10];
  memset(cnt, 0, sizeof cnt);
  for (int i = 0; i < n; i++) cnt[a[i] - '0']++;
  for (int i = 0; i < n; i++) cnt[b[i] - '0']--;
  int minf = 0, maxf = 0;
  for (int i = 0; i <= 9; i++) {
    if (cnt[i] < 0) minf -= cnt[i];
    if (cnt[i] > 0) maxf += cnt[i];
  }
  cout << minf << " " << maxf << endl;
  return 0;
}
