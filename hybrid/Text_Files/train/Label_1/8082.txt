#include <algorithm>
#include <iostream>
using namespace std;

int main() {
    long long a1, b1, a2, b2;
    cin >> a1 >> b1;
    cin >> a2 >> b2;
    long long g = __gcd(a1, b1), g1 = __gcd(a2, b2), g2 = __gcd(a1, b2),
              g3 = __gcd(b1, a2), gg = 1e18;
    if ((a1 * b1 / g) % (a2 * b2 / g1))
        gg = -1;
    else if ((a1 * b1 / g) % (b2 * a2 / g3))
        gg = -1;
    else if ((a2 * b2 / g1) % (b1 * a2 / g2))
        gg = -1;
    if (gg != 1e18) {
        long long ans = 0;
        if (a1 * b1 > a2 * b2) {
            while (a1 * b1 != a2 * b2) {
                if ((a1 * b1) % 2 == 0 && (a1 / 2) * b1 * g2 % (a2 * b2 / g1) ==
                                                 0) {
                    a1 /= 2;
                    ans++;
                } else if (a1 % 3 == 0 &&
                           ((a1 / 3) * b1 * g2) % ((a2 * b2) / g1) == 0) {
                    a1 /= 3;
                    ans++;
                } else if ((b1 * a1) % 2 == 0 &&
                           (a1 * (b1 / 2)) * g2 % ((a2 * b2) / g1) == 0) {
                    b1 /= 2;
                    ans++;
                } else if (b1 % 3 == 0 &&
                           (a1 * (b1 / 3)) * g2 % ((a2 * b2) / g1) == 0) {
                    b1 /= 3;
                    ans++;
                } else {
                    gg = -1;
                    break;
                }
            }
        } else if (a1 * b1 < a2 * b2) {
            while (a1 * b1 != a2 * b2) {
                if ((a2 * b2) % 2 == 0 && (a2 / 2) * b2 * g % (a1 * b1 / g1) ==
                                                 0) {
                    a2 /= 2;
                    ans++;
                } else if (a2 % 3 == 0 &&
                           ((a2 / 3) * b2 * g) % ((a1 * b1) / g1) == 0) {
                    a2 /= 3;
                    ans++;
                } else if ((b2 * a2) % 2 == 0 &&
                           (a2 * (b2 / 2)) * g % ((a1 * b1) / g1) == 0) {
                    b2 /= 2;
                    ans++;
                } else if (b2 % 3 == 0 &&
                           (a2 * (b2 / 3)) * g % ((a1 * b1) / g1) == 0) {
                    b2 /= 3;
                    ans++;
                } else {
                    gg = -1;
                    break;
                }
            }
        }
        if (gg != -1) {
            cout << ans << endl;
            cout << a1 << " " << b1 << endl;
            cout << a2 << " " << b2 << endl;
            return 0;
        }
    }
    cout << -1 << endl;
    return 0;
}
