#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  getline(cin, s);
  int i;
  for (i = 0; i < s.size() - 1; ++i) {
    if (s[i] >= 'a' && s[i] <= 'z') {
      if (s[i] == 'n') {
        continue;
      }
      if (s[i + 1] < 'a' || s[i + 1] > 'z' || s[i + 1] == 'n') {
        break;
      }
    }
  }
  if (i == s.size() - 1) {
    cout << "YES\n";
  } else {
    cout << "NO\n";
  }
  return 0;
}
