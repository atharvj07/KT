#include <bits/stdc++.h>
using namespace std;
int main() {
    int a, b, c, n, ans = 0;
    long long safes[1000001] = {};
    unordered_map<int, int> mp;
    cin >> a >> b >> c >> n;
    for (int i = 0; i < n; i++) {
        int pos;
        cin >> pos;
        safes[pos]++;
    }
    for (int i = a; i != b && i != c; i--) ans += safes[i];
    for (int i = a; i != b && i != c; i++) ans += safes[i];
    cout << ans << endl;
}
