import sys
n,k=map(int,input().split())
a=list(map(int,sys.stdin.readline().split()))
e=[set()for i in range(n)]
for i in range(n-1):
    u,v=map(int,sys.stdin.readline().split())
    e[u-1].add(v-1)
    e[v-1].add(u-1)
def dfs1(x,p):
    d[x][0]=a[x]
    for nxt in e[x]:
        if nxt!=p:
            dfs1(nxt,x)
            if d[nxt][0]>0:d[x][0]+=d[nxt][0]
            else:d[x][1]+=d[nxt][1]
def dfs2(x,p,keep):
    if keep:d[x][0]=a[x]
    else:d[x][0]=0
    for nxt in e[x]:
        if nxt!=p:
            if keep or d[x][0]==0:dfs2(nxt,x,keep)
            elif d[nxt][0]>0:dfs2(nxt,x,keep)
            elif d[nxt][1]>0:dfs2(nxt,x,1)
    if d[x][1]>0:d[x][0]+=d[x][1]
d=[[0,0]for i in range(n)]
dfs1(0,-1)
dfs2(0,-1,1)
print(max(d[i][0]for i in range(n)))
