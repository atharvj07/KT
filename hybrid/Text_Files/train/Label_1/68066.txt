n = int(input())
a = list(map(int, input().split()))

if sum(a) % 2 != 0:
    print(0)
else:
    a_sorted = sorted((val, idx) for idx, val in enumerate(a))
    for val, idx in a_sorted:
        if val % 2 != 0:
            print(1)
            print(idx + 1)
            break
        elif (sum(a) - val) % 2 != 0:
            print(1)
            print(idx + 1)
            break
    else:
        print(0)
