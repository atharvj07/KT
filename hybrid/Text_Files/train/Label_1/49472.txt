import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String[] names = new String[n];
        for (int i = 0; i < n; i++) {
            names[i] = input.next();
        }
        int maxLen = 0;
        for (int i = 0; i < n; i++) {
            int l = i, r = i, len = names[i].length();
            while (l >= 0 && r < n) {
                if (names[l].charAt(names[l].length() - 1) == names[r].charAt(0)) {
                    len += names[r].length();
                    l--;
                    r++;
                } else {
                    break;
                }
            }
            maxLen = Math.max(len, maxLen);
        }
        System.out.println(maxLen);
    }
}
