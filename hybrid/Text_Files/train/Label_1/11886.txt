#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 5, M = 1 << 6;
int32_t main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int n, m;
    cin >> n >> m;
    vector<vector<int>> adj(n + 1);
    for (int i = 1; i <= m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    vector<int> deg(n + 1);
    for (int i = 1; i <= n; i++) {
        deg[i] = adj[i].size();
    }
    vector<int> dp(1 << n, -1);
    function<int(int, int)> solve = [&](int mask, int cur) {
        if (dp[mask] != -1) return dp[mask];
        int &ret = dp[mask];
        ret = 0;
        for (auto &u : adj[cur]) {
            if (mask & (1 << u)) continue;
            if (deg[cur] % 2 == deg[u] % 2 || deg[cur] % 2 == 1)
                ret = max(ret, solve(mask | (1 << cur) | (1 << u), u) + 1);
        }
        return ret;
    };
    cout << solve(0, 1);
}
