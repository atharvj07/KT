#include<bits/stdc++.h>
using namespace std;
int main(){
  int n,m;
  cin>>n>>m;
  int a[n+1][m+1];
  for(int i=1;i<=n;i++)
    for(int j=1;j<=m;j++)cin>>a[i][j];
  int ans=0;
  for(int i=1;i<=n;i++)
    for(int j=1;j<=m;j++)
      for(int k=1;k<=n;k++)
        for(int l=1;l<=m;l++)
          if((i==k||j==l)&&(a[i][j]==a[k][l]))ans++;
  cout<<ans<<endl;
  return 0;
}
