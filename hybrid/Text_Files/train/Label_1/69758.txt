import sys
from collections import deque
input = sys.stdin.readline

t = int(input())

for _ in range(t):
    n = int(input())
    p = [int(x) for x in input().split()]

    q = deque()
    ans = []

    for x in p:
        if not q or q[0] > x:
            q.appendleft(x)
        else:
            while q and q[-1] < x:
                ans.append(q.pop())
            q.append(x)

    ans.extend(q)
    print(*ans)
