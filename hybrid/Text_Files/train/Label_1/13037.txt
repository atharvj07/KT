#include <bits/stdc++.h>
using namespace std;
#define int long long
const int md = 1000000007;
const int N = 200005;
int32_t main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    int n, q;
    cin >> n >> q;
    string s;
    cin >> s;
    bool ok = 1;
    for (int i = 1; i < n; i++) {
        ok &= (s[i] != s[i - 1]);
    }
    if (ok) {
        for (int i = 0; i < q; i++) {
            int l, r;
            cin >> l >> r;
            cout << 0 << endl;
        }
    } else {
        vector<int> pre(n + 1), suf(n + 1);
        for (int i = 0; i < n - 1; i++) {
            if (s[i] == s[i + 1]) {
                pre[i + 1] = pre[i] + 1;
            } else {
                pre[i + 1] = pre[i];
            }
        }
        for (int i = n - 1; i >= 1; i--) {
            if (s[i] == s[i - 1]) {
                suf[i - 1] = suf[i] + 1;
            } else {
                suf[i - 1] = suf[i];
            }
        }
        for (int i = 0; i < q; i++) {
            int l, r;
            cin >> l >> r;
            int lft = max(0LL, pre[l - 1]);
            int rgt = max(0LL, suf[r]);
            int ans = lft + rgt;
            ans = max(0LL, ans);
            if ((r - l + 1) % 2 == 0) {
                ans += 1;
            }
            cout << ans << endl;
        }
    }
}
