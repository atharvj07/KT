import sys
n,m=map(int,input().split())
a=[0]*pow(2,n)
for x in input().split():
    a[int(x)]=1
v=0
XOR=False
for x in range(n,0,-1):
    if XOR:
        for i in range(0,len(a),pow(2,x)):
            a[i]=(a[i]+a[i+pow(2,x)-1])%2
    else:
        for i in range(0,len(a),pow(2,x)):
            a[i]=(a[i]|a[i+pow(2,x)-1])
    XOR=not XOR
    v=a[0]
for x in range(m):
    i,y=map(int,sys.stdin.readline().split())
    a[i-1]=y
    XOR=False
    for x in range(n,0,-1):
        if XOR:
            for i in range(0,len(a),pow(2,x)):
                a[i]=(a[i]+a[i+pow(2,x)-1])%2
        else:
            for i in range(0,len(a),pow(2,x)):
                a[i]=(a[i]|a[i+pow(2,x)-1])
        XOR=not XOR
        v=a[0]
    print(v)
