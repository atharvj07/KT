import sys
input = sys.stdin.readline

def solve():
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    d = {}
    ans = 0
    for i in range(n):
        if a[i] % k == 0:
            if a[i] // k in d:
                ans += d[a[i] // k]
        if a[i] not in d:
            d[a[i]] = 0
        d[a[i]] += 1
    return ans

print(solve())
