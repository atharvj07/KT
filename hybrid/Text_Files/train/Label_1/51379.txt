import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long k = in.nextLong();
        long A = in.nextLong();
        long B = in.nextLong();
        long ans = 0;
        for (; n > 1; n %= k) {
            if (k * A < B) {
                n -= 1;
                ans += A;
            } else {
                ans += B;
            }
        }
        System.out.println(ans);
    }
}
