#include <bits/stdc++.h>
using namespace std;
#define MAX 100005
int t[MAX];
int main(){
  int n, ans = 0;
  cin >> n;
  for ( int i = 0; i < n; i++ ) cin >> t[i];
  int l = 0, r = 0, cnt1 = 0, cnt2 = 0;
  while ( r < n ){
    if ( t[r] == 1 ) cnt1++;
    else cnt2++;
    while ( cnt1 > cnt2 ){
      if ( t[l++] == 1 ) cnt1--;
    }
    ans = max(ans, r - l + 1);
    r++;
  }
  cout << ans;
}
