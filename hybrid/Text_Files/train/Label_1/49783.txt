import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        int[] v = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            v[i] = sc.nextInt();
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                long dx = x[j] - x[i];
                long dv = v[j] - v[i];
                if (dv == 0) {
                    continue;
                }
                long t = -dx / dv;
                if (t >= 0) {
                    sum += Math.abs(dx + t * dv);
                }
            }
        }
        System.out.println(sum);
    }
}
