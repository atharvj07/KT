#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n), b(n), off(n, 0);
    string s;
    cin >> s;
    for (int i = 0; i < n; i++) {
        cin >> a[i] >> b[i];
        if (s[i] == '0') off[i] = b[i] - 1;
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int cnt = 0;
        for (int j = 0; j < n; j++) {
            if (i >= off[j] && (i - off[j]) % a[j] == 0) cnt++;
        }
        ans = max(ans, cnt);
    }
    cout << ans << endl;
    return 0;
}
