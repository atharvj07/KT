def guess_number(n):
    l, r = -2e9, 2e9
    for i in range(n):
        sign, x, ans = input().split()
        x = int(x)
        if sign == '>':
            if ans == 'Y':
                l = max(l, x + 1)
            else:
                return "Impossible"
        elif sign == '<':
            if ans == 'Y':
                r = min(r, x - 1)
            else:
                return "Impossible"
        elif sign == '>=':
            if ans == 'Y':
                l = max(l, x)
            else:
                return "Impossible"
        else:
            if ans == 'Y':
                r = min(r, x)
            else:
                return "Impossible"
    if l <= r:
        return str(l)
    else:
        return "Impossible"

n = int(input())
print(guess_number(n))
