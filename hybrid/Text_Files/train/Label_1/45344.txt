import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long m = sc.nextLong();
        long[] a = new long[n];
        a[0] = sc.nextInt();
        long l = a[0], g = a[0];
        for (int i = 1; i < n; i++) {
            a[i] = sc.nextInt();
            g = gcd(a[i], g);
            l = lcm(a[i], l);
        }
        if (l > m) {
            System.out.println(0);
            return;
        }
        long ans = m / (l / g);
        System.out.println(ans);
    }
    public static long gcd(long a, long b) {
        return (b == 0) ? a : gcd(b, a % b);
    }
    public static long lcm(long a, long b) {
        return a * (b / gcd(a, b));
    }
}
