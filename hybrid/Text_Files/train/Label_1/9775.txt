#include <bits/stdc++.h>
#define ll long long
using namespace std;

const ll mxN = 2e5 + 5;
ll n, k, q, burles, sum, ans;
vector<ll> a;
map<ll, ll> dp;

int main() {
    cin >> n >> k;
    for (ll i = 0; i < n; i++) {
        cin >> burles;
        a.push_back(burles);
    }
    cin >> q;
    while (q--) {
        cin >> sum;
        if (sum % a[0] > k) cout << -1 << "\n";
        else if (dp.count(sum))
            cout << dp[sum] << "\n";
        else {
            ans = LLONG_MAX;
            for (ll i = 0; i < n; i++) {
                if (sum < a[i]) break;
                ans = min(ans, 1 + dp[sum - a[i]]);
            }
            dp[sum] = (ans == LLONG_MAX) ? -1 : ans;
            cout << dp[sum] << "\n";
        }
    }
}
