#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e6 + 10;
int n, a[maxn], t[maxn], c[maxn];
vector<pair<int, int>> ans;
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) { t[i] = a[i]; }
  sort(a + 1, a + 1 + n);
  for (int i = 1; i <= n; i++) {
    c[i] = lower_bound(a + 1, a + 1 + n, t[i]) - a;
    if (c[i] != i) { ans.push_back({c[i], i}); }
  }
  cout << ans.size() << endl;
  for (auto [x, y] : ans) { cout << x << " " << y << endl; }
  return 0;
}
