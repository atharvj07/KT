import sys
input = sys.stdin.readline

n = int(input())
s = [int(x) for x in input().split()]
c = [int(x) for x in input().split()]
ans = -1
for i in range(n):
    for j in range(i + 1, n):
        for k in range(j + 1, n):
            if s[i] < s[j] < s[k]:
                ans = min(ans, c[i] + c[j] + c[k]) if ans != -1 else c[i] + c[j] + c[k]
print(ans)
