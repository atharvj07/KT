#include<bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 998244353;
int32_t main() {
  int n;
  cin >> n;
  vector<int> a(n);
  bool ok = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (i) { a[i] += a[i - 1]; }
    if (a[i] >= n) { ok = 1; }
  }
  a.insert(a.begin(), 0);
  int ans = 0;
  for (int l = 0; l <= n; l++) {
    for (int r = l; r <= n; r++) {
      int s = l ? a[r - 1] - a[l - 1] : a[r - 1];
      if (s >= r - l && s + 1 == n - r + l) { (ans += ok || r == n ? 1 : mod - 1) %= mod; }
    }
  }
  cout << ans;
}
