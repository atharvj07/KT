#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    vector<int>a(n+1),freq(n+1),ans(n+1);
    map<int,int>m;
    for(int i=1; i<=n; i++){
        cin>>a[i];
        freq[a[i]]++;
        if(freq[a[i]]>freq[m[i]] or (freq[a[i]]==freq[m[i]] and a[i]<m[i]))
            m[i]=a[i];
    }
    for(int i=1; i<=n; i++){
        ans[m[i]]++;
    }
    for(int i=1; i<=n; i++){
        cout<<ans[i]<<" ";
    }
}
