#include <bits/stdc++.h>
#include <ext/pb_ds/assoc_container.hpp>
using namespace __gnu_pbds;
using namespace std;
#define int long long
#define pb push_back
const int mx=1e5+10;
int n,s[mx],t[mx];
typedef tree<int,null_type,less<int>,rb_tree_tag,tree_order_statistics_node_update> rn;
vector<rn> v;

void build()
{
    v.pb(*new rn());
    for(int i=1;i<=n;i++)
    {
        v.pb(*new rn());
        v[i].insert(s[i]);
    }
}

int get(int i, int l, int r)
{
    int x=*v[i].lower_bound(l);
    x-=*v[i].lower_bound(r+1);
    return x;
}

signed main()
{
    cin>>n;
    for(int i=1;i<=n;i++)
    {
        cin>>s[i];
        t[i]=s[i];
    }
    sort(t+1,t+1+n);
    
    build();
    int m;
    cin>>m;
    while(m--)
    {
        int x,y,z;
        cin>>x>>y>>z;
        if(x==1)
        {
            int ans=0;
            for(int i=y;i<=z;i++) ans+=s[i];
            cout<<ans<<'\n';
        }
        else
        {
            int ans=0;
            for(int i=y;i<=z;i++) ans+=t[get(i,y,z)];
            cout<<ans<<'\n';
        }
    }
}
