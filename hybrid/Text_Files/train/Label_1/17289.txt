#include <cstdio>
#include <algorithm>
#include <utility>

int main() {
	int n, x[4], y[4], r[4], ans, l[4], u[4], v[4];
	scanf("%d", &n);
	for (int i = 1; i <= n; ++i)
		scanf("%d%d%d", &x[i], &y[i], &r[i]);
	for (int i = 1; i <= n; ++i) {
		for (int j = 1; j <= n; ++j) {
			if (i == j)
				continue;
			l[i] = x[i] - r[i] - x[j] + r[j];
			u[i] = x[i] + r[i] - x[j] - r[j];
			v[i] = std::min(r[i], r[j]);
			if (l[i] > 0 || u[i] < 0 || v[i] * v[i] < l[i] * l[i] + u[i] * u[i])
				++ans;
		}
	}
	printf("%d", ans + n + 1);
	return 0;
}
