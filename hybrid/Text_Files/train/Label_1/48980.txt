import java.util.Scanner;

public class DotGame {

    private static final int MAX_COORDINATE = 200;
    private static final int MAX_VECTORS = 20;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int y = scanner.nextInt();
        int n = scanner.nextInt();
        int d = scanner.nextInt();

        int[][] vectors = new int[n][2];
        for (int i = 0; i < n; i++) {
            vectors[i][0] = scanner.nextInt();
            vectors[i][1] = scanner.nextInt();
        }

        boolean antonWins = isAntonWin(x, y, vectors, d);
        System.out.println(antonWins ? "Anton" : "Dasha");
    }

    private static boolean isAntonWin(int x, int y, int[][] vectors, int d) {
        if (x * x + y * y > d * d) {
            return false; // Anton already lost
        }

        for (int[] vector : vectors) {
            int newX = x + vector[0];
            int newY = y + vector[1];
            if (newX * newX + newY * newY > d * d) {
                return true; // Anton can win with this move
            }
        }

        // Anton can't win with any move, so Dasha can win
        return false;
    }
}
