#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int a, b, c, m;
  cin >> a >> b >> c >> m;
  struct mouses { int cost; string type; };
  vector<mouses> mouse(m);
  for (auto &i : mouse) cin >> i.cost >> i.type;
  sort(mouse.begin(), mouse.end(), [](mouses x, mouses y) { return x.cost < y.cost; });
  long long ans = 0; int cnt = 0;
  for (auto i : mouse) {
    if (i.type == "USB" && a > 0) { --a; ++cnt; ans += i.cost; }
    else if (i.type == "PS/2" && b > 0) { --b; ++cnt; ans += i.cost; }
    else if (c > 0) { --c; ++cnt; ans += i.cost; }
    else break;
  }
  cout << cnt << " " << ans << endl;
  return 0;
}
