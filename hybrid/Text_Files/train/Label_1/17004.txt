#include <bits/stdc++.h>

using namespace std;
#define ll long long
const int N = 1e6 + 5;

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, m, x, y, a, b;
    cin >> n >> m >> x >> y >> a >> b;
    if ((x + a <= n && y - b >= 1) || (x + a <= n && y + b <= m) ||
        (x - a >= 1 && y - b >= 1) || (x - a >= 1 && y + b <= m))
        cout << 1 << '\n';
    else if ((x + 2 * a <= n && y - 2 * b >= 1) ||
             (x + 2 * a <= n && y + 2 * b <= m) ||
             (x - 2 * a >= 1 && y - 2 * b >= 1) ||
             (x - 2 * a >= 1 && y + 2 * b <= m))
        cout << 2 << '\n';
    else
        cout << "Poor Inna and pony!\n";
    return 0;
}
