from collections import Counter

for t in range(int(input())):
    n = int(input())
    arr = Counter(int(x) for x in input().split())
    even = sum(list(arr[x] for x in arr if x % 2 == 0))
    odd = sum(list(arr[x] for x in arr if x % 2 == 1))
    if even == odd: print("Tie")
    elif even > odd: print("Alice")
    else: print("Bob")
