#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < n; ++i)
using namespace std;
int n, a[50], b[50];

int solve() {
  rep(i, n) b[i] = a[i] & 1;
  int cnt = 0;
  rep(i, n - 1) {
    if (b[i] != (i & 1)) {
      int j;
      for (j = i + 1; j < n; ++j) {
        if (b[j] == (i & 1)) {
          ++cnt;
          swap(b[i], b[j]);
          break;
        }
      }
      if (j == n) return -1;
    }
  }
  if (b[n - 1] != ((n - 1) & 1)) return -1;
  return cnt;
}

int main() {
  int t;
  cin >> t;
  rep(i, t) {
    cin >> n;
    rep(i, n) cin >> a[i];
    cout << solve() << endl;
  }
  return 0;
}
