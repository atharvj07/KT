import sys
input = sys.stdin.readline

n= int(input())
a, c, d, x = [0]*n, 0, 0, 0
for i in range(n):
    b=list(map(int,input().split()))
    if b[0]==1: a[c:c+b[1]]=[d]*b[1]; c+=b[1]; x+=b[1]*b[2]
    elif b[0]==2: x+=b[1]; c+=1; d=b[1]
    else: x-=d; c-=1; d=a[c]
    print(float("{0:.6f}".format(x/c)))
