n, a, b = map(int, input().split())
arthur_likes = set(map(int, input().split()))
alexander_likes = set(map(int, input().split()))

result = [0] * n
for i in range(1, n + 1):
    if i in arthur_likes:
        result[i - 1] = 1
    elif i in alexander_likes:
        result[i - 1] = 2

print(' '.join(map(str, result)))
