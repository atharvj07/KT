import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t1 = sc.nextInt();
        int t2 = sc.nextInt();
        int x1 = sc.nextInt();
        int x2 = sc.nextInt();
        int t0 = sc.nextInt();

        int delta = t2 - t0;
        if (x1 * delta >= x2 * (t2 - t1)) {
            System.out.println(0 + " " + x2);
        } else {
            int y1 = (t2 - t0) * x1 / (t2 - t1);
            int y2 = x2 - y1;
            System.out.println(y1 + " " + y2);
        }
    }
}
