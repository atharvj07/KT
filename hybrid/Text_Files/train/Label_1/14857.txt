#include<bits/stdc++.h>
using namespace std;
int main() {
    long long int n,a[1005],sum[1005]={0},check=1;
    cin>>n;
    for(int i=1;i<=n;i++) {
        cin>>a[i];
    }
    sort(a+1,a+n+1);
    sum[1]=a[1];
    sum[2]=a[1]+a[2];
    for(int i=3;i<=n;i++) {
        if(sum[i-1]+sum[i-2]==a[i]-1 || sum[i-1]+sum[i-2]==a[i]-2) {
            sum[i]=a[i];
            check=1;
        }
        else {
            check=0;
            break;
        }
    }
    if(check) cout<<n<<endl;
    else {
        n--;
        for(int i=3;i<=n;i++) {
            if(sum[i-1]+sum[i-2]==a[i]-1 || sum[i-1]+sum[i-2]==a[i]-2) {
                sum[i]=a[i];
                check=1;
            }
            else {
                check=0;
                break;
            }
        }
        if(check) cout<<n<<endl;
        else cout<<n-1<<endl;
    }
    return 0;
}
