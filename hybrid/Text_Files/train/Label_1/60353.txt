def min_operations(n, p):
    if p == sorted(p):
        return 0

    for i in range(2):
        for _ in range(n):
            if i == 0:
                p = [p[j+1] if j % 2 == 0 else p[j-1] for j in range(2*n)]
            else:
                p = p[n:] + p[:n]
            if p == sorted(p):
                return i + 1 + _

    return -1

n = int(input().strip())
p = list(map(int, input().strip().split()))

print(min_operations(n, p))
