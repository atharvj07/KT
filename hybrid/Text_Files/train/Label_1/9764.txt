#include <iostream>
#include <vector>
#include <cstring>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    char name[n + 1];
    cin >> name;
    vector<pair<char, char>> changes;
    for (int i = 0; i < m; i++) {
        char x, y;
        cin >> x >> y;
        changes.push_back({x, y});
    }
    for (int i = m - 1; i >= 0; i--) {
        for (int j = 0; j < n; j++) {
            if (name[j] == changes[i].first) {
                name[j] = changes[i].second;
            } else if (name[j] == changes[i].second) {
                name[j] = changes[i].first;
            }
        }
    }
    cout << name << endl;
    return 0;
}
