import java.util.*;

public class AlarmClocks {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[] times = new int[n];
        for (int i = 0; i < n; i++) {
            times[i] = scanner.nextInt();
        }

        Arrays.sort(times);

        int minToTurnOff = 0;
        int maxConsecutive = 0;

        for (int i = 0; i < n; i++) {
            int j = i + 1;
            while (j < n && times[j] - times[i] < m) {
                j++;
            }
            maxConsecutive = Math.max(maxConsecutive, j - i);
            if (maxConsecutive >= k) {
                break;
            }
        }

        minToTurnOff = n - maxConsecutive;

        System.out.println(minToTurnOff);
    }
}
