from collections import Counter
n=int(input());a=list(map(int,input().split()));c=Counter(a);f=0;b=a.copy();x,y=0,0
for i in range(n):
    if a[i]==i:f+=1
    if a[i]>i:
        x=a[i];y=i
        if y in c:c[y]-=1
        if x in c:c[x]-=1
for i,j in c.items():
    if j>0:f+=1
    if j==2:f+=1
if x>y and y in c:f+=1
print(f)
