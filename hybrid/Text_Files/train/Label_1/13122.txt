#include <bits/stdc++.h>
using namespace std;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    int flag = 1;
    for (int i = 0; i < n; i++) {
      if ((a[i] == '3' || a[i] == '4') &&
          (b[i] == '1' || b[i] == '2')) {
        flag = 0;
      }
      if ((a[i] == '1' || a[i] == '2') &&
          (b[i] == '3' || b[i] == '4')) {
        flag = 0;
      }
      if ((a[i] == '5' || a[i] == '6') &&
          (b[i] == '1' || b[i] == '2')) {
        flag = 0;
      }
      if ((a[i] == '1' || a[i] == '2') &&
          (b[i] == '5' || b[i] == '6')) {
        flag = 0;
      }
    }
    if (flag == 1) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
}
