#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main()
{
    int n,k;
    cin>>n>>k;
    vector<ll>a(n),dp(k+1);
    for(int i=0;i<n;i++)cin>>a[i];
    dp[0]=1;
    for(int i=0;i<n;i++)
    {
        for(int j=k;j>=a[i];j--)
        {
            dp[j]=dp[j] | dp[j-a[i]];
        }
    }
    int ans=0;
    for(int i=0;i<=k;i++)
    if(dp[i]) {ans++;}
    cout<<ans<<'\n';
    for(int i=0;i<=k;i++)
    {
        if(dp[i])cout<<i<<' ';
    }
    return 0;
}
