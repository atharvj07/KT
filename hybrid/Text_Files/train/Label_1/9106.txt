#include <iostream>

using namespace std;

int main() {
    long long n, k;
    cin >> n >> k;
    if (n % (k + 1) != 0) {
        cout << "0 0 " << n << endl;
    } else {
        long long diplomas = n / (k + 1);
        cout << diplomas << " " << diplomas * k << " " << n - diplomas * (k + 1) << endl;
    }
    return 0;
}
