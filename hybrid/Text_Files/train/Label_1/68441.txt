n, k, p = map(int, input().split())
a = sorted(list(map(int, input().split())))
b = sorted(list(map(int, input().split())))
ans = 0
i = j = 0
while i < n and j < k:
    if abs(a[i] - b[j]) < abs(p - b[j]):
        ans += abs(a[i] - b[j])
        j += 1
    else:
        ans += abs(p - b[j])
        i += 1
        j += 1
print(ans)
