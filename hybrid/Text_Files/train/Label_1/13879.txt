#include <bits/stdc++.h>
#define pb push_back
using namespace std;
int main() {
  int n, ans = 0;
  cin >> n;
  vector<int> ha, ga;
  for (int i = 0; i < n; i++) {
    int h, g;
    cin >> h >> g;
    ha.pb(h);
    ga.pb(g);
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i == j) continue;
      if (ha[i] == ga[j]) { ans++; }
    }
  }
  cout << ans << endl;
  return 0;
}
