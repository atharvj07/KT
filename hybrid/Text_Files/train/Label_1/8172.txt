#include <bits/stdc++.h>
using namespace std;
vector<int> a;
int main() {
    int n, m, w;
    cin >> n >> m >> w;
    a.resize(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    int ans = 0, cnt = w;
    for (int i = 0; i < n; i++) {
        if (cnt > 0) { ans = max(ans, a[i]), cnt--; }
    }
    for (int i = 0; i < n; i++) { a[i] = max(ans, a[i]); }
    while (m--) {
        ans = max(ans, a[0] + m);
        for (int i = 1; i < n; i++) { a[i] = max(ans, a[i]); }
    }
    cout << ans << endl;
    return 0;
}
