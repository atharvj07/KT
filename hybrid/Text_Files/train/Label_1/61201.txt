n, q = map(int, input().split())
arr = list(map(int, input().split()))
for _ in range(q):
    c, x = input().split()
    if c == '>':
        for i in range(n):
            if arr[i] > int(x):
                arr[i] = -arr[i]
    else:
        for i in range(n):
            if arr[i] < int(x):
                arr[i] = -arr[i]
for a in arr:
    print(a, end=' ')
