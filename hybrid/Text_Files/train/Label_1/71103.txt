n, m = map(int, input().split())
l = list(map(int, input().split()))
if sum(l) > n:
    print(-1)
else:
    ans = []
    for i in range(m - 1, -1, -1):
        if i == 0 or ans[-1] - l[i] >= l[i - 1]:
            ans.append(ans[-1] - l[i] + 1)
        else:
            ans.append(l[i] - l[i - 1] + ans[-1] + 1)
    print(*reversed(ans))
