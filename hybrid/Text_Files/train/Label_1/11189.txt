#include<bits/stdc++.h>
using namespace std;
const int N = 2e5+9;
string s;
int n, k, p[N], q[N];
int main() {
    cin >> n >> k;
    for(int i = 0;i < n;i++) {
        cin >> p[i];
        p[i]--;
    }
    for(int i = 0;i < n;i++) {
        cin >> q[i];
        q[i]--;
    }
    for(char a = 'a';a <= 'z';a++) {
        vector<char> v(n);
        for(int i = 0;i < n;i++) {
            if(p[i] > q[i]) v[p[i]] = v[q[i]] = a;
            else v[q[i]] = v[p[i]] = a;
        }
        int cnt = 0;
        for(int i = 0;i < n;i++) {
            if(v[i] != v[i-1]) cnt++;
            if(cnt == k) {
                for(int i = 0;i < n;i++) s += v[i];
                cout << "YES\n" << s << '\n';
                return 0;
            }
        }
    }
    cout << "NO\n";
    return 0;
}
