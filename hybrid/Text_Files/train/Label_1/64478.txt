from collections import deque, defaultdict

n, k = map(int, input().split())
a = set(int(x) for x in input().split())
adj = defaultdict(list)
for i in range(1, n + 1):
    l = list(map(int, input().split()))[1:]
    adj[i] = l
in_degree = [0] * (n + 1)
for l in adj.values():
    for x in l:
        in_degree[x] += 1
q = deque([x for x in a if in_degree[x] == 0])
ans = []
while q:
    x = q.popleft()
    ans.append(x)
    for y in adj[x]:
        in_degree[y] -= 1
        if in_degree[y] == 0:
            q.append(y)
if len(ans) != k:
    print(-1)
else:
    print(len(ans))
    print(*ans)
