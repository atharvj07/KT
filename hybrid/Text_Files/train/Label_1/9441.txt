#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 10;
int p[MAXN];

int main() {
    int n, q, a[MAXN];
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < a[i - 1]; j++) { p[j + a[i - 1]] = i; }
    }
    cin >> q;
    for (int i = 0; i < q; i++) {
        int x;
        cin >> x;
        cout << p[x - 1] << endl;
    }
    return 0;
}
