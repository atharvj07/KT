n = int(input())
s = list(map(int, input().split()))
c = list(map(int, input().split()))

min_cost = float('inf')

for j in range(1, n-1):
    min_left = float('inf')
    for i in range(j):
        if s[i] < s[j]:
            min_left = min(min_left, c[i])
    min_right = float('inf')
    for k in range(j+1, n):
        if s[j] < s[k]:
            min_right = min(min_right, c[k])
    if min_left != float('inf') and min_right != float('inf'):
        min_cost = min(min_cost, c[j] + min_left + min_right)

if min_cost == float('inf'):
    print(-1)
else:
    print(min_cost)
