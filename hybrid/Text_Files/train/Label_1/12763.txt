#include <iostream>
#include <vector>
using namespace std;

const int MOD = 1e9 + 7;

typedef long long ll;

int power(int n, int exp) {
    int ret = 1;
    while (exp > 0) {
        if (exp & 1) {
            ret = 1ll * ret * n % MOD;
        }
        n = 1ll * n * n % MOD;
        exp >>= 1;
    }
    return ret;
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    vector<int> freq(71, 0);
    for (int i = 1; i <= n; i++) {
        freq[a[i]]++;
    }
    int ans = 0;
    for (int i = 1; i <= 70; i++) {
        ans = (ans + 1ll * freq[i] * (freq[i] - 1) / 2 % MOD * power(2, freq[i] - 2) % MOD) % MOD;
    }
    cout << ans << endl;
    return 0;
}
