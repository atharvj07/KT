#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n);

    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    long long sum = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i];
    }
    if (sum != 0) {
        cout << -1 << endl;
        return 0;
    }

    long long ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] < 0) {
            ans++;
            a[i] = -a[i];
            a[(i + 1) % n] -= a[i];
            a[(i - 1 + n) % n] -= a[i];
        }
    }

    cout << ans << endl;

    return 0;
}
