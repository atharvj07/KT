import sys
input = sys.stdin.readline
t = int(input())

def func(s, l, r):
    x = 0
    st = set()
    for i in range(l - 1, r):
        if s[i] == '+':
            x += 1
        else:
            x -= 1
        st.add(x)
    return len(st)

for i in range(t):
    n, m = map(int, input().split())
    s = input().strip()
    for j in range(m):
        l, r = map(int, input().split())
        print(func(s, l, r))
