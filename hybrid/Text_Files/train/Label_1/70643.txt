n, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
low = 1
high = k // len(a)
while low < high:
    mid = (low + high) // 2
    tmp = 0
    for x in a:
        tmp += max(0, (x - 1) // mid)
    if tmp <= k:
        low = mid + 1
    else:
        high = mid
print(low - 1)
