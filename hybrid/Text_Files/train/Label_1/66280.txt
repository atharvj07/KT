from sys import stdin
input=stdin.readline
n,k=map(int,input().split())
a=list(map(int,input().split()))

ansl,ansr=1,1
cur=1
Freq={}
Freq[a[0]]=1
for i in range(1,n):
    if a[i] in Freq and Freq[a[i]]+1<=k: Freq[a[i]]+=1
    else: Freq[a[i]]=1
    if len(Freq)<=k: ansr=i
    else:
        while len(Freq)>k:
            Freq[a[cur]]-=1
            if Freq[a[cur]]==0: del Freq[a[cur]]
            cur+=1
        ansr=i
    if ansr-ansl+1>cur-1: ansl,ansr=cur,ansr
print(ansl,ansr)
