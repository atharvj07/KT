#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
    ll n; cin>>n;
    vector<ll> cnt(9); for(int i=1;i<=8;++i) cin>>cnt[i];
    ll mx = 0;
    for(int i=1;i<=8;++i){
        ll val = cnt[i];
        if(n-i>=0) val+=cnt[i]*(n/i);
        mx = max(mx,val*i);
    }
    cout<<mx<<endl;
}
