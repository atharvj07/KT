import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), g = sc.nextInt(), r = sc.nextInt();
        long[] l = new long[n + 2];
        for (int i = 0; i < n + 1; i++) {
            l[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            long t = sc.nextInt();
            long cur = t % (g + r);
            if (cur < g) {
                cur += g;
            }
            for (int j = 1; j < n + 2; j++) {
                cur += l[j];
                cur = (cur + (g + r) - 1) / (g + r) * (g + r);
            }
            System.out.println(cur);
        }
    }
}
