from collections import Counter

k = 0
b = 0
c = 0

s = Counter()

for _ in range(int(input())):
    x, y, z = map(int, input().split())
    k = max(x, y, z, k)
    b = max(x * y, y * z, z * x, b)
    c = max(min(x, y), min(y, z), min(z, x), c)
    s.update([x*y, y*z, z*x])

if b == c == k: print(1, s.most_common(1)[0][0])
else: print(2, s.most_common(2)[0][0], s.most_common(2)[1][0])
