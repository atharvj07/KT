#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long l, r;
        cin >> l >> r;
        if (l % 2 == 0 && l != r)
            cout << l << ' ' << l * 2 << '\n';
        else
            cout << l + 1 << ' ' << (l + 1) * 2 << '\n';
    }
    return 0;
}
