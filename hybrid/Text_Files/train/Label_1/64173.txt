import sys

def solve(n, m, d, c):
    if n == 0:
        return ["YES", "0"]
    if n < d:
        return ["NO"]
    pos = [0] * n
    for i in range(m):
        for j in range(c[i]):
            pos[j] = i + 1
    if pos[n - 1] == 0:
        return ["NO"]
    ans = [0]
    for i in range(1, n):
        if pos[i] == 0:
            if pos[i - 1] == ans[-1]:
                return ["NO"]
            ans.append(0)
        else:
            if ans[-1] == pos[i] and i < n - d:
                return ["NO"]
            ans.append(pos[i])
    return ["YES", " ".join(map(str, ans))]

n, m, d = map(int, input().split())
c = list(map(int, input().split()))
print(*solve(n, m, d, c))
