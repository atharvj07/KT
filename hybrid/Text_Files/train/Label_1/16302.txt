#include <bits/stdc++.h>
using namespace std;
#define MAX 200005
int power[MAX], ans[MAX];
bool vis[MAX];
vector<int> adj[MAX];
void dfs(int node) {
  vis[node] = true;
  for (auto u : adj[node]) {
    if (!vis[u]) dfs(u);
    ans[node] = max(ans[node], 1 + ans[u]);
  }
  power[node] += ans[node];
}
int main() {
  int n, m, q;
  cin >> n >> m;
  cin >> q;
  for (int i = 0; i < q; i++) {
    int a, b;
    cin >> a >> b;
    adj[a].push_back(b);
  }
  dfs(1);
  cout << power[n] + power[m] << endl;
  return 0;
}
