#include <iostream>
#include <vector>
#include <bitset>

const int mxN = 4e5;

std::vector<int> in[mxN+1], out[mxN+1];
std::string ans;
std::bitset<mxN+1> vis;

void dfs(int u, int p) {
    if (!vis[u]) {
        vis[u] = 1;
        for (int v : out[u]) {
            if (v != p) {
                dfs(v, u);
                if (!ans.empty()) return;
            }
        }
    } else {
        if (in[u].size() > 1) ans = "-1";
        else if (p != -1) ans = "2";
    }
}

int main() {
    int t; std::cin >> t;
    while(t--) {
        ans.clear();
        int n, m; std::cin >> n >> m;
        vis.reset();
        for (int i = 1; i <= n; ++i)
            in[i].clear(), out[i].clear();
        while(m--) {
            int u, v; std::cin >> u >> v;
            out[u].push_back(v);
            in[v].push_back(u);
        }
        dfs(1, -1);
        for (int i = 1; i <= n; ++i) {
            if (!vis[i]) ans += '0';
            else if (in[i].size() == 0) ans += '1';
            else if (in[i].size() == 1) ans += '1';
            else if (ans.empty()) ans = "-1";
        }
        std::cout << ans << '\n';
    }
    return 0;
}
