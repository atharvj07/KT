# include <bits/stdc++.h>
using namespace std ;
typedef long long LL ;
const int MOD = 1e9 ;
const int N = 2e5 + 5 ;
LL A[N] , F[N] ;
int main( ) {
  int n , q , t , x , y , z ;
  cin >> n >> q ;
  for( int i = 0 ; i < n ; i++ ) cin >> A[ i ] ;
  F[ 0 ] = F[ 1 ] = 1 ;
  for( int i = 2 ; i <= q ; i++ ) F[ i ] = ( F[ i - 1 ] + F[ i - 2 ] ) % MOD ;
  while( q -- ) {
    cin >> t ;
    if( t == 1 ) {
      cin >> x >> y ;
      x -- ;
      A[ x ] = y ;
    }
    else if( t == 2 ) {
      cin >> x >> y ;
      x -- , y -- ;
      LL sum = 0 ;
      for( int i = 0 ; i <= y - x ; i++ ) {
        sum = ( sum + ( A[ x + i ] * F[ i ] ) % MOD ) % MOD ;
      }
      cout << sum << "\n" ;
    }
    else {
      cin >> x >> y >> z ;
      x -- , y -- ;
      for( int i = x ; i <= y ; i++ ) {
        A[ i ] = ( A[ i ] + z ) % MOD ;
      }
    }
  }
  return 0 ;
}
