import sys

s = sys.stdin.readline().strip()
t = sys.stdin.readline().strip()
s = list(s)
t = list(t)

cnt_s = [0] * 26
cnt_t = [0] * 26
for i in range(len(s)):
    if s[i] != '?':
        cnt_s[ord(s[i]) - ord('a')] += 1
        cnt_t[ord(t[i]) - ord('a')] += 1

for i in range(len(s)):
    if s[i] == '?':
        for j in range(26):
            cnt_s[j] += 1
            tmp = cnt_s.copy()
            ok = True
            for k in range(len(t)):
                if tmp[ord(t[k]) - ord('a')] == 0:
                    ok = False
                    break
                tmp[ord(t[k]) - ord('a')] -= 1
            if ok:
                s[i] = chr(ord('a') + j)
                break
            cnt_s = tmp.copy()

print(''.join(s))
