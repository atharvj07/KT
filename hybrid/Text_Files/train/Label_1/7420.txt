#include <bits/stdc++.h>
using namespace std;

const int maxn = 2e5 + 10, MOD = 1e9 + 7;
vector<int> f(maxn, 0);
int vis[maxn];

int main() {
  int n; cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> f[i];
    if (f[i] == 0) {
      vis[i] = 1;
    }
  }
  vector<int> v;
  for (int i = 1; i <= n; i++) {
    if (f[i]) {
      v.push_back(i);
    }
  }
  int cur = v.size();
  for (int i = 1; i <= n; i++) {
    if (f[i] == 0) {
      while (vis[cur]) cur++;
      f[i] = cur++;
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << f[i] << " ";
  }
  return 0;
}
