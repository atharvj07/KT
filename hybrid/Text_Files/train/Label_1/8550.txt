#include <iostream>
#include <vector>
#include <map>

using namespace std;

int main() {
  int t;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int n;
    cin >> n;
    vector<string> users(n);
    map<string, bool> mentioned;
    for (int j = 0; j < n; j++) {
      cin >> users[j];
      mentioned[users[j]] = false;
    }
    int m;
    cin >> m;
    vector<string> messages(m);
    for (int j = 0; j < m; j++) {
      string msg;
      getline(cin, msg);
      messages[j] = msg;
      int pos = msg.find(':');
      if (pos == -1) continue;
      string user = msg.substr(0, pos);
      for (int k = 0; k < n; k++) {
        if (users[k] == user) mentioned[user] = true;
      }
    }
    bool possible = true;
    for (int j = 0; j < m; j++) {
      int pos = messages[j].find(':');
      if (pos == -1) {
        for (int k = 0; k < n; k++) {
          if (!mentioned[users[k]]) {
            messages[j] = users[k] + ":" + messages[j];
            mentioned[users[k]] = true;
            break;
          }
        }
        if (pos == -1) {
          possible = false;
          break;
        }
      }
    }
    if (possible) {
      for (int j = 0; j < m; j++) {
        cout << messages[j] << endl;
      }
    } else {
      cout << "Impossible" << endl;
    }
  }
  return 0;
}
