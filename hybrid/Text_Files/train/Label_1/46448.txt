import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }

            int l = 0, r = n - 1;
            while (l + 2 < r) {
                int m = (l + r) / 2;
                if (a[m] * 2 > a[l] + a[r]) {
                    r = m;
                } else {
                    l = m;
                }
            }

            if (a[l] + a[l + 1] > a[r] && a[l + 1] + a[r] > a[l] && a[r] + a[l] > a[l + 1]) {
                System.out.println(-1);
            } else {
                System.out.println((l + 1) + " " + (l + 2) + " " + (r + 1));
            }
        }
    }
}
