import java.util.Scanner;

public class FloodRisk {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int w = scanner.nextInt();
        int d = scanner.nextInt();
        int n = scanner.nextInt();

        int[][] altitudes = new int[w + 1][d + 1];

        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            int z = scanner.nextInt();
            altitudes[x][y] = z;
        }

        for (int i = 1; i <= w; i++) {
            for (int j = 1; j <= d; j++) {
                if (altitudes[i][j] == 0) {
                    if (altitudes[i - 1][j] != 0) {
                        altitudes[i][j] = altitudes[i - 1][j] - 1;
                    } else if (altitudes[i][j - 1] != 0) {
                        altitudes[i][j] = altitudes[i][j - 1] - 1;
                    } else if (altitudes[i + 1][j] != 0) {
                        altitudes[i][j] = altitudes[i + 1][j] + 1;
                    } else if (altitudes[i][j + 1] != 0) {
                        altitudes[i][j] = altitudes[i][j + 1] + 1;
                    } else {
                        System.out.println("No");
                        return;
                    }
                }
            }
        }

        int totalAltitude = 0;
        for (int i = 1; i <= w; i++) {
            for (int j = 1; j <= d; j++) {
                totalAltitude += altitudes[i][j];
            }
        }

        System.out.println(totalAltitude);
    }
}
