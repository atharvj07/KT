#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int cnt[300] = {0};
  for (char c : s) ++cnt[c];
  if (cnt['B'] >= 1 && cnt['u'] >= 1 && cnt['l'] >= 2 && cnt['b'] >= 1 &&
      cnt['a'] >= 1 && cnt['s'] >= 1 && cnt['a'] >= 1 && cnt['u'] >= 1 &&
      cnt['r'] >= 1) {
    int x = min(cnt['B'], min(min(min(cnt['u'], cnt['l']), cnt['b']),
                                 min(min(cnt['a'], cnt['s']), cnt['a'])));
    cout << x << endl;
  } else {
    cout << 0 << endl;
  }
}
