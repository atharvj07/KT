#include <bits/stdc++.h>
#define int long long
using namespace std;
int n, m, a[55][55];
bool f[55][55];
void print(int x, int y) {
  cout << x << " " << y << " ";
}
void solve() {
  cin >> n >> m;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      cin >> a[i][j];
    }
  }
  int ans = 0;
  for (int i = 1; i < n; ++i) {
    for (int j = 1; j < m; ++j) {
      if (a[i][j] && !f[i][j] && !f[i][j + 1] && !f[i + 1][j] &&
          !f[i + 1][j + 1]) {
        ++ans;
        print(i, j);
        f[i][j] = f[i][j + 1] = f[i + 1][j] = f[i + 1][j + 1] = 1;
      } else if (!a[i][j] && f[i][j] && f[i][j + 1] && f[i + 1][j] &&
                 f[i + 1][j + 1]) {
        ++ans;
        print(i, j);
        f[i][j] = f[i][j + 1] = f[i + 1][j] = f[i + 1][j + 1] = 0;
      }
    }
  }
  cout << ans << endl;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      if (a[i][j] != f[i][j]) {
        cout << -1 << endl;
        return;
      }
    }
  }
}
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  solve();
  return 0;
}
