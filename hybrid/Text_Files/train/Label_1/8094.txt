#include<bits/stdc++.h>
using namespace std;
#define int long long
int n, t;
string s;
int pos(char c) {
	return c - 'a';
}
int p(string &s, int l, int r) {
	if (l + 1 == r)return (1LL << pos(s[l]));
	int res = 0;
	while (l < r) {
		int m = l + rand() % (r - l);
		res -= p(s, l, m);
		l = m + 1;
	}
	return res;
}
signed main() {
	cin >> n >> t;
	cin >> s;
	if (p(s, 0, n) == t)cout << "Yes";
	else cout << "No";
	return 0;
}
