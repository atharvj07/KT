import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int k = input.nextInt();
        int[] digits = new int[k];
        for (int i = 0; i < k; i++) {
            digits[i] = input.nextInt();
        }
        int[] chosen = new int[k];
        int cnt = 0;
        for (int i = 0; i < k; i++) {
            if (digits[i] % 10 == 0) {
                chosen[cnt++] = digits[i];
            }
        }
        for (int i = 0; i < k; i++) {
            if (digits[i] % 10 != 0) {
                chosen[cnt++] = digits[i];
            }
        }
        System.out.println(cnt);
        for (int i = 0; i < cnt; i++) {
            System.out.print(chosen[i] + " ");
        }
        System.out.println();
    }
}
