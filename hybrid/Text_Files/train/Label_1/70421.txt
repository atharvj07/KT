def max_correct_subsequence(s, queries):
    n = len(s)
    open_bracket = [0] * (n + 1)
    close_bracket = [0] * (n + 1)
    for i in range(1, n + 1):
        open_bracket[i] = open_bracket[i - 1]
        close_bracket[i] = close_bracket[i - 1]
        if s[i - 1] == '(':
            open_bracket[i] += 1
        else:
            close_bracket[i] += 1
    
    for l, r in queries:
        open_count = open_bracket[r] - open_bracket[l - 1]
        close_count = close_bracket[r] - close_bracket[l - 1]
        min_count = min(open_count, close_count)
        print(min_count * 2)

# Read input
s = input().strip()
m = int(input().strip())
queries = [tuple(map(int, input().split())) for _ in range(m)]

max_correct_subsequence(s, queries)
