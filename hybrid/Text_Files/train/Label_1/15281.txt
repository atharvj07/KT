#include<bits/stdc++.h>
using namespace std;
const int MAXN = 50;
bool vis[MAXN];
int main() {
    int t, n, a[MAXN];
    cin >> t;
    while(t--) {
        cin >> n;
        for(int i = 0; i < n; i++) {
            cin >> a[i];
        }
        bool ok = true;
        for(int i = 0; i < n; i++) {
            if(vis[i]) continue;
            int last = a[i];
            int mx = 0;
            for(int j = i; j < n; j++) {
                if(vis[j]) continue;
                if(abs(last - a[j]) > 1) {
                    ok = false;
                    break;
                }
                vis[j] = true;
                mx = max(mx, a[j]);
                last = a[j];
            }
            if(mx > last) ok = false;
        }
        memset(vis, false, sizeof vis);
        puts(ok ? "YES" : "NO");
    }
    return 0;
}
