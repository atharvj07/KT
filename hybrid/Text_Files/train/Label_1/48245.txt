import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int x = sc.nextInt();
        long[][] dp = new long[n + 1][m + 1];

        dp[1][x] = 1;
        for (int i = 2; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                for (int k = j; k <= m; k++) {
                    dp[i][j] = (dp[i][j] + dp[i - 1][k]) % 1000000007;
                }
            }
        }

        long ans = 0;
        for (int i = 1; i <= m; i++) {
            ans = (ans + dp[n][i]) % 1000000007;
        }
        System.out.println(ans);
    }
}
