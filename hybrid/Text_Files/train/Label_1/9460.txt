#include <iostream>
#include <cmath>
using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  int aa = a.size(), bb = b.size(), ans = 0;
  for (int i = 0; i <= aa - bb; i++) {
    int diff = 0;
    for (int j = 0; j < bb; j++) { diff += a[i + j] != b[j]; }
    if (!(diff & 1)) ans++;
  }
  cout << ans << endl;
}
