#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, a, b;
  cin >> n >> a >> b;
  long long ans = 0, total = a;
  vector<long long> t(n);
  for (long long i = 0; i < n; i++) cin >> t[i];
  total += (b - a);
  for (long long i = 0; i < n; i++) {
    if (t[i] == 1) {
      if (total) {
        total--;
        continue;
      } else if (a) {
        a--;
        continue;
      } else {
        ans += 2;
      }
    } else {
      if (b) {
        b--;
        continue;
      } else {
        ans += 2;
      }
    }
  }
  cout << ans << endl;
}
