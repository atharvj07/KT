from sys import stdin, setrecursionlimit

setrecursionlimit(300000)
read = lambda: stdin.readline().strip()
n = int(read())
a = list(map(int, read().split()))
e = [[] for _ in range(n)]
for _ in range(n - 1):
    x, y = map(int, read().split())
    e[x - 1].append(y - 1)
    e[y - 1].append(x - 1)

ans = 1e9
t = [0] * n


def dfs(x, p):
    global ans
    t[x] = 1
    for y in e[x]:
        if y == p:
            continue
        dfs(y, x)
        a[x] += max(0, a[y])
        ans = min(ans, a[x])
    t[x] = 0


dfs(0, -1)
print(max(a))
