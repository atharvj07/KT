import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String s = input.next();
        int[] lower = new int[n];
        for (int i = 0; i < n; i++) {
            lower[i] = (s.charAt(i) >= 'a' && s.charAt(i) <= 'z') ? 1 : 0;
        }
        int[] dp = new int[n];
        dp[0] = lower[0];
        for (int i = 1; i < n; i++) {
            dp[i] = lower[i];
            if (lower[i] > 0 && s.charAt(i) != s.charAt(i - 1)) {
                dp[i] += dp[i - 1];
            }
        }
        int max = 0;
        for (int i = 0; i < n; i++) {
            max = Math.max(max, dp[i]);
        }
        System.out.println(max);
    }
}
