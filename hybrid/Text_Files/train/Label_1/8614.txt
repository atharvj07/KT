#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (int)(n); i++)

typedef long long ll;
using namespace std;

const ll MAX_N = 2e5;
ll N, M, K;
set<ll> G[MAX_N];
vector<string> ans;

bool dfs(ll p, ll c, ll par, vector<ll> &vec, string &cur) {
  vec.push_back(c);
  if ((ll)vec.size() == N) {
    bool ok = true;
    rep(i, N) {
      if (G[i].size() > 1) {
        ok = false;
        break;
      }
    }
    if (!ok) return false;
    ans.push_back(cur);
    return true;
  }
  for (ll to : G[c]) {
    if (to == par) continue;
    cur.push_back('1');
    if (!dfs(p, to, c, vec, cur)) return false;
    cur.pop_back();
    cur.push_back('0');
    if (!dfs(p, to, c, vec, cur)) return false;
    cur.pop_back();
  }
  vec.pop_back();
  return false;
}

int main() {
  cin >> N >> M >> K;
  rep(i, M) {
    ll a, b;
    cin >> a >> b;
    a--, b--;
    G[a].insert(b);
    G[b].insert(a);
  }
  vector<ll> vec;
  string cur;
  dfs(1, 0, -1, vec, cur);
  cout << ans.size() << endl;
  for (string a : ans) cout << a << endl;
}
