#include <bits/stdc++.h>
using namespace std;
const int mod = 998244353;

typedef long long ll;

inline int add(int a, int b) { return (a + b) % mod; }

inline int sub(int a, int b) { return (a - b + mod) % mod; }

inline int mul(int a, int b) { return (1ll * a * b) % mod; }

inline int pow(int a, int p) {
  int ans = 1;
  while (p) {
    if (p & 1) ans = mul(ans, a);
    a = mul(a, a);
    p >>= 1;
  }
  return ans;
}

int main() {
  int n, k;
  cin >> n >> k;
  cout << mul(pow(n, k), sub(1, pow(n - 1, k))) << '\n';
  return 0;
}
