import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.HashSet;
import java.util.LinkedList;
import java.util.List;
import java.util.Map;
import java.util.Map.Entry;
import java.util.Queue;
import java.util.Set;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = reader.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int k = Integer.parseInt(line1[1]);
        String initialStates = reader.readLine();
        List<Set<Integer>> subsets = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            int num = Integer.parseInt(reader.readLine());
            Set<Integer> subset = new HashSet<>();
            for (int j = 0; j < num; j++) {
                subset.add(Integer.parseInt(reader.readLine()) - 1);
            }
            subsets.add(subset);
        }

        Map<Integer, Set<Integer>> subsetsByLamp = new HashMap<>();
        for (int i = 0; i < n; i++) {
            subsetsByLamp.put(i, new HashSet<>());
        }
        for (int i = 0; i < k; i++) {
            for (int lamp : subsets.get(i)) {
                subsetsByLamp.get(lamp).add(i);
            }
        }

        int[] result = new int[n];
        Queue<Integer> queue = new LinkedList<>();
        boolean[] visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (initialStates.charAt(i) == '1') {
                result[i] = 0;
                queue.add(i);
                visited[i] = true;
            }
        }

        while (!queue.isEmpty()) {
            int currentLamp = queue.poll();
            for (int subsetIndex : subsetsByLamp.get(currentLamp)) {
                Set<Integer> subset = subsets.get(subsetIndex);
                boolean allLampsOn = true;
                for (int lamp : subset) {
                    if (!visited[lamp] && initialStates.charAt(lamp) == '0') {
                        allLampsOn = false;
                        break;
                    }
                }

                if (allLampsOn) {
                    for (int lamp : subset) {
                        if (!visited[lamp]) {
                            result[lamp] = result[currentLamp] + 1;
                            visited[lamp] = true;
                            queue.add(lamp);
                        }
                    }
                }
            }
        }

        for (int i = 0; i < n; i++) {
            System.out.println(result[i]);
        }
    }
}
