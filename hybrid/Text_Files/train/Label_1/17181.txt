#include <bits/stdc++.h>
using namespace std;
int main() {
    int q;
    cin >> q;
    for (int qi = 0; qi < q; qi++) {
        int n;
        cin >> n;
        unordered_map<int, int> cnt;
        vector<int> last(n + 1), ord;
        int l = n + 1;
        int mx = 0;
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            if (!last[x]) last[x] = i;
            cnt[x]++;
            mx = max(mx, cnt[x]);
            ord.push_back(x);
        }
        sort(ord.begin(), ord.end());
        int ans = n - mx;
        if (ans < 0) ans = 0;
        int r = n;
        int cr = n;
        int cl = -1;
        int k = -1;
        for (int i = 0; i < n; i++) {
            r = min(r, last[ord[i]]);
            if (last[ord[i]] != l && cnt[ord[i]] == mx) {
                cl = i;
            }
            if (cnt[ord[i]] == mx) {
                if (k == -1) k = i;
                cr = min(cr, last[ord[i]]);
            }
            if (i != n - 1 && ord[i + 1] != ord[i]) {
                ans = min(ans, i - k + n - cr);
                k = -1;
                cr = n;
                cl = -1;
            }
        }
        if (k != -1) {
            ans = min(ans, n - k + n - cr);
        }
        if (cl != -1) {
            ans = min(ans, n - cl + r);
        }
        cout << ans << endl;
    }
    return 0;
}
