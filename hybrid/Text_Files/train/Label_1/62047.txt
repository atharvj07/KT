n, k = map(int, input().split())

if k * (k - 1) < n:
    print("NO")
else:
    print("YES")
    count = 0
    for i in range(1, k + 1):
        for j in range(1, k + 1):
            if i != j:
                print(i, j)
                count += 1
                if count == n:
                    break
        if count == n:
            break
