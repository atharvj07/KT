import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Chouti {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] evens = new int[n / 2];
        for (int i = 0; i < n / 2; i++) {
            evens[i] = scanner.nextInt();
        }

        List<Integer> seq = restoreSequence(evens);

        if (seq == null) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            for (int num : seq) {
                System.out.print(num + " ");
            }
        }
    }

    static List<Integer> restoreSequence(int[] evens) {
        int n = evens.length * 2;
        List<Integer> seq = new ArrayList<>(n);

        int sum = 0;
        for (int even : evens) {
            sum += even;
            seq.add(even);
        }

        int target = 1;
        for (int i = 0; i < evens.length; i++) {
            int next = (int) Math.sqrt(sum + target);
            if (next * next != sum + target) {
                return null;
            }
            seq.add(1, next);
            sum += next;
            target++;
        }

        return seq;
    }
}
