#include <bits/stdc++.h>
using namespace std;
int a[200010], b[200010], d[200010];
bool u[200010];
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int n, k, x, y;
  cin >> n >> k;
  for (int i = 1; i <= k; i++) {
    cin >> x;
    a[i] = x;
    u[x] = 1;
  }
  x = k + 1;
  for (int i = 1; i <= n; i++)
    if (!u[i]) b[x++] = i;
  y = x;
  for (int i = 1; i <= k; i++) {
    b[i] = a[i];
  }
  for (int i = 1; i <= n; i++) {
    if (!u[i]) {
      while (1) {
        if (y == k + 1 || b[y] < i) {
          b[y++] = i;
          break;
        }
        if (d[b[y]] == 0) {
          d[b[y]] = i;
          y--;
        } else {
          i = d[b[y]];
          y++;
        }
      }
    }
  }
  for (int i = 1; i <= k; i++) {
    a[i] = b[i];
  }
  if (a[1] == 1) {
    for (int i = 1; i <= n; i++) cout << a[i] << " ";
  } else
    cout << -1;
  return 0;
}
