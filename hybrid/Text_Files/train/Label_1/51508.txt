import java.util.HashMap;
import java.util.LinkedList;
import java.util.Map;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    private static int[] p;
    private static int[] depth;
    private static boolean[] visited;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        p = new int[n + 1];
        depth = new int[n + 1];
        visited = new boolean[n + 1];
        Map<Integer, LinkedList<Integer>> mp = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            mp.put(i, new LinkedList<>());
        }
        for (int i = 1; i < n; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            mp.get(a).add(b);
            mp.get(b).add(a);
        }
        bfs(mp, 1);
        int maxDepth = 0;
        int farthest = 0;
        for (int i = 1; i <= n; i++) {
            if (depth[i] > maxDepth) {
                maxDepth = depth[i];
                farthest = i;
            }
        }
        bfs(mp, farthest);
        maxDepth = 0;
        for (int i = 1; i <= n; i++) {
            if (depth[i] > maxDepth) {
                maxDepth = depth[i];
                farthest = i;
            }
        }
        if (maxDepth <= k) {
            System.out.println(0);
        } else {
            System.out.println(maxDepth - k);
        }
    }

    private static void bfs(Map<Integer, LinkedList<Integer>> mp, int start) {
        Queue<Integer> q = new LinkedList<>();
        q.offer(start);
        visited[start] = true;
        while (!q.isEmpty()) {
            int current = q.poll();
            for (int next : mp.get(current)) {
                if (!visited[next]) {
                    q.offer(next);
                    visited[next] = true;
                    depth[next] = depth[current] + 1;
                    p[next] = current;
                }
            }
        }
    }
}
