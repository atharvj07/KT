#include <iostream>
#include <queue>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    priority_queue<long long, vector<long long>, greater<long long>> pq;
    for (int i = 0; i < n; i++) {
        long long s, m;
        cin >> s >> m;
        while (!pq.empty() && pq.top() <= s) pq.pop();
        if (pq.size() < k)
            cout << (pq.empty() ? s : pq.top()) + m << endl;
        else
            pq.push(pq.top() + m);
    }
}
