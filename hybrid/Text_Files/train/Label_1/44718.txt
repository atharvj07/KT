import java.util.Scanner;
public class Main {
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int []arr = new int[n];
        for(int i = 0;i<n;i++)arr[i]=in.nextInt();
        int[][] que = new int[m][2];
        for(int i = 0;i<m;i++)que[i][0]=in.nextInt()-1;
        for(int i = 0;i<m;i++)que[i][1]=in.nextInt()-1;
        int []sum = new int[n+1];
        for(int i = 0;i<n;i++)sum[i+1]=sum[i]+arr[i];
        for(int i = 0;i<m;i++){
            if (((sum[que[i][1]+1]-sum[que[i][0]])%2==0)){
                System.out.println("1");
            }else{
                System.out.println("0");
            }
        }
    }
}
