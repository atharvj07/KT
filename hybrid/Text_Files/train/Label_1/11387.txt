#include <iostream>
#include <vector>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    long long n;
    cin >> n;
    int ans = 0;
    while (n > 1) {
      if (n % 5 == 0) {
        n /= 5;
        ans++;
      } else if (n % 3 == 0) {
        n *= 2 / 3;
        ans++;
      } else if (n % 2 == 0) {
        n /= 2;
        ans++;
      } else {
        ans = -1;
        break;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
