import java.util.*;

public class Main {
    private static final int[] DX = {1, 0, -1, 0};
    private static final int[] DY = {0, 1, 0, -1};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int c = sc.nextInt();
        long[] dc = new long[c + 1];
        for (int i = 1; i <= c; i++) dc[i] = sc.nextInt();
        long[] dr = new long[c + 1];
        for (int i = 1; i <= c; i++) dr[i] = sc.nextInt();
        int[][] g = new int[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) g[i][j] = sc.nextInt();
        }

        boolean[][] visited = new boolean[n + 1][n + 1];
        Queue<Integer> rp = new LinkedList<>();
        rp.offer(1);
        rp.offer(1);
        long sum = 0;
        while (!rp.isEmpty()) {
            int px = rp.poll();
            int py = rp.poll();
            if (visited[px][py]) continue;
            visited[px][py] = true;
            long m = Long.MAX_VALUE;
            int mc = -1;
            for (int i = 1; i <= c; i++) {
                long sum2 = 0;
                for (int j = 1; j <= n; j++) {
                    for (int k = 1; k <= n; k++) {
                        if ((j + k) % 3 == (px + py) % 3) sum2 += dr[i] * (g[j][k] == i ? 0 : 1);
                        else sum2 += dc[i] * (g[j][k] == i ? 1 : 0);
                    }
                }
                if (sum2 < m) {
                    m = sum2;
                    mc = i;
                }
            }
            sum += m;
            for (int i = 0; i < DX.length; i++) {
                int nx = px + DX[i];
                int ny = py + DY[i];
                if (nx < 1 || nx > n || ny < 1 || ny > n || visited[nx][ny]) continue;
                rp.offer(nx);
                rp.offer(ny);
            }
        }

        System.out.println(sum);
    }
}
