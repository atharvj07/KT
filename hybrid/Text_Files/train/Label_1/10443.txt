#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int> > v(n);
  for (int i = 0; i < n; i++) {
    int w, h;
    cin >> w >> h;
    v[i] = make_pair(w, h);
  }
  sort(v.begin(), v.end());
  int ans = 0;
  for (int i = n / 2; i < n; i++) ans += v[i].first * v[i].second;
  for (int i = 0; i < n / 2; i++) ans += v[i].second * v[i].first;
  cout << ans << endl;
  return 0;
}
