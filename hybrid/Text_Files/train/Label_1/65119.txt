n = int(input())
cards = list(map(int, input().split()))

card_counts = {}
for card in cards:
    if card in card_counts:
        card_counts[card] += 1
    else:
        card_counts[card] = 1

for count in card_counts.values():
    if count % 2 == 1:
        print("Conan")
        break
else:
    print("Agasa")
