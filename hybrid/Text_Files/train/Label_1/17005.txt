#include <bits/stdc++.h>
 
using namespace std;
 
int main()
{
    long long int n, m, i, j, a, b;
    cin >> n >> m >> i >> j >> a >> b;
    if(i==1 || i==n || j==1 || j==m)  cout << 0 << "\n";
    else if ((i+a<=n && j-b>=1) || (i+a>=1 && j-b<=m) || (i-a<=n && j+b>=1) || (i-a>=1 && j+b<=m)) cout << 1 << "\n";
    else cout << "Poor Inna and pony!" << "\n";
    return 0;
}
