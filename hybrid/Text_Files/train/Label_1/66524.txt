import sys

def solve():
    n = int(input())
    p = [int(x) for x in input().split()]
    fingerprint = [p[i]+p[i+1] for i in range(n-1)] + [p[0]+p[n-1]]
    fingerprint.sort()
    for i in range(1,n-1):
        if fingerprint[i] == fingerprint[i-1]:
            p[i],p[i+1] = p[i+1],p[i]
            break
    else:
        p[0],p[1] = p[1],p[0]
    print(' '.join([str(x) for x in p]))

for _ in range(int(input())):
    solve()
