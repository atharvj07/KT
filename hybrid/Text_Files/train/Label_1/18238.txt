#include <iostream>
#include <vector>

using namespace std;

int main() {
  string hint;
  cin >> hint;
  vector<int> dp(hint.size(), 0);
  if (hint[0] == '?') {
    dp[0] = 10;
  } else {
    dp[0] = 1;
  }
  for (int i = 1; i < hint.size(); i++) {
    if (hint[i] == '?') {
      dp[i] = dp[i - 1] * 10;
    } else if (hint[i] >= 'A' && hint[i] <= 'J') {
      dp[i] = dp[i - 1];
    } else {
      dp[i] = dp[i - 1];
      if (i - 1 >= 0 && hint[i - 1] == hint[i]) {
        dp[i] -= dp[i - 2];
      }
    }
  }
  cout << dp[hint.size() - 1] << endl;
  return 0;
}
