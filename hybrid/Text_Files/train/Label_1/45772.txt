import java.util.*;
import java.io.*;
class Main {
    static long MOD = 1000000007;
    static int[] power = new int[200001];
    static void pre() {
        power[0] = 1;
        for(int i = 1 ; i <= 200000 ; i++) {
            power[i] = (power[i - 1] * 2) % (int)MOD;
        }
    }
    static long fastpow(long base, long pow) {
        long ans = 1;
        while(pow > 0) {
            if(pow % 2 == 1) {
                ans *= base;
                ans %= MOD;
            }
            base *= base;
            base %= MOD;
            pow /= 2;
        }
        return ans;
    }
    public static void main(String[] args) {
        pre();
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long t = sc.nextLong();
        String s = sc.next();
        int[] tarr = new int[k + 1];
        for(int i = 1 ; i <= k ; i++) {
            tarr[i] = sc.nextInt();
        }
        int[][] matrix = new int[k + 1][k + 1];
        for(int i = 1 ; i <= k ; i++) {
            for(int j = 1 ; j <= k ; j++) {
                matrix[i][j] = sc.nextInt();
            }
        }
        int[] dp = new int[21];
        int[] freq = new int[21];
        dp[0] = 0;
        dp[1] = 1;
        for(int i = 2 ; i <= 20 ; i++) {
            dp[i] = (dp[i - 1] + dp[i - 2]) % (int)MOD;
        }
        for(int i = 0 ; i < n ; i++) {
            int index = s.charAt(i) - 'A' + 1;
            freq[index]++;
        }
        long ans = 0;
        int total = 0;
        for(int i = 1 ; i <= k ; i++) {
            total += freq[i] > 0 ? matrix[i][i] : 0;
        }
        for(int mask = 0 ; mask < (1 << k) ; mask++) {
            long mul = 1;
            int cur = total;
            for(int i = 0 ; i < k ; i++) {
                if((mask & (1 << i)) > 0) {
                    mul *= dp[freq[i + 1]];
                    mul %= MOD;
                    cur -= matrix[i + 1][i + 1] * freq[i + 1];
                }
                else {
                    mul *= fastpow(2, freq[i + 1]);
                    mul %= MOD;
                    cur += tarr[i + 1] * freq[i + 1];
                }
            }
            if(__builtin_popcount(mask) % 2 == 0) {
                ans += mul;
                ans %= MOD;
            }
            else {
                ans -= mul;
                ans += MOD;
                ans %= MOD;
            }
        }
        ans %= MOD;
        System.out.println(ans);
    }
    static int __builtin_popcount(int n) {
        int c = 0;
        while(n > 0) {
            c += n & 1;
            n >>= 1;
        }
        return c;
    }
}
