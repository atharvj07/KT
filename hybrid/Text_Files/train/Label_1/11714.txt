#include <bits/stdc++.h>
using namespace std;

const int MX = 52;

int mat[MX][MX];

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            mat[i][j] = ((i + j) % 2) * n * n + (i * n + j + 1);
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << mat[i][j] << ' ';
        }
        cout << endl;
    }
    return 0;
}
