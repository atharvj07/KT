import sys

def solve():
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    adj = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)

    if k == n - 1: return "YES"
    vis = [False] * (n + 1)
    xor = 0
    def dfs(node, pre):
        nonlocal xor
        vis[node] = True
        xor ^= a[node - 1]
        for u in adj[node]:
            if not vis[u]:
                dfs(u, node)

    dfs(1, -1)
    if xor == 0: return "YES"
    return "NO"

for _ in range(int(input())):
    print(solve())
