import sys
sys.setrecursionlimit(10**9)

dp = {}
mod = 10**9 + 7

def go(id, last, score, painted):
    if id == n:
        if score % 2 == p:
            return 1
        else:
            return 0

    key = (id, last, score, painted)
    if key in dp:
        return dp[key]

    ans = 0
    if c[id] == -1:
        for col in range(2):
            if col != last:
                ans += go(id + 1, col, score + col, painted | (1 << id))

    if c[id] != 0:
        ans += go(id + 1, c[id], score + c[id], painted)

    dp[key] = ans % mod
    return ans % mod

n, p = map(int, input().split())
c = list(map(int, input().split()))
print(go(0, -1, 0, 0))
