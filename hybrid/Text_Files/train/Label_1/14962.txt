#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 2e5 + 5;
ll n, m, a, b, t, s[N], dp[N][2][2], dist[N];
void solve() {
  memset(dp, -1, sizeof(dp));
  memset(dist, -1, sizeof(dist));
  cin >> n >> m >> a >> b;
  for (int i = 0; i < m; i++) cin >> s[i];
  dist[a] = dist[b] = 0;
  queue<array<ll, 2>> q;
  q.push({0, a});
  q.push({0, b});
  dp[a][0][0] = dp[a][1][1] = dp[b][0][0] = dp[b][1][1] = 0;
  while (!q.empty()) {
    ll cell = q.front()[1], cost = q.front()[0];
    q.pop();
    if (dp[cell][dist[cell] & 1][dist[cell] % 2] != -1) continue;
    dp[cell][dist[cell] & 1][dist[cell] % 2] = cost;
    if (cell - 1 >= 1 && dist[cell - 1] == -1) {
      dist[cell - 1] = dist[cell] + 1;
      q.push({cost + 1, cell - 1});
    }
    if (cell + 1 <= n && dist[cell + 1] == -1) {
      dist[cell + 1] = dist[cell] + 1;
      q.push({cost + 1, cell + 1});
    }
  }
  ll ans = 0;
  for (int i = 0; i < m; i++) {
    for (int j = 0; j <= dist[a]; j++) {
      if ((j + s[i] >= dist[b]) || (j + s[i] == dist[b] && (dist[b] & 1) == (s[i] & 1)))
        ans = max(dp[a][j & 1][j % 2] + 1, ans);
    }
  }
  cout << ans << '\n';
}
int main() {
  cin >> t;
  while (t--) solve();
}
