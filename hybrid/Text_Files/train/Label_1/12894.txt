#include <algorithm>
#include <cstdio>
#include <set>
#include <vector>

using ll = long long;
const ll MX = 1ll * 1e9;

ll w[200005], x[200005];
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%lld %lld", &x[i], &w[i]);
  }
  ll ans = 0;
  for (int i = 0; i < n; ++i) {
    ll l = x[i] - w[i], r = x[i] + w[i];
    ll cnt = 1;
    std::set<ll> ss;
    ss.insert(l);
    ss.insert(r);
    for (int j = i + 1; j < n; ++j) {
      if (ss.find(x[j] - w[j]) != ss.end()) {
        ++cnt;
        ss.insert(x[j] - w[j]);
      }
      if (ss.find(x[j] + w[j]) != ss.end()) {
        ++cnt;
        ss.insert(x[j] + w[j]);
      }
      ans = std::max(ans, cnt);
    }
  }
  printf("%lld\n", ans);
  return 0;
}
