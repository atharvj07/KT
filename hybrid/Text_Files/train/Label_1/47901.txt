import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            String s = sc.next();
            int n = s.length();
            int[] dp = new int[n + 1];
            dp[0] = 0;
            dp[1] = s.charAt(0) == '0' ? 0 : a;
            for (int j = 2; j <= n; j++) {
                if (s.charAt(j - 1) == '0') {
                    dp[j] = dp[j - 1];
                } else {
                    dp[j] = Math.min(dp[j - 1] + a, dp[j - 2] + b + a);
                }
            }
            System.out.println(dp[n]);
        }
    }
}
