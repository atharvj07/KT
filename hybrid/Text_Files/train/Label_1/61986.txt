import sys
input = sys.stdin.readline

mod = 10 ** 9 + 7
n, p = map(int, input().split())
c = list(map(int, input().split()))
dp = [[[[[-1] * 3 for k in range(2)] for j in range(3)] for i in range(2)] for _ in range(n + 1)]

def solve(idx, prv, lst, painted, odd):
    if idx == n:
        if odd == p:
            return 1
        return 0
    if dp[idx][prv][lst][painted][odd] != -1:
        return dp[idx][prv][lst][painted][odd]
    res = 0
    if painted == 0:
        if c[idx] == 0:
            res += solve(idx + 1, prv, lst, painted, odd)
        elif c[idx] == 1:
            res += solve(idx + 1, prv, lst, painted, not odd)
            res += solve(idx + 1, idx, idx, painted + 1, odd)
        elif c[idx] == -1:
            res += solve(idx + 1, prv, lst, painted, odd)
            res += solve(idx + 1, prv, lst, painted, not odd)
            res += solve(idx + 1, idx, idx, painted + 1, odd)
            res += solve(idx + 1, idx, idx, painted + 1, not odd)
    else:
        if c[idx] == 1:
            res += solve(idx + 1, prv, lst, painted, not odd)
            res += solve(idx + 1, idx, lst, painted, odd)
        else:
            res += solve(idx + 1, prv, lst, painted, odd)
            res += solve(idx + 1, idx, lst, painted, not odd)
            if prv != idx:
                if lst == 1:
                    res += solve(idx + 1, prv, 0, painted, odd)
                else:
                    res += solve(idx + 1, prv, 0, painted, not odd)
    dp[idx][prv][lst][painted][odd] = res % mod
    return dp[idx][prv][lst][painted][odd]

print(solve(1, 0, 0, 0, 0))
