#include <bits/stdc++.h>
using namespace std;
int n, m, a[61], b[61], ans;
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= m; i++) cin >> b[i];
  sort(a + 1, a + n + 1);
  sort(b + 1, b + m + 1);
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) {
      int cnt = 0;
      if (!binary_search(a + 1, a + n + 1, b[j])) cnt++;
      if (!binary_search(b + 1, b + m + 1, a[i])) cnt++;
      ans = max(ans, cnt);
    }
  cout << ans;
  return 0;
}
