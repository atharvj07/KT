#include <iostream>
#include <vector>
using namespace std;

int main() {
  int m;
  cin >> m;
  vector<int> b(m);
  for (int i = 0; i < m; i++) { cin >> b[i]; }
  int ans = 0;
  for (int i = 0, j = -1; i < m; i++) {
    if (j < 0 || b[i] > b[j]) {
      j = i;
      ans++;
    }
  }
  cout << ans << endl;
  for (int i = 0, j = -1; i < m; i++) {
    if (j < 0 || b[i] > b[j]) {
      j = i;
      cout << b[j] << ' ';
    }
  }
  cout << endl;
  return 0;
}
