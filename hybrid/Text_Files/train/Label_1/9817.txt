#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> x(n);
  for (int i = 0; i < n; ++i) { std::cin >> x[i]; }

  std::sort(x.begin(), x.end());
  int ans = x[n - 1] - x[0] - 1;
  for (int i = 1; i < n; ++i) {
    ans -= x[i] - x[i - 1] - 1;
  }
  std::cout << ans << std::endl;
  return 0;
}
