import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
e = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    x, y = map(int, input().split())
    e[x].append(y)
    e[y].append(x)

vis = [False] * (n + 1)
ans = -1e9
st = []
for i in range(1, n + 1):
    if not vis[i]:
        q = [(i, a[i - 1])]
        ans = max(ans, a[i - 1])
        while q:
            u, cu = q.pop()
            if vis[u]: continue
            vis[u] = True
            for v in e[u]:
                if not vis[v]:
                    q.append((v, cu + 1))
                    ans = max(ans, cu + 1)
                    if cu + 1 > a[v - 1]:
                        ans = max(ans, max(a[v - 1], a[i - 1]))
print(ans)
