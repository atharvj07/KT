#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string n;
    cin >> n;
    int len = n.size();
    vector<int> dp(len + 1, len);
    dp[len] = 0;
    for (int i = len - 1; i >= 0; i--) {
      if (n[i] % 2 == 0) {
        dp[i] = 0;
        continue;
      }
      for (int j = i + 1; j <= len; j++) {
        if (j == len || n[j] % 2 == 0) {
          dp[i] = min(dp[i], dp[j] + 1);
        }
      }
    }
    if (dp[0] >= len) {
      cout << -1 << endl;
    } else {
      cout << dp[0] << endl;
    }
  }
  return 0;
}
