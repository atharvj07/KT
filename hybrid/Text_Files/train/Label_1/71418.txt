def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

n, m = map(int, input().split())
a = list(map(int, input().split()))
pairs = [tuple(map(int, input().split())) for _ in range(m)]

operations = 0
for i, j in pairs:
    g = gcd(a[i-1], a[j-1])
    while g != 1:
        operations += 1
        a[i-1] //= g
        a[j-1] //= g
        g = gcd(a[i-1], a[j-1])

print(operations)
