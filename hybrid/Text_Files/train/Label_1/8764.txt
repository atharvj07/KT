#include <iostream>
#include <vector>
using namespace std;

#define MOD 1000000007

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> dp(k + 1);
    dp[0] = 1;
    for (int i = 1; i <= n; i++) {
        for (int j = k; j >= 0; j--) {
            dp[j] = (dp[j] * i + dp[j - 1]) % MOD;
        }
    }
    cout << dp[k] << endl;
    return 0;
}
