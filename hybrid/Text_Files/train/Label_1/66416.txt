import math

def can_finish_on_time(n, d):
    # We need to find an x such that x + ceil(d / (x + 1)) <= n
    # Let's try all possible values of x from 0 to sqrt(d) because the minimum time can't be more than sqrt(d)
    for x in range(int(math.sqrt(d)) + 2):
        if x + math.ceil(d / (x + 1)) <= n:
            return "YES"
    return "NO"

T = int(input().strip())
for _ in range(T):
    n, d = map(int, input().strip().split())
    print(can_finish_on_time(n, d))
