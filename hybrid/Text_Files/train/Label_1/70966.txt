n=int(input())
a=[i for i in range(1,n+1)]
ans=a[:]
mn=n*2
for i in range(n):
    for j in range(i,n):
        swp=a[j]
        a[j]=a[i]
        a[i]=swp
        b=[a[k-1] for k in range(1,n+1) if k==1 or a[k]<a[k-1]]
        c=[a[k-1] for k in range(1,n+1) if k==1 or a[k]>a[k-1]]
        if len(b)+len(c)<mn:
            mn=len(b)+len(c)
            ans=a[:]
        a[j]=a[i]
        a[i]=swp
print(*ans)
