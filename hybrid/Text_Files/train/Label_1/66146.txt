import math
n, s = map(int, input().split())
pop, x, y = [0] * n, [0] * n, [0] * n
for i in range(n):
    x[i], y[i], pop[i] = map(int, input().split())
    s += pop[i]
if s < 1000000:
    print(-1)
else:
    l, r = 0, 1e9
    while r - l > 1e-6:
        m = (l + r) / 2
        ok = False
        for i in range(n):
            if x[i] ** 2 + y[i] ** 2 <= m ** 2:
                ok = True
                break
        if ok:
            r = m
        else:
            l = m
    print(math.sqrt(l))
