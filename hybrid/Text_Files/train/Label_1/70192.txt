import sys
n, m, q = map(int, input().strip().split(' '))
a = list(map(int, input().strip().split(' ')))
c = [0 for i in range(0, m + 1)]
for i in range(0, n): c[a[i]] += 1
ans = []
for i in range(0, q):
    y = int(input().strip())
    for j in range(1, m + 1):
        if c[j] <= (y - n - 1) % (m - 1) + 1: ans.append(j)
        else: break
print(*ans, sep = '\n')
