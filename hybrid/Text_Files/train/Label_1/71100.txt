import sys
input=sys.stdin.readline

n,m=map(int,input().split())
l=list(map(int,input().split()))
r=[-1]*n
ans=[-1]*m
for i in range(m):
    for j in range(max(0,n-l[i]+1),n):
        if r[j]==-1:
            r[j]=i
            ans[i]=j+1
            break
if -1 in r:
    print(-1)
else:
    print(*ans)
