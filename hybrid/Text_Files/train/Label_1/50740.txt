import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    static Map<Integer, Integer> a, b;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), v = sc.nextInt(), e = sc.nextInt();
        a = new HashMap<>();
        b = new HashMap<>();
        int sa = 0, sb = 0;
        for (int i = 1; i <= n; i++) {
            int ai = sc.nextInt();
            sa += ai;
            if (ai > 0) a.put(i, ai);
            int bi = sc.nextInt();
            sb += bi;
            if (bi > 0) b.put(i, bi);
        }
        if (sa != sb) {
            System.out.println("NO");
            return;
        }
        Map<Integer, Integer>[] rel = new Map[n + 1];
        for (int i = 1; i <= n; i++) rel[i] = new HashMap<>();
        for (int i = 1; i <= e; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            rel[x].merge(y, 1, Integer::sum);
            rel[y].merge(x, 1, Integer::sum);
        }
        int[] f = new int[n + 1];
        Arrays.fill(f, -1);
        int cnt = calc(1, f, rel);
        if (cnt == -1) {
            System.out.println("NO");
            return;
        }
        System.out.println(cnt);
        rec(1, f, rel);
    }

    static int calc(int u, int[] f, Map<Integer, Integer>[] rel) {
        if (f[u] != -1) return f[u];
        int ans = 0;
        for (Map.Entry<Integer, Integer> e : rel[u].entrySet()) {
            int t = calc(e.getKey(), f, rel);
            if (t == -1) return -1;
            ans = ans + t + e.getValue();
        }
        return f[u] = ans / 2;
    }

    static void rec(int u, int[] f, Map<Integer, Integer>[] rel) {
        for (Map.Entry<Integer, Integer> e : rel[u].entrySet()) {
            int t = f[u] + f[e.getKey()] + e.getValue();
            if (e.getValue() == 0) continue;
            int d = Math.min(e.getValue(), a.getOrDefault(u, 0));
            System.out.println(u + " " + e.getKey() + " " + d);
            a.put(u, a.get(u) - d);
            a.put(e.getKey(), a.getOrDefault(e.getKey(), 0) + d);
            rec(e.getKey(), f, rel);
            a.put(u, a.get(u) + d);
            a.put(e.getKey(), a.get(e.getKey()) - d);
            d = Math.min(t - e.getValue(), b.getOrDefault(e.getKey(), 0));
            System.out.println(u + " " + e.getKey() + " " + d);
            a.put(u, a.get(u) - d);
            a.put(e.getKey(), a.getOrDefault(e.getKey(), 0) + d);
            rec(e.getKey(), f, rel);
            a.put(u, a.get(u) + d);
            a.put(e.getKey(), a.get(e.getKey()) - d);
        }
    }
}
