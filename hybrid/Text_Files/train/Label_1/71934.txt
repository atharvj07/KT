from collections import defaultdict, deque

n, k = map(int, input().split())
if k == 0 or k * (n - 1) % n != 0:
    print(-1)
else:
    matches = n * (n - 1) // 2
    adj = defaultdict(list)
    for i in range(n):
        for j in range(k):
            adj[i].append((i + j + 1) % n)
    print(matches)
    for i in range(n):
        q = deque()
        q.append(i)
        while q:
            u = q.popleft()
            for v in adj[u]:
                if v != i:
                    print(u + 1, v + 1)
                    q.append(v)
                    adj[u].remove(v)
                    break
