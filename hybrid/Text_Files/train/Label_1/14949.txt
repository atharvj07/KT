#include <bits/stdc++.h>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    int n, m;
    cin >> n >> m;
    unordered_map<string, string> a;
    for (int i = 0; i < m; i++) {
        string s1, s2;
        cin >> s1 >> s2;
        a[s1] = s2;
    }
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        if (s.length() <= a[s].length())
            cout << s << " ";
        else
            cout << a[s] << " ";
    }
    return 0;
}
