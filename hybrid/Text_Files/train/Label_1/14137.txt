#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    string s;
    cin >> n >> s;
    vector<int> v = {0};
    int l = 0, r = 0;
    for (char c : s) {
        if (c == 'L') l++;
        else r++;
        v.push_back(r - l);
    }
    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());
    cout << v.size() << endl;
    return 0;
}
