#include<bits/stdc++.h>
using namespace std;
#define ll long long int
#define ull unsigned long long int
#define ld long double

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    ull t, i, j;
    cin >> t;
    while (t--) {
        ull n, a[2005][2], x[4], y[4], f = 0;
        cin >> n;
        for (i = 0; i < n; i++) {
            cin >> a[i][0] >> a[i][1];
        }
        for (i = 0; i < n; i++) {
            for (j = i + 1; j < n; j++) {
                for (ull k = j + 1; k < n; k++) {
                    if (a[i][0] == a[j][0]) {
                        x[0] = a[i][0];
                        y[0] = a[i][1];
                        x[1] = a[j][0];
                        y[1] = a[j][1];
                        for (int l = 0; l < n; l++) {
                            if (a[l][1] == a[k][1] && a[l][0] != a[i][0] && a[l][0] != a[j][0]) {
                                x[2] = a[l][0];
                                y[2] = a[l][1];
                                x[3] = a[k][0];
                                y[3] = a[k][1];
                                f = 1;
                                break;
                            }
                        }
                    } else if (a[i][0] == a[k][0]) {
                        x[0] = a[i][0];
                        y[0] = a[i][1];
                        x[1] = a[k][0];
                        y[1] = a[k][1];
                        for (int l = 0; l < n; l++) {
                            if (a[l][1] == a[j][1] && a[l][0] != a[i][0] && a[l][0] != a[k][0]) {
                                x[2] = a[l][0];
                                y[2] = a[l][1];
                                x[3] = a[j][0];
                                y[3] = a[j][1];
                                f = 1;
                                break;
                            }
                        }
                    } else if (a[j][0] == a[k][0]) {
                        x[0] = a[j][0];
                        y[0] = a[j][1];
                        x[1] = a[k][0];
                        y[1] = a[k][1];
                        for (int l = 0; l < n; l++) {
                            if (a[l][1] == a[i][1] && a[l][0] != a[j][0] && a[l][0] != a[k][0]) {
                                x[2] = a[l][0];
                                y[2] = a[l][1];
                                x[3] = a[i][0];
                                y[3] = a[i][1];
                                f = 1;
                                break;
                            }
                        }
                    }
                    if (f == 1) {
                        break;
                    }
                }
                if (f == 1) {
                    break;
                }
            }
            if (f == 1) {
                break;
            }
        }
        if (f == 1) {
            cout << 1 << endl;
        } else {
            cout << 0 << endl;
        }
    }
    return 0;
}
