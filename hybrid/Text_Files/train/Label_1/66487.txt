import sys
input = sys.stdin.readline

n,m = map(int,input().split())
edge = [[] for i in range(n+1)]
for i in range(n-1):
  u,v,w = map(int,input().split())
  edge[u].append((v,w))
  edge[v].append((u,w))

q = list(map(int,input().split()))
q.sort()
ans = [0] * m
size = [0] * (n+1)
weight = [0] * (n+1)

def dfs(x:int,p:int,w:int)->int:
  size[x] = 1
  weight[x] = w
  for nx,nw in edge[x]:
    if nx != p:
      size[x] += dfs(nx,x,max(w,nw))
  return size[x]

dfs(1,1,0)

def dfs2(x:int,p:int)->None:
  for nx,nw in edge[x]:
    if nx != p:
      if size[nx] > size[x]:
        weight[nx] = weight[x]
      dfs2(nx,x)

dfs2(1,1)

for i in range(m):
  l,r = 0,n
  idx = -1
  while l<=r:
    mid = (l+r) // 2
    if q[i]>=weight[mid]:
      idx = mid
      l = mid + 1
    else:
      r = mid - 1
  if idx == -1:
    ans[i] = n * (n-1) // 2
  else:
    ans[i] = size[idx] * (n - size[idx])
  
for x in ans:
  print(x)
