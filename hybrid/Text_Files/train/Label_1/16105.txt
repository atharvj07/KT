#include <bits/stdc++.h>

#define pb push_back
#define mp make_pair

using namespace std;

typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;

const int inf = 0x3f3f3f3f;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    a.resize(unique(a.begin(), a.end()) - a.begin());
    n = a.size();
    int ans = 0;
    int p, q;
    for (int x = 1; x <= n; x++) {
        for (int y = 1; x * y <= n; y++) {
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] <= x * y) {
                    cnt++;
                }
            }
            if (cnt >= x * y) {
                if (x * y > ans) {
                    ans = x * y;
                    p = x;
                    q = y;
                }
            }
        }
    }
    cout << ans << "\n" << p << " " << q << "\n";
    for (int i = 0; i < p; i++) {
        for (int j = 0; j < q; j++) {
            for (int k = 0; k < n; k++) {
                if (a[k] <= (i + 1) * (j + 1)) {
                    cout << a[k] << " ";
                    break;
                }
            }
        }
        cout << "\n";
    }
    return 0;
}
