#include <bits/stdc++.h>

using namespace std;

int main() {
  long long n;
  cin >> n;
  for (int k = 62; k >= 0; k--) {
    long long divisor = (1LL << k) - 1;
    long long divisor_squared = divisor * divisor;
    if (divisor_squared <= n && n % divisor_squared == 0) {
      cout << divisor << endl;
      return 0;
    }
  }
  return 0;
}
