#include <bits/stdc++.h>
using namespace std;
int n, ans;
string s;
int main() {
  cin >> s;
  n = s.length();
  vector<int> dp(n);
  stack<int> st;
  for (int i = 0; i < n; i++) {
    dp[i] = (i ? dp[i - 1] + 1 : 1);
    if (s[i] == '(' || s[i] == ')')
      ans += dp[i - 1];
    if (s[i] == '(')
      st.push(i);
    else if (!st.empty())
      st.pop();
    if (s[i] == '?' && !st.empty())
      dp[i] = dp[st.top() - (st.top() ? 1 : 0)], ans += dp[st.top() - (st.top() ? 1 : 0)];
  }
  cout << ans << endl;
  return 0;
}
