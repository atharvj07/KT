#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, d, h;
    cin >> n >> d >> h;
    if (d == 1 || d == n - 1) {
        cout << -1 << endl;
        return 0;
    }
    vector<pair<int, int>> edges;
    int u = 1, v = 2;
    while (v <= n) {
        edges.push_back(make_pair(u, v++));
    }
    u = n;
    int dist = 0;
    while (dist != d) {
        v = u - 1;
        edges.push_back(make_pair(u, v));
        dist++;
        u = v;
    }
    for (auto edge : edges) {
        cout << edge.first << " " << edge.second << endl;
    }
    return 0;
}
