n, m = map(int, input().split())
a = list(map(int, input().split()))

prefix_sum = [0] * (n + 1)
for i in range(n):
    if a[i] > m:
        prefix_sum[i + 1] = prefix_sum[i] + 1
    elif a[i] < m:
        prefix_sum[i + 1] = prefix_sum[i] - 1
    else:
        prefix_sum[i + 1] = prefix_sum[i]

count = {}
result = 0
for i in range(n + 1):
    if prefix_sum[i] in count:
        result += count[prefix_sum[i]]
    if a[i - 1] == m:
        count[prefix_sum[i - 1]] = count.get(prefix_sum[i - 1], 0) + 1
    elif i == 0:
        count[0] = 1

print(result)
