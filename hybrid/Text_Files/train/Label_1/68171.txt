import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
sa, sb = sum(a), sum(b)
a.sort()
b.sort()
av = bv = [0] * 2
idx = [0] * 2
ans = abs(sa - sb)
for j in range(2):
    for i in range(min(n, m)):
        v = abs(sa - sb + a[i] - b[i])
        if v < ans:
            ans = v
            av = [a[i], b[i]]
            bv = [b[i], a[i]]
            idx[j] = i
if ans == abs(sa - sb):
    print(ans, 0)
else:
    print(ans, 1)
    print(*idx)
