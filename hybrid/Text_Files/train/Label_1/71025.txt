import sys

h, q = map(int, input().split())
tree = [[[] for i in range(h + 1)] for j in range(2 ** h)]
for i in range(q):
    x, l, r, ans = map(int, input().split())
    for j in range(h + 1 - x):
        tree[1 << j][x].append([l, r, ans])

exit_node = 1

for i in range(1, h + 1):
    for j in range(1 << i):
        tree[j][i].sort()
        arr = tree[j][i]
        sz = len(arr)
        for k in range(sz - 1):
            if arr[k][0] <= exit_node <= arr[k][1]:
                if arr[k][2] == 0:
                    exit_node = -1
                    break
            elif arr[k][1] < exit_node:
                break
        if exit_node == -1:
            break

if exit_node == -1:
    print('Game cheated!')
elif exit_node == 1:
    print('Data not sufficient!')
else:
    print(exit_node)
