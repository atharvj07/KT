from collections import defaultdict

n = int(input())
genders = []
intervals = defaultdict(list)
for i in range(n):
    gender, a, b = input().split()
    genders.append(gender)
    intervals[a].append(b)
    intervals[b].append(a)

ans = 0

for day in sorted(intervals.keys()):
    invited = set()
    for girl in intervals[day]:
        if len(invited) >= ans:
            break
        ans += 1
        invited.add(girl)
        invited.add(day)

print(ans)
