t = int(input())
for _ in range(t):
    n = int(input())
    field = [list(input()) for _ in range(n)]

    # Find the positions of the two asterisks
    positions = [(i, j) for i in range(n) for j in range(n) if field[i][j] == '*']

    # If the asterisks are in the same row
    if positions[0][0] == positions[1][0]:
        if positions[0][0] == 0:
            field[1][positions[0][1]] = '*'
            field[1][positions[1][1]] = '*'
        else:
            field[0][positions[0][1]] = '*'
            field[0][positions[1][1]] = '*'
    # If the asterisks are in the same column
    elif positions[0][1] == positions[1][1]:
        if positions[0][1] == 0:
            field[positions[0][0]][1] = '*'
            field[positions[1][0]][1] = '*'
        else:
            field[positions[0][0]][0] = '*'
            field[positions[1][0]][0] = '*'
    else:
        field[positions[0][0]][positions[1][1]] = '*'
        field[positions[1][0]][positions[0][1]] = '*'

    for row in field:
        print(''.join(row))
