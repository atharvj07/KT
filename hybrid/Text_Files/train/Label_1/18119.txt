#include <iostream>
#include <string>
#include <cstring>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s, t;
    cin >> s >> t;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] != t[i]) {
            string w1 = s;
            string w2 = t;
            w1.erase(i, 1);
            w2.erase(i, 1);
            if (w1 == w2) {
                ans++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
