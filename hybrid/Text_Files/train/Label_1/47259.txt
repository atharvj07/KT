import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] colors = new int[9];
        for (int i = 0; i < n; i++) {
            if (a[i] >= 3200) {
                colors[8]++;
            } else if (a[i] >= 2800) {
                colors[7]++;
            } else if (a[i] >= 2400) {
                colors[6]++;
            } else if (a[i] >= 2000) {
                colors[5]++;
            } else if (a[i] >= 1600) {
                colors[4]++;
            } else if (a[i] >= 1200) {
                colors[3]++;
            } else if (a[i] >= 800) {
                colors[2]++;
            } else if (a[i] >= 400) {
                colors[1]++;
            } else {
                colors[0]++;
            }
        }
        int min = 0;
        for (int i = 0; i < 8; i++) {
            if (colors[i] > 0) {
                min++;
            }
        }
        int max = min + colors[8];
        System.out.println(min + " " + max);
    }
}
