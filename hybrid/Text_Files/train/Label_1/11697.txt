#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;

int n, l, r, x;
vector<int> c, dp;

int go(int i, int sum, int mx) {
    if(i == n) {
        if(sum >= l && sum <= r && mx - dp[i - 1] >= x)
            return 1;
        return 0;
    }
    if(dp[i] != -1)
        return dp[i];
    dp[i] = 0;
    dp[i] = (dp[i] + go(i + 1, sum + c[i], max(mx, c[i]))) % mod;
    dp[i] = (dp[i] + go(i + 1, sum, mx)) % mod;
    return dp[i];
}

int32_t main() {
    cin >> n >> l >> r >> x;
    c.resize(n);
    for(int i = 0; i < n; i++)
        cin >> c[i];
    sort(c.begin(), c.end());
    dp.resize(n + 1, -1);
    cout << go(1, c[0], c[0]) << "\n";
}
