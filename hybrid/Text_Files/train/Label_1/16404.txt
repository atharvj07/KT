#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int l = a[0], r = a[n - 1];
    if (r - l > 2 * k) {
      cout << "-1\n";
    } else {
      int mid = (l + r) / 2;
      if (l + k >= mid && r - k <= mid) {
        cout << mid << "\n";
      } else {
        if (l + k >= r - k) {
          cout << r - k << "\n";
        } else {
          cout << l + k << "\n";
        }
      }
    }
  }
  return 0;
}
