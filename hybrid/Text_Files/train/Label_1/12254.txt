#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
ll powmod(ll a, ll b) {
    ll ans = 1;
    while (b) {
        if (b & 1) ans = ans * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return ans;
}
int main() {
    ll n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    map<char, ll> mp;
    for (ll i = 0; i < k; i++) {
        char c;
        cin >> c;
        mp[c]++;
    }
    ll cnt = 0;
    for (ll i = 0; i < n; i++) {
        if (mp[s[i]] > 0) cnt++;
        else
            break;
    }
    for (ll i = n - 1; i >= 0; i--) {
        if (mp[s[i]] > 0) cnt++;
        else
            break;
    }
    cout << cnt * (cnt + 1) / 2 << "\n";
    return 0;
}
