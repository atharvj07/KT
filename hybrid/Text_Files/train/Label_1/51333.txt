import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt(), s = sc.nextInt();
        int mod = s % d;
        int base = s / d;
        int ans = -1;
        if (mod > 0) {
            int numDig = 9 - mod;
            if (base > 0) {
                int tempDig = numDig % 9;
                for (int i = 1; i <= tempDig; i++) {
                    ans = d * base + i;
                    numDig -= i;
                    if (numDig < 9) {
                        for (int j = 0; j < numDig; j++) {
                            ans *= 10;
                            ans += 9;
                        }
                        break;
                    }
                }
            } else {
                for (int i = 1; i <= numDig; i++) {
                    ans = i;
                    numDig -= i;
                    if (numDig < 9) {
                        for (int j = 0; j < numDig; j++) {
                            ans *= 10;
                            ans += 9;
                        }
                        break;
                    }
                }
            }
        } else {
            if (base > 0) {
                ans = d * base;
            }
        }
        System.out.println(ans);
    }
}
