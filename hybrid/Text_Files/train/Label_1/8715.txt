#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> v(n);
    for (int i = 0; i < n; ++i) {
        cin >> v[i];
    }

    sort(v.begin(), v.end());
    int ans = 1;
    long long sz = v[0];
    for (int i = 1; i < n; ++i) {
        if (v[i] > sz) {
            sz += v[i];
            ++ans;
        }
    }

    cout << ans << endl;

    return 0;
}
