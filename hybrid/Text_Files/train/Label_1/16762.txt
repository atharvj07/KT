#include<bits/stdc++.h>
using namespace std;
main(int argc, char** argv)                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                             {
    int n,k,x=0,i,a[100001]={0};
    char c[100001];
    cin>>n>>k;
    cin>>c;
    for(i=0;i<n;i++)
        a[c[i]-65]++;
    sort(a,a+26);
    for(i=25;i>=0&&k>0;i--){
        x+=a[i]*k;
        k--;
    }
    cout<<x<<endl;
    return 0;
}
