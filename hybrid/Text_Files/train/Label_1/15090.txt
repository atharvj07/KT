#include <bits/stdc++.h>
using namespace std;
typedef long long i64;
const int MAXN = 5001;
struct painter {
    int l, r;
    painter() {}
    painter(int _l, int _r) : l(_l), r(_r) { }
};
painter p[MAXN];
int n, q, dp[MAXN][4];
vector<int> G[MAXN];
int dfs(int u, int l, int r) {
    if(~dp[u][l]) return dp[u][l];
    int &ret = dp[u][l] = 0;
    for(auto v : G[u]) {
        if(v == l || v == r) continue;
        ret = max(ret, dfs(v, l, r) + 1);
    }
    return ret;
}
int main() {
    memset(dp, -1, sizeof(dp));
    scanf("%d%d", &n, &q);
    for(int i = 1; i <= q; ++i) {
        int l, r;
        scanf("%d%d", &l, &r);
        p[i] = painter(l, r);
    }
    for(int i = 1; i <= q; ++i) {
        for(int j = i + 1; j <= q; ++j) {
            if(p[i].l <= p[j].l && p[i].r >= p[j].r) G[i].push_back(j);
        }
    }
    int ans = 0;
    for(int i = 1; i <= q; ++i) {
        for(int j = i + 1; j <= q; ++j) {
            ans = max({ans, dfs(i, j, 0), dfs(j, i, 0)});
        }
    }
    printf("%d\n", ans);
    return 0;
}
