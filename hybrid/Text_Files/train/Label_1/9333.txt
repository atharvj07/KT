#include <iostream>
#include <string>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    string s, t;
    cin >> s >> t;
    int i = 0, j = 0;
    while (i < s.length() && j < t.length()) {
      if (s[i] == t[j]) {
        i++;
        j++;
      } else {
        i++;
      }
    }
    if (j == t.length()) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
}
