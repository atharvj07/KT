#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<pair<int, int>> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i].first >> arr[i].second;
    }
    vector<int> ans(n, 0);
    int cur = 0;
    for (int i = 0; i < n; i++) {
        if (arr[i].first >= cur) {
            ans[i] = cur + 1;
            cur++;
        } else {
            ans[i] = 0;
        }
        cur = max(cur, arr[i].second);
    }
    for (int i = 0; i < n; i++) {
        cout << ans[i] << " ";
    }
    cout << endl;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
