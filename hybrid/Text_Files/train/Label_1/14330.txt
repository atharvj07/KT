#include <bits/stdc++.h>
using namespace std;

struct p {
    int p, inf;
};

bool compare_inf(p a, p b) {
    if (a.p == b.p) return a.inf > b.inf;
    return a.p > b.p;
}

int main() {
    int n;
    cin >> n;
    vector<p> one, zero, both;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int inf;
        string s;
        cin >> s >> inf;
        if (s == "00") {
            zero.push_back({0, inf});
        } else if (s == "10") {
            one.push_back({1, inf});
        } else if (s == "01") {
            zero.push_back({0, inf});
        } else {
            both.push_back({2, inf});
        }
    }
    int szone = one.size(), szero = zero.size(), szboth = both.size();
    if (szone < n / 2 || szero < n / 2) {
        cout << 0 << endl;
        return 0;
    }
    sort(one.begin(), one.end(), compare_inf);
    sort(zero.begin(), zero.end(), compare_inf);
    sort(both.begin(), both.end(), compare_inf);

    int cntone = 0, cntzero = 0;
    for (int i = 0; i < min(n / 2, szone); i++) {
        ans += one[i].inf;
        cntone++;
    }
    for (int i = 0; i < min(n / 2, szero); i++) {
        ans += zero[i].inf;
        cntzero++;
    }
    if (cntone + cntzero < n / 2) {
        for (int i = 0; i < min(n / 2 - cntzero, szboth); i++) {
            ans += both[i].inf;
            cntzero++;
        }
    }

    cout << ans << endl;

    return 0;
}
