import sys, heapq

n, k = map(int, sys.stdin.readline().split())
t = []
for _ in range(n):
    a, b = map(int, sys.stdin.readline().split())
    t.append((a, b))
t.sort()
i = 0
ans = 0
b = []
while k:
    p = t[i]
    while i + 1 < n and p[0] == t[i + 1][0]:
        i += 1
        p = t[i]
    i += 1
    if not b:
        ans += p[0] * p[1]
    else:
        ans += p[0] * heapq.heappop(b)[1]
    heapq.heappush(b, p[1])
    k -= 1
print(ans)
