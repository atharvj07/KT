#include <vector>
#include <iostream>
using namespace std;

void dfs(vector<vector<int>>& adj, vector<int>& visited, int node, int parent){
    visited[node] = 1;
    for(int u: adj[node]){
        if(!visited[u]){
            if(parent != -1 && (adj[parent].size() > 2 || adj[u].size() > 2)) {
                cout << "unknown topology\n";
                exit(0);
            }
            dfs(adj, visited, u, node);
        }
        else if(u != parent){
            cout << "unknown topology\n";
            exit(0);
        }
    }
}

int main(){
    int n,m;
    cin >> n >> m;
    vector<vector<int>> adj(n+1);
    for(int i=1;i<=m;i++){
        int u,v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    vector<int> visited(n+1, 0);
    dfs(adj, visited, 1, -1);
    
    int cnt = 0, node = -1;
    for(int i=1;i<=n;i++){
        if(adj[i].size() > 2){
            cout << "unknown topology\n";
            return 0;
        }
        if(adj[i].size() == 1){
            cnt++;
            node = i;
        }
    }
    if(cnt == 2){
        cout << "bus topology\n";
    }
    else if(cnt == 0){
        cout << "ring topology\n";
    }
    else if(cnt == 1){
        cout << "star topology\n";
    }
    else{
        cout << "unknown topology\n";
    }
    return 0;
}
