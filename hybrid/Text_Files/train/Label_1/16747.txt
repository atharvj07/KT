#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    bool ok = true;
    for (int i = 0; i < n; i++) {
        if (i > 0 && a[i] < a[i - 1]) {
            ok = false;
        }
    }
    cout << (ok ? "Yes" : "No") << endl;
    return 0;
}
