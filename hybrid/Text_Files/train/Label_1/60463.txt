from sys import stdin
n=int(stdin.readline())
p=[list(map(int, stdin.readline().split())) for i in range(n)]
def ccw(p1,p2,p3):
    return (p2[0]-p1[0])*(p3[1]-p1[1])-(p3[0]-p1[0])*(p2[1]-p1[1])
def convexHull(p):
    p.sort()
    up=[]
    for i in p:
        while len(up)>=2 and ccw(up[-2], up[-1], i)<=0:
            up.pop()
        up.append(i)
    dn=[]
    for i in reversed(p):
        while len(dn)>=2 and ccw(dn[-2], dn[-1], i)<=0:
            dn.pop()
        dn.append(i)
    return up+dn[:-1]
def area(p):
    n=len(p)
    a=0
    for i in range(n):
        a+=p[i][0]*p[(i+1)%n][1]-p[(i+1)%n][0]*p[i][1]
    return abs(a)/2
p=convexHull(p)
print("%.9f"%(area(p)))
