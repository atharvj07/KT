import sys

inp = sys.stdin.readlines()
n = int(inp[0][:-1])
arr = []
b = {}
for i in range(0,n):
    x = inp[i+1][:-1].split(" ")
    a = int(x[0])
    bb = int(x[1])
    arr.append([a,bb])
    b[bb] = i
    b[a] = i

ans = []
for i in range(1,2*n+1):
    if i in b:
        ans.append(b[i]+1)

print(len(ans))
print(*ans)
