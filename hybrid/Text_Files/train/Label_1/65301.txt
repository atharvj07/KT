n = int(input())
s = input()
t = input()

s_a, s_b, t_a, t_b = [], [], [], []

for i in range(n):
    if s[i] == 'a' and t[i] == 'b':
        s_a.append(i + 1)
    elif s[i] == 'b' and t[i] == 'a':
        s_b.append(i + 1)

if (len(s_a) + len(s_b)) % 2 != 0:
    print(-1)
else:
    operations = []
    if len(s_a) % 2:
        operations.append([s_a[-1], s_a[-1]])
        s_a.append(s_a.pop())
        s_b.append(s_a[-1])

    for i in range(0, len(s_a) - 1, 2):
        operations.append([s_a[i], s_a[i + 1]])
    for i in range(0, len(s_b) - 1, 2):
        operations.append([s_b[i], s_b[i + 1]])

    print(len(operations))
    for op in operations:
        print(op[0], op[1])
