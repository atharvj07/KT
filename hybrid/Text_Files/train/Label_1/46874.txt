import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    long[] birth = new long[n];
    long[] death = new long[n];
    long maxAlive = 0;
    long year = 0;

    Map<Long, Long> alive = new HashMap<>();
    for (int i = 0; i < n; i++) {
      birth[i] = scanner.nextInt();
      death[i] = scanner.nextInt();
      alive.put(birth[i], alive.getOrDefault(birth[i], 0L) + 1);
      alive.put(death[i], alive.getOrDefault(death[i], 0L) - 1);
    }

    long aliveCount = 0;
    for (long key : alive.keySet()) {
      aliveCount += alive.get(key);
      if (aliveCount > maxAlive) {
        maxAlive = aliveCount;
        year = key;
      }
    }

    System.out.println(year + " " + maxAlive);
  }
}
