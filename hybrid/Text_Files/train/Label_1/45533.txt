import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int vX = in.nextInt();
        int vY = in.nextInt();
        int pX = in.nextInt();
        int pY = in.nextInt();
        int wX1 = in.nextInt();
        int wY1 = in.nextInt();
        int wX2 = in.nextInt();
        int wY2 = in.nextInt();
        int mX1 = in.nextInt();
        int mY1 = in.nextInt();
        int mX2 = in.nextInt();
        int mY2 = in.nextInt();
        if ((wX1 == vX && wY1 <= vY && vY <= wY2) || (wX2 == vX && wY1 <= vY && vY <= wY2)) {
            System.out.println("NO");
        } else if ((mY1 == vX && mX1 <= vY && vY <= mX2) || (mY2 == vX && mX1 <= vY && vY <= mX2)) {
            System.out.println("NO");
        } else {
            if (pX == vX) {
                if (vY > pY) {
                    System.out.println("NO");
                } else if (vY < pY) {
                    if (mY1 == mX1) {
                        System.out.println("NO");
                    } else if (vY > mY1 && vY < mY2) {
                        System.out.println("NO");
                    } else {
                        System.out.println("YES");
                    }
                } else {
                    System.out.println("YES");
                }
            } else if (pY == vY) {
                if (vX > pX) {
                    System.out.println("NO");
                } else if (vX < pX) {
                    if (mX1 == mY1) {
                        System.out.println("NO");
                    } else if (vX > mX1 && vX < mX2) {
                        System.out.println("NO");
                    } else {
                        System.out.println("YES");
                    }
                } else {
                    System.out.println("YES");
                }
            } else {
                System.out.println("NO");
            }
        }
    }
}
