#include<iostream>
#include<string>
#include<sstream>
using namespace std;
int a[101][101],n,m;
bool check()
{
    for(int i=1;i<=n;i++)
        for(int j=1;j<=m;j++)
        {
            if(a[i][j]=='*')
            {
                if(a[i+1][j]>='1'&&a[i+1][j]<='8')
                    a[i+1][j]++;
                if(a[i+1][j+1]>='1'&&a[i+1][j+1]<='8')
                    a[i+1][j+1]++;
                if(a[i][j+1]>='1'&&a[i][j+1]<='8')
                    a[i][j+1]++;
                if(a[i-1][j+1]>='1'&&a[i-1][j+1]<='8')
                    a[i-1][j+1]++;
                if(a[i-1][j]>='1'&&a[i-1][j]<='8')
                    a[i-1][j]++;
                if(a[i-1][j-1]>='1'&&a[i-1][j-1]<='8')
                    a[i-1][j-1]++;
                if(a[i][j-1]>='1'&&a[i][j-1]<='8')
                    a[i][j-1]++;
                if(a[i+1][j-1]>='1'&&a[i+1][j-1]<='8')
                    a[i+1][j-1]++;
            }
        }
    for(int i=1;i<=n;i++)
        for(int j=1;j<=m;j++)
            if(a[i][j]!='*'&&(a[i][j]<'0'||a[i][j]>'8'))
                return false;
    return true;
}
int main()
{
    int t;
    cin>>n>>m;
    string s;
    for(int i=1;i<=n;i++)
    {
        cin>>s;
        for(int j=0;j<s.size();j++)
        {
            if(s[j]=='*')
                a[i][j+1]='*';
            else if(s[j]>='0'&&s[j]<='8')
                a[i][j+1]=s[j]-'0';
            else
                a[i][j+1]=s[j];
        }
    }
    if(check())
        cout<<"YES"<<endl;
    else
        cout<<"NO"<<endl;
    return 0;
}
