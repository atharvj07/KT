#include <bits/stdc++.h>
using namespace std;

vector<pair<string, int> > v;
map<string, int> mp;
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n - 1; i++) {
    string s;
    cin >> s;
    v.push_back({s, 1});
  }
  for (int i = 0; i < n - 1; i++) {
    string s;
    cin >> s;
    v.push_back({s, -1});
  }
  sort(v.begin(), v.end());
  for (auto it: v) {
    mp[it.first] += it.second;
  }
  string ans = "";
  for (auto it: v) {
    if (mp[it.first] > 0) {
      ans += 'P';
      mp[it.first]--;
    } else {
      ans += 'S';
    }
  }
  cout << ans << endl;
}
