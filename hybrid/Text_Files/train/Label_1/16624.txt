#include <iostream>
using namespace std;
int main() {
  long long a, b;
  int c;
  cin >> a >> b >> c;
  long long l = 0;
  long long r = b * a;
  int d = -1;
  while (l <= r) {
    long long mid = (l + r) / b;
    if (a * mid % b == a % b) {
      long long x = (a * mid - a % b) * 10;
      d = mid + 1;
      while (x >= b) x -= b;
      if (x == c)
        break;
      if (x < c)
        l = mid + 1;
      else
        r = mid - 1;
    } else {
      l = mid + 1;
    }
  }
  cout << d << endl;
}
