from functools import reduce
n, k = map(int, input().split())
l = sorted([input() for i in range(n)])
s = [l[0]]
for x in l[1:]:
    if reduce(lambda a,b: a and b, [s[-len(y):] == y for y in l if y.startswith(s)]):
        s += [x]
    else:
        s += [x] * (k - 1)
print(['Second', 'First'][(k - 1) % len(s)])
