import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        sc.nextLine();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            sc.nextLine();
            String s = sc.nextLine();
            boolean[] vis = new boolean[26];
            for (int j = 0; j < n; j++) {
                vis[s.charAt(j) - 'a'] = true;
            }
            for (int j = 0; j < 26; j++) {
                if (!vis[j]) {
                    System.out.println((char) ('a' + j));
                    break;
                }
            }
        }
    }
}
