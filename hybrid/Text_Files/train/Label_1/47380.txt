import java.util.Scanner;

public class Mancala {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] holes = new int[14];
        int maxScore = 0;

        // Read the initial state of the board
        for (int i = 0; i < 14; i++) {
            holes[i] = scanner.nextInt();
        }

        // Find the hole with the maximum number of stones
        int maxStones = 0;
        int maxHole = -1;
        for (int i = 0; i < 14; i++) {
            if (holes[i] > maxStones && holes[i] % 2 != 0) {
                maxStones = holes[i];
                maxHole = i;
            }
        }

        // Simulate the move and calculate the score
        int score = 0;
        int currentHole = maxHole;
        while (maxStones > 0) {
            currentHole = (currentHole + 1) % 14;
            holes[currentHole]++;
            maxStones--;
            if (holes[currentHole] % 2 == 0) {
                score += holes[currentHole];
                holes[currentHole] = 0;
            }
        }

        // Output the maximum score
        System.out.println(score);
    }
}
