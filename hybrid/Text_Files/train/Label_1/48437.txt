import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    int n = scan.nextInt();
    int k = scan.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = scan.nextInt();
    }

    long ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        int xor = a[i] ^ a[j];
        int cnt = 0;
        for (int l = 0; l < 31; l++) {
          if (((xor >> l) & 1) == 1) {
            cnt++;
          }
        }
        if (cnt == k) {
          ans++;
        }
      }
    }

    System.out.println(ans);
  }
}
