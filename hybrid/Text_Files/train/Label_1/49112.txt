import java.util.Scanner;

public class JapaneseCrossword {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String row = scanner.next();

        int groupsCount = 0;
        int groupSize = 0;
        for (char ch : row.toCharArray()) {
            if (ch == 'B') {
                groupSize++;
            } else {
                if (groupSize > 0) {
                    groupsCount++;
                }
                groupSize = 0;
            }
        }
        if (groupSize > 0) {
            groupsCount++;
        }

        System.out.println(groupsCount);

        if (groupsCount > 0) {
            int[] groups = new int[groupsCount];
            groupsCount = 0;
            groupSize = 0;
            for (char ch : row.toCharArray()) {
                if (ch == 'B') {
                    groupSize++;
                } else {
                    if (groupSize > 0) {
                        groups[groupsCount++] = groupSize;
                    }
                    groupSize = 0;
                }
            }
            if (groupSize > 0) {
                groups[groupsCount++] = groupSize;
            }

            for (int i = 0; i < groupsCount; i++) {
                System.out.print(groups[i] + " ");
            }
        }
    }
}
