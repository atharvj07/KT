#include <bits/stdc++.h>
using namespace std;
int n,W,H,w[1001],h[1001],a[1001],b[1001];
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin>>n;
    for(int i=0;i<n;++i){
        cin>>w[i]>>h[i];
        a[i]=w[i];
        b[i]=h[i];
    }
    sort(a,a+n);
    sort(b,b+n);
    int ans=a[0]*b[n-1];
    for(int i=1,x=w[0],y=h[n-1];i<=n;++i){
        x+=a[i];
        ans=min(ans,x*y);
        swap(x,y);
        y+=b[n-1-i];
        ans=min(ans,x*y);
    }
    cout<<ans<<'\n';
    return 0;
}
