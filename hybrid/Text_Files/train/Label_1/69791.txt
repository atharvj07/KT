import sys
input=sys.stdin.readline

n,m,d=map(int,input().split())
a=[0]*n
b=[0]*m
for i in range(n):
    c,w=map(int,input().split())
    a[i]=(c,-w)
for i in range(m):
    c,w=map(int,input().split())
    b[i]=(c,-w)
a.sort()
b.sort()
ans=0
w=0
i=n-1
j=m-1
while i>=0 and j>=0:
    if -a[i][1]<=w:
        ans+=a[i][0]
        w-=a[i][1]
        i-=1
    elif -b[j][1]<=w:
        ans+=b[j][0]
        w-=b[j][1]
        j-=1
    else:
        break
if w<=d:
    ans+=a[i][0]
elif i>=0:
    ans+=a[i][0]
else:
    print(0)
    exit()
w=0
i=n-1
j=m-1
while i>=0 and j>=0:
    if -b[j][1]<=w:
        ans+=b[j][0]
        w-=b[j][1]
        j-=1
    elif -a[i][1]<=w:
        ans+=a[i][0]
        w-=a[i][1]
        i-=1
    else:
        break
if w<=d:
    ans+=b[j][0]
elif j>=0:
    ans+=b[j][0]
print(ans)
