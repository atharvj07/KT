from collections import Counter

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    count = Counter(a)
    if min(a) > n or max(a) < n or count[0] > 0 or any(count[x] > 1 for x in range(1, n + 1)):
        print(-1)
    else:
        print(sum(count[x] == 0 for x in range(1, n + 1)))
