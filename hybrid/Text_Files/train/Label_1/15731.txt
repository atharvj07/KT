#include <bits/stdc++.h>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) { std::cin >> a[i]; }
    std::sort(a.begin(), a.end());
    int l = 0, r = 0;
    int ans = 1;
    while (r < n) {
      if (a[r] - a[l] >= a[n - 1]) { ans = std::max(ans, r - l + 1); } else {
        l++;
      }
      r++;
    }
    std::cout << ans << '\n';
  }
  return 0;
}
