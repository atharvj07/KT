from collections import defaultdict
import hashlib

n = int(input())
a = []
for i in range(n):
    a.append(input())

b = defaultdict()

for i in range(n):
    for j in range(len(a[i])):
        s = a[i][j:]
        h = hashlib.sha1(s.encode()).hexdigest()
        if h in b:
            b[h].append(i)
        else:
            b[h] = [i]

ans = 0
for i in range(n):
    for j in range(len(a[i])):
        s = a[i][:j]
        h = hashlib.sha1(s.encode()).hexdigest()
        if h in b:
            for k in b[h]:
                if i == k: continue
                s1 = a[i] + a[k]
                s2 = a[k] + a[i]
                if s1 == s1[::-1] or s2 == s2[::-1]:
                    ans += 1

print(ans)
