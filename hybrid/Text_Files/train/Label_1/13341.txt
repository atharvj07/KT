#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<int> k(n);
    for (int i = 0; i < n; i++) cin >> k[i];
    vector<long long> c(m);
    for (int i = 0; i < m; i++) cin >> c[i];
    vector<long long> dp(m + 1, 1e18);
    dp[0] = 0;
    for (int i = 0; i < n; i++) {
      for (int j = m; j >= 0; j--) {
        if (j + k[i] <= m) dp[j + k[i]] = min(dp[j + k[i]], dp[j] + c[k[i] - 1]);
        dp[min(j + 1, k[i])] = min(dp[min(j + 1, k[i])], dp[j] + c[k[i] - 1]);
      }
    }
    cout << dp[m] << '\n';
  }
  return 0;
}
