import math

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

n = int(input())
a = list(map(int, input().split()))
t = []
for i in range(n):
    for j in range(i + 1, n):
        t.append(a[i] * a[j] // gcd(a[i], a[j]))
g = -1
for x in t:
    if g == -1:
        g = x
    g = gcd(g, x)
print(g)
