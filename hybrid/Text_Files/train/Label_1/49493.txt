import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long d = sc.nextLong();
        long n = sc.nextLong();
        int m = sc.nextInt();
        long[] stations = new long[m];
        long[] prices = new long[m];
        for (int i = 0; i < m; i++) {
            stations[i] = sc.nextLong();
            prices[i] = sc.nextLong();
        }
        long cur = 0;
        long minCost = 0;
        Arrays.sort(stations);
        Arrays.sort(prices);
        for (int i = 0; i < m; i++) {
            if (stations[i] - cur <= n) {
                minCost += (stations[i] - cur) * prices[i];
                cur = stations[i];
            } else {
                if (prices[i] >= prices[0]) {
                    System.out.println(-1);
                    return;
                } else {
                    minCost += (n - cur) * prices[i];
                    cur = stations[i];
                }
            }
        }
        if (d - cur <= n) {
            System.out.println(minCost);
        } else {
            System.out.println(-1);
        }
    }
}
