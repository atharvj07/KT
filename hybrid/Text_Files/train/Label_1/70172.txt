for _ in range(int(input())):
    n = int(input())
    s = input()
    if s == s[::-1]:
        print('YES')
    else:
        ok = True
        for i in range(n // 2 - 1):
            if s[i] != s[n - i - 1]:
                if s[i + 1] == s[n - i - 1] or s[i] == s[n - i - 2]:
                    ok &= True
                else:
                    ok &= False
        if s[n // 2 - 1] != s[n // 2]:
            ok &= False
        if ok:
            print('YES')
        else:
            print('NO')
