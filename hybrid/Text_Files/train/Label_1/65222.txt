def is_nearly_prime(x):
    if x == 1:
        return False

    for i in range(2, int(x ** 0.5) + 1):
        if x % i == 0:
            return True

    return False

for _ in range(int(input())):
    n = int(input())
    if n % 2 == 1 or n < 8:
        print("NO")
        continue

    a = 6
    b = 10
    c = min(n // 3, 221)
    d = n - a - b - c

    if is_nearly_prime(a) and is_nearly_prime(b) and is_nearly_prime(c):
        print("YES")
        print(a, b, c, d)
    else:
        print("NO")
