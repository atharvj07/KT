t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().strip().split())
    s = input().strip()
    first_star = s.find('*')
    last_star = s.rfind('*')
    if first_star == last_star:
        print(1)
        continue
    count = 2  # Include first and last '*' replacements
    current_pos = first_star
    while True:
        next_pos = current_pos + k
        if next_pos >= last_star:
            break
        while s[next_pos] != '*':
            next_pos -= 1
        count += 1
        current_pos = next_pos
    print(count)
