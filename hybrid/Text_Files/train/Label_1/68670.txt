n, m = map(int, input().split())
times = [0] * n
durations = [0] * n
for i in range(n):
    times[i], durations[i] = map(int, input().split())
moments = list(map(int, input().split()))
current_song = 0
current_time = 0
for i in moments:
    while current_time + durations[current_song] <= i:
        current_time += durations[current_song]
        current_song += 1
    print(current_song + 1)
