#include <iostream>
using namespace std;

int main() {
    long long x = 0;
    int l;
    cin >> l;
    while (l--) {
        string s;
        cin >> s;
        if (s == "for") {
            int n;
            cin >> n;
            while (n--) {
                string t;
                cin >> t;
                if (t == "add") x++;
                if (x < 0) {
                    cout << "OVERFLOW!!!" << endl;
                    return 0;
                }
            }
        } else if (s == "add")
            x++;
        if (x < 0) {
            cout << "OVERFLOW!!!" << endl;
            return 0;
        }
    }
    cout << x << endl;
    return 0;
}
