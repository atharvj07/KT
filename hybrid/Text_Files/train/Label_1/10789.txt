#include<bits/stdc++.h>
#include<vector>
#include<unordered_map>
using namespace std;
#define ll long long
#define fo(i,n) for(i=0;i<n;i++)
ll func(vector<ll> &st,vector<ll> &en,vector<ll> &uni,string t,string s)
{
    ll dp[st.size()][uni.size()];
    memset(dp, 0, sizeof(dp));
    dp[0][0] = 1;
    for (ll i = 0; i < st.size(); i++)
    {
        for (ll j = 0; j < uni.size(); j++)
        {
            //Skip
            dp[i + 1][j] += dp[i][j];
            //Put
            if (uni[j] == t[i])
            {
                dp[i + 1][j + 1] += dp[i][j];
            }
            //Remove
            if (st[i] <= en[i])
            {
                if (t[i] == s[en[i] - (st[i] - 1)])
                    dp[i + 1][j] += dp[en[i]][j];
            }
            else
            {
                if (t[i] == s[t.size() - (st[i] - en[i] - 1)])
                    dp[i + 1][j] += dp[en[i]][j];
            }
        }
    }
    return dp[st.size()][uni.size()];
}
int main()
{
    ll n;
    cin >> n;
    string s;
    cin >> s;
    // Storing start and end points of all unique elements
    ll i, j, k, m;
    vector<ll> st(s.size());
    vector<ll> en(s.size());
    unordered_map<char, ll> mp;
    for (i = 0; i < s.size(); i++)
    {
        if (mp.find(s[i]) == mp.end())
        {
            mp[s[i]] = 1;
        }
    }
    char c;
    for (auto it : mp)
    {
        c = it.first;
        vector<ll> uni;
        string t = "";
        ll ind = -1;
        for (i = n - 1; i >= 0; i--)
        {
            if (t == "")
            {
                t = s;
                reverse(t.begin(), t.end());
                for (j = 0; j < n; j++)
                {
                    t = t + s;
                }
            }
            if (ind == -1)
            {
                for (j = 0; j < t.size(); j++)
                {
                    if (t[j] == c)
                    {
                        ind = j;
                        uni.push_back(j);
                        break;
                    }
                }
            }
            else
            {
                for (j = ind + 1; j < t.size(); j++)
                {
                    if (t[j] == c)
                    {
                        uni.push_back(j);
                        break;
                    }
                }
            }
            if (uni.size() >= s.size())
                break;
        }
        for (j = 0; j < uni.size(); j++)
        {
            ll x = uni[j];
            st[j] = x;
            for (k = x + s.size(); k < t.size(); k += s.size())
            {
                if (t[k] != s[k - x])
                    break;
            }
            en[j] = k - 1;
        }
        cout << func(st, en, uni, t, s) << " ";
    }
    return 0;
}
