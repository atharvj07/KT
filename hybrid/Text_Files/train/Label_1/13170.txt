#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, w;
    cin >> n >> w;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.rbegin(), a.rend());
    vector<int> ans(n);
    int cur = 0;
    for (int i = 0; i < n; i++) {
        int x = min(a[i], w / 2);
        ans[i] = x;
        w -= x;
        cur += x;
    }
    if (cur < w) cout << -1 << endl;
    else {
        for (int x : ans) cout << x << " ";
        cout << endl;
    }
    return 0;
}
