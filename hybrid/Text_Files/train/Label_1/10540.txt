#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    vector<int> l(n), r(n);
    int m = a[0];
    for (int i = 0; i < n; ++i) {
        l[i] = m;
        m = min(m, a[i]);
    }
    m = a[n - 1];
    for (int i = n - 1; i >= 0; --i) {
        r[i] = m;
        m = min(m, a[i]);
    }
    for (int x = 1; x <= n; ++x) {
        int mx = 1e9;
        for (int i = 0; i + x - 1 < n; ++i) { mx = min(mx, max(l[i], r[i + x - 1])); }
        cout << mx << " ";
    }
    cout << "\n";
}
