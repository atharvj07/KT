#include <iostream>
using namespace std;
int main() {
  const int MAX_N = 500 + 1;
  char m[MAX_N][MAX_N];
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> m[i] + 1; }
  int ans = 0;
  for (int i = 2; i < n; i++) {
    for (int j = 2; j < n; j++) {
      if (m[i][j] == 'X' && m[i][j] == m[i - 1][j - 1] &&
          m[i][j] == m[i - 1][j + 1] && m[i][j] == m[i + 1][j - 1] &&
          m[i][j] == m[i + 1][j + 1])
        ans++;
    }
  }
  cout << ans << endl;
}
