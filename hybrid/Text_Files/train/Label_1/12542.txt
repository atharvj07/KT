#include <bits/stdc++.h>
using namespace std;
#define ll long long
int32_t main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<pair<int, int>> v(n);
  for (auto &i : v) cin >> i.first >> i.second;
  int ans = 0;
  for (int i = 0; i < 5; i++) {
    int on = 0;
    for (int j = 0; j < n; j++) {
      int t = ((i - v[j].second) % v[j].first + v[j].first) % v[j].first;
      if (t == 0 && s[j] == '1') on++;
    }
    ans = max(ans, on);
  }
  cout << ans << endl;
}
