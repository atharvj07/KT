#include<bits/stdc++.h>
using namespace std;
int n,k,ans=0,x=1;
char ar[105];
int main(){
    cin>>n>>k;
    cin>>ar;
    for(int i=0;i<n;i++){
        if(ar[i]=='Y') x=1;
        else if(ar[i]=='?') x=2;
        else{
            if(x==2) ans++;
            x=0;
        }
    }
    if(x==2) ans++;
    if(ans==k) cout<<"YES";
    else cout<<"NO";
}
