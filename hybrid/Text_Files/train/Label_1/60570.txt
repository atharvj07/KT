import sys
input = sys.stdin.readline

n = int(input())
adj = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

vis = [False] * (n+1)

def dfs(u, p):
    vis[u] = True
    res = 0
    for v in adj[u]:
        if v == p: continue
        if not vis[v]:
            res += 1 + dfs(v, u)
        else:
            res += 1
    return res

ans = dfs(1, 1)
print(ans)
