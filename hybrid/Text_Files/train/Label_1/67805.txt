from decimal import Decimal
import sys
rl = lambda : sys.stdin.readline().strip()
n, k = map(int,rl().split())
arr = list(map(int,rl().split()))
if k%2==n%2==0:
    x, A, B = 1, 1, 1
else:
    x, A, B = 0, 0, 1
for i in reversed(arr):
    x=(x*B)%1000000007
    A=(A*i)%1000000007
    B=(B*(1-i))%1000000007
print((x-A+1000000007)%1000000007)
