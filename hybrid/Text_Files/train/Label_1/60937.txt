def solve():
    n, k = map(int, input().split())
    p = list(map(int, input().split()))
    p = dict([(x, i) for i, x in enumerate(p)])
    e = [[0] * n for _ in range(n)]
    for i in range(0, n):
        for l in range(0, i):
            r = i
            if p[l] > p[r]:
                e[l][r] = 1
                e[r][l] = -1
    ans = 0
    for _ in range(k):
        l = i = 0
        r = n - 1
        while l < r:
            if l == r:
                i = l
                break
            if e[l][r]:
                ans += e[l][r]
                i += 1
            e[r][l] = 0
            e[l][r] = 0
            l += 1
            r -= 1
        for t in range(i, n):
            if p[t] < p[i]:
                e[t][i] = 1
                e[i][t] = -1
    print(ans / k)

solve()
