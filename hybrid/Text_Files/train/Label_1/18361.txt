#include <bits/stdc++.h>
using namespace std;
#define int long long
typedef pair<int, int> pi;

const int N = 2e5 + 5;
const int inf = 1e18 + 5;

vector<int> adj[N];
int val[N];
int c_xor[N], c_sz[N];
bool vis[N];

int dfs(int u, int p, int xor_taken) {
    vis[u] = true;
    c_xor[u] = val[u] ^ xor_taken;
    c_sz[u] = 1;
    for(int nbr: adj[u]) {
        if(nbr != p) {
            c_xor[u] = c_xor[u] ^ dfs(nbr, u, xor_taken);
            c_sz[u] += c_sz[nbr];
        }
    }
    return c_xor[u];
}

int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;
    while(t--) {
        int n, k;
        cin >> n >> k;
        for(int i = 1; i <= n; i++) {
            cin >> val[i];
            vis[i] = false;
            c_xor[i] = 0;
            c_sz[i] = 0;
            adj[i].clear();
        }

        for(int i = 1; i < n; i++) {
            int u, v;
            cin >> u >> v;
            adj[u].push_back(v);
            adj[v].push_back(u);
        }

        int xor_root = dfs(1, 1, 0);

        bool ok = true;
        for(int i = 1; i <= n; i++) {
            if(vis[i] == false) {
                ok = false;
            } else {
                if(c_xor[i] != xor_root) {
                    ok = false;
                }
            }
        }

        if(ok) {
            if(k == 2) {
                cout << "NO" << "\n";
            } else {
                cout << "YES" << "\n";
            }
        } else {
            cout << "NO" << "\n";
        }
    }

    return 0;
}
