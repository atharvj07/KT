import sys
input = sys.stdin.readline

n, m = map(int, input().split())
d = list(map(int, input().split()))
a = list(map(int, input().split()))

if any(a_i > n for a_i in a):
    print(-1)
    sys.exit()

not_prepared = set(range(1, m + 1))
days = [set() for _ in range(n + 1)]
for i in range(n):
    if d[i] != 0:
        days[i + 1].add(d[i])
        not_prepared.discard(d[i])

if not_prepared:
    print(-1)
    sys.exit()

dp = [10 ** 9] * (n + 1)
dp[0] = 0
for day in range(1, n + 1):
    for subject in days[day]:
        dp[day] = min(dp[day], dp[day - a[subject - 1]] + 1)

if dp[n] != 10 ** 9:
    print(dp[n])
else:
    print(-1)
