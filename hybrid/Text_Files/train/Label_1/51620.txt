import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int y = sc.nextInt(), k = sc.nextInt(), n = sc.nextInt();
        if ((n - y) % k == 0) {
            for (int i = y; i <= n - y; i += k) {
                System.out.print(i + " ");
            }
        } else {
            System.out.println(-1);
        }
    }
}
