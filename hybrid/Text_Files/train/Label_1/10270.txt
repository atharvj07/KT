#include <cstdio>
#include <cstring>
char s[110][110];
int c[110][110][5][2];
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%s", s[i] + 1);
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      for (int k = 1; k <= 4; k++) { c[i][j][k][0] = c[i][j][k][1] = 0; }
      if (s[i][j] == '#') c[i][j][1][0] = 1;
    }
  }
  int ans = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == '.') continue;
      if (c[i][j][1][0]) {
        if (j + 1 <= n) c[i][j + 1][2][0] = 1, c[i][j + 1][2][1] = j;
        if (i + 1 <= n) c[i + 1][j][3][0] = 1, c[i + 1][j][3][1] = i;
        if (i + 2 <= n) c[i + 2][j][4][0] = 1, c[i + 2][j][4][1] = i;
        if (j + 2 <= n) c[i][j + 2][4][0] = 1, c[i][j + 2][4][1] = j;
      } else {
        ans = 0;
        break;
      }
    }
  }
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == '#') cnt++;
    }
  }
  if (cnt % 5 == 0) ans = ans && !cnt;
  puts(ans ? "YES" : "NO");
  return 0;
}
