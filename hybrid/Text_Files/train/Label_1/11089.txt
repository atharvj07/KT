#include <bits/stdc++.h>
using namespace std;

struct node {
    bool leaf;
    unordered_map<char, node*> next;
};

node* build() {
    return new node{false};
}

void insert(node* root, string const& s) {
    auto cur = root;
    for (char c : s) {
        if (!cur->next.count(c)) {
            cur->next[c] = build();
        }
        cur = cur->next[c];
    }
    cur->leaf = true;
}

void dfs(node* root, string const& pre, vector<string>& ans) {
    if (root->leaf) {
        ans.emplace_back(pre);
    }
    for (auto const& [c, nx] : root->next) {
        dfs(nx, pre + c, ans);
    }
}

int main() {
    int n;
    cin >> n;
    vector<string> hs(n);
    for (int i = 0; i < n; ++i) {
        cin >> hs[i];
        hs[i] = hs[i].substr(1);
    }
    sort(hs.begin(), hs.end());
    node* r = build();
    for (auto const& s : hs) {
        insert(r, s);
    }
    vector<string> ans;
    dfs(r, "", ans);
    for (auto const& s : ans) {
        cout << "#" << s << '\n';
    }
}
