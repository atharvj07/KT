import java.util.*;

class Main {
    static int ROW, COL, ans;
    static char[][] map;
    static boolean[][] visited;
    static int[] dr = {-1, 1, 0, 0};
    static int[] dc = {0, 0, -1, 1};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        ROW = sc.nextInt();
        COL = sc.nextInt();
        map = new char[ROW][COL];
        visited = new boolean[ROW][COL];
        ans = Integer.MAX_VALUE;
        for (int i = 0; i < ROW; i++) {
            String line = sc.next();
            for (int j = 0; j < COL; j++) {
                map[i][j] = line.charAt(j);
            }
        }
        for (int i = 0; i < ROW; i++) {
            for (int j = 0; j < COL; j++) {
                if (map[i][j] != '#') {
                    bfs(i, j);
                }
            }
        }
        if (ans == Integer.MAX_VALUE) {
            ans = -1;
        }
        System.out.println(ans);
    }

    static void bfs(int r, int c) {
        Queue<int[]> queue = new LinkedList<>();
        queue.offer(new int[]{r, c});
        int steps = 0;
        while (!queue.isEmpty()) {
            int size = queue.size();
            while (size-- > 0) {
                int[] pos = queue.poll();
                int row = pos[0];
                int col = pos[1];
                visited[row][col] = true;
                if (map[row][col] != '.' && map[row][col] != map[r][c]) {
                    ans = Math.min(ans, steps);
                }
                for (int i = 0; i < 4; i++) {
                    int newRow = row + dr[i];
                    int newCol = col + dc[i];
                    if (newRow >= 0 && newRow < ROW && newCol >= 0 && newCol < COL && !visited[newRow][newCol] && map[newRow][newCol] != '#') {
                        queue.offer(new int[]{newRow, newCol});
                    }
                }
            }
            steps++;
        }
    }
}
