import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = in.nextInt();
        }
        int q = in.nextInt();
        int t[] = new int[q];
        for (int i = 0; i < q; i++) {
            t[i] = in.nextInt();
        }
        int c = 0;
        for (int v : t) {
            c += Arrays.binarySearch(s, v) >= 0 ? 1 : 0;
        }
        System.out.println(c);
    }
}
