for t in range(int(input())):
    n,d = map(int,input().split())
    a = list(map(int,input().split()))
    for x in range(n):
        if a[x] or a[(x-d+n)%n]:break
    else:print(0);continue
    for steps in range(1,n+1):
        for x in range(n):
            a[x] &= a[(x-d+n)%n]
        b = set(a)
        if len(b)==2 and 0 in b:break
    print(steps if 0 in b else -1)
