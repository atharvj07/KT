def count_years_with_one_zero(a, b):
    def count_helper(x):
        if x == 0:
            return 0
        length = x.bit_length() - 1
        return (1 << (length - 1)) * length + count_helper(x - (1 << length))

    return count_helper(b + 1) - count_helper(a)

a, b = map(int, input().split())
print(count_years_with_one_zero(a, b))
