t = int(input())
for _ in range(t):
    n = int(input())
    arr = list(map(int, input().split()))
    sum_arr = sum(arr)
    xor_arr = 0
    for num in arr:
        xor_arr ^= num
    if sum_arr == 2 * xor_arr:
        print(0)
        print()
    else:
        if xor_arr * 2 == sum_arr + xor_arr:
            print(1)
            print(xor_arr)
        else:
            print(2)
            print(xor_arr, sum_arr + xor_arr)
