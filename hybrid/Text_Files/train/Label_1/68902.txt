import sys
from heapq import heappush, heappop
input = sys.stdin.readline

for _ in range(int(input())):
    n, p = map(int, input().split())
    a = list(map(int, input().split()))
    g = [[] for _ in range(n)]
    for i in range(n-1):
        for j in range(i+1, n):
            mn = min(a[i:j+1])
            if mn == a[i] or gcd(mn, *a[i+1:j+1]) == mn:
                g[i].append((mn, j))
                g[j].append((mn, i))
    for i in range(n-1):
        g[i].append((p, i+1))
        g[i+1].append((p, i))

    vis = [False] * n
    mn = 10**12
    for i in range(n):
        if vis[i]:
            continue
        mn = 0
        q = [(0, i)]
        while q:
            w, u = heappop(q)
            if vis[u]:
                continue
            vis[u] = True
            mn += w
            for nw, v in g[u]:
                if not vis[v]:
                    heappush(q, (nw, v))
    print(mn)

def gcd(a, *b):
    ans = a
    for x in b:
        ans = gcd2(ans, x)
    return ans

def gcd2(a, b):
    if b == 0:
        return a
    return gcd2(b, a % b)
