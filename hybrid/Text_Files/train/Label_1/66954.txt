x, y, n = map(int, input().split())
minimal = -1
ansA = 0
ansB = 1
for i in range(1, n + 1):
    if i > y * (n + 1) // x:
        break
    A = i * x // y
    if abs(A / i - x / y) < minimal or minimal == -1:
        minimal = abs(A / i - x / y)
        ansA = A
        ansB = i
print(f"{ansA}/{ansB}")
