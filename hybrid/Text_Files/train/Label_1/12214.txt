#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> t(n), a(n), ind(n + 1, -1);
  for (int i = 0; i < n; i++) { cin >> t[i]; }
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<int> path;
  for (int i = n - 1; i >= 0; i--) {
    if (t[i] == 1) {
      path.push_back(i + 1);
      break;
    }
    if (a[i] == 0) continue;
    ind[a[i]] = i + 1;
  }
  int node = ind[path.back()];
  while (node != -1) {
    path.push_back(node);
    node = ind[node];
  }
  cout << path.size() << endl;
  for (auto& x : path) cout << x << " ";
  cout << endl;
  return 0;
}
