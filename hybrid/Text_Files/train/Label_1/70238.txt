import sys

t = int(input())
for _ in range(t):
    n = int(input())
    a = sorted(list(map(int, input().split())))
    if n % 2 == 0:
        for i in range(n - 1):
            print(a[i], a[n - 1 - i], end=' ')
    else:
        for i in range(n // 2):
            print(a[i], a[n - 1 - i], end=' ')
        print(a[n // 2])
    print()
