from collections import Counter
import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = []
for i in range(n):
  a.append(Counter(input()))

odd = 0
c = 0
for i in range(n):
  for j in range(m):
    a[i][0] += a[i][j]
  if a[i][0] % 2 == 0:
    odd += 1
  a[i][0] = a[i][0] % 2

if odd % 2 == 0:
  print(-1)
else:
  c = 0
  for i in range(n):
    for j in range(m - 1):
      if (j + 1) % 2 == 1:
        c += a[i][j + 1]
  if c % 2 == 1:
    print(0)
  else:
    print(1)
