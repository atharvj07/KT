#include <bits/stdc++.h>
using namespace std;
#define int long long
const int MAXN = 3e5 + 10;
char str[MAXN];
main() {
    int n;
    cin >> n;
    cin >> str;
    int l = -1, r = -1;
    for (int i = 0; i + 1 < n; i++) {
        if (str[i] > str[i + 1]) {
            l = i;
            break;
        }
    }
    for (int i = n - 1; i > 0; i--) {
        if (str[i] < str[i - 1]) {
            r = i;
            break;
        }
    }
    if (l == -1) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    cout << l + 1 << " " << r + 1 << endl;
}
