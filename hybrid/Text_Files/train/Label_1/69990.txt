import math
from collections import defaultdict

def solve(n, a, b):
    tasks = sorted(zip(a, b), reverse=True)
    grouped_tasks = defaultdict(list)
    for power, processors in tasks:
        grouped_tasks[processors].append(power)
    
    best_avgs = []
    for processors, powers in grouped_tasks.items():
        for i in range(0, len(powers), 2):
            best_avgs.append((sum(powers[i:i+2]), processors))
    
    best_avgs.sort(reverse=True)
    total_power, total_processors = 0, 0
    for power, processors in best_avgs:
        total_power += power
        total_processors += processors
        threshold = math.ceil((total_power / total_processors) * 1000)
    
    return threshold

n = int(input().strip())
a = list(map(int, input().strip().split()))
b = list(map(int, input().strip().split()))

print(solve(n, a, b))
