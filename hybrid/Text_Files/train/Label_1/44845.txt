import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int m = Integer.parseInt(line[0]);
        int n = Integer.parseInt(line[1]);
        int k = Integer.parseInt(br.readLine());
        char[][] map = new char[m][n];
        for (int i = 0; i < m; i++) {
            map[i] = br.readLine().toCharArray();
        }
        int[][][] cnt = new int[m + 1][n + 1][3];
        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= n; j++) {
                for (int x = 0; x < 3; x++) {
                    cnt[i][j][x] = cnt[i - 1][j][x] + cnt[i][j - 1][x] - cnt[i - 1][j - 1][x];
                }
                switch (map[i - 1][j - 1]) {
                    case 'J':
                        cnt[i][j][0]++;
                        break;
                    case 'O':
                        cnt[i][j][1]++;
                        break;
                    case 'I':
                        cnt[i][j][2]++;
                        break;
                }
            }
        }
        for (int i = 0; i < k; i++) {
            line = br.readLine().split(" ");
            int aj = Integer.parseInt(line[0]);
            int bj = Integer.parseInt(line[1]);
            int cj = Integer.parseInt(line[2]);
            int dj = Integer.parseInt(line[3]);
            for (int x = 0; x < 3; x++) {
                int ans = cnt[cj][dj][x] - cnt[aj - 1][dj][x] - cnt[cj][bj - 1][x] + cnt[aj - 1][bj - 1][x];
                System.out.print(ans + " ");
            }
            System.out.println();
        }
    }
}
