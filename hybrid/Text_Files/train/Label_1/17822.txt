#include<bits/stdc++.h>
using namespace std;
int32_t main()
{
    int n,k,i,ans=0,cnt=0;
    cin>>n>>k;
    char s[n+5];
    cin>>s;
    for(i=0;i<n;i++)
    {
        if(s[i]==s[i+1])cnt++;
        else
        {
            ans=max(ans,cnt+1);
            cnt=0;
        }
    }
    ans=max(ans,cnt+1);

    if(ans+k>=n)cout<<n<<endl;
    else if(ans%2==0)cout<<ans+k<<endl;
    else cout<<ans+k-1<<endl;
    return 0;
}
