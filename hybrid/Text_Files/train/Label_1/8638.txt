#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int sum = 0;
    int max_sum = 0;
    for (int i = 0; i < n; i++) {
        sum = max(x * a[i], sum + a[i]);
        max_sum = max(sum, max_sum);
    }
    cout << max_sum << endl;
    return 0;
}
