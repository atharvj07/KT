n, s = map(int, input().split())
a = list(map(int, input().split()))

a.sort()
mid = n // 2
operations = 0

if a[mid] < s:
    for i in range(mid, n):
        if a[i] < s:
            operations += s - a[i]
        else:
            break
elif a[mid] > s:
    for i in range(mid, -1, -1):
        if a[i] > s:
            operations += a[i] - s
        else:
            break

print(operations)
