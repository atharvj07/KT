#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 998244353;
int main() {
    int n;
    cin >> n;
    ll ans = 1;
    for (int i = 1; i <= n; i++) {
        ans = (ans * (n - i + 1)) % mod;
    }
    for (int i = 1; i <= n; i++) {
        ans = (ans * i) % mod;
    }
    cout << ans << endl;
    return 0;
}
