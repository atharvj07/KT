#include <iostream>

using namespace std;

int main() {
    int s, v1, v2, t1, t2;
    cin >> s >> v1 >> v2 >> t1 >> t2;

    int time1 = t1 + v1 * s;
    int time2 = t2 + v2 * s;

    if (time1 < time2) {
        cout << "First" << endl;
    } else if (time2 < time1) {
        cout << "Second" << endl;
    } else {
        cout << "Friendship" << endl;
    }

    return 0;
}
