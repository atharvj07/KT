#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> c(n * k);
    for (int& i : c) cin >> i;
    vector<int> f(n);
    for (int& i : f) cin >> i;
    vector<int> h(k);
    for (int& i : h) cin >> i;
    sort(c.begin(), c.end());
    sort(f.begin(), f.end());
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int low = lower_bound(c.begin(), c.end(), f[i]) - c.begin();
        int high = upper_bound(c.begin(), c.end(), f[i]) - c.begin() - 1;
        if (low + k <= high) {
            ans += h[k];
        } else {
            int left = k - (high - low + 1);
            ans += h[low - min(k, low) + 1] + h[left];
        }
    }
    cout << ans << endl;
}
