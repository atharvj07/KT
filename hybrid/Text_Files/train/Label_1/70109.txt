import numpy as np

n, m = list(map(int, input().split()))
mat = [list(map(int, list(input()))) for i in range(n)]
can = [True for i in range(n)]

for i in range(m):
    st = True
    for j in range(n):
        if mat[j][i] == 1:
            st = False
    for j in range(n):
        if mat[j][i] == 1:
            can[j] = can[j] and st

ans = "YES"
for t in can:
    if t == True:
        ans = "NO"

print(ans)
