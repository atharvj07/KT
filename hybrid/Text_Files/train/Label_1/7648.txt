#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll mod=998244353;
ll n,m;
ll f[(int)2e5+5],ifac[(int)2e5+5];
ll qpow(ll b,ll e)
{
    ll r=1;
    while(e)
    {
        if(e&1)
            r=(r*b)%mod;
        b=(b*b)%mod;
        e>>=1;
    }
    return r;
}
void pre()
{
    f[0]=f[1]=ifac[0]=ifac[1]=1;
    for(ll i=2;i<(int)2e5+1;i++)
    {
        f[i]=(f[i-1]*i)%mod;
        ifac[i]=(ifac[i-1]*qpow(i,mod-2))%mod;
    }
}
ll C(ll n,ll k)
{
    if(n<k)
        return 0;
    return (f[n]*ifac[n-k]%mod*ifac[k]%mod)%mod;
}
int main()
{
    pre();
    cin>>n>>m;
    vector<ll> a(n),b(m);
    for(ll &i:a)
        cin>>i;
    for(ll &i:b)
        cin>>i;
    ll ans=0,cnt=0,last=-2e9;
    for(ll i=0;i<n;i++)
    {
        if(a[i]==b[cnt])
        {
            cnt++;
            ans+=C(cnt+last,cnt);
            ans%=mod;
            last=cnt;
        }
    }
    if(cnt==m)
        cout<<ans;
    else
        cout<<0;
    cout<<endl;
    return 0;
}
