#include<iostream>
#include<bits/stdc++.h>
using namespace std;
int32_t main() {
	int n, m;
	cin >> n >> m;
	m--;
	int fac[n + 1];
	fac[0] = 1;
	for (int i = 1; i <= n; i++) fac[i] = fac[i - 1] * i;
	vector<int> b(n);
	for (int i = 0; i < n; i++) {
		int p = (m / fac[n - i - 1]);
		b[i] = p + 1;
		m = m % fac[n - i - 1];
	}
	for (int i = 0; i < n; i++) cout << b[i] << ' ';
	cout << endl;
}
