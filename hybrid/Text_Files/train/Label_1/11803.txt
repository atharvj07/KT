#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll MAX = 1e9;

unordered_map<ll, ll> mp;

ll f(ll a) {
  if (a % 2 == 0) return a / 2;
  return a - 1;
}

int main() {
  ll n, k;
  cin >> n >> k;
  for (ll i = 1; i <= n; i++) {
    ll a = i;
    ll cnt = 0;
    while (a >= 1) {
      if (a <= MAX) { mp[a] = 1; }
      a = f(a);
      cnt++;
    }
  }
  ll ans = 0;
  for (auto &p : mp) {
    if (p.second >= k) ans = p.first;
  }
  cout << ans << endl;
  return 0;
}
