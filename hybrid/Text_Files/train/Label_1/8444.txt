#include<bits/stdc++.h>
#define int long long
#define endl '\n'
using namespace std;
int32_t main()
{
    int n, sum=0, cnt[1000005]={0}, mx=0; cin>>n;
    for(int i=0;i<n;i++)
    {
        int x; cin>>x;
        sum+=x*i-cnt[x]*x;
        cnt[x]++;
        mx=max(mx,x);
    }
    for(int i=1;i<=mx;i++)
    {
        sum+=cnt[i]*(cnt[i]-1)*i/2;
    }
    cout<<sum<<endl;
    return 0;
}
