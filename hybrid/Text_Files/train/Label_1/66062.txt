import sys
input = sys.stdin.readline

n, m = map(int, input().split())
friends = [set(map(int, input().split())) for _ in range(n)]
pizzas = [set(map(int, input().split()[1:])) for _ in range(m)]
max_cnt = 0
ans = []
for i in range(m):
    for j in range(i + 1, m):
        cnt = sum(ing in pizzas[i] or ing in pizzas[j] for fr in friends for ing in fr)
        if cnt > max_cnt:
            max_cnt = cnt
            ans = [i + 1, j + 1]
print(*ans)
