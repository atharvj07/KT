#include <iostream>
#include <vector>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  vector<int> diff(n);
  for (int i = 1; i < n; i++) {
    diff[i] = (s[i] != s[i - 1]);
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (diff[i] == 0) continue;
    ans++;
    int j = i;
    while (j < n && diff[j] != 0) {
      j++;
    }
    i = j - 1;
  }
  cout << ans << endl;
  return 0;
}
