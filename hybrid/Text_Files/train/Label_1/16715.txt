#include <bits/stdc++.h>
using namespace std;

const long long p = 998244353LL;
long long dp[1000001], fac[1000001];

long long fastpow(long long base, long long pow) {
    long long ans = 1;
    while (pow > 0) {
        if (pow % 2 == 1) { ans = (ans * base) % p; }
        base = (base * base) % p;
        pow >>= 1;
    }
    return ans;
}

int main() {
    int n;
    cin >> n;
    long long sum = 1LL * n * (n + 1) / 2;
    fac[0] = dp[0] = 1;
    for (int i = 1; i <= n; i++) {
        fac[i] = (i * fac[i - 1]) % p;
        dp[i] = (fastpow(2, i - 1) - 1 + p) % p;
    }
    long long ans = (1LL * fac[n] * dp[n]) % p;
    cout << ans << endl;
    return 0;
}
