#include <cstdint>
#include <iostream>
#include <vector>

template<typename T>
std::vector<int64_t> prime_factorize(T n) {
  std::vector<int64_t> factors;
  for (int64_t i = 2; i * i <= n; ++i) {
    while (n % i == 0) {
      factors.push_back(i);
      n /= i;
    }
  }
  if (n > 1) {
    factors.push_back(n);
  }
  return factors;
}

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int64_t a, b;
    std::cin >> a >> b;
    if (a > b || (a & 1) != (b & 1)) {
      std::cout << "-1\n";
    } else {
      auto pf_b = prime_factorize(b ^ a);
      for (auto f : pf_b) {
        if (f > 8) {
          std::cout << "-1\n";
          return 0;
        }
      }
      std::cout << __builtin_popcount(b ^ a) << '\n';
    }
  }
  return 0;
}
