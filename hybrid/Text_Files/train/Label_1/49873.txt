import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long c = sc.nextLong();
        if (a == b) {
            System.out.println("inf");
        } else {
            long ans = BigInteger.valueOf(c).modPow(BigInteger.valueOf(b - a), BigInteger.valueOf(1000000007)).longValue();
            System.out.println(ans);
        }
    }
}
