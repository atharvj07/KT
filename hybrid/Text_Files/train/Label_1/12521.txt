#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    int arr[n];
    vector<int> odd, even;
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
        if (arr[i] % 2) {
            odd.push_back(arr[i]);
        } else {
            even.push_back(arr[i]);
        }
    }
    int ans = 0;
    while (!odd.empty() && !even.empty()) {
        int x = odd.back();
        odd.pop_back();
        x += even.back();
        even.pop_back();
        if (x % 2) {
            odd.push_back(x);
        } else {
            even.push_back(x);
        }
        ans++;
    }
    ans += odd.size() / 2;
    cout << ans << endl;
    return 0;
}
