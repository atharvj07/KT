#include <bits/stdc++.h>

using namespace std;

#define ll long long
const ll MOD = 1e9 + 7;

struct Ring {
    ll a, b, h;
    Ring(ll a, ll b, ll h) : a(a), b(b), h(h) {}
    bool operator<(const Ring &other) const { return b > other.b; }
};

int main() {
    ll n;
    cin >> n;
    vector<Ring> rings;
    for (ll i = 0; i < n; i++) {
        ll a, b, h;
        cin >> a >> b >> h;
        rings.emplace_back(a, b, h);
    }
    sort(rings.begin(), rings.end());
    ll dp[n];
    for (ll i = 0; i < n; i++) dp[i] = rings[i].h;
    for (ll i = 1; i < n; i++) {
        for (ll j = 0; j < i; j++) {
            if (rings[i].a > rings[j].b) dp[i] = max(dp[i], dp[j] + rings[i].h);
        }
    }
    ll ans = *max_element(dp, dp + n);
    cout << ans << endl;
    return 0;
}
