#include<bits/stdc++.h>
using namespace std;
vector<int> adj[200005];
bool zero[200005];
int dp[200005][2];
void dfs(int u,int p){
    dp[u][0] = 1;
    dp[u][1] = 0;
    for(int v:adj[u]){
        if(v==p)continue;
        dfs(v,u);
        dp[u][0] += max(dp[v][0] , dp[v][1]);
        dp[u][1] += dp[v][0];
    }
}
int main(){
    int n,x,y,c;
    cin >> n;
    for(int i=0;i<n-1;i++){
        cin >> x >> y >> c;
        adj[x].push_back(y);
        adj[y].push_back(x);
        zero[i] = c;
    }
    dfs(1,-1);
    cout << dp[1][0] +dp[1][1]-1;
}
