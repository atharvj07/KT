from collections import deque
n, k = map(int, input().split())
arr = list(map(int, input().split()))
dq = deque()
cnt = 0
ans = []
for x in arr:
    if not dq or x > dq[-1]:
        dq.append(x)
        cnt += 1
        if cnt > k:
            ans.append(dq.popleft())
    else:
        while dq and x <= dq[-1]:
            dq.pop()
            cnt -= 1
        dq.append(x)
        cnt += 1
        if cnt > k:
            ans.append(dq.popleft())
print(len(ans))
