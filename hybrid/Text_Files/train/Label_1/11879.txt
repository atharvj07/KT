#include <iostream>
#include <cstring>

constexpr int MAXN = 2e5 + 5;

int a[MAXN], b[MAXN], p[MAXN];

int main() {
    int n;
    std::cin >> n;
    for (int i = 0; i < n; ++i) {
        int temp;
        std::cin >> temp;
        a[i] = temp;
        b[temp] = i;
    }
    int cur = 0, l = 0, r = 0, lmax = 0, rmax = 0;
    for (int i = 0; i < n; ++i) {
        if (b[i] < b[cur]) {
            l++;
            if (l > lmax) {
                lmax = l;
                rmax = r;
            }
        } else {
            r++;
        }
        cur = i;
    }
    int k = lmax + rmax;
    std::cout << k << "\n";
    for (int i = 0; i < lmax; ++i) {
        std::cout << "L";
    }
    for (int i = 0; i < rmax; ++i) {
        std::cout << "R";
    }
    return 0;
}
