#include <bits/stdc++.h>
using namespace std;

const int MX = 2e4+5;
bool vis[MX];
vector<int> adj[MX];
int n, m, t, cnt;
map<string, int> id;
string s;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  cin >> n >> m >> t;
  for (int i = 0; i < n; i++) {
    cin >> s;
    if (!id.count(s)) {
      id[s] = ++cnt;
    }
    adj[id[s]].push_back(i);
  }

  vector<int> ans(n, -1);
  for (auto &[name, u] : id) {
    vector<pair<int, int>> events;
    for (int v : adj[u]) {
      events.push_back({v, 1});
      events.push_back({v + t, -1});
    }
    sort(events.begin(), events.end());
    int cur = 0;
    for (auto event : events) {
      cur += event.second;
      if (cur > m) {
        cout << "No solution\n";
        return 0;
      }
    }
    int last = -1;
    for (auto event : events) {
      if (event.second == 1) {
        ans[event.first] = last = u;
      }
    }
  }

  cout << cnt << '\n';
  for (int u : ans) cout << u << '\n';
}
