#include <iostream>

using namespace std;

void solve() {
  long long n, k;
  cin >> n >> k;
  if (n < 63 && k & 1 || n < 62 && k % 4 == 3) {
    cout << "NO\n";
  } else {
    cout << "YES " << (n - max(0LL, 60LL - k)) << '\n';
  }
}

int main() {
  int t;
  cin >> t;
  while (t--) { solve(); }
  return 0;
}
