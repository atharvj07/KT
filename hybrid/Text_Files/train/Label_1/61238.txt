def solve():
    import sys
    input = sys.stdin.readline

    N = int(input())
    p = [-1] * (2 * N + 1)
    count = [0] * (2 * N + 1)
    for _ in range(N):
        a, b = map(int, input().split())
        count[a] += 1
        count[b] += 1
        if p[a] == -1:
            p[a] = b
        elif p[b] == -1:
            p[b] = a
        else:
            return 0
    q = []
    for i in range(1, 2 * N + 1):
        if count[i] == 1:
            q.append(i)
    m = len(q)
    ans = 1
    for i in range(m // 2):
        ans *= q[i]
        ans %= 1000000007
        ans *= q[m - 1 - i]
        ans %= 1000000007
    return ans

print(solve())
