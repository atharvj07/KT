import sys

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

l, r = map(int, sys.stdin.readline().split())
if (r - l) % 2 == 0:
    print("NO")
else:
    print("YES")
    for i in range(l, r, 2):
        for j in range(i + 1, r + 1, 2):
            if gcd(i, j) == 1:
                print(i, j)
                break
