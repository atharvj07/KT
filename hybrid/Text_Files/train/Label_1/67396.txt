MOD = 10**9 + 7

def count_ways(field):
    # dp[i][j] stores the number of ways to fill the field up to i-th index
    # where j represents the state of the last square:
    # 0 - empty, 1 - contains a bomb, 2 - number of bombs in adjacent squares is 1,
    # 3 - number of bombs in adjacent squares is 2.
    dp = [[0, 0, 0, 0] for _ in range(len(field) + 1)]
    dp[0][0] = 1  # Base case: one way to fill an empty field

    for i in range(1, len(field) + 1):
        if field[i - 1] == '?' or field[i - 1] == '0':
            dp[i][0] = (dp[i - 1][0] + dp[i - 1][1]) % MOD
        if field[i - 1] == '?' or field[i - 1] == '*':
            dp[i][1] = (dp[i - 1][0] + dp[i - 1][2]) % MOD
        if field[i - 1] == '?' or field[i - 1] == '1':
            dp[i][2] = (dp[i - 1][0] + dp[i - 1][3] + dp[i - 1][1]) % MOD
        if field[i - 1] == '?' or field[i - 1] == '2':
            dp[i][3] = dp[i - 1][2] % MOD

    return sum(dp[len(field)]) % MOD

field = input().strip()
print(count_ways(field))
