import java.util.ArrayList;
import java.util.BitSet;
import java.util.Scanner;

class Main {
    static ArrayList<Integer>[] g;
    static int max = 0;

    static void dfs(int u, int p) {
        for (int v : g[u]) {
            if (v != p) {
                dfs(v, u);
                if (u > 1) {
                    int vCount = max;
                     BitSet vbs = new BitSet(vCount+1);
                    boolean outer = true;
                    for (int w : g[v]) {
                        if (w != u) {
                            int wCount = max;
                             BitSet wbs = new BitSet(wCount+1);
                            if (v > 1) {

                                for (int x : g[w]) {
                                    if (x != v)
                                        wbs.set(x);
                                }
                            }
                            if (wbs.cardinality() + 1 > vCount) {
                                vbs = wbs;
                                vCount = wbs.cardinality() + 1;
                                outer = false;
                            }
                        }
                    }
                    if (outer) {
                        vbs = new BitSet(vCount);
                        for (int x : g[v]) {
                            if (x != u)
                                vbs.set(x);
                        }
                        vCount++;
                    }
                    vbs.set(u);
                    if (vbs.cardinality() + 1 > max) {
                        max = vbs.cardinality() + 1;
                    }
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        g = new ArrayList[n + 1];
        for (int i = 0; i < n + 1; i++) {
            g[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            g[a].add(b);
            g[b].add(a);
        }
        dfs(1, 0);
        System.out.println(max);
    }
}
