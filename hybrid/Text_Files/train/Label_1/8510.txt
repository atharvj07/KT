#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> p(2 * n);
    for (int& x : p) std::cin >> x;

    std::vector<int> sorted = p;
    std::sort(sorted.begin(), sorted.end());

    int ans = 0;
    for (int i = 0; i < 2 * n; ++i) {
        if (p[i] != sorted[i]) {
            int j = -1;
            for (int k = i + 1; k < 2 * n; ++k) {
                if (p[k] == sorted[i]) {
                    j = k;
                    break;
                }
            }
            if (j == -1) {
                ans = -1;
                break;
            } else {
                std::swap(p[i], p[j]);
                ans++;
            }
        }
    }
    std::cout << ans << "\n";
    return 0;
}
