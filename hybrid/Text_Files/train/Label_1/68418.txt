import sys

def min_intimidation(arr):
    n = len(arr)
    best = n - 2
    for i in range(1, n - 1):
        if arr[i] > max(arr[i - 1], arr[i + 1]):
            best -= 1
        elif arr[i] < min(arr[i - 1], arr[i + 1]):
            best -= 1
    return best

t = int(input())
for _ in range(t):
    n = int(input())
    arr = list(map(int, input().split()))
    print(min_intimidation(arr))
