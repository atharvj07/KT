import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;
import java.text.DecimalFormat;

public class Main {
    private static final double EPS = 1e-9;
    private static final DecimalFormat FORMAT = new DecimalFormat("0.000000000");

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        long[] a = new long[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        Stack<long[]> stack = new Stack<>();
        for (int i = 0; i < n; i++) {
            while (!stack.isEmpty() && stack.peek()[1] > a[i]) {
                long[] top = stack.pop();
                long count = top[0];
                long sum = top[1];
                long avg = sum / count;
                for (int j = (int)top[2]; j <= i; j++) {
                    a[j] = avg;
                }
            }
            stack.push(new long[]{1, a[i], i});
        }
        while (!stack.isEmpty()) {
            long[] top = stack.pop();
            long count = top[0];
            long sum = top[1];
            long avg = sum / count;
            for (int j = (int)top[2]; j < n; j++) {
                a[j] = avg;
            }
        }
        for (long v : a) {
            System.out.println(FORMAT.format(v));
        }
    }
}
