from sys import stdin
A, B = map(int, stdin.readline().split())
pa, pb = [], []
L, R = 0, 0
for i in range(1, 10 ** 9 + 1):
    if L + i <= A:
        pa.append(i)
        L += i
    elif R + i <= B:
        pb.append(i)
        R += i
    else:
        break
print(len(pa))
print(' '.join(map(str, pa)))
print(len(pb))
print(' '.join(map(str, pb)))
