#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<string> grid(n);
  for (int i = 0; i < n; i++) {
    cin >> grid[i];
  }

  bool valid = true;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] != grid[0][j]) {
        valid = false;
        break;
      }
    }
    if (!valid) {
      break;
    }
  }

  for (int j = 0; j < m; j++) {
    for (int i = 0; i < n; i++) {
      if (grid[i][j] != grid[i][0]) {
        valid = false;
        break;
      }
    }
    if (!valid) {
      break;
    }
  }

  cout << (valid ? "YES" : "NO") << endl;

  return 0;
}
