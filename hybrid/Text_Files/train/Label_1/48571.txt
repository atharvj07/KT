import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    static long[] prefixA, prefixB;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        long[] arrA = Arrays.stream(reader.readLine().split(" ")).mapToLong(Long::parseLong).toArray();
        int m = Integer.parseInt(reader.readLine());
        long[] arrB = Arrays.stream(reader.readLine().split(" ")).mapToLong(Long::parseLong).toArray();
        prefixSum(arrA, arrB);
        long ans = -1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (prefixA[i] == prefixB[j]) {
                    ans = Math.max(ans, i + j + 1);
                }
            }
        }
        System.out.println(ans);
    }

    static void prefixSum(long[] arrA, long[] arrB) {
        prefixA = new long[arrA.length];
        prefixB = new long[arrB.length];
        prefixA[0] = arrA[0];
        prefixB[0] = arrB[0];
        for (int i = 1; i < arrA.length; i++) {
            prefixA[i] = prefixA[i - 1] + arrA[i];
        }
        for (int i = 1; i < arrB.length; i++) {
            prefixB[i] = prefixB[i - 1] + arrB[i];
        }
    }
}
