import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        // n is always an even number between 2 and 3 * 10^5
        int n = Integer.parseInt(reader.readLine());
        // a_i is a positive integer between 1 and 10^4
        int[] a = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();

        // Sort the array in ascending order
        Arrays.sort(a);

        // Initialize the sum of squares of group sums to 0
        long sumOfSquares = 0;
        // Initialize the number of groups to 0
        int numGroups = 0;

        // Iterate over the array in pairs, starting from the largest numbers
        for (int i = n - 1; i >= 1; i -= 2) {
            // Add the sum of the current pair to the sum of squares of group sums
            sumOfSquares += Math.pow(a[i] + a[i - 1], 2);
            // Increment the number of groups
            numGroups++;
        }

        // Print the minimum of the sum of the square of s_j
        System.out.println(sumOfSquares);
    }
}
