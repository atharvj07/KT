t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    even = [i for i, x in enumerate(a, 1) if x % 2 == 0]
    if even:
        print(1)
        print(even[0])
    else:
        if n == 1:
            print(-1)
        else:
            print(2)
            print(1, 2)
