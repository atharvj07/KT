#include <bits/stdc++.h>
using namespace std;

const int MAXN = 105;
int a[MAXN], b[MAXN];
int main() {
  int n;
  cin >> n;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    string s = to_string(a[i]);
    bool ok = 0;
    for (char c : s) {
      if (c == '0') ok = 1;
    }
    b[cnt++] = a[i] * ok;
  }
  cout << cnt << endl;
  for (int i = 0; i < cnt; i++) {
    cout << b[i] << " ";
  }
  cout << endl;
  return 0;
}
