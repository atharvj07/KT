#include <bits/stdc++.h>

using namespace std;

#define ll long long

ll n, k, a1, a2, a3, ans, b[1000001], c[1000001];

int main() {
    cin >> n >> k;
    while (n--) {
        cin >> a1;
        b[a1] = 1;
    }
    while (k--) {
        cin >> a1 >> a2 >> a3;
        for (ll i = a1; i <= a2; i++) c[i] = 1;
    }
    for (ll i = 1; i <= 1000000; i++) {
        if (c[i] == 1) {
            if (b[i] == 1)
                b[i] = 0;
            if (b[i + 1] == 1) { ans++; }
        }
    }
    cout << ans << endl;
}
