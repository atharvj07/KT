import sys

n, k = map(int, input().split())
a, b, c, d = map(int, input().split())

if k < n - 1:
    print(-1)
    sys.exit()

if k > 2 * n - 2:
    print(-1)
    sys.exit()

path1 = [a]
path2 = [c]

for i in range(1, n):
    if i % 2 == 0:
        path1.append(b)
        path2.append(d)
    else:
        path1.append(d)
        path2.append(b)

print(*path1)
print(*path2)
