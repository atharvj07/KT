a=[int(x) for x in input()]
if sum(a)%4==0:
    print(''.join(map(str,a)))
else:
    f=0
    l=0
    for i in range(len(a)):
        need=(4-(sum(a)%4))%4-a[i]
        if need<=0:
            need+=10
        if need<a[i]:
            f=1
            l=i
            break
        a[i]+=need
        if a[i]==10:
            a[i]=0
            f+=1
    if f==0:
        a+=[4]
    elif l+1<len(a):
        for i in range(l+1,len(a)):
            a[i]=0
    print(''.join(map(str,a)))
