#include<bits/stdc++.h>
using namespace std;
const int M=100;
const double eps=1e-3;
double a[M],b[M];
double C,T;
int main(){
    int tc;
    cin>>tc;
    while(tc--){
        int n;
        cin>>n;
        cin>>C>>T;
        double s=1.0;
        for(int i=0;i<n;++i){
            cin>>a[i]>>b[i];
            a[i]=ceil(a[i]/s);
        }
        double ans=0;
        for(int i=0;i<n;++i){
            if(T-a[i]>=0){
                double s1=(T-a[i]+eps)/(a[i]/s+eps),s2=(T-a[i]+eps)/(a[i]/s/0.9+eps);
                if(s2-s1>=1.0){
                    s=s/0.9;
                    T=T-10.0;
                }
            }
            if(T-a[i]>=0){
                ans+=b[i];
                T-=a[i]/s;
                s/=0.9;
                T-=10.0;
            }
        }
        printf("%.0f\n",ans);
    }
}
