#include <algorithm>
#include <cmath>
#include <cstdio>
#include <vector>

using namespace std;

struct Pt {
  double x, y;
  double dist2(const Pt& p) const { return hypot(p.x - x, p.y - y); }
};

int main() {
  int n;
  scanf("%d", &n);
  vector<Pt> pt(n);
  for (int i = 0; i < n; ++i) scanf("%lf %lf", &pt[i].x, &pt[i].y);
  sort(pt.begin(), pt.end(), [](const Pt& p1, const Pt& p2) { return p1.y < p2.y; });
  double mx = 0;
  for (int i = 0; i < n; ++i) mx = max(mx, pt[i].x);
  vector<Pt> lower, upper;
  for (int i = 0; i < n; ++i) {
    if (pt[i].y < 0) {
      lower.push_back(pt[i]);
    } else {
      upper.push_back(pt[i]);
    }
  }
  double minr = mx + 1e12;
  for (int i = 0; i < (int)lower.size(); ++i) {
    for (int j = 0; j < (int)upper.size(); ++j) {
      const Pt& a = lower[i];
      const Pt& b = upper[j];
      double da = a.dist2({0, -(a.y - a.x) / 2});
      double db = b.dist2({0, (b.y + b.x) / 2});
      double d = min(da, db);
      if (d >= pt[0].dist2({0, 0}))
        minr = min(minr, max(da, db));
    }
  }
  if (minr > mx + 1e10) {
    puts("-1");
  } else {
    printf("%.10lf\n", minr);
  }
  return 0;
}
