#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, x;
  cin >> n >> x;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  long long ans = 0, sum = 0;
  for (int i = 0; i < n; i++) {
    sum = max(sum + a[i], (long long)a[i] * x);
    ans = max(ans, sum);
  }
  cout << ans << endl;
  return 0;
}
