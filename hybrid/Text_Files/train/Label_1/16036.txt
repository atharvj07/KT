#include <iostream>
#include <vector>
#include <unordered_map>

using namespace std;

int main() {
    int n;
    cin >> n;
    string y;
    cin >> y;
    
    vector<int> cnt(n);
    unordered_map<int, int> seen;

    int x = 0;
    for (int i = 0; i < n; i++) {
        x = (x << 1) | (y[i] - '0');
        x &= ((1 << n) - 1);
        cnt[i] = seen[x]++;
    }

    long long ans = 0;
    for (int i = 0; i < n; i++) {
        ans += cnt[i] * (cnt[i] - 1) / 2;
        if (cnt[i] > 0) ans--;
    }

    cout << ans << '\n';
}
