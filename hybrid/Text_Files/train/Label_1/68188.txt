from math import pi

def distance(x1, y1, x2, y2):
    return ((x1 - x2) ** 2 + (y1 - y2) ** 2) ** 0.5

def cross_product(x1, y1, x2, y2, x3, y3):
    return (x2 - x1) * (y3 - y1) - (x3 - x1) * (y2 - y1)

def dot_product(x1, y1, x2, y2, x3, y3):
    return (x2 - x1) * (x3 - x1) + (y2 - y1) * (y3 - y1)

def point_line_distance(x1, y1, x2, y2, x3, y3):
    if dot_product(x1, y1, x2, y2, x3, y3) > 0 and dot_product(x2, y2, x1, y1, x3, y3) > 0:
        return abs(cross_product(x1, y1, x2, y2, x3, y3)) / distance(x1, y1, x2, y2)
    return min(distance(x1, y1, x3, y3), distance(x2, y2, x3, y3))

n, px, py = map(int, input().split())
vertices = [tuple(map(int, input().split())) for _ in range(n)]

max_dist = 0
min_dist = float('inf')

for i in range(n):
    x1, y1 = vertices[i]
    x2, y2 = vertices[(i + 1) % n]
    
    max_dist = max(max_dist, distance(px, py, x1, y1))
    min_dist = min(min_dist, point_line_distance(x1, y1, x2, y2, px, py))

area = pi * (max_dist ** 2 - min_dist ** 2)
print(f"{area:.20f}")
