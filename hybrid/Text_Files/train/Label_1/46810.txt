import java.util.Arrays;
import java.util.Scanner;

public class Main {

    static class Team implements Comparable<Team> {
        String name;
        int rating;

        public Team(String name, int rating) {
            this.name = name;
            this.rating = rating;
        }

        @Override
        public int compareTo(Team other) {
            return other.rating - rating;
        }
    }

    static class Basket {
        Team[] teams;
        int size;

        public Basket(int size) {
            teams = new Team[size];
            this.size = size;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long x = scanner.nextLong();
        long a = scanner.nextLong();
        long b = scanner.nextLong();
        long c = scanner.nextLong();
        Team[] teams = new Team[n];
        for (int i = 0; i < n; i++) {
            String name = scanner.next();
            int rating = scanner.nextInt();
            teams[i] = new Team(name, rating);
        }
        Arrays.sort(teams);
        Basket[] baskets = new Basket[4];
        for (int i = 0; i < 4; i++) {
            baskets[i] = new Basket(n / 4);
        }
        for (int i = 0; i < n; i++) {
            int basketIndex = i / (n / 4);
            baskets[basketIndex].teams[i % (n / 4)] = teams[i];
        }
        for (int groupIndex = 0; groupIndex < n / 4; groupIndex++) {
            System.out.println("Group " + (char) ('A' + groupIndex) + ":");
            for (int basketIndex = 0; basketIndex < 4; basketIndex++) {
                int teamIndex = (int) ((x % baskets[basketIndex].size) + 1) - 1;
                System.out.println(baskets[basketIndex].teams[teamIndex].name);
                baskets[basketIndex].size--;
                baskets[basketIndex].teams[teamIndex] = baskets[basketIndex].teams[baskets[basketIndex].size];
                x = (x * a + b) % c;
            }
        }
    }
}
