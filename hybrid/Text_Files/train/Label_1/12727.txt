#include<bits/stdc++.h>
using namespace std;
int gcd(int x, int y) {
	if (x == 0)
		return y;
	return gcd(y % x, x);
}
int lcm(int x, int y) {
	return (x * y) / gcd(x, y);
}
int main() {
	int la, ra, ta, lb, rb, tb;
	cin >> la >> ra >> ta >> lb >> rb >> tb;
	int cnt = 0, last = 0;
	for (int i = 0; i <= ra + ta; i++) {
		bool ok = 1;
		if (i <= ra)
			if ((i >= la && i <= ra) == 0)
				ok = 0;
		if (!ok)
			continue;
		if (i <= rb)
			if ((i >= lb && i <= rb) == 0)
				ok = 0;
		if (!ok)
			continue;
		cnt++;
		last = i;
	}
	cout << cnt << endl;
	return 0;
}
