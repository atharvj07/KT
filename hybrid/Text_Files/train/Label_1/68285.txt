import sys

n = int(input())
if n % 2:
    print(-1)
    sys.exit(0)

graph = [[]]
for _ in range(n):
    graph.append([])

for _ in range(n - 1):
    x, y = map(int, input().split())
    graph[x].append(y)
    graph[y].append(x)

even = 0
for v in range(1, n + 1):
    if len(graph[v]) % 2:
        even += 1

if even > 2:
    print(-1)
    sys.exit(0)

ans = (even + 1) // 2
print(n - 1 - ans)
