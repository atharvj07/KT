import collections
for _ in range(int(input())):
    n = int(input())
    a = []; b = []; diffa = []; diffb = []
    for i in range(n):
        aa,bb = map(int,input().split())
        a.append(aa); b.append(bb)
        diffa.append(aa-bb); diffb.append(bb-aa)
    diffa.sort(); diffb.sort()
    print(n*(n-1)*(n-2)//6 - collections.Counter(diffa).most_common(1)[0][1]*(collections.Counter(diffa).most_common(1)[0][1]-1)
* collections.Counter(diffb).most_common(1)[0][1]*(collections.Counter(diffb).most_common(1)[0][1]-1))
