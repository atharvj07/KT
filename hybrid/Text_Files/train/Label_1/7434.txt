#include <bits/stdc++.h>
using namespace std;
int a[4][7],vis[4],f[13]={0}, g[13]={0},h[13]={0},p,q;
int dfs1(int x){
	if(x==6){
		for (int i=1;i<=7;++i)
			if(f[i]==0) return 0;
	}
	for (int i=1;i<=3;++i){
		if(vis[i]) continue;
		if(x==0&&a[i][4]==0) continue;
		vis[i]=1;
		if(x) f[x]=f[x]*10+a[i][x];
		else f[x]=a[i][x];
		int ok=dfs1(x+1);
		vis[i]=0;
		if(ok) return 1;
	}
	return 0;
}
int dfs2(int x){
	if(x==6){
		for (int i=1;i<=7;++i)
			if(g[i]==0) return 0;
	}
	for (int i=3;i>=1;--i){
		if(vis[i]) continue;
		if(x==0&&a[i][4]==0) continue;
		vis[i]=1;
		if(x) g[x]=g[x]*10+a[i][x];
		else g[x]=a[i][x];
		int ok=dfs2(x+1);
		vis[i]=0;
		if(ok) return 1;
	}
	return 0;
}
int dfs3(int x){
	if(x==6){
		for (int i=1;i<=7;++i)
			if(h[i]==0) return 0;
	}
	for (int i=3;i>=1;--i){
		if(vis[i]) continue;
		if(x==0&&a[i][1]==0) continue;
		vis[i]=1;
		if(x) h[x]=h[x]*10+a[i][x];
		else h[x]=a[i][x];
		int ok=dfs3(x+1);
		vis[i]=0;
		if(ok) return 1;
	}
	return 0;
}
int main(){
	int n;
	cin>>n;
	for (int i=1;i<=n;++i){
		int x;
		a[i][7]=0;
		for (int j=1;j<=6;++j){
			cin>>x;
			a[i][j]=x;
		}
	}
	vis[1]=1;
	p=dfs1(1);
	vis[1]=0;
	vis[2]=1;
	q=dfs2(1);
	vis[2]=0;
	vis[3]=1;
	if(dfs3(1)){
		for (int i=1;i<=7;++i)
			if(h[i]>p){
				p=h[i];
				q=p;
			}
	}
	cout<<p+q<<endl;
}
