#include<bits/stdc++.h>
using namespace std;
long long n, m, ans;
int main() {
    cin >> n >> m;
    for (; n <= m; n++) {
        switch (n) {
        case 1:
        case 4:
            ans += 2;
            break;
        case 2:
        case 3:
        case 5:
        case 6:
        case 7:
        case 8:
        case 9:
            ans += 5;
            break;
        case 0:
            ans += 6;
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
