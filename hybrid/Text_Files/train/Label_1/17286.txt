#include <iostream>
#include <vector>
using namespace std;
int main() {
  string s, t;
  cin >> s >> t;
  vector<int> a(s.size());
  for (int i = 0; i < s.size(); i++) a[i] = (s[i] == t[i] ? 0 : 1);
  int sum = 0;
  for (int i : a) sum += i;
  if (sum % 2 == 0) {
    for (int i = 0; i < s.size(); i++) cout << (a[i] ? '0' : '1');
  } else {
    int ind = -1;
    for (int i = 0; i < s.size(); i++)
      if (a[i]) {
        ind = i;
        break;
      }
    if (ind == -1)
      cout << "impossible";
    else {
      for (int i = 0; i < s.size(); i++)
        cout << (i == ind ? '0' : s[i]);
    }
  }
  cout << endl;
}
