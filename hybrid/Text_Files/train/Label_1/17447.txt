#include <bits/stdc++.h>
using namespace std;
long long n, k;
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> k;
    long long ans = 0;
    while (n > 0) {
      if (n % k == 0) {
        ans++;
        n /= k;
      } else {
        long long mod = n % k;
        n -= mod;
        ans += mod;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
