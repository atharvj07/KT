#include <iostream>

// 0 - equal, 1 - A win, 2 - B win
int comp(int a, int b) {
    if (a == b) return 0;
    if ((a == 1 && b == 2) || (a == 2 && b == 3) || (a == 3 && b == 1)) return 1;
    else return 2;
}

int main() {
    long long k;
    int a, b;
    std::cin >> k >> a >> b;
    int A[4][4], B[4][4];
    for (int i = 1; i <= 3; ++i)
        for (int j = 1; j <= 3; ++j) {
            std::cin >> A[i][j];
            std::cin >> B[i][j];
        }
    int asa = a, bsb = b;
    int scoreA = 0, scoreB = 0;
    for (long long i = 0; i < k; ++i) {
        int curA = A[asa][bsb];
        int curB = B[asa][bsb];
        if (comp(curA, curB) == 1) scoreA++;
        if (comp(curA, curB) == 2) scoreB++;
        asa = curA;
        bsb = curB;
    }
    std::cout << scoreA << ' ' << scoreB << std::endl;
    return 0;
}
