for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    diff = [0] * (n + 1)
    for i in range(n):
        diff[i + 1] = diff[i] + (b[i] - a[i])

    min_prefix = diff[0]
    for i in range(1, n + 1):
        if diff[i] < min_prefix:
            min_prefix = diff[i]
    
    print(["NO", "YES"][min_prefix >= 0])
