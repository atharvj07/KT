import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), k = input.nextInt(), d = input.nextInt();
        if (d * n > k) System.out.println(-1);
        else {
            int[] buses = new int[n];
            for (int i = 0; i < d; i++) {
                for (int j = 0; j < n; j++) {
                    if (i == 0) buses[j] = j % k + 1;
                    else buses[j] = j % k + 1 != buses[j] ? buses[j] : k + (j % k + 1);
                    System.out.print(buses[j] + " ");
                }
                System.out.println();
            }
        }
    }
}
