import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    int[] a = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
    long[][] dp = new long[1 << n][n];
    for (int i = 0; i < (1 << n); i++) {
      Arrays.fill(dp[i], -1);
    }
    System.out.println(solve(a, 0, 0, dp));
  }

  private static long solve(int[] a, int mask, int last, long[][] dp) {
    if (dp[mask][last] != -1) {
      return dp[mask][last];
    }
    int sum = 0;
    for (int i = 0; i < a.length; i++) {
      if ((mask & (1 << i)) == 0) {
        sum += Math.abs(a[i] - a[last]);
      }
    }
    if ((sum % 2 == 0 && Integer.bitCount(mask) % 2 == 0) || (sum % 2 == 1 && Integer.bitCount(mask) % 2 == 1)) {
      dp[mask][last] = 1;
      for (int i = 0; i < a.length; i++) {
        if ((mask & (1 << i)) == 0 && Math.abs(a[i] - a[last]) <= 1) {
          dp[mask][last] += solve(a, mask | (1 << i), i, dp);
        }
      }
    } else {
      dp[mask][last] = 0;
    }
    return dp[mask][last];
  }
}
