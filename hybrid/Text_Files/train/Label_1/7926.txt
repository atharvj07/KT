#include <bits/stdc++.h>
using namespace std;
#define int long long

const int MAX = 3e5 + 1, INF = 1e9 + 7;
int n, a[MAX], ans;
vector<int> g[MAX];

void dfs(int x, int fa) {
    for(auto y : g[x]) {
        if(y == fa) continue;
        a[x] = min(a[x], a[y] - 1);
        dfs(y, x);
    }
}

signed main() {
    ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);

    cin >> n;
    for(int i = 1; i <= n; ++i) cin >> a[i];
    for(int i = 1, u, v; i < n; ++i) {
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 0);
    for(int i = 1; i <= n; ++i)
        if(a[i] > 0) ans = max(ans, a[i]);
    cout << ans << endl;
    return 0;
}
