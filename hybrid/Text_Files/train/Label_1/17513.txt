#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n,x;
        cin>>n;
        long long ct=0,ans=0;
        for(int i=1;i<=n;i++){
            cin>>x;
            if(x>0){
                ct++;
            }
            ans+=x;
        }
        if(ct==1)puts("NO");
        else if(ans>0)puts("YES");
        else puts("NO");
    }
    return 0;
}
