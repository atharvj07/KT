#include <bits/stdc++.h>
using namespace std;

const int N = 3e5 + 5;

int n, m, deg[N], ans[N], sz;

int main() {
    cin >> n >> m;
    for (int i = 1, x; i <= n; i++) { cin >> x; deg[i] = x + 2; }
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        deg[u]--, deg[v]--;
    }
    for (int i = 1; i <= n; i++) {
        if (deg[i] == 3) {
            deg[i] = 1;
            ans[++sz] = i;
        } else if (deg[i] == 2) {
            if (rand() % 2 == 0)
                deg[i] = 1, ans[++sz] = i;
        } else if (deg[i] != 0) {
            cout << -1;
            return 0;
        }
    }
    cout << sz << '\n';
    for (int i = 1; i <= sz; i++) { cout << ans[i] << '\n'; }
}
