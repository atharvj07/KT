import bisect

n = int(input())
b = list(map(int, input().split()))
c = list(map(int, input().split()))
b.sort()
c.sort()
if len(set(b)) != len(b) or len(set(c)) != len(c):
    print(-1)
else:
    res = []
    for i in range(n - 2):
        ind = bisect.bisect_right(c, b[i + 1]) - bisect.bisect_left(c, b[i])
        res.append(c[ind])
    print(*res, b[-1])
