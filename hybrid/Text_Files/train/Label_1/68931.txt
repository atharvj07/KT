def solve_fire_problem(n, items):
    items = sorted([(t, d, p, i) for i, (t, d, p) in enumerate(items, start=1)], key=lambda x: x[1])
    dp = [[0] * (2001) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(2001):
            t, d, p, idx = items[i - 1]
            if j < t or j >= d:
                dp[i][j] = dp[i - 1][j]
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - t] + p)

    ans_value = max(dp[n])
    ans_time = dp[n].index(ans_value)
    ans_items = []
    for i in range(n, 0, -1):
        t, d, p, idx = items[i - 1]
        if dp[i][ans_time] != dp[i - 1][ans_time]:
            ans_items.append(idx)
            ans_time -= t
    
    ans_items.reverse()
    print(ans_value)
    print(len(ans_items))
    print(" ".join(map(str, ans_items)))

n = int(input())
items = [list(map(int, input().split())) for _ in range(n)]
solve_fire_problem(n, items)
