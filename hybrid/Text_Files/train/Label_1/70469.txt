from collections import deque

def bfs(grid, sr, sc, tr, tc, n, k):
    vis = [[0 for i in range(n)] for j in range(5)]
    dis = [[0 for i in range(n)] for j in range(5)]
    q = deque()
    q.append([sr, sc, 0])
    vis[sr][sc] = 1
    while q:
        r, c, d = q.popleft()
        if r == tr and c == tc:
            return d
        for dr, dc in [[1, 0], [-1, 0], [0, 1], [0, -1]]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < 5 and 0 <= nc < n and grid[nr][nc] != -1 and not vis[nr][nc]:
                vis[nr][nc] = 1
                dis[nr][nc] = d + 1
                q.append([nr, nc, dis[nr][nc]])
    return -1

n, k = map(int, input().split())
grid = [[0 for i in range(n)] for j in range(5)]
cars = {}
for i in range(1, 5):
    t = [int(j) for j in input().split()]
    for j in range(len(t)):
        if t[j] != 0:
            cars[t[j]] = [i, j + 1]
            grid[i][j + 1] = -1
ans = 0
for i in range(1, k + 1):
    r, c = cars[i]
    ans += bfs(grid, r, c, i // n + 1, (i - 1) % n + 1, n, k)
if ans <= 20000:
    print(ans)
    for i in range(1, k + 1):
        r, c = cars[i]
        p = bfs(grid, r, c, i // n + 1, (i - 1) % n + 1, n, k)
        for j in range(p):
            cars[i] = [r, c]
            for dx, dy in [[1, 0], [-1, 0], [0, 1], [0, -1]]:
                nr, nc = r + dx, c + dy
                if 1 <= nr < 5 and 1 <= nc <= n and grid[nr][nc] == 0:
                    r, c = nr, nc
                    print(i, r, c)
                    break
else:
    print(-1)
