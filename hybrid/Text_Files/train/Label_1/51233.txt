import java.util.*;

public class Microwave {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        // Initialize variables
        int[] earliestTimeSteps = new int[m + 1];
        Arrays.fill(earliestTimeSteps, -1);
        earliestTimeSteps[0] = 0;

        // Process operations
        for (int i = 1; i <= n; i++) {
            int type = scanner.nextInt();
            int x = scanner.nextInt();
            int y = scanner.nextInt();

            // Calculate actual value of x
            double actualX = (double) x / 100000;

            // Handle type 1 operations
            if (type == 1) {
                for (int j = 1; j <= y; j++) {
                    for (int k = m; k >= x; k--) {
                        if (earliestTimeSteps[k - x] != -1) {
                            earliestTimeSteps[k] = Math.min(earliestTimeSteps[k], earliestTimeSteps[k - x] + 1);
                        }
                    }
                }
            }

            // Handle type 2 operations
            else if (type == 2) {
                for (int j = 1; j <= y; j++) {
                    for (int k = m; k >= 1; k--) {
                        if (earliestTimeSteps[k] != -1) {
                            earliestTimeSteps[(int) Math.ceil(k * actualX)] = Math.min(earliestTimeSteps[(int) Math.ceil(k * actualX)], earliestTimeSteps[k] + 1);
                        }
                    }
                }
            }
        }

        // Print output
        for (int i = 1; i <= m; i++) {
            System.out.print(earliestTimeSteps[i] + " ");
        }
    }
}
