import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        InputStream i = System.in;
        OutputStream o = System.out;
        int n, q, l, r;
        long[] a;
        try {
            Scanner s = new Scanner(i);
            n = s.nextInt();
            q = s.nextInt();
            a = new long[n];
            for (int j = 0; j < n; j++) {
                a[j] = s.nextInt();
            }
            for (int j = 0; j < q; j++) {
                l = s.nextInt();
                r = s.nextInt();
                int count = 0;
                for (long value : a) {
                    if (value >= l && value <= r) {
                        count++;
                    }
                }
                o.write(Integer.toString(count).getBytes());
                o.write('\n');
            }
            s.close();
            o.close();
        } catch (Throwable e) {
            e.printStackTrace();
        }
    }
}
