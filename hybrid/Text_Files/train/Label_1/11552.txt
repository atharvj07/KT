#include <iostream>
#include <vector>
using namespace std;

int n, k, a[105], pos[105], q[105], inv[105];
bool vis[105];

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) {
        cin >> q[i];
        inv[q[i]] = i;
    }
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        pos[a[i]] = i;
    }
    vis[1] = true;
    int cur = 1;
    bool ok = true;
    for (int i = 1; i < k; ++i) {
        if ((cur = inv[pos[cur]]) == 1) {
            ok = false;
            break;
        }
        vis[cur] = true;
    }
    if (cur == 1) ok = false;
    for (int i = 2; i <= n; ++i)
        if (!vis[i]) ok = false;
    cout << (ok ? "YES" : "NO") << endl;
    return 0;
}
