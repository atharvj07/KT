import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        while (true) {
            int n = in.nextInt();
            if (n == 0) {
                break;
            }

            int[][] grid = new int[4][4];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < 16; j++) {
                    grid[j / 4][j % 4] |= in.nextInt();
                }
            }

            boolean possible = true;
            for (int i = 0; i < 4; i++) {
                for (int j = 0; j < 4; j++) {
                    if (grid[i][j] == 0) {
                        possible = false;
                    }
                }
            }

            System.out.println(possible ? 1 : 0);
        }

        in.close();
    }
}
