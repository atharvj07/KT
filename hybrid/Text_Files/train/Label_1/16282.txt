#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  vector<int> b(3);
  for (int i = 0; i < n; i++) {
    b[i % 3] += a[i];
  }
  int max = *max_element(b.begin(), b.end());
  if (b[0] == max) {
    cout << "chest" << endl;
  } else if (b[1] == max) {
    cout << "biceps" << endl;
  } else {
    cout << "back" << endl;
  }
  return 0;
}
