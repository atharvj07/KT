#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin >> n;
    unordered_map<int, int> mp;
    ll ans = 0;
    int fr = -1;
    for (int i = 1; i <= n; ++i) {
        int el;
        cin >> el;
        if (el == -1)
            fr = i;
        else
            mp[el] = i;
    }
    int cur = n, w = 0;
    while (cur != 1) {
        vector<pair<int, int>> v;
        for (auto i : mp) {
            v.push_back({i.first, i.second});
        }
        sort(v.begin(), v.end());
        int sz = v.size();
        for (int i = 0; i < sz / 2; ++i) {
            if (w >= v[i].first)
                ans += v[i].first;
            else {
                if (v[i].second == fr)
                    w = v[i].first;
                else
                    ans += v[i].first;
            }
        }
        cur /= 2;
    }
    cout << ans << endl;
    return 0;
}
