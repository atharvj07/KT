#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    int cnt = 0;
    for (int i = 1; i < n - 1; i++) {
        if (a[i] > a[0] && a[i] < a[n - 1]) cnt++;
    }
    cout << cnt << endl;
    return 0;
}
