#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    int k = 0;
    vector<long long> a;
    for (int i = 2; n > 1; i++) {
      if (n % i == 0) {
        a.push_back(i);
        n /= i;
        k++;
      }
    }
    cout << k << '\n';
    for (auto &x : a) cout << x << ' ';
    cout << '\n';
  }
}
