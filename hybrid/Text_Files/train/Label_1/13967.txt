#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct seg {
    int l, r;
};
seg w[601];
int n, a[1201], b[1201];
bool vis[601];
bool solve(int l, int r) {
    if(l > r) { return true; }
    vector<pair<int, int>> v;
    for(int i = 1; i <= n; i++) {
        if(!vis[i] && w[i].l >= l && w[i].r <= r) {
            v.push_back(make_pair(w[i].l, w[i].r));
            vis[i] = true;
        }
    }
    sort(v.begin(), v.end());
    for(int i = 0; i < v.size(); i++) {
        if(v[i].first > l) { return false; }
        a[l] = v[i].second, b[v[i].second] = l;
        if(!solve(l + 1, v[i].second - 1)) { return false; }
        if(!solve(v[i].second + 1, r)) { return false; }
        return true;
    }
    return false;
}
int main() {
    cin >> n;
    for(int i = 1; i <= n; i++) { cin >> w[i].l >> w[i].r; }
    if(!solve(1, 2 * n)) { cout << "IMPOSSIBLE\n"; return 0; }
    for(int i = 1; i <= 2 * n; i++) { cout << (i <= n ? '(' : ')'); }
    cout << endl;
    return 0;
}
