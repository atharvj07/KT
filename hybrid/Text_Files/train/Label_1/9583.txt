#include <bits/stdc++.h>
using namespace std;
int n, k, a[105], ans = 1<<30;
int main() {
  cin >> n >> k;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < k; i++) {
    int cnt[3] = {0};
    for (int j = i; j < n; j += k) {
      cnt[a[j]]++;
    }
    int mn = 1e9;
    for (int j = 0; j <= 2; j++) {
      mn = min(mn, cnt[j]);
    }
    ans = min(ans, k - mn);
  }
  cout << ans << endl;
  return 0;
}
