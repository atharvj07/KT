#include <bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;
const int N = 1e5 + 5;
int dp[N][11][2];

int solve(const string& s, int pos, int cnt, bool f) {
  if (pos == s.size()) return (cnt == 0);
  int& ans = dp[pos][cnt][f];
  if (ans != -1) return ans;
  ans = 0;
  if (s[pos] == '?') {
    for (int nxt = 0; nxt < 10; nxt++) {
      if (f == false && nxt == 0) continue;
      ans = (ans + solve(s, pos + 1, cnt + (nxt == 0), f && nxt == 0)) % mod;
    }
  } else if (s[pos] >= '0' && s[pos] <= '9') {
    int d = s[pos] - '0';
    ans = solve(s, pos + 1, cnt + (d == 0), f && d == 0);
  } else {
    int ch = s[pos] - 'A';
    for (int nxt = 0; nxt < 10; nxt++) {
      if (f == false && nxt == 0) continue;
      if ((cnt > 0 && nxt != 0) || (cnt == 0 && nxt > 0))
        ans = (ans + solve(s, pos + 1, cnt + (nxt == 0), f && nxt == 0)) % mod;
    }
  }
  return ans;
}

int main() {
  string s;
  cin >> s;
  memset(dp, -1, sizeof(dp));
  cout << solve(s, 0, 0, true) << '\n';
  return 0;
}
