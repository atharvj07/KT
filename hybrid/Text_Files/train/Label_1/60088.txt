import sys

n = int(input())
l = list(map(int, input().split()))
s = input()
t = 0
stam = 0
for i in range(n):
    if s[i] != 'L':
        if s[i] == 'W': stam += 1
        t += l[i] * (3 if s[i] == 'W' else 1)
    else:
        if stam == 0: continue
        if stam >= l[i]: stam -= l[i]
        else: t += (l[i] - stam) * 5
        t += l[i]
sys.stdout.write(str(t) + '\n')
