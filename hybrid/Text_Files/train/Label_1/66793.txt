n, k = map(int, input().split())
a = list(map(int, input().split()))
b = [0 for i in range(k)]
for i in range(n):
    b[a[i] % k] = 1
c = 0
l = []
for i in range(k):
    if (i + k - b[i] % k) % k in b and i != b[i] % k:
        c += 1
        l.append(i)
    elif i == b[i] % k:
        c += 1
        l.append(i)
print(c)
print(*l)
