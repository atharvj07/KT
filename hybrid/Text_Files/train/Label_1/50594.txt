import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt();
            int k = scanner.nextInt();
            char[] arr = scanner.next().toCharArray();
            if (!check(arr, k)) {
                System.out.println(-1);
                continue;
            }
            int max = 0;
            int lastInd = 0;
            for (int i = 0; i < n; i++) {
                if (arr[i] > max && (n - i) % k == 0) {
                    max = arr[i];
                    lastInd = i;
                }
            }
            if (max == 0) {
                System.out.println(-1);
                continue;
            }
            for (int i = 0; i < lastInd; i++) {
                if (arr[i] < max) {
                    char temp = arr[i];
                    arr[i] = arr[lastInd];
                    arr[lastInd] = temp;
                    lastInd--;
                    i--;
                }
            }
            for (char c : arr) {
                System.out.print(c);
            }
            System.out.println();
        }
    }

    static boolean check(char[] arr, int k) {
        int[] freq = new int[26];
        for (char c : arr) {
            freq[c - 'a']++;
        }
        for (int f : freq) {
            if (f % k != 0) {
                return false;
            }
        }
        return true;
    }
}
