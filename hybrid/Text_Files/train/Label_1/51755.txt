import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int x = input.nextInt();
        int hh = input.nextInt();
        int mm = input.nextInt();
        int count = 0;
        while (!containsSeven(hh, mm)) {
            hh -= x;
            mm -= x;
            if (mm < 0) {
                mm += 60;
                hh--;
            }
            if (hh < 0) {
                hh += 24;
            }
            count++;
        }
        System.out.println(count);
    }

    private static boolean containsSeven(int hh, int mm) {
        return hh / 10 == 7 || hh % 10 == 7 || mm / 10 == 7 || mm % 10 == 7;
    }
}
