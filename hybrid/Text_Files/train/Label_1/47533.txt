import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String r = sc.next();
            String b = sc.next();
            int R = 0;
            int B = 0;
            for (int i = 0; i < n; i++) {
                R = R * 10 + r.charAt(i) - '0';
            }
            for (int i = 0; i < n; i++) {
                B = B * 10 + b.charAt(i) - '0';
            }
            if (R > B) {
                System.out.println("RED");
            } else if (R < B) {
                System.out.println("BLUE");
            } else {
                System.out.println("EQUAL");
            }
        }
    }
}
