import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        Map<String, Map<Integer, Integer>> teamCards = new HashMap<>();
        String homeTeam = br.readLine();
        String awayTeam = br.readLine();
        teamCards.put(homeTeam, new HashMap<>());
        teamCards.put(awayTeam, new HashMap<>());
        int n = Integer.parseInt(br.readLine());
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            int t = Integer.parseInt(line[0]);
            String team = line[1].equals("h") ? homeTeam : awayTeam;
            int player = Integer.parseInt(line[2]);
            String card = line[3];
            if (card.equals("y")) { 
                teamCards.get(team).put(player, teamCards.get(team).getOrDefault(player, 0) + 1);
                if (teamCards.get(team).get(player) == 2) {
                    System.out.println(team + " " + player + " " + t);
                }
            } else {
                System.out.println(team + " " + player + " " + t);
            }
        }
    }
}
