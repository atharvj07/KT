n = int(input())
a = list(map(int, input().split()))
cnt = [0] * (n+1)
for i in a:
    cnt[i]+=1
for i in range(1,n+1):
    cnt[i]+=cnt[i-1]
ans = 0
for l in range(1, n+1):
    for r in range(l, n+1):
        if l==r:
            ans+=cnt[l]
        else:
            ans+=cnt[r]-cnt[l-1]
print(ans)
