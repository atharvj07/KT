from sys import stdin
read = lambda : stdin.readline().strip()
n, x = map(int, read().split())
a = sorted([list(map(int, read().split())) for i in range(n)], key=lambda x:(x[0],x[1]))
mn = 1e9
for i in range(n):
    for j in range(i+1,n):
        if (mn > a[i][2] + a[j][2] and a[i][1]<a[j][0] or a[i][0]>a[j][1]):
            mn = a[i][2] + a[j][2]
print(-1 if mn == 1e9 else mn)
