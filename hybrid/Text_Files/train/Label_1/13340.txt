#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    vector<pair<long long, int>> v;
    for (int i = 0; i < n; i++) v.emplace_back(a[i], b[i]);
    sort(v.begin(), v.end());
    int ans = INT_MAX;
    long long cnt = 0;
    for (int i = 0; i < n; i++) {
      cnt += v[i].first;
      if (cnt >= m) {
        ans = min(ans, v[i].second);
      } else {
        ans = min(ans, v[i].second + 1);
      }
    }
    if (ans == INT_MAX)
      ans = -1;
    cout << ans << endl;
  }
  return 0;
}
