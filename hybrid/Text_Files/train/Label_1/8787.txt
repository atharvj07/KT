#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k, a, b;
  cin >> n >> m;
  vector<vector<int>> v(n, vector<int>(m));
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) cin >> v[i][j];
  cin >> k;
  while (k--) {
    cin >> a >> b;
    a--, b--;
    int flag = 0;
    for (int i = 0; i < m; i++) {
      if (v[b][i] < v[a][i]) continue;
      if (v[b][i] > v[a][i] + (b - a)) {
        flag = 1;
        break;
      }
    }
    cout << (flag ? "No\n" : "Yes\n");
  }
}
