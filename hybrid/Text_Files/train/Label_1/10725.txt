#include <iostream>
#include <algorithm>

using namespace std;

const int MAXN = 1e5 + 5;
int n, M, a[MAXN];

int main() {
  cin >> n >> M;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  a[n + 1] = M;

  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    ans += (a[i + 1] - a[i]) * (i % 2 == 0);
  }

  for (int i = 1; i <= n; i++) {
    long long cur = ans;
    cur -= (a[i + 1] - a[i]) * (i % 2 == 0);
    cur += (a[i + 1] - a[i - 1]) * (1 - (i - 1) % 2);
    ans = max(ans, cur);
  }

  cout << ans << endl;
  return 0;
}
