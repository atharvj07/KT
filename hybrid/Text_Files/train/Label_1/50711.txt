import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        int sumA = 0, sumB = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            sumA += Math.abs(a[i]);
            b[i] = sc.nextInt();
            sumB += Math.abs(b[i]);
        }
        if (sumA > sumB) System.out.println(n);
        else if (sumB > sumA) System.out.println(0);
        else System.out.println(Math.min(n, 1));
    }
}
