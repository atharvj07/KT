from math import factorial

def nCr(n, r):
    return factorial(n) // (factorial(r) * factorial(n - r))

def modInverse(a, m):
    return pow(a, m - 2, m)

def solve(n, m, mod, rows):
    # Count how many 1s in each column
    ones_count = [0] * n
    for row in rows:
        for i, val in enumerate(row):
            if val == '1':
                ones_count[i] += 1

    # Check if any column already has more than 2 ones
    if any(c > 2 for c in ones_count):
        return 0

    # Count how many columns have 0, 1, or 2 ones
    cols_with_0_ones, cols_with_1_one, cols_with_2_ones = 0, 0, 0
    for c in ones_count:
        if c == 0:
            cols_with_0_ones += 1
        elif c == 1:
            cols_with_1_one += 1
        else:
            cols_with_2_ones += 1

    free_rows = n - m
    ans = 1

    # Handle rows that need exactly 2 ones
    for _ in range(free_rows):
        # Choose 2 columns from columns with 0 ones or 1 column from columns with 0 ones and 1 column from columns with 1 one
        ways = nCr(cols_with_0_ones, 2) + cols_with_0_ones * cols_with_1_one
        ans = (ans * ways) % mod
        # Update counts
        cols_with_0_ones -= 2
        cols_with_1_one += 2

    # Multiply by the number of ways to arrange the remaining 1s
    remaining_ones = cols_with_1_one // 2
    for i in range(1, remaining_ones + 1):
        ans = (ans * i) % mod

    return ans

# Read input
n, m, mod = map(int, input().split())
rows = [input().strip() for _ in range(m)]

# Solve and print result
print(solve(n, m, mod, rows))
