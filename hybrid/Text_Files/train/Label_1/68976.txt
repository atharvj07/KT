n=int(input())
s=list(input())
a=[0]*n
if s[0]=='G':
    a[0]=1
for i in range(1,n):
    if s[i]=='G':
        a[i]=a[i-1]+1
b=[0]*n
if s[n-1]=='G':
    b[n-1]=1
for i in range(n-2,-1,-1):
    if s[i]=='G':
        b[i]=b[i+1]+1
ans=0
for i in range(n):
    if i>0 and i<n-1:
        if a[i]+b[i]-1>ans:
            ans=a[i]+b[i]-1
    if i>0:
        if a[i-1]+1+b[i]>ans:
            ans=a[i-1]+1+b[i]
    if i<n-1:
        if a[i]+b[i+1]+1>ans:
            ans=a[i]+b[i+1]+1
print(ans)
