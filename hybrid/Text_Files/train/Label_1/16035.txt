#include <iostream>
#include <vector>

using namespace std;

constexpr int MOD = 1000000007;
constexpr int MAXN = 200005;

long long pow[MAXN];

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  pow[0] = 1;
  for (int i = 1; i <= n; i++) pow[i] = pow[i - 1] * 2ll % MOD;

  vector<long long> h(n + 1, 0);
  for (int i = 0; i < n; i++) h[i + 1] = (h[i] * 2ll % MOD + (s[i] == '1' ? 1 : 0)) % MOD;

  int ans = 0;
  for (int i = 0; i < n; i++) {
    int l = 0, r = i;
    long long hash_i = h[i + 1];
    while (l <= r) {
      int mid = (l + r) / 2;
      long long hash_mid = (hash_i - h[i - mid] * pow[mid] % MOD + MOD) % MOD;
      if (hash_mid == h[mid]) {  // shift^mid(y) = y[i - mid...i]
        ans = (ans + 1) % MOD;
        l = mid + 1;
      } else {  // shift^mid(y) != y[i - mid...i]
        r = mid - 1;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
