#include <bits/stdc++.h>
using namespace std;
int main() {
  string s, t;
  cin >> s >> t;
  int n = s.size(), m = t.size();
  if (n != m) {
    cout << "need tree" << endl;
  } else {
    bool flag = true;
    for (int i = 0; i < n; i++) {
      if (s[i] == t[i]) flag = false;
    }
    if (flag) {
      cout << "need tree" << endl;
    } else {
      int cnt = 0;
      for (int i = 0; i < n; i++) {
        if (s[i] != t[i]) cnt++;
      }
      if (cnt == 2)
        cout << "array" << endl;
      else
        cout << "automaton" << endl;
    }
  }
}
