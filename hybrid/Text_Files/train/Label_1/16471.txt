#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; ++i) {
        cin >> v[i].first >> v[i].second;
    }
    double min_angle = acos(-1);
    int a = 0, b = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            double angle = acos(
                (v[i].first * v[j].first + v[i].second * v[j].second)
                / (sqrt(v[i].first * v[i].first + v[i].second * v[i].second)
                   * sqrt(v[j].first * v[j].first + v[j].second * v[j].second)));
            if (angle < min_angle) {
                min_angle = angle;
                a = i + 1;
                b = j + 1;
            }
        }
    }
    cout << a << " " << b << endl;
    return 0;
}
