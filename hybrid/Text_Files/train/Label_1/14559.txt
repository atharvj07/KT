#include <iostream>
#include <cstdio>
#include <vector>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false); cin.tie(nullptr);
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        long long sum = 0, mn = 1e9, mx = 0;
        vector<int> a(n);
        for (int &x : a) {
            cin >> x;
            sum += x;
            mn = min(mn, (long long)x);
            mx = max(mx, (long long)x);
        }
        if (sum % n == 0) {
            cout << (mx != mn ? 2 : 0) << "\n";
        } else {
            cout << "-1\n";
        }
    }
    return 0;
}
