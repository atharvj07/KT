import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    L = []
    R = []
    for i in range(n):
        l, r = map(int, input().split())
        L.append(l)
        R.append(r)
    L.sort()
    R.sort()
    print(min(R[i] - max(L[i], L[i-1]) for i in range(1, n)))
