#include <iostream>
#include <vector>
#include <map>

using namespace std;

typedef long long ll;

const int MOD = 1e9 + 7;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> a(n);
  vector<pair<int, int>> next(n, {0, 0});
  for (int i = 0; i < n; i++) {
    int u, v;
    cin >> u >> v;
    u--;
    v--;
    a[i] = {u, v};
  }
  int p = 0;
  for (int i = 0; i < n; i++) {
    if (a[i][0] == i) {
      p = i;
      break;
    }
  }
  for (int i = 0; i < n; i++) {
    int u = a[i][0];
    int v = a[i][1];
    if (p != i) {
      if (p == u)
        p = v;
      else if (p == v)
        p = u;
    } else {
      next[i].first = u;
      next[i].second = v;
    }
  }
  vector<int> ans(n);
  int idx = p;
  for (int i = 0; i < n; i++) {
    ans[i] = idx + 1;
    idx = next[idx].first;
    if (idx == p) idx = next[idx].second;
  }
  for (int i = 0; i < n; i++) cout << ans[i] << " ";
  cout << "\n";
}
