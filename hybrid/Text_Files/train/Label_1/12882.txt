#include <bits/stdc++.h>

#define ll long long

using namespace std;

const int LG = 64;

int n, a[100010];
ll dp[100010][LG];
int ans = 1e9;

void get_mex(int step, ll used_mask) {
  if (step == n) {
    ans = min(ans, __builtin_popcountll(used_mask));
    return;
  }

  ll mask = dp[step][__builtin_ctzll(used_mask)];

  if (mask != -1) {
    get_mex(step + 1, used_mask | mask);
    return;
  }

  ll cur_mex = used_mask;
  for (int i = 0; i < LG; i++) {
    ll new_used_mask = used_mask | (1LL << i);
    if ((new_used_mask & a[step]) == new_used_mask) {
      cur_mex = min(cur_mex, new_used_mask);
    }
  }

  dp[step][__builtin_ctzll(used_mask)] = cur_mex;
  get_mex(step + 1, used_mask | cur_mex);
}

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  memset(dp, -1, sizeof(dp));

  get_mex(0, 0);

  if (ans == 1e9)
    cout << -1 << endl;
  else
    cout << ans << endl;
  return 0;
}
