import sys
input=sys.stdin.readline

n=int(input())
par=[int(x) for x in input().split()]
g=[[] for x in range(n+1)]
for i in range(2,n+1):
    g[par[i]].append(i)

st=[0]*(n+1)
vis=[0]*(n+1)

def dfs(u,p):
    vis[u]=1
    st[u]=1
    for x in g[u]:
        if x==p:continue
        dfs(x,u)
        st[u]+=st[x]+1

dfs(1,1)
for x in st[1:]:
    print(x/n)
