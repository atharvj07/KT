#include <bits/stdc++.h>
#define LL long long
using namespace std;

const int N = 1e5 + 10;
LL a[N];
int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin >> n;
    unordered_map<LL, LL> cnt;
    for (int i = 0; i < n; i++) { cin >> a[i], cnt[a[i]]++; }
    LL ans = 0;
    for (auto x : cnt) {
        ans += (x.second * (x.second - 1) / 2);
    }
    cout << ans << "\n";
}
