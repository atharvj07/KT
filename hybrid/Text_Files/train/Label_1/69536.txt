n, k = list(map(int, input().split()))
pwrs = list(map(int, input().split()))
pwrs.sort(reverse=True)
min_pwr = sum(pwrs[:k - 1]) + sum(pwrs[k - 1::k])
start_task = 1
for i in range(1, n - k + 1):
    cur_pwr = sum(pwrs[i:i + k - 1]) + sum(pwrs[i + k - 1::k])
    if cur_pwr < min_pwr:
        min_pwr = cur_pwr
        start_task = i + 1
print(start_task)
