#include <iostream>
using namespace std;

int main() {
  int n;
  cin >> n;
  bool playing[3] = {0};
  playing[0] = 1;  // Alex starts playing
  for (int i = 0; i < n; ++i) {
    int winner;
    cin >> winner;
    if (winner == playing[0] || winner == playing[1] || winner == playing[2]) {
      cout << "NO" << endl;
      return 0;
    }
    playing[playing[1]] = 1;
    playing[playing[0]] = playing[1] = playing[2] = 0;
  }
  cout << "YES" << endl;
}
