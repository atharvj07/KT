#include <iostream>

int main() {
    int n;
    std::cin >> n;
    int a[n], health = 0, ans = 0;
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
        if (health + a[i] >= 0) {
            ans++;
            health += a[i];
        }
    }
    std::cout << ans << std::endl;

    return 0;
}
