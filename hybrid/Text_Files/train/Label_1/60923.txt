from itertools import permutations
import bisect

t = int(input())

for i in range(t):

    n, q = map(int, input().split())
    a = list(map(int, input().split()))

    b = []
    for j in range(q):
        l, r = map(int, input().split())
        b += [(l-1, r-1)]

    arr = sorted(a)
    mx = 0
    for comb in range(1, 1<<n):
        v = 0
        l = []
        for j in range(n):
            if comb>>j&1:
                l += [a[j]]
                v += arr[j] - arr[j-1] if j else arr[j]
        if len(l) > 1:
            mx = max(mx, v)

    print(mx)
    for l, r in b:
        a[l], a[r] = a[r], a[l]
        arr = sorted(a)
        mx = 0
        for comb in range(1, 1<<n):
            v = 0
            l = []
            for j in range(n):
                if comb>>j&1:
                    l += [a[j]]
                    v += arr[j] - arr[j-1] if j else arr[j]
            if len(l) > 1:
                mx = max(mx, v)
        print(mx)
