def solve(n, b):
    if any(b[i] > b[(i + 1) % n] and b[(i + 1) % n] != 0 for i in range(n)):
        return "NO"
    a = [None] * n
    for i in range(n):
        if b[i] == 0:
            a[i] = 1
        else:
            a[i] = b[i]
            if i > 0 and a[i - 1] is not None:
                a[i] += a[i - 1]
            elif i == 0:
                a[i] += b[n - 1]
    if a[n - 1] is None or a[n - 1] % a[0] != b[n - 1]:
        a[n - 1] = max(b[n - 1] + a[0], a[0] + 1)
    return "YES\n" + " ".join(str(x) for x in a)

n = int(input().strip())
b = list(map(int, input().strip().split()))
print(solve(n, b))
