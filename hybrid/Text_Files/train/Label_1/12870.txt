#include<bits/stdc++.h>
using namespace std;
const int maxN=2e5+5,inf=1<<30;
int n,k,p[maxN],vis[maxN];
vector<int> g[maxN];
bool ok;
void dfs1(int u,int fa) {
    for(auto v: g[u]) {
        if(v==fa) continue;
        if(vis[v]) { ok=1; return; }
        p[v]=u;
        vis[v]=1;
        dfs1(v,u);
    }
}
void dfs2(int u,int fa) {
    for(auto v: g[u]) {
        if(v==fa) continue;
        if(vis[v] && vis[v]!=vis[u]) { ok=1; return; }
        vis[v]=vis[u];
        dfs2(v,u);
    }
}
int main() {
    int t; cin>>t;
    while(t--) {
        cin>>n>>k;
        ok=0;
        for(int i=1;i<=n;++i) g[i].clear();
        memset(p,0,sizeof p);
        memset(vis,0,sizeof vis);
        for(int i=1;i<n;++i) {
            int u,v; cin>>u>>v;
            g[u].push_back(v);
            g[v].push_back(u);
        }
        for(int i=1;i<=k;++i) {
            int x; cin>>x;
            if(x==1) { cout<<"NO"<<endl; goto ed; }
        }
        dfs1(1,-1);
        if(ok) { cout<<"YES"<<endl; goto ed; }
        memset(vis,0,sizeof vis);
        vis[1]=1;
        dfs2(1,-1);
        if(ok) { cout<<"YES"<<endl; goto ed; }
        for(int i=2;i<=n;++i) {
            if(!vis[i]) { cout<<"YES"<<endl; goto ed; }
        }
        cout<<"NO"<<endl;
        ed:;
    }
    return 0;
}
