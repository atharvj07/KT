#include <bits/stdc++.h>
using namespace std;
const int N = 2e3 + 5;
int n, k, p;
int a[N], b[N];
int main() {
    cin >> n >> k >> p;
    for (int i = 1; i <= n; ++i) { cin >> a[i]; }
    for (int i = 1; i <= k; ++i) { cin >> b[i]; }
    sort(b + 1, b + k + 1);
    for (int i = 1; i <= n; ++i) {
        int now = a[i];
        int l = 1, r = k, pos = -1;
        while (l <= r) {
            int mid = (l + r) / 2;
            if (b[mid] >= now) {
                pos = mid;
                r = mid - 1;
            } else {
                l = mid + 1;
            }
        }
        if (pos == -1) {
            cout << "-1" << endl;
            return 0;
        }
        int L = abs(b[pos] - now) + abs(b[pos] - p);
        int R = INT_MAX;
        if (pos + 1 <= k) { R = abs(b[pos + 1] - now) + abs(b[pos + 1] - p); }
        now = p;
        if (R < L) { now = b[pos + 1], b[pos + 1] = b[pos], b[pos] = now; }
        L = abs(b[pos] - now) + abs(b[pos] - p);
        cout << L << "\n";
    }
    return 0;
}
