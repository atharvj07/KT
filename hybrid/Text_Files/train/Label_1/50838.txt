import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        if (k < n) {
            System.out.println(-1);
            return;
        }
        if (k == n) {
            for (int i = 0; i < n; i++) {
                System.out.print((i + 1) + " ");
            }
            System.out.println();
            for (int i = n - 1; i >= 0; i--) {
                System.out.print((i + 1) + " ");
            }
            System.out.println();
            return;
        }
        int sum = 0;
        for (int i = 1; i < n; i++) {
            if (sum + i + (n - i) <= k) {
                sum += i + (n - i);
            } else {
                sum += k - sum;
                break;
            }
        }
        System.out.println(sum);
        for (int i = 1; i < n; i++) {
            if (sum >= i && sum >= n - i) {
                System.out.print(i + " ");
                sum -= i;
            } else if (sum >= n - i) {
                System.out.print(n - i + " ");
                sum -= n - i;
            }
        }
        System.out.println();
        for (int i = n - 1; i >= 1; i--) {
            if (sum >= i && sum >= n - i) {
                System.out.print(i + " ");
                sum -= i;
            } else if (sum >= n - i) {
                System.out.print(n - i + " ");
                sum -= n - i;
            }
        }
        System.out.println();
    }
}
