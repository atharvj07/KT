#include <algorithm>
#include <climits>
#include <iostream>
#include <vector>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<std::vector<int>> votes(m, std::vector<int>(n));
  std::vector<int> cityWinners(m), cityVotes(m, INT_MIN);
  for (int i = 0; i < m; ++i) {
    for (int j = 0; j < n; ++j) {
      std::cin >> votes[i][j];
      if (votes[i][j] > cityVotes[i]) {
        cityVotes[i] = votes[i][j];
        cityWinners[i] = j;
      } else if (votes[i][j] == cityVotes[i]) {
        cityWinners[i] = std::min(cityWinners[i], j);
      }
    }
  }
  std::vector<int> electionVotes(n), electionCities(n);
  for (int i = 0; i < m; ++i) {
    ++electionVotes[cityWinners[i]];
    ++electionCities[cityWinners[i]];
  }
  int electionWinner = 0;
  for (int i = 1; i < n; ++i) {
    if (electionVotes[i] > electionVotes[electionWinner]) {
      electionWinner = i;
    } else if (electionVotes[i] == electionVotes[electionWinner]) {
      electionWinner = std::min(electionWinner, i);
    }
  }
  std::cout << electionWinner + 1 << std::endl;
  return 0;
}
