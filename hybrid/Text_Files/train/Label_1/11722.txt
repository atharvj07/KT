#include <iostream>
#include <vector>
using namespace std;

const int64_t MOD = 1e18;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int64_t> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];

  vector<int64_t> rem(m);
  for (int64_t x : a) rem[x % m]++;

  int64_t min_ops = 0;
  for (int i = 0; i < m; ++i) {
    if (rem[i] < n / m)
      min_ops += n / m - rem[i];
    if (rem[(i + m - 1) % m] > n / m)
      min_ops += rem[(i + m - 1) % m] - n / m;
  }
  cout << min_ops << endl;

  for (int i = 0; i < n; ++i) {
    int64_t x = a[i];
    while (x % m != 0) {
      rem[x % m]--;
      rem[(x + 1) % m]++;
      x++;
    }
    if (rem[(x + m - 1) % m] > n / m) {
      rem[(x) % m]++;
      rem[(x + m - 1) % m]--;
      x--;
    }
    a[i] = x;
  }

  for (int64_t x : a) cout << x << " ";
  return 0;
}
