for _ in range(int(input())):
    n, x = map(int, input().split())
    a = sorted(list(map(int, input().split())))
    l = [0] * 101
    for i in a:
        l[i] = 1
    for i in range(1, 101):
        if not l[i] and x:
            l[i] = 1
            x -= 1
    print(max(i for i in range(1, 101) if l[i]))
