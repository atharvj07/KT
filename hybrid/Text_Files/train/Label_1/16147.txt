#include <bits/stdc++.h>
using namespace std;
#define mp make_pair
#define pb push_back
#define fi first
#define se second

typedef long long ll;
typedef pair<int,int> pi;
typedef vector<int> vi;

const int N = 1e5 + 5;
int n, m, par[N], dis[N], ans[N];
bool act[N];
vector<pi> e[N];

int getpar(int x)
{
    if(par[x] == x) return x;
    return par[x] = getpar(par[x]);
}

void merge(int x, int y)
{
    int px = getpar(x), py = getpar(y);
    if(px == py) return;
    par[py] = px;
}

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cin >> n >> m;
    for(int i = 1; i <= n; i++)
    {
        par[i] = i;
        dis[i] = 1e9;
    }
    for(int i = 0; i < m; i++)
    {
        int x, y, z;
        cin >> x >> y >> z;
        if(z) e[x].pb(mp(y,i)), e[y].pb(mp(x,i));
        act[i] = z;
    }
    priority_queue<pi, vector<pi>, greater<pi>> q;
    q.push(mp(0,1));
    dis[1] = 0;
    while(q.size())
    {
        int u = q.top().se;
        q.pop();
        for(auto &x: e[u])
        {
            int v = x.fi, id = x.se;
            if(dis[v] > dis[u] + 1)
            {
                dis[v] = dis[u] + 1;
                q.push(mp(dis[v], v));
                merge(u,v);
                ans[id] = !act[id];
            }
            else if(getpar(u) == getpar(v) && dis[v] == dis[u] + 1)
                ans[id] = act[id];
        }
    }
    int cnt = 0;
    for(int i = 0; i < m; i++)
        cnt += ans[i];
    cout << cnt << "\n";
    for(int i = 0; i < m; i++)
    {
        if(ans[i])
        {
            int u, v;
            for(auto &x: e[i+1])
            {
                if(getpar(x.fi) == getpar(i+1))
                    u = x.fi;
                else
                    v = x.fi;
            }
            if(act[i])
            {
                swap(u,v);
                cout << u << " " << v << " 0\n";
            }
            else
                cout << u << " " << v << " 1\n";
        }
    }
    return 0;
}
