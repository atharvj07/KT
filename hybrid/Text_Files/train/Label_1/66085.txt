def modInverse(a, m) :
    m0, y, x = m, 0, 1
    while a > 1 :
        q = a // m
        t = m
        m = a % m
        a = t
        t = y
        y = x - q * y
        x = t
    return x % m0

def longestSubsequence(n):
    if n == 4: return [1, 3]
    result = []
    for i in range(1, n):
        if modInverse(i, n) > 0:
            result.append(i)
    return result

n = int(input().strip())
result = longestSubsequence(n)
print(len(result))
print(*result)
