import sys
from decimal import Decimal

for _ in range(int(sys.stdin.readline())):
    W, H = map(int, sys.stdin.readline().split())
    x1, y1, x2, y2 = map(int, sys.stdin.readline().split())
    w, h = map(int, sys.stdin.readline().split())
    
    ans = Decimal('-1')
    if h <= H - y2:
        ans = Decimal(0)
    elif w <= x2:
        ans = Decimal(x2 - w)
    else:
        if x1 == 0:
            m1 = (H - y1) / (x2 - x1)
            if m1 <= h / w:
                ans = Decimal(0)
            else:
                y = ((H - y1) * w) / h
                if y <= x2:
                    ans = Decimal(x2 - y)
        if y1 == 0:
            m2 = (W - x2) / (y2 - y1)
            if m2 <= w / h:
                ans = Decimal(0)
            else:
                x = ((W - x2) * h) / w
                if x <= y2:
                    ans = Decimal(y2 - x)

    print('%.6f' % ans if ans != Decimal('-1') else ans)
