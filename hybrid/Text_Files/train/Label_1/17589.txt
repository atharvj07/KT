#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, p, q, r, x;
    cin >> n >> p >> q >> r >> x;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    x = max({x, p * a[1] + q * a[2] + r * a[n], p * a[1] + q * a[n - 1] + r * a[n]});
    cout << x;
    return 0;
}
