import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int k = 0;
        for (int i = 1; i <= n / 2; i++) {
            boolean ok = true;
            for (int j = 0; j < i; j++) {
                if (a[j] != a[(i + j) % n]) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                k = i;
            }
        }
        System.out.println(k);
    }
}
