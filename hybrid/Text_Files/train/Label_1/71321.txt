from collections import defaultdict

def find_ancestors(fathers, node):
    ancestors = set()
    while node in fathers:
        node = fathers[node]
        ancestors.add(node)
    return ancestors

def solve(n, m, relations, wishes):
    fathers = {}
    children = defaultdict(list)
    for p, q in relations:
        fathers[q] = p
        children[p].append(q)
    
    for i in range(1, n + 1):
        if i not in fathers:
            root = i
            break
    
    ancestors_needed = defaultdict(set)
    for i, wish in enumerate(wishes, 1):
        if i != wish:
            ancestors_needed[wish].add(i)
    
    order = []
    visited = set()
    
    def dfs(node):
        visited.add(node)
        for child in children[node]:
            if child not in visited:
                dfs(child)
        order.append(node)
    
    dfs(root)
    
    candidates = set(order)
    for node in order:
        if ancestors_needed[node] - find_ancestors(fathers, node):
            return [-1]
        candidates &= set([node]) | find_ancestors(fathers, node)
    
    result_order = [node for node in order if node in candidates]
    return [len(result_order)] + result_order

n, m = map(int, input().split())
relations = [list(map(int, input().split())) for _ in range(m)]
wishes = list(map(int, input().split()))

print(*solve(n, m, relations, wishes), sep='\n')
