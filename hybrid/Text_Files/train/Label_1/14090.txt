#include <iostream>

int main() {
    long long n;
    std::cin >> n;
    if (n <= 2) {
        std::cout << -1;
        return 0;
    }
    if (n % 2 == 0) {
        std::cout << n / 2 - 1 << " " << n / 2 + 1;
    } else if ((n - 1) % 4 == 0) {
        std::cout << (n - 1) / 4 << " " << n;
    } else if ((n + 1) % 4 == 0) {
        std::cout << n << " " << 2 * (n + 1) / 4;
    } else {
        std::cout << -1;
    }
    return 0;
}
