n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

empty_a_index = a.index(0)
empty_b_index = b.index(0)

# Rotate the lists so that the empty pedestal is at the start
a_rotated = a[empty_a_index:] + a[:empty_a_index]
b_rotated = b[empty_b_index:] + b[:empty_b_index]

if a_rotated == b_rotated:
    print("YES")
else:
    print("NO")
