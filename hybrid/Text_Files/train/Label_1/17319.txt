#include <bits/stdc++.h>

using namespace std;

const long long MOD = 998244353;

void solve() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n + 1);
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  vector<long long> dp(n + 1);
  for (int i = 1; i <= n; i++) { dp[i] = dp[i - 1] + a[i]; }
  long long mxVal = dp[n];
  int numMxVal = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = i; j <= n; j++) {
      long long curVal = dp[j] - (i == 1 ? 0 : dp[i - 1]);
      if (curVal == mxVal) numMxVal++;
    }
  }
  cout << mxVal << " " << numMxVal << "\n";
}

int main() {
  solve();
  return 0;
}
