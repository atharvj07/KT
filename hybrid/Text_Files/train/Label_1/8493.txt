#include <bits/stdc++.h>

using namespace std;

const int MX = 100005;

int n, a[MX], b[MX], ans;

int32_t main() {
    cin >> n;
    string s, t;
    cin >> s >> t;
    for (int i = 0; i < n; ++i) {
        a[i] = s[i] - '0';
        b[i] = t[i] - '0';
    }
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            if (a[i] == a[j] && b[i] != b[j]) {
                ++ans;
            } else if (a[i] != a[j] && b[i] == b[j]) {
                ++ans;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
