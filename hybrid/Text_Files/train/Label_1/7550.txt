#include <iostream>
#include <vector>
using namespace std;

int solve(vector<char> &s) {
  int ans = 0, cur = 0, ones = 0;
  for (char c : s) {
    if (c == 'G') {
      cur++, ones++;
    } else {
      cur = 0;
    }
    ans = max(ans, cur);
  }
  ans = max(ans, ones);
  return ans;
}

int main() {
  int n;
  cin >> n;
  vector<char> s(n);
  for (char &c : s) cin >> c;
  cout << solve(s) << endl;
  return 0;
}
