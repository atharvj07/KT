import java.util.Scanner;

class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n + 1];
        int[] b = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 1; i <= n; i++) {
            b[i] = in.nextInt();
        }
        in.close();

        int[] diff = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            diff[i] = b[i] - a[i];
        }

        int sum = 0;
        for (int i = 1; i <= n; i++) {
            sum += Math.abs(diff[i]);
        }

        if (sum % 3 != 0) {
            System.out.println(-1);
            return;
        }

        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (diff[i] < 0) {
                ans++;
                diff[i] += 3;
                diff[i + 1] -= 3;
            }
        }

        System.out.println(ans);
    }
}
