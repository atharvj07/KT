from sys import stdin

n, r = map(int, stdin.readline().split())

a = []
for _ in range(n):
    a.append(list(map(int, stdin.readline().split())))

a.sort()
ans = 0
while a:
    x, y = a.pop(0)
    ans += 1
    i = 0
    while i < len(a):
        if abs(x - a[i][0]) + abs(y - a[i][1]) <= r:
            a.pop(i)
        else:
            i += 1

print(ans)
