import sys

n = int(input())
a = list(map(int, input().split()))

sum = 0
for i in range(n):
    sum += a[i] * a[i]
mini = 2e9
for i in range(n):
    for j in range(n):
        if i == j:
            continue
        cur = (i - j) ** 2
        for k in range(min(i, j) + 1, max(i, j)):
            cur += a[k] * a[k]
        mini = min(mini, cur)
print(mini)
