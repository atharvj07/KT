#include <iostream>
#include <vector>
using namespace std;
const int MAXN = 1e5 + 5;

int main() {
    long long n, k;
    cin >> n >> k;
    vector<int> p(n);

    for (int i = 0; i < n; i++) { p[i] = i + 1; }

    for (int i = 0; i < k; i++) {
        swap(p[i], p[i + 1]);
    }

    for (int i : p) {
        cout << i << " ";
    }
    cout << endl;
    return 0;
}
