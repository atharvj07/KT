#include <algorithm>
#include <vector>
#include <iostream>
#define ll long long
const int mxn = 2e5;
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, k;
        cin >> n >> m >> k;
        vector<ll> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a.begin(), a.end());
        ll ans = 0;
        ll p = 0;
        for (int i = 0; i < n; i++) {
            if (p + a[i] <= m) {
                ans += 1;
                m -= a[i];
                p += a[i];
                if (ans == k) break;
            } else {
                if (a[i] * (k - 1) <= m) {
                    ans += k;
                    m -= a[i] * (k - 1);
                    break;
                }
            }
        }
        cout << ans << endl;
    }
}
