import java.util.*;
import java.io.*;

class Main {
    private static final int BLACK = 0;
    private static final int WHITE = 1;

    public static void main(String[] args) {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            int N = Integer.parseInt(br.readLine());
            ArrayList<Integer>[] graph = new ArrayList[N + 1];
            for (int i = 0; i <= N; i++) graph[i] = new ArrayList<>();

            for (int i = 0; i < N - 1; i++) {
                StringTokenizer st = new StringTokenizer(br.readLine());
                int a = Integer.parseInt(st.nextToken());
                int b = Integer.parseInt(st.nextToken());
                graph[a].add(b);
                graph[b].add(a);
            }

            int[] colors = new int[N + 1];
            Arrays.fill(colors, -1);
            if (solve(1, -1, colors, graph)) System.out.println("First");
            else System.out.println("Second");
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private static boolean solve(int curr, int parent, int[] colors, ArrayList<Integer>[] graph) {
        colors[curr] = BLACK;
        for (int next : graph[curr]) {
            if (next == parent) continue;
            if (colors[next] == -1) {
                if (!solve(next, curr, colors, graph)) return true;
            } else if (colors[next] == WHITE) {
                return true;
            }
        }
        return false;
    }
}
