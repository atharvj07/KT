n = int(input())
s = input()
q = int(input())

for _ in range(q):
    m, c = input().split()
    m = int(m)
    pos = sorted([(i, ch) for i, ch in enumerate(s, 1) if ch != c])
    cov = 0
    for p, ch in pos:
        if cov < m:
            s = s[:p - 1] + c + s[p:]
            cov += 1
        else:
            break
    print(max(len(w) for w in s.split(c)))
