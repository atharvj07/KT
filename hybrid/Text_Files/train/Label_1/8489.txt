#include <bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    int a, b, c, d, ans = 0, n;
    cin >> n >> a >> b >> c >> d;
    for (int x = 1; x <= n; ++x) {
        if (a > x && b > x && c > x) ans++;
    }
    cout << ans << endl;
}
