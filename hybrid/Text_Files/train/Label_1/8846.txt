#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
    string s;
    cin >> s;
    int m;
    cin >> m;
    vector<int> shifts(m);
    for (int i = 0; i < m; i++) {
        int l, r, k;
        cin >> l >> r >> k;
        l--, r--;
        shifts[i] = (r - l + 1) - (k % (r - l + 1));
    }
    for (int i = 0, pos = 0; i < m; i++) {
        pos = (pos + shifts[i]) % s.length();
        cout << s[pos];
    }
    return 0;
}
