#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
int n, k, a[N];

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1);
    int l = 1, r = n;
    while (k && l < r) {
        int d = a[r] - a[l];
        if (d <= k) {
            a[l] += d;
            k -= d;
            r--;
        } else {
            a[l] += k;
            k = 0;
        }
    }
    cout << a[r] << endl;
    return 0;
}
