#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, v;
  cin >> n >> v;
  vector<int> cost(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> cost[i];
  }
  vector<int> dp(n + 1, 1e9);
  dp[1] = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = i + 1; j <= n && j <= i + v; j++) {
      dp[j] = min(dp[j], dp[i] + (j - i) * cost[i]);
    }
  }
  cout << dp[n] << endl;
  return 0;
}
