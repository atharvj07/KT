#include <iostream>
#include <vector>
#include <utility>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    if (n == 1) {
        cout << 1 << endl
             << 1 << endl;
        return 0;
    }
    vector<pair<int, int>> A;
    for (int i = 9; i >= 2; i--) {
        while (n >= i) {
            n -= i;
            A.push_back(make_pair(i, n + 1));
        }
    }
    sort(A.begin(), A.end());
    int ans = A.size();
    int lst = -1;
    cout << ans << endl;
    for (int i = 0; i < ans; i++) {
        if (lst == A[i].first)
            continue;
        lst = A[i].first;
        cout << A[i].second << " ";
    }
    cout << endl;
    return 0;
}
