#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, s, f;
    cin >> n >> s >> f;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    vector<long long> dp(n + 1);
    for (int l = s; l > 0; --l) {
        vector<long long> pref(n + 1);
        for (int i = 0; i < n; ++i) pref[i + 1] = pref[i] + a[(i + l - 1) % n];
        for (int r = max(l, f); r <= n; ++r) dp[l] = max(dp[l], pref[r] - pref[r - l]);
    }
    long long ans = 0;
    for (int l = s; l > 0; --l) ans = max(ans, dp[l]);
    cout << ans << endl;
    return 0;
}
