#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> b = a;
    sort(b.begin(), b.end());
    bool flag = 0;
    int len = 0;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == b[i]) len++;
        if (len == n - 1 && a[i] > a[i + 1]) {
            flag = 1;
            break;
        }
    }
    if (flag) cout << -1 << endl;
    else {
        for (int j = n - 1; j >= 0; j--) {
            if (a[j] != b[j]) ans++;
        }
        cout << ans << endl;
    }
}
