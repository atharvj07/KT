import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    private static int n, m;
    private static int[] a;
    private static Map<Integer, List<Integer>> b;
    private static int[] c;
    private static List<int[]> ans;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().trim().split(" ");
        n = Integer.parseInt(s[0]);
        m = Integer.parseInt(s[1]);
        a = new int[n + 1];
        b = new HashMap<>();
        c = new int[m + 1];
        ans = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            s = br.readLine().trim().split(" ");
            a[i] = Integer.parseInt(s[0]);
            for (int j = 1; j <= a[i]; j++) {
                int x = Integer.parseInt(s[j]);
                c[x]++;
                b.computeIfAbsent(x, k -> new ArrayList<>()).add(i);
            }
        }
        int mx = 0, mn = Integer.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            mx = Math.max(mx, a[i]);
            mn = Math.min(mn, a[i]);
        }
        if (mx - mn > 1) {
            System.out.println(-1);
            return;
        }
        if (mx == mn) {
            System.out.println(0);
            return;
        }
        for (int i = 1; i <= m; i++) {
            if (c[i] % 2 == 1) {
                System.out.println(-1);
                return;
            }
        }
        for (int i = 1; i <= n; i++) {
            if (a[i] == mn) {
                for (int j : b.keySet()) {
                    if (a[i] + 1 <= mx && c[j] >= 2) {
                        ans.add(new int[]{i, b.get(j).get(0), j});
                        a[i]++;
                        a[b.get(j).get(0)]--;
                        c[j] -= 2;
                        break;
                    }
                }
            }
        }
        System.out.println(ans.size());
        for (int[] p : ans) {
            System.out.println(Arrays.toString(p).replace(" ", ""));
        }
    }
}
