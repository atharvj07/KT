import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextLong();
        long b = in.nextLong();
        long c = in.nextLong();
        int d = in.nextInt();
        long mod = 1000000007;
        long ans = ((a - 1) * (b - 1) * (c - 1) + (a - 1) * (b - 1) + (a - 1) * (c - 1) + (b - 1) * (c - 1)) % mod;
        if (d >= 2) {
            ans = (ans + ((a - 2) * (b - 2) * (c - 2) + (a - 2) * (b - 2) + (a - 2) * (c - 2) + (b - 2) * (c - 2)) % mod) % mod;
        }
        if (d >= 3) {
            ans = (ans + ((a - 3) * (b - 3) * (c - 3)) % mod) % mod;
        }
        System.out.println(ans);
    }
}
