import sys

n = int(input())
q = [list(map(int, input().split())) for _ in range(n-2)]

e = []
for a, b, c in q:
    for x, y in [(a, b), (a, c), (b, c)]:
        if x == y: continue
        e.append((x, y))

g = {}
for x, y in e:
    if x not in g: g[x] = []
    if y not in g: g[y] = []
    g[x].append(y)
    g[y].append(x)

vis = [False] * (n+1)
p = []
def dfs(x):
    if vis[x]: return
    vis[x] = True
    p.append(x)
    for y in g[x]:
        dfs(y)

for i in range(1, n+1):
    if not vis[i]:
        dfs(i)

print(*p)
