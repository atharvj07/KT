import java.util.*;

public class LongestOnTV {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt(); // Number of teams
        int R = scanner.nextInt(); // Number of records
        int L = scanner.nextInt(); // Length of the contest

        // Initialize with team 1 and time 0
        int currentTeam = 1;
        int currentStart = 0;

        // Track the longest on-screen time and corresponding team
        int longestTeam = 1;
        int longestTime = 0;

        // Process each record
        for (int i = 0; i < R; i++) {
            int team = scanner.nextInt();
            int time = scanner.nextInt();
            int score = scanner.nextInt();

            // Update the current team and start time if necessary
            if (team != currentTeam) {
                currentTeam = team;
                currentStart = time;
            }

            // Update the longest team and time if needed
            int duration = time - currentStart;
            if (duration > longestTime) {
                longestTime = duration;
                longestTeam = team;
            }
        }

        // Print the result
        System.out.println(longestTeam);
    }
}
