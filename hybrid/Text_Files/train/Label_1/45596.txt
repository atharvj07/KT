import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        if (a % 2 == b % 2) {
            System.out.println(a * a - b * b);
            for (int i = 0; i < a; i++) {
                System.out.print("o");
            }
            for (int i = 0; i < b; i++) {
                System.out.print("x");
            }
        } else {
            System.out.println(-a * a - b * b);
            for (int i = 0; i < b; i++) {
                System.out.print("x");
            }
            for (int i = 0; i < a; i++) {
                System.out.print("o");
            }
        }
    }
}
