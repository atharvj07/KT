n = int(input())
a = []
b = []
for _ in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)
p = sorted(zip(a, b, range(n)))
a, b, ans = zip(*p)
a = [0] + list(a)
b = list(b) + [0]
pre = [0] * (n + 2)
for i in range(n):
    pre[i + 1] = pre[i] + a[i + 1] - b[i]
suf = [0] * (n + 2)
for i in range(n - 1, -1, -1):
    suf[i] = suf[i + 1] + b[i + 1] - a[i]
res = 1e18
for i in range(n):
    res = min(res, pre[i] + suf[i + 1])
print(int(res))
