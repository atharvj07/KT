#include <iostream>
#include <string>

int main() {
  unsigned long long int a, b;
  std::cin >> a >> b;
  std::string sa = std::to_string(a);
  std::string sb = std::to_string(b);

  int fza = 0;
  for (int i = 0; i < sa.size(); i++) {
    if (sa[i] == '0') fza++;
  }
  int fzb = 0;
  for (int i = 0; i < sb.size(); i++) {
    if (sb[i] == '0') fzb++;
  }

  std::cout << fzb - fza + (fza > 0);
  return 0;
}
