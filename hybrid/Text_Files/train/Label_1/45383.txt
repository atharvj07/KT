import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long t = scanner.nextLong();
        long n = scanner.nextLong();
        for (long i = 0; i < t; i++) {
            long a = scanner.nextLong();
            long b = scanner.nextLong();
            if (a > b) {
                long temp = a;
                a = b;
                b = temp;
            }
            if (b - a <= n) {
                System.out.println(b - a);
            } else if (a <= n && b <= 2 * n) {
                System.out.println(a + n - b);
            } else if (a <= n && 2 * n < b) {
                System.out.println(b - 2 * n + a);
            } else if (2 * n < a && b <= 3 * n) {
                System.out.println(a - 2 * n + b);
            } else if (2 * n < a && 3 * n < b) {
                System.out.println(b - 3 * n + a);
            } else if (3 * n < a && b <= 4 * n) {
                System.out.println(a - 3 * n + b);
            } else if (3 * n < a && 4 * n < b) {
                System.out.println(b - 4 * n + a);
            } else if (4 * n < a && b <= 5 * n) {
                System.out.println(a - 4 * n + b);
            } else {
                System.out.println(b - 5 * n + a);
            }
        }
        scanner.close();
    }
}
