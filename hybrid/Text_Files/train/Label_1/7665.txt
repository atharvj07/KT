#include <bits/stdc++.h>
#define int long long

using namespace std;

const int mod = 10;

int f(vector<int> a) {
    int n = a.size(), k = 0, ans = 0;
    while ((1LL << k) < n) k++;
    for (int i = 0; i < (1LL << k); i++) {
        int cur = (a[2 * i] + a[2 * i + 1]) % mod;
        ans += (cur >= 10);
        a[i] = cur;
    }
    a.resize((1LL << k));
    return ans;
}

int32_t main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int q;
    cin >> q;
    while (q--) {
        int l, r;
        cin >> l >> r;
        cout << f(vector<int>(a.begin() + l - 1, a.begin() + r)) << "\n";
    }
    return 0;
}
