def process(expr: str, start: int, end: int, plus: int, minus: int) -> int:
    if end - start + 1 <= 2:
        return int(expr[start:end+1])
    res = -10 ** 9
    for i in range(start + 1, end):
        if expr[i] == '+':
            res = max(res, process(expr, start, i-1, plus-1, minus) + process(expr, i+1, end, plus, minus))
        elif expr[i] == '-':
            res = max(res, process(expr, start, i-1, plus, minus-1) - process(expr, i+1, end, plus, minus))
    return res

expr = input()
p, m = map(int, input().split())
print(process(expr, 0, len(expr)-1, p, m))
