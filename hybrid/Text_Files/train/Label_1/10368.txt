#include <bits/stdc++.h>
using namespace std;

#define endl '\n'
#define ll long long
#define fastio ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0)

int main() {
    fastio;
    ll t;
    cin >> t;
    while (t--) {
        ll n, k;
        cin >> n >> k;
        ll q = n / k;
        ll r = n % k;
        for (ll i = 0; i < r; i++) {
            cout << q + 1 << " ";
        }
        for (ll i = 0; i < k - r; i++) {
            cout << q << " ";
        }
        cout << endl;
    }
    return 0;
}
