#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<string> v = {"vaporeon", "jolteon", "flareon", "espeon", "umbreon", "leafeon", "glaceon", "sylveon"};
  for (int i = 0; i < 8; i++) {
    if (v[i].size() == n) {
      int f = 0;
      for (int j = 0; j < n; j++) {
        if (s[j] == '.' || s[j] == v[i][j]) {
          continue;
        } else {
          f = 1;
        }
      }
      if (f == 0) {
        cout << v[i] << endl;
        return 0;
      }
    }
  }
  return 0;
}
