import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int ans = -1;
        for (int i = 0; i < n; i++) {
            if (a[i] > 0) {
                int g = a[i];
                for (int j = 0; j < n; j++) {
                    if (a[j] > 0) {
                        g = gcd(g, a[j]);
                    }
                }
                if (g > 1) {
                    ans = 0;
                    for (int j = 0; j < n; j++) {
                        if (a[j] > 0) {
                            ans += (a[j] - 1) / g;
                        }
                    }
                    break;
                }
            }
        }
        System.out.println(ans);
    }
    public static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
