import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class SmartBeaver {

    static class Node {
        int id;
        char letter;
        Node parent;
        List<Node> children = new ArrayList<>();

        public Node(int id, char letter, Node parent) {
            this.id = id;
            this.letter = letter;
            this.parent = parent;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();

        Node root1 = new Node(1, '\0', null);
        Node root2 = new Node(1, '\0', null);

        Map<Integer, List<Integer>> forwardPaths = new HashMap<>();
        Map<Integer, List<Integer>> backwardPaths = new HashMap<>();

        for (int i = 0; i < n; i++) {
            String[] operation = scanner.nextLine().split(" ");
            int tree = Integer.parseInt(operation[0]);
            int vertex = Integer.parseInt(operation[1]);
            char letter = operation[2].charAt(0);

            Node parent = tree == 1 ? root1 : root2;
            Node child = new Node(vertex, letter, parent);
            parent.children.add(child);

            updatePaths(forwardPaths, backwardPaths, child);

            int count = 0;
            for (int j = 1; j <= vertex; j++) {
                if (backwardPaths.get(j) != null) {
                    for (int k : backwardPaths.get(j)) {
                        if (forwardPaths.get(k) != null) {
                            count += forwardPaths.get(k).size();
                        }
                    }
                }
            }

            System.out.println(count);
        }
    }

    private static void updatePaths(Map<Integer, List<Integer>> forwardPaths, Map<Integer, List<Integer>> backwardPaths, Node child) {
        Node parent = child.parent;
        if (forwardPaths.get(child.id) == null) {
            forwardPaths.put(child.id, new ArrayList<>());
        }
        if (backwardPaths.get(parent.id) == null) {
            backwardPaths.put(parent.id, new ArrayList<>());
        }

        forwardPaths.get(child.id).add(parent.id);
        backwardPaths.get(parent.id).add(child.id);

        if (parent.parent != null) {
            updatePaths(forwardPaths, backwardPaths, parent);
        }
    }
}
