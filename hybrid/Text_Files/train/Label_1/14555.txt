#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int,int> pii;
typedef pair<int,pair<int,int>> piii;

const int mod = 1e9+7;
const int maxn = 2e5+5;

int n,k;
int a[maxn];
set<int> s[maxn];

bool check(int x){
    int cnt=0;
    for(auto idx : s[x]) cnt+=(a[idx]==x);
    return cnt>=(k-1);
}

int main(){
    cin>>n>>k;
    s[0].insert(0); // represents end of array
    for(int i=1;i<=n;++i){
        cin>>a[i];
        s[a[i]].insert(i);
    }
    bool flag=false;
    for(int i=1;i<=k;++i){
        if(check(i)) flag |= true;
        else{ flag=false; break;}
    }
    if(!flag){ cout<<-1<<endl; return 0; }
    int cnt=0;
    vector<vector<int>> ans;
    for(int i=1;i<=n;++i){
        int idx=-1;
        for(int j=1;j<=k;++j){
            if(a[i]==j && s[j].size()>=k){ idx=j; break; }
        }
        if(idx!=-1){ s[idx].erase(i); continue; }
        if(cnt!=k){
            ans.push_back({i});
            cnt++;
            continue;
        }
        for(int j=1;j<=k;++j){
            if(a[i]==j){
                ans.back().push_back(i);
                break;
            }
        }
    }
    for(auto vec : ans){
        for(auto i : vec) cout<<a[i]<<" ";
    }
    cout<<endl;
    return 0;
}
