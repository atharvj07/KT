n,q=map(int,input().split())
s=input()
t='abc'
for j in range(q):
    i,x=list(map(str,input().split()))
    i=int(i)
    idx=-1
    for j in range(3):
        if s[(i+j)%n]==t[j]:
            idx=j
            break
    ans=idx
    if ans==-1:
        ans=0
    else:
        for j in range(3):
            if x==t[(j-ans)%3]:
                ans=min(ans,j)
    print(ans)
