import sys
input = sys.stdin.readline

n,k = map(int,input().split())
a = list(map(int,input().split()))

if k==1:
    ans = 0
    for i in range(n):
        for j in range(i+1,n):
            if a[i]>a[j]:
                ans += 1
    ans-=n
    print(ans/n, end="")
elif n<=30 and k<=200:
    ans = 0
    den = k
    for i in range(n):
        for j in range(i):
            l,r = min(a[i],a[j]), max(a[i],a[j])
            if l!=r:
                ans += (l-1) * (n-r) / den
                ans += (n-l) * (r-1) / den
    print(ans, end="")
elif n<=100:
    dp=[[[-1]*3 for j in range(n+1)] for i in range(n+1)]
    def calc(l,r,f):
        if l==r:
            return 0
        if dp[l][r][f]!=-1:
            return dp[l][r][f]
        if f==0:
            ans=0
            for i in range(l,r+1):
                for j in range(i,r+1):
                    if a[i]>a[j]:
                        ans+=1
        elif f==1:
            ans=calc(l+1,r,0)
            ans-=(a[l]-1)*(n-a[l])/k
            for i in range(l+1,r+1):
                for j in range(i,r+1):
                    if a[i]>a[j]:
                        ans-=1
        elif f==2:
            ans=calc(l,r-1,0)
            ans+=(a[r]-1)*(n-a[r])/k
            for i in range(l,r):
                for j in range(i,r):
                    if a[i]>a[j]:
                        ans+=1
        dp[l][r][f]=ans
        return ans
    print(calc(0,n-1,0), end="")
else:
    print(0, end="")
