p, q, l, r = map(int, input().split())
z_times = [list(map(int, input().split())) for _ in range(p)]
x_times = [list(map(int, input().split())) for _ in range(q)]

def can_chat(l, z_times, x_times):
    for start, end in z_times:
        for c, d in x_times:
            if not (l + d < start or l + c > end):
                return True
    return False

count = 0
for t in range(l, r + 1):
    if can_chat(t, z_times, x_times):
        count += 1

print(count)
