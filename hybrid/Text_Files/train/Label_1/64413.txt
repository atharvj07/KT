import sys
input = sys.stdin.readline

def subtask1(m, var):
    mini, maxi = 0, 0
    for i in range(m):
        mask = (1 << i)
        min_set = 0
        for j in range(len(var)):
            if 'AND' in var[j]:
                a, b = var[j].split()
                min_set |= ((int(a, 2) & int(b, 2) & mask) != 0)
            elif 'XOR' in var[j]:
                a, b = var[j].split()
                min_set |= ((int(a, 2) ^ int(b, 2) & mask) != 0)
            elif 'OR' in var[j]:
                a, b = var[j].split()
                min_set |= ((int(a, 2) | int(b, 2) & mask) != 0)
            else:
                min_set |= ((int(var[j], 2) & mask) != 0)
        if min_set:
            mini |= mask
            maxi |= mask
    mini = bin(mini)[2:].zfill(m)
    maxi = bin(maxi)[2:].zfill(m)
    return mini, maxi

n, m = map(int, input().split())
var = [input()[:-1] for i in range(n)]
if m == 1:
    print(0, 0)
    exit()
mini, maxi = subtask1(m, var)
print(mini, maxi)
