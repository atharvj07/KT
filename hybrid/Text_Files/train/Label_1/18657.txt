#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n), b(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        for (int i = 0; i < n; i++) cin >> b[i];
        int sum = 0;
        for (int i = 0; i < n; i++) sum += (b[i] - a[i]);
        if (sum == 0) {
            cout << "YES" << endl;
        } else if (sum > 0) {
            cout << "NO" << endl;
        } else {
            vector<int> diff;
            int idx = -1, cnt = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] > b[i]) {
                    if (idx == -1) idx = i;
                    cnt++;
                    diff.push_back(a[i] - b[i]);
                }
            }
            if (cnt > 1 || sum < 0) {
                cout << "NO" << endl;
            } else {
                sort(diff.begin(), diff.end());
                int cur = b[idx] - diff[0];
                bool ok = true;
                for (int i = idx + 1; i < n; i++) {
                    if (a[i] > b[i]) {
                        ok = false;
                        break;
                    } else if (cur != a[i]) {
                        ok = false;
                        break;
                    }
                    cur++;
                }
                if (ok) {
                    cout << "YES" << endl;
                } else {
                    cout << "NO" << endl;
                }
            }
        }
    }
    return 0;
}
