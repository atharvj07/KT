import math
n, m, k = map(int, input().split())
sx, sy = 0, 0
fx, fy = n, m
ans = []
for _ in range(k):
    x, y = map(int, input().split())
    t = math.inf
    if x == sx:
        t = min(t, (y - sy) / math.sqrt(2))
    if x == fx:
        t = min(t, (sy - y) / math.sqrt(2))
    if y == sy:
        t = min(t, (x - sx) / math.sqrt(2))
    if y == fy:
        t = min(t, (sx - x) / math.sqrt(2))
    ans.append((t if t != math.inf else -1))
print(*ans)
