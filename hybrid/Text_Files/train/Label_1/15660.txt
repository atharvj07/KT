#include <bits/stdc++.h>

using namespace std;

const int N = 2e5 + 10;
int main() {
    int n, m, k, a[N];
    cin >> n >> m >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    int j = n - 1;
    int ans = 0;
    for (int i = 0; i < k; i++) {
        int sum = 0;
        for (int l = 0; l < m; l++) {
            sum += a[j--];
        }
        ans += sum;
    }
    cout << ans << endl;
    int pos = 0;
    for (int i = 1; i < k; i++) {
        pos += m;
        cout << pos << " ";
    }
    cout << endl;
}
