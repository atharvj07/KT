import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long d = sc.nextLong();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        java.util.Arrays.sort(a);
        long minCost = 0;
        for (int i = 1; i < n; i++) {
            minCost += (i * d + a[i] + a[i - 1]);
        }
        System.out.println(minCost);
    }
}
