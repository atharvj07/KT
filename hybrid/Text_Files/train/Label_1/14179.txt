#include <iostream>
#include <vector>
using namespace std;

vector<int> adj[1001];
int n, deg[1001];

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n;
  for (int u, v, i = 0; i < n - 1; i++) {
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
    deg[u]++, deg[v]++;
  }
  int cnt = 0;
  for (int i = 1; i <= n; i++)
    if (deg[i] == 1) cnt++;
  cout << cnt << '\n';
  return 0;
}
