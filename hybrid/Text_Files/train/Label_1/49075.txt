import java.util.*;
import java.io.*;

public class Main {
    static int n;
    static int[] par;
    static ArrayList<Integer>[] adj;

    public static void main(String[] args) {
        // 入力
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        int m = sc.nextInt();
        adj = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        int[] a = new int[n + 1];
        for (int i = 1; i <= m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj[u].add(v);
            adj[v].add(u);
            a[u]++; a[v]++;
        }

        // 連結成分に関係なく、全ての町を結ぶ道路が存在するか判定
        allRoads();

        // 存在する場合は、高速な道路のみを利用する
        if (ans) {
            System.out.println(1);
            return;
        }

        // 存在しない場合は、連結成分毎に道路と鉄道路の経路を判定
        par = new int[n + 1];
        for (int i = 1; i <= n; i++) par[i] = i;
        for (int i = 1; i <= n; i++) {
            int root = fin(i);
            if (a[i] % 2 == 0) continue;
            if (a[i] == 1) {
                System.out.println(-1);
                return;
            }
            int p = -1, q = -1;
            for (int j : adj[i]) {
                if (fin(j) == root) {
                    if (p == -1) p = j;
                    else q = j;
                }
            }
            if (p == -1 || q == -1) {
                System.out.println(-1);
                return;
            }
            uni(root, p);
            uni(root, q);
        }

        // 連結成分毎に高速な道路と鉄道路の経路を判定
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            if (fin(i) == i) {
                Set<Integer> st = new HashSet<>();
                solve(i, i, i, st);
                ans = Math.max(ans, dist[i]);
            }
        }
        System.out.println(ans);
    }

    static boolean ans;
    static int[] dist;

    static void allRoads() {
        dist = new int[n + 1];
        Arrays.fill(dist, 100000000);
        dist[1] = 0;
        Queue<Integer> qu = new LinkedList<>();
        qu.add(1);
        while (!qu.isEmpty()) {
            int now = qu.poll();
            if (now == n) ans = true;
            for (int u : adj[now]) {
                if (dist[u] > dist[now] + 1) {
                    dist[u] = dist[now] + 1;
                    qu.add(u);
                }
            }
        }
    }

    static void solve(int now, int st, int c, Set<Integer> stt) {
        if (now == n) {
            if (now != c) dist[c] = Math.max(dist[c], dist[now] + 1);
            else dist[c] = Math.max(dist[c], dist[now]);
            return;
        }
        stt.add(now);
        for (int u : adj[now]) {
            if (u == st) {
                if (now != c) dist[c] = Math.max(dist[c], dist[now] + 1);
                else dist[c] = Math.max(dist[c], dist[now]);
                continue;
            }
            if (stt.contains(u)) continue;
            solve(u, st, c, stt);
        }
        stt.remove(now);
    }

    static int fin(int x) {
        if (x == par[x]) return x;
        return par[x] = fin(par[x]);
    }

    static void uni(int x, int y) {
        par[fin(x)] = fin(y);
    }
}
