#include <iostream>
#include <string>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int ans = 0;
        bool ok = false;
        for (int i = 0; i < s.size(); ++i) {
            if (s[i] == '0')
                ans += !ok;
            else
                ok = true;
        }
        cout << ans << endl;
    }
    return 0;
}
