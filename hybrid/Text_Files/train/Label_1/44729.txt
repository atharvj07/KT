import java.util.Scanner;

public class VasilysTrip {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read Vasiliy's coordinates
        int a = scanner.nextInt();
        int b = scanner.nextInt();

        // Read the number of Beru-taxi cars
        int n = scanner.nextInt();

        // Find the minimum time
        double minTime = Double.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            // Read the i-th taxi's coordinates and speed
            int xi = scanner.nextInt();
            int yi = scanner.nextInt();
            int vi = scanner.nextInt();

            // Calculate the distance between Vasiliy and the i-th taxi
            double distance = Math.sqrt((a - xi) * (a - xi) + (b - yi) * (b - yi));

            // Calculate the time it takes for the i-th taxi to reach Vasiliy
            double time = distance / vi;

            // Update the minimum time
            minTime = Math.min(minTime, time);
        }

        // Print the minimum time
        System.out.println(minTime);
    }
}
