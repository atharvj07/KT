#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int k = 0;
  for (int i = 0; i < n / 2; i++) {
    if (s[2 * i] == s[2 * i + 1]) {
      k++;
      s[2 * i + 1] = ' ';
    }
  }
  cout << k << '\n';
  for (int i = 0; i < n; i++) {
    if (s[i] != ' ') cout << s[i];
  }
  cout << '\n';
}
