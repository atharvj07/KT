#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
#define ll long long
const ll inf = 1e16;

struct obj{
    ll value;
    ll ind;
    bool operator<(const obj &o)const{
        return value < o.value;
    }
};

int main() {
    ll t;
    cin >> t;
    while(t--){
        ll n;
        cin >> n;
        vector<obj>v(n);
        for(ll i=0;i<n;i++){
            ll x;
            cin >> x;
            v[i].value = x;
            v[i].ind = i;
        }
        sort(v.begin(), v.end());
        ll cnt = 0;
        for(ll i=0;i<n;i++){
            if(i != v[i].ind) cnt++;
        }
        cnt = cnt/2;
        cnt--;
        cout << cnt << endl;
    }
    return 0;
}
