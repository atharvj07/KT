#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, a;
  cin >> n >> a;
  vector<int> d(n);
  for (int i = 0; i < n; i++) cin >> d[i];
  vector<int> ans(n);
  int sum = 0;
  for (int i = 0; i < n; i++) { sum += d[i]; }
  for (int i = 0; i < n; i++) {
    ans[i] = max(0, min(sum - a + d[i], d[i] - 1));
  }
  for (int i = 0; i < n; i++) { cout << ans[i] << " "; }
  cout << endl;
}
