import java.util.*;

class Main {
    public static void main(String[] arg) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();

        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextLong();
        }

        long min = Integer.MAX_VALUE;
        int ind = 0;
        for (int i = 0; i < n; i++) {
            long cnt = 0;
            for (int j = i; j < n; j++) {
                cnt += Math.ceil((double) arr[j] / 7.0);
            }
            cnt += Math.ceil((double) arr[i] / 6.0);
            if (cnt < min) {
                min = cnt;
                ind = i;
            }
        }

        System.out.println(ind + 1);
    }
}
