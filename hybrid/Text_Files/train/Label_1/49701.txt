import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextLong();
        }
        long g = 0;
        for (int i = 1; i < n; i++) {
            g = gcd(g, a[i] - a[i - 1]);
        }
        if (g == 0) {
            System.out.println("1 " + a[0]);
        } else {
            long y = a[0] / g;
            long z = g;
            System.out.println(y + " " + z);
        }
    }

    private static long gcd(long a, long b) {
        while (b > 0) {
            long temp = a % b;
            a = b;
            b = temp;
        }
        return a;
    }
}
