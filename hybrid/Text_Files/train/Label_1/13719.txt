#include <cstdio>
#include <vector>
using namespace std;

const int MAXN = 1000;
int n, m;
char field[MAXN][MAXN];
int ans[MAXN][MAXN];

void dfs(int x, int y) {
  if(x < 0 || x >= n || y < 0 || y >= m || field[x][y] != '*') {
    return;
  }
  field[x][y] = '.';
  ans[x][y]++;
  dfs(x-1, y);
  dfs(x+1, y);
  dfs(x, y-1);
  dfs(x, y+1);
}

int main() {
  scanf("%d %d", &n, &m);
  for(int i = 0; i < n; i++) {
    scanf("%s", field[i]);
  }
  for(int i = 0; i < n; i++) {
    for(int j = 0; j < m; j++) {
      if(field[i][j] == '*') {
        dfs(i, j);
      }
    }
  }
  for(int i = 0; i < n; i++) {
    for(int j = 0; j < m; j++) {
      printf("%c", field[i][j] == '.' ? '.' : ans[i][j]%10 + '0');
    }
    printf("\n");
  }
  return 0;
}
