import sys
input = sys.stdin.readline

def cross(p, q, r):
    return (r[1]-p[1])*(q[0]-p[0])-(r[0]-p[0])*(q[1]-p[1])

x = []
y = []
ok = [False]

for i in range(int(input())):
    a, b = map(int, input().split())
    if a == 0: ok = [True]
    x.append(a)
    y.append(b)

if ok[0]:
    print("Yes")
else:
    X = []
    for i in range(1,len(x)):
        for j in range(i+1, len(x)):
            X.append(cross([x[0],y[0]], [x[i],y[i]], [x[j],y[j]]))
    if X.count(X[0]) == len(X): print("Yes")
    else: print("No")
