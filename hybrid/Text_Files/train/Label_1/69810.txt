import sys
input = sys.stdin.readline

n, m = map(int, input().split())
d = [[(x>0, abs(x)-1) for x in map(int, input().split())] for i in range(m)]

vis = [False] * (n+1)
for i in d:
    for x in i:
        if vis[x[1]]: flag = True; break
        vis[x[1]] = True
    vis = [False] * (n+1)
    if flag: break

print('YES' if flag else 'NO')
