import sys

n, m = map(int, input().split())
islands = [(list(map(int, input().split()))) for _ in range(n)]
bridges = list(map(int, input().split()))
islands.sort()
bridges.sort()

if bridges[-1] > islands[-1][1] - islands[-1][0] + 1:
    print("No")
    sys.exit()

j = 0
for i in range(n - 1):
    while bridges[j] < islands[i][1] - islands[i][0] + 1:
        j += 1
        if j == m:
            print("No")
            sys.exit()
    print(j + 1, end=" ")
print()
