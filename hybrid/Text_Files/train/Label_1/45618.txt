import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int min = (int) Math.ceil((double) (m + sum(a)) / n);
        int max = m + max(a);
        System.out.println(min + " " + max);
    }

    private static int sum(int[] a) {
        int sum = 0;
        for (int x : a) {
            sum += x;
        }
        return sum;
    }

    private static int max(int[] a) {
        int max = 0;
        for (int x : a) {
            max = Math.max(max, x);
        }
        return max;
    }
}
