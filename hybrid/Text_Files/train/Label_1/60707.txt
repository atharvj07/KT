import sys

def solve(n, k, s, t, cars, gas):
    cars.sort(key=lambda x: x[0])
    gas = [0] + gas + [s]
    best = -1

    for cost, cap in cars:
        time = 0
        fuel = cap
        prev = 0

        for i in range(1, len(gas)):
            dist = gas[i] - prev
            if fuel < dist:
                time += (dist - fuel) * 2
                if time > t:
                    break
                fuel = cap

            time += dist
            fuel -= dist
            prev = gas[i]

        if time <= t:
            best = cost

    return best

n, k, s, t = map(int, sys.stdin.readline().split())
cars = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]
gas = list(map(int, sys.stdin.readline().split()))

print(solve(n, k, s, t, cars, gas))
