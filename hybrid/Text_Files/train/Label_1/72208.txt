n = int(input())
k1, *a = map(int, input().split())
k2, *b = map(int, input().split())
assert len(a) == k1 and len(b) == k2
assert len(set(a + b)) == n
p1, p2 = 0, 0
t = 0
while True:
    t += 1
    if p1 == k1:
        print(-1)
        break
    if p2 == k2:
        print(-1)
        break
    if a[p1] > b[p2]:
        p1, p2 = p1+1, p2+1
        a.append(b[p2 - 1])
        b[p2 - 1] = a[p1 - 1]
    else:
        p1, p2 = p1+1, p2+1
        b.append(a[p1 - 1])
        a[p1 - 1] = b[p2 - 1]
else:
    print(t)
    if p1 == k1:
        print(2)
    else:
        print(1)
