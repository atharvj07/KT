A, B, C, N = map(int, input().split())

# Calculate the number of students who visited only one restaurant
only_BugDonalds = A - C
only_BeaverKing = B - C

# If the number of only one restaurant visitors is negative, it's impossible
if only_BugDonalds < 0 or only_BeaverKing < 0:
    print(-1)
else:
    # Calculate the total number of students who passed the exam
    passed_students = only_BugDonalds + only_BeaverKing + C
    
    # Calculate the number of students who didn't pass
    not_passed = N - passed_students

    # Check if the distribution is possible (at least one student didn't pass)
    if not_passed > 0:
        print(not_passed)
    else:
        print(-1)
