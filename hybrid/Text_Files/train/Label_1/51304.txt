import java.util.*;

public class LiegesOfLegendre {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        // Check if all piles have odd size
        boolean allOdd = true;
        for (int x : a) {
            if (x % 2 == 0) {
                allOdd = false;
                break;
            }
        }

        // If all piles have odd size, Nicky wins
        if (allOdd) {
            System.out.println("Nicky");
        } else {
            // Otherwise, Kevin wins
            System.out.println("Kevin");
        }
    }
}
