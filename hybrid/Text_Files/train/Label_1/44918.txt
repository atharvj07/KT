import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), r = in.nextInt();
        int[][] a = new int[n][2];
        for (int i = 0; i < n; i++) {
            a[i] = new int[]{in.nextInt(), in.nextInt()};
        }
        Arrays.sort(a, (p, q) -> Integer.compare(p[0], q[0]));
        int ans = 0, l = 0, j;
        for (int[] p : a) {
            while (l < p[0]) {
                j = l;
                while (j < n && a[j][1] <= l) j++;
                l = j + (j == n ? 0 : r);
            }
            if (l <= p[1]) ans++;
        }
        System.out.println(ans);
    }
}
