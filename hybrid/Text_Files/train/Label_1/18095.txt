#include <iostream>
#include <string>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int ans = 0, c = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'N' || (s[i] == '?' && k)) c++;
    else
      c = 0;
    ans = max(ans, c);
  }
  if (ans == k)
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
