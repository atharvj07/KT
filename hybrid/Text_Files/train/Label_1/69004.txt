import sys

n, m = map(int, input().split())
hates = dict()
for _ in range(m):
    a, b = map(int, input().split())
    if a not in hates:
        hates[a] = []
    hates[a].append(b)
    if b not in hates:
        hates[b] = []
    hates[b].append(a)

q = int(input())
salary = []
days = []

for _ in range(q):
    v = int(input())
    salary.append(v)
    days.append(list(set(hates.get(v, []) + [v])))

for i in range(1, n + 1):
    print(len({j for j in sys.stdin.readline().split() if j in days[salary.index(i)]}), end=' ')
