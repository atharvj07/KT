#include<bits/stdc++.h>
using namespace std;typedef long long LL;const LL mod=1e9+7;const int inf=0x3f3f3f3f;const double eps=1e-9;const LL INF=1e18;typedef pair<int,int>PII;typedef pair<LL,LL>PLL;typedef vector<int>VI;typedef vector<LL>VL;typedef vector<PII>VII;typedef vector<PLL>VLL;vector<VI>mp(100001);
int main(){
   int n,a[100001];
   cin>>n;
   for(int i=0;i<n;++i){
      cin>>a[i];
      mp[a[i]].push_back(i);
   }
   vector<int>b;
   b.push_back(a[0]);
   for(int i=1;i<n;++i){
      bool flag=0;
      for(int j=0;j<(int)mp[a[i]].size();++j){
         int pos=mp[a[i]][j];
         if(pos>0 && a[i]!=b[pos-1]){
            b.push_back(a[i]);
            mp[a[i]].erase(mp[a[i]].begin()+j);
            flag=1;
            break;
         }
      }
      if(!flag)return puts("No"),0;
   }
   cout<<"Yes"<<endl;
   for(auto x:b)cout<<x<<' ';
   return 0;
}
