#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> b(n);
  for (int i = 0; i < n; i++) cin >> b[i];
  int ans = 0;
  int l = 0, r = 0;
  for (int i = 0; i < k; i++) {
    r = min(m, b[i + 1] - 1);
    ans += r - l + 1;
    l = r + 1;
  }
  cout << ans << endl;
  return 0;
}
