import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        if (a < b) {
            System.out.println(-(b - a) * (b - a));
            for (int i = 0; i < b - a; i++) {
                System.out.print("x");
            }
            for (int i = 0; i < a; i++) {
                System.out.print("o");
            }
        } else if (a > b) {
            System.out.println((a - b) * (a - b));
            for (int i = 0; i < a - b; i++) {
                System.out.print("o");
            }
            for (int i = 0; i < b; i++) {
                System.out.print("x");
            }
        } else {
            System.out.println(0);
            for (int i = 0; i < a; i++) {
                System.out.print("o");
            }
            for (int i = 0; i < b; i++) {
                System.out.print("x");
            }
        }
    }
}
