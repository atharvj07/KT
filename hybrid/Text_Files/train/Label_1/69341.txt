n = int(input())
a = list(map(int, input().split()))

total_inv = 0
for i in range(n):
    for j in range(i + 1, n):
        if a[i] > a[j]:
            total_inv += 1

expected_inv = total_inv
for l in range(n):
    for r in range(l, n):
        k = r - l + 1
        inv_in_seg = 0
        for i in range(l, r + 1):
            for j in range(i + 1, r + 1):
                if a[i] > a[j]:
                    inv_in_seg += 1
        expected_inv += (inv_in_seg - k * (k - 1) / 4) / (n * (n + 1) / 2)

print(f'{expected_inv:.15f}')
