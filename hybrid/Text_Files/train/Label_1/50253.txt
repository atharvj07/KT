import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int z = scanner.nextInt();
        int killed = 0;
        for (int i = 1; i <= z; i++) {
            if ((i % n == 0) && (i % m != 0)) {
                killed++;
            }
        }
        System.out.println(killed);
    }
}
