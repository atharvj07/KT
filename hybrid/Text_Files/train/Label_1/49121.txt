import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    boolean arithmetic = true;
    boolean geometric = true;
    int d = in.nextInt();
    for (int i = 1; i < n; i++) {
      int x = in.nextInt();
      if (d != x) {
        arithmetic = false;
      }
      if (d % x != 0 && x % d != 0) {
        geometric = false;
      }
      d = x;
    }
    if (arithmetic) {
      System.out.println("0");
    } else if (geometric) {
      System.out.println("1");
    } else {
      System.out.println("2");
    }
  }
}
