import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long x = sc.nextLong();
        long y = sc.nextLong();
        if (2 * x <= y) {
            System.out.println(n * x);
        } else {
            System.out.println((n + 1) / 2 * y + (n + 1) % 2 * x);
        }
    }
}
