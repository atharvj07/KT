import sys

in1 = sys.stdin.readline().strip()
t = int(in1)
for i in range(t):
    in2 = sys.stdin.readline().strip()
    n = int(in2)
    in3 = sys.stdin.readline().strip().split()
    a = [int(j) for j in in3]
    s = set()
    for j in range(n):
        s.add(j)
    for j in range(n):
        if j + a[j] % n in s:
            s.remove(j + a[j] % n)
    if len(s) == 0:
        print("YES")
    else:
        print("NO")
