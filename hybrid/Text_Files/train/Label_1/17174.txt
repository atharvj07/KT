#include<bits/stdc++.h>
using namespace std;
#define vi vector<ll>
#define all(v) v.begin(),v.end()
#define pb push_back
typedef long long ll;

int32_t main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	ll n; cin >> n; vi v(n); for (ll &x : v)cin >> x;
	ll cnt1 = 0, ans = 0;
	for (ll x : v) {
		ll sq = sqrt(x);
		if (sq * sq == x)cnt1++;
	}
	if (cnt1 == n / 2) {
		cout << 0; return 0;
	}
	ans += cnt1; if (cnt1 >= n / 2)ans += cnt1 - n / 2;
	else ans += n / 2 - cnt1;
	cout << ans;
	return 0;
}
