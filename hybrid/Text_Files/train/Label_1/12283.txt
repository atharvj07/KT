#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll P = 998244353;
int main(){
    ios_base::sync_with_stdio(false); cin.tie(0); cout.tie(0);
    ll n, a[200005], b[200005], s[200005], ans;
    cin>>n;
    for(ll i = 1; i <= n; i++)    cin>>a[i];
    for(ll i = 1; i <= n; i++)    cin>>b[i], s[i] = s[i - 1] + b[i];
    sort(b + 1, b + n + 1);
    for(ll i = 1; i <= n; i++){
        ans = (ans + a[i] * (s[n] - s[i])) % P;
        ans = (ans + a[i] * (s[i - 1] + b[i] * i)) % P;
    }
    cout<<ans<<endl;
    return 0;
}
