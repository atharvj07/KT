mod = 1000000007

def factorial(n):
    f = 1
    for i in range(1, n + 1):
        f = (f * i) % mod
    return f

def solve(N, requests):
    fixed_positions = 0
    movable_positions = 0
    desk_map = {}
    for current_desk, desired_desk in requests:
        if current_desk == desired_desk:
            fixed_positions += 1
        else:
            if desired_desk in desk_map:
                return 0
            desk_map[desired_desk] = current_desk
    movable_positions = N - fixed_positions
    return factorial(movable_positions) % mod

N = int(input().strip())
requests = [tuple(map(int, input().strip().split())) for _ in range(N)]
print(solve(N, requests))
