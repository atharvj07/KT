import sys

n, k = map(int, input().split())
a = list(map(int, input().split()))
pre = [1]
for i in a:
    pre.append(pre[-1] * i)
suf = [1]
for i in a[::-1]:
    suf.append(suf[-1] * i)
suf.reverse()
ans = 0
for i in range(n):
    for j in range(n):
        if (pre[i] * suf[j]) % k == 0:
            ans += 1
print(ans)
