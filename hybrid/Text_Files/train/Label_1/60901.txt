for t in range(int(input())):
    s = input()
    if all(s[i] == s[-i -1] for i in range(len(s) // 2)):
        print('NO')
    else:
        print('YES')
        if s[0] != s[-1]:
            print(s + 'a')
        else:
            i = 0
            while s[i] == s[-(i + 1)]:
                i += 1
            print(s[:i] + 'a' + s[i:])
