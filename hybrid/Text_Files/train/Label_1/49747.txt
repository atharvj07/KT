import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.Arrays;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input1 = br.readLine().split(" ");
        int n = Integer.parseInt(input1[0]);
        int m = Integer.parseInt(input1[1]);
        Frog[] frogs = new Frog[n];
        for (int i = 0; i < n; i++) {
            String[] input2 = br.readLine().split(" ");
            frogs[i] = new Frog(Integer.parseInt(input2[0]), Integer.parseInt(input2[1]));
        }
        Mosquito[] mosquitoes = new Mosquito[m];
        for (int i = 0; i < m; i++) {
            String[] input3 = br.readLine().split(" ");
            mosquitoes[i] = new Mosquito(Integer.parseInt(input3[0]), Integer.parseInt(input3[1]));
        }
        Arrays.sort(mosquitoes);
        for (Mosquito mosquito : mosquitoes) {
            for (Frog frog : frogs) {
                if (frog.eat(mosquito)) {
                    break;
                }
            }
        }
        for (Frog frog : frogs) {
            System.out.println(frog.count + " " + frog.tongue);
        }
    }

    static class Frog {
        int position;
        int tongue;
        int count;

        public Frog(int position, int tongue) {
            this.position = position;
            this.tongue = tongue;
            this.count = 0;
        }

        public boolean eat(Mosquito mosquito) {
            if (mosquito.position >= position && mosquito.position <= position + tongue) {
                tongue += mosquito.size;
                count++;
                return true;
            }
            return false;
        }
    }

    static class Mosquito implements Comparable<Mosquito> {
        int position;
        int size;

        public Mosquito(int position, int size) {
            this.position = position;
            this.size = size;
        }

        @Override
        public int compareTo(Mosquito other) {
            return position - other.position;
        }
    }
}
