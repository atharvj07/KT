import sys

h, w = map(int, input().split())
grid = [list(input()) for _ in range(h)]
q = int(input())

for _ in range(q):
    r1, c1, r2, c2 = map(int, input().split())
    r1 -= 1
    c1 -= 1
    r2 -= 1
    c2 -= 1
    ans = 0
    for i in range(r1, r2):
        if grid[i][c1] == '.' and grid[i][c2] == '.':
            ans += 1
    for i in range(c1, c2):
        if grid[r1][i] == '.' and grid[r2][i] == '.':
            ans += 1
    print(ans)
