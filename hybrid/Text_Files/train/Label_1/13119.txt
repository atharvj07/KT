#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main()
{
  long long n, k, i, j;
  cin >> n >> k;
  vector<long long>v(n);
  if (k == 1)
  {
    for (i = 0; i < n; i++)
    {
      cout << i + 1 << " ";
    }
  }
  else if (k == 2)
  {
    for (i = 0; i < n; i += 2)
    {
      v[i] = i + 2;
    }
    for (i = 1; i < n; i += 2)
    {
      v[i] = i + 1;
    }
    for (i = 0; i < n; i++)
    {
      cout << v[i] << " ";
    }
  }
  else
  {
    long long a = 1, b = 3;
    for (i = 0; i < k - 1; i++)
    {
      for (j = a; j <= b; j++)
      {
        v[j] = j + 2 * (i + 1);
      }
      a += 2;
      b += 2;
    }
    v[0] = 2;
    for (i = 0; i < n; i++)
    {
      if (!v[i])
      {
        v[i] = i + 1;
      }
    }
    for (i = 0; i < n; i++)
    {
      cout << v[i] << " ";
    }
  }
  return 0;
}
