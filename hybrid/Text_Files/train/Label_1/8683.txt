#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 1e5 + 5;
vector<int> adj[N], adjt[N];
bool vis[N];
int n, m;
int ans;
void dfs(int node, vector<int> adj[]) {
  vis[node] = 1;
  for (auto it : adj[node]) {
    if (!vis[it]) {
      dfs(it, adj);
    }
  }
}
int32_t main() {
  cin >> n >> m;
  for (int i = 0; i < m; ++i) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adjt[v].push_back(u);
  }
  for (int i = 1; i <= n; ++i) {
    if (!vis[i]) {
      dfs(i, adj);
      ++ans;
    }
  }
  for (int i = 1; i <= n; ++i) {
    if (!vis[i]) {
      dfs(i, adjt);
      ++ans;
    }
  }
  cout << ans << endl;
  return 0;
}
