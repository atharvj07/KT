#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const ll mod = 998244353;
int main() {
    ll n, m, l;
    cin >> n >> m >> l;
    vector<ll> b(m);
    for (int i = 0; i < m; i++) {
        cin >> b[i];
    }
    ll ans = 1;
    for (ll x : b) {
        ans = ans * max(x, n - x) % mod;
    }
    cout << ans << endl;
    return 0;
}
