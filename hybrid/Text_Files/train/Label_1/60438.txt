from functools import lru_cache

def num_perm(n, arr):
    @lru_cache(maxsize=None)
    def dp(i, t):
        if i == n:
            return 1
        ans = (dp(i+1, t) * (t-arr[i]+1)) % mod
        for j in range(i+1, n):
            if arr[j] > t:
                break
            ans = (ans + (dp(j+1, t-arr[j])*arr[j])) % mod
        return ans
    mod = 1000000007
    ans = dp(0, sum(arr))
    return ans
n = int(input())
arr = list(map(int, input().split()))
print(num_perm(n, arr))
