#include <algorithm>
#include <iostream>

using namespace std;

int main() {
  int sum, t;
  cin >> t;
  while (t--) {
    int n, mx, mn;
    cin >> n;
    cin >> mx >> mn;
    sum = mx + mn;
    for (int i = 2; i < n; i++) {
      int val;
      cin >> val;
      mn = min(mn, val);
      mx = max(mx, val);
      sum += val;
    }
    if (sum % (n - 2) == 0)
      cout << sum / (n - 2) << endl;
    else
      cout << -1 << endl;
  }
  return 0;
}
