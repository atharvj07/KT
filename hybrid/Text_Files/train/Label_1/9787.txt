#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll n,p[1000005];
ll dp[1000005];
int main()
{
    cin >> n;
    for(ll i = 0; i < n; i++)
    {
        cin >> p[i];
        p[i+n] = p[i];
    }
    vector<ll> v;
    ll ans = 1e18, idx = 0;
    for(ll i = 0; i < n; i++)
    {
        dp[i+1] = dp[i]+abs(i+1-p[i]);
    }
    for(ll i = 0; i < n; i++)
    {
        dp[i+1] = min(dp[i+1], dp[i]+n-abs(i+1-p[i]));
    }
    for(ll i = 0; i < n; i++)
    {
        if(dp[i+1] < ans)
        {
            ans = dp[i+1];
            idx = i+1;
        }
    }
    cout << ans << ' ' << idx%n << endl;
}
