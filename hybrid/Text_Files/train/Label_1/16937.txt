#include <bits/stdc++.h>
using namespace std;
#define f first
#define s second
void go(int);
vector<pair<int, int>> adj[200002];
int par[200002], dep[200002], ans, n;
int vis[200002], mark[200002][2]={0};
int dfs(int u) {
  if (vis[u] == 1) { return par[u]; }
  vis[u] = 1;
  for (auto v : adj[u]) {
    par[v.f] = u;
    dep[v.f] = dep[u] + 1;
    if (v.s == 1)
      if (vis[v.f] == 2) ans++;
    int x = dfs(v.f);
    if (x != -1) return x;
  }
  vis[u] = 2;
  return -1;
}
int main() {
  int t, x, y, a, b, m;
  cin >> t;
  while (t--) {
    memset(adj, 0, sizeof adj);
    memset(par, 0, sizeof par);
    memset(dep, 0, sizeof dep);
    memset(mark, 0, sizeof mark);
    memset(vis, 0, sizeof vis);
    cin >> n >> m >> a >> b;
    int f1 = -1, f2 = -1;
    while (m--) {
      cin >> x >> y;
      adj[x].push_back({y, 0});
      adj[y].push_back({x, 0});
    }
    vis[b] = 1;
    dfs(b);
    vis[a] = 1;
    ans = 0;
    dfs(a);
    cout << ans << "\n";
  }
  return 0;
}
