#include <bits/stdc++.h>
using namespace std;
const int MM = 1e4 + 5;
int n, disc[MM], low[MM], par[MM], timer = 1;
vector<vector<int>> g;
void dfs(int u) {
    disc[u] = timer;
    low[u] = timer++;
    for (int v : g[u]) {
        if (!disc[v]) {
            par[v] = u;
            dfs(v);
            low[u] = min(low[u], low[v]);
        } else if (par[u] != v)
            low[u] = min(low[u], disc[v]);
    }
}
int main() {
    cin >> n;
    g.resize(n + 1);
    for (int i = 0, a, b; i < n - 1; i++) {
        cin >> a >> b;
        g[a].push_back(b);
        g[b].push_back(a);
    }
    dfs(1);
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        int ch = 0;
        for (int v : g[i]) {
            if (low[v] > disc[i]) ch++;
        }
        if (ch > 1) cnt += ch - 1;
    }
    cout << cnt << endl;
}
