#include <iostream>
#include <vector>
#include <limits>

using namespace std;

int main() {
  int w, l;
  cin >> w >> l;
  vector<int> stones(w - 1, 0);
  for (int i = 0; i < w - 1; i++) {
    cin >> stones[i];
  }
  vector<int> dp(w, 0);
  int max_frogs = 0;
  for (int i = l; i < w; i++) {
    int cur_max = 0;
    for (int j = i - l; j < i; j++) {
      if (j >= 0 && stones[j] > 0) {
        cur_max = max(cur_max, dp[j] + 1);
      }
    }
    dp[i] = cur_max;
    max_frogs = max(max_frogs, dp[i]);
  }
  cout << max_frogs << endl;
  return 0;
}
