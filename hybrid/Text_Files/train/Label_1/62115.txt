n, m = map(int, input().split())
a = []
b = []
# for managing the size of the files before and after compress
for _ in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)
# Sorting the songs to optimize
a.sort()
b.sort()
# try to compress each song with smallest size
ans, cur = 0, 0
for i in range(n):
    if cur + b[i] <= m:
        cur += b[i]
        ans += 1
    else:
        break
if cur + a[0] <= m:
    ans += 1
else:
    ans = -1
print(ans)
