#include <iostream>
#include <vector>
#include <algorithm>
#include <cstdint>
using namespace std;

int main() {
    int K;
    cin >> K;
    vector<uint32_t> ranks(K);
    for (uint32_t& r : ranks) cin >> r;
    sort(ranks.begin(), ranks.end());
    uint32_t l = ranks[0] - 1, r = 25 - K + 1;
    for (int i = 1; i < K; i++)
        l = max(l, ranks[i] - ranks[i-1] - 1);
    cout << l << endl;
    return EXIT_SUCCESS;
}
