import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), m = sc.nextInt();
    int[] pos = new int[n + 1];
    for (int i = 0; i < m; ++i) {
      pos[sc.nextInt()] = i;
    }
    int ans = m - 1;
    for (int i = 2; i <= n; ++i) {
      ans += Math.min(pos[i], pos[i - 1]) - Math.min(pos[i - 1], pos[i - 2]);
    }
    System.out.println(ans);
  }
}
