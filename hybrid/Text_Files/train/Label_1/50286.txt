import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        boolean isPrime = true;
        for (int i = 2; i <= Math.sqrt(m); i++) {
            if (m % i == 0) {
                isPrime = false;
            }
        }

        if (isPrime && n + 2 == m) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
