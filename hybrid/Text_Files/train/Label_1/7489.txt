#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using vi = vector<ll>;

int n, a[100005], b[100005], cnt = 0;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    vi v(n);
    for (int i = 0; i < n; ++i) {
        cin >> v[i];
        if (v[i] > 0) a[v[i]] = 1;
        else b[-v[i]] = 1;
    }
    bool found = true;
    for (int i = 1; i <= 1e6; ++i) {
        if (a[i] != b[i]) {
            found = false;
            break;
        }
    }
    if (!found) {
        cout << -1 << '\n';
        return 0;
    }
    for (int i = 0; i < n; ++i) {
        if (!v[i]) continue;
        if (v[i] > 0)
            ++cnt, b[v[i]] = 0;
        else
            b[-v[i]] = 0;
        if (b[v[i]] == 0) {
            cout << cnt << '\n';
            for (int j = i + 1; j < n; ++j) {
                if (v[j] == 0) continue;
                if (v[j] > 0)
                    ++cnt, b[v[j]] = 0;
                else
                    b[-v[j]] = 0;
                if (b[v[j]] == 0) {
                    cout << j - i << ' ';
                    i = j, cnt = 0;
                }
            }
            return 0;
        }
    }
    return 0;
}
