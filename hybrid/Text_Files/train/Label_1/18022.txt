#include <bits/stdc++.h>
#define int long long
using namespace std;
const int N = 5005, inf = 1e9;
vector<int> v[N];
int a[N], b[N], teams[N][2], x[N];
int32_t main() {
    int n, m, d, k;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) cin >> b[i], v[b[i]].push_back(a[i]);
    cin >> d;
    while (d--) {
        cin >> k;
        for (int j = 1; j <= m; j++) teams[j][0] = teams[j][1] = 0;
        v[b[k]].pop_back();
        for (int j = 1; j <= m; j++) {
            if (v[j].empty()) continue;
            teams[j][0] = *min_element(v[j].begin(), v[j].end());
            teams[j][1] = *max_element(v[j].begin(), v[j].end());
        }
        x[0] = inf;
        for (int j = 1; j <= m; j++) {
            if (teams[j][0] == 0) continue;
            x[teams[j][1]] = min(x[teams[j][1]], teams[j][0]);
        }
        int ans = 0;
        for (int i = 0; i < 5005; i++) {
            if (x[i] == 0) continue;
            if (x[i] != inf) ans = max(ans, i);
        }
        cout << ans << ' ';
    }
}
