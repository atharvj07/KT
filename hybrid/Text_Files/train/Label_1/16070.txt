#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 10;
long long a[MAXN], ans[MAXN];
int32_t main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  int cnt = 0, pos;
  for (int i = 1; i <= n; i++) {
    if (a[i] == 0) {
      cnt++;
      pos = i;
      continue;
    }
    if (cnt > 0) {
      ans[++cnt] = 1;
      ans[++cnt] = pos;
      ans[++cnt] = i;
      a[pos] = a[pos] * a[i];
      a[i] = 0;
    }
  }
  if (cnt == 1) {
    ans[++cnt] = 2;
    ans[++cnt] = pos;
  }
  for (int i = 1; i <= cnt; i++) {
    if (ans[i] == 1)
      cout << "1 " << ans[i + 1] << " " << ans[i + 2] << "\n";
    else
      cout << "2 " << ans[i + 1] << "\n";
    i += 2;
  }
}
