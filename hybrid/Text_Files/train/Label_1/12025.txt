#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> cities(n);
  for (int i = 0; i < n; i++) {
    int a, c;
    cin >> a >> c;
    cities[i] = {a, c};
  }
  vector<int> min_cost_to(n, 2e9);
  min_cost_to[0] = 0;
  for (int i = 1; i < n; i++) {
    for (int j = 0; j < i; j++) {
      min_cost_to[i] = min(min_cost_to[i], min_cost_to[j] + max(cities[j].second, cities[i].first - cities[j].first));
    }
  }
  cout << min_cost_to[n - 1] << endl;
  return 0;
}
