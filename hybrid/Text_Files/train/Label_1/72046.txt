def solve():
    n,m=map(int,input().split())
    a=[int(i) for i in input().split()]
    while m:
        m-=1
        p,b=map(int,input().split())
        a[p-1]=b
        ans=0
        for i in range(0,n):
            l=[]
            j=0
            if (1<<i) & 1:
                st=2
            else: st=1
            while j<=(1<<n)-1:
                l.append(a[j])
                j+=st
            if (1<<i) & 1:
                ans^=l[0]
            else: ans|=l[0]
            for k in range(1,len(l)):
                if (1<<i) & 1:
                    ans^=l[k]
                else: ans|=l[k]
        print(ans)
solve()
