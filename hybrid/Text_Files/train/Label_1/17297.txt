#include <bits/stdc++.h>

using namespace std;

const long long INF = (1LL << 60);
int n;
long long a[7001], b[7001];
vector<long long> adj[7001];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> b[i];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (i == j) continue;
      if ((a[i] & a[j]) == a[j]) adj[i].push_back(j);
    }
  }
  long long dp[7001];
  for (int i = 1; i <= n; i++) dp[i] = -INF;
  dp[1] = b[1];
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < adj[i].size(); j++) {
      int u = adj[i][j];
      dp[u] = max(dp[u], dp[i] + b[u]);
    }
  }
  long long ans = -INF;
  for (int i = 1; i <= n; i++) ans = max(ans, dp[i]);
  cout << (ans <= -INF ? 0 : ans) << endl;
  return 0;
}
