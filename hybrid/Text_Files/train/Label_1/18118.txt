#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s, t;
    cin >> s >> t;
    unordered_map<char, int> cnt_s, cnt_t;
    for (int i = 0; i < n; i++) {
        cnt_s[s[i]]++;
        cnt_t[t[i]]++;
    }
    int ans = 0;
    for (char i = 'a'; i <= 'z'; i++) {
        if (cnt_s[i] + 1 == cnt_t[i]) ans++;
        if (cnt_s[i] == cnt_t[i] + 1) ans++;
    }
    cout << ans;
    return 0;
}
