#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int mod = 998244353;

ll power(ll x, ll y) {
  ll ans = 1;
  while(y > 0) {
    if(y & 1) ans = (ans * x) % mod;
    x = (x * x) % mod;
    y >>= 1;
  }
  return ans;
}

int main() {
  ll n, m, L, R, ans;
  cin >> n >> m >> L >> R;
  if(n * m % 2 == 0) {
    ans = power(2, n * m / 2);
  } else {
    ll a = power(2, n * m / 2 - 1);
    ll b = power(2, n * m / 2 - 1);
    ll c = power(2, n * m / 2);
    ans = (a + b + c) % mod;
  }
  cout << ans << endl;
  return 0;
}
