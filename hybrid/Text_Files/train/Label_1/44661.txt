import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();

        // create adjacency list
        List<Integer>[] adj = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }

        // read input
        while (m-- > 0) {
            int a = input.nextInt();
            int b = input.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }

        int[] party = new int[n + 1];

        // try to split horses into two parties
        if (!dfs(adj, party, 1, 0)) {
            System.out.println(-1);
            return;
        }

        // print result
        for (int i = 1; i <= n; i++) {
            System.out.print(party[i]);
        }
        System.out.println();
    }

    // dfs to check if it is possible to split horses into two parties
    public static boolean dfs(List<Integer>[] adj, int[] party, int v, int p) {
        // mark the current horse as belonging to party p
        party[v] = p;

        // iterate over the neighbors of the current horse
        for (int u : adj[v]) {
            // if the neighbor is already in the same party, return false
            if (party[u] == p) {
                return false;
            }

            // if the neighbor is not yet assigned to a party, try to assign it to the other party
            if (party[u] == 0 && !dfs(adj, party, u, 1 - p)) {
                return false;
            }
        }

        // if all neighbors can be assigned to a different party, return true
        return true;
    }
}
