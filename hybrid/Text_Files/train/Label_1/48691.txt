import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), x = sc.nextInt(), a[] = new int[n + 2], cnt = 0, sum = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            sum += a[i];
            cnt += a[i] < x ? 1 : 0;
        }
        Arrays.sort(a, 0, n);
        int l = 0, r = n - 1, ans = n;
        while (l <= r) {
            int mid = (l + r) / 2;
            int tot = cnt + n - mid - 1 + (a[mid] == x ? 1 : 0);
            if (tot * 2 >= n) {
                ans = sum - a[mid] * mid - (n - mid - 1 - (a[mid] == x ? 0 : cnt - (a[mid] == x ? 1 : 0))) * x;
                r = mid - 1;
            } else
                l = mid + 1;
        }
        System.out.println(ans);
    }
}
