import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long k = sc.nextLong();
        
        if (n == 1) {
            System.out.println(0);
            return;
        }
        
        if (k < n - 1) {
            System.out.println(-1);
            return;
        }
        
        long ans = (n - 2 + k) / k - 1;
        System.out.println(ans);
    }
}
