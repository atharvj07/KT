#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> adj(n + 1);
  while (n--) {
    int a, b;
    cin >> a >> b;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  vector<bool> vis(n + 1);
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    if (!vis[i]) {
      cnt++;
      queue<int> q;
      q.push(i);
      vis[i] = 1;
      while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : adj[u]) {
          if (!vis[v]) {
            q.push(v);
            vis[v] = 1;
          }
        }
      }
    }
  }
  cout << cnt << "\n";
}
