#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define vll vector<ll>
#define pb push_back
#define pl pair<ll, ll>
#define vpll vector<pl>
#define all(x) x.begin(), x.end()
#define rep(i, n) for (ll i = 0; i < (n); i++)

int n, m;
vpll plants;

int main()
{
    priority_queue<ll> pq;
    cin >> n >> m;
    rep(i, n)
    {
        ll tag, pos;
        cin >> tag >> pos;
        --tag;
        if (tag < m - 1)
            pq.push(pos);
    }
    ll ans = 0;
    while (!pq.empty())
    {
        if (pq.top() < 0)
        {
            ans += -1 - pq.top();
            pq.push(pq.top() + 1);
        }
        pq.pop();
    }
    cout << ans << "\n";
}
