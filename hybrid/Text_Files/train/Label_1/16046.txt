#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> p, s;
vector<pair<int, int>> ans;
map<int, int> pos;
int main() {
    cin >> n;
    p.resize(n + 1), s.resize(n + 1);
    for (int i = 1; i <= n; i++) { cin >> p[i]; }
    for (int i = 1; i <= n; i++) { cin >> s[i], pos[s[i]] = i; }
    if (p == s) {
        cout << 0 << endl;
        cout << 0 << endl;
        return 0;
    }
    for (int i = 1; i <= n; i++) {
        if (p[i] != s[i]) {
            if (s[i] == p[pos[p[i]]]) {
                ans.push_back({i, pos[p[i]]});
                swap(p[i], p[pos[p[i]]]);
                pos[p[i]] = i;
                pos[p[pos[p[i]]]] = pos[p[i]];
            } else {
                ans.push_back({pos[p[i]], pos[s[i]]});
                swap(p[pos[p[i]]], p[pos[s[i]]]);
                pos[p[pos[p[i]]]] = pos[p[i]];
                pos[p[i]] = pos[s[i]];
                ans.push_back({i, pos[s[i]]});
                swap(p[i], p[pos[s[i]]]);
                pos[p[i]] = i;
                pos[p[pos[p[i]]]] = pos[p[i]];
            }
        }
    }
    cout << ans.size() << endl;
    cout << ans.size() << endl;
    for (auto x : ans) cout << x.first << ' ' << x.second << endl;
}
