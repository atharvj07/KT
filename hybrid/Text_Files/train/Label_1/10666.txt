#include <bits/stdc++.h>
using namespace std;
const int INF = 1<<30;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
ll a[200005];
int p[200005];
vector<int> g[200005];
inline int find(int u) {
    return p[u] == u ? u : p[u] = find(p[u]);
}
ll dp[200005];
void dfs(int u, int fa) {
    dp[u] = a[u];
    for (int v : g[u]) {
        if (v == fa) continue;
        dfs(v, u);
        dp[u] += max(dp[v], 0LL);
    }
}
ll dfs2(int u, int fa) {
    ll ret = dp[u];
    for (int v : g[u]) {
        if (v == fa) continue;
        ll pick = dp[u] - max(0LL, dp[v]) + dp[v];
        ll leave = dp[u];
        ret = max(ret, max(pick, leave) + dfs2(v, u));
    }
    return ret;
}
int main() {
    int n; cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        p[i] = i;
    }
    for (int i = 1; i < n; i++) {
        int u, v; cin >> u >> v;
        u = find(u);
        v = find(v);
        if (u == v) continue;
        p[u] = v;
        g[v].push_back(u);
    }
    dfs(1, 1);
    if (n == 1) {
        cout << max(a[1], 0LL) << '\n';
        return 0;
    }
    ll ans = dfs2(1, 1);
    ll other = dp[1];
    for (int v : g[1]) {
        other = max(other, dp[1] - max(0LL, dp[v]) + dp[v]);
    }
    ans = max(ans, other);
    cout << ans << '\n';
    return 0;
}
