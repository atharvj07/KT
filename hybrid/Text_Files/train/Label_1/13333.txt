#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  int pref = 0;
  for (int i = 0; i < n; i++) {
    pref += a[i];
    if (pref == 0) {
      ans++;
    } else if (pref < 0) {
      ans += abs(pref) + 1;
      pref = 1;
    }
  }
  cout << ans << endl;
  return 0;
}
