#include <cstdio>
#include <cstring>
int main() {
    char s[105];
    scanf("%s", s);
    int n = strlen(s), cnt = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'V' && (i + 1 < n && s[i + 1] == 'K')) {
            cnt++;
        }
    }
    printf("%d\n", cnt + (s[n - 1] == 'V' && s[0] == 'K'));
    return 0;
}
