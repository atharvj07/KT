#include <bits/stdc++.h>
using namespace std;
#define N 100000
int n, t, a[N], ans, ca;
int main() {
  cin >> n >> ca;
  for (int i = 0; i < n; i++) { cin >> t >> a[i]; }
  ans = (ca != 1);
  for (int i = n - 1; i >= 0; i--) {
    if (a[i] != 0) { ans &= !(a[i] % ca); }
    if (a[i] == 0) { ans &= (ca % n == 1); }
  }
  cout << (ans ? "Yes" : "No");
}
