import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int k = sc.nextInt();
    int[][] arr = new int[n][m];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        arr[i][j] = sc.nextInt();
      }
    }
    int[][] dp = new int[n][m];
    int max = 0;
    for (int i = k - 1; i < n - k; i++) {
      for (int j = k - 1; j < m - k; j++) {
        int sum = 0;
        if (i - k >= 0 && j - k >= 0) {
          sum -= dp[i - k][j - k];
        }
        if (i - k >= 0) {
          sum += dp[i - k][j];
        }
        if (j - k >= 0) {
          sum += dp[i][j - k];
        }
        sum += arr[i][j];
        dp[i][j] = sum;
        max = Math.max(max, sum);
      }
    }
    for (int i = k - 1; i < n - k; i++) {
      for (int j = k - 1; j < m - k; j++) {
        if (dp[i][j] == max) {
          System.out.println((i + 1) + " " + (j + 1));
          return;
        }
      }
    }
  }
}
