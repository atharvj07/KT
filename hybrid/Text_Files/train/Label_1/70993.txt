import sys

def solve():
  n, m = [int(x) for x in input().split()]
  a = input().split()
  b = input().split()
  for i in range(n):
    a[i] = int(a[i]) - int(b[i])
  graph = [[] for i in range(n)]
  for i in range(m):
    x, y = [int(x) for x in input().split()]
    x -= 1
    y -= 1
    graph[x].append(y)
    graph[y].append(x)

  ok = True
  for i in range(n):
    c = [False for j in range(n)]
    c[i] = True
    q = [i]
    sa = a[i]
    while len(q)>0:
      x = q.pop()
      for y in graph[x]:
        if c[y]: continue
        c[y] = True
        q.append(y)
        sa += a[y]
    if sa!=0:
      ok = False
  if ok:
    print("YES")
  else:
    print("NO")

for i in range(int(input())):
  solve()
