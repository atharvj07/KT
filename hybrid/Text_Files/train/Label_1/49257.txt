import java.util.ArrayList;
import java.util.Iterator;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        int d = sc.nextInt();
        ArrayList<Integer>[] clubs = new ArrayList[m];
        for (int i = 0; i < m; i++) {
            clubs[i] = new ArrayList<>();
        }
        for (int i = 0; i < n; i++) {
            clubs[c[i] - 1].add(i);
        }
        for (int i = 0; i < d; i++) {
            int k = sc.nextInt();
            k--;
            clubs[c[k] - 1].remove(Integer.valueOf(k));
            int max = -1;
            for (ArrayList<Integer> club : clubs) {
                if (!club.isEmpty()) {
                    int mex = 0;
                    Iterator<Integer> it = club.iterator();
                    while (it.hasNext()) {
                        int q = p[it.next()];
                        if (q == mex) {
                            mex++;
                            while (it.hasNext()) {
                                q = p[it.next()];
                                if (q <= mex) {
                                    mex++;
                                }
                            }
                            break;
                        } else {
                            mex = q + 1;
                        }
                    }
                    max = Math.max(max, mex);
                }
            }
            System.out.println(max);
        }
    }
}
