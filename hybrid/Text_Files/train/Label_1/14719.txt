#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  int n = a.size(), m = b.size();
  vector<vector<int>> dp(n + 1, vector<int>(m + 1, -1));
  dp[n][m] = 1;

  for (int i = n - 1; i >= 0; i--) {
    for (int j = m - 1; j >= 0; j--) {
      if (dp[i][j] == -1) continue;

      if (a[i] == '0' && b[j] == '0') dp[i][j] |= dp[i + 1][j];
      if (a[i] == '1' && (b[j] == '0' || b[j] == '1'))
        dp[i][j] |= dp[i + 1][j + 1];
      if (a[i] == '0' && b[j] == '0') dp[i][j] |= dp[i][j + 1];
    }
  }

  cout << (dp[0][0] ? "YES" : "NO") << endl;
  return 0;
}
