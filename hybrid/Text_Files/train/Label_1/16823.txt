#include <iostream>
#include <vector>
#include <cstring>
using namespace std;
const int mod = 1e9 + 7;
int nxt[1000010][26];
int f[1000010], lnk[1000010];

int main() {
    cin.tie(0);
    ios::sync_with_stdio(0);
    string s, t;
    cin >> s >> t;
    int u = 0, v;
    for (auto x : s) {
        v = ++u;
        for (int i = 0; i < 26; i++)
            nxt[v][i] = nxt[u][i];
        nxt[v][x - 'a'] = v + 1;
    }
    for (int j = 0; j < 26; j++)
        nxt[u][j] = u;
    u = v = 0;
    for (auto x : t) {
        u = nxt[u][x - 'a'];
        lnk[u] = v;
        v = u;
    }
    int cur = 0, L = s.length();
    vector<int> arr = {0};
    for (auto x : t) {
        cur = nxt[cur][x - 'a'];
        while (cur > L)
            cur = lnk[cur];
        if (cur == L) {
            arr.push_back(0);
            cur = 0;
        }
    }
    for (int i = 0; i < (int)arr.size(); i++) {
        cur = 0;
        for (int j = i + 1; j < (int)arr.size(); j++) {
            cur = nxt[cur][t[j - 1] - 'a'];
            while (cur > L)
                cur = lnk[cur];
            if (cur == L) {
                arr[i]++;
                arr[j]++;
            }
        }
    }
    int ans = 0;
    for (int i = 0; i < (int)arr.size(); i++)
        ans += (1LL * arr[i] * (arr[i] + 1)) / 2;
    cout << ans << "\n";
}
