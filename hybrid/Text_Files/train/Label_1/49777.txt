import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] f = new int[n + 1][n + 1];
        for (int j = 1; j <= n; j++) {
            f[1][j] = a[j];
        }
        for (int i = 2; i <= n; i++) {
            for (int j = i; j <= n; j++) {
                f[i][j] = Math.min(f[i - 1][j], f[i - 1][j - 1]) + a[j];
            }
        }
        int m = sc.nextInt();
        for (int i = 0; i < m; i++) {
            int xi = sc.nextInt();
            int yi = sc.nextInt();
            System.out.println(f[xi][yi]);
        }
    }
}
