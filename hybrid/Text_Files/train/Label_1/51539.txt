import java.util.*;

public class Main {
    static int[][] map;
    static int n, m;
    static Point start;
    static int[] price;
    static int cnt;
    static Set<Integer> inside;
    static int res;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        map = new int[n][m];
        inside = new HashSet<>();
        int t = 0, x = -1, y = -1;
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                char ch = s.charAt(j);
                if (ch == '#') {
                    map[i][j] = -1;
                } else if (ch >= '1' && ch <= '8') {
                    map[i][j] = ch - '0';
                    t++;
                } else if (ch == 'B') {
                    map[i][j] = -2;
                } else if (ch == 'S') {
                    map[i][j] = 0;
                    start = new Point(i, j);
                    x = i;
                    y = j;
                }
            }
        }
        price = new int[t + 1];
        for (int i = 1; i <= t; i++) price[i] = sc.nextInt();
        dfs(x, y, 1 << (map[x][y] == 0 ? 0 : map[x][y]), 0, 0);
        System.out.println(res);
    }

    static void dfs(int x, int y, int mask, int sum, int step) {
        if (x < 0 || x >= n || y < 0 || y >= m || map[x][y] == -2 || (mask & (1 << map[x][y])) != 0)
            return;
        inside.add(map[x][y]);
        mask |= (1 << map[x][y]);
        if (x == start.x && y == start.y && inside.size() > 1) {
            res = Math.max(res, sum - step);
        }
        sum += price[map[x][y]];
        step++;
        dfs(x + 1, y, mask, sum, step);
        dfs(x - 1, y, mask, sum, step);
        dfs(x, y + 1, mask, sum, step);
        dfs(x, y - 1, mask, sum, step);
    }

    static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
