#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int a = 0, b = 0, c = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '0') {
        if (i == n - i - 1) {
          c += 1;
        } else {
          a += 1;
        }
      }
      if (!i || s[i] != s[i - 1]) b = 1;
    }
    if (b == 1) {
      if (a <= c)
        cout << "ALICE\n";
      else
        cout << "BOB\n";
    } else if (n % 2)
      cout << "ALICE\n";
    else
      cout << "DRAW\n";
  }
  return 0;
}
