def move(n,s):
	sum=0
	while n>0:
		sum+=n%10
		n//=10
	if sum<=s:
		return 0
	ans=n=0
	arr=[0]*10
	while n<=s:
		arr[n%10]+=1
		n=10*n+9
	ans=sum-s
	while ans>0:
		if arr[s-(sum-ans)]:
			ans-=1
			arr[s-(sum-ans)]-=1
		else:
			ans+=1
	return ans
n=int(input())
for _ in range(n):
	print(move(*[int(x) for x in input().split()]))
