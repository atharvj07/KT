#include <bits/stdc++.h>
#define SZ(v) (int)(v).size()
using namespace std;
int n, ans[100005], lcp[100005], cur[26], sz[26], d[26][100005], mx[26][100005], q, cnt[100005];
string nam[100005], nick[100005];
int main() {
    ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> nam[i];
        int c = nam[i][0] - 'a';
        sz[c]++;
    }
    for (int i = 0; i < n; i++) {
        cin >> nick[i];
        int c = nick[i][0] - 'a';
        cnt[i] = ++cur[c];
    }
    for (int i = n - 1; i >= 0; i--) {
        int c = nick[i][0] - 'a';
        d[c][cnt[i]] = i;
    }
    for (int i = 0; i < n; i++) {
        int c = nam[i][0] - 'a';
        for (int j = 0; j < 26; j++) mx[j][i] = max(mx[j][i - 1], d[j][sz[j]]);
    }
    int res = 0;
    for (int i = 0; i < n; i++) {
        int c = nam[i][0] - 'a';
        lcp[i] = SZ(nam[i]);
        if (nick[mx[c][i]][0] == nam[i][0]) {
            for (int j = 0; j < SZ(nick[mx[c][i]]); j++) {
                if (j == SZ(nam[i])) break;
                lcp[i] = min(lcp[i], j + 1);
                if (nick[mx[c][i]][j] != nam[i][j]) break;
            }
        }
        res += lcp[i];
    }
    cout << res << "\n";
    for (int i = 0; i < n; i++) {
        int pos = mx[nam[i][0] - 'a'][i];
        ans[i] = pos + 1;
    }
    for (int i = 0; i < n; i++) cout << i + 1 << " " << ans[i] << "\n";
    return 0;
}
