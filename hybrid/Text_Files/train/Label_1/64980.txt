def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def eye_contact(t, data):
    results = []
    for i in range(t):
        n = data[i][0]
        points = data[i][1]
        angles = {}
        for x, y, u, v in points:
            dx, dy = u - x, v - y
            g = gcd(abs(dx), abs(dy))
            direction = (dx // g, dy // g)
            if direction in angles:
                angles[direction] += 1
            else:
                angles[direction] = 1
        count = 0
        for direction, freq in angles.items():
            opposite_direction = (-direction[0], -direction[1])
            if opposite_direction in angles:
                count += freq * angles[opposite_direction]
        results.append(count // 2)
    return results

# Example of how to use the function with input
t = int(input().strip())
data = []
for _ in range(t):
    n = int(input().strip())
    points = [list(map(int, input().strip().split())) for _ in range(n)]
    data.append((n, points))

results = eye_contact(t, data)
for result in results:
    print(result)
