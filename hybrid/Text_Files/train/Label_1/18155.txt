#undef _GLIBCXX_DEBUG
#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, a[100000], c = 1e9;
    cin >> n;
    for (long long i = 0; i < n; i++)
        cin >> a[i];
    vector<long long> ans;
    for (long long i = 1; i < n - 1; i++) {
        if (a[i] < a[i - 1] && a[i] < a[i + 1]) {
            ans.push_back(a[i]);
            c = min(c, a[i]);
        }
    }
    cout << ans.size() << endl;
    for (long long i = 0; i < n; i++)
        if (a[i] == c)
            cout << a[i] << ' ';
    cout << endl;
}
