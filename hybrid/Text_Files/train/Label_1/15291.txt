#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,a,b;
    cin>>n>>a>>b;
    string s;
    cin>>s;
    bool flag=false;
    int student=0;
    for(int i=0;i<n;i++){
        if(s[i]=='.'){
            if(!flag){
                flag=true;
                student++;
            }
            if(student==a+b){
                break;
            }
            if(i+1<n && s[i+1]=='.'){
                if(student>=min(a,b)){
                    student+=2;
                }
            }
        }
        else{
            flag=false;
        }
    }
    cout<<student;
}
