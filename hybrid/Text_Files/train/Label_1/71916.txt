import sys
input = sys.stdin.readline

n,m = list(map(int,input().split()))
relation={}
for i in range(n+1): relation[i] = set()
for i in range(m):
    a,b = list(map(int,input().split()))
    relation[a].add(b)
    relation[b].add(a)

check = [False for i in range(n+1)]
ans = 0
pairs = []
for i in range(1,n+1):
    if check[i]: continue
   
    check[i] = True
    que = [i]
    while len(que)>0:
        head = que.pop(0)
        for j in relation[head]:
            if i==j: continue
            if not check[j]:
                check[j] = True
                que.append(j)
            if check[j]==True:
                ans+=1
                pairs.append((i,j))
                relation[i].remove(j)
                relation[j].remove(i)
print(ans)
for i in pairs:
    print(*i)
