import sys
n,m,k=map(int,input().split())
a=[]
for _ in range(n):
    a.append(set())
for _ in range(n):
    l=list(map(int,input().split()))
    for i in range(m):
        if l[i]:
            if l[i] in a[_] or a[_]==set():
                a[_].add(l[i])
print(*map(lambda P:len(P) if len(P) else 0,a))
