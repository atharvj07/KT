#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    const ll mod = 1e9 + 7;
    ll n;
    cin >> n;
    vector<int> cur(n + 1), want(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> cur[i] >> want[i];
    }
    ll ans = 1;
    for (int i = 1; i <= n; i++) {
        if (cur[i] != want[i]) ans = (ans * 2) % mod;
    }
    cout << ans << endl;
    return 0;
}
