int main() {
    ll n, m;
    cin >> n >> m;
    vector<ll> t(n);
    rep(i, n) cin >> t[i];
    ll time = 0;
    rep(i, n) {
        time += t[i];
        ll need = max(0LL, time - m);
        ll ans = need;
        for (ll j = i - 1; j >= 0; j--) {
            if (ans <= 0) { break; }
            if (t[j] <= t[i]) { break; }
            need--;
            ans--;
        }
        cout << ans << " ";
        time -= t[i];
    }
    cout << endl;
    return 0;
}
