import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] dp = new int[n][61];
        dp[0][a[0]] = 1;
        for (int i = 1; i < n; i++) {
            for (int j = 1; j <= 60; j++) {
                dp[i][j] = dp[i - 1][j];
            }
            for (int j = 1; j <= 60; j++) {
                if (a[i] % j == 0) {
                    dp[i][j] = (dp[i][j] + dp[i - 1][a[i]]) % 1000000007;
                }
            }
        }
        int ans = 0;
        for (int j = 1; j <= 60; j++) {
            ans = Math.max(ans, dp[n - 1][j]);
        }
        System.out.println(ans);
    }
}
