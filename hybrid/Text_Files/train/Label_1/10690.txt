#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<string> v(n);
  map<char, int> mp;
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    for (int j = 0; j < v[i].size(); j++) mp[v[i][j]]++;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    set<char> s(v[i].begin(), v[i].end());
    int cnt = s.size();
    if (cnt <= 2) { ans += v[i].size(); }
  }
  cout << ans << endl;
}
