from collections import deque, defaultdict
n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
da, db = defaultdict(deque), defaultdict(deque)
for x in a:
    if x!=0: da[x].append(0)
for x in b:
    if x!=0: db[x].append(1)
ans = 0
for x in range(1, n+1):
    while da[x] and db[x]:
        if da[x][0]==db[x][0]:
            ans += 1
            da[x].popleft()
            db[x].popleft()
        elif da[x][0]==0:
            ans += 1
            da[x].popleft()
        else:
            db[x].popleft()
    ans += len(da[x])+len(db[x])
print(ans)
