import sys
tc=int(input())
for i in range(tc):
    n=int(input())
    C,T=map(float,input().split())
    prb=[list(map(int,input().split())) for i in range(n)]
    prb.sort(key=lambda x: (-x[1]/x[0]))
    #print(prb)
    sc=0.0
    T=T-10.0
    s=1.0
    while(s>0.1):
        a,b=0,0
        for p in prb:
            if(a+p[0]/s+b<=T):
                sc+=p[1]
                a+=p[0]/s
                b+=10.0
            else:
                s*=0.9
                break
        T-=b
    print(int(sc))
