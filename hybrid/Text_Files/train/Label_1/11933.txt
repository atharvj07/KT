#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    sort(a.begin(), a.end());
    int ans = abs(a[n / 2] - a[0]);
    // abs(a[n / 2] - a[0]) || abs(a[n / 2] - a[n - 1])
    int tmp = abs(a[n / 2] - a[n - 1]);
    if (tmp < ans) ans = tmp;
    cout << ans << endl;
    return 0;
}
