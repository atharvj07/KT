#include<bits/stdc++.h>
using namespace std;
typedef long long LL;
#define MAX 1000000

vector<int> prime;
bool vis[MAX+5];

void sieve(){
    for(int i=2;i*i<=MAX;++i){
        if(vis[i]) continue;
        for(int j=i;j<=MAX;j+=i){
            vis[j] = 1;
        }
    }
    for(int i=2;i<=MAX;++i){
        if(vis[i]) continue;
        prime.push_back(i);
    }
}
int main(){
    int n,x;
    scanf("%d",&n);

    set<int>st;
    for(int i=1;i<=n;++i){
        scanf("%d",&x);
        st.insert(x);
    }

    sieve();
    map<int,int>mp;
    int ans[4005];
    for(auto i:st){
        for(int j=0;j<prime.size();++j){
            if(prime[j]*prime[j]>i) break;
            if(i%prime[j]==0){
                while(i%prime[j]==0){
                   mp[prime[j]]++;
                   i/=prime[j];
                }
            }
        }
        if(i>1) mp[i]++;
    }
    
    int j=0;
    for(auto i:mp){
        for(int k=0;k<i.second;++k){
            ans[j++] = i.first;
        }
    }
    
    int len = j;

    if(len<=0){
        puts("-1");
        return 0;
    }

    printf("%d\n",len);
    for(int i=0;i<len;++i){
        if(i) putchar(' ');
        printf("%d",ans[i]);
    }
    puts("");

    return 0;
}
