from itertools import permutations

n, k = map(int, input().split())
r = [list(input().split()) for _ in range(k)]
p = permutations(range(1, n+1), n*2)
ans = 0
for x in p:
    x = [i for j in x for i in [j]*2]
    for a, b, c in r:
        a, b = int(a), int(b) - 1
        if (b < a) != (x[a] > x[b]) and not (x[a] == x[b] and c == '='):
            break
    else:
        ans += 1
print(ans)
