#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  int T; cin >> T;
  while (T--) {
    int n; cin >> n;
    vector<ll> a(n);
    for (auto &x : a) cin >> x;
    map<ll, ll> cnt;
    for (auto &x : a) ++cnt[x];
    ll mxm = *max_element(a.begin(), a.end());
    ll ans = 0;
    for (auto &x : cnt) {
      if (x.first == mxm) {
        ans += x.second * (x.second - 1) / 2;
      } else {
        ans += x.second * cnt[mxm];
      }
    }
    cout << ans << endl;
  }
  return 0;
}
