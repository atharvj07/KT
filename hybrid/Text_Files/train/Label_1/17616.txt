#include <vector>
#include <algorithm>
#include <iostream>

int main() {
  int n, k;
  std::cin >> n >> k;

  std::vector<int> v1(1, 1);
  for (int i = 0; i < n - 1; ++i) {
    std::vector<int> v2 = v1;
    std::reverse(v2.begin(), v2.end());
    int x = 1;
    while (std::find(v1.begin(), v1.end(), x) != v1.end()) ++x;
    v1.insert(v1.begin() + v1.size() / 2, x);
    v1.insert(v1.end(), v2.begin(), v2.end());
  }
  std::cout << v1[k - 1] << std::endl;

  return 0;
}
