#include <bits/stdc++.h>
#define ll long long int
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.size();
        string ans = s;
        for (int i = 0, j = n - 1; i < j; i++, j--) {
            if (s[i] != s[j]) {
                if (s[i] == '1' || s[j] == '1') {
                    if (s[i] == '1')
                        ans[i] = '0';
                    else
                        ans[j] = '0';
                } else {
                    if (s[i] == '0')
                        ans[i] = '1';
                    else
                        ans[j] = '1';
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
