t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    freq = {}
    for x in a:
        if x in freq:
            freq[x] += 1
        else:
            freq[x] = 1
    x, max_freq = max(freq.items(), key=lambda item: item[1])
    segments = 1
    for i in range(1, n):
        if a[i] == x and a[i-1] == x:
            continue
        if a[i] == x or a[i-1] == x:
            continue
        segments += 1
    print(segments - 1)
