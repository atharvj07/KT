n, x = map(int, input().split())
queue = [x]
for _ in range(n):
    op, d = input().split()
    d = int(d)
    if op == '+':
        queue.append(queue[-1] + d)
    else:
        if queue[-1] < d:
            queue.append(queue[-1])
        else:
            queue.append(queue[-1] - d)
print(queue[-1], queue.count(queue[-1]))
