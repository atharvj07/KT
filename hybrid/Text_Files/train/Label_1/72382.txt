import heapq
n, q = map(int, input().split())
tasks = []
for i in range(q):
    t, k, d = map(int, input().split())
    tasks.append((t, k, d))
tasks.sort()
heap = []
for t, k, d in tasks:
    while heap and heap[0][0] <= t:
        _, k = heappop(heap)
        k += 1
        if k > 0:
            heappush(heap, (t + d, k))
    if len(heap) >= k:
        ans = sum(map(lambda x: x[1], heapq.nsmallest(k, heap)))
        print(ans)
    else:
        print(-1)
