#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

typedef long long ll;
#define int ll

const int N = 303;
int n, a[N], b[N], c[N], xr[N], yr[N];
vector<int> g[N];

int get(int x, int y) {
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (a[i] * x + b[i] * y + c[i] == 0)
            ans |= 1 << i;
    }
    return ans;
}

int ans;
int dx[4] = {-1, 1, 0, 0};
int dy[4] = {0, 0, -1, 1};
void dfs(int v, int x, int y) {
    xr[v] = x;
    yr[v] = y;
    if (ans != -1)
        return;
    if (x == 0 && y == 0) {
        ans = v;
        return;
    }
    for (int i = 0; i < 4; i++) {
        int nx = x + dx[i];
        int ny = y + dy[i];
        int H = get(nx, ny);
        if ((H ^ v ^ xr[H] ^ yr[H]) == 0) {
            dfs(H, nx, ny);
            if (ans != -1)
                return;
        }
    }
}

int32_t main() {
    int x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2 >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i] >> c[i];
    }
    int st = get(x1, y1);
    int fin = get(x2, y2);
    ans = -1;
    dfs(st, x1, y1);
    if (ans == -1) {
        cout << -1;
        return 0;
    }
    int now = ans;
    vector<int> path;
    while (now != st) {
        path.push_back(now);
        int nx = xr[now] - x1;
        int ny = yr[now] - y1;
        int nv = get(nx, ny);
        now = nv;
    }
    path.push_back(st);
    now = fin;
    while (now != path[0]) {
        path.push_back(now);
        int nx = xr[now] - x2;
        int ny = yr[now] - y2;
        int nv = get(nx, ny);
        now = nv;
    }
    cout << (int)path.size() - 1 << '\n';
    for (int i = (int)path.size() - 2; i >= 0; i--) {
        int v = path[i];
        int nx = x1 + xr[v] - xr[path[i + 1]];
        int ny = y1 + yr[v] - yr[path[i + 1]];
        int H = get(nx, ny);
        if (a[H] != 0) {
            cout << "1 ";
        } else {
            cout << "2 ";
        }
    }
    return 0;
}
