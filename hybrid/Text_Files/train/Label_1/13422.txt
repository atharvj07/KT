#include <iostream>
#include <unordered_map>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  long long l;
  unordered_map<long long, long long> mp;
  for (long long i = 1; i <= n; ++i) {
    cin >> l;
    mp[l] = i;
  }
  long long pos = 0;
  for (long long i = 1; i <= n; ++i) {
    pos += mp[i];
    if (pos == k) {
      cout << i << "\n";
      break;
    }
  }
}
