def power(x, y):
    ret = 1
    while y > 0:
        if y % 2 == 0:
            x = x * x
            x = x % 1000000007
            y = y / 2
        else:
            ret = ret * x
            ret = ret % 1000000007
            y = y - 1
    return ret

n, k, d = map(int, input().split())
print((power(k, n) - k * power(k - 1, n - d) * power(k - d, d - 1)) % 1000000007)
