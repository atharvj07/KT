import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h1 = sc.nextInt();
        int a1 = sc.nextInt();
        int c1 = sc.nextInt();
        int h2 = sc.nextInt();
        int a2 = sc.nextInt();

        int n = (h2 + a2 - 1) / a1;
        int h = h1;
        StringBuilder strategy = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (h - a2 <= 0) {
                strategy.append("HEAL\n");
                h = h1;
            } else {
                strategy.append("STRIKE\n");
                h -= a2;
            }
            h2 -= a1;
        }
        System.out.println(n);
        System.out.print(strategy);
    }
}
