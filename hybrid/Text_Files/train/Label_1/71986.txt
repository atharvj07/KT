n, d = map(int, input().split())
s = input()
if s[n - 1] == '0':
    print(-1)
else:
    dp = [n] * (n + 1)
    dp[n - 1] = 0
    for i in range(n - 2, -1, -1):
        if s[i] == '1':
            for j in range(i + 1, min(i + d + 1, n)):
                if s[j] == '1':
                    dp[i] = min(dp[i], dp[j] + 1)
    print(-1 if dp[0] == n else dp[0])
