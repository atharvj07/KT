import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] cnt = new int[m + 1];
        for (int i = 0; i < n; i++) {
            cnt[a[i]]++;
        }
        long ans = 0;
        for (int i = 1; i <= m; i++) {
            ans += (long) cnt[i] * (cnt[i] - 1);
        }
        for (int i = 1; i <= m; i++) {
            for (int j = i + 1; j <= m; j++) {
                ans += (long) cnt[i] * cnt[j];
            }
        }
        System.out.println(ans);
    }
}
