from collections import defaultdict

n, d = map(int, input().split())
adj = defaultdict(list)
for _ in range(d):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

for i in range(1, d + 1):
    q = [(1, [1])]
    vis = set()
    ans = 0
    while q:
        lev, a = q.pop(0)
        if lev > ans:
            ans = lev
            vis = set(a)
        for ne in adj[a[0]]:
            if ne not in vis:
                q.append((lev + 1, a + [ne]))
    print(ans)
