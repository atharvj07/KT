import java.util.*;
import java.io.*;

public class Main {
    static class Team implements Comparable<Team> {
        String name;
        int rating;

        public Team(String name, int rating) {
            this.name = name;
            this.rating = rating;
        }

        @Override
        public int compareTo(Team other) {
            return other.rating - this.rating;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(reader.readLine());
        String[] randomNums = reader.readLine().split(" ");
        int x = Integer.parseInt(randomNums[0]);
        int a = Integer.parseInt(randomNums[1]);
        int b = Integer.parseInt(randomNums[2]);
        int c = Integer.parseInt(randomNums[3]);

        List<Team> teams = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] teamInfo = reader.readLine().split(" ");
            teams.add(new Team(teamInfo[0], Integer.parseInt(teamInfo[1])));
        }

        Collections.sort(teams);
        int m = n / 4;
        int[][] groups = new int[m][4];

        int teamIndex = 0;
        for (int i = 0; i < m - 1; i++) {
            for (int j = 0; j < 4; j++) {
                x = (a * x + b) % c;
                groups[i][j] = teamIndex + x % (teams.size() - 4 * i);
            }
            teamIndex += 4;
        }

        for (int j = 0; j < 4; j++) {
            groups[m - 1][j] = teamIndex++;
        }

        char groupLetter = 'A';
        for (int i = 0; i < m; i++) {
            System.out.println("Group " + groupLetter + ":");
            for (int j = 0; j < 4; j++) {
                System.out.println(teams.get(groups[i][j]).name);
            }
            groupLetter++;
        }
    }
}
