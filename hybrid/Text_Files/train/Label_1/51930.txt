import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int X = sc.nextInt();
        int Y = sc.nextInt();

        // Check if the total number of legs is even
        if (Y % 2 == 1) {
            System.out.println("No");
            return;
        }

        // Calculate the maximum number of cranes
        int maxCranes = Y / 4;

        // Iterate through the possible number of cranes
        for (int cranes = 0; cranes <= maxCranes; cranes++) {
            int turtles = X - cranes;

            // Check if the number of turtles is valid
            if (turtles >= 0 && turtles * 4 + cranes * 2 == Y) {
                System.out.println("Yes");
                return;
            }
        }

        // No valid combination found
        System.out.println("No");
    }
}
