from bisect import bisect_right

n = int(input())
l = [int(x) for x in input().split()]
c = [int(x) for x in input().split()]

l.sort()
c.sort()

ans = 1e9
jumps = 0
for i in range(n-1, -1, -1):
    idx = bisect_right(l, jumps, lo=i)
    if idx == i:
        break
    ans = min(ans, c[i] + c[idx])
    jumps += l[idx]

if jumps == 0:
    print(-1)
else:
    print(ans)
