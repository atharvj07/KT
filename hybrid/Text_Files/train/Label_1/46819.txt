import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] a = new int[n + 1];
        for (int i = 0; i < n; i++) {
            a[i + 1] = a[i] + (s.charAt(i) == '0' ? 1 : -1);
        }
        int min = n;
        for (int i = 0; i <= n; i++) {
            min = Math.min(min, Math.abs(a[i]));
        }
        System.out.println(min);
    }
}
