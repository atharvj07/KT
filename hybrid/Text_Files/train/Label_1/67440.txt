def snapreg():
    n = int(input())
    pos = []
    for i in range(n): pos.append(list(map(int, input().split())))
    used = [False]*n
    i = j = k = l = r = u = d = fi = fj = fk = fl = fr = fu = fd = 0
    ans = []
    for _ in range(n//2):
        for p in range(n):
            if used[p]: continue
            if i < pos[p][0]: i = pos[p][0]; fi = p
            if j < pos[p][1]: j = pos[p][1]; fj = p
            if k < pos[p][2]: k = pos[p][2]; fk = p
            if l > pos[p][0]: l = pos[p][0]; fl = p
            if r > pos[p][1]: r = pos[p][1]; fr = p
            if u > pos[p][2]: u = pos[p][2]; fu = p
            if d < pos[p][2]: d = pos[p][2]; fd = p
        if fi == fd or fj == fr or fk == fu: print(*sorted([fi+1, fd+1]), *sorted([fj+1, fr+1]), *sorted([fk+1, fu+1])); used[fi] = used[fd] = used[fj] = used[fr] = used[fk] = used[fu] = True; i = j = k = l = r = u = d = 0
        else: print(*sorted([fl+1, fi+1]), *sorted([fr+1, fj+1]), *sorted([fu+1, fk+1])); used[fl] = used[fi] = used[fr] = used[fj] = used[fu] = used[fk] = True; l = r = u = d = 0

snapreg()
