#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1e9 + 7;
int main()
{
    ll t;
    cin >> t;
    while (t--)
    {
        ll n;
        cin >> n;
        ll sum = n;
        while (n >= 10)
        {
            sum = sum + n / 10;
            n = n / 10 + n % 10;
        }
        cout << sum << endl;
    }
    return 0;
}
