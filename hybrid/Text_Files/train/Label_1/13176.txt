#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, mini, maxi, diff, ctr;
    cin >> n;
    int arr[n], ans[n];
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    sort(arr, arr + n);

    mini = arr[0];
    maxi = arr[n - 1];
    diff = maxi - mini;
    if (diff > 2) {
        cout << 0 << endl;
        return 0;
    }

    for (int i = 0; i < n; i++) {
        if (arr[i] == mini && ctr == 0) {
            ans[i] = mini + 1;
            ctr++;
            continue;
        } else if (arr[i] == maxi && ctr == 0) {
            ans[i] = maxi - 1;
            ctr++;
            continue;
        } else {
            ans[i] = arr[i];
        }
    }

    cout << ctr << endl;
    for (int i = 0; i < n; i++) {
        cout << ans[i] << " ";
    }
    return 0;
}
