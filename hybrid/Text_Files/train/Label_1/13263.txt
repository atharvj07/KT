#include <iostream>

int main() {
    int n, k, a[100000], ans = 0;
    std::cin >> n >> k;
    for (int i = 0; i < n; i++) { std::cin >> a[i]; }
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            int cnt = 0;
            for (int l = 0; l < 15; l++) {
                if (((a[i] >> l) & 1) != ((a[j] >> l) & 1)) { cnt++; }
            }
            if (cnt == k) { ans++; }
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
