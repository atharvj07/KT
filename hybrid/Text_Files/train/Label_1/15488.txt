#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int dx[4] = {0, 1, 0, -1};
const int dy[4] = {1, 0, -1, 0};

int main() {
  int n;
  cin >> n;
  vector<tuple<int, int, ll>> points(n);
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    points[i] = make_tuple(x, y, x - y);
  }
  sort(points.begin(), points.end());
  vector<ll> w(n);
  for (int i = 0; i < n; i++) cin >> w[i];
  for (int i = 0; i < n; i++) {
    auto [x, y, s] = points[i];
    if (s != w[i]) {
      cout << "NO" << endl;
      return 0;
    }
  }
  vector<tuple<int, int, int>> ans;
  for (int i = 0; i < n; i++) {
    int x, y;
    tie(x, y, ignore) = points[i];
    for (int k = 0; k < 4; k++) {
      int nx = x + dx[k];
      int ny = y + dy[k];
      auto it = lower_bound(points.begin(), points.end(), make_tuple(nx, ny, 0LL));
      if (it != points.end() && get<0>(*it) == nx && get<1>(*it) == ny) ans.push_back(make_tuple(x, y, w[i]));
    }
  }
  sort(ans.begin(), ans.end());
  cout << "YES" << endl;
  for (auto p : ans) {
    int x, y, w;
    tie(x, y, w) = p;
    cout << x << " " << y << endl;
  }
  return 0;
}
