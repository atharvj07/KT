#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        if (s == string(s.rbegin(), s.rend())) cout << -1 << endl;
        else {
            sort(s.begin(), s.end());
            cout << s << endl;
        }
    }
    return 0;
}
