#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll INF = 1ll << 60;
struct Edge {
    int to;
    ll w;
    Edge() {}
    Edge(int to, ll w) : to(to), w(w) {}
};
vector<vector<Edge>> g;
const int maxn = 2e5 + 10;
ll dis[maxn];
int vis[maxn];
void dijkstra(int s) {
    for (int i = 1; i <= maxn; i++) dis[i] = INF;
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll, int>>> pq;
    pq.push({0, s});
    dis[s] = 0;
    while (!pq.empty()) {
        int k = pq.top().second;
        ll w = pq.top().first;
        pq.pop();
        if (vis[k] || dis[k] < w) continue;
        vis[k] = 1;
        for (Edge e : g[k]) {
            if (!vis[e.to] && dis[e.to] > dis[k] + e.w) {
                dis[e.to] = dis[k] + e.w;
                pq.push({dis[e.to], e.to});
            }
        }
    }
}
int32_t main() {
    int n, m, k;
    cin >> n >> m >> k;
    g.resize(n + 1);
    while (m--) {
        int a, b, w;
        cin >> a >> b >> w;
        g[a].push_back(Edge(b, w));
        g[b].push_back(Edge(a, w));
    }
    dijkstra(1);
    vector<ll> ans;
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (i != j) ans.push_back(dis[i] + dis[j]);
        }
    }
    sort(ans.begin(), ans.end());
    cout << ans[k - 1] << "\n";
}
