#include <cstdio>
#include <iostream>
#include <algorithm>

using namespace std;
typedef long long ll;

int main() {
    ll n, m, a, b;
    cin >> n >> m >> a >> b;
    if (n % m == 0) {
        cout << 0 << endl;
        return 0;
    }
    ll d = n / m + 1 - n % m;
    ll cost = d * min(a, b);
    cout << cost << endl;
    return 0;
}
