#include<bits/stdc++.h>
#include<ext/pb_ds/assoc_container.hpp>
#include<ext/pb_ds/tree_policy.hpp>
using namespace __gnu_pbds;
using namespace std;
typedef tree<int, null_type, less<int>, rb_tree_tag, tree_order_statistics_node_update> indexed_set;
using ll = long long;
const ll P = 998244353;
ll pw(ll a, ll p) {
	if (p == 0) return 1;
	if (p == 1) return a;
	ll ans = pw(a * a % P, p >> 1);
	if (p & 1) ans = ans * a % P;
	return ans;
}
int main() {
	ll w, h;
	cin >> w >> h;
	ll ans = pw(2, w + h) - 2 * pw(2, (min(w, h) - 1)) + pw(2, max(w, h));
	ans = (ans % P + P) % P;
	cout << ans << '\n';
	return 0;
}
