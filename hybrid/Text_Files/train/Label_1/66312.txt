import sys

n, m = map(int, input().split())
house = [list(input()) for _ in range(n)]
T = [[[-1] * 3 for _ in range(m)] for _ in range(n)]

def solve(floor, cell, dir):
  if floor == 0:
    return 0
  res = T[floor][cell][dir]
  if res != -1:
    return res

  if dir == 0:
    if house[floor - 1][cell] == '.':
      res = solve(floor - 1, cell, dir)
      if res != -1:
        res += 1
    if cell + 1 < m and house[floor][cell + 1] == '.':
      res = solve(floor, cell + 1, dir)
      if res != -1:
        res += 1
    if cell + 1 < m and house[floor][cell + 1] == '+':
      res = solve(floor, cell + 1, 1 - dir)
      if res != -1:
        res += 1
  else:
    if house[floor - 1][cell] == '.':
      res = solve(floor - 1, cell, dir)
      if res != -1:
        res += 1
    if cell - 1 >= 0 and house[floor][cell - 1] == '.':
      res = solve(floor, cell - 1, dir)
      if res != -1:
        res += 1
    if cell - 1 >= 0 and house[floor][cell - 1] == '+':
      res = solve(floor, cell - 1, 1 - dir)
      if res != -1:
        res += 1

  T[floor][cell][dir] = res
  return res

res = solve(n - 1, 0, 0)
if res == -1:
  print("Never")
else:
  print(res)
