import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    for i in range(1, n - 1):
        if a[i] > a[i - 1] and a[i] > a[i + 1]:
            print(i + 1)
            break
    else:
        if a[0] > a[1]:
            print(1)
        elif a[-1] > a[-2]:
            print(n)
        else:
            print(-1)
