#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, u, v;
    cin >> n >> u >> v;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<vector<int>> dp(n + 1, vector<int>(1000002, 1e9));
    dp[1][a[0]] = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= 1000001; j++) {
        if (dp[i][j] == 1e9) continue;
        // Move up and down
        if (i + 1 <= n) dp[i + 1][j] = min(dp[i + 1][j], dp[i][j] + u);
        if (i - 1 >= 1) dp[i - 1][j] = min(dp[i - 1][j], dp[i][j] + u);
        // Move left and right
        if (j - 1 >= 1) dp[i][j - 1] = min(dp[i][j - 1], dp[i][j] + v);
        if (j + 1 <= 1000001) dp[i][j + 1] = min(dp[i][j + 1], dp[i][j] + v);
        // Move to obstacle
        if (a[i - 1] == j) dp[i][j] = min(dp[i][j], dp[i - 1][j]);
      }
    }
    cout << dp[n][1000001] << endl;
  }
  return 0;
}
