#include <cstdio>
#include <vector>

using namespace std;

int main() {
  int n, s;
  scanf("%d %d", &n, &s);
  vector<int> anc(n + 1);
  vector<int> deg(n + 1);
  for (int i = 1; i <= n; i++) {
    int p;
    scanf("%d", &p);
    anc[i] = p;
    deg[p]++;
  }
  int ans = 0;
  for (int i = 2; i <= n; i++) {
    if (i != s && deg[i] == 0) ans++;
  }
  printf("%d\n", ans);
}
