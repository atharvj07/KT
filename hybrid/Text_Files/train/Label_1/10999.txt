#include <cstdio>
#include <algorithm>
using namespace std;
int main() {
    int n; scanf("%d", &n);
    int a[n];
    for (int i = 0; i < n; i++) { scanf("%d", a + i); }
    sort(a, a + n);
    int mi = a[0], ans = -1;
    for (int i = 1; i < n; i++) {
        if (a[i] > mi + 1) { ans = mi + 1; break; }
        mi = a[i];
    }
    printf("%d\n", ans);
}
