import heapq

n, x, y = map(int, input().split())
tree = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    tree[a].append(b)
    tree[b].append(a)
    
q = [(0,1)]
dist = [10**18]*(n+1)
dist[1] = 0

while q:
    d, node = heapq.heappop(q)
    if d > dist[node]:
        continue
    for u in tree[node]:
        if d + (x if node in tree[u] else y) < dist[u]:
            dist[u] = d + (x if node in tree[u] else y)
            heapq.heappush(q, (d + (x if node in tree[u] else y), u))

print(max(dist[1:]))
