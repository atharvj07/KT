import sys
 
 
def dot(x, y): return sum(i * j for i, j in zip(x, y))
def dist(x, y): return ((x[0] - y[0]) ** 2 + (x[1] - y[1]) ** 2) ** .5
def cross(x, y): return x[0] * y[1] - x[1] * y[0]
def ccw(x, y, z): return (y[0] - x[0]) * (z[1] - y[1]) - (y[1] - x[1]) * (z[0] - y[0])
def line_seg_intersect(a, b, c, d):
    if (ccw(a, c, d) * ccw(b, c, d) < 0 and
            ccw(a, b, c) * ccw(a, b, d) < 0):
        return 1
    if a == c or b == c or a == d or b == d: return 1
    return 0
def inside(a, x):
    i = len(a)
    j = 0
    while j < i:
        if ccw(a[i - 1], a[j], x) >= 0: return False
        i, j = j, j + 1
    return True
def solve():
    n, m = map(int, input().split())
    a = []
    for i in range(n):
        x, y = map(float, input().split())
        a.append([x, y])
    ans = []
    for i in range(m):
        x1, y1, x2, y2 = map(float, input().split())
        b = []
        b.append([x1, y1])
        b.append([x2, y2])
        ans.append(0.)
        for j in range(n):
            if line_seg_intersect(a[j], a[(j + 1) % n], b[0], b[1]):
                ai, aj = a[j], a[(j + 1) % n]
                bi, bj = b[0], b[1]
                area = dot(ai, aj) + dot(aj, bi) + dot(bi, bj) - dot(bj, ai)
                an = dist(ai, aj)
                bn = dist(bi, bj)
                area /= (an + bn) * 2.
                if area > 0.:
                    if inside(a, bi) and inside(a, bj):
                        if dist(aj, bi) < dist(aj, bj):
                            ans[-1] = dist(bi, aj)
                        else:
                            ans[-1] = dist(aj, bj)
                    else: area = 0.
                if area < 0.:
                    if inside(a, ai) and inside(a, aj):
                        if dist(bj, ai) < dist(bj, aj):
                            ans[-1] = dist(bj, ai)
                        else:
                            ans[-1] = dist(bj, aj)
                    else: area = 0.
                if area != 0.: ans[-1] += area
    for x in ans: print(x)
solve()
