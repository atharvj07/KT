from collections import defaultdict

n, m = map(int, input().split())
roads = [tuple(map(int, input().split())) for _ in range(m)]
G = defaultdict(set)
for a, b in roads:
    G[a].add(b)
    G[b].add(a)
ans = 0
for i in range(1, n + 1):
    for j in G[i]:
        if i != j:
            for k in G[j]:
                if k != j and k != i and k in G[i]:
                    ans += 1
print(ans)
