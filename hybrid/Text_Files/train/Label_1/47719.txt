import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), b = sc.nextInt();
        int[] cost = new int[n], cnt = new int[m + 1], mon = new int[n];
        List<Integer>[] lst = new List[n];
        for (int i = 0; i < n; i++) lst[i] = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            cost[i] = sc.nextInt();
            mon[i] = sc.nextInt();
            int num = sc.nextInt();
            for (int j = 0; j < num; j++) {
                int index = sc.nextInt();
                lst[i].add(--index);
                cnt[index]++;
            }
        }
        int ans = -1, tot = 0;
        for (int i = 0; i < m; i++) if (cnt[i] == n) tot++;
        for (int i = 0; i < n; i++) if (mon[i] <= tot) {
            int cur = 0;
            for (int j = 0; j < n; j++) if (j != i) cur += cost[j];
            cur += b * mon[i];
            if (ans == -1 || cur < ans) ans = cur;
        }
        System.out.println(ans);
    }
}
