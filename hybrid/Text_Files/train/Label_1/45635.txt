import java.util.*;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        BigInteger a = new BigInteger(sc.next()), b = new BigInteger(sc.next()), c = new BigInteger(sc.next());
        BigInteger k = new BigInteger(sc.next()), ta = a, tb = b, tc = c;
        while (k.compareTo(BigInteger.ZERO) == 1) {
            if (k.remainder(BigInteger.valueOf(3)).equals(BigInteger.ZERO)) {
                ta = ta.add(tb).add(tc);
                tb = ta.add(tb).add(tc);
                tc = ta.add(tb).add(tc);
            }
            else if (k.remainder(BigInteger.valueOf(3)).equals(BigInteger.ONE)) {
                ta = ta.add(tb).add(tc);
                tb = ta.add(tc).add(tc);
                tc = tc.add(tc).add(tc);
            }
            else if (k.remainder(BigInteger.valueOf(3)).equals(BigInteger.valueOf(2))) {
                ta = ta.add(tb).add(tb);
                tb = ta.add(ta).add(tb);
                tc = ta.add(ta).add(ta);
            }
            k = k.divide(BigInteger.valueOf(3));
        }
        BigInteger abst = ta.subtract(tb).abs();
        if (abst.compareTo(BigInteger.valueOf(1000000000000000000L)) <= 0) {
            System.out.println(ta.subtract(tb));
        }
        else {
            System.out.println("Unfair");
        }
    }
}
