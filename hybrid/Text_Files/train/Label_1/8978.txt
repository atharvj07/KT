#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int &x : a) { cin >> x; }
        int c0 = 0, c1 = 0, c2 = 0;
        for (int x : a) {
            if (x % 3 == 0) {
                c0++;
            } else if (x % 3 == 1) {
                c1++;
            } else {
                c2++;
            }
        }
        int balance = n / 3;
        int ans = 0;
        if (c0 > balance) {
            ans += c0 - balance;
            c2 += c0 - balance;
            c0 -= c0 - balance;
        }
        if (c1 > balance) {
            ans += c1 - balance;
            c0 += c1 - balance;
            c1 -= c1 - balance;
        }
        if (c2 > balance) {
            ans += c2 - balance;
            c1 += c2 - balance;
            c2 -= c2 - balance;
        }
        cout << ans << "\n";
    }
}
