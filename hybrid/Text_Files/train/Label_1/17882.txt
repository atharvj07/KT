#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> ii;

int main() {
    int n;
    cin >> n;
    vector<int> a(n), sum;
    for (int &x : a)
        cin >> x;
    sum.emplace_back(a[0]);
    for (int i = 1; i < n; i++)
        sum.emplace_back(sum.back() + a[i]);
    map<int, ii> m;  // sum -> (start, end)
    m[0] = {0, 0};
    int ans = 0, last = 0;
    for (int i = 0; i < n; i++) {
        if (m.count(sum[i])) {
            int j = m[sum[i]].second + 1;
            if (i - j > ans) {
                ans = i - j;
                last = j;
            }
        }
        m[sum[i]] = {i, i};
    }
    vector<ii> res;
    int j = last + 1, s = sum[last];
    while (s != 0) {
        j -= (s == sum[j - 1]);
        int i = j;
        while (i > 0 && sum[i - 1] == s - a[i])
            i--;
        res.emplace_back(i, j);
        s = sum[i - 1];
        j = i - 1;
    }
    cout << res.size() << '\n';
    for (const ii &x : res)
        cout << x.first + 1 << ' ' << x.second + 1 << '\n';
}
