#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, x, mn, ans, cnt, ord;
    map<int, int> mp;
    vector<pair<int, int>> vx;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> x;
        mp[x]++;
    }
    for (auto it = mp.begin(); it != mp.end(); it++) {
        vx.push_back(make_pair(it->first, it->second));
    }
    sort(vx.begin(), vx.end());
    ans = 0;
    for (int i = 0; i < vx.size(); i++) {
        if (!vx[i].second) continue;
        cnt = 0;
        for (int j = i; j < vx.size(); j++) {
            if (vx[j].first % vx[i].first == 0) cnt += vx[j].second;
        }
        if (cnt) {
            ans += (cnt / vx[i].second);
            for (int j = i; j < vx.size(); j++) {
                if (vx[j].first % vx[i].first == 0) vx[j].second -= (cnt / vx[i].second) * vx[i].second;
                if (vx[j].second == 0) ord = j;
            }
            i = ord;
        }
    }
    cout << ans << "\n";
    return 0;
}
