import sys

def solve(s):
    x = y = 0
    mx = my = 0
    for c in s:
        if c == 'W': y -= 1
        elif c == 'S': y += 1
        elif c == 'A': x -= 1
        else: x += 1
        mx = max(mx, abs(x))
        my = max(my, abs(y))
    return (mx + 1) * (my + 1)

for line in sys.stdin:
    print(solve(line.strip()))
