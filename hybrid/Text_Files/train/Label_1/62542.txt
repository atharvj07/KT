def find_index_for_arithmetic_progression(n, b):
    if n <= 2:
        return 1
    
    candidates = [(b[1] - b[0], 0), (b[2] - b[1], 1), (b[2] - b[0], 2)]
    
    for diff, excluded_index in candidates:
        index_to_remove = -1
        last_value = None
        for i, value in enumerate(b):
            if i == excluded_index:
                continue
            if last_value is None:
                last_value = value
            else:
                if value - last_value != diff:
                    if index_to_remove == -1:
                        index_to_remove = i
                    else:
                        break
                last_value = value
        else:
            if index_to_remove == -1:
                return 1
            return index_to_remove + 1
    return -1

n = int(input().strip())
b = list(map(int, input().strip().split()))

print(find_index_for_arithmetic_progression(n, b))
