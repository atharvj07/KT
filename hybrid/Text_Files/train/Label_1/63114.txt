def is_adjacent(a, b):
    return abs(ord(a) - ord(b)) in [1, 13]

s = input()
g1 = [s[:13], s[13:]]
g2 = [''] * 2
for i in range(13):
    if g1[0][i] == g1[1][i]:
        g2[0] += s[i]
        g2[1] += s[26 - 1 - i]
    else:
        g2[1] += s[i]
        g2[0] += s[26 - 1 - i]
if all(len(set(g)) == 13 for g in [g1[0], g1[1], g2[0], g2[1]]):
    for g in [g1, g2]:
        for i in range(13):
            for j in range(2):
                if g[j][i] != '=':
                    g[j] = g[j][:i] + g[j][i].lower() + g[j][i + 1:]
        if all(j == g[0][i] for i in range(13)
                for j in [g[0][i], g[1][i]]) and all(
                    is_adjacent(g[0][i], g[0][i + 1])
                    for i in range(12)) and all(
                        is_adjacent(g[0][i], g[1][i])
                        for i in range(13)):
            print(''.join(g[0]))
            print(''.join(g[1]))
            exit()
print('Impossible')
