def find_max_score(n, m, scores, roads, s):
    from collections import defaultdict, deque
    
    # Creating adjacency list
    graph = defaultdict(list)
    for u, v in roads:
        graph[u].append(v)
        graph[v].append(u)
    
    visited = [False] * (n + 1)
    visited[s] = True
    
    queue = deque([(s, 0)])
    total_score = scores[s-1]
    
    # BFS to visit cities without repeating the immediate previous city
    while queue:
        current_city, prev_city = queue.popleft()
        for next_city in graph[current_city]:
            if not visited[next_city] and next_city != prev_city:
                visited[next_city] = True
                total_score += scores[next_city-1]
                queue.append((next_city, current_city))
    
    return total_score

n, m = map(int, input().split())
scores = list(map(int, input().split()))
roads = [tuple(map(int, input().split())) for _ in range(m)]
s = int(input())

print(find_max_score(n, m, scores, roads, s))
