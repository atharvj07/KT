#include <bits/stdc++.h>

using namespace std;
#define p 64
typedef long long ll;
#define umap unordered_map

vector<int> adj[p];

int main() {
    int n;
    cin >> n;
    vector<ll> x(n);
    vector<ll> ans;
    umap<ll, int> pos;
    for (int i = 0; i < n; i++) {
        cin >> x[i];
        pos[x[i]] = i;
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < p; j++) {
            if (x[i] & (1ll << j)) {
                adj[j].push_back(i);
            }
        }
    }
    for (int i = 0; i < p; i++) {
        vector<bool> vis(n, false);
        for (int vertex : adj[i]) {
            if (vis[vertex] == false) {
                vis[vertex] = true;
                ans.push_back(x[vertex]);
            }
        }
    }

    cout << ans.size() << endl;
    for (int j : ans) {
        cout << j << " ";
    }
    cout << endl;
    return 0;
}
