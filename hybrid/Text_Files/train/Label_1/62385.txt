t = int(input())
for _ in range(t):
    n = int(input())
    skills = list(map(int, input().split()))
    skill_count = {}
    for skill in skills:
        if skill in skill_count:
            skill_count[skill] += 1
        else:
            skill_count[skill] = 1
    max_count = max(skill_count.values())
    unique_skills = len(skill_count)
    print(min(max_count, unique_skills - 1) if max_count != unique_skills else max_count - 1)
