import java.util.Scanner;

public class DeepLearningCountries {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int m = scanner.nextInt();

        int[] entryCosts = new int[n];
        int[] layerCosts = new int[n];
        int[] exitCosts = new int[n];

        for (int i = 0; i < n; i++) {
            entryCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            layerCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            exitCosts[i] = scanner.nextInt();
        }

        long[][] dp = new long[l + 1][n];
        dp[0][0] = 1; // Base case: starting at the entry point

        for (int layer = 1; layer <= l; layer++) {
            for (int city = 0; city < n; city++) {
                for (int prevCity = 0; prevCity < n; prevCity++) {
                    int cost = (entryCosts[city] + layerCosts[prevCity] * (layer - 1) + exitCosts[prevCity]) % m;
                    if (cost == 0) {
                        dp[layer][city] = (dp[layer][city] + dp[layer - 1][prevCity]) % MOD;
                    }
                }
            }
        }

        long paths = 0;
        for (int city = 0; city < n; city++) {
            paths = (paths + dp[l][city]) % MOD;
        }

        System.out.println(paths);
    }
}
