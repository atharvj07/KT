#include<bits/stdc++.h>
using namespace std;

int main() {
  int n; cin >> n;
  string s; cin >> s;
  int a = 0, b = 0;
  for (int i = 0; i < n; ++i) {
    if (i % 2 == 0) a += (s[i] == 'a');
    else b += (s[i] == 'b');
  }
  int ans = abs(a - b) / 2;
  cout << ans << endl;
  for (int i = 0; i < n; ++i) {
    if (i % 2 == 0 && a > b) {
      a -= (s[i] == 'a');
      s[i] = 'b';
    }
    if (i % 2 == 1 && b > a) {
      b -= (s[i] == 'b');
      s[i] = 'a';
    }
  }
  cout << s << endl;
}
