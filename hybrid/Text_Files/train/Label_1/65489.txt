n, k = map(int, input().split())
p = []
e = []

for i in range(n):
    x, y = map(int, input().split())
    p.append(x)
    e.append(y)
    
p.sort()
e.sort(reverse = True)

t = 0
f = False
for i in range(n):
    if p[i] + 1 >= k or t + e[i] >= k - 1:
        f = True
        break
    t += e[i]
    p[i] += 1
    
if f:
    print(t)
else:
    print(-1)
