import math

def cross_product(o, a, b):
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

def solve_polygon(vertices):
    n = len(vertices)
    vertices.append(vertices[0])
    vertices.append(vertices[1])
    min_distance = float('inf')
    
    for i in range(1, n + 1):
        d1 = (vertices[i][0] - vertices[i - 1][0], vertices[i][1] - vertices[i - 1][1])
        d2 = (vertices[i + 1][0] - vertices[i][0], vertices[i + 1][1] - vertices[i][1])
        
        cross_prod = cross_product(vertices[i - 1], vertices[i], vertices[i + 1])
        
        area = abs(cross_prod)
        side_length = math.sqrt(d1[0] ** 2 + d1[1] ** 2)
        height = area / side_length
        min_distance = min(min_distance, height / 2)
    
    return min_distance

n = int(input().strip())
vertices = [tuple(map(int, input().strip().split())) for _ in range(n)]

print(f"{solve_polygon(vertices):.10f}")
