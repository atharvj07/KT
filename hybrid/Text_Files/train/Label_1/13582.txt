#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long t1,t2,t3,a[100001],ans=0,n,i,tot;
    cin>>t1>>t2>>t3>>n;
    tot=t1;
    for(i=0;i<n;i++)
    {
        cin>>a[i];
    }
    bool f=0;
    for(i=0;i<n;i++)
    {
        if(tot<=a[i])
        {
            tot=a[i]+t3;
            ans=a[i];
            f=1;
        }
        else
        {
            f=0;
            break;
        }
    }
    if(f==0)
    {
        if(t2>=tot)
        {
            ans=tot;
        }
        else
        {
            ans=t2;
        }
    }
    cout<<ans<<endl;
    return 0;
}
