import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static long[] t;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        t = new long[(n + 1) << 2];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        Arrays.sort(a);
        long ans = -1;
        long cur = 0;
        for (int i = n - 1; i >= 0; i -= 2) {
            if (n - i - 1 >= k) {
                ans = cur + ((n - i - 1 - k) >> 1);
                break;
            }
            cur += a[i];
            update(1, 0, n, i, i);
            update(1, 0, n, n - (i >> 1) - 1, n - (i >> 1) - 1);
        }
        System.out.println(ans);
    }

    static void update(int node, int l, int r, int x, int val) {
        if (x < l || r < x)
            return;
        if (l == r && l == x) {
            t[node] = val;
            return;
        }
        int mid = (l + r) >> 1;
        update(node << 1, l, mid, x, val);
        update(node << 1 | 1, mid + 1, r, x, val);
        t[node] = t[node << 1] + t[node << 1 | 1];
    }
}
