#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n), b(m), inv(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    inv[a[i] - 1] = i;
  }
  for (int i = 0; i < m; i++) cin >> b[i];
  long long ans = 0;
  int p = 0;
  for (int x : b) {
    int q = inv[x - 1];
    ans += (p > q) * ((p + k - q) / k) + (q - p);
    p = q;
  }
  cout << ans;
}
