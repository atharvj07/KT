import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] k = new int[3];
        for (int i = 0; i < 3; i++) {
            k[i] = scanner.nextInt();
        }
        int n = Arrays.stream(k).sum();
        int[] a1 = new int[k[0]];
        for (int i = 0; i < k[0]; i++) {
            a1[i] = scanner.nextInt();
        }
        int[] a2 = new int[k[1]];
        for (int i = 0; i < k[1]; i++) {
            a2[i] = scanner.nextInt();
        }
        int[] a3 = new int[k[2]];
        for (int i = 0; i < k[2]; i++) {
            a3[i] = scanner.nextInt();
        }
        int[] s = new int[n + 1];
        for (int i = 0; i < k[0]; i++) {
            s[a1[i]] = 1;
        }
        for (int i = 0; i < k[1]; i++) {
            s[a2[i]] = 2;
        }
        for (int i = 0; i < k[2]; i++) {
            s[a3[i]] = 3;
        }
        int last2 = 0;
        int last3 = 0;
        for (int i = 1; i <= n; i++) {
            if (s[i] == 2) {
                last2 = i;
            } else if (s[i] == 3) {
                last3 = i;
            }
        }
        int ans = 0;
        int cur = 0;
        for (int i = 1; i <= n; i++) {
            if (s[i] == 1) {
                cur++;
            } else if (s[i] == 3) {
                if (cur > 0) {
                    cur--;
                    ans++;
                } else if (i < last2) {
                    ans++;
                }
            } else if (s[i] == 2) {
                ans += cur;
                cur = 0;
                if (i < last3) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
