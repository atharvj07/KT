import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        try {
            BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
            String[] tokens = br.readLine().split(" ");
            int n = Integer.parseInt(tokens[0]);
            int q = Integer.parseInt(tokens[1]);
            tokens = br.readLine().split(" ");
            int[] coins = new int[n];
            for (int i = 0; i < n; i++) {
                coins[i] = Integer.parseInt(tokens[i]);
            }
            Arrays.sort(coins);
            for (int i = 0; i < q; i++) {
                tokens = br.readLine().split(" ");
                int value = Integer.parseInt(tokens[0]);
                System.out.println(solve(value, coins, n));
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public static int solve(int value, int[] coins, int n) {
        int count = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (value - coins[i] >= 0) {
                value -= coins[i];
                count++;
            }
        }
        if (value == 0) {
            return count;
        } else {
            return -1;
        }
    }
}
