#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n), b(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        for (int i = 0; i < n; i++) cin >> b[i];
        vector<pair<int, int>> v;
        for (int i = 0; i < n; i++) v.push_back({a[i], b[i]});
        sort(v.begin(), v.end());
        int ans = -1, sum = 0;
        for (int i = 0; i < n; i++) {
            sum += v[i].second;
            if (sum >= m) {
                ans = v[i].first;
                break;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
