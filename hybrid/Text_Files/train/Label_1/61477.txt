from collections import deque

def bfs(s, graph, n):
    distance = [-1] * (n + 1)
    queue = deque([s])
    distance[s] = 0
    while queue:
        v = queue.popleft()
        for w in graph[v]:
            if distance[w] == -1:
                distance[w] = distance[v] + 1
                queue.append(w)
    return distance

n, m = map(int, input().split())
graph = [[] for _ in range(n + 1)]

for _ in range(m):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

s1, t1, l1 = map(int, input().split())
s2, t2, l2 = map(int, input().split())

dist = [[0] * (n + 1) for _ in range(n + 1)]
for i in range(1, n + 1):
    dist[i] = bfs(i, graph, n)

if dist[s1][t1] > l1 or dist[s2][t2] > l2:
    print(-1)
else:
    ans = m - dist[s1][t1] - dist[s2][t2]
    for u in range(1, n + 1):
        for v in range(1, n + 1):
            for x in [s1], [t1]:
                for y in [s2], [t2]:
                    if dist[x[0]][u] + dist[u][v] + dist[v][y[0]] <= l1 and dist[y[1]][u] + dist[u][v] + dist[v][x[1]] <= l2:
                        ans = max(ans, m - (dist[s1][t1] + dist[s2][t2] + dist[u][v]))
    print(ans)
