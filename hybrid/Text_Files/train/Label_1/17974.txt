#include<bits/stdc++.h>
#define ll unsigned long long
using namespace std;
ll adj[100005]={}, sz[100005]={}, vis[100005]={}, pa[100005]={};
vector<ll>adjlist[100005];

void dfs1(ll x, ll p){
    pa[x]=p;
    vis[x]=1;
    for(auto u:adjlist[x]){
        if(vis[u]==0)
        dfs1(u,x);
    }
}

void dfs2(ll x,ll pre){
    sz[x]=1;
    for(auto u:adjlist[x]){
        if(u!=pre){
            dfs2(u,x);
            sz[x]+=sz[u];
        }
    }
}

int main(){
    ll n,m,ans=0;
    cin>>n>>m;
    while(m--){
        ll u,v;
        cin>>u>>v;
        adjlist[u].push_back(v);
        adjlist[v].push_back(u);
        adj[u]|=1<<v;
        adj[v]|=1<<u;
    }
    for(ll i=1;i<=n;i++){
        if(vis[i]==0){
            dfs1(i,i);
        }
    }
    for(ll i=1;i<=n;i++){
        if(vis[i]==0){
            ans+=((sz[i]*(sz[i]-1))/2);
        }
    }
    for(ll i=1;i<=n;i++){
        for(auto j:adjlist[i]){
            if(j!=pa[i]){
                ans+=sz[i]*sz[j];
            }
        }
    }
    cout<<ans<<endl;
    return 0;
}
