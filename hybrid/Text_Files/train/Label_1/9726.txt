#include <bits/stdc++.h>
using namespace std;
const int R = 2, C = 13;
typedef pair<int, int> pi;
char g[R][C];
bool vis[27], adj[27][27];
bool found = false;
bool isValid(int r, int c) {
    return (r >= 0 && r < R && c >= 0 && c < C);
}
void dfs(int r, int c, string s, int i) {
    if (i == 27) {
        found = true;
        return;
    }
    for (int nr = -1; nr <= 1; nr++) {
        for (int nc = -1; nc <= 1; nc++) {
            int nr2 = nr + r, nc2 = nc + c;
            if (isValid(nr2, nc2) && adj[s[i] - 'A'][g[nr2][nc2] - 'A'] && !vis[g[nr2][nc2] - 'A']) {
                vis[g[nr2][nc2] - 'A'] = true;
                dfs(nr2, nc2, s, i + 1);
                vis[g[nr2][nc2] - 'A'] = false;
            }
        }
    }
}
int main() {
    string s;
    cin >> s;
    for (int i = 0; i < 26; i++) { adj[i][i] = true; }
    for (int i = 0; i < 27; i++) {
        g[0][i % C] = s[i];
        adj[s[i] - 'A'][s[(i + 1) % 27] - 'A'] = true;
        adj[s[(i + 1) % 27] - 'A'][s[i] - 'A'] = true;
    }
    dfs(0, 0, s, 0);
    if (!found) {
        cout << "Impossible" << endl;
        return 0;
    }
    for (int i = 0; i < R; i++) {
        for (int j = 0; j < C; j++) { cout << g[i][j]; }
        cout << endl;
    }
    return 0;
}
