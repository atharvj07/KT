#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e3 + 10;
int dp[maxn][maxn];
int a[maxn], l[maxn], r[maxn];
int main() {
  int n, m;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  cin >> m;
  for (int i = 1; i <= m; i++) { cin >> l[i] >> r[i]; }
  memset(dp, -1, sizeof(dp));
  dp[0][0] = 0;
  for (int i = 1; i <= m; i++) {
    for (int j = 0; j <= n; j++) {
      if (dp[i - 1][j] >= 0) dp[i][j] = dp[i - 1][j];
      if (dp[i][j] < 0) continue;
      dp[i][j + 1] = max(dp[i][j + 1], dp[i][j] + a[j + 1]);
      if (dp[i][j] + a[j + 1] <= r[i]) dp[i][j + 1] = max(dp[i][j + 1], l[i]);
    }
  }
  int ans = -1;
  for (int i = 0; i <= n; i++) ans = max(ans, dp[m][i]);
  cout << ans << endl;
  return 0;
}
