for _ in range(int(input())):
    n=int(input())
    if n&1:print('NO')
    else:
        a=[]
        for i in range(2,int(n**0.5)+1):
            while n%i==0:
                a.append(i)
                n//=i
        if n>1:a.append(n)
        if len(a)>=3:print('YES',*a)
        else:print('NO')
