import java.util.Scanner;

public class CupsAndKey {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        long[] a = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextLong();
        }

        long n = 1;
        for (long ai : a) {
            n = (n * ai) % MOD;
        }

        long p = 1;
        for (int i = 0; i < k; i++) {
            p = (p * (n - a[i])) % MOD;
        }

        long q = MOD - 1;
        long gcd = gcd(p, q);
        p /= gcd;
        q /= gcd;

        System.out.println(p % MOD + "/" + q % MOD);
    }

    private static long gcd(long a, long b) {
        while (b != 0) {
            long temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
