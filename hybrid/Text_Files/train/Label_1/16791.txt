#include <bits/stdc++.h>
using namespace std;
int n, a[300001], b[300001], dp[300001], p[300001];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> b[i];
  dp[0] = -1e9;
  for (int i = 1; i <= n; i++) {
    dp[i] = -1e9;
    for (int j = max(0, i - b[i]); j <= i - 1; j++) {
      if (j + a[j] >= i) {
        if (dp[i] < dp[j] + (j != 0)) dp[i] = dp[j] + (j != 0), p[i] = j;
      }
    }
  }
  if (dp[n] == -1e9)
    cout << "-1";
  else {
    cout << dp[n] << "\n";
    vector<int> ans;
    for (int i = n; i != 0; i = p[i]) ans.push_back(i);
    for (int i = ans.size() - 1; i >= 0; i--) cout << ans[i] << " ";
  }
}
