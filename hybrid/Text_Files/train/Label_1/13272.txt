#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &x : a) { cin >> x; }
    sort(a.rbegin(), a.rend());
    for (int i = 1; i < n; i++) { a[i] -= a[i - 1]; }
    for (auto &x : a) { cout << x << " "; }
    cout << endl;
    return 0;
}
