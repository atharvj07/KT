import java.util.*;
import java.lang.*;
import java.math.*;
  
public class Main {
  
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int b = scanner.nextInt();
  
        if (a < b) {
            System.out.println(-(2 * b - 2 * a));
            for (int i = 0; i < b; i++) {
                System.out.print("o");
            }
            for (int i = 0; i < a; i++) {
                System.out.print("x");
            }
        } else {
            System.out.println(2 * a - 2 * b);
            for (int i = 0; i < a; i++) {
                System.out.print("x");
            }
            for (int i = 0; i < b; i++) {
                System.out.print("o");
            }
        }
        scanner.close();
    }
  
}
