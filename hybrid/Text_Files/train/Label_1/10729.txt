#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
template <typename T>
long long power(T a, long long b) {
	if (b == 0) return 1;
	if (b == 1) return a;
	if (b % 2 == 1) { return (power(a, b - 1) * a) % mod; }
	long long val = power(a, b / 2) % mod;
	return (val * val) % mod;
}
template <typename T>
long long mul(T a, T b) {
	return ((a % mod) * (b % mod)) % mod;
}
long long gcd(long long a, long long b) {
	if (a == 0) return b;
	return gcd(b % a, a);
}
long long get_max_val(vector<long long>& vec) {
	long long ans = 0;
	int n = vec.size();
	for (int i = 0; i < n; i++) {
		for (int j = i + 1; j < n; j++) {
			for (int k = j + 1; k < n; k++) {
				if (gcd(vec[i], vec[j]) == 1 && gcd(vec[i], vec[k]) == 1 && gcd(vec[j], vec[k]) == 1) {
					ans = max(ans, vec[i] + vec[j] + vec[k]);
				}
			}
		}
	}
	for (int i = 0; i < n; i++) {
		for (int j = i + 1; j < n; j++) {
			if (gcd(vec[i], vec[j]) == 1) {
				ans = max(ans, vec[i] + vec[j]);
			}
		}
	}
	for (int i = 0; i < n; i++) {
		ans = max(ans, vec[i]);
	}
	return ans;
}
int main() {
	long long q;
	cin >> q;
	while (q--) {
		long long n;
		cin >> n;
		vector<long long>vec(n);
		for (int i = 0; i < n; i++) {
			cin >> vec[i];
		}
		long long ans = get_max_val(vec);
		cout << ans << '\n';
	}
}
