#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 5;
int n, a[MAXN], b[MAXN], ans[MAXN], lst[MAXN], g[MAXN], fa[MAXN];

int find(int x) { return fa[x] == x ? x : fa[x] = find(fa[x]); }
void uni(int x, int y) { fa[find(x)] = find(y); }

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> b[i];
  for (int i = 1; i <= n; i++) fa[i] = i;
  for (int i = 1; i <= n; i++) {
    if (a[i] && b[i]) lst[a[i]] = b[i], uni(a[i], b[i]);
  }
  for (int i = 1; i <= n; i++) {
    if (!a[i]) {
      g[find(b[i])] = 1;
      if (lst[b[i]]) uni(find(lst[b[i]]), find(b[i]));
    }
  }
  int cnt = 0;
  for (int i = 1; i <= n; i++) cnt += g[i];
  cout << n - cnt << endl;
  return 0;
}
