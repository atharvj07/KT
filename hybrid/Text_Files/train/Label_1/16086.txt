#include<iostream>
#include<map>
#include<set>
#include<vector>
#include<string>
#include<algorithm>
using namespace std;
int main(){
  string s,sub;
  cin>>s>>sub;
  int len=0;
  for(int i=0;i<sub.length();i++){
    if(sub[i]=='0')  continue;
    len++;
  }
  int n=s.length()-1;
  if(sub[0]=='0'){
    s.erase(s.begin(),s.begin()+n-len);
  }
  else{
    s.erase(s.begin(),s.begin()+n-1-len);
  }
  string temp=s;
  sort(temp.begin(),temp.end());
  for(int i=0;i<sub.length();i++){
    int pos=0;
    for(int j=0;j<s.length();j++){
      if(s[j]==sub[i]){
        if(sub[i]=='0'){
          if(temp[pos]==s[j]){
            temp.erase(temp.begin()+pos);
            pos--;
          }
        }
        else{
          if(temp[pos]==s[j]){
            temp.erase(temp.begin()+pos);
          }
        }
        pos++;
      }
    }
  }
  while(temp.length()){
    if(temp[0]=='0'&&temp.length()>1){
      temp.erase(temp.begin());
    }
    else  break;
  }
  s+=temp;
  cout<<s<<endl;
  return 0;
}
