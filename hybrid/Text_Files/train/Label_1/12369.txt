#include <bits/stdc++.h>
#define int long long
const int M = 998244353;
int k, n, ans[1001], f[1001];
struct node {
  int v, p;
  bool operator<(const node &o) const { return v > o.v; }
} a[1001];
signed main() {
  std::cin >> n >> k;
  for (int i = 1; i <= n; ++i) std::cin >> a[i].v, a[i].p = i;
  std::sort(a + 1, a + 1 + n);
  for (int i = 1; i <= n - k + 1; ++i) {
    f[i] = f[i - 1] + a[i].v;
    ans[i] = (ans[i - 1] + f[i] - f[i - k + 1] - (a[i].v - a[i - k + 1].v) * (k - 1)) % M;
  }
  std::cout << ans[n - k + 1] << std::endl;
}
