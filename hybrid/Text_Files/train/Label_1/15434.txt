#include <bits/stdc++.h>

using namespace std;

int main() {
    long long ax, ay, bx, by, cx, cy;
    cin >> ax >> ay >> bx >> by >> cx >> cy;
    if ((cx - ax) * (ax - bx) + (cy - ay) * (ay - by) == 0 &&
        (cx - bx) * (bx - ax) + (cy - by) * (by - ay) == 0)
        cout << "Yes" << endl;
    else
        cout << "No" << endl;
    return 0;
}
