import sys

def main():
    t = int(input())
    for i in range(t):
        nr, ng, nb = map(int, input().split())
        a = list(map(int, input().split()))
        b = list(map(int, input().split()))
        c = list(map(int, input().split()))
        a.sort()
        b.sort()
        c.sort()
        print(min(pow(a[(nr - 1) // 2] - b[(ng - 1) // 2], 2) + pow(b[(ng - 1) // 2] - c[(nb - 1) // 2], 2) + pow(c[(nb - 1) // 2] - a[(nr - 1) // 2], 2), pow(a[nr // 2] - b[(ng - 1) // 2], 2) + pow(b[(ng - 1) // 2] - c[(nb - 1) // 2], 2) + pow(c[(nb - 1) // 2] - a[nr // 2], 2), pow(a[nr // 2] - b[ng // 2], 2) + pow(b[ng // 2] - c[(nb - 1) // 2], 2) + pow(c[(nb - 1) // 2] - a[nr // 2], 2), pow(a[nr // 2] - b[ng // 2], 2) + pow(b[ng // 2] - c[nb // 2], 2) + pow(c[nb // 2] - a[nr // 2], 2)))
main()
