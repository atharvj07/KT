import java.util.ArrayList;
import java.util.Collections;
import java.util.Scanner;
import java.util.stream.Collectors;
import java.util.stream.IntStream;
import java.util.stream.Stream;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        ArrayList<Card> cards = new ArrayList<Card>();
        for (int i = 0; i < n; i++) {
            String suit = sc.next();
            int rank = sc.nextInt();
            cards.add(new Card(suit, rank));
        }
        ArrayList<Card> missingCards = new ArrayList<Card>();
        for (String suit : new String[] { "S", "H", "C", "D" }) {
            for (int rank : IntStream.range(1, 14).toArray()) {
                if (cards.stream().noneMatch(card -> card.suit.equals(suit) && card.rank == rank)) {
                    missingCards.add(new Card(suit, rank));
                }
            }
        }
        Collections.sort(missingCards, (card1, card2) -> {
            int suitOrder = "SHCD".indexOf(card1.suit) - "SHCD".indexOf(card2.suit);
            return suitOrder == 0 ? card1.rank - card2.rank : suitOrder;
        });
        System.out.println(missingCards.stream().map(Card::toString).collect(Collectors.joining("\n")));
    }
}

class Card {
    String suit;
    int rank;

    Card(String suit, int rank) {
        this.suit = suit;
        this.rank = rank;
    }

    @Override
    public String toString() {
        return suit + " " + rank;
    }
}
