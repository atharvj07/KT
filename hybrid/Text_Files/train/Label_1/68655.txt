def min_moves_to_divisible_by_25(n):
    n = str(n)
    min_moves = float('inf')
    for d in ['00', '25', '50', '75']:
        moves, j = 0, len(n)
        for i in range(1, -1, -1):
            j = n.rfind(d[i], 0, j)
            if j == -1:
                moves = float('inf')
                break
            moves += len(n) - j - 2 + i
        min_moves = min(min_moves, moves)
    return min_moves if min_moves != float('inf') else -1

n = int(input())
print(min_moves_to_divisible_by_25(n))
