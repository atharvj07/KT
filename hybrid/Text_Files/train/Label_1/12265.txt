#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,m,k,i,j,u,v,l,a[100001],d[100001],c,dist[100001],x=1000000000,st[100001],vis[100001],par[100001];
    cin>>n>>m>>k;
    for(i=1;i<=n;i++)
    {
        dist[i]=1000000000;
        d[i]=1000000000;
        par[i]=i;
        st[i]=0;
    }
    for(i=1;i<=m;i++)
    {
        cin>>u>>v>>l;
        d[u]=min(d[u],l);
        d[v]=min(d[v],l);
    }
    for(i=1;i<=k;i++)
    {
        cin>>a[i];
        st[a[i]]=1;
    }
    for(i=1;i<=n;i++)
    {
        if(st[i])
            continue;
        dist[i]=0;
        c=0;
        j=i;
        while(par[j]!=j)
        {
            j=par[j];
            if(st[j])
            {
                c=1;
                break;
            }
        }
        if(c)
        {
            while(i!=j)
            {
                dist[i]=min(dist[i],d[i]);
                par[i]=j;
                i=par[i];
            }
        }
    }
    for(i=1;i<=n;i++)
    {
        if(st[i])
            continue;
        x=min(dist[i],x);
    }
    if(x!=1000000000)
        cout<<x<<endl;
    else
        cout<<-1<<endl;
    return 0;
}
