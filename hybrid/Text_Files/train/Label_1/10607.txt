#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    for (int i = 0; i < n; i++) {
        int sum = 0; 
        int maxi = 0;
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            maxi = max(maxi, v[j].second);
            sum += v[j].first;
        }
        cout << sum * maxi << " ";
    }
    cout << endl;
    return 0;
}
