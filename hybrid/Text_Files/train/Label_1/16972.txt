#include <bits/stdc++.h>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        string s; cin >> s;
        int l = -1, r = -1;
        int cnta = 0, cntb = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == 'a') cnta++;
            else cntb++;
            if (cnta == cntb) {
                l = i + 1;
                break;
            }
        }
        if (l != -1) {
            for (int i = n - 1; i >= 0; i--) {
                if (s[i] == 'a') cnta--;
                else cntb--;
                if (cnta == cntb) {
                    r = i + 1;
                    break;
                }
            }
        }
        cout << l << ' ' << r << '\n';
    }
    return 0;
}
