MOD = 10**9 + 7

def comb(n, r, mod):
    if r < 0 or r > n:
        return 0
    numerator = 1
    denominator = 1
    r = min(r, n - r)
    for i in range(1, r + 1):
        numerator = (numerator * (n - i + 1)) % mod
        denominator = (denominator * i) % mod
    return (numerator * pow(denominator, mod - 2, mod)) % mod

def solve(n, m):
    return comb(n + m, m, MOD)

n, m = map(int, input().split())
print(solve(n, m))
