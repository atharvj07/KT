from heapq import *

n, m, a, b = map(int, input().split())
g = [[] for i in range(n+1)]
for i in range(m):
    u, v, w = map(int, input().split())
    g[u].append((v,w))
    g[v].append((u,w))

pq = [(0, 1, 0)]
dist = [9e9] * (n+1)
dist[1] = 0
while pq:
    d, u, p = heappop(pq)
    if d > dist[u]: continue
    for v, w in g[u]:
        if v == p: continue
        if d+w < dist[v]:
            heappush(pq, (d+w, v, u))
            dist[v] = d+w

print(*dist[1:])
