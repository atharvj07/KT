#include <iostream>
#include <vector>
using namespace std;
vector<int> range[1005];
int main() {
  int n, m, p[1005], res[1005], last = 1;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    range[a].push_back(b);
    p[b]++;
  }
  for (int i = 1; i <= n; i++) {
    p[i] += p[i - 1];
    int sum1 = p[i], sum2 = m - p[i];
    if (sum1 > sum2) {
      res[i] = 0;
      last = 0;
    } else {
      res[i] = 1;
      last = 1;
    }
    for (int j = 0; j < range[i].size(); j++) {
      int sum1 = p[range[i][j]] - p[i - 1], sum2 = m - p[range[i][j]] + last;
      if (sum1 > sum2)
        res[i] = 0;
      else
        res[i] = 1;
    }
  }
  for (int i = 1; i <= n; i++) cout << res[i];
  return 0;
}
