#include <iostream>

int main() {
  int h1, h2, a, b, k = 0;
  std::cin >> h1 >> h2 >> a >> b;
  int cur = h1, day = 0;
  while (true) {
    cur += a * 10;
    if (cur >= h2) {
      break;
    }
    cur -= b * 14;
    day++;
  }
  if (cur >= h2) {
    k = day;
  } else {
    k = -1;
  }
  std::cout << k << std::endl;
  return 0;
}
