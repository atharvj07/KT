import java.util.*;
public class Main {
    static boolean[][] f;
    static Queue<int[]> q;
    static int[] dx = new int[]{1, -1, 0, 0};
    static int[] dy = new int[]{0, 0, 1, -1};
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        f = new boolean[n+1][m+1];
        int k = in.nextInt(); q = new LinkedList<>();
        for (int i = 0; i < k; i++) {
            int x = in.nextInt(), y = in.nextInt();
            f[x][y] = true; q.add(new int[]{x, y});
        }
        int tx = -1, ty = -1;
        while (!q.isEmpty()) {
            int[] t = q.poll();
            tx = t[0]; ty = t[1];
            for (int d = 0; d < 4; d++) {
                int x = t[0] + dx[d], y = t[1] + dy[d];
                if (x >= 1 && x <= n && y >= 1 && y <= m && !f[x][y]) {
                    f[x][y] = true; q.add(new int[]{x, y});
                }
            }
        }
        System.out.println(tx + " " + ty);
    }
}
