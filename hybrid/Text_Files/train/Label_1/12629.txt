int main() {
    fastio();
    ll t;
    cin>>t;
    while(t--){
        ll n;
        cin>>n;
        ll ans=-1;
        if(n==1){
            ans=0;
        }
        else{
            queue<ll>q;
            q.push(n);
            while(!q.empty()){
                ll node=q.front();
                q.pop();
                if(node%3==0){
                    q.push(node/3);
                }
                if(node%2==0){
                    q.push(node/2);
                }
                if(node==1){
                    ans=1;
                    break;
                }
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
