import sys

n, d = list(map(int, sys.stdin.readline().split()))
a = list(map(int, sys.stdin.readline().split()))

mn = 0
ans = 0
for x in a:
    mn = max(mn + x, 0)
    if mn > d:
        ans = -1
        break
    elif x == 0:
        if mn == 0:
            ans += 1
        mn = max(mn + x, 0)

print(ans)
