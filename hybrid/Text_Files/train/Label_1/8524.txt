#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

struct Mouse {
  int cost;
  string type;
  bool operator<(const Mouse& other) const { return cost < other.cost; }
};

int main() {
  int a, b, c, m;
  cin >> a >> b >> c >> m;
  vector<Mouse> v(m);
  for (int i = 0; i < m; i++) {
    cin >> v[i].cost >> v[i].type;
  }
  sort(v.begin(), v.end());
  int cnt = min(a, m);
  int cost = 0;
  for (int i = 0; i < cnt; i++) {
    cost += v[i].cost;
    if (v[i].type == "USB") a--;
  }
  cnt += min(b, m - cnt);
  for (int i = cnt; i < m; i++) {
    cost += v[i].cost;
    if (v[i].type == "PS/2") b--;
  }
  cnt += min(c, m - cnt);
  for (int i = cnt; i < m; i++) {
    cost += v[i].cost;
  }
  cout << cnt << " " << cost << endl;
  return 0;
}
