#include<bits/stdc++.h>
using namespace std;
int main(){
    ios_base::sync_with_stdio(0);
    int a,b,c,d;
    cin>>a>>b>>c>>d;
    int p=min(a,d),q=min(b,c);
    if(a+d<b+c){
        cout<<"Impossible";return 0;
    }
    string s="";
    for(int i=1;i<=p;++i)s+='0';
    for(int i=1;i<=q;++i)s+='1';
    while(s.size()<1000000){
        s+='0'+(s.back()=='1');
    }
    cout<<s<<endl;
}
