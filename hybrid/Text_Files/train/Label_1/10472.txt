#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int a, b, n, m, d, sum = 0, mx = -1, ans = -1;
    vector<int> v1, v2;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a;
        v1.push_back(a);
    }
    cin >> m;
    for (int i = 0; i < m; i++) {
        cin >> b;
        v2.push_back(b);
    }
    sort(v1.begin(), v1.end());
    sort(v2.begin(), v2.end());
    for (int i = 0; i < v1.size(); i++) {
        d = v1[i];
        sum = 0;
        for (int j = 0; j < v1.size(); j++) {
            if (v1[j] <= d)
                sum += 2;
            else
                sum += 3;
        }
        for (int j = 0; j < v2.size(); j++) {
            if (v2[j] <= d)
                sum -= 2;
            else
                sum -= 3;
        }
        if (sum > mx) {
            mx = sum;
            ans = d;
        }
    }
    for (int i = 0; i < v2.size(); i++) {
        d = v2[i] - 1;
        sum = 0;
        for (int j = 0; j < v1.size(); j++) {
            if (v1[j] <= d)
                sum += 2;
            else
                sum += 3;
        }
        for (int j = 0; j < v2.size(); j++) {
            if (v2[j] <= d)
                sum -= 2;
            else
                sum -= 3;
        }
        if (sum > mx) {
            mx = sum;
            ans = d;
        }
    }
    int sc1 = 0, sc2 = 0;
    for (int i = 0; i < v1.size(); i++) {
        if (v1[i] <= ans)
            sc1 += 2;
        else
            sc1 += 3;
    }
    for (int i = 0; i < v2.size(); i++) {
        if (v2[i] <= ans)
            sc2 += 2;
        else
            sc2 += 3;
    }
    cout << sc1 << ":" << sc2 << endl;
    return 0;
}
