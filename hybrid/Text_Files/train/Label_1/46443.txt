import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        int[] dp = new int[n];
        int[] prev = new int[n];
        Arrays.fill(dp, 1);
        Arrays.fill(prev, -1);
        int max = 1;
        int last = -1;
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (arr[i] > arr[j] && dp[i] < dp[j] + 1) {
                    dp[i] = dp[j] + 1;
                    prev[i] = j;
                    if (dp[i] > max) {
                        max = dp[i];
                        last = i;
                    }
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        while (last != -1) {
            if (prev[last] == -1) {
                sb.append('L');
            } else {
                sb.append('R');
            }
            last = prev[last];
        }

        System.out.println(max);
        System.out.println(sb.reverse().toString());
    }
}
