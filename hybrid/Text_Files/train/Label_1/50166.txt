import java.util.Scanner;
import java.util.ArrayList;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        ArrayList<Integer> unsorted = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        
        for (int i = 1; i < n; i++) {
            if (a[i] < a[i - 1]) {
                unsorted.add(i - 1);
                unsorted.add(i);
            }
        }

        if (unsorted.size() == 0) {
            System.out.println(-1);
        } else {
            System.out.println(unsorted.get(0) + " " + unsorted.get(1));
        }
    }
}
