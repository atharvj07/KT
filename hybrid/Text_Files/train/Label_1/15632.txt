#include <iostream>
#include <cstdio>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, k;
        cin >> n >> m >> k;
        if (n * m / 2 == k || (n * m) % 4 == 0 && (n * m / 2 - k) % 2 == 0 && k <= n * m / 2)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
