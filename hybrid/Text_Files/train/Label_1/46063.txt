import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

public class Main {

    private static final int INF = Integer.MAX_VALUE;

    public static void main(String[] args) throws Exception {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        String[] parts = reader.readLine().split(" ");
        int n = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);
        int q = Integer.parseInt(parts[2]);

        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            parts = reader.readLine().split(" ");
            int a = Integer.parseInt(parts[0]);
            int b = Integer.parseInt(parts[1]);
            int c = Integer.parseInt(parts[2]);
            edges.add(new Edge(a, b, c, i + 1));
        }

        Map<Integer, List<Edge>> graph = new HashMap<>();
        for (Edge edge : edges) {
            if (!graph.containsKey(edge.from)) {
                graph.put(edge.from, new ArrayList<>());
            }
            graph.get(edge.from).add(edge);
        }

        int[] dist = new int[n + 1];
        Arrays.fill(dist, INF);
        dist[1] = 0;

        for (int i = 0; i < n - 1; i++) {
            boolean updated = false;
            for (Edge edge : edges) {
                if (dist[edge.from] + edge.weight < dist[edge.to]) {
                    dist[edge.to] = dist[edge.from] + edge.weight;
                    updated = true;
                }
            }
            if (!updated) {
                break;
            }
        }

        while (q-- > 0) {
            parts = reader.readLine().split(" ");
            if (parts[0].equals("1")) {
                int v = Integer.parseInt(parts[1]);
                if (dist[v] == INF) {
                    System.out.println(-1);
                } else {
                    System.out.println(dist[v]);
                }
            } else {
                int c = Integer.parseInt(parts[1]);
                for (int i = 2; i < 2 + c; i++) {
                    int l = Integer.parseInt(parts[i]);
                    for (Edge edge : edges) {
                        if (edge.index == l) {
                            edge.weight++;
                        }
                    }
                }

                Arrays.fill(dist, INF);
                dist[1] = 0;

                for (int i = 0; i < n - 1; i++) {
                    boolean updated = false;
                    for (Edge edge : edges) {
                        if (dist[edge.from] + edge.weight < dist[edge.to]) {
                            dist[edge.to] = dist[edge.from] + edge.weight;
                            updated = true;
                        }
                    }
                    if (!updated) {
                        break;
                    }
                }
            }
        }
    }

    private static class Edge {
        public int from;
        public int to;
        public int weight;
        public int index;

        public Edge(int from, int to, int weight, int index) {
            this.from = from;
            this.to = to;
            this.weight = weight;
            this.index = index;
        }
    }
}
