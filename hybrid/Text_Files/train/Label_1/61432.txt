n, A, c_f, c_m, m = map(int, input().split())
a = sorted(map(int, input().split()))

min_a = a[0]
max_a = A
for i in range(n):
  if a[i] == min_a:
    max_a = a[i] + 1
    break

max_force = 0
if m >= c_f * (n - 1):
  max_force += c_f * (n - 1)
  max_a -= 1
  m -= c_f * (n - 1)
else:
  k = m // c_f
  max_force += c_f * k
  max_a -= k
  m -= c_f * k

if m >= min_a:
  max_force += c_m
  max_a -= 1

print(max_force)
for i in range(n):
  print(min(A, max_a) if a[i] < max_a else a[i], end=' ')
