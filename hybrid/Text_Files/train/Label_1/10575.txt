#include <iostream>
#include <string>
#include <queue>
using namespace std;

int main() {
    int r, c;
    cin >> r >> c;
    vector<string> grid(r);
    for (int i = 0; i < r; i++) { cin >> grid[i]; }
    int dr[] = {-1, 0, 1, 0};
    int dc[] = {0, 1, 0, -1};
    queue<pair<int, int>> q;
    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            if (grid[i][j] == 'W') { q.push({i, j}); }
        }
    }
    while (!q.empty()) {
        int row = q.front().first;
        int col = q.front().second;
        q.pop();
        for (int k = 0; k < 4; k++) {
            int nr = row + dr[k];
            int nc = col + dc[k];
            if (nr >= 0 && nr < r && nc >= 0 && nc < c) {
                if (grid[nr][nc] == 'S') { cout << "No" << endl; return 0; }
                else if (grid[nr][nc] == '.') {
                    grid[nr][nc] = 'D';
                    q.push({nr, nc});
                }
            }
        }
    }
    cout << "Yes" << endl;
    for (int i = 0; i < r; i++) { cout << grid[i] << endl; }
    return 0;
}
