n, m = map(int, input().split())
g = [list(input()) for _ in range(n)]
vis = [[0] * m for _ in range(n)]

def valid(i, j):
    return 0 <= i < n and 0 <= j < m and not vis[i][j] and g[i][j] != '.'

def dfs(i, j, c):
    if c == 4: return 1
    vis[i][j] = 1
    ans = 0
    for x, y in ((i, j - 1), (i, j + 1), (i - 1, j), (i + 1, j)):
        if valid(x, y) and g[x][y] == "DI"[c]:
            ans += dfs(x, y, c + 1)
    vis[i][j] = 0
    return ans

ans = -1
for i in range(n):
    for j in range(m):
        if g[i][j] == 'D':
            ans = max(ans, dfs(i, j, 0))
print((ans and ans) or "Poor Dima!" or "Poor Inna!")
