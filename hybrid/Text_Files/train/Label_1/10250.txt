#include <iostream>
#include <vector>
using namespace std;

typedef long long ll;
#define pb push_back

const ll MI = 1e9+7;
const int N = 1e5+10;
int n, q, a[N];

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> q;
    for (int i = 1; i <= n; i++) cin >> a[i];
    vector<int> lc(n+1);
    for (int i = 1; i < n; i++) {
        lc[i+1] = max(lc[i], abs(a[i+1]-a[i]));
    }
    while (q--) {
        int l, r; cin >> l >> r;
        cout << lc[r-1]-lc[l-1] << '\n';
    }
    return 0;
}
