import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong(), b = sc.nextLong(), mod = sc.nextLong();
        if (b == 0 || mod % 10 == 0) System.out.println(1 + " 000000000");
        else {
            long t = mod - (a % mod);
            if (t == mod) t -= mod;
            String f;
            if (t < 10) f = "00000000" + t;
            else if (t < 100) f = "0000000" + t;
            else if (t < 1000) f = "000000" + t;
            else if (t < 10000) f = "00000" + t;
            else if (t < 100000) f = "0000" + t;
            else if (t < 1000000) f = "000" + t;
            else if (t < 10000000) f = "00" + t;
            else if (t < 100000000) f = "0" + t;
            else f = "" + t;
            System.out.println(1 + " " + f);
        }
    }
}
