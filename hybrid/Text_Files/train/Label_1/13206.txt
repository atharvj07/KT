#include <bits/stdc++.h>
using namespace std;
using ll=long long;

const int N=2e5+5;
int n,k;
ll a[N],b[N],cnt[25005],sum;
int main(){
	cin>>k;
	for(int i=0;i<k;i++){
		cin>>n>>sum;
		for(int j=0;j<n;j++){
			cin>>a[j];
			cnt[a[j]+10001]++;
		}
		for(int j=0;j<n;j++){
			cnt[a[j]+10001]--;
			if(cnt[a[j]+10001]==0){
				b[i]=a[j];
				break;
			}
		}
		for(int j=0;j<n;j++)
			cnt[a[j]+10001]=0;
	}
	bool f=0;
	for(int i=0;i<k;i++){
		for(int j=0;j<k;j++){
			if(i==j) continue;
			if(b[i]+b[j]==sum){
				cout<<"YES\n";
				cout<<i+1<<" "<<a[i]+10001<<"\n"<<j+1<<" "<<b[j]+10001<<"\n";
				f=1;
				break;
			}
		}
		if(f) break;
	}
	if(!f) cout<<"NO\n";
}
