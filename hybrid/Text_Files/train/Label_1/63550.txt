def sieve(n):
    prime = [True for i in range(n+1)]
    p = 2
    while (p * p <= n):
        if (prime[p] == True):
            for i in range(p * p, n+1, p):
                prime[i] = False
        p += 1
    return prime

def solve(n):
    prime = sieve(n)
    a = [0] * (n+1)
    counter = 1
    for p in range(2, n+1):
        if prime[p]:
            for j in range(p, n+1, p):
                if a[j] == 0:
                    a[j] = counter
            counter += 1
    print(*a[2:])

n = int(input())
solve(n)
