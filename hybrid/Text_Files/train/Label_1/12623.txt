#include <iostream>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    bool consistent = true;
    for (int i = 1; i < s.size(); i++) {
      if (s[i] != s[i - 1]) {
        consistent = false;
        break;
      }
    }
    cout << (consistent ? "YES" : "NO") << endl;
  }
  return 0;
}
