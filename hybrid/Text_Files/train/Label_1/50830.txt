import java.util.*;

public class Main {
    private static final int MAXN = 100005;
    private static final int MAXM = 100005;
    private static int[] dis = new int[MAXN];
    private static Queue<Integer> que = new PriorityQueue<>(MAXN, (a, b) -> dis[a] - dis[b]);
    private static List<Integer>[] adj = new ArrayList[MAXN];
    private static int[] vis = new int[MAXN];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
            vis[i] = 0;
            dis[i] = MAXN;
            que.offer(i);
        }
        for (int i = 1; i <= m; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        dis[1] = 0;
        while (!que.isEmpty()) {
            int cur = que.poll();
            if (vis[cur] == 1) continue;
            vis[cur] = 1;
            System.out.print(cur + " ");
            for (Integer next : adj[cur]) {
                if (vis[next] == 0) {
                    if (dis[next] > dis[cur] + 1) {
                        dis[next] = dis[cur] + 1;
                        que.offer(next);
                    }
                }
            }
        }
    }
}
