import heapq
n, m, k = map(int, input().split())
g = [[]]
for _ in range(n): g.append([])
for _ in range(m):
	x, y, w = map(int, input().split())
	g[x].append((w, y))
	g[y].append((w, x))
d = [5e18] * (n+1)
vis = [False] * (n+1)
d[1] = 0
pq = [(0, 1)]
while pq:
	cd, cv = heapq.heappop(pq)
	if vis[cv]: continue
	vis[cv] = True
	for w, v in g[cv]:
		if not vis[v] and cd + w < d[v]:
			d[v] = cd + w
			heapq.heappush(pq, (d[v], v))
edges = []
for i in range(1, n+1):
	for j, w in g[i]:
		if d[i] == d[j] + w: edges.append((d[i], i, j))
edges.sort()
print(min(len(edges), k))
for i in range(min(len(edges), k)): print(edges[i][1], edges[i][2])
