import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    private static int[][] wanted;
    private static int[][] tested;
    private static int n;
    private static boolean[] ct;
    private static long[] dist;
    private static int[] parent;
    private static int[] topSort;
    private static int tp;
    private static int[] inDegree;
    private static long MAX_VAL = 2_640_000_000L;
    private static final long NOT_FOUND = 2_640_000_001L;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        wanted = new int[n][3];
        tested = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                wanted[i][j] = in.nextInt();
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                tested[i][j] = in.nextInt();
            }
        }
        ct = new boolean[n];

        if (!solve()) {
            System.out.println("Impossible");
            in.close();
            return;
        }
        System.out.println("Possible");
        for (boolean b : ct) {
            System.out.print(b ? 1 : 0 + " ");
        }
        System.out.println();
        int m = (int) dist[n];
        System.out.println(m);
        for (int i = 0; i < m; i++) {
            System.out.println(parent[topSort[i]] + " " + topSort[i]);
        }

        in.close();
    }

    private static boolean solve() {
        dist = new long[n + 1];
        parent = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            dist[i] = MAX_VAL;
        }
        dist[1] = 0;

        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v = 1; v <= n; v++) {
                if (u == v) {
                    continue;
                }
                long nw = dist[u] + getWeight(u, v);
                if (nw < dist[v]) {
                    parent[v] = u;
                    dist[v] = nw;
                    q.add(v);
                }
            }
        }

        tp = 0;
        topSort = new int[n];
        inDegree = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (parent[j] == i) {
                    inDegree[j]++;
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (inDegree[i] == 0) {
                topSort[tp++] = i;
                q.add(i);
            }
        }

        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v = 1; v <= n; v++) {
                if (u == v) {
                    continue;
                }
                if (parent[v] == u) {
                    inDegree[v]--;
                    if (inDegree[v] == 0) {
                        topSort[tp++] = v;
                        q.add(v);
                    }
                }
            }
        }

        for (int i = 0; i < tp; i++) {
            for (int j = i + 1; j < tp; j++) {
                if (getWeight(topSort[i], topSort[j]) > 0) {
                    return false;
                }
            }
        }

        return true;
    }

    private static long getWeight(int u, int v) {
        if (wanted[v - 1][0] == 1 && wanted[u - 1][0] == 0) {
            return NOT_FOUND;
        }
        if (wanted[v - 1][1] == 1 && wanted[u - 1][1] == 0) {
            return NOT_FOUND;
        }
        if (wanted[v - 1][2] == 1 && wanted[u - 1][2] == 0) {
            return NOT_FOUND;
        }
        if (tested[u - 1][0] == 0) {
            return 0;
        }
        if (wanted[v - 1][0] == 1 && wanted[u - 1][0] == 1 && tested[u - 1][0] == 1) {
            ct[u - 1] = true;
        }
        if (wanted[v - 1][1] == 1 && wanted[u - 1][1] == 1 && tested[u - 1][1] == 1) {
            ct[u - 1] = true;
        }
        if (wanted[v - 1][2] == 1 && wanted[u - 1][2] == 1 && tested[u - 1][2] == 1) {
            ct[u - 1] = true;
        }
        return 1;
    }
}
