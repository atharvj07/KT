#include <bits/stdc++.h>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    string s, t, p;
    cin >> s >> t >> p;
    int cnt[26] = {};
    int cnt2[26] = {};
    for (char ch : t) cnt[ch - 'a']++;
    for (char ch : s) cnt[ch - 'a']--;
    for (char ch : p) cnt2[ch - 'a']++;
    bool ok = true;
    for (int i = 0; i < 26; i++) {
      if (cnt[i] > 0 && cnt2[i] < cnt[i]) {
        ok = false;
        break;
      }
    }
    if (ok)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
