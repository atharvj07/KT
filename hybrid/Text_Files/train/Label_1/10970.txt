#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long a, b, c;
        cin >> a >> b >> c;
        cout << max(0LL, 1 + max(b, c) - a) << ' '
             << max(0LL, 1 + max(a, c) - b) << ' '
             << max(0LL, 1 + max(a, b) - c) << '\n';
    }
    return 0;
}
