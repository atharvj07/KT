#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int x;
    cin >> x;
    if (x == 1) {
      cout << 1 << endl;
      continue;
    }
    vector<int> digits;
    for (int i = 9; i >= 2; i--) {
      while (x >= i && digits.size() < i) {
        x -= i;
        digits.push_back(i);
      }
    }
    if (x != 0 || digits.size() == 0) {
      cout << -1 << endl;
    } else {
      sort(digits.begin(), digits.end());
      for (int i = 0; i < digits.size(); i++) {
        cout << digits[i];
      }
      cout << endl;
    }
  }
  return 0;
}
