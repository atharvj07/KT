import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        int[] b = new int[n + 1];
        StringTokenizer st = new StringTokenizer(reader.readLine());
        for (int i = 1; i <= n; i++) {
            b[i] = Integer.parseInt(st.nextToken());
        }
        int[] a = new int[n + 1];
        st = new StringTokenizer(reader.readLine());
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }

        List<int[]>[] graph = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }

        for (int i = 1; i < n; i++) {
            st = new StringTokenizer(reader.readLine());
            int x = Integer.parseInt(st.nextToken());
            int k = Integer.parseInt(st.nextToken());
            graph[x].add(new int[]{i + 1, k});
            graph[i + 1].add(new int[]{x, 1});
        }

        // Run BFS starting from sink node to find if there's a path from source to sink
        boolean[] visited = new boolean[n + 1];
        Arrays.fill(visited, false);

        bfs(graph, n, a, b, visited);

        boolean isPossible = true;
        for (int i = 1; i <= n; i++) {
            if (b[i] < a[i]) {
                isPossible = false;
                break;
            }
        }

        System.out.println(isPossible ? "YES" : "NO");
    }

    private static void bfs(List<int[]>[] graph, int n, int[] a, int[] b, boolean[] visited) {
        int[] queue = new int[n + 1];
        int head = -1, tail = -1;

        queue[++tail] = n;

        while (head != tail) {
            int u = queue[++head];

            if (visited[u]) {
                continue;
            }

            visited[u] = true;

            for (int[] e : graph[u]) {
                int v = e[0], k = e[1];

                b[v] -= (a[u] + k - 1) / k;
            }

            queue[++tail] = u;
        }
    }
}
