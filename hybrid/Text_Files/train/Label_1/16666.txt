#include <iostream>
#include <deque>
using namespace std;
int main() {
    int n, q; cin >> n >> q;
    deque<int> v; for (int i = 1; i <= n; i++) { int x; cin >> x; v.push_back(x); }
    while (q--) {
        int m; cin >> m;
        int a, b;
        if (m <= v.size()) {
            a = v.front();
            v.pop_front();
            b = v.front();
            v.pop_front();
        } else {
            m = m % v.size();
            for (int i = 1; i <= m; i++) {
                int c = v.front();
                v.pop_front();
                v.push_back(c);
            } a = v.front();
            v.pop_front();
            b = v.front();
            v.pop_front();
        }
        cout << a << ' ' << b << endl;
    }
    return 0;
}
