#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  string a;
  cin >> a;
  int n = a.size();
  vector<int> dp(n + 1);
  dp[n] = 1;
  for (int i = n - 1; i >= 0; i--) {
    dp[i] = dp[i + 1];
    if (a[i] == '9') dp[i] += dp[i + 1];
    if (i + 1 < n && (a[i] == '1' && a[i + 1] == '8' || a[i] == '8' && a[i + 1] == '1'))
      dp[i] += dp[i + 2];
  }
  cout << dp[0] << endl;
}
