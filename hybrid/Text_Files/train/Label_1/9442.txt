#include <iostream>

using namespace std;

const int maxn = 1e5 + 5, maxm = 1e6 + 5;

int a[maxn], q[maxm];
int s[maxn];

int main() {
  int n, m;
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    s[i] = s[i - 1] + a[i];
  }
  cin >> m;
  for (int i = 1; i <= m; ++i) cin >> q[i];
  for (int i = 1; i <= m; ++i) {
    for (int j = 1; j <= n; ++j)
      if (s[j] >= q[i]) { cout << j << endl; break; }
  }
  return 0;
}
