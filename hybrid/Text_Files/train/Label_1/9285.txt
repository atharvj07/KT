#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (n); ++i)

long long solve() {
  int n, s;
  cin >> n >> s;
  vector<tuple<int, int, int>> v(n);
  rep(i, n) {
    int a, b, c;
    cin >> a >> b >> c;
    v[i] = {a, b, c};
  }
  sort(v.rbegin(), v.rend());
  long long ans = 0;
  int cnt = 0;
  rep(i, n) {
    if (cnt >= s) break;
    int a, b, c;
    tie(a, b, c) = v[i];
    ans += min(a, s - cnt) * (long long)b;
    cnt += min(a, s - cnt);
  }
  cnt = 0;
  rep(i, n) {
    if (cnt >= s) break;
    int a, b, c;
    tie(a, b, c) = v[i];
    ans += min(a, s - cnt) * (long long)c;
    cnt += min(a, s - cnt);
  }
  return ans;
}

int main() { return cout << solve(), 0; }
