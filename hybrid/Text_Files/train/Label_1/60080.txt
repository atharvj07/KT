n = int(input())
a = []
for i in range(n):
    a.append([int(x) for x in input().split()])
a.sort(key=lambda x: x[1], reverse=True)
ans = 0
for i in range(n):
    l = i + 1
    r = n - 1
    ind = -1
    while l <= r:
        m = (l + r) // 2
        if a[m][1] <= a[i][0]:
            ind = m
            l = m + 1
        else:
            r = m - 1
    if ind == -1:
        ans += a[i][0]
    else:
        ans += a[ind][1]
print(ans / n)
