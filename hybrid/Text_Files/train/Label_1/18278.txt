#include <iostream>
#include <vector>
using namespace std;

int n, k, q;
vector<pair<int, int>> v;

int main() {
  cin >> n >> k >> q;
  for (int i = 0; i < n; i++) {
    int a, b;
    cin >> a >> b;
    v.push_back({a, b});
  }
  while (q--) {
    int a, b;
    cin >> a >> b;
    int ans = 0;
    for (auto x : v) {
      if (x.first <= a && x.second >= b) ans++;
      if (x.first > b) break;
    }
    if (ans >= k) cout << b - a + 1 << endl;
    else
      cout << 0 << endl;
  }
}
