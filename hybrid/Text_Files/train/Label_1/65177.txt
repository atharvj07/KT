def min_coins_for_permutation(n, p, s):
    position = {}
    for i in range(n):
        position[s[i]] = i
    ops = []
    for i in range(n):
        while p[i] != s[i]:
            swap_with = position[p[i]]
            ops.append((i + 1, swap_with + 1))
            p[i], p[swap_with] = p[swap_with], p[i]

    cost = sum(abs(i - j) for i, j in ops)
    print(cost)
    print(len(ops))
    for op in ops:
        print(*op)

n = int(input().strip())
p = list(map(int, input().strip().split()))
s = list(map(int, input().strip().split()))
min_coins_for_permutation(n, p, s)
