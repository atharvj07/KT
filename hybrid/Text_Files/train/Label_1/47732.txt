import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long k = scanner.nextLong();
        long counter = 0;
        long curNum = 0;
        for (int i = 1; i <= n; i++) {
            counter += i;
            if (counter >= k) {
                curNum = i;
                break;
            }
        }
        k -= counter - curNum;
        for (int i = 1; i <= n; i++) {
            if (k <= curNum) {
                System.out.println(i);
                return;
            }
            k -= curNum;
            curNum--;
        }
    }
}
