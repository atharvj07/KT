#include <bits/stdc++.h>
using namespace std;
#define fastio() ios::sync_with_stdio(false), cin.tie(nullptr), cout.tie(nullptr)
#define endl "\n"
#define all(x) begin(x), end(x)

int n, p[100005], vis[100005], fl;
vector<string> f, l;

int main() {
  fastio();
  cin >> n;
  f.resize(n), l.resize(n);
  for (int i = 0; i < n; ++i) {
    cin >> f[i];
    cin >> l[i];
  }
  for (int i = 0; i < n; ++i) {
    cin >> p[i];
  }
  for (int i = 0; i < n; ++i) {
    if (vis[p[i]]) {
      fl = 1;
    }
    vis[p[i]] = 1;
  }
  if (fl) {
    cout << "NO";
    return 0;
  }
  for (int i = 0; i < n - 1; ++i) {
    int x = p[i], y = p[i + 1];
    if (f[x - 1] > f[y - 1] && l[x - 1] > l[y - 1]) {
      fl = 1;
      break;
    }
    if (f[x - 1] == f[y - 1] && l[x - 1] > l[y - 1]) {
      fl = 1;
      break;
    }
    if (f[x - 1] > f[y - 1] && l[x - 1] == l[y - 1]) {
      fl = 1;
      break;
    }
  }
  if (fl) {
    cout << "NO";
  } else {
    cout << "YES";
  }
  return 0;
}
