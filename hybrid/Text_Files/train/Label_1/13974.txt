#include <iostream>
#include <map>
#include <vector>

using namespace std;

int main() {
  long long n, m;
  cin >> n >> m;
  map<long long, long long> balance;
  while (m--) {
    long long a, b, c;
    cin >> a >> b >> c;
    balance[a] -= c;
    balance[b] += c;
  }
  long long sum = 0;
  for (const auto &p : balance) {
    if (p.second > 0) sum += p.second;
  }
  cout << sum << endl;
  return 0;
}
