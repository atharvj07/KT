import math
import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, x = map(int, input().split())
    a = [int(i) for i in input().split()]
    mn = 0
    mx = 0

    for j in a:
        mn += math.ceil(j / x)
        mx += math.ceil(j / x) if j % x != 0 else math.ceil(j / x) - 1

    print(mn, mx)
