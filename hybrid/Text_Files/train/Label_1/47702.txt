import java.util.*;
import java.io.*;

class Main {
    static long MOD = 1000000007L;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        long[] fact = new long[n + 1];
        fact[0] = 1;
        for (int i = 1; i <= n; i++) {
            fact[i] = (i * fact[i - 1]) % MOD;
        }
        long ans = 0;
        for (int i = k + 1; i <= n; i++) {
            long tmp = (n - i + 1) % MOD;
            tmp = (tmp * (tmp - 1)) % MOD;
            tmp = (tmp * fact[n - i]) % MOD;
            ans = (ans + tmp) % MOD;
        }
        System.out.println(ans);
    }
}
