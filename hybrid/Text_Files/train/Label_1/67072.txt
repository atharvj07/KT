n = int(input())
L = list(map(int, input().split()))
L.insert(0, 0)
G = [[] for _ in range(n+1)]
for i in range(1, n):
    if L[i] > L[i+1]:
        G[L[i]].append(L[i+1])
        G[L[i+1]].append(L[i])
        L[i], L[i+1] = L[i+1], L[i]
mn = 1<<30
for i in range(1, n+1):
    mn = min(mn, len(G[i]))
print(n-mn)
