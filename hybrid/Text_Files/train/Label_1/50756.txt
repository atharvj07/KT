import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        boolean[] exist0 = new boolean[1000001];
        boolean[] exist1 = new boolean[1000001];
        for (int i = 0; i < n; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            if (a == 0) {
                exist0[b] = true;
            } else {
                exist1[b] = true;
            }
        }
        boolean ok = true;
        for (int i = 0; i < 1000001; i++) {
            if (!exist0[i] && !exist1[i]) {
                ok = false;
            }
        }
        if (ok) {
            System.out.println("Borna");
        } else {
            System.out.println("Ani");
        }
    }
}
