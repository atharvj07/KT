import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    mn = 1e9
    for i in range(n):
        if i + 1 != a[i]:
            mn = min(mn, a[i] % (i + 1))
    if mn == 1e9:
        print(-1)
    else:
        print(mn)
