def solve():
    n, m = map(int, input().split())
    s = input()
    t = input()
    if s.count('*') > 1 or (s.count('*') == 1 and s.count('*') > m - n + 1):
        print('NO')
    else:
        if s.count('*') == 1:
            s = s.replace('*', t[n - 1:])
        if s == t:
            print('YES')
        else:
            print('NO')

solve()
