import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long l = sc.nextLong();
        long r = sc.nextLong();
        long Max = 0;
        long ans = 0;
        for (long i = 2; i <= Math.sqrt(r); i++) {
            long count = 0;
            for (long j = l; j <= r; j++) {
                if (j % i == 0) {
                    count++;
                }
            }
            if (count > Max) {
                Max = count;
                ans = i;
            }
        }
        System.out.println(ans);
    }
}
