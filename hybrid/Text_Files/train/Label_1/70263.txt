from fractions import Fraction

n, m = map(int, input().split())
dist = []
for _ in range(n):
    dist.append(list(map(int, input().split())))
ans = 0
for v in dist:
    dp = [0] * (n + 1)
    for j in range(n):
        for i in range(n - 1, j - 1, -1):
            dp[i + 1] += dp[i]
            if v[i] >= j + 1:
                dp[j + 1] += 1
    ans += sum(dp)
print((ans * Fraction(n, m)) % 998244353)
