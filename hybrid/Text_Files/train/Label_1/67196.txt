n, sx, sy = map(int, input().split())
students = [list(map(int, input().split())) for _ in range(n)]

max_students = 0
best_location = (0, 0)

for dx, dy in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
    tent_x = sx + dx
    tent_y = sy + dy
    count = 0
    for x, y in students:
        if (abs(sx - x) + abs(sy - y) == abs(tent_x - x) + abs(tent_y - y) + 1):
            count += 1
    if count > max_students:
        max_students = count
        best_location = (tent_x, tent_y)

print(max_students)
print(best_location[0], best_location[1])
