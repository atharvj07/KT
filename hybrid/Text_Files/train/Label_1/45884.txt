import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();
            char[] arr = s.toCharArray();
            boolean flag = true;
            int countT = 0, countM = 0;
            for (char c : arr) {
                if (c == 'T') {
                    countT++;
                } else {
                    countM++;
                }
            }
            if (countT != countM || countT % 3 != 0) {
                flag = false;
            } else if (countT % 3 == 0) {
                int countTM = 0;
                for (char c : arr) {
                    if (c == 'T') {
                        countTM++;
                    } else {
                        countTM--;
                    }
                    if (countTM < 0) {
                        flag = false;
                        break;
                    }
                }
                if (countTM != 0) {
                    flag = false;
                }
            }
            if (flag) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
