n = int(input())
a = [[int(j=="X") for j in i] for i in [input() for x in range(n)]]
for i in range(n):
    for j in range(4):
        if j%2==0 and not (a[i][j] or a[i][j+1]):
            print("YES")
            a[i][j] = a[i][j+1] = 1
            for x in a: print(''.join(['XO' if y else '++' for y in x]))
            exit()
print("NO")
