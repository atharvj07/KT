import sys
test_cases = int(input())
for _ in range(test_cases):
    n = int(input())
    a = [int(i) for i in input().split()]
    b = [int(i) for i in input().split()]
    if sum(a) != sum(b):
        print(-1)
        continue
    ops = 0
    for i in range(n):
        if a[i] == b[i]:
            continue
        pos = b[i] - a[i]
        if pos < 0:
            print(-1)
            break
        ops += pos
        for j in range(n):
            if a[j] + pos <= 100:
                a[j] += pos
        a[i] = b[i]
    if ops == 0:
        print(0)
        continue
    print(ops)
    for i in range(n):
        if a[i] != b[i]:
            for j in range(n):
                if a[j] + 1 <= 100:
                    print(j + 1, i + 1)
                    a[j] += 1
                    a[i] -= 1
                    break
