n, m = map(int, input().split())
a = [input() for _ in range(n)]
ans = float('inf')
#Horizontal cutting
for i in range(n):
    for j in range(m):
        if a[i][j] == 'W':
            ans = min(ans, max(i, n-1-i, j, m-1-j))
for j in range(m):
    b = []
    for i in range(n):
        b += a[i][j]
    c = 0
    i = 0
    while i < n:
        if b[i] == 'W':
            c += 1
        else:
            c = 0
        ans = min(ans, c)
        i += 1
#Vertical cutting
for i in range(n):
    b = []
    for j in range(m):
        b += a[i][j]
    c = 0
    i = 0
    while i < m:
        if b[i] == 'W':
            c += 1
        else:
            c = 0
        ans = min(ans, c)
        i += 1
if ans == float('inf'):
    ans = -1
print(ans)
