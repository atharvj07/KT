import sys
from collections import Counter

lines = [list(map(int,line.split())) for line in sys.stdin]
cnt = Counter()
for line in lines: line.sort(); cnt[tuple(line)] += 1
if len(cnt)==8 and all(cnt[k]==1 for k in cnt.keys()):
    print("YES")
    for k in sorted(cnt.keys()): print(*k)
else: print("NO")
