import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }
        int maxAnd = 0;
        for (int i : arr) {
            maxAnd |= i;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if ((arr[i] & arr[j]) == 0) {
                    ans += n - j - 1;
                }
            }
        }
        if (maxAnd == 0) {
            ans = (n * (n - 1) / 2) % 1000000007;
        }
        System.out.println(ans % 1000000007);
    }
}
