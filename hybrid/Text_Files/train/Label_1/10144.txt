#include <iostream>
#include <vector>
using namespace std;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n, m;
    cin >> n >> m;
    vector<string> v(n);
    vector<bool> row(n), col(m);
    for (int i = 0; i < n; i++) {
      cin >> v[i];
      for (int j = 0; j < m; j++) {
        if (v[i][j] == '*') row[i] = col[j] = true;
      }
    }
    int ans = 0;
    for (int i = 0; i < n; i++)
      if (!row[i]) ans++;
    for (int j = 0; j < m; j++)
      if (!col[j]) ans++;
    cout << ans << '\n';
  }
}
