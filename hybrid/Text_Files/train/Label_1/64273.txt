for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    ans = 0
    for i in range(n):
        if a[i] == 0:
            ans += 1
            j = i + 1
            while j < n and a[j] == 0:
                j += 1
            if j == n:
                continue
            k = j - i - 1
            if j < n and a[j] == 1:
                ans += k - 1
    print(ans)
