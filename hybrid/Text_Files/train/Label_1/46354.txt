import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.PriorityQueue;

class Main {

    static int n, k, m;
    static List<Edge> graph[];
    static int[] vis;

    static boolean validPath(Edge e) {
        return (vis[e.from] + vis[e.to]) == 0;
    }

    public static void main(String[] args) throws Exception {
        // Read input
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        k = Integer.parseInt(line[1]);
        m = Integer.parseInt(br.readLine());
        graph = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++)
            graph[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int from = Integer.parseInt(line[0]);
            int to = Integer.parseInt(line[1]);
            int cost = Integer.parseInt(line[2]);
            graph[from].add(new Edge(from, to, cost));
            graph[to].add(new Edge(to, from, cost));
        }

        // Initialize vis array
        vis = new int[n + 1];

        // Initialize priority queue
        PriorityQueue<Edge> pq = new PriorityQueue<>((a, b) -> a.cost - b.cost);
        pq.add(new Edge(1, 1, 0));

        // Dijkstra's algorithm
        long minCost = Long.MAX_VALUE;
        while (!pq.isEmpty()) {
            Edge curr = pq.poll();
            if (vis[curr.to] != 0)
                continue;
            vis[curr.to] = curr.from;
            if (--k == 0) {
                minCost = Math.min(minCost, curr.cost);
                break;
            }
            for (Edge next : graph[curr.to])
                if (validPath(next))
                    pq.add(new Edge(curr.to, next.to, curr.cost + next.cost));
        }

        // Print result
        System.out.println(minCost == Long.MAX_VALUE ? -1 : minCost);
    }

    static class Edge {
        int from, to, cost;

        public Edge(int from, int to, int cost) {
            this.from = from;
            this.to = to;
            this.cost = cost;
        }
    }
}
