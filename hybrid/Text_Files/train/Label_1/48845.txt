import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int xor = 0;
            for (int i = 0; i < n; i++) {
                xor ^= sc.nextInt();
            }
            System.out.println(xor == 0 || n == 2 ? "YES" : "NO");
        }
    }
}
