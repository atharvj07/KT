n = int(input())
a = [int(x) for x in input().split()]
q = int(input())
for _ in range(q):
    op, x = input().split()
    if op == '+':
        a.append(int(x))
    else:
        a.remove(int(x))
    a.sort()
    if len(set(a)) != 2 or a[0] * 4 != sum(a):
        print('NO')
    else:
        print('YES')
