#include <iostream>
int main() {
    int n;
    std::cin >> n;
    int spect = 3;
    for (int i = 0; i < n; i++) {
        int a;
        std::cin >> a;
        if (a == spect) {
            std::cout << "NO\n";
            return 0;
        }
        spect = 6 - spect - a;
    }
    std::cout << "YES\n";
    return 0;
}
