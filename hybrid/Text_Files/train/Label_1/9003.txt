#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  vector<int> b(10);
  for (int i = 0; i < n; i++) {
    int x = 0;
    for (int j = 0; j < a[i].size(); j++) {
      x *= 10;
      x += b[a[i][j] - 'a'];
    }
    b[a[i][0] - 'a'] += 1;
  }
  int carry = 0;
  for (int i = 9; i >= 0; i--) {
    b[i] += carry;
    if (b[i] >= 10) {
      b[i] -= 10;
      carry += 1;
    } else {
      carry = 0;
    }
  }
  int sum = 0;
  for (int i = 0; i < 10; i++) sum += i * b[i];
  cout << sum << endl;
  return 0;
}
