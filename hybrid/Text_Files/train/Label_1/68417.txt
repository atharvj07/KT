h1, h2 = map(int, input().split())
a, b = map(int, input().split())

# From 2pm to 10pm, 8 hours of climbing on the first day
first_day_climb = 8 * a
h1 += first_day_climb

if h1 >= h2:
    print(0)
else:
    # Net gain per day after the first day
    net_gain = (12 * a) - (12 * b)
    
    if net_gain <= 0:
        print(-1)
    else:
        days_needed = (h2 - h1 - 1) // net_gain + 1
        print(days_needed)
