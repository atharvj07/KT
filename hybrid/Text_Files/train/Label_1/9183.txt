#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mxN = 3e5 + 5;
int sieve[mxN];
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    ll l, r;
    cin >> l >> r;
    if ((r - l + 1) % 2 != 0) {
        cout << "NO" << endl;
        return 0;
    }
    for (int p = 2; p * p <= r; p++) {
        if (sieve[p] == 0) {
            for (int i = p * p; i <= r; i += p)
                sieve[i] = p;
        }
    }
    cout << "YES" << endl;
    for (ll i = l; i <= r; i += 2) {
        if (sieve[i] == 0)
            sieve[i] = i;
        if (sieve[i + 1] == 0)
            sieve[i + 1] = i + 1;
        cout << i << " " << i + 1 << endl;
    }
    return 0;
}
