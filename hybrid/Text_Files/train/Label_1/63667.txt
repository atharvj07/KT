import sys
n = int(input())

res = []
for i in range(n):
    x = input()
    res.append(x[0])

res.sort()
ans = []
lst = []
for i in range(26):
    lst.append(chr(ord('a')+i))

new_lst = []
for x in res:
    if x not in new_lst:
        new_lst.append(x)

if len(new_lst) != len(lst):
    print('Impossible')
else:
    for i in range(len(new_lst)):
        ans.append(lst[new_lst.index(res[i])])
    print(''.join(ans))
