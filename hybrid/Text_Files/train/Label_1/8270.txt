#include<bits/stdc++.h>
using namespace std;
int g[2001][2001];
char c[2001][2001];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  for (int tt = 1; tt <= t; tt++) {
    int n, m;
    cin >> n >> m;
    memset(g, 0, sizeof(g));
    for (int i = 1; i <= n; ++i) {
      for (int j = 1; j <= m; ++j) {
        cin >> c[i][j];
      }
    }
    vector<tuple<int, int, int, int>> v;
    int co = 0;
    for (int i = 1; i <= n; ++i) {
      for (int j = 1; j <= m; ++j) {
        if (c[i][j] != '.') {
          if (g[i][j]) continue;
          int len = 1, x = i;
          while (x + 1 <= n && c[x][j] == c[x + 1][j] && !g[x + 1][j])
            x++, len++;
          for (int k = x; k >= i; k--) g[k][j] = 1;
          if (len == 1) {
            x = j;
            len = 1;
            while (x + 1 <= m && c[i][x] == c[i][x + 1] && !g[i][x + 1])
              x++, len++;
            for (int k = x; k >= j; k--) g[i][k] = 1;
          }
          v.emplace_back(i, j, x, j);
          co++;
        }
      }
    }
    cout << (co <= 26 ? "YES\n" : "NO\n");
    if (co <= 26) {
      cout << co << '\n';
      for (auto e : v) cout << get<0>(e) << ' ' << get<1>(e) << ' ' << get<2>(e)
                           << ' ' << get<3>(e) << '\n';
    }
  }
  return 0;
}
