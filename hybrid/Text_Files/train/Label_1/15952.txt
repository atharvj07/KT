#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n), b;
  for (int &i : a) cin >> i;
  sort(a.begin(), a.end());
  for (int i = 1; i < n - 1; i++) {
    if (a[i] == a[i - 1] || a[i] == a[i + 1]) b.push_back(a[i]), a.erase(a.begin() + i);
  }
  if (!a.empty()) b.push_back(a[0]), a.erase(a.begin());
  if (a.empty()) {
    for (int i : b) cout << i << ' ';
  } else {
    cout << -1;
  }
}
