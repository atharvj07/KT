#include <bits/stdc++.h>
using namespace std;
long long n, c0, c1, best[100000005], ans;
int main() {
  cin >> n >> c0 >> c1;
  best[0] = 0;
  for (long long i = 1; i <= n; i++) {
    best[i] = 1e18;
    if ((i - 1) % 2 == 0)
      best[i] = min(best[i], best[(i - 1) / 2] + c0);
    if (i % 2 == 0) best[i] = min(best[i], best[i / 2] + c0);
    best[i] = min(best[i], best[i - 1] + c1);
  }
  cout << best[n] << endl;
}
