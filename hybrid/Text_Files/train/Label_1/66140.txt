import sys
input = sys.stdin.readline
n,p=map(int,input().split())
a=list(map(int,input().split()))
a.sort()
g=[]
for i in range(p):
    k=i+1
    now=0
    f=0
    for j in range(n):
        if a[j]>k:
            break
        now+=1
        if now>=k:
            f=1
    if f==0:
        g.append(k)
print(len(g))
print(*g)
