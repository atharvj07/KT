#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int& ai: a) cin >> ai;
  vector<int> b(m);
  for (int& bi: b) cin >> bi;
  sort(b.begin(), b.end());
  sort(a.begin(), a.end());
  int score = 0;
  for (int i = 0; i < n; i++) {
    bool found = false;
    for (int j = 0; j < m; j++) {
      if (b[j] == i + 1) {
        if (score >= a[i]) {
          score += a[i];
          found = true;
        }
        break;
      }
    }
    if (!found) score += a[i];
  }
  cout << score << endl;
  return 0;
}
