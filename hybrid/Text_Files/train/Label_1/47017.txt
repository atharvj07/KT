import java.util.*;
import java.io.*;

public class Main {
    static void solve(int x, int y, int z) {
        if (x == y && y == z) {
            System.out.println("YES");
            System.out.println(x + " " + y + " " + z);
        } else if (x == y || y == z || z == x) {
            int max = Math.max(x, Math.max(y, z));
            if (max == x) {
                System.out.println("YES");
                System.out.println(max + " " + (max - 1) + " " + (max - 1));
            } else if (max == y) {
                System.out.println("YES");
                System.out.println((max - 1) + " " + max + " " + (max - 1));
            } else {
                System.out.println("YES");
                System.out.println((max - 1) + " " + (max - 1) + " " + max);
            }
        } else {
            System.out.println("NO");
        }
    }
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            int x = in.nextInt();
            int y = in.nextInt();
            int z = in.nextInt();
            solve(x, y, z);
        }
    }
}
