from collections import Counter

n, q = map(int, input().split())
s = input()
c = Counter(s)
ABC = ['abc', 'acb', 'bca', 'bac', 'cab', 'cba']

for _ in range(q):
    i, ch = input().split()
    i = int(i) - 1
    if s[i] == ch:
        print(0)
        continue
    
    s = s[:i] + ch + s[i+1:]
    sb = s[:-2]
    b = ''
    for j in range(len(sb)):
        if sb[j] == 'a' or (sb[j] == 'b' and ch == 'c') or (sb[j] == 'c' and ch == 'b'):
            b += sb[j]
    
    for j in range(len(ABC)):
        if b.find(ABC[j]) != -1:
            c[ABC[j][0]] -= 1
            c[ABC[j][1]] -= 1
            c[ABC[j][2]] -= 1
    
    flag = False
    for j in range(len(ABC)):
        if c[ABC[j][0]] >= 1 and c[ABC[j][1]] >= 1 and c[ABC[j][2]] >= 1:
            flag = True
            break
    
    print(1 if flag else 0)
