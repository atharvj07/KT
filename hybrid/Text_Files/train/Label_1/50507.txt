import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] cost = new long[n];
        for (int i = 0; i < n; i++) {
            cost[i] = sc.nextInt();
        }
        long cur = 0, ans = 0;
        for (long c : cost) {
            if (c <= cur) {
                ans += c;
                cur -= c;
            } else {
                ans += c - cur;
                cur = c * 9 / 10;
            }
        }
        System.out.println(ans);
    }
}
