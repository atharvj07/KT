#include <bits/stdc++.h>
using namespace std;
const int MAXN = 500;
typedef long long ll;

struct Monster {
    ll w;
    int idx;

    Monster(ll w, int idx) : w(w), idx(idx) {}
    bool operator<(const Monster &other) const { return w < other.w; }
};

int main() {
    int n;
    cin >> n;
    vector<ll> a(n), b;
    for (ll &x : a) cin >> x;
    int k;
    cin >> k;
    for (int i = 0; i < k; i++) {
        ll x;
        cin >> x;
        b.push_back(x);
    }
    if (a.size() != b.size()) {
        cout << "NO\n";
        return 0;
    }
    vector<int> p(n);
    for (int i = 0; i < n; i++) p[i] = i;
    auto eat = [&](int x, char d) {
        cout << x << " " << d << "\n";
        if (d == 'R') {
            swap(p[x], p[x + 1]);
            a[x] += a[x + 1];
            a.erase(a.begin() + x + 1);
            p.erase(p.begin() + x + 1);
        } else {
            swap(p[x - 1], p[x]);
            a[x - 1] += a[x];
            a.erase(a.begin() + x);
            p.erase(p.begin() + x);
        }
    };
    while (a.size() != b.size()) {
        vector<Monster> monsters;
        for (int i = 0; i < a.size(); i++) {
            if (i + 1 < a.size() && a[i] > a[i + 1]) monsters.push_back(Monster(a[i], i));
            if (i > 0 && a[i] > a[i - 1]) monsters.push_back(Monster(a[i], i));
        }
        sort(monsters.begin(), monsters.end());
        if (monsters.empty()) {
            cout << "NO\n";
            return 0;
        }
        eat(monsters[0].idx, monsters[0].idx == 0 ? 'R' : 'L');
    }
    for (int i = 0; i < a.size(); i++) {
        if (a[i] != b[i]) {
            cout << "NO\n";
            return 0;
        }
    }
    cout << "YES\n";
    return 0;
}
