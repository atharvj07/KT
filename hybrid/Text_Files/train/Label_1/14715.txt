#include <bits/stdc++.h>
using namespace std;

#define ll long long

int main() {
  ll n;
  cin >> n;
  if (n == 1 || (n & (n + 1)) == 0)
    cout << 0 << endl;
  else {
    int cnt = 0;
    while (n >= 1 << cnt) cnt++;
    for (int i = cnt - 2; i >= 0; i--) {
      if ((1LL << i) & n) n ^= 1LL << i;
    }
    cout << cnt - 1 << endl;
    for (int i = cnt - 2; i >= 0; i--) cout << i << ' ';
    cout << endl;
  }
  return 0;
}
