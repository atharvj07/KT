from sys import stdin
from heapq import *

rd = stdin.readline
n, k = map(int, rd().split())
g = [list(rd().rstrip()) for _ in range(n)]

h = [(-ord(g[0][i]), 0, i) for i in range(n)]
heapify(h)

while k and h:
    cost, r, c = heappop(h)
    g[r][c] = 'a'
    cost = -ord(g[r + 1][c]) if r + 1 < n else 0
    heappush(h, (cost, r + 1, c))
    cost = -ord(g[r][c + 1]) if c + 1 < n else 0
    heappush(h, (cost, r, c + 1))
    k -= 1

print(''.join(''.join(row) for row in g))
