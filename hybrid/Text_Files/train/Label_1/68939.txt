n = int(input())
board = [input() for _ in range(n)]

def check_even_o(board):
    for i in range(n):
        for j in range(n):
            count_o = 0
            for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                x, y = i + dx, j + dy
                if 0 <= x < n and 0 <= y < n and board[x][y] == 'o':
                    count_o += 1
            if count_o % 2 != 0:
                return "NO"
    return "YES"

print(check_even_o(board))
