#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;
const ll inf = 2e18;
ll a[305], L[305], R[305];
ll b[305];
vector<ll> seg[305];
int ans, sum;
void dfs(ll l, ll r, ll d) {
  if (l > r) return;
  for (ll i : seg[l]) {
    if (L[i] > r) continue;
    b[i] = d + 1;
    b[l] += d + 1;
    b[r + 1] -= d + 1;
    dfs(l + 1, min(r, R[i]), d + 1);
    b[i] = 0;
    b[l] -= d + 1;
    b[r + 1] += d + 1;
  }
  for (ll i : seg[r]) {
    if (L[i] > r) continue;
    b[i] = d + 1;
    b[l] += d + 1;
    b[r + 1] -= d + 1;
    dfs(max(l, L[i]), r - 1, d + 1);
    b[i] = 0;
    b[l] -= d + 1;
    b[r + 1] += d + 1;
  }
  for (ll i : seg[l]) {
    if (L[i] <= r && R[i] >= r) {
      b[l] += 1;
      b[r + 1] -= 1;
      dfs(l + 1, r - 1, d);
      b[l] -= 1;
      b[r + 1] += 1;
    }
  }
  for (ll i : seg[r]) {
    if (L[i] <= r && R[i] >= r) {
      b[l] += 1;
      b[r + 1] -= 1;
      dfs(l + 1, r - 1, d);
      b[l] -= 1;
      b[r + 1] += 1;
    }
  }
}
int main() {
  ll n, m;
  cin >> n >> m;
  for (ll i = 1; i <= n; i++) { cin >> a[i]; }
  for (ll i = 1; i <= m; i++) {
    cin >> L[i] >> R[i];
    for (ll j = L[i]; j <= R[i]; j++) { seg[j].push_back(i); }
  }
  for (ll i = 1; i <= n; i++) {
    for (ll j = 1; j <= m; j++) {
      if (L[j] <= i && R[j] >= i) { b[j] += a[i]; }
    }
  }
  dfs(1, n, 0);
  ll mx = 0;
  for (ll i = 1; i <= n; i++) { mx = max(mx, b[i]); }
  for (ll i = 1; i <= n; i++) { mx = min(mx, b[i]); }
  ans = 0;
  for (ll i = 1; i <= n; i++) { ans -= b[i]; }
  ans += 2 * mx;
  cout << ans << endl;
  for (ll i = 1; i <= m; i++) sum += (b[i] <= 0);
  cout << sum << endl;
  for (ll i = 1; i <= m; i++) {
    if (b[i] <= 0) { cout << i << ' '; }
  }
  return 0;
}
