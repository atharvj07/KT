import sys
import queue
r=sys.stdin.readline
for t in range(int(r())):
    n,a,b,da,db=map(int,r().split())
    vis=[False for _ in range(n+1)]
    vis[a]=True
    vis[b]=True
    g=[[] for _ in range(n+1)]
    for _ in range(n-1):
        u,v=map(int,r().split())
        g[u].append(v)
        g[v].append(u)
    q=queue.Queue()
    q.put((a,0))
    q.put((b,0))
    ans=False
    while not q.empty():
        u,d=q.get()
        if d<=da:
            for v in g[u]:
                if not vis[v]:
                    q.put((v,d+1))
                    vis[v]=True
                elif v==b:
                    ans=True
        if d<=db:
            for v in g[u]:
                if not vis[v]:
                    q.put((v,d+1))
                    vis[v]=True
                elif v==a:
                    ans=True
        if ans:
            break
    print("Alice" if ans else "Bob")
