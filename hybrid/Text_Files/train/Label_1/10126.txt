#include <bits/stdc++.h>
#define ll long long int
#define pb push_back
#define mp make_pair
#define pii pair<int,int>
#define fi first
#define se second
using namespace std;
const ll modn=998244353;
ll n,k,l[300001];
ll pre[300001],suf[300001];
int main()
{
    ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
    cin>>n>>k;
    for(int i=1;i<=n;i++)
    {
        ll x,y;
        cin>>x>>y;
        l[i]=x;
        pre[x]++; pre[y+1]--;
    }
    ll sum=0;
    for(int i=1;i<=300000;i++)
    {
        sum+=pre[i];
        suf[i]+=sum;
    }
    ll ans1=0,ans2=0;
    for(int i=1;i<=n;i++)
    {
        ans1+=suf[l[i]];
        ans1%=modn;
        ll x=l[i]+1,y=l[i]+k-1;
        if(y>300000) y=300000;
        ans2+=suf[y];
        ans2-=suf[x-1];
        ans2%=modn;
    }
    cout<<(ans1-ans2+modn)%modn<<endl;
    return 0;
}
