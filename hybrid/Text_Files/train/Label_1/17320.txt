#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 998244353;
int n, k;
ll p[200001];
ll dp[200001][201];
int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; i++) {
    cin >> p[i];
  }
  ll ans = 0;
  for (int i = 0; i <= n; i++) dp[i][0] = 0;
  for (int i = 1; i <= n; i++) {
    dp[i][1] = p[i];
  }
  for (int i = 2; i <= k; i++) {
    for (int j = i; j <= n; j++) {
      dp[j][i] = max(dp[j - 1][i], dp[j - 1][i - 1] + p[j]);
      ans = max(ans, dp[j][i]);
    }
  }
  int cnt = 0;
  for (int i = k; i <= n; i++) {
    if (dp[i][k] == ans) cnt++;
  }
  cout << ans << " " << cnt << "\n";
}
