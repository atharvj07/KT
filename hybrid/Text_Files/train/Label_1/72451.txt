from sys import stdin

n = int(stdin.readline())
a = list(map(int, stdin.readline().split()))
g = [0 for _ in range(n + 1)]
for i in range(1, n):
    g[i + 1] = g[i] + a[i]
ans = 5e18
for i in range(n):
    for j in range(i + 1, n):
        t = (i - j) ** 2 + (g[j] - g[i]) ** 2
        ans = min(t, ans)
print(int(ans))
