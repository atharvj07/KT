import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt(), w = sc.nextInt();
        int[][] a = new int[h][w + 1];
        int[] dp = new int[w + 1];
        int ans = 0;
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) a[i][j + 1] = a[i][j] + sc.nextInt();
        }
        for (int i = 0; i < h; i++) {
            for (int j = 0; j <= w; j++) {
                if (i > 0) dp[j] = Math.min(dp[j], a[i][j]);
                if (i == 0) dp[j] = a[i][j];
                ans = Math.max(ans, dp[j] * (j));
            }
        }
        System.out.println(ans);
    }
}
