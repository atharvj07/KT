import sys
input=sys.stdin.readline
n=int(input())
a=list(map(int,input().split()))
g=[[] for _ in range(n)]
for _ in range(n-1):
    u,c=map(int,input().split())
    g[u-1].append([c,u-1])
    g[u-1][-1][1]^=u-1
sa=-1
def dfs(v,p,ans,summ):
    global sa
    ans[v]=summ
    for x,y in g[v]:
        if y!=p:
            dfs(y,v,ans,summ+x)
    if not g[v]:
        if ans[v]>a[v]:
            sa=v
ans=[0]*n
dfs(0,-1,ans,0)
dfs(sa,-1,ans,0)
ans2=[0]*n
dfs(sa,-1,ans2,0)
ans3=0
for i in range(n):
    if (ans2[i]>a[i]):
        ans3+=1
print(ans3)
