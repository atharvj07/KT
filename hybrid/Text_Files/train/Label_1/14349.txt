#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
const ll oo = 1e18;
vector<pair<int, ll>> adj[N];
ll dist[N];
int par[N];
void dijkstra(int src, int n) {
    priority_queue<pair<ll, int>> pq;
    for (int i = 1; i <= n; i++) { dist[i] = oo; }
    pq.push({0, src});
    dist[src] = 0;
    while (!pq.empty()) {
        pair<ll, int> cur = pq.top();
        pq.pop();
        for (pair<int, ll> u : adj[cur.second]) {
            if (dist[u.first] > dist[cur.second] + u.second) {
                dist[u.first] = dist[cur.second] + u.second;
                pq.push({-dist[u.first], u.first});
                par[u.first] = cur.second;
            }
        }
    }
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        for (int i = 1; i <= n; i++) { adj[i].clear(); }
        for (int i = 0; i < 2 * n - 1; i++) {
            int a, b, c;
            cin >> a >> b >> c;
            adj[a].push_back({b, c});
            adj[b].push_back({a, c});
        }
        dijkstra(1, 2 * n);
        ll good = 0, bad = 0;
        for (int i = 1; i <= 2 * n; i++) {
            good += dist[i];
            swap(i, par[i]);
            bad += dist[i];
        }
        cout << good << " " << bad << endl;
    }
    return 0;
}
