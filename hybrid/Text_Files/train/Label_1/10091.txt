#include <iostream>
#include <unordered_set>
using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_set<int> watched;
  for (int i = 0; i < n - 1; i++) {
    int a;
    cin >> a;
    watched.insert(a);
  }
  for (int i = 1; i <= n; i++) {
    if (watched.find(i) == watched.end()) {
      cout << i << endl;
      return 0;
    }
  }
  return 0;
}
