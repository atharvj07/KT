#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int in[N];
int main() {
  int n, m;
  cin >> n >> m;
  int ans = 0;
  for (int i = 0; i < m; i++) {
    string s;
    int x;
    cin >> s >> x;
    if (s[0] == '+')
      in[x] = 1;
    else
      in[x] = 0;
  }
  for (int i = 1; i <= n; i++) ans += (in[i]);
  if (ans != n)
    cout << 0;
  else {
    cout << ans << endl;
    for (int i = 1; i <= n; i++) {
      if (in[i]) cout << i << " ";
    }
  }
}
