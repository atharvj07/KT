#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MOD = 1e9 + 7;
const int MAXN = 2e5 + 1;
int n;
vector<pair<int, int>> matryoshkas;
vector<vector<int>> min_space;
vector<int> dp;
int64_t recurse(int i, int prev) {
  if (i == n) { return 0; }
  if (dp[i] != -1) { return dp[i]; }
  int64_t ans = recurse(i + 1, prev);
  if (matryoshkas[i].second >= matryoshkas[prev].first) {
    ans = max(ans, recurse(i + 1, i) + matryoshkas[i].second - matryoshkas[prev].first);
  }
  return dp[i] = ans % MOD;
}
int main() {
  cin >> n;
  matryoshkas.resize(n);
  min_space.resize(n);
  dp.resize(n, -1);
  for (int i = 0; i < n; i++) { cin >> matryoshkas[i].first >> matryoshkas[i].second; }
  sort(matryoshkas.begin(), matryoshkas.end());
  cout << recurse(1, 0) << endl;
  return 0;
}
