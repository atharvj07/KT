import math

def factorize(n):
    factors = []
    for i in range(2, int(math.sqrt(n)) + 1):
        while n % i == 0:
            factors.append(i)
            n //= i
    if n > 1:
        factors.append(n)
    return factors

def solve(n):
    factors = factorize(n)
    k = len(factors)
    while k > 1 and factors[k - 1] == factors[k - 2]:
        k -= 1
    return k, factors[:k]

t = int(input())
for _ in range(t):
    n = int(input())
    k, factors = solve(n)
    print(k)
    print(*factors)
