#include <bits/stdc++.h>
using namespace std;
#define MIN(a, b) a < b ? a : b

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  string s;
  cin >> s;
  vector<bool> seen(n, false);
  vector<int> dp(n, -1);
  function<int(int)> solve = [&](int l) {
    if (seen[l]) return dp[l];
    seen[l] = true;
    if (l == n - 1) {
      dp[l] = a;
      return a;
    }
    if (n - 1 - l <= l) {
      dp[l] = a;
      return a;
    }
    int opt = INT_MAX;
    int i = l + 1;
    while (i < n) {
      opt = min(opt, solve(i));
      if (solve(i) != -1 && s.substr(i, n - i) == s.substr(l, n - i)) {
        opt = min(opt, solve(i) + b);
      }
      i++;
    }
    if (opt == INT_MAX) opt = a;
    dp[l] = opt;
    return opt;
  };
  cout << solve(0) << '\n';
  return 0;
}
