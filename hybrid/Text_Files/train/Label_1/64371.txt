n = int(input())
sherlock_card = sorted(input())
moriarty_card = sorted(input())

# Minimum flicks Moriarty will get
min_flicks = 0
used_indices = set()
for digit in sherlock_card:
    found = False
    for i, m_digit in enumerate(moriarty_card):
        if m_digit >= digit and i not in used_indices:
            used_indices.add(i)
            found = True
            break
    if not found:
        min_flicks += 1

# Maximum flicks Sherlock can get
max_flicks = 0
used_indices = set()
for digit in sherlock_card:
    found = False
    for i, m_digit in enumerate(moriarty_card):
        if m_digit > digit and i not in used_indices:
            used_indices.add(i)
            found = True
            break
    if found:
        max_flicks += 1

print(f"{min_flicks}\n{max_flicks}")
