#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
#define ll long long
vector<array<int,2>>ops;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, m;
  cin >> n >> m;
  vector<int> arr(n);
  for (auto &i: arr) cin >> i;
  int l, r;
  for (int i = 0; i < m; ++i) {
    cin >> l >> r;
    ops.push_back({l - 1, r - 1});
  }
  int ans = 0;
  for (auto &seg: ops) {
    ans = max(ans, arr[seg[1]] - arr[seg[0]]);
  }
  cout << ans << endl << ops.size() << endl;
  for (auto &seg: ops) {
    cout << seg[0] + 1 << " ";
  }
  cout << endl;
}
