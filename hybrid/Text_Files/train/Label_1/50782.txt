import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a = in.nextInt();
        int b = in.nextInt();
        int round = 1;
        while (n > 1) {
            if (a / 2 != b / 2) {
                System.out.print(round);
                return;
            }
            n /= 2;
            round++;
        }
        System.out.print("Final!");
    }
}
