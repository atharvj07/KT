#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int &x : a) cin >> x;
        long long sum = accumulate(a.begin(), a.end(), 0LL);
        long long avg = sum / n;
        for (int &x : a) { x -= avg; }
        sort(a.begin(), a.end());
        long long ans = 0;
        int l = 0, r = n - 1;
        while (l < r) {
            if (a[l] < 0) {
                if (-a[l] > a[r]) {
                    a[r] += a[l];
                    ans += -a[l];
                } else {
                    a[l] += a[r];
                    ans += a[r];
                }
                l++;
            } else {
                if (-a[l] > a[r]) {
                    a[r] += a[l];
                    ans += a[l];
                } else {
                    a[l] += a[r];
                    ans += -a[r];
                }
                r--;
            }
            if (l < r && a[l] == 0) l++;
            if (l < r && a[r] == 0) r--;
        }
        cout << ans << "\n";
    }
}
