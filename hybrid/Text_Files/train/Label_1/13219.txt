#include <bits/stdc++.h>
using namespace std;
int main() {
    int t, n, k;
    cin >> t;
    while (t--) {
        cin >> n >> k;
        string s(n, 'a');
        int pos = 0;
        for (int i = 0; i < k; i++) s[i] = (i + 'a');
        for (int i = k; i < n; i++) s[i] = s[pos], pos = (pos + 1) % k;
        cout << s << endl;
    }
}
