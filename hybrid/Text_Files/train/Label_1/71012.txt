t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().strip().split())
    a = list(map(int, input().strip().split()))
    
    s = 0
    for i in range(n):
        if i + 1 < n:
            diff = min(k + 1, 10**(a[i+1] - a[i]) - 1)
            s += diff * (10**a[i])
            k -= diff
        else:
            s += (k + 1) * (10**a[i])
            break
    print(s)
