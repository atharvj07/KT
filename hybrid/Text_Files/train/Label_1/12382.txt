#include <iostream>
#include <set>
#include <string>

using namespace std;

int main() {
    string n;
    cin >> n;

    set<string> possible;
    for (int i = 0; i < n.length(); i++) {
        string temp = n;
        possible.insert(temp);
        for (int j = i + 1; j < n.length(); j++) {
            swap(temp[i], temp[j]);
            possible.insert(temp);
        }
    }

    cout << possible.size() << endl;
    return 0;
}
