#include<bits/stdc++.h>
using namespace std;
const int N=300000+9;
int n,m,ans[N],mark[N];
int main(){
    cin>>n>>m;
    vector<int>vec(n);
    for(auto &x:vec)cin>>x;
    int l=0,r=0;
    for(int i=0;i<n;i++){
        if(mark[vec[i]]){
            ans[vec[i]]=-1;
            continue;
        }
        while(r<n&&vec[r]<=vec[i]){
            l=r++;
        }
        if(r==n)break;
        ans[vec[i]]=r-i-1;
        mark[vec[i]]=1;
    }
    int fin=INT_MAX;
    for(int i=0;i<m;i++){
        if(ans[i]==-1)continue;
        fin=min(fin,ans[i]);
    }
    if(fin==INT_MAX)cout<<0<<endl;
    else cout<<fin<<endl;
}
