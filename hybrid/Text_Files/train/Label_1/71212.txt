import sys

n = int(sys.stdin.readline())
board = [['.' for i in range(n)] for j in range(n)]
cnt = 0

def check(x, y):
    for i in range(x-2, x+3):
        for j in range(y-2, y+3):
            if i>=0 and i<n and j>=0 and j<n and abs(i-x) + abs(j-y) in [1,3]:
                return False
    return True

for i in range(n):
    for j in range(n):
        if check(i, j):
            board[i][j] = 'W'
            cnt += 1

for i in range(n):
    for j in range(n):
        if i-cnt>=0 and check(i-cnt, j):
            board[i-cnt][j] = 'B'

for i in range(n):
    print(''.join(board[i]))
