#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define vll vector<ll>

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin >> n;
    vll l(n), time(n, LLONG_MAX);
    string s;
    cin >> s;
    for (int i = 0; i < n; ++i) cin >> l[i];
    if (s[0] == 'W') time[0] = 3 * l[0];
    else
        time[0] = 5 * l[0];

    for (int i = 1; i < n; ++i) {
        if (s[i] == 'W') {
            time[i] = min({5 * l[i] + time[i - 1], 3 * l[i] + time[i - 1], time[i - 1] + 1 + 3 * (l[i] - 1)});
            if (s[i - 1] == 'L' && i != 1) time[i] = min({time[i], time[i - 2] + 1 + 3 * l[i]});
            else if (s[i - 1] == 'L' && i == 1)
                time[i] = min({time[i], 1 + 3 * l[i]});
            if (s[i - 1] == 'G') time[i] = min({time[i], time[i - 1] + 3 * l[i]});
        } else if (s[i] == 'G') {
            time[i] = min({5 * l[i] + time[i - 1], time[i - 1] + 1 + 5 * (l[i] - 1)});
            if (s[i - 1] == 'L' && i != 1) time[i] = min({time[i], time[i - 2] + 1 + 5 * l[i]});
            else if (s[i - 1] == 'L' && i == 1)
                time[i] = min({time[i], 1 + 5 * l[i]});
            if (s[i - 1] == 'W') time[i] = min({time[i], time[i - 1] + 5 * l[i]});
        } else {
            time[i] = min({time[i - 1] + 1, time[i]});
            if (s[i - 1] == 'G') time[i] = min({time[i], time[i - 1] + 1 + l[i]});
            else if (s[i - 1] == 'W') time[i] = min({time[i], time[i - 1] + 1 + 3 * l[i]});
        }
    }

    cout << time[n - 1] << endl;
    return 0;
}
