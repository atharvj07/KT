#include<iostream>

using namespace std;

int main() {
    int n, m, a[10000010];
    cin >> n >> m;
    a[1] = 1;
    a[2] = 2;
    for(int i = 3; i <= 10000000; i++) {
        a[i] = (a[i - 1] + a[i - 2]) % m;
        if(i == n) {
            cout << a[i] << endl;
            return 0;
        }
    }
    cout << "-1" << endl;
    return 0;
}
