def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

n = int(input())
if n == 4:
    print("NO")
elif n % 2 == 0 and n != 2:
    print("YES")
    print(1)
    print(1, n // 2)
else:
    if n % 3 == 0 and n // 3 != 1:
        print("YES")
        print(2)
        print(2, n // 3)
        print(1, n // 3)
    else:
        print("NO")
