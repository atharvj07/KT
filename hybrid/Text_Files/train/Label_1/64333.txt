n = int(input())
a = list(map(int, input().split()))

def is_valid(d):
    positives = sum(1 for x in a if x / d > 0)
    return positives >= -(-n // 2)

for d in range(1, 1001):
    if is_valid(d):
        print(d)
        break
    if is_valid(-d):
        print(-d)
        break
else:
    print(0)
