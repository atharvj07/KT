#include <iostream>

using namespace std;

int main() {
  bool isProper = true;
  char c, prv;
  prv = cin.get();
  for (int i = 0; i < 7; ++i) {
    for (int j = 0; j < 8; ++j) {
      c = cin.get();
      isProper &= ((prv == 'B' && c == 'W') || (prv == 'W' && c == 'B'));
      prv = c;
    }
  }
  cout << (isProper ? "YES" : "NO") << endl;
  return 0;
}
