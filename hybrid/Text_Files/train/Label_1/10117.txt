#include <iostream>

using namespace std;

int main() {
  long long a, b, c, d;
  cin >> a >> b >> c >> d;
  long double p1 = (long double)a / b;
  long double p2 = (long double)c / d;
  long double p1_win = p1 / (p1 + p2);
  long double p2_win = p2 / (p1 + p2);
  long double p1_win_first = 1;
  while (true) {
    long double temp = p1_win_first;
    p1_win_first = p1_win * p1_win_first + p2_win * (1 - p1_win_first);
    if (temp == p1_win_first) break;
  }
  cout << p1_win_first << endl;
  return 0;
}
