#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<int> dp(n);
    dp[n-1] = a[n-1];
    for (int i = n-2; i >= 0; i--) {
        if (a[i] > dp[i+1]) {
            dp[i] = a[i];
        } else {
            dp[i] = dp[i+1];
        }
    }
    for (int i = 0; i < n; i++) {
        cout << dp[i] << " ";
    }
    cout << endl;
    return 0;
}
