#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n), c(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  for (int i = 0; i < n; i++) c[i] = (a[i] + b[i] % n) % n;
  sort(c.begin(), c.end());
  sort(b.begin(), b.end());
  for (int i = 0; i < n; i++) cout << (c[i] - a[i] + n) % n << " ";
  return 0;
}
