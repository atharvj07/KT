#include <bits/stdc++.h>
using namespace std;
int main(){
    int x; cin>>x; int h,m; cin>>h>>m;
    int ans=0, hh=h, mm=m;
    while(true){
        if(hh<0) hh+=24;
        if(hh>23) hh-=24;
        if(hh==h && mm==m) break;
        ans++;
        hh-=x/60, mm-=x%60;
        if(mm<0) mm+=60;
    }
    cout<<ans;
    return 0;
}
