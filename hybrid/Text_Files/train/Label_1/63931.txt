n, m, i, j, a, b = map(int, input().split())

def min_moves(n, m, i, j, a, b):
    if (i - 1) % a != 0 or (j - 1) % b != 0 or (n - i) % a != 0 or (m - j) % b != 0:
        return "Poor Inna and pony!"
    if (n == a and (i == 1 or i == n)) or (m == b and (j == 1 or j == m)):
        return "Poor Inna and pony!"
    if (abs(i - 1) // a) % 2 != (abs(j - 1) // b) % 2 or (abs(n - i) // a) % 2 != (abs(m - j) // b) % 2:
        return "Poor Inna and pony!"

    moves = []
    if (i - a > 0 and j - b > 0) or (i + a <= n and j + b <= m):
        moves.append(max((abs(i - 1) + a - 1) // a, (abs(j - 1) + b - 1) // b))
    if (i - a > 0 and j + b <= m) or (i + a <= n and j - b > 0):
        moves.append(max((abs(n - i) + a - 1) // a, (abs(m - j) + b - 1) // b))
    
    if moves:
        return min(moves)
    else:
        return "Poor Inna and pony!"

print(min_moves(n, m, i, j, a, b))
