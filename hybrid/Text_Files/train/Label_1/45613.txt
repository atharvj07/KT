import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        int[] deg = new int[n + 1];
        List<Integer>[] adj = new List[n + 1];
        for (int i = 0; i < n + 1; i++) {
            adj[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            deg[a]++;
            deg[b]++;
            adj[a].add(b);
            adj[b].add(a);
        }

        if (n % 2 == 1) {
            System.out.println(-1);
            return;
        }

        int ans = n / 2;
        for (int i = 1; i <= n; i++) {
            if (deg[i] % 2 == 1) {
                ans--;
            }
        }

        System.out.println(ans);
    }
}
