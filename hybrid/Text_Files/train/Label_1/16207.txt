#include <bits/stdc++.h>
using namespace std;
#define rep(i, a, n) for (int i = a; i < n; i++)
#define ll long long
#define pb push_back
#define all(a) a.begin(), a.end()

const int N = 3e5 + 10;
vector<vector<int>> adj;
int up[N][20], col[N];
vector<int> dfs(int node, int par = -1, int c = 0) {
    col[node] = c;
    vector<int> ans(2);
    ans[c]++;
    for (auto adj : adj[node]) {
        if (adj != par) {
            auto a = dfs(adj, node, c ^ 1);
            ans[0] += a[0];
            ans[1] += a[1];
            up[adj][0] = node;
        }
    }
    return ans;
}
int get(int u, int v) {
    int ans = 0;
    if (col[u] == 0) {
        swap(u, v);
    }
    for (int j = 19; j >= 0; j--) {
        if ((1 << j) <= up[v][0] && col[up[v][j]] == 1) {
            ans += (1 << j);
            v = up[v][j];
        }
    }
    return ans + (col[v] == 0);
}
