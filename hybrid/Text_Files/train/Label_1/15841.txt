#include<bits/stdc++.h>
using namespace std;
vector<vector<int>>e,ne;
int n,ans,vis[100002]={},cnt[100002]={},lvl[100002]={};
void dfs(int x,int p){
    for(auto y:e[x]){
        if(y!=p){
            ne[x].push_back(y);
            dfs(y,x);
            cnt[x]+=cnt[y];
        }
    }
    if(cnt[x]==0)cnt[x]=1;
    cnt[x]+=lvl[x];
}
void dia(int x,int p,int xx){
    lvl[x]=xx;
    int mx=0,idx=-1;
    for(auto y:ne[x]){
        if(y!=p && cnt[y]>mx){
            mx=cnt[y];
            idx=y;
        }
    }
    if(idx!=-1){
        dia(idx,x,xx+1);
        ans+=mx;
        for(auto y:ne[x]){
            if(y==p || y==idx)continue;
            dia(y,x,1);
        }
    }
}
int main(){
    int i,x,y;
    cin>>n;
    e.resize(n+1);
    ne.resize(n+1);
    for(i=1;i<n;i++){
        cin>>x>>y;
        e[x].push_back(y);
        e[y].push_back(x);
    }
    vis[1]=1;
    dfs(1,1);
    dia(1,1,0);
    cout<<ans<<"\n";
    return 0;
}
