#include<bits/stdc++.h>
using namespace std;
#define ll long long
const double EPS=1e-12;
const double PI=acos(-1.0);
int main() {
    int n,m;
    double ans=0;
    cin>>n>>m;
    int P[n],inv[n];
    for (int i=0; i<n; i++) {P[i]=i;cin>>inv[i];}
    sort(inv,inv+n);
    for (int i=0; i<m; i++) {
        int x,y;cin>>x>>y;x--,y--;
        swap(P[x],P[y]);
    }
    for (int i=0; i<n; i++) {
        ans+=(double)(abs(i-inv[P[i]]));
    }
    cout<<fixed<<setprecision(9)<<ans<<endl;
    return 0;
}
