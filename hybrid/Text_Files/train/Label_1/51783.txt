import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] costs = { "100", "200", "300" };

        String s = sc.next();

        int result = 700;
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == 'o') {
                result += Integer.parseInt(costs[i]);
            }
        }

        System.out.println(result);
    }
}
