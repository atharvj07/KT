#include <bits/stdc++.h>
using namespace std;
#define ll long long
struct point { ll x, v; };

ll dist(const point &a, const point &b) { return abs(a.x - b.x); }

int main() {
  ll n;
  cin >> n;
  vector<point> p(n);
  for (ll i = 0; i < n; i++) cin >> p[i].x;
  for (ll i = 0; i < n; i++) cin >> p[i].v;
  sort(p.begin(), p.end(), [](const point &a, const point &b) { return a.x * b.v < a.v * b.x; });
  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    const point &a = p[i];
    for (ll j = 0; j < n; j++) {
      const point &b = p[j];
      if (b.x * a.v < b.v * a.x) break;
      ans += dist(a, b);
    }
  }
  cout << ans << endl;
  return 0;
}
