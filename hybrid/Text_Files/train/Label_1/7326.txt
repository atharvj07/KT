#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    if (n % 2 == 1 || s[0] == '0' || s[n - 1] == '0')
      cout << "NO\n";
    else {
      string a, b;
      int open = 0, close = 0;
      for (char c : s) {
        if (c == '0') {
          if (open == 0) {
            cout << "NO\n";
            return 0;
          }
          a += ')';
          b += '(';
          open--;
          close++;
        } else {
          a += '(';
          b += '(';
          open++;
        }
      }
      for (int i = 0; i < close; i++) a += ')';
      cout << "YES\n" << a << "\n" << b << "\n";
    }
  }
}
