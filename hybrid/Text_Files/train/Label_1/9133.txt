#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> t(n);
    for (int &ti : t) cin >> ti;
    int d = t[1] - t[0];
    bool is_ap = true;
    for (int i = 2; i < n; i++) {
        is_ap &= t[i] - t[i - 1] == d;
    }
    if (is_ap) cout << t[n - 1] + d << endl;
    else cout << t[n - 1] << endl;
    return 0;
}
