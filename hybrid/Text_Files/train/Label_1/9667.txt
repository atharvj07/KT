#include <bits/stdc++.h>
using namespace std;
int n, m, k, g[10005], t, r;
bool v[10005];
int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= k; i++) { cin >> g[i]; }
  while (m--) {
    int x, y;
    cin >> x >> y;
    if (!v[x] && !v[y]) {
      t++;
      v[x] = v[y] = 1;
    }
  }
  cout << n - 1 - t << endl;
  return 0;
}
