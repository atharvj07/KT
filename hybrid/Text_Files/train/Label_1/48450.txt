import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            String s = input.next();
            int oddCount = 0;
            int evenCount = 0;
            for (int i = 0; i < s.length(); i++) {
                if (s.charAt(i) - '0' % 2 == 0) evenCount++;
                else oddCount++;
            }
            if (oddCount % 2 == 0) System.out.println("DA");
            else if (evenCount % 2 == 0) System.out.println("DA");
            else System.out.println("NET");
        }
    }
}
