import java.util.*;
import java.math.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int K = sc.nextInt();
        long[] A = new long[N + 1];
        for (int i = 1; i <= N; i++) A[i] = sc.nextLong();
        long[][] dp = new long[N + 1][K + 1];
        for (int kind = 1; kind <= K; kind++) {
            dp[1][kind] = Math.max(dp[1][kind - 1], A[1]);
        }
        for (int i = 2; i <= N; i++) {
            for (int kind = 1; kind <= K; kind++) {
                dp[i][kind] = Math.max(dp[i - 1][kind], A[i] + dp[i - 2][kind - 1]);
            }
        }
        System.out.println(dp[N][K]);
    }
}
