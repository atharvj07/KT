def max_pillows(n, m, k):
    left = max(k - 1, n - k)
    right = min(k - 1, n - k)
    lo, hi = 1, m
    while lo < hi:
        mid = (lo + hi + 1) // 2
        if mid <= left + 1:
            total = (mid * (mid + 1)) // 2 + ((mid - 1) * mid) // 2 + (n - 2 * mid + 1)
        else:
            total = ((2 * mid - left - 1) * (left + 1)) // 2 + ((2 * mid - right - 1) * (right + 1)) // 2
        if total <= m:
            lo = mid
        else:
            hi = mid - 1
    return lo

n, m, k = map(int, input().split())
print(max_pillows(n, m, k))
