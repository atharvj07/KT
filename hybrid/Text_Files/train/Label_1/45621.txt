import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int min = 0;
        int max = 0;
        for (int i = 0; i < n; i++) {
            min = Math.max(min, (a[i] + m + n - 1) / n);
            max += a[i];
        }
        max = Math.max(max + m, min);
        System.out.println(min + " " + max);
    }
}
