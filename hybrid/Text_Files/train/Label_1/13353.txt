#include <bits/stdc++.h>
using namespace std;
vector<int> get(vector<int>&a, vector<int>&b) {
  int n = a.size(), m = b.size(), i = 0, j = 0;
  vector<int>ret;
  while (i < n && j < m) {
    if (a[i] == b[j]) ret.push_back(a[i]), ++i, ++j;
    else a[i] < b[j] ? ++i : ++j;
  }
  return ret;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<int>a(n), b(m);
    for (int& x : a) cin >> x;
    for (int& x : b) cin >> x;
    vector<int> c = get(a, b);
    cout << (c.empty() ? "NO" : "YES") << endl;
    if (!c.empty()) {
      cout << c.size() << endl;
      for (int& x : c) cout << x << " ";
      cout << endl;
    }
  }
}
