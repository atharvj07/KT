#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, t;
  cin >> n >> t;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int days = (t + 86400 - 1) / 86400;
  while (days > 0) {
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      cnt += min(86400, t);
      t -= min(86400, t);
      t += 86400 - a[i];
    }
    if (cnt <= 0) break;
    days--;
  }
  cout << days + 1 << endl;
  return 0;
}
