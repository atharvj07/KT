for _ in range(int(input())):
    n, a, b = list(map(int, input().split()))
    pattern = ""
    i = 0
    while i < b:
        pattern += chr(ord('a') + i)
        i += 1
    temp = len(pattern)
    while temp < a:
        pattern += pattern[temp - b:temp]
        temp = len(pattern)
    string = pattern
    while len(string) < n:
        string += pattern
    print(string[:n])
