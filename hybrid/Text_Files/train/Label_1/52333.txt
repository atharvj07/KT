import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[][] songs = new int[n][2];
        for (int i = 0; i < n; i++) {
            songs[i][0] = scanner.nextInt();
            songs[i][1] = scanner.nextInt();
        }
        Arrays.sort(songs, (a, b) -> {
            if (a[0] == b[0]) {
                return Integer.compare(b[1], a[1]);
            } else {
                return Integer.compare(a[0], b[0]);
            }
        });
        long maxPleasure = 0;
        int minBeauty = Integer.MAX_VALUE;
        long totalLength = 0;
        for (int i = 0; i < Math.min(k, n); i++) {
            totalLength += songs[i][0];
            minBeauty = Math.min(minBeauty, songs[i][1]);
        }
        maxPleasure = totalLength * minBeauty;
        System.out.println(maxPleasure);
    }
}
