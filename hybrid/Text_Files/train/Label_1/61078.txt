n, w = map(int, input().split())
arr = []
for i in range(n):
    l, *a = list(map(int, input().split()))
    arr.append((l, a))
ans = [0 for i in range(w)]
pref = [[0 for i in range(w + 1)] for i in range(n + 1)]
for i in range(1, n + 1):
    l, a = arr[i - 1]
    for j in range(1, w + 1):
        pref[i][j] = pref[i - 1][j]
    for j in range(l):
        if j + 1 <= w:
            pref[i][j + 1] = max(pref[i][j + 1], pref[i - 1][max(0, j + 1 - l)] + a[j])
for i in range(w):
    ans[i] = max(pref[j][i + 1] for j in range(n + 1))
print(*ans)
