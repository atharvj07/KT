m9=998244353
def pow(a,b):
    if(b==0):return 1
    if(b==1):return a
    if(b%2==0):
        val=pow(a,b/2)
        val=val*val
        val%=m9
        return val
    else:
        val=pow(a,b-1)
        val=val*a
        val%=m9
        return val
def inv(x):
    return pow(x,m9-2)

n,m=map(int,input().split())
l=[]
for i in range(n):
    l.append(list(map(int,input().split())))
a=1
b=1
for i in range(n):
    for j in range(m):
        if(l[i][j]<=n):
            a=a*l[i][j]
            b=b*(n+1-l[i][j])
            a%=m9
            b%=m9
print((a*inv(b))%m9)
