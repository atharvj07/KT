import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int n = s.length();
        int ans = n;
        for (int i = 1; i <= n / 2; i++) {
            StringBuilder t = new StringBuilder();
            int cnt = 0;
            for (int j = 0; j < n; j += i) {
                String cur = s.substring(j, Math.min(j + i, n));
                if (cnt == 0 || t.toString().equals(cur)) {
                    cnt++;
                } else {
                    t.append(cnt).append(cur);
                    cnt = 1;
                }
            }
            t.append(cnt).append(s.substring(n - n % i, n));
            ans = Math.min(ans, t.length());
        }
        System.out.println(ans);
    }
}
