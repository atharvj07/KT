#include <cstdio>
#include <algorithm>
#include <vector>
using namespace std;
int t, n, m;
int main(){
    scanf("%d", &t);
    for(int _ = 1; _ <= t; ++_){
        scanf("%d %d", &n, &m);
        vector<int> a[n + 1];
        for(int i = 0; i < m; ++i){
            int x, y;
            scanf("%d %d", &x, &y);
            a[x].push_back(y), a[y].push_back(x);
        }
        vector<int> e;
        for(int i = 1; i <= n; ++i){
            if(a[i].size() % 2 == 1) e.push_back(i);
        }
        for(int i = 1; i < e.size(); i += 2){
            a[e[i]].push_back(e[i - 1]), a[e[i - 1]].push_back(e[i]);
        }
        printf("%d\n", n - e.size());
        for(int i = 1; i <= n; ++i){
            for(int j = 1; j < a[i].size(); ++j){
                printf("%d %d\n", i, a[i][j]);
            }
        }
    }
    return 0;
}
