#include <bits/stdc++.h>
using namespace std;
const int mxn = 5e2 + 10;
char a[mxn][mxn];
int dis[mxn][mxn];
bool vis[mxn][mxn];
int n, m, k;
int dx[] = {0, 1, 0, -1, 1, 1, -1, -1};
int dy[] = {1, 0, -1, 0, 1, -1, 1, -1};
void bfs(int x, int y)
{
    queue<pair<int, int>> q;
    q.push({x, y});
    memset(vis, 0, sizeof vis);
    vis[x][y] = true;
    while (!q.empty())
    {
        pair<int, int> t = q.front();
        q.pop();
        for (int i = 0; i < 8; i++)
        {
            int xx = t.first + dx[i];
            int yy = t.second + dy[i];
            if (xx > 0 && yy > 0 && xx <= n && yy <= m && vis[xx][yy] == false)
            {
                vis[xx][yy] = true;
                dis[xx][yy] = dis[t.first][t.second] + 1;
                q.push({xx, yy});
            }
        }
    }
}
int main()
{
    cin >> n >> m >> k;
    int x, y;
    for (int i = 1; i <= k; i++)
    {
        cin >> x >> y;
        a[x][y] = '*';
    }
    int mx = -1;
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= m; j++)
        {
            if (a[i][j] != '*')
            {
                bfs(i, j);
                for (int p = 1; p <= n; p++)
                {
                    for (int q = 1; q <= m; q++)
                    {
                        if (a[p][q] == '*')
                            continue;
                        mx = max(mx, dis[p][q]);
                    }
                }
            }
        }
    }
    cout << mx << endl;
    return 0;
}
