#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;

    vector<int> log(n);
    for (int i = 0; i < n; i++) {
        cin >> log[i];
    }

    int alex = 0, bob = 1, carl = 2;
    bool possible = true;
    for (int i = 0; i < n; i++) {
        if (log[i] == carl) {
            possible = false;
            break;
        }
        if (log[i] == bob) {
            swap(alex, bob);
        } else {
            swap(alex, carl);
        }
    }

    cout << (possible ? "YES" : "NO") << endl;

    return 0;
}
