import sys

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort(key=lambda x: (-x, x))  # Sort the array in decreasing order of (a_i, i)
    cur = 0
    for i in range(n):
        cur += a[i] / (i + 1)
    if cur == m:
        print("YES")
    else:
        print("NO")
