import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] edges = new int[m][2];
        for (int i = 0; i < m; i++) {
            edges[i][0] = scanner.nextInt();
            edges[i][1] = scanner.nextInt();
        }
        boolean[] visited = new boolean[n + 1];
        ArrayList<Integer>[] components = new ArrayList[n + 1];
        int componentCount = 0;
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                components[componentCount] = new ArrayList<>();
                dfs(i, visited, components[componentCount], edges);
                componentCount++;
            }
        }
        System.out.println(componentCount);
        for (int i = 0; i < componentCount; i++) {
            System.out.print(components[i].size() + " ");
        }
    }

    public static void dfs(int node, boolean[] visited, ArrayList<Integer> component, int[][] edges) {
        visited[node] = true;
        component.add(node);
        for (int[] edge : edges) {
            if (edge[0] == node && !visited[edge[1]]) {
                dfs(edge[1], visited, component, edges);
            } else if (edge[1] == node && !visited[edge[0]]) {
                dfs(edge[0], visited, component, edges);
            }
        }
    }
}
