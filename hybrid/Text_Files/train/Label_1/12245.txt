#include<bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 10;
typedef long long ll;

int n, k;
ll h[MAXN];

int main() {
    cin >> n >> k;
    for (int i = 0; i < n; ++i) cin >> h[i];

    sort(h, h + n);
    ll ans = 0;
    for (int i = 1; i < n; ++i) if (h[i] - h[0] > k) ans += h[i] - h[0] - k;
    cout << ans << endl;
    return 0;
}
