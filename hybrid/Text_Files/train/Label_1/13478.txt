#include <algorithm>
#include <cmath>
#include <cstdio>
#include <vector>

using namespace std;

#define all(x) (x).begin(), (x).end()

using ll = long long;
using ld = long double;

const ld eps = 1e-9;

struct pt {
    vector<ll> x;
    pt() = default;
    pt(vector<ll> x_) : x(x_) {}
    ll len() {
        ll sum = 0;
        for (ll i : x) sum += i * i;
        return sum;
    }
    ld dot(const pt &rhs) {
        ld s = 0;
        for (int i = 0; i < 5; i++) s += x[i] * rhs.x[i];
        return s;
    }
    ld dist(const pt &rhs) {
        return sqrt((double)(dot(rhs)));
    }
};
int main() {
    int n;
    scanf("%d", &n);
    vector<pt> pts(n);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 5; j++) {
            ll x;
            scanf("%lld", &x);
            pts[i].x.push_back(x);
        }
    }
    vector<int> ans;
    for (int i = 0; i < n; i++) {
        bool good = true;
        for (int j = 0; j < n; j++) {
            for (int k = j + 1; k < n; k++) {
                if (i == j || i == k || j == k) continue;
                ld x = (pts[i].dot(pts[j]) - pts[i].dot(pts[k])) / (pts[i].dist(pts[j]) * pts[i].dist(pts[k]) - eps);
                if (abs(x) <= 1) good = false;
            }
        }
        if (good) ans.push_back(i);
    }
    printf("%d\n", (int)ans.size());
    for (int x : ans) printf("%d\n", x);
}
