#include<bits/stdc++.h>
using namespace std;
int T,n,sum[200010],a[200010],flag;
vector<pair<int,int>>ans;
int main(){
    cin>>T;
    while(T--){
        cin>>n;
        memset(sum,0,sizeof sum);
        for(int i=0;i<n;i++){
            cin>>a[i];
            sum[a[i]]++;
        }
        flag=0;
        ans.clear();
        for(int i=1;i<=n-1;i++){
            if(sum[i]==0) flag=1;
            if(sum[i]%2==1){
                int p1=(i+1)/2;
                int p2=(i-1)/2;
                if(sum[p1]%2==1&&sum[p2]%2==1&&sum[p1]>0&&sum[p2]>0) ans.push_back(make_pair(p1,p2));
            }
        }
        if(sum[0]%2==1||flag==1){
            cout<<0<<endl;
            continue;
        }
        int p1=0,p2=0;
        for(int i=1;i<=n-1;i++){
            if(sum[i]%2==0) p1=i;
            if(sum[i]>0) p2=i;
        }
        cout<<ans.size()<<endl;
        for(int i=0;i<ans.size();i++){
            cout<<ans[i].first<<" "<<ans[i].second<<endl;
        }
    }
    return 0;
}
