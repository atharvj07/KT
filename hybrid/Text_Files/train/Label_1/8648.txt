#include <iostream>
#include <vector>
using namespace std;

typedef unsigned long long ll;

const ll MOD = 1000000007;

ll gcd(ll a, ll b) {
  while (b) {
    a %= b;
    swap(a, b);
  }
  return a;
}

int main() {
  ll n, x;
  cin >> n >> x;
  vector<ll> a(n);
  for (auto &x : a) cin >> x;
  ll sum = 0;
  for (auto &x : a) sum += x;
  ll num = 1;
  ll deno = 1;
  for (auto &x : a) num *= (num * x) % MOD;
  for (ll i = 0; i < sum; i++) deno = (deno * x) % MOD;
  ll g = gcd(num, deno);
  cout << g % MOD << endl;
  return 0;
}
