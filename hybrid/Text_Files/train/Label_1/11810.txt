#include<bits/stdc++.h>
using namespace std;
#define ll long long int

int main(){
    int n;
    cin>>n;
    vector<int>x(n+1);
    vector<vector<ll>>adj(n+1);
    for(int i=1;i<=n;++i){
        cin>>x[i];
        if(x[i]!=-1) adj[x[i]].push_back(i);
    }
    ll a=0,b=0;
    function<void(int,int)>dfs=[&](int node,int par){
        vector<ll>v;
        for(auto i:adj[node]){
            if(i==par) continue;
            dfs(i,node);
            v.push_back(adj[i].size());
        }
        if(node!=1){
            ll p=1;
            for(auto i:v) p*=i+1;
            a+=p;
        }
        if(v.size()){
            sort(v.begin(),v.end());
            ll p=1,q=1,s=0;
            for(auto i:v){
                p*=i+1;
                q*=i;
                s+=i;
            }
            b+=p/(q*s);
        }
    };
    dfs(1,1);
    cout<<(a+b)%((ll)1e9+7)<<"\n";
    return 0;
}
