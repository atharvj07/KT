from sys import stdin

n,a,b=map(int,stdin.readline().split())
G=[(int(x),int(vx),int(vy)) for x,vx,vy in map(lambda x:x.split(),stdin)]
G.sort()
EX=0
for x,vx,vy in G:
  l=x+vx-a*vy
  r=x+vx-a*vy+1
  p=[]
  for x2,vx2,vy2 in G:
    p.append((x2+vx2-a*vy2,x2+vx2-a*vy2+1))
    if x2 < x and x2 + vx2 <= x + vx:
      EX+=1
    elif x2 > x and x + vx <= x2 + vx2:
      EX+=1
    elif x2 > x and l <= x2 + vx2 and x2 + vx2 <= r:
      EX+=1
    elif x2 < x and l <= x2 + vx2 and x2 + vx2 <= r:
      EX+=1
    elif x2 == x:
      if vy*vx2 > vx*vy2: EX+=1
  p.sort()
  L=l
  for PL,PR in p:
    if PL <= L < PR:
      EX+=1
    else:
      L=PR
print(EX)
