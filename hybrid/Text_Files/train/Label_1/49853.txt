import java.util.Scanner;

public class ErrorMinimization {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k1 = scanner.nextInt();
        int k2 = scanner.nextInt();

        int[] a = new int[n];
        int[] b = new int[n];

        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }

        int minError = solve(a, b, k1, k2);
        System.out.println(minError);
    }

    private static int solve(int[] a, int[] b, int k1, int k2) {
        int sumA = 0;
        int sumB = 0;
        for (int i = 0; i < a.length; i++) {
            sumA += a[i];
            sumB += b[i];
        }

        int targetA = (sumA + k1) / (a.length + k1 + k2);
        int targetB = (sumB + k2) / (a.length + k1 + k2);

        int error = 0;
        for (int i = 0; i < a.length; i++) {
            error += Math.abs(a[i] - targetA) + Math.abs(b[i] - targetB);
        }

        return error;
    }
}
