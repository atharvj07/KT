import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int K = sc.nextInt();
        int[] A = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextInt();
        }
        Arrays.sort(A);

        int min = N;
        for (int i = 0; i < N - K + 1; i++) {
            min = Math.min(min, A[i + K - 1] - A[i] + 1);
        }
        System.out.println(min);
    }
}
