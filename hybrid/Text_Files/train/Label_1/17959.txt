#include <iostream>

int f[300][300], MOD = 1e9 + 7;

int main() {
  int n;
  std::cin >> n;

  f[0][0] = 1;
  for (int i = 1; i <= 2 * n - 2; i++) {
    for (int j = 0; j <= n - 1; j++) {
      f[i][j] = (f[i - 1][j] * 2) % MOD;
      if (j) f[i][j] = (f[i][j] + f[i - 1][j - 1]) % MOD;
    }
  }

  std::cout << f[2 * n - 2][n - 1] << '\n';

  return 0;
}
