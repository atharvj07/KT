n, m = map(int, input().split())
a = [input() for _ in range(n)]

lamp_counts = [0] * m
for row in a:
    for j, val in enumerate(row):
        if val == '1':
            lamp_counts[j] += 1

for row in a:
    if all(lamp_counts[j] > 1 if val == '1' else True for j, val in enumerate(row)):
        print("YES")
        break
else:
    print("NO")
