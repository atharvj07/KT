#include <bits/stdc++.h>
#include <vector>
using namespace std;

int n, p[100005];
vector<int> adj[100005];
int leafCnt, mxNode, dp[100005], pre[100005];
bool flag;

void dfs1(int u, int par) {
    for (auto v : adj[u]) {
        if (v == par)
            continue;
        dfs1(v, u);
        leafCnt += (dp[v] == 1);
        mxNode = max(mxNode, dp[v]);
    }
    if (leafCnt == 0) {
        dp[u] = 1;
        pre[u] = 0;
    } else if (leafCnt == 1) {
        dp[u] = 2;
        pre[u] = 1;
    } else {
        dp[u] = mxNode + 1;
        pre[u] = mxNode;
    }
    leafCnt = 0;
    mxNode = 0;
}

void dfs2(int u, int par) {
    for (auto v : adj[u]) {
        if (v == par)
            continue;
        int newPre = (par != 1 ? max(pre[u], pre[v] + 1) : pre[v] + 1);
        int newDp = (newPre < dp[u] ? dp[u] : newPre);
        int newLeaf = leafCnt;
        if (dp[v] == 1)
            newLeaf++;
        dfs2(v, u);
        if (leafCnt == newLeaf) {
            dp[u] = newDp;
            pre[u] = newPre;
        } else {
            dp[u] = mxNode + 1;
            pre[u] = mxNode;
        }
        leafCnt = 0;
        mxNode = 0;
    }
    leafCnt = 0;
    mxNode = 0;
}

void solve() {
    cin >> n;
    for (int i = 2; i <= n; ++i) {
        cin >> p[i];
        adj[p[i]].push_back(i);
        adj[i].push_back(p[i]);
    }
    leafCnt = mxNode = 0;
    dfs1(1, 1);
    leafCnt = mxNode = 0;
    dfs2(1, 1);
    for (int i = 1; i <= n; ++i)
        cout << dp[i] << " ";
}

int main() {
    solve();
    return 0;
}
