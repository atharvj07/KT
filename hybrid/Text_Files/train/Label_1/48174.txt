import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long d = scanner.nextLong();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextLong();
        }

        long sum = 0;
        int ans = 0;
        for (long ai : a) {
            sum += ai;
            if (sum < 0) {
                ans = -1;
                break;
            } else if (sum > d) {
                ans++;
                sum = ai;
            }
        }

        System.out.println(ans);
    }
}
