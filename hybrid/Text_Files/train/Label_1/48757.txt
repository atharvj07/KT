import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            b[i] = in.nextInt();
        }
        System.out.println(isPossible(a, b) ? "YES" : "NO");
    }

    public static boolean isPossible(long[] a, long[] b) {
        long sum = 0;
        for (int i = 0; i < a.length; i++) {
            sum += a[i];
        }
        long max1 = 0;
        long max2 = 0;
        for (int i = 0; i < b.length; i++) {
            if (b[i] > max1) {
                max2 = max1;
                max1 = b[i];
            } else if (b[i] > max2) {
                max2 = b[i];
            }
        }
        return sum <= max1 + max2;
    }
}
