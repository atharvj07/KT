import java.util.*;

public class BooleanFunction {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String expression = scanner.nextLine();
        int n = Integer.parseInt(scanner.nextLine());
        int[][] values = new int[n][5];
        for (int i = 0; i < n; i++) {
            String[] line = scanner.nextLine().split(" ");
            for (int j = 0; j < 5; j++) {
                values[i][j] = Integer.parseInt(line[j]);
            }
        }
        System.out.println(countValidExpressions(expression, values));
    }

    private static int countValidExpressions(String expression, int[][] values) {
        int[][][] dp = new int[expression.length() + 1][values.length][2];
        for (int i = expression.length() - 1; i >= 0; i--) {
            if (expression.charAt(i) == '?') {
                for (int j = 0; j < values.length; j++) {
                    for (int k = 0; k < 2; k++) {
                        dp[i][j][k] = (dp[i + 1][j][0] + dp[i + 1][j][1]) % MOD;
                    }
                }
            } else if (expression.charAt(i) == '(') {
                int closingBracketIndex = findClosingBracketIndex(expression, i);
                for (int j = 0; j < values.length; j++) {
                    for (int k = 0; k < 2; k++) {
                        for (int l = 0; l < 2; l++) {
                            if (k == 0 && values[j][4] == 0) {
                                continue;
                            }
                            if (k == 1 && values[j][4] == 1) {
                                continue;
                            }
                            dp[i][j][k] = (dp[i][j][k] + dp[closingBracketIndex + 1][j][l]) % MOD;
                        }
                    }
                }
                i = closingBracketIndex;
            } else {
                char variable = expression.charAt(i);
                int variableIndex = variable == 'A' ? 0 : variable == 'B' ? 1 : variable == 'C' ? 2 : 3;
                for (int j = 0; j < values.length; j++) {
                    for (int k = 0; k < 2; k++) {
                        if (k == 0 && values[j][4] == 0 && values[j][variableIndex] == 1) {
                            continue;
                        }
                        if (k == 1 && values[j][4] == 1 && values[j][variableIndex] == 0) {
                            continue;
                        }
                        dp[i][j][k] = (dp[i][j][k] + 1) % MOD;
                    }
                }
            }
        }
        return dp[0][0][1];
    }

    private static int findClosingBracketIndex(String expression, int openingBracketIndex) {
        int balance = 1;
        for (int i = openingBracketIndex + 1; i < expression.length(); i++) {
            if (expression.charAt(i) == '(') {
                balance++;
            } else if (expression.charAt(i) == ')') {
                balance--;
                if (balance == 0) {
                    return i;
                }
            }
        }
        throw new IllegalArgumentException("Invalid expression: missing closing bracket.");
    }
}
