#include <bits/stdc++.h>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    string s;
    cin >> n >> s;
    bool ok = false;
    for (int i = 1; i < n; i++) {
      if (s[i] > s[i - 1]) {
        cout << "YES" << endl;
        cout << 2 << endl;
        cout << s.substr(0, i) << " " << s.substr(i) << endl;
        ok = true;
        break;
      }
    }
    if (!ok) cout << "NO" << endl;
  }
  return 0;
}
