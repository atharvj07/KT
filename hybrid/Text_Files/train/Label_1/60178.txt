import sys

n, s, l = map(int, input().split())
a = list(map(int, input().split()))

if max(a) - min(a) <= s:
    print(1)
else:
    a.sort()
    i = 0
    j = l
    ans = 0
    while j <= n:
        if abs(a[j - 1] - a[i]) <= s:
            j += 1
        else:
            i += 1
            j = i + l
            ans += 1
    if i != j - 1:
        ans += 1
    print(ans if j <= n else -1)
