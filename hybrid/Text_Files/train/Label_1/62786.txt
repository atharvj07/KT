import sys

n, k = map(int, input().split())
d = list(map(int, input().split()))
if k * (k - 1) / 2 < n - 1:
    print(-1)
    sys.exit()
edges = []
for i in range(n):
    for j in range(n):
        if i == j:
            continue
        if d[i] + d[j] == d[n - 1]:
            edges.append((i + 1, j + 1))
            edges.append((j + 1, i + 1))
print(len(edges))
for edge in edges:
    print(*edge)
