#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 1e5;

int main() {
  int n, k, m;
  cin >> n >> k >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());

  int l = 0, r = 0;
  vector<int> cnt(MAXN, 0);
  bool ok = true;
  for (int i = 0; i < k; i++) {
    while (r < n && cnt[a[r] - a[l]] == 0) {
      cnt[a[r] - a[l]]++;
      r++;
    }
    if (r == n) {
      ok = false;
      break;
    }
    cnt[a[r] - a[l]]--;
    cnt[a[r]]++;
    r++;
  }
  if (!ok) {
    cout << "No" << endl;
    return 0;
  }
  vector<int> ans;
  for (int i = l + m - 1; i < r; i += m) { ans.push_back(a[i]); }
  cout << "Yes" << endl;
  for (int x : ans) { cout << x << " "; }
  cout << endl;
  return 0;
}
