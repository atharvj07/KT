#include <bits/stdc++.h>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int b, p, f, h, c; cin >> b >> p >> f >> h >> c;
        cout << max({0, min(b / 2, p) * h + min(b - min(b / 2, p) * 2, f) * c, min(b / 2, f) * c + min(b - min(b / 2, f) * 2, p) * h}) << endl;
    }
    return 0;
}
