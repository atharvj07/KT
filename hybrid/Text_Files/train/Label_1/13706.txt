#include <bits/stdc++.h>
using namespace std;

bool check(vector<int> &a, int n, int d) {
  auto b = a;
  for (int i = 0; i < n; i++) { b[i] &= a[(i + n - d) % n]; }
  return a == b;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, d;
    cin >> n >> d;
    vector<int> a(n);
    for (auto &x : a) { cin >> x; }
    int cnt = 0;
    while (!check(a, n, d)) {
      cnt++;
      auto b = a;
      for (int i = 0; i < n; i++) { b[i] &= a[(i + n - d) % n]; }
      a = b;
    }
    if (any_of(a.begin(), a.end(), [](int x) { return x; })) {
      cout << -1 << "\n";
    } else {
      cout << cnt << "\n";
    }
  }
}
