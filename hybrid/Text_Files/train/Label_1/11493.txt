#include <cstdio>
#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  getline(cin, s);

  int n = s.size(), start = 0;
  while (s[start] == '0' && start < n) start++;
  if (start == n) {
    cout << 0 << endl;
    return 0;
  }

  int first = -1;
  for (int i = start; i < n; i++) {
    if (s[i] == '.') {
      first = i;
      break;
    }
  }

  long long a = 0;
  while (start < n) {
    if (start != first) a = a * 10 + s[start] - '0';
    start++;
  }

  int b = 0;
  if (first != -1) {
    b = first - start;
  }

  if (a % 10 == 0) {
    while (a % 10 == 0) {
      b++;
      a /= 10;
    }
    a /= 10;
    b--;
  }

  if (b == 0) {
    cout << a << endl;
  } else {
    cout << a / 10.0 << "E" << b << endl;
  }
  return 0;
}
