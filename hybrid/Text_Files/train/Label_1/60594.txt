import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
a.sort()

l = 0
r = n // 3 - 1
best = 1e9
ans = []

while l <= r:
    m = (l + r) // 2
    b = [[] for i in range(m + 1)]
    cur = 0
    for i in range(n):
        b[cur].append(a[i])
        cur += 1
        if cur > m: cur = 0
    
    tot = 0
    for i in b:
        if len(i) > 2: tot += i[-1] - i[0] 
    
    if tot < best:
        best = tot
        ans = b
    
    if len(b[-1]) > 3:
        l = m + 1
    else:
        r = m - 1

print(best, len(ans))
for i in ans:
    for j in i:
        print(ans.index(i) + 1, end = ' ')
    print()
