import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    List<Meteor> meteors = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      String[] line = br.readLine().split(" ");
      long t1 = Long.parseLong(line[0]);
      long x1 = Long.parseLong(line[1]);
      long y1 = Long.parseLong(line[2]);
      long t2 = Long.parseLong(line[3]);
      long x2 = Long.parseLong(line[4]);
      long y2 = Long.parseLong(line[5]);
      meteors.add(new Meteor(t1, x1, y1, t2, x2, y2));
    }
    Map<Long, List<Point>> map = new HashMap<>();
    for (Meteor meteor : meteors) {
      long t = meteor.t1;
      while (t <= meteor.t2) {
        long x = meteor.x1 + (t - meteor.t1) * (meteor.x2 - meteor.x1) / (meteor.t2 - meteor.t1);
        long y = meteor.y1 + (t - meteor.t1) * (meteor.y2 - meteor.y1) / (meteor.t2 - meteor.t1);
        Point point = new Point(x, y);
        if (!map.containsKey(t)) {
          map.put(t, new ArrayList<>());
        }
        map.get(t).add(point);
        t++;
      }
    }
    int max = 0;
    for (List<Point> points : map.values()) {
      max = Math.max(max, points.size());
    }
    System.out.println(max);
  }

  static class Meteor {
    long t1, x1, y1, t2, x2, y2;

    Meteor(long t1, long x1, long y1, long t2, long x2, long y2) {
      this.t1 = t1;
      this.x1 = x1;
      this.y1 = y1;
      this.t2 = t2;
      this.x2 = x2;
      this.y2 = y2;
    }
  }

  static class Point {
    long x, y;

    Point(long x, long y) {
      this.x = x;
      this.y = y;
    }
  }
}
