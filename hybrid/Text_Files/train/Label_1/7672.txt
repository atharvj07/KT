#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> deg(n + 1);
  for (int i = 0; i < m; i++) {
    int x, y;
    cin >> x >> y;
    deg[x]++;
    deg[y]++;
  }
  int cnt1 = 0, cnt2 = 0;
  for (int i = 1; i <= n; i++) {
    if (deg[i] == 1) cnt1++;
    if (deg[i] > 2) cnt2++;
  }
  if (cnt1 == 2 && cnt2 == 0)
    cout << "bus topology" << endl;
  else if (cnt1 == n)
    cout << "star topology" << endl;
  else if (cnt2 == 0)
    cout << "ring topology" << endl;
  else
    cout << "unknown topology" << endl;
  return 0;
}
