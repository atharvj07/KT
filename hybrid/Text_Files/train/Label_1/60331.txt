usbs, ps2s, both = list(map(int, input().split()))
m = int(input())
prices = []
for i in range(m):
    price, port = input().split()
    prices.append((int(price), port))
prices.sort()
computers = 0
total = 0
for price, port in prices:
    if (port == "USB" and usbs > 0):
        computers += 1
        usbs -= 1
        total += price
    elif (port == "PS/2" and ps2s > 0):
        computers += 1
        ps2s -= 1
        total += price
    elif (both > 0):
        computers += 1
        both -= 1
        total += price
    if usbs == 0 and ps2s == 0 and both == 0:
        break
print(computers, total)
