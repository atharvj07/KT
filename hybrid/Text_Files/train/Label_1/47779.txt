import java.util.*;

public class Main {

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    char[] row = scanner.next().toCharArray();

    int countStand = 0;
    for (int i = 0; i < n; i++) {
      if (row[i] == 'X') {
        countStand++;
      }
    }

    int countSit = n - countStand;
    int result = Math.abs(countSit - countStand) / 2;
    System.out.println(result);

    char[] optimal = new char[n];
    if (countStand <= countSit) {
      int counter = 0;
      for (int i = 0; i < n; i++) {
        if (counter < countStand) {
          optimal[i] = 'X';
          counter++;
        } else {
          optimal[i] = 'x';
        }
      }
    } else {
      int counter = 0;
      for (int i = 0; i < n; i++) {
        if (counter < countSit) {
          optimal[i] = 'x';
          counter++;
        } else {
          optimal[i] = 'X';
        }
      }
    }
    System.out.println(String.valueOf(optimal));
  }
}
