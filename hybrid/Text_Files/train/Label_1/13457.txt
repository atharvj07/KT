#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> grid(n, vector<int>(n));
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        for (int j = 0; j < n; j++) {
            grid[i][j] = s[j] - '0';
        }
    }
    bool valid = true;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            int cnt = 0;
            if (i > 0) cnt += grid[i - 1][j];
            if (j > 0) cnt += grid[i][j - 1];
            if (i > 0 && j > 0) cnt -= grid[i - 1][j - 1];
            if (cnt != grid[i][j]) {
                valid = false;
            }
        }
    }
    cout << (valid ? "Yes" : "No") << "\n";
    return 0;
}
