#include <iostream>

int main() {
    int l, a;
    std::cin >> l >> a;
    std::string prefix, s = std::to_string(a + 1);
    while (prefix.size() < l) prefix += s;
    for (int i = 0; i + l <= (int)s.size(); i++) {
        std::string temp = s.substr(i, l);
        if (temp < prefix) prefix = temp;
    }
    std::cout << prefix << std::endl;
    return 0;
}
