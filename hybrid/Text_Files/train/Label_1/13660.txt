#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
int a[maxn], b[maxn], c[maxn];
int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        b[a[i]] = i;
        c[a[i]] = 1;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (c[a[i]] == 0) {
            ans++;
        } else {
            int j = b[a[i] * k];
            if (j > i) {
                ans++;
            }
        }
        c[a[i]] = 0;
    }
    cout << ans << endl;
    return 0;
}
