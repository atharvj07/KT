#include <bits/stdc++.h>
using namespace std;
int n, r;
char s[200001];
char t[200001];
int cnt[3];
int main() {
    cin >> n;
    cin >> s;
    for (int i = 0; i < n; i++) {
        s[i] == 'R' ? cnt[0]++ : s[i] == 'G' ? cnt[1]++ : cnt[2]++;
    }
    int mx = max({cnt[0], cnt[1], cnt[2]});
    r = mx - 1;
    for (int i = 0; i < n; i++) {
        if (r == 0) {
            t[i] = s[i];
            continue;
        }
        if (s[i] == 'R') {
            if (r && cnt[1] > cnt[2]) {
                t[i] = 'G';
                cnt[1]--;
                r--;
            } else if (r && cnt[2] >= cnt[1]) {
                t[i] = 'B';
                cnt[2]--;
                r--;
            } else {
                t[i] = 'R';
            }
        } else if (s[i] == 'G') {
            if (r && cnt[0] > cnt[2]) {
                t[i] = 'R';
                cnt[0]--;
                r--;
            } else if (r && cnt[2] >= cnt[0]) {
                t[i] = 'B';
                cnt[2]--;
                r--;
            } else {
                t[i] = 'G';
            }
        } else {
            if (r && cnt[0] > cnt[1]) {
                t[i] = 'R';
                cnt[0]--;
                r--;
            } else if (r && cnt[1] >= cnt[0]) {
                t[i] = 'G';
                cnt[1]--;
                r--;
            } else {
                t[i] = 'B';
            }
        }
    }
    cout << r << "\n";
    for (int i = 0; i < n; i++) {
        cout << t[i];
    }
    return 0;
}
