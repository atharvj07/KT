import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Scanner;
import java.util.Set;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] s = sc.nextLine().toCharArray();
        char[] t = sc.nextLine().toCharArray();

        // Build a mapping from characters in s to their corresponding characters in t
        Map<Character, Character> charMap = new HashMap<>();
        for (int i = 0; i < s.length; i++) {
            charMap.putIfAbsent(s[i], t[i]);
        }

        // Find all pairs of characters that are out of place
        Set<Character> outOfPlaceChars = new HashSet<>();
        for (int i = 0; i < s.length; i++) {
            if (s[i] != t[i]) {
                outOfPlaceChars.add(s[i]);
                outOfPlaceChars.add(t[i]);
            }
        }

        // If there are an odd number of out-of-place characters, it is impossible to fix the keyboard
        if (outOfPlaceChars.size() % 2 == 1) {
            System.out.println(-1);
            return;
        }

        // Find pairs of characters that need to be swapped
        List<String> swaps = new ArrayList<>();
        for (char c1 : outOfPlaceChars) {
            for (char c2 : outOfPlaceChars) {
                if (c1 != c2 && charMap.get(c1) == c2 && charMap.get(c2) == c1) {
                    swaps.add(String.format("%c %c", c1, c2));
                }
            }
        }

        // Print the number of pairs and the pairs themselves
        System.out.println(swaps.size());
        for (String swap : swaps) {
            System.out.println(swap);
        }
    }
}
