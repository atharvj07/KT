#include <bits/stdc++.h>
using namespace std;
int main() {
  int q;
  cin >> q;
  for (int k = 0; k < q; k++) {
    int n;
    cin >> n;
    map<vector<int>, int> M;
    vector<vector<int>> A(n);
    for (int i = 0; i < n; i++) {
      string s;
      cin >> s;
      for (auto &ch : s) A[i].push_back(ch - '0');
    }
    for (auto &a : A) {
      int m = a.size();
      vector<int> V(m);
      for (int i = 0; i < m; i++) {
        if (i < m / 2) V[m - i - 1] = a[i];
        else if (i == m / 2 && m % 2 == 1) continue;
        else V[i] = a[i];
      }
      M[V]++;
    }
    int mx = 0;
    for (auto &it : M) mx = max(mx, it.second);
    cout << mx << endl;
  }
  return 0;
}
