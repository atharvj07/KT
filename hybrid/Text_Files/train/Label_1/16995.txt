#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9+7;
#define pb(x) push_back(x)

ll n,x,ans,mn,a[100005];

int main() {
    cin>>n>>x;
    for(int i=1; i<=n; i++) {
        cin>>a[i];
    }
    mn=x;
    ans=a[1]*mn;
    for(int i=2; i<=n; i++) {
        mn=max(mn-1,1LL);
        ans+=a[i]*mn;
    }
    cout<<ans<<"\n";
    return 0;
}
