#include<bits/stdc++.h>
using namespace std;
int ab(char c){
    if(c=='a'){
        return 27;
    }
    return c-96;
}
int32_t main() {
    int n; cin>>n;
    string f; cin>>f;
    vector<int> adj;
    for(int i=1;i<(int)f.size();i++){
        if(ab(f[i])==ab(f[i-1])-1 || ab(f[i])==ab(f[i-1])+1){
            adj.push_back(i);
        }
    }
    cout<<adj.size()<<endl;

    return 0;
}
