import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long g = 0;
        for (int i = 0; i < n; i++) {
            g = gcd(g, a[i]);
        }
        if (g == 1) {
            System.out.println("1 " + a[0]);
        } else {
            long ans = 0;
            for (int i = 0; i < n; i++) {
                ans += a[i] / g;
            }
            System.out.println(ans + " " + g);
        }
    }

    public static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
