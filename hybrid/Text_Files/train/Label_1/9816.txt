#include <bits/stdc++.h>
using namespace std;
int n, x[100001];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> x[i];
    sort(x + 1, x + n + 1);
    int ans = 0;
    for (int i = 1; i < n; i++) {
        if (x[i] + 1 != x[i + 1]) ans++;
    }
    cout << ans << endl;
    return 0;
}
