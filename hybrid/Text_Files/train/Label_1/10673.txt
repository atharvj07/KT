#include <bits/stdc++.h>
using namespace std;
int a[100001], n, m;
pair<int, int> q[100001];
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= m; i++) cin >> q[i].first >> q[i].second;
  for (int i = 1; i <= m; i++) {
    int l = q[i].first;
    int r = q[i].second;
    int lmax = 0, rmin = 1e9;
    for (int j = l; j <= r; j++) {
      lmax = max(lmax, a[j]);
      rmin = min(rmin, a[j]);
    }
    bool f = 1;
    for (int j = l; j <= r; j++) {
      if (a[j] < lmax) f = 0;
      if (a[j] > rmin) f = 0;
    }
    if (f) cout << "Yes" << endl; else cout << "No" << endl;
  }
}
