#include <iostream>
#include <sstream>
using namespace std;
int main() {
  string s, a, b;
  getline(cin, s);
  bool f1 = 0, f2 = 0;
  for (char c : s) {
    int x = c - '0';
    if (x >= 0 && x <= 9) b += c, f2 = 1;
    else if (c == ',' || c == ';')
      b += c;
    else
      a += c, f1 = 1;
  }
  if (f1)
    cout << '"' << a << '"' << endl;
  else
    cout << '-' << endl;
  if (f2)
    cout << '"' << b << '"' << endl;
  else
    cout << '-' << endl;
}
