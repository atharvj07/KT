import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for(int i = 0; i < n; i++) a[i] = sc.nextInt();
        sc.close();
        Arrays.sort(a);
        BigInteger ans = BigInteger.ZERO;
        for(int i = 0; i < n - 1; i++) ans = ans.add(BigInteger.valueOf((long) a[i]));
        System.out.println(ans.add(BigInteger.valueOf((long) a[n - 1] - 1)));
    }
}
