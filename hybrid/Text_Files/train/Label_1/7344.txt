#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;

ll power(ll a, ll b, ll p = mod) {
    ll r = 1;
    for (; b; b >>= 1, a = (a * a) % p) {
        if (b & 1) { r = (r * a) % p; }
    }
    return r;
}

int main() {
    ll k, p;
    cin >> k >> p;
    ll ans = 0;
    for (ll i = 1; i <= k; i++) {
        ll num = i;
        ll rev_num = 0;
        while (num > 0) {
            rev_num = (10 * rev_num) + (num % 10);
            num /= 10;
        }
        if (i == rev_num && __builtin_popcountll(i) % 2 == 0) {
            ans = (ans + i) % p;
        }
    }
    cout << ans % p << '\n';
    return 0;
}
