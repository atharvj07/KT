import java.util.*;
class Main {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for(int i=0; i<n; i++) a[i] = sc.nextLong();
        Arrays.sort(a);
        long ans = 0;
        for(int i=0; i+2<n; i++){
            int j = i+1;
            int k = i+2;
            while(k<n){
                if(a[i]>a[j] && a[j]>a[k]) ans += n-k;
                k++;
            }
        }
        System.out.println(ans);
    }
}
