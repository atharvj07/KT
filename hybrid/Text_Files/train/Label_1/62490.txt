from collections import deque, defaultdict
n = int(input())
pos = defaultdict(list)
for i in range(n):
    x, y = map(int, input().split())
    pos[x].append(y)
    pos[y].append(x)
odd_chairs = deque()
for i in range(1, 2 * n + 1):
    if len(pos[i]) % 2 == 1:
        odd_chairs.append(i)
if len(odd_chairs) % 2 == 1:
    print(-1)
else:
    result = [[1, 2]] * n
    chairs = deque([i for i in range(1, 2 * n + 1) if len(pos[i]) % 2 == 0])
    while chairs:
        x = chairs[0]
        y = chairs[-1]
        result[pos[x][0] - 1][0] = 2
        result[pos[y][0] - 1][1] = 2
        chairs.popleft()
        chairs.pop()
        pos[x].remove(pos[x][0])
        pos[y].remove(pos[y][0])
    for i in result:
        print(*i)
