#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  long long ans = 0;
  vector<int> a;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    a.push_back(x);
  }
  sort(a.begin(), a.end());
  for (int i = 0, j = n - 1; i <= j; i++, j--) { ans += (a[i] + a[j]) * (a[i] + a[j]); }
  cout << ans << endl;
  return 0;
}
