import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int min = n;
        int ops = 0;
        while(true) {
            if(n % 4 == 0) {
                n /= 4;
                ops += 2;
            } else if(n % 2 == 0) {
                n /= 2;
                ops++;
            } else {
                break;
            }
            min = Math.min(min, n);
        }
        System.out.println(min + " " + ops);
    }
}
