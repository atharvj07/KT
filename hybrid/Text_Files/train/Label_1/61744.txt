from collections import Counter

n = int(input())
a = map(int, input().split(' '))
cnt = Counter(a) 
x = list(cnt.keys())
ans = 0
for i in range(1, len(x)):
    if cnt[x[i-1]] < cnt[x[i]]:
        if ans == 0 or cnt[x[i-1]] * 2 == cnt[x[i]]:
            ans = cnt[x[i-1]]
        else: break
if ans != 0: print(ans * (len(x)-1))
else: print(cnt[x[0]])
