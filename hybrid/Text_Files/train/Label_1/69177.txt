from sys import stdin
rl=lambda:stdin.readline()
n=int(rl().strip())
p=list(map(int,rl().strip().split()))
r=[[int(x) for x in rl().strip()]for i in range(n)]
d=[0 for i in range(n)]
for i in range(n):
    for j in range(n):
        if r[i][j] and p[i]>p[j]:d[j]+=1
ans=[]
while len(ans)<n:
    for i in range(n):
        if not d[i]:
            ans.append(p[i])
            break
    for j in range(n):
        if r[i][j] and p[i]<p[j]:d[j]-=1
print(*ans)
