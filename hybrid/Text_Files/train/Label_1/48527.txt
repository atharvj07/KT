import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {
    static int[] d, id, ans;
    static List<int[]>[] g;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        g = new List[n + 1];
        for (int i = 0; i <= n; i++) g[i] = new ArrayList<>();
        for (int i = 2; i <= n; i++) {
            int p = sc.nextInt();
            int w = sc.nextInt();
            g[i].add(new int[]{p, w});
            g[p].add(new int[]{i, w});
        }
        d = new int[n + 1];
        id = new int[n + 1];
        ans = new int[q];
        int next_id = 1;
        dfs(1, next_id);
        for (int i = 0; i < q; i++) {
            int v = sc.nextInt();
            int l = sc.nextInt();
            int r = sc.nextInt();
            int min_d = Integer.MAX_VALUE;
            for (int j = l; j <= r; j++)
                if (id[j] > 0 && d[j] < min_d)
                    min_d = d[j];
            for (int[] e : g[v])
                if (d[e[0]] <= min_d) {
                    min_d = d[e[0]];
                    break;
                }
            ans[i] = min_d;
        }
        for (int i = 0; i < q; i++) System.out.println(ans[i]);
    }

    static void dfs(int v, int next_id) {
        id[v] = next_id++;
        for (int[] e : g[v])
            if (id[e[0]] == 0) {
                d[e[0]] = d[v] + e[1];
                dfs(e[0], next_id);
            }
    }
}
