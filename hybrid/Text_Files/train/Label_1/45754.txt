import java.util.*;
public class Main {
  static void paint(int k) {
    char[][][] cube = new char[k][k][k];
    for (int i = 0; i < k; i++) 
      for (int j = 0; j < k; j++) 
        for (int h = 0; h < k; h++) 
          cube[i][j][h] = 'w';
    int s = k * k;
    if (k % 2 == 1) {
      for (int i = 0; i < s / 2; i++) 
        for (int j = i + 1; j < s / 2; j++) 
          if (i / k == j / k) {
            cube[i / k][i % k][j % k] = cube[i / k][j % k][i % k] = 'b';
            break;
          }
    }
    s /= 2;
    for (int i = 0; i < s / 2; i++) 
      for (int j = i + 1; j < s / 2; j++) 
        if (i / k == j / k) {
          cube[i / k][i % k][k - 1 - j % k] = cube[i / k][j % k][k - 1 - i % k] = 'b';
          break;
        }
    for (int i = 0; i < k; i++) {
      for (int j = 0; j < k; j++) {
        System.out.print(cube[i][j]);
        if (j == k - 1) 
          System.out.println();
      }
    }
    System.out.println();
  }
  public static void main(String[] a) {
    Scanner sc = new Scanner(System.in);
    int k = sc.nextInt();
    if (k % 2 == 0) 
      paint(k);
    else 
      System.out.println(-1);
  }
}
