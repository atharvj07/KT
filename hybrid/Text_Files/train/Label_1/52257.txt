import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    long n = input.nextLong();
    int minOps = 0;
    long min = n;
    while (n > 1) {
      if (n % 4 == 0) {
        mul(n, 4);
      } else if (n % 2 == 0) {
        mul(n, 2);
      } else {
        sqrt(n);
      }
    }
    System.out.println(min + " " + minOps);
  }

  private static void mul(long n, long x) {
    n *= x;
  }

  private static void sqrt(long n) {
    n = (long) Math.sqrt(n);
  }
}
