#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define inf 1e9
const int MAXN = 2e5 + 5;
ll a[MAXN], b[MAXN], c[MAXN], know[1000005], seen[1000005];
int main() {
  ll n, m, x;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i], know[a[i]] = 1;
  cin >> m;
  for (int i = 0; i < m; i++) {
    cin >> b[i] >> c[i];
    seen[b[i]] = 1;
    seen[c[i]] = 1;
  }
  x = 0;
  for (int i = 0; i < m; i++)
    if (know[b[i]]) x = max(x, seen[c[i]]);
  for (int i = 0; i < m; i++) {
    if (know[b[i]] && seen[c[i]] == x) { cout << i + 1 << endl; return 0; }
    if (know[c[i]] && seen[b[i]] == x) { cout << i + 1 << endl; return 0; }
  }
  return 0;
}
