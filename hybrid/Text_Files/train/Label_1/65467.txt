import queue
n,b=[int(i)for i in input().split()]
q=queue.PriorityQueue()
last_processed=0
for i in range(n):
 t,d=[int(i)for i in input().split()]
 while not q.empty()and q.queue[0][0]<=t:
  last_processed=q.get()[0]
 if q.qsize()<b:
  q.put([max(last_processed,t)+d,i])
 print(q.get()[0]if q.qsize()else-1)
