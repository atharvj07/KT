import java.util.*;
import java.lang.*;

class Main {
    public static void main (String[] args) throws java.lang.Exception {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int p = in.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = in.nextInt();
        }
        int[] ans = new int[n];
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        int tcur = 0;
        for (int i = 0; i < n; i++) {
            while (!pq.isEmpty() && pq.peek() <= t[i]) {
                pq.poll();
                tcur += p;
            }
            ans[i] = tcur + p;
            pq.add(ans[i]);
        }
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
