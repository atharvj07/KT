n,m,k=map(int,input().split())
h=sorted(list(map(int,input().split())))
for i in range(k):
    l,r=map(int,input().split())
    if l<r:
        l,r=r,l
    if l<=h[0] and r>=h[-1]:
        print(h[0])
        break
    if l<h[0]:
        inds=bisect.bisect(h,r)
        if inds==0:
            print(h[inds])
            break
        else:
            print(h[inds-1])
            break
