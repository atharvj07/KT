from collections import deque
def main():
    n, k = map(int, input().split())
    ms = set(map(int, input().split()))
    e = [[] for i in range(n+1)]
    f = [0] * (n+1)
    for i in range(2, n+1):
        p = list(map(int, input().split()))
        if not p: continue
        for j in p[1:]:
            e[j].append(i)
        f[i] += len(p[1:])
    q = deque([(x,0) for x in ms])
    t = 0
    ans = []
    while q:
        v,cnt = q.popleft()
        if cnt>t:
            ans.append(v)
            t = cnt
        for i in e[v]:
            f[i] -= 1
            if f[i] == 0: q.append((i,cnt+1))
    if len(ans) < k: return print(-1)
    print(len(ans))
    print(*ans)
main()
