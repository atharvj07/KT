#include <bits/stdc++.h>
using namespace std;
using ll=long long;

const int mod=1e9+7;

int main() {
    string s;
    cin>>s;
    ll p[s.length()+1]={0},last[s.length()+1][26]={0};
    for(int i=s.length()-1;i>=0;i--) {
        p[i]=(p[i+1]+(s[i]-'a'+1))%mod;
        for(int j=0;j<26;j++)
            last[i][j]=last[i+1][j];
        last[i][s[i]-'a']=i;
    }
    ll cnt=0;
    for(int i=0;i<s.length();i++) {
        ll tmp=1;
        for(int j=0;j<26;j++) {
            ll x=last[i][j];
            if(x==0) continue;
            tmp=(tmp*((p[i]-p[x]+mod)%mod)%mod)%mod;
            x=last[x][j];
            if(x==0) continue;
            tmp=(tmp*((p[i]-p[x]+mod)%mod)%mod)%mod;
            i=x;
            break;
        }
        cnt=(cnt+tmp)%mod;
    }
    cout<<cnt<<endl;
    return 0;
}
