def find_lexicographically_smallest_string(s, k):
    unique_chars = sorted(set(s))
    t = [unique_chars[0]] * k

    for i in range(k):
        for char in unique_chars:
            if char > s[i] if i < len(s) else t[i]:
                t[i] = char
                return ''.join(t[:i + 1] + [unique_chars[0]] * (k - i - 1))
    return ''.join(t)
