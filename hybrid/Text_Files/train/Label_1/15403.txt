#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        vector<int> ans(n, 1);
        for (int i = 0; i < n; i++) {
            if (s[i] == '>' && i < n - 1 && s[i + 1] != '<')
                ans[i] = 0;
            if (s[i] == '<' && i && s[i - 1] != '>')
                ans[i] = 0;
        }
        long long ret = 0;
        for (int i : ans) ret += i;
        cout << ret << endl;
    }
}
