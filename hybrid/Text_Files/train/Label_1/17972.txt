#include <iostream>
#include <vector>

using namespace std;

const long long MOD = 1e9 + 7;
long long f[100005][2];
int n, m, u[100005], v[100005];
vector<int>e[100005];

int main() {
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        cin >> u[i] >> v[i];
        e[u[i]].push_back(v[i]);
        e[v[i]].push_back(u[i]);
    }

    f[1][0] = f[1][1] = 1;

    for (int i = 2; i <= n; i++) {
        for (int v: e[i]) {
            (f[i][0] += f[v][1]) %= MOD;
            (f[i][1] += f[v][0] + f[v][1]) %= MOD;
        }
        (f[i][0] += f[i][1]) %= MOD;
    }

    cout << f[n][1];

    return 0;
}
