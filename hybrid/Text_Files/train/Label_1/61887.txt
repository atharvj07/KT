from sys import stdin
input=lambda:stdin.readline()   
n=int(input())
a=[int(x) for x in input().split()]
b=[a[0]]
p=[-1]*n
ans=[0]*n
for x in a[1:]:
 b.append(x)
 for j in range((len(b)-1)//2,-1,-1):
  y=j
  while 2*y+1<len(b):
   if len(b)>2*y+2 and b[2*y+2]<b[2*y+1]:
    y=2*y+2
   elif b[2*y+1]<b[y]:
    y=2*y+1
   else:break
  if y!=j:
   p[b.pop(j)]=b[y]
   b[y]=b[-1]
   b.pop()
   ans[len(b)-1]+=1
 for x in ans:print(x,end=' ')
