import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int q = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < q; i++) {
            long l = in.nextInt();
            long r = in.nextInt();
            long ans = 0;
            for (int j = 0; j < n; j++) {
                if (a[j] >= l && a[j] <= r) {
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
