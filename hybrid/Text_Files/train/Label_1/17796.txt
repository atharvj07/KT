#include<iostream>
#include<vector>
using namespace std;
int main() {
    int n, ans = 0, cnt = 0;
    cin >> n;
    int last = -1e9;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        if (x > last) {
            cnt++;
        } else cnt = 1;
        last = x;
        ans = max(ans, cnt);
    }
    cout << ans << endl;
}
