#include <bits/stdc++.h>
#define pb push_back

using namespace std;

const int maxn = 2e5+5;
vector<int> g[4*maxn];
bool vis[4*maxn];

int get_id(int x, int y){
    return 2*x + y;
}

int main(){
    int q; cin >> q;
    while(q--){
        int n; cin >> n;
        for(int i=1; i<4*maxn; i++){
            g[i].clear();
            vis[i] = false;
        }
        string s1, s2; cin >> s1 >> s2;
        for(int i=1; i<=n; i++){
            if(s1[i-1] == '1' or s1[i-1] == '2') continue;
            g[get_id(1, i)].pb(get_id(1, i+1));
            g[get_id(1, i+1)].pb(get_id(1, i));
        }
        for(int i=1; i<=n; i++){
            if(s1[i-1] == '1' or s1[i-1] == '2') continue;
            g[get_id(1, i)].pb(get_id(2, i));
            g[get_id(2, i)].pb(get_id(1, i));
        }
        for(int i=1; i<=n; i++){
            if(s2[i-1] == '1' or s2[i-1] == '2') continue;
            g[get_id(2, i)].pb(get_id(2, i+1));
            g[get_id(2, i+1)].pb(get_id(2, i));
        }
        for(int i=1; i<=n; i++){
            if(s2[i-1] == '1' or s2[i-1] == '2') continue;
            g[get_id(2, i)].pb(get_id(1, i));
            g[get_id(1, i)].pb(get_id(2, i));
        }

        queue<int> Q;
        Q.push(get_id(1, 0));
        vis[get_id(1, 0)] = true;
        bool ok = false;
        while(!Q.empty()){
            int u = Q.front(); Q.pop();
            if(u == get_id(2, n+1)){
                ok = true;
                break;
            }
            for(auto v : g[u]){
                if(vis[v]) continue;
                Q.push(v);
                vis[v] = true;
            }
        }
        cout << (ok ? "YES" : "NO") << endl;
    }
}
