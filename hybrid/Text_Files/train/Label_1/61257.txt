n = int(input())
q = []
for i in range(n - 2):
    q.append(sorted(map(int, input().split())))
    
q.sort()
n_lst = [0 for i in range(n)]
ans = []
for i in range(n - 2):
    if q[i][0] not in ans:
        ans.append(q[i][0])
    if q[i][1] not in ans:
        ans.append(q[i][1])
    if q[i][2] not in ans:
        ans.append(q[i][2])
    n_lst[q[i][0] - 1] += 1
    n_lst[q[i][1] - 1] += 1
    n_lst[q[i][2] - 1] += 1

for i in range(n):
    if n_lst[i] == 1:
        ans.append(i + 1)
print(*ans)
