import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int h = scanner.nextInt();
        int w = scanner.nextInt();
        int[][] ranges = new int[h][2];
        for (int i = 0; i < h; i++) {
            ranges[i][0] = scanner.nextInt();
            ranges[i][1] = scanner.nextInt();
        }
        int[] dp = new int[h + 1];
        dp[0] = 0;
        for (int i = 1; i <= h; i++) {
            dp[i] = -1;
            for (int j = i - 1; j >= 0; j--) {
                if (dp[j] >= 0 && ranges[j][0] - 1 <= dp[j] && dp[j] <= ranges[j][1] - 1) {
                    dp[i] = dp[j] + 1;
                    break;
                } else if (dp[j] >= 0 && dp[j] > ranges[j][1]) {
                    dp[i] = dp[j] + 1;
                    break;
                }
            }
        }
        for (int i = 1; i <= h; i++) {
            System.out.println(dp[i]);
        }
    }
}
