#include<bits/stdc++.h>
using namespace std;
int main()
{
	int t;
	cin >> t;
	while(t--)
	{
		int n;
		cin >> n;
		vector<int> ans;
		for(int i = 2; i * i <= n; i++)
		{
			while(n % i == 0)
			{
				ans.push_back(i);
				n /= i;
			}
		}
		if(n > 1)
		{
			ans.push_back(n);
		}
		if(ans.size() < 3)
		{
			cout << "NO" << endl;
		}
		else
		{
			cout << "YES" << endl;
			for(int x : ans)
				cout << x << " ";
			cout << endl;
		}
	}
	return 0;
}
