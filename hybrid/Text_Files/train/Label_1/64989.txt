import sys
input = sys.stdin.readline
a = [int(x) for x in input().split()]
n, q = a[0], a[1]
a = [int(x) for x in input().split()]
g = [[]for i in range(n+2)]
for i in range(n-1):
    if a[i]&a[i+1]>0:
        g[i+1]+=[i+2]
        g[i+2]+=[i+1]
for i in range(q):
    x, y = map(int, input().split())
    vis = [False]*(n+2)
    vis[x]=True
    q = [x]
    while q:
        v = q.pop()
        if v==y:
            print("Shi")
            break
        for u in g[v]:
            if vis[u]==False:
                vis[u]=True
                q+=[u]
    else:
        print("Fou")
