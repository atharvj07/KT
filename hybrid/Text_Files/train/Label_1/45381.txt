import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int D1 = sc.nextInt();
        int D2 = sc.nextInt();
        int[][] points = new int[N * N][2];
        int idx = 0;
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                points[idx][0] = i;
                points[idx][1] = j;
                idx++;
            }
        }
        for (int i = 0; i < N * N; i++) {
            for (int j = i + 1; j < N * N; j++) {
                int dx = points[i][0] - points[j][0];
                int dy = points[i][1] - points[j][1];
                int dist = dx * dx + dy * dy;
                if (dist == D1 || dist == D2) {
                    points[i][0] = -1;
                    points[j][0] = -1;
                    break;
                }
            }
        }
        idx = 0;
        for (int i = 0; i < N * N; i++) {
            if (points[i][0] != -1) {
                System.out.println(points[i][0] + " " + points[i][1]);
                idx++;
            }
        }
    }
}
