#include <iostream>
using namespace std;
int main() {
  int a[3], b[3], n, i, ans = 1;
  cin >> a[0] >> a[1] >> a[2] >> b[0] >> b[1] >> b[2] >> n;
  for (i = 0; i < 3; i++)
    ans &= (a[i] <= 5) && (b[i] <= 10);
  cout << (n >= (a[0] + a[1] + a[2] + b[0] + b[1] + b[2]) && ans ? "YES" : "NO");
  return 0;
}
