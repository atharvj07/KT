import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class ColorfulRibbon {
    static class SegmentTree {
        long[] tree;
        long[] lazy;

        public SegmentTree(int n) {
            this.tree = new long[4 * n];
            this.lazy = new long[4 * n];
        }

        private void build(int node, int start, int end, int[] arr) {
            if (start == end) {
                tree[node] = arr[start];
            } else {
                int mid = (start + end) / 2;
                build(2 * node, start, mid, arr);
                build(2 * node + 1, mid + 1, end, arr);
                tree[node] = tree[2 * node] + tree[2 * node + 1];
            }
        }

        private void update(int node, int start, int end, int l, int r, int val) {
            if (lazy[node] != 0) {
                tree[node] += (end - start + 1) * lazy[node];
                if (start != end) {
                    lazy[2 * node] += lazy[node];
                    lazy[2 * node + 1] += lazy[node];
                }
                lazy[node] = 0;
            }

            if (start > end || start > r || end < l) {
                return;
            }

            if (start >= l && end <= r) {
                tree[node] += (end - start + 1) * val;
                if (start != end) {
                    lazy[2 * node] += val;
                    lazy[2 * node + 1] += val;
                }
                return;
            }

            int mid = (start + end) / 2;
            update(2 * node, start, mid, l, r, val);
            update(2 * node + 1, mid + 1, end, l, r, val);
            tree[node] = tree[2 * node] + tree[2 * node + 1];
        }

        private long query(int node, int start, int end, int l, int r) {
            if (lazy[node] != 0) {
                tree[node] += (end - start + 1) * lazy[node];
                if (start != end) {
                    lazy[2 * node] += lazy[node];
                    lazy[2 * node + 1] += lazy[node];
                }
                lazy[node] = 0;
            }

            if (start > end || start > r || end < l) {
                return 0;
            }

            if (start >= l && end <= r) {
                return tree[node];
            }

            int mid = (start + end) / 2;
            long left = query(2 * node, start, mid, l, r);
            long right = query(2 * node + 1, mid + 1, end, l, r);
            return left + right;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        int[] colors = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            colors[i] = i;
        }

        SegmentTree st = new SegmentTree(n);
        st.build(1, 1, n, colors);

        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int type = Integer.parseInt(line[0]);

            if (type == 1) {
                int l = Integer.parseInt(line[1]);
                int r = Integer.parseInt(line[2]);
                int x = Integer.parseInt(line[3]);

                st.update(1, 1, n, l, r, x - colors[l]);
                Arrays.fill(colors, l, r + 1, x);
            } else {
                int l = Integer.parseInt(line[1]);
                int r = Integer.parseInt(line[2]);

                long sum = st.query(1, 1, n, l, r);
                System.out.println(sum);
            }
        }
    }
}
