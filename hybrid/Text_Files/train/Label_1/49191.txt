import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        char[] shapes = sc.next().toCharArray();
        int[] counts = new int[3];
        for (char shape : shapes) {
            counts[shape - 'R']++;
        }
        int ans = 0;
        for (int i = 0; i < 3; i++) {
            ans += counts[i] % 2;
        }
        System.out.println(ans);
        for (int i = 0; i < q; i++) {
            int p = sc.nextInt() - 1;
            char c = sc.next().charAt(0);
            int oldShape = shapes[p] - 'R';
            int newShape = c - 'R';
            counts[oldShape]--;
            counts[newShape]++;
            ans += (counts[newShape] % 2) - (counts[oldShape] % 2);
            System.out.println(ans);
        }
    }
}
