import java.util.Scanner;

public class AutomaticDoor {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int a = scanner.nextInt();
        long d = scanner.nextLong();

        long[] employees = new long[n];
        long[] clients = new long[m];

        for (int i = 0; i < n; i++) {
            employees[i] = (long) (i + 1) * a;
        }

        for (int i = 0; i < m; i++) {
            clients[i] = scanner.nextLong();
        }

        int employeeIndex = 0;
        int clientIndex = 0;
        int openCount = 0;
        long closeTime = 0;

        while (employeeIndex < n || clientIndex < m) {
            long nextEventTime = Long.MAX_VALUE;

            if (employeeIndex < n && employees[employeeIndex] <= nextEventTime) {
                nextEventTime = employees[employeeIndex];
            }

            if (clientIndex < m && clients[clientIndex] <= nextEventTime) {
                nextEventTime = clients[clientIndex];
            }

            if (nextEventTime == Long.MAX_VALUE) {
                break;
            }

            if (nextEventTime <= closeTime) {
                employeeIndex++;
                clientIndex++;
            } else {
                openCount++;
                closeTime = nextEventTime + d;
            }
        }

        System.out.println(openCount);
    }
}
