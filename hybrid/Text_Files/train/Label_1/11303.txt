#include <iostream>

using namespace std;

typedef long long ll;
const ll mod = 1000000009;

ll power(ll n) {
  ll res = 1;
  while (n > 0) {
    if (n & 1) res = (res * 2) % mod;
    n >>= 1;
  }
  return res;
}

ll get_min_score(ll n, ll m, ll k) {
  ll score = 1;
  ll inc = 0;
  for (ll i = 1; i <= n; i++) {
    if (inc < k && i < n) {
      score = (score * 2) % mod;
      inc++;
    } else if (inc == k || i == n) {
      score = (score * 2) % mod;
      score = (score + 1) % mod;
      inc = 0;
    }
    if (i >= m) score = (score - power(i - m)) % mod;
  }
  return score;
}

int main() {
  ll n, m, k;
  cin >> n >> m >> k;
  cout << get_min_score(n, m, k) << endl;
  return 0;
}
