import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            String s = sc.next();
            String t = sc.next();
            int flag = 0;
            if (s.equals(t)) {
                System.out.println("YES");
                continue;
            }
            for (int j = 0; j < n; j++) {
                int cnt = 0;
                for (int k = j; k < n; k++) {
                    if (s.charAt(k) == t.charAt(k))
                        cnt++;
                    else
                        break;
                }
                if (cnt % 2 != 0) {
                    flag = 1;
                    break;
                }
            }
            if (flag == 1)
                System.out.println("NO");
            else
                System.out.println("YES");
        }
    }
}
