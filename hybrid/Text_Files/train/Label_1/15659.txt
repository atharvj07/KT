#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<long long> a(n), b;
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  for (int i = n - 1; i >= 0; i -= m) {
    b.push_back(a[i]);
    if (b.size() >= (size_t)k) break;
  }
  if (b.size() != (size_t)k) {
    cout << 0 << '\n';
    return 0;
  }
  cout << accumulate(b.begin(), b.end(), 0LL) << '\n';
  for (int i = 0; i + 1 < (int)b.size(); i++) cout << i + 1 << ' ';
  cout << '\n';
}
