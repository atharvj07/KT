import sys

def solve():
    t = int(input())
    for _ in range(t):
        s = input()
        t = input()
        s = ''.join(sorted(s))
        i = 0
        j = 0
        while i < len(s) and j < len(t):
            if s[i] == t[j]:
                j += 1
            i += 1
        if j == len(t):
            s = s[i:]
        print(s)

solve()
