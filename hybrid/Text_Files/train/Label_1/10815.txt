#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> w(n);
    for (int i = 0; i < n; i++) {
        cin >> w[i];
    }
    vector<pair<int, int>> f(m);
    for (int i = 0; i < m; i++) {
        cin >> f[i].first >> f[i].second;
        --f[i].first;
        --f[i].second;
    }
    vector<int> ans;
    vector<int> deg(m);
    for (int i = 0; i < m; i++) {
        w[f[i].first]--;
        w[f[i].second]--;
        if (w[f[i].first] == 0) {
            deg[i]++;
        }
        if (w[f[i].second] == 0) {
            deg[i]++;
        }
    }
    set<int> q;
    for (int i = 0; i < m; i++) {
        if (deg[i] == 0) {
            q.insert(i);
        }
    }
    while (!q.empty()) {
        int u = *q.begin();
        q.erase(q.begin());
        ans.push_back(u + 1);
        if (w[f[u].first] == 0) {
            w[f[u].second]++;
            if (w[f[u].second] > 0) {
                deg[u]--;
                if (deg[u] == 0) {
                    q.insert(u);
                }
            }
        } else {
            w[f[u].first]++;
            if (w[f[u].first] > 0) {
                deg[u]--;
                if (deg[u] == 0) {
                    q.insert(u);
                }
            }
        }
    }
    if (ans.size() == m) {
        cout << "ALIVE" << endl;
        for (int i = 0; i < m; i++) {
            cout << ans[i] << " ";
        }
        cout << endl;
    } else {
        cout << "DEAD" << endl;
    }
    return 0;
}
