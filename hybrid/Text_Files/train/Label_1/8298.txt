#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main(){
    double a,b,c,d;cin>>a>>b>>c>>d;
    if(a==c&&b==d)cout<<"0.0000000000"<<endl;
    else if((a*d-b*c)==0)cout<<fixed<<setprecision(10)<<sqrt((a-c)*(a-c)+(b-d)*(b-d))/2<<endl;
    else cout<<max(fabs(a-c),fabs(b-d))<<endl;
    return 0;
}
