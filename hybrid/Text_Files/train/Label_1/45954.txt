import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + ans + 1; j <= n; j++) {
                boolean valid = true;
                for (int k = i; k < j; k++) {
                    if (s.charAt(k) >= 'A' && s.charAt(k) <= 'Z') {
                        valid = false;
                        break;
                    }
                    for (int l = k + 1; l < j; l++) {
                        if (s.charAt(k) == s.charAt(l)) {
                            valid = false;
                            break;
                        }
                    }
                }
                if (valid) {
                    ans = Math.max(ans, j - i);
                    break;
                }
            }
        }
        System.out.println(ans);
    }
}
