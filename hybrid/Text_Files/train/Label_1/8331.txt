#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      int x;
      cin >> x;
      if (x != -1) {
        if (i == 0)
          a[0] = x;
        else {
          if (x != a[0]) {
            a[i] = x;
            a[j] = a[0];
          }
        }
      }
    }
  }
  for (int i = 0; i < n; ++i) cout << a[i] << " ";
  return 0;
}
