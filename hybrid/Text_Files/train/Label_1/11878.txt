#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n + 2);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }
    for (int i = 1; i <= n; ++i) {
        if (a[i - 1] > a[i]) {
            swap(a[i - 1], a[i]);
        }
        if (a[i + 1] > a[i]) {
            swap(a[i], a[i + 1]);
        }
    }
    int k = 0;
    for (int i = 1; i <= n; ++i) {
        if (a[i] > a[i - 1]) {
            ++k;
        }
    }
    cout << k << endl;
    for (int i = 1; i <= k; ++i) {
        if (a[i] < a[i + 1]) {
            cout << "L";
        } else {
            cout << "R";
        }
    }
    cout << endl;
    return 0;
}
