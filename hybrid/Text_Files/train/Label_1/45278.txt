import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int[][] grid = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = input.nextInt();
            }
        }
        int[][] pairIndex = new int[n * m + 1][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int shoe = grid[i][j];
                if (pairIndex[shoe][0] == 0) {
                    pairIndex[shoe][0] = i;
                    pairIndex[shoe][1] = j;
                } else {
                    int x1 = pairIndex[shoe][0];
                    int y1 = pairIndex[shoe][1];
                    int x2 = i;
                    int y2 = j;
                    if (x1 == x2) {
                        if (y1 > y2) {
                            grid[x1][y1] = grid[x2][y2];
                            grid[x2][y2] = shoe;
                        }
                    } else if (y1 == y2) {
                        if (x1 > x2) {
                            grid[x1][y1] = grid[x2][y2];
                            grid[x2][y2] = shoe;
                        }
                    }
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int shoe = grid[i][j];
                if (pairIndex[shoe][0] != i || pairIndex[shoe][1] != j) {
                    ans++;
                }
            }
        }
        System.out.println(ans / 2);
    }
}
