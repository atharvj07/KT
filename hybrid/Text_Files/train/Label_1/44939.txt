import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        PrintWriter pw = new PrintWriter(System.out);
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] in = new int[n], out = new int[n];
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1, b = sc.nextInt() - 1;
            in[a]++;
            out[b]++;
        }
        int g = -1;
        for (int i = 0; i < n; i++) {
            if (in[i] != out[i]) {
                pw.println("No");
                pw.close();
                return;
            }
            if (in[i] > 0 && g == -1) g = in[i];
            else if (in[i] > 0 && g != in[i]) {
                pw.println("No");
                pw.close();
                return;
            }
        }
        pw.println("Yes");
        pw.close();
    }
}
