t, sx, sy, ex, ey = map(int, input().split())
wind_directions = input()

# Calculate the required moves in each direction
dx = ex - sx
dy = ey - sy

# Initialize the time counter
time = 0

# Iterate through the wind directions
for wind in wind_directions:
    if dx > 0 and wind == 'E':
        dx -= 1
    elif dx < 0 and wind == 'W':
        dx += 1
    elif dy > 0 and wind == 'N':
        dy -= 1
    elif dy < 0 and wind == 'S':
        dy += 1
    
    # Increment the time
    time += 1
    
    # Check if the destination is reached
    if dx == 0 and dy == 0:
        print(time)
        break
else:
    print(-1)
