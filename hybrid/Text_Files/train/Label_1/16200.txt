#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, p, q, x, y;
    cin >> n >> p;
    unordered_set<int> a;
    for (int i = 0; i < p; i++) {
        cin >> x;
        a.insert(x);
    }
    cin >> q;
    for (int i = 0; i < q; i++) {
        cin >> y;
        if (a.find(y) != a.end()) {
            cout << "I become the guy." << endl;
            return 0;
        }
    }
    cout << "Oh, my keyboard!" << endl;
    return 0;
}
