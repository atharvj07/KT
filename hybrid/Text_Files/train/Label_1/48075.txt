import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] c = new int[5];
        for (int i = 0; i < 5; i++)
            c[i] = in.nextInt();
        in.close();

        boolean valid = true;
        for (int i = 0; i < 5; i++) {
            if ((c[i] - i) % 5 != 0) {
                valid = false;
                break;
            }
        }

        System.out.println(valid ? (c[0] - 0) / 5 : -1);
    }
}
