import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt(), b = in.nextInt(), k = in.nextInt();
        ArrayList<Integer> a1 = new ArrayList<>(), b1 = new ArrayList<>();
        for (int i = 0; i < a; i++) {
            a1.add(in.nextInt());
        }
        for (int i = 0; i < b; i++) {
            b1.add(in.nextInt());
        }
        int ans = 0;
        for (int i = 0; i < k; i++) {
            if (!a1.get(i % a).equals(b1.get(i % b)))
                ans++;
        }
        System.out.println(((k - ans) / a) * b + ans);
    }
}
