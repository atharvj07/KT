import sys
INF = 10**9
n = list(map(int, input().split()))
a = [0] + list(map(int, input().split()))
b = [0] + list(map(int, input().split()))
c = [0] + list(map(int, input().split()))
d = [0] + list(map(int, input().split()))

m = int(input())
ab = {}
for i in range(m):
    x, y = map(int, input().split())
    ab[(x, y)] = True

m = int(input())
bc = {}
for i in range(m):
    x, y = map(int, input().split())
    bc[(x, y)] = True

m = int(input())
cd = {}
for i in range(m):
    x, y = map(int, input().split())
    cd[(x, y)] = True

dp = [[[[INF] * 11 for _ in range(11)] for _ in range(11)] for _ in range(11)]
dp[0][0][0][0] = 0

for fa in range(1, n[0]+1):
    for fb in range(1, n[1]+1):
        for fc in range(1, n[2]+1):
            for fd in range(1, n[3]+1):
                if (fa, fb) in ab: continue
                if (fb, fc) in bc: continue
                if (fc, fd) in cd: continue
                dp[fa][fb][fc][fd] = min(dp[fa][fb][fc][fd], 
                                         dp[fa-1][fb][fc][fd] + a[fa],
                                         dp[fa][fb-1][fc][fd] + b[fb],
                                         dp[fa][fb][fc-1][fd] + c[fc],
                                         dp[fa][fb][fc][fd-1] + d[fd])

ans = dp[n[0]][n[1]][n[2]][n[3]]
if ans == INF:
    print(-1)
else:
    print(ans)
