import sys

n, m, k = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
lock = [0] * (k + 1)

for i in range(n):
    for j in range(m):
        if lock[a[i][j]]:
            print(lock[a[i][j]])
            lock[a[i][j]] = 0
            for l in range(i):
                for h in range(m):
                    if a[l][h] == a[i][j]:
                        print(lock[a[l][h]])
                        lock[a[l][h]] = 0
            break
        elif a[i][j]:
            lock[a[i][j]] = j + 1
    else:
        print(0)
