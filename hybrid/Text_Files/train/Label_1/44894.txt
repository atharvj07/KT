import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);

		long l1 = in.nextLong();
		long r1 = in.nextLong();
		long l2 = in.nextLong();
		long r2 = in.nextLong();

		long min1 = Math.max(1, l1 + r1 - (r2 + l2));
		long max1 = Math.min(r1, l2 + r2 - (l1 + r1));

		long common = Math.max(0, max1 - min1 + 1);
		System.out.println(common);
	}
}
