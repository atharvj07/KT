import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine());
        HashMap<Integer, List<Integer>> adj = new HashMap<>();
        int[] a = new int[n - 1];
        int[] b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            String[] s = sc.nextLine().split(" ");
            a[i] = Integer.parseInt(s[0]);
            b[i] = Integer.parseInt(s[1]);
            List<Integer> l1 = adj.getOrDefault(a[i], new ArrayList<>());
            l1.add(b[i]);
            adj.put(a[i], l1);
            List<Integer> l2 = adj.getOrDefault(b[i], new ArrayList<>());
            l2.add(a[i]);
            adj.put(b[i], l2);
        }

        int[] color = new int[n + 1];
        Arrays.fill(color, -1);
        dfs(1, 0, color, adj);

        LinkedList<Integer> q = new LinkedList<>();
        for (int i = 1; i <= n; i++) {
            if (color[i] == 0) {
                q.add(i);
            }
        }

        while (!q.isEmpty()) {
            int v = q.poll();
            for (int u : adj.getOrDefault(v, new ArrayList<>())) {
                if (color[v] == color[u]) {
                    System.out.println("Second");
                    return;
                } else if (color[u] == -1) {
                    color[u] = 1 - color[v];
                    q.add(u);
                }
            }
        }

        System.out.println("First");
    }

    static void dfs(int v, int p, int[] color, HashMap<Integer, List<Integer>> adj) {
        color[v] = 0;
        for (int u : adj.getOrDefault(v, new ArrayList<>())) {
            if (u != p) {
                dfs(u, v, color, adj);
            }
        }
    }
}
