from collections import deque

n, d = map(int, input().split())
adj = [[] for i in range(n + 1)]
deg = [0] * (n + 1)
for i in range(d):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)
    deg[a] += 1
    deg[b] += 1
ans = [0] * d
q = deque([i for i in range(1, n + 1) if deg[i] == 1])
for i in range(d):
    while q:
        cur = q.popleft()
        ans[i] = max(ans[i], deg[cur])
        for nxt in adj[cur]:
            deg[nxt] -= 1
            if deg[nxt] == 1:
                q.append(nxt)
print(*ans)
