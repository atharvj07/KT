import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[] a = new int[n];
            int[] b = new int[m];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            for (int j = 0; j < m; j++) {
                b[j] = sc.nextInt();
            }
            int[][] dp = new int[n + 1][m + 1];
            for (int j = 1; j <= n; j++) {
                for (int k = 1; k <= m; k++) {
                    if (a[j - 1] == b[k - 1]) {
                        dp[j][k] = dp[j - 1][k - 1] + 1;
                    } else {
                        dp[j][k] = Math.max(dp[j - 1][k], dp[j][k - 1]);
                    }
                }
            }
            if (dp[n][m] == 0) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
                int j = n;
                int k = m;
                ArrayList<Integer> ans = new ArrayList<>();
                while (j > 0 && k > 0) {
                    if (a[j - 1] == b[k - 1]) {
                        ans.add(a[j - 1]);
                        j--;
                        k--;
                    } else {
                        if (dp[j - 1][k] > dp[j][k - 1]) {
                            j--;
                        } else {
                            k--;
                        }
                    }
                }
                Collections.reverse(ans);
                System.out.println(ans.size());
                for (int l = 0; l < ans.size(); l++) {
                    System.out.print(ans.get(l) + " ");
                }
                System.out.println();
            }
        }
    }
}
