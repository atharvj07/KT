#include<iostream>
#include<string>

using namespace std;

int main(){
    string s, t;
    int q;
    cin >> q;
    while(q--){
        cin >> s >> t;
        bool ans = true;
        for(int i = 0; i < s.size(); i++)
            ans &= (s[i] == t[i] || s[(i+1)%s.size()] == t[i] || t[(i+1)%t.size()] == s[i]);
        cout << (ans? "YES" : "NO") << endl;
    }
    return 0;
}
