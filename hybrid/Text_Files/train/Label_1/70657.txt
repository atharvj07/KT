n = int(input())
requests = list(map(int, input().split()))

max_length = 0
current_sum = 0
for i in range(n):
    current_sum += requests[i]
    if current_sum > 100 * (i + 1):
        max_length = max(max_length, i + 1)
    for j in range(i):
        current_sum -= requests[j]
        if current_sum > 100 * (i - j):
            max_length = max(max_length, i - j)

print(max_length)
