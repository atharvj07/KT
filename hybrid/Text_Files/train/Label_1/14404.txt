#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, ans = 0;
    string s;
    cin >> n >> s;
    for (int l = -1, r = -1, i = 0; i <= n; i++) {
        if (s[i] != '.' || i == n) {
            if (l < 0 || r < 0) {
                ans += i - max(l, r);
            } else if (l == r) {
                ans += i - r;
            }
            l = i, r = i;
        }
        if (s[i] == 'R')
            r++;
        else if (s[i] == 'L')
            l++;
    }
    cout << ans << endl;
}
