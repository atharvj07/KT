#include <bits/stdc++.h>

using namespace std;

const int MAX_N = 120;
const int MAX_M = 5;
const int mod = 1e9 + 7;
const int INF = 1 << 30;

int n, a[MAX_N][MAX_M], b[MAX_M];

int solve() {
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 5; j++) {
      if (i == 1) {
        b[j] = a[i][j];
      } else if (b[j] == -1 || a[i][j] < b[j]) {
        b[j] = a[i][j];
      }
    }
  }
  sort(b, b + 5, greater<int>());
  int ans = 0;
  for (int i = 0; i < 5; i++) {
    if (b[i] >= 30) {
      int x = (119 - b[i]) / 2;
      ans += x * (3000 - i * 500) / 250;
    }
  }
  ans++;
  for (int i = 0; i < n; i++) {
    if (i == 1) {
      continue;
    }
    int sum = 0;
    for (int j = 0; j < 5; j++) {
      if (a[i][j] != -1) {
        int x = (119 - a[i][j]) / 2;
        sum += x * (3000 - j * 500) / 250;
      }
    }
    sum++;
    if (sum >= ans) {
      return -1;
    }
  }
  return ans;
}

int main() {
  memset(a, -1, sizeof(a));
  cin >> n;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 5; j++) {
      cin >> a[i][j];
    }
  }
  cout << solve() << endl;
  return 0;
}
