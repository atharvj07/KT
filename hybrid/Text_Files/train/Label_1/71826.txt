from collections import defaultdict, deque

n, d = map(int, input().split())
graph = defaultdict(list)

for _ in range(d):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

max_acquaintances = [0] * d
visited = [False] * (n + 1)

for i in range(d):
    queue = deque()
    for node in graph:
        if not visited[node]:
            queue.append((node, 1))
            visited[node] = True

    while queue:
        current_node, acquaintances = queue.popleft()
        max_acquaintances[i] = max(max_acquaintances[i], acquaintances)

        for neighbor in graph[current_node]:
            if not visited[neighbor]:
                queue.append((neighbor, acquaintances + 1))
                visited[neighbor] = True

    for j in range(1, n + 1):
        visited[j] = False

for acquaintance in max_acquaintances:
    print(acquaintance)
