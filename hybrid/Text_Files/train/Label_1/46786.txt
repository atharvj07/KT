import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), M = (int) 1e9, pwSeq[] = new int[N];
        for (int i = 0; i < N; i++) {
            pwSeq[i] = sc.nextInt();
            M = Math.min(M, pwSeq[i]);
        }
        Arrays.sort(pwSeq);
        long res = 0;
        for (int i = 0; i < N; i++) {
            if (pwSeq[i] % M != 0)
                res += Math.min(pwSeq[i] % M, M - (pwSeq[i] % M));
        }
        System.out.println(res);
    }
}
