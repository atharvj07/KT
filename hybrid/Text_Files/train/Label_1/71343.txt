import heapq, sys
input = sys.stdin.readline

d, n, m = map(int, input().split())
gas = [(0, 0)]
for i in range(m):
    x, p = map(int, input().split())
    gas.append((x, p))
gas.append((d, 0))

ans = 0
pos = 0
heap = []
for i in range(1, len(gas)):
    need = gas[i][0] - pos - n
    if need > 0:
        while heap and need > 0:
            cost, x = heapq.heappop(heap)
            buy = min(x, need)
            ans += cost * buy
            need -= buy
            x -= buy
            if x > 0:
                heapq.heappush(heap, (cost, x))
        if need > 0:
            print(-1)
            exit(0)
    pos = gas[i - 1][0]
    heapq.heappush(heap, (gas[i][1], gas[i][0] - pos))

print(ans)
