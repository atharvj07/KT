#include <bits/stdc++.h>

using namespace std;
using ll = long long;
const int N = 2e5 + 5;

ll a[N];

int main() {
    cin.tie(nullptr);
    ios_base::sync_with_stdio(false);
    ll n, sum = 0;
    cin >> n;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
        sum += a[i];
    }
    cout << abs(sum) / 2 << endl;
    return 0;
}
