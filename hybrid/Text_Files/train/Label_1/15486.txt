#include <bits/stdc++.h>

using namespace std;

struct point {
  int x, y;
  int special;
};

const int maxn = 100500;

point points[maxn];
bool mark[maxn];
int ans[maxn];
vector<int> g[maxn];
int in[maxn];
int n;

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> points[i].x >> points[i].y;
    points[i].special = -1e9;
  }
  for (int i = 0; i < n; i++) { cin >> points[i].special; }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (points[i].x <= points[j].x && points[i].y <= points[j].y) {
        g[i].push_back(j);
        in[j]++;
      }
    }
  }
  queue<int> q;
  for (int i = 0; i < n; i++) {
    if (points[i].special == 0) {
      q.push(i);
      mark[i] = true;
    }
  }
  while (!q.empty()) {
    int now = q.front();
    q.pop();
    for (auto x : g[now]) {
      if (points[now].special + 1 == points[x].special && !mark[x]) {
        q.push(x);
        mark[x] = true;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (!mark[i]) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (mark[j] && ans[j] == 0) {
        ans[j] = i + 1;
        break;
      }
    }
  }
  for (int i = 0; i < n; i++) { cout << points[ans[i] - 1].x << " " << points[ans[i] - 1].y << endl; }
  return 0;
}
