import sys

input = sys.stdin.readline

m, k, n, s = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = [0] * 500001
for j in b: c[j] += 1
for j in a: c[j] -= 1
if min(c) < 0: print(-1); exit()
cnt = 0
for j in c:
  cnt += max(0, -j)
  if cnt == n: break
print(cnt)
for j in range(len(c)):
  if c[j] < 0:
    cnt -= 1
    print(j, end=' ')
    if cnt == 0: break
