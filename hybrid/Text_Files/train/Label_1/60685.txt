import sys
input = sys.stdin.readline

n, k, q = (int(x) for x in input().split())
cnt = [0] * 200000
ans = []
for _ in range(n):
    l, r = (int(x) for x in input().split())
    cnt[l-1] += 1
    cnt[r] -= 1
for i in range(1, 200000):
    cnt[i] += cnt[i - 1]
    if cnt[i] >= k:
        ans.append(i)

for _ in range(q):
    a, b = (int(x) for x in input().split())
    l = a - 1 if a > 1 else 0
    r = b
    print(ans[r] - ans[l] if l <= r else 0)
