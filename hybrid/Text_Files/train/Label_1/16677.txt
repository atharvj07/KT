#include <bits/stdc++.h>
using namespace std;
vector<int> solve(int n, vector<int> &q) {
    if (n == 2) return q[0] > 0 ? vector<int>{2, 1} : vector<int>{1, 2};
    vector<int> p(n);
    p[0] = 1;
    for (int i = 0; i < n - 1; i++) {
        p[i + 1] = p[i] + q[i];
        if (p[i + 1] < 1 || p[i + 1] > n) return {};
    }
    return p;
}
int main() {
    int n;
    cin >> n;
    vector<int> q(n - 1);
    for (int i = 0; i < n - 1; i++) cin >> q[i];
    vector<int> ans = solve(n, q);
    if (ans.size() == 0) {
        cout << -1 << endl;
    } else {
        for (int i = 0; i < n; i++) cout << ans[i] << " ";
        cout << endl;
    }
    return 0;
}
