import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long x = sc.nextLong();
            if (x == 1 || x == 4 || x == 6 || x == 8) {
                System.out.println("NO");
            } else if (x % 2 == 0 && x % 4 == 2) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
            }
        }
    }
}
