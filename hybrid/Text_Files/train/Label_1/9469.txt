#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll maxn = 2e5 + 5;
#define pii pair<ll, ll>
#define fi first
#define se second
vector<ll> a;
ll n, r[maxn];
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    a.resize(n);
    for (ll &i : a) cin >> i;
    set<ll> s;
    for (ll i = 0; i < n; i++) {
        auto it = s.lower_bound(a[i]);
        if (it != s.begin()) {
            it--;
            if (*it + 1 == a[i])
                r[i] = 1;
            else
                r[i] = 0;
        } else
            r[i] = 0;
        s.insert(a[i]);
    }
    ll inc = 0, dec = 0;
    vector<ll> tin, tdec;
    for (ll i = 0; i < n; i++) {
        if (r[i] == 0) {
            dec++;
            tdec.push_back(a[i]);
        } else {
            inc++;
            tin.push_back(a[i]);
        }
    }
    if (inc == 0 || dec == 0 || inc + dec != n)
        cout << "NO" << endl;
    else {
        cout << "YES" << endl;
        cout << inc << endl;
        for (auto &i : tin) cout << i << " ";
        cout << endl;
        cout << dec << endl;
        for (auto &i : tdec) cout << i << " ";
        cout << endl;
    }
    return 0;
}
