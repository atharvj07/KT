t = [0 for i in range(65)]
n = int(input())
a = list(map(int, input().split()))
ans = 0
for i in a:
    for j in range(64):
        if (i & (1 << j)): t[j] += 1
for i in a:
    for j in range(64):
        if not t[j]: continue
        if i & (1 << j):
            ans += t[j] - 1
        else:
            ans += t[j]
print(ans)
