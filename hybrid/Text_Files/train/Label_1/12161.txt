#include <bits/stdc++.h>
using namespace std;
#define EPS 1e-6

long long a, h, w;
long double brute() {
  // brute a solution to find a value that satisfies the conditions

  for (long double x = 0;; x += EPS) {
    long double col = a + 2 * x;
    long double row = a + 2 * x;
    long long mx_cols = floor((w + x) / col);
    if (mx_cols == 0) break;
    long long mx_rows = floor((h + x) / row);
    if (mx_rows == 0) break;
    if (fabs(mx_cols * col - w) <= x && fabs(mx_rows * row - h) <= x) {
      return x;
    }
  }
  return -1;
}

int main() {
  cin >> a >> h >> w;
  long double ans = brute();
  if (ans == -1)
    cout << -1 << endl;
  else
    cout << setprecision(10) << fixed << ans << endl;
  return 0;
}
