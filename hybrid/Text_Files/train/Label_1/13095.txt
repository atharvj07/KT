#include <iostream>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int press = 0;
  while (n != m) {
    if (n > m) {
      n--;
      press++;
    } else {
      if (n % 2 == 0) {
        n /= 2;
        press++;
      } else {
        n--;
        press++;
      }
    }
  }
  cout << press;
  return 0;
}
