def solve(s):
    s = list(s)
    n = len(s)
    for i in range(n):
        if s[i] == '?':
            for ch in 'abc':
                if (i == 0 or s[i-1] != ch) and (i == n-1 or s[i+1] != ch):
                    s[i] = ch
                    break
    for i in range(1, n):
        if s[i] == s[i-1]:
            return "-1"
    return "".join(s)

t = int(input().strip())
for _ in range(t):
    s = input().strip()
    print(solve(s))
