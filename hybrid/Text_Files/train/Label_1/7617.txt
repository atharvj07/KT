#include <bits/stdc++.h>
using namespace std;
const int M = 1e9 + 7;
vector<int> d;
int main() {
  string p;
  cin >> p;
  for (char c : p) d.push_back(c - '0');
  int n = d.size(), i = n - 1, a = 0, b = 0, ans = 0;
  while (i >= 0) {
    if (i > 0) {
      a = (a * 10 + d[i]) % 25;
      b = (d[i] + b * 10) % 25;
      if (a == 0 || b == 0) {
        if (d[i - 1] % 2 == 0 || ((d[i - 1] - '0') * 10 + d[i]) % 25 == 0)
          ans++;
        else if ((d[i] - '0') * 10 + d[i - 1] % 25 == 0) {
          swap(d[i], d[i - 1]);
          i--;
          ans++;
        } else {
          i = -1;
          break;
        }
      }
    } else {
      if ((d[i] - '0') % 25 == 0)
        ans++;
      else
        i = -1;
    }
    i--;
  }
  reverse(d.begin(), d.end());
  string s = "";
  i = 0;
  while (i < n && d[i] == 0) i++;
  while (i < n) s += to_string(d[i++]);
  if (s == "")
    s = "0";
  if (i == -1)
    cout << -1 << endl;
  else
    cout << ans << endl;
  return 0;
}
