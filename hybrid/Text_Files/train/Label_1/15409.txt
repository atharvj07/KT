#include <bits/stdc++.h>

typedef long long LL;

#define x first
#define y second
#define pb push_back

const LL N = 1e6 + 5;
const LL Mod = 1e6;

std::vector<std::pair<LL, LL>> spells;

LL solve(LL t, LL h) {
    LL ans = 0, need = t;
    for (auto spell : spells) {
        LL sec = need / spell.y;
        if (sec >= 1) {
            ans += sec * spell.x;
            need -= sec * spell.y;
        }
    }
    return ans >= h;
}

int main() {
    LL q, m;
    std::cin >> q >> m;
    while (q--) {
        LL k, a, b;
        std::cin >> k >> a >> b;
        a = (a + spells.size()) % Mod + 1;
        b = (b + spells.size()) % Mod + 1;
        if (k == 1) spells.pb(std::make_pair(a, b));
        else std::cout << (solve(a, b) ? "YES\n" : "NO\n");
    }
    return 0;
}
