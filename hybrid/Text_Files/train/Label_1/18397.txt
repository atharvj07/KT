#include <algorithm>
#include <vector>
#include <iostream>
#include <cmath>
using namespace std;
const int MAXN = 1e6 + 1;
int a[MAXN], p[MAXN], pcnt;
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    p[a[i]] = i + 1;
  }
  for (int i = 2; i <= m; i++) {
    if (!p[i]) continue;
    for (int j = i * 2; j <= m; j += i) p[j] = p[i];
  }
  vector<int> primes;
  for (int i = 2; i <= m; i++) {
    if (p[i] > 0) primes.push_back(i);
  }
  int l = 1;
  vector<int> ans;
  for (auto &v : primes) {
    while (l * v <= m && p[l * v] > 0) {
      if (p[l * v] > p[l]) {
        l *= v;
        ans.clear();
      }
      if (p[l * v] == p[l]) ans.push_back(p[l * v]);
      p[l * v] = -1;
    }
  }
  cout << l << " " << ans.size() << "\n";
  for (auto &v : ans) cout << v << " ";
}
