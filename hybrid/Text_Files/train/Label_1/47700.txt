import java.util.*;
public class Main {
    final static int mod = 1000000007;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        System.out.println(fastpow(n, k) - 1);
    }
    public static long fastpow(int n, int k) {
        if (k == 0) {
            return 1;
        }
        if (k == 1) {
            return n;
        }
        long v = fastpow(n, k / 2);
        v = (v * v) % mod;
        if (k % 2 == 1) {
            v = (v * n) % mod;
        }
        return v;
    }
}
