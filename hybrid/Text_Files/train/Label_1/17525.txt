#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  string s;
  cin >> n >> s;
  vector<int> v[256];
  for (int i = 0; i < n; i++) v[s[i]].push_back(i);
  int k = 0;
  string t = "";
  for (int i = 'a'; i <= 'z'; i++) {
    if (v[i].size() % 2) {
      int pos = v[i].size() / 2;
      t += s[v[i][pos]];
      v[i].erase(v[i].begin() + pos);
      k++;
    }
    for (int j = 0; j < v[i].size(); j += 2) {
      t += s[v[i][j]];
      t += s[v[i][j + 1]];
      k++;
    }
  }
  for (int i = 'A'; i <= 'Z'; i++) {
    if (v[i].size() % 2) {
      int pos = v[i].size() / 2;
      t += s[v[i][pos]];
      v[i].erase(v[i].begin() + pos);
      k++;
    }
    for (int j = 0; j < v[i].size(); j += 2) {
      t += s[v[i][j]];
      t += s[v[i][j + 1]];
      k++;
    }
  }
  for (int i = '0'; i <= '9'; i++) {
    if (v[i].size() % 2) {
      int pos = v[i].size() / 2;
      t += s[v[i][pos]];
      v[i].erase(v[i].begin() + pos);
      k++;
    }
    for (int j = 0; j < v[i].size(); j += 2) {
      t += s[v[i][j]];
      t += s[v[i][j + 1]];
      k++;
    }
  }
  cout << k << '\n' << t;
  return 0;
}
