import java.io.*;
import java.util.*;

class Main {
  int n, m;
  List<Integer>[] adj;
  boolean[] used;

  void read() {
    Scanner in = new Scanner(System.in);
    n = in.nextInt();
    m = in.nextInt();
    adj = new List[n + 1];
    for (int i = 0; i <= n; i++) adj[i] = new ArrayList<>();
    for (int i = 0; i < m; i++) {
      int u = in.nextInt(), v = in.nextInt();
      adj[u].add(v);
      adj[v].add(u);
    }
  }

  boolean dfs(int u) {
    used[u] = true;
    for (int v : adj[u]) {
      if (!used[v]) {
        if (!dfs(v)) return false;
      } else if (used[v] && !inCycle[v]) {
        return false;
      }
    }
    inCycle[u] = true;
    return true;
  }

  int[] parent;
  boolean[] inCycle;

  void findCycle() {
    parent = new int[n + 1];
    inCycle = new boolean[n + 1];
    used = new boolean[n + 1];
    for (int i = 1; i <= n; i++) {
      if (!used[i] && !dfs(i)) {
        List<Integer> cycle = new ArrayList<>();
        int u = i;
        while (!used[u]) {
          cycle.add(u);
          used[u] = true;
          u = parent[u];
        }
        cycle.add(u);
        for (int j = 0; j < cycle.size(); j++) {
          int v = cycle.get(j);
          int w = cycle.get((j + 1) % cycle.size());
          if (v < w) {
            adj[v].add(w);
            adj[w].add(v);
          }
        }
      }
    }
  }

  void addEdge(int u, int v) {
    if (u < v) {
      adj[u].add(v);
      adj[v].add(u);
    } else {
      adj[v].add(u);
      adj[u].add(v);
    }
  }

  void solve() {
    findCycle();
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
      if (adj[i].size() < 2) {
        cnt++;
      }
    }
    System.out.println("YES");
    System.out.println(cnt / 2);
    for (int i = 1; i <= n; i++) {
      if (adj[i].size() == 1) {
        addEdge(i, adj[i].get(0));
        cnt -= 2;
        if (cnt == 0) break;
      }
    }
  }

  public static void main(String[] args) {
    Main m = new Main();
    m.read();
    m.solve();
  }
}
