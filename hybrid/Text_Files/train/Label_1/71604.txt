w, h = map(int, input().split())
im = []
for _ in range(h):
    im.append(input())
rot_im = [['.'] * w for _ in range(h)]
for i in range(h):
    for j in range(w):
        rot_im[i][j] = im[h - j - 1][i]
h_im = [['.'] * w for _ in range(h)]
for i in range(h):
    for j in range(w):
        h_im[i][j] = rot_im[i][w - j - 1]
z_im = [['.'] * (2 * w) for _ in range(2 * h)]
for i in range(2 * h):
    for j in range(2 * w):
        z_im[i][j] = h_im[i // 2][j // 2]
for i in range(2 * h):
    print(''.join(z_im[i]))
