import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long k = sc.nextLong();
        int dig = 1, len = 1;
        long a = 1, b = 10;
        while (k > len) {
            k -= len;
            a = b;
            b *= 10;
            len = (int) (b - a) * dig;
            dig++;
        }
        System.out.println((a + (k - 1) / dig + "").charAt((int) ((k - 1) % dig)));
    }
}
