#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int maxn = 2e5 + 5;
const ll mod = 1e9 + 7;
ll ans[maxn];

int main() {
  int T;
  cin >> T;
  while (T--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
    }
    memset(ans, 0, sizeof(ans));
    for (int i = 0; i < n; i++) {
      ans[a[i]]++;
    }
    ll res = 0;
    for (int i = 0; i < maxn; i++) {
      res += (ans[i] * (ans[i] - 1)) / 2;
    }
    cout << res << '\n';
  }
  return 0;
}
