from sys import stdin

n, I = map(int, stdin.readline().split())
a = [int(x) for x in stdin.readline().split()]
a.sort()
K = len(set(a))

l, r = 0, 10 ** 9
ans = n

while l <= r:
    m = (l + r) // 2
    ind = [i for i in range(n) if a[i] < m or a[i] > m + 1]

    k = len(set([a[i] for i in ind] + [m, m + 1]))
    
    if k * ((n - len(ind))//8 + 1) <= I:
        ans = len(ind)
        r = m - 1
    else:
        l = m + 1

print(ans)
