#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, A, B;
  cin >> n >> A >> B;
  vector<int> s(n);
  for (int i = 0; i < n; i++) cin >> s[i];
  int sum = 0;
  for (int i : s) sum += i;
  int mx = A * s[0] / sum;
  for (int i = 0; i < n; i++)
    if (A * s[0] / (sum - s[i]) > mx) mx = A * s[0] / (sum - s[i]);
  cout << (mx < B ? (A - mx + B - 1) / B : 0) << endl;
  return 0;
}
