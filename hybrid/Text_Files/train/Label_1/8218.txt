#include<bits/stdc++.h>
#define LL long long
using namespace std;
LL a, b;
void solve(){
    cout << "YES\n";
    vector<LL>ans;
    ans.push_back(a);
    while(a < b){
        LL x = b / a;
        if(b % a == 0){ x--; }
        while(x > 1){
            ans.push_back(a * x);
            a *= x;
            x = b / a;
            if(b % a == 0){ x--; }
        }
        a = a * 10 + 1;
        ans.push_back(a);
    }
    cout << ans.size() << endl;
    for(LL p : ans){
        cout << p << ' ';
    }
    cout << endl;
}
int main(){
    cin >> a >> b;
    if(a > b || (b - a) % 9 != 0){
        cout << "NO\n";
    }
    else{ solve(); }
    return 0;
}
