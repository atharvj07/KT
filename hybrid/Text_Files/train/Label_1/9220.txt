#include <iostream>
#include <vector>
using namespace std;
typedef long long LL;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> ans(n);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j <= i; j++) {
      bool is_ok = true;
      for (int k = i; k > j; k--) {
        if (s[k] < s[k - j + i - k]) {
          is_ok = false;
          break;
        }
      }
      if (is_ok) {
        ans[i] = 1;
        break;
      }
    }
  }
  for (auto i : ans) cout << i << ' ';
  cout << '\n';
  return 0;
}
