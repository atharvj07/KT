import sys
input = sys.stdin.readline

def isprime(n):
    if n < 2: return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0: return False
    return True

n = int(input())
a = list(map(int, input().split()))
ans = []
for i in range(n - 1):
    for j in range(i + 1, n):
        if isprime(j - i + 1):
            a[i], a[j] = a[j], a[i]
            ans.append((i + 1, j + 1))
print(len(ans))
for i in ans: print(*i)
