#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int max_chest = 0, max_biceps = 0, max_back = 0;
    for (int i = 0; i < n; i++) {
        if (i % 3 == 0)
            max_chest = max(max_chest, a[i]);
        else if (i % 3 == 1)
            max_biceps = max(max_biceps, a[i]);
        else
            max_back = max(max_back, a[i]);
    }
    if (max_chest > max_biceps && max_chest > max_back)
        cout << "chest";
    else if (max_biceps > max_chest && max_biceps > max_back)
        cout << "biceps";
    else
        cout << "back";
}
