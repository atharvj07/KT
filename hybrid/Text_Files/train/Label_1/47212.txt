import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int[][] dp = new int[n + 1][a * n + 1];
        dp[0][0] = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= a * n; j++) {
                dp[i + 1][j] += dp[i][j];
                if (j >= x[i]) {
                    dp[i + 1][j] += dp[i][j - x[i]];
                }
            }
        }
        System.out.println(dp[n][a * n]);
    }
}
