#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < n; ++i)
#define REP(i, a, n) for (int i = a; i < n; ++i)
#define all(x) (x).begin(), (x).end()
#define trav(a, x) for (auto &a : x)
#define sz(x) (int) (x).size()

int main()
{
    cin.tie(0)->sync_with_stdio(0);
    cin.exceptions(cin.failbit);
    int n; cin >> n;
    string s; cin >> s;
    vector<int> ans;
    int x = -1, y = -1;
    for (int i = 0; i < n; ++i)
    {
        if (i > 0 && s[i] == s[i - 1])
        {
            if (x == -1) ans.push_back(i), x = i;
            y = i;
        }
    }
    if (x == -1) cout << 0 << '\n' << s << '\n';
    else
    {
        if (x == 0) s[x] = (s[x] == 'R' ? 'G' : 'R');
        else if (y == n - 1) s[y] = (s[y] == 'G' ? 'R' : 'G');
        else
        {
            if (s[x] == s[x + 1]) s[x] = (s[x] == 'B' ? 'G' : 'B');
            else s[y] = (s[y] == 'B' ? 'R' : 'B');
        }
        cout << 1 << '\n' << s << '\n';
    }
}
