#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
const ll INF = 2e18;

int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<ll> a(n);
    for (auto &x : a) { cin >> x; }
    sort(a.begin(), a.end());
    vector<pair<ll, ll>> pref(n);
    for (int i = 0; i < n; i++) {
      pref[i] = {a[i], a[i]};
      if (i) {
        pref[i].first += pref[i - 1].first;
        pref[i].second += pref[i - 1].second;
      }
    }
    ll ans = 0;
    for (int i = 0; i < n; i++) {
      ll l = 0, r = n - 1;
      ll l_ans = 0;
      while (l <= r) {
        ll mid = (l + r) / 2;
        ll sum = 0;
        if (mid) sum = pref[mid - 1].second;
        if (k >= mid) {
          l_ans = max(l_ans, pref[mid].first - sum);
          l = mid + 1;
        } else
          r = mid - 1;
      }
      l = 0, r = n - 1;
      ll r_ans = 0;
      while (l <= r) {
        ll mid = (l + r) / 2;
        ll sum = 0;
        if (mid) sum = pref[mid - 1].first;
        if (k >= n - mid - 1) {
          r_ans = max(r_ans, pref[n - 1].second - sum);
          r = mid - 1;
        } else
          l = mid + 1;
      }
      ans = max(ans, max(l_ans, r_ans));
    }
    cout << ans << '\n';
  }
  return 0;
}
