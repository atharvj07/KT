import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), b = sc.nextInt();
        int[] p = new int[n], k = new int[n], a = new int[n], c = new int[m + 1], d = new int[m + 1];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
            k[i] = sc.nextInt();
            a[i] = sc.nextInt();
            for (int j = 0; j < a[i]; j++) {
                c[sc.nextInt()] = i;
            }
        }
        int[] q = new int[m + 1];
        for (int i = 1; i <= m; i++) {
            if (c[i] != 0) {
                d[i] = p[c[i]];
                q[i] = d[i];
            }
        }
        for (int i = 2; i <= m; i++) {
            for (int j = 1; j < i; j++) {
                if (c[j] != 0 && c[i] != 0 && c[j] != c[i]) {
                    if (d[j] + d[i] > d[i]) {
                        d[i] = d[j] + d[i];
                        q[i] = q[j] + k[c[i]];
                    }
                }
            }
        }
        int ans = (int) 1e9;
        for (int i = 1; i <= m; i++) {
            if (d[i] == 0) {
                continue;
            }
            ans = Math.min(ans, q[i] + b * k[c[i]]);
        }
        System.out.println(ans == (int) 1e9 ? -1 : ans);
    }
}
