#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin >> n;
    int a[n];
    for (int &i : a) { cin >> i; }
    pair<int, int> b[6] = {{4, 0}, {8, 0}, {15, 0}, {16, 0}, {23, 0}, {42, 0}};
    for (int &i : a) {
        if (i == 4)
            b[0].second++;
        if (i == 8)
            b[1].second++;
        if (i == 15)
            b[2].second++;
        if (i == 16)
            b[3].second++;
        if (i == 23)
            b[4].second++;
        if (i == 42)
            b[5].second++;
    }
    int cnt = 0;
    int mx = 0;
    for (int i = 0; i < 6; i++) {
        mx = max(b[i].second / 6, mx);
    }
    for (int i = 0; i < 6; i++) { b[i].second %= 6; }
    sort(b, b + 6);
    for (int i = 0, j = 5; i <= 1; i++, j--) {
        mx += min(b[i].second, b[j].second);
    }
    cnt = n - mx * 6;
    cout << cnt << '\n';
    return 0;
}
