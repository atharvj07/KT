import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = [0] * n
covered = [0] * n

for _ in range(m):
    l, r = map(int, input().split())

    for j in range(l-1, r):
        covered[j] += 1
        if a[j] == 0:
            a[j] = -1

mex = 0
for i in range(n):
    if covered[i] == 0:
        a[i] = mex
        mex += 1

    elif a[i] == -1:
        a[i] = mex
        mex += 1

print(mex)
print(*a)
