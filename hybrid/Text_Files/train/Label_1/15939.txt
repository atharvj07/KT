#include<bits/stdc++.h>
#define ll long long int
using namespace std;
map<ll,ll> mp;
int main()
{
    ll m;
    cin>>m;
    ll x;
    for(ll i=0;i<m;i++)
    {
        cin>>x;
        ll ans=0;
        for(ll j=599;j>=0;j--)
        {
            if(!(x&(1ll<<j))) continue;
            else
            {
                if(mp.count(x^(1ll<<j)))
                {
                    ans++;
                    if(ans==1) cout<<mp[x^(1ll<<j)]+1;
                    else cout<<" "<<mp[x^(1ll<<j)]+1;
                }
            }
        }
        mp[x]=i;
        if(ans==0) cout<<"0";
        cout<<endl;
    }
}
