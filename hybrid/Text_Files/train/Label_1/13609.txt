#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main()
{
    int n,v; cin>>n>>v;
    vector<vector<int>> a(n);
    for (int i=0;i<n;i++)
    {
        int k; cin>>k;
        for (int j=0;j<k;j++)
        {
            int t; cin>>t;
            a[i].push_back(t);
        }
        sort(a[i].begin(),a[i].end());
    }
    vector<int> ans;
    for (int i=0;i<n;i++)
    {
        if (a[i][0]<=v) ans.push_back(i+1);
    }
    cout<<ans.size()<<endl;
    for (int x: ans) cout<<x<<" ";
    cout<<endl;
    return 0;
}
