n, q = map(int, input().split())
a = sorted(list(map(int, input().split())))
b = []
for i in range(q):
    l, r = map(int, input().split())
    b.append([r - l + 1, l - 1, r])

b = sorted(b)
a = reversed(a)
ans = 0
for x, y, z in b:
    t = min(x, next(a))
    ans += t * (z - y)
    x -= t
print(ans)
