import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] a = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextInt();
        }
        int x = 0;
        for (int i = 0; i < N; i++) {
            x ^= a[i];
        }
        for (int i = 0; i < N; i++) {
            System.out.print((x ^ a[i]) + " ");
        }
        System.out.println();
    }
}
