import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class SegmentTree {

    private static class Segment implements Comparable<Segment> {
        int left;
        int right;

        public Segment(int left, int right) {
            this.left = left;
            this.right = right;
        }

        @Override
        public int compareTo(Segment other) {
            return Integer.compare(this.left, other.left);
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        Segment[] segments = new Segment[n];
        for (int i = 0; i < n; i++) {
            int left = scanner.nextInt();
            int right = scanner.nextInt();
            segments[i] = new Segment(left, right);
        }

        Arrays.sort(segments);

        List<Integer> connected = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (intersects(segments[i], segments[j])) {
                    connected.add(i);
                    connected.add(j);
                }
            }
        }

        int[] parent = new int[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
        }

        for (int i = 0; i < connected.size(); i += 2) {
            int u = connected.get(i);
            int v = connected.get(i + 1);
            union(parent, u, v);
        }

        int components = 0;
        for (int i = 0; i < n; i++) {
            if (parent[i] == i) {
                components++;
            }
        }

        System.out.println(components == 1 ? "YES" : "NO");
    }

    private static boolean intersects(Segment s1, Segment s2) {
        return s1.right > s2.left && s2.right > s1.left && s1 != s2 && !contains(s1, s2) && !contains(s2, s1);
    }

    private static boolean contains(Segment s1, Segment s2) {
        return s1.left <= s2.left && s1.right >= s2.right;
    }

    private static int find(int[] parent, int u) {
        if (parent[u] == u) {
            return u;
        }
        return parent[u] = find(parent, parent[u]);
    }

    private static void union(int[] parent, int u, int v) {
        u = find(parent, u);
        v = find(parent, v);
        if (u != v) {
            parent[u] = v;
        }
    }
}
