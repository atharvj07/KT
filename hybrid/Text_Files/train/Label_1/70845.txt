import sys

n = int(input())
a = [int(x) for x in input().split()]
b = [0] * n
c = [0] * n

if len(set(a)) > len(a) // 2 + len(a) % 2:
    print(-1)
    sys.exit()

l = r = 0
for i in range(n):
    if i and a[i] != a[i - 1]:
        r = i
        b[r] = l
        l = r
        c[r] = b[r] - c[r]
        c[b[r]] += c[r]
else:
    b[r] = l
    c[r] = b[r] - c[r]
    c[b[r]] += c[r]

print(r - l + 1)
for i in range(l, r + 1):
    print(b[i] + 1, i + 1)
