import sys
r = sys.stdin.readline
n, m, s = map(int, r().split())
r = [[]] + [[int(x) for x in r().split()] for i in range(m)]
g, p = [1 if x == s else 0 for x in range(n + 1)], [s]
for u, v in r[1:]:
    g[v] = 1
    p += [v] if g[u] and not g[v] else []
    g[u] = 1
print(len([1 for g in p if not g]))
