def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def gcdSum(x):
    return gcd(x, sum([int(digit) for digit in str(x)]))

def find_gcdSum(n):
    while gcdSum(n) == 1:
        n += 1
    return n

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    print(find_gcdSum(n))
