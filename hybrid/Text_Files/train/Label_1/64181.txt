import sys
t=int(input())
for _ in range(t):
    n=int(input())
    g=[[]for _ in range(n)]
    for _ in range(n-1):
        u,v=map(int,input().split())
        u-=1
        v-=1
        g[u]+=[v]
        g[v]+=[u]
    vis=[0 for _ in range(n)]
    cur=0
    q=[]
    vis[cur]=1
    q+=[(cur,0)]
    relab=[0 for _ in range(n)]
    idx=0
    while q:
        now,par=q.pop(0)
        relab[idx]=now
        idx+=1
        for u in g[now]:
            if vis[u]:
                continue
            vis[u]=1
            if now^u<=min(now,u):
                q+=[(u,now)]
            else:
                q+=[(u,par)]
    print(*[x+1 for x in relab])
