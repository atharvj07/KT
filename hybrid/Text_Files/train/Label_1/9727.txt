#include <bits/stdc++.h>

using namespace std;

int main() {
    string s;
    cin >> s;
    vector<vector<int>> pos(26);
    for (int i = 0; i < 27; i++) {
        cin >> s;
        for (int j = 0; j < 13; j++) {
            pos[s[j] - 'A'].push_back(i * 13 + j);
        }
    }
    vector<int> ans;
    vector<bool> vis(26, false);
    function<bool(int)> dfs = [&](int u) {
        if (u == 26) return true;
        for (int v : pos[u]) {
            if (!vis[v]) {
                vis[v] = true;
                ans.push_back(v);
                if (dfs(u + 1)) return true;
                vis[v] = false;
                ans.pop_back();
            }
        }
        return false;
    };
    if (dfs(0)) {
        for (int i = 0; i < 2; i++) {
            for (int j = 0; j < 13; j++) {
                cout << (char)(s[ans[i * 13 + j]] + 'A');
            }
            cout << endl;
        }
    } else {
        cout << "Impossible" << endl;
    }
    return 0;
}
