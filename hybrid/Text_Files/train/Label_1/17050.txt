#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 998244353;
struct portal {
    ll pos, teleport, init;
};
istream& operator>>(istream& is, portal& a) {
    return is >> a.pos >> a.teleport >> a.init;
}
ll modsum(ll a, ll b) { return (a + b) % mod; }
ll modmul(ll a, ll b) { return (a * b) % mod; }
int main() {
    ll n;
    cin >> n;
    vector<portal> port(n);
    for (auto& a : port) cin >> a;
    ll ans = 0, cur = 0;
    for (auto a : port) {
        if (!a.init) {
            ans = modsum(ans, a.pos - cur);
            cur = a.pos;
            continue;
        }
        ans = modsum(ans, modmul(a.pos - cur, 2));
        ans = modsum(ans, a.teleport - a.pos);
        cur = a.teleport;
    }
    ans = modsum(ans, port.back().pos + 1 - cur);
    cout << ans << '\n';
    return 0;
}
