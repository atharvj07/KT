#include<bits/stdc++.h>
using namespace std;
const int N=4010;
int n;
double a[N],ans;
bool vis[N];
int main(){
    cin>>n;
    for(int i=1;i<=2*n;++i) cin>>a[i];
    sort(a+1,a+1+2*n,greater<double>());
    for(int i=1;i<=n;++i){
        double x=max(floor(a[i]),floor(a[2*n-i+1])),y=min(ceil(a[i]),ceil(a[2*n-i+1]));
        if(x+y>a[i]+a[2*n-i+1]) swap(x,y);
        ans+=abs(a[i]-x);
        ans+=abs(a[2*n-i+1]-y);
        a[i]=x;
        a[2*n-i+1]=y;
        vis[i]=vis[2*n-i+1]=1;
    }
    for(int i=1;i<=2*n;++i){
        if(!vis[i]){
            a[i]=floor(a[i]);
            ans+=abs(a[i]-floor(a[i]));
        }
    }
    cout<<fixed<<setprecision(3)<<ans<<endl;
    return 0;
}
