#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 4, M = 1e5 + 4;
int n, m, q, cnt[N][N];
string s, t;
int main() {
    cin >> n >> m >> q;
    cin >> s >> t;
    for(int i = 1;i <= n;i++) {
        for(int j = 1;j <= n - m + 1;j++) {
            cnt[i][j] = cnt[i - 1][j];
            if(s.substr(j, m) == t) {
                cnt[i][j]++;
            }
        }
    }
    while(q--) {
        int l, r;
        cin >> l >> r;
        cout << cnt[r][l - m + 1] - cnt[l - 1][l - m + 1] << '\n';
    }
    return 0;
}
