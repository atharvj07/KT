import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] edges = new int[m][3];
        for (int i = 0; i < m; i++) {
            edges[i][0] = sc.nextInt() - 1;
            edges[i][1] = sc.nextInt() - 1;
            edges[i][2] = sc.nextInt();
        }
        sc.close();

        int[] deg = new int[n];
        for (int[] edge : edges) {
            deg[edge[0]]++;
            deg[edge[1]]++;
        }

        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (deg[i] == 1) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
