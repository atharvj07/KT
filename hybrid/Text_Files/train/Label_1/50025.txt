import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
    static final int MAX = 501;
    static final int[] dr = {-1, 0, -1, 0};
    static final int[] dc = {0, -1, 0, 1};
    static int N, r, d;
    static int[][] pizza = new int[MAX][MAX];
    static int[][] pizza2 = new int[MAX][MAX];
    static boolean[][] flag = new boolean[MAX][MAX];
    static int ans;

    public static void dfs(int x, int y) {
        for (int i = 0; i < 4; i++) {
            int nr = x + dr[i];
            int nc = y + dc[i];
            if (nr < 0 || nr >= MAX || nc < 0 || nc >= MAX)
                continue;
            if (pizza2[nr][nc] == -1 || flag[nr][nc])
                continue;
            if (pizza2[nr][nc] == 0 && pizza[nr][nc] > 0)
                ans++;
            flag[nr][nc] = true;
            dfs(nr, nc);
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        r = Integer.parseInt(st.nextToken());
        d = Integer.parseInt(st.nextToken());
        N = Integer.parseInt(br.readLine());
        for (int i = 0; i < MAX; i++) {
            for (int j = 0; j < MAX; j++) {
                if (j - d < i - r)
                    pizza[i][j] = -1;
                else if (j + d > i + r)
                    pizza[i][j] = -1;
                else
                    pizza[i][j] = 0;
            }
        }
        for (int i = 0; i < MAX; i++)
            for (int j = 0; j < MAX; j++)
                pizza2[i][j] = pizza[i][j];
        for (int i = 0; i < N; i++) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            int rr = Integer.parseInt(st.nextToken());
            for (int j = 0; j < MAX; j++)
                for (int k = 0; k < MAX; k++) {
                    if (j - d < k - r - rr)
                        pizza2[j][k] = -1;
                    else if (j + d > k + r + rr)
                        pizza2[j][k] = -1;
                    else
                        pizza2[j][k] = 0;
                }
        }
        ans = 0;
        flag = new boolean[MAX][MAX];
        for (int i = 0; i < MAX; i++) {
            for (int j = 0; j < MAX; j++) {
                if (pizza2[i][j] == -1)
                    continue;
                if (!flag[i][j] && pizza2[i][j] > 0) {
                    dfs(i, j);
                }
            }
        }
        System.out.print(ans);
    }
}
