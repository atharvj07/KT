import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] firstLine = br.readLine().split(" ");
        int n = Integer.parseInt(firstLine[0]);
        int platformStrength = Integer.parseInt(firstLine[1]);

        ArrayList<Parcel> parcels = new ArrayList<>();
        Map<Integer, Integer> outTimeMap = new HashMap<>();

        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            int inTime = Integer.parseInt(line[0]);
            int outTime = Integer.parseInt(line[1]);
            int weight = Integer.parseInt(line[2]);
            int strength = Integer.parseInt(line[3]);
            int value = Integer.parseInt(line[4]);

            parcels.add(new Parcel(inTime, outTime, weight, strength, value));
            outTimeMap.put(outTime, i);
        }

        Collections.sort(parcels, (a, b) -> a.inTime - b.inTime);

        int platformWeight = 0;
        int totalValue = 0;
        ArrayList<Parcel> platform = new ArrayList<>();

        for (Parcel parcel : parcels) {
            while (!platform.isEmpty() && platform.get(0).outTime <= parcel.inTime) {
                platformWeight -= platform.remove(0).weight;
            }

            if (platformWeight + parcel.weight <= platformStrength && parcel.strength >= platformWeight + parcel.weight) {
                platform.add(parcel);
                platformWeight += parcel.weight;
                totalValue += parcel.value;
            } else {
                int outTime = outTimeMap.get(parcel.outTime);
                parcels.set(outTime, new Parcel(parcel.inTime, parcel.outTime, parcel.weight, parcel.strength, 0));
            }
        }

        System.out.println(totalValue);
    }

    private static class Parcel {
        int inTime;
        int outTime;
        int weight;
        int strength;
        int value;

        public Parcel(int inTime, int outTime, int weight, int strength, int value) {
            this.inTime = inTime;
            this.outTime = outTime;
            this.weight = weight;
            this.strength = strength;
            this.value = value;
        }
    }
}
