import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        int A = scan.nextInt();
        int B = scan.nextInt();
        int H = scan.nextInt();
        int M = scan.nextInt();

        double A1 = Math.toRadians((30 * H) % 360);
        double B1 = Math.toRadians((6 * M) % 360);

        double x1 = A * Math.cos(A1);
        double y1 = A * Math.sin(A1);
        double x2 = B * Math.cos(B1);
        double y2 = B * Math.sin(B1);

        System.out.println(Math.sqrt(Math.pow((x2 - x1), 2) + Math.pow((y2 - y1), 2)));
    }
}
