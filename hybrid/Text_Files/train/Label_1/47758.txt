import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String firstLine = br.readLine();
        String[] firstLineSplit = firstLine.split(" ");
        int n = Integer.parseInt(firstLineSplit[0]);
        int k = Integer.parseInt(firstLineSplit[1]);
        int A = Integer.parseInt(firstLineSplit[2]);
        List<Senator> senators = new ArrayList<>();
        long sumB = 0;
        for (int i = 0; i < n; i++) {
            String[] senatorStr = br.readLine().split(" ");
            int b = Integer.parseInt(senatorStr[0]);
            int l = Integer.parseInt(senatorStr[1]);
            sumB += b;
            senators.add(new Senator(b, l, false));
        }
        Collections.sort(senators);
        int candies = 0;
        long sumLoyalty = 0;
        long sumBPlusCandies = sumB;
        for (Senator senator : senators) {
            if (candies == k)
                break;
            int loyalty = 100 - senator.l;
            if (candies + loyalty <= k) {
                candies += loyalty;
                sumLoyalty += senator.b;
                sumBPlusCandies += loyalty * senator.b / 10;
                senator.bribed = true;
            }
        }
        double prob = (double) A / (double) (A + sumBPlusCandies - sumLoyalty);
        System.out.printf("%.10f", prob);
    }

    static class Senator implements Comparable<Senator> {
        int b;
        int l;
        boolean bribed;

        public Senator(int b, int l, boolean bribed) {
            this.b = b;
            this.l = l;
            this.bribed = bribed;
        }

        @Override
        public int compareTo(Senator o) {
            if (this.l != o.l)
                return Integer.compare(o.l, this.l);
            else
                return Integer.compare(this.b, o.b);
        }
    }
}
