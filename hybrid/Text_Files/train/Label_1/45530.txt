import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int xv = sc.nextInt();
        int yv = sc.nextInt();
        int xp = sc.nextInt();
        int yp = sc.nextInt();
        int xw1 = sc.nextInt();
        int yw1 = sc.nextInt();
        int xw2 = sc.nextInt();
        int yw2 = sc.nextInt();
        int xm1 = sc.nextInt();
        int ym1 = sc.nextInt();
        int xm2 = sc.nextInt();
        int ym2 = sc.nextInt();
        boolean canSee = true;
        if (xw1 == xw2) {
            if (xv > xw1 || xp > xw1) {
                canSee = false;
            }
        } else if (yw1 == yw2) {
            if (yv > yw1 || yp > yw1) {
                canSee = false;
            }
        } else {
            double slope = (double) (yw2 - yw1) / (xw2 - xw1);
            double intercept = yw1 - slope * xw1;
            if (yv > slope * xv + intercept || yp > slope * xp + intercept) {
                canSee = false;
            }
        }
        if (canSee) {
            if (xm1 == xm2) {
                if ((xv > xm1 && xp < xm1) || (xv < xm1 && xp > xm1)) {
                    canSee = false;
                }
            } else if (ym1 == ym2) {
                if ((yv > ym1 && yp < ym1) || (yv < ym1 && yp > ym1)) {
                    canSee = false;
                }
            } else {
                double slope = (double) (ym2 - ym1) / (xm2 - xm1);
                double intercept = ym1 - slope * xm1;
                if ((yv > slope * xv + intercept && yp < slope * xp + intercept) || (yv < slope * xv + intercept && yp > slope * xp + intercept)) {
                    canSee = false;
                }
            }
        }
        System.out.println(canSee ? "YES" : "NO");
    }
}
