#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
vector<int> adj[N];
int n, val[N], l[N], r[N], mark[N];

int go(int c) {
    if (c == -1) return 0;
    if (mark[c]) return mark[c];
    return mark[c] = 1 + go(l[c]) + go(r[c]);
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int i;
    cin >> n;
    for (i = 1; i <= n; i++) {
        int x, y, z;
        cin >> x >> y >> z;
        val[i] = x;
        l[i] = y;
        r[i] = z;
    }
    for (i = 1; i <= n; i++) {
        if (!mark[i]) {
            cout << go(i) - 1 << endl;
        }
    }
    return 0;
}
