import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        char[] s = in.next().toCharArray();
        int[] pref = new int[s.length + 1];
        int unbalanced = 0;
        for (int i = 0; i < s.length; i++) {
            if (s[i] != '?') {
                pref[i + 1] = pref[i] + (s[i] - '0' == 1 ? 1 : -1);
                unbalanced = Math.max(unbalanced, Math.abs(pref[i + 1]));
            }
        }
        int minUnbalanced = unbalanced;
        for (int i = 0; i < s.length; i++) {
            if (s[i] == '?') {
                int tmpUnbalanced = Math.max(Math.abs(pref[i] + 1), Math.abs(pref[i] - 1));
                minUnbalanced = Math.min(minUnbalanced, tmpUnbalanced);
            }
        }
        System.out.println(minUnbalanced);
    }
}
