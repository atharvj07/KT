import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt();
        int M = in.nextInt();
        int[] a = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = in.nextInt();
        }
        int count = 0;
        for (int i = 1; i <= M; i++) {
            boolean isSemiCommonMultiple = true;
            for (int j = 0; j < N; j++) {
                if ((i % a[j]) % 2 != 0) {
                    isSemiCommonMultiple = false;
                    break;
                }
            }
            if (isSemiCommonMultiple) {
                count++;
            }
        }
        System.out.println(count);
    }
}
