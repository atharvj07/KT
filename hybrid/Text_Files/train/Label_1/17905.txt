#include <bits/stdc++.h>
using namespace std;
const int maxn = 300000 + 5;
int n, m, q, a[maxn], b[maxn], st[maxn], en[maxn];
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= m; i++) cin >> b[i];
    sort(a + 1, a + n + 1, greater<int>());
    for (int i = 1; i <= m; i++) {
        int l = 1, r = n;
        while (l <= r) {
            int mid = (l + r) >> 1;
            if (b[i] >= a[mid])
                l = mid + 1;
            else
                r = mid - 1;
        }
        st[i] = l;
    }
    int mx = 0;
    for (int i = m; i >= 1; i--) {
        mx = max(mx, a[st[i]]);
        en[i] = mx;
    }
    cin >> q;
    while (q--) {
        int x, y;
        cin >> x >> y;
        if (x == 1) {
            a[y] = y;
            sort(a + 1, a + n + 1, greater<int>());
            for (int i = 1; i <= m; i++) {
                int l = 1, r = n;
                while (l <= r) {
                    int mid = (l + r) >> 1;
                    if (b[i] >= a[mid])
                        l = mid + 1;
                    else
                        r = mid - 1;
                }
                st[i] = l;
            }
            int mx = 0;
            for (int i = m; i >= 1; i--) {
                mx = max(mx, a[st[i]]);
                en[i] = mx;
            }
        } else {
            for (int i = 1; i <= m; i++) {
                if (st[i] == st[i + y]) en[i] = en[i + y];
            }
            b[y] = x;
        }
        int ans = -1;
        for (int i = 1; i <= m; i++) {
            ans = max(ans, en[i]);
        }
        cout << ans << '\n';
    }
    return 0;
}
