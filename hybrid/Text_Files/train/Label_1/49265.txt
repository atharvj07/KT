import java.util.*;
class Main {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt(), m = sc.nextInt(), sum = 0;
            int[] a = new int[n], b = new int[n];
            long[] c = new long[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }
            for (int i = 0; i < n; i++) {
                b[i] = sc.nextInt();
                c[i] = (long)a[i] + ((long)b[i]) * 1000000007;
            }
            Arrays.sort(c);
            for (int i = 0; i < n; i++)
                if (m > 0) {
                    if (c[i] % 1000000007 <= m) {
                        m -= c[i] % 1000000007;
                        sum += c[i] / 1000000007;
                    } else {
                        sum += m / c[i] % 1000000007;
                        break;
                    }
                }
            if (m > 0) System.out.println(-1);
            else System.out.println(sum);
        }
    }
}
