#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int last = 0;
    bool ok = true;
    for (char c : s) {
      if (c - last > 1) ok = false;
      last = c;
    }
    cout << (ok ? "YES" : "NO") << endl;
  }
}
