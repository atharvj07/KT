import sys

n,m=map(int,input().split())
l=list(map(int,input().split()))
l.sort()
if len(set(l))>m or sum(l)>n:
	print(-1)
	sys.exit()
c=[-1]*n
ci=0
for mi in l:
	for i in range(mi):
		if c[i]==-1:
			c[i]=ci
			ci+=1
	if c[-mi:]==[-1]*(mi):
		for i in range(n-mi,n):
			if c[i]==-1:
				c[i]=ci
				ci+=1
print(*[i+1 for i in c])
