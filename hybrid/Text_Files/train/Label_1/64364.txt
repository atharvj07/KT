def solve_flower_festival(m, k, n, s, liana, schematic):
    from collections import Counter

    def can_make_schematic(workpiece):
        workpiece_count = Counter(workpiece)
        for flower, required_count in schematic_count.items():
            if workpiece_count[flower] < required_count:
                return False
        return True

    schematic_count = Counter(schematic)
    result = []
    extra_flowers = m - n * k

    for start in range(k):
        if start + (n-1)*k >= m:
            break

        to_remove = []
        current_schematic = schematic_count.copy()
        end = start
        removed = 0

        while end < m and removed <= extra_flowers:
            if current_schematic[liana[end]] > 0:
                current_schematic[liana[end]] -= 1
            else:
                to_remove.append(end + 1)
                removed += 1

            if sum(current_schematic.values()) == 0:
                segment_length = end - start + 1
                if segment_length - removed <= k:
                    needed = k - (segment_length - removed)
                    for i in range(start-1, -1, -1):
                        if liana[i] not in schematic:
                            to_remove.append(i + 1)
                            removed += 1
                            needed -= 1
                            if needed == 0:
                                break

                    if needed == 0 and len(result) == 0 or len(to_remove) < len(result):
                        result = to_remove
                break

            end += 1

    if result:
        return str(len(result)) + "\n" + " ".join(map(str, result))
    else:
        return "-1"

# Reading input and executing the solution
m, k, n, s = map(int, input().split())
liana = list(map(int, input().split()))
schematic = list(map(int, input().split()))

print(solve_flower_festival(m, k, n, s, liana, schematic))
