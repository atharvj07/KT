#include <bits/stdc++.h>

#define endl '\n'
#define ll long long

using namespace std;

typedef pair<ll, ll> cake;

const double pi = 3.1415926535897932;

ll n;
vector<cake> cakes;
double ans;

bool compare(cake a, cake b) {
    return a.first > b.first;
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);

    cin >> n;
    cakes.resize(n);
    for (ll i = 0; i < n; i++) {
        cin >> cakes[i].first >> cakes[i].second;
        cakes[i].first *= cakes[i].first;
    }
    sort(cakes.begin(), cakes.end(), compare);

    vector<double> dp(n);
    dp[0] = pi * cakes[0].first * cakes[0].second;
    for (ll i = 1; i < n; i++) {
        dp[i] = pi * cakes[i].first * cakes[i].second;
        for (ll j = i - 1; j >= 0; j--) {
            if (cakes[i].first > cakes[j].first) dp[i] = max(dp[i], dp[j] + pi * cakes[i].first * cakes[i].second);
        }
    }
    ans = *max_element(dp.begin(), dp.end());
    cout << fixed << setprecision(10) << ans << endl;
    return 0;
}
