import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.HashSet;
import java.util.Set;

public class N_ishLanguage {

    public static void main(String[] args) throws Exception {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String sentence = reader.readLine();
        int k = Integer.parseInt(reader.readLine());
        Set<Character> forbidden = new HashSet<>();
        for (int i = 0; i < k; i++) {
            String pair = reader.readLine();
            forbidden.add(pair.charAt(0));
            forbidden.add(pair.charAt(1));
        }

        int deletions = 0;
        for (int i = 0; i < sentence.length() - 1; i++) {
            char left = sentence.charAt(i);
            char right = sentence.charAt(i + 1);
            if (forbidden.contains(left) && forbidden.contains(right)) {
                deletions++;
                i++; // Skip the next character as well
            }
        }

        System.out.println(deletions);
    }
}
