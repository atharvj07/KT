n, m, k = map(int, input().split())
ma = [input() for _ in range(n)]
s = sum(x.count('.') for x in ma)
for i, row in enumerate(ma):
    for j, ch in enumerate(row):
        if ch == '.' and s > k:
            ma[i] = ma[i][:j] + 'X' + ma[i][j+1:]
            s -= 1
            k -= 1
        if k == 0: break
for row in ma: print(row)
