#include<bits/stdc++.h>
using namespace std;
void dfs(int u, int par, vector<int>& vis, vector<int>& ans, vector<vector<int>>& adj) {
	ans.push_back(u);
	vis[u] = 1;
	for (int e : adj[u]) {
		if (!vis[e]) {
			dfs(e, u, vis, ans, adj);
		}
	}
}
int main() {
	int n, m;
	cin >> n >> m;
	vector<vector<int>> adj(n + 1);
	for (int i = 0; i < m; i++) {
		int u, v;
		cin >> u >> v;
		adj[u].push_back(v);
	}
	vector<int> vis(n + 1), ans;
	bool ok = 1;
	for (int i = 1; i <= n; i++) {
		if (!vis[i]) {
			dfs(i, 0, vis, ans, adj);
		}
	}
	for (int i = 1; i <= n; i++) {
		for (int e : adj[i]) {
			if (find(ans.begin(), ans.end(), e) > find(ans.begin(), ans.end(), i)) {
				ok = 0;
			}
		}
	}
	if (!ok) {
		cout << -1 << endl;
	} else {
		for (auto x : ans) {
			cout << x << " ";
		}
		cout << endl;
	}
	return 0;
}
