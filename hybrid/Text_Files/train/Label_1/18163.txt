#include <cstdio>
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n, sum = 0;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) {
      int tmp = i, digit = i % 10;
      bool flag = true;
      while (tmp > 0) {
        if (tmp % 10 != digit) {
          flag = false;
          break;
        }
        tmp /= 10;
      }
      if (flag) sum++;
    }
    printf("%d\n", sum);
  }
  return 0;
}
