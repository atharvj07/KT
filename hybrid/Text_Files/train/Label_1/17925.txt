#include <bits/stdc++.h>
using namespace std;

int n, a, b;
vector<int> p;

int main() {
  cin >> n >> a >> b;
  if ((n + a - b) % 2) {
    cout << -1 << endl;
    return 0;
  }
  p.resize(n + 1);
  int c = (n + a - b) / 2;
  int u = 1;
  for (int i = 1; i <= n; i++) {
    p[i] = i;
    if (c) { p[i] = p[a], p[a] = i, c--; }
    if (c) { p[i] = p[b], p[b] = i, c--; }
  }
  for (int i = 1; i <= n; i++) cout << p[i] << " ";
  cout << endl;
}
