n, m, a, b = map(int, input().split())

# Calculate cost if buying single tickets for all rides
cost_single_tickets = n * a

# Calculate cost if buying m-ride tickets only
cost_m_ride_tickets = (n // m) * b + min((n % m) * a, b)

# Print the minimum of the two calculated costs
print(min(cost_single_tickets, cost_m_ride_tickets))
