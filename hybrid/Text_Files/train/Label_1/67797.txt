from collections import defaultdict, deque
import sys

def dfs(adj, vis, ans, u, p):
    vis[u] = True
    ans.append((u, p))
    for v in adj[u]:
        if not vis[v]:
            dfs(adj, vis, ans, v, u)

def bfs(adj, deg, lst):
    while lst:
        u, d = lst.popleft()
        for v in adj[u]:
            if deg[v] > 0:
                lst.append((v, deg[v]-1))
                deg[v] -= 1
                deg[u] -= 1
                ans.append((u, v))

n, m, d = map(int, input().split())
adj = defaultdict(list)
for i in range(m):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

vis = [False] * (n + 1)
ans = []
dfs(adj, vis, ans, 1, 1)
ans.reverse()

if ans[-1][1] != 1:
    print("NO")
    sys.exit(0)
deg = [0] * (n+1)
lst = deque([(1, d)])
ans.pop()

bfs(adj, deg, lst)
print("YES")
for x, y in ans:
    print(x, y)
