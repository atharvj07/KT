import java.util.Scanner;

public class OddSumArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            scanner.nextLine(); // Consume newline

            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
            }

            boolean isOddSumPossible = isOddSumPossible(a);
            System.out.println(isOddSumPossible ? "YES" : "NO");
        }
    }

    private static boolean isOddSumPossible(int[] a) {
        int sum = 0;
        for (int element : a) {
            sum += element;
        }

        return sum % 2 != 0; // Check if the sum is odd
    }
}
