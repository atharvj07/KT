import heapq

n = int(input())
loc = [list(map(int, input().split())) for _ in range(n)]
c = list(map(int, input().split()))
k = list(map(int, input().split()))
adj = [[] for _ in range(n)]
for i in range(n):
    for j in range(n):
        if i == j:
            continue
        dist = abs(loc[i][0]-loc[j][0]) + abs(loc[i][1]-loc[j][1])
        adj[i].append((dist*k[i]+dist*k[j], j))

ans = 1e18

for mask in range(1<<n):
    cost = 0
    for i in range(n):
        if mask&(1<<i):
            cost += c[i]
    cities = [i for i in range(n) if mask&(1<<i)]
    if len(cities) == 0:
        continue
    pq = [(0, cities[0])]
    vis = [0]*n
    vis[cities[0]] = 1
    while pq:
        cnt, node = heapq.heappop(pq)
        cost += cnt
        for w, nxt in adj[node]:
            if vis[nxt]:
                continue
            vis[nxt] = 1
            heapq.heappush(pq, (w, nxt))
    ans = min(ans, cost)

print(ans)
mask = bin(ans)[2:].zfill(n)
print(mask.count('1'))
for i in range(n):
    if mask[i] == '1':
        print(i+1)
print(0)
