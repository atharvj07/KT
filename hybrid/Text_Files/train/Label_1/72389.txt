for _ in range(int(input())):
    s,c=input().split()
    if s==c:
        print(s)
    else:
        for i in range(len(s)):
            for j in range(i+1,len(s)):
                if s[i]>s[j]:
                    a=list(s)
                    a[i],a[j]=a[j],a[i]
                    if a<list(c):
                        print(''.join(a))
                        break
            else:
                continue
            break
        else:
            print('---')
