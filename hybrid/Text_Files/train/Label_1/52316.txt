import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] strengths = new int[n];
        for (int i = 0; i < n; i++) {
            strengths[i] = scanner.nextInt();
        }

        int m = scanner.nextInt();
        List<Action> actions = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            char action = scanner.next().charAt(0);
            int team = scanner.nextInt();
            actions.add(new Action(action, team));
        }

        // Sort the heroes in descending order of strength
        Arrays.sort(strengths);

        // Initialize the teams' strengths
        int team1Strength = 0;
        int team2Strength = 0;

        // Process the actions
        for (Action action : actions) {
            if (action.action == 'p') {
                // Pick the hero with the highest strength
                if (action.team == 1) {
                    team1Strength += strengths[n - 1];
                } else {
                    team2Strength += strengths[n - 1];
                }
                n--;
            } else {
                // Ban the hero with the highest strength
                n--;
            }
        }

        // Output the difference in strengths
        System.out.println(team1Strength - team2Strength);
    }

    private static class Action {
        char action;
        int team;

        public Action(char action, int team) {
            this.action = action;
            this.team = team;
        }
    }
}
