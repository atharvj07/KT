import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = scan.nextInt();

        int ans = 0;
        int p1 = 1, p2 = 106;
        for (int i = 0; i < n; i++) {
            if (Math.abs(a[i] - p1) < Math.abs(a[i] - p2)) {
                p1 = a[i];
                ans += Math.abs(p1 - p2);
            } else {
                p2 = a[i];
                ans += Math.abs(p1 - p2);
            }
        }

        System.out.println(ans);
    }
}
