#include <bits/stdc++.h>
using namespace std;
int k, n;
string s;
bool palindrome(string s) {
    for (int i = 0, j = s.length() - 1; i < j; i++, j--) if (s[i] != s[j]) return false;
    return true;
}
bool sameLength(string s, string t) { return s.length() == t.length(); }
int main() {
    cin >> s >> k;
    n = s.length();
    if (n % k != 0) cout << "NO", exit(0);
    int ans = 1;
    for (int i = 1; i < k; i++) ans |= sameLength(s.substr(0, i), s.substr(i, i));
    ans &= palindrome(s.substr(0, n / k));
    cout << (ans ? "YES" : "NO");
    return 0;
}
