#include <iostream>
#include <vector>
#include <unordered_map>

using namespace std;

int main() {
  int n, x0, y0;
  cin >> n >> x0 >> y0;
  vector<pair<int, int>> points;
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    points.push_back({x, y});
  }
  unordered_map<double, int> m;
  for (auto& [x, y] : points) {
    if (x == x0) m[1e9]++;
    else m[(y - y0) * 1.0 / (x - x0)]++;
  }
  int ans = m.size();
  cout << ans << endl;
  return 0;
}
