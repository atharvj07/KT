#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, p, t;
    cin >> n;
    vector<int> q(n), s(n);
    for (int i = 0; i < n; ++i) {
        cin >> q[i];
        for (int j = 0; j < q[i]; ++j) {
            cin >> p;
            s[i] += 5 * p + 15;
        }
    }
    cout << *min_element(s.begin(), s.end()) << endl;
    return 0;
}
