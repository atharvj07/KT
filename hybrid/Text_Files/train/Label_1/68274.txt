import heapq
INF = 10**15
def dijkstra(start, n, adj):
    dist = [INF] * (n+1)
    dist[start] = 0
    q = [(0, start)]
    while q:
        d, v = heapq.heappop(q)
        if d > dist[v]:
            continue
        for u, w in adj[v]:
            nv = d + w
            if nv < dist[u]:
                dist[u] = nv
                heapq.heappush(q, (nv, u))
    return dist
n, m, a, b = map(int, input().split())
adj = [[] for i in range(n+1)]
for i in range(m):
    x, y, w = map(int, input().split())
    adj[x].append((y, w))
    adj[y].append((x, w))
ans = []
for i in range(1, n+1):
    adj[1].append((i, a))
    res = dijkstra(1, n, adj)
    ans.append(res[i])
    adj[1].pop()
    adj[i].append((1, a))
    res = dijkstra(i, n, adj)
    ans[i-1] = min(ans[i-1], res[1])
    adj[i].pop()
print(*ans)
