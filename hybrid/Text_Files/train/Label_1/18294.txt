#include <iostream>
#include <string>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  int count = 0;
  for (int i = 1; i <= n - 1; i++) {
    string s1, s2;
    cin >> s1 >> s2;
    for (int j = 1; j <= m - 1; j++) {
      if (s1[j] == 'a' && s1[j - 1] == 'f' && s2[j] == 'c' && s2[j - 1] == 'e')
        count++;
    }
  }
  cout << count;
  return 0;
}
