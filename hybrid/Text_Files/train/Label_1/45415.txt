import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.nextLine();
        String t = input.nextLine();
        input.close();
        int[] cnt0 = new int[128];
        int[] cnt1 = new int[128];
        for (char c : t.toCharArray()) {
            if (s.charAt(0) == '0') cnt0[c]++;
            else cnt1[c]++;
        }
        int max0 = 0, max1 = 0, cnt = 0;
        char chr0 = 'a', chr1 = 'a';
        for (int i = 'a'; i <= 'z'; i++) {
            if (cnt0[i] > max0) {
                max0 = cnt0[i];
                chr0 = (char) i;
            }
            if (cnt1[i] > max1) {
                max1 = cnt1[i];
                chr1 = (char) i;
            }
        }
        if (chr0 != chr1) {
            if (s.charAt(0) == '0') cnt = max0;
            else cnt = max1;
        }
        System.out.println(cnt);
    }
}
