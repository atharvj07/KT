#include <bits/stdc++.h>
using namespace std;
const int N = 4005;
int n, m, p[N], c[N];
vector<int> g[N];
void dfs(int u) {
    for (int v : g[u])
        if (p[v] == -1) {
            p[v] = u;
            c[u]++;
            dfs(v);
        }
}
int main() {
    cin >> n >> m;
    while (m--) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    memset(p, -1, sizeof(p));
    for (int i = 1; i <= n; i++)
        if (p[i] == -1) {
            p[i] = i;
            dfs(i);
        }
    int ans = N;
    for (int i = 1; i <= n; i++)
        for (int j = i + 1; j <= n; j++)
            for (int k = j + 1; k <= n; k++)
                if (p[i] == p[j] && p[j] == p[k])
                    ans = min(ans, c[i] + c[j] + c[k]);
    if (ans == N)
        cout << -1 << endl;
    else
        cout << ans << endl;
    return 0;
}
