import math
for _ in range(int(input())):
    x = int(input())
    ans = 0
    while x % 2 == 0:
        ans += 1
        x //= 2
    while x % 3 == 0:
        ans += 2
        x //= 3
    while x % 5 == 0:
        ans += 4
        x //= 5
    if x == 1:
        print(ans)
    else:
        print(-1)
