#include <iostream>
using namespace std;
typedef long long ll;

int main() {
  ll x, y;
  cin >> x >> y;
  if(1LL * y * x / 2 >= x) {
    ll cur = x / (y+1);
    cout << cur << ' ' << cur * y << ' ' << x - cur - cur * y << endl;
  } else {
    cout << 0 << ' ' << 0 << ' ' << x << endl;
  }
  return 0;
}
