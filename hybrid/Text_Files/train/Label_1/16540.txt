#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> x(n);
    for (int i = 0; i < n; i++) {
        cin >> x[i];
    }
    sort(x.begin(), x.end());
    int min_houses = n;
    for (int i = 0; i < n - 1; i++) {
        if (x[i] + 1 == x[i + 1]) {
            min_houses--;
        }
    }
    int max_houses = n;
    for (int i = 0; i < n - 1; i++) {
        if (x[i] + 1 < x[i + 1]) {
            max_houses++;
        }
    }
    cout << min_houses << " " << max_houses << endl;
    return 0;
}
