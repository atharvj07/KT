#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
	ll n,k;
	cin >> n >> k;
	vector<ll> v(n);
	for(ll i=0;i<n;i++) cin >> v[i];
	sort(v.begin(),v.end());
	ll ans=0;
	for(ll i=0;i<n-k;i++){
		if(v[i]==v[i+1]) continue;
		ans+=v[i];
		v[i+1]-=v[i];
	}
	cout << ans << endl;
	return 0;
}
