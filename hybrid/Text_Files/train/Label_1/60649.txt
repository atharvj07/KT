n = int(input())
s = input()
t = input()

# Check if the number of '1's and '0's are the same in both strings
if s.count('1') != t.count('1') or s.count('0') != t.count('0'):
    print(-1)
else:
    # Count the number of differences in '1's and '0's positions
    ones_diff = zeros_diff = 0
    for i in range(n):
        if s[i] != t[i]:
            if s[i] == '1':
                ones_diff += 1
            else:
                zeros_diff += 1
    # The minimum number of operations is the maximum of these differences
    print(max(ones_diff, zeros_diff))
