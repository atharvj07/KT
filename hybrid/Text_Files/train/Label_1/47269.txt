import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt();
        char[][] arr = new char[n][m];
        for(int i = 0; i < n; i++) arr[i] = input.next().toCharArray();
        input.close();
        int ans = 0;
        for(int i = 0; i < n - 1; i++) {
            for(int j = 0; j < m - 1; j++) {
                ans += (arr[i][j] == 'f' && arr[i][j + 1] == 'a' && arr[i + 1][j] == 'c' && arr[i + 1][j + 1] == 'e')? 1:0;
            }
        }
        System.out.println(ans);
    }
}
