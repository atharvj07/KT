#include<bits/stdc++.h>
using namespace std;

int main() {
    char a[4][4];
    for (int i = 0; i < 4; i++)
        for (int j = 0; j < 4; j++)
            cin >> a[i][j];
    for (int i = 0; i < 4; i++) {
        int cnt1 = 0, cnt2 = 0;
        for (int j = 0; j < 4; j++) {
            if (a[i][j] == '.') cnt1++;
            if (a[j][i] == '.') cnt2++;
        }
        if (cnt1 == 1 || cnt2 == 1) {
            cout << "YES";
            return 0;
        }
    }
    for (int i = 0; i < 2; i++) {
        int cnt1 = 0, cnt2 = 0;
        for (int j = 0; j < 4; j++) {
            if (a[i + j][j] == '.') cnt1++;
            if (a[i + j][3 - j] == '.') cnt2++;
        }
        if (cnt1 == 1 || cnt2 == 1) {
            cout << "YES";
            return 0;
        }
    }
    cout << "NO";
    return 0;
}
