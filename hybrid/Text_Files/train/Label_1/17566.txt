#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1000000007;
ll gcd(ll a,ll b){
    if(b==0)return a;
    return gcd(b,a%b);
}
int main(){
    ll t;cin>>t;
    while(t--){
        ll num;cin>>num;
        while(1){
            ll sum=0,temp=num;
            while(temp>0){
                sum+=temp%10;
                temp=temp/10;
            }
            if(gcd(num,sum)>1){
                cout<<num<<endl;
                break;
            }
            num++;
        }
    }
    return 0;
}
