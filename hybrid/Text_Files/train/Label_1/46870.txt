import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] b = new long[n];
        long[] d = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextLong();
            d[i] = sc.nextLong();
        }
        long y = -1;
        int k = 0;
        TreeMap<Long, Integer> map = new TreeMap<>();
        for (int i = 0; i < n; i++) {
            map.put(b[i], map.getOrDefault(b[i], 0) + 1);
            map.put(d[i], map.getOrDefault(d[i], 0) - 1);
        }
        int cur = 0;
        for (long year : map.keySet()) {
            cur += map.get(year);
            if (cur > k) {
                k = cur;
                y = year;
            }
        }
        System.out.println(y + " " + k);
    }
}
