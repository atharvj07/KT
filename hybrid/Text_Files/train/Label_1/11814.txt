#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
const ll inf = 1e9;
ll n, par[505], sz[505];
ll modpow(ll b, ll e) {
    if (!e)
        return 1;
    if (e == 1)
        return b;
    ll ret = modpow(b, e / 2);
    ret = ret * ret % mod;
    if (e & 1) ret = ret * b % mod;
    return ret;
}
ll gcd(ll a, ll b) {
    if (!b) return a;
    return gcd(b, a % b);
}
int main() {
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    cin >> n;
    vector<vector<int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> par[i];
        if (par[i] == -1)
            v[i].push_back(i);
        else
            v[par[i]].push_back(i);
    }
    ll deg = 0;
    for (int i = 0; i < n; i++) {
        sz[i] = v[i].size();
        deg += (sz[i] == 1);
    }
    ll num = deg * (deg - 1) % mod;
    ll den = 2;
    ll g = gcd(num, den);
    num /= g, den /= g;
    cout << (num * modpow(den, mod - 2)) % mod << endl;
    return 0;
}
