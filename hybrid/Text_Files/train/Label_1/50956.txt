import java.util.Scanner;

public class TreeConstruct {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        long[][] dp = new long[n + 1][k + 1];
        dp[0][0] = 1;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                // If we connect the current vertex to the first vertex, then we can have at most two more edges from this vertex.
                if (i > 1 && j > 0) {
                    dp[i][j] = dp[i - 1][j - 1] * 2 % MOD;
                }
                // If we don't connect the current vertex to the first vertex, then we can have at most three more edges from this vertex.
                dp[i][j] = (dp[i][j] + dp[i - 1][j] * (3 - (i == 1 ? 1 : 0))) % MOD;
            }
        }

        System.out.println(dp[n][k]);
    }
}
