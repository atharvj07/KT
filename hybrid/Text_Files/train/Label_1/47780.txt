import java.util.*;
public class Main {
public static void main(String[] args) {
Scanner sc=new Scanner(System.in);
int n=sc.nextInt();
char[] ch=sc.next().toCharArray();
int co=0,a=0;
for(char c:ch) {
if(c=='X') co++;
if(c=='x') a++;
}
int mi=Math.abs(co-a)/2;
System.out.println(mi);
int c=0,b=0;
for(int i=0;i<n;i++) {
if(c==mi) {
if(ch[i]=='X') {
ch[i]='x';
b++;
}
}
else {
if(ch[i]=='x') {
ch[i]='X';
c++;
}
}
}
System.out.println(new String(ch));
}
}
