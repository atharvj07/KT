#include <bits/stdc++.h>
using namespace std;

const int N = 4;
int A[N][N][N], B[N][N][N];

inline bool ok() {
  vector<int> v(24);
  for (int i = 0; i < N; i++)
    for (int j = 0; j < N; j++)
      for (int k = 0; k < N; k++)
        v[i * 4 + j * 2 + k] = A[i][j][k];
  int sz = int(v.size());
  for (int i = 0; i < sz; i++)
    for (int j = i + 1; j < sz; j++)
      if (v[i] > v[j]) {
        swap(v[i], v[j]);
        j--;
      }
  for (int i = 0; i < N; i++)
    for (int j = 0; j < N; j++)
      for (int k = 0; k < N; k++)
        B[i][j][k] = v[i * 4 + j * 2 + k];
  for (int i = 0; i < N; i++)
    for (int j = 0; j < N; j++)
      for (int k = 0; k < N; k++)
        if (B[i][j][k] != i) return 0;
  for (int i = 0; i < N; i++)
    for (int j = 0; j < N; j++)
      if (B[j][i][0] != i) return 0;
  for (int i = 0; i < N; i++)
    if (B[0][i][i] != i) return 0;
  return 1;
}

int main() {
  for (int i = 0; i < N; i++)
    for (int j = 0; j < N; j++)
      for (int k = 0; k < N; k++) {
        cin >> A[i][j][k];
        B[i][j][k] = A[i][j][k];
      }
  cout << (ok() ? "YES" : "NO") << endl;
  return 0;
}
