#include <bits/stdc++.h>
#define int long long
using namespace std;
int mod=1e9+7;
int power(int x,int y)
{
    int ret=1;
    while(y)
    {
        if(y&1)ret=(ret*x)%mod;
        y/=2;
        x=(x*x)%mod;
    }
    return ret;
}
int32_t main()
{
    int n;
    cin>>n;
    int ans;
    if(n==1)ans=1;
    else if(n==2)ans=3;
    else if(n==3)ans=10;
    else
    {
        ans=(power(3,n-2)+power(3,n-1)+power(3,n))%mod;
    }
    cout<<ans;
    return 0;
}
