def mmod(x):
    return x % (10**9 + 7)

n, m = map(int, input().split())
s = input()

p_arr = [0] * m
q_arr = [0] * m
sum_ = 0
for i in range(m):
    if s[i] == '(':
        p_arr[i] = p_arr[i - 1] + 1
        q_arr[i] = q_arr[i - 1]
    else:
        q_arr[i] = q_arr[i - 1] + 1
        sum_ += p_arr[i - 1]

res = mmod(q_arr[m - 1] * sum_)
if n == m:
    res = mmod(res + 1)
print(res)
