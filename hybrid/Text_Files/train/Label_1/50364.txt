import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = scanner.nextInt();
        }
        scanner.close();
        int count = 0;
        while (h[0] != 0) {
            count++;
            for (int i = 0; i < n; i++) {
                if (i > 0 && h[i] == h[i - 1]) {
                    h[i]--;
                }
                if (i < n - 1 && h[i] == h[i + 1]) {
                    h[i]--;
                }
                h[i]--;
            }
        }
        System.out.println(count);
    }
}
