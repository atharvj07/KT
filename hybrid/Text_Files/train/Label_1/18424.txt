#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
typedef long long ll;
int32_t main() {
  ll n, m;  cin >> n >> m;
  vector<int> a(m);
  for (int i = 0; i < m; i++) { cin >> a[i]; }
  vector<int> dp(n + 1, 0);
  for (int i = 0; i <= n; i++) dp[i] = 1;
  for (int p : a) {
    if (p != 1) dp[p] = (dp[p] + dp[p - 1]) % mod;
    if (p != n) dp[p] = (dp[p] + dp[p + 1]) % mod;
  }
  cout << dp[1] << "\n";
}
