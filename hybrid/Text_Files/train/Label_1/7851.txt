#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> s(n);
    for (int i = 0; i < n; i++) cin >> s[i];
    int l = 1, r = 1e9;
    while (l < r) {
        int mid = (l + r) / 2;
        int cnt = 0, sum = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] > mid) break;
            if (sum + s[i] <= mid) {
                sum += s[i];
            } else {
                sum = s[i];
                cnt++;
            }
        }
        if (cnt > k) l = mid + 1;
        else r = mid;
    }
    cout << l << endl;
}
