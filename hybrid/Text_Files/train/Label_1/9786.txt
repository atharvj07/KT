#include <bits/stdc++.h>
#define int long long 
using namespace std;
const int mod=1e9+7;
int32_t main() {
int n;cin>>n;
int arr[n],sum=0;
int ans=1e18,idi;
for(int i=0;i<n;++i){
    cin>>arr[i];
    sum+=abs(arr[i]-i-1);
}
for(int i=0;i<n;++i){
    sum=sum-abs(arr[i]-i-1)+abs(arr[(i+n-1)%n]-(i+n-1)%n);
    if(sum<ans){
        ans=sum;
        idi=i;
    }
}
cout<<ans<<" "<<idi<<endl;
    return 0;
}
