import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < m; i++) {
            int s = sc.nextInt() - 1;
            c[s] = sc.nextInt();
        }
        boolean flag = false;
        for (int i = 0; i < n; i++) {
            if (c[i] == 0 && i != n - 1) {
                flag = true;
                break;
            }
        }
        if (flag) {
            System.out.println(-1);
            return;
        }
        for (int i = n - 1; i >= 0; i--) {
            if (c[i] == 0) {
                System.out.print('0');
            } else {
                System.out.print(c[i]);
                break;
            }
        }
    }
}
