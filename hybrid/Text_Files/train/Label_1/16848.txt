#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    map<int, int> num;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        num[x]++;
    }
    for (int i = 0; i < m; i++) {
        int x;
        cin >> x;
        cout << (num[x - 1] || num[x + 1]) << endl;
    }
}
