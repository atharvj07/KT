b1, q, l, m = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
cur = b1
ans = 0
for i in range(1, 10 ** 9 + 1):
    if cur <= l and cur != a[0]:
        ans += 1
        cur *= q
        a.pop(0)
    else:
        break

if ans == 10 ** 9:
    print('inf')
else:
    print(ans)
