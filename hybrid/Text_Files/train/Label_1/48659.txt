import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[][] arr = new long[n + 1][2];
        for (int i = 1; i <= n; i++) {
            arr[i][0] = sc.nextInt();
            arr[i][1] = sc.nextInt();
        }
        long ans = 0;
        for (int i = 1; i < n; i++) {
            if (arr[i][1] < arr[i + 1][0]) {
                ans += arr[i + 1][0] - arr[i][1];
                arr[i + 1][0] = arr[i][1];
            }
        }
        System.out.println(ans);
    }
}
