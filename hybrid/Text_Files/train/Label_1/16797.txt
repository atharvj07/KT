#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int mod = 998244353;

int n, k, sa[200005];
ll f[200005], invf[200005];
ll qpow(ll x, ll y) {
  ll r = 1;
  while (y > 0) {
    if (y % 2 == 0) {
      x = x * x % mod;
      y /= 2;
    } else {
      r = r * x % mod;
      y -= 1;
    }
  }
  return r;
}

void init() {
  f[0] = f[1] = invf[0] = invf[1] = 1;
  for (int i = 2; i <= 200000; i++) {
    f[i] = f[i - 1] * i % mod;
    invf[i] = invf[i - 1] * qpow(i, mod - 2) % mod;
  }
}

vector<int> stk;
void get() {
  stk.clear();
  int la = -1;
  for (int i = 1; i <= n; i++) {
    while (stk.size() > 0 && i - sa[stk.back()] > k) stk.pop_back();
    while (stk.size() > 0 && sa[stk.back()] >= sa[i]) stk.pop_back();
    if (stk.size() > 0) la = sa[stk.back()];
    stk.push_back(i);
    sa[i] = la;
  }
}

int main() {
  init();
  cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> sa[i];
  get();
  for (int i = 1; i <= n; i++) {
    int l = sa[i] + 1;
    int r = n - i + 1;
    ll res = f[r];
    res = res * invf[l] % mod;
    res = res * invf[r - l] % mod;
    cout << res << endl;
  }
  return 0;
}
