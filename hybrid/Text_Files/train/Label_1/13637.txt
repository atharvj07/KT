void run()
{
        int n; cin >> n;
        vi a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(all(a));
        deque<int> dq;
        for (int i = 0; i < n; i++)
        {
                while (!dq.empty() && a[i] >= a[dq.back()])
                        dq.pop_back();
                dq.pb(i);
                while (dq.front() < i - 1) dq.pop_front();
                if (i == n - 1 || a[i] != a[dq.front()])
                {
                        cout << a[dq.front()] << " " << n - 1 - i << '\n';
                        for (int j = i + 1; j < n; ++j)
                                if (a[j] != a[dq.front()])
                                        cout << j + 1 << " ";
                        return;
                }
        }
}
