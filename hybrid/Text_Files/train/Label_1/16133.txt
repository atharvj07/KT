#include <bits/stdc++.h>
using namespace std;

const int maxN = 20, maxM = 20;
int n, m;
string s[maxN];
int c[maxN][maxM];
bool u[maxN][maxM];

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> s[i];
        for (int j = 0; j < m; j++) {
            cin >> c[i][j];
        }
    }
    long long ans = 0;
    for (int j = 0; j < m; j++) {
        int cnt[26] = {};
        for (int i = 0; i < n; i++) {
            cnt[s[i][j] - 'a']++;
        }
        int mx = -1;
        for (int k = 0; k < 26; k++) {
            mx = max(mx, cnt[k]);
        }
        ans += (n - mx);
    }
    cout << ans << '\n';
    return 0;
}
