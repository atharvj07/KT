x1, y1 = map(int, input().split())
x2, y2 = map(int, input().split())
n = int(input())
s = input()
dx = [0, 0, -1, 1]
dy = [1, -1, 0, 0]
ans = 1e9
for i in range(n):
    for j in range(4):
        nx = x1 + dx[j] * s.count(chr(ord('U') + j))
        ny = y1 + dy[j] * s.count(chr(ord('U') + j))
        if i == n - 1:
            if nx == x2 and ny == y2:
                ans = min(ans, s.count(chr(ord('U') + j)))
        else:
            ans = min(ans, s.count(chr(ord('U') + j)) + (abs(x2 - nx) + abs(y2 - ny) + n - 1) // n)
if ans == 1e9:
    ans = -1
print(ans)
