#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll f[1000001];
int32_t main() {
    for (int i = 2; i <= 1000000; i++) {
        if (f[i] == 0) {
            for (int j = i; j <= 1000000; j += i) {
                if (f[j] == 0) f[j] = i;
            }
        }
    }
    int t;
    cin >> t;
    while (t--) {
        ll n, k;
        cin >> n >> k;
        n += f[n] * k;
        cout << n << "\n";
    }
}
