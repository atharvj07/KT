import java.util.*;
import java.math.BigInteger;

public class Main {
    public static final BigInteger MOD = BigInteger.valueOf(1000000007L);

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int[] broken = new int[M];
        for(int i=0; i<M; i++) {
            broken[i] = sc.nextInt();
        }
        sc.close();

        BigInteger[] dp = new BigInteger[N+1];
        Arrays.fill(dp, BigInteger.ZERO);
        dp[0] = BigInteger.ONE;
        dp[1] = BigInteger.ONE;
        for(int i=2; i<=N; i++) {
            boolean isBroken = false;
            for(int j=0; j<M; j++) {
                if(broken[j]==i) {
                    isBroken = true;
                    break;
                }
            }
            if(!isBroken) {
                dp[i] = dp[i-1].add(dp[i-2]).mod(MOD);
            }
        }
        System.out.println(dp[N]);
    }
}
