#pragma GCC optimize ("O3")
#include <bits/stdc++.h>
using namespace std;
int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int n;
    cin >> n;
    vector<int32_t> a(n);
    for (int i = 0;i < n;i++) cin >> a[i];
    vector<int32_t> last;
    vector<int32_t> dp(n, 0);
    for (int i = 1;i < n - 1;i++) {
        if (a[i] == a[i - 1]) continue;
        last.push_back(i);
    }
    if (last.empty() == true) {
        cout << a[0] << ' ' << 0 << '\n';
        return 0;
    }
    if (a[0] != a[1]) dp[0] = 1;
    if (a[n - 2] != a[n - 1]) dp[n - 1] = 1;
    for (int i = 1;i < n - 1;i++) {
        if (a[i] == a[i + 1] and a[i] == a[i - 1]) continue;
        if (a[i] != a[i - 1]) dp[i] = dp[i - 1] + 1;
        else dp[i] = dp[i - 1];
    }
    int best = 0;
    int ind = 0;
    for (int i = 0;i < last.size();i++) {
        if (dp[last[i]] > best) {
            best = dp[last[i]];
            ind = last[i];
        }
    }
    int32_t sum = 0;
    for (int i = 0;i < n;i++) {
        if (i != ind) sum += a[i];
    }
    cout << sum << ' ' << best + 1 << '\n';
    if (dp[ind] > 0) cout << ind << ' ';
    for (int i = ind + dp[ind];i < n - 1;i++) {
        if (a[i] != a[i + 1]) break;
        else cout << i + 1 << ' ';
    }
    return 0;
}
