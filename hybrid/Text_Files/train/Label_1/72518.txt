import sys

def is_k_beautiful(x, k):
    vis = [False] * 10
    while x > 0:
        d = x % 10
        vis[d] = True
        x //= 10
    cnt = 0
    for i in range(10):
        if vis[i]:
            cnt += 1
        if cnt > k:
            return False
    return True

def solve():
    n, k = map(int, input().split())
    if k == 1:
        return n
    x = n
    while not is_k_beautiful(x, k):
        x += 1
    return x

t = int(input())
for _ in range(t):
    print(solve())
