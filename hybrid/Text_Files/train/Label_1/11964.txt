#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  int min1, max1, min2, max2, min3, max3;
  cin >> min1 >> max1;
  cin >> min2 >> max2;
  cin >> min3 >> max3;
  int sum_min = min1 + min2 + min3;
  int sum_max = max1 + max2 + max3;
  if (sum_min > n || sum_max < n) {
    cout << "-1"
      << "\n";
    return 0;
  }
  int first = n - min2 - min3;
  first = max(min1, first);
  first = min(max1, first);
  int second = n - first - min3;
  second = max(min2, second);
  second = min(max2, second);
  int third = n - first - second;
  third = max(min3, third);
  third = min(max3, third);
  cout << first << " " << second << " " << third << "\n";
  return 0;
}
