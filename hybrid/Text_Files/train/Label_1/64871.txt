from collections import defaultdict

n=int(input())
a=[0]*n
b=[0]*n
c=[0]*n
l=defaultdict(list)
for i in range(n):
 a[i],b[i],c[i]=[int(j) for j in input().split()]
for j in range(n-1):
 x,y=map(int,input().split())
 l[x].append(y)
 l[y].append(x)
def solve(node,p):
 global ans
 if node!=1:
  if b[node]!=c[node]:
   ans+=a[p]
 return
 for j in l[node]:
  if j!=p:
   solve(j,node)
ans=-1
solve(1,1)
print(ans)
