#include <bits/stdc++.h>
#define nl "\n"
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n;
        string s, t;
        cin >> n >> s >> t;
        sort(s.begin(), s.end());
        sort(t.begin(), t.end());
        if (s == t)
            cout << "YES" << nl;
        else
            cout << "NO" << nl;
    }
    return 0;
}
