#include <bits/stdc++.h>

#define endl '\n'
using namespace std;
typedef long long ll;

ll mod = 1e9 + 7;
vector<vector<int>> graph;
vector<int> vis, color;
ll dfs(int node, int d, ll mask) {
    if (vis[node]) {
        if (color[node] == d) return (mask & (1 << d));
        return 0;
    }
    vis[node] = 1;
    color[node] = d;
    ll ch1 = dfs(node, d, mask | (1 << d));
    ll ch2 = dfs(node, !d, mask);
    return (ch1 + ch2) % mod;
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n, m;
    cin >> n >> m;
    graph.resize(n);
    while (m--) {
        int u, v;
        cin >> u >> v;
        u--, v--;
        graph[u].push_back(v);
        graph[v].push_back(u);
    }
    ll ans = 1;
    vis.resize(n);
    color.resize(n);
    for (int i = 0; i < n; i++) {
        if (vis[i]) continue;
        ans = ans * dfs(i, 0, 0) % mod;
    }
    cout << ans << endl;
    return 0;
}
