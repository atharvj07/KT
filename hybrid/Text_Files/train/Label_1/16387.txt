#include<bits/stdc++.h>
#define ll long long int
using namespace std;

vector<ll>adj[100000];
ll par[100000];
bool vis[100000];
ll black[100000];

void dfs(ll node){
	vis[node]=true;
	for(ll child:adj[node]){
		if(vis[child]==false){
			par[child]=node;
			dfs(child);
		}
	}
}

int main(){
	ll n;
	cin>>n;
	for(ll i=1;i<n;i++){
		ll l,r;
		cin>>l>>r;
		adj[l].push_back(r);
		adj[r].push_back(l);
	}
	for(ll i=0;i<n;i++){
		ll x;
		cin>>x;
		black[i]=x;
	}
	
	dfs(0);
	ll ans=0;
	for(ll i=1;i<n;i++){
		if(black[i]!=black[par[i]]){
			ans++;
		}
	}
	
	cout<<(1<<ans)%1000000007;
	
}
