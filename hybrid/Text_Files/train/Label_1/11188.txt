#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> p(n), q(n);
    for (auto &it : p) cin >> it;
    for (auto &it : q) cin >> it;
    vector<char> v(26);
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        int x = p[i] - 1, y = q[i] - 1;
        if (x > y) swap(x, y);
        if (v[x] == 0) cnt++;
        if (v[x] == 0 || v[y] == 0) v[x] = 'a' + cnt;
        v[y] = v[x];
    }
    if (cnt < k) {
        cout << "NO" << endl;
    } else {
        cout << "YES" << endl;
        for (auto it : v) cout << it;
        cout << endl;
    }
    return 0;
}
