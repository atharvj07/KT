import java.util.*;
import java.io.*;

class Main {
    static PrintWriter out = new PrintWriter(System.out);

    int[] par, rank, destroyed;
    int n;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();

        Main p = new Main(n);

        for (int i = 0; i < k; i++) {
            int type = sc.nextInt();
            int u = sc.nextInt(), v = sc.nextInt();
            if (type == 1) {
                p.unite(u, v);
            } else if (type == 2) {
                p.destroyed[p.find(u)] = p.destroyed[p.find(v)] = Math.max(p.destroyed[p.find(u)], p.destroyed[p.find(v)]);
            } else {
                out.println(p.canConnect(u, v) ? "YES" : "NO");
            }
        }

        out.flush();
    }

    public Main(int n) {
        this.n = n;
        par = new int[n];
        rank = new int[n];
        destroyed = new int[n];
        for (int i = 0; i < n; i++) par[i] = i;
    }

    int find(int x) {
        if (par[x] == x) return x;
        else return par[x] = find(par[x]);
    }

    void unite(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) return;
        if (rank[x] < rank[y]) par[x] = y;
        else {
            par[y] = x;
            if (rank[x] == rank[y]) rank[x]++;
        }
    }

    boolean canConnect(int x, int y) {
        if (x == y) return true;
        x = find(x);
        y = find(y);
        if (x == y && destroyed[x] >= 0) return true;
        else return false;
    }
}
