n, m = map(int, input().split())
grid = [input() for _ in range(n)]
vis = [[False] * m for _ in range(n)]
for x in range(n):
    for y in range(m):
        for dx, dy in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
            if vis[x][y]: continue
            cx, cy = x, y
            color = grid[x][y]
            lst = [(x, y)]
            while True:
                cx += dx
                cy += dy
                if cx < 0 or cx >= n or cy < 0 or cy >= m or vis[cx][cy]: break
                if grid[cx][cy] != color: break
                vis[cx][cy] = True
                lst.append((cx, cy))
            if len(lst) >= 4 and lst[0] == lst[-1]: print('Yes'); exit()
print('No')
