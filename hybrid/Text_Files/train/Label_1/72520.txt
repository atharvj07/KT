def solve(n, k, p, q):
    # Find the intersection points of the two permutations to determine the minimum
    # number of segments that can be created which respect both permutations.
    order = [0] * n
    for i, val in enumerate(q):
        order[val-1] = i

    segments = 1
    for i in range(1, n):
        if order[p[i-1]-1] > order[p[i]-1]:
            segments += 1

    if segments > k:
        return "NO"

    # Construct the string with the required number of distinct letters
    result = ['a'] * n
    current_char = ord('a')
    for i in range(1, n):
        if order[p[i-1]-1] > order[p[i]-1]:
            current_char += 1
        result[p[i-1]-1] = chr(current_char)
        if current_char - ord('a') + 1 == k:
            break
    result[p[-1]-1] = chr(current_char)

    return "YES\n" + ''.join(result)

# Input reading
n, k = map(int, input().split())
p = list(map(int, input().split()))
q = list(map(int, input().split()))

# Solve the problem
print(solve(n, k, p, q))
