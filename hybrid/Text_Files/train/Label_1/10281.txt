#include <iostream>
#include <cmath>
using namespace std;

int main() {
    int n, x, taken[2 * 1000 + 1] = {0}, ans = 1000000000, cur;
    cin >> n >> x;
    for (int i = 0; i < n; i++) {
        cin >> cur;
        taken[cur + x]++;
    }
    taken[x] = n;
    for (int i = 1; i <= 2 * x; i++) {
        if (taken[i] >= 1) {
            ans = min(ans, taken[i]);
        }
        else if (taken[i] == 0) {
            if (taken[2 * x - i] >= 1) {
                ans = min(ans, taken[2 * x - i] + 1);
            }
        }
    }
    cout << ans << endl;
    return 0;
}
