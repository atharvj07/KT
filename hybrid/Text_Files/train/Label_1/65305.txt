import sys

def solve():
    n = int(input())
    s = input()
    t = input()
    if s == t:
        print(0)
        return
    a = []
    b = []
    for i in range(n):
        if s[i] != t[i]:
            if s[i] == 'a':
                a.append(i)
            else:
                b.append(i)
    if len(a) != len(b):
        print(-1)
        return
    print(len(a))
    for i in range(len(a)):
        print(a[i] + 1, b[i] + 1)

solve()
