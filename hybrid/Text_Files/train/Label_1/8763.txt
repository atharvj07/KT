#include <bits/stdc++.h>
#define int long long
using namespace std;
const int P = 1e9 + 7;
int f[1005][1005], g[1005][1005];
signed main() {
  int n, k;
  cin >> n >> k;
  f[2][1] = 1, g[2][1] = 1;
  for (int i = 3; i <= n; i++) {
    for (int j = 1; j < i; j++) {
      int pre = f[i - 1][j];
      int suf = g[i - 1][j - 1];
      suf = (suf + g[i - 1][j]) % P;
      f[i][j] = (pre * j + suf * (i - j) % P) % P;
      pre = f[i - 1][j - 1];
      suf = g[i - 1][j];
      suf = (suf + g[i - 1][j - 1]) % P;
      g[i][j] = (pre * j + suf * (i - j - 1) % P) % P;
    }
  }
  cout << f[n][k] << endl;
  return 0;
}
