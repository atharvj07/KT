#include<bits/stdc++.h>
using namespace std;
const int nax=5e2+5;
int d[nax],h[nax],n;
int main(){
    cin>>n;
    for(int i=1;i<=n;++i)cin>>h[i],d[i]=h[i];
    sort(d+1,d+n+1);
    int mx=0,ans=0,k;
    for(int i=1;i<=n;++i)for(int j=1;j<=n;++j){
            if(h[i]==d[j])k=j;
        }
    int x=h[k];
    for(int i=1;i<=n;++i){
        if(h[i]>d[k])ans=max(ans,h[i]-d[k]);
        else ans=max(ans,d[k+1]-h[i]);
    }
    mx=d[k];
    cout<<ans<<endl;
    cout<<mx<<" "<<d[k+1]-d[k]<<endl;
    return 0;
}
