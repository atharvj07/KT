#include <bits/stdc++.h>
#define int long long int
using namespace std;

int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    unordered_map<int, int> mp;
    int val = 0, occ = 0;
    for (int i = 0; i < n; i++) {
      int a;
      cin >> a;
      if (mp[a] == 0) occ++;
      mp[a]++;
      if (mp[a] > val) val = mp[a], occ = 1;
      else if (mp[a] == val)
        occ++;
    }
    cout << n - val * occ << endl;
  }
}
