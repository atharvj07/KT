import sys
input = sys.stdin.readline

n = int(input())
s = input().strip()
t = input().strip()
u = input().strip()
a = [s]
b = [t]
c = [u]
sb = [0] * len(s)
tb = [0] * len(t)
ub = [0] * len(u)
mb, bt = 0, 0
for i in range(len(s)):
    if i > 0: sb[i] = sb[i - 1]
    if s[i] == s[i - 1] or i == 0: sb[i] += 1
    mb = max(mb, sb[i])
for i in range(len(t)):
    if i > 0: tb[i] = tb[i - 1]
    if t[i] == t[i - 1] or i == 0: tb[i] += 1
    mb = max(mb, tb[i])
for i in range(len(u)):
    if i > 0: ub[i] = ub[i - 1]
    if u[i] == u[i - 1] or i == 0: ub[i] += 1
    mb = max(mb, ub[i])
for i in range(n):
    na, nb, nc = [], [], []
    for j in range(len(a[i])):
        if a[i][j] == t[j] or b[i][j] == t[j]: na.append(t[j])
        else: na.append(a[i][j])
    for j in range(len(a[i])):
        if a[i][j] == u[j] or c[i][j] == u[j]: nb.append(u[j])
        else: nb.append(a[i][j])
    for j in range(len(t[i])):
        if b[i][j] == u[j] or c[i][j] == u[j]: nc.append(u[j])
        else: nc.append(b[i][j])
    sb = [0] * len(na)
    tb = [0] * len(nb)
    ub = [0] * len(nc)
    for j in range(len(na)):
        if j > 0: sb[j] = sb[j - 1]
        if na[j] == na[j - 1] or j == 0: sb[j] += 1
        mb = max(mb, sb[j])
    for j in range(len(nb)):
        if j > 0: tb[j] = tb[j - 1]
        if nb[j] == nb[j - 1] or j == 0: tb[j] += 1
        mb = max(mb, tb[j])
    for j in range(len(nc)):
        if j > 0: ub[j] = ub[j - 1]
        if nc[j] == nc[j - 1] or j == 0: ub[j] += 1
        mb = max(mb, ub[j])
    a.append(na)
    b.append(nb)
    c.append(nc)
if sb[-1] == tb[-1] and tb[-1] == ub[-1]: print("Draw")
elif sb[-1] > tb[-1] and sb[-1] > ub[-1]: print("Kuro")
elif tb[-1] > sb[-1] and tb[-1] > ub[-1]: print("Shiro")
else: print("Katie")
