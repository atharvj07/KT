import java.io.*;
import java.util.*;

public class Main {

    static class Pair {
        int x, y;
        Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static char[][] board, target;
    static int n, ans;
    static boolean[][] vis;
    static ArrayList<Pair> list = new ArrayList<>();

    static boolean check() {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] != target[i][j]) return false;
            }
        }
        return true;
    }

    static void dfs(int x, int y) {
        if (x == n || check()) {
            if (check()) ans = list.size();
            return;
        }

        vis[x][y] = true;

        int nx = x + 1;
        int ny = y;

        list.add(new Pair(x, y));
        for (int i = 0; i < n; i++) {
            if (!vis[x][i]) {
                for (int j = 0; j < n; j++) {
                    if (nx < n && board[nx][i] == target[nx][i]) {
                        if (board[x][j] == target[x][j]) board[x][j] = '0';
                        else board[x][j] = '1';
                    }
                }
            }

            if (!vis[i][y]) {
                for (int j = 0; j < n; j++) {
                    if (ny < n && board[i][ny] == target[i][ny]) {
                        if (board[j][y] == target[j][y]) board[j][y] = '0';
                        else board[j][y] = '1';
                    }
                }
            }
        }

        dfs(nx, ny);
        vis[x][y] = false;
        list.remove(list.size() - 1);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(in.readLine());

        board = new char[n][n];
        target = new char[n][n];
        vis = new boolean[n][n];

        for (int i = 0; i < n; i++) {
            board[i] = in.readLine().toCharArray();
        }

        for (int i = 0; i < n; i++) {
            target[i] = in.readLine().toCharArray();
        }

        char[] bar = in.readLine().toCharArray();

        dfs(0, 0);

        if (ans == 0) System.out.println(-1);
        else {
            System.out.println(ans);
            for (Pair p : list) {
                if (p.x == p.y) {
                    System.out.println("row " + p.x);
                } else {
                    System.out.println("col " + p.x);
                }
            }
        }
    }
}
