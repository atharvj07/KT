import java.util.*;
import java.io.*;

public class Main {
    static final int MAXN = 100005;

    static int[] perm;
    static boolean[] used;
    static int[] p;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        PrintWriter out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));

        int n = sc.nextInt();

        perm = new int[MAXN];
        used = new boolean[MAXN];
        p = new int[MAXN];
        Arrays.fill(used, false);

        for (int i = 0; i < n; i++) {
            p[sc.nextInt()] = i;
        }

        int cnt = 1;
        for (int i = 0; i < n; i++) {
            if (used[i])
                continue;

            while (perm[i] == 0) {
                perm[i] = cnt++;
                used[i] = true;
                i = p[i];
            }
        }

        for (int i = 0; i < n; i++)
            out.printf("%d ", perm[i]);

        out.println();
        out.close();
    }
}
