import sys
import math
input = sys.stdin.readline

n = int(input())
mx = list(map(float, input().split()))
mn = list(map(float, input().split()))

for i in range(n):
    if mx[i] != 1 and mn[i] != 1:
        s = 1 / (mx[i] + mn[i] - 1)
        mx[i] = s * mx[i]
        mn[i] = s * mn[i]

    p = (1 - mx[i]) / (n - i)
    for j in range(i + 1, n):
        mx[j] = mx[j] * (1 - p)
        mn[j] = mn[j] * (1 - p) + p * mn[i]

f = mx[0]
for i in range(n):
    mn[i] = (mn[i] - f * (1 - mn[i])) / f
    print('%.8f' % mx[i], '%.8f' % mn[i])
