#include <bits/stdc++.h>
using namespace std;
 
int main() {
    int t; cin >> t;
    while (t--) {
        int l, r; cin >> l >> r;
        if (l & 1) {
            if (l <= r-2) {
                cout << l+1 << ' ' << l+2 << endl;
            } else {
                cout << -1 << ' ' << -1 << endl;
            }
        } else {
            if (l+3 <= r) {
                cout << l << ' ' << l+3 << endl;
            } else {
                cout << -1 << ' ' << -1 << endl;
            }
        }
    }
}
