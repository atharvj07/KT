#include <stdio.h>

struct Segment {
  long long l, r;
  bool contained;
};

void solve(int n, Segment* seg) {
  for (int i = 0; i < n; ++i) {
    seg[i].contained = false;
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      if (seg[i].l >= seg[j].l && seg[i].r <= seg[j].r) {
        seg[i].contained = true;
        break;
      }
    }
  }
  bool found = false;
  for (int i = 0; i < n; ++i) {
    if (!seg[i].contained) {
      for (int j = 0; j < n; ++j) {
        if (seg[j].contained) {
          printf("%d %d\n", i + 1, j + 1);
          found = true;
          break;
        }
      }
      break;
    }
  }
  if (!found) {
    printf("-1 -1\n");
  }
}

int main() {
  int n;
  scanf("%d", &n);
  Segment seg[n];
  for (int i = 0; i < n; ++i) {
    scanf("%lld %lld", &seg[i].l, &seg[i].r);
  }
  solve(n, seg);
  return 0;
}
