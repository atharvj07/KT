good_letters = input().strip()
pattern = input().strip()
n = int(input())

for _ in range(n):
    query = input().strip()
    if '*' in pattern:
        prefix, suffix = pattern.split('*', 1)
    else:
        prefix, suffix = pattern, ''
    
    if len(query) < len(pattern.replace('*', '')):
        print("NO")
        continue
    
    if not all(c in good_letters or c == '?' for c in prefix + suffix):
        print("NO")
        continue
    
    i, j = 0, 0
    while i < len(prefix) and j < len(query) and (prefix[i] == query[j] or prefix[i] == '?'):
        i += 1
        j += 1
    
    if i < len(prefix):
        print("NO")
        continue
    
    i, j = len(suffix) - 1, len(query) - 1
    while i >= 0 and j >= 0 and (suffix[i] == query[j] or suffix[i] == '?'):
        i -= 1
        j -= 1
    
    if i >= 0:
        print("NO")
        continue
    
    if '*' in pattern:
        if any(c in good_letters for c in query[j-len(suffix)+1:j+1]):
            print("NO")
        else:
            print("YES")
    else:
        print("YES")
