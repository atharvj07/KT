n = int(input())
a = list(map(int, input().split()))

diffs = [a[i] - a[i-1] for i in range(1, n)]

possible_k = []

for k in range(1, n+1):
    ok = True
    for i in range(1, n-k+1):
        if diffs[i] != diffs[i+k-1]:
            ok = False
            break
    if ok:
        possible_k.append(k)

print(len(possible_k))
print(*possible_k)
