n, k = map(int, input().split())
skills = list(map(int, input().split()))
quarrels = [tuple(map(int, input().split())) for _ in range(k)]

quarrel_set = set(quarrels)

mentor_count = [0] * n

for i in range(n):
    for j in range(n):
        if i != j and skills[i] > skills[j] and (i+1, j+1) not in quarrel_set and (j+1, i+1) not in quarrel_set:
            mentor_count[i] += 1

print(*mentor_count)
