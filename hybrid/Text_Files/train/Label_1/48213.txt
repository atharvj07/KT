import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }

        int pos = -1;
        for (int i = 0; i < n - 1; i++) {
            if (a[i] > a[i + 1]) {
                if (pos == -1) {
                    pos = i;
                } else {
                    System.out.println("NO");
                    return;
                }
            }
        }

        if (pos == -1) {
            System.out.println("YES");
        } else if (pos == 0 || pos == n - 2) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
