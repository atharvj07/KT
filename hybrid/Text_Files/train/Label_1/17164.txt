#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> s(n);
  for (int i = 0; i < n; ++i) cin >> s[i];
  int res = 0, last = 0;
  for (int i = 0; i < n; ++i) {
    if (s[i] == 0 && last == 1) continue;
    res += 1;
    last = s[i];
  }
  cout << res << endl;
  return 0;
}
