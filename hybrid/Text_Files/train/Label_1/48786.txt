import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class SubnetMask {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        scanner.nextLine(); // Consume the newline

        List<Integer> ipAddresses = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String ipAddress = scanner.nextLine();
            ipAddresses.add(convertIpAddressToInt(ipAddress));
        }

        Collections.sort(ipAddresses);

        int subnetMask = findSubnetMask(ipAddresses, k);
        if (subnetMask == -1) {
            System.out.println("-1");
        } else {
            System.out.println(convertIntToIpAddress(subnetMask));
        }
    }

    private static int findSubnetMask(List<Integer> ipAddresses, int k) {
        int maxDiff = 0;
        for (int i = 0; i < ipAddresses.size() - 1; i++) {
            maxDiff = Math.max(maxDiff, ipAddresses.get(i + 1) - ipAddresses.get(i));
        }

        int subnetMask = 0;
        for (int i = 31; i >= 0; i--) {
            if ((maxDiff & (1 << i)) != 0) {
                subnetMask |= (1 << i);
            } else {
                break;
            }
        }

        if (countNetworks(ipAddresses, subnetMask) != k) {
            return -1;
        }

        return subnetMask;
    }

    private static int countNetworks(List<Integer> ipAddresses, int subnetMask) {
        int currentNetwork = -1;
        int count = 0;
        for (int ipAddress : ipAddresses) {
            int networkAddress = ipAddress & subnetMask;
            if (networkAddress != currentNetwork) {
                count++;
                currentNetwork = networkAddress;
            }
        }
        return count;
    }

    private static int convertIpAddressToInt(String ipAddress) {
        String[] parts = ipAddress.split("\\.");
        int result = 0;
        for (int i = 0; i < parts.length; i++) {
            result |= Integer.parseInt(parts[i]) << (24 - 8 * i);
        }
        return result;
    }

    private static String convertIntToIpAddress(int ipAddress) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 4; i++) {
            sb.append((ipAddress >> (24 - 8 * i)) & 0xFF).append(".");
        }
        return sb.substring(0, sb.length() - 1);
    }
}
