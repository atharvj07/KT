import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String[] parts = scanner.nextLine().split(" ");
        int n = Integer.parseInt(parts[parts.length - 1]);
        boolean possible = false;
        StringBuilder result = new StringBuilder();
        for (int a = 1; a <= n; a++) {
            for (int b = 1; b <= n; b++) {
                for (int c = 1; c <= n; c++) {
                    for (int d = 1; d <= n; d++) {
                        int e = n - a - b - c + d;
                        if (e >= 1 && e <= n) {
                            possible = true;
                            result.append(a).append(" + ").append(b).append(" - ").append(c).append(" + ").append(d).append(" + ").append(e).append(" = ").append(n);
                            break;
                        }
                    }
                    if (possible) {
                        break;
                    }
                }
                if (possible) {
                    break;
                }
            }
            if (possible) {
                break;
            }
        }
        System.out.println(possible ? "Possible\n" + result : "Impossible");
    }
}
