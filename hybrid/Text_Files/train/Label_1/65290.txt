N = int(input())
G = [[1e9 for i in range(N)] for j in range(N)]
for i in range(N):
    L = [int(x) for x in input().split()]
    for j in range(N):
        G[i][j] = min(G[i][j], L[j])
for k in range(N):
    for i in range(N):
        for j in range(N):
            G[i][j] = min(G[i][j], G[i][k] + G[k][j])
print(max(max(gg) for gg in G))
