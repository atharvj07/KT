k = int(input())
q = input()

distinct_chars = []
substrings = []
current_string = q[0]
distinct_chars.append(q[0])

for char in q[1:]:
    if char in distinct_chars:
        current_string += char
    else:
        substrings.append(current_string)
        current_string = char
        distinct_chars.append(char)
substrings.append(current_string)

if len(substrings) < k:
    print("NO")
else:
    print("YES")
    for i in range(k):
        if i < len(substrings) - 1:
            print(substrings[i])
        else:
            print(''.join(substrings[i:]))
            break
