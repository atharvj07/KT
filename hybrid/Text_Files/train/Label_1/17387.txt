#include <vector>
#include <algorithm>
#include <iostream>

using namespace std;

int solve(int n, int m, int x, int y, vector<vector<char>>& grid) {
    int dp[n + 1][m + 1];
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= m; j++) {
            dp[i][j] = 0;
        }
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (grid[i - 1][j - 1] == '.') {
                dp[i][j] = x + min(dp[i - 1][j], dp[i][j - 1]);
                if (j >= 2) {
                    dp[i][j] = min(dp[i][j], y + dp[i][j - 2]);
                }
            }
        }
    }

    return dp[n][m];
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, x, y;
        cin >> n >> m >> x >> y;
        vector<vector<char>> grid(n, vector<char>(m));
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                cin >> grid[i][j];
            }
        }
        cout << solve(n, m, x, y, grid) << endl;
    }
    return 0;
}
