#include <iostream>
using namespace std;
int main() {
    int n;
    cin >> n;
    int a[n], b[n];
    for (int i = 0; i < n - 1; i++) {
        cin >> a[i];
        cin >> b[i];
    }
    int t[n];
    bool flag = true;
    for (int i = 0; i < n - 1; i++) {
        if ((a[i] & b[i]) > 0) {
            flag = false;
            cout << "NO" << endl;
            break;
        }
        t[i] = a[i];
        t[i + 1] = (t[i] | b[i]);
    }
    if (flag) {
        t[n - 1] = b[n - 2];
        cout << "YES" << endl;
        for (int i = 0; i < n; i++) {
            cout << t[i] << " ";
        }
        cout << endl;
    }
}
