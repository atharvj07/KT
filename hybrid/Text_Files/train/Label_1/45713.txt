import java.util.*;
import static java.lang.Math.*;

public class Main {

    static final long MOD = 1000000007;
    static final int MAXN = 110;

    static class bit {
        long[] a = new long[MAXN];

        void upd(int i, long v) {
            for (; i < a.length; i |= i + 1)
                a[i] = (a[i] + v) % MOD;
        }

        long get(int i) {
            long v = 0;
            for (; i >= 0; i = (i & (i + 1)) - 1)
                v = (v + a[i]) % MOD;
            return v;
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt(), m = in.nextInt();
        long[] ar = new long[n], ac = new long[m];

        for (int i = 0; i < n; ++i)
            ar[i] = in.nextLong();

        for (int i = 0; i < m; ++i)
            ac[i] = in.nextLong();

        long sumc = 0;
        for (int i = 0; i < m; ++i)
            sumc += ac[i];

        bit b = new bit();

        for (int i = 0; i < n; ++i)
            b.upd(i, ar[i]);

        long ans = 0;

        for (int i = 0; i < m; ++i) {
            long s = 0;
            for (int j = 0; j < n; ++j)
                s += b.get(j) * ((ac[i] + s) % MOD);
            if (s % MOD != sumc) {
                System.out.println("NO");
                return;
            }
            ans += ac[i] % MOD;
        }

        System.out.println("YES");
        System.out.println(ans % MOD);

        for (int i = 0; i < n; ++i)
            for (int j = 0; j < m; ++j) {
                long x = j == 0 ? ar[i] : ac[j];
                for (int k = j - 1; k >= 0; --k)
                    x ^= ac[k];
                for (int k = i - 1; k >= 0; --k)
                    x ^= b.get(k);
                System.out.print((x + MOD) % MOD + " ");
            }
    }
}
