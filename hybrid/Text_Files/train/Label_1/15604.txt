#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    char a[n][m];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++) {
            if (i % 2 == 0) {
                if (j % 2 == 0)
                    a[i][j] = 'A';
                else
                    a[i][j] = 'B';
            } else {
                if (j % 2 == 1)
                    a[i][j] = 'A';
                else
                    a[i][j] = 'C';
            }
        }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) cout << a[i][j];
        cout << endl;
    }
    return 0;
}
