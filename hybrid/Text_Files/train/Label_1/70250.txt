import sys

class UnionFind:
    def __init__(self, n):
        self.par = [i for i in range(n)]
        self.sz = [1 for i in range(n)]

    def find(self, x):
        if self.par[x] != x:
            self.par[x] = self.find(self.par[x])
        return self.par[x]

    def union(self, x, y):
        rx, ry = self.find(x), self.find(y)
        if rx != ry:
            if self.sz[rx] > self.sz[ry]:
                self.par[ry] = rx
                self.sz[rx] += self.sz[ry]
            else:
                self.par[rx] = ry
                self.sz[ry] += self.sz[rx]

for _ in range(int(input())):
    n,a,b,da,db = map(int, input().split())
    edges = [map(int,sys.stdin.readline().split()) for _ in range(n-1)]
    uf = UnionFind(n+1)
    for u,v in edges:
        uf.union(u,v)
    a,b = uf.find(a), uf.find(b)
    if a==b:
        print('Alice')
    else:
        if da<db:
            print('Bob')
        else:
            print('Alice')
