#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll n, k, a, b;
    cin >> n >> k >> a >> b;
    ll ans = LLONG_MAX;
    for (int i = 0; i <= 100000; i += 1) {
        if ((n - i) % k == 0) {
            ans = min(ans, (n - i) / k * b + i * a);
        }
    }
    cout << ans << endl;
    return 0;
}
