import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int p = sc.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        TreeSet<Integer> q = new TreeSet<>();
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            Integer y = q.ceiling(t[i]);
            if (y == null) {
                ans[i] = t[i] + p;
            } else {
                ans[i] = y + p;
            }
            q.add(ans[i]);
        }
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
