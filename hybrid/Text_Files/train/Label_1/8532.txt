#include <bits/stdc++.h>
using namespace std;
#define int long long

const int N = 1e5 + 5;
int n, m;
int a[N];

int32_t main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] > m) {
            ans += a[i] - m;
            a[i] = m;
        }
    }
    int mx = 0;
    for (int i = 0; i < n; i++) { mx = max(a[i], mx); }
    for (int i = 0; i < n; i++) { ans += mx - a[i]; }
    cout << ans << "\n";
}
