#include <bits/stdc++.h>
using namespace std;

int main() {
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<string> s(n);
        for (string& str : s) cin >> str;
        vector<int> pref(52, 0);
        for (int i = 0, j = 0; i <= 50; i++) {
            while (j < 50 && s[0][j] == i % 2) ++j;
            pref[i] = j;
        }
        int ans = n;
        for (int i = 1; i < n; ++i) {
            int mismatch = 0;
            for (int j = 0, k = 0; j <= 50; ++j) {
                while (k < 50 && (s[i][k] == j % 2 || (s[0][k] == j % 2 && s[i][k] != s[0][k]))) ++k;
                mismatch += (50 - k);
            }
            ans = min(ans, mismatch / 2 + 1);
        }
        cout << ans << '\n';
    }
}
