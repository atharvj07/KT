def solve():
  t = int(input())
  for _ in range(t):
    n = int(input())
    s = input()
    a = [0] * (n + 1)
    for i in range(n):
      a[i + 1] = a[i]
      if s[i] == 'a': a[i + 1] += 1
    l = 1
    r = n
    while l <= r:
      m = (l + r) // 2
      if a[m] == a[n] - a[m]:
        print(m, n)
        return
      if a[m] < a[n] - a[m]:
        l = m + 1
      else:
        r = m - 1
    print(-1, -1)

solve()
