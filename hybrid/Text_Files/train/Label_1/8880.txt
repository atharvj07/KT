#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
const ld eps=1e-9;
ll n, l, v2, k;
ld v1;
int main(){
    cin>>n>>l>>v1>>v2>>k;
    if(v1>=v2){
        cout<<fixed<<setprecision(9)<<(ld)l/v1<<endl;
        return 0;
    }
    ll ans=0;
    while(n){
        if(n<=k) ans+=(l-ans*v1)/v2;
        else ans+=l/(v2*(n/k)+(v1*(n%k)));
        n-=min(n,k);
    }
    cout<<fixed<<setprecision(9)<<ans<<endl;
}
