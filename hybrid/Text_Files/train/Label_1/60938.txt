n, m = map(int, input().split())
scores = [list(map(int, input().split())) for _ in range(n)]

hate_pairs = set()
for _ in range(m):
    u, v = sorted(map(int, input().split()))
    hate_pairs.add((u-1, v-1))

total_scores = [0] * n
for i in range(n):
    for j in range(i+1, n):
        if (i, j) not in hate_pairs:
            total_scores[i] += min(scores[i][0] + scores[j][1], scores[i][1] + scores[j][0])
            total_scores[j] += min(scores[i][0] + scores[j][1], scores[i][1] + scores[j][0])

print(*total_scores)
