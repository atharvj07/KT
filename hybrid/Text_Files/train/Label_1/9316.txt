#include <bits/stdc++.h>
using namespace std;
#define fastio ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
#define rep(i, a, n) for (int i = a; i < n; i++)
#define per(i, a, n) for (ll i = n - 1; i >= a; i--)

int main() {
    fastio;
    int t; cin >> t;
    while (t--) {
        string s, ans = ""; cin >> s;
        int m; cin >> m; vector<int> b(m); rep(i, 0, m) { cin >> b[i]; }
        int sum = 0; for (int i = 0; i < m; i++) { sum += b[i]; }
        int n = s.size();
        int pref[n] = {0}; pref[0] = 0;
        for (int i = 1; i < n; i++) {
            pref[i] = pref[i - 1];
            if (s[i - 1] < s[i]) pref[i] += abs(i - i);
            else pref[i] += abs(i - (n - 1));
        }
        for (int i = 1; i < n; i++) {
            if (sum - pref[i] + abs(i - (n - 1)) >= b[i]) {
                sum -= abs(i - (n - 1));
                ans += s[i];
            }
            else {
                sum -= pref[i];
                ans += s[i - 1];
                int l = i + 1, r = n - 1;
                while (l <= r) {
                    int mid = (l + r) / 2;
                    if (sum - pref[mid] + abs(mid - i) >= b[i]) {
                        l = mid + 1;
                    }
                    else {
                        sum -= pref[mid];
                        ans += s[mid];
                        r = mid - 1;
                    }
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
