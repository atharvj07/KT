#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  bool ok = true;
  for (int i = 1; i < n; i++) {
    if (s[i] < s[i - 1]) {
      ok = false;
    }
  }
  if (ok) {
    cout << "YES" << endl;
    for (int i = 0; i < n; i++) {
      cout << "0";
    }
    cout << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
