#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define vi vector<int>
#define vll vector<ll>
#define pi pair<int, int>
#define pll pair<ll, ll>
#define pb push_back
#define mp make_pair
#define F first
#define S second
#define rep(i, a, b) for (int i = (int)(a); i < (int)(b); i++)
#define per(i, b, a) for (int i = (int)(b); i > (int)(a); i--)
#define all(x) x.begin(), x.end()
#define sz(x) (int)(x).size()
#define endl '\n'

void solve()
{
    int n;
    cin >> n;
    string s;
    cin >> s;
    map<pair<int, int>, int> m;
    m[make_pair(0, 0)]++;
    int x = 0, y = 0, ans = 0;
    for (int i = 0; i < n; i++)
    {
        if (s[i] == 'U')
            y++;
        if (s[i] == 'D')
            y--;
        if (s[i] == 'L')
            x--;
        if (s[i] == 'R')
            x++;
        ans += m[make_pair(x, y)];
        m[make_pair(x, y)]++;
    }
    cout << ans << endl;
}

int main()
{
    solve();

    return 0;
}
