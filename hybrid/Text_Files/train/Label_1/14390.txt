#include <bits/stdc++.h>
using namespace std;
#define MAXN 1010
const int dy[] = {0, 0, 1, -1};
const int dx[] = {1, -1, 0, 0};
vector<vector<int>> grid;
vector<vector<bool>> vis;
vector<int> p;
int n, m, p_cnt;
bool valid(int y, int x) {
  return y >= 0 && y < n && x >= 0 && x < m;
}
void bfs(int y, int x, int speed, int player) {
  queue<pair<int, int>> q;
  q.push({y, x});
  vis[y][x] = true;
  while (!q.empty()) {
    auto [y1, x1] = q.front();
    q.pop();
    grid[y1][x1] = player;
    for (int i = 0; i < 4; ++i) {
      int ny = y1 + dy[i], nx = x1 + dx[i];
      if (valid(ny, nx) && !vis[ny][nx] && grid[ny][nx] != '#' &&
          ((abs(y1 - ny) + abs(x1 - nx)) <= speed)) {
        vis[ny][nx] = true;
        q.push({ny, nx});
      }
    }
  }
}
int main() {
  int speed, player;
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n >> m >> p_cnt;
  p.resize(p_cnt + 1);
  for (int i = 0; i < p_cnt; ++i) cin >> p[i + 1];
  grid.resize(n, vector<int>(m));
  vis.resize(n, vector<bool>(m, false));
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      cin >> grid[i][j];
      if (grid[i][j] != '#' && grid[i][j] != '.' && grid[i][j] != '0') {
        player = grid[i][j];
        speed = p[player];
        bfs(i, j, speed, player);
      }
    }
  }
  for (int i = 0; i < p_cnt; ++i) {
    int ans = 0;
    for (int j = 0; j < n; ++j) {
      for (int k = 0; k < m; ++k) {
        if (grid[j][k] == i + 1) ans++;
      }
    }
    cout << ans << ' ';
  }
}
