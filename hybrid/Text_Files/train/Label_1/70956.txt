import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    i = 0
    ans = 0
    while i < n - 1:
        j = i + 1
        while j < n and p[j] > p[j - 1]:
            j += 1
        if j < n:
            ans += 1
        i = j
    print(ans)
