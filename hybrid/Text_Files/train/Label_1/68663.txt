n, m = map(int, input().split())
table = [list(map(int, input().split())) for _ in range(n)]

# Preprocess to find columns that are sorted up to each row
sorted_until = [[0] * m for _ in range(n)]
for j in range(m):
    for i in range(1, n):
        sorted_until[i][j] = sorted_until[i-1][j] + (table[i][j] >= table[i-1][j])

# Find the maximum row up to which each column is sorted
max_sorted_row = [0] * m
for j in range(m):
    for i in range(n):
        if sorted_until[i][j] == i:
            max_sorted_row[j] = i

k = int(input())
for _ in range(k):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    if any(max_sorted_row[j] >= r for j in range(m)):
        print("Yes")
    else:
        print("No")
