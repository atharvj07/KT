#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> h(n);
    for (int i = 0; i < n; i++) { cin >> h[i]; }
    int cur = 0, ans = 0;
    for (int i = 0; i < n; i++) {
        if (cur != h[i]) { ans += abs(cur - h[i]); }
        cur = h[i];
        ans++;
    }
    cout << ans << endl;
    return 0;
}
