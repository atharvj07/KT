#include <bits/stdc++.h>
using namespace std;
typedef unsigned long long ll;
int main() {
  ll n, m, cnt = 0;
  cin >> n >> m;
  unordered_map<ll, vector<ll>> h;
  for (ll i = 0; i < n; ++i) {
    ll x;
    cin >> x;
    h[x].push_back(i);
  }
  vector<bool> v(n);
  for (ll i = 0; i < m; ++i) {
    ll x1, x2, y;
    cin >> x1 >> x2 >> y;
    --y;
    if (x1 == x2) {
      v[y] = 1;
      ++cnt;
    } else if (h[x1].size() && h[x2].size()) {
      ll l = lower_bound(h[x1].begin(), h[x1].end(), y) - h[x1].begin();
      ll r = lower_bound(h[x2].begin(), h[x2].end(), y) - h[x2].begin();
      if (l == r) ++cnt;
    }
  }
  cout << cnt << endl;
  return 0;
}
