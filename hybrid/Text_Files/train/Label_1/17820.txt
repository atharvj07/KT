#include <iostream>
int main() {
  long n, k;
  std::cin >> n >> k;
  std::string s;
  std::cin >> s;
  long ans = 1;
  for (long i = 0; i < n - 1; i++) {
    long cnt = 1;
    while (i + cnt < n && s[i + cnt] == s[i]) cnt++;
    if (cnt % 2) { cnt--; }
    if (i > 0 && s[i] != s[i - 1]) cnt++;
    cnt = std::min(cnt, 2 * k + 1);
    ans = std::max(ans, cnt);
  }
  std::cout << ans << "\n";
}
