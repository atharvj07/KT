#include <iostream>
#define ull unsigned long long

const ull M = 1000000007ULL;

int main() {
  ull n, k;
  std::cin >> n >> k;
  ull dp[n+1][3];
  dp[1][0] = 1ULL;
  dp[1][1] = 0ULL;
  dp[1][2] = 0ULL;
  for (ull i = 2; i <= n; ++i) {
    if (i <= k) {
      dp[i][0] = (dp[i-1][0] + dp[i-1][1]) % M;
      dp[i][1] = (dp[i-1][0] + dp[i-1][1] + dp[i-1][2]) % M;
      dp[i][2] = 0ULL;
    } else {
      dp[i][0] = (dp[i-1][0] + dp[i-1][1] + dp[i-1][2]) % M;
      dp[i][1] = dp[i-1][0] % M;
      dp[i][2] = (dp[i-1][0] + dp[i-1][1] + dp[i-1][2]) % M;
    }
  }
  
  std::cout << dp[n][0] % M << '\n';
  return 0;
}
