n, m = map(int, input().split())
friends = [set(map(int, input().split()[1:])) for _ in range(n)]
pizzas = [tuple(map(int, input().split())) for _ in range(m)]

max_pleased = 0
min_cost = float('inf')
best_pair = (0, 0)

for i in range(m):
    for j in range(i+1, m):
        combined_ingredients = set(pizzas[i][2:]).union(pizzas[j][2:])
        pleased = sum(all(ingredient in combined_ingredients for ingredient in friend) for friend in friends)
        cost = pizzas[i][0] + pizzas[j][0]
        if pleased > max_pleased or (pleased == max_pleased and cost < min_cost):
            max_pleased, min_cost = pleased, cost
            best_pair = (i + 1, j + 1)

print(*best_pair)
