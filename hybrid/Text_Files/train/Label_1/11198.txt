#include<bits/stdc++.h>
using namespace std;
typedef long long lld;

lld f(lld n,lld w,lld k)
{
    lld ans=n;
    for(lld i=1;i<=w;i++)
    {
        ans=ans-i*k;
        if(ans<0)
        {
            return (lld)(i*k-n);
        }
    }
    return (lld)0;
}

int main()
{
    lld k,n,w;
    cin>>k>>n>>w;
    cout<<(lld)f(n,w,k)<<endl;
    return 0;
}
