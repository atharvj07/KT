import heapq

n, m, k = map(int, input().split())
g = [[] for _ in range(n+1)]
for _ in range(m):
  a, b, w = map(int, input().split())
  g[a].append((b, w))
  g[b].append((a, w))
vis = [False for _ in range(n+1)]
dist = [[10**9 for _ in range(n+1)] for _ in range(n+1)]
for i in range(n+1):
  dist[i][i] = 0
q = [(0, 1)]
while q:
  d, u = heapq.heappop(q)
  if vis[u]:
    continue
  vis[u] = True
  for v, w in g[u]:
    if not vis[v]:
      if dist[1][v] > d+w:
        dist[1][v] = d+w
        heapq.heappush(q, (d+w, v))
      if dist[u][1] > d+w:
        dist[u][1] = d+w
        heapq.heappush(q, (d+w, u))
res = []
for i in range(1, n+1):
  for j in range(i+1, n+1):
    if dist[i][j] != 10**9:
      res.append(dist[i][j])
res.sort()
print(res[k-1])
