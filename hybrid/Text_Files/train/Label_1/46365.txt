import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = line.charAt(j) - '0';
            }
        }

        boolean[][] canBeIgnored = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == 1) {
                    canBeIgnored[i][j] = true;
                    for (int k = 0; k < n; k++) {
                        if (a[k][j] == 1 && k != i) {
                            canBeIgnored[i][j] = false;
                            break;
                        }
                    }
                }
            }
        }

        boolean canIgnore = false;
        for (int i = 0; i < n; i++) {
            boolean canIgnoreThis = true;
            for (int j = 0; j < m; j++) {
                if (a[i][j] == 1 && !canBeIgnored[i][j]) {
                    canIgnoreThis = false;
                    break;
                }
            }
            if (canIgnoreThis) {
                canIgnore = true;
                break;
            }
        }

        System.out.println(canIgnore ? "YES" : "NO");
    }
}
