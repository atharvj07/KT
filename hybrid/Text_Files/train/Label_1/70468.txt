import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    islands = [tuple(map(int, sys.stdin.readline().split())) for _ in range(n)]
    bridges = [int(sys.stdin.readline()) for _ in range(m)]

    bridges.sort()

    i = 0
    used = [False] * m

    for l, r in islands[:-1]:
        while i < m and bridges[i] < r - l:
            i += 1
        if i == m:
            return "No"
        used[i] = True
        i += 1

    return "Yes\n" + " ".join(map(str, [i + 1 for i, u in enumerate(used) if u]))

print(solve())
