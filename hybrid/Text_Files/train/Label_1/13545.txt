#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;

int n, m;
vector<int> a, b;

int main() {
    cin >> n >> m;
    a.resize(n);
    for (int &ai : a) cin >> ai;
    b.resize(n + 1);
    for (int i = 1; i <= n; ++i) b[i] = a[i - 1];
    for (int i = 0; i < m; ++i) {
        int l, r, x;
        cin >> l >> r >> x;
        if (b[r] != x)
            cout << r << endl;
        else if (b[l - 1] != x)
            cout << l << endl;
        else
            cout << -1 << endl;
    }
    return 0;
}
