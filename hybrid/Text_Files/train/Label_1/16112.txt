#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
const ll inf = 1e18;

struct gadget{
	int t, c;
	gadget(){}
	gadget(int t, int c): t(t), c(c){}
	bool operator < (const gadget & a){
		return c < a.c;
	}
};

int n, m, k, s;
ll a[N], b[N];
gadget v[N];
ll dp[N]; // day, number of bought gadget

int main(){
	cin >> n >> m >> k >> s;
	for(int i = 1; i <= n; i ++){
		cin >> a[i];
	}
	for(int i = 1; i <= n; i ++){
		cin >> b[i];
	}
	for(int j = 1; j <= m; j ++){
		int t, c; cin >> t >> c;
		v[j] = gadget(t, c);
	}

	sort(v + 1, v + m + 1);
	for(int i = 1; i <= k; i ++){
		dp[i] = inf;
	}
	dp[0] = 0;

	for(int i = 1; i <= m; i ++){
		for(int j = k; j >= v[i].c; j --){
			dp[j] = min(dp[j], dp[j - v[i].c] + 1ll * v[i].t * (v[i].t == 1 ? a[j] : b[j]));
		}
	}

	if(dp[k] != inf){
		cout << dp[k] << endl;
		int j = k;
		vector<pair<int, int>> ans;
		for(int i = m; i >= 1; i --){
			if(j - v[i].c >= 0 && dp[j] == dp[j - v[i].c] + 1ll * v[i].t * (v[i].t == 1 ? a[j] : b[j])){
				ans.emplace_back(i, dp[j] - dp[j - v[i].c]);
				j -= v[i].c;
			}
		}
		sort(ans.begin(), ans.end());
		for(auto p: ans){
			cout << p.first << " " << p.second << endl;
		}
	} else{
		cout << -1 << endl;
	}
	return 0;
}
