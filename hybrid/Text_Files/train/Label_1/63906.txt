n = int(input().strip())
a = list(map(int, input().strip().split()))

def solve(a):
    operations = []
    max_count = max(a, key=a.count)
    index_max = a.index(max_count)
    
    for i in range(index_max - 1, -1, -1):
        if a[i] != max_count:
            operations.append((1, i + 1, i + 2))
            a[i] = max_count
            
    for i in range(index_max + 1, len(a)):
        if a[i] != max_count:
            operations.append((1, i + 1, i))
            a[i] = max_count
            
    return operations

ops = solve(a)
print(len(ops))
for op in ops:
    print(*op)
