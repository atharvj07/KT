import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.List;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        // Read the number of strings
        int n = Integer.parseInt(reader.readLine());

        // Read the strings
        List<String> strings = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            strings.add(reader.readLine());
        }

        // Read the number of queries
        int q = Integer.parseInt(reader.readLine());

        // Read the queries
        for (int i = 0; i < q; i++) {
            String[] query = reader.readLine().split(" ");
            int k = Integer.parseInt(query[0]);
            String order = query[1];

            // Create a comparator based on the order
            Comparator<String> comparator = new Comparator<String>() {
                @Override
                public int compare(String s1, String s2) {
                    int len = Math.min(s1.length(), s2.length());
                    for (int j = 0; j < len; j++) {
                        int index1 = order.indexOf(s1.charAt(j));
                        int index2 = order.indexOf(s2.charAt(j));
                        if (index1 != index2) {
                            return index1 - index2;
                        }
                    }
                    return s1.length() - s2.length();
                }
            };

            // Sort the strings
            strings.sort(comparator);

            // Output the sequence of the string
            System.out.println(strings.indexOf(strings.get(k - 1)) + 1);
        }
    }
}
