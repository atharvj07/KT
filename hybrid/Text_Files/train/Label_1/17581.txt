#include <bits/stdc++.h>
using namespace std;
int t, q, a, b;
map<int, int> m;
int main() {
  cin >> q;
  while (q--) {
    cin >> t >> a;
    if (t == 1) {
      if (m.find(a) == m.end()) { cout << a << " "; }
      m[a]++;
    } else {
      cin >> b;
      auto it = m.find(a);
      if (it == m.end()) continue;
      int temp = it->second;
      m.erase(it);
      m[b] += temp;
    }
  }
  return 0;
}
