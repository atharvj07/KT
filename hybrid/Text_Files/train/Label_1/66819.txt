def get_min_hamming(s: str, t: str) -> tuple:
    n = len(s)
    dist = sum(s[i] != t[i] for i in range(n))
    if dist <= 1:
        return dist, (-1, -1)
    ans = dist - 2
    a = -1
    b = -1
    for i in range(n):
        for j in range(n):
            if i != j and s[i] != t[i] and s[j] != t[j] and s[i] == t[j] and s[j] == t[i]:
                if ans < dist - 2:
                    ans = dist - 2
                    a = i
                    b = j
    return dist - ans, (a + 1, b + 1)

n = int(input())
s = input()
t = input()
ans, pos = get_min_hamming(s, t)
print(ans)
print(*pos)
