#include <bits/stdc++.h>
using namespace std;
int m[5005][5005], t;
char s[5005];
int main() {
  for (cin >> t; t--;) {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> m[i][i];
    for (int l = 2; l <= n; l++) {
      for (int i = 1; i <= n - l + 1; i++) {
        int j = i + l - 1;
        if (l == 2)
          m[i][j] = m[i][j] == m[i + 1][j];
        else
          m[i][j] = m[i][j] == m[i + 1][j] && m[i][j] == m[i][j - 1];
      }
    }
    bool f = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = i + 2; j <= n; j++) if (m[i][j]) f = 1;
    }
    puts(f ? "YES" : "NO");
  }
}
