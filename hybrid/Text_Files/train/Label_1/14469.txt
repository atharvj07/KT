#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  ll n, a[100005];
  map<ll, ll> mp;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    mp[a[i]]++;
  }
  ll cnt = 0;
  for (int i = 0; i < n; i++) {
    if (!mp.count(cnt + 1)) {
      while (mp[a[i]] > 0) {
        a[i] = cnt + 1;
        mp[a[i]]--;
        cnt++;
      }
    }
    cout << a[i] << " ";
  }
}
