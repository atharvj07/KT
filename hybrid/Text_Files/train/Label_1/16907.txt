#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 1000001;
int n, u, v, l[N];
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> u >> v;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    memset(l, 0, sizeof(l));
    fill(l, l + a[0], v);
    for (int i = 1; i < n; i++) {
      l[a[i]] = v;
      if (a[i] > a[i - 1]) {
        for (int j = a[i - 1] + 1; j < a[i]; j++) l[j] = min({l[j], u + l[j - 1], v + l[j + 1]});
      } else {
        for (int j = a[i - 1] - 1; j > a[i]; j--) l[j] = min({l[j], u + l[j + 1], v + l[j - 1]});
      }
    }
    cout << l[1000000] << endl;
  }
}
