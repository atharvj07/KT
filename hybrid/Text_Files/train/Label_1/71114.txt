n, m = map(int, input().split())
adj = [[] for _ in range(n + 1)]
labels = [[] for _ in range(n + 1)]
indegree = [0] * (n + 1)

for _ in range(m):
    t = input().split()
    if t[0] == '+':
        u, v, c = int(t[1]), int(t[2]), t[3]
        adj[u].append(v)
        labels[u].append(c)
        indegree[v] += 1
    elif t[0] == '-':
        u, v = int(t[1]), int(t[2])
        ind = adj[u].index(v)
        adj[u].pop(ind)
        labels[u].pop(ind)
        indegree[v] -= 1
    else:
        k = int(t[1])
        vis = [False] * (n + 1)
        last = [{} for _ in range(n + 1)]
        cur = [{} for _ in range(n + 1)]
        cycle = False

        for i in range(1, n + 1):
            if indegree[i] == 0:
                vis[i] = True
                cur[i][labels[i][0]] = [i]

        while True:
            nxt = [{} for _ in range(n + 1)]
            for i in range(1, n + 1):
                if vis[i]:
                    for j in range(len(adj[i])):
                        v = adj[i][j]
                        if not vis[v]:
                            for c in labels[i][j]:
                                if c not in nxt[v]:
                                    nxt[v][c] = []
                                nxt[v][c].extend(cur[i][c])
                                if len(nxt[v][c]) == k - 1:
                                    if v == nxt[v][c][0]:
                                        cycle = True
                                    else:
                                        last[v][c] = nxt[v][c].pop()
            if all(vis):
                break
            cur = nxt
            vis = [v != [] for v in vis]

        if cycle:
            print('YES')
        else:
            while last != [{} for _ in range(n + 1)]:
                cycle = False
                for i in range(1, n + 1):
                    for c in labels[i]:
                        if last[i] != {} and last[i][c] != {}:
                            last[i], last[last[i][c]] = last[last[i][c]], {}
                            if i == last[i][c]:
                                cycle = True
                if all(v == {} for v in last):
                    break
                if cycle:
                    break
            if cycle:
                print('YES')
            else:
                print('NO')
