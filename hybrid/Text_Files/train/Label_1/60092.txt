n=int(input())
l=[int(x) for x in input().split()]
t=[1]*n
s=input()
t[0]=0
t[n-1]=0
for i in range(1,n):
    if s[i-1]!=s[i]:
        t[i]=1
    else:
        t[i]=t[i-1]
if s[0]=='W':
    t[0]=3
if s[n-1]=='W':
    t[n-1]=3
for i in range(n-2,-1,-1):
    if s[i]==s[i+1]:
        t[i]=min(t[i+1],t[i])
    else:
        t[i]=t[i+1]
ans=0
for i in range(n-1):
    if s[i]=='G':
        ans+=5*l[i]
    elif s[i]=='W':
        ans+=3*l[i]
    else:
        ans+=l[i]
    ans+=t[i]
print(ans)
