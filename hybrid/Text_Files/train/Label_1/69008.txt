import sys

n,m=map(int,input().split())
a=list(map(int,input().split()))
lcms=[0 for i in range(0,n+2)]

cur=1
for i in range(0,n):
    cur=a[i]//cur*cur
    lcms[i+1]=cur
    cur=cur//a[i]*(a[i],cur)

for i in range(n-1,-1,-1):
    lcms[i]=max(lcms[i],lcms[i+1])

ans=[0,0]
l=-1

for i in range(0,n):
    if lcms[i]<=m and lcms[i]>=l:
        l=lcms[i]
        ans=[l,i+1]

print(*ans)

cur=1
lst=[]
for i in a:
    if cur<=l and cur*i<=l:
        cur=cur*i//cur*cur
        lst.append(i)
    if cur>l:
        break

print(*lst)
