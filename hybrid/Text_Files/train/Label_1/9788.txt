#include <bits/stdc++.h>
using namespace std;
#define ll long long int
int main() {
  ll n;
  cin >> n;
  vector<ll> a(n), diff(n), ans(n);
  for (ll i = 0; i < n; i++) {
    cin >> a[i];
    diff[i] = abs(a[i] - (i + 1));
  }
  ll ans1 = 0, ans2 = 0;
  for (ll i = 1; i < n; i++) {
    if (diff[i] < diff[ans1]) ans1 = i;
  }
  ll sum = 0;
  for (ll i = 0; i < n; i++) sum += abs(a[(i + ans1) % n] - (i + 1));
  ans2 = sum;
  for (ll i = 0; i < n; i++) {
    if (diff[i] == diff[ans1]) {
      ll x = 0;
      for (ll j = 0; j < n; j++) x += abs(a[(j + i) % n] - (j + 1));
      if (x < ans2) {
        ans1 = i;
        ans2 = x;
      }
    }
  }
  cout << ans2 << " " << ans1 << '\n';
}
