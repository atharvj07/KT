t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    even_count = sum(1 for x in a if x % 2 == 0)
    odd_count = n - even_count

    if abs(even_count - odd_count) > 1:
        print(-1)
    else:
        even_positions = [i for i, x in enumerate(a) if x % 2 == 0]
        odd_positions = [i for i, x in enumerate(a) if x % 2 == 1]
        even_swaps = sum(abs(even_positions[i] - i) for i in range(len(even_positions)))
        odd_swaps = sum(abs(odd_positions[i] - i) for i in range(len(odd_positions)))
        
        if even_count == odd_count:
            print(min(even_swaps, odd_swaps) // 2)
        elif even_count > odd_count:
            print(even_swaps // 2)
        else:
            print(odd_swaps // 2)
