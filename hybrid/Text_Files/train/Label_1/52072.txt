import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    String l = in.next();
    while (l.length() < n) {
      l = '0' + l;
    }
    String r = in.next();
    long ans = 0;
    for (int i = n - 1; i >= 0; i--) {
      int bitl = l.charAt(i) - '0';
      if (r.charAt(i) == '1') {
        ans |= (1L << i);
      } else if (bitl == 0) {
        ans |= (1L << i);
      }
    }
    System.out.println(Long.toBinaryString(ans));
  }
}
