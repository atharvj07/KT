#include<bits/stdc++.h>
#define PB push_back
#define ll long long
using namespace std;
int n,m,a[1000005],b[1000005],c[1000005],cnt;
int main()
{
    int x,x1,x2,y;
    cin >> n >> m;
    while(n--)
    {
        cin >> x;
        a[x]=1;
    }
    while(m--)
    {
        cin >> x1 >> x2 >> y;
        b[x1]=1,c[x2]=1;
    }
    for(int i=1; i<=1000000; i++)
    {
        if(a[i]) cnt++;
        else 
        {
            if(b[i]&&c[i]) cnt+=2;
        }
    }
    cout << cnt << endl;
    return 0;
}
