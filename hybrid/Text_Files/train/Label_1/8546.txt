#include<bits/stdc++.h>
using namespace std;

int main()
{
    int t;
    cin >> t;
    while(t--)
    {
        unordered_map<string, int> mp;
        int n;
        cin >> n;
        vector<string> names;
        for(int i=0; i<n; i++)
        {
            string s;
            cin >> s;
            names.push_back(s);
            mp[s] = 0;
        }
        int m;
        cin >> m;
        vector<string> ans;
        for(int i=0; i<m; i++)
        {
            string s;
            getline(cin, s);
            string name, mess;
            istringstream iss(s);
            int f = 0;
            iss >> name;
            iss.clear();
            iss.str(s);
            iss >> mess;
            if(name.back() == ':')
            {
                name.pop_back();
                ans.push_back(name + ":" + mess);
                mp[name] = 1;
            }
            else
            {
                for(auto x:names)
                {
                    if(mp[x] == 0 && mess.find(x) == string::npos)
                    {
                        ans.push_back(x + ":" + mess);
                        mp[x] = 1;
                        f = 1;
                        break;
                    }
                }
                if(f == 0)
                {
                    cout << "Impossible" << endl;
                    break;
                }
            }
        }
        if(ans.size() == m)
        {
            for(auto x:ans)
                cout << x << endl;
        }
        mp.clear();
        names.clear();
        ans.clear();
    }
    return 0;
}
