#include <bits/stdc++.h>
#define ll long long
#define mod 1000000007
using namespace std;

int32_t main() {
  int n;
  cin >> n;
  vector<pair<ll, ll>> v; // price, type (0: accept or 1: add)
  ll odd = 0, even = 1;
  while (n--) {
    string s;
    ll x;
    cin >> s >> x;
    if (s == "ADD")
      v.push_back({x, 1});
    else {
      v.push_back({x, 0});
      odd ^= 1; // max 3 adds and 4 accepts -> max xor odd = 1
    }
  }
  sort(v.begin(), v.end());
  int cnt = 0;
  for (auto x : v) {
    if (x.second)
      cnt++;
    else
      odd ^= 1;
    if (cnt > odd) {
      cout << 0;
      return 0;
    }
  }
  cout << (even << odd) % mod;
}
