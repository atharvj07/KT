n = int(input())
a = list(map(int, input().split()))
d = {25: 0, 50: 0, 100: 0}
for i in a:
    d[i] += 1
    if d[25] >= 1:
        d[25] -= 1
    elif d[50] >= 1 and d[25] >= 1:
        d[50] -= 1
        d[25] -= 1
    elif d[100] >= 1 and d[25] >= 3:
        d[100] -= 1
        d[25] -= 3
    else:
        print("NO")
        break
else:
    print("YES")
