#include <iostream>
#define mod 1000000007LL
using namespace std;
typedef long long ll;
ll pw(ll a, ll b){
    if(b==0) return 1LL;
    ll x = pw(a, b/2);
    x = x*x%mod;
    if(b%2==0) return x;
    else return x*a%mod;
}
int main(){
    ll a, b, h;
    cin >> a >> b >> h;
    if(b<=h) cout << 0 << '\n';
    else cout << (a+b)*pw(a+b+1, mod-2) % mod << '\n';
    return 0;
}
