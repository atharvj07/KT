import java.util.Scanner;

public class CokeParty {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int desiredConcentration = scanner.nextInt();
        int cokeTypes = scanner.nextInt();
        int[] cokeConcentrations = new int[cokeTypes];
        for (int i = 0; i < cokeTypes; i++) {
            cokeConcentrations[i] = scanner.nextInt();
        }

        int minVolume = findMinVolume(desiredConcentration, cokeConcentrations);
        System.out.println(minVolume);
    }

    private static int findMinVolume(int desiredConcentration, int[] cokeConcentrations) {
        int minVolume = Integer.MAX_VALUE;
        for (int i = 0; i < cokeConcentrations.length; i++) {
            int remainingConcentration = desiredConcentration - cokeConcentrations[i];
            if (remainingConcentration >= 0) {
                int volume = 1;
                while (remainingConcentration > 0) {
                    volume++;
                    remainingConcentration = desiredConcentration - volume * cokeConcentrations[i];
                }
                if (volume < minVolume) {
                    minVolume = volume;
                }
            }
        }

        if (minVolume == Integer.MAX_VALUE) {
            return -1;
        } else {
            return minVolume;
        }
    }
}
