n = int(input())
a = list(map(int, input().split()))

neg = [x for x in a if x < 0]
pos = [x for x in a if x > 0]
zero = [x for x in a if x == 0]

if len(pos) == 0:
    pos.append(neg.pop())
    pos.append(neg.pop())

if len(neg) % 2 == 0:
    zero.append(neg.pop())

print(len(neg), ' '.join(map(str, neg)))
print(len(pos), ' '.join(map(str, pos)))
print(len(zero), ' '.join(map(str, zero)))
