import sys
input = sys.stdin.readline

n = int(input())
legs = list(map(int, input().split()))
del_cost = list(map(int, input().split()))
legs.sort(reverse=True)
del_cost.sort()

cost = 0
for i in range(n):
    if i >= n//2 and legs[i] != legs[n//2 - 1]:
        cost += del_cost[i]
    else:
        break

print(cost)
