#include <cstdio>
#include <vector>
#include <algorithm>
const int mod = 1000000007;
int n, cnt[100005], pw[100005], ans;
bool c[100005];
std::vector<int> g[100005];
void dfs(int u, int f) {
    cnt[u] = c[u];
    for (int i = 0; i < g[u].size(); ++i) {
        int v = g[u][i];
        if (v != f) {
            dfs(v, u);
            cnt[u] += cnt[v];
        }
    }
}
void sub1() {
    dfs(0, 0);
    int res = 0;
    for (int i = 0; i < n; ++i) {
        if (cnt[i] > 0) ++res;
    }
    pw[0] = 1;
    for (int i = 1; i <= n; ++i) {
        pw[i] = pw[i - 1] * 2;
        if (pw[i] >= mod) pw[i] -= mod;
    }
    ans = pw[res] - 1;
    if (ans < 0) ans += mod;
}
void sub2() {
    std::sort(cnt, cnt + n);
    int res = 0;
    for (int i = 0; i < n; ++i) {
        if (cnt[i] > 0) {
            res++;
            if (i == 0 || cnt[i] != cnt[i - 1]) {
                ans = ans * 2;
                if (ans >= mod) ans -= mod;
            }
        }
    }
    ans--;
    if (ans < 0) ans += mod;
}
int main() {
    scanf("%d", &n);
    for (int i = 0; i + 1 < n; ++i) {
        int x;
        scanf("%d", &x);
        g[i + 1].push_back(x);
        g[x].push_back(i + 1);
    }
    for (int i = 0; i < n; ++i) scanf("%d", &c[i]);
    if (n <= 5000) sub1();
    else sub2();
    printf("%d\n", ans);
    return 0;
}
