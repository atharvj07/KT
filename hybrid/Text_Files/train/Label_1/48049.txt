import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int X = sc.nextInt();
        int N = sc.nextInt();
        int[] p = new int[N];
        for (int i = 0; i < N; i++) {
            p[i] = sc.nextInt();
        }
        Arrays.sort(p);
        int ans = X;
        for (int i = 0; i < N; i++) {
            if (Math.abs(X - p[i]) < Math.abs(X - ans)) {
                ans = p[i];
            }
        }
        if (X < p[0]) {
            ans = p[0];
        } else if (X > p[N - 1]) {
            ans = p[N - 1];
        }
        System.out.println(ans);
    }
}
