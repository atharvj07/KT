#include <iostream>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    char a[n][m];
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
        }
    }
    int cntR = 0, cntG = 0, cntB = 0;
    for (int i = 0; i < n; i++) {
        char c = a[i][0];
        int cnt = 1;
        for (int j = 1; j < m; j++) {
            if (c != a[i][j]) {
                cnt++;
                c = a[i][j];
            }
        }
        if (c == 'R') cntR = cnt;
        else if (c == 'G') cntG = cnt;
        else if (c == 'B') cntB = cnt;
    }
    if (cntR == cntG && cntR == cntB && cntR > 0) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
