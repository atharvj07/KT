#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[1005];
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  int cnt = 0;
  for (int i = 2; i + 1 <= n; i++) {
    if ((a[i - 1] > a[i] && a[i] < a[i + 1]) ||
        (a[i - 1] < a[i] && a[i] > a[i + 1])) {
      cnt++;
    }
  }
  cout << cnt << endl;
}
