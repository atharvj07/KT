#include <iostream>
#include <vector>
using namespace std;

const int N = 1001;
int n, m, p, s[N], grid[N][N];
int dx[] = {-1, 0, 1, 0};
int dy[] = {0, 1, 0, -1};
vector<pair<int, int>> castles[N];

int main() {
  cin >> n >> m >> p;
  for (int i = 0; i < p; i++) { cin >> s[i]; }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      char c;
      cin >> c;
      grid[i][j] = c == '.' ? 0 : c - '0';
      if (grid[i][j] > 0) { castles[grid[i][j]].push_back({i, j}); }
    }
  }
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < n; j++) {
      for (int k = 0; k < s[grid[i][j]]; k++) {
        for (int dir = 0; dir < 4; dir++) {
          int x = i + dx[dir] * k;
          int y = j + dy[dir] * k;
          if (x >= 0 && x < m && y >= 0 && y < n) {
            grid[y][x] |= grid[i][j];
            castles[grid[y][x]].push_back({y, x});
          }
        }
      }
    }
  }
  for (int i = 1; i <= p; i++) { cout << castles[i].size() << ' '; }
  return 0;
}
