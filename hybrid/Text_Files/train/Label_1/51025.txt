import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int ans = 0;
        for (int i = 0; i < s.length(); i++) {
            if (Character.isLetter(s.charAt(i))) {
                ans += s.charAt(i) - 'A' + 1;
            } else if (s.charAt(i) == '.') {
                ans *= -1;
            }
        }
        System.out.println(ans);
    }
}
