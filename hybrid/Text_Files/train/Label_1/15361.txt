#include <iostream>
#include <unordered_map>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
int main() {
    ll n, m;
    cin >> n >> m;
    string s;
    cin >> s;
    unordered_map<int, ll> cnt;
    cnt[0] = 1;
    ll ans = 0;
    ll sum = 0;
    for (int i = 0; i < m; i++) {
        sum += (s[i] == '(' ? 1 : -1);
        if (cnt.count(sum))
            ans = (ans + cnt[sum]) % mod;
        cnt[sum]++;
    }
    cout << ans << endl;
    return 0;
}
