import sys

m = int(input())
a = list(map(str.lower, input().split()))
n = int(input())
d = {}
for i in range(n):
    x, y = map(str.lower, input().split())
    if x not in d: d[x] = set()
    if y not in d: d[y] = set()
    d[x].add(y)
    d[y].add(x)

# DP + DAG shortest path
dp = [(0, len(a[0]))] + [(1e9, 1e9)] * m

for i in range(1, m):
    for j in range(i):
        if a[i] in d and a[j] in d[a[i]]:
            if dp[j][0] + 1 < dp[i][0]:
                dp[i] = (dp[j][0] + 1, dp[j][1] + len(a[i]))
            elif dp[j][0] + 1 == dp[i][0]:
                dp[i] = (dp[i][0], min(dp[i][1], dp[j][1] + len(a[i])))

    dp[i] = (dp[i][0], dp[i][1] + len(a[i]))
    if dp[i][0] > dp[i-1][0]:
        dp[i] = dp[i - 1]

ans = dp[m-1]
print(ans[0], ans[1])
