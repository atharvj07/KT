#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const int N = 2e3 + 1;
int n;
ll a[N][3];
vector<pair<int, int>> ans;

void init() {
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i][0] >> a[i][1] >> a[i][2];
    }
}

int main() {
    init();
    vector<int> id;
    for (int i = 1; i <= n; ++i) { id.push_back(i); }
    while (id.size() > 1) {
        bool ok = false;
        for (int i = 0; i < (int)id.size() && !ok; ++i) {
            for (int j = i + 1; j < (int)id.size(); ++j) {
                int u = id[i], v = id[j];
                bool good = true;
                for (int k = 0; k < (int)id.size(); ++k) {
                    if (k == i || k == j) continue;
                    int w = id[k];
                    for (int d = 0; d < 3; ++d) {
                        if (min(a[u][d], a[v][d]) <= a[w][d] &&
                            a[w][d] <= max(a[u][d], a[v][d])) {
                            good = false;
                            break;
                        }
                    }
                }
                if (good) {
                    ans.push_back({u, v});
                    id.erase(id.begin() + i);
                    id.erase(id.begin() + j - 1);
                    ok = true;
                    break;
                }
            }
        }
        if (!ok) {
            cout << "-1\n";
            return 0;
        }
    }
    for (auto p : ans) {
        cout << p.first << " " << p.second << "\n";
    }
}
