#include <iostream>
#include <vector>

using namespace std;

#define ll long long
#define mod 998244353

ll fastpow(ll b, ll p) {
  if (!p) return 1;
  if (p == 1) return b;
  ll ans = fastpow(b, p / 2);
  ans = (ans * ans) % mod;
  if (p % 2 == 1) ans = (ans * b) % mod;
  return ans;
}

ll inv(ll b) { return fastpow(b, mod - 2); }

int main() {
  ll n, m;
  cin >> n >> m;
  vector<vector<ll>> a(n, vector<ll>(m));
  for (ll i = 0; i < n; i++) {
    for (ll j = 0; j < m; j++) { cin >> a[i][j]; }
  }
  ll ans = 0;
  for (auto &i : a) {
    ll sum = 0;
    for (auto j : i) { sum += (j <= n); }
    ans += (sum * sum) % mod;
    ans %= mod;
  }
  cout << (ans * inv(n)) % mod << endl;
  return 0;
}
