#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  unordered_map<int, int> freq;
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    ans += freq[a[i]]++;
    for (int j = i + 1; j < n; j++) {
      if (freq[a[j]] < freq[a[i]]) ans++;
    }
  }
  cout << ans << endl;
}
