from collections import deque

t = int(input())

for _ in range(t):
    r, c = [int(x) for x in input().split()]
    grid = []
    for _ in range(r):
        grid.append(list(input()))

    if any('P' in row for row in grid):
        print('MORTAL')
    else:
        print(0)
        continue

    q = deque()
    for i in range(r):
        for j in range(c):
            if grid[i][j] == 'P':
                q.append((i, j, 0))
                q.append((i, j, 1))

    ans = 0
    while q:
        i, j, d = q.popleft()
        if i < 0 or i >= r or j < 0 or j >= c:
            continue
        if grid[i][j] == 'A':
            continue

        if d == 0:
            for k in range(i, r):
                if grid[k][j] == 'A':
                    break
                else:
                    grid[k][j] = 'A'
            ans += 1
        else:
            for k in range(j, c):
                if grid[i][k] == 'A':
                    break
                else:
                    grid[i][k] = 'A'
            ans += 1

    print(ans)
