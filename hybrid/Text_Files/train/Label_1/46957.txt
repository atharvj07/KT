import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        Graph graph = new Graph(n);
        for (int i = 0; i < m; i++) {
            int a = input.nextInt() - 1;
            int b = input.nextInt() - 1;
            graph.addEdge(a, b);
        }
        if (graph.canBeInteresting()) {
            System.out.println("YES");
            System.out.println(graph.getAdditionalEdges());
            for (Edge edge : graph.getEdgesToAdd()) {
                System.out.println((edge.a + 1) + " " + (edge.b + 1));
            }
        } else {
            System.out.println("NO");
        }
    }

    static class Graph {
        int n;
        List<List<Integer>> adj;
        int[] color;
        int[] cycleStart;
        int[] cycleLength;
        List<Edge> edgesToAdd;

        Graph(int n) {
            this.n = n;
            adj = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                adj.add(new ArrayList<>());
            }
            color = new int[n];
            cycleStart = new int[n];
            cycleLength = new int[n];
            edgesToAdd = new ArrayList<>();
        }

        void addEdge(int a, int b) {
            adj.get(a).add(b);
            adj.get(b).add(a);
        }

        boolean canBeInteresting() {
            // Mark cycles in the graph by dfs
            dfs(0, -1);

            // Check if all vertices belong to exactly one cycle
            for (int i = 0; i < n; i++) {
                if (color[i] == 0) {
                    return false;
                }
                if (color[i] > 1) {
                    return false;
                }
            }

            // Add edges to make each vertex belong to exactly one cycle
            for (int i = 0; i < n; i++) {
                if (cycleLength[i] == 1) {
                    for (int j : adj.get(i)) {
                        if (cycleLength[j] != 1) {
                            edgesToAdd.add(new Edge(i, j));
                        }
                    }
                }
            }

            return true;
        }

        int getAdditionalEdges() {
            return edgesToAdd.size();
        }

        List<Edge> getEdgesToAdd() {
            return edgesToAdd;
        }

        void dfs(int vertex, int parent) {
            color[vertex] = 1;
            cycleStart[vertex] = vertex;
            cycleLength[vertex] = 1;
            for (int adjacent : adj.get(vertex)) {
                if (adjacent == parent) {
                    continue;
                }
                if (color[adjacent] == 0) {
                    dfs(adjacent, vertex);
                    if (cycleStart[vertex] != cycleStart[adjacent]) {
                        edgesToAdd.add(new Edge(vertex, adjacent));
                    }
                } else if (color[adjacent] == 1) {
                    cycleStart[vertex] = cycleStart[adjacent];
                    cycleLength[vertex] = cycleLength[adjacent] + 1;
                }
            }
            color[vertex] = 2;
        }
    }

    static class Edge {
        int a;
        int b;

        Edge(int a, int b) {
            this.a = Math.min(a, b);
            this.b = Math.max(a, b);
        }
    }
}
