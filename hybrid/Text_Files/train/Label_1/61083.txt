from collections import deque

n = int(input())
s = input()
stack = deque()
ans = []
for c in s:
    if c == 'L':
        if stack:
            stack.pop()
    elif c in '()':
        stack.append(c)
        if len(stack) >= 2 and stack[-1] + stack[-2] in '()':
            stack.pop()
            stack.pop()
    else:
        stack.clear()
    ans.append(-1 if stack else len(set(stack)))
print(*ans)
