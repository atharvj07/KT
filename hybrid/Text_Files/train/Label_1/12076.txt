#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n;
    scanf("%d", &n);
    vector<int> arr(n);
    for (int i = 0; i < n; i++) scanf("%d", &arr[i]);
    long long cur = arr.back();
    long long ans = 0;
    for (int i = n - 2; i >= 0; i--) {
      if (arr[i] > cur)
        ans++;
      else
        cur = arr[i];
    }
    printf("%lld\n", ans);
  }
  return 0;
}
