#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<string> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (a[i][j] == 'C') {
        for (int k = j + 1; k < n; k++) {
          if (a[i][k] == 'C') { ans++; }
          if (a[k][j] == 'C') { ans++; }
        }
      }
    }
  }
  cout << ans << endl;
}
