import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] h = new long[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }
        long ans = 0;
        while (true) {
            boolean flag = false;
            for (int i = 1; i < n - 1; i++) {
                if (h[i] > h[i - 1] && h[i] > h[i + 1]) {
                    h[i]--;
                    flag = true;
                }
            }
            if (!flag) {
                break;
            }
            ans++;
        }
        System.out.println(ans);
    }
}
