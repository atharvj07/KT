from collections import deque

n, m = map(int, input().split())
mat = [input() for _ in range(n)]
q = deque()
dist = [[(n + 1) ** 2 for _ in range(m)] for _ in range(n)]

for i in range(n):
    for j in range(m):
        if mat[i][j] == 'G':
            q.append((i, j, 0))
            dist[i][j] = 0

while q:
    i, j, step = q.popleft()
    if mat[i][j] == 'S':
        print(step)
        exit(0)
    step += 1
    if i > 0 and dist[i - 1][j] > step and mat[i - 1][j] != '#':
        q.append((i - 1, j, step))
        dist[i - 1][j] = step
    if i < n - 1 and dist[i + 1][j] > step and mat[i + 1][j] != '#':
        q.append((i + 1, j, step))
        dist[i + 1][j] = step
    if j < m - 1 and dist[i][j + 1] > step and mat[i][j + 1] != '#':
        q.append((i, j + 1, step))
        dist[i][j + 1] = step
print(-1)
