def solve(nr, ng, nb, r, g, b):
    r.sort()
    g.sort()
    b.sort()

    def find_min(a, b, c):
        return (a-b)**2 + (b-c)**2 + (c-a)**2

    def binary_search(arr, x):
        l, r = 0, len(arr) - 1
        ans = arr[0]
        while l <= r:
            mid = (l + r) // 2
            if arr[mid] == x:
                return arr[mid]
            elif arr[mid] < x:
                l = mid + 1
            else:
                r = mid - 1
            if abs(arr[mid] - x) < abs(ans - x):
                ans = arr[mid]
        return ans

    min_diff = float('inf')
    for arrays in [[r, g, b], [r, b, g], [g, b, r], [g, r, b], [b, r, g], [b, g, r]]:
        for i in range(len(arrays[0])):
            for j in range(len(arrays[1])):
                x = arrays[0][i]
                y = arrays[1][j]
                z = binary_search(arrays[2], (x + y) / 2)
                min_diff = min(min_diff, find_min(x, y, z))

    return min_diff

t = int(input().strip())
for _ in range(t):
    nr, ng, nb = map(int, input().strip().split())
    r = list(map(int, input().strip().split()))
    g = list(map(int, input().strip().split()))
    b = list(map(int, input().strip().split()))
    print(solve(nr, ng, nb, r, g, b))
