import java.io.*;
import java.util.*;

public class Main {

    static int[] solve(int n, int m, int[] a, int[] x) {
        int[] ans = new int[m];
        int sum = 0;
        for (int j = 0; j < n; j++) {
            sum += a[j];
            if (sum >= x[0]) break;
        }
        if (sum < x[0]) {
            for (int i = 0; i < m; i++) ans[i] = -1;
        } else {
            List<Integer> swaps = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                if (sum >= x[0]) swaps.add(j);
                sum -= a[j];
                sum += a[(j + 1) % n];
            }
            Map<Integer, Integer> swapMap = new HashMap<>();
            int first = -1;
            for (int j = 0; j < swaps.size(); j++) {
                int swap = swaps.get(j);
                if (first == -1) first = swap;
                swapMap.put(swap, j - first);
            }
            for (int i = 0; i < m; i++) {
                if (x[i] > sum) ans[i] = -1;
                else {
                    int steps = swapMap.getOrDefault(0, 0);
                    while (sum < x[i]) {
                        sum += a[(steps + first) % n];
                        steps++;
                    }
                    ans[i] = steps;
                }
            }
        }
        return ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < t; i++) {
            int[] nm = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            int[] a = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            int[] x = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            int[] ans = solve(nm[0], nm[1], a, x);
            for (int j = 0; j < nm[1]; j++) {
                sb.append(ans[j]).append(" ");
            }
            sb.append("\n");
        }
        System.out.print(sb);
    }
}
