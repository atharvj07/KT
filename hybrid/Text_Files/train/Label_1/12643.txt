#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        int a[n];
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a, a + n);
        double sum = 0;
        for (int i = 0; i < n; i++) {
            sum += double(a[i]) / (i + 1);
        }
        if (sum == double(m))
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
