#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int n, d;
    cin >> n >> d;
    vector<string> s(d);
    for (int i = 0; i < d; i++) {
        cin >> s[i];
    }
    int ans = 0;
    int cur = 0;
    for (int i = 0; i < d; i++) {
        int cnt = 0;
        for (int j = 0; j < n; j++) {
            cnt += (s[i][j] == '0');
        }
        if (cnt == n) {
            cur = 0;
            continue;
        } else {
            cur++;
            ans = max(ans, cur);
        }
    }
    cout << ans << endl;
    return 0;
}
