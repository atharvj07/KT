#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[200005];
int main() {
    ll n;
    cin >> n;
    for(ll i = 1; i <= n; i++) cin >> a[i];
    vector<ll> c;
    ll sum = 0;
    for(ll i = 1; i <= n; i++) {
        sum += a[i];
    }
    for(ll i = 1; i <= n; i++) {
        if(sum - 2 * a[i] == a[i]) c.push_back(i);
    }
    cout << c.size() << '\n';
    for(ll i : c) cout << i << " ";
}
