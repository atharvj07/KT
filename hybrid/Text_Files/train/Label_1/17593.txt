#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll n, p, q, r, v;
    cin >> n >> p >> q >> r;
    vector<ll> a;
    for (int i = 0; i < n; i++) {
        cin >> v;
        a.push_back(v);
    }
    sort(a.begin(), a.end());
    ll ans = LLONG_MIN;
    for (int i = 0; i < n; i++) {
        ll l = i, r = n - 1, idx = -1;
        while (l <= r) {
            ll mid = (l + r) >> 1;
            if (a[mid] >= a[i] * q) {
                idx = mid;
                r = mid - 1;
            } else {
                l = mid + 1;
            }
        }
        if (idx != -1) {
            ans = max(ans, a[i] * p + a[idx] * q + a[n - 1] * r);
        }
    }
    cout << ans << endl;
    return 0;
}
