#include <iostream>
#include <vector>
using namespace std;
const int N = 1e5 + 1;
int n, m, a[N], b[N], c[N], ans;
vector<int> adj[N];
int dfs(int nd) {
    int mx = a[nd];
    int s = adj[nd].size();
    for (int i = 0; i < s; i++) {
        int u = adj[nd][i];
        a[nd] = max(a[u] + 1, a[nd]);
        mx = max(a[nd], mx);
    }
    for (int i = 0; i < s; i++) {
        int u = adj[nd][i];
        if (a[u] + 1 != mx) dfs(u);
    }
    return mx;
}
int main() {
    ios_base::sync_with_stdio(0);
    cin >> n >> m;
    for (int i = 1; i <= n; i++) adj[i].clear(), a[i] = 0;
    for (int i = 1; i <= m; i++) {
        int x, y;
        cin >> x >> y;
        adj[y].push_back(x);
        b[x]++, c[y]++;
    }
    for (int i = 1; i <= n; i++) {
        if (!a[i] && !b[i]) {
            ans = max(ans, dfs(i));
        }
    }
    for (int i = 1; i <= n; i++) {
        if (c[i] > 1) {
            cout << -1;
            return 0;
        }
    }
    cout << ans;
    return 0;
}
