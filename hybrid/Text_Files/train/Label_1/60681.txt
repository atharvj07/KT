n = int(input())
stones = [list(map(int, input().split())) for _ in range(n)]

# Sort each stone's dimensions to easily compare and match faces
for stone in stones:
    stone.sort()

best_volume = 0
best_stones = [1]

# Check all stones individually
for i, stone in enumerate(stones):
    min_side = min(stone)
    volume = min_side**3
    if volume > best_volume:
        best_volume = volume
        best_stones = [i+1]

# Check all pairs of stones
for i in range(n):
    for j in range(i+1, n):
        # Matching faces
        if stones[i][0] == stones[j][0] and stones[i][1] == stones[j][1]:
            glued_height = stones[i][2] + stones[j][2]
            min_side = min(stones[i][0], glued_height)
            volume = min_side**3
            if volume > best_volume:
                best_volume = volume
                best_stones = [i+1, j+1]

print(len(best_stones))
print(*best_stones)
