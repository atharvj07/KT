#include <iostream>
using namespace std;
#define ll long long int

const int p = 1e9 + 7;

ll fact(int n) {
  if (n == 0)
    return 1;
  return (n % p) * fact(n - 1) % p;
}
ll pwr(int a, int b) {
  if (b == 0)
    return 1;
  if (b == 1)
    return a;
  ll val = pwr(a, b / 2);
  val = (val * val) % p;
  if (b % 2 != 0)
    val = (val * a) % p;
  return val;
}
int main() {
  int n;
  cin >> n;
  ll a = fact(n) * fact(n) % p;
  ll b = pwr(2, n);
  b = (b - 1 + p) % p;
  a = (a * b) % p;
  cout << a << endl;
  return 0;
}
