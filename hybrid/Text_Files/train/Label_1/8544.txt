#include <iostream>
#include <deque>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int p[n];
        for (int i = 0; i < n; i++) {
            cin >> p[i];
        }
        deque<int> dq;
        for (int i = 0; i < n; i++) {
            if (dq.empty() || dq.front() > p[i]) {
                dq.push_front(p[i]);
            } else {
                dq.push_back(p[i]);
            }
        }
        for (auto& e : dq) {
            cout << e << ' ';
        }
        cout << endl;
    }
}
