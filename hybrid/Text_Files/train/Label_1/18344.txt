#include <bits/stdc++.h>
using namespace std;

int n, v[105], vis[105], p[105];
vector<pair<int, int>> ans;

void solve(int l, int r) {
  if (l >= r) return;
  int mn = l;
  for (int i = l + 1; i <= r; i++)
    if (v[i] < v[mn])
      mn = i;
  if (mn == l) {
    solve(l + 1, r);
    return;
  }
  p[mn] = l;
  for (int i = l; i < mn; i++) p[i + 1] = i;
  ans.push_back(make_pair(l, mn));
  solve(l + 1, mn);
  solve(mn + 1, r);
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> v[i];
    vis[i] = false;
  }
  solve(1, n);
  for (auto x : ans) { cout << x.first << ' ' << x.second << '\n'; }
  return 0;
}
