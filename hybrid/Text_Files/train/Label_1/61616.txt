for t in range(int(input())):
    n=int(input())
    l=[int(x) for x in input().split()]
    b=[0]*2*n
    for x in l: b[x-1]=1
    if 0 in b[n:]:
        print(-1)
        continue
    ans=[]
    for i in range(n):
        mn=1e9
        for j in range(2*n):
            if not b[j] and j+1<mn:
                mn=j+1
        ans.append(mn)
        b[mn-1]=1
        for j in range(n):
            if b[j] and l[j]==mn:
                ans.append(j+1)
                b[j]=0
                break
    print(*ans)
