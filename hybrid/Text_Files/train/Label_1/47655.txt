import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt(), d = in.nextInt();
        if (k < d)
            System.out.println(-1);
        else {
            for (int i = 0; i < d; i++) {
                for (int j = 0; j < n; j++)
                    System.out.print(((j / k) + 1 + i) % k + 1 + " ");
                System.out.println();
            }
        }
    }
}
