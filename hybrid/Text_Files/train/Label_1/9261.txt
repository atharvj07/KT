#include <bits/stdc++.h>

using namespace std;

const int mod = 1e9 + 7;

int main() {
    int n;
    cin >> n;
    map<int, int> m;
    for (int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        m[a] = b;
    }
    long long ans = 1;
    for (int i = 1; i <= 2 * n; i++) {
        if (m[i] != 0 && i != m[i]) {
            ans *= 2;
            ans %= mod;
        }
    }
    cout << ans << endl;
    return 0;
}
