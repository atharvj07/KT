from math import sqrt

def distance(x1, y1, x2, y2):
    return sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)

def solve(ax, ay, bx, by, tx, ty, bottles):
    base_distance = 0
    for x, y in bottles:
        base_distance += 2 * distance(x, y, tx, ty)
    
    best_savings = float('-inf')
    for first_pick in range(2):
        for i, (x, y) in enumerate(bottles):
            pick_distance = distance(ax if first_pick == 0 else bx, ay if first_pick == 0 else by, x, y) 
            return_distance = distance(x, y, tx, ty)
            savings = return_distance - pick_distance 
            if savings > best_savings:
                best_savings = savings
                first_bottle = i
    
    if best_savings < 0: 
        best_savings = 0

    best_extra_savings = float('-inf')
    for second_pick in range(2):
        for i, (x, y) in enumerate(bottles):
            if i == first_bottle: 
                continue
            pick_distance = distance(bx if second_pick == 0 else ax, by if second_pick == 0 else ay, x, y)
            return_distance = distance(x, y, tx, ty)
            extra_savings = return_distance - pick_distance
            if extra_savings > best_extra_savings:
                best_extra_savings = extra_savings
    
    if best_extra_savings < 0:
        best_extra_savings = 0

    return base_distance - best_savings - best_extra_savings

ax, ay, bx, by, tx, ty = map(int, input().split())
n = int(input())
bottles = [tuple(map(int, input().split())) for _ in range(n)]

print(solve(ax, ay, bx, by, tx, ty, bottles))
