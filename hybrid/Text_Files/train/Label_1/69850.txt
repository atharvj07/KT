import sys
input = sys.stdin.readline

for t in range(int(input())):
    n, l, r = map(int, input().split())
    lst = sorted(map(int, input().split()))
    rcol = [l]
    ans = 0
    for i in range(l, n):
        if lst[i] in lst[:i]:
            rcol.append(i)
        ans += 1
    ans += min(l, n - r - 1)
    print(ans)
