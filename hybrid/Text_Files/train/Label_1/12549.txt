#include <bits/stdc++.h>

#define ll long long

using namespace std;

const int MAXN = 2e5 + 10;

int n, k;
ll dp[MAXN][2][2];
ll a[MAXN], b[MAXN], t[MAXN];

ll solve(int ind, bool aa, bool bb) {
  if (ind == n) {
    if (aa and bb) return 0;
    return LLONG_MAX;
  }
  ll &ret = dp[ind][aa][bb];
  if (ret != -1) return ret;

  ret = LLONG_MAX;
  ret = min(ret, solve(ind + 1, aa | a[ind], bb | b[ind]) + t[ind]);
  ret = min(ret, solve(ind + 1, aa, bb));
  return ret;
}

int main() {
  memset(dp, -1, sizeof(dp));
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    cin >> t[i] >> a[i] >> b[i];
  }

  ll ans = solve(0, 0, 0);
  if (ans == LLONG_MAX) {
    cout << -1 << endl;
  } else
    cout << ans << endl;
}
