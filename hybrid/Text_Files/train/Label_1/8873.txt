#include<bits/stdc++.h>
#define endl '\n'
#define ll long long 
using namespace std;
typedef pair<int, int> pi;
 
vector<vector<int>> adj;
vector<bool> vis;
vector<int> comp;
vector<int> value;
vector<int> vec;
vector<int> appear;
int k, p;
 
void dfs(int node){
    vis[node] = 1;
    comp[node] = k;
    value[k] += appear[node];
    for(int next : adj[node])
        if(!vis[next])dfs(next);
}
 
int main(){
    int n; cin >> n;
    adj.resize(n + 1);
    vis.resize(n + 1);
    comp.resize(n + 1);
    value.resize(n + 1);
    vec.resize(n + 1);
    appear.resize(n + 1);
    for(int i = 1; i <= n; i++){
        int a; cin >> a;
        vec[i] = a;
        appear[a]++;
    }
    for(int i = 1; i <= n; i++){
        int a; cin >> a;
        adj[vec[i]].push_back(a);
        adj[a].push_back(vec[i]);
    }
    k = 0;
    for(int i = 1; i <= n; i++){
        if(!vis[i]){
            k++;
            dfs(i);
        }
    }
    int maxi = 0;
    for(int i = 0; i <= k; i++)maxi = max(maxi, value[i]);
    int ans = 0, winner = 1;
    for(int i = 0; i <= k; i++){
        if(value[i] == maxi){
            if(p)winner = 2;
            p++;      
        }
        if(value[i] == 0)ans = -1;
    }
    if(ans == -1)cout << ans << endl;
    else{
        ans = n;
        ans /= maxi;
        cout << ans << ' ' << winner << endl;
    }
    
}
