import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    a = list(map(int, input().split()))
    b = sorted(a)
    ans = 0
    for i in range(n):
        if a[i] != b[i]:
            ans += 1
            if a[i] == b[n - 1]:
                j = n - 1
                while a[j] != b[i]:
                    j -= 1
                a[i], a[j] = a[j], a[i]
            else:
                j = i + 1
                while a[j] != b[i]:
                    j += 1
                a[i], a[j] = a[j], a[i]
    return ans

t = int(input())
for _ in range(t):
    print(solve())
