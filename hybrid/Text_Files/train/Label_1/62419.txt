import sys
s = sys.stdin.readline().strip()
k = int(sys.stdin.readline().strip())
n = len(s)
if n % k != 0:
    print("NO")
else:
    s += s[0]
    f = [True] * (n+1)
    for i in range(1,n+1):
        l = i
        r = i + n // k - 1
        while l < r:
            if s[l] != s[r]:
                f[i] = False
            l += 1
            r -= 1
    ok = True
    for i in range(1, k+1):
        if f[n // k * i] == False:
            ok = False
    if ok: print("YES")
    else: print("NO")
