#include <cstdio>
int a[200'005];
int main() {
  int t, n, k;
  scanf("%d", &t);
  while (t--) {
    scanf("%d%d", &n, &k);
    for (int i = 1; i <= n; i++)
      scanf("%d", a + i);
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      if (k - a[i] > 0) {
        ans++;
        k -= a[i];
      }
      if (k <= 0) break;
    }
    if (k > 0) ans += (k + n - 1) / n;
    printf("%d\n", ans);
  }
}
