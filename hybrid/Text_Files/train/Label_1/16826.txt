#include <bits/stdc++.h>
#define ll long long
using namespace std;

ll n, d, m;
ll a[100005], b[100005], c[100005];

int main() {
  cin >> n >> d >> m;
  for (ll i = 0; i < n; i++) cin >> a[i];
  for (ll i = 0; i < d; i++) b[i] = -1;
  for (ll i = n - 1; i >= 0; i--) {
    if (i + d < n) b[i + d] = max(b[i + d], b[i]);
    c[i] = b[i + d] + 1;
    if (a[i] > m) b[i] = c[i] + 1;
  }
  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    if (c[i] == 0) ans += a[i];
  }
  cout << ans << endl;
  return 0;
}
