#include<bits/stdc++.h>
using namespace std;
long long int min(long long int a,long long int b){
    if(a<b)
        return a;
    return b;
}

long long int minPerimeter(long long int a,long long int b){
    return 2*(min(a,b)+min(a,a+b)+min(b,a+b));
}

int main(){
    long long int a,b;
    cin>>a>>b;
    cout<<minPerimeter(a,b)<<endl;
    return 0;
}
