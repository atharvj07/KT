from collections import defaultdict
s = input()
d = defaultdict(int)
d[0] = 1
l = 0
r = 0
ans = []
pref = [0] * len(s)
for r in range(len(s)):
    pref[r] = (pref[r - 1] * 26 + ord(s[r]) - ord('A')) % 1000000007
for l in range(len(s)):
    cur = (pref[r] - pref[l - 1] * pow(26, r - l, 1000000007)) % 1000000007
    ans.append((r - l + 1, d[cur]))
    d[cur] += 1
    r += 1
ans.sort()
print(len(ans))
for x in ans:
    print(*x)
