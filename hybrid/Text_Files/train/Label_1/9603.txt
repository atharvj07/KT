#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const ll mod = 998244353;

ll quickpow(ll a, ll n) {
    ll ans = 1;
    while (n) {
        if (n % 2) ans = ans * a % mod;
        a = a * a % mod;
        n /= 2;
    }
    return ans;
}

ll solve(vector<int> &a) {
    int n = a.size();
    vector<int> b(n);
    bool flag = false;
    for (int i = 0; i < n; i++) {
        if (a[i] > 1) flag = true;
        b[i] = a[i] - 1;
    }
    ll ans = 1;
    for (int i = 0; i < n; i++) {
        if (b[i] == 0) continue;
        ans = ans * b[i] % mod;
    }
    if (flag) ans = ans * 2 % mod;
    return ans;
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    cout << solve(a) << endl;
    return 0;
}
