from collections import defaultdict, deque

def solve(n, m, edges):
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    for u, v in edges:
        graph[u].append(v)
        indegree[v] += 1

    # BFS to find all vertices reachable from 1
    reachable = [False] * (n + 1)
    queue = deque([1])
    while queue:
        node = queue.popleft()
        if not reachable[node]:
            reachable[node] = True
            for neigh in graph[node]:
                queue.append(neigh)

    # Detect cycles
    in_cycle = [False] * (n + 1)
    visited = [False] * (n + 1)
    def dfs(node, path):
        if visited[node]:
            if node in path:
                cycle_start = path.index(node)
                for i in range(cycle_start, len(path)):
                    in_cycle[path[i]] = True
            return
        visited[node] = True
        path.append(node)
        for neigh in graph[node]:
            dfs(neigh, path)
        path.pop()

    for i in range(1, n + 1):
        if not visited[i]:
            dfs(i, [])

    # Determine path counts
    path_counts = [0] * (n + 1)  # 0: no path, 1: exactly one path, 2: more than one but finite, -1: infinite
    path_counts[1] = 1  # From 1 to 1 there's always exactly one path (empty path)
    queue = deque([1])
    while queue:
        node = queue.popleft()
        for neigh in graph[node]:
            if reachable[neigh]:
                if in_cycle[neigh] or in_cycle[node] or path_counts[node] > 1:
                    path_counts[neigh] = -1
                elif path_counts[neigh] == 0:
                    path_counts[neigh] = path_counts[node]
                    queue.append(neigh)
                elif path_counts[neigh] == 1:
                    path_counts[neigh] = 2

    return ' '.join(map(str, path_counts[1:]))

t = int(input().strip())
results = []
for _ in range(t):
    input()  # empty line
    n, m = map(int, input().split())
    edges = [tuple(map(int, input().split())) for _ in range(m)]
    results.append(solve(n, m, edges))

for result in results:
    print(result)
