import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        if (k > n || k == 1) {
            System.out.println(-1);
            return;
        }
        int x = a[k - 1] - 1;
        int y = a[0];
        System.out.println(x + " " + y);
    }
}
