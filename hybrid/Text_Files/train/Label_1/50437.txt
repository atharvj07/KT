import java.util.Scanner;

public class PiApproximation {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            double allowedError = scanner.nextDouble();
            if (allowedError == 0.0) {
                break;
            }

            Fraction fraction = approximatePi(allowedError);
            System.out.println(fraction.numerator + "/" + fraction.denominator);
        }

        scanner.close();
    }

    private static Fraction approximatePi(double allowedError) {
        int numerator = 3;
        int denominator = 1;

        while (Math.abs(Math.PI - ((double) numerator / denominator)) > allowedError) {
            if (Math.PI - ((double) numerator / denominator) > 0) {
                numerator += 2;
                denominator *= 2;
            } else {
                numerator -= 1;
                denominator -= 1;
            }
        }

        return new Fraction(numerator, denominator);
    }

    private static class Fraction {
        int numerator;
        int denominator;

        public Fraction(int numerator, int denominator) {
            this.numerator = numerator;
            this.denominator = denominator;
        }
    }
}
