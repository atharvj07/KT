#include <iostream>
#include <vector>
#include <queue>

using namespace std;

const int inf = 1e9;
vector<int> g[1001];
vector<int> val(1001, 0);

int prim(int s, int n) {
    vector<int> key(n + 1, inf);
    vector<int> parent(n + 1, -1);
    vector<bool> vis(n + 1, false);

    key[s] = 0;
    for (int i = 1; i <= n; i++) {
        int u = -1;
        for (int j = 1; j <= n; j++) {
            if (!vis[j] && (u == -1 || key[j] < key[u])) {
                u = j;
            }
        }

        vis[u] = true;
        for (int j = 0; j < g[u].size(); j++) {
            int v = g[u][j];
            if (!vis[v] && key[u] + val[v] < key[v]) {
                key[v] = key[u] + val[v];
                parent[v] = u;
            }
        }
    }

    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (parent[i] != -1) {
            ans += val[i];
        }
    }
    return ans;
}

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> val[i];
    }
    for (int i = 0; i < m; i++) {
        int x, y;
        cin >> x >> y;
        g[x].push_back(y);
        g[y].push_back(x);
    }
    cout << prim(1, n) << '\n';
    return 0;
}
