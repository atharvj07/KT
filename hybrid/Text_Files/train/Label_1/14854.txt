#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1005;
int a[MAXN];
int pre[MAXN];
int prv[MAXN];

int main() {
    int n;
    cin >> n;

    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    int ans = 0;
    pre[0] = 0;
    for (int i = 0; i < n; i++) {
        int sum = 0;
        for (int j = i + 1; j <= n; j++) {
            sum += a[j];
            if (sum == a[i]) {
                pre[j] = i;
                break;
            }
        }
    }

    prv[n + 1] = n + 1;
    for (int i = n; i >= 0; i--) {
        int sum = 0;
        for (int j = i - 1; j >= 0; j--) {
            sum += a[j];
            if (sum == a[i]) {
                prv[j] = i;
                break;
            }
        }
    }

    for (int i = 1; i <= n; i++) {
        int l = pre[i];
        int r = prv[i];
        if (l == 0 || r == n + 1) {
            continue;
        }
        ans = max(ans, r - l);
    }

    cout << ans << endl;

    return 0;
}
