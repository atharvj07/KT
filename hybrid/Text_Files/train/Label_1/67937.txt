def solve(a, s):
    sa, ss = str(a), str(s)
    n, m = len(sa), len(ss)
    sb = sa
    if n == m:
        for i in range(n - 1, -1, -1):
            if int(sb[i]) + int(ss[i]) >= 10:
                if i - 1 >= 0: sb = sb[:i - 1] + str(int(sb[i - 1]) + 1) + sb[i:]
                else: sb = '1' + sb
            sb = sb[:i] + str(int(sb[i]) + int(ss[i]) % 10) + sb[i + 1:]
    else:
        sb = '0' * abs(n - m) + sa
        for i in range(n - 1, -1, -1):
            if int(sb[i]) + int(ss[i]) >= 10:
                if i - 1 >= 0: sb = sb[:i - 1] + str(int(sb[i - 1]) + 1) + sb[i:]
                else: sb = '1' + sb
            sb = sb[:i] + str(int(sb[i]) + int(ss[i]) % 10) + sb[i + 1:]
    if sb[0] == '0': sb = sb[1:]
    return sb if sb == ss else -1

for _ in range(int(input())): a, s = [int(_) for _ in input().split()] ; print(solve(a, s))
