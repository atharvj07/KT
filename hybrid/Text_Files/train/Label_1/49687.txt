import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]), m = Integer.parseInt(line[1]);
        List<int[]> ranges = new ArrayList<>();
        int[] deleted = new int[m];
        Arrays.fill(deleted, -1);
        int skipped = 0;
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            ranges.add(new int[]{Integer.parseInt(line[0]), Integer.parseInt(line[1])});
        }
        for (int i = 0; i < m; i++) {
            int[] range = ranges.get(i);
            int start = range[0], end = range[0] + range[1] - 1;
            boolean canSkip = true;
            for (int j = start; j <= end; j++) {
                if (j < start || j > end) continue;
                for (int k = 0; k < m; k++) {
                    if (k == i) continue;
                    int[] otherRange = ranges.get(k);
                    if (deleted[k] == 1) continue;
                    if (otherRange[0] <= j && otherRange[0] + otherRange[1] - 1 >= j) {
                        canSkip = false;
                        break;
                    }
                }
                if (!canSkip) break;
            }
            if (canSkip) {
                skipped++;
                deleted[i] = 1;
            }
        }
        System.out.println(skipped);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < m; i++) {
            if (deleted[i] == 1) {
                sb.append(i + 1).append(" ");
            }
        }
        System.out.println(sb.toString().trim());
    }
}
