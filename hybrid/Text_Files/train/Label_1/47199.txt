import java.util.*;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
public class Main {
    public static void main(String [] args){
        Scanner sc= new Scanner(System.in);
        int n= sc.nextInt();
        int q= sc.nextInt();
        int []arr=new int[n+1];
        int unread=0;
        for(int i=0;i<q;i++){
            int type=sc.nextInt();
            if(type==1){
                int app=sc.nextInt();
                arr[app]++;
                unread++;
            }
            else if(type==2){
                int app=sc.nextInt();
                unread-=arr[app];
                arr[app]=0;
            }
            else{
                int k=sc.nextInt();
                for(int j=1;j<=n && k>0;j++){
                    if(arr[j]>0){
                        k--;
                        arr[j]--;
                        unread--;
                    }
                }
            }
            System.out.println(unread);
        }
    }
}
