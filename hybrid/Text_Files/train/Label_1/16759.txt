# include <bits/stdc++.h>
using namespace std;
const int maxn = 100000 + 10;
char s[maxn];
int cnt[maxn];
int main() {
  int n, k;
  cin >> n >> k;
  cin >> s;
  for (int i = 0; i < n; ++i) cnt[s[i] - 'A'] ++;
  sort(cnt, cnt + maxn, greater<int>());
  int ans = 0;
  for (int i = 0; i < k; ++i) {
    if (!cnt[i]) break;
    ans += (cnt[i] - i - 1) * cnt[i] / 2 + cnt[i];
  }
  cout << ans << endl;
  return 0;
}
