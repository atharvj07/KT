import sys

n, k, A, B = map(int, input().split())
a = [0] * (2 ** n + 1)
for i in map(int, input().split()):
    a[i] = 1
dp = [10 ** 9] * (2 ** n + 1)
dp[0] = 0

for i in range(1, 2 ** n + 1):
    dp[i] = dp[i - 1] + A if not a[i] else dp[i - 1] + B * a[i]
    if i % 2 == 0:
        dp[i] = min(dp[i], dp[i // 2] + dp[i - i // 2])
print(dp[2 ** n])
