#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s1, s2;
    cin >> s1 >> s2;
    int ans = 0, cnt0 = 0, cnt1 = 0;
    for (int i = 0; i < n; ++i) {
      if (s1[i] == '0') cnt0++;
      if (s2[i] == '1') cnt1++;
      if (cnt0 > 0 && cnt1 > 0) {
        ans += 2;
        cnt0 = cnt1 = 0;
      }
    }
    if (cnt0 > 0) ans += 1;
    if (cnt1 > 0) ans += 1;
    cout << ans << endl;
  }
}
