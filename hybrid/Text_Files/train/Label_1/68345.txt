import sys
input = sys.stdin.readline

n, x, y = map(int, input().split())
g = [[] for _ in range(n + 1)]
dis, sub = [0] * (n + 1), [0] * (n + 1)
vis = [0] * (n + 1)

def dfs1(v, p):
    sub[v] = 1
    for u in g[v]:
        if u == p or vis[u]: continue
        dfs1(u, v)
        sub[v] += sub[u]

def dfs2(v, p):
    vis[v] = 1
    for u in g[v]:
        if u == p or vis[u]: continue
        dis[u] = dis[v] + sub[v] - sub[u]
        dfs2(u, v)

for _ in range(n - 1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

dfs1(1, 1)
dis[1] = n
dfs2(1, 1)

ans = n * (n - 1) // 2
for v in range(1, n + 1):
    if v in [x, y]: continue
    ans -= sub[v] * (n - sub[v])
    ans -= dis[v]
print(ans)
