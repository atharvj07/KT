import java.util.*;

public class Main {
    static int n, k;
    static List<Integer>[] adj;
    static int[] dist;
    static boolean[] vis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        adj = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 1; i < n; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj[u].add(v);
            adj[v].add(u);
        }
        dist = new int[n + 1];
        vis = new boolean[n + 1];
        dfs(1, -1, 0);
        int maxDist = 0;
        int maxDistIdx = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] > maxDist) {
                maxDist = dist[i];
                maxDistIdx = i;
            }
        }
        dfs(maxDistIdx, -1, 0);
        maxDist = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] > maxDist) {
                maxDist = dist[i];
                maxDistIdx = i;
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] > k) {
                ans++;
            }
        }
        System.out.println(ans);
    }

    static void dfs(int node, int parent, int depth) {
        vis[node] = true;
        dist[node] = depth;
        for (int next : adj[node]) {
            if (next != parent && !vis[next]) {
                dfs(next, node, depth + 1);
            }
        }
    }
}
