import bisect

n, m = map(int, input().split())
a = list(map(int, input().split()))

if sum(ai % 2 for ai in a) == n // 2:
    for i in range(n):
        print(a[i], end=' ')
    exit(0)

a.sort()
ans = []
unevens = []

for ai in a:
    if ai % 2 == 1:
        unevens.append(ai)
    else:
        ans.append(ai)

if n // 2 - len(unevens) > m:
    print(-1)
else:
    for u in unevens[:n // 2 - len(ans)]:
        idx = bisect.bisect_left(ans, u)
        if idx < len(ans):
            ans[idx], u = u, ans[idx]
        unevens.remove(u)
    print(n // 2 - len(ans))
    print(*ans + unevens)
