import sys
n=int(input())
a=list(map(int, input().split()))
a.sort()
b=[]
c=[]
for i in a:
    if i not in b:
        b.append(i)
        c.append(1)
    else: c[-1]+=1
p=c.index(max(c))
q=0
for i in range(len(b)):
    if c[i]==c[p]: q+=1
print(c[p]*q)
print(q, c[p])
for i in range(q):
    for j in range(c[p]):
        print(b[p], end=' ')
    print()
