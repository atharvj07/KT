#include <bits/stdc++.h>
using namespace std;

const int N = 105;
int n, m, mx[N], cnt[N];
int ans, ans_cnt;

int main() {
    cin >> n >> m;
    memset(mx, -1, sizeof mx);
    while (m--) {
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            if (x > mx[i]) {
                mx[i] = x;
                cnt[i] = 1;
            } else if (mx[i] == x) {
                cnt[i]++;
            }
        }
    }
    for (int i = 0; i < n; i++) {
        if (cnt[i] > ans_cnt) {
            ans = i + 1;
            ans_cnt = cnt[i];
        } else if (cnt[i] == ans_cnt) {
            ans = min(ans, i + 1);
        }
    }
    cout << ans << endl;
    return 0;
}
