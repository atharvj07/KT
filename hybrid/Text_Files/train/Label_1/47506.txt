import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextLong();
        }
        Arrays.sort(a);
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += a[i] * i;
        }
        if (a[n - 1] > 0) {
            ans -= a[n - 1] * (n - 1);
            ans += a[n - 1] * n;
        }
        System.out.println(ans);
    }
}
