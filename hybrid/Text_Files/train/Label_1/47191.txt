import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int[] honey = new int[n];
        for (int i = 0; i < n; i++) {
            honey[i] = in.nextInt();
        }
        int eaten = 0;
        int givenToPiglet = 0;
        Arrays.sort(honey);
        for (int i = n - 1; i >= 0; i--) {
            if (honey[i] < k || eaten == 3) {
                givenToPiglet += honey[i];
            } else {
                eaten = 0;
                honey[i] -= k;
                eaten++;
            }
        }
        System.out.println(givenToPiglet);
    }
}
