import java.util.Arrays;
import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), d = sc.nextInt();
        int[] F = new int[n], t = new int[n];
        for (int i = 0; i < n; i++) {
            F[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        Arrays.sort(F);
        Arrays.sort(t);
        PriorityQueue<Integer> q = new PriorityQueue<>();
        int m = 0, r = 0, ans = 0;
        for (int i = 0, j = 0; i < n || !q.isEmpty(); ) {
            if (q.isEmpty()) {
                m = j;
            }
            if (i < n && t[i] < m) {
                i++;
                continue;
            }
            int f = q.isEmpty() ? 1 : q.poll();
            if (f < F[j]) {
                ans = -1;
                break;
            }
            if (j == n) {
                r += d - q.size();
                break;
            }
            ans += F[j] - f;
            if (q.size() < d) {
                q.add(F[j]);
            } else {
                r += F[j] - f;
            }
            j++;
        }
        if (r == 0) {
            System.out.println(ans);
        } else {
            System.out.println(ans + r + d);
        }
    }
}
