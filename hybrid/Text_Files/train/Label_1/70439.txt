def max_operations(t, cases):
    results = []
    for n, s in cases:
        zero_group, one_group = 0, 0
        prev = s[0]
        if prev == '0':
            zero_group += 1
        else:
            one_group += 1
            
        for i in range(1, n):
            if s[i] != prev:
                if s[i] == '0':
                    zero_group += 1
                else:
                    one_group += 1
                prev = s[i]
                
        results.append(min(zero_group, one_group) + 1 if zero_group > 0 and one_group > 0 else 1)
    return results

t = int(input().strip())
cases = [(int(input().strip()), input().strip()) for _ in range(t)]
results = max_operations(t, cases)
for result in results:
    print(result)
