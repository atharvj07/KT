import java.util.Scanner;

public class CompetitionEarnings {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int codingContestPlace = scanner.nextInt();
        int robotManeuverPlace = scanner.nextInt();

        int codingContestEarnings = getEarnings(codingContestPlace);
        int robotManeuverEarnings = getEarnings(robotManeuverPlace);
        int bonus = (codingContestPlace == 1 && robotManeuverPlace == 1) ? 400000 : 0;

        int totalEarnings = codingContestEarnings + robotManeuverEarnings + bonus;

        System.out.println(totalEarnings);
    }

    private static int getEarnings(int place) {
        switch (place) {
            case 1:
                return 300000;
            case 2:
                return 200000;
            case 3:
                return 100000;
            default:
                return 0;
        }
    }
}
