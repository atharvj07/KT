#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> grid(n + 2, vector<int>(m + 2, 0));
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= m; ++j) cin >> grid[i][j];
  int ans = n * m;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      if (grid[i][j]) {
        for (int x1 = 1; x1 <= n; ++x1) {
          for (int y1 = 1; y1 <= m; ++y1) {
            if (!grid[x1][y1] && i != x1 && j != y1) {
              int cur = 0;
              for (int x2 = 1; x2 <= n; ++x2) {
                for (int y2 = 1; y2 <= m; ++y2) {
                  if (min(i, x1) <= x2 && x2 <= max(i, x1) &&
                      min(j, y1) <= y2 && y2 <= max(j, y1))
                    ++cur;
                }
              }
              ans = min(ans, cur);
            }
          }
        }
      }
    }
  }
  cout << ans << "\n";
  return 0;
}
