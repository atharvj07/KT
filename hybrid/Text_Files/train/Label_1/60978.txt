def solve(n, a):
    adj = [[]] * n
    b = a[:]
    b.sort()
    for i in range(n):
        for j in range(i + 1, n):
            if a[i] > a[j]:
                adj[i].append(j)
            else:
                adj[j].append(i)
    dp = [-1] * n
    vis = [False] * n
    ans = []

    def dfs(u):
        if vis[u]:
            return True
        vis[u] = True
        for v in adj[u]:
            if dp[v] == -1 or dfs(dp[v]):
                dp[v] = u
                ans.append((u + 1, v + 1))
                return True
        return False

    for i in range(n):
        if dfs(i):
            break
    if vis[0] == False:
        return -1
    return len(ans), ans

n = int(input())
a = list(map(int, input().split()))
ans = solve(n, a)
print(ans[0])
for pair in ans[1]:
    print(*pair)
