for _ in range(int(input())):
    n = int(input())
    print(n * (n - 1) // 2)
    print(' '.join([str(i + 1) for i in range(n)]))
    print(n - 1)
    a = list(range(1, n + 1))
    for j in range(2, n + 1):
        for i in range(1, n + 1):
            if not check(a):
                if a[i - 1] == j:
                    print(i, i - 1)
                    a[i - 1], a[i] = a[i], a[i - 1]
                    break
