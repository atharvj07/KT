import java.util.Scanner;

public class Restaurant {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();

        int denied = 0;
        int oneSeaterOccupied = 0;
        int twoSeaterOccupied = 0;

        for (int i = 0; i < n; i++) {
            int groupSize = scanner.nextInt();

            if (groupSize == 1) {
                if (oneSeaterOccupied < a) {
                    oneSeaterOccupied++;
                } else if (twoSeaterOccupied < b) {
                    twoSeaterOccupied++;
                } else {
                    denied++;
                }
            } else {
                if (twoSeaterOccupied < b) {
                    twoSeaterOccupied++;
                } else {
                    denied += 2;
                }
            }
        }

        System.out.println(denied);
    }
}
