def expected_inversions(n, k):
    # Calculate the total number of inversions in the initial permutation
    # Since the permutation is given to be 1 to n, it initially has 0 inversions.
    
    # The probability that a single operation does not change the number of inversions
    prob_no_change = 1/2
    # The probability that a single operation reverses a section, which on average has half of the possible inversions reversed
    prob_half_reversed = 1/2
    
    # Calculate the expected value after k operations
    expected_value = n * (n - 1) / 4 * (1 - (prob_no_change + prob_half_reversed * (-1)) ** k)
    
    return expected_value

# Processing input
n, k = map(int, input().split())
p = list(map(int, input().split()))

# Calculate and print the expected number of inversions
print(expected_inversions(n, k))
