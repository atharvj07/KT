#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 10;

int a[N], dp[N];

int main() {
  int n, q;
  cin >> n >> q;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = n - 1; i >= 1; i--) {
    dp[i] = dp[i + 1];
    if (a[i] & a[i + 1]) dp[i] = i + 1;
  }
  while (q--) {
    int l, r;
    cin >> l >> r;
    puts(dp[l] >= r ? "Shi" : "Fou");
  }
  return 0;
}
