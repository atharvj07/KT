#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (auto &item : a) cin >> item;
    long long sum = 0;
    for (auto &item : a) sum += (1ll << item);
    long long ans = 0;
    long long target = 2;
    while (sum < target - 1) {
        sum += target;
        ans++;
        target *= 2;
    }
    cout << ans << endl;
    return 0;
}
