#include <iostream>
#include <cmath>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t, n, x;
    cin >> t;
    while (t--) {
        cin >> n >> x;
        vector<int> favs(n);
        for (int i = 0; i < n; i++) cin >> favs[i];
        sort(favs.begin(), favs.end());
        int ans = (x - 1) / favs[0] + 1;
        for (int i = 1; i < n; i++) {
            double need = (double)x / favs[i];
            ans = min(ans, (int)ceil(need) + 1);
        }
        cout << ans << endl;
    }
    return 0;
}
