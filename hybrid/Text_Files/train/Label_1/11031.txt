#include <iostream>
using namespace std;

int main() {
  int n, r, Ans=0;
  bool ch[1001];
  cin >> n >> r;
  for(int i=1; i<=n; i++) {
    int x;
    cin >> x;
    if(x==0) ch[i]=1;
  }
  for(int i=1; i<=n;) {
    int j;
    if(ch[i]) {
      i++;
      continue;
    }
    for(j=i; j<=n; j++) if(ch[j]==0) break;
    if(j==n+1) {
      Ans=-1;
      break;
    }
    Ans++;
    int l=max(i, j-r+1);
    for(int x=l; x<=j; x++) ch[x]=1;
    i=j+1;
  }
  cout << Ans << endl;
  return 0;
}
