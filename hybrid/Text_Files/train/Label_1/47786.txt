import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        char[] hamsters = scanner.next().toCharArray();
        int standing = 0;
        int sitting = 0;
        for (char hamster : hamsters) {
            if (hamster == 'X') {
                standing++;
            } else {
                sitting++;
            }
        }
        int minutes = 0;
        while (standing != n / 2 || sitting != n / 2) {
            minutes++;
            if (standing > n / 2) {
                standing--;
                sitting++;
            } else if (standing < n / 2) {
                standing++;
                sitting--;
            }
        }
        System.out.println(minutes);
        for (int i = 0; i < n; i++) {
            if (i < (n / 2)) {
                hamsters[i] = 'X';
            } else {
                hamsters[i] = 'x';
            }
        }
        System.out.println(hamsters);
    }
}
