import sys

n, l, r = list(map(int, input().split()))
a = list(map(int, input().split()))
p = list(map(int, input().split()))

b = list(range(1, n + 1))

for i in range(n):
    b[i] += (p[i] + a[i] - i - 1) % (r - l + 1)
    b[i] = l + b[i] % (r - l + 1)

if len(set(b)) == n:
    print(*b)
else:
    print(-1)
