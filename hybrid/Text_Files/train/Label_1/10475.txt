#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, d, ansa = 0, ansb = 0;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    cin >> m;
    vector<int> b(m);
    for (int i = 0; i < m; i++) cin >> b[i];
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    for (d = 0;; d++) {
        int sa = 0, sb = 0;
        for (int i = 0; i < n; i++) if (a[i] <= d) sa += 2; else sa += 3;
        for (int i = 0; i < m; i++) if (b[i] <= d) sb += 2; else sb += 3;
        if (sa - sb > ansa - ansb) ansa = sa, ansb = sb;
        else break;
    }
    cout << ansa << ":" << ansb << endl;
    return 0;
}
