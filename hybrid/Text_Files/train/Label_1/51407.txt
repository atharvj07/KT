import java.util.Arrays;
import java.util.Scanner;

public class Main {

    private static int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner stdin = new Scanner(System.in);
        int n = stdin.nextInt();
        int k = stdin.nextInt();
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            w[i] = stdin.nextInt();
        }

        int[][] dp = new int[n + 1][k + 1];
        int[][] sol = new int[n + 1][k + 1];
        Arrays.fill(dp[n], -1);
        dp[n][0] = 0;
        sol[n][0] = 1;

        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = dp[i + 1][j];
                sol[i][j] = sol[i + 1][j];
                if (j >= w[i]) {
                    int newDp = dp[i + 1][j - w[i]];
                    if (newDp != -1) {
                        if (dp[i][j] == -1) {
                            dp[i][j] = newDp + 1;
                            sol[i][j] = sol[i + 1][j - w[i]];
                        } else if (dp[i][j] == newDp + 1) {
                            sol[i][j] = (sol[i][j] + sol[i + 1][j - w[i]]) % MOD;
                        }
                    }
                }
            }
        }

        System.out.println(dp[0][k]);
        System.out.println(sol[0][k]);
    }
}
