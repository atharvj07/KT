#include<iostream>
using namespace std;
bool can[305][305],can1[305];
int32_t main(){
	int n,a[305]={0};
	cin>>n;
	for(int i=0,x;i<n;++i){
		cin>>x,a[x]++;
	}
	for(int i=1;i<=300;++i)
		for(int j=1;j<=300;++j){
			if(i==j){
				can1[i]=true;
				continue;
			}
			if(a[i]==0||a[j]==0)continue;
			a[i]--,a[j]--;
			can[i][j]|=can1[j];
			a[i]++,a[j]++;
		}
	for(int i=300;i>=1;--i)
		for(int j=i;j<=300;++j)
			can1[i]|=can[i][j];
	cout<<(can1[1]?"BitLGM":"BitAryo")<<endl;
	return 0;
}
