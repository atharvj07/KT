import re

n = int(input())
p = list(map(int, input().split()))
for _ in range(n):
    s = input()
    if not re.fullmatch(r"([aeiouy]*[A-Za-z]*){2,}$", s):
        print("NO")
        exit()
    i = 0
    cnt = 0
    while i < len(s):
        if s[i] in 'aeiouy':
            cnt += 1
            i += 1
            while i < len(s) and s[i] not in 'aeiouy':
                i += 1
        else:
            i += 1
    if cnt != p[n - 1]:
        print("NO")
        exit()
print("YES")
