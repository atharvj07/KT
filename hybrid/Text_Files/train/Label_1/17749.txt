#include <iostream>

int main() {
  long long p, q, a[100], d, r;
  int n;
  std::cin >> p >> q;
  std::cin >> n;
  if (n == 1) {
    std::cin >> a[1];
    if (p * a[1] == q) std::cout << "YES" << std::endl; else std::cout << "NO" << std::endl;
    return 0;
  }
  for (int i = n; i >= 1; --i) std::cin >> a[i];
  d = 0;
  r = 1;
  for (int i = n; i >= 1; --i) {
    long long tmp = d;
    d = r;
    r = tmp + a[i] * r;
  }
  if (r * q == d * p) std::cout << "YES" << std::endl; else std::cout << "NO" << std::endl;
  return 0;
}
