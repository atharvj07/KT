n = int(input())
a = list(map(int, input().split()))
a.sort(reverse=True)
mx = min(a)
ans = 0
lst = []
i = 0
while i < n:
    j = i
    cnt = 0
    while j < n and a[j] == a[i]:
        cnt += 1
        j += 1
    if cnt >= mx:
        ans += mx * 2
    else:
        ans += cnt * 2
    i = j
print(ans)
