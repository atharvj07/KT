import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[] arr = Arrays.stream(in.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int ans = -1;
        int[] sorted = Arrays.stream(arr).sorted().toArray();
        int[] cyclicShift = new int[n];
        for (int i = 0; i < n; i++) cyclicShift[i] = sorted[(i + 1) % n];
        for (int i = 0; i < n; i++) {
            if (!Arrays.equals(arr, sorted) && !Arrays.equals(arr, cyclicShift)) break;
            if (Arrays.equals(arr, sorted) || Arrays.equals(arr, cyclicShift)) ans = i;
            int[] tmp = new int[n];
            for (int j = 0; j < n - 1; j++) tmp[j] = arr[j + 1];
            tmp[n - 1] = arr[0];
            arr = tmp;
        }
        System.out.println(ans);
    }
}
