#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, p, s, a[3001], b[3001], ans = 0, sum[3001], ansa[3001], ansb[3001];
  cin >> n >> p >> s;
  for (int i = 1; i <= n; ++i) cin >> a[i];
  for (int i = 1; i <= n; ++i) cin >> b[i];
  for (int i = 1; i <= n; ++i) sum[i] = a[i] + b[i];
  sort(sum + 1, sum + 1 + n, greater<int>());
  for (int i = 1; i <= p; ++i) ans += a[ansa[i] = i];
  for (int i = p + 1; i <= p + s; ++i) ans += b[ansb[i - p] = i];
  cout << ans << endl;
  for (int i = 1; i <= p; ++i) cout << ansa[i] << " ";
  cout << endl;
  for (int i = 1; i <= s; ++i) cout << ansb[i] << " ";
  cout << endl;
}
