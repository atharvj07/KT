n = int(input())
ar = list(map(int, input().split()))
ar.sort()
if ar[0]==0: print(0)
else:
    sum_odd = 0
    sum_even = 0
    odd_max = 0
    for i in range(n):
        if ar[i]%2==1:
            sum_odd+=ar[i]
            odd_max = max(odd_max,ar[i])
        else: sum_even+=ar[i]
    sum_all = sum_even + sum_odd
    if sum_even%2==1 and sum_odd!=0:
        print(sum_all - odd_max + 1)
    else: print(sum_all)
