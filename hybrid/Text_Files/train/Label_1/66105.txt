def find_sets(cards, k):
    sets_count = 0
    n = len(cards)
    for i in range(n):
        for j in range(i+1, n):
            for l in range(j+1, n):
                is_set = True
                for feature in range(k):
                    # Check if all features are same or all different
                    if not (cards[i][feature] == cards[j][feature] == cards[l][feature] or
                            (cards[i][feature] != cards[j][feature] and cards[i][feature] != cards[l][feature] and cards[j][feature] != cards[l][feature])):
                        is_set = False
                        break
                if is_set:
                    sets_count += 1
    return sets_count

n, k = map(int, input().split())
cards = [input().strip() for _ in range(n)]
print(find_sets(cards, k))
