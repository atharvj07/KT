#include <bits/stdc++.h>
using namespace std;

long long n, ans;
unordered_map<long long, int> mp;
vector<int> v;

int main() {
    cin >> n;
    while (n) {
        v.push_back(n % 10);
        mp[n % 10]++;
        n /= 10;
    }
    reverse(v.begin(), v.end());
    int zero = 0;
    for (int i = 0; i < v.size(); i++)
        if (v[i] == 0) zero++;
    ans = mp[0] - zero;
    for (long long s = 1; s <= v.size(); s++) {
        ans += s * mp[s];
        if (ans >= 1000000007) ans -= 1000000007;
    }
    cout << ans;
    return 0;
}
