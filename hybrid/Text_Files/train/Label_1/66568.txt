for i in range(int(input())):
    a = list(map(int,input().split()))
    a.reverse()
    b = [a[0]]
    c = []
    for ai in a[1:]:
        if ai>b[-1]:
            c.append(ai)
        else:
            b.append(ai)
    if len(c)==0:
        print(0)
    elif len(b)==0:
        print(1)
    else:
        print(2)
