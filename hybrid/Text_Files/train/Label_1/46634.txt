import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] a = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = sc.nextInt();
        }
        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                if (i == j) {
                    continue;
                }
                for (int k = 0; k < 4; k++) {
                    if (i == k || j == k) {
                        continue;
                    }
                    for (int l = 0; l < 4; l++) {
                        if (i == l || j == l || k == l) {
                            continue;
                        }
                        ans = Math.min(ans, Math.abs(a[i] + a[j] - (a[k] + a[l])));
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
