#include<bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 10;
int main() {
    int n, s;
    cin >> n >> s;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    int ans = n;
    for (int i = 1; i <= n; i++) {
        if (i != s) {
            if (a[a[i]] != a[i] - 1) {
                ans--;
            }
        } else {
            if (a[i] != n - 1) {
                ans--;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
