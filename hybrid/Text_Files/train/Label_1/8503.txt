#include<bits/stdc++.h>
using namespace std;
int main(){
    typedef long long ll;
    ll a,b,ans1,ans2;
    cin>>a>>b;
    if(!(a%2==b%2)){
        cout<<-1<<endl;
        return 0;
    }
    if(a+b==0){
        cout<<0<<endl;
        return 0;
    }
    ans1=(a+b)/2;
    ans2=(a-b)/2;
    if(ans1<0 || ans2<0){
        cout<<-1<<endl;
        return 0;
    }
    cout<<2<<endl;
    cout<<ans1<<" "<<ans2<<endl;
    return 0;
}
