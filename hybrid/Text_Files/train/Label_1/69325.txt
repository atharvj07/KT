import datetime

n = int(input())
events = []
for _ in range(n):
    m, d, p, t = map(int, input().split())
    e1 = datetime.date(2013, m, d) - datetime.timedelta(t - 1)
    e2 = datetime.date(2013, m, d) - datetime.timedelta(1)
    events.append((e1, p, e2))

events.sort()
need = 0
cur = []
for e1, p, e2 in events:
    while cur and e1 >= cur[0][0]:
        need -= cur.pop(0)[1]
    if e1 == e2:
        need += p
        cur.append((e1, p))
    else:
        cur.append((e1, p))
        need = max(need, p)

print(need)
