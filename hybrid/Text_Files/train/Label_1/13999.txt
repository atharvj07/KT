#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;
int a[N], L[N], R[N], ans;
bool vis[N];

void dfs(int x) {
    vis[x] = 1;
    if (L[x] != -1) dfs(L[x]);
    if (R[x] != -1) dfs(R[x]);
}

int main() {
    int n, x;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i] >> L[i] >> R[i];
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) dfs(i);
        memset(vis, 0, sizeof vis);
        x = a[i];
        while (1) {
            if (a[i] == x) break;
            if ((a[i] < x && R[i] != -1) || (a[i] > x && L[i] != -1))
                i = (a[i] < x ? R[i] : L[i]);
            else {
                ans++;
                break;
            }
        }
    }
    cout << ans << '\n';
}
