#include <iostream>
#include <cmath>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        long long x1, p1, x2, p2;
        std::cin >> x1 >> p1 >> x2 >> p2;
        long long n1 = (x1 * pow(10, p1)), n2 = (x2 * pow(10, p2));
        if (n1 < n2)
            std::cout << "<\n";
        else if (n1 > n2)
            std::cout << ">\n";
        else
            std::cout << "=\n";
    }
    return 0;
}
