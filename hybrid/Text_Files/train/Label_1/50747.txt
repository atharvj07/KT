import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.nextLine();
        int n = s.length();
        int[] pre = new int[3];
        int[] suf = new int[3];
        int[][] dp = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                pre[j] = dp[i][j];
                suf[j] = dp[n - i - 1][j];
            }
            int ch = s.charAt(i) - '0';
            if (ch == 0) {
                dp[i][0] = pre[0] + 1;
                dp[i][1] = pre[1] + 1;
                dp[i][2] = pre[2] + 1;
            } else {
                dp[i][0] = pre[0];
                dp[i][1] = pre[1];
                dp[i][2] = pre[2];
            }
        }
        int ans = 0;
        for (int i = n - 1; i >= 0; i--) {
            int ch = s.charAt(i) - '0';
            if (ch == 0) {
                ans += dp[i][0] * suf[0] + dp[i][1] * suf[1] + dp[i][2] * suf[2];
            }
        }
        System.out.println(ans);
    }
}
