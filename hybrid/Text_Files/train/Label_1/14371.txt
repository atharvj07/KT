#include <bits/stdc++.h>
#define ll long long

using namespace std;

const ll mod = (ll)1e9 + 7;
const ll MAXN = 2e3 + 5;

inline ll add(ll a, ll b){
  return ((a % mod) + (b % mod)) % mod;
}
inline ll mul(ll a, ll b){
  return ((a % mod) * (b % mod)) % mod;
}

ll dp[MAXN][MAXN], pw[MAXN];

int main(){
  ios_base::sync_with_stdio(false); cin.tie(nullptr); cout.tie(nullptr);
  ll n, p;
  cin >> n >> p;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  pw[0] = 1;
  for (ll i = 1; i <= n; i++) pw[i] = mul(pw[i - 1], p);
  for (ll k = 0; k < n; k++) {
    dp[0][k] = 1;
    for (ll j = 1; j <= k; j++) {
      if (j < n and a[j] <= a[k]) dp[j][k] = dp[j - 1][k - 1];
      dp[j][k] = add(dp[j][k], dp[j][k - 1]);
    }
  }
  vector<ll> res;
  for (ll x = 1; x <= 2000; x++) {
    ll val = 0;
    for (ll k = 0; k < n; k++) {
      if (x <= a[k]) val = add(val, dp[k][k]);
    }
    if (val % p) res.push_back(x);
  }
  cout << res.size() << '\n';
  for (auto x: res) cout << x << ' ';
  cout << '\n';
}
