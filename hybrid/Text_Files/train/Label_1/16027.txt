#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;

int main() {
  ll n;
  cin >> n;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];
  ll ans = 0;
  vector<ll> st;
  for (ll i = 0; i < n; i++) {
    if (st.empty() || a[i] > st.back()) {
      st.push_back(a[i]);
      continue;
    }
    ll bal = 0;
    while (!st.empty() && a[i] <= st.back()) {
      bal += st.back();
      st.pop_back();
    }
    if (bal <= a[i]) ans += bal + 1;
    else ans += a[i];
    st.push_back(a[i]);
  }
  ans += st.size();
  cout << ans << "\n";
}
