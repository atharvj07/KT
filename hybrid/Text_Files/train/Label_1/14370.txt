#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        long long n,k;
        cin>>n>>k;
        int cnt[10]={};
        long long tmp=n,ans=n;
        while(tmp){
            cnt[tmp%10]++;
            tmp/=10;
        }
        if(k==1){
            ans=(n%10==0)?n+1:n;
        }else{
            while(cnt[10-k]){
                long long x=n%10;
                n/=10;
                cnt[x]--;
                ans=n*10+(x+1);
            }
            while(k!=0){
                if(cnt[k]){
                    ans=n*10+k;
                    break;
                }
                k--;
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
