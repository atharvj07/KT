#include <iostream>

const int MAXN = 50;

using namespace std;

int n, m;

int a[MAXN][MAXN], b[MAXN][MAXN];

bool equal() {
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
      if (a[i][j] != b[i][j]) return false;
  return true;
}

void print() {
  int c = 0;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
      if (a[i][j]) {
        if (!c++) cout << endl;
        cout << i + 1 << " " << j + 1 << endl;
      }
}

int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) cin >> a[i][j];
  while (true) {
    bool flag = false;
    for (int i = 0; i < n - 1; i++)
      for (int j = 0; j < m - 1; j++)
        if (!a[i][j] && !a[i][j + 1] && !a[i + 1][j] && a[i + 1][j + 1]) {
          b[i][j] = b[i][j + 1] = b[i + 1][j] = b[i + 1][j + 1] = 1;
          flag = true;
        }
    if (!flag) break;
  }
  if (equal())
    print();
  else
    cout << -1 << endl;
  return 0;
}
