#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAXN = 50 + 5;
const int MAXQ = 1e5 + 5;

int h, q;
unordered_map<ll, unordered_map<ll, pair<int, int>>> mp;

bool check(ll x, ll l, ll r, int ans) {
  if (mp[x].find(l) != mp[x].end() && mp[x][l].second == ans)
    return 1;
  if (mp[x].find(r) != mp[x].end() && mp[x][r].second == 1 - ans)
    return 1;
  return 0;
}

int main() {
  cin >> h >> q;
  for (int i = 1; i <= q; i++) {
    ll x, l, r, ans;
    cin >> x >> l >> r >> ans;
    mp[x][l] = {r, ans};
  }
  for (ll i = 1; i <= (1 << h) - 1; i++) {
    if (mp[h].find(i) != mp[h].end()) {
      if (mp[h][i].second) {
        ll l = 1, r = (1 << h) - 1;
        while (l < r) {
          ll mid = (l + r) / 2;
          if (check(h - 1, mid, r, mp[h][i].second))
            r = mid;
          else
            l = mid + 1;
        }
        if (mp[h][i].second == check(h - 1, l, l + 1, mp[h][i].second))
          cout << i << '\n';
        else {
          cout << "Game cheated!\n";
          return 0;
        }
        return 0;
      }
    }
  }
  cout << "Data not sufficient!\n";
  return 0;
}
