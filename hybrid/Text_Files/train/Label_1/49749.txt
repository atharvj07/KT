import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String password = sc.nextLine();
        int n = sc.nextInt();
        boolean found = false;
        for (int i = 0; i < n; i++) {
            String word = sc.nextLine();
            if (password.contains(word)) {
                found = true;
                break;
            }
        }
        System.out.println(found ? "YES" : "NO");
    }
}
