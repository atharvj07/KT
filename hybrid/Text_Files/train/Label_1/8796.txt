#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    long long x;
    std::cin >> n >> x;
    std::vector<long long> a(n);
    for (auto &e : a) std::cin >> e;
    std::sort(a.begin(), a.end());
    long long tot = 0, ans = 0;
    for (auto e : a) {
      tot += e;
      if (tot >= x) ans++;
      else
        break;
    }
    std::cout << ans << '\n';
  }
  return 0;
}
