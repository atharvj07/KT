#include<bits/stdc++.h>
using namespace std;
signed main() {
    int t; cin >> t;
    while (t--) {
        int n, m; cin >> n >> m;
        vector<long long> a(n+1);
        for (int i = 0; i < n; i++) cin >> a[i+1];
        for (int i = 0; i < m; i++) {
            long long x; cin >> x;
            long long sum = 0, cnt = 0, ans = -1;
            for (long long j = 1; j <= n; j++) {
                sum += a[j];
                cnt++;
                if (sum >= x) {
                    ans = cnt;
                    break;
                }
            }
            cout << ans << ' ';
        }
        cout << '\n';
    }
}
