def solve(n, h):
    MOD = 1000000007
    dp = [[0] * 2 for _ in range(n)]
    dp[0][0] = 1
    dp[0][1] = 0
    for i in range(1, n):
        dp[i][0] = ((2 * dp[i - 1][0] + (1 if h[i - 1] == h[i] else 0)) % MOD)
        dp[i][1] = (dp[i][0] + dp[i - 1][1]) % MOD
    return dp[n - 1][1]

n = int(input())
h = list(map(int, input().split()))
print(solve(n, h))
