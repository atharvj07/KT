#include <bits/stdc++.h>
using namespace std;

const int N = 5010, inf = 1e9;

int n, m, a[N], ans[N];
vector<pair<int, pair<int, int>>> b;

int main() {
    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        int t, l, r, d;
        cin >> t >> l >> r;
        if (t == 1) cin >> d, b.push_back({t, {l, d}});
        else {
            cin >> d;
            b.push_back({t, {l, r}});
            ans[l] = max(ans[l], d);
            ans[r + 1] = max(ans[r + 1], -d);
        }
    }
    int cur = 0;
    bool flag = 1;
    for (int i = 1; i <= n; i++) {
        cur += ans[i];
        if (cur < 0) flag = 0;
        a[i] = -inf;
    }
    cur = 0;
    for (auto x : b) {
        if (x.first == 1) cur += x.second.second;
        else if (cur + x.second.first > ans[x.second.first])
            a[x.second.first] = cur + x.second.first;
    }
    for (int i = 1; i <= n; i++) cur += ans[i];
    if (cur == 0) {
        for (int i = 1; i <= n; i++)
            if (a[i] == -inf) a[i] = 0;
        cout << "YES\n";
        for (int i = 1; i <= n; i++) cout << a[i] << " ";
    } else
        cout << "NO\n";
}
