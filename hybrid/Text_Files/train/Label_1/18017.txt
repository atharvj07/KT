#include <iostream>
#include <vector>
using namespace std;

int main() {
    string s;
    cin >> s;
    int cnt_a = 0, cnt_b = 0, cnt_c = 0;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] == 'a') cnt_a++;
        else if (s[i] == 'b') cnt_b++;
        else if (s[i] == 'c') cnt_c++;
    }
    if ((cnt_c == cnt_a && cnt_c != cnt_b) || (cnt_c == cnt_b && cnt_c != cnt_a)) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
