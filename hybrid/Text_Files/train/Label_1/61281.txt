import sys

input = sys.stdin.readline
for _ in range(int(input())):
    n = int(input())
    a = input().strip()
    b = input().strip()

    if a == b:
        print(0)
    elif a[0] == b[0] and a[-1] == b[-1]:
        print(1)
    elif a[0] == b[-1] and a[-1] == b[0]:
        print(1)
    else:
        print(2)
