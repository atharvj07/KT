#include <algorithm>
#include <bitset>
#include <iostream>

typedef long long ll;

int main() {
  int n;
  std::cin >> n;
  while (n--) {
    ll p, q, b;
    std::cin >> p >> q >> b;
    std::bitset<64> vis;
    p %= q;
    while (p != 0) {
      if (vis[p])
        break;
      vis[p] = true;
      p *= b;
      p /= q;
    }
    std::cout << (p == 0 ? "Finite\n" : "Infinite\n");
  }
  return 0;
}
