#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    string n;
    cin >> n;

    int len = n.length();
    bool hasFive = false;
    bool hasZero = false;
    for (int i = 0; i < len; i++) {
        if (n[i] == '5') hasFive = true;
        if (n[i] == '0') hasZero = true;
    }

    if (!hasFive) {
        cout << -1 << endl;
        return 0;
    }

    if (!hasZero) {
        cout << 0 << endl;
        return 0;
    }

    int lastZero = -1;
    int steps = 0;
    for (int i = len - 1; i >= 0; i--) {
        if (n[i] == '0') {
            lastZero = i;
            break;
        }
    }

    for (int i = lastZero - 1; i >= 0; i--) {
        if (n[i] == '5') {
            swap(n[i], n[lastZero]);
            lastZero = i;
            steps++;
        }
    }

    cout << steps << endl;

    return 0;
}
