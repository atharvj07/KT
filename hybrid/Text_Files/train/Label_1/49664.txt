import java.util.Scanner;

class Main {

    private static final int[] WEEK = { 0, 1, 1, 1, 1, 1, 0 };

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = Integer.parseInt(s.next());
        long[] a = new long[n + 2];
        a[0] = Integer.MAX_VALUE;
        for (int i = 1; i <= n; ++i) {
            a[i] = s.nextLong();
        }
        int ans = 1;
        long min = a[1];
        for (int i = 2; i <= n; ++i) {
            long cur = a[i];
            for (int j = 1; j <= 6; ++j) {
                if (WEEK[j] <= cur) {
                    cur -= WEEK[j];
                }
            }
            if (cur < min) {
                min = cur;
                ans = i;
            }
        }
        System.out.print(ans);
    }
}
