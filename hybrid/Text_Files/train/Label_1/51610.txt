import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long n = sc.nextInt();
    long ans = 0;
    for (long i = 1; i <= n; i++) {
      String s = String.valueOf(i);
      boolean has7 = false, has5 = false, has3 = false;
      for (char c : s.toCharArray()) {
        if (c == '7') has7 = true;
        if (c == '5') has5 = true;
        if (c == '3') has3 = true;
      }
      if (has7 && has5 && has3) ans++;
    }
    System.out.println(ans);
  }
}
