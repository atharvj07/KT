import java.util.*;
import java.lang.*;
import java.io.*;

public class Main {
  public static void main(String args[]) throws IOException {
    Scanner in = new Scanner(System.in);
    String s = in.next();
    int[] cnt = new int[26];
    for (int i = 0; i < s.length(); i++) {
      cnt[s.charAt(i) - 'A']++;
    }
    long ans = 0;
    for (int i = 0; i < 26; i++) {
      ans += cnt[i] * (cnt[i] - 1) / 2;
    }
    System.out.println(ans);
  }
}
