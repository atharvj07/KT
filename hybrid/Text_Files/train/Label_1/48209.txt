import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int i = 0, j = n - 1, cnt = 0;
        while (i < n && a[i] <= a[i + 1]) {
            i++;
        }
        while (j >= 0 && a[j] >= a[j - 1]) {
            j--;
        }
        if (i < j) {
            cnt = 1;
        }
        while (i < j) {
            if (a[i] > a[j]) {
                cnt++;
            }
            i++;
            j--;
        }
        if (cnt <= 1) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
