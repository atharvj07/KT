#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    string w;
    for (int i = 0; i < n; i++) {
      if (s[i] == s[n + i]) w += s[i];
      else w += "0";
    }
    cout << w << endl;
  }
  return 0;
}
