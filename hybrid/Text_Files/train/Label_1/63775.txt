from bisect import bisect_left, bisect_right
n = int(input())
a = []
b = []
for i in range(n):
    x, y = map(int, input().split())
    a.append(x)
    b.append(y)
    
a.extend(b)
a.sort()
ans = n
for i in range(n):
    x = a[i]
    l = bisect_left(a, x)
    r = bisect_right(a, x)
    if l != i and r != i + n:
        ans -= 1
    
if ans == n:
    print(-1)
else:
    print(ans)
