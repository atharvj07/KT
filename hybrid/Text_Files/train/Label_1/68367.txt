def calc_occurrences(s, p):
    dp = [[0 for _ in range(len(s) + 1)] for _ in range(len(s) + 1)]
    for i in range(len(s) + 1):
        dp[i][0] = i // len(p)
    for i in range(1, len(s) + 1):
        for j in range(1, i + 1):
            dp[i][j] = dp[i - 1][j]
            if s[i - 1] == p[(i - j) % len(p)]:
                if (i - j) % len(p) == 0:
                    dp[i][j] = max(dp[i][j], dp[i - len(p)][j - len(p)] + 1)
                else:
                    dp[i][j] = max(dp[i][j], dp[i - 1][j - 1])
    return dp

def solve(s, p):
    dp = calc_occurrences(s, p)
    answer = [0] * (len(s) + 1)
    for j in range(len(s) + 1):
        for i in range(len(s) + 1):
            answer[j] = max(answer[j], dp[i][j])
    print(*answer)

s = input().strip()
p = input().strip()
solve(s, p)
