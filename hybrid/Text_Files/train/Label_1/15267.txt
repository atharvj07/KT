#include <bits/stdc++.h>
#define ll long long
#define mod 998244353
using namespace std;
vector<ll> f1(3e5 + 5), f2(3e5 + 5);
void preCalc() {
  f1[0] = 1;
  f2[0] = 1;
  for (ll i = 1; i < 3e5 + 5; ++i) {
    f1[i] = f1[i - 1] * i;
    f1[i] %= mod;
    f2[i] = f1[i] * f1[i];
    f2[i] %= mod;
  }
}
ll bigMod(ll b, ll p) {
  if (p == 0) return 1;
  if (p == 1) return b;
  ll po = bigMod(b, p / 2);
  po = (po * po) % mod;
  if (p % 2 == 1) po = (po * b) % mod;
  return po;
}
int main() {
  ll n, p, k;
  preCalc();
  cin >> n >> p >> k;
  vector<ll> a(n);
  map<ll, ll> mp;
  for (ll i = 0; i < n; ++i) {
    cin >> a[i];
    mp[a[i]] = i;
  }
  sort(a.begin(), a.end());
  ll ans = 0;
  for (ll i = 0; i < n; ++i) {
    for (ll j = i + 1; j < n; ++j) {
      ll s = a[i] + a[j];
      if (s > k) {
        ll need = (k - a[i] * s % p + p) % p;
        if (mp.find(need) != mp.end() && mp[need] > j) {
          ans++;
        }
      } else {
        ll need = (k - a[i] * s % p + p) % p * bigMod(a[i], p - 2) % p;
        if (mp.find(need) != mp.end() && mp[need] > j) {
          ans++;
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
