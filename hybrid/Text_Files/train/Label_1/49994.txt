import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String[] s = new String[n];
        for (int i = 0; i < n; i++) {
            s[i] = scanner.next();
        }
        String[] t = new String[n];
        boolean isIncreasing = true;
        boolean isSolvable = true;
        for (int i = 1; i < n; i++) {
            isIncreasing = isIncreasing && compare(s[i - 1], s[i]) < 0;
            t[i] = s[i];
        }
        if (!isIncreasing) {
            System.out.println("NO");
            return;
        }
        for (int i = 0; i < n - 1; i++) {
            if (s[i].length() < s[i + 1].length()) {
                int len = s[i + 1].length() - s[i].length();
                for (int j = 0; j < len; j++) {
                    s[i] = "0" + s[i];
                }
            } else if (s[i].length() > s[i + 1].length()) {
                int len = s[i].length() - s[i + 1].length();
                for (int j = 0; j < len; j++) {
                    s[i + 1] = "0" + s[i + 1];
                }
            }
        }
        for (int i = 0; i < n - 1; i++) {
            if (!s[i].equals("?")) {
                int j = i + 1;
                while (j < n && t[j].equals("?")) {
                    j++;
                }
                if (j < n) {
                    if (s[i].length() == t[j].length()) {
                        for (int k = 0; k < s[i].length(); k++) {
                            if (s[i].charAt(k) != t[j].charAt(k)) {
                                if (s[i].charAt(k) == '?') {
                                    s[i] = s[i].substring(0, k) + t[j].charAt(k) + s[i].substring(k + 1);
                                } else {
                                    isSolvable = false;
                                }
                            }
                        }
                    } else {
                        isSolvable = false;
                    }
                }
            }
        }
        if (isSolvable) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                s[i] = s[i].replaceAll("\\?", "0");
                System.out.println(s[i]);
            }
        } else {
            System.out.println("NO");
        }
    }

    private static int compare(String s1, String s2) {
        int i = 0;
        while (i < s1.length() && i < s2.length()) {
            if (s1.charAt(i) == '?' || s2.charAt(i) == '?') {
                i++;
            } else if (s1.charAt(i) < s2.charAt(i)) {
                return -1;
            } else if (s1.charAt(i) > s2.charAt(i)) {
                return 1;
            } else {
                i++;
            }
        }
        return s1.length() - s2.length();
    }
}
