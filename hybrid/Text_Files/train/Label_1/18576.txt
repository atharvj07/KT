#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    string a;
    cin >> a;
    while (m--) {
      string b = a;
      for (int i = 0; i < n; i++) {
        int cnt = 0;
        if (i > 0) cnt += a[i - 1] == '1';
        if (i < n - 1) cnt += a[i + 1] == '1';
        if (a[i] == '0' && cnt == 1) b[i] = '1';
      }
      a = b;
    }
    cout << a << '\n';
  }
  return 0;
}
