for t in range(int(input())):
    n = int(input())
    se = [list(map(int, input().split())) for i in range(n - 1)]
    p = [0] * n
    for l, *x in sorted(se):
        for v in x:
            if not p[v - 1]:
                p[v - 1] = v
                break
    for i, v in enumerate(p):
        if not v:
            for j in range(n):
                if j + 1 not in p:
                    p[i] = j + 1
                    break
    print(*p)
