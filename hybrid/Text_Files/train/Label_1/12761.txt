#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    vector<int> a(t);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    if (a.back() != t - 1) {
        cout << -1 << endl;
        return 0;
    }
    int n = 0, m = 0;
    for (int i = 0; i < t; ++i) {
        if (!a[i]) n = i;
        else if (a[i] == a[i + 1]) m++;
    }
    m += n * 2 + 1;
    cout << n + 1 << " " << m + 1 << endl;
    cout << n + 1 << " " << m / 2 + 1 << endl;
}
