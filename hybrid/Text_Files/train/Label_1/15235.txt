#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.length();
    int i = 0;
    bool found = false;
    while (i < n / 2 && !found) {
        if (s[i] != s[n - i - 1]) {
            if (s[i + 1] == s[n - i - 1]) {
                s.insert(i, 1, s[n - i - 1]);
                found = true;
            } else if (s[i] == s[n - i - 2]) {
                s.insert(n - i, 1, s[i]);
                found = true;
            } else {
                cout << "NA" << endl;
                return 0;
            }
        }
        i++;
    }
    if (!found) {
        s.insert(n / 2, 1, s[n / 2]);
    }
    cout << s << endl;
    return 0;
}
