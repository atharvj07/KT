import sys

def check(a, i, j, k):
    return (a[j] - a[i] < a[k] - a[j]) or (a[j] - a[i] > a[k] - a[j] + a[i] - 1)

for t in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    for i in range(1, n - 1):
        for j in range(i + 1, n):
            l = i - 1
            r = j + 1
            while l >= 0 and r < n:
                if check(a, l, i, j):
                    break
                if check(a, i, j, r):
                    l -= 1
                else:
                    r += 1
            if l >= 0 and r < n:
                print(l + 1, i + 1, r + 1)
                break
    else:
        print(-1)
