#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  int n, k;
  cin >> n >> k;
  vector<ll> a(n), b(n);
  for (int i = 0; i < n; ++i) cin >> a[i] >> b[i];
  ll ans = -1;
  int cnt = 0;
  for (ll prv = 0, cur = 0, i = 0; i < 32; ++i) {
    int s = 0;
    for (int j = 0; j < n; ++j) {
      if ((bool)(a[j] & (1 << i)) != (bool)(b[j] & (1 << i))) s++;
    }
    if (!i) {
      if (s == n || s == n - k || s == n - k + 1) {
        if (prv || s == n || s == n - k + 1) cnt++, ans &= (1 << i);
      } else
        cnt = 0;
    } else {
      if (s == n - cnt || s == n - k - cnt || s == n - k - cnt + 1)
        cnt++, ans &= (1 << i);
      else
        cnt = 0;
    }
    prv = cur;
    cur = s;
  }
  if (cnt == k)
    cout << bitset<32>(ans).to_string() << "\n";
  else
    cout << "-1\n";
}
