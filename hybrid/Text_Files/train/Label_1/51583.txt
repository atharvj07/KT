import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int prize = 0;

        if (x == 1) {
            prize += 300000;
        } else if (x == 2) {
            prize += 200000;
        } else if (x <= 3) {
            prize += 100000;
        }

        if (y == 1) {
            prize += 300000;
        } else if (y == 2) {
            prize += 200000;
        } else if (y <= 3) {
            prize += 100000;
        }

        if (x == 1 && y == 1) {
            prize += 400000;
        }

        System.out.println(prize);
    }
}
