import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        String s = sc.next();
        char[] c = s.toCharArray();
        int[] a = new int[m];
        for (char cc : c) {
            a[cc - 'a']++;
        }
        Arrays.sort(a);
        int ans = 0;
        for (int i = 1; i < m; i++) {
            ans += a[i];
        }
        System.out.println(ans);
    }
}
