#include <iostream>
#include <vector>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;

  for (int i = 0; i < t; i++) {
    int k;
    cin >> k;

    string s;
    cin >> s;

    int last_angry = 0;
    for (int j = 0; j < k; j++) {
      if (s[j] == 'A') {
        last_angry = j + 1;
      }
    }

    cout << last_angry << endl;
  }

  return 0;
}
