import sys
input = sys.stdin.readline

n, s, k = map(int, input().split())
a = list(map(int, input().split()))
c = input().strip()

if a[s - 1] < k:
    print(-1)
    exit(0)

ans = 0
lst = 0
lstc = c[s - 1]
for i in range(s - 1, -1, -1):
    if c[i] == lstc:
        continue
    else:
        if a[i] >= lst:
            ans += 1
            lst = a[i]
            lstc = c[i]

lst = 0
lstc = c[s - 1]
for i in range(s - 1, n):
    if c[i] == lstc:
        continue
    else:
        if a[i] >= lst:
            ans += 1
            lst = a[i]
            lstc = c[i]

print(ans)
