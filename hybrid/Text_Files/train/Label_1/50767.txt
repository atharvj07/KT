import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int k = in.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
    }
    int m = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] > 0) {
        m += a[i];
      }
    }
    if (k % 2 == 0) {
      for (int i = 0; i < n; i++) {
        if (a[i] < 0) {
          m += a[i];
        }
      }
    } else {
      int min = 0;
      for (int i = 0; i < n; i++) {
        if (a[i] < 0) {
          min = Math.min(min, a[i]);
        }
      }
      m += min;
    }
    System.out.println(m);
  }
}
