#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a;
  for (int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    a.push_back(x);
  }
  int q;
  cin >> q;
  while (q--) {
    int x;
    cin >> x;
    if (x == 0) {
      cout << 0 << endl;
      continue;
    }
    int ans = -1;
    int j = n - 1;
    int cnt = 0;
    for (int i = n - 1; i >= 0; --i) {
      if (j < i) break;
      while (j >= i && a[j] + a[i] <= x) {
        --j;
        ++cnt;
      }
      if (j >= i && a[j] + a[i] == x) {
        ans = cnt + 1;
        break;
      }
    }
    if (cnt <= k && ans != -1)
      cout << ans << endl;
    else
      cout << -1 << endl;
  }
  return 0;
}
