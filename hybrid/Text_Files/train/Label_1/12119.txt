#include <iostream>
#include <vector>
int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }
  long long ans = 0;
  int freq = 0;
  int max_freq = 0;
  int min_val = 1e9;
  int max_val = -1e9;
  for (auto &x : a) {
    freq += (x == min_val || x == max_val);
    max_freq = std::max(max_freq, freq);
    min_val = std::min(min_val, x);
    max_val = std::max(max_val, x);
  }
  if (max_freq >= k) {
    std::cout << 0 << std::endl;
    return 0;
  }
  ans = (n - max_freq) / 2;
  ans += (n - max_freq) % 2;
  std::cout << ans << std::endl;
  return 0;
}
