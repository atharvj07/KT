import sys

input = sys.stdin.readline
INF = 10**9

# Read input
t = int(input())
for _ in range(t):
    n, k = map(int, input().strip().split())
    x = list(map(int, input().strip().split()))
    y = list(map(int, input().strip().split()))

    # Sort points by y-coordinate
    points = list(zip(x, y))
    points.sort(key=lambda p: p[1])

    # Find the lowest y-coordinate of any point
    min_y = INF
    for point in points:
        if point[1] < min_y:
            min_y = point[1]

    # Find the highest y-coordinate of any point
    max_y = -INF
    for point in points:
        if point[1] > max_y:
            max_y = point[1]

    # Calculate the number of points that can be saved
    count = 0
    for point in points:
        if point[1] >= min_y and point[1] <= max_y:
            count += 1

    # Print the answer
    print(count)
