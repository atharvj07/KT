#include <bits/stdc++.h>
using namespace std;
int n, m;
char arr[2002][2002];
void dfs(int x, int y, int f) {
  if (f) arr[x][y] = '+';
  if (x == 0 || y == 0 || x == n + 1 || y == m + 1) f = 1;
  if (x > 0 && arr[x - 1][y] == 'w') dfs(x - 1, y, f);
  if (y > 0 && arr[x][y - 1] == 'w') dfs(x, y - 1, f);
  if (x < n && arr[x + 1][y] == 'w') dfs(x + 1, y, f);
  if (y < m && arr[x][y + 1] == 'w') dfs(x, y + 1, f);
}
int main() {
  cin >> n >> m;
  if (n == 0 || m == 0) return cout << -1, 0;
  for (int i = 0; i <= n + 1; i++)
    for (int j = 0; j <= m + 1; j++) arr[i][j] = '.';
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) cin >> arr[i][j];
  dfs(1, 1, 0);
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) cout << arr[i][j];
    cout << '\n';
  }
}
