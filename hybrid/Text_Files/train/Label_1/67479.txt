import sys

input = sys.stdin.readline

n, k = map(int, input().split())
a = list(map(int, input().split()))
b = [set() for i in range(n)]
for i in range(k):
    x, y = map(int, input().split())
    x -= 1
    y -= 1
    b[x].add(y)
    b[y].add(x)
c = [0 for i in range(n)]
for i in range(n):
    for j in range(n):
        if i == j or i in b[j]:
            continue
        if a[i] > a[j]:
            c[i] += 1
for i in c:
    print(i)
