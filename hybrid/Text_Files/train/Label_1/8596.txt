#include <iostream>
#include<vector>
#include<algorithm>
using namespace std;
const int PINF=1e9+7;
int main() {
  int n,ans=-1,p1=0,p2=0;
  cin>>n;
  vector<int> v[n];
  for(int i=0;i<n;++i){
    int x;
    for(int j=0;j<5;++j){
      cin>>x;
      v[i].push_back(x);
      if(x!=-1)
        p1+=max(0,(119-x)/250);
    }
  }
  int mx=0;
  for(int j=0;j<5;++j){
    int k=0;
    for(int i=0;i<n;++i){
      if(v[i][j]==-1)k++;
    }
    if((double)k/(double)n<=0.5)mx=500;
    if((double)k/(double)n>0.5 && (double)k/(double)n<=0.25)mx=1000;
    if((double)k/(double)n>0.25 && (double)k/(double)n<=0.125)mx=1500;
    if((double)k/(double)n>0.125 && (double)k/(double)n<=0.0625)mx=2000;
    if((double)k/(double)n>0.0625 && (double)k/(double)n<=0.03125)mx=2500;
    if((double)k/(double)n>0.03125)mx=3000;
  }
  for(int j=0;j<5;++j){
    int k=0;
    for(int i=0;i<n;++i){
      if(v[i][j]==-1)k++;
      else{
        p2+=max(0,(119-v[i][j])/250)*mx/1000;
      }
    }
    if((double)k/(double)n<=0.5)mx=500;
    if((double)k/(double)n>0.5 && (double)k/(double)n<=0.25)mx=1000;
    if((double)k/(double)n>0.25 && (double)k/(double)n<=0.125)mx=1500;
    if((double)k/(double)n>0.125 && (double)k/(double)n<=0.0625)mx=2000;
    if((double)k/(double)n>0.0625 && (double)k/(double)n<=0.03125)mx=2500;
    if((double)k/(double)n>0.03125)mx=3000;
  }
  for(int i=0;i<n;++i){
    for(int j=0;j<5;++j){
      if(v[i][j]!=-1){
        p2+=max(0,(119-v[i][j])/250)*mx/1000;
      }
    }
  }
  if(p2<p1)
    for(int i=3;;++i){
      p2+=2*mx/1000;
      if(p2>p1){
        ans=i;
        break;
      }
    }
  cout<<ans<<endl;
  return 0;
}
