#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  int ans = 1;
  for (int i = 2, j = 0; i <= n; i++) {
    if ((a[i - 1] ^ a[j]) != 0) { j = i - 1; }
    ans = max(ans, i - j);
  }
  cout << ans << '\n';
  return 0;
}
