import sys

def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)

def lcm(a, b):
    return (a * b) // gcd(a, b)

for _ in range(int(input())):
    n = int(input())
    divisors = [int(x) for x in input().split()]
    divisors.sort()
    if len(divisors) != n:
        print(-1)
        continue

    x = divisors[0]
    for i in range(1, n - 1):
        x = lcm(x, divisors[i])

    if divisors[-1] % x == 0:
        print(x)
    else:
        print(-1)
