#include <bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;
int n, p, dp[55][2][2][2];
vector<int> g[55];

inline int add(int a, int b) { return (a + b) % mod; }
inline int mul(int a, int b) { return (1LL * a * b) % mod; }

void dfs(int v, int prv = -1, int c = 0, int ev = 0) {
    dp[v][p][c][ev] = 1;
    for (int u : g[v]) {
        if (u == prv) continue;
        dfs(u, v, c ^ 1, (ev ^ 1) | (c == 0));
        dp[v][p][c][ev] = add(dp[v][p][c][ev], dp[u][p][c ^ 1][ev ^ 1]);
    }
}

int32_t main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> p;
    int c;
    for (int i = 1; i <= n; i++) {
        cin >> c;
        if (c == -1) g[i].push_back(i);
        else if (c == 0)
            g[i].push_back(i);
        else {
            if (p)
                g[i].push_back(i);
            else
                g[0].push_back(i);
        }
    }
    dfs(0);
    cout << dp[0][p][0][0] << '\n';
}
