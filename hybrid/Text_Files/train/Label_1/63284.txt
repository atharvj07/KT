n = int(input())
a = list(map(int, input().split()))
s, f = map(int, input().split())

contest_duration = f - s
max_participants = -1
start_time = 0

for i in range(n):
    current_participants = sum(a[(s - i - 1) % n : (f - i - 1) % n])
    if current_participants > max_participants:
        max_participants = current_participants
        start_time = i + 1

print(start_time)
