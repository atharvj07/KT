#include <bits/stdc++.h>

using namespace std;

#define ll long long
#define mod 1000000007LL

ll binpow(ll a, ll b) {
    a %= mod;
    ll ans = 1;
    while (b > 0) {
        if (b % 2 == 1)
            ans = ans * a % mod;
        a = a * a % mod;
        b /= 2;
    }
    return ans;
}

int main() {
    ll n, m, a, b, g0, x, y, z;
    cin >> n >> m >> a >> b >> g0 >> x >> y >> z;
    vector<vector<ll>> dp(n + 1, vector<ll>(m + 1, 0));
    dp[1][1] = g0;
    for (ll i = 2; i <= m; i++) { dp[1][i] = (dp[1][i - 1] * x + y) % z; }
    for (ll i = 2; i <= n; i++) { dp[i][1] = (dp[i - 1][1] * x + y) % z; }
    for (ll i = 2; i <= n; i++) {
        for (ll j = 2; j <= m; j++) { dp[i][j] = (dp[i - 1][j] + dp[i][j - 1] - dp[i - 1][j - 1] + z) % z; }
    }
    ll ans = 0;
    for (ll i = 1; i <= n - a + 1; i++) {
        for (ll j = 1; j <= m - b + 1; j++) {
            ll tl = dp[i - 1][j - 1], tr = dp[i - 1][j + b - 1], bl = dp[i + a - 1][j - 1], br = dp[i + a - 1][j + b - 1];
            ans = (ans + br - bl - tr + tl + z) % z;
        }
    }
    cout << ans << '\n';
    return 0;
}
