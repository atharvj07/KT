from collections import defaultdict

n, m, k = map(int, input().split())
table = defaultdict(int, {(x, y): 0 for x in range(1, n + 1) for y in range(1, m + 1)})
for _ in range(k):
    op, a, b = map(int, input().split())
    if op == 1:
        for y in range(1, m + 1):
            table[(a, y)] = b
    else:
        for x in range(1, n + 1):
            table[(x, b)] = b
for x in range(1, n + 1):
    print(' '.join(str(table[(x, y)]) for y in range(1, m + 1)))
