#include <iostream>
#include <string>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::string d;
  std::cin >> d;
  std::vector<int> l(n);
  for (auto &i : l) std::cin >> i;
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    if (d[i] == '<')
      ans -= l[i] == 0 ? 0 : l[i];
    else
      ans += l[i] == 0 ? 0 : l[i];
    if (ans < 0 || ans > n) {
      std::cout << "INFINITE" << std::endl;
      return 0;
    }
  }
  std::cout << "FINITE" << std::endl;
}
