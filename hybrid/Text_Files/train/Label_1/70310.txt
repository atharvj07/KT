n, m = map(int, input().split())
a = [input() for _ in range(n)]
x1, x2, y1, y2 = n, 0, m, 0
for i in range(n):
    for j in range(m):
        if a[i][j] == 'B':
            x1, x2, y1, y2 = min(x1, i), max(x2, i), min(y1, j), max(y2, j)
w = x2 - x1 + 1
h = y2 - y1 + 1
if w == h and w > 0:
    print(w * h - sum(len(r) - r.count('B') for r in a))
else:
    print(-1)
