#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 3e5 + 5;
ll a[maxn];

int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    ll ans = 0;
    for (int i = 0; i < k; i++) {
        ans += abs(a[i]);
        ans += abs(a[n - i - 1]);
    }
    for (int i = k; i < n - k; i++) { ans += a[i]; }
    cout << ans << endl;
    return 0;
}
