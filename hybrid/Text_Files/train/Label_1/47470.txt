import java.util.*;
import java.io.*;
class Main{
	public static void main(String[] args)throws Exception{
		BufferedReader bf=new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st=new StringTokenizer(bf.readLine());
		int n=Integer.parseInt(st.nextToken()),c=Integer.parseInt(st.nextToken());
		st=new StringTokenizer(bf.readLine());
		int[] a=new int[n];
		for(int i=0;i<n;i++)a[i]=Integer.parseInt(st.nextToken());
		int[] l=new int[c+1],r=new int[c+1],cnt=new int[c+1];
		Arrays.fill(r,-1);
		int ans[]=new int[n+1];
		for(int i=0;i<n;i++){
			if(l[a[i]]==-1)l[a[i]]=i;
			r[a[i]]=i;
			cnt[a[i]]++;
		}
		for(int l1=0;l1<=c;l1++){
			if(cnt[l1]==0)continue;
			if(l[l1]==-1){
				for(int i=1;i<=c;i++)if(cnt[i]>0){
					if(r[i]>l[i])ans[1]+=n-r[i]+1;
				}
				continue;
			}
			int d=l[l1];
			for(int i=1;i<=c;i++){
				if(cnt[i]==0||l[i]>d)continue;
				ans[1]+=r[i]-l[i]+1;
				if(r[i]==n){
					for(int j=l1;j<=c;j++)if(cnt[j]>0){
						assert l[j]>=d;
						if(l[j]==d){
							ans[1]-=n-d;
						}else ans[1]-=r[j]-d;
					}
					break;
				}else{
					d=r[i];
				}
			}
		}
		for(int p=2;p<=n;p++){
			for(int l1=0;l1<=c;l1++){
				if(cnt[l1]==0)continue;
				if(l[l1]==-1){
					for(int i=1;i<=c;i++)if(cnt[i]>0){
						ans[p]+=((long)(r[i]-l[i]+1))*cnt[i]%998244353;
						ans[p]%=998244353;
					}
				}
			}
			int d=r[1];
			for(int i=1;i<=c;i++){
				if(cnt[i]==0||l[i]>d)continue;
				ans[p]+=((long)(r[i]-l[i]+1))*cnt[i]%998244353;
				ans[p]%=998244353;
				if(r[i]==n)break;
				d=r[i];
			}
		}
		for(int i=0;i<=n;i++)System.out.println(ans[i]);
	}
}
