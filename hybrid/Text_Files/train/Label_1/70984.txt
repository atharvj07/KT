n, a, b = map(int, input().split())
c = list(map(int, input().split()))
if c[0] != c[-1]:
    print(-1)
else:
    ans = 0
    for i in range(1, (n + 1) // 2):
        if c[i] == c[-(i + 1)]:
            continue
        elif c[i] == 2:
            ans += min(a, b)
            c[i] = c[-(i + 1)]
        elif c[-(i + 1)] == 2:
            ans += min(a, b)
            c[-(i + 1)] = c[i]
        else:
            print(-1)
            exit()
    print(ans)
