#include <iostream>
#include <vector>
#include <algorithm>
#include <iterator>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> alarms(n);
    for (auto& [h, m] : alarms) {
        string s;
        cin >> s;
        h = stoi(s.substr(0, 2));
        m = stoi(s.substr(3, 2));
    }
    sort(begin(alarms), end(alarms));
    pair<int, int> cur = {0, 0};
    int ans = 0;
    for (auto [h, m] : alarms) {
        int h_ = (h - cur.first) * 60 + (m - cur.second);
        if (h_ > ans) ans = h_;
        cur = {h, m};
    }
    int h_ = (24 - cur.first) * 60 + (60 - cur.second);
    if (h_ > ans) ans = h_;
    cout << (ans / 60) << ':' << (ans % 60 == 0 ? "00" : to_string(ans % 60)) << '\n';
    return 0;
}
