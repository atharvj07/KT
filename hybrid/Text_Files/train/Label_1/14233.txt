#include<bits/stdc++.h>
#define MOD 1000000007
using namespace std;
typedef long long ll;

ll compute(vector<ll>& v, ll sum) {
    ll n = v.size();
    vector<ll> dp(sum + 1, 0);
    dp[0] = 1;
    for (auto x : v) {
        for (ll j = sum; j >= x; j--) {
            dp[j] = (dp[j] + dp[j - x]) % MOD;
        }
    }
    return dp[sum];
}

int main() {
    ll h, w;
    cin >> h >> w;
    vector<ll> r(h), c(w);
    for (ll i = 0; i < h; i++) cin >> r[i];
    for (ll i = 0; i < w; i++) cin >> c[i];
    ll sum1 = 0, sum2 = 0;
    for (auto x : r) sum1 += x;
    for (auto x : c) sum2 += x;
    if (sum1 != sum2) {
        cout << 0 << "\n";
        return 0;
    }
    cout << compute(r, sum1) << "\n";
    return 0;
}
