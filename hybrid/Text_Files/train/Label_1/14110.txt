#include <bits/stdc++.h>
using namespace std;
int main(){ long long t; cin>>t; while(t--){ long long n, x; cin>>n>>x; unordered_map<long long, long long> m; for(long long i=0;i<n;i++){ long long v; cin>>v; m[v]++; } long long ans = x, g = 0; for(auto it = m.begin(); it!=m.end(); it++){ long long a = it->first; for(long long j=a;j<=x;j+=a){ g = (j%a==0 && j<=x) ? __gcd(g, j/a) : 1; } if(g!=1){ ans = x/g; break; } } cout<<ans<<"\n"; } }
