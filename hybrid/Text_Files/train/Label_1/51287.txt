import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int M = scanner.nextInt();
        int[] blowers = new int[M];
        for (int i = 0; i < M; i++) {
            blowers[i] = scanner.nextInt();
        }
        int Q = scanner.nextInt();
        int[] queries = new int[Q];
        for (int i = 0; i < Q; i++) {
            queries[i] = scanner.nextInt();
        }

        for (int query : queries) {
            System.out.println(solve(N, M, blowers, query));
        }
    }

    static int solve(int N, int M, int[] blowers, int query) {
        int[] dp = new int[N + 1];
        dp[0] = 0;
        for (int i = 1; i <= N; i++) {
            dp[i] = 1000000000;
        }
        for (int i = 1; i <= N; i++) {
            for (int j = 0; j < M; j++) {
                if (i <= blowers[j]) {
                    dp[i] = Math.min(dp[i], dp[i - 1]);
                } else {
                    dp[i] = Math.min(dp[i], dp[blowers[j]] + 1);
                }
            }
        }
        if (dp[N] <= query) {
            return dp[N];
        } else {
            return -1;
        }
    }
}
