#include <bits/stdc++.h>
#define int long long
using namespace std;

const int maxn = 3e5 + 5;
const int mod = 1e9 + 7;

vector<int> adj[maxn];
int n, q;
int sub[maxn], ans = 1;

void dfs(int u, int p) {
    sub[u] = 1;
    for (auto v : adj[u]) {
        if (v != p) {
            dfs(v, u);
            sub[u] += sub[v];
        }
    }
    ans = max(ans, sub[u]);
}

int32_t main() {
    cin >> q;
    while (q--) {
        cin >> n;
        for (int i = 0; i <= n; i++) { adj[i].clear(), sub[i] = 0; }
        for (int i = 1; i < n; i++) {
            int u, v;
            cin >> u >> v;
            adj[u].push_back(v);
            adj[v].push_back(u);
        }
        dfs(1, 1);
        cout << ans << endl;
        ans = 1;
    }
    return 0;
}
