import java.util.*;
public class Main {
    static int l0, l1, a0, a1, res;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        l0 = in.nextInt();
        l1 = in.nextInt();
        in.nextLine();
        int[] t = new int[l0], s = new int[l1];
        char[] c0 = new char[l0], c1 = new char[l1];
        for (int i = 0; i < l0; i++) {
            String[] tmp = in.next().split("-");
            t[i] = Integer.parseInt(tmp[0]);
            c0[i] = tmp[1].charAt(0);
        }
        in.nextLine();
        for (int i = 0; i < l1; i++) {
            String[] tmp = in.next().split("-");
            s[i] = Integer.parseInt(tmp[0]);
            c1[i] = tmp[1].charAt(0);
        }
        for (int i = 0; i + l1 - 1 < l0; i++) {
            boolean f = true;
            for (int j = 0; j < l1; j++) {
                if (c1[j] != c0[i + j] || s[j] != t[i + j]) {
                    f = false;
                    break;
                }
            }
            if (f) {
                res++;
            }
        }
        System.out.println(res);
    }
}
