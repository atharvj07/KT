#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main()
{
    int n, m, k;
    cin >> n >> m >> k;

    vector<int> armies(m + 1);
    for (int i = 0; i < m + 1; i++)
    {
        cin >> armies[i];
    }

    int ans = 0;
    for (int i = 0; i < m; i++)
    {
        int diffBits = __builtin_popcount(armies[i] ^ armies[m]);
        if (diffBits <= k)
        {
            ans++;
        }
    }

    cout << ans << endl;

    return 0;
}
