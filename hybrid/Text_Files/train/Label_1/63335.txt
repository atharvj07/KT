t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    # Calculate the number of elements in the descending part of sequence a
    dec = n - k
    # First, generate the increasing part of p
    p = list(range(1, k - dec))
    # Then, generate the decreasing part of p, to make b lexicographically maximum
    p.extend(range(k, k - dec - 1, -1))
    print(*p)
