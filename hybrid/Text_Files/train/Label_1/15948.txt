#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(k);
  for (int i = 0; i < k; i++) { cin >> a[i]; }
  int p = 0, q = 1;
  while (p < q) {
    int x = a[p % k];
    a[p % k] = a[q % k];
    a[q++ % k] = x;
    if (p++ % n == 0) q -= n;
  }
  for (int i = 0; i < n * k; i++) cout << a[i] << ' ';
  cout << endl;
}
