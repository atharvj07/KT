#include <algorithm>
#include <array>
#include <bitset>
#include <cmath>
#include <cstdio>
#include <vector>
using namespace std;
typedef long long ll;
template < typename T > void Read(T &x) {
  char c;
  while (!isdigit(c = getchar()))
    ;
  for (x = 0; isdigit(c); c = getchar())
    x = x * 10 + c - '0';
}
int main() {
  ll x;
  Read(x);
  int t;
  vector< int > a;
  if (x == 1) {
    t = 0;
  } else {
    ll y = x;
    int c = 0;
    while ((y = (y + 1) / 2) > 1)
      if (y % 2ll)
        ++c;
    t = c * 2;
    if (x != 3)
      a.push_back(0);
    for (int i = 1; i <= t / 2; ++i)
      if ((y = (y + 1) / 2) % 2ll)
        a.push_back(i);
  }
  printf("%d\n", t);
  for (int i = 0; i < t / 2; ++i)
    printf("%d ", a[i]);
  printf("\n");
  return 0;
}
