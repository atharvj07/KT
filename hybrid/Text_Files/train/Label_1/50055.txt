import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int a = input.nextInt();
    int b = input.nextInt();
    int h = input.nextInt();
    int w = input.nextInt();
    int n = input.nextInt();
    int[] ai = new int[n];
    for (int i = 0; i < n; i++)
      ai[i] = input.nextInt();
    Arrays.sort(ai);
    int ans = -1;
    for (int i = 0; i < n; i++) {
      if (h * ai[i] >= a && w >= b || h >= a && w * ai[i] >= b) {
        ans = i + 1;
        break;
      }
    }
    if (h >= a && w >= b)
      ans = 0;
    System.out.println(ans);
  }
}
