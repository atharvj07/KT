import java.util.*;

public class TrailWalk {

    static class Edge {
        int u, v;

        Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static class Graph {
        int n;
        List<List<Integer>> adj;

        Graph(int n) {
            this.n = n;
            adj = new ArrayList<>();
            for (int i = 0; i <= n; i++) {
                adj.add(new ArrayList<>());
            }
        }

        void addEdge(int u, int v) {
            adj.get(u).add(v);
            adj.get(v).add(u);
        }

        boolean isEulerian() {
            int oddDegreeCount = 0;
            for (int i = 1; i <= n; i++) {
                if (adj.get(i).size() % 2 != 0) {
                    oddDegreeCount++;
                }
            }
            return oddDegreeCount == 0 || oddDegreeCount == 2;
        }

        List<Edge> findEulerianCircuit() {
            if (!isEulerian()) {
                return null;
            }

            List<Edge> circuit = new ArrayList<>();
            Stack<Integer> stack = new Stack<>();
            stack.push(1);

            while (!stack.isEmpty()) {
                int u = stack.peek();
                if (adj.get(u).isEmpty()) {
                    circuit.add(new Edge(stack.pop(), u));
                } else {
                    int v = adj.get(u).remove(0);
                    stack.push(v);
                }
            }

            Collections.reverse(circuit);
            return circuit;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        Graph graph = new Graph(n);
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            graph.addEdge(u, v);
        }

        List<Edge> circuit = graph.findEulerianCircuit();
        if (circuit == null) {
            System.out.println(n - graph.adj.stream().mapToInt(List::size).sum() / 2);
        } else {
            System.out.println(0);
        }
    }
}
