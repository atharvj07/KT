#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
  ll n, x1, y1, x2, y2;
  cin >> n >> x1 >> y1 >> x2 >> y2;
  vector<ll> x(n), y(n);
  for (auto &i : x) cin >> i;
  for (auto &i : y) cin >> i;
  vector<pair<ll, ll>> pt{
      {x1, y1}, {x2, y2},
  };
  for (ll i = 0; i < n; i++) { pt.push_back({x[i], y[i]}); }
  ll ans = 0;
  for (ll i = 0; i < 2; i++) {
    for (ll j = i + 1; j < 2; j++) {
      ll sum = 0;
      for (ll k = 2; k < pt.size(); k++) {
        sum += min({
            (pt[i].first - pt[k].first) *
                (pt[i].first - pt[k].first) +
                (pt[i].second - pt[k].second) *
                    (pt[i].second - pt[k].second),
            (pt[j].first - pt[k].first) *
                (pt[j].first - pt[k].first) +
                (pt[j].second - pt[k].second) *
                    (pt[j].second - pt[k].second),
        });
      }
      ans = max(ans, sum);
    }
  }
  cout << ans << endl;
}
