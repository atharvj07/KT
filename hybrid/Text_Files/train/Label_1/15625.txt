#include <iostream>
#include <vector>

using namespace std;

bool f(int n, vector<long long> &v) {
  if (n <= 0) {
    return false;
  }
  bool flag = false;
  for (int i = 1;; ++i) {
    if (n >= 1LL * i * (i - 1) / 2) {
      flag = true;
      long long rem = n - 1LL * i * (i - 1) / 2;
      if (rem % i == 0) {
        v.push_back(i);
      }
    } else {
      break;
    }
  }
  return flag;
}
int main() {
  long long n;
  cin >> n;
  vector<long long> v;
  if (f(n, v)) {
    for (int i = 0; i < v.size(); ++i) {
      cout << 1LL * v[i] + 2LL * v[i] / 2 << endl;
    }
  } else {
    cout << "-1" << endl;
  }
  return 0;
}
