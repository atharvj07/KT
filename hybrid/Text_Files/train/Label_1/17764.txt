#include <bits/stdc++.h>
using namespace std;

int32_t main() {
  int n, k;
  cin >> n >> k;
  vector<string> s(n);
  for (int i = 0; i < n; i++) cin >> s[i];
  int ans = 0;
  for (int i = 0; i < n - 2; i++)
    for (int j = i + 1; j < n - 1; j++)
      for (int l = j + 1; l < n; l++) {
        int same = 0, diff = 0;
        for (int t = 0; t < k; t++) {
          if (s[i][t] == s[j][t] && s[j][t] == s[l][t]) same++;
          if (s[i][t] != s[j][t] && s[i][t] != s[l][t] && s[j][t] != s[l][t]) diff++;
        }
        if (same == k || diff == k) ans++;
      }
  cout << ans;
  return 0;
}
