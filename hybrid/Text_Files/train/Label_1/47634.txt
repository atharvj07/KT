import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int h = sc.nextInt();
        int d = sc.nextInt();
        System.out.println(modPow(2, n * (h / d) + n - 1, 1000000007));
    }

    private static long modPow(long n, long p, int mod) {
        long ret = 1;
        while (p > 0) {
            if ((p & 1) == 1) {
                ret = (ret * n) % mod;
            }
            n = (n * n) % mod;
            p >>= 1;
        }
        return ret;
    }
}
