#include <bits/stdc++.h>
using namespace std;
const int N = 200002;
int main() {
    int n, b;
    cin >> n >> b;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    deque<int> dq;
    vector<int> ans(n);
    for (int i = 0, last = 0; i < n; i++) {
        while (!dq.empty() && dq.front() <= last) dq.pop_front();
        if (dq.size() < b) {
            ans[i] = v[i].first + v[i].second;
            dq.push_back(ans[i]);
        } else {
            ans[i] = -1;
        }
        last = max(last, v[i].first);
        if (!dq.empty() && dq.front() <= last) {
            last = dq.front();
            dq.pop_front();
        }
    }
    for (int i = 0; i < n; i++) {
        cout << ans[i] << endl;
    }
}
