import math

x1, y1, x2, y2 = map(int, input().split())
v, t = map(int, input().split())
v1, v2 = map(int, input().split())
w1, w2 = map(int, input().split())

def dist(x, y):
    return math.sqrt(x * x + y * y)

ans = dist(x2 - x1, y2 - y1) / v
for d in [v1, v2, w1, w2]:
    if d > 0:
        ans = min(ans, dist(d * t - (x2 - x1), d * t - (y2 - y1)) / v)
    elif d < 0:
        ans = min(ans, dist(-d * (t - 1000) - (x2 - x1), -d * (t - 1000) - (y2 - y1)) / v)

print(ans)
