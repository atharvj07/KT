#include <bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    if (v[n - 1] < 15)
        cout << "UP" << endl;
    else if (v[n - 1] > 0)
        cout << "DOWN" << endl;
    else
        cout << -1 << endl;
    return 0;
}
