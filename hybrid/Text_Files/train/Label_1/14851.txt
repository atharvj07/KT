#include <bits/stdc++.h>
using namespace std;
int main() {
  int a, b, c, d;
  cin >> a >> b >> c;
  vector<int> q(a + b + c + 1);
  for (int i = 0; i < a; i++) {
    cin >> d;
    q[d] = 1;
  }
  int ans = 0;
  for (int i = 0; i < b; i++) {
    cin >> d;
    q[d] = 2;
    ans += q[d] != q[d - 1];
  }
  for (int i = 0; i < c; i++) {
    cin >> d;
    q[d] = 3;
    ans += q[d] != q[d - 1];
  }
  cout << ans;
}
