#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  long long n;
  std::cin >> n;
  std::vector<long long> x(n);
  for (auto &xi : x) std::cin >> xi;
  std::sort(x.begin(), x.end());
  long long ans = 0;
  for (auto &xi : x) ans += std::abs(xi - x[n / 2]);
  std::cout << x[n / 2] << std::endl;
  return 0;
}
