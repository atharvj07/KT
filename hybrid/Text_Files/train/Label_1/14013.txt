#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
int MinSplit(ll n, ll s, ll l, ll* arr) {
  bool ok = true;
  for (ll i = 0; i + l - 1 < n; i++) {
    ok &= (arr[i + l - 1] - arr[i] <= s);
  }
  if (!ok) return -1;
  return (n + l - 1) / l;
}
int main() {
  ll n, s, l;
  cin >> n >> s >> l;
  ll arr[n];
  for (auto& it : arr) cin >> it;
  sort(arr, arr + n);
  cout << MinSplit(n, s, l, arr) << endl;
  return 0;
}
