#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> div;
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            div.push_back(x);
        }
        sort(div.begin(), div.end());
        int ans = 1;
        for (int i = 0; i < n; i++) {
            ans = ans * div[i] / __gcd(ans, div[i]);
        }
        if (ans > 1)
            cout << ans << endl;
        else
            cout << -1 << endl;
    }
    return 0;
}
