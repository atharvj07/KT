#include<bits/stdc++.h>
using namespace std;
int main(){
    int n; cin >> n;
    vector<int> a(n);
    for(int i=0; i<n; i++) cin >> a[i];
    long long odd_prefix = 0, even_prefix = 0, ans = LLONG_MIN;
    for(int i=0; i<n; i++){
        if(i&1){
            odd_prefix += a[i];
            ans = max(ans, odd_prefix);
        }
        else{
            even_prefix += a[i];
            odd_prefix = max(odd_prefix, even_prefix + a[i]);
        }
    }
    cout << ans << endl;
}
