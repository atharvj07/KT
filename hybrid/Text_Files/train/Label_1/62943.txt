t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    if sum(a) != sum(b):
        print(-1)
        continue
    
    operations = []
    for i in range(n):
        while a[i] > b[i]:
            for j in range(n):
                if a[j] < b[j]:
                    diff = min(a[i] - b[i], b[j] - a[j])
                    a[i] -= diff
                    a[j] += diff
                    for _ in range(diff):
                        operations.append((i + 1, j + 1))
                    break
    
    if a != b:
        print(-1)
    else:
        print(len(operations))
        for op in operations:
            print(op[0], op[1])
