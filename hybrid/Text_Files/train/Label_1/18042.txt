#include<bits/stdc++.h>
using namespace std;
const int maxN = 105;
int main() {
    int n; cin>>n;
    int a[n][maxN], t[maxN]={};
    for(int i=0; i<n; i++) {
        int m; cin>>m;
        for(int j=0; j<m; j++) {
            cin>>a[i][j];
            t[a[i][j]]=1;
        }
    }
    for(int i=1; i<=100; i++) {
        if(t[i]) cout<<i<<" ";
    }
}
