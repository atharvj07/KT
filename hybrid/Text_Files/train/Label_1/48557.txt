import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[12];
        for (int i = 0; i < 12; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        boolean ok = true;
        for (int i = 0; i < 4; i++) {
            if (arr[i] != arr[4]) ok = false;
        }
        for (int i = 4; i < 8; i++) {
            if (arr[i] != arr[8]) ok = false;
        }
        System.out.println(ok ? "yes" : "no");
    }
}
