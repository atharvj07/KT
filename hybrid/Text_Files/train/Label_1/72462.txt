n = int(input())
g = [[-1, -1] for i in range(n + 1)]
for i in range(1, n + 1):
    l, r = map(int, input().split())
    g[i][0] = l
    g[i][1] = r
t = -1
h = -1
for i in range(1, n + 1):
    if g[i][0] == -1:
        t = i
    if g[i][1] == -1:
        h = i
if h != -1:
    g[h][1] = t
    g[t][0] = h
for i in range(1, n + 1):
    print(*g[i])
