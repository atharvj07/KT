#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string a, s;
    cin >> a >> s;
    int n = max(a.length(), s.length());
    string b(n, '0');
    int carry = 0;
    for (int i = n - 1; i >= 0; i--) {
      int da = i < a.length() ? a[i] - '0' : 0;
      int db = i < b.length() ? b[i] - '0' : 0;
      int ds = s[i] - '0';
      int sum = da + db + carry;
      b[i] = '0' + sum % 10;
      carry = sum / 10;
    }
    if (carry) {
      cout << -1 << endl;
      continue;
    }
    bool leading = true;
    for (int i = 0; i < n; i++) {
      if (b[i] != '0') leading = false;
      if (!leading) cout << b[i];
    }
    if (leading) cout << -1;
    cout << endl;
  }
  return 0;
}
