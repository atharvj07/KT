#include <cstdio>
#include <vector>
#include <algorithm>

int main() {
    int n, m;
    scanf("%d %d", &n, &m);
    std::vector<std::pair<int, double>> plants;
    for (int i = 0; i < n; ++i) {
        int s; double x;
        scanf("%d %lf", &s, &x);
        plants.emplace_back(s, x);
    }
    int ans = 0;
    for (int i = 0; i < n - 1; ++i) {
        if (plants[i].first != plants[i + 1].first) {
            ++ans;
        }
    }
    printf("%d\n", ans);
    return 0;
}
