t = int(input().strip())
for _ in range(t):
    a1, b1 = map(int, input().strip().split())
    a2, b2 = map(int, input().strip().split())
    if a1 == a2 and b1 + b2 == a1:
        print("Yes")
    elif a1 == b2 and b1 + a2 == a1:
        print("Yes")
    elif b1 == a2 and a1 + b2 == b1:
        print("Yes")
    elif b1 == b2 and a1 + a2 == b1:
        print("Yes")
    else:
        print("No")
