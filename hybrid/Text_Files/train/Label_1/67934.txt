n, K = map(int, input().split())
a = list(map(int, input().split()))

a.sort()
remaining = [True] * n

for i in range(n):
    if remaining[i]:
        for j in range(i + 1, n):
            if a[j] > a[i] and a[j] <= a[i] + K:
                remaining[j] = False
            elif a[j] > a[i] + K:
                break

print(sum(remaining))
