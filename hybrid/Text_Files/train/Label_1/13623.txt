#include<bits/stdc++.h>
using namespace std;
#define pb push_back
#define eb emplace_back
#define fi first
#define se second
#define all(x) (x).begin(),(x).end()
typedef pair<int,int> pii;
typedef long long ll;
typedef vector<int> vi;
const int inf=2e9+11;
const int mod=1e9+7;
void solve(){
	int n,q;
	cin>>n>>q;
	vector<ll>a(n);
	for(int i=0;i<n;i++) cin>>a[i];
	for(int i=0;i<q;i++){
		ll x;
		cin>>x;
		ll mi=*min_element(all(a)),ma=*max_element(all(a)),mid=(ma+mi)/2;
		if((x-mi)*(ma-mid)>(ma-mi)*(mid-x)) puts("No");
		else puts("Yes");
	}
}
int main(){
	ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
	int t;cin>>t;while(t--) solve();
	return 0;
}
