n,m,k=map(int,input().split())
a=[list(map(int,input().split())) for _ in range(n)]
ans,cnt=0,0
for j in range(m):
    i,mn=n+1,0
    for x in range(n):
        if a[x][j]==1:
            i=x
            break
    if i==n+1:continue
    cnt+=i-1
    ans+=min(k,n-i+1)
    for x in range(i,min(i+k,n)):
        if a[x][j]==0:
            cnt-=1
            a[x][j]=1
print(ans,cnt)
