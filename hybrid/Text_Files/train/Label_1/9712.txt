#include <bits/stdc++.h>
using namespace std;
int a[300005], b[300005];
int32_t main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    int bal = 0;
    for (int j = 0; j < s.size(); j++) {
      bal += s[j] == '(' ? +1 : -1;
      a[i] -= bal;
      b[i] += bal;
    }
  }
  map<int, int> freq;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    ans += freq[b[i]];
    freq[a[i]]++;
  }
  cout << ans << endl;
}
