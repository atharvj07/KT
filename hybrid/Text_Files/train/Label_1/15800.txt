#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n,s=0;
        cin>>n;
        vector<int>b(n+2);
        for(int i=0;i<n+2;i++)cin>>b[i],s+=b[i];
        if(s%2)cout<<"-1\n";
        else{
            int x=s/2;
            for(int i=n;i>=0;i--){
                if(x-b[i]>=0)x-=b[i],b[i]=0;
                else b[i]-=x,x=0;
            }
            if(x==0){
                for(int i=0;i<n;i++)cout<<b[i]<<" ";
                cout<<"\n";
            }
            else cout<<"-1\n";
        }
    }
    return 0;
}
