#include <iostream>
#include <string>

using namespace std;

bool is_leap_year(int year) {
  return (year % 4 == 0) && ((year % 100 != 0) || (year % 400 == 0));
}

int main() {
  string s;
  cin >> s;
  int x;
  if (s.back() == 'k') {
    x = stoi(s.substr(0, s.size() - 2));
    cout << (is_leap_year(2016) ? 53 : 52) << endl;
  } else {
    x = stoi(s.substr(0, s.size() - 5));
    int days = (is_leap_year(2016) ? 366 : 365);
    if (x == 29 || (x == 30 && days == 365))
      cout << 0 << endl;
    else
      cout << days / x << endl;
  }
  return 0;
}
