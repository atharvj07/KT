import java.util.*;
class Main {
    static final long mod = 998244353;
    public static void main(String[] arg) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        int[] idx = new int[n + 1];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
            if (p[i] != -1) {
                idx[p[i]] = i;
            }
        }
        long[] npref = new long[n + 1];
        long[] nsuff = new long[n + 1];
        for (int i = n - 1; i >= 0; i--) {
            if (p[i] == -1) {
                nsuff[i] = nsuff[i + 1] + (i + 1);
            } else {
                nsuff[i] = nsuff[i + 1];
            }
        }
        for (int i = 0; i < n; i++) {
            if (p[i] == -1) {
                npref[i + 1] = npref[i] + (n - i);
            } else {
                npref[i + 1] = npref[i];
            }
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (p[i] != -1) {
                int cur = p[i];
                ans += cur * (npref[idx[cur]] + nsuff[idx[cur] + 1] - (idx[cur] + 1));
                ans %= mod;
            }
        }
        System.out.println(ans);
    }
}
