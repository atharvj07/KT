import sys

for _ in range(int(input())):
    n, d = map(int, input().split())
    a = list(map(int, input().split()))
    b = [0] * n
    cycles = 0
    while a != b:
        cycles += 1
        for i in range(n):
            b[i] = a[i] & a[(i - d % n + n) % n]
        a = b.copy()
    print(-1 if 1 in a else cycles)
