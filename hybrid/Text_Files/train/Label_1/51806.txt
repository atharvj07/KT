import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class Queue {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] people = new int[n][2];
        for (int i = 0; i < n; i++) {
            people[i][0] = scanner.nextInt();
            people[i][1] = scanner.nextInt();
        }
        scanner.close();

        Arrays.sort(people, Comparator.comparingInt(a -> a[0]));

        int[] queue = new int[n];
        int currentPosition = 0;
        for (int i = 0; i < n; i++) {
            int importance = people[i][0];
            int moves = people[i][1];
            while (currentPosition > 0 && importance < queue[currentPosition - 1] && moves > 0) {
                queue[currentPosition] = queue[currentPosition - 1];
                currentPosition--;
                moves--;
            }
            queue[currentPosition] = importance;
            currentPosition++;
        }

        for (int i = 0; i < n; i++) {
            System.out.print(queue[i] + " ");
        }
    }
}
