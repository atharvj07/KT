n=int(input())
a=[int(x) for x in input().split()]
has=[0]*n
cool=[0]*n
mn=-1
for i in range(n):
    if has[a[i]-1]==0:
        has[a[i]-1]=i+1
        if mn==-1:
            mn=i
        cool[i]=min(cool[mn],a[i])
    else:
        mn=has[a[i]-1]-1
        cool[i]=min(cool[mn],a[i])
ans=[0]*n
for i in range(n):
    if cool[i]<=a[i]/2:
        ans[i]=-1
    else:
        mn=-1
        for j in range(i,n):
            if mn==-1:
                mn=j
            if cool[mn]<=a[j]/2:
                mn=j+1
                break
            if j==n-1:
                mn=n
        ans[i]=mn-i
print(*ans)
