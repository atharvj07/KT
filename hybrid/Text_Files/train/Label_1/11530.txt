#include <iostream>
#include <set>

using namespace std;

int main() {
  long long n, k;
  cin >> n >> k;
  if (n % k == 0) {
    cout << "No" << endl;
    return 0;
  }
  set<long long> remainders;
  for (long long i = 1; i <= k; i++) { remainders.insert(n % i); }
  if (remainders.size() == k) {
    cout << "Yes" << endl;
  } else {
    cout << "No" << endl;
  }
  return 0;
}
