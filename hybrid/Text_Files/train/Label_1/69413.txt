import sys

input = sys.stdin.readline

m, n, k, t = map(int, input().split())

a = list(map(int, input().split()))

l = []
r = []
d = []

for i in range(k):
    x, y, z = map(int, input().split())
    l.append(x)
    r.append(y)
    d.append(z)

a.sort()

ans = 0

for i in range(m):
    if a[i] >= d[0]:
        ans += 1
        t -= n - l[0]
        t -= l[0]
        l.pop(0)
        r.pop(0)
        d.pop(0)
    else:
        break

for i in range(len(l)):
    if t >= r[i] - l[i]:
        t -= r[i] - l[i]
        l.pop(i)
        r.pop(i)
        d.pop(i)
        ans += 1
    else:
        break

print(ans)
