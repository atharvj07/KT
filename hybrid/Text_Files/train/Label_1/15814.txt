#include <iostream>
#include <map>
#include <vector>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            cin >> a[i];
        }
        map<int, int> cnt;
        int l = -1, r = -1;
        int len = n + 1;
        int mx = 0;
        for (int i = 0, j = 0; j < n; ++j) {
            ++cnt[a[j]];
            if (cnt[a[j]] > mx) {
                mx = cnt[a[j]];
                l = i;
                r = j;
                len = j - i + 1;
            } else if (cnt[a[j]] == mx) {
                if (j - i + 1 < len) {
                    l = i;
                    r = j;
                    len = j - i + 1;
                }
            }
            while (i <= j && cnt[a[j]] > mx) {
                --cnt[a[i]];
                ++i;
            }
        }
        if (len > n || l == -1) {
            cout << -1 << endl;
        } else {
            cout << len << endl;
        }
    }
}
