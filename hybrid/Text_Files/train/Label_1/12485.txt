#include <iostream>

using namespace std;

int main()
{
    int h1, h2, a, b, hours;
    cin >> h1 >> h2 >> a >> b;
    hours = (h2 - h1 + b - 1) / (a + b);
    if((h2 - h1) % (a + b) == 0)
        hours--;
    if(hours < 0)
        cout << -1 << endl;
    else
        cout << hours << endl;
    return 0;
}
