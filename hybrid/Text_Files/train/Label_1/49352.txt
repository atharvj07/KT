import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int[][] players = new int[4][2];
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 2; j++) {
                players[i][j] = input.nextInt();
            }
        }
        int[] team1 = new int[2];
        int[] team2 = new int[2];
        for (int i = 0; i < 2; i++) {
            team1[i] = Math.max(players[i][0], players[i + 2][0]);
            team2[i] = Math.max(players[i + 2][1], players[i][1]);
        }
        if (team1[0] > team2[1] && team1[1] > team2[0]) {
            System.out.println("Team 1");
        } else if (team2[0] > team1[1] && team2[1] > team1[0]) {
            System.out.println("Team 2");
        } else {
            System.out.println("Draw");
        }
    }
}
