#include <iostream>
#include <cstdio>
#include <algorithm>
using namespace std;

int main() {
    int n, ans = 1;
    cin >> n;
    for (int i = 29; i >= 0; i--) {
        if (n % (1 << (i + 1)) == 0 && (1 << i) + (1 << (i + 1)) - 1 == n / (1 << (i + 1))) {
            ans = n / (1 << (i + 1));
        }
    }
    cout << ans << endl;
    return 0;
}
