import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
	private static final int MAX_FLAGS = 64;
	private static long state = 0; // Represent state by decimal integer.

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

		// Read the number of queries.
		int q = Integer.parseInt(br.readLine());

		// Handle each query.
		for (int i = 0; i < q; i++) {
			String[] query = br.readLine().split(" ");
			int op = Integer.parseInt(query[0]);
			int arg = 0;
			if (query.length > 1) {
				arg = Integer.parseInt(query[1]);
			}
			
			switch (op) {
			case 0:
				System.out.println(test(arg));
				break;
			case 1:
				set(arg);
				break;
			case 2:
				clear(arg);
				break;
			case 3:
				flip(arg);
				break;
			case 4:
				System.out.println(all() ? 1 : 0);
				break;
			case 5:
				System.out.println(any() ? 1 : 0);
				break;
			case 6:
				System.out.println(none() ? 1 : 0);
				break;
			case 7:
				System.out.println(count());
				break;
			case 8:
				System.out.println(val());
				break;
			}
		}
	}
	
	// Returns 1 if the i-th flag is ON, otherwise 0.
	private static int test(int i) {
		return (state & (1 << i)) > 0 ? 1 : 0;
	}

	// Set the i-th flag to ON.
	private static void set(int i) {
		state |= (1 << i);
	}

	// Set the i-th flag to OFF.
	private static void clear(int i) {
		state &= ~(1 << i);
	}

	// Inverse the i-th flag.
	private static void flip(int i) {
		state ^= (1 << i);
	}

	// Returns 1 if all flags are ON, otherwise 0.
	private static boolean all() {
		return state == (1 << MAX_FLAGS) - 1;
	}

	// Returns 1 if at least one flag is ON, otherwise 0.
	private static boolean any() {
		return state > 0;
	}

	// Returns 1 if all flags are OFF, otherwise 0.
	private static boolean none() {
		return state == 0;
	}

	// Returns the number of ON flags.
	private static int count() {
		int count = 0;
		
		for (int i = 0; i < MAX_FLAGS; i++) {
			count += test(i);
		}
		
		return count;
	}
	
	// Returns the decimal value of the state.
	private static long val() {
		return state;
	}
}
