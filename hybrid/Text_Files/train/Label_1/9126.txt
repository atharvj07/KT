#include <bits/stdc++.h>
using namespace std;

const int N = 1e3 + 10;

int n, s, m, k;
int a[N], f[N];
vector<pair<int, int>> seg[N];
int mi[N][N];

int read() {
    int ret = 0, f = 1;
    char ch = getchar();
    while ((ch < '0' || ch > '9') && ch != '-') ch = getchar();
    if (ch == '-') f = -1, ch = getchar();
    while (ch >= '0' && ch <= '9') ret = ret * 10 + ch - '0', ch = getchar();
    return ret * f;
}

void solve() {
    n = read(), s = read(), m = read(), k = read();
    for (int i = 1; i <= n; ++i) a[i] = read();
    for (int i = 1; i <= s; ++i) {
        int l = read(), r = read();
        seg[l].emplace_back(r, i);
    }
    for (int i = 1; i <= n; ++i) {
        for (auto [r, id] : seg[i]) { f[id] = max(f[id], f[r] + 1); }
    }
    for (int i = 1; i <= s; ++i) {
        for (int j = 1; j <= s; ++j) { mi[i][j] = 1e9; }
        mi[i][i] = f[i];
    }
    for (int l = 1; l < s; ++l) {
        for (int i = 1; i + l <= s; ++i) {
            int j = i + l;
            for (int k = i; k < j; ++k) { mi[i][j] = min(mi[i][j], mi[i][k] + mi[k + 1][j]); }
        }
    }
    int ans = 1e9;
    for (int i = 1; i + m - 1 <= s; ++i) { ans = min(ans, mi[i][i + m - 1]); }
    if (ans == 1e9)
        ans = -1;
    else {
        ans = n + 1;
        for (int i = 1; i + m - 1 <= s; ++i) {
            int j = i + m - 1;
            int val = 0;
            for (int k = i; k <= j; ++k) val += f[k];
            ans = min(ans, a[val]);
        }
    }
    cout << ans << endl;
}

int main() {
    solve();
    return 0;
}
