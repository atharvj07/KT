#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;
int cnt[N], a[N];
vector<tuple<int, int, int>> ans;

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        cnt[a[i]]++;
    }
    int x = 0;
    for (int i = 0; i < N; i++) {
        if (cnt[i] % 3) {
            x = i;
            break;
        }
    }
    if (x == 0) {
        cout << "NO" << '\n';
        return 0;
    }
    cout << "YES" << '\n';
    int cnt0 = cnt[x] / 3, cnt1 = 0, cnt2 = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == x) {
            if (cnt0) {
                cnt0--;
                ans.emplace_back(i + 1, i + 1, i + 1);
            } else if (cnt1) {
                cnt1--, cnt2++;
                ans.emplace_back(i + 1, i + 1, i + 2);
            } else {
                cnt1++, cnt2--;
                ans.emplace_back(i + 1, i + 2, i + 2);
            }
        } else if (a[i] == x ^ 1) {
            if (cnt2) {
                cnt2--;
                ans.emplace_back(i + 1, i + 1, i + 2);
            } else {
                cout << "-1\n";
                return 0;
            }
        } else {
            if (cnt1) {
                cnt1--;
                ans.emplace_back(i + 1, i + 1, i + 3);
            } else {
                cout << "-1\n";
                return 0;
            }
        }
    }
    cout << ans.size() << '\n';
    for (auto [i, j, k] : ans) {
        cout << i << ' ' << j << ' ' << k << '\n';
    }
    return 0;
}
