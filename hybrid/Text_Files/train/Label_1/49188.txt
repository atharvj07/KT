import java.util.Arrays;
import java.util.Scanner;

class Main {

    static int solve(int[] a, int[] l, int[] r) {
        int ans = 0, n = a.length;
        for (int i = 0; i < n; i++) {
            l[i] = a[i] == 0 || i == 0 ? 0 : l[i - 1] - 1;
            r[i] = a[i] == 2 || i == n - 1 ? 0 : r[i + 1] - 1;
            if (l[i] <= 0 && r[i] <= 0) ans++;
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt();
        int[] a = new int[n], l = new int[n], r = new int[n];
        Arrays.fill(l, n);
        Arrays.fill(r, n);
        String s = sc.next();
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == 'R') a[i] = 0;
            if (s.charAt(i) == 'P') a[i] = 1;
            if (s.charAt(i) == 'S') a[i] = 2;
        }
        System.out.println(solve(a, l, r));
        for (int i = 0; i < q; i++) {
            int p = sc.nextInt() - 1;
            char c = sc.next().charAt(0);
            if (c == 'R') a[p] = 0;
            if (c == 'P') a[p] = 1;
            if (c == 'S') a[p] = 2;
            System.out.println(solve(a, l, r));
        }
        sc.close();
    }
}
