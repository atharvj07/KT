#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, k, cnt[]{0, 0, 0}, ans = 0, x;
    cin >> n >> k;
    while (n--) {
        cin >> x;
        ++cnt[x % k];
    }
    cout << min({cnt[0], cnt[k / 2] * (k % 2 == 0) / 2,
                 min(cnt[k / 2], cnt[k] / 2) * (k % 2 != 0)}) + min(cnt[0] % 2, cnt[k])
         << '\n';
}
