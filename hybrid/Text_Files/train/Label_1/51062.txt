import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        in.nextLine();
        String[] tokens = in.nextLine().split(" ");
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(tokens[i]);
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int k = n - i - j - 1;
                if (k < 0 || k >= n || a[i] == 0 || a[j] == 0 || a[k] == 0) {
                    continue;
                }
                a[i]--;
                a[j]--;
                a[k]--;
                ans++;
            }
        }
        System.out.println(ans);
    }
}
