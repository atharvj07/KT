for _ in range(int(input())):
    n = int(input())
    L = list(map(int, input().split()))
    a = [x % 2 for x in L]
    ans = -1
    for i in range(n - 1):
        if a[i] == a[i + 1]:
            if ans == -1:
                ans = i
            else:
                ans = -1
                break
    print(ans)
