import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] prime = new int[100001];
        for (int i = 2; i <= 100000; i++) {
            if (prime[i] == 0) {
                for (int j = i * 2; j <= 100000; j += i) {
                    prime[j] = 1;
                }
            }
        }
        List<List<Integer>> tables = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (prime[a[i] + a[j]] == 0) {
                    List<Integer> table = new ArrayList<>();
                    table.add(i + 1);
                    table.add(j + 1);
                    tables.add(table);
                }
            }
        }
        if (tables.size() < n / 3) {
            System.out.println("Impossible");
            return;
        }
        System.out.println(tables.size());
        for (List<Integer> table : tables) {
            System.out.print(table.size() + " ");
            for (Integer fox : table) {
                System.out.print(fox + " ");
            }
            System.out.println();
        }
    }
}
