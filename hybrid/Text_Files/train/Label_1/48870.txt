import java.util.*;

class Main {
	final static int N = 27;
	static char[][] ans = new char[2][N];
	static boolean[] taken = new boolean[N + 1];
	static int x = 0;
	static int y = 0;
	static boolean isValid(int dx, int dy) {
		return dx >= 0 && dx < 2 && dy >= 0 && dy < N && !taken[ans[dx][dy] - 'A' + 1];
	}
	static boolean dfs(String s) {
		if (s.length() == 0)
			return true;
		for (int i = 0; i < 2; i++) {
			for (int j = 0; j < N; j++) {
				if (taken[ans[i][j] - 'A' + 1] || ans[i][j] != s.charAt(0))
					continue;
				if (i + 1 == x && j == y || x + 1 == i && j == y)
					continue;
				int dx = i + (i - x);
				int dy = j + (j - y);
				if (!isValid(i, j) || !isValid(dx, dy))
					continue;
				x = i;
				y = j;
				taken[ans[i][j] - 'A' + 1] = true;
				ans[i][j] = ' ';
				ans[dx][dy] = s.charAt(0);
				if (dfs(s.substring(1)))
					return true;
				taken[ans[i][j] - 'A' + 1] = false;
				ans[i][j] = s.charAt(0);
				ans[dx][dy] = ' ';
			}
		}
		return false;
	}
	static void solve() {
		Scanner input = new Scanner(System.in);
		String s = input.next();
		for (int i = 0; i < 2; i++)
			Arrays.fill(ans[i], ' ');
		int[] frq = new int[N + 1];
		for (char c : s.toCharArray()) {
			frq[c - 'A' + 1]++;
			if (frq[c - 'A' + 1] > 2) {
				System.out.println("Impossible");
				return;
			}
		}
		boolean flag = false;
		if (frq['A' - 'A' + 1] >= 1) {
			flag = true;
			x = 0;
			y = 0;
			ans[x][y] = 'A';
			taken[1] = true;
			frq['A' - 'A' + 1]--;
		}
		if (frq['B' - 'A' + 1] >= 1) {
			flag = true;
			x = 1;
			y = 0;
			ans[x][y] = 'B';
			taken[2] = true;
			frq['B' - 'A' + 1]--;
		}
		if (frq['C' - 'A' + 1] >= 1) {
			flag = true;
			x = 1;
			y = N - 1;
			ans[x][y] = 'C';
			taken[3] = true;
			frq['C' - 'A' + 1]--;
		}
		int extra = 0;
		for (int i = 1; i <= N; i++) {
			extra += frq[i];
		}
		if (extra > 18 || extra % 2 == 1 || !flag) {
			System.out.println("Impossible");
			return;
		}
		int idx = 0;
		for (int i = 1; i <= N; i++) {
			while (frq[i] > 0) {
				char c = (char) (i + 'A' - 1);
				int x = -1;
				int y = -1;
				if (frq[i] == 1) {
					if (ans[0][s.indexOf(c) + 1] == ' ' && ans[1][s.indexOf(c)] == ' ' && s.indexOf(c) != y) {
						x = 0;
						y = s.indexOf(c) + 1;
					} else if (ans[1][s.indexOf(c)] == ' ' && ans[1][s.indexOf(c) + 1] == ' ' && s.indexOf(c) != y) {
						x = 1;
						y = s.indexOf(c);
					}
				}
				if (y == -1) {
					while (ans[++idx][y = idx] != ' ');
					x = idx % 2;
				}
				ans[x][y] = c;
				frq[i]--;
			}
		}
		taken[ans[1][1] - 'A' + 1] = false;
		if (dfs(s)) {
			for (int i = 0; i < 2; i++)
				System.out.println(Arrays.toString(ans[i]).replaceAll(" ", ""));
		} else
			System.out.println("Impossible");
	}
	public static void main(String[] args) {
		solve();
	}
}
