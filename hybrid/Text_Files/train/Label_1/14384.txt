#include <iostream>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;

    int a[n];
    for (int i = 0; i < n; ++i) {
        std::cin >> a[i];
    }

    std::sort(a, a + n);

    int day = 0;
    int problems = 0;
    for (int i = n - 1; i >= 0; --i) {
        if (problems + a[i] >= day + 1) {
            ++day;
            problems += a[i];
        } else {
            break;
        }
    }

    std::cout << day << std::endl;

    return 0;
}
