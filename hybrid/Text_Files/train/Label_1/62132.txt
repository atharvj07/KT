MOD = 10**9 + 7

def pow_mod(x, n, mod=MOD):
    result = 1
    while n > 0:
        if n % 2 == 1:
            result = result * x % mod
        x = x * x % mod
        n //= 2
    return result

def solve(k, n, nodes):
    color_pairs = {
        'white': ('white', 'yellow'),
        'yellow': ('white', 'yellow'),
        'green': ('green', 'blue'),
        'blue': ('green', 'blue'),
        'red': ('red', 'orange'),
        'orange': ('red', 'orange')
    }
    total_nodes = 2**k - 1
    edge_count = [0] * (total_nodes + 1)
    for node, color in nodes:
        forbidden = color_pairs[color]
        for forbidden_color in forbidden:
            edge_count[node] |= 1 << ('whiteyellowgreenblueredorange'.index(forbidden_color) // 6)
    
    def dfs(node=1):
        if node > total_nodes:
            return 1, 0
        left_mult, left_free = dfs(node * 2)
        right_mult, right_free = dfs(node * 2 + 1)
        if edge_count[node]:
            free_colors = 4 - bin(edge_count[node]).count('1')
            if node <= total_nodes // 2:
                return (left_mult * right_mult * pow_mod(free_colors, left_free + right_free, MOD)) % MOD, 0
            else:
                return 1, left_free + right_free + 1
        else:
            if node <= total_nodes // 2:
                return (left_mult * right_mult * pow_mod(4, left_free + right_free, MOD)) % MOD, 0
            else:
                return 1, left_free + right_free + 1

    mult, free = dfs()
    return mult * pow_mod(4, free, MOD) % MOD

k = int(input().strip())
n = int(input().strip())
nodes = []
for _ in range(n):
    v, s = input().strip().split()
    nodes.append((int(v), s))

print(solve(k, n, nodes))
