#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);
    int n, k;
    cin >> n >> k;
    vector<int32_t> a(n);
    for (int32_t i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int32_t ans = 0;
    for (int32_t i = 0; i < n && k < a[i] / 2; i++) ans++;
    cout << ans;
    return 0;
}
