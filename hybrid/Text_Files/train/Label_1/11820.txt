#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        sort(a.rbegin(), a.rend());
        int ans = 0;
        for (int i = 1; i < n; i++) {
            ans += max(0, a[i - 1] - a[i]);
        }
        cout << ans << endl;
    }
    return 0;
}
