import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[][] ab = new long[n][2];
        for (int i = 0; i < n; i++) {
            ab[i][0] = sc.nextInt();
            ab[i][1] = sc.nextInt();
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            long x = (ab[i][0] - 1 + ab[i][1] - 1) / ab[i][1];
            ans += x;
            for (int j = i + 1; j < n; j++) {
                ab[j][0] += x * ab[i][1];
            }
        }
        System.out.println(ans);
    }
}
