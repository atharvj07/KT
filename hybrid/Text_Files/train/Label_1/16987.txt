#include <iostream>
#include <vector>
#include <algorithm>
#include <limits>
using namespace std;

typedef pair<int, int> game;
const int mxN = 1e3;

int n, m, k;
vector<int> w;
vector<game> g;

int main() {
    cin >> n >> m >> k;
    w.resize(n);
    for (int &wi : w) cin >> wi;
    g.resize(m);
    for (game &gi : g) cin >> gi.first >> gi.second;
    vector<vector<int>> games(n, vector<int>(k, 0)), ans(m);
    vector<bool> vis(m, false);
    long long mx = 0;
    for (int i = 0; i < n; ++i) {
        mx += max(w[i], w[(i + 1) % n]);
        for (int j = 0; j < k; ++j) {
            if (i == k - 1) games[i][j] = w[i];
            else games[i][j] = w[i] + w[(i + 1) % n];
        }
    }
    for (int i = 0; i < k; ++i) {
        vector<pair<int, int>> gm;
        for (int j = 0; j < n; ++j) gm.push_back({games[j][i], j});
        sort(gm.begin(), gm.end());
        int bal = gm[0].second - gm[n - 1].second;
        if (bal <= 2) {
            for (int j = 0; j < n; ++j) {
                int team = gm[j].second;
                games[team][i] = 0;
                for (int l = 0; l < m; ++l) {
                    if (!vis[l] && (g[l].first == team + 1 || g[l].second == team + 1)) {
                        ans[l].push_back(i + 1);
                        vis[l] = true;
                        break;
                    }
                }
            }
        }
    }
    for (auto v : ans) {
        for (auto vi : v) cout << vi << ' ';
        cout << '\n';
    }
    return 0;
}
