n = int(input())
degree = []
s = []
edges = []
for i in range(n):
    d, x = map(int, input().split())
    degree.append(d)
    s.append(x)

while True:
    v = -1
    for i in range(n):
        if degree[i] == 1:
            v = i
            break
    if v == -1:
        break

    u = s[v]
    edges.append((v, u))
    s[u] ^= v
    degree[u] -= 1
    degree[v] -= 1
    s[v] = 0

print(len(edges))
for edge in edges:
    print(edge[0], edge[1])
