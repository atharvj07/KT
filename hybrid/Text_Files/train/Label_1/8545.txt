#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> p(n);
        for (int &i : p) cin >> i;
        deque<int> dq;
        for (int i : p) {
            if (dq.empty())
                dq.push_back(i);
            else if (i < dq.front())
                dq.push_front(i);
            else
                dq.push_back(i);
        }
        for (int i : dq) cout << i << ' ';
        cout << '\n';
    }
    return 0;
}
