#include <iostream>
using namespace std;
int main() {
  int n, m, foundr = -1, foundg = -1, foundb = -1;
  char c;
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> c;
      if (c == 'R') {
        if (foundr == -1) foundr = i;
        else if (foundr != i) { cout << "NO"; return 0; }
      } else if (c == 'G') {
        if (foundg == -1) foundg = i;
        else if (foundg != i) { cout << "NO"; return 0; }
      } else {
        if (foundb == -1) foundb = i;
        else if (foundb != i) { cout << "NO"; return 0; }
      }
    }
  }
  cout << "YES";
  return 0;
}
