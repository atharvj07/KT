#include<iostream>
using namespace std;
int main(){
    int n, m, s = 0, c = 0; cin >> n >> m;
    for (int i = 1;; i++){
        int in; while(n--){
            cin >> in;
            if(s + in <= m) s += in;
            else{c++; s = in;}
        }
        cout << c + 1;
        return 0;
    }
}
