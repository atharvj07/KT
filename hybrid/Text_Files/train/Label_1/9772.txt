#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int &i : a) { cin >> i; }
  int q;
  cin >> q;
  while (q--) {
    int x;
    cin >> x;
    int ans = -1;
    for (int i = n - 1; i >= 0; i--) {
      int num = x / a[i];
      if (num > k) continue;
      if (x % a[i] == 0) {
        ans = num;
        break;
      } else if ((x % a[i]) % a[i - 1] == 0) {
        ans = num + (x % a[i]) / a[i - 1];
        break;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
