#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
    int n;
    cin >> n;
    map<int, int> m;
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        m[a]++;
    }
    int q;
    cin >> q;
    while (q--) {
        char c;
        int x;
        cin >> c >> x;
        if (c == '+') m[x]++;
        else m[x]--;
        if (m.size() < 2) cout << "NO" << endl;
        else if (m.size() == 2) {
            if (m.begin()->second != m.rbegin()->second) cout << "NO" << endl;
            else if (m.begin()->second % 2 == 0) cout << "YES" << endl;
            else cout << "NO" << endl;
        } else {
            auto it = m.begin();
            auto it2 = m.rbegin();
            if ((it->second % 2 == 0 && it2->second % 2 == 0) || (it->second % 2 != 0 && it2->second % 2 != 0)) cout << "YES" << endl;
            else cout << "NO" << endl;
        }
    }
    return 0;
}
