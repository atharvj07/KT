
#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    if (a[n - 1] == 0) {
        cout << "NO" << endl;
        return 0;
    }
    string ans = "";
    for (int i = 0; i < n - 1; i++) {
        if (a[i] == 0) {
            ans += "((";
        } else {
            ans += "(";
        }
    }
    for (int i = 0; i < n; i++) {
        if (i) {
            ans += "->";
        }
        ans += to_string(a[i]);
    }
    for (int i = 0; i < n - 1; i++) {
        ans += ")";
    }
    cout << "YES" << endl;
    cout << ans << endl;
    return 0;
}
