import sys
import copy
input = sys.stdin.readline

def dfs(cx, cy):
    visited[cx][cy] = True
    stack.append((cx, cy))
    for k in range(4):
        nx = cx + dx[k]
        ny = cy + dy[k]
        if nx < 0 or nx >= n or ny < 0 or ny >= m:
            continue
        if grid[nx][ny] == '#':
            continue
        if visited[nx][ny]:
            continue
        dfs(nx, ny)

n, m, k = map(int, input().split())
s = 0
grid = [list(input().rstrip()) for _ in range(n)]
for i in range(n):
    for j in range(m):
        if grid[i][j] == '.':
            s += 1
dx = [0, 0, 1, -1]
dy = [1, -1, 0, 0]
for i in range(n):
    for j in range(m):
        if grid[i][j] == '.':
            visited = [[False] * m for _ in range(n)]
            stack = []
            dfs(i, j)
            if len(stack) == s - k:
                for cx, cy in stack:
                    grid[cx][cy] = 'X'
        elif grid[i][j] == 'X':
            k -= 1
            if k == 0:
                for i in range(n):
                    print(''.join(grid[i]))
                exit(0)
print(''.join(grid[i]))
