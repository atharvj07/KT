import java.util.Scanner;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int[] D = new int[N];
        int[] C = new int[M];
        for (int i = 0; i < N; i++) {
            D[i] = sc.nextInt();
        }
        for (int i = 0; i < M; i++) {
            C[i] = sc.nextInt();
        }
        long[][] dp = new long[M + 1][N + 1];
        for (int i = 0; i <= M; i++) {
            for (int j = 0; j <= N; j++) {
                dp[i][j] = Long.MAX_VALUE;
            }
        }
        dp[0][0] = 0;
        for (int i = 0; i < M; i++) {
            for (int j = 0; j <= N; j++) {
                if (dp[i][j] == Long.MAX_VALUE) {
                    continue;
                }
                dp[i + 1][j] = Math.min(dp[i + 1][j], dp[i][j]);
                if (j < N) {
                    dp[i + 1][j + 1] = Math.min(dp[i + 1][j + 1], dp[i][j] + D[j] * C[i]);
                }
            }
        }
        System.out.println(dp[M][N]);
    }
}
