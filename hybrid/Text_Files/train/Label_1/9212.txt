#include <bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;
int32_t main() {
  int n, c, a[N], b[N], dp[N];
  cin >> n >> c;
  for (int i = 1; i < n; i++) { cin >> a[i]; }
  for (int i = 1; i < n; i++) { cin >> b[i]; }
  dp[1] = 0;
  for (int i = 2; i <= n; i++) {
    dp[i] = dp[i - 1] + a[i - 1];
    dp[i] = min(dp[i], c + b[i - 1] + dp[i - 1]);
  }
  for (int i = 1; i <= n; i++) { cout << dp[i] << " "; }
  return 0;
}
