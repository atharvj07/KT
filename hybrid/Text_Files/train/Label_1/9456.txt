#include <bits/stdc++.h>
using namespace std;
const int N = 3010, mod = 1e9 + 7;
typedef long long ll;
ll n, a[N], b[N];
int main() {
    cin >> n;
    for (ll i = 0; i < n; i++) cin >> a[i];
    for (ll i = 0; i < n; i++) cin >> b[i];
    vector<pair<ll, ll>> f(n + 2);
    vector<pair<ll, ll>> g(n + 2);
    for (ll i = n - 1; i >= 0; i--) {
        ll p = lower_bound(f.begin(), f.end(), make_pair(a[i], 0ll)) - f.begin();
        f[i] = make_pair(a[i], b[i] + (p ? f[p - 1].second : 0ll));
    }
    for (ll i = 0; i < n; i++) {
        ll p = lower_bound(g.begin(), g.end(), make_pair(a[i], 0ll)) - g.begin();
        g[i] = make_pair(a[i], b[i] + (p ? g[p - 1].second : 0ll));
    }
    ll res = 1e18;
    for (ll i = 0; i < n; i++) {
        ll p = lower_bound(g.begin(), g.end(), make_pair(a[i], 0ll)) - g.begin();
        ll q = lower_bound(f.begin(), f.end(), make_pair(a[i], 0ll)) - f.begin();
        if (p && q && p != q) res = min(res, g[p - 1].second + f[q - 1].second);
    }
    if (res == 1e18)
        cout << -1 << endl;
    else
        cout << res << endl;
}
