def check(arr):
    for i in range(len(arr)//2):
        if arr[2*i] < arr[2*i+1]:
            arr[2*i], arr[2*i+1] = arr[2*i+1], arr[2*i]
            return arr
    return False

t = int(input())
while t:
    t -= 1
    n = int(input())
    a = list(map(int, input().split()))
    r = check(a)
    if r:
        for i in r:
            print(i, end = " ")
        print()
    else:
        print(-1)
