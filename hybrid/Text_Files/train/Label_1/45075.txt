import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] A = new int[N];
        int[] B = new int[N];
        for (int i = 0; i < N; i++) A[i] = sc.nextInt();
        for (int i = 0; i < N; i++) B[i] = sc.nextInt();
        int[] C = new int[N];
        for (int i = 0; i < N; i++) C[(i + 1) % N] += A[i];
        boolean[] ok = new boolean[N];
        for (int i = 0; i < N; i++) {
            if (C[i] == B[i]) ok[i] = true;
        }
        int last = -1;
        for (int i = 0; i < N; i++) {
            if (ok[i]) {
                last = i;
                break;
            }
        }
        int ans = -1;
        if (last != -1) {
            ans = 0;
            int cur = last;
            while (true) {
                C[cur] = B[cur];
                ans++;
                cur = (cur - 1 + N) % N;
                if (C[cur] == B[cur] && ok[cur]) break;
            }
        }
        System.out.println(ans);
    }
}
