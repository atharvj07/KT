import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] b = new long[n + 1];
        long[] a = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            b[i] = in.nextInt();
            a[i] = in.nextInt();
        }
        long[] p = new long[n + 1];
        for (int i = 1; i < n; i++) {
            int x = in.nextInt();
            long k = in.nextInt();
            p[i + 1] = k;
            p[x] = -k;
        }
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            ans += p[i] * b[i];
            ans -= a[i];
        }
        if (ans == 0) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
