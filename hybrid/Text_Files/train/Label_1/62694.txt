import math
import sys

n = int(input())
l = []
for i in range(n): l.append(list(map(int, input().split())))

p = [0 for i in range(n)]

for i in range(n):
    if l[i][1] != 0: p[i] = 1
    else: p[i] = 0
    l[i][0] = math.atan2(l[i][1], l[i][0])

l.sort(key=lambda x: (x[1], x[0]))
max = math.pi/2;
for i in range(n - 1):
    if (p[i] == 0 and p[i + 1] == 1):
        if (l[i + 1][0] - l[i][0] < 0): max = max - abs(math.pi + l[i + 1][0] - l[i][0])
        else: max = max - abs(l[i + 1][0] - l[i][0])

print(max * 180 / math.pi)
