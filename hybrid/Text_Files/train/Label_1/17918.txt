#include <iostream>
#include <bitset>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, m; cin >> n >> m;
    vector<unsigned int> db(n);
    for (auto &i : db) {
        string s; cin >> s;
        for (auto c : s) i |= (1u << (c - 'a'));
    }
    for (int i = 0; i < m; ++i) {
        string s; cin >> s;
        bool ok = false;
        for (auto &j : db)
            if (bitset<26>(j).count() == s.size() && bitset<26>(j ^ (1u << (s[0] - 'a'))).count() == s.size() - 1)
                ok = true;
        cout << (ok ? "YES\n" : "NO\n");
    }
    return 0;
}
