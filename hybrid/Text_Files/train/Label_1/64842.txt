n = int(input())
a_success, a_total, b_success, b_total = 0, 0, 0, 0

for _ in range(n):
    t, x, y = map(int, input().split())
    if t == 1:
        a_success += x
        a_total += 10
    else:
        b_success += x
        b_total += 10

if a_success >= a_total / 2:
    print("LIVE")
else:
    print("DEAD")

if b_success >= b_total / 2:
    print("LIVE")
else:
    print("DEAD")
