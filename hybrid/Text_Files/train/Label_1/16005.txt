#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> arr(n);
  for (int i = 0; i < n; i++) { cin >> arr[i]; }
  int sum = 0;
  for (int i : arr) { sum += i; }
  if (sum % 2 == 0) {
    cout << "First" << endl;
  } else {
    cout << "Second" << endl;
  }
  return 0;
}
