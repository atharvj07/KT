import java.util.Scanner;

public class SellOut {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int f = scanner.nextInt();

        int[] products = new int[n];
        int[] clients = new int[n];
        for (int i = 0; i < n; i++) {
            products[i] = scanner.nextInt();
            clients[i] = scanner.nextInt();
        }

        int[][] dp = new int[n + 1][f + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= f; j++) {
                dp[i][j] = dp[i - 1][j];
                if (j > 0) {
                    dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - 1] + Math.min(clients[i - 1], 2 * products[i - 1]));
                }
            }
        }

        System.out.println(dp[n][f]);
    }
}
