#include <iostream>
#include <string>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    int ok = 0;
    for (int i = 0; i < n / 2; i++) {
      if (s.substr(i, n / 2) == s.substr(n / 2, n / 2)) {
        ok = 1;
        break;
      }
    }
    if (ok)
      cout << "YES\n";
    else
      cout << "NO\n";
  }
  return 0;
}
