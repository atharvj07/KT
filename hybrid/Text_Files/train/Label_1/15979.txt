#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n,m,a[1000001],b[1000001],c[1000001];
int main(){
    cin>>n>>m;
    for(ll i=1;i<=n;i++){
        cin>>a[i] >> b[i];
        c[i]=c[i-1]+a[i]*b[i];
    }
    for(ll i=1;i<=m;i++){
        cin>>a[i];
        for(ll j=1;j<=n;j++){
            if(a[i]<=c[j]){
                cout<<j<<'\n';
                break;
            }
        }
    }
    return 0;
}
