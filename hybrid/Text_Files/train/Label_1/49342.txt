import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int[] player1 = new int[2];
    player1[0] = scanner.nextInt();
    player1[1] = scanner.nextInt();
    int[] player2 = new int[2];
    player2[0] = scanner.nextInt();
    player2[1] = scanner.nextInt();
    int[] player3 = new int[2];
    player3[0] = scanner.nextInt();
    player3[1] = scanner.nextInt();
    int[] player4 = new int[2];
    player4[0] = scanner.nextInt();
    player4[1] = scanner.nextInt();

    int team1Attack = Math.max(player1[1], player2[1]);
    int team1Defense = Math.max(player1[0], player2[0]);
    int team2Attack = Math.max(player3[1], player4[1]);
    int team2Defense = Math.max(player3[0], player4[0]);

    if (team1Attack > team2Defense && team1Defense > team2Attack) {
      System.out.println("Team 1");
    } else if (team2Attack > team1Defense && team2Defense > team1Attack) {
      System.out.println("Team 2");
    } else {
      System.out.println("Draw");
    }
  }
}
