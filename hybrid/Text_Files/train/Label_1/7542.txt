#define ll long long
#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int lb = 0, rb = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(')
      lb++;
    else
      rb++;
    if (rb > lb) {
      cout << "No" << endl;
      return 0;
    }
  }
  if (lb != rb)
    cout << "No" << endl;
  else
    cout << "Yes" << endl;
  return 0;
}
