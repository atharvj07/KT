import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int z = scanner.nextInt();
        int count = 0;
        for (int i = m; i <= z; i += m) {
            if ((i % n) == 0) {
                count++;
            }
        }
        System.out.println(count);
    }
}
