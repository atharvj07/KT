t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    max_diff = 0
    for i in range(n-1, 0, -1):
        if a[i] < a[i-1]:
            max_diff = max(max_diff, a[i-1] - a[i])
    if max_diff == 0:
        print(0)
    else:
        print(max_diff.bit_length())
