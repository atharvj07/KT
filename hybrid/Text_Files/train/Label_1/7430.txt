#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n, ans = 0;
    cin >> n;
    map<int, bool> mp[7];
    while(n--)
    {
        int x;
        for(int i = 0; i < 6; i++){
            cin >> x;
            mp[i][x] = true;
        }
    }
    for(int i = 1;; i++)
    {
        bool fl = true;
        for(int j = 0; j < min(i, 7); j++)
        {
            if(mp[j][i % 10] == false)
            {
                fl = false;
                break;
            }
            i /= 10;
        }
        if(fl){
            ans = i - 1;
        }
        else{
            break;
        }
    }
    cout << ans;
}
