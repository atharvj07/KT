n, d = map(int, input().split())
s = input()

current_pos = 0
jumps = 0

while current_pos < n - 1:
    found = False
    for jump in range(d, 0, -1):
        if current_pos + jump < n and s[current_pos + jump] == '1':
            current_pos += jump
            jumps += 1
            found = True
            break
    if not found:
        jumps = -1
        break

print(jumps)
