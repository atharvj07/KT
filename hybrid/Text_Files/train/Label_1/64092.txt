import sys
N, M = map(int, input().split())
C = int(input())
hotels = [tuple(map(int, input().split())) for i in range(C)]
H = int(input())
restaurants = [tuple(map(int, input().split())) for i in range(H)]
sys.stdout.write(' '.join(map(str, [min(abs(x-rx)+abs(y-ry) for x,y in hotels for rx,ry in restaurants), restaurants.index(min(restaurants, key=lambda r: min(abs(x-r[0])+abs(y-r[1]) for x,y in hotels)))+1])))
