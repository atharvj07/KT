import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[] times = new long[n];
        for (int i = 0; i < n; i++) {
            times[i] = sc.nextLong();
        }
        ArrayList<Long> periods = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            periods.add(sc.nextLong());
        }
        periods.sort(null);
        long min = times[0];
        boolean can = false;
        for (long period : periods) {
            long y = min % period == 0 ? min : min - min % period + period;
            boolean ok = true;
            for (long time : times) {
                if ((time - y) % period != 0) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                System.out.println("YES");
                System.out.println(y + " " + (periods.indexOf(period) + 1));
                can = true;
                break;
            }
        }
        if (!can) {
            System.out.println("NO");
        }
    }
}
