#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;

    vector<int> lastPos(26, -1);
    vector<int> cnt(26, 0);

    int ans = 0;
    for (int i = 0; i < n; i++) {
        int chr = s[i] - 'a';
        if (lastPos[chr] != -1) {
            for (int j = lastPos[chr]; j < i; j++) {
                cnt[s[j] - 'a']--;
            }
            ans = max(ans, i - lastPos[chr]);
        }
        cnt[chr]++;
        lastPos[chr] = i;
    }

    for (int i = 0; i < 26; i++) {
        if (cnt[i] > 1) {
            ans = -1;
            break;
        }
    }

    cout << ans << endl;

    return 0;
}
