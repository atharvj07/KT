n = int(input())
a = list(map(int, input().split()))
s = input()[::-1]

max_val = 0
current_sum = 0

for i in range(n):
    if s[i] == '1':
        current_sum += a[i]
    max_val = max(max_val, current_sum)

for i in range(n):
    if s[i] == '0':
        max_val = max(max_val, current_sum + a[i])

print(max_val)
