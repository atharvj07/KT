#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const int N=1e5+5;
int n, k, l;
ll a[N];
ll solve() {
	sort(a, a+n*k);
	int s=0, e=n-1, mid, ans=0;
	while(s<=e) {
		mid=(s+e)/2;
		int mn=mid, mx=mid+k-1;
		if(a[mx]-a[mn]<=l) {
			ans=a[mn]*(k);
			s=mid+1;
		} else e=mid-1;
	}
	if(ans) return ans;
	return 0;
}
int main() {
	cin >> n >> k >> l;
	for(int i=0; i<n*k; i++) cin >> a[i];
	cout << solve() << endl;
	return 0;
}
