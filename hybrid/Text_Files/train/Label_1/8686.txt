#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<int> b;
    for (int i = 0; i < k - 1; i++) {
        int idx = min_element(a.begin() + i, a.end()) - a.begin();
        b.push_back(idx);
        reverse(a.begin() + i, a.begin() + idx + 1);
    }
    int ans = *min_element(a.begin(), a.end());
    for (int i = 0; i < b.size(); i++) {
        ans = max(ans, a[b[i]]);
    }
    cout << ans << "\n";
    return 0;
}
