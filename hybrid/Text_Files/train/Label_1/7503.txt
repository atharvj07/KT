#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;

pii getDiff(string &s, string &t) {
    int tot = 0;
    for(int i = 0; i < s.size(); ++i) {
        if(s[i] != t[i]) {
            tot += (s[i] < t[i]);
        }
    }
    return {tot, tot ? -1 : 0};
}
int main() {
    string s, t;
    cin >> s >> t;
    pii diff = getDiff(s, t);
    if(diff.second < 0) cout << -1 << endl;
    else {
        cout << diff.first << endl;
        for(int i = 0; i < s.size(); ++i) {
            if(s[i] != t[i]) {
                cout << i + 1 << " " << i + diff.second + 1 << endl;
            }
        }
    }
    return 0;
}
