t = int(input())
for i in range(t):
    s = input()
    s = s.replace("?","a")
    for j in range(1,len(s)):
        if s[j] == s[j-1]:
            s = s.replace("?","b")
    for j in range(1,len(s)):
        if s[j] == s[j-1]:
            s = s.replace("?","c")
    for j in range(1,len(s)):
        if s[j] == s[j-1]:
            s = -1
            break
    print(s)
