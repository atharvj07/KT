import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    var sc = new Scanner(System.in);
    int[] p1 = {sc.nextInt(), sc.nextInt()};
    int[] p2 = {sc.nextInt(), sc.nextInt()};
    int[] p3 = {sc.nextInt(), sc.nextInt()};
    int[] p4 = {sc.nextInt(), sc.nextInt()};
    boolean team1 = false, team2 = false;
    if ((p1[0] > p4[1] && p1[1] > p4[0]) || (p1[0] > p3[1] && p1[1] > p3[0])) team1 = true;
    if ((p2[0] > p3[1] && p2[1] > p3[0]) || (p2[0] > p4[1] && p2[1] > p4[0])) team2 = true;
    if (team1 && team2) System.out.print("Draw");
    else if (team1) System.out.print("Team 1");
    else if (team2) System.out.print("Team 2");
    else System.out.print("Draw");
  }
}
