#include<iostream>
#include<algorithm>
using namespace std;
int main() {
    int n, c, d, ans = 0;
    cin >> n >> c >> d;
    pair<pair<int, char>, int> fnt[n + 1];
    for (int i = 0;i < n;i++) {
        cin >> fnt[i].first.first >> fnt[i].first.second >> fnt[i].second;
    }
    sort(fnt, fnt + n);
    for (int i = n - 1, j = n - 1;i >= 0;i--) {
        if (fnt[i].first.second == 'C') {
            if (c >= fnt[i].first.first) {
                j = i;
                c -= fnt[i].first.first;
                ans += fnt[i].second;
            }
        }
        else if (fnt[i].first.second == 'D') {
            if ((fnt[i].second <= j) && (d >= fnt[i].first.first)) {
                d -= fnt[i].first.first;
                ans += fnt[i].second;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
