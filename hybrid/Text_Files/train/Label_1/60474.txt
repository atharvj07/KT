def sum_of_digits(x):
    return sum(map(int, str(x)))

def find_min_x(n, k):
    for x in range(1000):
        total_sum = sum(sum_of_digits(x + i) for i in range(k + 1))
        if total_sum == n:
            return x
    return -1

t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().split())
    print(find_min_x(n, k))
