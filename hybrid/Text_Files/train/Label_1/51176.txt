import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int q = sc.nextInt();
            String s = sc.next();
            for (int i = 0; i < q; i++) {
                int l = sc.nextInt() - 1;
                int r = sc.nextInt() - 1;
                boolean found = false;
                for (int j = l + 1; j < r; j++) {
                    if (s.charAt(j) != s.charAt(l) && s.charAt(j) != s.charAt(r)) {
                        found = true;
                        break;
                    }
                }
                System.out.println(found ? "YES" : "NO");
            }
        }
    }
}
