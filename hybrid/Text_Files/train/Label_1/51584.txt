import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int prizeCodingContest = 0;
        int prizeRobotManeuver = 0;
        int bonus = 0;

        switch (x) {
            case 1:
                prizeCodingContest = 300000;
                break;
            case 2:
                prizeCodingContest = 200000;
                break;
            case 3:
                prizeCodingContest = 100000;
                break;
            default:
                prizeCodingContest = 0;
        }

        switch (y) {
            case 1:
                prizeRobotManeuver = 300000;
                break;
            case 2:
                prizeRobotManeuver = 200000;
                break;
            case 3:
                prizeRobotManeuver = 100000;
                break;
            default:
                prizeRobotManeuver = 0;
        }

        if (x == 1 && y == 1) {
            bonus = 400000;
        }

        int totalPrize = prizeCodingContest + prizeRobotManeuver + bonus;
        System.out.println(totalPrize);
    }
}
