import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long k = input.nextLong();

        int numDigits = 1;
        long num = 9L;
        while (k > num * numDigits) {
            k -= num * numDigits;
            numDigits++;
            num *= 10;
        }

        long start = num / 10;
        long digitIndex = k / numDigits;
        long digit = start + digitIndex;
        System.out.println(digit % 10);
    }
}
