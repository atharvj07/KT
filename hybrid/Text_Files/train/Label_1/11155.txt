#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(m), b(m);
  for (int i = 0; i < m; i++) { cin >> a[i] >> b[i]; }
  for (int k = 1; k < n; k++) {
    bool ok = true;
    for (int i = 0; i < m; i++) {
      if ((b[i] - a[i] + n) % n != k) {
        ok = false;
        break;
      }
    }
    if (ok) {
      cout << "Yes" << endl;
      return 0;
    }
  }
  cout << "No" << endl;
  return 0;
}
