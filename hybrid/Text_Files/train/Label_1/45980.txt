import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            int p = scanner.nextInt();
            String s = scanner.next();

            int n = s.length();
            int[] dp = new int[n + 1];

            dp[n] = 0;
            for (int j = n - 1; j >= 0; j--) {
                dp[j] = Math.min(a, b);

                if (s.charAt(j) == 'A') {
                    for (int k = j + 1; k < n && s.charAt(k) == 'A'; k++) {
                        dp[j] = Math.min(dp[k + 1] + a, dp[j]);
                    }
                }

                if (s.charAt(j) == 'B') {
                    for (int k = j + 1; k < n && s.charAt(k) == 'B'; k++) {
                        dp[j] = Math.min(dp[k + 1] + b, dp[j]);
                    }
                }
            }

            for (int j = 1; j <= n; j++) {
                if (dp[j] <= p) {
                    System.out.println(j);
                    break;
                }
            }
        }
    }
}
