import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        int[] sum = new int[n + 1];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            sum[i + 1] = sum[i] + a[i];
        }
        while (m-- > 0) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt();
            System.out.println((sum[r] - sum[l]) % 2 == 0 ? 1 : 0);
        }
    }
}
