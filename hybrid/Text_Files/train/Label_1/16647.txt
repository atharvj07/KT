#include <bits/stdc++.h>
using namespace std;

int32_t main() {
  int n, k;
  cin >> n >> k;
  vector<int> skills(n);
  for (int i = 0; i < n; i++) cin >> skills[i];
  unordered_set<int> rivals[n];
  for (int i = 0; i < k; i++) {
    int x, y;
    cin >> x >> y;
    rivals[x - 1].insert(y - 1);
    rivals[y - 1].insert(x - 1);
  }
  vector<int> ans(n);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (skills[i] > skills[j] && !rivals[i].count(j)) ans[i] += 1;
    }
    cout << ans[i] << " ";
  }
  return 0;
}
