import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p1 = new int[5], p2 = new int[5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                int t = in.nextInt();
                if (i == 0) p1[j] = t;
                else if (i == 1) p2[j] = t;
            }
        }
        double x = 0;
        for (int i = 0; i < 5; i++) {
            if (p1[i] == -1) p1[i] = 120;
            if (p2[i] == -1) p2[i] = 120;
            x += Math.max(500, 2000 - Math.max(0, p1[i] - 48) * 2000 / 250) - Math.max(500, 2000 - Math.max(0, p2[i] - 48) * 2000 / 250);
        }
        System.out.println(x < EPS ? -1 : (int) (x / 500 + EPS));
    }
    static final double EPS = 1e-9;
}
