#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long int> h(n);
  for (int i = 0; i < n; i++) { cin >> h[i]; }
  vector<long long int> ans(n);
  long long int mx = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (h[i] > mx)
      mx = h[i];
    else
      ans[i] = mx - h[i];
  }
  for (int i = 0; i < n; i++) { cout << ans[i] << " "; }
  cout << endl;
  return 0;
}
