#include <bits/stdc++.h>
using namespace std;
int main() {
    int test;
    cin >> test;
    while (test--) {
        int n, m = 0;
        cin >> n;
        vector<pair<int, int>> op;
        for (int i = 2; i <= n; i++) {
            if (i % 2) {
                if (n % 2)
                    op.push_back({n, i}), m++, n++, i--;
                else
                    op.push_back({n - 1, i}), m++, n--;
            }
        }
        cout << m << endl;
        for (auto &x : op) {
            cout << x.first << " " << x.second << endl;
        }
    }
    return 0;
}
