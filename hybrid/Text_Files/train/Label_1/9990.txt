#include <iostream>
#include <iomanip>
#include <cmath>
using namespace std;

int main() {
    double y1, y2, yw, xb, yb, r;
    cin >> y1 >> y2 >> yw >> xb >> yb >> r;
    double h = yw - yb - r;
    if (yb / h - floor(yb / h) > 0.5) {
        h = yw - yb + r;
    }
    if (xb * xb / h / h + 1 <= (y2 - yw) / (yw - y1)) {
        cout << fixed << setprecision(9) << sqrt(h * h / (xb * xb / h / h + 1) + xb * xb) << endl;
    } else {
        cout << -1 << endl;
    }
    return 0;
}
