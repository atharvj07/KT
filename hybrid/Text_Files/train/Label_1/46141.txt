import java.util.*;
import java.lang.*;
import java.io.*;

public class Main {
    static int[] p;
    static int[] sz;
    static int n;

    static int find(int x) {
        if (p[x] == x) return x;
        return p[x] = find(p[x]);
    }

    static void union(int x, int y) {
        int rx = find(x);
        int ry = find(y);
        if (rx == ry) return;
        if (sz[rx] > sz[ry]) {
            p[ry] = rx;
            sz[rx] += sz[ry];
        } else {
            p[rx] = ry;
            sz[ry] += sz[rx];
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        n = Integer.parseInt(s[0]);
        int m = Integer.parseInt(s[1]);
        int d = Integer.parseInt(s[2]);
        s = br.readLine().split(" ");
        p = new int[n + 1];
        sz = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            p[i] = i;
            sz[i] = 1;
        }
        Set<Integer> set = new HashSet<>();
        for (int i = 0; i < m; i++) {
            set.add(Integer.parseInt(s[i]));
        }
        for (int i = 0; i < n - 1; i++) {
            s = br.readLine().split(" ");
            int x = Integer.parseInt(s[0]);
            int y = Integer.parseInt(s[1]);
            union(x, y);
        }
        int ans = 0;
        for (int x : set) {
            if (find(x) == x) ans++;
        }
        System.out.println(ans);
    }
}
