import sys
input = sys.stdin.readline
inf = float('inf')

d, n, m = map(int, input().split())
p = [inf] * (d+1)
for _ in range(m):
    x, y = map(int, input().split())
    p[x] = min(p[x], y)
ans = 0
a = 0
while a < d:
    b = a+n
    if b > d:
        b = d
    if p[a] == inf:
        ans = -1
        break
    ans += min(p[a] * (b-a), p[a]*(d-b))
    a = b
print(ans)
