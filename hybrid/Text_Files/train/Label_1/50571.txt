import java.util.*;
import java.lang.*;

class Main
{
	public static void main(String[] args)
	{
		Scanner in= new Scanner(System.in);
		
		int t=in.nextInt();
		
		for(int i=0; i<t; i++)
		{
			int n=in.nextInt();
			
			int[] a=new int[n+1];
			
			for(int j=1; j<=n; j++)
				a[j]=in.nextInt();
			
			boolean ok=true;
			
			for(int j=1; j<=n; j++)
			{
				if(a[j]==j)
					continue;
				
				if(a[j]==j-1 || a[j]==j+1)
					continue;
				
				ok=false;
			}
			
			if(ok)
				System.out.println("YES");
			else
				System.out.println("NO");
		}
	}
}
