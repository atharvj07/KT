#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, s, t;
  cin >> n >> s >> t;

  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
  }

  vector<int> pos(n);
  for (int i = 0; i < n; i++) {
    pos[p[i] - 1] = i + 1;
  }

  int cur = s - 1;
  int cnt = 0;
  while (cur != t - 1) {
    cur = pos[cur] - 1;
    cnt++;
    if (cnt > n) {
      cout << -1 << endl;
      return 0;
    }
  }

  cout << cnt << endl;
  return 0;
}
