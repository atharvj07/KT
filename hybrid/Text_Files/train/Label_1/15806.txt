#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, sum1 = 0, sum2 = 0, minDiff = 360;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        sum1 += a[i];
    }
    for (int i = 0; i < n; i++) {
        sum2 += a[i];
        sum1 -= a[i];
        minDiff = min(minDiff, abs(sum1 - sum2));
    }
    cout << minDiff << endl;
}
