#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        bool canSort = true;
        for (int i = 0; i < n - 1; i++) {
            if (a[i] > a[i + 1] && i + x >= n) {
                canSort = false;
                break;
            }
        }
        cout << (canSort ? "YES" : "NO") << endl;
    }
    return 0;
}
