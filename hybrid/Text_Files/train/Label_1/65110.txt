import sys

t = int(sys.stdin.readline())

for _ in range(t):
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    dp = [0 for _ in range(n)]
    dp[0] = 0
    for i in range(1, n):
        dp[i] = dp[i - 1]
        if a[i] < a[i - 1]:
            dp[i] += 1
    print(dp[n - 1])
