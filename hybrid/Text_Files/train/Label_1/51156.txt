import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] b = new int[n];
            for (int i = 0; i < n; i++) {
                b[i] = sc.nextInt();
            }
            StringBuilder res = new StringBuilder();
            int[] a = new int[n + 1];
            a[n / 2 + 1] = b[0];
            int cur = n / 2 + 2;
            int from = 1;
            for (int i = 1; i < n; i++) {
                if (cur > n) {
                    cur = 1;
                }
                if (from > n / 2) {
                    from = 1;
                }
                if (b[i] < b[from] && cur <= n / 2) {
                    a[cur] = b[i];
                    cur++;
                } else {
                    a[from] = b[i];
                    from++;
                }
            }
            for (int i = 1; i <= n; i++) {
                res.append(a[i] + " ");
            }
            System.out.println(res.toString());
        }
    }
}
