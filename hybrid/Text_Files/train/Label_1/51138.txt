import java.util.*;

class Main {

    private static int n, l;
    private static int[] distances;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        l = scanner.nextInt();
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int[] distances = new int[n];
        for (int i = 0; i < n; i++) {
            distances[i] = scanner.nextInt();
        }

        ArrayList<Integer> newDistances = new ArrayList<>();

        if (!existsDistance(distances, x)) {
            newDistances.add(x);
        }
        if (!existsDistance(distances, y)) {
            newDistances.add(y);
        }

        System.out.println(newDistances.size());
        for (int newDistance : newDistances) {
            System.out.print(newDistance + " ");
        }
        System.out.println();
    }

    private static boolean existsDistance(int[] distances, int distance) {
        int startIndex = 0;
        int endIndex = distances.length - 1;
        while (startIndex <= endIndex) {
            int middleIndex = (startIndex + endIndex) / 2;
            if (distances[middleIndex] == distance) {
                return true;
            } else if (distances[middleIndex] < distance) {
                startIndex = middleIndex + 1;
            } else {
                endIndex = middleIndex - 1;
            }
        }
        return false;
    }
}
