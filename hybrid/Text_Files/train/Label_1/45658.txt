import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = br.readLine().split(" ");
        int n = Integer.parseInt(tokens[0]);
        int k = Integer.parseInt(tokens[1]);
        boolean[][] walls = new boolean[2][n];
        String[] left = br.readLine().split("");
        String[] right = br.readLine().split("");
        for (int i = 0; i < n; i++) {
            walls[0][i] = left[i].equals("X");
            walls[1][i] = right[i].equals("X");
        }
        boolean[][] visited = new boolean[2][n];
        boolean[][] water = new boolean[2][n];
        Queue<int[]> queue = new LinkedList<>();
        queue.add(new int[] {0, 0, 0});
        while (!queue.isEmpty()) {
            int[] state = queue.poll();
            int wall = state[0];
            int height = state[1];
            int time = state[2];
            if (height >= n) {
                System.out.println("YES");
                return;
            }
            if (visited[wall][height]) {
                continue;
            }
            visited[wall][height] = true;
            if (time == 0) {
                water[wall][height] = true;
            }
            if (!walls[wall][height] && !water[wall][height]) {
                queue.add(new int[] {wall, height + 1, time});
                queue.add(new int[] {wall, height - 1, time});
                queue.add(new int[] {1 - wall, height + k, time + 1});
            }
        }
        System.out.println("NO");
    }
}
