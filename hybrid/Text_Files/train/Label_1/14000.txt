#include<bits/stdc++.h>
using namespace std;
int n,u,v,x,y;
map<int,int>mp;
int main(){
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>u>>v>>x;
        if(v!=-1)mp[v]++;
        if(x!=-1)mp[x]++;
    }
    y=0;
    for(auto z:mp){
        if(z.second==1)y++;
    }
    cout<<y<<endl;
    return 0;
}
