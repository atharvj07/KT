import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int[][] subarrays = new int[m][2];
        for (int i = 0; i < m; i++) {
            subarrays[i][0] = input.nextInt();
            subarrays[i][1] = input.nextInt();
        }
        int[] count = new int[n];
        for (int[] subarray : subarrays) {
            for (int i = subarray[0] - 1; i < subarray[1]; i++) {
                count[i]++;
            }
        }
        int max = 0;
        for (int i = 0; i < n; i++) {
            max += a[i] * count[i];
        }
        System.out.println(max);
    }
}
