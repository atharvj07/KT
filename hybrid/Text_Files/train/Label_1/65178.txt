n = int(input())
p = [int(i) for i in input().split()]
s = [int(i) for i in input().split()]
d = {}
for i in range(n):
    d[p[i]] = i
ans = []
for i in range(n):
    if p[i] != s[i]:
        j = d[s[i]]
        ans += [[i, j]]
        p[i], p[j] = p[j], p[i]
        d[p[i]] = i
        d[p[j]] = j
print(len(ans))
print(len(ans))
for i in ans:
    print(*i)
