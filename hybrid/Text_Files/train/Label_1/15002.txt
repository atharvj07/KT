#include <bits/stdc++.h>
using namespace std;
#define endl "\n"

const int MAXN = 1001;
const int MAXS = 11;
char ip[MAXN][MAXS], name[MAXN][MAXS], cmd[MAXN][MAXS];
int ip_of_name[MAXN];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> name[i] >> ip[i];
        ip_of_name[i] = i;
    }
    for (int i = 1; i <= m; i++) cin >> cmd[i];
    for (int i = 1; i <= m; i++) cout << cmd[i] << ' ' << ip[ip_of_name[i]] << "; #" << name[ip_of_name[i]] << endl;
    return 0;
}
