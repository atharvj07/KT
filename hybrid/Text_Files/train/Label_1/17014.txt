#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int d = -1000; d <= 1000; d++) {
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (d * a[i] > 0) { cnt++; }
    }
    if (d != 0 && cnt >= (n + 1) / 2) {
      cout << d << endl;
      return 0;
    }
  }
  cout << 0 << endl;
  return 0;
}
