import java.util.Scanner;

public class CodeforcesContest {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int c = scanner.nextInt();
        int d = scanner.nextInt();

        int mishaPoints = getPoints(a, c);
        int vasyaPoints = getPoints(b, d);

        if (mishaPoints > vasyaPoints) {
            System.out.println("Misha");
        } else if (mishaPoints < vasyaPoints) {
            System.out.println("Vasya");
        } else {
            System.out.println("Tie");
        }
    }

    private static int getPoints(int problemCost, int submissionTime) {
        int reducedCost = problemCost - (submissionTime / 250) * 250;
        return Math.max(reducedCost, 0);
    }
}
