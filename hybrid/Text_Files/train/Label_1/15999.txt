#include<bits/stdc++.h>
using namespace std;
string s;
int n,ans,a[1005];
int main()
{
    cin>>n>>s;
    int i=1,j=1;
    while(i<=n)
    {
        int cntz=0,cnto=0;
        while(i<=n && ((s[i-1]=='0' && cntz==0) || (s[i-1]=='1' && cnto==0)))
        {
            if(s[i-1]=='0')cntz=1;
            if(s[i-1]=='1')cnto=1;
            a[j]=i-1;
            i++;
        }
        j++;
        while(i<=n && ((s[i-1]=='0' && cntz==1) || (s[i-1]=='1' && cnto==1)))
        {
            if(s[i-1]=='0')cntz=0;
            if(s[i-1]=='1')cnto=0;
            i++;
        }
    }
    cout<<j-1<<endl;
    for(int i=1;i<j;i++)
    {
        for(int j=a[i];j>=a[i]-((a[i+1]-a[i])/2);j--)
        {
            cout<<s[j];
        }
        cout<<" ";
    }
    cout<<endl;
    return 0;
}
