#include<bits/stdc++.h>
#define int long long int
using namespace std;
int32_t main(){
	int n, m, r, c, x, y;
	cin >> n >> m >> r >> c >> x >> y;
	r--;
	c--;
	queue<pair<int, pair<int, int>>> bfs;
	bfs.push({0, {r, c}});
	vector<vector<bool>> vis(n, vector<bool>(m, false));
	vis[r][c] = true;
	while (!bfs.empty()) {
		pair<int, pair<int, int>> p = bfs.front();
		bfs.pop();
		pair<int, int> u = p.second;
		int cur_x = u.first;
		int cur_y = u.second;
		if (cur_x - 1 >= 0 && !vis[cur_x - 1][cur_y] && p.first < x) {
			bfs.push({p.first + 1, {cur_x - 1, cur_y}});
			vis[cur_x - 1][cur_y] = true;
		}
		if (cur_x + 1 < n && !vis[cur_x + 1][cur_y] ) {
			bfs.push({p.first + 1, {cur_x + 1, cur_y}});
			vis[cur_x + 1][cur_y] = true;
		}
		if (cur_y - 1 >= 0 && !vis[cur_x][cur_y - 1] && p.first < y) {
			bfs.push({p.first + 1, {cur_x, cur_y - 1}});
			vis[cur_x][cur_y - 1] = true;
		}
		if (cur_y + 1 < m && !vis[cur_x][cur_y + 1]) {
			bfs.push({p.first + 1, {cur_x, cur_y + 1}});
			vis[cur_x][cur_y + 1] = true;
		}
	}
	int ans = 0;
	for (auto x : vis)
		for (bool y : x)
			if (y) ans++;
	cout << ans << endl;
}
