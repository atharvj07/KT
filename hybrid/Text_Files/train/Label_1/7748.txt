#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, t;
    cin >> n >> t;
    vector<long long> a(n - 1);
    for (long long i = 0; i < n - 1; i++) {
        cin >> a[i];
    }
    vector<bool> vis(n + 1, false);
    vis[1] = true;
    queue<long long> q;
    q.push(1);
    while (!q.empty()) {
        long long x = q.front();
        q.pop();
        if (x == t) {
            cout << "YES" << endl;
            return 0;
        }
        for (long long i = 0; i < n - 1; i++) {
            if (!vis[x + a[i]] && x + a[i] <= n) {
                vis[x + a[i]] = true;
                q.push(x + a[i]);
            }
        }
    }
    cout << "NO" << endl;
    return 0;
}
