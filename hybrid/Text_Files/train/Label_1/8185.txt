#include <bits/stdc++.h>
using namespace std;
const int N=2e5+5;
int main(){
    int n,u,v;
    cin>>n;
    vector<int> adj[N];
    for(int i=1;i<n;++i){
        cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    vector<int> vis(n+1,0),dis(n+1,0);
    function<void(int,int)> dfs=[&](int node, int par){
        for(auto u: adj[node]){
            if(u==par)continue;
            dis[u]=dis[node]+1;
            dfs(u,node);
        }
    };
    dfs(1,1);
    int maxm=0,node;
    for(int i=1;i<=n;++i)if(dis[i]>maxm){maxm=dis[i];node=i;}
    fill(vis.begin(),vis.end(),0);
    fill(dis.begin(),dis.end(),0);
    dfs(node,node);
    maxm=0;
    for(int i=1;i<=n;++i)if(dis[i]>maxm){maxm=dis[i];}
    cout<<(maxm*(maxm+1))/2<<endl;
}
