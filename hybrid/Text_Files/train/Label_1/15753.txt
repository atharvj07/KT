#include <bits/stdc++.h>
#define ll long long

using namespace std;
map<ll, int>mp;

int main() {
    int n;
    cin >> n;
    vector<ll>v(n);
    for(int i = 0; i < n; i++) {
        ll x;
        cin >> x;
        v[i] = x;
        mp[x]++;
    }
    for(int i = 0; i < n; i++) {
        cout << v[i] << ' ';
        ll x = v[i];
        while(mp[x / 3]) {
            x /= 3;
        }
        mp[x]--;
    }
    return 0;
}
