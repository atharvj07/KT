from collections import defaultdict

n, m = map(int, input().split())
w = list(map(int, input().split()))
graph = defaultdict(list)
for _ in range(m):
    u, v = map(int, input().split())
    graph[u].append(v)
    graph[v].append(u)
s = int(input())
vis = [False for _ in range(n+1)]
vis[s] = True
ans = w[s-1]
mx = 0
for node in graph[s]:
    mx = max(mx, w[node-1])
ans += mx
for node in graph[s]:
    if w[node-1] != mx: continue
    path = [node]
    skip = False
    while True:
        lst = []
        for u in path[-1]:
            if vis[u]: skip = True; break
            lst.extend(graph[u])
        path.append(lst)
        if skip: break
    for u in path[-1]:
        if vis[u]: continue
        vis[u] = True
        ans += w[u-1]
        mx = max(mx, w[u-1])
print(ans)
