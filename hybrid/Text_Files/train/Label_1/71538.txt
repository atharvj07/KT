import math

n, m = map(int, input().split())
x = [0 for _ in range(n + 1)]
y = [0 for _ in range(n + 1)]
ang = 0

for _ in range(m):
    t, i, v = map(int, input().split())
    if t == 1:
        x[i] += v * math.cos(ang * math.pi / 180)
        y[i] += v * math.sin(ang * math.pi / 180)
    else:
        ang += v
    print(*[round(_, 10) for _ in (x[n] + math.cos(ang * math.pi / 180), y[n] + math.sin(ang * math.pi / 180))])
