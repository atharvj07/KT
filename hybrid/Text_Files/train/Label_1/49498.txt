import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long d = sc.nextLong(), n = sc.nextLong(), m = sc.nextLong();

        ArrayList<Station> arr = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            arr.add(new Station(sc.nextLong(), sc.nextLong()));
        }
        Collections.sort(arr);

        long prev = 0, cost = 0;
        for (Station s : arr) {
            long dist = s.pos - prev;
            if (n < dist) {
                cost = -1;
                break;
            }

            n -= dist;
            cost += dist * s.price;
            prev = s.pos;
        }

        if (n < d - prev) cost = -1;
        System.out.println(cost);
    }

    static class Station implements Comparable<Station> {
        long pos, price;

        public Station(long pos, long price) {
            this.pos = pos;
            this.price = price;
        }

        @Override
        public int compareTo(Station other) {
            return Long.compare(pos, other.pos);
        }
    }
}
