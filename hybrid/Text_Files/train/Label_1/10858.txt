#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
map<int, int> mp;
int main() {
    long long n, k;
    cin >> n >> k;
    unordered_map<long long, int> cnt;
    for (long long a_i, i = 1; i <= n; i++) {
        cin >> a_i;
        long long x = (long long)pow(a_i, k);
        cnt[x]++;
    }
    long long ans = 0;
    for (auto kv : cnt) {
        long long x = kv.first;
        int freq = kv.second;
        ans += (long long)freq * (freq - 1) / 2;
    }
    cout << ans << '\n';
}
