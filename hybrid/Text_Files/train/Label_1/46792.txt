import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] > 1) {
                a[i]--;
                cnt++;
            } else {
                break;
            }
        }
        Arrays.sort(a);
        for (int i = 0; i < n; i++) {
            if (a[i] < 150000) {
                a[i]++;
                cnt++;
            } else {
                break;
            }
        }
        System.out.println(cnt);
    }
}
