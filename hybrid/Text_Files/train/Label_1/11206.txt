#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int a[4005], b[4005];
vector<ll> c;
bitset<1000001> mark;

int main() {
  int m;
  cin >> m;
  for (int i = 0; i < m; ++i) {
    cin >> a[i];
    c.push_back(a[i]);
  }
  sort(c.begin(), c.end());
  c.resize(unique(c.begin(), c.end()) - c.begin());

  bool flag = false;
  for (int i = 0; i < m; ++i) {
    for (int j = 0; j < int(c.size()); ++j) {
      int gc = c[j];
      for (int k = j + 1; k < int(c.size()); ++k) {
        if (i > k) continue;
        gc = __gcd(a[i], gc);
        if (gc == c[k]) continue;
        b[i] = -1;
        flag = true;
        break;
      }
    }
  }
  if (flag) {
    cout << -1 << endl;
    return 0;
  }

  for (int ans = 1; 1LL * ans * (ans + 1) / 2 <= a[m - 1]; ++ans) {
    for (int i = 0; i < ans; ++i) { a[i] = 1LL * ans * (i + 1); }

    bool flag = true;
    for (int i = 0; i < m; ++i) {
      if (b[i] == -1) continue;

      int gc = a[b[i] - 1];
      mark.reset();
      for (int j = b[i]; j < ans; ++j) {
        gc = __gcd(a[j], gc);
        mark.set(gc);
        if (!mark.test(a[j])) {
          flag = false;
          break;
        }
      }
    }
    if (flag) {
      cout << ans << endl;
      for (int i = 0; i < ans; ++i) { cout << a[i] << " "; }
      cout << endl;
      return 0;
    }
  }

  cout << -1 << endl;
  return 0;
}
