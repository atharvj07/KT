#include<bits/stdc++.h>
using namespace std;

int v[505], n, a, b, c; 

int main(){
    cin >> n;
    for(int i = 1; i <= n; i++){
        cin >> a;
        v[a]++;
    }
    for(int i = 1; i <= n; i++){
        b += v[i] / 2;
        c += (v[i] % 2 != 0);
    }
    cout << b + c << endl;
    return 0;
}
