#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  if (s.size() != t.size()) {
    cout << "need tree" << endl;
  } else if (s == t) {
    cout << "array" << endl;
  } else {
    vector<int> cnt(26);
    for (char c : s) cnt[c - 'a']++;
    for (char c : t) cnt[c - 'a']--;
    int odd = 0;
    for (int x : cnt) odd += (x & 1);
    if (odd == 0 || odd == 2) {
      cout << "automaton" << endl;
    } else {
      cout << "both" << endl;
    }
  }
}
