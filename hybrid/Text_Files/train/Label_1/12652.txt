#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
int main(){
    string s;
    cin >> s;
    long long cnt_a = 0, ans = 0;
    for(char ch : s){
        if(ch == 'a') cnt_a++;
        else{
            ans = (ans + cnt_a * (cnt_a - 1) / 2) % mod;
        }
    }
    cout << ans << "\n";
    return 0;
}
