def solve(r: int, c: int, grid: list[str]) -> int:
    if any('P' in row for row in grid):
        return 'MORTAL'

    def floodfill(row, col):
        if row < 0 or row >= r or col < 0 or col >= c:
            return
        if grid[row][col] == 'A':
            return
        grid[row][col] = 'A'
        floodfill(row + 1, col)
        floodfill(row - 1, col)
        floodfill(row, col + 1)
        floodfill(row, col - 1)

    floodfill(0, 0)
    return grid.count('P')

t = int(input())
for _ in range(t):
    r, c = map(int, input().split())
    grid = [input() for _ in range(r)]
    print(solve(r, c, grid))
