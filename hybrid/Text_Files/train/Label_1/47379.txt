import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static class Node {
        int parent;
        List<Node> children;

        Node(int parent) {
            this.parent = parent;
            this.children = new ArrayList<>();
        }
    }

    static int n;
    static Node[] mainGrid;
    static Node[] reserveGrid;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();

        int a = scanner.nextInt();
        mainGrid = new Node[a + 1];
        for (int i = 2; i <= a; i++) {
            mainGrid[i] = new Node(scanner.nextInt());
            mainGrid[i].children.add(mainGrid[i - 1]);
        }
        int[] mainGridDevices = new int[n];
        for (int i = 0; i < n; i++) {
            mainGridDevices[i] = scanner.nextInt();
        }

        int b = scanner.nextInt();
        reserveGrid = new Node[b + 1];
        for (int i = 2; i <= b; i++) {
            reserveGrid[i] = new Node(scanner.nextInt());
            reserveGrid[i].children.add(reserveGrid[i - 1]);
        }
        int[] reserveGridDevices = new int[n];
        for (int i = 0; i < n; i++) {
            reserveGridDevices[i] = scanner.nextInt();
        }

        int maxWiresToRemove = 0;
        for (int i = 0; i < n; i++) {
            int mainGridDepth = getDepth(mainGrid, mainGridDevices[i]);
            int reserveGridDepth = getDepth(reserveGrid, reserveGridDevices[i]);
            maxWiresToRemove += Math.min(mainGridDepth, reserveGridDepth);
        }

        System.out.println(maxWiresToRemove);
    }

    static int getDepth(Node[] grid, int deviceNode) {
        int depth = 0;
        while (deviceNode != 1) {
            depth++;
            deviceNode = grid[deviceNode].parent;
        }
        return depth;
    }
}
