#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n);
  while (m--) {
    int a, b;
    cin >> a >> b;
    a--, b--;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  bool ok = true;
  for (int i = 0; i < n; i++) {
    vector<int> seen(n);
    for (int j : adj[i]) {
      int diff = (j - i + n) % n;
      if (seen[diff]) {
        ok = false;
        break;
      }
      seen[diff] = true;
    }
  }
  cout << (ok ? "Yes" : "No") << '\n';
}
