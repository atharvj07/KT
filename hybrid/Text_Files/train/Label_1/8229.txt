#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, s, k;
        cin >> n >> s >> k;
        set<int>x;
        while (k--) {
            int a;
            cin >> a;
            x.insert(a);
        }
        if (x.count(s)) {
            if (x.lower_bound(s) != x.end()) cout << *x.lower_bound(s) - s << '\n';
            else {
                auto it = x.begin();
                cout << *it - s << '\n';
            }
        } else {
            auto it = x.lower_bound(s);
            if (it == x.begin()) cout << 0 << '\n';
            else {
                it--;
                cout << s - *it << '\n';
            }
        }
    }
}
