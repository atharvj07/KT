def can_distribute_pencils(n, k, d, pencils):
    pencils.sort()
    dp = [0] * (n + 1)
    j = 0
    dp[0] = 1
    for i in range(1, n + 1):
        while pencils[i-1] - pencils[j] > d:
            j += 1
        if j > 0 and dp[j-1]:
            if i - j >= k:
                dp[i] = 1
    return "YES" if dp[n] else "NO"

n, k, d = map(int, input().split())
pencils = list(map(int, input().split()))
print(can_distribute_pencils(n, k, d, pencils))
