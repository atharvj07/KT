import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] colors = new int[n][2];
        for (int i = 0; i < n; i++) {
            colors[i][0] = in.nextInt();
            colors[i][1] = in.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }
                if (colors[i][0] == colors[j][1]) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
