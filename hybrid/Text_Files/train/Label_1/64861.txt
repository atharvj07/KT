def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def count_ambiguous_pairs(t, test_cases):
    results = []
    for m, d, w in test_cases:
        if m == 1 or d == 1:
            results.append(0)
            continue
        min_md = min(m, d)
        w_gcd = gcd(w, d - 1)
        period = w // w_gcd
        full_cycles = min_md // period
        remainder = min_md % period
        result = (full_cycles * (full_cycles - 1) // 2) * period + full_cycles * remainder
        results.append(result)
    return results

# Read input
t = int(input())
test_cases = [tuple(map(int, input().split())) for _ in range(t)]

# Solve and print results
results = count_ambiguous_pairs(t, test_cases)
for result in results:
    print(result)
