#include <bits/stdc++.h>
using namespace std;
long long n, a[100001], b[100001];
vector<long long> v[100001];
bool vis[100001];
vector<long long> res;
void dfs(long long x, long long par) {
  vis[x] = 1;
  bool ok = 1;
  for (auto u : v[x]) {
    if (u == par) continue;
    if (!vis[u]) {
      dfs(u, x);
      ok &= a[u];
      b[x] += b[u];
    }
  }
  a[x] = a[x] && ok;
  b[x] += a[x] ^ 1;
  if (!a[x] && b[x] == 0 && x != 1) res.push_back(x);
}
int main() {
  cin >> n;
  for (long long i = 1; i <= n; i++) {
    long long p, t;
    cin >> p >> t;
    v[p].push_back(i);
    a[i] = t;
  }
  dfs(1, 1);
  if (res.size() == 0)
    cout << -1 << endl;
  else {
    sort(res.begin(), res.end());
    for (auto x : res) cout << x << " ";
    cout << endl;
  }
}
