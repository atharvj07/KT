#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int lsb = 31 - __builtin_clz(a[n - 1]);
    vector<int> dp(1 << lsb), cnt(1 << lsb);
    dp[0] = 1;
    cnt[0] = 1;
    for (int i = 0; i < n; i++) {
        for (int j = (1 << lsb) - 1; j >= 0; j--) {
            if (dp[j] && cnt[j] < i + 1) {
                int k = j | (a[i] >> (31 - lsb));
                if (!dp[k]) {
                    dp[k] = 1;
                    cnt[k] = i + 1;
                }
            }
        }
    }
    int best = 0;
    for (int i = 0; i < (1 << lsb); i++) {
        if (dp[i]) {
            best = max(best, cnt[i]);
        }
    }
    cout << best << '\n';
    vector<int> ans;
    for (int i = n - 1; i >= 0; i--) {
        for (int j = (1 << lsb) - 1; j >= 0; j--) {
            if (dp[j] && cnt[j] == best && (a[i] >> (31 - lsb)) == j) {
                ans.push_back(a[i]);
                dp[j] = 0;
                cnt[j] = 0;
                best--;
                break;
            }
        }
    }
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i] << ' ';
    }
    cout << '\n';
}
