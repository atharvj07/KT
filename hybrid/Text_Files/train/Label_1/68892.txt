def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def construct_graph(n, m):
    edges = []
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            if gcd(i, j) == 1:
                edges.append((i, j))
            if len(edges) == m:
                break
        if len(edges) == m:
            break
    
    if len(edges) < m:
        return "Impossible"

    result = ["Possible"]
    for edge in edges:
        result.append(f"{edge[0]} {edge[1]}")
    return "\n".join(result)

n, m = map(int, input().split())
print(construct_graph(n, m))
