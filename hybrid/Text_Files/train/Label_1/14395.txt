#include <bits/stdc++.h>
using namespace std;
int n, a, b;
void f(int x, int y) {
  if (x == a && y == b) {
    cout << "Final!\n";
    exit(0);
  }
  if (x == y) return;
  int z = (n + 1) / 2;
  if (x != 1 && y != 1) f(x / z, y / z);
  if ((x / z) != (y / z)) f(x % z, y % z);
}
int main() {
  cin >> n >> a >> b;
  if (a == b) {
    cout << "Final!\n";
    return 0;
  }
  if (a > b) swap(a, b);
  f(a, b);
  cout << 1 << endl;
  return 0;
}
