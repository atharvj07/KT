import sys
input = sys.stdin.readline

def solve():
    a, b, p = map(int, input().split())
    s = input().strip()
    n = len(s)
    dp = [-1] * n
    dp[0] = 0
    for i in range(1, n):
        if s[i] == s[i - 1]:
            dp[i] = dp[i - 1]
        else:
            if s[i] == 'A':
                dp[i] = dp[i - 1] + a
            else:
                dp[i] = dp[i - 1] + b
    mn = 1e9
    ans = -1
    for i in range(n - 1, -1, -1):
        if dp[i] <= p:
            if dp[i] < mn:
                mn = dp[i]
                ans = i
    print(ans + 1)

t = int(input())
for _ in range(t):
    solve()
