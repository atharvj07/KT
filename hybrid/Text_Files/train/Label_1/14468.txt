#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    map<int, int> m;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        m[a[i]]++;
    }
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (m[a[i]] > 1) {
            while (m[cnt + 1] > 0) cnt++;
            a[i] = cnt + 1;
            m[a[i]]++;
            m[a[i] - 1]--;
        }
    }
    for (int i = 0; i < n; i++) {
        cout << a[i] << " ";
    }
    cout << endl;
    return 0;
}
