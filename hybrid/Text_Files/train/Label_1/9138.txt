#include <bits/stdc++.h>
using namespace std;

long long t, a[200005], n;

bool check(long long x)
{
    return t >= x;
}

int main()
{
    cin >> n >> t;
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    a[n + 1] = a[1];
    long long l = 1, r = 1e18, ans = 0;
    while (l <= r)
    {
        long long mid = (l + r) / 2;
        if (check(a[n] * mid))
        {
            ans = mid;
            l = mid + 1;
        }
        else
            r = mid - 1;
    }
    long long res = ans;
    t -= ans * a[n];
    for (int i = 1; i <= n && t >= *min_element(a + 1, a + n + 1); i++)
    {
        res += t / a[i];
        t -= (t / a[i]) * a[i];
    }
    cout << res << endl;
    return 0;
}
