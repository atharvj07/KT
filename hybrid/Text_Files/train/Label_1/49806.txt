import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

class Main {
  static final int MOD = 1000000007;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] nkq = br.readLine().split(" ");
    int n = Integer.parseInt(nkq[0]), q = Integer.parseInt(nkq[1]), t = Integer.parseInt(nkq[2]);
    String[] a = br.readLine().split(" ");
    int[] ai = Arrays.stream(a).mapToInt(Integer::parseInt).toArray();
    Map<Integer, Integer> bi = new HashMap<>(), ci = new HashMap<>();
    for (int i = 0; i < q; i++) {
      String[] bc = br.readLine().split(" ");
      bi.put(Integer.parseInt(bc[0]), i);
      ci.put(Integer.parseInt(bc[1]), i);
    }
    long[][] dp = new long[n + 1][t + 1];
    dp[0][0] = 1;
    for (int i = 1; i <= n; i++) {
      for (int j = 0; j <= t; j++) {
        dp[i][j] = dp[i - 1][j];
        if (j >= ai[i - 1]) {
          int biPos = (bi.containsKey(i) ? bi.get(i) : -1), ciPos = (ci.containsKey(i) ? ci.get(i) : -1);
          if (biPos == -1 || ciPos == -1 || biPos < ciPos) {
            dp[i][j] += dp[i - 1][j - ai[i - 1]];
          }
        }
        dp[i][j] %= MOD;
      }
    }
    System.out.println(dp[n][t]);
  }
}
