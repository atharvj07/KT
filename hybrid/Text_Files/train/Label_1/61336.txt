n, m, a = map(int, input().split())
b = list(map(int, input().split()))
p = list(map(int, input().split()))
p.sort()
b.sort()
r = 0
s = 0
while r < m and (a >= p[r] or s + b[r] >= p[r]):
    if a >= p[r]:
        r += 1
    else:
        a += b[r]
        s += b[r]
        b.pop(0)
        r += 1
print(r, s)
