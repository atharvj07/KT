def find_smaller_name(s, c):
    # If s is already smaller, no swap needed
    if s < c:
        return s
    # Try finding a swap that makes s smaller than c
    for i in range(len(s)):
        for j in range(i + 1, len(s)):
            # Swap characters at positions i and j
            swapped = list(s)
            swapped[i], swapped[j] = swapped[j], swapped[i]
            swapped = ''.join(swapped)
            # Check if the swapped string is smaller than c and s
            if swapped < c:
                return swapped
    # If no swap can make s smaller, return impossible
    return "---"

# Read the number of test cases
t = int(input().strip())
for _ in range(t):
    s, c = input().strip().split()
    print(find_smaller_name(s, c))
