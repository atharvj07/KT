#include<bits/stdc++.h>
using namespace std;
int f[6],a[500005],ans=1e9;
int main()
{
	int n;cin>>n;
	for(int i=1;i<=n;i++)
	{
		cin>>a[i];
		f[a[i]/4]++;
	}
	ans=min(ans,f[2]+f[6]);
	ans=min(ans,f[1]+f[3]+f[5]);
	if(f[1]%3==0) ans=min(ans,f[4]+f[5]);
	if(f[1]%3==1) ans=min(ans,f[4]+f[6]);
	if(f[1]%3==2) ans=min(ans,f[5]+f[6]);
	if(ans==f[6]+f[2]) ans+=f[1]%3;
	if(ans==f[1]+f[3]+f[5]) ans+=f[4];
	if(ans==f[4]+f[5]) ans+=(f[1]-f[1]%3)/3;
	if(ans==f[4]+f[6]) ans+=(f[1]-f[1]%3-1)/3+1;
	if(ans==f[5]+f[6]) ans+=(f[1]-f[1]%3-2)/3+2;
	cout<<ans<<endl;
}
