#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    unordered_set<string> vis;
    cin >> s;
    int ans = 0;
    string cur = "0,0";
    for (char c : s) {
      if (c == 'N') cur[0]++;
      else if (c == 'S') cur[0]--;
      else if (c == 'E') cur[2]++;
      else if (c == 'W') cur[2]--;
      if (vis.find(cur) == vis.end()) {
        ans += 5;
      } else {
        ans += 1;
      }
      vis.insert(cur);
    }
    cout << ans << endl;
  }
}
