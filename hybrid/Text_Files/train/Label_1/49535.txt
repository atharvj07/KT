import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.LinkedList;
import java.util.Queue;

class Main {
    private static int n, q;
    private static boolean[][] spe, part;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(in.readLine());
        par = new int[n + 1];
        type = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            String[] line = in.readLine().split(" ");
            par[i] = Integer.parseInt(line[0]);
            type[i] = Integer.parseInt(line[1]);
        }
        q = Integer.parseInt(in.readLine());
        spe = new boolean[n + 1][n + 1];
        part = new boolean[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            bfs(i, i, type[i]);
        }
        for (int i = 0; i < q; i++) {
            String[] line = in.readLine().split(" ");
            int t = Integer.parseInt(line[0]);
            int u = Integer.parseInt(line[1]);
            int v = Integer.parseInt(line[2]);
            if (t == 1) {
                System.out.println(spe[u][v] ? "YES" : "NO");
            } else {
                System.out.println(part[u][v] ? "YES" : "NO");
            }
        }
    }

    private static int[] par, type;

    private static void bfs(int s, int cur, int t) {
        Queue<Integer> q = new LinkedList<>();
        q.add(s);
        while (!q.isEmpty()) {
            int p = q.poll();
            if (t == 0) {
                spe[p][cur] = true;
                for (int i = 1; i <= n; i++) {
                    if (i == p) continue;
                    if (par[i] == p) {
                        q.add(i);
                    }
                }
            } else {
                part[p][cur] = true;
                if (par[p] > 0) {
                    q.add(par[p]);
                }
            }
        }
    }
}
