#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int M=2e5, N=1e9;
ll a[M], b[M];
ll dp[M][1010];

int main() {
    int tt;
    cin>>tt;
    while(tt--) {
        int n, m;
        cin>>n>>m;
        for(int i=0;i<m;i++)
            cin>>a[i]>>b[i];
        for(int j=1;j<=n;j++)
            for(int i=0;i<m;i++)
                dp[j][i]=max(dp[j][i], dp[j-1][i]);
        for(int j=1;j<=n;j++) {
            for(int i=0;i<m;i++) {
                dp[j][i+1]=max(dp[j][i+1], dp[j-1][i]+a[i]+(j-1)*b[i]);
            }
        }
        cout<<dp[n][m]<<'\n';
    }
    return 0;
}
