#include<iostream>
#include<vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> s(n);
    for (int i = 0; i < n; i++) {
        cin >> s[i];
    }
    int ans = 0;
    int last = -1;
    for (int i = 0; i < n; i++) {
        if (s[i] == 0) {
            if (last == -1) {
                ans++;
            }
        } else {
            last = i;
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
