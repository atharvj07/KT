n, k = map(int, input().split())
a = list(map(int, input().split()))

operations = {}
for number in a:
    current = number
    ops = 0
    while current > 0:
        if current in operations:
            operations[current].append(ops)
        else:
            operations[current] = [ops]

        current //= 2
        ops += 1

min_ops = float('inf')
for key, value in operations.items():
    value.sort()
    if len(value) >= k:
        min_ops = min(min_ops, sum(value[:k]))

print(min_ops)
