#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main()
{
    ll n,d,ans=0,cur=0;
    cin>>n>>d;
    vector<ll>v(n+1);
    for(ll i=1;i<=n;i++)
    {
        cin>>v[i];
        if(v[i]==0)
        {
            if(cur<0)
            {
                ans++;
                cur+=d;
            }
        }
        cur+=v[i];
        if(cur>d)
        {
            cout<<"-1";
            return 0;
        }
    }
    cout<<ans;
    return 0;
}
