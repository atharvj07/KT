const int N = 1e3;

int32_t main() {
	int t = 0; cin >> t;
	while (t--) {
		int n = 0, m = 0; cin >> n >> m;
		int sum = 0; for (int i = 0; i < n; i++) { int x; cin >> x; sum += x; }
		int avg = sum / n;
		cout << m - avg + sum % n << "\n";
	}
}
