import sys

n = int(input())
a = list(map(int, input().split()))
if sum(a[1:]) % a[0] != 0:
    print('NO')
    sys.exit()
else:
    print('YES')

x = sum(a[1:]) // a[0]
b = []
for i in range(1, n):
    if a[i] < x:
        b.append((1, i + 1, n, a[i] ^ x))
        a[i] = x
    elif a[i] > x:
        b.append((i + 1, n, 1, a[i] ^ x))
        a[i] = x

print(len(b))
for i in b:
    print(*i)
