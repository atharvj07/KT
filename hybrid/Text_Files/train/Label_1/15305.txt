#include <bits/stdc++.h>

using namespace std;

struct Edge {
  int u, v, w;
  Edge() {}
  Edge(int u, int v, int w) : u(u), v(v), w(w) {}
};

vector<Edge> edges;
vector<int> p, sz, ans;

int get(int x) {
  return (x == p[x] ? x : p[x] = get(p[x]));
}

void unite(int x, int y) {
  x = get(x);
  y = get(y);
  if (x == y) return;
  if (sz[x] > sz[y]) {
    p[y] = x;
    sz[x] += sz[y];
  } else {
    p[x] = y;
    sz[y] += sz[x];
  }
}

int kruskal(vector<Edge> &edges, int cost) {
  int ans = 0;
  p.resize(edges.size() + 1);
  iota(begin(p), end(p), 0);
  sz.resize(edges.size() + 1, 1);
  sort(begin(edges), end(edges), [](Edge &a, Edge &b) { return a.w < b.w; });
  for (Edge &e : edges) {
    if (get(e.u) != get(e.v)) {
      unite(e.u, e.v);
      ans += e.w;
      if (ans == cost) break;
    }
  }
  return ans;
}

int main() {
  int n, m, k = 0;
  cin >> n >> m;
  p.resize(n + 1);
  iota(begin(p), end(p), 0);
  sz.resize(n + 1, 1);
  for (int i = 0; i < m; ++i) {
    int u, v, w;
    cin >> u >> v >> w;
    k += w;
  }
  int l = 0, r = 1e9, mid;
  while (l < r) {
    mid = (l + r) >> 1;
    int cur = kruskal(edges, k + mid);
    if (cur == k)
      r = mid;
    else
      l = mid + 1;
  }
  cout << l << endl;
  return 0;
}
