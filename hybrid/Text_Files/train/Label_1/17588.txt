#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, c;
    cin >> n >> c;
    vector<int> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
    }
    int ans = 1, t = v[0];
    for (int i = 1; i < n; i++) {
        if (v[i] <= t + c) {
            ans++;
        } else {
            ans = 1;
        }
        t = v[i];
    }
    cout << ans << '\n';
    return 0;
}
