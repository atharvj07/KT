import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }
        Arrays.sort(c);
        int max = 0;
        for (int i = 0; i < n; i++) {
            if (c[i] > 0) {
                max += c[i];
            }
        }
        for (int i = 0; i < n; i++) {
            if (max % c[i] == 0) {
                System.out.print(c[i] + " ");
            }
        }
        System.out.println();
    }
}
