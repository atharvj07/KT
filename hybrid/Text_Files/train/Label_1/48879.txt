import java.util.*;

class Main {
    static int n, m;
    static int[] dist;
    static List<Integer>[] graph;

    static void bfs(int s) {
        Queue<Integer> Q = new LinkedList<Integer>();
        Q.add(s);
        dist[s] = 0;
        while (!Q.isEmpty()) {
            int v = Q.poll();
            for (int u : graph[v]) {
                if (dist[u] > dist[v] + 1) {
                    dist[u] = dist[v] + 1;
                    Q.add(u);
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        int s = in.nextInt() - 1;
        int t = in.nextInt() - 1;
        dist = new int[n];
        Arrays.fill(dist, Integer.MAX_VALUE);
        graph = new List[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<Integer>();
        }
        while (m-- > 0) {
            int u = in.nextInt() - 1;
            int v = in.nextInt() - 1;
            graph[u].add(v);
            graph[v].add(u);
        }
        bfs(s);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (graph[i].contains(j)) {
                    continue;
                }
                if (dist[i] + 1 == dist[j] || dist[j] + 1 == dist[i]) {
                    continue;
                }
                ans++;
            }
        }
        System.out.println(ans);
    }
}
