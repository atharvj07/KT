#include <cstdio>
#include <vector>
using namespace std;
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n;
    scanf("%d", &n);
    vector<int> a(n + 1);
    int s = 0;
    for (int i = 1; i <= n; i++) {
      scanf("%d", &a[i]);
      s += a[i];
    }
    if (s & 1)
      printf("NO\n");
    else
      printf("YES\n");
  }
  return 0;
}
