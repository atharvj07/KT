#include <iostream>
#include <string>

int main() {
  int n;
  std::cin >> n;
  std::string s;
  std::cin >> s;
  int ans = 0;
  for (int i = 0; i < n / 2; i++) {
    if (s[i] != s[n - i - 1]) {
      ans++;
      int j = i;
      while (s[j] != s[n - i - 1]) j++;
      for (int k = j; k > i; k--) {
        std::swap(s[k], s[k - 1]);
      }
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
