import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int d = sc.nextInt(), w = sc.nextInt();
            if (d == 0 && w == 0) {
                break;
            }
            int[][] map = new int[d][w];
            for (int i = 0; i < d; i++) {
                for (int j = 0; j < w; j++) {
                    map[i][j] = sc.nextInt();
                }
            }
            int maxCapacity = 0;
            for (int i = 1; i < d - 1; i++) {
                for (int j = 1; j < w - 1; j++) {
                    int minOuter = Integer.MAX_VALUE;
                    for (int k = i - 1; k <= i + 1; k++) {
                        for (int l = j - 1; l <= j + 1; l++) {
                            if (k != i || l != j) {
                                minOuter = Math.min(minOuter, map[k][l]);
                            }
                        }
                    }
                    if (minOuter > map[i][j]) {
                        maxCapacity = Math.max(maxCapacity, minOuter - map[i][j]);
                    }
                }
            }
            System.out.println(maxCapacity);
        }
    }
}
