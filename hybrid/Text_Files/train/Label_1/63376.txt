from collections import deque, defaultdict

def dijkstra(graph, start):
  dist = {v: float('inf') for v in graph}
  dist[start] = 0
  q = deque([start])
  while q:
    v = q.popleft()
    for u in graph[v]:
      alt = dist[v] + 1
      if alt < dist[u]:
        dist[u] = alt
        q.append(u)
  return dist

n, m = map(int, input().split())
graph = defaultdict(list)
for _ in range(m):
  x, y, z = map(int, input().split())
  graph[x].append(y)
  graph[y].append(x)

dist = dijkstra(graph, 1)
path = [n]
v = n
while v != 1:
  for u in graph[v]:
    if dist[u] == dist[v] - 1:
      path.append(u)
      v = u
      break

k = 0
for i in range(1, len(path)):
  x, y = path[i - 1], path[i]
  if (x, y, 0) in graph[(x, y, 1)]:
    k += 1
    graph[(x, y, 1)].remove((x, y, 0))
  elif (y, x, 0) in graph[(x, y, 1)]:
    k += 1
    graph[(x, y, 1)].remove((y, x, 0))
  elif (x, y, 1) not in graph:
    k += 1
    graph[x].append((y, 0))
    graph[y].append((x, 0))

print(k)
for (x, y, z) in graph[(x, y, 1)]:
  print(x, y, z)
