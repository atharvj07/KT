#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define all(s) s.begin(), s.end()

int main() {
    int n, m, k;
    cin >> n >> m >> k;
    ll a[m + 1], f;
    cin >> f;
    int cnt = 0;
    for (int i = 0; i < m; i++) {
        cin >> a[i];
        ll tmp = f ^ a[i];
        int c = __builtin_popcountll(tmp);
        if (c <= k) cnt++;
    }
    cout << cnt << endl;
    return 0;
}
