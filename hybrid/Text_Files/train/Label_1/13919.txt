#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;

  std::vector<int> rates(n);
  int sum = 0;
  int mx = 0;
  for (int i = 0; i < n; ++i) {
    std::cin >> rates[i];
    sum += rates[i];
    mx = std::max(mx, rates[i]);
  }
  std::sort(rates.begin(), rates.end());

  int r = sum / n;
  if (r == mx) {
    std::cout << r << '\n';
    std::cout << 0 << '\n';
    return 0;
  }

  mx = rates[n - 1];
  int x = mx - r;
  std::vector<int> pos;
  for (int i = 0; i < n; ++i)
    if (rates[i] == mx) {
      pos.push_back(i + 1);
    }

  int nl = pos.size();
  int rem = x * nl;
  int needs = rem + (r == 0 ? nl : nl - 1);
  std::cout << r << '\n';
  std::cout << needs << '\n';
  for (int i = 0; i < n; ++i) {
    int p = i + 1;
    int cnt = 0;
    if (rates[i] == mx) {
      while (cnt < x) {
        for (int j = 0; j < nl; ++j) {
          if (pos[j] == p) {
            ++cnt;
            std::cout << '1';
          } else {
            std::cout << '0';
          }
        }
        --needs;
        if (needs == 0) return 0;
        std::cout << '\n';
      }
      pos.erase(std::remove(pos.begin(), pos.end(), p), pos.end());
    } else {
      while (needs > 0) {
        --needs;
        for (int j = 0; j < nl; ++j)
          std::cout << '0';
        std::cout << '1';
        std::cout << '\n';
      }
    }
  }
  return 0;
}
