#include<bits/stdc++.h>
using namespace std;
int n;
bool a[100005];
int main(){
    cin>>n;
    if(n==2){cout<<"3\n2 1 2\n";return 0;}
    int p=0;
    cout<<n<<"\n";
    for(int i=1;i<=n;i++){
        if((i&1)==1){
            cout<<i+1<<" ";
            a[i+1]=1;
            if(i+2<=n)a[i+2]=1;
        }
        else{
            if(p!=1){
                cout<<i<<" ";
                a[i]=1;
                p++;
            }
            else{
                cout<<i+1<<" ";
                a[i+1]=1;
                if(i+2<=n)a[i+2]=1;
            }
        }
    }
    return 0;
}
