def count_k_interesting_pairs(n, k, sequence):
    def bits_diff(x, y):
        return bin(x ^ y).count('1')
    
    count = 0
    for i in range(n):
        for j in range(i+1, n):
            if bits_diff(sequence[i], sequence[j]) == k:
                count += 1
    return count

n, k = map(int, input().split())
sequence = list(map(int, input().split()))
print(count_k_interesting_pairs(n, k, sequence))
