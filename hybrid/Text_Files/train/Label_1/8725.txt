#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll n,m,ans=2e9,v,t[205][205];
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++) cin>>t[i][0],t[i][0]=__builtin_popcountl(t[i][0]);
    for(int i=1;i<=m;i++) cin>>t[0][i],t[0][i]=__builtin_popcountl(t[0][i]);
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            v=__builtin_popcountl(t[i][0]&t[0][j]);
            ans=min(ans,v);
        }
    }
    cout<<ans<<endl;
    return 0;
}
