#include <iostream>

using namespace std;

int main() {
    int a1, b1, c1, a2, b2, c2, d1, d2, d3;
    cin >> a1 >> b1 >> c1 >> a2 >> b2 >> c2 >> d1 >> d2 >> d3;
    int ans = (d2 > d3 && b1 < b2 ? (b2 - b1 + d2 - 1) / d2 : 0) + (d1 > d3 && a1 < a2 ? (a2 - a1 + d1 - 1) / d1 : 0);
    cout << ans << endl;
    return 0;
}
