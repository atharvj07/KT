#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007

long long bigMod(long long base, long long pow){
    long long ret = 1;
    while(pow){
        if(pow&1==1) ret=(ret*base)%mod;
        base=(base*base)%mod;
        pow>>=1;
    }
    return ret;
}

int main()
{
    string s;
    cin>>s;
    long long ans=0;
    long long last=1;
    for(int i=s.size()-1; i>=0; i--){
        ans=(ans+((last*((s[i]-'0')%mod))%mod))%mod;
        last=((last*4)%mod);
        last=(last+mod)%mod;
    }
    cout<<ans<<endl;
    return 0;
}
