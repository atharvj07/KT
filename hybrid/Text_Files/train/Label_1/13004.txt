#include <bits/stdc++.h>
using namespace std;
const int M = 200005;
const int N = 45;
string s[M];
char dir[M];
int n, st[N], fi[N];
const char *dr = "udlr";
#define endl "\n"
bool can() {
  int cnt = 0;
  for (int i = 1; i <= n; i++) { cnt += fi[i] < st[i]; }
  return cnt % 2 == 0;
}
void dfs(int id) {
  if (id >= (int)s[n].size()) {
    if (can()) {
      cout << "SURGERY COMPLETE" << endl;
      int i = 0, j = 0;
      while (i < n) {
        for (int k = 1; k <= n; k++) {
          for (int t = 1; t <= fi[k]; t++) {
            if (s[i][j] == 'E') { j++; } else if (st[k] > fi[k]) {
              s[i][j] = s[i][j + 1];
              s[i][j + 1] = 'E';
              dir[id + j * n + i] = 'r';
              j++;
              t--;
            } else {
              s[i][j] = s[i + 1][j];
              s[i + 1][j] = 'E';
              dir[id + j * n + i] = 'd';
              i++;
              t--;
            }
          }
          j -= fi[k], fi[k] = n;
        }
        i++, j = 0;
      }
      string str;
      for (int i = 0; i < n; i++) {
        int l = 0, r = 0;
        for (int j = 0; j < n; j++) {
          if (s[i][j] != 'E') l = j;
        }
        for (int j = n - 1; j >= 0; j--) {
          if (s[i][j] != 'E') r = j;
        }
        while (l < r) {
          str += 'r';
          l++;
        }
      }
      for (int i = 0; i < n; i++) {
        int l = 0, r = 0;
        for (int j = 0; j < n; j++) {
          if (s[j][i] != 'E') l = j;
        }
        for (int j = n - 1; j >= 0; j--) {
          if (s[j][i] != 'E') r = j;
        }
        while (l < r) {
          str += 'd';
          l++;
        }
      }
      for (int i = 0; i < str.size(); i++) {
        if (str[i] != str[(i + 1) % str.size()]) {
          int cnt = str.size() / str.substr(i, str.size()).size();
          char C = str[(i + 1) % str.size()];
          cout << C << ' ' << cnt << endl;
        }
      }
      cout << "DONE" << endl;
    } else {
      cout << "SURGERY FAILED" << endl;
    }
    exit(0);
  }
  int i = id % n, j = id / n;
  for (int k = 0; k < 4; k++) {
    int ni = i + dir[k] - 'l', nj = j;
    if (ni < 0 || ni >= n || nj < 0 || nj >= n) continue;
    if (st[s[j][i] - '0'] >= fi[s[j][i] - '0']) continue;
    swap(s[j][i], s[nj][ni]);
    fi[s[j][i] - '0']++;
    st[s[nj][ni] - '0']--;
    dfs(id + 1);
    swap(s[j][i], s[nj][ni]);
    fi[s[j][i] - '0']--;
    st[s[nj][ni] - '0']++;
  }
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 0; i < n; i++) {
      cin >> s[i];
      for (int j = 0; j < n; j++) {
        if (s[i][j] != 'E') st[s[i][j] - '0']++;
      }
    }
    for (int i = 0; i < n; i++) {
      cin >> s[i];
      for (int j = 0; j < n; j++) {
        if (s[i][j] != 'E') fi[s[i][j] - '0']++;
      }
    }
    dfs(0);
  }
  return 0;
}
