#include <iostream>

using namespace std;

int main() {
    int m, h1, a1, x1, y1, h2, a2, x2, y2;
    cin >> m >> h1 >> a1 >> x1 >> y1 >> h2 >> a2 >> x2 >> y2;
    int x = (a2 - h2) % m;
    int y = (a1 - h1) % m;
    if ((x2 - x1) * y != (y2 - y1) * x) {
        cout << -1 << endl;
        return 0;
    }
    int num = (y2 - y1) * h1 - (x2 - x1) * h2;
    int den = (x2 - x1) * y - (y2 - y1) * x;
    cout << (num % den + den) % den << endl;
    return 0;
}
