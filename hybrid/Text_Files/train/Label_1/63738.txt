import sys
input = sys.stdin.readline
# 200001
arr = [0] * 200001
L = [0] * 200001
R = [0] * 200001
cnt = [0] * 200001
val = [0] * 200001
N, Q = list(map(int, input().split()))
A = [0] + list(map(int, input().split()))
not_nice = []
for i in range(1, N + 1):
  if L[A[i]] == 0:
    L[A[i]] = i
    R[A[i]] = i
  else:
    R[A[i]] = i
  cnt[A[i]] += 1
  if cnt[A[i]] > 1:
    if val[A[i]] == 0:
      not_nice.append(A[i])
    del val[val[A[i]]]
  val[A[i]] = i
if len(not_nice) == 0:
  print(0)
else:
  print(min(cnt[i] for i in not_nice))
for _ in range(Q):
  I, X = list(map(int, input().split()))
  cnt[A[I]] -= 1
  if cnt[A[I]] == 0:
    del val[val[A[I]]]
    if val[X] == 0:
      not_nice.remove(A[I])
    else:
      # print(val[A[I]], val[X], L[A[I]], R[A[I]], L[X], R[X])
      if R[X] < L[A[I]] or L[X] > R[A[I]]:
        not_nice.remove(A[I])
  L[X] = min(L[X], L[A[I]])
  R[X] = max(R[X], R[A[I]])
  A[I] = X
  cnt[X] += 1
  if cnt[X] == 1:
    val[X] = I
  if cnt[X] == 2:
    # print(cnt)
    not_nice.append(X)
  if len(not_nice) == 0:
    print(0)
  else:
    print(min(cnt[i] for i in not_nice))
