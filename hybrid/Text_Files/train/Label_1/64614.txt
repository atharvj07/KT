n=input()
l=[]
for i in range(int(n)):
    l.append(input())
l.sort(key=lambda i: len(i),reverse=True)
lnof = 0
lnot = 0
answer = ""
flag = True
for i in l:
    if len(set(list(i))) == 1 and lnof == 0:
        answer += i
        lnof += 1
        flag = False
    elif len(set(list(i))) == 2 and lnot == 0:
        answer += i
        lnot += 1
        flag = False
    elif len(set(list(i))) == 1 and lnof == 1 and flag:
        answer += i
    elif len(set(list(i))) == 2 and lnot == 1 and flag:
        answer += i
    flag = True
    
print(len(answer))
