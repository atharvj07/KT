#include <iostream>
#include <unordered_set>
using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_set<string> seen;
  for (int i = 0; i < n; i++) {
    string color;
    cin >> color;
    if (color == "purple") {
      seen.insert("Power");
    } else if (color == "green") {
      seen.insert("Time");
    } else if (color == "blue") {
      seen.insert("Space");
    } else if (color == "orange") {
      seen.insert("Soul");
    } else if (color == "red") {
      seen.insert("Reality");
    } else if (color == "yellow") {
      seen.insert("Mind");
    }
  }
  cout << 6 - seen.size() << endl;
  for (string gem : {"Time", "Mind", "Soul", "Power", "Reality", "Space"}) {
    if (!seen.count(gem)) {
      cout << gem << endl;
    }
  }
  return 0;
}
