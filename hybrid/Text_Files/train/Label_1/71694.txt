import sys

def main():
    n = int(input())
    b = [int(x) for x in input().split()]
    c = [int(x) for x in input().split()]
    for i in range(n - 1):
        if b[i] > c[i]:
            b[i], c[i] = c[i], b[i]
    a = [b[0]]
    for i in range(n - 1):
        if b[i] != a[-1]:
            a.append(b[i])
        else:
            a.append(c[i])
    print(*a)

main()
