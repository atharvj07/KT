import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
if n - a[0] - b[0] > 0:
    print(-1)
else:
    d = [0]
    j = 0
    while j < n and d[-1] >= n - a[j + 1] - b[j + 1]:
        d.append(max(0, d[-1] + a[j + 1] - b[j + 1]))
        j += 1
    if d[-1] >= n:
        print(j)
        print(*d[:-1])
    else:
        print(-1)
