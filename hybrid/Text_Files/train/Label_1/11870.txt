#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAX_SIZE = 2000;
const char BLACK = '.';
const char WHITE = 'w';
const char FRAME = '+';

int main() {
    int n, m;
    cin >> n >> m;

    vector<string> screen(n);
    for (int i = 0; i < n; i++) {
        cin >> screen[i];
    }

    // Find the smallest possible frame that contains all white pixels.
    int min_size = MAX_SIZE;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (screen[i][j] == BLACK) {
                continue;
            }

            // Check if this pixel can be part of the frame.
            bool can_be_frame = true;
            if (i > 0 && screen[i - 1][j] == WHITE) {
                can_be_frame = false;
            }
            if (i < n - 1 && screen[i + 1][j] == WHITE) {
                can_be_frame = false;
            }
            if (j > 0 && screen[i][j - 1] == WHITE) {
                can_be_frame = false;
            }
            if (j < m - 1 && screen[i][j + 1] == WHITE) {
                can_be_frame = false;
            }

            if (can_be_frame) {
                int size = 1;
                if (i > 0 && screen[i - 1][j] == BLACK) {
                    size++;
                }
                if (i < n - 1 && screen[i + 1][j] == BLACK) {
                    size++;
                }
                if (j > 0 && screen[i][j - 1] == BLACK) {
                    size++;
                }
                if (j < m - 1 && screen[i][j + 1] == BLACK) {
                    size++;
                }
                min_size = min(min_size, size);
            }
        }
    }

    // If no frame exists, print -1.
    if (min_size == MAX_SIZE) {
        cout << -1 << endl;
        return 0;
    }

    // Print the screen with the frame.
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (screen[i][j] == WHITE) {
                cout << WHITE;
            } else if (screen[i][j] == BLACK && min_size == 1) {
                cout << FRAME;
            } else {
                cout << BLACK;
            }
        }
        cout << endl;
    }

    return 0;
}
