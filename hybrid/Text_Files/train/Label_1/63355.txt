import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
m = int(input())
g = [[] for i in range(n+1)]
for i in range(m) :
    x, y = map(int, input().split())
    g[x].append(y)
    g[y].append(x)

dep = [-1 for i in range(n+1)]
def dfs(x) :
    if dep[x] != -1 :
        return
    dep[x] = 1
    for y in g[x] :
        if dep[y] == -1 :
            dfs(y)
            dep[x] = max(dep[x], dep[y]+1)

ans1 = 0
ans2 = 0
for i in range(1, n+1) :
    if dep[i] == -1 :
        dfs(i)
        cur = 0
        for j in range(1, n+1) :
            if dep[j] == dep[i] :
                cur += a[j-1]
        ans1 += min(cur, sum(a)-cur)
        if cur <= sum(a)-cur :
            ans2 += 1
print(ans1, ans2)
