#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;

    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    sort(a.begin(), a.end());

    int ans = 0;
    for (int i = 0; i < n; i += 2) {
        ans += (a[i] + a[i + 1]) * (a[i] + a[i + 1]);
    }

    cout << ans << endl;

    return 0;
}
