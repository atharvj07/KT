from sys import stdin

def make_songs(idx, so_far, genre_played, mod):
    if idx == n: return 1 if so_far == t else 0
    if memo[idx][so_far][genre_played] != -1: return memo[idx][so_far][genre_played]

    total = make_songs(idx + 1, so_far, genre_played, mod)
    if genre[idx] != genre_played:
        total = (total + make_songs(idx + 1, so_far + song[idx], genre[idx], mod)) % mod
    memo[idx][so_far][genre_played] = total
    return total

n, t = map(int, stdin.readline().split())
song = []
genre = []
for _ in range(n):
    tmp_song, tmp_genre = map(int, stdin.readline().split())
    song.append(tmp_song)
    genre.append(tmp_genre)
