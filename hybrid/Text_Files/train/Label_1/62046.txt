from collections import deque
n, k = map(int, input().split())
if n > k + 1 or n % 2 == k % 2:
    print("NO")
else:
    print("YES")
    lst = list(range(1, k + 1)) * (n // 2) + list(range(2, k + 1)) * (n % 2)
    ans = deque(lst[:n])
    lst.reverse()
    for el in lst:
        ans.appendleft(el)
    for el in ans:
        print(*el)
