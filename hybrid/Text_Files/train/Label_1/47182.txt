import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) throws Exception {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();

        Set<Character> possibleLetters = new HashSet<>();
        for (int i = 0; i < 26; i++) {
            possibleLetters.add((char) ('a' + i));
        }

        int excessiveShocks = 0;
        for (int i = 0; i < n; i++) {
            String action = scanner.nextLine();
            char firstChar = action.charAt(0);

            if (firstChar == '.') {
                // Word spoken without shock, remove letters not in the word
                String word = action.substring(2);
                for (char c : word.toCharArray()) {
                    possibleLetters.remove(c);
                }
            } else if (firstChar == '!') {
                // Word spoken with shock, remove all other letters
                possibleLetters.clear();
            } else if (firstChar == '?') {
                // Guess made, check if only one letter remains
                char guess = action.charAt(2);
                if (possibleLetters.size() == 1 && possibleLetters.contains(guess)) {
                    // Letter correctly guessed, no excessive shocks
                } else {
                    // Incorrect guess, shock is excessive
                    excessiveShocks++;
                }
                break; // Stop after the last guess
            }
        }

        System.out.println(excessiveShocks);
    }
}
