#include <iostream>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int gpos;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'G') gpos = i;
    }
    if (gpos % k == 0 || (n - gpos - 1) % k == 0) cout << "YES";
    else cout << "NO";
    return 0;
}
