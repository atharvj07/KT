#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, a, b, k;
  cin >> n >> a >> b >> k;
  vector<long long> h(n);
  for (auto &x : h) cin >> x;
  vector<int> dp(n + 1);
  for (int i = n - 1; i >= 0; --i) {
    if (h[i] % (a + b) == 0) {
      dp[i] = dp[i + 1];
      continue;
    }
    if (h[i] % (a + b) <= a) {
      dp[i] = dp[i + 1] + 1;
      continue;
    }
    if (h[i] % (a + b) > a && k) {
      dp[i] = max(dp[i + 1] + 1, dp[i + 1]);
      --k;
    } else {
      dp[i] = dp[i + 1];
    }
  }
  cout << dp[0] << endl;
}
