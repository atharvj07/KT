import datetime as dt

n = int(input())
reqs = []
for _ in range(n):
    _, _, p, d = map(int, input().split())
    reqs.append([d, p, dt.date(dt.date.today().year-1 if d < 15 else dt.date.today().year, p, d) - dt.timedelta(d-1)])

reqs.sort()
t = j = 0
for d, p, _ in reqs:
    if d <= t: p = max(p - j, 0)
    j = max(p, j)
    t = d + j
print(j)
