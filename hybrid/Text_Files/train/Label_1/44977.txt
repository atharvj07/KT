import java.util.Arrays;
import java.util.Scanner;

public class TeamFormation {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int x = scanner.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
            }

            int maxTeams = solve(a, x);
            System.out.println(maxTeams);
        }
        scanner.close();
    }

    static int solve(int[] a, int x) {
        Arrays.sort(a); // Sort programmers by skill in ascending order

        int i = 0, j = 0, maxTeams = 0;
        while (i < a.length && j < a.length) {
            if (a[j] * (j - i + 1) >= x) { // Check if current team meets the minimum skill requirement
                maxTeams++;
                i = j + 1; // Move to the next potential team
            } else {
                j++; // Increase the team size by adding the next programmer
            }
        }

        return maxTeams;
    }
}
