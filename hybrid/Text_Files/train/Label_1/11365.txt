#include <bits/stdc++.h>

using namespace std;

const int mod = 1e6 + 3;

int main() {
  int n, c;
  cin >> n >> c;
  vector<int> dp(n + 1);
  dp[0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= c; j++) {
      dp[i] = (dp[i] + dp[i - j]) % mod;
    }
  }
  cout << dp[n] << endl;
  return 0;
}
