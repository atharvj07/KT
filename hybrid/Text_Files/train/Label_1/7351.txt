#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;
typedef long long ll;
typedef pair<ll, ll> pi;

int main() {
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<pi> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i].first >> a[i].second;
        }
        sort(a.begin(), a.end());
        ll ans = 0;
        for (int i = 1; i < n; i++) {
            if (a[i].first == a[i - 1].first) {
                ans += min(a[i].second, a[i - 1].second);
                a[i].first++;
            }
        }
        cout << ans << '\n';
    }
}
