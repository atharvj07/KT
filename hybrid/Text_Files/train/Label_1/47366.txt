import java.util.*;

public class ArithmeticProgression {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        List<Integer> firstProgression = new ArrayList<>();
        List<Integer> secondProgression = new ArrayList<>();

        int[] diff = new int[n - 1];
        for (int i = 1; i < n; i++) {
            diff[i - 1] = a[i] - a[i - 1];
        }

        boolean possible = true;
        int commonDiff = diff[0];
        for (int i = 1; i < n - 1; i++) {
            if (diff[i] != commonDiff) {
                possible = false;
                break;
            }
        }

        if (possible) {
            firstProgression.add(a[0]);
            secondProgression.add(a[1]);
            for (int i = 2; i < n; i++) {
                if ((a[i] - a[i - 1]) == commonDiff) {
                    firstProgression.add(a[i]);
                } else {
                    secondProgression.add(a[i]);
                }
            }

            System.out.println(firstProgression);
            System.out.println(secondProgression);
        } else {
            System.out.println("No solution");
        }
    }
}
