#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct block {
    int l, r, sum;
    block() {}
    block(int _l, int _r, int _sum) : l(_l), r(_r), sum(_sum) {}
    bool operator<(const block& other) const {
        return l < other.l;
    }
};

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<block> blocks;
    int sum = 0;
    int l = 0;
    for (int r = 0; r < n; r++) {
        sum += a[r];
        if (r == n - 1 || sum != a[r + 1]) {
            blocks.push_back(block(l, r, sum));
            l = r + 1;
            sum = 0;
        }
    }
    sort(blocks.begin(), blocks.end());
    int k = 0;
    int best_sum = -1e9;
    for (block b : blocks) {
        if (b.sum > best_sum) {
            best_sum = b.sum;
            k++;
        }
    }
    cout << k << endl;
    for (block b : blocks) {
        if (b.sum == best_sum) {
            cout << b.l + 1 << " " << b.r + 1 << endl;
        }
    }
    return 0;
}
