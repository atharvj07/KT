import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] x = new int[n], y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = in.nextInt();
            y[i] = in.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int dx1 = x[(i + 1) % n] - x[i];
            int dy1 = y[(i + 1) % n] - y[i];
            int dx2 = x[(i + 2) % n] - x[(i + 1) % n];
            int dy2 = y[(i + 2) % n] - y[(i + 1) % n];
            int cross = dx1 * dy2 - dy1 * dx2;
            if (cross > 0) ans++;
            else if (cross < 0) ans--;
        }
        if (ans == 0) System.out.println(0);
        else System.out.println(1);
    }
}
