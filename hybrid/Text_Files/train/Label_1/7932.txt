#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  vector<int> ans(n - 1, 0);

  for (int k = 1; k < n; k++) {
    for (int i = 1; i + k <= n; i++) {
      if (a[i] < a[i / k]) { ans[k - 1]++; }
    }
  }

  for (int i = 0; i < n - 1; i++) { cout << ans[i] << " "; }
  cout << endl;

  return 0;
}
