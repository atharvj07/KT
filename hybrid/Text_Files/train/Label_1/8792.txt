#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  vector<int> arthur(a), alexander(b);
  set<int> arthur_set, alexander_set;
  for (int i = 0; i < a; i++) {
    cin >> arthur[i];
    arthur_set.insert(arthur[i]);
  }
  for (int i = 0; i < b; i++) {
    cin >> alexander[i];
    alexander_set.insert(alexander[i]);
  }
  for (int i = 1; i <= n; i++) {
    if (arthur_set.count(i))
      cout << 1;
    else
      cout << 2;
  }
  cout << endl;
  return 0;
}
