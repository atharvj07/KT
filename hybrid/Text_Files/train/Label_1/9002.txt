#include <bits/stdc++.h>

using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<string> strs(n);
    for (string &s : strs)
        cin >> s;
    map<char, int> mp;
    for (string s : strs)
        for (char c : s)
            mp[c]++;
    int digit = 0;
    for (pair<char, int> p : mp)
    {
        if (p.second <= n)
            mp[p.first] = digit++;
    }
    long long ans = 0;
    for (string s : strs)
    {
        long long pw = 1;
        for (int i = s.size() - 1; i >= 0; i--)
        {
            ans += mp[s[i]] * pw;
            pw *= 10;
        }
    }
    cout << ans << endl;
    return 0;
}
