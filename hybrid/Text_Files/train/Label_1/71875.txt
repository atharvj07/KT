import sys
n,m,k = map(int,input().split())
l = sorted([int(i) for i in input().split()]+[0,n])
a = sorted([int(i) for i in input().split()])
ans = -1
for i in range(k):
    need = 0
    last = l[0]+a[i]-1
    for j in range(1,len(l)):
        if l[j]>last+1: need+=1; last = l[j]+a[i]-1
    if need==0: ans=a[i]*need; break
print(ans)
