from sys import stdin
from typing import List

def gcd(a: int, b: int) -> int:
    return a if b == 0 else gcd(b, a % b)

def solve(sticks: List[int]) -> List[int]:
    sticks.sort()

    ans = []
    for i in range(0, len(sticks), 2):
        j = i + 1
        while j < len(sticks) and sticks[j] == sticks[i]:
            j += 1
        
        if j < len(sticks):
            a = sticks[i]
            b = sticks[j]

            if a == b:
                ans.append(a)
                ans.append(a)
                ans.append(b)
                ans.append(b)
            else:
                c = gcd(a, b)
                if c == 1:
                    ans.append(a)
                    ans.append(a)
                    ans.append(b)
                    ans.append(b)
                else:
                    ans.append(a)
                    ans.append(a)
                    ans.append(c)
                    ans.append(b - c)

            return ans

    return []

for _ in range(int(stdin.readline())):
    _ = int(stdin.readline())
    sticks = [int(x) for x in stdin.readline().split()]
    ans = solve(sticks)
    if ans:
        print(*ans)
    else:
        print(-1)
