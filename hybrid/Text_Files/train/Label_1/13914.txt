#include <bits/stdc++.h>
using namespace std;
const int N=1e5+100;
vector<int> adj[N];
int a[N];
void dfs(int x,int p,int d){
    for(int i=0;i<(int)adj[x].size();i++){
        int y=adj[x][i];
        if(y==p)continue;
        a[d]=y;
        dfs(y,x,d+1);
    }
}
int main() {
    int n;
    cin >> n;
    for(int i=1;i<n;i++){
        int x,y;cin>>x>>y;
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    dfs(1,1,0);
    set<int>st;
    for(int i=0;i<n-1;i++){
        while(st.count(a[i]))a[i]++;
        st.insert(a[i]);
        cout<<a[i]<<' ';
    }
    return 0;
}
