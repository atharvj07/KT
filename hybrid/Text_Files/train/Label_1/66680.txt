Q = int(input().strip())
S = []
for _ in range(Q):
    query = list(map(int, input().strip().split()))
    if query[0] == 1:
        S.append(query[1])
    else:
        if len(S) == 1:
            print("0.0000000000")
        else:
            max_val = S[-1]
            mean_val = sum(S) / len(S)
            print(f"{max_val - mean_val:.10f}")
