import bisect

n, k, q = map(int, input().split())
temp = []
for _ in range(n):
    l, r = map(int, input().split())
    for a in range(l, r + 1):
        temp.append(a)
temp.sort()

for _ in range(q):
    a, b = map(int, input().split())
    print(bisect.bisect_right(temp, b) - bisect.bisect_left(temp, a))
