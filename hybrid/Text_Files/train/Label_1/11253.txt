#include <iostream>
using namespace std;
int a[20],n,x;
int main() {
    cin>>n;
    for (int i=0;i<n;i++) cin>>a[i];
    for (int i=0;i<n;i++) {
        for (int j=0;j<n;j++) {
            if (a[i]<a[j]) {
                if (a[j]-a[i]>x) x=a[j]-a[i];
            } else {
                if (a[i]-a[j]>x) x=a[i]-a[j];
            }
        }
    }
    x=x/2;
    for (int i=0;i<n;i++) {
        for (int j=0;j<n;j++) {
            if (a[i]-a[j]<=x) {
                for (int k=a[i]-x;k<=a[i]+x;k++) {
                    if (k==a[j]) cout<<k;
                }
            }
        }
    }
    cout<<endl;
    return 0;
}
