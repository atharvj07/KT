#include<bits/stdc++.h>
#define ll long long
const int mod = 1e9 + 7;
using namespace std;
int pw[1000010];
int main() {
    ll a, b, n;
    cin >> a >> b >> n;
    a--, b--;
    pw[0] = 1;
    for(int i = 1;i <= 1000000;++i){
        pw[i] = 2 * pw[i - 1];
        pw[i] %= mod;
    }
    if(b == a + 1){
        cout << pw[n] << '\n';
        return 0;
    }
    ll ans = pw[n];
    ans -= pw[n - 1];
    ll deg = b - a - 1;
    deg %= mod;
    deg += mod;
    deg %= mod;
    ans *= deg;
    ans %= mod;
    cout << ans << '\n';
    return 0;
}
