import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static Scanner sc = new Scanner(System.in);

    static int[] p, q;
    static int[] id;
    static int[] size;
    static int[] ans;

    public static void main(String[] args) {
        int n = sc.nextInt();
        p = new int[n];
        q = new int[n];
        for (int i = 0; i < n; ++i) {
            p[i] = sc.nextInt() - 1;
            q[i] = sc.nextInt() - 1;
        }
        int m = sc.nextInt();
        ans = new int[m];
        dsUnionInit(n);
        for (int i = 0; i < m; ++i) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            int c = sc.nextInt() - 1;
            int d = sc.nextInt() - 1;
            ans[i] = solve(a, b, c, d);
        }
        for (int i = 0; i < m; ++i) {
            System.out.println(ans[i]);
        }
    }

    static void dsUnionInit(int n) {
        id = new int[n];
        size = new int[n];
        for (int i = 0; i < n; ++i) {
            id[i] = i;
            size[i] = 1;
        }
    }

    static int root(int i) {
        while (i != id[i]) {
            id[i] = id[id[i]];
            i = id[i];
        }
        return i;
    }

    static void dsUnion(int a, int b) {
        int rootA = root(a);
        int rootB = root(b);
        if (rootA == rootB) {
            return;
        }
        if (size[rootA] > size[rootB]) {
            id[rootB] = rootA;
            size[rootA] += size[rootB];
        } else {
            id[rootA] = rootB;
            size[rootB] += size[rootA];
        }
    }

    static int solve(int a, int b, int c, int d) {
        int cnt = 0;
        for (int i = 0; i < p.length; ++i) {
            if (p[i] >= a && p[i] <= b && q[i] >= c && q[i] <= d) {
                ++cnt;
            }
        }
        return cnt;
    }
}
