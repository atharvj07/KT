#include <iostream>
using namespace std;
long long det(long long a1, long long a2, long long a3, long long a4) {
  return a1 * a4 - a2 * a3;
}
int main() {
  long long a1, a2, b1, b2, c1, c2;
  cin >> a1 >> a2 >> b1 >> b2 >> c1 >> c2;
  if (det(a1, a2, b1, b2) * det(a1, a2, c1, c2) < 0)
    cout << "Yes";
  else
    cout << "No";
  return 0;
}
