import sys
input = sys.stdin.readline
n, m = map(int,input().split())
g = [[] for i in range(n+1)]
for i in range(m):
    u, v, c = map(int,input().split())
    g[u].append((v,c))
    g[v].append((u,c))
used = [False]*(n+1)
par = [0]*(n+1)
size = [1]*(n+1)
ans = [0]*m
res = []
def dfs(v,p):
    global ans,res
    used[v] = True
    par[v] = p
    for x,c in g[v]:
        if not used[x]:
            size[v]+=dfs(x,v)
            ans[c]+=1
        elif x!=p:
            cycle = []
            cur = v
            while cur!=x:
                cycle.append(cur)
                cur = par[cur]
            cycle.append(x)
            cycle.reverse()
            mi = 1e9
            for j in cycle:
                mi = min(mi,ans[j])
            for j in cycle:
                ans[j]-=mi
            res+=cycle[:len(cycle)-1]
    return size[v]
dfs(1,0)
print(*ans[:max(ans)+1],len(res),*res)
