#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (auto &ai : a) {
    std::cin >> ai;
  }
  int i = 0;
  while (i < n && a[i] < a[(i + 1) % n]) {
    i++;
  }
  if (i < n && a[i] == a[(i + 1) % n]) {
    std::cout << -1 << std::endl;
  } else if (i < n) {
    std::cout << "DOWN" << std::endl;
  } else {
    std::cout << "UP" << std::endl;
  }
  return 0;
}
