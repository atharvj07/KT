#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<bool> v(m, 0);
  while (n--) {
    int l, r;
    cin >> l >> r;
    v[l] = 1, v[r + 1] = 1;
  }
  int ans = 0;
  for (int i = 1; i <= m; i++) ans += v[i] == 0;
  cout << ans << "\n";
  for (int i = 1; i <= m; i++) {
    if (v[i] == 0) cout << i << ' ';
  }
  return 0;
}
