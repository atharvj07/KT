#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  cin >> T;
  while (T--) {
    int n;
    cin >> n;
    map<int, int> a;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      a[x]++;
    }
    int odd = 0;
    for (auto &p : a) {
      if (p.second % 2 == 1) odd++;
    }
    if (odd <= 1) cout << 0 << endl; else cout << 1 << endl;
  }
  return 0;
}
