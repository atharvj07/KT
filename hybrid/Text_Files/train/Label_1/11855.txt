#include <iostream>
#include <vector>
int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) { std::cin >> a[i]; }
  int mn = 0, mx = 0;
  for (int i = 1; i < n; ++i) {
    if (a[i] > a[i - 1]) {
      ++mx;
    } else {
      break;
    }
  }
  for (int i = n - 2; i >= 0; --i) {
    if (a[i] > a[i + 1]) {
      ++mn;
    } else {
      break;
    }
  }
  std::cout << (mn + mx + 1 == n ? "YES" : "NO") << "\n";
}
