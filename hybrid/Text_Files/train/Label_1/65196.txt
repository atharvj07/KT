n = int(input())
segments = [list(map(int, input().split())) for _ in range(n)]

alexey_segment = segments[0]
occupied = [0] * 101

for i in range(1, n):
    for j in range(segments[i][0], segments[i][1]):
        occupied[j] = 1

length = 0
for i in range(alexey_segment[0], alexey_segment[1]):
    if not occupied[i]:
        length += 1

print(length)
