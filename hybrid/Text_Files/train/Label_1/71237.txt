a = list(map(int, input().split()))
n = int(input())
b = list(map(int, input().split()))

b.sort()

min_diff = float('inf')
for i in range(6):
    frets = [note - a[i] for note in b]
    min_fret = min(frets)
    max_fret = max(frets)
    min_diff = min(min_diff, max_fret - min_fret)

print(min_diff)
