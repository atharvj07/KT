import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();

        Queue<Integer>[] lanes = new Queue[N];
        for (int i = 0; i < N; i++) {
            lanes[i] = new LinkedList<>();
        }

        int[] carNum = new int[10000];
        Arrays.fill(carNum, -1);

        for (int i = 0; i < M; i++) {
            int type = sc.nextInt();
            if (type == 0) {
                int lane = sc.nextInt() - 1;
                System.out.println(lanes[lane].poll());
            } else {
                int car = sc.nextInt();
                int minLane = -1;
                int minSize = Integer.MAX_VALUE;
                for (int j = 0; j < N; j++) {
                    if (lanes[j].size() < minSize) {
                        minSize = lanes[j].size();
                        minLane = j;
                    }
                }
                lanes[minLane].add(car);
                carNum[car] = minLane;
            }
        }
    }
}
