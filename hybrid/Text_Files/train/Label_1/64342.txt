import sys
input = sys.stdin.readline

n, m = map(int, input().split())
g = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b, c = map(int, input().split())
    g[a].append((b, c))
    g[b].append((a, c))

mod = 10 ** 9 + 7
dp = [[[-1] * 3 for _ in range(3)] for _ in range(n + 1)]

def rec(x, p, l):
    if dp[x][p][l] != -1:
        return dp[x][p][l]
    if not g[x]:
        if l == 1:
            dp[x][p][l] = 1
        else:
            dp[x][p][l] = 0
        return dp[x][p][l]
    ret = 0
    for y, c in g[x]:
        if y != p:
            if l == 0:
                ret += rec(y, x, c)
            elif l == 1:
                ret += rec(y, x, 2 - c)
            else:
                ret += rec(y, x, c)
            ret %= mod
    dp[x][p][l] = ret
    return ret

print(rec(1, 0, 0))
