n, k = map(int, input().split())

bits = []
for i in range(63, -1, -1):
    if (n >> i) & 1:
        bits.append(i)

if len(bits) > k or k > n:
    print("No")
else:
    print("Yes")
    while len(bits) < k:
        x = bits.pop()
        bits.append(x-1)
        bits.append(x-1)
    bits.sort(reverse=True)
    print(*bits)
