import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int H = sc.nextInt();
        int W = sc.nextInt();
        int[][] a = new int[H + 1][W + 1];
        int[][] b = new int[H + 1][W + 1];
        for (int i = 1; i <= H; i++) {
            for (int j = 1; j <= W; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        for (int i = 1; i <= H; i++) {
            for (int j = 1; j <= W; j++) {
                b[i][j] = sc.nextInt();
            }
        }
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[][] c = new int[h + 1][w + 1];
        for (int i = 1; i <= h; i++) {
            for (int j = 1; j <= w; j++) {
                c[i][j] = sc.nextInt();
            }
        }
        int ans = 0;
        for (int i = 1; i <= H - h + 1; i++) {
            for (int j = 1; j <= W - w + 1; j++) {
                boolean flag = true;
                for (int k = 1; k <= h; k++) {
                    for (int l = 1; l <= w; l++) {
                        if (b[i + k - 1][j + l - 1] != c[k][l]) {
                            flag = false;
                        }
                    }
                }
                if (flag) {
                    int sum = 0;
                    for (int k = 1; k <= h; k++) {
                        for (int l = 1; l <= w; l++) {
                            sum += a[i + k - 1][j + l - 1];
                        }
                    }
                    ans = Math.max(ans, sum);
                }
            }
        }
        if (ans == 0) {
            System.out.println("NA");
        } else {
            System.out.println(ans);
        }
    }
}
