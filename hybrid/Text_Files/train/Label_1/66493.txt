def maximize_profit(n, requests, k, tables):
    requests = sorted([(c, p, i) for i, (c, p) in enumerate(requests, start=1)], key=lambda x: -x[1])
    tables = sorted([(r, i) for i, r in enumerate(tables, start=1)])
    used_tables = [False] * k
    accepted_requests = []

    for c, p, req_id in requests:
        for j in range(k):
            if tables[j][0] >= c and not used_tables[j]:
                used_tables[j] = True
                accepted_requests.append((req_id, tables[j][1]))
                break

    accepted_requests.sort()
    total_money = sum(requests[req_id-1][1] for req_id, _ in accepted_requests)
    print(len(accepted_requests), total_money)
    for req_id, table_id in accepted_requests:
        print(req_id, table_id)

n = int(input().strip())
requests = [tuple(map(int, input().split())) for _ in range(n)]
k = int(input().strip())
tables = list(map(int, input().split()))
maximize_profit(n, requests, k, tables)
