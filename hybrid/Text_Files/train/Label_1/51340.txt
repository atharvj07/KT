import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;
import java.util.concurrent.ArrayBlockingQueue;

public class Main {
    static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static int dx[] = {1, -1, 0, 0};
    public static int dy[] = {0, 0, 1, -1};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int h = sc.nextInt();
        int w = sc.nextInt();
        int k = sc.nextInt();

        char[][] a = new char[h + 2][w + 2];
        for (int i = 1; i <= h; i++) {
            a[i] = (" " + sc.next() + " ").toCharArray();
        }

        Point s = null;
        for (int i = 1; i <= h; i++) {
            for (int j = 1; j <= w; j++) {
                if (a[i][j] == 'S') {
                    s = new Point(i, j);
                }
            }
        }

        int[][][] dist = new int[h + 2][w + 2][k + 1];
        for (int[][] d : dist) {
            for (int[] dd : d) {
                Arrays.fill(dd, -1);
            }
        }
        dist[s.x][s.y][0] = 0;

        Queue<Point> que = new LinkedList<>();
        que.add(s);

        while (!que.isEmpty()) {
            Point now = que.poll();

            for (int i = 0; i < 4; i++) {
                int nx = now.x + dx[i];
                int ny = now.y + dy[i];

                if (nx < 1 || nx > h || ny < 1 || ny > w) continue;
                if (a[nx][ny] == '#') continue;

                for (int rest = 0; rest <= k; rest++) {
                    if (dist[nx][ny][rest] != -1) continue;

                    dist[nx][ny][rest] = dist[now.x][now.y][rest] + 1;
                    que.add(new Point(nx, ny));
                }

                if (a[nx][ny] == '.' && dist[now.x][now.y][k] == -1) {
                    dist[now.x][now.y][k] = dist[nx][ny][k] + 1;
                    que.add(now);
                }
            }
        }

        int ans = Integer.MAX_VALUE;
        for (int i = 0; i <= k; i++) {
            if (dist[h][w][i] != -1) {
                ans = Math.min(ans, dist[h][w][i]);
            }
        }

        if (ans == Integer.MAX_VALUE) {
            ans = -1;
        }
        System.out.println(ans);
    }
}
