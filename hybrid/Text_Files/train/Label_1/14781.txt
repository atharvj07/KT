#include <bits/stdc++.h>
using namespace std;
const int maxn = 1005;
char s[maxn], t[maxn];
int cnt[30];
vector<pair<char, char>> ans;
bool check() {
  for (int i = 0; i < 26; i++) {
    if (cnt[i] & 1) return false;
  }
  return true;
}
int main() {
  cin >> s >> t;
  int n = strlen(s);
  for (int i = 0; i < n; i++) {
    cnt[s[i] - 'a']++;
    if (s[i] == t[i]) continue;
    if (s[i] == t[i] - 32 || s[i] == t[i] + 32) continue;
    ans.push_back({s[i], t[i]});
  }
  if (check()) {
    cout << ans.size() << endl;
    for (auto x : ans) {
      cout << x.first << ' ' << x.second << endl;
    }
  } else
    cout << -1 << endl;
  return 0;
}
