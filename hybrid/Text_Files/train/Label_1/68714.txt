import sys

n = int(input())
a = []
for i in range(n):
    x, y = map(int, input().split())
    a.append((y-x, (x, y)))
a.sort()
w = list(map(int, input().split()))
for i in range(n):
    if a[i][0] != w[i]:
        print('NO')
        sys.exit()
print('YES')
a.sort(key=lambda x: x[1])
for i in range(n):
    print(*a[i][1])
