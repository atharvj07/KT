#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long n, k, sum = 0, s;
  cin >> n >> k;
  vector<long long> a(n);
  for (auto &ai : a) {
    cin >> ai;
    sum += ai;
  }
  if (sum % k != 0) {
    cout << "No" << endl;
    return 0;
  }
  s = sum / k;
  cout << "Yes" << endl;
  for (long long i = 0, j = 0; i < n; i += j) {
    for (j = 0; j < n - i && a[i + j] <= s; j++)
      ;
    cout << j << " ";
  }
  cout << endl;
  return 0;
}
