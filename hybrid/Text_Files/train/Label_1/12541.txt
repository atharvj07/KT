#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 1e2 + 5;
ll a[N], b[N], on[N], st[N * 25];
int main() {
  ll n;
  cin >> n;
  string s;
  cin >> s;
  for (ll i = 0; i < n; i++) {
    cin >> a[i] >> b[i];
    for (ll j = b[i]; j < b[i] + a[i] * 25; j++) st[j % (a[i] * 25)]++;
    if (s[i] == '1') on[i] = 1;
  }
  ll ans = 0;
  for (ll i = 0; i < N * 25; i++) {
    ll tmp = 0;
    for (ll j = 0; j < n; j++) {
      tmp += on[j] * st[i % (a[j] * 25)];
    }
    ans = max(ans, tmp);
  }
  cout << ans << endl;
  return 0;
}
