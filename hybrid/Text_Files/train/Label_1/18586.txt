#include<bits/stdc++.h>
#define ll long long
using namespace std;
struct node{
	int x;
	ll v;
	bool operator<(const node& other)const{
		return v<other.v;
	}
};
int n,m,p[55],ans[55];
vector<ll>a;
priority_queue<node>pq;
ll c[55][55];
int main(){
	cin>>n>>m;
	for(ll i=1;i<=n;i++)a.push_back(i);
	for(int i=1;i<=n;i++){
		for(int j=i;j<=n;j++){
			c[i][j]=0;
			for(int k=i;k<=j;k++)c[i][j]+=a[k-1];
		}
	}
	for(int i=1;i<=n;i++)pq.push({i,c[1][i]});
	for(int i=1;i<=n;i++){
		node tmp=pq.top();pq.pop();
		ans[i]=tmp.x;
		for(int j=tmp.x+1;j<=n;j++)pq.push({j,c[tmp.x+1][j]-c[tmp.x][j]});
	}
	for(int i=1;i<=n;i++){
		cout<<ans[i];
		if(i<n)cout<<" ";
	}
	return 0;
}
