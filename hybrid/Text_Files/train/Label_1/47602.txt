import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] s = sc.nextLine().toCharArray();
        int n = s.length;
        int l = 0, r = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == '(') {
                l++;
            } else {
                r++;
            }
        }
        StringBuilder ans = new StringBuilder();
        while (l > 0 || r > 0) {
            if (l > 0) {
                ans.append('(');
                l--;
            } else {
                ans.append(')');
                r--;
            }
        }
        System.out.println(ans);
    }
}
