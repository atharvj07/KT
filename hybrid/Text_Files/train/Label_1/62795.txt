from collections import deque, Counter
for _ in range(int(input())):
    s = deque(input())
    bal = Counter([x for x in s])['('] - Counter([x for x in s])[')']
    r = 0
    t = 0
    while (s):
        x = s.popleft()
        if bal < 0:
            x, bal = ')', bal + 1
        elif bal > 0:
            x, bal = '(', bal - 1
        if x == '(':
            bal += 1
        if x == ')' and bal < 0:
            t += 1
            bal = 0
    r += max(0, bal) + t
    r += t
    r += max(0, -bal)
    print(r)
