#include<bits/stdc++.h>
using namespace std;
int main(){
	int T;
	cin >> T;
	while(T--){
		string n;
		int s;
		cin >> n >> s;
		int ans = 0;
		int sum = 0;
		for(char c : n){
			sum += c - '0';
		}
		if(sum > s){
			int diff = sum - s;
			int sz = n.size();
			for(int i = 0; i < sz; i++){
				int l = n[i] - '0';
				if(l > 9){
					if(l - 9 <= diff){
						ans++;
						diff -= l - 9;
						n[i] = '9';
					}
					else{
						ans += diff;
						l -= diff;
						n[i] = l + '0';
						break;
					}
				}
			}
			sum = 0;
			for(char c : n){
				sum += c - '0';
			}
			if(sum > s){
				ans += sum - s;
			}
		}
		cout << ans <<'\n';
	}
	return 0;
}
