import sys
n = int(sys.stdin.readline())
a, b, c, d = [0] * 4
for i in range(4 * n + 1):
    x, y = map(int, sys.stdin.readline().split())
    a += (x == 0)
    b += (x == n - 1)
    c += (y == 0)
    d += (y == n - 1)
x = (a == n)
y = (b == n)
if x and not y:
    x, y = 2 * n - 1, 0
elif y and not x:
    x, y = 0, 2 * n - 1
elif not x and not y:
    x, y = n - 1, n - 1
print(x, y)
