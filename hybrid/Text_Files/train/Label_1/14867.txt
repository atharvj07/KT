#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll n,i,a,b,c,msk[200005],ans;
int main()
{
    cin>>n;
    for(i=1;i<=n;i++)
    {
        cin>>a>>b;
        msk[i]=msk[i-1];c=1<<i;
        if(a>b)   msk[i]|=c;
    }
    for(i=n;i>=1;i--)
    {
        c=1<<i;
        ans+=msk[i]==msk[i-1];
    }
    cout<<ans<<endl;
    return 0;
}
