#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  map<int, int> a, b;
  for (int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    ++a[x];
  }
  int ans = -1;
  for (auto e : a) b[e.second]++;
  for (int i = 2; i * i <= n; ++i)
    if (n % i == 0) {
      int x = 0;
      for (auto e : b) x += e.second % (i * i) > 0;
      if (x == 1) ans = max(ans, i);
      for (auto e : b) x += e.second / (i * i) > 0;
      if (x == 1) ans = max(ans, n / i);
    }
  if (ans == -1) ans = 1;
  cout << ans << endl;
  return 0;
}
