import java.util.Scanner;

public class InnopolisHills {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] heights = new int[n];
        for (int i = 0; i < n; i++) {
            heights[i] = scanner.nextInt();
        }

        int[] minHours = new int[(n + 1) / 2];
        int[] prevHeights = new int[n];

        for (int k = 1; k <= n / 2; k++) {
            int maxHeight = Integer.MIN_VALUE;
            for (int i = 0; i < n; i++) {
                if (heights[i] > maxHeight) {
                    maxHeight = heights[i];
                }
                prevHeights[i] = maxHeight;
            }

            int hours = 0;
            for (int i = n - 1; i >= 0; i--) {
                if (prevHeights[i] >= heights[i] - k + 1) {
                    hours += prevHeights[i] - (heights[i] - k + 1);
                    prevHeights[i] = heights[i] - k + 1;
                }
            }
            minHours[k - 1] = hours;
        }

        for (int hours : minHours) {
            System.out.print(hours + " ");
        }
    }
}
