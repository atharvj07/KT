#include<bits/stdc++.h>
#include<limits.h>
using namespace std;
int main()
{
    long long int n,c,d,tot,dif,b,p,max=0,max2=0,cmax=INT_MIN,dmax=INT_MIN;
    cin>>n>>c>>d;
    for(int i=0;i<n;i++)
    {
        cin>>b>>p;
        char type;
        cin>>type;
        if(type=='C')
        {
            if(p<=c)
            {
                if(b>cmax)
                {
                    cmax=b;
                }
            }
        }
        else
        {
            if(p<=d)
            {
                if(b>dmax)
                {
                    dmax=b;
                }
            }
        }
    }
    if(cmax==INT_MIN || dmax==INT_MIN)
        cout<<"0"<<endl;
    else
    {
        max=cmax+dmax;
        cout<<max<<endl;
    }
    return 0;
}
