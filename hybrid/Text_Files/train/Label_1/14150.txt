#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
int a[maxn], b[maxn], dp[maxn];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
    }
    for (int i = 1; i <= n; i++) {
        dp[i] = dp[i - 1];
        for (int j = 1; j <= n; j++) {
            if (dp[j] + b[i] <= m) {
                dp[i] = dp[j] + a[i] - b[i];
            }
        }
    }
    int ans = *max_element(dp + 1, dp + n + 1);
    if (ans == 0) {
        cout << -1 << endl;
    } else {
        cout << n - ans << endl;
    }
    return 0;
}
