import sys

def solve():
    t = int(sys.stdin.readline().strip())
    for _ in range(t):
        a, b, n = map(int, sys.stdin.readline().strip().split())
        a_arr = list(map(int, sys.stdin.readline().strip().split()))
        b_arr = list(map(int, sys.stdin.readline().strip().split()))
        can_kill = True
        for b_i in b_arr:
            if b_i > a:
                can_kill = False
        if can_kill:
            print('YES')
        else:
            print('NO')

solve()
