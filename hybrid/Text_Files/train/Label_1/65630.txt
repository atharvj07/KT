from collections import defaultdict
import sys

for t in range(int(input())):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    ps = [0] * (n+1)
    for i in range(n):
        ps[i + 1] = (a[i] + ps[i]) % x
    dic = defaultdict(list)
    dic[0].append(-1)
    mx = 0
    for i in range(n + 1):
        if ps[i] in dic:
            mx = max(mx, i - dic[ps[i]][0])
        else:
            dic[ps[i]].append(i)
    sys.stdout.write(str(mx) if mx else '-1' + '\n')
