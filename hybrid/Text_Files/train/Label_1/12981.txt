#include <iostream>
using namespace std;

long long m, a, b;

int main() {
    cin >> m >> a >> b;
    if (a < b) {
        long long t = a;
        a = b;
        b = t;
    }
    if (a + b <= m) {
        cout << (m + 1) * (m + 2) / 2 << endl;
        return 0;
    }
    long long l, r, ans = 0;
    l = (m + b) / a;
    r = m / a;
    ans = ans + (l + r) * (r - l + 1) / 2;
    if (l - 1 >= 0) {
        l = (m / b + 1);
        r = m / b;
        ans = ans + (l + r) * (r - l + 1) / 2;
    }
    cout << ans << endl;
    return 0;
}
