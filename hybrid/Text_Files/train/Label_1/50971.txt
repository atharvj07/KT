import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        scanner.close();
        long m = 2;
        while (BigInteger.valueOf(n).divide(BigInteger.valueOf(m)).multiply(BigInteger.valueOf(n - m + 1)).divide(BigInteger.valueOf(m - 1)).remainder(BigInteger.valueOf(2)).equals(BigInteger.ONE)) {
            m++;
        }
        System.out.println(m);
    }
}
