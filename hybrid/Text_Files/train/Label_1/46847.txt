import java.util.*;
class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    int s = scan.nextInt();
    int b = scan.nextInt();
    int[] a = new int[s];
    for (int i = 0; i < s; i++) {a[i] = scan.nextInt();}
    int[][] bd = new int[b][2];
    for (int i = 0; i < b; i++) {
      bd[i][0] = scan.nextInt();
      bd[i][1] = scan.nextInt();
    }
    Arrays.sort(a);
    Arrays.sort(bd, (x, y) -> x[0] - y[0]); // Sort based on defensive power
    int[] ans = new int[s];
    for (int i = 0; i < s; i++) {
      int cur = 0;
      int j = 0;
      while (j < b && bd[j][0] <= a[i]) {
        cur += bd[j][1];
        j++;
      }
      ans[i] = cur;
    }
    StringBuilder sb = new StringBuilder();
    for (int num : ans) {sb.append(num).append(' ');}
    System.out.println(sb);
  }
}
