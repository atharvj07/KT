#include<bits/stdc++.h>
using namespace std;
int main(){
    int w,h,u1,d1,u2,d2; cin>>w>>h>>u1>>d1>>u2>>d2;
    if(d1<d2){
        while(h){
                w+=h--;
                if(h==d1)w-=u1;
            if(h==d2)w-=u2;
            }
        }
    else{
        while(h){
            w+=h--;
            if(h==d2)w-=u2;
            if(h==d1)w-=u1;
        }
    }
    cout<<w;
    return 0;
}
