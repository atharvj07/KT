#include<bits/stdc++.h>
using namespace std;
int main(){
    int t,a,b,c,m,n;
    cin>>t;
    while(t--){
        cin>>a>>b>>c>>m;
        n=a+b+c;
        if(n%2==1){
            cout<<"NO"<<endl;
            continue;
        }
        if(m>a+b+c-1 || a+c<m || b+c<m ){
            cout<<"NO"<<endl;
            continue;
        }
        cout<<"YES"<<endl;
    }
}
