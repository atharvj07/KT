import random
import sys
from collections import defaultdict

t = int(input())
for _ in range(t):
    n = int(input())
    b = list(map(int, input().split()))
    if b[n + 1] != sum(b[:n]) or b[n] == b[n + 1]:
        print(-1)
        continue
    random.shuffle(b[:n])
    a = [b[n + 1] - b[n]]
    for i in range(1, n):
        a.append(b[i] - a[i - 1])
    print(*a)
