t = int(input())
for _ in range(t):
    n = int(input())
    heights = list(map(int, input().split()))
    can_increase = True
    for i in range(1, n):
        if heights[i-1] >= heights[i]:
            if heights[i-1] - heights[i] + 1 > heights[i]:
                can_increase = False
                break
            else:
                heights[i] = heights[i-1] + 1
    print("YES" if can_increase else "NO")
