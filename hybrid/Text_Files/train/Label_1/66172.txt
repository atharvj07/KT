n, k = map(int, input().split())
a = list(map(int, input().split()))
p = 0
ans = 0
for x in a:
    if not p:
        if x==k:
            p=1
    else:
        if x==k*p:
            p+=1
            if p==3:
                ans+=1
        else:
            p=1
            if x==k:
                p=1
print(ans)
