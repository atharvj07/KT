for nt in range(int(input())):
    n = int(input())
    e = [[] for i in range(n+1)]
    for i in range(n-1):
        u, v = map(int, input().split())
        e[u].append(v)
        e[v].append(u)
    vis = [False for i in range(n+1)]
    vis[1] = True
    q = [1]
    ans = []
    while q:
        u = q.pop(0)
        ans.append(u)
        for v in e[u]:
            if vis[v] == False:
                vis[v] = True
                q.append(v)
    print(*ans)
