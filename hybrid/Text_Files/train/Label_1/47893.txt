import java.util.*;
import java.math.*;
class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    if(k == 0) {
      for(int i = 1; i <= n; i++) {
        System.out.print(i + " ");
      }
    } else if(n%2 == 0 && k%2 == 0 && k <= n) {
      if(k == 2) {
        System.out.println(-1);
      } else {
        System.out.print(1 + " ");
        for(int i = 3; i <= k + 2; i += 2) {
          System.out.print(i + " " + (i-1) + " ");
        }
        for(int i = k + 3; i <= n; i++) {
          System.out.print(i + " ");
        }
      }
    } else if(n%2 == 1 && k%2 == 1 && k <= n) {
      if(k == 1) {
        System.out.println(-1);
      } else {
        System.out.print(1 + " ");
        for(int i = 3; i <= k + 2; i += 2) {
          System.out.print(i + " " + (i-1) + " ");
        }
        for(int i = k + 3; i <= n; i++) {
          System.out.print(i + " ");
        }
      }
    } else {
      System.out.println(-1);
    }
  }
}
