n = int(input().strip())
birds_on_wire = list(map(int, input().strip().split()))
m = int(input().strip())

for _ in range(m):
    x, y = map(int, input().strip().split())
    wire = x - 1
    if wire > 0:
        birds_on_wire[wire - 1] += y - 1
    if wire < n - 1:
        birds_on_wire[wire + 1] += birds_on_wire[wire] - y
    birds_on_wire[wire] = 0

for birds in birds_on_wire:
    print(birds)
