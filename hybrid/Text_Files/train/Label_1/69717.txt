import sys
input = sys.stdin.readline
for _ in range(int(input())):
    n, t, a, b = map(int, input().split())
    l = list(map(int, input().split()))
    tm = list(map(int, input().split()))
    ans = 0
    i = 0
    for j in range(t):
        while i < n and tm[i] <= j:
            if l[i] == 0:
                ans += 1
            i += 1
        if i < n and tm[i] > j:
            if l[i] == 0:
                if tm[i] - j <= a:
                    ans += 1
            else:
                if tm[i] - j <= b:
                    ans += 1
            i += 1
    print(ans)
