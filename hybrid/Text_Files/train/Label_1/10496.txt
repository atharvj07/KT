#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n * 2);
  for (auto &x : a) cin >> x;
  vector<vector<int>> pos(n + 1);
  for (int i = 0; i < n * 2; ++i) { pos[a[i]].emplace_back(i); }
  int ans = 0;
  for (int i = 1; i <= n; ++i) {
    int sz = pos[i].size();
    ans += (sz - 1) / 2;
  }
  cout << ans << "\n";
}
