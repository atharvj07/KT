n, c = map(int, input().split())
p = list(map(int, input().split()))
t = list(map(int, input().split()))
limaks = 0
radewoosh = 0
for i in range(0, n):
    limaks += max(0, p[i] - c * (i + 1 + t[i]))
    radewoosh += max(0, p[n - i - 1] - c * (i + 1 + t[n - i - 1]))
if limaks > radewoosh:
    print("Limak")
elif limaks == radewoosh:
    print("Tie")
else:
    print("Radewoosh")
