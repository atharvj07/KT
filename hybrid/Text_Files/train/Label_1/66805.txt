from sys import stdin
n=int(stdin.readline())
a=[int(x) for x in stdin.readline().split()]
p=[-1]*n
for _ in range(n-1):
    x,y=map(int,stdin.readline().split())
    p[y-1]=x-1
r=0
ans=[0]*n
for i in range(n-1,-1,-1):
    if p[i]!=-1:a[i]=a[i]%a[p[i]]
    ans[i]=max(a[i],ans[p[i]] if p[i]!=-1 else 0)
print(*ans)
