import java.util.*;

public class Main {
    static int N, K, C, T;
    static long[][] dp;
    static List<Integer>[] shuffle;
    static long mod = (int)1e9 + 7;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt(); K = sc.nextInt(); C = sc.nextInt(); T = sc.nextInt();
        dp = new long[N+1][T+1];
        shuffle = new List[N+1];
        for (int i=1; i<=N; i++) shuffle[i] = new ArrayList<>();

        for (int i=0; i<K; i++) {
            int a = sc.nextInt(), b = sc.nextInt(), t = sc.nextInt();
            for (int j=a; j<a+b; j++) shuffle[j].add(t);
        }

        dp[C][0] = 1;
        for (int i=1; i<=T; i++) {
            for (int j=1; j<=N; j++) {
                for (int t : shuffle[j]) {
                    if (i - t >= 0) dp[j][i] = (dp[j][i] + dp[C][i-t]) % mod;
                }
            }
        }

        System.out.println(dp[C][T]);
    }
}
