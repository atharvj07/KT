#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

vector<int> build_suffix_array(const string& s) {
    int n = s.size();
    vector<int> p(n), c(n), lcp(n);
    {
        vector<pair<char, int>> a(n);
        for (int i = 0; i < n; ++i) a[i] = {s[i], i};
        sort(a.begin(), a.end());
        for (int i = 0; i < n; ++i) p[i] = a[i].second;
        c[p[0]] = 0;
        for (int i = 1; i < n; ++i) c[p[i]] = c[p[i - 1]] + (a[i].first != a[i - 1].first);
    }
    int k = 0;
    while ((1 << k) < n) {
        for (int i = 0; i < n; ++i) p[i] = (p[i] - (1 << k) + n) % n;
        vector<pair<pair<int, int>, int>> a(n);
        for (int i = 0; i < n; ++i) a[i] = {{c[p[i]], c[(p[i] + (1 << k)) % n]}, i};
        sort(a.begin(), a.end());
        for (int i = 0; i < n; ++i) p[i] = a[i].second;
        c[p[0]] = 0;
        for (int i = 1; i < n; ++i) c[p[i]] = c[p[i - 1]] + (a[i].first != a[i - 1].first);
        ++k;
    }
    for (int i = 0; i < n - 1; ++i) lcp[i] = max(0, k - 1 - (int)(__builtin_clz(c[p[i]] ^ c[p[i + 1]])));
    return p;
}

int main() {
    int q;
    cin >> q;
    while (q--) {
        string s;
        cin >> s;
        int m;
        cin >> m;
        vector<int> b(m);
        for (int& x : b) cin >> x;
        auto p = build_suffix_array(s);
        string t;
        int cur = 0;
        for (int i = 1; i < s.size(); ++i) {
            while (cur + 1 < s.size() && b[cur] == b[cur + 1]) ++cur;
            if (cur + 1 >= s.size() || s[p[cur + 1]] < s[p[i]]) t += s[p[i]];
            ++cur;
        }
        cout << t << "\n";
    }
    return 0;
}
