import queue
def solve():
    n,k,l = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort()
    if (a[-1]-a[0])>l:
        return 0
    ans = 0
    q = queue.Queue()
    for x in a:
        q.put(x)
    for i in range(n):
        sum = 0
        for j in range(k):
            sum+=q.get()
        ans+=sum-sum%k
        q.put(sum%k)
    return ans
print(solve())
