#include <algorithm>
#include <cstdio>

constexpr int MOD = 998244353,MAXN=500;
long long dp[MAXN+1][MAXN+1];
int a[MAXN+1];

int main() {
    int n, m;
    scanf("%d %d", &n, &m);
    for (int i = 0; i < m; i++) {
        scanf("%d", &a[i]);
    }
    dp[0][0] = 1;
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            if (i != 0 and j == a[i-1]) continue;
            dp[i+1][j] = (dp[i+1][j] + dp[i][j]) % MOD;
            dp[i+1][a[i]] = (dp[i+1][a[i]] + dp[i][j]) % MOD;
        }
    }
    printf("%d\n", dp[m][0]);
    return 0;
}
