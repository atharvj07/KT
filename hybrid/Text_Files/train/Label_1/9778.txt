#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
  int n, k, x;
  cin >> n >> k >> x;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];

  vector<long long> dp(n + 1, -1e9);
  dp[n] = 0;
  for (int i = n - 1; i >= 0; i--) {
    dp[i] = dp[i + 1];
    if (i + k - 1 < n) dp[i] = max(dp[i], a[i] + dp[i + k]);
  }

  long long sum = 0;
  for (int i = 0; i < n && x > 0; i++) {
    if (i + k - 1 < n && a[i] + dp[i + k] > dp[i]) {
      sum += a[i];
      x--;
    }
  }

  if (x > 0) cout << -1 << endl;
  else cout << sum << endl;

  return 0;
}
