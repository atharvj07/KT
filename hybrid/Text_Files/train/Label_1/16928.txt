#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<std::vector<int>> table(100, std::vector<int>(100, 0));
  for (int i = 0; i < n; i++) {
    int x1, y1, x2, y2;
    std::cin >> x1 >> y1 >> x2 >> y2;
    for (int j = y1 - 1; j < y2; j++) {
      for (int k = x1 - 1; k < x2; k++) {
        table[j][k]++;
      }
    }
  }
  int sum = 0;
  for (int i = 0; i < 100; i++) {
    for (int j = 0; j < 100; j++) {
      sum += table[i][j];
    }
  }
  std::cout << sum << std::endl;
  return 0;
}
