t = int(input())
for _ in range(t):
    n = int(input())
    adj = [[] for _ in range(n+2)]
    for i in range(n-1):
        adj[i+1].append(i+2)
        adj[i+2].append(i+1)
    a = list(map(int,input().split()))
    for i in range(n):
        if a[i]==0:
            adj[i+1].append(n+1)
        else:
            adj[n+1].append(i+1)
    visited = [False]*(n+2)
    def dfs(u):
        visited[u] = True
        print(u,end=" ")
        for v in adj[u]:
            if not visited[v]:
                dfs(v)
    dfs(1)
    print()
