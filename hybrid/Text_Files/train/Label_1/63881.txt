import sys

def solve():
    n = int(input())
    points = []
    for i in range(n):
        x, y = map(int, input().split())
        if (x, y) in points:
            return "NO"
        points.append((x, y))

    slope = {}
    intercept = {}

    for i in range(n):
        x1, y1 = points[i]
        for j in range(i + 1, n):
            x2, y2 = points[j]
            if x2 - x1 == 0:
                if (x1, None) not in slope:
                    slope[(x1, None)] = []
                slope[(x1, None)].append(y1)
                slope[(x1, None)].append(y2)
            else:
                k = (y2 - y1) / (x2 - x1)
                if k not in slope:
                    slope[k] = []
                slope[k].append(y1 - k * x1)
                slope[k].append(y2 - k * x2)
    
    if len(slope) >= 2:
        return "YES"
    else:
        for k in slope:
            if len(set(slope[k])) > 1:
                return "YES"
    
    return "NO"

print(solve())
