import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextLong();
            b[i] = in.nextLong();
        }
        Arrays.sort(b);
        long ans = 0;
        long cnt = 0;
        for (int i = 0; i < n; i++) {
            if (cnt >= a[i]) continue;
            long need = a[i] - cnt;
            long rem = b[i] - cnt;
            ans += need * 2;
            cnt += need + (rem >= 0 ? need / rem * b[i] : 0);
        }
        System.out.println(ans);
    }
}
