#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
int main(){
    string s;  cin >> s;
    ll cnt_w = 0, cnt_m = 0;
    for(char c: s){
        if(c == 'w') cnt_w++;
        else if(c == 'm') cnt_m++;
    }
    ll res = 1;
    for(ll i = 0; i < cnt_w; i++) res = (res * 2ll) % mod;
    for(ll i = 0; i < cnt_m; i++) res = (res * 3ll) % mod;
    cout << res % mod << endl;
}
