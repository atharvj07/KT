import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] primes = {2,3,5,7};
        int len = n;
        int ans = 0;
        while(len-- > 0){
            int cur = (int) (ans * Math.pow(10, len));
            for (int prime : primes) {
                if(cur % prime != 0){
                    ans++;
                    len = n;
                    break;
                }
            }
        }

        if(ans == 0) ans = -1;
        System.out.println(ans);
    }
}
