n = int(input())
verse_pattern = list(map(int, input().split()))
text = [input() for _ in range(n)]

vowels = set("aeiouy")

def count_syllables(word):
    return sum(1 for letter in word if letter in vowels)

def check_verse_pattern(text, verse_pattern):
    for line, pattern in zip(text, verse_pattern):
        words = line.split()
        syllables_count = sum(count_syllables(word) for word in words)
        if syllables_count != pattern:
            return "NO"
    return "YES"

print(check_verse_pattern(text, verse_pattern))
