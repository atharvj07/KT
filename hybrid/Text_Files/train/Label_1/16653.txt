#include <algorithm>
#include <vector>
#include <cstdio>
using namespace std;
#define ll long long

ll n, v, ans;

struct tree {
    ll a, b;
};

bool cmp(tree a, tree b) {
    return a.a < b.a;
}

int main() {
    scanf("%lld%lld", &n, &v);
    vector<tree> t(n);
    for (ll i = 0; i < n; i++) {
        scanf("%lld%lld", &t[i].a, &t[i].b);
    }
    sort(t.begin(), t.end(), cmp);
    for (ll i = 0, cur = 0; i < n; i++) {
        if (t[i].a != cur) {
            ans += min(v, t[i].b);
            cur++;
        } else {
            ans += min(v - t[i - 1].b, t[i].b);
        }
    }
    printf("%lld", ans);
    return 0;
}
