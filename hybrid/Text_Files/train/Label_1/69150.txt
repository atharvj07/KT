import sys
input = sys.stdin.readline

def check(x1, y1, x2, y2):
    if x2-x1==1 or y2-y1==1: return 1
    else: return 2

n = int(input())
edges = []
for _ in range(n):
    x1, y1, x2, y2 = map(int, input().split())
    if x1>x2: x1, x2 = x2, x1
    if y1>y2: y1, y2 = y2, y1
    edges.append([(x1, y1, x2, y2), check(x1, y1, x2, y2)])
edges.sort(key = lambda x:x[0])
colors = [0]*n
used = set()
for i in range(n):
    x1, y1, x2, y2 = edges[i][0]
    c = edges[i][1]
    if not used:
        colors[i] = 1
        used.add(1)
    for j in range(4):
        if (j not in used) and ((x1==edges[j][0][0] and y2==edges[j][0][3]) or (x1==edges[j][0][2] and y2==edges[j][0][1]) or (x2==edges[j][0][0] and y1==edges[j][0][3]) or (x2==edges[j][0][2] and y1==edges[j][0][1])):
            colors[i] = j+1
            used.add(j+1)
            break
    if colors[i]==0:
        print('NO')
        break
else:
    print('YES')
    print(*colors)
