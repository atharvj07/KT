n = int(input())
ev = []
for i in range(n):
    s, d = map(int, input().split())
    ev.append((s, i, 1))
    ev.append((s + d, i, 2))
ev.sort()
cur = [0] * n
for t, i, tp in ev:
    if tp == 1:
        for j in range(n):
            if cur[j] <= t:
                cur[j] = t + d
                print(t, t + d - 1)
                break
        assert cur[i] > 0
    else:
        assert cur[i] > 0
        cur[i] = 0
