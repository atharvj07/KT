import sys
n,m=map(int,input().split())
a=[list(x) for x in sys.stdin.readlines()]
if n%2==1: a+=[['0']*m]
if m%2==1: [x.append('0') for x in a]
ans=0
for i in range(n):
    for j in range(m):
        if a[i][j]=='0': continue
        if (i+j)&1==0 and a[i][j]!=a[i][0]: ans+=1
        if (i+j)&1==1 and a[i][j]!=a[0][j]: ans+=1
print(ans)
