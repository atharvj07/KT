#include <bits/stdc++.h>

using namespace std;
#define rep(i, a, b) for (auto i = (a); i < (b); ++i)
#define all(c) begin(c), end(c)

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    string s, t;
    cin >> s >> t;
    rep(i, 0, n) { a[i] = s[i] - '0', b[i] = t[i] - '0'; }
    int ans = 0;
    vector<int> v;
    rep(i, 0, n) {
      if (a[i] != b[i]) {
        ans += 2;
        continue;
      }
      if (v.empty() || v.back() != a[i]) v.push_back(a[i]);
    }
    if (v.size() == 1) ans += 1;
    if (v.size() == 2) ans += 2;
    cout << ans << '\n';
  }
  return 0;
}
