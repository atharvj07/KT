import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[][] arr = new long[n][2];
        for (int i = 0; i < n; i++) {
            arr[i][0] = in.nextLong();
            arr[i][1] = in.nextLong();
        }
        Arrays.sort(arr, (a, b) -> {
            double t = 1.0 * a[0] / (a[0] - a[1]);
            double s = 1.0 * b[0] / (b[0] - b[1]);
            return Double.compare(t, s);
        });
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i][0] <= arr[i][1]) ans += arr[i][0] * 2;
            else {
                ans += arr[i][0] - arr[i][1];
                arr[i][0] = arr[i][1];
            }
        }
        System.out.println(ans);
    }
}
