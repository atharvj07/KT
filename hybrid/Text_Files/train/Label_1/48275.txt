import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[][] subjects = new int[m][3];
        for (int i = 0; i < m; i++) {
            subjects[i][0] = scanner.nextInt();
            subjects[i][1] = scanner.nextInt();
            subjects[i][2] = scanner.nextInt();
        }

        Arrays.sort(subjects, (a, b) -> Integer.compare(a[2], b[2]));

        long totalExercises = 0;
        int[] chosenSubjects = new int[n];
        int[] exercises = new int[n];
        for (int i = 0; i < n; i++) {
            int subjectIndex = -1;
            long maxExercises = -1;
            for (int j = 0; j < subjects.length; j++) {
                if (subjects[j][0] < subjects[j][1]) {
                    long currentExercises;
                    if (i == 0) {
                        currentExercises = subjects[j][0];
                    } else {
                        if (exercises[i - 1] % subjects[j][0] == 0) {
                            currentExercises = exercises[i - 1] + k;
                        } else {
                            currentExercises = exercises[i - 1] * k;
                        }
                    }
                    if (currentExercises <= subjects[j][1] && currentExercises > maxExercises) {
                        subjectIndex = j;
                        maxExercises = currentExercises;
                    }
                }
            }
            if (subjectIndex == -1) {
                System.out.println("NO");
                return;
            }
            chosenSubjects[i] = subjectIndex + 1;
            exercises[i] = (int) maxExercises;
            totalExercises += maxExercises;
            subjects[subjectIndex][0] = subjects[subjectIndex][1];
        }

        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            System.out.println(chosenSubjects[i] + " " + exercises[i]);
        }
        System.out.println("Total exercises: " + totalExercises);
    }
}
