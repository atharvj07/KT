#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> g(m + 1);
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    while (x--) {
      int y;
      cin >> y;
      g[y].push_back(i);
    }
  }
  vector<int> vis(m + 1, 0);
  int sum = 0;
  for (int i = 1; i <= m; i++) {
    if (vis[i]) continue;
    queue<int> q;
    q.push(i);
    vis[i] = 1;
    sum++;
    while (!q.empty()) {
      int cur = q.front();
      q.pop();
      for (int x : g[cur]) {
        if (vis[x]) continue;
        q.push(x);
        vis[x] = 1;
      }
    }
  }
  if (sum == 1)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
