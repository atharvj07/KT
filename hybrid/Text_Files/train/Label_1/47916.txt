import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++)
            a[i] = scanner.nextLong();

        long ans = 0;
        for (int i = 0; i < n; i++) {
            long x = 0;
            for (int j = i; j < n; j++) {
                x ^= a[j];
                if (x == 0)
                    ans += j - i + 1;
            }
        }

        System.out.println(ans);
    }
}
