for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    cnt = 0
    for i in range(n-2):
        for j in range(i+1, n-1):
            p = a[i] + 1
            q = a[j] + 1
            l = j+1
            r = n-1
            pos = -1
            while l <= r:
                mid = (l+r)//2
                if q <= a[mid]:
                    r = mid - 1
                    pos = mid
                else: l = mid + 1
            if pos != -1 and a[pos] <= p + 2: cnt += pos - j
    print(cnt)
