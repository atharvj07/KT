def find_minimal_n_gon(ang):
    for n in range(3, 998244354):
        if (n-2)*180 % n == 0 and (n-2)*180 // n >= ang and ang % (180 - (n-2)*180 // n) == 0:
            return n
    return -1

T = int(input().strip())
for _ in range(T):
    ang = int(input().strip())
    print(find_minimal_n_gon(ang))
