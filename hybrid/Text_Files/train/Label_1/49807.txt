import java.util.*;

public class Main {
  static final int MOD = 1_000_000_007;
  static int n, q, t;
  static int[] a;
  static List<Pair> lst = new ArrayList<>();

  static class Pair implements Comparable<Pair> {
    int x, y;

    Pair(int x, int y) {
      this.x = x;
      this.y = y;
    }

    @Override
    public int compareTo(Pair o) {
      int cmp = Integer.compare(x, o.x);
      return cmp == 0 ? Integer.compare(y, o.y) : cmp;
    }
  }

  public static void main(String[] args) {
    Scanner cin = new Scanner(System.in);
    n = cin.nextInt();
    q = cin.nextInt();
    t = cin.nextInt();
    a = new int[n];
    for (int i = 0; i < n; i++) a[i] = cin.nextInt();
    for (int i = 0; i < q; i++) {
      int x = cin.nextInt() - 1;
      int y = cin.nextInt() - 1;
      lst.add(new Pair(x, y));
    }
    if (!valid()) System.out.println(0);
    else System.out.println(solve(0, 0, 0, new int[n]));
    cin.close();
  }

  static boolean valid() {
    int[] cnt = new int[n];
    PriorityQueue<Pair> pq = new PriorityQueue<>();
    for (Pair p : lst) {
      if (++cnt[p.x] > cnt[p.y]) pq.add(p);
    }
    while (!pq.isEmpty()) {
      Pair p = pq.poll();
      if (--cnt[p.y] > cnt[p.x]) return false;
    }
    return true;
  }

  static int solve(int idx, int sum, int msk, int[] cnt) {
    if (sum > t) return 0;
    if (idx == n) return sum == t ? 1 : 0;
    int ret = solve(idx + 1, sum + a[idx], msk | 1 << idx, cnt);
    for (int i = 0; i < q; i++) {
      Pair p = lst.get(i);
      if ((msk & 1 << p.x) == 0) {
        boolean ok = true;
        if ((msk & 1 << p.y) != 0) {
          cnt[p.x]++;
          cnt[p.y]--;
          if (cnt[p.y] < cnt[p.x]) ok = false;
          cnt[p.x]--;
          cnt[p.y]++;
        }
        if (ok) ret = (ret + solve(idx + 1, sum + a[idx], msk | 1 << idx, cnt)) % MOD;
      }
    }
    return ret;
  }
}
