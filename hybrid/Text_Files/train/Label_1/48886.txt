import java.util.*;

public class Main {

    static int[] dist;
    static int[][] adj;
    static boolean[] vis;
    static int n, s, t;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        int m = in.nextInt();
        s = in.nextInt() - 1;
        t = in.nextInt() - 1;
        dist = new int[n];
        Arrays.fill(dist, -1);
        adj = new int[n][n];
        vis = new boolean[n];
        for (int i = 0; i < m; i++) {
            int u = in.nextInt() - 1;
            int v = in.nextInt() - 1;
            adj[u][v] = adj[v][u] = 1;
        }
        bfs();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (adj[i][j] == 0 && dist[i] == dist[j]) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }

    static void bfs() {
        Queue<Integer> q = new LinkedList<>();
        q.add(s);
        vis[s] = true;
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (int i = 0; i < n; i++) {
                if (adj[cur][i] == 1 && !vis[i]) {
                    q.add(i);
                    vis[i] = true;
                    dist[i] = dist[cur] + 1;
                }
            }
        }
    }
}
