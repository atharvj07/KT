k = int(input())
s = input()
t = input()

total = 0
for i in range(k):
    total += (ord(t[i]) - ord(s[i])) * (26 ** (k - i - 1))

total //= 2

median = ['a'] * k
for i in range(k):
    median[i] = chr(ord('a') + total // (26 ** (k - i - 1)))
    total %= 26 ** (k - i - 1)

for i in range(k):
    print(chr(ord(s[i]) + ord(median[i]) - ord('a')), end='')
print()
