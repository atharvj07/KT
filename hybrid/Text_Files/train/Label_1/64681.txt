from collections import defaultdict

for _ in range(int(input())):
    n = int(input())
    inds = defaultdict(list)
    to_add = []
    added = set()
    for _ in range(n - 1):
        a = list(map(int, input().split()))
        for i in range(1, len(a)): inds[a[i]].append(a[0:i])
    for i in range(1, n + 1):
        if i not in inds:
            to_add.append(i)
            added.add(i)
        for j in inds[i]:
            for k in j:
                if k not in added:
                    to_add.append(k)
                    added.add(k)
    print(*to_add)
