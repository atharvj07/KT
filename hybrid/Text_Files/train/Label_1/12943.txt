#include<bits/stdc++.h>
#define mod 1000000007
using namespace std;
int pre[1001], suf[1001], p[1001];
int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> p[i];
    suf[n] = 1;
    for (int i = n - 1; i >= 1; i--) {
        suf[i] = (1LL * suf[p[i + 1]] * 2) % mod;
    }
    pre[1] = 1;
    for (int i = 2; i <= n; i++) {
        pre[i] = (1LL * pre[i - 1] * 2) % mod;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        ans = (ans + 1LL * pre[i] * suf[i]) % mod;
    }
    cout << ans << '\n';
}
