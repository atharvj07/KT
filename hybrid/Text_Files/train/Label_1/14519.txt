#include <bits/stdc++.h>
using namespace std;
const int mod = 2e9;
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    int n, x;
    cin >> n >> x;
    if (n == 1 && x == 1)
        cout << 0 << endl;
    else if (n == 1) {
        cout << 2 << endl
             << "1 3" << endl;
    } else {
        cout << 3 << endl
             << x << ' ' << (mod + 1 - x) / 2 << ' ' << (mod + 1 - x) / 2 << endl;
    }
}
