#include <bits/stdc++.h>
using namespace std;

const int mxN = 1010;
int n, l[mxN], r[mxN], id = 0;

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    l[i] = i + 1, r[i] = i + 1;
  }
  if (n % 2 == 0) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  for (int i = 0; i < n; i += 2) {
    swap(l[i], r[i]);
    cout << i + 1 << ' ' << i + 2 << endl;
  }
  for (int i = 1; i < n; i += 2) {
    if (i == n - 1) continue;
    swap(l[i], r[i + 1]);
    cout << i + 1 << ' ' << i + 2 << endl;
  }
  return 0;
}
