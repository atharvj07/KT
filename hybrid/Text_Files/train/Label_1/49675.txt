import java.util.*;
public class Main{
    public static void main(String[]args){
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        sc.nextLine();
        String arr[] = new String[k];
        for(int i=0;i<k;i++){
            arr[i] = sc.nextLine();
        }
        int[] freq = new int[256];
        for(int i=0;i<k;i++){
            for(int j=0;j<(n+1)/2;j++){
                freq[arr[i].charAt(j)]++;
            }
        }
        int oddcnt = 0, idx = -1;
        for(int i=0;i<256;i++){
            if(freq[i]%k==1){
                oddcnt++;
                idx = i;
            }else if(freq[i]%k!=0){
                System.out.println(-1);
                return;
            }
        }
        if(oddcnt>1){
            System.out.println(-1);
            return;
        }
        for(int i=0;i<k;i++){
            for(int j=0;j<(n+1)/2;j++){
                if(arr[i].charAt(j)!=(char)idx&&arr[i].charAt(n-j-1)!=(char)idx){
                    for(int l=j;l<=n-j-1;l++){
                        if(arr[i].charAt(l)==arr[i].charAt(n-l-1)&&(arr[i].charAt(l)!=(char)idx||arr[i].charAt(n-l-1)!=(char)idx)){
                            for(int m=0;m<k;m++){
                                if(arr[m].charAt(l)!=arr[m].charAt(n-l-1)){
                                    System.out.println(-1);
                                    return;
                                }
                            }
                        }
                        else{
                            System.out.println(-1);
                            return;
                        }
                    }
                }
            }
        }
        char ch = (char)idx;
        String s = "";
        for(int i=0;i<n;i++){
            if(ch!=0){
                s+=ch;
                ch = 0;
            }
            else{
                s+='a';
            }
        }
        System.out.println(s);
    }
}
