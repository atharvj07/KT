#include <iostream>
#include <string>
#include <unordered_map>

int main() {
  std::string pass;
  int n;
  std::cin >> pass >> n;
  std::unordered_map<std::string, bool> dict;
  for (int i = 0; i < n; ++i) {
    std::string word;
    std::cin >> word;
    dict[word] = true;
  }
  int len = pass.size(), i = 0, j = 0;
  while (i < len && j < n) {
    std::string word = pass.substr(i, 2);
    if (dict[word]) i += 2;
    ++j;
  }
  std::cout << (i == len ? "YES" : "NO") << std::endl;
  return 0;
}
