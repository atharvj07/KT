from collections import deque

n = int(input())
t = []
for _ in range(n):
    t.append(deque(input().strip()))

ans = []
while t:
    s = t.pop()
    if not ans:
        ans.append(s)
    elif s[-1] == 's':
        ans.append(s)
    else:
        ans.insert(0, s)

noise = 0
for i in range(len(ans) - 1):
    noise += ans[i].count('s') * ans[i + 1].count('h')

print(noise)
