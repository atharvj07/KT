#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

// Find the maximum length of a non-overlapping interval between [l, r].
int max_length(const vector<pair<int, int>>& intervals) {
  int max_end = 0;
  int max_length = 0;
  for (auto& interval : intervals) {
    if (interval.first > max_end) {
      max_length = max(max_length, interval.second - interval.first + 1);
      max_end = interval.second;
    }
  }
  return max_length;
}

int main() {
  int n, x;
  cin >> n >> x;
  vector<pair<int, int>> intervals;
  for (int i = 0; i < n; i++) {
    int l, r;
    cin >> l >> r;
    intervals.push_back(make_pair(l, r));
  }
  sort(intervals.begin(), intervals.end());
  int ans = max_length(intervals);
  // Watch the rest of the movie by skipping x minutes.
  ans += (intervals.back().second + x - intervals.front().second) / x;
  cout << ans << endl;
  return 0;
}
