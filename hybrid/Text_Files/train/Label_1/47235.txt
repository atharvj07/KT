import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long count = 0;
        for (long i = 2; i * i <= n; i++) {
            while (n % i == 0) {
                n /= i;
                count++;
            }
        }
        if (n > 1) {
            count++;
        }
        System.out.println(count);
    }
}
