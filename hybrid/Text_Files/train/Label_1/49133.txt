import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        if (isGeometricProgression(a)) {
            System.out.println(0);
        } else if (canBecomeGeometricProgression(a)) {
            System.out.println(1);
        } else {
            System.out.println(2);
        }
    }

    private static boolean isGeometricProgression(int[] a) {
        for (int i = 1; i < a.length; i++) {
            if (a[i] * a[i - 1] < 0) {
                return false;
            }
        }
        return true;
    }

    private static boolean canBecomeGeometricProgression(int[] a) {
        int[] aa = new int[a.length];
        for (int i = 1; i < a.length; i++) {
            aa[i] = a[i] - a[i - 1];
        }
        return isGeometricProgression(aa);
    }
}
