import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        char[] hamsters = scanner.next().toCharArray();
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (hamsters[i] == 'x') {
                count++;
            }
        }
        if (count == n / 2) {
            System.out.println(0);
            System.out.println(String.valueOf(hamsters));
        } else {
            int min = Math.min(count, n - count);
            System.out.println(min);
            if (count < n - count) {
                for (int i = 0; i < n; i++) {
                    if (hamsters[i] == 'x') {
                        hamsters[i] = 'X';
                        min--;
                        if (min == 0) {
                            break;
                        }
                    }
                }
            } else {
                for (int i = 0; i < n; i++) {
                    if (hamsters[i] == 'X') {
                        hamsters[i] = 'x';
                        min--;
                        if (min == 0) {
                            break;
                        }
                    }
                }
            }
            System.out.println(String.valueOf(hamsters));
        }
    }
}
