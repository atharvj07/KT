n = int(input())

# If n is less than 10, it can only be split into digits equal to itself
if n < 10:
    print(1)
    print(n)
else:
    # Otherwise, we try to use as many 9s as possible, since 9 is the largest digit
    nines_count = n // 9
    remainder = n % 9
    
    if remainder != 0:
        # If there's a remainder, we need one additional digit
        print(nines_count + 1)
        print(' '.join(['9'] * nines_count + [str(remainder)]))
    else:
        # If no remainder, we just print all 9s
        print(nines_count)
        print(' '.join(['9'] * nines_count))
