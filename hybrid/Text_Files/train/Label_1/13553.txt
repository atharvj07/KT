#include <bits/stdc++.h>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<int> p(n + 1), ans(n + 1);
    iota(p.begin(), p.end(), 0);
    for (int i = 1; i <= n; ++i) {
      cin >> p[i];
    }
    for (int i = 1; i <= n; ++i) {
      if (!ans[i]) {
        ans[i] = ans[p[i]] + 1;
      }
    }
    for (int i = 1; i <= n; ++i) {
      cout << ans[i] << " ";
    }
    cout << '\n';
  }
}
