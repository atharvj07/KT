import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int digits = 0;
        int base = 1;

        while (n > 0) {
            int count = n / base;
            if (n % base > 0) {
                count++;
            }
            digits += count * base;
            n -= count * base;
            base *= 10;
        }

        System.out.println(digits);
    }
}
