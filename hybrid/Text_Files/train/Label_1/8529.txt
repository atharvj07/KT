#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ll n, m;
    cin >> n >> m;
    ll sum = 0, ans = 0;
    vector<ll> v(n);
    for (ll i = 0; i < n; i++) {
        cin >> v[i];
        if (v[i] <= m) {
            sum += v[i];
            ans = max(ans, v[i]);
        } else {
            ans = max(ans, m);
        }
    }
    cout << sum - ans << endl;
    return 0;
}
