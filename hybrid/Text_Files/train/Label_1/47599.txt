import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] deg = new int[n + 1];
        List<Integer>[] adj = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            deg[u]++;
            deg[v]++;
            adj[u].add(v);
            adj[v].add(u);
        }
        int leaves = 0;
        int root = 0;
        for (int i = 1; i <= n; i++) {
            if (deg[i] == 1) {
                leaves++;
            }
            if (deg[i] > 2) {
                root = i;
            }
        }
        if (leaves % 2 == 1 || root == 0) {
            System.out.println(-1);
        } else {
            int ans = leaves / 2 - 1;
            System.out.println(ans);
        }
    }
}
