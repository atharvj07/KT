n, m = map(int, input().split())
covered_points = set()

for _ in range(n):
    l, r = map(int, input().split())
    covered_points.update(range(l, r + 1))

uncovered_points = [x for x in range(1, m + 1) if x not in covered_points]
print(len(uncovered_points))
if len(uncovered_points) > 0:
    print(*uncovered_points)
