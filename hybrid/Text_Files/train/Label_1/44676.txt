import java.util.*;

public class Main {
    public static long maxSteps(int n, int h) {
        // dp[i][j] = the maximum number of steps needed to jump over all trees with i nodes and height j
        long[][] dp = new long[n + 1][h + 1];
        dp[1][1] = 1;
        for (int i = 2; i <= n; i++) {
            for (int j = 1; j <= h; j++) {
                // Check if the tree is a valid binary search tree
                if (i <= (1 << j) - 1) {
                    dp[i][j] = dp[i - 1][j];
                    dp[i][j] = Math.max(dp[i][j], 1 + dp[i - 1][j - 1]);
                }
            }
        }
        return dp[n][h];
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int h = scanner.nextInt();
        System.out.println(maxSteps(n, h));
    }
}
