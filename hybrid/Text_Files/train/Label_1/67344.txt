import sys

n, m = map(int, input().split())
graph = [[] for _ in range(n+1)]
for _ in range(m):
    u, v = map(int, sys.stdin.readline().split())
    graph[u].append(v)
    graph[v].append(u)

vis = [False] * (n+1)
color = [0] * (n+1)
odd = []

def dfs(node, col):
    vis[node] = True
    color[node] = col
    if col == 0:
        odd.append(node)
    for u in graph[node]:
        if not vis[u]:
            dfs(u, 1-col)

def solve():
    for i in range(1, n+1):
        if not vis[i]:
            dfs(i, 0)
    if len(odd) % 2 == 0:
        print(1)
        print(1, *odd)
        print(n-1)
        print(*[i for i in range(1, n+1) if i not in odd])
    else:
        print(-1)

solve()
