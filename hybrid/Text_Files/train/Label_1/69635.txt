import sys

def modpow(x, n, mod):
    if not n:
        return 1
    elif n % 2:
        return x * modpow(x, n-1, mod) % mod
    else:
        return modpow(x*x % mod, n//2, mod)

def solve():
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    pa = [0] * (n + 1)
    for i in range(n):
        pa[a[i]] = i
    ans = 1
    l = n - k
    for bi in b:
        ans = ans * (pa[bi] - l) % mod
        l += 1
        pa[bi] = -1
    return ans

mod = 998244353
for t in range(int(input())):
    print(solve())
