import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        int m = sc.nextInt();
        long[] b = new long[m];
        for (int i = 0; i < m; i++) {
            b[i] = sc.nextLong();
        }
        if (!Arrays.equals(a, b)) {
            System.out.println(-1);
            return;
        }
        long g = 0;
        for (long l : a) {
            g = gcd(g, l);
        }
        System.out.println(g == 1 ? n : 1);
    }

    private static long gcd(long a, long b) {
        if (b == 0) return a;
        return gcd(b, a % b);
    }
}
