import sys

n, m, d = map(int, input().split())
g = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

q = []
q.append(1)
deg = [0] * (n + 1)
deg[1] = 1

while q:
    cur = q.pop()
    if deg[cur] == d and cur == 1:
        print("YES")
        for i in range(1, n + 1):
            for j in g[i]:
                if j < i:
                    print(i, j)
        sys.exit(0)
    for i in g[cur]:
        if deg[i] >= d:
            continue
        q.append(i)
        deg[i] = deg[cur] + 1

print("NO")
