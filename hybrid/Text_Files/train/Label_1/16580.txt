#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  for (int i = 0; i < n; ++i) { cin >> b[i]; }
  for (int i = 0; i < q; ++i) {
    int l, r;
    cin >> l >> r;
    --l, --r;
    if (a[l] != b[l] || a[r] != b[r]) {
      cout << "-1\n";
    } else {
      int diff = (b[r] - a[l] + 1) / 2;
      if (diff % 2 == 0) { cout << diff / 2 << '\n'; } else { cout << "-1\n"; }
    }
  }
  return 0;
}
