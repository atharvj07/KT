#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pii pair<ll, ll>
const ll mod = 1e9 + 7;

int main() {
    ll n;
    cin >> n;
    pii P[n];
    for (ll i = 0; i < n; i++) {
        ll a, c;
        cin >> a >> c;
        P[i] = {a, c};
    }
    sort(P, P + n);
    ll ans = 0;
    ll mx1 = P[0].second, mx2 = 0;
    for (ll i = 1; i < n; i++) {
        ll a = P[i].first, c = P[i].second;
        if (a < mx1) {
            ans += mx1 - a;
        } else {
            ans += c - mx1;
            mx1 = c;
        }
        mx2 = max(mx2, c);
    }
    cout << ans << "\n";
    return 0;
}
