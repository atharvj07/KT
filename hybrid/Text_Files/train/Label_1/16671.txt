#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, p;
    cin >> n >> p;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    vector<vector<int>> dp(n + 1, vector<int>(2, 0));
    dp[0][0] = a[0] % p;
    dp[0][1] = a[0] % p;
    for (int i = 1; i < n; ++i) {
        dp[i][0] = max(dp[i - 1][0] + a[i], a[i]);
        dp[i][1] = max(dp[i - 1][1] + a[i], dp[i - 1][0]);
    }
    cout << dp[n - 1][1] << endl;
    return 0;
}
