#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll llinf = 1e18;
ll n, ans;
int main() {
  cin >> n;
  int sqr = sqrt(n);
  for (ll i = 2; i <= sqr; ++i) {
    if (n % i == 0 && n / i != i) ans = max(ans, n / i);
  }
  ans = max(ans, n);
  cout << ans << endl;
  return 0;
}
