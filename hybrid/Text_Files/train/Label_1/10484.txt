#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    sort(v.begin(), v.end());
    int ans = 0;
    for (int i = 0; i < n - 10; i++) {
        if (v[i] == 0) continue;
        int j = i + 1;
        bool ok = true;
        while (j <= i + 10) {
            if (v[j] != j - i) {
                ok = false;
                break;
            }
            j++;
        }
        if (ok) {
            ans++;
            i = j - 1;
        }
    }
    cout << ans << endl;
    return 0;
}
