#include <algorithm>
#include <iostream>

using namespace std;

int main() {
    int m, k;
    cin >> m >> k;
    int d[1001], s[1001];
    for (int i = 1; i < m; i++) cin >> d[i];
    for (int i = 1; i < m; i++) cin >> s[i];

    int ans = 0;
    int fuel = s[1];
    for (int i = 1; i < m; i++) {
        fuel -= d[i];
        if (fuel <= 0) {
            ans += k;
            fuel = s[i];
        }
        fuel += s[i];
    }

    cout << ans << endl;
}
