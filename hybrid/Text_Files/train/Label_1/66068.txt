from collections import deque, defaultdict
import sys

n, m = map(int, input().split())
a = list(map(int, input().split()))
w = defaultdict(int)
valC = 0

for _ in range(m):
    x, y, c = map(int, input().split())
    w[(x, y)] = w[(y, x)] = c
    valC += c

if m == 0:
    print(0.000000000000000)
else:
    e = [[0] * (n + 1) for _ in range(n + 1)]
    f = [[0] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(1, n + 1):
            if i == j: continue
            if (i, j) in w:
                e[i][j] = w[(i, j)]
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                e[i][j] = max(e[i][j], min(e[i][k], e[k][j]))
    for i in range(1, n + 1):
        for j in range(1, n + 1):
            if e[i][j] > 0:
                f[i][j] = f[j][i] = 1
    q = deque()
    q.append(1)
    f[1][1] = 1
    ans = 0
    valN = a[0]
    while len(q) > 0:
        x = q.popleft()
        for y in range(1, n + 1):
            if f[x][y] == 0 and e[x][y] > 0:
                q.append(y)
                f[x][y] = f[y][x] = 1
                valN += a[y - 1]
                ans += (e[x][y] + e[y][x])
    print(ans / valN)
