import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h1 = sc.nextInt();
        int a1 = sc.nextInt();
        int c1 = sc.nextInt();
        int h2 = sc.nextInt();
        int a2 = sc.nextInt();

        int n = (h2 + a2 - 1) / a1;
        System.out.println(n);

        for (int i = 0; i < n; i++) {
            if ((h2 <= a2) || (h1 < c1)) {
                System.out.println("STRIKE");
                h2 -= a1;
            } else {
                System.out.println("HEAL");
                h1 += c1;
            }
        }
    }
}
