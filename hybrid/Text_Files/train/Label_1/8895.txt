#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> ans(n, -1);
    for (int i = 0; i < n; i++) {
        if (s[i] == 'U') {
            int cnt = 1;
            for (int j = i + 1; j < n; j++) {
                if (s[j] == 'U')
                    cnt += 2;
                else
                    break;
            }
            ans[i] = cnt;
        }
    }
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == 'D') {
            int cnt = 1;
            for (int j = i - 1; j >= 0; j--) {
                if (s[j] == 'D')
                    cnt += 2;
                else
                    break;
            }
            ans[i] = cnt;
        }
    }
    for (int i = 0; i < n; i++) cout << ans[i] << " ";
    return 0;
}
