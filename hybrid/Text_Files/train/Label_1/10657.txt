#include <bits/stdc++.h>
using namespace std;
int main(){
    int ttt;
    cin >> ttt;
    while (ttt--){
        int n;
        cin >> n;
        int f=0;
        for (int i=1; i<=n; i++){
            int x;
            cin >> x;
            if (x==1){
                cout << "YES" << endl;
                f=1;
                break;
            }
            else if (x%2==0) continue;
            else{
                cout << "YES" << endl;
                f=1;
                break;
            }
        }
        if (f==0) cout << "NO" << endl;
    }
}
