import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class DynamicArrays {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int q = Integer.parseInt(line[1]);

        List<List<Integer>> arrays = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            arrays.add(new ArrayList<>());
        }

        for (int i = 0; i < q; i++) {
            line = br.readLine().split(" ");
            int op = Integer.parseInt(line[0]);
            int t = Integer.parseInt(line[1]);
            if (op == 0) {
                int x = Integer.parseInt(line[2]);
                pushBack(arrays.get(t), x);
            } else if (op == 1) {
                dump(arrays.get(t));
            } else if (op == 2) {
                clear(arrays.get(t));
            }
        }
    }

    private static void pushBack(List<Integer> array, int x) {
        array.add(x);
    }

    private static void dump(List<Integer> array) {
        if (array.isEmpty()) {
            System.out.println();
        } else {
            for (int i = 0; i < array.size(); i++) {
                System.out.print(array.get(i) + " ");
            }
            System.out.println();
        }
    }

    private static void clear(List<Integer> array) {
        array.clear();
    }
}
