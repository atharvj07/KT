#include <iostream>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, m;
    std::cin >> n >> m;
    if (m > 1) {
      if (m % 2 == 0) {
        if (m % n == 0) {
          std::cout << m / n - 1 << "\n";
        } else {
          std::cout << m / n << "\n";
        }
      } else {
        if (m % n == 0) {
          std::cout << m / n - 1 << "\n";
        } else {
          std::cout << m / n << "\n";
        }
      }
    } else {
      std::cout << 0 << "\n";
    }
  }
}
