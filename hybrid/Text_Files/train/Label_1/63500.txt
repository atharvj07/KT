import sys

input = sys.stdin.readline

def main():
    t = int(input())
    for _ in range(t):
        n, m = map(int, input().split())
        a = [list(map(int, input().split())) for _ in range(n)]
        ans = 0
        for j in range(m):
            mx = 0
            for i in range(n):
                mx = max(mx, a[i][j])
            ans += mx
        print(ans)

main()
