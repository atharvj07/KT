import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt(), t = in.nextInt();
        String s = in.next();
        int[] a = new int[k];
        for (int i = 0; i < k; ++i) {
            a[i] = in.nextInt();
        }
        int[][] b = new int[k][k];
        for (int i = 0; i < k; ++i) {
            for (int j = 0; j < k; ++j) {
                b[i][j] = in.nextInt();
            }
        }
        int cnt = 0;
        for (int i = 1 << k; i > 0; i--) {
            int sum = 0;
            boolean used = false;
            for (int j = 0; j < k; ++j) {
                if ((i & (1 << j)) != 0) {
                    if (used) {
                        sum += b[j][cnt - 1];
                    }
                    sum += a[j];
                    used = true;
                }
            }
            cnt++;
            if (sum <= t) {
                break;
            }
        }
        System.out.println(cnt);
    }
}
