#include <cstdio>
#include <cstring>
#include <vector>
#include <algorithm>

using namespace std;

const int MAXN = 1002;
int main() {
  char s[MAXN], t[MAXN];
  scanf("%s%s", s, t);
  int n = strlen(s);
  vector<pair<int, int>> ans;
  for (int i = 0; i < n; i++)
    if (s[i] != t[i]) {
      for (int j = i + 1; j < n; j++)
        if (s[i] == t[j] && s[j] == t[i]) {
          ans.push_back(make_pair(s[i], s[j]));
          swap(t[i], t[j]);
          break;
        }
      if (s[i] != t[i]) {
        puts("-1");
        return 0;
      }
    }
  printf("%d\n", ans.size());
  for (auto v : ans) printf("%c %c\n", v.first, v.second);
  return 0;
}
