n=int(input())
s=list(input())
m=int(input())
cnt=[0]*n
for i in range(0,n):
    if(s[i]=="?"):
        cnt[i]=cnt[i-1]+1
    else:
        cnt[i]=cnt[i-1]
ans=0
sm1=0
sm2=0
for i in range(0,n):
    if(i%2==0):
        if(s[i]=="?"):
            ans=ans+sm2
        elif(s[i]=="b"):
            sm1=sm1+1
        else:
            sm2=sm2+1
    else:
        if(s[i]=="?"):
            ans=ans+sm1
        elif(s[i]=="a"):
            sm1=sm1+1
        else:
            sm2=sm2+1
print(ans+cnt[n-1])
