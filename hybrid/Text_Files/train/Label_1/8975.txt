#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2e5 + 5;
char s[MAXN];
int n;

int main() {
    cin >> n >> s;
    int ans = 0;
    int cnt_a = 0, cnt_b = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'a') cnt_a++;
        else cnt_b++;
        if (i > 0) {
            if (i % 2 == 0) {
                if (cnt_a != cnt_b) ans++;
            } else {
                if (cnt_a > cnt_b) ans += cnt_a - (i + 1) / 2;
                else if (cnt_b > cnt_a) ans += cnt_b - (i + 1) / 2;
            }
        }
    }
    cout << ans << endl;
    for (int i = 0; i < n; i++) {
        if (i % 2 == 0) {
            s[i] = 'a';
        } else {
            if (cnt_a < cnt_b) s[i] = 'a', cnt_a++;
            else if (cnt_b < cnt_a) s[i] = 'b', cnt_b++;
            else {
                if (i + 1 == n) s[i] = 'a';
                else {
                    if (s[i + 1] == 'a') s[i] = 'b', cnt_b++;
                    else s[i] = 'a', cnt_a++;
                }
            }
        }
    }
    cout << s << endl;
    return 0;
}
