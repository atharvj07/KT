import java.util.Scanner;

public class AddEdges {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        // Create an adjacency matrix to represent the graph.
        boolean[][] adjMatrix = new boolean[n + 1][n + 1];

        // Read the edges from the input.
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            adjMatrix[a][b] = true;
            adjMatrix[b][a] = true;
        }

        // Check if Vertex 1 and 2 are connected.
        boolean isConnected = isConnected(adjMatrix, 1, 2);

        // If Vertex 1 and 2 are not connected, the maximum number of edges that can be added is n - 1 - m.
        // Otherwise, the maximum number of edges that can be added is n - 2 - m.
        int maxEdges = isConnected ? n - 2 - m : n - 1 - m;

        // Print the maximum number of edges that can be added.
        System.out.println(maxEdges);
    }

    // Check if Vertex u and Vertex v are connected.
    private static boolean isConnected(boolean[][] adjMatrix, int u, int v) {
        boolean[] visited = new boolean[adjMatrix.length];
        return dfs(adjMatrix, u, v, visited);
    }

    // Depth-first search to check if Vertex u and Vertex v are connected.
    private static boolean dfs(boolean[][] adjMatrix, int u, int v, boolean[] visited) {
        visited[u] = true;
        if (u == v) {
            return true;
        }
        for (int i = 1; i < adjMatrix.length; i++) {
            if (adjMatrix[u][i] && !visited[i]) {
                if (dfs(adjMatrix, i, v, visited)) {
                    return true;
                }
            }
        }
        return false;
    }
}
