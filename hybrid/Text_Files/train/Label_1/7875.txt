#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int mxN = 2e5 + 5, mxM = 2e7 + 5;
int n, m, t[mxN], c[mxN];

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; ++i) {
        cin >> t[i];
    }
    int u = 0, cnt = 0;
    for (int i = n - 1; i >= 0; i--) {
        u += t[i];
        if (u > m) c[i] = ++cnt;
    }
    for (int i = 0; i < n; ++i) { cout << c[i] << ' '; }

    return 0;
}
