#include <bits/stdc++.h>

using namespace std;

const int MAXN = 100;

int n, a[MAXN + 5];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i < n; i++) {
    if (a[i] > a[i + 1]) {
      cout << i << " " << i + 1 << "\n";
      swap(a[i], a[i + 1]);
      if (i > 1 && a[i] < a[i - 1]) i -= 2;
    }
  }
  return 0;
}
