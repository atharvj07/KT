#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    long long ans = 0;
    vector<long long> d = {2, 4, 8, 6, 5, 4, 3, 2, 1, 0};
    int v = -1;
    for (int i = 0; i < 10; i++) {
      if (n % 10 == d[i]) {
        v = i;
        break;
      }
    }
    if (v == -1) {
      cout << 0 << endl;
      continue;
    }
    if (n / 10 == 0) {
      if (v) {
        cout << (v + 9) / 2 << endl;
      } else {
        cout << 1 << endl;
      }
      continue;
    }
    int x = n / 10;
    n %= 10;
    while (x / 2) {
      x /= 2;
      if (n < 5 || !v) {
        ++n;
      } else {
        n += 2;
      }
      ans += n - 5;
      v = 0;
    }
    if (!x) {
      if (v) {
        ++ans;
      }
    } else if (v) {
      if (n != 5) {
        ans += 2;
      } else {
        ans += 1;
      }
      ++v;
      if (v < 5 || !x) {
        ++x;
      } else {
        x += 2;
      }
    } else {
      ++x;
    }
    ans += x - 1;
    cout << ans << endl;
  }
  return 0;
}
