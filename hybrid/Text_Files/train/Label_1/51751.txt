import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] s = sc.nextLine().toCharArray();
        int n = s.length;
        int qMarks = 0;
        for (char c : s)
            if (c == '?')
                qMarks++;
        System.out.println((n - qMarks + 1) / 2);
    }
}
