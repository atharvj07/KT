import random
def get_max_seg_inv(l,r):
    # pre-calculate max segment inv
    max_inv = [(i - 1) * i // 2 for i in range(1,100001)]
    return max_inv[r] - max_inv[l - 1]
def inv_add(inv,add,L):
    # update inv
    for a,b in zip(add,range(L,L + len(add))):
        inv += a * b + b * (b - 1) // 2
    return inv
def gen_seg(L):
    inv = 0
    seg = []
    while L > 1:
        l = random.randint(1,L)
        r = random.randint(l,L)
        seg.append((l,r))
        s_inv = get_max_seg_inv(l,r)
        add = [random.randint(1,s_inv+1) for _ in range(s_inv)]
        inv = inv_add(inv,add,l)
        L -= 1
    return seg,inv
for _ in range(int(input())):
    L = int(input())
    seg,inv = gen_seg(L)
    print(inv / L)
