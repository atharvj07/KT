#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 2e5+5, M = 2e5+5;
vector<int>adj[N];
int vis[N],sub[N];
int dfs(int u,int p){
    vis[u]=1;
    sub[u]=1;
    for(auto v:adj[u]){
        if(v==p)continue;
        sub[u]+=dfs(v,u);
    }
    return sub[u];
}
vector<pair<int,int> >edge;
int32_t main() {
    int n;
    cin>>n;
    for(int i=1;i<n;i++){
        int u,v;
        cin>>u>>v;
        edge.push_back({u,v});
        adj[u].push_back(v),adj[v].push_back(u);
    }
    dfs(1,-1);
    int ans=0;
    vector<pair<int,int> >res;
    for(int i=0;i<edge.size();i++){
        int u=edge[i].first,v=edge[i].second;
        if(sub[u]>sub[v])swap(u,v);
        res.push_back({sub[u],i+1});
        ans+=sub[u];
    }
    cout<<ans<<endl;
    sort(res.begin(),res.end());
    for(auto x:res){
        cout<<x.first<<" ";
        for(int i=0;i<x.first;i++)cout<<x.second<<" ";
        cout<<endl;
    }
    return 0;
}
