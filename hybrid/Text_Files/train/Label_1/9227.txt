#include<bits/stdc++.h>
using namespace std;
char a[5][5];
int main() {
  for (int i = 1; i <= 4; i++) {
    for (int j = 1; j <= 4; j++) { cin >> a[i][j]; }
  }
  for (int i = 1; i <= 4; i++) {
    for (int j = 1; j <= 4; j++) {
      if (a[i][j] == '.') {
        if ((a[i][1] == 'x' && a[i][2] == 'x' && a[i][3] == 'x') ||
            (a[1][j] == 'x' && a[2][j] == 'x' && a[3][j] == 'x') ||
            (i == j && a[1][1] == 'x' && a[2][2] == 'x' && a[3][3] == 'x') ||
            (i + j == 5 && a[1][4] == 'x' && a[2][3] == 'x' && a[3][2] == 'x')) {
          cout << "YES" << endl;
          return 0;
        }
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
