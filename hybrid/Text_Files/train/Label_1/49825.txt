import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        char[][] grid = new char[n][m];
        int[][] componentSize = new int[n][m];
        for (int i = 0; i < n; i++) {
            line = reader.readLine().split("");
            for (int j = 0; j < m; j++) {
                grid[i][j] = line[j].charAt(0);
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '*') {
                    grid[i][j] = '.';
                    dfs(i, j, grid, componentSize);
                    grid[i][j] = '*';
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                sb.append(componentSize[i][j] % 10 + ".");
            }
            sb.deleteCharAt(sb.length() - 1);
            sb.append("\n");
        }
        System.out.print(sb);
    }

    private static void dfs(int i, int j, char[][] grid, int[][] componentSize) {
        if (i < 0 || i >= grid.length || j < 0 || j >= grid[0].length || grid[i][j] == '*') {
            return;
        }
        if (componentSize[i][j] != 0) {
            return;
        }
        componentSize[i][j] += 1;
        dfs(i - 1, j, grid, componentSize);
        dfs(i + 1, j, grid, componentSize);
        dfs(i, j - 1, grid, componentSize);
        dfs(i, j + 1, grid, componentSize);
    }
}
