import java.io.PrintWriter;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        int[] s = new int[4];
        int[] a = new int[100000];
        for (int i = 0; i < 4; i++) {
            s[i] = in.nextInt();
            for (int j = 0; j < s[i]; j++) { a[j + i] = i; }
        }
        if ((s[1] + s[2] + s[3]) % 2 != 0 || s[1] == 0 || s[2] == 0 || s[3] == 0) { out.print("NO"); }
        else {
            int[] b = new int[100000];
            for (int j = (s[1] + s[2]) / 2, i = 0; i < s[1] - 1; i += 2, j += 2) { b[i] = 1; b[j + 1] = 1; }
            for (int j = (s[1] + s[2] + s[3]) / 2, i = s[1] - 1; i < s[1] + s[2] - 1; i += 2, j += 2) { b[i] = 2; b[j + 1] = 2; }
            out.print("YES");
            for (int j = 0; j < s[0]; j++) { out.print(" " + 0); }
            for (int j = 0; j < s[1]; j++) { out.print(" " + b[j]); }
            for (int j = 0; j < s[2]; j++) { out.print(" " + (b[j + s[1]] + 1)); }
            for (int j = 0; j < s[3]; j++) { out.print(" " + (b[j + s[1] + s[2]] + 2)); }
        }
        out.println();
        out.flush();
        out.close();
    }
}
