def find_sequence(x, y):
    sequence = []
    while x > 0 and y > 0:
        if x < y:
            y, x = x, y
            sequence.append((x - y, 'B'))
        else:
            sequence.append((x // y, 'A'))
        x %= y
    if x != 0 or y != 1:
        return "Impossible"
    result = []
    for count, letter in sequence[::-1]:
        if count > 0:
            result.append(f"{count}{letter}")
    return ''.join(result)

x, y = map(int, input().split())
print(find_sequence(x, y))
