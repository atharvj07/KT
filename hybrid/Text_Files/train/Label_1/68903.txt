import sys
from collections import defaultdict
from math import gcd

input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n, p = map(int, input().split())
    a = list(map(int, input().split()))

    g = defaultdict(list)
    for i in range(n - 1):
        mn = a[i]
        for j in range(i + 1, n):
            mn = gcd(mn, a[j])
            if mn == min(a[i : j + 1]):
                g[i].append((j, mn))
    
    for i in range(n - 1):
        g[i].append((i + 1, p))
    
    vis = [False] * n
    vis[0] = True
    mn_span = 0

    def dfs(node, par):
        nonlocal mn_span
        for u, w in g[node]:
            if not vis[u]:
                mn_span += w
                vis[u] = True
                dfs(u, node)
    
    dfs(0, -1)
    print(mn_span)
