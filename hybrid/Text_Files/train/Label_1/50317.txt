import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long n = sc.nextLong();
    int max = 0;
    for (long i = 1; i * i <= n; i++) {
      if (n % i == 0) {
        max = (int) i;
      }
    }
    System.out.println(max);
  }
}
