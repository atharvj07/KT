import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[] colors = new int[n];
        for (int i = 0; i < n; i++) colors[i] = in.nextInt();
        Map<Integer, Set<Integer>> graph = new HashMap<>();
        Map<Integer, Set<Integer>> adj = new HashMap<>();
        for (int i = 0; i < n; i++) {
            graph.put(colors[i], new HashSet<>());
            adj.put(colors[i], new HashSet<>());
        }
        for (int i = 0; i < m; i++) {
            int a = in.nextInt(), b = in.nextInt();
            graph.get(colors[a-1]).add(colors[b-1]);
            graph.get(colors[b-1]).add(colors[a-1]);
            adj.get(colors[a-1]).add(a);
            adj.get(colors[a-1]).add(b);
            adj.get(colors[b-1]).add(a);
            adj.get(colors[b-1]).add(b);
        }
        int max = 0, color = 0;
        for (int i : colors) {
            int count = 0;
            for (int j : graph.get(i)) {
                count += adj.get(j).size();
            }
            if (count > max) {
                max = count;
                color = i;
            }
        }
        System.out.println(color);
    }
}
