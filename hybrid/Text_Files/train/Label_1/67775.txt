import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    p = list(map(int, input().split()))
    s = list(map(int, input().rstrip()))

    q = [0] * n
    q_good = [0] * n
    q_bad = [0] * n
    idx_good = 0
    idx_bad = 0
    
    for i in range(n):
        if s[i] == 1:
            q_good[idx_good] = p[i]
            idx_good += 1
        else:
            q_bad[idx_bad] = p[i]
            idx_bad += 1
            
    idx_good -= 1
    idx_bad -= 1
    for i in range(n - 1, -1, -1):
        if idx_good >= 0 and q_good[idx_good] >= q_bad[idx_bad]:
            q[i] = q_good[idx_good]
            idx_good -= 1
        else:
            q[i] = q_bad[idx_bad]
            idx_bad -= 1
            
    return q

t = int(input())
for _ in range(t):
    print(*solve())
