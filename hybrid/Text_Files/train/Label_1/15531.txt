#include <iostream>
#include <vector>
using namespace std;

typedef unsigned long long ull;
int N, Q, T, X, Y, sm[500005], cnt[1005];
vector<ull> a[500005];

int main() {
  cin >> Q;
  for (int q = 1; q <= Q; q++) {
    cin >> T >> X >> Y;
    if (T == 1) {
      a[X].push_back(Y);
      cnt[Y % X]++;
    } else {
      for (auto v : a[X]) sm[v % X] += v;
      cout << sm[Y % X] + 1ll * Y * cnt[Y % X] << endl;
    }
  }
  return 0;
}
