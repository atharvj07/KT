#include <bits/stdc++.h>
using namespace std;
int const maxn = 1e4 + 5;
vector<int> vec;
int32_t main() {
  for (int i = 0, p = 1, x; i < maxn; p *= 3, i++) {
    x = (1 << i) * p;
    if (x >= maxn) break;
    vec.push_back(x);
  }
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    int ans = n - 1;
    for (int x : vec) {
      if (ans + x >= n) {
        ans += x;
        break;
      }
    }
    cout << ans << '\n';
  }
}
