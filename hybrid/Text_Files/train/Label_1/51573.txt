import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt(), p = in.nextInt();
        int[] a = new int[n], b = new int[k];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();
        for (int i = 0; i < k; i++) b[i] = in.nextInt();

        TreeSet<Integer> keys = new TreeSet<>(), people = new TreeSet<>();
        for (int x : b) keys.add(x);
        for (int x : a) people.add(x);

        int ans = 0;
        while (!people.isEmpty()) {
            Integer x = people.pollFirst();
            Integer y = keys.ceiling(x);
            ans = Math.max(ans, y == null ? p - x : y - x);
            if (y != null) keys.remove(y);
        }

        System.out.println(ans);
    }
}
