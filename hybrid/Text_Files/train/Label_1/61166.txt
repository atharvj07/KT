import sys
data = sys.stdin.readlines()
t = int(data[0])
for _ in range(t):
    n, q = map(int, data[1].split())
    a = list(map(int, data[2].split()))
    Sums = list(map(int, data[3:3+q]))
    for s in Sums:
        l, r = 0, n-1
        while l <= r:
            idx = (l + r) // 2
            m = (a[l] + a[r]) // 2
            if m == s:
                print('Yes')
                break
            elif m < s:
                l = idx + 1
            else:
                r = idx - 1
        else:
            print('No')
