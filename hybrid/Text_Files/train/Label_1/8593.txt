#include <iostream>
#include <map>
#include <vector>
#include <cmath>

using namespace std;

// calculate the max score for a problem with x participants solved
int maxScore(int x, int n) {
    double ratio = (double)x / n;
    if (ratio > 1. / 2) return 500;
    if (ratio > 1. / 4) return 1000;
    if (ratio > 1. / 8) return 1500;
    if (ratio > 1. / 16) return 2000;
    if (ratio > 1. / 32) return 2500;
    return 3000;
}

// calculate the score for a participant with time t for a problem with max score s
int score(int t, int s) {
    return (int)round(s * (1. - t / 250.));
}

int main() {
    int n;
    cin >> n;
    vector<vector<int>> times(n, vector<int>(5));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 5; j++) {
            cin >> times[i][j];
        }
    }

    // calculate the max score for each problem
    vector<int> maxScores(5);
    for (int i = 0; i < 5; i++) {
        int solved = 0;
        for (int j = 0; j < n; j++) {
            if (times[j][i] != -1) solved++;
        }
        maxScores[i] = maxScore(solved, n);
    }

    // calculate the score for Vasya and Petya for each problem
    vector<int> vasyaScores(5), petyaScores(5);
    for (int i = 0; i < 5; i++) {
        vasyaScores[i] = score(times[0][i], maxScores[i]);
        petyaScores[i] = score(times[1][i], maxScores[i]);
    }

    // check if Vasya can beat Petya
    int vasyaTotal = 0, petyaTotal = 0;
    for (int i = 0; i < 5; i++) {
        vasyaTotal += vasyaScores[i];
        petyaTotal += petyaScores[i];
    }
    if (vasyaTotal <= petyaTotal) {
        cout << -1 << endl;
        return 0;
    }

    // calculate the number of new accounts Vasya needs
    int newAccounts = 0;
    for (int i = 0; i < 5; i++) {
        if (vasyaScores[i] < petyaScores[i]) {
            newAccounts++;
        }
    }

    cout << newAccounts << endl;
    return 0;
}
