import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            boolean possible = true;
            for (int j = 1; j < n; j++) {
                if (a[j] != a[j - 1]) {
                    if (a[j] == a[j - 1] + 1 || a[j] == a[j - 1] - 1) {
                        possible = true;
                    } else {
                        possible = false;
                        break;
                    }
                }
            }
            System.out.println(possible ? "YES" : "NO");
        }
    }
}
