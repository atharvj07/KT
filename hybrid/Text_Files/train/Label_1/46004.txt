import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        long cnt = 0;
        for (long i = 1; i <= n; i++) {
            if (gcd(i, n) == 1) cnt++;
        }
        System.out.println(cnt);
    }
    public static long gcd(long a, long b) {
        if (b == 0) return a;
        else return gcd(b, a % b);
    }
}
