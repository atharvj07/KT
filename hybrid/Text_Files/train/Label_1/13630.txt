#include <bits/stdc++.h>
using namespace std;

const int maxn = 1e6 + 5;
int p[maxn], a[maxn], b[maxn], root[maxn], cnt;
vector<int> adj[maxn];
bool vis[maxn];

int find(int x) {
    if (root[x] == x) return x;
    return root[x] = find(root[x]);
}

void unite(int x, int y) {
    x = find(x);
    y = find(y);
    root[x] = y;
}

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> p[i], root[i] = i;
    for (int i = 1; i <= m; i++) {
        cin >> a[i] >> b[i];
        unite(a[i], b[i]);
    }
    for (int i = 1; i <= n; i++) {
        int t = find(i);
        adj[t].push_back(p[i]);
    }
    for (int i = 1; i <= n; i++) sort(adj[i].rbegin(), adj[i].rend());
    int j = 1;
    for (int i = 1; i <= n; i++) {
        for (int x : adj[i]) p[j++] = x;
    }
    for (int i = 1; i <= n; i++) cout << p[i] << ' ';
    cout << '\n';
    return 0;
}
