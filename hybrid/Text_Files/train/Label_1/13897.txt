#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main()
{
 ll m,k,i,count,count1,ans,j,p,q;
 cin>>m>>k;
 for(ans=1;;ans++)
 {
  count=0;
  count1=0;
  for(i=ans+1;i<=2*ans;i++)
  {
   p=i;
   while(p>0)
   {
    q=p%2;
    p=p/2;
    count1+=q;
   }
   if(count1==k)
   {
    count++;
   }
   count1=0;
  }
  if(count==m)
  {
   cout<<ans<<endl;
   break;
  }

 }
 return 0;
}
