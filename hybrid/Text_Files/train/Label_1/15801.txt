#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main() {
    int T;
    cin >> T;
    while (T--) {
        int n;
        cin >> n;
        vector<ll> b(n + 2);
        for (ll &x : b) { cin >> x; }
        ll sum = b[n + 1];
        sort(b.begin(), b.end(), greater<ll>());
        bool ok = b[0] != sum;
        for (int i = 1; i + 1 < n + 2; i++) {
            if (b[i] >= b[i - 1]) {
                ok = false;
                break;
            }
            sum -= b[i];
        }
        sum -= b[n + 1];
        if (sum <= 0 || ok == false) {
            cout << "-1\n";
        } else {
            for (int i = 0; i + 1 < n + 2; i++) {
                cout << b[i] << " ";
            }
            cout << "\n";
        }
    }
}
