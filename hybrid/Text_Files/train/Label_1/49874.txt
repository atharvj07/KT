import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    int ans = 0;
    HashMap<Integer, Integer> map = new HashMap<>();
    for (int i = 0; i < n; i++) {
      if (a[i] > i + 1) {
        ans = -1;
        break;
      }
      if (map.containsKey(a[i])) {
        ans++;
      } else {
        map.put(a[i], i);
      }
    }
    System.out.println(ans);
  }
}
