#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> h(n);
    for (int i = 0; i < n; i++) cin >> h[i];
    vector<int> last_min(n, 1e9), last_max(n, -1);
    last_min[n - 1] = h[n - 1];
    last_max[n - 1] = h[n - 1];
    for (int i = n - 2; i >= 0; i--) {
        last_min[i] = min(last_min[i + 1], h[i]);
        last_max[i] = max(last_max[i + 1], h[i]);
    }
    int k = 0;
    int last_pos = 0;
    for (int i = 1; i < n; i++) {
        if (last_max[i] < h[last_pos] || last_min[i] > h[last_pos]) {
            k++;
            last_pos = i;
        }
    }
    cout << k << endl;
    return 0;
}
