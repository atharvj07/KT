#include <bits/stdc++.h>

using namespace std;

const int MXN = 110;

int n;
char s[MXN];

int main() {
    memset(s, 0, sizeof s);
    cin >> n >> (s + 1);

    int cnt[10];
    memset(cnt, 0, sizeof cnt);

    int ans = 0;
    for (int i = 1; i <= n; i++) {
        int c = s[i] - '0';
        if (cnt[c] < 2) cnt[c]++;
    }

    if (cnt[0] == 2 && cnt[1] == 2 && cnt[2] >= 3 && cnt[3] >= 1 &&
        cnt[4] >= 1 && cnt[5] >= 1 && cnt[6] >= 1 && cnt[7] >= 1)
        ans++;

    cnt[0] = cnt[1] = 0;

    for (int i = 1; i <= n; i++) {
        int c = s[i] - '0';
        if (cnt[c] < 3) cnt[c]++;
    }

    if (cnt[0] == 3 && cnt[1] == 3 && cnt[2] >= 2 && cnt[3] >= 2 &&
        cnt[4] >= 2 && cnt[5] >= 2 && cnt[6] >= 2 && cnt[7] >= 2 &&
        cnt[8] >= 2 && cnt[9] >= 2)
        ans++;

    cout << ans << '\n';

    return 0;
}
