#include <iostream>
#include <cstdlib>

using namespace std;

int main() {
  int n, m, i, j, a, b;
  cin >> n >> m >> i >> j >> a >> b;
  int x1 = 1, x2 = n, y1 = 1, y2 = m;
  if ((i + j) % 2 != (x1 + y1) % 2 || (i + j) % 2 != (x1 + y2) % 2 || (i + j) % 2 != (x2 + y1) % 2 || (i + j) % 2 != (x2 + y2) % 2) {
    cout << "Poor Inna and pony!";
    return 0;
  }
  int ans = abs(i - x1) + abs(j - y1);
  ans = min(ans, abs(i - x1) + abs(j - y2));
  ans = min(ans, abs(i - x2) + abs(j - y1));
  ans = min(ans, abs(i - x2) + abs(j - y2));
  cout << ans;
  return 0;
}
