a,b,c,d = map(int,input().split())
if a+b+c+d<=2 or a+c+d==0 or b+c+d==0:
    print("NO")
else:
    print("YES")
    ans = []
    while a>0:
        ans.append(0)
        a-=1
    while b>1:
        ans.append(1)
        ans.append(2)
        b-=1
    while c>0:
        ans.append(2)
        c-=1
    while d>0:
        ans.append(3)
        d-=1
    print(*ans)
