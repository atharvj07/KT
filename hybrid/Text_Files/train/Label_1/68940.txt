n = int(input())
board = [input() for _ in range(n)]
valid = True
for i in range(n):
    for j in range(n):
        count = 0
        if i - 1 >= 0 and board[i - 1][j] == 'o':
            count += 1
        if i + 1 < n and board[i + 1][j] == 'o':
            count += 1
        if j - 1 >= 0 and board[i][j - 1] == 'o':
            count += 1
        if j + 1 < n and board[i][j + 1] == 'o':
            count += 1
        if count % 2 == 1:
            valid = False
            break
if valid:
    print("YES")
else:
    print("NO")
