#include <iostream>

using namespace std;

int main() {
  int n, x1, y1, x2, y2, a[101][101] = {0};
  cin >> n;
  while (n--) {
    cin >> x1 >> y1 >> x2 >> y2;
    a[x1][y1]++;
    a[x1][y2 + 1]--;
    a[x2 + 1][y1]--;
    a[x2 + 1][y2 + 1]++;
  }
  int sum = 0;
  for (int i = 1; i <= 100; i++) {
    for (int j = 1; j <= 100; j++) {
      a[i][j] += a[i - 1][j] + a[i][j - 1] - a[i - 1][j - 1];
      sum += a[i][j];
    }
  }
  cout << sum << endl;
  return 0;
}
