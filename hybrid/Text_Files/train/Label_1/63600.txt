from collections import deque

def bfs(adj, start):
    n = len(adj)
    dist = [float('inf')] * n
    dist[start] = 0
    q = deque([start])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if dist[v] == float('inf'):
                dist[v] = dist[u] + 1
                q.append(v)
    return dist

def solve():
    t = int(input())
    for _ in range(t):
        n, m, a, b, c = map(int, input().split())
        p = list(map(int, input().split()))
        p.sort()
        pref = [0]
        for x in p:
            pref.append(pref[-1] + x)
        
        adj = [[] for _ in range(n)]
        for _ in range(m):
            u, v = map(int, input().split())
            adj[u-1].append(v-1)
            adj[v-1].append(u-1)
        
        dist_a = bfs(adj, a-1)
        dist_b = bfs(adj, b-1)
        dist_c = bfs(adj, c-1)

        ans = float('inf')
        for i in range(n):
            if dist_a[i] + dist_b[i] + dist_c[i] < len(pref):
                ans = min(ans, pref[dist_b[i]] + pref[dist_a[i] + dist_b[i] + dist_c[i]])
        print(ans)

solve()
