import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int codingContestPrize = calculatePrize(x);
        int robotManeuverPrize = calculatePrize(y);
        int bonus = (x == 1 && y == 1) ? 400000 : 0;

        int totalPrize = codingContestPrize + robotManeuverPrize + bonus;
        System.out.println(totalPrize);
    }

    private static int calculatePrize(int rank) {
        switch (rank) {
            case 1:
                return 300000;
            case 2:
                return 200000;
            case 3:
                return 100000;
            default:
                return 0;
        }
    }
}
