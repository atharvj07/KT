#include <iostream>
#include <vector>
using namespace std;

typedef long long ll;

ll dfs(vector<pair<ll, ll> >& v, vector<vector<int> >& adj, vector<int>& dp, int node, ll sum){
    if(dp[node] != -1) return dp[node];
    dp[node] = 1;
    for(int u: adj[node]){
        if(v[u].first <= sum + v[node].second){
            dp[node] += dfs(v, adj, dp, u, sum + v[node].second - v[u].first);
        }
    }
    return dp[node];
}

int main(){
    int n;
    cin >> n;
    vector<pair<ll, ll> > v(n);
    vector<vector<int> > adj(n);
    for(int i = 0; i < n; i++){
        cin >> v[i].first >> v[i].second;
    }
    for(int i = 0; i < n; i++){
        for(int j = 0; j < n; j++){
            if(v[i].first <= v[j].second){
                adj[i].push_back(j);
            }
        }
    }

    vector<int> dp(n, -1);
    ll ans = 0;

    for(int i = 0; i < n; i++){
        ans += dfs(v, adj, dp, i, 0);
    }

    cout << ans % (1000000007) << endl;
}
