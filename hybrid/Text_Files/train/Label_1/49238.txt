import java.util.Scanner;

public class NoorasSellOut {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int f = scanner.nextInt();
        long[] ki = new long[n];
        long[] li = new long[n];
        for (int i = 0; i < n; i++) {
            ki[i] = scanner.nextLong();
            li[i] = scanner.nextLong();
        }

        long[][] dp = new long[n + 1][f + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= f; j++) {
                dp[i][j] = dp[i - 1][j];
                if (j > 0) {
                    dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - 1] + Math.min(ki[i - 1] * 2, li[i - 1]));
                }
            }
        }

        System.out.println(dp[n][f]);
    }
}
