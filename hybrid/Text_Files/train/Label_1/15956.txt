#include <bits/stdc++.h>
using namespace std;

const int mxN = 3e5 + 5, lg = 30;

int n, c, a[mxN], pre[lg][mxN], prv[mxN], ans;
pair<int, int> seg[mxN];

int main() {
    cin >> n >> c;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j < lg; j++) {
            pre[j][i] = (i + (1 << j) < n ? pre[j][i + (1 << j)] : 0);
            if (j) pre[j][i] += pre[j - 1][i];
        }
    }
    for (int i = 1; i < n; i++) {
        prv[i] = prv[i - 1];
        while (prv[i] < i && a[prv[i]] <= a[i]) prv[i] = prv[prv[i]];
    }
    for (int l = 0, r = 0; l < n; l = r) {
        while (r < n && (r - l + 1) % c) {
            if (l) seg[r] = {prv[r], pre[lg - 1][r] - pre[lg - 1][l - 1]};
            r++;
        }
        if (l == r) seg[r++] = {l, 0};
        else {
            int p = r;
            while (r < n && !(r - l + 1) % c && seg[p].first <= l) p++;
            if (r == n) break;
            seg[r] = {seg[p].first, pre[lg - 1][r] - pre[lg - 1][seg[p].first - 1]};
            r++;
        }
    }
    for (int i = 0; i < n; i++) {
        int lo = i, hi = n - 1, p = -1;
        while (lo <= hi) {
            int mid = lo + (hi - lo) / 2;
            if (seg[mid].first >= i) {
                p = mid;
                hi = mid - 1;
            } else lo = mid + 1;
        }
        if (~p) ans += max(0, seg[p].second - (a[i] * ((seg[p].first - i) / c)));
    }
    cout << ans << endl;
}
