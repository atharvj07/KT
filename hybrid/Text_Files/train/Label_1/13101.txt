#include <iostream>
#include <cmath>
#include <vector>

using namespace std;

struct Point {
    int x, y;
};

double dist(Point a, Point b) {
    return sqrt((a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y));
}

int main() {
    int n, s;
    cin >> n >> s;

    vector<Point> points(n);
    for (int i = 0; i < n; i++) {
        cin >> points[i].x >> points[i].y;
    }

    double ans = -1;
    for (int i = 0; i < n; i++) {
        double r = dist({0, 0}, points[i]);
        int sum = s;
        for (int j = 0; j < n; j++) {
            if (dist({0, 0}, points[j]) <= r) {
                sum += points[j].y;
            }
        }
        if (sum >= 1000000) {
            if (ans == -1 || ans > r) {
                ans = r;
            }
        }
    }

    cout << ans << endl;
    return 0;
}
