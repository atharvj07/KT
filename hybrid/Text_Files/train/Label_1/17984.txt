#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll solve(vector<ll>& a) {
    int n = a.size();
    vector<ll> pp;
    for(int i = 0; i < n; i++) {
        ll x = a[i];
        vector<ll> v;
        for(ll j = 1; j * j <= x; j++) {
            if(x % j == 0) {
                v.push_back(j);
                if(j * j != x) {
                    v.push_back(x / j);
                }
            }
        }
        ll sum = 0;
        for(int j = 0; j < v.size(); j++) {
            sum += v[j];
        }
        if(sum == x) {
            pp.push_back(x);
        }
    }
    if(pp.size() == n) {
        vector<ll> dp(n, 1);
        for(int i = 1; i < n; i++) {
            for(int j = 0; j < i; j++) {
                if(pp[i] % pp[j] == 0) {
                    dp[i] = max(dp[i], dp[j] + 1);
                }
            }
        }
        ll ans = 0;
        for(int i = 0; i < n; i++) {
            ans = max(ans, dp[i]);
        }
        return ans;
    }
    return -1;
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin >> n;
    vector<ll> a(n);
    for(int i = 0; i < n; i++) {
        cin >> a[i];
    }
    ll ans = solve(a);
    cout << ans << endl;
    return 0;
}
