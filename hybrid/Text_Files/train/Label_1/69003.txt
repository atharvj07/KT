import sys
input = sys.stdin.readline

n, m = map(int, input().split())
g = [[]] + [[] for _ in range(n)]
for _ in range(m): a, b = map(int, input().split()); g[a].append(b); g[b].append(a)
q = int(input())

ans = [0]
for i in range(1, q + 1):
    v = int(input())
    L = [v]
    for u in L:
        for x in g[u]:
            if x != v and x not in L:
                L.append(x)
    ans.append(0)
    for a in L:
        for b in L:
            for c in L:
                if a != b and b != c and c != a and a < b and b < c: ans[-1] += (a, b, c) in g

print(*ans)
