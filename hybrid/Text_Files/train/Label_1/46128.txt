import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int H = sc.nextInt();
        int W = sc.nextInt();
        int N = sc.nextInt();
        int s_r = sc.nextInt() - 1;
        int s_c = sc.nextInt() - 1;
        String S = sc.next();
        String T = sc.next();

        boolean[][][][] dp = new boolean[N + 1][H][W][4];
        dp[N][s_r][s_c][0] = true;
        dp[N][s_r][s_c][1] = true;
        dp[N][s_r][s_c][2] = true;
        dp[N][s_r][s_c][3] = true;

        for (int i = N - 1; i >= 0; i--) {
            for (int r = 0; r < H; r++) {
                for (int c = 0; c < W; c++) {
                    for (int d = 0; d < 4; d++) {
                        int nr = r, nc = c;
                        if (d == 0) {
                            nc--;
                        } else if (d == 1) {
                            nc++;
                        } else if (d == 2) {
                            nr--;
                        } else if (d == 3) {
                            nr++;
                        }

                        if (nr < 0 || nr >= H || nc < 0 || nc >= W) {
                            continue;
                        }

                        boolean f = false;
                        if (d == 0) {
                            f = (S.charAt(i) == 'L');
                        } else if (d == 1) {
                            f = (S.charAt(i) == 'R');
                        } else if (d == 2) {
                            f = (S.charAt(i) == 'U');
                        } else if (d == 3) {
                            f = (S.charAt(i) == 'D');
                        }

                        if (f) {
                            dp[i][r][c][d] |= dp[i + 1][nr][nc][d];
                        }

                        f = false;
                        if (d == 0) {
                            f = (T.charAt(i) == 'L');
                        } else if (d == 1) {
                            f = (T.charAt(i) == 'R');
                        } else if (d == 2) {
                            f = (T.charAt(i) == 'U');
                        } else if (d == 3) {
                            f = (T.charAt(i) == 'D');
                        }

                        if (f) {
                            dp[i][r][c][d] |= dp[i + 1][r][c][d];
                        }
                    }
                }
            }
        }

        if (dp[0][s_r][s_c][0] || dp[0][s_r][s_c][1] || dp[0][s_r][s_c][2] || dp[0][s_r][s_c][3]) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
