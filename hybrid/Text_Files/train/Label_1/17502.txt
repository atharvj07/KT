#include <cstdio>
int a[100005], b[100005], t[100005];
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) scanf("%d", a + i);
  for (int i = n, p = 0, x; i; i--) {
    x = p + a[i];
    t[i] = x / m;
    p = x % m;
  }
  for (int i = 1; i <= n; i++) printf("%d ", t[i]);
  puts("");
  return 0;
}
