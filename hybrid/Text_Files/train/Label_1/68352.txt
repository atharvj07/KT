n, t = map(int, input().split())
dp = [[0 for _ in range(i)] for i in range(1, n + 1)]

full_glasses = 0
dp[0][0] = t
for i in range(n):
    for j in range(len(dp[i])):
        if dp[i][j] > 1:
            overflow = dp[i][j] - 1
            dp[i][j] = 1
            if i + 1 < n:
                dp[i + 1][j] += overflow / 2
                dp[i + 1][j + 1] += overflow / 2
            full_glasses += 1
        elif dp[i][j] == 1:
            full_glasses += 1

print(full_glasses)
