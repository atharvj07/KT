d = int(input())
n, m = map(int, input().split())
sofas = [set() for x in range(n)]
cols = [set() for x in range(m)]
f = [0, 1, 0, -1, 1, 0, -1, 0]
ans = -1
l = r = u = d = 0
for i in range(d):
    x1, y1, x2, y2 = map(int, input().split())
    if y1==y2:
        sofas[x1-1].add((x2, y1))
        cols[y1-1].add((x1, y1))
    else:
        sofas[x1-1].add((x2, y1))
        sofas[x2-1].add((x1, y1))
a = [int(x) for x in input().split()]
for x in range(n):
    y = sofas[x]
    for xj, yj in y:
        if xj<x: l+=1
        if xj>x: r+=1
        if yj>yj: u+=1
        if yj<yj: d+=1
    if l==a[0] and r==a[1] and u==a[2] and d==a[3]: ans=x+1
print(ans)
