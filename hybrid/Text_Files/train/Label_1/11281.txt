#include <bits/stdc++.h>
using namespace std;

const int N = 1e5+9;

vector<int> adj[N];
int in[N];
int dp_sz[N];
int dp_beauty[N];
bool visited[N];
int n,m;

void dfs(int node){
    visited[node] = true;
    dp_sz[node] = 1;
    dp_beauty[node] = 0;
    for(int i=0; i<adj[node].size(); i++){
        int child = adj[node][i];
        if(!visited[child]){
            dfs(child);
            dp_sz[node] = max(dp_sz[node], dp_sz[child]+1);
            dp_beauty[node] = max(dp_beauty[node], dp_beauty[child]+1);
        }
    }
}

int main() {
    cin>>n>>m;
    while(m--){
        int u, v;
        cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
        in[u]++;
        in[v]++;
    }
    int ans = 0;
    for(int i=1; i<=n; i++){
        if(in[i]==1){
            memset(visited, false, sizeof visited);
            dfs(i);
            ans = max(ans, dp_beauty[i]*dp_sz[i]);
        }
    }
    cout<<ans<<endl;
	return 0;
}
