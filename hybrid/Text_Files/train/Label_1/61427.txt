for _ in range(int(input())):
    n = int(input())
    b = list(map(int, input().split()))
    if n + 2 < sum(b) or b[n] <= n or b[n + 1] > sum(b) - n:
        print(-1)
        continue
    x = b[n + 1]
    for i in range(n):
        if b[i] > x - n:
            x = b[i]
    for i in range(n):
        if x - b[n] <= b[i] <= x - n:
            print(b[i], end=" ")
    print()
