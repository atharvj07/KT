#include <bits/stdc++.h>
using namespace std;
int main() {
  int T, n;
  cin >> T;
  while (T--) {
    cin >> n;
    string s = to_string(n);
    for (int i = 0; i < (int)s.length(); i++) {
      int d = s[i] - '0';
      if (d != 0 && n % d != 0) {
        int j = i;
        while (s[j] == s[i]) j++;
        int x = n / d + 1;
        d = s[j] - '0';
        while (d != 0 && (n % d || s[j] == s[i])) {
          d = d == 9 ? 0 : d + 1;
          x++;
        }
        s[j] = s[i] = (d ? d : '1');
        for (; j < (int)s.length(); j++) {
          if (s[j] != s[i]) {
            d = s[j] - '0';
            while (d != 0 && (x % d || s[j] == s[i])) {
              d = d == 9 ? 0 : d + 1;
              x++;
            }
            s[j] = (d ? d : '0');
          }
        }
        break;
      }
    }
    cout << s << endl;
  }
  return 0;
}
