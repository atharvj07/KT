#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  set<char> st;
  for (int i = 0; i < k; ++i) {
    char c;
    cin >> c;
    st.insert(c);
  }
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = i; j < n; ++j) {
      if (st.count(s[j])) ans++; else break;
    }
  }
  cout << ans << endl;
  return 0;
}
