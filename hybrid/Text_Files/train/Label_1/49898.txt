import java.util.Scanner;

public class RockPaperScissorsTournament {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int q = scanner.nextInt();
        scanner.nextLine();
        char[] shapes = scanner.nextLine().toCharArray();
        int[] champions = new int[q + 1];
        champions[0] = countChampions(shapes);

        for (int i = 0; i < q; i++) {
            int player = scanner.nextInt() - 1;
            char newShape = scanner.next().charAt(0);
            shapes[player] = newShape;
            champions[i + 1] = countChampions(shapes);
        }

        for (int i = 0; i <= q; i++) {
            System.out.println(champions[i]);
        }
    }

    private static int countChampions(char[] shapes) {
        int result = 0;
        for (int i = 0; i < shapes.length; i++) {
            if (canWin(shapes, i)) {
                result++;
            }
        }
        return result;
    }

    private static boolean canWin(char[] shapes, int player) {
        int left = (player - 1 + shapes.length) % shapes.length;
        int right = (player + 1) % shapes.length;
        while (left != right) {
            if (shapes[left] != shapes[right]) {
                return shapes[player] == getWinningShape(shapes[left], shapes[right]);
            }
            left = (left - 1 + shapes.length) % shapes.length;
            right = (right + 1) % shapes.length;
        }
        return true;
    }

    private static char getWinningShape(char shape1, char shape2) {
        if (shape1 == shape2) {
            return shape1; // It's a draw
        } else if (shape1 == 'R' && shape2 == 'S' || shape1 == 'P' && shape2 == 'R' || shape1 == 'S' && shape2 == 'P') {
            return shape1;
        } else {
            return shape2;
        }
    }
}
