n, c1, c2 = map(int, input().split(' '))
s = input()
dp = [10**9]*(n+2)
dp[0] = 0
dp[1] = c1
adults = [i for i in range(n) if s[i] == '1']
for i in range(2, n+1):
    for j in adults:
        if i-j >= 1:
            dp[i] = min(dp[i], dp[j] + c1 + c2 * (i-j-1)**2)
    if i == n:
        dp[i+1] = dp[i]
print(dp[n+1])
