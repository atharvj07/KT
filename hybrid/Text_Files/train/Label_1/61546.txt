def sort_permutation(t, test_cases):
    for i in range(t):
        n = test_cases[i][0]
        a = test_cases[i][1]
        
        if sorted(a) == a:
            print(0)
            continue

        if a[0] == 1 or a[-1] == n:
            print(-1)
            continue

        idx_1 = a.index(1)
        idx_n = a.index(n)

        if (idx_1 % 2 == 0 and idx_n % 2 == 1) or (idx_1 % 2 == 1 and idx_n % 2 == 0):
            print(-1)
            continue

        steps = []
        if idx_1 % 2 == 0:
            steps.append(idx_1 + 1)
            a = a[:idx_1+1][::-1] + a[idx_1+1:]
            idx_n = a.index(n)

        if idx_n != n - 1:
            steps.append(idx_n + 2)
            a = a[:idx_n+2][::-1] + a[idx_n+2:]

        steps.append(n)
        a = a[::-1]

        if sorted(a) != a:
            print(-1)
            continue

        print(len(steps))
        print(' '.join(map(str, steps)))

t = int(input())
test_cases = []
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    test_cases.append((n, a))

sort_permutation(t, test_cases)
