#include <bits/stdc++.h>
using namespace std;

int main() {
  int tt;
  cin >> tt;
  while (tt--) {
    int n;
    cin >> n;
    vector<int> p(n << 1);
    for (int &x : p) cin >> x;
    bool ok = false;
    for (int i = 0; i < n; i++) ok |= p[i] > p[i + n];
    cout << (ok ? "YES" : "NO") << endl;
  }
}
