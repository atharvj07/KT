#include<bits/stdc++.h>
using namespace std;
char grid[510][510];
int n, m, k;
int dx[] = {0, 0, 1, -1};
int dy[] = {1, -1, 0, 0};
bool vis[510][510];
bool valid(int x, int y) {
  return (x >= 0 && x < n && y >= 0 && y < m);
}
void dfs(int x, int y) {
  vis[x][y] = 1;
  for (int i = 0; i < 4; ++i) {
    int nx = x + dx[i];
    int ny = y + dy[i];
    if (valid(nx, ny) && grid[nx][ny] != '#' && !vis[nx][ny])
      dfs(nx, ny);
  }
}
int main() {
  cin >> n >> m >> k;
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < m; ++j) cin >> grid[i][j];
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      if (grid[i][j] == '.' && !vis[i][j]) {
        dfs(i, j);
        if (ans < k) {
          grid[i][j] = 'X';
          ans++;
        }
      }
    }
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) { cout << grid[i][j]; }
    cout << "\n";
  }
  return 0;
}
