x, y, z = map(int, input().split())

total_coconuts_no_exchange = x // z + y // z
remainder_sasha = x % z
remainder_masha = y % z

if remainder_sasha + remainder_masha < z:
    print(total_coconuts_no_exchange, 0)
else:
    print(total_coconuts_no_exchange + 1, min(z - remainder_sasha, z - remainder_masha))
