import sys
input = sys.stdin.readline

n, k = map(int, input().split())
road_map = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    road_map[a].append(b)
    road_map[b].append(a)
uni = list(map(int, input().split()))

# bfs
dist = [-1] * (n+1)
dist[uni[0]] = 0
q = [uni[0]]
while q:
    cur = q.pop()
    for nxt in road_map[cur]:
        if dist[nxt] == -1:
            dist[nxt] = dist[cur] + 1
            q.append(nxt)

ans = 0
idx = [0] * (k+1)
for u in uni:
    ans += dist[u]

    for i in range(k+1):
        if dist[u] <= dist[uni[idx[i]]]:
            idx[i] = uni.index(u)
            break

print(ans)
