#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> cnt(256);
  for (char c : s) { cnt[c]++; }
  int odd = 0;
  for (int i = 0; i < 256; i++) { odd += cnt[i] & 1; }
  if (odd > 1) { cout << -1 << '\n'; return 0; }
  vector<string> ans;
  while (n > 0) {
    int i, j;
    for (i = '0'; i <= '9'; i++) {
      if (cnt[i]) {
        string t = "";
        cnt[i]--;
        n--;
        t += i;
        for (; j = n, cnt[j]; j--) {
          cnt[j]--;
          n--;
          t += j;
          if (!cnt[j]) { break; }
          t += j;
        }
        ans.push_back(t);
        break;
      }
    }
    for (i = 'a'; i <= 'z'; i++) {
      if (cnt[i]) {
        string t = "";
        cnt[i]--;
        n--;
        t += i;
        for (; j = n, cnt[j]; j--) {
          cnt[j]--;
          t += j;
          if (cnt[j] == 0) { break; }
          n--;
          t += j;
        }
        ans.push_back(t);
        break;
      }
    }
    for (i = 'A'; i <= 'Z'; i++) {
      if (cnt[i]) {
        string t = "";
        cnt[i]--;
        n--;
        t += i;
        for (; j = n, cnt[j]; j--) {
          cnt[j]--;
          t += j;
          if (cnt[j] == 0) { break; }
          n--;
          t += j;
        }
        ans.push_back(t);
        break;
      }
    }
  }
  cout << ans.size() << '\n';
  for (auto s : ans) { cout << s << ' '; }
  return 0;
}
