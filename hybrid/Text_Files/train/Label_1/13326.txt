#include <bits/stdc++.h>
using namespace std;
#define FASTIO ios::sync_with_stdio(false); cin.tie(0); cout.tie(0);
const int MAXN = 2e5 + 10;
vector<int> adj[MAXN], clr(MAXN), vis(MAXN);
int ans = 0;
void dfs(int u, int p, int c) {
    if (vis[u]) return;
    vis[u] = 1;
    if (clr[u] != clr[p] || ! clr[p]) ans++;
    for (int v: adj[u]) {
        dfs(v, u, c);
    }
}
int main() {
    FASTIO;
    int n; cin >> n;
    for (int i = 1; i <= n; i++) cin >> clr[i];
    for (int i = 1; i < n; i++) {
        int u, v; cin >> u >> v;
        adj[u].emplace_back(v);
        adj[v].emplace_back(u);
    }
    dfs(1, 1, -1);
    cout << (ans + 1) / 2 << '\n';
    return 0;
}
