import java.util.Stack;

public class Main {
    public static void main(String[] args) throws Exception {
        char[] s = new java.io.BufferedReader(new java.io.InputStreamReader(System.in)).readLine().toCharArray();
        int ans = 0, dep = 0;
        Stack<Integer> st = new Stack<>();
        for (int i = 0; i < s.length; i++) {
            if (s[i] == '(') {
                st.push(dep);
                dep++;
            } else {
                dep = st.pop();
                if (!st.isEmpty() && dep != st.peek()) {
                    ans++;
                    dep = st.peek();
                }
            }
        }
        System.out.println(ans);
    }
}
