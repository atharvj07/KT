#include<bits/stdc++.h>
#define int long long
using namespace std;
int mod, dp[505][505][2], pw[505][505];
int n, m, matriz[505][505];
int solve(int i, int cont, bool first) {
	if(i == n) {
		if(cont == 2 && first) return 1;
		return 0;
	}
	int &ans = dp[i][cont][first];
	if(~ans) return ans;
	ans = 0;
	for(int j = 0; j < n; j++) {
		int aux = cont;
		if(matriz[i][j] == 1) aux++;
		if(!first && matriz[i][j] == 1) continue;
		if(matriz[i][j] == 0 && first) {
			for(int l = i + 1; l < n; l++) {
				if(matriz[l][j] == 1) {
					ans += solve(l, aux + 1, false) * pw[l][j] % mod;
					ans %= mod;
				}
			}
		}
		else if(matriz[i][j] == 1) ans += solve(i + 1, aux, first) * pw[i][j] % mod, ans %= mod;
		else ans += solve(i + 1, aux, false) * pw[i][j] % mod, ans %= mod;
	}
	return ans;
}
main() {
	cin >> n >> m >> mod;
	for(int i = 0; i < m; i++) {
		for(int j = 0; j < n; j++) {
			char c;
			cin >> c;
			matriz[i][j] = c - '0';
		}
	}
	for(int i = 0; i < 505; i++) {
		for(int j = 0; j < 505; j++) {
			pw[i][j] = 1;
			for(int l = j + 1; l < 505; l++) pw[i][j] = pw[i][j] * 2 % mod;
		}
	}
	memset(dp, -1, sizeof dp);
	cout << solve(m, 0, true) << endl;
}
