for _ in range(int(input())):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    i = j = ans = 0
    while i<n and j<n:
        if a[i]+b[j]<=x:
            i += 1
            j += 1
            ans += 1
        else:
            j += 1
    print("YES" if ans==n else "NO")
