#include <bits/stdc++.h>

using namespace std;
const int MAX = 500;
int n, a[MAX], dp[MAX];

int func(int sum) {
  if (sum == n) return 0;
  int &ret = dp[sum];
  if (ret != -1) return ret;
  ret = n + 1;
  if (sum + 1 < n && a[sum] == a[sum + 1]) ret = min(ret, func(sum + 2) + 1);
  if (sum + 2 < n && a[sum] == a[sum + 2]) ret = min(ret, func(sum + 3) + 1);
  return ret;
}

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  memset(dp, -1, sizeof dp);
  int ans = func(0);
  if (ans == n + 1) cout << n << endl;
  else
    cout << ans << endl;
  return 0;
}
