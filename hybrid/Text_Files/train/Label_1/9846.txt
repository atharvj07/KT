#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
    ll n, k;
    cin >> n >> k;
    ll a[n], cum = 0;
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
        cum += a[i];
        if (cum >= k) {
            cout << i + 1 << endl;
            return 0;
        }
        cum = min(cum, 8LL);
    }
    cout << -1 << endl;
    return 0;
}
