#include <iostream>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b;
    cin >> a >> b;
    if (a % b == 0)
      cout << a / b;
    else {
      for (long long i = a / b; i > 0; i--) {
        if (a % i == 0 && b % i != 0) {
          cout << i;
          break;
        }
      }
    }
    cout << "\n";
  }
  return 0;
}
