import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        long n = s.nextInt();
        long[] birth = new long[(int) n];
        long[] death = new long[(int) n];
        Map<Long, Integer> m = new HashMap<>();
        for (int i = 0; i < n; i++) {
            birth[i] = s.nextLong();
            death[i] = s.nextLong();
        }
        for (long b : birth) {
            m.merge(b, 1, Integer::sum);
            m.merge(b + 1, -1, Integer::sum);
        }
        for (long d : death) {
            m.merge(d, 1, Integer::sum);
            m.merge(d + 1, -1, Integer::sum);
        }
        int max = 0;
        long year = 0;
        int current = 0;
        for (long key : m.keySet()) {
            current += m.get(key);
            if (current > max || (current == max && year > key)) {
                max = current;
                year = key;
            }
        }
        System.out.print(year + " " + max);
    }
}
