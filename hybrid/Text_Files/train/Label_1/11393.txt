#include <iostream>
#include <vector>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<vector<int>> dp(n, vector<int>(n, 0));
  for (int i = n - 1; i >= 0; i--) {
    for (int j = i; j < n; j++) {
      if (i == j) {
        dp[i][j] = 1;
      } else if (s[i] == s[j]) {
        dp[i][j] = dp[i + 1][j - 1] + 2;
      } else {
        dp[i][j] = max(dp[i + 1][j], dp[i][j - 1]);
      }
    }
  }
  int max_len = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      max_len = max(max_len, dp[i][j]);
    }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      if (dp[i][j] == max_len) {
        ans += dp[i + max_len / 2 - 1][j - max_len / 2 + 1];
      }
    }
  }
  cout << ans << endl;
  return 0;
}
