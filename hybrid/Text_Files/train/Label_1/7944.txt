#include <bits/stdc++.h>
using std::vector;
using std::pair;
using std::queue;
int main() {
  int n;
  std::cin >> n;
  vector<int> p(n), d(n, 1), c(n), del;
  for (int i = 0; i < n; ++i) {
    std::cin >> p[i] >> c[i];
    if (p[i] != -1) ++d[p[i] - 1];
  }
  queue<int> q;
  for (int i = 0; i < n; ++i)
    if (d[i] == 0) q.push(i);
  while (q.size()) {
    int u = q.front();
    q.pop();
    del.push_back(u);
    if (p[u] != -1) {
      --d[p[u] - 1];
      if (d[p[u] - 1] == 0 && c[p[u] - 1] == 1) q.push(p[u] - 1);
    }
  }
  if (del.empty())
    std::cout << -1 << std::endl;
  else {
    for (int i : del) std::cout << i + 1 << ' ';
    std::cout << std::endl;
  }
  return 0;
}
