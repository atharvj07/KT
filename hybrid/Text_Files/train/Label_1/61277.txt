from collections import Counter

def solve():
    for _ in range(int(input())):
        n = int(input())
        s, t = input(), input()
        if Counter(s) != Counter(t):
            print(-1)
            continue
        op = 0
        i = n // 2
        while i >= 1:
            if s[i] != t[i]:
                if s[i] == t[n - i + 1]:
                    op += i
                else:
                    op += n - i
            i -= 1
        print(op)

solve()
