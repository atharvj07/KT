import sys
a,b,c,d=map(int, sys.stdin.readline().strip().split())
l=[(x,y) for x in range(1,a+1) for y in range(1,b+1)]
if (a+b)&1: l=[(c,d)]+l+[(x,y) for x in range(max(1,c-a+1),min(a,c+a-1)+1)+range(max(1,c-b+1),min(b,c+b-1)+1) if x!=c for y in range(1,b+1)]
else: l=[[c,y] for y in range(1,b+1)]+l if c<=(a+b)//2 else l+[(c,y) for y in range(1,b+1)][::-1]
for x,y in l: sys.stdout.write(f'{x} {y}\n')
