#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  string s, t;
  cin >> s >> t;
  vector<bool> s_used(n, false);
  vector<bool> t_used(m, false);
  int ans = 0;
  for (int i = 0; i < n;) {
    int match_idx = -1;
    for (int j = 0; j < m; j++) {
      if (t[j] == s[i] && !s_used[i] && !t_used[j]) {
        match_idx = j;
        s_used[i] = t_used[j] = true;
        bool found = false;
        for (int l = i + 1; l < n; l++) {
          if (!s_used[l] && t[match_idx + l - i] == s[l]) {
            s_used[l] = true;
            found = true;
          } else {
            found = false;
            break;
          }
        }
        if (found) {
          ans += n - i;
          break;
        } else {
          s_used[i] = t_used[j] = false;
        }
      }
    }
    if (match_idx == -1)
      i++;
    else
      i = n;
  }
  cout << ans << endl;
}
