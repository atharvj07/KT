import sys
input = sys.stdin.readline
mod = 10**9 + 7

n = int(input())
a = list(map(int, input().split()))
ans = n - 1
for i in range(1, n + 1):
    if a[i - 1] == -1:
        continue
    c = 0
    x = i
    while x != -1:
        c += 1
        x = a[x - 1]
    ans += n // c - 1
print(ans % mod)
