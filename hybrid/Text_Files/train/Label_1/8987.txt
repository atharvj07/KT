#include <bits/stdc++.h>
using namespace std;
#define int long long

signed main() {
    int n; cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = n - 1; i >= 0; i--) {
        b[i] = n - a[i] - 1;
    }
    vector<int> rem;
    for (int i = 0; i < n; i++) {
        if (b[i] == 0) {
            rem.push_back(i + 1);
        }
    }
    for (int i = 0; i < n; i++) {
        if (b[i] > rem.size()) {
            cout << "Impossible" << endl;
            return 0;
        }
        for (int j = 0; j < b[i]; j++) {
            rem.erase(remove(rem.begin(), rem.end(), a[i] + 1), rem.end());
        }
    }
    cout << "Possible" << endl;
    for (int i = 0; i < n; i++) {
        cout << rem[i] << ' ';
    }
    cout << endl;
}
