import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class PeopleWalking {

    private static class Person {
        int position;
        int direction;

        public Person(int position, int direction) {
            this.position = position;
            this.direction = direction;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int t = scanner.nextInt();
        int q = scanner.nextInt();

        List<Person> people = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int ai = scanner.nextInt();
            int di = scanner.nextInt();
            people.add(new Person(ai, di));
        }

        List<Integer> importantPeople = new ArrayList<>();
        for (int i = 0; i < q; i++) {
            importantPeople.add(scanner.nextInt() - 1);
        }

        for (int i : importantPeople) {
            Person person = people.get(i);
            int finalPosition = person.position + person.direction * t;
            System.out.println(finalPosition);
        }
    }
}
