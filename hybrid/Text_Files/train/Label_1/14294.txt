#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
const ld EPS = 1e-7;
const ll INF = 1001001001001001001ll;
#define endl "\n"
int main() {
  ll m, t, r;
  cin >> m >> t >> r;
  vector<ll> v(m);
  for (ll& i : v) { cin >> i; }
  ll ans = INF;
  for (ll i = v[0]; i <= 299; i++) {
    ll cnt = 0, cur = i;
    for (ll j = 0; j < m; j++) {
      if (cur < v[j]) { cur = v[j]; }
      if (cur + t < v[j] + 1) { cnt++; }
      cur += t;
    }
    if (cnt >= r) { ans = min(ans, cnt); }
  }
  cout << (ans == INF ? -1 : ans) << endl;
}
