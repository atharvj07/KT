#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.rbegin(), a.rend());
  long long sum = 0;
  for (int i = 0; i < n; i++) {
    cout << max(a[i], sum + 1) << ' ';
    sum += max(a[i], sum + 1) - a[i];
  }
  cout << endl;
  return 0;
}
