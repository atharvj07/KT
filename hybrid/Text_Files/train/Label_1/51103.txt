import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            int d = sc.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }
            int[] b = new int[k + 1];
            int j = 0;
            while (j < d && j < n) {
                b[a[j]]++;
                j++;
            }
            int ans = 0;
            if (j == d) {
                ans = k;
            } else {
                ans = k - 1;
            }
            while (j < n) {
                b[a[j - d]]--;
                b[a[j]]++;
                if (b[0] == 0) {
                    ans--;
                }
                j++;
                if (j - d >= 0 && b[a[j - d]] == 0) {
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
