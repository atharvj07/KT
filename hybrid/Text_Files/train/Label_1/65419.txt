import sys

n, m = map(int, sys.stdin.readline().split())
edges = [list(map(int, sys.stdin.readline().split())) for _ in range(m)]

graph = [[] for _ in range(n+1)]
for u, v in edges:
    graph[u].append(v)

visited = [False] * (n+1)
on_stack = [False] * (n+1)
stack = []
cycle = False

def dfs(v):
    global cycle
    visited[v] = True
    on_stack[v] = True
    stack.append(v)
    for u in graph[v]:
        if not visited[u]:
            if dfs(u):
                return True
        elif on_stack[u]:
            cycle = True
            return True
    on_stack[v] = False
    return cycle

for i in range(1, n+1):
    if not visited[i] and dfs(i):
        break

if cycle:
    print('YES')
else:
    print('NO')
