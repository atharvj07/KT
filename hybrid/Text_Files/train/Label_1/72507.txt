from collections import deque

for _ in range(int(input())):
    n = int(input())
    q = deque()
    q.append((n, 0))
    vis = set()
    ans = -1
    while q:
        cur, cnt = q.popleft()
        if cur == 1:
            ans = cnt
            break
        if cur in vis:
            continue
        vis.add(cur)
        if cur % 2 == 0 and cur // 2 not in vis:
            q.append((cur // 2, cnt + 1))
        if cur % 3 == 0 and cur // 3 not in vis:
            q.append((cur // 3, cnt + 1))
        if cur % 5 == 0 and cur // 5 not in vis:
            q.append((cur // 5, cnt + 1))
    print(ans)
