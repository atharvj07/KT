def find_best_arrival_time(ts, tf, t, n, arrival_times):
    best_time = ts
    min_wait = float('inf')
    
    current_time = ts
    for i in range(n):
        if current_time < arrival_times[i]:
            wait_time = max(0, arrival_times[i] - current_time)
            if wait_time < min_wait:
                min_wait = wait_time
                best_time = current_time
            current_time = arrival_times[i]
        elif current_time == arrival_times[i]:
            current_time += t
        else:
            wait_time = current_time - arrival_times[i]
            if wait_time < min_wait:
                min_wait = wait_time
                best_time = arrival_times[i]
            current_time += t
    
    # After all visitors
    if current_time < tf and (tf - current_time) >= t:
        wait_time = current_time - ts
        if wait_time < min_wait:
            best_time = current_time
    
    return best_time

# Input reading
ts, tf, t = map(int, input().split())
n = int(input())
arrival_times = list(map(int, input().split())) if n > 0 else []

# Processing and Output
print(find_best_arrival_time(ts, tf, t, n, arrival_times))
