#include<bits/stdc++.h>
using namespace std;
#define N 105
#define M 10005
int T=1;
int dx[]={0,0,-1,-1,-1,1,1,1};
int dy[]={-1,1,-1,0,1,-1,0,1};
int vis[N][M];
int n,m,ans;
char arr[N][M];
void dfs(int x,int y,int dir){
  if(x==n){ans=T;return;}
  if(vis[x][y]!=T)return;
  vis[x][y]=T;
  int xx=x+dx[dir],yy=y+dy[dir];
  if(arr[xx][yy]=='.')dfs(xx,yy,dir);
  else if(arr[xx][yy]=='+'){dfs(x,y,(dir+4)%8);arr[xx][yy]='.';}
  else if(arr[xx][yy]=='#')dfs(x,y,(dir+4)%8);
  return ;
}
int main(){
  cin>>n>>m;
  for(int i=1;i<=n;i++){
    for(int j=1;j<=m;j++){
      cin>>arr[i][j];
      if(arr[1][j]=='.')dfs(1,j,1);
    }
  }
  if(ans)cout<<ans<<endl;
  else cout<<"Never"<<endl;
  return 0;
}
