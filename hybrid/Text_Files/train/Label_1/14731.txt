#include <bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    vector<pair<int, int>> v;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        v.emplace_back(a[i], i);
    }
    sort(v.begin(), v.end());
    int l = 0, r = n - 1;
    while (l < r) {
        cout << v[l].first << ' ';
        l++;
        cout << v[r].first << ' ';
        r--;
    }
    if (l == r) { cout << v[l].first << ' '; }
    cout << '\n';
    return 0;
}
