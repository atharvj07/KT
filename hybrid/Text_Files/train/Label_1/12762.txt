#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
const int M=1e9+7;
int f[72],fr[72];
int main(){
	int n;
	cin>>n;
	for(int i=0; i<n; i++){
		int a;
		cin>>a;
		fr[a]++;
	}
	ll s=0;
	for(int i=1; i<=70; i++){
		for(int j=i; j<=70; j+=i)
			(s+=1LL*fr[i]*fr[j])%=M;
	}
	cout<<s%M;
}
