import java.util.*;
import java.lang.*;

public class Main {
    static int[] a;
    static int[] b;
    static double[][] dp;
    static double eps = 1e-9;
    static int k;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        k = sc.nextInt();
        dp = new double[n][n * 100 + 1];
        a = new int[n];
        b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        double ans = solve(0, 0);
        if (ans < eps) {
            System.out.println(-1);
            return;
        }
        System.out.println((int) ans);
    }

    static double solve(int pos, int sumCalories) {
        if (pos == a.length) {
            if (Math.abs((double) sumCalories / 1.0 / k - 1.0) < eps) {
                return 0;
            } else {
                return -1e9;
            }
        }
        if (dp[pos][sumCalories] != 0) {
            return dp[pos][sumCalories];
        }
        dp[pos][sumCalories] = Math.max(solve(pos + 1, sumCalories), solve(pos + 1, sumCalories + b[pos]) + a[pos]);
        dp[pos][sumCalories] = Math.max(dp[pos][sumCalories], solve(pos + 1, sumCalories + b[pos] * 2) + a[pos]);
        return dp[pos][sumCalories];
    }
}
