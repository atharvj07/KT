#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<pair<int, int>> exams(m);
  vector<int> s(m), d(m), c(m);
  for (int i = 0; i < m; i++) {
    cin >> s[i] >> d[i] >> c[i];
    exams[i] = {s[i], i + 1};
  }
  sort(exams.begin(), exams.end());
  vector<int> schedule(n + 1, 0);
  for (int i = 0; i < m; i++) {
    int start = exams[i].first;
    int end = d[exams[i].second - 1];
    if (end - start < c[exams[i].second - 1]) {
      cout << -1;
      return 0;
    }
    for (int j = start; j < end; j++) {
      if (schedule[j] != 0) {
        cout << -1;
        return 0;
      }
    }
    for (int j = start; j < start + c[exams[i].second - 1]; j++) {
      schedule[j] = exams[i].second;
    }
    for (int j = start + c[exams[i].second - 1]; j < end; j++) {
      schedule[j] = m + 1;
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << schedule[i] << " ";
  }
  return 0;
}
