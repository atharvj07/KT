#include <iostream>
#include <vector>
using namespace std;
const int MAXN = 100005;
vector<int> g[MAXN];
bool vis[MAXN];
int level[MAXN], s[MAXN], e[MAXN], t = 0;
void dfs(int u, int p, int d) {
  s[u] = ++t;
  level[u] = d;
  for (int v : g[u]) {
    if (v == p) continue;
    dfs(v, u, d + 1);
  }
  e[u] = t;
}
int main() {
  int n, u, v;
  cin >> n;
  for (int i = 1; i < n; i++) {
    cin >> u >> v;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  dfs(1, 1, 0);
  vector<pair<int, int>> nested;
  for (int i = n; i > 0; i--) {
    if (vis[i]) continue;
    int ans = 1;
    int l = level[i];
    int r = level[i];
    for (int v : g[i]) {
      if (s[v] < s[i] || e[v] > e[i]) continue;
      ans++;
      l = min(l, level[v]);
      r = max(r, level[v]);
      vis[v] = true;
    }
    if (ans > 1) nested.push_back({l, r});
  }
  cout << nested.size() << endl;
  return 0;
}
