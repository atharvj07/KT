import java.util.*;
public class Main {
public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    boolean isConnect[][] = new boolean[n+1][n+1];
    for (int i = 0; i < m; i++) {
        int u = sc.nextInt();
        int v = sc.nextInt();
        isConnect[u][v] = true;
    }
    int d[][] = new int[n+1][n+1];
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            d[i][j] = Integer.MAX_VALUE;
        }
    }
    for (int i = 1; i <= n; i++) {
        dfs(i, i, 0, d, isConnect);
    }
    int max = 0;
    int ans1 = 0, ans2 = 0, ans3 = 0, ans4 = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = i+1; j <= n; j++) {
            for (int k = j+1; k <= n; k++) {
                for (int l = k+1; l <= n; l++) {
                    if (d[i][j] != Integer.MAX_VALUE && d[j][k] != Integer.MAX_VALUE && d[k][l] != Integer.MAX_VALUE && (d[i][j]+d[j][k]+d[k][l]+d[l][i] > max)) {
                        max = d[i][j]+d[j][k]+d[k][l]+d[l][i];
                        ans1 = i;
                        ans2 = j;
                        ans3 = k;
                        ans4 = l;
                    }
                }
            }
        }
    }
    System.out.println(ans1+" "+ans2+" "+ans3+" "+ans4);
}
public static void dfs(int s, int cur, int dis, int d[][], boolean isConnect[][]) {
    if (dis > d[s][cur]) return;
    d[s][cur] = dis;
    for (int i = 1; i <= isConnect.length-1; i++) {
        if (isConnect[cur][i] == true) dfs(s, i, dis+1, d, isConnect);
    }
}
}
