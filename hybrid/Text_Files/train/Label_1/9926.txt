#include<bits/stdc++.h>
using namespace std;
int n,m,q,cnt,has[200005],tot[200005];
int get(int x){
    if(has[x]==x)return x;
    return has[x]=get(has[x]);
}
void mer(int x,int y){
    has[get(x)]=get(y);
}
bool vis[200005];
int main(){
    cin>>n>>m>>q;
    for(int i=1;i<=n*m;i++)has[i]=i;
    while(q--){
        int x,y;
        cin>>x>>y;
        if(vis[x*m+y])continue;
        vis[x*m+y]=1;
        cnt++;
        tot[get(x*m+y)]++;
    }
    bool asd=0;
    for(int i=1;i<=n*m;i++){
        if(has[i]==i){
            if(asd){
                cout<<cnt<<endl;
                exit(0);
            }
            asd=1;
            if(tot[i]>1){
                cout<<cnt<<endl;
                exit(0);
            }
        }
    }
    cout<<cnt+1<<endl;
}
