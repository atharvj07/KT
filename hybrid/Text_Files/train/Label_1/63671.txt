import sys

n, m = map(int, input().split())
a = list(map(int, input().split()))

l = [0] * (m + 1)
r = [0] * (m + 1)
for k in range(m):
    li, ri = map(int, input().split())
    l[k] = li
    r[k] = ri
    for j in range(li, ri + 1):
        a[j - 1] = min(a[j - 1], k + 1)

# print(a)

mxL = minV = a[0]
for v in a:
    minV = min(minV, v)
    mxL = max(mxL, v)

ans = mxL - minV
line = [mxL - minV]

cnt = 0
for now in range(m):
    for j in range(l[now], r[now] + 1):
        if a[j - 1] > now + 1:
            cnt += 1
            line.append(now + 1)
            mxL = max(mxL, now + 1)
            ans = mxL - minV
            break

print(ans)
print(cnt)
for b in line[1:]:
    print(b, end=' ')
    
sys.stdout.flush()
