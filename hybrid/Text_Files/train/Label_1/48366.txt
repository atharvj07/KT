import java.util.Scanner;

public class WordSolver {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();

        StringBuilder numbers = new StringBuilder();
        StringBuilder otherWords = new StringBuilder();
        boolean hasNumbers = false;
        boolean hasOtherWords = false;

        for (String word : s.split("[,;]")) {
            if (isNumber(word)) {
                hasNumbers = true;
                numbers.append(word).append(",");
            } else {
                hasOtherWords = true;
                otherWords.append(word).append(",");
            }
        }

        if (hasNumbers) {
            numbers.deleteCharAt(numbers.length() - 1); // Remove trailing comma
        }
        if (hasOtherWords) {
            otherWords.deleteCharAt(otherWords.length() - 1); // Remove trailing comma
        }

        String result1 = hasNumbers ? "\"" + numbers + "\"" : "-";
        String result2 = hasOtherWords ? "\"" + otherWords + "\"" : "-";

        System.out.println(result1);
        System.out.println(result2);
    }

    private static boolean isNumber(String word) {
        if (word.isEmpty()) {
            return false;
        }
        for (char c : word.toCharArray()) {
            if (!Character.isDigit(c)) {
                return false;
            }
        }
        return true;
    }
}
