n = int(input())
s = input()

# Count occurrences of each nucleotide and question marks
counts = {'A': 0, 'C': 0, 'G': 0, 'T': 0, '?': 0}
for char in s:
    counts[char] += 1

# Number of question marks
question_marks = counts['?']

# Calculate how many nucleotides need to be added to each to make them equal
needed = [max(0, (n // 4) - counts[nucleotide]) for nucleotide in "ACGT"]

# If the sum of needed nucleotides is not equal to the number of question marks, or n is not divisible by 4, print '==='
if sum(needed) != question_marks or n % 4 != 0:
    print("===")
else:
    # Replace '?' in the string with the needed nucleotides
    result = list(s)
    for i, char in enumerate(result):
        if char == '?':
            for j in range(4):
                if needed[j] > 0:
                    result[i] = "ACGT"[j]
                    needed[j] -= 1
                    break
    print(''.join(result))
