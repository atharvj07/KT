import sys
input = sys.stdin.readline

def solve(n, l, r):
  if r - l + 1 == 1: return [1]
  start = (l + 1) // 2
  out = [start]
  vis = [0] * (n + 1)
  vis[start] = 1
  cur = start
  for i in range(n * (n - 1) - 1):
    if i + 1 == r: out.append(cur)
    for j in range(1, n + 1):
      if i + 1 >= l and j == cur: continue
      if vis[j] == 0 and cur != j:
        out.append(j)
        vis[j] = 1
        cur = j
        break
  if r == n * (n - 1): out.append(start)
  return out[l - 1:]

for i in range(int(input())):
  n, l, r = map(int, input().split())
  print(*solve(n, l, r))
