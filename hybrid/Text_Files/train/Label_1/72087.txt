from sys import stdin

for _ in range(int(stdin.readline())):
    n, k = map(int, stdin.readline().split())
    a = list(map(int, stdin.readline().split()))
    for i in range(n):
        if k == 0:
            break
        if a[i] == 0:
            continue
        pos = -1
        for j in range(i + 1, n):
            if a[j] > 0:
                pos = j
                break
        if pos == -1:
            break
        a[i] -= 1
        a[pos] += 1
        k -= 1
    print(*a)
