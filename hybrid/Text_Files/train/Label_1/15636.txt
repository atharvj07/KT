#include <bits/stdc++.h>
#define ll long long
#define mod 1000000007
using namespace std;
ll power(ll a, ll b, ll p) {
    if (b == 0) return 1;
    if (b == 1) return a;
    ll ans = power(a, b / 2, p);
    if (b % 2 == 0) {
        return ((ans * ans) % p);
    } else {
        return (((ans * ans) % p) * a) % p;
    }
}
ll modInverse(ll a, ll p) { return power(a, p - 2, p); }
int main() {
    ll n, k;
    cin >> n >> k;
    ll arr[n + 1];
    unordered_map<ll, ll> freq;
    ll zero_count = 0;
    for (ll i = 0; i < n; i++) {
        cin >> arr[i];
        if (arr[i] == 0) zero_count++;
        freq[arr[i]]++;
    }
    ll ans = (modInverse(2, freq[0] + freq[1]) % mod) % mod;
    ll val = (power(ans, k, mod) % mod * zero_count % mod) % mod;
    ll inv = modInverse(freq[0] + freq[1], mod);
    cout << (val * inv) % mod << endl;
}
