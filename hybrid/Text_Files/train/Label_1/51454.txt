import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), p = sc.nextInt(), q = sc.nextInt();
        long[] c = new long[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }

        long[][] dp = new long[n + 1][3];
        dp[n][0] = q;
        dp[n][1] = dp[n][2] = Long.MIN_VALUE / 2;
        for (int i = n - 1; i >= 0; i--) {
            dp[i][0] = Math.max(dp[i + 1][0], Math.max(dp[i + 1][1], dp[i + 1][2]));
            dp[i][1] = c[i] + dp[i + 1][0];
            dp[i][2] = p * dp[i + 1][1];
        }

        System.out.println(Math.max(dp[0][0], Math.max(dp[0][1], dp[0][2])));
    }
}
