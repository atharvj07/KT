#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int &x : p) cin >> x;
  int neg = 0;
  for (int i = 0; i < n; i++) {
    if (p[i] < 0) neg++;
  }
  int ans = neg * 1LL * (neg - 1) / 2;
  int cur = 0;
  for (int i = 0; i < n; i++) {
    if (p[i] > 0) ans += cur;
    if (p[i] < 0) cur++;
  }
  cout << ans << endl;
}
