#include<bits/stdc++.h>
using namespace std;

int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n,i,arr[305],mini=INT_MAX;
        cin>>n;
        for(i=0;i<n;i++)
        {
            cin>>arr[i];
            mini=min(mini,arr[i]);
        }
        int ans=(arr[0]*arr[1])/mini;
        for(i=0;i<n;i++)
        {
            if(ans%arr[i]!=0)
            {
                ans=-1;
                break;
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
