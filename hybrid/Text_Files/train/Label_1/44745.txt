import java.util.Scanner;

public class RemainingNumber {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int T = scanner.nextInt();

        for (int i = 0; i < T; i++) {
            int n = scanner.nextInt();
            int x = scanner.nextInt();

            // Find the first step when x is wiped
            int firstWipeStep = (x + 1) / 2;

            // Find the number of elements remaining after wiping
            int remainingElements = n - firstWipeStep;

            // Find the position of x in the remaining elements
            int positionInRemaining = x - (firstWipeStep - 1) * 2;

            // Find the value of the x-th remaining number
            int value = positionInRemaining * 2;

            System.out.println(value);
        }
    }
}
