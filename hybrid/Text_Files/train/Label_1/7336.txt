#include <iostream>

using namespace std;

int main() {
  int a, b;
  cin >> a >> b;
  int ans = 0;
  for (int i = a; i <= b; i++) {
    int num = i, seg = 0;
    while (num > 0) {
      int digit = num % 10;
      if (digit == 0 || digit == 6 || digit == 9)
        seg += 6;
      else if (digit == 2 || digit == 3 || digit == 5 || digit == 7 || digit == 8)
        seg += 5;
      else
        seg += 2;
      num /= 10;
    }
    ans += seg;
  }
  cout << ans << endl;
  return 0;
}
