#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  if (k > a.back() - a[0] + 1) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    for (int i = 0; i < k; i++) {
      cout << (lower_bound(a.begin(), a.end(), a[0] + i) - a.begin()) + 1 << " ";
    }
    cout << endl;
  }
  return 0;
}
