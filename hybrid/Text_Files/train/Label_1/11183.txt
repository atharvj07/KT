#include<bits/stdc++.h>
using namespace std;
const int maxn=5e4+5;
int n,l,r,a[maxn];
int main(){
    cin >> n >> l >> r;
    for(int i=1;i<=r;++i) a[i]=i;
    for(int i=r+1;i<=n;++i) a[i]=a[i-1]+2;
    int mm=0,Mi=0;
    for(int i=1;i<=l;++i)
    { if(i!=l) mm+=a[i],Mi+=a[i];
      else mm+=1,Mi+=a[i];
    }
    for(int i=l+1;i<=r;++i)
    { if(i!=r) mm+=a[i],Mi+=a[i-l];
      else mm+=1,Mi+=a[i-l];
    }
    for(int i=r+1;i<=n;++i)
    { if(i!=n) mm+=a[i],Mi+=a[i-r];
      else mm+=1,Mi+=a[i-r];
    }
    cout << Mi << " " << mm << endl;
    return 0;
}
