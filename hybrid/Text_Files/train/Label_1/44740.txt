import java.util.Scanner;

public class Main {
    static int[] par;
    static int[] rank;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        par = new int[n];
        rank = new int[n];
        for (int i = 0; i < n; i++) {
            par[i] = i;
        }
        long res = 0;
        for (int i = 0; i < q; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            long w = (long) (c + 6) * (long) (c + 3) / 2;
            res += w;
            combine(a, b);
            combine(b, (a + 1) % n);
        }
        System.out.println(res);
    }

    static int find(int x) {
        if (par[x] == x) {
            return x;
        } else {
            return par[x] = find(par[x]);
        }
    }

    static void combine(int a, int b) {
        int x = find(a);
        int y = find(b);
        if (x != y) {
            if (rank[x] < rank[y]) {
                par[x] = y;
            } else if (rank[x] > rank[y]) {
                par[y] = x;
            } else {
                par[y] = x;
                rank[x]++;
            }
        }
    }
}
