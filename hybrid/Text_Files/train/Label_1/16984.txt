#include<iostream>
#include<string>
using namespace std;
int main() {
  int n;
  cin >> n;
  string a;
  cin >> a;
  for (int i = 0; i < n; ++i) {
    if (a[i] == 'a' || a[i] == 'e' || a[i] == 'i' || a[i] == 'o' || a[i] == 'u' || a[i] == 'y') {
      if (a[i] == a[i - 1]) {
        a.erase(i, 1);
        --i;
      }
    }
  }
  cout << a << '\n';
}
