n = int(input())
b = list(map(int, input().split()))
c = list(map(int, input().split()))

a = [0] * n
a[0] = b[0]
for i in range(1, n-1):
    a[i] = min(b[i], c[i-1])

a[-1] = c[-1]
for i in range(n-1):
    if b[i] != min(a[i], a[i+1]) or c[i] != max(a[i], a[i+1]):
        print(-1)
        break
else:
    print(*a)
