import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class UniverseCount {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        Set<Integer> xCoordinates = new HashSet<>();
        Set<Integer> yCoordinates = new HashSet<>();

        for (int i = 0; i < n; i++) {
            String[] coordinates = scanner.nextLine().split(" ");
            int x = Integer.parseInt(coordinates[0]);
            int y = Integer.parseInt(coordinates[1]);
            xCoordinates.add(x);
            yCoordinates.add(y);
        }

        int universes = 0;

        // Check for empty rows or columns
        if (xCoordinates.size() == 1 || yCoordinates.size() == 1) {
            universes = 1;
        } else {
            // Check for splits
            for (int x : xCoordinates) {
                if (!xCoordinates.contains(x + 1) && !xCoordinates.contains(x - 1)) {
                    universes++;
                }
            }
            for (int y : yCoordinates) {
                if (!yCoordinates.contains(y + 1) && !yCoordinates.contains(y - 1)) {
                    universes++;
                }
            }
        }

        System.out.println(universes);
    }
}
