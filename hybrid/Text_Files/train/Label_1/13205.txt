#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007
ll p;
bool f[1000010];
int main()
{
    string s;
    int a,b;
    cin>>s>>a>>b;
    for (int i=1e6; i>0; --i)
    {
        if (s.size()%i==0)
        {
            string aa=s.substr(0,s.size()/i),bb=s.substr(s.size()/i);
            long long x=0LL,y=0LL;
            for (int j=aa.size()-1; j>=0; --j)
            {
                x=(x*10LL+(aa[j]-'0'))%a;
            }
            for (int j=bb.size()-1; j>=0; --j)
            {
                y=(y*10LL+(bb[j]-'0'))%b;
            }
            if ((x==0)&&(y==0))
            {
                cout<<"YES"<<endl;
                cout<<aa<<endl<<bb<<endl;
                return 0;
            }
        }
    }
    cout<<"NO"<<endl;
    return 0;
}
