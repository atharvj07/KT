import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[][] e = new int[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
            int a = in.nextInt(), b = in.nextInt(), c = in.nextInt(), d = in.nextInt();
            e[a][b] = c;
            e[b][a] = d;
        }
        boolean[] vis = new boolean[n + 1];
        int[] f = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            if (vis[i]) continue;
            Queue<Integer> q = new LinkedList<>();
            q.offer(i);
            int sum = 0;
            while (!q.isEmpty()) {
                int u = q.poll();
                vis[u] = true;
                for (int j = 1; j <= n; j++) {
                    if (e[u][j] != 0) {
                        sum += e[u][j];
                        if (!vis[j]) q.offer(j);
                    }
                }
            }
            if (sum > 0) {
                f[i] = f[i - 1] + 1;
            } else {
                f[i] = f[i - 1];
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (f[i] > f[ans]) ans = i;
        }
        System.out.println(ans == 0 ? 0 : f[ans]);
    }
}
