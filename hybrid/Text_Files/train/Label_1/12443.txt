#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int MAXN = 1e5 + 10;

int n, m;
int a[MAXN], c[MAXN];
priority_queue<int, vector<int>, greater<int> > pq;

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> c[i];
        pq.push(c[i]);
    }
    while (m--) {
        int t, d;
        cin >> t >> d;
        t--;
        if (a[t] >= d) {
            a[t] -= d;
            cout << d * c[t] << endl;
        } else {
            int ans = 0;
            while (d) {
                int tmp = pq.top();
                pq.pop();
                if (a[t]) {
                    ans += c[t];
                    d--;
                    a[t]--;
                } else {
                    if (tmp != c[t]) {
                        ans += tmp;
                        a[t]++;
                        t = find(c, c + n, tmp) - c;
                    }
                }
            }
            cout << ans << endl;
        }
    }
    return 0;
}
