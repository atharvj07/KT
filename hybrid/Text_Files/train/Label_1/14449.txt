#include <iostream>

using namespace std;

int main() {
    long long k, l, r, t, x, y;
    cin >> k >> l >> r >> t >> x >> y;
    while (t--) {
        k -= x;
        k += y;
        if (k < l || k > r) {
            cout << "No" << endl;
            return 0;
        }
    }
    cout << "Yes" << endl;
    return 0;
}
