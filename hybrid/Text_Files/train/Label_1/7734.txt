#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k1, k2;
    cin >> n >> k1 >> k2;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) cin >> b[i];
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    int ans = 0;
    for (int i = 0; i < n; ++i) ans += (a[i] - b[i]) * (a[i] - b[i]);
    int sum = n * (a[0] - b[0]);
    for (int i = 0; i < n - 1; ++i) sum += (a[i] - b[0]) - (a[i + 1] - b[0]);
    ans = min(ans, sum);
    for (int i = 1; i < n; ++i) {
        sum += n * (a[i] - b[i]) - (a[i - 1] - b[i]) + (a[i] - b[0]) - (a[i + 1] - b[0]);
        ans = min(ans, sum);
    }
    sum = n * (a[n - 1] - b[n - 1]);
    for (int i = n - 2; i >= 0; --i) sum += (a[i] - b[n - 1]) - (a[i + 1] - b[n - 1]);
    ans = min(ans, sum);
    for (int i = n - 2; i >= 0; --i) {
        sum += n * (a[i] - b[i]) - (a[i - 1] - b[i]) + (a[i] - b[n - 1]) - (a[i + 1] - b[n - 1]);
        ans = min(ans, sum);
    }
    cout << ans << endl;
    return 0;
}
