n = int(input())

if n < 3:
    print(-1)
else:
    # Generate a counter-example
    counter_example = [i for i in range(1, n + 1)]
    # Swap the last two elements to create a simple counter-example
    counter_example[-1], counter_example[-2] = counter_example[-2], counter_example[-1]
    print(*counter_example)
