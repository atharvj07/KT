import java.io.*;
import java.util.*;

public class Main {

    static int n;
    static int k;
    static char[][] map;
    static boolean[][] visited;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        n = Integer.parseInt(tokenizer.nextToken());
        k = Integer.parseInt(tokenizer.nextToken());
        map = new char[2][n];
        visited = new boolean[2][n];
        tokenizer = new StringTokenizer(reader.readLine());
        map[0] = tokenizer.nextToken().toCharArray();
        tokenizer = new StringTokenizer(reader.readLine());
        map[1] = tokenizer.nextToken().toCharArray();
        if (dfs(0, 0)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    static boolean dfs(int wall, int height) {
        if (height == n) {
            return true;
        }
        if (visited[wall][height] || map[wall][height] == 'X') {
            return false;
        }
        visited[wall][height] = true;
        // Try up
        if (dfs(wall, height + 1)) {
            return true;
        }
        // Try down
        if (height > 0 && dfs(wall, height - 1)) {
            return true;
        }
        // Try jump
        if (dfs((wall + 1) % 2, height + k)) {
            return true;
        }
        return false;
    }
}
