#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t, n;
  string s;
  cin >> t;
  for (int i = 0; i < t; ++i) {
    cin >> s;
    int ans = 0;
    for (auto val : s) {
      if (val == 'L') ans++;
    }
    cout << (int)(ans + 1) << endl;
  }
  return 0;
}
