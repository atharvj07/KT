#include <bits/stdc++.h>

using namespace std;

const int MAXN = 200005;
vector<int> adj[MAXN];
int dp[MAXN][11];
int ans;
int n, k;

void dfs(int u, int p) {
  for (int v : adj[u]) {
    if (v == p) continue;
    for (int i = 1; i <= k; i++) {
      dp[v][i] = min(dp[u][i] + 1, dp[u][i - 1] + 2);
    }
    dfs(v, u);
  }
  for (int i = 1; i <= k; i++) {
    ans += dp[u][i];
  }
}

int main() {
  cin >> n >> k;
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs(0, 0);
  cout << ans << endl;
  return 0;
}
