#include <bits/stdc++.h>
using namespace std;
const int mxN = 5005;
int p[mxN], c[mxN], l[mxN], r[mxN];
vector<int> adj[mxN];
bool vis[mxN];
int ans[mxN];
void dfs(int u) {
    vis[u] = 1;
    l[u] = r[u] = p[u];
    for (auto &v : adj[u]) {
        if (!vis[v]) {
            dfs(v);
            l[u] = min(l[u], l[v]);
            r[u] = max(r[u], r[v]);
        }
    }
}
int main() {
    int n, m, d;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> p[i];
    for (int i = 1; i <= n; i++) cin >> c[i], adj[c[i]].push_back(i);
    cin >> d;
    for (int i = 1; i <= d; i++) cin >> l[i], vis[l[i]] = 1;
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) dfs(i);
    }
    for (int i = 1; i <= d; i++) {
        int u = l[i];
        if (vis[c[u]]) ans[i] = r[c[u]];
        else {
            vis[c[u]] = 1;
            dfs(c[u]);
            ans[i] = r[c[u]];
        }
        c[u] = 0;
    }
    for (int i = 1; i <= d; i++) cout << ans[i] << '\n';
    return 0;
}
