import sys
n,k = map(int,sys.stdin.readline().split())
a = [list(map(int,sys.stdin.readline().split())) for _ in range(n)]
a.sort(key = lambda x : x[1])
l = []
for x in a :
    if x[0]<= x[1]:
        l.append((x[0]-1,x[1]))
    else:
        l.append((x[0]-1,1e9))
        l.append((1,x[1]))
l.sort()
ans = cnt = 0
for x in l:
    ans+=cnt*(x[0]-ans%1e9)
    ans%=1e9
    cnt+=1
    ans+=cnt*(x[1]-ans%1e9)
    ans%=1e9
print(ans%998244353)
