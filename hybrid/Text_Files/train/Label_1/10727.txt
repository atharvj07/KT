#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  long long ans = 0, now = 0, pre = 0;
  for (int i = 0; i < n; i++) {
    ans = max(ans, now + m - a[i]);
    pre = now;
    now = m - a[i];
  }
  ans = max(ans, pre);
  cout << ans << endl;
  return 0;
}
