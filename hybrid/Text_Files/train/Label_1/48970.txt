import java.util.Scanner;

public class UniqueSubstring {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        // Build the string by alternating 0s and 1s, starting with the character that makes the length of the minimal unique substring equal to k.
        char startChar = (n % k == 0) ? '0' : '1';
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            sb.append(startChar);
            startChar = (startChar == '0') ? '1' : '0';
        }

        // Adjust the string if necessary to ensure the minimal unique substring has length k.
        if (n % k != 0) {
            int extraChars = n % k;
            for (int i = 0; i < extraChars; i++) {
                sb.setCharAt(n - i - 1, sb.charAt(n - i - 1) == '0' ? '1' : '0');
            }
        }

        System.out.println(sb.toString());
    }
}
