#include <bits/stdc++.h>
using namespace std;
const long long mod=998244353;
int main(){
  int t;cin>>t;
  while(t--){
    int n;cin>>n;
    vector<long long>a(n,0);
    for(int i=0;i<n;i++){cin>>a[i];}
    long long ans=1,sum=0;
    for(int i=0;i<n;i++){
      sum+=a[i];
      if(sum>=n-1){
        ans=ans*(sum-n+2)%mod;
        sum=sum-n+a[i];
      }
    }
    if(sum!=0){ans=0;}
    cout<<ans<<endl;
  }
  return 0;
}
