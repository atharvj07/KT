#include<bits/stdc++.h>
typedef long long ll;
const ll maxn=1e5+5;
const ll mod=1e9+7;
using namespace std;

struct Node
{
    int next[26];
    int len;
    int fail;
    int num;
    Node(){memset(next,0,sizeof next),len=0,fail=0,num=0;}
}trie[maxn];

int cnt=0;

void insert(string &s,int n)
{
    int p=0;
    for(int i=0;i<s.size();++i)
    {
        int c=s[i]-'a';
        if(!trie[p].next[c]) trie[cnt]=Node(),trie[p].next[c]=cnt++;
        p=trie[p].next[c];
    }
    trie[p].num=n;
}

void Get_fail()
{
        queue<int> q;
        trie[0].fail=-1;
        for(int i=0;i<26;++i) if(trie[0].next[i]) q.push(trie[0].next[i]);
        while(!q.empty())
        {
            int p=q.front();q.pop();
            for(int i=0;i<26;++i)
            {
                if(trie[p].next[i])
                {
                    trie[trie[p].next[i]].fail=trie[p].fail?trie[trie[p].fail].next[i]:0;
                    q.push(trie[p].next[i]);
                }
                else trie[p].next[i]=trie[trie[p].fail].next[i];
            }
        }
}

int dfs(int p,int pos,string s)
{
    while(p&&trie[p].next[s[pos]-'a']==0) p=trie[p].fail;
    if(trie[p].next[s[pos]-'a']) p=trie[p].next[s[pos]-'a'];
    if(trie[p].num) return max(trie[p].num,trie[trie[p].fail].num);
    return 0;
}

map<string,int> ans;
string A[maxn],B[maxn];

int main()
{
    int n;
    cin>>n;
    for(int i=1;i<=n;++i) cin>>A[i];
    for(int i=1;i<=n;++i) cin>>B[i];
    for(int i=1;i<=n;++i) insert(A[i],i);
    Get_fail();
    int mx=0;
    for(int i=1;i<=n;++i) mx=max(mx,dfs(0,0,B[i]));
    cout<<mx<<endl;
    for(int i=1;i<=n;++i)
    {
        int p=0;
        for(int j=0;j<B[i].size();++j)
        {
            while(p&&trie[p].next[B[i][j]-'a']==0) p=trie[p].fail;
            if(trie[p].next[B[i][j]-'a']) p=trie[p].next[B[i][j]-'a'];
            if(trie[p].num==dfs(0,0,B[i])) {ans[A[i]]=trie[p].num;break;}
        }
    }
    for(int i=1;i<=n;++i) cout<<i<<' '<<ans[A[i]]<<endl;
    return 0;
}
