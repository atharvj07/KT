import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int H, W;
        while ((H = sc.nextInt()) != 0 && (W = sc.nextInt()) != 0) {
            for (int i = 0; i < H; i++) {
                for (int j = 0; j < W; j++) {
                    System.out.print(i == 0 || i == H - 1 || j == 0 || j == W - 1 ? "#" : ".");
                }
                System.out.println();
            }
            System.out.println();
        }
    }
}
