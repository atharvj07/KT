#include <iostream>
#include <vector>
#include <map>
using namespace std;
int main() {
    int n; cin >> n;
    vector<int> u(n), pref(n+1), suf(n+1, 2e9);
    map<int, int> frq;
    for (int i = 0; i < n; i++) { cin >> u[i]; pref[i+1] = pref[i] + (frq[u[i]]++ == 0); }
    for (int i = n-1; i >= 0; i--) { suf[i] = suf[i+1] - (frq[u[i]]-- == 1); }
    int mx = 0;
    for (int j = 0; j < n; j++) {
        for (int i = j + 1; i < n; i++) {
            if (suf[i+1] - suf[i] <= 1 && pref[i] - pref[j] <= 1) mx = max(mx, i-j+1);
        }
    }
    cout << mx << endl;
}
