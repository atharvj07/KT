#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  long long cur = 0;
  int ans = 1;
  for (int i = 0; i < n; i++) {
    cur += a[i];
    if (cur > 0) ans = i + 1;
    cur--;
  }
  cout << ans << endl;
  return 0;
}
