import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        int h = in.nextInt();
        int m = in.nextInt();
        double hourAngle = Math.PI * (h / 12.0 + m / 720.0);
        double minuteAngle = Math.PI * m / 30.0;
        double diffAngle = Math.abs(hourAngle - minuteAngle);
        double distance = Math.sqrt(a * a + b * b - 2 * a * b * Math.cos(diffAngle));
        System.out.println(distance);
    }
}
