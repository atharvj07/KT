#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i]; }
    int ans = 0;
    int last = -1;
    for (int i = 0; i < n; i++) {
        if (v[i] == 1) {
            ans++;
            last = i;
        } else if (last != -1) {
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
