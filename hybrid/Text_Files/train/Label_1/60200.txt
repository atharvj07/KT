from collections import deque
n = int(input())
dq = deque([1])
for i in range(n-1):
  a = dq.pop()
  b = dq.pop()
  while a==b:
    dq.append(a+1)
    a=dq.pop()
    b=dq.pop()
  dq.append(b)
  dq.append(a)
while len(dq)>=2:
  a = dq.pop()
  b = dq.pop()
  while a==b:
    dq.append(a+1)
    a=dq.pop()
    b=dq.pop()
  dq.append(b)
  dq.append(a)
if len(dq)==0: print(0)
else: print(*dq)
