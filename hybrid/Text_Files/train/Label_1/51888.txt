import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String home = in.nextLine(), away = in.nextLine(), p;
        Map<String, Map<Integer, Integer>> m = new HashMap<>();
        for (int n = in.nextInt(); n-- > 0; ) {
            int t = in.nextInt();
            p = in.next();
            int i = in.nextInt();
            if (p.equals("h")) {
                m.putIfAbsent(home, new HashMap<>());
                if (!m.get(home).containsKey(i) || m.get(home).get(i) >= t - 1) {
                    m.get(home).put(i, t);
                }
            } else {
                m.putIfAbsent(away, new HashMap<>());
                if (!m.get(away).containsKey(i) || m.get(away).get(i) >= t - 1) {
                    m.get(away).put(i, t);
                }
            }
            in.next();
        }
        m.forEach((t, x) -> x.forEach((k, v) -> System.out.printf("%s %d %d%n", t, k, v)));
    }
}
