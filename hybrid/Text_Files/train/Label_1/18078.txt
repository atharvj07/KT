#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n;
        cin >> n;
        vector<long long> v;
        for (long long i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                v.emplace_back(i);
                while (n % i == 0)
                    n /= i;
            }
        }
        if (n > 1)
            v.emplace_back(n);
        cout << v.size() << '\n';
        for (auto x : v)
            cout << x << ' ';
        cout << '\n';
    }
    return 0;
}
