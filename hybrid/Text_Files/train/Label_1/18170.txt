#include <iostream>
#include <string>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int num = 0, cur = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '1') {
      cur++;
    } else {
      num += cur;
      cur = 0;
    }
  }
  num += cur;
  cout << num << '\n';
  return 0;
}
