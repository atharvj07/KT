import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] s = new int[m], c = new int[m];
        for (int i = 0; i < m; ++i) {
            s[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        int num = 0, rev = 0, flag = 0;
        for (int i = 0; i < n; ++i) {
            num *= 10;
            int now = flag;
            for (int j = 0; j < m; ++j) {
                if (s[j] == i + 1) {
                    num += c[j];
                    flag = 1;
                    if (now == 0) rev = rev * 10 + c[j];
                }
            }
            if (flag == 1) {
                rev *= 10;
            }
        }
        if (rev == num) System.out.println(rev);
        else System.out.println(-1);
    }
}
