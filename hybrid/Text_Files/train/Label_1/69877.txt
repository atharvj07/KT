n = int(input())
s = input()
l = [0] * n
for i in range(n - 1):
    if s[i] == '>' and s[i + 1] == '<':
        l[i + 1] = l[i] + 1
    else:
        l[i + 1] = l[i]
last = n
for i in range(n - 1, -1, -1):
    if s[i] == '<' and s[i - 1] == '>':
        last = i - 1
        l[i - 1] = max(l[i - 1], last)
    else:
        l[i - 1] = max(l[i - 1], last)
print(len([i for i in l if i == 0]))
