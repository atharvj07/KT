#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  int mx = *max_element(a, a + n);
  int ans = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] > mx - a[0]) {
      ans += a[i] - mx + a[0];
      a[i] = mx - a[0];
    }
  }
  cout << ans;
  return 0;
}
