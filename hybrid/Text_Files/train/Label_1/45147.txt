import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        double ans = 0.0;
        if (n % 2 == 0 && n > 1) {
            ans = (n / 2.0)/n;
        } else {
            ans = (n/2.0+1.0)/n;
        }
        System.out.println(ans);
    }
}
