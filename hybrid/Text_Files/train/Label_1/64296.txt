n, m = map(int, input().split())
s = [list(input()) for _ in range(n)]

dr = [1,-1,0,0]
dc = [0,0,1,-1]
def valid(r, c):
    if 0<=r<n and 0<=c<m:
        return True
    return False

ans = "NO"
for r in range(1,n-1):
    for c in range(1,m-1):
        ok = True
        for d in range(4):
            nr = r+dr[d]
            nc = c+dc[d]
            if not valid(nr,nc): ok=False
            if s[nr][nc]=="#": ok=False
        if ok: ans = "YES"
print(ans)
