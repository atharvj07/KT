import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

class Main {
    public static void main(String[] args) throws IOException {

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        st = new StringTokenizer(br.readLine());
        int s = Integer.parseInt(st.nextToken()) - 1;
        int t = Integer.parseInt(st.nextToken()) - 1;

        int[] p = new int[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            p[i] = Integer.parseInt(st.nextToken());
        }

        List<List<Edge>> adjList = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            adjList.add(new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            int w = Integer.parseInt(st.nextToken());
            adjList.get(u).add(new Edge(v, w));
            adjList.get(v).add(new Edge(u, w));
        }

        int[] distS = dijkstra(adjList, s);
        int[] distT = dijkstra(adjList, t);

        int scoreTavas = 0;
        int scoreNafas = 0;

        boolean[] used = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (distS[i] == Integer.MAX_VALUE || distT[i] == Integer.MAX_VALUE) continue;
            int minDist = Math.min(distS[i], distT[i]);
            for (Edge edge : adjList.get(i)) {
                int v = edge.to;
                if (distS[v] == Integer.MAX_VALUE || distT[v] == Integer.MAX_VALUE) continue;
                int newDist = Math.min(minDist, edge.weight + Math.min(distS[v], distT[v]));
                if (distS[i] == newDist || distT[i] == newDist) {
                    if (!used[v]) {
                        used[v] = true;
                        if (p[v] > 0) {
                            scoreTavas += p[v];
                        } else {
                            scoreNafas += p[v];
                        }
                    }
                }
            }
        }

        if (scoreTavas > scoreNafas) {
            System.out.println("Break a heart");
        } else if (scoreNafas > scoreTavas) {
            System.out.println("Cry");
        } else {
            System.out.println("Flowers");
        }
    }

    static int[] dijkstra(List<List<Edge>> adjList, int s) {
        int[] dist = new int[adjList.size()];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[s] = 0;
        boolean[] visited = new boolean[adjList.size()];

        for (int i = 0; i < adjList.size(); i++) {
            int u = -1;
            for (int j = 0; j < adjList.size(); j++) {
                if (!visited[j] && (u == -1 || dist[j] < dist[u])) {
                    u = j;
                }
            }

            if (u == -1) break;
            visited[u] = true;

            for (Edge edge : adjList.get(u)) {
                int v = edge.to;
                int weight = edge.weight;
                dist[v] = Math.min(dist[v], dist[u] + weight);
            }
        }

        return dist;
    }

    static class Edge {
        int to;
        int weight;

        public Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }
}
