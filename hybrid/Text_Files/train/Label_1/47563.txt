import java.util.*;

public class StringPermutation {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        sc.nextLine(); // Consume newline

        String[] strings = new String[N];
        for (int i = 0; i < N; i++) {
            strings[i] = sc.nextLine();
        }

        int Q = sc.nextInt();
        sc.nextLine(); // Consume newline

        for (int i = 0; i < Q; i++) {
            int k = sc.nextInt();
            String permutation = sc.nextLine().trim();

            // Create a map from characters to their indices in the permutation
            Map<Character, Integer> charIndexMap = new HashMap<>();
            for (int j = 0; j < 26; j++) {
                charIndexMap.put(permutation.charAt(j), j);
            }

            // Sort the strings lexicographically based on the permutation
            Arrays.sort(strings, (s1, s2) -> {
                for (int j = 0; j < Math.min(s1.length(), s2.length()); j++) {
                    int index1 = charIndexMap.get(s1.charAt(j));
                    int index2 = charIndexMap.get(s2.charAt(j));
                    if (index1 != index2) {
                        return index1 - index2;
                    }
                }
                return s1.length() - s2.length();
            });

            // Find the index of the k-th string in the sorted array
            int index = Arrays.binarySearch(strings, strings[k - 1]);

            // Output the sequence of the string
            System.out.println(index + 1);
        }
    }
}
