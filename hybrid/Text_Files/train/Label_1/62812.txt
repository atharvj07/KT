r,c=map(int,input().split())
a=[['0' for j in range(c)]for i in range(r)]
b=[list(map(int,input()))for i in range(r)]
for i in range(r):
 for j in range(c):
  a[i][j]=str(b[i][j])
ans=0
for i in range(r):
 for j in range(c):
  if(i%2==0 and j%2==0):ans+=int(a[i][j])
  if(i%2==1 and j%2==1):ans+=int(a[i][j])
print(ans)
