#include<bits/stdc++.h>
using namespace std;
vector<pair<int,int>> g[5005];
int dis[5005],vis[5005],par[5005],n,m,T;
void dij()
{
    priority_queue<pair<int,int>> pq;
    pq.push({0,1});
    dis[1]=0;
    while(!pq.empty()){
        int u=pq.top().second,d=-pq.top().first;
        pq.pop();
        if(vis[u])continue;
        vis[u]=1;
        for(auto x:g[u])
        {
            int v=x.first,w=x.second;
            if(dis[u]+w<dis[v]){
                dis[v]=dis[u]+w;
                par[v]=u;
                pq.push({-dis[v],v});
            }
        }
    }
}
int main()
{
    cin>>n>>m>>T;
    for(int i=0;i<m;i++)
    {
        int u,v,w;
        cin>>u>>v>>w;
        g[u].push_back({v,w});
    }
    memset(dis,0x3f,sizeof(dis));
    dij();
    if(dis[n]>T){ cout<<"IMPOSSIBLE\n"; return 0; }
    vector<int> v;
    int u=n;
    while(u!=1){ v.push_back(u); u=par[u]; }
    v.push_back(1);
    reverse(v.begin(),v.end());
    cout<<v.size()<<'\n';
    for(auto x:v) cout<<x<<" ";
}
