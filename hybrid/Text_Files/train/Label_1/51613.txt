import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while(t-- > 0) {
            String s = sc.next();
            int n = s.length();
            char[] c = s.toCharArray();
            Arrays.sort(c);
            int i = n/2 - 1;
            while(i >= 0 && c[i] == c[i+1]) {
                i--;
            }
            if(i < 0) {
                System.out.println(s.substring(0, n-1));
                continue;
            }
            c[i+1] = c[i];
            i--;
            int j = n-1;
            while(i >= 0) {
                c[j] = c[i];
                i--;
                j--;
            }
            System.out.println(new String(c));
        }
    }
}
