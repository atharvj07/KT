#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  string s;
  cin >> s;
  long long ans = 0, last_v = -1, cnt = 0;
  for (long long i = 0; i < s.size(); i++) {
    if (s[i] == 'v') {
      cnt++;
      if (last_v != -1) ans += last_v * (cnt - last_v);
      last_v = cnt;
    } else
      last_v = -1;
  }
  cout << ans << endl;
}
