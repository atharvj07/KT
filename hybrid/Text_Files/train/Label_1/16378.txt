#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = i; j < n; j++) {
      cnt += (a[j] == 1);
      if (j - i + 1 - cnt < cnt) { break; }
      ans = max(ans, cnt);
    }
  }
  cout << ans << endl;
  return 0;
}
