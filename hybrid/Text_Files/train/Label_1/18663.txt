#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef long long ll;
const ll mod = 100000007;
vector<ll> a;

ll dp[2100000];

ll fastpow(ll x, ll pow) {
    if (pow == 0) return 1;
    if (pow == 1) return x % mod;
    ll val = fastpow(x, pow / 2);
    val = (val * val) % mod;
    if (pow % 2 == 1) {
        val = (val * x) % mod;
    }
    return val % mod;
}

int main() {
    int n, m, k;
    cin >> n >> m >> k;
    a.resize(n + 1);
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    sort(a.begin(), a.end());
    ll ans = 0;
    if (k == 1) {
        dp[1] = a[n];
        for (int i = 2; i <= n; i++)
            dp[i] = (dp[i - 1] + a[n] - a[n - i + 1]) % mod;
        if (m <= n) {
            for (int i = 1; i <= m; i++) {
                ans = (ans + dp[i]) % mod;
            }
        } else {
            ll times = m / n;
            times %= mod;
            ans = (ans + (times * dp[n]) % mod) % mod;
            m = m % n;
            for (int i = 1; i <= m; i++) {
                ans = (ans + dp[i]) % mod;
            }
        }
    } else {
        dp[1] = a[n];
        dp[2] = (a[n] + mod - a[n - 1]) % mod;
        for (int i = 3; i <= k; i++) {
            dp[i] = (mod - dp[i - 1] + dp[i - 2] + a[n] - a[n - i + 1]) % mod;
        }
        for (int i = k + 1; i <= n; i++) {
            dp[i] = (dp[i - 1] + mod - dp[i - k] + a[n] - a[n - i + 1]) % mod;
        }
        ll times = m / k;
        times %= mod;
        ans = (ans + (times * dp[k]) % mod) % mod;
        m = m % k;
        for (int i = 1; i <= m; i++) {
            ans = (ans + dp[i]) % mod;
        }
    }
    cout << ans << "\n";
    return 0;
}
