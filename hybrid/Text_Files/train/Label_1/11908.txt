#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 2000;
int n;
int parent[MAXN + 1], size[MAXN + 1];

struct Point {
    int x, y, z;
};

Point points[MAXN + 1];

bool cmpX(const Point &a, const Point &b) {
    return a.x < b.x;
}

bool cmpY(const Point &a, const Point &b) {
    return a.y < b.y;
}

bool cmpZ(const Point &a, const Point &b) {
    return a.z < b.z;
}

// DSU
int find(int a) {
    if (parent[a] == a) return a;
    return parent[a] = find(parent[a]);
}

void merge(int a, int b) {
    a = find(a);
    b = find(b);
    if (a != b) {
        if (size[a] < size[b]) swap(a, b);
        parent[b] = a;
        size[a] += size[b];
    }
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> points[i].x >> points[i].y >> points[i].z;
        parent[i] = i;
        size[i] = 1;
    }
    sort(points + 1, points + n + 1, cmpX);
    for (int i = 1; i < n; i++) {
        if (points[i].x == points[i + 1].x &&
            points[i].y == points[i + 1].y &&
            points[i].z == points[i + 1].z) {
            cout << -1 << endl;
            return 0;
        }
    }
    sort(points + 1, points + n + 1, cmpY);
    for (int i = 1; i < n; i++) {
        if (points[i].x == points[i + 1].x &&
            points[i].y == points[i + 1].y &&
            points[i].z == points[i + 1].z) {
            cout << -1 << endl;
            return 0;
        }
    }
    sort(points + 1, points + n + 1, cmpZ);
    for (int i = 1; i < n; i++) {
        if (points[i].x == points[i + 1].x &&
            points[i].y == points[i + 1].y &&
            points[i].z == points[i + 1].z) {
            cout << -1 << endl;
            return 0;
        }
    }
    for (int i = 1; i < n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (points[i].x <= points[j].x &&
                points[i].y <= points[j].y &&
                points[i].z <= points[j].z) {
                merge(i, j);
            }
        }
    }
    vector<pair<int, int>> pairs;
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (find(i) != find(j)) {
                pairs.push_back({i, j});
                merge(i, j);
            }
        }
    }
    for (auto &pair : pairs) {
        cout << pair.first << ' ' << pair.second << endl;
    }
    return 0;
}
