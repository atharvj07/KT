#include <bits/stdc++.h>
using namespace std;
int main(){
    int a,b,c,l; cin>>a>>b>>c>>l;
    int ans=0;
    for(int i=0;i<=l;i++){
        for(int j=0;j<=l;j++){
            if(i+j<=l && a+i>0 && b+j>0 && c>0){
                int p=a+i,q=b+j,r=c;
                if(p+q>r && q+r>p && r+p>q){
                    ans++;
                }
            }
        }
    }
    cout<<ans<<endl;
}
