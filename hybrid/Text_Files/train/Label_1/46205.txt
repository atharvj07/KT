import java.util.*;
class Main {
	public static void main( String[] args ) {
		Scanner sc = new Scanner( System.in );
		int t = sc.nextInt();
		while( ( t-- ) > 0 ) {
			int n = sc.nextInt();
			int m = sc.nextInt();
			boolean[][] grid = new boolean[n][m];
			for( int i = 0; i < n; ++i ) {
				for( int j = 0; j < m; ++j ) {
					int num = sc.nextInt();
					grid[i][j] = num == 1;
				}
			}
			boolean ashishTurn = true;
			boolean ashishWins = false;
			while( ashishTurn ) {
				boolean moved = false;
				for( int i = 0; i < n; ++i ) {
					for( int j = 0; j < m; ++j ) {
						if( grid[i][j] ) continue;
						boolean ok = true;
						for( int k = 0; k < n; ++k ) {
							if( grid[k][j] ) { ok = false; break; }
						}
						for( int k = 0; k < m; ++k ) {
							if( grid[i][k] ) { ok = false; break; }
						}
						if( ok ) {
							grid[i][j] = true;
							moved = true;
							ashishTurn = false;
							break;
						}
					}
					if( moved ) break;
				}
				if( !moved ) {
					ashishWins = false;
					ashishTurn = false;
				}
			}
			while( !ashishTurn ) {
				boolean moved = false;
				for( int i = 0; i < n; ++i ) {
					for( int j = 0; j < m; ++j ) {
						if( grid[i][j] ) continue;
						boolean ok = true;
						for( int k = 0; k < n; ++k ) {
							if( grid[k][j] ) { ok = false; break; }
						}
						for( int k = 0; k < m; ++k ) {
							if( grid[i][k] ) { ok = false; break; }
						}
						if( ok ) {
							grid[i][j] = true;
							moved = true;
							ashishTurn = true;
							break;
						}
					}
					if( moved ) break;
				}
				if( !moved ) {
					ashishWins = true;
					ashishTurn = true;
				}
			}
			System.out.println( ashishWins ? "Ashish" : "Vivek" );
		}
	}
}
