#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 1;
int n;
string s;

int get(char c, char d) {
  int pos = 0;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == d) pos = i;
    if (s[i] == c) {
      if (pos != 0 && (i - pos) % 3 != 0) cnt++;
    }
  }
  return cnt;
}

int main() {
  cin >> n >> s;
  int ans = 1e9;
  string t;
  for (int i = 97; i <= 99; i++) {
    for (int j = 97; j <= 99; j++) {
      int cnt = 0;
      for (int k = 0; k < n; k++) {
        if (s[k] == char(i)) cnt += (k % 3 != j % 3);
      }
      cnt += get(char(i), char(j));
      if (cnt < ans) {
        ans = cnt;
        t = s;
        for (int k = 0; k < n; k++) {
          if (t[k] == char(i)) t[k] = char(j);
        }
      }
    }
  }
  cout << ans << '\n';
  cout << t << '\n';
}
