import java.util.*;
import static java.lang.Math.*;
import static java.util.Arrays.*;
import static java.util.stream.Collectors.*;

class Main {
    public static void main(String[] a) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), S = sc.nextInt(), l = 2 * n, ans = 0;
        Parcel[] arr = new Parcel[l];
        for (int i = 0; i < n; i++) {
            int x1 = sc.nextInt(), x2 = sc.nextInt(), w = sc.nextInt(), s = sc.nextInt(), v = sc.nextInt();
            arr[x1] = new Parcel(w, s, v, true);
            arr[x2] = new Parcel(w, s, v, false);
        }
        Parcel plat = new Parcel(0, S);
        for (int i = 0; i < l; i++) {
            if (arr[i] == null) continue;
            if (arr[i].in) {
                while (plat.w + arr[i].w > plat.s) {
                    if (plat.b) {
                        ans += plat.v;
                        plat.w = 0;
                        plat.b = false;
                        continue;
                    }
                    break;
                }
                plat.w += arr[i].w;
                if (plat.s < arr[i].s) plat.b = true;
            } else if (plat.w == arr[i].w && plat.b) {
                ans += arr[i].v;
                plat.w = 0;
                plat.b = false;
                plat.s = S;
            }
        }
        System.out.println(ans);
    }
    static class Parcel {
        int w, s, v;
        boolean in, b;
        Parcel(int w, int s) {
            this.w = w;
            this.s = s;
        }
        Parcel(int w, int s, int v, boolean in) {
            this.w = w;
            this.s = s;
            this.v = v;
            this.in = in;
        }
    }
}
