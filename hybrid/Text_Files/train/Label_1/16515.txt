#include <iostream>
#include <vector>
using namespace std;
int main() {
    long long int n, k, sum = 0, prod = 1, ans = 0;
    cin >> n >> k;
    vector<long long int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        prod *= a[i];
    }
    for (int l = 0, r = 0; r < n; r++) {
        sum += a[r];
        while (l <= r && sum * k < prod) {
            sum -= a[l];
            l++;
        }
        if (sum * k == prod) ans++;
    }
    cout << ans << "\n";
}
