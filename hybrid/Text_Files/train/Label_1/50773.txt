import java.util.*;
import java.io.*;

public class Main {
    static long x;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        x = sc.nextLong();
        long i = 2, ans1 = 1, ans2 = x;
        while (i * i <= x) {
            if (x % i == 0) {
                if (x / i <= i) {
                    ans1 = i;
                    ans2 = x / i;
                } else {
                    ans1 = x / i;
                    ans2 = i;
                }
            }
            i++;
        }
        System.out.println(ans1 + " " + ans2);
    }
}
