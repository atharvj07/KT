#include <vector>
#include <cmath>
#include <iostream>
#include <algorithm>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  string s, t;
  cin >> s >> t;
  vector<int> ans;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] != t[i]) {
      cnt++;
      ans.push_back(i + 1);
    }
  }
  cout << cnt << endl;
  for (int i = 0; i < ans.size(); i++)
    cout << ans[i] << " ";
  cout << endl;
  return 0;
}
