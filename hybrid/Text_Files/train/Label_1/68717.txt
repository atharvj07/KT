import sys
input = sys.stdin.readline

n = int(input())
d = {}
for i in range(n):
    a, b = map(int, input().split())
    d[(a, b)] = None

ws = list(map(int, input().split()))

def f(a, b):
    return a * n + b

def g(x):
    return b[x] * n + a[x]

ans = [None] * n
idx = 0
widx = 0
for a, b in sorted(d.keys()):
    while widx < n and b != ws[widx]: widx += 1
    if widx == n: break
    ans[idx] = (a, b)
    a += 1
    while (a, b) in d:
        ans[idx] = (a, b)
        a += 1
        b += 1
    idx += 1

if idx == n:
    print("YES")
    for a, b in ans:
        print(f(a, b))
else:
    print("NO")
