#include<bits/stdc++.h>
using namespace std;

#define i64 long long
#define MOD 1000000007

int main() {
    i64 n, m; cin >> n >> m;
    vector<i64> a(n + 1);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.rbegin(), a.rend());
    i64 ans = 0;
    for (int i = 1; i < n; i++) {
        ans = (ans + (a[i - 1] - a[i]) % m * (i % m)) % m;
    }
    cout << ans << "\n";
    return 0;
}
