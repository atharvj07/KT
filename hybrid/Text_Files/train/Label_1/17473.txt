#include<bits/stdc++.h>

using namespace std;
#define ll long long

int main()
{
	ll n,b;
	cin>>n>>b;
	ll a[n], ev=0, od=0, sum=0;
	
	for(ll i=0; i<n; i++)
	{
		cin>>a[i];
		if(a[i]%2==0) ev++;
		else od++;
	}
	
	if(ev!=od) { cout<<0; return 0; }
	
	sort(a,a+n);
	for(ll i=1; i<n && b>=abs(a[i]-a[i-1]); i++)
	{
		b-=abs(a[i]-a[i-1]);
		sum++;
	}
	
	cout<<sum<<endl;
}
