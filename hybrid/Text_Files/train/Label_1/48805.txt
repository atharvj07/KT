import java.util.*;

class Main {

    static int n;
    static int[] permutation;
    static ArrayList<Integer>[] graph;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        n = input.nextInt();
        permutation = new int[n + 1];
        graph = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
        for (int i = 1; i <= n; i++) permutation[i] = input.nextInt();

        bubbleSortGraph();
        System.out.println(maxIndependentSetSize());
    }

    public static void bubbleSortGraph() {
        boolean swapped = true;
        while (swapped) {
            swapped = false;
            for (int i = 1; i < n; i++) {
                if (permutation[i] > permutation[i + 1]) {
                    graph[permutation[i]].add(permutation[i + 1]);
                    graph[permutation[i + 1]].add(permutation[i]);
                    int temp = permutation[i];
                    permutation[i] = permutation[i + 1];
                    permutation[i + 1] = temp;
                    swapped = true;
                }
            }
        }
    }

    public static int maxIndependentSetSize() {
        boolean[] visited = new boolean[n + 1];
        int maxSize = 0;
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                int size = dfs(i, visited);
                maxSize = Math.max(maxSize, size);
            }
        }
        return maxSize;
    }

    public static int dfs(int node, boolean[] visited) {
        if (visited[node]) return 0;
        visited[node] = true;
        int size = 1;
        for (int neighbor : graph[node]) {
            size += dfs(neighbor, visited);
        }
        return size;
    }
}
