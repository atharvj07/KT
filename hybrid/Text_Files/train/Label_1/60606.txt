import sys

n, h = map(int, input().split())
a = list(map(int, input().split()))

# dp[i][j] = number of ways to make all numbers in the sequence equal to h,
# using only the first i elements and a total of j operations
dp = [[0 for j in range(n + 1)] for i in range(n + 1)]
dp[0][0] = 1

for i in range(1, n + 1):
    for j in range(n + 1):
        dp[i][j] = dp[i - 1][j]
        if j > 0 and a[i - 1] < h:
            dp[i][j] += dp[i - 1][j - 1]

print(dp[n][n] % 1000000007)
