n,m=map(int,input().split())
t=input().split()
s=input().split()
i=j=ans=0
while i<n and j<m:
    if t[i][1]==s[j][1]:
        if i==n-1 and j==m-1:
            ans+=1
            break
        elif i==n-1:
            j+=1
            continue
        elif j==m-1:
            i+=1
            j=0
        else:
            i+=1
            j+=1
    else:
        j=0
        i+=1
print(ans)
