import sys

def get_sector(x, y, n, m):
    if x == 1:
        return y
    else:
        return n + y

def get_x(sector, n, m):
    if sector <= n:
        return 1
    else:
        return 2

def get_y(sector, n, m):
    if sector <= n:
        return sector
    else:
        return sector - n

def can_move(s_x, s_y, e_x, e_y, n, m):
    s_sector = get_sector(s_x, s_y, n, m)
    e_sector = get_sector(e_x, e_y, n, m)
    if s_sector == e_sector:
        return True
    if s_x != e_x:
        return False
    if s_sector % n == 0:
        return True
    if e_sector % n == 1:
        return True
    return False

n, m, q = map(int, input().split())
for _ in range(q):
    s_x, s_y, e_x, e_y = map(int, input().split())
    print("YES" if can_move(s_x, s_y, e_x, e_y, n, m) else "NO")
