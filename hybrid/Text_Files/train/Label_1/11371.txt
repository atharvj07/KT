#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll ans;
unordered_map<ll,ll>mp;
unordered_map<ll,vector<ll>> adj;
void dfs(ll u,ll p)
{
    for(auto it:adj[u])
    {
        if(it!=p)
        {
            if(mp[u]==0)
               mp[it]=1;
            else
               mp[it]=0;
            dfs(it,u);
        }
    }
}
int main()
{
    ll n;
    cin >> n;
    ll a, b;
    while (n > 1)
    {
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
        n--;
    }
    dfs(1,1);
    for(auto it:mp)
    {
      for(auto it2:adj[it.first])
      {
          if(it2>it.first&&mp[it2]!=it.second)
             ans++;
      }
    }
    cout<< ans/2 <<endl;
}
