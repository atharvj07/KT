import java.util.Scanner;

public class PiApproximation {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            double allowedError = scanner.nextDouble();
            if (allowedError == 0.0) {
                break;
            }

            Fraction fraction = approximatePi(allowedError);
            System.out.println(fraction.numerator + "/" + fraction.denominator);
        }

        scanner.close();
    }

    private static Fraction approximatePi(double allowedError) {
        double lowerBound = Math.PI - allowedError;
        double upperBound = Math.PI + allowedError;

        int numerator = 0;
        int denominator = 1;

        while (true) {
            double fractionValue = (double) numerator / denominator;
            if (fractionValue >= lowerBound && fractionValue <= upperBound) {
                return new Fraction(numerator, denominator);
            }

            if (fractionValue < lowerBound) {
                numerator++;
            } else {
                denominator++;
            }
        }
    }

    private static class Fraction {
        int numerator;
        int denominator;

        public Fraction(int numerator, int denominator) {
            this.numerator = numerator;
            this.denominator = denominator;
        }
    }
}
