n = int(input())
a = list(map(int, input().split()))
a.sort(reverse=True)
o, i = 0, 0
for k in range(n):
    if k % 2 == 0:
        if i == n - k - 1: o += a[i]
        else: o += max(a[i], a[n - k - 1])
    else:
        if i == n - k - 1: i += 1
        else: i = n - k - 1
        o += max(a[i], a[n - k - 1])
print(*a[i + (k % 2):])
