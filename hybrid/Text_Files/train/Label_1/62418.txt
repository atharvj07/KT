from collections import deque
from sys import stdin
read = stdin.readline

def BFS(v:int, w:int, d:str, vis:list[list[list[bool]]], grid:list[list[str]], cnt:int, c:int):
    ok = False
    q = deque()
    q.append((v,w,d))
    vis[v][w][d] = True
    while q:
        nv, nw, nd = q.popleft()
        if grid[nv][nw] == 'A': continue
        grid[nv][nw] = 'A'
        cnt += 1
        if cnt == c:
            ok = True
            break
        if d == 'E':
            if nw+1 < w and not vis[nv][nw+1]['E']: q.append((nv,nw+1,'E'))
            if nw-1 >= 0 and not vis[nv][nw-1]['W']: q.append((nv,nw-1,'W'))
        elif d == 'W':
            if nw-1 >= 0 and not vis[nv][nw-1]['W']: q.append((nv,nw-1,'W'))
            if nw+1 < w and not vis[nv][nw+1]['E']: q.append((nv,nw+1,'E'))
        elif d == 'N':
            if nv-1 >= 0 and not vis[nv-1][nw]['N']: q.append((nv-1,nw,'N'))
            if nv+1 < v and not vis[nv+1][nw]['S']: q.append((nv+1,nw,'S'))
        else:
            if nv+1 < v and not vis[nv+1][nw]['S']: q.append((nv+1,nw,'S'))
            if nv-1 >= 0 and not vis[nv-1][nw]['N']: q.append((nv-1,nw,'N'))
    return ok, cnt

for _ in range(int(read())):
    h,w = map(int, read().split())
    grid = [list(read().strip()) for _ in range(h)]
    c = 0
    for i in range(h):
        for j in range(w):
            c += grid[i][j]=='A'
    if c == h*w: print(0)
    else:
        vis = [[[False, False, False, False] for _ in range(w)] for _ in range(h)]
        ans = float('inf')
        for i in range(h):
            for j in range(w):
                if grid[i][j] == 'P':
                    a, b = BFS(i,j,'E',vis,grid,c,c)
                    if a: ans = min(ans,b)
                    vis = [[[False, False, False, False] for _ in range(w)] for _ in range(h)]
                    a, b = BFS(i,j,'W',vis,grid,c,c)
                    if a: ans = min(ans,b)
                    vis = [[[False, False, False, False] for _ in range(w)] for _ in range(h)]
                    a, b = BFS(i,j,'N',vis,grid,c,c)
                    if a: ans = min(ans,b)
                    vis = [[[False, False, False, False] for _ in range(w)] for _ in range(h)]
                    a, b = BFS(i,j,'S',vis,grid,c,c)
                    if a: ans = min(ans,b)
        
        if ans == float('inf'): print('MORTAL')
        else: print(ans)
