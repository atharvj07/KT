#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 101;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    int ans = 1e9;
    for (int x = 1; x <= n; x++) {
        int cur = 0;
        for (int i = 1; i <= n; i++) {
            cur += abs(x - i) * a[i - 1];
        }
        ans = min(ans, cur);
    }

    cout << ans << endl;
}
