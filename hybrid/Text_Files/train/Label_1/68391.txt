import math

def count_perfect_powers(L, R):
    max_base = int(R**0.5) + 1
    perfect_powers = set()
    
    for base in range(2, max_base):
        power = 2
        while True:
            result = base ** power
            if result > R:
                break
            if result >= L:
                perfect_powers.add(result)
            power += 1
    return len(perfect_powers)

Q = int(input().strip())
for _ in range(Q):
    L, R = map(int, input().strip().split())
    print(count_perfect_powers(L, R))
