#include<iostream>
#include<cstdio>
#include<algorithm>
#define ll long long
using namespace std;
int main()
{
 ll t,n,a,b,c,d,sum;
 scanf("%lld",&t);
 while(t--)
 {
  scanf("%lld%lld%lld%lld%lld",&n,&a,&b,&c,&d);
  sum=max({a-b,c-d});
  sum=min({a+b,c+d,sum+n-1});
  if(sum>=0) puts("Yes");
  else puts("No");
 }
 return 0;
}
