#include<bits/stdc++.h>
using namespace std;
const int maxn = 1e6 + 5;
int n, k;
char s[maxn];
int main() {
    int q; cin >> q;
    while(q--) {
        cin >> n >> k;
        cin >> s + 1;
        if(k & 1) {
            for(int i = 1; i < n; i++) {
                if(s[i] > s[i + 1]) {
                    swap(s[i], s[i + 1]);
                    k--;
                    if(k == 0) break;
                }
            }
            for(int i = n; i > 1; i--) {
                if(s[i] < s[i - 1]) {
                    swap(s[i], s[i - 1]);
                    k--;
                    if(k == 0) break;
                }
            }
        } else {
            for(int i = 1; i < n; i++) {
                if(s[i] < s[i + 1]) {
                    swap(s[i], s[i + 1]);
                    k--;
                    if(k == 0) break;
                }
            }
            for(int i = n; i > 1; i--) {
                if(s[i] > s[i - 1]) {
                    swap(s[i], s[i - 1]);
                    k--;
                    if(k == 0) break;
                }
            }
        }
        cout << s + 1 << endl;
    }
    return 0;
}
