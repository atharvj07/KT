#include <iostream>

using namespace std;

int main() {
    long long n, k, m, d;
    cin >> n >> k >> m >> d;
    if (m < k - 1 || m < d) {
        cout << -1 << endl;
        return 0;
    }
    long long lo = 1, hi = min(m, n / d);
    while (lo <= hi) {
        long long mid = (lo + hi) / 2;
        if (mid * k <= n && ((n - mid * k) % mid == 0 || (n - mid * k) % mid <= m)) {
            lo = mid + 1;
        } else {
            hi = mid - 1;
        }
    }
    cout << hi * (k - 1) << endl;
    return 0;
}
