#include <bits/stdc++.h>
using namespace std;
double a[100005];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  double L = 0, R = 0;
  for (int i = 1; i <= n; i++) L = max(L, a[i] - a[i - 1]);
  for (int i = 1; i <= n; i++) R = min(R, a[i] - a[i - 1]);
  int ans = 0;
  double cur = R;
  for (int i = 1; i <= n; i++) {
    cur = max(cur, a[i] - a[i - 1]);
    if (cur <= L) ans = max(ans, i);
  }
  cout << ans << endl;
}
