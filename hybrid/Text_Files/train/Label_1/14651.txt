#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

long long n, k, a[2000], b[2000];
unordered_map<long long, long long> cnt;

int main() {
    cin >> n >> k;
    for (long long i = 0; i < k; i++) cin >> a[i];
    for (long long i = 0; i < n; i++) {
        cin >> b[i];
        cnt[b[i]]++;
    }
    long long ans = 0, pre = 0;
    for (long long i = 0; i < k; i++) {
        pre += a[i];
        if (cnt[pre] > 0) ans += cnt[pre], cnt[pre]--;
    }
    cout << ans << endl;
}
