for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    if a[0]!=1 or a[-1]!=n: print(-1) 
    else:
        ans = []
        l,r = 1,n
        for x in a:
            if x==l: ans.append(x); l+=1
            elif x==r: ans.append(x); r-=1
            else: print(-1); break
        for i in range(l,r+1): ans.append(i)
        print(*ans)
