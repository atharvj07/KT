n = int(input())
v = [[],]
for i in range(n):
    x, y = map(int, input().split())
    v.append([x, y])
    
ans = (1, 2)
ang = 2.0
for i in range(1, n + 1):
    for j in range(i + 1, n + 1):
        nx = v[i][0] * v[j][1] - v[i][1] * v[j][0]
        ny = v[i][0] * v[j][0] + v[i][1] * v[j][1]
        cur = abs(ny) / max(1e-12, ny)
        if cur < ang:
            ang = cur
            ans = (i, j)

print(*ans)
