class Main {
    public static void main(String[] args) throws Exception {
        java.util.Scanner in = new java.util.Scanner(System.in);
        int t = in.nextInt();
        int[] l = new int[100001], r = new int[100001];
        for (int i = 0; i < t; ++i) {
            int n = in.nextInt();
            long ll = Long.MAX_VALUE, rr = Long.MIN_VALUE;
            for (int j = 0; j < n; ++j) {
                l[j] = in.nextInt();
                r[j] = in.nextInt();
                ll = Math.min(ll, l[j]);
                rr = Math.max(rr, r[j]);
            }
            System.out.println(Math.min(Math.max(rr, ll) - Math.min(rr, ll), rr - ll + 1));
        }
    }
}
