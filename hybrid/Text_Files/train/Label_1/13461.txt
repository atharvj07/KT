#include <iostream>
#include <vector>

size_t count_ones(size_t x) {
    size_t result = 0;
    while (x > 0) {
        if (x % 2 == 1) {
            result++;
        }
        x /= 2;
    }
    return result;
}

int main() {
    size_t t;
    std::cin >> t;
    std::vector<size_t> answers(t);
    for (size_t i = 0; i < t; i++) {
        size_t n;
        std::cin >> n;
        size_t unfairness = 0;
        for (size_t j = 0; j < n; j++) {
            unfairness += count_ones(j) + count_ones(j + 1);
        }
        answers[i] = unfairness;
    }
    for (size_t i = 0; i < t; i++) {
        std::cout << answers[i] << std::endl;
    }
    return 0;
}
