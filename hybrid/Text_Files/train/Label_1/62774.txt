for _ in range(int(input())):
    a = list(map(lambda x: int(x)//2, input().split()))
    a.sort(reverse=True)
    x, y = 0, 0
    for c in a:
        if c&1:
            y += c
        else:
            x += c
    if x > y:
        print("Alice")
    elif y > x:
        print("Bob")
    else:
        print("Tie")
