#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for(int i = 0; i < n; i++)
#define rev(i, n) for(int i = n-1; i >= 0; i--)
typedef long long int ll;
const int INF = 1<<30;
typedef pair<int, int> pi;

int n, q;
bool g[3][100001], flag;

int main() {
    cin >> n >> q;
    rep(i, q) {
        int r, c; cin >> r >> c;
        g[r][c] = !g[r][c];
        if(flag) continue;
        for(int i = 1; i <= n && !flag; i++) {
            if(g[1][i] && (g[2][i-1] || g[2][i+1])) flag = 1;
        }
        if(!flag) cout << "Yes" << endl;
        else cout << "No" << endl;
    }
}
