import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int[] cnt = new int[26];
        for (char c : s.toCharArray()) cnt[c - 'A']++;
        boolean[] used = new boolean[26];
        int[] grid = new int[26];
        int[][] adj = new int[26][26];
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < 26; j++) {
                if (i != j && (cnt[i] > 0 || cnt[j] > 0)) adj[i][j] = 1;
            }
        }
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            int idx = c - 'A';
            if (used[idx]) {
                System.out.println("Impossible");
                return;
            }
            used[idx] = true;
            cnt[idx]--;
            for (int j = 0; j < 26; j++) {
                if (grid[j] > 0 && adj[idx][j] > 0) {
                    adj[idx][j]--;
                    adj[j][idx]--;
                }
            }
            grid[idx]++;
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 26; i++) {
            if (grid[i] > 0) {
                for (int j = 0; j < grid[i]; j++) sb.append((char) (i + 'A'));
            }
        }
        String t = sb.toString();
        System.out.println(t.substring(0, 13));
        System.out.println(t.substring(13));
    }
}
