#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  bool color = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      char c;
      cin >> c;
      color |= (c != 'W' && c != 'B' && c != 'G');
    }
  }
  if (color)
    cout << "#Color" << endl;
  else
    cout << "#Black&White" << endl;
  return 0;
}
