import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        String[] program = new String[n];
        for (int i = 0; i < n; i++) {
            program[i] = sc.nextLine().trim();
        }
        sc.close();

        boolean[] active = new boolean[n];
        String exception = null;
        String message = null;
        for (int i = n - 1; i >= 0; i--) {
            if (program[i].startsWith("throw")) {
                exception = program[i].substring(6, program[i].length() - 1);
            } else if (program[i].startsWith("catch")) {
                String[] parts = program[i].substring(6, program[i].length() - 1).split(",");
                if (exception != null && exception.equals(parts[0])) {
                    message = parts[1].substring(1, parts[1].length() - 1);
                    break;
                }
            } else if (program[i].startsWith("try")) {
                active[i] = true;
            }
        }

        if (message == null) {
            message = "Unhandled Exception";
        }

        System.out.println(message);
    }
}
