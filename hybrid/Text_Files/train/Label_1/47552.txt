import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int cnt = 0;
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            int a = scanner.nextInt();
            if (a % 2 != 0) {
                ok = false;
            }
            while (a % 2 == 0) {
                a /= 2;
                cnt++;
            }
        }
        if (!ok) {
            cnt = 0;
        }
        System.out.println(cnt);
    }
}
