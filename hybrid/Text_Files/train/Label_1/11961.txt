#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, min1, max1, min2, max2, min3, max3;
    cin >> n >> min1 >> max1 >> min2 >> max2 >> min3 >> max3;
    if (n - min1 - min2 < max3) {
        max3 = n - min1 - min2;
    }
    if (n - min1 - max3 < max2) {
        max2 = n - min1 - max3;
    }
    cout << max1 << " " << max2 << " " << max3 << endl;
    return 0;
}
