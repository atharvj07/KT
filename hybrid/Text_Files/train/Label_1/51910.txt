import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[][] lines = new int[n][3];
    for (int i = 0; i < n; i++) {
      lines[i][0] = sc.nextInt();
      lines[i][1] = sc.nextInt();
      lines[i][2] = sc.nextInt();
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        for (int k = j + 1; k < n; k++) {
          if (isOriginal(lines[i], lines[j], lines[k])) {
            ans++;
          }
        }
      }
    }
    System.out.println(ans);
  }

  private static boolean isOriginal(int[] a, int[] b, int[] c) {
    long det = a[1] * (b[2] - c[2]) + b[1] * (c[2] - a[2]) + c[1] * (a[2] - b[2]);
    return det != 0;
  }
}
