#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, x;
  cin >> n >> x;
  vector<pair<char, int>> v(n);
  for (int i = 0; i < n; i++) {
    char c;
    int d;
    cin >> c >> d;
    v[i] = {c, d};
  }
  int cnt = 0;
  for (auto i : v) {
    if (i.first == '+')
      x += i.second;
    else {
      if (i.second <= x) {
        x -= i.second;
      } else
        cnt++;
    }
  }
  cout << x << " " << cnt;
  return 0;
}
