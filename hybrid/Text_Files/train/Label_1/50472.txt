import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        int[] d = new int[k];
        for (int i = 0; i < k; i++) {
            d[i] = in.nextInt();
        }
        Arrays.sort(d);
        int ans = 0;
        boolean[] used = new boolean[k];
        for (int i = 0; i < k; i++) {
            if (used[i]) {
                continue;
            }
            ans++;
            for (int j = i + 1; j < k; j++) {
                if (d[i] != 0 && d[j] != 0 && (d[i] % 10 == 0 || d[j] % 10 == 0)) {
                    used[j] = true;
                }
            }
        }
        System.out.println(ans);
        for (int i = 0; i < k; i++) {
            if (!used[i]) {
                System.out.print(d[i] + " ");
            }
        }
        System.out.println();
    }
}
