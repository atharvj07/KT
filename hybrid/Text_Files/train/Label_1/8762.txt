#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 1;
int a[N], b[N];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++)
    cin >> a[i];
  copy(a + 1, a + n + 1, b + 1);
  sort(b + 1, b + n + 1);
  for (int i = 1, j = n; i <= n; i++, j--) {
    if (a[i] != b[j]) {
      cout << i - 1 << endl;
      return 0;
    }
  }
  cout << n << endl;
  return 0;
}
