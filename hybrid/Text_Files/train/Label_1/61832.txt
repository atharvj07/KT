def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(n, m, x, y, vx, vy):
    if vx == 0:
        if x == 0 or x == n:
            return f"{x} {0 if vy == -1 else m}"
        return "-1"
    if vy == 0:
        if y == 0 or y == m:
            return f"{0 if vx == -1 else n} {y}"
        return "-1"
    dx = n if vx == 1 else -n
    dy = m if vy == 1 else -m
    X = x + min((n-x)//vx, (m-y)//vy) * dx
    Y = y + min((n-x)//vx, (m-y)//vy) * dy
    gcd_nm = gcd(n, m)
    if ((X // n) + (Y // m)) % 2 == 0:
        return f"{0 if x * dx < 0 else n} {0 if y * dy < 0 else m}"
    if (n // gcd_nm) % 2 == 0 and (m // gcd_nm) % 2 == 0:
        return "-1"
    return f"{X % (2*n) % n} {Y % (2*m) % m}"

n, m, x, y, vx, vy = map(int, input().split())
print(solve(n, m, x, y, vx, vy))
