#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N=1005;
ll n,m,visit[N][N],a[N],cur=0,dp[N][N];
inline ll calc(int l,int r)
{
    return a[r]-a[l-1];
}
int main()
{
    cin>>n>>m;
    memset(dp,0,sizeof dp);
    for(int i=0;i<m;i++)
    {
        int l,r;
        cin>>l>>r;
        ++visit[l][r];
    }
    for(int r=1;r<=n;r++)
    {
        for(int l=r;l>=1;l--)
        {
            dp[l][r]=dp[l][r-1]+dp[l+1][r]+calc(l,r)*(visit[l][r]%2);
            if(calc(l,r))
            {
                if(l!=r) dp[l][r]=max(dp[l][r],dp[l+1][r-1]+calc(l,r)*(visit[l][r]%2));
            }
            else
            {
                if(l!=r+1) dp[l][r]=max(dp[l][r],dp[l+1][r-1]);
            }
        }
    }
    ll mx=0;
    for(int l=1;l<=n;l++) mx=max(mx,dp[l][n]);
    cerr<<mx<<'\n';
    int l=1,r=n;
    while(l<=r)
    {
        if(!calc(l,r))
        {
            cur+=visit[l][r]%2;
            a[l]=cur;
            a[r]=cur;
            ++l,--r;
            continue;
        }
        ll mx=0,pos=l;
        for(int i=l+1;i<r;i++)
        {
            if(dp[l][r]==dp[l][i-1]+dp[i+1][r]+calc(l,r)*(visit[l][r]%2))
            {
                mx=dp[l][i-1];
                pos=i;
            }
            if(dp[l][r]==dp[l+1][i-1]+dp[i][r]+calc(l,r)*(visit[l][r]%2))
            {
                mx=dp[l+1][i-1];
                pos=i;
            }
        }
        cur+=visit[l][r]%2;
        a[l]=cur;
        a[r]=cur;
        l=pos+1,r=pos-1;
    }
    for(int i=1;i<=n;i++)
    {
        if(a[i]%2==0) cout<<0;
        else cout<<1;
    }
}
