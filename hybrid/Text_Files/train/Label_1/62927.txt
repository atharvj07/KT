import math

n = int(input())

sm = 0
k = 0
for i in range(1, int(math.sqrt(n))):
    if n % i == 0:
        t = n // i
        if t <= 9:
            k += 1
        else:
            sum_t = 0
            while t > 0:
                sum_t += t % 10
                t /= 10
            if i + sum_t == n:
                k += 1
        if n // i <= 9:
            k += 1
        else:
            sum_t = 0
            t = n // i
            while t > 0:
                sum_t += t % 10
                t /= 10
            if i + sum_t == n:
                k += 1
print(k)
