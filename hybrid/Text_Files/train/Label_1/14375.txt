#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
#define ff first
#define ss second
#define mp make_pair
#define pii pair<int,int>
#define uu first
#define vv second
#define all(v) (v).begin(),(v).end()
#define fastio ios::sync_with_stdio(false),cin.tie(0),cout.tie(0);
#define SIZE 2005
#define MOD 1000000007
map<ll,ll>mm;
ll val(ll x)
{
    int nn=0;
    while(x%2==0)
    {
        x/=2;
        nn++;
    }
    return nn;
}

int main()
{
    ll n,p;
    cin>>n>>p;

    vector<ll>vec;

    for(ll i=1; i<=n; i++)
    {
        ll x;
        cin>>x;
        if(!x) continue;
        if(mm.find(x)==mm.end())
            vec.pb(x);
        mm[x]++;
    }

    sort(all(vec));

    ll siz=vec.size();
    bool fl=0;
    vector<ll>ans;
    for(ll i=0; i<siz; i++)
    {
        if(val(mm[vec[i]])%p==0)
        {
            continue;
        }
        if(mm[vec[i]]%p==0)
        {

            ll x=p-mm[vec[i]]%p;

            if(x%2==0 && mm[vec[i]]+x<=n)
            {
                continue;
            }
            else
            {
                if(fl==0)
                {
                    ans.pb(vec[i]);
                    fl=1;
                }
                else
                {
                    ll check=0;

                    for(ll j=0; j<siz; j++)
                    {
                        if(vec[i]+vec[j]>n)
                        {
                            continue;
                        }
                        if(val(mm[vec[i]+vec[j]]+val(mm[vec[i]]))%p==0)
                            check++;
                    }
                    if(!check)
                    {
                        ans.pb(vec[i]);
                    }
                }

            }

        }

        else if(mm[vec[i]]==1)
        {
            ll cnt=0;
            for(ll j=0; j<siz; j++)
            {
                if(vec[i]+vec[j]>n)
                {
                    continue;
                }
                if(val(mm[vec[i]+vec[j]])%p==0)
                    cnt++;
            }

            if(cnt==siz)
            {
                if(fl==0)
                {
                    ans.pb(vec[i]);
                    fl=1;
                }
                else
                {

                    if(val(mm[vec[i]])%p==0)
                        ans.pb(vec[i]);

                }
            }

        }

    }

    cout<<ans.size()<<endl;
    for(ll i=0; i<ans.size(); i++)
    {
        cout<<ans[i]<<" ";
    }
    cout<<endl;

    return 0;
}
