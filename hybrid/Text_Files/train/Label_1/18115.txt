#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

struct Rose {
    int beauty;
    char color;
    bool operator<(const Rose& other) const { return beauty < other.beauty; }
};

int main() {
    int n, k;
    cin >> n >> k;
    vector<Rose> roses(n);
    for (auto& r : roses) {
        cin >> r.beauty;
    }
    string colors;
    cin >> colors;
    for (int i = 0; i < n; ++i) {
        roses[i].color = colors[i];
    }
    sort(roses.begin(), roses.end());
    int maxBeauty = 0;
    vector<int> prefix(n);
    for (int i = 0; i < n; ++i) {
        prefix[i] = roses[i].beauty;
        if (i) {
            prefix[i] += prefix[i - 1];
        }
    }
    for (int l = 0; l < n; ++l) {
        for (int r = l; r < n; ++r) {
            if (l == r) {
                maxBeauty = max(maxBeauty, roses[l].beauty);
            } else if (roses[l].color != roses[r].color) {
                int rangeBeauty = prefix[r] - (l ? prefix[l - 1] : 0);
                if (rangeBeauty <= k) {
                    maxBeauty = max(maxBeauty, rangeBeauty);
                } else {
                    int left = l, right = r - 1;
                    while (left <= right) {
                        int mid = (left + right) / 2;
                        int midBeauty = prefix[mid] - (l ? prefix[l - 1] : 0);
                        if (midBeauty <= k) {
                            left = mid + 1;
                            maxBeauty = max(maxBeauty, midBeauty);
                        } else {
                            right = mid - 1;
                        }
                    }
                }
            }
        }
    }
    cout << (maxBeauty ? maxBeauty : -1) << endl;
    return 0;
}
