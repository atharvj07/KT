#include <bits/stdc++.h>
#define ll long long
using namespace std;
char s[2003];
int main() {
    int n;
    cin >> n >> s;
    stack<char> st;
    for (int i = n - 1; i >= 0; i--) {
        st.push(s[i]);
        for (int j = i - 1; j >= 0; j--) {
            if (s[i] == s[j]) s[j] = '0';
        }
        while (st.top() == '0') st.pop();
    }
    while (!st.empty()) { cout << st.top(); st.pop(); }
    cout << endl;
    return 0;
}
