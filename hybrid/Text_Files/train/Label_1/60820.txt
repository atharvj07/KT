n,t = map(int,input().split())
s1 = input()
s2 = input()
if(t>n-1 or t<(n%2==1 and n//2)):
    print(-1)
else:
    s3 = (s1[0] if s1[0]==s2[0] else chr(ord('a')+((ord('z')-ord('a')+1)%2)))
    s3 += s1 if n%2==0 else s2
    for i in range(1,n,2):
        s3 += s2[i] if s1[i]==s2[i] else chr(ord('a')+((ord('z')-ord('a')+1)%2))
    for i in range(0,n,2):
        s3 += s1[i] if s1[i]==s2[i] else chr(ord('a')+((ord('z')-ord('a')+1)%2))
    print(s3)
