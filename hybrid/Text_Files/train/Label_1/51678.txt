import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] b = new long[n + 1];
        long[] a = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            b[i] = sc.nextInt();
        }
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        ArrayList<Pair>[] adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 2; i <= n; i++) {
            int u = sc.nextInt();
            int w = sc.nextInt();
            adj[u].add(new Pair(i, w));
            adj[i].add(new Pair(u, 1));
        }
        long[] dist = new long[n + 1];
        PriorityQueue<Pair> pq = new PriorityQueue<>((x, y) -> Long.compare(x.first, y.first));
        pq.add(new Pair(1, 0));
        while (!pq.isEmpty()) {
            Pair cur = pq.poll();
            if (cur.second > dist[cur.first]) {
                continue;
            }
            dist[cur.first] = cur.second;
            for (Pair next : adj[cur.first]) {
                pq.add(new Pair(next.first, cur.second + next.second));
            }
        }
        boolean flag = true;
        for (int i = 1; i <= n; i++) {
            if (a[i] > b[i] + dist[i]) {
                flag = false;
                break;
            }
        }
        System.out.println(flag ? "YES" : "NO");
    }

    static class Pair {
        int first;
        long second;

        public Pair(int first, long second) {
            this.first = first;
            this.second = second;
        }
    }
}
