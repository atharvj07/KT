import java.util.*;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int[] a = new int[n];
		for (int i = 0; i < n; i++) {
			a[i] = sc.nextInt();
		}
		int max = 0;
		for (int i = 0; i < n; i++) {
			double sqrt = Math.sqrt(a[i]);
			if (Math.floor(sqrt) != sqrt) {
				max = Math.max(max, a[i]);
			}
		}
		System.out.println(max);
	}
}
