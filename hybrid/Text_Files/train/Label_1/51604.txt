import java.util.Scanner;

public class CapsLock {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String word = scanner.nextLine();

        // Check if the word only contains uppercase letters or all letters except for the first one are uppercase.
        if (word.toUpperCase().equals(word) || word.substring(1).toUpperCase().equals(word.substring(1))) {
            // Change the case of all letters to the opposite case.
            word = word.toLowerCase();
        }

        // Print the result.
        System.out.println(word);
    }
}
