import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), L = sc.nextInt(), s = sc.nextInt(), t = sc.nextInt();
        edge[] edg = new edge[m];
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt(), b = sc.nextInt(), w = sc.nextInt();
            edg[i] = new edge(a, b, w);
        }
        int[] dis = new int[n];
        int[] par = new int[n];
        for (int i = 0; i < n; i++) {
            dis[i] = 1 << 30;
        }
        Queue<Integer> q = new LinkedList<Integer>();
        q.add(s);
        dis[s] = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int i = 0; i < m; i++) {
                if ((edg[i].a == u && edg[i].b != par[u]) || (edg[i].b == u && edg[i].a != par[u])) {
                    if (dis[u] + edg[i].w < dis[edg[i].a]) {
                        dis[edg[i].a] = dis[u] + edg[i].w;
                        par[edg[i].a] = u;
                        q.add(edg[i].a);
                    }
                    if (dis[u] + edg[i].w < dis[edg[i].b]) {
                        dis[edg[i].b] = dis[u] + edg[i].w;
                        par[edg[i].b] = u;
                        q.add(edg[i].b);
                    }
                }
            }
        }
        if (dis[t] != L) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        for (int i = 0; i < m; i++) {
            if (edg[i].w == 0) {
                edg[i].w = L - dis[edg[i].a] - dis[edg[i].b];
            }
            System.out.println(edg[i].a + " " + edg[i].b + " " + edg[i].w);
        }
    }

    static class edge {
        int a, b, w;

        public edge(int a, int b, int w) {
            this.a = a;
            this.b = b;
            this.w = w;
        }
    }
}
