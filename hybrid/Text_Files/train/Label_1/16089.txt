#include <bits/stdc++.h>
using namespace std;
int n, ans, a[100005];
int solve(int l, int r) {
  if (l > r) return 0;
  if (l == r) return 1;
  if (a[l] == a[r]) return solve(l + 1, r - 1);
  return max({solve(l + 1, r), solve(l, r - 1), r - l + 1});
}
int main() {
  cin >> n;
  char x;
  for (int i = 1; i <= n; i++) {
    cin >> x;
    a[i] = x - '0';
  }
  ans = solve(1, n);
  cout << ans << endl;
  return 0;
}
