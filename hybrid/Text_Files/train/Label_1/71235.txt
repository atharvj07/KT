def main():
    s = input()
    t = input()
    n = len(s)
    m = len(t)
    s_suf = [0] * n
    for i in range(n-1,-1,-1):
        if i < n - 1:
            s_suf[i] = s_suf[i+1]
        if s[i] == s[n-1]:
            s_suf[i]+=1
    ans = 0
    for i in range(m):
        L = i
        R = m - i - 1
        if R >= n:
            ans += s_suf[n-1] + (t[i:i+n-1]==s)
            if i + n - 1 < m:
                ans -= (t[i+n:i+n+1]==s)
    print(ans)

main()
