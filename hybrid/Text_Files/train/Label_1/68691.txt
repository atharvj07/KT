n = int(input().strip())
arr = list(map(int, input().strip().split()))

dp = [[float('inf') for _ in range(n)] for _ in range(n)]
for i in range(n):
    dp[i][0] = abs(arr[i] - 1)

for i in range(1, n):
    min_val = float('inf')
    for j in range(n):
        min_val = min(min_val, dp[j][i-1])
        dp[j][i] = min_val + abs(arr[j] - (i + 1))

print(min(dp[j][n-1] for j in range(n)))
