import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] a = new int[1 << n];
        for (int i = 0; i < (1 << n); i++) {
            a[i] = sc.nextInt();
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < (1 << (n - i - 1)); j++) {
                a[j] -= a[j + (1 << (n - i - 1))];
                if (a[j] < 0) {
                    a[j] = 0;
                }
            }
        }

        System.out.println(a[0]);
    }
}
