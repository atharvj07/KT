#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
        }
    }
    int ans = n;
    for (int i = 0; i < n / 2; i++) {
        bool ok = true;
        for (int j = 0; j < m; j++) {
            if (a[i][j] != a[n - i - 1][j]) {
                ok = false;
            }
        }
        if (ok) {
            ans = n / 2;
        }
    }
    cout << ans << endl;
    return 0;
}
