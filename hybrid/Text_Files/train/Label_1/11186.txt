#include <bits/stdc++.h>
#define nl "\n"
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> p(n + 1), q(n + 1);
  for (int i = 1; i <= n; i++) { cin >> p[i]; }
  for (int i = 1; i <= n; i++) { cin >> q[i]; }
  if (k > n) {
    cout << "NO" << nl;
  } else {
    cout << "YES" << nl;
    string s(n, 'a');
    vector<int> used(26);
    for (int i = 1, j = 1; i <= n; i++) {
      if (p[i] == q[i]) {
        s[p[i] - 1] = 'a' + j;
        used[j]++;
        j++;
      }
    }
    bool taken[26] = {false};
    for (int i = 1; i <= n; i++) {
      if (used[s[q[i] - 1] - 'a'] < 2) {
        for (int ch = 'a'; ch <= 'z'; ch++) {
          if (!taken[ch - 'a'] && (ch != s[p[i] - 1])) {
            taken[ch - 'a'] = true;
            s[q[i] - 1] = ch;
            used[s[q[i] - 1] - 'a']++;
            break;
          }
        }
      }
    }
    for (int i = 1; i <= n; i++) {
      if (!used[s[p[i] - 1] - 'a']) {
        for (int ch = 'a'; ch <= 'z'; ch++) {
          if (!taken[ch - 'a']) {
            s[p[i] - 1] = ch;
            taken[ch - 'a'] = true;
            break;
          }
        }
      }
    }
    cout << s << nl;
  }
  return 0;
}
