T = int(input())
for _ in range(T):
    n, m = map(int, input().split())
    weights = list(map(int, input().split()))

    if n == 2 and m == 1:
        print(-1)
        continue
    elif m < n:
        print(-1)
        continue

    total_cost = 2 * sum(weights)
    print(total_cost)
    for i in range(1, n):
        print(i, i + 1)
    print(n, 1)
