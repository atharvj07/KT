from collections import Counter

t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    f = Counter(a)
    if f[0]%2 and sum(a)==0:
        print("NO")
        continue
    b = [a[0]]
    for e in sorted(set(a)):
        if e*len(b)+sum(b)<0:
            b.append(e)
        else:
            b.insert(0, e)
    print("YES")
    print(*b)
