import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        long[] costToWin = new long[m + 1];
        for (int i = 0; i < n; i++) {
            int party = input.nextInt();
            long cost = input.nextInt();
            costToWin[party] = Math.min(costToWin[party], cost);
        }
        long minCost = Long.MAX_VALUE;
        for (int i = 1; i <= m; i++) {
            minCost = Math.min(minCost, costToWin[i]);
        }
        System.out.println(minCost);
    }
}
