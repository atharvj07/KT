#include <cstdio>
#include <vector>
using namespace std;

int N, M;
vector<bool> V;

int main() {
  scanf("%d%d", &N, &M);
  V.assign(N + 1, 0);

  printf("%d", N);
  V[N] = 1;
  int ac = 1;
  for (int i = N - 1, cnt = 0; i >= 1; i--) {
    int rem = N - i;
    int cur = cnt + (rem * (rem + 1) / 2);
    if (cur < M) {
      printf(" %d", i);
      cnt += rem;
      V[i] = 1;
    } else {
      for (int j = i + 1; j <= N; j++) {
        if (!V[j]) {
          printf(" %d", j);
          V[j] = 1;
          ac++;
          break;
        }
      }
    }
  }
  puts("");
  return 0;
}
