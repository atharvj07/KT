#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  int t[n], w[n], ans = 0;
  for (int i = 0; i < n; i++) {
    cin >> t[i] >> w[i];
    ans += t[i];
  }
  sort(t, t + n);
  sort(w, w + n, greater<>());
  int tot = 0;
  for (int i = 0; i < n; i++) {
    if (tot + t[i] <= w[i]) {
      tot += t[i];
      ans -= t[i];
    } else {
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
