#include <bits/stdc++.h>
using namespace std;
int n,k;
string s;
int main(){
    cin>>n>>k;
    cin>>s;
    int ans=0;
    int cnt=(n+k)/2;
    int fr[26];
    for(int i=0;i<=26;i++) fr[i]=0;
    for(int i=0;i<n;i++) fr[s[i]-'A']++;
    for(int i=0;i<=26;i++){
        if(fr[i]<=cnt) ans+=fr[i]*fr[i];
        else ans+=cnt*cnt;
    }
    cout<<ans<<endl;
}
