r, c = [int(x) for x in input().split()]
grid = []
for i in range(r):
    grid.append([x for x in input()])
ans = 0
for i in range(r):
    if 'S' not in grid[i]:
        for j in range(c):
            if grid[i][j] != 'S':
                ans += 1
for j in range(c):
    f = True
    for i in range(r):
        if grid[i][j] == 'S':
            f = False
            break
    if f:
        for i in range(r):
            if grid[i][j] != 'S':
                ans += 1
print(ans)
