import sys
input = sys.stdin.readline

n = int(input())
p = list(map(int, input().split()))
ans = []
for i in range(n):
    for j in range(i + 1, n):
        if abs(i - j) >= n // 2:
            ans.append((i + 1, j + 1))
            p[i], p[j] = p[j], p[i]
print(len(ans))
for a, b in ans:
    print(a, b)
