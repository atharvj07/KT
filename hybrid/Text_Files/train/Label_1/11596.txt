#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1e9+7;
int a[26];
ll dp[26][100001][2], f[100001], ff[100001];
int k, n, s;
ll solve (int nn, ll sum, int taken) {
    if (nn == n + 1) {
        if (sum == s) return 1;
        return 0;
    }
    if (dp[nn][sum][taken] != -1) return dp[nn][sum][taken];
    ll ans = solve(nn + 1, sum, taken);
    ans += solve(nn + 1, sum + a[nn], taken + 1);
    if (taken < k) ans += solve(nn + 1, sum + f[a[nn]], taken + 1);
    ans %= mod;
    return dp[nn][sum][taken] = ans;
}
int main() {
    cin >> n >> k >> s;
    for (int i = 1; i <= n; i++) cin >> a[i];
    f[1] = 1;
    for (int i = 2; i <= 100000; i++) f[i] = (f[i - 1] * i) % mod;
    for (int i = 1; i <= 100000; i++) ff[i] = (ff[i - 1] * f[i]) % mod;
    memset(dp, -1, sizeof(dp));
    cout << solve(1, 0, 0) << '\n';
    return 0;
}
