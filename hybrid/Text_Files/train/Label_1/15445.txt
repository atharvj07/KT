#include <bits/stdc++.h>
using namespace std;

int a[1<<24], b[1<<24], l, r;

int gcd(int a, int b) {
    while(b) {
        int t=b;
        b=a%b;
        a=t;
    }
    return a;
}

int check(int x) {
    for(int i=l; i<=r; i++) {
        if((a[i]%x==0) || (b[i]%x==0)) {
            return 1;
        }
    }
    return 0;
}

int main() {
    l=0;
    cin>>r;
    for(int i=1; i<=r; i++) {
        int d=gcd(i,r);
        if((r/d)>(1<<24-1)) continue;
        a[++l]=i/d;
        b[l]=r/d;
    }
    for(int i=2; i<=r; i++) {
        if((r%i==0) && check(i)) {
            cout<<i<<endl;
            return 0;
        }
    }
    cout<<-1<<endl;
    return 0;
}
