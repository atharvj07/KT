import java.util.*;

class Main {

    static int n, m, k, s;
    static char[][] a;
    static boolean[][] u;
    static Queue<int[]> q = new LinkedList<>();

    static boolean valid(int i, int j) {
        return i >= 0 && i < n && j >= 0 && j < m && a[i][j] == '.';
    }

    static void floodfill(int x, int y) {
        u[x][y] = true;
        q.add(new int[]{x, y});
        while (!q.isEmpty()) {
            int[] cur = q.poll();
            if (valid(cur[0], cur[1] - 1) && !u[cur[0]][cur[1] - 1]) {
                u[cur[0]][cur[1] - 1] = true;
                q.add(new int[]{cur[0], cur[1] - 1});
            }
            if (valid(cur[0], cur[1] + 1) && !u[cur[0]][cur[1] + 1]) {
                u[cur[0]][cur[1] + 1] = true;
                q.add(new int[]{cur[0], cur[1] + 1});
            }
            if (valid(cur[0] - 1, cur[1]) && !u[cur[0] - 1][cur[1]]) {
                u[cur[0] - 1][cur[1]] = true;
                q.add(new int[]{cur[0] - 1, cur[1]});
            }
            if (valid(cur[0] + 1, cur[1]) && !u[cur[0] + 1][cur[1]]) {
                u[cur[0] + 1][cur[1]] = true;
                q.add(new int[]{cur[0] + 1, cur[1]});
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        s = 0;
        a = new char[n][m];
        u = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            String line = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = line.charAt(j);
                if (a[i][j] == '.') {
                    s++;
                }
            }
        }
        int cx = -1, cy = -1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == '.' && !u[i][j]) {
                    floodfill(i, j);
                    cx = i;
                    cy = j;
                }
            }
        }
        List<int[]> l = new ArrayList<>();
        if (s - k == 0) {
            System.out.println("impossible");
            return;
        }
        floodfill(cx, cy);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == '.' && !u[i][j]) {
                    l.add(new int[]{i, j});
                }
            }
        }
        if (l.size() < k) {
            System.out.println("impossible");
            return;
        }
        for (int i = 0; i < k; i++) {
            int x = l.get(i)[0];
            int y = l.get(i)[1];
            a[x][y] = 'X';
        }
        for (char[] row : a) {
            System.out.println(new String(row));
        }
    }
}
