import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] nm = br.readLine().split(" ");
    int n = Integer.parseInt(nm[0]);
    int m = Integer.parseInt(nm[1]);
    long[] x = new long[n + 1];
    long[] y = new long[m + 1];
    String[] xs = br.readLine().split(" ");
    String[] ys = br.readLine().split(" ");
    for (int i = 0; i < n; i++) {
      x[i + 1] = Long.parseLong(xs[i]);
    }
    for (int i = 0; i < m; i++) {
      y[i + 1] = Long.parseLong(ys[i]);
    }
    BigInteger ans = BigInteger.ZERO;
    for (int i = 1; i <= n; i++) {
      for (int j = i + 1; j <= n; j++) {
        for (int k = 1; k <= m; k++) {
          for (int l = k + 1; l <= m; l++) {
            ans = ans.add(BigInteger.valueOf(x[j] - x[i]).multiply(BigInteger.valueOf(y[l] - y[k])));
          }
        }
      }
    }
    System.out.println(ans.mod(BigInteger.valueOf(1000000007)));
  }
}
