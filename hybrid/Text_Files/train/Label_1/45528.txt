import java.util.Scanner;

public class Main {

    static int cw(int x1, int y1, int x2, int y2) {
        return x1 * y2 - x2 * y1;
    }

    static boolean collinear(int x1, int y1, int x2, int y2, int x3, int y3) {
        return cw(x2 - x1, y2 - y1, x3 - x1, y3 - y1) == 0;
    }

    static int side(int x1, int y1, int x2, int y2, int x3, int y3) {
        return cw(x2 - x1, y2 - y1, x3 - x1, y3 - y1);
    }

    static boolean between(int x1, int y1, int x2, int y2, int x3, int y3) {
        return side(x1, y1, x2, y2, x3, y3) == 0 && cw(x1 - x2, y1 - y2, x3 - x2, y3 - y2) <= 0;
    }

    static boolean intersect(int x1, int y1, int x2, int y2, int x3, int y3, int x4, int y4) {
        return side(x1, y1, x2, y2, x3, y3) * side(x1, y1, x2, y2, x4, y4) < 0
                && side(x3, y3, x4, y4, x1, y1) * side(x3, y3, x4, y4, x2, y2) < 0;
    }

    static boolean see(int xv, int yv, int xp, int yp, int xw1, int yw1, int xw2, int yw2, int xm1, int ym1, int xm2, int ym2) {
        if (intersect(xw1, yw1, xw2, yw2, xv, yv, xp, yp) || intersect(xm1, ym1, xm2, ym2, xv, yv, xp, yp)) {
            return false;
        }
        if (!collinear(xv, yv, xp, yp, xm1, ym1)) {
            if (between(xm1, ym1, xm2, ym2, xv, yv) != between(xm1, ym1, xm2, ym2, xp, yp)) {
                return true;
            }
            if (between(xm1, ym1, xm2, ym2, xv, yv) && between(xm1, ym1, xm2, ym2, xp, yp)) {
                return intersect(xm1, ym1, xm2, ym2, xw1, yw1, xw2, yw2);
            }
        }
        return false;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int xv = scanner.nextInt();
        int yv = scanner.nextInt();
        int xp = scanner.nextInt();
        int yp = scanner.nextInt();
        int xw1 = scanner.nextInt();
        int yw1 = scanner.nextInt();
        int xw2 = scanner.nextInt();
        int yw2 = scanner.nextInt();
        int xm1 = scanner.nextInt();
        int ym1 = scanner.nextInt();
        int xm2 = scanner.nextInt();
        int ym2 = scanner.nextInt();
        scanner.close();
        System.out.println(see(xv, yv, xp, yp, xw1, yw1, xw2, yw2, xm1, ym1, xm2, ym2) ? "YES" : "NO");
    }
}
