#include <bits/stdc++.h>

#define ll long long

using namespace std;

const int mod = 1e9 + 7;
int dp[1000001][3];

int n;
int m[1000001];

int f(int pos, int rem) {
  if (pos == n) { return rem == 0; }
  int &ans = dp[pos][rem];
  if (ans != -1) { return ans; }
  ans = 0;
  for (int i = 0; i <= m[pos]; i++) {
    ans = (ans + f(pos + 1, rem - i)) % mod;
  }
  return ans;
}

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> m[i]; }

  memset(dp, -1, sizeof dp);
  cout << f(0, 2) << "\n";
  return 0;
}
