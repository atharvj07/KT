import sys
n, m = map(int, input().split())
a = []
for i in range(n):
    a.append([int(x) for x in input()])
b = [[0 for j in range(m)] for i in range(n)]
for i in range(n):
    for j in range(m):
        for k in range(n):
            if a[i][j] == 1 and a[k][j] == 1:
                b[i][j] = 1
                b[k][j] = 1
for i in range(n):
    for j in range(m):
        if b[i][j] == 0:
            print("NO")
            sys.exit()
print("YES")
