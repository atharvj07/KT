import java.util.Scanner;

public class Main {
    static final long MIN = 2;
    static long n = 0;
    static long a[][] = new long[100001][2];

    public static void dfs(int i, long cnt) {
        if (i == 0) {
            a[(int) cnt + 1][0] = Math.min(a[(int) cnt + 1][0], n);
            a[(int) cnt + 1][1] = Math.max(a[(int) cnt + 1][1], n);
            return;
        }
        for (int j = 1; j <= 50000; j++) {
            n = cnt + a[j][1];
            if ((n % i) == 0) dfs(i - 1, n);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        for (int i = 0; i < 100001; i++) {
            a[i][0] = 1000000000000000000L;
            a[i][1] = 0;
        }
        for (int i = 0; i < k; i++) {
            a[sc.nextInt()][1] = 1000000000000000000L;
        }
        n = MIN - 1;
        dfs(k, MIN - 1);
        if (a[k][0] > a[k][1]) System.out.println(-1);
        else System.out.println(a[k][0] + " " + a[k][1]);
    }
}
