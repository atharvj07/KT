import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextInt();
        long b = in.nextInt();
        long f = in.nextInt();
        long k = in.nextInt();
        long refuel = (a - f + b - 1) / b;
        System.out.println(refuel * k - (f > b ? 1 : 0));
    }
}
