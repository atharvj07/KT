import java.io.*;
import java.util.*;

class Main {
    static int n;
    static int[] p;
    static boolean[] vis;

    static boolean canDance() {
        Arrays.fill(vis, false);
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (!vis[i]) {
                int cur = i;
                int next = p[cur] - 1;
                while (next != i) {
                    vis[cur] = true;
                    cnt++;
                    cur = next;
                    next = p[cur] - 1;
                }
                if (cnt > 2) {
                    return false;
                }
                cnt = 0;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            n = sc.nextInt();
            p = new int[n];
            vis = new boolean[n];
            for (int i = 0; i < n; i++) {
                p[i] = sc.nextInt();
            }
            if (canDance()) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
