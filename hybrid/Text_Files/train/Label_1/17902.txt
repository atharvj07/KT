#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll GCD(ll a, ll b)
{
    if (!b)
        return a;
    return GCD(b, a % b);
}
map<pair<ll, ll>, ll> ans;
ll get_kth(ll x, ll p, ll k)
{
    if (ans.count({x, p}))
        return ans[{x, p}];
    ll prv = 0;
    for (ll y = x + 1;; y++)
    {
        if (GCD(p, y) == 1)
        {
            prv++;
            if (prv == k)
            {
                ans[{x, p}] = y;
                return y;
            }
        }
    }
}
int main()
{
    ll t, x, p, k;
    cin >> t;
    while (t--)
    {
        cin >> x >> p >> k;
        cout << get_kth(x, p, k) << '\n';
    }
    return 0;
}
