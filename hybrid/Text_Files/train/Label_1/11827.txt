#include<bits/stdc++.h>
#define MX 3010
#define pb push_back
using namespace std;
vector<int>adj[MX],adj2[MX];
int vis[MX],chk[MX],res,n,m;
void dfs(int x){
    vis[x]=1;
    for(int y:adj[x]){
        if(!vis[y]){
            dfs(y);
        }
    }
}
void dfs2(int x,int p){
    chk[x]=p;
    for(int y:adj2[x]){
        if(!chk[y]){
            dfs2(y,p);
        }
    }
}
int main(){
    cin>>n>>m;
    for(int i=1;i<=m;++i){
        int x,y;
        cin>>x>>y;
        adj[x].pb(y),adj[y].pb(x);
    }
    dfs(1);
    int cnt=0;
    for(int i=1;i<=n;++i){
        if(!vis[i]){
            ++cnt;
        }
    }
    if(cnt>1){
        cout<<-1<<endl;
        return 0;
    }
    int s1,t1,l1,s2,t2,l2;
    cin>>s1>>t1>>l1>>s2>>t2>>l2;
    int dis[n+1];
    queue<int>q;
    memset(dis,-1,sizeof(dis));
    dis[s1]=0,q.push(s1);
    while(!q.empty()){
        int u=q.front();
        q.pop();
        for(int i=0;i<adj[u].size();++i){
            int v=adj[u][i];
            if(dis[v]==-1){
                dis[v]=dis[u]+1;
                q.push(v);
            }
        }
    }
    if(dis[t1]>l1||dis[t2]>l2) cout<<-1<<endl;
    else{
        for(int i=1;i<=m;++i){
            int x,y;
            cin>>x>>y;
            if(x==s1&&y==t1||y==s1&&x==t1||x==s2&&y==t2||x==t2&&y==s2){
                ++res;
            }
            else{
                adj2[x].pb(y),adj2[y].pb(x);
            }
        }
        dfs2(1,1);
        int p1=chk[s1],p2=chk[s2];
        if(p1!=p2){
            int ans=min(res,m-res);
            cout<<ans<<endl;
        }
        else{
            cout<<-1<<endl;
        }
    }
}
