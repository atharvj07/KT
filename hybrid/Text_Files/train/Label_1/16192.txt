#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, a[2005], b[2005];
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i];
    }
    sort(a, a + n);
    sort(b, b + n);
    int ans = (b[0] - a[0] + m) % m;
    for (int i = 0; i < n; i++) {
        if ((a[i] + ans) % m != b[i]) {
            ans = -1;
            break;
        }
    }
    cout << ans << endl;
}
