import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        Map<String, Integer> a = new HashMap<>();
        for (int i = 0; i < N; i++) {
            String s = sc.next();
            int x = sc.nextInt();
            a.put(s, x);
        }
        int M = sc.nextInt();
        Map<String, String> s = new HashMap<>();
        Map<String, String> t = new HashMap<>();
        for (int i = 0; i < M; i++) {
            String ss = sc.next();
            String tt = sc.next();
            s.put(ss, tt);
            t.put(tt, ss);
        }
        int ans = 0;
        List<String> list = new ArrayList<>();
        for (String key : a.keySet()) {
            if (list.contains(key)) {
                continue;
            }
            list.add(key);
            int min = a.get(key);
            String cur = key;
            while (t.containsKey(cur)) {
                cur = t.get(cur);
                min = Math.min(min, a.get(cur));
            }
            ans += min;
        }
        System.out.println(ans);
    }
}
