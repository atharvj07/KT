import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long q = sc.nextLong();
        if (q % 2 == 0) {
            System.out.println(1);
            System.out.println(q / 2);
        } else if (q % 3 == 0) {
            System.out.println(1);
            System.out.println(q / 3);
        } else {
            System.out.println(2);
        }
    }
}
