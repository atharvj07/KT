#include <iostream>
#include <unordered_map>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, ans = 1e9;
    vector<int> a, b, dp;
    cin >> n >> m;
    unordered_map<int, int> ma, mb;
    for (int i = 0; i < n; ++i) {
      int x;
      cin >> x;
      a.push_back(x);
      ma[x] = 1;
    }
    for (int i = 0; i < m; ++i) {
      int x;
      cin >> x;
      b.push_back(x);
      mb[x] = 1;
    }
    int l = 0, r = 0;
    for (int i = 0; i < m; ++i) {
      if (ma[b[i]]) {
        if (l + 1 > r)
          r = l + 1;
        l = r;
        while (l <= r && mb[a[l]]) l++;
        if (l - r < ans) ans = l - r, dp = vector<int>(a.begin() + r, a.begin() + l);
      }
    }
    if (ans == 1e9) {
      cout << "NO\n";
      continue;
    }
    cout << "YES\n";
    cout << ans << ' ';
    for (auto x : dp) cout << x << ' ';
    cout << '\n';
  }
}
