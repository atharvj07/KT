import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int[] arr = new int[k];
        boolean[] zero = new boolean[k];
        for (int i = 0; i < k; i++) {
            arr[i] = sc.nextInt();
            zero[i] = arr[i] % 10 == 0;
        }
        int[] ans = new int[k];
        int idx = 0;
        for (int i = 0; i < k; i++) {
            if (zero[i]) {
                ans[idx++] = arr[i];
            }
        }
        for (int i = 0; i < k; i++) {
            if (!zero[i]) {
                ans[idx++] = arr[i];
            }
        }
        System.out.println(idx);
        for (int i = 0; i < idx; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
