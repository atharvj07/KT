#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, a[1005][1005];
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
  }
  int f[1005][1005][2][2];
  f[1][1][0][0] = a[1][1];
  f[n][1][1][0] = a[n][1];
  for (int j = 2; j <= m; j++) {
    f[1][j][0][1] = f[1][j - 1][0][0];
    f[1][j][1][0] = f[1][j - 1][0][0] + a[1][j];
  }
  for (int i = 2; i <= n; i++) {
    f[i][1][1][1] = f[i - 1][1][1][0];
    f[i][1][0][0] = f[i - 1][1][1][0] + a[i][1];
  }
  for (int i = 2; i <= n; i++) {
    for (int j = 2; j <= m; j++) {
      f[i][j][0][0] = max(f[i - 1][j][1][0] + a[i][j], f[i][j - 1][0][0] + a[i][j]);
      f[i][j][0][1] = max(f[i][j - 1][0][0], f[i - 1][j][1][0]);
      f[i][j][1][0] = max(f[i - 1][j][1][0] + a[i][j], f[i][j - 1][0][0]);
      f[i][j][1][1] = max(f[i][j - 1][0][0], f[i - 1][j][1][0]);
    }
  }
  cout << max({f[n][m - 1][0][0], f[n][m - 1][0][1], f[n][m - 1][1][0], f[n][m - 1][1][1]});
  return 0;
}
