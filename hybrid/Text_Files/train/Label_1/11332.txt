#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin >> n;
    if (n==3) cout << "-1";
    else {
        vector<int> p;
        p.push_back(n);
        for(int i = 1; i < n-1; ++i)
            p.push_back(i);
        p.push_back(n-1);
        for (auto& x : p) cout << x << " ";
    }
}
