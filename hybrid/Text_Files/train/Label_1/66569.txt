import sys

input = sys.stdin.readline

def solve():
    n = int(input())
    arr = list(map(int, input().strip().split()))
    ans = 0
    for i in range(n - 1):
        if arr[i] <= arr[i + 1]:
            continue
        ans += 1
        arr = arr[:i + 1] + arr[i:i + 1][::-1] + arr[i + 1:]
    return ans

for _ in range(int(input())):
    print(solve())
