t = int(input())
for _ in range(t):
    s = input()
    max_gap = 1
    current_gap = 0
    for char in s:
        if char == 'R':
            max_gap = max(max_gap, current_gap + 1)
            current_gap = 0
        else:
            current_gap += 1
    max_gap = max(max_gap, current_gap + 1)
    print(max_gap)
