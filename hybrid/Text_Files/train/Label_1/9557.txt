#include <iostream>
#include <vector>

int gcd(int a, int b) {
  if (b == 0)
    return a;
  return gcd(b, a % b);
}

int main() {
  int t;
  std::cin >> t;

  for (int i = 0; i < t; i++) {
    int a, b, c;
    std::cin >> a >> b >> c;

    int sum = a + b + c;
    int g = gcd(gcd(a, b), c);
    std::cout << (g != 1 && sum % g == 0 ? "YES" : "NO") << "\n";
  }

  return 0;
}
