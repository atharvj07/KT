#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
  long long a, b;
  cin >> a >> b;
  vector<long long> c = {3, 4, 5, 6, 7, 8, 9, 11, 12, 14, 15, 18, 21, 24, 27, 30, 33, 36, 42, 48, 54, 56, 63, 72, 84, 96, 126, 168};
  vector<long long> s = {2, 2, 1, 2, 1, 1, 2, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 2, 1, 2, 1, 2, 1, 2, 1};
  long long t = a + b, ans = 2;
  if (a && b) ans += 1;
  for (long long i = 0; i < 28; i++) {
    if (t % c[i] == 0 && s[i] == 1) ans += 1;
  }
  long long sq = static_cast<long long>(sqrt(t));
  for (long long i = 3; i <= sq; i += 2) {
    if (t % i == 0) ans += 1;
    if (t / i != i && t % (t / i) == 0) ans += 1;
  }
  cout << ans << endl;
  return 0;
}
