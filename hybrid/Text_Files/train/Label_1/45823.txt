import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n1 = sc.nextInt(), n2 = sc.nextInt(), k1 = sc.nextInt(), k2 = sc.nextInt();
        int mod = 100000000;
        long[][][] dp = new long[3][n1 + n2 + 1][k1 + k2 + 1];
        dp[0][0][0] = 1;
        for (int i = 0; i < n1 + n2; i++) {
            for (int j = 0; j <= Math.min(i, k1); j++) {
                dp[1][i + 1][j] = (dp[1][i + 1][j] + dp[0][i][j]) % mod;
                dp[0][i + 1][j + 1] = (dp[0][i + 1][j + 1] + dp[0][i][j]) % mod;
                dp[2][i + 1][j] = (dp[2][i + 1][j] + dp[1][i][j]) % mod;
            }
            for (int j = 0; j <= Math.min(i, k2); j++) {
                dp[2][i + 1][j] = (dp[2][i + 1][j] + dp[2][i][j]) % mod;
            }
        }
        System.out.println(dp[2][n1 + n2][0]);
    }
}
