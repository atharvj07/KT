#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> left(n, 0), right(n, 0);
  for (int i = 0; i < n; i++) {
    if (s[i] == 'L') {
      left[i] = 1;
      if (i > 0) right[i - 1] = 1;
    } else if (s[i] == 'R') {
      right[i] = 1;
      if (i > 0) left[i - 1] = 1;
    }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (!left[i] && !right[i]) ans++;
  }
  cout << ans << endl;
  return 0;
}
