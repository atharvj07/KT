import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] generator = new int[4];
        for (int i = 0; i < 4; i++)
            generator[i] = scanner.nextInt();
        Team[] teams = new Team[n];
        for (int i = 0; i < n; i++)
            teams[i] = new Team(scanner.next(), scanner.nextInt());
        Arrays.sort(teams, Collections.reverseOrder());
        int[][] baskets = new int[4][n / 4];
        for (int i = 0, idx = 0; i < n / 4; i++)
            for (int j = 0; j < 4; j++, idx++)
                baskets[j][i] = idx;
        int[][] groups = new int[n / 4][4];
        for (int i = 0; i < n / 4 - 1; i++) {
            for (int j = 0; j < 4; j++) {
                int rnd = (generator[0] * generator[1] + generator[2]) % generator[3];
                groups[i][j] = baskets[j][rnd % (n / 4 - i)];
                baskets[j][rnd % (n / 4 - i)] = baskets[j][n / 4 - i - 1];
                generator[0] = rnd;
            }
        }
        for (int j = 0; j < 4; j++)
            groups[n / 4 - 1][j] = baskets[j][0];
        char ch = 'A';
        for (int[] group : groups) {
            System.out.println("Group " + ch++ + ":");
            for (int idx : group)
                System.out.println(teams[idx].name);
            System.out.println();
        }
    }

    static class Team implements Comparable<Team> {
        String name;
        int rating;

        Team(String name, int rating) {
            this.name = name;
            this.rating = rating;
        }

        @Override
        public int compareTo(Team o) {
            return this.rating - o.rating;
        }
    }
}
