import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        in.nextLine();
        String s = in.nextLine();
        int[] p = new int[m];
        for (int i = 0; i < n; i++) {
            p[s.charAt(i) - 'a']++;
        }
        int ans = 0;
        for (int i = 1; i < m; i++) {
            ans += Math.abs(p[i] - p[i - 1]);
        }
        System.out.println(ans);
    }
}
