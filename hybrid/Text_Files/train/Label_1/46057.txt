import java.util.*;

public class Main {
    private static class Node {
        int index;
        int parent;
        List<Integer> adj;
        int piece;

        public Node(int index, int parent) {
            this.index = index;
            this.parent = parent;
            this.adj = new ArrayList<>();
            this.piece = 0;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            nodes[i] = new Node(i, -1);
        }
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            nodes[a].adj.add(b);
            nodes[b].adj.add(a);
        }
        for (int i = 0; i < n; i++) {
            nodes[i + 1].piece = s.charAt(i) - '0';
        }
        if (!canUnify(nodes, 1)) {
            System.out.println(-1);
            return;
        }
        int ans = dfs(nodes, 1, -1, 0);
        System.out.println(ans);
    }

    private static int dfs(Node[] nodes, int u, int p, int ans) {
        int sum = nodes[u].piece;
        for (int v : nodes[u].adj) {
            if (v != p) {
                sum += dfs(nodes, v, u, ans);
            }
        }
        if (sum != 0) {
            ans += sum - 1;
        }
        return sum;
    }

    private static boolean canUnify(Node[] nodes, int u) {
        int sum = nodes[u].piece;
        for (int v : nodes[u].adj) {
            if (!canUnify(nodes, v)) {
                return false;
            }
            sum += nodes[v].piece;
        }
        return sum == 0 || sum == nodes[u].piece;
    }
}
