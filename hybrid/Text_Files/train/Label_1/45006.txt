import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = s.nextInt();
        }
        int m = s.nextInt();
        for (int i = 0; i < m; i++) {
            int p = s.nextInt();
            int x = s.nextInt();
            int sum = 0;
            for (int j = 0; j < n; j++) {
                if (j + 1 == p) {
                    sum += x;
                } else {
                    sum += t[j];
                }
            }
            System.out.println(sum);
        }
    }
}
