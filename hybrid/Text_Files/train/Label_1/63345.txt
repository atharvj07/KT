n, m = map(int, input().split())
f = list(map(int, input().split()))
b = list(map(int, input().split()))

f_index = {}
for i, val in enumerate(f, start=1):
    if val not in f_index:
        f_index[val] = [i]
    else:
        f_index[val].append(i)

result = []
ambiguous = False
for val in b:
    if val not in f_index:
        print("Impossible")
        exit(0)
    elif len(f_index[val]) > 1:
        ambiguous = True
    else:
        result.append(f_index[val][0])

if ambiguous:
    print("Ambiguity")
else:
    print("Possible")
    print(" ".join(map(str, result)))
