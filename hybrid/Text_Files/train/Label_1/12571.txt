#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<unsigned> a(n);
    for (int i = 0; i < n; ++i) {
        string s;
        cin >> s;
        for (char c: s) {
            if (('0' <= c && c <= '9'))
                a[i] = a[i] * 16 + c - '0';
            else
                a[i] = a[i] * 16 + c - 'A' + 10;
        }
    }
    unsigned ans = 1;
    for (int x = 2; x * x <= n; ++x) {
        if (n % x == 0) {
            bool ok = true;
            unsigned t = 0;
            for (int i = 0; i < n; ++i) {
                for (int j = 0; j < n; ++j) {
                    bool v = a[i] & 1;
                    if (i % x > 0 || j % x > 0) v = t == v;
                    t = v;
                    a[i] >>= 1;
                    if (!v) ok = false;
                }
            }
            if (ok) ans = x;
        }
    }
    cout << ans << endl;
    return 0;
}
