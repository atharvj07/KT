import java.util.*;
import java.io.*;
class Main {
    static boolean[][] visited;
    static int[][] grid;
    static int n, m;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        try {
            n = in.nextInt();
            m = in.nextInt();
            in.nextLine();
            grid = new int[n][m];
            for (int i = 0; i < n; i++) {
                String line = in.nextLine();
                for (int j = 0; j < m; j++) {
                    if (line.charAt(j) == '#') {
                        grid[i][j] = 1;
                    }
                }
            }
            visited = new boolean[n][m];
            int ans = n * m;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (dfs(i, j, 1)) {
                        ans = 0;
                    }
                }
            }
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (dfs(i, j, 0)) {
                        ans = Math.min(ans, dfsCnt);
                    }
                }
            }
            System.out.println(ans);
        } catch (Exception e) {}
    }
    static int[][] dirs = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
    static int dfsCnt = 0;
    static boolean dfs(int x, int y, int c) {
        if (x < 0 || x >= n || y < 0 || y >= m || visited[x][y] || grid[x][y] == c) {
            return false;
        }
        visited[x][y] = true;
        dfsCnt++;
        boolean res = false;
        for (int[] dir : dirs) {
            res |= dfs(x + dir[0], y + dir[1], c);
        }
        return res;
    }
}
