import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, k = map(int, input().split())
    a = [*map(int, input().split())]
    b = a[:]
    b.sort()
    cnt = 0
    for i in b:
      if cnt >= i:
        break
      cnt += 1
    if cnt < k:
      print(-1)
    else:
      ans = 0
      for i in range(n-1):
        if a[i] != a[i+1]:
          ans += 1
      print(ans)
