#include<iostream>
#include<vector>
using namespace std;
int main()
{
    int n, m, k, a, b, c;
    cin >> n >> m >> k;
    vector<vector<int>> v(n + 1, vector<int>(m + 1, 0));
    for (int i = 1; i <= k; i++)
    {
        cin >> a >> b >> c;
        if (a == 1)
        {
            for (int j = 1; j <= m; j++)
            {
                v[b][j] = c;
            }
        }
        else
        {
            for (int j = 1; j <= n; j++)
            {
                v[j][b] = c;
            }
        }
    }
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= m; j++)
        {
            cout << v[i][j] << " ";
        }
        cout << endl;
    }
    return 0;
}
