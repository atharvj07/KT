#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
    unsigned n;
    cin >> n;
    vector<pair<unsigned, unsigned>> hor, ver;
    for (unsigned i = 0; i < n; ++i) {
        unsigned x, y, c;
        cin >> x >> y >> c;
        if (x > y)
            swap(x, y);
        if (x == 1)
            ver.push_back({y, c});
        else if (y == 1)
            hor.push_back({x, c});
    }
    sort(hor.begin(), hor.end());
    sort(ver.begin(), ver.end());
    unsigned ans = 0;
    for (unsigned i = 0, j = 0; i < hor.size(); ++i) {
        while (j < ver.size() && ver[j].first <= hor[i].first) j++;
        ans += hor[i].second * (j == ver.size() ? 0 : ver[j].second);
    }
    cout << ans << endl;
    return 0;
}
