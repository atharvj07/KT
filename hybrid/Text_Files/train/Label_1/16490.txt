#include <iostream>
#include <vector>
#include <set>
using namespace std;
typedef pair<int,int> par;
const int N=502;
int n,m,a[N][N],b[N][N];
set<par> vis;
bool valid(int x, int y, int size){
	if(x+size-1>=n||y+size-1>=m) return false;
	for(int i=0;i<size;++i){
		for(int j=0;j<size;++j){
			if(a[x+i][y+j]!=b[x+j][y+i]) return false;
		}
	}
	return true;
}
int main(){
	cin>>n>>m;
	for(int i=0;i<n;++i){
		for(int j=0;j<m;++j){
			cin>>a[i][j];
		}
	}
	for(int i=0;i<n;++i){
		for(int j=0;j<m;++j){
			cin>>b[i][j];
		}
	}
	for(int i=0;i<n;++i){
		for(int j=0;j<m;++j){
			for(int k=1;k<=min(n-i,m-j);++k){
				if(!valid(i,j,k)) continue;
				vis.insert({i,j});
				for(int x=0;x<k;++x){
					for(int y=0;y<k;++y){
						swap(a[i+x][j+y],a[i+y][j+x]);
					}
				}
				break;
			}
			if(vis.find({i,j})!=vis.end()){
				vis.erase({i,j});
			}else{
				cout<<"NO";
				return 0;
			}
		}
	}
	cout<<"YES";
}
