def round(x):
    if x > 0:
        return int(x + 0.5)
    else:
        return int(x - 0.5)

n = int(input())
s = [int(input()) for i in range(n)]

od = sum([i & 1 for i in s])
ev = sum([round(i / 2) for i in s])
s.sort(reverse=True)
for i in range(n):
    if od == 0:
        print(round(s[i] / 2))
    elif od > 0 and s[i] & 1 == 1:
        print(-round(s[i] / 2) - 1)
        od -= 1
    else:  # od < 0 and s[i] & 1 == 0:
        print(round(s[i] / 2) + 1)
        od += 1
