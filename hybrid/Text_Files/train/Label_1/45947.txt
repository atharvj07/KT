import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        pq.add(1);
        for (int i = 1; i < k; i++) {
            int n = pq.poll();
            System.out.println(n);
            int sn = sumDigits(n);
            pq.add(n * 10);
            pq.add(n * 10 + 1);
            pq.add(n * 10 + sn);
        }
        System.out.println(pq.poll());
    }

    static int sumDigits(int n) {
        int sum = 0;
        while (n > 0) {
            sum += n % 10;
            n /= 10;
        }
        return sum;
    }
}
