#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int lim = 1e5+1;
vector<ll> a;
int main() {
  int n, k; cin >> n >> k; a.resize(n);
  for (auto &x: a) cin >> x;
  ll ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      ll g = __gcd(a[i], a[j]);
      ll y = a[i] / g * a[j];
      ans += (ll)pow((double)y, 1.0 / k) == (ll)y / g;
    }
  }
  cout << ans << '\n';
  return 0;
}
