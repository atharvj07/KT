import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
c = list(map(int, input().split()))
p = [list(map(int, input().split())) for _ in range(n)]

ans = -1
for i in range(1<<n):
    if bin(i).count("1") == k:
        mn = 1e9
        for j in range(m):
            s = 0
            for l in range(n):
                if (i>>l)&1 and c[l]==0: s += p[l][j]
            mn = min(mn, s)
        ans = max(ans, mn)

print(ans)
