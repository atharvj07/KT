import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    List<Integer> a = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      a.add(sc.nextInt());
    }
    for (int d = -1000; d <= 1000; d++) {
      if (d == 0) continue;
      int count = 0;
      for (int x : a) {
        if (x / d > 0) count++;
      }
      if (count >= (n + 1) / 2) {
        System.out.println(d);
        return;
      }
    }
    System.out.println(0);
  }
}
