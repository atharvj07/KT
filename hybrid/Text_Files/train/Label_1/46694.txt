import java.util.Scanner;

public class Frames {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        char[][] grid = new char[n][m];

        for (int i = 0; i < n; i++) {
            String row = scanner.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = row.charAt(j);
            }
        }

        boolean hasFrames = false;
        int x1 = 0, y1 = 0, x2 = 0, y2 = 0;

        // Check for the first frame
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '#' && isFrame(grid, i, j)) {
                    hasFrames = true;
                    x1 = i;
                    y1 = j;
                    break;
                }
            }
            if (hasFrames) {
                break;
            }
        }

        // Check for the second frame
        if (hasFrames) {
            for (int i = x1 + 1; i < n; i++) {
                for (int j = y1 + 1; j < m; j++) {
                    if (grid[i][j] == '#' && isFrame(grid, i, j)) {
                        x2 = i;
                        y2 = j;
                        break;
                    }
                }
                if (x2 != 0 && y2 != 0) {
                    break;
                }
            }
        }

        // Output
        if (hasFrames && x2 != 0 && y2 != 0) {
            System.out.println("YES");
            System.out.println((x1 + 1) + " " + (y1 + 1) + " " + (x2 + 1) + " " + (y2 + 1));
        } else {
            System.out.println("NO");
        }
    }

    // Checks if a square is part of a frame
    private static boolean isFrame(char[][] grid, int x, int y) {
        if (x == 0 || x == grid.length - 1 || y == 0 || y == grid[0].length - 1) {
            return true;
        }

        if (grid[x - 1][y] == '.' && grid[x + 1][y] == '.' && grid[x][y - 1] == '.' && grid[x][y + 1] == '.') {
            return true;
        }

        return false;
    }
}
