n, m = map(int, input().split())
p = list(map(int, input().split()))

for _ in range(m):
    l, r, x = map(int, input().split())
    original_value = p[x - 1]
    count = 0
    for i in range(l - 1, r):
        if p[i] < original_value:
            count += 1
    if l + count == x:
        print("Yes")
    else:
        print("No")
