#include <iostream>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        while (k--) {
            long long f = 2;
            while (f * f <= n) {
                if (n % f == 0) {
                    n += f;
                    break;
                }
                f++;
            }
            if (f * f > n) {
                n += n;
            }
        }
        cout << n << endl;
    }
    return 0;
}
