#include <bits/stdc++.h>
using namespace std;
using i64 = long long;

i64 v[100005], u[100005], pre[100005], suf[100005];

int main() {
  i64 n, m;
  cin >> n;
  for (i64 i = 1; i <= n; i++) { cin >> v[i]; }
  cin >> m;
  vector<tuple<i64, i64, i64>> q(m);
  for (i64 i = 0; i < m; i++) {
    i64 t, l, r;
    cin >> t >> l >> r;
    q[i] = make_tuple(t, l, r);
  }
  sort(v + 1, v + n + 1);
  for (i64 i = 1; i <= n; i++) { u[i] = v[i]; }
  for (i64 i = 1; i <= n; i++) { pre[i] = pre[i - 1] + v[i]; }
  for (i64 i = n; i >= 1; i--) { suf[i] = suf[i + 1] + v[i]; }
  for (const auto &[t, l, r] : q) {
    if (t == 1)
      cout << pre[r] - pre[l - 1] << endl;
    else
      cout << suf[l] - suf[r + 1] << endl;
  }
  return 0;
}
