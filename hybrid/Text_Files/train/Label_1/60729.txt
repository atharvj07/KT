from collections import Counter, defaultdict

n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

MOD = 998244353

# Make a dictionary of the counts of each element in array a
a_count = Counter(a)

# Make a dictionary of the indices of each element in array b
b_idx = defaultdict(list)
for i, x in enumerate(b):
    b_idx[x].append(i)

# Check if it is possible to partition array a such that each subarray
# has a minimum equal to the corresponding element in array b
for x in b:
    if a_count[x] < len(b_idx[x]):
        print(0)
        exit()

# Initialize the dp array
dp = [0] * (n + 1)
dp[0] = 1

# Iterate over the elements of array a
for i in range(1, n + 1):
    # Iterate over the elements of array b
    for x in b:
        # Check if it is possible to partition array a such that the
        # minimum of the subarray ending at index i is equal to x
        if a[i - 1] >= x:
            # Update the dp array
            dp[i] = (dp[i] + dp[b_idx[x][-1]]) % MOD
    # Update the last index of element x in array b
    for x in b:
        if a[i - 1] == x:
            b_idx[x].pop()

# Print the dp array
print(dp[n])
