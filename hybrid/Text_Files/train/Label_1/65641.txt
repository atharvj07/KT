def is_ladder(subsegment):
    increasing = True
    for i in range(1, len(subsegment)):
        if subsegment[i] < subsegment[i - 1]:
            increasing = False
        elif not increasing and subsegment[i] > subsegment[i - 1]:
            return False
    return True

n, m = map(int, input().split())
a = list(map(int, input().split()))

for _ in range(m):
    l, r = map(int, input().split())
    if is_ladder(a[l-1:r]):
        print("Yes")
    else:
        print("No")
