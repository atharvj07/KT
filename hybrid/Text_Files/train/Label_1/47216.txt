import java.util.*;

public class Main {
    private static final long MOD = 1000000007L;
    private static int n;
    private static int m;
    private static int[] v;
    private static int[] s;
    private static int[] t;
    private static long[][] grf;

    private static void floydWarshall() {
        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                if (grf[i][k] == -1) {
                    continue;
                }
                for (int j = 0; j < n; j++) {
                    if (i == j || j == k || grf[k][j] == -1) {
                        continue;
                    }
                    if (grf[i][j] == -1) {
                        grf[i][j] = grf[i][k] + grf[k][j];
                    } else {
                        grf[i][j] = Math.min(grf[i][j], grf[i][k] + grf[k][j]);
                    }
                    grf[i][j] %= MOD;
                }
            }
        }
    }

    private static boolean isPossible(int idx) {
        return grf[v[idx] - 1][v[idx] - 1] != -1 && (s[idx] - grf[v[idx] - 1][v[idx] - 1]) % t[idx] == 0L;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        grf = new long[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(grf[i], -1);
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            int l = sc.nextInt();
            grf[a][b] = l;
        }
        floydWarshall();
        int q = sc.nextInt();
        v = new int[q];
        s = new int[q];
        t = new int[q];
        for (int i = 0; i < q; i++) {
            v[i] = sc.nextInt();
            s[i] = sc.nextInt();
            t[i] = sc.nextInt();
        }
        for (int i = 0; i < q; i++) {
            if (isPossible(i)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
