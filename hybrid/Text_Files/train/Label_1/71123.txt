import sys
input = sys.stdin.readline

def factor(n):
    i=2
    ans=[1]
    while i*i<=n:
        if n%i==0:
            ans.append(i)
            while n%i==0:
                n=int(n/i)
        i+=1
    if n!=1:
        ans.append(n)
    return ans

n=int(input())
f=factor(n)
ans=0
for x in f:
    ans=max(ans,x-1)
print(ans*2)
