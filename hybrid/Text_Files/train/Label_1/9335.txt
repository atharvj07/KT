#include<bits/stdc++.h>
using namespace std;
int main() {
  int q;
  cin >> q;
  while (q--) {
    string s, t;
    cin >> s >> t;
    int i = 0, j = 0;
    while (i < s.size() && j < t.size()) {
      if (s[i] == t[j]) {
        i++;
        j++;
      } else {
        i++;
      }
    }
    if (j == t.size())
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
