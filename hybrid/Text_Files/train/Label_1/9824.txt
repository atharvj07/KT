#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, h;
  cin >> n >> m >> h;
  vector<int> u(n);
  for (int &a : u) cin >> a;
  map<pair<int, int>, int> e;
  for (int i = 0; i < m; ++i) {
    int a, b;
    cin >> a >> b;
    e[{a, b}] = 1;
  }
  vector<bool> use(n, 0);
  for (auto &p : e) {
    if (u[p.first.first] == u[p.first.second]) {
      use[p.first.first] = true;
      use[p.first.second] = true;
    }
  }
  vector<int> ans;
  for (int i = 0; i < n; ++i) {
    if (!use[i]) ans.push_back(i + 1);
  }
  cout << ans.size() << endl;
  for (auto &a : ans) cout << a << " ";
}
