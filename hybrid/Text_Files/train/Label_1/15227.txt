#include<bits/stdc++.h>
#define int long long
using namespace std;
const int mod=1e9+7;
int32_t main(){
    int n;
    cin>>n;
    int a[n+5];
    for(int i=1;i<=n;i++) cin>>a[i];
    sort(a+1,a+n+1);
    int sum1=0,sum2=0,ans=0;
    for(int i=1;i<n;++i){
        sum1+=a[i];
        sum2+=a[n-i];
        if(sum1==sum2) ans+=2;
        else if((sum1+a[n-i+1])==(sum2+a[i])) ans+=2;
    }
    cout<<ans<<endl;
}
