#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    multiset<pair<int, int>> t;
    for (int i = 0; i < n; i++) {
        int x, a;
        cin >> x >> a;
        t.insert({x, a});
        t.insert({-x, a});
    }
    int ans = 0, pos = 0;
    for (auto it = t.begin(); it != t.end(); it++) {
        int x = it->first, a = it->second;
        ans += a;
        if ((pos > 0 && x > 0) || (pos < 0 && x < 0)) {
            break;
        }
        pos = x;
    }
    cout << ans << endl;
    return 0;
}
