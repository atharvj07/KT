import sys

for _ in range(int(input())):
    x, y = map(int, input().split())
    c = list(map(int, input().split()))
    print(min([
        3 * min(c[0], c[1]) + abs(x - 1) * min(c[0], c[5]),
        3 * min(c[0], c[2]) + abs(x + y) * min(c[0], c[2]),
        3 * min(c[0], c[3]) + abs(x - y) * min(c[0], c[3]),
        3 * min(c[0], c[4]) + abs(x) * min(c[0], c[4])
    ]))
