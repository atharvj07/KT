import sys

for case in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    t = 0; aa = bb = 0;
    while a:
        t += 1
        alice = sum(a[:]) - sum(a[1:])
        aa += alice
        a = a[1:] if alice > bb else []
        bob = sum(a) - sum(a[:-1])
        bb += bob
        a = a[:-1] if bob > aa else []
    print(t, aa, bb)
