#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    if (k == 0) {
        int ans = 0;
        for (int i = 0; i < n; i++) ans += (a[i] > 0);
        cout << ans << endl;
        for (int i = 0; i < n; i++) cout << a[i] << ' ';
        return 0;
    }
    int i = 0, j = 0, ans = 0, cnt = 0;
    while (j < n) {
        if (a[j] == 1) cnt++;
        while (cnt > k) {
            if (a[i] == 1) cnt--;
            i++;
        }
        ans = max(ans, j - i + 1);
        j++;
    }
    cout << ans << endl;
    i = 0, j = 0, cnt = 0;
    while (j < n) {
        if (a[j] == 1) cnt++;
        while (cnt > k) {
            if (a[i] == 1) cnt--;
            a[i++] = 0;
        }
        a[j++] = 1;
    }
    for (int i = 0; i < n; i++) cout << a[i] << ' ';
    return 0;
}
