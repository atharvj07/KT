n = int(input())

max_width = 0
max_height = 0

for _ in range(n):
    query = input().split()
    if query[0] == '+':
        x, y = int(query[1]), int(query[2])
        max_width = max(max_width, min(x, y))
        max_height = max(max_height, max(x, y))
    else:
        h, w = int(query[1]), int(query[2])
        if max_width <= min(h, w) and max_height <= max(h, w):
            print("YES")
        else:
            print("NO")
