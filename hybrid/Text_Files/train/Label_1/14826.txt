#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<string> chat_list;
    unordered_map<string, int> chat_idx;
    for (int i = 0; i < n; i++) {
        string name;
        cin >> name;
        if (chat_idx.find(name) == chat_idx.end()) {
            chat_list.push_back(name);
            chat_idx[name] = chat_list.size() - 1;
        } else {
            int idx = chat_idx[name];
            if (idx != 0) {
                swap(chat_list[idx], chat_list[idx - 1]);
                chat_idx[chat_list[idx]] = idx - 1;
                chat_idx[chat_list[idx - 1]] = idx;
            }
        }
    }
    for (string name : chat_list) {
        cout << name << endl;
    }
    return 0;
}
