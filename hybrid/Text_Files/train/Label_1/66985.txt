import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n, T = map(int, input().split())
    a = list(map(int, input().split()))
    b = [a[i] for i in range(n) if i%2==0]
    c = [a[i] for i in range(n) if i%2==1]
    b.sort(reverse=True)
    c.sort(reverse=True)
    res = []
    for i in range(len(b)):
        if b[i]+c[i]<=T:
            res.append(0)
        else:
            res.append(1)
            c[i]-=T-b[i]
    print(*res)
