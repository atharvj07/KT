import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = [i for i in range(1, n + 1) if i not in a]
op = 0
for i in c:
    if i == b[-1]:
        op += 1
        b.pop()
        continue
    if i in a:
        for j in range(len(a)):
            if a[j] == i:
                a[j] = b[-1]
                op += 1
                b.pop()
                break
    else:
        for j in range(len(b)):
            if b[j] == i:
                op += len(b[j:])
                b = b[:j]
                break
print(op)
