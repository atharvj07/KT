#include <iostream>
#include <vector>

int days[13] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
int leap_days[13] = {0, 31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

int is_leap(int y) {
    return y % 4 == 0 && (y % 100 != 0 || y % 400 == 0);
}

int main() {
    int y1, m1, d1, y2, m2, d2;
    char c;
    std::cin >> y1 >> c >> m1 >> c >> d1 >> y2 >> c >> m2 >> c >> d2;
    int res = 0;
    for (int i = y1; i < y2; ++i) {
        res += is_leap(i) ? 366 : 365;
    }
    for (int i = 1; i < m1; ++i) {
        res -= days[i];
    }
    res -= d1;
    for (int i = 1; i < m2; ++i) {
        res += is_leap(y2) ? leap_days[i] : days[i];
    }
    res += d2;
    std::cout << res << '\n';
    return 0;
}
