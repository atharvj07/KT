#include <iostream>

using namespace std;

int main() {
    int w, h, u1, d1, u2, d2, wt = 0;
    cin >> w >> h;
    cin >> u1 >> d1;
    cin >> u2 >> d2;
    while (h > 0) {
        wt += h;
        if (h == d1) wt -= u1;
        if (h == d2) wt -= u2;
        if (wt < 0) wt = 0;
        h--;
    }
    cout << wt << endl;
    return 0;
}
