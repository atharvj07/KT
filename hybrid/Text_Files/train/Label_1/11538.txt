#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e3 + 5;
#define ll long long
const ll inf = 1e18;
ll a[maxn], sm[200001], ans;
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        sm[a[i]]++;
    }
    for (int i = 0; i < 100000; i++) {
        ans = max(ans, min(sm[i], sm[i + 1]));
    }
    cout << ans << endl;
}
