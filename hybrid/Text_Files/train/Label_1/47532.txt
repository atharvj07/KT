import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        for(int t=0; t<T; t++){
            int n = sc.nextInt();
            char[] r = sc.next().toCharArray();
            char[] b = sc.next().toCharArray();
            int res = 0;
            for(int i=0; i<n; i++){
                if(r[i] > b[i]) res++;
                else if(r[i] < b[i]) res--;
            }
            if(res > 0) System.out.println("RED");
            else if(res < 0) System.out.println("BLUE");
            else System.out.println("EQUAL");
        }
    }
}
