import java.util.Scanner;
import java.util.List;
import java.util.ArrayList;
import java.util.Map;
import java.util.HashMap;
import java.util.Set;
import java.util.HashSet;

@SuppressWarnings("unchecked")
public class Main {
    static char[][] grid;
    static int[] dx = { 0, 0, -1, 1 };
    static int[] dy = { 1, -1, 0, 0 };
    static int m, n;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        m = sc.nextInt();
        n = sc.nextInt();
        grid = new char[m + 2][n + 2];
        for (int i = 1; i <= m; i++) {
            String s = sc.next();
            for (int j = 1; j <= n; j++)
                grid[i][j] = s.charAt(j - 1);
        }

        int ans = 0;
        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= n; j++) {
                if (grid[i][j] == '.') {
                    int cnt = 0;
                    Map<Character, Integer> color2idx = new HashMap<>();
                    int[] rowIdxes = new int[4];
                    int[] colIdxes = new int[4];
                    Set<Integer> ids = new HashSet<>();
                    for (int d = 0; d < 4; d++) {
                        int row = i, col = j;
                        int idx = -1;
                        while (row <= m && row >= 1 && col <= n && col >= 1) {
                            if (grid[row][col] != '.') {
                                if (!ids.contains(idx)) {
                                    if (color2idx.containsKey(grid[row][col])) {
                                        idx = color2idx.get(grid[row][col]);
                                    } else {
                                        idx = color2idx.size();
                                        color2idx.put(grid[row][col], idx);
                                        ids.add(idx);
                                    }
                                }
                                rowIdxes[idx] = row;
                                colIdxes[idx] = col;
                                cnt++;
                            }
                            row += dx[d];
                            col += dy[d];
                        }
                    }

                    if (cnt <= 1)
                        continue;
                    ans += 2 * (cnt - List.of(color2idx.values()).stream().distinct().count());
                }
            }
        }

        System.out.println(ans);
    }
}
