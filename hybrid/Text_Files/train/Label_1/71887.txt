import sys

gems = {"purple": "Power", "green": "Time", "blue": "Space", "orange": "Soul", "red": "Reality", "yellow": "Mind"}

n = int(input())
seen = set()
for _ in range(n):
    color = input()
    seen.add(color)

answer = []
for color in gems:
    if color not in seen:
        answer.append(gems[color])

print(len(answer))
print("\n".join(answer))
