n,l,r=map(int,input().split())
a=list(map(int,input().split()))
p=list(map(int,input().split()))
b=[0 for i in range(n)]
for i in range(n-1,-1,-1):
    lo=l;hi=r
    while lo<=hi:
        mid=(lo+hi)//2
        cur=sum([1 for j in range(i+1,n) if a[j]-mid<=a[i]-b[j]])
        if cur==p[i]-1:
            b[i]=mid
            break
        if cur<p[i]-1:
            lo=mid+1
        else:
            hi=mid-1
    if b[i]==0:
        print(-1)
        exit()
print(*b)
