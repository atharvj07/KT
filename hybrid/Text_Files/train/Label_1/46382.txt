import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();

        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }

        sc.close();

        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (h[i] > 0) {
                long tmp = h[i] - 1 + b;
                ans += tmp / a;
                if (tmp % a != 0) {
                    ans++;
                }
            }
        }

        System.out.println(ans);
    }
}
