import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader bfr = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(bfr.readLine());
        int[] a = Arrays.stream(bfr.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int m = Integer.parseInt(bfr.readLine());
        int[] b = Arrays.stream(bfr.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int g = gcd(a[i], b[j]);
                if (a[i] / g == b[j] / g) ans++;
            }
        }
        System.out.println(ans);
    }

    private static int gcd(int a, int b) {
        if (a == 0) return b;
        return gcd(b % a, a);
    }
}
