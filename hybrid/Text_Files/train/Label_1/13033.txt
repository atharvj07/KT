#include <bits/stdc++.h>
using namespace std;

int a[2005][2005];
int b[2005];
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      char c;
      cin >> c;
      a[i][j] = c - '0';
    }
  }
  for (int i = 1; i <= m; i++) {
    bool f = false;
    for (int j = 1; j <= n; j++) {
      if (a[j][i]) {
        b[j] = 1;
        f = true;
      }
    }
    if (!f) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
