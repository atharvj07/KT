import sys
n, q = map(int, sys.stdin.readline().split())
a = [0] * (n+1)
for i in range(q):
    t, x = map(int, sys.stdin.readline().split())
    if t == 1:
        a[x] += 1
    elif t == 2:
        a[x] = 0
    else:
        for j in range(1, n+1):
            a[j] -= min(x, a[j])
    print(sum(a))
