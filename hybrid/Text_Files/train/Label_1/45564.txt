import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextInt();
        long b = in.nextInt();
        long f = in.nextInt();
        int k = in.nextInt();
        long x = Math.min(f, a - f);
        if (2 * x <= b) {
            System.out.println(k / 2);
        } else if (x + f <= b) {
            System.out.println(k / 2 + k % 2);
        } else {
            System.out.println(-1);
        }
    }
}
