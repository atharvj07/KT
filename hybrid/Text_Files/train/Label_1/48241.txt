import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] a = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        int max = 0;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum = 0;
            for (int j = 0; j < n; j++) {
                sum += a[i][j];
                if (max < sum) {
                    max = sum;
                }
                if (sum < 0) {
                    sum = 0;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            sum = 0;
            for (int j = 0; j < n; j++) {
                sum += a[j][i];
                if (max < sum) {
                    max = sum;
                }
                if (sum < 0) {
                    sum = 0;
                }
            }
        }
        System.out.println(max);
    }
}
