#include <bits/stdc++.h>

using namespace std;

const int MAXN = 5000;
const int MAXM = 100;
const int INF = 1e9;

int n, m;
vector<int> c;
vector<int> p;
vector<int> color;
vector<pair<int, int>> ans;

int main() {
  cin >> n >> m;
  c.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> c[i];
    c[i]--;
  }
  p.resize(m);
  for (int i = 0; i < m; i++) {
    p[i] = i;
  }
  while (p.size() > 1) {
    int u = p.back();
    p.pop_back();
    int v = p.back();
    p.pop_back();
    ans.emplace_back(u, v);
  }
  color.resize(m);
  int cnt = 0;
  for (auto &x : ans) {
    if (color[x.first] == 0 && color[x.second] == 0) {
      color[x.first] = ++cnt;
      color[x.second] = ++cnt;
    } else if (color[x.first] == 0) {
      color[x.first] = color[x.second];
    } else if (color[x.second] == 0) {
      color[x.second] = color[x.first];
    }
  }
  cout << cnt << endl;
  for (int i = 0; i < n; i++) {
    cout << color[c[i]] << " " << (color[c[i]] == color[c[(i + 1) % n]] ? color[c[i]] + cnt : color[c[(i + 1) % n]]) << endl;
  }
  return 0;
}
