n, m = map(int, input().split())
a = list(map(int, input().split()))
b = [0] * m
for ai in a:
    b[ai-1] += 1
mn = min(b)
c = 0
for ai in a:
    if b[ai-1] > mn:
        c += 1
        b[ai-1] -= 1
        b[mn-1] += 1
print(mn, c)
for ai in a:
    if b[ai-1] == mn:
        print(mn, end=" ")
    else:
        print(ai, end=" ")
