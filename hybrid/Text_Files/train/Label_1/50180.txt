import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] even = new int[n / 2];
        for (int i = 0; i < n / 2; i++) {
            even[i] = in.nextInt();
        }

        int[] odd = new int[n / 2];
        BigInteger sum = BigInteger.ZERO;
        for (int i = 0; i < n / 2; i++) {
            odd[i] = even[i] - (int) sum.sqrt().longValue();
            sum = sum.add(BigInteger.valueOf(odd[i]));
        }

        if (sum.sqrt().pow(2).equals(sum)) {
            System.out.println("Yes");
            for (int i = 0; i < n; i++) {
                System.out.print(i % 2 == 0 ? even[i / 2] : odd[i / 2] + " ");
            }
            System.out.println();
        } else {
            System.out.println("No");
        }
    }
}
