n = int(input().strip())
r = list(map(int, input().strip().split()))

# Find the minimum rating
min_rating = min(r)

# Strategy to maximize the final rating
matches = []
for i in range(n):
    difference = r[i] - min_rating
    while difference > 0:
        party = ['0'] * n
        # Form a party of up to 5 members starting from the current friend
        for j in range(i, min(i + 5, n)):
            if r[j] > min_rating:
                party[j] = '1'
                r[j] -= 1
            if sum(int(x) for x in party) == 5:
                break
        matches.append(''.join(party))
        difference -= 1

# Calculate the number of matches
t = len(matches)

# Output the results
print(min_rating)
print(t)
for match in matches:
    print(match)
