#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    unordered_set<string> s;
    while (n--) {
        string str;
        cin >> str;
        if (s.count(str))
            cout << "YES\n";
        else
            cout << "NO\n", s.insert(str);
    }
    return 0;
}
