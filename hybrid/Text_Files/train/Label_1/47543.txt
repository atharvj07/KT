import java.util.Scanner;
import java.util.Set;
import java.util.HashSet;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Set<String> hard = new HashSet<>();
        for (int i = 0; i < n; i++) {
            String name = sc.next();
            String type = sc.next();
            if (type.equals("hard")) {
                hard.add(name);
            }
        }
        System.out.println(hard.size());
    }
}
