import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }

        int[] pos = new int[n];
        for (int i = 0; i < n; i++) {
            pos[p[i] - 1] = i;
        }

        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if ((pos[j] - pos[i]) % 2 == 1) {
                    ans++;
                }
            }
        }

        System.out.println(ans);
    }
}
