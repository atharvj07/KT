#include <bits/stdc++.h>
#define ll long long
using namespace std;

int main() {
    ll a; cin >> a;
    ll ans = 0;
    while (a > 0) {
        if (a % 10 == 0) {
            a /= 10;
            continue;
        }
        ans++;
        a = a / 10 - a % 10;
    }
    cout << ans << endl;
    return 0;
}
