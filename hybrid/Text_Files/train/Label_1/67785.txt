import sys

rl = lambda: sys.stdin.readline().strip()
wl = lambda x: sys.stdout.write(x + '\n')

def lis(a):
    n = len(a)
    dp = [1] * n
    for i in range(n):
        for j in range(i):
            if a[i] > a[j]:
                dp[i] = max(dp[i], dp[j] + 1)
    return dp

for _ in range(int(rl())):
    rl()
    a = list(map(int, rl().split()))
    dp = lis(a)
    wl('YES' if sum(dp) & 1 == 0 else 'NO')
