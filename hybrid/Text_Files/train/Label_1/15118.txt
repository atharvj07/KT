#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x, y, ans = 0;
  cin >> n >> x >> y;
  vector<int> a(n);
  for (int &i : a) cin >> i;
  sort(a.begin(), a.end());
  for (int i : a) {
    if (i <= x) {
      ++ans;
    } else if (i - x <= y) {
      ++ans;
      i -= y;
    }
    if (i - x > y) break;
  }
  cout << ans << '\n';
}
