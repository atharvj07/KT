#include <bits/stdc++.h>
using namespace std;

int n;
string s;

int main() {
    cin >> n;
    while (n--) {
        cin >> s;
        stack<char> st;
        int ans = 0;
        for (int i = 0; i < s.size(); ++i) {
            if (s[i] == '(' || s[i] == '[') st.push(s[i]);
            else {
                if (!st.empty() && (st.top() == '(' && s[i] == ')' || st.top() == '[' && s[i] == ']')) {
                    st.pop();
                    if (st.empty()) ans++;
                } else {
                    st.push(s[i]);
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
