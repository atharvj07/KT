import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int t = sc.nextInt();
        long[] ans = new long[t];
        for (int i = 0; i < t; i++) {
            int xi = sc.nextInt();
            int yi = sc.nextInt();
            int idx = Math.abs(xi + yi);
            if (idx < t) {
                ans[idx]++;
            }
        }
        for (int i = 0; i < t; i++) {
            if (i > n) {
                ans[i] += (i - n) / 4;
            }
            ans[i] += (ans[i] + 3) / 4;
            System.out.println(ans[i]);
        }
    }
}
