#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> ans;
    for (int i = 1; i <= n; i += 2 * k + 1) ans.push_back(i);
    for (int i = n - k; i >= 1; i -= 2 * k + 1) ans.push_back(i);
    cout << ans.size() << endl;
    for (auto &x : ans) cout << x << ' ';
    return 0;
}
