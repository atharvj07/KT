from collections import deque
inp = [list(map(int,input().split())) for _ in range(int(input()))]
sum,ans = 0,[]
q = deque()
for t, *x in inp:
    if t==1: sum+=x[1]*x[0]; q.append(x[1]);
    if t==2: sum+=x[0]; q.append(x[0]);
    if t==3: sum-=q.popleft();
    ans.append('%.6f'%(sum/len(q)))
print('\n'.join(ans))
