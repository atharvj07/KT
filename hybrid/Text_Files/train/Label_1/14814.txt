#include <bits/stdc++.h>
using namespace std;

char s[1<<10][1<<10];
int k,d;
void play(int x,int y,int b){
  if(x==d){ for(int j=0;j<d;++j) s[b][j]="+"[y>>j&1]; return;}
  play(x+1,y,b);
  play(x+1,(~1ULL)&y<<1,b|(1<<(x+1)));
  play(x+1,(~~2ULL)&y<<1,b|(1<<x)|(1<<(x+1)));
}

int main(){
  cin>>k,d=1<<k;
  play(0,0,0);
  for(int i=0;i<d;++i) cout<<s[i]<<endl;
}
