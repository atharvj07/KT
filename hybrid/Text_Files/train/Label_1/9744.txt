#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 1; i < n; i++) {
        if (a[0] <= a[i]) ans += a[i] - a[0] + 1;
    }
    cout << ans << endl;
    return 0;
}
