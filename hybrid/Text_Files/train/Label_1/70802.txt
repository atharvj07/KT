for t in range(int(input())):
    n = int(input())
    s = input()
    if '?' not in s:
      print(s)
      continue
    cnt = 0
    ans = ''
    for i in s:
      if i == '?':
        if cnt % 2 == 0:
          ans += 'R'
          cnt += 1
        else:
          ans += 'B'
          cnt += 1
      else:
        ans += i
        if i == 'R':
          cnt += 1
    print(ans)
