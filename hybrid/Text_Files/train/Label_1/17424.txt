#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll N = 2e5 + 9, INF = 1e18 + 9;

ll a[N], ans[N];
int main() {
  ll n, m, x, p, k;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  while (m--) {
    cin >> x >> p;
    if (x == 1) {
      cin >> k;
      for (int i = p; i <= n; i++) {
        if (ans[i] + k <= a[i])
          ans[i] += k;
        else {
          k = k - (a[i] - ans[i]);
          ans[i] = a[i];
        }
      }
    } else {
      cout << ans[p] << endl;
    }
  }
  return 0;
}
