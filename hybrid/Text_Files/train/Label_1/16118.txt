#include <iostream>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int c, d;
    std::cin >> c >> d;
    if (c > d) std::swap(c, d);
    if ((c + d) % 2 == 0 && (d - c) % 2 == 0)
      std::cout << (d - c) / 2 << std::endl;
    else
      std::cout << -1 << std::endl;
  }
  return 0;
}
