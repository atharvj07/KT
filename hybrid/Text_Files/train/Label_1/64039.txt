for _ in range(int(input())):
    n = int(input())
    a = tuple(map(int, input().split()))
    m = int(input())
    b = []
    for _ in range(m):
        p, s = map(int, input().split())
        if all(p < j for j in a):
            b += [s] * n
        elif p >= min(a):
            b += [s]
        else:
            b += [0]
    print(-1 if max(b) < n else max(b))
