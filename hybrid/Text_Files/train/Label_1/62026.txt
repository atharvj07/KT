import sys

t = int(input())
for _ in range(t):
    a, b = map(int, input().split())
    s = input()
    n = len(s)

    dp = [1000000000] * (n + 1)
    dp[n] = 0
    for i in range(n - 1, -1, -1):
        if s[i] == '0':
            dp[i] = dp[i + 1]
        else:
            dp[i] = min(dp[i + 1] + a, dp[max(0, i - 1)] + b)
    print(dp[0])
