#include<bits/stdc++.h>
const long long mod = 1e9 + 7;
long long g(long long x, long long p) {
    while (x % p == 0) x /= p;
    return x;
}
long long f(long long x, long long y) {
    long long ans = 1;
    for (long long p : {2, 3, 5, 7}) {
        long long temp = y; 
        while (temp % p == 0) temp /= p;
        if (x % p != 0) temp = 1;
        ans = ans * temp % mod;
    }
    return ans;
}
int main() {
    long long x, n;
    std::cin >> x >> n;
    long long ans = 1;
    for (long long i = 1; i <= n; i++) { ans = ans * f(x, i) % mod; }
    std::cout << ans << std::endl;
    return 0;
}
