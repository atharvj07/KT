import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] p = new int[n];
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
            t[i] = sc.nextInt();
        }
        List<Integer> groups[] = new List[51];
        for (int i = 0; i <= 50; i++) {
            groups[i] = new ArrayList<Integer>();
        }
        for (int i = 0; i < n; i++) {
            groups[p[i]].add(t[i]);
        }
        int ans = 0;
        int cur = 1;
        for (int i = 50; i >= 1; i--) {
            groups[i].sort(null);
            for (int j = 0; j < groups[i].size(); j++) {
                if (cur == k) ans++;
                cur++;
            }
            if (!groups[i].isEmpty()) cur--;
        }
        System.out.println(ans);
    }
}
