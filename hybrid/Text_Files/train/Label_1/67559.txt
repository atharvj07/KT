import sys
input = sys.stdin.readline

n = int(input())
minx, miny, maxx, maxy = 1e9, 1e9, -1e9, -1e9
for _ in range(n):
    x1, y1, x2, y2 = map(int, input().split())
    minx = min([minx, x1, x2])
    miny = min([miny, y1, y2])
    maxx = max([maxx, x1, x2])
    maxy = max([maxy, y1, y2])
x, y = (minx + maxx) // 2, (miny + maxy) // 2
print(x, y)
