import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        int cur = 0, ans = 0;
        for (int i = 0; i <= n; i++) {
            if (i > 0) cur -= a[i - 1];
            while (cur < i * k && i < n) {
                cur += a[i];
                i++;
            }
            if (cur == i * k && i < n) ans++;
        }
        System.out.println(ans);
    }
}
