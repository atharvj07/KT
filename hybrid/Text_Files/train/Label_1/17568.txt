#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
typedef vector<ll> vll;
#define FOR(i, a, b) for(ll i = ll(a); i <= ll(b); i++)
#define ROF(i, a, b) for(ll i = ll(a); i >= ll(b); i--)
#define inf 1e9 + 7
#define endl '\n'

ll n, a[200010], b[200010], c[200010], pref[200010], suf[200010];
pair<ll, ll> ans[200010];

int main() {
    ll t, x, l1, l2, k;
    cin >> t;
    while(t--) {
        cin >> n;
        FOR(i, 1, n) {
            cin >> a[i];
            c[a[i]] = i;
            pref[i] = pref[i - 1];
            if(i == c[i]) pref[i]++;
        }
        for(ll i = n; i >= 1; i--) {
            suf[i] = suf[i + 1];
            if(i == c[a[i]]) suf[i]++;
        }
        x = 0, k = 0;
        FOR(i, 1, n) {
            if(c[i] < c[x]) x = i;
        }
        l1 = pref[c[x] - 1];
        l2 = suf[c[x] + 1];
        b[1] = x;
        FOR(i, 2, l1) {
            l2 += (c[x] < c[a[c[x] - i]]);
            b[i] = a[c[x] - i];
        }
        FOR(i, 1, n - l1) {
            x = a[c[x] + i];
            b[++l1] = x;
        }
        if(l1 != n) {
            cout << 0 << endl;
        }
        else {
            k = 0;
            FOR(i, 1, n) {
                l2 -= (c[a[i]] > c[x]);
                if(l2 == 0) ans[k++] = {i, n - i};
                x = a[i];
            }
            cout << k << endl;
            FOR(i, 0, k - 1) cout << ans[i].first << " " << ans[i].second << endl;
        }
    }
    return 0;
}
