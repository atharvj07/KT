import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            String s = sc.next();
            String t = sc.next();
            boolean equ = true;
            for (int j = 0; j < n; j++) {
                if (s.charAt(j) != t.charAt(j)) {
                    equ = false;
                    break;
                }
            }
            if (equ) System.out.println("YES");
            else System.out.println("NO");
        }
    }
}
