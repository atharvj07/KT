#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> gold;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'G') gold.push_back(i);
    }
    int ans = 0;
    for (int i = 0; i < gold.size(); i++) {
        int l = i - 1 >= 0 ? gold[i] - gold[i - 1] : gold[i] + 1;
        int r = i + 1 < gold.size() ? gold[i + 1] - gold[i] : n - gold[i];
        ans = max(ans, l + r + 1);
    }
    cout << ans << endl;
    return 0;
}
