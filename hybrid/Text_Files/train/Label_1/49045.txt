import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long max = -1;
        for (int i = 0; i < n; i++) {
            long x = in.nextInt();
            if (Math.sqrt(x) != (int) Math.sqrt(x)) {
                max = Math.max(max, x);
            }
        }
        System.out.println(max);
    }
}
