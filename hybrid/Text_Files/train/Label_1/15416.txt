#include <bits/stdc++.h>

using namespace std;
#define ll long long
#define DEBUG(x) cerr << #x << ": " << (x) << "\n";
#define REP(i,n) for (int i=0;i<(n);i++)
const ll MOD = 1e9 + 9;

struct Cube {
    ll x;
    ll y;
    ll val;
    bool operator<(const Cube& other) const {
        return other.val < val;
    }
};

int main() {
    ll m;
    cin >> m;
    vector<Cube> cubes(m);
    REP(i, m) {
        ll x, y;
        cin >> x >> y;
        cubes[i] = {x, y, i};
    }
    sort(cubes.begin(), cubes.end());
    ll ans = 0;
    ll p = 1;
    for (auto& cube : cubes) {
        ans += p * cube.val;
        p *= m;
        p %= MOD;
        ans %= MOD;
    }
    cout << ans << endl;
    return 0;
}
