#include <bits/stdc++.h>
using namespace std;
int btns[103], lights[103], res[103];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) cin >> btns[i];
    for (int i = m - 1; i >= 0; i--) {
        int btn = btns[i];
        for (int light = btn; light <= n; light++) {
            if (lights[light] == 0) {
                res[light] = btn;
                lights[light] = 1;
            }
        }
    }
    for (int i = 1; i <= n; i++) cout << res[i] << " ";
    cout << endl;
}
