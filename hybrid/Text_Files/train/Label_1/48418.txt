import java.util.*;
public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), t = in.nextInt();
    Map<Long, Integer> m = new HashMap<>();
    m.put(0l, n);
    for (int i = 1; i <= t; i++) {
      long p = in.nextLong() * 1000 + in.nextLong();
      System.out.println(m.getOrDefault(p, 0));
      for (long j = 1; j <= 4; j++) {
        long q = p + j * 1000;
        m.put(q, m.getOrDefault(q, 0) + m.getOrDefault(p, 0) / 4);
      }
    }
  }
}
