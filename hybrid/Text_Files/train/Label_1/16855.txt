#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 100;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<char>> field(n, vector<char>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> field[i][j];
        }
    }

    bool valid = true;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            int num_bombs = 0;
            if (field[i][j] == '.') {
                for (int di = -1; di <= 1; di++) {
                    for (int dj = -1; dj <= 1; dj++) {
                        if (i + di >= 0 && i + di < n && j + dj >= 0 && j + dj < m && field[i + di][j + dj] == '*') {
                            num_bombs++;
                        }
                    }
                }
                if (num_bombs > 0) {
                    valid = false;
                    break;
                }
            } else if (isdigit(field[i][j])) {
                int num = field[i][j] - '0';
                for (int di = -1; di <= 1; di++) {
                    for (int dj = -1; dj <= 1; dj++) {
                        if (i + di >= 0 && i + di < n && j + dj >= 0 && j + dj < m && field[i + di][j + dj] == '*') {
                            num_bombs++;
                        }
                    }
                }
                if (num_bombs != num) {
                    valid = false;
                    break;
                }
            }
        }
    }

    if (valid) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }

    return 0;
}
