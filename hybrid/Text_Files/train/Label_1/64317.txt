import sys

n,m=map(int,input().split())
a=[list(map(int,input().split()))for i in range(m)]
m=len(a)
ans=[-1]*n
for sta in range(n):
    now=sta+1
    use=[0]*m
    tim=0
    for p in a:
        if use[p[0]-1]==0:
            tim+=abs(now-p[0])
            now=p[0]
            use[p[0]-1]=1
        tim+=abs(now-p[1])
        now=p[1]
    ans[sta]=tim
print(*ans)
