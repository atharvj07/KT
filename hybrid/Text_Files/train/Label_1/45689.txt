import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner scanner = new Scanner(System.in);
		int n = scanner.nextInt();
		int[] a = new int[n];
		int max = 0;
		for (int i = 0; i < n; i++) {
			a[i] = scanner.nextInt();
			max = Math.max(max, a[i]);
		}
		for (int i = max; i < 101; i++) {
			long sum = 0;
			for (int j = 0; j < n; j++) {
				sum += i - a[j];
			}
			if (sum > 0) {
				System.out.println(i);
				break;
			}
		}
	}
}
