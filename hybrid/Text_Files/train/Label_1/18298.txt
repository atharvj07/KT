#include <bits/stdc++.h>
int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<std::string> f(n);
  for (std::string &s : f)
    std::cin >> s;
  int ans = 0;
  for (int i = 0; i < n - 1; ++i)
    for (int j = 0; j < m - 1; ++j)
      if (f[i][j] == 'f' && f[i + 1][j] == 'a' && f[i][j + 1] == 'c' &&
          f[i + 1][j + 1] == 'e')
        ans++;
  std::cout << ans << std::endl;
}
