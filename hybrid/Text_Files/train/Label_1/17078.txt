#include <bits/stdc++.h>

#define MOD 1000000007
#define MAX_N 16

using namespace std;

long long int dp[MAX_N][1 << MAX_N];

long long int get_count(int n, int mask) {
  if (n == 0) {
    return 1;
  }
  if (dp[n][mask] != -1) {
    return dp[n][mask];
  }
  long long int count = 0;
  for (int i = 1; i <= n; i++) {
    if ((mask & (1 << (i - 1))) == 0) {
      count = (count + get_count(n - 1, mask | (1 << (i - 1)))) % MOD;
    }
  }
  return dp[n][mask] = count;
}

int main() {
  int n;
  cin >> n;

  memset(dp, -1, sizeof(dp));

  long long int ans = (get_count(n, 0) * get_count(n, 0)) % MOD;

  cout << ans << endl;

  return 0;
}
