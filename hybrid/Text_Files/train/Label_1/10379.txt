#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
#define fastIO ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
typedef long long int ll;
const ll inf = 1e18;

int main() {
    fastIO;
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = -1;
    int sum = 0;
    for (int i : a) sum += i;
    for (int k = 2; k <= sum; k++) {
        int cur = 0;
        for (int i : a) cur += i % k == 0;
        if (cur == n) {
            ans = 0;
            break;
        } else if (cur > 0) {
            ans = 1;
            break;
        }
    }
    if (sum == 0) ans = -1;
    cout << ans << endl;
    return 0;
}
