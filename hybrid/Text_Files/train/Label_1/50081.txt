import java.util.*;

public class MyonCoins {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long[] A = new long[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextLong();
        }

        // Find the maximum value among the prices
        long maxPrice = 0;
        for (long price : A) {
            maxPrice = Math.max(maxPrice, price);
        }

        // Find the minimum number of coins needed to pay the maximum price
        int minCoins = 0;
        while (maxPrice > 0) {
            // Find the largest coin that is less than or equal to the maximum price
            long coinValue = 1;
            while (coinValue * 10 <= maxPrice) {
                coinValue *= 10;
            }

            // Add the coin to the minimum coins count
            minCoins++;

            // Subtract the coin value from the maximum price
            maxPrice -= coinValue;
        }

        System.out.println(minCoins);
    }
}
