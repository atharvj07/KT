import sys

n = int(input())
a = []
b = []
for i in range(n):
    a.append(int(input()))
for i in range(n):
    b.append(int(input()))
c = 0
ans = -1
for i in range(n):
    while a[i]:
        t = min(b[ans + 1], a[i])
        b[ans + 1] -= t
        a[i] -= t
        c += t
        if not b[ans + 1]: ans += 1
print(ans + 1, c)
