#include <bits/stdc++.h>
#define int long long
using namespace std;

int32_t main() {
  int n, c1, c2;
  cin >> n >> c1 >> c2;
  string s;
  cin >> s;
  int a = 0, p = 0;
  for (auto e : s) { a += (e == '1'); }
  int dp[n + 1][a + 1];
  for (int i = 0; i <= n; i++) {
    for (int j = 0; j <= a; j++) { dp[i][j] = 1e18; }
  }
  dp[0][0] = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= a; j++) {
      dp[i][j] = min(dp[i][j], dp[i - 1][j]);
      if (j > 0 or s[i - 1] == '1') { dp[i][j] = min(dp[i][j], dp[i - 1][j - 1] + c1); }
      if (j > 0) { dp[i][j] = min(dp[i][j], dp[i - 1][j] + c1 + c2); }
    }
  }
  cout << dp[n][a] << endl;
}
