import sys
input = sys.stdin.readline

R, B = map(int, input().split())
rx = []
ry = []
bx = []
by = []
for i in range(R):
    x, y = map(int, input().split())
    rx.append(x)
    ry.append(y)
for i in range(B):
    x, y = map(int, input().split())
    bx.append(x)
    by.append(y)
ok = False
for i in range(R):
    for j in range(B):
        if i == j:
            continue
        if (rx[i] - rx[j]) * (by[i] - by[j]) == (ry[i] - ry[j]) * (bx[i] - bx[j]):
            ok = True
            break
    if ok:
        break
print('Yes' if not ok else 'No')
