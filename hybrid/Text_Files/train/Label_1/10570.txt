#include <bits/stdc++.h>
using namespace std;
template <unsigned int M>
struct ModInt {
    unsigned int v;
    ModInt() : v(0) {}
    ModInt(signed v) : v(v % M + (v < 0 ? M : 0)) {}
    ModInt(unsigned v) : v((v % M) + (v >= M ? M : 0)) {}
    ModInt& operator+=(const ModInt& rhs) { return v += rhs.v, v >= M ? v -= M : v, *this; }
    ModInt& operator-=(const ModInt& rhs) { return v -= rhs.v, v < 0 ? v += M : v, *this; }
    ModInt& operator*=(const ModInt& rhs) { return v = ((v * 1ull * rhs.v) % M + (v * 1ull * rhs.v) / M * M) % M, *this; }
    ModInt inverse() const { return pow(M - 2); } // assuming M is prime
    ModInt pow(unsigned long long e) const {
        if (e == 0) return ModInt(1);
        if (e == 1) return *this;
        return (*this * *this).pow(e >> 1) * (*this).pow(e & 1);
    }
    operator unsigned() const { return v; }
    ModInt operator+(const ModInt& rhs) const { return ModInt(*this) += rhs; }
    ModInt operator-(const ModInt& rhs) const { return ModInt(*this) -= rhs; }
    ModInt operator*(const ModInt& rhs) const { return ModInt(*this) *= rhs; }
};
const unsigned int mod = 998244353;
using mint = ModInt<mod>;
const int maxn = 20, maxm = 5e4;
mint comb[maxm + maxn][maxn];
int ans;
int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> db(n, vector<int>(m + 1));
    for (int i = 0; i < n; i++)
        for (int j = 1; j <= m; j++) {
            int x;
            cin >> x;
            db[i][x] = j;
        }
    comb[0][0] = 1;
    for (int i = 1; i <= n + m; i++) {
        comb[i][0] = comb[i][i] = 1;
        for (int j = 1; j < i; j++) comb[i][j] = comb[i - 1][j - 1] + comb[i - 1][j];
    }
    mint f[maxn + 1][maxm + maxn];
    f[n + 1][0] = 1;
    for (int i = n; i >= 1; i--)
        for (int j = 0; j <= m + i - 1; j++) {
            int lower = max(0, j - i + 1);
            int upper = j;
            for (int k = lower + 1; k <= upper; k++)
                f[i][j] += f[i + 1][k] * comb[j - lower][upper - lower] * comb[j - lower][k - lower - 1];
        }
    for (int i = 1; i <= m; i++) {
        int cnt = 0;
        for (int j = 0; j < n; j++) if (db[j][i] <= n) cnt++;
        if (cnt == 0) continue;
        mint tmp = f[1][n];
        for (int j = 1; j < n; j++) {
            if (db[j][i] <= n)
                tmp *= (mint(cnt) - mint(1)) * mint(db[j][i]) * mint(mint(j) * mint(db[j][i]) - mint(j + 1) * mint(db[j][i] - 1)) * mint(db[j][i] - 1).inverse();
            else
                tmp *= (mint(cnt) - mint(1)) * mint(db[j][i]) * mint(mint(j) * mint(db[j][i]) - mint(j + 1) * mint(db[j][i] - 1)) * mint(j + 1).inverse() * mint(db[j][i] - n).inverse();
        }
        ans += (int)(tmp.v);
    }
    cout << ans % mod << endl;
}
