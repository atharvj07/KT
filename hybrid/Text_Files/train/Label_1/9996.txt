#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e6;

int n;
int a[MAXN];
int tp[MAXN];
int inp[MAXN];
vector<int> adj[MAXN];
bool vis[MAXN];
bool ans[MAXN];

int f(int u, int v) {
    if (tp[u] == 0) return v;
    if (tp[u] == 1) return !v;
    if (tp[u] == 2) return v & inp[u];
    if (tp[u] == 3) return v | inp[u];
    if (tp[u] == 4) return v ^ inp[u];
    return -1;
}

int dfs(int u, int p, int val) {
    inp[u] = val;
    int ret = 0;
    if (tp[u] == 1) ret = 1;
    for (auto v : adj[u]) {
        if (v == p) continue;
        ret = f(u, dfs(v, u, ret));
    }
    return ret;
}

int main() {
    cin >> n;
    for (int u = 1; u <= n; u++) {
        string s;
        cin >> s;
        if (s == "IN") {
            int v;
            cin >> v;
            inp[u] = v;
        } else if (s == "NOT") {
            tp[u] = 1;
            int v;
            cin >> v;
            adj[u].push_back(v);
        } else if (s == "AND") {
            tp[u] = 2;
            int v1, v2;
            cin >> v1 >> v2;
            adj[u].push_back(v1);
            adj[u].push_back(v2);
        } else if (s == "OR") {
            tp[u] = 3;
            int v1, v2;
            cin >> v1 >> v2;
            adj[u].push_back(v1);
            adj[u].push_back(v2);
        } else if (s == "XOR") {
            tp[u] = 4;
            int v1, v2;
            cin >> v1 >> v2;
            adj[u].push_back(v1);
            adj[u].push_back(v2);
        }
    }
    dfs(1, 0, 0);
    for (int u = 1; u <= n; u++) {
        if (tp[u] != 0) {
            inp[u] ^= 1;
            dfs(1, 0, 0);
            ans[u] = inp[1];
            inp[u] ^= 1;
        }
    }
    for (int u = 1; u <= n; u++) {
        if (tp[u] == 0) {
            cout << ans[u];
        }
    }
    cout << endl;
    return 0;
}
