#include <iostream>
#include <vector>
#include <unordered_map>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<char> commands(n);
  for (int i = 0; i < n; i++) {
    cin >> commands[i];
  }
  unordered_map<long long, int> cnt;
  long long cur = 0;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (commands[i] == 'U') cur += 1;
    else if (commands[i] == 'R') cur += 1000000;
    else if (commands[i] == 'D') cur += 1000000000;
    else cur += 1000000000000;
    ans += cnt[cur - 1];
    cnt[cur]++;
  }
  cout << ans << endl;
}
