#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int k = 0;
  vector<pair<int, int>> ans;
  int l = -1, r = -1;
  for (int i = 0; i < n; i++) {
    if (l == -1) {
      l = i;
      r = i;
    } else if (a[i] == a[r]) {
      r++;
    } else {
      if (r - l >= 1) {
        ans.push_back({l + 1, r + 1});
        k++;
      }
      l = i;
      r = i;
    }
  }
  if (r - l >= 1) {
    ans.push_back({l + 1, r + 1});
    k++;
  }
  if (k == 0) {
    cout << "-1\n";
  } else {
    cout << k << "\n";
    for (auto p : ans) { cout << p.first << " " << p.second << "\n"; }
  }
  return 0;
}
