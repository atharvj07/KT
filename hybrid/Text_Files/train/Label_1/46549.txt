import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[] b = Arrays.copyOf(a, n);

        int f = 0;
        int l = 0;
        int r = 0;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                r++;
            } else {
                if (cnt < k) {
                    r++;
                    cnt++;
                }
            }

            if (r - l > f) {
                f = r - l;
                l = i - f + 1;
            }
        }

        for (int i = l; i < l + f; i++) {
            b[i] = 1;
        }

        System.out.println(f);
        for (int x : b) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
