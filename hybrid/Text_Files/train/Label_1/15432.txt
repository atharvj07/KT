#include <bits/stdc++.h>
using namespace std;

struct task {
    int a, t;
    bool operator<(const task &o) {
        return a > o.a;
    }
};

int main() {
    int n, t;
    cin >> n >> t;
    vector<task> tasks(n);
    for (int i = 0; i < n; ++i) cin >> tasks[i].a >> tasks[i].t;
    sort(tasks.begin(), tasks.end());
    int score = 0;
    vector<int> solution;
    for (int i = 0; i < n; ++i) {
        if (t < tasks[i].t) break;
        t -= tasks[i].t;
        score++;
        solution.push_back(i + 1);
    }
    cout << score << endl;
    cout << solution.size() << endl;
    for (int x : solution) cout << x << ' ';
    return 0;
}
