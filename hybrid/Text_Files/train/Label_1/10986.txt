#include <bits/stdc++.h>

#define pb push_back
#define N 2000010

using namespace std;

typedef long long ll;

ll a[N], b[N];
ll p[N];
ll sz[N];
ll ans[N];

int main(){
    ll n, q, x, y;
    cin >> n;
    for(ll i=1; i<=n; i++){
        cin >> q >> x >> y;
        if(q == 1){
            ll u = x, v = y;
            ll idx = upper_bound(p+1, p+n+1, x) - p;
            ll idx2 = upper_bound(p+1, p+n+1, y) - p;
            if(idx2 <= idx) ans[i] = 1;
            else{
                ll u_p = p[idx], u_s = p[idx-1];
                ll v_p = p[idx2], v_s = p[idx2-1];
                ans[i] = (u_s < u && u < v_p) || (u_s < v && v < v_p);
            }
            p[idx] = y;
            a[idx] = x;
            b[idx] = y;
            sz[idx] = y-x+1;
        }
        else{
            if(ans[x] && ans[y]) cout << "YES" << endl;
            else cout << "NO" << endl;
        }
    }

    return 0;
}
