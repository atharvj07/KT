from math import factorial
from collections import Counter

def mod_factorial(n, p, factorial_mod):
    if n >= p:
        return 0
    return factorial_mod[n]

def solve(n, p, a):
    a.sort()
    factorial_mod = [1] * (n + 1)
    for i in range(1, n + 1):
        factorial_mod[i] = (factorial_mod[i - 1] * i) % p

    good = []
    cnt = Counter(a)
    unique_a = sorted(set(a))
    dp = [0] * (n + 1)
    dp[0] = 1
    cum = 0
    for x in unique_a:
        if dp[cum] and mod_factorial(n - cum, p, factorial_mod):
            good.append(x - 1)
        cum += cnt[x]
        for i in range(cum, 0, -1):
            dp[i] = (dp[i] + dp[i - 1] * cnt[x]) % p

    if dp[cum] and mod_factorial(n - cum, p, factorial_mod):
        good.append(unique_a[-1] + n - 1 - cum)

    print(len(good))
    print(*good)

n, p = map(int, input().split())
a = list(map(int, input().split()))
solve(n, p, a)
