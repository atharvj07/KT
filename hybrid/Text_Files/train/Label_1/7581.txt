#include <bits/stdc++.h>

using namespace std;

const int maxn = 105;
int n;
string s;
int b[maxn];
vector<int> a[26];

int main() {
    cin >> n;
    cin >> s;
    for (int i = 0; i < n; i++) {
        cin >> b[i];
    }
    for (int i = 0; i < n; i++) {
        a[s[i] - 'a'].push_back(b[i]);
    }
    int ans = 0;
    for (int i = 0; i < 26; i++) {
        sort(a[i].rbegin(), a[i].rend());
        for (int j = 0; j < a[i].size() && j < n / 2; j++) {
            ans += a[i][j];
        }
    }
    cout << ans << endl;
    return 0;
}
