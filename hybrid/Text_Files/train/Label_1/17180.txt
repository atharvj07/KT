#include <bits/stdc++.h>
using namespace std;
#define LL long long
#define pb push_back
#define eb emplace_back
const int MAXN = 100001;
const LL INF = 0x3f3f3f3f3f3f3f3fLL;
const LL MOD = 1000000007LL;
string x, y;
int main() {
    int T;
    cin >> T;
    while (T--) {
        cin >> x >> y;
        while (x.size() < y.size()) x = '0' + x;
        while (y.size() < x.size()) y = '0' + y;
        int n = x.size();
        bool flg = false;
        for (int i = n - 1; i >= 0; i--) {
            if (x[i] > y[i]) break;
            if (x[i] < y[i]) {
                flg = true;
                break;
            }
        }
        if (!flg) {
            cout << 0 << endl;
            continue;
        }
        int k = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (x[i] == '1' && y[i] == '0') k++;
            else break;
        }
        cout << k << endl;
    }
    return 0;
}
