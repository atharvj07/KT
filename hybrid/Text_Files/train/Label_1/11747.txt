#include <vector>
#include <cstring>
#include <iostream>
#include <algorithm>

int main() {
    int a;
    std::cin >> a;
    std::string s;
    std::cin >> s;
    int n = s.size();
    std::vector<std::vector<long long>> b(n, std::vector<long long>(n));
    for (int i = 0; i < n; ++i) {
        for (int j = i; j < n; ++j) {
            int u = s[i] - '0';
            int v = s[j] - '0';
            if (i + 1 < n && j + 1 < n) {
                u *= s[i + 1] - '0';
                v *= s[j + 1] - '0';
            }
            b[i][j] = (long long)u * (long long)v;
        }
    }
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = i; j < n; ++j) {
            for (int x = i; x <= j; ++x) {
                for (int y = j; y >= x; --y) {
                    ans += (long long)(b[i][y] * b[x][j]) == a;
                }
            }
        }
    }
    std::cout << ans << '\n';
    return 0;
}
