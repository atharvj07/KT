import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int xv = scanner.nextInt();
        int yv = scanner.nextInt();
        int xp = scanner.nextInt();
        int yp = scanner.nextInt();
        int xw1 = scanner.nextInt();
        int yw1 = scanner.nextInt();
        int xw2 = scanner.nextInt();
        int yw2 = scanner.nextInt();
        int xm1 = scanner.nextInt();
        int ym1 = scanner.nextInt();
        int xm2 = scanner.nextInt();
        int ym2 = scanner.nextInt();
        boolean canSee = true;
        if (xw1 != xw2) {
            if (xv > xw1 && xp > xw1) {
                canSee = false;
            } else if (xv < xw1 && xp < xw1) {
                canSee = false;
            }
            if (yv >= yw1 && yv <= yw2 && yp >= yw1 && yp <= yw2) {
                canSee = true;
            }
        } else if (yw1 != yw2) {
            if (yv > yw1 && yp > yw1) {
                canSee = false;
            } else if (yv < yw1 && yp < yw1) {
                canSee = false;
            }
            if (xv >= xw1 && xv <= xw2 && xp >= xw1 && xp <= xw2) {
                canSee = true;
            }
        }
        if (xm1 != xm2) {
            if (xv > xm1 && xp > xm1) {
                canSee = false;
            } else if (xv < xm1 && xp < xm1) {
                canSee = false;
            }
            if (yv >= ym1 && yv <= ym2 && yp >= ym1 && yp <= ym2) {
                canSee = true;
            }
        } else if (ym1 != ym2) {
            if (yv > ym1 && yp > ym1) {
                canSee = false;
            } else if (yv < ym1 && yp < ym1) {
                canSee = false;
            }
            if (xv >= xm1 && xv <= xm2 && xp >= xm1 && xp <= xm2) {
                canSee = true;
            }
        }
        System.out.println(canSee ? "YES" : "NO");
    }
}
