import java.util.*;

class Main {
  public static void main(String[] args) {
    var sc = new Scanner(System.in);
    int[] def = new int[4], atk = new int[4];
    for (var i = 0; i < 4; i++) {
      def[i] = sc.nextInt();
      atk[i] = sc.nextInt();
    }
    Arrays.sort(def);
    Arrays.sort(atk);
    if (def[0] < atk[3] && def[1] < atk[2]) System.out.println("Team 2");
    else if (def[2] > atk[0] && def[3] > atk[1]) System.out.println("Team 1");
    else System.out.println("Draw");
  }
}
