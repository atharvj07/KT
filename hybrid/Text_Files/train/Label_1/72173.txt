MOD = 10**9 + 7

def sum_mod(n, m):
    result = 0
    prev = -1
    for i in range(1, m + 1):
        current = n % i
        if current == prev:  # Early stopping if the pattern repeats
            break
        result = (result + current) % MOD
        prev = current
    return result

n, m = map(int, input().split())
print(sum_mod(n, m))
