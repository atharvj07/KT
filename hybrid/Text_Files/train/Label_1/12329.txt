#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  ll n;
  cin >> n;
  ll arr[n];
  for (ll i = 0; i < n; i++) { cin >> arr[i]; }
  sort(arr, arr + n);
  if (n % 2 == 1) {
    cout << arr[n / 2] << endl;
  } else {
    cout << (arr[n / 2 - 1] + arr[n / 2]) / 2 << endl;
  }
}
