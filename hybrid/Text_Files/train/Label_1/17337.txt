#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, q;
  cin >> n >> q;
  vector<int> data(n);
  for (auto &i : data) cin >> i;
  for (int i = 0, t, x; i < q; i++) {
    char c;
    cin >> c >> x;
    for (auto &j : data) {
      if (c == '>' && j > x) j *= -1;
      if (c == '<' && j < x) j *= -1;
    }
  }
  for (int i = 0; i < n; i++) cout << data[i] << ' ';
  return 0;
}
