#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e5+5;
const int inf = 2e9;
vector<int> adj[N];
int a, b, da, db;
ll dis[N];
bool vis[N];
queue<int> q;
int main() {
    int t;
    cin >> t;
    while(t--) {
        int n;
        cin >> n >> a >> b >> da >> db;
        for(int i=1; i<=n; i++) {
            adj[i].clear();
            dis[i] = inf;
            vis[i] = false;
        }
        for(int i=1; i<=n-1; i++) {
            int u, v;
            cin >> u >> v;
            adj[u].push_back(v);
            adj[v].push_back(u);
        }
        q.push(a);
        dis[a] = 0;
        while(!q.empty()) {
            int cur = q.front();
            q.pop();
            vis[cur] = true;
            for(int v:adj[cur]) {
                if(!vis[v] && dis[v]>dis[cur]+1 && dis[cur]+1<=da) {
                    dis[v] = dis[cur]+1;
                    q.push(v);
                }
            }
        }
        ll mx = 0;
        for(int i=1; i<=n; i++) mx = max(mx, dis[i]);
        while(!q.empty()) q.pop();
        q.push(b);
        dis[b] = 0;
        while(!q.empty()) {
            int cur = q.front();
            q.pop();
            vis[cur] = true;
            for(int v:adj[cur]) {
                if(!vis[v] && dis[v]>dis[cur]+1 && dis[cur]+1<=db) {
                    dis[v] = dis[cur]+1;
                    q.push(v);
                }
            }
        }
        mx = 0;
        for(int i=1; i<=n; i++) mx = max(mx, dis[i]);
        if(mx>da) cout<<"Bob"<<endl;
        else if(mx>db) cout<<"Alice"<<endl;
        else cout<<"Alice"<<endl;
    }
    return 0;
}
