#include<bits/stdc++.h>
using namespace std;typedef long long ll;const ll N = 1e6 + 7;
int a[N], p[N], c[N], k;ll n;
int main(){for(cin >> n; n; n >>= 1) a[++k] = n & 1;
    for(int i = k; i >= 1; --i){
        if(!c[i]){
            if(a[i]) p[++p[0]] = 1 << (i - 1);
            else{
                for(int j = p[0]; j >= 1; --j)
                    if(p[j] < (1 << (i - 1))){
                        c[i] = c[j] + 1;
                        break;
                    }
                if(!c[i]) c[i] = 1;
            }
        }p[0] += c[i];
    }cout << p[0];}
