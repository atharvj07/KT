#include <bits/stdc++.h>
#define int long long
using namespace std;
const int maxn = 1e5 + 5;
int p[maxn];
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    for (int i = 1; i <= n; ++i) { cin >> p[i]; }
    int l = 1, r = n, ans = 0;
    while (l < r) {
      if (p[l] > p[r])
        r--;
      else
        l++;
      ans = max(ans, abs(p[l] - p[r]));
    }
    l = 1, r = n;
    while (l < r) {
      if (p[l] < p[r])
        l++;
      else
        r--;
      ans = max(ans, abs(p[l] - p[r]));
    }
    int mx = 0;
    int id1 = 0, id2 = 0;
    for (int i = 1; i <= n; ++i) {
      for (int j = i + 1; j <= n; ++j) {
        if (abs(p[i] - p[j]) > mx) {
          id1 = i;
          id2 = j;
          mx = abs(p[i] - p[j]);
        }
      }
    }
    cout << 2 << '\n' << p[id1] << ' ' << p[id2] << '\n';
  }
  return 0;
}
