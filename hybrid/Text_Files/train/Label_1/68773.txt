from sys import stdin

n, m = map(int, stdin.readline().split())
a = [list(map(int, stdin.readline().split())) for _ in range(n)]
if any(a[i][0] >= a[i][j] or a[i][j] >= a[i][-1] or a[0][j] >= a[i][j] or a[i][j] >= a[-1][j] for i in range(n) for j in range(m)):
    print(-1)
else:
    a = [sorted(filter(lambda x: x != 0, r)) for r in a]
    for i in range(n):
        for j in range(m):
            if a[i][j] == 0:
                a[i][j] = (a[i][j - 1] + a[i][j + 1] - 1) // 2
    print(sum(sum(r) for r in a))
