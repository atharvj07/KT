#include <bits/stdc++.h>
using namespace std;
int main() {
    pair<int, int> cen1, cen2;
    int r1, r2;
    double dis, ans;
    cin >> cen1.first >> cen1.second >> r1 >> cen2.first >> cen2.second >> r2;
    dis = sqrt(pow(cen1.first - cen2.first, 2) + pow(cen1.second - cen2.second, 2));
    if (dis > (r1 + r2)) {
        cout << 0 << endl;
        return 0;
    }
    if ((r1 + r2) < dis) {
        cout << 0 << endl;
        return 0;
    }
    if (r2 + r1 == dis) {
        ans = r1 * r1 * acos((pow(dis, 2) + pow(r1, 2) - pow(r2, 2)) / (2.0 * dis * r1));
        ans += r2 * r2 * acos((pow(dis, 2) + pow(r2, 2) - pow(r1, 2)) / (2.0 * dis * r2));
        ans -= sqrt((dis + r1 + r2) * (dis + r1 - r2) * (dis - r1 + r2) * (-dis + r1 + r2)) * 0.5;
    } else if (r1 > r2 + dis || r2 > r1 + dis) {
        ans = min(r1, r2) * min(r1, r2) * M_PI;
    } else {
        double a1 = (pow(r1, 2) - pow(r2, 2) + pow(dis, 2)) / (2 * r1 * dis);
        double a2 = (pow(r2, 2) - pow(r1, 2) + pow(dis, 2)) / (2 * r2 * dis);
        double angle1 = acos(a1);
        double angle2 = acos(a2);
        ans = r1 * r1 * (angle1 - sin(angle1) * cos(angle1)) + r2 * r2 * (angle2 - sin(angle2) * cos(angle2));
    }
    cout << setprecision(20) << ans << endl;
}
