def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

n = int(input().strip())
l = list(map(int, input().strip().split()))
c = list(map(int, input().strip().split()))

dp = {0: 0}
for i in range(n):
    dp2 = dp.copy()
    for g in dp:
        new_g = gcd(g, l[i])
        if new_g in dp2:
            dp2[new_g] = min(dp2[new_g], dp[g] + c[i])
        else:
            dp2[new_g] = dp[g] + c[i]
    dp = dp2

if 1 in dp:
    print(dp[1])
else:
    print(-1)
