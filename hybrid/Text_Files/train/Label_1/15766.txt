#include<bits/stdc++.h>
using namespace std;
int main(){
    double sum=0,a;
    int n,k;
    cin>>n>>k;
    for(int i=0;i<n;i++){
        cin>>a;
        sum+=a;
        if(i>=k){
            sum-=a;
        }
        if(i>=k-1){
            cout<<fixed<<setprecision(10)<<sum/k<<endl;
        }
    }
}
