from functools import reduce
import sys
input=sys.stdin.readline

def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)

for _ in range(int(input())):
    n = int(input())
    a = sorted([int(x) for x in input().split()], reverse = True)
    g = 1
    for x in a:
        g = gcd(g, x)
        print(x // g, end = ' ')
    print()
