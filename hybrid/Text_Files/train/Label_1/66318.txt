n = int(input())
g = {i: [] for i in range(1, n + 1)}
for _ in range(n - 1):
    a, b = tuple(map(int, input().split()))
    g[a].append(b)
    g[b].append(a)
cnt = 0
vis = [False] * (n + 1)
for i in g:
    if vis[i]:
        continue
    vis[i] = True
    q = [i]
    while q:
        v = q.pop(0)
        for u in g[v]:
            if vis[u]:
                continue
            q.append(u)
            vis[u] = True
            cnt += 1
print(cnt)
