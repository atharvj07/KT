import java.util.Scanner;

public class LongestUncommonSubsequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String a = scanner.nextLine();
        String b = scanner.nextLine();

        int maxLength = Math.max(a.length(), b.length());
        boolean[] isSubsequence = new boolean[maxLength + 1];

        int longestUncommonSubsequenceLength = -1;
        for (int i = 0; i < a.length(); i++) {
            for (int j = b.length() - 1; j >= 0; j--) {
                if (a.charAt(i) == b.charAt(j)) {
                    isSubsequence[j + 1] = true;
                } else if (isSubsequence[j]) {
                    break;
                }
            }
        }

        for (int i = 0; i <= maxLength; i++) {
            if (!isSubsequence[i]) {
                longestUncommonSubsequenceLength = Math.max(longestUncommonSubsequenceLength, i);
            }
        }

        System.out.println(longestUncommonSubsequenceLength);
    }
}
