import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.extend(a)
    inc = [1] * (2 * n)
    for i in range(1, 2 * n):
        for j in range(i):
            if a[i] > a[j] and inc[i] < inc[j] + 1:
                inc[i] = inc[j] + 1
    print(max(inc))
