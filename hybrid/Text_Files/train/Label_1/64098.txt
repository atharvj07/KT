import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())

    t = [0]
    x = [0]
    for _ in range(n):
        ti, xi = map(int, input().split())
        if ti >= t[-1]:
            t.append(ti)
            x.append(xi)

    ans = 0
    for i in range(1, len(t)):
        if t[i] <= x[i - 1] + i - 1:
            ans += 1

    print(ans)
