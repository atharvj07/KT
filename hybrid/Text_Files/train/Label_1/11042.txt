#define REP(a, b) for (ll a = 0; a < (b); a++)
#define REV(a, b) for (ll a = (b) - 1; a >= 0; a--)
#define LOG(x) cout << x << endl;

#define vc vector

int main() {
    ll n;
    cin >> n;
    ll temp;
    ll Xor = 0;
    bool found = false;
    unordered_map<ll, vi> indices;
    REP(i, n) {
        cin >> temp;
        if(indices.count(temp) == 0){
            vi possIndices = {i};
            indices[temp] = possIndices;
        }
        else{
            indices[temp].pb(i);
        }
        Xor = Xor ^ temp;
    }
    if(Xor == 0){
        cout << "YES" << endl;
        cout << "0" << endl;
    }
    else{
        if(indices.count(Xor) != 0){
            cout << "YES" << endl;
            cout << "1" << endl;
            cout << indices[Xor][0] + 1 << " " << indices[Xor][1] + 1 << " " << indices[Xor][2] + 1 << endl;
        }
        else{
            cout << "NO" << endl;
        }
    }
}
