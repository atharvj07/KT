#include <iostream>
using namespace std;

int t[1000001];

int main() {
    long long w;
    cin >> w;
    int m[9];
    for (int i = 0; i < 8; i++) { cin >> m[i]; }
    for (int i = 0; i <= w; i++) { t[i] = 0; }
    for (int p = 1; p < 9; p++) {
        for (long long i = w; i >= p; i--) {
            t[i] = max(t[i], t[i - p] + p * m[p - 1]);
        }
    }
    cout << t[w] << "\n";
    return 0;
}
