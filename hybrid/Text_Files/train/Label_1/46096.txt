import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            if (s.charAt(i) == s.charAt(i + 1)) {
                ans++;
                i++;
            }
            for (int j = i + 2; j < n; j++) {
                if (s.charAt(i) == s.charAt(j) && s.charAt(i + 1) == s.charAt(j - 1)) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
