from sys import stdin

for test in range(int(stdin.readline())):
    n = int(stdin.readline())
    l = list(map(int, stdin.readline().split()))

    if l[-1] != n:
        print(-1)
        continue

    p = [l[-1]]
    l.pop()
    while l:
        if not l or l[-1] <= p[-1]:
            print(-1)
            break
        p.append(l[-1])
        l.pop()
    else:
        p.reverse()
        print(*p)
