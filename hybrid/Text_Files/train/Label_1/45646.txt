import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      String binary = Integer.toBinaryString(i);
      boolean flag = true;
      for (char c : binary.toCharArray()) {
        if (c != '0' && c != '1') {
          flag = false;
          break;
        }
      }
      if (flag) {
        ans++;
      }
    }
    System.out.println(ans);
  }
}
