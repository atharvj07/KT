from sys import stdin
k, n = map(int, input().split())
S = [input() for i in range(k)]

dic = {}
for s in S:
    for i in range(len(s)):
        for j in range(len(s)):
            if i != j:
                if not (s[i], s[j]) in dic:
                    dic[(s[i], s[j])] = set([s])
                else:
                    if s not in dic[(s[i], s[j])]:
                        dic[(s[i], s[j])].add(s)

vcnt = 0
for v in dic.values():
    vcnt += len(v) > 2
if vcnt > 0:
    print(-1)
else:
    ans = ""
    for a, b in dic.keys():
        if len(dic[(a, b)]) == 1:
            ans += a
        if len(dic[(a, b)]) == 2:
            ans += b

    if len(ans) == 0 or len(ans) == len(S):
        print(ans)
    else:
        print(-1)
