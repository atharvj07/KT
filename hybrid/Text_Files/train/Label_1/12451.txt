#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, a[200005];
    vector<int> ans;
    cin >> n;
    cin >> a[1];
    for (int i = 1, last = -1, r = 0; i < n + 1; i++) {
        while (r < n && a[r + 1] <= a[r]) r++;
        if (a[r] <= a[r + 1]) {
            if (last != 0) ans.push_back(last);
            last = i;
        }
        if (a[i] <= a[i + 1]) {
            if (last != 0) ans.push_back(last);
            last = 0;
        }
    }
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++)
        cout << (ans[i] != 0 ? 'R' : 'L');
    return 0;
}
