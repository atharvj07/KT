import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) a[i] = sc.nextLong();
    if (n == 1) {
      System.out.println(0);
      return;
    }
    long d = a[1] - a[0];
    long l = (a[n - 1] - a[n - 2]) / d + 1;
    if (l == n) {
      System.out.println(0);
      return;
    }
    int ans = 2;
    for (int i = 0; i < n; i++) {
      long r = (a[i] - a[(i - 1 + n) % n]) / d + 1;
      long l2 = (a[(i + 1) % n] - a[i]) / d + 1;
      if (r == n || l2 == n) {
        ans = 1;
      }
    }
    System.out.println(ans);
  }
}
