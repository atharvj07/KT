from collections import deque

def bfs(start, graph, parity):
    queue = deque([(start, 0)])
    visited = set()
    visited.add(start)
    while queue:
        current, depth = queue.popleft()
        if parity[current] != parity[start]:
            return depth
        for neighbor in graph[current]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, depth + 1))
    return -1

n = int(input())
a = list(map(int, input().split()))

graph = [[] for _ in range(n)]
parity = [x % 2 for x in a]

for i in range(n):
    for direction in [-1, 1]:
        if 0 <= i + direction * a[i] < n:
            graph[i].append(i + direction * a[i])

results = [bfs(i, graph, parity) for i in range(n)]

print(*results)
