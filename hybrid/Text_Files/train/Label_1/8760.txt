#include <vector>
#include <iostream>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }
  int ans = 0;
  for (int i = 0; i < n - 1; ++i) {
    if (a[i] > a[i + 1]) {
      ++ans;
      int val = a[i];
      for (int j = i + 1; j < n; ++j) {
        if (val > a[j]) {
          a[j - 1] = a[j];
        } else {
          a[j - 1] = val;
          break;
        }
      }
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
