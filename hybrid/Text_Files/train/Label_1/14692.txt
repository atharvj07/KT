#include <bits/stdc++.h>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int x1, p1, x2, p2;
        std::cin >> x1 >> p1 >> x2 >> p2;
        if (x1 * pow(10, p1) < x2 * pow(10, p2))
            std::cout << '<';
        else if (x1 * pow(10, p1) > x2 * pow(10, p2))
            std::cout << '>';
        else
            std::cout << '=';
        std::cout << '\n';
    }
    return 0;
}
