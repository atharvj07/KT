#include <vector>
#include <iostream>

using namespace std;

vector<vector<int>> g;
int n, m;
vector<pair<int,int>> ans;

int solve(vector<int>& a, int sum, int mask, int cursum) {
    if (cursum > sum) {
        return -1;
    }
    if (cursum == sum && mask == (1 << a.size()) - 1) {
        return 0;
    }
    for (int i = 0; i < a.size(); ++i) {
        if (!(mask & (1 << i))) {
            int status = solve(a, sum, mask | (1 << i), cursum + a[i]);
            if (status != -1) {
                ans.emplace_back(1, i + 1);
                return status + 1;
            }
        }
    }
    return -1;
}

int main() {
    int a;
    cin >> n >> m;
    g.resize(n, vector<int>(m));
    vector<int> rows(n), cols(m);
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            cin >> g[i][j];
            rows[i] += g[i][j];
            cols[j] += g[i][j];
        }
    }
    int ans_rows = solve(rows, m, 0, 0);
    if (ans_rows == -1) {
        cout << -1 << "\n";
    } else {
        cout << ans_rows + solve(cols, n, 0, 0) << "\n";
        for (auto&& e : ans) {
            cout << "row " << e.second << "\n";
        }
        for (auto&& e : ans) {
            cout << "col " << e.second << "\n";
        }
    }
    return 0;
}
