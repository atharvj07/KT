import sys

input = sys.stdin.readline

n = int(input())
points = [list(map(int, input().split())) for _ in range(n)]
count = 0
ps = set()
for x, y in points:
    for nx, ny in points:
        if x == nx and y == ny:
            continue
        g = (y - ny) * (x + nx) + (x - nx) * (y + ny)
        if not g and ((x <= nx and nx <= x + (x - nx)) or (x >= nx and nx >= x + (x - nx))):
            if (g, x, y, nx, ny) not in ps:
                count += 1
                ps.add((g, x, y, nx, ny))
print(count)
