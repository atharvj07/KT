#include <bits/stdc++.h>
using namespace std;
const int INF = 1e9;
int n, c1, c2;
vector<int> a;
int main() {
    cin >> n >> c1 >> c2;
    a.assign(n, -1);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        a[i] ^= 1;
    }
    vector<long long> dp(n + 1, INF);
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < i; j++) {
            if (a[j]) dp[i] = min(dp[i], dp[j] + c1 + c2 * (i - j - 1) * (i - j - 1));
        }
    }
    cout << dp[n] << '\n';
}
