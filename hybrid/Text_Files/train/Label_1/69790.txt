n, m, d = map(int, input().split())
p = sorted([(c, w) for c, w in (map(int, input().split()) for _ in range(n))], reverse=True)
i = sorted([(c, w) for c, w in (map(int, input().split()) for _ in range(m))], reverse=True)
ans = 0
j = 0
for c, w in p:
    if w <= d:
        ans += c
        d -= w
        j += 1
    else: break
s = [c for c, w in i]
if j > 0 and s:
    ans += sum(s[:j])
print(ans)
