import java.util.*;

public class Main {

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    while (scanner.hasNext()) {
      long n = scanner.nextInt();
      long ans = (n + 1) * (n + 2) * (n + 3) / 6;
      System.out.println(ans);
    }
    scanner.close();
  }
}
