import sys

t = int(input())

for _ in range(t):
    n = int(input())
    p = [int(x) for x in input().split()]
    a = []
    b = []
    for x in p:
        if not a or x > a[-1]:
            a.append(x)
        else:
            b.append(x)
    if len(a) == n and len(b) == n and set(a).isdisjoint(set(b)):
        print("YES")
    else:
        print("NO")
