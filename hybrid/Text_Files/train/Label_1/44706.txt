import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        long n = sc.nextLong();

        long last = 1;
        long i = 1;

        while (i + last < n) {
            i += last;
            last++;
        }

        System.out.println((n - i) % last + 1);
    }
}
