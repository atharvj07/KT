#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int MAXN = 2e5 + 5;
ll n, t, t_i[MAXN];

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n >> t;
  for (int i = 1; i <= n; i++) cin >> t_i[i];
  ll last = 0, ans = 0;
  for (int i = 1; i <= n; i++) {
    if (last + 1 >= t_i[i]) {
      ans++;
      last++;
    } else if (last + 1 < t_i[i] && last + 1 + t_i[i] <= t) {
      ans++;
      last = last + 1 + t_i[i];
    } else
      break;
  }
  cout << ans << endl;
  return 0;
}
