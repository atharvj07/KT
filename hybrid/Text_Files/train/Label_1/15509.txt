#include<bits/stdc++.h>
using namespace std ;
const int mxN = 22 , mxM = 22 ;
bool vis[mxN][mxM] ;
int main()
{
    int n , m ;
    cin >> n >> m ;
    int arr[mxN][mxM] ;
    vector<int> row[mxN] ;
    for(int i = 0 ; i < n ; i++)
    {
        for(int j = 0 ; j < m ; j++)
        {
            cin >> arr[i][j] ;
        }
    }

    for(int i = 0 ; i < n ; i++)
    {
        for(int j = 0 ; j < m ; j++)
        {
            vis[i][j] = 0 ;
        }
    }

    for(int i = 0 ; i < n ; i++)
    {
        for(int j = 1 ; j <= m ; j++)
        {
            for(int k = 0 ; k < m ; k++)
            {
                if(arr[i][k] == j)
                {
                    row[i].push_back(k) ;
                    vis[i][k] = 1 ;
                    break ;
                }
            }

        }
        vector<int> temp ;
        int limit = 0 ;
        for(int j = 0 ; j < m ; j++)
        {

            if(vis[i][j] == 0)
            {
                temp.push_back(j) ;
                if(j == m - 1)
                {
                    limit = 1 ;
                }
            }
        }

        if(limit == 1)
        {
            if(temp.size() % 2 != 0)
            {
                cout << "NO" << endl ;
                return 0 ;
            }
            else
            {
                int j = 0 , k = temp.size() - 1 ;
                while(j < k)
                {
                    swap(arr[i][row[i][temp[j]]],arr[i][row[i][temp[k]]]) ;
                    j+=2 ;
                    k-=2 ;
                }
            }
        }
    }

    for(int i = 0 ; i < m ; i++)
    {
        for(int j = 1 ; j <= m ; j++)
        {
            for(int k = 0 ; k < n ; k++)
            {
                if(arr[k][i] == j)
                {
                    vis[k][i] = 1 ;
                    break ;
                }
            }
        }
        vector<int> temp ;
        int limit = 0 , last = -1 ;
        for(int j = 0 ; j < n ; j++)
        {
            if(vis[j][i] == 0)
            {
                temp.push_back(j) ;
                last = j ;
                if(j == n - 1)
                {
                    limit = 1 ;
                }
            }
        }
        if(limit == 1)
        {
            if(last == 0)
            {
                if(temp.size() % 2 != 0)
                {
                    cout << "NO" << endl ;
                    return 0 ;
                }
                else
                {
                    int j = 0 , k = temp.size() - 1 ;
                    while(j < k)
                    {
                        swap(arr[temp[j]][i],arr[temp[k]][i]) ;
                        j+=2 ;
                        k-=2 ;
                    }
                }
            }
            else
            {
                if(temp.size() <= 2)
                {
                    if(temp.size() == 1 && i != 0)
                    {
                        cout << "NO" << endl ;
                        return 0 ;
                    }
                    else
                    {
                        swap(arr[last][i],arr[temp[0]][i]) ;
                    }
                }
                else
                {
                    if(temp.size() % 2 != 0)
                    {
                        cout << "NO" << endl ;
                        return 0 ;
                    }
                    else
                    {
                        int k = temp.size() - 1 , j = 0 ;
                        while(j < k)
                        {

                            swap(arr[temp[j]][i],arr[temp[k]][i]) ;
                            j+=2 ;
                            k-=2 ;
                        }
                    }
                }

            }
        }
    }

    cout << "YES" << endl ;
    return 0 ;
}
