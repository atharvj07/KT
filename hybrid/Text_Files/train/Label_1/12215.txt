#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> type(n);
  for (int i = 0; i < n; i++) { cin >> type[i]; }
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int k = 0;
  for (int i = 0; i < n; i++) {
    if (type[i] == 1) {
      int j = i + 1;
      while (j < n && type[j] == 0 && a[j] != 0) {
        j = a[j];
      }
      if (j == n) k = max(k, j - i);
      i = j - 1;
    }
  }
  cout << k << endl;
  for (int i = 0; i < n; i++) {
    if (type[i] == 1) {
      int j = i + 1;
      while (j < n && type[j] == 0 && a[j] != 0) {
        cout << j << " ";
        j = a[j];
      }
      cout << j << endl;
      break;
    }
  }
  return 0;
}
