import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] deg = new int[n + 1];
        for (int i = 0; i < n - 1; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            deg[x]++;
            deg[y]++;
        }
        int dead = 0;
        for (int i = 1; i <= n; i++) {
            if (deg[i] == 1) {
                dead++;
            }
        }
        System.out.println((dead + 1) / 2);
    }
}
