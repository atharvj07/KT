def lucky_numbers_count(n):
    def is_lucky(n, x, y):
        for digit in str(n):
            if digit not in (x, y):
                return False
        return True

    def count_lucky_numbers(n, prefix="", x="0", y="0"):
        if prefix != "" and int(prefix) > n:
            return 0
        if prefix != "" and x != "0" and y != "0":
            return 1
        count = 0
        for digit in "0123456789":
            if digit == x or digit == y or x == "0" or y == "0":
                if x == "0" and digit != "0":
                    count += count_lucky_numbers(n, prefix + digit, digit, y)
                elif y == "0" and digit != "0" and digit > x:
                    count += count_lucky_numbers(n, prefix + digit, x, digit)
                elif x != "0" and y != "0":
                    count += count_lucky_numbers(n, prefix + digit, x, y)
        return count

    return count_lucky_numbers(n)

n = int(input().strip())
print(lucky_numbers_count(n))
