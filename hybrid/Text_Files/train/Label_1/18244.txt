#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;

ll pw(ll x,ll y){
    x%=mod;
    ll ans=1;
    while(y){
        if(y&1) ans = (ans*x)%mod;
        x=(x*x)%mod;
        y>>=1;
    }
    return ans;
}


int main()
{
    ll n, x;
    cin >> n;

    map<ll, ll> mp;
    for(ll i = 0; i < n; i++)
    {
        cin >> x;
        if(x == -1) mp[i + 1] = 1ll;
    }

    if(mp.empty()) cout << 0;
    else{
        ll z = mp.size();
        ll fact = 1;
        for(ll i = 2; i <= z; i++) fact = (fact * i) % mod;
        ll den = pw(fact, mod - 2);

        ll ans = ((ll)(n - z) * fact ) % mod;
        ans = (ans * den) % mod;

        cout << ans << "\n";
    }
    return 0;
}
