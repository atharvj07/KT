#include <iostream>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    long long a, b, c, d;
    std::cin >> a >> b >> c >> d;
    if (b >= a) {
      std::cout << b << std::endl;
    } else {
      long long res = b + (((a - b - d) / (c + d)) + 1) * (c + d);
      std::cout << res << std::endl;
    }
  }
  return 0;
}
