#include <iostream>
#include <queue>
#include <vector>
using namespace std;

int main() {
  int n, m, ans = 0, sum = 0;
  vector<pair<int, int>> a;
  cin >> n >> m;
  for (int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    a.push_back({x, i + 1});
  }
  while (!a.empty()) {
    int cnt = 0;
    for (int i = 0; i < (int)a.size(); ++i) {
      sum += m;
      if (a[i].first > sum) {
        sum -= m;
        a.push_back(a[i]);
      } else {
        cnt++;
        if (cnt == (int)a.size()) ans = a[i].second;
      }
    }
    a.erase(a.begin(), a.begin() + cnt);
  }
  cout << ans << endl;
}
