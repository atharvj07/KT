n = int(input().strip())
draws = 1  # Count the starting moment of the match as a draw
prev_a, prev_b = 0, 0

for _ in range(n):
    a, b = map(int, input().strip().split())
    if a > b:
        draws += max(0, min(a, b) - max(prev_a, prev_b) + 1)
    else:
        draws += max(0, min(a, b) - max(prev_a, prev_b) + 1)
    prev_a, prev_b = a, b

print(draws)
