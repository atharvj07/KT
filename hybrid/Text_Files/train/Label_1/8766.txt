#include <bits/stdc++.h>
#define ll long long
#define ull unsigned long long
using namespace std;
const ll mod = 1000000007;
int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> dp(k + 1);
  dp[0] = 1;
  dp[1] = n * (n - 1);
  for (int i = 2; i <= k; i++) dp[i] = (dp[i - 1] * (n - i) % mod + dp[i - 2] * (i - 1) % mod) % mod;
  if (k % 2 == 0)
    cout << dp[k];
  else
    cout << (dp[k] - dp[k - 1] + mod) % mod;
  return 0;
}
