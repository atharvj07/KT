#include <bits/stdc++.h>
using namespace std;
#define ll long long

const int N = 5e5 + 5;

int n;
vector<ll> a, b;

int main() {
    cin >> n;
    a.resize(n), b.resize(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n;) {
        ll x = a[i];
        int j = i;
        while (j < n && a[j] % 2 == 0) {
            a[j] /= 2;
            j++;
        }
        if (j > i) {
            b[i] = x;
            i = j;
        } else {
            b[j++] = a[i++];
        }
    }
    sort(b.begin(), b.end());
    for (auto x : b) cout << x << ' ';
    cout << '\n';
}
