#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n; cin >> n;
  vector<int> a(n); for (int& x : a) cin >> x;
  int l = 0, r = n - 1, ans = 0;
  string s;
  while (l <= r) {
    if ((l != 0 && a[l - 1] > a[l]) || (r != n - 1 && a[r + 1] < a[r])) {
      if (a[l] < a[r]) {
        s += 'L'; ans++; l++;
      } else {
        s += 'R'; ans++; r--;
      }
    } else if (a[l] < a[r]) {
      s += 'L'; ans++; l++;
    } else {
      s += 'R'; ans++; r--;
    }
  }
  cout << ans << '\n' << s << '\n';
}
