#include <cstdio>
#include <vector>
#include <queue>

using namespace std;

int main() {
  int n, x;
  scanf("%d", &n);
  long long score = 0;
  vector<int> b;
  priority_queue<int> q;
  for (int i = 0; i < n; i++) {
    scanf("%d", &x);
    if (x == 1)
      score += x;
    else
      q.push(x);
  }
  while (q.size() > 1) {
    int a = q.top();
    q.pop();
    int b = q.top();
    q.pop();
    score += a + b;
    q.push(a + b);
  }
  if (!q.empty()) score += q.top();
  printf("%lld\n", score);
  return 0;
}
