#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
vector<int> g[N], w[N];
int depth[N], p[N], sz[N];
int n, m;

int get(int x) {
  if (p[x] != x) p[x] = get(p[x]);
  return p[x];
}

void unite(int a, int b) {
  a = get(a), b = get(b);
  if (a == b) return;
  if (depth[a] > depth[b]) swap(a, b);
  p[a] = b;
  if (depth[a] == depth[b]) depth[b]++;
  sz[b] += sz[a];
}

int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) { depth[i] = 1; p[i] = i; sz[i] = 1; }
  for (int i = 1; i <= m; i++) {
    int u, v;
    cin >> u >> v;
    g[u].push_back(v), w[u].push_back(1);
    g[v].push_back(u), w[v].push_back(1);
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < g[i].size(); j++) {
      int u = i, v = g[i][j];
      if (get(u) == get(v)) continue;
      ans += w[i][j];
      unite(u, v);
    }
  }
  cout << ans << endl;
  return 0;
}
