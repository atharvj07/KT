#include<iostream>
#include<cmath>
#define MAXT 10000001
#define MAXN 101
using namespace std;
int n,t,ar[MAXT],ans,mxt;
int main(){
    cin>>n>>t;
    for(int i=1;i<=n;++i){
        cin>>ar[i];
        ar[i+n]=ar[i]-n;
    }
    for(int i=1;i<=2*n;++i){
        if(ar[i-1]<=ar[i]){
            ans++;
            if(ans>mxt)
                mxt=ans;
        }
        else
            ans=1;
    }
    cout<<mxt;
    return 0;
}
