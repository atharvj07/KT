#include <bits/stdc++.h>
using namespace std;
int t,r,c,ma,mb,ans;
char g[65][65],tmp;
int main(){
    cin>>t;
    while(t--){
        cin>>r>>c;
        ans=0;ma=0;mb=0;
        memset(g,0,sizeof g);
        for(int i=1;i<=r;i++){
            for(int j=1;j<=c;j++){
                cin>>tmp;g[i][j]=tmp;
                if(tmp=='P')ma+=1;
                else mb+=1;
            }
        }
        if(r%3==1&&c%3==1){
            if(ma%3==1||mb%3==1){
                cout<<"MORTAL"<<endl;
                continue;
            }
            ans+=ma/3;
            ans+=mb/3;
            int x=r/3;
            int y=c/3;
            ans+=x*y;
        }
        else if(r%3==1||c%3==1){
            int x=r%3+c%3;
            int y=0;
            if(r%3==0){
                x+=1;
                y+=r/3;
            }
            else if(c%3==0){
                x+=1;
                y+=c/3;
            }
            ans+=ma/x;
            ans+=mb/x;
            ans+=y;
        }
        else{
            ans+=ma/3;
            ans+=mb/3;
            int x=r/3;
            int y=c/3;
            ans+=x*y;
        }
        cout<<ans<<endl;
    }
}
