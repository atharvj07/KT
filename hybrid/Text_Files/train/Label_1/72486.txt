import sys
input = sys.stdin.readline

n = int(input())
a = [[] for i in range(2*n+1)]
for i in range(n):
    x,y = map(int, input().split())
    a[x].append(y)
    a[y].append(x)

ans = []
for i in range(2*n,0,-1):
    if len(a[i])>=2:
        ans.append(a[i][0])
        ans.append(a[i][1])
        a[a[i][0]].remove(i)
        a[a[i][1]].remove(i)
        
print(len(ans))
print(*ans)
