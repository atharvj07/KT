#include <iostream>
using namespace std;
int main() {
    int n, m;
    char a[110][110];
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) cin >> a[i] + 1;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j)
            if (a[i][j] == '.') {
                if ((i + j) % 2)
                    a[i][j] = 'B';
                else
                    a[i][j] = 'W';
            }
        cout << a[i] + 1 << endl;
    }
    return 0;
}
