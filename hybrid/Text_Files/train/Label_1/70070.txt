n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))

# Calculate the distances between stops
distances = [0] * (n - 1)
for i in range(n - 1):
    distances[i] = a[i + 1] - a[i]

# Calculate the sum of distances for visited stops
visited_sum = 0
for i in range(1, m):
    if b[i] == b[i - 1]:
        continue
    elif b[i] > b[i - 1]:
        visited_sum += sum(distances[b[i - 1] - 1:b[i] - 1])
    else:
        visited_sum += sum(distances[b[i] - 1:b[i - 1] - 1])

# Check if the path is uniquely determined
if m > 2 and b[0] == b[1] and b[m - 2] == b[m - 1] and (b[0] != 1 or b[m - 1] != n):
    print(-1)
else:
    print(visited_sum)
