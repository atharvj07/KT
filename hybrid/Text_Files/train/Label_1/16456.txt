#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    while(getline(cin,s)){
        s='0'+s;
        reverse(s.begin(),s.end());
        int mx=0;
        int cnt=0;
        bool neg=0;
        for(int i=0;i<s.length();){
            int j=i;
            while(j<s.length()&&s[j]=='0')j++;
            if(j==s.length())break;
            int p=1;
            while(j<s.length()&&s[j]=='1'){j++;p++;}
            if(s[j]=='0'){
                if(p>1&&p%2==0)mx=max(mx,p/2);
                else if(p==1)neg=1;
                if(p%2==1){
                    if(neg){cnt++;neg=0;}
                    else cnt+=2;
                }
            }
            i=j;
        }
        cout<<cnt+neg<<endl;
    }
}
