import sys

n = int(input())
a = [int(x) for x in input().split()]

if len(a) > len({a[x - 1] for x in range(1, n + 1)}):
    print(-1)
    sys.exit(0)

m = {}
for x in range(1, n + 1):
    m[a[x - 1]] = x

c = [0 for x in range(1, n + 1)]

for x in range(n):
    if n % 2 == 1:
        c[x] = m.pop(a[x])
    else:
        if x == n - 1:
            c[x] = m.pop(a[x])
        else:
            c[x] = m.popitem()[1]

for x in range(n):
    print(c[x], end=' ')
