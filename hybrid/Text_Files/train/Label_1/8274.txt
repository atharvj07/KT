#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1005;
int n, k, m, s;
int c[MAXN], p[MAXN], r[MAXN], choose[MAXN];
int dp[MAXN][MAXN];
vector<pair<int, int>>ans;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> c[i] >> p[i]; }
    cin >> k;
    for (int i = 1; i <= k; i++) { cin >> r[i]; }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= k; j++) {
            dp[i][j] = dp[i - 1][j];
            if (c[i] <= r[j]) { dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] + p[i]); }
        }
    }
    for (int i = n, j = k; i >= 1; i--) {
        if (dp[i][j] != dp[i - 1][j]) {
            choose[i] = j;
            j--;
        }
    }
    for (int i = 1; i <= n; i++) {
        if (choose[i]) {
            m++;
            ans.push_back(make_pair(i, choose[i]));
            s += p[i];
        }
    }
    cout << m << " " << s << endl;
    for (auto x : ans) { cout << x.first << " " << x.second << endl; }
    return 0;
}
