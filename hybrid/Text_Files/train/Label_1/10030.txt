#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

int main() {
  long long int n, tmp, ans;
  std::cin >> n;
  std::vector<long long int> arr;
  while (n--) {
    std::cin >> tmp;
    arr.emplace_back(tmp);
  }
  ans = *std::max_element(arr.begin(), arr.end(),
                         [](long long int a, long long int b) -> bool {
                           if (a == b) return false;
                           long long int sr = sqrt(a), sb = sqrt(b);
                           return sr * sr == a && sb * sb != b;
                         });
  std::cout << ans << std::endl;
  return 0;
}
