n = int(input())
p = list(map(int, input().split()))
f = [0] * (n + 1)
for i in range(n - 1, -1, -1):
    for j in range(i + 1):
        if i == n - 1:
            f[n - i] = 0
        elif p[j] == i + 1:
            f[n - i] = f[n - i - 1]
            break
        else:
            f[n - i] = f[n - i - 1] + 1
            p[j], p[j + 1] = p[j + 1], p[j]
for i in f:
    print(i)
