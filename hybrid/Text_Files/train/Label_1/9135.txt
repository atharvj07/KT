#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, res;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    int d = a[1] - a[0];
    bool ok = true;
    for (int i = 1; i < n - 1; ++i) {
        int j = a[i + 1] - a[i];
        ok &= j == d;
    }
    res = a.back() + d;
    if (!ok) res = a.back();
    cout << res;
}
