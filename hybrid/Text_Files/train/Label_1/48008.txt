import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            int sum = 0;
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
                sum += a[j];
            }
            int ans = sum / n;
            if (sum % n != 0) {
                ans++;
            }
            System.out.println(ans);
        }
    }
}
