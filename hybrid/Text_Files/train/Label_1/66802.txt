import sys
sys.setrecursionlimit(1000000)
n=int(input())
a=list(map(int,input().split()))
g=[[]]
for i in range(n-1):
    x,y=list(map(int,input().split()))
    g[x-1].append(y-1)
    g[y-1].append(x-1)
ans=[0]*n
def dfs(i,p,mn):
    ans[i]=mn
    mn=max(mn,a[i])
    for j in g[i]:
        if j==p:continue
        dfs(j,i,mn)
dfs(0,-1,a[0])

def dfs2(i,p,mn):
    ans[i]=max(ans[i],mn)
    for j in g[i]:
        if j==p:continue
        dfs2(j,i,max(mn,a[i]))
dfs2(0,-1,a[0])
print(*ans)
