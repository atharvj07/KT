#include <iostream>
#include <vector>
using namespace std;

const int digits[] = {6, 2, 5, 5, 4, 5, 6, 3, 7, 6};
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        for (int i = 9; i >= 0; i--) {
            if (n >= digits[i]) {
                cout << i;
                n -= digits[i];
                break;
            }
        }
        for (int i = 0; i < n; i++) cout << 8;
        cout << endl;
    }
    return 0;
}
