import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int b1 = input.nextInt();
        int q = input.nextInt();
        int l = input.nextInt();
        int m = input.nextInt();

        Set<Integer> bad = new HashSet<>();
        for (int i = 0; i < m; i++) {
            bad.add(input.nextInt());
        }

        long bi = b1;
        int count = 0;
        while (true) {
            if (bi <= l && bi >= -l && !bad.contains(bi)) {
                count++;
            } else {
                break;
            }
            bi *= q;
        }

        if (bi <= l && q == 1) {
            System.out.println("inf");
        } else {
            System.out.println(count);
        }
    }
}
