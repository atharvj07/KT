for t in range(int(input())):
    n = int(input())
    a = [int(i) for i in input().split()]
    b = []
    ans = 1
    f = 0
    for i in range(1, n):
        if a[i] == a[i - 1]:
            b.append(a[i - 1])
            f = 1
        else:
            if f == 1:
                if len(b) % 2 == 0:
                    b.append(a[i - 1])
                b.reverse()
                ans += len(b)
            b = [a[i]]
            f = 0
    if f == 1:
        if len(b) % 2 == 0:
            b.append(a[n - 1])
        b.reverse()
        ans += len(b)
    print(ans)
