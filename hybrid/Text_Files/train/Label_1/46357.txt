import java.util.*;
import java.lang.*;

class Main {
    static int N, K, M, a[][], vis[], cur[];
    static long ans = Long.MAX_VALUE;

    static void dfs(int v, int cnt, long sum) {
        if (cnt == K) {
            ans = Math.min(ans, sum);
            return;
        }
        for (int u = 1; u <= N; u++) {
            if (vis[u] == 0 && (cur[u] == 0 || u == v)) {
                cur[v] = cur[u] = 1;
                vis[u]++;
                dfs(u, cnt + 1, sum + a[v][u]);
                vis[u]--;
                cur[v] = cur[u] = 0;
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        K = sc.nextInt();
        M = sc.nextInt();
        a = new int[N + 1][N + 1];
        vis = new int[N + 1];
        cur = new int[N + 1];
        for (int i = 0; i < M; i++) {
            int x = sc.nextInt(), y = sc.nextInt(), c = sc.nextInt();
            a[x][y] = a[y][x] = c;
        }
        dfs(1, 1, 0);
        if (ans > 1e9)
            System.out.println(-1);
        else
            System.out.println(ans);
    }
}
