MOD = 10**9 + 7

def mod_inv(x, mod=MOD):
    return pow(x, mod - 2, mod)

def solve(n, T, times):
    prefix_sum = [0] * (n + 1)
    for i in range(n):
        prefix_sum[i + 1] = prefix_sum[i] + times[i]

    total_ways = pow(2, n, MOD)
    ans = 0
    for i in range(n + 1):
        time_required = prefix_sum[i] + i
        if time_required <= T:
            remaining_time = T - time_required
            additional = min(remaining_time, n - i)
            prob = (pow(2, additional, MOD) if additional >= 0 else 1)
            ans += i * prob
            ans %= MOD
        else:
            break

    return ans * mod_inv(total_ways) % MOD

n, T = map(int, input().split())
times = list(map(int, input().split()))
print(solve(n, T, times))
