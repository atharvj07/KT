import java.util.Scanner;

public class Main {
    static int[][] dp;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            String s = scanner.next();
            dp = new int[s.length() + 1][2];
            System.out.println(solve(s, a, b, 0, 0));
        }
    }

    public static int solve(String s, int a, int b, int index, int state) {
        if (index >= s.length()) {
            return 0;
        }
        if (dp[index][state] != 0) {
            return dp[index][state];
        }
        int ans;
        if (state == 0) {
            // place
            if (s.charAt(index) == '0') {
                ans = b + solve(s, a, b, index + 1, 1);
            } else {
                ans = solve(s, a, b, index + 1, 0);
            }
            // activate
            if (s.charAt(index) == '1') {
                ans = Math.min(ans, a + solve(s, a, b, index + 1, 0));
            }
        } else {
            // dont place
            ans = solve(s, a, b, index + 1, 1);
            // place
            if (s.charAt(index) == '0') {
                ans = Math.min(ans, b + solve(s, a, b, index + 1, 1));
            }
            // activate
            if (s.charAt(index) == '1') {
                ans = Math.min(ans, a + solve(s, a, b, index + 1, 0));
            }
        }
        dp[index][state] = ans;
        return ans;
    }
}
