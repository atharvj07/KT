import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = sc.nextInt();
        }
        int[][] ratios = new int[m][n];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                ratios[i][j] = b[i] / a[j];
            }
        }
        Arrays.sort(ratios, (o1, o2) -> o2[0] - o1[0]);
        int maxRatio = ratios[0][0];
        int count = 0;
        for (int i = 0; i < m; i++) {
            if (ratios[i][0] == maxRatio) {
                count++;
            } else {
                break;
            }
        }
        System.out.println(count);
    }
}
