#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 998244353
ll dp[1000001], pre[1000001];

int main() {
  int n;
  cin >> n;
  vector<int> v(n + 1);
  for (int i = 1; i <= n; i++) { cin >> v[i]; }
  dp[n] = pre[n] = 1;
  for (int i = n - 1; i >= 1; i--) {
    if (v[i] + 1 == v[i + 1]) {
      dp[i] = pre[i + 2];
      dp[i] = (dp[i] - dp[i + 1] + mod) % mod;
    } else
      dp[i] = 0;
    pre[i] = (pre[i + 1] + dp[i]) % mod;
  }
  cout << pre[1];
  return 0;
}
