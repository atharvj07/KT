from collections import Counter
from math import factorial

MOD = 998244353

def mod_factorial(n):
    return factorial(n) % MOD

def solve(n, pairs):
    total_perms = mod_factorial(n)
    first_sorted = sorted(pairs)
    second_sorted = sorted(pairs, key=lambda x: (x[1], x[0]))
    
    first_counter = Counter(first_sorted)
    second_counter = Counter(second_sorted)
    
    first_bad = all(first_sorted[i][0] <= first_sorted[i + 1][0] for i in range(n - 1))
    second_bad = all(second_sorted[i][1] <= second_sorted[i + 1][1] for i in range(n - 1))
    
    first_perms = second_perms = 1
    if first_bad:
        for key in first_counter:
            first_perms = (first_perms * mod_factorial(first_counter[key])) % MOD
    if second_bad:
        for key in second_counter:
            second_perms = (second_perms * mod_factorial(second_counter[key])) % MOD
    
    both_sorted = sorted(pairs)
    both_counter = Counter(both_sorted)
    both_bad = first_bad and second_bad and all(both_sorted[i] == both_sorted[i + 1] for i in range(n - 1))
    
    both_perms = 1
    if both_bad:
        for key in both_counter:
            both_perms = (both_perms * mod_factorial(both_counter[key])) % MOD
    
    if not first_bad and not second_bad:
        return total_perms
    if both_bad:
        return (total_perms - first_perms - second_perms + both_perms) % MOD
    return (total_perms - first_perms - second_perms) % MOD

n = int(input())
pairs = [tuple(map(int, input().split())) for _ in range(n)]
print(solve(n, pairs))
