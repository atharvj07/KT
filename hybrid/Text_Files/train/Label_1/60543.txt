def go(arr, n, d):
    c = arr.copy()
    arr.reverse()
    for i in range(n-1,n-1-d,-1):
        arr[i%n]&=c[i%n]
    arr.reverse()
    for i in arr:
        if i>0: return -1
    return 1

for i in range(int(input())):
    n, d=map(int, input().split())
    arr=list(map(int, input().split()))
    print(go(arr,n,d))
