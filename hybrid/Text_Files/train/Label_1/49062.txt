import java.util.*;

class Main {
    private static long f, t, t0, a1, t1, p1, a2, t2, p2;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        f = in.nextLong();
        t = in.nextLong();
        t0 = in.nextLong();
        a1 = in.nextLong();
        t1 = in.nextLong();
        p1 = in.nextLong();
        a2 = in.nextLong();
        t2 = in.nextLong();
        p2 = in.nextLong();

        boolean done = false;
        long cost = 0;
        int times = 0;
        while (!done) {
            if (a1 == 0 && a2 == 0) {
                if (f % t0 == 0 && f / t0 <= t) {
                    cost = (f / t0) * p1;
                    done = true;
                } else if (f % t0 == 0 && (f / t0 + 1) * t0 <= t) {
                    cost = (f / t0 + 1) * p1;
                    done = true;
                } else {
                    cost = -1;
                    done = true;
                }
            } else if (a1 != 0 && a2 == 0) {
                while (f > 0) {
                    if (f >= a1 && (f - a1) / t0 + t1 <= t) {
                        f -= a1;
                        t -= t1;
                        cost += p1;
                    } else {
                        times = (int) (Math.ceil((f - a1) / t0));
                        if (times + t1 <= t) {
                            cost += times * p1;
                            done = true;
                            f = 0;
                            t -= times * t1;
                        } else if (a1 / t0 + t1 <= t) {
                            cost += p1;
                            done = true;
                            f -= a1;
                            t -= t1;
                        } else {
                            cost = -1;
                            done = true;
                        }
                    }
                }
            } else if (a1 == 0 && a2 != 0) {
                while (f >= a2) {
                    f -= a2;
                    t -= t2;
                    cost += p2;
                }
                if (f > 0 && a2 / t0 + t2 <= t) {
                    cost += p2;
                    done = true;
                    f -= a2;
                    t -= t2;
                } else if (f > 0) {
                    cost = -1;
                    done = true;
                }
            } else if (a1 != 0 && a2 != 0) {
                while (f > a2 && (f - a2) / t0 + t2 <= t) {
                    f -= a2;
                    t -= t2;
                    cost += p2;
                }
                if (f >= a2 && a2 / t0 + t2 <= t) {
                    cost += p2;
                    f -= a2;
                    t -= t2;
                } else if (f < a2 && a1 / t0 + t1 <= t) {
                    cost += p1;
                    f -= a1;
                    t -= t1;
                } else {
                    times = (int) (Math.ceil((f - a2) / t0));
                    if (times * t1 + t2 <= t && f - a2 >= 0) {
                        cost += p2 + times * p1;
                        done = true;
                        f -= a2 + times * a1;
                        t -= t2 + times * t1;
                    } else if (a2 / t0 + t2 <= t) {
                        cost += p2;
                        done = true;
                        f -= a2;
                        t -= t2;
                    } else if (a1 / t0 + t1 <= t) {
                        cost += p1;
                        done = true;
                        f -= a1;
                        t -= t1;
                    } else {
                        cost = -1;
                        done = true;
                    }
                }
            }
        }
        System.out.println(cost);
    }
}
