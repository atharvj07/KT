import sys

input = sys.stdin.readline

n = int(input().strip())
a = list(map(int, input().strip().split()))
assert(len(set(a)) == n)

ans = 0
ans_id = 0
mn = 10**9

for i in range(n):
    cur = sum(abs(a[j] - (j + 1)) for j in range(n))
    if cur < mn:
        mn = cur
        ans = i

print(mn, ans)
