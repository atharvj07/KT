#include <bits/stdc++.h>
using namespace std;
typedef unsigned long long ll;
const ll m = 1000000007;
ll t, p, a[1010], b[1010], f[1010];
int main() {
  cin >> p;
  for (int i = 0; i < p; i++) cin >> a[i];
  b[p - 1] = 1;
  for (int i = p - 2; i >= 0; i--) b[i] = b[i + 1] * (a[i] + 1) % m;
  f[0] = 1;
  for (int i = 1; i <= p; i++) {
    f[i] = f[i - 1] * i;
    if (f[i] >= m) f[i] %= m;
  }
  t = 1;
  for (int i = 0; i < p; i++) {
    t = t * f[a[i]] % m;
  }
  t = t * b[0] % m;
  cout << t;
  return 0;
}
