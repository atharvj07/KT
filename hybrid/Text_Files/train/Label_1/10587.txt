#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        int sum = 0;
        for (int &x : a) { cin >> x, sum += x; }
        assert(sum == 0);
        long long ans = 0;
        for (auto val : a) {
            if (val < 0) {
                int tmp = abs(val);
                if (tmp % 2) {
                    ans++;
                    tmp++;
                }
                ans += tmp / 2;
            }
        }
        cout << ans << endl;
    }
}
