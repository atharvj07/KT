import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long a = input.nextLong();
        long b = input.nextLong();
        long n = input.nextLong();
        System.out.println(a + b <= n + 1 ? a + b : 0);
    }
}
