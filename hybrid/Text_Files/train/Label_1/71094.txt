import sys,heapq
input=sys.stdin.readline
n,k=map(int,input().split())
G=[[]for _ in range(n)]
for _ in range(n-1):
    u,v=map(int,input().split())
    G[u-1].append(v-1)
    G[v-1].append(u-1)
D=[-1]*n
H=[]
def dijkstra(u):
    H.append((0,u))
    while H:
        d,u=heapq.heappop(H)
        if D[u]!=-1:continue
        D[u]=d
        for v in G[u]:
            if D[v]==-1:heapq.heappush(H,(d+1,v))
dijkstra(0)
ans=0
for d in D:
    if d!=-1:ans+=d
print(ans*(ans+1)//2)
