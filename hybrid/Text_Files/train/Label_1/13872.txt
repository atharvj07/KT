#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const ll mod = 1e9 + 7;

int main() {
  ll n, b, k, x;
  cin >> n >> b >> k >> x;
  vector<ll> freq(10);
  ll sum = 0;
  for (ll i = 0; i < n; i++) {
    ll d;
    cin >> d;
    freq[d]++;
    sum += d;
  }
  vector<ll> dp(k + 1);
  dp[0] = 1;
  for (ll i = 0; i < 10; i++) {
    for (ll rem = k; rem >= 0; rem--) {
      ll cur = (rem + i * freq[i] % x) % x;
      dp[cur] = (dp[cur] + dp[rem] * freq[i]) % mod;
    }
  }
  cout << dp[k] << endl;
  return 0;
}
