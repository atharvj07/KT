import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            if (n % 2 == 0) {
                for (int j = 1; j <= n; j++) {
                    System.out.print(2 * j - 1 + " ");
                }
            } else {
                for (int j = 1; j < n; j++) {
                    System.out.print(2 * j - 1 + " ");
                }
                System.out.print(n);
            }
            System.out.println();
        }
    }
}
