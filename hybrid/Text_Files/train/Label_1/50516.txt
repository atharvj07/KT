import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int[] a = new int[6];
        for (int i = 0; i < 6; i++) {
            a[i] = in.nextInt();
        }

        int n = in.nextInt();
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = in.nextInt();
        }

        int min = Integer.MAX_VALUE;
        int max = Integer.MIN_VALUE;
        for (int i = 0; i < n; i++) {
            int fret = b[i] - a[0];
            min = Math.min(min, fret);
            max = Math.max(max, fret);
        }

        System.out.println(max - min);
    }
}
