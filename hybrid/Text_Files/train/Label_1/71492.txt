import sys
input = lambda: sys.stdin.readline().strip()

mod = 1000000007

for _ in range(int(input())):
    n, s = map(int, input().split())
    a = []
    for i in range(n):
        l, r = map(int, input().split())
        a.append([l, r])
    a.sort(key = lambda x: x[0])

    l, r = 0, 0
    for i in range(n):
        l += a[i][0]
        r += a[i][1]

    if l <= s:
        ans = a[n // 2][1]
    else:
        l -= s
        mn = min(a[n // 2][0] - l, a[n // 2][1] - a[n // 2][0])
        ans = a[n // 2][0] + mn

    print(ans)
