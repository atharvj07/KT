#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5+10;
string s[MAXN];
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> s[i];
    }
    string res = s[0];
    for (int i = 1; i < n; i++) {
        int l = 0, r = s[i].length()-1;
        while (l < res.length() && r >= 0 && res[l] == s[i][r]) {
            l++;
            r--;
        }
        res += s[i].substr(r+1);
    }
    cout << res << endl;
    return 0;
}
