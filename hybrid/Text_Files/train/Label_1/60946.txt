def find_char(n, k):
    f_0_len = 75
    prefix = "What are you doing while sending "
    suffix = "? Are you busy? Will you send "
    prefix_len = len(prefix)
    suffix_len = len(suffix) + 2  # accounting for the quotes
    if n == 0:
        if k > f_0_len:
            return '.'
        else:
            return "What are you doing at the end of the world? Are you busy? Will you save us?"[k - 1]
    lengths = [f_0_len]
    for i in range(1, n + 1):
        next_len = 2 * lengths[i - 1] + prefix_len + suffix_len
        if next_len > 10**18:  # Avoid integer overflow
            next_len = 10**18 + 1
        lengths.append(next_len)
        if lengths[i] >= k:
            break
    for i in range(n, 0, -1):
        if k <= prefix_len:
            return prefix[k - 1]
        k -= prefix_len
        if k <= lengths[i - 1]:
            return find_char(i - 1, k)
        k -= lengths[i - 1]
        if k <= 1:
            return '"'
        k -= 1
        if k <= lengths[i - 1]:
            return find_char(i - 1, k)
        k -= lengths[i - 1]
        if k <= suffix_len - 2:
            return suffix[k - 1]
        k -= (suffix_len - 2)
        return '.'
    return '.'

q = int(input())
answers = []
for _ in range(q):
    n, k = map(int, input().split())
    answers.append(find_char(n, k))
print(''.join(answers))
