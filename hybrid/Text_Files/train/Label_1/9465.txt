#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll mod=1e9+7;
int main()
{
  ll t;
  cin>>t;
  while(t--)
  {
    ll n;
    cin>>n;
    vector<vector<ll>>a(2,vector<ll>(n));
    for(ll i=0;i<2;i++)
    for(ll j=0;j<n;j++)
    cin>>a[i][j];
    vector<vector<ll>>dp(2,vector<ll>(n,0));
    dp[0][0]=a[0][0];
    for(ll i=1;i<n;i++)
    {
      dp[0][i]=a[0][i]+dp[0][i-1];
    }
    dp[1][0]=a[1][0];
    for(ll i=1;i<n;i++)
    dp[1][i]=a[1][i]+dp[1][i-1];
    ll ans=dp[1][n-1];
    for(ll i=0;i<n-1;i++)
    {
      ll val=a[1][i];
      if(i!=0)
      val+=dp[0][i-1];
      ll rem1=dp[0][n-1]-val,rem2=dp[1][n-1]-val;
      ans=max(ans,val+min(rem1,rem2));
    }
    cout<<ans<<endl;
  }
}
