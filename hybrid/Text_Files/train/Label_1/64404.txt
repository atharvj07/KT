import sys
input = sys.stdin.readline

n = int(input())
s = input().strip()
a = []
b = []
for i in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)

ans = 0
for i in range(n):
    if s[i] == '1':
        for j in range(i+1, n):
            if s[j] == '1':
                if (b[i]-b[j])%a[j] == 0 or (b[j]-b[i])%a[i] == 0:
                    ans += 1
print(ans)
