import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    a = [list(map(int, input().split())) for _ in range(n)]
    for j in range(m):
        a[0][j], a[n//2][j] = a[n//2][j], a[0][j]
        if n%2: a[n-1][j], a[(n+1)//2][j] = a[(n+1)//2][j], a[n-1][j]     
    print(sum(max(row) for row in a))
