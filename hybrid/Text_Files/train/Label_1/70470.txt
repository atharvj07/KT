import sys

def solve():
    n, k = [int(x) for x in input().split()]
    cars_start = [0]*k
    cars_end = [0]*k
    for r in range(1, 5):
        line = [int(x) for x in input().split()]
        for i in range(n):
            x = line[i]
            if x != 0:
                if r == 1 or r == 4:
                    cars_end[x-1] = (r, i)
                else:
                    cars_start[x-1] = (r, i)
    graph = [[0]*51 for _ in range(51)]
    for i in range(k):
        sr, sc = cars_start[i]
        er, ec = cars_end[i]
        graph[sr][sc] += 1
        graph[er][ec] -= 1
        for j in range(sr+1, er+1):
            graph[j][sc] += 2
        for j in range(sc+1, ec+1):
            graph[sr][j] += 2
        graph[sr][ec+1] -= 1
        graph[er+1][sc] -= 1
    flow = [0]*51
    for r in range(1, 5):
        for c in range(1, n+1):
            flow[r] += graph[r][c]
            flow[c+n] -= graph[r][c]
    for c in range(1, n+k+1):
        if flow[c] != 0:
            return -1
    vis = [False]*51
    matching = [-1]*51
    path = [0]*51
    def dfs(cur):
        if vis[cur]:
            return False
        vis[cur] = True
        for nxt in range(1, 51):
            if graph[cur][nxt] > 0 and (matching[nxt] == -1 or dfs(matching[nxt])):
                matching[nxt] = cur
                path[cur] = nxt
                return True
        return False
    ans = []
    for r in range(1, 5):
        for c in range(1, n+1):
            if graph[r][c] == 1:
                if dfs(r):
                    while path[r] != 0:
                        nr, nc = path[r]
                        ans.append((nc-n, nr, r))
                        r = nr
                        c = nc
                else:
                    return -1
    return len(ans), ans
result = solve()
if result == -1:
    print(-1)
else:
    l, ans = result
    print(l)
    for a in ans:
        print(*a)
