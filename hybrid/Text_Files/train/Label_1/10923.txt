#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        if (m % 2 && n % 2 == 0) {
            cout << "NO" << endl;
            continue;
        }
        int a[n][4];
        bool ok = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 4; j++) {
                cin >> a[i][j];
            }
            if (a[i][0] != a[i][3] || a[i][1] != a[i][2]) ok = 0;
        }
        if (ok)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
}
