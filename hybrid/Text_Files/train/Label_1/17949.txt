#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &i : a) cin >> i;
  stack<int> st;
  int l[n], r[n];
  for (int i = 0; i < n; i++) {
    while (!st.empty() && a[st.top()] * a[i] >= 0) st.pop();
    if (st.empty())
      l[i] = -1;
    else
      l[i] = st.top();
    st.push(i);
  }
  while (!st.empty()) st.pop();
  for (int i = n - 1; i >= 0; i--) {
    while (!st.empty() && a[st.top()] * a[i] >= 0) st.pop();
    if (st.empty())
      r[i] = n;
    else
      r[i] = st.top();
    st.push(i);
  }
  long long neg = 0, pos = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] < 0) {
      neg += 1LL * (r[i] - i) * (i - l[i]);
      pos += 1LL * (r[i] - i) * (i - l[i] - 1);
    } else {
      pos += 1LL * (r[i] - i) * (i - l[i]);
      neg += 1LL * (r[i] - i) * (i - l[i] - 1);
    }
  }
  cout << neg << " " << pos << "\n";
}
