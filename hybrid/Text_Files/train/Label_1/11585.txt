#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int l = -1, r = -1;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '0') {
        if (i - k > l && i + k < r)
          continue;
        ans++;
        l = i - k;
        r = i + k;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
