#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=3e5+5;
ll a[maxn],c[maxn];
int main()
{
	int n,k;
	cin>>n>>k;
	for(int i=1;i<=n;i++) cin>>a[i];
	for(int i=1;i<=n-1;i++) c[i]=a[i+1]-a[i];
	sort(c+1,c+n);
	ll ans=0;
	for(int i=1;i+k-2<=n-1;i++) ans+=c[i];
	cout<<ans<<endl;
	return 0;
}
