#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 998244353;
int main() {
    int n, m;
    cin >> n >> m;
    string a, b;
    cin >> a >> b;
    ll ans = 0;
    for (; m; m--) {
        ll t = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == '1' && b[i] == '1') t |= 1ll << (i);
        }
        ans = (ans + t % mod) % mod;
        string c;
        for (int i = 0; i < m - 1; i++) { c += b[i]; }
        for (int i = m - 1; i < m; i++) {
            if (b[i] == '1') c += '0';
            else c += '1';
        }
        b = c;
    }
    cout << ans << endl;
    return 0;
}
