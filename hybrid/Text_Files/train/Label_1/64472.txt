n, m = map(int, input().split())
a = []
for i in range(n):
    a.append(input())

ans = "YES"
x = -1
y = -1
for i in range(n):
    for j in range(m):
        if a[i][j] == 'B':
            if x == -1:
                x, y = i, j
            elif abs(i - x) + abs(j - y) > 1:
                ans = "NO"
print(ans)
