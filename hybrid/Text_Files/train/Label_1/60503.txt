import sys
input = sys.stdin.readline

n, m = map(int,input().split())
a = list(map(int,input().split()))
b = sorted(((a[i]-i) % m for i in range (n)), reverse=True)
ans = 0
for k in b:
    need = (k+(m-1))%m
    if need <= k: continue
    ans+=need-k
    for i in range(n):
        a[i] = (a[i] - (a[i]-i)%m + need)%m
print(ans)
print(*a)
