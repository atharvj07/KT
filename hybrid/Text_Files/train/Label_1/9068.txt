#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        string s;
        cin >> s;
        sort(s.begin(), s.end());
        vector<int> freq(26);
        for (char c : s) freq[c - 'a']++;
        int ans = 0;
        for (int i = 0; i < 26; i++) {
            if (freq[i] >= k) ans += freq[i] / k * k;
            else if (freq[i] >= 1) ans += freq[i];
        }
        cout << ans << endl;
    }
    return 0;
}
