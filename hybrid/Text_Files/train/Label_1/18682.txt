#include<bits/stdc++.h>
using namespace std;
const int maxn=200000+10;
int v[maxn],b[maxn];
int main(){
    int n;cin>>n;
    for(int i=1;i<=n;i++)
        cin>>b[i];
    for(int i=1;i<n;i++){
        v[i]+=b[i+1]-b[i];
    }
    int ans=0;
    for(int i=1;i<n;i++){
        if(v[i]-v[i-1]!=b[i]){
            ans++;
            v[i]++;
        }
    }
    cout<<ans;
}
