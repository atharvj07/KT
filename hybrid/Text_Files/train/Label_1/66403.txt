t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    sorted_a = sorted(a)
    actions = []
    for i in range(n):
        if a == sorted_a:
            break
        if a[i] != sorted_a[i]:
            for j in range(i + 1, n):
                if a[j] == sorted_a[i]:
                    d = j - i
                    a = a[:i] + a[i:i+d+1][d:] + a[i:i+d+1][:d] + a[i+d+1:]
                    actions.append((i + 1, j + 1, d))
                    break
    print(len(actions))
    for action in actions:
        print(*action)
