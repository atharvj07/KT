import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class DotGame {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int x = Integer.parseInt(line[0]);
        int y = Integer.parseInt(line[1]);
        int n = Integer.parseInt(line[2]);
        int d = Integer.parseInt(line[3]);

        int[][] vectors = new int[n][2];
        for (int i = 0; i < n; i++) {
            line = reader.readLine().split(" ");
            vectors[i][0] = Integer.parseInt(line[0]);
            vectors[i][1] = Integer.parseInt(line[1]);
        }

        boolean antonWins = false;
        for (int[] vector : vectors) {
            int newX = x + vector[0];
            int newY = y + vector[1];
            if (Math.sqrt(newX * newX + newY * newY) > d) {
                antonWins = true;
                break;
            }
        }

        if (antonWins) {
            System.out.println("Anton");
        } else {
            System.out.println("Dasha");
        }
    }
}
