#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  vector<int> pos(26);
  for (int i = 0; i < 26; i++) {
    pos[s[i] - 'a'] = i;
  }
  int ans = 0;
  int cur = 0;
  for (int i = 0; i < n; i++) {
    int next = pos[s[i] - 'a'];
    ans += min(abs(next - cur), 26 - abs(next - cur));
    cur = next;
  }
  cout << ans << endl;
  return 0;
}
