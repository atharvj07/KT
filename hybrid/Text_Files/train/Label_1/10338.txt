#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mx = 1e9;
main() {
  int w;
  cin >> w;
  vector<int> cnt(9);
  for (int &x : cnt) cin >> x;
  vector<int> dp(w + 1);
  for (int i = 1; i <= 8; i++) {
    for (int j = w; j >= i; j--) {
      dp[j] = max(dp[j], dp[j - i] + i * cnt[i]);
      if (dp[j] >= mx) dp[j] = mx;
    }
  }
  cout << dp[w] << "\n";
}
