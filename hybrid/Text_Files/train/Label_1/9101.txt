#include <cstdio>

int main(int argc, char const* argv[]) {
    int x[3], y[3], s = 0, dx, dy;
    for (int i = 0; i < 3; i++)
        scanf("%d%d", &x[i], &y[i]);
    dx = (x[0] == x[1]) ? (x[1] == x[2]) : 0;
    dy = (y[0] == y[1]) ? (y[1] == y[2]) : 0;
    s += 2 - (dx + dy);
    if (dx) {
        x[1] = x[2];
        x[2] = x[0];
        x[0] = -2e9;
    }
    if (dy) {
        y[1] = y[2];
        y[2] = y[0];
        y[0] = -2e9;
    }
    if ((x[1] - x[0]) * (x[2] - x[1]) < 0) s++;
    if ((y[1] - y[0]) * (y[2] - y[1]) < 0) s++;
    printf("%d\n", s);
    return 0;
}
