#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (auto &x : v) {
        cin >> x;
    }
    sort(v.rbegin(), v.rend());
    int ans = v[0];
    for (int i = 1; i < n; i++) {
        ans = (ans + 1) / 2;
    }
    cout << ans << endl;
}
