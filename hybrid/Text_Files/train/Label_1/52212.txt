import java.util.*;
import java.io.*;

class Main {
  static int[] p = new int[200001], c = new int[200001];
  static boolean[] f = new boolean[200001];

  public static void main(String[] args) {
    try {
      BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
      int n = Integer.parseInt(br.readLine());
      StringTokenizer st = new StringTokenizer(br.readLine());
      for (int i = 1; i <= n; i++) {
        c[i] = Integer.parseInt(st.nextToken());
        p[c[i]] = i;
      }
      int j = 0;
      for (int i = 1; i <= n; i++) {
        if (p[i] != i) {
          j = i;
          break;
        }
      }
      if (j == 0) {
        System.out.println(0);
      } else {
        System.out.println(n + 1);
        for (int i = 1; i <= n; i++) {
          if (p[i] == i) continue;
          System.out.println(i + " " + p[i]);
          int tmp = p[i];
          p[i] = p[c[i]];
          p[c[i]] = tmp;
          f[i] = !f[i];
          f[c[i]] = !f[c[i]];
        }
        System.out.println(j + " " + p[j]);
        f[j] = !f[j];
        f[p[j]] = !f[p[j]];
        for (int i = 1; i <= n; i++) {
          if (f[i]) System.out.println(i + " " + p[i]);
        }
      }
    } catch (IOException e) {
      e.printStackTrace();
    }
  }
}
