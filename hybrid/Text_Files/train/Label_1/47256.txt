import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        Map<Integer, Integer> map = new HashMap<>();
        for (int x : a) {
            if (x < 3200) {
                int c;
                if (x <= 399) {
                    c = 0;
                } else if (x <= 799) {
                    c = 1;
                } else if (x <= 1199) {
                    c = 2;
                } else if (x <= 1599) {
                    c = 3;
                } else if (x <= 1999) {
                    c = 4;
                } else if (x <= 2399) {
                    c = 5;
                } else if (x <= 2799) {
                    c = 6;
                } else {
                    c = 7;
                }
                map.put(c, map.getOrDefault(c, 0) + 1);
            }
        }

        int min = map.size();
        int max = map.size() + 1;

        System.out.println(min + " " + max);
    }
}
