#include <bits/stdc++.h>

using namespace std;

const int COIN = 0, DIAMOND = 1;

int main() {
  int n, c, d;
  cin >> n >> c >> d;
  vector<pair<int, int>> cnd(n);
  for (int i = 0; i < n; ++i) {
    int b, p;
    cin >> b >> p;
    string s;
    cin >> s;
    if (s[0] == 'C') {
      cnd[i] = {b, COIN};
      c -= p;
    } else if (s[0] == 'D')
      cnd[i] = {b, DIAMOND};
  }
  sort(cnd.rbegin(), cnd.rend());
  int ans = 0;
  for (auto &p : cnd) {
    if (!c && !d) break;
    int b = p.first, type = p.second;
    if ((type == COIN && c >= 0) || (type == DIAMOND && d >= 0)) {
      ans += b;
      if (type == COIN) c--;
      else if (type == DIAMOND) d--;
    }
  }
  cout << ans << endl;
  return 0;
}
