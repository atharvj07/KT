import sys
n = int(sys.stdin.readline())
a = list(map(int, sys.stdin.readline().split()))

dic = {}
for i in a:
    if i not in dic:
        dic[i] = []
    dic[i].append(a.index(i))

for i in a:
    for j in a:
        if i == j:
            continue
        if i + j in dic:
            l1 = dic[i]
            l2 = dic[j]
            l3 = dic[i + j]
            for x in l1:
                for y in l2:
                    for z in l3:
                        if x != y and y != z and z != x:
                            print("YES")
                            print(x + 1, y + 1, z + 1, l3[0] + 1)
                            sys.exit()
print("NO")
