  int h1, h2, m1, m2;
  char c;
  cin >> h1 >> c >> m1 >> h2 >> c >> m2;
  if (h2 < h1 || (h2 == h1 && m2 < m1)) {
    h2 += 24;
  }
  h2 -= h1;
  m2 -= m1;
  if (m2 < 0) {
    m2 += 60;
    h2--;
  }
  if (h2 < 10) {
    cout << "0";
  }
  cout << h2 << ":";
  if (m2 < 10) {
    cout << "0";
  }
  cout << m2 << endl;
  return 0;
}
