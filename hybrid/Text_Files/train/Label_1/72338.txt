from sys import stdin

def read_int():
    return int(stdin.readline())

def read_ints():
    return list(map(int, stdin.readline().split()))

for t in range(read_int()):
    n = read_int()
    print(1 if n == 2 else n * n - 1)
    print(' '.join([str(a) for a in range(1, n + 1)]))
    print(n - 1)
    for a in range(2, n + 1):
        print(1, a)
