import java.io.*;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter pw = new PrintWriter(System.out);

        int n = Integer.parseInt(br.readLine());
        boolean[] used = new boolean[n];
        int l = 0;
        for (int i = 0; i < n; i++) {
            l += br.readLine().length();
        }
        String s = br.readLine();
        int m = s.length();
        int ind = 2;
        for (int i = 0; i < m; i++) {
            char c = s.charAt(i);
            if (c == '<') {
                if (s.charAt(i + 1) == '3') {
                    if (ind < m && s.charAt(ind) == '<') {
                        used[(ind - 2) / 3] = true;
                        ind += 3;
                    }
                } else {
                    pw.printf("no");
                    pw.close();
                    return;
                }
            }
        }
        int cnt = 0;
        for (boolean x : used) {
            if (x) cnt++;
        }
        pw.printf(cnt == n ? "yes" : "no");
        pw.close();
    }
}
