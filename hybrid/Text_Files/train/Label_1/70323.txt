import bisect
import sys

def min_segment(segs):
    l = [s[0] for s in segs]
    r = [s[1] for s in segs]
    l.sort()
    r.sort()
    # l is the reverse of the r.
    # So we can just check whether every seg has common point with [l[i], r[i]].
    res = float('inf')
    for i, seg in enumerate(segs):
        st = seg[0]
        en = seg[1]
        ji = bisect.bisect_right(r, st)
        if ji > 0 and r[ji - 1] >= en:
            res = min(res, r[ji - 1] - st)
        else:
            res = min(res, l[bisect.bisect_left(l, en)] - st)
    return res

for _ in range(int(input())):
    n = int(input())
    segs = [list(map(int, input().split())) for _ in range(n)]
    print(min_segment(segs))
