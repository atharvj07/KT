import java.io.*;
import java.util.*;

class Main {
    static final int INF = 987654321;
    static int n, m;
    static char[][] map, origin;
    static int[][] memo;
    static int[][] dir = new int[][]{{0, 1}, {0, -1}, {1, 0}, {-1, 0}};

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter out = new BufferedWriter(new OutputStreamWriter(System.out));
        StringTokenizer st = new StringTokenizer(in.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        map = new char[n][m];
        origin = new char[n][m];
        memo = new int[n][m * 2];
        for (int i = 0; i < n; i++) {
            origin[i] = in.readLine().toCharArray();
            map[i] = origin[i].clone();
        }

        int ans = dfs(0, 0);
        out.write(ans + "\n");
        StringBuilder sb = new StringBuilder();
        print(0, 0, ans, sb);
        out.write(sb.toString());
        out.flush();
        in.close();
        out.close();
    }

    static int dfs(int r, int cnt) {
        if (r == n) {
            return score();
        } else if (memo[r][cnt] != -1) {
            return memo[r][cnt];
        }

        int ret = dfs(r + 1, cnt);

        int c = r * m + cnt;
        if (map[r][cnt] == '.') {
            ret = Math.max(ret, dfs(r + 1, cnt + 1) + 100);
            if (cnt > 0 || hasAdjacent(r, cnt)) {
                ret = Math.max(ret, dfs(r + 1, cnt + 1) + 200);
            }
        }

        return memo[r][cnt] = ret;
    }

    static void print(int r, int c, int ans, StringBuilder sb) {
        if (r == n) {
            return;
        }

        int c2 = r * m + c;
        int cur = memo[r][c];
        if (cur == ans) {
            if (map[r][c] == '.' && cur == dfs(r + 1, c) + 100) {
                sb.append(String.format("B %d %d\n", r + 1, c + 1));
                map[r][c] = 'B';
            } else if (map[r][c] == '.' && cur == dfs(r + 1, c) + 200) {
                sb.append(String.format("R %d %d\n", r + 1, c + 1));
                map[r][c] = 'R';
            }
        } else {
            if (map[r][c] == '.' && cur == dfs(r + 1, c + 1) + 100) {
                sb.append(String.format("B %d %d\n", r + 1, c + 1));
            } else if (map[r][c] == '.' && cur == dfs(r + 1, c + 1) + 200) {
                sb.append(String.format("R %d %d\n", r + 1, c + 1));
            } else if (map[r][c] != '.') {
                sb.append(String.format("D %d %d\n", r + 1, c + 1));
            }
            map[r][c] = origin[r][c];
        }

        print(r + 1, c2 % m, ans, sb);
    }

    static int score() {
        int ret = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] == 'B') {
                    ret += 100;
                } else if (map[i][j] == 'R') {
                    ret += 200;
                }
            }
        }

        return ret;
    }

    static boolean hasAdjacent(int sr, int sc) {
        for (int d = 0; d < 4; d++) {
            int nr = sr + dir[d][0];
            int nc = sc + dir[d][1];
            if (nr >= 0 && nr < n && nc >= 0 && nc < m && map[nr][nc] == 'B') {
                return true;
            }
        }

        return false;
    }
}
