import java.util.*;
import java.math.BigInteger;

class Main {
    static int n, m, k, inf = 1 << 30;
    static int[] in, a, b, c, p;
    static boolean[] vis;
    static List<Edge>[] edge;

    static class Edge {
        int u, d, w;
        Edge(int u, int d, int w) { this.u = u; this.d = d; this.w = w; }
    }

    static int dijkstra(int s) {
        int[] dis = new int[n + 1];
        Arrays.fill(dis, inf);
        dis[s] = 0;
        PriorityQueue<Edge> pq = new PriorityQueue<>((a, b) -> a.w - b.w);
        pq.offer(new Edge(0, s, 0));
        while (!pq.isEmpty()) {
            Edge e = pq.poll();
            if (e.d == n) return e.w;
            if (vis[e.d]) continue;
            vis[e.d] = true;
            for (Edge ee : edge[e.d]) {
                if (vis[ee.u]) continue;
                if (ee.w + e.w < dis[ee.u]) {
                    dis[ee.u] = ee.w + e.w;
                    pq.offer(new Edge(e.d, ee.u, dis[ee.u]));
                }
            }
        }
        return -1;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        k = in.nextInt();
        a = new int[n + 1];
        b = new int[n + 1];
        c = new int[n + 1];
        vis = new boolean[n + 1];
        p = new int[n + 1];
        edge = new List[n + 1];
        for (int i = 1; i <= n; i++) edge[i] = new ArrayList<Edge>();
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
            b[i] = in.nextInt();
            c[i] = in.nextInt();
            p[i] = -1;
        }
        for (int i = 1; i <= m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            edge[u].add(new Edge(u, v, 0));
        }
        int res = -1;
        for (int i = 1; i <= n; i++) {
            if (p[i] != -1) continue;
            if (k >= a[i]) {
                p[i] = 0;
                k += b[i];
                res = Math.max(res, c[i]);
                for (Edge e : edge[i]) {
                    if (vis[e.u]) continue;
                    int tmp = dijkstra(e.u);
                    if (tmp != -1) {
                        p[e.u] = i;
                        k += b[e.u];
                        res = Math.max(res, tmp + c[e.u]);
                    }
                }
            }
        }
        System.out.println(res);
    }
}
