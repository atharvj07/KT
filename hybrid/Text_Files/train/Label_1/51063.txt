import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        sc.close();

        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (a[i] > 0 && a[j] > 0 && a[k] > 0) {
                        a[i]--;
                        a[j]--;
                        a[k]--;
                        ans++;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
