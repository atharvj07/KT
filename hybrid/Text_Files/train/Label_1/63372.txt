n, m = [int(i) for i in input().split()]
a = [list(input()) for _ in range(n)]
chk = [[-1]*m for _ in range(n)]
dx, dy = [-1,0,1,0], [0,1,0,-1]

def dfs(x, y, d, c):
    chk[x][y] = c
    res = []
    res.append((x,y))
    for i in range(4):
        nx, ny = x+dx[i], y+dy[i]
        if 0<=nx<n and 0<=ny<m:
            if chk[nx][ny] == -1 and a[nx][ny] == a[x][y]:
                res += dfs(nx, ny, d+1, c)
    chk[x][y] = -1
    if d>=4:
        for xx,yy in res:
            chk[xx][yy] = -1
    return res

ans = 'No'
for i in range(n):
    for j in range(m):
        if chk[i][j]==-1:
            if dfs(i, j, 1, 1):
                ans = 'Yes'
print(ans)
