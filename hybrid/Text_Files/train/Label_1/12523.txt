#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0), cout.tie(0);
    int n, ans = 0, sum = 0, cnt = 0;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> sum;
        if (sum % 2) {
            cnt++;
            continue;
        }
        ans += (cnt / 2), cnt %= 2;
    }
    ans += cnt / 2;
    cout << ans << endl;
    return 0;
}
