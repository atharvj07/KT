import java.util.*;
import java.io.*;

class Main {
    static long[] a;
    
    public static void main(String[] args) {
        long ans = 0;
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            int n = Integer.parseInt(br.readLine());
            a = new long[n+2];
            String[] s = br.readLine().split(" ");
            long sum = 0;
            for (int i = 1; i <= n; i++) {
                a[i] = Integer.parseInt(s[i-1]);
                sum += a[i] * i;
            }
            if (n <= 2) {
                ans = sum;
            } else {
                long l = a[1], r = a[n];
                sum -= a[1];
                for (int i = 2; i <= n; i++) {
                    sum += (a[i]-a[i-1]) * (1+i-n);
                }
                ans = sum;
                for (int i = 2; i < n; i++) {
                    sum += (a[i]-a[i-1]) * (n);
                    if (sum > ans) ans = sum;
                }
                sum = 0;
                for (int i = n - 1; i >= 1; i--) {
                    sum += (a[i]-a[i+1]) * i;
                }
                for (int i = n - 2; i >= 1; i--) {
                    sum += (a[i]-a[i+1]) * (i-1);
                    if (sum > ans) ans = sum;
                }
            }
            System.out.println(ans);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
