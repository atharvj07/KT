import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        in.nextLine();
        String[] cards = in.nextLine().split(" ");
        int sereja = 0, dima = 0;
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) {
                sereja += Integer.parseInt(cards[i]);
            } else {
                dima += Integer.parseInt(cards[i]);
            }
        }
        System.out.println(sereja + " " + dima);
    }
}
