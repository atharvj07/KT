#include <bits/stdc++.h>
using namespace std;
#define int long long
int n,x,tot;
string s;
int32_t main() {
    int t; cin >> t;
    while(t--) {
        cin >> n >> x;
        int a[n],sum = 0,ans=0;
        cin >> s;
        for(int i=0;i<n;++i) {
           a[i] = s[i]-'0';
           sum += a[i];
           if(sum == x) ans++;
        }
        if(sum != x) {
            cout << 0 << '\n';
            continue;
        }
        sum = 0;
        int b[n];
        for(int i=n-1;i>=0;--i) {
            sum += a[i];
            b[i] = sum;
        }
        int cnt = 0;
        for(int i=0;i<n;++i) {
            int l = i,r = n-1,id = -1;
            while(l<=r) {
               int mid = (l+r)/2;
               if(b[mid]-sum+a[i] >= x) id=mid, r=mid-1;
               else l=mid+1;
            }
            if(id != -1) cnt += n-id;
        }
        ans += cnt;
        if(ans != 0) cout << ans << '\n';
        else cout << -1 << '\n';
    }
}
