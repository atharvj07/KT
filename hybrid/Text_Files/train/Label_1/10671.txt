#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    int l[n], d[n];
    for (int i = 0; i < n; i++) cin >> l[i];
    for (int i = 0; i < n; i++) cin >> d[i];
    sort(l, l + n);
    sort(d, d + n);
    int ans = 0, cnt = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (l[i] == l[n - 1]) {
            if (cnt > n / 2) break;
            cnt++;
        } else {
            ans += d[i];
            cnt++;
        }
    }
    cout << ans << endl;
}
