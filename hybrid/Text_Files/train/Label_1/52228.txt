import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = input.nextInt();
        }
        int[] cntA = new int[6];
        int[] cntB = new int[6];
        for (int i = 0; i < n; i++) {
            cntA[a[i]]++;
            cntB[b[i]]++;
        }
        int ans = 0;
        for (int i = 1; i < 6; i++) {
            if (cntA[i] != cntB[i]) {
                int diff = Math.abs(cntA[i] - cntB[i]);
                if (diff % 2 == 1) {
                    System.out.println(-1);
                    return;
                }
                ans += diff / 2;
            }
        }
        System.out.println(ans);
    }
}
