#include <bits/stdc++.h>

using namespace std;

vector<int> l, r, w;

int main() {
  int n; cin >> n;
  l.assign(n, 0); r.assign(n, 0); w.assign(n, 0);
  for (int i = 0; i < n; i++) cin >> l[i];
  for (int i = 0; i < n; i++) cin >> r[i];

  for (int i = 0; i < n; i++) {
    if (l[i] + r[i] >= n) {
      cout << "NO\n";
      return 0;
    }
  }

  queue<int> q;
  for (int i = 0; i < n; i++) {
    if (l[i] == 0) q.push(i);
  }

  int curL = 0, curR = n - 1;
  while (!q.empty()) {
    int u = q.front(); q.pop();
    w[u] = ++curL;
    curR--;
    if (w[u] > 1 && l[u] != l[u - 1]) {
      cout << "NO\n";
      return 0;
    }
    if (u + 1 < n && r[u] != r[u + 1]) {
      cout << "NO\n";
      return 0;
    }
    if (r[u] > 0) {
      l[u + 1] -= 1;
      r[u + 1] -= 1;
      if (l[u + 1] == 0) q.push(u + 1);
    }
  }

  cout << "YES\n";
  for (int i = 0; i < n; i++) cout << w[i] << " ";
  cout << endl;
}
