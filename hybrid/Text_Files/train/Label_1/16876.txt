#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1000000007;
ll fastpow(ll x, ll p) {
    if (p == 0) return 1;
    if (p == 1) return x;
    ll ans = fastpow(x, p >> 1);
    ans = (ans * ans) % mod;
    if (p & 1) ans = (ans * x) % mod;
    return ans;
}
ll mminvprime(ll x) {
    return fastpow(x, mod - 2);
}
const int N = 2e5 + 10;
int n, k;
vector<int> a;
int main() {
    cin >> n >> k;
    a.resize(n);
    for (int i = 0;i < n;i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = 0;
    int num_same = 0;
    for (int i = 0;i < n;i++) {
        if (i != 0 && a[i] == a[i - 1]) {
            num_same++;
        }
        else num_same = 0;
        if (num_same + 1 >= k) {
            break;
        }
        if (a[i] % 2 == 0) {
            ans++;
            a[i] /= 2;
        }
        else {
            if (a[i] - 1 >= a[i] / 2) {
                ans++;
                a[i] = (a[i] - 1) / 2;
            }
            else {
                ans += 2;
                a[i] /= 2;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
