import java.io.*;
import java.util.*;

public class Main {
    static int n, m;
    static int[] a;
    static int[] last;
    static int[] first;
    static int[] memory;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        a = new int[n + 1];
        last = new int[n + 1];
        first = new int[n + 1];
        memory = new int[n + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
            first[a[i]] = i;
        }
        for (int i = 1; i <= n; i++) {
            if (first[a[i]] == i) {
                memory[i] = 0;
            } else {
                memory[i] = i - last[a[i]];
            }
            memory[i] += memory[i - 1];
            last[a[i]] = i;
        }
        while (m-- > 0) {
            st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            int l = Integer.parseInt(st.nextToken());
            int r = Integer.parseInt(st.nextToken());
            if (type == 2) {
                System.out.println(memory[r] - memory[l - 1]);
            } else {
                int old = a[l];
                a[l] = r;
                if (first[old] == l) {
                    first[old] = -1;
                    memory[l - 1] -= memory[l];
                } else {
                    memory[l - 1] -= l - last[old];
                }
                last[old] = -1;
                if (first[a[l]] == l) {
                    memory[l - 1] += 0;
                } else {
                    memory[l - 1] += l - last[a[l]];
                }
                last[a[l]] = l;
                for (int i = l + 1; i <= n; i++) {
                    if (a[i] == old) {
                        last[old] = i;
                        if (first[old] == -1) {
                            first[old] = i;
                            memory[i - 1] -= memory[i];
                        } else {
                            memory[i - 1] -= i - last[old];
                        }
                        last[old] = i;
                        if (first[a[i]] == i) {
                            memory[i - 1] += 0;
                        } else {
                            memory[i - 1] += i - last[a[i]];
                        }
                        last[a[i]] = i;
                    } else if (a[i] == a[l]) {
                        if (first[a[i]] == i) {
                            memory[i - 1] += 0;
                        } else {
                            memory[i - 1] += i - last[a[i]];
                        }
                        last[a[i]] = i;
                    }
                }
            }
        }
    }
}
