n, m = map(int, input().split())
first_pairs = list(map(int, input().split()))
second_pairs = list(map(int, input().split()))

first_pairs = [(first_pairs[i], first_pairs[i+1]) for i in range(0, len(first_pairs), 2)]
second_pairs = [(second_pairs[i], second_pairs[i+1]) for i in range(0, len(second_pairs), 2)]

common_numbers = set()

for fp in first_pairs:
    for sp in second_pairs:
        common = set(fp).intersection(set(sp))
        if common:
            common_numbers.update(common)

if len(common_numbers) == 1:
    print(common_numbers.pop())
else:
    first_uniques = set()
    second_uniques = set()
    for fp in first_pairs:
        first_uniques.update(fp)
    for sp in second_pairs:
        second_uniques.update(sp)
    
    if common_numbers == first_uniques.intersection(second_uniques):
        print(0)
    else:
        print(-1)
