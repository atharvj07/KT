#include<bits/stdc++.h>
using namespace std;
int main()
{
    int h1,a1,c1,h2,a2,Moves,h1s,h2s;
    cin>>h1>>a1>>c1>>h2>>a2;
    h1s=h1;
    h2s=h2;
    if(a1>=a2)
    {
        Moves=1;
    }
    else if(((h2s%a1)>0&&h2s%c1>0)||(h2s%a1==0))
    {
        Moves=((h2s/a1)+1);
    }
    else if(h2s%a1==0&&h2s%c1==0)
    {
        Moves=h2s/a1;
    }
    cout<<Moves<<endl;
    while(Moves!=0)
    {
        if (h1s>a2)
        {
            h2s=h2s-a1;
            cout<<"STRIKE"<<endl;
        }
        else
        {
            h1s=h1s+c1;
            cout<<"HEAL"<<endl;
        }
        Moves--;
        h1s=h1s-a2;
    }
    return 0;
}
