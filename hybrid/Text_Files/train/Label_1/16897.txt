#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 1001;
int n, a[MAXN][MAXN];
int main() {
    cin >> n;
    int l = 1, r = n + 1, ans = -1;
    while (l <= r) {
        int mid = (l + r) / 2;
        bool ok = 1;
        for (int i = 1; i <= mid; ++i) {
            for (int j = 1; j <= mid; ++j) {
                if (!ok) break;
                for (int ii = 1; ii <= n; ++ii) {
                    for (int jj = ii + 1; jj <= n; ++jj) {
                        if (abs(ii - jj) <= abs(a[i][j] - a[ii][jj])) {
                            ok = 0;
                            break;
                        }
                    }
                }
            }
        }
        if (ok) {
            ans = mid;
            r = mid - 1;
        }
        else l = mid + 1;
    }
    cout << ans << endl;
    for (int i = 1; i <= n; ++i) {
        int r = 1, c = 1;
        for (int j = 1; j <= ans; ++j) {
            for (int jj = 1; jj <= ans; ++jj) {
                if (!a[j][jj]) {
                    r = j;
                    c = jj;
                    a[j][jj] = i;
                    break;
                }
            }
        }
        cout << r << ' ' << c << endl;
    }
    return 0;
}
