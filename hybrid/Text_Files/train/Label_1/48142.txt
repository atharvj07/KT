import java.io.*;
import java.util.*;

public class Main {

    static int calculateHungryMice(int numMice, int numCheese, int y0, int y1, int[] micePositions, int[] cheesePositions) {
        int[][] distance = new int[numMice][numCheese];
        for (int i = 0; i < numMice; i++) {
            for (int j = 0; j < numCheese; j++) {
                distance[i][j] = Math.abs(micePositions[i] - cheesePositions[j]) + Math.abs(y1 - y0);
            }
        }

        Arrays.sort(distance, Comparator.comparingInt(a -> a[0]));
        int cheeseCount = 0;
        int hungryMice = 0;
        for (int[] dist : distance) {
            if (cheeseCount < numCheese) {
                cheeseCount++;
            } else {
                hungryMice++;
            }
        }
        return hungryMice;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] input = reader.readLine().split(" ");
        int numMice = Integer.parseInt(input[0]);
        int numCheese = Integer.parseInt(input[1]);
        int y0 = Integer.parseInt(input[2]);
        int y1 = Integer.parseInt(input[3]);
        
        int[] micePositions = Arrays.stream(reader.readLine().split(" "))
                .mapToInt(Integer::parseInt)
                .toArray();
        
        int[] cheesePositions = Arrays.stream(reader.readLine().split(" "))
                .mapToInt(Integer::parseInt)
                .toArray();

        int hungryMice = calculateHungryMice(numMice, numCheese, y0, y1, micePositions, cheesePositions);
        System.out.println(hungryMice);
    }
}
