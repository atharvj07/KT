n = int(input().strip())
s = input().strip()

# count the number of each character
zeros = s.count('z')
ones = s.count('o')

# compute the maximum possible number
num = []
while zeros > 0:
    num.append(0)
    zeros -= 1
while ones > 0:
    num.append(1)
    ones -= 1

# print the maximum possible number
print(' '.join(str(x) for x in num))
