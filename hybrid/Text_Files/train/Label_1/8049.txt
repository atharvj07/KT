#include <iostream>
int h, n, r, l, p, c, ans;
int main() {
  std::cin >> h >> n;
  for (c = 0; c < h; c++) {
    r = n % 2;
    n /= 2;
    if (n != p) {
      l++;
      ans++;
    }
    p = n;
    if (l == 2) {
      ans += 2;
      l = 0;
    }
    ans += r;
  }
  std::cout << ans << std::endl;
}
