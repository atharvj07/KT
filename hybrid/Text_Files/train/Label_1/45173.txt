import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        ArrayList<Edge>[] adj = new ArrayList[N + 1];
        for (int i = 0; i <= N; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < M; i++) {
            int L = sc.nextInt();
            int R = sc.nextInt();
            int C = sc.nextInt();
            adj[L].add(new Edge(R, C));
        }
        long[] dist = dijkstra(1, adj, N);
        if (dist[N] == Long.MAX_VALUE) System.out.println(-1);
        else System.out.println(dist[N]);
    }

    public static long[] dijkstra(int s, ArrayList<Edge>[] adj, int N) {
        long[] dist = new long[N + 1];
        Arrays.fill(dist, Long.MAX_VALUE);
        dist[s] = 0;
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        pq.add(new Edge(s, 0));
        while (!pq.isEmpty()) {
            Edge cur = pq.poll();
            if (cur.w > dist[cur.to]) continue;
            for (Edge next : adj[cur.to]) {
                long newDist = cur.w + next.w;
                if (newDist < dist[next.to]) {
                    dist[next.to] = newDist;
                    pq.add(new Edge(next.to, newDist));
                }
            }
        }
        return dist;
    }

    public static class Edge implements Comparable<Edge> {
        public int to;
        public long w;
        public Edge(int t, long w) {
            this.to = t;
            this.w = w;
        }
        public int compareTo(Edge o) {
            return Long.compare(this.w, o.w);
        }
    }
}
