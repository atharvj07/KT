#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'h' && i + 5 < n) {
      if (s.substr(i, 6) == "heavy") {
        ans += n - i - 5;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
