import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();

        for (int i = 0; i < t; i++) {
            int n = input.nextInt();
            int[] b = new int[n];
            for (int j = 0; j < n; j++) {
                b[j] = input.nextInt();
            }

            int[] a = new int[n];
            a[0] = b[n - 1];
            a[n - 1] = b[0];

            for (int j = 1; j < n - 1; j++) {
                if (j % 2 == 1) {
                    a[j] = b[n - j - 2];
                } else {
                    a[j] = b[j - 1];
                }
            }

            for (int j = 0; j < n; j++) {
                System.out.print(a[j] + " ");
            }
            System.out.println();
        }
    }
}
