n = int(input())
a = []
l = []
for _ in range(n):
    li, *ai = map(int, input().split())
    l.append(li)
    a.append(ai)
ans = 0
for i in range(n):
    for j in range(n):
        if i == j:
            for k in range(l[i] - 1):
                if a[i][k] < a[i][k + 1]:
                    ans += 1
                    break
        else:
            if a[i][l[i] - 1] < a[j][0]:
                ans += 1
print(ans)
