#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k; cin >> n >> k;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    int ans = 0;
    while (a[k - 1] != a[n - 1]) {
        if (a[k - 1] == a[k]) a[k]++;
        else a[n - 1]--;
        ans++;
    }
    cout << ans << endl;
    return 0;
}
