from collections import defaultdict, deque
from sys import stdin

n, m, k = map(int, stdin.readline().split())
a = {int(i) for i in stdin.readline().split()}

g = defaultdict(list)
for u, v in map(lambda x: map(int, x.split()), stdin):
    g[u].append(v)
    g[v].append(u)

q = deque([(1, 0)])
dis = [-1] * (n+1)
vis = [False] * (n+1)
dis[1] = 0
vis[1] = True

while q:
    u, dep = q.popleft()
    for v in g[u]:
        if not vis[v]:
            dis[v] = dep + 1
            vis[v] = True
            q.append((v, dep+1))

dis2 = [-1] * (n+1)
vis = [False] * (n+1)
dis2[n] = 0
vis[n] = True

q = deque([(n, 0)])

while q:
    u, dep = q.popleft()
    for v in g[u]:
        if not vis[v]:
            dis2[v] = dep + 1
            vis[v] = True
            q.append((v, dep+1))

ans = -1
for i in a:
    for j in a:
        if i == j: continue
        if dis[i] + dis2[j] + 1 > ans: ans = dis[i] + dis2[j] + 1

print(ans)
