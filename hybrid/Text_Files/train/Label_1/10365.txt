
#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 1;
vector<int> adj[N];
int f[N], in[N];
char s[N];
int main() {
    int n, m, mx = 0;
    cin >> n >> m;
    cin >> s + 1;
    for (int i = 1; i <= m; ++i) {
        int x, y;
        cin >> x >> y;
        adj[x].emplace_back(y);
        in[y]++;
    }
    queue<int> qu;
    for (int i = 1; i <= n; ++i) {
        if (!in[i])
            qu.emplace(i);
    }
    while (!qu.empty()) {
        int u = qu.front();
        qu.pop();
        for (int v : adj[u]) {
            f[v] = max({f[v], f[u], f[u] + (s[v] == s[u])});
            in[v]--;
            if (!in[v]) {
                qu.emplace(v);
            }
        }
        mx = max(mx, f[u]);
    }
    cout << (mx == (int)1e9 ? -1 : mx) << endl;
}
