#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n,x;
        cin>>n;
        long long *b = new long long[n+3];
        for(int i=1; i<=n+2; i++)
        {
            cin>>b[i];
        }
        int flag=0;
        long long sum=0;
        for(int i=1; i<=n; i++)
        {
            sum+=b[i];
        }
        if(sum!=b[n+1])
        {
            flag=1;
        }
        else
        {
            long long r=b[n+2];
            long long div=r/n;
            long long rem=r%n;
            if(div==0)
            {
                flag=1;
            }
            else
            {
                for(int i=n; i>=1; i--)
                {
                    if(b[i]>div)
                    {
                        flag=1;
                    }
                    else
                    {
                        rem-=b[i];
                    }
                }
                if(rem!=0)
                    flag=1;
            }
        }
        if(flag==1)
            cout<<"-1"<<endl;
        else
        {
            long long r=b[n+2];
            long long div=r/n;
            for(int i=n; i>=1; i--)
            {
                b[i]=div;
            }
            sort(b+1,b+n+1);
            for(int i=1; i<=n; i++)
            {
                cout<<b[i]<<" ";
            }
            cout<<endl;
        }
    }
    return 0;
}
