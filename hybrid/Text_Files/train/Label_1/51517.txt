import java.util.*;
import java.io.*;

public class Main {
    static char[][] pic;
    static int n, m;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        pic = new char[n][m];
        for (int i = 0; i < n; i++) {
            pic[i] = in.next().toCharArray();
        }
        int cnt = 0;
        int minX = 100, minY = 100;
        for (int x = 1; x <= n; x++) {
            for (int y = 1; y <= m; y++) {
                if (n % x == 0 && m % y == 0) {
                    if (good(x, y)) {
                        cnt++;
                        if (x * y < minX * minY) {
                            minX = x;
                            minY = y;
                        }
                        if (x * y == minX * minY && x < minX) {
                            minX = x;
                            minY = y;
                        }
                    }
                }
            }
        }
        System.out.println(cnt);
        System.out.println(minX + " " + minY);
    }

    static boolean good(int x, int y) {
        int n = pic.length;
        int m = pic[0].length;
        HashSet<String> set = new HashSet<String>();
        for (int i = 0; i <= n - x; i++) {
            for (int j = 0; j <= m - y; j++) {
                String cur = "";
                for (int a = i; a < i + x; a++) {
                    for (int b = j; b < j + y; b++) {
                        cur += pic[a][b];
                    }
                }
                if (set.contains(cur)) return false;
                set.add(cur);
            }
        }
        return true;
    }
}
