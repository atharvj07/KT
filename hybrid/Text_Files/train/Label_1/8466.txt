#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll mod = 998244353;
vector<int> a; ll dp[501][501];
ll solve(int l, int r) {
  if (dp[l][r] != -1) return dp[l][r];
  if (l > r) return dp[l][r] = 1;
  if (l == r) return dp[l][r] = (a[l] != a[l - 1] ? 1 : 0);
  ll ans = (a[l] != a[l - 1] ? solve(l + 1, r) : 0);
  ans = (ans + (a[r] != a[r + 1] ? solve(l, r - 1) : 0)) % mod;
  for (int mid = l; mid < r; mid++) {
    if (a[mid] == a[mid + 1]) continue;
    ans = (ans + solve(l, mid) * solve(mid + 1, r)) % mod;
  }
  return dp[l][r] = ans;
}
int main() {
  int n; cin >> n; a.resize(n + 1);
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  memset(dp, -1, sizeof dp);
  cout << solve(1, n) << endl;
}
