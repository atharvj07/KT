#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,k;
    cin>>n>>k;
    map<int,int>mp;
    for(int i=0, x; i<n; i++)
    {
        cin>>x;
        mp[x]++;
    }
    long long ans=0;
    for(auto i: mp)
    {
        ans+=min(i.second, n/2);
    }
    cout<<ans<<endl;
    return 0;
}
