import sys
input = sys.stdin.readline
n, m = map(int, input().split())
a = list(map(int, input().split()))
b = [0] * m
for x in a: b[x - 1] += 1
mb = min(b)
cnt = 0
for i in range(n):
	if a[i] <= m and b[a[i] - 1] > mb: a[i], cnt = m + 1, cnt + 1
	elif a[i] > m and b[a[i] - 1] <= mb: a[i], cnt = mb, cnt + 1
print(mb, cnt)
print(*a)
