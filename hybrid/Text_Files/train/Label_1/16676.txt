#include<iostream>
#include<cstring>
using namespace std;
int main()
{
    char s[1501],type[27];
    int k;
    cin>>s>>type>>k;
    int n=strlen(s),ans=0;
    int f[26];
    for(int i=0;i<n;i++)
    {
        memset(f,0,sizeof(f));
        for(int j=i;j<n;j++)
        {
            if(type[s[j]-97] == '0')
                f[s[j]-97]++;
            else
            {
                if(f[s[j]-97]==0)
                    k--;
                if(k>=0)
                    ans++;
                else
                    break;
            }
        }
    }
    cout<<ans;
    return 0;
}
