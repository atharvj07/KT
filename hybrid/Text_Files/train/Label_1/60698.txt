import sys
input=sys.stdin.readline
n,m=map(int,input().split())
a=[list(map(int,input().split())) for i in range(n)]
a.sort(key=lambda x:x[1])
vote=[0]*m
ans=0
for p,c in a:
    if p!=1:
        if ans+c<=vote[p]: continue
        ans+=c
        vote[p]+=1
    else:
        ans+=1
        vote[p]+=1
print(ans)
