using namespace std;

ll mod=1e9+7;
int main(){
	fast;
	ll t;
	cin>>t;
	while(t--){
		ll n;
		cin>>n;
		vi a(n),b(n);
		for(int i=0;i<n;i++){
			cin>>a[i];
		}
		for(int i=0;i<n;i++){
			cin>>b[i];
		}
		int ans=0;
		map<int,int> m;
		for(int i=0;i<n;i++){
			if(b[i]%3!=0){
				ans=-1;
				break;
			}
			else{
				m[a[i]]++;
				m[b[i]/3]--;
			}
		}
		int flag=1;
		for(auto x:m){
			if(x.second!=0){
				flag=0;
				break;
			}
		}
		if(ans==0) ans=flag;
		if(ans==1) cout<<"YES"<<endl;
		else cout<<"NO"<<endl;
	}
	return 0;
}
