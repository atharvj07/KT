#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string a;
    cin >> a;
    vector<int> pre(n + 1);
    for (int i = 0; i < n; i++) {
      pre[i + 1] = pre[i] + (a[i] - '0');
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j <= n; j++) {
        if (pre[j] - pre[i] == j - i) ans++;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
