#include<bits/stdc++.h>
using namespace std;
int main(){
	int n,m,a[505][505],i,j,f,ans[505];
	cin>>n>>m;
	for(i=1;i<=n;i++){
		for(j=1;j<=m;j++){
			cin>>a[i][j];
		}
	}
	f=0;
	for(i=1;i<=n;i++){
		for(j=1;j<=m;j++){
			if(a[i][j]){
				ans[i]=j;
				f=1;
			}
		}
	}
	if(!f){
		cout<<"NIE";
	}
	else{
		cout<<"TAK"<<"\n";
		for(i=1;i<=n;i++){
			cout<<ans[i]<<" ";
		}
	}
	return 0;
}
