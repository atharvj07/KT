#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int a, b, n;
    cin >> a >> b >> n;
    if (n < 2)
      cout << (n ? b : a) << endl;
    else
      cout << ((n & 1) ? a : b) << endl;
  }
  return 0;
}
