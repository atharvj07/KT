import sys

input = sys.stdin.readline

n, x0, y0 = map(int, input().split())

def ccw(a, b, c):
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])

ans = 0
ps = [(0, 0)] * n
for i in range(n):
    x, y = map(int, input().split())
    if x == x0:
        if y > y0:
            ans += 1
    else:
        ps[i] = (x, y)
        if ans == 0:
            ans += 1

ps.sort(key=lambda p: (p[1], p[0]))
last = (y0, -1e9)
for p in ps:
    if ccw(last, (x0, y0), p) < 0:
        ans += 1
        last = p

print(ans)
