#include <bits/stdc++.h>

using namespace std;
using ll = long long;
using ld = long double;

int main() {
  int n;
  cin >> n;
  vector<ll> a(n + 1), dp(n + 1);
  for (int i = 1; i <= n; ++i) cin >> a[i];
  ll sm = 0;
  int l = 1, r = 1;
  for (int i = 1; i <= n; ++i) {
    sm += a[i];
    if (1.0 * sm / i > 1.0 * dp[i - 1] / (i - 1)) {
      dp[i] = sm;
      l = i - dp[i - 1] + 1;
      r = i;
    } else
      dp[i] = dp[i - 1];
  }
  cout << r - l + 1 << "\n";
  return 0;
}
