import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    static class Cell {
        int x, y;

        Cell(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            grid[i] = br.readLine().toCharArray();
        }
        StringBuilder[] ans = new StringBuilder[n];
        for (int i = 0; i < ans.length; i++) ans[i] = new StringBuilder();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '*') {
                    boolean[][] visited = new boolean[n][m];
                    int size = dfs(i, j, grid, visited);
                    ans[i].append(size % 10);
                } else {
                    ans[i].append('.');
                }
            }
        }
        for (StringBuilder a : ans) System.out.println(a);
    }

    static int[] dx = {1, -1, 0, 0};
    static int[] dy = {0, 0, -1, 1};

    static int dfs(int x, int y, char[][] grid, boolean[][] visited) {
        if (x < 0 || x >= grid.length || y < 0 || y >= grid[0].length || visited[x][y] || grid[x][y] == '*') return 0;
        visited[x][y] = true;
        int ans = 1;
        for (int i = 0; i < 4; i++) {
            ans += dfs(x + dx[i], y + dy[i], grid, visited);
        }
        return ans;
    }
}
