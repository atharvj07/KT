import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long max = a[0];
        int index = 1;
        for (int i = 1; i < n; i++) {
            if (a[i] > max) {
                max = a[i];
                index = i + 1;
            }
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }
        if (sum == max) {
            index = -1;
        }
        System.out.println(index);
    }
}
