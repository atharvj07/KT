import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int r = input.nextInt();
        int k = input.nextInt();
        double ans = (double) k / (n - r + 1) / (m - r + 1);
        System.out.println(ans);
    }
}
