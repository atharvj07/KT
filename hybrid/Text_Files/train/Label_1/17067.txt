#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int dp[n][n];
    if (s[0] == '(')
        dp[0][0] = 1;
    else
        dp[0][0] = 1e9;
    for (int i = 1; i < n; i++) {
        if (s[i] == '(')
            dp[i][i] = 1;
        else
            dp[i][i] = 1e9;
    }
    for (int len = 2; len <= n; len++) {
        for (int l = 0; l < n - len + 1; l++) {
            int r = l + len - 1;
            dp[l][r] = 1e9;
            dp[l][r] = min(dp[l][r], dp[l + 1][r] + (s[l] == ')'));
            dp[l][r] = min(dp[l][r], dp[l][r - 1] + (s[r] == '('));
            for (int j = l; j < r; j++) {
                dp[l][r] = min(dp[l][r], dp[l][j] + dp[j + 1][r]);
            }
        }
    }
    if (dp[0][n - 1] >= 1e9)
        cout << -1;
    else
        cout << dp[0][n - 1];
    return 0;
}
