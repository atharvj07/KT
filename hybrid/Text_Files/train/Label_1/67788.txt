from bisect import bisect_right
for t in range(int(input())):
    a = [int(i) for i in input().split()]
    n = len(a)
    lis = [1] * n
    lds = [1] * n
    for i in range(1, n):
        for j in range(i):
            if a[i] > a[j]: lis[i] = max(lis[i], lis[j] + 1)
            elif a[i] < a[j]: lds[i] = max(lds[i], lds[j] + 1)
    l = [lis[i] + lds[i] - 1 for i in range(n)]
    l.sort(reverse=True)
    if l[0] % 2 == 0 and sum(l) % 2 == 0: print('YES')
    else: print('NO')
