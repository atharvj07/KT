import sys
n = int(input())
s = input()
x = 0
for i in range(n):
    if s[i] == 'X':
        x += 1
y = n // 2
if x == y:
    print(0)
    print(s)
else:
    print(1)
    if x < y:
        for i in range(n):
            if s[i] == 'x':
                s = s[:i] + 'X' + s[i + 1:]
                break
    else:
        for i in range(n):
            if s[i] == 'X':
                s = s[:i] + 'x' + s[i + 1:]
                break
    print(s)
