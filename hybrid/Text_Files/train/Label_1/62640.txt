import sys
input = sys.stdin.readline

n, m = map(int, input().split())
p = list(map(int, input().split()))
lst, lsti = [], []
for i in range(m):
    a, b = map(int, input().split())
    a, b = a - 1, b - 1
    lst.append([p[a], p[b]])
    lsti.append([a, b])
lst.sort()
p = sorted(p)
ans = [0] * n
for i in range(m):
    ans[lsti[i][0]] = lst[i][0]
    ans[lsti[i][1]] = lst[i][1]
for i in range(n):
    if ans[i] == 0:
        ans[i] = p.pop(0)
print(*ans)
