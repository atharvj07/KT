#include<bits/stdc++.h>
using namespace std;
int mod = 1e9+7;
typedef long long ll;
int main() {
  int n, m, b, md;
  cin >> n >> m >> b >> md;
  mod = md;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  vector<vector<ll>> dp(n + 1, vector<ll>(m + 1, 0));
  dp[0][0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j <= m; j++) {
      for (int k = 0; k <= b; k++) {
        if (j + k <= m) { dp[i + 1][j + k] = (dp[i + 1][j + k] + dp[i][j]); }
        if (k - a[i] >= 0) { dp[i + 1][j] = (dp[i + 1][j] + dp[i][j] - dp[i][j - k + a[i]]); }
      }
    }
  }
  cout << dp[n][m] % mod;
}
