#include<iostream>
#include<bits/stdc++.h>
using namespace std;

int main(){
  int t;
  cin>>t;
  while(t--){
    int n,m;
    cin>>n;
    map<string , int> mp;
    vector<string> name;
    for(int i=0;i<n;i++){
      string s;
      cin>>s;
      name.push_back(s);
      mp[s]=0;
    }
    cin>>m;
    vector<pair<string,string>> chat;
    for(int i=0;i<m;i++){
      string s,x="";
      getline(cin,s);
      int l=s.size();
      for(int j=0;j<l;j++){
        if(s[j]==':'){
          chat.push_back({x,s.substr(j+1,l-j-1)});
          break;
        }
        x+=s[j];
      }
    }
    int fl=0;
    for(auto it:chat){
      string p=it.first;
      if(p[0]=='?'){
        for(auto x:name){
          if(mp[x]==0){
            mp[x]=1;
            chat[chat.size()-1].first=x;
            break;
          }
        }
      }
      if(chat[chat.size()-1].first==chat[chat.size()-2].first){
        fl=1;
        cout<<"Impossible"<<endl;
        break;
      }
    }
    if(fl==0){
      for(auto it:chat)
        cout<<it.first<<":"<<it.second<<endl;
    }
  }
}
