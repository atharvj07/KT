import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] submissions = new int[n][5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                submissions[i][j] = sc.nextInt();
            }
        }
        int[] petya = submissions[1];
        int[][] max = new int[5][2];
        for (int j = 0; j < 5; j++) {
            int solved = 0;
            for (int i = 0; i < n; i++) {
                if (submissions[i][j] != -1) {
                    solved++;
                }
            }
            double ratio = solved / (double) n;
            if (ratio >= 1.0) {
                max[j][0] = 1000;
                max[j][1] = 0;
            } else {
                double x = 1000 / (1.0 - ratio);
                max[j][0] = (int) Math.round(x);
                max[j][1] = (int) Math.round(x / 250.0);
            }
        }
        long[] score = new long[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                if (submissions[i][j] != -1) {
                    score[i] += max[j][0] - max[j][1] * submissions[i][j];
                }
            }
        }
        Arrays.sort(score);
        int ans = -1;
        if (score[n - 1] > score[n - 2]) {
            ans = 0;
        } else {
            for (int i = 0; i < 5; i++) {
                long x = petya[i] != -1 ? max[i][0] - max[i][1] * petya[i] : 0;
                for (int j = 0; j < n; j++) {
                    score[j] -= x;
                }
                Arrays.sort(score);
                if (score[n - 1] > score[n - 2]) {
                    ans = i + 1;
                    break;
                }
            }
        }
        System.out.println(ans);
    }
}
