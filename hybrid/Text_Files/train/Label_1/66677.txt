from collections import Counter, defaultdict

for _ in range(int(input())):
    n = int(input())
    codes = []
    for _ in range(n):
        codes.append(input())
    cnt = Counter(codes)
    if len(cnt) == len(codes):
        print('0')
        for code in codes:
            print(code)
    else:
        res = defaultdict(list)
        for i, code in enumerate(codes):
            res[code].append(i)
        lst = list(res.items())
        lst.sort(key=lambda x: len(x[1]))
        lst_len = len(lst)
        changed = [p[1][0] for p in lst if len(p[1]) > 1]
        ch_cnt = len(changed)
        print(ch_cnt)
        for i, p in enumerate(lst):
            code, idxs = p
            if len(idxs) > 1:
                for j in range(1, len(idxs)):
                    codes[idxs[j]] = code[:3] + str((int(code[3]) + j) % 10)
            print(codes[i])
