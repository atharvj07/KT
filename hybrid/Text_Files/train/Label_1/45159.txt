import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int xStart = sc.nextInt();
        int yStart = sc.nextInt();
        int xEnd = sc.nextInt();
        int yEnd = sc.nextInt();
        ArrayList<int[]> segments = new ArrayList<>();
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int[] segment = { sc.nextInt(), sc.nextInt(), sc.nextInt() };
            segments.add(segment);
        }
        System.out.println(bfs(xStart, yStart, xEnd, yEnd, segments));
    }

    private static int bfs(int xStart, int yStart, int xEnd, int yEnd, ArrayList<int[]> segments) {
        int[][] distance = new int[109][109];
        for (int i = 0; i < 109; i++) {
            Arrays.fill(distance[i], -1);
        }
        distance[xStart][yStart] = 0;
        ArrayList<int[]> queue = new ArrayList<>();
        queue.add(new int[] { xStart, yStart });
        while (!queue.isEmpty()) {
            int[] current = queue.remove(0);
            int x = current[0];
            int y = current[1];
            if (x == xEnd && y == yEnd) {
                return distance[x][y];
            }
            for (int i = x - 1; i <= x + 1; i++) {
                for (int j = y - 1; j <= y + 1; j++) {
                    if (i >= 1 && i <= 109 && j >= 1 && j <= 109) {
                        if (distance[i][j] == -1) {
                            boolean allowed = false;
                            for (int[] segment : segments) {
                                if (segment[0] == i && j >= segment[1] && j <= segment[2]) {
                                    allowed = true;
                                    break;
                                }
                            }
                            if (allowed) {
                                distance[i][j] = distance[x][y] + 1;
                                queue.add(new int[] { i, j });
                            }
                        }
                    }
                }
            }
        }
        return -1;
    }
}
