def find_min_days(k, schedule):
    total_classes = sum(schedule)
    full_weeks = k // total_classes
    remaining_classes = k % total_classes

    if remaining_classes == 0:
        # If no remaining classes, find the last class in the week.
        for i in range(6, -1, -1):
            if schedule[i] == 1:
                return full_weeks * 7 - (6 - i)
    else:
        max_days = 0
        for start_day in range(7):
            current_day = start_day
            classes_count = 0
            days_passed = 0
            while classes_count < remaining_classes:
                if schedule[current_day] == 1:
                    classes_count += 1
                days_passed += 1
                current_day = (current_day + 1) % 7
            max_days = max(max_days, days_passed)
        return full_weeks * 7 + max_days

t = int(input().strip())
for _ in range(t):
    k = int(input().strip())
    schedule = list(map(int, input().strip().split()))
    print(find_min_days(k, schedule))
