#include <iostream>
#include <string>
#include <unordered_map>
using namespace std;

int main() {
  int k, n;
  cin >> k >> n;
  unordered_map<string, int> m;
  unordered_map<string, int> bm;
  for (int i = 0; i < k; i++) {
    string s;
    int a;
    cin >> s >> a;
    m[s]++;
    bm[s] = max(bm[s], a);
  }
  int ans = 0;
  for (const auto &p : m) {
    string s = p.first;
    int cnt = p.second;
    int b = bm[s];
    if (s.size() % 2 == 0) {
      if (cnt % 2 == 0)
        ans += cnt / 2 * b;
      else
        ans += (cnt / 2) * b + b / 2;
    } else {
      if (cnt % 2 == 0)
        ans += (cnt / 2) * b;
      else {
        ans += (cnt / 2) * b + b / 2;
        cnt--;
        s = s.substr(0, (s.size() - 1) / 2) + s.substr((s.size() + 3) / 2 - 1);
        ans += (cnt / 2) * bm[s];
      }
    }
  }
  cout << ans << endl;
  return 0;
}
