#include <iostream>
using namespace std;

int main() {
  long long l, r;
  cin >> l >> r;
  long long p = 0;
  while (l % 3 == 0) {
    l /= 3;
    p++;
  }
  while (r % 3 == 0) {
    r /= 3;
    p++;
  }
  if (l % 2 == r % 2) {
    p /= 2;
  } else {
    p /= 2;
    p++;
  }
  cout << p << endl;
}
