#include <algorithm>
#include <vector>
#include <iostream>

using namespace std;

void merge(int *arr1, int n1, int *arr2, int n2) {
   int gap = n1 + n2;
   for (int i = gap; i > 0; i--) {
      for (int j = 0; j + i < gap; j++) {
         if (arr1[j] > arr1[j + i]) {
            int temp = arr1[j];
            arr1[j] = arr1[j + i];
            arr1[j + i] = temp;
         } else if (arr2[j] > arr2[j + i]) {
            int temp = arr2[j];
            arr2[j] = arr2[j + i];
            arr2[j + i] = temp;
         }
      }
   }
}

int main() {
   int t;
   cin >> t;
   while (t--) {
      int n;
      cin >> n;
      int a[n], b[n], p[2 * n];
      for (int i = 0; i < 2 * n; i++) {
         cin >> p[i];
      }
      bool flag = false;
      for (int i = 1; i < 2 * n; i += 2) {
         if (p[i] < p[i - 1]) {
            a[flag] = p[i - 1];
            b[flag] = p[i];
            flag = true;
         }
      }
      if (flag == false) {
         cout << "NO" << endl;
         continue;
      }
      merge(a, flag, b, flag);
      bool valid = true;
      for (int i = 0; i < flag; i++) {
         if (a[i] != p[i] || b[i] != p[i + flag]) {
            valid = false;
            break;
         }
      }
      if (valid)
         cout << "YES" << endl;
      else
         cout << "NO" << endl;
   }
   return 0;
}
