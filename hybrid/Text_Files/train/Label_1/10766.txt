#include <iostream>
using namespace std;
int main() {
  string s;
  cin >> s;
  for (int i = 0; i < s.size() - 1; i++)
    if (s.substr(i, 2) == "AB" && s.substr(i + 2, 2) == "BA")
      return cout << "YES", 0;
  for (int i = 0; i < s.size() - 1; i++)
    if (s.substr(i, 2) == "BA" && s.substr(i + 2, 2) == "AB")
      return cout << "YES", 0;
  return cout << "NO", 0;
}
