#include <iostream>
#include <vector>
#include <queue>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> d(n);
  for (int i = 0; i < n; i++) { cin >> d[i]; }
  vector<vector<int>> adj(n);
  vector<int> deg(n, 0);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    u--;
    v--;
    adj[u].push_back(v);
    adj[v].push_back(u);
    deg[u]++;
    deg[v]++;
  }
  vector<int> ans;
  queue<int> q;
  for (int i = 0; i < n; i++) {
    if (d[i] == -1) {
      q.push(i);
    } else if (deg[i] % 2 != d[i]) {
      cout << -1 << endl;
      return 0;
    }
  }
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    for (int v : adj[u]) {
      deg[v]--;
      if (d[v] == -1) {
        q.push(v);
      } else if (deg[v] % 2 != d[v]) {
        ans.push_back(u + 1);
        ans.push_back(v + 1);
        swap(d[u], d[v]);
        deg[u]++;
        deg[v]++;
        q.push(u);
        q.push(v);
      }
    }
  }
  cout << ans.size() / 2 << endl;
  for (int i = 0; i < ans.size(); i += 2) { cout << ans[i] << " " << ans[i + 1] << endl; }
  return 0;
}
