t = int(input().strip())
for _ in range(t):
    n, m = map(int, input().strip().split())
    s = list(input().strip())
    for _ in range(min(m, 2)):  # After 2 iterations, no more changes occur
        new_s = s.copy()
        for i in range(n):
            if s[i] == '0':
                left_alive = i > 0 and s[i - 1] == '1'
                right_alive = i < n - 1 and s[i + 1] == '1'
                if left_alive ^ right_alive:
                    new_s[i] = '1'
        s = new_s
    print("".join(s))
