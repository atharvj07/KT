#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int res = 0;
    int cur = 1;
    int last = n % 10;
    while (last != 0 && last % 2 != 0) {
      res++;
      cur++;
      last = (n / cur) % 10;
    }
    if (last == 0)
      cout << -1 << endl;
    else
      cout << res << endl;
  }
  return 0;
}
