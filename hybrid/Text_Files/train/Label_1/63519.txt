n, m, q = map(int, input().split())
x0 = [1, 2]
for i in range(q):
    sx, sy, ex, ey = map(int, input().split())
    if sx == ex and x0[sx - 1] < min(sy, ey):
        x0[sx - 1] = max(sy, ey)
    if x0[sx - 1] < min(sy, ey) or x0[ex - 1] > max(sy, ey):
        print('NO')
    else:
        print('YES')
