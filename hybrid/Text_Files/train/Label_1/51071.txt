import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt() - 1;
        int[] s = new int[n];
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = s[i] + p[i];
        }
        Arrays.sort(t);
        int r = 0;
        for (int i = n - 1; i >= 0; i--) {
            r++;
            if (t[i] == t[d]) {
                break;
            }
        }
        System.out.println(r);
    }
}
