#include<bits/stdc++.h>
#define ii pair<int,int>
using namespace std;
typedef long long ll;

const int nmax = 1e5 + 5;
const ll inf = 1LL << 62;
vector<ii> g[nmax];
vector<int> storage;
int n,m,k;
int dis[nmax];
bool vis[nmax];

void dijkstra(int v,int flag = 0){
    fill(dis,dis+n+1,inf);
    memset(vis,0,sizeof vis);
    priority_queue<ii,vector<ii>,greater<ii> > q;
    q.push({0,v});
    dis[v] = 0;
    while(!q.empty()){
        auto u = q.top(); q.pop();
        if(vis[u.second])continue;
        vis[u.second] = 1;
        for(auto tmp : g[u.second]){
            if(vis[tmp.first])continue;
            if(dis[u.second] + tmp.second < dis[tmp.first]){
                dis[tmp.first] = dis[u.second] + tmp.second;
                q.push({dis[tmp.first],tmp.first});
            }
        }
    }
    if(!flag){
        for(int i=1;i<=n;++i){
            cout<<dis[i]<<" ";
        }
        cout<<endl;
    }
}
int main() {
    cin>>n>>m>>k;
    for(int i=1;i<=m;++i){
        int x,y,w;
        cin>>x>>y>>w;
        g[x].push_back({y,w});
        g[y].push_back({x,w});
    }
    for(int i=1;i<=k;++i){
        int x;  cin>>x;
        storage.push_back(x);
    }
    if(k==0){
        cout<<0<<endl;
        return 0;
    }
    ll ans = inf;
    for(int i=1;i<=n;++i){
        bool check = 0;
        for(auto tmp : storage){
            if(tmp==i){
                check = 1;
            }
        }
        if(check)continue;
        dijkstra(i,1);
        ll tmp = 0;
        for(auto tmps : storage){
            tmp = min(tmp,1LL*dis[tmps]);
        }
        if(tmp != inf){
            ans = min(ans,tmp);
        }

    }
    if(ans == inf){
        ans = -1;
    }
    cout<<ans<<endl;
    return 0;
}
