import java.util.Scanner;

class Main {

	private static final char[][] FACE_MATRIX = new char[][] { { 'f', 'a' }, { 'c', 'e' } };

	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int m = in.nextInt();
		int count = 0;
		char[][] image = new char[n][m];
		for (int i = 0; i < n; i++) {
			image[i] = in.next().toCharArray();
		}
		for (int i = 0; i < n - 1; i++) {
			for (int j = 0; j < m - 1; j++) {
				if (image[i][j] == FACE_MATRIX[0][0] && image[i + 1][j] == FACE_MATRIX[0][1]
						&& image[i][j + 1] == FACE_MATRIX[1][0] && image[i + 1][j + 1] == FACE_MATRIX[1][1]) {
					count++;
				}
			}
		}
		System.out.println(count);
	}

}
