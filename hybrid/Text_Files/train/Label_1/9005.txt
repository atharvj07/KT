#include <bits/stdc++.h>

using namespace std;

typedef unsigned long long ull;

int main() {
    int n;
    cin >> n;
    unordered_map<string, int> nums;
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        sort(s.begin(), s.end());
        nums[s]++;
    }

    ull sum = 0;
    for (auto &num : nums) {
        int cnt = 0;
        for (char c : num.first) {
            cnt += c - 'a';
        }
        sum += (num.second * cnt);
    }
    cout << sum << endl;
    return 0;
}
