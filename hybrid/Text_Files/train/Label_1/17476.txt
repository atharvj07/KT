#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, B;
  cin >> n >> B;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int cnt = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] % 2 != a[i - 1] % 2) {
      if (B >= abs(a[i] - a[i - 1])) {
        cnt += 1;
        B -= abs(a[i] - a[i - 1]);
      }
    }
  }
  cout << cnt << endl;
  return 0;
}
