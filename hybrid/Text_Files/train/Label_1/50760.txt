import java.util.*;

public class Main {
    static int n, m;
    static List<Integer>[] adj, reverseAdj;

    static int dfs1(int node, int[] path) {
        boolean[] visited = new boolean[n + 1];
        visited[node] = true;
        path[0] = node;
        LinkedList<Integer> queue = new LinkedList<>();
        queue.add(node);
        while (!queue.isEmpty()) {
            int curr = queue.removeFirst();
            for (int neighbor : adj[curr]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    queue.add(neighbor);
                    path[0] = neighbor;
                }
            }
        }
        return path[0];
    }

    static void dfs2(int node, boolean[] visited, int[] path, int index) {
        visited[node] = true;
        path[index] = node;
        for (int neighbor : reverseAdj[node]) {
            if (!visited[neighbor]) {
                dfs2(neighbor, visited, path, index + 1);
            }
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        adj = new List[n + 1];
        reverseAdj = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
            reverseAdj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            adj[u].add(v);
            reverseAdj[v].add(u);
        }
        int[] path = new int[n + 1];
        int start = dfs1(1, path);
        boolean[] visited = new boolean[n + 1];
        dfs2(start, visited, path, 1);
        int[] result = new int[4];
        result[0] = path[path.length - 1];
        result[1] = path[path.length - 2];
        result[2] = path[path.length - 3];
        result[3] = path[path.length - 4];
        for (int num : result) {
            System.out.print(num + " ");
        }
    }
}
