#include <bits/stdc++.h>
using namespace std;
#define mod 1000000007
int32_t main() {
  string s, t;
  cin >> s >> t;
  int n = s.size(), m = t.size();
  vector<int> dp(n + 1);
  vector<int> oc(n + 1);
  for (int i = m - 1, j; i < n; i++) {
    for (j = i - m + 1; j >= 0 && s[j] == t[j - i + m - 1]; j--)
      ;
    oc[i + 1] = (j < 0);
  }
  dp[0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = oc[i]; j <= i; j++) {
      dp[i + 1] = (dp[i + 1] + dp[j]) % mod;
    }
  }
  cout << dp[n] << endl;
}
