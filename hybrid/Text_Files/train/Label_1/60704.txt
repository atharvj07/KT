import sys

n, k, s, t = map(int, input().split())
cars = []
for _ in range(n):
  c, v = map(int, input().split())
  cars.append((c, v))
gas = sorted([int(x) for x in input().split()])

min_cost = -1
for c, v in cars:
  time = 0
  fuel = v
  for g in gas:
    if g > time:
      fuel -= (g - time) * (time % 2 + 1)
    if fuel <= 0:
      break
    time = g
  fuel -= (s - time) * (time % 2 + 1)
  if fuel > 0 and time + (s - time) * (time % 2 + 1) <= t:
    min_cost = min(min_cost, c) if min_cost != -1 else c

print(min_cost)
