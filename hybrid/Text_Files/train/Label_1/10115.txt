#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007

const int maxN = 1010, maxM = maxN;
vector<int> adj[maxN], weight[maxN];
vector<ll> dist(maxN, 1e18);
ll ans;
bool visited[maxN];

void dfs(int u, int d, int k) {
  if (d > k) return;
  visited[u] = true;
  if (d == k) ans++;
  for (int i = 0; i < adj[u].size(); i++) {
    int v = adj[u][i], w = weight[u][i];
    if (!visited[v]) dfs(v, d + w, k);
  }
}

int main() {
  ll n, k;
  cin >> n >> k;
  n++;
  int len = 0;
  while (n > 0) {
    if (n % 2 == 1) adj[len].push_back(len + 1), weight[len].push_back(1);
    n /= 2;
    len++;
  }
  dfs(0, 0, k);
  cout << ans % mod << endl;
  return 0;
}
