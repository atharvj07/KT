import math

def solve(R, x1, y1, x2, y2):
    distance = math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
    if distance >= R:
        return f"{x1} {y1} {R}"
    else:
        r = (distance + R) / 2
        if distance == 0:
            return f"{x1} {y1 + r} {r}"
        else:
            x_ap = x1 + (x2 - x1) * (R - r) / distance
            y_ap = y1 + (y2 - y1) * (R - r) / distance
            return f"{x_ap} {y_ap} {r}"

R, x1, y1, x2, y2 = map(int, input().split())
print(solve(R, x1, y1, x2, y2))
