def get_pokemon_name(n, s):
    possibilities = []
    for pokemon in ["vaporeon", "jolteon", "flareon", "espeon", "umbreon", "leafeon", "glaceon", "sylveon"]:
        if len(pokemon) == n:
            found = True
            for i in range(n):
                if (s[i] != '.') and (s[i] != pokemon[i]):
                    found = False
                    break
            if found:
                possibilities.append(pokemon)
    return possibilities[0]
n = int(input())
s = input()
print(get_pokemon_name(n, s))
