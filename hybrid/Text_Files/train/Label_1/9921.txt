#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int ROOT = 1;
const int inf = 1e9;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> adj(n + 1);
    vector<pair<int, int>> indeg(n + 1, {0, 0});
    bool cycle = false;
    while (m--) {
      int a, b;
      cin >> a >> b;
      adj[a].push_back(b);
      indeg[a].second++;
    }
    queue<int> q;
    for (int i = 1; i <= n; i++) {
      if (indeg[i].second == 0) {
        q.push(i);
        indeg[i].first = 1;
      }
    }
    vector<int> ans(n + 1, 0);
    while (!q.empty()) {
      int u = q.front();
      q.pop();
      if (u == ROOT)
        ans[u] = 1;
      else {
        if (indeg[u].first > 1) {
          ans[u] = 2;
          cycle = true;
        } else
          ans[u] = 1;
      }
      for (auto v : adj[u]) {
        indeg[v].first++;
        if (indeg[v].first > 1)
          cycle = true;
        indeg[v].second--;
        if (indeg[v].second == 0)
          q.push(v);
      }
    }
    for (int i = 1; i <= n; i++) {
      if (indeg[i].second > 0)
        ans[i] = -1;
      cout << ans[i] << ' ';
    }
    cout << endl;
  }
  return 0;
}
