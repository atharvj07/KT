import sys
input = sys.stdin.readline

n, x, y = map(int, input().split())
adj = [[] for _ in range(n+1)]
visited = [False] * (n+1)

for _ in range(n-1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

def dfs(node, dist, ans):
    visited[node] = True
    ans.append(dist)
    for u in adj[node]:
        if visited[u] == False:
            dfs(u, dist + (x if u not in adj[node] else y), ans)

ans = []
dfs(1, 0, ans)
print(max(ans))
