import sys
input = sys.stdin.readline

n = int(input())
pos = [0] * n
col = [0] * n
for i in range(n):
    p, c = input().split()
    pos[i] = int(p)
    col[i] = c
    
p = [0] * 3
for i in range(n):
    if col[i] == 'R':
        p[0] = i
    elif col[i] == 'G':
        p[1] = i
    else:
        p[2] = i

ans = pos[p[1]] - pos[p[0]] + pos[p[2]] - pos[p[1]]
print(ans)
