import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, m = map(int, input().split())
    s = input()
    p = [int(x) for x in input().split()]
    ans = [0] * 26
    cnt = [0] * n
    for i in range(m):
        cnt[p[i] - 1] += 1
    for i in range(n):
        ans[ord(s[i]) - ord('a')] += (m + 1) - cnt[i]
    print(*ans)
