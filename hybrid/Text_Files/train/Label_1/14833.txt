#include <bits/stdc++.h>
using namespace std;
#define ll long long
string s;
int main()
{
    cin >> s;
    int ans = 0;
    for (int i = 0; i < s.size(); i++)
    {
        if (i == s.size() - 1)
            ans++;
        else if (s[i] == s[i + 1])
        {
            if (ans % 2 == 0)
                ans++;
        }
        else
            ans = 1;
    }
    cout << ans - 1 << "\n";
}
