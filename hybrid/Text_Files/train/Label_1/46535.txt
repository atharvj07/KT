import java.util.ArrayList;
import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] s = new long[n];
    for (int i = 0; i < n; i++) {
      s[i] = sc.nextInt();
    }
    ArrayList<Integer> p = new ArrayList<>();
    for (int i = n; i >= 1; i--) {
      long cnt = 0;
      int j = 0;
      for (; j < p.size() && cnt + p.get(j) < s[i - 1]; j++) {
        cnt += p.get(j);
      }
      p.add(j, i);
    }
    for (int i : p) {
      System.out.print(i + " ");
    }
    System.out.println();
  }
}
