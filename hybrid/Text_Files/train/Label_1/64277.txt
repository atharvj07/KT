import sys
def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        cnt, pre, ans = 0, 0, 0
        for i in range(n):
            if a[i]:
                if pre == 0:
                    ans += cnt // 2
                    cnt = 1
                else:
                    cnt += 1
            else:
                pre = cnt
                cnt = 0
        ans += cnt // 2
        print(ans)
solve()
