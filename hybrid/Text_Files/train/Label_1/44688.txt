import java.util.Scanner;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.HashMap;
import java.util.Map.Entry;

class Main {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] e = new int[n + 1][n + 1];
        for (int i = 0; i < n; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            e[u][v] = 1;
            e[v][u] = 1;
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            System.out.println(bfs(e, a, b));
        }
    }

    public static int bfs(int[][] e, int a, int b) {
        int[] d = new int[e.length];
        Arrays.fill(d, -1);
        Queue<Integer> q = new LinkedList<>();
        q.add(a);
        d[a] = 0;
        while (!q.isEmpty()) {
            int v = q.poll();
            if (v == b) {
                return d[v];
            }
            for (int i = 1; i <= e.length - 1; i++) {
                if (e[v][i] == 1 && d[i] == -1) {
                    q.add(i);
                    d[i] = d[v] + 1;
                }
            }
        }
        return -1;
    }
}
