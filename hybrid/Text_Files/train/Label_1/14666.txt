#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll INF = 0x3f3f3f3f3f3f3f3f;
int main() {
  ll n;
  cin >> n;
  vector<ll> a(n);
  for (auto &ai : a) cin >> ai;
  ll ans = INF;
  for (ll i = 0; i < n; i++) {
    if (a[i] != -1) {
      ll L = i + 1;
      ll R = n - i - 1;
      ll cur = 0;
      while (L > 0 && R > 0) {
        if (a[L] < a[R]) {
          cur += a[L];
          L--;
        } else {
          cur += a[R];
          R--;
        }
      }
      while (L > 0) {
        cur += a[L];
        L--;
      }
      while (R > 0) {
        cur += a[R];
        R--;
      }
      ans = min(ans, cur);
    }
  }
  cout << ans << '\n';
}
