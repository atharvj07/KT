t = int(input())
for _ in range(t):
    n = int(input())
    a = input().strip()
    b = input().strip()
    
    if a == b:
        print(0)
        continue
    
    if '1' not in a or '1' not in b:
        print(-1)
        continue
    
    if a.count('1') != b.count('1') and a.count('0') != b.count('0'):
        print(-1)
        continue
    
    diff = 0
    for x, y in zip(a, b):
        if x != y:
            diff += 1
            
    if diff % 2 == 0:
        print(2 if a.count('1') != b.count('1') else 1)
    else:
        print(-1)
