#include <iostream>
#include <cmath>

using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        int a, b, c;
        cin >> a >> b >> c;

        int sum = a + 2 * b + 3 * c;
        int half = sum / 2;

        int minDiff = abs(half - sum + half);

        cout << minDiff << "\n";
    }
}
