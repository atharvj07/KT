import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] p = new int[n];
        p[n - 1] = a[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            p[i] = p[i + 1] * a[i];
        }
        int[] pr = new int[n];
        pr[0] = a[0];
        for (int i = 1; i < n; i++) {
            pr[i] = pr[i - 1] * a[i];
        }
        int neg = 0, pos = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                long prod = (i == 0 ? 1L : pr[i - 1]) * (j == n - 1 ? 1L : p[j + 1]);
                if (prod < 0) {
                    neg++;
                } else if (prod > 0) {
                    pos++;
                }
            }
        }
        System.out.println(neg + " " + pos);
    }
}
