#include <iostream>
#include <cstdio>
#include <cstdlib>
using namespace std;
typedef long long ll;

int main() {
    ll x, y, n;
    const int mod = int(1e9) + 7;
    cin >> x >> y >> n;
    if (n % 2 == 0) {
        ll ans = (y - x) % mod * (n / 2) % mod + x;
        ans = (ans + mod) % mod;
        cout << ans << endl;
    } else {
        ll ans = (y - x + mod) % mod * (n / 2) % mod + x;
        ans = (ans + mod) % mod;
        cout << ans << endl;
    }
    return 0;
}
