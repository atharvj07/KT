#include <bits/stdc++.h>
using namespace std;
const int MAXN = 3e5 + 10;
int n, A[MAXN], sum[MAXN], dp[MAXN];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> A[i];
        sum[i] = sum[i - 1] + A[i];
    }
    dp[1] = dp[2] = A[1] + A[2];
    for (int i = 3; i <= n; i++) {
        dp[i] = max(dp[i - 1], sum[i] + dp[i - 2]);
    }
    cout << dp[n] << endl;
    return 0;
}
