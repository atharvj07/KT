import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int d = input.nextInt();
        int s = input.nextInt();
        int n = 0;
        int sum = 0;
        while (true) {
            n += d;
            int tmp = n;
            sum = 0;
            while (tmp > 0) {
                sum += tmp % 10;
                tmp /= 10;
            }
            if (sum == s) {
                break;
            }
            if (sum > s) {
                n = -1;
                break;
            }
        }
        System.out.println(n);
    }
}
