#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5+10;
int n,a[N],ps[N];
int main() {
    cin >> n;
    for(int i=1;i<=n;i++) cin >> a[i];
    for(int i=1;i<=n;i++) ps[i] = ps[i-1]+a[i];
    int sum = ps[n];
    int ans = 0;
    int sum_l=0,sum_r=0;
    for(int i=1;i<=n;i++){
        sum_l+=a[i];
        sum_r=sum-sum_l;
        if(sum_l==sum_r) ans = sum_l;
    }
    cout << ans << '\n';
    return 0;
}
