import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int k = Integer.parseInt(input[1]);

        String lamps = br.readLine();
        char[] lampsArr = lamps.toCharArray();

        List<int[]> subsets = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            input = br.readLine().split(" ");
            int size = Integer.parseInt(input[0]);
            int[] subset = new int[size];
            String[] subsetArr = br.readLine().split(" ");
            for (int j = 0; j < size; j++) {
                subset[j] = Integer.parseInt(subsetArr[j]) - 1;
            }
            subsets.add(subset);
        }

        int[] minOps = new int[n];
        Arrays.fill(minOps, Integer.MAX_VALUE);

        minOps[0] = 0;
        for (int i = 1; i < n; i++) {
            for (int[] subset : subsets) {
                boolean[] subsetLamps = new boolean[n];
                for (int j = 0; j < subset.length; j++) {
                    subsetLamps[subset[j]] = true;
                }

                if (subsetLamps[i] && minOps[i - 1] != Integer.MAX_VALUE) {
                    minOps[i] = Math.min(minOps[i], minOps[i - 1] + 1);
                }
            }
        }

        for (int i = 0; i < n; i++) {
            System.out.println(minOps[i]);
        }
    }
}
