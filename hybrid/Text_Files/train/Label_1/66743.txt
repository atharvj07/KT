from sys import stdin
n=int(stdin.readline())
s=stdin.readline().strip()
ans=[0]*n
a=[0]*n
a[0]=1
ans[0]=1
for i in range(1,n):
    if s[i]!=s[i-1]: a[i]=a[i-1]+1
    else: a[i]=a[i-1]
b=[0]*n
b[n-1]=1
ans[n-1]=1
for i in range(n-2,-1,-1):
    if s[i]!=s[i+1]: b[i]=b[i+1]+1
    else: b[i]=b[i+1]
for i in range(n):
    ans[i]=a[i]+b[i]
    if ans[i]%2==0: ans[i]=-1
for i in ans: print(i)
