import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int cnt = 0;
                for (int l = 0; l < 15; l++) {
                    if (((a[i] >> l) & 1) != ((a[j] >> l) & 1)) {
                        cnt++;
                    }
                }
                if (cnt == k) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
