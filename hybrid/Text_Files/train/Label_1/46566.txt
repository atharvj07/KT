import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long a00 = scanner.nextLong();
        long a01 = scanner.nextLong();
        long a10 = scanner.nextLong();
        long a11 = scanner.nextLong();
        if (a01 != a10 || a00 + a11 != a01 + a10) {
            System.out.println("Impossible");
            return;
        }
        long n = a01;
        StringBuilder sb = new StringBuilder();
        for (long i = 0; i < n; i++) {
            sb.append('0');
            sb.append('1');
        }
        long m = a00 - n;
        for (long i = 0; i < m; i++) {
            sb.append('0');
        }
        System.out.println(sb.toString());
    }
}
