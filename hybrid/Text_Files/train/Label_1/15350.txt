#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const int N = 1e5;
int n, a[N + 5], dp[N + 5];
string ans;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    dp[n] = -1;
    for (int i = n - 1; i >= 1; i--) {
        dp[i] = -1;
        for (int j = i; j <= n; j += a[i])
            if (dp[j] != -1) {
                if (dp[i] == -1 || dp[i] == 0) dp[i] = 1 - dp[j];
                else if (dp[i] == 1 - dp[j]) continue;
                else return cout << "impossible" << endl, 0;
            }
    }
    for (int i = 1; i <= n; i++) ans += dp[i] == 1 ? 'A' : 'B';
    cout << ans << endl;
    return 0;
}
