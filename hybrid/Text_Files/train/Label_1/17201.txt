#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

ll gcd(ll a, ll b) {
  while (b) {
    a %= b;
    swap(a, b);
  }
  return a;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    ll n, r;
    cin >> n >> r;
    if (n < r)
      cout << 1 << endl;
    else if (n % r == 0)
      cout << (r + 1) / 2 << endl;
    else {
      ll g = gcd(n, r);
      n /= g;
      r /= g;
      if (n % r != 0)
        cout << r << endl;
      else
        cout << r + 1 << endl;
    }
  }
  return 0;
}
