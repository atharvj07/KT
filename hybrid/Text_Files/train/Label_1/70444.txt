n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
sum = [0]
for el in a:
    sum.append(sum[-1] + el)
for el in b:
    l, r = 0, n
    while l < r:
        m = (l + r) // 2
        if sum[m] >= el:
            r = m
        else:
            l = m + 1
    print(l, el - sum[l - 1])
