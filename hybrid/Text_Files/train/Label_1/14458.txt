#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 1e2 + 5;
ll n, a[MAXN], dp[MAXN][MAXN];
string s;
ll solve(int l, int r) {
    if (l > r) return 0;
    ll &ret = dp[l][r];
    if (ret != -1) return ret;
    ret = 0;
    for (int i = l; i <= r; i++) {
        if (s[i] == s[l]) ret = max(ret, solve(i + 1, r) + a[r - i + 1]);
    }
    return ret;
}
int main() {
    memset(dp, -1, sizeof(dp));
    cin >> n >> s;
    for (int i = 1; i <= n; i++) cin >> a[i];
    cout << solve(0, n - 1) << endl;
    return 0;
}
