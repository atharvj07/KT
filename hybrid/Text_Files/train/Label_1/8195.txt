#include <cstdio>
#include <vector>
using namespace std;

const int MAXN = 100 + 10;
char g[MAXN][MAXN];

int main() {
    int n, m;
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i++) {
        scanf("%s", g[i] + 1);
    }
    bool ok = true;
    vector<pair<int, int>> v;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (g[i][j] == '*') {
                bool ff = false;
                for (int k = 0; k <= 1; k++) {
                    for (int l = -1; l <= 1; l += 2) {
                        int ti = i + k * l, tj = j + (1 - k) * l;
                        if (ti < 1 || tj < 1 || ti > n || tj > m) {
                            continue;
                        }
                        if (g[ti][tj] == '*') {
                            ff = true;
                            break;
                        }
                    }
                    if (ff) {
                        break;
                    }
                }
                if (!ff) {
                    v.push_back(make_pair(i, j));
                    for (int k = 0; k <= 1; k++) {
                        for (int l = -1; l <= 1; l += 2) {
                            int ti = i + k * l, tj = j + (1 - k) * l;
                            if (ti < 1 || tj < 1 || ti > n || tj > m) {
                                continue;
                            }
                            g[ti][tj] = '*';
                        }
                    }
                } else {
                    ok = false;
                }
            }
        }
    }
    if (!ok) {
        puts("-1");
        return 0;
    }
    printf("%d\n", (int)v.size());
    for (auto x : v) {
        printf("%d %d 1\n", x.first, x.second);
    }
    return 0;
}
