#include <bits/stdc++.h>
using namespace std;

const long long MAXN = 5000;

int n;
long long s[MAXN + 2];

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> s[i];
    int ans = 0;
    while (true) {
      int f = 0;
      for (int i = 1; i <= n; i++) {
        if (s[i] != 1) {
          ans++;
          f = 1;
          while (i <= n && s[i] != 1) {
            s[i] = max(s[i] - 1, (long long)1);
            i += s[i];
          }
          break;
        }
      }
      if (!f) break;
    }
    cout << ans << endl;
  }
  return 0;
}
