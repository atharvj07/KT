import java.util.Scanner;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0){
			int n=sc.nextInt(),a[]=new int[n];
			for(int i=0;i<n;a[i]=i+1,i++);
			for(int i=n-1;i>0;i--){
				int x=a[i-1],y=a[i];
				if((x|y)<i){
					a[i-1]=y;
					a[i]=x;
				}
			}
			for(int x:a)System.out.print(x+" ");
			System.out.println();
		}
	}
}
