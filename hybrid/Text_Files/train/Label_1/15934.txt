#include <bits/stdc++.h>
using namespace std;

#define vi(x)  vector<x>
#define pb(x)  push_back(x)
#define all(x) (x).begin(),(x).end()
#define forn(i, n) for(int i = 0; i < (int)n; i++)

const int MAXN = 1e5 + 5;
const int MOD  = 1e9 + 7;
const int INF  = 2e9;

void solve() {
    int n;
    cin >> n;
    string s, t;
    cin >> s >> t;
    map<char, vi(int)> pos;
    forn(i, n) {
        pos[s[i]].pb(i);
    }
    int ans = 0;
    forn(i, n) {
        if (s[i] == t[i])
            continue;
        int ind = lower_bound(all(pos[t[i]]), i) - pos[t[i]].begin();
        if (ind == pos[t[i]].size()) {
            ans = -1;
            break;
        }
        ans += min(i - pos[t[i]][ind], n + pos[t[i]][ind] - i);
    }
    cout << ans << '\n';
}

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int q;
    cin >> q;
    while (q--) {
        solve();
    }
    return 0;
}
