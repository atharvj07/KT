import sys

p = [0, 3, 2, 1]
dp = [[[-1] * 4 for _ in range(4)] for _ in range(3)]
k, a, b = map(int, sys.stdin.readline().split())
A = [list(map(int, sys.stdin.readline().split())) for _ in range(3)]
B = [list(map(int, sys.stdin.readline().split())) for _ in range(3)]

def f(a, b, s):
    if s == k:
        return 0, 0
    if dp[a][b][s] != -1:
        return dp[a][b][s]
    dp[a][b][s] = []
    for i in range(1, 4):
        for j in range(1, 4):
            if (i == j):
                pt = f(i, j, s + 1)
            elif p[i] > p[j]:
                pt = f(i, j, s + 1)
                pt[0] += 1
            else:
                pt = f(i, j, s + 1)
                pt[1] += 1
            dp[a][b][s].append(pt)
    dp[a][b][s] = min(dp[a][b][s])
    return dp[a][b][s]

print(*f(A[a - 1][b - 1], B[a - 1][b - 1], 1))
