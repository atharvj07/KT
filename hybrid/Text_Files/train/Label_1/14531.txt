#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const int N = 1000005;
const ll MOD = 1e9 + 7;
ll dp[N];

inline ll cnt(ll x) { return __builtin_popcountll(x); }

int main() {
  ll n, ans = 0;
  cin >> n;
  unordered_map<ll, ll> freq;
  for (int i = 0; i < n; i++) {
    ll x;
    cin >> x;
    freq[dp[x]]++;
    dp[x]++;
    dp[x] %= MOD;
  }
  for (auto &x : freq) ans = (ans + x.second * (x.second - 1) / 2) % MOD;
  cout << ans << endl;
  return 0;
}
