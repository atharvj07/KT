#include<bits/stdc++.h>
using namespace std;
int main(){
  int n,l,r,ans=0;
  cin>>n;
  priority_queue<int,vector<int>,greater<int>> pq;
  for(int i=1;i<=n;++i){
    cin>>l>>r;
    l++;
    r++;
    pq.push(l);
    pq.push(r);
  }
  int sum=0;
  int cnt=0;
  while(!pq.empty()){
    sum+=pq.top();
    pq.pop();
    cnt++;
    if(sum>=cnt){
      ans=cnt;
    }
    else break;
  }
  cout<<ans<<endl;
  return 0;
}
