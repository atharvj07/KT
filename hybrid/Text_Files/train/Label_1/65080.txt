m=int(input())
q=[int(x) for x in input().split()]
n=int(input())
a=[int(x) for x in input().split()]
a.sort()
for i in range(m):
    for j in range(max(0,q[i]-2),q[i]):
        x=a[j]
        for p in range(j):
            for pp in range(p+1,j):
                x=min(x,a[p]+a[pp])
        for p in range(j+1,n):
            x=min(x,a[j]+a[p])
        a[j]=x
print(sum(a))
