#include<iostream>
using namespace std;
int main() {
  long long n, a[100005], b[100005];
  cin >> n;
  for (long long i = 1; i <= n; i++) cin >> a[i];
  bool ok = true;
  for (long long i = 1; i <= n; i++) {
    b[i] = b[i - 1] + a[i];
    if (b[i] <= b[i % n] || b[i] <= b[(i + 1) % n]) {
      ok = false;
    }
  }
  cout << (ok ? "YES\n" : "NO\n");
  if (ok) {
    for (long long i = 1; i <= n; i++) {
      if (b[i] > b[i % n] && b[i] > b[(i + 1) % n]) {
        for (long long j = i; j <= n; j++) cout << a[j] << " ";
        for (long long j = 1; j < i; j++) cout << a[j] << " ";
        cout << "\n";
        return 0;
      }
    }
  }
}
