from sys import stdin

# Read input
n = int(stdin.readline())
actions = (stdin.readline().strip().split() for _ in range(n))

# Initialize data structures
buy_prices = []
sell_prices = []
count = 1
mod = 10**9 + 7

# Process actions
for action, price in actions:
    price = int(price)
    if action == 'ADD':
        # Add price to the appropriate list
        if not sell_prices or price > sell_prices[0]:
            sell_prices.append(price)
        else:
            buy_prices.append(price)
    else:
        # Check if price is in the appropriate list
        if price in sell_prices:
            sell_prices.pop(0)
        elif price in buy_prices:
            buy_prices.pop(0)
        else:
            # Invalid action
            count = 0
            break

        # Update count
        count *= 2
        count %= mod

# Check if all orders were processed
if len(buy_prices) != len(sell_prices):
    count = 0

# Output result
print(count)
