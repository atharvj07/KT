import sys

n, m = map(int, input().split())
a = [[int(x) for x in input().split()] for _ in range(n)]
if any(any(x != a[x][y] for x in range(i, i + 2)) for i in range(n - 1) for y in range(m)):
    sys.exit(print(-1))
for i in range(n - 1):
    for y in range(m - 1):
        if a[i][y] and not a[i + 1][y] and not a[i][y + 1]:
            print(1, i + 1, y + 1)
        if not a[i][y] and a[i + 1][y] and a[i][y + 1]:
            print(2, i + 1, y + 1)
