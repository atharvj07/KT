#include<bits/stdc++.h>
using namespace std;
int main() {
  long long n, i, a[100005], mx=-1e9, mn=1e9;
  cin>>n;
  for(i=0;i<n;i++) {
    cin>>a[i];
    mx=max(mx, a[i]);
    mn=min(mn, a[i]);
  }
  long long ans=mx-mn;
  for(i=0;i<n;i++) {
    if(a[i]==mx) {
      ans=min(ans, mx-mn+1);
      if(i+1<n) ans=min(ans, mx-a[i+1]);
      if(i-1>=0) ans=min(ans, mx-a[i-1]);
    }
    else if(a[i]==mn) {
      ans=min(ans, mx-mn-1);
      if(i+1<n) ans=min(ans, a[i+1]-mn);
      if(i-1>=0) ans=min(ans, a[i-1]-mn);
    }
  }
  cout<<ans<<'\n';
}
