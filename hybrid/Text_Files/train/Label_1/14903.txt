#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<pair<int, int>> songs(n);
    for (int i = 0; i < n; i++) {
        cin >> songs[i].first >> songs[i].second;
    }
    sort(songs.begin(), songs.end(), [](pair<int, int> a, pair<int, int> b) { return a.second > b.second; });
    long long ans = 0;
    for (int i = 0; i < min(k, n); i++) {
        ans += (long long)songs[i].first * songs[i].second;
    }
    cout << ans << endl;
    return 0;
}
