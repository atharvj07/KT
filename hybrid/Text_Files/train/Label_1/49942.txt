import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long ans = 0;
        for (long i = 1000000000000000000L; i > 0; i /= 10) {
            ans += n / i;
            n %= i;
        }
        System.out.println(ans);
    }
}
