import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int k = scanner.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = scanner.nextInt();
    }

    int[] count = new int[n + 1];
    int ans = 0;
    for (int i = 0; i < n; i++) {
      count[a[i]]++;
      if (count[a[i]] == 1) {
        ans++;
      }
      if (ans > k) {
        break;
      }
    }

    System.out.println(ans);
  }
}
