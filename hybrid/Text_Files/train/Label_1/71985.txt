import sys

n, d = map(int, input().split())
s = input()

dp = [10**10] * (n + 1)
dp[1] = 0
for i in range(1, n + 1):
    if s[i - 1] == '0':
        continue
    for j in range(1, d + 1):
        if i + j <= n and s[i + j - 1] == '1':
            dp[i + j] = min(dp[i + j], dp[i] + 1)

if dp[n] == 10**10:
    print(-1)
else:
    print(dp[n])
