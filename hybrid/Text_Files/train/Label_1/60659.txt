r, c, n, k = map(int, input().split())
violas = [tuple(map(int, input().split())) for _ in range(n)]

def count_viola_in_rectangle(top, bottom, left, right):
    count = 0
    for viola in violas:
        if top <= viola[0] <= bottom and left <= viola[1] <= right:
            count += 1
    return count

def count_pictures():
    pictures = 0
    for top in range(1, r + 1):
        for bottom in range(top, r + 1):
            for left in range(1, c + 1):
                for right in range(left, c + 1):
                    if count_viola_in_rectangle(top, bottom, left, right) >= k:
                        pictures += 1
    return pictures

print(count_pictures())
