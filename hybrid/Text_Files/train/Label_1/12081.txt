#include <bits/stdc++.h>
using namespace std;
int main() {
  int a[6], n, b[100005];
  for (int i = 0; i < 6; i++) {
    cin >> a[i];
    ;
  }
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  int ans = 1e9;
  for (int i = 0; i < 6; i++) {
    ans = min(ans, b[0] - a[i]);
    for (int j = 1; j < n; j++) {
      ans = min(ans, b[j] - b[j - 1]);
    }
  }
  cout << ans << endl;
  return 0;
}
