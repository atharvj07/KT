def convert_scientific_notation_to_decimal(sci_notation):
    a, eb = sci_notation.split('e')
    e, b = 'e', int(eb)
    a_parts = a.split('.')
    if len(a_parts) == 1 or b >= len(a_parts[1]):
        result = a_parts[0] + a_parts[1][:b].ljust(b, '0')
    else:
        result = a_parts[0] + a_parts[1][:b] + '.' + a_parts[1][b:]
    return str(int(result) if result[-1] == '.' else result)

sci_notation = input()
print(convert_scientific_notation_to_decimal(sci_notation))
