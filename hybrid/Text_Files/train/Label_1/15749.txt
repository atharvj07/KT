#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> v(2 * n - 2);
  for (int i = 0; i < 2 * n - 2; i++) {
    cin >> v[i];
  }
  sort(v.begin(), v.end());
  string ans(2 * n - 2, ' ');
  for (int i = 0; i < 2 * n - 2; i += 2) {
    if (v[i] != v[i + 1]) {
      ans[i] = 'S';
      ans[i + 1] = 'P';
    }
  }
  cout << ans << endl;
  return 0;
}
