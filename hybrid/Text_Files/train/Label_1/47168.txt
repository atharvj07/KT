import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        System.out.println(pow(k, n));
    }

    static long pow(long x, long y) {
        long res = 1;
        while (y > 0) {
            if ((y & 1) == 1) {
                res *= x;
                res = res % (long) (1e9 + 7);
            }
            x *= x;
            x = x % (long) (1e9 + 7);
            y >>= 1;
        }
        return res;
    }
}
