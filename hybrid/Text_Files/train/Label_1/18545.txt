#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 998244353;
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  ll dp[n][2][2][k + 1]; // dp[i][same][prev][diff]
  memset(dp, 0, sizeof(dp));
  dp[0][1][1][0] = m;
  for (int i = 0; i < n - 1; i++) {
    // keep same color
    for (int same = 0; same <= 1; same++) {
      for (int prev = 0; prev <= 1; prev++) {
        for (int diff = 0; diff <= k; diff++) {
          (dp[i + 1][same][prev][diff] += dp[i][same][prev][diff]) %= mod;
          if (diff + 1 <= k) (dp[i + 1][same][prev][diff + 1] += dp[i][same][prev][diff] * m % mod) %= mod;
        }
      }
    }
    // change color
    for (int same = 0; same <= 1; same++) {
      for (int prev = 0; prev <= 1; prev++) {
        for (int diff = 0; diff <= k; diff++) {
          for (int color = 0; color < m; color++) {
            int cur_prev = color == prev;
            (dp[i + 1][!same][cur_prev][diff + 1] += dp[i][same][prev][diff]) %= mod;
          }
        }
      }
    }
  }
  cout << dp[n - 1][0][0][k] << "\n";
  return 0;
}
