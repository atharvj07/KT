#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int k = 0;
    for (int i = 1; i < n; i++) {
        if (a[i] - a[i - 1] <= 2 * k) {
            k++;
        }
    }
    cout << k << endl;
    return 0;
}
