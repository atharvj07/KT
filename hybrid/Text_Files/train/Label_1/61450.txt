import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    k = 0
    for x in a:
        if x % 2:
            k += 1
    odd = 0
    b = []
    for x in a:
        if (odd < k and x % 2) or (odd == k and x % 2 == 0):
            odd += x % 2
            b.append(x)
    if odd != k:
        print('NO')
    else:
        for x in a:
            if x not in b:
                b.append(x)
        print('YES')
        print(*b)
