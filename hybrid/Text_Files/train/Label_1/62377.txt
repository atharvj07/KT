n = int(input())
a = list(map(int, input().split()))
edges = [list(map(int, input().split())) for _ in range(n - 1)]

adj_list = [[] for _ in range(n)]
for u, v in edges:
    adj_list[u - 1].append(v - 1)
    adj_list[v - 1].append(u - 1)

def neighbors_strength_increase(a, adj_list):
    max_strength = [0] * n
    for i in range(n):
        for neighbor in adj_list[i]:
            max_strength[neighbor] = max(max_strength[neighbor], a[i] + 1)
            for second_neighbor in adj_list[neighbor]:
                if second_neighbor != i:
                    max_strength[second_neighbor] = max(max_strength[second_neighbor], a[i] + 1)
    return max_strength

max_strength_after_hacks = neighbors_strength_increase(a, adj_list)
min_strength_needed = max(max_strength_after_hacks)
print(min_strength_needed)
