import java.util.*;

public class Uni {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n - 1];
        int[] b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }

        int[] depth = new int[n + 1];
        dfs(1, 0, depth, a, b);

        int k = 0;
        for (int i = 1; i <= n; i++) {
            k = Math.max(k, depth[i]);
        }

        System.out.println(k);
    }

    static void dfs(int v, int p, int[] depth, int[] a, int[] b) {
        depth[v] = 0;
        for (int i = 0; i < a.length; i++) {
            if (a[i] == v && b[i] != p) {
                dfs(b[i], v, depth, a, b);
                depth[v] = Math.max(depth[v], depth[b[i]] + 1);
            } else if (b[i] == v && a[i] != p) {
                dfs(a[i], v, depth, a, b);
                depth[v] = Math.max(depth[v], depth[a[i]] + 1);
            }
        }
    }
}
