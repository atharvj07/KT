#include <bits/stdc++.h>
#define ll long long
using namespace std;

int main() {
    ll n, ans = 0, t;
    cin >> n;
    priority_queue<ll, vector<ll>, greater<ll>> pq;
    for (int i = 1; i <= n; i++) {
        cin >> t;
        ans += t;
        pq.push(t);
    }
    for (int i = 1; i <= n; i++) {
        ans -= 2 * pq.top();
        pq.pop();
    }
    cout << (ans + 10007) % 10007 << endl;
    return 0;
}
