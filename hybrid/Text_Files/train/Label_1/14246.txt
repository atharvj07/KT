#include<iostream>
using namespace std;
int n, a[1001], que[1001], head = 0, tail = 0, l = 0, r = 0, ans = 0;
int main() {
    cin >> n;
    for(int i = 1;i <= n; i++) {
        cin >> a[i];
        que[tail ++] = i;
        r = max(r, i + a[i]);
        while(head < tail && que[head] < r) {
            l = max(l, que[head] + a[que[head]]);
            head ++;
        }
        if(l >= n) {
            ans ++;
            l = r = que[tail - 1];
            head = tail = 0;
            que[tail ++] = i;
            r = max(r, i + a[i]);
        }
    }
    cout << ans << endl;
    return 0;
}
