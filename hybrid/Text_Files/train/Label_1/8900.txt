#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    unordered_set<int> s;
    for (int i = 0; i < n; i++) {
      int a;
      cin >> a;
      s.insert(a);
    }
    if (s.size() == 1)
      cout << 0 << endl;
    else
      cout << 1 << endl;
  }
  return 0;
}
