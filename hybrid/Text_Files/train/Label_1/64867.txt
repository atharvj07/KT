from collections import defaultdict

def dfs(node, parent, graph, cost, b, c):
    global total_cost, impossible
    ones_to_zero, zeros_to_one = 0, 0

    if b[node-1] != c[node-1]:
        if b[node-1] == 1:
            ones_to_zero += 1
        else:
            zeros_to_one += 1

    for child in graph[node]:
        if child == parent:
            continue
        child_ones_to_zero, child_zeros_to_one = dfs(child, node, graph, cost, b, c)
        ones_to_zero += child_ones_to_zero
        zeros_to_one += child_zeros_to_one

    if ones_to_zero and zeros_to_one:
        total_cost += cost[node-1] * (ones_to_zero + zeros_to_one)
        ones_to_zero = zeros_to_one = 0
    elif ones_to_zero or zeros_to_one:
        if node == 1:
            impossible = True

    return ones_to_zero, zeros_to_one

n = int(input())
cost, b, c = [], [], []
graph = defaultdict(list)
for _ in range(n):
    a_i, b_i, c_i = map(int, input().split())
    cost.append(a_i)
    b.append(b_i)
    c.append(c_i)

for _ in range(n-1):
    u, v = map(int, input().split())
    graph[u].append(v)
    graph[v].append(u)

total_cost = 0
impossible = False
dfs(1, -1, graph, cost, b, c)

if impossible:
    print(-1)
else:
    print(total_cost)
