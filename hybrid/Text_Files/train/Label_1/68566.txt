def find_lucky_permutation(n):
    if n % 4 in [1, 2]: return -1
    p = [0] * n
    for i in range(1, n + 1, 4):
        if i + 3 <= n:
            p[i-1], p[i], p[i+1], p[i+2] = i+1, i+3, i, i+2
    if n % 4 == 3:
        p[-3], p[-2], p[-1] = n-1, n, n-2
    return p

n = int(input().strip())
result = find_lucky_permutation(n)
if result == -1:
    print(-1)
else:
    print(' '.join(map(str, result)))
