#include <iostream>
#include <vector>
#include <cstring>
using namespace std;

int n, m;
vector<vector<char>> g;
vector<vector<vector<int>>> dp;
int f(int x, int y, int pre) {
    if (x == n) return 1;
    if (dp[x][y][pre] != -1) return dp[x][y][pre];
    int &ret = dp[x][y][pre];
    ret = 0;
    if (pre == -1) {
        ret = f(x + 1, y, -1);
        if (y + 2 < m) ret += f(x + 1, y + 1, g[x][y]);
        if (y + 1 < m) ret += f(x + 1, y + 1, g[x][y + 1]);
    } else if (g[x][y] != g[x - 1][y]) {
        ret = f(x + 1, y, g[x][y]);
        if (y + 1 < m) ret += f(x + 1, y + 1, g[x][y + 1]);
    } else if (y + 1 < m && g[x][y] != g[x - 1][y + 1]) {
        ret = f(x + 1, y + 1, g[x][y]);
    }
    return ret;
}

int main() {
    cin >> n >> m;
    g.resize(n, vector<char>(m));
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++) cin >> g[i][j];
    dp.resize(n + 1, vector<vector<int>>(m + 1, vector<int>(3, -1)));
    cout << f(0, 0, -1) << endl;
    return 0;
}
