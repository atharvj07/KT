n, m, k = map(int, input().split()); p = list(map(int, input().split()))
p.insert(0, 0)
s = [sum(p[i:i+m+1]) for i in range(1,n-m+2)]
ans = 0
for i in range(k):
    ans += max(s)
    j = s.index(max(s))
    s.pop(j)
    s[j+m-1] -= p[j]
    for k in range(1,m): s[j+k] -= p[j+k] - p[j+k-1]
print(ans)
