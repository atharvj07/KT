#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main()
{
    string s;
    cin >> s;
    set<string> st;
    for (int i = 0; i < 26; i++)
    {
        for (int j = 0; j <= s.size(); j++)
        {
            string a = s.substr(0, j);
            string b = s.substr(j);
            char c = 'a' + i;
            st.insert(a + c + b);
            st.insert(a + b + c);
        }
    }
    cout << st.size() << endl;
}
