#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    long long n, k;
    cin >> n >> k;
    vector<long long> a(n);
    for (long long i = 1; i <= n; ++i) a[i - 1] = i;
    if (k == 1) {
        cout << a[0];
        for (long long i = 1; i <= n - 1; ++i) cout << " " << i + 1;
    } else if (k == n - 1) {
        cout << a[n - 2];
        for (long long i = 0; i <= n - 3; ++i) cout << " " << a[i];
        cout << " " << a[n - 1];
    } else {
        cout << a[0];
        for (long long i = 1; i <= n - 1; ++i) cout << " " << a[i] + i % 2;
    }
    cout << endl;
}
