#include<iostream>
#include<cstdio>
#include<algorithm>
using namespace std;

int main()
{
    int n, k, arr[1005], mx = 0;
    cin >> n >> k;
    for(int i = 1; i <= n; i++) { cin >> arr[i]; mx = max(mx, arr[i]); }
    sort(arr+1, arr+1+n);
    int ans = 0;
    for(int i = n; i >= 1; i--)
    {
        if(arr[i] > 2 * k) ans++;
        else break;
        k = max(k, arr[i] / 2);
    }
    cout << ans << endl;
    return 0;
}
