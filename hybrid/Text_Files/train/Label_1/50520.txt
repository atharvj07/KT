import java.util.Scanner;

public class JoyStickGame {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a1 = scanner.nextInt();
        int a2 = scanner.nextInt();

        int maxMinutes = 0;
        while (a1 > 0 || a2 > 0) {
            if (a1 <= 1) {
                a1 += 1;
                a2 -= 2;
            } else if (a2 <= 1) {
                a2 += 1;
                a1 -= 2;
            } else {
                a1 -= 2;
                a2 -= 2;
            }
            maxMinutes++;
        }

        System.out.println(maxMinutes);
    }
}
