import java.util.Arrays;
import java.util.Scanner;

public class MrChanek {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int K = scanner.nextInt();

        int[] A = new int[N];
        int[] D = new int[N];
        int[] E = new int[N];

        for (int i = 0; i < N; i++) {
            A[i] = scanner.nextInt();
        }

        for (int i = 0; i < N; i++) {
            D[i] = scanner.nextInt();
        }

        for (int i = 0; i < N - 1; i++) {
            E[i] = i + 1;
        }

        // Calculate maximum energy for each bond change
        int[][] maxEnergy = new int[N][K + 1];
        for (int i = 0; i < N; i++) {
            for (int j = 0; j <= K; j++) {
                maxEnergy[i][j] = -1;
            }
        }

        // Calculate maximum energy for each bond change and excited atoms
        for (int i = 0; i < N; i++) {
            for (int j = 0; j <= K; j++) {
                if (j == 0) {
                    maxEnergy[i][j] = 0;
                } else {
                    for (int k = 0; k < i; k++) {
                        int newEnergy = A[i] - D[i] + maxEnergy[k][j - 1];
                        if (newEnergy > maxEnergy[i][j]) {
                            maxEnergy[i][j] = newEnergy;
                        }
                    }
                }
            }
        }

        // Find the maximum energy across all bond changes
        int maxTotalEnergy = 0;
        for (int i = 0; i < N; i++) {
            maxTotalEnergy = Math.max(maxTotalEnergy, maxEnergy[i][K]);
        }

        System.out.println(maxTotalEnergy);
    }
}
