import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (x[i] < x[i - 1]) {
                ans += x[i - 1] - x[i];
                x[i] = x[i - 1];
            }
        }
        System.out.println(ans);
    }
}
