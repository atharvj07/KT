#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, q;
  cin >> n >> q;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  while (q--) {
    char op;
    int x;
    cin >> op >> x;
    for (int i = 0; i < n; i++) {
      if (op == '>' && a[i] > x) { a[i] = -a[i]; }
      if (op == '<' && a[i] < x) { a[i] = -a[i]; }
    }
  }
  for (int i = 0; i < n; i++) { cout << a[i] << ' '; }
  return 0;
}
