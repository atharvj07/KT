import java.util.ArrayList;
import java.util.HashMap;
import java.util.Scanner;

public class Main {
    static class Node {
        int index;
        long gas;

        Node(int index, long gas) {
            this.index = index;
            this.gas = gas;
        }
    }

    static int n;
    static long[] gas;
    static ArrayList<Node>[] adj;

    static long dfs(int u, int p) {
        long maxGas = gas[u];

        for (Node v : adj[u]) {
            if (v.index == p) {
                continue;
            }

            maxGas = Math.max(maxGas, dfs(v.index, u) + v.gas - (v.gas < 0 ? v.gas : 0));
        }

        return maxGas;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        gas = new long[n];

        for (int i = 0; i < n; i++) {
            gas[i] = sc.nextInt();
        }

        adj = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }

        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            long w = sc.nextInt();
            adj[u].add(new Node(v, -w));
            adj[v].add(new Node(u, w));
        }

        sc.close();

        long maxGas = 0;
        for (int i = 0; i < n; i++) {
            maxGas = Math.max(maxGas, dfs(i, -1) + gas[i]);
        }

        System.out.println(maxGas);
    }
}
