import sys

m, k, n, s = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = [0] * (max(a) + 1)
for x in a: c[x] += 1
for x in b: c[x] -= 1
d = 0
for i in range(len(c)):
    d += max(c[i], 0)
    if d > s:
        print(-1)
        sys.exit(0)
d = 0
for i in range(len(a)):
    if c[a[i]] <= 0:
        d += 1
        print(i + 1, end=' ')
        c[a[i]] += 1
    else: c[a[i]] -= 1
