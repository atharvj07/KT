
#include <bits/stdc++.h>
using namespace std;
int n, a[5001], ans[5001];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) {
    for (int j = i; j <= n; j++) {
      int hi = 0, maxcnt = 0;
      for (int k = i; k <= j; k++) {
        if (a[k] == hi)
          maxcnt++;
        else if (maxcnt == 0) {
          hi = a[k];
          maxcnt = 1;
        } else
          maxcnt--;
      }
      ans[hi]++;
    }
  }
  for (int i = 1; i <= n; i++) cout << ans[i] << " ";
  return 0;
}
