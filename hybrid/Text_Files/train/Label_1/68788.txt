import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))

prefix = [0] * (n + 1)
for i in range(n):
    if i != n - 1:
        prefix[i + 1] = prefix[i] + a[i]
for i in range(m):
    l, r = map(int, input().split())
    if prefix[r] - prefix[l - 1] == 0:
        print(1)
    else:
        print(0)
