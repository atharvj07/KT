#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(n), sortedb(n);
  map<int, int> numa, numb;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    numa[a[i]]++;
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
    numb[b[i]]++;
    sortedb[i] = b[i];
  }
  sort(sortedb.begin(), sortedb.end());
  int x = 0;
  for (int i = 0; i < n; i++) {
    if (numa[sortedb[i]] >= numb[sortedb[i]]) {
      numa[sortedb[i]] -= numb[sortedb[i]];
      x += sortedb[i] * numb[sortedb[i]];
      numb[sortedb[i]] = 0;
    } else
      x = -1;
  }
  if (x == -1)
    cout << -1;
  else
    cout << x % m;
}
