import java.util.Scanner;

public class InnopolisHills {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] heights = new int[n];

        for (int i = 0; i < n; i++) {
            heights[i] = scanner.nextInt();
        }

        int[] answers = new int[n / 2 + 1];

        // Calculate the minimum time required for each number of houses
        for (int i = 1; i <= n / 2; i++) {
            int minHours = 0;
            int left = 0, right = n - 1;

            // Find the leftmost and rightmost hills that need to be leveled
            while (left < right && heights[left] >= heights[left + 1]) {
                left++;
            }
            while (left < right && heights[right] <= heights[right - 1]) {
                right--;
            }

            // Level the hills until the required number of houses can be built
            while (right - left + 1 > 2 * i) {
                if (heights[left] < heights[right]) {
                    minHours += heights[left + 1] - heights[left];
                    left++;
                } else {
                    minHours += heights[right] - heights[right - 1];
                    right--;
                }
            }

            answers[i] = minHours;
        }

        // Print the answers
        for (int answer : answers) {
            System.out.print(answer + " ");
        }
    }
}
