import java.util.*;
import static java.lang.Math.*;

class Main {
  public static void main(String[] args) {
    Scanner s = new Scanner(System.in);
    int h1 = s.nextInt();
    int a1 = s.nextInt();
    int c1 = s.nextInt();
    int h2 = s.nextInt();
    int a2 = s.nextInt();
    int n;
    if (h1 >= h2)
      n = 1;
    else if (h2 % a1 == 0)
      n = h2 / a1;
    else
      n = h2 / a1 + 1;
    System.out.println(n);
    for (int j = 0; j < n; j++) {
      if (h1 - a2 <= 0)
        System.out.println("HEAL");
      else
        System.out.println("STRIKE");
      h1 += c1 - a2;
    }
  }
}
