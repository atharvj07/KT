import java.util.*;

public class Main {
  public static int gcd(int a, int b) {
    if (b == 0)
      return a;
    return gcd(b, a % b);
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int[] arr = new int[1000001];
    int n = sc.nextInt();
    int k = sc.nextInt();
    int sum = 0;
    for (int i = 2; i <= 1000000; i++) {
      if (arr[i] == 0) {
        for (int j = i + i; j <= 1000000; j += i) {
          arr[j] = 1;
        }
        if (sum + i <= k) {
          sum += i;
          System.out.print(i + " ");
          n--;
        }
      }
      if (n == 0) {
        break;
      }
    }
    if (n > 0) {
      System.out.println(-1);
    }
    sc.close();
  }
}
