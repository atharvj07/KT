def f(p):
    n=len(p)
    s=[p[(i+1)%n]+p[i] for i in range(n)]
    s.sort()
    return s

for _ in range(int(input())):
    n=int(input())
    p=list(map(int,input().split()))
    for i in range(1,n):
        if f(p)!=f(p[1:]+[p[0]]):
            break
        p=p[1:]+[p[0]]
    print(' '.join(map(str,p)))
