#include<iostream>
using namespace std;
int32_t main(){
    int n, m;
    cin >> n >> m;
    for(int i = 1; i <= n; i++){
        if(i != n){
            for(int j = 1; j <= m; j++){
                if(i % 2) cout << i << ' ' << j << endl;
                else cout << i << ' ' << m - j + 1 << endl;
            }
        }
        else{
            for(int j = 1; j <= m; j++) cout << i << ' ' << j << endl;
        }
    }
    return 0;
}
