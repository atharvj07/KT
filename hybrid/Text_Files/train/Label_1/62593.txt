import sys
sys.setrecursionlimit(3000)

def get_aca_pos(s, i=0):
    if i == len(s):
        return 0, []
    if s[i].lower() == 'a' or s[i] == '?':
        j, pos = get_aca_pos(s, i+1)
        if j > 0:
            return i+1, pos
    elif s[i] == 'b':
        if i+1 < len(s) and (s[i+1].lower() == 'a' or s[i+1] == '?'):
            j, pos = get_aca_pos(s, i+2)
            if j > 0:
                return i+1, pos
    elif s[i] == 'c':
        if i+1 < len(s) and (s[i+1].lower() == 'a' or s[i+1] == '?'):
            if i+2 < len(s) and (s[i+2].lower() == 'b' or s[i+2] == '?'):
                if i+3 < len(s) and (s[i+3].lower() == 'a' or s[i+3] == '?'):
                    j, pos = get_aca_pos(s, i+4)
                    if j > 0:
                        return i+1, pos
    return 0, []

def solve(s):
    i, pos = get_aca_pos(s)
    if i > 0:
        t = []
        for j in range(len(s)):
            if j in pos:
                t.append('a')
            elif i == j:
                t.append('b')
            elif s[j] != '?':
                t.append(s[j])
            else:
                t.append('c')
        return ''.join(t)
    return 'No'

for t in range(int(input())):
    n = int(input())
    s = input()
    print(solve(s))
