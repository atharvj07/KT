from heapq import heappop, heappush
from collections import defaultdict

def dijkstra(graph, start):
    distance = defaultdict(lambda: float('inf'))
    distance[start] = 0
    queue = [(0, start)]
    
    while queue:
        dist, current = heappop(queue)
        
        if dist > distance[current]:
            continue
        
        for neighbor, cost in graph[current].items():
            new_cost = dist + cost
            if new_cost < distance[neighbor]:
                distance[neighbor] = new_cost
                heappush(queue, (new_cost, neighbor))
                
    return distance

n, m, s, t = map(int, input().split())
edges = []
graph = defaultdict(lambda: defaultdict(int))
reverse_graph = defaultdict(lambda: defaultdict(int))

for _ in range(m):
    a, b, l = map(int, input().split())
    edges.append((a, b, l))
    graph[a][b] = l
    reverse_graph[b][a] = l

distance_from_s = dijkstra(graph, s)
distance_from_t = dijkstra(reverse_graph, t)

min_distance = distance_from_s[t]

for a, b, l in edges:
    if distance_from_s[a] + l + distance_from_t[b] == min_distance:
        print("YES")
    elif distance_from_s[a] < float('inf') and distance_from_t[b] < float('inf'):
        reduced_time = max(1, min_distance - distance_from_s[a] - distance_from_t[b])
        if reduced_time < l:
            print("CAN", l - reduced_time)
        else:
            print("NO")
    else:
        print("NO")
