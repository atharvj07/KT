import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int w = in.nextInt();
        int b = in.nextInt();
        double p = b * 1.0 / (w + b);
        double pp = p;
        for (int i = b; i > 1; i--) {
            p *= pp * (i - 1.0) / i;
        }
        System.out.println(1 - p);
    }
}
