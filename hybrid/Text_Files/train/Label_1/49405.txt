import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int[] weight = new int[n];
        for (int i = 0; i < n; i++) weight[i] = input.nextInt();
        int[] order = new int[m];
        for (int i = 0; i < m; i++) order[i] = input.nextInt() - 1;
        int[] next = new int[n];
        Arrays.fill(next, -1);
        for (int i = m - 2; i >= 0; i--) next[order[i]] = order[i + 1];
        int[] dp = new int[n];
        Arrays.fill(dp, -1);
        dp[n - 1] = 0;
        for (int i = n - 2; i >= 0; i--) {
            dp[i] = dp[i + 1];
            if (next[order[i]] != -1) dp[i] = Math.max(dp[i], weight[next[order[i]]] + dp[next[order[i]]]);
        }
        int sum = 0;
        for (int i = 0; i < m; i++) {
            int j = order[i];
            sum += weight[j];
            if (next[j] != -1) sum -= dp[next[j]];
        }
        System.out.println(sum);
    }
}
