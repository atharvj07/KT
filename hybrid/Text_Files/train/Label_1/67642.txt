import sys
input = sys.stdin.readline

n = int(input())
heap = []
res = []
for _ in range(n):
    op, *x = input().split()
    if op == 'insert':
        heap.append(int(x[0]))
        res.append(op + ' ' + x[0])
    elif op == 'getMin':
        if not heap:
            print(-1)
            exit()
        if heap[0] == int(x[0]):
            res.append(op + ' ' + x[0])
        else:
            while heap and heap[0] < int(x[0]):
                res.append('removeMin')
                heap.pop(0)
            if heap and heap[0] == int(x[0]):
                res.append(op + ' ' + x[0])
            else:
                res.append('insert ' + x[0])
                heap.append(int(x[0]))
    else:
        if not heap:
            print(-1)
            exit()
        if heap[0] == int(x[0]):
            res.append(op)
            heap.pop(0)
        else:
            while heap and heap[0] < int(x[0]):
                res.append('removeMin')
                heap.pop(0)
            if heap and heap[0] == int(x[0]):
                res.append(op)
                heap.pop(0)
            else:
                print(-1)
                exit()
print(len(res))
for op in res:
    print(op)
