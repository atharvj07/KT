#include <bits/stdc++.h>
using namespace std;

const int MAXN = 110;

int n, a[MAXN], ans;

void solve() {
  sort(a + 1, a + n / 2 + 1);
  for (int i = 2; i <= n / 2; i++) {
    ans += min(a[i] - a[i - 1], n - a[i] + a[i - 1]);
  }
  printf("%d", ans);
}

int main() {
  cin >> n;
  for (int i = 1; i <= n / 2; i++) { cin >> a[i]; }
  solve();
  return 0;
}
