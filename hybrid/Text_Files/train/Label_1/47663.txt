import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();

        // If n is even, the maximum score is 4 * (n / 2) = 2 * n
        // If n is odd, the maximum score is 4 * (n / 2) + 2 = 2 * n + 2
        System.out.println(n % 2 == 0 ? 2 * n : 2 * n + 2);
    }
}
