#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
const int MAXN = 100005;
const int MAXM = 26;
int n, ans, fail[MAXN];
char s[MAXN];
map<int, int> mp[MAXN];

int main() {
    scanf("%s", s + 1);
    n = strlen(s + 1);
    int j = 0;
    for (int i = 2; i <= n; i++) {
        while (j && s[i] != s[j + 1]) j = fail[j];
        if (s[i] == s[j + 1]) j++;
        fail[i] = j;
    }
    for (int i = n; i; i--) {
        int len = i;
        mp[len][fail[i]]++;
        if (fail[i]) fail[i] = fail[fail[i]];
    }
    for (int i = 1; i <= n; i++) mp[i][i] += mp[i][fail[i]];
    for (int i = 1; i <= n; i++) if (mp[i][0]) ans++;
    printf("%d\n", ans);
    for (int i = 1; i <= n; i++) if (mp[i][0]) printf("%d %d\n", i, mp[i][0]);
    return 0;
}
