import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[][] seg = new long[n][2];
            for (int i = 0; i < n; i++) {
                seg[i][0] = sc.nextLong();
                seg[i][1] = sc.nextLong();
            }
            long l = 1000000000000000000L, r = 0;
            for (int i = 0; i < n; i++) {
                l = Math.min(l, seg[i][0]);
                r = Math.max(r, seg[i][1]);
            }
            System.out.println(Math.max(0, r - l + 1 - 2 * (n - 1)));
        }
    }
}
