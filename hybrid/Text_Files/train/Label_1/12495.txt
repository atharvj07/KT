#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, r, avg;
    cin >> n >> r >> avg;
    vector<pair<long long, long long>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    sort(v.begin(), v.end());
    long long sum = 0, cnt = 0;
    for (int i = 0; i < n; i++) {
        sum += v[i].first;
        long long need = (avg * n - sum - (r - v[i].first) * (n - i - 1) + v[i].first) / v[i].second;
        if (need <= 0) {
            cout << cnt << '\n';
            return 0;
        }
        sum += v[i].second * need;
        cnt += need;
    }
    cout << cnt << '\n';
    return 0;
}
