import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		
		int[] deg = new int[n + 1];
		for (int i = 1; i < n; i++) {
			int x = in.nextInt();
			int y = in.nextInt();
			deg[x]++;
			deg[y]++;
		}
		
		int oddDeg = 0;
		for (int i = 1; i <= n; i++) {
			if (deg[i] % 2 == 1) {
				oddDeg++;
			}
		}
		
		System.out.println(oddDeg / 2);
	}
}
