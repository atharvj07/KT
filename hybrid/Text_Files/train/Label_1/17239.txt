#include <bits/stdc++.h>

using namespace std;

int main() {
  long long b1, q, l, m;
  cin >> b1 >> q >> l >> m;
  vector<long long> a(m);
  for (int i = 0; i < m; i++) cin >> a[i];
  sort(a.begin(), a.end());
  long long i = 1, ans = 0;
  while (1) {
    long long x = b1 * pow(q, i - 1);
    if (abs(x) > l) break;
    if (x == 0) {
      ans++;
    } else if (binary_search(a.begin(), a.end(), x)) {
      ans++;
      b1 = q;
      i = 0;
    } else {
      ans++;
      b1 = x;
      i++;
    }
  }
  if (q == 0 || b1 == 0) cout << "inf" << endl; else cout << ans << endl;
}
