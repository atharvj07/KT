import java.util.Scanner;
import java.util.HashMap;
import java.util.ArrayList;
import java.util.LinkedList;

public class Main {
    static class Food {
        int id;
        int cnt;
        LinkedList<Integer> friends;

        public Food(int id, int cnt) {
            this.id = id;
            this.cnt = cnt;
            friends = new LinkedList<>();
        }
    }

    static Food[] foods;
    static boolean[] visited;
    static ArrayList<Integer> ans;
    static HashMap<Integer, Integer> map;

    public static void dfs(int cur) {
        if (visited[cur]) return;
        visited[cur] = true;

        Food food = foods[cur];
        ans.add(food.id);
        for (Integer friend : food.friends) {
            if (map.get(friend) == 2) dfs(friend);
            else map.put(friend, map.get(friend) + 1);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        foods = new Food[n + 1];
        for (int i = 1; i <= n; i++) {
            foods[i] = new Food(i, sc.nextInt());
        }

        map = new HashMap<>();
        for (int i = 0; i < m; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            foods[y].friends.add(x);
            foods[x].friends.add(y);
            map.put(x, 1);
            map.put(y, 1);
        }

        visited = new boolean[n + 1];
        ans = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                if (map.get(i) == 2) dfs(i);
                else {
                    System.out.println("DEAD");
                    return;
                }
            }
        }

        System.out.println("ALIVE");
        StringBuilder sb = new StringBuilder();
        for (int id : ans) sb.append(id + " ");
        System.out.println(sb.substring(0, sb.length() - 1));
    }
}
