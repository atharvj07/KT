import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String n = sc.next();
        char[] arr = n.toCharArray();
        Arrays.sort(arr);
        String max1 = new StringBuilder(new String(arr)).reverse().toString();
        String max2 = max1;
        int i = n.length() - 1;
        while (i >= 0 && arr[i] == '0') i--;
        if (i >= 0) {
            char temp = arr[i];
            arr[i] = arr[0];
            arr[0] = temp;
            max2 = new String(arr);
        }
        System.out.println(max1);
        System.out.println(max2);
    }
}
