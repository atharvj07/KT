import sys

n, l, r, x = map(int, sys.stdin.readline().split())
c = list(map(int, sys.stdin.readline().split()))
c.sort()
ans = 0
for i in range(1 << n):
    tot = 0
    mn = 1e9
    mx = -1
    for j in range(n):
        if (i >> j) & 1:
            tot += c[j]
            mn = min(mn, c[j])
            mx = max(mx, c[j])
    if l <= tot <= r and mx - mn >= x:
        ans += 1

print(ans)
