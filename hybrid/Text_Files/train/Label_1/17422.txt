#include <bits/stdc++.h>
using namespace std;

const int mxN = 2e5;

int a[mxN], ans[mxN];
long long b[mxN];

int main() {
    int n, m;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    cin >> m;
    while (m--) {
        int ty, u, v;
        cin >> ty >> u >> v;
        if (ty == 1) {
            b[u] += v;
        } else {
            long long cur = b[u];
            for (int j = u; j <= n && cur > 0; j++) {
                cur = min(cur, 1LL * a[j] - ans[j]);
                ans[j] += cur;
                cur -= cur;
            }
            cout << ans[u] << '\n';
        }
    }
}
