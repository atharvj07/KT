import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        sc.nextLine();
        long[][] robots = new long[n][3];
        for (int i = 0; i < n; i++) {
            String[] line = sc.nextLine().split(" ");
            robots[i][0] = Long.parseLong(line[0]);
            robots[i][1] = Long.parseLong(line[1]);
            robots[i][2] = Long.parseLong(line[2]);
        }
        Arrays.sort(robots, (a, b) -> Long.compare(a[0], b[0]));
        long count = 0;
        for (int i = 0; i < n; i++) {
            long left = Math.max(robots[i][0] - robots[i][1], 0);
            long right = robots[i][0] + robots[i][1];
            int l = i - 1;
            int r = i + 1;
            while (l >= 0) {
                if (robots[l][0] < left) {
                    break;
                }
                else if (Math.abs(robots[l][2] - robots[i][2]) <= k) {
                    count++;
                }
                l--;
            }
            while (r < n) {
                if (robots[r][0] > right) {
                    break;
                }
                else if (Math.abs(robots[r][2] - robots[i][2]) <= k) {
                    count++;
                }
                r++;
            }
        }
        System.out.println(count);
    }
}
