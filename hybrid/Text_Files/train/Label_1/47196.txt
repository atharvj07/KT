import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt(), app = 0, unread = 0;
        int[] apps = new int[n];
        while (q-- > 0) {
            int type = sc.nextInt();
            if (type == 1) {
                int x = sc.nextInt() - 1;
                if (apps[x]++ == 0) unread++;
            } else if (type == 2) {
                int x = sc.nextInt() - 1;
                unread -= --apps[x];
            } else {
                int t = sc.nextInt();
                unread = Math.max(unread - apps[app++], 0);
            }
            System.out.println(unread);
        }
    }
}
