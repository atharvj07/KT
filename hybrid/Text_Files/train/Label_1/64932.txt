import sys
input = sys.stdin.readline

def solve():
    n, m, k = map(int, input().split())
    grid = [input().strip() for _ in range(n)]

    if grid[0][0] == '*' and (grid[0][1] == '.' or grid[1][0] == '.'):
        return 'NO'

    for i in range(n):
        for j in range(m):
            if grid[i][j] == '*':
                if (i == 0 or grid[i - 1][j] == '.') and (j == 0 or grid[i][j - 1] == '.'):
                    return 'NO'

    return 'YES'

for _ in range(int(input())):
    print(solve())
