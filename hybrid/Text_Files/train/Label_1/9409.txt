#include <bits/stdc++.h>
#define mp make_pair
using namespace std;
typedef long long ll;

const int MAXN = 105;
int cand[MAXN], city[MAXN], n, m;

int main() {
  cin >> n >> m;
  for (int i = 1; i <= m; ++i) {
    vector<pair<int, int>> v;
    for (int j = 1; j <= n; ++j) {
      int k; cin >> k;
      v.push_back(mp(k, j));
    }
    sort(v.rbegin(), v.rend());
    city[v[0].second]++, cand[v[0].second] += v[0].first;
  }
  vector<pair<int, int>> v;
  for (int i = 1; i <= n; ++i) { v.push_back(mp(cand[i], i)); }
  sort(v.rbegin(), v.rend());
  vector<pair<int, int>> c;
  for (int i = 1; i <= m; ++i) { c.push_back(mp(city[i], i)); }
  sort(c.rbegin(), c.rend());
  cout << (v[0].first == c[0].first ? min(v[0].second, c[0].second)
                                   : max(v[0].second, c[0].second))
       << endl;
  return 0;
}
