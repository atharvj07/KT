import sys
input = sys.stdin.readline

n = int(input())
y = list(sorted(map(int, input().split())))

# c = [1] * (n + 1)
c = [[1] * (n + 1) for _ in range(3)]

for i in range(1, n + 1):
    for j in range(1, 3):
        for k in range(i - 1, -1, -1):
            if y[i - 1] > c[j][k] * 2 + j:
                break
            c[j][i] = max(c[j][i], c[j][k] * 2 + j)

    for j in range(1, 3):
        if c[j][i] == y[i - 1]:
            for k in range(i):
                c[j][k] = 0
            break

x = []
j = 2
for i in range(n - 1, -1, -1):
    if c[j][i] == y[i]:
        x.append(c[j][i] // 2)
        if c[j][i] % 2 == 1:
            j = 1
    else:
        x.append(c[j][i])

print(*reversed(x))
