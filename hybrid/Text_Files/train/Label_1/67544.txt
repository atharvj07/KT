from itertools import combinations

n, l, r, x = map(int, input().split())
c = list(map(int, input().split()))

count = 0
for i in range(2, n + 1):
    for combo in combinations(c, i):
        if l <= sum(combo) <= r and max(combo) - min(combo) >= x:
            count += 1
            
print(count)
