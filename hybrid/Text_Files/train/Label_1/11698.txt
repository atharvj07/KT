#include<bits/stdc++.h>
using namespace std;

#define endl '\n'
#define ld long double
#define ll long long

const int maxn = 15;
const int mod = 1e9 + 7;
int n, l, r, x;
int c[maxn];

vector<int> a;

int32_t main() {
    cin >> n >> l >> r >> x;
    for (int i = 0; i < n; i++) {
        cin >> c[i];
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j) {
                continue;
            }
            int tot = c[i] + c[j];
            if (tot >= l && tot <= r && abs(c[i] - c[j]) >= x) {
                cout << 1 << endl;
                return 0;
            }
        }
    }
    for (int i = 0; i < n; i++) {
        a.push_back(c[i]);
    }
    sort(a.begin(), a.end());
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            for (int k = j + 1; k < n; k++) {
                int tot = a[i] + a[j] + a[k];
                if (tot >= l && tot <= r && a[j] - a[i] >= x && a[k] - a[j] >= x) {
                    cout << 1 << endl;
                    return 0;
                }
            }
        }
    }
    cout << 0 << endl;
    return 0;
}
