#include <bits/stdc++.h>
#define ll long long

using namespace std;

const ll M=1e18;
const ll N=2e5+5;
int n;
ll l[N],r[N];
ll cnt[N]={};
vector<ll> pos;

int main() {
  cin>>n;
  for(int i=1;i<=n;++i){
    cin>>l[i]>>r[i];
  }
  for(int i=1;i<=n;++i){
    pos.push_back(l[i]);
    pos.push_back(r[i]+1);
  }
  sort(pos.begin(),pos.end());
  pos.erase(unique(pos.begin(),pos.end()),pos.end());
  ll last=0;
  for(int i=1;i<pos.size();++i){
    ll p=pos[i];
    for(int j=1;j<=n;++j){
      if(l[j]<=p&&p<=r[j]){
        ++cnt[j];
      }
    }
    ll cnt_all=p-last;
    for(int k=1;k<=n;++k){
      cnt_all-=cnt[k];
    }
    last=p;
    cnt_all+=cnt[n];
    cnt[n]=cnt_all;
  }
  for(int i=1;i<=n;++i){
    cout<<cnt[i]<<' ';
  }
  cout<<endl;
  return 0;
}
