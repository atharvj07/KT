import sys
input = sys.stdin.readline

def solve(n):
    if n & 1:
        print("NO")
        return

    print("YES")
    for i in range(2, n+1, 2):
        print(i, end=' ')
    for i in range(1, n+1, 2):
        print(i, end=' ')
    print()

    for i in range(1, n+1):
        print(n+1-i, end=' ')
    print()

n = int(input())
solve(n)
