#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> ans(n, 1);
  vector<bool> vis(n + 1);
  for (int i = 2; i <= n; i++) {
    if (vis[i]) continue;
    vector<int> vec;
    for (int j = i; j <= n; j += i) {
      vis[j] = 1;
      vec.push_back(ans[j - 1]);
    }
    int g = 0;
    for (auto &x : vec) g = __gcd(g, x);
    for (int j = i; j <= n; j += i) { ans[j - 1] = g; }
  }
  for (auto &x : ans) cout << x << ' ';
  cout << '\n';
}
