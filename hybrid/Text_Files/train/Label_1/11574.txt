#include<bits/stdc++.h>
using namespace std;
int main() {
  char a[100][100];
  int n, i, j;
  cin >> n;
  for (i = 0; i < n; i++)
    if (i % 2 == 0) {
      for (j = 0; j < n; j++)
        if (j % 2 == 0)
          a[i][j] = 'W';
        else
          a[i][j] = 'B';
    } else {
      for (j = 0; j < n; j++)
        if (j % 2 == 0)
          a[i][j] = 'B';
        else
          a[i][j] = 'W';
    }
  for (i = 0; i < n; i++)
    cout << a[i] << "\n";
  return 0;
}
