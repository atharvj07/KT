import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long[][] data = new long[n][2];
        for (int i = 0; i < n; i++) {
            data[i][0] = sc.nextInt();
            data[i][1] = sc.nextInt();
        }
        Arrays.sort(data, (a, b) -> Long.compare(a[0] * a[1], b[0] * b[1]));
        long ans = 0L;
        for (int i = 0; i < k; i++) {
            ans += data[i][0] * data[i][1];
        }
        System.out.println(ans);
    }
}
