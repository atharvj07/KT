n = int(input())
solutions = [list(map(int, input().split())) for _ in range(n)]

participant_submissions = {}

for x, k in solutions:
    if k in participant_submissions:
        if participant_submissions[k] != x:
            print("NO")
            break
        participant_submissions[k] += 1
    else:
        if x != 0:
            print("NO")
            break
        participant_submissions[k] = 1
else:
    print("YES")
