import java.util.Scanner;

public class Mancala {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] holes = new int[14];

        // Read the number of stones in each hole
        for (int i = 0; i < 14; i++) {
            holes[i] = scanner.nextInt();
        }

        int maxScore = 0;

        // Try all possible moves
        for (int i = 0; i < 14; i++) {
            if (holes[i] > 0) {
                int[] newHoles = simulateMove(holes, i);
                int score = calculateScore(newHoles);
                maxScore = Math.max(maxScore, score);
            }
        }

        System.out.println(maxScore);
    }

    // Simulate a move from a given hole
    private static int[] simulateMove(int[] holes, int holeIndex) {
        int stones = holes[holeIndex];
        holes[holeIndex] = 0;

        int currentIndex = holeIndex;
        while (stones > 0) {
            currentIndex = (currentIndex + 1) % 14;
            holes[currentIndex]++;
            stones--;
        }

        return holes;
    }

    // Calculate the score after a move
    private static int calculateScore(int[] holes) {
        int score = 0;
        for (int i = 0; i < 14; i++) {
            if (holes[i] % 2 == 0) {
                score += holes[i];
            }
        }
        return score;
    }
}
