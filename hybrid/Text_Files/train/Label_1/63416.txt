import sys
input = sys.stdin.readline

n = int(input())
a = []
b = []
for _ in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)
c = [(a[i] + b[i] - 1) // b[i] + a[i] for i in range(n)]
c.sort()
ans = sum(c)
for i in range(n):
    if a[i] == c[i] or ans < b[i]:
        break
    ans -= (a[i] - 1) // b[i]
print(ans)
