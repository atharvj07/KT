def check_constraints(i, j, constraints, heights):
    for c in constraints:
        x, sign, y = c
        x, y = x-1, y-1
        if i <= x <= j and i <= y <= j:
            if sign == '=' and heights[x] != heights[y]:
                return False
            elif sign == '<' and heights[x] >= heights[y]:
                return False
            elif sign == '>' and heights[x] <= heights[y]:
                return False
            elif sign == '<=' and heights[x] > heights[y]:
                return False
            elif sign == '>=' and heights[x] < heights[y]:
                return False
    return True

def count_mausoleums(n, k, constraints):
    dp = [[-1] * (2*n) for _ in range(2*n)]
    
    def dfs(i, j):
        if i >= j: return 1
        if dp[i][j] != -1: return dp[i][j]
        total = 0
        for height in range(1, n+1):
            for di, dj in ((i+1, j), (i, j-1)):
                heights[i], heights[j] = height, height
                if check_constraints(i, j, constraints, heights):
                    total += dfs(di, dj)
                heights[i], heights[j] = 0, 0
        dp[i][j] = total
        return total
    
    heights = [0] * (2*n)
    return dfs(0, 2*n-1)

n, k = map(int, input().split())
constraints = [input().split() for _ in range(k)]
for i in range(k):
    constraints[i][0] = int(constraints[i][0])
    constraints[i][2] = int(constraints[i][2])

print(count_mausoleums(n, k, constraints))
