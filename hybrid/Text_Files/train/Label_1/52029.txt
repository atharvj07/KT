import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        try {
            BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
            String[] nk = br.readLine().split(" ");
            int n = Integer.parseInt(nk[0]);
            int k = Integer.parseInt(nk[1]);
            long b = Long.parseLong(br.readLine());
            String[] aiS = br.readLine().split(" ");
            long[] ai = new long[n];
            for (int i = 0; i < n; ++i) {
                ai[i] = Long.parseLong(aiS[i]);
            }

            Arrays.sort(ai);
            long limit = 0;
            int idx = -1;
            for (int i = 0; i < n; ++i) {
                long tmp = limit;
                limit += ai[i];
                if (b < tmp || b < limit) {
                    idx = i;
                    break;
                }
            }

            System.out.println(idx == -1 ? n : idx);
        } catch (IOException e) {
            e.printStackTrace();
            System.exit(1);
        }
    }
}
