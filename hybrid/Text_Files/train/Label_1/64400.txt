import math

n = int(input())
lights = input()
l = []
for i in range(n):
    a, b = map(int, input().split())
    l.append((a, b))

mx = 0
for i in range(100):
    cnt = 0
    for j in range(n):
        a, b = l[j]
        if (i - b) % a == 0:
            cnt += lights[j] == '1'
    mx = max(mx, cnt)
    
print(mx)
