#include <bits/stdc++.h>

long long fast_div(long long a, long long b) {
  return (a + b - 1) / b;
}

int main() {
  long long x, y, z, mn;
  std::cin >> x >> y >> z;
  mn = std::min({fast_div(x, z), fast_div(y, z), (x + y) / z});
  std::cout << mn << " "
            << (fast_div(x + y, z) - mn) * (std::min(x, y) % z != 0) << std::endl;
  return 0;
}
