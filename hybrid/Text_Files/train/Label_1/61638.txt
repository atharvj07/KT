import sys
input = sys.stdin.readline

n,k=map(int,input().split())
a=[]
b=[]
t=[]
for i in range(n):
    ai,bi,ti=map(int,input().split())
    a.append(ai)
    b.append(bi)
    t.append(ti)
t.sort()
ans=1e18
sum=0
aa=a.count(1)
bb=b.count(1)
if aa<k or bb<k: print(-1)
else:
    for i in range(n):
        if a[i]==1: aa-=1
        if b[i]==1: bb-=1
        if aa>=k and bb>=k: ans=min(ans,sum)
        sum+=t[i]
    print(ans)
