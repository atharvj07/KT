n, k = map(int, input().split())
slogan = input()

actions = []

if k - 1 <= n - k:
    actions += ["LEFT"] * (k - 1)
    for i in range(k - 1):
        actions.append("PRINT {}".format(slogan[i]))
        if i < k - 2:
            actions.append("RIGHT")
else:
    actions += ["RIGHT"] * (n - k)
    for i in range(n - 1, k - 2, -1):
        actions.append("PRINT {}".format(slogan[i]))
        if i > k - 1:
            actions.append("LEFT")

for action in actions:
    print(action)
