#include <bits/stdc++.h>
using namespace std;
int n, a[25];
int main() {
    cin >> n;
    for (int i = 0; i < n * n; i++) cin >> a[i];
    sort(a, a + n * n);
    if (n % 2 == 1) {
        for (int i = 0, j = n * n - 1, k = 0; k < n * n / 2; i++, j--, k++) {
            if (a[i] != a[j]) {
                cout << "NO" << endl;
                return 0;
            }
            if (k < n / 2) {
                cout << a[i] << " ";
            } else if (k == n / 2) {
                cout << a[i] << endl;
            } else {
                cout << a[j] << " ";
            }
        }
    } else {
        for (int i = 0, j = n * n - 1; i < n * n; i++, j--) {
            if (a[i] != a[j]) {
                cout << "NO" << endl;
                return 0;
            }
            if (i < n / 2) {
                cout << a[i] << " ";
            } else if (i == n / 2) {
                cout << a[i] << endl;
            } else {
                cout << a[j] << " ";
            }
        }
    }
    return 0;
}
