#include <bits/stdc++.h>
using namespace std;
const int MX = 1 << 18;
int a[MX], b[MX], x, n;
int main() {
  cin >> n >> x;
  for (int i = 0; i < (1 << n); ++i) {
    b[i] = b[i >> 1] ^ (i & 1);
  }
  if (b[x]) {
    puts("0");
    return 0;
  }
  for (int i = (1 << n) - 1; i; i--) {
    if ((b[i] ^ b[i ^ x]) > b[i]) {
      a[i] = 1;
      x ^= i ^ b[i ^ x];
    }
  }
  int len = 0;
  for (int i = (1 << n) - 1; i; i /= 2) len += a[i];
  cout << len << endl;
  for (int i = (1 << n) - 1; i; i /= 2)
    if (a[i]) cout << i << ' ';
  cout << endl;
  return 0;
}
