import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int t = s.nextInt();
        while(t-->0){
            int n = s.nextInt();
            int[] a = new int[n+1];
            for(int i=1;i<=n;i++){
                a[i] = s.nextInt();
            }
            int ans = 0;
            for(int i=1;i<=n;i++){
                if(a[i]!=i){
                    ans++;
                    i = a[i];
                }
            }
            System.out.println(ans/2);
        }
    }
}
