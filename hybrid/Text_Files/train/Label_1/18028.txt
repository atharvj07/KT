#include<bits/stdc++.h>
#define ll long long
using namespace std;
bool a[105];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        int x, y;
        cin >> x >> y;
        for (int j = x; j <= y; j++)
            a[j] = 1;
    }
    int k = 0;
    for (int i = 1; i <= m; i++)
        if (!a[i]) k++;
    cout << k << endl;
    for (int i = 1; i <= m; i++)
        if (!a[i]) cout << i << " ";
}
