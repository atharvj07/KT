for _ in range(int(input())):
    n, k = map(int, input().split())
    if n == k:
        print(0)
    else:
        a = 0
        if n < k:
            a = k - n
            n += k - n
        else:
            a = n - k
            n -= a
        if n % 2 != a % 2:
            n -= 1
            a += 1
        print((n - a) // 2 + a // 2)
