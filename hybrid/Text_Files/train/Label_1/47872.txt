import java.util.*;

public class Main {
    static int[] a;
    static int[] res;
    static int[] depth;
    static int[] size;
    static int[] mxDepth;
    static ArrayList<Integer>[] edges;
    static ArrayList<Integer>[] depthEdges;
    static int maxDepth;
    static int maxDiff;
    static int root;

    static void dfs1(int v, int p) {
        size[v] = 1;
        for (int u : edges[v]) {
            if (u == p) continue;
            depth[u] = depth[v] + 1;
            dfs1(u, v);
            size[v] += size[u];
            if (size[u] > size[mxDepth[v]]) {
                mxDepth[v] = u;
            }
        }
    }

    static void dfs2(int v, int p) {
        res[v] = maxDiff;
        for (int u : edges[v]) {
            if (u == p || u == mxDepth[v]) continue;
            maxDiff += a[u] == 1 ? 2 : -2;
            dfs2(u, v);
            maxDiff -= a[u] == 1 ? 2 : -2;
        }
        if (mxDepth[v] != 0) {
            maxDiff += a[mxDepth[v]] == 1 ? 2 : -2;
            dfs2(mxDepth[v], v);
            maxDiff -= a[mxDepth[v]] == 1 ? 2 : -2;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        a = new int[n + 1];
        res = new int[n + 1];
        depth = new int[n + 1];
        size = new int[n + 1];
        mxDepth = new int[n + 1];
        edges = new ArrayList[n + 1];
        depthEdges = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
            edges[i] = new ArrayList<>();
            depthEdges[i] = new ArrayList<>();
        }
        for (int i = 1; i < n; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            edges[u].add(v);
            edges[v].add(u);
        }
        dfs1(1, 1);
        for (int i = 1; i <= n; i++) {
            if (size[i] > size[root]) {
                root = i;
            }
        }
        dfs1(root, root);
        for (int i = 1; i <= n; i++) {
            maxDepth = depth[i];
            maxDiff = a[i] == 1 ? 1 : -1;
            dfs2(i, i);
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(res[i] + " ");
        }
    }
}
