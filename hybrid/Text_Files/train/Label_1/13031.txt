#include <bits/stdc++.h>
using namespace std;
#define rep(i, j, k)for(int i = j; i <= k; i++)
int n, m;
int a[2005][2005];
int b[2005];
int main() {
        cin >> n >> m;
        rep(i, 1, n) {
                rep(j, 1, m) {
                        char x; cin >> x;
                        if(x == '1') a[i][j] = 1;
                }
        }
        rep(i, 1, m) {
                rep(j, 1, n) {
                        if(a[j][i] == 1) {
                                b[i] = 1;
                                break;
                        }
                }
        }
        bool f = 0;
        rep(i, 1, m) {
                if(b[i] == 0) {
                        cout << "YES" << endl;
                        f = 1;
                        break;
                }
        }
        if(f) return 0;
        else cout << "NO" << endl;
}
