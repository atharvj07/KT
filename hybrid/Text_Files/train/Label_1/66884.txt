import sys
home, away = input(), input()
n = int(sys.stdin.readline())
players = {}
for _ in range(n):
    t, s, m, c = sys.stdin.readline().split()
    if s == home:
        players.setdefault(m, [t, c]).sort()
    else:
        players.setdefault(m, [t, c]).sort(reverse=True)
for m, j in players.items():
    if 'r' in j:
        print(home if j[0] == 'y' else away, m, j[1])
