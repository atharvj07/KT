#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main()
{
    int l, a;
    cin >> l >> a;
    a++;
    string s = to_string(a);
    int p = s.size();
    while (p < l)
    {
        s += s;
        p = s.size();
    }
    int p2 = p / 2;
    if (p % 2 == 0)
    {
        for (int i = 0; i < p2; i++)
        {
            if (s[i] != s[i + p2])
            {
                while (s[i] != s[i + p2])
                {
                    if (s[i] > '9')
                    {
                        s[i] = '0';
                        if (s[i - 1] < '9')
                            s[i - 1]++;
                        else
                        {
                            i--;
                        }
                    }
                    else
                    {
                        s[i]++;
                    }
                }
            }
        }
    }
    else
    {
        for (int i = 0; i < p2; i++)
        {
            if (s[i] != s[i + p2])
            {
                while (s[i] != s[i + p2])
                {
                    if (s[i] > '9')
                        s[i] = '0';
                    else
                        s[i]++;
                }
            }
        }
        if (s[p2] > '9')
            s[p2] = '0';
        else
            s[p2]++;
    }
    cout << s << endl;
    return 0;
}
