import java.util.*;

class Main {
    static Scanner in = new Scanner(System.in);
    static int h, w;
    static int[] left, right;
    static int[] arr;
    static int maxArea;
    static int[][] mat;

    public static void main(String[] args) {
        h = in.nextInt(); w = in.nextInt();
        mat = new int[h][w];
        left = new int[w];
        right = new int[w];
        for (int i = 0; i < h; i++)
            for (int j = 0; j < w; j++)
                mat[i][j] = in.nextInt();
        for (int i = 0; i < h; i++) {
            Arrays.fill(right, w);
            Arrays.fill(left, -1);
            arr = mat[i];
            for (int j = 0; j < w; j++) {
                if (arr[j] == 0) left[j] = Math.max(left[j], j);
            }
            for (int j = w - 1; j >= 0; j--)
                if (arr[j] == 0) right[j] = Math.min(right[j], j);

            for (int j = 0; j < w; j++) {
                if (arr[j] == 0)
                    maxArea = Math.max(maxArea, (right[j] - left[j] - 1) * (i + 1));
            }
        }
        System.out.println(maxArea);
    }
}
