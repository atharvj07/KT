m = int(input())
cards = list(map(int, input().split()))

cards_count = {}
for card in cards:
    if card in cards_count:
        cards_count[card] += 1
    else:
        cards_count[card] = 1

ascending = []
descending = []

for card in sorted(cards_count.keys()):
    ascending.append(card)
    if cards_count[card] > 1:
        descending.append(card)

result = ascending + descending[::-1]
print(len(result))
print(*result)
