#include <iostream>
#include <vector>
#include <cstdio>
#define int long long
using namespace std;
vector<int> a;
int32_t main() {
    int q;
    cin >> q;
    while (q--) {
        int n, k;
        cin >> n >> k;
        a.resize(n);
        bool ok = 1;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            sum += a[i];
            if (a[i] % 2 == 0)
                ok = 0;
        }
        sum %= 2;
        if (sum == 0 || (n - 1) % 2 == 0 || k % 2 == 0 || ok == 0) {
            cout << "NO" << endl;
            continue;
        }
        cout << "YES" << endl;
        for (int i = 0; i < k - 1; i++)
            cout << i + 1 << " ";
        cout << n << endl;
    }
    return 0;
}
