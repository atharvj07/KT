#include <iostream>
#include <algorithm>

int main() {
    int n, s;
    std::cin >> n >> s;
    long long arr[n];
    for (int i = 0; i < n; ++i) {
        std::cin >> arr[i];
    }
    std::sort(arr, arr + n);
    long long ans = 0, sum = 0;
    for (int i = 0; i < n; ++i) {
        sum += arr[i];
        if (sum >= s) {
            ans = arr[i];
            break;
        }
    }
    if (sum < s) {
        std::cout << -1 << std::endl;
    } else {
        std::cout << ans << std::endl;
    }
    return 0;
}
