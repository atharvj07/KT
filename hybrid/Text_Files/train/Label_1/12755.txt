#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<long long> f(n + 1), g(n + 1);
  for (int i = 0; i < n; i++) {
    g[i + 1] = g[i];
    if (s[i] == '=') {
      continue;
    }
    if (s[i] == '^') {
      f[i + 1] = f[i];
      continue;
    }
    int j = i;
    while (j < n && s[j] != '=' && s[j] != '^') {
      j++;
    }
    long long w = 0;
    for (int k = i; k < j; k++) {
      w *= 10;
      w += s[k] - '0';
    }
    if (s[i] == '^') {
      f[i + 1] = f[i] + w;
    } else {
      g[i + 1] = g[i] + w;
    }
    i = j - 1;
  }
  if (f[n] == g[n]) {
    cout << "balance";
  } else if (f[n] > g[n]) {
    cout << "left";
  } else {
    cout << "right";
  }
}
