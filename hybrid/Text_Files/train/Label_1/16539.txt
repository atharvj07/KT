#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define int ll

const int N = 2e5 + 5;

int n, a[N];
set<int> pref[N], suff[N];

void solve() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];

  for (int i = 1; i <= n; i++) {
    pref[i] = pref[i - 1];
    pref[i].insert(a[i]);
  }
  for (int i = n; i >= 1; i--) {
    suff[i] = suff[i + 1];
    suff[i].insert(a[i]);
  }

  int l = 1, r = n;
  while (l < r && pref[l] == suff[r] && pref[l].size() == 1) l++, r--;
  int min_ans = pref[l].size();

  l = 1, r = n;
  while (l < r && (pref[l] == suff[r] || pref[l].size() > 1 || suff[r].size() > 1)) l++, r--;
  int max_ans = min(pref[l].size(), suff[r].size()) + (l != r);

  cout << min_ans << " " << max_ans << '\n';
}
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  solve();
}
