#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MaxN = 2005;
ll x[MaxN], y[MaxN];

inline ll cross(ll x0, ll y0, ll x1, ll y1, ll x2, ll y2) {
    return (x1 - x0) * (y2 - y0) - (x2 - x0) * (y1 - y0);
}

ll solve(int n) {
    unordered_map<ll, ll> m;
    for (int i = 1; i <= n; ++i) {
        for (int j = i + 1; j <= n; ++j) {
            ll tmp = x[j] - x[i];
            if (tmp < 0) tmp = -tmp;
            if (tmp > 1e9) tmp = 1e9;
            ++m[tmp];
        }
    }
    ll ans = 0;
    for (auto it = m.begin(); it != m.end(); ++it) {
        ll tmp = it->second;
        ans += tmp * (tmp - 1) / 2;
    }
    return ans;
}

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> x[i] >> y[i];
    cout << solve(n);
    return 0;
}
