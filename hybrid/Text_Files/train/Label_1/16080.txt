#include <iostream>
#include <vector>
using namespace std;
int main() {
  long long n, L, a, ans = 0, t, l;
  vector<pair<long long, long long>> v;
  cin >> n >> L >> a;
  while (n--) {
    cin >> t >> l;
    v.push_back({t, l});
  }
  for (long long i = 0, st = 0; i < v.size(); i++) {
    if (v[i].first >= st) {
      ans++;
      st = v[i].first + v[i].second;
    } else
      st = min(st + a, v[i].first + v[i].second);
  }
  cout << ans;
}
