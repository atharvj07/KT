from sys import stdin,stdout
a,b,c,d=map(int,stdin.readline().split())
cost1=[int(x) for x in stdin.readline().split()]
cost2=[int(x) for x in stdin.readline().split()]
cost3=[int(x) for x in stdin.readline().split()]
cost4=[int(x) for x in stdin.readline().split()]
m=int(stdin.readline())
block1=[list(map(int,stdin.readline().split())) for x in range(m)]
m=int(stdin.readline())
block2=[list(map(int,stdin.readline().split())) for x in range(m)]
m=int(stdin.readline())
block3=[list(map(int,stdin.readline().split())) for x in range(m)]
ans=1e30
for x in range(a):
    if x+1 in [y for y,z in block1 if z==1]:continue
    for y in range(b):
        if x+1 in [p for p,q in block1 if q==y+1]:continue
        if y+1 in [p for p,q in block2 if q==1]:continue
        for z in range(c):
            if y+1 in [p for p,q in block2 if q==z+1]:continue
            if z+1 in [p for p,q in block3 if q==1]:continue
            for w in range(d):
                if z+1 in [p for p,q in block3 if q==w+1]:continue
                ans=min(ans,cost1[x]+cost2[y]+cost3[z]+cost4[w])
if ans==1e30:print(-1)
else:print(int(ans))
