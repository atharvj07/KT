from sys import stdin

t = int(stdin.readline())
for _ in range(t):
    n, l, r = map(int, stdin.readline().split())
    c = [int(x) for x in stdin.readline().split()]
    assert(0 <= l <= n)
    assert(0 <= r <= n)
    assert(l + r == n)
    assert(len(c) == n)
    for c_i in c:
        assert(1 <= c_i <= n)
    ls = sorted([c_i for c_i in c if c_i in c[l:]])
    rs = sorted([c_i for c_i in c if c_i in c[:l]], reverse=True)
    cost = 0
    while ls and rs:
        if ls[-1] == rs[-1]:
            ls.pop()
            rs.pop()
            continue
        cost += 1
        ls[-1] = rs[-1]
    cost += len(ls) + len(rs)
    print(cost)
