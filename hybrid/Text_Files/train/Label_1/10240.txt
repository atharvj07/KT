#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = -1;
  bool del = false;
  for (int i = 0; i < n; i++) {
    if (i + 1 < n && s[i] > s[i + 1] && !del) {
      if (ans == -1 || s[i - 1] > s[ans]) {
        ans = i;
      }
      del = true;
    }
  }
  if (ans == -1) {
    cout << s;
  } else {
    s.erase(ans, 1);
    cout << s;
  }
  return 0;
}
