t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    good_subarrays = n + (n - 1)  # Subarrays of length 1 and 2 are always good
    for i in range(n):
        for j in range(i + 2, n):
            if abs(a[i] - a[j]) < (j - i):
                good_subarrays += 1
            else:
                break
    print(good_subarrays)
