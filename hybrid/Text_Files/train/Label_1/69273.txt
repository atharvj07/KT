n, r = map(int, input().split())
projects = []
for _ in range(n):
    a, b = map(int, input().split())
    projects.append((a, b))

projects.sort(key=lambda x: (max(x[0], x[0] + x[1]), x[1]))

for a, b in projects:
    if r < a:
        print("NO")
        break
    else:
        r += b
        if r < 0:
            print("NO")
            break
else:
    print("YES")
