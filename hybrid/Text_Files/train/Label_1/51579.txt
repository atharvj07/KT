import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[][] a = new long[n][2], b = new long[m][3];
        for (int i = 0; i < n; i++) {
            a[i][0] = sc.nextLong();
            a[i][1] = sc.nextLong();
        }
        for (int i = 0; i < m; i++) {
            b[i][0] = sc.nextLong();
            b[i][1] = sc.nextLong();
            b[i][2] = sc.nextLong();
        }
        long max = 0, x = 0, y = 0, z = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                long l = Math.max(a[i][0], b[j][0]), r = Math.min(a[i][1], b[j][1]);
                if (l <= r) {
                    long cur = (r - l) * b[j][2];
                    if (cur > max) {
                        max = cur;
                        x = i + 1;
                        y = j + 1;
                        z = l;
                    }
                }
            }
        }
        System.out.println(max);
        if (max > 0) {
            System.out.println(x + " " + y + " " + z);
        }
    }
}
