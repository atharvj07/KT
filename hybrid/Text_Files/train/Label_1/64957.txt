import sys
input = sys.stdin.readline

n, m = map(int, input().strip().split())
a = [[x for x in input().strip()] for _ in range(n)]
vis = [[0] * m for _ in range(n)]

for i in range(n):
    for j in range(m):
        if a[i][j] != 'G':
            continue
        for k in range(j, m):
            if a[i][k] == 'G':
                break
            elif a[i][k] == 'S':
                vis[i][j] = 1
                break

ans = -1
for i in range(n):
    if sum(vis[i]) == 0:
        continue
    cnt = 0
    for j in range(m):
        if vis[i][j] == 1:
            cnt += 1
        elif vis[i][j] == 0 and cnt > 0:
            ans = max(ans, cnt)
            cnt = 0
    if cnt > 0:
        ans = max(ans, cnt)

print(ans)
