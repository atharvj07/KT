#include <bits/stdc++.h>
using namespace std;
#define int long long
int n, a[5010], pre[5010], suf[5010], mx;
vector<int> ans;
main(void) {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    pre[i] = a[i];
    if (i) pre[i] += pre[i - 1];
    mx = max(mx, a[i]);
  }
  for (int i = n - 1; i >= 0; i--) {
    suf[i] = a[i];
    if (i + 1 < n) suf[i] += suf[i + 1];
  }
  for (int i = 0; i <= n / 2; i++) {
    int res = pre[n - 1] + a[n - 1] * ((mx - a[n - 1]) * (i - 1) - (mx - a[n - 1]) * i);
    int now = suf[i + 1] + a[i + 1] * ((mx - a[i + 1]) * ((i - 1) * (i)) - (mx - a[i + 1]) * (i * (i + 1)));
    if (i) res += pre[i - 1] + a[i - 1] * ((mx - a[i - 1]) * (i - 1) - (mx - a[i - 1]) * i);
    ans.push_back(min(res, now));
  }
  for (auto i : ans) cout << i << " ";
}
