#include <vector>
#include <iostream>
#include <queue>
#include <algorithm>
 
int main() {
    int n, m, D;
    std::cin >> n >> m >> D;
    std::vector<int> deg(n + 1, 0);
    std::vector<std::vector<int>> g(n + 1);
    int tt = 0;
    for (int i = 0; i < m; ++i) {
        int u, v;
        std::cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
        tt += (u == 1 && deg[v] == 0);
        tt += (v == 1 && deg[u] == 0);
        deg[u]++;
        deg[v]++;
    }
    if (D == tt) {
        std::cout << "YES\n";
        std::queue<int> q;
        q.push(1);
        deg[1] = 0;
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            for (const auto &u : g[v]) {
                if (deg[u]) {
                    q.push(u);
                    deg[u]--;
                    std::cout << v << " " << u << "\n";
                }
            }
        }
    } else {
        std::cout << "NO\n";
    }
    return 0;
}
