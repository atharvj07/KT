import math

n = int(input())

def f(x):
    return x * x + x * sum(int(c) for c in str(x)) - n

def binary_search(l, r):
    while l <= r:
        mid = (l + r) // 2
        if f(mid) == 0:
            return mid
        elif f(mid) > 0:
            r = mid - 1
        else:
            l = mid + 1
    return -1

ans = binary_search(1, int(math.sqrt(n)))
print(ans)
