#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, p, k;
    cin >> n >> p >> k;
    assert(k == 2);
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end(), greater<>());
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (i < n - 1) {
        if (a[i] <= p and a[i + 1] <= p) {
          p -= a[i] + a[i + 1];
          ans += 2;
          i++;
        } else if (a[i] <= p) {
          p -= a[i];
          ans++;
        }
      } else {
        if (a[i] <= p) {
          p -= a[i];
          ans++;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
