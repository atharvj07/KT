import java.util.*;
import java.io.*;

public class Main {
  static int N;
  static int[] bit;

  public static void main(String[] args) {
    try {
      Scanner sc = new Scanner(System.in);
      N = sc.nextInt();
      int[] A = new int[N];
      for (int i = 0; i < N; i++) {
        A[i] = sc.nextInt();
      }
      bit = new int[N + 1];
      long ans = 0;
      for (int i = 0; i < N; i++) {
        ans += A[i] * (sum(N) - sum(A[i] - 1));
        update(A[i], 1);
      }
      System.out.println(ans % 1000000007);
    } catch (Exception e) {
    }
  }

  static void update(int x, int val) {
    while (x <= N) {
      bit[x] += val;
      x += (x & (-x));
    }
  }

  static long sum(int x) {
    long ans = 0;
    while (x > 0) {
      ans += bit[x];
      x -= (x & (-x));
    }
    return ans;
  }
}
