#include <iostream>
#include <set>
using namespace std;

int main() {
  int n;
  cin >> n;
  set<int> xs, ys;
  for (int i = 0; i < 4 * n + 1; i++) {
    int x, y;
    cin >> x >> y;
    xs.insert(x), ys.insert(y);
  }
  for (int x: xs) {
    if (xs.count(x - 1) || xs.count(x + 1)) continue;
    for (int y: ys) {
      if (ys.count(y - 1) || ys.count(y + 1)) continue;
      cout << x << ' ' << y << endl;
      return 0;
    }
  }
  return 0;
}
