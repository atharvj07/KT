import sys
input = sys.stdin.readline

n = int(input())
g = [[] for i in range(n)]
for i in range(n - 1):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    g[u].append(v)
    g[v].append(u)

cnt = [0] * n
vis = [False] * n

def dfs(u, p):
    vis[u] = True
    cnt[u] = 1
    for v in g[u]:
        if not vis[v]:
            dfs(v, u)
            cnt[u] += cnt[v]

dfs(0, -1)

ans = 0
for i in range(n):
    if cnt[i] % 2:
        ans += 1

if ans % 2:
    print(-1)
else:
    print(ans // 2)
