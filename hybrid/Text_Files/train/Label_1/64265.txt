MOD = 10**9 + 7

def solve(m, d, a, b):
    def count_magic_numbers(x):
        n = len(x)
        dp = [[[0 for _ in range(m)] for __ in range(2)] for ___ in range(n+1)]
        dp[0][0][0] = 1
        
        for i in range(n):
            for j in range(m):
                for k in range(2):
                    for digit in range(10):
                        if digit == d and i % 2 == 0:
                            continue
                        if digit != d and i % 2 == 1:
                            continue
                        ni, nj, nk = i + 1, (j * 10 + digit) % m, k
                        if k == 0:
                            if digit > int(x[i]):
                                break
                            if digit < int(x[i]):
                                nk = 1
                        dp[ni][nk][nj] += dp[i][k][j]
                        dp[ni][nk][nj] %= MOD
        return (dp[n][0][0] + dp[n][1][0] - 1) % MOD

    a = str(int(a) - 1)
    b = str(b)
    return (count_magic_numbers(b) - count_magic_numbers(a)) % MOD

m, d = map(int, input().split())
a = input().strip()
b = input().strip()
print(solve(m, d, a, b))
