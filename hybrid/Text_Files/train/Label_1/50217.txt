import java.util.*;
public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int t = sc.nextInt();
    int k = sc.nextInt();
    int[] a = new int[t];
    for (int i = 0; i < t; i++) {
      a[i] = sc.nextInt();
    }
    int[] d = new int[n + 1];
    for (int i = 0; i < t; i++) {
      for (int j = 0; j < a[i]; j++) {
        d[n - a[i] + j] = t - i;
      }
    }
    if (d[2] < k) {
      System.out.println(-1);
      return;
    }
    d[2] -= k;
    d[1] += k;
    d[2]++;
    System.out.println(n);
    for (int i = 2; i <= n; i++) {
      for (int j = 1; j < d[i]; j++) {
        System.out.println(j + " " + (j + 1));
      }
    }
  }
}
