#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef unsigned long long ll;

ll gcd(ll a, ll b) {
  if (b == 0)
    return a;
  return gcd(b, a % b);
}

int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> a(k);
  for (int i = 0; i < k; i++) { cin >> a[i]; }

  ll g = 0;
  for (int i = 0; i < k; i++) { g = gcd(g, a[i]); }

  ll ans = n / g;
  ll type = -1;
  for (int i = 0; i < k; i++) {
    if (a[i] % g == 0 && n % a[i] == 0) {
      if (type == -1 || a[i] < a[type]) { type = i; }
    }
  }

  cout << type + 1 << ' ' << ans / a[type] << endl;
  return 0;
}
