#include <bits/stdc++.h>
using namespace std;

int main() {
  int hy, ay, dy, hm, am, dm, h, a, d, ans = 0;
  cin >> hy >> ay >> dy >> hm >> am >> dm >> h >> a >> d;
  if (ay > dm) {
    if (hy > 0) ans = 0;
    else {
      if (h <= a) ans += hy * h;
      else {
        ans += hm * h;
        hy -= hm;
        if (ay - dm > 0) ans += (hy + 1) / (ay - dm) * a;
        else {
          ans += (hy + 1) * h / a;
          hm -= (hy + 1) / (ay - dm);
          if (am > dy) ans += hm * a;
          else ans += hm * d;
        }
      }
    }
  } else {
    if (hy - am > 0) ans += hy * h;
    else {
      if (h <= a) ans += (hy + 1) * h;
      else {
        ans += (hy + 1) * d;
        hm -= (hy + 1) / (ay - dm);
        if (hm > 0) ans += hm * h;
        else {
          ans += (hm + 1) * d / h;
          dm -= (hm + 1) / (am - dy);
          if (ay > dm) ans += hm * h;
          else ans += hm * a;
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
