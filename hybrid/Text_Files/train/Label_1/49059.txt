import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int f = scanner.nextInt();
        int t = scanner.nextInt();
        int t0 = scanner.nextInt();
        int a1 = scanner.nextInt();
        int t1 = scanner.nextInt();
        int p1 = scanner.nextInt();
        int a2 = scanner.nextInt();
        int t2 = scanner.nextInt();
        int p2 = scanner.nextInt();
        int result = -1;
        for (int i = 0; i <= f; i += a1) {
            int time = i / t0 + (f - i) / t1;
            if (time <= t) {
                result = Math.min(result == -1 ? Integer.MAX_VALUE : result, i / a1 * p1 + (f - i) / t1 * p1);
            }
        }
        for (int i = 0; i <= f; i += a2) {
            int time = i / t0 + (f - i) / t2;
            if (time <= t) {
                result = Math.min(result == -1 ? Integer.MAX_VALUE : result, i / a2 * p2 + (f - i) / t2 * p2);
            }
        }
        System.out.println(result);
    }
}
