#include <iostream>
#include <cmath>

using namespace std;

int main() {
  int h1, a1, c1;
  int h2, a2;
  cin >> h1 >> a1 >> c1;
  cin >> h2 >> a2;
  int t = ceil(double(h2) / a1);
  int h = h1;
  if (h - t * a2 > 0)
    cout << t << endl;
  else
    cout << t + 1 << endl;
  for (int i = 0; i < t; i++) {
    if (h > c1 - a2) {
      cout << "STRIKE" << endl;
      h -= a2;
    } else {
      cout << "HEAL" << endl;
      h += c1 - a2;
    }
  }
  if (h > 0) cout << "STRIKE" << endl;
  return 0;
}
