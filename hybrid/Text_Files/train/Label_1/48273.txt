import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;

public class Main {
    static class Subject {
        int complexity;
        int minExercises;
        int maxExercises;

        Subject(int complexity, int minExercises, int maxExercises) {
            this.complexity = complexity;
            this.minExercises = minExercises;
            this.maxExercises = maxExercises;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        int k = Integer.parseInt(line[2]);

        Subject[] subjects = new Subject[m];
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int minExercises = Integer.parseInt(line[0]);
            int maxExercises = Integer.parseInt(line[1]);
            int complexity = Integer.parseInt(line[2]);
            subjects[i] = new Subject(complexity, minExercises, maxExercises);
        }

        // Sort subjects by complexity
        Arrays.sort(subjects, Comparator.comparingInt(s -> s.complexity));

        long totalExercises = 0;
        long[][] dp = new long[n + 1][m + 1];

        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                dp[i][j] = Long.MIN_VALUE;
            }
        }

        // Base case: no subjects
        dp[0][0] = 0;

        // Fill in the DP table
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                Subject subject = subjects[j - 1];

                // Try both options: k times more exercises and k more exercises
                dp[i][j] = Math.max(
                        dp[i - 1][j - 1] + subject.maxExercises,
                        dp[i - 1][j - 1] + k * subject.minExercises
                );

                // Update total exercises if a valid solution is found
                if (dp[i][j] >= subject.minExercises && dp[i][j] <= subject.maxExercises) {
                    totalExercises = Math.max(totalExercises, dp[i][j]);
                }
            }
        }

        // Check if a valid solution exists
        if (totalExercises == 0) {
            System.out.println("NO");
            return;
        }

        // Print the timetable
        System.out.println("YES");
        int day = n;
        int subjectIndex = m;

        while (day > 0 && subjectIndex > 0) {
            Subject subject = subjects[subjectIndex - 1];

            // Determine the number of exercises based on the DP solution
            long exercises = Math.max(
                    dp[day - 1][subjectIndex - 1] + subject.maxExercises,
                    dp[day - 1][subjectIndex - 1] + k * subject.minExercises
            );

            System.out.println(subject.complexity + " " + exercises);

            // Update day and subject index
            day--;
            subjectIndex--;
        }
    }
}
