#include<bits/stdc++.h>
using namespace std;
#define endl "\n"

struct Edge{
    int a, b, w;
    Edge(int a, int b, int w) {
        this->a = a;
        this->b = b;
        this->w = w;
    }
    bool operator< (Edge& o) {
        return w < o.w;
    }
};
vector<Edge> edges;
vector<int> par;

int getPar(int n) {
    if(par[n] == n)    return n;
    return par[n] = getPar(par[n]);
}

void merge(int a, int b) {
    a = getPar(a);
    b = getPar(b);
    if(a != b)    par[b] = a;
    return;
}

void init(int n) {
    par.resize(n);
    for(int i=0; i<n; i++)    par[i] = i;
    return;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;  cin >> n;
    int deg[n], sum[n];

    for(int i=0; i<n; i++)    cin >> deg[i] >> sum[i];
    for(int i=0; i<n; i++) {
        for(int j=i+1; j<n; j++) {
            Edge ed(i, j, sum[i] ^ sum[j]);
            edges.push_back(ed);
        }
    }

    sort(edges.begin(), edges.end());
    init(n);
    int cnt = 0;

    for(auto x : edges) {
        if(getPar(x.a) != getPar(x.b)) {
            merge(x.a, x.b);
            cnt++;
            cout << x.a << " " << x.b << endl;
        }
    }

    cout << cnt << endl;

    return 0;
}
