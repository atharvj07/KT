#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  long long ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      for (int l = j + 1; l < n; l++) {
        if (a[j] == 1LL * a[i] * k && a[l] == 1LL * a[j] * k) {
          ans++;
        }
      }
    }
  }

  cout << ans << endl;
  return 0;
}
