#include<bits/stdc++.h>
using namespace std;
int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        unordered_map<int, int> freq;
        int mx = 0, same = 0;
        for (int i = 0, x; i < n; i++) {
            cin >> x;
            freq[x]++;
        }
        for (auto& it : freq) mx = max(mx, it.second);
        for (auto& it : freq) if (it.second == mx) same++;
        int ans = n - mx * same;
        if (ans % 2 == 0 or !same) cout << ans << "\n";
        else cout << ans + 1 << "\n";
    }
}
