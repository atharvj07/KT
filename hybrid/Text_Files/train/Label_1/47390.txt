import java.util.*;

public class Main {
    private static final Scanner scanner = new Scanner(System.in);

    public static void main(String[] args) {
        while (true) {
            String correct = scanner.nextLine();
            if (correct.equals(".")) {
                break;
            }
            while (true) {
                String answer = scanner.nextLine();
                if (answer.equals(".")) {
                    break;
                } else {
                    System.out.println(correct.equals(answer) ? "yes" : "no");
                }
            }
            System.out.println(".");
        }
    }
}
