#include <iostream>
#include <vector>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  bool lucky = false;
  for (int i = 1; i < n; i++) {
    int sum1 = 0, sum2 = 0;
    for (int j = 0; j < i; j++) { sum1 += a[j]; }
    for (int j = i; j < n; j++) { sum2 += a[j]; }
    if (sum1 == sum2) { lucky = true; }
  }
  cout << (lucky ? "YES" : "NO") << endl;
  return 0;
}
