import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        String s = in.next();

        int[] dp = new int[n + 2];
        int[] lastA = new int[n + 2];
        int[] lastB = new int[n + 2];
        dp[0] = dp[1] = 1;
        if (s.charAt(0) == 'A') {
            lastA[1] = 0;
            lastB[1] = -1;
        } else {
            lastA[1] = -1;
            lastB[1] = 0;
        }

        for (int i = 2; i <= n; i++) {
            if (s.charAt(i - 1) == 'A') {
                if (lastA[i - 1] != -1) {
                    dp[i] = dp[lastA[i - 1]] + dp[i - 1];
                } else {
                    dp[i] = dp[i - 1];
                }
                lastA[i] = i - 1;
                lastB[i] = lastB[i - 1];
            } else {
                if (lastB[i - 1] != -1) {
                    dp[i] = dp[lastB[i - 1]] + dp[i - 1];
                } else {
                    dp[i] = dp[i - 1];
                }
                lastA[i] = lastA[i - 1];
                lastB[i] = i - 1;
            }
        }

        System.out.println(dp[n]);
    }
}
