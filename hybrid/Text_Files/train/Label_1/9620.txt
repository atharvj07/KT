#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, s;
    cin >> n >> s;
    vector<pair<int, int>> arr(n);
    for (int i = 0; i < n; i++)
        cin >> arr[i].first >> arr[i].second;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (i > 0)
            ans = max(ans, arr[i].second);
        ans += arr[i].first - (i == 0 ? s : arr[i - 1].first);
    }
    cout << ans + 1 << endl;
}
