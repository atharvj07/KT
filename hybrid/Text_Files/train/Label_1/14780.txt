#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>

using namespace std;

string s, t;
vector<pair<char, char>> ans;
unordered_map<char, char> mp;

int main() {
    cin >> s >> t;
    for (int i = 0; i < s.length(); i++) {
        if (s[i] != t[i]) {
            if (mp.count(s[i])) {
                if (mp[s[i]] != t[i]) {
                    cout << -1 << endl;
                    return 0;
                }
            } else {
                mp[s[i]] = t[i];
                mp[t[i]] = s[i];
                ans.push_back({s[i], t[i]});
            }
        }
    }
    cout << ans.size() << endl;
    for (auto p : ans) {
        cout << p.first << ' ' << p.second << endl;
    }
    return 0;
}
