import java.util.Scanner;

public class VovaAndNumber {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int count = 0;
        for (int x = 1; x <= n; x++) {
            int sum = 0;
            int temp = x;
            while (temp > 0) {
                sum += temp % 10;
                temp /= 10;
            }
            if (x + sum == n) {
                count++;
                System.out.println(x);
            }
        }

        if (count == 0) {
            System.out.println(0);
        }
    }
}
