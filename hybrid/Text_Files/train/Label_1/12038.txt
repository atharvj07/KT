#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  if (n == 2) {
    if (a[1] == 1) {
      cout << "perfect" << endl;
    } else {
      cout << "ambiguous" << endl;
      cout << "0 1" << endl;
      cout << "0 1 2" << endl;
    }
  } else {
    int l = n - 2, r = n - 1;
    while (l >= 0 && r >= 0 && a[l] >= 2) {
      a[l] -= 2;
      a[r] -= 2;
      l--;
      r--;
    }
    if (l >= 0 || r >= 0) {
      cout << "ambiguous" << endl;
      cout << "0 1";
      for (int i = 1; i < n - 1; i++) {
        cout << " " << i + 1;
      }
      cout << endl;
      cout << "0";
      for (int i = 1; i < n; i++) {
        cout << " " << i;
      }
      cout << endl;
    } else {
      cout << "perfect" << endl;
    }
  }
  return 0;
}
