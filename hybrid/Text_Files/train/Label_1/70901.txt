n, k = map(int, input().split())
tables = list(map(int, input().split()))
formulas = list(map(int, input().split()))

total_pages = sum(tables) + sum(formulas)
tables_possible = (sum(tables) + k - 1) // k
formulas_possible = (sum(formulas) + k - 1) // k

if tables_possible + formulas_possible <= total_pages // k:
    print("YES")
else:
    print("NO")
