#include <bits/stdc++.h>

using namespace std;

const int mod = 998244353;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt[26] = {};
  for (char c : s) cnt[c - 'a']++;
  int mx = 0, sum = 0;
  for (int i = 0; i < 26; i++) {
    mx = max(mx, cnt[i]);
    sum += cnt[i] - cnt[i] % 2;
  }
  long long ans = 1ll * mx * sum % mod;
  cout << ans << '\n';
  return 0;
}
