n, B = map(int, input().split())
a = list(map(int, input().split()))

cuts = []
even, odd = 0, 0

for i in range(n - 1):
    if a[i] % 2 == 0:
        even += 1
    else:
        odd += 1
    if even == odd:
        cuts.append(abs(a[i] - a[i + 1]))

cuts.sort()
cuts_made = 0

for cost in cuts:
    if B >= cost:
        B -= cost
        cuts_made += 1
    else:
        break

print(cuts_made)
