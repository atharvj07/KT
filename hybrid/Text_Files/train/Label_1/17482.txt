#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  multiset<int> s;
  for (int i = 0, a; i < n; ++i) {
    cin >> a;
    s.insert(a);
  }
  while (k--) {
    int x = *s.begin();
    s.erase(s.begin());
    s.emplace(x + 1);
  }
  cout << *s.rbegin() << endl;
  return 0;
}
