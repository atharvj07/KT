#include <bits/stdc++.h>
#include <vector>

using namespace std;

int main() {
  int n, m;
  string s, t;
  cin >> n >> s >> m;
  vector<string> v(m);
  for (int i = 0; i < m; i++) cin >> v[i];
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = 0; j < m; j++) cnt += (s[i] == '*' || s[i] == v[j][i]);
    if (cnt == 0) {
      cout << m;
      return 0;
    }
  }
  vector<int> f(26, 0);
  sort(v.begin(), v.end());
  for (int i = 0; i < m; i++) {
    t += v[i][0];
    for (int j = 0; j < n; j++) f[v[i][j] - 'a']++;
  }
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    if (s[i] != '*') {
      for (int j = 0; j < m; j++) cnt += (s[i] == v[j][i]);
      if (cnt == 0) {
        cout << m;
        return 0;
      }
    } else {
      for (int j = 0; j < 26; j++) {
        if (f[j] != m) {
          for (int k = 0; k < m; k++) {
            if (v[k][i] == (char)('a' + j)) {
              cout << m - f[j];
              return 0;
            }
          }
        }
      }
    }
  }
  cout << 0;
  return 0;
}
