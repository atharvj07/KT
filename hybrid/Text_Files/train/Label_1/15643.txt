#include <iostream>
#include <string>
#include <map>
using namespace std;
int main() {
    int k;
    cin >> k;
    for (int i = 0; i < k; i++) {
        int n;
        cin >> n;
        string s, t;
        cin >> s >> t;
        map<char, int> m;
        for (char x : s) {
            m[x]++;
        }
        bool can = true;
        for (char x : t) {
            if (m[x] > 0) {
                m[x]--;
            } else {
                can = false;
                break;
            }
        }
        cout << (can ? "Yes" : "No") << endl;
    }
    return 0;
}
