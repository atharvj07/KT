from sys import stdin

n, m, k = map(int, stdin.readline().split())
c = [int(x) for x in stdin.readline().split()]
p = [[int(x) for x in stdin.readline().split()] for i in range(n)]

dp = [[[-1] * (k + 1) for j in range(m + 1)] for i in range(n + 1)]
for i in range(n):
    for j in range(m + 1):
        for x in range(k + 1):
            if i == 0:
                if j == 0:
                    dp[i][j][x] = 0
            elif c[i - 1] == 0 or c[i - 1] == j:
                if x == 0:
                    dp[i][j][x] = dp[i - 1][j][x]
                else:
                    dp[i][j][x] = min(dp[i - 1][j][x], dp[i - 1][j][x - 1] + p[i - 1][j - 1])
            else:
                dp[i][j][x] = dp[i - 1][j][x]
print(dp[n][m][k] if dp[n][m][k] != -1 else -1)
