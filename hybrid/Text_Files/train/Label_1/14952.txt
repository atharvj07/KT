#include <bits/stdc++.h>
#define int long long

using namespace std;

const int mxN = 2e5 + 5;

int n, a[mxN], sum[mxN], par[mxN], sz[mxN], ans;
vector<int> adj[mxN];

int find(int x) { return x == par[x] ? x : par[x] = find(par[x]); }

void unite(int x, int y) {
    x = find(x);
    y = find(y);
    if (x == y) return;
    if (sz[x] > sz[y]) {
        sz[x] += sz[y];
        par[y] = x;
    } else {
        sz[y] += sz[x];
        par[x] = y;
    }
}

void dfs(int node, int p) {
    sum[node] = a[node];
    for (int next : adj[node]) {
        if (next == p) continue;
        dfs(next, node);
        sum[node] += sum[next];
        unite(node, next);
    }
    ans = max(ans, sum[find(node)]);
}

int32_t main() {
    cin >> n;
    for (int i = 2; i <= n; i++) {
        int x;
        cin >> x;
        adj[x].push_back(i);
        adj[i].push_back(x);
    }
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        par[i] = i;
        sz[i] = 1;
    }
    dfs(1, 1);
    cout << ans << endl;
}
