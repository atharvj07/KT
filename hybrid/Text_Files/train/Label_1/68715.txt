from typing import List, Tuple

def main():
    n = int(input())
    points: List[Tuple[int, int]] = []
    for _ in range(n):
        points.append(tuple(map(int, input().split())))
    w = list(map(int, input().split()))

    points.sort(key=lambda p: (p[0], p[1]))

    for i in range(n):
        if points[i][0] > w[i] or points[i][1] - points[i][0] > w[i]:
            print("NO")
            return

    print("YES")
    for i in range(n):
        print(*points[i])

main()
