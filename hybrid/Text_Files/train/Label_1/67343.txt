import sys

n, m = map(int, input().split())
edges = [[] for _ in range(n+1)]
for _ in range(m):
    u, v = map(int, input().split())
    edges[u].append(v)
    edges[v].append(u)

if any(len(edges[i]) >= n for i in range(1, n+1)):
    print(-1)
    sys.exit()

color = [0] * (n + 1)
for u in range(1, n+1):
    if color[u] == 0:
        q = [(u, 0)]
        while q:
            u, c = q.pop()
            if color[u] == 0:
                color[u] = c ^ 1
            for v in edges[u]:
                if color[v] == color[u]:
                    print(-1)
                    sys.exit()
                if color[v] == 0:
                    q.append((v, color[u]))

a = []
b = []
for u in range(1, n+1):
    if color[u] == 1:
        a.append(u)
    else:
        b.append(u)

print(len(a))
print(*a)
print(len(b))
print(*b)
