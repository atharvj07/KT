import java.util.*;
import java.lang.*;

class Main {
  static boolean is_valid(char[] s, int n, int p) {
    if (s[0] != s[n - 1]) return false;
    int i = 0, j = n - 1;
    while (i < j) {
      if (s[i] != s[j]) return false;
      i += 1;
      j -= 1;
    }
    return true;
  }

  static void next_palindrome(char[] s, int n, int p) {
    if (n == 1) {
      System.out.println("NO");
      return;
    }
    boolean f = false;
    for (int i = n - 2; i >= 0; i--) {
      if (s[i] != s[i + 1]) {
        f = true;
        char t = s[i + 1];
        s[i + 1] = s[i];
        s[i] = t;
        break;
      }
    }
    if (!f) {
      System.out.println("NO");
      return;
    }
    if (is_valid(s, n, p)) {
      for (char c : s) System.out.print(c);
      return;
    }
    char c = s[n - 2];
    s[n - 2] = s[n - 1];
    s[n - 1] = 'a';
    for (int i = n - 3; i >= 0; i--) {
      if (s[i] != s[i + 1]) break;
      s[i] -= 1;
      if (s[i] < 'a') {
        s[i] = 'z';
        char t = s[i + 1];
        s[i + 1] = s[i];
        s[i] = t;
      }
    }
    if (is_valid(s, n, p)) {
      for (char d : s) System.out.print(d);
      return;
    }
    int pos = -1;
    for (int i = n - 2; i >= 0; i--) {
      if (s[i] != s[i + 1]) {
        pos = i;
        break;
      }
    }
    c -= 1;
    if (c < 'a') c = 'z';
    s[pos + 1] = c;
    for (char d : s) System.out.print(d);
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int p = in.nextInt();
    char[] s = in.next().toCharArray();
    next_palindrome(s, n, p);
  }
}
