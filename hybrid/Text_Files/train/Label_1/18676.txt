#include <cmath>
#include <iostream>
using namespace std;
bool is_prime(long long n) {
    if (n <= 1) {
        return false;
    }
    if (n % 2 == 0) {
        return (n == 2);
    }
    long long i;
    for (i = 3; i <= sqrt(n); i = i + 2) {
        if (n % i == 0) {
            return false;
        }
    }
    return true;
}
int main() {
    long long n;
    int count = 0;
    cin >> n;
    while (n > 0) {
        long long i;
        for (i = 2; i <= n; i++) {
            if (n % i == 0 && is_prime(i)) {
                n = n - i;
                count = count + 1;
                break;
            }
        }
    }
    cout << count << endl;
    return 0;
}
