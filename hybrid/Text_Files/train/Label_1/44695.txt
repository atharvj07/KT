import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int E = sc.nextInt();
        int T = sc.nextInt();
        int[] x = new int[N];
        for (int i = 0; i < N; i++) {
            x[i] = sc.nextInt();
        }
        sc.close();
        long ans = 0;
        int last = 0;
        for (int i = 0; i < N; i++) {
            ans += x[i] - last;
            ans += T;
            last = x[i];
        }
        ans += E - last;
        System.out.println(ans);
    }
}
