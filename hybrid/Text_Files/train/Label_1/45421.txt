import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        String s = br.readLine();
        int[] lamps = new int[n];
        int[] minOps = new int[n + 1];
        for (int i = 0; i < n; i++) lamps[i] = s.charAt(i) - '0';
        List<int[]> subsets = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            int c = Integer.parseInt(br.readLine());
            int[] subset = new int[c];
            st = new StringTokenizer(br.readLine());
            for (int j = 0; j < c; j++) {
                subset[j] = Integer.parseInt(st.nextToken()) - 1;
            }
            subsets.add(subset);
        }
        Queue<Integer> q = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            if (lamps[i] == 1) q.add(i);
        }
        while (!q.isEmpty()) {
            int idx = q.poll();
            for (int[] subset : subsets) {
                boolean allOn = true;
                for (int j : subset) {
                    if (j < idx) {
                        if (minOps[j] == 0) allOn = false;
                    } else if (lamps[j] == 0) allOn = false;
                }
                if (allOn) {
                    for (int j : subset) {
                        if (j > idx && minOps[j] == 0) {
                            q.add(j);
                            minOps[j] = minOps[idx] + 1;
                        }
                    }
                }
            }
        }
        for (int i = 1; i <= n; i++) System.out.println(minOps[i - 1]);
    }
}
