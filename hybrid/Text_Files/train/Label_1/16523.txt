#include<bits/stdc++.h>
using namespace std;
#define N 4003
int n, m, vis[N], know[N], ans = INT_MAX;
vector<int> adj[N];
bool dfs(int u, int p) {
    vis[u] = 1;
    for (int i = 0; i < (int)adj[u].size(); i++) {
        int v = adj[u][i];
        if (v == p)
            continue;
        if (vis[v] == 1)
            return true;
        if (vis[v] == 0 && dfs(v, u))
            return true;
    }
    vis[u] = 2;
    return false;
}
int main() {
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v, u--, v--;
        adj[u].push_back(v), adj[v].push_back(u);
    }
    for (int i = 0; i < n; i++) {
        if (vis[i] == 0) {
            if (dfs(i, -1)) {
                cout << "-1" << endl;
                return 0;
            }
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < (int)adj[i].size(); j++) {
            int v = adj[i][j];
            if (i < v)
                know[i]++;
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            for (int k = j + 1; k < n; k++) {
                if (know[i] + know[j] + know[k] < ans)
                    ans = know[i] + know[j] + know[k];
            }
        }
    }
    cout << ans << endl;
    return 0;
}
