def solve(n, m, a):
    for i in range(n):
        for j in range(m):
            h = sorted({a[i][k] for k in range(m)} | {a[k][j] for k in range(n)})
            for k in range(m):
                print(h.index(a[i][k]) + 1, end=' ')
            print()

n, m = map(int, input().split())
a = [list(map(int, input().split())) for i in range(n)]
solve(n, m, a)
