import sys

def solve(n, m, k):
    adj = [set() for i in range(n + 1)]
    for i in range(m):
        x, y = map(int, sys.stdin.readline().split())
        adj[x].add(y)
        adj[y].add(x)

    ans = [0 for i in range(m + 1)]
    for day in range(m):
        visited = set()
        for i in range(1, n + 1):
            if i in visited:
                continue
            group = set([i])
            queue = [i]
            while queue:
                u = queue.pop()
                visited.add(u)
                for v in adj[u]:
                    if v not in visited:
                        queue.append(v)
                        group.add(v)
            if len(group) >= k:
                ans[day] = max(ans[day], len(group))
    return ans

n, m, k = map(int, sys.stdin.readline().split())
print('\n'.join(map(str, solve(n, m, k))))
