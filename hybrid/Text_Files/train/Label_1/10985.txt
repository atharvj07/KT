#include <iostream>
#include <vector>
#include <algorithm>
#include <set>
using namespace std;

const int maxn = 1e9;

struct node
{
    int l, r;
};

vector<node>v;

set<int>s;

bool cmp(node &a, node &b)
{
    return a.r < b.r;
}

int find(int k)
{
    set<int>::iterator it = s.lower_bound(k);
    if(it == s.end()) return 0;
    return *it;
}

int main()
{
    int n;
    cin >> n;
    for(int i=0;i<n;i++)
    {
        int t;
        cin >> t;
        if(t==1)
        {
            int x, y;
            cin >> x >> y;
            v.push_back({x, y});
            s.insert(x);
            s.insert(y);
        }
        else
        {
            int x, y;
            cin >> x >> y;
            int p = find(v[x-1].l);
            int r = find(v[y-1].r);
            if(p == r && p) cout << "YES" << endl;
            else cout << "NO" << endl;
        }
    }
    return 0;
}
