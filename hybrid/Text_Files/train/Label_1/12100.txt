#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m,p,a[200005],b[200005],x=0,y=0,flag=0;
    cin>>n>>m>>p;
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    for(int i=1;i<=m;i++){
        cin>>b[i];
    }
    for(int i=1;i<n;i++){
        if(a[i]>a[i+1]&&flag==0){
            x=i;
            flag=1;
        }
        if(a[i]>a[i+1]){
            y=i;
        }
    }
    if(flag==0){
        cout<<"0\n";
        return 0;
    }
    else{
        cout<<(y-x)/p+1<<endl;
        for(int i=x;i<=y;i=i+p){
            cout<<(i-x)/p+1<<" ";
        }
        cout<<endl;
    }
}
