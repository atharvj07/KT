#include <bits/stdc++.h>
using namespace std;

int n, h, m;
vector<array<int, 3>> restr;
int memo[51][51][51][11];

int dp(int pos, int lmax, int rmax, int hl) {
  if (pos == n + 1)
    return 0;
  if (memo[pos][lmax][rmax][hl] != -1)
    return memo[pos][lmax][rmax][hl];
  int opt1 = dp(pos + 1, lmax, rmax, hl);
  int opt2 = dp(pos + 1, max(lmax, hl), max(rmax, hl), hl) + hl * hl;
  for (auto &r : restr) {
    int ll = r[0], rr = r[1], rh = r[2];
    if (ll <= pos && pos <= rr) {
      opt2 = max(opt2, dp(pos + 1, max(lmax, rh), max(rmax, rh), rh) + rh * rh);
    }
  }
  memo[pos][lmax][rmax][hl] = max(opt1, opt2);
  return memo[pos][lmax][rmax][hl];
}

int main() {
  memset(memo, -1, sizeof(memo));
  cin >> n >> h >> m;
  int l, r, x;
  while (m--) {
    cin >> l >> r >> x;
    restr.push_back({l, r, x});
  }
  cout << dp(1, 0, 0, 0) << endl;
  return 0;
}
