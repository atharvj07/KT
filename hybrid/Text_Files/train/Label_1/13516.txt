#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> l(m);
  for (int i = 0; i < m; i++) cin >> l[i];
  int j = 0;
  for (int i = 0; i < n; i++) {
    if (j < m && l[j] <= i) {
      j++;
      i--;
    } else
      cout << i + 1 << " ";
  }
  return 0;
}
