import sys

INF = int(1e18)

for n in sys.stdin.read().splitlines()[1:]:
    n = int(n)
    ans = INF
    if n % 3 == 0:
        ans = min(ans, n - n % 3)
        if n % 5 == 0:
            ans = min(ans, n - 5 * (n // 5))
    elif n % 5 == 0:
        ans = min(ans, n - n % 5)
    if n % 2 == 0:
        ans = min(ans, n - n // 2)
        if n % 3 == 0:
            ans = min(ans, n - 3 * (n // 3))
            if n % 5 == 0:
                ans = min(ans, n - 15 * (n // 15))
        if n % 5 == 0:
            ans = min(ans, n - 10 * (n // 10))
    print([-1, ans][ans != INF])
