import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int a = in.nextInt();
    int b = in.nextInt();
    int c = in.nextInt();
    int d = in.nextInt();
    boolean ok = false;
    if (a + b > c && a + c > b && b + c > a) ok = true;
    if (b + c > d && b + d > c && c + d > b) ok = true;
    if (c + d > a && c + a > d && d + a > c) ok = true;
    if (d + a > b && d + b > a && a + b > d) ok = true;
    if (a + b == c) ok = true;
    if (a + c == b) ok = true;
    if (a + d == b) ok = true;
    if (a + d == c) ok = true;
    if (b + c == d) ok = true;
    if (b + d == c) ok = true;
    System.out.println(ok ? "TRIANGLE" : (a + b + c + d == 6 ? "SEGMENT" : "IMPOSSIBLE"));
  }
}
