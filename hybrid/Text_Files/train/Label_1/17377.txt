#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) {
  return b == 0 ? a : gcd(b, a % b);
}
int main() {
  long long x;
  cin >> x;
  long long a = 1, b = x;
  long long g = gcd(a, b);
  while (g != 1) {
    a = (a + 1) / g * g;
    b = x / a;
    g = gcd(a, b);
  }
  cout << a << ' ' << b << endl;
}
