#include<bits/stdc++.h>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    long long n, ans = 0, a[1000005], b[1000005], g;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        b[i] = b[i - 1] + a[i];
    }
    for (int i = 1; i <= n; i++) {
        g = __gcd(g, b[i] - a[i]);
    }
    for (int i = 1; i * i <= g; i++) {
        if (g % i == 0) {
            if (i > 1)
                ans++;
            long long h = g / i;
            if (h != i && h > 1)
                ans++;
        }
    }
    cout << ans << endl;
    for (int i = 1; i * i <= g; i++) {
        if (g % i == 0) {
            if (i > 1)
                cout << i << " ";
            long long h = g / i;
            if (h != i && h > 1)
                cout << h << " ";
        }
    }
    cout << endl;
    return 0;
}
