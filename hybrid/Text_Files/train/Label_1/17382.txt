#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int a[n + 1], b[n + 1], tm[n + 1];
    for (int i = 1; i <= n; i++) { cin >> a[i] >> b[i]; }
    for (int i = 1; i <= n; i++) cin >> tm[i];
    int ans = 0;
    for (int i = 1; i < n; i++) {
      int x = (b[i] - a[i]) / 2;
      int y = max(x, tm[i]);
      ans = max(ans, a[i] + y);
    }
    cout << max(ans, a[n]) << endl;
  }
  return 0;
}
