#include <bits/stdc++.h>
#include <ext/pb_ds/assoc_container.hpp>
#include <ext/pb_ds/tree_policy.hpp>
#define int long long
using namespace std;
using namespace __gnu_pbds;

typedef tree<int, null_type, less<int>, rb_tree_tag, tree_order_statistics_node_update> ordered_set;

struct edge {
    int to, c;
};

const int mxN = 1005;
const int INF = 1e18;
int n, m, s, b, k, h;
vector<int> adj[mxN];
vector<array<int, 3>> ships;
vector<array<int, 2>> bases;
vector<int> dist[mxN];

void dij(int s) {
    dist[s].assign(n + 1, INF);
    dist[s][s] = 0;
    priority_queue<pair<int, int>> pq;
    pq.push({0, s});
    while (!pq.empty()) {
        int d = -pq.top().first;
        int u = pq.top().second;
        pq.pop();
        if (dist[s][u] != d) continue;
        for (int v : adj[u]) {
            int w = 1;
            if (dist[s][v] > d + w) {
                dist[s][v] = d + w;
                pq.push({-dist[s][v], v});
            }
        }
    }
}

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    while (m--) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    cin >> s >> b >> k >> h;
    ships.resize(s);
    bases.resize(b);
    for (auto &ship : ships) cin >> ship[0] >> ship[1] >> ship[2];
    for (auto &base : bases) cin >> base[0] >> base[1];
    for (int i = 1; i <= n; i++) dij(i);
    vector<vector<int>> dp(s + 1, vector<int>(b + 1, INF));
    dp[0][0] = 0;
    for (int i = 1; i <= s; i++) {
        dp[i][0] = dp[i - 1][0];
        for (int j = 1; j <= b; j++) {
            dp[i][j] = min(dp[i - 1][j], dp[i][j - 1] + k);
            for (int base = 1; base <= b; base++) {
                if (ships[i - 1][1] >= bases[base - 1][1] && dist[ships[i - 1][0]][bases[base - 1][0]] <= ships[i - 1][2]) {
                    dp[i][j] = min(dp[i][j], dp[i - 1][j - 1]);
                }
            }
        }
    }
    int ans = dp[s][b];
    if (ans == INF) {
        ans = -1;
    }
    cout << ans << "\n";
    return 0;
}
