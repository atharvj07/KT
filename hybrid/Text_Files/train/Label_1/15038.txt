#include<bits/stdc++.h>
using namespace std;
int main(){
    long long w,m,k,ans=0;
    cin>>w>>m>>k;
    while(w>=m){
        long long tmp=m;
        int cnt=0;
        while(tmp>0){
            tmp/=10;
            cnt++;
        }
        ans++;
        w-=k*cnt;
        m++;
    }
    cout<<ans<<endl;
}
