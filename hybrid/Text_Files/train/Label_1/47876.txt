import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

class Main {
    public static long solve(int[] a, int n) {
        Map<Integer, Integer> sPos = new HashMap<>();
        for (int i = 0; i < n; i++)
            if (a[i] == 4 || a[i] == 7)
                sPos.put(a[i], i);
        int p4 = -1, p7 = -1;
        long ans = n * (long)(n - 1) / 2;
        for (int i = 0; i < n; i++) {
            if (a[i] == 4)
                p4 = i;
            if (a[i] == 7)
                p7 = i;
            if (p4 != -1 && p7 != -1) {
                int l = Math.max(sPos.get(4), sPos.get(7));
                if (l == Math.min(p4, p7))
                    ans -= (long)(n - l) * (n - l - 1) / 2;
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        sc.close();
        System.out.println(solve(a, n));
    }
}
