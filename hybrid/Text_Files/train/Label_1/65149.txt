field = [input() for i in range(4)]
for x in range(4):
    for y in range(4):
        if field[x][y] == '.':
            for dx, dy in [(-1, -1), (0, -1), (1, -1)]:
                if 0 <= x + dx < 4 and 0 <= y + dy < 4 and \
                        field[x + dx][y + dy] == 'x' and \
                        (x + 2 * dx < 0 or x + 2 * dx >= 4 or field[x + 2 * dx][y + 2 * dy] == 'o'):
                    print("YES")
                    exit()
print("NO")
