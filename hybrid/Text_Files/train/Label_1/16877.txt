#pragma GCC optimize("Ofast")
#pragma GCC target("avx,avx2,fma")
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef unsigned long long llu;
#define rep(i,a,b) for(int i=a;i<=b;++i)
#define rrep(i,b,a) for(int i=b;i>=a;--i)
ll rem[200001];
int main() {
  int n,k;
  scanf("%d%d",&n,&k);
  rep(i,1,n) {
    int x;
    scanf("%d",&x);
    for(int j=2;j*j<=x;++j) {
      while(x%j==0) {
        rem[j]++;
        x/=j;
      }
    }
    if(x>1) rem[x]++;
  }
  ll ans=1e9;
  for(int i=1;i<=200000;++i) {
    if(rem[i]) {
      ll step=0,cnt=0;
      for(;cnt<k;) {
        if(rem[i]) cnt++,rem[i]--,step++;
        else break;
        i*=i;
      }
      ans=min(ans,step);
    }
  }
  printf("%lld\n",ans);
  return 0;
}
