#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int n, b, k, x, a[50005];
int powmod(int a, int b) {
    int ans = 1;
    while (b > 0) {
        if (b % 2) ans = (1LL * ans * a) % mod;
        a = (1LL * a * a) % mod;
        b /= 2;
    }
    return ans;
}
int main() {
    cin >> n >> b >> k >> x;
    for (int i = 1; i <= n; i++) cin >> a[i];
    if (k % x != 0) {
        cout << "0\n";
        return 0;
    }
    int sum = 0;
    for (int i = 1; i <= n; i++) sum += a[i];
    sum %= x;
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        ans = (ans + powmod((sum - a[i] % x + x) % x, b - 1) % mod * a[i] % mod) % mod;
    }
    cout << ans << "\n";
    return 0;
}
