n,m=map(int,input().split());a=[]
for i in range(m):a.append(list(map(int,input().split())))
x=[];y=[];z=0
for i in range(n):
    x.append(0);y.append(0)
for i in a:
    if x[i[0]-1]==0 and y[i[1]-1]==0:z+=1
    x[i[0]-1]=1;y[i[1]-1]=1
if z&1:print(-1)
else:
    print(z);
    for i in range(n):
        if x[i]==1 or y[i]==1:print(i+1,end=' ');
    print('');
    for i in range(n):
        if x[i]==0 and y[i]==0:print(i+1,end=' ');
