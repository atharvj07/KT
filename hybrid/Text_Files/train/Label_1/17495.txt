#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, q, m;
  cin >> n >> q >> m;
  vector<int> a(n), b(m), p(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    p[i] = i;
  }
  for (int i = 0; i < m; i++) {
    cin >> b[i];
    b[i]--;
  }
  for (int i = 0; i < q; i++) {
    int t, l, r;
    cin >> t >> l >> r;
    l--, r--;
    if (t == 1) {
      reverse(p.begin() + l, p.begin() + r + 1);
    } else {
      int len = r - l + 1;
      for (int j = 0; j < len / 2; j++) {
        swap(p[l + j], p[r - j]);
      }
    }
  }
  for (int i = 0; i < m; i++) {
    cout << a[p[b[i]]] << " ";
  }
  cout << endl;
  return 0;
}
