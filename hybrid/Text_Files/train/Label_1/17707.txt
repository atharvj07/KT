#include <bits/stdc++.h>
using namespace std;
const int N = 101;
int a[N][N], n, k, t;
bool ok() {
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < n; ++j)
            if (!((i == j) ? a[i][j] : a[j][i])) return false;
    return true;
}
int main() {
    ios::sync_with_stdio(false);
    cin >> n >> k;
    if (k > n * (n - 1) / 2) {
        cout << -1 << endl;
        return 0;
    }
    while (k--) {
        if (t < n) a[t][t++] = 1;
        else a[t - 1][t++] = 1;
        if (!ok()) {
            k++;
            if (t > 1) t -= 2;
            else t = 0;
            a[t][t] = 0;
        }
    }
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) cout << a[i][j] << ' ';
        cout << endl;
    }
}
