#include <bits/stdc++.h>
using namespace std;
int main() {
	string s, t;
	cin >> s >> t;
	int s_idx = s.size() - 1, t_idx = t.size() - 1, ans = 0;
	while (s_idx >= 0 && t_idx >= 0) {
		ans += s[s_idx] == t[t_idx];
		s_idx--;
		if (s[s_idx] == t[t_idx]) t_idx--;
	}
	cout << s.size() - ans << "\n";
}
