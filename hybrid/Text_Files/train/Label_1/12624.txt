#include<bits/stdc++.h>
using namespace std;
#define ll long long
bool check(ll x, ll y, ll z)
{
    if(x>y)
        swap(x,y);
    if(y>z)
        swap(y,z);
    if(x==y&&x==z)
        return true;
    else
        return false;
}
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    ll t;
    cin>>t;
    while(t--)
    {
        ll x,y,z;
        cin>>x>>y>>z;
        if(check(x,y,z))
            cout<<"YES\n"<<x<<" "<<y<<" "<<z<<"\n";
        else
            cout<<"NO\n";
    }
    return 0;
}
