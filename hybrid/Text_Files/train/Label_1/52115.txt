import java.util.*;

public class Main {
    private static final long MOD = 1000000007L;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        long[] sorted = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
            sorted[i] = a[i];
        }
        Arrays.sort(sorted);

        long[] rank = new long[n];
        for (int i = 0; i < n; i++) {
            rank[Arrays.binarySearch(sorted, a[i])] = i + 1;
        }

        long inv = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (rank[i] > rank[j]) {
                    inv += (rank[j] - rank[i]) * (rank[j] - rank[i]) % MOD;
                    inv %= MOD;
                }
            }
        }

        System.out.println(inv);
    }
}
