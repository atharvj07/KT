import sys

n = int(next(sys.stdin))
a = list(map(int, next(sys.stdin).split()))

if len(set(a)) >= n - 1:
    print(-1)
    sys.exit()

i = 0
ans = []
while i < n - 1:
    j = i + 1
    while j < n and a[j] != a[i]:
        j += 1
    if j < n:
        ans.append((i + 1, j + 1))
        i = j
        j += 1
    else:
        i = j

print(len(ans))
for l, r in ans:
    print(l, r)
