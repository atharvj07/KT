#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> log;
  while (cin.peek() != '\n') {
    string s;
    getline(cin, s);
    log.push_back(s);
  }
  int cnt = 0;
  string ans = "-1";
  for (int i = 0; i < log.size(); ++i) {
    int j = max(0, i - n + 1);
    string s = log[i].substr(11, 8);
    if (i == 0 || s != log[i - 1].substr(11, 8)) {
      cnt = 1;
    } else {
      cnt++;
    }
    if (cnt >= m) {
      ans = log[i].substr(0, 19);
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
