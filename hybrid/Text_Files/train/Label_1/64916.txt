import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int,input().split()))

edges = [[] for _ in range(n+1)]
for _ in range(n-1):
    u,v = map(int,input().split())
    edges[u].append(v)
    edges[v].append(u)

vis = [False]*(n+1)
size = [0]*(n+1)
def dfs(x:int,p:int):
    vis[x] = True
    size[x] = a[x-1]
    for y in edges[x]:
        if not vis[y] and y!=p:
            dfs(y,x)
            size[x]+=size[y]

dfs(1,1)

ans = 0
for i in range(1,n+1):
    ans = max(ans,size[i]*(n-size[i]))

print(ans)
