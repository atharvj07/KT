#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int ans = 0, last = -1;
    for (int i = 0; i < n; i++) {
      if (a[i] > x) {
        if (last != -1 && a[i] < a[last]) {
          ans = -1;
          break;
        }
        ans++;
        last = i;
      }
    }
    cout << ans << endl;
  }
}
