n=int(input())
a=[]
s=t=0
for i in range(n):
    x=int(input())
    a.append(x)
    s+=x
    t+=x*x
a.sort()
if n==4:
    if a[0]==a[1] and a[1]==a[2] and a[2]==a[3]: print('YES')
    else: print('NO')
elif n==3:
    x=s-2*a[0]
    if x>=1 and x%2==0 and x//2>=a[1] and x//2<=a[2]: print('YES\n',x//2,'\n',x//2)
    else: print('NO')
elif n==2:
    x=s-a[0]
    y=t+a[0]*a[0]-2*a[0]*x
    d=(x*x-y)**.5
    if d==int(d) and (d+x)%2==0 and (d-x)%2==0 and (d+x)//2>=a[1] and (d-x)//2>=1: print('YES\n',(d+x)//2,'\n',(d-x)//2)
    else: print('NO')
else:
    x=s
    y=t
    d=(x*x-y)**.5
    if d==int(d) and (d+x)%2==0 and (d-x)%4==0 and (d+x)//2>=1 and (d-x)//4>=1: print('YES\n',(d+x)//2,'\n',(d+x)/4-1,'\n',(d+x)/4-1)
    else: print('NO')
