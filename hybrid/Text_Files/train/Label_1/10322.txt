#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 998244353;
int f(int x, int y) {
  int sz1 = 0;
  int sz2 = 0;
  vector<int> a, b;
  while (x > 0) {
    a.push_back(x % 10);
    sz1++;
    x /= 10;
  }
  while (y > 0) {
    b.push_back(y % 10);
    sz2++;
    y /= 10;
  }
  vector<int> v;
  if (sz1 >= sz2) {
    for (int i = 0; i <= sz1 - sz2 - 1; i++) {
      v.push_back(a[i]);
    }
    for (int i = 0; i < sz2; i++) {
      v.push_back(b[i]);
      v.push_back(a[sz1 - sz2 + i]);
    }
  } else {
    for (int i = 0; i <= sz2 - sz1 - 1; i++) {
      v.push_back(b[i]);
    }
    for (int i = 0; i < sz1; i++) {
      v.push_back(a[i]);
      v.push_back(b[sz2 - sz1 + i]);
    }
  }
  int ans = 0;
  for (auto it : v) {
    ans *= 10;
    ans += it;
    ans %= mod;
  }
  return ans;
}
int32_t main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      ans += f(v[i], v[j]);
      ans %= mod;
    }
  }
  cout << ans << "\n";
}
