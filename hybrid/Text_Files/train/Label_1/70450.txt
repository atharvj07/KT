t = int(input())
for _ in range(t):
    n = int(input())
    q = list(map(int, input().split()))
    p = []
    available = set(range(1, n+1))
    for i in range(n):
        if i == 0 or q[i] != q[i-1]:
            p.append(q[i])
            available.remove(q[i])
        else:
            found = False
            for x in available:
                if x < q[i]:
                    p.append(x)
                    available.remove(x)
                    found = True
                    break
            if not found:
                p = [-1]
                break
    if p[0] == -1:
        print(-1)
    else:
        print(' '.join(map(str, p)))
