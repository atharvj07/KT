n = int(input())
a = list(map(int, input().split()))

# Sort the list for easier comparison
a.sort()

# Count the number of ways Jerry can win
wins = 0

# Total number of possible games
total_games = 0

for i in range(n):
    for j in range(i + 1, n):
        # Andrew wins the first two rounds
        andrew_wins = j
        # Counts how many numbers are greater than andrew_wins (for Jerry to win the third round)
        jerry_wins_third = n - 1 - j
        # Jerry needs to draw a number less than andrew_wins in the first two rounds
        # and a number greater in the last round to have a higher total sum
        jerry_wins_total = andrew_wins
        # Increment the wins and total_games counters
        wins += jerry_wins_third * jerry_wins_total * jerry_wins_total
        total_games += jerry_wins_third * (j - i) * (j - i)

# Calculate the probability
probability = wins / total_games if total_games > 0 else 0

# Print the result
print(f"{probability:.10f}")
