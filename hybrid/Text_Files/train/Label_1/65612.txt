n=int(input())
v=[0]*n
for i in range(n):
    v[i]=input()
cnt=0
for i in range(n):
    for j in range(n):
        if i!=j:
            t=""
            for c in v[i]:
                t+=c+'1+'
            for c in v[j]:
                t+=c+'1+'
            if t[:-2]=='1+)*'*(len(v[i]+v[j])//2-1):
                cnt+=1
        else:
            t=""
            for c in v[i]:
                t+=c+'1+'
            if t[:-2]=='1+)*'*(len(v[i])//2):
                cnt+=1
print(cnt)
