#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
#define vi(x) vector<x>
#define pb push_back
const ll mod = 1e9+7;
const char nl = '\n';
ll n,m,a,b,x,y,st,en;
vi(ll)song;
int main(){
    cin>>n>>m;
    for(int i=0;i<n;++i){
        cin>>a>>b;
        song.pb(a-b);
    }
    sort(song.begin(),song.end());
    for(int i=0;i<n;++i){
        if(m>=song[i]){
            m-=song[i];
            ++x;
        }
    }
    cout<<(m? -1 : n-x);
    return 0;
}
