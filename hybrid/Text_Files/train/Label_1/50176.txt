import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class ChoutiAndTheStrangeSequence {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        int[] evenNumbers = new int[n / 2];
        String[] evenNumbersStr = reader.readLine().split(" ");
        for (int i = 0; i < n / 2; i++) {
            evenNumbers[i] = Integer.parseInt(evenNumbersStr[i]);
        }

        List<Long> oddNumbers = new ArrayList<>();
        long sum = 0;
        for (int evenNumber : evenNumbers) {
            sum += evenNumber;
            long squareRoot = (long) Math.sqrt(sum);
            if (squareRoot * squareRoot != sum) {
                System.out.println("No");
                return;
            }
            oddNumbers.add(squareRoot - evenNumber);
        }

        System.out.println("Yes");
        for (long oddNumber : oddNumbers) {
            System.out.print((oddNumber + 1) + " ");
        }
        for (int evenNumber : evenNumbers) {
            System.out.print(evenNumber + " ");
        }
    }
}
