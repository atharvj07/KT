import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        long[] h = new long[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }
        long ans = 0;
        for (long x : h) {
            ans += (x + b - 1) / a;
        }
        System.out.println(ans);
    }
}
