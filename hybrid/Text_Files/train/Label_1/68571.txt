from math import ceil
n, a, b, k = map(int, input().split())
shots = input()
if b * a <= n - k:
    print(b * a)
    for i in range(a):
        for j in range(b):
            if shots[i + j * a] == '0':
                print(i + j * a + 1, end=' ')
else:
    x = ceil((n - k) / a)
    print(a * x)
    for i in range(a):
        for j in range(x + 1 if i < b else x):
            if shots[i + j * a] == '0':
                print(i + j * a + 1, end=' ')
