#include <iostream>
#include <string>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    string s, t;
    cin >> s >> t;
    int i = 0, j = 0, wild = -1;
    while (i < n && j < m) {
        if (s[i] == '*') { wild = i; i++; continue; }
        if (s[i] != t[j]) {
            if (wild == -1) { cout << "NO"; return 0; }
            i = wild;
        } else
            i++, j++;
    }
    if (i == n && j == m) cout << "YES";
    else cout << "NO";
    return 0;
}
