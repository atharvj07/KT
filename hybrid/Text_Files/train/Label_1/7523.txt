#include<bits/stdc++.h>
using namespace std;
const int N=110;
int n,a[N],L,ans;
int main(){
	cin>>n;
	for(int i=1;i<=n;i++){
		cin>>a[i];
	}
	sort(a+1,a+1+n);
	L=unique(a+1,a+1+n)-a-1;
	for(int i=2;i<=L;i++){
		if(a[i]-a[i-1]>1)return puts("Alice"),0;
	}
	for(int i=1;i<L;i++){
		if(a[i]+a[i+1]<=a[L])ans^=1;
	}
	if(ans)puts("Bob");
	else puts("Alice");
	return 0; 
}
