#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, el;
    cin >> n;
    vector<int> a;
    for (int i = 0; i < n; i++) {
        cin >> el;
        if (!a.empty() && a.back() == el)
            a.pop_back();
        else
            a.push_back(el);
    }
    cout << a.size() << '\n';
    for (int e : a) cout << e << ' ';
    cout << '\n';
    return 0;
}
