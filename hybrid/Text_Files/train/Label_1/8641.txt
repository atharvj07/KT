#include <bits/stdc++.h>
using namespace std ;
int32_t main(){
    int n,q; cin >> n >> q;
    string s; cin >> s;
    while(q--){
        int l,r; cin >> l >> r;
        --l,--r;
        map<int,int> mp;
        for(int i=l;i<=r;++i){
            if(isdigit(s[i]))
                mp[s[i]-'0']++;
            else if(s[i]=='<')
                --i;
        }
        for(int i=0;i<10;++i)
            cout << mp[i] << " ";
        cout << endl;
    }
    return 0;
}
