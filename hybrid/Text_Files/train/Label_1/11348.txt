#include <iostream>
#include <unordered_map>
using namespace std;
int main() {
    int color, count = 0;
    unordered_map<string, int> mp = {{"purple", 1}, {"green", 1}, {"blue", 1}, {"orange", 1}, {"red", 1}, {"yellow", 1}};
    cin >> color;
    for (int i = 0; i < color; i++) {
        string c;
        cin >> c;
        mp[c] = 0;
    }
    for (auto ptr = mp.begin(); ptr != mp.end(); ptr++) {
        if (ptr->second) { count++; }
    }
    cout << count << endl;
    for (auto ptr = mp.begin(); ptr != mp.end(); ptr++) {
        if (ptr->second) { cout << ptr->first << endl; }
    }
    return 0;
}
