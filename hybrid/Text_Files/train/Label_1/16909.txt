#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, u, v;
    ll ans = 0;
    cin >> n >> u >> v;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    priority_queue<int, vector<int>, greater<int>> pq;
    for (int i = 0; i < n; i++) { pq.push(a[i]); }
    pq.push(0);
    pq.push(1e6 + 1);
    while (!pq.empty()) {
      int x = pq.top();
      pq.pop();
      if (x == 1e6 + 1) break;
      int y = pq.top();
      if (y < x) break;
      pq.pop();
      ans += min({v, u + v, v + u + v});
      if (x - 1 > y) pq.push(x - 1);
      if (x + 1 < y) pq.push(x + 1);
    }
    cout << ans << "\n";
  }
  return 0;
}
