#include <iostream>

using namespace std;

int main() {
  int x1, y1, x2, y2, x3, y3;
  cin >> x1 >> y1 >> x2 >> y2 >> x3 >> y3;
  int ans = 0;
  if (x1 == x2 || x2 == x3 || x1 == x3) {
    ans++;
    if (y1 == y2 || y2 == y3 || y1 == y3) ans++;
  } else if (y1 == y2 || y2 == y3 || y1 == y3) {
    ans++;
    if (x1 == x2 || x2 == x3 || x1 == x3) ans++;
  } else
    ans = 3;
  cout << ans << endl;
  return 0;
}
