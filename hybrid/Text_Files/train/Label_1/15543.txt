#include <cstdio>
#include <vector>
#include <algorithm>
#include <queue>
using namespace std;

const int Maxn = 1e5+10;
typedef long long ll;

vector<pair<int, ll>> G[Maxn]; //(v, w)
ll dis[Maxn], v[Maxn], ans[Maxn];
int main() {
    int n, m, k; scanf("%d%d%d", &n, &m, &k);
    for(int i = 1; i <= n; ++i) G[i].clear();
    for(int i = 1; i <= k; ++i) scanf("%lld", &v[i]);
    int x, y, w; ll z;
    for(int i = 0; i < m; ++i) {
        scanf("%d%d%lld", &x, &y, &z);
        G[x].push_back({y, z}); G[y].push_back({x, z});
    }

    for(int i = 1; i <= k; ++i) {
        priority_queue<ll, vector<ll>, greater<ll>> q;
        for(int j = 1; j <= n; ++j) dis[j] = -1;
        dis[v[i]] = 0; q.push(v[i]);
        while(!q.empty()) {
            ll f = q.top(); q.pop();
            if(dis[f] > ans[i]) ans[i] = dis[f];
            for(auto x: G[f]) if(dis[x.first] < 0 or dis[x.first] > dis[f] + x.second) {
                dis[x.first] = dis[f] + x.second;
                q.push(x.first);
            }
        }
    }
    for(int i = 1; i <= k; ++i) printf("%lld ", ans[i]);
    return 0;
}
