#include <iostream>
#include <cmath>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k1, k2;
  cin >> n >> k1 >> k2;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }

  sort(a.begin(), a.end());
  sort(b.begin(), b.end());

  int i = 0, j = 0, ans = 0;
  while (i < n && j < n && k1 > 0 && k2 > 0) {
    if (abs(a[i] - b[j]) <= abs(a[i + 1] - b[j]) && abs(a[i] - b[j]) <= abs(a[i] - b[j + 1])) {
      ans += abs(a[i] - b[j]) * abs(a[i] - b[j]);
      i++;
      k1--;
    } else if (abs(a[i] - b[j]) <= abs(a[i] - b[j + 1]) && abs(a[i] - b[j]) <= abs(a[i + 1] - b[j])) {
      ans += abs(a[i] - b[j]) * abs(a[i] - b[j]);
      j++;
      k2--;
    } else if (abs(a[i + 1] - b[j]) <= abs(a[i] - b[j + 1]) && abs(a[i + 1] - b[j]) <= abs(a[i] - b[j])) {
      ans += abs(a[i + 1] - b[j]) * abs(a[i + 1] - b[j]);
      i++;
      k1--;
    } else if (abs(a[i] - b[j + 1]) <= abs(a[i + 1] - b[j]) && abs(a[i] - b[j + 1]) <= abs(a[i] - b[j])) {
      ans += abs(a[i] - b[j + 1]) * abs(a[i] - b[j + 1]);
      j++;
      k2--;
    }
  }

  while (i < n && k1 > 0) {
    ans += abs(a[i] - b[j]) * abs(a[i] - b[j]);
    i++;
    k1--;
  }

  while (j < n && k2 > 0) {
    ans += abs(a[i] - b[j]) * abs(a[i] - b[j]);
    j++;
    k2--;
  }

  cout << ans << "\n";

  return 0;
}
