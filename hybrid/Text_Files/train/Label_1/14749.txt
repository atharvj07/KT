#include <bits/stdc++.h>
using namespace std;
char a[2005][2005];
int n,k,ans;
int main(){
    cin>>n>>k;
    for(int i=1;i<=n;++i){
        for(int j=1;j<=n;++j) cin>>a[i][j];
    }
    for(int i=1;i<=n;++i){
        int sumr=0;
        for(int j=1;j<=n;++j) sumr+=a[i][j]=='B';
        ans+=sumr==0;
    }
    for(int j=1;j<=n;++j){
        int sumc=0;
        for(int i=1;i<=n;++i) sumc+=a[i][j]=='B';
        ans+=sumc==0;
    }
    cout<<ans<<endl;
    return 0;
}
