import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int,input().split()))
b = [0]*n
for i in range(n):
    b[a[i]-1] = i
ans = []
for i in range(n):
    if i==b[i]:
        continue
    j=i
    while j!=b[i]:
        ans.append((j+1,b[i]+1))
        a[j],a[b[i]]=a[b[i]],a[j]
        b[a[j]]=b[a[i]]
        b[a[i]]=j
        j=b[i]
print(len(ans))
for i in ans:
    print(*i)
