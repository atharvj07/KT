from collections import Counter
n = int(input())
a = list(map(int, input().split()))
b = Counter()
ans = 0
l = 0
for r in range(n):
    b[a[r]] += 1
    Min = min(b.values())
    Max = max(b.values())
    if Max - Min <= 1:
        ans = r - l + 1
    else:
        while Max - Min > 1 and l <= r:
            b[a[l]] -= 1
            if b[a[l]] == 0:
                del b[a[l]]
            Min = min(b.values())
            Max = max(b.values())
            l += 1
print(ans)
