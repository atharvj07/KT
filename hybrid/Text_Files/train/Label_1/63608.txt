def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def sequence_occurs_in_G(n, m, k, a):
    if k == 1:
        if a[0] <= n or a[0] <= m:
            return "YES"
        else:
            return "NO"
    
    g = a[0]
    for i in range(1, k):
        g = gcd(g, a[i])
    
    for i in range(1, min(n, m) + 1):
        if gcd(i, g) == g:
            if i <= n and g + k - 1 <= m:
                return "YES"
            if i <= m and g + k - 1 <= n:
                return "YES"
    return "NO"

n, m, k = map(int, input().split())
a = list(map(int, input().split()))

print(sequence_occurs_in_G(n, m, k, a))
