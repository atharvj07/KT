x1,y1 = map(int,input().split())
x2,y2 = map(int,input().split())
x3,y3 = map(int,input().split())
ans = x1+1+y1+1
if (x2!=x1 or y2!=y3) and max(x2,x3)>x1 and min(x2,x3)>x1: ans+=2
if y2!=y1 and max(y2,y3)>y1 and min(y2,y3)>y1: ans+=2
print(ans)
for i in range(x1+1): print(i,0)
for i in range(y1+1): print(x1,i)
if x2!=x1 or y2!=y3:
    if x2>x1: print(x1+1,max(y2,y3)-1)
    else: print(x1,max(y2,y3))
if y2!=y1:
    if y2>y1: print(x3,y1+1)
    else: print(x3,y1)
