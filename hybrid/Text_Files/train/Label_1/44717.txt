import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer in = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(in.nextToken()), m = Integer.parseInt(in.nextToken());
        int[] a = new int[n+1];
        in = new StringTokenizer(br.readLine());
        for(int i = 1; i <= n; i++){
            a[i] = Integer.parseInt(in.nextToken());
        }
        while(m-->0){
            in = new StringTokenizer(br.readLine());
            int l = Integer.parseInt(in.nextToken()), r = Integer.parseInt(in.nextToken());
            if((r-l+1)%2==1 || (a[r]==1 && r-l+1==2) || (a[l]==-1 && r-l+1==2)){
                System.out.println(0);
            } else {
                System.out.println(1);
            }
        }
    }
}
