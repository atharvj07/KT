#include <iostream>
#include <vector>

using namespace std;

int main() {
  int T;
  cin >> T;

  while (T--) {
    int n, x;
    cin >> n >> x;
    string s;
    cin >> s;

    long long sum = 0;
    vector<int> pref(n);

    for (int i = 0; i < n; i++) {
      sum += s[i] == '0' ? -1 : 1;
      pref[i] = sum;
    }

    bool found = false;
    for (int i = 0; i < n; i++) {
      if (x % n == 0 && (pref[i] + x / n * n) % n == 0) {
        cout << -1 << endl;
        found = true;
        break;
      }
    }
    if (!found) {
      int ans = 0;
      for (int i = 0; i < n; i++) {
        if (x % n == 0 && (pref[i] + x / n * n) % n == 0) ans++;
      }
      cout << ans << endl;
    }
  }

  return 0;
}
