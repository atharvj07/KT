#include <iostream>
typedef long long ll;
ll GCD(ll a, ll b){
    if(a==0)return b;
    return GCD(b%a,a);
}
int main(){
    ll a[3],b[3];
    for(int i=0;i<3;i++)std::cin>>a[i]>>b[i];
    ll t=GCD(a[2],b[2]);
    ll l=a[0]%t,r=b[0]%t;
    if(l<r)std::cout<<r-l+1<<std::endl;
    else std::cout<<t+r-l+1<<std::endl;
    return 0;
}
