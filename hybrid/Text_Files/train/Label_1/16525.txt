#include<bits/stdc++.h>
#define FOR(a,b,c) for(int a=b;a<c;++a)

using namespace std;

const int maxn = 4001, oo=1e9;
int n,m,a[maxn];
bool vis[maxn],know[maxn][maxn];

int main() {
    cin>>n>>m;
    FOR(i,1,m+1) {
        int x,y; cin>>x>>y;
        know[x][y]=know[y][x]=true;
    }
    int ans=oo;
    FOR(i,1,n+1) {
        memset(vis,false,sizeof vis);
        FOR(j,1,n+1) a[j]=j==i ? 0 : oo;
        queue<int> q;
        q.push(i); vis[i]=true;
        while(!q.empty()) {
            int u=q.front();q.pop();
            FOR(k,1,n+1) if(!vis[k] && know[u][k]) {
                vis[k]=true; q.push(k);
                a[k]=min(a[k],a[u]+1);
            }
        }
        FOR(j,1,n+1) if(j!=i) ans=min(ans,a[j]);
    }
    if(ans==oo) ans=-1;
    cout<<ans<<'\n';
    return 0;
}
