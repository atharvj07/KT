#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    int ans = 0;
    while (m--) {
        int x;
        cin >> x;
        x--;
        a[x]++;
        if (a[x] == 1) ans++;
        for (int i = 0; i < n; i++) {
            a[i]--;
            if (a[i] > 0) { ans++; break; }
        }
    }
    cout << ans << endl;
}
