#include <bits/stdc++.h>
using namespace std;
using vvi=vector<vector<int>>;
using vi=vector<int>;
using ll=long long;
const ll mod=1e9+7;
ll n,m,x[300005],y[300005],cur[300005],u,v,ans[300005];
int main(){
	cin>>n>>m;
	for(int i=1;i<=n;++i){
		cin>>x[i]>>y[i];
	}
	vvi g(n+1);
	for(int i=1;i<=m;++i){
		cin>>u>>v;
		g[u].push_back(v);
		g[v].push_back(u);
	}
	vector<bool> vst(n+1);
	for(int i=1;i<=n;++i){
		if(vst[i])continue;
		queue<int> c;
		vvi vcur={vi(),vi()};
		c.push(i);
		vst[i]=1;
		while(!c.empty()){
			int p=c.front();
			c.pop();
			vcur[x[p]>y[p]].push_back(p);
			for(auto nx:g[p])if(!vst[nx]){
				c.push(nx);
				vst[nx]=1;
			}
		}
		for(auto a:vcur[0]){
			cur[a]=x[a];
		}
		for(auto a:vcur[1]){
			cur[a]=y[a];
		}
		for(auto a:vcur[0]){
			for(auto b:vcur[1]){
				ans[a]+=cur[b];
				ans[b]+=x[a];
			}
		}
	}
	for(int i=1;i<=n;++i){
		cout<<ans[i]<<' ';
	}
	cout<<endl;
}
