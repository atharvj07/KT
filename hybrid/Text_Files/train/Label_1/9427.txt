#include<bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while(t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for(auto &el: a) cin >> el;
    vector<int> dp(n, 0), dp1(n, 0);
    if(a[0] > 0) dp[0] = a[0];
    if(a[0] < 0) dp1[0] = a[0];
    for(int i = 1; i < n; i++) {
      if(a[i] > 0) {
        dp[i] = max({dp[i], dp[i - 1], dp1[i - 1] + a[i]});
        dp1[i] = dp1[i - 1];
      } else if(a[i] < 0) {
        dp1[i] = max({dp1[i], dp1[i - 1], dp[i - 1] + a[i]});
        dp[i] = dp[i - 1];
      }
    }
    cout << max(dp[n - 1], dp1[n - 1]) << endl;
  }
  return 0;
}
