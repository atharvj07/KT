def process_slimes(n):
    slimes = []
    for _ in range(n):
        if slimes and slimes[-1] == 1:
            value = 2
            slimes.pop()
            while slimes and slimes[-1] == value:
                slimes.pop()
                value += 1
            slimes.append(value)
        else:
            slimes.append(1)
    return slimes

n = int(input().strip())
result = process_slimes(n)
print(" ".join(map(str, result)))
