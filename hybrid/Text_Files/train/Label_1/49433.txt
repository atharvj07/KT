import java.util.Scanner;
import java.util.HashMap;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int m = in.nextInt(), d = in.nextInt();
        int[][][] bases = new int[m][][];
        for (int i = 0; i < m; i++) {
            int k = in.nextInt();
            int[][] bs = new int[k][d];
            for (int j = 0; j < k; j++) {
                for (int a = 0; a < d; a++) {
                    bs[j][a] = in.nextInt();
                }
            }
            Arrays.sort(bs, (a, b) -> Arrays.hashCode(a) - Arrays.hashCode(b));
            bases[i] = bs;
        }
        HashMap<String, Integer> group = new HashMap<>();
        for (int i = 0; i < m; i++) {
            StringBuilder sb = new StringBuilder();
            for (int[] b : bases[i]) {
                sb.append(Arrays.toString(b));
            }
            if (!group.containsKey(sb.toString())) {
                group.put(sb.toString(), group.size() + 1);
            }
            System.out.print(group.get(sb.toString()) + " ");
        }
    }
}
