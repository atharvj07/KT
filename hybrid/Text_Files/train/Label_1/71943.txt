import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))

odd = [0] * (n+1)
even = [0] * (n+1)

for i in range(n - 1):
    if (i % 2 == 0):
        odd[i + 1] = odd[i] + abs(a[i] - a[i + 1])
    else:
        even[i + 1] = even[i] + abs(a[i] - a[i + 1])

ans = -1e9
for i in range(1, n + 1):
    for j in range(i + 1, n + 1):
        s = even[j - 1] - even[i - 1]
        if (i % 2 == 1):
            s -= odd[j - 1] - odd[i - 1]
        ans = max(ans, s)

print(ans)
