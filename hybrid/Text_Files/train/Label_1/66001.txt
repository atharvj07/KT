    print("NO")
else:
    print("YES")
    for i in range(int2):
        print(1, unsorted.pop())
    for i in range(int1-int2):
        print(1, unsorted.pop())
    for i in range(int2-int3):

    if int3 > 0:
        print(int1-2*(int2-int3), *unsorted)
