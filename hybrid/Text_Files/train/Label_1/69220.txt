def solve(s : str, x : int) -> str:
    n = len(s)
    w = ['0'] * n
    for i in range(n):
        if (i - x >= 0 and s[i - x] == '1') or (i + x < n and s[i + x] == '1'):
            w[i] = '1'
    if all(w[i] == '0' for i in range(x, n - x)):
        return '-1'
    return ''.join(w)

for _ in range(int(input())):
    s = input()
    x = int(input())
    print(solve(s, x))
