#include <cstdio>
#include <cstring>
#include <cmath>
#include <algorithm>
using namespace std;
#define MP make_pair
#define PB emplace_back
typedef long long ll;
const int N = 100005;
const ll inf = 1e18;
int n, m, k;
ll a[N][4];
int main() {
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            scanf("%lld", &a[i][j]);
        }
    }
    scanf("%d", &k);
    for (int i = 1; i <= k; i++) {
        int l, r;
        scanf("%d%d", &l, &r);
        bool flag = 0;
        for (int j = 1; j <= m; j++) {
            flag = 1;
            for (int k = l + 1; k <= r; k++) {
                if (a[k][j] - a[k - 1][j] > 1) {
                    flag = 0;
                    break;
                }
            }
            if (flag) {
                break;
            }
        }
        puts(flag ? "Yes" : "No");
    }
    return 0;
}
