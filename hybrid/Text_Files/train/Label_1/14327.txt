#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pi;
int main() {
  int n;
  cin >> n;
  vector<pi> v;
  int tot = 0;
  for (int i = 0; i < n; i++) {
    string s;
    int x;
    cin >> s >> x;
    v.emplace_back(x, stoi(s, 0, 2));
    if (s[0] == '0') tot += x;
  }
  int ans = 0;
  sort(v.rbegin(), v.rend());
  int aCnt = 0, bCnt = 0;
  for (auto &e : v) {
    if ((e.second & 1) && aCnt < n / 2)
      ans += e.first, aCnt++;
    else if (!(e.second & 1) && bCnt < n / 2)
      ans += e.first, bCnt++;
    else if (tot > 0)
      ans += e.first, tot -= e.first;
  }
  cout << ans << '\n';
  return 0;
}
