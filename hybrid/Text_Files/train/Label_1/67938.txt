def add(a,b):
    a=str(a);b=str(b)
    x,y=len(a),len(b)
    if x<y:a='0'*(y-x)+a;x=y
    if x>y:b='0'*(x-y)+b;y=x
    c=''
    for i in range(x):
        s=int(a[x-i-1]) + int(b[x-i-1])
        c=str(s%10) + c
        s/=10
        s=int(s)
        if i+1==x and s>0:c=str(s) + c
    return c
for _ in range(int(input())):
    a,b=input().split()
    x=add(a,b)
    if len(x)==len(b):print(x)
    else:print(-1)
