import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] data = br.readLine().split(" ");
        int n = Integer.parseInt(data[0]), m = Integer.parseInt(data[1]);
        Voter[] voters = new Voter[n];
        for (int i = 0; i < n; i++) {
            data = br.readLine().split(" ");
            voters[i] = new Voter(Integer.parseInt(data[0]), Integer.parseInt(data[1]));
        }
        Arrays.sort(voters, Comparator.comparingInt(Voter::getCost));
        int sum = 0, cnt = 0, max = 0;
        for (Voter voter : voters) {
            if (voter.party == 1) {
                cnt++;
            } else {
                max = Math.max(max, voter.count);
            }
            if (cnt > max) {
                break;
            }
            sum += voter.cost;
            cnt++;
        }
        System.out.println(sum);
    }

    private static class Voter {
        int party, cost, count = 1;

        public Voter(int party, int cost) {
            this.party = party;
            this.cost = cost;
        }

        public int getCost() {
            return cost;
        }
    }
}
