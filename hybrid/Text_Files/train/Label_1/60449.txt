def max_total_damage(n, a, b, creatures):
    creatures.sort(key=lambda x: max(x[0]-x[1], 0), reverse=True)
    total_damage = sum(dmg for hp, dmg in creatures)
    max_increase = 0

    if b > 0:
        for i in range(min(n, b)):
            hp, dmg = creatures[i]
            max_increase = max(max_increase, max(hp * (2**a) - dmg, 0))

        for i in range(min(n, b), n):
            _, dmg = creatures[i]
            max_increase = max(max_increase, -dmg)

        total_damage += sum(max(hp - dmg, 0) for hp, dmg in creatures[:min(n, b)])
    else:
        max_increase = max([hp * (2**a) - dmg for hp, dmg in creatures] + [0])

    total_damage += max_increase
    return total_damage

n, a, b = map(int, input().split())
creatures = [tuple(map(int, input().split())) for _ in range(n)]
print(max_total_damage(n, a, b, creatures))
