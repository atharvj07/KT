#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> a(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> a[i][j];
    }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      ans |= a[i][j];
    }
  }
  if (ans == 0) {
    cout << "NIE" << endl;
  } else {
    cout << "TAK" << endl;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (a[i][j] > 0) {
          cout << j + 1 << " ";
          break;
        }
      }
    }
    cout << endl;
  }
  return 0;
}
