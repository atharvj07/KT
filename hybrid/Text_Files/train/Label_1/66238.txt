def max_saved_points(t, test_cases):
    results = []
    for n, k, x_coords, y_coords in test_cases:
        # Sorting points by x-coordinate
        points = sorted(zip(x_coords, y_coords), key=lambda p: p[0])
        x_sorted = [x for x, y in points]

        # Precompute the maximum points each platform can cover starting from each point
        max_covers = [1] * n
        right = 0
        for left in range(n):
            while right < n and x_sorted[right] - x_sorted[left] <= k:
                right += 1
            max_covers[left] = right - left

        # Precompute the maximum points can be covered from the right side for each point
        max_from_right = [0] * (n + 1)
        for i in range(n - 1, -1, -1):
            max_from_right[i] = max(max_from_right[i + 1], max_covers[i])

        # Calculate the maximum points can be saved
        max_saved = 0
        for i in range(n):
            current_saved = max_covers[i] + (max_from_right[i + max_covers[i]] if i + max_covers[i] < n else 0)
            max_saved = max(max_saved, current_saved)

        results.append(max_saved)

    return results

# Reading input and outputting the results
t = int(input().strip())
test_cases = []
for _ in range(t):
    n, k = map(int, input().strip().split())
    x_coords = list(map(int, input().strip().split()))
    y_coords = list(map(int, input().strip().split()))
    test_cases.append((n, k, x_coords, y_coords))

results = max_saved_points(t, test_cases)
for result in results:
    print(result)
