import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int a = sc.nextInt();
    int b = sc.nextInt();
    int h = sc.nextInt();
    int m = sc.nextInt();
    System.out.println(Math.sqrt(a * a + b * b - 2 * a * b * Math.cos(Math.PI * (30 * h + 0.5 * m - 6 * m) / 180)));
  }
}
