#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> M(n, vector<int>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) { cin >> M[i][j]; }
  }
  vector<int> a(n);
  a[0] = 1;
  for (int i = 1; i < n; i++) {
    a[i] = M[0][i] / a[0];
  }
  for (int i = 0; i < n; i++) {
    cout << a[i] << ' ';
  }
  cout << endl;
  return 0;
}
