n = int(input())
a = sorted(list(map(int, input().split())))
i = 0
j = n - 1
inc = []
dec = []
while i < j:
    if a[i] <= a[i + 1]:
        inc.append(a[i])
        i += 1
    if a[j] >= a[j - 1]:
        dec.append(a[j])
        j -= 1
    else:
        break
if i == j:
    inc.append(a[i])
if len(inc) + len(dec) == n:
    print("YES")
    print(len(inc))
    print(*inc)
    print(len(dec))
    print(*dec)
else:
    print("NO")
