import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] ciel = new int[n];
        int[] jiro = new int[n];
        for (int i = 0; i < n; i++) {
            int s = in.nextInt();
            int[] a = new int[s];
            for (int j = 0; j < s; j++) {
                a[j] = in.nextInt();
            }
            Arrays.sort(a);
            for (int j = 0; j < s; j++) {
                if (j % 2 == 0) {
                    ciel[i] += a[j];
                } else {
                    jiro[i] += a[j];
                }
            }
        }
        int cielSum = 0;
        int jiroSum = 0;
        for (int i = 0; i < n; i++) {
            cielSum += ciel[i];
            jiroSum += jiro[i];
        }
        System.out.println(cielSum + " " + jiroSum);
    }
}
