import java.util.*;

public class Main {
    static int n;
    static int[] a;
    static int[] dp;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
        }
        dp = new int[n + 1];
        Arrays.fill(dp, -1);
        System.out.println(solve(1, n));
    }

    static int solve(int l, int r) {
        if (l > r) {
            return 0;
        }
        if (dp[l] != -1) {
            return dp[l];
        }
        int ans = a[l];
        for (int i = l + 1; i <= r; i++) {
            ans = Math.max(ans, a[i] + solve(i + 1, r));
        }
        dp[l] = ans;
        return ans;
    }
}
