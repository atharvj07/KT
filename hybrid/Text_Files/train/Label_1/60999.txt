import sys
input = sys.stdin.readline
n,m=map(int,input().split())
g=[[] for _ in range(n+1)]
for _ in range(m):
    a,b=map(int,input().split())
    g[a].append(b)
    g[b].append(a)
vis=[0]*(n+1)
ans=[0]*(m+1)
def dfs(x,p,d):
    vis[x]=1
    for u in g[x]:
        if vis[u]==0:ans[d]=-ans[d]
        elif u==p:
            if ans[d]==ans[d^1]:return False
        else:
            if ans[d]==ans[d^1]:return False
            ans[d]=-ans[d]
        if not dfs(u,x,d^1):return False
    return True
if not dfs(1,0,0):print('NO')
else:
    print('YES')
    print(*ans[1:])
