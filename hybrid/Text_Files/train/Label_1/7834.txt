#include <bits/stdc++.h>
using namespace std;
const int N = 25;
vector<int> v[N];
vector<int> ans;
int vis[N], cnt;
void dfs(int u, int p) {
  if (vis[u] != -1) {
    if (!vis[p]) { ans.push_back(u); }
    vis[u] = vis[p] + 1;
    return;
  }
  vis[u] = 0;
  for (auto to : v[u]) {
    dfs(to, u);
  }
}
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    v[a].push_back(b);
    v[b].push_back(a);
  }
  memset(vis, -1, sizeof vis);
  dfs(1, 1);
  cout << ans.size() << endl;
  for (auto x : ans) { cout << x << " "; }
  cout << endl;
  return 0;
}
