n,m,d=map(int,input().split())
c1=[]
w1=[]
for i in range(n):
    a,b=map(int,input().split())
    c1.append(a)
    w1.append(b)
c2=[]
w2=[]
for i in range(m):
    a,b=map(int,input().split())
    c2.append(a)
    w2.append(b)
c1.sort(reverse=True)
w1.sort(reverse=True)
c2.sort(reverse=True)
w2.sort(reverse=True)
s1,s2,ans=0,0,0
i1,i2=0,0
while(i1<n and i2<m):
    if(d>0):
        if(w1[i1]<=d and w2[i2]<=d):
            if(c1[i1]>=c2[i2]):
                ans+=c1[i1]
                d-=w1[i1]
                i1+=1
            else:
                ans+=c2[i2]
                d-=w2[i2]
                i2+=1
        elif(w1[i1]<=d):
            ans+=c1[i1]
            d-=w1[i1]
            i1+=1
        elif(w2[i2]<=d):
            ans+=c2[i2]
            d-=w2[i2]
            i2+=1
        else:
            break
    else:
        break
while(i1<n):
    if(w1[i1]<=d):

        d-=w1[i1]

    else:

while(i2<m):
