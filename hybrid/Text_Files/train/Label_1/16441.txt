#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n; cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  sort(a.begin(), a.end());
  vector<int> ans;
  int mn = 2e9, mx = 0;
  for (int i = 0; i < n; ++i) {
    mn = min(mn, a[i]);
    mx = max(mx, a[i]);
  }
  if (mx - mn > 1) { cout << 1 << endl << a[0] << endl; return 0; }
  for (int i = 0; i < n; ++i) {
    if (a[i] == mn) ans.push_back(a[i]);
    else break;
  }
  reverse(ans.begin(), ans.end());
  for (int i = 0; i < n; ++i) {
    if (a[i] == mx) ans.push_back(a[i]);
    else break;
  }
  cout << ans.size() << endl;
  for (auto v : ans) cout << v << " ";
  cout << endl;
  return 0;
}
