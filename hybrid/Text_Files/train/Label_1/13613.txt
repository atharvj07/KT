#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, v, p = 0, q[51];
    cin >> n >> v;
    for (int i = 0; i < n; i++) {
        int k;
        cin >> k;
        int mi = 100000;
        for (int j = 0; j < k; j++) {
            int t;
            cin >> t;
            mi = min(mi, t);
        }
        if (mi <= v) { q[p++] = i + 1; }
    }
    cout << p << endl;
    for (int i = 0; i < p; i++) {
        if (i) cout << " ";
        cout << q[i];
    }
    cout << endl;
}
