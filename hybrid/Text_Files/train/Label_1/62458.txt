t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    x = list(map(int, input().split()))

    total_sum = sum(a)
    max_sum = 0
    prefix_sum = [0]
    for i in range(n):
        prefix_sum.append(prefix_sum[-1] + a[i])
        max_sum = max(max_sum, prefix_sum[-1])

    for xi in x:
        if total_sum > 0:
            cycles = (xi - max_sum + total_sum - 1) // total_sum
            remaining_sum = xi - cycles * total_sum
        else:
            cycles = 0
            remaining_sum = xi

        if remaining_sum <= max_sum:
            ans = n * cycles
            for i in range(1, len(prefix_sum)):
                if prefix_sum[i] >= remaining_sum:
                    ans += i
                    break
        else:
            ans = -1

        print(ans, end=" ")
    print()
