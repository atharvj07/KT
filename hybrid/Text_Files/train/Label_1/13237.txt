#include <bits/stdc++.h>
using namespace std;

long long ncr(long long n, long long r) {
	double res = lgamma(n + 1) - lgamma(r + 1) - lgamma(n - r + 1);
	return llround(exp(res));
}

int main() {
	int t;
	cin >> t;
	while(t--) {
		long long x;
		cin >> x;
		if(x == 0) cout << "1 1\n";
		else if(x & 1) cout << "-1\n";
		else {
			long long n = sqrt((double)x * 2 / 3.);
			long long m = (3 * x) / (n * (n + 1));
			if(n * (n + 1) * m != 3 * x) cout << "-1\n";
			else cout << n << ' ' << m << '\n';
		}
	}
}
