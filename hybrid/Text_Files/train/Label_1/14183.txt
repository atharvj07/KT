#include <bits/stdc++.h>
using namespace std;
#define MOD 1000000007
#define ll long long
int main() {
    ll n, x, pos;
    cin >> n >> x >> pos;
    ll fact[n + 2];
    fact[0] = 1;
    for (int i = 1; i <= n + 1; i++)
        fact[i] = (fact[i - 1] * i) % MOD;
    ll res = (fact[n] * pos) % MOD;
    res = (res * ((fact[n - 1 - pos] * (n - pos)) % MOD)) % MOD;
    cout << res;
    return 0;
}
