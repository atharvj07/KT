#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(m);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < m; i++) cin >> b[i];
  int ans = -1;
  for (int v = 1; v <= 100; v++) {
    bool ok1 = true, ok2 = false, ok3 = true;
    for (int i = 0; i < n; i++) {
      if (a[i] > v) ok1 = false;
      if (a[i] <= v / 2) ok2 = true;
    }
    for (int i = 0; i < m; i++) {
      if (b[i] <= v) ok3 = false;
    }
    if (ok1 && ok2 && ok3) {
      ans = v;
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
