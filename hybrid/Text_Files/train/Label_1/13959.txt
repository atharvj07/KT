#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long a, b, x, y, n;
        cin >> a >> b >> x >> y >> n;
        if (n > 0) {
            if (a - x > b - y) {
                a = x + (n % 2);
                b -= (n - (n % 2));
            } else {
                b = y + (n % 2);
                a -= (n - (n % 2));
            }
        }
        cout << a * b << endl;
    }
    return 0;
}
