#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
struct {ll a, b;} p[100005];
int main() {
    ios::sync_with_stdio(false);
    ll n, s, ans = 0, sum = 0;
    cin >> n >> s;
    for (int i = 0; i < n; i++) {
        cin >> p[i].a;
        p[i].b = p[i].a + i;
    }
    sort(p, p + n, [&](const auto& a, const auto& b) { return a.b < b.b; });
    for (int i = 0; i < n; i++) {
        if (sum + p[i].a <= s) {
            sum += p[i].a;
            ans++;
        } else if (sum + p[i].b <= s) {
            sum += p[i].b;
            ans++;
            break;
        } else
            break;
    }
    cout << ans << " " << sum << endl;
    return 0;
}
