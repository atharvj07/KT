#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    sort(a.begin(), a.end());
    bool alice = false, bob = false, tie = false;
    long long alice_score = 0, bob_score = 0;
    for (const auto &x : a) {
      if (!tie) {
        if (x % 2 != 0) alice = true;
        else bob = true;
        if (alice && bob) tie = true;
      }
      if (x % 2 == 0) alice_score += x;
      else bob_score += x;
    }
    if (tie)
      cout << "Tie" << endl;
    else if (alice_score > bob_score)
      cout << "Alice" << endl;
    else if (alice_score == bob_score)
      cout << "Tie" << endl;
    else
      cout << "Bob" << endl;
  }
  return 0;
}
