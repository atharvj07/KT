#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) { std::cin >> a[i]; }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == 1 || a[i] == 2) { ans += 2 - a[i]; }
  }
  if (ans > 2) { ans = -1; }
  std::cout << ans << std::endl;
  return 0;
}
