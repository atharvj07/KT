import java.util.*;
import java.io.*;

class Main {
    static int n, m, k;
    static int[] x;
    static List[] graph;
    static boolean[] vis;

    static class Pair {
        int num;
        int len;

        public Pair(int a, int b) {
            num = a;
            len = b;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());
        x = new int[k];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < k; ++i)
            x[i] = Integer.parseInt(st.nextToken());
        graph = new ArrayList[n + 1];
        for (int i = 0; i <= n; ++i)
            graph[i] = new ArrayList<Integer>();
        for (int i = 1; i <= m; ++i) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            int w = Integer.parseInt(st.nextToken());
            graph[u].add(new Pair(v, w));
            graph[v].add(new Pair(u, w));
        }
        vis = new boolean[n + 1];
        int[] ans = new int[k];
        Arrays.fill(ans, -1);
        for (int i = 0; i < k; ++i) {
            int stt = x[i];
            Queue<Pair> q = new LinkedList<>();
            q.add(new Pair(stt, 0));
            ans[i] = 0;
            vis = new boolean[n + 1];
            vis[stt] = true;
            while (!q.isEmpty()) {
                Pair h = q.poll();
                ans[i] = Math.max(ans[i], h.len);
                for (int j = 0; j < graph[h.num].size(); ++j) {
                    Pair p = (Pair) graph[h.num].get(j);
                    if (!vis[p.num]) {
                        q.add(new Pair(p.num, Math.max(p.len, h.len)));
                        vis[p.num] = true;
                    }
                }
            }
        }
        vis = new boolean[n + 1];
        for (int i = 0; i < k; ++i) {
            int stt = x[i];
            Queue<Pair> q = new LinkedList<>();
            q.add(new Pair(stt, 0));
            vis = new boolean[n + 1];
            vis[stt] = true;
            while (!q.isEmpty()) {
                Pair h = q.poll();
                ans[i] = Math.max(ans[i], h.len);
                for (int j = 0; j < graph[h.num].size(); ++j) {
                    Pair p = (Pair) graph[h.num].get(j);
                    if (!vis[p.num] && (ans[i] == p.len && ans[i] == ans[j])) {
                        q.add(new Pair(p.num, Math.max(p.len, h.len)));
                        vis[p.num] = true;
                    }
                }
            }
        }
        for (int i = 0; i < k; ++i)
            System.out.print(ans[i] + " ");
    }
}
