import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) h[i] = scan.nextInt();
        int ans = 1;
        int mn = h[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            if (h[i] < mn) {
                ans++;
                mn = h[i];
            }
        }
        System.out.println(ans);
    }
}
