import sys, math

def read():
    return sys.stdin.readline()

def is_prime(n):
    if n < 2: return False
    for i in range(2, int(math.sqrt(n))+1):
        if n % i == 0:
            return False
    return True

n = int(read())
a = list(map(int, read().split()))
a_prime = [i for i in a if is_prime(i)]
m = 0
b = []

for i in range(len(a)):
    if a[i] in a_prime:
        b.append(a[i])
        if len(b) > m: m = len(b)
        for j in range(i+1, len(a)):
            if a[j] in a_prime and a[i] + a[j] not in a_prime:
                b.pop()
                break

print(m)
print(*b)
