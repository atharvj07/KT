#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> last(26, -1);
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int c = s[i] - 'a';
    if (last[c] == -1) {
      ans++;
    }
    last[c] = i;
  }
  cout << ans << endl;
  return 0;
}
