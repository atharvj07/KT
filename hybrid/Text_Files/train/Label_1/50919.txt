import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] min = new int[k];
        for (int i = 0, j = 0; i < n; i++) {
            if (j < k) {
                if (i == 0 || min[j] > a[i]) {
                    min[j] = a[i];
                }
            }
            if (i == n - 1 || a[i] != a[i + 1]) {
                j++;
            }
        }
        int ans = Integer.MIN_VALUE;
        for (int i = 0; i < k; i++) {
            ans = Math.max(ans, min[i]);
        }
        System.out.println(ans);
    }
}
