#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define M 1000000007
ll pw(ll x,ll y)
{
	if(y==0)
		return 1LL;
	if(y==1)
		return x%M;
	ll ans=pw(x,y/2);
	ans=ans*ans;
	ans%=M;
	if(y%2==1) ans=ans*x,ans%=M;
	return ans;
}
int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	ll n,m;
	cin>>n>>m;
	string s;
	cin>>s;
	ll o=0,c=0,ans=1;
	for(int l=0,r=0;l<m;l++)
	{
		if(s[l]=='(') o++;
		else c++;
		if(s[l]=='(' || l==m-1)
		{
			while(r<m && s[r]==')' && c>o)
			{
				c--,r++;
			}
			if(o==c)
			{
				if(l==0 || s[r]!=s[l-1] || l-2>=0 && s[r]==s[l-2])
					ans=ans*2;
				ans%=M;
				o--,c++,l--,r++;
			}
		}
	}
	cout<<(ans-1+M)%M;
}
