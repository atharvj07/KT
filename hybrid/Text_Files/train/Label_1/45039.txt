import java.util.Scanner;

public class SecondPriceAuction {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] prices = new int[n];

        for (int i = 0; i < n; i++) {
            prices[i] = scanner.nextInt();
        }

        int winnerIndex = 0;
        int highestPrice = 0;

        for (int i = 0; i < n; i++) {
            if (prices[i] > prices[winnerIndex]) {
                winnerIndex = i;
                highestPrice = prices[i];
            }
        }

        int secondHighestPrice = 0;
        for (int i = 0; i < n; i++) {
            if (i != winnerIndex && prices[i] > secondHighestPrice) {
                secondHighestPrice = prices[i];
            }
        }

        System.out.println((winnerIndex + 1) + " " + secondHighestPrice);
    }
}
