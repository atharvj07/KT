def solve_next_permutation(n, next_values):
    stack = []
    permutation = [0] * n
    for i in range(n - 1, -1, -1):
        while stack and stack[-1][0] >= next_values[i]:
            stack.pop()
        if next_values[i] == -1 or (stack and stack[-1][1] == next_values[i]):
            permutation[i] = -1
        else:
            permutation[i] = len(stack)
        stack.append((next_values[i], i + 1))
    lowest = 1
    for i in range(n):
        if permutation[i] == -1:
            permutation[i] = lowest
            lowest += 1
        else:
            lowest = max(lowest, permutation[i] + 1)
    if len(set(permutation)) != n:
        return [-1]
    return permutation

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    next_values = list(map(int, input().strip().split()))
    result = solve_next_permutation(n, next_values)
    if result == [-1]:
        print(-1)
    else:
        # Adjust permutation to match the required conditions
        value_to_pos = {v: i for i, v in enumerate(sorted(result))}
        final_perm = [value_to_pos[v] + 1 for v in result]
        print(" ".join(map(str, final_perm)))
