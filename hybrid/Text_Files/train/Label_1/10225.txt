#include <iostream>
#include <vector>

using namespace std;

int ans = 0;
bool vis[2500], p[2500];
vector<int> v;

int main() {
    int n;
    cin >> n;
    p[0] = p[1] = 1;
    for (int i = 2; i <= n; i++) {
        if (!p[i]) {
            v.push_back(i);
            for (int j = i * 2; j <= n; j += i) p[j] = 1;
        }
    }
    for (int i = 0; i < v.size(); i++)
        for (int j = i; j < v.size(); j++)
            for (int k = j; k < v.size(); k++) {
                int x = v[i], y = v[j], z = v[k];
                if (x + y <= z || x ^ y ^ z) continue;
                if ((x + y - z) % 2 == 0 && (x - z + y) % 2 == 0 &&
                    (z - x + y) % 2 == 0)
                    ans++;
            }
    cout << ans << endl;
    return 0;
}
