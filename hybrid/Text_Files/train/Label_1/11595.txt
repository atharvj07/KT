#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int dp[26][26][100006];
long long pre[26];
long long f[26];
int32_t main() {
  int n, k, S;
  cin >> n >> k >> S;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    f[i] = x;
    pre[i + 1] = pre[i] + x;
  }
  dp[0][0][0] = 1;
  for (int i = 0; i <= n; i++) {
    for (int j = 0; j <= n; j++) {
      for (int h = 0; h <= S; h++) {
        if (i) {
          dp[i][j][h] += dp[i - 1][j][h];
          dp[i][j][h] %= mod;
        }
        if (j < k && h >= f[i]) {
          dp[i][j + 1][h] += dp[i][j][h - f[i]];
          dp[i][j + 1][h] %= mod;
        }
      }
    }
  }
  cout << dp[n][k][S] << '\n';
}
