n = int(input())

alarms = []
for _ in range(n):
    time = input().strip()
    hours, minutes = map(int, time.split(":"))
    alarms.append(hours * 60 + minutes)

alarms.sort()

max_sleep = 0
for i in range(1, len(alarms)):
    max_sleep = max(max_sleep, alarms[i] - alarms[i-1] - 1)

# Check the sleep time between the last and first alarm across midnight
max_sleep = max(max_sleep, 1440 - alarms[-1] + alarms[0] - 1)

hours = max_sleep // 60
minutes = max_sleep % 60

print(f"{hours:02d}:{minutes:02d}")
