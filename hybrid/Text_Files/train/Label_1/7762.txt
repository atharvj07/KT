#include <iostream>
#include <cmath>
#include <vector>
#include <set>
using namespace std;

vector<int>v;
set<int>ans;

bool check(int k, int n)
{
    int prv = 0;
    for (int i = 1; i < n; i++)
    {
        int cur = (v[i] - prv + k) % k;
        prv = v[i];
        if (!cur) return false;
    }
    return true;
}

int main()
{
    int n;
    cin >> n;
    v.resize(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    for (int i = 1; i <= n; i++)
    {
        if (check(i, n)) ans.insert(i);
    }
    cout << ans.size() << endl;
    for (auto x : ans) cout << x << " ";
    cout << endl;
    return 0;
}
