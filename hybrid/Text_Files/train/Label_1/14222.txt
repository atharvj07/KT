#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int MOD = 1e9 + 7;
const int MAXN = 1e5 + 5;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<int> dp(n + 1);
    dp[n] = 1;
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == 'w' || s[i] == 'm') {
            dp[i] = (2 * dp[i + 1]) % MOD;
        } else {
            dp[i] = dp[i + 1];
        }
    }
    cout << dp[0] << endl;
    return 0;
}
