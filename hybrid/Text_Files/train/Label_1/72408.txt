from collections import Counter

n, k = map(int, input().split())
a = []
for _ in range(n):
    a.append(int(input()))
cnt = Counter()
for i in a:
    cnt[i] += 1
ans = 0
for i in a:
    if str(i).count('0') == 0 and i <= (10 ** (k + 1)) - 1:
        s = set(str(i))
        t = 0
        for j in s:
            j = int(j)
            if j <= k:
                t += 1
        if t == len(s):
            ans += cnt[i]
print(ans)
