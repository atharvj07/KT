#include <bits/stdc++.h>

using namespace std;
int a[200001], b[200001], t, n, p, ans;

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> t;
        b[i] = t;
    }
    p = 0;
    for (int i = 1; i <= n; i++) {
        if (b[i] == 'P') {
            if (p != 0) ans += min(a[i] - a[p], a[i + 1] - a[i]);
            p = i;
        } else if (b[i] == 'B')
            if (b[i - 1] == 'P') ans += min(a[i] - a[i - 1], a[i + 1] - a[i]);
    }
    p = 0;
    for (int i = 1; i <= n; i++) {
        if (b[i] == 'P') {
            if (p != 0) ans += min(a[i] - a[p], a[i + 1] - a[i]);
            p = i;
        } else if (b[i] == 'R')
            if (b[i - 1] == 'P') ans += min(a[i] - a[i - 1], a[i + 1] - a[i]);
    }
    cout << ans << endl;
}
