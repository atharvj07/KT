import sys

n, k = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
a.sort(reverse=True)
ans = ''
i = 0
while n:
    if i + k >= n:
        ans += '1' * n
        break
    l = max(0, i - k)
    r = min(n, i + k + 1)
    if a[i] > a[l:r]:
        ans += '1' * (r - l)
        n -= r - l
    else:
        ans += '2' * (r - l)
        n -= r - l
    i += k + 1
print(ans)
