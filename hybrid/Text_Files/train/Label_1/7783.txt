#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod=1e9+7;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        ll n;
        cin>>n;
        ll arr[2*n];
        ll sum=0;
        for(ll i=0;i<2*n;i++)
        {
            cin>>arr[i];
            sum+=arr[i];
        }
        if(sum%2!=0)
        cout<<"NO"<<"\n";
        else cout<<"YES"<<"\n";
    }
    return 0;
}
