#include<bits/stdc++.h>
using namespace std;
int n,m,x[1005],y[1005];
int main(){
    cin>>n;
    for(m=1;;m++){
        for(int i=1;i<=n;i++)
            for(int j=1;j<=n;j++)
                if(i+j<=n+1||i-j>=1)
                    x[i]=y[j]=m;
        bool ok=true;
        for(int i=1;i<=n;i++)
            for(int j=i+1;j<=n;j++)
                if(abs(x[i]-x[j])+abs(y[i]-y[j])<abs(i-j))
                    ok=false;
        if(ok)break;
    }
    cout<<m<<endl;
    for(int i=1;i<=n;i++)
        cout<<x[i]<<' '<<y[i]<<endl;
}
