#include <iostream>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long a, b;
        cin >> a >> b;
        if (a == b)
            cout << "YES\n";
        else if (a < b)
            cout << "NO\n";
        else if (a > b && (a - b) % 3 == 0)
            cout << "YES\n";
        else
            cout << "NO\n";
    }
    return 0;
}
