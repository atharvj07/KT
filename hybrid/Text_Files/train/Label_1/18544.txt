#include<bits/stdc++.h>
using namespace std;
#define mod 998244353
int main() {
    int n, m, k;
    cin >> n >> m >> k;
    int vals[n+1];
    vals[0] = 1;
    vals[1] = m;
    for(int i = 2; i <= n; i++) {
        if(i <= k+1) {
            vals[i] = vals[i-1] * 1LL * (m-1) % mod;
        }
        else {
            vals[i] = vals[i-1] * 1LL * m % mod;
        }
    }
    cout << vals[n] << endl;
    return 0;
}
