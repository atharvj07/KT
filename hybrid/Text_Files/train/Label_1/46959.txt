import java.util.LinkedList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static int V, E;
    static List<Integer>[] graph;
    static List<Edge> addedEdges;
    static boolean[] addedEdgeUsed;
    static boolean[] visited;
    static int[] parent;
    static int cnt = 0;

    static class Edge {
        int from, to;

        public Edge(int from, int to) {
            this.from = from;
            this.to = to;
        }
    }

    static void dfs(int u) {
        visited[u] = true;
        for (int v : graph[u]) {
            if (!visited[v]) {
                parent[v] = u;
                dfs(v);
            } else if (parent[u] != v) {
                cnt++;
                if (cnt == 1) {
                    addedEdges.add(new Edge(u, v));
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        V = in.nextInt();
        E = in.nextInt();
        graph = new LinkedList[V + 1];
        for (int i = 0; i <= V; i++) {
            graph[i] = new LinkedList<>();
        }
        for (int i = 0; i < E; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            graph[a].add(b);
            graph[b].add(a);
        }
        addedEdges = new LinkedList<>();
        addedEdgeUsed = new boolean[V + 1];
        visited = new boolean[V + 1];
        parent = new int[V + 1];
        for (int i = 1; i <= V; i++) {
            if (!visited[i]) {
                dfs(i);
            }
        }
        if (cnt > V) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            System.out.println(cnt);
            for (Edge edge : addedEdges) {
                System.out.println(edge.from + " " + edge.to);
            }
        }
    }
}
