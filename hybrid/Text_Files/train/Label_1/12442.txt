#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 10;

vector<int> a(MAXN);
int c[MAXN], res[MAXN];
vector<pair<int, int>> q[MAXN];
int b[MAXN];
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) { cin >> c[i]; }
  for (int i = 1; i <= m; i++) {
    int a, b;
    cin >> a >> b;
    q[a].push_back({b, i});
  }
  priority_queue<pair<int, int>> pq;
  for (int i = 1; i <= n; i++) {
    if (a[i] > 0) { pq.push({c[i], i}); }
  }
  int cur = 1;
  while (pq.size()) {
    pair<int, int> now = pq.top();
    pq.pop();
    int id = now.second;
    while (cur <= m && q[id].size()) {
      int cnt = q[id][0].first, pos = q[id][0].second;
      if (a[id] > 0 && cnt > 0) {
        a[id]--;
        cnt--;
        res[pos] += c[id];
        if (cnt > 0) { q[id][0].first = cnt; } else {
          q[id].erase(q[id].begin());
        }
      } else {
        if (pq.size()) {
          pair<int, int> nw = pq.top();
          if (nw.first == now.first) {
            a[nw.second]--;
            cnt--;
            res[pos] += nw.first;
            pq.pop();
            if (cnt > 0) { q[id][0].first = cnt; } else {
              q[id].erase(q[id].begin());
            }
            if (a[nw.second] > 0) { pq.push(nw); }
          } else {
            break;
          }
        } else {
          break;
        }
      }
      cur++;
    }
    if (a[id] > 0) { pq.push(now); }
  }
  for (int i = 1; i <= m; i++) { cout << res[i] << endl; }
  return 0;
}
