import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();
        int[] hasR = new int[n + 1];
        for (int i = n - 1; i >= 0; i--) {
            if (s.charAt(i) == 'W' || i + k < n && hasR[i + k] == 1) {
                hasR[i] = 1;
            }
        }
        int ans = 0;
        for (int i = 0; i <= n - k; i++) {
            if (s.charAt(i) == 'B') {
                ans = (ans + (hasR[i + k] == 1 ? 1 : 2)) % 1000000007;
            }
        }
        System.out.println(ans);
    }
}
