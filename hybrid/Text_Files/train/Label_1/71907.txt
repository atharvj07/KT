n,m=map(int,input().split())
a=[input() for i in range(n)]
for i in range(1,n):
    if a[i]!=a[i-1]:
        print("NO")
        exit()
c=[0,0,0]
for i in a[0]:
    if i=='R':
        c[0]+=1
    elif i=='G':
        c[1]+=1
    elif i=='B':
        c[2]+=1
if c[0]>0 and c[1]>0 and c[2]>0:
    print("YES")
else:
    print("NO")
