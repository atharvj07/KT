#include<bits/stdc++.h>
#define mo 1000000007
#define int long long
using namespace std;
map<pair<int,int>,int>mp;
int32_t main() {
	int n; cin >> n;
	while (n--) {
		int x, y; cin >> x >> y;
		mp[{x,y}]++;
	}
	int ans = 1;
	for (auto &x : mp) {
		ans = ans * ((x.second * 3ll) % mo) % mo;
	}
	cout << ans;

}
