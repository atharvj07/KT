import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int s = in.nextInt() - 1;
        int t = in.nextInt() - 1;
        int[][] d = new int[n][n];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                d[i][j] = i == j ? 0 : Integer.MAX_VALUE / 2;
        for (int i = 0; i < m; i++) {
            int a = in.nextInt() - 1;
            int b = in.nextInt() - 1;
            d[a][b] = d[b][a] = 1;
        }
        for (int k = 0; k < n; k++)
            for (int i = 0; i < n; i++)
                for (int j = 0; j < n; j++)
                    if (d[i][k] < Integer.MAX_VALUE / 2 && d[k][j] < Integer.MAX_VALUE / 2)
                        d[i][j] = Math.min(d[i][j], d[i][k] + d[k][j]);
        int ans = 0;
        for (int i = 0; i < n; i++)
            for (int j = i + 1; j < n; j++)
                if (d[i][j] == Integer.MAX_VALUE / 2 || d[i][s] + d[s][j] + d[j][t] == d[s][t])
                    ans++;
        System.out.println(ans);
    }
}
