#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mx = 1005;
int a[mx][mx],mn[mx][mx],id[mx][mx];
int main() {
    int n, m;
    cin >> n >> m;
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            cin >> a[i][j];
        }
    }
    int ans = 0;
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            int now = 0;
            for(int k = 1; k <= m; k++) {
                if(a[i][j] > a[i][k]) id[i][j] |= (1 << (k - 1));
                if(a[i][j] < a[i][k]) id[i][j] |= (1 << (m + k - 1));
                if(a[j][i] > a[k][i]) id[j][i] |= (1 << (k - 1));
                if(a[j][i] < a[k][i]) id[j][i] |= (1 << (m + k - 1));
                now = max(now,a[i][k]);
                now = max(now,a[k][i]);
            }
            mn[i][j] = now;
        }
    }
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) cout << mn[i][j] << " ";
        cout << endl;
    }
}
