#include <cmath>
#include <cstdio>
#include <vector>
#include <iostream>
#include <map>

using namespace std;

const int MOD = 1e9 + 7;
const int MAX_STR = 1e5 + 5;
const int MAX_SUM = 505;

#define pb push_back
#define ll long long

int main() {
  int n;
  cin >> n;

  vector<string> s(n);
  for (int i = 0; i < n; i++) {
    cin >> s[i];
  }

  map<int, vector<int>> hm;

  for (int i = 0; i < n; i++) {
    int h = 0;
    for (char ch : s[i]) {
      h += (1ll * (ch - 'a') * (i + 1)) % MOD;
      h %= MOD;
    }
    hm[h].pb(i);
  }

  ll ans = 0;

  for (auto it : hm) {
    vector<int> arr = it.second;

    for (int i = 0; i < int(arr.size()); i++) {
      for (int j = i; j < int(arr.size()); j++) {
        if (i != j || s[arr[i]].size() % 2 == 1) {
          string t = s[arr[i]] + s[arr[j]];
          bool palindrome = 1;
          for (int k = 0, l = int(t.size()) - 1; k < l; k++, l--) {
            if (t[k] != t[l]) {
              palindrome = 0;
              break;
            }
          }
          ans += palindrome;
        }
      }
    }
  }

  cout << ans << endl;

  return 0;
}
