import heapq
n = int(input())
a = []
ans = []
for i in range(n):
    x = input().split()
    a.append((x[0], int(x[1])))
ans.append(a[0])
a.pop(0)
for i in a:
    if i[0] == 'insert':
        ans.append(i)
        heapq.heappush(ans, i[1])
    elif i[0] == 'removeMin':
        heapq.heappop(ans)
        ans.append(i)
    else:
        x = heapq.heappop(ans)
        ans.append(i)
        if x != i[1]:
            heapq.heappush(ans, i[1])
            ans.append(('insert', x))
print(len(ans))
for i in ans:
    print(*i)
