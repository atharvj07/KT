#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  if (k * (n - 1) % n != 0) {
    cout << -1 << endl;
    return 0;
  }
  int m = k * (n - 1);
  cout << m << endl;
  int p1 = 1;
  int p2 = 2;
  for (int i = 1; i <= m; i++) {
    if (i % (n - 1) == 0) {
      p2++;
      if (p2 > n) p2 = 1;
      if (p1 == p2) p2++;
      if (p2 > n) p2 = 1;
    }
    cout << p1 << " " << p2 << endl;
    p1++;
    if (p1 > n) p1 = 1;
    if (p1 == p2) p1++;
    if (p1 > n) p1 = 1;
  }
  return 0;
}
