import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextInt();
        long m = sc.nextInt();
        BigInteger ans = BigInteger.ONE.shiftLeft((int) (n - m + 1)).subtract(m == 1 ? BigInteger.ONE : BigInteger.ONE.shiftLeft((int) m));
        ans = ans.mod(BigInteger.valueOf(1000000007L));
        System.out.println(ans);
    }
}
