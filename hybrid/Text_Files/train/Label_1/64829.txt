import sys
mod = 998244353
n = int(sys.stdin.readline())
a = [int(x) for x in sys.stdin.readline().split()]
l = len(str(max(a)))
a = [str(x).zfill(l) for x in a]
ans = 0
for i in range(l):
    d = []
    for j in range(n):
        d += a[j][i]
    for j in range(n):
        for k in range(n):
            f = d[j::n]+d[k::n]
            ans += int(''.join(f))
ans %= mod
print(ans)
