import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int q = in.nextInt();

        for (int i = 0; i < q; ++i) {
            int n = in.nextInt();
            int[] prices = new int[n];
            int sum = 0;

            for (int j = 0; j < n; ++j) {
                prices[j] = in.nextInt();
                sum += prices[j];
            }

            int minPrice = (int) Math.ceil((double) sum / n);
            System.out.println(minPrice);
        }
    }
}
