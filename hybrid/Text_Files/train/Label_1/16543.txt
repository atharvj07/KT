#include <iostream>
#include <vector>
using namespace std;

vector<int> a, s;
int main() {
  int n;
  cin >> n;
  a.resize(n + 1);
  s.resize(n, 0);
  for (int i = 2; i <= n; ++i) cin >> a[i];
  for (int i = n - 2; i >= 0; --i) s[i] = s[i + 1] + a[i + 2];
  for (int i = 1; i <= n - 1; ++i) {
    if (s[i - 1] <= 0 || s[i - 1] > n - i)
      cout << -1 << endl;
    else
      cout << s[i - 1] + 2 * a[i] << endl;
  }
}
