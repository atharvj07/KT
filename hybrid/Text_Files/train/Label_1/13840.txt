#include <iostream>
#include <string>
using namespace std;
int main() {
  int t, ans = 0;
  string s;
  cin >> t >> s;
  for (char &i : s) {
    if (t == stoi(string(1, i)))
      ++ans;
    if (t == stoi(string(1, i)) + 1 || t == stoi(string(1, i)) - 1)
      if (i != '6' && i != '9' && i != '5' && i != '2')
        ++ans;
  }
  cout << ans << endl;
}
