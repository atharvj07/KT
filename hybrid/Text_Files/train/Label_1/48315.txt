import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] winners = new int[n];

        for (int i = 0; i < n; i++) {
            int[] powers = new int[k];
            for (int j = 0; j < k; j++) {
                powers[j] = scanner.nextInt();
            }
            Arrays.sort(powers);
            int maxPower = powers[k - 1];
            int count = 0;
            for (int j = 0; j < k; j++) {
                if (powers[j] == maxPower) {
                    count++;
                }
            }
            winners[i] = count;
        }

        for (int i = 0; i < n; i++) {
            System.out.println(winners[i]);
        }
    }
}
