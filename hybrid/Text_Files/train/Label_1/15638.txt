#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;

ll fastpow(ll x, ll p) {
  ll ans = 1;
  while (p) {
    if (p % 2) ans = (ans * x) % mod;
    x = (x * x) % mod;
    p >>= 1;
  }
  return ans;
}

int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> a(n);
  ll sum = 0;
  for (auto &x : a) cin >> x, sum += (x == 0);
  if (k < sum || (n - 1 - sum) * 2 + k < n - 1) {
    cout << 0 << endl;
    return 0;
  }

  ll p = 0, q = 1, inv = fastpow(2, mod - 2);
  for (ll i = 0; i < sum; i++) p += (i + 1) * inv % mod, q = (q * inv) % mod;
  for (ll i = n - 1 - sum; i >= 0; i--) p += (n - i) * inv % mod, q = (q * inv) % mod;
  cout << (p % mod + mod) * fastpow(q, mod - 2) % mod << endl;
  return 0;
}
