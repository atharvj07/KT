n, m, k = map(int, input().split())
alarms = list(map(int, input().split()))

alarms.sort()
diffs = [alarms[i] - alarms[i-1] for i in range(1, n)]
prefix_sums = [0] * (n-1)
prefix_sums[0] = diffs[0]

for i in range(1, n-1):
    prefix_sums[i] = prefix_sums[i-1] + diffs[i]

ans = n
for i in range(n-k+1):
    if i == 0:
        total_time = prefix_sums[k-2] if k-2 >= 0 else 0
    else:
        total_time = prefix_sums[i+k-2] - prefix_sums[i-1]
    if total_time < m:
        ans = min(ans, k-1)

print(ans)
