#include <bits/stdc++.h>
using namespace std;
#define ll long long
pair<ll, ll> fun(int a, int b) {
  if (a < 1 || b < 1 || (!a && b == 1) || (!b && a == 1)) {
    return {-1, -1};
  }
  if (a == 10) a--;
  if (b == 10) b--;
  ll x = (ll)a * 10 + b;
  ll y = x + 1;
  return {x, y};
}
int main() {
  int da, db;
  cin >> da >> db;
  auto [a, b] = fun(da, db);
  if (a == -1 || b == -1) {
    cout << -1 << endl;
  } else {
    cout << a << " " << b << endl;
  }
  return 0;
}
