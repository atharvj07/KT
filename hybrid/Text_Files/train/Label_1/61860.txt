import sys

n = int(input())
adj = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

cnt = [0] * (n + 1)

def dfs(v, p):
    for u in adj[v]:
        if u == p: continue
        cnt[v] += 1
        dfs(u, v)

dfs(1, 1)
ans = n - 1 - max(cnt)
print(ans)
