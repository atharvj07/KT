import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        Scanner sc = new Scanner(br);

        int s = sc.nextInt();
        int b = sc.nextInt();

        int[] a = new int[s];
        int[] d = new int[b];
        long[] g = new long[b];
        long[] ans = new long[s];

        for (int i = 0; i < s; i++) {
            a[i] = sc.nextInt();
        }

        Arrays.sort(a);

        for (int i = 0; i < b; i++) {
            d[i] = sc.nextInt();
            g[i] = sc.nextInt();
        }

        Arrays.sort(d);
        Arrays.sort(g);

        int ind = 0;
        for (int i = 0; i < s; i++) {
            while (ind < b && d[ind] <= a[i]) {
                ans[i] += g[ind++];
            }
        }

        for (long l : ans) {
            System.out.println(l);
        }
    }
}
