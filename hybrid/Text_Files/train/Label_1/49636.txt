import java.util.Scanner;
import java.util.TreeMap;

public class ProgrammingContest {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt(); // Number of teams
        int R = scanner.nextInt(); // Number of records
        int L = scanner.nextInt(); // Length of the contest

        // Create a TreeMap to store team ID and their on-screen time
        TreeMap<Integer, Long> teamOnScreenTime = new TreeMap<>();

        // Process each record in the log
        for (int i = 0; i < R; i++) {
            int teamId = scanner.nextInt();
            int time = scanner.nextInt();
            int scoreChange = scanner.nextInt();

            // Update the on-screen time for the team
            if (!teamOnScreenTime.containsKey(teamId)) {
                teamOnScreenTime.put(teamId, 0L);
            }
            teamOnScreenTime.put(teamId, teamOnScreenTime.get(teamId) + (long) time);

            // Remove any team with a lower score from the map
            for (int j = 1; j <= N; j++) {
                if (j != teamId && teamOnScreenTime.containsKey(j)) {
                    if (scoreChange > 0) {
                        teamOnScreenTime.remove(j);
                    }
                }
            }
        }

        // Find the team with the longest on-screen time
        long maxOnScreenTime = 0;
        int teamWithMaxOnScreenTime = 0;
        for (int teamId : teamOnScreenTime.keySet()) {
            long onScreenTime = teamOnScreenTime.get(teamId);
            if (onScreenTime > maxOnScreenTime) {
                maxOnScreenTime = onScreenTime;
                teamWithMaxOnScreenTime = teamId;
            }
        }

        // Output the team ID with the longest on-screen time
        System.out.println(teamWithMaxOnScreenTime);
    }
}
