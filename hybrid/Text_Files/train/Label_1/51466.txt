import java.util.HashMap;
import java.util.HashSet;
import java.util.Scanner;

public class ForbiddenPairs {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String sentence = scanner.nextLine();
        int forbiddenPairsCount = scanner.nextInt();
        scanner.nextLine(); // Consume the newline after the count

        HashMap<Character, Character> forbiddenPairs = new HashMap<>();
        for (int i = 0; i < forbiddenPairsCount; i++) {
            String pair = scanner.nextLine();
            forbiddenPairs.put(pair.charAt(0), pair.charAt(1));
        }

        HashSet<Character> removedLetters = new HashSet<>();
        int deletions = 0;

        for (int i = 0; i < sentence.length() - 1; i++) {
            char current = sentence.charAt(i);
            char next = sentence.charAt(i + 1);

            if (forbiddenPairs.containsKey(current) && forbiddenPairs.get(current) == next) {
                if (!removedLetters.contains(current)) {
                    deletions++;
                    removedLetters.add(current);
                }
            } else if (forbiddenPairs.containsKey(next) && forbiddenPairs.get(next) == current) {
                if (!removedLetters.contains(next)) {
                    deletions++;
                    removedLetters.add(next);
                }
            }
        }

        System.out.println(deletions);
    }
}
