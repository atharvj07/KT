import sys

n, m, d = list(map(int, input().split()))
c = list(map(int, input().split()))
a = [0 for i in range(n)]
x = [0] * m
y = [0] * m
for i in range(m):
    for j in range(c[i]):
        a[x[i] + j] = i + 1
    y[i] = x[i] + c[i] - 1
    x[i] += d
if x[m - 1] < n+1:
    print("NO")
    sys.exit()
print("YES")
print(*a)
