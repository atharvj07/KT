s=input()
t=input()
if len(s)!=len(t):
    print(0)
else:
    z=[0]*26
    o=[0]*26
    for i in s:
        if i=='0':
            z[ord(t[s.index(i)])-ord('a')]+=1
        else:
            o[ord(t[s.index(i)])-ord('a')]+=1
    ans=0
    for i in range(26):
         for j in range(26):
             if i!=j and z[i]!=0 and o[j]!=0:
                 ans+=1
    print(ans)
