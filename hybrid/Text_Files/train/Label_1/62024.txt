import sys
input = sys.stdin.readline

def f(x: int) -> int:
    ans = 0
    while x:
        ans = 10 * ans + x % 10
        x //= 10
    while ans % 10 == 0:
        ans //= 10
    return ans

def g(x: int) -> int:
    return x // f(f(x))

d = {}
for _ in range(int(input())):
    n = int(input())
    i = 1
    while n >= i:
        d[g(i)] = 1
        i = 10 * i + 1
    print(len(d))
