#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.length();
    vector<int> pref(n, 0), suff(n, 0);
    for (int i = 0; i < n; i++) {
      if (i > 0) { pref[i] = pref[i - 1]; }
      if (s[i] == '1') { pref[i]++; }
    }
    for (int i = n - 1; i >= 0; i--) {
      if (i < n - 1) { suff[i] = suff[i + 1]; }
      if (s[i] == '0') { suff[i]++; }
    }
    int ans = n;
    for (int i = 0; i < n; i++) {
      ans = min(ans, pref[i] + suff[i] - (s[i] == '0'));
    }
    cout << ans << endl;
  }
  return 0;
}
