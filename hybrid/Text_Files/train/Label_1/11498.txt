#include <iostream>
using namespace std;
int main() {
  long long a, b, c, d, ans = 0;
  cin >> a >> b >> c >> d;
  for (long long x = a; x <= b; x++)
    for (long long y = b; y <= c; y++)
      for (long long z = c; z <= d; z++)
        if (x < y && x < z && y < z) ans++;
  cout << ans << endl;
}
