def max_points(m, test_cases):
    results = []
    for case in test_cases:
        n, T, a, b, types, times = case
        events = sorted([(times[i], types[i]) for i in range(n)] + [(T+1, 0)])
        easy, hard = types.count(0), types.count(1)
        points, max_points, time_spent = 0, 0, 0
        
        for time, type in events:
            if time_spent < time:
                can_solve_easy = min(easy, (time - 1 - time_spent) // a)
                can_solve_hard = min(hard, (time - 1 - time_spent - can_solve_easy * a) // b)
                
                max_points = max(max_points, points + can_solve_easy + can_solve_hard)
                
                if type == 0:
                    if easy == 0: break
                    easy -= 1
                    time_spent += a
                else:
                    if hard == 0: break
                    hard -= 1
                    time_spent += b
                points += 1
            else:
                if type == 0:
                    easy -= 1
                else:
                    hard -= 1
                points += 1
        
        results.append(max_points)
    
    return results

# Input parsing
m = int(input().strip())
test_cases = []
for _ in range(m):
    n, T, a, b = map(int, input().strip().split())
    types = list(map(int, input().strip().split()))
    times = list(map(int, input().strip().split()))
    test_cases.append((n, T, a, b, types, times))

# Calculating and printing results
results = max_points(m, test_cases)
for result in results:
    print(result)
