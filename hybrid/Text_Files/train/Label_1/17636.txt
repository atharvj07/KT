#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];

  int k = 0, t = 0;
  vector<bool> marked(n, 0);
  for (int i = 0; i < n; i++) {
    if (!marked[i]) {
      k++;
      int cur = b[i];
      int curT = 0;
      while (true) {
        bool found = false;
        for (int j = i + 1; j < n; j++) {
          if (!marked[j] && cur >= a[j]) {
            found = true;
            cur += a[j] - cur;
            curT += a[j];
            marked[j] = true;
          }
        }
        if (!found) break;
      }
      t += curT;
    }
  }
  cout << k << " " << t << endl;
  return 0;
}
