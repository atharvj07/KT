#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e5+10;
int n;
int b[MAXN], c[MAXN], b2[MAXN], c2[MAXN], a[MAXN];
int main() {
  cin >> n;
  for(int i = 1; i < n; i++) {
    cin >> b[i];
  }
  for(int i = 1; i < n; i++) {
    cin >> c[i];
  }
  b2[1] = b[1];
  for(int i = 1; i < n-1; i++) {
    int x = min(b[i], c[i]), y = max(b[i], c[i]);
    if(x != b[i+1] || y != c[i+1]) {
      cout << -1 << endl;
      return 0;
    }
    b2[i+1] = x, c2[i+1] = y;
  }
  for(int i = 1; i < n; i++) {
    if(b[i] != b2[i] || c[i] != c2[i]) {
      cout << -1 << endl;
      return 0;
    }
  }
  a[1] = b[1];
  for(int i = 2; i <= n; i++) {
    if(a[i-1] < b[i]) a[i] = a[i-1]+1;
    else a[i] = a[i-1]-1;
  }
  for(int i = 1; i <= n; i++) {
    cout << a[i] << " ";
  }
  return 0;
}
