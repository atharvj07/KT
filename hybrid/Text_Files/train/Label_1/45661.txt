import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int num = (k % n == 0) ? (k / n) * m : (k / n) * m + (k % n);
        System.out.println(num);
    }
}
