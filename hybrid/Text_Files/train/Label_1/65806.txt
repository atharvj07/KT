n, m = map(int, input().split())
adj = [[] for i in range(n+1)]
for i in range(m):
  a, b, c = input().split()
  adj[int(a)].append((int(b), c))
  adj[int(b)].append((int(a), c))
vis = [0] * (n+1)
ans = []
def dfs(node = 1, col = 'R', prv = -1):
  vis[node] = 1
  ans.append(node)
  for u, c in adj[node]:
    if u!=prv and (vis[u]==0 or (col!=c and vis[u]==1)):
      if vis[u]==0:
        dfs(u, c, node)
      else:
        if col=='R':
          ans.append(u)
          dfs(u, 'B', node)
        else:
          ans.append(u)
          dfs(u, 'R', node)
dfs()
if len(ans)==n:
  print(len(set(ans))//2)
  print(*ans)
else:
  print(-1)
