t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    l = list(map(int, input().split()))
    unlocked_values = [a[i] for i in range(n) if l[i] == 0]
    unlocked_values.sort(reverse=True)
    j = 0
    for i in range(n):
        if l[i] == 0:
            a[i] = unlocked_values[j]
            j += 1
    print(*a)
