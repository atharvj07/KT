import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] adj = new int[n + 1][n + 1];
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt(), v = sc.nextInt();
            adj[u][v] = adj[v][u] = 1;
        }
        int[] degree = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                degree[i] += adj[i][j];
            }
        }
        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            if (degree[i] == 1) {
                cnt++;
            }
        }
        if (cnt > 2) {
            System.out.println(-1);
        } else {
            System.out.println(cnt - 1);
        }
    }
}
