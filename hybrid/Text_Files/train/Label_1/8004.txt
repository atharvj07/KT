#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    if (k == 1) {
      if (a[0] > 0) {
        cout << -1 << endl;
      } else {
        cout << 1 << endl;
      }
    } else {
      int ans = (a[n - 1] + k - 1) / k;
      cout << ans << endl;
    }
  }
  return 0;
}
