import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    String s = sc.next();
    int cntR = 0;
    for (char c : s.toCharArray()) {
      if (c == 'R') {
        cntR++;
      }
    }
    System.out.println(cntR > n - cntR ? "Yes" : "No");
  }
}
