import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long k = sc.nextLong();
        if (k > b) {
            System.out.println(0);
            return;
        }
        if (a % k == 0) {
            System.out.println(((b - a) / k) + 1);
            return;
        }
        long ans = ((b - a) / k);
        long inc = k - (a % k);
        System.out.println(ans + inc);
    }
}
