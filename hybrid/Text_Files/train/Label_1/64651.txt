import bisect

n, y1 = [int(c) for c in input().split()]
a = sorted([int(c) for c in input().split()])

m, y2 = [int(c) for c in input().split()]
b = sorted([int(c) for c in input().split()])

ymax = y2 - y1
xmax = max(max(*a) - min(*b), max(*b) - min(*a))

ans = 0
for x in a:
    if -x <= xmax and x <= xmax:
        ans += 1
        
for x in b:
    if -x <= xmax and x <= xmax:
        ans += 1

print(ans)
