#include <bits/stdc++.h>
using namespace std;
long long n, s, a[100001], b[100001], c[100001], ans1[100001], ans2[100001], anss1[100001], anss2[100001];
int main() {
    cin >> n >> s;
    for (int i = 1; i <= n; i++) cin >> a[i], cin >> b[i], cin >> c[i];
    for (int i = s; i >= 1; i--) { ans1[i] = max(ans1[i + 1], anss1[i + 1]); }
    for (int i = s; i >= 1; i--) {
        anss1[i] = anss1[i + 1];
        for (int j = 1; j <= n; j++) {
            if (i >= a[j]) anss1[i] = max(anss1[i], ans1[i - a[j]] + b[j] * c[j]);
            if (i >= b[j]) anss1[i] = max(anss1[i], ans1[i - b[j]] + a[j] * c[j]);
        }
    }
    for (int i = s; i >= 1; i--) { ans2[i] = max(ans2[i + 1], anss2[i + 1]); }
    for (int i = s; i >= 1; i--) {
        anss2[i] = anss2[i + 1];
        for (int j = 1; j <= n; j++) {
            if (i >= a[j]) anss2[i] = max(anss2[i], ans2[i - a[j]] + b[j] * c[j]);
            if (i >= b[j]) anss2[i] = max(anss2[i], ans2[i - b[j]] + a[j] * c[j]);
        }
    }
    cout << max(anss1[1], anss2[1]);
    return 0;
}
