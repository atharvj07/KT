#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        if (n % 2) {
            int ans = (n - 1) / 2 + (n - 1) / 2;
            cout << ans << endl;
        } else {
            int ans = n / 2 + n / 2 - 1;
            cout << ans << endl;
        }
    }
    return 0;
}
