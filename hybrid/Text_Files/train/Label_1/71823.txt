from collections import deque
import heapq

n, d = map(int, input().split())
cond = [None] * d
for i in range(d):
    x, y = map(int, input().split())
    cond[i] = (x, y)

g = [set() for _ in range(n + 1)]
for x, y in cond:
    g[x].add(y)
    g[y].add(x)

deg = [0] * (n + 1)
for x in g:
    deg[len(x)] += 1

ans = [1] * d
que = deque()
for i in range(1, n + 1):
    if len(g[i]) == 1:
        que.append(i)

for i in range(d):
    while que:
        x = que.popleft()
        for y in g[x]:
            g[y].remove(x)
            deg[len(g[y])] -= 1
            if deg[len(g[y])] == 1:
                que.append(y)
    for x in g:
        if x:
            ans[i] = max(ans[i], len(x))

for x in ans:
    print(x)
