import collections

def solve(n, k):
    ns = dict()
    ns[0] = collections.deque()
    
    for i in range(n):
        id = int(input())
        if id in ns:
            ns[id].appendleft(ns[id].popleft())
        else:
            if len(ns) < k:
                ns[id] = collections.deque([id])
            else:
                ns.popitem()
                ns[id] = collections.deque([id])
    
    print(len(ns))
    print(' '.join(map(str, sorted(ns))))

n, k = map(int, input().split())
solve(n, k)
