import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int x0 = in.nextInt();
            int y0 = in.nextInt();
            int x1 = in.nextInt();
            int y1 = in.nextInt();
            int x2 = in.nextInt();
            int y2 = in.nextInt();
            int x3 = in.nextInt();
            int y3 = in.nextInt();
            if ((x1 == x0 && y1 == y0) || (x2 == x3 && y2 == y3)) {
                System.out.println(0);
            } else {
                double a1, b1, c1, a2, b2, c2;
                a1 = y1 - y0;
                b1 = x0 - x1;
                c1 = a1 * x0 + b1 * y0;
                a2 = y3 - y2;
                b2 = x2 - x3;
                c2 = a2 * x2 + b2 * y2;

                if (a1 * b2 - a2 * b1 != 0) {
                    double d = a1 * b2 - a2 * b1;
                    double dx = c1 * b2 - c2 * b1;
                    double dy = a1 * c2 - a2 * c1;
                    if (dx / d >= 0 && dx / d <= 1 && dy / d >= 0 && dy / d <= 1) {
                        System.out.println(1);
                    } else {
                        System.out.println(0);
                    }
                } else {
                    if (c1 == c2) {
                        System.out.println(1);
                    } else {
                        System.out.println(0);
                    }
                }
            }
        }
    }
}
