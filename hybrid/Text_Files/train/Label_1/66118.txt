import sys

in_stream = sys.stdin
out_stream = sys.stdout

n, d, p = map(int, in_stream.readline().split())

f = {}
for _ in range(d):
    x, y = map(int, in_stream.readline().split())
    if x not in f:
        f[x] = set()
    if y not in f:
        f[y] = set()
    f[x].add(y)
    f[y].add(x)

out_stream.write("0" * (d-1))
for i in range(d-1, len(f)):
    vis = [0 for _ in range(n + 1)]

    cur = []
    mx = 0
    for x in f:
        if vis[x] == 0:
            cur.append(x)
            mx += 1
    out_stream.write(str(mx) + "\n")

    while cur:
        y = cur.pop()
        vis[y] = 1
        for z in f[y]:
            if vis[z] == 0:
                cur.append(z)
