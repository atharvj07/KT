def check(field, n, m):
    for i in range(n):
        for j in range(m):
            if field[i][j] == '.':
                continue
            if field[i][j] == '*':
                if i > 0 and field[i - 1][j] != '*':
                    field[i - 1][j] = int(field[i - 1][j]) + 1
                if i < n - 1 and field[i + 1][j] != '*':
                    field[i + 1][j] = int(field[i + 1][j]) + 1
                if j > 0 and field[i][j - 1] != '*':
                    field[i][j - 1] = int(field[i][j - 1]) + 1
                if j < m - 1 and field[i][j + 1] != '*':
                    field[i][j + 1] = int(field[i][j + 1]) + 1
                if i > 0 and j > 0 and field[i - 1][j - 1] != '*':
                    field[i - 1][j - 1] = int(field[i - 1][j - 1]) + 1
                if i > 0 and j < m - 1 and field[i - 1][j + 1] != '*':
                    field[i - 1][j + 1] = int(field[i - 1][j + 1]) + 1
                if i < n - 1 and j > 0 and field[i + 1][j - 1] != '*':
                    field[i + 1][j - 1] = int(field[i + 1][j - 1]) + 1
                if i < n - 1 and j < m - 1 and field[i + 1][j + 1] != '*':
                    field[i + 1][j + 1] = int(field[i + 1][j + 1]) + 1
    for i in range(n):
        for j in range(m):
            if field[i][j] != '.':
                if field[i][j] != int(field[i][j]):
                    return 'NO'
    return 'YES'

n, m = map(int, input().split())
field = []
for i in range(n):
    field.append(list(input()))
print(check(field, n, m))
