import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[] deg = new int[n+1];
        for (int i = 0; i < m; i++) {
            deg[in.nextInt()]++;
            deg[in.nextInt()]++;
        }
        Arrays.sort(deg);
        int ans = 0;
        for (int i = n; i >= 1; i--) {
            if (deg[i] > 0) {
                ans += deg[i] - 1;
            }
        }
        System.out.println(ans);
    }
}
