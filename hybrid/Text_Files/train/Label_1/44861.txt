import java.util.*;
import java.io.*;

public class Main {
    static int[] p = new int[100000];
    static void sieve() {
        Arrays.fill(p, 1);
        p[0] = p[1] = 0;
        for (int i = 2; i < 100000; i++) {
            if (p[i] == 1) {
                for (int j = i + i; j < 100000; j += i) p[j] = 0;
            }
        }
    }

    public static void main(String[] args) throws IOException {
        sieve();
        InputStreamReader isr = new InputStreamReader(System.in);
        BufferedReader br = new BufferedReader(isr);
        StringTokenizer st = new StringTokenizer(br.readLine());
        int foxes = Integer.parseInt(st.nextToken());
        st = new StringTokenizer(br.readLine());
        int[] ages = new int[foxes];
        int[] agecnt = new int[100000];
        for (int i = 0; i < foxes; i++) {
            ages[i] = Integer.parseInt(st.nextToken());
            agecnt[ages[i]]++;
        }
        boolean[] used = new boolean[100000];
        int[] ans = new int[foxes]; int ansIdx = 0;
        for (int i = 99999; i >= 2; i--) {
            if (p[i] == 1 && agecnt[i] > 1) {
                used[i] = true;
                ans[ansIdx++] = i; agecnt[i] -= 2;
                for (int j = i+i; j < 100000; j += i) {
                    if (p[j] == 1 && agecnt[j] > 0) {
                        used[j] = true;
                        ans[ansIdx++] = j; agecnt[j]--;
                    }
                }
            }
        }
        int rem = 0;
        int[] remCnt = new int[100000];
        for (int i = 0; i < 100000; i++) {
            remCnt[i] = agecnt[i];
            rem += agecnt[i];
        }
        ArrayList<Integer> sol = new ArrayList<Integer>();
        ArrayList<Integer> acc = new ArrayList<Integer>();
        boolean found = true;
        while (rem > 0) {
            int prv = -1;
            int prv2 = -1;
            sol.clear();
            acc.clear();
            for (int i = 2; i < 100000; i++) {
                if (remCnt[i] > 0) {
                    remCnt[i]--;
                    sol.add(i);
                    acc.add(i);
                    if (p[i + prv2] != 1) {
                        remCnt[i]--;
                        sol.add(i);
                        acc.add(i);
                    }
                    rem -= 2;
                    remCnt[i]--;
                    prv2 = prv;
                    prv = i;
                    if (remCnt[i] == 0) break;
                }
            }
            if (sol.size() == 0) {
                found = false;
                break;
            }
            int x = sol.get(sol.size()-1);
            sol.add(x);
            if (p[acc.get(0) + x] == 1) {
                acc.add(0, x);
            } else {
                acc.add(x);
            }
            rem -= sol.size() - 3;
            remCnt[acc.get(0)]--;
            remCnt[acc.get(acc.size()-1)]--;
            for (int age : sol) {
                remCnt[age] -= 1;
                if (remCnt[age] == 0) used[age] = false;
            }
        }
        if (!found) System.out.println("Impossible");
        else {
            int cnt = (ansIdx + sol.size()) / 3;
            System.out.println(cnt);
            int start = 0;
            for (int i = 0; i < cnt; i++) {
                System.out.print(3 + " ");
                int end = start + 3;
                if (ansIdx > start) {
                    for (int j = ansIdx - 1; j >= start; j--) {
                        System.out.print(ans[j] + " ");
                        end++;
                    }
                    ansIdx = start;
                }
                for (int j = start; j < end; j++) {
                    System.out.print(sol.get(j) + " ");
                }
                start = end;
                System.out.println();
            }
        }
        br.close();
        isr.close();
    }
}
