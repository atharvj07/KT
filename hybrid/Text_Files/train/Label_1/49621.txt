            System.out.println("2-type");
            System.out.println(d % 2 == 0 ? Math.min(2, b) : d % 5 == 0 ? 1 : 0);
        } else if (d % 3 == 0 || d % 9 == 0) {
            System.out.println("3-type");

            System.out.println("11-type");
        } else if (d % 6 == 0 && b % 2 == 0) {

        }
