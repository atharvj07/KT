#include<bits/stdc++.h>
#define f first
#define s second
#define ll long long
#define all(x) x.begin(), x.end()
#define rall(x) x.rbegin(),x.rend()
using namespace std;
bool check(ll x, ll y){
    ll t = x;
    while(t){
        if(t%10 == y)return 1;
        t/=10;
    }
    return 0;
}
int main() {
    ll n,m,s,f;
    cin >> n >> m >> s >> f;
    ll t[m+1], l[m+1], r[m+1];
    vector<ll>v;
    for(ll i=1; i<=m; i++)
    {
        cin >> t[i] >> l[i] >> r[i];
        v.push_back(t[i]);
        v.push_back(t[i]-1);
    }
    map<ll,ll>mp,mp1;
    for(ll i=1; i<=m; i++){
        if(l[i]==r[i]){
            mp[l[i]]=1;
            mp1[t[i]]=1;
            mp1[t[i]-1]=1;
        }
    }
    v.push_back(1e9);
    sort(v.begin(),v.end());
    ll t1=(lower_bound(v.begin(),v.end(),1)-v.begin());

    string ans;
    ll x=s;
    map<ll,ll> vis;
    ll idx=0;
    while(x!=f){
        if(idx == t1 and check(t[idx],x)){
            mp[x]=1;
        }
        if(vis[x]==0){
            vis[x]=1;
            if(mp[x]==0){
                if(x>f and mp[x-1]==0)
                    x--,ans.push_back('L');
                else if(x<f and mp[x+1]==0)
                    x++,ans.push_back('R');
                else{
                    ans=' ';
                    break;
                }
            }
            else{
                while(mp[x]==1 and x!=f){
                    if(x>f)x--,ans.push_back('L');
                    else x++,ans.push_back('R');
                }
                if(idx != t1)
                    idx++;
                else if(idx == t1){
                    ll now=0;
                    for(ll j=idx; j<=m; j++){
                        if(l[j]==r[j] and t[j]>=v[idx] and t[j]<v[idx+1]){
                            now=1;
                            idx++;
                            break;
                        }
                    }
                    if(now==0){
                        ans=' ';
                        break;
                    }
                }
            }
        }
        else{
            ans = ' ';
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
