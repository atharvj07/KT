import sys

n = int(input())
s = input()
a = [int(x) for x in s if x != '?']
b = [int(x) if x != '?' else 0 for x in s]

if len(a) == n:
    if sum(a[:len(a) // 2]) == sum(a[len(a) // 2:]):
        print('Bicarp')
    else:
        print('Monocarp')
else:
    if sum(a[:len(a) // 2]) + (n - len(a)) // 2 * 9 <= sum(a[len(a) // 2:]):
        print('Bicarp')
    else:
        print('Monocarp')
