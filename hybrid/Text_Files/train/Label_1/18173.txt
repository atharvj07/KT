#include <bits/stdc++.h>

#define ll long long

using namespace std;

const int MAX_N = 1e5 + 5;

vector<int> primes;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    if (m > n * (n - 1) / 2) {
        cout << "Impossible" << endl;
        return 0;
    }

    cout << "Possible" << endl;

    // find all primes less than n
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            primes.push_back(i);
            while (n % i == 0) {
                n /= i;
            }
        }
    }

    if (n > 1) {
        primes.push_back(n);
    }

    // generate a relatively prime graph
    vector<vector<int>> adj(n + 1);
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            bool isRelativelyPrime = true;
            for (int prime : primes) {
                if (i % prime == 0 && j % prime == 0) {
                    isRelativelyPrime = false;
                    break;
                }
            }

            if (isRelativelyPrime) {
                adj[i].push_back(j);
                adj[j].push_back(i);
                m--;
                if (m == 0) {
                    break;
                }
            }
        }
        if (m == 0) {
            break;
        }
    }

    // output the graph
    for (int i = 1; i <= n; i++) {
        for (int j : adj[i]) {
            if (i < j) {
                cout << i << ' ' << j << endl;
            }
        }
    }

    return 0;
}
