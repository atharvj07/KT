#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9;
typedef pair<int, int> pii;
#define int long long
int32_t main() {
    int n, k;
    cin >> n >> k;
    vector<pii> a(n);
    for (auto &i : a) cin >> i.first >> i.second;
    sort(a.begin(), a.end());
    int ans = inf, sum = 0;
    for (auto i : a) {
        int rem = (k - sum - i.first) % k;
        rem = rem == 0 ? 0 : k - rem;
        ans = min(ans, rem);
        sum += rem;
    }
    cout << ans << '\n';
}
