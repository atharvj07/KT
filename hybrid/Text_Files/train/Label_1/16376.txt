#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  vector<char> t(n);
  int r = 0;
  for (int i = 0; i < n; i++) {
    if (i == 0) {
      t[i] = s[i];
    } else {
      if (t[i - 1] == s[i]) {
        r++;
        if (s[i] == 'R') {
          t[i] = 'G';
        } else if (s[i] == 'G') {
          t[i] = 'B';
        } else {
          t[i] = 'R';
        }
      } else {
        t[i] = s[i];
      }
    }
  }

  cout << r << endl;
  for (char c : t) {
    cout << c;
  }
  cout << endl;

  return 0;
}
