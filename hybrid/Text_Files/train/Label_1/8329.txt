#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, flag = 0;
  cin >> n;
  vector<int> ans, b[n];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      int x;
      cin >> x;
      if (i != j && x == -1) {
        cout << -1;
        return 0;
      } else if (i == j)
        flag = x;
      else
        b[i].push_back(x);
    }
  }
  if (flag == -1) {
    cout << -1;
    return 0;
  }
  ans.push_back(flag);
  for (int i = 1; i < n; i++) {
    int x = ans[i - 1];
    for (int j = 0; j < n - 1; j++) ans.push_back(x & b[i - 1][j]);
    x = ans[b[i - 1][n - 1] % ans.size()];
    ans.push_back(x);
  }
  for (int i = 0; i < n; i++) cout << ans[i] << " ";
  return 0;
}
