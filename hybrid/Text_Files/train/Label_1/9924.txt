#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
const int inf=1e6;
int n,m,q,ans=inf;
bool vis[200001];
int Size[200001];
map<pair<int,int>,int> mp;
map<int,vector<pair<int,int>>> r,c;
int findpar(int x) {
    if(x==vis[x]) return x;
    return vis[x]=findpar(vis[x]);
}
int main() {
    cin>>n>>m>>q;
    for(int i=1; i<=200000; i++) vis[i]=i;
    while(q--) {
        int x,y;
        cin>>x>>y;
        mp[{x,y}]=1;
        r[x].push_back({y,1});
        c[y].push_back({x,1});
    }
    for(int i=1; i<=n; i++) {
        int x=findpar(r[i][0].first);
        for(auto it:r[i]) {
            int y=findpar(it.first);
            if(x!=y) {
                vis[x]=y;
                Size[y]+=Size[x];
            }
        }
    }
    for(int i=1; i<=m; i++) {
        int x=findpar(c[i][0].first);
        for(auto it:c[i]) {
            int y=findpar(it.first);
            if(x!=y) {
                vis[x]=y;
                Size[y]+=Size[x];
            }
        }
    }
    for(int i=1; i<=n; i++) for(auto it:r[i]) {
        int x=findpar(it.first),y=findpar(i);
        if(x!=y) ans=min(ans,Size[x]-mp[{i,it.first}]+Size[y]-mp[{i,it.first}]);
    }
    for(int i=1; i<=m; i++) for(auto it:c[i]) {
        int x=findpar(it.first),y=findpar(i);
        if(x!=y) ans=min(ans,Size[x]-mp[{it.first,i}]+Size[y]-mp[{it.first,i}]);
    }
    if(ans==inf) ans=0;
    cout<<ans<<endl;
    return 0;
}
