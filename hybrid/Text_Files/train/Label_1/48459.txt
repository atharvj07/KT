import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long a = sc.nextLong();
            long b = sc.nextLong();
            long k = sc.nextLong();
            if ((a + b) % 2 == 0 && k % 2 == 0) {
                System.out.println("YES");
            } else if ((a + b) % 2 != 0 && (a + b) % 2 != k % 2) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
