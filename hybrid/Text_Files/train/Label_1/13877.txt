#include<bits/stdc++.h>
using namespace std;
int a[105];
int main()
{
    int n;
    cin>>n;
    for(int i=0; i<n; i++)
    {
        cin>>a[i];
    }
    int cnt=0;
    int ans=1;
    bool flag=false;
    for(int i=0; i<n; i++)
    {
        if(a[i]<0)  cnt++;
        if(cnt>=3)
        {
            flag=true;
            ans++;
            cnt=1;
        }
    }
    if(!flag)    ans=1;
    cout<<ans<<endl;
    int val=0;
    for(int i=0; i<n; i++)
    {
        if(a[i]>=0)   val++;
        else if(cnt>=3)
        {
            val=0;
            i--;
            cnt=1;
            ans++;
        }
        else if(a[i]<0)
        {
            val++;
            cnt++;
            if(cnt==3)
            {
                val=0;
                i--;
                cnt=1;
                ans++;
            }
        }
    }
    cout<<val<<" ";
    ans--;
    while(ans--)
    {
        cout<<"3 ";
    }
    cout<<endl;
    return 0;
}
