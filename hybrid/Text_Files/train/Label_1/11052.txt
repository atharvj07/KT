#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    bool ok = true;
    int coupon = 0; // number of consecutive days with discount
    for (int i = 0; i < n; i++) {
        if (coupon) {
            coupon--;
            a[i]--;
        }
        if (a[i] >= 2) {
            coupon++;
            a[i] -= 2;
        }
        if (a[i] > 0) { ok = false; }
    }
    if (coupon) { ok = false; }
    cout << (ok ? "YES" : "NO");
    return 0;
}
