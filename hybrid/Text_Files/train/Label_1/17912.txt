#include <cstdio>
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

const int MM = 60, HH = 24;
int main() {
    int h;
    scanf("%d", &h);
    string s, t;
    cin >> s;
    t = s;
    int c = 0;
    if (h == 12) {
        if (s[0] == '0') {
            if (s[1] > '1') {
                t[0] = '1', c = 1;
            } else if (s[1] == '1' && s[3] > '2') {
                t[3] = '2', c = 1;
            } else if (s[1] == '1' && s[3] == '2' && s[4] == '6') {
                t[3] = t[4] = '0', c = 2;
            }
        } else if (s[0] > '1') {
            t[0] = '0', c = 1;
        } else if (s[0] == '1' && s[1] > '1') {
            t[1] = '1', c = 1;
        } else if (s[0] == '1' && s[1] == '1' && s[3] > '2') {
            t[3] = '2', c = 1;
        } else if (s[0] == '1' && s[1] == '1' && s[3] == '2' && s[4] == '6') {
            t[3] = t[4] = '0', c = 2;
        }
        if (c < 2) {
            if (s[1] == '2' || s[1] == '5' || s[1] == '8') {
                s[1] = s[1] - 3;
            } else if (s[1] == '0' || s[1] == '3' || s[1] == '6' || s[1] == '9') {
                s[1] = s[1] + 1;
            }
        }
    } else {
        if (s[0] > '3') {
            t[0] = '0', c = 1;
        } else if (s[0] == '3' && s[1] > '9') {
            t[1] = '0', c = 1;
        }
        if (s[0] == '2' || s[0] == '5' || s[0] == '8') {
            s[0] = s[0] - 3;
        } else if (s[0] == '0' || s[0] == '3' || s[0] == '6' || s[0] == '9') {
            s[0] = s[0] + 1;
        }
    }
    for (int i = 2; i < 4; i++) {
        if (s[i] == '6') {
            s[i] = '0';
        } else {
            s[i] = s[i] + 2;
        }
    }
    cout << (c ? t : s) << endl;
    return 0;
}
