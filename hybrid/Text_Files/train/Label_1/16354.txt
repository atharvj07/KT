#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<char>> a(n, vector<char>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> a[i][j]; }
  }
  vector<int> ans1(n), ans2(m);
  bool ok = true;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (a[i][j] == '>' || a[i][j] == '=') {
        if (ans1[i] <= ans2[j]) {
          if (ans1[i] == 0) ans1[i] = ans2[j] + 1;
        } else
          ok = false;
      } else {
        if (ans1[i] >= ans2[j]) {
          if (ans2[j] == 0) ans2[j] = ans1[i] + 1;
        } else
          ok = false;
      }
    }
  }
  if (ok) {
    cout << "Yes" << endl;
    for (int i = 0; i < n; i++) cout << ans1[i] << " ";
    cout << endl;
    for (int i = 0; i < m; i++) cout << ans2[i] << " ";
    cout << endl;
  } else
    cout << "No" << endl;
  return 0;
}
