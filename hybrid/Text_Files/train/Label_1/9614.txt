#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[4][2][2];
  cin >> n;
  for (int i = 0; i < 4; i++) for (int j = 0; j < 2; j++) cin >> a[i][j][0] >> a[i][j][1];
  for (int i = 0; i < 4; i++) {
    if ((a[i][0][0] + a[i][1][0]) <= n || (a[i][0][1] + a[i][1][1]) <= n) {
      cout << i + 1 << " ";
      if (a[i][0][0] + a[i][1][0] <= n) cout << a[i][0][0] << " " << a[i][1][0] << "\n";
      else cout << a[i][0][1] << " " << a[i][1][1] << "\n";
      return 0;
    }
  }
  cout << -1 << "\n";
  return 0;
}
