#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<char>> v(n, vector<char>(m));
  bool color = false;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> v[i][j];
      if (v[i][j] == 'C' || v[i][j] == 'M' || v[i][j] == 'Y') {
        color = true;
      }
    }
  }
  if (color) {
    cout << "#Color" << endl;
  } else {
    cout << "#Black&White" << endl;
  }
  return 0;
}
