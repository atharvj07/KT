import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    String n = sc.nextLine();
    String m = sc.nextLine();
    String sorted = n.chars().sorted().mapToObj(c -> String.valueOf((char) c)).reduce("", String::concat);
    System.out.println(sorted.equals(m) ? "OK" : "WRONG_ANSWER");
  }
}
