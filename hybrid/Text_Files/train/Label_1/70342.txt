t = int(input())
for _ in range(t):
    n = int(input())
    weights = list(map(int, input().split()))
    weight_counts = {}
    for w in weights:
        weight_counts[w] = weight_counts.get(w, 0) + 1
    max_teams = 0
    for s in range(2, 2*n + 1):
        teams = 0
        for i in range(1, (s+1)//2):
            if i < s - i:
                teams += min(weight_counts.get(i, 0), weight_counts.get(s-i, 0))
        if s % 2 == 0:
            teams += weight_counts.get(s//2, 0) // 2
        max_teams = max(max_teams, teams)
    print(max_teams)
