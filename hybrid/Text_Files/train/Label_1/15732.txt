#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
typedef pair<ll, ll> pi;

const ll MAXN = 100005;

ll n;
ll a[MAXN];

int main() {
    ll t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (ll i = 0; i < n; i++) cin >> a[i];
        sort(a, a + n);
        ll dp[n + 1];
        dp[1] = 1;
        ll ans = 1;
        for (ll i = 2; i <= n; i++) {
            dp[i] = 1;
            for (ll j = i - 1; j >= 1; j--) {
                if (abs(a[i - 1] - a[j - 1]) >= a[n - 1]) {
                    dp[i] = max(dp[i], dp[j] + 1);
                }
            }
            ans = max(ans, dp[i]);
        }
        cout << ans << '\n';
    }
    return 0;
}
