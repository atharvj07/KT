#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (n); ++i)

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n + 1), b(k);
  rep(i, k) { cin >> b[i]; }
  iota(a.begin() + 1, a.end(), 1);
  vector<int> st(n + 1);
  for (int i = k - 1; i >= 0; --i) {
    a[b[i]] = 0;
  }
  int sz = 0;
  rep(i, n + 1) {
    if (a[i]) st[sz++] = i;
  }
  vector<int> ans;
  for (int i = 0; i < k; ++i) { ans.push_back(b[i]); }
  for (int i = sz - 1; i >= 0; --i) {
    if (st[i] >= ans.back()) {
      ans.push_back(st[i]);
    } else {
      for (int j = ans.size() - 1; j >= 0; --j) {
        if (st[i] > ans[j]) {
          swap(ans[j], st[i]);
          break;
        }
      }
    }
  }
  sort(a.begin() + 1, a.end());
  for (int i = 0; i < k; ++i) {
    if (b[i] != a[i + 1]) {
      cout << -1 << endl;
      return 0;
    }
  }
  for (int x : ans) cout << x << " ";
  cout << endl;
}
