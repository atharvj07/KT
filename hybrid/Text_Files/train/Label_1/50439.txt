import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        sc.close();
        int l = 0, r = n - 1, f = 0, b = 0;
        while (l <= r) {
            if (l == r) {
                f = b = 1;
                break;
            } else if (d[l] >= r - l) {
                f = 1;
                l++;
            } else if (d[r] >= r - l) {
                b = 1;
                r--;
            } else {
                break;
            }
        }
        System.out.println(f == 1 && b == 1 ? "yes" : "no");
    }
}
