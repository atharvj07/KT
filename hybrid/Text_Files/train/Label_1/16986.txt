#include <bits/stdc++.h>
#define pb push_back
using namespace std;
vector<int>adj[1001];
int w[1001], a[1001], b[1001], c[1001][1001], vis[1001];
int main(){
    ios_base :: sync_with_stdio(false);
    cin.tie(0);
    int n, m, k, i, j, x, y, mx, id;
    cin >> n >> m >> k;
    for(i = 1; i <= n; i++) cin >> w[i];
    for(i = 1; i <= m; i++) cin >> a[i] >> b[i];
    for(i = 1; i <= k; i++) for(j = 1; j <= n; j++) vis[j] = 0;
    for(i = 1; i <= m; i++){
        mx = 0;
        for(j = 1; j <= k; j++){
            if(vis[a[i]] == 0 || vis[b[i]] == 0) c[j][a[i]]++, c[j][b[i]]++;
            else c[j][a[i]]++, c[j][b[i]]++;
            if(c[j][a[i]] > mx) mx = c[j][a[i]], id = j;
            if(c[j][b[i]] > mx) mx = c[j][b[i]], id = j;
        }
        for(j = 1; j <= k; j++){
            if(j == id) continue;
            if(c[id][a[i]] == c[j][a[i]]) a[i] = -1;
            else if(c[id][b[i]] == c[j][b[i]]) b[i] = -1;
        }
        if(mx > 2) a[i] = b[i] = -1;
        vis[a[i]] = 1, vis[b[i]] = 1;
    }
    for(i = 1; i <= m; i++){
        if(a[i] != -1 && b[i] != -1){
            for(j = 1; j <= k; j++){
                if(c[j][a[i]] == c[j][b[i]]) cout << j << "\n";
            }
        }
        else cout << "0\n";
    }
    return 0;
}
