import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int E = sc.nextInt();
        int T = sc.nextInt();
        long[] x = new long[N];
        for (int i = 0; i < N; i++) {
            x[i] = sc.nextLong();
        }
        long ans = 0;
        for (int i = 0; i < N; i++) {
            if (i > 0) {
                ans = Math.max(ans, x[i] - x[i - 1]);
            }
            ans = Math.max(ans, E - x[i]);
        }
        ans += T * N;
        System.out.println(ans);
    }
}
