#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    string s, t;
    cin >> s >> t;
    int sidx = 0, tidx = 0, ans = 0;
    while (sidx < s.length() && tidx < t.length()) {
        if (s[sidx] == t[tidx]) {
            tidx++;
        }
        sidx++;
        ans = max(ans, sidx - tidx);
    }
    cout << ans << endl;
    return 0;
}
