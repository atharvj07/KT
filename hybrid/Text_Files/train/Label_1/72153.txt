from sys import stdin
input=lambda:stdin.readline().strip()
n,x,y=[int(i) for i in input().split()]
adj=[[] for i in range(n)]
for i in range(n-1):
    a,b=[int(i)-1 for i in input().split()]
    adj[a].append(b)
    adj[b].append(a)
q=[(i,0) for i in range(n)]
vis=[False for i in range(n)]
ans=1e18
while q:
    v,d=q.pop(0)
    if vis[v]:continue
    vis[v]=True
    ans=min(ans,d)
    for u in adj[v]:
        if not vis[u]:q.append((u,d+x))
for i in range(n):
    vis[i]=False
while q:
    v,d=q.pop(0)
    if vis[v]:continue
    vis[v]=True
    ans=min(ans,d)
    for u in adj[v]:
        if not vis[u]:q.append((u,d+y))
print(ans)
