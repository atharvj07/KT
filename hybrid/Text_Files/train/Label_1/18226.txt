#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
	int n, m;
	cin >> n >> m;
	vector<int> p1[10], p2[10];
	for (int i = 0; i < n; i++) {
		int a, b;
		cin >> a >> b;
		p1[a].push_back(b);
		p1[b].push_back(a);
	}
	for (int i = 0; i < m; i++) {
		int a, b;
		cin >> a >> b;
		p2[a].push_back(b);
		p2[b].push_back(a);
	}
	int cnt = 0, ans;
	for (int i = 1; i <= 9; i++) {
		sort(p1[i].begin(), p1[i].end());
		sort(p2[i].begin(), p2[i].end());
		if (p1[i] == p2[i]) {
			cnt++;
			ans = i;
		}
	}
	if (cnt == 1)
		cout << ans;
	else if (cnt == 0)
		cout << -1;
	else
		cout << 0;
	return 0;
}
