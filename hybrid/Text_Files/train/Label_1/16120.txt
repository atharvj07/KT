#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int c, d;
    cin >> c >> d;
    if (c == d) {
      cout << 0 << endl;
    } else if ((c - d) % 2 == 0) {
      cout << 1 << endl;
    } else {
      cout << -1 << endl;
    }
  }
  return 0;
}
