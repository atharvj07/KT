#include<bits/stdc++.h>
using namespace std;
int dp[100005];
int main(){
    int n, k;
    int mx = 0;
    cin >> n >> k;
    int i, j;
    if(k > 2 * n - 2){
        cout << -1;
        return 0;
    }
    for(i = 0; i < n; i++){
        if(k){
            dp[i] = i + 1;
            k--;
        }
        else dp[i] = i;
        mx = max(mx, dp[i]);
    }
    if(k == 1){
        cout << -1;
        return 0;
    }
    if(mx != n){
        cout << -1;
        return 0;
    }
    for(i = 0; i < n; i++){
        cout << dp[i] << " ";
    }
    cout << endl;
}
