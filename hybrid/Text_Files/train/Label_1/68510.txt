from sys import stdin
read = lambda : stdin.readline().strip()
write = lambda val : sys.stdout.write(val)

k = int(read())
L = [read().split() for _ in range(k)]
L = [(len(l) - 1, sum([int(a) for a in l[1:]])) for l in L]
L.sort()

avg = L[0][1] // L[0][0]
if any(l[1] != avg * l[0] for l in L): write("No"); exit()
write("Yes\n")
for l in L:
    a, s = l
    c, p = (s - avg * a) // a, (avg * a - s) // a
    write(f"{int(L[p][1])} {int(L[c][1])}\n")
