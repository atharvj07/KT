import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] coins = new int[5];
        for (int i = 0; i < 5; i++) {
            coins[i] = in.nextInt();
        }
        int min = coins[0], max = coins[0];
        for (int i = 1; i < 5; i++) {
            if (coins[i] < min) {
                min = coins[i];
            } else if (coins[i] > max) {
                max = coins[i];
            }
        }
        if ((max - min) % 4 != 0) {
            System.out.println(-1);
            return;
        }
        System.out.println((max - min) / 4);
    }
}
