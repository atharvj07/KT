n = int(input())
names = []
for _ in range(n):
    names.append(input())

pos = {}
for name in names:
    for i, c in enumerate(name):
        pos[c] = max(pos.get(c, 0), i)
last = 0
order = []
vis = [False] * 26
for i in range(26):
    c = chr(ord('a') + i)
    if vis[i]:
        continue
    if pos.get(c, 0) <= last:
        order.append(c)
        vis[i] = True
    else:
        for j in range(i + 1, 26):
            if vis[j]:
                continue
            d = chr(ord('a') + j)
            if pos.get(d, 0) <= last:
                order.append(d)
                vis[j] = True
            else:
                last = pos[d]
                order.append(d)
                vis[j] = True
                break
        if not vis[i]:
            order.append(c)
            vis[i] = True
if len(order) == 26:
    print(''.join(order))
else:
    print("Impossible")
