import sys
n, m = map(int, input().split())
e = [list() for _ in range(n+1)]
for _ in range(m):
    a, b = map(int, input().split())
    e[a].append(b)
    e[b].append(a)
for i in range(n):
    if len(e[i])%2:
        cnt = 0
        for j in e[i]:
            if len(e[j])%2:
                cnt+=1
        if cnt == 1:
            print(1,1)
            sys.exit()
print(0,1)
