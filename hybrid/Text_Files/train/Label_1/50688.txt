import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long x = in.nextLong(), y = in.nextLong(), z = in.nextLong();
        long c = x / z + y / z;
        long min = 0;
        if (x % z == 0 && y % z == 0) {
            min = 0;
        } else if (x % z != 0 && y % z == 0) {
            min = z - x % z;
        } else if (x % z == 0 && y % z != 0) {
            min = z - y % z;
        } else {
            min = Math.min((z - x % z) + (y % z), (z - y % z) + (x % z));
        }
        System.out.println(c + " " + min);
    }
}
