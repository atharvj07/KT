def min_poison_duration(t, test_cases):
    results = []
    for i in range(t):
        n, h = test_cases[i][0]
        attacks = test_cases[i][1]
        
        # Binary search for the minimum k
        left, right = 1, h
        while left < right:
            k = (left + right) // 2
            damage = 0
            for j in range(n-1):
                damage += min(k, attacks[j+1] - attacks[j])
            damage += k  # Add damage from the last attack
            
            if damage >= h:
                right = k
            else:
                left = k + 1
        
        results.append(left)
    return results

# Read input
t = int(input().strip())
test_cases = []
for _ in range(t):
    n, h = map(int, input().strip().split())
    attacks = list(map(int, input().strip().split()))
    test_cases.append(((n, h), attacks))

# Solve and print output
results = min_poison_duration(t, test_cases)
for result in results:
    print(result)
