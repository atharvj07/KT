#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define vvi vector<vi>
#define FOR(i, a, b) for(int i = a; i < b; i++)
#define REP(i, n) FOR(i, 0, n)
#define ll long long
#define pb push_back
const int INF = 0x3f3f3f3f;
int main() {
    int n;
    cin >> n;
    vi a(n);
    REP(i, n) cin >> a[i];
    int r1, c1, r2, c2;
    cin >> r1 >> c1 >> r2 >> c2;
    vvi d(n+1, vi(max(c1, c2)+1, INF));
    d[r1][c1] = 0;
    priority_queue<tuple<int, int, int>> pq;
    pq.push({ 0, r1, c1 });
    while(!pq.empty()) {
        int dist, r, c;
        tie(dist, r, c) = pq.top();
        pq.pop();
        dist = -dist;
        if(r == r2 && c == c2) {
            cout << dist << endl;
            return 0;
        }
        if(r - 1 >= 1 && d[r-1][c] > dist + 1) {
            pq.push({ -dist-1, r-1, c });
            d[r-1][c] = dist + 1;
        }
        if(r + 1 <= n && d[r+1][c] > dist + 1) {
            pq.push({ -dist-1, r+1, c });
            d[r+1][c] = dist + 1;
        }
        if(c - 1 >= 1 && d[r][c-1] > dist + 1) {
            pq.push({ -dist-1, r, c-1 });
            d[r][c-1] = dist + 1;
        }
        if(c + 1 <= a[r] && d[r][c+1] > dist + 1) {
            pq.push({ -dist-1, r, c+1 });
            d[r][c+1] = dist + 1;
        }
    }
    cout << -1 << endl;
    return 0;
}
