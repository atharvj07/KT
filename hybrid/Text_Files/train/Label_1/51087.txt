import java.util.*;
import java.lang.*;

public class Main {

    static class Edge {
        int to, weight;
        Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    static List<Edge>[] adj;
    static int n, m, k;
    static int[] dist;
    static int[] malls;
    static final int INF = (int) 1e9;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        adj = new List[n+1];
        for (int i = 1; i <= n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int l = sc.nextInt();
            adj[a].add(new Edge(b, l));
            adj[b].add(new Edge(a, l));
        }
        malls = new int[k];
        for (int i = 0; i < k; i++) malls[i] = sc.nextInt();
        dist = new int[n+1];
        Arrays.fill(dist, INF);
        dijkstra();
        int ans = INF;
        for (int mall : malls) ans = Math.min(ans, dist[mall]);
        System.out.println(ans);
    }

    static void dijkstra() {
        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> b[1] - a[1]);
        dist[1] = 0;
        pq.offer(new int[] {1, 0});
        while (!pq.isEmpty()) {
            int[] cur = pq.poll();
            int node = cur[0];
            int weight = cur[1];
            if (weight > dist[node]) continue;
            for (Edge e : adj[node]) {
                int next = e.to;
                int newWeight = weight + e.weight;
                if (newWeight < dist[next]) {
                    dist[next] = newWeight;
                    pq.offer(new int[] {next, newWeight});
                }
            }
        }
    }
}
