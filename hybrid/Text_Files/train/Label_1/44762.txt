import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Stack;

public class Main {

    private static final boolean SAME = true;
    private static final boolean DIFF = false;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] nm = reader.readLine().split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);

        List<List<Integer>> adj = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            adj.add(new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            String[] ab = reader.readLine().split(" ");
            int a = Integer.parseInt(ab[0]);
            int b = Integer.parseInt(ab[1]);
            adj.get(a - 1).add(b);
        }

        boolean[] visited = new boolean[n];
        boolean same = true;

        for (int i = 1; i <= n; i++) {
            if (visited[i - 1]) continue;
            same = dfs(i, i, visited, adj) && dfs2(i, new boolean[n], adj) && same;
        }

        for (int i = 0; i < m; i++) {
            System.out.println(same ? "same" : "diff");
        }
    }

    private static boolean dfs(int v, int s, boolean[] visited, List<List<Integer>> adj) {
        visited[v - 1] = true;
        for (int u : adj.get(v - 1)) {
            if (!visited[u - 1]) return dfs(u, s, visited, adj);
            if (u != s) return false;
        }
        return true;
    }

    private static boolean dfs2(int v, boolean[] visited, List<List<Integer>> adj) {
        Stack<Integer> s = new Stack<>();
        s.push(v);
        visited[v - 1] = true;
        boolean found = false;

        while (!s.isEmpty()) {
            int u = s.pop();
            if (u == v) {
                found = !found;
                if (found) return true;
            }
            for (int a : adj.get(u - 1)) {
                if (!visited[a - 1]) {
                    s.push(a);
                    visited[a - 1] = true;
                }
            }
        }
        return false;
    }
}
