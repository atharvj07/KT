#include <bits/stdc++.h>

using namespace std;

using ll = long long;

int main() {
  ll n, m;
  cin >> n >> m;
  vector<ll> a(n);
  for (auto &e : a) { cin >> e; }
  sort(a.begin(), a.end());
  vector<ll> psa(n + 1);
  for (ll i = 0; i < n; ++i) { psa[i + 1] = psa[i] + a[i]; }
  const ll inf = 1e18;
  vector<ll> dp(n + 1, inf);
  dp[0] = 0;
  for (int i = 1; i <= n; ++i) {
    for (int j = i; j >= max(1ll, i - m); --j) {
      dp[i] = min(dp[i], psa[i] - psa[j - 1] + dp[j - 1]);
    }
  }
  for (int i = 1; i <= n; ++i) { cout << dp[i] << ' '; }
  return 0;
}
