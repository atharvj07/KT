import java.util.*;
import java.io.*;

public class Main {

    private static class Item {
        int seat, timeIn, timeOut;

        public Item(int seat, int timeIn) {
            this.seat = seat;
            this.timeIn = timeIn;
            this.timeOut = timeIn;
        }

        @Override
        public String toString() {
            return "Item{" +
                    "seat=" + seat +
                    ", timeIn=" + timeIn +
                    ", timeOut=" + timeOut +
                    '}';
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int p = scanner.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = scanner.nextInt();
        }
        Deque<Item> queue = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            if (queue.isEmpty()) {
                queue.offer(new Item(i + 1, t[i]));
                continue;
            }
            while (!queue.isEmpty() && queue.peekLast(). seat <= i + 1 && queue.peekLast().timeOut <= t[i]) {
                queue.pollLast();
            }
            if (queue.isEmpty()) {
                queue.offer(new Item(i + 1, t[i]));
            } else {
                int timeLeft = t[i] - queue.peekLast().timeOut;
                p -= timeLeft;
                while (p < 0) {
                    p += p;
                    queue.pollLast();
                }
                queue.offer(new Item(i + 1, t[i]));
            }
        }
        ArrayList<Integer> result = new ArrayList<>();
        while (!queue.isEmpty()) {
            result.add(queue.poll().timeOut + p);
        }
        for (int i = 0; i < n; i++) {
            System.out.println(result.get(i));
        }
    }
}
