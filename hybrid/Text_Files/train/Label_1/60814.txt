n = int(input())
a = list(map(int, input().split()))
s = sum(a)
if s % 2 or n < 3: print(0)
else:
    prv = -1
    l = [0] * n
    for i in range(n):
        if a[i] % 2:
            if prv == -1: prv = i
            l[i] = 1
    if prv != -1 and sum(a[(prv + 1):n]) % 2: l[prv] = 0
    mx = (s // 2 - sum(a[prv + 1:n])) // 2
    mn = -1
    for i in range(prv + 1, n):
        if l[i] and mx:
            mn = i
            mx -= 1
    if mn != -1:
        for i in range(prv, mn - 1, -1):
            l[i] = 1
            l[prv] = 0
            if mx:
                prv = i
                mx -= 1
            else: break
    if sum([i for i in a if i % 2]) % 2: l[prv] = 0
    print(sum([x for x, y in zip(a, l) if y]))
