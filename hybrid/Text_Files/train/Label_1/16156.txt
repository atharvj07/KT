#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int open = 0;
  vector<int> pos;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(') {
      open++;
    } else {
      open--;
      if (open == 0 && pos.size() < k / 2) {
        pos.push_back(i);
      }
    }
  }
  string res;
  res.resize(k);
  for (int i = 0; i < k / 2; i++) {
    res[2 * i] = '(';
    res[2 * i + 1] = s[pos[i]];
  }
  cout << res << endl;
  return 0;
}
