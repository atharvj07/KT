#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
const int N = 2e5 + 5;
vi adj[N];
int w[N];
bool vis[N];
vi p;
int n, m;
void dfs(int node, int par, int sum) {
  vis[node] = 1;
  p.push_back(sum);
  for (int child : adj[node]) {
    if (child == par) continue;
    dfs(child, node, sum + w[child]);
  }
}
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++) adj[i].clear(), vis[i] = 0;
    p.clear();
    for (int i = 1; i <= n - 1; i++) {
      int u, v, wt;
      cin >> u >> v >> wt;
      adj[u].push_back(v);
      adj[v].push_back(u);
      w[u] = wt;
      w[v] = wt;
    }
    dfs(1, 1, 0);
    sort(p.begin(), p.end());
    int ans1 = 0, ans2 = 0;
    for (int i = 0; i < p.size(); i++) {
      ans1 += p[i];
      ans2 += p[p.size() - i - 1];
    }
    cout << ans1 << " " << ans2 << endl;
  }
  return 0;
}
