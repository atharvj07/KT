#include <bits/stdc++.h>
using namespace std;
long long a[200005], n, k, x, b[200005], c, d, j;
int main() {
  cin >> n >> k >> x;
  for (long long i = 1; i <= n; i++) {
    cin >> a[i];
    b[i] = a[i];
  }
  sort(a + 1, a + n + 1);
  if (x != 1) {
    for (long long i = 1; i <= n; i++) {
      if (i > 1 and a[i] - a[i - 1] > x) {
        if (k > 0) {
          k--;
          a[i] = a[i - 1] + x;
        } else {
          c++;
        }
      }
    }
    sort(a + 1, a + n + 1);
    for (long long i = 1; i <= n; i++) {
      if (i > 1 and a[i] - a[i - 1] > x) {
        c++;
      }
    }
  }
  for (long long i = 1; i <= n; i++) {
    if (a[i] != b[i]) {
      swap(a[i], a[i + 1]);
      d++;
    }
  }
  cout << c + d;
  return 0;
}
