#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int,int> pi;
const int mod=1e9+7;
void solve()
{
    int n,k;
    cin>>n>>k;
    if(k>(n-1))
    {
        cout<<-1<<endl;
        return;
    }
    int a,b,c,d;
    cin>>a>>b>>c>>d;

    set<pi> st;
    vector<vector<int>> g(n+1);

    for(int i=0;i<k;++i)
    {
        int x,y;
        cin>>x>>y;
        if(x==y or st.count({x,y}) or st.count({y,x}))
        {
            cout<<-1<<endl;
            return;
        }
        st.insert({x,y});
        st.insert({y,x});
        g[x].push_back(y);
        g[y].push_back(x);
    }
    int now=a;
    vector<int> v={a};
    int cnt=0;
    while(cnt<n-1)
    {
        int flag=0;
        for(auto nbr:g[now])
        {
            if(nbr==v.back())
            {
                continue;
            }
            if(nbr==b)
            {
                flag=2;
                now=nbr;
                v.push_back(nbr);
                break;
            }
            else
            {
                flag=1;
                now=nbr;
                v.push_back(nbr);
                break;
            }

        }
        if(flag==0)
        {
            cout<<-1<<endl;
            return;
        }
        if(flag==2)
        {
            break;
        }
        cnt++;
    }
    now=c;
    vector<int> u={c};
    cnt=0;
    while(cnt<n-1)
    {
        int flag=0;
        for(auto nbr:g[now])
        {
            if(nbr==u.back())
            {
                continue;
            }
            if(nbr==d)
            {
                flag=2;
                now=nbr;
                u.push_back(nbr);
                break;
            }
            else
            {
                flag=1;
                now=nbr;
                u.push_back(nbr);
                break;
            }

        }
        if(flag==0)
        {
            cout<<-1<<endl;
            return;
        }
        if(flag==2)
        {
            break;
        }
        cnt++;
    }
    for(auto x:v)
    {
        cout<<x<<" ";
    }
    cout<<endl;
    for(auto x:u)
    {
        cout<<x<<" ";
    }
    cout<<endl;

}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  solve();
  return 0;
}
