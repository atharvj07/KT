import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++) {
            board[i] = scanner.next().toCharArray();
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.' && canFit(board, i, j)) {
                    board[i][j] = '#';
                    board[i][j + 1] = '#';
                    board[i][j + 2] = '#';
                    board[i + 1][j + 1] = '#';
                    board[i + 2][j + 1] = '#';
                }
            }
        }
        boolean allFilled = true;
        for (char[] row : board) {
            if (Arrays.toString(row).contains(".")) {
                allFilled = false;
                break;
            }
        }
        System.out.println(allFilled ? "YES" : "NO");
    }

    public static boolean canFit(char[][] board, int x, int y) {
        return x + 2 < board.length && y + 2 < board[0].length && board[x][y] == '.' && board[x][y + 1] == '.' && board[x][y + 2] == '.' && board[x + 1][y + 1] == '.' && board[x + 2][y + 1] == '.';
    }
}
