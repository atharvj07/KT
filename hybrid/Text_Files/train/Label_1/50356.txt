import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = in.nextInt();
        }
        int ans = 0;
        boolean[] isBoundary = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (i == 0 || i == n - 1) {
                isBoundary[i] = true;
            } else if (h[i] > h[i - 1] || h[i] > h[i + 1]) {
                isBoundary[i] = true;
            }
        }
        while (true) {
            boolean hasBoundary = false;
            for (int i = 0; i < n; i++) {
                if (isBoundary[i]) {
                    hasBoundary = true;
                    h[i]--;
                    if (h[i] == 0) {
                        isBoundary[i] = false;
                    }
                }
            }
            if (!hasBoundary) {
                break;
            }
            ans++;
        }
        System.out.println(ans);
    }
}
