#include <cstring>
#include <iostream>

int main() {
  char s[11];
  std::cin >> s;
  int l = strlen(s);
  for (int i = 0; i < l; i++) {
    if (s[i] != '1' && s[i] != '4' && s[i] != '*') {
      std::cout << "NO" << std::endl;
      return 0;
    }
    if (s[i] == '*') {
      if (i % 2 == 0) {
        if (s[i - 1] != '1') {
          std::cout << "NO" << std::endl;
          return 0;
        }
      } else {
        if (s[i - 1] != '4') {
          std::cout << "NO" << std::endl;
          return 0;
        }
      }
    }
  }
  std::cout << "YES" << std::endl;
  return 0;
}
