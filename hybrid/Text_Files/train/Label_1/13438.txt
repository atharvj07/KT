#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
long long x[300005], ans;
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> x[i];
  sort(x, x + n);
  for (int i = 1; i < n; i++) ans = (ans + x[i] - x[i - 1]) % mod;
  ans = ans * n % mod;
  cout << ans << '\n';
  return 0;
}
