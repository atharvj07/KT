#include<bits/stdc++.h>
using namespace std;
#define int long long
const int maxn=3e3+1;
vector<int> adj[maxn];
vector<bool> vis(maxn);

void dfs(int u,int p){
    vis[u]=true;
    for(auto x:adj[u]){
        if(vis[x]==false && x!=p)
            dfs(x,u);
    }
}
int32_t main()
{
    int n,m;
    cin>>n>>m;
    while(m--){
        int a,b;
        cin>>a>>b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    int cc=0;
    for(int i=1;i<=n;i++){
        if(vis[i]==false){
            dfs(i,i);
            cc++;
        }
    }
    if(cc>2)
        cc=cc*(cc-1)*(cc-2)/6;
    else cc=0;
    cout<<cc;
}
