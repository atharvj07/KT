import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    char[][] image = new char[n + 2][m + 2];
    for (int i = 1; i <= n; i++) {
      Arrays.fill(image[i], '!');
      char[] c = sc.next().toCharArray();
      for (int j = 1; j <= m; j++) {
        image[i][j] = c[j - 1];
      }
    }
    int ans = 0;
    for (int i = 2; i <= n; i++) {
      for (int j = 2; j <= m; j++) {
        if (image[i][j] == 'a' && image[i][j - 1] == 'f' && image[i - 1][j] == 'c' && image[i - 1][j - 1] == 'e')
          ans++;
        if (image[i][j] == 'e' && image[i][j - 1] == 'c' && image[i - 1][j] == 'f' && image[i - 1][j - 1] == 'a')
          ans++;
      }
    }
    System.out.println(ans);
  }
}
