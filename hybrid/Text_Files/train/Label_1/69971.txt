t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    q = int(input())
    counts = {}
    for num in a:
        if num in counts:
            counts[num] += 1
        else:
            counts[num] = 1
    new_a = [counts[num] for num in a]
    counts2 = {}
    for num in new_a:
        if num in counts2:
            counts2[num] += 1
        else:
            counts2[num] = 1
    final_a = [counts2[num] for num in new_a]
    for _ in range(q):
        x, k = map(int, input().split())
        if k == 0:
            print(a[x-1])
        elif k == 1:
            print(new_a[x-1])
        else:
            print(final_a[x-1])
