import java.util.*;

public class Main {
    private static int solve(int[] yang, int[] monster, int[] prices) {
        int[] remain = new int[3];
        for (int i = 0; i < 3; i++)
            remain[i] = monster[i] - yang[i];
        int result = 0;
        for (int i = 0; i < 3; i++) {
            if (remain[i] > 0) {
                int add = (int) Math.ceil((double) remain[i] / (prices[i]));
                result += add * prices[i];
                for (int j = 0; j < 3; j++)
                    yang[j] += add;
            }
        }
        return result;
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int[] yang = new int[3], monster = new int[3], prices = new int[3];
        for (int i = 0; i < 3; i++)
            yang[i] = input.nextInt();
        for (int i = 0; i < 3; i++)
            monster[i] = input.nextInt();
        for (int i = 0; i < 3; i++)
            prices[i] = input.nextInt();
        System.out.println(solve(yang, monster, prices));
    }
}
