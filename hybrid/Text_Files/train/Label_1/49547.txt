import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[1 << n];
        for(int i = 0; i < (1 << n); i++) {
            a[i] = sc.nextInt();
        }

        while((1 << n) > 1) {
            for(int i = 0; i < (1 << n) / 2; i++) {
                if(a[i * 2] > a[i * 2 + 1]) {
                    a[i] = a[i * 2] - a[i * 2 + 1];
                } else if(a[i * 2] < a[i * 2 + 1]) {
                    a[i] = a[i * 2 + 1] - a[i * 2];
                } else {
                    a[i] = a[i * 2];
                }
            }
            n--;
        }

        System.out.println(a[0]);
    }
}
