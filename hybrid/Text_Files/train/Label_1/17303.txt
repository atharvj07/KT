#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> cards(n);
    for (int i = 0; i < n; i++) {
        cin >> cards[i].first >> cards[i].second;
    }

    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) {
        a[i] = cards[i].first;
        b[i] = cards[i].second;
    }

    sort(a.begin(), a.end());
    sort(b.begin(), b.end(), greater<int>());

    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] != cards[i].first || b[i] != cards[i].second) {
            ans++;
        }
    }

    if (ans % 2 == 1) {
        cout << "-1" << endl;
    } else {
        cout << ans / 2 << endl;
    }

    return 0;
}
