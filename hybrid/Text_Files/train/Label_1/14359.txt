#include <bits/stdc++.h>
using namespace std;

const int MX = 2e5 + 5;
int n, ans, l[MX], r[MX];
vector<pair<int, int>> seg;

int main() {
    int t;
    cin >> t;
    while (t--) {
        seg.clear();
        cin >> n;
        while (n--) {
            int a, b;
            cin >> a >> b;
            if (a > b) swap(a, b);
            seg.push_back({a, b});
        }
        sort(seg.begin(), seg.end());
        int al = -2e9, ar = -2e9;
        for (auto x : seg) {
            int L = x.first, R = x.second;
            if (L > al && R > ar) {
                ans++;
                al = L;
                ar = R;
            }
        }
        cout << ans << endl;
        ans = 0;
    }
    return 0;
}
