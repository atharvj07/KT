import java.util.Arrays;
import java.util.Scanner;

public class RomaGrind {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();

        int[] weaponAttack = new int[n];
        int[] weaponCost = new int[n];
        for (int i = 0; i < n; i++) {
            weaponAttack[i] = scanner.nextInt();
            weaponCost[i] = scanner.nextInt();
        }

        int[] armorDefense = new int[m];
        int[] armorCost = new int[m];
        for (int i = 0; i < m; i++) {
            armorDefense[i] = scanner.nextInt();
            armorCost[i] = scanner.nextInt();
        }

        int[][] monster = new int[p][3];
        for (int i = 0; i < p; i++) {
            monster[i][0] = scanner.nextInt();
            monster[i][1] = scanner.nextInt();
            monster[i][2] = scanner.nextInt();
        }

        Arrays.sort(weaponAttack);
        Arrays.sort(armorDefense);
        Arrays.sort(monster, (a, b) -> Integer.compare(b[2], a[2]));

        int maxProfit = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int profit = 0;
                for (int k = 0; k < p; k++) {
                    if (weaponAttack[i] > monster[k][0] && armorDefense[j] > monster[k][1]) {
                        profit += monster[k][2];
                    }
                }
                maxProfit = Math.max(maxProfit, profit - weaponCost[i] - armorCost[j]);
            }
        }

        System.out.println(maxProfit);
    }
}
