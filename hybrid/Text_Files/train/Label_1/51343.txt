import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        while (true) {
            int n = input.nextInt();
            if (n == 0)
                break;
            int minX = 0, maxX = 0, minY = 0, maxY = 0;
            int[] right = new int[n];
            int[] down = new int[n];
            for (int i = 1; i < n; ++i) {
                int p = input.nextInt(), d = input.nextInt();
                if (d == 0) {
                    minX = Math.min(minX, right[p]);
                    maxX = Math.max(maxX, right[p]);
                    right[i] = right[p] - 1;
                    down[i] = down[p];
                } else if (d == 1) {
                    minY = Math.min(minY, down[p]);
                    maxY = Math.max(maxY, down[p]);
                    right[i] = right[p];
                    down[i] = down[p] + 1;
                } else if (d == 2) {
                    minX = Math.min(minX, right[p]);
                    maxX = Math.max(maxX, right[p]);
                    right[i] = right[p] + 1;
                    down[i] = down[p];
                } else {
                    minY = Math.min(minY, down[p]);
                    maxY = Math.max(maxY, down[p]);
                    right[i] = right[p];
                    down[i] = down[p] - 1;
                }
            }
            System.out.println((maxX - minX + 1) + " " + (maxY - minY + 1));
        }
    }
}
