#include<bits/stdc++.h>
using namespace std;
int n, k,x,f[210],c[210],a[210],ans;
int main(){
    cin>>n>>k>>x;
    for(int i=1;i<=n;++i) cin>>a[i];
    if(x>=k){
        for(int i=1;i<=n;++i) f[i]=f[i-1]+a[i],c[i]=c[i-1]+1;
        int now=0;
        for(int i=k;i<=n;++i){
            ans=max(ans,f[i]-f[max(0,i-k-now)]);
            if(c[i]-c[max(0,i-k-now)]<k){
                now=i-k;
            }
        }
        cout<<ans<<endl;
    }else cout<<-1<<endl;
    return 0;
}
