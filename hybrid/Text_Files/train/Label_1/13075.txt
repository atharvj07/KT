#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 2e5 + 5;

int n;
string s, t;
vector<pair<int, int>> ans;

int main() {
  cin >> n >> s >> t;

  int diff = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] != t[i]) {
      diff++;
      ans.emplace_back(i, -1);
    }
  }

  if (diff <= 1) {
    cout << diff << '\n';
    if (diff == 1) {
      cout << ans[0].first + 1 << ' ' << ans[0].second + 1 << '\n';
    } else {
      cout << "-1 -1\n";
    }
    return 0;
  }

  int cnt = 0;
  for (auto &p : ans) {
    if (p.second == -1) {
      cnt++;
      if (cnt > 2) {
        break;
      }
      for (auto &q : ans) {
        if (p.first != q.first && p.first != q.second && q.first != q.second && p.second != q.first && p.second != q.second) {
          p.second = q.first;
          q.second = p.first;
          break;
        }
      }
    }
  }

  cout << 2 << '\n';
  for (auto &p : ans) {
    if (p.second != -1) {
      cout << p.first + 1 << ' ' << p.second + 1 << '\n';
    }
  }

  return 0;
}
