#include <bits/stdc++.h>
using namespace std;
const int maxp=1e5;
int n,xf,xt,yf,yt;
struct point{
    long long x, y;
    bool operator<(const point &a)const{return(x<a.x);}
}p[maxp];
signed main(){
    ios_base::sync_with_stdio(0), cin.tie(0);
    cin>>n;
    for(int i=1;i<=n;i++)cin>>p[i].x>>p[i].y;
    sort(p+1,p+1+n);
    xf=p[1].x, xt=p[n].x, yf=p[1].y, yt=p[n].y;
    cout<<(xt-xf+1)*(yt-yf+1)<<"\n";
    return 0;
}
