t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    matrix = [list(map(int, input().split())) for _ in range(n)]
    row_claimed = [False] * n
    col_claimed = [False] * m
    for i in range(n):
        for j in range(m):
            if matrix[i][j] == 1:
                row_claimed[i] = True
                col_claimed[j] = True
    free_rows = row_claimed.count(False)
    free_cols = col_claimed.count(False)
    moves = min(free_rows, free_cols)
    if moves % 2 == 0:
        print("Vivek")
    else:
        print("Ashish")
