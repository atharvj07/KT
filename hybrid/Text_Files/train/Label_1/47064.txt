import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Main {
    static class Contestant implements Comparable<Contestant> {
        int s, a, b;
        
        public Contestant(int s, int a, int b) {
            this.s = s;
            this.a = a;
            this.b = b;
        }

        @Override
        public int compareTo(Contestant that) {
            double happiness1 = this.s * (this.a - this.b) / 2.0;
            double happiness2 = that.s * (that.a - that.b) / 2.0;
            
            return Double.compare(happiness2, happiness1);
        }
    }
    
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        String[] parts = reader.readLine().split(" ");
        int n = Integer.parseInt(parts[0]);
        int s = Integer.parseInt(parts[1]);
        
        List<Contestant> contestants = new ArrayList<>();
        
        for (int i = 0; i < n; i++) {
            parts = reader.readLine().split(" ");
            int si = Integer.parseInt(parts[0]);
            int ai = Integer.parseInt(parts[1]);
            int bi = Integer.parseInt(parts[2]);
            contestants.add(new Contestant(si, ai, bi));
        }
        
        Collections.sort(contestants);
        
        long totalHappiness = 0;
        int type1 = 0, type2 = 0;
        
        for (Contestant contestant : contestants) {
            if (type1 + contestant.s <= s) {
                type1 += contestant.s;
                totalHappiness += contestant.a * contestant.s;
            } else if (type2 + contestant.s <= s) {
                type2 += contestant.s;
                totalHappiness += contestant.b * contestant.s;
            } else {
                break;
            }
        }
        
        System.out.println(totalHappiness);
    }
}
