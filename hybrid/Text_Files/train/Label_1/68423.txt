def calculate_intimidation(a, n):
    intimidation = 0
    for i in range(1, n - 1):
        if a[i] > a[i + 1] and a[i] > a[i - 1] or a[i] < a[i + 1] and a[i] < a[i - 1]:
            intimidation += 1
    return intimidation

def minimum_intimidation_value(n, a):
    original_intimidation = calculate_intimidation(a, n)
    if original_intimidation == 0:
        return 0
    min_intimidation = original_intimidation
    for i in range(1, n - 1):
        original_value = a[i]
        # Try setting a[i] to either a[i-1] or a[i+1]
        for new_value in [a[i - 1], a[i + 1]]:
            a[i] = new_value
            current_intimidation = calculate_intimidation(a, n)
            min_intimidation = min(min_intimidation, current_intimidation)
        a[i] = original_value
    return min_intimidation

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print(minimum_intimidation_value(n, a))
