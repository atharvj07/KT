import java.util.*;

public class Main {

    private static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    private static class Turn {
        int index, type; // 1 for left, -1 for right

        public Turn(int index, int type) {
            this.index = index;
            this.type = type;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            points[i] = new Point(x, y);
        }
        String s = scanner.nextLine();
        ArrayList<Turn> turns = new ArrayList<>();
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == 'L') {
                turns.add(new Turn(i + 1, 1));
            } else {
                turns.add(new Turn(i + 1, -1));
            }
        }
        int[] permutation = new int[n];
        for (int i = 0; i < n; i++) {
            permutation[i] = i + 1;
        }
        boolean found = false;
        while (!found) {
            boolean valid = true;
            for (Turn turn : turns) {
                int i = turn.index;
                int type = turn.type;
                if (((permutation[i] - permutation[i - 1]) * (permutation[i + 1] - permutation[i]) < 0) != (type == 1)) {
                    valid = false;
                    break;
                }
            }
            if (valid) {
                found = true;
            } else {
                nextPermutation(permutation);
            }
        }
        if (found) {
            for (int p : permutation) {
                System.out.print(p + " ");
            }
        } else {
            System.out.println(-1);
        }
    }

    private static void nextPermutation(int[] permutation) {
        int i = permutation.length - 2;
        while (i >= 0 && permutation[i] >= permutation[i + 1]) {
            i--;
        }
        if (i >= 0) {
            int j = permutation.length - 1;
            while (j >= 0 && permutation[j] <= permutation[i]) {
                j--;
            }
            int temp = permutation[i];
            permutation[i] = permutation[j];
            permutation[j] = temp;
        }
        reverse(permutation, i + 1);
    }

    private static void reverse(int[] permutation, int start) {
        int end = permutation.length - 1;
        while (start < end) {
            int temp = permutation[start];
            permutation[start] = permutation[end];
            permutation[end] = temp;
            start++;
            end--;
        }
    }
}
