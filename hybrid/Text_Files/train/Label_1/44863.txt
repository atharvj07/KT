import java.util.*;
class Main {
  static final int N=201, M=100001;
  static int n, m, a[]=new int[N], sum[][]=new int[N][N], first[]=new int[N], order[]=new int[N];
  static boolean vis[]=new boolean[N], flag[]=new boolean[M], prime[]=new boolean[M];
  static void dfs(int p, int sum, int next, int min) {
    vis[p]=true;
    if(first[p]!=0 && sum-first[p]==min) {
      m++;
      order[m]=vis[0]?m:0;
      for(int i=first[p]; i!=p; i=first[i])
        order[m]=order[m]==0?i:order[m]*100+i;
      return;
    }
    for(int i=next; i<=n; i++)
      if(!vis[i] && (sum+a[i]-a[p]>=min && flag[sum+a[i]-a[p]]))
        dfs(i, sum+a[i], i+1, min);
  }
  public static void main(String[] args) {
    Scanner in=new Scanner(System.in);
    n=in.nextInt();
    int i, j;
    for(i=2; i<=100000; i++)
      if(!prime[i]) {
        for(j=i+i; j<=100000; j+=i) prime[j]=true;
        flag[i]=true;
      }
    for(i=1; i<=n; i++) a[i]=in.nextInt();
    Arrays.sort(a, 1, n+1);
    for(i=1; i<=n; i++)
      for(j=i+1; j<=n; j++) sum[i][j]=sum[i][j-1]+a[j]-a[i];
    int min=a[n]-a[1];
    for(i=1; i<n; i++)
      for(j=i+1; j<=n; j++)
        if((i>2 || j<n) && flag[sum[i][j]]) {
          if((j-i+1)%2==0 && flag[sum[i][j]/2])
            min=Math.min(min, a[j]-a[i]);
          else min=Math.min(min, sum[i][j]);
        }
    dfs(1, 0, 2, min);
    if(order[m]==0) System.out.println("Impossible");
    else {
      System.out.println(m);
      for(i=1; i<=m; i++) {
        System.out.print(order[i]/100+" ");
        for(j=order[i]%100; j>0; j/=100)
          System.out.print(j+" ");
        System.out.println();
      }
    }
  }
}
