#include <cstdio>
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

bool is_prime(int n) {
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) { return false; }
    }
    return true;
}

int main() {
    int t, n, k;
    cin >> t;
    while (t--) {
        cin >> k >> n;
        vector<int> digits;
        while (n) {
            digits.push_back(n % 10);
            n /= 10;
        }
        reverse(digits.begin(), digits.end());

        int ans = 1;
        for (int l = 1; l < digits.size(); l++) {
            int num = 0;
            bool f = false;
            for (int i = 0; i < l; i++) {
                num = num * 10 + digits[i];
                if (!is_prime(num)) { f = true; break; }
            }
            if (f) {
                ans = l;
                break;
            }
        }
        cout << ans << '\n';
        for (int i = 0; i < ans; i++) {
            cout << digits[i];
        }
        cout << '\n';
    }
    return 0;
}
