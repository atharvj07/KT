#include <bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;
vector<int> adj[N];
vector<int> w(N);
vector<bool> vis(N);
int ans = 0;

void dfs(int u, int par) {
  if (vis[u]) return;
  vis[u] = true;
  ans += w[u];
  for (int v : adj[u]) {
    if (v == par) continue;
    dfs(v, u);
  }
}

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> w[i];
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  int s;
  cin >> s;
  dfs(s, s);
  cout << ans << endl;
  return 0;
}
