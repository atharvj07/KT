#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[101], i, j, x;
    cin >> n;
    for (i = 1; i <= n; i++) cin >> a[i];
    while (1) {
        x = -1;
        for (i = 1; i * 2 + 1 <= n; i++) {
            if (a[i] || a[i * 2] || a[i * 2 + 1]) {
                x = i;
                break;
            }
        }
        if (x == -1) break;
        a[x]--, a[x * 2]--, a[x * 2 + 1]--;
    }
    bool ok = true;
    for (i = 1; i <= n; i++) {
        if (a[i] > 0) ok = false;
    }
    cout << (ok ? x : -1) << endl;
}
