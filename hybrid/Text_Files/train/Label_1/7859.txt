#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int min_k = *max_element(a.begin(), a.end());
    int max_k = *max_element(a.begin(), a.end()) + m;
    cout << min_k << ' ' << max_k << endl;
    return 0;
}
