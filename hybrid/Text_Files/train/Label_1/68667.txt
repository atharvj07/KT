import sys
input = sys.stdin.readline

n, m = map(int, input().split())
songs = []
for _ in range(n):
    songs.append(list(map(int, input().split())))
moments = list(map(int, input().split()))

idx = 0
for moment in moments:
    while moment > songs[idx][1]:
        moment -= songs[idx][1] * songs[idx][0]
        idx += 1
    print(idx + 1)
