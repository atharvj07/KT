import sys

n, k = map(int, sys.stdin.readline().split())
books = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]
books.sort(key=lambda b: b[0])  # Sort by reading time

a_count, b_count = 0, 0
total_time = 0

for t, a, b in books:
    if (a or b) and (a_count < k or b_count < k):
        total_time += t
        a_count += a
        b_count += b
        if a_count >= k and b_count >= k:
            break
if a_count < k or b_count < k:
    print(-1)
else:
    print(total_time)
