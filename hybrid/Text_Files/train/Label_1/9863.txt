#include <iostream>
#include <vector>
#include <algorithm>
#define ll long long

using namespace std;

vector<ll> freqa, freqb, freqc;
ll n, m, x;

int main() {
    cin >> n;
    freqa.resize(1000001, 0);
    for (ll i = 0; i < n; i++) {
        cin >> x;
        freqa[x]++;
    }
    cin >> m;
    freqb.resize(1000001, 0);
    freqc.resize(1000001, 0);
    for (ll i = 0; i < m; i++) {
        cin >> x;
        freqb[x]++;
    }
    for (ll i = 0; i < m; i++) {
        cin >> x;
        freqc[x]++;
    }
    ll mx1 = -1, mx2 = -1, ans = -1;
    for (ll i = 1; i <= 1000000; i++) {
        ll tmp1 = freqa[i] + freqb[i];
        if (tmp1 > mx1) {
            mx1 = tmp1;
            mx2 = freqa[i] + freqc[i];
            ans = i;
        } else if (tmp1 == mx1) {
            tmp1 = freqa[i] + freqc[i];
            if (tmp1 > mx2) {
                mx1 = tmp1;
                ans = i;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
