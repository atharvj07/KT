#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
 ll n,q,x,y; cin>>n>>q;
 ll arr[n][n];
 ll cnt = 0;
 for(int i=0;i<(n*n+1)/2;i++){
  for(int j=0;j<n;j++){
   for(int k=0;k<n;k++){
    if((j+k+2)%2==0){
     cnt++;
     arr[j][k] = cnt;
    }
   }
  }
 }
 ll p = (n*n)/2 + 1;
 for(int i=0;i<(n*n)/2;i++){
  for(int j=0;j<n;j++){
   for(int k=0;k<n;k++){
    if((j+k+2)%2==1){
     cnt++;
     arr[j][k] = cnt;
    }
   }
  }
 }
 for(int i=0;i<q;i++){
  cin>>x>>y;
  cout<<arr[x-1][y-1]<<endl;
 }
}
