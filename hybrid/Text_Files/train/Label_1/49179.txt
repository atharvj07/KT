import java.util.*;
import java.io.*;
import java.nio.charset.StandardCharsets;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
        int n = Integer.parseInt(reader.readLine());
        Map<String, Integer> blues = new HashMap<>();
        Map<String, Integer> reds = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String s = reader.readLine();
            blues.put(s, blues.getOrDefault(s, 0) + 1);
        }
        int m = Integer.parseInt(reader.readLine());
        for (int i = 0; i < m; i++) {
            String t = reader.readLine();
            reds.put(t, reds.getOrDefault(t, 0) + 1);
        }
        int ans = 0;
        for (String key : blues.keySet()) {
            if (reds.containsKey(key)) {
                ans += blues.get(key) - reds.get(key);
            } else {
                ans += blues.get(key);
            }
        }
        System.out.println(ans);
    }
}
