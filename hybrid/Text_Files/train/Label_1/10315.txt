#include <cstdio>
#include <vector>
#include <algorithm>

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n, k;
    scanf("%d %d", &n, &k);
    std::vector<long long> p(n);
    for (int i = 0; i < n; i++) scanf("%lld", &p[i]);
    double cur = p[0] / 100.0, sum = 0.0;
    for (int i = 1; i < n; i++) {
      if (cur * (1 + k / 100.0) < p[i]) {
        sum += p[i] - cur * (1 + k / 100.0);
        cur = p[i];
      } else
        cur = p[i];
    }
    printf("%.0lf\n", sum);
  }
}
