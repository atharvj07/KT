#include<bits/stdc++.h>
using namespace std;
const long long N = 1e5 + 100;
long long c[N], t[N], pos[N], tt;
int main() {
    long long n, m;
    cin >> n >> m;
    for(long long i = 1; i <= n; i++) {
        cin >> c[i] >> t[i];
        pos[i] = pos[i - 1] + c[i];
        tt += t[i] * c[i];
    }
    long long now = 1;
    for(long long i = 1; i <= m; i++) {
        long long x;
        cin >> x;
        while(x > pos[now] * t[now]) now++;
        cout << now << endl;
    }
    return 0;
}
