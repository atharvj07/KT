#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long
const int mod = 1e9 + 7;

int main() {
    int n, r, avg;
    cin >> n >> r >> avg;
    ll sum = 0, tot = 0;
    vi a(n), b(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i] >> b[i];
        sum += a[i];
        tot += b[i];
    }
    int ans = max(0LL, ((avg * n) - sum + avg - 1) / avg);
    if (ans <= tot)
        cout << ans << endl;
    else
        cout << -1 << endl;
    return 0;
}
