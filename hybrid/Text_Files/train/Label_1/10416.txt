#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
const int N = 1e3 + 3;
const int M = 1e5 + 3;
int n, m;
vector<int> ed[N];
int deg[N];
int dis[N];
int fa[N];
vector<pair<int, int>> ans;

int dfs(int u, int p = 0) {
    for (int v : ed[u]) {
        if (v != p) {
            dis[v] = dis[u] + 1;
            fa[v] = u;
            int ret = dfs(v, u);
            if (ret != -1) {
                ans.emplace_back(u, v);
                if (ret != u) ans.emplace_back(v, ret);
                return ret;
            }
        }
    }
    return -1;
}

int main() {
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        ed[a].push_back(b);
        ed[b].push_back(a);
        deg[a]++, deg[b]++;
    }
    for (int i = 1; i <= n; i++) {
        if (deg[i] % 2 == 1) ans.emplace_back(i, i);
    }
    dfs(1);
    reverse(ans.begin(), ans.end());
    cout << (ans.size() + m) / 2 << '\n';
    for (auto &p : ans) { cout << p.first << ' ' << p.second << '\n'; }
    return 0;
}
