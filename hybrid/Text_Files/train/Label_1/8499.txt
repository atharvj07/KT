#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> w(n);
    for (int i = 0; i < n; i++) {
        cin >> w[i];
    }
    int ans1 = -1, ans2 = 0;
    for (int i = 1; i <= n; i *= 2) {
        vector<int> cnt(k + 1);
        for (int j = 0; j < n; j++) {
            if (w[j] == 100) {
                cnt[100]++;
            }
        }
        for (int j = 0; j <= k - 100; j++) {
            for (int l = 0; l <= cnt[100]; l++) {
                if (l * 100 + j <= k && cnt[50] >= i - l) {
                    if (ans1 == -1 || ans1 > l + (i - l) / 2) {
                        ans1 = l + (i - l) / 2;
                        ans2 = 1;
                    } else if (ans1 == l + (i - l) / 2) {
                        ans2 = ans2 * (i - l % 2) / 2 % 1000000007;
                    }
                }
            }
            cnt[j + 50]++;
            cnt[j]--;
        }
    }
    cout << ans1 << '\n' << ans2 << '\n';
    return 0;
}
