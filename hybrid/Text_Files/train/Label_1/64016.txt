from sys import stdin
mod = int(1e9) + 7
n, k = map(int, stdin.readline().split())
p = [int(i) for i in stdin.readline().split()]
b = [int(i) for i in stdin.readline().split()]
dp = [[[-1] * 11 for _ in range(11)] for _ in range((n // k) + 5)]

def calc(pos, last, pre):
    if pos == (n // k): return 1
    ret = dp[pos][last][pre]
    if ret != -1: return ret
    ret = 0
    for i in range(10):
        if (i != b[pos] or pre != 0) and (p[pos] == 1 or (i % p[pos] == 0)):
            ret += calc(pos + 1, i, i != 0)
    dp[pos][last][pre] = ret % mod
    return ret % mod

print(calc(0, 0, 0))
