using namespace std;
#include<bits/stdc++.h>
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        long long x,y,p,q;
        cin>>x>>y>>p>>q;
        if(x*q <= y*p)
        {
            cout<<(y*p-x*q)+1<<endl;
        }
        else
        {
            cout<<-1<<endl;
        }
    }
}
