#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1e9 + 7;
int main() {
  ll n, ans = 1;
  cin >> n;
  vector<pair<ll, ll>> p(n);
  map<ll, ll> x, y;
  for (int i = 0; i < n; i++) {
    cin >> p[i].first >> p[i].second;
    x[p[i].first]++;
    y[p[i].second]++;
  }
  for (auto i : x) ans = (ans * i.second) % mod;
  for (auto i : y) ans = (ans * i.second) % mod;
  cout << ans;
}
