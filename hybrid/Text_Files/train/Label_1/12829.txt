#include <bits/stdc++.h>
using namespace std;
#define sz(x) (int)(x).size()
#define FOR(i, a, b) for (int i = (a); i < (b); ++i)
typedef long long ll;

const int mx = 101;
ll arr[mx][mx], dp[mx][mx];
int main() {
    int n, m, k;
    cin >> n >> m >> k;
    FOR(i, 0, n) {
        FOR(j, 0, m) {
            cin >> arr[i][j];
        }
    }

    FOR(i, 0, n) { dp[i][m] = 0; }
    FOR(j, 0, m) { dp[n][j] = 0; }

    for (int i = n - 1; i >= 0; i--) {
        for (int j = m - 1; j >= 0; j--) {
            if (arr[i][j] == 1) {
                dp[i][j] = 1 + dp[i + 1][j];
                if (i + k < n) dp[i][j] = max(dp[i][j], 1 + dp[i + k][j]);
            }
            else {
                dp[i][j] = max(dp[i + 1][j], dp[i + 1][j + 1]);
            }
        }
    }

    ll mxScore = 0, mnChanges = LLONG_MAX;
    FOR(j, 0, m) { mxScore += dp[0][j]; }
    FOR(i, 0, n) {
        int totOnes = 0;
        FOR(j, 0, m) {
            totOnes += arr[i][j] == 1;
            arr[i][j] = totOnes <= dp[i][j];
        }

        ll curScore = 0, numChanges = 0;
        FOR(j, 0, m) {
            if (arr[i][j] != 1) { numChanges += (1 - arr[i][j]); }
            curScore += dp[i][j];
        }
        if (curScore == mxScore) {
            mnChanges = min(mnChanges, numChanges);
        }
    }

    cout << mxScore << " " << mnChanges << "\n";
    return 0;
}
