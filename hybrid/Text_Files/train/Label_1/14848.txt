#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int k1, k2, k3;
    std::cin >> k1 >> k2 >> k3;

    std::vector<int> a1(k1), a2(k2), a3(k3);
    for (int i = 0; i < k1; i++) std::cin >> a1[i];
    for (int i = 0; i < k2; i++) std::cin >> a2[i];
    for (int i = 0; i < k3; i++) std::cin >> a3[i];

    std::sort(a1.begin(), a1.end());
    std::sort(a2.begin(), a2.end());
    std::sort(a3.begin(), a3.end());

    int ans = 0;
    int i = 0, j = 0, k = 0;
    while (k < k3) {
        if (a3[k] < a1[i]) {
            ans++;
            k++;
        } else if (a3[k] < a2[j]) {
            ans += 2;
            k++;
        } else break;
        i++;
        j++;
    }

    std::cout << ans << std::endl;

    return 0;
}
