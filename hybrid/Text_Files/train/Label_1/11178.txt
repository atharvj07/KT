#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int sum = 0;
    for (int i : a) {
        sum += i;
    }
    int need = k * n - sum;
    if (need <= 0) {
        cout << 0 << endl;
    } else {
        cout << need << endl;
    }
    return 0;
}
