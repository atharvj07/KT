#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    string s;
    cin >> s;

    string ans = s;
    int n = s.size();
    for (int i = 0; i < n; i++) {
        string t = s;
        for (int j = i; j < n; j++) {
            if (t[j] == 'a') t[j] = 'z';
            else t[j]--;
        }
        ans = min(ans, t);
    }

    cout << ans << endl;

    return 0;
}
