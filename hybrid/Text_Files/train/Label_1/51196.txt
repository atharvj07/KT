import java.util.*;

public class Main {
    static int n, m;
    static ArrayList<Integer>[] a;
    static boolean[] can;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        a = new ArrayList[n + 1];
        can = new boolean[n + 1];
        for (int i = 1; i <= n; i++) { a[i] = new ArrayList<>(); }
        for (int i = 0; i < m; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            a[x].add(y);
            a[y].add(x);
        }
        for (int i = 1; i <= n; i++) {
            if (a[i].size() < 4) { continue; }
            Queue<Integer> qu = new LinkedList<>();
            qu.offer(i);
            can[i] = true;
            while (!qu.isEmpty()) {
                int v = qu.poll();
                for (int j : a[v]) {
                    if (can[j] || a[j].size() >= 4) { continue; }
                    qu.offer(j);
                    can[j] = true;
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (!can[i]) { System.out.println(-1); return; }
        }
        int k = 0;
        List<Integer> aa = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (a[i].size() == 1) { k++; aa.add(i); }
        }
        System.out.println(k);
        for (int i : aa) { System.out.print(i + " "); }
        System.out.println();
        k = 0;
        aa = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (a[i].size() == 1) { continue; }
            aa.add(i);
            k++;
        }
        System.out.println(k);
        for (int i : aa) { System.out.print(i + " "); }
        System.out.println();
    }
}
