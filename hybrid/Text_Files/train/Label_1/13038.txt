#include <bits/stdc++.h>
using namespace std;
int ans[200005];
int main() {
  int n, m;
  cin >> n >> m;
  string s;
  cin >> s;
  int t=0,i,j;
  for(i=0;i<n;i++){
    if(s[i]==s[t]) t++;
    else{
      s[t+1]=s[i];
      t++;
    }
    if(t>=1){
      if(s[t]!=s[t-1]) ans[i]++;
    }
    if(t>=2){
      if(s[t]==s[t-2]) ans[i]++;
    }
  }
  t=0;
  s=s.substr(0,t+1);
  for(i=0;i<n;i++){
    if(s.length()>0&&s[s.length()-1]==s[i]) s+=s[i];
    else{
      s+=s[i];
      t++;
    }
    if(t>=1){
      if(s[t]!=s[t-1]) ans[i-1]--;
    }
    if(t>=2){
      if(s[t]==s[t-2]) ans[i-1]--;
    }
  }
  for(i=0;i<m;i++){
    cin>>j;
    cout<<ans[j-1]<<'\n';
  }
}
