#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  int ans = v[0];
  int win = 0;
  while (k > win) {
    int win1 = v[0];
    if (win1 > v[1]) {
      ans = win1;
    }
    v.erase(v.begin());
    v.push_back(win1);
    win++;
  }
  cout << ans << endl;
}
