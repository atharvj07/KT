#include <bits/stdc++.h>
using namespace std;
const int maxT = 86400;
const int inf = 1e9;
int T, M, N;
int st[maxT+1], en[maxT+1];
map<int, int> mp;
vector<pair<int, int> > v;
int main() {
  cin >> N >> M >> T;
  for(int i=0; i<N; ++i) {
    int h, m, s;
    scanf("%d:%d:%d", &h, &m, &s);
    int t = (h*60*60 + m*60 + s);
    st[t] = 1;
    en[t+T] = 1;
    v.push_back({t, t+T});
  }
  for(int i=1; i<=maxT; ++i) {
    st[i] += st[i-1];
    en[i] += en[i-1];
    st[i] = min(st[i], M);
  }
  for(int t=maxT; t>=0; --t) {
    st[t] = min(st[t], st[t+1]);
    if(st[t] == M) mp[t] = 1;
  }
  for(int t=0; t<=maxT; ++t) {
    en[t] = min(en[t], en[t-1]);
    if(en[t] == M) mp[t] = 1;
  }
  int Max = 0;
  int idx = inf;
  for(auto x: mp) {
    int cur = max(st[x.first], en[x.first]);
    if(cur > Max) {
      Max = cur;
      idx = x.first;
    }
  }
  if(Max < M) {
    puts("No solution");
  } else {
    int id = 1;
    for(auto x: v) {
      if(en[x.second] >= idx) cout << id << endl;
      else id++;
    }
  }
  return 0;
}
