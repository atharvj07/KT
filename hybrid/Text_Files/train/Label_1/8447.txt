#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,c,i,l,r,k,mn=2e9,f=0,ans=0;
    cin>>n>>c;
    int a[n+1];
    for(i=1;i<=n;++i){
        cin>>a[i];
        mn=min(mn,a[i]);
    }
    for(i=1;i<=n;++i){
        if(a[i]==mn) f=1;
    }
    if(mn==c) ans=n;
    else if(f==0) ans=0;
    else{
        l=r=0;
        while(1){
            while(r+1<=n && a[r+1]==mn) r++;
            if(r==n) break;
            l=r+1;
            while(r+1<=n && a[r+1]!=mn) r++;
            if((mn+c)%2==0){
                if(abs(mn+c-a[l])%2==1){
                    k=(mn+c-a[l])/2+1;
                }
                else k=(mn+c-a[l])/2;
                if(k>=0){
                    ans=max(ans,r-l+1);
                }
            }
            l=r+1;
        }
    }
    cout<<ans<<endl;
    return 0;
}
