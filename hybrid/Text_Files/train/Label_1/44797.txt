import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] friends = new int[26];
        int ans = 0;
        String handle = "";
        for (int i = 0; i < n; i++) {
            int type = sc.nextInt();
            if (type == 1) {
                handle = "";
            } else {
                String friend = sc.next();
                if (handle.equals(friend)) {
                    ans++;
                } else if (friends[friend.charAt(0) - 'a'] == 0) {
                    friends[friend.charAt(0) - 'a'] = 1;
                    handle = friend;
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
