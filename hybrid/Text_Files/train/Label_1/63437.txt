import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
a = list(map(int, input().split()))
d = [[] for i in range(n+1)]
for i in range(m):
    x, y = map(int, input().split())
    d[x].append(y)
    d[y].append(x)

dis_1 = [0] * (n+1)
vis_1 = [0] * (n+1)
q = [1]
vis_1[1] = 1
while q:
    x = q.pop(0)
    for y in d[x]:
        if vis_1[y] == 0:
            q.append(y)
            vis_1[y] = 1
            dis_1[y] = dis_1[x] + 1

dis_n = [0] * (n+1)
vis_n = [0] * (n+1)
q = [n]
vis_n[n] = 1
while q:
    x = q.pop(0)
    for y in d[x]:
        if vis_n[y] == 0:
            q.append(y)
            vis_n[y] = 1
            dis_n[y] = dis_n[x] + 1

ans = 0
for i in a:
    for j in a:
        if i != j:
            ans = max(ans, dis_1[i] + dis_n[j] - 1)

print(ans)
