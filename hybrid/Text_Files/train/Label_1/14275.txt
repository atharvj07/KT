#include <iostream>
using namespace std;
string s;
int main() {
  cin >> s;
  for (int i = 0; i < (int)s.size(); i++) {
    if (s[i] == '1') {
      for (int j = i + 1; j < (int)s.size(); j++) {
        if (s[j] == '0') {
          s[i] = '0';
          s[j] = '1';
          break;
        }
      }
    }
  }
  cout << s << endl;
}
