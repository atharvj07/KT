import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        Map<Integer, List<String>> regionMap = new HashMap<>();
        for (int i = 0; i < n; i++) {
            line = br.readLine().split(" ");
            String name = line[0];
            int region = Integer.parseInt(line[1]);
            int score = Integer.parseInt(line[2]);
            if (!regionMap.containsKey(region)) {
                regionMap.put(region, new ArrayList<>());
            }
            regionMap.get(region).add(name + "," + score);
        }
        for (int i = 1; i <= m; i++) {
            if (!regionMap.containsKey(i) || regionMap.get(i).size() < 2) {
                System.out.println("?");
            } else {
                List<String> team = regionMap.get(i);
                team.sort(Comparator.comparing(a -> Integer.parseInt(a.substring(a.indexOf(',') + 1))));
                System.out.println(team.get(team.size() - 1).substring(0, team.get(team.size() - 1).indexOf(',')) + " " + team.get(team.size() - 2).substring(0, team.get(team.size() - 2).indexOf(',')));
            }
        }
    }
}
