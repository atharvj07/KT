#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> b(n);
  for (int i = 0; i < n; i++) cin >> b[i];
  int ans = 0;
  map<char, int> cnt;
  for (auto c : s) cnt[c]++;
  for (auto c : s) {
    int r = n - cnt[c];
    if (r < n / 2) {
      ans += b[r];
    } else {
      ans += b[r - 1];
    }
    cnt[c]--;
  }
  cout << ans << endl;
}
