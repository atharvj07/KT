import java.util.Scanner;
import java.util.TreeSet;

public class StrongPassword {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        // Create a Trie to store all the dangerous password substrings
        Trie trie = new Trie();
        for (int i = 0; i < n; i++) {
            String dangerousPassword = scanner.nextLine();
            trie.insert(dangerousPassword);
        }

        // Generate the strong password
        StringBuilder strongPassword = new StringBuilder();
        char currentChar = 'a';
        while (true) {
            // Check if the current character is not contained in any dangerous password substring
            if (!trie.containsPrefix(strongPassword.toString() + currentChar)) {
                strongPassword.append(currentChar);
                break;
            }

            currentChar++;
        }

        // Print the strong password
        System.out.println(strongPassword);
    }

    // Trie to store all the dangerous password substrings
    static class Trie {
        TrieNode root;

        Trie() {
            root = new TrieNode();
        }

        void insert(String word) {
            TrieNode node = root;
            for (char c : word.toCharArray()) {
                if (node.children[c - 'a'] == null) {
                    node.children[c - 'a'] = new TrieNode();
                }
                node = node.children[c - 'a'];
            }
            node.isEndOfWord = true;
        }

        boolean containsPrefix(String prefix) {
            TrieNode node = root;
            for (char c : prefix.toCharArray()) {
                if (node.children[c - 'a'] == null) {
                    return false;
                }
                node = node.children[c - 'a'];
            }
            return true;
        }
    }

    // Trie node
    static class TrieNode {
        TrieNode[] children;
        boolean isEndOfWord;

        TrieNode() {
            children = new TrieNode[26];
            isEndOfWord = false;
        }
    }
}
