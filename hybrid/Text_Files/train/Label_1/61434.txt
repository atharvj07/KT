def solve():
    n, A, cf, cm, m = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort()
    p = []
    for i in range(n):
        if a[i] == A:
            p.append(i)
    add = [0] * n
    ans = sum(a) * cm
    inc = 0
    for i in range(n - 1, -1, -1):
        add[i] = min(a[i] + 1, A)
        if p and p[-1] == i:
            p.pop()
            ans += (add[i] - a[i]) * (cm + cf)
            inc += 1
        else:
            ans += (add[i] - a[i]) * cm
    if m < ans - sum(a) * cm:
        return 'MISSION_IMPOSSIBLE'
    inc += (m - (ans - sum(a) * cm)) // (A * cm)
    m -= (m - (ans - sum(a) * cm)) // (A * cm) * A * cm
    if m > 0:
        inc += 1
    for i in range(n - 1, -1, -1):
        if m == 0:
            break
        if add[i] < A:
            m -= A - add[i]
            add[i] = A
            inc += 1
    for i in range(n):
        a[i] = add[i]
    print(ans)
    print(*a)

solve()
