import java.util.Scanner;

public class Main {
    static int t;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        t = sc.nextInt();
        while (t-- > 0) {
            long a = sc.nextLong();
            long b = sc.nextLong();
            if (a == b)
                System.out.println("First");
            else if (a > b)
                System.out.println(((b % a) | a) != 0 ? "First" : "Second");
            else
                System.out.println(((a % b) | b) != 0 ? "Second" : "First");
        }
        sc.close();
    }
}
