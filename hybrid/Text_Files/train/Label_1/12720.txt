#include <bits/stdc++.h>
using namespace std;
int main() {
    int a[6], s = 0;
    for (int i = 0; i < 6; i++) cin >> a[i], s += a[i] * a[i + 1];
    for (int x = 1; x <= 101; x++) {
        int t = x * x;
        if (t == s) {
            cout << x << '\n';
            for (int i = 0; i < x; i++) {
                for (int j = 0; j < x;) {
                    for (int k = 0; k < a[1] || j + k < x; k++) {
                        cout << char('A' + (i < a[0] && j + k < x));
                        j++;
                    }
                    if (j + a[3] > x) break;
                    for (int k = 0; k < a[3] || i + k < x; k++) {
                        cout << char('C' + (i + k < x));
                        j++;
                    }
                }
                cout << '\n';
            }
            return 0;
        }
    }
    return cout << -1, 0;
}
