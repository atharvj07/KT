import sys

MOD = int(1e9) + 7

def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)

def count_coprimes(arr):
    n = len(arr)
    dp = [[[0 for _ in range(3)] for _ in range(2)] for _ in range(n + 1)]
    dp[n][0][0] = 1
    dp[n][0][1] = 1
    dp[n][0][2] = 1
    for i in range(n - 1, -1, -1):
        for picked in range(2):
            for g in range(3):
                dp[i][picked][g] = dp[i + 1][picked][g]
                if picked == 0:
                    dp[i][picked][gcd(arr[i], g)] += dp[i + 1][1][g]
                dp[i][picked][g] %= MOD
    return dp[0][0][1]

n = int(input())
arr = [int(num) for num in input().split()]
print(count_coprimes(arr))
