#include <bits/stdc++.h>
using namespace std;
#define pi 3.1415926535897932
long double dis(pair<long long,long long>a,pair<long long,long long>b){
	return sqrt((a.first-b.first)*(a.first-b.first)+(a.second-b.second)*(a.second-b.second));
}
int main(){
	long long n,x,y,k,sum=0,i;
	cin>>n>>sum;
	while(n--){
		cin>>x>>y>>k;
		sum+=k;
	}
	if(sum<1e6){
		cout<<-1<<"\n";
		return 0;
	}
	long long mx=0;
	long double res=0;
	for(i=0;;i++){
		if(mx<i*i){
			res=i;
			mx=i*i;
		}
		if(res*res*pi>=sum)break;
	}
	cout<<fixed<<setprecision(9)<<res<<"\n";
}
