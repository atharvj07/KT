n = int(input())
a = input()
i = 1
d = {'D':1, 'R':-1}[a[0]]
while(n>1):
    if(d==1):
        while(i<=n and a[i-1]=='R'):
            i += 1
        if(i<=n):
            d = -d
        n -= 1
        i += 1
    else:
        while(i<=n and a[i-1]=='D'):
            i += 1
        if(i<=n):
            d = -d
        n -= 1
        i += 1
print(['R','D'][d==-1])
