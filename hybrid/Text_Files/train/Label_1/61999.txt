import io
n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
c = list(map(int, input().split()))
a.sort()
b.sort()
c.sort()
i, j = 0, 0
ans = 0
best = 0
for aa in a:
    while i < m and b[i] < aa:
        i += 1
    while j < m and c[j] < aa:
        j += 1
    if i < m and b[i] == aa:
        ans += 1
    elif j < m and c[j] == aa:
        best += 1
if ans + best == 0:
    print(1)
else:
    ans += best
    print(ans + (best > 0))
