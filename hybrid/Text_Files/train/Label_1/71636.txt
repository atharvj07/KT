import sys

for line in sys.stdin.readlines()[1:]:
    s = list(line.strip())
    for i in range(1, len(s)):
        if s[i] == '?' and s[i - 1] == s[i - 1 if i - 1 >= 0 else len(s) - 1]:
            for j in 'abc':
                if j != s[i - 1]:
                    s[i] = j
                    break
    print(''.join(s).replace('?', 'a') if all(s[i] != s[i - 1] for i in range(1, len(s))) else -1)
