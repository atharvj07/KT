import math

def main():
    n, a, b, k = map(int, input().split())
    misses = input()
    misses = [i for i, x in enumerate(misses) if x == '1']
    ans = math.ceil((n - k) / (a + b - 1))
    pos = []
    for i in range(ans):
        if i * (a + b - 1) + b <= n:
            pos.append(i * (a + b - 1) + b)
        else:
            pos.append(n)
    print(ans)
    print(*pos)

main()
