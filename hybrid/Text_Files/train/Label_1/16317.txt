#include <iostream>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
const int maxn = 1e5 + 5;
int n, k;
ll a[maxn], pw[maxn], inv[maxn];
ll powMod(ll a, ll b) {
    ll ans = 1;
    while (b) {
        if (b & 1) { ans = ans * a % mod; }
        b >>= 1;
        a = a * a % mod;
    }
    return ans;
}
int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    pw[0] = 1;
    for (int i = 1; i <= n; i++) pw[i] = pw[i - 1] * a[i] % mod;
    inv[n] = powMod(pw[n], mod - 2);
    for (int i = n - 1; i >= 0; i--) inv[i] = inv[i + 1] * a[i + 1] % mod;
    ll ans = 0, tmp = (pw[n] - 1 + mod) % mod * inv[1] % mod;
    for (int i = 0; i <= n; i++) {
        ans = (ans + pw[i] * inv[i + 1] % mod * tmp) % mod;
        tmp = tmp * inv[i + 2] % mod;
    }
    for (int i = 1; i <= n; i++) {
        ans = (ans + k % mod * pw[n - i - 1] % mod * inv[n - i] % mod) % mod;
    }
    ans = (ans % mod + mod) % mod;
    cout << ans << endl;
}
