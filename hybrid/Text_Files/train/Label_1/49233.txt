import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();
        char[] cs = s.toCharArray();
        char[] rs = new char[n];
        for (int i = 0; i < n; i++) {
            rs[i] = cs[n - i - 1];
        }
        char[] us = new char[2 * n];
        for (int i = 0; i < n; i++) {
            us[i] = cs[i];
            us[i + n] = rs[i];
        }
        char[] ans = new char[n];
        int p = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < i + n; j++) {
                if (us[j] < us[p]) {
                    p = j;
                }
            }
            ans[i] = us[p];
            p++;
            p = p % (2 * n);
        }
        int l = k % (2 * n - 1);
        char[] ans2 = new char[n];
        for (int i = 0; i < n; i++) {
            ans2[(i + l) % n] = ans[i];
        }
        System.out.println(new String(ans2));
    }
}
