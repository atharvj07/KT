#include <bits/stdc++.h>
using namespace std;
const int inf = 1000000000;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int l = 0, r = 0, k = 0;
  vector<pair<int, int>> ans;
  while (r < n) {
    while (r < n && a[r] == a[l]) r++;
    if (r == n) break;
    k++;
    ans.push_back({l + 1, r});
    l = r;
  }
  if (k == 0) cout << -1 << endl;
  else {
    cout << k << endl;
    for (auto p : ans) cout << p.first << " " << p.second << endl;
  }
  return 0;
}
