import java.util.*;
import java.io.*;

public class Main {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int[] a = new int[n+1];
		Set<Integer> set = new HashSet<>();
		int cnt = 0;
		for (int i = 1; i <= n; i++) {
			int x = in.nextInt();
			a[x] = i;
			if (set.add(x)) {
				cnt++;
			}
		}
		int ans = cnt;
		int last = 1;
		for (int i = 1; i <= n; i++) {
			if (a[i] > last) {
				last = a[i];
				ans--;
			}
		}
		System.out.println(ans);
	}
}
