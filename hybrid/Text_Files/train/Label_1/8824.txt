#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  if (n & 1) {
    cout << "No" << endl;
    return 0;
  }
  bool f = 0, l = 0;
  for (int x : a) f |= x % 2;
  reverse(a.begin(), a.end());
  for (int x : a) l |= x % 2;
  if (f && l)
    cout << "Yes" << endl;
  else
    cout << "No" << endl;
}
