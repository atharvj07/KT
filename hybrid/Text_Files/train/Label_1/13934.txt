#include <bits/stdc++.h>
using namespace std;
const int maxa = 2e5 + 6;
vector<int> pos[maxa];
int p[maxa], a[maxa];
int main() {
  int n, m, q;
  cin >> n >> m >> q;
  for (int i = 0; i < n; i++) {
    cin >> p[i];
    pos[p[i]].push_back(i);
  }
  for (int i = 0; i < m; i++) {
    cin >> a[i];
  }
  while (q--) {
    int l, r;
    cin >> l >> r;
    cout << (pos[a[l - 1]].size() && pos[a[r - 1]].size() && pos[a[r - 1]][lower_bound(pos[a[r - 1]].begin(), pos[a[r - 1]].end(), pos[a[l - 1]][0] + n - r + l, greater<int>()) - pos[a[r - 1]].begin()] - 1) << "\n";
  }
  return 0;
}
