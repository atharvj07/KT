#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> dp(n);
  dp[n - 2] = n % 2;
  dp[n - 1] = 1;
  for (int i = n - 3; i >= 0; i--) {
    if (s[i] == s[i + 2]) {
      dp[i] = dp[i + 2] + 2;
    } else {
      dp[i] = dp[i + 1];
    }
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (dp[i] > 0) {
      ans += dp[i];
    } else {
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
