#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> words(n);
  for (int i = 0; i < n; i++) {
    cin >> words[i];
  }
  sort(words.begin(), words.end(), [](const string &a, const string &b) {
    return a.size() > b.size();
  });
  int ans = 0;
  for (const string &word : words) {
    bool used[26] = {};
    bool ok = true;
    for (char c : word) {
      if (used[c - 'a']) {
        ok = false;
        break;
      }
      used[c - 'a'] = true;
    }
    if (ok) {
      ans += word.size();
    }
  }
  cout << ans << endl;
  return 0;
}
