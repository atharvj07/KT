#include <bits/stdc++.h>
using namespace std;
const int N = 7010, LEN = 60;
typedef long long ll;
ll a[N], b[N], dp[N][1 << LEN], pw[LEN];
ll getbit(ll x, int bit) {
    return (x >> bit) & 1ll;
}
ll f(int pos, int mask) {
    if(pos == 0) {
        if(mask > 0)
            return 0;
        else
            return -1e18;
    }
    ll &ret = dp[pos][mask];
    if(ret != -1)
        return ret;
    ret = f(pos - 1, mask);
    ll ans = 0;
    for(int x = 0; x <= 59; x++) {
        if(getbit(mask, x) == 0) {
            if(getbit(a[pos], x) == 1) {
                ans = max(ans, f(pos - 1, mask | (1 << x)) + b[pos]);
            }
        }
    }
    ret = max(ret, ans);
    return ret;
}
int main() {
    memset(dp, -1, sizeof dp);
    int n;
    cin >> n;
    for(int i = 1; i <= n; i++)
        cin >> a[i];
    for(int i = 1; i <= n; i++)
        cin >> b[i];
    pw[0] = 1ll;
    for(int i = 1; i <= 59; i++)
        pw[i] = pw[i - 1] * 2ll;
    cout << f(n, 0) << "\n";
    return 0;
}
