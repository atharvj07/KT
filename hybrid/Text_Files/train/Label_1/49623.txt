import java.util.Scanner;

public class Main {
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        int b = sc.nextInt(), d = sc.nextInt();
        boolean[] div2 = new boolean[b];
        boolean[] div4 = new boolean[b * b];
        boolean[] div10 = new boolean[b];

        int sum = 0, last = 0;
        for (int i = 0; i < b; i++) {
            last = (last * b + i) % d;
            if (i % 2 == 0) sum = (sum + i) % d;
            div2[i] = last == 0;
            div4[i * b] = last == 0;
            div10[i] = i == 0;
        }

        for (int i = 1; i < b; i++) {
            sum = (sum + last) % d;
            last = (last * b + last) % d;
            div4[(i * b) + i] = last == 0;
        }

        if (d == 2) System.out.println("2-type\n1");
        else if (d == 3 || d == 9) System.out.println("3-type");
        else if (d == 11) System.out.println("11-type");
        else if (d == 4 || d == 8) {
            if (div2[b - 1] && div4[(b - 1) * b + (b - 1)])
                System.out.println("2-type\n" + ((d == 4) ? 2 : 3));
            else
                System.out.println("7-type");
        } else if (d == 6) {
            if (div2[b - 1] && div3(sum))
                System.out.println("6-type");
            else
                System.out.println("7-type");
        } else if (d == 5 || d == 10) {
            if (div2[b - 1] && div10[b - 1])
                System.out.println("2-type\n1");
            else
                System.out.println("7-type");
        } else if (d == 15) {
            if (div3(sum) && div2[b - 1] && div4[(b - 1) * b + (b - 1)])
                System.out.println("6-type");
            else
                System.out.println("7-type");
        } else {
            System.out.println("7-type");
        }
    }

    private static boolean div3(int sum) {
        while (sum > 0) {
            if (sum % 3 != 0) return false;
            sum /= 10;
        }
        return true;
    }
}
