#include <bits/stdc++.h>
using namespace std;
vector<pair<int, int>> ans;
void f(vector<vector<int>> &a, int x, int y) {
  ans.emplace_back(x, y);
  a[x][y] ^= 1, a[x][y + 1] ^= 1, a[x + 1][y] ^= 1, a[x + 1][y + 1] ^= 1;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, m;
  cin >> n >> m;
  vector<vector<int>> a(n + 2, vector<int>(m + 2)), b(n + 2, vector<int>(m + 2));
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) cin >> a[i][j];
  }
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      if (a[i][j] != b[i][j]) f(b, i - 1, j - 1);
    }
  }
  bool ok = 1;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) ok &= (a[i][j] == b[i][j]);
  }
  if (!ok) {
    cout << -1 << endl;
    return 0;
  }
  cout << ans.size() << endl;
  for (const auto &p : ans) cout << p.first + 1 << ' ' << p.second + 1 << '\n';
}
