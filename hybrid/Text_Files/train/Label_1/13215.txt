#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    int a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 0;
    int mx = x;
    for (int i = 0; i < n; i++) {
      if (a[i] > mx) ans++;
      mx = max(mx, a[i]);
    }
    cout << (ans <= n ? ans : -1) << endl;
  }
  return 0;
}
