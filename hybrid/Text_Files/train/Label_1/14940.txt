#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define pii pair<int,int>
#define vii vector<pii>
#define ll long long

const int N = 1e5+10;
bool vis[N];

int32_t main(){
    int n,q;
    cin >> n >> q;
    set<int>s;
    for(int i=0;i<=n;i++) s.insert(i);
    for(int i=1;i<=n;i++) vis[i]=0;
    while(q--){
        int x,y,z;
        cin>>x>>y>>z;
        set<int>::iterator it;
        int ans=0;
        for(int j=1;j<=n;j++){
            if(vis[j]) continue;
            if(s.find(x)!=s.end()){
                if(y==0){
                    cout<<-1<<"\n";
                    goto l;
                }
                it=s.lower_bound(x);
                ans+=*it;
                s.erase(it);
                vis[j]=1;
                x+=z;
                y--;
            }
        }
        if(y>0) cout<<-1<<"\n";
        else cout<<ans<<"\n";
        l:;
    }
    return 0;
}
