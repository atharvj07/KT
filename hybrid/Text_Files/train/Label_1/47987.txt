import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int s = sc.nextInt();
        int[] li = new int[s];
        int[] ri = new int[s];
        int[] qi = new int[s];
        for (int i = 0; i < s; i++) {
            li[i] = sc.nextInt();
            ri[i] = sc.nextInt();
            qi[i] = sc.nextInt();
        }
        int[] ans = new int[s];
        for (int i = 0; i < s; i++) {
            int min = Integer.MAX_VALUE;
            for (int j = li[i]; j <= ri[i]; j++) {
                int sum = 0;
                for (int k = j; k < j + m; k++) {
                    if (a[k - 1] < qi[i]) {
                        sum++;
                    }
                }
                min = Math.min(min, sum);
            }
            ans[i] = min;
        }
        for (int i = 0; i < s; i++) {
            System.out.println(ans[i]);
        }
    }
}
