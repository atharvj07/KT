import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        for (int i = 0; i < n; i++) {
            int[][] mesh = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    mesh[j][k] = in.nextInt();
                }
            }
            if (checkMesh(mesh)) {
                System.out.println("true");
            } else {
                System.out.println("false");
            }
        }
        in.close();
    }

    public static boolean checkMesh(int[][] mesh) {
        // Check if the mesh is a cube.
        for (int i = 0; i < 5; i++) {
            for (int j = 0; j < 5; j++) {
                if (i == 0 || i == 4 || j == 0 || j == 4) {
                    if (mesh[i][j] == 0) {
                        return false;
                    }
                }
            }
        }

        // Check if the sum of two numbers on the opposite faces is always 7.
        for (int i = 0; i < 5; i++) {
            if (mesh[i][0] + mesh[i][4] != 7) {
                return false;
            }
            if (mesh[0][i] + mesh[4][i] != 7) {
                return false;
            }
        }

        // Check if the numbers on the faces of the cube are different from each other.
        for (int i = 0; i < 5; i++) {
            for (int j = 0; j < 5; j++) {
                for (int k = i + 1; k < 5; k++) {
                    for (int l = j + 1; l < 5; l++) {
                        if (mesh[i][j] == mesh[k][l]) {
                            return false;
                        }
                    }
                }
            }
        }

        return true;
    }
}
