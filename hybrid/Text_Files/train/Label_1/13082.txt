#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int M = 5e5+5;
int a[M];
ll dp[M];
const ll inf = -1e15;
int main(){
    int n;cin>>n;
    for(int i=1;i<=n;i++)cin>>a[i];
    dp[0]=0;
    dp[1]=0;
    dp[2]=a[1]+a[2];
    for(int i=3;i<=n;i++){
        dp[i]=max(dp[i-1],dp[i-2]+max({a[i-1],a[i]}));
    }
    cout<<dp[n]<<endl;
    return 0;
}
