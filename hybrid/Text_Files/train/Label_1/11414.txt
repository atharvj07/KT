#include<bits/stdc++.h>
using namespace std;
struct Queue
{
	int l,r,len,cnt;
}q[500005];
int main()
{
	int n,k,l=1,r=1,ans_l=1,ans_r,cnt=0;
	scanf("%d %d",&n,&k);
	int a[n+5],vis[1000005]={0},ans=0;
	for(int i=1;i<=n;i++)	
	{  
		scanf("%d",&a[i]);
		if(!vis[a[i]]) cnt++;
		vis[a[i]]++;
	}
	q[1].l=1,q[1].r=1,q[1].cnt=1,q[1].len=1;
	for(int i=1;i<=n;i++)
	{
		int temp=q[i].len;
		while(temp&&q[temp].cnt<=k&&q[temp].r<i)
		{
			l=q[temp].l,r=q[temp].r,ans=temp,temp--;
		}
		if(r-l+1>ans) ans_l=l,ans_r=r;
		if(vis[a[i]]>1) q[i].cnt=q[i-1].cnt-1;
		else q[i].cnt=q[i-1].cnt+1;
		q[i].l=i,q[i].r=i,q[i].len=i;
		vis[a[i]]--;
	}
	printf("%d %d\n",ans_l,ans_r);
	return 0;
}
