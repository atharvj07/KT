#include <vector>
#include <algorithm>
#include <iostream>

typedef std::vector<int> vi;
typedef std::vector<vi> vvi;

vi solve(int n, int c0, int c1) {
    vi dp(n + 1);
    std::vector<bool> vis(1 << n);
    const int inf = 1e9;
    dp[0] = 0;
    for (int mask = 0; mask < (1 << n); ++mask) {
        const int total0 = __builtin_popcount(~mask);
        const int total1 = n - total0;
        const int cost = total0 * c0 + total1 * c1;
        dp[total0] = std::min(dp[total0], cost);
        for (int i = 0; i < n; ++i) {
            if (((mask >> i) & 1) == 0) {
                const int newMask = mask | (1 << i);
                if (!vis[newMask]) {
                    vis[newMask] = true;
                    dp[total0 + 1] = std::min(dp[total0 + 1], cost);
                }
            }
        }
    }
    return dp;
}

int main() {
    int n, c0, c1;
    std::cin >> n >> c0 >> c1;
    const vi& dp = solve(n, c0, c1);
    std::cout << dp[n] << std::endl;
    return 0;
}
