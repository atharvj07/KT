#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
int main()
{
    ll t, n, ans;
    cin>>t;
    for(ll test=1; test<=t; test++)
    {
        cin>>n;
        if(n<=3){
            ans=n-1;
        }
        else if(n%2==0)
        {
            ans=(n/2);
        }
        else{
            ans=(n/2)+1;
        }
        cout<<ans<<endl;
    }
}
