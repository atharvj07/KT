#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    if (k > n * (n + 1) / 2) {
        cout << -1 << endl;
        return 0;
    }
    int r = n, c = (n + 1) / 2;
    vector<vector<int>> matrix(n, vector<int>(n));
    while (k > 0) {
        matrix[r - 1][c - 1] = 1;
        k--;
        r--;
        c++;
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << matrix[i][j] << " ";
        }
        cout << endl;
    }
    return 0;
}
