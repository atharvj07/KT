import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    static long mod = 998244353;

    static List<Integer> restorePermutation(int[] p, int n) {
        List<Integer> result = new ArrayList<>();
        int[] notPresent = new int[n + 1];
        Arrays.fill(notPresent, 1);
        for (int i : p) {
            if (i != -1) {
                notPresent[i] = 0;
                result.add(i);
            }
        }

        for (int i = 1; i <= n; i++) {
            if (notPresent[i] == 1) {
                result.add(i);
            }
        }

        return result;
    }

    static long countInversions(List<Integer> perm) {
        int[] p = new int[perm.size()];
        for (int i = 0; i < perm.size(); i++) {
            p[i] = perm.get(i);
        }
        long ans = 0;
        for (int i = 0; i < p.length; i++) {
            for (int j = i + 1; j < p.length; j++) {
                if (p[i] > p[j]) {
                    ans++;
                }
            }
        }
        return ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String[] line = br.readLine().split(" ");
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = Integer.parseInt(line[i]);
        }
        List<Integer> perm = restorePermutation(p, n);
        long inversions = countInversions(perm);
        long pow = modPow(2, n * (n - 1) / 2 - inversions, mod);
        long result = (inversions * pow) % mod;
        System.out.println(result);
    }

    static long modPow(long base, long pow, long mod) {
        if (pow == 0) {
            return 1;
        }
        if (pow == 1) {
            return base;
        }
        if (pow % 2 == 0) {
            long half = modPow(base, pow / 2, mod);
            return (half * half) % mod;
        }
        return (base * modPow(base, pow - 1, mod)) % mod;
    }
}
