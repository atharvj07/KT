n, m = map(int, input().split())
matrix = [list(map(int, input().split())) for _ in range(n)]

def min_moves_to_sort_matrix(n, m, matrix):
    # Initialize the counter for moves
    moves = 0
    
    # Process each column independently
    for j in range(m):
        # Create a list to keep track of the number of shifts needed for each row to be in its correct position
        shift_counts = [0] * n
        for i in range(n):
            correct_value = (i * m) + (j + 1)  # Calculate the correct value that should be in cell (i, j)
            if (matrix[i][j] - 1) % m == j:  # Check if the current cell value could possibly be corrected by shifting
                correct_row = (matrix[i][j] - 1) // m  # Calculate the row where the current cell value should be
                shift_needed = (i - correct_row) % n  # Calculate the number of shifts needed to get it there
                shift_counts[shift_needed] += 1
                
        # For the current column, find the minimum number of shifts needed by considering the maximum number of cells
        # that can be placed correctly with the same number of shifts. The remaining cells will have to be changed individually.
        max_correct_cells = max(shift_counts)
        moves += n - max_correct_cells  # Add the minimum number of moves needed for the current column
    
    return moves

print(min_moves_to_sort_matrix(n, m, matrix))
