#include<bits/stdc++.h>
using namespace std;
int main() {
    int tt; cin >> tt;
    while (tt--) {
        int n; cin >> n;
        vector<int> p(n);
        for (int i = 0; i < n; i++) {
            cin >> p[i];
        }
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            if (p[i] > p[i + 1]) {
                int j = i + 1;
                while (j < n && p[j] < p[i]) {
                    j++;
                }
                ans = max(ans, j - i);
            }
        }
        cout << ans << '\n';
    }
}
