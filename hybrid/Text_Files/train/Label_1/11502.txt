#include<bits/stdc++.h>
using namespace std;
const int MAXN=100;
int k;
char s[MAXN];
bool palindrome(){
    int len=strlen(s);
    for(int l=0,r=len-1;l<=r;l++,r--)if(s[l]!=s[r])return 0;
    return 1;
}
int main(){
    cin>>s>>k;
    int len=strlen(s);
    if(len%k)return cout<<"NO",0;
    for(int i=len/k;i<len;i+=len/k)if(!palindrome())return cout<<"NO",0;
    return cout<<"YES",0;
}
