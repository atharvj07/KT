#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n, m, q;
  cin >> n >> m >> q;
  string s, t;
  cin >> s >> t;
  vector<int> pref(n + 1, 0);
  for (int i = 0; i < m; i++) {
    pref[0] += (s[i] == t[0]);
  }
  for (int i = 1; i < n; i++) {
    pref[i] = pref[i - 1];
    if (i + m - 1 < n) {
      pref[i] -= (s[i - m] == t[0]);
      pref[i] += (s[i + m - 1] == t[0]);
    }
  }
  for (int i = 0; i < q; i++) {
    int l, r;
    cin >> l >> r;
    l--, r--;
    cout << (l == 0 ? pref[r] : pref[r] - pref[l - 1]) << endl;
  }
  return 0;
}
