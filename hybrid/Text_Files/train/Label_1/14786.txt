#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int flag = 0;
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            if (x % (i + 2) == 0) {
                flag = 1;
                break;
            }
        }
        if (!flag) {
            cout << "YES" << '\n';
        } else {
            cout << "NO" << '\n';
        }
    }
    return 0;
}
