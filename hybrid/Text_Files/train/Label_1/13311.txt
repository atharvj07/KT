#include <iostream>

int main() {
  int k;
  std::cin >> k;
  int sol = 0, sum = 0, digit;
  for (sol = 19; sol <= 100000; sol++) {
    sum = 0;
    for (int i = sol; i != 0; i /= 10) {
      digit = i % 10;
      sum += digit;
    }
    if (sum == 10) k--;
    if (k == 0) break;
  }
  std::cout << sol << std::endl;
}
