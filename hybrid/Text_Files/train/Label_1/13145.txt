#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, L;
  cin >> n >> L;
  vector<int> v(n);
  for (auto& x : v) { cin >> x; }
  sort(v.begin(), v.end());
  int sum = 0, i = 0;
  while (sum < L) {
    sum += (1 << i) - 1;
    i++;
  }
  cout << v[i - 1] << endl;
  return 0;
}
