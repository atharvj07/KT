import sys

n,d,h = map(int,sys.stdin.readline().split())
if n <= 3 or h <= 1 or h*(h+1)/2 < n or d > (n-h - (n&1)):
    print(-1)
else:
    if n&1:
        print('1 2')
        print('2',n)
        n-=2
        le=3
    else:
        print('1 2')
        n-=2
        le=2
    st=[]
    cur=2
    for i in range(1,h):
        tmp=n-i
        if tmp%2 == 0:
            print(2,cur+1)
            print(2,cur+2)
            cur+=2
            n-=2
        else:
            print('2',cur+1)
            st.append(cur+1)
            cur+=1
            n-=1
    print(le,cur+1)
    while st and n>0:
        print(le,st.pop())
        le+=1
        n-=1
