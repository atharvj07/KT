#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> f(n);
  for (auto &x : f) cin >> x;
  int ans = 0, cur = 0, pre = 0;
  for (int j = 0; j < m; ++j) {
    for (int i = 0; i < n; ++i) {
      if (f[i][j] == '.') continue;
      if (pre - cur > 1) cur = 0;
      if (pre == cur) ++cur;
      ++pre;
      if (ans < cur) ans = cur;
    }
    cur = 0;
    pre = 0;
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      if (f[i][j] == '.') continue;
      if (pre - cur > 1) cur = 0;
      if (pre == cur) ++cur;
      ++pre;
      if (ans < cur) ans = cur;
    }
    cur = 0;
    pre = 0;
  }
  if (ans > 1 && (ans == n || ans == m)) {
    puts("YES");
  } else {
    puts("NO");
  }
}
