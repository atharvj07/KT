#include <iostream>
#include <vector>
#include <utility>

using namespace std;

int main() {
  int n, s;
  cin >> n >> s;

  vector<pair<int, int>> landings;
  for (int i = 0; i < n; i++) {
    int h, m;
    cin >> h >> m;
    landings.push_back(make_pair(h, m));
  }

  for (int i = 0; i < n - 1; i++) {
    if (landings[i + 1].first - landings[i].first > 1 ||
        (landings[i + 1].first - landings[i].first == 1 &&
         landings[i + 1].second - landings[i].second >= s)) {
      cout << landings[i].first + 1 << " " << landings[i].second << endl;
      return 0;
    }
  }

  cout << landings[n - 1].first + 1 << " " << landings[n - 1].second << endl;
  return 0;
}
