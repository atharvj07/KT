#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1000000007;
ll n, ans, k;
void sub1() {
  for (ll x = 0; x <= 1000000000000000000LL; x++) {
    ll cur = 0, tmp = x;
    while (tmp) {
      cur = cur * n + (tmp % 10LL);
      tmp /= 10LL;
    }
    if (cur == k) {
      ans = x;
      return;
    }
  }
}
int main() {
  cin >> n >> k;
  sub1();
  cout << ans;
  return 0;
}
