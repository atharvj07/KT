#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;
    vector<int> l(m), p(m);
    for(auto &x:l) cin>>x;
    for(int i=0;i<m;i++){
        if(n-l[i]<i) {cout<<"-1"; return 0;}
        p[i]=n-l[i]+1-i;
        for(int j=0;j<l[i];j++){
            if(p[i]+j<=n){
                for(int k=p[i];k<p[i]+l[i];k++){
                    if(k==p[i]+j) cout<<i+1<<" ";
                    else cout<<0<<" ";
                }
                cout<<"\n";
            }
            else{
                for(int k=0;k+j<l[i];k++) cout<<0<<" ";
                for(int k=0;k<p[i]+l[i]-n;k++) cout<<i+1<<" ";
                cout<<"\n";
            }
        }
        n-=l[i];
    }
    return 0;
}
