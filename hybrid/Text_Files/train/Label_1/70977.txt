MOD = 10**9 + 7

n = int(input())
s = input()
a = list(map(int, input().split()))

# Precompute the max valid length for each position
max_valid = [0] * n
for i in range(n):
    current_max = 0
    for j in range(i, n):
        current_max = max(current_max, a[ord(s[j]) - ord('a')])
        if j - i + 1 > current_max:
            break
        max_valid[i] = max(max_valid[i], j - i + 1)

# Compute the number of ways to split
dp_ways = [0] * (n + 1)
dp_ways[0] = 1
for i in range(1, n + 1):
    for j in range(1, max_valid[i - 1] + 1):
        dp_ways[i] = (dp_ways[i] + dp_ways[i - j]) % MOD

# Compute the minimum number of substrings
dp_min = [float('inf')] * (n + 1)
dp_min[0] = 0
for i in range(1, n + 1):
    for j in range(1, max_valid[i - 1] + 1):
        dp_min[i] = min(dp_min[i], dp_min[i - j] + 1)

print(dp_ways[n])
print(max(max_valid))
print(dp_min[n])
