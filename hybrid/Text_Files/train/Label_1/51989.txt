import java.util.*;
public class Main {
    public static void main(String[] args){
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while(t-->0){
            int n = input.nextInt();
            int[] a = new int[n];
            for(int i=0; i<n; i++) a[i] = input.nextInt();
            int ans = 0;
            for(int i=0; i<n-1; i++){
                if(a[i]>a[i+1]) ans++;
            }
            System.out.println(ans);
        }
    }
}
