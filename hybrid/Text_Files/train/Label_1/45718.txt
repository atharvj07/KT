import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
  public static void main(String[] args) throws java.lang.Exception {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[] a = new int[n];
    int[] b = new int[m];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    for (int i = 0; i < m; i++) {
      b[i] = sc.nextInt();
    }
    int[][] ans = new int[n][m];
    boolean flag = false;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (i == 0) {
          ans[i][j] = b[j];
        } else {
          ans[i][j] = a[i] ^ ans[i - 1][j];
        }
        if (ans[i][j] < 0 || ans[i][j] > 2 * Math.pow(10, 9)) {
          flag = true;
          break;
        }
      }
      if (flag) {
        break;
      }
    }
    for (int i = 0; i < m; i++) {
      for (int j = 0; j < n; j++) {
        if (i == 0) {
          ans[j][i] = a[j];
        } else {
          ans[j][i] = b[i] ^ ans[j][i - 1];
        }
        if (ans[j][i] < 0 || ans[j][i] > 2 * Math.pow(10, 9)) {
          flag = true;
          break;
        }
      }
      if (flag) {
        break;
      }
    }
    if (flag) {
      System.out.println("NO");
    } else {
      System.out.println("YES");
      for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
          System.out.print(ans[i][j] + " ");
        }
        System.out.println();
      }
    }
  }
}
