import java.util.Scanner;

public class VasyaAndRectangles {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        char[][] grid = new char[n][m];

        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = line.charAt(j);
            }
        }

        boolean hasTwoFrames = findTwoFrames(grid);
        System.out.println(hasTwoFrames ? "YES" : "NO");

        if (hasTwoFrames) {
            int[] frame1 = findFrame(grid);
            int[] frame2 = findFrame(grid);
            System.out.println(frame1[0] + " " + frame1[1] + " " + frame1[2] + " " + frame1[3]);
            System.out.println(frame2[0] + " " + frame2[1] + " " + frame2[2] + " " + frame2[3]);
        }
    }

    static boolean findTwoFrames(char[][] grid) {
        int[] frame1 = findFrame(grid);
        int[] frame2 = findFrame(grid);
        return frame1 != null && frame2 != null;
    }

    static int[] findFrame(char[][] grid) {
        for (int i = 0; i < grid.length; i++) {
            for (int j = 0; j < grid[0].length; j++) {
                if (grid[i][j] == '#' && isFrame(grid, i, j)) {
                    return new int[] { i, j, i, j };
                }
            }
        }
        return null;
    }

    static boolean isFrame(char[][] grid, int x, int y) {
        if (x == 0 || x == grid.length - 1 || y == 0 || y == grid[0].length - 1) {
            return false;
        }

        if (grid[x][y] != '#') {
            return false;
        }

        int minX = x, maxX = x, minY = y, maxY = y;
        while (minX > 0 && grid[minX - 1][y] == '#') {
            minX--;
        }
        while (maxX < grid.length - 1 && grid[maxX + 1][y] == '#') {
            maxX++;
        }
        while (minY > 0 && grid[x][minY - 1] == '#') {
            minY--;
        }
        while (maxY < grid[0].length - 1 && grid[x][maxY + 1] == '#') {
            maxY++;
        }

        if (maxX - minX >= 2 && maxY - minY >= 2) {
            for (int i = minX + 1; i < maxX; i++) {
                if (grid[i][minY] != '#' || grid[i][maxY] != '#') {
                    return false;
                }
            }
            for (int i = minY + 1; i < maxY; i++) {
                if (grid[minX][i] != '#' || grid[maxX][i] != '#') {
                    return false;
                }
            }
            return true;
        }
        return false;
    }
}
