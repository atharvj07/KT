#include <bits/stdc++.h>
#define ll long long
using namespace std;

ll powMod(ll base, ll ex, ll mod) {
    if (ex == 0) return 1LL;
    if (ex == 1) return base % mod;
    ll tmp = powMod(base, ex / 2, mod);
    tmp = (tmp * tmp) % mod;
    if (ex % 2 == 1) {
        tmp = (tmp * base) % mod;
    }
    return tmp;
}

int main() {
    ll a, b, n, x;
    cin >> a >> b >> n >> x;
    cout << powMod(a, n, 1000000007) * x % 1000000007 + powMod(a, n - 1, 1000000007) * b % 1000000007 << '\n';
}
