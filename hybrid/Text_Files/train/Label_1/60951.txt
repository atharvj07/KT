n, k = [int(x) for x in input().split()]
arr = [int(x) for x in input().split()]
pre = [0 for i in range(n+1)]
for i in range(n):
    pre[i+1] = pre[i] + arr[i]
ans = pre[k]
idx = 1
for i in range(k, n+1):
    if ans > pre[i] - pre[i-k]:
        ans = pre[i] - pre[i-k]
        idx = i - k + 1
print(idx)
