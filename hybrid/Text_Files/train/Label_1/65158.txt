def overlap():
    ok = False
    for i in range(1, n + 2):
        x = bin(xmask & mask[i])[2:]
        y = bin(ymask & mask[i])[2:]
        if len(x) < 16:
            x = '0' * (16 - len(x)) + x
        if len(y) < 16:
            y = '0' * (16 - len(y)) + y
        if x[8:] == y[8:]:
            ok = True
    return ok

n = int(input())
mask = [0] * (n + 3)
xmask = 0
ymask = 0
for i in range(1, n + 1):
    x1, y1, x2, y2 = map(int, input().split())
    xmask |= (1 << (x2 + 1)) - 1 ^ (1 << x1)
    ymask |= (1 << (y2 + 1)) - 1 ^ (1 << y1)
    mask[i] = (1 << (x2 + 1)) - 1 ^ (1 << x1) ^ ((1 << (y2 + 1)) - 1 ^ (1 << y1))
if overlap():
    print('YES')
else:
    print('NO')
