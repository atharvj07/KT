#include<bits/stdc++.h>
using namespace std;
int main()
{
    int i,n,x=0,j,d=0,a[11],f,s,ans;
    a[0]=a[1]=a[2]=a[3]=a[4]=a[5]=a[6]=a[7]=a[8]=a[9]=1;
    cin>>n;
    for(i=2;i<=10;i++)
    {
        if(a[i]==1)
        {
            for(j=2*i;j<=10;j+=i)a[j]=0;
        }
    }
    for(i=1;i<=n;i++)
    {
        d=0;
        x=i;
        while(x>1)
        {
            f=x/2;
            s=x/5;
            if(f==s)
            {
                d=1;
                break;
            }
            else if(x%2==0)x=x/2;
            else if(x%5==0)x=x/5;
            else break;
        }
        if(d==0&&x==1)ans++;
    }
    cout<<ans<<endl;
    return 0;
}
