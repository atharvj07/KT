import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), t = sc.nextInt(), ans = 0;
        String s = sc.next();
        char[] a = s.toCharArray();
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'B') {
                ans += i;
                if (ans >= t) {
                    break;
                }
                a[i] = 'G';
                a[i + 1] = 'B';
            }
        }
        System.out.println(new String(a));
    }
}
