import java.util.*;

class Main {
    static int MOD = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[][] dist = new long[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = sc.nextInt();
            }
        }
        long ans = 0;
        for (long[] d : dist) {
            Arrays.sort(d);
            for (long z : d) {
                long b = 0, e = n + 1, k = 0;
                while (b <= e) {
                    long mid = (b + e) / 2;
                    if (z <= mid * (n - 1) + mid + 2 * (mid * mid + mid - 2) / 2) {
                        k = mid;
                        b = mid + 1;
                    } else {
                        e = mid - 1;
                    }
                }
                ans += k;
            }
        }
        ans = ans * pow(n, MOD - 2, MOD) % MOD;
        System.out.println(ans);
    }

    private static long pow(long base, long n, long mod) {
        if (n == 0) {
            return 1;
        }
        if (n == 1) {
            return base;
        }
        long half = pow(base, n / 2, mod);
        if (n % 2 == 0) {
            return half * half % mod;
        }
        return half * half % mod * base % mod;
    }
}
