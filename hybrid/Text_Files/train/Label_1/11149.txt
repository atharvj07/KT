#include <iostream>
#include <string>
#include <stack>
using namespace std;

int main() {
  string a;
  cin >> a;
  stack<char> s;
  for (char c : a) {
    if (!s.empty() && s.top() == c)
      s.pop();
    else
      s.push(c);
  }
  while (!s.empty()) {
    a += s.top();
    s.pop();
  }
  cout << a << endl;
}
