#include <iostream>
#include <string>
#include <unordered_map>

int main() {
  std::string pw;
  std::cin >> pw;
  int n;
  std::cin >> n;
  std::unordered_map<std::string, int> cnt;
  for (int i = 0; i < n; i++) {
    std::string s;
    std::cin >> s;
    cnt[s]++;
  }
  bool ok = false;
  for (int i = 0; i < pw.size(); i++) {
    ok |= cnt[pw.substr(i, 1)] > 0;
  }
  std::cout << (ok ? "YES" : "NO") << '\n';
}
