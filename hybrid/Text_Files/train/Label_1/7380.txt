#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int MAX = 1e5 + 5;

int n, l, r, ql, qr;
ll w[MAX], pref[MAX], suf[MAX];

void solve() {
  cin >> n >> l >> r >> ql >> qr;
  for (int i = 1; i <= n; i++) cin >> w[i];
  pref[1] = w[1] * l;
  for (int i = 2; i <= n; i++) pref[i] = pref[i - 1] + w[i] * l;
  suf[n] = w[n] * r;
  for (int i = n - 1; i >= 1; i--) suf[i] = suf[i + 1] + w[i] * r;
  ll ans = pref[n];
  for (int i = 2; i < n; i++) {
    ans = min(ans, pref[i - 1] + suf[i + 1] + ql * i + qr * (n - i));
  }
  cout << ans << '\n';
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  solve();
  return 0;
}
