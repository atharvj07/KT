from sys import stdin

t = int(stdin.readline())
for i in range(t):
    w, h = map(int, stdin.readline().split())
    x1, y1, x2, y2 = map(int, stdin.readline().split())
    w2, h2 = map(int, stdin.readline().split())
    dx = [x1, w-x2, w2 - y1, h - y2]
    dy = [y1, h - y2, h2 - x1, w - x2]
    print(min(dx) if min(dx) >= 0 else -1)
