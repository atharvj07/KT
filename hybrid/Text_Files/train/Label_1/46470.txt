import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigDecimal;
import java.math.MathContext;
import java.math.RoundingMode;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        long[] a = new long[n];
        String[] line = br.readLine().split("\\s");
        for (int i = 0; i < n; i++) {
            a[i] = Long.parseLong(line[i]);
        }

        BigDecimal sum = BigDecimal.ZERO;
        for (long ai : a) sum = sum.add(BigDecimal.valueOf(ai));

        BigDecimal minWeakness = BigDecimal.valueOf(1e20);
        for (long ai : a) {
            long currentSum = 0;
            BigDecimal currentWeakness = BigDecimal.ZERO;

            for (int i = 0; i < n; i++) {
                currentSum += (a[i] - ai);
                currentWeakness = currentWeakness.add(BigDecimal.valueOf(Math.abs(currentSum)));
            }

            minWeakness = minWeakness.min(currentWeakness);
        }

        System.out.println(minWeakness.setScale(15, RoundingMode.HALF_UP).toPlainString());
    }
}
