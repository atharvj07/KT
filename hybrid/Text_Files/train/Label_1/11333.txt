#include <bits/stdc++.h>
#pragma GCC optimize ("O3")
using namespace std;
typedef long long int ll;
typedef unsigned long long int ull;

#define mid(l, u) ((l+u)/2)
#define mod(x) (x%1000000007)

const ll linf = LLONG_MAX;
const ll inf = INT_MAX;
const char nl = '\n';

istream & in = cin;
ostream & out = cout;
const int max_row = 5005, max_col = 5005, max_op = 100005;
ll row[max_row], col[max_col];
int op[max_op][3], tot = 0;
void addop(int t, int x, int y)
{ op[tot][0] = t, op[tot][1] = x, op[tot][2] = y; tot++; }

int main() {
    int n, m, k;
    while(in >> n >> m >> k){
        memset(row, 0, sizeof row);
        memset(col, 0, sizeof col);
        tot = 0;

        for(int i = 0; i < k; i++){
            int t, x, y;
            in >> t >> x >> y;
            addop(t, x, y);
        }

        for(int i = 0; i < tot; i++){
            if(op[i][0] == 1){
                for(int j = 1; j <= m; j++) col[j] = op[i][2];
            }
            else{
                for(int j = 1; j <= n; j++) row[j] = op[i][2];
            }
        }

        for(int i = 1; i <= n; i++){
            for(int j = 1; j <= m; j++)
                out << row[i] + col[j] << ' ';
            out << nl;
        }
    }
    return 0;
}
