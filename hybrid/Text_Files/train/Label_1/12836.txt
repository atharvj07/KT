#include <cstdio>
#include <cstdlib>
#include <cstdint>
#include <algorithm>
#include <vector>
#include <set>
using namespace std;

int main() {
    int n, I;
    scanf("%d %d", &n, &I);
    I *= 8;
    vector<int> a;
    for (int i = 0; i < n; i++) {
        int x;
        scanf("%d", &x);
        a.push_back(x);
    }
    sort(a.begin(), a.end());
    int mx = *max_element(a.begin(), a.end());
    int mn = *min_element(a.begin(), a.end());
    if (mx - mn <= I) {
        printf("0\n");
        return 0;
    }
    int64_t l = mn, r = mx, ans = n;
    auto f = [&] (int64_t l, int64_t r) {
        if (r - l > I) {
            ans = min(ans, l - mn + mx - r);
        }
    };
    while (true) {
        int64_t ll, rr;
        set<int> s(a.begin(), a.end());
        int64_t l1 = l, r1 = r;
        auto it = s.upper_bound(l);
        if (it == s.begin()) {
            ll = l;
        } else {
            it--;
            ll = *it;
            if (ll - l > I) {
                f(l, ll - 1);
                ans = min(ans, ll - l);
            }
        }
        it = s.lower_bound(r);
        if (it == s.end()) {
            rr = r;
        } else {
            rr = *it;
            if (r - rr > I) {
                f(rr + 1, r);
                ans = min(ans, r - ll);
            }
        }
        if (l1 == ll && r1 == rr) break;
        l = ll, r = rr;
    }
    printf("%lld\n", ans);
    return 0;
}
