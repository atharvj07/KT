import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    while (true) {
      int n = in.nextInt();
      if (n == 0) {
        break;
      }
      if (n <= 15) {
        System.out.println(n + (n > 6 ? 4 : 0));
      } else {
        if (n % 10 == 9) {
          n -= 2;
        }
        n = n / 5 * 4 + n % 5 + (n % 15 == 13 ? 4 : 0);
        System.out.println(n);
      }
    }
  }
}
