#include <bits/stdc++.h>
using namespace std;

const int N = 100;
int n, r[N];

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> r[i];
  sort(r, r + n, greater<int>());
  int mx = *max_element(r, r + n);
  cout << mx << endl;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (r[i] == mx) continue;
    if (r[i] > 0) {
      int k = (mx - r[i] + 4) / 5;
      for (int j = 0; j < k && r[i] > 0; j++) {
        cout << string(i, '0') + string(min(n - i, 5), '1') + '\n';
        r[i]--;
        cnt++;
      }
    }
  }
  cout << cnt << endl;
  return 0;
}
