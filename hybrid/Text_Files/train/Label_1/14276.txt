#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  ll n, sum = 0;
  cin >> n;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) {
    cin >> a[i];
    sum += a[i];
  }
  if (sum % 2 == 0) {
    for (ll i = 0; i <= sum / 2; i++) cout << i << " ";
  } else {
    cout << -1;
  }
  cout << endl;
  return 0;
}
