#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    set<int> adj[n + 1];
    while (m--) {
        int x, y;
        cin >> x >> y;
        adj[x].insert(y);
        adj[y].insert(x);
    }
    for (int i = 1; i <= n; i++) {
        for (auto x : adj[i]) {
            for (auto y : adj[i]) {
                if (x == y) continue;
                for (auto z : adj[x]) {
                    if (z == y || z == i) continue;
                    cout << i + x + z << endl;
                    return 0;
                }
            }
        }
    }
    cout << "-1" << endl;
}
