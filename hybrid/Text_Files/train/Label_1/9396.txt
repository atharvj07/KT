#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  for (int l = 0, r = n - 1, cur = 0; l <= r;) {
    if (cur == 0) {
      if (a[l] == 0) {
        cur += 1;
        l += 1;
      } else
        l += 1;
    } else {
      if (a[r] == 1) {
        cur += 1;
        r -= 1;
      } else
        r -= 1;
    }
    ans += cur;
  }
  cout << ans << endl;
  return 0;
}
