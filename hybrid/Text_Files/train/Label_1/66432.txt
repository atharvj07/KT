from queue import PriorityQueue
n, m, T = map(int, input().split())
adj = [[] for _ in range(n+1)]
for _ in range(m):
    a, b, c = map(int, input().split())
    adj[a].append((b,c))
vis = [False] *(n+1)
q = PriorityQueue()
q.put((0,1))
ans = [1]
mn = float('inf')
while not q.empty():
    t, node = q.get()
    if node == n:
        mn = min(mn,t)
        continue
    if t > mn:
        continue
    vis[node] = True
    for u, w in adj[node]:
        if vis[u] == False:
            q.put((t+w,u))
            if t+w <= mn:
                ans.append(u)
print(len(ans))
print(*ans)
