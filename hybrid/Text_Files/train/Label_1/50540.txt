import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
  static int[] que(int n, int p, int[] time) {
    int[] ans = new int[n];

    PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> a[0] - b[0]);
    boolean[] seen = new boolean[n];

    int cur = 0;
    for (int i = 0; i < n; i++) {
      int ct = time[i];
      while (!pq.isEmpty() && pq.peek()[0] <= ct) {
        cur = pq.poll()[1];
        seen[cur] = false;
      }
      if (seen[i]) continue;
      seen[i] = true;
      pq.offer(new int[]{ct + p, i});
      ans[i] = ct + p;
    }

    return ans;
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int p = sc.nextInt();
    int[] time = new int[n];
    for (int i = 0; i < n; i++) {
      time[i] = sc.nextInt();
    }
    int[] res = que(n, p, time);
    for (int i : res) {
      System.out.print(i + " ");
    }
  }
}
