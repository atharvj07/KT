#include <iostream>
#include <vector>

int main() {
    int n, m;
    std::cin >> n >> m;

    if (n > m) {
        std::cout << m << '\n';
        for (int i = 0; i < m; ++i) {
            std::cout << i + 1 << ' ' << i + 1 << '\n';
        }
    } else {
        std::cout << n << '\n';
        for (int i = 0; i < n; ++i) {
            std::cout << i + 1 << ' ' << i + 1 << '\n';
        }
    }

    return 0;
}
