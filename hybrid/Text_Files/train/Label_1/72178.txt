def can_transform(n, a, b):
    diff = 0
    same = 0
    for i in range(n):
        if a[i] == b[i]:
            same += 1
        else:
            diff += 1
        if same < diff:
            return False
    return True

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = input().strip()
    b = input().strip()

    if a.count('0') != b.count('0') or a.count('1') != b.count('1'):
        print("NO")
    elif a == b:
        print("YES")
    else:
        if can_transform(n, a, b):
            print("YES")
        else:
            print("NO")
