#include <algorithm>
#include <iostream>
#include <string>

int main() {
  std::string s;
  int m;
  std::cin >> s >> m;
  int a[m];
  for (int i = 0; i < m; ++i) { std::cin >> a[i]; }
  std::reverse(s.begin(), s.begin() + a[0] - 1);
  for (int i = 1; i < m; ++i) {
    std::reverse(s.begin() + a[i] - 1, s.begin() + s.size() - a[i]);
  }
  std::cout << s << '\n';
  return 0;
}
