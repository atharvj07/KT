#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll N = 55;

vector<pair<ll, ll>> ask[N];
ll a[N], b[N];

int main() {
	ll h, q;
	cin >> h >> q;
	while (q--) {
		ll i, l, r, ans;
		cin >> i >> l >> r >> ans;
		ask[i].push_back({ l, r });
		a[i] |= 1 << (l - 1);
		b[i] |= 1 << (r - 1);
	}
	for (ll i = h - 1; i >= 1; i--) { // parent levels [h - 1 .. 1]
		for (auto x : ask[i]) {
			ll l = x.first, r = x.second;
			if ((a[i] & b[i + 1]) && (a[i + 1] & b[i]) && !(a[i + 1] & (a[i] ^ b[i])) && !(b[i + 1] & (a[i] ^ b[i])) && (!l || (a[i + 1] & (1 << (l - 1)))) && (!r || (b[i + 1] & (1 << (r - 1))))) {
				cout << "Game cheated!" << endl;
				return 0;
			}
			if ((a[i] & b[i + 1]) && (a[i + 1] & b[i]) && !(a[i + 1] & (a[i] ^ b[i])) && (!l || (a[i + 1] & (1 << (l - 1)))) && (!r || (b[i + 1] & (1 << (r - 1))))) {
				a[i] &= ~((1 << (l - 1)) | (1 << (r - 1)));
			}
			if ((a[i] & b[i + 1]) && (a[i + 1] & b[i]) && !(b[i + 1] & (a[i] ^ b[i])) && (!l || (a[i + 1] & (1 << (l - 1)))) && (!r || (b[i + 1] & (1 << (r - 1))))) {
				b[i] &= ~((1 << (l - 1)) | (1 << (r - 1)));
			}
		}
	}
	if (__builtin_popcountll(a[1]) == 1) {
		cout << (ll)__builtin_ctzll(a[1]) + 1 << endl;
	}
	else if (__builtin_popcountll(b[1]) == 1) {
		cout << (ll)__builtin_ctzll(b[1]) + 1 << endl;
	}
	else {
		cout << "Data not sufficient!" << endl;
	}
}
