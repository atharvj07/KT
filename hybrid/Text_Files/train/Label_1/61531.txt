n, c = map(int, input().split())
a = list(map(int, input().split()))

def min_partition_sum(n, c, a):
    if c >= n:
        return sum(a)
    
    dp = [0] * (n + 1)
    for i in range(1, n + 1):
        dp[i] = dp[i - 1] + a[i - 1]
        if i > c:
            max_val = a[i - 1]
            min_val = dp[i] - dp[i - 1]
            for j in range(2, c + 1):
                max_val = max(max_val, a[i - j])
                min_val = min(min_val, dp[i] - dp[i - j] + max_val)
            dp[i] = min(dp[i], min_val)
    
    return dp[n]

print(min_partition_sum(n, c, a))
