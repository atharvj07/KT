import java.util.*;

public class UniverseCounter {

    static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Point> planets = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            planets.add(new Point(scanner.nextInt(), scanner.nextInt()));
        }
        System.out.println(countUniverses(planets));
    }

    static int countUniverses(List<Point> planets) {
        int minX = Integer.MAX_VALUE, maxX = Integer.MIN_VALUE;
        int minY = Integer.MAX_VALUE, maxY = Integer.MIN_VALUE;
        for (Point planet : planets) {
            minX = Math.min(minX, planet.x);
            maxX = Math.max(maxX, planet.x);
            minY = Math.min(minY, planet.y);
            maxY = Math.max(maxY, planet.y);
        }
        return countUniverses(planets, minX, maxX, minY, maxY);
    }

    static int countUniverses(List<Point> planets, int minX, int maxX, int minY, int maxY) {
        if (minX > maxX || minY > maxY) {
            return 0;
        }
        boolean hasPlanetOnRow = false, hasPlanetOnColumn = false;
        for (Point planet : planets) {
            if (planet.x == minX || planet.x == maxX) {
                hasPlanetOnColumn = true;
            }
            if (planet.y == minY || planet.y == maxY) {
                hasPlanetOnRow = true;
            }
        }
        if (hasPlanetOnRow && hasPlanetOnColumn) {
            return countUniverses(planets, minX, maxX - 1, minY, maxY) + countUniverses(planets, minX + 1, maxX, minY, maxY);
        }
        return 1;
    }
}
