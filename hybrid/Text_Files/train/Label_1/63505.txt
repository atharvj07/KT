from sys import stdin

n, k = map(int, stdin.readline().split())

dp = [[0] * (k+2) for i in range(n+1)]

for i in range(n+1):
    dp[i][1] = 1

for i in range(1, n+1):
    for rem in range(k+1):
        dp[i][rem+1] += dp[i-1][rem] * i
        if rem:
            dp[i][rem+1] += dp[i-1][rem] * (n-i)

print(dp[n][k])
