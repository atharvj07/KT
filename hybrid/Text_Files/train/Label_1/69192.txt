n, m = list(map(int, input().split()))
a = [list(map(int, input().split())) for i in range(n)]
b = [list(map(int, input().split())) for i in range(n)]
if a==b:
  print("YES")
else:
  for i in range(n):
    for j in range(i+1, n):
      y1, x1, y2, x2 = i, j, m -1 - j, n -1 -i
      for k in range(m):
        if a[y1][k]!=b[x1][k]:  break
        if a[y2][k]!=b[x2][k]:  break
      else:
        for k in range(n):
          a[k][y1], a[k][y2] = a[k][y2], a[k][y1]
        if a==b:  
          print("YES");  exit()
  print("NO")
