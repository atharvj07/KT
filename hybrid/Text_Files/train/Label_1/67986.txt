import queue

# read input
n, m = map(int, input().split())
grid = [input() for _ in range(n)]
r1, c1 = map(int, input().split())
r2, c2 = map(int, input().split())

# define directions
di = [0, 1, 0, -1]
dj = [1, 0, -1, 0]

# check if cell is valid
def valid(i, j):
    return 0 <= i < n and 0 <= j < m and grid[i][j] != 'X'

# perform BFS
q = queue.Queue()
q.put((r1 - 1, c1 - 1))
visited = [[False] * m for _ in range(n)]
visited[r1 - 1][c1 - 1] = True

while not q.empty():
    i, j = q.get()
    if i == r2 - 1 and j == c2 - 1:
        print('YES')
        exit()
    for k in range(4):
        ni, nj = i + di[k], j + dj[k]
        if valid(ni, nj) and not visited[ni][nj]:
            q.put((ni, nj))
            visited[ni][nj] = True

# if no path found
print('NO')
