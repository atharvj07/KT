#include <bits/stdc++.h>

using namespace std;

const int N = 1e6+5;
vector<int>adj[N];
bool vis[N], hole[N];
int n, m, k;

void dfs(int u) {
    vis[u] = true;
    for(auto v : adj[u]) {
        if(!vis[v]) {
            dfs(v);
        }
    }
}

int main() {
    cin >> n >> m >> k;
    for(int i = 0; i < m; i++) {
        int x; cin >> x;
        hole[x] = true;
    }
    for(int i = 0; i < k; i++) {
        int u, v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1);
    for(int i = 1; i <= n; i++) {
        if(vis[i] && !hole[i]) {
            cout << i << endl;
            return 0;
        }
    }
    return 0;
}
