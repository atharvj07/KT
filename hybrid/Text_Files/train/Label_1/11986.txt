#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1073741824;
int f[1100000];
int main() {
  int a, b, c;
  cin >> a >> b >> c;
  int mx = a * b * c;
  for (int i = 1; i <= mx; i++) {
    for (int j = i; j <= mx; j += i) f[j]++;
  }
  ll ans = 0;
  for (int i = 1; i <= a; i++) {
    for (int j = 1; j <= b; j++) {
      for (int k = 1; k <= c; k++) { ans += f[i * j * k]; }
    }
  }
  cout << ans % mod << endl;
  return 0;
}
