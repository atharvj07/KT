import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        java.util.Scanner sc = new java.util.Scanner(System.in);
        int k = sc.nextInt();
        BigInteger l = new BigInteger(sc.next());
        BigInteger r = new BigInteger(sc.next());

        BigInteger[] binomCoeff = new BigInteger[k + 1];
        binomCoeff[0] = new BigInteger("1");
        for (int i = 1; i <= k; i++) {
            binomCoeff[i] = binomCoeff[i - 1].multiply(r.subtract(l.add(BigInteger.valueOf(i - 1))).divide(BigInteger.valueOf(i)));
        }

        BigInteger answer = new BigInteger("0");
        for (int i = 0; i <= k; i++) {
            answer = answer.add(binomCoeff[i].multiply(l.pow(k - i).multiply(r.pow(i))));
        }

        System.out.println(answer.mod(new BigInteger("1000000007")));
    }
}
