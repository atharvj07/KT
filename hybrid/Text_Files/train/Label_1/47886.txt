import java.util.Scanner;

public class Main {
    private static final int MOD = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        System.out.println(pow(m, n - k) % MOD);
    }

    private static long pow(long a, int n) {
        if (n == 0) {
            return 1;
        }
        if (n == 1) {
            return a;
        }
        long x = pow(a, n / 2);
        x = x * x % MOD;
        if (n % 2 == 1) {
            x = x * a % MOD;
        }
        return x;
    }
}
