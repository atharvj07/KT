#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (n); ++i)
using namespace std;
typedef long long ll;
const ll INF = 1LL << 60;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  rep(i, n) cin >> a[i];
  
  int mn = a[0];
  rep(i, n) mn = min(mn, a[i]);
  cout << mn << '\n';
  
  rep(i, n) a[i] -= mn;

  vector<string> ans;
  int mx = 0;
  while (true) {
    ++mx;
    int cnt = 0;
    string s(n, '0');
    rep(i, n) {
      if (a[i] == mx) {
        s[i] = '1';
        ++cnt;
      }
    }
    if (cnt >= 2 && cnt <= 5) {
      ans.push_back(s);
      rep(i, n) {
        if (s[i] == '1') a[i] = mn;
      }
    }
    int ok = 1;
    rep(i, n) ok &= a[i] <= mx;
    if (ok) break;
  }
  cout << ans.size() << '\n';
  rep(i, ans.size()) cout << ans[i] << '\n';
}
