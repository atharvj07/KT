import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Jigsaw {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int N = Integer.parseInt(line[0]);
        int H = Integer.parseInt(line[1]);

        List<Piece> pieces = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            line = reader.readLine().split(" ");
            int A_i = Integer.parseInt(line[0]);
            int B_i = Integer.parseInt(line[1]);
            int C_i = Integer.parseInt(line[2]);
            int D_i = Integer.parseInt(line[3]);
            pieces.add(new Piece(A_i, B_i, C_i, D_i, H));
        }

        Collections.sort(pieces);

        boolean possible = true;
        int currentHeight = 0;
        for (Piece piece : pieces) {
            if (piece.bottom + piece.A > currentHeight) {
                possible = false;
                break;
            }
            currentHeight = Math.max(currentHeight, piece.bottom + piece.B);
        }

        System.out.println(possible ? "YES" : "NO");
    }

    static class Piece implements Comparable<Piece> {
        int A, B, C, D, H;
        int bottom;

        public Piece(int A, int B, int C, int D, int H) {
            this.A = A;
            this.B = B;
            this.C = C;
            this.D = D;
            this.H = H;
            this.bottom = H - A - B;
        }

        @Override
        public int compareTo(Piece other) {
            return Integer.compare(this.bottom, other.bottom);
        }
    }
}
