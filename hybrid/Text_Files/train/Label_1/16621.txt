int main() {
    OJ;
    int n;
    cin >> n;
    vll l(n);
    for (ll &x : l) cin >> x;
    string s;
    cin >> s;
    vll pref(n, 0), suff(n, 0);
    for (int i = 0; i < n; i++) {
        if (s[i] == 'G' || s[i] == 'W') {
            pref[i] = 1;
        }
        if (i > 0) pref[i] += pref[i - 1];
    }
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == 'G' || s[i] == 'W') {
            suff[i] = 1;
        }
        if (i < n - 1) suff[i] += suff[i + 1];
    }
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        ll sec = l[i] * 5;
        if (s[i] == 'W') sec = l[i] * 3;
        if (s[i] == 'L') {
            if (i == 0 || i == n - 1) {
                cout << -1 << endl;
                rr 0;
            }
            if (i > 0) sec = ((pref[i - 1] > suff[i - 1] ? pref[i - 1] : suff[i - 1]) * 3 + l[i]);
            if (i < n - 1) sec = ((pref[i + 1] > suff[i + 1] ? pref[i + 1] : suff[i + 1]) * 3 + l[i]);
        }
        ans += sec;
    }
    cout << ans << endl;
}
