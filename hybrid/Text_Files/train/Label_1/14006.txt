#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define N 500005
int n, ans[N], col[N], sz[N];
pair<int, int> a[N];
vector<int> g[N];
void dfs(int u, int p) {
    for (auto &v : g[u]) {
        if (v == p) continue;
        dfs(v, u);
        sz[u] += sz[v];
    }
    for (auto &v : g[u]) {
        if (v == p) continue;
        if (sz[v] > n / 4) {
            cout << "NO" << endl;
            exit(0);
        }
    }
}
void color(int u, int p, int c) {
    ans[u] = c;
    for (auto &v : g[u]) {
        if (v == p) continue;
        color(v, u, c ^ 1);
    }
}
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i].first >> a[i].second;
    }
    sort(a + 1, a + n + 1);
    for (int i = 1; i <= n; i++) {
        a[i].first *= 2, a[i].second *= 2;
    }
    vector<int> vx, vy;
    for (int i = 1; i <= n; i++) {
        vx.emplace_back(a[i].first);
        vx.emplace_back(a[i].first + 1);
        vy.emplace_back(a[i].second);
        vy.emplace_back(a[i].second + 1);
    }
    sort(vx.begin(), vx.end());
    sort(vy.begin(), vy.end());
    vx.erase(unique(vx.begin(), vx.end()), vx.end());
    vy.erase(unique(vy.begin(), vy.end()), vy.end());
    int mxX = vx.size(), mxY = vy.size();
    int dx[] = {-1, 1, 0, 0};
    int dy[] = {0, 0, -1, 1};
    for (int i = 1; i <= n; i++) {
        int x = lower_bound(vx.begin(), vx.end(), a[i].first) - vx.begin() + 1;
        int y = lower_bound(vy.begin(), vy.end(), a[i].second) - vy.begin() + 1;
        for (int j = 0; j < 4; j++) {
            int nx = x + dx[j], ny = y + dy[j];
            if (nx < 1 || nx > mxX || ny < 1 || ny > mxY) continue;
            int v = (nx - 1) * mxY + ny;
            g[v].emplace_back(i);
            g[i].emplace_back(v);
        }
    }
    dfs(1, 1);
    color(1, 1, 1);
    cout << "YES" << endl;
    for (int i = 1; i <= n; i++) cout << ans[i] << endl;
    return 0;
}
