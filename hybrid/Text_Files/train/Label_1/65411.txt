n,m,k = map(int,input().split())
a = [list(input()) for i in range(n)]
vis = [[0]*m for i in range(n)]
for times in range(2000):
    for i in range(n):
        for j in range(m):
            if not(0<=i<n and 0<=j<m): continue
            if a[i][j]=='.': continue
            cnt = 0
            if a[i][j]=='L' and not(vis[i][j-1]): cnt+=1
            if a[i][j]=='R' and not(vis[i][j+1]): cnt+=1
            if a[i][j]=='U' and not(vis[i-1][j]): cnt+=1
            if a[i][j]=='D' and not(vis[i+1][j]): cnt+=1
            k-=cnt
            vis[i][j] = 1
            if cnt>0:
                if a[i][j]=='L': a[i][j-1]+=a[i][j]
                if a[i][j]=='R': a[i][j+1]+=a[i][j]
                if a[i][j]=='U': a[i-1][j]+=a[i][j]
                if a[i][j]=='D': a[i+1][j]+=a[i][j]
                a[i][j]='.'
    if k==0: break
ans = [0]*m
for i in range(m):
    for j in range(n):
        if a[j][i]=='.': continue
        if a[j][i]=='L': ans[i]+=1
        if a[j][i]=='R': ans[i+1]+=1
        if a[j][i]=='U': ans[i]+=1
        if a[j][i]=='D': ans[i+1]+=1
print(*ans)
