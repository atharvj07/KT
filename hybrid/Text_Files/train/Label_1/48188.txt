import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int numDatasets;
        while ((numDatasets = sc.nextInt()) != 0) {
            int years = sc.nextInt();
            double maxInterest = -1;
            int maxBank = -1;
            for (int i = 0; i < numDatasets; i++) {
                int bank = sc.nextInt();
                int rate = sc.nextInt();
                int type = sc.nextInt();
                double interest = type == 1 ? calculateSimpleInterest(rate, years) : calculateCompoundInterest(rate, years);
                if (interest > maxInterest) {
                    maxInterest = interest;
                    maxBank = bank;
                }
            }
            System.out.println(maxBank);
        }
        sc.close();
    }

    private static double calculateSimpleInterest(int rate, int years) {
        return (rate / 100.0) * years;
    }

    private static double calculateCompoundInterest(int rate, int years) {
        return Math.pow((1 + (rate / 100.0)), years) - 1;
    }
}
