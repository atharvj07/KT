import java.util.Scanner;

public class NooraAndProductSales {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int f = scanner.nextInt();

        int[][] days = new int[n][2];
        for (int i = 0; i < n; i++) {
            days[i][0] = scanner.nextInt();
            days[i][1] = scanner.nextInt();
        }

        int maxProductsSold = 0;
        for (int mask = 0; mask < (1 << f); mask++) {
            int productsSold = 0;
            for (int i = 0; i < n; i++) {
                int products = days[i][0];
                if ((mask & (1 << i)) > 0) {
                    products *= 2;
                }
                productsSold += Math.min(products, days[i][1]);
            }
            maxProductsSold = Math.max(maxProductsSold, productsSold);
        }

        System.out.println(maxProductsSold);
    }
}
