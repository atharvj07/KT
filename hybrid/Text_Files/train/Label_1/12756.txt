#include <iostream>

using namespace std;

const int N = 1 << 21;

int main() {
    long long a[N], b[N];
    bool f = 0;
    int p = 0, sum = 0, ok = 0;
    string s;
    cin >> s;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] == '^') {
            p = i;
            f = 1;
        }
        if (!f) a[i]++;
        else b[i - p]++;
        sum += (s[i] > '1' && s[i] <= '9');
    }
    for (int i = 0; i < p; i++) {
        if (a[i] > b[p - i - 1]) {
            ok = 1;
            cout << "left" << endl;
            return 0;
        }
        if (b[p - i - 1] > a[i]) {
            ok = 1;
            cout << "right" << endl;
            return 0;
        }
    }
    if (!ok && sum) cout << "right" << endl;
    else cout << "balance" << endl;
    return 0;
}
