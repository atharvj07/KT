import sys
input = sys.stdin.readline

for _ in range(int(input())):
    r, c = map(int, input().split())
    grid = [input().strip() for _ in range(r)]
    ans = 0
    for i in range(r):
        if 'P' in grid[i]:
            for j in range(c):
                if grid[i][j] == 'P':
                    grid[i] = 'A' * c
                    ans += 1
                    break
    for j in range(c):
        col = ''.join([grid[i][j] for i in range(r)])
        if 'P' in col:
            for i in range(r):
                if grid[i][j] == 'P':
                    for k in range(i, r):
                        grid[k][j] = 'A'
                    ans += 1
                    break
    if 'P' in ''.join(grid):
        ans = -1
    print(ans)
