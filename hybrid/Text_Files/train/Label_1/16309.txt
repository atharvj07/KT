#include <bits/stdc++.h>
using namespace std;
int main() {
    int t, n, a[101], sum1 = 0, sum2 = 0;
    vector<int> ans;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            if (a[i] % 2) sum1++;
            else sum2++;
        }
        if (sum2 > 0 || sum1 % 2 == 0) {
            cout << sum1 + sum2 << endl;
            for (int i = 0; i < n; i++)
                if (a[i] % 2) {
                    ans.push_back(i + 1);
                } else {
                    ans.push_back(i + 1);
                    break;
                }
            for (int i = 0; i < ans.size(); i++) cout << ans[i] << " ";
            cout << endl, ans.clear();
        } else
            cout << -1 << endl;
        sum1 = sum2 = 0;
    }
    return 0;
}
