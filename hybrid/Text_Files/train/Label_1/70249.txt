from collections import deque, defaultdict
from math import inf

for _ in range(int(input())):
    n, a, b, da, db = map(int, input().split())
    adj = defaultdict(set)
    for _ in range(n - 1):
        u, v = map(int, input().split())
        adj[u].add(v); adj[v].add(u)
    dist_a = [inf] * (n + 1)
    dist_b = [inf] * (n + 1)
    vis = [False] * (n + 1)
    q = deque()
    q.append((a, 0))
    while q:
        u, d = q.popleft()
        if vis[u]:
            continue
        vis[u] = True
        dist_a[u] = d
        for v in adj[u]:
            if not vis[v]:
                q.append((v, d + 1))
    q = deque()
    q.append((b, 0))
    while q:
        u, d = q.popleft()
        if vis[u]:
            continue
        vis[u] = True
        dist_b[u] = d
        for v in adj[u]:
            if not vis[v]:
                q.append((v, d + 1))
    ans = "Bob"
    if dist_a[b] <= da * 10**100:
        ans = "Alice"
    for u in range(1, n + 1):
        if dist_a[u] + dist_b[u] <= da * 10**100 + db * 10**100:
            ans = "Alice"
            break
    print(ans)
