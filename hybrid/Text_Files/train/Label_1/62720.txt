N, S = map(int, input().split())
total_slices = 0
happiness_diff = []
total_happiness = 0

for _ in range(N):
    s, a, b = map(int, input().split())
    total_slices += s
    happiness_diff.append(a - b)
    total_happiness += s * max(a, b)

happiness_diff.sort(reverse=True)
pizzas_needed = (total_slices + S - 1) // S
slices_used = 0

for i in range(pizzas_needed):
    if slices_used < total_slices:
        slices_used += S
    else:
        break
    if happiness_diff[i] > 0:
        total_happiness -= happiness_diff[i] * min(S, slices_used - total_slices + S)

print(total_happiness)
