#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  for (int i = n - 1; i >= 4; i--) {
    bool found = false;
    for (int j = 2; j * j <= i; j++) {
      if (i % j == 0) {
        found = true;
        break;
      }
    }
    if (!found) {
      cout << i << endl;
      break;
    }
  }
  return 0;
}
