#include <bits/stdc++.h>

using namespace std;

inline string& operator+=(string &a, char x)
{
     return a.push_back(x), a;
}

const int MAXN = 1e5 + 10, MAXC = 26;
int n, q, cnt[MAXN][MAXC];

int main()
{
    string s;
    cin >> s >> q;
    n = s.size();
    for (int i = n - 1; i >= 0;  -- i)
    {
        for (int j = 0; j < MAXC; j ++)
        {
            cnt[i][j] = cnt[i + 1][j];
        }
        cnt[i][s[i] - 'a'] ++;
    }
    while (q --)
    {
        int type;
        cin >> type;
        if (type == 1)
        {
            int pos;
            char c;
            cin >> pos >> c;
            pos --;
            for (int i = 0; i < MAXC; i ++)
            {
                cnt[pos][i] = cnt[pos + 1][i];
            }
            cnt[pos][c - 'a'] ++;
            s[pos] = c;
        }
        else
        {
            int l, r;
            cin >> l >> r;
            l --, r --;
            int ans = 0;
            for (int i = 0; i < MAXC; i ++)
            {
                ans += (cnt[l][i] > cnt[r][i]);
            }
            cout << ans << endl;
        }
    }
    return 0;
}
