import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int open = 0, ans = 0;
        boolean flag = true;
        for (char c : s.toCharArray()) {
            if (c == '(') {
                open++;
            } else if (c == ')') {
                open--;
            } else if (c == '#') {
                ans++;
            }
            if (open < 0) {
                flag = false;
                break;
            }
        }
        if (open != 0 || !flag) {
            System.out.println(-1);
        } else {
            for (char c : s.toCharArray()) {
                if (c == '#') {
                    System.out.println(ans);
                    ans--;
                }
            }
        }
    }
}
