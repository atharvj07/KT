def check_win(matrix):
    directions = [(0, 1), (1, 0), (1, 1), (1, -1)]
    for i in range(10):
        for j in range(10):
            if matrix[i][j] == '.':
                for dx, dy in directions:
                    count = 1
                    x, y = i + dx, j + dy
                    while 0 <= x < 10 and 0 <= y < 10 and matrix[x][y] == 'X':
                        count += 1
                        x, y = x + dx, y + dy
                    x, y = i - dx, j - dy
                    while 0 <= x < 10 and 0 <= y < 10 and matrix[x][y] == 'X':
                        count += 1
                        x, y = x - dx, y - dy
                    if count >= 5:
                        return 'YES'
    return 'NO'

matrix = [input() for _ in range(10)]
print(check_win(matrix))
