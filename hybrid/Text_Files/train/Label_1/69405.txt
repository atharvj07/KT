import sys
from collections import defaultdict

n = int(input())
g = defaultdict(list)
for _ in range(n - 1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

visited = [False] * (n + 1)
paths = []
path = []

def dfs(node, prev):
    visited[node] = True
    path.append(node)
    for neighbor in g[node]:
        if not visited[neighbor]:
            dfs(neighbor, node)
        elif neighbor != prev:
            paths.append(path[path.index(neighbor) :])
    path.pop()

dfs(1, 1)
if len(paths) == 0:
    print("No")
else:
    print("Yes")
    print(len(paths))
    for path in paths:
        print(*path)
