def is_palindrome(s):
    return s == s[::-1]

def solve(s):
    if len(s) == 1:
        return 'Impossible'
    if len(s) == 2:
        return 1 if s[0] != s[1] else 'Impossible'
    for i in range(1, len(s) - 1):
        l, r = s[:i], s[i:]
        if is_palindrome(l) and is_palindrome(r):
            return 1
    if is_palindrome(s[1:-1]):
        return 1
    return 'Impossible'

s = input()
print(solve(s))
