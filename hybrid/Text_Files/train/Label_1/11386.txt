#include <iostream>

using namespace std;

typedef long long ll;

int main() {
  int q;
  cin >> q;
  while (q--) {
    ll n;
    cin >> n;
    if (n < 1) {
      cout << "-1" << endl;
    } else {
      int step = 0;
      for (; n % 2 == 0; step++) n /= 2;
      for (; n % 3 == 0; step++) n = n * 2 / 3;
      for (; n % 5 == 0; step++) n = n * 4 / 5;
      if (n == 1) {
        cout << step << endl;
      } else {
        cout << "-1" << endl;
      }
    }
  }
  return 0;
}
