#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  cin >> T;
  while (T--) {
    int n, m;
    cin >> n >> m;
    vector<int> ax(n), ay(m);
    for (int i = 0; i < n; i++) { cin >> ax[i]; }
    for (int i = 0; i < m; i++) { cin >> ay[i]; }
    int ans = 0;
    set<int> ax2, ay2;
    for (auto &x : ax) {
      if (ax2.count(x)) ans++;
      ax2.insert(x);
    }
    for (auto &x : ay) {
      if (ay2.count(x)) ans++;
      ay2.insert(101 - x);
    }
    cout << ans << endl;
  }
}
