import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = [1 << int(x) for x in input().split()]
    dict = {}
    for i in range(n):
        dict[a[i]] = i
    for i in range(n):
        for j in range(i + 1, n):
            for k in range(n):
                if j <= k:
                    continue
                if a[j] - a[i] == a[k] - a[j] or a[i] + a[k] == a[j] + a[k]:
                    print("YES")
                    exit()
    print("NO")
