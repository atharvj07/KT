from sys import stdin
input=stdin.readline

n,m,k,s=map(int,input().split())
a=[[int(ch)for ch in input().split()]for i in range(n)]
notes=list(map(int,input().split()))

ans=-1
for x in range(n):
    for y in range(m-1):
        for z in range(x,n):
            for w in range(max(y+1,m-1)):
                ans=max(ans,abs(a[x][y]-a[z][w])+abs(x-z)+abs(y-w))
print(ans)
