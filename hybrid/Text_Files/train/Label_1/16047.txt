#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> low, up;
    for (int i = 0; i < n; ++i) {
        if (islower(s[i]))
            low.push_back(i);
        else
            up.push_back(i);
    }
    int ans = 0;
    for (int i = 0, j = 0; i < low.size(); ++i) {
        while (j < up.size() && up[j] < low[i])
            ++j;
        if (j == up.size())
            ans = max(ans, (int)low.size());
        else
            ans = max(ans, (int)low.size() - (i - j));
    }
    cout << ans << endl;
    return 0;
}
