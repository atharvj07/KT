import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int p = sc.nextInt();
        int[] points = { 500, 1000, 5000, 10000 };
        int max = 0;
        for (int point : points) {
            max += p / point * point;
            p %= point;
        }
        System.out.println(max);
    }
}
