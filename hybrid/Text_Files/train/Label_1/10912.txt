#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  string t = "CODEFORCES";
  int i = 0, j = 0;
  while (i < s.size() && j < t.size()) {
    if (s[i] == t[j]) {
      j++;
    }
    i++;
  }
  if (j == t.size()) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
