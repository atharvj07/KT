import java.util.*;

public class Main {
    static void solve() {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long k = in.nextLong();
        long l = in.nextLong();
        long m = in.nextLong();
        long x1[] = new long[(int) (l + 2)];
        long x2[] = new long[(int) (l + 2)];
        for (int i = 0; i < (1L << l); i++) {
            long temp = i;
            for (long j = 0; j < l; j++) {
                int tmp = (int) (1L << j);
                if ((i & tmp) != 0) {
                    temp -= (1L << j);
                }
            }
            if (temp % 2 == 0) {
                x1[(int) (temp / (1L << k))]++;
            } else {
                x2[(int) (temp / (1L << k))]++;
            }
        }
        long temp[] = new long[(int) (n + 2)];
        for (long i = 0; i <= n; i++) {
            for (long j = 0; j < (1L << l); j++) {
                long t = j;
                for (long l1 = 0; l1 < l; l1++) {
                    int tmp = (int) (1L << (int) l1);
                    if ((t & tmp) != 0) {
                        t += (1L << (int) l1);
                    }
                }
                if (t > i) {
                    break;
                }
                long p = x1[(int) ((i - t) / (1 << k))];
                long q = x2[(int) ((i - t) / (1 << k))];
                temp[(int) (i)] += ((p * q) % m);
                if (i > 0) {
                    temp[(int) (i)] += ((temp[(int) (i - 1)]) % m);
                }
            }
        }
        System.out.println(temp[(int) (n)]);
    }

    public static void main(String[] args) {
        solve();
    }
}
