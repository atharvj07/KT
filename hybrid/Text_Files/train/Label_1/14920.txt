#include <iostream>
#include <string>

using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  if (s.size() != t.size()) cout << "need tree" << endl;
  else if (s == t) cout << "array" << endl;
  else {
    int diff = 0;
    for (int i = 0; i < s.size(); i++) {
      if (s[i] != t[i]) diff++;
    }
    if (diff == 0) cout << "need tree" << endl;
    else if (diff == 1 || diff == 2) cout << "both" << endl;
    else cout << "automaton" << endl;
  }
  return 0;
}
