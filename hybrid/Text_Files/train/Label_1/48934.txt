import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (scanner.hasNext()) {
            int num = scanner.nextInt();
            int ans = (num - 1) % 39 + 1;
            System.out.printf("3C%02d\n", ans);
        }

        scanner.close();
    }
}
