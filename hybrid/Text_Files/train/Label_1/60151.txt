import sys
input=sys.stdin.readline
n,m,k=map(int,input().split())
c=[0]*n; a=[0]*n; b=[0]*n
for i in range(n): a[i],b[i],c[i]=map(int,input().split())
for i in range(m): u,v=map(int,input().split()); u,v=u-1,v-1; c[u],c[v]=c[u]+c[v],0
dp=[[0]*(k+1) for i in range(n)]
for i in range(k,k+b[0],1): dp[0][i]+=c[0]
for i in range(1,n,1):
 dp2=[[0]*(k+1) for j in range(i)]
 for l in range(i):
  for j in range(k+1):
   if dp[l][j]>0:
    if j+a[i]<=k: dp2[l][j+a[i]]+=dp[l][j]+c[i]
    if j+1<=k: dp2[i][j+1]+=dp[l][j]
 for l in range(i):
  for j in range(k+1):
   dp[l][j]=max(dp[l][j],dp2[l][j])
print(max([max(i) for i in dp])) if max([max(i) for i in dp])>0 else print(-1)
