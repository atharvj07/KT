#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> seq(n);
  for (int i = 0; i < n; i++) {
    int l;
    cin >> l;
    seq[i].resize(l);
    for (int j = 0; j < l; j++) { cin >> seq[i][j]; }
  }

  long long ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i == j) {
        for (int k = 1; k < seq[i].size(); k++) {
          if (seq[i][k] > seq[i][k - 1]) { ans += n - 1; }
        }
      } else {
        for (int k = 0; k < seq[i].size(); k++) {
          for (int l = 0; l < seq[j].size(); l++) {
            if (seq[i][k] < seq[j][l]) { ans++; }
          }
        }
      }
    }
  }

  cout << ans << endl;

  return 0;
}
