#include <iostream>
#include <vector>

using namespace std;

int gcd(int a, int b) {
    while (b) {
        a %= b;
        swap(a, b);
    }
    return a;
}

int main() {
    int x, y, n;
    cin >> x >> y >> n;
    int ans_x = x, ans_y = y;
    for (int i = 2; i * i <= n; ++i) {
        if (n % i == 0) {
            while (n % i == 0) n /= i;
            int nx = x / i, ny = y / i;
            if (gcd(nx, ny) == 1 && (abs(nx / ny - x / y) < abs(ans_x / ans_y - x / y) || (
                                              abs(nx / ny - x / y) == abs(ans_x / ans_y - x / y) && ny < ans_y))) {
                ans_x = nx;
                ans_y = ny;
            }
        }
    }
    if (n > 1) {
        int nx = x / n, ny = y / n;
        if (gcd(nx, ny) == 1 && (abs(nx / ny - x / y) < abs(ans_x / ans_y - x / y) || (
                                          abs(nx / ny - x / y) == abs(ans_x / ans_y - x / y) && ny < ans_y))) {
            ans_x = nx;
            ans_y = ny;
        }
    }
    cout << ans_x << '/' << ans_y << endl;
}
