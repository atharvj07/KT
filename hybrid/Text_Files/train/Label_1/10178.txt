#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  ll n;
  cin >> n;
  string s = to_string(n);
  bool f = false;
  for (int i = 0; i < s.length() - 1; i++) {
    for (int j = i + 1; j < s.length(); j++) {
      if (s[i] < s[j]) {
        swap(s[i], s[j]);
        cout << s << endl;
        f = true;
        return 0;
      }
    }
  }
  if (!f) cout << -1 << endl;
  return 0;
}
