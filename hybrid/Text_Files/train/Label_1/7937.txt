#include <bits/stdc++.h>
using namespace std;

int X[110], Y[110];

int main() {
  string s;
  cin >> s;

  int n = s.size();
  int bal = 0;

  for (int i = 0; i < n; i++) {
    if (s[i] == '|') {
      ++bal;
    } else if (s[i] == '+' || s[i] == '=') {
      X[bal] = i;
      bal = 0;
    }
  }

  Y[bal] = n;

  if (X[1] - X[0] == Y[1] - X[1]) {
    cout << s;
    return 0;
  }

  if (Y[1] - X[0] <= Y[0] - X[0]) {
    s[Y[1] - 1] = '|';
  } else {
    s[X[1] + 1] = ' ';
  }

  cout << s;

  return 0;
}
