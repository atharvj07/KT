a = input()
b = input()
ans = 0
while len(a) and len(b):
    if a[0] == b[0]:
        a = a[1:]
        b = b[1:]
    else:
        if a[0] in b:
            idx = b.index(a[0])
            b = b[:idx] + b[idx+1:]
            ans += idx + 1
        elif b[0] in a:
            idx = a.index(b[0])
            a = a[:idx] + a[idx+1:]
            ans += idx + 1
        else:
            ans += len(a) + len(b)
            break
ans += max(len(a), len(b))
print(ans)
