#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        std::string s;
        std::cin >> s;
        int n = s.size();
        std::vector<int> pref0(n + 1), pref1(n + 1);
        for (int i = 1; i <= n; i++) {
            pref0[i] = pref0[i - 1];
            pref1[i] = pref1[i - 1];
            if (s[i - 1] == '0') pref0[i]++;
            else pref1[i]++;
        }
        std::string ans = s;
        for (int l = 0; l < n; l++) {
            for (int r = l + 1; r <= n; r++) {
                int c0 = pref0[r] - pref0[l];
                int c1 = pref1[r] - pref1[l];
                if (c0 == c1) {
                    std::string t = s.substr(l, r - l);
                    std::reverse(t.begin(), t.end());
                    for (int i = 0; i < t.size(); i++) t[i] = (t[i] == '0') ? '1' : '0';
                    ans = std::min(ans, s.substr(0, l) + t + s.substr(r));
                }
            }
        }
        std::cout << ans << '\n';
    }
    return 0;
}
