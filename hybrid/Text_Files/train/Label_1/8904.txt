#include <vector>
#include <queue>
#include <set>
#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        set<int> s;
        for (int i = 0; i < n; i++) s.insert(a[i]);
        if (s.count(x)) cout << 0 << endl;
        else cout << 1 << endl;
    }
}
