#include <iostream>

int main() {
  int n;
  std::string s;
  std::cin >> n;
  while (n--) {
    std::cin >> s;
    bool flag = true;
    for (int i = 1; i < s.size() && flag; i++) {
      if (s[i - 1] != s[i] - 1 || s[i] == 'z') {
        flag = false;
      }
    }
    std::cout << (flag ? "Yes\n" : "No\n");
  }
  return 0;
}
