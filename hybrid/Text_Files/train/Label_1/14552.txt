#include <iostream>
#include <algorithm>
#include <vector>

static const int MAXN = 1e6 + 10;

int main() {
    int n, k;
    std::vector<int> perm;
    std::cin >> n >> k;
    if (n < 3 * k || k == 1) {
        std::cout << "-1" << std::endl;
        return 0;
    }
    int t = n / k;
    for (int i = 0; i < k; ++i) {
        for (int j = 0; j < t; ++j) {
            perm.push_back(i + 1);
        }
        if (i * t + t <= n) {
            perm.push_back(i + 1);
        }
    }
    while (perm.size() < n) {
        perm.push_back(1);
    }
    std::random_shuffle(perm.begin(), perm.end());
    for (int& x : perm) {
        std::cout << x << " ";
    }
    std::cout << std::endl;
    return 0;
}
