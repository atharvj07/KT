import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        for (int i = 0; i < t; i++) {
            int n = input.nextInt();
            int x = input.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = input.nextInt();
            }
            Arrays.sort(a);
            int ans = 0;
            for (int j = 0; j < n; j++) {
                if (a[j] < x) {
                    ans++;
                    a[j] = x;
                }
            }
            System.out.println(ans);
        }
    }
}
