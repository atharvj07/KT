import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
            }
            int ans = 0;
            int max = a[n - 1];
            for (int i = n - 2; i >= 0; i--) {
                if (a[i] > max) {
                    ans++;
                } else {
                    max = a[i];
                }
            }
            System.out.println(ans);
        }
    }
}
