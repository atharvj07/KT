#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll mod=1e9+7;

ll F[205], I[205];

ll fastpow(ll a, ll p)
{
    ll ans=1;
    while(p>0)
    {
        if(p%2==1) ans=(ans*a)%mod;
        a=(a*a)%mod;
        p/=2;
    }
    return ans;
}

int main()
{
    ll n, k, ans;
    cin>>n>>k;
    F[0]=1; for(ll i=1;i<=200;i++) F[i]=(F[i-1]*i)%mod;
    for(ll i=0;i<=200;i++) I[i]=fastpow(F[i], mod-2);
    for(ll i=1;i<=k;i++)
    {
        ll x=(F[n]*I[n-i])%mod;
        x=(x*I[i])%mod;
        cout<<x<<" ";
    }
    cout<<endl;
    return 0;
}
