#include <iostream>
#include <cmath>
int main() {
    int a, b, c, d;
    std::cin >> a >> b >> c >> d;
    double Ac = std::max(a * 3 / 10.0, a - a * c / 250.0);
    double Bc = std::max(b * 3 / 10.0, b - b * d / 250.0);
    if (Ac > Bc) {
        std::cout << "Misha" << std::endl;
    } else if (Ac == Bc) {
        std::cout << "Tie" << std::endl;
    } else {
        std::cout << "Vasya" << std::endl;
    }
    return 0;
}
