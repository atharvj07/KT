import sys

n, k = list(map(int, sys.stdin.readline().split()))
a = [0] * n
dic = {}
for i in range(n):
    a[i] = int(sys.stdin.readline())
    t = a[i]**k
    if t not in dic: dic[t] = 0
    dic[t] += 1

ans = 0
for val in dic.values():
    ans += val * (val - 1) // 2
for i in range(n):
    if a[i]**k in dic and dic[a[i]**k] > 0:
        ans -= 1
        dic[a[i]**k] -= 1
print(ans)
