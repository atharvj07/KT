from collections import deque

def bfs(dy, dx):
    while q:
        y, x, t = q.popleft()
        for dy_ in dy:
            for dx_ in dx:
                next_y, next_x = y + dy_, x + dx_
                if 0 <= next_y < n and 0 <= next_x < m and grid[next_y][next_x] == '.':
                    if t == 0:  # 1번째로 방문
                        grid[next_y][next_x] = '*'
                        q.append((next_y, next_x, 1))
                    elif t == 1:
                        grid[next_y][next_x] = 'X'
                    elif t == 2:
                        print("YES")
                        exit()
                        

n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]

dy = [1, 1, 1, -1, -1, -1, 0, 0]
dx = [1, -1, 0, 1, -1, 0, 1, -1]
q = deque()

for i in range(n):
    for j in range(m):
        if grid[i][j] == '.':
            q.append((i, j, 0))
bfs(dy, dx)
print("NO")
