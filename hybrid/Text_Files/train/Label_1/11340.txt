#include <bits/stdc++.h>
using namespace std;
const int MAXN = 26;
vector<int> a;

int main() {
    int t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        a.resize(2 * n);
        for (int i = 0; i < 2 * n; i++) cin >> a[i];
        sort(a.begin(), a.end());
        int p = 0, q = 2 * n - 1;
        for (int i = 1; i <= n; i++) {
            cout << a[p] << ' ';
            p++;
            cout << a[q] << ' ';
            q--;
        }
        cout << endl;
    }
    return 0;
}
