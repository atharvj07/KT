#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<pair<int, int>> segments(n);
    for (int i = 0; i < n; ++i) {
      cin >> segments[i].first >> segments[i].second;
    }
    sort(segments.begin(), segments.end(), [](pair<int, int> a, pair<int, int> b) {
      return a.second < b.second;
    });
    int max_num_segments = 0;
    int cur_right = segments[0].first - 1;
    for (int i = 0; i < n; ++i) {
      if (segments[i].first > cur_right) {
        ++max_num_segments;
        cur_right = segments[i].second;
      }
    }
    cout << max_num_segments << endl;
  }
  return 0;
}
