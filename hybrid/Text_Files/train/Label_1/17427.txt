#include<bits/stdc++.h>
using namespace std;
const int mo=1e9+7;
typedef long long ll;
ll n,ans,a[100005],pos[100005],p;
vector<ll>v[100005];

void solve( ll m, ll p ){
	if( m == n ){
		ans++;
		ans %= mo;
		return ;
	}
	solve(m+1, p);
	if( __gcd(p, a[m] ) == 1 ) solve(m+1, a[m]);
}

int main(){
  cin >> n;
  for(int i=0;i<n;i++) cin >> a[i];
  solve(0, 1);
  cout << ans << endl;
  return 0; 
}
