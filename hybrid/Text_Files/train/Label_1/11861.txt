#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;
int n, m, d;
pair<int,int> p[N], i[N];

int main() {
    cin >> n >> m >> d;
    for(int j = 0; j < n; j++) cin >> p[j].first >> p[j].second;
    for(int j = 0; j < m; j++) cin >> i[j].first >> i[j].second;
    sort(p, p+n);
    sort(i, i+m);
    int j = n-1, ans = 0;
    for(int k = m-1; k >= 0 && j >= 0;) {
        if(p[j].second + i[k].second <= d) ans += p[j].first + i[k].first, j--, k--;
        else {
            if(i[k].second <= d) ans += i[k].first, k--;
            else break;
        }
    }
    cout << ans << '\n';
}
