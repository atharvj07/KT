import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int sum = 0;
        for (int i : a) {
            sum += i;
        }
        double half = (double) sum / 2;
        int day = n;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            cur += a[i];
            if (cur >= half) {
                day = i + 1;
                break;
            }
        }
        System.out.println(day);
    }
}
