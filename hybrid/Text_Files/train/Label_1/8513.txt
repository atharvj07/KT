#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
	int n;
	cin >> n;
	vector<int> a(2 * n);
	for (int &i : a) cin >> i;
	int ans = -1;
	if (a[0] == 1) {
		for (int i = 1; i < a.size(); i++) {
			if ((i % 2 == 0 && a[i] > a[i - 1]) || (i % 2 == 1 && a[i] < a[i - 1])) ans = i - 1;
		}
		for (int i = n; i < a.size() - 2; i += 2) {
			if (a[i] > a[i + 1]) ans = (ans == -1) ? i : -1;
		}
	}
	cout << ans << endl;
	return 0;
}
