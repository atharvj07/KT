import sys, collections
for _ in range(int(input())):
    n = int(input())
    arr = list(map(int, input().split()))
    dp = collections.defaultdict(int)
    diversity = 1
    dp[arr[0]] = diversity
    for i in range(1, n):
        if dp.get(arr[i]+1):
            dp[arr[i]] = max(dp[arr[i]], dp[arr[i]+1])
        else:
            diversity += 1
            dp[arr[i]] = diversity
    print(diversity)
