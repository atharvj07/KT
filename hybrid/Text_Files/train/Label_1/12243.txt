#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
const int MAXN = 2e5 + 10;
int n, k, h[MAXN];
int main() {
    cin >> n >> k;
    vector<int> taller;
    for (int i = 1; i <= n; ++i) { cin >> h[i]; if (h[i] > k) taller.emplace_back(h[i]); }
    sort(taller.begin(), taller.end());
    int ans = 0, sum = 0;
    for (int t : taller) {
        int need = t - 1;
        if (sum + need <= k) sum += need, ans++;
        else break;
    }
    cout << ans << endl;
    return 0;
}
