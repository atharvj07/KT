def find_trailing_zeros(n, b):
    # Prime factorization of b
    factors = {}
    for factor in range(2, int(b**0.5)+1):
        while b % factor == 0:
            b //= factor
            if factor in factors:
                factors[factor] += 1
            else:
                factors[factor] = 1
    if b > 1:
        factors[b] = 1
    
    # Find minimum trailing zeros for each factor
    res = float('inf')
    for factor, count in factors.items():
        temp = n
        zeros_for_factor = 0
        while temp:
            temp //= factor
            zeros_for_factor += temp
        res = min(res, zeros_for_factor // count)
    
    return res

n, b = map(int, input().split())
print(find_trailing_zeros(n, b))
