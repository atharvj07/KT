from collections import Counter

def max_fibish_prefix_length():
    n = int(input())
    a = list(map(int, input().split()))
    a_counter = Counter(a)
    max_length = 2 if n >= 2 else n
    for i in range(n):
        for j in range(i + 1, n):
            temp_counter = a_counter.copy()
            f0, f1 = a[i], a[j]
            temp_length = 2
            temp_counter[f0] -= 1
            temp_counter[f1] -= 1
            while temp_counter[f0 + f1] > 0:
                f0, f1 = f1, f0 + f1
                temp_counter[f1] -= 1
                temp_length += 1
            max_length = max(max_length, temp_length)
    print(max_length)

max_fibish_prefix_length()
