import sys

q, m = map(int, input().split())
spells = []
ans = []
for i in range(q):
	k, a, b = map(int, input().split())
	if k == 1:
		spells.append((a, b))
	else:
		t, h = a, b
		mn_t = sys.maxsize
		for d, c in spells:
			if h % (d * t) == 0 and h // (d * t) <= m:
				mn_t = min(mn_t, t)
		ans.append("YES" if mn_t != sys.maxsize else "NO")
print(*ans, sep="\n")
