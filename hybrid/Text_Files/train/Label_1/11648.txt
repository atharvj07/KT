#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
  }
  vector<int> prefMax(n);
  prefMax[n - 1] = p[n - 1];
  for (int i = n - 2; i >= 0; i--) {
    prefMax[i] = max(prefMax[i + 1], p[i]);
  }
  long long ans = 0;
  for (int i = 0; i < n - 1; i++) {
    if (p[i] + p[i + 1] == prefMax[i]) {
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
