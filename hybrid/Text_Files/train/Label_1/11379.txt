#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

struct Person {
  string name;
  int region;
  int score;
};

int main() {
  int n, m;
  cin >> n >> m;
  vector<Person> people(n);
  for (int i = 0; i < n; i++) {
    cin >> people[i].name >> people[i].region >> people[i].score;
  }
  sort(people.begin(), people.end(),
       [](const Person &a, const Person &b) { return a.region < b.region; });
  for (int i = 0; i < m; i++) {
    vector<int> scores;
    for (int j = i * 2; j < i * 2 + 2; j++) {
      scores.push_back(people[j].score);
    }
    sort(scores.begin(), scores.end());
    if (scores[1] != scores[0]) {
      cout << people[i * 2].name << " " << people[i * 2 + 1].name << endl;
    } else {
      cout << "?" << endl;
    }
  }
  return 0;
}
