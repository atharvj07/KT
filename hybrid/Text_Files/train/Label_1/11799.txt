#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> c(n);
  for (int i = 0; i < n; i++) {
    cin >> c[i];
  }

  vector<pair<int, int>> ans(n, {-1, -1});
  vector<int> cnt(m + 1, 0);
  int match = 0;
  for (int i = 0; i < n; i++) {
    if (++cnt[c[i]] == 2) {  // match found
      for (int j = 0; j < n; j++) {
        if (ans[j].first == -1 && c[j] == c[i]) {
          ans[j].first = c[i];
          ans[i].second = c[i];
          match++;
          break;
        }
      }
    }
  }

  cout << match << endl;
  for (int i = 0; i < n; i++) {
    if (ans[i].first == -1) {
      for (int j = 1; j <= m; j++) {
        if (cnt[j] < 2) {
          ans[i].first = j;
          cnt[j]++;
          break;
        }
      }
    }
    if (ans[i].second == -1) {
      for (int j = 1; j <= m; j++) {
        if (cnt[j] < 2 && j != ans[i].first) {
          ans[i].second = j;
          cnt[j]++;
          break;
        }
      }
    }
    cout << ans[i].first << " " << ans[i].second << endl;
  }
  return 0;
}
