#include <bits/stdc++.h>

using ll = long long;

const ll MOD = 1e9 + 7;

struct Data {
  ll zeroes;
  ll ones;
};

struct BinMatrix {
  int n;
  int m;
  std::vector<Data> rows;
  std::vector<Data> columns;
};

int main() {
  int n, m, mod;
  std::cin >> n >> m >> mod;
  BinMatrix binMatrix;
  binMatrix.n = n;
  binMatrix.m = m;
  char ch;
  for (int i = 0; i < m; i++) {
    binMatrix.rows.push_back({0, 0});
    binMatrix.columns.push_back({0, 0});
    for (int j = 0; j < n; j++) {
      std::cin >> ch;
      if (ch == '0') {
        binMatrix.rows.back().zeroes++;
        binMatrix.columns[j].zeroes++;
      } else {
        binMatrix.rows.back().ones++;
        binMatrix.columns[j].ones++;
      }
    }
  }
  ll ans = 1;
  for (int i = 0; i < m; i++) {
    if (binMatrix.rows[i].zeroes != n - 2 ||
        binMatrix.rows[i].ones != 2 || binMatrix.columns[i].zeroes != n - 2 ||
        binMatrix.columns[i].ones != 2) {
      ans = 0;
      break;
    }
  }
  for (int i = m; i < n; i++) {
    ans = ans * 2;
    ans %= mod;
  }
  std::cout << ans << std::endl;
  return 0;
}
