import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        int l = 0;
        int r = 0;
        int ans = n;
        for (int i = 0; i < n; i++) {
            while (l < n && a[l] < x) {
                l++;
            }
            while (r < n && a[r] <= x) {
                r++;
            }
            ans = Math.min(ans, l + n - r);
        }
        System.out.println(ans);
    }
}
