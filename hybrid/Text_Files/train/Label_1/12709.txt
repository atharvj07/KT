#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<pair<int, int>, pair<int, int>>> rectangles(n);
    for (int i = 0; i < n; i++) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        rectangles[i] = make_pair(make_pair(x1, y1), make_pair(x2, y2));
    }
    int minX = rectangles[0].first.first, maxX = rectangles[0].first.first, minY = rectangles[0].first.second, maxY = rectangles[0].first.second;
    for (int i = 1; i < n; i++) {
        minX = min(minX, rectangles[i].first.first);
        maxX = max(maxX, rectangles[i].second.first);
        minY = min(minY, rectangles[i].first.second);
        maxY = max(maxY, rectangles[i].second.second);
    }
    int side = maxX - minX;
    if (side != maxY - minY) {
        cout << "NO" << endl;
        return 0;
    }
    for (int i = 0; i < n; i++) {
        if (rectangles[i].first.first != minX && rectangles[i].second.first != maxX && rectangles[i].first.second != minY && rectangles[i].second.second != maxY) {
            cout << "NO" << endl;
            return 0;
        }
    }
    cout << "YES" << endl;
    return 0;
}
