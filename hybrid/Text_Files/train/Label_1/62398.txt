n, k = map(int, input().split())
cards = input().strip()

card_count = {}
for card in cards:
    if card in card_count:
        card_count[card] += 1
    else:
        card_count[card] = 1

max_coins = 0
for count in sorted(card_count.values(), reverse=True):
    if k > 0:
        take = min(k, count)
        max_coins += take * take
        k -= take
    else:
        break

print(max_coins)
