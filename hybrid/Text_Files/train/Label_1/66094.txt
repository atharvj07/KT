v = int(input())
a = list(map(int, input().split()))

if v == 0:
    print(-1)
else:
    min_liters = min(a)
    if v < min_liters:
        print(-1)
    else:
        digit = a.index(min_liters) + 1
        max_length = v // min_liters
        remaining_paint = v % min_liters
        result = str(digit) * max_length

        for i in range(max_length):
            for j in range(8, digit - 1, -1):
                if remaining_paint + min_liters >= a[j]:
                    remaining_paint -= a[j] - min_liters
                    result = result[:i] + str(j + 1) + result[i + 1:]
                    break
        print(result)
