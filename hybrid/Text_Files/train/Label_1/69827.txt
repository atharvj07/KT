n = int(input())
a = list(map(int, input().split()))

a.sort()
diffs = [a[i+1] - a[i] for i in range(n-1)]

def check(k):
    segments = 1
    length = 0
    for diff in diffs:
        if diff < k:
            length += 1
        else:
            if length > 0:
                if segments == 1:
                    first_segment_length = length
                elif length != first_segment_length:
                    return False
            length = 0
            segments += 1
    if length > 0 and segments > 1 and length != first_segment_length:
        return False
    return True

l, r = 1, max(a) + 1
while l < r:
    mid = (l + r) // 2
    if check(mid):
        l = mid + 1
    else:
        r = mid

print(l - 1)
