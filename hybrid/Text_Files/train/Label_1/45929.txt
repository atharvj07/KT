import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class SegmentsAndTrees {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Segment> segments = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            segments.add(new Segment(scanner.nextInt(), scanner.nextInt()));
        }
        System.out.println(isTree(segments) ? "YES" : "NO");
    }

    private static boolean isTree(List<Segment> segments) {
        segments.sort((a, b) -> a.l - b.l);
        int[] parent = new int[segments.size() + 1];
        Arrays.fill(parent, -1);
        for (int i = 0; i < segments.size(); i++) {
            Segment segment = segments.get(i);
            for (int j = i + 1; j < segments.size(); j++) {
                Segment other = segments.get(j);
                if (segment.r < other.l || segment.l > other.r) {
                    break;
                }
                if (parent[i] == -1) {
                    parent[i] = j;
                } else if (parent[j] == -1) {
                    parent[j] = i;
                } else {
                    return false;
                }
            }
        }
        for (int i = 0; i < parent.length; i++) {
            if (parent[i] != -1 && parent[parent[i]] != -1) {
                return false;
            }
        }
        return true;
    }

    private static class Segment {
        int l, r;

        public Segment(int l, int r) {
            this.l = l;
            this.r = r;
        }
    }
}
