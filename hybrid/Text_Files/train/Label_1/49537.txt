import java.util.*;

public class Main {
  static final int SPECIAL_CASE = 0, PART_OF = 1;
  static int[] parent;
  static int[] type;
  static int[][] anc;
  static int[][] isAnc;
  static int log;

  static void dfs(int x, int p) {
    parent[x] = p;
    anc[x][0] = p;
    isAnc[x][p] = 1;
    for (int i = 1; i <= log; i++) {
      anc[x][i] = anc[anc[x][i - 1]][i - 1];
      isAnc[x][anc[x][i]] = 1;
    }
    for (int i = 0; i < type[x]; i++) {
      dfs(i, x);
    }
  }

  static boolean isSpecialCase(int u, int v) {
    return isAnc[v][u] > 0;
  }

  static boolean isPartOf(int u, int v) {
    return isAnc[v][u] > 0 && type[u] == PART_OF;
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    parent = new int[n + 1];
    type = new int[n + 1];
    anc = new int[n + 1][20];
    isAnc = new int[n + 1][n + 1];
    for (int i = 1; i <= n; i++) {
      parent[i] = in.nextInt();
      type[i] = in.nextInt();
    }
    for (int i = 1; i <= n; i++) {
      if (parent[i] == -1) dfs(i, 0);
    }
    log = 0;
    while ((1 << log) <= n) log++;
    for (int j = 1; j <= log; j++) {
      for (int i = 1; i <= n; i++) {
        anc[i][j] = anc[anc[i][j - 1]][j - 1];
        isAnc[i][anc[i][j]] = 1;
      }
    }
    int q = in.nextInt();
    for (int i = 0; i < q; i++) {
      int type = in.nextInt(), u = in.nextInt(), v = in.nextInt();
      if (type == 1) {
        if (isSpecialCase(u, v)) System.out.println("YES");
        else System.out.println("NO");
      } else {
        if (isPartOf(u, v)) System.out.println("YES");
        else System.out.println("NO");
      }
    }
  }
}
