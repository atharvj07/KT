#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    long long ans = 0, sum = 0;
    cin >> n;
    vector<int> a(n);
    for (auto &x : a) cin >> x, sum += x;
    sort(begin(a), end(a));
    for (auto &x : a) ans += max(x - abs(sum / n), 0LL);
    cout << ans / 2 << '\n';
  }
}
