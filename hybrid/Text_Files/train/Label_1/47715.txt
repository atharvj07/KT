import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        char[] s = new char[1000000];
        int len = 0;
        for (int i = 0; i < q; i++) {
            int p = sc.nextInt();
            char c = sc.next().charAt(0);
            int n = sc.nextInt();
            for (int j = 0; j < n; j++) {
                s[len + j] = c;
            }
            len += n;
            int l = 0, r = 0;
            for (int j = 0; j < len; j++) {
                if (s[j] == '(') {
                    l++;
                } else {
                    r++;
                    if (r > l) {
                        System.out.println("No");
                        break;
                    }
                }
            }
            if (l == r) {
                System.out.println("Yes");
            }
        }
    }
}
