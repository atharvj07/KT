#include <iostream>
#include <vector>
using namespace std;
using ll = long long;

int main() {
  int n;
  cin >> n;
  vector<ll> b(n - 1), c(n - 1), p(n - 2), a(n);
  for (int i = 0; i < n - 1; i++) cin >> b[i];
  for (int i = 0; i < n - 1; i++) cin >> c[i];
  for (int i = 0; i < n - 2; i++) cin >> p[i];
  if (c[0] > b[0]) return cout << -1, 0;
  a[0] = b[0], a[1] = c[0];
  for (int i = 1; i < n - 1; i++) {
    ll prv = p[i - 1];
    if (c[prv] > b[i]) return cout << -1, 0;
    a[i + 1] = c[prv];
    swap(b[prv], c[prv]);
    a[prv + 1] = b[prv];
  }
  for (const auto& x : a) cout << x << ' ';
  cout << '\n';
}
