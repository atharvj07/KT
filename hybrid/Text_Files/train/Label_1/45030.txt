import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), Q = sc.nextInt();
        long[] a = new long[N];
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextLong();
        }
        while (Q-- > 0) {
            long l = sc.nextLong(), r = sc.nextLong();
            int ans = 0;
            for (long v : a) {
                if (l <= v && v <= r) {
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
