mod = 10**9 + 7

n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]

dp = [[0] * (m+1) for _ in range(n+1)]
dp[n][m] = 1

for i in range(n-1,-1,-1):
    for j in range(m-1,-1,-1):
        if grid[i][j] == 'R':
            continue
        dp[i][j] = (dp[i][j+1] + dp[i+1][j]) % mod

print(dp[0][0])
