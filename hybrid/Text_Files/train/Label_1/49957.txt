import java.util.Scanner;

public class UnimodalArray {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] array = new int[n];
        for (int i = 0; i < n; i++) {
            array[i] = scanner.nextInt();
        }

        boolean isUnimodal = true;
        int increasingIndex = -1;
        int decreasingIndex = -1;

        for (int i = 1; i < n; i++) {
            if (array[i] < array[i - 1]) {
                if (increasingIndex == -1) {
                    increasingIndex = i - 1;
                }
                if (decreasingIndex == -1) {
                    decreasingIndex = i;
                }
            }
        }

        if (increasingIndex != -1 && decreasingIndex != -1 && increasingIndex + 1 != decreasingIndex) {
            isUnimodal = false;
        }

        System.out.println(isUnimodal ? "YES" : "NO");
    }
}
