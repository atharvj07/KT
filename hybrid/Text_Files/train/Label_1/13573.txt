#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
int n, mark[maxn], l[maxn], r[maxn], ans;

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> mark[i];
    for (int i = 1; i <= n; ++i) { // mark[i] <= i-1
        l[mark[i] + 1]++;
        r[mark[i]]--;
    }
    int sum = 0;
    for (int i = 1; i <= n; ++i) {
        sum += l[i] - r[i - 1];
        ans = min(ans, sum);
    }
    cout << ans << endl;
    return 0;
}
