#include <bits/stdc++.h>
using namespace std;
int a[5][100];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < 5; i++)
    for (int j = 0; j < 100; j++) {
      int x = j;
      if (j >= 10) a[i][j] = !a[i][j / 10];
      if (x & (1 << i)) a[i][j] = !a[i][j];
    }
  int ans = 0;
  for (int j = 0; j < 100; j++) {
    int ok = 1;
    for (int i = 0; i < 5; i++) ok &= a[i][n] == a[i][j];
    ans += ok;
  }
  cout << ans << endl;
}
