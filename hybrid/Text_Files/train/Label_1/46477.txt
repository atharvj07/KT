import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            boolean[][] net = new boolean[5][5];
            int[][] faceValue = new int[6][2];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    int value = sc.nextInt();
                    net[j][k] = value != 0;
                    if (value != 0) {
                        faceValue[value - 1][0] = j;
                        faceValue[value - 1][1] = k;
                    }
                }
            }
            Map<Integer, Integer> oppositeFaces = new HashMap<>();
            for (int j = 0; j < 6; j++) {
                for (int k = j + 1; k < 6; k++) {
                    if (faceValue[j][0] == faceValue[k][0]) {
                        oppositeFaces.put(faceValue[j][0], faceValue[k][1]);
                        oppositeFaces.put(faceValue[k][0], faceValue[j][1]);
                    } else if (faceValue[j][1] == faceValue[k][1]) {
                        oppositeFaces.put(faceValue[j][1], faceValue[k][0]);
                        oppositeFaces.put(faceValue[k][1], faceValue[j][0]);
                    }
                }
            }
            boolean isValid = true;
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (net[j][k]) {
                        if (k != 4 && net[j][k + 1] || j != 4 && net[j + 1][k] || j != 0 && net[j - 1][k] || k != 0 && net[j][k - 1])
                            isValid = false;
                        if (oppositeFaces.containsKey(j) && oppositeFaces.get(j) == k ||
                            oppositeFaces.containsKey(k) && oppositeFaces.get(k) == j) {
                            if (j == 0 || k == 0 || j == 4 || k == 4) isValid = false;
                        }
                    }
                }
            }
            for (int j = 0; j < 6 && isValid; j++) {
                if (faceValue[j][0] == -1 || faceValue[j][1] == -1) isValid = false;
            }
            System.out.println(isValid);
        }
    }
}
