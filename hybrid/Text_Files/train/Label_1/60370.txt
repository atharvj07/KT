import sys
input = sys.stdin.readline

def msleep(S,K,I):
    S.sort()
    # two pointers
    li,ri = 0,len(S)-1
    # try all possible (l,r) pairs
    ans = len(S)
    while li<=ri:
        l,r = S[li],S[ri]
        tmp = (r-l+1)*len(S) - sum([max(0,x-l) for x in S]) - sum([max(0,r-x) for x in S])
        if tmp<=8*I:
            ans = min(ans, sum([x>r or x<l for x in S]))
        if tmp<=8*I: ri-=1
        else: li+=1
    return ans

n,I = map(int,input().split())
S = list(map(int,input().split()))
print(msleep(S,n,I))
