#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        unordered_map<int,int>mp;
        int mx=0;
        for(int i=1;i<=n;i++){
            int x;
            cin>>x;
            mp[x]++;
            mx=max(mx,mp[x]);
        }
        int sz=(n-mx+1)/2;
        if(sz>0 && sz<=mx)
            cout<<sz<<endl;
        else
            cout<<0<<endl;
        
    }
}
