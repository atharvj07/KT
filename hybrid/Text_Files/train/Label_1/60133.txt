def find_min_days(n, p, q, projects):
    # Sort projects based on their efficiency (experience+money) to total ratio
    projects.sort(key=lambda x: -((p*x[0] + q*x[1]) / (x[0]**2 + x[1]**2)))
    
    exp, money, days = 0, 0, 0
    for a, b in projects:
        # Check if adding this project would exceed the requirements
        need_exp, need_money = max(0, p - exp), max(0, q - money)
        if a * need_money >= b * need_exp:  # If the project can contribute to the goal
            required_days = max(need_exp / a, need_money / b)
            exp += required_days * a
            money += required_days * b
            days += required_days
            if exp >= p and money >= q:
                break
    
    return days

# Input reading
n, p, q = map(int, input().split())
projects = [tuple(map(int, input().split())) for _ in range(n)]

# Calculate and print result
print(f"{find_min_days(n, p, q, projects):.12f}")
