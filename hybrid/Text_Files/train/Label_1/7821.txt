#include <bits/stdc++.h>
#include<algorithm>
using namespace std;
typedef long long ll;
typedef pair<int, int> pi;
typedef pair<ll, int> pli;
#define endl '\n'
const int mod = 1e9 + 7;

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    int n; 
    cin >> n;
    vector<pair<ll, ll>>v;
    for (int i = 0; i < n; i++) {
        ll x, y;
        cin >> x >> y;
        v.push_back({ x,y });
    }
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (i == j) continue;
            for (int k = j + 1; k < n; k++) {
                if (i == k || j == k) continue;
                int c1 = 0, c2 = 0;
                if ((v[i].first - v[j].first) * (v[i].first - v[k].first) < 0) c1 = 1;
                if ((v[i].second - v[j].second) * (v[i].second - v[k].second) < 0) c2 = 1;
                if (c1 * c2 > 0)ans++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
