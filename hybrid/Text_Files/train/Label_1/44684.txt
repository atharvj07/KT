import java.io.*;
import java.util.*;

public class Main {
    static long[] a;
    public static void main(String[] args) throws IOException {
        BufferedReader buf = new BufferedReader(new InputStreamReader(System.in));
        String[] line = buf.readLine().split(" ");
        int[] arr = {Integer.parseInt(line[0]), Integer.parseInt(line[1]), Integer.parseInt(line[2])};
        line = buf.readLine().split(" ");
        a = new long[arr[0] + arr[1] + arr[2] + 1];
        for (String s : line) a[Integer.parseInt(s)] = 1;
        line = buf.readLine().split(" ");
        for (String s : line) a[Integer.parseInt(s)] = -1;
        line = buf.readLine().split(" ");
        for (String s : line) a[Integer.parseInt(s)] = 1;
        int ans = 0;
        int c = 0;
        for (int i = 0; i < a.length; i++) {
            if (a[i] == 1) {
                if (c != 0) ans += (c > 0 ? c - 1 : 1);
                c = 1;
            } else if (a[i] == -1) {
                if (c != 0) ans += (c > 0 ? c - 1 : 1);
                c = -1;
            } else c = 0;
        }
        System.out.println(ans);
    }
}
