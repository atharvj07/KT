#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> ans(n + 1, 0), p(n + 1, 0);
    for (int i = 1; i <= n; ++i) cin >> p[i];
    for (int i = 1; i <= n; ++i) {
        vector<bool> vis(n + 1, 0);
        vis[i] = 1;
        int x = i;
        while (!vis[p[x]]) {
            vis[x] = 1;
            x = p[x];
        }
        ans[i] = x;
    }
    for (int i = 1; i <= n; ++i) cout << ans[i] << ' ';
    cout << '\n';
    return 0;
}
