#include <bits/stdc++.h>

using namespace std;

#define pb push_back
#define mp make_pair

struct DSU {
    vector<int> parent;
    vector<int> size;

    DSU(int n) : parent(vector<int>(n)), size(vector<int>(n, 1)) {
        for (int i = 0; i < n; i++) parent[i] = i;
    }

    int find_set(int v) {
        if (v == parent[v])
            return v;
        return parent[v] = find_set(parent[v]);
    }

    void union_sets(int a, int b) {
        a = find_set(a);
        b = find_set(b);
        if (a != b) {
            if (size[a] < size[b])
                swap(a, b);
            parent[b] = a;
            size[a] += size[b];
        }
    }
};

int main() {
    int n, m;
    cin >> n >> m;
    vector<pair<int, int>> edges;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        u--;
        v--;
        edges.pb(mp(u, v));
    }
    DSU dsu(n);
    for (auto& edge : edges) {
        int u = edge.first;
        int v = edge.second;
        dsu.union_sets(u, v);
    }
    int ans = 0;
    vector<int> order;
    for (int i = 0; i < n; i++) {
        if (dsu.find_set(i) == i) {
            ans++;
            order.pb(i + 1);
        }
    }
    cout << ans << endl;
    for (auto& v : order) cout << v << " ";
    cout << endl;
}
