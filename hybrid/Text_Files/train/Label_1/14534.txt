#include<bits/stdc++.h>
using namespace std;
const int N = 1e5+10;
int n,x[N],h[N],ans=0,mx=0;
pair<int,int> tree[N];
vector<pair<int,int>> seg;
int main(){
	cin >> n;
	for(int i=0;i<n;i++){
		cin >> x[i] >> h[i];
		tree[i] = {x[i]-h[i],x[i]+h[i]};
	}
	sort(tree,tree+n);
	for(int i=0;i<n;i++){
		if(tree[i].first>mx){
			ans++;
			mx=tree[i].second;
		}
	}
	cout << ans;
	return 0;
}
