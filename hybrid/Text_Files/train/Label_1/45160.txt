import java.util.*;
import java.io.*;

public class Main {

    static class Segment {
        int row, left, right;

        Segment(int row, int left, int right) {
            this.row = row;
            this.left = left;
            this.right = right;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        String[] s = br.readLine().split(" ");
        int x0 = Integer.parseInt(s[0]);
        int y0 = Integer.parseInt(s[1]);
        int x1 = Integer.parseInt(s[2]);
        int y1 = Integer.parseInt(s[3]);

        int n = Integer.parseInt(br.readLine());
        List<Segment> segments = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            s = br.readLine().split(" ");
            int row = Integer.parseInt(s[0]);
            int left = Integer.parseInt(s[1]);
            int right = Integer.parseInt(s[2]);
            segments.add(new Segment(row, left, right));
        }

        // Calculate the minimum number of moves using BFS
        Queue<int[]> queue = new LinkedList<>();
        int[][] dist = new int[110][110];
        for (int i = 0; i < 110; i++) {
            for (int j = 0; j < 110; j++) {
                dist[i][j] = -1;
            }
        }

        queue.add(new int[]{x0, y0});
        dist[x0][y0] = 0;
        while (!queue.isEmpty()) {
            int[] curr = queue.poll();
            int x = curr[0];
            int y = curr[1];

            // Check if we have reached the destination
            if (x == x1 && y == y1) {
                System.out.println(dist[x][y]);
                return;
            }

            // Try all possible moves from the current position
            for (int dx = -1; dx <= 1; dx++) {
                for (int dy = -1; dy <= 1; dy++) {
                    if (dx == 0 && dy == 0) continue;
                    int nx = x + dx;
                    int ny = y + dy;

                    // Check if the new position is within the allowed cells
                    boolean allowed = false;
                    for (Segment segment : segments) {
                        if (segment.row == nx && segment.left <= ny && ny <= segment.right) {
                            allowed = true;
                        }
                    }

                    // If the new position is allowed and we haven't visited it yet, add it to the queue
                    if (allowed && dist[nx][ny] == -1) {
                        queue.add(new int[]{nx, ny});
                        dist[nx][ny] = dist[x][y] + 1;
                    }
                }
            }
        }

        // If we haven't found a path, print -1
        System.out.println(-1);
    }
}
