import java.util.*;
import java.math.*;

public class Main {
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        int n = sc.nextInt();
        int q = sc.nextInt();
        String s = sc.next();
        char[] arr = s.toCharArray();
        while (q-- > 0) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            int[] ans = new int[10];
            int[] freq = new int[10];
            int dir = 1;
            int ptr = l;
            while (ptr >= 0 && ptr < r) {
                if (Character.isDigit(arr[ptr])) {
                    int dig = arr[ptr] - '0';
                    ans[dig]++;
                    arr[ptr]--;
                    if (arr[ptr] == '0') {
                        if (ptr > 0 && arr[ptr - 1] == arr[ptr]) {
                            arr[ptr - 1]--;
                            ptr--;
                        } else if (ptr < r - 1 && arr[ptr + 1] == arr[ptr]) {
                            arr[ptr + 1]--;
                            ptr++;
                        } else {
                            continue;
                        }
                    } else {
                        freq[dig]++;
                    }
                } else if (arr[ptr] == '>')
                    dir = 1;
                else if (arr[ptr] == '<')
                    dir = -1;
                ptr += dir;
            }
            for (int i = 0; i < 10; i++)
                ans[i] = freq[i] + ans[i];
            for (int i = 0; i < 10; i++) {
                System.out.print(ans[i] + " ");
            }
            System.out.println();
        }
    }
}
