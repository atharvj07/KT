#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, q;
  cin >> n;
  vector<int> x(n);
  for (int i = 0; i < n; i++) { cin >> x[i]; }
  cin >> q;
  vector<int> m(q);
  for (int i = 0; i < q; i++) { cin >> m[i]; }
  for (int i = 0; i < q; i++) {
    int cnt = 0;
    for (int j = 0; j < n; j++) {
      if (x[j] <= m[i]) cnt++;
    }
    cout << cnt << '\n';
  }
  return 0;
}
