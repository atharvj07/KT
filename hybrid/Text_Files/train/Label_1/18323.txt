#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, ans = 0;
    cin >> n;
    vector<int> rows(n);
    for (int i = 0; i < n; i++) {
        cin >> rows[i];
    }
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        bool ok = true;
        for (int j = 0; j < n; j++) {
            if ((rows[j] >> i) & 1)
                ok = false;
        }
        if (ok) {
            cnt++;
        }
    }
    for (int i = 0; i < n; i++) {
        bool ok = true;
        for (int j = 0; j < n; j++) {
            if ((rows[j] >> i) & 1 == 0)
                ok = false;
        }
        if (ok) {
            cnt++;
        }
    }
    cout << cnt << endl;
    return 0;
}
