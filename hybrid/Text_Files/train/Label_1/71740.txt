n, m = map(int, input().split())

if m >= n:
    print(n)
else:
    # Calculate the day when the barn will become empty
    # Using binary search to find the day when total grain eaten by sparrows exceeds or equals the initial grain plus the total grain added minus m
    left, right = 1, 2*n
    while left < right:
        mid = (left + right) // 2
        total_eaten = mid * (mid + 1) // 2
        if total_eaten >= n + (mid - m):
            right = mid
        else:
            left = mid + 1
    print(left)
