#include<iostream>
using namespace std;
struct s
{int n,pos,num;bool p;}e[800002];
int a[400002],vis[400002],q;
bool d[400002];
int main()
{
    int i,j,r=0,n,k,t;
    cin>>n>>k;
    for(i=1;i<=n;i++)
    {
        cin>>a[i];
        if(!vis[a[i]]) {t=i;vis[a[i]]=i;}
        e[i]={n+1-i,t,a[i],0};
        t++;
        if(!vis[a[i]]) {t=i;vis[a[i]]=i;}
        e[n+1-i]={i,t,a[i],1};
        t++;
    }
    for(i=1;i<=n+n;i++)
    {
        if(!d[e[i].num])
        {
            if(q&&!e[q].p) r++;
            q=i;
            d[e[i].num]=1;
        }
        else
        {
            if(i-q>=k) {d[e[i].num]=0;d[e[q].num]=0;q=0;}
            else {if(q&&!e[q].p) r++;}
        }
    }
    cout<<r<<'\n';
    return 0;
}
