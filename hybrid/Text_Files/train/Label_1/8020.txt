#include <iostream>
#include <utility>
#include <algorithm>
using namespace std;
int main() {
    int n, a, b, c = 0;
    cin >> n >> a >> b;
    pair<int, int> s[n];
    for (int i = 0; i < n; i++) {
        int x, y;
        cin >> x >> y;
        if (min(x, y) <= a && max(x, y) <= b)
            c += x * y, s[i] = {x, y};
        else if (min(x, y) <= b && max(x, y) <= a)
            c += x * y, s[i] = {y, x};
    }
    sort(s, s + n);
    for (int i = 0, j = n - 1; i < j;) {
        if (s[i].first + s[j].first > a)
            j--;
        else {
            c = max(c, s[i].first * s[j].first + s[i].second * s[j].second);
            i++;
        }
    }
    cout << c << endl;
    return 0;
}
