#include <iostream>
#include <vector>

int main()
{
    int f, i, t;
    std::cin >> f >> i >> t;
    std::vector<std::string> v(f);
    for (auto& s : v) {
        std::cin >> s;
    }
    int ans = 0;
    for (int j = 0; j < i; j++) {
        int cnt = 0;
        for (const auto& s : v) {
            cnt += s[j] == 'Y';
        }
        ans += cnt >= t;
    }
    std::cout << ans << std::endl;
}
