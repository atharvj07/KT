import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        int k = 1;
        for (int i = n - 1; i >= 0; i--) {
            if (k <= a[i]) {
                ans++;
                k++;
            }
        }
        System.out.println(ans);
    }
}
