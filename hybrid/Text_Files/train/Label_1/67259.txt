def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)
x, y = list(map(int, input().split()))
if x < y:  x, y = y, x
if x == y:  print("Impossible")
elif x % y == 0:  print(str(y) + 'B')
else:
    g = gcd(x, y)
    a, b = x // g, y // g
    if (a + b) % 3 != 0:
        print("Impossible")
    else:
        print(str(b // 3) + 'A' + str(a // 3 + b // 3) + 'B' + str(b // 3) + 'A' + str(a // 3) + 'B')
