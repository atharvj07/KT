from collections import defaultdict

def solve():
    n, m = map(int, input().split())
    edges = defaultdict(list)
    for _ in range(m):
        u, v, c = input().split()
        edges[int(u)].append((int(v), c))
        edges[int(v)].append((int(u), c))

    vis = [False] * (n + 1)
    color = [None] * (n + 1)

    def dfs(node, clr):
        if vis[node]:
            return clr == color[node]
        vis[node] = True
        color[node] = clr
        for adj, edge_color in edges[node]:
            if not dfs(adj, 'R' if edge_color == 'B' else 'B'):
                return False
        return True

    if not dfs(1, 'R'):
        print(-1)
        return

    print(sum(vis) - 1)
    for i in range(1, n + 1):
        if vis[i]:
            print(i, end=' ')

solve()
