#include <iostream>
#include <vector>
#include <map>
#include <utility>

struct Node {
  int v;
  Node* n[2];
  Node(): v(0), n{nullptr, nullptr} {}
};

int main() {
  using namespace std;
  int n; cin >> n;
  vector<Node*> r(n + 1);
  map<pair<int, int>, int> id;
  for (int i = 1; i <= n; i++) {
    int a, b; cin >> a >> b;
    r[i] = new Node;
    r[i]->v = i;
    if (!id.count({a, b})) {
      id[{a, b}] = i;
    } else {
      int idx = id[{a, b}];
      if (!r[idx]->n[0]) {
        r[idx]->n[0] = r[i];
      } else {
        r[idx]->n[1] = r[i];
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    if (!r[i]->n[0]) {
      Node* c = r[i];
      while (c != nullptr) {
        cout << c->v << " ";
        c = c->n[1];
      }
    }
  }
  cout << endl;
  return 0;
}
