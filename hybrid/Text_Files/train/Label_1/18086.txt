#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
int main(){
    int t;
    cin>>t;
    while(t--){
        int n,m,k;
        cin>>n>>m>>k;
        vector<int>v(n,0);
        map<int,int>mp;
        for(int i=0;i<n;i++){
            int x;
            cin>>x;
            v[i]=x;
            mp[x]++;
        }
        if(k==0){
            long long ans=1;
            for(int i=0;i<m;i++){
                ans=ans*mp[v[i]]%mod;
            }
            cout<<ans<<"\n";
            continue;
        }
        sort(v.begin(),v.end());
        int st=0,en=0,cnt=0;
        long long ans=0;
        while(en<n){
            while(v[en]-v[st]<=k and en<n){
                cnt+=mp[v[en]];
                en++;
            }
            if(en-st>=m){
                ans+=1LL*mp[v[st]]*cnt%mod;
            }
            mp[v[st]]--;
            cnt-=mp[v[st]];
            st++;
        }
        cout<<ans%mod<<"\n";
    }
    return 0;
}
