MOD = 1000000007

def solve(t, k, queries):
    # Precompute the number of ways to eat exactly i flowers
    dp = [0] * 100001
    dp[0] = 1
    for i in range(1, 100001):
        dp[i] = dp[i-1]
        if i >= k:
            dp[i] = (dp[i] + dp[i-k]) % MOD
    
    # Precompute the cumulative sum
    for i in range(1, 100001):
        dp[i] = (dp[i] + dp[i-1]) % MOD
    
    # Answer the queries
    result = []
    for a, b in queries:
        result.append((dp[b] - dp[a-1] + MOD) % MOD)
    return result

# Input handling
t, k = map(int, input().split())
queries = [list(map(int, input().split())) for _ in range(t)]

# Solve and output
answers = solve(t, k, queries)
for ans in answers:
    print(ans)
