import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] p = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] q = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int m = Integer.parseInt(br.readLine());
        int[] ans = new int[m];
        for (int i = 0; i < m; i++) {
            int[] a = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            int x = i > 0 ? ans[i - 1] : 0;
            int l1 = Math.min((a[0] - 1 + x) % n + 1, (a[1] - 1 + x) % n + 1);
            int r1 = Math.max((a[0] - 1 + x) % n + 1, (a[1] - 1 + x) % n + 1);
            int l2 = Math.min((a[2] - 1 + x) % n + 1, (a[3] - 1 + x) % n + 1);
            int r2 = Math.max((a[2] - 1 + x) % n + 1, (a[3] - 1 + x) % n + 1);
            for (int j = 0; j < n; j++) {
                if (l1 <= p[j] && p[j] <= r1 && l2 <= q[j] && q[j] <= r2) {
                    ans[i]++;
                }
            }
        }
        for (int an : ans) {
            System.out.println(an);
        }
    }
}
