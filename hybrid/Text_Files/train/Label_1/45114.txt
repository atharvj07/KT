import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int q = sc.nextInt();
        int[] count = new int[26];
        for (int i = 0; i < s.length(); i++) count[s.charAt(i) - 'a']++;
        StringBuilder t = new StringBuilder(s);
        while (q-- > 0) {
            int type = sc.nextInt();
            if (type == 1) {
                int i = sc.nextInt() - 1;
                char c = sc.next().charAt(0);
                count[t.charAt(i) - 'a']--;
                t.setCharAt(i, c);
                count[c - 'a']++;
            } else {
                int l = sc.nextInt() - 1;
                int r = sc.nextInt() - 1;
                String y = sc.next();
                int ans = 0;
                int[] cnt = new int[26];
                for (int i = l; i <= r; i++) {
                    cnt[t.charAt(i) - 'a']++;
                    if (cnt[t.charAt(i) - 'a'] == count[t.charAt(i) - 'a'] / (r - l + 1)) ans++;
                }
                if (ans == y.length()) System.out.println(1);
                else System.out.println(0);
            }
        }
    }
}
