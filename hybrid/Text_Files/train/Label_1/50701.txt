import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n], b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextLong();
        }
        long sum = 0, min = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            sum += b[i];
            min = Math.min(min, b[i]);
        }
        int ans = 0;
        if (sum == 0) {
            ans = n;
        } else {
            for (int i = 0; i < n; i++) {
                if (a[i] == 0) {
                    ans++;
                }
                if (sum - b[i] + a[i] == 0) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
