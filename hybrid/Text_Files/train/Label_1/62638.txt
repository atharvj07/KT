import sys

n, m = map(int, input().split())
a = list(map(int, input().split()))
p = list(range(1, n + 1))
for i in range(m):
    x, y = map(int, input().split())
    p[x - 1], p[y - 1] = p[y - 1], p[x - 1]
b = [0] * (n + 1)
for i in range(n):
    b[a[i]] = p[i]
for i in range(1, n + 1):
    print(b[i], end=" ")
