import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    long[] arr = new long[n];
    for (int i = 0; i < n; i++) {
      arr[i] = sc.nextInt();
    }
    for (int i = 0; i < m; i++) {
      int ti = sc.nextInt();
      if (ti == 1) {
        int vi = sc.nextInt();
        long xi = sc.nextInt();
        arr[vi - 1] = xi;
      } else if (ti == 2) {
        long yi = sc.nextInt();
        for (int j = 0; j < n; j++) {
          arr[j] += yi;
        }
      } else {
        int qi = sc.nextInt();
        System.out.println(arr[qi - 1]);
      }
    }
  }
}
