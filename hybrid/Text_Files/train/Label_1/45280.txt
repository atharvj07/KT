import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.Map;

public class Main {
    static Map<Long, Integer> cntMap;
    static int n, m, ans, minx, maxx, miny, maxy;
    static long[][] arr;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");

        n = Integer.parseInt(line[0]);
        m = Integer.parseInt(line[1]);
        cntMap = new HashMap<>();
        arr = new long[n][m];
        for (int i = 0; i < n; i++) {
            line = br.readLine().split(" ");
            for (int j = 0; j < m; j++) {
                arr[i][j] = Long.parseLong(line[j]);
                cntMap.put(arr[i][j], cntMap.getOrDefault(arr[i][j], 0) + 1);
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                long val = arr[i][j];
                if (cntMap.get(arr[i][j]) % 2 == 1) {
                    System.out.println(-1);
                    return;
                }

                int cnt = cntMap.get(val) / 2, j1 = j;
                while (cnt > 0 && j1 < m) {
                    if (arr[i][j1] == val) {
                        cnt--;
                        if (j1 < j) {
                            ans++;
                            minx = Math.min(minx, i);
                            miny = Math.min(miny, j);
                            maxx = Math.max(maxx, i);
                            maxy = Math.max(maxy, j1);
                        }
                    }
                    j1++;
                }

                if (cnt > 0) {
                    val = val - 1;
                    j1 = j - 1;
                    while (cnt > 0 && j1 >= 0) {
                        if (arr[i][j1] == val) {
                            cnt--;
                            if (j1 > j) {
                                ans++;
                                minx = Math.min(minx, i);
                                miny = Math.min(miny, j1);
                                maxx = Math.max(maxx, i);
                                maxy = Math.max(maxy, j);
                            }
                        }
                        j1--;
                    }
                }
            }
        }

        System.out.println(ans);
    }
}
