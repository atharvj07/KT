import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int t = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int ans = 0;
        int time = 0;
        for (int i = 0; i < n; i++) {
            if (time + a[i] <= t) {
                time += a[i];
                ans++;
            } else {
                break;
            }
        }
        System.out.println(ans);
    }
}
