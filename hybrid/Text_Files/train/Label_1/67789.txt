def check(x, y):
    for i in range(2):
        for j in range(2):
            if not grid[x + i][y + j]:
                return False
    return True

n, m, k = map(int, input().split())
grid = [[False] * (m + 1) for _ in range(n + 1)]

for idx in range(k):
    r, c = map(int, input().split())
    grid[r][c] = True
    if check(r, c):
        print(idx + 1)
        exit()

print(0)
