#include <cstdio>
int main() {
  long long n;
  scanf("%lld", &n);
  long long ans;
  if (n & 1) {
    if ((n * n + 1) % 8 == 0) {
      ans = (n * n + 1) / 8;
      printf("%lld %lld\n", ans * 3, ans * 4);
    } else {
      printf("-1\n");
    }
  } else {
    if ((n / 2) & 1) {
      printf("-1\n");
    } else {
      ans = n / 2;
      if ((ans * ans + 1) % 8 == 0) {
        ans = (ans * ans + 1) / 8;
        printf("%lld %lld\n", ans * 3, ans * 4);
      } else {
        ans = (ans * ans - 1) / 8;
        printf("%lld %lld\n", ans * 3, ans * 4);
      }
    }
  }
  return 0;
}
