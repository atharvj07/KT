#include <bits/stdc++.h>
using namespace std;
int main() {
    int t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        vector<int> a(n);
        for (int &x : a) cin >> x;
        sort(a.begin(), a.end());
        long long ans = 1LL * a[0] * a[n - 1];
        for (int i = 0; i < n - 1; i++) ans = max(ans, 1LL * a[i] * a[i + 1]);
        cout << ans << '\n';
    }
    return 0;
}
