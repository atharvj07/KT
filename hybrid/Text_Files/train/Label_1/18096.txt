#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main()
{
    cin.tie(nullptr);
    ios::sync_with_stdio(false);

    int t;
    cin >> t;
    while (t--)
    {
        int n;
        cin >> n;

        vector<int> a(n);
        for (int i = 0; i < n; i++)
        {
            cin >> a[i];
        }

        bool found = false;
        for (int i = 1; i < n - 1; i++)
        {
            if (a[i + 1] - a[i] > a[i] - a[i - 1])
            {
                cout << i << " " << i + 1 << " " << i + 2 << "\n";
                found = true;
                break;
            }
        }
        if (!found)
        {
            cout << -1 << "\n";
        }
    }

    return 0;
}
