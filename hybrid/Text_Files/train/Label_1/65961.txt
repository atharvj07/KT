for t in range(int(input())):
    n = int(input())
    l = [int(x) for x in input().split()]
    d = {}
    for x in l:
        if x not in d: d[x] = 0
        d[x] += 1
    l.sort()
    if len(d) == 1:
        print(n)
    elif len(d) == 2:
        x = list(d.keys())
        if x[1] == x[0] + 1 and d[x[0]] == 1:
            print(d[x[1]] - 1)
        else:
            print(min(d.values()) * 2)
    else:
        print(max(d.values()))
