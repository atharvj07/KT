#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

const int maxn = 1e3+5;
const int maxm = 6+1;

vector<int> graph[maxn];

int total, n, m, d[maxm], seen[maxn];

bool valid(int u, int times, int mismatch) {
    if(times == m) return mismatch <= 1;
    for(int v : graph[u]) {
        if(seen[v]) continue;
        seen[v] = 1;
        if(d[times] == d[times+1])
            if(valid(v, times+1, mismatch))
                return true;
        if(d[times] != d[times+1])
            if(valid(v, times+1, mismatch+1))
                return true;
        seen[v] = 0;
    }
    return false;
}

int main() {
    int t, ans = 0;
    cin >> n;
    for(int i=0; i<n; i++) {
        cin >> t;
        for(int j=0; j<m; j++) {
            d[j] = t%10;
            t /= 10;
        }
        graph[i].resize(n);
        for(int j=0; j<n; j++) {
            for(int k=0; k<m; k++) {
                cin >> t;
                if(t == d[k]) {
                    graph[i][j] = 1;
                    break;
                }
            }
        }
    }
    for(int i=0; i<n; i++) {
        fill(seen, seen+maxn, 0);
        if(valid(i, 0, 0)) ans++;
    }
    cout << ans << endl;
    return 0;
}
