import java.util.Scanner;
import java.util.Stack;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String golorpName = scanner.nextLine();

        Stack<Character> jaws = new Stack<>();
        Stack<Character> stomach = new Stack<>();

        for (char c : golorpName.toCharArray()) {
            switch (c) {
                case '(':
                case '[':
                    jaws.push(c);
                    break;
                case ')':
                case ']':
                    if (jaws.isEmpty() || jaws.pop() != c) {
                        System.out.println("false");
                        return;
                    }
                    break;
                case '>':
                    if (!jaws.isEmpty()) {
                        stomach.push(jaws.pop());
                    } else {
                        System.out.println("false");
                        return;
                    }
                    break;
                case '<':
                    if (!stomach.isEmpty()) {
                        jaws.push(stomach.pop());
                    } else {
                        System.out.println("false");
                        return;
                    }
                    break;
                default:
                    if (Character.isDigit(c) && c >= '0' && c <= '9') {
                        jaws.push(c);
                    } else {
                        System.out.println("false");
                        return;
                    }
            }
        }

        StringBuilder result = new StringBuilder();
        while (!jaws.isEmpty()) {
            result.append(jaws.pop());
        }
        while (!stomach.isEmpty()) {
            result.append(stomach.pop());
        }

        System.out.println(result);
    }
}
