#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  int64_t n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  set<char> letters;
  for (auto c : s) letters.insert(c);
  string ans(k, 'a');
  int64_t i = k - 1;
  for (auto c = letters.rbegin(); c != letters.rend() && i >= 0; c++) {
    ans[i] = *c;
    i--;
  }
  cout << ans << "\n";
  return 0;
}
