#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9;

struct num {
    int val = 0;
    int dig[10] = {};
};
vector<num> arr;

int main() {
    int n, x;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> x;
        arr.push_back({x});
        int tmp = x;
        while (tmp > 0) {
            arr.back().dig[tmp % 10]++;
            tmp /= 10;
        }
    }
    for (int i = 0; i < 10; i++) { arr[0].dig[i] += arr[1].dig[i]; }

    for (int i = 2; i < n; i++) {
        bool flag = 0;
        for (int j = 0; j < 10; j++) {
            if (arr[0].dig[j] && arr[i].dig[j] > 0) { flag = 1; }
        }
        if (flag) { arr[0] = arr[i]; }
    }
    int cnt = 0;
    for (int i = 0; i < 10; i++) { cnt += (arr[0].dig[i] > 0); }
    cout << cnt << endl;
    for (int i = 0; i < 10; i++) {
        if (arr[0].dig[i] > 0) { cout << i << " "; }
    }
    cout << endl;
    return 0;
}
