# coding: utf-8
from typing import Tuple, List

dx = [0, 1, 0, -1]
dy = [1, 0, -1, 0]
symbols = ['R', 'D', 'L', 'U']
visited = [[False for _ in range(1002)] for _ in range(1002)]

def get_min_path(x: int, y: int, k: int, maz: List[List[str]]) -> Tuple[int, List[str]]:
    if visited[x][y] or maz[x][y] == '*':
        return int(1e9), []

    if k == 1:
        return 0, [maz[x][y]]

    visited[x][y] = True
    ans = int(1e9), []
    for i in range(4):
        nx = x + dx[i]
        ny = y + dy[i]
        ok, cur_path = get_min_path(nx, ny, k - 1, maz)
        visited[x][y] = False
        if ok < int(1e9):
            ans = min(ans, (ok, [symbols[i]] + cur_path))

    return ans

n, m, k = map(int, input().split())
maze = [['.'] * (m + 2) for _ in range(n + 2)]

for i in range(1, n + 1):
    tmp = list(input())
    for j in range(1, m + 1):
        if tmp[j - 1] == 'X':
            x, y = i, j
        maze[i][j] = tmp[j - 1]

ok, path = get_min_path(x, y, k, maze)

if ok == int(1e9):
    print('IMPOSSIBLE')
else:
    print(''.join(path))
