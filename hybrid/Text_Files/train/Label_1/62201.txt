import bisect
from math import pi

n,r = map(int,input().split())
x = [0] + list(map(int,input().split()))
L = [1e100] * (n+1)
for i in range(1,n+1):
 L[i] = min(L[i], 2*r)
 for j in range(i-1,-1,-1):
  if x[i] == x[j]:
   continue
  c = (x[i]-x[j])**2 - (2*r)**2
  if c <= 0:
   continue
  L[i] = min(L[i], (2*r)*(2*r - (x[i]-x[j])**2 / (x[i]-x[j])) / c)
R = [1e100] * (n+1)
for i in range(n,0,-1):
 R[i] = min(R[i], 2*r)
 for j in range(i+1,n+1):
  if x[i] == x[j]:
   continue
  c = (x[i]-x[j])**2 - (2*r)**2
  if c <= 0:
   continue
  R[i] = min(R[i], (2*r)*(2*r - (x[i]-x[j])**2 / (x[i]-x[j])) / c)
ans = []
for i in range(1,n+1):
 if L[i] < R[i]:
  ans.append(L[i])
 else:
  ind = bisect.bisect_right(x,x[i]-2*r)
  if ind > i:
   ind -= 1
  if x[ind] == x[i]-2*r:
   ans.append(2*r)
  else:
   c = (x[i]-x[ind])**2 - (2*r)**2
   if c < 0:
    c = 0
   ans.append( (2*r)*(2*r - c / (x[i]-x[ind])) / c )

print(*ans)
