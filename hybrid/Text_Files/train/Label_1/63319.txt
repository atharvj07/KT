def min_days_to_reach(x1, y1, x2, y2, n, s):
    def check(mid):
        x_shift = mid // n * dx + sum(x_dir[:mid % n])
        y_shift = mid // n * dy + sum(y_dir[:mid % n])
        return abs(x2 - x1 - x_shift) + abs(y2 - y1 - y_shift) <= mid

    dx = dy = 0
    x_dir = []
    y_dir = []

    for direction in s:
        if direction == 'U':
            dy += 1
            y_dir.append(1)
            x_dir.append(0)
        elif direction == 'D':
            dy -= 1
            y_dir.append(-1)
            x_dir.append(0)
        elif direction == 'L':
            dx -= 1
            x_dir.append(-1)
            y_dir.append(0)
        else:  # direction == 'R'
            dx += 1
            x_dir.append(1)
            y_dir.append(0)

    left, right = 0, 2 * 10**18
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if check(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer

x1, y1 = map(int, input().split())
x2, y2 = map(int, input().split())
n = int(input())
s = input().strip()

print(min_days_to_reach(x1, y1, x2, y2, n, s))
