#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5+5;
bool vis[maxn],tmp[maxn];
vector<int> vec[maxn];
int main() {
    int n, m, q; cin >> n >> m >> q;
    while(q--) {
        int x, y; cin >> x >> y;
        vis[x*m+y] = 1;
    }
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            if(vis[i*m+j]) {
                tmp[i] = tmp[j] = 1;
            }
        }
    }
    int ans = n*m;
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            if(!vis[i*m+j]) ans -= tmp[i]+tmp[j];
        }
    }
    cout << ans << endl;
}
