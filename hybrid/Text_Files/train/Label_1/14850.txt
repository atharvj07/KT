#include<bits/stdc++.h>
using namespace std;
int main(){
    int a,b,c,sum=0,ans=0,x[200010];
    cin>>a>>b>>c;
    sum=a+b+c;
    int fa=0,fb=a,fc=fb+b;
    for(int i=0; i<a; i++){
        cin>>x[i];
        if(x[i]>fc) ans++;
    }
    for(int i=a; i<fb; i++){
        cin>>x[i];
        if(x[i]<fa || x[i]>fc) ans++;
    }
    fa=fc,fb=sum;
    for(int i=fb; i<sum; i++){
        cin>>x[i];
        if(x[i]<fa) ans++;
    }
    cout<<ans;
}
