#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, d, ans = 0, l = 1, r = 1, cnt = 0;
    cin >> n >> k >> d;
    vector<int> a(n + 1);
    vector<bool> seen(k + 1, false);
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
    }
    seen[a[l]] = true;
    cnt = 1;
    while (l <= r) {
      if (cnt >= k) {
        if (r - l + 1 >= d) {
          ans = k;
          break;
        } else {
          r++;
          if (r > n) break;
          if (!seen[a[r]]) {
            cnt++;
            seen[a[r]] = true;
          }
        }
      } else {
        r++;
        if (r > n) break;
        if (!seen[a[r]]) {
          cnt++;
          seen[a[r]] = true;
        }
      }
      if (r - l + 1 >= d) {
        while (r - l + 1 >= d) {
          if (seen[a[l]]) cnt--;
          seen[a[l]] = false;
          l++;
        }
        ans = min(ans, cnt);
      }
    }
    if (ans == 0) ans = k;
    cout << ans << endl;
  }
  return 0;
}
