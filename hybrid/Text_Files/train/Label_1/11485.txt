#include <bits/stdc++.h>
using namespace std;typedef long long ll;
ll mod=1e9+7;ll n;vector<ll>a;
int main(){
  cin>>n;a.resize(n);for(ll&x:a)cin>>x;
  ll cnt[2]={0,0};
  for(ll x:a)cnt[x&1]++;
  if(cnt[0]!=cnt[1]){
    cout<<(cnt[0]+cnt[1]-2)%mod*cnt[0]%mod*cnt[1]%mod;
  }else{
    ll sum=0,cnt=0;
    for(ll x:a){
      sum+=x%2==1?-1:1;
      if(sum==0){
        cnt++;
      }
    }
    cnt=(cnt*(cnt-1))/2;
    cout<<(cnt+mod)%mod;
  }
}
