#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    int price[n], quality[n];
    for (int i = 0; i < n; i++) {
        cin >> price[i] >> quality[i];
    }
    sort(price, price + n);
    sort(quality, quality + n);
    bool found = false;
    for (int i = 0; i < n - 1; i++) {
        if (price[i] < price[i + 1] && quality[i] > quality[i + 1]) {
            found = true;
            break;
        }
    }
    cout << (found ? "Happy Alex" : "Poor Alex") << endl;
    return 0;
}
