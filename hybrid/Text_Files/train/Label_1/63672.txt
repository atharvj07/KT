from sys import stdin
input = stdin.readline

N, M = map(int, input().split())
a = list(map(int, input().split()))
st = [[j for j in map(int, input().split())] for i in range(M)]
sum_st = [0] * N
sum = 0
for l, r in st:
    l -= 1; r -= 1
    sum += a[l] - a[r]
    sum_st[l] += 1
    sum_st[r + 1] -= 1
for i in range(1, N):
    sum_st[i] += sum_st[i - 1]

a.sort(); sum_st.sort()
sum_max = 0
ans = []
for i in range(M):
    if sum_st[i] <= 0: break
    sum_max += sum_st[i] * (a[i] - a[i + M - 1])
    ans.append(i + 1)

print(sum_max)
print(len(ans))
print(*ans)
