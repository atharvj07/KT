#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<int> a(n);
        for (auto& x : a) cin >> x;
        bool flag = false;
        for (int i = 1; i + 2 < n; ++i) {
            int side1 = a[i], side2 = a[i + 1], side3 = a[i + 2];
            if (side1 + side2 <= side3 || side1 + side3 <= side2 || side2 + side3 <= side1) {
                cout << i << ' ' << i + 1 << ' ' << i + 2 << '\n';
                flag = true;
                break;
            }
        }
        if (!flag) cout << "-1\n";
    }
}
