#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    vector<bool> used(n, false);
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (used[i]) {
            continue;
        }
        int sz = 1;
        used[i] = true;
        for (int j = i + 1; j < n; j++) {
            if (used[j]) {
                continue;
            }
            if (a[i] % a[j] == 0 || a[j] % a[i] == 0) {
                sz++;
                used[j] = true;
            }
        }
        ans = max(ans, sz);
    }

    cout << ans << endl;
    return 0;
}
