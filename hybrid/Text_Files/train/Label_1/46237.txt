import java.util.Scanner;

public class MoneyWithdrawal {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();

        // Initialize the number of operations required
        int operations = 0;

        // Withdraw amounts in descending order of powers of 6 and 9
        while (N > 0) {
            if (N >= Math.pow(6, operations)) {
                N -= Math.pow(6, operations);
            } else if (N >= Math.pow(9, operations)) {
                N -= Math.pow(9, operations);
            } else {
                operations++;
            }
        }

        // Print the number of operations required
        System.out.println(operations);
    }
}
