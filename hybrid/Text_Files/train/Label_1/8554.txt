#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  vector<int> xor_values;
  int last = -1;
  for (int i = 0; i < n; i++) {
    if (a[i] != last) {
      xor_values.push_back(a[i]);
      last = a[i];
    }
  }
  int k = xor_values.size();
  cout << k << endl;
  for (int i = 0; i < k; i++) {
    cout << xor_values[i] << " ";
  }
  cout << endl;
  return 0;
}
