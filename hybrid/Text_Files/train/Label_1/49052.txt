import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
            }
            int[] cnt = new int[n + 1];
            for (int i = 0; i < n; i++) {
                cnt[a[i]]++;
            }
            long ans = 0;
            for (int i = 1; i <= n; i++) {
                ans += (long)cnt[i] * (cnt[i] - 1) / 2 * (cnt[i] - 2) / 3 * (cnt[i] - 3);
            }
            System.out.println(ans);
        }
    }
}
