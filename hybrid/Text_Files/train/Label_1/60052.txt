n, m = list(map(int, input().split()))
x = [2] * (n + 1)
for i in range(1, n + 1):
  x += [i, i + n]
s = []
for i in range(m):
  a = x[1]
  s.append(a)
  x.pop(1)
  x.append(a)
  for j in range(1, n + 1):
    for t in range(2):
      if x[1] == i + 1:
        s.append(x[1])
        x.pop(1)
        x.append(x[1])
print(*s)
