import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(bf.readLine());
        int[] arr = Arrays.stream(bf.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] sorted = Arrays.copyOf(arr, n);
        Arrays.sort(sorted);
        int[] lPos = new int[n];
        for (int i = 0; i < n; i++) lPos[sorted[i] - 1] = i;
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            int l = 0, r = n - 1, mid;
            while (l < r) {
                mid = (l + r) / 2;
                if (lPos[mid] < i) l = mid + 1;
                else r = mid;
            }
            int maxDist = Math.max(i - lPos[l], lPos[r] - i);
            if (l > 0 && lPos[l - 1] < i) maxDist = Math.max(maxDist, i - lPos[l - 1]);
            if (r < n - 1 && lPos[r + 1] > i) maxDist = Math.max(maxDist, lPos[r + 1] - i);
            ans[i] = maxDist;
        }
        PrintWriter pw = new PrintWriter(System.out);
        for (int i = 0; i < n; i++) pw.print(ans[i] + " ");
        pw.flush();
    }
}
