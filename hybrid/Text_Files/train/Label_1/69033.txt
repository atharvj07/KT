import math

n, k = map(int, input().split())
segments = []
for _ in range(n):
    l, r = map(int, input().split())
    segments.append((l, r))

# Shift each segment's right endpoint to the nearest multiple of k
for i in range(n):
    r_new = math.ceil(segments[i][1] / k) * k
    segments[i] = (segments[i][0], r_new)

# Calculate the minimum number of moves needed to shift each segment's left endpoint to the nearest multiple of k
ans = 0
for l, r in segments:
    ans += math.ceil(l / k) * k - l

print(ans)
