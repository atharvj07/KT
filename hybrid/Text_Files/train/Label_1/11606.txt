#include<bits/stdc++.h>
using namespace std;
int main() {
    int t, n, a[100001], b[100001];
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 1; i <= n; i++) {
            cin >> a[i] >> b[i];
        }
        vector<pair<int, int>> v1(n + 1), v2(n + 1);
        for (int i = 1; i <= n; i++) {
            v1[i].first = a[i];
            v1[i].second = i;
            v2[i].first = b[i];
            v2[i].second = i;
        }
        sort(v1.begin(), v1.end());
        sort(v2.begin(), v2.end());
        int win1 = v1[n].second, win2 = v2[n].second;
        for (int i = 1; i <= n; i++) {
            if (v1[i].second == win1 || v2[i].second == win2) cout << 1;
            else cout << 0;
        }
        cout << endl;
    }
    return 0;
}
