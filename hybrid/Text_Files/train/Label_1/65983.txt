import sys
input = sys.stdin.readline

n,k = map(int, input().split())
s = list(input().strip())
stack = []
open_doors = 0
for c in s:
    if stack and c == stack[-1]:
        stack.pop()
        open_doors -= 1
    else:
        stack.append(c)
        open_doors += 1
        if open_doors > k:
            print('YES')
            exit()

print('NO')
