#include<bits/stdc++.h>
using namespace std;
#define ll long long int

const ll MAXN = 100005;
const ll oo = 1<<30;

struct node{
    ll x, y;
}a[MAXN], b[MAXN];

ll n, i, j, sum[MAXN];

inline ll is_intersect(const node& a, const node& b){
    ll s = a.x * b.y - a.y * b.x;
    if(s == 0)    return 0;
    if(s > 0)    return 1;
    return -1;
}

int main(){
    cin >> n;
    for(i = 1; i <= n; i++){
        cin >> a[i].x;
        a[i].y = (i==1? oo: a[i-1].y);
    }
    sort(a+1, a+n+1, [](const node& x, const node& y){ return x.x < y.x; });
    for(i = 1; i <= n; i++){
        b[i] = a[i];
        b[i].y += n;
    }
    ll ans = 0;
    for(i = 1; i <= n; i++){
        for(j = i+1; j <= n; j++){
            ans += is_intersect(a[i], b[j]) * is_intersect(a[j], b[i]);
            if(ans){
                puts("yes");
                return 0;
            }
        }
    }
    puts("no");
    return 0;
}
