import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            if (n == 0 && k == 0) break;
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) {
                arr[i] = sc.nextInt();
            }
            int[] dp = new int[n];
            int max = 0;
            for (int i = 0; i < n; i++) {
                if (i < k) {
                    dp[i] = arr[i];
                    max = Math.max(max, dp[i]);
                } else {
                    dp[i] = Math.max(dp[i - k] * arr[i], arr[i]);
                    max = Math.max(max, dp[i]);
                }
            }
            for (int i = k - 1; i < n; i++) {
                int prod = 1;
                for (int j = i; j >= i - k + 1; j--) {
                    prod *= arr[j];
                }
                max = Math.max(max, prod);
            }
            if (max < 0) System.out.println("NO GAME");
            else System.out.println(max);
        }
    }
}
