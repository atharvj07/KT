for _ in range(int(input())):
    n, k = map(int, input().split())
    even = (n&1) == 0
    if k == 1:
        print('YES')
    elif k == 2:
        if even:
            print('YES')
            print(n//2, n//2)
        else:
            print('NO')
    elif k == 3:
        if not even or n % 3 != 0:
            print('NO')
        else:
            print('YES')
            print(n//3, n//3, n - n//3 * 2)
    elif even and n % 4 == 0:
        print('YES')
        n -= 4
        k -= 2
        if k & 1:
            print(4, end=' ')
            k -= 1
        while k > 0:
            print(2, end=' ')
            k -= 1
        print(n)
    else:
        print('NO')
