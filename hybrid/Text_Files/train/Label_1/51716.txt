import java.util.*;

public class Main {
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int I = in.nextInt();
        int[] arr = new int[n];
        for(int i = 0; i < n; i++){
            arr[i] = in.nextInt();
        }
        int l = 0;
        int r = 0;
        int k = 0;
        for(int i = 0; i < n; i++){
            if(k == 0){
                l = arr[i];
                r = arr[i];
                k = 1;
            }
            else{
                if(arr[i] < l){
                    l = arr[i];
                }
                else if(arr[i] > r){
                    r = arr[i];
                }
            }
        }
        int count = 0;
        for(int i = 0; i < n; i++){
            if(arr[i] != l && arr[i] != r){
                count++;
            }
        }
        long disk = I * 8;
        long log2 = (long)Math.ceil(Math.log(n) / Math.log(2));
        if(disk >= log2 * n){
            count = 0;
        }
        System.out.println(count);
    }
}
