import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();

        int cnt = 0, p = 0, q = 0;
        for (int i = 0; i < n; i++) {
            while (q < n && p + a[q] <= i) {
                q++;
                p += a[q - 1];
            }
            if (k < q && q * 100 / n == (i + 1) * 100 / n) {
                cnt++;
            }
        }
        System.out.println(cnt);
    }
}
