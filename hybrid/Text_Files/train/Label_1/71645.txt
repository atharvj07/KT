import sys

def solve():
    coords = []
    for _ in range(8):
        coords.append(sorted(map(int, sys.stdin.readline().split())))
    n = coords[0][2]
    coords.sort(key=lambda x: x[0])
    if coords[0][1] == n:
        return (
            0, n, n - coords[0][0],
            n, 0, n - coords[0][0],
            n, n, n - coords[0][0],
            0, 0, n - coords[0][0],
            0, n, 0,
            n, n, 0,
            n, 0, 0,
            0, 0, 0,
        )
    elif coords[0][1] == 0:
        return (
            0, 0, n - coords[0][0],
            n, 0, n - coords[0][0],
            n, n, n - coords[0][0],
            0, n, n - coords[0][0],
            0, 0, 0,
            n, n, 0,
            n, n, n - coords[0][0],
            0, n, n - coords[0][0],
        )
    return 'NO'

print(solve())
