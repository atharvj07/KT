import sys

class DisjointSet:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root != y_root:
            if self.rank[x_root] < self.rank[y_root]:
                self.parent[x_root] = y_root
            else:
                self.parent[y_root] = x_root
                if self.rank[x_root] == self.rank[y_root]:
                    self.rank[x_root] += 1

n, q = map(int, sys.stdin.readline().split())
grid = [[False] * (n + 2) for _ in range(3)]
dset = DisjointSet(n + 2)

for _ in range(q):
    x, y = map(int, sys.stdin.readline().split())
    if x == 2:
        y -= 1
    grid[x][y] = not grid[x][y]
    if grid[x][y]:
        dset.union(y, y + 1)
    else:
        dset.parent[y] = y
        dset.parent[y + 1] = y + 1
    if dset.find(1) == dset.find(n):
        print("Yes")
    else:
        print("No")
