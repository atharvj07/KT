#include <bits/stdc++.h>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    string s;
    int m;
    cin >> s >> m;
    vector<int> b(m);
    for (int i = 0; i < m; i++) cin >> b[i];
    string t;
    vector<char> v;
    for (char c : s) v.push_back(c);
    for (int i = 0; i < m; i++) {
      int j = 0;
      while (j < (int)v.size() && v[j] <= v[b[i]]) j++;
      if (j < (int)v.size()) t += v[j], v.erase(v.begin() + j);
    }
    cout << t << endl;
  }
  return 0;
}
