import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][][] cube = new int[n][n][n];
        int num = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    int l = (j + k) % 3;
                    if (l == 0) {
                        cube[i][j][k] = num++;
                    } else if (l == 1) {
                        cube[k][i][n - 1 - j] = num++;
                    } else {
                        cube[n - 1 - j][k][i] = num++;
                    }
                }
            }
        }
        for (int[][] layer : cube) {
            for (int[] row : layer) {
                for (int el : row) {
                    System.out.print(el + " ");
                }
                System.out.println();
            }
            System.out.println();
        }
    }
}
