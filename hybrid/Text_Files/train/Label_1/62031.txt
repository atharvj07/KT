import sys
from collections import defaultdict

n, m, k = map(int, sys.stdin.readline().split())
c = list(map(int, sys.stdin.readline().split()))
p = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]
f = [[[1e9 for _ in range(m + 1)] for _ in range(m + 1)] for _ in range(n + 1)]
f[0][0][0] = 0
for i in range(n):
    for j in range(m + 1):
        for x in range(m + 1):
            if x == 0:
                f[i + 1][x][j] = min(f[i + 1][x][j], f[i][x][j])
            else:
                f[i + 1][x][j] = min(f[i + 1][x][j], f[i][x][j] + p[i][x - 1])
            if c[i] == 0:
                if x == j:
                    f[i + 1][x][j + 1] = min(f[i + 1][x][j + 1], f[i][x][j])
                else:
                    f[i + 1][x][j + 1] = min(f[i + 1][x][j + 1], f[i][x][j] + p[i][x - 1])
            else:
                if x == c[i]:
                    f[i + 1][x][j] = min(f[i + 1][x][j], f[i][x][j])
if f[n][m][k] == 1e9:
    print(-1)
else:
    print(f[n][m][k])
