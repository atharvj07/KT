#include <bits/stdc++.h>
using namespace std;
#define ll long long

const ll mod = 1e9 + 7;
const ll size = 2e5 + 5;

vector<ll> pref(size), suf(size);
int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    ll ans = 0, cnt[26] = {};
    for (ll i = 0; i < n; i++) {
      cnt[s[i] - 'a']++;
      if (k % 2 == 0 || cnt[s[i] - 'a'] % 2 == 0) {
        ans++;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
