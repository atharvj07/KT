from collections import Counter

def find_positions(n, m, p, a, b):
    b_counter = Counter(b)
    valid_positions = []

    for start in range(p):
        window = Counter()
        matches = 0
        for i in range(start, n, p):
            val = a[i]
            window[val] += 1
            if window[val] == b_counter[val]:
                matches += 1
            elif window[val] == b_counter[val] + 1:
                matches -= 1
            
            if i >= m*p:
                left_val = a[i - m*p]
                window[left_val] -= 1
                if window[left_val] == b_counter[left_val]:
                    matches += 1
                elif window[left_val] == b_counter[left_val] - 1:
                    matches -= 1
            
            if matches == len(b_counter):
                valid_positions.append(i - (m-1)*p + 1)
    
    return valid_positions

n, m, p = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

valid_positions = find_positions(n, m, p, a, b)
print(len(valid_positions))
print(*sorted(valid_positions))
