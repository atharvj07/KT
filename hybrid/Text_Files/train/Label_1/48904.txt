import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int q = sc.nextInt();
        int[] d = new int[k];
        for (int i = 0; i < k; i++) {
            d[i] = sc.nextInt();
        }
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            int x = sc.nextInt();
            int m = sc.nextInt();
            long[] a = new long[n];
            a[0] = x;
            for (int j = 1; j < n; j++) {
                a[j] = a[j - 1] + d[(j - 1) % k];
            }
            int count = 0;
            for (int j = 0; j < n - 1; j++) {
                if ((a[j] % m) < (a[j + 1] % m)) {
                    count++;
                }
            }
            System.out.println(count);
        }
    }
}
