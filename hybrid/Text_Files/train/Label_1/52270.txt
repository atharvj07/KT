import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int[][] a = new int[n][n];
		for (int i = 0; i < n; i++) {
			String s = sc.next();
			for (int j = 0; j < n; j++) {
				a[i][j] = s.charAt(j) - '0';
			}
		}
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				if (a[i][j] == 1 && a[j][i] == 1) {
					for (int k = 0; k < n; k++) {
						if (a[j][k] == 1 && a[k][i] == 1) {
							System.out.println((i + 1) + " " + (j + 1) + " " + (k + 1));
							return;
						}
					}
				}
			}
		}
		System.out.println("-1");
		sc.close();
	}
}
