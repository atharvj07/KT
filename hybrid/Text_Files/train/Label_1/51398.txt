import java.util.*;

public class Main {
    static int W, H, N;
    static int cable[][][];
    static int staff[][];
    static int pattern[][];

    static int[] dx = {0, 0, -1, 1};
    static int[] dy = {-1, 1, 0, 0};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        W = sc.nextInt();
        H = sc.nextInt();
        N = sc.nextInt();

        int x1 = sc.nextInt(), y1 = sc.nextInt();
        int x2 = sc.nextInt(), y2 = sc.nextInt();

        int xx1 = (x1 == W ? W - 1 : x1);
        int xx2 = (x2 == W ? W - 1 : x2);
        int yy1 = (y1 == H ? H - 1 : y1);
        int yy2 = (y2 == H ? H - 1 : y2);

        cable = new int[H][W][4];
        for (int i = yy1; i <= yy2; i++) {
            cable[i][xx1][3] = cable[i][xx1][2] = 1;
            cable[i][xx2][3] = cable[i][xx2][2] = 1;
        }
        for (int i = xx1; i <= xx2; i++) {
            cable[yy1][i][1] = cable[yy1][i][0] = 1;
            cable[yy2][i][1] = cable[yy2][i][0] = 1;
        }

        staff = new int[N][2];
        pattern = new int[N][];
        for (int i = 0; i < N; i++) {
            staff[i][0] = sc.nextInt();
            staff[i][1] = sc.nextInt();
            pattern[i] = new int[sc.nextInt()];
            String s = sc.next();
            for (int j = 0; j < s.length(); j++) {
                if (s.charAt(j) == 'U') pattern[i][j] = 0;
                else if (s.charAt(j) == 'D') pattern[i][j] = 1;
                else if (s.charAt(j) == 'L') pattern[i][j] = 2;
                else pattern[i][j] = 3;
            }
        }

        int ans = 1000000000;
        for (int yy = 0; yy < H; yy++) {
            for (int xx = 0; xx < W; xx++) {
                int tmp = 0;
                for (int i = 0; i < N; i++) {
                    int x = staff[i][0], y = staff[i][1], dir = 0, loop = pattern[i].length;
                    for (int j = 0; j < 10000000; j++) {
                        tmp += cable[y][x][dir];
                        x += dx[dir];
                        y += dy[dir];
                        if (x == -1 || x == W || y == -1 || y == H) {
                            dir = (dir + 2) % 4;
                            x += dx[dir];
                            y += dy[dir];
                        }
                        if (x == xx && y == yy) {
                            dir = (dir + 3) % 4;
                            x += dx[dir];
                            y += dy[dir];
                        }
                        if (j % loop == loop - 1) dir = pattern[i][j % loop];
                    }
                }
                ans = Math.min(ans, tmp);
            }
        }

        System.out.println(ans);
    }
}
