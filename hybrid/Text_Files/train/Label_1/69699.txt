def count_distinct_numbers(b):
    distinct_numbers = 0
    a = 1
    while a * a <= b:
        if b % a == 0:
            distinct_numbers += 1
            if a * a != b:
                distinct_numbers += 1
        a += 1
    return distinct_numbers

b = int(input().strip())
print(count_distinct_numbers(b))
