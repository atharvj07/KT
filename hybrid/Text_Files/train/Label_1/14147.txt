#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<pair<int, int>> songs(n);
    for (int i = 0; i < n; i++) {
        cin >> songs[i].first >> songs[i].second;
    }
    sort(songs.begin(), songs.end());
    long long sum = 0, cnt = 0;
    for (int i = 0; i < n; i++) {
        if (sum + songs[i].second <= m) {
            sum += songs[i].second;
        } else {
            cnt++;
            sum = songs[i].second;
        }
    }
    if (sum > m) cout << -1 << endl;
    else cout << cnt << endl;
}
