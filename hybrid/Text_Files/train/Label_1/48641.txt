import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner scn = new Scanner(System.in);
        int n = scn.nextInt();
        long[] h = new long[n+1];
        for (int i = 1; i <= n; i++) {
            h[i] = scn.nextLong();
        }
        long ans = 0;
        for (int i = 2; i <= n; i++) {
            if (h[i-1] > h[i]) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
