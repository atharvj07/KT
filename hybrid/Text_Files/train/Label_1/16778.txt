#include <iostream>
#include <vector>
using namespace std;

vector<int> get_cyclic_shifts(int num) {
  vector<int> shifts;
  int lz = 0;  // leading zeros
  while (num % 10 == 0) {
    num /= 10;
    lz++;
  }
  int cur = num;
  for (int i = 0; i < lz; i++) {
    shifts.emplace_back(cur);
    cur *= 10;
  }
  for (int i = 0; i < lz; i++) {
    cur /= 10;
    shifts.emplace_back(cur);
  }
  return shifts;
}

int get_min_remainder(int num, int m) {
  vector<int> shifts = get_cyclic_shifts(num);
  int res = num % m;
  for (auto s : shifts) res = min(res, s % m);
  return res;
}

int main() {
  int num;
  cin >> num;
  int m;
  cin >> m;
  cout << get_min_remainder(num, m) << endl;
  return 0;
}
