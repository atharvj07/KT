import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int s = 0, d = 0;
        boolean sTurn = true;
        while (n > 0) {
            if (sTurn) {
                if (a[0] > a[n - 1]) {
                    s += a[0];
                    n--;
                } else {
                    s += a[n - 1];
                    n--;
                }
            } else {
                if (a[0] > a[n - 1]) {
                    d += a[0];
                    n--;
                } else {
                    d += a[n - 1];
                    n--;
                }
            }
            sTurn = !sTurn;
        }
        System.out.println(s + " " + d);
    }
}
