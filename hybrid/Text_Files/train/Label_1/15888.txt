#include <cstdio>
#include <vector>
using namespace std;
#define vi vector<int>

int main(){
    int n, a, b, cnt1, cnt2;
    scanf("%d %d %d", &n, &a, &b);
    vi p(n);
    for(int i = 0; i < n; i++) scanf("%d", &p[i]);
    cnt1 = 0;
    cnt2 = 0;
    for(int i = 0; i < n; i++){
        if(p[i] == 1){
            if(cnt1) cnt1--;
            else if(cnt2){
                if(cnt2 > 1) cnt2--;
                else if(a) a--;
                else cnt2--;
            }
            else if(a) a--;
            else cnt2++;    
        }
        else{
            if(cnt2) cnt2--;
            else if(b) b--;
            else cnt2 += 2;
        }
    }
    cnt1 += cnt2;
    cnt2 = cnt1 + cnt2;
    printf("%d\n", cnt2);
    return 0;
}
