import sys

n = int(input())
a = list(map(int, input().split()))
e = [[] for i in range(n)]
for i in range(n - 1):
    x, y, z = map(int, input().split())
    x -= 1; y -= 1
    e[x].append((y, z))
    e[y].append((x, z))

d = [-1] * n
f = [0] * n

def dfs1(v, p):
    d[v] = p
    for u, w in e[v]:
        if d[u] == -1:
            dfs1(u, v)

def dfs2(v, p):
    mx = a[v]
    for u, w in e[v]:
        if u == p:
            continue
        dfs2(u, v)
        f[v] = max(f[v], f[u] + a[v] - w)
        mx = max(mx, f[u] + a[v])
    f[v] = mx

dfs1(0, -1)
dfs2(0, -1)

print(max(f))
