#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MOD = 1e9 + 7;
ll a[100005], e, ans;
map<ll, ll> cnt;
int main() {
    ll n;
    cin >> n;
    for (ll i = 1; i <= n; i++) {
        cin >> a[i];
        e += (a[i] == 1);
        cnt[a[i]]++;
    }
    ans = 1;
    for (int i = 2; i <= 70; i++) {
        for (ll x = i, cnt2 = 1; cnt[x]; x *= i, cnt2++) {
            ans += (1LL * cnt[x] * (cnt[x] - 1) / 2 % MOD) * cnt2 % MOD;
            ans %= MOD;
        }
    }
    ans = (ans + e * (e - 1) % MOD * e % MOD / 2) % MOD;
    cout << (ans + MOD) % MOD << endl;
}
