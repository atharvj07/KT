#include <iostream>
#include <cstring>

int main() {
  int n;
  std::cin >> n;
  char s[10];
  std::cin >> s;
  std::string pokemons[] = {"vaporeon", "jolteon", "flareon", "espeon", "umbreon", "leafeon", "glaceon", "sylveon"};
  for (std::string& pokemon : pokemons) {
    int mismatch = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] != '.' && s[i] != pokemon[i]) {
        mismatch = 1;
        break;
      }
    }
    if (!mismatch) {
      std::cout << pokemon << std::endl;
      return 0;
    }
  }
  return 0;
}
