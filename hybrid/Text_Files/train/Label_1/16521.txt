#include <bits/stdc++.h>
using namespace std;
struct bug { int a, c; };
bool cmp(bug a, bug b) { return a.c > b.c; }
int main() {
  int n, m, s;
  cin >> n >> m >> s;
  int a[m], b[n], c[n];
  vector<bug> av;
  for (int i = 0; i < m; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  for (int i = 0; i < n; i++) { cin >> c[i]; av.push_back({b[i], c[i]}); }
  sort(av.begin(), av.end(), cmp);
  vector<int> res(m, -1);
  int used = 0;
  for (int i = 0; i < m && used < s; i++) {
    for (int j = 0; j < n; j++) {
      if (av[j].a >= a[i] && av[j].c + used <= s && res[i] == -1) {
        res[i] = j;
        av[j].c = -1;
        used += av[j].c;
        break;
      }
    }
  }
  if (used < s) cout << "NO" << endl;
  else {
    cout << "YES" << endl;
    for (int i = 0; i < m; i++) cout << res[i] + 1 << ' ';
    cout << endl;
  }
  return 0;
}
