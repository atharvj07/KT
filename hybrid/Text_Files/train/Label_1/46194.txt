import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = reader.readLine().split(" ");
        int n = Integer.parseInt(parts[0]), m = Integer.parseInt(parts[1]);
        Tree[] trees = new Tree[n];
        Mushroom[] mushrooms = new Mushroom[m];
        for (int i = 0; i < n; i++) {
            parts = reader.readLine().split(" ");
            trees[i] = new Tree(Integer.parseInt(parts[0]), Integer.parseInt(parts[1]), Integer.parseInt(parts[2]), Integer.parseInt(parts[3]));
        }
        for (int i = 0; i < m; i++) {
            parts = reader.readLine().split(" ");
            mushrooms[i] = new Mushroom(Integer.parseInt(parts[0]), Integer.parseInt(parts[1]));
        }
        Arrays.sort(mushrooms);
        double sum = 0;
        for (Mushroom mushroom : mushrooms) {
            double probability = 1.0;
            for (Tree tree : trees) {
                probability *= tree.getProbability(mushroom.x);
            }
            sum += probability * mushroom.z;
        }
        System.out.printf("%.9f", sum);
    }

    private static class Tree {
        int x, h, l, r;

        public Tree(int x, int h, int l, int r) {
            this.x = x;
            this.h = h;
            this.l = l;
            this.r = r;
        }

        public double getProbability(int x) {
            if (x <= this.x - h) {
                return 1.0 * (100 - l) / 100;
            } else if (x >= this.x + h) {
                return 1.0 * (100 - r) / 100;
            } else {
                return 0.0;
            }
        }
    }

    private static class Mushroom implements Comparable<Mushroom> {
        int x, z;

        public Mushroom(int x, int z) {
            this.x = x;
            this.z = z;
        }

        @Override
        public int compareTo(Mushroom other) {
            return Integer.compare(x, other.x);
        }
    }
}
