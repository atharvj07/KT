#include <iostream>
#include <cmath>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int x = n / 7;
    if (n % 7 == 0) {
      cout << x << " " << 0 << " " << x << endl;
      continue;
    }
    for (int i = x; i >= 0; i--) {
      int j = (n - 7 * i) % 5;
      int k = (n - 7 * i - 5 * j) / 3;
      if (7 * i + 5 * j + 3 * k == n) {
        cout << k << " " << j << " " << i << endl;
        break;
      }
    }
    if (7 * x + 5 * 0 + 3 * 0 != n) { cout << -1 << endl; }
  }
  return 0;
}
