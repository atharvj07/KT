import sys

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    if n % 2 == 0 or sum(a) % 2 == 0:
        print(-1)
        continue
    b = sorted(a)
    m = 0
    for i in range(n):
        if a[i] != b[i]:
            for j in range(n):
                if a[j] == b[i]:
                    if (j - i) % 2 == 0:
                        print(-1)
                        sys.exit()
                    m += 1
                    a[j:i:-1] = a[j:i:-1]
                    break
    print(m)
    for i in range(n):
        if a[i] != b[i]:
            for j in range(n):
                if a[j] == b[i]:
                    if (j - i) % 2 != 0:
                        print(j - i)
                        a[j:i:-1] = a[j:i:-1]
