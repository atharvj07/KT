n, m, S_x, S_y = map(int, input().split())

# Visit all cells in the starting column first.
for i in range(1, n+1):
    print(i, S_y)

# For the remaining columns, visit them in a zig-zag manner.
for col in range(1, m+1):
    # Skip the starting column since we've already visited it.
    if col == S_y:
        continue
    if col % 2 == 1:
        # For odd columns (relative to the starting point), go bottom to top.
        for row in range(n, 0, -1):
            print(row, col)
    else:
        # For even columns, go top to bottom.
        for row in range(1, n+1):
            print(row, col)
