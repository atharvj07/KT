#include <iostream>

using namespace std;

long long S(long long n) {
  long long ans = 0;
  while (n) {
    ans++;
    n /= 10;
  }
  return ans;
}

int main() {
  long long w, m, k;
  cin >> w >> m >> k;
  long long ans = 0;
  while (w >= (ans + 1) * k * S(m + ans)) {
    ans++;
  }
  cout << ans << endl;
  return 0;
}
