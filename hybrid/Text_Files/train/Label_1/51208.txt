import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    static BigInteger gcd(BigInteger a, BigInteger b) {
        if (a.compareTo(BigInteger.ZERO) == 0) return b;
        return gcd(b.mod(a), a);
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        BigInteger n = BigInteger.valueOf(input.nextInt());
        BigInteger x = BigInteger.valueOf(input.nextInt());
        BigInteger m = BigInteger.valueOf(input.nextInt());
        BigInteger y = BigInteger.valueOf(input.nextInt());
        x = x.subtract(y).abs();
        n = n.min(m);
        BigInteger g = gcd(x, n);
        BigInteger ans = (x.divide(g).multiply(n)).divide(g);
        System.out.println(ans);
    }
}
