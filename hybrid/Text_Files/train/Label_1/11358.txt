#include <iostream>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  if (m % n != 0) {
    cout << -1 << endl;
    return 0;
  }
  int ans = 0;
  while (m > n) {
    if (m % 3 == 0) {
      m /= 3;
      ans++;
    } else if (m % 2 == 0) {
      m /= 2;
      ans++;
    } else {
      cout << -1 << endl;
      return 0;
    }
  }
  cout << ans << endl;
  return 0;
}
