#include <bits/stdc++.h>
#define ll long long
using namespace std;

const ll MAXN = 1e5+10;
ll n, a[MAXN], pos[MAXN], cnt[MAXN], ans[MAXN];

int main() {
    cin >> n;
    ll sz = 0;
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
        if (!pos[a[i]]) pos[a[i]] = ++sz;
        cnt[pos[a[i]]]++;
    }
    for (ll i = 1; i <= sz; i++) {
        if (cnt[i] == 1) {
            ans[pos[i]] = 0;
            continue;
        }
        ll last = i -1;
        ll j = i + 1;
        while (j <= sz) {
            if (cnt[j] == 0) {
                j++;
                continue;
            }
            ans[pos[i]] = pos[j] - last;
            last = pos[j];
            j++;
        }
    }
    ll idx = 0;
    for (ll i = 0; i < n; i++) {
        if (ans[pos[a[i]]] != 0) {
            idx++;
            cout << a[i] << " " << ans[pos[a[i]]] << endl;
        }
    }
    cout << idx << endl;
    return 0;
}
