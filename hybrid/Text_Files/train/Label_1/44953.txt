import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String A = sc.next();
        String B = sc.next();
        boolean ans = true;

        for (int i = 0; i < B.length(); i++) {
            int index = A.indexOf(B.charAt(i));

            if (index > -1) {
                A = A.substring(index + 1);
            } else {
                ans = false;
                break;
            }
        }

        System.out.println(ans ? "Yes" : "No");
    }
}
