#include <iostream>
using namespace std;
typedef long long ll;

int main() {
  int n; cin >> n;
  ll sum = 0, lst = 0, pre = -2;
  while (n--) {
    ll h; cin >> h;
    sum += h;
    if (pre == h) sum -= 2;
    else if (h == lst) sum += 2;
    if (h != lst) lst = pre, pre = h;
  }
  cout << (sum % 3 ? "NO" : "YES") << endl;
  return 0;
}
