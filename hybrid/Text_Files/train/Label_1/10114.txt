#include <bits/stdc++.h>
using namespace std;
#define int long long
typedef long long ll;
const int MOD = 1e9 + 7;
ll powmod(ll a, ll b) {
    ll res = 1;
    while (b) {
        if (b & 1) res = (res * a) % MOD;
        a = (a * a) % MOD;
        b >>= 1;
    }
    return res;
}
bool is_special(int n, int k) {
    int d = 0;
    while (n) {
        if (n & 1) d++;
        n >>= 1;
    }
    if (k == 0) {
        if (d == 1) return true;
        return false;
    } else {
        if (d == 1 || d == 2) return true;
        return false;
    }
}
