import sys
input = sys.stdin.readline

n = int(input())
g = [[] for i in range(n + 1)]
for i in range(n - 1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

vis = [False] * (n + 1)
dis = [0] * (n + 1)
q = [(1, 0)]
while q:
    now, d = q.pop(0)
    if vis[now]:
        continue
    dis[now] = d
    vis[now] = True
    for i in g[now]:
        q.append((i, d + 1))

print(sum(dis) * 2 - n)
