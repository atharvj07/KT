#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int x, n, m;
        cin >> x >> n >> m;
        if (n > 0 && x % 25 <= 10 * n)
            cout << "YES" << endl;
        else if (m > 0 && x % 10 <= 10 * m)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
}
