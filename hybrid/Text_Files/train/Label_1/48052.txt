import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        int minDiff = Integer.MAX_VALUE;
        int nearest = 0;
        for (int i = 0; i < n; i++) {
            int diff = Math.abs(x - arr[i]);
            if (diff < minDiff) {
                minDiff = diff;
                nearest = arr[i];
            }
        }
        if (x < arr[0]) {
            nearest = arr[0] - minDiff;
        } else if (x > arr[n - 1]) {
            nearest = arr[n - 1] + minDiff;
        }
        System.out.println(nearest);
    }
}
