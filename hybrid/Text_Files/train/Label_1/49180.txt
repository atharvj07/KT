import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] s = new String[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.next();
        }
        int m = sc.nextInt();
        String[] t = new String[m];
        for (int i = 0; i < m; i++) {
            t[i] = sc.next();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (s[i].equals(t[j])) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
