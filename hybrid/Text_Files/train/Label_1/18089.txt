#include <bits/stdc++.h>
#define ll long long

const ll mod = 1e9 + 7;

int main() {
    int t; std::cin >> t;
    while (t--) {
        int n, m, k; std::cin >> n >> m >> k;
        std::vector<int> a(n);
        for (int i = 0; i < n; i++) { std::cin >> a[i]; }
        std::vector<int> freq(n + 1);
        for (int x : a) { freq[x]++; }
        ll ans = 0;
        for (int i = 1; i <= n; i++) {
            int left = std::max(1, i - k), right = std::min(n, i + k);
            if (left <= right) {
                ll cnt1 = 0, cnt2 = 0;
                for (int j = left; j <= i; j++) { cnt1 += freq[j]; }
                for (int j = i; j <= right; j++) { cnt2 += freq[j]; }
                ans = (ans + (cnt1 * cnt2) % mod) % mod;
            }
        }
        std::cout << ans << '\n';
    }
    return 0;
}
