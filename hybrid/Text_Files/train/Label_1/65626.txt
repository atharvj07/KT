n, c = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

dp = [[0, 0] for _ in range(n)]
dp[0][1] = c

for i in range(1, n):
    dp[i][0] = min(dp[i-1][0] + a[i-1], dp[i-1][1] + a[i-1])
    dp[i][1] = min(dp[i-1][0] + b[i-1] + c, dp[i-1][1] + b[i-1])

for i in range(n):
    print(min(dp[i]), end=' ')
