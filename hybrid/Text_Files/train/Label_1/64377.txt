n, m = map(int, input().split())
floors = [input() for _ in range(n)]

time = 0
for i in range(n-1, -1, -1):
    left_most = floors[i].find('1')
    right_most = floors[i].rfind('1')
    if right_most == -1:
        continue
    if i == n-1:
        time += right_most
    else:
        time += min(2*left_most, 2*(m+1-right_most)) + m + 1
print(time)
