#include <iostream>
#define ll long long

int main() {
    ll n, m, q;
    std::cin >> n >> m >> q;
    while (q--) {
        ll s_x, s_y, e_x, e_y;
        std::cin >> s_x >> s_y >> e_x >> e_y;
        if (s_x == e_x) { std::cout << "YES\n"; }
        else if (s_x == 1 && e_x == 2) { std::cout << "YES\n"; }
        else if (s_x == 2 && e_x == 1 && (s_y == e_y || n == 1 || e_y == 1)) { std::cout << "YES\n"; }
        else { std::cout << "NO\n"; }
    }
    return 0;
}
