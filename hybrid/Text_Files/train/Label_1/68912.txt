import sys

max_rows = int(input())
ad = input()

hyphenated = ad.split('-')
words = hyphenated[0].split()
for part in hyphenated[1:]:
    words.append('-')
    words.append(part)

words.append(' ')  # sentinel

min_width = 0
num_rows = 1
current_row_width = 0
for word in words:
    if current_row_width + len(word) <= max_rows:
        current_row_width += len(word)
    else:
        num_rows += 1
        current_row_width = len(word)

    min_width = max(min_width, current_row_width)

print(min_width)
