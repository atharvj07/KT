import sys
input=sys.stdin.readline
n,m,k=map(int,input().split())
s=sorted([0]+list(map(int,input().split()))+[n])
a=list(map(int,input().split()))
ans=-1; cur=[0]*k
for r in s[1:]:
    l=s[s.index(r)-1]
    if any(map(lambda x:l<=x<=r,cur)):continue
    lst,idx=min([(a[x],x)for x in range(k)if x+1<=r-l], key=lambda x:x[0])
    if lst*(r-l)<ans or ans==-1:ans=lst*(r-l); cur[idx]=r
print(ans)
