#include <cstdio>
const int N = 2000;
int phi[N], p[N];
bool notprime[N];
void sieve() {
  notprime[0] = notprime[1] = true;
  for (long long i = 2; i < N; ++i) {
    if (!notprime[i]) {
      p[++p[0]] = i;
      phi[i] = i - 1;
    }
    for (long long j = 1; j <= p[0] && i * p[j] < N; ++j) {
      notprime[i * p[j]] = true;
      if (i % p[j] == 0) {
        phi[i * p[j]] = phi[i] * p[j];
        break;
      }
      phi[i * p[j]] = phi[i] * (p[j] - 1);
    }
  }
}
int main() {
  int tar;
  scanf("%d", &tar);
  sieve();
  printf("%d\n", phi[tar - 1]);
}
