import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n, m, k = map(int, input().split())
    x = [0] + list(map(int, input().split())) + [1e6]
    y = [0] + list(map(int, input().split())) + [1e6]
    ans = 0
    for i in range(k):
        xi, yi = map(int, input().split())
        L = 0
        R = 1e6
        while R - L > 1:
            M = (L + R) / 2
            if M - xi >= yi and M - yi > xi:
                L = M
            else:
                R = M
        if L > 0 and R < 1e6:
            ans += 2
    print(ans)
