#include <iostream>
#include <vector>
using namespace std;

vector<string> n1 = {"", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine"};
vector<string> n10 = {"", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety"};
vector<string> n11 = {"ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen", "eighteen", "nineteen"};

int main() {
  int a;
  cin >> a;
  if (a >= 20) { cout << n10[a / 10] << (a % 10 ? "-" : "") << n1[a % 10]; }
  else if (a >= 10) { cout << n11[a - 10]; }
  else { cout << n1[a]; }
  cout << endl;
  return 0;
}
