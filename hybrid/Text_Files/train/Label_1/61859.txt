import queue
# read input
n = int(input())
edges = [[] for _ in range(n+1)]
for _ in range(n-1):
    u, v = map(int, input().split())
    edges[u].append(v)
    edges[v].append(u)
# run BFS to find a leaf furthest from a random starting node
start = 1
dist = [-1] * (n+1)
q = queue.Queue()
q.put((start, 0))
while not q.empty():
    node, d = q.get()
    dist[node] = d
    for adj in edges[node]:
        if dist[adj] == -1:
            q.put((adj, d+1))
# find the farthest leaf from the leaf found in the previous step
farthest = -1
max_dist = -1
for i in range(1, n+1):
    if dist[i] > max_dist:
        max_dist = dist[i]
        farthest = i
# run BFS again to find the farthest leaf from the farthest leaf found in the previous step
dist = [-1] * (n+1)
q = queue.Queue()
q.put((farthest, 0))
while not q.empty():
    node, d = q.get()
    dist[node] = d
    for adj in edges[node]:
        if dist[adj] == -1:
            q.put((adj, d+1))
# the farthest leaf found is the center of the star
center = -1
max_dist = -1
for i in range(1, n+1):
    if dist[i] > max_dist:
        max_dist = dist[i]
        center = i
# remove all edges from the center to its neighbors (to convert to a star)
total = 0
for adj in edges[center]:
    edges[adj].remove(center)
    total += 1
# output minimum operations needed
print(total)
