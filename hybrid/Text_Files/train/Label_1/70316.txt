from collections import deque

def bfs(graph, start):
    visited = set()
    queue = deque([(start, 0)])
    visited.add(start)
    last = start, 0
    while queue:
        vertex, depth = queue.popleft()
        last = vertex, depth
        for neighbor in graph[vertex]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, depth + 1))
    return last

def min_tree_diameter(n, m, edges):
    if m == n - 1:
        print(n - 1)
        return
    
    graph = {i: [] for i in range(1, n + 1)}
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    # Finding all trees in the forest
    visited = set()
    trees = []
    for i in range(1, n + 1):
        if i not in visited:
            vertex, _ = bfs(graph, i)
            _, depth1 = bfs(graph, vertex)
            _, depth2 = bfs(graph, vertex)
            trees.append((depth1, i))
            visited.add(i)
            visited.update(graph[i])
    
    # Sorting trees by diameter in decreasing order
    trees.sort(reverse=True)
    
    # Connecting all trees to form minimal diameter
    print(trees[0][0] + len(trees) - 1)
    for i in range(1, len(trees)):
        print(trees[0][1], trees[i][1])

n, m = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(m)]
min_tree_diameter(n, m, edges)
