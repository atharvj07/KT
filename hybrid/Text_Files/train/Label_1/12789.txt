#include <vector>
#include <cstdio>
#include <tuple>
#include <map>
using namespace std;
int32_t main() {
  map<int32_t, int32_t> dp;
  int32_t n, k;
  scanf("%d%d", &n, &k);
  dp[0] = 1;
  for (int32_t i = 0; i < n; ++i) {
    int32_t coin;
    scanf("%d", &coin);
    map<int32_t, int32_t> ndp;
    for (auto [sum, count] : dp) {
      ndp[sum + coin] += count;
      ndp[sum] += count;
    }
    dp = ndp;
  }
  vector<int32_t> answer;
  for (auto [sum, count] : dp) {
    if (sum <= k) answer.push_back(sum);
  }
  printf("%d\n", (int32_t)answer.size());
  for (int32_t x : answer) {
    printf("%d ", x);
  }
  puts("");
  return 0;
}
