n, d = map(int, input().split())
v = sorted([list(map(int, input().split())) for _ in range(n)], key=lambda x: x[1])
ans = -1
for l, r, c in v:
    if l > d: break
    if d - (r - l + 1) in (x[1] for x in v):
        ans = ans if ans != -1 else c + min(x[2] for x in v if x[1] == d - (r - l + 1))
n = len(v)
for i, (l1, r1, c1) in enumerate(v):
    for j in range(i + 1, n):
        l2, r2, c2 = v[j]
        if l1 <= l2 <= r1 or l2 <= l1 <= r2: continue
        if r1 - l1 + 1 + r2 - l2 + 1 == d:
            ans = ans if ans != -1 else c1 + c2
print(ans)
