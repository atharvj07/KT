#include <iostream>
#define ll long long
using namespace std;

ll n, a, b, s;

int main() {
    cin >> n;
    cin >> a >> b;
    s = a < b ? a : b;
    for (int i = 2; i <= n; i++) {
        ll x, y; cin >> x >> y;
        if (x > s) s = x;
        if (y < b) b = y;
    }
    cout << (s <= b ? 0 : s - b) << endl;
    return 0;
}
