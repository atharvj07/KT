#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    deque<char> dq;
    for (auto c : s) {
      if (c == '(' || c == '[')
        dq.push_back(c);
      else if (dq.empty() || (c == ')' && dq.back() != '(') ||
               (c == ']' && dq.back() != '['))
        continue;
      else
        dq.pop_back();
    }
    cout << dq.empty() << '\n';
  }
  return 0;
}
