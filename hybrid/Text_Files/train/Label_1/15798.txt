#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> b(n + 2);
        for (int i = 0; i < n + 2; i++) cin >> b[i];
        int sum = b[n + 1] - b[n];
        int diff = b[n] - b[0];
        vector<int> a(n, diff);
        if (sum % n != 0) {
            cout << "-1\n";
        } else {
            a[0] += sum / n;
            for (int i = 1; i < n; i++) a[i] += a[i - 1];
            for (int i = 0; i < n; i++)
                cout << a[i] << " ";
            cout << endl;
        }
    }
    return 0;
}
