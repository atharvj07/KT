#include<bits/stdc++.h>
using namespace std;

int a[300005], b[300005];
int main() {
  int n, k; cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];

  int ans, l, r; ans = l = r = 0;
  for (int i = 1, j = 1; i <= n; i++) {
    int d = (a[i - 1] == a[i]);
    if (!a[i]) {
      if (j == i) j = i + 1;
      else {
        if (i - j > k) j = i - k;
        else {
          d = a[i - 1];
          b[i] = 1;
          k -= i - j;
          if (ans < i - j + 1) {
            ans = i - j + 1;
            l = j; r = i;
          }
          j = i + 1;
        }
      }
    }
  }
  if (b[n]) {
    if (n - b[n] > k) b[n] = 0;
    else {
      ans = n - b[n] + 1;
      l = b[n]; r = n;
    }
  }
  cout << ans << endl;
  for (int i = 1; i <= n; i++) {
    if (l <= i && i <= r) cout << 1;
    else cout << a[i];
  }
  return 0;
}
