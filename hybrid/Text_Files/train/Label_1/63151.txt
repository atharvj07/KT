import sys

for i in range(int(sys.stdin.readline())):
    n, m, k = map(int, sys.stdin.readline().split())
    x = [0, 1000000] + [int(j) for j in sys.stdin.readline().split()]
    y = [0, 1000000] + [int(j) for j in sys.stdin.readline().split()]
    a = []
    for j in range(k):
        a.append(list(map(int, sys.stdin.readline().split())))
    ans = 0
    for p in a:
        for q in a:
            if p == q:
                continue
            if (p[1] == q[1] and p[0] != q[0]) or (p[0] == q[0] and p[1] != q[1]):
                ans += 1
                continue
            if abs(p[0] - q[0]) + abs(p[1] - q[1]) > 1000000:
                ans += 1
    print(ans)
