import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }

            int maxScore = 0;
            for (int i = 0; i < n; i++) {
                int score = 0;
                int j = i;
                while (j < n) {
                    score += a[j];
                    j += a[j];
                }
                maxScore = Math.max(maxScore, score);
            }

            System.out.println(maxScore);
        }
    }
}
