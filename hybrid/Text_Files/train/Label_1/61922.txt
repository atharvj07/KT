import math

n = int(input())
for i in range(3, 60):
    if n == i * (i - 1) / 2:
        print(i)
    elif n < i * (i - 1) / 2:
        break
    if n == i * (i - 1) / 2 + (i - 1):
        print(2 * i - 1)
    elif n < i * (i - 1) / 2 + (i - 1):
        break

if n == 0:
    print(0)
if n == 1:
    print(1)
if n == 2:
    print(-1)
