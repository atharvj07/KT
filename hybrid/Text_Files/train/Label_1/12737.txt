#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> x(n);
  for (int i = 0; i < n; i++) cin >> x[i];
  sort(x.begin(), x.end());
  long long ans = 0;
  if (x.size() == 1)
    cout << ans << endl;
  else {
    long long mid = x[(n - 1) / 2];
    cout << ans;
    for (int i = 0; i < n; ans += abs(mid - x[i++]) % 2)
      cout << "|1";
    cout << endl;
  }
  return 0;
}
