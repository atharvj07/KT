#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, c;
  cin >> n >> c;
  vector<int> x(n);
  for (int i = 0; i < n; i++) cin >> x[i];
  int mx = 0;
  for (int i = 0, j = 0; i < n; i = j) {
    j++;
    while (j < n && x[j] >= x[j - 1]) j++;
    if (j < n) {
      mx = max(mx, x[j] - x[i] - c);
    }
  }
  cout << mx << endl;
  return 0;
}
