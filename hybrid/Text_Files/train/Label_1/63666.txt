from collections import Counter
import string

n = int(input())
order = string.ascii_lowercase
first_diff = []
for _ in range(n):
    s = input()
    for t in first_diff:
        for i in range(len(s)):
            if s[i] != t[i]:
                order = order.replace(s[i], t[i]).replace(t[i], s[i])
                break
        else:
            order = order.replace(t[0], s[0]).replace(s[0], t[0])
            break
    else:
        for i in range(len(s)):
            if s[i] != order[i]:
                first_diff.append(s)
                break
        else:
            order = s
if Counter(order) == Counter(string.ascii_lowercase):
    print(order)
else:
    print("Impossible")
