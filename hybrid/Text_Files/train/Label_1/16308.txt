#include <bits/stdc++.h>
using namespace std;
int a[105];
int main(){
    int t;
    cin >> t;
    while(t--){
        int n;
        cin >> n;
        for(int i=1;i<=n;++i){
            cin >> a[i];
        }
        int sum=0;
        for(int i=1;i<=n;++i){
            sum += a[i];
        }
        if(sum % 2 == 1){
            cout << -1 << endl;
            continue;
        }
        vector<int> v;
        for(int i=1;i<=n;++i){
            if(a[i] % 2 == 0){
                v.push_back(i);
            }
        }
        int sz = (int)v.size();
        if(sz == 0){
            v.push_back(1);
        }
        cout << sz << endl;
        for(auto it: v){
            cout << it << " ";
        }
        cout << endl;
    }

    return 0;
}
