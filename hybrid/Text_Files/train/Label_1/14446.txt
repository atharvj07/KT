#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int s, n;
    cin >> s >> n;
    vector<int> ans;
    for (int i = n - 1; i >= 0; i--) {
      int cur = min(s, 10);
      s -= cur;
      ans.push_back(cur);
    }
    for (auto x : ans) cout << x << " ";
    cout << endl;
  }
  return 0;
}
