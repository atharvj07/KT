import sys
input = sys.stdin.readline

n = int(input())
money = list(map(int, input().split()))
events = []
for _ in range(int(input())):
    cmd = input().split()    
    events.append([cmd[0], cmd[1:]])

for cmd, x in events:
    x = int(x[0])
    if cmd == '1':
        money[x-1] = int(x[1])
    else:
        for i in range(len(money)):
            if money[i] < x:
                money[i] = x

print(*money)
