import java.util.*;

public class Main {

    static class Spider {
        int x, y, dir;
        public Spider(int x, int y, int dir) { this.x = x; this.y = y; this.dir = dir; }
    }

    static final int[] dx = { -1, 1, 0, 0 };
    static final int[] dy = { 0, 0, 1, -1 };

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) grid[i] = sc.next().toCharArray();

        List<Spider> spiders = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] != '.') {
                    int dir = 0;
                    switch (grid[i][j]) {
                        case 'L': dir = 0; break;
                        case 'R': dir = 1; break;
                        case 'D': dir = 2; break;
                        case 'U': dir = 3; break;
                    }
                    spiders.add(new Spider(i, j, dir));
                }
            }
        }

        int[] ans = new int[m];
        for (int j = 0; j < m; j++) {
            Set<Integer> seen = new HashSet<>();
            for (Spider spider : spiders) {
                int x = spider.x + dx[spider.dir], y = spider.y + dy[spider.dir];
                while (x >= 0 && x < n && y >= 0 && y < m) {
                    if (x == n - 1) {
                        seen.add((x - 1) * m + y);
                    } else {
                        seen.add(x * m + y);
                    }
                    x += dx[spider.dir];
                    y += dy[spider.dir];
                }
            }
            ans[j] = seen.size();
        }

        for (int num : ans) System.out.print(num + " ");
        System.out.println();
    }
}
