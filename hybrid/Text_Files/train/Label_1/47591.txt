import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long maxx = sc.nextLong(), maxy = sc.nextLong(), w = sc.nextLong();
        long left = 1, right = (long) Math.sqrt(w), ansX = -1, ansY = -1;
        while (left <= right) {
            long midX = (left + right) / 2;
            if (midX * maxy >= w) {
                ansX = midX;
                ansY = (w + ansX - 1) / ansX;
                if (ansX <= maxx && ansY <= maxy) {
                    left = midX + 1;
                } else {
                    right = midX - 1;
                }
            } else {
                right = midX - 1;
            }
        }
        if (ansX == -1) {
            System.out.println(-1);
        } else {
            System.out.println(ansX + " " + ansY);
        }
    }
}
