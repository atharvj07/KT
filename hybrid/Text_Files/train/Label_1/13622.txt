#include <algorithm>
#include <array>
#include <iostream>
#include <vector>

int main() {
  int t, n, q;
  std::cin >> t;
  for (int i = 0; i < t; i++) {
    std::cin >> n >> q;
    std::vector<int> a(n), sum(q);
    for (int& e : a) std::cin >> e;
    for (int& e : sum) std::cin >> e;
    std::sort(a.begin(), a.end());
    for (int& e : sum) {
      if (e < a[0] || e > a[n - 1])
        std::cout << "No" << std::endl;
      else
        std::cout << "Yes" << std::endl;
    }
  }
  return 0;
}
