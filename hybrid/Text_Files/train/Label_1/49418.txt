import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int x = 0, y = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int cnt = 0;
                for (int l = 0; l < n; l++) {
                    if ((a[l] >= a[i] && a[l] >= a[j]) || (a[l] <= a[i] && a[l] <= a[j])) {
                        cnt++;
                    }
                }
                if (cnt == k) {
                    x = a[i];
                    y = a[j];
                    break;
                }
            }
        }
        if (x == 0 && y == 0) {
            System.out.println(-1);
        } else {
            System.out.println(x + " " + y);
        }
    }
}
