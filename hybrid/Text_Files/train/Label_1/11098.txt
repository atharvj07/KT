#include <bits/stdc++.h>
using namespace std;
const int N = 5010;
int n, m;
int dep[N], fa[N];
int get(int x) { return fa[x] == x ? x : fa[x] = get(fa[x]); }
vector<pair<int, int>> g[N];
vector<int> s[N];
int val[N], ans[N], opt[N], ord[N], cur, t;
int a[N], b[N];
void dfs1(int x) {
  for (auto o : g[x]) {
    int y = o.first;
    if (dep[y] > 0) {
      if (s[x].size() < s[y].size()) swap(s[x], s[y]);
      for (int x : s[y]) s[x].clear(), fa[x] = x;
      s[x].insert(s[x].end(), s[y].begin(), s[y].end());
      s[y].clear();
    } else {
      dep[y] = dep[x] + 1;
      dfs1(y);
    }
  }
}
void dfs2(int x, int sum) {
  s[x].push_back(x);
  for (auto o : g[x]) {
    int y = o.first;
    if (dep[y] > 0) continue;
    dfs2(y, sum ^ o.second);
  }
  ans[x] = sum;
  if (s[x].size() > 1) dfs1(x);
  for (int i : s[x]) ord[t++] = i;
  s[x].clear();
  if (s[x].empty()) fa[x] = x;
}
void dfs3(int x, int sum) {
  if (x == -1) return;
  val[x] = sum;
  for (auto o : g[x]) {
    int y = o.first;
    dfs3(y, sum ^ o.second);
  }
}
void dfs4(int x, int sum) {
  if (x == -1) return;
  int id = get(x);
  if (ans[id] != sum) dfs3(id, sum), opt[id] = 1;
  dfs4(fa[id], sum);
}
int main() {
  cin >> n >> m;
  for (int i = 1; i < n; i++) {
    int u, v, w;
    string s, op;
    cin >> s >> s >> s >> u >> op >> v;
    v = op == "?" ? n : v;
    g[u].push_back({v, op == "AND" ? 0 : (op == "OR" ? 1 : 2)});
    fa[i] = i;
  }
  dfs2(1, 0);
  for (int i = 0; i < t; i++) {
    int x = ord[i], y = ord[i ^ 1];
    int u = get(x), v = get(y);
    if (u == v) continue;
    if (s[u].size() < s[v].size()) swap(u, v);
    for (int x : s[v]) fa[x] = u, s[u].push_back(x);
    s[v].clear();
  }
  memset(fa, 0, sizeof(fa)), memset(ans, 0, sizeof(ans)), memset(opt, 0, sizeof(opt));
  for (int i = 1; i <= n; i++) {
    int u = get(i);
    if (ans[u] != 0) dfs4(fa[u], ans[u]);
  }
  for (int i = 1; i <= n; i++) {
    a[i] = val[i] ^ (opt[i] ? ans[get(i)] : 0);
    b[i] = val[i];
  }
  int mask = (1 << m) - 1;
  for (int i = m; i > 0; i--) {
    cout << ((a[n] >> (i - 1)) & 1);
  }
  cout << "\n";
  for (int i = m; i > 0; i--) {
    cout << ((b[n] >> (i - 1)) & 1);
  }
  cout << "\n";
}
