t = int(input())
for _ in range(t):
    n = int(input())
    if n % 2 == 0:
        print((n // 2) ** 2)
        p = list(range(1, n + 1))
        # Rearrange p to match the worst-case scenario
        for i in range(1, n, 2):
            p[i - 1], p[i] = p[i], p[i - 1]
        print(*p)
        print(n // 2)
        for i in range(2, n + 1, 2):
            print(i, i - 1)
    else:
        print((n // 2) ** 2 + (n // 2))
        p = list(range(1, n + 1))
        # Rearrange p to match the worst-case scenario
        for i in range(1, n - 1, 2):
            p[i - 1], p[i] = p[i], p[i - 1]
        print(*p)
        print(n // 2 + 1)
        for i in range(2, n, 2):
            print(i, i - 1)
        print(n, n - 1)
