import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    long[] t = new long[n + 1];
    for (int i = 1; i <= n; i++) {
      t[i] = scanner.nextLong();
    }
    int m = scanner.nextInt();
    int[][] p = new int[m + 1][2];
    for (int i = 1; i <= m; i++) {
      p[i][0] = scanner.nextInt();
      p[i][1] = scanner.nextInt();
    }
    long[] d = new long[m + 1];
    for (int i = 1; i <= m; i++) {
      long sum = 0;
      for (int j = 1; j <= n; j++) {
        if (p[i][0] == j) {
          sum += p[i][1];
        } else {
          sum += t[j];
        }
      }
      d[i] = sum;
    }
    for (int i = 1; i <= m; i++) {
      System.out.println(d[i]);
    }
  }
}
