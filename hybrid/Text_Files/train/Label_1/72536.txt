n = int(input().strip())
cities = list(map(int, input().strip().split()))

for i in range(n):
    if i == 0:
        min_cost = abs(cities[i] - cities[i+1])
        max_cost = abs(cities[i] - cities[-1])
    elif i == n-1:
        min_cost = abs(cities[i] - cities[i-1])
        max_cost = abs(cities[i] - cities[0])
    else:
        min_cost = min(abs(cities[i] - cities[i+1]), abs(cities[i] - cities[i-1]))
        max_cost = max(abs(cities[i] - cities[0]), abs(cities[i] - cities[-1]))
    print(min_cost, max_cost)
