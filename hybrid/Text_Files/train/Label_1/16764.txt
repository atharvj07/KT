#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int INF=1e9;
const int MAXN=2e3+5;
int dp[MAXN][2][2][2];
vector<pair<pair<int,int>,int>> v;
int n,x;
int go(int idx,int prv,int a,int b){
    if(idx==n)return 0;
    int &ret=dp[idx][prv][a][b];
    if(~ret)return ret;
    ret=go(idx+1,prv,a,b);
    if((!prv or a) and v[idx].first.first==0)ret=max(ret,go(idx+1,0,a,b)+v[idx].second);
    if((!prv or b) and v[idx].first.first==1)ret=max(ret,go(idx+1,1,a,b)+v[idx].second);
    if(v[idx].first.second<=x)ret=max(ret,go(idx+1,prv,1,1)+v[idx].second);
    return ret;
}
int main(){
    memset(dp,-1,sizeof dp);
    cin>>n>>x;
    v.resize(n);
    for(int i=0;i<n;i++){
        int tp,h,w;
        cin>>tp>>h>>w;
        v[i]={{tp,h},w};
    }
    sort(v.begin(),v.end());
    cout<<go(0,0,0,0)<<endl;
    return 0;
}
