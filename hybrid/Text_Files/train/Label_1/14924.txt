#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
#define endl "\n"
const int N = 1e5 + 5;
ll x[N], c[N], dp[2][N];

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    ll n, k;
    cin >> n >> k;
    for (ll i = 1; i <= n; i++)
    {
        cin >> x[i];
    }
    ll a;
    cin >> a;
    for (ll i = 1; i <= n; i++)
    {
        cin >> c[i];
    }
    dp[0][0] = 0;
    dp[0][1] = c[1];
    for (ll i = 1; i <= n; i++)
    {
        dp[i % 2][0] = dp[(i - 1) % 2][0];
        dp[i % 2][1] = min(dp[(i - 1) % 2][0] + c[i], dp[(i - 1) % 2][1]);
        ll min_val = min(k + a, x[i]);
        for (ll j = 2; j <= min_val; j++)
        {
            dp[i % 2][j] = min(dp[(i - 1) % 2][j], dp[(i - 1) % 2][j - 1] + c[i]);
        }
        if (dp[i % 2][x[i]] >= mod)
            dp[i % 2][x[i]] = -1;
    }
    ll ans = dp[n % 2][x[n]];
    if (ans == -1)
        cout << -1;
    else
        cout << ans << endl;
    return 0;
}
