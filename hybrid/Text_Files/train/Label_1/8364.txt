#include<bits/stdc++.h>
using namespace std;
int main()
{
    int p,q,l,r;
    cin >> p >> q >> l >> r;
    int a[p][2],b[q][2];
    for(int i=0; i<p; i++) cin >> a[i][0] >> a[i][1];
    for(int i=0; i<q; i++) cin >> b[i][0] >> b[i][1];
    int ans=0;
    for(int i=l; i<=r; i++)
    {
        bool ck=0;
        for(int j=0; j<p; j++)
        {
            if(i>=a[j][0] && i<=a[j][1])
            {
                ck=1;
                break;
            }
        }
        if(ck)
        {
            for(int j=0; j<q; j++)
            {
                if(i+b[j][0]>=b[j][1])continue;
                ans++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
