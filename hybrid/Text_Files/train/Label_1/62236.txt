n=int(input())
a=list(map(int,input().split()))
b=list(map(int,input().split()))

summ1=sum(a)
summ2=sum(b)
if((summ1+summ2)%2==0):
    print(0)
else:
    if(summ1%2==0):
        f=summ2%2
        for i in range(0,n):
            if(b[i]%2!=f):
                print(1)
                exit()
    else:
        f=summ1%2
        for i in range(0,n):
            if(a[i]%2!=f):
                print(1)
                exit()
    print(-1)
