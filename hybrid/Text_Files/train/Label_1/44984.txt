import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String n = br.readLine();
        String m = br.readLine();
        int[] t = new int[m.length() + 1];
        for (int i = 1; i <= m.length(); i++) {
            for (int j = 1; j <= n.length(); j++) {
                if (n.charAt(j - 1) == m.charAt(i - 1)) {
                    t[i] = t[i - 1] + 1;
                } else {
                    t[i] = Math.max(t[i], t[i - 1]);
                }
            }
        }
        Set<Character> s = new HashSet<>();
        int k = m.length(), j = t[m.length()];
        for (int i = m.length(); i >= 1; i--) {
            if (t[i] == j && !s.contains(m.charAt(i - 1))) {
                s.add(m.charAt(i - 1));
                j--;
            }
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n.length(); i++) {
            if (!s.contains(n.charAt(i))) {
                sb.append(n.charAt(i));
            }
        }
        for (int i = 0; i < m.length(); i++) {
            if (!s.contains(m.charAt(i))) {
                sb.append(m.charAt(i));
            }
        }
        char[] c = sb.toString().toCharArray();
        Arrays.sort(c);
        System.out.println(c);
    }
}
