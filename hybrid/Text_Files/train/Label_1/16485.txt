#include <iostream>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n), b(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        for (int i = 0; i < n; i++) {
            cin >> b[i];
        }
        bool found = true;
        for (int i = 0; i < n; i++) {
            if (a[i] + b[n - 1 - i] != b[i] + a[n - 1 - i]) {
                found = false;
                break;
            }
        }
        if (!found) {
            cout << "-1\n";
        } else {
            int ans = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] < b[i]) {
                    cout << ans << "\n";
                    for (int j = 0; j < n; j++) {
                        if (i == j) continue;
                        cout << i + 1 << " " << j + 1 << "\n";
                        ans++;
                    }
                    break;
                }
            }
        }
    }
    return 0;
}
