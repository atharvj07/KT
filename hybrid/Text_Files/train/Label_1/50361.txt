import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }
        int cnt = 0;
        while (true) {
            boolean f = false;
            for (int i = 1; i < n - 1; i++) {
                if (h[i] > h[i - 1] && h[i] > h[i + 1]) {
                    h[i]--;
                    f = true;
                }
            }
            if (!f) {
                break;
            }
            cnt++;
        }
        System.out.println(cnt);
    }
}
