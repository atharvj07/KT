import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class ForbiddenPairs {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String sentence = scanner.nextLine();
        int forbiddenPairsCount = scanner.nextInt();
        scanner.nextLine(); // Consume the empty line

        Set<Character> forbiddenPairs = new HashSet<>();
        for (int i = 0; i < forbiddenPairsCount; i++) {
            String pair = scanner.nextLine();
            forbiddenPairs.add(pair.charAt(0));
            forbiddenPairs.add(pair.charAt(1));
        }

        int deletions = 0;
        for (int i = 0; i < sentence.length() - 1; i++) {
            if (forbiddenPairs.contains(sentence.charAt(i)) && forbiddenPairs.contains(sentence.charAt(i + 1))) {
                deletions++;
                i++; // Skip the next character as it's already been deleted
            }
        }

        System.out.println(deletions);
    }
}
