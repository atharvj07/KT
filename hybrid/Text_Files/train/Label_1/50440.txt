import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        int right = 0;
        int left = 0;
        while (right < n - 1) {
            int r = d[right];
            if (left + r >= n - 1) {
                System.out.println("yes");
                return;
            }
            int max = 0;
            int j = right + 1;
            while (j < n && d[j] <= r) {
                max = Math.max(max, d[j]);
                j++;
            }
            if (max == 0) {
                System.out.println("no");
                return;
            }
            left = right;
            right = j - 1;
        }
        System.out.println("no");
    }
}
