#include<bits/stdc++.h>
using namespace std;
int main(){
    string s; cin >> s;
    int n = s.size();
    int l = -1, r = -1;
    for (int i = 0; i < n; i++){
        if (s[i] == 'z') continue;
        if (l == -1){
            l = i; r = i;
        }
        r = i;
    }
    if (l == -1) l = 0;
    for (int i = l; i <= r; i++) s[i] = (s[i] == 'a' ? 'z' : s[i] - 1);
    cout << s << endl;
    return 0;
}
