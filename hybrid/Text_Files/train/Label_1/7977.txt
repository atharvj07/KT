#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n; cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> b;
  for (int i = 0; i < n; i++) if (a[i] != 0) b.push_back(a[i]);
  int ans = 0;
  for (int i = 0; i + 1 < (int) b.size(); i++) ans += (b[i] % 2) ^ (b[i + 1] % 2);
  cout << ans << endl;
}
