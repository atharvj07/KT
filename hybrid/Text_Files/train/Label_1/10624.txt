#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;

  vector<vector<bool>> a(n, vector<bool>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      int x;
      cin >> x;
      a[i][j] = x;
    }
  }

  vector<int> cnt(n, 0);
  for (int i = 0; i < k; i++) {
    int x, y;
    cin >> x >> y;
    for (int j = 0; j < n; j++) {
      if (a[j][y - 1] && j != x - 1) {
        cnt[j]++;
      }
    }
  }

  for (int i = 0; i < n; i++) {
    cout << cnt[i] << " ";
  }
  cout << endl;
  return 0;
}
