import java.util.Scanner;

public class AutomaticDoor {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read input
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();
        int[] clients = new int[m];
        for (int i = 0; i < m; i++) {
            clients[i] = scanner.nextInt();
        }

        // Calculate the number of times the door will open
        int count = 0;
        int lastOpenTime = -d; // Initialize with a negative value to indicate that the door is initially closed
        for (int i = 0; i < n; i++) {
            int employeeTime = i * a;
            if (employeeTime >= lastOpenTime + d) {
                count++;
                lastOpenTime = employeeTime;
            }
        }
        for (int clientTime : clients) {
            if (clientTime >= lastOpenTime + d) {
                count++;
                lastOpenTime = clientTime;
            }
        }

        // Print output
        System.out.println(count);
    }
}
