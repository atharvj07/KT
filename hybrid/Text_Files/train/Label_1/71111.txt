def solve_triangle_problem(n, m, k):
    if 2 * n * m % k != 0:
        print("NO")
    else:
        print("YES")
        if k % 2 == 0:
            k //= 2
            if n % k == 0:
                print(f"0 0\n{ n // k } 0\n0 1")
            else:
                print(f"0 0\n0 { m // k }\n1 0")
        else:
            if n % k == 0:
                print(f"0 0\n{ n // k } 0\n0 2")
            elif m % k == 0:
                print(f"0 0\n0 { m // k }\n2 0")
            else:
                print(f"0 0\n{ n // (k // 2) } 0\n0 2")

n, m, k = map(int, input().split())
solve_triangle_problem(n, m, k)
