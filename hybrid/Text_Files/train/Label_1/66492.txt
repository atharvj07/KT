import bisect

n = int(input())
requests = []
for _ in range(n):
    c, p = map(int, input().split())
    requests.append((c, p))
    
k = int(input())
capacities = list(map(int, input().split()))
capacities.sort()

requests.sort(reverse=True)
taken = [False] * k
revenue = 0
result_requests = []
result_tables = []

for c, p in requests:
    i = bisect.bisect_left(capacities, c)
    if i < k and not taken[i]:
        revenue += p
        taken[i] = True
        result_requests.append(requests.index(c, p) + 1)
        result_tables.append(i + 1)

print(len(result_requests), revenue)
for r, t in zip(result_requests, result_tables):
    print(r, t)
