import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int[] a = new int[n+1];
		for(int i = 1; i <= n; i++)
			a[i] = sc.nextInt();

		int[] ans = new int[n+1];
		for(int i = 1; i <= n; i++){
			if(a[i] == i){
				ans[i] = 0;
				continue;
			}
			int l = i, r = i;
			while(l > 1 && a[l-1] >= a[l])
				l--;
			while(r < n && a[r+1] >= a[r])
				r++;

			ans[i] = Math.max(Math.abs(i - l), Math.abs(i - r));
		}

		for(int i = 1; i <= n; i++)
			System.out.print(ans[i] + " ");
	}
}
