#include <cstdio>
#include <vector>
#include <queue>
using namespace std;
int n, s, a[200001], ans = 0;
bool f[200001];
vector<int> adj[200001];
queue<int> q;
int main() {
    scanf("%d%d", &n, &s);
    for (int i = 1; i <= n; ++i) {
        scanf("%d", &a[i]);
        if (a[i] < 0 || a[i] > n - 1) {
            ++ans;
            f[i] = 1;
            q.push(i);
        }
    }
    while (!q.empty()) {
        int now = q.front();
        q.pop();
        if (now != s) a[now]--;
        for (int i = 0; i < adj[now].size(); ++i) {
            int v = adj[now][i];
            if (a[v] <= 0) {
                a[v]--;
                if (v != s) {
                    if (!f[v]) {
                        q.push(v);
                        ++ans;
                        f[v] = 1;
                    }
                }
            }
        }
    }
    printf("%d", ans);
    return 0;
}
