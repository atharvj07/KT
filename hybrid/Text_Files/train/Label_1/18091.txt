#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int ans = 0;
  int now = 0;
  for (int i = 0; i < n; i++) {
    now = (s[i] == 'N' || s[i] == '?');
    ans = max(ans, now);
    ans += (s[i] == 'N' && now == 0);
  }
  ans = max(ans, now);
  cout << (ans == k ? "YES" : "NO");
  return 0;
}
