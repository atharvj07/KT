#include<bits/stdc++.h>
using namespace std;
int n, m, type, l, r, a[100005];
inline long long maxl[100005], minl[100005];
int main() {
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> a[i];
	sort(a + 1, a + n + 1);
	for (int i = n; i >= 1; i--) maxl[i] = maxl[i + 1] + a[i];
	for (int i = 1; i <= n; i++) minl[i] = minl[i - 1] + a[i];
	cin >> m;
	for (int i = 0; i < m; i++) {
		cin >> type >> l >> r;
		cout << (type == 1 ? maxl[l] - maxl[r + 1] : minl[r] - minl[l - 1]) << '\n';
	}
	return 0;
}
