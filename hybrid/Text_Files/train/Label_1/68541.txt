import sys
for t in range(int(sys.stdin.readline())):
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    x = 0
    for i in range(n):
        if a[i] == a[i] % (n + 1):
            x = a[i]
            break
    ans = 0
    for i in range(n):
        if a[i] != x:
            ans += 1
    print(ans)
