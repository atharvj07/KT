#include<bits/stdc++.h>
using namespace std;
const int N=1e3+15,M=1e6+15;
char maze[N][N];
int n,m,k,x,y;
int cycle[M];
char dir[]="LURD";
int main(){
    ios::sync_with_stdio(false);cin.tie(0);
    cin>>n>>m>>k;
    for(int i=0;i<n;++i){
        cin>>(maze[i]+1);
        for(int j=1;j<=m;++j)if(maze[i][j]=='X'){x=i,y=j;}
    }
    int last=0,p=0;
    while(last+4<=k){
        for(int j=0;j<4;++j){
            if(maze[x][y+dir[j]-65]=='.'){
                cycle[last]=dir[j]-65;
                last++;
                maze[x][y]=dir[j]-65+'0';
                if(dir[j]=='U')--x;
                if(dir[j]=='D')++x;
                if(dir[j]=='L')--y;
                if(dir[j]=='R')++y;
                if(x==1&&y==1){p=last;break;}
            }
        }
        if(p)break;
    }
    if(!p){cout<<"IMPOSSIBLE";return 0;}
    cycle[last]=cycle[0];cycle[last+1]=0;
    int cyc=last-p+1;
    int t=(k-p)/cyc;
    k-=p+t*cyc;
    if(k){cout<<cycle;for(int i=1;i<=k;++i)cout<<cycle[i%cyc];}
    else{for(int i=p;i<=last;++i)cout<<cycle[i];}
    return 0;
}
