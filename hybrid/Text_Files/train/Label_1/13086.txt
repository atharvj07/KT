#include <iostream>
const int mod = 1e9 + 7;
void add( int &x, int y ) { x = ( x + y ) % mod; }
int main() {
    int a, b, n;
    std::cin >> a >> b >> n;
    int dp[ n + 1 ][ 2 ][ 2 ];
    dp[ 1 ][ 1 ][ 0 ] = dp[ 1 ][ 0 ][ 1 ] = dp[ 1 ][ 0 ][ 0 ] = 1;
    for( int i = 2; i <= n; i++ ) {
        for( int j = 0; j < 2; j++ ) {
            for( int k = 0; k < 2; k++ ) {
                int sum = j + k;
                add( dp[ i ][ j ][ k ] , dp[ i - 1 ][ j ][ k ] );
                if( sum % 2 == 0 || sum == 1 ) add( dp[ i ][ j ][ k ] , dp[ i - 1 ][ j ][ 1 ] );
                if( sum % 2 == 0 || sum == a ) add( dp[ i ][ j ][ k ] , dp[ i - 1 ][ 1 ][ k ] );
                if( sum % 2 == 0 || sum == b ) add( dp[ i ][ j ][ k ] , dp[ i - 1 ][ 1 ][ 1 ] );
            }
        }
    }
    std::cout << dp[ n ][ 1 ][ 1 ] << '\n';
    return 0;
}
