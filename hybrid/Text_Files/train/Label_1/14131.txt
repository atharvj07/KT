#include <bits/stdc++.h>
using namespace std;
#define rep(i, a, b) for(int i = a; i < (b); ++i)
#define trav(a, x) for(auto& a : x)
#define all(x) begin(x), end(x)
#define sz(x) (int)(x).size()
typedef long long ll;
typedef pair<int, int> pii;
typedef vector<int> vi;

const int MAXN = 5005;

int main() {
	int n, m, T;
	cin >> n >> m >> T;
	vector<vi> g(n);
	vector<array<int, 2>> edges;
	rep(i, 0, m) {
		int u, v, t;
		cin >> u >> v >> t;
		--u;
		--v;
		g[u].push_back(v);
		g[v].push_back(u);
		edges.push_back({u, v});
	}

	vector<int> dist(n, (int)1e9);
	dist[0] = 0;
	priority_queue<pii> pq;
	pq.push({0, 0});
	while(pq.size()) {
		int d, u;
		tie(d, u) = pq.top();
		pq.pop();
		d = -d;
		if(d > dist[u])
			continue;
		trav(v, g[u]) {
			if(d + 1 < dist[v]) {
				dist[v] = d + 1;
				pq.push({-(d + 1), v});
			}
		}
	}

	if(dist[n - 1] > T) {
		cout << "IMPOSSIBLE";
		return 0;
	}

	vector<int> path;
	int u = n - 1;
	while(u != 0) {
		trav(v, g[u]) {
			if(dist[u] == dist[v] + 1) {
				path.push_back(v);
				u = v;
			}
		}
	}

	cout << sz(path) + 1 << "\n";
	trav(v, path)
		cout << v + 1 << " ";
	cout << n << "\n";

	return 0;
}
