#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long n, a, b;
  cin >> n >> a >> b;
  vector<long long> s(n), sort_s(n);
  for (long long i = 0; i < n; i++) {
    cin >> s[i];
    sort_s[i] = s[i];
  }
  sort(sort_s.begin(), sort_s.end());
  long long sum = 0;
  for (long long i = 0; i < n - 1; i++) {
    sum += sort_s[i];
  }
  long long cnt = 0;
  for (long long i = n - 1; i >= 0; i--) {
    if (sum * b >= s[i] * (a - b)) {
      sum -= s[i];
      cnt++;
    } else {
      break;
    }
  }
  cout << cnt << endl;
  return 0;
}
