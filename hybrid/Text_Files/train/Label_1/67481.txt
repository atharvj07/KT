n, k = map(int, input().split())
r = list(map(int, input().split()))
q = [[] for i in r]
for i in range(k):
    a, b = map(int, input().split())
    a -= 1
    b -= 1
    q[a].append(b)
    q[b].append(a)
l = [0] * n
for i in range(n):
    for j in range(n):
        if r[i] > r[j] and i not in q[j]:
            l[i] += 1
print(*l)
