#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, c0, c1, h;
    string s;
    cin >> n >> c0 >> c1 >> h;
    cin >> s;
    int ans = 0;
    int sum0 = 0, sum1 = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '0')
        sum0++;
      else
        sum1++;
    }
    ans += min(c0 * sum0, c1 * sum0 + h);
    ans += min(c1 * sum1, c0 * sum1 + h);
    cout << ans << endl;
  }
  return 0;
}
