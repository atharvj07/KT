n = int(input().strip())
m = list(map(int, input().strip().split()))

min_marks = [0] * n
min_marks[0] = m[0]
for i in range(1, n):
    min_marks[i] = max(min_marks[i-1], m[i])

sum_di = 0
for i in range(n-1, 0, -1):
    min_marks[i] = max(min_marks[i], min_marks[i-1] + 1)
    sum_di += min_marks[i] - m[i] - 1

print(sum_di)
