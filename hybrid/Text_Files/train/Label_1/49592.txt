import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            char[] s = scanner.next().toCharArray();
            int[] last = new int[26];
            for (int j = 0; j < n; j++) {
                last[s[j] - 'a'] = j;
            }
            boolean[] used = new boolean[26];
            StringBuilder mex = new StringBuilder();
            for (int j = 0; j < n; j++) {
                if (!used[s[j] - 'a']) {
                    char c = (char) (s[j] - 1);
                    while (c >= 'a' && last[c - 'a'] >= j) {
                        c--;
                    }
                    if (c >= 'a') {
                        mex.append(c);
                        used[c - 'a'] = true;
                    } else {
                        break;
                    }
                }
            }
            if (mex.length() == 0) {
                mex.append((char) (s[n - 1] + 1));
            }
            System.out.println(mex);
        }
    }
}
