n = int(input())
x = 0
y = 20000

for i in range(n):
    t, s = input().split()
    t = int(t)
    
    if y == 20000:
        if s == 'North':
            y -= t
        elif s == 'South':
            pass
        else:
            print('NO')
            exit()
    elif y == 0:
        if s == 'South':
            y += t
        elif s == 'North':
            pass
        else:
            print('NO')
            exit()
    else:
        if s in ['South', 'North']:
            y -= t
        else:
            x += t

if x == 0 and y == 20000:
    print('YES')
else:
    print('NO')
