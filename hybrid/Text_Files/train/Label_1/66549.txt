def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

n = int(input().strip())
a = list(map(int, input().strip().split()))

counts = {}
for number in a:
    counts[number] = counts.get(number, 0) + 1

max_subset = []
if 1 in counts:
    ones_count = counts[1]
    max_subset = [1] * ones_count
    max_prime = 0
    for number in counts:
        if number != 1 and is_prime(number + 1):
            if number > max_prime:
                max_prime = number
    if max_prime > 0:
        max_subset.append(max_prime)
else:
    for number in counts:
        max_subset = [number]
        break

print(len(max_subset))
print(" ".join(map(str, max_subset)))
