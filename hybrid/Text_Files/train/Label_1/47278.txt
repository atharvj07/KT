import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        String[] grid = new String[n];
        sc.nextLine();
        for (int i = 0; i < n; i++) grid[i] = sc.nextLine();
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < m - 1; j++) {
                if (grid[i].charAt(j) == 'f' && grid[i].charAt(j + 1) == 'a' && grid[i + 1].charAt(j) == 'c' && grid[i + 1].charAt(j + 1) == 'e') ans++;
            }
        }
        System.out.println(ans);
    }
}
