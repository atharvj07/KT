#include <bits/stdc++.h>

using namespace std;

int main() {
    double a, b, ans = 1e9;
    int n;
    cin >> a >> b >> n;
    while (n--) {
        double x, y, v;
        cin >> x >> y >> v;
        double dx = x - a, dy = y - b;
        double dist = sqrt(dx * dx + dy * dy);
        double time = dist / v;
        ans = min(ans, time);
    }
    cout << fixed << setprecision(17) << ans << endl;
    return 0;
}
