import java.util.Scanner;
class Main {
    public static void main(String[] args){
        Scanner input=new Scanner(System.in);
        long n=input.nextLong();
        int k=input.nextInt();
        int ans=0;
        long x=n;
        while(true){
            ans++;
            if(x%10==0)x/=10;
            else{
                x=x/10+1;
                x%=10;
            }
            if(x%k==0){
                System.out.println(ans);
                break;
            }
        }
    }
}
