#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<pair<long long, long long>> v(n);
        vector<bool> success(n, false);
        for (int i = 0; i < n; i++) {
            cin >> v[i].first >> v[i].second;
        }
        for (int i = 0; i < n - 1; i++) {
            long long l = v[i].first, r = v[i + 1].first;
            long long x = v[i].second;
            if (x >= l && x <= r) success[i] = true;
        }
        cout << count(success.begin(), success.end(), true) << "\n";
    }
    return 0;
}
