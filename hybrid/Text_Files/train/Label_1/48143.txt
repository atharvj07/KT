import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), y0 = sc.nextInt(), y1 = sc.nextInt();
        int[] mx = new int[n], my = new int[m];
        for (int i = 0; i < n; i++) mx[i] = sc.nextInt();
        for (int i = 0; i < m; i++) my[i] = sc.nextInt();
        int[] dist = new int[n];
        int[] first = new int[n];
        for (int i = 0; i < n; i++) {
            int dm = Integer.MAX_VALUE, id = -1;
            for (int j = 0; j < m; j++) {
                int d = Math.abs(mx[i] - my[j]) + Math.abs(y0 - y1);
                if (d < dm) {
                    dm = d;
                    id = j;
                }
            }
            dist[i] = dm;
            first[i] = id;
        }
        int[] cnt = new int[m];
        int res = n;
        for (int i = 0; i < n; i++) {
            if (++cnt[first[i]] == 1) res--;
        }
        System.out.println(res);
    }
}
