#include <bits/stdc++.h>
#define rep(i, n) for(int i = 0; i < n; ++i)
using namespace std;
typedef long long llong;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, W;
    cin >> n >> W;
    vector<llong> v(n);
    rep(i, n) cin >> v[i];
    sort(v.begin(), v.end());
    llong ans = 0;
    int j = 0, cnt = 0;
    rep(i, n) {
      if (ans + v[i] > W) break;
      ans += v[i];
      ++cnt;
      ++j;
    }
    if (ans >= 0.5 * W) {
      cout << cnt << endl;
      rep(i, cnt) cout << j - i - 1 << ' ';
      cout << endl;
    } else {
      cout << -1 << endl;
    }
  }
  return 0;
}
