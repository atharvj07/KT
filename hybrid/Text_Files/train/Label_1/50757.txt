import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int N = input.nextInt();
        int[] a = new int[N], b = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = input.nextInt();
            b[i] = input.nextInt();
        }
        boolean flag = false;
        for (int i = 0; i < N; i++) {
            int j = 0;
            while (j < N && !(a[j] == a[i] && b[j] == b[i])) {
                j++;
            }
            if (j == N) {
                flag = true;
                break;
            }
        }
        if (flag) {
            System.out.println("Ani");
        } else {
            System.out.println("Borna");
        }
    }
}
