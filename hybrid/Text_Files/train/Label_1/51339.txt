import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int L = sc.nextInt();
        List<Integer> kefa = new ArrayList<Integer>();
        List<Integer> sasha = new ArrayList<Integer>();
        for (int i = 0; i < n; i++) {
            kefa.add(sc.nextInt());
        }
        for (int i = 0; i < n; i++) {
            sasha.add(sc.nextInt());
        }
        Collections.sort(kefa);
        Collections.sort(sasha);
        boolean flag = true;
        for (int i = 0; i < n; i++) {
            if (kefa.get(i) != sasha.get(i)) {
                flag = false;
                break;
            }
        }
        System.out.println(flag ? "YES" : "NO");
    }
}
