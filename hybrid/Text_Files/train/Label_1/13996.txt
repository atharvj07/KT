#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e5 + 5;

int n;
vector<pair<int, pair<int, int>>> g[MAXN];
int ans;

void dfs(int u, int p = 0) {
  set<int> st;
  for (auto& [v, e] : g[u]) {
    if (v == p) continue;
    dfs(v, u);
    st.insert(e.first);
    st.insert(e.second);
  }
  if (st.size() > 2) ans++;
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int v, l, r;
    cin >> v >> l >> r;
    if (l != -1) g[i].emplace_back(l, make_pair(v, -1));
    if (r != -1) g[i].emplace_back(r, make_pair(-1, v));
  }
  dfs(1);
  cout << ans << endl;
  return 0;
}
