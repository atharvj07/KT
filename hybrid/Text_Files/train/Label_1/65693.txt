import heapq

def min_time_eating_carrots(n, k, a):
    # Initial heap for carrots, we use negative because Python has a min heap by default
    carrots = [-x for x in a]
    heapq.heapify(carrots)
    
    # The total number of pieces we need to split into is k - n since we already have n pieces
    total_splits = k - n
    
    while total_splits > 0:
        # Get the largest carrot
        carrot = -heapq.heappop(carrots)
        
        # Calculate optimal split for this carrot
        # We find the number of pieces to split into by finding the divisor that minimizes the time
        # for the current carrot, which minimizes (carrot // pieces)^2 * pieces + (carrot % pieces)^2
        best_split = 1
        min_time = float('inf')
        for pieces in range(1, total_splits + 2):  # +2 because range is exclusive and we start from 1
            large_piece = carrot // pieces
            small_piece = carrot % pieces
            time = (large_piece ** 2) * (pieces - small_piece) + (large_piece + 1) ** 2 * small_piece
            if time < min_time:
                min_time = time
                best_split = pieces
        
        # Split the carrot into best_split pieces and add back to the heap
        large_piece = carrot // best_split
        small_piece_size = large_piece + 1
        num_small_pieces = carrot % best_split
        
        for _ in range(best_split - num_small_pieces):
            heapq.heappush(carrots, -large_piece)
        for _ in range(num_small_pieces):
            heapq.heappush(carrots, -small_piece_size)
        
        total_splits -= best_split - 1  # We replace one carrot with best_split pieces
    
    # Calculate total time
    total_time = 0
    while carrots:
        total_time += (-heapq.heappop(carrots)) ** 2
    
    return total_time

n, k = map(int, input().split())
a = list(map(int, input().split()))

print(min_time_eating_carrots(n, k, a))
