import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class ForbiddenPairs {

    public static void main(String[] args) throws IOException {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            String sentence = br.readLine();
            int numPairs = Integer.parseInt(br.readLine());
            Map<Character, Character> forbiddenPairs = new HashMap<>();
            for (int i = 0; i < numPairs; i++) {
                String[] pair = br.readLine().split("");
                forbiddenPairs.put(pair[0].charAt(0), pair[1].charAt(0));
            }

            int deletions = 0;
            for (int i = 1; i < sentence.length(); i++) {
                char left = sentence.charAt(i - 1);
                char right = sentence.charAt(i);
                if (forbiddenPairs.containsKey(left) && forbiddenPairs.get(left) == right) {
                    deletions++;
                }
            }

            System.out.println(deletions);
        }
    }
}
