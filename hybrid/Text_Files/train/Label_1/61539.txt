t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    pos = {a[i]: i for i in range(n)}
    last = -1
    time = 0
    
    for i in range(m):
        if pos[b[i]] > last:
            time += 2 * (pos[b[i]] - i) + 1
            last = pos[b[i]]
        else:
            time += 1
            
    print(time)
