#include <bits/stdc++.h>
using namespace std;

const int maxn = 2e5 + 1;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> r(n), g[maxn], ans(n);
    unordered_set<int> s[maxn];
    for (int& x : r) cin >> x;
    for (int i = 0; i < k; i++) {
        int x, y;
        cin >> x >> y;
        x--, y--;
        s[x].insert(y);
        s[y].insert(x);
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (r[j] < r[i] && !s[i].count(j)) g[i].push_back(j);
        }
    }
    for (int i = 0; i < n; i++) {
        sort(g[i].begin(), g[i].end());
        for (int j = 0; j < g[i].size(); j++) {
            int pos = upper_bound(g[i].begin(), g[i].end(), g[i][j]) - g[i].begin();
            ans[i] += pos;
        }
    }
    for (int& x : ans) cout << x << ' ';
    cout << '\n';
}
