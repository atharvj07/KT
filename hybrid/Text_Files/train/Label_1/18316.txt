#include <iostream>
#include <vector>
using namespace std;

const int maxn = 1e6 + 1;
int val[maxn], divisor[maxn], f[maxn], tmp[maxn];

int find(int x) { return x == f[x] ? x : f[x] = find(f[x]); }

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> val[i]; divisor[val[i]]++; }
  int size = 0;
  for (int i = 1; i <= n; i++) {
    int x = find(val[i]);
    f[x] = find(divisor[val[i]]);
    tmp[x]++;
    size = max(size, tmp[x]);
  }
  cout << size << endl;
  return 0;
}
