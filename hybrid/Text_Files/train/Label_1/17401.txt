#include <iostream>
#include <vector>
#include <utility>

using namespace std;

const int mod = 1e9 + 7;

long long pow_mod(long long base, long long exp) {
    if (exp == 0) { return 1; }
    long long ans = pow_mod(base, exp / 2);
    ans = (ans * ans) % mod;
    if (exp % 2) { ans = (ans * base) % mod; }
    return ans;
}

long long inv_mod(long long x) { return pow_mod(x, mod - 2); }

pair<vector<long long>, vector<long long>> get_factorial_inverse() {
    vector<long long> factorial(1001);
    vector<long long> inverse_factorial(1001);
    factorial[0] = 1, factorial[1] = 1;
    inverse_factorial[0] = 1, inverse_factorial[1] = 1;
    for (int i = 2; i < 1001; ++i) {
        factorial[i] = (i * factorial[i - 1]) % mod;
        inverse_factorial[i] = inv_mod(factorial[i]);
    }
    return make_pair(factorial, inverse_factorial);
}

int main() {
    int n, k;
    cin >> n >> k;
    const auto [factorial, inverse_factorial] = get_factorial_inverse();
    cout << (factorial[n] * inverse_factorial[n - k]) % mod << endl;
    return 0;
}
