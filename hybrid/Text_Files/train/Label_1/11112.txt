const int MOD = 1000000007;

const double PI = 3.14159265358979323846;
const ll sz = 1e6+7;
const int MAXN = 1<<20;
int a[sz];
int b[sz];
void solve() {
  ll k,aa,bb,ansA=0,ansB=0,i;
  cin>>k>>aa>>bb;
  for(i=1;i<=3;i++) cin>>a[i]>>a[i+3]>>a[i+6];
  for(i=1;i<=3;i++) cin>>b[i]>>b[i+3]>>b[i+6];
  while(k--) {
    aa = a[1+aa];
    bb = b[1+bb];
    if(aa==bb) continue;
    if(aa==1 && bb==2) ansA++;
    if(aa==2 && bb==3) ansA++;
    if(aa==3 && bb==1) ansA++;
    if(aa==2 && bb==1) ansB++;
    if(aa==3 && bb==2) ansB++;
    if(aa==1 && bb==3) ansB++;
  }
  cout<<ansA<<" "<<ansB<<endl;
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  solve();
  return 0;
}
