#include <bits/stdc++.h>
using namespace std; typedef long long ll;
const ll MAXN = 3e5 + 5; ll n, k, cost[MAXN], dp[MAXN];
int main() { cin >> n >> k; for (int i = 1; i <= n; i++) cin >> cost[i];
    priority_queue<pair<ll, int>> pq; for (int i = 1; i <= n; i++) pq.push({-cost[i], i});
    ll ans = 0; for (int i = k + 1; i <= k + n; i++) {
        pair<ll, int> tmp = pq.top(); pq.pop();
        ans -= tmp.first; dp[tmp.second] = i;
    }
    cout << ans << endl; for (int i = 1; i <= n; i++) cout << dp[i] << " ";
    return 0;
}
