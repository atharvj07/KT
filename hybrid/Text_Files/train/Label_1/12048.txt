#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9+7;
const char nl = '\n';

ll gcd(ll a, ll b) {
    while (b) {
        ll t = a;
        a = b;
        b = t % b;
    }
    return a;
}
int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for (ll i = 0; i < n; ++i) {
        cin >> a[i];
        if (i > 0 && gcd(a[i], a[i-1]) == 1) {
            cout << "No" << endl;
            return 0;
        }
    }
    cout << "Yes" << endl;
    return 0;
}
