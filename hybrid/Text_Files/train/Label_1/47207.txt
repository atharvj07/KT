import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int vp = sc.nextInt();
        int vd = sc.nextInt();
        int t = sc.nextInt();
        int f = sc.nextInt();
        int c = sc.nextInt();
        int time = t + f;
        int bijous = (int) Math.ceil((double) c / (vd - vp) * time / t);
        System.out.println(bijous);
    }
}
