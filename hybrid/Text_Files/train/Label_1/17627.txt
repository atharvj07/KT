#include<bits/stdc++.h>
#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  string a[] = {"vaporeon", "jolteon", "flareon", "espeon", "umbreon", "leafeon", "glaceon", "sylveon"};
  for (auto x : a) {
    if (x.size() == n) {
      bool f = true;
      for (int j = 0; j < n; j++) {
        if (s[j] != '.' && s[j] != x[j]) {
          f = false;
          break;
        }
      }
      if (f == true) {
        cout << x << "\n";
        return 0;
      }
    }
  }
}
