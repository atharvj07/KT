#include <bits/stdc++.h>
using namespace std;
#define rep(i,n)for(int i=0;i<n;i++)
const int mod=1e9+7;
int main() {
    int q; cin >> q;
    string s0="What are you doing at the end of the world? Are you busy? Will you save us?";
    rep(i,q) {
        int n,k; cin >> n >> k;
        if(n==0) cout << s0[k%s0.size()-1];
        else {
            long long l=pow(k+1,n);
            long long t=l-1;
            for(int i=0;!(1<<i>t);i++) l>>=1;
            string sn=s0;
            rep(i,n) sn+="What are you doing while sending \""+sn+"\"? ";
            sn+="Are you busy? Will you send \""+sn+"\"?";
            cout << sn[t-l]-'.';
        }
    }
    return 0;
}
