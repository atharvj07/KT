import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int k = sc.nextInt();
        String s = String.valueOf(n);
        int m = 0;
        for (int i = s.length() - 1; i >= 0; i--) {
            int x = s.charAt(i) - '0';
            if (x % k != 0) {
                m = i;
                break;
            }
        }
        long ans = 0;
        long p = 1;
        for (int i = m; i < s.length(); i++) {
            int x = s.charAt(i) - '0';
            if (x % k == 0) {
                ans += p * x;
                p *= 10;
            }
        }
        if (ans % (long) Math.pow(10, s.length() - m - 1) == 0) {
            System.out.println(s.length() - 1 - m);
        } else {
            System.out.println(s.length() - m);
        }
    }
}
