t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    
    counts = {}
    for i, num in enumerate(a):
        if num not in counts:
            counts[num] = [i+1, 1]  # Store index and occurrence
        else:
            counts[num][1] += 1
    
    winner = -1
    min_val = float('inf')
    for num, details in counts.items():
        if details[1] == 1 and num < min_val:
            min_val = num
            winner = details[0]
    
    print(winner)
