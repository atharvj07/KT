#include <bits/stdc++.h>
using namespace std;
#define fastio ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);

int main() {
    fastio;
    int n, sum = 0;
    cin >> n;
    vector<int> v(n);
    for(auto& e : v) cin >> e;
    for(int i = 1; i < n - 1; i++) sum += min(v[i], min(v[i-1], v[i+1]));
    cout << sum << endl;
}
