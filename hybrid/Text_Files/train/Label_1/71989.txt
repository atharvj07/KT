n = int(input().strip())
a = list(map(int, input().strip().split()))
b = list(map(int, input().strip().split()))

for i in range(n):
    a[i] *= (i + 1) * (n - i)
a.sort()
b.sort(reverse=True)

mod = 998244353
answer = 0
for i in range(n):
    answer = (answer + (a[i] % mod) * (b[i] % mod)) % mod

print(answer)
