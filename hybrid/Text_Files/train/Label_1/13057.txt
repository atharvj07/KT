#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, t;
    cin >> n >> t;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (t >= a[i]) {
            t -= a[i];
            ans++;
        } else
            break;
    }
    cout << ans << endl;
    return 0;
}
