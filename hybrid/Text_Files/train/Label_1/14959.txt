#include <bits/stdc++.h>
using namespace std;
const int maxn=1e5,INF=2e9;
const long long llinf=1ll*INF*INF;
int n;
struct segtree{
    int t[4*maxn];
    void build(int p,int l,int r,int v[]){
        if(l==r){
            t[p]=v[l];
            return;
        }
        int mid=(l+r)/2;
        build(p*2,l,mid,v);
        build(p*2+1,mid+1,r,v);
        t[p]=max(t[p*2],t[p*2+1]);
    }
    int query(int p,int l,int r,int ll,int rr){
        if(l>rr||r<ll){
            return -INF;
        }
        if(l>=ll&&r<=rr){
            return t[p];
        }
        int mid=(l+r)/2;
        int x=query(p*2,l,mid,ll,rr);
        int y=query(p*2+1,mid+1,r,ll,rr);
        return max(x,y);
    }
}st1,st2;
struct node{
    int l,r,ty,id;
};
vector<node>v;
int main(){
    cin>>n;
    vector<pair<int,int>>v1,v2;
    for(int i=1;i<=n;++i){
        int l1,r1,l2,r2;cin>>l1>>r1>>l2>>r2;
        v1.emplace_back(l1,l2);
        v1.emplace_back(r1+1,r2+1);
        v2.emplace_back(l2,l1);
        v2.emplace_back(r2+1,r1+1);
    }
    for(int i=0;i<(int)v1.size();++i){
        if(v1[i].first<=v1[i].second){
            v.push_back({v1[i].first,v1[i].second,0,i});
        }
    }
    sort(v.begin(),v.end(),[](const node&a,const node&b)->bool{
        return a.l<b.l;
    });
    int cur=-INF;
    for(int i=0;i<(int)v.size();++i){
        if(v[i].l<=cur){
            cout<<"NO";
            return 0;
        }
        cur=v[i].r;
    }
    v.clear();
    for(int i=0;i<(int)v2.size();++i){
        if(v2[i].first<=v2[i].second){
            v.push_back({v2[i].first,v2[i].second,1,i});
        }
    }
    sort(v.begin(),v.end(),[](const node&a,const node&b)->bool{
        return a.l<b.l;
    });
    cur=-INF;
    for(int i=0;i<(int)v.size();++i){
        if(v[i].l<=cur){
            cout<<"NO";
            return 0;
        }
        cur=v[i].r;
    }
    cout<<"YES";
    return 0;
}
