#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1000000007;
const int N = 2e5 + 10;
#define endl "\n"
#define faster  ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);

int cost[N];
int dp[N][3];
const int inf = 1e18;

int solve(int i,int stat, int n, int a, int b)
{
	if(i == n + 1)
	{
		if(stat == 2) return 0;
		return inf;
	}
	if(dp[i][stat]!=-1) return dp[i][stat];
	int ans = inf;
	if(cost[i] != 2)
	{
		if(stat!= 2)
		{
			if(cost[i] == 0)
			{
				ans = solve(i + 1,0,n,a,b);
			}
			else ans = solve(i + 1,1,n,a,b);
		}
	}
	else
	{
		if(stat != 2)
		{
			ans = min(solve(i + 1,0,n,a,b) + a,solve(i + 1,1,n,a,b) + b);
		}
		else
		{
			ans = min(solve(i + 1,2,n,a,b),min(solve(i + 1,0,n,a,b) + a,solve(i + 1,1,n,a,b) + b));
		}
	}
	return dp[i][stat] = ans;
}
int32_t main()
{
	faster;
	int n,a,b;
	cin >> n >> a >> b;
	memset(dp,-1,sizeof dp);
	for(int i = 1; i <= n; i++)
	{
		cin >> cost[i];
	}
	int ans;
	ans=solve(1,2,n,a,b);
	if(ans>=(1e18)) ans = -1;
	cout << ans << endl;
	return 0;
}
