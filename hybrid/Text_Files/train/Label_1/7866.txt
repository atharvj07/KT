#include <iostream>
#include <vector>
using namespace std;
int main() {
  int N, M;
  cin >> N >> M;
  vector<string> grid(N);
  for (auto &row : grid) cin >> row;
  int first_x = -1, first_y = -1, last_x = -1, last_y = -1;
  for (int i = 0; i < N; ++i) {
    for (int j = 0; j < M; ++j) {
      if (grid[i][j] == 'B') {
        if (first_x == -1) {
          first_x = i;
          first_y = j;
        }
        last_x = i;
        last_y = j;
      }
    }
  }
  if (last_x == -1) { cout << -1 << endl; } else
    cout << (last_x - first_x + 1) * (last_y - first_y + 1) -
              (N * M - (grid[first_x][first_y] == 'B'))
        << endl;
}
