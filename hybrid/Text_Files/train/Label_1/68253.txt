h,m,s,t1,t2=map(int,input().split())
f=[0]*13
f[t1]=1
f[t2]=-1
for i in range(1,h+1):
    if i%12 in (m,s):
        f[i%12]+=1
        f[(i+1)%12]+=1
    elif i%12 not in (t1,t2):
        f[i%12]+=2
        f[(i+1)%12]+=2
for i in range(1,h+1):
    if f[i%12]==1 and f[(i+1)%12]==1 and f[(i+2)%12]==1:
        print("YES")
        break
    if f[i%12]==1 and f[(i+1)%12]==2 and f[(i-1)%12]==1:
        print("YES")
        break
else:
    print("NO")
