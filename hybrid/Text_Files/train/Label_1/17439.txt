#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

int main() {
      string s, t;
      cin >> s >> t;
      vector<int> pos[26];
      for (int i = 0; i < s.length(); i++) {
            if (s[i] == '?') continue;
            pos[s[i] - 'a'].push_back(i);
      }
      for (int i = 0; i < t.length(); i++) {
            char c = t[i];
            int j = 0;
            int p = 0;
            while (j < pos[c - 'a'].size() && p < i) {
                  int pj = pos[c - 'a'][j];
                  if (pj >= i) {
                        s[pj] = c;
                        p++;
                        swap(pos[c - 'a'][j], pos[c - 'a'].back());
                        pos[c - 'a'].pop_back();
                  } else {
                        j++;
                  }
            }
      }
      cout << s << endl;
}
