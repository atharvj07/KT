import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Line> lines = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            long a = scanner.nextInt();
            long b = scanner.nextInt();
            long c = scanner.nextInt();
            lines.add(new Line(a, b, c));
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (lines.get(i).isTriangleOriginal(lines.get(j), lines.get(k))) {
                        ans++;
                    }
                }
            }
        }
        System.out.println(ans);
    }

    public static class Line {
        private long a;
        private long b;
        private long c;

        public Line(long a, long b, long c) {
            this.a = a;
            this.b = b;
            this.c = c;
        }

        public boolean isTriangleOriginal(Line line1, Line line2) {
            long a1 = this.a;
            long b1 = this.b;
            long c1 = this.c;
            long a2 = line1.a;
            long b2 = line1.b;
            long c2 = line1.c;
            long a3 = line2.a;
            long b3 = line2.b;
            long c3 = line2.c;
            long x1 = (b2 * c3 - b3 * c2) / (a2 * b3 - a3 * b2);
            long y1 = (c2 * a3 - c3 * a2) / (b2 * a3 - b3 * a2);
            if (a1 * x1 + b1 * y1 == c1) {
                return true;
            } else {
                return false;
            }
        }
    }
}
