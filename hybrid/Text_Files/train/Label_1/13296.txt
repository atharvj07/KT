#include <iostream>
#include <vector>
#include <cstring>

using namespace std;

const int INF = 1e9;

struct t_shirt {
    int price, a, b;
};

vector<t_shirt> A;
vector<int> B;
int C[200005];

int main() {
    int n, m;
    cin >> n;
    A.resize(n);
    B.resize(n);
    for(int i = 0; i < n; i++) {
        cin >> A[i].price;
        cin >> A[i].a >> A[i].b;
        B[i] = A[i].a * 3 + A[i].b;
    }
    cin >> m;
    for(int i = 0; i < m; i++) cin >> C[i];

    for(int i = 0; i < m; i++) {
        int ans = INF;
        for(int j = 0; j < n; j++) {
            if(A[j].a != C[i] && A[j].b != C[i]) continue;
            ans = min(ans, A[j].price);
        }

        if(ans < INF) cout << ans << " ";
        else cout << -1 << " ";
    }
    cout << endl;

    return 0;
}
