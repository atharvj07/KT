#include <bits/stdc++.h>
using namespace std;

int main() {
    int l, r;
    cin >> l >> r;
    if ((r - l + 1) % 2 == 1) {
        cout << "NO" << endl;
    } else {
        cout << "YES" << endl;
        for (int i = l; i <= r; i += 2) {
            if (i + 1 <= r) cout << i << " " << i + 1 << endl;
            else cout << i << " " << l << endl;
        }
    }
    return 0;
}
