#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int mx = 0;
  vector<int> cnt(1000005);
  for (int i = 0; i < n; i++) {
    cnt[a[i]]++;
    mx = max(mx, cnt[a[i]]);
  }
  int l = 0, r = 0;
  int now = 0;
  for (int i = 0; i < n; i++) {
    if (cnt[a[i]] == mx && now == 0) {
      l = i;
      now = 1;
    }
    if (cnt[a[i]] == mx && now == 1) {
      r = i;
    }
  }
  cout << l + 1 << " " << r + 1 << endl;
  return 0;
}
