import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        String s = sc.next();
        char[] a = s.toCharArray();
        int[] ans = new int[10];
        while (q-- > 0) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt();
            int[] cnt = new int[10];
            for (int i = l; i < r; i++) {
                char ch = a[i];
                if (ch == '<' || ch == '>') {
                    continue;
                }
                cnt[ch - '0']++;
            }
            for (int i = l; i < r; i++) {
                char ch = a[i];
                if (ch == '<' || ch == '>') {
                    continue;
                }
                if (cnt[ch - '0'] > 0) {
                    ans[ch - '0']++;
                    cnt[ch - '0']--;
                    if (cnt[ch - '0'] == 0) {
                        a[i] = ' ';
                    }
                }
            }
            for (int i = 0; i < 10; i++) {
                System.out.print(ans[i] + " ");
            }
            System.out.println();
            ans = new int[10];
        }
    }
}
