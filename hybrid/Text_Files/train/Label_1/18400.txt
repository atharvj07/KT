#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1005;
const int MAXM = 1005;
int dp[MAXN][MAXM][2];
int a[MAXN][MAXM];

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
  }
  dp[1][1][0] = a[1][1];
  dp[n][1][1] = a[n][1];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      dp[i][j][0] = max(dp[i - 1][j][0], dp[i - 1][j][1]) + a[i][j];
      dp[i][j][1] = max(dp[i + 1][j][1], dp[i + 1][j][0]) + a[i][j];
    }
  }
  cout << max(dp[n][m][0], dp[n][m][1]) << endl;
  return 0;
}
