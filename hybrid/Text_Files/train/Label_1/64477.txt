import sys

n, k = list(map(int, input().split()))
need = set(list(map(int, input().split())))
graph = [[]] + [list(map(int, input().split()[1:])) for i in range(n)]
vis = [False] * (n + 1)
ans = []

def dfs(v):
    ans.append(v)
    vis[v] = True
    for u in graph[v]:
        if not vis[u]:
            dfs(u)

need_i = k
for i in range(len(vis)):
    if i in need and not vis[i]:
        dfs(i)
        need_i -= 1
        if need_i == 0:
            break

print(len(ans) if need_i == 0 else -1)
if need_i == 0:
    print(*ans)
