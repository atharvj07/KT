#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n), c(n + 1, 0);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<int> f(n + 1, 0);
  for (int i = 0; i < n; i++) {
    int mx = 0, mx_col = n;
    for (int j = i; j < n; j++) {
      c[a[j]]++;
      if (c[a[j]] > mx || (c[a[j]] == mx && a[j] < mx_col)) {
        mx = c[a[j]];
        mx_col = a[j];
      }
      f[mx_col] += j - i + 1;
    }
    for (int j = i; j < n; j++) { c[a[j]]--; }
  }
  for (int i = 1; i <= n; i++) { cout << f[i] << " "; }
  cout << endl;
  return 0;
}
