#include <bits/stdc++.h>

using namespace std;

const int MAXN = 200000;

int main() {
  int n, a[MAXN + 1], x, f;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  cin >> x >> f;
  long long ans = 0;
  for (int i = 0; i < n; i++) ans += (a[i] + f - 1) / x * f;
  cout << ans << endl;
  return 0;
}
