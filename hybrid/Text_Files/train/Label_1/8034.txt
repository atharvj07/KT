#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> v(n + 1);
  for (int i = 1; i <= n; i++) cin >> v[i];
  vector<int> dp(n + 1);
  for (int i = 3; i <= n; i++) {
    if (v[i] == v[i - 1] - 1 && v[i] == v[i - 2] - 2) dp[i] = dp[i - 2] + 1;
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) ans = max(ans, dp[i]);
  cout << ans + 2 << endl;
  return 0;
}
