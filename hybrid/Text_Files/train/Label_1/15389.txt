#include <bits/stdc++.h>

using namespace std;

#define ll long long

const int mxN = 1e5 + 5;

int n;
ll a[mxN];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }

    sort(a + 1, a + 1 + n);

    ll ans = 0;
    int j = n;
    for (int i = 1; i <= n; i++) {
        while (j > i && a[j] > a[i] + 1) j--;
        if (j <= i) break;
        ans += a[i] * a[j];
        j--;
    }

    cout << ans << endl;

    return 0;
}
