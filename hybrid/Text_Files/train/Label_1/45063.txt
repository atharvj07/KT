import java.util.*;
import java.lang.*;


class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();

        int[][] a = new int[n][n];
        int[] row = new int[n];
        int[] col = new int[n];

        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < n; j++) {
                a[i][j] = s.charAt(j) - '0';
            }
        }

        int res = 0;

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (a[i][j] == 0) {
                    if (row[i] == 0 && col[j] == 0) {
                        row[i] = 1;
                        col[j] = 1;
                        res++;
                    }
                }

                if (a[i][j] == 1) {
                    if (row[i] == 1 && col[j] == 1) {
                        row[i] = 0;
                        col[j] = 0;
                        res--;
                    }
                }
            }
        }

        System.out.println(res);
    }
}
