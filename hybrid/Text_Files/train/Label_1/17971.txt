#include <bits/stdc++.h>
using namespace std;
int n, p, q;
string s;
vector<string> v;
int main() {
  cin >> n >> p >> q;
  cin >> s;
  if ((n - 1) % (p + q) != 0) {
    return cout << "-1", 0;
  }
  int x = (n - 1) / (p + q);
  for (int i = 1; i < s.length(); i++) {
    if (v.size() < x) {
      v.push_back(s.substr(i - 1, p));
      i += p - 1;
    } else if (v.size() < 2 * x) {
      v.push_back(s.substr(i - 1, q));
      i += q - 1;
    } else {
      break;
    }
  }
  cout << v.size() << endl;
  for (auto x : v) {
    cout << x << endl;
  }
}
