def max_expected_score(N, prob):
    teams = 1 << N
    dp = [[0.0 for _ in range(teams)] for _ in range(N+1)]
    dp[0] = [1.0] * teams
    
    def win_prob(team1, team2, round):
        segment_length = 1 << round
        segment_start = (team1 // segment_length) * segment_length
        opponents = [i for i in range(segment_start, segment_start + segment_length) if i != team1]
        return sum(prob[team1][i] * dp[round-1][i] for i in opponents)
    
    for round in range(1, N+1):
        for team in range(teams):
            dp[round][team] = dp[round-1][team] * win_prob(team, None, round) / 100.0
    
    score = 0
    for round in range(1, N+1):
        score += sum(dp[round][team] for team in range(teams)) * (1 << (round-1))
    
    return score

N = int(input().strip())
prob = [list(map(int, input().strip().split())) for _ in range(1 << N)]
print(f"{max_expected_score(N, prob):.9f}")
