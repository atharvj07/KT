import bisect

n, m, ta, tb, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

bn = b + [b[-1] + tb]
for i in range(n):
    idx = bisect.bisect_right(bn, a[i] + ta)
    if idx <= k:
        k -= idx
        a[i] = -1

if k > 0:
    a.sort(reverse=True)
    for i in range(min(k, n)):
        if a[i] > 0:
            a[i] = -1

ans = -1
for i in range(n):
    if a[i] > 0:
        idx = bisect.bisect_right(bn, a[i] + ta)
        ans = max(ans, bn[idx - 1] + tb)

print(ans)
