#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, l, r, k;
    cin >> n >> l >> r >> k;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int ans = 0;
    long long total = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] >= l && a[i] <= r) {
        if (total + a[i] <= k) {
          ans++;
          total += a[i];
        } else {
          break;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
