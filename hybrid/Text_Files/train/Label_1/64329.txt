t = int(input())
for _ in range(t):
    n = int(input())
    sizes = list(map(int, input().split()))
    max_size = max(sizes)
    dominant_piranha = -1
    for i in range(n):
        if sizes[i] == max_size:
            if (i > 0 and sizes[i-1] < sizes[i]) or (i < n-1 and sizes[i+1] < sizes[i]):
                dominant_piranha = i + 1
                break
    print(dominant_piranha)
