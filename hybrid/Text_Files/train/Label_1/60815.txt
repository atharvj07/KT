n = int(input())
a = list(map(int, input().split()))
pref = [0]
sum = 0
for i in range(n):
    sum += a[i]
    pref.append(sum)
l, r = -1, -1
ans = 0
for i in range(n - 1):
    for j in range(i + 1, n):
        cur = pref[j] - pref[i]
        if j > 0:
            cur -= pref[j - 1]
        if j == n - 1:
            cur -= pref[j]
        if cur <= pref[n - 1] - cur and pref[n - 1] - cur >= 0:
            if j - i > ans:
                l, r = i, j
                ans = j - i
print(pref[l] - pref[i])
