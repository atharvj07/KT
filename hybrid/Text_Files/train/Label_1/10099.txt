#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
ll p[1000005];
ll quickpow(ll a, ll n) {
    ll ret = 1;
    while (n) {
        if (n & 1) ret = (ret * a) % mod;
        a = (a * a) % mod;
        n >>= 1;
    }
    return ret;
}
ll c(ll a, ll b) {
    if (a < b) return 0;
    return ((p[a] * quickpow(p[a - b], mod - 2)) % mod * quickpow(p[b], mod - 2)) % mod;
}
int main() {
    ll n, l, r;
    cin >> n >> l >> r;
    p[0] = 1;
    for (ll i = 1; i <= r; ++i) p[i] = (p[i - 1] * i) % mod;
    if ((r - l + 1) % 3 != 0) {
        cout << 0 << endl;
        return 0;
    }
    ll ans = c(r - l + 1, (r - l + 1) / 3);
    cout << ans << endl;
    return 0;
}
