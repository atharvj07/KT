#include <iostream>

using namespace std;

int main() {
  long long n;
  cin >> n;
  if (n == 2)
    cout << -1 << endl;
  else {
    while (n % 2 == 0) n /= 2;
    if (n == 1)
      cout << 3 << endl;
    else {
      for (long long i = 3;; i += 2) {
        if (n == i * (i - 1) / 2) {
          cout << i << endl;
          return 0;
        }
      }
    }
  }
}
