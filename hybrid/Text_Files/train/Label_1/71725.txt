import sys
import math

n = int(input())
p = []

for i in range(n):
    x, y = map(int, input().split())
    p.append((x, y))

def cross(o, a, b):
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

mn = math.inf
for i in range(n):
    for j in range(n):
        if i == j or (i + 1) % n == j:
            continue
        a, b = p[i], p[j]
        d = math.inf
        for k in range(n):
            c = p[k]
            if cross(a, b, c) < 0:
                d = min(d, ((c[1] - a[1])**2 + (c[0] - a[0])**2) / (cross(a, b, c)**2))
        mn = min(mn, math.sqrt(d))

print(mn / 2)
