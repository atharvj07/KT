#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int n, sum;
bool seen[100];
vector<pair<int, int>> v[100];
void dfs(int x, int prv, int sum_, int &ans) {
  ans = max(ans, sum_);
  seen[x] = 1;
  for (auto p : v[x]) {
    if (p.first == prv) continue;
    if (!seen[p.first]) dfs(p.first, x, sum_ + p.second, ans);
  }
}
int main() {
  cin >> n;
  for (int i = 0; i < n - 1; i++) {
    int x, y, z;
    cin >> x >> y >> z;
    v[x].emplace_back(y, z);
    v[y].emplace_back(x, z);
    sum += z;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (!seen[i]) { dfs(i, i, 0, ans); }
  }
  cout << ans;
  return 0;
}
