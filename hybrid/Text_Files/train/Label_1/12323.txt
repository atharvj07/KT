#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1005;
ll n, l[N], r[N], deg[N];
vector<int> ans;
int main() {
    cin >> n;
    for (ll i = 0; i < n; i++) cin >> l[i];
    for (ll i = 0; i < n; i++) cin >> r[i];
    for (ll i = 0; i < n; i++) {
        deg[i + 1] = r[i];
        deg[i] += l[i];
    }
    ans.resize(n);
    for (ll i = 1; i <= n; i++) {
        if (deg[i] == 0) {
            cout << "NO\n";
            return 0;
        }
        ans[i - 1] = deg[i];
        if (i != n) deg[i + 1] -= deg[i];
    }
    cout << "YES\n";
    for (ll i = 0; i < n; i++) cout << ans[i] << " ";
    return 0;
}
