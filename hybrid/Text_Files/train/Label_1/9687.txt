#include <vector>
#include <iostream>
using namespace std;
typedef long long ll;

const int mxN = 40;
int n;
ll p;
ll a = 0;
vector<string> v;

int main() {
    cin >> n >> p;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        v.push_back(s);
    }
    ll cnt = 0;
    for (int i = 0; i < n; i++) {
        string s = v[i];
        if (cnt % 2 == 0) {
            cnt = (cnt + 1) / 2;
        }
        if (s == "halfplus") {
            a += p / 2;
            cnt++;
        }
        else {
            cnt = (cnt + 1) / 2;
        }
    }
    cout << a << '\n';
    return 0;
}
