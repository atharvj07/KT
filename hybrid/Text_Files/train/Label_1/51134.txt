import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        BigInteger[] a = new BigInteger[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextBigInteger();
        }
        BigInteger[] b = new BigInteger[n];
        for (int i = 0; i < n; i++) {
            BigInteger sum = BigInteger.ZERO;
            for (int j = 0; j < n; j++) {
                sum = sum.xor(a[j]);
            }
            sum = sum.xor(a[i]);
            b[i] = sum;
        }
        for (int i = 0; i < n; i++) {
            System.out.print(b[i] + " ");
        }
        System.out.println();
    }
}
