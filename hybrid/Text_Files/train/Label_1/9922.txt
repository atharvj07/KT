#include <bits/stdc++.h>
#define fi first
#define se second
using namespace std;

const int N = 2e5+5;
int n, m, q, a[N], b[N], c[N], vis[N];

void answer(int z) {
    cout<<z;
    exit(0);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin>>n>>m>>q;
    for(int i = 0; i < q; i++) {
        cin>>a[i]>>b[i];
        c[(a[i]-1)*m+b[i]] = 1;
    }
    int ans = n*m - q;
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            int cur = (i)*m + j;
            if(c[cur] || vis[cur]) continue;
            bool f = 0, g = 0, h = 0, p = 0;
            if(i-1 >= 0) {
                vis[(i-1)*m+j] = 1;
                if(c[(i-1)*m+j] == 0) f = 1;
            }
            if(i+1 < n) {
                vis[(i+1)*m+j] = 1;
                if(c[(i+1)*m+j] == 0) g = 1;
            }
            if(j-1 >= 0) {
                vis[(i)*m+j-1] = 1;
                if(c[(i)*m+j-1] == 0) h = 1;
            }
            if(j+1 < m) {
                vis[(i)*m+j+1] = 1;
                if(c[(i)*m+j+1] == 0) p = 1;
            }
            if(f && g && h || f && g && p || g && h && p || f && h && p) ans--;
        }
    }
    answer(ans);
    return 0;
}
