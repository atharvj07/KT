import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println(solve(sc.next()));
        sc.close();
    }

    static int solve(String s) {
        int[] cnt = new int[26];
        for (char c : s.toCharArray()) {
            cnt[c - 'a']++;
        }
        int ans = 0, odd = 0;
        for (int i : cnt) {
            if (i % 2 == 1) {
                odd++;
            }
            ans += i / 2;
        }
        ans += (odd + 1) / 2;
        return ans;
    }
}
