#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, x, y;
  std::cin >> n >> x >> y;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }

  std::sort(a.begin(), a.end());
  int ans = 1;
  for (int i = x; i < n - y; ++i) {
    if (a[i] < a[i - x] && a[i] < a[i + y]) {
      ans = i + 1;
      break;
    }
  }

  std::cout << ans << std::endl;
  return 0;
}
