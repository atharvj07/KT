def is_valid(grid, n, m):
    for i in range(n):
        for j in range(m):
            if grid[i][j] == '.':
                cnt_w, cnt_r = 0, 0
                if i - 1 >= 0:
                    cnt_r += grid[i - 1][j] == 'R'
                    cnt_w += grid[i - 1][j] == 'W'
                if i + 1 < n:
                    cnt_r += grid[i + 1][j] == 'R'
                    cnt_w += grid[i + 1][j] == 'W'
                if j - 1 >= 0:
                    cnt_r += grid[i][j - 1] == 'R'
                    cnt_w += grid[i][j - 1] == 'W'
                if j + 1 < m:
                    cnt_r += grid[i][j + 1] == 'R'
                    cnt_w += grid[i][j + 1] == 'W'
                if cnt_r > 0 and cnt_w > 0:
                    return False
    return True

def dfs(grid, i, j, n, m):
    if i == n:
        if is_valid(grid, n, m):
            for row in grid:
                print(''.join(row))
            return True
        return False

    if j == m:
        return dfs(grid, i + 1, 0, n, m)

    if grid[i][j] != '.':
        return dfs(grid, i, j + 1, n, m)

    grid[i][j] = 'R'
    if dfs(grid, i, j + 1, n, m):
        return True
    grid[i][j] = 'W'
    if dfs(grid, i, j + 1, n, m):
        return True
    grid[i][j] = '.'
    return False

t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    grid = [list(input()) for _ in range(n)]
    if dfs(grid, 0, 0, n, m):
        print('YES')
    else:
        print('NO')
