#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<pair<char, long long>> a(n);
    for (int i = 0; i < n; i++) {
        char c;
        long long x;
        cin >> c >> x >> c;
        a[i] = {c, x};
    }
    long long l = -2e9, r = 2e9;
    for (int i = 0; i < n; i++) {
        if (a[i].first == '>') {
            l = max(l, a[i].second + 1);
        } else if (a[i].first == '<') {
            r = min(r, a[i].second - 1);
        } else if (a[i].first == '<=') {
            r = min(r, a[i].second);
        } else {
            l = max(l, a[i].second);
        }
    }
    if (l > r)
        cout << "Impossible" << endl;
    else
        cout << l << endl;
}
