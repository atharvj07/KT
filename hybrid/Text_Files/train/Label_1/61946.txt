import sys
input = sys.stdin.readline

N, M = map(int, input().split())
A = []
for i in range(N): A.append(input())
Q = int(input())

for i in range(N):
  for j in range(M):
    if (i==0 or A[i-1][j]=='.' or j==0 or A[i][j-1]=='.') and A[i][j]=='.': A[i][j] = 'E'

for i in range(Q):
  x1, x2 = map(int, input().split())
  x1 -= 1
  x2 -= 1
  ans = 'YES'
  for j in range(x1, x2):
    for k in range(N):
      if A[k][j]=='.' and (
        (k==0 or A[k-1][j]=='X' or j==0 or A[k][j-1]=='X') or
        (k==N-1 or A[k+1][j]=='X' or j==M-1 or A[k][j+1]=='X')
      ):
        ans = 'NO'
        break
    if ans=='NO': break
  print(ans)
