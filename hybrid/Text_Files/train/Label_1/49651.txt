import java.util.*;

class Main {
    public static void main(String[] args) throws Exception {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int week = n / 7;
        int i = 0;
        long ink = 0;
        for (int j = 1; j <= n; j++) {
            ink += in.nextInt();
            if (j % 7 != 0 || week == i) {
                if (ink < 7 - j % 7) {
                    i = j;
                }
            }
            ink -= 1;
        }
        System.out.print(i);
    }
}
