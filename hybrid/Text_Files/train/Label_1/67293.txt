n = int(input())
asphalting_days = []
asphalted_h = set()
asphalted_v = set()

for i in range(1, n**2 + 1):
    h, v = map(int, input().split())
    if h not in asphalted_h and v not in asphalted_v:
        asphalting_days.append(i)
        asphalted_h.add(h)
        asphalted_v.add(v)

print(*asphalting_days)
