#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k, q;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  cin >> q;
  vector<int> x(q);
  for (int i = 0; i < q; i++) {
    cin >> x[i];
  }
  for (int i = 0; i < q; i++) {
    int ans = -1;
    for (int j = 0; j < n; j++) {
      if (x[i] >= a[j]) {
        int cnt = x[i] / a[j];
        if (cnt <= k) {
          if (ans == -1) {
            ans = cnt;
          } else {
            ans = min(ans, cnt);
          }
        }
      }
    }
    for (int j = 0; j < n; j++) {
      for (int l = j; l < n; l++) {
        if (x[i] >= a[j] + a[l]) {
          int cnt1 = x[i] / a[j];
          int cnt2 = (x[i] - cnt1 * a[j]) / a[l];
          if (cnt1 + cnt2 <= k) {
            if (ans == -1) {
              ans = cnt1 + cnt2;
            } else {
              ans = min(ans, cnt1 + cnt2);
            }
          }
        }
      }
    }
    cout << ans << endl;
  }
}
