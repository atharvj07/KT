import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        long a = s.nextLong(), b = s.nextLong(), f = s.nextLong(), k = s.nextLong();
        if (b < f || (a - f) % b != 0) {
            System.out.println(-1);
        } else {
            System.out.println(k * (f - 1) / (a - f));
        }
    }
}
