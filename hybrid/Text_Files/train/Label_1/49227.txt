import java.util.Scanner;
import java.math.BigDecimal;
import java.math.RoundingMode;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int w = in.nextInt();
        int b = in.nextInt();
        BigDecimal num = new BigDecimal(w);
        BigDecimal den = new BigDecimal(w + b);
        BigDecimal res = num.divide(den, 9, RoundingMode.HALF_UP);
        res = res.multiply(num.add(BigDecimal.ONE));
        res = res.divide(den.add(BigDecimal.ONE), 9, RoundingMode.HALF_UP);
        System.out.println(res);
    }
}
