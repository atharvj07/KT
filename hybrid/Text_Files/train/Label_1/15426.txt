#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;
int main() {
    int n, sum = 0;
    cin >> n;
    vector<int> a(n), p(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
        sum += a[i];
    }
    if (sum % 2 == 0)
        cout << -1 << '\n';
    else {
        sort(a.begin(), a.end());
        for (int i = 0; i < n; ++i)
            if (sum - a[i] <= sum / 2) {
                cout << i + 1 << '\n';
                for (int j = 0; j < n; ++j)
                    if (i != j)
                        cout << j + 1 << ' ';
                return 0;
            }
        cout << -1 << '\n';
    }
    return 0;
}
