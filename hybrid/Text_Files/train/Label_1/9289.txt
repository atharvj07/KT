#include <iostream>
#include <vector>
using namespace std;
const int N = 2000 + 10;
typedef long long ll;
struct edge {
    int v, w;
};
vector<edge> g[N];
bool vis[N];
ll dfs(int u) {
    if (vis[u]) return 0;
    vis[u] = 1;
    ll ans = 1;
    for (edge t : g[u]) {
        ans += dfs(t.v);
    }
    return ans;
}
int main() {
    int n, m, k;
    char c;
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> c;
            if (c == '.') g[i].push_back({j, 1});
        }
    }
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (vis[j]) continue;
            if (dfs(j) >= k) ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
