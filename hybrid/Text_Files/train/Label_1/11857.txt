#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int>a(n);
    for (int j = 0; j < n; ++j)
        cin >> a[j];
    int i = 0, j = n - 1;
    while (i < j) {
        if (a[i] < a[i + 1]) ++i;
        else if (a[j] < a[j - 1]) --j;
        else break;
    }
    if (i == j) cout << "YES";
    else cout << "NO";
}
