#include <bits/stdc++.h>

#define endl '\n'

typedef long long ll;

using namespace std;

const int MAX_N = 2e5 + 5;

int n, k, x;
vector<int> a;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> k >> x;
    a.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    // dp[i][j] = maximum bitwise OR of the first i elements after performing j operations
    vector<vector<int>> dp(n + 1, vector<int>(k + 1, 0));

    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= k; j++) {
            dp[i][j] = dp[i - 1][j];
            if (j > 0) {
                dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] | a[i - 1]);
                dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] | (a[i - 1] * x));
            }
        }
    }

    cout << dp[n][k] << endl;

    return 0;
}
