n = int(input())
v = list(map(int, input().split()))
u = sorted(v)
m = int(input())

prefix_v = [0] * (n + 1)
prefix_u = [0] * (n + 1)
for i in range(1, n + 1):
    prefix_v[i] = prefix_v[i - 1] + v[i - 1]
    prefix_u[i] = prefix_u[i - 1] + u[i - 1]

for _ in range(m):
    type, l, r = map(int, input().split())
    if type == 1:
        print(prefix_v[r] - prefix_v[l - 1])
    else:
        print(prefix_u[r] - prefix_u[l - 1])
