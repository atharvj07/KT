#include <cstdio>
#include <cstring>
#include <algorithm>
using namespace std;

typedef long long ll;
const ll inf = (ll)1e18;
const int N = 2e5 + 5;
int n, m, k, s, a[N], b[N], t[N], c[N], r[N];
ll f[N];

int main() {
    scanf("%d%d%d%d", &n, &m, &k, &s);
    for (int i = 1; i <= n; ++i) scanf("%d", &a[i]);
    for (int i = 1; i <= n; ++i) scanf("%d", &b[i]);
    for (int i = 1; i <= m; ++i) scanf("%d%d", &t[i], &c[i]);
    memset(f, 0x3f, sizeof f);
    f[0] = 0;
    for (int i = 1; i <= m; ++i) {
        for (int j = k; j >= 0; --j) {
            ll cost;
            if (!r[c[i]]) r[c[i]] = i;
            if (t[i] == 1) cost = 1ll * a[r[c[i]]] * c[i];
            else cost = 1ll * b[r[c[i]]] * c[i];
            if (cost <= s && j + 1 <= k) f[j + 1] = min(f[j + 1], f[j] + cost);
        }
        if (f[k] <= s) {
            printf("%lld\n", r[c[i]]);
            for (int j = 1; j <= k; ++j) {
                int x = lower_bound(f, f + k + 1, s - f[j - 1]) - f;
                printf("%d %lld\n", c[x], r[c[x]]);
                s -= f[x - 1];
            }
            break;
        }
    }
    if (f[k] > s) puts("-1");
}
