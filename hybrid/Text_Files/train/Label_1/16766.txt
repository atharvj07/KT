#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

typedef long long ll;

int main() {
    int n, x;
    cin >> n >> x;
    vector<pair<int, pair<int, int>>> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i].first >> a[i].second.first >> a[i].second.second;
    }
    sort(a.begin(), a.end());
    int ans = 0;
    bool last = false;
    for (int i = 0; i < n; i++) {
        if (a[i].second.first <= x) {
            if (last == a[i].first) continue;
            x += a[i].second.second;
            last = a[i].first;
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
