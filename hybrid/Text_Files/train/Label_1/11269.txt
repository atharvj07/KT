#include<bits/stdc++.h>
using namespace std;
const int maxn = 200010;
long long n, a[maxn], ans = LLONG_MIN;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    for (int i = 1; i <= n; i++) {
        if (i == n)
            ans = max(ans, a[i]);
        else
            ans = max({ans, a[i], a[i] + a[i + 1]});
    }
    cout << ans << endl;
    return 0;
}
