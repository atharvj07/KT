import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        for (int i = 0; i < t; i++) {
            String s = br.readLine();
            int n = s.length();
            boolean[] bad = new boolean[n];
            for (int j = 0; j < n - 2; j++) {
                if (s.charAt(j) == 't' && s.charAt(j + 1) == 'w' && s.charAt(j + 2) == 'o') {
                    bad[j] = true;
                    bad[j + 1] = true;
                    bad[j + 2] = true;
                } else if (s.charAt(j) == 'o' && s.charAt(j + 1) == 'n' && s.charAt(j + 2) == 'e') {
                    bad[j] = true;
                    bad[j + 1] = true;
                    bad[j + 2] = true;
                }
            }
            int cnt = 0;
            for (boolean b : bad)
                if (b) cnt++;
            System.out.println(cnt);
            for (int j = 0; j < n; j++)
                if (bad[j]) System.out.print((j + 1) + " ");
            System.out.println();
        }
    }
}
