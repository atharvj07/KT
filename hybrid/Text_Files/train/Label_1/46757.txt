import java.util.*;

public class Main {
    static class Slime {
        int x, y;

        public Slime(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static int n, w, h;
    static int[][] grid;
    static int[] dx = {-1, 0, 1, 0};
    static int[] dy = {0, -1, 0, 1};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        w = sc.nextInt();
        h = sc.nextInt();
        grid = new int[h + 2][w + 2];
        List<Slime> slimes = new ArrayList<>();
        int sx = -1, sy = -1;
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            grid[y][x] = 1;
            slimes.add(new Slime(x, y));
            if (sx == -1) {
                sx = x;
                sy = y;
            }
        }
        int ans = bfs(sx, sy);
        System.out.println(ans);
    }

    static int bfs(int sx, int sy) {
        int dist[][] = new int[h + 2][w + 2];
        for (int i = 0; i <= h + 1; i++) {
            for (int j = 0; j <= w + 1; j++) {
                dist[i][j] = -1;
            }
        }
        Queue<Slime> q = new LinkedList<>();
        q.offer(new Slime(sx, sy));
        dist[sy][sx] = 0;
        while (!q.isEmpty()) {
            Slime slime = q.poll();
            int x = slime.x, y = slime.y;
            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i], ny = y + dy[i];
                if (nx < 1 || nx > w || ny < 1 || ny > h) continue;
                if (grid[ny][nx] == 0 && dist[ny][nx] == -1) {
                    dist[ny][nx] = dist[y][x] + 1;
                    q.offer(new Slime(nx, ny));
                }
            }
        }
        int maxDist = 0;
        for (int i = 0; i <= h + 1; i++) {
            for (int j = 0; j <= w + 1; j++) {
                if (grid[i][j] == 1) {
                    maxDist = Math.max(maxDist, dist[i][j]);
                }
            }
        }
        return maxDist;
    }
}
