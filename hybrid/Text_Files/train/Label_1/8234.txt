#include <bits/stdc++.h>
#define endl '\n'

using namespace std;
typedef long long ll;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  ll t;
  cin >> t;
  while (t--) {
    ll n;
    string s;
    cin >> n >> s;
    ll cnt = 1, i = 0, cur = s[i] - '0';
    while (i < n && s[i] - '0' <= cur) i++, cnt++;
    if (i == n) {
      cout << "NO" << endl;
      continue;
    }
    string div = s.substr(0, cnt);
    cout << "YES" << endl;
    cout << cnt << endl;
    cout << div << " ";
    for (ll j = cnt; j < n; j++) {
      div = s.substr(j, n - j);
      cout << div << " ";
    }
    cout << endl;
  }
  return 0;
}
