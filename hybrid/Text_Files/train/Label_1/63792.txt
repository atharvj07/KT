from collections import defaultdict
for k in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    b=defaultdict(list)
    for x in a:
        b[x].append(len(b[x])%2)
    if n!=len(b):
        print(-1)
        continue
    ans=0
    for k in b:
        if len(b[k])%2:
            ans+=1
    print(ans//2)
