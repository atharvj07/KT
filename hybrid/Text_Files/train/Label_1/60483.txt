from collections import Counter

def find_min_palindromes(s):
    freq = Counter(s)
    odd_count = sum(1 for count in freq.values() if count % 2 != 0)
    if odd_count == 0:
        palindrome_length = 1
    else:
        palindrome_length = odd_count

    parts = [''] * palindrome_length
    odd_chars = [char for char, count in freq.items() if count % 2 != 0]
    even_chars = [char for char, count in freq.items() if count % 2 == 0]

    for char in odd_chars:
        index = odd_chars.index(char)
        parts[index] += char * (freq[char] // palindrome_length)
        freq[char] %= palindrome_length

    for char in even_chars:
        count = freq[char] // palindrome_length
        half_count = count // 2
        for i in range(palindrome_length):
            parts[i] = char * half_count + parts[i] + char * half_count

    if odd_count > 0:
        for char in odd_chars:
            for i in range(freq[char]):
                parts[i % palindrome_length] = parts[i % palindrome_length][:len(parts[i % palindrome_length]) // 2] + char + parts[i % palindrome_length][len(parts[i % palindrome_length]) // 2:]

    return len(parts), ' '.join(parts)

n = int(input())
s = input()

k, palindromes = find_min_palindromes(s)
print(k)
print(palindromes)
