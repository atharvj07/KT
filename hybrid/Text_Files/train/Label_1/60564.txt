mod = int(1e9) + 7

n, q = map(int, input().split())
s = input()
pref = [0] * (n + 1)
for i in range(n):
    pref[i + 1] = pref[i] + (s[i] == '1')
dp = [0] * (n + 1)
for l in range(n - 1, -1, -1):
    dp[l] = max(pref[r + 1] - pref[l] + dp[r + 1] for r in range(l, n))
for i in range(q):
    l, r = map(int, input().split())
    print(dp[l - 1])
