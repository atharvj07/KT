#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (n); ++i)

using namespace std;
using ll = long long int;

int main() {
  int n, x, y;
  cin >> n >> x >> y;
  vector<ll> a(n);
  rep(i, n) cin >> a[i];
  set<ll> s(a.begin(), a.end());
  if (x >= y) {
    for (auto v : s) cout << v << " ";
    cout << endl;
  } else {
    rep(i, n) { cout << (a[i] + x + y - 1) / (x + y) * (x + y) + (y - x) << " "; }
    cout << endl;
  }
  return 0;
}
