#include <iostream>
#include <vector>

using namespace std;

int sum(const vector<int> &v, int i) {
  int res = 0;
  for (int j = 0; j <= i; j++) res += v[j];
  return res;
}

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> v(n);
  for (auto &i : v) cin >> i;
  if (v[0] >= k) {
    cout << "0\n";
    return 0;
  }
  int i = 0;
  while (i < n) {
    int l = 0, r = n - 1, mid;
    while (l <= r) {
      mid = (l + r) / 2;
      if (sum(v, mid) >= k) r = mid - 1; else l = mid + 1;
    }
    if (l == n) l--;
    if (sum(v, l) >= k) {
      cout << i + 1 << "\n";
      return 0;
    }
    for (int j = 0; j <= l; j++) v[j] += v[j + 1];
    v.pop_back();
    n--;
    i++;
  }
  cout << "-1\n";
  return 0;
}
