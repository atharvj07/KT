#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<long long> cap(n);
    for (int i = 0; i < n; i++) { std::cin >> cap[i]; }
    int m;
    std::cin >> m;
    std::vector<long long> amt(n);
    for (int i = 0; i < m; i++) {
        int q;
        std::cin >> q;
        if (q == 1) {
            int p;
            long long x;
            std::cin >> p >> x;
            p--;
            long long over = x;
            while (over && p < n - 1) {
                amt[p] += cap[p] - amt[p];
                over -= cap[p] - amt[p];
                amt[p + 1] += over;
                p++;
            }
            amt[p] += std::min(cap[p], x);
        } else {
            int p;
            std::cin >> p;
            p--;
            while (p >= 1 && amt[p] == cap[p]) { p--; }
            std::cout << amt[p] << '\n';
        }
    }
    return 0;
}
