#include <iostream>
#include <string>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int ans = 0;
    bool flag = false;
    for (char c : s) {
      if (c == '1') {
        flag = true;
      } else if (flag) {
        ans++;
      }
    }
    cout << ans << endl;
  }
}
