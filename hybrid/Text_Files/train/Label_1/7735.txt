#include <bits/stdc++.h>
using namespace std;
long long n, x, ans, a[1000001], l[1000001], r[1000001], freq[1000001];
int main() {
  cin >> n >> x;
  for (long long i = 1; i <= n; i++) {
    cin >> a[i];
    freq[a[i]]++;
  }
  for (long long i = 1; i <= x; i++) {
    freq[i] += freq[i - 1];
    l[i] = freq[i - 1] + 1;
    r[i] = l[i] - 1 + freq[i];
  }
  for (long long i = 1; i <= x; i++) {
    for (long long j = 1; j <= x; j++) {
      if (i <= j && r[j] - l[i] + 1 >= j - i + 1) ans++;
    }
  }
  cout << ans;
  return 0;
}
