import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int[] d = new int[n];
        for (int i = 1; i < n; i++) {
            d[i] = a[i] - a[i - 1];
        }
        int m = 1;
        int p = 0;
        for (int i = 1; i < n; i++) {
            if (d[i] == d[p]) {
                m++;
            } else {
                if (m > 1) {
                    int t = gcd(m);
                    if (t > 1) {
                        m = m / t + 1;
                    }
                }
                p = i;
                m = 1;
            }
        }
        if (m > 1) {
            int t = gcd(m);
            if (t > 1) {
                m = m / t + 1;
            }
        }
        System.out.println(m);
        int[] b = new int[m];
        int j = 0;
        for (int i = n - 1; i >= 0 && j < m; i--) {
            if (i < n - 1 && d[i] == d[i + 1]) {
                continue;
            } else {
                b[j++] = a[i];
            }
        }
        for (int i = 0; i < m; i++) {
            System.out.print(b[i] + " ");
        }
    }

    static int gcd(int a) {
        while (a != 1) {
            if (a % 2 == 0) {
                a /= 2;
            } else {
                return 1;
            }
        }
        return 2;
    }
}
