import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int q = sc.nextInt();
            String s = sc.next();
            while (q-- > 0) {
                int l = sc.nextInt() - 1;
                int r = sc.nextInt() - 1;
                boolean flag = false;

                for (int i = l; i < r; i++)
                    for (int j = i + 2; j <= r; j++)
                        if (s.charAt(i) == s.charAt(j)) flag = true;
                if (flag) System.out.println("YES");
                else System.out.println("NO");
            }
        }
    }
}
