#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const int maxn = 3e5 + 10;

int main() {
  int n; cin >> n;
  vector<int> color(n);
  vector<int> last(maxn);
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int x; cin >> x; x--;
    color[i] = x;
    if (last[x] != -1) ans = max(ans, i - last[x]);
    last[x] = i;
  }
  cout << ans << '\n';
  return 0;
}
