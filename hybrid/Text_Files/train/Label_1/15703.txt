#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, d;
    cin >> n >> d;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 0, j = n - 1; i < j; i++, j--) {
        if (a[j] - a[i] > d) {
            ans++, j++;
        }
    }
    cout << ans << endl;
}
