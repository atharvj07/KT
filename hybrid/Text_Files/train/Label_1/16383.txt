#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007
ll pw(ll a, ll b){
    if(b==0)return 1;
    if(b==1)return a;
    ll x=pw(a,b/2);
    x=(x*x)%mod;
    if(b%2)x=(x*a)%mod;
    return x;
}
int32_t main(){
    ll x,k;cin>>x>>k;
    cout<<(x*pw(2,k))%mod<<endl;
    return 0;
}
