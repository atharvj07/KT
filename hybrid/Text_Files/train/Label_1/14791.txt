#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;

int n, q, a[N], b[N];

void solve()
{
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    cin >> q;
    for (int i = 1; i <= q; i++)
    {
        int type, x, y;
        cin >> type >> x;
        if (type == 1)
        {
            cin >> y;
            b[x] = y;
        }
        else
        {
            for (int j = 1; j <= n; j++) b[j] = max(b[j], x);
        }
    }
    for (int i = 1; i <= n; i++)
    {
        cout << b[i] << " ";
    }
}
int main()
{
    solve();
}
