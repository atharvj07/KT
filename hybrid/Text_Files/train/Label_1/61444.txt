import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    c = list(map(int, input().split()))
    if sorted(a) == sorted(b):
        vis = [False] * (n + 1)
        ans = []
        for i in range(m):
            for j in range(n):
                if a[j] == c[i] and not vis[b[j]]:
                    ans.append(j + 1)
                    vis[b[j]] = True
                    break
        if len(ans) == m:
            print("YES")
            print(*ans)
        else:
            print("NO")
    else:
        print("NO")
