#include <bits/stdc++.h>
using namespace std;
char a[1005];
int n,fr[30];
bool check()
{
  for(int i=0;i<26;i++) if(fr[i]>n/2) return 0;
  return 1;
}
int main()
{
  cin>>n;
  cin>>a;
  for(int i=0;i<n;i++) fr[a[i]-'a']++;
  if(!check())
  {
    cout<<"NO"<<endl;
    return 0;
  }
  cout<<"YES"<<endl;
  int l=0,r=0;
  while(r<n)
  {
    fr[a[r]-'a']++;
    while(!check() && l<r) fr[a[l]-'a']--,l++;
    r++;
    if(check())
    {
      for(int i=l;i<r;i++) cout<<a[i];
      cout<<endl;
      return 0;
    }
  }
}
