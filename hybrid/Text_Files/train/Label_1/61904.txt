a=input();b=input();
if len(a)!=len(b):print('NO')
else:
    p0=[-1]*(1<<11);p0[0]=0;p1=[-1]*(1<<11);p1[0]=0;
    for i in range(0,1<<10):
        p0[i+1]=p0[i];p1[i+1]=p1[i];
        if (i&(1<<10))==0:
            if (i&(1<<9))==0:p1[i+1]=p1[i]+1
            else:p0[i+1]=p0[i]+1
    st=0
    for x in a:
        st+=(1<<(ord(x)-ord('0')))
    for x in b:
        st^=(1<<(ord(x)-ord('0')))
    if st!=(1<<10):print('NO')
    else:
        if p0[st]==p1[st]:print('YES')
        else:print('NO')
