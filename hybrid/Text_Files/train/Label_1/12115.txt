#include <iostream>
#include <vector>

using namespace std;

int main() {
  int f, i, t;
  cin >> f >> i >> t;
  vector<string> friends(f);
  for (int j = 0; j < f; j++) cin >> friends[j];
  int ans = 0;
  for (int j = 0; j < i; j++) {
    int count = 0;
    for (int k = 0; k < f; k++) {
      if (friends[k][j] == 'Y') count++;
    }
    if (count >= t) ans++;
  }
  cout << ans << endl;
  return 0;
}
