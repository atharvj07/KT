#include<iostream>
#include<vector>
using namespace std;
bool ok=true;
void dfs(vector<vector<int>>&g,int v,int p,int cnt3,int k){
    if(cnt3>3)ok=false;
    if(g[v].size()==1&&v!=1){
        if(k)dfs(g,g[v][0],v,cnt3-1,k-1);
        else ok=false;
    }
    for(int u:g[v])if(u!=p)dfs(g,u,v,cnt3+(g[v].size()==1),k);
}
int main(){
    int n,k;
    cin>>n>>k;
    vector<vector<int>>g(n+1);
    for(int i=0;i<n-1;i++){
        int u,v;
        cin>>u>>v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(g,1,0,1,k);
    cout<<(ok?"Yes":"No");
    return 0;
}
