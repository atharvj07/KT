#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < n; i++)
using namespace std;
int main() {
  string t;
  cin >> t;
  if (2 * t.size() != t.length() + count(t.begin(), t.end(), 'a'))
    cout << ":(" << endl;
  else {
    string s;
    rep(i, t.size()) {
      if (t[i] != 'a') s += t[i];
      else s += t[i + t.size()];
    }
    cout << s << endl;
  }
}
