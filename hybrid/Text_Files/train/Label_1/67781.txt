import collections
MAX_L = 100000
S = [input() for _ in range(int(input()))]
F = collections.defaultdict(list)
for i, s in enumerate(S):
    for j in range(len(s) + 1):
        F[s[:j]].append(i)
res = 0
for i, s in enumerate(S):
    L, R = [], []
    for x in F[s[::-1]]:
        if x == i:
            res += len(L) + len(R) - 1
        else:
            L.append(x)
            R = [x] + R
    if s == s[::-1]:
        res -= 1
print(res)
