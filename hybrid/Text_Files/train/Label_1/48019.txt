import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            int[] arr = new int[n];
            for (int j = 0; j < n; j++) {
                arr[j] = sc.nextInt();
            }
            Arrays.sort(arr);
            int sum = 0;
            for (int j = n - 1; j >= 0; j--) {
                sum += arr[j];
                if (sum >= arr[j] * (n - j)) {
                    System.out.println(arr[j]);
                    break;
                }
            }
        }
        sc.close();
    }
}
