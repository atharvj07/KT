#include <bits/stdc++.h>
using namespace std;
#define pb push_back
typedef long long ll;
const int N=2e5+10;
vector<int>v[N];
int a[N],dp[N][30];
int solve(int idx,int rem,int lim){
    if(idx==lim){
        if(rem>0)return a[idx];
        else return INT_MAX;
    }
    int &ret=dp[idx][rem];
    if(~ret)return ret;
    ret=INT_MAX;
    ret=min(ret,solve(idx+1,rem,lim));
    ret=min(ret,solve(idx+1,rem-1,lim)+a[idx]);
    return ret;
}
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n,k;
    cin>>n>>k;
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    memset(dp,-1,sizeof(dp));
    int ans=solve(1,k,n);
    cout<<ans<<endl;
    return 0;
}
