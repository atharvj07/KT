#include <iostream>
#include <vector>

int main() {
  int t; std::cin >> t;
  while (t--) {
    int n, m; std::cin >> n >> m;
    std::vector<int> a(n), x(m);
    for (int i = 0; i < n; ++i) std::cin >> a[i];
    for (int i = 0; i < m; ++i) std::cin >> x[i];

    int sum = 0;
    std::vector<int> ans(m);
    for (int i = 0; i < m; ++i) {
      int j = 0;
      sum = 0;
      while (sum < x[i]) {
        sum += a[j];
        j = (j + 1) % n;
        ans[i]++;
        if (j == 0) {
          ans[i] = -1;
          break;
        }
      }
    }

    for (int u: ans) std::cout << u << ' ';
    std::cout << '\n';
  }
  return 0;
}
