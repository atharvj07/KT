import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        long[][] a = new long[k][];
        for (int i = 0; i < k; i++) {
            int n = sc.nextInt();
            a[i] = new long[n];
            for (int j = 0; j < n; j++) {
                a[i][j] = sc.nextLong();
            }
        }

        long sum = 0;
        for (long[] b : a) {
            sum += b[0];
        }

        boolean ok = false;
        for (int i = 0; i < k; i++) {
            for (int j = i + 1; j < k; j++) {
                if (sum - a[i][0] - a[j][0] == a[i][0] + a[j][0]) {
                    System.out.println("YES");
                    System.out.println((i + 1) + " " + 1);
                    System.out.println((j + 1) + " " + 1);
                    ok = true;
                    break;
                }
                for (int x = 1; x < a[i].length; x++) {
                    for (int y = 1; y < a[j].length; y++) {
                        if (sum - a[i][x] - a[j][y] == a[i][x] + a[j][y]) {
                            System.out.println("YES");
                            System.out.println((i + 1) + " " + (x + 1));
                            System.out.println((j + 1) + " " + (y + 1));
                            ok = true;
                            break;
                        }
                    }
                }
            }
        }

        if (!ok) {
            System.out.println("NO");
        }
    }
}
