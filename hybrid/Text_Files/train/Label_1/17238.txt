#include <bits/stdc++.h>
using namespace std;
#define int long long

int32_t main() {
    int n, k;
    cin >> n >> k;
    vector<int> v(n);
    map<int, int> f, s;
    for (int i = 0; i < n; i++) {
        cin >> v[i];
        f[v[i]]++;
    }
    int ans = 0;
    for (auto it : v) {
        f[it]--;
        s[it / k]++;
        ans += (f[it / k] * s[it * k]);
    }
    cout << ans << endl;
    return 0;
}
