#include <iostream>
using namespace std;
#define MOD 1000000007
long long pow(long long a, long long b) {
    long long ans = 1;
    while (b > 0) {
        if (b % 2 == 0) {
            a = (a * a) % MOD;
            b /= 2;
        } else {
            ans = (ans * a) % MOD;
            b--;
        }
    }
    return ans;
}
int main() {
    long long n, k;
    cin >> n >> k;
    cout << (pow(2, n + k - 1) - 1 + MOD) % MOD;
}
