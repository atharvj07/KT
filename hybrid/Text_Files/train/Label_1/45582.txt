import java.util.Scanner;

class Main {
    static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), m = scanner.nextInt();
        int[] deg = new int[n + 1];
        int[][] grid = new int[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt(), b = scanner.nextInt(), c = scanner.nextInt();
            grid[a][b] = grid[b][a] = c;
            deg[a]++;
            deg[b]++;
        }
        boolean flag = false;
        for (int i = 1; i <= n; i++) {
            if (deg[i] % 2 == 1) {
                flag = true;
                break;
            }
        }
        if (flag) {
            System.out.println(0);
            return;
        }
        long[][] dp = new long[n + 1][4];
        for (int i = 0; i <= n; i++) {
            dp[i][3] = 1;
        }
        for (int i = n - 1; i >= 1; i--) {
            for (int j = 0; j < 3; j++) {
                int cur = j + 2;
                for (int k = i + 1; k <= n; k++) {
                    if (grid[i][k] == 1) {
                        dp[i][j] = (dp[i][j] + dp[k][cur]) % MOD;
                    } else if (grid[i][k] == 0) {
                        dp[i][j] = (dp[i][j] + (dp[k][cur] * 2) % MOD) % MOD;
                    }
                }
            }
        }
        System.out.println(dp[1][0]);
    }
}
