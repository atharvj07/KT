#include <iostream>
using namespace std;

int main() {
  int w = 0, b = 0;
  for (int i = 0; i < 8; i++) {
    for (int j = 0; j < 8; j++) {
      char c = getchar();
      if (c == 'Q') w += 9;
      else if (c == 'R') w += 5;
      else if (c == 'B') w += 3;
      else if (c == 'N') w += 3;
      else if (c == 'P') w += 1;
      else if (c == 'q') b += 9;
      else if (c == 'r') b += 5;
      else if (c == 'b') b += 3;
      else if (c == 'n') b += 3;
      else if (c == 'p') b += 1;
    }
    getchar();
  }
  if (w > b)
    cout << "White";
  else if (w < b)
    cout << "Black";
  else
    cout << "Draw";
  return 0;
}
