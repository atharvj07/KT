def permutation(n, k):
    a = list(range(1, n + 1))
    if k > (n - 1) // 2:
        return -1
    if k == 0:
        return a
    i = 0
    while i < k:
        if i == 0:
            a[0], a[1] = a[1], a[0]
            i += 1
        elif i == k - 1:
            a[n - 1], a[n - 2] = a[n - 2], a[n - 1]
            i += 1
        else:
            a[i], a[i + 1] = a[i + 1], a[i]
            i += 2
    return a

t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    res = permutation(n, k)
    print(*res) if res != -1 else print(-1)
