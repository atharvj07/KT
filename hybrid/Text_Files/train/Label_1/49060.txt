import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long f = in.nextLong();
        long T = in.nextLong();
        long t0 = in.nextLong();
        long a1 = in.nextLong();
        long t1 = in.nextLong();
        long p1 = in.nextLong();
        long a2 = in.nextLong();
        long t2 = in.nextLong();
        long p2 = in.nextLong();

        if (t0 * f <= T) {
            System.out.println(0);
            return;
        }

        long ans = Long.MAX_VALUE;
        for (long i = 0; i * a1 <= f; i++) {
            long rem = f - i * a1;
            long cost = i * p1;
            long time = i * t1 + Math.max(0, rem * t0 - T);
            if (time <= T) {
                ans = Math.min(ans, cost);
            }

            if (t2 < t0) {
                long j = (rem + a2 - 1) / a2;
                cost += j * p2;
                time = i * t1 + j * t2 + Math.max(0, (rem - j * a2) * t0 - T);
                if (time <= T) {
                    ans = Math.min(ans, cost);
                }
            }
        }

        if (ans == Long.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }
}
