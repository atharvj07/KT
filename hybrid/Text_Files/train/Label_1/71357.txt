def find_inversions(p):
    inversions = 0
    for i in range(len(p)):
        for j in range(i + 1, len(p)):
            if p[i] > p[j]:
                inversions += 1
    return inversions

n, m = map(int, input().strip().split())
p = list(map(int, input().strip().split()))
moves = []
for _ in range(m):
    moves.append(tuple(map(int, input().strip().split())))

inversions = find_inversions(p)

for a, b in moves:
    if p[a-1] > p[b-1]:
        inversions -= 1
    else:
        inversions += 1
    inversions *= 0.5  # Because there's a 50% chance the move is actually performed

print("{:.9f}".format(inversions))
