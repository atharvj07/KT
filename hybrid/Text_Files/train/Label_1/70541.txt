t = int(input())
for i in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    count = 0
    for j in range(n):
        if a[j] < j + 1:
            k = j + 1
            while k < n:
                if a[k] * a[j] == k + j:
                    count += 1
                k += 1
    print(count)
