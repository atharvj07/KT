#include <iostream>
#include <string>
#include <vector>
#include <stack>
#include <queue>

using namespace std;

int main() {
  string name;
  getline(cin, name);

  int n = name.size();
  vector<int> values(n, -1);
  stack<int> s;
  queue<int> q;
  int curr = 0;
  for (int i = 0; i < n; i++) {
    if (name[i] == '_') {
      s.push(curr);
    } else if (name[i] == '-') {
      if (!s.empty()) {
        q.push(s.top());
        s.pop();
      } else {
        values[i] = 0;
        curr++;
      }
    } else if (name[i] == '*') {
      int a = q.front();
      q.pop();
      int b = q.front();
      q.pop();
      values[a] = values[b] = curr++;
    } else if (name[i] == '+') {
      if (!s.empty()) {
        q.push(s.top());
        s.pop();
      } else {
        q.push(curr++);
      }
    }
  }

  while (!s.empty()) {
    q.push(s.top());
    s.pop();
  }

  while (!q.empty()) {
    int idx = q.front();
    q.pop();
    values[idx] = curr++;
  }

  for (int i = 0; i < n; i++) {
    if (values[i] == -1) {
      cout << "false" << endl;
      return 0;
    }
  }

  for (int i = 0; i < n; i++) {
    cout << values[i];
  }
  cout << endl;

  return 0;
}
