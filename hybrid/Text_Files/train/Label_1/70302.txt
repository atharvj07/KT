n, m, k = map(int, input().split())
p = list(map(int, input().split()))

operations = 0
discarded = 0

while m > 0:
    currentPage = (p[0] - discarded - 1) // k + 1
    count = 0
    for item in p:
        if (item - discarded - 1) // k + 1 == currentPage:
            count += 1
        else:
            break
    p = p[count:]
    m -= count
    discarded += count
    operations += 1

print(operations)
