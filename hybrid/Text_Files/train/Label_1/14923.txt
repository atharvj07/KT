#include <vector>
#include <iostream>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> x(n), c(n);
    int a;
    cin >> a;
    for (int &i : x) cin >> i;
    for (int &i : c) cin >> i;
    int pre = k;
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        if (pre >= x[i]) {
            pre = pre + x[i];
        } else {
            int need = (x[i] - pre + a - 1) / a;
            if (i == 0) {
                if (need * c[i] < x[i] - k) {
                    pre = x[i];
                    ans += need * c[i];
                } else {
                    ans = -1;
                    break;
                }
            } else {
                if (need * c[i] < x[i] - pre) {
                    pre = x[i];
                    ans += need * c[i];
                } else {
                    ans = -1;
                    break;
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
