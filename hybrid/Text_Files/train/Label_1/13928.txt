#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n, k, x; cin >> n >> k >> x;
    long long s = 0;
    int a[101];
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        s += a[i];
    }
    sort(a + 1, a + 1 + n);
    for (int i = n; i > n - k; i--) {
        s = s - a[i] + x;
    }
    cout << s << endl;
    return 0;
}
