#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mxN = 1e18;
ll get_min(ll a, ll b) {
    if (a == b)return 0;
    if (b % a != 0)return -1;
    ll x = b / a;
    ll ans = 0;
    while (x != 1) {
        if (x % 8 == 0) {
            x /= 8; ans++;
        }
        else if (x % 4 == 0) {
            x /= 4; ans++;
        }
        else if (x % 2 == 0) {
            x /= 2; ans++;
        }
        else {
            return -1;
        }
    }
    return ans;
}
int main() {
    ll t; cin >> t;
    while (t--) {
        ll a, b; cin >> a >> b;
        cout << get_min(a, b) << '\n';
    }
}
