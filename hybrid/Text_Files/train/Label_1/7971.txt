#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    long long sum = accumulate(a.begin(), a.begin() + k + 1, 0LL);
    long long ans = sum;
    if (k % 2)
      ans -= *a.begin();
    else
      ans += *a.rbegin();
    cout << ans << "\n";
  }
  return 0;
}
