#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
int32_t main() {
  int n, x, sum;
  cin >> n;
  vector<int> p(n);
  string s;
  bool f;
  for (int i = 0; i < n; i++) {
    cin >> p[i];
    getline(cin, s);
    stringstream ss(s);
    sum = 0, f = 0;
    while (!ss.eof()) {
      ss >> s;
      string ss1;
      for (char c : s) {
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' ||
            c == 'y')
          sum++, f = 1;
        else
          ss1.push_back(c);
      }
      sum += f;
      f = 0;
    }
    if (sum != p[i]) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
