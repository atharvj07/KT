def parse_line(line):
  (name, operation) = line.split(":=")
  name = name.strip()
  operation = operation.strip()
  if operation.startswith("?"):
    return None
  elif "AND" in operation:
    (op1, op2) = operation.split("AND")
    return (name, "AND", op1.strip(), op2.strip())
  elif "OR" in operation:
    (op1, op2) = operation.split("OR")
    return (name, "OR", op1.strip(), op2.strip())
  elif "XOR" in operation:
    (op1, op2) = operation.split("XOR")
    return (name, "XOR", op1.strip(), op2.strip())
  else:
    return (name, "CONSTANT", operation)

def evaluate_line(line, variables, value):
  (name, operation, op1, op2) = line
  if operation == "CONSTANT":
    variables[name] = int(op1, 2)
  elif operation == "AND":
    variables[name] = variables[op1] & variables[op2]
  elif operation == "OR":
    variables[name] = variables[op1] | variables[op2]
  elif operation == "XOR":
    variables[name] = variables[op1] ^ variables[op2]

def dfs(variables, value, minimum, maximum):
  if len(variables) == 0:
    if value < minimum:
      minimum = value
    if value > maximum:
      maximum = value
    return (minimum, maximum)
  (name, line) = variables.popitem()
  if line is None:
    (minimum, maximum) = dfs(variables, value, minimum, maximum)
    variables[name] = line
    (minimum, maximum) = dfs(variables, value, minimum, maximum)
  else:
    evaluate_line(line, variables, value)
    (minimum, maximum) = dfs(variables, value, minimum, maximum)
    del variables[name]
  return (minimum, maximum)

(n, m) = map(int, input().split())
variables = {}
for i in range(n):
  line = parse_line(input())
  variables[line[0]] = line
value = 0
(minimum, maximum) = dfs(variables, value, 2**m - 1, 0)
print(bin(minimum)[2:].zfill(m))
print(bin(maximum)[2:].zfill(m))
