from collections import defaultdict

def dfs(node, parent, graph, subtree_sum, total_sum, values):
    subtree_sum[node] = values[node - 1]
    for child in graph[node]:
        if child != parent:
            dfs(child, node, graph, subtree_sum, total_sum, values)
            subtree_sum[node] += subtree_sum[child]
    if parent != -1:
        total_sum[0] = max(total_sum[0], total_sum[0] + subtree_sum[node])

def solve(n, values, edges):
    if n == 1:
        return 0
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    subtree_sum = [0] * (n + 1)
    total_sum = [sum(values)]
    dfs(1, -1, graph, subtree_sum, total_sum, values)
    return total_sum[0]

n = int(input().strip())
values = list(map(int, input().strip().split()))
edges = [list(map(int, input().strip().split())) for _ in range(n - 1)]
print(solve(n, values, edges))
