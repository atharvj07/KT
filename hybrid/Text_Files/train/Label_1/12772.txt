#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n % 3 != 0) {
      cout << -1 << endl;
    } else {
      int tmp = n / 3;
      for (int a = 0; a <= tmp; a++) {
        for (int b = 0; b <= tmp; b++) {
          int c = tmp - a - b;
          if (a * 3 + b * 5 + c * 7 == n) {
            cout << a << ' ' << b << ' ' << c << endl;
            goto out;
          }
        }
      }
      cout << -1 << endl;
    out:;
    }
  }
  return 0;
}
