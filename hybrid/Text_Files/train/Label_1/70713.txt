n = int(input())
w = list(map(int, input().split()))
a = list(input())
t = ['']*n
e = []
for i in range(2*n):
    if a[i]=='0':
        t[w.index(min(w))] = '1'
        w[w.index(min(w))]=10**9+i
    else:
        e.append(w.index(max(w)))
        w[w.index(max(w))]=0
for i in e:
    t[i] = '1'
print(*[i+1 for i in range(n) if t[i]=='1'])
