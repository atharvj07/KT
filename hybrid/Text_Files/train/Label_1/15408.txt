#include <iostream>
#include <vector>
#include <map>
using namespace std;

const int MOD = 1e6;

int main() {
    int q, m; cin >> q >> m;
    vector<pair<int, int>> diff(q);
    for(int i = 0; i < q; i++) {
        int k, a, b; cin >> k >> a >> b;
        diff[i] = {k, (a + b) % MOD + 1};
    }
    
    map<int, pair<int, int>> spells;
    bool win[q];
    
    int last_win = -1;
    for(int i = 0; i < q; i++) {
        if(diff[i].first == 1) {
            spells[diff[i].second % MOD + 1].first += diff[i].second;
            spells[diff[i].second % MOD + 1].second -= diff[i].second;
        } else {
            int t = diff[i].second;
            bool flag = false;
            for(auto p : spells) {
                int d = p.first, mp = p.second.first, mc = p.second.second;
                if(t > mp || m < mc) continue;
                int cnt = (t - (m - mc)) / mp;
                if(cnt >= 1) {
                    m -= mc + mp * (cnt - 1);
                    t -= d * cnt;
                    if(t <= 0) {
                        flag = true;
                        break;
                    }
                }
            }
            
            if(flag) {
                win[i] = true;
                last_win = i;
            } else {
                win[i] = false;
            }
        }
    }
    
    for(int i = 0; i < q; i++) {
        if(diff[i].first == 2) {
            cout << (win[i] || i == last_win ? "YES" : "NO") << endl;
        }
    }
}
