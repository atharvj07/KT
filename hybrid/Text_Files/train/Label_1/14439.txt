#include <iostream>
#include <vector>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n;
        std::cin >> n;
        std::vector<int> a(n);
        for (auto& x : a) std::cin >> x;
        int mx = a[0];
        int ind = 0;
        for (int i = 1; i < n; i++) {
            if (a[i] > mx) {
                mx = a[i];
                ind = i;
            }
        }
        bool ok = true;
        for (int i = 0; i < ind; i++) {
            if (a[i] > a[i + 1]) {
                ok = false;
                break;
            }
        }
        for (int i = ind; i < n - 1; i++) {
            if (a[i] > a[i + 1]) {
                ok = false;
                break;
            }
        }
        if (ok)
            std::cout << ind + 1 << std::endl;
        else
            std::cout << -1 << std::endl;
    }
}
