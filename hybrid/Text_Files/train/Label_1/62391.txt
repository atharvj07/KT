from collections import defaultdict, deque

n, q = map(int, input().split())
a = list(map(int, input().split()))
t = list(map(int, input().split()))
d = defaultdict(deque)
ans = []
for i in range(n - 1, -1, -1):
    d[a[i]].append(i + 1)
for color in t:
    index = d[color].popleft()
    ans.append(index)
    d[color].append(index)
print(*ans)
