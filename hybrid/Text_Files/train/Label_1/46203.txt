import java.util.*;

public class Main {

    static int n, m;
    static int[][] grid;
    static boolean[][] visited;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            n = sc.nextInt();
            m = sc.nextInt();
            grid = new int[n][m];
            visited = new boolean[n][m];
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < m; k++) {
                    grid[j][k] = sc.nextInt();
                    if (grid[j][k] == 1) {
                        visited[j][k] = true;
                    }
                }
            }
            if (canWin(0, true)) {
                System.out.println("Ashish");
            } else {
                System.out.println("Vivek");
            }
        }
    }

    static boolean canWin(int score, boolean ashishTurn) {
        if (!ashishTurn) {
            score += 1;
        }
        if (score >= n * m) {
            return !ashishTurn;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!visited[i][j]) {
                    if (isValid(i, j)) {
                        visited[i][j] = true;
                        if (canWin(score, !ashishTurn)) {
                            visited[i][j] = false;
                            return !ashishTurn;
                        }
                        visited[i][j] = false;
                    }
                }
            }
        }
        return ashishTurn;
    }

    static boolean isValid(int x, int y) {
        for (int i = 0; i < m; i++) {
            if (visited[x][i]) {
                return false;
            }
        }
        for (int i = 0; i < n; i++) {
            if (visited[i][y]) {
                return false;
            }
        }
        return true;
    }
}
