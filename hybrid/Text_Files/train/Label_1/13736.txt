#include <bits/stdc++.h>
using namespace std;
#define ll long long

const int MAXN = 2e5 + 5;

vector<int> adj[MAXN];
int deg[MAXN], vis[MAXN];

int dfs(int u) {
    vis[u] = true;
    for (auto v : adj[u]) {
        if (!vis[v]) deg[v] = dfs(v);
        deg[u] += deg[v];
    }
    return deg[u];
}

int main() {
    int n;
    cin >> n;
    int a[n + 1];
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        adj[a[i]].push_back(i);
    }
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) {
            dfs(i);
            ans += 1LL * deg[i] * (deg[i] - 1) / 2;
        }
    }
    cout << ans << "\n";
}
