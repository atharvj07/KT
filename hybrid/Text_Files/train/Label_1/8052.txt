#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll n,w[100005];
vector< vector<ll> >a(100005);
ll fun(ll ver,ll pre,ll sum,ll lv){
if(lv==1) return sum;
ll mx=0;
for(auto u:a[ver]){
    if(u==pre) continue;
    mx=max(mx,fun(u,ver,sum+1,lv-1));
}
return mx;
}
int main(){
    ll t;
    cin>>t;
    while(t--){
        cin>>n;
        for(ll i=0;i<n;i++) cin>>w[i];
        for(ll i=0;i<n;i++){
            a[w[i]].push_back(i);
        }
        ll ans=0;
        vector<ll>cnt(n+1);
        for(ll i=0;i<n;i++){
            if(cnt[w[i]]) ans=max(ans,i-cnt[w[i]]);
            cnt[w[i]]=i;
        }
        for(ll i=0;i<n;i++) a[w[i]].clear();
        cout<<ans<<endl;
    }
    return 0;
}
