#include <iostream>
#include <string>
using namespace std;
int main() {
  int n, p;
  cin >> n >> p;
  string s;
  cin >> s;
  int sm = p;
  int pos = -1;
  for (int i = n - 2; i >= 0; i--) {
    if (s[i] == s[i + 1]) continue;
    if (s[i] + 1 < 'a' + sm) {
      pos = i;
      sm = s[i] - 'a' + 1;
    }
  }
  if (pos == -1) {
    cout << "NO";
  } else {
    s[pos]++;
    for (int i = pos + 1; i < n; i++) s[i] = 'a';
    cout << s;
  }
  return 0;
}
