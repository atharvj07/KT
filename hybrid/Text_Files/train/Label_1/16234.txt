#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main()
{
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    int ans = a[n-1];
    for (int i = n-2; i >= 0; i--)
    {
        if (ans > a[i]) ans--;
        else break;
    }

    cout << ans << endl;
    return 0;
}
