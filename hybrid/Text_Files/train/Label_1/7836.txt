#include <iostream>
#include <vector>
#include <algorithm>

std::vector<std::pair<char, int> > v;

int main() {
  int n;
  std::cin >> n;
  for (int i = 0; i < n; ++i) {
    char c;
    int x;
    std::cin >> c >> x;
    v.push_back(std::make_pair(c, x));
  }
  std::sort(v.begin(), v.end());
  int f = -1, g = -1;
  for (int i = 0; i < n; ++i) {
    if (f == -1 && v[i].second == 1) f = i;
    if (g == -1 && v[i].second == 0) g = i;
  }
  std::cout << (f == -1 ? 0 : 1) + (g == -1 ? 0 : 1) << "\n";
  if (f != -1) std::cout << v[f].first << " " << v[f].second << "\n";
  if (g != -1) std::cout << v[g].first << " " << v[g].second << "\n";
  return 0;
}
