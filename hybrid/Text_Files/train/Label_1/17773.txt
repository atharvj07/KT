#include <bits/stdc++.h>
using namespace std;

int main() {
  int A, B, n;
  cin >> A >> B >> n;
  while (n--) {
    int l, t, m;
    cin >> l >> t >> m;
    int x = (A - 1) / B + 1;
    if (x > t * m || x > l)
      cout << -1 << endl;
    else
      cout << l + t * m - 1 << endl;
  }
  return 0;
}
