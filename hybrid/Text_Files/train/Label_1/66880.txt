home_team = input().strip()
away_team = input().strip()
n = int(input().strip())
cards = {}
red_cards = []

for _ in range(n):
    minute, team, player_num, card_type = input().split()
    minute = int(minute)
    player = (home_team if team == 'h' else away_team, player_num)
    
    if player in cards:
        if card_type == 'y':
            if cards[player]['y'] == 1:
                cards[player]['r'] = 1
                red_cards.append((player[0], player[1], minute))
        elif card_type == 'r':
            if cards[player]['r'] == 0:
                cards[player]['r'] = 1
                red_cards.append((player[0], player[1], minute))
    else:
        cards[player] = {'y': 0, 'r': 0}
        if card_type == 'y':
            cards[player]['y'] = 1
        elif card_type == 'r':
            cards[player]['r'] = 1
            red_cards.append((player[0], player[1], minute))

for team, player_num, minute in red_cards:
    print(f"{team} {player_num} {minute}")
