import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] a = new int[6], cnt = {0, 0, 0};
        for (int i = 0; i < 6; i++)
            a[i] = in.nextInt();
        for (int i = 0; i < 6; i += 2)
            for (int j = i + 2; j < 6; j += 2)
                for (int k = j + 2; k < 6; k += 2) {
                    int dx = a[j] - a[i], dy = a[j + 1] - a[i + 1];
                    int ex = a[k] - a[i], ey = a[k + 1] - a[i + 1];
                    int fx = a[j] - a[k], fy = a[j + 1] - a[k + 1];
                    cnt[dx * ey - dy * ex == 0 ? 2 : dx * fx + dy * fy == 0 ? 0 : 1]++;
                }
        System.out.println(cnt[2] + cnt[0] == 3 ? (cnt[0] == 1 ? "ALMOST" : "RIGHT") : "NEITHER");
    }
}
