from collections import deque

def bfs(start, graph, color):
    queue = deque([start])
    color[start] = 1
    while queue:
        v = queue.popleft()
        for u in graph[v]:
            if color[u] == 0:
                color[u] = -color[v]
                queue.append(u)
            elif color[u] == color[v]:
                return False
    return True

def solve(n, m, edges):
    graph = [[] for _ in range(n)]
    color = [0] * n
    for a, b in edges:
        graph[a-1].append(b-1)
        graph[b-1].append(a-1)
    
    bipartite = True
    for i in range(n):
        if color[i] == 0:
            if not bfs(i, graph, color):
                bipartite = False
                break
    
    if bipartite:
        if m == n*(n-1)//2:
            return (0, 1)
        missing_edges = n*(n-1)//2 - m
        return (1, missing_edges)
    else:
        return (0, 1)

n, m = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(m)]
t, w = solve(n, m, edges)
print(t, w)
