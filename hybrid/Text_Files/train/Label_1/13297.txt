#include <bits/stdc++.h>
using namespace std;

const int MAX = 200005;
int p[MAX], a[MAX], b[MAX], c[MAX], ans[MAX];
map<pair<int, int>, int> mp;

int main() {
  int n, m;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> p[i];
  }
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  for (int i = 0; i < n; ++i) {
    cin >> b[i];
  }
  cin >> m;
  for (int i = 0; i < m; ++i) {
    cin >> c[i];
  }
  for (int i = 0; i < n; ++i) {
    mp[make_pair(a[i], b[i])] = p[i];
  }
  for (int i = 0; i < m; ++i) {
    if (mp.count(make_pair(c[i], c[i])) > 0) {
      ans[i] = mp[make_pair(c[i], c[i])];
    } else if (mp.count(make_pair(c[i], 0)) > 0) {
      ans[i] = mp[make_pair(c[i], 0)];
    } else if (mp.count(make_pair(0, c[i])) > 0) {
      ans[i] = mp[make_pair(0, c[i])];
    } else {
      ans[i] = -1;
    }
  }
  for (int i = 0; i < m; ++i) {
    cout << ans[i] << " ";
  }
  cout << endl;
  return 0;
}
