def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    
    diff_set = set()
    for i in range(1, n):
        diff_set.add(abs(a[i] - a[0]))
    
    if len(diff_set) == 0:
        print(-1)
    else:
        k = max(diff_set)
        for diff in diff_set:
            k = gcd(k, diff)
        print(k)
