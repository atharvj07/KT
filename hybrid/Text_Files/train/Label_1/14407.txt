#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int maxn = 2e5 + 10;
const ll inf = 1e18;

vector<pair<int, ll>> g[maxn];
int n, m;
ll d[maxn];
bool vis[maxn];

void dijkstra() {
  for (int i = 1; i <= n; ++i) {
    d[i] = inf;
    vis[i] = false;
  }
  d[1] = 0;
  multiset<pair<ll, int>> q;
  q.insert({0, 1});
  ll ans = 0;
  while (!q.empty()) {
    int u = q.begin()->second;
    ll w = q.begin()->first;
    q.erase(q.begin());
    if (vis[u]) continue;
    vis[u] = true;
    ans = d[u];
    for (auto& p : g[u]) {
      int v = p.first;
      ll weight = p.second;
      if (d[v] > d[u] + weight) {
        q.erase({d[v], v});
        d[v] = d[u] + weight;
        q.insert({d[v], v});
      }
    }
  }
  cout << ans << endl;
}

int main() {
  int k;
  cin >> n >> m >> k;
  while (m--) {
    int a, b, c;
    cin >> a >> b >> c;
    g[a].push_back({b, c});
    g[b].push_back({a, c});
  }
  dijkstra();
  return 0;
}
