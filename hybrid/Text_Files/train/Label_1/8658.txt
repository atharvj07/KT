#include<bits/stdc++.h>
using namespace std;
#define pb push_back
int n,L=1,R,sz[200005],par[200005],ans=0,sum[200005];
vector<int>adj[200005];
void dfs(int u,int p){
	sz[u]=1;
	for(int i=0;i<(int)adj[u].size();i++){
		int v=adj[u][i];
		if(v!=p){
			dfs(v,u);
			sum[u]+=sum[v];
			sz[u]+=sz[v];
		}
	}
}
void dfs2(int u,int p){
	for(int i=0;i<(int)adj[u].size();i++){
		int v=adj[u][i];
		if(v!=p){
			if(sz[v]>sz[R]){
				R=v;
			}
			dfs2(v,u);
		}
	}
}
int main(){
	cin>>n;
	for(int i=1;i<n;++i){
		int u,v;
		cin>>u>>v;
		ans+=par[u];
		ans+=par[v];
		par[v]++;
		adj[u].pb(v);
		adj[v].pb(u);
	}
	dfs(1,0);
	L=1;
	for(int i=1;i<=n;i++){
		if(sz[L]<sz[i]){
			L=i;
		}
	}
	dfs2(L,0);
	dfs2(R,0);
	int cnt=0;
	dfs(L,0);
	for(int i=1;i<=n;i++){
		if(sum[i]>ans){
			ans=sum[i];
		}
	}
	cout<<ans<<endl;
	dfs(1,0);
	for(int i=1;i<n;++i){
		for(int j=0;j<(int)adj[i].size();j++){
			int v=adj[i][j];
			if(sum[i]+sum[v]-2*par[i]==ans){
				cout<<i<<" "<<v<<" ";
				if(sum[i]>sum[v]){
					cout<<i<<endl;
					par[i]--;
					ans-=par[i];
					sum[i]=0;
					dfs(i,0);
				}
				else{
					cout<<v<<endl;
					par[v]--;
					ans-=par[v];
					sum[v]=0;
					dfs(v,0);
				}
				break;
			}
		}
	}
	return 0;
}
