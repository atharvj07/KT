#include<bits/stdc++.h>
#define fast ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
using namespace std;

int main() {
    fast;
    long long n, a, b, c;
    cin >> n >> a >> b >> c;
    long long ans = LLONG_MAX;
    if (n % 4 == 0) ans = 0;
    else if (n % 4 == 1) ans = min({a, b + c, 2 * b});
    else if (n % 4 == 2) ans = min({2 * a, b});
    else ans = min({3 * a, b + c});
    cout << ans;
}
