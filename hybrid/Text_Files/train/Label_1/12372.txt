#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;
const int mod = 998244353;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  ll ans = 0;
  for (int i = 0; i < n - k + 1; i++) { ans += 1LL * (a[i + k - 1] - a[i]) * i % mod; }
  for (int i = n - k + 1; i < n; i++) { ans += 1LL * (a[i] - a[i - k]) * (n - k - i) % mod; }
  cout << ans % mod << endl;
  return 0;
}
