n = int(input())
a = [int(x) for x in input().split()]
if n >= 29:
    if a[n - 1] < a[n - 29]:
        print("DOWN")
    elif a[n - 1] > a[n - 29]:
        print("UP")
    else:
        print(-1)
elif n >= 15:
    if a[n - 1] < a[n - 15]:
        print("DOWN")
    elif a[n - 1] > a[n - 15]:
        print("UP")
    else:
        print(-1)
else:
    if a[n - 1] <= a[0]:
        print("DOWN")
    else:
        print("UP")
