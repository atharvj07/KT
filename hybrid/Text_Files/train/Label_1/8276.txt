#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef vector<int> vi;
typedef pair<int, int> pi;

#define F first
#define S second

struct req {
    int c, p;
    bool operator<(const req& b) const { return p > b.p; }
};

int main() {
    int n, k;
    cin >> n;
    vector<req> r(n);
    for (auto& x : r) cin >> x.c >> x.p;
    cin >> k;
    vi t(k);
    for (auto& x : t) cin >> x;
    sort(r.begin(), r.end());
    sort(t.begin(), t.end());
    int m = 0, s = 0;
    vector<pi> out;
    for (auto& x : r) {
        auto it = lower_bound(t.begin(), t.end(), x.c);
        if (it == t.end()) continue;
        s += x.p;
        m++;
        int ind = it - t.begin();
        t[ind] -= x.c;
        if (t[ind] == 0) t.erase(t.begin() + ind);
        out.push_back({x.c, ind + 1});
    }
    cout << m << " " << s << endl;
    for (auto& x : out) cout << x.F << " " << x.S << endl;
    return 0;
}
