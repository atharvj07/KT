import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt(), cnt = 0, l = 0;
        int[] a = new int[m];
        for (int i = 0; i < m; i++)
            a[i] = in.nextInt();
        for (int i = 0; i < m; i++)
            if (a[i] <= l)
                cnt++;
            else
                l = a[i] + in.nextInt() - 1;
        System.out.println(cnt);
        for (int i = 0; i < m; i++)
            if (a[i] > l)
                System.out.print((i + 1) + " ");
    }
}
