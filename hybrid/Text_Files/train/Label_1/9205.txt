#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int& i : a) cin >> i;
  vector<int> p(n + 1);
  for (int i = 0; i < n; ++i) p[a[i]] = i;
  int ans = 0;
  for (int i = 1; i < n; ++i) {
    if (p[i] < p[i + 1]) ++ans;
  }
  cout << ans << endl;
}
