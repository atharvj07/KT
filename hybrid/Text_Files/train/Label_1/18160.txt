#include <cstdio>
#include <vector>
using namespace std;
int main(){
  long long x;
  scanf("%lld",&x);
  long long t=0;
  for(long long i=1;i*i<=x;++i){
    long long rem=x-i*i;
    if(rem==0){
      ++t;
    }else{
      long long sq=0;
      for(long long j=1;j*j<=rem;++j){
        if(rem%j==0){
          sq+=j;
          if(rem/j!=j)sq+=rem/j;
        }
      }
      t+=sq;
    }
  }
  printf("%lld\n",t);
  for(long long i=1;i*i<=x;++i){
    long long rem=x-i*i;
    if(rem==0){
      printf("%lld %lld\n",i,i);
    }else{
      long long sq=0;
      for(long long j=1;j*j<=rem;++j){
        if(rem%j==0){
          sq+=j;
          if(rem/j!=j)sq+=rem/j;
        }
      }
      for(long long j=1;j*j<=rem;++j){
        if(rem%j==0){
          printf("%lld %lld\n",j,rem/j);
          if(rem/j!=j)printf("%lld %lld\n",rem/j,j);
        }
      }
    }
  }
  return 0;
}
