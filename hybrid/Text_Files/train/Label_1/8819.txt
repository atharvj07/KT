#include <iostream>
#include <vector>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<int> a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        int ans = -1;
        for (int i = 0; i < n; i++) {
            if (a[i] == i + 1) { k--; }
            if (k == 0) {
                ans = i;
                break;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
