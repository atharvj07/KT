import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int max = -1;
        int pos = -1;
        for (int i = 0; i < n; i++) {
            if (a[i] > max) {
                max = a[i];
                pos = i;
            }
        }
        boolean possible = true;
        for (int i = 0; i < n; i++) {
            if (Math.abs(pos - i) != a[i]) {
                possible = false;
                break;
            }
        }
        System.out.println(possible ? "Possible" : "Impossible");
    }
}
