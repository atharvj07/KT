import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextInt();
        long x = sc.nextInt();
        long y = sc.nextInt();

        long ans = 0;
        if (y <= x) {
            ans = n * x;
        } else {
            while (n > 0) {
                if (y * (n / 2) <= x * (n - (n / 2))) {
                    n /= 2;
                    ans += y;
                } else {
                    n--;
                    ans += x;
                }
            }
        }

        System.out.println(ans);
    }
}
