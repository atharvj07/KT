#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, ans = 0;
  cin >> n;
  vector<vector<char>> v(n, vector<char>(n));
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++) cin >> v[i][j];
  for (int i = 1; i < n - 1; i++)
    for (int j = 1; j < n - 1; j++)
      ans += v[i][j] == 'X' && v[i - 1][j - 1] == 'X' &&
             v[i - 1][j + 1] == 'X' && v[i + 1][j - 1] == 'X' &&
             v[i + 1][j + 1] == 'X';
  cout << ans << "\n";
  return 0;
}
