#include <bits/stdc++.h>
using namespace std;
int n, a[200002], b[200002], cc, vis[200002];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    if (!a[i]) cc++;
  }
  for (int i = 1; i <= n; i++) {
    if (a[i]) b[i] = a[i];
  }
  for (int i = 1; i <= n; i++) {
    if (!a[i]) {
      while (1) {
        int x = rand() % n + 1;
        if (x != i && !vis[x] && x != a[b[x]]) {
          b[i] = x;
          vis[x] = 1;
          break;
        }
      }
    }
  }
  for (int i = 1; i <= n; i++) { cout << b[i] << " "; }
  return 0;
}
