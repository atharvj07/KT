import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), k = scanner.nextInt();
        int[][] data = new int[n][3];
        for (int i = 0; i < n; i++) {
            data[i][0] = scanner.nextInt();
            data[i][1] = scanner.nextInt();
            data[i][2] = scanner.nextInt();
        }
        Arrays.sort(data, Comparator.comparingInt(a -> a[0]));
        int[] dp = new int[n];
        int[] iq = new int[n];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            iq[i] = data[i][2];
        }
        for (int i = 0; i < n; i++) {
            int j = Arrays.binarySearch(iq, i, n, data[i][2]);
            if (j < 0) j = -j - 1;
            if (j > 0 && data[i][2] - iq[j - 1] <= k) dp[i] = dp[j - 1];
            if (j < n && iq[j] - data[i][2] <= k) dp[i] = Math.max(dp[i], dp[j]);
            dp[i]++;
            ans = Math.max(ans, dp[i]);
        }
        System.out.println(ans - 1);
    }
}
