import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long k = sc.nextLong();

        long ans = (long) Math.ceil((double) k / m);
        ans *= m - Math.ceil((double) k / n) * n + k % m;
        System.out.println(ans);
    }
}
