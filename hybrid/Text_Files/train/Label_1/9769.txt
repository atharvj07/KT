#include <bits/stdc++.h>
using namespace std;
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int even = 0, odd = 0;
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            if (x & 1) odd++; else even++;
        }
        if (even == 0 || odd == 0) cout << "Alice" << endl; else if (odd % 2 == 0) cout << "Tie" << endl; else cout << "Bob" << endl;
    }
    return 0;
}
