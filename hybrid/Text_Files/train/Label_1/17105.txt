ll powmod(ll x,ll y){ll t;for(t=1;y;y>>=1,x=x*x%mod)if(y&1)t=t*x%mod;return t;}
ll gcd(ll x,ll y){ll q,t;while(y){q=x%y;t=x;x=y;y=q;}return x;}
int main()
{
	ll n,a,b,c,ans=0;
	cin>>n>>a>>b>>c;
	if(b-c<=a){cout<<n/b<<endl;return 0;}
	ans=n/(b-c);
	n=n-ans*(b-c);
	ans+=n/a;
	cout<<ans<<endl;
	return 0;
}
