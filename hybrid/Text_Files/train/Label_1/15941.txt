#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N=2e5+5;
struct node{
    ll val,pos;
    bool operator<(const node &o)const{
        return val<o.val;
    }
};
int main(){
    ll n,k,x; cin>>n>>k>>x;
    vector<node>v;
    for(ll i=0;i<n;i++){
        ll a; cin>>a;
        v.push_back({a,i});
    }
    sort(v.begin(),v.end());
    for(ll i=0;i<n;i++){
        while(v[i].val*x<=v[i+1].val&&k){
            v[i].val*=x;
            k--;
        }
    }
    ll ans=0;
    for(ll i=0;i<n;i++){
        ans|=v[i].val;
    }
    cout<<ans;
    return 0;
}
