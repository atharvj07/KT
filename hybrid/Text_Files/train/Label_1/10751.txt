#include<vector>
#include<cstdio>
#include<algorithm>
using namespace std;

const int N = 5e5 + 10;
int n, m, k, s;
int a[N], b[N];
int cnt[N];

vector<int> rem;
vector<int> a1;
vector<int> b1;

int main() {
  scanf("%d%d%d%d", &m, &k, &n, &s);
  for (int i = 1; i <= m; ++i) {
    scanf("%d", &a[i]);
    cnt[a[i]]++;
  }
  for (int i = 1; i <= s; ++i){ 
    scanf("%d", &b[i]);
    b1.push_back(b[i]);
  }
  sort(b1.begin(), b1.end());

  int j = 0;
  for (int i = 1; i <= m; ++i) {
    if (cnt[a[i]] < k) continue;
    cnt[a[i]] -= k;
    if (j < s && a[i] == b[j]) {
      j++;
    } else {
      a1.push_back(a[i]);
    }
  }
  if (j != s) {
    puts("-1");
    return 0;
  }

  sort(a1.begin(), a1.end());
  sort(b1.begin(), b1.end());
  int cnt1 = 0, cnt2 = 0;
  int tot = 0;

  for (int i = 0; i < a1.size(); ++i) {
    if (a1[i] < b1[0]) {
      rem.push_back(a1[i]);
      tot++;
      continue;
    }
    if (cnt1 < s && b1[cnt1] == a1[i]) {
      ++cnt1;
      tot++;
      continue;
    } else
      break;
  }

  for (int i = a1.size() - 1; i >= 0; --i) {
    if (a1[i] > b1[s - 1]) {
      rem.push_back(a1[i]);
      tot++;
      continue;
    }
    if (cnt2 < s && b1[s - cnt2 - 1] == a1[i]) {
      ++cnt2;
      tot++;
      continue;
    } else
      break;
  }

  printf("%d\n", tot);
  for (int i = 0; i < rem.size(); ++i) {
    printf("%d ", rem[i]);
  }
  puts("");
  return 0;
}
