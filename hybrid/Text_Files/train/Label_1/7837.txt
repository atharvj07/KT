#include <iostream>
using namespace std;
int main() {
    int n, a[5] = {0, 0, 0, 0, 0}, x;
    char c;
    cin >> n;
    while (n--) {
        cin >> c >> x;
        if (c == '&')
            a[4] ^= x;
        else if (c == '|')
            a[3] ^= x;
        else
            a[2] ^= x;
    }
    if (a[2] == 0 && a[3] == 0) {
        cout << 1 << endl;
        cout << "& " << a[4] << endl;
        return 0;
    }
    if (a[3] == 0) {
        cout << 2 << endl;
        cout << (a[2] == a[4] ? "^ " : "& ") << a[2] << endl;
        cout << "| " << a[4] << endl;
        return 0;
    }
    if (a[2] == 0) {
        cout << 2 << endl;
        cout << "| " << a[3] << endl;
        cout << "& " << a[4] << endl;
        return 0;
    }
    cout << 3 << endl;
    cout << "^ " << a[2] << endl;
    cout << "| " << a[3] << endl;
    cout << "& " << a[4] << endl;
    return 0;
}
