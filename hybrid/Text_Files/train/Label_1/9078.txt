#include<bits/stdc++.h>

using namespace std;

struct Node{
	int octet[4];
	int mask;
	bool type;
	Node(){
		mask=0;
		for(int &x : octet)
			x=0;
	}
	Node(int a,int b,int c,int d,int m,bool t){
		octet[0]=a;
		octet[1]=b;
		octet[2]=c;
		octet[3]=d;
		mask=m;
		type=t;
	}
	
	bool operator < (const Node& o) const{
		for(int i=0;i<4;i++){
			if(octet[i]!=o.octet[i])
				return octet[i]<o.octet[i];
		}
		return mask<o.mask;
	}
};

int main(){
	int n;
	cin>>n;
	vector<Node>vec;
	while(n--){
		string s;
		cin>>s;
		bool type;
		int octet[4];
		int mask;
		if(s[0]=='+')
			type=1;
		else
			type=0;
		s=s.substr(1);
		int idx=s.find('/');
		if(idx!=-1){
			mask=stoi(s.substr(idx+1));
			s=s.substr(0,idx);
		}
		else
			mask=32;
		
		int cnt=0;
		for(int i=0;i<4;i++){
			int p=s.find('.');
			if(p!=-1){
				octet[cnt++]=stoi(s.substr(0,p));
				s=s.substr(p+1);
			}
			else{
				octet[cnt++]=stoi(s);
			}
		}
		vec.push_back(Node(octet[0],octet[1],octet[2],octet[3],mask,type));
			
	}
	sort(vec.begin(),vec.end());
	int sz=vec.size();
	bool flag=0;
	for(int i=0;i<sz;i++){
		if(vec[i].type==0){
			int j=i;
			while(j<sz&&vec[j].type==0 && vec[j].mask<=vec[i].mask){
				if(j+1<sz){
					for(int k=0;k<4;k++){
						if(vec[j+1].octet[k]<vec[i].octet[k])
							flag=1;
					}	
				}
				j++;
			}
			i=j-1;
			if(flag)
				break;
		}
		else{
			int j=i;
			while(j<sz&&vec[j].type==1 && vec[j].mask<=vec[i].mask){
				if(j+1<sz){
					for(int k=0;k<4;k++){
						if(vec[j+1].octet[k]>vec[i].octet[k])
							flag=1;
					}	
				}
				j++;
			}
			i=j-1;
			if(flag)
				break;
		}
	}
	if(flag){
		cout<<-1<<endl;
	}
	else{
		vector<Node>ans;
		for(auto &node : vec){
			if(node.type==0)
				ans.push_back(node);
		}
		int x=ans.size();
		cout<<x<<endl;
		for(auto &node : ans){
			if(node.mask==32)
				cout<<node.octet[0]<<"."<<node.octet[1]<<"."<<node.octet[2]<<"."<<node.octet[3]<<endl;
			else
				cout<<node.octet[0]<<"."<<node.octet[1]<<"."<<node.octet[2]<<"."<<node.octet[3]<<"/"<<node.mask<<endl;
		}
	}
	
	return 0;
}
