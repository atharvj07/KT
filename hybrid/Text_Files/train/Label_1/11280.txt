#include <bits/stdc++.h>

using namespace std;

struct Link{
    int to;
    long long w;
};

const int mxN=2e5+5,mxL=1e5;
vector<Link> graph[mxN];
bool vis[mxN];
long long dis[mxN],parent[mxN];
int sub[mxN];
int n,m,ans,tail;

void dfs1(int u,int p){
    vis[u]=1;
    parent[u]=p;
    for(Link v:graph[u]){
        if(vis[v.to]) continue;
        dis[v.to]=dis[u]+v.w;
        dfs1(v.to,u);
    }
}
void dfs2(int u){
    vis[u]=1;
    long long mx=0;
    for(Link v:graph[u]){
        if(vis[v.to]) continue;
        dfs2(v.to);
        if(mx<dis[v.to]){
            tail=v.to;
            mx=dis[v.to];
        }
    }
}
void dfs3(int u){
    vis[u]=1;
    if(u!=tail) sub[u]=1;
    for(Link v:graph[u]){
        if(vis[v.to]) continue;
        if(v.to!=parent[u]) dfs3(v.to);
        sub[u]+=sub[v.to];
    }
}
int main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cin>>n>>m;
    while(m--){
        int x,y;
        cin>>x>>y;
        graph[x].push_back({y,1});
        graph[y].push_back({x,1});
    }
    dfs1(1,1);
    memset(vis,0,sizeof vis);
    dfs2(1);
    ans=dis[tail];
    memset(vis,0,sizeof vis);
    dfs3(1);
    ans*=sub[tail];
    cout<<ans<<"\n";
}
