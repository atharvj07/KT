import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Scanner;

public class Main {
    static long[][][] mem;
    static long[][] a;
    static long[][] b;
    static long[][] c;

    static long func(int x, int y, int z) {
        if (x == 0 || y == 0 || z == 0) {
            return 0;
        }
        if (mem[x][y][z] != -1) {
            return mem[x][y][z];
        }
        long max = 0;
        for (int i = 0; i < x; i++) {
            for (int j = 0; j < y; j++) {
                for (int k = 0; k < z; k++) {
                    max = Math.max(max, a[i][0] + b[j][0] + c[k][0] + func(i, j + 1, k + 1));
                }
            }
        }
        return mem[x][y][z] = max;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int y = scanner.nextInt();
        int z = scanner.nextInt();
        int k = scanner.nextInt();
        a = new long[x][2];
        b = new long[y][2];
        c = new long[z][2];
        for (int i = 0; i < x; i++) {
            a[i][0] = scanner.nextLong();
            a[i][1] = i;
        }
        for (int i = 0; i < y; i++) {
            b[i][0] = scanner.nextLong();
            b[i][1] = i;
        }
        for (int i = 0; i < z; i++) {
            c[i][0] = scanner.nextLong();
            c[i][1] = i;
        }
        mem = new long[x + 1][y + 1][z + 1];
        for (int i = 0; i < x + 1; i++) {
            for (int j = 0; j < y + 1; j++) {
                for (int l = 0; l < z + 1; l++) {
                    mem[i][j][l] = -1;
                }
            }
        }
        List<Long> ans = new ArrayList<>();
        if (func(x, y, z) > 0) {
            List<Integer> aa = new ArrayList<>();
            List<Integer> bb = new ArrayList<>();
            List<Integer> cc = new ArrayList<>();
            while (x > 0 && y > 0 && z > 0) {
                long max = 0;
                for (int i = 0; i < x; i++) {
                    for (int j = 0; j < y; j++) {
                        for (int l = 0; l < z; l++) {
                            if (max < a[i][0] + b[j][0] + c[l][0] + func(i, j + 1, l + 1)) {
                                max = a[i][0] + b[j][0] + c[l][0] + func(i, j + 1, l + 1);
                                aa.add(i);
                                bb.add(j);
                                cc.add(l);
                            }
                        }
                    }
                }
                x = aa.get(aa.size() - 1);
                y = bb.get(bb.size() - 1);
                z = cc.get(cc.size() - 1);
                ans.add(a[x][0] + b[y][0] + c[z][0]);
                a[x][0] = -1;
                b[y][0] = -1;
                c[z][0] = -1;
                aa.clear();
                bb.clear();
                cc.clear();
            }
        }
        ans.sort(Comparator.reverseOrder());
        if (ans.isEmpty()) {
            System.out.println(0);
        } else {
            for (int i = 0; i < Math.min(ans.size(), k); i++) {
                System.out.println(ans.get(i));
            }
        }
    }
}
