#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n, k;
    std::cin >> n >> k;
    std::vector<int> a(n);
    for (int &aa : a) std::cin >> aa;
    std::sort(a.begin(), a.end());
    for (int i = 0; i < n; i++) {
        if (k < (a[i] + 1) / 2) {
            std::cout << n - i << std::endl;
            return 0;
        }
        k = a[i];
    }
    std::cout << 0 << std::endl;
    return 0;
}
