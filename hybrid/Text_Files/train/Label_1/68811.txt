from math import ceil
n, k = map(int, input().split())
a = list(map(int, input().split()))
ans = 0
t = [0] * (n + 1)
for i in range(1, n + 1):
    t[i] = t[i - 1] + a[i - 1] // k + 1
for i in range(1, n + 1):
    for j in range(1, a[i - 1] + 1):
        if ceil(t[i] - j / k * 100 / n) == j:
            ans += 1
print(ans)
