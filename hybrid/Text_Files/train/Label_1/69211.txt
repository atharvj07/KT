n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

# Combine lists with player tags
combined = [(val, 'A') for val in a] + [(val, 'B') for val in b]
# Sort combined list in descending order
combined.sort(reverse=True, key=lambda x: x[0])

a_score, b_score = 0, 0

# Simulate the game
for i, (val, player) in enumerate(combined):
    if i % 2 == 0:  # Player A's turn
        if player == 'A':
            a_score += val
    else:  # Player B's turn
        if player == 'B':
            b_score += val

print(a_score - b_score)
