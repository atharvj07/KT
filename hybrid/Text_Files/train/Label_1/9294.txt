#include <bits/stdc++.h>
using namespace std;
const int maxn = 105;
int n, x, d[maxn], h[maxn];
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        cin >> n >> x;
        for (int i = 1; i <= n; i++) cin >> d[i] >> h[i];
        sort(d + 1, d + n + 1);
        sort(h + 1, h + n + 1);
        int ans = 0, cur = x, nd = 0, nh = 0;
        for (int i = 1; i <= n; i++)
        {
            nd = max(0, d[i] - max(nd, x - cur));
            nh = max(nh, h[i] - cur);
            if (nd > cur)
            {
                ans = -1;
                break;
            }
            nh = min(cur, nh);
            cur = nh;
            ans += nd;
        }
        cur = x;
        for (int i = n; i >= 1; i--)
        {
            nd = max(0, d[i] - max(nd, x - cur));
            nh = max(nh, h[i] - cur);
            if (nd > cur)
            {
                ans = -1;
                break;
            }
            nh = min(cur, nh);
            cur = nh;
            ans += nd;
        }
        cout << ans << endl;
    }
    return 0;
}
