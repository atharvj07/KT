import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }
        int minDeviation = Integer.MAX_VALUE;
        int minId = 0;
        for (int i = 0; i < n; i++) {
            int deviation = 0;
            for (int j = 0; j < n; j++) {
                int pos = (j + i) % n;
                deviation += Math.abs(p[j] - p[pos]);
            }
            if (deviation < minDeviation) {
                minDeviation = deviation;
                minId = i;
            }
        }
        System.out.println(minDeviation + " " + minId);
    }
}
