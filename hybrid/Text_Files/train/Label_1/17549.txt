#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
    int d;
    cin >> d;
    int n, m;
    cin >> n >> m;
    vector<tuple<int, int, int, int>> sofas(d);
    for (int i = 0; i < d; ++i) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        sofas[i] = make_tuple(x1, y1, x2, y2);
    }
    int cnt_l, cnt_r, cnt_t, cnt_b;
    cin >> cnt_l >> cnt_r >> cnt_t >> cnt_b;
    bool found = false;
    for (int i = 0; i < d; ++i) {
        int x1, y1, x2, y2;
        tie(x1, y1, x2, y2) = sofas[i];
        bool left = x1 == 1 || (x1 == x2 && y1 == 1);
        bool right = x2 == n || (x1 == x2 && y2 == m);
        bool top = y1 == 1 || (y1 == y2 && x1 == 1);
        bool bottom = y2 == m || (y1 == y2 && x2 == n);
        int l = 0, r = 0, t = 0, b = 0;
        for (int j = 0; j < d; ++j) {
            int x11, y11, x22, y22;
            tie(x11, y11, x22, y22) = sofas[j];
            if (x11 < x1 && x22 <= x1) {
                ++l;
            }
            if (x22 > x2 && x11 >= x2) {
                ++r;
            }
            if (y11 < y1 && y22 <= y1) {
                ++t;
            }
            if (y22 > y2 && y11 >= y2) {
                ++b;
            }
        }
        if (l == cnt_l && r == cnt_r && t == cnt_t && b == cnt_b) {
            cout << i + 1 << '\n';
            found = true;
            break;
        }
    }
    if (!found) {
        cout << -1 << '\n';
    }
    return 0;
}
