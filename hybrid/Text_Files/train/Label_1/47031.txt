import java.util.Scanner;

public class LemmingsProtection {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] shields = new int[n];
        for (int i = 0; i < n; i++) {
            shields[i] = scanner.nextInt();
        }

        int[][] dp = new int[n + 1][n * (n - 1) / 2 + 1];

        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= n * (n - 1) / 2; j++) {
                dp[i][j] = -1;
            }
        }

        System.out.println(calculateMaxProtection(shields, 0, 0, dp));
    }

    private static int calculateMaxProtection(int[] shields, int currentPosition, int shieldsUsed, int[][] dp) {
        if (currentPosition == shields.length) {
            return 0;
        }

        if (dp[currentPosition][shieldsUsed] != -1) {
            return dp[currentPosition][shieldsUsed];
        }

        // Option 1: Keep the shield
        int option1 = calculateMaxProtection(shields, currentPosition + 1, shieldsUsed, dp);

        // Option 2: Give the shield to the right
        int option2 = 0;
        if (currentPosition + 1 < shields.length && shields[currentPosition + 1] == 0) {
            option2 = calculateMaxProtection(shields, currentPosition + 2, shieldsUsed + 1, dp) + countProtectedPairs(shields, currentPosition + 1);
        }

        // Option 3: Give the shield to the left
        int option3 = 0;
        if (currentPosition > 0 && shields[currentPosition - 1] == 0) {
            option3 = calculateMaxProtection(shields, currentPosition + 1, shieldsUsed + 1, dp) + countProtectedPairs(shields, currentPosition - 1);
        }

        dp[currentPosition][shieldsUsed] = Math.max(Math.max(option1, option2), option3);
        return dp[currentPosition][shieldsUsed];
    }

    private static int countProtectedPairs(int[] shields, int shieldPosition) {
        int count = 0;
        for (int i = shieldPosition - 1; i >= 0; i--) {
            if (shields[i] == 1) {
                break;
            }
            for (int j = shieldPosition + 1; j < shields.length; j++) {
                if (shields[j] == 1) {
                    break;
                }
                count++;
            }
        }
        return count;
    }
}
