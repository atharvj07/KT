#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;

vector<int> calc_powers() {
    vector<int> res(51);
    res[0] = 1;
    for (ll p = 1; p <= 50; p++) res[p] = res[p - 1] * 2;
    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin >> n;
    vector<ll> nums(n);
    for (ll &x : nums) cin >> x;
    vector<int> powers = calc_powers();
    for (int i = 0; i < n; i++) {
        ll num = nums[i];
        int k = 0;
        ll pw = powers[k];
        bool found = false;
        while (!found && pw <= 1e18) {
            string s = to_string(pw);
            string x = to_string(num);
            if (s.substr(s.size() - x.size()) == x) {
                found = true;
                break;
            }
            k++;
            pw = powers[k];
        }
        cout << k << endl;
    }
    return 0;
}
