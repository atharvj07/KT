#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long x, y, a, b;
    cin >> x >> y >> a >> b;
    if (x == 0 && y == 0) {
      cout << 0 << endl;
    } else if (x == y) {
      cout << a << endl;
    } else if ((x > 0 && y > 0) || (x < 0 && y < 0)) {
      cout << min(abs(x - y) * a, b) << endl;
    } else {
      cout << abs(x) * a + b << endl;
    }
  }
  return 0;
}
