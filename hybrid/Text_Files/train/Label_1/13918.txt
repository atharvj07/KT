#include <bits/stdc++.h>
using namespace std;
const int maxn=1e5+5;
int n,fa[maxn],sz[maxn],a[maxn];
bool vis[maxn];
vector<array<int,2>> adj[maxn];
int find(int x) { return x==fa[x]?x:fa[x]=find(fa[x]); }
void dfs(int u) {
	sz[u]=1;
	for (auto e:adj[u]) if (!vis[e[1]]) {
		a[e[0]]=e[1];
		vis[e[1]]=true;
		fa[e[1]]=u;
		dfs(e[1]);
		sz[u]+=sz[e[1]];
	}
}
int main() {
	ios::sync_with_stdio(false); cin.tie();
	cin>>n;
	for (int i=1; i<=n; i++) fa[i]=i;
	for (int i=1; i<n; i++) {
		int u,v; cin>>u>>v;
		adj[u].push_back({i,v});
		adj[v].push_back({i,u});
	}
	dfs(1);
	for (int u=1; u<=n; u++) {
		vector<bool> occ(n);
		for (auto e:adj[u]) if (e[1]!=fa[u]) {
			occ[a[e[0]]]=true;
		}
		int cur=0;
		for (auto e:adj[u]) if (e[1]!=fa[u]) {
			while (occ[cur]) cur++;
			a[e[0]]=cur++;
		}
	}
	for (int i=1; i<n; i++) cout<<a[i]<<' ';
	cout<<'\n';
}
