n = int(input())
planks = list(map(int, input().split()))
q = int(input())

plank_counts = {}
for plank in planks:
    plank_counts[plank] = plank_counts.get(plank, 0) + 1

squares = 0
rectangles = 0
for count in plank_counts.values():
    squares += count // 4
    rectangles += count // 2

for _ in range(q):
    event, x = input().split()
    x = int(x)

    if event == '+':
        if x in plank_counts:
            rectangles -= plank_counts[x] // 2
            squares -= plank_counts[x] // 4
            plank_counts[x] += 1
            rectangles += plank_counts[x] // 2
            squares += plank_counts[x] // 4
        else:
            plank_counts[x] = 1
            rectangles += plank_counts[x] // 2
    else:
        rectangles -= plank_counts[x] // 2
        squares -= plank_counts[x] // 4
        plank_counts[x] -= 1
        rectangles += plank_counts[x] // 2
        squares += plank_counts[x] // 4

        if plank_counts[x] == 0:
            del plank_counts[x]

    if squares >= 1 and rectangles >= 4:
        print("YES")
    else:
        print("NO")
