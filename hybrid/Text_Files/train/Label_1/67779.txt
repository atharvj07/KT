from collections import defaultdict

n = int(input())
a = [input() for _ in range(n)]
pref = [defaultdict(int) for _ in range(n+1)]
suf = [defaultdict(int) for _ in range(n+1)]
for i in range(n-1,-1,-1):
    suf[i] = suf[i+1].copy()
    suf[i][a[i][-1]] += 1
for i in range(n):
    pref[i+1] = pref[i].copy()
    pref[i+1][a[i][0]] += 1
ans = 0
for i in range(n):
    for c in pref[i]:
        if c == a[i][-1] and suf[i+1][c] > 0 and (i + 1 != n or pref[i][c] > 1):
            ans += pref[i][c] * suf[i+1][c]
    if a[i] == a[i][::-1]:
        ans += pref[i][a[i][0]] * suf[i+1][a[i][-1]] - 1
print(ans)
