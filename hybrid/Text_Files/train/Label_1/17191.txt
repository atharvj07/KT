#include <iostream>
#include <vector>
#include <queue>
using namespace std;

const int MAXN = 3e4 + 5;
vector<int> g[MAXN], ord;
int in[MAXN];

int main() {
  int n, m;
  cin >> n >> m;
  while (m--) {
    int a, b;
    cin >> a >> b;
    g[b].push_back(a);
    ++in[a];
  }
  queue<int> q;
  for (int i = 1; i <= n; ++i)
    if (in[i] == 0) q.push(i);
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    ord.push_back(u);
    for (int v : g[u]) {
      if (--in[v] == 0) q.push(v);
    }
  }
  if (ord.size() < n)
    cout << -1;
  else
    for (int x : ord) cout << x << ' ';
  return 0;
}
