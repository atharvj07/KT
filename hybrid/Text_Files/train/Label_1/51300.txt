import java.util.Scanner;

public class LiegesOfLegendre {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] piles = new int[n];
        for (int i = 0; i < n; i++) {
            piles[i] = scanner.nextInt();
        }

        if (canWin(piles, k)) {
            System.out.println("Kevin");
        } else {
            System.out.println("Nicky");
        }
    }

    static boolean canWin(int[] piles, int k) {
        int sum = 0;
        for (int pile : piles) {
            sum += pile;
        }
        return sum % (k + 1) != 0;
    }
}
