#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;

const int MAX_N = 501;

vector<vector<int>> A(MAX_N, vector<int>(MAX_N));
vector<vector<int>> B(MAX_N, vector<int>(MAX_N));
vector<vector<bool>> visited(MAX_N, vector<bool>(MAX_N));

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> A[i][j];
        }
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> B[i][j];
        }
    }

    bool possible = true;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (A[i][j] != B[i][j]) {
                if (visited[i][j]) {
                    possible = false;
                    break;
                }
                visited[i][j] = true;
                int minVal = min(n - i + 1, m - j + 1);
                int k = 1;
                while (k <= minVal && A[i][j] != B[i][j]) {
                    for (int l = 0; l < k; l++) {
                        swap(A[i + l][j + k - 1 - l], A[i + k - 1 - l][j + l]);
                    }
                    k++;
                }
                if (A[i][j] != B[i][j]) {
                    possible = false;
                    break;
                }
            }
        }
    }

    if (possible) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
