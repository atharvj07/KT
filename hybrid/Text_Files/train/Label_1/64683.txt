for i in range(int(input())):
    n = int(input())
    a = []
    for _ in range(n-1):
        b = list(map(int, input().split()))
        if len(b) >= 2: b.sort()
        a.append(b)
    a.sort(key=len, reverse=True)
    p = [0]*n
    for x in a:
        for v in x:
            if p[v-1] == 0:
                for j in range(n):
                    if p[j] == 0:
                        p[j] = v
                        break
    print(*p)
