#include <iostream>
#include <vector>
int main()
{
    int n;
    std::cin >>n;
    std::vector<int> a(n);
    for (auto &x:a) std::cin >>x;
    int ans=0;
    std::vector<int> b(n+1);
    for (auto i=0;i<n;++i) b[a[i]]=i;
    for (auto i=2;i<=n;++i)
        ans+=(b[i]<b[i-1]);
    std::cout <<ans<<'\n';
    return 0;
}
