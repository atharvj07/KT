import sys

n = int(input())
l = list(map(int,input().split()))
r = list(map(int,input().split()))
suff_r = [0]*n
suff_r[n-1] = r[n-1]
for i in range(n-2,-1,-1):
    suff_r[i] = r[i] + suff_r[i+1]
    
if any(x>y for x,y in zip(l,suff_r)):
    print('NO')
    sys.exit()

a = [0]*n
l_count = 0
for i in range(n):
    a[i] = max(l_count+1,n-suff_r[i]-1)
    l_count += a[i]>l[i]
    
print('YES')
print(*a)
