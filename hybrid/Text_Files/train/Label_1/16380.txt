#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      int cnt = 0;
      for (int k = i; k <= j; k++) {
        cnt += (a[k] ^ 1);
      }
      ans = max(ans, cnt);
    }
  }
  cout << ans << endl;
  return 0;
}
