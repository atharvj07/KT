import sys
n=int(input())
s=input()
t=input()
if s==t:
    print(0)
else:
    a,b,c,d,cnt=0,0,0,0,0
    for i in range(n):
        a+=s[i]=='a'
        b+=s[i]=='b'
        c+=t[i]=='a'
        d+=t[i]=='b'
        if s[i]!=t[i]:
            cnt+=1
    if a==c and b==d and cnt%2==0:
        print((cnt+1)//2)
        for i in range(n):
            if s[i]!=t[i]:
                print(i+1,i+1)
    else: print(-1)
