#include <bits/stdc++.h>
using namespace std;
const int N = 200005;
int n, m, a[N], x[N], y[N], f[N], g[N];
vector<pair<int, int>> vec;
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  cin >> m;
  for (int i = 1; i <= m; i++) { cin >> x[i] >> y[i]; }
  sort(a + 1, a + 1 + n);
  for (int i = 1; i <= n; i++) { f[i] = f[i - 1] + a[i]; }
  for (int i = n; i >= 1; i--) { g[i] = g[i + 1] + a[i]; }
  for (int i = 1; i <= m; i++) {
    int l = lower_bound(a + 1, a + 1 + n, x[i]) - a;
    int r = upper_bound(a + 1, a + 1 + n, y[i]) - a - 1;
    vec.push_back({l, r});
  }
  sort(vec.begin(), vec.end());
  long long ans = 0;
  for (auto p : vec) {
    int l = p.first, r = p.second;
    if (l == 0)
      ans += x[p.second];
    else
      ans += min(f[l - 1], g[r + 1]);
  }
  cout << ans << endl;
}
