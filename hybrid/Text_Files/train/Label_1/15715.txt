#include <iostream>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        int n, k;
        cin >> n >> k;

        if (n - 1 < k || k & 1) {
            cout << -1 << endl;
            continue;
        }

        vector<int> a(n);
        int mid = (n + 1) / 2;

        for (int i = 0; i < k / 2; i++) {
            a[i * 2] = mid + 1 + i;
            a[i * 2 + 1] = mid - 1 - i;
        }

        for (int i = k / 2 * 2; i < n; i++) {
            a[i] = i + 1;
        }

        for (int i = 0; i < n; i++) {
            cout << a[i] << " ";
        }
        cout << endl;
    }

    return 0;
}
