import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), k = input.nextInt(), a = input.nextInt(), b = input.nextInt(), q = input.nextInt();

        long[] orders = new long[n + 1];
        Map<Integer, Long> map = new HashMap<>();
        for (int i = 0; i < q; i++) {
            int type = input.nextInt();
            if (type == 1) {
                int d = input.nextInt(), o = input.nextInt();
                orders[d] += o;
            } else {
                int p = input.nextInt();
                long ans = 0;
                for (int j = p - 1; j <= p + k - 2; j++) {
                    ans += map.getOrDefault(j, 0L);
                }
                for (int j = Math.max(p + k, p); j <= n; j++) {
                    ans += Math.min(a, orders[j]);
                    map.put(j - k, map.getOrDefault(j - k, 0L) + Math.min(a, orders[j]));
                    ans -= map.getOrDefault(j - k - (k - 1), 0L);
                }
                System.out.println(ans);
            }
        }
        input.close();
    }
}
