#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        int x, y;
        cin >> x >> y;
        v[i] = {x, y};
    }
    double minAngle = M_PI;
    int a = -1, b = -1;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            int dx1 = v[i].first, dy1 = v[i].second;
            int dx2 = v[j].first, dy2 = v[j].second;
            double dot = dx1 * dx2 + dy1 * dy2;
            double det = dx1 * dy2 - dx2 * dy1;
            double angle = acos(dot / sqrt((dx1 * dx1 + dy1 * dy1) * (dx2 * dx2 + dy2 * dy2)));
            if (fabs(angle) < minAngle) {
                minAngle = fabs(angle);
                a = i + 1;
                b = j + 1;
            }
        }
    }
    cout << a << ' ' << b << endl;
    return 0;
}
