import java.util.*;

public class Main {

    static int[] a, b;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        a = new int[n + 2];
        b = new int[n + 2];
        for (int i = 1; i <= n; i++)
            a[i] = sc.nextInt();
        int ans = 0;
        while (true) {
            boolean flag = false;
            for (int i = 2; i < n; i++)
                if (a[i - 1] != a[i + 1] && a[i - 1] != a[i]) {
                    b[i] = a[i];
                    flag = true;
                } else
                    b[i] = a[i - 1];
            b[1] = a[1];
            b[n] = a[n];
            if (!flag)
                break;
            for (int i = 1; i <= n; i++)
                a[i] = b[i];
            ans++;
        }
        if (ans == 0)
            System.out.println(1);
        else {
            System.out.println(ans);
            for (int i = 1; i <= n; i++)
                System.out.print(a[i] + " ");
        }
    }
}
