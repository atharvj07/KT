#include <bits/stdc++.h>
using namespace std;
 
#define ll long long
const int MX = 2e5 + 9;
ll a[MX];
ll cnt[MX];
ll n;
vector<pair<ll, ll>> ans;
 
void solve() {
    memset(cnt, 0, sizeof(cnt));
    ans.clear();
    for(int i = 0; i < n; ++i) cnt[a[i]]++;
    for(int i = 0; i < n; ++i) {
        if(cnt[i] == 0) {
            ll len = 0;
            for(int j = i + 1; j < n; ++j) {
                if(cnt[j] == 0) len++;
                else break;
            }
            if(len){
                if(len == n - i - 1){
                    ans.push_back({len + 1, n - len - 1});
                }
            }
            else {
                ll len = 0;
                for(int j = i - 1; j >= 0; --j) {
                    if(cnt[j] == 0) len++;
                    else break;
                }
                if(len){
                    if(len == i){
                        ans.push_back({i + 1, n - i - 1});
                    }
                }
            }
        }
    }
    cout << ans.size() << '\n';
    for(auto x : ans) cout << x.first << ' ' << x.second << '\n';
}
 
int main() {
    int tc;
    cin >> tc;
    while(tc--){
        cin >> n;
        for(int i = 0; i < n; ++i) cin >> a[i];
        solve();
    }
}
