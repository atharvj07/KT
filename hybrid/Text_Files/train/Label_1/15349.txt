#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, neg = 0, zero = 0, pos = 0, a[105];
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (a[i] > 0) pos++;
        else if (a[i] == 0) zero++;
        else neg++;
    }
    cout << neg << ' ';
    for (int i = 0; i < n; i++)
        if (a[i] < 0) cout << a[i] << ' ';
    cout << '\n' << pos << ' ';
    for (int i = 0; i < n; i++)
        if (a[i] > 0) cout << a[i] << ' ';
    cout << '\n' << zero << ' ';
    for (int i = 0; i < n; i++)
        if (a[i] == 0) cout << a[i] << ' ';
    cout << '\n';
    return 0;
}
