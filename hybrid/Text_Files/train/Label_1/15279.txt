#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ll n, k;
    cin >> n >> k;
    vector<pair<ll, pair<ll, ll>>> v(n);
    for (auto &p : v) { cin >> p.second.first >> p.second.second >> p.first; }
    sort(v.begin(), v.end());
    ll ans = 0, last = 0, cur = 0;
    for (const auto &p : v) {
        if (last <= p.second.first) {
            ans += ceil(1.0 * p.first / k) * k;
            last = p.second.second;
            cur = p.second.first;
        } else {
            last = p.second.second;
            cur = p.second.first;
        }
    }
    cout << ans << endl;
}
