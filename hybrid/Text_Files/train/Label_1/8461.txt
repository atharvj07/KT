#include <bits/stdc++.h>
using namespace std;

const int MAXN = 105;

int n;
vector<int> a(MAXN);
int f[MAXN][5005];

int dp(int pos, int coins) {
    if(pos > n) {
        if(coins == 0) return 0;
        return -1;
    }
    if(f[pos][coins] != -1) return f[pos][coins];
    int ans = -1;
    if(coins >= a[pos]) {
        ans = dp(pos + 1, coins - a[pos]) + 1;
        if(ans != -1) {
            return f[pos][coins] = ans;
        }
    }
    if(coins >= a[2 * pos]) {
        ans = dp(pos + 1, coins - a[2 * pos]) + 1;
        if(ans != -1) {
            return f[pos][coins] = ans;
        }
    }
    if(coins >= a[2 * pos + 1]) {
        ans = dp(pos + 1, coins - a[2 * pos + 1]) + 1;
        if(ans != -1) {
            return f[pos][coins] = ans;
        }
    }
    return f[pos][coins] = ans;
}

int main() {
    int coins = 0;
    cin >> n;
    for(int i = 1; i <= n; i++) {
        cin >> a[i];
        coins += a[i];
    }
    memset(f, -1, sizeof(f));
    cout << dp(1, coins) << endl;
    return 0;
}
