#include <bits/stdc++.h>
using namespace std;
#define ll long long
vector<ll>suf_max;
ll psum[101][101];
string get_min_lex(vector<ll>&v, ll k){
    string ans="";
    ll sum=0;
    for(auto x:v){
        if(sum+x<=k){
            sum+=x;
            ans.push_back('1');
        }
        else{
            ans.push_back('0');
        }
    }
    return ans;
}
int main(){
    ll n,k,pos,m;
    cin>>n;
    ll arr[n+1];
    vector<ll>v;
    vector<string>vec;
    for(ll i=0;i<n;i++){
        cin>>arr[i];
    }
    for(ll i=n-1;i>=0;i--){
        if(i==n-1){
            suf_max.push_back(arr[i]);
        }
        else{
            suf_max.push_back(max(arr[i],suf_max.back()));
        }
    }
    reverse(suf_max.begin(),suf_max.end());
    for(ll i=0;i<n;i++){
        for(ll j=i;j<n;j++){
            if(i==j){
                psum[i][j]=arr[i];
            }
            else{
                psum[i][j]=psum[i][j-1]+arr[j];
            }
        }
    }
    for(ll i=1;i<=n;i++){
        v.push_back(suf_max[i-1]);
        vec.push_back(get_min_lex(v,i));
    }
    cin>>m;
    while(m--){
        cin>>k>>pos;
        cout<<psum[n-k][n-1]-psum[n-k][n-k+pos-1]<<endl;
    }
    return 0;
}
