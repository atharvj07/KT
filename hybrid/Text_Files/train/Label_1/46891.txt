import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;

public class Main {
    private static class Graph {
        private List<List<Integer>> graph;
        private List<Integer> degree;

        public Graph(int n) {
            graph = new ArrayList<>();
            degree = new ArrayList<>();
            for (int i = 0; i <= n; i++) {
                graph.add(new ArrayList<>());
                degree.add(0);
            }
        }

        public void addEdge(int u, int v) {
            graph.get(u).add(v);
            graph.get(v).add(u);
            degree.set(u, degree.get(u) + 1);
            degree.set(v, degree.get(v) + 1);
        }

        public List<Integer> getNeighbors(int u) {
            return graph.get(u);
        }

        public int getDegree(int u) {
            return degree.get(u);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(reader.readLine());
        for (int i = 0; i < t; i++) {
            String[] line = reader.readLine().split(" ");
            int n = Integer.parseInt(line[0]);
            int m = Integer.parseInt(line[1]);
            int k = Integer.parseInt(line[2]);
            Graph graph = new Graph(n);
            for (int j = 0; j < m; j++) {
                line = reader.readLine().split(" ");
                int u = Integer.parseInt(line[0]);
                int v = Integer.parseInt(line[1]);
                graph.addEdge(u, v);
            }
            List<Integer> clique = findClique(graph, k);
            List<Integer> subset = findSubset(graph, k);
            if (clique != null) {
                System.out.println("2");
                for (int vertex : clique) {
                    System.out.print(vertex + " ");
                }
                System.out.println();
            } else if (subset != null) {
                System.out.println("1 " + subset.size());
                for (int vertex : subset) {
                    System.out.print(vertex + " ");
                }
                System.out.println();
            } else {
                System.out.println("-1");
            }
        }
    }

    private static List<Integer> findClique(Graph graph, int k) {
        for (int vertex = 1; vertex <= graph.graph.size() - 1; vertex++) {
            if (graph.getDegree(vertex) < k) {
                continue;
            }
            Queue<Integer> queue = new LinkedList<>();
            queue.add(vertex);
            List<Integer> clique = new ArrayList<>();
            while (!queue.isEmpty()) {
                int u = queue.poll();
                clique.add(u);
                for (int v : graph.getNeighbors(u)) {
                    if (!clique.contains(v) && graph.getDegree(v) >= k) {
                        queue.add(v);
                    }
                }
            }
            if (clique.size() == k) {
                return clique;
            }
        }
        return null;
    }

    private static List<Integer> findSubset(Graph graph, int k) {
        for (int vertex = 1; vertex <= graph.graph.size() - 1; vertex++) {
            int count = 0;
            for (int neighbor : graph.getNeighbors(vertex)) {
                if (graph.getDegree(neighbor) >= k) {
                    count++;
                }
            }
            if (count >= k) {
                List<Integer> subset = new ArrayList<>();
                subset.add(vertex);
                return subset;
            }
        }
        return null;
    }
}
