#include<iostream>
#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
typedef long long int ll;
map<ll, ll> mp;
const ll maxN = 72;
ll power(ll x, ll y) {
	if (y == 0) return 1;
	x %= mod;
	if (y % 2) return (x * power(x, y - 1)) % mod;
	return power((x * x) % mod, y / 2) % mod;
}
int main() {
	ll n;
	cin >> n;
	ll ans = 0;
	for (ll i = 0; i < n; i++) {
		ll x;
		cin >> x;
		mp[x]++;
	}
	for (auto it : mp) {
		ll x = it.first;
		ll y = it.second;
		ans = (ans + power(2, y) - 1 + mod) % mod;
		// cout << x << " " << y << " " << power(2, y) << endl;
	}
	cout << ans << endl;
	return 0;
}
