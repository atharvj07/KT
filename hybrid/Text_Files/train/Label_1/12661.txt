#include <iostream>
#include <cmath>
using namespace std;
int main() {
    int T; cin >> T;
    for (int i = 0; i < T; i++) {
        long long n, d; cin >> n >> d;
        d = (long long)ceil((double)d / (n - 1));
        if (d <= n) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
}
