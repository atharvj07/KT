#include <bits/stdc++.h>
using namespace std;
long long n, k, a[1005], b[1005], x;
int main() {
    cin >> n >> k;
    for (long long i = 0; i < n; i++) cin >> a[i];
    for (long long i = 0; i < n; i++) cin >> b[i];
    for (long long i = 0; i < n; i++) {
        if (b[i] % a[i]) x = x + (a[i] - (b[i] % a[i]));
    }
    if (x <= k) cout << b[n - 1] / a[n - 1] + k / x;
    else cout << b[n - 1] / a[n - 1];
    return 0;
}
