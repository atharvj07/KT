#include <bits/stdc++.h>

using namespace std;
using ll = long long;
template<class T> using v = vector<T>;

const int MX = 2e5 + 1, LG = 18;

int up[LG][MX], par[MX], n;

int find_par(int x) {
    return x == par[x] ? x : par[x] = find_par(par[x]);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;
    iota(par, par + n + 1, 0);
    iota(up[0], up[0] + n + 1, 0);
    v<pair<int, int>> edges;
    for (int i = 1; i <= 2 * n; i++) {
        int a, b;
        cin >> a >> b;
        if (find_par(a) == find_par(b)) {
            cout << "-1\n";
            return 0;
        }
        edges.emplace_back(a, b);
        par[find_par(a)] = par[find_par(b)];
    }
    for (auto& e : edges) {
        int a = e.first, b = e.second;
        for (int j = LG - 1; j >= 0; j--) {
            if (up[j][b]) {
                b = up[j][b];
            }
        }
        up[0][a] = b;
        for (int j = 1; j < LG; j++) {
            up[j][a] = up[j - 1][up[j - 1][a]];
        }
    }
    for (int i = 1; i <= n; i++) {
        int now = i;
        for (int j = LG - 1; j >= 0; j--) {
            if (up[j][now]) {
                now = up[j][now];
            }
        }
        cout << now << ' ';
    }
    cout << '\n';
}
