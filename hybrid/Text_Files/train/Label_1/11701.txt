#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, d;
    cin >> n >> k >> d;
    vector<int> shows(n);
    for (int i = 0; i < n; i++) {
      cin >> shows[i];
    }
    int cnt = 0;
    int last = 0;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (shows[i] != last) {
        cnt = 1;
        last = shows[i];
      } else {
        cnt++;
      }
      if (cnt >= d) {
        ans++;
        cnt = 0;
        last = 0;
      }
    }
    cout << ans << endl;
  }
}
