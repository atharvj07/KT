#include <bits/stdc++.h>
using namespace std;
int main() {
	string s;
	cin >> s;
	string t = "abcdefghijklmnopqrstuvwxyz";
	int pos = 0;
	for (char c : s) {
		if (c > t[pos]) break;
		if (c == t[pos]) ++pos;
	}
	if (pos == t.size()) {
		cout << t << '\n';
	} else {
		cout << "-1\n";
	}
	return 0;
}
