#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n), t(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) cin >> t[i];
    int ans = 0, sum = 0;
    for (int i = 0; i < k; ++i) sum += a[i] * t[i];
    for (int i = k; i < n; ++i) sum = sum + a[i] * t[i] - a[i - k] * t[i - k];
    ans = max(ans, sum);
    for (int i = 1; i <= n - k; ++i) sum = sum + a[i - 1] * t[i - 1] - a[i + k - 1] * t[i + k - 1];
    ans = max(ans, sum);
    cout << ans << '\n';
    return 0;
}
