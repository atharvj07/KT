#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k, m, a;
  cin >> n >> k >> m >> a;

  vector<int> votes(n + 1, 0);
  vector<int> last_vote(n + 1, 0);

  for (int i = 0; i < a; i++) {
    int candidate;
    cin >> candidate;
    votes[candidate]++;
    last_vote[candidate] = i;
  }

  vector<pair<int, int>> sorted_candidates;

  for (int i = 1; i <= n; i++) {
    sorted_candidates.push_back({votes[i], last_vote[i]});
  }

  sort(sorted_candidates.begin(), sorted_candidates.end(), greater<pair<int, int>>());

  vector<int> result(n + 1, 3);

  int top_k_votes = sorted_candidates[k - 1].first;

  for (int i = 0; i < n; i++) {
    if (sorted_candidates[i].first >= top_k_votes) {
      result[i + 1] = 1;
    } else if (sorted_candidates[i].first > 0) {
      result[i + 1] = 2;
    }
  }

  for (int i = 1; i <= n; i++) {
    cout << result[i] << " ";
  }
  cout << endl;

  return 0;
}
