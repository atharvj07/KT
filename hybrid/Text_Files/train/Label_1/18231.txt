#include <iostream>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    int c[n], k[m];
    for (int i = 0; i < n; i++) cin >> c[i];
    for (int i = 0; i < m; i++) cin >> k[i];
    int l = 0, r = 0, t[m] = {};
    for (int i = 0; i < n; i++) {
        t[c[i] - 1]++;
        while (l <= r && t[c[i] - 1] > k[c[i] - 1]) t[c[l++] - 1]--;
        if (l <= r) {
            for (int j = 0; j < m; j++)
                if (t[j] < k[j]) { l = i + 1; break; }
            if (l > r)
                r = i;
        }
    }
    cout << (l <= r ? "YES" : "NO") << endl;
    return 0;
}
