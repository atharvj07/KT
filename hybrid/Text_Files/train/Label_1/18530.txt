#include <iostream>
#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int z, o, t, tw;
  scanf("%d %d %d %d", &z, &o, &t, &tw);
  if (z + t < o + tw) {
    puts("Impossible");
    return 0;
  }
  vector<string> res;
  int k = o + z - t - tw;
  for (int i = 0; i < z + k; i++) res.push_back("0");
  for (int i = 0; i < o + k; i++) res.push_back("1");
  for (int i = z + k; i < z + k + t; i++) res.push_back("0");
  for (int i = o + k; i < o + k + tw; i++) res.push_back("1");
  random_shuffle(res.begin(), res.end());
  for (string s : res) cout << s;
  return 0;
}
