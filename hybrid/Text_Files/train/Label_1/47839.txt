import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long v = sc.nextLong();
        long b = sc.nextLong();
        long w = sc.nextLong();
        long t = sc.nextLong();
        long dist = Math.abs(b-a);
        long time = dist/(v+w);
        if(time<=t){
            System.out.println("YES");
        }else{
            System.out.println("NO");
        }
    }
}
