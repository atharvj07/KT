#include <bits/stdc++.h>
typedef long long ll;
const int mod = 1e9 + 7;
int main() {
  int n;
  std::cin >> n;
  std::vector<int> p(2 * n + 1);
  std::vector<int> in(2 * n + 1, 0);
  for (int i = 0; i < n; ++i) {
    int a, b;
    std::cin >> a >> b;
    p[a] = b;
    p[b] = a;
    in[a] = 1;
    in[b] = 1;
  }
  int c = 0;
  for (int i = 1; i <= 2 * n; ++i) c += in[i];
  c = c / 2;
  ll ans = 1;
  for (int i = 1; i <= c; ++i) ans = ans * i % mod;
  std::cout << ans << std::endl;
  return 0;
}
