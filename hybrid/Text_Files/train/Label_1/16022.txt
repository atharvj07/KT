#include <cstdio>
#include <set>
#include <vector>
using namespace std;
int main() {
  set<int> a, b;
  int n, m, x;
  vector<int> ans;
  scanf("%d%d", &n, &m);
  while (n--) {
    scanf("%d", &x);
    a.insert(x);
  }
  while (m--) {
    scanf("%d", &x);
    b.insert(x);
  }
  m = 0x3f3f3f3f, n = 0;
  for (int x : a) { m = min(x, m), n++; }
  for (int x : b) { m = min(x, m), n++; }
  while (n--) {
    ans.push_back(m);
    m++;
  }
  for (int x : ans) printf("%d", x);
  return 0;
}
