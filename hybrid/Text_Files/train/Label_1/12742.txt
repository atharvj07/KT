#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> show(n);
    for (int i = 0; i < n; i++) {
        cin >> show[i].first >> show[i].second;
    }
    sort(show.begin(), show.end());
    int i = 0, j = 0;
    while (i < n && j < n) {
        if (show[i].second <= show[j].first) {
            i++;
            continue;
        }
        if (show[j].second <= show[i].first) {
            j++;
            continue;
        }
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    return 0;
}
