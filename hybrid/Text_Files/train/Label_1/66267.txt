n, m = map(int, input().split())
mines = input()
for i in range(1, n - 1):
    line = input()
    if line[0] == '*' or line[-1] == '*':
        mines += "*"
    else:
        mines += str(int(line[0]) + int(line[-1]))
    for j in range(1, m - 1):
        if line[j] == '*':
            mines += '*'
        else:
            cnt = 0
            a = input()[j]
            if a == '*':
                cnt += 1
            b = mines[i - 1][j]
            if b == '*':
                cnt += 1
            if mines[i - 1][j - 1] == '*':
                cnt += 1
            if mines[i - 1][j + 1] == '*':
                cnt += 1
            if mines[i + 1][j - 1] == '*':
                cnt += 1
            if mines[i + 1][j] == '*':
                cnt += 1
            if mines[i + 1][j + 1] == '*':
                cnt += 1
            if line[j] == str(cnt):
                mines += line[j]
            else:
                mines += '.'
        mines += input()[j]
if input()[0] == '*' or input()[-1] == '*':
    mines += "*"
else:
    mines += str(int(input()[0]) + int(input()[-1]))
print("YES" if mines.count('*') == n * m // 6 else "NO")
