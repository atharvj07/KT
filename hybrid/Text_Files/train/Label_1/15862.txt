#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    if (a[0] || a[n - 1])
      cout << "NO\n";
    else {
      cout << "YES\n";
      vector<int> ans;
      for (int i = 1; i < n - 1; i++) {
        if (a[i]) {
          ans.push_back(i);
          a[i - 1] ^= 1;
          a[i + 1] ^= 1;
          a[i] = 0;
        }
      }
      cout << ans.size() << endl;
      for (int i : ans) cout << i << " ";
      cout << "\n";
    }
  }
  return 0;
}
