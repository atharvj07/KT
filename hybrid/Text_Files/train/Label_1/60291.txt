def bit(l, r):
    while l < r:
        swp[l], swp[r] = swp[r], swp[l]
        l += 1
        r -= 1
 
n, q, m = map(int, input().split())
a = list(map(int, input().split()))
swp = [0] * (n+1)
for i in range(q):
    t, l, r = map(int, input().split())
    if t == 1:
        l -= 1
        r -= 1
        bit(l, r)
    else:
        bit(l-1, r-1)
        swp[l-1], swp[r] = swp[r], swp[l-1]
for b in map(int, input().split()):
    print(swp[b-1] or a[b-1])
