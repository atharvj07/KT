#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ll n, m, k;
    cin >> n >> m >> k;
    vector<ll> b(n);
    for (auto &x : b) cin >> x;
    ll last = 0;
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
        if (last >= b[i]) continue;
        ll start = (i ? b[i - 1] + 1 : 1);
        ll end = b[i];
        ll len = end - start + 1;
        if (len <= k) {
            ans += len;
            last = end;
        } else {
            ll rem = k;
            if (start + rem - 1 >= last) {
                ans += rem;
                last = start + rem - 1;
            }
        }
    }
    if (m > last) {
        ll len = m - last;
        if (len <= k) ans += len;
    }
    cout << ans << endl;
}
