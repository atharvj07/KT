#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define ll long long
#define pii pair<int, int>
#define all(x) x.begin(), x.end()
const int maxn = 2e5 + 5;
vector<int> a, pre[maxn];
int main() {
    int n, m;
    cin >> n >> m;
    a.resize(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
        pre[a[i]].pb(i);
    }
    for (int i = 0; i < m; ++i) {
        int l, r, x;
        cin >> l >> r >> x;
        l--, r--;
        if (a[l] != x)
            cout << l + 1 << '\n';
        else if (a[r] != x)
            cout << r + 1 << '\n';
        else {
            int p1 = upper_bound(all(pre[x]), l) - pre[x].begin();
            if (p1 == (int)pre[x].size())
                cout << -1 << '\n';
            else
                cout << pre[x][p1] + 1 << '\n';
        }
    }
}
