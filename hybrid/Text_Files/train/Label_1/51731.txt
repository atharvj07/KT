import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] p = new String[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.next();
        }
        long res = 0, len = 0, ans = 0;
        for (int i = 0; i < n; i++) {
            for (char c : p[i].toCharArray()) {
                if (len > 0 && c != p[i].charAt((int) (len - 1))) {
                    res = Math.max(res, len);
                    len = 0;
                }
                len++;
            }
            len = Math.max(len, 1);
            res = Math.max(res, len);
            ans += len;
            len = 0;
        }
        System.out.println(ans + res);
    }
}
