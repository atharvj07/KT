#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const ll inf = 1e18;

int main() {
    ll n, ans = inf;
    cin >> n;
    unordered_map<ll, ll> m;
    while (n--) {
        ll a, c;
        cin >> a >> c;
        // assume that pref[a][c] is the minimum cost to get a city with beauty a and price floor c
        if (m.find(c) == m.end()) { m[c] = inf; }
        ans = min(ans, m[c] + a);
        // we can get a city with beauty a and price floor c by flying from a city with beauty x and price floor c
        m[c] = min(m[c], a);
    }
    cout << ans << "\n";
}
