#include <iostream>
#include <limits>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    long long a, b, c, d;
    std::cin >> a >> b >> c >> d;
    long long totalTime = b;
    while (totalTime < a) { totalTime += c + d; }
    totalTime -= d;  // remove unnecessary falling asleep time
    if (totalTime >= a) { std::cout << totalTime << '\n'; } else { std::cout << -1 << '\n'; }
  }
  return 0;
}
