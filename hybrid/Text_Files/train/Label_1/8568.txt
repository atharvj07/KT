#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, t1, w1;
    long long ans = 0, sum = 0;
    cin >> n;
    cin >> t1 >> w1;
    vector<pair<long long, long long>> v;
    for (int i = 0; i < n - 1; ++i) {
        long long t, w;
        cin >> t >> w;
        v.emplace_back(w - t, w);
    }
    sort(v.begin(), v.end());
    for (int i = 0; i < v.size(); ++i) {
        if (sum + v[i].first > w1) break;
        sum += v[i].first;
        ans += v[i].second;
    }
    cout << ans + 1 << endl;
}
