#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;
    std::vector<std::pair<int, int>> exams(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> exams[i].first >> exams[i].second;
    }
    std::sort(exams.begin(), exams.end());
    int day = exams[0].second;
    for (int i = 1; i < n; ++i) {
        day = std::max(day + 1, exams[i].first);
    }
    std::cout << day << std::endl;
    return 0;
}
