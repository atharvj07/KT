#include <iostream>
#include <vector>
#include <queue>

using namespace std;

int main() {
  int n, b;
  cin >> n >> b;
  vector<pair<long long, long long>> queries(n);
  for (int i = 0; i < n; i++) {
    cin >> queries[i].first >> queries[i].second;
  }
  priority_queue<pair<long long, long long>, vector<pair<long long, long long>>, greater<pair<long long, long long>>> pq;
  long long time = 0;
  for (int i = 0; i < n; i++) {
    if (pq.empty() || pq.top().first <= queries[i].first) {
      time = max(time, queries[i].first);
      pq.push({time + queries[i].second, i});
    } else if (pq.size() < b) {
      pq.push({pq.top().first + queries[i].second, i});
    } else {
      cout << -1 << endl;
      continue;
    }
    cout << pq.top().first << endl;
    pq.pop();
  }
}
