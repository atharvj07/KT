#include<bits/stdc++.h>
using namespace std;
#define pb push_back
typedef long long ll;

const int N=1e5+5;
int n,x,k;

int main() {
    cin>>n>>k>>x;
    vector<int> a(n);
    for(int i=0; i<n; i++) cin>>a[i];
    ll sum=0;
    int idx=-1;
    for(int i=0; i<k; i++) {
        idx=i;
        sum+=x;
    }
    if(idx==-1) {
        for(int i=0; i<n; i++) sum+=a[i];
    }
    else {
        for(int i=0; i<n; i++) {
            if(i<=idx) continue;
            sum+=a[i];
        }
    }
    cout<<sum<<"\n";
    return 0;
}
