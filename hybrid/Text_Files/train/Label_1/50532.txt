import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int ntest = Integer.parseInt(br.readLine());
    while (ntest-- > 0) {
      String[] t = br.readLine().split(" ");
      int n = Integer.parseInt(t[0]), m = Integer.parseInt(t[1]);
      int[] cost = {Integer.parseInt(t[2]), Integer.parseInt(t[3])};
      char[][] c = new char[n][];
      for (int i = 0; i < n; i++) c[i] = br.readLine().toCharArray();
      int ans = 0;
      if (c[0][0] == '.') {
        int col = 0;
        for (; col < m && c[0][col] == '.'; col++) ;
        ans += (col + 1) / 2 * cost[1];
        for (int i = 1; i < n; i++) {
          if (c[i][col] == '.') {
            ans += cost[0];
            col++;
            for (; col < m && c[i][col] == '.'; col++) ;
            ans += (col - (i % 2 == col % 2 ? 1 : 0)) / 2 * cost[1];
          }
        }
      } else {
        ans += cost[0];
        for (int i = 1; i < n; i++) {
          for (int j = 0; j < m; j++) {
            if (c[i][j] == '.' && c[i - 1][j] == '.') ans += cost[0];
          }
        }
      }
      System.out.println(ans);
    }
  }
}
