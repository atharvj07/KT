#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<vector<int>> a(n, vector<int>(5));
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < 5; j++) { cin >> a[i][j]; }
    }
    bool ok = false;
    for (int i = 0; i < 5; i++) {
      for (int j = i + 1; j < 5; j++) {
        int cnt = 0;
        for (int k = 0; k < n; k++) { cnt += (a[k][i] && a[k][j]); }
        if (cnt == n / 2) ok = true;
      }
    }
    cout << (ok ? "YES" : "NO") << '\n';
  }
}
