import sys

q, m = map(int, input().split())

last_good = 0
dm = [[], []]
spells = []

for i in range(1, q + 1):
    k, a, b = map(int, input().split())

    if k == 1:
        a += last_good
        b += last_good
        spells.append((a % 1000000 + 1, b % 1000000 + 1))
    
    else:
        a += last_good
        b += last_good

        t = a % 1000000 + 1
        h = b % 1000000 + 1

        winner = "NO"

        for d, mn in dm:
            for dd, mm in spells:
                if mm * t <= m and h <= dd * t:
                    winner = "YES"
                    last_good = i
                    break
            else:
                continue
            break

        dm = [[]]

        for d, mn in spells:
            if d * t <= h <= dd * t:
                dm[-1].append((d, mn))
            else:
                if d * t > h:
                    dm.append([(d, mn)])
                
        print(winner)
