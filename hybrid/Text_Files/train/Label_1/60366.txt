from collections import Counter
n,m=map(int,input().split())
l=sorted(list(Counter(map(int,input().split())).keys()))
if m*8<n*32:
    m=m*8-n*32
    ans=(l[0]==l[1])+(l[-1]==l[-2])
    if m<0:
        if l[1]-l[0]>l[-1]-l[-2]:
            ans+=max(0,-m//2)
            l[1]-=m//2
        else:
            ans+=max(0,-m//2)
            l[-2]+=m//2
    print(ans)
else:
    print(0)
