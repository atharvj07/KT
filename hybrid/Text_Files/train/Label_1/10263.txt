#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> f(n);
    for (int i = 0; i < n; i++) { cin >> f[i]; }

    sort(f.begin(), f.end());
    int cur = 0;
    int res = 0;
    multiset<int> s;
    for (auto x : f) {
        auto it = s.upper_bound(x);
        int t;
        if (it == s.end())
            t = x - cur, s.insert(x), cur = x + 1;
        else
            t = x - *it, s.insert(x), s.erase(it), cur = x + 1;
        res += t;
    }
    res += cur;
    cout << res << endl;
    return 0;
}
