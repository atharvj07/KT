import java.io.*;
import java.util.*;
class Main {
    static int[][] points;
    static int n, k;
    static boolean[] fear;
    static long ccw(int[] a, int[] b, int[] c) {
        return (long)(b[0] - a[0]) * (c[1] - a[1]) - (long)(b[1] - a[1]) * (c[0] - a[0]);
    }
    static boolean pointOnSegment(int[] a, int[] b, int[] c) {
        return ccw(a, b, c) == 0 &&
            (long)(a[0] - c[0]) * (b[0] - c[0]) <= 0 &&
            (long)(a[1] - c[1]) * (b[1] - c[1]) <= 0;
    }
    static boolean isIntersect(int[] a, int[] b, int[] c, int[] d) {
        return ccw(a, b, c) * ccw(a, b, d) <= 0 &&
               ccw(c, d, a) * ccw(c, d, b) <= 0;
    }
    static void getFear() {
        int[] a = new int[2];
        int[] b = new int[2];
        int[] c = new int[2];
        for (int j = n; j < n + k; j++) {
            a[0] = points[j][0];
            a[1] = points[j][1];
            for (int i = 0; i < n; i++) {
                b[0] = points[i][0];
                b[1] = points[i][1];
                for (int l = 0; l <= i; l++) {
                    c[0] = points[l][0];
                    c[1] = points[l][1];
                    if (pointOnSegment(a, b, c)) {
                        fear[i] = true;
                        break;
                    }
                }
                if (fear[i]) {
                    continue;
                }
                for (int l = 0; l <= i; l++) {
                    c[0] = points[l][0];
                    c[1] = points[l][1];
                    if (isIntersect(a, b, b, c)) {
                        fear[i] = true;
                        break;
                    }
                }
            }
        }
    }
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        k = Integer.parseInt(st.nextToken());
        n = Integer.parseInt(st.nextToken());
        fear = new boolean[n];
        points = new int[n + k][2];
        for (int i = 0; i < k; i++) {
            st = new StringTokenizer(br.readLine());
            points[i][0] = Integer.parseInt(st.nextToken());
            points[i][1] = Integer.parseInt(st.nextToken());
        }
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            points[i + k][0] = Integer.parseInt(st.nextToken());
            points[i + k][1] = Integer.parseInt(st.nextToken());
        }
        getFear();
        int res = 0;
        for (boolean a : fear) {
            if (a) {
                res++;
            }
        }
        System.out.println(res);
    }
}
