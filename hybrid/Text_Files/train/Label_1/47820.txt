import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        while (q-- > 0) {
            int xp0 = scanner.nextInt();
            int yp0 = scanner.nextInt();
            int xp1 = scanner.nextInt();
            int yp1 = scanner.nextInt();
            int xp2 = scanner.nextInt();
            int yp2 = scanner.nextInt();
            int xp3 = scanner.nextInt();
            int yp3 = scanner.nextInt();
            int dx = xp1 - xp0;
            int dy = yp1 - yp0;
            int dx2 = xp3 - xp2;
            int dy2 = yp3 - yp2;
            int denominator = dx * dy2 - dx2 * dy;
            double ua = (dx * (yp2 - yp0) - dy * (xp2 - xp0)) * 1.0 / denominator;
            double ub = (dx2 * (yp0 - yp2) - dy2 * (xp0 - xp2)) * 1.0 / denominator;
            if (ua >= 0 && ua <= 1 && ub >= 0 && ub <= 1) {
                System.out.println("1");
            } else {
                System.out.println("0");
            }
        }
    }
}
