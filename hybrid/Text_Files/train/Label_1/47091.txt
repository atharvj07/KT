import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        boolean[][] graph = new boolean[n][n];
        for (int i = 0; i < m; i++) {
            int x = in.nextInt() - 1;
            int y = in.nextInt() - 1;
            graph[x][y] = true;
            graph[y][x] = true;
        }
        int[] size = new int[n];
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (size[i] == 0) {
                count++;
                dfs(graph, i, size, count);
            }
        }
        System.out.println(count);
        for (int i = 1; i <= count; i++) {
            int cnt = 0;
            for (int j = 0; j < n; j++) {
                if (size[j] == i) {
                    cnt++;
                }
            }
            System.out.print(cnt + " ");
        }
    }

    private static void dfs(boolean[][] graph, int i, int[] size, int count) {
        if (size[i] != 0) {
            return;
        }
        size[i] = count;
        for (int j = 0; j < graph.length; j++) {
            if (!graph[i][j]) {
                dfs(graph, j, size, count);
            }
        }
    }
}
