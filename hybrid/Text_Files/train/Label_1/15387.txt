#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    long long ans = 0;
    for (int i = 0; i < n; i += 4) {
        if (i + 3 < n && a[i] == a[i + 1] && a[i + 2] == a[i + 3] && a[i] + 1 == a[i + 1]) {
            ans += 1LL * a[i] * a[i + 1];
        }
    }
    cout << ans << '\n';
    return 0;
}
