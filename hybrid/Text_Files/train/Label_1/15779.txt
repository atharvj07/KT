#include <iostream>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  string t;
  cin >> t;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (t[j] == t[(j + i) % n]) {
        cout << t[j];
        break;
      }
    }
  }
  cout << endl;
  return 0;
}
