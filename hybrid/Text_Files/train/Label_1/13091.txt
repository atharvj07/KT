#include<iostream>
#include<vector>
#include<algorithm>

using namespace std;

int main() {
    int n, x1, y1, x2, y2;
    cin >> n >> x1 >> y1 >> x2 >> y2;
    vector<pair<long long, long long>> flows;
    for(int i = 0 ; i < n; ++i) {
        int x, y;
        cin >> x >> y;
        flows.emplace_back((x - x1) * (x - x1) + (y - y1) * (y - y1), (x - x2) * (x - x2) + (y - y2) * (y - y2));
    }
    sort(flows.begin(), flows.end());
    long long ans = 0;
    for(int i = 0, j = n - 1; i < n; ++i) {
        while(j >= 0 && flows[i].second < flows[j].first) {
            --j;
        }
        if(i >= j) break;
        ans = flows[i].first + flows[j].second;
    }
    cout << ans << endl;
    return 0;
}
