n, q = map(int, input().split())
a = list(map(int, input().split()))

def calculate_difficulty(a):
    count = {}
    max_count = 0
    total_diff_blocks = 0
    last = None
    for i in a:
        if i in count:
            count[i] += 1
        else:
            count[i] = 1
        max_count = max(max_count, count[i])
        if i != last:
            total_diff_blocks += 1
            last = i
    return total_diff_blocks - max_count

print(calculate_difficulty(a))
