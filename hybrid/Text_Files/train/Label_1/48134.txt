import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = in.readLine().split(" ");
        int n = Integer.parseInt(tokens[0]);
        int k = Integer.parseInt(tokens[1]);
        int a = Integer.parseInt(tokens[2]);
        int b = Integer.parseInt(tokens[3]);
        int q = Integer.parseInt(tokens[4]);
        int[] orders = new int[n + 1];
        for (int i = 0; i < q; i++) {
            tokens = in.readLine().split(" ");
            int type = Integer.parseInt(tokens[0]);
            int day = Integer.parseInt(tokens[1]);
            int count = 0;
            if (type == 1) {
                count = Integer.parseInt(tokens[2]);
            }
            if (day + k <= n) {
                orders[day] += count;
                orders[day + k] -= count;
            }
        }
        int[] prefix = new int[n + 1];
        prefix[0] = orders[0];
        for (int i = 1; i <= n; i++) {
            prefix[i] = prefix[i - 1] + orders[i];
        }
        int[] max = new int[n + 1];
        int current = Math.min(a, b);
        for (int i = 1; i <= n; i++) {
            current += prefix[i];
            current = Math.min(current, a);
            max[i] = Math.max(max[i - 1], current);
        }
        for (int i = 1; i <= n - k + 1; i++) {
            System.out.println(max[i + k - 1]);
        }
    }
}
