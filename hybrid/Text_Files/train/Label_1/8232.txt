#include<bits/stdc++.h>
using namespace std ;
int main()
{
   long long  n,s,k,a[1005],t;
   cin >> t ;

   while(t--)
   {
      cin >> n >> s >> k ;
      for(int i=0 ; i<k ; i++) cin >> a[i] ;
      sort(a,a+k) ;
      bool f=1 ;
      long long ans=0 ;
      for(int i=0 ; i<k ; i++)
      {
         if(a[i]<=s)
         {
            if(s-a[i]<=k)
            {
               ans=max(ans,k-s+a[i]) ;
               f=0 ;
               break ;
            }
         }
         else
         {
            ans=max(ans,a[i]-s) ;
            f=0 ;
            break ;
         }
      }
      if(f) cout << 0 << endl ;
      else cout << ans << endl ;
   }
   return 0 ;
}
