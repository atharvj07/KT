#include <iostream>
#include <cmath>

using namespace std;
typedef signed long long int ll;

ll x, y, nx, ny;
ll n, m;
ll vx, vy;
bool wall[10001] = {false};

inline ll min(ll a, ll b) { return (a < b) ? a : b; }

int main()
{
    cin >> n >> m >> x >> y >> vx >> vy;
    if (vx == 0 && vy == 0) {
        cout << -1 << endl;
        return 0;
    } else if ((vx == 0) ^ (vy == 0)) {
        if (vx == 0) {
            nx = y;
            ll t = n - x;
            ll nt = (y / t) * t;
            if ((t - nt) == (y - nt)) {
                cout << x << " " << 0 << endl;
            } else {
                cout << x << " " << t << endl;
            }
        } else {
            nx = x;
            ll t = m - y;
            ll nt = (x / t) * t;
            if ((t - nt) == (x - nt)) {
                cout << 0 << " " << y << endl;
            } else {
                cout << t << " " << y << endl;
            }
        }
        return 0;
    } else if (abs(vx) == abs(vy)) {
        if (vx > 0 && vy > 0) {
            nx = n;
            ny = m;
            if (x < y) {
                wall[y - x] = true;
            } else if (y < x) {
                wall[n + m - x - y] = true;
            }
        } else if (vx > 0 && vy < 0) {
            nx = n;
            ny = 0;
            if (x < m - y) {
                wall[n - x + y] = true;
            } else if (m - y < x) {
                wall[x - (m - y)] = true;
            }
        } else if (vx < 0 && vy < 0) {
            nx = 0;
            ny = 0;
            if (n - x < m - y) {
                wall[m - y - n + x] = true;
            } else if (m - y < n - x) {
                wall[x - n + y + m] = true;
            }
        } else if (vx < 0 && vy > 0) {
            nx = 0;
            ny = m;
            if (n - x < y) {
                wall[y - x + n] = true;
            } else if (y < n - x) {
                wall[n + m - x - y] = true;
            }
        }
        ll t1 = n * m;
        ll t2 = 0;
        for (ll i = 1; i <= t1; i++) {
            if (wall[i]) {
                t2 = i;
            }
        }
        nx -= x;
        ny -= y;
        ll temp = (nx * nx) + (ny * ny);
        temp = sqrt(temp);
        t2 /= temp;
        x = nx * t2;
        y = ny * t2;
        while (true) {
            if (x == 0 || x == n || y == 0 || y == m) {
                cout << x << " " << y << endl;
                return 0;
            } else {
                if (nx > 0 && ny > 0) {
                    if (x < y) {
                        y -= x;
                        x = 0;
                        continue;
                    } else if (y < x) {
                        x -= y;
                        y = 0;
                        continue;
                    }
                } else if (nx > 0 && ny < 0) {
                    if (x < m - y) {
                        y += x;
                        x = 0;
                        continue;
                    } else if (m - y < x) {
                        x -= m - y;
                        y = m;
                        continue;
                    }
                } else if (nx < 0 && ny < 0) {
                    if (n - x < m - y) {
                        y += n - x;
                        x = n;
                        continue;
                    } else if (m - y < n - x) {
                        x += m - y;
                        y = 0;
                        continue;
                    }
                } else if (nx < 0 && ny > 0) {
                    if (n - x < y) {
                        y -= n - x;
                        x = n;
                        continue;
                    } else if (y < n - x) {
                        x += y;
                        y = m;
                        continue;
                    }
                }
            }
        }
    }
    cout << -1 << endl;
    return 0;
}
