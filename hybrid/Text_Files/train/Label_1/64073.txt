MOD = 10**9 + 7

def count_set_bits(n):
    count = 0
    while n:
        count += n & 1
        n >>= 1
    return count

def comb(n, r, fact, inv_fact):
    if r > n or r < 0:
        return 0
    return fact[n] * inv_fact[r] * inv_fact[n-r] % MOD

def precompute(n):
    fact = [1] * (n+1)
    for i in range(1, n+1):
        fact[i] = fact[i-1] * i % MOD
    inv_fact = [1] * (n+1)
    inv_fact[n] = pow(fact[n], MOD-2, MOD)
    for i in range(n-1, 0, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % MOD
    return fact, inv_fact

def special_numbers(n, k):
    if k == 0:
        return 1 if n == '1' else 0
    fact, inv_fact = precompute(len(n))
    total = 0
    set_bits = n.count('1')
    for i in range(len(n)):
        if n[i] == '1':
            if k-1 == 0:
                total = (total + 1) % MOD
            else:
                total += comb(set_bits-1, k-1, fact, inv_fact)
                total %= MOD
            set_bits -= 1
    if k == 1:
        total += 1
    dp = [0] * (len(n) + 1)
    dp[1] = 1
    for i in range(2, len(n) + 1):
        dp[i] = dp[count_set_bits(i)] + 1
    for i in range(1, len(n) + 1):
        if dp[i] == k and i < int(n, 2):
            total += 1
            total %= MOD
    return total

n = input().strip()
k = int(input().strip())
print(special_numbers(n, k))
