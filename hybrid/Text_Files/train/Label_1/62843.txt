n, m, p = map(int, input().split())
sequence = input()
operations = input()

# Stack to store the indexes of opening brackets
stack = []
# Pairing stores the index of the paired bracket for each bracket
pairing = {}

# Generate pairing information
for i, bracket in enumerate(sequence):
    if bracket == '(':
        stack.append(i)
    else:
        opening = stack.pop()
        pairing[i] = opening
        pairing[opening] = i

# Convert sequence to list for easier manipulation
sequence_list = list(sequence)

# Initialize cursor position (0-indexed)
cursor = p - 1

for operation in operations:
    if operation == 'L':
        cursor -= 1
    elif operation == 'R':
        cursor += 1
    elif operation == 'D':
        # Find the paired bracket's index
        paired_index = pairing[cursor]
        
        # Determine the range to be deleted
        start, end = sorted([cursor, paired_index])
        
        # Delete the brackets
        del sequence_list[start:end + 1]
        
        # Update cursor position after deletion
        if end + 1 < len(sequence_list):
            cursor = start
        else:
            cursor = start - 1

# Convert list back to string
final_sequence = ''.join(sequence_list)
print(final_sequence)
