import java.util.Scanner;
import java.util.HashSet;

public class Shockers {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        HashSet<Character> possibleLetters = new HashSet<>();
        HashSet<Character> excludedLetters = new HashSet<>();
        int excessiveShocks = 0;

        for (int i = 0; i < n; i++) {
            String action = scanner.nextLine();
            char firstChar = action.charAt(1);

            if (action.startsWith("!")) {
                // Electric shock
                possibleLetters.add(firstChar);
            } else if (action.startsWith(".")) {
                // No electric shock
                excludedLetters.add(firstChar);
            } else {
                // Guess
                char guess = firstChar;

                if (possibleLetters.size() == 1 && possibleLetters.contains(guess)) {
                    // Correct guess, no excessive shocks
                } else {
                    // Incorrect guess, count excessive shocks
                    for (char ch : possibleLetters) {
                        if (!excludedLetters.contains(ch) && ch != guess) {
                            excessiveShocks++;
                        }
                    }
                }

                // Update possible letters based on the guess
                possibleLetters.retainAll(excludedLetters);
                possibleLetters.add(guess);
            }
        }

        System.out.println(excessiveShocks);
    }
}
