n = int(input())
board = []
for i in range(n):
    board.append(input())
for x in range(n):
    for y in range(n):
        count = 0
        for i in range(max(0, x - 1), min(n, x + 2)):
            for j in range(max(0, y - 1), min(n, y + 2)):
                if board[i][j] == 'o':
                    count += 1
        if count % 2 == 1:
            print("NO")
            exit()
print("YES")
