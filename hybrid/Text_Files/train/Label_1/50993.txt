import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        // get the number of marbles
        long n = sc.nextLong();
        long MOD = 1000000007;
        // calculate the number of ways to pick a zebroid subsequence
        long ans = (n + 1) / 2 * (n + 2) / 2 % MOD;
        // print the answer
        System.out.println(ans);
    }
}
