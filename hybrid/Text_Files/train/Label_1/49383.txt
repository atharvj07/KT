import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int min = 1, max = n;
            while (min < max) {
                int mid = (min + max) / 2;
                System.out.println("? " + min + " " + mid);
                System.out.flush();
                String c = in.next();
                if (c.equals("<")) max = mid - 1;
                else if (c.equals("=")) {
                    System.out.println("! " + min + " " + min);
                    System.out.flush();
                    break;
                } else min = mid + 1;
            }
            if (min == max) {
                System.out.println("! " + min + " " + min);
                System.out.flush();
            }
        }
    }
}
