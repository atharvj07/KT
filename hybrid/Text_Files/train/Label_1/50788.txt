import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt() - 1;
        if (x == 0) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            int[] a = new int[2 * n - 1];
            for (int i = 0; i < n; i++) {
                a[i] = 2 * (n - i + 1) - 1;
            }
            for (int i = n - 1; i < 2 * n - 2; i++) {
                a[i + 1] = 2 * (n - i - 1);
            }
            a[2 * n - 2] = x + 1;
            for (int i : a) {
                System.out.println(i);
            }
        }
    }
}
