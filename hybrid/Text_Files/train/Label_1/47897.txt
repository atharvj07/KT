import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        if (k % 2 == 1 && n % 2 == 0 || k % 2 == 0 && n % 2 == 1 || k > n * (n - 1) / 2) {
            System.out.println(-1);
            return;
        }
        int[] a = new int[n];
        if (n % 2 == 1) {
            for (int i = 0; i < n; i++) {
                a[i] = i * 2 + 1;
            }
            k -= n * (n - 1) / 2;
            a[n - 1] += k;
        } else {
            for (int i = 0; i < n; i++) {
                a[i] = i * 2 + 2;
            }
            k -= n * (n - 1) / 2;
            a[n - 1] += k / 2;
            a[n - 2] += k / 2;
        }
        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
    }
}
