#include <bits/stdc++.h>
#define mod 1000000007
#define pb push_back
using namespace std;

int main() {
  int64_t n;
  cin >> n;
  string s;
  cin >> s;
  int64_t dp[n + 2][2]; // dp[i][0] = subarray begins from i, last was 'a'
                        // dp[i][1] = subarray begins from i, last was 'b'
  dp[n][0] = dp[n][1] = 1;
  for (int64_t i = n - 1; i >= 0; i--) {
    if (s[i] == 'a') {
      dp[i][0] = (dp[i + 1][0] + dp[i + 1][1]) % mod;
      dp[i][1] = 0;
    } else {
      dp[i][1] = (dp[i + 1][1] + dp[i + 1][0]) % mod;
      dp[i][0] = 0;
    }
  }
  cout << dp[0][0] << '\n';
}
