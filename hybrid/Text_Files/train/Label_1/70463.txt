def is_possible(n, m, li, ri, a):
    a.sort()
    for i in range(n - 1):
        if a and li[i] + 1 <= ri[i] and li[i + 1] <= ri[i + 1] and ri[i] - li[i] <= a[-1] and ri[i + 1] - li[i + 1] <= a[0]:
            a.pop()
            a.pop(0)
        else:
            return False
    return True

n, m = map(int, input().split())
li = []
ri = []
for i in range(n):
    l, r = map(int, input().split())
    li.append(l)
    ri.append(r)
a = list(map(int, input().split()))
if is_possible(n, m, li, ri, a):
    print("Yes")
    print(*[j + 1 for j in range(n - 1) if ri[j] - li[j] > a[-1]])
else:
    print("No")
