#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 5;
vector<int> adj[N];
int deg[N];

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
    deg[u]++;
    deg[v]++;
  }
  int buses = 0, rings = 0, stars = 0;
  for (int i = 1; i <= n; i++) {
    if (deg[i] > 1) {
      if (deg[i] == n - 1) stars++;
      else if (deg[i] == 2) buses++;
      else rings++;
    }
  }
  if (buses == 1 && rings == 0 && stars == 0) cout << "bus topology" << endl;
  else if (buses == 0 && rings == 1 && stars == 0) cout << "ring topology" << endl;
  else if (buses == 0 && rings == 0 && stars == 1) cout << "star topology" << endl;
  else cout << "unknown topology" << endl;
}
