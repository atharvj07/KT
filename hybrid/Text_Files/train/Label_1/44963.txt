import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt(), b = in.nextInt();
        int maxLen = 0;
        for (int base = 15; base > 1; base--) {
            int num = a + b;
            int len = 0;
            while (num > 0) {
                num /= base;
                len++;
            }
            maxLen = Math.max(maxLen, len);
        }
        System.out.println(maxLen);
    }
}
