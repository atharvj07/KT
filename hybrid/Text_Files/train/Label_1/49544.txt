import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        PriorityQueue<Long> pq = new PriorityQueue<>();
        for (int i = 1; i <= k; i++) {
            pq.offer(0L);
        }
        for (int i = 0; i < n; i++) {
            int s = sc.nextInt();
            int m = sc.nextInt();
            pq.offer(pq.poll() + m);
            System.out.println(pq.peek() + s);
        }
    }
}
