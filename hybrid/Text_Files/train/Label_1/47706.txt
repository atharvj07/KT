import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();
        int a = x % 4;
        char b = 'A';
        if (a == 3 || a == 1) {
            b = 'A';
            a = 0;
        } else if (a == 2) {
            b = 'C';
            a = 1;
        } else {
            b = 'B';
            a = 1;
        }
        System.out.println(a + " " + b);
    }
}
