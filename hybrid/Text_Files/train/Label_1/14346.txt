#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
typedef long long ll;
#define rep(i, n) for (int i = 0; i < (n); ++i)

const ll inf = 1e18;
struct edge {
    int to, w;
    edge() {}
    edge(int to, int w) : to(to), w(w) {}
};

vector<edge> g[400005];
ll dist[400005];

void dijkstra(int s) {
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll, int>>> pq;
    pq.push(make_pair(0, s));
    fill(dist, dist + 400005, inf);
    dist[s] = 0;
    while (!pq.empty()) {
        ll co = pq.top().first;
        int cu = pq.top().second;
        pq.pop();
        for (auto edge : g[cu]) {
            ll w = edge.w, to = edge.to;
            if (dist[cu] + w < dist[to]) {
                dist[to] = dist[cu] + w;
                pq.push(make_pair(dist[to], to));
            }
        }
    }
}

int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int k;
        cin >> k;
        for (int i = 0; i < 400005; ++i) g[i].clear();
        set<int> vis;
        for (int i = 0; i < 2 * k - 1; ++i) {
            int u, v, w;
            cin >> u >> v >> w;
            g[u].push_back(edge(v, w));
            g[v].push_back(edge(u, w));
            vis.insert(u);
            vis.insert(v);
        }
        dijkstra(*vis.begin());
        ll min = 0, max = 0;
        for (int i = 0; i < 400005; ++i) {
            if (vis.find(i) != vis.end()) {
                min += dist[i];
                max += dist[i];
            }
        }
        dijkstra(*vis.rbegin());
        for (int i = 0; i < 400005; ++i) {
            if (vis.find(i) != vis.end()) {
                max += dist[i];
            }
        }
        cout << min << " " << max << endl;
    }
    return 0;
}
