n = int(input())
L = list(map(int, input().split()))
l = 0; r = n - 1; k = 0; s = ''
while l <= r:
    if L[l] < L[r]:
        if L[l] > k:
            k = L[l]
            s += 'L'
            l += 1
        elif L[r] > k:
            k = L[r]
            s += 'R'
            r -= 1
        else: r -= 1
    else:
        if L[r] > k:
            k = L[r]
            s += 'R'
            r -= 1
        elif L[l] > k:
            k = L[l]
            s += 'L'
            l += 1
        else: l += 1
print(len(s))
print(s)
