#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, T;
    cin >> n >> T;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    vector<int> b;
    for (int x : a) {
      if (x <= T / 2) b.push_back(x);
    }
    sort(b.begin(), b.end());
    int m = b.size();
    vector<int> p(n);
    int l = 0, r = m - 1;
    for (int i = 0; i < n; i++) {
      while (l < r && b[l] + b[r] > T) r--;
      if (l < r && b[l] + b[r] == T) {
        p[i] = 1;
        l++;
      }
      if (a[i] > T / 2) p[i] = 1;
    }
    for (int x : p) cout << x << " ";
    cout << endl;
  }
  return 0;
}
