def solve_snakes(n, m, grid):
    def is_valid(snakes):
        test_grid = [['.' for _ in range(m)] for _ in range(n)]
        for idx, (r1, c1, r2, c2) in enumerate(snakes):
            if r1 == r2:  # Horizontal snake
                for c in range(min(c1, c2)-1, max(c1, c2)):
                    test_grid[r1-1][c] = chr(97 + idx)
            else:  # Vertical snake
                for r in range(min(r1, r2)-1, max(r1, r2)):
                    test_grid[r][c1-1] = chr(97 + idx)
        return all(test_grid[i][j] == grid[i][j] for i in range(n) for j in range(m))

    letters = {}
    for i in range(n):
        for j in range(m):
            if grid[i][j] != '.':
                if grid[i][j] not in letters:
                    letters[grid[i][j]] = [i+1, j+1, i+1, j+1]
                else:
                    letters[grid[i][j]][0] = min(letters[grid[i][j]][0], i+1)
                    letters[grid[i][j]][1] = min(letters[grid[i][j]][1], j+1)
                    letters[grid[i][j]][2] = max(letters[grid[i][j]][2], i+1)
                    letters[grid[i][j]][3] = max(letters[grid[i][j]][3], j+1)

    snakes = [letters[chr(97 + i)] for i in range(len(letters))]
    for snake in snakes:
        r1, c1, r2, c2 = snake
        if r1 != r2 and c1 != c2:
            return "NO\n"

    if not is_valid(snakes):
        return "NO\n"

    result = "YES\n"
    result += f"{len(snakes)}\n"
    for snake in snakes:
        result += " ".join(map(str, snake)) + "\n"
    return result

t = int(input().strip())
for _ in range(t):
    n, m = map(int, input().split())
    grid = [input().strip() for _ in range(n)]
    print(solve_snakes(n, m, grid), end='')
