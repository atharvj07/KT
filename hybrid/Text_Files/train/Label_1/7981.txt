#include <bits/stdc++.h>
using namespace std;
const int N = 32;
int n, cnt[N][N];
string ans;
int main() {
    cin >> n;
    while (n--) {
        string s;
        cin >> s;
        for (int i = 0; i < s.size(); i++) cnt[s[i] - 'a'][i] = 1;
    }
    for (int i = 0; i < N; i++) {
        int j = 0;
        while (j < N && cnt[i][j]) j++;
        if (j < N) {
            ans += char(i + 'a');
            break;
        }
    }
    if (ans.empty()) ans += 'a';
    cout << ans << '\n';
}
