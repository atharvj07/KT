#include <iostream>
using namespace std;
const int MOD = 998244353, MX = 2e5 + 5;
long long A[MX]; bool flag;
int main() {
    long long n, k, s = 0; cin >> n >> k;
    for (int i = 0; i < n; i++) {
        cin >> A[i];
        s += A[i];
    }
    if (s >= k) {
        cout << 0 << endl;
        return 0;
    }
    long long add = 0, i;
    for (i = 0; i < n; i++) {
        add += A[i];
        if (add >= k) {
            cout << i + 1 << endl;
            return 0;
        }
        A[i] = (A[i] + add) % MOD;
    }
    while (!flag) {
        add = 0;
        for (int i = 0; i < n; i++) {
            add += A[i];
            if (add >= k) {
                cout << n + i + 1 << endl;
                flag = 1;
                break;
            }
            A[i] = (A[i] + add) % MOD;
        }
        n += n;
    }
}
