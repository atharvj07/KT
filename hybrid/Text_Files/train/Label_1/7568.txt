#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> out(m);
  int p = 1;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < 2; ++j) {
      for (int k = 0; k < 4; ++k) {
        if (p <= m) { out[p - 1] = p; }
        ++p;
      }
      p += 2 * (n - i - 1);
    }
  }
  for (auto x : out) { cout << x << ' '; }
  return 0;
}
