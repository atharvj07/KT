import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] dp = new int[n];
        dp[n - 1] = a[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            if (a[i] == 1) {
                dp[i] = dp[i + 1] + 1;
            } else {
                dp[i] = 0;
            }
        }
        int max = 0;
        int end = 0;
        for (int i = 0; i < n; i++) {
            if (dp[i] > max) {
                max = dp[i];
                end = i;
            }
        }
        if (k == 0) {
            System.out.println(max);
            for (int i = 0; i < n; i++) {
                System.out.print(a[i]);
            }
            System.out.println();
        } else {
            int j = end;
            int count = 0;
            while (count < k && j >= 0) {
                if (a[j] == 0) {
                    a[j] = 1;
                    count++;
                }
                j--;
            }
            int ans = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] == 1) {
                    ans++;
                }
            }
            System.out.println(ans);
            for (int i = 0; i < n; i++) {
                System.out.print(a[i]);
            }
            System.out.println();
        }
    }
}
