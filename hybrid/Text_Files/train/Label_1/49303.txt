import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int q = Integer.parseInt(reader.readLine());
        for (int i = 0; i < q; i++) {
            String[] line = reader.readLine().split(" ");
            int n = Integer.parseInt(line[0]);
            int m = Integer.parseInt(line[1]);
            List<Customer> customers = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                line = reader.readLine().split(" ");
                customers.add(new Customer(Integer.parseInt(line[0]), Integer.parseInt(line[1]), Integer.parseInt(line[2])));
            }
            System.out.println(isPossible(m, customers) ? "YES" : "NO");
        }
    }

    private static boolean isPossible(int m, List<Customer> customers) {
        int temperature = m;
        for (Customer customer : customers) {
            if (customer.t == 0) {
                if (temperature < customer.l || temperature > customer.h) {
                    return false;
                }
            } else {
                if (temperature < customer.l) {
                    int steps = customer.l - temperature;
                    if (steps > customer.t) {
                        return false;
                    }
                    temperature += steps;
                } else if (temperature > customer.h) {
                    int steps = temperature - customer.h;
                    if (steps > customer.t) {
                        return false;
                    }
                    temperature -= steps;
                }
            }
        }
        return true;
    }

    private static class Customer {
        int t;
        int l;
        int h;

        public Customer(int t, int l, int h) {
            this.t = t;
            this.l = l;
            this.h = h;
        }
    }
}
