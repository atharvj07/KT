#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 1e9 + 7;
struct edge {
    int to, w;
};
vector<vector<edge>> g;
vector<int> ms;
int n;
void dfs1(int u, int p, int d = 0) {
    ms[u] = d;
    for (auto &e : g[u]) {
        if (e.to == p) continue;
        dfs1(e.to, u, d + e.w);
    }
}
int dfs2(int u, int p, int ans = 0) {
    int mx = -1e18, sum = ans;
    for (auto &e : g[u]) {
        if (e.to == p) continue;
        sum += e.w;
        int v = dfs2(e.to, u, ans + e.w);
        mx = max(mx, v);
        sum -= e.w;
    }
    mx = max(mx, sum);
    return ms[u] = mx;
}
int32_t main() {
    cin >> n;
    g.resize(n + 1), ms.resize(n + 1);
    unordered_map<int, int> mp;
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        mp[i] = x;
    }
    for (int i = 1; i < n; i++) {
        int x, y;
        cin >> x >> y;
        g[x].emplace_back((edge){y, mp[y]});
        g[y].emplace_back((edge){x, mp[x]});
    }
    dfs1(1, 1);
    int ans = -1e18;
    for (int i = 1; i <= n; i++) {
        ans = max(ans, dfs2(i, i));
    }
    if (ans < 0) cout << "Impossible" << "\n";
    else
        cout << ans << "\n";
}
