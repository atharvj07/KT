n=int(input())
a=[int(i) for i in input().split()]
b = [a[0]]
i=1
while(i<n-1):
    if a[i]==a[i-1] and a[i]==a[i+1]:
        i+=1
        if i+1==n:b.append(a[n-1])
    else:
        b.append(a[i])
        i+=1
        if i+1==n:b.append(a[n-1])
if len(b)==n:
    print(0)
    print(*b)
else:
    print(1)
