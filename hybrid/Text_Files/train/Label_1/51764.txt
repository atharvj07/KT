import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();
        int hh = sc.nextInt();
        int mm = sc.nextInt();
        int ans = (hh * 60 + mm - 7) / x + 1;
        System.out.println(ans);
    }
}
