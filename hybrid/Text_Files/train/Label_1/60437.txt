MOD = 1000000007

def factorial_mod(n):
    result = 1
    for i in range(1, n + 1):
        result = (result * i) % MOD
    return result

def solve(n, a):
    ones = a.count(1)
    twos = n - ones
    
    # For each student that can participate in 2 throws, we have two options: keep the ball or swap it.
    # Therefore, for 'twos' students, there are 2^twos ways to rearrange balls among these students.
    twos_ways = pow(2, twos, MOD)
    
    # For the students that can only participate in 1 throw, we treat it as a permutation problem.
    # There are factorial(ones) ways to arrange these balls.
    ones_ways = factorial_mod(ones)
    
    # The total number of distinct ways is the product of the two possibilities, modulo MOD.
    return (twos_ways * ones_ways) % MOD

n = int(input().strip())
a = list(map(int, input().strip().split()))

print(solve(n, a))
