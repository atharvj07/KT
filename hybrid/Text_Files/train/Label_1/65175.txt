x, y = map(int, input().split())

if x == 0 and y == 0:
    print(0)
else:
    turns = 0
    layer = max(abs(x), abs(y))
    if y == layer and x != -layer:
        turns = (layer-1)*4
    elif x == layer:
        turns = layer*4 - 3
    elif y == -layer:
        turns = layer*4 - 2
    elif x == -layer:
        turns = layer*4 - 1
    print(turns)
