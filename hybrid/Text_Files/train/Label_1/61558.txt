n, k = list(map(int, input().split()))
a = list(map(int, input().split()))
cnt = [0] * (k + 1)
for x in a: cnt[x] += 1
occ = []
for i in range(1, k + 1):
    if cnt[i]: occ.append(i)
mn = n
ex = 0
for i in range(len(occ)):
    cur = 0
    for j in range(i + 1, len(occ)):
        cur += cnt[occ[i]] * cnt[occ[j]]
    if cur < mn:
        mn = cur
        ex = occ[i]
print(ex)
