import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int police = 0, untreated = 0;
        for(int i=0; i<n; i++) {
            int c = input.nextInt();
            if(c == -1) {
                if(police > 0) --police;
                else untreated++;
            } else police += c;
        }
        System.out.println(untreated);
    }
}
