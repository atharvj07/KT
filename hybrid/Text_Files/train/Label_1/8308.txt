#include <cstring>
#include <iostream>
#include <sstream>
using namespace std;
typedef long long ll;
const int MX = 19;
ll dp[MX][2][2][MX];
string s;
int l;
ll solve(int pos, bool lead, bool tight, int sum) {
  if (pos == l) {
    if (sum == 0 && !tight)
      return 0;
    return 1e9;
  }
  ll &ret = dp[pos][lead][tight][sum];
  if (ret != -1)
    return ret;
  int lim;
  if (!tight)
    lim = s[pos] - '0';
  else
    lim = 9;
  ret = 1e9;
  if (lead) {
    for (int prv = 0; prv <= lim; prv++) {
      ret = min(ret, 1 + solve(pos + 1, 0, tight & (prv == lim), sum + prv));
    }
  } else {
    ret = min(ret, solve(pos + 1, 1, tight & (lim == 0), sum));
    for (int prv = 0; prv < lim; prv++)
      ret = min(ret, 1 + solve(pos + 1, 0, tight & (prv == lim), sum + prv));
  }
  return ret;
}
int main() {
  cin >> s;
  l = s.size();
  memset(dp, -1, sizeof(dp));
  cout << solve(0, 1, 1, 0) << endl;
  return 0;
}
