#include <bits/stdc++.h>
using namespace std;
int n, x, a[1005], vis[1005], ans;
int main() {
    cin >> n >> x;
    for (int i = 1; i <= n; i++) cin >> a[i];
    while (!vis[x]) {
        ans++;
        vis[x] = 1;
        for (int i = 1; i <= n; i++)
            if (!vis[a[i]]) vis[a[i] = a[i] + x - a[vis[a[i]]]] = 1;
    }
    cout << ans << endl;
}
