import sys
input = sys.stdin.readline

t = int(input())

for _ in range(t):
    n = int(input())

    a = list(map(int, input().split()))
    b = [0] * (n*2+1)

    j = n
    while j >= 1:
        if a[j-1] == 1:
            b[j] = 1
        b[j] += b[j+1]
        j -= 1

    j = n+1
    while j <= n*2:
        if a[j-1] == 2:
            b[j] = 1
        b[j] += b[j-1]
        j += 1

    ans = n*2

    for i in range(1, n*2+1):
        if b[i] == b[n]:
            ans = min(ans, i-1)

    print(ans)
