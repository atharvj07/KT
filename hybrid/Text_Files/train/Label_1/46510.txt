import java.util.Scanner;

public class Restaurant {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();

        int[] groups = new int[n];
        for (int i = 0; i < n; i++) {
            groups[i] = scanner.nextInt();
        }

        int denied = 0;
        int single = 0; // Track available single seats at two-seater tables

        for (int group : groups) {
            if (group == 1) { // Group of one person
                if (a > 0) { // Vacant one-seater table available
                    a--;
                } else if (b > 0) { // Vacant two-seater table available
                    b--;
                } else if (single > 0) { // Available single seat at two-seater table
                    single--;
                } else { // No available tables
                    denied++;
                }
            } else { // Group of two people
                if (b > 0) { // Vacant two-seater table available
                    b--;
                } else { // No available tables
                    denied += 2;
                }
            }
        }

        System.out.println(denied);
    }
}
