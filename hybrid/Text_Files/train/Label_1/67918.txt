import sys

n = int(input())
a = []
for i in range(n):
    a.append(list(map(int, input().split())))

a.sort(key=lambda x: x[1])

s = []
for i in range(n):
    if len(s) == 0 or s[-1][1] <= a[i][0]:
        s.append(a[i])
    elif len(s) == 1 or s[-2][1] <= a[i][0]:
        s[-1] = a[i]
    else:
        print('NO')
        sys.exit()

print('YES')
