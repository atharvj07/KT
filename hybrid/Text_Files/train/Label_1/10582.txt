#include<iostream>
#include<vector>
using namespace std;
int main(){
  int n;
  cin>>n;
  vector<vector<int>> graph(n,vector<int>(n));
  for(int i=0;i<n;++i){
    for(int j=0;j<n;++j){
      int x;
      cin>>x;
      graph[i][j]=x;
      graph[j][i]=x;
    }
  }
  for(int k=0;k<n;++k){
    for(int i=0;i<n;++i){
      for(int j=0;j<n;++j){
        if(graph[i][k]!=0&&graph[k][j]!=0){
          if(graph[i][j]==0||graph[i][j]>graph[i][k]+graph[k][j]){
            graph[i][j]=graph[i][k]+graph[k][j];
          }
        }
      }
    }
  }
  int ans=0;
  for(int i=0;i<n;++i){
    for(int j=0;j<n;++j){
      ans=max(ans,graph[i][j]);
    }
  }
  cout<<ans;
  return 0;
}
