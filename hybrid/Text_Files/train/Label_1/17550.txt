#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int maxn = 100010;

int n, m, d, sofas[maxn][4], id[maxn], l[maxn], r[maxn], t[maxn], b[maxn];

int main() {
    cin >> d >> n >> m;
    for (int i = 1; i <= d; i++) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        if (x1 == x2) {
            if (y1 > y2) swap(y1, y2);
            sofas[i][0] = y1;
            sofas[i][1] = y2;
        } else {
            if (x1 > x2) swap(x1, x2);
            sofas[i][2] = x1;
            sofas[i][3] = x2;
        }
    }
    int cntl, cntr, cntt, cntb;
    cin >> cntl >> cntr >> cntt >> cntb;
    for (int i = 1; i <= d; i++) {
        for (int j = 1; j <= d; j++) {
            if (i != j && sofas[i][1] > sofas[j][0] && sofas[i][3] > sofas[j][2] && sofas[i][0] < sofas[j][1] && sofas[i][2] < sofas[j][3])
                l[i]++;
            if (i != j && sofas[i][1] > sofas[j][0] && sofas[i][3] < sofas[j][2] && sofas[i][0] < sofas[j][1] && sofas[i][2] > sofas[j][3])
                t[i]++;
            if (i != j && sofas[i][1] < sofas[j][0] && sofas[i][3] > sofas[j][2] && sofas[i][0] > sofas[j][1] && sofas[i][2] < sofas[j][3])
                r[i]++;
            if (i != j && sofas[i][1] < sofas[j][0] && sofas[i][3] < sofas[j][2] && sofas[i][0] > sofas[j][1] && sofas[i][2] > sofas[j][3])
                b[i]++;
        }
        if (l[i] == cntl && r[i] == cntr && t[i] == cntt && b[i] == cntb)
            id[i]++;
    }
    int ans = -1;
    for (int i = 1; i <= d; i++)
        if (id[i]) ans = i;
    cout << ans << endl;
    return 0;
}
