#include <bits/stdc++.h>
using namespace std;
const int mxN = 1e5 + 1;
typedef long long ll;

int p[mxN], idx[mxN];
vector<tuple<int, int, int>> q;

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n - 2; ++i) {
        int a, b, c;
        cin >> a >> b >> c;
        q.emplace_back(a, b, c);
    }
    sort(q.begin(), q.end());
    for (int i = 0; i < n - 2; ++i) {
        int a, b, c;
        tie(a, b, c) = q[i];
        p[i + 1] = a;
        p[i + 2] = b;
        p[i + 3] = c;
    }
    vector<int> v(n);
    for (int i = 1; i <= n; ++i) v[p[i]] = i;
    for (int i = 1; i <= n; ++i) cout << v[i] << ' ';
}
