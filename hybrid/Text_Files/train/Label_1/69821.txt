n = int(input())
a = list(map(int, input().split()))

if n == 1:
    print(-1)
else:
    a_sorted = sorted((val, idx) for idx, val in enumerate(a))
    if a_sorted[0][0] != a_sorted[-1][0]:
        print(1, a_sorted[0][1] + 1)
    else:
        all_same = True
        for i in range(1, n):
            if a_sorted[i][0] != a_sorted[i-1][0]:
                all_same = False
                break
        if all_same:
            print(-1)
        else:
            print(2, a_sorted[0][1] + 1, a_sorted[1][1] + 1)
