#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
string s;
map<ll,ll> mp;
int main() {
	ll n; cin >> n;
	for(ll i = 0; i < 3; i++) {
		mp.clear();
		for(ll j = 0; j < n - i; j++) {
			ll a; cin >> a;
			mp[a]++;
		}
		for(auto it: mp) {
			if(it.second == (i + 1)) cout << it.first << " ";
		}
		cout << endl;
	}
}
