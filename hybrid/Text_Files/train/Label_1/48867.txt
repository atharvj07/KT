import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        char[] g = new char[26];
        for (int i = 0; i < 26; i++) {
            g[i] = s.charAt(i);
        }
        int[][] adj = new int[26][26];
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < 26; j++) {
                if (i == j) {
                    continue;
                }
                if (Math.abs(i - j) == 1 || (i / 13 == j / 13 && Math.abs(i - j) <= 2)) {
                    adj[i][j] = 1;
                    adj[j][i] = 1;
                }
            }
        }
        int[] p = new int[26];
        int[] vis = new int[26];
        int[] deg = new int[26];
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < 26; j++) {
                deg[i] += adj[i][j];
            }
        }
        int t = 0;
        for (int i = 0; i < 26; i++) {
            if (deg[i] % 2 == 1) {
                t++;
            }
        }
        if (t > 0 && t % 2 == 0) {
            System.out.println("Impossible");
            return;
        }
        for (int i = 0; i < 26; i++) {
            if (vis[i] == 0) {
                dfs(i, p, vis, adj);
            }
        }
        String r1 = "";
        String r2 = "";
        for (int i = 0; i < 26; i++) {
            if (p[i] == 0) {
                r1 += g[i];
            } else {
                r2 += g[i];
            }
        }
        System.out.println(r1);
        System.out.println(r2);
    }

    private static void dfs(int i, int[] p, int[] vis, int[][] adj) {
        vis[i] = 1;
        for (int j = 0; j < 26; j++) {
            if (adj[i][j] == 1 && vis[j] == 0) {
                p[j] = i;
                dfs(j, p, vis, adj);
            }
        }
    }
}
