#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  long long ans = 0;
  for (int i = 1; i < n; i++) {
    long long diff = a[i] - a[i - 1];
    long long cur = (diff + i - 1) / i;
    ans = max(ans, cur);
  }
  cout << ans << '\n';
  return 0;
}
