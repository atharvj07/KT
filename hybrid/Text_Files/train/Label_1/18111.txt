#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> b(n), w(n), o(n), r(n);
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  string s;
  cin >> s;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'W')
      w[i] = 1;
    else if (s[i] == 'O')
      o[i] = 1;
    else
      r[i] = 1;
  }
  if (k == 1) {
    cout << *max_element(b.begin(), b.end()) << endl;
    return 0;
  }
  if (k == 2) {
    int mx = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        int sum = b[i];
        if (w[i] != w[j])
          sum += b[j];
        else if (o[i] != o[j])
          sum += b[j];
        else if (r[i] != r[j])
          sum += b[j];
        mx = max(sum, mx);
      }
    }
    cout << mx << endl;
    return 0;
  }
  if (w[0] + w[1] + w[2] >= k) {
    vector<int> ans;
    for (int i = 0; i < n; i++) {
      if (w[i] && ans.size() < k) ans.push_back(b[i]);
    }
    sort(ans.rbegin(), ans.rend());
    int sum = 0;
    for (int i = 0; i < k; i++) { sum += ans[i]; }
    cout << sum << endl;
    return 0;
  }
  if (o[0] + o[1] + o[2] >= k) {
    vector<int> ans;
    for (int i = 0; i < n; i++) {
      if (o[i] && ans.size() < k) ans.push_back(b[i]);
    }
    sort(ans.rbegin(), ans.rend());
    int sum = 0;
    for (int i = 0; i < k; i++) sum += ans[i];
    cout << sum << endl;
    return 0;
  }
  if (r[0] + r[1] + r[2] >= k) {
    vector<int> ans;
    for (int i = 0; i < n; i++) {
      if (r[i] && ans.size() < k) ans.push_back(b[i]);
    }
    sort(ans.rbegin(), ans.rend());
    int sum = 0;
    for (int i = 0; i < k; i++) { sum += ans[i]; }
    cout << sum << endl;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    if (w[i]) {
      int idx1 = -1, idx2 = -1;
      for (int j = i + 1; j < n; j++) {
        if (s[j] != s[i]) {
          if (idx1 == -1)
            idx1 = j;
          else {
            idx2 = j;
            break;
          }
        }
      }
      if (idx1 != -1 && idx2 != -1) {
        vector<int> ans;
        ans.push_back(b[i]), ans.push_back(b[idx1]), ans.push_back(b[idx2]);
        sort(ans.rbegin(), ans.rend());
        if (ans[0] + ans[1] + ans[2] >= k) {
          int sum = 0;
          for (int i = 0; i < k; i++) { sum += ans[i]; }
          cout << sum << endl;
          return 0;
        }
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (o[i]) {
      int idx1 = -1, idx2 = -1;
      for (int j = i + 1; j < n; j++) {
        if (s[j] != s[i]) {
          if (idx1 == -1)
            idx1 = j;
          else {
            idx2 = j;
            break;
          }
        }
      }
      if (idx1 != -1 && idx2 != -1) {
        vector<int> ans;
        ans.push_back(b[i]), ans.push_back(b[idx1]), ans.push_back(b[idx2]);
        sort(ans.rbegin(), ans.rend());
        if (ans[0] + ans[1] + ans[2] >= k) {
          int sum = 0;
          for (int i = 0; i < k; i++) { sum += ans[i]; }
          cout << sum << endl;
          return 0;
        }
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (r[i]) {
      int idx1 = -1, idx2 = -1;
      for (int j = i + 1; j < n; j++) {
        if (s[j] != s[i]) {
          if (idx1 == -1)
            idx1 = j;
          else {
            idx2 = j;
            break;
          }
        }
      }
      if (idx1 != -1 && idx2 != -1) {
        vector<int> ans;
        ans.push_back(b[i]), ans.push_back(b[idx1]), ans.push_back(b[idx2]);
        sort(ans.rbegin(), ans.rend());
        if (ans[0] + ans[1] + ans[2] >= k) {
          int sum = 0;
          for (int i = 0; i < k; i++) { sum += ans[i]; }
          cout << sum << endl;
          return 0;
        }
      }
    }
  }
  cout << "-1" << endl;
  return 0;
}
