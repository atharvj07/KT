#include <iostream>

using namespace std;

long long binpow(long long n, long long k) {
    if (k == 0) {
        return 1;
    }
    long long ans = binpow(n, k / 2);
    ans = (ans * ans) % (1LL << 63);
    if (k % 2 != 0) {
        ans = (ans * n) % (1LL << 63);
    }
    return ans;
}

int main() {
    long long n, m;
    cin >> n >> m;
    long long ans = binpow(2, n);
    cout << m % (ans) << "\n";
    return 0;
}
