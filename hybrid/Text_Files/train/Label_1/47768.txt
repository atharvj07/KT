import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] x = new int[n], y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = in.nextInt();
            y[i] = in.nextInt();
        }
        Arrays.sort(x);
        Arrays.sort(y);
        long ans = 0;
        for (int i = 0; i < n - 1; i++) {
            ans += x[i + 1] - x[i];
            ans += y[i + 1] - y[i];
        }
        System.out.println(ans);
    }
}
