#include <bits/stdc++.h>
using namespace std;

int n, m, k, d[1000001], b[1000001], c[1000001];

int main() {
    cin >> n >> m >> k;
    for (int i = 0; i < m; i++) {
        cin >> d[i];
    }
    b[0] = d[0];
    for (int i = 1; i < m; i++) {
        b[i] = d[i] - d[i - 1] - 1;
    }
    b[m] = n - d[m - 1] - 1;
    for (int i = 1; i <= k; i++) {
        cin >> c[i];
    }
    int ans = -1;
    for (int i = 1; i <= k; i++) {
        int sum = c[i];
        for (int j = 0; j <= m; j++) {
            if(b[j] % i == 0) {
                sum += c[i] * ((b[j] / i) - 1);
            }
            else {
                sum += c[i] * (b[j] / i);
                if(b[j]%i != 0) sum += c[b[j]%i];
            }
        }
        if(ans == -1) ans = sum;
        else ans = min(ans, sum);
    }
    cout << ans;
}
