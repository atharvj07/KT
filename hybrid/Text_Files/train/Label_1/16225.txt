#include<cstdio>
#include<iostream>
#include<cstring>
#include<algorithm>
using namespace std;
int a[1000007],dp[1000007],n,T;
int main()
{
    scanf("%d%d",&n,&T);
    for(int i=0;i<n;i++)
    {
        scanf("%d",&a[i]);
    }
    dp[0]=1;
    for(int i=0;i<T;i++)
    {
        if(i+n>=T)
        {
            dp[i+n]=dp[i];
        }
        else
        {
            dp[i+n]=dp[i]+1;
            for(int j=0;j<n;j++)
            {
                if(a[i]<a[j])
                {
                    dp[i+n]=max(dp[i+n],dp[j]+1);
                }
            }
        }
    }
    int ans=0;
    for(int i=0;i<T;i++)
    {
        ans=max(ans,dp[i]);
    }
    cout<<ans<<endl;
}
