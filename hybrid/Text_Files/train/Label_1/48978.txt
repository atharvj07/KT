import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextInt();
        }
        int ans = n - 1;
        for (int i = 0; i < k - 1; i++) {
            if (a[i + 1] - a[i] > 1) {
                ans--;
            }
        }
        System.out.println(ans);
    }
}
