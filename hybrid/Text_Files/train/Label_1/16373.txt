#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> ans;
    for (int i = 1; i < n; i++) {
        if (s[i] == s[i - 1]) {
            ans.push_back(i);
            if (i + 1 < n) {
                s[i] = (s[i] == 'R' ? 'G' : 'R');
            } else {
                s[i] = 'B';
            }
        }
    }
    cout << ans.size() << "\n";
    cout << s << "\n";
    return 0;
}
