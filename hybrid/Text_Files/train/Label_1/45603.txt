import java.util.*;

class Main {
    static List<Integer>[] g;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        g = new List[N + 1];
        for (int i = 0; i <= N; i++) {
            g[i] = new ArrayList<>();
        }
        for (int i = 0; i < M; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            g[a].add(b);
            g[b].add(a);
        }
        int[] col = new int[N + 1];
        int ans = 0;
        for (int i = 1; i <= N; i++) {
            if (col[i] == 0) {
                ans += dfs(i, 1, col);
            }
        }
        if (ans == N - 1) {
            System.out.println(ans);
        } else {
            System.out.println(-1);
        }
    }

    static int dfs(int u, int colr, int[] col) {
        if (col[u] != 0) {
            return 0;
        }
        col[u] = colr;
        int ok = 1;
        for (int v : g[u]) {
            ok &= dfs(v, 3 - colr, col);
        }
        return ok;
    }
}
