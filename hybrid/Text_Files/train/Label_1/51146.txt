import java.util.*;
class Main {
  public static void main(String[] argv) {
    Scanner sc = new Scanner(System.in);
    int cm = sc.nextInt();
    int fe = round(cm/30);
    int inc = round((((cm/30.0) - fe)*30));
    System.out.println(fe + " " + inc);
  }
  public static int round(double x) {
    return (int)(Math.round(x/1.0));
  }
}
