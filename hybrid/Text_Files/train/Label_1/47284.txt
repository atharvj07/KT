import java.util.Arrays;
import java.util.HashMap;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        for (int i = 1; i < n; i++) {
            p[i] = sc.nextInt() - 1;
        }
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }

        int[] w = new int[n];
        boolean[] v = new boolean[n];
        Arrays.fill(w, -1);
        v[0] = true;
        for (int i = 1; i < n; i++) {
            if (w[p[i]] == -1) {
                System.out.println("IMPOSSIBLE");
                return;
            }
            w[i] = x[p[i]] - w[p[i]];
            if (w[i] < 0 || w[i] > 5000) {
                System.out.println("IMPOSSIBLE");
                return;
            }
            v[p[i]] = true;
        }

        for (int i = 0; i < n; i++) {
            if (!v[i]) {
                System.out.println("IMPOSSIBLE");
                return;
            }
        }

        System.out.println("POSSIBLE");
    }
}
