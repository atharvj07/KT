#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  int ans = n / 3;
  ans += (n % 3 > 0);
  cout << ans << endl;
  for (int i = 1; i <= n; i += 3) {
    cout << i << " ";
  }
  if (n % 3 == 1) {
    cout << n << " ";
  }
  if (n % 3 == 2) {
    cout << n - 1 << " " << n << " ";
  }
  cout << endl;
  return 0;
}
