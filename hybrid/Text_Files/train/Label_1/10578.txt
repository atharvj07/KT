#include <bits/stdc++.h>

using namespace std;

int main() {
  int r, c;
  cin >> r >> c;
  vector<string> a(r);
  for (int i = 0; i < r; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < r; i++) {
    for (int j = 0; j < c; j++) {
      if (a[i][j] == 'S') {
        for (int di = -1; di <= 1; di++) {
          for (int dj = -1; dj <= 1; dj++) {
            if (di == 0 && dj == 0) continue;
            int ii = i + di, jj = j + dj;
            if (ii >= 0 && ii < r && jj >= 0 && jj < c && a[ii][jj] != 'D' && a[ii][jj] != 'S') {
              a[ii][jj] = 'D';
            }
          }
        }
      }
    }
  }
  bool ok = true;
  for (int i = 0; i < r; i++) {
    for (int j = 0; j < c; j++) {
      if (a[i][j] == 'S') {
        for (int di = -1; di <= 1; di++) {
          for (int dj = -1; dj <= 1; dj++) {
            if (di == 0 && dj == 0) continue;
            int ii = i + di, jj = j + dj;
            if (ii >= 0 && ii < r && jj >= 0 && jj < c && a[ii][jj] == 'W') {
              ok = false;
            }
          }
        }
      }
    }
  }
  if (!ok) {
    cout << "No" << endl;
  } else {
    cout << "Yes" << endl;
    for (int i = 0; i < r; i++) {
      cout << a[i] << endl;
    }
  }
  return 0;
}
