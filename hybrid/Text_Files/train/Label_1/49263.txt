import java.util.*;

public class Main {
    static Scanner sc = new Scanner(System.in);

    static class Pair {
        int a;
        int b;

        public Pair(int a, int b) {
            this.a = a;
            this.b = b;
        }
    }

    static int n, m;
    static Pair[] arr;

    public static void main(String[] args) {
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            input();
            int ans = solve();
            System.out.println(ans);
        }
    }

    static void input() {
        n = sc.nextInt();
        m = sc.nextInt();
        arr = new Pair[n];
        for (int i = 0; i < n; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            arr[i] = new Pair(a, b);
        }
    }

    static int solve() {
        Arrays.sort(arr, (a, b) -> {
            if (a.a * b.b != a.b * b.a) return a.a * b.b - a.b * b.a;
            return a.b - b.b;
        });
        int sum = 0;
        int ans = Integer.MAX_VALUE;
        for (Pair pair : arr) {
            if (sum + pair.a >= m) {
                ans = Math.min(ans, sum + pair.a - m);
            }
            sum += pair.a;
        }
        return sum >= m ? ans : -1;
    }
}
