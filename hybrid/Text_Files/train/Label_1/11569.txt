#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> z(n + 1), o(n + 1), e(n + 1);
    for (int i = 1; i <= n; i++) {
        z[i] = z[i - 1];
        o[i] = o[i - 1];
        e[i] = e[i - 1];
        if (s[i - 1] == 'z' or s[i - 1] == 'e') z[i]++;
        if (s[i - 1] == 'o' or s[i - 1] == 'e') o[i]++;
        if (s[i - 1] == 'n' or s[i - 1] == 'e') e[i]++;
    }
    vector<int> now;
    for (int i = n; i >= 1; i--) {
        if (o[i] - o[i - 1] > 0 and e[i] - e[i - 1] > 0) {
            o[i - 1]++;
            e[i - 1]++;
            now.push_back(1);
        } else if (z[i] - z[i - 1] > 0 and o[i] - o[i - 1] > 0) {
            z[i - 1]++;
            o[i - 1]++;
            now.push_back(0);
        }
    }
    reverse(now.begin(), now.end());
    for (int f : now) cout << f << " ";
    return 0;
}
