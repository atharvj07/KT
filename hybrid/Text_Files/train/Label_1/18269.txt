#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int n, a[N], freq[11];

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    string s;
    cin >> s;
    fill(freq, freq + 11, 0);
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
      freq[s[i] - '0']++;
      ans += freq[s[i] - '0'] == s[i] - '0';
    }
    cout << ans << endl;
  }
  return 0;
}
