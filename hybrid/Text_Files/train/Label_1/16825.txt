#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;
const int MX = 1e6 + 5;
int n, m;
int pw[MX], invpw[MX], h[MX], rh[MX];
int gethash(int l, int r) {
  int x = (h[r] - (l ? h[l - 1] : 0LL) + mod) % mod;
  return (x * invpw[l]) % mod;
}
bool check(int len) {
  set<int> hs;
  for (int i = 0; i + m - 1 < n; i++) { hs.insert(gethash(i, i + m - 1)); }
  int curh = gethash(0, m - 1);
  for (int i = m; i + m - 1 < n; i++) {
    curh = ((curh - (rh[m - 1] * (gethash(i, i + m - 1))) % mod) % mod + mod) % mod;
    curh = (curh * pw[1]) % mod;
    hs.insert(curh);
  }
  return hs.size() == n - m + 1;
}
int32_t main() {
  int base = 31;
  pw[0] = invpw[0] = 1LL;
  for (int i = 1; i <= 1000000; i++) {
    pw[i] = (pw[i - 1] * base) % mod;
    invpw[i] = ((pw[i] - 1) % mod + mod) % mod;
  }
  string a, b;
  cin >> a >> b;
  n = a.size(), m = b.size();
  for (int i = 0; i < n; i++) { h[i] = (h[i - 1] * base + (a[i] - 'a')) % mod; }
  for (int i = m - 1; i >= 0; i--) { rh[i] = (rh[i + 1] * base + (b[i] - 'a')) % mod; }
  int l = 0, r = n, ans = 0;
  while (l <= r) {
    int mid = (l + r) >> 1LL;
    if (check(mid)) {
      ans = mid;
      l = mid + 1;
    } else
      r = mid - 1;
  }
  cout << ans << endl;
}
