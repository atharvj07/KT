n = int(input())
words = [input() for _ in range(n)]
message = input()
if message.count('<3') != n:
    print("no")
else:
    encoded = ''.join(['<3' + w + '<3' for w in words])
    if encoded in message:
        print("yes")
    elif encoded[::-1] in message:
        print("yes")
    else:
        print("no")
