t = int(input())
for _ in range(t):
    n, k, m = map(int, input().split())
    b = list(map(int, input().split()))
    if (n - m) % (k - 1) != 0:
        print("NO")
        continue
    total_removed = n - m
    required_steps = total_removed // (k - 1)
    is_possible = False
    for i in range(m):
        left_elements = b[i] - 1 - i
        right_elements = (n - b[i]) - (m - 1 - i)
        max_removal_left = left_elements + (right_elements // (k - 1)) * (k - 1)
        max_removal_right = right_elements + (left_elements // (k - 1)) * (k - 1)
        if left_elements + right_elements == total_removed and (max_removal_left >= required_steps * (k - 1) or max_removal_right >= required_steps * (k - 1)):
            is_possible = True
            break
    if is_possible:
        print("YES")
    else:
        print("NO")
