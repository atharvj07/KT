#include <bits/stdc++.h>
using namespace std;
int main() {
    string s, t = "heidi", sub = "";
    cin >> s, sub.push_back(s[0]);
    for (int i = 1, j = 0; i < s.size(); i++) {
        if (sub[j] == t[j] && j + 1 < t.size()) sub.push_back(s[i]), j++;
    }
    if (sub == t) cout << "YES" << endl; else cout << "NO" << endl;
    return 0;
}
