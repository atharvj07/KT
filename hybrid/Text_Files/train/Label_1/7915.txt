#include<bits/stdc++.h>
using namespace std;
#define fio ios_base::sync_with_stdio(false)
#define ll long long

ll a, b, m, ans = 1;
char s[100001];

int main() {
    fio;
    cin >> a >> b >> s;
    for (int i = 0; i < a; i++) {
        if (s[i] == 'a') {
            m++;
        } else {
            ans *= m;
            m = 1;
        }
    }
    ans *= m;
    cout << ans << endl;
    return 0;
}
