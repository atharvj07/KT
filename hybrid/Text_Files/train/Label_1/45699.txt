import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int d = sc.nextInt();
            int a = 2;
            while (true) {
                int cnt = 0;
                for (int i = 1; i * i <= a; i++) {
                    if (a % i == 0) {
                        if (i * i == a) {
                            cnt++;
                        } else {
                            cnt += 2;
                        }
                    }
                }
                if (cnt >= 4) {
                    int last = 1;
                    boolean ok = true;
                    for (int i = 2; i * i <= a; i++) {
                        if (a % i == 0) {
                            if (i * i == a) {
                                if (i - last < d) {
                                    ok = false;
                                    break;
                                }
                            } else {
                                if (i - last < d || a / i - last < d) {
                                    ok = false;
                                    break;
                                }
                            }
                            last = i;
                        }
                    }
                    if (ok) {
                        System.out.println(a);
                        break;
                    }
                }
                a++;
            }
        }
    }
}
