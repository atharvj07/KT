n, m = map(int, input().split())
if n * m % 2 == 1:
    print("NO")
else:
    print("YES")
    mat = [[0] * m for i in range(n)]
    num = 1
    for i in range(0, n, 2):
        for j in range(m):
            mat[i][j] = num
            num += 1
    for i in range(1, n, 2):
        for j in range(m - 1, -1, -1):
            mat[i][j] = num
            num += 1
    for i in range(n):
        for j in range(m):
            print(mat[i][j], end=" ")
        print()
