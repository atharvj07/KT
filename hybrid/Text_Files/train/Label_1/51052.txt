import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long[] dp = new long[n + 1];
        dp[0] = 1L;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (a[i] == j - i - 1) {
                    dp[j] += dp[i];
                    dp[j] %= 1000000007;
                }
            }
        }
        System.out.println(dp[n]);
    }
}
