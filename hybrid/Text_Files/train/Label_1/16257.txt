#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N=1e5+10;
ll n,a[N],A,cf,cm,m,ans;
struct nd{
    ll v,p;
}tr[N];
bool cmp(nd x,nd y){
    return x.v>y.v;
}
int main(){
    cin>>n>>A>>cf>>cm>>m;
    for(int i=1;i<=n;++i){
        cin>>a[i];
        tr[i]={a[i],i};
    }
    sort(tr+1,tr+1+n,cmp);
    for(int i=1;i<=n;++i){
        if(a[tr[i].p]==A) continue;
        a[tr[i].p]++;
        ans+=cm;
        if(!--m) break;
    }
    sort(tr+1,tr+1+n,cmp);
    for(int i=1;i<=n;++i){
        if(tr[i].v==A) continue;
        ll need=A-tr[i].v;
        if(need<=m){
            m-=need;
            ans+=need*cf;
            a[tr[i].p]=A;
        }
        else break;
    }
    cout<<ans<<endl;
    for(int i=1;i<=n;++i) cout<<a[i]<<" ";
}
