import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[][] grid = new int[n][m];
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < m; k++) {
                    grid[j][k] = sc.nextInt();
                }
            }
            boolean ans = false;
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < m; k++) {
                    if (grid[j][k] == 0) {
                        boolean ok = true;
                        for (int l = 0; l < n; l++) {
                            if (grid[l][k] == 1) {
                                ok = false;
                            }
                        }
                        for (int l = 0; l < m; l++) {
                            if (grid[j][l] == 1) {
                                ok = false;
                            }
                        }
                        if (ok) {
                            ans = true;
                        }
                    }
                }
            }
            System.out.println(ans ? "Ashish" : "Vivek");
        }
    }
}
