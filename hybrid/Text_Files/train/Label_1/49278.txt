import java.util.Scanner;

public class AutomaticDoor {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();

        long[] employees = new long[n];
        employees[0] = a;
        for (int i = 1; i < n; i++) {
            employees[i] = employees[i - 1] + a;
        }

        long[] clients = new long[m];
        for (int i = 0; i < m; i++) {
            clients[i] = scanner.nextLong();
        }

        int openCount = 0;
        long lastOpened = -1;

        for (long employee : employees) {
            if (employee <= lastOpened + d) {
                continue;
            }
            openCount++;
            lastOpened = employee;
        }

        for (long client : clients) {
            if (client <= lastOpened + d) {
                continue;
            }
            openCount++;
            lastOpened = client;
        }

        System.out.println(openCount);
    }
}
