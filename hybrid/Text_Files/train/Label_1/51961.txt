import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

    int[] sorted = arr.clone();
    Arrays.sort(sorted);

    int[] left = new int[n];
    left[0] = sorted[0];
    for (int i = 1; i < n; i++) {
      left[i] = Math.max(left[i - 1], sorted[i]);
    }

    int[] right = new int[n];
    right[n - 1] = sorted[n - 1];
    for (int i = n - 2; i >= 0; i--) {
      right[i] = Math.max(right[i + 1], sorted[i]);
    }

    int[] ans = new int[n];
    for (int i = 0; i < n; i++) {
      ans[i] = Math.max(right[i] - sorted[i], sorted[i] - left[i]);
    }

    for (int i : ans) System.out.print(i + " ");
  }
}
