#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> friends(n);
  for (int i = 0; i < n; i++) {
    int w, h;
    cin >> w >> h;
    friends[i] = {w, h};
  }
  sort(friends.begin(), friends.end());
  int min_area = 0;
  int max_height = 0;
  for (int i = 0; i < n; i++) {
    min_area += friends[i].first;
    max_height = max(max_height, friends[i].second);
  }
  cout << min_area * max_height << endl;
}
