import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        permute(1, n, new int[n], new boolean[n + 1]);
    }

    private static void permute(int index, int n, int[] arr, boolean[] used) {
        if (index == n + 1) {
            for (int i = 1; i <= n; i++) {
                System.out.print(arr[i - 1] + " ");
            }
            System.out.println();
            return;
        }

        for (int i = 1; i <= n; i++) {
            if (!used[i]) {
                arr[index - 1] = i;
                used[i] = true;
                permute(index + 1, n, arr, used);
                used[i] = false;
            }
        }
    }
}
