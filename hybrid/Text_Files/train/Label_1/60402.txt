    ans+=int(n[i])*(10**i)
    i-=1
if ans%8==0:
    print("YES")
    print(ans)

    while i>=0:
        if int(n[i])%8==0:
            flag=True
            break
        i-=1
    if flag==False:
        print("NO")
    else:
        print("YES")
        ans=int(n[:i+1])
        i-=1
