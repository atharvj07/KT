#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define pb push_back

int main()
{
    int n, m;
    cin >> n>> m;
    vector<vi> a(n + 2, vi(m + 2, 0));
    vector<vi> dp(n + 2, vi(m + 2, 0));
    for(int i = 1; i <= n; i++)
    {
        for(int j = 1; j <= m; j++)
        {
            cin >> a[i][j];
            if(a[i][j] > 0)
            {
                dp[i][j] = a[i][j];
            }
            else
            {
                dp[i][j] = -1e9;
            }
        }
    }
    for(int j = 1; j <= m; j++)
    {
        for(int i = 1; i <= n; i++)
        {
            if(a[i][j] > 0)
            {
                dp[i][j] = a[i][j];
            }
            else
            {
                dp[i][j] = max({dp[i - 1][j], dp[i][j - 1], dp[i + 1][j], dp[i][j + 1]}) + 1;
            }
        }
    }
    long long sum = 0;
    for(int i = 1; i <= n; i++)
    {
        for(int j = 1; j <= m; j++)
        {
            if(dp[i][j] < 0)
            {
                cout << -1;
                return 0;
            }
            else
            {
                sum += dp[i][j];
            }
        }
    }
    cout << sum;
    return 0;
}
