import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[][] holidays = new int[N][4];
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < 4; j++) {
                holidays[i][j] = sc.nextInt();
            }
        }

        int[] dp = new int[367];
        for (int i = 1; i <= 365; i++) dp[i] = dp[i - 1];
        for (int i = 0; i < N; i++) {
            for (int j = holidays[i][1]; j <= holidays[i][1] + holidays[i][2] - 1; j++) {
                dp[j] = Math.max(dp[j], holidays[i][3]);
            }
        }
        int min = 361;
        for (int i = 1; i <= 365; i++) {
            min = Math.min(min, dp[i]);
        }
        System.out.println(min);
    }
}
