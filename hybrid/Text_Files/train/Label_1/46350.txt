import java.util.Arrays;
import java.util.Scanner;

public class RyoukoMemoryNote {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] pages = new int[m];
        for (int i = 0; i < m; i++) {
            pages[i] = scanner.nextInt();
        }
        scanner.close();

        Arrays.sort(pages);

        int minTurns = Integer.MAX_VALUE;
        for (int i = 0; i < m - 1; i++) {
            int mergeCost = pages[i + 1] - pages[i] - 1;
            int turnsAfterMerge = 0;
            for (int j = 0; j < m; j++) {
                if (pages[j] <= pages[i]) {
                    turnsAfterMerge += pages[i + 1] - pages[j];
                } else if (pages[j] >= pages[i + 1]) {
                    turnsAfterMerge += pages[j] - pages[i + 1];
                } else {
                    turnsAfterMerge += pages[i] - pages[j];
                }
            }
            minTurns = Math.min(minTurns, turnsAfterMerge + mergeCost);
        }

        System.out.println(minTurns);
    }
}
