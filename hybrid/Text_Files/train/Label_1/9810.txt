#include<bits/stdc++.h>
#define ll long long
using namespace std;
map<string,vector<string>>mp;
string ans;
int n,m;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cin>>n>>m;
    int cnt=0;
    string x;
    while(cin>>x){
        string y="";
        for(int i=0,j=x.size(); i<j; i++){
            if(x[i]==':'){
                y+=' ';
                continue;
            }
            y+=x[i];
        }
        mp[y.substr(0,10)].push_back(y);
        cnt++;
    }
    if(cnt<m){
        cout<<-1;
        return 0;
    }
    cnt=0;
    for(auto it=mp.rbegin(); it!=mp.rend(); it++){
        for(auto jt=it->second.rbegin(); jt!=it->second.rend(); jt++){
            if(cnt+it->second.size()>=m){
                ans=*jt;
                break;
            }
            cnt+=it->second.size();
        }
        if(cnt>=m){
            break;
        }
    }
    cout<<ans;
    return 0;
}
