import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] b = new int[n];
            for (int i = 0; i < n; i++) {
                b[i] = sc.nextInt();
            }
            int[] a = new int[n];
            a[0] = b[0];
            int left = 1;
            int right = n - 1;
            for (int i = 1; i < n; i++) {
                if (i % 2 == 1) {
                    a[right--] = b[i];
                } else {
                    a[left++] = b[i];
                }
            }
            for (int i = 0; i < n; i++) {
                System.out.print(a[i] + " ");
            }
            System.out.println();
        }
    }
}
