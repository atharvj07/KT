#include <bits/stdc++.h>
using namespace std;
int n, k, t[105], ans;
int main() {
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> t[i];
    for (int j = 0; j < k; j++) {
        int e = 0, s = 0;
        for (int i = j; i < n; i += k) e += t[i] == 1, s += t[i] == -1;
        ans = max(ans, abs(e - s));
    }
    cout << ans << '\n';
}
