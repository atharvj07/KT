import sys, math
input = sys.stdin.readline

def init(l, r, node):
    if l == r:
        tree[node] = [int(a[l-1]), int(a[l-1])]
        return tree[node]
    
    mid = (l+r) // 2
    left = init(l, mid, node*2)
    right = init(mid+1, r, node*2+1)

    tree[node] = [0, 0]
    x = left[0]
    y = right[0]
    tree[node][0] = max(x, y)
    tree[node][0] = max(tree[node][0], x + right[1])
    tree[node][0] = max(tree[node][0], left[1] + y)

    x = left[1]
    y = right[1]
    tree[node][1] = max(x, y)
    tree[node][1] = max(tree[node][1], x + right[0])
    tree[node][1] = max(tree[node][1], left[0] + y)

    return tree[node]

def query(l, r, ql, qr, node):
    if l > qr or r < ql:
        return [0, 0]

    if l >= ql and r <= qr:
        return tree[node]

    mid = (l+r) // 2
    left = query(l, mid, ql, qr, node*2)
    right = query(mid+1, r, ql, qr, node*2+1)
    
    x = left[0]
    y = right[0]
    ans = [0, 0]
    ans[0] = max(x, y)
    ans[0] = max(ans[0], x + right[1])
    ans[0] = max(ans[0], left[1] + y)

    x = left[1]
    y = right[1]
    ans[1] = max(x, y)
    ans[1] = max(ans[1], x + right[0])
    ans[1] = max(ans[1], left[0] + y)

    return ans

mod = 1000000007
n, q = map(int, input().split())
a = input().strip()
tree = [[0, 0] for i in range(4*n+5)]

init(1, n, 1)

for i in range(q):
    l, r = map(int, input().split())
    x = query(1, n, l, r, 1)
    print(x[0]%mod)
