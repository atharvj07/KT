def find_parent(parent, i):
    if parent[i] == i:
        return i
    return find_parent(parent, parent[i])

def union(parent, rank, x, y):
    xroot = find_parent(parent, x)
    yroot = find_parent(parent, y)
    
    if rank[xroot] < rank[yroot]:
        parent[xroot] = yroot
    elif rank[xroot] > rank[yroot]:
        parent[yroot] = xroot
    else:
        parent[yroot] = xroot
        rank[xroot] += 1

n, m = map(int, input().split())
parent = list(range(m+1))
rank = [0] * (m+1)
languages_known = [False] * (m+1)
for _ in range(n):
    data = list(map(int, input().split()))
    for i in range(1, len(data)):
        if not languages_known[data[i]]:
            languages_known[data[i]] = True
        for j in range(i+1, len(data)):
            x = find_parent(parent, data[i])
            y = find_parent(parent, data[j])
            if x != y:
                union(parent, rank, x, y)

connected_components = set()
for i in range(1, m+1):
    if languages_known[i]:
        connected_components.add(find_parent(parent, i))

if all(not lang for lang in languages_known[1:]):
    print(n)
else:
    print(len(connected_components) - 1)
