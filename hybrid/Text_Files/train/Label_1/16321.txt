#include <iostream>
#include <vector>
using namespace std;
#define endl "\n"
int main() {
  string a, b;
  cin >> a >> b;
  int ans = 0;
  vector<int> pref(b.size());
  vector<int> suf(b.size());
  for (int i = 0; i < b.size(); i++) {
    if (i != 0) pref[i] = pref[i - 1];
    if (a[0] != b[i]) pref[i]++;
  }
  for (int i = b.size() - 1; i >= 0; i--) {
    if (i != b.size() - 1) suf[i] = suf[i + 1];
    if (a[a.size() - 1] != b[i]) suf[i]++;
  }
  for (int i = 0; i <= b.size() - a.size(); i++) {
    int sum = 0;
    if (i != 0) sum += pref[i - 1];
    sum += suf[i + a.size() - 1];
    ans += sum;
  }
  cout << ans << endl;
  return 0;
}
