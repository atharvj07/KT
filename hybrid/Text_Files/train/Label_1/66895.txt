def find_max_cost(n, m, k, array):
    max_cost = 0
    for l in range(n):
        sum_subarray = 0
        for r in range(l, n):
            sum_subarray += array[r]
            cost = sum_subarray - k * ((r - l + m) // m)
            max_cost = max(max_cost, cost)
    return max_cost

n, m, k = map(int, input().split())
array = list(map(int, input().split()))
print(find_max_cost(n, m, k, array))
