import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
b = [[0]*m for _ in range(n+1)]
for j in range(m):
    for i in range(1, n+1):
        b[i][j] = b[i-1][j] + a[i-1][j]
k = int(input())
for _ in range(k):
    l, r = map(int, input().split())
    mn = float('inf')
    for j in range(m):
        if b[r][j] - b[l-1][j] == r - l:
            print('Yes')
            break
        mn = min(mn, b[r][j] - b[l-1][j])
    else:
        if mn == r - l + 1:
            print('Yes')
        else:
            print('No')
