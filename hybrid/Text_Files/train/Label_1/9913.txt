#include <iostream>

using namespace std;

int main() {
  int t, n, a[100100];
  cin >> t;
  while (t--) {
    cin >> n;
    int odd = 0, even = 0;
    for (int i = 0; i < n; ++i) {
      cin >> a[i];
      if (a[i] % 2)
        odd++;
      else
        even++;
    }
    if (!odd || (odd % 2 == 0 && even == 0))
      cout << "NO" << "\n";
    else
      cout << "YES" << "\n";
  }
  return 0;
}
