#include <cstdio>
#include <cstring>
using namespace std;
int n, m, a[1010][1010], ans, hs[30];
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) {
    char s[1010];
    scanf("%s", s + 1);
    for (int j = 1; j <= m; j++) a[i][j] = s[j] - 'a';
    for (int j = 1; j <= m - 2; j++)
      for (int k = j + 1; k <= m - 1; k++) {
        memset(hs, 0, sizeof hs);
        hs[a[i][j]] = 1;
        hs[a[i][k]] = 1;
        hs[a[i][k + 1]] = 1;
        if (hs[0] + hs[1] + hs[2] == 3) ans++;
      }
  }
  printf("%d\n", ans);
}
