#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int N = 3e5 + 5;
vector<int> g[N], ans;
int n, a[N], u, p[N], l[N], r[N];

int gd(int a, int b) {
    return b ? gd(b, a % b) : a;
}

void dfs(int u) {
    for (int v : g[u]) if (!l[v]) {
        l[v] = l[u];
        dfs(v);
    }
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) if (!p[i]) {
        for (int j = i; j <= n; j += i) p[j] = i;
    }
    for (int i = 1; i <= n; i++) {
        for (int j = a[i]; j > 1; j = p[j]) g[p[j]].push_back(i);
    }
    int best = 0;
    for (int i = 1; i <= n; i++) {
        if (!l[i]) l[i] = i, dfs(i);
        int j = l[i], cnt = 0;
        for (int k = i; k <= n; k += a[j]) {
            if (l[k] == l[i]) cnt++, r[k] = i;
            else break;
        }
        if (cnt > best) best = cnt, u = i;
    }
    cout << best << ' ' << r[u] - l[u];
    for (int i = l[u]; i <= n; i += a[l[u]]) if (r[i] == l[u]) ans.push_back(i);
    cout << '\n';
    for (int i : ans) cout << i << ' ';
    return 0;
}
