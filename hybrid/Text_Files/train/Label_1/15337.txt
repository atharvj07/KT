#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e2 + 5;
int t[maxn], c[maxn];
int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> t[i];
    for (int i = 0; i < n; i++) cin >> c[i];
    double T = 0, C = 0;
    for (int i = 0; i < n; i++) {
        T += t[i];
        if (C * k <= T * (k - 1)) {
            C += c[i];
            T += t[i];
        }
    }
    if (C * k <= T * (k - 1))
        cout << -1 << endl;
    else
        cout << (int)T << endl;
    return 0;
}
