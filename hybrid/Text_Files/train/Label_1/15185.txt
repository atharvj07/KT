#include <bits/stdc++.h>
using namespace std;
int n;
char c[100005];
bool ok;
int main() {
  cin >> n;
  cin >> c;
  map<char, int> m;
  for (int i = 0; i < n; i++) m[c[i]]++;
  for (map<char, int>::iterator it = m.begin(); it != m.end(); it++) {
    if (it->second >= 2) ok = true;
  }
  if (ok)
    cout << "Yes";
  else
    cout << "No";
  return 0;
}
