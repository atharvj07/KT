def min_hamming_distance():
    n = int(input())
    s = input()
    t = input()

    # Initial Hamming distance
    hamming_distance = sum([1 for i in range(n) if s[i] != t[i]])
    if hamming_distance == 0:
        print(0)
        print("-1 -1")
        return

    # Mismatch tracking
    s_mismatches, t_mismatches = {}, {}
    for i in range(n):
        if s[i] != t[i]:
            s_mismatches.setdefault(s[i], []).append(i)
            t_mismatches.setdefault(t[i], []).append(i)

    # Check for a direct swap
    for ch in s_mismatches:
        if ch in t_mismatches and len(s_mismatches[ch]) == 1 and len(t_mismatches[ch]) == 1:
            idx1 = s_mismatches[ch][0]
            idx2 = t_mismatches[ch][0]
            if s[idx1] == t[idx2] and t[idx1] == s[idx2]:
                print(hamming_distance - 2)
                print(idx1 + 1, idx2 + 1)
                return

    # Check for a beneficial single swap reducing distance by 1
    for ch in s_mismatches:
        for idx in s_mismatches[ch]:
            if t[idx] in s_mismatches and s[idx] in t_mismatches:
                print(hamming_distance - 1)
                swap_idx = s_mismatches[t[idx]][0] + 1
                print(idx + 1, swap_idx)
                return

    print(hamming_distance)
    print("-1 -1")

min_hamming_distance()
