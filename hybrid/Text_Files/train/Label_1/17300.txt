#include <bits/stdc++.h>
using namespace std;
int C[400010], D[400010];
struct node {
  int a, b, c, d, ord_a, ord_b;
};
node A[400010];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> A[i].a >> A[i].b;
    C[A[i].a] = i, D[A[i].b] = i;
  }
  for (int i = 1; i <= n; i++) {
    A[i].ord_a = C[i], A[i].ord_b = D[i];
  }
  sort(A + 1, A + n + 1, [](const node &l, const node &r) {
    if (l.a == r.a) { return l.b > r.b; }
    return l.a < r.a;
  });
  int Max[400010] = {0};
  int ans = 0, Max_b = -1;
  for (int i = 1; i <= n; i++) {
    if (A[i].ord_b > Max_b) { ans++; }
    Max_b = max(Max_b, A[i].ord_b);
  }
  cout << ans << endl;
  return 0;
}
