#include <vector>
#include <cstring>
#include <cmath>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n, c;
  cin >> n >> c;
  vector<int> p(n), t(n);
  for (auto &i : p) cin >> i;
  for (auto &i : t) cin >> i;
  int x = p[0] - c * t[0];
  int y = p[n-1] - c * t[n-1];
  for (int i = 1; i < n; ++i) {
    x += max(p[i] - c * (t[i] + max(t[i-1], t[0])), 0);
    y += max(p[n-1-i] - c * (t[n-1-i] + max(t[n-i], t[n-1])), 0);
  }
  if (x > y) {
    cout << "Limak" << endl;
  } else if (x == y) {
    cout << "Tie" << endl;
  } else {
    cout << "Radewoosh" << endl;
  }
  return 0;
}
