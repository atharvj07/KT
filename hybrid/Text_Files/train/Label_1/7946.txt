#include<bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b){
    return b?gcd(b,a%b):a;
}
long long lcm(long long a, long long b){
    return a*b/gcd(a,b);
}
int main(){
    long long t;cin>>t;
    while(t--){
        long long l,r;cin>>l>>r;
        if(l*2>r){
            cout<<-1<<" "<<-1<<endl;
        }
        else if(l%2){
            if(l+1<=r)cout<<l<<" "<<l+1<<endl;
            else cout<<-1<<" "<<-1<<endl;
        }
        else{
            if(l+2<=r)cout<<l<<" "<<l+2<<endl;
            else cout<<-1<<" "<<-1<<endl;
        }
    }
}
