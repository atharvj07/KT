#include <bits/stdc++.h>

using namespace std;

string solve(int n, string s) {
  int cnt[4];
  for (int i = 0; i < 4; i++) cnt[i] = 0;
  for (int i = 0; i < n; i++) {
    char c = s[i];
    if (c == 'A')
      cnt[0]++;
    else if (c == 'C')
      cnt[1]++;
    else if (c == 'G')
      cnt[2]++;
    else if (c == 'T')
      cnt[3]++;
  }
  int avg = (n + 3) / 4;
  for (int i = 0; i < 4; i++) {
    if (cnt[i] > avg) return "===";
  }
  for (int i = 0; i < n; i++) {
    char &c = s[i];
    if (c == '?') {
      for (int j = 0; j < 4; j++) {
        if (cnt[j] < avg) {
          cnt[j]++;
          if (j == 0)
            c = 'A';
          else if (j == 1)
            c = 'C';
          else if (j == 2)
            c = 'G';
          else
            c = 'T';
          break;
        }
      }
    }
  }
  return s;
}

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  string ans = solve(n, s);
  cout << ans << endl;
  return 0;
}
