#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 5;
vector<int> g[MAXN];
int par[MAXN], sz[MAXN];

int find(int u) { return par[u] = (u == par[u] ? u : find(par[u])); }
void unite(int u, int v) {
  u = find(u);
  v = find(v);
  if (u == v) return;
  if (sz[u] > sz[v]) {
    par[v] = u;
    sz[u] += sz[v];
  } else {
    par[u] = v;
    sz[v] += sz[u];
  }
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, a, b;
    cin >> n >> m >> a >> b;
    for (int i = 1; i <= n; i++) {
      par[i] = i;
      sz[i] = 1;
      g[i].clear();
    }
    while (m--) {
      int u, v;
      cin >> u >> v;
      g[u].push_back(v);
      g[v].push_back(u);
    }
    set<int> ab;
    for (auto &u : g[a]) unite(u, a), ab.insert(find(u));
    for (auto &u : g[b]) unite(u, b), ab.insert(find(u));
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
      if (i == a || i == b) continue;
      cnt += ab.count(find(i)) == 2;
    }
    cout << cnt / 2 << '\n';
  }
  return 0;
}
