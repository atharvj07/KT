n, k = map(int, input().split())
a = list(map(int, input().split()))

operations = {}
for number in a:
    steps = 0
    while number > 0:
        if number in operations:
            operations[number].append(steps)
        else:
            operations[number] = [steps]
        number //= 2
        steps += 1
    if 0 in operations:
        operations[0].append(steps)
    else:
        operations[0] = [steps]

min_operations = float('inf')
for key in operations:
    operations[key].sort()
    if len(operations[key]) >= k:
        min_operations = min(min_operations, sum(operations[key][:k]))

print(min_operations)
