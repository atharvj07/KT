f = [input() for i in range(10)]
for i in range(10):
    for j in range(10):
        if f[i][j] == '.' and (
            ('X' * 5 in ''.join(f[i - k][j - k:j + 1] for k in range(min(i, j + 1))))
            or ('X' * 5 in ''.join(f[i + k][j - k:j + 1] for k in range(9 - max(i, j))))
            or ('X' * 5 in ''.join(f[i][j - k:j + 1] for k in range(min(i, 10 - j))))
            or ('X' * 5 in ''.join(f[i][j + k:j + 1] for k in range(9 - max(i, 10 - j))))
        ):
            print('YES')
            exit(0)
print('NO')
