import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextLong();
            long k = sc.nextLong();
            long b = (k - 1) % (2 * n - 2) + 1;
            if (b > n) b -= n;
            System.out.println(b);
        }
    }
}
