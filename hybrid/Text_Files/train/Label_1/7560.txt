#include <bits/stdc++.h>
using namespace std;
string read() {
    string s;
    cin >> s;
    return s;
}
int readInt() {
    int a;
    cin >> a;
    return a;
}
int main() {
    int t = readInt();
    while (t--) {
        int n = readInt();
        vector<int> v(n);
        for (int i = 0; i < n; i++) { v[i] = readInt(); }
        int mx = v[n - 1], ans = 0;
        for (int i = n - 2; i >= 0; i--) {
            if (v[i] > mx) {
                ans++;
            } else {
                mx = v[i];
            }
        }
        cout << ans << "\n";
    }
}
