import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        char[] s = in.next().toCharArray();
        char[] t = in.next().toCharArray();
        int diffCount = 0;
        int[] diffPos = new int[n + 1];
        for (int i = 0; i < n; i++) {
            if (s[i] != t[i]) {
                diffCount++;
                diffPos[diffCount] = i + 1;
            }
        }
        if (diffCount % 2 == 1) {
            System.out.println(-1);
        } else {
            System.out.println(diffCount / 2);
            for (int i = 1; i <= diffCount / 2; i++) {
                System.out.println(diffPos[i] + " " + diffPos[diffCount - i + 1]);
            }
        }
    }
}
