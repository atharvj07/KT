import java.util.Scanner;

public class InnopolisConstruction {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] heights = new int[n];
        for (int i = 0; i < n; i++) {
            heights[i] = scanner.nextInt();
        }

        int[] minHours = new int[n / 2 + 1];
        minHours[0] = 0;
        for (int i = 1; i <= n / 2; i++) {
            int minTime = Integer.MAX_VALUE;
            for (int j = 0; j < n; j++) {
                if (heights[j] > i) {
                    minTime = Math.min(minTime, heights[j] - i);
                }
            }
            minHours[i] = minHours[i - 1] + minTime;
        }

        for (int i = 0; i <= n / 2; i++) {
            System.out.print(minHours[i] + " ");
        }
    }
}
