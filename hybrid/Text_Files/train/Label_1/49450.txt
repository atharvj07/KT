import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int k = sc.nextInt();

        int[] h = new int[2 * n];
        for (int i = 0; i < 2 * n; i++)
            h[i] = i / 2 + 1;

        List<int[]> reqs = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            int x = sc.nextInt() - 1;
            String s = sc.next();
            int y = sc.nextInt() - 1;
            int sign = "><=>".indexOf(s.charAt(1));
            reqs.add(new int[]{x, sign, y});
        }

        long ans = 0;
        int[] ord = new int[2 * n];
        do {
            boolean ok = true;
            for (int[] req : reqs)
                ok &= (h[ord[req[0]]] << req[1]) > h[ord[req[2]]] - 1;
            if (ok)
                ans++;
        } while (nextPermutation(ord));

        System.out.println(ans);
    }

    static boolean nextPermutation(int[] p) {
        int i = p.length - 1;
        while (i > 0 && p[i] <= p[i - 1])
            i--;
        if (i <= 0)
            return false;
        int j = p.length - 1;
        while (p[j] <= p[i - 1])
            j--;
        int t = p[i - 1];
        p[i - 1] = p[j];
        p[j] = t;
        while (i < p.length) {
            j = p.length - 1;
            while (p[j] <= p[i])
                j--;
            t = p[i];
            p[i] = p[j];
            p[j] = t;
            i++;
        }
        return true;
    }
}
