#include <iostream>
#include <string>
int main() {
  int n;
  std::string s0;
  std::cin >> n;
  for (int i = 0; i < n; ++i) {
    std::string s;
    std::cin >> s;
    if (i != 0) {
      int n = 0;
      for (int j = 0; j < s.size(); ++j) {
        if (j >= s0.size() || s[j] != s0[s0.size() - j - 1]) { break; }
        ++n;
      }
      s0 = s0.substr(0, s0.size() - n) + s;
    } else {
      s0 = s;
    }
  }
  std::cout << s0 << std::endl;
}
