#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const ll M = 998244353;
ll powmod(ll x, ll p) {
  if (p == 0) return 1;
  ll t = powmod(x, p / 2);
  t = t * t % M;
  if (p % 2) t = t * x % M;
  return t;
}

int main() {
  ll n;
  cin >> n;
  string s;
  cin >> s;
  map<char, ll> cnt;
  for (int i = 0; i < n; i++) cnt[s[i]]++;
  ll sum = 0, mx = cnt[s.front()];
  for (auto [_, c] : cnt) {
    sum += c;
    mx = max(mx, c);
  }
  if (sum - mx <= 1) return cout << n * (n - 1) % M << endl, 0;
  return cout << mx * (sum - mx) % M << endl, 0;
}
