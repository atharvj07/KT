n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
ans = 0
for i in range(n):
	if not any(x % 2 == 0 for x in a[i]): continue
	for j in range(m):
		if a[i][j] % 2 == 0: ans += 1
		a[i][j] += 1
for j in range(m):
	if not any(a[i][j] % 2 == 0 for i in range(n)): continue
	for i in range(n):
		if a[i][j] % 2 == 0: ans += 1
		a[i][j] += 1
print(ans)
