def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def find_sets(n, k):
    m = 4 * n * k
    sets = []
    for i in range(n):
        sets.append([k * (4 * i + j + 1) for j in range(4)])
    return m, sets

n, k = map(int, input().split())
m, sets = find_sets(n, k)
print(m)
for s in sets:
    print(*s)
