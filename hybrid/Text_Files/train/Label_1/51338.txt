import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int L = sc.nextInt();
        int[] Kefa = new int[n];
        int[] Sasha = new int[n];
        for (int i = 0; i < n; i++) {
            Kefa[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            Sasha[i] = sc.nextInt();
        }
        boolean flag = false;
        for (int i = 0; i < L - 1; i++) {
            int[] newKefa = new int[n];
            for (int j = 0; j < n; j++) {
                newKefa[(j + i + 1) % n] = Kefa[j];
            }
            if (Arrays.equals(newKefa, Sasha)) {
                flag = true;
                break;
            }
        }
        System.out.println(flag ? "YES" : "NO");
    }
}
