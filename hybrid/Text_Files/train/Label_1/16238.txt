#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
int main(){
    int t; cin>>t;
    while(t--){
        int n; cin>>n;
        vector<int>a(n),b(n);
        for(int i=0; i<n; i++) cin>>a[i];
        for(int i=0; i<n; i++) cin>>b[i];
        int asum=0,bsum=0,a0=0,a1m=0,b1=0;
        for(int i=0; i<n; i++){
            asum+=a[i];
            bsum+=b[i];
            if(a[i]==0) a0++;
            if(a[i]==1) a1m=max(a1m,b[i]);
            if(b[i]==1) b1++;
        }
        if(asum!=bsum) cout<<"NO"<<endl;
        else if(a0+a1m>=b1) cout<<"YES"<<endl;
        else cout<<"NO"<<endl;
    }
}
