import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    int[] A = new int[1 << N];
    for (int i = 0; i < (1 << N); i++) {
      A[i] = sc.nextInt();
    }
    int[] sortedA = A.clone();
    Arrays.sort(sortedA);
    int result = sortedA[(1 << N) - 1];
    for (int i = (1 << N) - 2; i >= 0; i--) {
      if (result > sortedA[i]) {
        result -= sortedA[i];
      } else {
        break;
      }
    }
    System.out.println(result);
  }
}
