#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;
const int mod = 1e9 + 7;

int main() {
  int n, m, dp[1005][1005];
  cin >> n >> m;
  vector<int> a(m);
  for (int i = 0; i < m; ++i) cin >> a[i];
  for (int i = 0; i < m; ++i) {
    for (int j = 0; j < m; ++j) {
      if (i == j)
        dp[i][j] = 1;
      else if (a[i] + 1 == a[j] || a[i] - 1 == a[j])
        dp[i][j] = 1;
      else
        dp[i][j] = 0;
    }
  }
  for (int i = m; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      if (a[j] + 1 == i || a[j] - 1 == i)
        dp[i][j] = dp[a[j] - 1][j];
      else
        dp[i][j] = 0;
    }
  }
  int ans = 0;
  for (int i = 0; i < m; ++i) {
    ans = (ans + dp[n - 1][i]) % mod;
  }
  cout << ans << endl;
  return 0;
}
