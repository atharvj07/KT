#include <iostream>
#include <vector>
#include <queue>
using namespace std;
int main() {
    int n, d;
    cin >> n >> d;
    vector<vector<int>> adj(n + 1);
    for (int i = 1; i <= d; i++) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    for (int k = 1; k <= d; k++) {
        vector<int> dist(n + 1, -1);
        queue<int> q;
        dist[1] = 0;
        q.push(1);
        int ans = 0;
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            ans = max(ans, dist[u]);
            for (auto v : adj[u]) {
                if (dist[v] == -1) {
                    dist[v] = dist[u] + 1;
                    q.push(v);
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
