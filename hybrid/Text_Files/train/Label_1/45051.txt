import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        for (int i = 0; i < d; i++) {
            int n = sc.nextInt();
            for (int j = n; j > 0; j--) {
                for (int k = 0; k < n; k++) {
                    if (k + j == n) {
                        System.out.print("#");
                    } else if (k == n - j) {
                        System.out.print("#");
                    } else {
                        System.out.print(" ");
                    }
                }
                System.out.println();
            }
            System.out.println();
        }
    }
}
