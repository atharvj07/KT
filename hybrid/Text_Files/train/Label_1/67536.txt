import sys

t = int(input())

for test in range(t):
    n = int(input())
    a = list(map(int, input().split()))

    ans = 0

    for i in range(n):
        for j in range(i + 1, n):
            if abs(a[i] - a[j]) <= 2:
                for k in range(j + 1, n):
                    if abs(max(a[i], a[j], a[k]) - min(a[i], a[j], a[k])) <= 2:
                        ans += 1

    print(ans)
