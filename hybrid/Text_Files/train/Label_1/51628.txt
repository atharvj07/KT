import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int numCases = input.nextInt();
        for (int i = 0; i < numCases; i++) {
            int numCols = input.nextInt();
            int numTrains = input.nextInt();
            char[][] grid = new char[3][numCols];
            boolean[][] trainPos = new boolean[3][numCols];
            int startRow = 0;
            int startCol = 0;
            for (int j = 0; j < 3; j++) {
                String row = input.next();
                grid[j] = row.toCharArray();
                for (int k = 0; k < numCols; k++) {
                    if (grid[j][k] == 's') {
                        startRow = j;
                        startCol = k;
                    } else if (grid[j][k] != '.') {
                        trainPos[j][k] = true;
                    }
                }
            }
            boolean[][] visited = new boolean[3][numCols];
            boolean possible = dfs(grid, trainPos, visited, startRow, startCol, numCols);
            if (possible) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }

    public static boolean dfs(char[][] grid, boolean[][] trainPos, boolean[][] visited, int row, int col, int numCols) {
        if (row < 0 || row >= 3 || col < 0 || col >= numCols) {
            return false;
        }
        if (trainPos[row][col] || visited[row][col]) {
            return false;
        }
        if (col == numCols - 1) {
            return true;
        }
        visited[row][col] = true;
        boolean canMove = false;
        if (row > 0) {
            canMove |= dfs(grid, trainPos, visited, row - 1, col + 1, numCols);
        }
        canMove |= dfs(grid, trainPos, visited, row, col + 1, numCols);
        if (row < 2) {
            canMove |= dfs(grid, trainPos, visited, row + 1, col + 1, numCols);
        }
        visited[row][col] = false;
        return canMove;
    }
}
