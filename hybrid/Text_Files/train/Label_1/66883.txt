home, away = input(), input()
n = int(input())
players = {}
for _ in range(n):
    time, side, num, color = input().split()
    side = home if side == 'h' else away
    if side not in players:
        players[side] = {}
    if num not in players[side]:
        players[side][num] = 0
    if color == 'y':
        players[side][num] += 1
        if players[side][num] == 2:
            print(side, num, time)
    else:
        print(side, num, time)
