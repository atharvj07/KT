#include <cstdio>
#include <cmath>

const int mod = 998244353;
const int nax = 1005;
int fact[nax], invfact[nax];

int fpow(int x, int y) {
    int r = 1;
    while(y) {
        if(y&1) r = (long long)r * x % mod;
        x = (long long)x * x % mod;
        y >>= 1;
    }
    return r;
}

void pre() {
    fact[0] = 1;
    for(int i = 1; i < nax; i++) fact[i] = (long long)i * fact[i-1] % mod;
    invfact[0] = invfact[1] = 1;
    for(int i = 2; i < nax; i++) invfact[i] = (long long)invfact[mod%i] * (mod - mod/i) % mod;
    for(int i = 2; i < nax; i++) invfact[i] = (long long)invfact[i] * invfact[i-1] % mod;
}

int choose(int n, int r) {
    return (long long)fact[n] * invfact[r] % mod * invfact[n-r] % mod;
}

int main() {
    pre();
    int n, k; scanf("%d%d", &n, &k);
    int ans = 0;
    for(int i = 1; i <= n; i += 2) if(k % i == 0) ans = (ans + (long long)choose(n, i) * fpow(2, i*k/i - k) % mod) % mod;
    printf("%d\n", ans);
    return 0;
}
