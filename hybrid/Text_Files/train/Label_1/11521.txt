#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int tt;
    cin >> tt;
    while (tt--) {
        int n;
        cin >> n;
        vector<ll> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a.rbegin(), a.rend());
        ll sum = 0, ans = 0;
        for (auto &e : a) {
            ans += e - 1;
            sum += max(0LL, e - 1);
        }
        cout << sum << '\n';
    }
}
