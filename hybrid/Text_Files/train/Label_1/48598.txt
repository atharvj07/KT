import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class JigsawArrangement {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int H = scanner.nextInt();

        List<Piece> pieces = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            int A_i = scanner.nextInt();
            int B_i = scanner.nextInt();
            int C_i = scanner.nextInt();
            int D_i = scanner.nextInt();
            pieces.add(new Piece(A_i, B_i, C_i, D_i));
        }

        boolean possible = isArrangementPossible(pieces, H);
        System.out.println(possible ? "YES" : "NO");
    }

    static boolean isArrangementPossible(List<Piece> pieces, int H) {
        // Sort pieces by height of center part in descending order
        pieces.sort((p1, p2) -> Integer.compare(p2.H, p1.H));

        // Initialize table with empty spaces
        boolean[][] table = new boolean[H + 1][2 * H + 1];

        // Place the first piece on the table
        if (!placePiece(table, pieces.get(0), 0, 0)) {
            return false;
        }

        // Place the remaining pieces
        for (int i = 1; i < pieces.size(); i++) {
            Piece piece = pieces.get(i);
            for (int j = 0; j <= H; j++) {
                for (int k = 0; k <= H; k++) {
                    if (placePiece(table, piece, j, k)) {
                        break;
                    }
                }
            }
            if (!isTableFull(table)) {
                return false;
            }
        }

        return true;
    }

    static boolean placePiece(boolean[][] table, Piece piece, int row, int col) {
        // Check if the piece fits on the table
        if (row + piece.H > table.length || col + piece.A + piece.B > table[0].length) {
            return false;
        }

        // Check if the piece overlaps with any other piece
        for (int i = row; i < row + piece.H; i++) {
            for (int j = col; j < col + piece.A + piece.B; j++) {
                if (table[i][j]) {
                    return false;
                }
            }
        }

        // Place the piece on the table
        for (int i = row; i < row + piece.H; i++) {
            for (int j = col; j < col + piece.A; j++) {
                table[i][j] = true;
            }
        }
        for (int i = row + piece.C; i < row + piece.H; i++) {
            for (int j = col + piece.A; j < col + piece.A + piece.B; j++) {
                table[i][j] = true;
            }
        }

        return true;
    }

    static boolean isTableFull(boolean[][] table) {
        for (int i = 0; i < table.length; i++) {
            for (int j = 0; j < table[0].length; j++) {
                if (!table[i][j]) {
                    return false;
                }
            }
        }
        return true;
    }

    static class Piece {
        int A, B, C, D, H;

        Piece(int A, int B, int C, int D) {
            this.A = A;
            this.B = B;
            this.C = C;
            this.D = D;
            this.H = A + B + C + D;
        }
    }
}
