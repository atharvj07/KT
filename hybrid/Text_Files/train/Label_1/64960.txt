n, m = map(int, input().split())
a = [input() for i in range(n)]
for i in range(n):
    for j in range(m):
        if a[i][j] == 'G':
            if j > 0 and a[i][j - 1] == '*':
                a[i][j], a[i][j - 1] = a[i][j - 1], a[i][j]

ans = 0
for i in range(n):
    for j in range(m):
        if a[i][j] == 'G':
            if a[i][j + 1] == 'S':
                ans += 1
                break
            elif a[i][j + 1] == '*':
                a[i][j], a[i][j + 1] = a[i][j + 1], a[i][j]
print(-1 if ans != n else ans)
