n = int(input())
a = list(map(int, input().split()))
if sum(a) == 0:
    print('NO')
else:
    print('YES')
    print(len(a))
    l = 0
    while l < n:
        r = l
        s = 0
        while r < n and s + a[r] >= 0:
            s += a[r]
            r += 1
        if s == 0:
            print('NO')
            break
        print(l + 1, r)
        l = r
