#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, c, best = 0, profit = 0;
    cin >> n >> c;
    vector<int> x(n);
    for (int i = 0; i < n; ++i) cin >> x[i];
    for (int i = 0; i < n - 1; ++i) {
        profit += x[i + 1] - x[i];
        if (profit + x[i + 1] - c > best) best = profit + x[i + 1] - c;
        if (profit < 0) profit = 0;
    }
    cout << best << endl;
    return 0;
}
