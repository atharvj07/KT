#include<bits/stdc++.h>
using namespace std;
int main(){
    ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int n, ans=0;
    cin >> n;
    vector<long long> x(n), ord;
    map<long long, int> mp;
    for(int i=0; i<n; i++){
        cin >> x[i];
        ord.push_back(x[i]);
        mp[x[i]]=1;
    }
    sort(ord.begin(), ord.end());
    vector<int> ansindx;
    for(int i=1; i<n; i++){
        long long d=ord[i]-ord[i-1];
        for(int j=1; (1LL<<j)<=d; j++){
            if((d&(1LL<<j)) && mp[ord[i]-((1LL<<j))]) {
                ans=max(ans, j+1);
                ansindx.clear();
                ansindx.push_back(i);
                ansindx.push_back(lower_bound(ord.begin(), ord.end(), ord[i]-((1LL<<j)) ) - ord.begin());
                int k=j+1;
                while(k<=ans){
                    if(mp[ord[i]-((1LL<<k))]){
                        ans=k;
                        ansindx.push_back(lower_bound(ord.begin(), ord.end(), ord[i]-((1LL<<k))) - ord.begin());
                    }
                    k++;
                }
            }
        }
    }
    cout << ans << endl;
    for(auto it: ansindx) cout << x[it] << " "; cout << endl;
}
