import sys

L = []
for i in range(int(input())):
  li = list(input())
  L.append(li)
  assert len(li) == 2
  assert li[0] in 'BW?'
  assert li[1] in 'BW?'

n = len(L)
mod = 998244353

# dp[i][j] = number of ways to color the first i dominoes,
#            with the right cell of the i-th domino colored j
dp = [[0 for j in range(2)] for i in range(n+1)]
dp[0][0] = dp[0][1] = 1
for i in range(1,n+1):
    # try color i-th domino with right cell color 0 (black)
    if i == n or L[i][1] != 'W':
        dp[i][0] += dp[i-1][1]
    # try color i-th domino with right cell color 1 (white)
    if i == n or L[i][1] != 'B':
        dp[i][1] += dp[i-1][0]
    dp[i][0] %= mod
    dp[i][1] %= mod
print(dp[n][0])
