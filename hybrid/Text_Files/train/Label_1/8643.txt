#include<bits/stdc++.h>
using namespace std;

void answer(string& s, int l, int r, int f[]) {
  deque<int> d;
  int cnt[256] = {};
  for(int i = l-1; i < r; ++i) {
    char ch = s[i];
    cnt[ch]++;
    if(ch=='<') d.push_front(i);
    else if(ch=='>') { d.push_front(i); d.pop_front(); }
    else {
      int p = d.empty() ? l-1 : d.front()-1;
      f[ch-'0'] += p - i + 1;
      --cnt[ch];
      if(cnt[ch] == 0) d.pop_front();
    }
  }
}

int main() {
  int n, q; cin>>n>>q;
  string s; cin>>s;
  while(q--) {
    int l, r; cin>>l>>r;
    int f[10] = {};
    answer(s, l, r, f);
    for(int& i : f) cout<<i<<' ';
    cout<<'\n';
  }
  return 0;
}
