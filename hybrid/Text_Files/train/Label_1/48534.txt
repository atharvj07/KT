import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long N = scanner.nextLong();
        long K = scanner.nextLong();
        int[] A = new int[(int) N];
        for (int i = 0; i < N; i++) A[i] = scanner.nextInt();

        // Count the frequency of each number
        HashMap<Integer, Integer> count = new HashMap<>();
        for (int i = 0; i < N; i++) {
            count.put(A[i], count.getOrDefault(A[i], 0) + 1);
        }

        // If already within K, print 0
        if (count.size() <= K) {
            System.out.println(0);
            return;
        }

        // Sort the frequency in ascending order
        List<Integer> frequencies = new ArrayList<>(count.values());
        Collections.sort(frequencies);

        // Find the minimum number of balls to rewrite to satisfy K different numbers
        int ans = 0;
        for (int f : frequencies) ans += Math.max(f - 1, 0);

        System.out.println(ans);
    }
}
