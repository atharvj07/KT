for t in range(int(input())):
    n = int(input())
    w = list(map(int, input().split()))
    w.sort()
    mn = float('inf')
    fr = [0] * (n + 2)
    for i in w:
        fr[i] += 1
        mn = min(mn, i)
    ans = 0
    if mn == 1:
        print(n // 2)
        continue
    x = 0
    for i in range(mn, n + 1):
        ans += fr[i] // 2
        x += fr[i] % 2
    if x % 2 and fr[2 * mn]:
        x //= 2
        ans += 1
        x %= 2
    ans += x // 2
    print(ans)
