#include <cstdio>

int check(char c) {
    return c == '_' || c == 'X';
}

int main() {
    char s[100];
    scanf("%s", s);
    int len = 0, cntx = 0;
    while (s[len]) {
        if (check(s[len])) cntx++;
        len++;
    }
    int ans = 0, mod = 0;
    for (int i = len - 1; i >= 0; i--) {
        if (i == len - 1) mod = s[i] - '0';
        else {
            mod = mod * 10;
            if (check(s[i])) mod += 0;
            else mod += s[i] - '0';
        }
        mod %= 25;
    }
    if (mod == 0) {
        int sum = 0;
        for (int i = 0; i < len; i++) {
            if (s[i] == 'X') sum += 4;
            if (s[i] == '_') sum++;
        }
        ans = 1;
        while (sum--) ans *= 10;
    }
    printf("%d\n", ans);
    return 0;
}
