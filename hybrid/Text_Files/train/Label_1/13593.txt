#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    bool ok = true;
    for (int i = 0; i < n; i++) {
      bool found = false;
      for (int j = 0; j < n; j++) {
        for (int k = 0; k < n; k++) {
          if (j == k) continue;
          if (a[i] == a[j] - a[k]) {
            found = true;
            break;
          }
        }
      }
      ok &= found;
    }
    cout << (ok ? "YES" : "NO") << '\n';
  }
  return 0;
}
