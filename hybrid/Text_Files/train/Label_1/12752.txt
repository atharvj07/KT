#include <iostream>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    int pivot = 0;
    long long sum1 = 0, sum2 = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '^') {
            pivot = i;
        } else if (s[i] != '=') {
            int w = s[i] - '0';
            if (i < pivot)
                sum1 += (long long)w * (pivot - i);
            else
                sum2 += (long long)w * (i - pivot);
        }
    }
    if (sum1 == sum2)
        cout << "balance" << endl;
    else if (sum1 > sum2)
        cout << "left" << endl;
    else
        cout << "right" << endl;
    return 0;
}
