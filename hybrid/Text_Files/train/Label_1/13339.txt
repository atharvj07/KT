#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;
int32_t main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n), b(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        for (int i = 0; i < n; i++) cin >> b[i];
        int ans = INT_MAX, sum = 0;
        sort(a.begin(), a.end());
        sort(b.begin(), b.end());
        for (int i = 0; i < n; i++) {
            if (sum + a[i] >= m) {
                ans = min(ans, sum + b[i] - a[i]);
            }
            sum += b[i];
        }
        if (sum >= m)
            cout << ans << '\n';
        else
            cout << "-1\n";
    }
    return 0;
}
