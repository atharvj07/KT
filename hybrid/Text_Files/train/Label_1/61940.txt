n = int(input())

position = 0
valid = True

for _ in range(n):
    t, dir = input().split()
    t = int(t)

    if dir == "South":
        position += t
    elif dir == "North":
        position -= t

    if position < 0 or position > 20000:
        valid = False
        break

    if (position == 0 or position == 20000) and dir in ["East", "West"]:
        valid = False
        break

if position != 0:
    valid = False

print("YES" if valid else "NO")
