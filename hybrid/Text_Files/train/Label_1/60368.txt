n, I = map(int, input().split())
a = list(map(int, input().split()))
K = len(set(a))
k = (K+1).bit_length() - 1
B = (n*k+7)//8
if B <= I: print(0)
else:
    a.sort()
    l1, r1 = a[B-I//2], a[n-1-(I-B+I//2)]
    l2, r2 = a[I//2], a[n-1-B]
    if (r1-l1) < (r2-l2): l, r = l1, r1
    else: l, r = l2, r2
    ans = 0
    for x in a:
        if x<l: ans += l-x
        elif x>r: ans += x-r
    print(ans)
