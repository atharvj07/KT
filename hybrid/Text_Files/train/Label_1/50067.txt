import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.next();
        int n = s.length();
        long ans = 0;
        for (int i = 0; i < n; ++i) {
            for (int j = i + 1; j < n; ++j) {
                if (s.charAt(i) == '1' && s.charAt(j) == '0') {
                    ans++;
                }
            }
        }
        System.out.println(ans % 1000000007L);
    }
}
