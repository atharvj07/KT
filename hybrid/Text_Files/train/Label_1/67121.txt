import sys
input = sys.stdin.readline

q = int(input())

for _ in range(q):
    n = int(input())
    p = list(map(int,input().split()))
    ans = [0 for i in range(n)]
    vis = [0 for i in range(n)]

    for i in range(n):
        if vis[i]==0:
            j = i
            cnt = 0
            while vis[j]==0:
                vis[j] = 1
                cnt += 1
                j = p[j]-1
            ans[i] = cnt

    for x in ans:
        print(x, end=' ')
    print('')
