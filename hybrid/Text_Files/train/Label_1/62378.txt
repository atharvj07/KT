import sys
input = sys.stdin.readline

n = int(input())
a = [int(x) for x in input().split()]
e = [[] for i in range(n)]
for i in range(n - 1):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    e[u].append(v)
    e[v].append(u)
vis = [False] * n
ans = -10**10
q = []
for i in range(n):
    if len(e[i]) == 1:
        q.append(i)
        vis[i] = True
while q:
    v = q.pop(0)
    if ans < a[v]:
        ans = a[v]
    for x in e[v]:
        if not vis[x]:
            vis[x] = True
            a[x] += 1
            q.append(x)
print(ans)
