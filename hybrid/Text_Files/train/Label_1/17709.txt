#include <bits/stdc++.h>
using namespace std;

int main()
{
    cin.tie(0);
    ios_base::sync_with_stdio(0);
    int n, a[100005], b[1000005];
    cin >> n;
    for (int i = 0; i < n; i++)
        cin >> a[i];
    int ans = 0;
    for (int l = 0; l < n; l++)
    {
        int tmp = 0;
        for (int r = l; r < n; r++)
        {
            tmp |= a[r];
            if (!b[tmp])
            {
                ans++;
                b[tmp] = 1;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
