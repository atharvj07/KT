import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t1 = in.nextInt();
        int t2 = in.nextInt();
        int x1 = in.nextInt();
        int x2 = in.nextInt();
        int t0 = in.nextInt();
        long y2 = 0, y1 = 0;
        for (int i = 0; i <= x1; i++) {
            int cur = t0 - t1 * i;
            for (int j = 0; j <= x2; j++) {
                if (t2 * j >= cur && (t2 * j - cur) * (1000000000L) / x2 >= (t0 - t1 * i) * (1000000000L) / x1) {
                    y1 = i;
                    y2 = j;
                }
            }
        }
        System.out.println(y1 + " " + y2);
    }
}
