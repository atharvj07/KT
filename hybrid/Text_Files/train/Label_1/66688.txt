import sys

n, m, k = map(int, input().split())
w = list(map(int, input().split()))

games = {}

for i in range(1, m + 1):
    a, b = map(int, input().split())
    games[i] = [a, b]

ans = [-1] * m

l = [0] * n

for i in range(1, m + 1):
    if l[games[i][0] - 1] == l[games[i][1] - 1]:
        ans[i - 1] = 1 + min(l[games[i][0] - 1], k - l[games[i][0] - 1])
    else:
        ans[i - 1] = 1 + min(l[games[i][0] - 1], l[games[i][1] - 1])
    l[games[i][0] - 1] = ans[i - 1]
    l[games[i][1] - 1] = ans[i - 1]

for i in range(0, m):
    print(ans[i])
