#include <bits/stdc++.h>

using namespace std;

using ll = long long;
const ll M = 1e9;

int main() {
  ll n;
  cin >> n;
  vector<ll> ans(n);

  map<ll, ll> cnt;
  for (ll i = 0; i < n; i++) {
    ll v;
    cin >> v;
    cnt[v]++;
  }
  ll x = 0;
  for (const auto& [v, c] : cnt) {
    for (ll i = 0; i < c; i++) {
      ans[x] = v;
      if (x > 0 && v > ans[x - 1] && ans[x] % ans[x - 1] == 0)
        swap(ans[x - 1], ans[x]);
      x++;
    }
  }
  for (ll i = 0; i < n; i++) {
    cout << ans[i] << " ";
  }
  cout << "\n";
}
