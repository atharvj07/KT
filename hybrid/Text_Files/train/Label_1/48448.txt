import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            String s = input.next();
            System.out.println(s.charAt(0) != s.charAt(s.length() - 1) ? "DA" : "NET");
        }
    }
}
