#include <bits/stdc++.h>
using namespace std;
int n, m, l[100005], c[100005], p[100005];
bool vis[100005];
int main() {
  cin >> n >> m;
  for (int i = 0; i < m; ++i) cin >> l[i];
  vector<int> v;
  for (int i = 0; i < m; ++i) {
    int x;
    cin >> x;
    v.push_back(x);
  }
  int cnt = 0;
  for (int i = 0; i <= n; ++i) {
    if (vis[i] == 0) {
      p[cnt] = i;
      for (int j = i; j < i + l[cnt]; ++j) {
        vis[j] = 1;
        c[j] = cnt + 1;
      }
      cnt++;
    }
  }
  if (cnt == m) {
    for (int i = 0; i < m; ++i) cout << p[i] << " ";
    cout << endl;
  } else
    cout << "-1" << endl;
  return 0;
}
