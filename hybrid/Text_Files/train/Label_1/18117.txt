#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007
int main()
{
    int n, cnt[26];
    cin >> n;
    string s, t;
    cin >> s >> t;
    for(int i = 0; i < n; i++)
    {
        if(s[i] != t[i])
        {
            cnt[s[i] - 'a']++;
            cnt[t[i] - 'a']++;
        }
    }
    ll ans = 0;
    for(int i = 0; i < 26; i++)
    {
        if(cnt[i] == 2) ans++;
    }
    cout << ans << endl;
    return 0;
}
