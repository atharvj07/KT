#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int m;
  cin >> m;
  while (m--) {
    int l, r;
    cin >> l >> r;
    int cnt[3] = {};
    for (int i = l - 1; i < r; i++) { cnt[s[i] - 'x']++; }
    cout << ((cnt[0] != cnt[1]) || (cnt[0] != cnt[2]) || (cnt[1] != cnt[2])
                  ? "YES"
                  : "NO")
         << endl;
  }
  return 0;
}
