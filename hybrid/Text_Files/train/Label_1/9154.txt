#include <bits/stdc++.h>
#define mod 998244353
using namespace std;
long long dp[300005][2];
char s[300005];
int main() {
  int n, m, sum = 0;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> s + 1;
    for (int j = 1; j <= m; j++) {
      sum += (s[j] == 'o');
    }
  }
  long long ans = 1;
  for (int i = 0; i < sum; i++) ans = ans * 2 % mod;
  cout << (ans + mod) % mod << endl;
  return 0;
}
