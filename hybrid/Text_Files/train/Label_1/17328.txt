#include <bits/stdc++.h>

using namespace std;

const int N = 505, M = 505, INF = (int)1e9;

int n, m, a[N][M], f[N][1 << 11], ans[N];
bool ok;

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            cin >> a[i][j];
        }
    }
    memset(f, -1, sizeof f);
    f[0][0] = 0;
    for (int i = 1; i <= n; ++i) {
        for (int mask = 0; mask < (1 << 11); ++mask) {
            if (f[i - 1][mask] == -1) continue;
            f[i][mask] = max(f[i][mask], f[i - 1][mask]);
            for (int j = 1; j <= m; ++j) {
                int newmask = mask | a[i][j];
                if (newmask != mask) {
                    f[i][newmask] = max(f[i][newmask], f[i - 1][mask] + 1);
                    ans[i] = j;
                }
            }
        }
    }
    ok = (f[n][(1 << 11) - 1] != -1);
    if (ok) {
        puts("TAK");
        for (int i = n; i >= 1; --i) {
            cout << ans[i] << ' ';
        }
        cout << '\n';
    } else {
        puts("NIE");
    }
    return 0;
}
