#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
    int tt;
    cin >> tt;
    while (tt--) {
        int n, k;
        cin >> n >> k;
        int ans[k];
        int sum = n / k;
        int rem = n % k;
        for (int i = 0; i < rem; i++) {
            ans[i] = sum + 1;
        }
        for (int i = rem; i < k; i++) {
            ans[i] = sum;
        }
        for (int i = 0; i < k; i++) { cout << ans[i] << " "; }
        cout << endl;
    }
}
