#include<bits/stdc++.h>
using namespace std;

#define pb push_back

vector<int> adj[100005];
vector<int> color;

bool dfs(int node, int c){
	if(color[node] != -1){
		if(color[node] != c){
			return false;
		}
		return true;
	}
	color[node] = c;
	for(int x : adj[node]){
		if(!dfs(x, c ^ 1)) return false;
	}
	return true;
}

int main(){
	int n, m;
	cin >> n >> m;
	for(int i = 0; i < m; i++){
		int u, v;
		cin >> u >> v;
		u--, v--;
		adj[u].pb(v);
		adj[v].pb(u);
	}
	color.resize(n, -1);
	for(int i = 0; i < n; i++){
		if(color[i] == -1){
			if(!dfs(i, 0)){
				cout << "-1\n";
				return 0;
			}
		}
	}
	vector<int> part1, part2;
	for(int i = 0; i < n; i++){
		if(color[i] == 0) part1.pb(i + 1);
		else part2.pb(i + 1);
	}
	cout << part1.size() << '\n';
	for(int i : part1) cout << i << ' ';
	cout << '\n' << part2.size() << '\n';
	for(int i : part2) cout << i << ' ';
	cout << '\n';
	return 0;
}
