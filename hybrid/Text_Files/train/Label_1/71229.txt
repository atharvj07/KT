import sys

s, t = list(sys.stdin.readline().strip())+['0'], list(sys.stdin.readline().strip())+['0']
n = len(s); m = len(t)
a = [[0]*m for i in range(n)]
l = []
for i in range(1, n):
    for j in range(1, m):
        if s[i] == t[j]:
            a[i][j] = a[i-1][j-1] + 1
if a[n-1][m-1] == 0:
    print(-1)
    exit()
print(a[n-1][m-1])
i, j = n-1, m-1
while i != 0 and j != 0:
    if a[i][j] == a[i-1][j]:
        i -= 1
    elif a[i][j] == a[i][j-1]:
        j -= 1
    else:
        l.append((i, j))
        i -= 1
        j -= 1
l.reverse()
for x, y in l:
    if x < y:
        print(x, y)
    else:
        print(y, x)
