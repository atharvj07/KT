n = int(input())
semifinal1 = []
semifinal2 = []
for _ in range(n):
    a, b = map(int, input().split())
    semifinal1.append(a)
    semifinal2.append(b)

all_results = sorted(semifinal1 + semifinal2)
cutoff = all_results[n - 1]

output1 = ['1' if x <= cutoff else '0' for x in semifinal1]
output2 = ['1' if x <= cutoff else '0' for x in semifinal2]

print("".join(output1))
print("".join(output2))
