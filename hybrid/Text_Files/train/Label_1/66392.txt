import sys

A, B, C = map(int, sys.stdin.readline().split())
M = int(sys.stdin.readline())
L = list(map(int, sys.stdin.readline().split()))

pre = [0] * (A + 1)
for i in L:
    pre[i] = 1
for i in range(2, A + 1):
    pre[i] += pre[i - 1]

for i in range(1, M + 1):
    if pre[L[i - 1]] - (0 if i == 1 else pre[L[i - 2]]) == 0:
        print(i)
        exit()
print(-1)
