#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=2e5+5;
ll a[N],t[N];
int main(){
    int n; cin>>n;
    vector<ll> v;
    for(int i=0;i<n;i++) cin>>a[i];
    for(int i=0;i<n;i++) cin>>t[i];
    ll ct=0;
    for(int i=0;i<n;i++){
        if(a[i]%2){
            v.push_back(t[i]);
            ct++;
        }
    }
    sort(v.begin(),v.end());
    ll ans=0;
    int cnt=0;
    for(auto x:v){
        if(cnt==ct) break;
        if(ans+x*2>ans){
            ans+=x*2;
            cnt++;
        }
    }
    cout<<ans<<endl;
}
