import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int[] a = new int[12];
    for (int i = 0; i < 6; i++) {
      a[i * 2] = sc.nextInt();
      a[i * 2 + 1] = sc.nextInt();
    }
    Arrays.sort(a);
    System.out.println((a[0] == a[2] && a[2] == a[4] && a[4] == a[6] && a[0] == a[8] && a[8] == a[10])
        || (a[0] == a[4] && a[4] == a[8] && a[1] == a[5] && a[5] == a[9] && a[1] == a[11]) ? "yes" : "no");
  }
}
