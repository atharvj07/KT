import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long q = input.nextLong();
        if (q % 2 == 1 || q == 2) {
            System.out.println(1);
            System.out.println(0);
        } else {
            System.out.println(2);
            System.out.println(q / 2);
        }
    }
}
