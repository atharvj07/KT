from collections import deque
n, m = map(int, input().split())
r, c = map(int, input().split())
x, y = map(int, input().split())
grid = [input() for i in range(n)]
dist = [[10**9]*m for i in range(n)]
q = deque([(r-1, c-1)])
dist[r-1][c-1] = 0
while q:
    i, j = q.popleft()
    for ni, nj in ((i-1, j), (i+1, j), (i, j-1), (i, j+1)):
        if 0 <= ni < n and 0 <= nj < m and grid[ni][nj] == '.' and dist[ni][nj] > dist[i][j]+1:
            dist[ni][nj] = dist[i][j]+1
            if nj < j:
                x -= 1
            elif nj > j:
                y -= 1
            if x >= 0 and y >= 0: q.append((ni, nj))
print(sum(sum(dist[i][j] < 10**9 for j in range(m)) for i in range(n)))
