#include <bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    int ans=0;
    for(int i=1;i<=n;i++){
        bool flag =true;
        for(int j=2;j<=10;j++){
            if(i%j!=0){
                flag=false;
                break;
            }
        }
        ans+=flag;
    }
    cout<<ans<<endl;
    return 0;
}
