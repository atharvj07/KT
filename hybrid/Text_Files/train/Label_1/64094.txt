import sys

n, m = map(int, input().split())
ho = []
for _ in range(int(input())):
    x, y = map(int, input().split())
    ho.append((x, y))
re = []
for _ in range(int(input())):
    x, y = map(int, input().split())
    re.append((x, y))

mi = 2e9
for x, y in ho:
    for nx, ny in re:
        mx = max(abs(x - nx), abs(y - ny))
        mi = min(mi, mx)

print(mi)
for x, y in re:
    for nx, ny in ho:
        mx = max(abs(x - nx), abs(y - ny))
        if mx == mi:
            print(re.index((x, y)) + 1)
            sys.exit()
