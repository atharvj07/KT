#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int solve(const vector<string>& s) {
    vector<int> odd(s.size(), 0);
    for (int i = 0; i < s.size(); i++) {
        int o = 0;
        for (int j = 0; j < s[i].size(); j++) {
            if (s[i][j] == '1') o ^= 1<<j;
        }
        odd[i] = o;
    }
    int common = 0;
    for (int i = 0; i < odd.size(); i++) {
        common |= odd[i];
    }
    int ans = 0;
    for (int i = 0; i < odd.size(); i++) {
        if ((common & odd[i]) == odd[i]) ans++;
    }
    if ((common > 0) && ((1<<s[0].size()) - 1 == common)) ans--;
    return ans;
}

int main() {
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<string> s(n);
        for (int i = 0; i < n; i++) {
            cin >> s[i];
        }
        cout << solve(s) << endl;
    }
    return 0;
}
