import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int d = 0;
        for (int i = -1000; i <= 1000; i++) {
            if (i == 0) {
                continue;
            }
            int cnt = 0;
            for (int j = 0; j < n; j++) {
                if ((double) a[j] / i > 0) {
                    cnt++;
                }
            }
            if (cnt >= (n + 1) / 2) {
                d = i;
                break;
            }
        }
        System.out.println(d);
    }
}
