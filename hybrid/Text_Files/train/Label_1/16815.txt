#include <bits/stdc++.h>
#define pb push_back
#define mp make_pair
#define X first
#define Y second
typedef long long ll;
using namespace std;
const int inf=1<<30;
const int mod=1e9+7;
const int N=1000;

int main(){
    int n,m,k,x,y,ans=0,xd,yd;
    string s;
    cin>>n>>m>>k;
    vector<vector<char>> v(n+2, vector<char> (m+2)); 
    for(int i=1;i<=n;i++){
        string s;cin>>s;
        for(int j=1;j<=m;j++){
          v[i][j]=s[j-1];
          if(s[j-1]=='X') {
            x=i;y=j;
          }
        }
      }
    int dx[]={0,0,1,-1};
    int dy[]={1,-1,0,0};
    char dir[]={'R','L','D','U'};
    int px[k],py[k];
    for(int i=0;i<k;i++){
      int fx,fy;
      while(1){
        int nd=rand()%4;
        fx=x+dx[nd];fy=y+dy[nd];
        if(1<=fx && fx<=n && 1<=fy && fy<=m && v[fx][fy]!='*'){
          v[fx][fy]='*';
          px[i]=fx;
          py[i]=fy;
          s+=dir[nd];
          x=fx;y=fy;
          break;
        }
      }
    }
    for(int i=0;i<k;i++) v[px[i]][py[i]]='.';
    cout<<s;
    return 0;
}
