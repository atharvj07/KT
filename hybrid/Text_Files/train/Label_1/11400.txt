#include <iostream>
#include<vector>
int main() {
  long long a, b, c;
  std::cin >> a >> b >> c;
  std::vector<long long> l = {a, b, c};
  long long i = 0;
  for (std::vector<long long>::iterator it = l.begin();
       it != l.end() && i <= 3000000000000000000LL;
       ++it) {
    i = i + *it;
  }
  if (i <= 3000000000000000000LL)
    std::cout << "NO";
  else
    std::cout << "YES";
  std::cout << std::endl;
  return 0;
}
