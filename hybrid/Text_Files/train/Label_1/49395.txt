import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int[] x = new int[n];
    for (int i = 0; i < n; i++) {
      x[i] = scanner.nextInt();
    }
    int s = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if (x[i] > x[j]) {
          int tmp = x[i];
          x[i] = x[j];
          x[j] = tmp;
          s += x[i];
        }
      }
    }
    System.out.println(s);
  }
}
