#include <iostream>
#include <vector>

int main() {
  int n, m, q;
  std::cin >> n >> m >> q;
  std::vector<int> v(m);
  for (int i = 0; i < n; i++) {
    int a;
    std::cin >> a;
    v[a - 1]++;
  }
  int ans = 0;
  for (int i = 0; i < q; i++) {
    int k;
    std::cin >> k;
    while (k > n + v[ans]) k -= v[ans]++;
    ans = (ans + 1) % m;
    std::cout << ans + 1 << std::endl;
  }
  return 0;
}
