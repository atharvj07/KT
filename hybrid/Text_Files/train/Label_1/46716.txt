import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-->0) {
            int n = sc.nextInt(), x = sc.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) a[i] = sc.nextInt();
            Arrays.sort(a);
            int cnt = 0;
            for (int i = 0; i < n; i++) cnt+=a[i]<x?1:-1;
            System.out.println(Math.abs(cnt)+((cnt!=0)?1:0));
        }
    }
}
