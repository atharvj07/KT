from sys import stdin

n,m=map(int,stdin.readline().split())
arr=[]
ans=[]
for i in range(n):
    a=list(map(int,stdin.readline().split()))
    arr.append(a)
    x=[]
    for j in a:
        if j not in x:
            x.append(j)
    x.sort()
    a=dict()
    for j in range(len(x)):
        a[x[j]]=j+1
    ans.append(a)
for i in arr:
    for j in i:
        print(ans[arr.index(i)][j],end=" ")
    print()
