n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
c = list(map(int, input().split()))
a += b + c
a.sort()
ans, cnt1, cnt2 = 0, 0, 0
for j in range(1, len(a)):
    if a[j] == a[j - 1]:
        cnt1 += 1
        if cnt1 > cnt2:
            cnt2 = cnt1
            ans = cnt1 + 1
    else: cnt1, cnt2 = 0, 0
print(ans)
