#include<bits/stdc++.h>
#define ll long long
#define N 100005
using namespace std;
int n,m,ans,t,a[N],l[N],r[N],b[N],idx[N];
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++)cin>>a[i];
    for(int i=1;i<=m;i++){
        cin>>l[i]>>r[i];
        b[r[i]]++;
        b[l[i]-1]--;
    }
    for(int i=1;i<=n;i++){
        b[i]+=b[i-1];
        idx[i]=b[i];
    }
    sort(idx+1,idx+1+n);
    sort(a+1,a+1+n);
    for(int i=n,j=m;i>=1 && j>=1;){
        if(idx[j]==b[i]){
            t+=abs(a[i]-a[j]),j--,i--;
            ans++;
        }
        else i--;
    }
    cout<<t<<endl<<ans<<endl;
    for(int i=n,j=1;i>=1 && j<=m;){
        if(idx[j]==b[i] && abs(a[i]-a[j])==t){
            cout<<j<<" ",i--,j++;
        }
        else i--;
    }
    cout<<endl;
    return 0;
}
