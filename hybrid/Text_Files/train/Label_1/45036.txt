import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        int[] lanes = new int[n + 1];
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        for (int i = 0; i < m; i++) {
            int type = sc.nextInt();
            if (type == 0) {
                int lane = sc.nextInt();
                System.out.println(pq.poll());
                lanes[lane] = 0;
            } else {
                int car = sc.nextInt();
                pq.offer(car);
                lanes[findLane(lanes)] = car;
            }
        }
    }

    private static int findLane(int[] lanes) {
        int lane = -1;
        int minCars = Integer.MAX_VALUE;
        for (int i = 1; i < lanes.length; i++) {
            if (lanes[i] == 0 || lanes[i] < minCars) {
                lane = i;
                minCars = lanes[i];
            }
        }
        return lane;
    }
}
