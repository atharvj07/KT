import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] times = new int[n][5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                times[i][j] = sc.nextInt();
            }
        }
        int[] petya = new int[5];
        for (int i = 0; i < 5; i++) {
            petya[i] = times[1][i];
            if (petya[i] == -1) petya[i] = 120;
        }
        int[] best = new int[5];
        for (int i = 0; i < 5; i++) {
            best[i] = 1000;
            for (int j = 0; j < n; j++) {
                if (times[j][i] != -1) {
                    best[i] = Math.min(best[i], times[j][i]);
                }
            }
            best[i] = Math.max(best[i], 1);
        }
        long petyaScore = 0;
        for (int i = 0; i < 5; i++) {
            petyaScore += 1000 * (1 - (double) petya[i] / 250);
        }
        long vasyaScore = 0;
        for (int i = 0; i < 5; i++) {
            vasyaScore += 1000 * (1 - (double) best[i] / 250);
        }
        if (vasyaScore > petyaScore) System.out.println(0);
        else {
            int cnt = 0;
            for (int i = 0; i < 5; i++) {
                if (best[i] < petya[i]) cnt++;
            }
            System.out.println(cnt);
        }
    }
}
