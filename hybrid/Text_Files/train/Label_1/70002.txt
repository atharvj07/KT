mod = int(1e9) + 7
in_ = input().split()
n = int(in_[0])
m = int(in_[1])
g = [[] for _ in range(n)]
for i in range(n):
    g[i] += map(int, input().split()[1:])
fact = [1]
for i in range(1, m + 1):
    fact.append(fact[-1] * i % mod)
f = fact[m]
for gy in g:
    x = [0] * (m + 1)
    for gy_ in gy:
        x[gy_] += 1
    for k in x:
        if k == 1:
            f = f * k % mod
        else:
            f = f * fact[k] % mod
print(f)
