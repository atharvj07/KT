#include <iostream>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  bool prev = false;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    bool isvowel = (s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' || s[i] == 'u' || s[i] == 'y');
    if (isvowel) {
      if (i == n - 1) {
        if (cnt >= 1) cout << s[i];
      } else {
        if (s[i] == s[i + 1]) {
          cnt++;
          if (cnt == 2) continue;
        } else {
          if (cnt == 1) cout << s[i];
          else if (cnt == 3) cout << s[i] << s[i];
          cnt = 1;
        }
      }
    } else {
      if (cnt == 1) cout << s[i - 1];
      else if (cnt == 3) cout << s[i - 1] << s[i - 1];
      cnt = 0;
      cout << s[i];
    }
  }
  cout << endl;
  return 0;
}
