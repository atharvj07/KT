import java.util.*;
import java.io.*;

public class NikolayAndStaircases {
    static int n;
    static String s;
    static boolean[][] visited;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        while (t-- > 0) {
            n = Integer.parseInt(br.readLine());
            s = br.readLine();
            visited = new boolean[2][n + 1];
            int ans = 0;
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= 1; j++) {
                    if (!visited[j][i]) {
                        ans = Math.max(ans, dfs(j, i));
                    }
                }
            }
            System.out.println(ans);
        }
    }

    static int dfs(int floor, int room) {
        visited[floor][room] = true;
        int ans = 1;
        if (room > 1 && !visited[floor][room - 1]) {
            ans += dfs(floor, room - 1);
        }
        if (room < n && !visited[floor][room + 1]) {
            ans += dfs(floor, room + 1);
        }
        if (s.charAt(room - 1) == '1' && !visited[floor ^ 1][room]) {
            ans += dfs(floor ^ 1, room);
        }
        return ans;
    }
}
