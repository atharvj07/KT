n = int(input().strip())
a = list(map(int, input().strip().split()))
m = int(input().strip())
q = list(map(int, input().strip().split()))

a.sort(reverse=True)
prefix_sum = [0] * (n + 1)
for i in range(n):
    prefix_sum[i + 1] = prefix_sum[i] + a[i]

for qi in q:
    print(prefix_sum[n] - (prefix_sum[qi] - prefix_sum[qi - 1]))
