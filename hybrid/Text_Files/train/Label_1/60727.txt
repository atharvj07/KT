from functools import reduce

N, M = (int(x) for x in input().split())
A = [int(x) for x in input().split()]
B = [int(x) for x in input().split()]

F = [0] * (M + 1)
F[0] = 1
for i in range(M):
    for j in range(N - 1, -1, -1):
        if A[j] >= B[i]:
            F[i] += F[i-1]
    F[i] %= 998244353

print(F[M - 1])
