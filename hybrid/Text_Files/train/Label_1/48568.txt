import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n];
    boolean[] isSq = new boolean[n];
    int ans = 0;
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
      double sq = Math.sqrt(a[i]);
      isSq[i] = (sq == (int) sq);
    }
    int sqCnt = 0;
    for (int i = 0; i < n; i++) {
      if (isSq[i]) {
        sqCnt++;
      }
    }
    int rem = n / 2 - sqCnt;
    for (int i = 0; i < n; i++) {
      if (!isSq[i] && rem > 0) {
        ans += a[i] % 2;
        rem--;
      } else {
        ans += a[i] % 2 + a[i] / 2;
      }
    }
    System.out.println(ans);
  }
}
