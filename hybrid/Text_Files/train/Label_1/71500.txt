# Implementation of the solution in Python

import math

def is_lovely(x):
    for i in range(2, int(math.sqrt(x)) + 1):
        if x % (i**2) == 0:
            return False
    return True

n = int(input())
lovely = 1
for i in range(2, int(math.sqrt(n)) + 1):
    if n % i == 0 and is_lovely(i):
        lovely = i
print(lovely)
