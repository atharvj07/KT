n = int(input())
p = list(map(int, input().split()))

positions = [0] * (n + 1)
for i, v in enumerate(p):
    positions[v] = i

longest = 0
current_length = 1

for i in range(2, n + 1):
    if positions[i] > positions[i - 1]:
        current_length += 1
    else:
        longest = max(longest, current_length)
        current_length = 1

longest = max(longest, current_length)
print(n - longest)
