import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }
        int d = 0;
        int positiveCount = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] > 0) {
                positiveCount++;
            }
        }
        if (positiveCount < n / 2) {
            System.out.println(0);
        } else {
            for (d = -1000; d <= 1000; d++) {
                int tempPositiveCount = 0;
                for (int i = 0; i < n; i++) {
                    if ((double) arr[i] / d > 0) {
                        tempPositiveCount++;
                    }
                }
                if (tempPositiveCount >= n / 2) {
                    break;
                }
            }
            System.out.println(d);
        }
    }
}
