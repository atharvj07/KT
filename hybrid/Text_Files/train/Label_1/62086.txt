n, q = map(int, input().split())
a = list(map(int, input().split()))
ans = []
l = [0] * (10**6 + 7)
r = [0] * (10**6 + 7)
for i in range(n):
    l[a[i]] = i
    r[a[i]] = i
for i in range(q):
    x, y, z = map(int, input().split())
    if l[z] < x or y < r[z]:
        ans.append(l[z] if l[z] >= x and l[z] <= y else r[z])
    else:
        ans.append(-1)
print(*ans, sep='\n')
