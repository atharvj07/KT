import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), p = sc.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        int[] ans = new int[n];
        int time = 0;
        for (int i = 0; i < n; i++) {
            if (i > 0 && t[i] <= time) {
                time += p;
            }
            ans[i] = time + p;
            time = Math.max(time, t[i]);
        }
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
