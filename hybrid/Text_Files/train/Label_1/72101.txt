for _ in range(int(input())):
    n = int(input())
    is_valid = True
    for i in range(n):
        p, c = map(int, input().split())
        if p < i or c > p or (i > 0 and p < last_p) or (i > 0 and c < last_c):
            is_valid = False
            break
        last_p = p
        last_c = c
    print("YES" if is_valid else "NO")
