n = int(input())
stack = []
placement = []
inconsistent = False

for _ in range(2*n):
    event = input().split()
    if event[0] == '+':
        stack.append('+')
    else:
        price = int(event[1])
        if not stack or stack[-1] != '+':
            inconsistent = True
            break
        stack.pop()
        placement.append(price)
        for i in range(len(stack)-1, -1, -1):
            if stack[i] == '+':
                stack[i] = price
                break
        else:
            stack.append(price)

if inconsistent or sorted(placement, reverse=True) != placement:
    print("NO")
else:
    print("YES")
    print(" ".join(map(str, reversed(placement))))
