#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    int k;
    cin >> k;
    vector<int> a(k), b(n - k);
    for (int i = 0; i < k; i++) { cin >> a[i]; }
    for (int i = 0; i < n - k; i++) { cin >> b[i]; }
    int x = 0, y = 0, ans = 0;
    while (x < k && y < n - k) {
        ans++;
        if (a[x] > b[y]) {
            y++;
            a.push_back(b[y - 1]);
            b.erase(b.begin() + y - 1);
        } else {
            x++;
            b.push_back(a[x - 1]);
            a.erase(a.begin() + x - 1);
        }
    }
    if (x == k) {
        cout << -1 << endl;
    } else {
        cout << ans << " " << 2 << endl;
    }
}
