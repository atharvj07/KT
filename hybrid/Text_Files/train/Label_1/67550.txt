n,a,b=map(int,input().split())
arr=[]
for i in range(n):
    x,y=map(int,input().split())
    arr.append([x,y])
    arr.append([y,x])
ans=0
for i in range(n*2):
    for j in range(n*2):
        if i==j:continue
        x1,y1=arr[i]
        x2,y2=arr[j]
        if x1<=a and y1+y2<=b:ans=max(ans,x1*y1+x2*y2)
        elif x2<=a and y1+y2<=b:ans=max(ans,x1*y1+x2*y2)
        elif x1<=b and y1+y2<=a:ans=max(ans,x1*y1+x2*y2)
        elif x2<=b and y1+y2<=a:ans=max(ans,x1*y1+x2*y2)
print(ans)
