#include<bits/stdc++.h>
using namespace std;
int c=0,sum;
void f1(string s,string p)
{
    sum=0;
    int sz=p.size();
    for(int i=0; i<sz; i++)
    {
        if(s[i]!=p[i]) sum++;
    }
    if(sum%2==0) c++;

}
int main()
{
    string a,b;
    cin>>a>>b;
    int sb=b.size();
    for(int i=0; i<=a.size()-sb; i++)
    {
        string t=a.substr(i,sb);
        f1(t,b);
    }
    cout<<c<<endl;
    return 0;
}
