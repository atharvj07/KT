import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        boolean[][] canDivide = new boolean[n][n];
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;
            canDivide[x][y] = canDivide[y][x] = true;
        }
        int[] divisors = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = 2; j <= Math.sqrt(a[i]); j++) {
                if (a[i] % j == 0) {
                    divisors[i]++;
                    while (a[i] % j == 0) {
                        a[i] /= j;
                    }
                }
            }
            if (a[i] > 1) {
                divisors[i]++;
            }
        }
        int answer = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (canDivide[i][j]) {
                    answer += Math.min(divisors[i], divisors[j]);
                }
            }
        }
        System.out.println(answer);
    }
}
