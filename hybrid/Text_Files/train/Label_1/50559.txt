import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int n = s.length(), ans = n;
        for (int i = 1; i <= n; i++) {
            StringBuilder t = new StringBuilder();
            int cnt = 0, pre = 0, j = 0;
            while (j < n) {
                int st = j;
                while (j < n && s.charAt(j) == s.charAt(st)) j++;
                int cur = j - st;
                if (cnt == 0 || cur != pre) {
                    if (cnt > 0) t.append(cnt).append(s.charAt(pre - 1));
                    cnt = cur;
                    pre = cur;
                } else {
                    cnt += cur;
                }
            }
            if (cnt > 0) t.append(cnt).append(s.charAt(pre - 1));
            ans = Math.min(ans, t.length());
        }
        System.out.println(ans);
    }
}
