import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] x = new int[n];
        int[] t = new int[n];
        int[] p = new int[m];
        int[] b = new int[m];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            t[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            p[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        Arrays.sort(p);
        int[] ans = new int[n];
        for (int i = 0; i < m; i++) {
            int j = Arrays.binarySearch(x, p[i]);
            if (j < 0) {
                j = -j - 1;
            }
            if (j > 0 && x[j - 1] == p[i]) {
                j--;
            }
            if (j < n && t[j] >= p[i] - x[j]) {
                ans[j]++;
                t[j] += b[i];
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.println(ans[i] + " " + t[i]);
        }
    }
}
