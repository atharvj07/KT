#include<bits/stdc++.h>
using namespace std;
int n,m,k,a[55],mn=1e9,sum;
int main(){
    cin>>n>>m>>k;
    for(int i=1;i<=n;i++)cin>>a[i],sum+=a[i];
    if(sum<m)return cout<<-1,0;
    for(int i=1;i<=n;i++){
        int x;
        if(m<=k)x=0;
        else{
            x=m-k;
            x=(x+a[i]-1)/a[i];
        }
        mn=min(mn,x+1);
    }
    cout<<mn;
    return 0;
}
