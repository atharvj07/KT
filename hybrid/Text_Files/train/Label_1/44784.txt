import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Jaber {

    static class City {
        int row;
        int col;
        int color;

        public City(int row, int col, int color) {
            this.row = row;
            this.col = col;
            this.color = color;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] grid = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = scanner.nextInt();
            }
        }

        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int r1 = scanner.nextInt() - 1;
            int c1 = scanner.nextInt() - 1;
            int r2 = scanner.nextInt() - 1;
            int c2 = scanner.nextInt() - 1;

            int time = bfs(grid, n, m, r1, c1, r2, c2);
            System.out.println(time);
        }
    }

    static int bfs(int[][] grid, int n, int m, int r1, int c1, int r2, int c2) {
        boolean[][] visited = new boolean[n][m];
        Queue<City> queue = new LinkedList<>();

        queue.offer(new City(r1, c1, grid[r1][c1]));
        visited[r1][c1] = true;

        int time = 0;
        while (!queue.isEmpty()) {
            int size = queue.size();
            for (int i = 0; i < size; i++) {
                City current = queue.poll();

                if (current.row == r2 && current.col == c2) {
                    return time;
                }

                for (int dx = -1; dx <= 1; dx++) {
                    for (int dy = -1; dy <= 1; dy++) {
                        int newRow = current.row + dx;
                        int newCol = current.col + dy;

                        if (isValid(newRow, newCol, n, m) && !visited[newRow][newCol] &&
                                (Math.abs(dx) + Math.abs(dy) == 1 || grid[newRow][newCol] == current.color)) {
                            queue.offer(new City(newRow, newCol, grid[newRow][newCol]));
                            visited[newRow][newCol] = true;
                        }
                    }
                }
            }
            time++;
        }

        return -1;
    }

    static boolean isValid(int row, int col, int n, int m) {
        return row >= 0 && row < n && col >= 0 && col < m;
    }
}
