#include<iostream>
#define ll long long 
using namespace std;
int main(){
    ll q,k;
    cin>>q;
    while(q--){
        cin>>k;
        ll ans=0,x=0,y=0;
        while(1){
            x=1LL*(ans*(ans+1))/2;
            y=1LL*((ans+1)*(ans+2))/2;
            if(k<=y){
                k=k-x;
                break;
            }
            ans++;
        }
        cout<<(k+ans)%10<<endl;
    } 
    return 0;
}
