import sys
import collections
n = int(input())
points = []
for i in range(n):
    x, c = sys.stdin.readline().split()
    x = int(x)
    if c == 'R': points.append((x, 'R'))
    elif c == 'G': points.append((x, 'G'))
    elif c == 'B': points.append((x, 'B'))

points.sort()

size = 0
ans = 0
for p in points:
    if size == 0: size, ans = 1, 0
    elif p[1] == points[size - 1][1]: continue
    else: size, ans = size + 1, ans + p[0] - points[size - 1][0]
print(ans)
