def min_time(n, m, volcanoes):
    # No direct way to compute the path due to the size constraints and presence of volcanoes.
    # Use dynamic programming for small n or check if there's a blocking row or column for large n.

    if n > 500:  # Threshold for switching strategy based on constraints
        # For large n, checking for a complete blocking row or column is more efficient.
        blocked_rows = set()
        blocked_cols = set()
        for x, y in volcanoes:
            blocked_rows.add(x)
            blocked_cols.add(y)
        if len(blocked_rows) == n or len(blocked_cols) == n:
            return -1
        else:
            return 2*n - 2
    else:
        # Use dynamic programming for smaller n.
        dp = [[float('inf')] * n for _ in range(n)]
        dp[0][0] = 0
        vol_set = {(x-1, y-1) for x, y in volcanoes}
        for i in range(n):
            for j in range(n):
                if (i, j) in vol_set:
                    continue
                if i > 0:
                    dp[i][j] = min(dp[i][j], dp[i-1][j] + 1)
                if j > 0:
                    dp[i][j] = min(dp[i][j], dp[i][j-1] + 1)
        return dp[-1][-1] if dp[-1][-1] != float('inf') else -1

n, m = map(int, input().split())
volcanoes = [tuple(map(int, input().split())) for _ in range(m)]
print(min_time(n, m, volcanoes))
