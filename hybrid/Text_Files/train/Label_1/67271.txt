n = int(input())
friends = [list(map(int, input().split())) for _ in range(n)]

total_width = sum(w for w, h in friends)
max_heights = sorted([h for w, h in friends], reverse=True)

for i in range(n):
    width = total_width - friends[i][0]
    height = max_heights[1] if max_heights[0] == friends[i][1] else max_heights[0]
    print(width * height, end=' ')
