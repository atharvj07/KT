#include<bits/stdc++.h>
using namespace std;
int a[100005];
int main()
{
    int n;
    cin>>n;
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
    }
    int ans=2;
    for(int i=0;i<n;i++)
    {
        if(a[i]!=1)
        {
            ans=max(ans,a[i]);
        }
    }
    for(int i=0;i<n;i++)
    {
        if(a[i]==1)
        {
            cout<<ans<<" ";
            ans++;
        }
        else
        {
            cout<<a[i]<<" ";
        }
    }
}
