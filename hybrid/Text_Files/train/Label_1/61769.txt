import sys
input = sys.stdin.readline

mod = 10**9 + 7
n, T = map(int, input().split())
songs = [(t, g) for t, g in zip(*[map(int, input().split()) for _ in range(n)])]

dp = [[0] * 4 for _ in range(T+1)]
dp[0][0] = 1
for t in range(1, T+1):
  for last in range(4):
    for song in songs:
      if t >= song[0] and last != song[1]:
        dp[t][last] += dp[t - song[0]][song[1]]
        dp[t][last] %= mod
print(dp[T][0])
