n=int(input())
a=list(map(int, input().split()))
l=[0 for i in range(n)]
for i in range(n):
    for j in range(i):
        if a[j]<=a[i]:
            l[i]=max(l[i], l[j])
    l[i]+=1
m=[0 for i in range(n)]
for i in range(n-1,-1,-1):
    for j in range(n-1, i, -1):
        if a[j]>=a[i]:
            m[i]=max(m[i], m[j])
    m[i]+=1
for i in range(n):
    if l[i]==m[i] and l[i]==max(l):
        print(3, end='')
    elif l[i]==m[i]:
        print(2, end='')
    else:
        print(1, end='')
