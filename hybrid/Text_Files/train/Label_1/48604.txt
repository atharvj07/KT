import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }

        int ans = 0;
        for (int i = 0; i < m; i++) {
            int j = i % n;
            if (i % 2 == 0) {
                ans += a[j];
            } else {
                ans += b[j];
            }
        }

        System.out.println(ans);
    }
}
