#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int k = 0;
    for (int i = 0; i < n - 1; i++) {
      if (a[i] > a[i + 1]) { k++; }
    }
    cout << k << endl;
  }
  return 0;
}
