import java.util.Scanner;

public class KitchenTiling {
    private static final int MOD = 998244353;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int w = scanner.nextInt();
        int h = scanner.nextInt();

        // Precompute factorials and inverses
        long[] factorial = new long[w + h];
        long[] inverse = new long[w + h];
        factorial[0] = 1;
        inverse[0] = 1;
        for (int i = 1; i < w + h; i++) {
            factorial[i] = (i * factorial[i - 1]) % MOD;
            inverse[i] = modInverse(factorial[i], MOD);
        }

        long result = 0;
        // Iterate over possible orientations of the first row and the first column
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                // Calculate the number of valid tilings with a specific orientation
                long count = calculateTilingCount(w, h, i, j, factorial, inverse);
                result = (result + count) % MOD;
            }
        }

        System.out.println(result);
    }

    // Calculate the number of valid tilings with a specific orientation
    private static long calculateTilingCount(int w, int h, int firstRowOrientation, int firstColumnOrientation, long[] factorial, long[] inverse) {
        long count = 1;
        for (int i = 1; i < w; i++) {
            count = (count * 2) % MOD; // Two possible orientations for each tile
        }
        for (int i = 1; i < h; i++) {
            count = (count * 2) % MOD;
        }

        // Adjust for restrictions due to orientation
        if (firstRowOrientation == firstColumnOrientation) {
            count = (count * inverse[w + h - 2]) % MOD;
        } else {
            count = (count * inverse[w + h - 1]) % MOD;
        }

        return count;
    }

    // Calculate the modular inverse of a number
    private static long modInverse(long a, long m) {
        long m0 = m;
        long y = 0, x = 1;

        if (m == 1) {
            return 0;
        }

        while (a > 1) {
            // q is quotient
            long q = a / m;
            long t = m;

            // m is remainder now, process same as Euclid's algo
            m = a % m;
            a = t;
            t = y;

            // Update y and x
            y = x - q * y;
            x = t;
        }

        // Make x positive
        if (x < 0) {
            x += m0;
        }

        return x;
    }
}
