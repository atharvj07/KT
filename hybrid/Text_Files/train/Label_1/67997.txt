def solve():
    t = int(input())
    for _ in range(t):
        s = input()
        ans = 0
        i = 0
        while i < len(s):
            if s[i] == "1":
                cnt = 1
                while i + 1 < len(s) and s[i + 1] == s[i]:
                    i += 1
                    cnt += 1
                if cnt % 2 == 1:
                    ans += cnt // 2 + 1
            i += 1
        print(ans)

solve()
