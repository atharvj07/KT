#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  if (n > k * (k - 1))
    return cout << "NO" << endl, 0;
  cout << "YES" << endl;
  vector<bool> taken(k + 1);
  for (int i = 0, j = 1; i < n; i++, j = -j) {
    int a, b;
    a = b = 1;
    for (int x = 1; x <= k; x++) {
      if (taken[x]) continue;
      if (j > 0) {
        a = x;
        if (j == 1 && i - 2 >= 0 && b == taken[b]) a++;
      } else {
        b = x;
        if (i - 1 >= 0 && a == taken[a]) b++;
      }
      if (a != b) break;
    }
    taken[a] = taken[b] = 1;
    cout << a << ' ' << b << "\n";
  }
}
