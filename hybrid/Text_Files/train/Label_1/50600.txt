import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) throws IOException{
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] nk = br.readLine().split(" ");
    int numInstrument = Integer.parseInt(nk[0]);
    int numDays = Integer.parseInt(nk[1]);
    String[] days = br.readLine().split(" ");
    int[] dayArrays = Arrays.stream(days).mapToInt(Integer::parseInt).toArray();

    Arrays.sort(dayArrays);
    int numCanBeLearned = 0;
    for(int d : dayArrays){
      if(numDays >= d){
        numCanBeLearned++;
        numDays -= d;
      } else {
        break;
      }
    }

    System.out.println(numCanBeLearned);
    if(numCanBeLearned != 0){
      for(int i = 0; i < numCanBeLearned; i++){
        System.out.print((i+1) + " ");
      }
    }
  }
}
