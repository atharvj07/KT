import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] piles = new int[n][];
        for (int i = 0; i < n; i++) {
            int size = scanner.nextInt();
            piles[i] = new int[size];
            for (int j = 0; j < size; j++) {
                piles[i][j] = scanner.nextInt();
            }
        }
        for (int[] pile : piles) {
            Arrays.sort(pile);
        }
        int CielSum = 0, JiroSum = 0;
        boolean CielTurn = true;
        while (true) {
            int bestPile = -1;
            int bestCard = -1;
            for (int i = 0; i < n; i++) {
                if (piles[i].length > 0) {
                    if (CielTurn) {
                        if (bestCard < piles[i][0]) {
                            bestCard = piles[i][0];
                            bestPile = i;
                        }
                    } else {
                        if (bestCard < piles[i][piles[i].length - 1]) {
                            bestCard = piles[i][piles[i].length - 1];
                            bestPile = i;
                        }
                    }
                }
            }
            if (bestPile == -1) {
                break;
            }
            if (CielTurn) {
                CielSum += bestCard;
                piles[bestPile] = Arrays.copyOfRange(piles[bestPile], 1, piles[bestPile].length);
            } else {
                JiroSum += bestCard;
                piles[bestPile] = Arrays.copyOfRange(piles[bestPile], 0, piles[bestPile].length - 1);
            }
            CielTurn = !CielTurn;
        }
        System.out.println(CielSum + " " + JiroSum);
    }
}
