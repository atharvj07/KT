#include <iostream>
#include <string>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    int i = 0, j = n - 1;
    bool flag = true;
    while (i <= j) {
      if (s[i] != 'a' + i || s[j] != 'a' + n - 1 - i) {
        flag = false;
      }
      i++, j--;
    }
    if (flag) { cout << "YES" << endl; } else { cout << "NO" << endl; }
  }
  return 0;
}
