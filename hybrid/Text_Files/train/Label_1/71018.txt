MOD = 10**9 + 7

def nCr_mod_p(n, r, p):
    if r > n:
        return 0
    if r == 0 or r == n:
        return 1
    num = den = 1
    for k in range(r):
        num = (num * (n - k)) % p
        den = (den * (k + 1)) % p
    return num * pow(den, p - 2, p) % p

def solve(n, m, k, a):
    a.sort()
    ans = 0
    for i in range(n):
        low, high = i, n - 1
        while low < high:
            mid = (low + high + 1) // 2
            if a[mid] - a[i] <= k:
                low = mid
            else:
                high = mid - 1
        ans += nCr_mod_p(low - i, m - 1, MOD)
        ans %= MOD
    return ans

t = int(input().strip())
for _ in range(t):
    n, m, k = map(int, input().strip().split())
    a = list(map(int, input().strip().split()))
    print(solve(n, m, k, a))
