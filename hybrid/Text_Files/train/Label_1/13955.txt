#include <iostream>
#include <vector>
#include <set>
using namespace std;

int n, k;

int main() {
    cin >> n >> k;

    vector<int> ans;

    set<int> cur_turned, nxt_turned;
    for (int i = 1; i <= n; i++) {
        cur_turned.insert(i);
    }

    while (cur_turned.size()) {
        ans.push_back(*cur_turned.begin());

        for (int i = *cur_turned.begin() - 1; i >= max(1, *cur_turned.begin() - k); i--) {
            if (cur_turned.find(i) != cur_turned.end()) {
                nxt_turned.insert(i);
            }
        }

        for (int i = *cur_turned.begin() + 1; i <= min(n, *cur_turned.begin() + k); i++) {
            if (cur_turned.find(i) != cur_turned.end()) {
                nxt_turned.insert(i);
            }
        }

        cur_turned.clear();
        cur_turned.swap(nxt_turned);
    }

    cout << ans.size() << "\n";
    for (int el : ans) {
        cout << el << " ";
    }

    return 0;
}
