for _ in range(int(input())):
    n, s, k = map(int, input().split())
    a = list(map(int, input().split()))
    l, r = s, s
    ans = 10**9
    for x in a:
        if x < s: l = min(l, x)
        if x > s: r = min(r, x)
    if l != s: ans = min(ans, s - l)
    if r != s: ans = min(ans, r - s)
    print(ans)
