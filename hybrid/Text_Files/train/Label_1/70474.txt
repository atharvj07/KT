import sys

n,k = list(map(int,input().split()))
a = [list(map(int,input().split())) for i in range(4)]

ok = True
for j in range(n):
    if a[0][j] != a[3][j] or a[1][j] != a[2][j]:
        ok = False
if not ok:
    sys.stdout.write('-1')
    sys.exit()

car = {}
for i in range(1,n+1):
    car[i] = []
    for j in range(4):
        for k in range(n):
            if a[j][k] == i:
                car[i].append((j,k))

ans = []

def bfs(u, v, car, dist):
    q = [(u,v,dist)]
    while q:
        u,v,dist = q.pop(0)
        if u == 3 and v == car[u-1]:
            return dist
        if u == 0 and v == car[u+1]:
            return dist
        for dx, dy in [(1,0),(-1,0),(0,1),(0,-1)]:
            x = u + dx
            y = v + dy
            if 0 <= x < 4 and 0 <= y < n and a[x][y] == 0:
                q.append((x,y,dist+1))
    return 1e9

def check(u, v):
    for dx, dy in [(1,0),(-1,0),(0,1),(0,-1)]:
        x = u + dx
        y = v + dy
        if 0 <= x < 4 and 0 <= y < n and a[x][y] == 0:
            return True
    return False

def move(i):
    u,v = car[i][0]
    if check(u,v):
        ans.append((i,u+1,v+1))
        a[u][v] = 0
        a[u+1][v] = i
        car[i][0] = (u+1,v)
        return
    d = bfs(u,v,car,0)
    if d > 20000:
        sys.stdout.write('-1')
        sys.exit()
    u,v = car[i][1]
    for j in range(d):
        ans.append((i,u+1,v+1))
        a[u][v] = 0
        a[u+1][v] = i
        car[i][1] = (u+1,v)
        u,v = car[i][1]

for i in range(1,n+1):
    while len(car[i]) > 1:
        move(i)
    u,v = car[i][0]
    if u == 3:
        d = bfs(3,v,car,0)
    else:
        d = bfs(0,v,car,0)
    if d > 20000:
        sys.stdout.write('-1')
        sys.exit()
    for j in range(d):
        ans.append((i,u+1,v+1))
        a[u][v] = 0
        if u == 3:
            u = 2
        else:
            u = 1
        a[u][v] = i
        car[i][0] = (u,v)

sys.stdout.write('{:}\n'.format(len(ans)))
for p, q, r in ans:
    sys.stdout.write('{: {: {:}}}\n'.format(p,q,r))
