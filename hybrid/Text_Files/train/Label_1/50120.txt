import java.util.Scanner;

public class CaracalVsMonster {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long H = scanner.nextLong();
        scanner.close();

        long attacks = 0;
        while (H > 0) {
            attacks++;
            long x = H;
            while (x > 1) {
                H -= Math.floor(x / 2);
                x /= 2;
                attacks++;
            }
            H -= 1;
        }

        System.out.println(attacks);
    }
}
