#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n), b(m);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < m; i++) cin >> b[i];
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    int ans = 0;
    for (int i = 0, j = 0; i < n && j < m; i++) {
        while (j < m && (a[i] + b[j]) % 2 == 0) j++;
        if (j < m) {
            ans++;
            j++;
        }
    }
    cout << ans << endl;
    return 0;
}
