#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;
const ll MAXN = 1e5+10;
const ll INF = 1e18;
ll l[MAXN], n;
char t[MAXN];
ll dp[MAXN][2][2]; // dp[i][swim][fly] = shortest time to get to the left of segment i, while currently having (not having) the ability to swim and fly (corresponding to swim==0 and swim==1)
vector<pair<int, int>> adj[MAXN]; // adj[i] = [segment length, segment terrain type]

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> l[i];
    }
    cin >> t;
    for (int i = 1; i <= n; i++) {
        int x;
        if (t[i-1] == 'G') x = 0;
        else if (t[i-1] == 'W') x = 1;
        else x = 2;
        adj[i] = { {l[i], x} };
    }

    for (int i = 0; i <= n; i++) {
        dp[i][0][0] = dp[i][0][1] = dp[i][1][0] = dp[i][1][1] = INF;
    }
    dp[0][0][0] = dp[0][0][1] = dp[0][1][0] = 0;
    for (int i = 1; i <= n; i++) {
        ll segLength = adj[i][0].first, segType = adj[i][0].second;
        for (int swim = 0; swim < 2; swim++) {
            for (int fly = 0; fly < 2; fly++) {
                if (segType == 0 || swim == 1) {
                    dp[i][swim][fly] = min(dp[i][swim][fly], (fly ? 5 : 0) + dp[i-1][swim][fly]);
                }
                if (segType == 1 || fly == 1) {
                    dp[i][swim][fly] = min(dp[i][swim][fly], (fly ? 3 : 0) + dp[i-1][swim][fly]);
                }
                if (segType == 2) {
                    bool canFly = (fly == 1 && i > 1 && dp[i-1][swim][fly] - dp[i-2][swim][fly] >= segLength);
                    if (swim == 1) {
                        dp[i][swim][fly] = min(dp[i][swim][fly], 3 * segLength + dp[i-1][swim][!canFly]);
                    }
                    if (fly == 1) {
                        dp[i][swim][fly] = min(dp[i][swim][fly], 3 * segLength + dp[i-1][!swim][!canFly]);
                    }
                }
            }
        }
    }

    cout << dp[n][1][1] << '\n';
}
