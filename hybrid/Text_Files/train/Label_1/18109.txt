#include <bits/stdc++.h>
using namespace std;

int n, a[100001], lst[100001];
stack<int> st;

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = n; i >= 1; i--) {
    if (st.empty() || st.top() < a[i]) {
      st.push(a[i]);
      lst[a[i]] = i;
    }
  }
  for (int i = 1; i <= n; i++) {
    while (!st.empty() && lst[st.top()] < i) st.pop();
    if (st.empty())
      cout << endl;
    else
      cout << st.top() << endl;
  }
}
