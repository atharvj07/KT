import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), x = sc.nextInt();
        int[] a = new int[n];
        int[] cnt = new int[106];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            cnt[a[i]]++;
        }
        sc.close();
        Arrays.sort(a);
        int sum = n / 2, mn = n;
        for (int i = 105; i >= 1 && sum < n / 2; i--) {
            int t = cnt[i] - Math.min(cnt[i], sum - (n - cnt[i]) / 2);
            mn = Math.min(mn, (x <= i ? t : t + 1));
        }
        System.out.println(mn);
    }
}
