y1, m1, d1 = map(int, input().split(':'))
y2, m2, d2 = map(int, input().split(':'))
days = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
is_leap = lambda x: not (x % 400 or (not x % 100 and x % 4))
days_y1 = sum(days[:m1-1]) + d1 - (1 if is_leap(y1) and m1 <= 3 else 0)
days_y2 = sum(days[:m2-1]) + d2 - (1 if is_leap(y2) and m2 <= 3 else 0)
print(365 * (y2 - y1 - (is_leap(y1) and m1 <= 3)) + (is_leap(y2) and m2 <= 3) + days_y2 - days_y1)
