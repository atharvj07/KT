n,k=map(int,input().split())
a=list(map(int,input().split()))
l=[[-1 for i in range (k+1)] for j in range (n+1)]
r=[[-1 for i in range (k+1)] for j in range (n+1)]
L=[0]*(n+1)
R=[0]*(n+1)
mod=998244353
for i in range (n-1,-1,-1):
    if a[i]==-1:
        for j in range (1,k+1):
            l[i][j]=i
            r[i][j]=i
            if L[i+1]!=-1:
                l[i][j]=min(l[i][j],L[i+1])
            if R[i+1]!=-1:
                r[i][j]=max(r[i][j],R[i+1])
        L[i]=-1
        R[i]=-1
    else:
        L[i]=i
        R[i]=i
        for j in range (1,k+1):
            if l[i+1][j]==-1 and r[i+1][j]==-1:
                l[i][j]=-1
                r[i][j]=-1
            elif l[i+1][j]!=-1 and r[i+1][j]!=-1:
                if a[i]==a[l[i+1][j]]:
                    l[i][j]=l[i+1][j]
                if a[i]==a[r[i+1][j]]:
                    r[i][j]=r[i+1][j]
            else:
                assert False
for j in range (1,k+1):
    if l[0][j]!=-1 and r[0][j]!=-1 and l[0][j]<r[0][j]:
        for i in range (l[0][j],r[0][j]+1):
            if a[i]!=a[l[0][j]]:
                assert False
print(pow(k,sum(1 for x in a if x == -1),mod))
