#include <iostream>

int main() {
  int q;
  std::cin >> q;
  while (q--) {
    int l1, r1, l2, r2;
    std::cin >> l1 >> r1 >> l2 >> r2;
    if (l1 == r1) std::cout << l1 << ' ' << l2 << '\n';
    else if (l2 == r2) std::cout << l1 << ' ' << l2 << '\n';
    else std::cout << l1 << ' ' << r2 << '\n';
  }
}
