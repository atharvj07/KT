#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int maxN = 100001, inf = 0x3f3f3f3f;
int n, m, vis[maxN], r, c, x, y, ans;
char mp[maxN];
struct node
{
    int x, y, r, l;
    node(int x_, int y_, int r_, int l_)
    {
        x = x_;
        y = y_;
        r = r_;
        l = l_;
    }
};
queue<node> q;
int d[4][2] = {{-1, 0}, {0, 1}, {1, 0}, {0, -1}};
void bfs()
{
    memset(vis, 0, sizeof(vis));
    ans = 1;
    q.push(node(r, c, 0, 0));
    vis[r * m + c] = 1;
    while(!q.empty())
    {
        node now = q.front();
        q.pop();
        for(int k = 0; k < 4; k++)
        {
            int dx = now.x + d[k][0];
            int dy = now.y + d[k][1];
            if(dx >= 1 && dx <= n && dy >= 1 && dy <= m)
            {
                if(mp[(dx - 1) * m + dy] == '*')
                    continue;
                if(k == 1 && now.r == x)
                    continue;
                if(k == 3 && now.l == y)
                    continue;
                if(!vis[dx * m + dy])
                {
                    ans++;
                    q.push(node(dx, dy, k == 0 ? now.r + 1 : now.r, k == 3 ? now.l + 1 : now.l));
                    vis[dx * m + dy] = 1;
                }
            }
        }
    }
}
int main()
{
    cin >> n >> m;
    cin >> r >> c;
    cin >> x >> y;
    for(int i = 1; i <= n; i++)
        for(int j = 1; j <= m; j++)
            cin >> mp[(i - 1) * m + j];
    bfs();
    cout << ans << endl;
    return 0;
}
