#include <bits/stdc++.h>
using namespace std;

const int inf = 1e9;
const int dr[] = {0, 0, 1, -1};
const int dc[] = {1, -1, 0, 0};

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<string> grid(n);
  vector<vector<int>> spiders(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    cin >> grid[i];
  }

  // preprocess the position of spiders
  for (int i = n - 2; i >= 0; i--) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] != '.') {
        int dir = grid[i][j] == 'L' ? 0 : grid[i][j] == 'R' ? 1 : grid[i][j] == 'U' ? 3 : 2;
        spiders[i][j] = spiders[i + dr[dir]][j + dc[dir]] + 1;
      }
    }
  }

  // find the minimum number of spiders seen from each cell in the first row
  vector<int> ans(m, inf);
  for (int j = 0; j < m; j++) {
    ans[j] = spiders[0][j];
    for (int i = 1; i < n - 1; i++) {
      for (int dir = 0; dir < 4; dir++) {
        if (j + dc[dir] >= 0 && j + dc[dir] < m) {
          ans[j] = min(ans[j], spiders[i][j + dc[dir]] + 1);
        }
      }
    }
  }

  for (int j = 0; j < m; j++) {
    cout << ans[j] << ' ';
  }
  cout << '\n';

  return 0;
}
