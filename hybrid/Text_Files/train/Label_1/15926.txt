#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
      cin >> v[i].first >> v[i].second;
    }
    sort(v.begin(), v.end());
    int ans = 0, l = -1, r = -1;
    for (auto &i : v) {
      if (i.first <= l || i.second <= r) ans++;
      l = max(l, i.first);
      r = max(r, i.second);
    }
    cout << ans << endl;
  }
  return 0;
}
