#include <bits/stdc++.h>
using namespace std;

int main()
{
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    if (c > 1 || d > 1)
        cout << "NO" << endl;
    else
    {
        if (a == b + 1 && c == d && b >= c)
            cout << "YES" << endl;
        else if (b == a + 1 && d == c && a >= d)
            cout << "YES" << endl;
        else if (a == c + 1 && b == d && a >= b)
            cout << "YES" << endl;
        else if (b == c + 1 && a == d && c >= a)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;

        if (cout << "YES")
        {
            for (int i = 1; i <= a; i++)
                cout << 0 << " ";
            for (int i = 1; i <= b; i++)
                cout << 1 << " ";
            for (int i = 1; i <= c; i++)
                cout << 2 << " ";
            for (int i = 1; i <= d; i++)
                cout << 3 << " ";
            cout << endl;
        }
    }
    return 0;
}
