import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = scanner.nextInt();
        }
        int k = 0;
        int max = h[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            if (h[i] > max) {
                k++;
            }
            max = Math.max(max, h[i]);
        }
        System.out.println(k);
    }
}
