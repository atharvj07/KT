#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<long long> q(n);
  long long sum = 0;
  for (int i = 0; i < n; ++i) {
    std::cin >> q[i];
    sum += q[i];
  }
  long long p = 1;
  int ans = 1;
  while (p < sum) {
    p += q[ans - 1];
    ++ans;
  }
  std::cout << ans << '\n';
  return 0;
}
