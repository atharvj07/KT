import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

class Edge implements Comparable<Edge> {
    int from, to, weight;

    public Edge(int from, int to, int weight) {
        this.from = from;
        this.to = to;
        this.weight = weight;
    }

    @Override
    public int compareTo(Edge other) {
        return Integer.compare(this.weight, other.weight);
    }
}

class Graph {
    int n;
    List<Edge>[] edges;

    public Graph(int n) {
        this.n = n;
        edges = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            edges[i] = new ArrayList<>();
        }
    }

    public void addEdge(int from, int to, int weight) {
        from %= n;
        to %= n;
        edges[from].add(new Edge(from, to, weight));
        edges[to].add(new Edge(to, from, weight));
    }

    public long kruskal() {
        long totalWeight = 0;
        int[] parent = new int[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
        }

        Edge[] allEdges = new Edge[0];
        for (List<Edge> edgeList : edges) {
            allEdges = mergeArrays(allEdges, edgeList.toArray(new Edge[0]));
        }
        Arrays.sort(allEdges);

        for (Edge edge : allEdges) {
            int fromRoot = findRoot(parent, edge.from);
            int toRoot = findRoot(parent, edge.to);
            if (fromRoot != toRoot) {
                parent[fromRoot] = toRoot;
                totalWeight += edge.weight;
            }
        }

        return totalWeight;
    }

    private int findRoot(int[] parent, int node) {
        if (parent[node] == node) {
            return node;
        } else {
            return findRoot(parent, parent[node]);
        }
    }

    private Edge[] mergeArrays(Edge[] arr1, Edge[] arr2) {
        Edge[] merged = new Edge[arr1.length + arr2.length];
        System.arraycopy(arr1, 0, merged, 0, arr1.length);
        System.arraycopy(arr2, 0, merged, arr1.length, arr2.length);
        return merged;
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int q = scanner.nextInt();
        Graph graph = new Graph(n);
        for (int i = 0; i < q; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            int c = scanner.nextInt();
            for (int j = 0; j < 7; j++) {
                graph.addEdge(a + j, b + j, c + j);
            }
        }
        System.out.println(graph.kruskal());
    }
}
