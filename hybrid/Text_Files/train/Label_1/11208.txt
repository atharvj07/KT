#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    ll m; cin >> m;
    vector<ll>v(m);
    for(int i = 0; i < m; i++) {
        cin >> v[i];
    }
    if(v[m - 1] % 2 == 1) {
        cout << -1 << endl;
        return 0;
    }
    vector<ll>ans = {v[0] / 2};
    for(int i = 1; i < m; i++) {
        ans.push_back(ans.back() + (v[i] - v[i - 1]) / 2);
    }
    cout << ans.size() << endl;
    for(int i = 0; i < ans.size(); i++) {
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
