#include<bits/stdc++.h>
using namespace std;

typedef long long ll;

ll gcd(ll a, ll b){
    if(!a) return b;
    return gcd(b%a,a);
}

int main(){
  ll n,i,j,st[200002],ans;
  cin >> n;
  for(i=0;i<n;++i){ cin >> st[i]; }
  sort(st,st+n);
  ans=0;
  for(i=0;i<n;++i){
    for(j=i+1;j<n;++j){
      if(gcd(st[i],st[j])==1) continue;
      for(ll k=j+1;k<n;++k){
        if(gcd(st[i],gcd(st[j],st[k]))==1) continue;
        ans+=gcd(st[i],gcd(st[j],st[k]));
      }
      ans+=2*gcd(st[i],st[j]);
    }
    ans+=st[i];
  }
  cout << ans%1000000007 << endl;
  return 0;
}
