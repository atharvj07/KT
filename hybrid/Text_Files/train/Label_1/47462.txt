import java.util.Scanner;

public class Osu {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        double[] p = new double[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextDouble();
        }

        double expectedScore = 0;
        int currentLength = 0;
        for (int i = 0; i < n; i++) {
            if (p[i] >= 0.5) {
                currentLength++;
            } else {
                expectedScore += Math.pow(currentLength, 2);
                currentLength = 0;
            }
        }
        expectedScore += Math.pow(currentLength, 2);

        System.out.println(expectedScore);
    }
}
