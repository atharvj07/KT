import java.util.Scanner;

public class RockPaperScissorsTournament {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int q = scanner.nextInt();

        char[] shapes = scanner.next().toCharArray();

        int[] champions = new int[n + 1];
        champions[1] = n;

        for (int i = 0; i < q; i++) {
            int player = scanner.nextInt() - 1;
            char newShape = scanner.next().charAt(0);

            // Update the number of possible champions after the request.
            champions[1] -= countLosers(player, shapes, newShape);
            champions[2] -= countLosers(player + 1, shapes, newShape);

            shapes[player] = newShape;

            // Print the result.
            System.out.println(champions[1]);
        }

        // Print the final result.
        System.out.println(champions[1]);
    }

    // Counts the number of players who would lose to the given player with the given shape.
    private static int countLosers(int player, char[] shapes, char newShape) {
        int count = 0;

        // Check the left neighbor.
        if (player > 0) {
            char leftShape = shapes[player - 1];
            if (newShape == 'R' && leftShape == 'S' ||
                newShape == 'P' && leftShape == 'R' ||
                newShape == 'S' && leftShape == 'P') {
                count++;
            } else if (newShape == leftShape) {
                // If it's a draw, both players have a chance of winning.
                count += 2;
            }
        }

        // Check the right neighbor.
        if (player < shapes.length - 1) {
            char rightShape = shapes[player + 1];
            if (newShape == 'R' && rightShape == 'S' ||
                newShape == 'P' && rightShape == 'R' ||
                newShape == 'S' && rightShape == 'P') {
                count++;
            } else if (newShape == rightShape) {
                // If it's a draw, both players have a chance of winning.
                count += 2;
            }
        }

        return count;
    }
}
