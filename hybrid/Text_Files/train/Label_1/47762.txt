import java.util.*;

class Main {
    static int solve(String a, String b) {
        return a.length() - b.length() + 1;
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        String t = sc.next();
        System.out.println(solve(s, t));
    }
}
