#include <bits/stdc++.h>
using namespace std;
string s;
int n, w[300010];
char c[1000010];
int main() {
  cin >> n;
  cin >> s;
  n--;
  int l = 0;
  for (int i = 0; i < n; i++) { s[i] == '?' ? w[i] = 3 : w[i] = s[i] == ')' ? 2 : 1; }
  for (int i = n - 1; i >= 0; i--)
    s[i] == ')' ? l-- : l++, w[i] += l, l = max(0, l);
  for (int i = 0; i < n; i++)
    if (w[i] == 3) {
      if (s[i + 1] == '(' || w[i + 1] == 2)
        s[i] = ')', w[i] = 2;
      else
        s[i] = '(', w[i] = 1;
    }
  if (w[n] == 1)
    cout << s << endl;
  else
    cout << ":(" << endl;
  return 0;
}
