#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int m, n;
    cin >> m;
    vector<int> q(m);
    for (int i = 0; i < m; i++) { cin >> q[i]; }
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    vector<vector<int>> dp(n + 1, vector<int>(m + 1, 1e9));
    dp[0][0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= m; j++) {
            dp[i][j] = dp[i - 1][j];
            if (j) dp[i][j] = min(dp[i][j], dp[max(0, i - q[j - 1] - 2)][j - 1] + min(a[i - 1], a[min(i - 2, i - 1)] * 2));
        }
    }
    cout << dp[n][m];
}
