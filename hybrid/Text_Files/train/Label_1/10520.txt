#include<bits/stdc++.h>
using namespace std;
int main() {
    int m, n;
    cin >> m >> n;
    vector<vector<int>> b(m, vector<int>(n));
    for (vector<int> &i : b) {
        for (int &j : i)
            cin >> j;
    }
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            bool ok = true;
            for (int k = 0; k < m; k++) {
                if (b[k][j] == 1 && k != i)
                    ok = false;
            }
            for (int k = 0; k < n; k++) {
                if (b[i][k] == 1 && j != k)
                    ok = false;
            }
            if (b[i][j] == 0 && ok)
                ok = false;
            if (b[i][j] == 1 && !ok)
                ok = false;
            if (!ok) {
                cout << "NO";
                return 0;
            }
        }
    }
    cout << "YES\n";
    for (vector<int> &i : b) {
        for (int j : i)
            cout << j << ' ';
        cout << '\n';
    }
}
