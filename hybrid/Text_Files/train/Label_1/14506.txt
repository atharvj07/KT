#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define vpii vector<pair<int,int>>
#define si set<int>
#define ll long long
vpii ids;
int n, k;
int main() {
    cin >> n >> k;
    ids.resize(k);
    map<int, int> mp;
    int last = 0;
    for (int i = 0; i < n; i++) {
        int id;
        cin >> id;
        if (mp[id]) {
            auto itr = mp.find(id);
            int pos = itr->second;
            if (pos == last)continue;
            for (int j = pos; j < last; j++) {
                ids[j] = ids[j + 1];
            }
            ids[last] = { id,mp[id] };
            (mp[id])++;
        }
        else {
            if (last == k) {
                mp.erase(ids[last].first);
                last--;
            }
            last++;
            (mp[id])++;
            ids[last] = { id,mp[id] };
        }
    }
    cout << last << endl;
    for (int i = 1; i <= last; i++) {
        cout << ids[i].first << " ";
    }
    cout << endl;
}
