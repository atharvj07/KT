#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define all(v) v.begin(), v.end()
#define endl "\n"
#define int long long

const int MAXN = 2e5 + 5;
int n, k, arr[MAXN];
vi ans;
map<int, int> last_pos;

int32_t main() {
  cin >> n >> k;
  for (int i = 0; i < n; ++i) { cin >> arr[i]; }
  for (int i = 0; i < n; ++i) {
    int id = arr[i];
    if (last_pos.count(id)) {
      ans.erase(ans.begin() + last_pos[id] - 1);
    } else if (ans.size() == k) {
      ans.pop_back();
    }
    ans.insert(ans.begin(), id);
    last_pos[id] = ans.size();
  }
  int sz = ans.size();
  cout << sz << endl;
  for (int i = 0; i < sz; ++i) { cout << ans[i] << " "; }
  cout << endl;
  return 0;
}
