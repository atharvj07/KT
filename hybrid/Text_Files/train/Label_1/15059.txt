#include <iostream>
 
using namespace std;
 
int main() {
    int n, m;
    cin >> n >> m;
    for (int p = 1, w = 1, z = 1; p <= n; p++, w ^= 1) {
        for (int q = 1; q <= m; q++, z ^= 1)
            cout << ((p == 1 || q == m || p == n || (p >= 3 && !w && q <= z)) ?'#' : '.');
        cout << endl;
    }
    return 0;
}
