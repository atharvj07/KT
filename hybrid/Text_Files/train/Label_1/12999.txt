#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main()
{
    int t;
    cin >> t;
    while(t--) {
        string s;
        cin >> s;
        int n = s.size();
        int f = 1;
        for(int i = 0; i < n - 1; i++) {
            if(s[i] != s[i + 1]) f = 0;
        }
        if(f) cout << "NET" << endl;
        else cout << "DA" << endl;
    }
    return 0;
}
