#include <iostream>

int main() {
    int n;
    std::cin >> n;
    if (n % 2 == 0) {
        for (int i = 0; i < n; i += 2) {
            std::cout << i + 2 << " ";
        }
        for (int i = 1; i < n; i += 2) {
            std::cout << i + 1 << " ";
        }
        std::cout << std::endl;
    } else {
        std::cout << -1 << std::endl;
    }
    return 0;
}
