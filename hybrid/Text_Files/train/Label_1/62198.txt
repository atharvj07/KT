import math

n, r = list(map(int, input().split()))
x = list(map(int, input().split()))
i = 0
while i < n:
    j = i + 1
    while j < n and x[j] - x[i] < 2 * r:
        j += 1
    if j < n:
        y1 = (x[j] - x[i]) ** 2 / 4. / r
        y2 = math.sqrt(y1 * y1 - x[i] * x[j] + r * r)
        x[i] = (y2 + y1 - x[i] * r / y2 + x[j] * r / y2) / (2./y2)  
        x[j] = r ** 2 / x[i]
        i = j
    else:
        x[i] = 0.
        i += 1
print(*x, sep=" ")
