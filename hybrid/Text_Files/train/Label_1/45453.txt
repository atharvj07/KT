import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int d = sc.nextInt();
            if (d == 0) {
                break;
            }
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[] di = new int[n];
            for (int i = 1; i < n; i++) {
                di[i] = sc.nextInt();
            }
            Arrays.sort(di);
            int ans = 0;
            for (int i = 0; i < m; i++) {
                int k = sc.nextInt();
                int dist = d;
                for (int j = 0; j < n; j++) {
                    int dist1 = di[j] - k;
                    if (dist1 < 0) {
                        dist1 += d;
                    }
                    int dist2 = k - di[j];
                    if (dist2 < 0) {
                        dist2 += d;
                    }
                    dist = Math.min(dist, Math.min(dist1, dist2));
                }
                ans += dist;
            }
            System.out.println(ans);
        }
    }
}
