#include <iostream>
using namespace std;
int main() {
  long long n, k;
  cin >> n >> k;
  bool r[k + 1];
  fill(r, r + k + 1, false);
  for (long long i = 1; i <= k; i++) {
    if (r[n % i]) {
      cout << "No" << endl;
      return 0;
    }
    r[n % i] = true;
  }
  cout << "Yes" << endl;
  return 0;
}
