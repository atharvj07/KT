for t in range(int(input())):
    n = int(input())
    l = list(input())
    a = b = 0
    for x in l:
        if x == 'B':
            b += 1
        elif x == 'R':
            a += 1
    l = [x for x in l if x == '?']
    ans = ['B'] * (b + len(l) // 2) + ['R'] * (a + (len(l) + 1) // 2)
    print(''.join(ans))
