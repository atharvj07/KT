import java.util.Scanner;

public class Mancala {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] holes = new int[14];

        for (int i = 0; i < 14; i++) {
            holes[i] = scanner.nextInt();
        }

        int maxScore = 0;
        for (int i = 0; i < 14; i++) {
            if (holes[i] > 0) {
                int stones = holes[i];
                holes[i] = 0;

                int j = (i + 1) % 14;
                while (stones > 0) {
                    holes[j]++;
                    stones--;
                    j = (j + 1) % 14;
                }

                int score = 0;
                for (int k = 0; k < 14; k++) {
                    if (holes[k] % 2 == 0) {
                        score += holes[k];
                        holes[k] = 0;
                    }
                }

                maxScore = Math.max(maxScore, score);

                // Reset the board for the next iteration
                holes[i] = stones;
                for (int k = 0; k < 14; k++) {
                    holes[k] += (k == i) ? stones : 0;
                }
            }
        }

        System.out.println(maxScore);
    }
}
