#include <bits/stdc++.h>
using namespace std;

vector<int> solve(vector<int> &h, int n) {
  vector<int> ans(n), l(n), r(n);
  for (int i = 1, j = 0; i < n; i++) {
    while (j > 0 && h[l[j - 1]] < h[r[j - 1]]) j--;
    l[j] = r[j] = i;
    while (j > 0 && h[l[j - 1]] + 2 >= h[r[j - 1]]) {
      ans[r[j - 1]] = ans[l[j - 1]] = (ans[l[j - 1]] + h[r[j - 1]]) / 2;
      j--;
    }
    j++;
    l[j] = i;
  }
  for (int i = 0; i < n; i++) ans[l[i]] = h[l[i]];
  return ans;
}

vector<int> read(int n) {
  vector<int> h(n + 1);
  scanf("%d", &h[1]);
  for (int i = 2; i <= n; i++) {
    scanf("%d", &h[i]);
    h[i] += h[i - 1];
  }
  return h;
}

int main() {
  int n;
  scanf("%d", &n);
  vector<int> h = read(n);
  vector<int> ans = solve(h, n);
  for (int i = 1; i <= n; i++) printf("%d ", ans[i]);
  printf("\n");
  return 0;
}
