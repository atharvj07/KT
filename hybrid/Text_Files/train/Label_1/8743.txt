#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k, m;
  cin >> n >> k >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = n * m;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    cnt++;
    if (cnt == k) {
      ans -= k;
      cnt = 0;
    }
  }
  cout << ans << endl;
  return 0;
}
