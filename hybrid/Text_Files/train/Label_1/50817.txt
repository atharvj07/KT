import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            ArrayList<Integer> lst = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                String s = sc.next();
                if (s.equals("x")) {
                    lst.add(i);
                } else {
                    lst.add(Integer.parseInt(s));
                }
            }
            if (lst.size() > 2) {
                if (lst.get(1) < lst.get(0) && lst.get(2) > lst.get(0)) {
                    if (lst.get(1) == lst.get(2) && lst.size() % 2 == 0) {
                        System.out.println("none");
                        continue;
                    }
                    System.out.println(lst.get(0));
                    continue;
                }
                if (lst.get(1) > lst.get(0) && lst.get(2) < lst.get(0)) {
                    if (lst.get(1) == lst.get(2) && lst.size() % 2 == 0) {
                        System.out.println("none");
                        continue;
                    }
                    System.out.println(lst.get(0));
                    continue;
                }
                if (lst.get(lst.size() - 2) < lst.get(lst.size() - 3) && lst.get(lst.size() - 1) > lst.get(lst.size() - 3)) {
                    if (lst.get(lst.size() - 2) == lst.get(lst.size() - 1) && lst.size() % 2 == 0) {
                        System.out.println("none");
                        continue;
                    }
                    System.out.println(lst.get(lst.size() - 3));
                    continue;
                }
                if (lst.get(lst.size() - 2) > lst.get(lst.size() - 3) && lst.get(lst.size() - 1) < lst.get(lst.size() - 3)) {
                    if (lst.get(lst.size() - 2) == lst.get(lst.size() - 1) && lst.size() % 2 == 0) {
                        System.out.println("none");
                        continue;
                    }
                    System.out.println(lst.get(lst.size() - 3));
                    continue;
                }
            }
            if (lst.size() == 2) {
                if (lst.get(0) == lst.get(1)) {
                    System.out.println("none");
                    continue;
                }
                System.out.println(lst.get(0));
                continue;
            }
            System.out.println("ambiguous");
        }
    }
}
