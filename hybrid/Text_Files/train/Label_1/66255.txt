import sys
input = sys.stdin.readline
n, m = map(int, input().split())
pos = {i:j-1 for j, i in enumerate(list(map(int, input().split())))}
adj = [[] for i in range(n)]
for u, v in (map(int, input().split()) for i in range(m)):
    u, v = u-1, v-1
    adj[pos[u]+1].append(pos[v])
    adj[pos[v]+1].append(pos[u])
    pos[u], pos[v] = pos[v], pos[u]
vis = [False] * n
ans = 0
for i in range(n):
    if vis[i]: continue
    vis[i] = True
    ans += 1
    j = i
    while len(adj[j]) == 2:
        vis[j] = True
        j = adj[j][1-adj[j].index(j)]
    vis[j] = True
print(ans-1)
