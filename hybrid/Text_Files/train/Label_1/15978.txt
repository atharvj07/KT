#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int k;
  cin >> k;
  vector<int> d(k);
  for (int i = 0; i < k; i++) {
    cin >> d[i];
  }
  sort(d.begin(), d.end());
  int n = 0;
  vector<int> ans;
  for (int i = 0; i < k; i++) {
    if (d[i] > 0) {
      ans.push_back(d[i]);
      n++;
    }
  }
  cout << n << endl;
  for (int i = 0; i < n; i++) {
    cout << ans[i] << " ";
  }
  cout << endl;
  return 0;
}
