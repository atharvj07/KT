#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
ll a[200500];
int main() {
    ll n, k;
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    ll l = 1, r = 1e6;
    while (l != r) {
        ll mid = (l + r) / 2;
        ll ct = 0, i = 0, j = n - 1;
        while (i <= j) {
            if (a[j] + a[i] <= mid) {
                j--;
                ct++;
                i++;
            } else i++;
        }
        if (ct < k) l = mid + 1;
        else r = mid;
        if (ct == k && l == r) cout << r << endl;
    }
}
