#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> c(n), a(m);
    for (int &x : c) cin >> x;
    for (int &x : a) cin >> x;
    int bought = 0, i = 0, j = 0;
    while (i < n && j < m) {
        if (a[j] >= c[i]) {
            bought++;
            j++;
        }
        i++;
    }
    cout << bought << endl;
    return 0;
}
