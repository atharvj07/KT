import sys
input=sys.stdin.readline;n=int(input())
s=list(input().strip());p=list(input().strip())
i=j=ans=0
while i<n:
 if s[i]==p[j]:i+=1;j+=1
 else:
  if i+1<n and s[i]==p[j+1] and s[i+1]==p[j]:
   i+=2;j+=2;ans+=1
  elif j+1<n and s[j+1]==p[j] and s[j]==p[j+1]:
   i+=2;j+=2;ans+=1
  else:
   i+=1
print(ans)
