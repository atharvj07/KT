n = int(input())
a = list(map(int, input().split()))

count_decreasing = 0
position = 0

for i in range(1, n):
    if a[i] < a[i - 1]:
        count_decreasing += 1
        position = i

if count_decreasing > 1:
    print(-1)
else:
    if count_decreasing == 0:
        print(0)
    elif a[0] >= a[-1]:
        print(n - position)
    else:
        print(-1)
