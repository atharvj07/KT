#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> l(m);
  for (int i = 0; i < m; i++) cin >> l[i];
  sort(l.begin(), l.end());
  vector<int> p(m);
  int i, j, cnt = 1, color = 1;
  for (i = 0, j = 0; i < m; i++) {
    if (j == cnt) {
      j = 0;
      color++;
    }
    p[i] = j + 1;
    cnt += l[i] - 1;
    j++;
  }
  if (cnt > n) {
    cout << "-1\n";
    return 0;
  }
  for (int i = 0; i < m; i++) {
    cout << p[i] << ' ';
  }
  cout << '\n';
}
