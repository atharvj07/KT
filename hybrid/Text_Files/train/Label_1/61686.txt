import math

a1, b1 = map(int, input().split())
a2, b2 = map(int, input().split())

gcdb = math.gcd(a1, b1)
gcdn = math.gcd(a2, b2)

if gcdb == gcdn:
    print(0)
    print("{} {}".format(a1, b1))
    print("{} {}".format(a2, b2))
elif gcdb % gcdn != 0 and gcdn % gcdb != 0:
    print(-1)
else:
    if gcdb % gcdn == 0:
        if a1 % (gcdb // gcdn) != 0 or b1 % (gcdb // gcdn) != 0:
            print(-1)
        else:
            a1 //= (gcdb // gcdn)
            b1 //= (gcdb // gcdn)
            print(1)
            print("{} {}".format(a1, b1))
            print("{} {}".format(a2, b2))
    elif gcdn % gcdb == 0:
        if a2 % (gcdn // gcdb) != 0 or b2 % (gcdn // gcdb) != 0:
            print(-1)
        else:
            a2 //= (gcdn // gcdb)
            b2 //= (gcdn // gcdb)
            print(1)
            print("{} {}".format(a1, b1))
            print("{} {}".format(a2, b2))
