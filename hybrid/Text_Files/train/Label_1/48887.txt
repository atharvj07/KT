import java.util.*;

public class Main {
    static class Edge {
        int u, v;

        Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static int n, m, s, t;
    static List<Edge> edges = new ArrayList<>();
    static List<Integer>[] adj;

    static int[] dist;
    static boolean[] vis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        s = sc.nextInt() - 1;
        t = sc.nextInt() - 1;
        adj = new List[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            edges.add(new Edge(u, v));
            adj[u].add(v);
            adj[v].add(u);
        }
        dist = new int[n];
        vis = new boolean[n];
        bfs(s);
        int ans = 0;
        for (Edge edge : edges) {
            int u = edge.u;
            int v = edge.v;
            if (dist[u] == dist[v]) continue;
            ans++;
        }
        System.out.println(ans);
    }

    static void bfs(int start) {
        Queue<Integer> q = new LinkedList<>();
        q.add(start);
        vis[start] = true;
        dist[start] = 0;
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (int next : adj[cur]) {
                if (!vis[next]) {
                    q.add(next);
                    vis[next] = true;
                    dist[next] = dist[cur] + 1;
                }
            }
        }
    }
}
