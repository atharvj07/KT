import sys

def solve():
    t = int(sys.stdin.readline().strip())
    for _ in range(t):
        n, m = [int(x) for x in sys.stdin.readline().strip().split()]
        bottom = [int(x) for x in sys.stdin.readline().strip().split()]
        left = [int(x) for x in sys.stdin.readline().strip().split()]
        crashes = 0
        for i in range(1, 101):
            if i in bottom and i in left:
                crashes += 1
        print(crashes // 2)

solve()
