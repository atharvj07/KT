import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int m = input.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = input.nextInt();
        }
        if (!canMakeEqual(a, n, b, m)) {
            System.out.println(-1);
            return;
        }
        int maxLength = 0;
        int i = 0;
        int j = 0;
        long sumA = 0;
        long sumB = 0;
        while (i < n && j < m) {
            if (sumA == sumB) {
                maxLength = Math.max(maxLength, i + j);
            }
            if (sumA <= sumB) {
                sumA += a[i];
                i++;
            } else {
                sumB += b[j];
                j++;
            }
        }
        while (i < n) {
            sumA += a[i];
            i++;
        }
        while (j < m) {
            sumB += b[j];
            j++;
        }
        if (sumA == sumB) {
            maxLength = Math.max(maxLength, i + j);
        }
        System.out.println(maxLength);
    }

    private static boolean canMakeEqual(int[] a, int n, int[] b, int m) {
        long sumA = 0;
        long sumB = 0;
        for (int i = 0; i < n; i++) {
            sumA += a[i];
        }
        for (int i = 0; i < m; i++) {
            sumB += b[i];
        }
        long diff = Math.abs(sumA - sumB);
        if (diff % 2 == 1) {
            return false;
        }
        diff /= 2;
        Arrays.sort(a);
        Arrays.sort(b);
        int i = 0;
        int j = m - 1;
        while (diff > 0 && i < n && j >= 0) {
            if (a[i] <= diff && b[j] >= diff) {
                diff -= a[i];
                i++;
            } else if (a[i] >= diff && b[j] <= diff) {
                diff -= b[j];
                j--;
            } else if (a[i] <= diff) {
                diff -= a[i];
                i++;
            } else {
                diff -= b[j];
                j--;
            }
        }
        if (diff == 0) {
            return true;
        }
        return false;
    }
}
