#include<iostream>
#include<vector>
using namespace std;
int main(){
    int n, a, b = 0, c = 0;
    cin >> n;
    vector<int> d(n);
    for (int i = 0; i < n; i++){
        cin >> a;
        d[i] = a - b;
        b = a + c;
        if(d[i] == 0 && n > 1){
            cout << "NO" << endl;
            return 0;
        }
        else if(d[i] > 0) c -= min(c, (d[i] + 1) / 2);
        else if(d[i] < 0) c -= min(c, -d[i]);
    }    
    if(c != 0) cout << "NO" << endl;
    else cout << "YES" << endl;
    return 0;
}
