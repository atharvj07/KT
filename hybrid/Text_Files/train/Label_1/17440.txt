#include<bits/stdc++.h>
#define eb emplace_back
using namespace std;

int main() {
    string s, t;
    while(getline( cin, s) ) {
        getline( cin, t);
        int ns = s.size(), nt = t.size();
        vector<int> pos[26];
        for ( int i = 0; i < ns; i++) if( s[i] != '?') pos[ s[i] - 'a'].eb( i);
        int best = 0, bst_idx = 0;
        for ( int i = 0; i < 26; i++) {
            int j, k, nc = pos[i].size();
            for ( j = 0, k = 0; j < nt && k < nc; j++) {
                while( k < nc && pos[i][k] < j) k++;
                if(k < nc && pos[i][k] == j) k++, best += nt - j;
            }
            if(best > bst_idx) bst_idx = i;
        }
        for ( int i = 0; i < ns; i++) {
            if(s[i] == '?') cout << (char) (bst_idx + 'a');
            else cout << s[i];
        }
        cout << '\n';
    }
    return 0;
}
