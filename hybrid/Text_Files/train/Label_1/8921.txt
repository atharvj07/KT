#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> cola(n + 1);
    int tot = 0;
    for (int x = 1; x <= n; x++) {
        cin >> cola[x].first;
        cin >> cola[x].second;
        tot += cola[x].first;
    }

    if (tot % 2 == 1) {
        cout << "NO";
        return 0;
    }
    for (int x = 1; x <= n; x++) {
        if (cola[x].first > cola[x].second || 2 * cola[x].first > tot) {
            cout << "NO";
            return 0;
        }
    }
    cout << "YES";
    return 0;
}
