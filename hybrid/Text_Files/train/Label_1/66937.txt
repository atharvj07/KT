from collections import Counter

s=input()
t=input()
pos=[]
for i in range(len(s)):
    if s[i]=='?': pos.append(i)
st=Counter(t)
cnt=[0]*len(pos)
for i in pos:
    for j in range(97,97+26):
        st[chr(j)]-=1
        if st[chr(j)]>=0: cnt[i]+=1
        st[chr(j)]+=1
ans=list(s)
for i in range(len(pos)):
    c=0
    for j in range(97,97+26):
        st[chr(j)]-=1
        if st[chr(j)]>=0:
            if cnt[i]==c:
                ans[pos[i]]=chr(j)
                c+=1
            cnt[i]-=1
        st[chr(j)]+=1
print(''.join(ans))
