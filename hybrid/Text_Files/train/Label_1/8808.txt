#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 10;
int main() {
  int n;
  cin >> n;
  int a[n], b[n], sz = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (i == 0 || a[i] != a[i - 1]) b[sz++] = a[i];
  }
  for (int i = 0, j = 0; i < sz; i++) {
    while (j < sz && __gcd(b[i], b[j]) == 1) j++;
    ans = max(ans, j - i);
  }
  cout << ans << endl;
  return 0;
}
