#define _CRT_SECURE_NO_WARNINGS
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    //freopen("input.txt", "r", stdin);
    //freopen("output.txt", "w", stdout);

    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> freq(n + 1, 0);
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            freq[x]++;
        }
        int x = -1;
        for (int i = 1; i <= n; i++) {
            if (freq[i] > 0) {
                x = i;
                break;
            }
        }
        if (x == -1) {
            cout << 0 << "\n";
            continue;
        }
        int ans = n - freq[x];
        cout << ans << "\n";
    }

    return 0;
}
