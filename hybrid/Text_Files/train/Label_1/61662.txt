n, m = map(int, input().split())
sequence = [int(x) for x in input().split()]
fingerprints = set([int(x) for x in input().split()])
max_code = ""
for i in range(n):
    code = ""
    for j in range(i, n):
        if sequence[j] in fingerprints:
            code += str(sequence[j])
        else:
            break
    if len(code) > len(max_code):
        max_code = code
print(*map(int, list(max_code)))
