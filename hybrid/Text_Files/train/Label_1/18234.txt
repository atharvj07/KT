#include <bits/stdc++.h>
using namespace std;
const int N = 100;
int t, n, m, a[N], cnt[N], f[N][N][N];
int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        cnt[a[i]]++;
    }
    bool ok = true;
    for (int j = 1; j <= m; j++) ok &= (cnt[j] >= t);
    if (!ok)
        return cout << "NO" << endl, 0;
    memset(f, 0, sizeof(f));
    f[0][0][0] = 1;
    for (int i = 0; i < n; i++)
        for (int j = 0; j <= n; j++)
            for (int k = 0; k <= m; k++) {
                f[i + 1][j][k] |= f[i][j][k];
                if (j + 1 <= cnt[a[i]] && k == a[i]) f[i + 1][j + 1][k] |= f[i][j][k];
            }
    if (f[n][t][m])
        return cout << "YES" << endl, 0;
    else
        return cout << "NO" << endl, 0;
}
