n, U = map(int, input().split())
E = list(map(int, input().split()))

max_efficiency = -1

for i in range(n):
    for k in range(i+2, n):
        if E[k] - E[i] <= U:
            j = i + 1  # j is always between i and k
            efficiency = (E[k] - E[j]) / (E[k] - E[i])
            max_efficiency = max(max_efficiency, efficiency)
        else:
            break

if max_efficiency == -1:
    print(-1)
else:
    print(f'{max_efficiency:.9f}')
