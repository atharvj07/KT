#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MOD = 1e9 + 7;
ll qpow(ll base, ll p = MOD - 2) { ll ans = 1; for (; p; p >>= 1) { if (p & 1) ans = ans * base % MOD; base = base * base % MOD; } return ans % MOD; }
int main() {
    ll n, f1, f2, f3, c;
    cin >> n >> f1 >> f2 >> f3 >> c;
    c = qpow(c);
    ll ans = f3;
    ans = (ans * c) % MOD;
    ans = (ans * f2) % MOD;
    ans = (ans * f1) % MOD;
    for (ll i = 4; i < n; i++) {
        ans = (ans * c) % MOD;
    }
    cout << ans << endl;
    return 0;
}
