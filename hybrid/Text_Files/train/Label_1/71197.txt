direy=[-1,1,0,0]
direx=[0,0,-1,1]
r,c=map(int,input().split())
arr=[list("." for i in range(c))for j in range(r)]
ans=0
for i in range(r):
    arr[i]=list(input())

for i in range(r):
    for j in range(c):
        if arr[i][j]==".":
            cnt=0
            for k in range(4):
                x,y=i+direy[k],j+direx[k]
                if 0<=x<r and 0<=y<c:
                    if arr[x][y]==".":
                        cnt+=1
            if cnt>0:
                ans+=1
print(ans)
