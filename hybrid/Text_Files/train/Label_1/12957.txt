#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    multiset<int> s;
    for (int i = 0; i < n; i++) {
      int tmp;
      cin >> tmp;
      s.insert(tmp);
    }
    for (int i = 1; i <= 100; i++) {
      auto it = s.find(i);
      if (it == s.end()) {
        if (!x) {
          cout << i - 1 << endl;
          break;
        }
        x--;
      } else {
        s.erase(it);
      }
    }
  }
  return 0;
}
