import java.util.*;
import java.lang.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int sum = 0;
        boolean odd = true;
        for (int i = 0; i < n; i++) {
            if ((a[i] % 2 == 0) != odd) {
                sum += a[i];
                odd = !odd;
            }
        }
        System.out.println(sum);
    }
}
