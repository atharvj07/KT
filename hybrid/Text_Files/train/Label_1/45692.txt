import java.util.Arrays;
import java.util.Scanner;

public class VovaAndTests {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int p = scanner.nextInt();
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int[] marks = new int[k];
        for (int i = 0; i < k; i++) {
            marks[i] = scanner.nextInt();
        }

        Arrays.sort(marks);

        int sum = 0;
        for (int mark : marks) {
            sum += mark;
        }

        int remainingTests = n - k;
        int remainingSum = x - sum;

        if (remainingSum < 0 || remainingSum % remainingTests != 0) {
            System.out.println("-1");
            return;
        }

        int remainingMedian = y;
        if (remainingTests % 2 == 0) {
            remainingMedian = (y + marks[k / 2]) / 2;
        }

        int targetMark = remainingSum / remainingTests;
        int[] result = new int[remainingTests];
        for (int i = 0; i < remainingTests; i++) {
            if (i < remainingTests / 2) {
                result[i] = targetMark;
            } else {
                result[i] = remainingMedian;
            }
        }

        Arrays.sort(result);

        for (int mark : result) {
            System.out.print(mark + " ");
        }
    }
}
