#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll mod=1000000007;
ll fact[2003],inv[2002];
ll p(ll x, ll y){
    if(y==0) return 1LL;
    ll t=p(x,y/2);
    ll ans=((t%mod)*(t%mod))%mod;
    if(y&1){
        ans=((ans%mod)*(x%mod))%mod;
    }
    return ans;
}
ll pw(ll x, ll y){
    if(y==0) return 1LL;
    ll t=pw(x,y/2);
    ll ans=((t%mod)*(t%mod))%mod;
    if(y&1){
        ans=((ans%mod)*(x%mod))%mod;
    }
    return ans;
}
ll _inv(ll x){
    if(inv[x]!=-1) return inv[x];
    ll y=mod-2;
    return inv[x]=((p(x,y)%mod));
}
ll nCr(ll x, ll y){
    if(x<0 || x<y || y<0) return 0LL;
    return (((fact[x]%mod)*((_inv(fact[y]%mod)%mod)*(_inv(fact[x-y]%mod)%mod)))%mod);
}
int main(){
    fact[0]=inv[0]=1;
    for(ll i=1;i<=2000;i++){
        fact[i]=(fact[i-1]%mod*i)%mod;
        inv[i]=-1;
    }
    ll n;
    cin>>n;
    ll arr[n];
    for(ll i=0;i<n;i++){
        cin>>arr[i];
        if(arr[i]==-1) arr[i]=0;
    }
    ll cnt0=0,cnt1=0;
    for(ll i=0;i<n;i++){
        if(arr[i]==0) cnt0++;
        if(arr[i]) cnt1++;
    }
    cout<<(((p(2,cnt1)%mod)*nCr(cnt0+cnt1-1,cnt1)%mod))<<endl;
    return 0;
}
