#include <bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, i, x = 0, y = 0;
        cin >> n;
        for (i = 0; i < n; i++) {
            int a;
            cin >> a;
            x = __gcd(x, a);
            y = min(y, a);
        }
        if (x > 1)
            cout << -1 << endl;
        else
            cout << max(1, y) << endl;
    }
    return 0;
}
