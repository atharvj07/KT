#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 998244353
int main() {
  ll n, m;
  cin >> n >> m;
  vector<ll> a(n), b(m);
  for (auto &i : a) cin >> i;
  for (auto &i : b) cin >> i;
  vector<ll> ct(n);
  for (ll i = 0; i < n; ++i) {
    if (i && b[lower_bound(b.begin(), b.end(), a[i]) - b.begin()] == a[i]) {
      ct[i] = ct[i - 1] + (a[i] == a[i - 1]);
    } else {
      ct[i] = 1ll;
    }
  }
  ll ans = 1ll, last = -1, cnt = 0;
  for (ll i = 0; i < n; ++i) {
    if (a[i] == b[last + 1]) {
      ++cnt;
    } else {
      cnt = 1, last++;
    }
    ans = (ans * ct[i]) % mod;
    if (cnt > 1) {
      ans = (ans * cnt) % mod;
    }
    if (last + 1 == m) {
      break;
    }
  }
  if (last + 1 < m) ans = 0ll;
  cout << ans << '\n';
  return 0;
}
