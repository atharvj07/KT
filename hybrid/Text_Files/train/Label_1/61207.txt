from sys import stdin

for _ in range(int(stdin.readline())):
  n = int(stdin.readline().strip())
  grid = [list(stdin.readline().strip()) for _ in range(n)]
  for i in range(n):
    for j in range(n):
      if grid[i][j] == '*':
        xx, yy = i, j
        break
  for i in range(n):
    for j in range(n):
      if grid[i][j] == '*' and (i != xx or j != yy):
        grid[xx][j] = grid[i][yy] = '*'
        break
  for r in grid:
    print(''.join(r))
