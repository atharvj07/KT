#include <iostream>
#include <vector>
using namespace std;

int main() {
    int a[200001] = {}, n;
    vector<int> ans;
    cin >> n;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        a[x] = i + 1;
    }
    for (int i = 1, l = -1, r = n + 1; i <= 200000; i++) {
        if (a[i] > l && a[i] < r) {
            l = a[i];
            ans.push_back(1);
        } else if (a[i] > l && a[i] < r) {
            r = a[i];
            ans.push_back(0);
        }
    }
    cout << ans.size() << endl;
    for (int i : ans) cout << (i ? 'R' : 'L');
    return 0;
}
