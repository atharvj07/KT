#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  string num;
  cin >> num;
  long long pref[n + 1], suf[n + 1];
  pref[0] = 0;
  for (int i = 0; i < n; i++) pref[i + 1] = pref[i] + num[i] - '0';
  suf[n] = 0;
  for (int i = n - 1; i >= 0; i--) suf[i] = suf[i + 1] + num[i] - '0';
  bool found = false;
  for (int i = 1; i < n; i++) {
    if (pref[i] == suf[i]) {
      found = true;
      break;
    }
  }
  cout << (found ? "YES" : "NO") << endl;
}
