#include<iostream>

using namespace std;

int main(){
    int x, y, a, b, red[2005], pink[2005];
    cin >> x >> y >> a >> b;
    for (int i = x; i <= b; i += x) red[i] = 1;
    for (int i = y; i <= b; i += y) pink[i] = 1;
    int ans = 0;
    for (int i = a; i <= b; i++) if (red[i] && pink[i]) ans++;
    cout << ans << endl;
    return 0;
}
