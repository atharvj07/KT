#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 998244353;
const int maxn = 2e5 + 10;
ll a[maxn];
ll dp[maxn][2];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  int m;
  cin >> m;
  for (int i = 1; i <= m; i++) {
    int x, y;
    cin >> x >> y;
    ll ans1 = 0, ans2 = 0;
    for (int j = 0; j <= n; j++) dp[j][0] = dp[j][1] = 1e18;
    dp[0][0] = 0;
    dp[0][1] = 0;
    for (int j = 1; j <= n; j++) {
      dp[j][0] = min(dp[j][0], dp[j - 1][0]);
      if (a[j] >= x) dp[j][0] = min(dp[j][0], dp[j - 1][1]);
      dp[j][1] = min(dp[j][1], dp[j - 1][1] + 1);
      if (a[j] >= x) dp[j][1] = min(dp[j][1], dp[j - 1][0] + 1);
    }
    ans1 = dp[n][0];
    ans2 = dp[n][1];
    dp[0][0] = 0;
    dp[0][1] = 0;
    for (int j = 1; j <= n; j++) {
      dp[j][0] = min(dp[j][0], dp[j - 1][0]);
      if (a[j] >= y) dp[j][0] = min(dp[j][0], dp[j - 1][1]);
      dp[j][1] = min(dp[j][1], dp[j - 1][1] + 1);
      if (a[j] >= y) dp[j][1] = min(dp[j][1], dp[j - 1][0] + 1);
    }
    ans1 = min(ans1, dp[n][0]);
    ans2 = min(ans2, dp[n][1]);
    cout << ans1 << endl;
  }
  return 0;
}
