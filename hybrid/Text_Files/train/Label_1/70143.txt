m = int(input())
x1, a1 = map(int, input().split())
p1, q1 = map(int, input().split())
x2, a2 = map(int, input().split())
p2, q2 = map(int, input().split())

def extgcd(a, b):
    if b == 0:
        return 1, 0, a
    x1, y1, gcd = extgcd(b, a % b)
    x, y = y1, x1 - (a // b) * y1
    return x, y, gcd

x, y, gcd = extgcd(p1, p2)
if (a1 - a2) % gcd != 0:
    print(-1)
else:
    ans = int((x * (a1 - a2) / gcd) % (p2 / gcd))
    print(ans)
