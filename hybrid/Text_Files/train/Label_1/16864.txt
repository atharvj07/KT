#include<bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    sort(a.begin(), a.end());
    int sum = 0, cnt = 0;
    for (int i = 0; i < n; i++) {
      if (1ll * cnt * a[i] >= 1ll * sum + a[i])
        break;
      sum += a[i];
      cnt++;
    }
    printf("%d\n", n - cnt);
  }
  return 0;
}
