import sys
input = sys.stdin.readline

n, x = map(int, input().split())

g = [[] for _ in range(n+1)]
vis = [0] * (n+1)

for _ in range(n-1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

q = [(1, 0)]
ans = 0

while q:
    u, d = q.pop(0)
    if u == x:
        ans = d+1
    vis[u] = 1
    for v in g[u]:
        if vis[v] == 0:
            q.append((v, d+1))

print(ans)
