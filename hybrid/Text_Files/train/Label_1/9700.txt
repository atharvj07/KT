#include <bits/stdc++.h>
using namespace std;

const int N = 2e5+5;

int n, b[N], dp[N];
long long ans;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> n;
    for(int i = 1; i <= n; ++i) cin >> b[i];

    for(int i = 1; i <= n; ++i) {
        if(dp[i - 1] >= 0) dp[i] = dp[i - 1] + b[i];
        else dp[i] = b[i];
        ans = max(ans, 1LL * dp[i]);
    }
    cout << ans << endl;
    return 0;
}
