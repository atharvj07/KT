import sys

n, sx, sy = map(int, input().split())
a = []
for _ in range(n):
    a.append(tuple(map(int, input().split())))

ans = 0
ans_x, ans_y = 0, 0
for x, y in a:
    if abs(x - sx) + abs(y - sy) != 0:
        ans_x = (sx + x) // 2
        ans_y = (sy + y) // 2
        ans += 1
sys.stdout.write(str(ans) + '\n' + str(ans_x) + ' ' + str(ans_y))
