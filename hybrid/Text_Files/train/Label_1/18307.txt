#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int M = 1e9 + 7;
int n, t, ans, l, r;
ll p[300005], m[300005];
int main() {
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> t;
        p[t]++;
        if (p[t] == 2) {
            ans++;
            l = m[t];
            r = i;
            if (l > 0) cout << l << " " << r << endl;
            m[t] = 0;
        } else m[t] = i;
    }
    for (int i = 1; i <= 300000; i++) {
        if (p[i] == 1) {
            cout << "-1" << endl;
            return 0;
        }
    }
    cout << ans << endl;
}
