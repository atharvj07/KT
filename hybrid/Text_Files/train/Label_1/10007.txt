#include <iostream>
#include <unordered_map>
using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_map<string, int> f = {
      {"Tetrahedron", 4}, {"Cube", 6},   {"Octahedron", 8},
      {"Dodecahedron", 12}, {"Icosahedron", 20}};
  int ans = 0;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    ans += f[s];
  }
  cout << ans << endl;
  return 0;
}
