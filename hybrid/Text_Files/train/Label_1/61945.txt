n,m=map(int,input().split())
s=[]
for i in range(n): s.append([x for x in input()])
q=int(input())
for i in range(q):
    l,r=map(int,input().split())
    if s[0][r-1]=='.': print('YES')
    else:
        f=False
        for j in range(l-1,r):
            if s[0][j]=='.': f=True
        print(['NO','YES'][f])
