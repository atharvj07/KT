#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    multiset<int> s;
    while (n--) {
        int x, y;
        cin >> x >> y;
        s.insert(min(x, y));
        s.insert(max(x, y));
    }
    int last = *s.rbegin();
    s.erase(s.find(last));
    while (!s.empty()) {
        if (*s.rbegin() >= last) {
            cout << "NO\n";
            return 0;
        }
        last = *s.rbegin();
        s.erase(s.find(last));
    }
    cout << "YES\n";
    return 0;
}
