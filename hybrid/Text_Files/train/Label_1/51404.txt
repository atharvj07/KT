import java.util.*;

public class Main {
	static Scanner in = new Scanner(System.in);

	public static void main(String[] args) {
		int n = in.nextInt(), k = in.nextInt();
		int[] a = new int[n];
		for (int i = 0; i < n; i++) {
			a[i] = in.nextInt();
		}

		int[] max = new int[n];
		max[n - 1] = a[n - 1];
		for (int i = n - 2; i >= 0; i--) {
			max[i] = Math.max(a[i], max[i + 1]);
		}

		int[] dp = new int[n + 1];
		dp[n] = 0;
		for (int i = n - 1; i >= 0; i--) {
			dp[i] = Integer.MAX_VALUE;
			for (int j = i; j < n; j++) {
				dp[i] = Math.min(dp[i], dp[j + 1] + (j - i + 1) * max[i]);
			}
		}

		System.out.println(dp[0]);
	}
}
