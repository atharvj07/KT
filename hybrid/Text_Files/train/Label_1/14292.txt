#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int m, t, r;
    cin >> m >> t >> r;
    vector<int> w(m);
    for (int i = 0; i < m; i++) cin >> w[i];
    if (t < r) {
        cout << -1 << endl;
        return 0;
    }
    sort(w.begin(), w.end());
    int cnt = 0, cur = 0;
    for (int i = 0; i < m; i++) {
        cnt += cur + t < w[i];
        cur = max(cur + t, w[i]) - 1;
    }
    cout << cnt << endl;
    return 0;
}
