#include <cmath>
#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

const int MXN=1e2+3;
int N, A[MXN];

int main(){
    int T; scanf("%d",&T);
    while(T--){
        scanf("%d", &N);
        for(int i=0; i<N; i++){
            scanf("%d", &A[i]);
        }
        sort(A, A+N);
        int sum=0, prod=1, neg=0, zero=0;
        for(int i=0; i<N; i++){
            sum+=A[i];
            prod*=A[i];
            zero+=(A[i]==0);
            neg+=(A[i]<0);
        }
        if(prod && zero) prod=0;
        if(neg&1){
            if(!prod && sum!=0) prod=1;
            else if(!prod && sum==0){
                if(A[0]>0) prod=1;
                for(int i=1; i<N; i++) if(A[i]<0){ prod=-1; break; }
            }
        }
        if(prod) printf("%d\n", 0);
        else if(sum) printf("%d\n", 1);
        else printf("%d\n", 2);
    }
    return 0;
}
