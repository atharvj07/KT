#include <bits/stdc++.h>
using namespace std;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    if (k % 2 == 0) { s.push_back(s[k - 1]); }
    int flip = 0;
    for (int i = 0; i < k - 1; i++) { flip += (s[i] != s[i + 1]); }
    if (s[0] == s[k - 1])
      flip++;
    cout << flip / 2 << endl;
  }
}
