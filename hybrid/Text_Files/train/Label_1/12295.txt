#include <bits/stdc++.h>
using namespace std;
int main() {
    int k;
    cin >> k;
    if (k == 2) cout << "NO\n";
    else {
        cout << "YES\n";
        cout << k + 2 << ' ' << k + 1 << '\n';
        for (int i = 1; i <= k + 1; ++i) { cout << i << ' ' << i + 1 << '\n'; }
        cout << k + 1 << ' ' << k + 2 << '\n';
    }
    return 0;
}
