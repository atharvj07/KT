#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    int day = 0;
    for (int i = 1; i <= n; i++) {
        k -= min(k, 8);
        k -= a[i];
        day++;
        if (k <= 0) { break; }
    }
    cout << (k <= 0 ? day : -1) << endl;
    return 0;
}
