#include <bits/stdc++.h>
using namespace std;
const int N=2e3+5,M=2e6+5;
int n,ans;
bool bad[N];
int freq[M];
int main(){
   int t;cin>>t;
   while(t--){
      cin>>n;
      ans=0;
      memset(freq,0,sizeof freq);
      for(int i=1;i<=n;i++){
         int x;cin>>x;
         if(x%2==0)bad[i]=1;
         freq[x]++;
      }
      for(int i=2;i<=2e5;i++){
         int x=0;
         for(int j=i;j<=2e5;j+=i){
            x+=freq[j];
            if(x>=2)ans+=x*(x-1)/2;
         }
      }
      for(int i=1;i<=n;i++){
         if(bad[i]==0)
            for(int j=2;j<=2e5;j+=2){
               ans+=freq[j]*(freq[j]-1)*(freq[j*2]>0);
            }
      }
      cout<<ans<<endl;
   }
   return 0;
}
