n, m = map(int, input().split())
x = list(map(int, input().split()))
y = list(map(int, input().split()))

i, j, files = 0, 0, 0
while i < n and j < m:
    sum_x, sum_y = x[i], y[j]
    while sum_x != sum_y:
        if sum_x < sum_y:
            i += 1
            if i < n:
                sum_x += x[i]
        else:
            j += 1
            if j < m:
                sum_y += y[j]
    i += 1
    j += 1
    files += 1

print(files)
