import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int h = sc.nextInt();
        int m = sc.nextInt();
        double alpha = 2 * Math.PI * h / 12 + 2 * Math.PI * (double) m / 60 / 12;
        double d = Math.sqrt(a * a + b * b - 2 * a * b * Math.cos(alpha));
        System.out.println(d);
    }
}
