#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (i + 1 < n && a[i] < a[i + 1]) ans++;
      else
        break;
    }
    cout << ans << endl;
  }
}
