n,m=map(int,input().split())
a=[0]*n
d=[0]*1000001
for i in range(n):
    a[i]=int(input())
    d[a[i]]+=1
mx=0
for i in range(m):
    mx=max(mx,d[i+1])
mn=1000000000
for i in range(m):
    mn=min(mn,d[i+1])
print(mn,mx-mn)
for i in range(n):
    if d[a[i]]<=mn:
        for j in range(m):
            if d[j+1]>mn:
                a[i]=j+1
                d[a[i]]+=1
                d[a[i]-1]-=1
                break
for i in range(n):
    print(a[i],end=" ")
