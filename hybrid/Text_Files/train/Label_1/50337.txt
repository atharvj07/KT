import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        System.out.println((s.charAt(0) == 'A' && s.indexOf('C') > 2 && s.indexOf('C') < s.length() - 2
                && s.substring(1, s.indexOf('C')).toLowerCase().equals(s.substring(1, s.indexOf('C')))
                && s.substring(s.indexOf('C') + 1).toLowerCase()
                        .equals(s.substring(s.indexOf('C') + 1))) ? "AC" : "WA");
    }
}
