#include <bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, q;
    cin >> n >> q;
    vector<pair<int, int>> a(q);
    for (int i = 0; i < q; i++)
    {
        int x;
        cin >> x;
        if (x <= 2)
            cin >> a[i].first;
        else
            cin >> a[i].second;
        a[i].second = x;
    }
    int cnt[n + 1] = {}, sum = 0;
    for (int i = 0; i < q; i++)
    {
        if (a[i].second == 1)
            cnt[a[i].first]++;
        if (a[i].second == 2)
            cnt[a[i].first] = 0;
        if (a[i].second == 3)
        {
            sum = 0;
            for (int j = 0; j <= n; j++)
            {
                sum += min(a[i].first, cnt[j]);
            }
        }
        cout << sum << "\n";
    }
}
