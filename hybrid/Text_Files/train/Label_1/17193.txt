#include <iostream>
#include <vector>
#include <cmath>
#include <cstdio>
using namespace std;
int main() {
  vector<int> graph[3001];
  bool visited[3001] = {};
  int n, m;
  cin >> n >> m;
  while (m--) {
    int a, b;
    cin >> a >> b;
    graph[a].push_back(b);
  }
  long long ans = 0;
  for (int i = 1; i < n; i++) {
    for (int j : graph[i]) {
      for (int k : graph[j]) {
        if (i == k) continue;
        for (int l : graph[k]) {
          if (j == l || i == l) continue;
          ans++;
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
