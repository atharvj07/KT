n, m = map(int, input().split())
matrix = [input() for _ in range(n)]

if n == 1 or m == 1:
    print(0)
else:
    cnt_0 = cnt_1 = 0
    for i in range(n):
        for j in range(m):
            if (i + j) % 2 == 0:
                cnt_0 += matrix[i][j] == '0'
                cnt_1 += matrix[i][j] == '1'
            else:
                cnt_0 += matrix[i][j] == '1'
                cnt_1 += matrix[i][j] == '0'
    if (n * m) % 2 == 0 and cnt_0 != cnt_1:
        print(-1)
    else:
        print(min(cnt_0, cnt_1))
