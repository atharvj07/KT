import sys
in1 = list(map(int, input().split()))
n = in1[0]
k = in1[1]
fighters = []
for _ in range(n): fighters.append(list(map(int, input().split())))
fighters.sort(key = lambda x: x[0], reverse = True)
effort = 0
for i, fighter in enumerate(fighters):
    if k > i + 1 and fighter[0] > 0:
        effort += fighter[1]
if fighters[k - 1][0] < 0: effort = -1
print(effort)
