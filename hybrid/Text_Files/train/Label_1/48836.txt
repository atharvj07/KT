import java.util.Scanner;

public class LongestUncommonSubsequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String a = scanner.nextLine();
        String b = scanner.nextLine();

        int[][] dp = new int[a.length() + 1][b.length() + 1];

        for (int i = 1; i <= a.length(); i++) {
            for (int j = 1; j <= b.length(); j++) {
                if (a.charAt(i - 1) == b.charAt(j - 1)) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                } else {
                    dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
                }
            }
        }

        int longestCommonSubsequenceLength = dp[a.length()][b.length()];
        int longestUncommonSubsequenceLength = Math.max(a.length(), b.length()) - longestCommonSubsequenceLength;

        System.out.println(longestUncommonSubsequenceLength == 0 ? "-1" : longestUncommonSubsequenceLength);
    }
}
