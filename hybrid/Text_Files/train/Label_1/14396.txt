#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007
bool prime[200001];
vector<ll> primes;
 
void sieve(ll n)
{
    memset(prime, true, sizeof(prime));
 
    for (ll p = 2; p * p <= n; p++)
    {
        if (prime[p] == true)
        {
            for (ll i = p * p; i <= n; i += p)
                prime[i] = false;
        }
    }
    for (ll p = 2; p <= n; p++)
        if (prime[p])
            primes.push_back(p);
}
 
int main()
{
    sieve(200001);
    ll t;
    cin>>t;
    while(t--)
    {
        ll n,x=0;
        cin>>n;
        if(n&1)
        {
            cout<<"NO\n";
            continue;
        }
        if(n<=6)
        {
            cout<<"NO\n";
            continue;
        }
        if(n%2==0)
        {
            n=n-6;
            cout<<"YES 6 2 ";
        }
        ll ctr=0;
        for(ll i=0;i<primes.size();i++)
        {
            if(primes[i]*primes[i]<=n)
            {
                cout<<primes[i]*primes[i]<<" ";
                ctr++;
            }
            if(ctr==2)
                break;
        }
        cout<<n<<endl;
    }
    return 0;
}
