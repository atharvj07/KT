import java.io.*;
import java.math.BigInteger;
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        BigInteger n = new BigInteger(sc.next());
        int k = Integer.parseInt(sc.next());
        BigInteger ans = BigInteger.ZERO;
        for (int i = k; i <= 3; i++) {
            BigInteger mul_9 = BigInteger.valueOf(9);
            BigInteger mul = mul_9.pow(i - 1);
            BigInteger tmp = new BigInteger(n.toString()).divide(mul);
            BigInteger div_9 = new BigInteger(n.toString()).divide(mul_9);
            ans = ans.add(tmp.subtract(div_9));
        }
        System.out.println(ans);
    }
}
