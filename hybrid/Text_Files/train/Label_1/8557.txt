#include<bits/stdc++.h>
using namespace std;
const long long mod = 1000000007;
int main() {
    int n; cin >> n;
    if (n % 3 != 0) cout << 0 << endl;
    else {
        long long ans = 1;
        for (int i = 0; i < n; i++) {
            ans *= 3;
            ans %= mod;
        }
        cout << ans << endl;
    }
}
