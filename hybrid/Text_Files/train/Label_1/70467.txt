n, m = map(int, input().split())
islands = [list(map(int, input().split())) for _ in range(n)]
bridges = list(map(int, input().split()))

# Calculate the minimum and maximum possible lengths for each bridge
bridge_lengths = []
for i in range(1, n):
    min_length = islands[i][0] - islands[i - 1][1]
    max_length = islands[i][1] - islands[i - 1][0]
    bridge_lengths.append((min_length, max_length))

# Sort bridge lengths by their minimum required length
bridge_lengths.sort(key=lambda x: x[0])

# Sort bridges
bridges.sort()

# Assign bridges
bridge_indices = []
for min_length, max_length in bridge_lengths:
    found = False
    for i, bridge in enumerate(bridges):
        if min_length <= bridge <= max_length:
            bridge_indices.append(i + 1)  # Store index (1-based)
            bridges[i] = float('inf')  # Mark as used
            found = True
            break
    if not found:
        print("No")
        break
else:
    print("Yes")
    print(" ".join(map(str, bridge_indices)))
