#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, cnta = 0, cntb = 0, cntc = 0;
    string s;
    cin >> s;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] == 'A') cnta++;
        else if (s[i] == 'B') cntb++;
        else if (s[i] == 'C') cntc++;
    }
    if (cnta > 0 && cntb > 0 && cntc > 0) cout << "Yes";
    else cout << "No";
    return 0;
}
