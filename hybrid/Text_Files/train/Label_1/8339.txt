#include <bits/stdc++.h>
using namespace std;
int32_t main() {
    string s;
    cin >> s;
    int k;
    cin >> k;
    vector<int> w(26);
    for (auto &i : w) cin >> i;
    vector<int> suf(s.size(), 0);
    int sum = 0;
    for (int i = s.size() - 1; i >= 0; i--) {
        sum += w[s[i] - 'a'];
        suf[i] = sum;
    }
    int ans = sum;
    for (int i = 0; i < s.size() && k; i++) {
        ans += (i + 1) * w[s[i] - 'a'];
        if (i && ans < suf[i] + (i + 1) * w[s[i] - 'a']) {
            ans = suf[i] + (i + 1) * w[s[i] - 'a'];
            sum = suf[i];
        }
        ans = max(ans, sum + (i + k + 1) * w[s[i] - 'a']);
        sum += w[s[i] - 'a'];
        k--;
    }
    cout << ans << endl;
}
