#include <iostream>
#include <cmath>

using namespace std;

int main() {
  long long n;
  cin >> n;
  long long l = floor((sqrt(8 * n + 1) - 1) / 2);
  long long d = l * (l + 1) / 2;
  long long dx = 0, dy = -l;
  if (n >= d) {
    n -= d;
    dx = l;
    dy = -l;
    l++;
  }
  while (n >= l) {
    n -= l;
    dx--;
  }
  dy += n;
  cout << dx << " " << dy << endl;
  return 0;
}
