#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    int sum1 = 0, sum2 = 0;
    while (n--) {
        sum1 += a[0] > a[n - 1] ? (a[0]), a[0] : (a[n - 1]), a[n - 1];
        sum2 += (a[0] > a[n - 1] ? (a[n - 1]), a[n - 1] : (a[0]), a[0]);
    }
    cout << sum1 << " " << sum2;
}
