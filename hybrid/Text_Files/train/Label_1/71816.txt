from decimal import Decimal, getcontext
getcontext().prec = 60
y1, y2, yw, xb, yb, r = map(int, input().split())
d = Decimal(y2 - y1) / 2
e = Decimal(yw - yb) - d
if e <= r:
    print(-1)
    exit()
v = Decimal(3 * r - d + e) / 2
ans = Decimal(xb) + v
if y1 < v < y2:
    print(ans)
else:
    print(-1)
