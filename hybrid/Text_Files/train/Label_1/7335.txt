#include <iostream>
using namespace std;
int main() {
    int a, b, ans = 0;
    cin >> a >> b;
    for (int i = a; i <= b; i++) {
        string s = to_string(i);
        for (int j = 0; j < s.length(); j++) {
            if (s[j] == '1' || s[j] == '4')
                ans += 2;
            else if (s[j] == '0' || s[j] == '6' || s[j] == '9')
                ans += 6;
            else if (s[j] == '2' || s[j] == '3' || s[j] == '5')
                ans += 5;
            else if (s[j] == '7')
                ans += 3;
            else if (s[j] == '8')
                ans += 7;
        }
    }
    cout << ans << endl;
}
