#include <cstdio>
#include <vector>
#include <queue>

int n, m, k;
const int maxn = 500;
char g[maxn][maxn];
bool vis[maxn][maxn];
int dx[] = {0, 1, 0, -1}, dy[] = {1, 0, -1, 0};

struct point { int x, y; };
point r;

void flood_fill(int x, int y) {
    if (x < 0 || x >= n || y < 0 || y >= m || vis[x][y] || g[x][y] != '.') return;
    vis[x][y] = true;
    r = { x, y };
    for (int i = 0; i < 4; ++i) flood_fill(x + dx[i], y + dy[i]);
}

void bfs(std::vector<point> &a) {
    int cnt = 0;
    std::queue<point> q;
    for (auto pt : a) q.push(pt);
    while (!q.empty()) {
        auto p = q.front(); q.pop();
        g[p.x][p.y] = 'X';
        ++cnt;
        if (cnt == k) return;
        for (int i = 0; i < 4; ++i) {
            int nx = p.x + dx[i], ny = p.y + dy[i];
            if (nx >= 0 && nx < n && ny >= 0 && ny < m && g[nx][ny] == '.')
                q.push({nx, ny});
        }
    }
}

int main() {
    scanf("%d %d %d", &n, &m, &k);
    for (int i = 0; i < n; ++i) scanf("%s", g[i]);
    flood_fill(0, 0);
    g[r.x][r.y] = '.';

    std::vector<point> v;
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j)
            if (g[i][j] == '.') v.push_back({i, j});
    bfs(v);

    for (int i = 0; i < n; ++i) printf("%s\n", g[i]);
    return 0;
}
