from collections import deque

n,m = map(int,input().split())
a = [[x for x in input()] for i in range(n)]
dx = [0,0,-1,1]
dy = [-1,1,0,0]
ans = 0

def valid(x,y):
    return 0<=x<n and 0<=y<m and a[x][y]=='.'

def dfs(x,y,go,cnt):
    global ans
    if cnt==4:
        ans+=1
        return
    for k in range(4):
        nx,ny = x+dx[k], y+dy[k]
        if valid(nx,ny) and (not go or go!=k) and (not go or go!=(k+2)%4):
            a[nx][ny]='*'
            dfs(nx,ny,k,cnt+1)
            a[nx][ny]='.'

for i in range(n):
    for j in range(m):
        if (i==0 or i==n-1) and (j!=0 and j!=m-1):
            if a[i][j]=='.':
                a[i][j]='*'
                dfs(i,j,0,1)
                a[i][j]='.'
        elif (j==0 or j==m-1) and (i!=0 and i!=n-1):
            if a[i][j]=='.':
                a[i][j]='*'
                dfs(i,j,0,1)
                a[i][j]='.'
print(ans//2)
