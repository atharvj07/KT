import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] l = new int[n];
        int[] s = new int[n];
        int[] c = new int[m + n];
        for (int i = 0; i < n; i++) {
            l[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            s[i] = scanner.nextInt();
        }
        for (int i = 0; i < n + m; i++) {
            c[i] = scanner.nextInt();
        }
        int[][] dp = new int[n + 1][m + n + 1];
        boolean[][] in = new boolean[n + 1][m + n + 1];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 1; j <= m + n; j++) {
                int best = dp[i + 1][j];
                boolean b = in[i + 1][j];
                if (!b || j != l[i]) {
                    int profit = dp[i][j] + c[j - 1];
                    if (j != l[i]) {
                        profit += c[j - 1];
                    }
                    if (profit > best) {
                        best = profit;
                        b = true;
                    }
                }
                dp[i][j] = best;
                in[i][j] = b;
            }
        }
        int max = 0;
        for (int j = 1; j <= m + n; j++) {
            max = Math.max(max, dp[0][j] - s[0]);
        }
        System.out.println(max);
    }
}
