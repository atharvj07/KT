#include <iostream>
#include <cstring>
#include <vector>
using namespace std;
int main() {
    int n; cin >> n;
    string s; cin >> s;
    if (s[0] == '_') s = '_' + s;
    if (s[n - 1] == '_') s = s + '_';
    vector<int> cnt(3, 0);
    bool in = false;
    for (int i = 0; i < n; i++) {
        if (s[i] == '_' || s[i] == '(' || s[i] == ')') {
            in = false;
            cnt[0] = max(cnt[0], cnt[1]);
            cnt[1] = 0;
        } else if (s[i] >= 'a' && s[i] <= 'z') {
            if (!in) cnt[1]++, in = true;
        } else {
            if (!in) cnt[2]++, in = true;
        }
    }
    cout << cnt[0] << ' ' << cnt[2] << endl;
}
