n = int(input())
a = list(map(int, input().split()))

def max_segments(a):
    basis = []
    for num in a:
        tmp = num
        for b in basis:
            tmp = min(tmp, tmp ^ b)
        if tmp > 0:
            basis.append(tmp)
    if len(basis) == 0:
        return -1
    return len(basis)

print(max_segments(a))
