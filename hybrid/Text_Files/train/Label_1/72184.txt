import sys
for t in range(int(input())):
    input()
    n, m = list(map(int, input().split()))
    g = [[] for _ in range(n+1)]
    vis = [False]*(n+1)
    for _ in range(m):
        a, b = list(map(int,input().split()))
        g[a].append(b)
    def dfs(v):
        if vis[v] == True: return '-1'
        if len(g[v])==0: return '1'
        vis[v] = True
        ans = []
        for u in g[v]:
            ans.append(dfs(u))
        vis[v] = False
        if ans.count('0')>0: return '0'
        if ans.count('-1')>0: return '-1'
        if len(set(ans))==1: return ans[0]
        return '2'
    print(*[dfs(v) for v in range(1,n+1)])
