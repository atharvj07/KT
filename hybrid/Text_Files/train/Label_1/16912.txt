#include <cstdio>
#include <vector>

using namespace std;

typedef long double ld;

const int MAXN = 100005;

int n;
vector<int> g[MAXN];
ld dp[MAXN];

ld dfs(int u, int p) {
  dp[u] = 1;
  for (int v : g[u]) {
    if (v == p) continue;
    dp[u] += dfs(v, u) / (n - 1);
  }
  return dp[u];
}

int main() {
  scanf("%d", &n);
  for (int i = 0; i < n - 1; ++i) {
    int a, b;
    scanf("%d%d", &a, &b);
    g[a].push_back(b);
    g[b].push_back(a);
  }
  dfs(1, 1);
  printf("%.10Lf\n", dp[1]);
  return 0;
}
