import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    static final int INF = 1<<30;
    static int[] dijkstra(List<List<Edge>> g, int s) {
        int n = g.size();
        int[] dist = new int[n];
        PriorityQueue<Edge> q = new PriorityQueue<>((a, b) -> a.cost - b.cost);
        Arrays.fill(dist, INF);
        dist[s] = 0;
        q.offer(new Edge(s, 0));
        while (!q.isEmpty()) {
            Edge e = q.poll();
            if (e.cost > dist[e.v]) continue;
            for (Edge ne: g.get(e.v)) {
                int w = e.cost + ne.cost;
                if (dist[ne.v] > w) {
                    q.offer(new Edge(ne.v, w));
                    dist[ne.v] = w;
                }
            }
        }
        return dist;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<List<Edge>> g = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            g.add(new ArrayList<>());
        }
        for (int i = 0; i < n; i++) {
            String[] arr = br.readLine().split(" ");
            int src = Integer.parseInt(arr[0]);
            int k = Integer.parseInt(arr[1]);
            for (int j = 0; j < k; j++) {
                int dst = Integer.parseInt(arr[2 + j * 2]);
                int cost = Integer.parseInt(arr[3 + j * 2]);
                g.get(src).add(new Edge(dst, cost));
            }
        }
        int[] dist = dijkstra(g, 0);
        for (int i = 0; i < n; i++) {
            System.out.println(i + " " + dist[i]);
        }
    }

    static class Edge {
        int v, cost;
        public Edge(int v, int cost) {
            this.v = v;
            this.cost = cost;
        }
    }
}
