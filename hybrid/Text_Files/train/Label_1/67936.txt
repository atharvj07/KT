from collections import defaultdict

def dfs(v, parent):
    for child in tree[v]:
        if child != parent:
            depth[child] = depth[v] + 1
            dfs(child, v)

def update_node_value(x, val):
    if depth[x] % 2 == 0:
        even_values[x] += val
        odd_values[x] -= val
    else:
        even_values[x] -= val
        odd_values[x] += val

def propagate_value_to_children(v, parent, val):
    for child in tree[v]:
        if child != parent:
            update_node_value(child, -val)
            propagate_value_to_children(child, v, -val)

n, m = map(int, input().split())
a = list(map(int, input().split()))

tree = defaultdict(list)
for _ in range(n - 1):
    u, v = map(int, input().split())
    tree[u].append(v)
    tree[v].append(u)

depth = [0] * (n + 1)
even_values = [0] * (n + 1)
odd_values = [0] * (n + 1)

dfs(1, 0)

for i in range(1, n + 1):
    if depth[i] % 2 == 0:
        even_values[i] = a[i - 1]
    else:
        odd_values[i] = a[i - 1]

for _ in range(m):
    query = input().split()
    if query[0] == '1':
        x, val = map(int, query[1:])
        update_node_value(x, val)
        propagate_value_to_children(x, 0, val)
    else:
        x = int(query[1])
        if depth[x] % 2 == 0:
            print(even_values[x] + odd_values[1])
        else:
            print(odd_values[x] + even_values[1])
