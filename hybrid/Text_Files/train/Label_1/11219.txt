#include <iostream>
#include <vector>
#include <unordered_map>

using namespace std;

int main() {
    string days[] = {"monday", "tuesday", "wednesday", "thursday", "friday", "saturday", "sunday"};
    unordered_map<string, int> dayOfWeek;
    for (int i = 0; i < 7; i++) dayOfWeek[days[i]] = i;
    
    string firstDay, secondDay;
    cin >> firstDay >> secondDay;
    
    cout << ((dayOfWeek[firstDay] + 1) % 7 == dayOfWeek[secondDay] ? "YES" : "NO") << endl;
    
    return 0;
}
