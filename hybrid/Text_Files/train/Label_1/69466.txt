n, k = map(int, input().split())
a = list(map(int, input().split()))

if k == 2:
    print(min(a))
else:
    min_cost = float('inf')
    for i in range(n - k + 1):
        subseq_max = max(a[i:i+k])
        for j in range(i, i + k):
            if a[j] == subseq_max:
                left = max(a[i:j:2] + [0])
                right = max(a[j+1:i+k:2] + [0])
                cost = min(subseq_max, max(left, right))
                min_cost = min(min_cost, cost)
                break
    print(min_cost)
