import java.util.Scanner;

class MartianDollars {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int b = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[] dp = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            dp[i] = dp[i - 1];
            for (int j = 0; j < i; j++) {
                dp[i] = Math.max(dp[i], dp[j] + a[i - 1] * (b / a[j]));
            }
        }

        System.out.println(dp[n]);
    }
}
