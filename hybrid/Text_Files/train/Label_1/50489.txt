import java.util.*;

public class Main {
  static int n, m, x;
  static char[][] keyboard;
  static int[][] distance = new int[60][60];
  static final int INF = (int) 1e9;

  public static void main(String[] args) throws Exception {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    m = sc.nextInt();
    x = sc.nextInt();
    keyboard = new char[n][m];
    for (int i = 0; i < n; i++) {
      String line = sc.next();
      for (int j = 0; j < m; j++) {
        keyboard[i][j] = line.charAt(j);
        if (line.charAt(j) == 'S')
          for (int dx = -1; dx <= 1; dx++)
            for (int dy = -1; dy <= 1; dy++)
              distance[dx + 30][dy + 30] = Math.min(distance[dx + 30][dy + 30], Math.max(Math.abs(dx), Math.abs(dy)));
      }
    }
    for (int dx = -1; dx <= 1; dx++)
      for (int dy = -1; dy <= 1; dy++)
        if (Math.abs(dx) + Math.abs(dy) <= x)
          distance[dx + 30][dy + 30] = 0;
    for (int k = 0; k < 60; k++)
      for (int i = 0; i < 60; i++)
        for (int j = 0; j < 60; j++)
          distance[i][j] = Math.min(distance[i][j], distance[i][k] + distance[k][j]);
    String text = sc.next();
    int ans = 0;
    for (int i = 0; i < text.length(); i++) {
      char ch = text.charAt(i);
      if (Character.isUpperCase(ch))
        ch = Character.toLowerCase(ch);
      boolean found = false;
      outer: for (int dx = -1; dx <= 1; dx++)
        for (int dy = -1; dy <= 1; dy++)
          if (Math.abs(dx) + Math.abs(dy) <= x) {
            int pos = (dx + 30) * m + (dy + 30);
            for (int j = 0; j < m; j++)
              if (keyboard[(dx + n / 2) % n][j] == ch || keyboard[(dx + n / 2 + 1) % n][j] == ch) {
                found = true;
                if (keyboard[(dx + n / 2) % n][j] == 'S' || keyboard[(dx + n / 2 + 1) % n][j] == 'S')
                  ans++;
                break outer;
              }
            for (int j = 0; j < m; j++)
              if (keyboard[(dx + n / 2) % n][j] == 'S' && distance[pos][(dy + 30) * m + (j + 30)] <= x) {
                found = true;
                ans++;
                break outer;
              }
            for (int j = 0; j < m; j++)
              if (keyboard[(dx + n / 2 + 1) % n][j] == 'S' && distance[pos][(dy + 30) * m + (j + 30)] <= x) {
                found = true;
                ans++;
                break outer;
              }
          }
      if (!found) {
        System.out.println(-1);
        return;
      }
    }
    System.out.println(ans);
  }
}
