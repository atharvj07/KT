#include <iostream>
#include <vector>
using namespace std;

long long n, k;

int main() {
  cin >> n >> k;
  vector<long long> ans;
  for (long long a = 1, b; a <= n; a = b + 1) {
    b = n / (n / a);
    if (n / a >= k) ans.push_back(n / a);
  }
  if (ans.empty())
    cout << 1 << endl;
  else
    cout << ans[0] << endl;
  return 0;
}
