#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    long long n, k;
    cin >> n >> k;
    vector<long long> a(n), b(n);
    for (long long i = 0; i < n; i++) cin >> a[i];
    for (long long i = 0; i < n; i++) cin >> b[i];
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    long long ans = 0;
    for (long long i = 0; i < k; i++) ans += a[i];
    for (long long i = k; i < n; i++) ans += min(a[i], b[i]);
    cout << ans << endl;
    return 0;
}
