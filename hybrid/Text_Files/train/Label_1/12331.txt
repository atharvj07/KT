#include <bits/stdc++.h>
using namespace std;
const int maxn = 1005;
int n,a[maxn],dp[maxn][maxn],ms;
int rec(int l,int r){
    if(l == r)return a[l];
    int &ret = dp[l][r];
    if(ret != -1)return ret;
    ret = 1e9;
    ret = min(ret,max(rec(l+1,r),a[l]));
    ret = min(ret,max(rec(l,r-1),a[r]));
    return ret;
}
int main(){
    cin >> n;
    for(int i=1; i<=n; i++)cin >> a[i];
    memset(dp,-1,sizeof dp);
    cout << rec(1,n) << endl;
}
