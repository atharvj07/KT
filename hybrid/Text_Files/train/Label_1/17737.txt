#include <cstdio>
using namespace std;
#define min(_, __) ((_)<(__)?(_):(__))
const int MAXN = 16;
int f[MAXN][102], g[MAXN][102], a[MAXN][102], n, m;
int main() {
  scanf("%d%d", &n, &m);
  for (int i = n; i >= 1; i--) {
    scanf("%s", &a[i][1]);
  }
  for (int j = m; j >= 1; j--) {
    for (int i = n; i >= 1; i--) {
      if (a[i][j] == '0' && a[i][j + 1] == '0') {
        f[i][j] = f[i][j + 1] + 1;
        g[i][j] = g[i][j + 1] + 1;
      } else if (a[i][j] == '0' && a[i][j + 1] == '1') {
        f[i][j] = f[i][j + 1] + 1;
        g[i][j] = min(g[i][j + 1] + 1, g[i][j]);
      } else if (a[i][j] == '1' && a[i][j + 1] == '0') {
        f[i][j] = min(f[i][j], f[i + 1][j + 1]);
        g[i][j] = (g[i][j] == 0) ? 1 : min(g[i + 1][j + 1], g[i][j]);
      } else if (a[i][j] == '1' && a[i][j + 1] == '1') {
        f[i][j] = min(f[i + 1][j] + 1, f[i][j] + 1);
        g[i][j] = min(g[i + 1][j] + 1, min(g[i][j], g[i][j + 1] + 1));
      }
    }
  }
  printf("%d", g[1][1]);
  return 0;
}
