import sys

n, m, k = map(int, sys.stdin.readline().split())
a = []
for i in range(n):
    a.append(list(map(int, sys.stdin.readline().split())))

dp = [[[(1 << 30) for _ in range(2)] for _ in range(m)] for _ in range(n)]
dp[0][0][0] = 0

for i in range(n):
    for j in range(m):
        for x in range(2):
            if x == 0:  # xor from (0, 0) to (i, j)
                dp[i][j][x] = dp[i][j][x] & a[i][j]
                if i > 0:
                    dp[i][j][x] = min(dp[i][j][x], dp[i - 1][j][x] & a[i][j])
                if j > 0:
                    dp[i][j][x] = min(dp[i][j][x], dp[i][j - 1][x] & a[i][j])
            else:  # xor from (0, 0) to (i, j) == k
                dp[i][j][x] = dp[i][j][x] ^ a[i][j]
                if i > 0:
                    dp[i][j][x] = min(dp[i][j][x], dp[i - 1][j][x] ^ a[i][j])
                if j > 0:
                    dp[i][j][x] = min(dp[i][j][x], dp[i][j - 1][x] ^ a[i][j])
ans = 0
for x in range(2):
    if dp[n - 1][m - 1][x] == k:
        ans += 1
print(ans)
