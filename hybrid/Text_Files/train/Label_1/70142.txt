m = int(input())
h1, a1 = map(int, input().split())
x1, y1 = map(int, input().split())
h2, a2 = map(int, input().split())
x2, y2 = map(int, input().split())
d1 = (a1 - h1) % m
d2 = (a2 - h2) % m
if d1 == d2 == 0:
    print(1)
    exit()
if (x1 - x2) % m == 0 and (y1 - y2) % m == 0:
    print(-1)
    exit()
k = ((d2 - d1) * pow((x2 - x1) % m, m - 2, m)) % m
print(k if k > 0 else m + k)
