from math import gcd
from functools import reduce

def lcm(a, b):
    return a * b // gcd(a, b)

def lcm_list(lst):
    return reduce(lcm, lst)

N = int(input())
M = list(map(int, input().split()))
R = list(map(int, input().split()))

lcm_M = lcm_list(M)
days = [0] * lcm_M

for i in range(N):
    for j in range(R[i], lcm_M, M[i]):
        days[j] = 1

print(sum(days) / lcm_M)
