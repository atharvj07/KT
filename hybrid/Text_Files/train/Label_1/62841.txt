n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
used = [False] * n
k = 0
t = 0
for i in range(n):
    if not used[i]:
        k += 1
        rem = a[i]
        j = i
        while rem > 0:
            used[j] = True
            need = b[j] - a[j]
            if need <= rem:
                rem -= need
                t += need
                j = (j + 1) % n
                if used[j]:
                    break
            else:
                t += rem
                a[j] += rem
                rem = 0
print(k, t)
