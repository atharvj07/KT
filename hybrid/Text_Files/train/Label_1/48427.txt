import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int[] cards = new int[n];
    for (int i = 0; i < n; i++) {
      cards[i] = scanner.nextInt();
    }
    int sereja = 0, dima = 0;
    boolean serejaTurn = true;
    while (n > 0) {
      if (serejaTurn) {
        if (cards[0] > cards[n - 1]) {
          sereja += cards[0];
          cards[0] = -1;
        } else {
          sereja += cards[n - 1];
          cards[n - 1] = -1;
        }
      } else {
        if (cards[0] > cards[n - 1]) {
          dima += cards[0];
          cards[0] = -1;
        } else {
          dima += cards[n - 1];
          cards[n - 1] = -1;
        }
      }
      serejaTurn = !serejaTurn;
      n--;
      boolean allNeg = true;
      for (int i = 0; i < n; i++) {
        allNeg &= (cards[i] == -1);
      }
      if (allNeg) {
        n = 0;
      }
    }
    System.out.println(sereja + " " + dima);
  }
}
