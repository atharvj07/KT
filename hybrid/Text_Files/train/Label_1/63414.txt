n = int(input())
products = [list(map(int, input().split())) for _ in range(n)]
products.sort(key=lambda x: min(x[0], x[1]))

total_cost = 0
total_bought = 0
i = 0
j = n - 1

while i <= j:
    if total_bought >= products[i][1]:
        total_cost += products[i][0]
        total_bought += products[i][0]
        i += 1
    else:
        if i != j and total_bought + products[j][0] > products[i][1]:
            buy = products[i][1] - total_bought
            products[j][0] -= buy
            total_cost += buy * 2
            total_bought += buy
        else:
            total_cost += products[j][0] * 2
            total_bought += products[j][0]
            j -= 1

print(total_cost)
