import sys
input = sys.stdin.readline

mod = 10**9 + 7
n, m = map(int, input().split())
d = [0]*(n+1)
for _ in range(m):
    a, b, c = map(int, input().split())
    d[a] |= (1 << b)
    d[b] |= (1 << a)
    if c: d[a] |= d[b]

for i in range(1, n+1):
    x = d[i]
    d[i] = (d[i] ^ (d[i] >> 1)) & x
    d[i] = (d[i] ^ (d[i] >> 2)) & x
    d[i] = (d[i] ^ (d[i] >> 4)) & x
    d[i] = (d[i] ^ (d[i] >> 8)) & x
    d[i] = (d[i] ^ (d[i] >> 16)) & x
    d[i] %= mod

# assume that subset of i has happy triangle
ans = 1
for i in range(1, n+1):
    ans = (ans * d[i]) % mod

print(ans)
