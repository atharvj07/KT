import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] firstLine = br.readLine().split(" ");
        String[] secondLine = br.readLine().split(" ");

        int l1 = Integer.parseInt(firstLine[0]);
        int r1 = Integer.parseInt(firstLine[1]);
        int l2 = Integer.parseInt(secondLine[0]);
        int r2 = Integer.parseInt(secondLine[1]);

        int overlappedLength = 0;
        if (l1 <= l2) {
            if (r1 >= l2) {
                overlappedLength = Math.min(r1, r2) - l2;
            }
        } else {
            if (r2 >= l1) {
                overlappedLength = Math.min(r1, r2) - l1;
            }
        }
        System.out.println(overlappedLength);
    }
}
