#include <iostream>
#include <map>
#include <vector>
using namespace std;
int main() {
    int n, m;
    string a, b;
    cin >> a >> b;
    n = a.length(), m = b.length();
    vector<int> cnt(26);
    for (int i = 0; i < n; i++) cnt[a[i] - 'a']++;
    int sum = 0;
    bool ok = true;
    for (int i = 0; i < m; i++) {
        if (!cnt[b[i] - 'a']) ok = false;
        cnt[b[i] - 'a']--;
        sum++;
    }
    cout << (ok ? sum : -1);
}
