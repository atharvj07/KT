#include <bits/stdc++.h>

using namespace std;

#define endl '\n'

// cost: (n - t.length())
// subtask 1: (1 <= n, k <= 100)
int minCostToMakeKSubsequences(string& s, int n, int k) {
    vector<pair<char, int>> freq(26);
    for (const auto& c : s) {
        ++freq[c - 'a'].second;
    }
    sort(rbegin(freq), rend(freq));
    int rem = k;
    int cost = 0;
    for (const auto& [c, f] : freq) {
        int take = min(f, rem);
        if (take == 0) {
            break;
        }
        cost += (n - take);
        rem -= take;
    }
    if (rem == 0) {
        return cost;
    }
    return -1;
}

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    cout << minCostToMakeKSubsequences(s, n, k) << endl;
    return 0;
}
