import random

t = int(input())

for _ in range(t):
    n = int(input())
    b = list(map(int, input().split()))
    b.sort()
    if b[n] != sum(b[:n]):
        print(-1)
        continue
    a = b[:n]
    a.sort()
    if b[n+1] == sum(a) and b[n+1] <= 10**9:
        print(*a)
    else:
        print(-1)
