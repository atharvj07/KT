for t in range(int(input())):
    input()
    a=list(map(int,input().split()))
    ans=-1
    fr=[0]*len(a)
    for i in range(len(a)):
        fr[i]=[a[i],0]
    for i in range(len(a)):
        fr[i][1]+=1
        if i>0:
            fr[i][1]+=fr[i-1][1]
    for x in fr:
        if (x[1]==len(a)-fr.count(x)):
            if ans==-1:ans=1
            ans=min(x[1],ans)
    for i in range(len(fr)-1):
        if fr[i][0]==fr[i+1][0]:
            ans=min(ans,i+1)
    if ans!=-1:ans+=1
    print(ans)
