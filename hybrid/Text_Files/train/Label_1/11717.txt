#include<bits/stdc++.h>
using namespace std;
int n,a[51][51],b[51],ans=0;
bool ok(int x,int y){
	for(int i=1; i<x; ++i)
		if(a[i][y]==a[x][y]) return 0;
	for(int j=1; j<y; ++j)
		if(a[x][j]==a[x][y]) return 0;
	return 1;
}
bool check(){
	int sum=0;
	for(int i=1; i<=n; ++i){
		for(int j=1; j<=n; ++j)
			sum^=a[i][j];
		if(sum%2==0) return 0;
		sum=0;
	}
	for(int i=1; i<=n; ++i){
		for(int j=1; j<=n; ++j)
			sum^=a[j][i];
		if(sum%2==0) return 0;
		sum=0;
	}
	sum=0;
	for(int i=1; i<=n; ++i)
		sum^=a[i][i];
	if(sum%2==0) return 0;
	sum=0;
	for(int i=1; i<=n; ++i)
		sum^=a[i][n-i+1];
	if(sum%2==0) return 0;
	return 1;
}
void Try(int x){
	if(x==n*n+1){
		if(check()){
			for(int i=1; i<=n; ++i){
				for(int j=1; j<=n; ++j)
					cout<<a[i][j]<<" ";
				cout<<endl;
			}
			exit(0);
		}
		else
			ans++;
	}
	else{
		for(int i=1; i<=n*n; ++i){
			if(b[i]==0){
				b[i]=1;
				a[x/n+1][x%n+1]=i;
				if(ok(x/n+1,x%n+1))
					Try(x+1);
				b[i]=0;
			}
		}
	}
}
int main() {
	cin>>n;
	Try(1);
	cout<<ans<<endl;
	return 0;
}
