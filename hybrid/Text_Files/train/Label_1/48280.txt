import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int[] v = new int[n];
        for (int i = 0; i < n; i++) {
            v[i] = in.nextInt();
        }
        Arrays.sort(v);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (k == 0) {
                break;
            }
            if (v[i] >= 0) {
                ans += v[i];
                k--;
            } else {
                if (i == n - 1 || v[i + 1] >= 0) {
                    ans += v[i];
                    k--;
                }
            }
        }
        System.out.println(ans);
    }
}
