#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a, b, c;
    cin >> n >> a >> b >> c;
    if ((n + 1) % 4 == 0)
        cout << n / 4 * a;
    else if ((n + 2) % 4 == 0) {
        if (n + 2 - 3 * b < 0)
            cout << a;
        else
            cout << min(a, b + (n + 2 - 3 * b) / 4 * a);
    } else if ((n + 3) % 4 == 0)
        cout << b + (n + 3 - 3 * c) / 4 * a;
    else
        cout << min(a + b, b + 2 * c);
    return 0;
}
