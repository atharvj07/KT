#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int res = 0;
    for (int i = 0, j = n - 1; i < j; i++) {
        if (a[i] != a[i + 1]) res++;
        else i++;
    }
    cout << res + 1 << endl;
    return 0;
}
