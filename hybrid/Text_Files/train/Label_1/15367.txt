#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<ll> a(n);
        for (int i = 0; i < n; ++i) { cin >> a[i]; }
        multiset<ll> s(a.begin(), a.end());
        ll mx = *s.rbegin();
        if (k > mx) { cout << "no\n"; } else { cout << "yes\n"; }
    }
}
