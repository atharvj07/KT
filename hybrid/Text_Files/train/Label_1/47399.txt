import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            long p = in.nextLong();
            long q = in.nextLong();
            long x = p;
            while (x % q == 0) {
                x /= q;
            }
            System.out.println(x);
        }
    }
}
