#include <bits/stdc++.h>
#define fastio ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0)
using namespace std;
typedef long long ll;
typedef pair<int,int> pii;

ll fact[210005];

void pre() {
	fact[0] = fact[1] = 1;
	for (ll i = 2; i < 210005; i++) fact[i] = fact[i - 1] * i;
}

int main() {
	fastio;
	pre();
	int n; cin >> n;
	vector<int> a(n), b(n), ord(n), sum(n);
	for (int i = 0; i < n; i++) cin >> a[i];
	for (int i = 0; i < n; i++) cin >> b[i];
	for (int i = 0; i < n; i++) {
		int cnt = 0;
		for (int j = i + 1; j < n; j++) {
			if (a[j] < a[i]) cnt++;
		}
		ord[i] = cnt;
	}
	for (int i = 0; i < n; i++) {
		for (int j = 0; j < i; j++) {
			if (b[j] > b[i]) sum[i] += fact[n - 1 - j];
		}
		sum[i] += ord[b[i]];
	}
	for (int x : sum) cout << x << ' ';
	cout << endl;
	return 0;
}
