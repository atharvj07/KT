#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (abs(a[i]) % 2 == 1) {
      ans += a[i];
    }
  }
  sort(a.begin(), a.end());
  for (int i = 0; i < n; i++) {
    if (abs(a[i]) % 2 == 0) {
      ans += a[i];
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
