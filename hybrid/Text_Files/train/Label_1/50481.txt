import java.io.*;
import java.util.*;

class Main {
    static final long MOD = 998244353L;
    static long[][] dp;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String S = br.readLine();
        int N = S.length();
        dp = new long[N + 2][3];
        dp[0][0] = 1L;
        for (int i = 1; i <= N; i++) {
            int s = Integer.parseInt(String.valueOf(S.charAt(i - 1)));
            for (int j = 0; j < 3; j++) {
                for (int k = 0; k < 3; k++) {
                    if (j == k || (s != 2 && j == 0)) {
                        dp[i][j] += dp[i - 1][k];
                        dp[i][j] %= MOD;
                    }
                }
            }
        }
        System.out.println(dp[N][0]);
    }
}
