import java.util.*;

class Main {
  private static final int[][] moves = new int[][]{{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
  private static int[][] maze, dist;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int h = sc.nextInt();
    int w = sc.nextInt();
    maze = new int[h][w];
    dist = new int[h][w];
    sc.nextLine();
    for (int i = 0; i < h; i++) {
      String line = sc.nextLine();
      for (int j = 0; j < w; j++) {
        maze[i][j] = line.charAt(j) == '.' ? 0 : 1;
        dist[i][j] = -1;
      }
    }
    Queue<int[]> queue = new LinkedList<>();
    for (int i = 0; i < h; i++) {
      for (int j = 0; j < w; j++) {
        if (maze[i][j] == 0) {
          dist[i][j] = 0;
          queue.add(new int[]{i, j});
        }
      }
    }
    while (!queue.isEmpty()) {
      int[] cur = queue.poll();
      int x = cur[0];
      int y = cur[1];
      for (int[] move : moves) {
        int nextX = x + move[0];
        int nextY = y + move[1];
        if (nextX >= 0 && nextX < h && nextY >= 0 && nextY < w && maze[nextX][nextY] == 0 && dist[nextX][nextY] == -1) {
          dist[nextX][nextY] = dist[x][y] + 1;
          queue.add(new int[]{nextX, nextY});
        }
      }
    }
    int max = 0;
    for (int[] row : dist) {
      for (int d : row) {
        max = Math.max(max, d);
      }
    }
    System.out.println(max);
  }
}
