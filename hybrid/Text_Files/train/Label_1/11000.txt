#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9+7;
int main()
{
    ll n,m;
    cin>>n>>m;
    ll a[n],b[m];
    ll ans = 0;
    ll sum1=0,sum2=0;
    for(ll i=0;i<n;i++)
    {
        cin>>a[i];
        sum1+=a[i];
    }
    for(ll j=0;j<m;j++)
    {
        cin>>b[j];
        sum2+=b[j];
    }
    if(sum1!=sum2)
    cout<<0;
    else
    {
        sort(a,a+n);
        sort(b,b+m);
        ll i=n-1,j=m-1;
        while(i>=0 && j>=0)
        {
            if(a[i]==b[j])
            {
                ans++;
                i--;
                j--;
            }
            else if(a[i]>b[j])
            i--;
            else
            j--;
        }
        cout<<ans;
    }
}
