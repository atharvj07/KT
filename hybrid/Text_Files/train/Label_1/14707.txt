#include<cstdio>
#include<vector>
#include<iostream>
#include<set>
using namespace std;
const int maxn=510;
vector<vector<pair<int,int> > >G(maxn);
bool vis[maxn];
int n,deg[maxn],f[maxn][maxn][2],d[maxn],ans=0,last=0;
set<int>link;
void dfs(int u,int fa=0,int dep=0){
    vis[u]=true;
    f[u][1][1]=0;
    for(int i=0;i<G[u].size();++i){
        int v=G[u][i].first,id=G[u][i].second;
        if(v==fa) continue;
        if(vis[v]) ans=max(ans,dep+d[v]-2);
        else{
            dfs(v,u,dep+1);
            for(int j=0;j<=dep;++j)
                for(int k=0;k<=min(j+1,deg[v]);++k){
                    f[u][j+1][1]=max(f[u][j+1][1],f[u][j][0]+f[v][k+1][1]);
                    f[u][j+1][0]=max(f[u][j+1][0],f[u][j][0]+f[v][k+1][0]);
                }
            for(int j=0;j<=dep;++j)
                for(int k=0;k<=min(deg[u]-deg[v]-1,j);++k)
                    f[u][j][0]=max(f[u][j][0],f[u][j-1][1]+f[v][k+1][1]),
                    f[u][j][1]=max(f[u][j][1],f[u][j-1][1]+f[v][k+1][0]);
        }
    }
    d[u]=f[u][0][1]=max(f[u][0][1],f[u][0][0]);
    for(int i=0;i<G[u].size();++i){
        int v=G[u][i].first,id=G[u][i].second;
        if(v==fa) continue;
        if(f[u][0][1]==f[u][0][0]+f[v][deg[v]][1]) link.insert(id);
    }
}
int main(){
    scanf("%d",&n);
    for(int i=1;i<=n;++i) scanf("%d",&deg[i]);
    for(int i=1;i<=n;++i){
        if(deg[i]%2==0) last=i;
        else if(deg[i]!=deg[last]){
            puts("NO");
            return 0;
        }
    }
    if(deg[last]%2){
        deg[last]--;
        deg[last+1]++;
    }
    vector<int>rec;
    for(int i=1;i<=deg[last];++i) rec.push_back(last),rec.push_back(last+1);
    for(int i=2;i<=last;++i){
        for(int j=1;j<=deg[i]/2;++j) rec.push_back(i-1),rec.push_back(i);
        if(deg[i]%2) rec.push_back(i-1),rec.push_back(i),rec.push_back(i+1);
    }
    for(int i=1;i<=last;++i) deg[i]--;
    for(int i=1;i<=n;++i){
        for(int j=1;j<=deg[i];++j){
            if(rec.empty()){
                puts("NO");
                return 0;
            }
            G[i].push_back(make_pair(rec[0],rec[1]));
            G[rec[0]].push_back(make_pair(i,rec[1]));
            G[rec[1]].push_back(make_pair(i,rec[0]));
            rec.erase(rec.begin()),rec.erase(rec.begin());
        }
    }
    dfs(1);
    printf("YES %d\n",ans+1);
    int tot=0;
    for(int i=1;i<=n;++i)
        for(int j=0;j<G[i].size();++j)
            tot+=(G[i][j].first<i)&&link.count(G[i][j].second);
    printf("%d\n",tot);
    for(int i=1;i<=n;++i)
        for(int j=0;j<G[i].size();++j)
            if(G[i][j].first<i) printf("%d %d\n",G[i][j].first,G[i][j].second);
    return 0;
}
