import sys
s='((',')(', '()', '))'
a=[0,0,0,0]
for x in sys.stdin: a[int(x)]+=1
ans=1
while a[1]+a[3]!=0:
    if a[1]==0 and a[3]==0: ans=0; break
    if a[1]>0 and a[3]>0: a[1]-=1; a[3]-=1
    elif a[1]>0 and a[2]>0: a[1]-=1; a[2]-=1
    else: ans=0; break
while a[2]>0:
    if a[2]%2==1: ans=0; break
    a[1]+=a[2]//2
    a[2]=0
while a[1]>0:
    if a[1]%2==1: ans=0; break
    a[2]+=a[1]//2
    a[1]=0
print(ans)
