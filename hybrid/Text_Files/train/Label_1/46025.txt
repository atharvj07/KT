import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextLong();
        }
        if (!verify(n, b)) System.out.println("No");
        else {
            System.out.println("Yes");
            long[] a = new long[n];
            a[0] = b[0];
            for (int i = 1; i < n; i++) {
                a[i] = a[i - 1] ^ b[i];
            }
            Arrays.sort(b);
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (a[j] == b[i]) {
                        System.out.print(b[i] + " ");
                        continue;
                    }
                }
            }
        }
    }
    public static boolean verify(int n, long[] b) {
        long[] a = new long[n];
        a[0] = b[0];
        for (int i = 1; i < n; i++) {
            a[i] = a[i - 1] ^ b[i];
        }
        for (int i = 0; i < n - 1; i++) {
            if (a[i] >= a[i + 1]) return false;
        }
        return true;
    }
}
