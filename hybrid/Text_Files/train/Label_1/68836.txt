n = int(input())

# Since 1^n % 5 = 1, 2^n % 5 = 2^1 = 2, 3^n % 5 = 3^1 = 3, 4^n % 5 = 4^1 = 4 for n % 4 == 0
# and 2^2 % 5 = 4, 3^2 % 5 = 4, 4^2 % 5 = 1, for n % 4 == 1
# and 2^3 % 5 = 3, 3^3 % 5 = 2, 4^3 % 5 = 4, for n % 4 == 2
# and 2^4 % 5 = 1, 3^4 % 5 = 1, 4^4 % 5 = 1, for n % 4 == 3
# Summing these up, we find a repeating pattern for (1^n + 2^n + 3^n + 4^n) mod 5 depending on n mod 4

if n == 0:
    result = 1
else:
    result = [4, 4, 0, 0][n % 4]

print(result)
