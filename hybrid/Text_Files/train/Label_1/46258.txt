import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), t = in.nextInt();
        String str = in.next();
        int hCount = 0;
        for (int i = 0; i < n; i++) {
            if (str.charAt(i) == 'H') hCount++;
        }
        int k = Math.max(0, (2 * hCount - t + 1) / 2);
        System.out.println(k);
    }
}
