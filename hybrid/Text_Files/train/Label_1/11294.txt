#include<iostream>
#include<algorithm>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int x, n, m;
        cin >> x >> n >> m;
        int ans = 0;
        while (x > 0 && n != 0) {
            if (x % 2 == 0) {
                x = x / 2 + 10;
                n--, ans++;
                continue;
            }
            x -= 10;
            m--, ans++;
        }
        while (x > 0 && m != 0) {
            x -= 10;
            m--, ans++;
        }
        if (x <= 0) cout << "YES" << endl;
        else cout << "NO" << endl;

    }
    return 0;
}
