import java.util.Arrays;
import java.util.Scanner;

class Main {
    static int n, res;
    static char[] s;
    static int[] c;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        s = in.next().toCharArray();
        c = new int[n];
        Arrays.fill(c, 1);
        for (int i = 0; i < n; i++) {
            if (i != 0 && s[i] != s[i - 1]) continue;
            int j = i + 1;
            while (j < n && s[j] == s[i]) j++;
            int r = j, l = i;
            int col = 1;
            int prv = 0;
            for (int k = l; k < r; k++) {
                if (c[k] == prv) col++;
                c[k] = col;
                prv = col;
            }
            res = Math.max(res, col);
        }
        System.out.println(res);
        for (int i : c) System.out.print(i + " ");
    }
}
