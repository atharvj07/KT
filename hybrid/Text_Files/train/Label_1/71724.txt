def min_path_cost(n, rs, cs):
    points = sorted(zip(rs, cs))
    ans = 0
    r, c = 1, 1
    for nr, nc in points:
        dr, dc = nr - r, nc - c
        if (r + c) % 2:
            if dc >= dr:
                ans += (dc - dr + 1) // 2
            else:
                ans += (dr - dc) // 2
        else:
            if dc > dr:
                ans += (dc - dr) // 2
            else:
                ans += (dr - dc + 1) // 2
        r, c = nr, nc
    return ans

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    rs = list(map(int, input().strip().split()))
    cs = list(map(int, input().strip().split()))
    print(min_path_cost(n, rs, cs))
