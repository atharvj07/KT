import java.util.Arrays;
import java.util.Scanner;

public class BattleLemmings {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] shields = new int[n];
        for (int i = 0; i < n; i++) {
            shields[i] = scanner.nextInt();
        }

        int maxProtection = n * (n - 1) / 2;
        int[] protection = new int[maxProtection + 1];

        // Calculate protection for each possible number of orders
        for (int orders = 0; orders <= maxProtection; orders++) {
            protection[orders] = calculateProtection(shields, orders);
        }

        // Print the results
        for (int value : protection) {
            System.out.print(value + " ");
        }
    }

    // Function to calculate protection for a given state and number of orders
    private static int calculateProtection(int[] shields, int orders) {
        int n = shields.length;
        int protection = 0;

        // Iterate over all pairs of lemmings
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 2; j < n; j++) {
                // Check if the pair is protected
                if (shields[i] == 0 && shields[j] == 0 && shields[i + 1] == 1) {
                    protection++;
                }
            }
        }

        // Give orders until we reach the desired number of orders
        for (int order = 0; order < orders; order++) {
            boolean shieldGiven = false;

            // Iterate over all lemmings with shields
            for (int i = 0; i < n; i++) {
                if (shields[i] == 1) {
                    // Try to give the shield to the left neighbor
                    if (i > 0 && shields[i - 1] == 0) {
                        shields[i - 1] = 1;
                        shields[i] = 0;
                        shieldGiven = true;
                        break;
                    }

                    // Try to give the shield to the right neighbor
                    if (i < n - 1 && shields[i + 1] == 0) {
                        shields[i + 1] = 1;
                        shields[i] = 0;
                        shieldGiven = true;
                        break;
                    }
                }
            }

            // If no shield was given, we cannot increase protection anymore
            if (!shieldGiven) {
                break;
            }

            // Recalculate protection after giving the order
            protection = calculateProtection(shields, orders - order - 1);
        }

        return protection;
    }
}
