#include <bits/stdc++.h>
using namespace std;
int main() {
  int f;
  string t;
  cin >> f >> t;
  if (f == 12) {
    if (t[0] == '1' && t[1] >= '3')
      t[0] = '0';
    if (t[1] > '5')
      t[1] = '0';
    if (t == "12:00") t = "00:00";
  }
  if (f == 24 && t[0] > '2') t[0] = '0';
  cout << t << endl;
  return 0;
}
