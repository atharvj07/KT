import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
            }
            int max = a[0], min = a[0];
            boolean found = false;
            for (int i = 1; i < n; i++) {
                max = Math.max(max, a[i]);
                min = Math.min(min, a[i]);
                if (max - min >= i + 1) {
                    found = true;
                    System.out.println("YES");
                    System.out.println((i + 1) + " " + n);
                    break;
                }
            }
            if (!found) {
                System.out.println("NO");
            }
        }
    }
}
