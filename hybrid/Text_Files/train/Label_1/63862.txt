import sys
input = sys.stdin.readline

def dfs(u, par, cnt):
    vis[u] = True
    ans = cnt
    for v in adj[u]:
        if v == par:
            continue
        if vis[v]:
            ans = min(ans, cnt - depth[v])
        else:
            depth[v] = cnt + 1
            ans = min(ans, dfs(v, u, cnt + 1))
    return ans

n = int(input())
adj = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)

vis = [False] * (n + 1)
depth = [0] * (n + 1)
depth[1] = 1
ans = dfs(1, 1, 1)
print(n - 1 - ans)
