import java.util.Scanner;
public class Main {
    static long count,sum,total,n,a;
    static int[] arr;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n=sc.nextInt(); a=sc.nextInt();
        arr = new int[(int)n];
        for(int i=0; i<n; i++){
            arr[i]=sc.nextInt(); sum+=arr[i];
        }
        total=(a*n)-sum;
        solve(1,0,0);
        System.out.println(count);
    }
    public static void solve(int ind, long curSum, int cnt){
        if(ind==n){
            if(curSum==total && cnt>0){
                count++;
            }
            return;
        }
        solve(ind+1, curSum+arr[ind], cnt+1);
        solve(ind+1, curSum, cnt);
    }
}
