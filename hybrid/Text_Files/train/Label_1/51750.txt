import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        int n = s.length();
        int[] a = new int[n + 1];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '0') a[i + 1] = a[i] + 1;
            else if (s.charAt(i) == '1') a[i + 1] = a[i] - 1;
            else a[i + 1] = a[i];
        }
        int ans = 0;
        for (int l = 1; l <= n; l++) {
            for (int r = l; r <= n; r++) {
                ans = Math.max(ans, Math.abs(a[r] - a[l - 1]));
            }
        }
        System.out.println(ans);
    }
}
