import java.util.Scanner;

public class RageOfEmpires {

    static class Warrior {
        int personality;

        public Warrior(int personality) {
            this.personality = personality;
        }
    }

    static class Commander {
        int personality;
        int leadership;

        public Commander(int personality, int leadership) {
            this.personality = personality;
            this.leadership = leadership;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();

        // Initialize army and commander list
        Warrior[] army = new Warrior[100000];
        int armySize = 0;
        Commander[] commanders = new Commander[100000];
        int commanderCount = 0;

        for (int i = 0; i < q; i++) {
            int type = scanner.nextInt();
            int p = scanner.nextInt();

            if (type == 1) {
                // Add warrior to army
                army[armySize++] = new Warrior(p);
            } else if (type == 2) {
                // Remove warrior from army
                for (int j = 0; j < armySize; j++) {
                    if (army[j].personality == p) {
                        army[j] = army[armySize - 1];
                        armySize--;
                        break;
                    }
                }
            } else if (type == 3) {
                // Try to hire commander and count respected warriors
                int l = scanner.nextInt();
                Commander commander = new Commander(p, l);
                int respected = 0;
                for (int j = 0; j < armySize; j++) {
                    if ((commander.personality ^ army[j].personality) < commander.leadership) {
                        respected++;
                    }
                }
                System.out.println(respected);
            }
        }
    }
}
