#include <iostream>
const int mod = 1e9 + 7;
using namespace std;

int quick_pow(long long x, long long p) {
	if (p == 0) return 1;
	if (p == 1) return x;
	long long tmp = quick_pow(x, p / 2);
	tmp = tmp * tmp % mod;
	if (p % 2) tmp = tmp * x % mod;
	return tmp;
}

int main() {
	int a, b, k, t;
	cin >> a >> b >> k >> t;
	if (a > b)
		cout << quick_pow(2 * k + 1, 2 * t) << endl;
	else
		cout << 0 << endl;
	return 0;
}
