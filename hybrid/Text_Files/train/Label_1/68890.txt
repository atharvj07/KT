import sys

def gcd(a, b):
    if a == 0:
        return b
    return gcd(b % a, a)

def is_relatively_prime(a, b):
    return gcd(a, b) == 1

n, m = map(int, sys.stdin.readline().split())

if n * (n - 1) / 2 < m:
    print("Impossible")
    exit()

edges = []
for i in range(2, n + 1):
    edges.append((1, i))
m -= n - 1

i = 2
j = n
while m > 0:
    if is_relatively_prime(i, j):
        edges.append((i, j))
        m -= 1
    i += 1
    if i > j:
        i = 2
        j += 1

print("Possible")
for edge in edges:
    print(*edge)
