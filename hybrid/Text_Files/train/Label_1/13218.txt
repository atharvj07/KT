#include <iostream>
#include <string>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    string s(n, 'a');
    for (int i = 0; i < k; i++) { s[i] = 'a' + i; }
    for (int i = k; i < n; i++) { s[i] = s[i - k]; }
    cout << s << endl;
  }
  return 0;
}
