import sys;

def lis(l):
    # find longest non-decreasing subsequence
    n = len(l)
    i = 0; seq = [0] * n
    while i < n:
        cur = l[i]; j = i + 1
        while j < n:
            if cur <= l[j]:
                if seq[j] < seq[i] + 1: seq[j] = seq[i] + 1
            j += 1
        i += 1
    ans = max(seq); return ans

n, m = map(int, input().split()); r = [0] * (n + 1)
for _ in range(m):
    l, rr = map(int, input().split()); r[l - 1] += 1; r[rr] -= 1
r = [i for i in r if i != 0]; ans = lis(r)
print(0 * (len(r) - ans) + 1 * ans)
