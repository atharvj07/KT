import sys
k=int(input())
a=input()
b=input()
c=input()
d=input()
if k==5:
    if a[0]==a[1]==a[2]==b[0]==c[0]==c[1]==c[2]==d[0]: print("YES")
    else: print("NO")
else:
    m=[]
    for i in range(0,10):
        m.append(0)
    m[int(a[0])]+=1
    m[int(a[1])]+=1
    m[int(a[2])]+=1
    m[int(a[3])]+=1
    m[int(b[0])]+=1
    m[int(b[1])]+=1
    m[int(b[2])]+=1
    m[int(b[3])]+=1
    m[int(c[0])]+=1
    m[int(c[1])]+=1
    m[int(c[2])]+=1
    m[int(c[3])]+=1
    m[int(d[0])]+=1
    m[int(d[1])]+=1
    m[int(d[2])]+=1
    m[int(d[3])]+=1
    for i in range(0,10):
        if m[i]==0:
            print("NO")
            sys.exit(0)
    t=[[0]*10 for _ in range(11)]
    t[0][0]=1
    for i in range(0,10):
        for j in range(1,10):
            if i-j>=0: t[j][i]=t[j-1][i-j]
            if i+j<=9: t[j][i]+=t[j-1][i+j]
    ans=0
    for i in range(0,4):
        for j in range(0,4):
            if a[j]=='.': continue
            ans+=t[k][int(a[j])]
    if ans==0: print("NO")
    else: print("YES")
