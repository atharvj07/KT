import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            w[i] = scanner.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            int min = w[i];
            int idx = i;
            for (int j = i + 1; j < n; j++) {
                if (w[j] < min) {
                    min = w[j];
                    idx = j;
                }
            }
            if (idx != i) {
                ans += w[i] + w[idx];
                w[idx] = w[i];
                w[i] = min;
            }
        }
        System.out.println(ans);
    }
}
