#include <iostream>
#include <cmath>

using namespace std;

int main() {
    int a, h, w;
    cin >> a >> h >> w;

    long double dx = (w - a) / 2.0L;
    long double dy = (h - a) / 2.0L;

    if (fmodl(dx, 1) != 0 || fmodl(dy, 1) != 0) {
        cout << -1 << endl;
    } else {
        cout << max(dx, dy) << endl;
    }

    return 0;
}
