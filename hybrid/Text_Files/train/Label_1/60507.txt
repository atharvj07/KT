def mod_pow(a: int, n: int, m: int):
    res = 1
    while n > 0:
        if n % 2 == 1:
            res = res * a % m
        n /= 2
        a = a * a % m
    return res % m

mod = 1000000007
A, B, n, x = map(int, input().split())
ans = mod_pow(A, n, mod) * x + mod_pow(A, n - 1, mod) * B
print(ans % mod)
