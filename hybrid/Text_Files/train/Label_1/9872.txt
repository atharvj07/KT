#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, d;
        cin >> n >> d;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        int m = 0;
        for (int x: a) m = max(m, x);
        if (d < m) cout << "NO" << endl;
        else cout << "YES" << endl;
    }
}
