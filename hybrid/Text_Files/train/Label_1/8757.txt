#include <iostream>
#include <vector>
#include <queue>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> e(n);
    vector<vector<int>> g(n);
    for (int i = 0; i < n; ++i) cin >> e[i];
    for (int i = 0; i < m; ++i) {
        int a, b;
        cin >> a >> b;
        --a, --b;
        g[b].push_back(a);
    }

    vector<int> d(n, -1);
    queue<int> q;
    for (int i = 0; i < n; ++i) {
        if (e[i] == 0) {
            d[i] = 0;
            q.push(i);
        }
    }

    int ans = 0;
    while (!q.empty()) {
        int v = q.front();
        q.pop();
        ++ans;

        for (int u : g[v]) {
            if (d[u] != -1) continue;
            d[u] = d[v] + 1;
            if (e[u] == 0) {
                q.push(u);
            }
        }
    }

    cout << ans << endl;
}
