#include <iostream>
#include <vector>
#include <algorithm>
#include <bitset>
using namespace std;
const int SZ = 1000000;
bitset<SZ> isPrime;
void sieve() {
    isPrime.set();
    isPrime[0] = isPrime[1] = false;
    for (int i = 2; i * i < SZ; ++i) if (isPrime[i])
        for (int j = i * i; j < SZ; j += i)
            isPrime[j] = false;
}
int main() {
    sieve();
    int n; cin >> n;
    vector<int> a(n); for (auto& x : a) cin >> x;
    vector<bool> state(SZ);
    vector<vector<int>> ans; for (int mask = 1; mask < 1<<n; ++mask) {
        vector<int> vec;
        for (int i = 0; i < n; ++i) if (mask & (1<<i)) {
            vec.push_back(a[i]);
            if (vec.size() > 1) {
                int sum = 0;
                for (auto& x : vec) sum += x;
                if (!isPrime[sum]) {
                    state[mask] = true;
                    break;
                }
            }
        }
        if (!state[mask]) ans.push_back(vec);
    }
    for (auto& x : ans) sort(x.begin(), x.end());
    sort(ans.begin(), ans.end(), [](const vector<int>& x, const vector<int>& y) { return x.size() > y.size(); });
    cout << ans[0].size() << '\n';
    for (auto& x : ans[0]) cout << x << ' ';
}
