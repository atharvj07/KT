#include <bits/stdc++.h>
using namespace std;
int n, ans;
char a[1005][1005];
void dfs(int x, int y)
{
    if (x > n || y > n || x < 1 || y < 1 || a[x][y] == 'C')
        return;
    a[x][y] = 'C';
    ans++;
    dfs(x + 1, y);
    dfs(x - 1, y);
    dfs(x, y + 1);
    dfs(x, y - 1);
}
int main()
{
    cin >> n;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            a[i][j] = '.';
    ans = 0;
    dfs(1, 1);
    cout << ans << endl;
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
            cout << a[i][j];
        cout << endl;
    }
    return 0;
}
