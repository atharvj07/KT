import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long c = sc.nextLong();
    long hr = sc.nextLong();
    long hb = sc.nextLong();
    long wr = sc.nextLong();
    long wb = sc.nextLong();
    long ans = 0;
    while (c >= wb && hb > 0) {
      ans += hb;
      c -= wb;
    }
    while (c >= wr && hr > 0) {
      ans += hr;
      c -= wr;
    }
    System.out.println(ans);
  }
}
