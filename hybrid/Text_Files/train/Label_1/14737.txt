#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  sort(v.begin(), v.end());
  int ans = v[k - 1] - v[0];
  for (int i = 1; i <= n - k; i++) {
    ans = min(ans, v[i + k - 1] - v[i]);
  }
  cout << ans << endl;
  return 0;
}
