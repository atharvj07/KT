#include<iostream>
#include<vector>
#define ll long long
using namespace std;
const int N = 3e5 + 10;
ll n, q;
vector<int> adj[N];
typedef pair<int, int> pi;
int size[N], par[N], ans;
bool done[N];

int dfs(int node, int p = -1) {
	done[node] = true;
	size[node] = 1;
	for (int next : adj[node]) {
		if (next == p) { continue; }
		size[node] += dfs(next, node);
	}
	return size[node];
}

void calc(int node, int p = -1) {
	ans = max(ans, size[node]);
	for (int next : adj[node]) {
		if (next == p) { continue; }
		par[next] = node;
		calc(next, node);
	}
}

vector<int> path;
void gepath(int node) {
	if (node == -1) { return; }
	path.push_back(node);
	gepath(par[node]);
}

int main() {
	cin >> q;
	while (q--) {
		cin >> n;
		ans = 0;
		for (int i = 1; i <= n; i++) { adj[i].clear(); done[i] = false; par[i] = -1; }
		for (int i = 0; i < n - 1; i++) {
			int u, v;
			cin >> u >> v;
			adj[u].push_back(v);
			adj[v].push_back(u);
		}
		dfs(1);
		calc(1);
		path.clear();
		gepath(1);
		for (int node : path) {
			size[node] = 0;
		}
		dfs(1);
		calc(1);
		cout << ans << '\n';
	}
	return 0;
}
