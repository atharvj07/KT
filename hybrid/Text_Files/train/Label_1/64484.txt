import sys
input = sys.stdin.readline

T = int(input())
for _ in range(T):
    n = int(input())
    a = [0]
    b = [0]
    for _ in range(n):
        x, y = map(int, input().split())
        a.append(x)
        b.append(y)
    a.append(0)
    b.append(0)
    dp = [0] * (n + 1)
    dp[n] = a[n]
    for i in range(n - 1, 0, -1):
        dp[i] = (n + i - 1) % n + 1
        if a[i] > 1:
            dp[i] = min(dp[i], (n + i - 1) % n + 1 + a[i] - 1)
        if a[i] > b[i + 1] + 1:
            dp[i] = min(dp[i], dp[i + 1] + a[i])
        dp[i] += b[(n + i - 1) % n + 1]
    print(dp[1])
