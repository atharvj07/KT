#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int ans = 0;
  int pos = 1;
  int pos2 = 1000000;
  for (int i = 0; i < n; i++) {
    int cur = a[i];
    ans += min(abs(pos - cur), abs(pos2 - cur));
    if (abs(pos - cur) < abs(pos2 - cur)) {
      pos = cur;
    } else {
      pos2 = cur;
    }
  }
  cout << ans << endl;
  return 0;
}
