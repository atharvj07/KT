T = int(input().strip())
for _ in range(T):
    r, g, b, w = map(int, input().split())
    odd_count = sum([r % 2, g % 2, b % 2, w % 2])
    if odd_count <= 1 or (odd_count == 2 and min(r, g, b) == 0):
        print("Yes")
    else:
        if r > 0 and g > 0 and b > 0:
            r -= 1
            g -= 1
            b -= 1
            w += 3
            odd_count = sum([r % 2, g % 2, b % 2, w % 2])
            if odd_count <= 1:
                print("Yes")
            else:
                print("No")
        else:
            print("No")
