#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<pair<int, int>> teleports(n);
  for (int i = 0; i < n; i++) { cin >> teleports[i].first >> teleports[i].second; }
  for (int i = 1; i < n; i++) {
    if (teleports[i].first < teleports[i - 1].first ||
        teleports[i].second < teleports[i - 1].second) {
      cout << "NO" << endl;
      return 0;
    }
  }
  if (teleports.empty() || teleports.back().second < m) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
  }
  return 0;
}
