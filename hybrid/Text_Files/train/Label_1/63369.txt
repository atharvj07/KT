n, m = map(int, input().split())
board = [list(input()) for _ in range(n)]
dirs = [(0, 1), (1, 0), (0, -1), (-1, 0)]
flag = False
for x in range(n):
    for y in range(m):
        color = board[x][y]
        for dx, dy in dirs:
            nx = (x + dx) % n
            ny = (y + dy) % m
            if board[nx][ny] == color:
                board[nx][ny] = '*'
                if [color] * 4 == [board[(x + dx + offset) % n][(y + dy + offset) % m] for offset in range(4)]:
                    flag = True
                    break
                board[nx][ny] = color
        if flag:
            break
print('Yes' if flag else 'No')
