#include <bits/stdc++.h>

using namespace std;

const int N = 2e5 + 5;

int n, m, k;
vector<int> adj[N];
vector<int> ord;
int deg[N], ordPos[N], ans[N];
bool vis[N];

void dfs(int u){
    vis[u] = true;
    ord.push_back(u);
    ordPos[u] = ord.size() - 1;
    for(int v : adj[u]){
        if(!vis[v]){
            dfs(v);
        }
    }
}

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    cin >> n >> m >> k;
    for(int i = 0;i < m;i++){
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    for(int i = 1;i <= n;i++){
        if(!vis[i]){
            dfs(i);
        }
    }

    reverse(ord.begin(), ord.end());

    for(int u : ord){
        for(int v : adj[u]){
            if(ordPos[u] < ordPos[v]){
                deg[u]++;
            }
        }
        int mx = 0;
        if(deg[u] >= k){
            mx++;
        }
        for(int v : adj[u]){
            if(ordPos[u] < ordPos[v]){
                mx = max(mx, ans[v]);
            }
        }
        ans[u] = mx + 1;
    }

    for(int i = 1;i <= m;i++){
        cout << ans[i] << "\n";
    }

    return 0;
}
