#include <bits/stdc++.h>
using namespace std;
int64_t n, k, ans, sumTaste = 0, sumCal = 0, tot = 0;
vector<pair<int64_t, int64_t>> v;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n >> k;
  for (int64_t i = 0; i < n; i++) {
    int64_t a, b;
    cin >> a >> b;
    tot += b;
    v.push_back({a, b});
  }
  sort(v.begin(), v.end(), [&](pair<int64_t, int64_t> &x,
                                 pair<int64_t, int64_t> &y) {
    return x.first * y.second < y.first * x.second;
  });
  for (auto x : v) {
    int64_t taste = x.first, cal = x.second;
    if (sumCal + cal > tot * k)
      break;
    sumTaste += taste;
    sumCal += cal;
  }
  if (sumCal == 0)
    ans = -1;
  else
    ans = sumTaste;
  cout << ans << '\n';
  return 0;
}
