#include<bits/stdc++.h>
using namespace std;
int n, a[2000], dp0[2000], dp1[2000], dp2[2000];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    int ans = 1;
    for (int i = 1; i <= n; i++) {
        dp0[i] = dp1[i] = dp2[i] = 1;
        if (a[i] == a[i - 1]) dp0[i] = max(dp0[i], dp0[i - 1] + 1);
        if (a[i] == 2 && a[i - 1] == 1) dp1[i] = max(dp1[i], dp0[i - 1] + 1);
        if (a[i] == 1 && a[i - 1] == 2) dp2[i] = max(dp2[i], dp1[i - 1] + 1);
        ans = max(ans, max(dp0[i], max(dp1[i], dp2[i])));
    }
    cout << ans << endl;
    return 0;
}
