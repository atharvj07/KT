#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int n;
vector<pair<long long, long long>> v;
vector<pair<long long, long long>> ans;
int main() {
  cin >> n;
  v.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i].first >> v[i].second;
  }
  for (auto &i : v) {
    long long st = i.first, en = i.first + i.second - 1;
    int ind = lower_bound(ans.begin(), ans.end(), make_pair(st, -1ll)) - ans.begin();
    if (ind != ans.size() && ans[ind].first <= st && st <= ans[ind].second) {
      while (1) {
        ind = lower_bound(ans.begin(), ans.end(), make_pair(st, -1ll)) - ans.begin();
        if (ind == ans.size() || ans[ind].first > st) break;
        st = ans[ind].second + 1;
        en = st + i.second - 1;
      }
    }
    ans.emplace_back(st, en);
    cout << st << " " << en << endl;
  }
  return 0;
}
