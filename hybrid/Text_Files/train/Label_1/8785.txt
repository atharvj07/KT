#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n, m, k;
  cin >> n >> m;
  vector<vector<int>> v(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> v[i][j]; }
  }
  cin >> k;
  while (k--) {
    int l, r;
    cin >> l >> r;
    l--, r--;
    bool ok = false;
    for (int j = 0; j < m; j++) {
      bool flag = true;
      for (int i = l; i < r; i++) {
        if (v[i][j] > v[i + 1][j]) {
          flag = false;
          break;
        }
      }
      if (flag) {
        ok = true;
        break;
      }
    }
    cout << (ok ? "Yes\n" : "No\n");
  }
  return 0;
}
