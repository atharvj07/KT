import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();
        int hh = sc.nextInt();
        int mm = sc.nextInt();
        int ans = 0;
        while (true) {
            String time = String.format("%02d:%02d", hh, mm);
            if (time.contains("7")) {
                break;
            }
            mm -= x;
            if (mm < 0) {
                mm += 60;
                hh--;
                if (hh < 0) {
                    hh += 24;
                }
            }
            ans++;
        }
        System.out.println(ans);
    }
}
