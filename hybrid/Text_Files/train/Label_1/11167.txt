#include<bits/stdc++.h>
using namespace std;

int p[110][110];

int main() {
    int n;
    cin >> n;
    for(int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        for(int j = i; j >= 1; j--) {
            p[i][j] = p[i - 1][j] ^ x;
        }
    }

    int ans = 0;
    for(int i = 1; i <= n; i++) {
        for(int j = i; j >= 1; j--) {
            ans = max(ans, p[i][j]);
        }
    }

    cout << ans << endl;
}
