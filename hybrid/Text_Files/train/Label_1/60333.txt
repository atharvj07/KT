m, s, w = map(int, input().split())
a = {2: 0, 4: 1, 8: 2, 16: 4, 32: 8, 64: 16, 128: 32, 256: 64, 512: 128, 1024: 256}
c = sorted([(int(x), y) for y, x in [x.split() for x in [input() for i in range(int(input()))]]] + [(i, 0) for i in a])
i = m = x = 0
while i < len(c):
    if c[i][1] == 0 and (1 << m & w) > 0:
        x += c[i][0]
        m += 1
        i += 1; s -= 1
        if m == 30: break
    elif ((1 << c[i][1]) & s) > 0:
        x += c[i][0]
        m += a[1 << c[i][1]]
        s -= 1 << c[i][1]
        i += 1
        if s <= 0: break
    else: i += 1
print(m, x)
