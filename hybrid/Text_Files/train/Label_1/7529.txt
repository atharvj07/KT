#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
int main() {
  ll a[4], b[4], n;
  cin >> a[1] >> a[2] >> a[3] >> b[1] >> b[2] >> b[3] >> n;
  vector<ll> aCups(n + 1, 0), bMedals(n + 1, 0);
  for (ll i = 1; i <= n; i++)
    cin >> aCups[i] >> bMedals[i];
  bool can = true;
  for (ll i = 1; i <= n; i++)
    if (aCups[i] > 5 || bMedals[i] > 10) can = false;
  for (ll i = 1; i <= n; i++)
    if ((aCups[i] > 0 && bMedals[i] > 0) || aCups[i] + bMedals[i] > a[1] + a[2] + a[3] +
                                               b[1] + b[2] + b[3])
      can = false;
  cout << (can ? "YES" : "NO") << endl;
  return 0;
}
