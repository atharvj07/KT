import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int k = input.nextInt();
    int[][] powers = new int[n][k];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < k; j++) {
        powers[i][j] = input.nextInt();
      }
    }
    int[] winners = new int[n];
    for (int i = 0; i < n; i++) {
      Set<Integer> winnersSet = new HashSet<>();
      for (int j = 0; j < k; j++) {
        int maxPower = -1;
        int maxPowerIndex = -1;
        for (int l = 0; l < i + 1; l++) {
          if (powers[l][j] > maxPower) {
            maxPower = powers[l][j];
            maxPowerIndex = l;
          }
        }
        winnersSet.add(maxPowerIndex);
      }
      winners[i] = winnersSet.size();
    }
    for (int winner : winners) {
      System.out.println(winner);
    }
  }
}
