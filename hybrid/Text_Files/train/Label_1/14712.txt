#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n;
        cin >> n;
        string str = to_string(n);
        int len = str.length();
        int ans = -1;
        int first_odd = -1;
        for (int i = 0; i < len; i++) {
            if ((str[i] - '0') % 2) {
                ans = 1;
                if (i % 2 == 0) {
                    ans = 0;
                } else if (first_odd == -1) {
                    first_odd = i;
                } else if (i % 2) {
                    ans = 2;
                }
                break;
            }
        }
        if (ans == -1) ans = 0;
        if (first_odd != -1) ans = min(ans, 2);
        cout << ans << "\n";
    }
    return 0;
}
