def max_median_salary(t, test_cases):
    results = []
    for i in range(t):
        n, s = test_cases[i][0]
        salaries = test_cases[i][1]
        salaries.sort(key=lambda x: x[0])
        total_min = sum([x[0] for x in salaries])
        left = s - total_min
        median_pos = n // 2
        median = salaries[median_pos][0]
        for j in range(median_pos, n):
            if j == n - 1:
                next_diff = salaries[j][1] - salaries[j][0]
            else:
                next_diff = salaries[j + 1][0] - salaries[j][0]
            actual_diff = min(left // (j - median_pos + 1), next_diff)
            median += actual_diff
            left -= actual_diff * (j - median_pos + 1)
            if actual_diff < next_diff:
                break
        results.append(median)
    return results

# Read input
t = int(input().strip())
test_cases = []
for _ in range(t):
    n, s = map(int, input().strip().split())
    salaries = [list(map(int, input().strip().split())) for __ in range(n)]
    test_cases.append(((n, s), salaries))

# Get results
results = max_median_salary(t, test_cases)

# Print results
for result in results:
    print(result)
