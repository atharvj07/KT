import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        long n = scan.nextLong();

        long result = 0;
        for (long i = 1; i * i <= n; i++) {
            if (n % i == 0) {
                result++;
                if (i * i != n) {
                    result++;
                }
            }
        }
        System.out.println(result);
    }
}
