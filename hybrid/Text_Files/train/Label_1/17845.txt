#include <bits/stdc++.h>
using namespace std;
vector<vector<int>> g;
int u[200005];
void dfs(int x) {
  cout << x << ' ';
  u[x] = 1;
  for (int i : g[x]) {
    if (!u[i]) dfs(i);
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int n, x, y;
  cin >> n;
  g.resize(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> x >> y;
    g[x].push_back(y);
    g[y].push_back(x);
  }
  for (int i = 1; i <= n; i++)
    if (!u[i]) dfs(i);
  cout << '\n';
  return 0;
}
