n = int(input())
if n % 2 == 0:
    print(n)
    arr = []
    for i in range(2, n + 1, 2):
        arr.append(i)
    for i in range(1, n + 1, 2):
        arr.append(i)
    print(*arr)
else:
    print(n - 1)
    arr = []
    for i in range(3, n + 1, 2):
        arr.append(i)
    arr.append(1)
    for i in range(2, n + 1, 2):
        arr.append(i)
    print(*arr)
