def get_bound(lst, val, left=0, right=None):
  if right is None:
    right = len(lst) - 1
  if left > right:
    return -1
  mid = (left + right) // 2
  if lst[mid] == val:
    while mid > 0 and lst[mid - 1] == val:
      mid -= 1
    while mid < len(lst) - 1 and lst[mid + 1] == val:
      mid += 1
    return mid
  elif lst[mid] > val:
    return get_bound(lst, val, left, mid - 1)
  else:
    return get_bound(lst, val, mid + 1, right)

s = list(input())
n = len(s)
cnt = [0] * 27
for i in s:
  cnt[ord(i) - ord('a')] += 1
times = [i // 2 for i in cnt]

ans = 'Impossible'
for i in range(26):
  if times[i] > 0:
    x = get_bound(s, chr(i + ord('a')))
    times[i] -= 1
    y = get_bound(s, chr(i + ord('a')), x + 1)
    times[i] += 1
    if x == -1 or y == -1:
      continue
    if y - x + 1 > 1 and times[i] > 0:
      ans = '1'
      break
    for j in range(26):
      if i == j:
        continue
      if times[j] > 1:
        ans = '2'
        break
      elif times[j] > 0:
        ans = '1'
        break

print(ans)
