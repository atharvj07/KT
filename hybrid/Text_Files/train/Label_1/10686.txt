#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int& x : a) cin >> x;

    int g = a[0];
    for (int x : a) g = __gcd(g, x);

    cout << max(1, g - k) << '\n';
}
