import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        char[] a = s.toCharArray();
        int n = s.length();
        int[] ans = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            int j = i;
            while (j <= n && a[j - 1] == 'a')
                j++;
            if (j <= n && a[j - 1] == 'b') {
                j--;
                while (j > 0 && a[j - 1] == 'b')
                    j--;
                if (j > 0) {
                    ans[i] = 1;
                    for (int k = (j - 1) / 2; k < i; k++) {
                        char temp = a[k];
                        a[k] = a[i - k - 1];
                        a[i - k - 1] = temp;
                    }
                }
            }
        }
        for (int i = 1; i <= n; i++)
            System.out.println(ans[i]);
    }
}
