#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;

int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll a, b, c;
    cin >> a >> b >> c;
    vector<tuple<ll, ll>> v;
    for (ll i = 0; i < c; i++) {
      ll x, y;
      cin >> x >> y;
      v.push_back(make_tuple(x, y));
    }
    sort(v.begin(), v.end());
    ll ans = 0;
    for (ll i = 0; i < c; i++) {
      for (ll j = i + 1; j < c; j++) {
        ll boy1 = get<0>(v[i]);
        ll girl1 = get<1>(v[i]);
        ll boy2 = get<0>(v[j]);
        ll girl2 = get<1>(v[j]);
        if (boy1 != boy2 && girl1 != girl2) ans++;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
