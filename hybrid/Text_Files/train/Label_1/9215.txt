#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
// const int INF = 1e9;
long long n, c, a[N], b[N], t[N], f[N];
int main() {
  cin >> n >> c;
  for (int i = 0; i < n - 1; i++) cin >> a[i];
  for (int i = 0; i < n - 1; i++) cin >> b[i];
  for (int i = 0; i < n; i++) {
    t[i] = f[i] = 1e18;
  }
  t[0] = f[0] = 0;
  for (int i = 0; i < n; i++) {
    if (i + 1 < n) { f[i + 1] = min(f[i + 1], f[i] + a[i]); }
    if (i + 1 < n) { t[i + 1] = min(t[i + 1], t[i] + b[i] + c); }
    if (i - 1 > 0) { f[i - 1] = min(f[i - 1], f[i] + a[i - 1]); }
    if (i - 1 > 0) { t[i - 1] = min(t[i - 1], t[i] + b[i - 1] + c); }
  }
  for (int i = 0; i < n; i++) { cout << min(f[i], t[i]) << ' '; }
  return 0;
}
