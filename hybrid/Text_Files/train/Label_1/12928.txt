#include <bits/stdc++.h>
using namespace std;

const int mxN = 1005;
int par[mxN], sz[mxN], n;

int getpar(int x) {
    return par[x] == x ? x : par[x] = getpar(par[x]);
}
bool unite(int a, int b) {
    a = getpar(a), b = getpar(b);
    if (a != b) {
        if (sz[a] < sz[b]) swap(a, b);
        par[b] = a, sz[a] += sz[b];
        return true;
    }
    return false;
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) par[i] = i, sz[i] = 1;
    vector<pair<int, int>> v;
    for (int i = 1, a, b; i < n; i++) {
        cin >> a >> b;
        v.push_back({a, b});
    }
    for (auto [a, b] : v) {
        if (!unite(a, b)) {
            cout << "NO" << endl;
            return 0;
        }
    }
    cout << "YES" << '\n';
    for (auto [a, b] : v) cout << a << ' ' << b << '\n';
}
