#include <bits/stdc++.h>

using namespace std;

string ans;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        ans = s;
        for (int k = 1; k <= n; k++) {
            string t = s;
            for (int i = 0; i < n - (k - 1); i++)
                reverse(t.begin() + i, t.begin() + i + k), i++;
            if (t <= ans)
                ans = t;
        }
        cout << ans << endl;
        cout << (int)(ans[0] - s[0]) + 1 << endl;
    }
}
