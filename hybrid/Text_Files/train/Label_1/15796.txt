#include <bits/stdc++.h>
using namespace std;
long long fastpow(long long base, long long p) {
  if (p == 0) return 1;
  if (p == 1) return base;
  long long val = fastpow(base, p / 2);
  val = (val * val) % 10;
  if (p % 2 != 0) val = (val * base) % 10;
  return val;
}
long long fastmod(long long base, long long p, long long mod) {
  if (p == 0) return 1;
  if (p == 1) return base;
  long long val = fastmod(base, p / 2, mod);
  val = (val * val) % mod;
  if (p % 2 != 0) val = (val * base) % mod;
  return val;
}
int main() {
  long long a, b;
  cin >> a >> b;
  cout << fastpow(fastmod(b, b - a, 10), 10) << endl;
}
