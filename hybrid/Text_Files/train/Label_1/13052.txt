#include <algorithm>
#include <cstdio>
#include <vector>

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n;
    scanf("%d", &n);
    std::vector<std::vector<int>> a(n);
    std::vector<bool> used(n, false);
    for (int i = 0; i < n; ++i) {
      int k;
      scanf("%d", &k);
      a[i].resize(k);
      for (int j = 0; j < k; ++j) {
        scanf("%d", &a[i][j]);
        a[i][j]--;
        used[a[i][j]] = true;
      }
    }
    std::vector<bool> free_pr(n, true);
    std::vector<int> pair(n, -1);
    for (int i = 0; i < n; ++i) {
      for (size_t j = 0; j < a[i].size(); ++j) {
        if (free_pr[a[i][j]]) {
          free_pr[a[i][j]] = false;
          pair[i] = a[i][j];
          break;
        }
      }
    }
    int unpaired = 0;
    for (int i = 0; i < n; ++i) {
      if (pair[i] == -1) {
        unpaired++;
      }
    }
    if (unpaired > 0) {
      for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
          if (used[j]) {
            continue;
          }
          a[i].push_back(j + 1);
          std::sort(a[i].begin(), a[i].end());
          bool ok = true;
          for (int k = 0; k < a[i].size(); ++k) {
            if (a[i][k] == pair[i]) {
              ok = false;
              break;
            }
          }
          if (ok) {
            printf("IMPROVE\n");
            printf("%d %d\n", i + 1, j + 1);
            goto next_test;
          }
          a[i].pop_back();
        }
      }
    }
    printf("OPTIMAL\n");
  next_test:;
  }
  return 0;
}
