import sys

def perm(N, K):
    p = [1] * (N + 1)
    ans = 1
    for i in range(2, N + 1):
        p[i] = i
    for i in range(N - K, N + 1):
        ans *= p[i]
        p[i] = p[i - K]
    return ans

N = int(sys.stdin.readline().strip())
ans = 1
for i in range(N):
    p, q = map(int, sys.stdin.readline().strip().split())
    if p != q:
        ans *= perm(2 * N, q - p - 1)
print(ans % 1000000007)
