import java.util.*;

class Main {
  static ArrayList<Integer>[] adj;
  static boolean[][] visitedPair;
  static String ans;
  static int[] color;

  static void dfs(int u, int c) {
    color[u] = c;
    ans = ans + c;
    for (int v : adj[u]) {
      if (!visitedPair[u][v]) {
        visitedPair[u][v] = true;
        visitedPair[v][u] = true;
        if (color[v] == -1) {
          dfs(v, c ^ 1);
        } else if (color[u] == color[v]) {
          ans = "-1";
          return;
        }
      }
    }
  }

  public static void main(String[] args) {
    var sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    visitedPair = new boolean[n + 1][n + 1];
    adj = new ArrayList[n + 1];
    for (int i = 0; i <= n; ++i)
      adj[i] = new ArrayList<>();
    for (int i = 0; i < m; ++i) {
      int u = sc.nextInt();
      int v = sc.nextInt();
      adj[u].add(v);
      adj[v].add(u);
    }

    color = new int[n + 1];
    Arrays.fill(color, -1);
    ans = "";
    for (int i = 1; i <= n; ++i) {
      if (color[i] == -1)
        dfs(i, 0);
    }

    System.out.println(ans);
  }
}
