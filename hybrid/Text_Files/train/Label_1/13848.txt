#include <iostream>
#include <vector>
using namespace std;
int main() {
    string s, t;
    cin >> s >> t;
    int n = s.length(), m = t.length();
    vector<int> pos(m), idx(m);
    for (int i = 0, j = 0; i < n && j < m; i++) {
        if (s[i] == t[j]) pos[j++] = i;
    }
    int ans = n - m;
    for (int i = 0; i < m; i++) {
        for (int j = i + 1; j < m; j++) {
            int d = pos[i] - pos[j];
            if (d >= m - 1) {
                ans = max(ans, pos[i] - idx[i] - (m - 1));
            } else {
                ans = max(ans, d);
            }
        }
        idx[i] = pos[i];
    }
    cout << ans << endl;
    return 0;
}
