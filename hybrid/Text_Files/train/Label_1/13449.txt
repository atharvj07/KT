#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define pb push_back
#define ll long long
const int mxn = 505;
int n, l, r, s, arr[mxn];
int main() {
    int t; cin >> t;
    while(t--) {
        cin >> n >> l >> r >> s;
        ll sum = 0;
        for(int i = 1; i <= r; i++) sum += i;
        for(int i = r + 1; i <= n; i++) sum -= i;
        if(s < sum || (s - sum) % (r - l + 1)) {
            cout << -1 << endl;
            continue;
        }
        int x = (s - sum) / (r - l + 1);
        for(int i = 1; i <= l - 1; i++) {
            arr[i] = i;
            sum += i;
        }
        for(int i = r + 1; i <= n; i++) {
            arr[i] = i;
            sum -= i;
        }
        for(int i = l; i <= r; i++) {
            arr[i] = x + sum;
            sum += x;
        }
        for(int i = 1; i <= n; i++) cout << arr[i] << " ";
        cout << endl;
    }
}
