#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
const int N=2e5+10;
vector<int> g[N];
int ans=0;
bool vis[N];
void dfs(int u){
	vis[u]=1;
	for(int to:g[u]){
		if(!vis[to]){
			dfs(to);
		}
	}
}
int32_t main(){
	ios_base::sync_with_stdio(false); cin.tie(0);
	int n,m; cin>>n>>m;
	for(int e=0;e<m;++e){
		int a,b; cin>>a>>b;
		g[a].push_back(b);
		g[b].push_back(a);
	}
	for(int i=1;i<=n;++i){
		if(!vis[i]){
			dfs(i);
			ans++;
		}
	}
	cout<<ans-1;
	return 0;
}
