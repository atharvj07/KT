import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            long l = sc.nextLong();
            long r = sc.nextLong();
            long d = sc.nextLong();
            long x = (l + d - 1) / d * d;
            if (x <= r) {
                x += d;
            }
            System.out.println(x);
        }
    }
}
