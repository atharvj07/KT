#include <iostream>
#include <limits>
using namespace std;

int main() {
    int n; cin >> n;
    int a_max = 0;
    int a_sum = 0;
    for (int i = 0; i < n; i++) {
        int a; cin >> a;
        a_max = max(a_max, a);
        a_sum += a;
    }
    int k = numeric_limits<int>::max();
    for (int i = a_max + 1; i <= 2 * a_max; i++) {
        if (a_sum * 2 < i * n) {
            k = i;
            break;
        }
    }
    cout << k << endl;
    return 0;
}
