#include <bits/stdc++.h>
using namespace std;
typedef long long LL;

const int N = 110;
int n, ansL[N], ansR[N];
bool vis[N];

void dfs(int x){
  if(vis[x]) return;
  vis[x] = true;
  ansR[x] = ansR[ansL[x]];
  ansL[ansR[x]] = x;
  dfs(ansR[x]);
}

int main(){
  cin >> n;
  for(int i = 1; i <= n; i ++){
    int l, r;
    cin >> l >> r;
    ansL[i] = l, ansR[i] = r;
  }
  for(int i = 1; i <= n; i ++){
    if(!vis[i]) dfs(i);
  }
  for(int i = 1; i <= n; i ++){
    cout << ansL[i] << ' ' << ansR[i] << endl;
  }
  return 0;
}
