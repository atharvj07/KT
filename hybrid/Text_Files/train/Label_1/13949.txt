#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
typedef long double ld;
template <typename T> using ordered_set = tree<T, null_type, less<T>, rb_tree_tag, tree_order_statistics_node_update>;
mt19937 mt(chrono::steady_clock::now().time_since_epoch().count());
#define rand(l, r) uniform_int_distribution<ll>(l, r)(mt)
#define all(a) (a).begin(), (a).end()
#define sz(a) ((int)((a).size()))
#define F first
#define S second
#define PB push_back
#define MP make_pair
#define fori(a, b, c) for (int(a) = (b); (a) < (c); ++(a))
#define fork(a, b, c) for (int(a) = (b); (a) <= (c); ++(a))

const ll MOD = 1e9 + 7;
const int K = 2e5 + 5;
vector<ll> wt(K);
vector<vector<ll>> dp(K, vector<ll>(11, -1));
int n, k;

ll cal(int idx, int rem)
{
  if (idx == n)
  {
    if (rem == 0)
      return 1;
    return 0;
  }
  ll &ans = dp[idx][rem];
  if (ans != -1)
    return ans;
  ans = cal(idx + 1, rem);
  ans += cal(idx + 1, rem - 1) * wt[idx] % MOD;
  ans %= MOD;
  return ans;
}

int main()
{
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> k;
  fori(i, 0, n) cin >> wt[i];
  cout << cal(0, k);
  return 0;
}
