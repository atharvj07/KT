#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> a(n);
        long long sum = 0;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            sum += a[i];
        }
        vector<long long> b(n);
        if (sum % 2 == 0) {
            for (int i = 0; i < n; i++) {
                if (a[i] % 2 == 0) b[i] = a[i] / 2;
                else b[i] = a[i];
            }
        } else {
            for (int i = 0; i < n; i++) {
                if (a[i] % 2 == 1) {
                    b[i] = a[i];
                    sum--;
                    break;
                }
            }
            for (int i = 0; i < n; i++) {
                if (a[i] % 2 == 0) {
                    b[i] = a[i] / 2;
                    sum--;
                    if (sum % 2 == 0) break;
                }
            }
            for (int i = 0; i < n; i++) {
                if (a[i] % 2 == 1) {
                    b[i] = a[i] / 2;
                    sum--;
                    if (sum % 2 == 0) break;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            cout << b[i] << " ";
        }
        cout << endl;
    }
    return 0;
}
