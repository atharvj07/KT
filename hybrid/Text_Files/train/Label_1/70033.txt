s = input().strip()
letters = {}
question_marks = 0

for char in s:
    if char == "?":
        question_marks += 1
    elif char.isalpha():
        letters[char] = 0

# Count of possibilities for each letter (10 for the first, 9 for the second, etc.)
possibilities = 1
for i in range(10, 10 - len(letters), -1):
    possibilities *= i

# If the first character is a letter, it can't be 0, so we handle this separately.
if s[0] in letters:
    possibilities *= 9 // 10

# Calculate the total possibilities including question marks
total_possibilities = possibilities * (10 ** question_marks)

# Adjust for the first character if it's a question mark
if s[0] == "?":
    total_possibilities *= 9 // 10

print(total_possibilities)
