#include<bits/stdc++.h>
using namespace std;
const long long INF = 1e15;
const int MAXN = 1e5+1;
long long T;
string S;
bool solve(long long l,long long r,long long sum){
    if(r-l+1<=0)return sum==T;
    sum-=1LL*(S[l]-'a');
    for(long long i=l;i<r;i++){
        if(solve(l,i,sum)||solve(i+1,r,sum))return 1;
        sum+=1LL*(S[i+1]-'a');
    }
    return 0;
}
int main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cin>>S>>T;
    long long n=S.length();
    long long pref[MAXN];
    pref[0]=0LL;
    for(long long i=0;i<n;i++){
        pref[i+1]=pref[i]+1LL*(S[i]-'a');
    }
    if(solve(0,n-1,pref[n]))cout<<"Yes\n";
    else cout<<"No\n";
    return 0;
}
