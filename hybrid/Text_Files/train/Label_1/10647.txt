#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

const ll M = 998244353;

ll modpow(ll base, ll exp) {
  ll res = 1;
  while (exp > 0) {
    if (exp % 2 == 1) res = (res * base) % M;
    base = (base * base) % M;
    exp /= 2;
  }
  return res;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    vector<int> cnt(n + 1, 0);
    for (int x : a) cnt[x]++;

    ll ans = 0;
    for (int k = 1; k <= n; k++) {
      if (cnt[k] > 1) ans = (ans + modpow(k, cnt[k] - 1)) % M;
    }

    cout << ans << '\n';
  }
  return 0;
}
