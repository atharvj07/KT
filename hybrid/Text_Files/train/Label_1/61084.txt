n = int(input())
s = input()
a = [0] * n
for i in range(n):
    if s[i] == 'R':
        a[i] = a[i - 1] + 1
    elif s[i] == 'L':
        a[i] = a[i - 1] - 1
    else:
        a[i] = a[i - 1]
        if s[i] == '(' or (a[i] and s[i] == ')'):
            a[i] -= 1
        else:
            a[i] += 1
aa = []
ans = [0] * n
cnt = 0
for i in range(n):
    if a[i] == 0:
        cnt += 1
    else:
        if cnt > 0:
            ans[i - cnt] = cnt
            ans[i] = cnt
        cnt = 0
    if a[i] != 0:
        aa.append(s[i])
    if len(aa) > 0 and aa[-1] == ')' and s[i] == ')':
        aa.pop()
    if len(aa) > 0 and aa[-1] == '(' and s[i] == '(':
        aa.pop()
    if len(aa) > 0:
        ans[i] = -1
for i in ans:
    print(i, end=' ')
