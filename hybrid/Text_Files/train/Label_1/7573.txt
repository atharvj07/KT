#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i].first >> v[i].second; }
  for (int i = 0; i < n; i++) {
    int cnt1 = 0, cnt2 = 0;
    for (int j = 0; j < n; j++) {
      if (i != j) {
        if (v[i].first == v[j].first || v[i].first == v[j].second)
          cnt2++;
        if (v[i].second == v[j].first || v[i].second == v[j].second)
          cnt1++;
      }
    }
    cout << n - 1 - cnt1 << " " << n - 1 - cnt2 << endl;
  }
  return 0;
}
