#include<bits/stdc++.h>
using namespace std;

#define sc scanf
#define pr printf
#define ll long long
#define mx 1000000007

int ar[3005];

int main()
{
    int n,i,j,k=0;
    cin>>n;
    for(i=1;i<=n;i++)
    {
        cin>>ar[i];
        if(ar[i]>ar[k]) k=i;
    }

    for(i=1;i<=n;i++)
    {
        if(i==k) continue;
        if(ar[i]==ar[k]) ar[i]++;
    }
    cout<<ar[k]-1<<endl;
}
