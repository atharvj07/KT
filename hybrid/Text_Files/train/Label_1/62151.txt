for _ in range(int(input())):
    n, k, z = map(int, input().split())
    a = list(map(int, input().split()))
    l, r = 0, n - 1
    while k:
        if z > 0 and l < r and a[l] < a[r]:
            l += 1
            z -= 1
        elif r > l and a[l] >= a[r]:
            r -= 1
        else:
            l += 1
        k -= 1
    print(sum(a[l:r + 1]))
