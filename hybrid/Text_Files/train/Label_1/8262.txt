#include <iostream>
using namespace std;
long long gcd(long long a, long long b) {
    if (a == 0) { return b; }
    return gcd(b % a, a);
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        long long g = gcd(n, k);
        if (g != 1) {
            cout << (k / g) << '\n';
        } else {
            cout << (k * (n - 1)) << '\n';
        }
    }
}
