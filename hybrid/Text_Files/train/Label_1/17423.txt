#include<bits/stdc++.h>
using namespace std;
#define LL long long
const int mx=2e5+5;
const LL llinf=(LL)1e18;
LL n,m,ar[mx],cap[mx];
LL ans[mx],typ[mx],q[mx];
void addFlow(LL x, LL am){
    while(x<=n){
        ans[x]+=am;
        LL rem=min(am,cap[x]-ans[x]);
        x++, am-=rem;
    }
}
int main(){
    ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
    cin>>n>>m;
    for(LL i=1;i<=n;i++) cin>>ar[i];
    for(LL i=1;i<=m;i++) cin>>typ[i]>>q[i];
    for(LL i=1;i<=n;i++) cap[i]=ar[i];
    for(LL i=1;i<=n;i++) cap[i]+=cap[i-1];
    for(LL i=1;i<=m;i++){
        if(typ[i]==1) addFlow(q[i],q[i]);
        else{
            LL b=0,e=n,last=0;
            while(b<=e){
                LL mid=(b+e)/2;
                if(cap[mid]>=q[i]) e=mid-1,last=mid;
                else b=mid+1;
            }
            addFlow(q[i],-llinf);
            addFlow(last,-ans[last]);
            addFlow(q[i],ans[last]);
            cout<<ans[q[i]]<<"\n";
        }
    }
}
