n, m, k = map(int, input().split())
b = list(map(int, input().split()))
ans = 0
if 0 in b:
    ans += 1
    b.remove(0)
i = 0
j = 0
while i < n:
    j = i + 1
    while j < n and b[j] - b[i] <= k:
        j += 1
    if j - i > 0:
        ans += b[j - 1] - b[i] + 1
        i = j
    else:
        i += 1
print(ans)
