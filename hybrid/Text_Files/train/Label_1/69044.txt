n, k = map(int, input().split())

if k == 1:
    # If k is 1, then we can simply return a string of alternating '0's and '1's
    s = ''.join(['01'[i % 2] for i in range(n)])
else:
    # Start with a unique substring of length k
    s = '1' * k
    # Fill the rest of the string alternating between '0's and the unique substring
    while len(s) < n:
        s += '0' + '1' * (k - 1)

print(s[:n])
