#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxa = 46;
const int maxb = 46;
const ll mod = 1e9 + 7;
ll dp[maxa][maxa][maxb];
ll t[maxa];
int main() {
  ll n, k, m;
  cin >> n >> k >> m;
  for (int i = 1; i <= k; i++) { cin >> t[i]; }
  dp[0][0][0] = 1;
  for (int i = 0; i < n; i++) {
    for (int taken = 0; taken <= k; taken++) {
      for (int rem = 0; rem <= m; rem++) {
        dp[i + 1][taken + 1][rem + t[taken + 1]] =
            (dp[i + 1][taken + 1][rem + t[taken + 1]] + dp[i][taken][rem]) %
            mod;
        if (rem >= t[taken + 1]) {
          dp[i][taken][rem] = (dp[i][taken][rem] + dp[i][taken][rem - t[taken + 1]]) % mod;
        }
      }
    }
  }
  ll ans = 0;
  for (int i = 0; i <= n; i++) {
    for (int j = 0; j <= k; j++) {
      for (int l = 0; l <= m; l++) {
        ans = (ans + (dp[i][j][l] * (j + min(i, 1))) % mod) % mod;
      }
    }
  }
  cout << ans << '\n';
  return 0;
}
