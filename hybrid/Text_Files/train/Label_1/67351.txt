n = int(input())
a = list(map(int, input().split()))

res = [0] * n
inc, dec = float('-inf'), float('inf')

for i in range(n):
    if a[i] > inc and a[i] < dec:
        if i + 1 < n and a[i] < a[i + 1]:
            inc = a[i]
            res[i] = 0
        else:
            dec = a[i]
            res[i] = 1
    elif a[i] > inc:
        inc = a[i]
        res[i] = 0
    elif a[i] < dec:
        dec = a[i]
        res[i] = 1
    else:
        print('NO')
        break
else:
    print('YES')
    print(*res)
