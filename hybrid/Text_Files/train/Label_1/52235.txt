import java.util.*;

class Main {
    static int n, pop[], cnt, cur;
    static ArrayList<ArrayList<Integer>> adj;
    static boolean vis[];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        pop = new int[n];
        for (int i = 0; i < n; i++)
            pop[i] = sc.nextInt();
        adj = new ArrayList<>();
        for (int i = 0; i < n; i++)
            adj.add(new ArrayList<>());
        vis = new boolean[n];
        for (int i = 1; i < n; i++) {
            int u = sc.nextInt() - 1, v = sc.nextInt() - 1;
            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        dfs(0, -1);
        System.out.println(cnt);
    }

    static void dfs(int node, int par) {
        vis[node] = true;
        if (pop[node] > cur) {
            cnt++;
            cur = pop[node];
        }
        for (int nbr : adj.get(node))
            if (!vis[nbr])
                dfs(nbr, node);
    }
}
