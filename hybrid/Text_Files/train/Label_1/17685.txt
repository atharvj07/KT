#include <bits/stdc++.h>
#include <vector>
using namespace std;

// Returns the minimum cost to transform the given array
// into a power sequence.
long long minCost(vector<long long>& arr, int n) {
  sort(arr.begin(), arr.end());
  long long ans = 0;
  for (int i = 1; i < n; i++) {
    ans += arr[i] - arr[i - 1] - 1;
  }
  return ans;
}

int main() {
  int n;
  cin >> n;
  vector<long long> arr(n);
  for (int i = 0; i < n; i++) { cin >> arr[i]; }
  cout << minCost(arr, n) << endl;
  return 0;
}
