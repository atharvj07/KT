import sys
input = sys.stdin.readline

n, k = map(int, input().split())
id_lst = [int(i) for i in input().split()]

c = 1 if k > 0 else 0
ans = []
for id in id_lst:
    try:
        idx = ans.index(id)
        del ans[idx]
        ans.insert(0, id)
    except ValueError:
        if c < k:
            ans.insert(0, id)
            c += 1
        else:
            ans.pop()
            ans.insert(0, id)

print(len(ans))
print(*ans)
