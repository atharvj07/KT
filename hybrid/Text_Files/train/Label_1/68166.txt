# we can use the character not included in s or t as a base to start our pattern.
# Find the character not in s or t, or if all characters are included, find the one
# that is not the starting character of both s and t.
for c in 'abc':
    if c not in s and c not in t:
        base_char = c
        break
else:
    if s[0] != t[0]:
        base_char = 'abc'.replace(s[0], '').replace(t[0], '')
    else:
        base_char = s[0]

# Strategy to build the string based on the base character
if base_char == 'a':
    if 'ab' not in s+t and 'ba' not in s+t:

    else:
        res = 'a'*n + 'b'*n + 'c'*n

else:
