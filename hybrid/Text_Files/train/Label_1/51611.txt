import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {

       Scanner sc=new Scanner(System.in);

       long n=sc.nextLong();

       long ans=0;

      for(long i=1;i<=n;i++){

         boolean a=false,b=false,c=false;

         long m=i;

         while(m>0){

            if(m%10==7){
               a=true;
            }
            if(m%10==5){
               b=true;
            }

            if(m%10==3){
               c=true;
            }

           m=m/10;
         }

         if(a==true && b==true && c==true){
            ans++;
         }


      }
      System.out.println(ans);
    }
}
