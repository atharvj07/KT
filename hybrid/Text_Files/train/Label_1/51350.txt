import java.util.Scanner;

public class AntonAndArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] b = new int[n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        int[] a = new int[n];
        boolean possible = true;
        for (int i = 0; i < n; i++) {
            int min = c[i] & b[i];
            int max = c[i] | b[i];
            if (min > max) {
                possible = false;
                break;
            }
            a[i] = min;
        }

        if (possible) {
            for (int i : a) {
                System.out.print(i + " ");
            }
        } else {
            System.out.println(-1);
        }
    }
}
