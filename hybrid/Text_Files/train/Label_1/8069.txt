#include <bits/stdc++.h>
using namespace std;
const int dx[] = {0, 0, -1, 1};
const int dy[] = {1, -1, 0, 0};
const int maxN = 55;
char g[maxN][maxN];
int n, m;
int sx, sy, ex, ey;
bool vis[maxN][maxN][1 << 4];
bitset<4> mp[11];
bool valid(int x, int y) { return 1 <= x && x <= n && 1 <= y && y <= m && g[x][y] != '#'; }
int dfs(int x, int y, int msk) {
    if (x == ex && y == ey) return 1;
    if (vis[x][y][msk]) return 0;
    vis[x][y][msk] = 1;
    int ret = 0;
    for (int i = 0; i < 4; i++) {
        if (mp[msk][i]) {
            int nx = x + dx[i], ny = y + dy[i];
            if (valid(nx, ny)) ret += dfs(nx, ny, msk);
        }
    }
    return ret;
}
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> g[i][j];
            if (g[i][j] == 'S') sx = i, sy = j;
            if (g[i][j] == 'E') ex = i, ey = j;
        }
    }
    string s;
    cin >> s;
    int ans = 0;
    for (int msk = 0; msk < (1 << 4); msk++) {
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                if ((msk & (1 << j)) && i != j) mp[msk][i] = 1;
            }
        }
        int sz = s.size();
        for (int i = 0; i < sz; i++) {
            int d = mp[msk][s[i] - '0'];
            int nx = sx + dx[d], ny = sy + dy[d];
            memset(vis, 0, sizeof vis);
            if (valid(nx, ny) && dfs(nx, ny, msk)) {
                ans++;
                break;
            }
        }
    }
    cout << ans << endl;
}
