#include<bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    char c;
    int x;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        cin >> c >> x;
        if (c == '|') {
            ans |= x;
        }
    }
    cout << 1 << endl;
    cout << "| " << ans << endl;
}
