#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        bool ok = true;
        if (s[0] != '8') ok = false;
        for (int i = 1; i < n; i++) {
            if (i == 1) {
                if (s[i] == '0') ok = false;
            } else {
                if (s[i] < '0' || s[i] > '9') ok = false;
            }
        }
        if (n != 11) ok = false;
        if (ok) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
