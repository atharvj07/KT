import math

for j in range(int(input())):
    x = int(input())
    z = int(math.sqrt(x))
    l = 0
    for i in range(1, z + 1):
        if x % i == 0:
            y = x // i
            if math.gcd(i, y) == 1:
                l = i
                break
    if l == 0:
        print(int(z), int(x / z))
    else:
        print(l, x - l)
