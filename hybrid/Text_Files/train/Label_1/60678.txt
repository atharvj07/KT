import sys

n = int(input())
stones = []
for i in range(n):
    a, b, c = map(int, input().split())
    stones.append([a, b, c])

# Merge all possible pairs of stones
candidates = []
for i in range(n):
    for j in range(i + 1, n):
        if stones[i][:] == stones[j][:]:
            continue
        a = stones[i][:]
        b = stones[j][:]
        # Check if there is common side between i and j
        for k in range(3):
            for l in range(3):
                if a[k] == b[l]:
                    # Make common side between a and b
                    a[k], a[(k + 1) % 3], a[(k + 2) % 3] = a[(k + 1) % 3], a[(k + 2) % 3], a[k]
                    b[k], b[(k + 1) % 3], b[(k + 2) % 3] = b[(k + 1) % 3], b[(k + 2) % 3], b[k]
                    common_side = [a[k], a[(k + 1) % 3]]
                    # Try flip one stone
                    if [a[k], b[(k + 1) % 3]] == common_side:
                        a[k], a[(k + 1) % 3], a[(k + 2) % 3] = a[(k + 1) % 3], a[(k + 2) % 3], a[k]
                    elif [b[k], a[(k + 1) % 3]] == common_side:
                        b[k], b[(k + 1) % 3], b[(k + 2) % 3] = b[(k + 1) % 3], b[(k + 2) % 3], b[k]
                    # Check size of new candidate
                    if a[1] == b[1] and a[2] == b[2]:
                        candidates.append([a[0] + b[0], a[1], a[2]])
                    elif a[1] == b[1] and a[0] == b[2]:
                        candidates.append([a[0], a[1], a[2] + b[2]])
                    elif a[1] == b[2] and a[0] == b[0]:
                        candidates.append([a[0], a[1] + b[1], a[2]])
                    elif a[2] == b[2] and a[0] == b[1]:
                        candidates.append([a[0], a[1] + b[1], a[2]])
                    elif a[2] == b[0] and a[0] == b[1]:
                        candidates.append([a[0] + b[2], a[1], a[2]])
                    elif a[2] == b[1] and a[1] == b[0]:
                        candidates.append([a[0] + b[2], a[1], a[2]])

# Sort candidates by volume
candidates.sort(key=lambda x: x[0] * x[1] * x[2], reverse=True)

# Output answer
if candidates[0][0] * candidates[0][1] * candidates[0][2] > stones[0][0] * stones[0][1] * stones[0][2]:
    print(2)
    print(candidates[0][:2])
else:
    print(1)
    print(1)
