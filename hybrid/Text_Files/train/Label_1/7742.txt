#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;
    int i, j, cnt, sum = 0, x = 0;
    char a[n + 5];
    cin >> a;
    for (i = 0; i < n; i++) {
        if (i & 1) {
            if (a[i] == 'r') {
                x++;
            }
        } else {
            if (a[i] == 'b') {
                x++;
            }
        }
    }
    if (x > n - x) {
        cnt = x;
        for (i = 0; i < n; i++) {
            if (i & 1) {
                if (a[i] == 'b') {
                    sum++;
                }
            } else {
                if (a[i] == 'r') {
                    sum++;
                }
            }
        }
    } else {
        cnt = n - x;
        for (i = 0; i < n; i++) {
            if (i & 1) {
                if (a[i] == 'r') {
                    sum++;
                }
            } else {
                if (a[i] == 'b') {
                    sum++;
                }
            }
        }
    }
    cout << sum << endl;

    return 0;
}
