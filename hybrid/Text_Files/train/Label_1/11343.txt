#include <iostream>
int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int a, b, c, m;
    std::cin >> a >> b >> c >> m;
    if (m == 0) {
      if (a <= 1 && b <= 1 && c <= 1)
        std::cout << "YES" << std::endl;
      else
        std::cout << "NO" << std::endl;
    } else {
      if (a >= 1 && b >= 1 && c >= 1 && 2 * m <= a + b + c - 1)
        std::cout << "YES" << std::endl;
      else
        std::cout << "NO" << std::endl;
    }
  }
  return 0;
}
