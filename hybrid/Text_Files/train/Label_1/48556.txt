import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] edges = new int[12];
        for (int i = 0; i < 12; i++) {
            edges[i] = sc.nextInt();
        }
        Arrays.sort(edges);
        if (edges[0] == edges[11]) {
            System.out.println("yes");
        } else if (edges[0] == edges[5] && edges[6] == edges[11]) {
            System.out.println("yes");
        } else if (edges[0] == edges[3] && edges[4] == edges[7] && edges[8] == edges[11]) {
            System.out.println("yes");
        } else if (edges[0] == edges[1] && edges[2] == edges[3] && edges[4] == edges[7] && edges[8] == edges[11]) {
            System.out.println("yes");
        } else if (edges[0] == edges[1] && edges[2] == edges[5] && edges[6] == edges[9] && edges[10] == edges[11]) {
            System.out.println("yes");
        } else {
            System.out.println("no");
        }
    }
}
