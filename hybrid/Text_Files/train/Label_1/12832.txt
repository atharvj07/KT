#include<bits/stdc++.h>
using namespace std;
const int N=105;
int n,m,k,a[N][N],v[N][N];
int main(){
    cin>>n>>m>>k;
    for(int i=1;i<=n;++i)for(int j=1;j<=m;++j)cin>>a[i][j],v[i][j]=a[i][j];
    for(int j=1;j<=m;++j){
        int id=0;
        for(int i=1;i<=n&&id<k;++i)if(a[i][j])id=i;
        for(int i=id;i>=1&&i>=id-k+1;--i)a[i][j]=0;
    }
    int ans=0,cnt=0;
    for(int j=1;j<=m;++j){
        int id=0;
        for(int i=1;i<=n&&id<k;++i)if(a[i][j])id=i;
        for(int i=id;i>=1&&i>=id-k+1;--i){
            if(a[i][j])++ans;
            else if(v[i][j])++cnt;
        }
    }
    cout<<ans<<" "<<cnt<<'\n';
}
