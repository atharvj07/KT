#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int T, n, x, k;
    cin >> T;
    while (T--) {
        cin >> n;
        vector<int> v(n);
        long long sum = 0;
        for (int i = 0; i < n; i++) {
            cin >> x;
            v[i] = x;
            sum += x;
        }
        k = sum / n;
        sort(v.begin(), v.end());
        int ans = 0;
        for (int i = 1; i < n - 1; i++) {
            int l = 0, r = i - 1;
            while (l < r) {
                int avg = (sum - v[i] - v[l] - v[r]) / (n - 3);
                if (avg == k) ans += r - l;
                if (avg >= k) r--;
                else l++;
            }
        }
        cout << ans << '\n';
    }
}
