import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] events = new int[n];
        
        for (int i = 0; i < n; i++) {
            events[i] = scanner.nextInt();
        }
        
        int police = 0;
        int untreated = 0;
        
        for (int i = 0; i < n; i++) {
            if (events[i] == -1) {
                if (police == 0) {
                    untreated++;
                } else {
                    police--;
                }
            } else {
                police += events[i];
            }
        }
        
        System.out.println(untreated);
    }
}
