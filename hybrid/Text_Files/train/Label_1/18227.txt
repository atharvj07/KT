#include <iostream>
#include <unordered_set>
#include <unordered_map>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  unordered_map<int, unordered_set<int>> one, two;
  while (n--) {
    int a, b;
    cin >> a >> b;
    one[a].insert(b);
    one[b].insert(a);
  }
  while (m--) {
    int a, b;
    cin >> a >> b;
    two[a].insert(b);
    two[b].insert(a);
  }
  for (auto p : one) {
    for (auto q : two) {
      if (p.first == q.first) continue;
      if (p.second.size() != q.second.size()) cout << -1 << endl, exit(0);
      if (p.second.size() == 1) cout << *q.second.begin() << endl, exit(0);
      int common = -1;
      for (auto x : q.second) {
        if (p.second.find(x) != p.second.end()) {
          if (common == -1) common = x;
          else common = -1;
        }
      }
      if (common == -1) cout << -1 << endl, exit(0);
      else if (common != -1) cout << 0 << endl, exit(0);
    }
  }
}
