n, a = map(int, input().split())
t = [int(i) for i in input().split()]
cr = [0] * (n + 1)
for i in range(n):
    if t[i]:
        cr[i + 1] = 1
for i in range(2, n + 1):
    cr[i] += cr[i - 1]
ans = 0
ti = t[a - 1]
for i in range(a - 1, -1, -1):
    if ti + cr[i] < ti + cr[a] - cr[i]:
        ti += 1
        ans += 1
for i in range(a, n):
    if ti + cr[n] - cr[i] < ti + cr[a] - cr[i]:
        ti += 1
        ans += 1
print(ans)
