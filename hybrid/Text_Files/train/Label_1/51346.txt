import java.util.Scanner;

public class AntonAndArrays {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] b = new int[n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = (b[i] & c[i]) | (b[i] ^ c[i]);
        }

        for (int i = 0; i < n; i++) {
            if (a[i] < 0 || a[i] > 1000000000) {
                System.out.println(-1);
                return;
            }
        }

        for (int i : a) {
            System.out.print(i + " ");
        }
    }
}
