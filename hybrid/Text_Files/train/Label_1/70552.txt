n = int(input().strip())
a = list(map(int, input().strip()))

for target_sum in range(1, sum(a)):
    partial_sum = 0
    segments = 0
    for digit in a:
        partial_sum += digit
        if partial_sum > target_sum:
            break
        elif partial_sum == target_sum:
            partial_sum = 0
            segments += 1
    if partial_sum == 0 and segments >= 2:
        print("YES")
        break
else:
    print("NO")
