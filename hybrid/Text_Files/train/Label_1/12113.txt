#include <bits/stdc++.h>
using namespace std;
char str[11][11];
int main() {
  int f, i, t;
  cin >> f >> i >> t;
  for (i = 0; i < f; i++) { cin >> str[i]; }
  int ans = 0;
  for (int item = 0; item < i; item++) {
    int cnt = 0;
    for (int friend_ = 0; friend_ < f; friend_++) {
      if (str[friend_][item] == 'Y') { cnt++; }
    }
    if (cnt >= t) { ans++; }
  }
  cout << ans << endl;
  return 0;
}
