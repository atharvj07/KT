#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  vector<bool> lava(n + 1, false);
  for (int i = 0; i < q; i++) {
    int r, c;
    cin >> r >> c;
    lava[c] = !lava[c];
    if (r == 1 && lava[c]) {
      cout << "No" << endl;
    } else if (r == 2 && lava[c - 1]) {
      cout << "No" << endl;
    } else {
      cout << "Yes" << endl;
    }
  }
  return 0;
}
