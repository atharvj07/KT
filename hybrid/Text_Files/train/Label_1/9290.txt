#include <cstdio>
#include <cstring>
const int N = 2100;
char s[N][N];
int n, m, k, ans, f[N], g[N];

int main() {
    scanf("%d%d%d", &n, &m, &k);
    for (int i = 1; i <= n; i++) {
        scanf("%s", s[i] + 1);
        for (int j = 1; j <= m; j++) {
            f[j] = f[j - 1] + (s[i][j] == '*');
        }
        for (int j = 1; j <= m; j++) {
            g[j] = s[i][j] == '.' ? g[j - 1] + 1 : 0;
        }
        for (int j = 1; j <= m; j++) {
            if (g[j] >= k) ans += f[j] - f[j - k];
        }
    }
    for (int j = 1; j <= m; j++) {
        int cn = 0;
        for (int i = 1; i <= n; i++) {
            cn = s[i][j] == '.' ? cn + 1 : 0;
            if (cn >= k) { ans += n - i + 1; }
        }
    }
    printf("%d\n", ans);
    return 0;
}
