import sys
input = sys.stdin.readline

INF = 2e9
def solve():
    n, a, b, da, db = map(int, input().split())
    adj = [[] for _ in range(n+1)]
    for _ in range(n-1):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)

    # dp[i][j] = the minimum number of moves to get from vertex i to vertex j
    dp = [[INF] * (n+1) for _ in range(n+1)]
    dp[a][a] = 0

    # bfs to calculate the minimum number of moves to get from vertex a to all other vertices
    q = [a]
    while q:
        cur = q.pop(0)
        for nxt in adj[cur]:
            if dp[a][nxt] > dp[a][cur] + 1:
                dp[a][nxt] = dp[a][cur] + 1
                q.append(nxt)

    # dp2[i][j] = the minimum number of moves to get from vertex i to vertex j, given that the other player is at vertex j
    dp2 = [[INF] * (n+1) for _ in range(n+1)]
    dp2[b][b] = 0

    # bfs to calculate the minimum number of moves to get from vertex b to all other vertices, given that the other player is at vertex b
    q = [b]
    while q:
        cur = q.pop(0)
        for nxt in adj[cur]:
            if dp2[b][nxt] > dp2[b][cur] + 1:
                dp2[b][nxt] = dp2[b][cur] + 1
                q.append(nxt)

    # if there is a vertex that Alice can reach in fewer moves than Bob, then Alice wins
    for i in range(1, n+1):
        if dp[a][i] <= da and dp2[b][i] > db:
            return "Alice"

    # otherwise, Bob wins
    return "Bob"


t = int(input())
for _ in range(t):
    print(solve())
