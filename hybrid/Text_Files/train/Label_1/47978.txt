import java.util.Scanner;

public class JabberIDChecker {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String jabberID = scanner.nextLine();

        if (isValidJabberID(jabberID)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isValidJabberID(String jabberID) {
        String[] parts = jabberID.split("@");
        if (parts.length != 2) {
            return false;
        }

        String username = parts[0];
        String hostname = parts[1];

        if (!isValidUsername(username) || !isValidHostname(hostname)) {
            return false;
        }

        if (jabberID.contains("[/") && !jabberID.endsWith("]")) {
            return false;
        }

        if (jabberID.contains("[") && !jabberID.contains("]")) {
            return false;
        }

        String resource = jabberID.substring(jabberID.indexOf("[") + 1, jabberID.indexOf("]"));
        if (!isValidResource(resource)) {
            return false;
        }

        return true;
    }

    private static boolean isValidUsername(String username) {
        return username.matches("^[a-zA-Z0-9_]+$") && username.length() >= 1 && username.length() <= 16;
    }

    private static boolean isValidHostname(String hostname) {
        String[] words = hostname.split("\\.");
        for (String word : words) {
            if (!isValidUsername(word)) {
                return false;
            }
        }
        return hostname.length() >= 1 && hostname.length() <= 32;
    }

    private static boolean isValidResource(String resource) {
        return resource.matches("^[a-zA-Z0-9_]+$") && resource.length() >= 1 && resource.length() <= 16;
    }
}
