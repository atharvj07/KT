import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
            if ((int) Math.sqrt(arr[i]) * (int) Math.sqrt(arr[i]) != arr[i])
                ans = Math.max(ans, arr[i]);
        }
        System.out.println(ans);
    }
}
