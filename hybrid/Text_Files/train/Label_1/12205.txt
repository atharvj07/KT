#include <cstdio>
#include <set>

int main() {
    int n, m;
    char c[101][101] = {0};

    scanf("%d%d", &n, &m);
    for (int i = 0; i < n; ++i) {
        scanf("%s", c[i]);
    }
    std::set<char> color;
    int h = n / 3;
    for (int i = 0; i < n; i += h) {
        for (int j = 0; j < m; ++j) {
            color.insert(c[i][j]);
        }
    }
    if (color.size() == 3) {
        puts("YES");
    } else {
        puts("NO");
    }
    return 0;
}
