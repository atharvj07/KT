# This is an implementation of the solution to the coding problem: How Many Days to Merge All Startups

import sys


def modinv(a, m):
    """return the modular inverse of a mod m"""
    x, y, gcd = extended_gcd(a, m)
    if gcd != 1:
        raise ValueError
    return (x % m + m) % m


def extended_gcd(a, b):
    """return (gcd, x, y) such that a*x + b*y = gcd"""
    if b == 0:
        return 1, 0, a
    x1, y1, gcd = extended_gcd(b, a % b)
    x, y = y1, x1 - (a // b) * y1
    return gcd, x, y


def matrix_mul(a, b, mod):
    """multiply two 2x2 matrices"""
    return [[(a[0][0]*b[0][0] + a[0][1]*b[1][0]) % mod, (a[0][0]*b[0][1] + a[0][1]*b[1][1]) % mod],
            [(a[1][0]*b[0][0] + a[1][1]*b[1][0]) % mod, (a[1][0]*b[0][1] + a[1][1]*b[1][1]) % mod]]


def matrix_pow(a, p, mod):
    """raise a 2x2 matrix to the power of p"""
    if p == 0:
        return [[1, 0], [0, 1]]
    if p == 1:
        return a
    if p % 2 == 0:
        return matrix_pow(matrix_mul(a, a, mod), p//2, mod)
    return matrix_mul(a, matrix_pow(matrix_mul(a, a, mod), p//2, mod), mod)


def solve(n, a):
    """return the expected number of days"""
    p = n-1
    q = n*(n-1)

    gcd = 0
    for i in range(n):
        if a[i] == -1:
            continue
        for j in range(n):
            if a[j] == -1:
                continue
            if i == j:
                continue
            if a[i] == a[j]:
                p -= 1
                q -= 1
            else:
                q -= 1
    return (p*modinv(q, 10**9 + 7)) % (10**9 + 7)


n = int(input())
a = list(map(int, sys.stdin.readline().split()))
print(solve(n, a))
