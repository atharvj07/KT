from functools import reduce
MOD = 10**9+7
def add(x, y):
    return (x+y)%MOD
def mul(x, y):
    return (x*y)%MOD
def solve():
    n, m, k = map(int, input().split())
    grid = [list(map(int, input().split())) for _ in range(n)]
    dp = [[[0 for l in range(2)] for j in range(m)] for i in range(n)]
    # 0: at some point in the path, xor is different from k
    # 1: at every point in the path, xor is equal to k
    dp[0][0][1] = 1
    for i in range(n):
        for j in range(m):
            x = grid[i][j]
            for l in range(2):
                # If the current cell is not on the path
                dp[i][j][l] = dp[i][j][0]
                # Otherwise
                if l==1 and x^k==0:
                    dp[i][j][l] = add(dp[i][j][l], dp[i][j][1])
                if i+1<n:
                    # Move down
                    dp[i][j][l] = add(dp[i][j][l], dp[i+1][j][l])
                    # Move down and update xor
                    if l==1 and x^k==0:
                        dp[i][j][l] = add(dp[i][j][l], dp[i+1][j][0])
                if j+1<m:
                    # Move right
                    dp[i][j][l] = add(dp[i][j][l], dp[i][j+1][l])
                    # Move right and update xor
                    if l==1 and x^k==0:
                        dp[i][j][l] = add(dp[i][j][l], dp[i][j+1][0])
    return dp[n-1][m-1][1]
print(solve())
