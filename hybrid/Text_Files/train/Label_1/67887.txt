import sys
input = sys.stdin.readline
for i in range(int(input())):
    n = int(input())
    a = [int(j) for j in input().split()]
    b = []
    for x in a:
        flag = False
        for y in b:
            if x % y == 0:
                flag = True
                break
        if flag == False:
            b.append(x)
    if len(b) == len(a):
        print(0)
        continue
    print(b[0])
