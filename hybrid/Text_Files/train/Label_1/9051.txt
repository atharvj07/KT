#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 998244353
const int mxN=1e5+10;
int n,k,len;
ll dp[mxN][2][2];
int a[mxN];
void input(){
  cin>>n>>k>>len;
  for(int i=1;i<=n;i++) cin>>a[i];
}
ll solve(int i, int prv, int prv1, int prv2){
  if(i==n+1){
    if(prv2) return 0;
    return 1LL;
  }
  ll &ret=dp[i][prv][prv1];
  if(ret!=-1) return ret;
  ret=0;
  if(a[i]==-1){
    for(int j=1;j<=k;j++){
      if((j==prv && prv1) || (j==prv2)) continue;
      ret+=solve(i+1,j,prv,prv1);
      ret%=mod;
    }
  }
  else if(a[i]!=prv && a[i]!=prv2){
    ret+=solve(i+1,a[i],prv,prv1);
    ret%=mod;
  }
  return ret;
}
int main(){
  input();
  memset(dp,-1,sizeof(dp));
  cout<<solve(1,0,0,0)%mod<<endl;
  return 0;
}
