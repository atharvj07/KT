#include <iostream>
#include <unordered_map>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        string s;
        cin >> s;
        unordered_map<char, int> m;
        for (char x : s)
            m[x]++;
        int ans = 0;
        for (auto p : m)
            ans = max(ans, p.second / 2);
        cout << ans << '\n';
    }
}
