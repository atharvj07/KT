n, s = list(map(int, input().split())), list(map(int, input().split()))
q = int(input())
for l, r in zip(*[iter(map(int, input().split())) for _ in range(q)]):
    l, r = l - 1, r - 1
    candies = 0
    while l < r:
        candies += s[l // 2 + (l % 2)] + s[r // 2 + (r % 2)] >= 10
        s = s[:(l + r + 2) // 2] + [sum(s[i:i + 2]) % 10 for i in range(0, len(s), 2)]
        l, r = l // 2, r // 2
    print(candies)
