t = int(input())
for _ in range(t):
    n = int(input())
    prices = list(map(int, input().split()))
    min_future_price = prices[-1]
    bad_days = 0
    for price in reversed(prices[:-1]):
        if price > min_future_price:
            bad_days += 1
        min_future_price = min(min_future_price, price)
    print(bad_days)
