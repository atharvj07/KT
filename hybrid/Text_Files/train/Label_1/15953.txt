#include <bits/stdc++.h>

int a[25];
bool vis[25];

int main() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &a[i]);
    vis[i] = 1;
  }
  std::sort(a, a + n);
  int cur = a[n - 1];
  std::vector<int> ans;
  for (int i = n - 1; i >= 0; --i) {
    if (vis[i] == 0) continue;
    for (int j = 0; j < n; ++j) {
      if (vis[j] == 0 || a[j] == cur) continue;
      ans.push_back(a[j]);
      vis[j] = 0;
      break;
    }
    ans.push_back(cur);
    vis[i] = 0;
    cur = a[i];
  }
  for (auto it : ans) {
    printf("%d ", it);
  }
  puts("");
  return 0;
}
