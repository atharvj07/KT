import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int h = in.nextInt();
        int w = in.nextInt();
        int sum = 0;
        for (int i = 0; i < n; i++) {
            int x = in.nextInt();
            if (i % 2 == 0) sum += h * x;
            else sum -= h * x;
        }
        System.out.println(Math.abs(sum));
    }
}
