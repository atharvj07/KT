#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, d;
    cin >> n >> d;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    if (*max_element(begin(a), end(a)) > d)
      cout << "NO" << endl;
    else
      cout << "YES" << endl;
  }
  return 0;
}
