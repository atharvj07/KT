t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    mods = set()
    operations = 0
    for i in range(n):
        mod = a[i] % n
        while mod in mods:
            a[i] += n - mod
            mod = a[i] % n
            operations += 1
            if a[i] > n*n:
                print(-1)
                break
        else:
            mods.add(mod)
            continue
        break
    else:
        print(operations)
