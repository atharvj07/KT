#include <iostream>
#include <vector>
#include <stack>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> open(n), close(n);
  stack<int> st;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(') {
      st.push(i);
    } else {
      if (!st.empty()) {
        open[i] = st.top();
        close[st.top()] = i;
        st.pop();
      }
    }
  }
  if (st.size() > 0) {
    cout << "-1" << endl;
    return 0;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(') {
      ans += i - open[i];
    } else {
      ans += close[i] - i;
    }
  }
  cout << ans << endl;
  return 0;
}
