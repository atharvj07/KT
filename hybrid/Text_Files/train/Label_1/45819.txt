import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n1 = in.nextInt();
        int n2 = in.nextInt();
        int k1 = in.nextInt();
        int k2 = in.nextInt();
        System.out.println(solve(n1, n2, k1, k2));
    }

    private static long solve(int n1, int n2, int k1, int k2) {
        long[][] dp = new long[n1 + 1][n2 + 1];
        dp[0][0] = 1;
        for (int i = 0; i <= n1; i++) {
            for (int j = 0; j <= n2; j++) {
                if (i > 0) {
                    dp[i][j] += dp[i - 1][j] * (k1 + 1);
                }
                if (j > 0) {
                    dp[i][j] += dp[i][j - 1] * (k2 + 1);
                }
            }
        }
        return dp[n1][n2] % 100000000;
    }
}
