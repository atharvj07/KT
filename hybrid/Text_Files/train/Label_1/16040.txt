#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll inf = 1e15;
struct point {
  ll x, y, t;
  bool operator<(point p) const { return t < p.t; }
};
int main() {
  int t;
  cin >> t;
  while (t--) {
    ll n, k;
    cin >> n >> k;
    priority_queue<point> p;
    for (ll i = 0; i < n; i++) {
      point x;
      cin >> x.x >> x.y >> x.t;
      p.push(x);
    }
    ll ans = p.top().t;
    while (!p.empty()) {
      point top = p.top();
      p.pop();
      if (top.t > ans) break;
      ans = min(ans, top.t);
      ans = min(ans, top.t + min(abs(top.x + k), abs(top.x - k)) +
                           min(abs(top.y + k), abs(top.y - k)));
    }
    cout << ans << '\n';
  }
}
