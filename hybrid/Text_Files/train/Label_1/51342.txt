import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            if (n == 0) break;
            int[][] directions = new int[n][2];
            int[] counts = new int[4];
            for (int i = 1; i < n; i++) {
                directions[i][0] = scanner.nextInt();
                directions[i][1] = scanner.nextInt();
                counts[directions[i][1]]++;
            }
            int width = 1, height = 1;
            int end = 0;
            int count = 0;
            for (int i = 1; i < n; i++) {
                if (directions[i][0] == end) {
                    count++;
                } else {
                    count = 1;
                    end = directions[i][0];
                }
                if (directions[i][1] == 0) {
                    width += count;
                }
                if (directions[i][1] == 1) {
                    height += count;
                }
            }
            System.out.println(width + " " + height);
        }
    }
}
