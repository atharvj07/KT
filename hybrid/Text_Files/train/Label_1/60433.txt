n = int(input())
a = [*map(int,input().split())]
adj = [[]]
for _ in range(n-1):
    adj.append(list(map(int,input().split())))
par = [0]*n
def dfs(node = 1, p = 0, lev = 0):
    par[node] = p
    for i in adj[node]:
        if i==p: continue
        dfs(i,node,1-lev)
dfs()
odd = [0]*n
for i in range(n-1,0,-1):
    odd[par[i]] += (odd[i] + a[i]%2)%2
pref = [0]*n
for i in range(n-1,-1,-1):
    pref[par[i]] += pref[i] + a[i]
ans = 0
for i in range(1,n):
    if odd[i]%2 == 0 and a[i]%2==0: ans+=1
    if odd[i]%2 == 0 and a[par[i]]%2==1: ans+=1
    if odd[i]%2==1 and a[i]%2 == 1 and odd[par[i]]%2==0 and pref[i]%2==0: ans+=1
print(ans)
