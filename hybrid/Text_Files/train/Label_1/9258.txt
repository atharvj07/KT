#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < n; i++)

const int mod = 1e9 + 7;
const int maxN = 1e3 + 10;
int dp[maxN][maxN][11];

void kmp(const string& s, int pos, string& pat, vector<int>& lps) {
  if (pos == pat.size()) {
    lps.push_back(pos - 1);
    return;
  }
  int j = lps[pos - 1];
  while (j > 0 && pat[pos] != pat[j]) j = lps[j - 1];
  (j == 0 && pat[pos] == pat[j]) ? lps.push_back(1) : lps.push_back(j);
  kmp(s, pos + 1, pat, lps);
}

int solve(string& s, string& pat, int l, int r) {
  if (dp[l][r][pat.size()] != -1) return dp[l][r][pat.size()];
  dp[l][r][pat.size()] = 0;
  if (r - l + 1 < pat.size()) return 0;
  int j = dp[l + 1][r][pat.size() - 1];
  while (j > 0 && pat[pat.size() - 1] != s[r]) j = dp[l + 1][r][j - 1];
  if (j == 0 && s[r] == pat[pat.size() - 1]) {
    dp[l][r][pat.size()] = 1 + solve(s, pat, l + 1, r - (pat.size() - 1));
  }
  return dp[l][r][pat.size()] %= mod;
}

int main() {
  int n, m, q;
  cin >> n >> m >> q;
  string s, pat;
  cin >> s >> pat;
  memset(dp, -1, sizeof dp);
  vector<int> lps(m);
  kmp(s, 0, pat, lps);
  while (q--) {
    int l, r;
    cin >> l >> r;
    cout << solve(s, pat, l - 1, r - 1) << '\n';
  }
  return 0;
}
