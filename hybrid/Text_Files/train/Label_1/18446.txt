#include <iostream>
#include <vector>
int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, q;
    std::string s;
    std::cin >> n >> q >> s;
    std::vector<int> sum(n + 1, 0);
    for (int i = 0; i < n; i++) sum[i + 1] = sum[i] + (s[i] == '+' ? 1 : -1);
    while (q--) {
      int l, r;
      std::cin >> l >> r;
      std::cout << ((sum[r] - sum[l - 1]) % 2 == 0
                        ? std::abs((sum[r] - sum[l - 1]) / 2)
                        : (sum[r] - sum[l - 1]) / 2 + 1)
                << '\n';
    }
  }
  return 0;
}
