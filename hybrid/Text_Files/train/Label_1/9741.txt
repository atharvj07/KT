#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  long long a[n + 1];
  for (int i = 1; i <= n; ++i) { cin >> a[i]; }
  sort(a + 1, a + 1 + n);
  long long ans = 0, sum = a[n];
  for (int i = n - 1; i >= 1; --i) {
    ans += max(sum - a[i], 0ll);
    sum += a[i];
  }
  cout << ans << endl;
}
