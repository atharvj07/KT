def solve():
    n = int(input())
    ls = []
    for _ in range(n):
        ls.append(input())
    if n % 2 != 0:
        for i in range(n):
            if ls[i][0] == '?':
                ls[i] = 'B' if ls[(i + 1) % n][1] == 'W' else 'W'
    else:
        x = False
        y = False
        for i in range(n):
            if ls[i][0] == '?':
                if ls[i][1] == 'B' or (not y and ls[(i + 1) % n][1] == 'W'):
                    ls[i] = 'B'
                    y = True
                else:
                    ls[i] = 'W'
                    x = True
        if x and y:
            return 0
    return 1
print(solve())
