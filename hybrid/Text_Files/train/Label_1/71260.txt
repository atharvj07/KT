import sys

n,m = map(int,input().split())
a = [[]]
for i in range(n):
	a.append([0]*n)

for i in range(m):
	x,y = map(int,input().split())
	a[x][y] = a[y][x] = 1

mn = sys.maxsize
in3 = [0]*n
for i in range(1,n+1):
	for j in range(i+1,n+1):
		if a[j][i] == 0:
			continue
		for k in range(j+1,n+1):
			if a[j][k] == 0 or a[i][k] == 0:
				continue
			mn = min(mn, i + j + k - a[i][j] - a[j][k] - a[i][k] )

if mn == sys.maxsize:
	print(-1)
else:
	print(mn)
