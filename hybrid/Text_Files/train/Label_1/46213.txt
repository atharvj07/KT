import java.util.*;

public class Main {
    static class Pair {
        int a, b;

        Pair(int a, int b) {
            this.a = a;
            this.b = b;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] arr = new int[n + 1][2];
        for (int i = 1; i <= n; i++) {
            arr[i][0] = sc.nextInt();
            arr[i][1] = sc.nextInt();
        }
        int count = 0, boyGirl = 0;
        List<Pair> pairs = new ArrayList<>();
        boolean[] vis = new boolean[n + 1];
        for (int i = 1; i <= n; i++) {
            if (vis[i]) continue;
            int j = arr[i][0];
            if (vis[j]) continue;
            if (arr[i][1] != arr[j][1]) boyGirl++;
            pairs.add(new Pair(i, j));
            vis[i] = vis[j] = true;
            count++;
        }
        System.out.println(count + " " + boyGirl);
        for (Pair p : pairs) {
            System.out.println(p.a + " " + p.b);
        }
    }
}
