#include <iostream>

const int modulus = 1e9 + 7;

int main() {
    long long a, b, n;
    std::cin >> a >> b >> n;
    long long x = b;
    long long y = (a + b) % modulus;
    n -= 2;
    if (n <= 0)
        std::cout << x << '\n';
    else if (n & 1)
        std::cout << ((x - y * ((n - 1) / 2 % modulus)) % modulus + modulus) % modulus << '\n';
    else
        std::cout << y << '\n';
    return 0;
}
