#include <cstdio>
#include <cstring>
#include <vector>
#include <map>

typedef unsigned long long ull;

int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  std::map<ull, int> times;
  std::vector<int> msgs;
  char buf[80];
  ull t, now = 0, prev = 0;
  msgs.push_back(0);
  while (fgets(buf, 80, stdin)) {
    sscanf(buf, "%llu:%*s", &t);
    msgs.push_back(0);
    auto it = times.find(t - n);
    if (it != times.end()) {
      msgs[msgs.size() - 1] = msgs[msgs.size() - 2] - it->second;
      times.erase(it);
    } else {
      msgs[msgs.size() - 1] = msgs[msgs.size() - 2];
    }
    times[t] = msgs[msgs.size() - 1];
    msgs[msgs.size() - 1] += 1;
    if (msgs.back() >= m) {
      now = t;
      break;
    }
  }
  if (now == 0) {
    puts("-1");
  } else {
    now -= n;
    ull sec = now % 60;
    ull min = (now % 3600) / 60;
    ull hr = (now % 86400) / 3600;
    ull day = now / 86400;
    printf("%04d-%02d-%02d %02d:%02d:%02d\n", 2012, day + 1, day + 1, hr, min, sec);
  }
  return 0;
}
