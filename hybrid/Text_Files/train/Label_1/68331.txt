s, b = map(int, input().split())
a = list(map(int, input().split()))
d = []
g = []
for _ in range(b):
    di, gi = map(int, input().split())
    d.append(di)
    g.append(gi)
d.sort()
g.sort(reverse=True)
ans = [0] * s
for i in range(s):
    j = 0
    while j < b and a[i] >= d[j]:
        ans[i] += g[j]
        j += 1
print(*ans)
