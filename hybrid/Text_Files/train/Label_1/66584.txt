n,m,k=[int(x) for x in input().split()]
q=[{'x': 0, 'y': 0, 'd': 0}]
a=[[0]*m for i in range(n)]
for i in range(n):
	a[i]=list(input())
	for j in range(m):
		if a[i][j]=='.':
			q.append({'x': i, 'y': j, 'd': 0})
while k>0 and q:
	pp=q[0]
	q=q[1:]
	if a[pp['x']][pp['y']]=='.':
		k-=1
		a[pp['x']][pp['y']]='X'
	for i,j in [[0,1],[0,-1],[1,0],[-1,0]]:
		nx=pp['x']+i
		ny=pp['y']+j
		if 0<=nx<n and 0<=ny<m and a[nx][ny]=='.':
			q.append({'x': nx, 'y': ny, 'd': 0})
for i in a:
	print(''.join(i))
