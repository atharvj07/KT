#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int x;
        cin>>x;
        int a[10]={0};
        int ans=0;
        int cnt=0;
        while(x>=10)
        {
            int temp=x%10;
            if(a[temp]==0)
            {
                a[temp]=1;
                ans+=temp;
                cnt++;
            }
            x/=10;
        }
        if(x!=0)
        {
            if(a[x]==0)
            {
                ans+=x;
                cnt++;
            }
        }
        if(cnt>10 || x+ans!=x)
        cout<<-1<<endl;
        else
        cout<<ans<<endl;
    }
    return 0;
}
