#include <iostream>
#include <vector>
using namespace std;
const int mo = 1000000007;
int main() {
    string s;
    int n;
    cin >> s >> n;
    vector<pair<int, string>> p(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> p[i].first >> p[i].second;
        p[i].first -= '0';
    }
    int m = s.size();
    vector<int> np(10);
    for (int i = 0; i < 10; ++i) { np[i] = i; }
    for (int j = 0, k = 0; j < 10; ++j) {
        if (p[k].first != j) { continue; }
        for (int i = 0; i < m; ++i) {
            if (s[i] == '0' + j) {
                s.erase(i, 1);
                i -= 1;
                ++m;
            }
        }
        if (p[k].second.size()) {
            string t(p[k].second);
            int mt = t.size();
            for (int i = 0; i < m; ++i) {
                if (s[i] == '0' + j) {
                    s.replace(i, 1, p[k].second);
                    i += mt - 1;
                    ++m;
                }
            }
        }
        ++k;
    }
    int x = 0, w = 1;
    while (m--) {
        x += (s[m] - '0') * w;
        x %= mo;
        w = 10LL * w % mo;
    }
    cout << x << "\n";
}
