#include <vector>
#include <iostream>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; ++i)
        std::cin >> a[i];
    int ans = 0;
    for (int i = 0; i < n - 1; ++i) {
        if (a[i] == a[i + 1])
            continue;
        ans += 1;
        i += 1;
    }
    std::cout << ans << std::endl;
    return 0;
}
