#include <bits/stdc++.h>
using namespace std;
#define pb push_back

const int mod = 998244353;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> p(n);
    vector<pair<int, bool>> pr;
    for (int i = 0; i < n; ++i) {
        int x, y, s;
        cin >> x >> y >> s;
        p[i] = {x, y};
        pr.pb({x, s});
    }
    sort(pr.begin(), pr.end());
    vector<int> c(n);
    int t = 0;
    for (int i = 0; i < n; ++i) {
        t = (t + p[i].first - pr[i].first) % mod;
        if (pr[i].second) {
            auto it = lower_bound(p.begin(), p.end(), p[i]);
            c[it - p.begin()] = 1;
        }
    }
    t = (t + p[n - 1].first + 1) % mod;
    for (int i = n - 2; i >= 0; --i) {
        t = (t + (p[i].first - p[i + 1].first) * c[i + 1] % mod) % mod;
    }
    cout << t << '\n';
}
