#include <bits/stdc++.h>
using namespace std;
vector<int>g[305],ord;
int mark[305],vis[305],a[305][305],tmp[305][305],n,k;
void dfs(int x){
    ord.push_back(x);
    vis[x]=1;
    for(auto u:g[x])
        if(!vis[u])
            dfs(u);
}
int f(int x){
    int mx=0;
    for(int i=1;i<=n;i++)
        if(tmp[x][i]){
            mx=max(mx,tmp[i][x]);
        }
    return mx;
}
int main(){
    cin>>n;
    int p[n+5];
    for(int i=1;i<=n;i++){
        cin>>p[i];
    }
    for(int i=1;i<=n;i++)
        for(int j=1;j<=n;j++)
            cin>>a[i][j];
    for(int i=1;i<=n;i++)
        for(int j=1;j<=n;j++)
            if(i!=j&&a[i][j])
                g[i].push_back(j);
    for(int i=1;i<=n;i++)
        if(!vis[i])
            dfs(i);
    reverse(ord.begin(),ord.end());
    for(auto x:ord){
        k++;
        tmp[k][x]=p[x];
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            int tnp=f(j);
            if(tnp>tmp[i][j]&&a[j][i]){
                int tt=tmp[i][j];
                tmp[i][j]=tmp[j][i];
                tmp[j][i]=tt;
            }
        }
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            if(tmp[i][j])
                cout<<tmp[i][j]<<" ";
        }
    }
}
