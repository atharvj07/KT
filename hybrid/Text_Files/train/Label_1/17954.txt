#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k, m, a;
  cin >> n >> k >> m >> a;
  vector<int> g(a);
  for (int &x : g) cin >> x;
  sort(g.begin(), g.end());
  int cnt = 1;
  int last = g[0];
  vector<int> ans(n + 1, 3);
  for (int x : g) {
    if (x != last) {
      last = x;
      cnt++;
    }
    if (cnt <= k) ans[x] = 1;
  }
  int rem = k - cnt;
  last = -1;
  int cnt1 = 0;
  for (int i = a - 1; i >= 0; i--) {
    int x = g[i];
    if (x != last) {
      cnt1++;
    }
    last = x;
    if (cnt1 <= rem) ans[x] = 2;
  }
  for (int i = 1; i <= n; i++) cout << ans[i] << " ";
  return 0;
}
