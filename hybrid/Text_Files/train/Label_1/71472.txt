import sys
input = sys.stdin.readline

n, m = map(int,input().split())
g = [list(map(int,input().split())) for i in range(n)]
f = [[[0 for j in range(m)] for i in range(n)] for k in range(2)]

f[0][0][0] = g[0][0]
f[1][n-1][0] = g[n-1][0]
for i in range(1,n):
    f[0][i][0] = f[0][i-1][0] + g[i][0]
for i in range(1,m):
    f[1][n-1][i] = f[1][n-1][i-1] + g[n-1][i]

for i in range(1,n):
    for j in range(1,m):
        f[0][i][j] = max(f[0][i-1][j],f[0][i][j-1]) + g[i][j]
        f[1][n-1-i][j] = max(f[1][n-1-i][j-1],f[1][n-1-i+1][j]) + g[n-1-i][j]

mx = 0
for i in range(n):
    for j in range(m):
        if i!=n-1 or j!=m-1:
            mx = max(mx,f[0][i][j]+f[1][n-1-i][j]-g[i][j])

print(mx)
