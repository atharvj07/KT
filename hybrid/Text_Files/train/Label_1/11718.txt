#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> cnt(m);
    for (int i = 0; i < n; i++) cnt[a[i] % m]++;
    int ans = 0;
    int mx = 0;
    for (int i = 0; i < m; i++) mx = max(mx, cnt[i]);
    for (int i = 0; i < m; i++) {
        if (cnt[i] == mx) continue;
        ans += mx - cnt[i];
    }
    cout << ans << endl;
    for (int i = 0; i < n; i++) a[i] += ((ans + cnt[a[i] % m] - 1) / m) * m;
    for (int x : a) cout << x << " ";
    cout << endl;
    return 0;
}
