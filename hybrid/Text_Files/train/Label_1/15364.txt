#include<iostream>
const long long mod=1e9+7;
using namespace std;
int prv[100001],nxt[100001];
long long fastpow(long long a, long long b)
{
    if(b==0) return 1;
    if(b==1) return a;
    long long v=fastpow(a,b/2);
    v=(v*v)%mod;
    if(b%2) return (v*a)%mod;
    return v;
}
int main()
{
    long long n,m;
    cin>>n>>m;
    string s;
    cin>>s;
    long long ans=0;
    for(int i=0;i<m;i++)
    {
        if(s[i]=='(')
        {
            prv[i]=prv[i-1]+1;
            nxt[i]=nxt[i-1]+1;
        }
        else
        {
            if(prv[i-1]>0)
            {
                prv[i]=prv[i-1]-1;
                nxt[i]=nxt[i-1]-1;
            }
            else
            {
                prv[i]=0;
                nxt[i]=nxt[i-1]+1;
            }
        }
    }
    ans=fastpow(2,prv[m-1]);
    ans=(ans-1+mod)%mod;
    ans=(ans*fastpow(2,nxt[m-1]))%mod;
    cout<<ans<<endl;
    return 0;
}
