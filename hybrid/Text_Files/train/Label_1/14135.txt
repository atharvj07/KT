#include<bits/stdc++.h>
using namespace std;
typedef pair<int,int> pi;

int main()
{
    int n,m,T,u,v,t;
    cin>>n>>m>>T;
    vector<pi> adj[n+1];
    for(int i=0;i<m;i++)
    {
        cin>>u>>v>>t;
        adj[u].push_back({v,t});
    }
    priority_queue<pi> pq;
    pq.push({0,1});
    vector<int> dist(n+1,INT_MAX),par(n+1,-1);
    dist[1]=0;
    while(!pq.empty())
    {
        int u=pq.top().second;
        int w=-pq.top().first;
        pq.pop();
        if(w>dist[u]) continue;
        for(auto node:adj[u])
        {
            v=node.first;
            t=node.second;
            if(dist[u]+t<dist[v])
            {
                dist[v]=dist[u]+t;
                pq.push({-dist[v],v});
                par[v]=u;
            }
        }
    }
    if(dist[n]==INT_MAX) cout<<"1\n1";
    else
    {
        vector<int> path;
        int x=n;
        while(x!=-1)
        {
            path.push_back(x);
            x=par[x];
        }
        reverse(path.begin(),path.end());
        cout<<path.size()<<"\n";
        for(int p:path) cout<<p<<" ";
    }
}
