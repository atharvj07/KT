import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        int maxProd = 1;
        int[] maxProdIndices = new int[n];

        for (int i = 0; i < n; i++) {
            int prod = 1;
            int[] prodIndices = new int[n];
            for (int j = i; j < n; j++) {
                prod *= arr[j];
                if (prod > maxProd) {
                    maxProd = prod;
                    maxProdIndices = prodIndices;
                }
                prodIndices[j - i] = arr[j];
            }
        }

        for (int i = 0; i < maxProdIndices.length; i++) {
            if (maxProdIndices[i] != 0) {
                System.out.print(maxProdIndices[i] + " ");
            }
        }

        sc.close();
    }
}
