#include <algorithm>
#include <iostream>
#include <vector>

struct Point {
  long long x, y;
  Point() {}
  Point(long long x, long long y) : x(x), y(y) {}
};

bool cmp1(const Point &a, const Point &b) { return a.y < b.y; }
bool cmp2(const Point &a, const Point &b) { return a.x < b.x; }

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, k;
    std::cin >> n >> k;
    std::vector<Point> points(n);
    for (int i = 0; i < n; ++i) { std::cin >> points[i].x; }
    for (int i = 0; i < n; ++i) { std::cin >> points[i].y; }
    std::sort(points.begin(), points.end(), cmp1);
    long long ans = 0, y1 = -1e18, y2 = -1e18, cnt = 0;
    for (int l = 0, r = 0; l < n; ++l) {
      while (r < n && points[r].y <= points[l].y + k) ++r;
      if (r > l) {
        if (points[l].x >= y1 && points[l].x + k <= y2) ++cnt;
        if (points[r - 1].x >= y1 &&
            points[r - 1].x + k <= y2) ++cnt;
        if (l != r - 1 && points[l].x + k >= y1 &&
            points[r - 1].x <= y2) cnt += r - l - 1;
      }
      if (points[l].x < y1 || points[l].x + k > y2) {
        y1 = points[l].x, y2 = points[l].x + k, cnt = 1;
      }
    }
    std::sort(points.begin(), points.end(), cmp2);
    y1 = -1e18, y2 = -1e18, cnt = 0;
    for (int l = 0, r = 0; l < n; ++l) {
      while (r < n && points[r].x <= points[l].x + k) ++r;
      if (r > l) {
        if (points[l].y >= y1 && points[l].y + k <= y2) ++cnt;
        if (points[r - 1].y >= y1 &&
            points[r - 1].y + k <= y2) ++cnt;
        if (l != r - 1 && points[l].y + k >= y1 &&
            points[r - 1].y <= y2) cnt += r - l - 1;
      }
      if (points[l].y < y1 || points[l].y + k > y2) {
        y1 = points[l].y, y2 = points[l].y + k, cnt = 1;
      }
    }
    ans = std::max(ans, cnt);
    std::cout << ans << std::endl;
  }
  return 0;
}
