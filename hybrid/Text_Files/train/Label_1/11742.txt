#include <bits/stdc++.h>
using namespace std;
int n, k, ans = -1, cur[200001], ord[200001];
vector<pair<int, int>> a;
int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; i++) {
    int p, e;
    cin >> p >> e;
    a.push_back({p, e});
  }
  int all = 0;
  sort(a.rbegin(), a.rend());
  for (auto p : a) {
    int pnt = p.first, ef = p.second;
    all += ef;
    if (pnt <= cur[k]) {
      ans = all;
      break;
    }
    for (int i = 1; i <= n; i++) {
      if (ord[i] && cur[i] == pnt) { cur[i] = pnt - 1; }
      if (!ord[i] && cur[i] < pnt) {
        ord[i] = 1;
        cur[i] = pnt;
        break;
      }
    }
  }
  cout << ans << endl;
}
