x, k = map(int, input().split())
attended_rounds = [False] * x
for _ in range(k):
    round_info = list(map(int, input().split()))
    if round_info[0] == 1:
        attended_rounds[round_info[1]] = True
        attended_rounds[round_info[2]] = True
    else:
        attended_rounds[round_info[1]] = True

min_missed = max_missed = 0
i = 1
while i < x:
    if not attended_rounds[i]:
        min_missed += 1
        if i+1 < x and not attended_rounds[i+1]:
            i += 1
    i += 1

for i in range(1, x):
    if not attended_rounds[i]:
        max_missed += 1

print(min_missed, max_missed)
