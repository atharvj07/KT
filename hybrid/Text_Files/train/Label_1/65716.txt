from collections import defaultdict

n = int(input())
ranges = defaultdict(set)

for _ in range(n):
    gender, a, b = input().split()
    a, b = int(a), int(b)
    ranges[gender].add((a, b))

def solve():
    female, male = ranges["F"], ranges["M"]
    if not female or not male:
        return 0
    female.sort()
    male.sort()
    ans = 0
    i, j = 0, 0
    while i < len(female) and j < len(male):
        a1, b1 = female[i]
        a2, b2 = male[j]
        if a2 <= b1:
            ans += 1
            i += 1
            j += 1
        else:
            j += 1
    return ans * 2

print(solve())
