#include <bits/stdc++.h>
#define endl '\n'
using namespace std;

typedef long long ll;
const int maxn = 1e5;
int a[maxn], c[maxn], ans[maxn];

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    int cnt = 0, day = 0, p = 0, negCnt = 0;
    for (int i = 0; i < n; i++) {
        cnt += (a[i] > 0);
        negCnt -= (a[i] < 0);
        if (cnt == negCnt) { c[day] = i - p + 1; p = i + 1; day++; }
    }
    if (cnt == 0) { cout << -1 << endl; return 0; }
    cout << day << endl;
    for (int i = 0; i < day; i++) { cout << c[i] << ' '; }
    cout << endl;
    return 0;
}
