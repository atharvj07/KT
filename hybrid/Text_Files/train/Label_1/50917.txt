import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.nextInt();
        }
        int min = Integer.MAX_VALUE;
        int ans = 0;
        for (int mask = 0; mask < (1 << n); mask++) {
            int sum = 0, cnt = 0;
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) != 0) {
                    sum += w[i];
                    cnt++;
                }
            }
            if (sum <= k && cnt > 0) {
                min = Math.min(min, cnt);
                if (cnt == min) {
                    ans++;
                }
            }
        }
        System.out.println(min == Integer.MAX_VALUE ? -1 : min);
        System.out.println(min == Integer.MAX_VALUE ? 0 : ans % 1000000007);
    }
}
