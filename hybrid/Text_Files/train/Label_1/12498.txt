#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;
int main() {
    int t; cin >> t;
    while (t--) {
        long long W, H, x1, y1, x2, y2, w, h;
        cin >> W >> H >> x1 >> y1 >> x2 >> y2 >> w >> h;
        long long ans = -1, d = 1e18;
        if (abs(x1 - x2) + w <= W) {
            if (y2 >= h) d = min(d, y2 - h);
            if (H - y1 >= h) d = min(d, H - y1 - h);
        }
        if (abs(y1 - y2) + h <= H) {
            if (x2 >= w) d = min(d, x2 - w);
            if (W - x1 >= w) d = min(d, W - x1 - w);
        }
        ans = (d == 1e18) ? -1 : d;
        cout << fixed << ans << endl;
    }
    return 0;
}
