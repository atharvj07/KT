#include <iostream>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int ang;
    std::cin >> ang;
    if (3 * ang == 360 || 9 * ang == 360)
      std::cout << -1 << "\n";
    else if (180 % ang == 0) {
      int v = 180 / ang;
      if (180 % v == 0)
        std::cout << v << "\n";
      else
        std::cout << -1 << "\n";
    } else
      std::cout << (360 / ang) << "\n";
  }
  return 0;
}
