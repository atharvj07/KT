#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    string ow, nw;
    cin >> ow >> m;
    for (int i = 0; i < m; i++) {
        cin >> nw;
        for (int j = 0; j < n; j++) {
            if (ow[j] == '*') {
                ow[j] = nw[j];
            }
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (ow[i] == '*') {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
