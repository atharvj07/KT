import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long ans = -1, a, b;
        for (int i = 0; i < n; i++) {
            a = in.nextLong();
            b = in.nextLong();
            if (ans == -1) ans = a;
            else while (ans % b != a) ans += b;
        }
        System.out.println(ans % 1000000007);
    }
}
