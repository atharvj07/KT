n, k = map(int, input().split())
s = input()
if k == 1:
    print(ord(s[0]) - ord('a') + 1)
else:
    ans = 0
    rem = set()
    for i in s:
        if i in rem:
            continue
        rem.add(i)
        if k == 0:
            break
        while len(rem) > 1:
            x = min(rem)
            rem.remove(x)
            k -= 1
            ans += ord(x) - ord('a') + 1
            if k == 0:
                break
        rem.clear()
    if k > 0:
        print(-1)
    else:
        print(ans)
