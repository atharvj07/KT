#include <bits/stdc++.h>
using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    int i, j;
    for (i = 0; i < n - 1; i++) {
        if (s[i] > s[i + 1]) break;
    }
    if (i == n - 1) {
        s[n - 1] = 'a';
    } else {
        for (j = i; j < n; j++) {
            if (s[j] == 'a') s[j] = 'z';
            else s[j]--;
        }
    }
    cout << s << endl;
    return 0;
}
