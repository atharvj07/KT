#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int n, m;
vector<int> a;

int main() {
    cin >> n >> m;
    a.resize(n);
    for (int &x : a) { cin >> x; }
    vector<vector<long long>> dp(m + 1, vector<long long>(n + 1, 0));
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            dp[i][j] = dp[i][j - 1];
            if (i == 1) { dp[i][j] = dp[i][j - 1] + 1ll * j * a[j - 1]; }
            else { dp[i][j] = min(dp[i][j], dp[i - 1][j - 1] + 1ll * i * a[j - 1]); }
        }
    }
    for (int i = 1; i <= n; i++) { cout << dp[m][i] << ' '; }
    cout << '\n';
}
