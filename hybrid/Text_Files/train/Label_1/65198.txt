n = int(input())
a = []
b = []
for _ in range(n):
    i, j = map(int, input().split())
    a.append(i)
    b.append(j)
ans = 0
for i in range(0, a[0]):
    f = 1
    for j in range(n):
        if a[j] <= i <= b[j]:
            f = 0
            break
    if f == 1:
        ans += 1
for i in range(b[0] + 1, 101):
    f = 1
    for j in range(n):
        if a[j] <= i <= b[j]:
            f = 0
            break
    if f == 1:
        ans += 1
print(ans)
