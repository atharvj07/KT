#include <iostream>
#include <string>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    int cnt = 0;
    bool odd = false;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'o') {
            cnt++;
            if (cnt % 2 == 1) odd = true;
        }
    }
    if (cnt == 0) {
        cout << "NO\n";
        return 0;
    }
    if (n % 2 == 0) {
        if (odd) {
            cout << "NO\n";
            return 0;
        }
    } else {
        if (cnt % 2 == 1) {
            cout << "NO\n";
            return 0;
        }
    }
    cout << "YES\n";
    return 0;
}
