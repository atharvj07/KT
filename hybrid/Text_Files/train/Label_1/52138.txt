import java.io.*;
import java.util.*;
import java.text.*;
import java.math.*;
import java.util.regex.*;

class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String a = in.next();
        String b = in.next();
        int alen = a.length();
        int blen = b.length();
        int idx = 0;
        int count = 0;
        while (idx < blen) {
            if (a.indexOf(b.charAt(idx)) != -1) {
                count++;
            }
            idx++;
        }
        System.out.println(count);
    }
}
