import java.util.*;
import java.io.*;

public class Main {
  static int[] dice = {1, 2, 3, 4, 5, 6};
  static int[][][] nets;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    nets = new int[n][5][5];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < 5; j++) {
        for (int k = 0; k < 5; k++) {
          nets[i][j][k] = sc.nextInt();
        }
      }
      if (isValid(i)) {
        System.out.println("true");
      } else {
        System.out.println("false");
      }
    }
  }

  private static boolean isValid(int i) {
    Set<Integer> set = new HashSet<>();
    for (int j = 0; j < 5; j++) {
      set.add(nets[i][j][0] + nets[i][j][4]);
      for (int k = 0; k < 5; k++) {
        set.add(nets[i][j][k] + nets[i][4 - j][4 - k]);
      }
      set.add(nets[i][j][2] + nets[i][0][2]);
      set.add(nets[i][j][3] + nets[i][0][3]);
    }
    if (set.size() == 6) {
      for (int num : set) {
        if (num != 7) {
          return false;
        }
      }
      return true;
    }
    return false;
  }
}
