import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long m = in.nextLong();
        long k = in.nextLong();

        long row = (k / (2 * m - 2)) % n + 1;
        long col = (k % (2 * m - 2) + m) % m + 1;

        System.out.println(row + " " + col);
    }
}
