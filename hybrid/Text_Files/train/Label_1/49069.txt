import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int h = sc.nextInt();
            if (n == 0 && h == 0) {
                break;
            }
            int[][][] cube = new int[n][n][n];
            for (int i = 0; i < h; i++) {
                String c = sc.next();
                int a = sc.nextInt() - 1;
                int b = sc.nextInt() - 1;
                if (c.equals("xy")) {
                    for (int k = 0; k < n; k++) {
                        cube[a][b][k] = -1;
                        cube[a][b][k] = -1;
                    }
                } else if (c.equals("xz")) {
                    for (int k = 0; k < n; k++) {
                        cube[a][k][b] = -1;
                        cube[a][k][b] = -1;
                    }
                } else {
                    for (int k = 0; k < n; k++) {
                        cube[k][a][b] = -1;
                        cube[k][a][b] = -1;
                    }
                }
            }
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    for (int k = 0; k < n; k++) {
                        if (cube[i][j][k] >= 0) {
                            cnt++;
                        }
                    }
                }
            }
            System.out.println(cnt);
        }
    }
}
