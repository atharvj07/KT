t = int(input().strip())
for _ in range(t):
    n, m = map(int, input().split())
    possible = m % 2 == 0
    for _ in range(n):
        top = list(map(int, input().split()))
        bottom = list(map(int, input().split()))
        if top[1] == bottom[0]:
            possible = True
    if possible:
        print("YES")
    else:
        print("NO")
