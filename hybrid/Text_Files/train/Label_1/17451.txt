#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;

ll solve(ll n, ll k){
    ll ans = 0;
    while(n > 0){
        if(n % k == 0) n /= k, ans++;
        else{
            ll rem = n % k;
            n -= rem;
            ans += rem;
        }
    }
    return ans;
}

int main(){
    int t; cin >> t;
    while(t--){
        ll n, k; cin >> n >> k;
        cout << solve(n, k) << endl;
    }
    return 0;
}
