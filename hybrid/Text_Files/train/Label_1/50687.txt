import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long x = scanner.nextLong();
        long y = scanner.nextLong();
        long z = scanner.nextLong();
        long max = (x + y) / z;
        long min = Math.min(x % z, y % z);
        System.out.println(max + " " + min);
    }
}
