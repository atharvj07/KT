for _ in range(int(input())):
    r, g, b = map(int, input().split())
    rr = sorted([int(x) for x in input().split()])
    gg = sorted([int(x) for x in input().split()])
    bb = sorted([int(x) for x in input().split()])
    print(min((a - b)**2 + (b - c)**2 + (c - a)**2 for a in rr for b in gg for c in bb))
