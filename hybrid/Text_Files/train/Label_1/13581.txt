#include <bits/stdc++.h>

using namespace std;

typedef long long int lli;

const lli MAXN = 1e5+5;

lli n, s, f, p[MAXN], t;
bool vis[MAXN];

int main() {
    cin >> s >> f >> t;
    cin >> n;
    for (lli i=0;i<n;i++) {
        cin >> p[i];
    }

    lli st = s, ans = s, cnt = 0;
    for (lli i=0;i<n;i++) {
        if (!vis[i]) {
            if (st + t <= p[i]) {
                for (lli j=i;j<n;j++) {
                    if (p[j] >= st && p[j] <= st+t) {
                        vis[j] = true;
                        cnt++;
                    }
                }
                st += cnt * t;
                ans = max(ans, st - t - s);
                cnt = 0;
            }
            else {
                ans = max(ans, p[i] + t - s);
            }
        }
    }

    cout << ans << endl;

    return 0;
}
