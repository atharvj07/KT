#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> x(n), y(m);
    for (int i = 0; i < n; i++) cin >> x[i];
    for (int i = 0; i < m; i++) cin >> y[i];
    unordered_set<int> s(y.begin(), y.end());
    stack<int> ans;
    for (int i = n - 1; i >= 0; i--) {
        if (s.count(x[i])) {
            ans.push(x[i]);
        }
    }
    while (!ans.empty()) {
        cout << ans.top() << ' ';
        ans.pop();
    }
    return 0;
}
