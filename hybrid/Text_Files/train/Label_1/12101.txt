#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, m, p;
  cin >> n >> m >> p;
  vector<int> a(n);
  vector<int> b(m);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < m; i++) cin >> b[i];
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    int j = i;
    bool ok = true;
    for (int k = 0; k < m; k++) {
      if (j < n && a[j] == b[k]) j++;
      else {
        ok = false;
        break;
      }
    }
    if (ok) cnt++;
  }
  cout << cnt << endl;
  for (int i = 0; i < n; i++) {
    int j = i;
    bool ok = true;
    for (int k = 0; k < m; k++) {
      if (j < n && a[j] == b[k]) j++;
      else {
        ok = false;
        break;
      }
    }
    if (ok) cout << i + 1 << " ";
  }
  cout << endl;
  return 0;
}
