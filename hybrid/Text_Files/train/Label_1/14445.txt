#include <bits/stdc++.h>
using namespace std;

int get_max_digit(int sum,int cnt)// get maximum digit for current number
{
    int ans = min(9,sum/cnt);
    sum-=ans;
    cnt--;
    return ans;
}

int main(){
    int t,s,n;
    cin>>t;
    while(t--)
    {
        cin>>s>>n;
        for(int i=n-1;i>=0;i--)
        {
            int dgt = get_max_digit(s,i+1);
            cout<<dgt<<(i?" ":"");
            s-=dgt;
        }
        cout<<endl;
    }
    return 0;
}
