#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> friends(n);
  for (int i = 0; i < n; i++) {
    cin >> friends[i].first >> friends[i].second;
  }
  sort(friends.begin(), friends.end());
  int min_area = friends[0].first * friends[0].second;
  for (int i = 1; i < n / 2; i++) {
    min_area += friends[i].second * friends[i].first;
  }
  for (int i = n / 2; i < n; i++) {
    min_area += friends[i].first * friends[i].second;
  }
  cout << min_area << endl;
  return 0;
}
