import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int x = input.nextInt();
    int y = input.nextInt();
    double ratio = (double) y / 100;
    int diff = x;
    if (ratio * n > x) {
      diff = (int) Math.ceil(ratio * n) - x;
    }
    System.out.println(diff);
  }
}
