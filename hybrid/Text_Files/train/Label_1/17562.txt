#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t,n,a,b;
    cin>>t;
    while(t--)
    {
        cin>>n;
        int x=0,y=0;
        while(n--)
        {
            cin>>a>>b;
            if(a<x || b<y)
            {
                cout<<"NO\n";
                goto label;
            }
            x=a;
            y=b;
        }
        cout<<"YES\n";
        label:;
    }
}
