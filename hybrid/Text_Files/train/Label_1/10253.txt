#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;

const int maxn = 100005;
int n, q, h[maxn];
ll lc[maxn];
vector<int> adj[maxn];

int main() {
    cin >> n >> q;
    for (int i = 1; i <= n; i++) cin >> h[i];
    for (int i = 2; i <= n; i++) {
        ll diff = abs(h[i] - h[i - 1]);
        lc[i] = lc[i - 1] + diff;
    }
    while (q--) {
        int l, r;
        cin >> l >> r;
        auto ans = lc[r] - (l - 1 < 1 ? 0 : lc[l - 1]);
        cout << ans << endl;
    }
    return 0;
}
