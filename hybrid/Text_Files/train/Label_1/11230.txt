#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9+7;
ll add(ll a, ll b){
    return ((a+b)%mod+mod)%mod;
}
ll mul(ll a, ll b){
    return ((a*b)%mod+mod)%mod;
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n;
    ll sum = 0;
    cin >> n;
    for(int i=0; i<n; i++){
        int t;
        cin >> t;
        if(t==1){
            ll a, x;
            cin >> a >> x;
            sum = add(sum, mul(a, x));
        }
        else if(t==2){
            ll k;
            cin >> k;
            sum = add(sum, k);
        }
        else{
            sum = add(sum, -1ll);
        }
        cout << fixed << setprecision(6) << (double)sum << "\n";
    }
    return 0;
}
