import java.util.*;

class Main {
    static int n, m, l, e, v;
    static Set<Integer> stairs, lifts;
    static int[][] dist;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        l = sc.nextInt();
        e = sc.nextInt();
        v = sc.nextInt();
        stairs = new HashSet<>();
        lifts = new HashSet<>();
        for (int i = 0; i < l; i++) stairs.add(sc.nextInt());
        for (int i = 0; i < e; i++) lifts.add(sc.nextInt());
        pre();
        int q = sc.nextInt();
        while (q-- > 0) {
            int x1 = sc.nextInt() - 1, y1 = sc.nextInt() - 1;
            int x2 = sc.nextInt() - 1, y2 = sc.nextInt() - 1;
            System.out.println(dist[x1][y1] + dist[x2][y2]);
        }
    }

    private static void pre() {
        dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (stairs.contains(j + 1))
                    dist[i][j] = i;
                else {
                    int idx = binarySearch(j + 1, lifts);
                    if (idx == -1)
                        dist[i][j] = Integer.MAX_VALUE;
                    else {
                        int nearest = lifts.toArray(lifts.toArray(new Integer[0]))[idx];
                        if (j + 1 - nearest <= nearest - j)
                            dist[i][j] = Math.min(i + 1, Math.abs(i - (nearest - j - 1) / v));
                        else
                            dist[i][j] = Math.min(i + 1, Math.abs(i - (nearest - j + 1) / v));
                    }
                }
            }
        }
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = Math.min(dist[i][j], dist[i - 1][j] + 1);
            }
        }
        for (int i = n - 2; i >= 0; i--) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = Math.min(dist[i][j], dist[i + 1][j] + 1);
            }
        }
    }

    private static int binarySearch(int v, Set<Integer> set) {
        Integer[] arr = set.toArray(set.toArray(new Integer[0]));
        int n = arr.length;
        int l = 0, r = n - 1, mid;
        while (l <= r) {
            mid = (l + r) / 2;
            if (arr[mid] < v)
                l = mid + 1;
            else
                r = mid - 1;
        }
        return r >= 0 && arr[r] == v ? r : -1;
    }
}
