#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const int MAXN = 2e5 + 5;
const ll MOD = 1e9 + 7;

int n, m;
vector<int> ev, od;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        if (x % 2 == 0)
            ev.push_back(x);
        else
            od.push_back(x);
    }
    int sz_ev = ev.size();
    int sz_od = od.size();
    sort(ev.begin(), ev.end());
    sort(od.begin(), od.end());
    if (sz_ev == sz_od)
        cout << "0\n";
    else if (sz_ev == sz_od - 1) {
        cout << "1\n";
        vector<int> ans;
        for (int i = 0; i < n; i++) {
            if (i < sz_ev)
                ans.push_back(ev[i]);
            else
                ans.push_back(od[i]);
        }
        for (auto e : ans) cout << e << " ";
        cout << "\n";
    } else if (sz_od == sz_ev - 1) {
        cout << "1\n";
        vector<int> ans;
        for (int i = 0; i < n; i++) {
            if (i < sz_od)
                ans.push_back(od[i]);
            else
                ans.push_back(ev[i]);
        }
        for (auto e : ans) cout << e << " ";
        cout << "\n";
    } else
        cout << "-1\n";
    return 0;
}
