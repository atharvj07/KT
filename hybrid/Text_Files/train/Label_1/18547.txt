#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 998244353
ll powmod(ll a, ll x){
    ll ans=1;
    while(x){
        if(x&1) ans=(ans*a)%mod;
        x>>=1;
        a=(a*a)%mod;
    }
    return ans;
}
int main(){
    ll n, m, k;
    cin>>n>>m>>k;
    ll ans;
    if(!k){
        ans = powmod(m, n);
    }
    else{
        ll f[m][m]={0};
        for(int i=0;i<m;i++){
            for(int j=0;j<m;j++){
                if(i==j) f[i][j]=m-1;
                else f[i][j]=1;
            }
        }
        ll mat[m][m];
        for(int i=0;i<m;i++){
            for(int j=0;j<m;j++){
                mat[i][j]=f[i][j];
            }
        }
        for(int pw=2;pw<=k;pw*=2){
            if(pw==k) break;
            for(int i=0;i<m;i++){
                for(int j=0;j<m;j++){
                    for(int k=0;k<m;k++){
                        mat[i][j]=(mat[i][j]+(f[i][k]*mat[k][j])%mod)%mod;
                    }
                }
            }
            for(int i=0;i<m;i++){
                for(int j=0;j<m;j++){
                    f[i][j]=mat[i][j];
                }
            }
        }
        ans=powmod(m, n-k-1);
        for(int i=0;i<m;i++){
            ans=(ans*mat[i][0])%mod;
        }
    }
    cout<<ans<<endl;
    return 0;
}
