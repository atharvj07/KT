import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.StringTokenizer;

public class Main {
    private static class Node {
        int parent;
        int type;
        boolean isSpecial;
        boolean isPart;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int parent = Integer.parseInt(st.nextToken());
            int type = Integer.parseInt(st.nextToken());
            nodes[i] = new Node();
            nodes[i].parent = parent;
            nodes[i].type = type;
        }

        // Preprocess the nodes to determine if each node is a special case or a part of another node
        for (int i = 1; i <= n; i++) {
            Node node = nodes[i];
            if (node.parent == -1) {
                continue;
            }

            Node parentNode = nodes[node.parent];
            if (node.type == 0) {
                parentNode.isSpecial = true;
                node.isSpecial = true;
            } else if (node.type == 1) {
                parentNode.isPart = true;
                node.isPart = true;
            }
        }

        // Process the queries
        int q = Integer.parseInt(br.readLine());
        for (int i = 0; i < q; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());

            Node nodeU = nodes[u];
            Node nodeV = nodes[v];

            if (type == 1) {
                // Check if node v is a special case of node u
                if (nodeV.isSpecial && (nodeU.isSpecial || nodeU.parent == v)) {
                    System.out.println("YES");
                } else {
                    System.out.println("NO");
                }
            } else if (type == 2) {
                // Check if node v is a part of node u
                if (nodeV.isPart && (nodeU.isPart || nodeU.parent == v)) {
                    System.out.println("YES");
                } else {
                    System.out.println("NO");
                }
            }
        }
    }
}
