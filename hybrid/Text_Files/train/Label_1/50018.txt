import java.util.Scanner;

public class KilaniGame {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();
        int[] speeds = new int[p];
        for (int i = 0; i < p; i++) {
            speeds[i] = scanner.nextInt();
        }
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            grid[i] = scanner.next().toCharArray();
        }

        int[][][] reachable = new int[n][m][p];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                for (int k = 0; k < p; k++) {
                    if (grid[i][j] != '.' && grid[i][j] != '#') {
                        reachable[i][j][k] = 1;
                    }
                }
            }
        }

        for (int round = 0; round < 10000; round++) {
            boolean changed = false;
            for (int i = 0; i < p; i++) {
                for (int j = 0; j < n; j++) {
                    for (int k = 0; k < m; k++) {
                        if (reachable[j][k][i] > 0) {
                            for (int dx = -speeds[i]; dx <= speeds[i]; dx++) {
                                for (int dy = -speeds[i]; dy <= speeds[i]; dy++) {
                                    int nx = j + dx;
                                    int ny = k + dy;
                                    if (nx >= 0 && nx < n && ny >= 0 && ny < m && grid[nx][ny] == '.') {
                                        reachable[nx][ny][i] = 1;
                                        changed = true;
                                    }
                                }
                            }
                        }
                    }
                }
            }
            if (!changed) {
                break;
            }
        }

        int[] controlled = new int[p];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                for (int k = 0; k < p; k++) {
                    if (reachable[i][j][k] > 0) {
                        controlled[k]++;
                    }
                }
            }
        }

        for (int i = 0; i < p; i++) {
            System.out.print(controlled[i] + " ");
        }
        System.out.println();
    }
}
