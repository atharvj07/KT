import java.util.Scanner;
import java.util.HashSet;
import java.util.Set;

public class ForbiddenPairs {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int k = scanner.nextInt();
        scanner.nextLine(); // Consume the empty line

        Set<Character> forbidden = new HashSet<>();
        for (int i = 0; i < k; i++) {
            String pair = scanner.nextLine();
            forbidden.add(pair.charAt(0));
            forbidden.add(pair.charAt(1));
        }

        int deletions = 0;
        for (int i = 0; i < s.length() - 1; i++) {
            char ch1 = s.charAt(i);
            char ch2 = s.charAt(i + 1);
            if (forbidden.contains(ch1) && forbidden.contains(ch2)) {
                deletions++;
                i++; // Skip the next character since it's already removed
            }
        }

        System.out.println(deletions);
    }
}
