#include <iostream>
#include <vector>
#include <limits>

using namespace std;

vector<vector<int>> memo;
vector<vector<bool>> computed;
vector<vector<int>> costs;

int solve(int start, int end, int n) {
  if (computed[start][end])
    return memo[start][end];
  if (n == 1)
    return costs[start][end];
  int best_cost = numeric_limits<int>::max();
  for (int i = 0; i < 3; i++) {
    if (i != start && i != end) {
      int cost = solve(start, i, n - 1) + solve(i, end, 1) + costs[start][end];
      if (cost < best_cost)
        best_cost = cost;
    }
  }
  memo[start][end] = best_cost;
  computed[start][end] = true;
  return best_cost;
}

int main() {
  for (int i = 0; i < 3; i++) {
    costs.push_back(vector<int>(3, 0));
    memo.push_back(vector<int>(3, 0));
    computed.push_back(vector<bool>(3, false));
  }
  for (int i = 0; i < 3; i++) {
    cin >> costs[i][0] >> costs[i][1] >> costs[i][2];
  }
  int n;
  cin >> n;
  cout << solve(0, 2, n) << endl;
  return 0;
}
