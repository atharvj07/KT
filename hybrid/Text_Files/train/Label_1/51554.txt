import java.util.*;
import java.math.*;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    scanner.close();

    int[] digits = {2, 3, 5, 7};
    int min = 1;

    for (int i = 0; i < n; i++) {
      min *= 10;
      for (int digit : digits) {
        while (min % digit != 0) {
          min++;
        }
      }
    }

    if (min > Math.pow(10, n) - 1) {
      System.out.println("-1");
    } else {
      System.out.println(min);
    }
  }
}
