import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        List<Integer> pos = new ArrayList();
        for (int i = 0; i < n / 2; i++) {
            pos.add(input.nextInt());
        }
        Collections.sort(pos);
        int ans = 0;
        for (int x : pos) {
            ans += Math.min(x - 1, n - x);
        }
        System.out.println(ans);
    }
}
