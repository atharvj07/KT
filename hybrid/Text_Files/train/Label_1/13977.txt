#include <iostream>
#include <vector>

#define ll long long

using namespace std;

const int mxN = 105;
int n, m, sum = 0;
vector<int> adj[mxN];
int bal[mxN];

int add(int x, int y) {
    bal[x] += y;
    bal[y] -= y;
}

int dfs(int node) {
    int s = 0;
    for (int next : adj[node]) {
        if (bal[next] > 0) {
            int mn = min(bal[node], bal[next]);
            add(next, mn);
            s += mn;
        }
    }
    return s;
}

int main() {
    cin >> n >> m;
    while (m--) {
        int a, b, c;
        cin >> a >> b >> c;
        adj[a].push_back(b);
        add(a, c);
        sum += c;
    }

    int ans = 0;
    for (int i = 1; i <= n; ++i) ans += dfs(i);

    cout << sum - ans;
    return 0;
}
