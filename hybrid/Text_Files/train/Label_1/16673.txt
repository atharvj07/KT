#include <bits/stdc++.h>
using namespace std;
const int mx=1502;
int f[mx][mx],good[mx],dp[mx][mx],bad,n,ans,k;
string s;
int main(){
    cin>>s;
    for(int i=0;i<26;i++){
        cin>>good[i];
    }
    cin>>k;
    n=s.size();
    for(int i=1;i<=n;i++){
        for(int j=i;j<=n;j++){
            if(i==j){
                if(good[s[i-1]-'a']==1) dp[i][j]=1;
                continue;
            }
            dp[i][j]=dp[i][j-1];
            if(good[s[j-1]-'a']==1) dp[i][j]++;
            if(dp[i][j]<=k) f[i][j]=1;
            if(f[i][j-1]==1){
                dp[i][j]=dp[i][j-1];
                f[i][j]=1;
            }
        }
    }
    int last[27];
    memset(last,-1,sizeof last);
    for(int i=1;i<=n;i++){
        for(int j=1;j<=i;j++){
            if(f[j][i]==1){
                ans+=i-max(last[s[i-1]-'a'],j)+1;
                last[s[i-1]-'a']=i;
            }
        }
    }
    cout<<ans;
}
