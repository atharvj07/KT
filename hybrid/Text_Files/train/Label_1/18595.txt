#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, k;
    cin >> n >> m >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int x = 0;
    if (k >= 1) {
      if (m == 1)
        x = a[n - 2];
      else if (m == n)
        x = a[1];
      else
        x = max(a[k - 1], a[n - k - 1]);
    }
    cout << x << endl;
  }
  return 0;
}
