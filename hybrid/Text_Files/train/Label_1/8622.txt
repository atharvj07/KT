#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define int long long
int32_t main() {
    ll h, n, cur = 0;
    cin >> h >> n;
    vector<int> p(n);
    for (int i = 0; i < n; i++) cin >> p[i];
    for (int i = 0; i <= 1e6; i++) {
        ll sum = 0;
        for (int j = 0; j < n; j++) sum += p[(i + j) % n];
        cur += sum;
        if (cur <= 0) return cout << i + 1 << endl, 0;
    }
    return cout << -1 << endl, 0;
}
