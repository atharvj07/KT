#include <cstdio>
#include <algorithm>
#include <vector>

using namespace std;

int n, k, a[105];

int main() {
  scanf("%d%d", &n, &k);
  for (int i = 0; i < n; i++) {
    scanf("%d", &a[i]);
  }
  int ans = 0;
  for (int b = 1; b <= n; b++) {
    vector<int> keep;
    for (int c = b; c <= n; c += k) {
      keep.push_back(c);
    }
    int cnt = 0;
    for (int c: keep) {
      cnt += a[c - 1];
    }
    ans = max(ans, abs(cnt));
  }
  printf("%d\n", ans);
  return 0;
}
