from collections import defaultdict
n,m=map(int,input().split())
s=[input() for i in range(n)]

d=defaultdict(int)
for t in s:
    for i in range(m):
        for j in range(i+1,m+1):
            d[t[i:j]]+=1

t=[]
p=0
for i in d:
    if len(i) % 2 == 0 and d[i] * len(i) > p:
        t=[i] * d[i]
        p=len(i) * d[i]
    elif d[i] > p // len(i):
        t=[i+''] * (d[i] // 2)
        if d[i] % 2:
            t.append(i[::-1])
        p=len(i) * (d[i] // 2) + len(t[-1])

t.sort()
print(len(''.join(t)))
print(''.join(t))
