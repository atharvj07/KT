import sys

def solve():
    n = int(input())
    segs = []
    for _ in range(n):
        l, r = map(int, input().split())
        segs.append((l, r))
    segs.sort(key=lambda seg: seg[1])

    l = segs[0][0]
    r = segs[0][1]
    groups = [1]
    for i in range(1, n):
        l1, r1 = segs[i]
        if l1 > r:
            groups.append(2)
            l = l1
            r = r1
        else:
            groups.append(1)
    if len(set(groups)) == 2:
        return groups
    else:
        return [-1]

for _ in range(int(input())):
    res = solve()
    print(*res)
