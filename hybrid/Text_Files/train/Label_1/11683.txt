#include <bits/stdc++.h>
#define lld long long int
using namespace std;
int ar[1001][1001];
int main() {
    int n, k;
    cin >> n >> k;
    if (k > (n * (n - 1)) / 2) {
        puts("-1");
        return 0;
    }
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (k) {
                ar[i][j] = 1;
                ar[j][i] = 1;
                cnt++;
                k--;
            }
        }
    }
    cout << cnt << endl;
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (ar[i][j]) cout << i << " " << j << endl;
        }
    }
    return 0;
}
