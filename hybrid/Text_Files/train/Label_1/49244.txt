import java.util.Scanner;

public class SummerJob {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int f = scanner.nextInt();

        int[] sales = new int[n];
        int[] clients = new int[n];

        for (int i = 0; i < n; i++) {
            sales[i] = scanner.nextInt();
            clients[i] = scanner.nextInt();
        }

        int maxSales = 0;
        for (int mask = 0; mask < (1 << f); mask++) {
            int currentSales = 0;
            for (int i = 0; i < n; i++) {
                if (((mask >> i) & 1) == 1) {
                    currentSales += Math.min(sales[i] * 2, clients[i]);
                } else {
                    currentSales += Math.min(sales[i], clients[i]);
                }
            }
            maxSales = Math.max(maxSales, currentSales);
        }

        System.out.println(maxSales);
    }
}
