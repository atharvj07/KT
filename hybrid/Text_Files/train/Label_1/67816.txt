def dfs(x,lst,adj,vis,g,ct):
    vis[x]=1
    lst.append(x)
    for i in adj[x]:
        if not vis[i]:
            if dfs(i,lst,adj,vis,g,ct+1):
                return True
        elif i==g and ct>k:
            return True
    vis[x]=0
    lst.pop()
    return False

    adj[a].append(b)
    adj[b].append(a)
ans=[]
