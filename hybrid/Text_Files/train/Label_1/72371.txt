import sys
from typing import List, Tuple

# Reading input data
n = int(sys.stdin.readline())
points = [tuple(map(int, sys.stdin.readline().split())) for _ in range(n)]

# Preprocessing data
# Sorting points by x-coordinate, then by y-coordinate
points.sort(key=lambda p: (p[0], p[1]))

# Storing points in clockwise order
convex_hull = []
for point in points:
    while len(convex_hull) >= 2 and (
        (convex_hull[-1][0] - convex_hull[-2][0]) * (point[1] - convex_hull[-2][1])
        - (convex_hull[-1][1] - convex_hull[-2][1]) * (point[0] - convex_hull[-2][0])
        <= 0
    ):
        convex_hull.pop()
    convex_hull.append(point)

# Calculating the maximum perimeter for each possible number of points
perimeters = [0] * (n + 1)
for k in range(3, n + 1):
    # Calculating the maximum perimeter for a polygon with k vertices
    min_perimeter = float('inf')
    for i in range(k):
        # Calculating the perimeter of the polygon formed by points from i to i+k-1 (in clockwise order)
        perimeter = 0
        for j in range(i, i + k):
            perimeter += abs(convex_hull[j % k][0] - convex_hull[(j + 1) % k][0]) + abs(convex_hull[j % k][1] - convex_hull[(j + 1) % k][1])
        min_perimeter = min(min_perimeter, perimeter)
    perimeters[k] = min_perimeter

# Printing the maximum perimeters for each possible number of points
for k in range(3, n + 1):
    print(perimeters[k])
