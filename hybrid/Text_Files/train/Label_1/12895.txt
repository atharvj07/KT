#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, ans = 1;
  cin >> n;
  vector<pair<int, int>> points(n);
  for (int i = 0; i < n; i++) { cin >> points[i].first >> points[i].second; }
  sort(points.begin(), points.end());
  int r = points[0].second;
  for (int l = 0; l < n; l++) {
    while (r < points[l].first) { r = points[++l].second; }
    if (points[l].second >= r) { r = points[l].second; ans++; }
  }
  cout << ans;
  return 0;
}
