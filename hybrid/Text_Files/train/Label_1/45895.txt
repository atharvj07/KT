import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        int n = Integer.parseInt(tokenizer.nextToken());
        int m = Integer.parseInt(tokenizer.nextToken());
        List<Set<Integer>> boxes = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            boxes.add(new HashSet<>());
            int count = Integer.parseInt(tokenizer.nextToken());
            for (int j = 0; j < count; j++) {
                boxes.get(i).add(Integer.parseInt(tokenizer.nextToken()));
            }
        }
        int[][] minMax = new int[n][2];
        for (int i = 0; i < n; i++) {
            minMax[i][0] = boxes.get(i).size();
            minMax[i][1] = boxes.get(i).size();
        }
        int min = minMax[0][0];
        int max = minMax[0][1];
        for (int i = 1; i < n; i++) {
            min = Math.min(min, minMax[i][0]);
            max = Math.max(max, minMax[i][1]);
        }
        int k = max - min;
        List<String> moves = new ArrayList<>();
        while (k > 0) {
            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {
                    int[] t1 = minMax[i];
                    int[] t2 = minMax[j];
                    if (t1[0] + 1 <= t2[1] && t2[0] - 1 >= t1[1]) {
                        for (int kind : boxes.get(i)) {
                            if (!boxes.get(j).contains(kind)) {
                                String move = String.format("%d %d %d", i + 1, j + 1, kind);
                                moves.add(move);
                                minMax[i][1]++;
                                minMax[j][0]--;
                                min = Math.min(min, minMax[i][0]);
                                max = Math.max(max, minMax[j][1]);
                                k--;
                                break;
                            }
                        }
                    }
                }
            }
        }
        System.out.println(moves.size());
        for (String move : moves) {
            System.out.println(move);
        }
    }
}
