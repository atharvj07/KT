import java.util.*;
import java.io.*;

class Main {
    private static int n, m;
    private static int[][] a, b;

    private static boolean dfs(int i, int j, int cx, int cy) {
        if (i < 0 || i >= n || j < 0 || j >= m || a[i][j] != 0 || (i == cx && j == cy)) {
            return false;
        }
        b[i][j] = 1;
        boolean canMove = false;
        canMove |= dfs(i - 1, j, cx, cy);
        canMove |= dfs(i + 1, j, cx, cy);
        canMove |= dfs(i, j - 1, cx, cy);
        canMove |= dfs(i, j + 1, cx, cy);
        return canMove;
    }

    private static boolean check(int cx, int cy) {
        for (int i = 0; i < n; i++) {
            Arrays.fill(b[i], 0);
        }
        return dfs(cx, cy, cx, cy);
    }

    public static void main(String[] args) throws IOException {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            n = scanner.nextInt();
            m = scanner.nextInt();
            a = new int[n][m];
            b = new int[n][m];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    a[i][j] = scanner.nextInt();
                }
            }
            boolean canAshishWin = false;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (a[i][j] == 0) {
                        canAshishWin |= check(i, j);
                    }
                }
            }
            System.out.println(canAshishWin ? "Ashish" : "Vivek");
        }
    }
}
