#include<bits/stdc++.h>
#define L long long
using namespace std;
L solve(string s, L n, L p, L a){
    L ans = 0;
    if(p == 1)
        for(L i = 0; i < n; i++)
            if(s[i] == '.')
                ans++;
    else{
        vector<L> v;
        for(L i = 0; i < n; i++)
            if(s[i] == '.')
                v.push_back(i);
        L m = v.size();
        if(m == 0)
            ans = 0;
        else{
            sort(v.begin(), v.end());
            L cnt = 0;
            if(m == 1){
                if(a > 1)
                    ans = 1;
                else
                    ans = 0;
            }
            else{
                for(L i = 0; i < m; i++){
                    cnt++;
                    if(i == m - 1 || v[i] + 1 != v[i + 1]){
                        if(cnt >= p){
                            ans += cnt / p;
                            cnt = cnt % p;
                        }
                        else if(cnt < p && a > 0){
                            L x = p - cnt;
                            if(x <= a){
                                ans++;
                                a -= x;
                            }
                            cnt = 0;
                        }
                    }
                }
            }
        }
    }
    return ans;
}
int main(){
    L n, p, a;
    cin >> n >> p >> a;
    string s;
    cin >> s;
    cout << solve(s, n, p, a) << '\n';
    return 0;
}
