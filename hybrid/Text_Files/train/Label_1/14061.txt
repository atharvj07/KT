#include <bits/stdc++.h>
using namespace std;

const int maxn = 500+5;
int a[maxn];

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  int delta = 0;
  for (int i = 0; i < n-1; i++)
    if (a[i] == a[i+1]) {
      delta++;
      i++;
    }
  cout << n - delta << endl;
  return 0;
}
