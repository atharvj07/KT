import sys
input=sys.stdin.readline
n=int(input())
adj=[[] for i in range(n)]
a,b,c=[[int(j) for j in input().split()]for i in range(n)]
ds=[0]*n
for i in range(n-1):
    x,y=map(int,input().split())
    adj[x-1].append(y-1)
    adj[y-1].append(x-1)
    ds[x-1]+=1
    ds[y-1]+=1
ans=0
def dfs(cur,p,l):
    global ans
    if b[cur]!=c[cur]:
        ans+=a[p]*l
        b[cur]=c[cur]
    for nx in adj[cur]:
        if nx!=p:
            dfs(nx,cur,l+1)
def find(cur,p):
    for nx in adj[cur]:
        if nx!=p:
            if b[cur]==c[cur] and b[nx]==c[nx]:
                continue
            if b[cur]!=b[nx] or c[cur]!=c[nx]:
                return False
            find(nx,cur)
    return True
if not find(0,-1):
    print(-1)
else:
    dfs(0,-1,1)
    print(ans)
