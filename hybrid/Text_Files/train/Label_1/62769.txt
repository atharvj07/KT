n=int(input())
l=sorted([int(i) for i in input().split()])
r=[""]*n; t=0
for i in range(n):
    for j in range(n):
        if(l[i]==r[j]): t=j
        if(r[j]==''):
            r[j]=str(l[i])
            break
        if(l[i]*3==r[j]):
            r[t]=str(l[i])
            t=j; break
        if(l[i]*2==r[j]):
            r[t]=str(l[i])
            t=j; break
t=-1
for i in range(n):
    if(r[i]!=""):
        t=i
        break
for i in range(n):
    if(r[i]==''):
        r[i]=str(l[(t+i)%n])
print(" ".join(r))
