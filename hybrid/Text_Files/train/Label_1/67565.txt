n = int(input())
r = list(map(int, input().split()))
b = list(map(int, input().split()))

r_points = 0
b_points = 0
common_problems = 0

for i in range(n):
    if r[i] == 1 and b[i] == 0:
        r_points += 1
    elif r[i] == 0 and b[i] == 1:
        b_points += 1
    elif r[i] == 1 and b[i] == 1:
        common_problems += 1

if r_points == 0:
    print(-1)
else:
    print(max(1, (b_points + r_points - common_problems) // r_points + 1))
