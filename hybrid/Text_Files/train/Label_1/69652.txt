import sys
n, m = map(int, input().split())
l = [0] + list(map(int, input().split()))
a = [0] * (n + 1)
for i in range(1, m + 1):
    a[l[i]] = (a[l[i]] + i) % n
    l[i] = (l[i] + a[l[i]]) % n
if set(a[1:]) == set(range(1, n + 1)):
    for i in range(1, n + 1):
        sys.stdout.write(str(a[i]) + ' ')
else:
    print(-1)
