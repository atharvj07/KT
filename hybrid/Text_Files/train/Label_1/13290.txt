#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) { cin >> x; }
    map<int, int> cnt;
    int l = 0, r = 0, ans = 0;
    while (l < n) {
        while (r < n && cnt[a[r]] < 2) { cnt[a[r]]++, r++; }
        if (r - l > ans) { ans = r - l; }
        if (cnt[a[l]] == 1) { r--, cnt[a[l]]--; }
        l++;
    }
    cout << ans << endl;
}
