import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    static int n, v, e;
    static int[] a, b;
    static List<int[]>[] adjList;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        v = Integer.parseInt(st.nextToken());
        e = Integer.parseInt(st.nextToken());
        a = new int[n + 1];
        b = new int[n + 1];
        adjList = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            adjList[i] = new ArrayList<>();
        }
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) {
            b[i] = Integer.parseInt(st.nextToken());
        }
        for (int i = 0; i < e; i++) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            adjList[x].add(new int[]{y, 0});
            adjList[y].add(new int[]{x, 0});
        }
        if (!bfs()) {
            System.out.println("NO");
            return;
        }
        System.out.println(cnt);
        for (int[] edge : edges) {
            System.out.println(edge[0] + " " + edge[1] + " " + edge[2]);
        }
    }

    static int cnt = 0;
    static List<int[]> edges = new ArrayList<>();

    static boolean bfs() {
        int[] q = new int[n + 1];
        int head = 0, tail = 0;
        boolean[] vis = new boolean[n + 1];
        q[tail++] = 1;
        vis[1] = true;
        while (head < tail) {
            int cur = q[head++];
            for (int[] next : adjList[cur]) {
                int[] edge = new int[]{cur, next[0], 0};
                if (!vis[next[0]]) {
                    q[tail++] = next[0];
                    vis[next[0]] = true;
                }
                if (a[cur] + b[next[0]] <= v) {
                    edge[2] = Math.min(v - a[cur], b[next[0]]);
                    a[cur] += edge[2];
                    a[next[0]] -= edge[2];
                    cnt++;
                    edges.add(edge);
                    if (cnt > 2 * n * n) {
                        return false;
                    }
                } else if (a[next[0]] + b[cur] <= v) {
                    edge[2] = Math.min(v - a[next[0]], b[cur]);
                    a[cur] -= edge[2];
                    a[next[0]] += edge[2];
                    cnt++;
                    edges.add(edge);
                    if (cnt > 2 * n * n) {
                        return false;
                    }
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (a[i] != b[i]) {
                return false;
            }
        }
        return true;
    }
}
