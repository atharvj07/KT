#include <iostream>

using namespace std;

int main() {
    long long k, l, r, t, x, y;
    cin >> k >> l >> r >> t >> x >> y;

    if (t * x > r - l)
        cout << "No" << endl;
    else if ((y + max(0ll, (r - k - t * x)) % (r - l) - x) / (r - l) * (r - l) +
             max(0ll, (r - k - t * x)) % (r - l) >= l)
        cout << "Yes" << endl;
    else
        cout << "No" << endl;
    return 0;
}
