#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<vector<int>> permutations(k, vector<int>(n));
  for (int i = 0; i < k; i++) {
    for (int j = 0; j < n; j++) {
      cin >> permutations[i][j];
    }
  }

  vector<int> lcs(n, 0);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < k; j++) {
      if (i == 0 || j == 0) {
        lcs[i] = (permutations[j][i] == permutations[0][i]);
      } else if (permutations[j][i] == permutations[0][i]) {
        lcs[i] = lcs[i - 1] + 1;
      } else {
        lcs[i] = max(lcs[i - 1], lcs[i]);
      }
    }
  }

  cout << lcs[n - 1] << endl;

  return 0;
}
