n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)
vis = [0] * n
def dfs(v, p, rot):
    vis[v] = 1
    ret = [v]
    for u in adj[v]:
        if u == p: continue
        if not vis[u]:
            ret += dfs(u, v, rot+1)
        elif (rot+1)%2 != (vis[u]+1)%2:
            return []
    return ret
rot = dfs(0, -1, 0)
print("Yes" if rot else "No")
