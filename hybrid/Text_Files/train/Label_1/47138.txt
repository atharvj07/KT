import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long[][] holidays = new long[N][4];
        for (int i = 0; i < N; i++) {
            holidays[i][0] = sc.nextInt();
            holidays[i][1] = sc.nextInt();
            holidays[i][2] = sc.nextInt();
            holidays[i][3] = sc.nextInt();
        }
        long[] congestion = new long[366];
        for (int i = 0; i < 366; i++) {
            for (long[] holiday : holidays) {
                int sm = (int) Math.max(1, holiday[0] - holiday[2]);
                int em = (int) Math.min(12, holiday[0] + holiday[2] - 1);
                int sd = (int) Math.max(1, holiday[1] - holiday[2]);
                int ed = (int) Math.min(30 + ((holiday[0] == 2) ? 1 : 0), holiday[1] + holiday[2] - 1);
                if (i >= sm && i <= em && i % 30 >= sd && i % 30 <= ed) {
                    congestion[i] = Math.max(congestion[i], holiday[3]);
                } else {
                    long days = Math.abs(i - em * 30 - holiday[1]);
                    congestion[i] = Math.max(congestion[i], holiday[3] - Math.min(holiday[2], days));
                }
            }
        }
        long ans = congestion[0];
        for (long c : congestion) {
            ans = Math.min(ans, c);
        }
        System.out.println(ans);
    }
}
