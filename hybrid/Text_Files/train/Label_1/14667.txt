#include <bits/stdc++.h>
using namespace std;
const long long INF = 1e18;
long long a[1<<19], n;
int main() {
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; if (a[i] == -1) a[i] = INF; }
    for (n >>= 1, sort(a, a + n); n > 1; n >>= 1)
        for (int i = 0, j = n; i < n; i++, j++)
            a[i] = min(a[i], a[j] + a[i << 1]);
    cout << (a[0] == INF ? 0LL : a[0]);
}
