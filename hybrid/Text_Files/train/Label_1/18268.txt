#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int a, b;
    std::cin >> a >> b;
    std::vector<int> ans;
    if (b == 0) {
      std::cout << 1 << '\n';
      continue;
    } else if (a == 1) {
      std::cout << 2 << '\n';
      continue;
    } else {
      ans.push_back(0);
      if (a == 2 && (b & 1) == 1) {
        a++;
      }
      b ^= a;
      ans.push_back(b);
      ans.push_back(a);
    }
    std::cout << ans.size() << '\n';
  }
  return 0;
}
