#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, q, type, x, t, unread = 0;
  cin >> n >> q;
  unordered_set<int> read;
  while (q--) {
    cin >> type;
    if (type == 1) {
      cin >> x;
      ++unread;
    } else if (type == 2) {
      cin >> x;
      if (read.count(x)) --unread;
      read.insert(x);
    } else {
      cin >> t;
      unread -= min(t, unread);
    }
    cout << unread << endl;
  }
  return 0;
}
