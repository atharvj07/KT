#include<bits/stdc++.h>

using namespace std;

typedef long long ll;

#define maxm 305

ll vis[maxm];
vector<ll>ev;

int main(){

    ll m,t,r,i,j,k,l,n,x,sum=0;
    cin>>m>>t>>r;

    for(i=1;i<=m;i++){

        cin>>x;
        ev.push_back(x);
        ev.push_back(x);
    }

    ev.push_back(-1);
    ev.push_back(0);

    sort(ev.begin(),ev.end());

    ll now=0,cnt=0;

    for(i=0;i<ev.size();i++){

        if(vis[ev[i]]==1) continue;
        if(now>=ev[i]){

            cnt+=t;
            now=ev[i]+t;
        }
        vis[ev[i]]=1;
        while(cnt>=t){

            cnt-=t;
            now-=t;
        }
        now++;
        cnt++;
    }

    if(cnt<r) cout<<-1<<endl;
    else cout<<now<<endl;

    return 0;
}
