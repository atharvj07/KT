import java.util.Scanner;

public class PashasGame {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        boolean[][] grid = new boolean[n][m];
        int moveNumber = 0;

        for (int i = 0; i < k; i++) {
            int row = scanner.nextInt() - 1;
            int col = scanner.nextInt() - 1;

            grid[row][col] = true;

            if (checkLoss(grid, row, col)) {
                moveNumber = i + 1;
                break;
            }
        }

        System.out.println(moveNumber);
    }

    private static boolean checkLoss(boolean[][] grid, int row, int col) {
        if (row > 0 && grid[row - 1][col] && col > 0 && grid[row][col - 1]) {
            return true;
        }
        if (row < grid.length - 1 && grid[row + 1][col] && col > 0 && grid[row][col - 1]) {
            return true;
        }
        if (row > 0 && grid[row - 1][col] && col < grid[0].length - 1 && grid[row][col + 1]) {
            return true;
        }
        if (row < grid.length - 1 && grid[row + 1][col] && col < grid[0].length - 1 && grid[row][col + 1]) {
            return true;
        }
        return false;
    }
}
