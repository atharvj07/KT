#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,s,t;cin>>n>>s>>t;
    vector<int>p(n);
    for(int&x:p)cin>>x;
    if(s==t){
        cout<<0;
        return 0;
    }
    vector<int>r(n+1,-1);
    for(int i=0;i<n;i++)r[p[i]]=i;
    int c=0;
    int pos=s;
    while(pos!=t){
        pos=r[pos];
        if(pos==-1){
            cout<<-1;
            return 0;
        }
        c++;
    }
    cout<<c;
}
