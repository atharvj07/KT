import heapq
inf = float('inf')

n, p = map(int, input().split())
a = list(map(int, input().split()))
b = [0] * n

q = []
for i in range(n):
    heapq.heappush(q, (a[i], i))

t = 0
while q:
    t0, i = heapq.heappop(q)
    b[i] = t
    t = max(t, t0 + p)
    if i + 1 < n and a[i + 1] <= t:
        heapq.heappush(q, (a[i + 1], i + 1))
print(*b)
