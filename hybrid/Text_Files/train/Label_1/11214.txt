#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[2005], dp[2005];
  cin >> n;
  dp[0] = 0;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    dp[i] = -1e9;
    if (dp[i - 1] > 0) dp[i] = dp[i - 1] + a[i];
    dp[i] = max(dp[i], a[i]);
  }
  int ans = -1;
  for (int i = 1; i <= n; i++) {
    if (dp[i] >= 0) ans = i;
  }
  cout << ans << endl;
}
