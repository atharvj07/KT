#include <bits/stdc++.h>
#include <algorithm>
#include <set>
#include <sstream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    unordered_set<int> s;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      s.insert(x);
    }
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
      if (s.count(i)) cnt++;
    }
    cout << cnt << endl;
  }
}
