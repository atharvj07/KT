for _ in range(int(input())):
    n, m = map(int, input().split())
    M = [list(map(int, input().split())) for _ in range(n)]
    a = sum(sum(x) for x in M)
    if (a & 1) or (a / 2 <= sum(M[(n + 1) // 2 - 1]) + sum(M[i][(m + 1) // 2 - 1] for i in range(n))): print('Ashish')
    else: print('Jeel')
