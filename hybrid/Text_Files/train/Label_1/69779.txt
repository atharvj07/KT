import sys

input = sys.stdin.readlines()
n = int(input[0])
s = input[1]
b = [int(x) for x in input[2].split()]
lst = {i: [] for i in range(1, 27)}
for i in range(n):
    lst[ord(s[i]) - 96].append(i)
ans = 0
j = n // 2
for i in range(1, 27):
    while lst[i] and j < n:
        ans += b[lst[i].pop(0)]
        j += 2
    while lst[i] and j >= 0:
        ans += b[lst[i].pop()]
        j -= 2
print(ans)
