#include <iostream>
#include <algorithm>

using namespace std;

int gcd(int a, int b) {
    if (b == 0) return a;
    return gcd(b, a % b);
}

int main() {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    if (a * d < b * c) swap(a, b), swap(c, d);
    int g = gcd(a * d - b * c, a * d);
    cout << (a * d - b * c) / g << "/" << a * d / g << endl;
    return 0;
}
