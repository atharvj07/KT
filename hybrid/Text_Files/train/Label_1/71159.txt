def solve(n, a, b):
    if a != 1 and b != 1:
        return "NO"
    if a == 1 and b == 1 and n > 1:
        return "NO"
    
    matrix = [[1]*n for _ in range(n)]
    for i in range(n):
        matrix[i][i] = 0
    
    if a == 1:
        for i in range(n-b):
            for j in range(i+1, n-b):
                matrix[i][j] = matrix[j][i] = 0
    else:
        for i in range(n-a):
            for j in range(i+1, n-a):
                matrix[i][j] = matrix[j][i] = 1
        for i in range(n):
            for j in range(n):
                matrix[i][j] = 1 - matrix[i][j]
    
    result = "YES\n" + "\n".join("".join(map(str, row)) for row in matrix)
    return result

n, a, b = map(int, input().split())
print(solve(n, a, b))
