import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        List<Pair> questions = new ArrayList<>();
        long l = 0, r = 0;
        for (int i = 0; i < n; i++) {
            Pair q = new Pair(in.next().charAt(0), in.nextLong());
            q.ans = in.next().charAt(0);
            questions.add(q);
            if (q.ans == 'Y') {
                if (q.sign == '>' || q.sign == '=') l = Math.max(l, q.x + 1);
                if (q.sign == '<' || q.sign == '=') r = Math.min(r, q.x - 1);
            }
        }
        if (l <= r)
            System.out.println(l);
        else
            System.out.println("Impossible");
    }

    static class Pair {
        char sign;
        long x;
        char ans;

        public Pair(char sign, long x) {
            this.sign = sign;
            this.x = x;
        }
    }
}
