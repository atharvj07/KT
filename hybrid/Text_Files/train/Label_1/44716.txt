import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        while (m-- > 0) {
            int l = sc.nextInt() - 1, r = sc.nextInt() - 1;
            int sum = 0;
            for (int i = l; i <= r; i++) {
                sum += a[i];
            }
            System.out.println(sum == 0 ? 1 : 0);
        }
    }
}
