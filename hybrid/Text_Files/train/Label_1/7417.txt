#include <cstdio>
#include <vector>
#include <algorithm>

using namespace std;

int n, m;
vector<vector<int>>g;

int main() {
  scanf("%d", &n);
  m = n + 1;
  g.resize(m, vector<int>(m));
  for (int i = 1; i < m; ++i) {
    for (int j = 1; j <= i; ++j) {
      scanf("%d", &g[j][i]);
      g[i][j] = g[j][i];
    }
  }
  int P[] = {0, 2, 3, 1, 5, 4};
  int ans = 0;
  for (int i = 1; i < m; ++i) {
    for (int j = 1; j <= i; ++j) {
      ans += g[P[i]][P[j]];
      if (P[i] < P[j]) ans += g[P[i]][P[j]];
      swap(P[i], P[j]);
    }
  }
  printf("%d\n", ans);
  return 0;
}
