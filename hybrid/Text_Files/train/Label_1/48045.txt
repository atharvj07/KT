import java.util.*;
import java.math.*;

public class Main {

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        BigInteger p = cin.nextBigInteger(), k = cin.nextBigInteger();
        if (!p.equals(p.remainder(k))) {
            System.out.println(-1);
            return;
        }
        int last = p.mod(k).intValue();
        p = p.divide(k);
        List<Integer> ans = new ArrayList<>();
        while (!p.equals(BigInteger.ZERO)) {
            ans.add(last);
            if (last == 0 && p.mod(k).equals(BigInteger.ZERO)) {
                p = p.divide(k).subtract(BigInteger.ONE);
            } else {
                p = p.divide(k);
            }
            last = p.mod(k).intValue();
        }
        ans.add(last);
        Collections.reverse(ans);
        System.out.println(ans.size());
        for (int i : ans) {
            System.out.print(i + " ");
        }
        System.out.println();
    }
}
