#include <bits/stdc++.h>

using namespace std;

#define ll long long

const int maxn = 100005;
const int maxc = 30;
const ll mod = 1000000007;

int n;
string s[maxn];
ll h[maxn][maxc], pw[maxn][maxc];

ll addmod(ll a, ll b) {
  return (a + b) % mod;
}

ll mulmod(ll a, ll b) {
  return a * b % mod;
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> s[i];
  }
  pw[0][0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < maxc; j++) {
      pw[i][j] = mulmod(pw[i - 1][j], 31);
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = s[i].size() - 1; j >= 0; j--) {
      for (int k = 0; k < maxc; k++) {
        h[i][k] = addmod(h[i][k], mulmod(s[i][j] - 'a' + 1, pw[j][k]));
      }
    }
  }
  ll ans = 0;
  for (int i = 1; i <= n; i++) {
    ll cur = 0;
    ll hs[maxc];
    memset(hs, 0, sizeof hs);
    for (int j = 0; j < maxc; j++) {
      hs[j] = h[i][j];
    }
    for (int j = 0; j < s[i].size() / 2; j++) {
      cur = addmod(cur, h[i][s[i][j] - 'a' + 1]);
      cur = mulmod(cur, pw[s[i].size() - 2 * j - 1][s[i][j] - 'a' + 1]);
      for (int k = 0; k < maxc; k++) {
        hs[k] = addmod(hs[k], -mulmod(s[i][j] - 'a' + 1, pw[s[i].size() - 2 * j - 1][k]));
        for (int l = 1; l <= n; l++) {
          if (l == i) {
            continue;
          }
          bool ok = true;
          for (int p = 0; p < maxc; p++) {
            if (addmod(hs[p], h[l][p]) != 0) {
              ok = false;
              break;
            }
          }
          if (!ok) {
            continue;
          }
          for (int p = (int)s[i].size() - 2 * j - 1; p < (int)s[i].size(); p++) {
            if (s[i][p] != s[l][p - ((int)s[i].size() - 2 * j - 1)]) {
              ok = false;
              break;
            }
          }
          if (ok) {
            ans++;
          }
        }
      }
    }
    for (int j = 0; j < s[i].size(); j++) {
      if (j > 0) {
        cur = mulmod(cur, pw[s[i].size() - 1 - j][s[i][j] - 'a' + 1]);
      }
      for (int k = 0; k < maxc; k++) {
        hs[k] = addmod(hs[k], mulmod(s[i][j] - 'a' + 1, pw[s[i].size() - 1 - j][k]));
        for (int l = 1; l <= n; l++) {
          if (l == i) {
            continue;
          }
          bool ok = true;
          for (int p = 0; p < maxc; p++) {
            if (addmod(hs[p], h[l][p]) != 0) {
              ok = false;
              break;
            }
          }
          if (!ok) {
            continue;
          }
          for (int p = 0; p < s[l].size(); p++) {
            if (s[i][j + p] != s[l][p]) {
              ok = false;
              break;
            }
          }
          if (ok) {
            ans++;
          }
        }
      }
    }
  }
  cout << ans / 2 << endl;
  return 0;
}
