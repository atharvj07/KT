#include <iostream>
#include <cstdio>
using namespace std;
int a[55][55],n,ans;
int main()
{
    cin>>n;
    if(n==1) {puts("1");return 0;}
    if(n==3) {puts("2 1 4\n3 5 7\n6 9 8");return 0;}
    if(n&1) {ans=n*n+1;a[1][1]=ans+n*n-1;ans++;}
    else {ans=n*n;a[1][n]=ans+n*n-1;ans++;}
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++) if(a[i][j]==0)
        {
            a[i][j]=ans;
            ans++;
        }
    }
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++) printf("%d ",a[i][j]);
        printf("\n");
    }
    return 0;
}
