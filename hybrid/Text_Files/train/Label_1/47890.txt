import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a = n / 2 - 1, b = n - a;
        while (gcd(a, b) != 1) {
            a--;
            b++;
        }
        System.out.println(a + " " + b);
    }

    public static int gcd(int a, int b) {
        return b == 0 ? a : gcd(b, a % b);
    }
}
