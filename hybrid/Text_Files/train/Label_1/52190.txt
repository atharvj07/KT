import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int y = scanner.nextInt();
        int w = scanner.nextInt();

        int numerator = 1;
        int denominator = 2;
        if (y == w) {
            numerator = 1;
            denominator = 3;
        } else if (y > w) {
            numerator = 0;
            denominator = 1;
        }

        System.out.println(numerator + "/" + denominator);
    }
}
