from collections import defaultdict

n, m = [int(x) for x in input().split()]
debts = defaultdict(lambda: defaultdict(int))

for _ in range(m):
    a, b, d = [int(x) for x in input().split()]
    debts[a][b] += d

total = 0
for a in debts:
    for b in debts[a]:
        if a != b:
            total += debts[a][b]

nc = 0
res = []

for a in debts:
    s = sorted(debts[a].items(), key=lambda x: x[1], reverse=True)
    for b, d in s:
        if a == b or d == 0:
            continue
        nc += 1
        res.append((a, b, d))
        for c in debts[b]:
            if c == a:
                continue
            total -= debts[c][a]
            debts[c][a] = 0
            debts[a][c] += debts[c][a]
        debts[b][a] = 0
        debts[a][b] -= d
        if debts[a][b] == 0:
            del debts[a][b]

print(nc)
for a, b, d in res:
    print(a, b, d)
    total -= d

print(total)
