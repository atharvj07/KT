import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        int min = Integer.MAX_VALUE;
        int cur = 0;
        for (char c : s.toCharArray()) {
            if (c == '-') {
                cur--;
            } else {
                cur++;
            }
            min = Math.min(min, cur);
        }
        System.out.println(min);
    }
}
