#include <iostream>
#include <vector>
using namespace std;

struct Box {
  int w, h;
};

struct Stair {
  int height;
  vector<Box> boxes;
};

int main() {
  int n, m, h, w;
  cin >> n;
  vector<Stair> stairs(n);
  for (auto &stair : stairs) { cin >> stair.height; }
  cin >> m;
  for (int i = 0; i < m; i++) {
    cin >> w >> h;
    int maxh = -1;
    for (int j = 0; j < w; j++) { maxh = max(maxh, stairs[j].height); }
    int k = 0;
    while (k < stairs[w - 1].boxes.size() &&
           maxh + h <= stairs[w - 1].boxes[k].h + stairs[w - 1].height) {
      k++;
    }
    stairs[w - 1].boxes.insert(stairs[w - 1].boxes.begin() + k, Box{w, h});
    cout << maxh + h << endl;
  }
  return 0;
}
