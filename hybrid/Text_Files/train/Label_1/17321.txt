#include <bits/stdc++.h>
using namespace std;
#define fast ios::sync_with_stdio(0);cin.tie(0);cout.tie(0);
#define endl '\n'
#define ll long long
const int md=998244353;
ll fpow(ll x,ll p){
    ll ans=1;while(p){if(p%2)ans=(ans*x)%md;x=(x*x)%md;p/=2;}return ans;
}
int main() {
    fast;
    int n,k;cin>>n>>k;
    vector<int> v(n,0);
    for(int i=0;i<n;i++){cin>>v[i];}
    vector<vector<int>> dp(k+1,vector<int>(n+1,0));
    vector<vector<int>> mx(k+1,vector<int>(n+1,0));
    dp[1][0]=1;
    for(int i=1;i<=k;i++){
        for(int j=0;j<n;j++){
            mx[i][j]=v[j];
            for(int p=j;p<n;p++){
                mx[i][j]=max(mx[i][j],v[p]);
                dp[i][j]=(dp[i][j]+dp[i-1][p])%md;
            }
        }
    }
    cout<<mx[k][0]<<" "<<dp[k][0]<<endl;
}
