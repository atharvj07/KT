#include <iostream>
#include <utility>
#include <vector>

int main() {
    long n;
    std::cin >> n;
    long curr;
    std::vector<long> heights(n + 1);
    for (long i = 0; i < n; i++) {
        std::cin >> curr;
        heights[i + 1] = curr;
    }
    long q;
    std::cin >> q;
    while (q--) {
        long w, h;
        std::cin >> w >> h;
        std::cout << heights[w] + h << '\n';
        heights[w] = heights[w] + h;
    }
}
