import java.util.*;
import java.lang.*;

class Main {
  static HashMap<Long, Long> freq = new HashMap<>();

  public static int run(long[] a) {
    long mex = 0;
    for (long i : a) {
      freq.put(i, freq.getOrDefault(i, 0L) + 1);
      while (freq.containsKey(mex)) {
        mex++;
      }

      if (i == mex)
        return (int) i + 1;
    }
    return -1;
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextLong();
    }
    in.close();
    System.out.print(run(a));
  }
}
