#include <iostream>
#include <cassert>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; ++i) {
    cin >> p[i];
  }
  vector<pair<int, int>> st;
  vector<int> l_max(n);
  for (int i = 0; i < n; ++i) {
    while (!st.empty() && st.back().first < p[i]) {
      st.pop_back();
    }
    if (!st.empty()) {
      l_max[i] = st.back().second;
    }
    st.emplace_back(p[i], i);
  }
  st.clear();
  vector<int> r_max(n);
  for (int i = n - 1; i >= 0; --i) {
    while (!st.empty() && st.back().first <= p[i]) {
      st.pop_back();
    }
    if (!st.empty()) {
      r_max[i] = st.back().second;
    }
    st.emplace_back(p[i], i);
  }
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    if (l_max[i] <= i && i <= r_max[i]) {
      ++ans;
    }
  }
  cout << ans << endl;
  return 0;
}
