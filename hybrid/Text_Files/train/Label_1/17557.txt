#include <iostream>

typedef long long ll;

static const ll mod = 1e9 + 7;
ll pow2[100001];
ll inv2[100001];

ll fastpow(ll base, ll exp) {
    if (!exp) return 1;
    if (exp == 1) return base % mod;
    ll half = fastpow(base, exp >> 1);
    half = (half * half) % mod;
    if (exp & 1) half = (half * base) % mod;
    return half;
}

void precompute() {
    pow2[0] = 1;
    for (int i = 1; i < 100001; i++) pow2[i] = (pow2[i - 1] << 1) % mod;
    inv2[100000] = fastpow(pow2[100000], mod - 2);
    for (int i = 100000 - 1; i >= 0; i--) inv2[i] = (inv2[i + 1] << 1) % mod;
}

int main() {
    precompute();
    int n, m;
    std::cin >> n >> m;
    std::cout << (pow2[n] * inv2[n - 1]) % mod * (pow2[m] * inv2[m - 1]) % mod << std::endl;
    return 0;
}
