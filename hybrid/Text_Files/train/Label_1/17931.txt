#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 10005;
int a[N][105];
int main() {
  int n, a_, b;
  cin >> n >> a_ >> b;
  if (n % 2 != a_ * b % 2) return cout << -1, 0;
  int t = (n % 2);
  int u = (a_ * b - n) / 2;
  for (int i = 0; i < a_; ++i) {
    for (int j = 0; j < b; ++j) {
      if (t) {
        cout << (i % 2 ^ j % 2 ? n : 0) + 1 << ' ';
        t--;
      } else if (u) {
        cout << (i % 2 ^ j % 2 ? n : 0) + 1 << ' ';
        u--;
      } else {
        cout << 0 << ' ';
      }
    }
    cout << endl;
  }
}
