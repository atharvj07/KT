import math

n, s = map(int, input().split())
a = []
for i in range(n):
    a += [list(map(int, input().split()))]
a.sort(key=lambda x: (x[0] ** 2 + x[1] ** 2, x[1]))
# a = [x[0] for x in a]
x = s
for i in a:
    if i[2] + x >= 1e6:
        print(math.sqrt((i[0] ** 2 + i[1] ** 2)))
        break
    x += i[2]
else:
    print(-1)
