import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class Main {

    static class Edge {

        int to;
        int weight;

        public Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    static class Castle {

        int numWarriorsToCapture;
        int numWarriorsToHire;
        int importance;

        public Castle(int numWarriorsToCapture, int numWarriorsToHire, int importance) {
            this.numWarriorsToCapture = numWarriorsToCapture;
            this.numWarriorsToHire = numWarriorsToHire;
            this.importance = importance;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());

        Castle[] castles = new Castle[n + 1];
        for (int i = 1; i <= n; i++) {
            st = new StringTokenizer(br.readLine());
            int numWarriorsToCapture = Integer.parseInt(st.nextToken());
            int numWarriorsToHire = Integer.parseInt(st.nextToken());
            int importance = Integer.parseInt(st.nextToken());
            castles[i] = new Castle(numWarriorsToCapture, numWarriorsToHire, importance);
        }

        List<Edge>[] adjList = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adjList[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adjList[u].add(new Edge(v, castles[v].importance));
        }

        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);

        int ans = dfs(1, k, adjList, castles, dp);
        System.out.println(ans);
    }

    private static int dfs(int currentCastle, int numWarriors, List<Edge>[] adjList, Castle[] castles, int[] dp) {
        if (dp[currentCastle] != -1) {
            return dp[currentCastle];
        }

        int maxImportance = 0;
        if (numWarriors >= castles[currentCastle].numWarriorsToCapture) {
            maxImportance = castles[currentCastle].importance;
            numWarriors += castles[currentCastle].numWarriorsToHire;

            for (Edge edge : adjList[currentCastle]) {
                maxImportance = Math.max(maxImportance, dfs(edge.to, numWarriors - 1, adjList, castles, dp) + edge.weight);
            }
        }

        dp[currentCastle] = maxImportance;
        return maxImportance;
    }
}
