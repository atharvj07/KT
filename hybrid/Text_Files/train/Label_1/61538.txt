from sys import stdin, stdout

mod = int(1e9)+7
def modpow(a: int, b: int) -> int:
    ret = 1
    while b:
        if b&1: ret = (ret * a) %mod
        a = (a * a) %mod
        b >>= 1
    return ret
def inv(x: int) -> int:
    return modpow(x, mod-2)

n, m = map(int, stdin.readline().split())
p = [*map(int, stdin.readline().split())]
foe = [[] for _ in range(n+1)]
for _ in range(m):
    a, b = map(int, stdin.readline().split())
    foe[a].append(b)
    foe[b].append(a)
    
idx = [0] * (n+1)
for i in range(1, n+1):
    idx[p[i-1]] = i
    
inv_sum = [1] * (n+1)
for i in range(n-1, 0, -1):
    inv_sum[i] = (inv_sum[i+1] + inv(i+1)) % mod
    
cnt = [0] * (n+1)
for i in range(1, n+1):
    cnt[i] = (cnt[i-1] + 1) % mod
    
ans = 0
for i in range(1, n+1):
    app = 1
    for e in foe[i]:
        if idx[e] < idx[i]:
            app = (app * cnt[idx[e]-1]) % mod
            app = (app * inv_sum[idx[e]]) % mod
    
    rem = 1
    for e in foe[i]:
        if idx[e] > idx[i]:
            rem = (rem * cnt[idx[e]-1]) % mod
            rem = (rem * inv_sum[idx[e]]) % mod
            
    ans = (ans + (app*rem) % mod) % mod
print(ans)
