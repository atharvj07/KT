#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> gov(k);
  for (int i = 0; i < k; i++) {
    cin >> gov[i];
  }
  vector<bool> vis(n + 1, false);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    vis[u] = vis[v] = true;
  }
  int cnt = 0;
  for (int i = 0; i < k; i++) {
    if (vis[gov[i]]) {
      cnt++;
    }
  }
  cout << m + k - cnt << endl;
  return 0;
}
