t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    p = list(map(int, input().split()))
    total_changes = 0
    current_price = p[0]
    for i in range(1, n):
        max_allowed_increase = (current_price * k) // 100
        if p[i] > max_allowed_increase:
            required_base = (p[i] * 100 + k - 1) // k
            changes = required_base - current_price
            total_changes += changes
            current_price += changes
        current_price += p[i]
    print(total_changes)
