#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
const ll inf = 1e18;

int main() {
    int n;
    cin >> n;
    vector<ll> x(n), y(n), z(n), p(n), q(n);
    for (int i = 0; i < n; i++) {
        cin >> x[i] >> y[i] >> z[i] >> p[i] >> q[i];
    }
    vector<int> good;
    for (int i = 0; i < n; i++) {
        bool flag = true;
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            for (int k = 0; k < n; k++) {
                if (j == k || i == k) continue;
                ll dot1 = (x[i] - x[j]) * (x[k] - x[j]) +
                          (y[i] - y[j]) * (y[k] - y[j]) +
                          (z[i] - z[j]) * (z[k] - z[j]) +
                          (p[i] - p[j]) * (p[k] - p[j]) +
                          (q[i] - q[j]) * (q[k] - q[j]);
                ll dot2 = (x[i] - x[j]) * (x[j] - x[k]) +
                          (y[i] - y[j]) * (y[j] - y[k]) +
                          (z[i] - z[j]) * (z[j] - z[k]) +
                          (p[i] - p[j]) * (p[j] - p[k]) +
                          (q[i] - q[j]) * (q[j] - q[k]);
                ll norm1 = (x[i] - x[j]) * (x[i] - x[j]) +
                            (y[i] - y[j]) * (y[i] - y[j]) +
                            (z[i] - z[j]) * (z[i] - z[j]) +
                            (p[i] - p[j]) * (p[i] - p[j]) +
                            (q[i] - q[j]) * (q[i] - q[j]);
                ll norm2 = (x[k] - x[j]) * (x[k] - x[j]) +
                            (y[k] - y[j]) * (y[k] - y[j]) +
                            (z[k] - z[j]) * (z[k] - z[j]) +
                            (p[k] - p[j]) * (p[k] - p[j]) +
                            (q[k] - q[j]) * (q[k] - q[j]);
                if (dot1 * norm2 < norm1 * dot2) {
                    flag = false;
                    break;
                }
            }
            if (!flag) break;
        }
        if (flag) good.push_back(i + 1);
    }
    cout << good.size() << '\n';
    for (int i : good) cout << i << '\n';
}
