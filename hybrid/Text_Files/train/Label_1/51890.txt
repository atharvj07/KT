import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String home = br.readLine();
    String away = br.readLine();
    int n = Integer.parseInt(br.readLine());
    Map<Integer, Integer> homeYellows = new HashMap<>();
    Map<Integer, Integer> awayYellows = new HashMap<>();
    for (int i = 0; i < n; i++) {
      String[] parts = br.readLine().split(" ");
      int time = Integer.parseInt(parts[0]);
      String team = parts[1];
      int num = Integer.parseInt(parts[2]);
      String card = parts[3];
      if (card.equals("y")) {
        if (team.equals("h")) {
          homeYellows.put(num, time);
        } else {
          awayYellows.put(num, time);
        }
      } else {
        if (team.equals("h")) {
          if (homeYellows.containsKey(num)) {
            System.out.println(home + " " + num + " " + time);
          }
        } else {
          if (awayYellows.containsKey(num)) {
            System.out.println(away + " " + num + " " + time);
          }
        }
      }
    }
  }
}
