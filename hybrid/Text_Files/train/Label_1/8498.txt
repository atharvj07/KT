#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007
vector<int> v; ll n, k;
bool check(int m){
    ll c = 0, sum = 0, ans;
    for(int i=n-1; i>=0; i--){
        if(sum + v[i] <= k){
            c++;
            sum += v[i];
            if(c==m) break;
        }
        else sum = v[i];
    }
    if(c==m) return 1;
    else return 0;
}
int main(){
    cin >> n >> k;
    v.resize(n);
    for(int i=0; i<n; i++){
        cin >> v[i];
    }
    sort(v.rbegin(), v.rend());
    ll hi = n, lo = 1, ans = lo;
    while(lo <= hi){
        ll mid = (lo+hi)>>1;
        if(check(mid)){
            ans = mid;
            hi = mid-1;
        }
        else lo = mid+1;
    }
    if(ans == 1){
        cout << ans << endl;
        cout << "1" << endl;
        return 0;
    }
    if(check(ans)){
        ll x = n-1, y = ans, a = 0, b = 0, c = 0;
        while(x>=0 && y>0){
            if(c == 0){
                c = v[x];
                x--;
                a++;
            }
            else if(c + v[x] <= k){
                c += v[x];
                x--;
                a++;
            }
            else{
                c = v[x];
                x--;
                b++;
            }
            if(a==y && b==ans-y){
                a = b = c = 0;
                y--;
            }
        }
        ll temp = a*(ans-1);
        temp %= mod;
        for(int i=1; i<=ans-1; i++){
            temp *= i;
            temp %= mod;
        }
        ll res = 1;
        for(int i=1; i<=b; i++){
            res *= i;
            res %= mod;
        }
        temp *= res;
        temp %= mod;
        cout << ans << endl;
        cout << temp << endl;
    }
    else{
        cout << "-1" << endl;
        cout << "0" << endl;
    }
    return 0;
}
