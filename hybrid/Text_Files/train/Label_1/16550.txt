#include <iostream>
#include <vector>

std::vector<int> lps(const std::string& p) {
    const int m = p.length();
    std::vector<int> lps(m, 0);
    int len = 0;
    lps[0] = 0;
    int i = 1;
    while (i < m) {
        if (p[i] == p[len]) {
            len++;
            lps[i] = len;
            i++;
        } else {
            if (len != 0) {
                len = lps[len - 1];
            } else {
                lps[i] = 0;
                i++;
            }
        }
    }
    return lps;
}

std::string solve(const std::string& a) {
    const std::string b = "trygub";
    std::vector<int> la = lps(a);
    std::vector<int> lb = lps(b);
    int n = a.length();
    int m = b.length();
    std::string c = a;
    int j = 0;
    for (int i = 0; i < n; i++) {
        while (j > 0 && a[i] != b[j]) {
            j = lb[j - 1];
        }
        if (a[i] == b[j]) j++;
        if (j == m) {
            int k = j - 1;
            while (k >= 0) {
                if (k >= la[i]) c[i - k] = a[i];
                k--;
            }
            j = lb[j - 1];
        }
    }
    return c;
}

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n;
        std::cin >> n;
        std::string a;
        std::cin >> a;
        std::cout << solve(a) << '\n';
    }
    return 0;
}
