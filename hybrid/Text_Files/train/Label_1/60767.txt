import sys
from math import atan2, sqrt
input=sys.stdin.readline

n=int(input())
c=[(int(x),int(y)) for x,y in zip(*[[int(x) for x in j.split()]for j in[input()for i in range(n)]])]

cnt=0
for i in c:
    for j in c[c.index(i)+1:]:
        for k in c[c.index(j)+1:]:
            if (k[0]-i[0])*(j[1]-i[1])==(k[1]-i[1])*(j[0]-i[0]):
                if(atan2(j[1]-i[1],j[0]-i[0])==atan2(k[1]-j[1],k[0]-j[0]) and sqrt((i[0]-j[0])**2+(i[1]-j[1])**2)==sqrt((j[0]-k[0])**2+(j[1]-k[1])**2)):cnt+=1
print(cnt)
