import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextInt();
        long k = in.nextInt();
        long a = in.nextInt();
        long b = in.nextInt();
        long ans = 0;
        while (n > 1) {
            if (n % k == 0 && a + b < b) {
                ans += a;
                n--;
            } else {
                ans += b;
                n /= k;
            }
        }
        System.out.println(ans);
    }
}
