import sys

n = int(input())
a = list(map(int, input().split()))
m = int(input())
times = []
for i in range(m):
    times.append(list(map(int, input().split())))

a.sort()
times.sort()

t = 0
j = 0
for i in a:
    if j < m and times[j][0] <= t:
        t = max(t, times[j][1])
        j += 1
    if j == m:
        print(-1)
        sys.exit(0)
    t += i

print(t)
