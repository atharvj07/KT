#include <bits/stdc++.h>
using namespace std;

#define fi first
#define se second
#define ll long long

typedef pair<int, int> pi;

const int INF = 1e9;
const ll LINF = 1e18;
const int MAXN = 3e5 + 10;

int main() {
    int n;
    cin >> n;
    vector<pi> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].fi >> v[i].se;
    }
    sort(v.rbegin(), v.rend());
    vector<bool> used(n, false);
    vector<int> ans;
    bool type = 1;
    for (int i = 0; i < n; i++) {
        int a = v[i].fi;
        int b = v[i].se;
        bool ok = 0;
        for (int j = 0; j < n; j++) {
            if (used[j]) {
                continue;
            }
            int aa = v[j].fi;
            int bb = v[j].se;
            if (((type && aa < a && b < bb) || (!type && aa > a && b > bb)) && !(aa == a && bb == b)) {
                ok = 1;
                used[j] = used[i] = 1;
                ans.push_back(j + 1);
                ans.push_back(i + 1);
                break;
            }
        }
        if (!ok) {
            for (int j = 0; j < n; j++) {
                if (used[j]) {
                    continue;
                }
                int aa = v[j].fi;
                int bb = v[j].se;
                if (((type && aa > a && b < bb) || (!type && aa < a && b > bb)) && !(aa == a && bb == b)) {
                    ok = 1;
                    used[j] = used[i] = 1;
                    ans.push_back(i + 1);
                    ans.push_back(j + 1);
                    break;
                }
            }
        }
        if (ok) {
            type ^= 1;
        }
    }
    cout << ans.size() << "\n";
    for (int x : ans) {
        cout << x << " ";
    }
    cout << "\n";
    return 0;
}
