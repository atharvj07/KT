#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
const int mod=1e9+7;
const int N=3e5;
const int oo=1e9+5;
int32_t main(){
    ll n,m,x,y,ans=0;
    cin>>n>>m;
    vector<pair<ll,ll>>v;
    for(int i=0;i<m;i++){
        cin>>x>>y;
        x--,y--;
        v.pb({x,y});
    }
    sort(v.begin(),v.end());
    ll prv=0;
    for(int i=0;i<m;i++){
        if(v[i].first!=prv)prv=v[i].first,ans+=v[i].second-i;
    }
    cout<<ans;

}
