#include <bits/stdc++.h>
using namespace std;
using ll = int64_t;
using ld = long double;
void solve() {
  ll n, m, a = 0, b = 0;
  cin >> n >> m;
  for (int i = 0; i < n; ++i) {
    char ch;
    cin >> ch;
    a <<= 1;
    a += ch - '0';
  }
  for (int i = 0; i < m; ++i) {
    char ch;
    cin >> ch;
    b <<= 1;
    b += ch - '0';
  }
  ll ans = 0;
  while (b) {
    ans = (ans + (a & b)) % 998244353;
    b >>= 1;
  }
  cout << ans << '\n';
}
int main() {
  solve();
}
