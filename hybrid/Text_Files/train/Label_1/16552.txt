#include <iostream>
#include <vector>
#include <algorithm>
#include <set>

const int MAXN = 2e5 + 10;

int n, k, a[MAXN];
std::set<int> alive;

int main() {
  std::cin >> n >> k;
  for (int i = 1; i <= n; ++i) {
    std::cin >> a[i];
    alive.insert(i);
  }
  std::sort(a + 1, a + n + 1);
  for (int i = 1; i <= n; ++i)
    for (int j = i + 1; j <= n; ++j) {
      if (a[i] > a[j] && a[i] <= a[j] + k) alive.erase(j);
    }
  std::cout << alive.size() << std::endl;
}
