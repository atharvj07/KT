#include <cstdio>
#include <algorithm>
using namespace std;

int bitcount(int x) {
  int ret = 0;
  while (x) {
    if (x & 1) ret++;
    x >>= 1;
  }
  return ret;
}

int main() {
  int q;
  scanf("%d", &q);
  while (q--) {
    int n;
    scanf("%d", &n);
    int a[100];
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    sort(a, a + n);
    int cnt = 0;
    for (int i = 0; i < n; i++) if (a[i] < 2048) cnt += bitcount(2048 / a[i]);
    printf((cnt & 1) ? "NO\n" : "YES\n");
  }
}
