import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] p = new int[n];
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (i > 0) p[i] = p[i - 1];
            if (i > 0 && s.charAt(i) == s.charAt(i - 1)) p[i]++;
        }
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                if (i == 0 || s.charAt(i - 1) != s.charAt(j) && (i == j || s.charAt(i) != s.charAt(j - 1)) && (j + 1 == n || s.charAt(j) != s.charAt(j + 1))) {
                    if (i == 0 || j == n - 1 || s.charAt(i - 1) != s.charAt(j + 1)) {
                        count++;
                    } else if (p[j] > p[i - 1]) {
                        count++;
                    }
                }
            }
        }
        System.out.println(count);
    }
}
