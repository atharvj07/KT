import java.util.*;

public class Main {
    static class Snuke implements Comparable<Snuke> {
        public int n;
        public int s;

        public Snuke(int n) {
            this.n = n;
            this.s = sumDigits(n);
        }

        private int sumDigits(int n) {
            int sum = 0;
            while (n > 0) {
                sum += n % 10;
                n /= 10;
            }
            return sum;
        }

        @Override
        public int compareTo(Snuke other) {
            double ratio1 = (double) n / s;
            double ratio2 = (double) other.n / other.s;
            return Double.compare(ratio1, ratio2);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        PriorityQueue<Snuke> snukeNumbers = new PriorityQueue<>();
        for (int n = 1; n <= 1000000000; n++) {
            Snuke snuke = new Snuke(n);
            if (snukeNumbers.size() < k) {
                snukeNumbers.add(snuke);
            } else {
                Snuke smallest = snukeNumbers.peek();
                if (snuke.compareTo(smallest) < 0) {
                    snukeNumbers.poll();
                    snukeNumbers.add(snuke);
                } else {
                    break;
                }
            }
        }
        while (!snukeNumbers.isEmpty()) {
            System.out.println(snukeNumbers.poll().n);
        }
    }
}
