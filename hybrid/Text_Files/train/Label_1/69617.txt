from collections import deque

n, m = map(int, input().split())

f = [[False for j in range(n+1)] for i in range(n+1)]

for i in range(m):
    u, v = map(int, input().split())
    f[u][v] = f[v][u] = True

q = deque([(x,[]) for x in range(1,n+1)])
ans = []

while len(q)>1:
    x,l = q.popleft()
    ans.append(x)
    
    for i in range(1,n+1):
        f[x][i] = f[i][x] = True
        
    for j in range(len(q)):
        y,ll = q[j]
        if f[x][y]:
            ll.append(x)
            q[j] = (y,ll)
    
    while q and f[q[0][0]][q[-1][0]]:
        q.popleft()
        q.append(q.popleft())

print(len(ans))
print(*ans)
