#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a, b;
    cin >> n >> a >> b;
    if (a == b) return cout << 1, 0;
    int t = 0;
    while (n > 1) {
        if ((n & 1) && a != n && b != n) n++;
        n >>= 1;
        t++;
        if ((n & 1) && a == n) a = 0;
        if ((n & 1) && b == n) b = 0;
        if (a == b) return cout << t, 0;
    }
    cout << "Final!";
}
