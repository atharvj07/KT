#include <iostream>
using namespace std;
int main() {
    int n; cin >> n;
    int a[n], ans[n];
    for (int i = 0; i < n; ++i) cin >> a[i], --a[i];
    bool vis[n];
    for (int i = 0; i < n; ++i) {
        ans[i] = 0; vis[i] = 0;
    }
    for (int i = 0; i < n; ++i) {
        if (vis[i]) continue;
        int p = i;
        while (!vis[p]) {
            vis[p] = true;
            p = (p + a[p]) % n;
        }
        ans[i] = (ans[i] == ans[p] ? 0 : 1);
    }
    for (int i = 0; i < n; ++i) cout << (ans[i] ? 'A' : 'B');
    cout << "\n";
    return 0;
}
