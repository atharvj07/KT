import heapq
n,m=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
a.sort()
heapq.heapify(b)
q=int(input())
for i in range(q):
    x,y,z=map(int,input().split())
    if x==1:
        a[y-1]=z
    else:
        k=0
        while k<len(b) and b[0]>=z:
            heapq.heappop(b)
            k+=1
    a.sort()
    if a and (not b or a[-1]>b[0]):
        print(a[-1])
    else:
        print(-1)
