#include <bits/stdc++.h>
using namespace std;

long long n, a[200005], x;

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a, a + n);
  x = a[0];
  for (int i = 1; i < n; i++) {
    x = __gcd(x, a[i] - a[0]);
  }
  cout << ((a[n - 1] - a[0] + x) / x) << " " << x << endl;
  return 0;
}
