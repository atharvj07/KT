import java.util.*;
public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    ArrayList<Integer> a = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      int tmp = sc.nextInt();
      a.add(tmp);
    }
    int[] color = new int[8];
    for (int i = 0; i < n; i++) {
      int rating = a.get(i);
      if (rating < 400) {
        color[0] = 1;
      } else if (rating < 800) {
        color[1] = 1;
      } else if (rating < 1200) {
        color[2] = 1;
      } else if (rating < 1600) {
        color[3] = 1;
      } else if (rating < 2000) {
        color[4] = 1;
      } else if (rating < 2400) {
        color[5] = 1;
      } else if (rating < 2800) {
        color[6] = 1;
      } else if (rating < 3200) {
        color[7] = 1;
      }
    }
    int min = 0, max = 0;
    for (int i = 0; i < 8; i++) {
      min += color[i];
      max += (color[i] > 0 ? 1 : 0);
    }
    System.out.println(min + " " + max);
  }
}
