#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef vector<ll> vi;
typedef pair<ll, ll> pii;
typedef unordered_map<ll, ll> mii;
#define inf 1e18
#define mod 1000000007
#define pb push_back
#define ff first
#define ss second
#define foru(i,a,b) for(int i=a; i<=b; i++)
#define ford(i,a,b) for(int i=a; i>=b; i--)
#define all(a) a.begin(),a.end()

void dfs(ll u, ll pr, vi adj[], vector<bool>& vis, vector<vector<ll>>& dp, const vector<vi>& G) {
  vis[u] = 1;
  for (ll i : adj[u]) {
    if (i == pr || vis[i] == 0) continue;
    dfs(i, u, adj, vis, dp, G);
  }
  dp[u][1] = 1;
  for (ll i : adj[u]) {
    if (i == pr || vis[i] == 0) continue;
    for (ll j = 2; j < G[0].size(); j++) {
      dp[u][j] = (dp[u][j] + (dp[u][j - 1] * dp[i][G[0].size() - j]) % mod) % mod;
    }
  }
}

int main() {
  ll n; cin >> n;
  vector<vi> G(n, vi(n));
  foru(i, 0, n - 1) {
    foru(j, 0, n - 1) {
      cin >> G[i][j];
    }
  }
  vi adj[n];
  foru(i, 0, n - 1) {
    foru(j, 0, n - 1) {
      if (i == j || G[i][j] == 0) continue;
      adj[i].pb(j);
    }
  }
  vector<bool> vis(n, 0);
  vector<vector<ll>> dp(n, vector<ll>(G[0].size(), 0));
  dfs(0, 0, adj, vis, dp, G);
  cout << dp[0][G[0].size() - 1] % mod << "\n";
  return 0;
}
