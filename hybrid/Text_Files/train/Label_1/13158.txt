#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  vector<int> x(8), y(8);
  for (int i = 0; i < 8; i++) cin >> x[i] >> y[i];
  sort(x.begin(), x.end());
  sort(y.begin(), y.end());
  cout << (x[1] == x[4] && y[1] == y[4] ? "respectable" : "ugly") << endl;
  return 0;
}
