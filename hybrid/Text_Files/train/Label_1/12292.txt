#include <iostream>
#include <queue>
#include <stack>
#include <vector>

using namespace std;

int main() {
  int k;
  cin >> k;
  if (k == 2)
    cout << "NO" << endl;
  else {
    cout << "YES" << endl;
    int n = 2 * k + 1, m = 2 * k * k;
    cout << n << " " << m << endl;
    for (int i = 1; i <= k; i++) {
      for (int j = 1; j <= k; j++) {
        if (i == j)
          continue;
        cout << i << " " << j + k << endl;
        cout << i + k << " " << j << endl;
      }
    }
  }
  return 0;
}
