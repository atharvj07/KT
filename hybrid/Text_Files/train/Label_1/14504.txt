#include <bits/stdc++.h>
using namespace std;
#define int long long

const int maxn = 1e5 + 5;
int k, n;
int a[maxn];
int id[maxn];
int m = 0;

int32_t main() {
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    if (id[a[i]] == 0) {
      if (m < k) {
        id[a[i]] = ++m;
      } else {
        for (int j = k - 1; j >= 1; j--) id[a[i - j]] = id[a[i - j + 1]];
        id[a[i]] = 1;
      }
    }
  }
  cout << m << '\n';
  for (int i = 1; i <= m; i++) cout << id[a[n - i]] << ' ';
  return 0;
}
