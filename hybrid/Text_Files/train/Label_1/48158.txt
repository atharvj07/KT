import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        Arrays.sort(a);
        long ans = 0;
        int idx = 0;
        for (int i = 0; i < m; i++) {
            if (idx == n) {
                break;
            }
            ans += a[idx] >> i;
            idx++;
        }
        for (int i = idx; i < n; i++) {
            ans += a[i];
        }
        System.out.println(ans);
    }
}
