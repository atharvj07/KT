#include <cstdio>
using namespace std;

constexpr int MM = 3e5 + 1;
inline char gc() {
    static char buf[MM], *p1 = buf, *p2 = buf;
    return p1 == p2 && (p2 = (p1 = buf) + fread(buf, 1, MM, stdin), p1 == p2) ? EOF : *p1++;
}
inline int rd() {
    int ans = 0;
    char c = gc();
    while (c < '0' || c > '9')
        c = gc();
    while (c >= '0' && c <= '9')
        ans = ans * 10 + (c ^ 48), c = gc();
    return ans;
}

int n, a[MM], cnt[6], dir = -1;
bool vis[MM];
char ch;

int main() {
    int t = rd();
    while (t--) {
        n = rd();
        cnt[0] = cnt[2] = cnt[3] = cnt[4] = cnt[5] = 0;
        for (int i = 0; i < n; i++) {
            ch = gc();
            if (ch == '>') {
                dir = -1, a[i] = 1;
            } else if (ch == '<') {
                dir = 1, a[i] = 2;
            } else
                a[i] = 0;
            cnt[a[i]]++;
        }
        for (int i = 1; i <= 5; i++) {
            cnt[i] += cnt[i - 1];
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if ((a[i] == 0) || (a[i] == 1 && dir == 1) || (a[i] == 2 && dir == -1))
                ans++;
        }
        printf("%d\n", ans);
    }
    return 0;
}
