#include <bits/stdc++.h>
using namespace std;

int main() {
  long long int t;
  cin >> t;
  while (t--) {
    long long int n, m;
    cin >> n >> m;
    long long int a[m];
    for (long long int i = 0; i < m; i++) { cin >> a[i]; }
    long long int sum = 0;
    for (long long int i = 0; i < m; i++) { sum += a[i]; }
    if (sum < n) {
      cout << -1 << endl;
      continue;
    }
    long long int ans = 0;
    for (long long int i = 0; i < m; i++) {
      while (a[i] > n) {
        ans++;
        a[i] /= 2;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
