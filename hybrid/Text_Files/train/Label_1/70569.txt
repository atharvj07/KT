from collections import deque

n = int(input())
a = deque(map(int, input().split()))

if a[0] > a[-1]:
    a.reverse()

flag = True
while a:
    if flag:
        if a[0] > a[-1]:
            a.pop()
        else:
            a.popleft()
    else:
        if a[0] > a[-1]:
            a.popleft()
        else:
            a.pop()
    flag = not flag

if flag:
    print("Alice")
else:
    print("Bob")
