#include <bits/stdc++.h>
using namespace std;

int getCost(string &a, string &b) {
  int cost = 0;
  for (int i = 0; i < a.size(); i++) {
    if (a[i] != b[i]) {
      if (a[i] == '1')
        cost++;
      else
        cost += 2;
    }
  }
  return cost;
}

int main() {
  string a, b;
  cin >> a >> b;
  cout << getCost(a, b) << endl;
  return 0;
}
