import java.util.*;
public class Main {
    static final int mod = 1000000007;
    static int[][] pwr;
    static int[][] count;
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.nextLine();
        String t = input.nextLine();
        pwr = new int[s.length()+1][t.length()+1];
        count = new int[s.length()+1][t.length()+1];
        for(int i=0;i<=s.length();i++)
            for(int j=0;j<=t.length();j++)
                pwr[i][j] = i>0 && j>0 ? (pwr[i-1][j]*2)%mod : 1;
        for(int i=1;i<=s.length();i++)
            for(int j=1;j<=t.length();j++)
                count[i][j] = s.charAt(i-1) == t.charAt(j-1) ?
                        (count[i-1][j]+count[i][j-1]-count[i-1][j-1] + pwr[i-1][j-1])%mod
                        : count[i][j-1];
        System.out.println(count[s.length()][t.length()]);
    }
}
