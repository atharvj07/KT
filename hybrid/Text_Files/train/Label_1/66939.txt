import string

s = input()
t = input()
t = list(t)
s = list(s.replace('?', 'a'))
i = j = 0
ct = len(t)
while i < len(s) and j != ct:
    if s[i] == t[j]:
        j += 1
    i += 1
ans = ''.join(s)
i = 0
ct = len(t)
j = len(s) - 1
while i != ct and j >= 0:
    if s[j] == t[i]:
        i += 1
    j -= 1
ans = max(ans, ''.join(s))
print(ans)
