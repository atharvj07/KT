#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, t, ans = 0;
  cin >> n >> t;
  vector<int> level(n + 1, 0);
  while (t--) {
    for (int i = 1; i <= n; i++) {
      int p = (level[i] + 1) / 2;
      level[i] = (level[i] + 1) % 2;
      level[i + 1] += p;
      level[i - 1] += p;
    }
  }
  for (int i = 0; i <= n; i++) ans += level[i] == 0;
  cout << ans << endl;
  return 0;
}
