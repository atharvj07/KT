import java.util.*;

class Main {
    static final long MOD = 1_000_000_007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int k = sc.nextInt() - 1;
        long[][] dp = new long[h + 1][1 << w];
        dp[0][1 << k] = 1;
        for (int i = 1; i <= h; i++) {
            for (int mask = 0; mask < 1 << w; mask++) {
                if (dp[i - 1][mask] == 0) continue;
                for (int j = 0; j < w; j++) {
                    if ((mask >> j & 1) == 1) continue;
                    if (j != 0 && (mask >> (j - 1) & 1) == 1) continue;
                    dp[i][mask | (1 << j)] += dp[i - 1][mask];
                    dp[i][mask | (1 << j)] %= MOD;
                }
            }
        }
        System.out.println(dp[h][(1 << w) - 1]);
    }
}
