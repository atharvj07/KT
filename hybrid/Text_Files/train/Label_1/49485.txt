import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

public class Main {

    private static final int MOD = 998244353;
    private static int[] arr;
    private static long[] memo;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(reader.readLine());
        Map<Integer, List<Integer>> adj = new HashMap<>();
        for (int i = 0; i < n; i++) adj.put(i + 1, new ArrayList<>());
        for (int i = 0; i < n - 1; i++) {
            String[] edge = reader.readLine().split(" ");
            int u = Integer.parseInt(edge[0]);
            int v = Integer.parseInt(edge[1]);
            adj.get(u).add(v);
            adj.get(v).add(u);
        }

        arr = new int[n + 1];
        memo = new long[n + 1];
        for (int i = 1; i <= n; i++) arr[i] = i;
        System.out.println(dfs(1, 0, adj));
    }

    private static long dfs(int curr, int par, Map<Integer, List<Integer>> adj) {
        if (memo[curr] != 0) return memo[curr];
        Set<Integer> set = new HashSet<>();
        for (int next : adj.get(curr)) if (next != par) set.add((int) dfs(next, curr, adj));
        long ans = 0;
        for (int num : set) {
            ans += num;
            if (ans >= MOD) ans -= MOD;
        }
        ans++;
        if (ans >= MOD) ans -= MOD;
        return memo[curr] = ans;
    }
}
