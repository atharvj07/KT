#include <bits/stdc++.h>

using namespace std;
typedef pair<int, int> ii;

const int N = 1e5 + 10;
int n, k, a, b;
char ans[N];

bool solve() {
  priority_queue<ii> pq;
  pq.push({k + 1, 0});
  pq.push({k + 1, 1});
  for (int i = 0; i < n; ++i) {
    auto p = pq.top();
    pq.pop();
    p.first--;
    if (p.second == 0)
      ans[i] = 'G', a--;
    else
      ans[i] = 'B', b--;
    if (p.first > 0) pq.push(p);
    if (a < 0 || b < 0) return false;
  }
  return true;
}

int main() {
  cin >> n >> k >> a >> b;
  if (!solve())
    cout << "NO" << endl;
  else
    cout << ans << endl;
  return 0;
}
