#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    sort(v.rbegin(), v.rend());
    int L = 0, R = 0;
    for (int x : v) {
      R = max(R, L + x);
      L = max(L - x, 0);
    }
    cout << R - L << "\n";
  }
  return 0;
}
