#include <bits/stdc++.h>
using namespace std;
#define ll long long

const ll maxn = 2e5 + 5;
ll freq[2][maxn], a[maxn], need[maxn];

int main() {
    ll n, m, even = 0, odd = 0;
    cin >> n >> m;
    for (ll i = 1; i <= n; i++) {
        cin >> a[i];
        if (a[i] % 2 == 0)
            even++;
        else
            odd++;
    }
    for (ll i = 1; i <= m; i++) {
        freq[0][i] = freq[1][i] = 0;
    }
    for (ll i = 1; i <= n; i++) {
        freq[a[i] % 2][a[i]]++;
    }
    need[0] = ((n - even) % 2 == 0) ? ((n - even) / 2) : ((n - even - 1) / 2);
    need[1] = ((n - odd) % 2 == 0) ? ((n - odd) / 2) : ((n - odd - 1) / 2);
    ll ex = 0;
    vector<pair<ll, ll>> ans;
    for (ll i = 1; i <= m; i++) {
        if (i % 2 == 0 && freq[0][i] < need[0]) {
            ll x = min(need[0] - freq[0][i], freq[1][i]);
            need[0] -= x;
            need[1] += x;
            ex += x * 2;
            for (ll j = 1; j <= n; j++) {
                if (a[j] == i && freq[0][i] < need[0]) {
                    ans.push_back({j, i});
                    freq[0][i]++;
                    freq[1][i]--;
                    a[j] = i + 1;
                }
            }
        } else if (i % 2 == 1 && freq[1][i] < need[1]) {
            ll x = min(need[1] - freq[1][i], freq[0][i]);
            need[1] -= x;
            need[0] += x;
            ex += x * 2;
            for (ll j = 1; j <= n; j++) {
                if (a[j] == i && freq[1][i] < need[1]) {
                    ans.push_back({j, i});
                    freq[1][i]++;
                    freq[0][i]--;
                    a[j] = i - 1;
                }
            }
        }
    }
    if (need[0] > 0 || need[1] > 0)
        cout << "-1" << endl;
    else {
        cout << ex << endl;
        for (ll i = 1; i <= n; i++) {
            cout << a[i] << " ";
        }
        cout << endl;
    }
    return 0;
}
