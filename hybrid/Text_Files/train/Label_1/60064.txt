n,x=map(int,input().split())
m=2**n-1
if x==m:
    print(0)
else:
    smx=2**n
    for i in range(1,n):
        if((smx^x)%m!=0):
            smx=2**i
    print(smx)
    st=1
    while st*2<=smx:
        st*=2
    l=[]
    while smx:
        if smx>=st:
            l.append(st)
            smx^=st
        st//=2
    for i in l:
        print(i,end=" ")
