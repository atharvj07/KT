import sys

input = sys.stdin.readline

m = int(input())
seq, pref, copies = [1], [0], [0]
for _ in range(m):
    t, *a = map(int, input().split())
    if t == 1:
        seq.append(a[0])
    else:
        l, c = a
        pref.append(len(seq) - l)
        copies.append(c)
for _ in range(int(input())):
    i = int(input()) - 1
    while pref[i] > 0:
        i = pref[i] - 1 + copies[i] * (i + l - pref[i])
    print(seq[i], end=' ')
