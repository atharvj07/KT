import sys

n = int(input())
adj = [[]]
for _ in range(n):
    adj.append([])

for i in range(n - 1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

ans = [1] * (n + 1)

def dfs(v, p):
    cur = 1
    for u in adj[v]:
        if u == p: continue
        dfs(u, v)
        cur = max(cur, ans[u] + 1)
    ans[v] = cur

dfs(1, 1)
print(max(ans))
