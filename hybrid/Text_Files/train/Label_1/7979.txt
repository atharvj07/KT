#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin >> n;
    int a[n + 5], b[2], o = 0, e = 0;
    for(int i = 1; i <= n; i++){
        cin >> a[i];
        if(a[i] == 0) continue;
        if(a[i] % 2) o++;
        else e++;
    }
    cout << abs(o - e) / 2 << endl;
    return 0;
}
