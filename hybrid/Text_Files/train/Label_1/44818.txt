import java.util.*;

public class Main {
    static final int INF = Integer.MAX_VALUE / 2;

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int v = cin.nextInt();
        ArrayList<Node> graph[] = new ArrayList[v];
        for (int i = 0; i < v; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < v; i++) {
            int u = cin.nextInt();
            int degree = cin.nextInt();
            for (int j = 0; j < degree; j++) {
                int w = cin.nextInt();
                int weight = cin.nextInt();
                graph[u].add(new Node(w, weight));
            }
        }
        PriorityQueue<Node> q = new PriorityQueue<>((a, b) -> a.w - b.w);
        q.add(new Node(0, 0));
        int dist[] = new int[v];
        Arrays.fill(dist, INF);
        dist[0] = 0;
        while (!q.isEmpty()) {
            Node cur = q.poll();
            if (cur.w > dist[cur.idx]) continue;
            for (Node next : graph[cur.idx]) {
                int newW = cur.w + next.w;
                if (newW < dist[next.idx]) {
                    dist[next.idx] = newW;
                    q.add(new Node(next.idx, newW));
                }
            }
        }
        for (int i = 0; i < v; i++) {
            System.out.printf("%d %d\n", i, dist[i] == INF ? -1 : dist[i]);
        }
    }

    static class Node {
        int idx, w;

        public Node(int idx, int w) {
            this.idx = idx;
            this.w = w;
        }
    }
}
