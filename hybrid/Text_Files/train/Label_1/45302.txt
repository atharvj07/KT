import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] a = new int[N - 1];
        int[] b = new int[N - 1];
        for (int i = 0; i < N - 1; i++) {
            a[i] = sc.nextInt() - 1;
            b[i] = sc.nextInt() - 1;
        }
        String s = sc.next();
        int[][] g = new int[N][];
        for (int i = 0; i < N; i++) {
            g[i] = new int[0];
        }
        for (int i = 0; i < N - 1; i++) {
            g[a[i]] = Arrays.copyOf(g[a[i]], g[a[i]].length + 1);
            g[a[i]][g[a[i]].length - 1] = b[i];
            g[b[i]] = Arrays.copyOf(g[b[i]], g[b[i]].length + 1);
            g[b[i]][g[b[i]].length - 1] = a[i];
        }
        int[] dist = new int[N];
        Arrays.fill(dist, -1);
        Queue<Integer> q = new LinkedList<Integer>();
        for (int i = 0; i < N; i++) {
            if (s.charAt(i) == '1') {
                q.add(i);
                dist[i] = 0;
            }
        }
        while (!q.isEmpty()) {
            int v = q.poll();
            for (int u : g[v]) {
                if (dist[u] == -1) {
                    q.add(u);
                    dist[u] = dist[v] + 1;
                }
            }
        }
        int maxDist = 0;
        for (int i = 0; i < N; i++) {
            maxDist = Math.max(maxDist, dist[i]);
        }
        long ans = 1L << maxDist;
        System.out.println(ans);
    }
}
