import sys

n = int(input())
l = []

for i in range(n):
    l.append(input().split())

l.sort(key=lambda x: int(x[1]))

ans = []
mn = -2e9
mx = 2e9

for j in range(n):

    x = int(l[j][1])
    y = l[j][2]

    if y == "Y":
        if l[j][0] == ">=":
            mn = max(mn, x)
        if l[j][0] == ">":
            mn = max(mn, x + 1)

    else:
        if l[j][0] == "<=":
            mx = min(mx, x)
        if l[j][0] == "<":
            mx = min(mx, x - 1)

if mn > mx:
    print("Impossible")
    sys.exit()

print(mn)
