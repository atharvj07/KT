#include<bits/stdc++.h>
using namespace std;

int main(){
    int n;
    cin>>n;
    if(n<=3){
        cout<<-1;
        return 0;
    }
    int a[n+1];
    a[1]=n;
    a[2]=1;
    for(int i=3;i<=n;i++){
        if(i%2==0){
            a[i]=i/2;
        }
        else{
            a[i]=i+1;
        }
    }
    for(int i=1;i<=n;i++){
        cout<<a[i]<<" ";
    }
    return 0;
}
