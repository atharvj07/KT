from collections import defaultdict
import heapq

n, m = map(int, input().split())
a = list(map(int, input().split()))
g = defaultdict(list)
for i in range(m):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)
s = int(input()) - 1

q = [(-a[s], s)]
b = set([s])
ans = 0
while q:
    cur, p = heapq.heappop(q)
    ans -= cur
    for v in g[p]:
        if v not in b:
            heapq.heappush(q, (-a[v], v))
            b.add(v)

print(ans)
