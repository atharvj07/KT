#include <algorithm>
#include <iostream>
#include <stack>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n), ans(n);
  for (int i = 0; i < n; ++i) std::cin >> a[i];
  std::stack<int> s;
  for (int i = 0; i < n; ++i) {
    while (!s.empty() && a[s.top()] > a[i]) {
      ans[s.top()] += a[i];
      s.pop();
    }
    ans[i] += a[i];
    s.push(i);
  }
  for (int i = 0; i < n; ++i) std::cout << ans[i] << ' ';
  return 0;
}
