#include <bits/stdc++.h>
using namespace std;
struct stud {
  int a, b, id;
};
bool cmp(stud x, stud y) { return x.a + x.b > y.a + y.b; }
int main() {
  int n, p, s;
  cin >> n >> p >> s;
  vector<stud> v(n);
  for (int i = 0; i < n; i++) {
    int aa, bb;
    cin >> aa >> bb;
    v[i] = {aa, bb, i + 1};
  }
  sort(v.begin(), v.end(), cmp);
  int ans = 0;
  vector<int> pa, sa;
  for (int i = 0; i < p; i++) pa.push_back(v[i].id), ans += v[i].a;
  for (int i = p; i < p + s; i++) sa.push_back(v[i].id), ans += v[i].b;
  cout << ans << '\n';
  for (auto p : pa) cout << p << ' ';
  cout << '\n';
  for (auto p : sa) cout << p << ' ';
  return 0;
}
