#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct pt { int x, v; };

int main() {
    int n; cin >> n;
    vector<pt> v(n);
    for (int i = 0; i < n; ++i)
        cin >> v[i].x >> v[i].v;
    sort(v.begin(), v.end(),
         [](const pt& lhs, const pt& rhs) {
             return lhs.x < rhs.x;
         });
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            ans += min(v[i].v, v[j].v) * (v[j].x - v[i].x);
        }
    }
    cout << ans << "\n";
    return 0;
}
