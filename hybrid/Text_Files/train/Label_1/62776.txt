t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))

    # Sort in descending order
    a.sort(reverse=True)

    alice_score = sum(a[i] for i in range(0, n, 2) if a[i] % 2 == 0)
    bob_score = sum(a[i] for i in range(1, n, 2) if a[i] % 2 == 1)

    if alice_score > bob_score:
        print("Alice")
    elif alice_score < bob_score:
        print("Bob")
    else:
        print("Tie")
