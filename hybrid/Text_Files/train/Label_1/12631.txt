#include <bits/stdc++.h>
using namespace std;
int const INF = 1e9;
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin >> t;
    for (int z = 0; z < t; z++) {
        long long n;
        cin >> n;
        if (n % 3 != 0) {
            cout << -1 << endl;
            continue;
        }
        int ans = 0;
        while (n % 6 == 0) {
            n /= 6;
            ans++;
        }
        while (n % 3 == 0) {
            n *= 2;
            n /= 3;
            ans++;
        }
        if (n == 1)
            cout << ans << endl;
        else
            cout << -1 << endl;
    }
    return 0;
}
