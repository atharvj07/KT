import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a1 = sc.nextInt();
        int b1 = sc.nextInt();
        int a2 = sc.nextInt();
        int b2 = sc.nextInt();
        int a3 = sc.nextInt();
        int b3 = sc.nextInt();
        sc.close();

        boolean canFit = false;

        if (a1 >= a2 + a3 && b1 >= Math.max(b2, b3)) {
            canFit = true;
        } else if (a1 >= Math.max(b2, b3) && b1 >= a2 + a3) {
            canFit = true;
        }

        System.out.println(canFit ? "YES" : "NO");
    }
}
