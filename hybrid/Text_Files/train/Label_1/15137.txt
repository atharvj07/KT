#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  ll n, x;
  cin >> n >> x;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  ll cur = 0;
  for (ll i = 0; i < n; i++) {
    if (cur + a[i] <= x) {
      cur += a[i];
    } else {
      cout << "NO" << endl;
      return 0;
    }
  }
  if (cur == x)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
