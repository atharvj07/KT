#include<bits/stdc++.h>
using namespace std;
string s;
int main()
{
    cin>>s;
    int l=s.size(),x=0;
    for(int i=0;i<l;i++)
    {
        if(s[i]=='0')
        {
            x=1;
            break;
        }
    }
    if(x==1)
        cout<<-1<<endl;
    else
        cout<<l<<endl;
    return 0;
}
