#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  int n, m, d;
  cin >> n;
  ll t1[n], t2[n];
  for (int i = 0; i < n; i++) {
    cin >> t1[i];
    t2[i] = t1[i];
  }
  sort(t1, t1 + n, greater<ll>());
  cin >> m;
  for (int i = 0; i < m; i++) {
    cin >> d;
    t2[i] = d;
  }
  sort(t2, t2 + n, greater<ll>());
  ll a = 0, b = 0;
  for (int i = 0; i < n; i++) {
    a += t1[i];
    b += t2[i];
  }
  cout << a << ":" << b << "\n";
  return 0;
}
