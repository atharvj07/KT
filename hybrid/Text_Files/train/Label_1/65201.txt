n = int(input())
t = list(map(int, input().split()))

cnt = [0, 0, 0]
for i in t:
    cnt[i - 1] += 1

w = min(cnt)

print(w)

for i in range(w):
    for j in range(3):
        while cnt[j] > 0:
            print(t.index(j + 1) + 1, end=' ')
            t[t.index(j + 1)] = 0
            cnt[j] -= 1
            break
    print()
