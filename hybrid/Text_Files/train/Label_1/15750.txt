#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<string> s(2 * n - 2);
    for (int i = 0; i < 2 * n - 2; i++) { cin >> s[i]; }
    sort(s.begin(), s.end());
    string t = s[0];
    for (int i = 1; i < 2 * n - 2; i++) {
        if (s[i].size() == t.size() + 1 && s[i].substr(0, t.size()) == t) { t = s[i]; }
    }
    string ans;
    for (int i = 0; i < 2 * n - 2; i++) {
        if (s[i].size() == t.size() && s[i] == t.substr(0, s[i].size())) { ans += 'P'; } else { ans += 'S'; }
    }
    cout << ans << endl;
    return 0;
}
