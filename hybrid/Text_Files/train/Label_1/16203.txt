#include <iostream>
using namespace std;
int main() {
  int t, L, v, l, r, ans;
  cin >> t;
  while (t--) {
    ans = 0;
    cin >> L >> v >> l >> r;
    if (l > v) ans += L / v;
    else if (r < L % v) ans += (L % v) / v;
    else if (r >= L - 1) ans = 0;
    else {
      if ((L % v) > r) {
        ans += L / v;
        ans -= (r - l + 1) / v;
        ans += (L % v - r - 1) / v;
      } else {
        ans += L / v;
        ans -= (L % v - l) / v;
        ans -= (r - L % v + 1) / v;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
