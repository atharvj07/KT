import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }
        int k = 0;
        int min = h[n - 1];
        int max = h[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            if (h[i] < min && h[i] < max) {
                k++;
            } else if (h[i] > min && h[i] > max) {
                k++;
            }
            min = Math.min(min, h[i]);
            max = Math.max(max, h[i]);
        }
        System.out.println(k);
    }
}
