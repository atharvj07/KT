import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        in.close();
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = a[i];
        }
        for (int i = 1; i < n; i++) {
            for (int j = i - 1; j >= 0; j--) {
                if (a[i] % a[j] == 0) {
                    b[i] = Math.max(b[i], b[j] + 1);
                }
            }
        }
        int max = 0;
        int num = 0;
        for (int i = 0; i < n; i++) {
            if (b[i] > max) {
                max = b[i];
                num = 1;
            } else if (b[i] == max) {
                num++;
            }
        }
        System.out.println(num);
    }
}
