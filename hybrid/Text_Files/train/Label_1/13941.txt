#include <bits/stdc++.h>
using namespace std;
int main() {
    const int mx = 1e9;
    int n; cin >> n;
    vector<pair<int, int>> v;
    for (int i = 0; i < n; i++) {
        int a, b; cin >> a >> b;
        v.push_back({a, 1});
        v.push_back({b, -1});
    }
    int cnt = 0, ans = 0, ansy = 0, pre = 0;
    sort(v.begin(), v.end());
    for (auto x : v) {
        cnt += x.second;
        if (cnt > pre) {
            pre = cnt;
            ans = x.first;
            ansy = cnt;
        }
    }
    cout << ans << ' ' << ansy << '\n';
}
