#include<bits/stdc++.h>
using namespace std;
const int MAXN=110;
int a[MAXN],b[MAXN],q[MAXN],head=1,tail;
vector<int>ans;
int main(){
    int n;
    cin >> n;
    memset(a,0,sizeof(a));
    int cnt0=0;
    for(int i=1;i<=n;++i){
        cin >> b[i];
        if(!b[i]) cnt0++;
        a[b[i]]=i;
    }
    sort(b+1,b+n+1);
    for(int i=1;i<=n;++i){
        if(b[i]==0) q[++tail]=a[0];
        else{
            while(head<=tail&&q[head]<a[b[i]]) head++;
            if(head<=tail) ans.push_back(q[head]);
            ans.push_back(a[b[i]]);
            q[++tail]=a[0];
        }
    }
    int t=0;
    for(int i=0;i<(int)ans.size()-1;++i){
        t+=(ans[i]%2)!=(ans[i+1]%2);
    }
    cout << t << endl;
    return 0;
}
