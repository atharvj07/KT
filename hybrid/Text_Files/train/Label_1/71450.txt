n, k = list(map(int, input().split()))
a = list(map(int, input().split()))
b = list(map(int, input().split()))
need = []
for i in range(n):
    if a[i] > b[i]:
        need.append([a[i] - b[i], i])
need.sort()
ans = 0
for j in need:
    if k >= j[0]:
        k -= j[0]
        ans += 1
    else:
        break
print(ans)
