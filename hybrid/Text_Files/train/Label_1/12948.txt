#include <cstdio>
#include <vector>
const int INF=1<<30;
const long long MOD=1000000007;
int n, m, k;
int cur, fl[550][550];
std::vector<std::pair<int, int> > ans;
int main() {
    scanf("%d%d%d", &n, &m, &k);
    for (int i=1; i<=n; i++)
        for (int j=1; j<=m; j++)
            fl[i][j]=INF;
    for (int i=1; i<=n-1; i++)
        for (int j=1; j<=m; j++)
            if ((i+j)%2==k%2) {
                fl[i][j]=1;
            }
    for (int i=1; i<=n; i++)
        for (int j=1; j<=m-1; j++)
            if ((i+j)%2==k%2) {
                fl[i][j]=1;
            }
    for (int i=n; i>=1; i--)
        for (int j=m; j>=1; j--) {
            if (fl[i][j]==INF)
                continue;
            if (i+1<=n && fl[i][j]+1<fl[i+1][j]) {
                fl[i+1][j]=fl[i][j]+1;
            }
            if (j+1<=m && fl[i][j]+1<fl[i][j+1]) {
                fl[i][j+1]=fl[i][j]+1;
            }
        }
    if (fl[1][1]==INF) {
        printf("NO\n");
        return 0;
    }
    cur=1;
    while (fl[1][1]!=0) {
        if (fl[cur+1][cur]==fl[cur][cur]-1) {
            fl[cur][cur]--;
            ans.push_back(std::make_pair(1, cur));
            cur++;
        }
        else {
            fl[cur][cur]--;
            ans.push_back(std::make_pair(0, cur));
            cur++;
        }
    }
    printf("YES\n");
    printf("%d\n", ans.size());
    for (int i=0; i<ans.size(); i++) {
        printf("%d %c\n", ans[i].first, ans[i].second=='m'?'R':'D');
    }
    return 0;
}
