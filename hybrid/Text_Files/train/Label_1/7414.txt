#include <bits/stdc++.h>
using namespace std;
const double EPS = 1e-9;
struct Edge { int u, w; };
vector<vector<Edge>> G;
vector<int> V;

int find(int x) { return x == V[x] ? x : V[x] = find(V[x]); }
void Union(int x, int y) { V[find(x)] = find(y); }

int main() {
  int n, m;
  cin >> n >> m;
  V.resize(n + 1);
  G.resize(n + 1);
  for (int i = 0; i <= n; i++) V[i] = i;
  vector<tuple<int, int, int>> E;
  long long tot = 0;
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    tot += x;
  }
  for (int i = 0; i < m; i++) {
    int u, v, w;
    cin >> u >> v >> w;
    G[u].push_back({v, w});
    G[v].push_back({u, w});
    E.push_back({u, v, w});
  }
  long long ans = 0;
  for (auto [u, v, w] : E) {
    if (find(u) != find(v)) {
      ans += w;
      Union(u, v);
    }
  }
  double res = (tot > EPS && m > EPS) ? ans * 1.0 / tot : 0.0;
  cout << fixed << setprecision(15) << res << endl;
  return 0;
}
