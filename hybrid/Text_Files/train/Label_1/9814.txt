#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x, y, a[100001], v[100001];
  cin >> n >> x >> y;
  for (int i = 1; i <= n; i++) cin >> a[i];
  deque<pair<int, int>> dq;
  for (int i = 1; i <= n; i++) {
    while (!dq.empty() && a[i] < a[dq.back().first]) dq.pop_back();
    dq.push_back({i, a[i]});
  }
  for (int i = 1; i <= n; i++) {
    while (dq.front().first < i - x) dq.pop_front();
    while (!dq.empty() && dq.back().first > i + y) dq.pop_back();
    v[i] = dq.front().second;
  }
  int mn = v[1];
  for (int i = 2; i <= n; i++) mn = min(mn, v[i]);
  for (int i = 1; i <= n; i++)
    if (v[i] == mn) { cout << i; return 0; }
}
