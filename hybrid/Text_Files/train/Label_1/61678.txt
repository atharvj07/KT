for t in range(int(input())):
    n=int(input())
    a=[input() for i in range(n)]
    b=[[0]*(n) for i in range(n)]
    c=[]
    for i in range(n):
        for j in range(n):
            if a[i][j]=='0':
                c.append((i,j))
            elif a[i][j]=='S'or a[i][j]=='F':
                continue
            else:
                b[i][j]=1
    x,y=c[0]
    b[x][y]^=1
    if b[n-1][n-1]!=b[x][y]:
        c.remove((x,y))
        x,y=c[0]
        b[x][y]^=1
    print(len(c))
    for x,y in c:
        print(x+1,y+1)
