#include <bits/stdc++.h>
#define MOD 1000000007
using namespace std;

int main() {
  string a;
  int k;
  cin >> a >> k;
  k %= 4;
  int ans = 0;
  for (int i = 0; i < (int)a.size(); i++) {
    if (a[i] == '0' && i != 0) continue;
    int x = (a[i] - '0') * i;
    if (k == 0) ans += x % 5 == 0;
    else if (k == 1) ans += x % 5 == 1;
    else if (k == 2) ans += x % 5 == 2;
    else if (k == 3) ans += x % 5 == 3;
    else ans += x % 5 == 4;
    ans %= MOD;
  }
  cout << ans << endl;
  return 0;
}
