#include <bits/stdc++.h>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a.begin(), a.end());
        long long diff = a[n - 1] - a[0];
        if (diff == 0) {
            cout << -1 << endl;
            continue;
        }
        bool valid = false;
        for (int i = 1; i < n; i++) {
            if ((a[n - 1] - a[i]) % diff != 0) {
                valid = true;
                break;
            }
        }
        if (!valid) {
            cout << diff << endl;
        } else {
            cout << -1 << endl;
        }
    }
    return 0;
}
