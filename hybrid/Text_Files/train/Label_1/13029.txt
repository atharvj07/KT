#include <cstdio>
#include <cstring>
using namespace std;
int n, m, a[2001][2001], b[2001];
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; ++i) {
    char str[2001];
    scanf("%s", str);
    for (int j = 1; j <= m; ++j) a[i][j] = str[j - 1] - '0';
  }
  for (int j = 1; j <= m; ++j) {
    int cnt = 0;
    for (int i = 1; i <= n; ++i) if (a[i][j]) ++cnt;
    if (cnt == 1) {
      for (int i = 1; i <= n; ++i)
        if (a[i][j]) { b[i] = 1; break; }
    }
  }
  int flag = false;
  for (int i = 1; i <= n; ++i) {
    if (b[i]) continue;
    int cnt = 0;
    for (int j = 1; j <= m; ++j) if (a[i][j]) ++cnt;
    if (cnt == 1) {
      flag = 1;
      for (int j = 1; j <= m; ++j) {
        if (!a[i][j]) { b[i] = 1; break; }
      }
    }
  }
  for (int j = 1; j <= m; ++j) {
    int tot = 0;
    for (int i = 1; i <= n; ++i) if (!b[i]) tot += a[i][j];
    if (tot == 0) {
      flag = 0;
      break;
    }
  }
  if (flag)
    puts("YES");
  else
    puts("NO");
  return 0;
}
