import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        double n = cin.nextInt(), p = cin.nextDouble(), t = cin.nextInt();
        double ans = 0;
        for (int i = 0; i <= t; ++i) {
            ans += n * Math.pow(p, i) * Math.pow(1 - p, t - i);
        }
        System.out.println(ans);
    }
}
