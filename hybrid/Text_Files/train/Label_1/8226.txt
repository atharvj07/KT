#include<bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
    int h, w, n; cin >> h >> w >> n;
    vector<pair<int,int>> pos(n);
    for(int i = 0; i < n; i++) cin >> pos[i].first >> pos[i].second;
    vector<vector<int>> dp(h+1,vector<int>(w+1,0));
    dp[1][1] = 1;
    for(int i = 1; i <= h; i++) {
        for(int j = 1; j <= w; j++) {
            if(i == 1 && j == 1) continue;
            pair<int,int> p;
            bool f = true;
            for(int x = 0; x < n; x++) {
                if(pos[x].first == i && pos[x].second == j) {f = false; break;}
            }
            if(f) dp[i][j] = ((dp[i][j] + dp[i-1][j]%1000000007)%1000000007 + dp[i][j-1]%1000000007)%1000000007;
        }
    }
    cout << dp[h][w] % 1000000007;
}
