import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            String s = sc.next();
            int n = s.length();
            int ans = 0;
            for (int i = 1; i < n - 1; i++) {
                if (s.charAt(i) == '1') {
                    if (s.charAt(i - 1) == '0' && s.charAt(i + 1) == '0') {
                        ans += a;
                    }
                }
            }
            if (s.charAt(0) == '1') {
                ans += a;
            }
            if (s.charAt(n - 1) == '1') {
                ans += a;
            }
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                if (s.charAt(i) == '0') {
                    cnt++;
                }
            }
            ans += Math.min(cnt * b, a);
            System.out.println(ans);
        }
    }
}
