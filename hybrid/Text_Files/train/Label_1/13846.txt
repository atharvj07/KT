#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  cin >> s;
  vector<long long> dp(s.size() + 1, 0);
  for (int i = s.size() - 1; i >= 0; i--) {
    if (s[i] == 'Q') dp[i] = dp[i + 1] * 2 + 1;
    else dp[i] = dp[i + 1];
    if (i + 1 < s.size() && s[i] == 'Q' && s[i + 1] == 'A') dp[i] += dp[i + 2];
    if (i + 2 < s.size() && s[i] == 'Q' && s[i + 1] == 'A' && s[i + 2] == 'Q') dp[i] += dp[i + 3];
  }
  cout << dp[0] << endl;
  return 0;
}
