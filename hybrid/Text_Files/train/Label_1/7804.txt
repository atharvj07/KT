#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int t;
    std::cin >> t;
    for (int i = 0; i < t; i++) {
        long long int p, f, cnt_s, cnt_w, s, w;
        std::cin >> p >> f >> cnt_s >> cnt_w >> s >> w;
        long long int ans = 0;
        long long int totalw = p + f;
        std::vector<long long int> weapon;
        for (int j = 0; j < cnt_s; j++) {
            weapon.push_back(s);
        }
        for (int j = 0; j < cnt_w; j++) {
            weapon.push_back(w);
        }
        std::sort(weapon.begin(), weapon.end());
        for (int j = 0; j < weapon.size(); j++) {
            if (weapon[j] <= totalw) {
                totalw -= weapon[j];
                ans++;
            } else {
                break;
            }
        }
        std::cout << ans << std::endl;
    }
    return 0;
}
