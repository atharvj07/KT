import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (true) {
            int a = sc.nextInt();
            int d = sc.nextInt();
            int n = sc.nextInt();

            if (a == 0 && d == 0 && n == 0) {
                break;
            }

            int cnt = 0;
            int num = a;
            while (true) {
                if (isPrime(num)) {
                    cnt++;
                }
                if (cnt == n) {
                    System.out.println(num);
                    break;
                }
                num += d;
            }
        }
    }

    private static boolean isPrime(int n) {
        if (n <= 1) {
            return false;
        }
        for (int i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                return false;
            }
        }
        return true;
    }
}
