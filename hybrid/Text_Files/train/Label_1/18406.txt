#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  string s, t;
  cin >> s >> t;
  int n = s.size(), m = t.size();
  vector<int> z(m);
  z[0] = 0;
  int l = 0, r = 0;
  for (int i = 1; i < m; i++) {
    if (i <= r) {
      z[i] = min(r - i + 1, z[i - l]);
    }
    while (i + z[i] < m && t[z[i]] == t[i + z[i]]) {
      z[i]++;
    }
    if (i + z[i] - 1 > r) {
      l = i;
      r = i + z[i] - 1;
    }
  }
  int j = 0;
  vector<pair<int, int>> ans;
  for (int i = 0; i < n; i++) {
    while (j < m && t[j] != s[i]) {
      j++;
    }
    if (j < m) {
      int k = j;
      while (k < m && s[i + k - j] == t[k]) {
        k++;
      }
      if (k == m) {
        ans.push_back({i, i + k - j - 1});
        j = 0;
      } else {
        j++;
      }
    }
  }
  if (j == m) {
    cout << ans.size() << endl;
    for (auto it : ans) {
      cout << it.first + 1 << ' ' << it.second + 1 << endl;
    }
  } else {
    cout << -1 << endl;
  }
}
