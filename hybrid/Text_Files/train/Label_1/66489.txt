import sys

def dfs(v, p, w, sa):
  sa[v] = w
  for u, ww in adj[v]:
    if u == p: continue
    dfs(u, v, max(w, ww), sa)

def dfs2(v, p, w, sa, ans):
  ans[v] = sa[v]
  for u, ww in adj[v]:
    if u == p: continue
    dfs2(u, v, max(w, ww), sa, ans)
    ans[v] = min(ans[v], ans[u])

n, m = map(int, input().split())
adj = [[] for _ in range(n+1)]
for i in range(n-1):
    a, b, w = map(int, input().split())
    adj[a].append((b, w))
    adj[b].append((a, w))

sa = [0] * (n+1)
ans = [0] * (n+1)
dfs(1, 1, 0, sa)
dfs2(1, 1, 0, sa, ans)

ans = [0] * m
for i in range(m):
    w = int(input())
    for j in range(n):
        if sa[j+1] <= w:
            ans[i] += n - 1 - j

print(*ans)
