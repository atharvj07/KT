#include <cstdio>
#include <iostream>

using namespace std;

const long long MOD = 998244353;
const long long N = 250000;
long long f[N + 5], invf[N + 5], powk[N + 5];

long long fastpow(long long x, long long p) {
  long long res = 1;
  while (p) {
    if (p & 1) res = res * x % MOD;
    x = x * x % MOD;
    p >>= 1;
  }
  return res;
}

int main() {
  int n, k;
  cin >> n >> k;
  f[0] = 1;
  for (int i = 1; i <= n; i++) f[i] = i * f[i - 1] % MOD;
  invf[n] = fastpow(f[n], MOD - 2);
  for (int i = n - 1; i >= 0; i--) invf[i] = invf[i + 1] * (i + 1) % MOD;
  powk[0] = 1;
  for (int i = 1; i <= n; i++) powk[i] = powk[i - 1] * k % MOD;
  int x;
  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    cin >> x;
    long long tot = powk[n - i] * invf[i] % MOD;
    tot = tot * invf[n - i] % MOD;
    tot = tot * (f[i] - 1 + MOD) % MOD;
    ans = ans + tot;
  }
  ans = (ans + MOD) % MOD;
  cout << ans << endl;
  return 0;
}
