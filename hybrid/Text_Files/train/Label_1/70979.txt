import sys

# Read the input
n = int(sys.stdin.readline())
s = sys.stdin.readline().strip()
a = list(map(int, sys.stdin.readline().split()))

# Initialize the dp table
dp = [[-1] * (n + 1) for _ in range(n + 1)]

# Function to calculate the number of ways to split the string
def solve(i, j):
    # Base case
    if i == j:
        return 1

    # Memoization
    if dp[i][j] != -1:
        return dp[i][j]

    ans = 0

    # Try all possible split points
    for k in range(i, j + 1):
        # Check if the substring from i to k is valid
        if k - i + 1 <= a[ord(s[i]) - ord('a')]:
            ans += solve(i, k) * solve(k + 1, j)
            ans %= 1000000007

    # Store the result in the dp table
    dp[i][j] = ans

    return ans

# Calculate the number of ways to split the string
num_ways = solve(0, n - 1)

# Find the length of the longest substring
max_len = 0
for i in range(n):
    for j in range(i, n):
        if j - i + 1 <= a[ord(s[i]) - ord('a')]:
            max_len = max(max_len, j - i + 1)

# Find the minimum number of substrings
min_substrings = n
for i in range(n):
    for j in range(i, n):
        if j - i + 1 <= a[ord(s[i]) - ord('a')]:
            min_substrings = min(min_substrings, 1 + solve(j + 1, n - 1))

# Print the results
print(num_ways)
print(max_len)
print(min_substrings)
