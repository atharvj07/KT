def dfs(x, y, step):
    global result, infinite
    if dp[x][y] != -1:
        if not visited[x][y]:
            infinite = True
        else:
            result = max(result, dp[x][y])
        return dp[x][y]

    visited[x][y] = False
    dp[x][y] = 0
    for dx, dy in directions:
        nx, ny = x + dx, y + dy
        if 0 <= nx < n and 0 <= ny < m and table[nx][ny] == order[(step + 1) % 4]:
            dp[x][y] = max(dp[x][y], dfs(nx, ny, step + 1) + (step == 3))
    visited[x][y] = True
    result = max(result, dp[x][y])
    return dp[x][y]

n, m = map(int, input().split())
table = [input() for _ in range(n)]

order = "DIMA"
directions = [(0, 1), (1, 0), (-1, 0), (0, -1)]
dp = [[-1] * m for _ in range(n)]
visited = [[True] * m for _ in range(n)]
result = 0
infinite = False

for i in range(n):
    for j in range(m):
        if table[i][j] == "D":
            dfs(i, j, 0)

if infinite:
    print("Poor Inna!")
elif result == 0:
    print("Poor Dima!")
else:
    print(result)
