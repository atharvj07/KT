import sys

n, q = map(int, input().split())
a = [int(x) for x in input().split()]
b = [int(input()) for _ in range(q)]

# Largest coin value that is less than or equal to the query value
dp = [0] * (max(b) + 1)
ans = []
for val in a:
    for i in range(val, len(dp)):
        if dp[i] == 0:
            dp[i] = dp[i - val] + 1
for query in b:
    ans.append(dp[query] if dp[query] else -1)
print(*ans, sep='\n')
