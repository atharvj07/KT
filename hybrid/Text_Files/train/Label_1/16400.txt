#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int n,m,p[200001];
const ll inf = 1e18;
ll dist(ll a, ll b){
  return a > b ? a - b : b - a;
}
int f(ll x){
  return x == p[x] ? x : p[x] = f(p[x]);
}
int main(){
  cin >> n >> m;
  ll a[n+1], ans = 0;
  for(int i = 1;i <= n;i++){
    cin >> a[i];
    p[i] = i;
  }
  for(int i = 1;i <= m;i++){
    ll x, y, w;
    cin >> x >> y >> w;
    if(w < dist(a[x], a[y])){
      a[f(x)] = a[f(y)] = min(a[f(x)], a[f(y)]);
      ans += w - dist(a[f(x)], a[f(y)]);
      p[f(x)] = f(y);
    }
  }
  for(int i = 1;i <= n;i++)
    if(p[i] == i)
      ans += a[i];
  cout << ans << '\n';
  return 0;
}
