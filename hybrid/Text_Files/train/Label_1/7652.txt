#include <bits/stdc++.h>
#define ll long long
#define MOD 998244353
using namespace std;
int main() {
    ll n, m;
    cin >> n >> m;
    vector<ll> a(n), b(m);
    for (ll &i : a) cin >> i;
    for (ll &i : b) cin >> i;
    vector<vector<ll>> dp(n + 1, vector<ll>(m + 1, 0));
    dp[0][0] = 1;
    for (ll i = 0; i < n; i++) {
        for (ll j = 0; j < m; j++) {
            dp[i + 1][j + 1] = dp[i][j + 1];
            if (a[i] >= b[j]) { dp[i + 1][j + 1] = (dp[i][j + 1] + dp[i][j]) % MOD; }
        }
    }
    cout << dp[n][m] << endl;
    return 0;
}
