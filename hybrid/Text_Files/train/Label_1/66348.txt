n = int(input())
max_salaries = []
total_employees = 0

for _ in range(n):
    company = list(map(int, input().split()))
    max_salaries.append(max(company[1:]))
    total_employees += company[0]

max_salary = max(max_salaries)
total_increase = (max_salary * total_employees) - sum([company[0] * company[1] for company in zip(max_salaries, [1]*n)])

print(total_increase)
