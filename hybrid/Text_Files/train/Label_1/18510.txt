#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int N = 2e5+9;
int n, m, q, p[N];
vector<int> a[N];

int find(int x) {
  return x == p[x] ? x : p[x] = find(p[x]);
}

int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> m;
  for(int i = 1;i <= n;++i) p[i] = i;
  while(m--) {
    int u, v; cin >> u >> v;
    u = find(u), v = find(v);
    if(u != v) p[v] = u;
  }
  cin >> q;
  while(q--) {
    int x, y; cin >> x;
    if(x == 1) {
      cin >> x >> y;
      x = find(x), y = find(y);
      if(x != y) p[y] = x;
    } else if(x == 2) {
      cin >> x >> y;
      x = find(x), y = find(y);
      if(x != y) a[x].push_back(y), a[y].push_back(x);
    } else {
      int ans = n;
      for(int i = 1;i <= n;++i) {
        p[i] = i;
        bool ok = 0;
        for(auto &x : a[i]) {
          if(find(x) != i) ok = 1;
        }
        if(!a[i].empty() && ok) p[find(i)] = 0, ans--;
      }
      for(int i = 1;i <= n;++i) for(auto &x : a[i]) a[i].clear();
      cout << ans << '\n';
    }
  }
  return 0;
}
