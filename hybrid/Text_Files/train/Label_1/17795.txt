#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  int ans = 1;
  int cur = 1;
  for (int i = 1; i < n; i++) {
    if (a[i] > a[i - 1])
      cur++;
    else
      cur = 1;
    ans = max(ans, cur);
  }
  cout << ans << endl;
  return 0;
}
