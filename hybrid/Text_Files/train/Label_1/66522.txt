import sys
input = sys.stdin.readline
for t in range(int(input())):
    n = int(input())
    p = sorted([int(x) for x in input().strip().split()])
    f = []
    for i in range(n-1):
        f.append(p[i] + p[i+1])
    f.sort()
    print(*p)
    for x in f:
        i = p.index(x - p[0])
        p[0], p[i] = p[i], p[0]
        print(*p)
