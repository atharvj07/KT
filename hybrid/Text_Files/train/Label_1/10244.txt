#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  while (n--) {
    string s;
    cin >> s;
    int l = s.length();
    bool diverse = true;
    for (int i = 1; i < l; i++) {
      if (s[i] != s[i - 1] + 1) {
        diverse = false;
        break;
      }
    }
    if (diverse) {
      cout << "Yes" << endl;
    } else {
      cout << "No" << endl;
    }
  }
  return 0;
}
