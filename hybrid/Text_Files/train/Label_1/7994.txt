#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> digits;
  while (k > 0) {
    digits.push_back(k % n);
    k /= n;
  }
  ll ans = 0;
  for (int i = 0; i < digits.size(); i++) {
    ll base = 1;
    for (int j = 0; j < i; j++) {
      base *= n;
    }
    ans += base * digits[i];
  }
  cout << ans << endl;
  return 0;
}
