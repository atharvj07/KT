import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.nextLine();
        int n = s.length();
        int cnt = 0;
        int mid = n / 2;
        for (int i = 0; i < mid; i++) {
            if (s.charAt(i) != s.charAt(n - i - 1)) {
                cnt++;
            }
        }
        System.out.println(cnt == 1 ? "YES" : "NO");
    }
}
