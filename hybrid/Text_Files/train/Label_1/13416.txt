#include <bits/stdc++.h>
#define fori(i, a, b) for(int i = a; i < b; ++i)
#define ford(i, a, b) for(int i = a; i >= b; i--)
#define pb push_back
#define mp make_pair
using namespace std;
vector<int> p[200010];
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        fori(i, 0, 200010) p[i].clear();
        int n, ans = 0;
        cin >> n;
        fori(i, 0, n)
        {
            int x; cin >> x;
            p[x].pb(i);
        }
        fori(i, 1, 100001)
        {
            if (p[i].empty()) continue;
            fori(j, 2*i, 100001)
            {
                if (p[j].empty()) continue;
                for(int k : p[j]) if (k > p[i].back()) ans++;
            }
        }
        cout << ans << endl;
    }
}
