#include <bits/stdc++.h>
#define endl '\n'
using namespace std;

typedef long long ll;

void solve() {
  ll n;
  cin >> n;
  string s, t;
  cin >> s >> t;
  int ans = 0;
  vector<pair<int, int>> v;
  for (int i = 0; i < n; i++) {
    if (s[i] != t[i]) {
      bool ok = false;
      for (int j = i + 1; j < n; j++) {
        if (s[j] == s[i] && t[j] == t[i]) {
          v.push_back({i + 1, j + 1}), swap(s[i], t[j]), swap(s[j], t[i]), ans++, ok = true;
          break;
        }
      }
      if (!ok) {
        cout << -1 << endl;
        return;
      }
    }
  }
  cout << ans << endl;
  for (auto x : v) cout << x.first << ' ' << x.second << endl;
}

int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int t = 1;
  while (t--) solve();
}
