#include <bits/stdc++.h>
using namespace std;
#define int long long
const int INF = 1e18;
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, a, b, c;
        cin >> n >> m >> a >> b >> c;
        vector<vector<pair<int, int>>> g(n + 1);
        vector<int> p(m);
        for (int i = 0; i < m; i++) {
            cin >> p[i];
        }
        for (int i = 0; i < m; i++) {
            int v, u;
            cin >> v >> u;
            g[v].push_back({u, i});
            g[u].push_back({v, i});
        }
        vector<int> dist(n + 1, INF);
        queue<int> q;
        dist[a] = 0;
        q.push(a);
        while (!q.empty()) {
            int cur = q.front();
            q.pop();
            for (auto& edge : g[cur]) {
                int to = edge.first;
                int id = edge.second;
                if (dist[cur] + 1 < dist[to]) {
                    dist[to] = dist[cur] + 1;
                    q.push(to);
                }
            }
        }
        sort(p.begin(), p.end());
        int ans = 0;
        for (int i = 0; i < dist[b] + dist[c] - 1; i++) {
            ans += p[i];
        }
        cout << ans << endl;
    }
    return 0;
}
