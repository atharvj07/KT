n, m = map(int, input().split())
s = [list(input()) for _ in range(n)]
for i in range(1, n - 1):
    for j in range(1, m - 1):
        if s[i][j] == '.' and s[i - 1][j] == '#' and s[i + 1][j] == '#' and s[i][j - 1] == '#' and s[i][j + 1] == '#':
            print('YES')
            exit()
print('NO')
