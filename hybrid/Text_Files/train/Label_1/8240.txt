#include <bits/stdc++.h>

using namespace std;

const int MAX_N = 1e9 + 7;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, m;
    cin >> n >> m;
    vector<pair<long long, long long>> flowers(m);
    for (int i = 0; i < m; i++) { cin >> flowers[i].first >> flowers[i].second; }
    sort(flowers.begin(), flowers.end(),
         [](pair<long long, long long> &a, pair<long long, long long> &b) {
           return (a.first + a.second) > (b.first + b.second);
         });
    long long ans = 0;
    for (int i = 0; i < m && n > 0; i++) {
      long long take = min(flowers[i].first + flowers[i].second, n);
      ans += take * (take - 1) / 2;
      ans += (flowers[i].first + (take - 1) * flowers[i].second) % MAX_N;
      ans %= MAX_N;
      n -= take;
    }
    cout << ans << endl;
  }
  return 0;
}
