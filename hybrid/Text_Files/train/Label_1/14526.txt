#include <iostream>
#include <vector>
#include <string>

using namespace std;

int main() {
  int L, A;
  cin >> L >> A;
  string P = to_string(A + 1);
  int n = P.size();
  while (n < L) {
    P += P;
    n = P.size();
  }
  if (n == L) {
    cout << P;
  } else {
    P = P.substr(0, L);
    cout << P;
  }
  return 0;
}
