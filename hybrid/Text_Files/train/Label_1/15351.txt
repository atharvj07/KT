#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 10;

vector<int> G[MAXN];
int a[MAXN], dp[MAXN], n;

int main() {
    cin >> n;
    for(int i = 1; i <= n; ++i) cin >> a[i];
    for(int i = 1; i <= n; ++i) {
        for(int j = i + a[i]; j <= n; j += a[i]) G[i].push_back(j);
        G[i].push_back(i);
    }
    for(int i = 1; i <= n; ++i) {
        vector<int> vis(n + 3, 0);
        for(int j = 0; j < (int)G[i].size(); ++j) {
            vis[a[G[i][j]]] = true;
        }
        for(int j = 1; j <= n; ++j) {
            if(!vis[j]) {
                dp[i] = j;
                break;
            }
        }
    }
    for(int i = 1; i <= n; ++i) {
        if(dp[i] & 1) cout << "A";
        else cout << "B";
    }
    return 0;
}
