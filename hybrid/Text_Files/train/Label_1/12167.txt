#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
const int MAXN=1e4+5;
vector<LL> v;
void sieve()
{
    v.push_back(1);
    for(int i=1;i<=31;++i) v.push_back(3ll*v[i-1]);
}
int main()
{
    sieve();
    int q;
    cin>>q;
    while(q--)
    {
       LL n;
       cin>>n;
       LL ans=n;
       for(int i=0;i<=31;++i)
       {
           if(n<=v[i])
           {
               ans=v[i];
               break;
           }
       }
       cout<<ans<<endl;
    }
    return 0;
}
