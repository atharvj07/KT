#include<bits/stdc++.h>
using namespace std;

int main(){
    int n;
    cin>>n;
    vector<int> deg(n+1,0);
    vector<int> prv(n+1,0);
    vector<int> tp(n+1,0);
    for(int i=1; i<=n; i++){
        cin>>tp[i];
    }
    for(int i=1; i<=n; i++){
        int x;
        cin>>x;
        if(x){
            deg[i]++;
            deg[x]--;
            prv[i]=x;
        }
    }
    int c=0;
    int lst=0;
    for(int i=1; i<=n; i++){
        if(deg[i]==1){
            c++;
            lst=i;
        }
    }
    cout<<c<<endl;
    while(c--){
        cout<<lst<<" ";
        lst=prv[lst];
    }
    return 0;
}
