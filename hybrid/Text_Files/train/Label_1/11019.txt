#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> field(n);
  for (int i = 0; i < n; i++) { cin >> field[i]; }
  pair<int, int> first, last;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    first = {n, m}, last = {0, 0};
    bool flag = false;
    for (int j = 0; j < m; j++) {
      if (field[i][j] == 'G') {
        first = {i, j + 1};
        if (j + 1 < m && field[i][j + 1] == 'S') { flag = true; }
      }
      if (field[i][j] == 'S') { last = {i, j}; }
    }
    if (!flag && first.first != n && last.first != 0) ans++;
  }
  if (ans == 0) ans = -1;
  cout << ans << endl;
  return 0;
}
