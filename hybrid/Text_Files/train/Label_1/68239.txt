for _ in range(int(input())):
    n, k, l = map(int, input().split())
    d = [int(x) for x in input().split()]
    p = [x for x in range(k)]
    p.extend([x for x in range(k - 1, -1, -1)])
    for i in range(n + 1):
        if i > 0 and d[i - 1] + p[i] > l:
            print("No")
            break
    else:
        print("Yes")
