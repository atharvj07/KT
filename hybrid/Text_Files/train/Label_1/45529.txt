import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int xv = scanner.nextInt();
    int yv = scanner.nextInt();
    int xp = scanner.nextInt();
    int yp = scanner.nextInt();
    int xw1 = scanner.nextInt();
    int yw1 = scanner.nextInt();
    int xw2 = scanner.nextInt();
    int yw2 = scanner.nextInt();
    int xm1 = scanner.nextInt();
    int ym1 = scanner.nextInt();
    int xm2 = scanner.nextInt();
    int ym2 = scanner.nextInt();
    if (xv == xp && yv == yp) {
      System.out.println("NO");
      return;
    }
    boolean wall = false;
    boolean mirror = false;
    if (yw1 == yw2) {
      if ((xw1 <= xv && xv <= xw2) || (xw1 <= xp && xp <= xw2)) {
        wall = true;
      }
    } else if (xw1 == xw2) {
      if ((yw1 <= yv && yv <= yw2) || (yw1 <= yp && yp <= yw2)) {
        wall = true;
      }
    }
    if (ym1 == ym2) {
      if ((xm1 <= xv && xv <= xm2) || (xm1 <= xp && xp <= xm2)) {
        mirror = true;
      }
    } else if (xm1 == xm2) {
      if ((ym1 <= yv && yv <= ym2) || (ym1 <= yp && yp <= ym2)) {
        mirror = true;
      }
    }
    if (wall && mirror) {
      int dxv = xv - xm1;
      int dyp = yp - ym1;
      int dxp = xp - xm1;
      int dym = yv - ym1;
      if ((dxv * dyp < 0 && dxp * dym < 0) || (xv == xp && yv == yp)) {
        System.out.println("NO");
        return;
      }
    }
    System.out.println("YES");
  }
}
