#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const ll MOD = 998244353;

ll fact = 1;
ll pwr[500005];
ll rpw[500005];

ll quick_pow(ll b, ll p) {
  if (p == 0) return 1;
  ll ans = quick_pow(b, p / 2);
  ans = ans * ans;
  ans %= MOD;
  if (p % 2) ans = ans * b;
  return ans % MOD;
}

map<ll, ll> m;

ll solve(ll n, ll k) {
  if (k > n) return 0;
  if (k == 1) return fact;
  if (k == 2) return fact * (n - 1) % MOD;
  ll v = m[k];
  if (v != 0) return v;
  ll ans = fact * (n - k) % MOD;
  v = ans * pwr[k - 2] % MOD;
  m[k] = v;
  return v;
}

int main() {
  ll n, k;
  cin >> n >> k;
  fact = 1;
  for (ll i = 1; i <= n; i++) {
    fact = fact * i;
    fact %= MOD;
  }
  pwr[0] = 1;
  for (ll i = 1; i <= n; i++) {
    pwr[i] = pwr[i - 1] * k;
    pwr[i] %= MOD;
  }
  rpw[0] = 1;
  for (ll i = 1; i <= n; i++) {
    rpw[i] = rpw[i - 1] * quick_pow(k, MOD - 2);
    rpw[i] %= MOD;
  }
  ll ans = solve(n, k);
  ans = ans * rpw[n - k] % MOD;
  cout << ans << endl;
  return 0;
}
