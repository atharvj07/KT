import sys

n,m=map(int,input().split())
a=list(map(int,input().split()))
m-=1
if not m:ans=[max(a)-min(a)];print(*ans,0);sys.exit()
scv=[0]*n
for l,r in [list(map(int,input().split())) for i in range(m)]:
    scv[l-1]+=1
    scv[r]-=1
sc=0
ma=0
hd=[]
for x in scv:
   sc+=x   
   hd.append(ma)
   ma=max(ma,sc)
ans=[max(a)+ma]
for k,sc in enumerate(hd):
   if not ma==sc:continue
   v=ans[-1]
   b=sorted([j for j in range(k+1,n) if scv[j]]+[(1<<31)])
   for j in a:
      j+=ma
      if j>v:ans+=[j,b[0]];b=b[1:]
   ans+=[v]
print(*ans,len(ans)-1,*[a for a in range(len(ans)-1) if a%2])
