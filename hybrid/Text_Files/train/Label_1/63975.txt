def find_max_k(promocodes):
    max_k = 6
    for i in range(len(promocodes)):
        for j in range(i+1, len(promocodes)):
            diff = sum(a != b for a, b in zip(promocodes[i], promocodes[j]))
            max_k = min(max_k, (diff - 1) // 2)
    return max_k

n = int(input())
promocodes = [input().strip() for _ in range(n)]
print(find_max_k(promocodes))
