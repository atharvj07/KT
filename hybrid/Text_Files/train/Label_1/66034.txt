def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(n, m, x, y, a, b):
    g = gcd(a, b)
    a //= g
    b //= g
    k = min(n // a, m // b)
    x1 = max(0, x - a * k)
    y1 = max(0, y - b * k)
    if x1 + a * k > n:
        x1 = n - a * k
    if y1 + b * k > m:
        y1 = m - b * k
    x2 = x1 + a * k
    y2 = y1 + b * k
    return x1, y1, x2, y2

n, m, x, y, a, b = map(int, input().split())
print(*solve(n, m, x, y, a, b))
