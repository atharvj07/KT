import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        BigInteger gcd = new BigInteger("1000000007");
        BigInteger ans = new BigInteger(Long.MAX_VALUE + "");
        while (n-- > 0) {
            BigInteger a = new BigInteger(sc.next());
            BigInteger b = new BigInteger(sc.next());
            BigInteger tmp = a;
            while (tmp.compareTo(ans) < 0) {
                if (a.compareTo(b) != -1 && ans.compareTo(tmp) != -1) {
                    ans = tmp;
                }
                tmp = tmp.multiply(b);
            }
            if (a.compareTo(b) == 1) {
                ans = ans.min(a);
            }
        }
        if (ans.compareTo(new BigInteger("1000000007")) == -1) {
            System.out.println(ans.remainder(gcd));
        } else {
            System.out.println(-1);
        }
    }
}
