#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, ans = 0;
    cin >> n;
    vector<pair<int, int>> s(n);
    for (int i = 0; i < n; i++) cin >> s[i].first >> s[i].second;
    sort(s.begin(), s.end());
    int l = s[0].first, r = s[0].second;
    for (int i = 1; i < n; i++) {
      if (s[i].first > r) {
        ans++;
        l = s[i].first;
        r = s[i].second;
      } else
        l = max(l, s[i].first), r = min(r, s[i].second);
    }
    cout << ans << endl;
  }
}
