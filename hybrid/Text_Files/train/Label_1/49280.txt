import java.util.Scanner;

public class Door {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();
        int[] t = new int[m];
        for (int i = 0; i < m; i++) {
            t[i] = scanner.nextInt();
        }
        int open = 0;
        int time = 0;
        for (int i = 0; i < n; i++) {
            int employeeTime = i * a;
            if (employeeTime >= time) {
                open++;
                time = employeeTime + d;
            }
        }
        for (int i = 0; i < m; i++) {
            if (t[i] >= time) {
                open++;
                time = t[i] + d;
            }
        }
        System.out.println(open);
    }
}
