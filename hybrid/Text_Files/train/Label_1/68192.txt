import math
def power(n):
    return int(math.pow(n, 0.5))

n=int(input())
g=[[0]*n for i in range(n)]
for i in range(n):
    g[i]=[int(x) for x in input().split()]
a=[0 for i in range(n)]
for i in range(n):
    for j in range(n):
        if g[i][j]!=0:
            a[i]=int(g[i][j]/a[j]) if a[j]!=0 else power(g[i][j])
print(' '.join(map(str,a)))
