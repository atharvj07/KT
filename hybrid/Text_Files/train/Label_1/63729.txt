for _ in range(int(input())):
    n = int(input())
    b = list(map(int,input().split()))
    for i in range(n):
        b[i]=1<<b[i]
    pfx=[0 for i in range(n+1)]
    for i in range(n):
        pfx[i+1]=pfx[i]+b[i]
    st=set()
    for i in range(n):
        for j in range(i+1,n+1):
            if pfx[j]-pfx[i] in st:
                print("YES")
                exit()
            else: st.add(pfx[j]-pfx[i])
    print("NO")
