def transform_string(s, t):
    if sorted(s) != sorted(t):
        return -1, []
    moves = []
    for i in range(len(t)):
        j = i
        while s[j] != t[i]:
            j += 1
        while j > i:
            s = s[:j-1] + s[j] + s[j-1] + s[j+1:]
            moves.append(j)
            j -= 1
    return len(moves), moves

n = int(input())
s = input()
t = input()

k, moves = transform_string(s, t)
if k == -1:
    print(k)
else:
    print(k)
    print(' '.join(map(str, moves)))
