import sys

n, m, k = map(int, sys.stdin.readline().split())
holes = [int(x) for x in sys.stdin.readline().split()]
bone = 1
for i in range(k):
    u, v = map(int, sys.stdin.readline().split())
    if bone == u:
        bone = v
    elif bone == v:
        bone = u
    if bone in holes:
        bone = 1
print(bone)
