#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int n, a[710];
ll c[710][710];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];

  for (int i = 1; i <= n; i++) {
    for (int j = i + 1; j <= n; j++) {
      ll g = __gcd(a[i], a[j]);
      c[i][j] = g;
      c[j][i] = g;
    }
  }

  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      for (int k = 1; k <= n; k++) {
        if (c[i][j] > 1 && c[i][k] > 1 && __gcd(c[i][j], c[i][k]) == 1) {
          cout << "No" << endl;
          return 0;
        }
      }
    }
  }

  cout << "Yes" << endl;
}
