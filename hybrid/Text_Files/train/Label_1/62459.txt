import sys, math
input = sys.stdin.readline

def is_infinite(a, x):
    sum = 0
    for i in range(len(a)):
        sum += a[i]
        if sum >= x:
            return False
    return True

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    x = list(map(int, input().split()))
    for i in x:
        if is_infinite(a, i):
            print(-1)
        else:
            sum = 0
            sec = 0
            while sum < i:
                sum += a[sec%n]
                sec += 1
            print(sec)
