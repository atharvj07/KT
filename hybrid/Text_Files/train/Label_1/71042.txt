n, p = map(int, input().split())
times = list(map(int, input().split()))

events = [(times[i], i, 1) for i in range(n)]  # (time, index, type)
events.sort()

ans = [0] * n
queue = []
current_time = 0

for time, i, type in events:
    while queue and queue[0][0] <= time:
        _, index = queue.pop(0)
        current_time = max(current_time, ans[index]) + p
        ans[index] = current_time

    if not queue:
        current_time = max(current_time, time) + p
        ans[i] = current_time
    else:
        queue.append((times[i], i))

    queue.sort(key=lambda x: (x[0], x[1]))

while queue:
    _, index = queue.pop(0)
    current_time = max(current_time, ans[index]) + p
    ans[index] = current_time

print(' '.join(map(str, ans)))
