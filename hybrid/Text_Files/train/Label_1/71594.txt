n, m = map(int, input().split())
a = [list(map(int, input().split())) for i in range(m)]
s = [0]*n

white = [set() for j in range(m)]
red = [set() for j in range(m)]
blue = [set() for j in range(m)]

for i in range(m):
    for j in a[i]:
        if s[j-1]==0:
            white[i].add(j)
            s[j-1] = 1
        elif s[j-1]==1:
            red[i].add(j)
            s[j-1] = 2
        else:
            blue[i].add(j)
