#include<bits/stdc++.h>
#define ll long long
#define ull unsigned long long
const int mod=998244353;
const int p=31;
const int ms=1e5+10;
using namespace std;
int power[ms],h1[ms],h2[ms],n;
 
void init(){
    power[0]=1;
    for(int i=1; i<=n; i++){
        power[i]=(power[i-1]*p)%mod;
    }
}
 
void pre(string &s){
    h1[0]=0;
    for(int i=1; i<=n; i++){
        h1[i]=(h1[i-1]*p+(s[i-1]-'0'))%mod;
    }
}
 
int gethash(int l,int r){
    return (h1[r]-((ll)h1[l-1]*power[r-l+1])%mod+mod)%mod;
}
 
int main(){
    string s;
    cin>>s;
    cin>>n;
    n=s.size();
    init();
    pre(s);
    while(n--){
        int l,r;
        char c;
        string t;
        cin>>c>>t;
        l=c-'0';
        r=t.size();
        for(int i=1; i<=n; i++){
            if(s[i-1]-'0'==l){
                int hl=gethash(max(i-r+1,1),i);
                int hr=gethash(1,r);
                if(hl==hr){
                    s.replace(i-1,1,t);
                    pre(s);
                    break;
                }
            }
        }
    }
    ull ans=0;
    for(int i=1; i<=n; i++){
        ans=(ans*10ll+(s[i-1]-'0'))%mod;
    }
    cout<<ans<<endl;
}
