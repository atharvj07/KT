import java.util.*;

public class Main {
    static int N, M, K;
    static int[] A, B, C;
    static ArrayList<Integer>[] G;
    static int[][][] dp;
    static boolean[][] vis;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        N = in.nextInt();
        M = in.nextInt();
        K = in.nextInt();
        A = new int[N];
        B = new int[N];
        C = new int[N];
        G = new ArrayList[N];
        for (int i = 0; i < N; i++) {
            G[i] = new ArrayList<>();
        }
        for (int i = 0; i < N; i++) {
            A[i] = in.nextInt();
            B[i] = in.nextInt();
            C[i] = in.nextInt();
        }
        for (int i = 0; i < M; i++) {
            int u = in.nextInt() - 1;
            int v = in.nextInt() - 1;
            G[u].add(v);
        }
        dp = new int[N][N][K + 1];
        vis = new boolean[N][N];
        int ans = dfs(0, 0, K);
        if (ans == -1) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }

    static int dfs(int u, int v, int k) {
        if (u == N) {
            if (k == 0) {
                return 0;
            } else {
                return -1;
            }
        }
        if (vis[u][v]) {
            return dp[u][v][k];
        }
        vis[u][v] = true;
        dp[u][v][k] = -1;
        if (k >= A[u]) {
            dp[u][v][k] = Math.max(dp[u][v][k], dfs(u + 1, u, k - A[u]) + C[u]);
        }
        if (v != u && k >= 1) {
            dp[u][v][k] = Math.max(dp[u][v][k], dfs(u + 1, v, k - 1) + C[u]);
        }
        for (int w : G[u]) {
            if (k >= 1) {
                dp[u][v][k] = Math.max(dp[u][v][k], dfs(u + 1, w, k - 1) + C[u]);
            }
        }
        if (k + B[u] <= K) {
            dp[u][v][k] = Math.max(dp[u][v][k], dfs(u, u, k + B[u]));
        }
        return dp[u][v][k];
    }
}
