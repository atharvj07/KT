#include <bits/stdc++.h>
using namespace std;
typedef pair<string, string> a;
int main() {
  int n;
  cin >> n;
  map<string, string> m;
  while (n--) {
    string s;
    cin >> s;
    m[s] = s;
    map<string, string>::iterator it;
    it = m.find(s);
    m.erase(it);
    m.insert(make_pair(s, s));
  }
  for (auto it = m.rbegin(); it != m.rend(); ++it) cout << it->first << endl;
}
