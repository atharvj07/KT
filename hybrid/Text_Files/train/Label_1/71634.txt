for _ in range(int(input())):
    s = input()
    if s[0] == '?' or s[-1] == '?' or '??' in s:
        print(-1)
        continue
    c = None
    chk = {'a', 'b', 'c'}
    for x in s:
        if x == '?':
            if c in chk:
                chk.remove(c)
            c = chk.pop()
            print(c, end="")
        else:
            c = x
            print(x, end="")
    print()
