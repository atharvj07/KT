#include <cstdio>
#include <iostream>

using namespace std;

int main() {
  long long k1, q, max, m;
  long long skips[100001];
  cin >> k1 >> q >> max >> m;
  for (long long i = 1; i <= m; i++) { cin >> skips[i]; }
  if (k1 == 0 && q == 0) {
    cout << "inf" << endl;
    return 0;
  }
  if ((k1 > 0 && q < 0) || (k1 < 0 && q > 0)) {
    cout << "inf" << endl;
    return 0;
  }
  long long counter = 0;
  long long badskip = 0;
  while (true) {
    k1 = k1 * q;
    if (k1 > 0 && max < k1) {
      break;
    }
    if (k1 < 0 && k1 < -max) {
      break;
    }
    for (long long j = 1; j <= m; j++) {
      if (skips[j] == k1) {
        badskip = 1;
        break;
      }
    }
    if (badskip != 1) { counter++; }
    badskip = 0;
  }
  cout << counter << endl;
  return 0;
}
