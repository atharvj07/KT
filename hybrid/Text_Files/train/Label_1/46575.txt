import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int sum = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            sum += a[i];
        }
        int ans = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            cur += a[i];
            if (cur >= sum / 2) {
                ans = i + 1;
                break;
            }
        }
        System.out.println(ans);
    }
}
