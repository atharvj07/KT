import java.util.*;

class Main {
    static int dfs(int cur, int prev, int[] vis, List<List<Edge>> g, int[] ans) {
        vis[cur] = 1;
        int sum = 0;
        for (Edge e : g.get(cur)) {
            if (vis[e.to] == 0 && e.to != prev) {
                sum += dfs(e.to, cur, vis, g, ans);
            } else if (e.to != prev) {
                ans[0] = Math.min(ans[0], e.cost);
            }
        }
        return sum + 1;
    }

    static class Edge implements Comparable<Edge> {
        int to;
        int cost;

        public Edge(int to, int cost) {
            this.to = to;
            this.cost = cost;
        }

        @Override
        public int compareTo(Edge o) {
            return cost - o.cost;
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        List<List<Edge>> g = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            g.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            int c = in.nextInt();
            g.get(a).add(new Edge(b, c));
            g.get(b).add(new Edge(a, c));
        }
        int[] vis = new int[n + 1];
        int[] ans = new int[]{1 << 30};
        int sum = dfs(1, 1, vis, g, ans);
        if (sum != n) {
            System.out.println(-1);
            return;
        }
        System.out.println(ans[0]);
    }
}
