import java.util.Scanner;
import java.util.stream.IntStream;
public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        String s = cin.next();
        System.out.println(IntStream.range(0, n - 1).filter(i -> s.charAt(i) != s.charAt(i + 1)).count());
    }
}
