#include <vector>
#include <iostream>

int main() {
    size_t n, m, k;
    std::cin >> n >> m >> k;

    std::vector<bool> a(n), b(m);
    for (size_t i = 0; i < n; i++) {
        int z; std::cin >> z; a[i] = z;
    }
    for (size_t i = 0; i < m; i++) {
        int z; std::cin >> z; b[i] = z;
    }

    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(m + 1, 0));
    for (size_t i = 1; i <= n; i++) {
        for (size_t j = 1; j <= m; j++) {
            if (a[i - 1] and b[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = 0;
            }
        }
    }

    long long ans = 0;
    for (size_t i = 1; i <= n; i++) {
        for (size_t j = 1; j <= m; j++) {
            if (dp[i][j] < k) {
                continue;
            }

            size_t l = 1, r = dp[i][j];
            while (l <= r) {
                size_t mid = (l + r) / 2;

                bool ok = true;
                for (size_t x = i - mid; x <= i; x++) {
                    for (size_t y = j - mid; y <= j; y++) {
                        if (!a[x] || !b[y]) {
                            ok = false;
                        }
                    }
                }

                if (ok) {
                    l = mid + 1;
                } else {
                    r = mid - 1;
                }
            }

            ans += r;
        }
    }
    std::cout << ans << '\n';
}
