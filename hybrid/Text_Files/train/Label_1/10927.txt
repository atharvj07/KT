#include <bits/stdc++.h>
using namespace std;
const int mxN = 2003;
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);
    int n, h, l, r;
    cin >> n >> h >> l >> r;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    vector<bool> dp(h, false);
    for (int i = 0; i < h; ++i) {
        if (i >= l && i < r) { dp[i % h] = true; }
    }
    vector<int> pre(h, 0);
    for (int i = 0; i < h; ++i) { pre[i] = dp[i] + (i ? pre[i - 1] : 0); }
    int maxGood = 0;
    for (int i = 0; i < n; ++i) {
        int remain = (a[i] - 1) % h;
        maxGood += dp[remain] + (remain ? pre[remain - 1] : 0);
    }
    cout << maxGood << '\n';
}
