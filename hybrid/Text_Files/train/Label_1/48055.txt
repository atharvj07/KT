import java.util.*;
public class Main {
    static long[] dpL, dpR;
    static int n, l, r, ql, qr;
    static int[] w;
    static long solve(int i, boolean fromLeft) {
        if (i == n)
            return 0;
        if (fromLeft) {
            if (dpL[i] != -1)
                return dpL[i];
            long takeL = w[i] * l + (i == 0 ? 0 : ql);
            long takeR = w[n - i - 1] * r + (i == 0 ? 0 : qr) + solve(i + 1, false);
            return dpL[i] = Math.min(takeL, takeR);
        } else {
            if (dpR[i] != -1)
                return dpR[i];
            long takeL = w[i] * l + (i == 0 ? 0 : ql) + solve(i + 1, true);
            long takeR = w[n - i - 1] * r + (i == 0 ? 0 : qr);
            return dpR[i] = Math.min(takeL, takeR);
        }
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        l = sc.nextInt();
        r = sc.nextInt();
        ql = sc.nextInt();
        qr = sc.nextInt();
        dpL = new long[n];
        dpR = new long[n];
        Arrays.fill(dpL, -1);
        Arrays.fill(dpR, -1);
        w = new int[n];
        for (int i = 0; i < n; i++)
            w[i] = sc.nextInt();
        long res = solve(0, true);
        System.out.println(res);
    }
}
