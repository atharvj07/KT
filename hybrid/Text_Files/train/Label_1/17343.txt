#include<bits/stdc++.h>
using namespace std ; 
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin>>t;
    while(t--)
    {
        int n,a[1001]={0},x,y,s=0; 
        string st;
        cin>>n>>st;
        for(int i=0;i<n;i++)
        {
            if(st[i]=='1')
            {
                if(a[i]==0 and a[i+1]==0)
                {
                    a[i]=1;
                    a[i+1]=1;
                    s+=2;
                }
            }
        }
        for(int i=0;i<n;i++)
        {
            if(a[i]==0)
            {
                s++;
                a[i]=1;
            }
        }
        cout<<s<<"\n";
    }
    return 0 ; 
}
