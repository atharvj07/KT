import java.util.*;

class Main {
  static int N, M, tick = 1;
  static int[] featureRequirement = new int[256];
  static int[] featureFunctionality = new int[256];
  static int[][] cd = new int[256][256];
  static int[][] ct = new int[256][256];
  static ArrayList<Integer> active = new ArrayList<>();

  static void dfs(int node) {
    if (active.contains(node) || ct[1][node] == 0) {
      return;
    }
    ct[1][node] = 1;
    active.add(node);
    for (int i = 1; i <= N; i++) {
      if (i == node) {
        continue;
      }
      if (cd[node][i] > 0 && ct[i][node] == 0) {
        dfs(i);
      }
    }
  }

  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    N = input.nextInt();
    for (int i = 1; i <= N; i++) {
      for (int j = 1; j <= 3; j++) {
        featureRequirement[i] = featureRequirement[i] * 2 + input.nextInt();
      }
      for (int j = 1; j <= 3; j++) {
        featureFunctionality[i] = featureFunctionality[i] * 2 + input.nextInt();
      }
    }

    for (int i = 1; i <= N; i++) {
      if ((featureRequirement[1] & featureFunctionality[i]) != featureRequirement[1]) {
        System.out.println("Impossible");
        return;
      }
    }

    active.add(1);
    for (int i = 1; i <= N; i++) {
      if (i == 1) {
        continue;
      }
      ct[1][i] = 1;
      for (int j = 1; j <= N; j++) {
        if (i == j) {
          continue;
        }
        cd[i][j] = tick;
        tick++;
        if (tick > 264) {
          System.out.println("Impossible");
          return;
        }
        cd[j][i] = tick;
        tick++;
        if (tick > 264) {
          System.out.println("Impossible");
          return;
        }
        M += 2;
      }
      dfs(i);
    }

    int[] hasCT = new int[N + 1];
    for (int i = 1; i <= N; i++) {
      for (int j = 1; j <= N; j++) {
        hasCT[i] |= ct[i][j];
      }
    }

    System.out.println("Possible");
    for (int i = 1; i <= N; i++) {
      System.out.print(hasCT[i] + " ");
    }
    System.out.println();
    System.out.println(M);
    for (int i = 1; i <= N; i++) {
      for (int j = 1; j <= N; j++) {
        if (cd[i][j] != 0) {
          System.out.println(i + " " + j);
        }
      }
    }
  }
}
