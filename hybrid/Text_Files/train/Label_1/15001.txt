#include <bits/stdc++.h>
using namespace std;
const int mx=5005;
int n,a[mx],ans;
int main(){
  ios::sync_with_stdio(false);cin.tie(0);
  cin>>n;
  for(int i=1;i<=n;i++)cin>>a[i];
  int l=1,r=1,sum=a[1];
  while(r<=n){
    if(sum<=100*r)r++;
    else sum-=a[l++];
    ans=max(ans,r-l+1);
  }
  cout<<ans<<endl;
}
