import sys
input = sys.stdin.readline

def solve(l, r):
  ret = 0
  while l != r:
    mid = (l + r + 1) // 2
    if mid == r: mid -= 1
    ret += int(sa[mid] >= 10)
    r = mid
  return ret

n = int(input())
s = input().strip()
sa = []; cur = 0
for i in range(n):
  cur += int(s[i])
  sa.append(cur)
for i in range(int(input())):
  print(solve(*map(int, input().split())))
