for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    i = 0
    j = 1
    k = 2
    while k < n:
        if a[j]+a[k] <= a[i]:
            j += 1
            k += 1
        else:
            break
    if i+1 == j or j+1 == k or k == n:
        print(-1)
    else:
        print(' '.join(map(str, [i+1, j+1, k+1])))
