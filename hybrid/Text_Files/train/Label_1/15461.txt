#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  int p1 = 1, p2 = 1e6;
  for (int i = 0; i < n; i++) {
    if (abs(a[i] - p1) < abs(a[i] - p2)) {
      ans += abs(a[i] - p1);
      p1 = a[i];
    } else {
      ans += abs(a[i] - p2);
      p2 = a[i];
    }
  }
  cout << ans << endl;
  return 0;
}
