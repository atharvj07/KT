import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    segments = [[]]
    for _ in range(n-1):
        k, *arr = map(int, input().split())
        segments.append(arr)
    visited = [False for _ in range(n+1)]
    result = []
    for i in range(1,n+1):
        if not visited[i]:
            for segment in segments:
                if i in segment:
                    visited[i] = True
                    result += segment
                    break
    return result

t = int(input())
for _ in range(t):
    print(*solve())
