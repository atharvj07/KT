import random

x, y = map(int, input().split())
if x and not y:
    print(-1)
else:
    s = bin(x)[2:]
    if s == "0":
        k = 0
    else:
        k = max(s.rfind('1'), s.rfind('0')) + 1
    l = []
    while k > 0:
        if k <= 18:
            l.append(1 << k)
            y -= 1 << k
            k -= 1
        elif random.randint(1, 2**k + 10**5) % 2 == 0:
            l.append(y)
            break
        else:
            l.append(1 << (k - 1))
            y -= 1 << (k - 1)
            k -= 1
    if y > 0:
        l.append(y)
    print(len(l))
    print(*l)
