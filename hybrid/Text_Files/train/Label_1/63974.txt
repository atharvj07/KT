n = int(input())
s = [list(input()) for _ in range(n)]
cnt = [[0 for _ in range(11)] for _ in range(11)]
for i in range(n):
    for j in range(1,6):
        cnt[int(s[i][j])][int(s[i][j-1])] += 1
ans = 0
for j in range(10):
    for i in range(10):
        ans = max(ans, cnt[i][j] + cnt[j][i])
print(ans)
