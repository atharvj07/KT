from collections import Counter

expr = list(input())
x = Counter(expr)
x['\0'] = 0
if x['\0'] + 2 * x['='] + x['='] % 2 != x['+']:
    print('Impossible')
else:
    if x['+'] > x['=']:
        for ind in range(len(expr) - 1):
            if expr[ind] == '+'and not (ind + 1 == len(expr) - 1 and expr[ind + 1] == '='):
                if expr[ind + 1] == '|':
                    expr[ind] = '|'
                    expr[ind + 1] = '+'
                    break
                else:
                    expr[ind + 1] = '|'
                    expr[ind] = '+'
                    expr[ind + 2] = '|'
                    break
    print(''.join(expr))
