import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int k = sc.nextInt();

            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }

            int[] gaps = new int[n];
            for (int j = 0; j < n - 1; j++) {
                gaps[j] = a[j + 1] - a[j];
            }

            int[] freqs = new int[101];
            for (int j = 0; j < n - 1; j++) {
                freqs[gaps[j]]++;
            }

            Arrays.sort(freqs);
            int rem = n - 1 - k;
            while (rem > 0 && freqs[freqs.length - 1] > 0) {
                freqs[freqs.length - 1]--;
                freqs[freqs.length - 2]++;
                rem--;
            }

            int sum = 0;
            for (int j = 0; j < freqs.length - 1; j++) {
                sum += freqs[j];
            }

            if (sum == n - 1 - k) {
                System.out.println(k + 1);
            } else {
                System.out.println(-1);
            }
        }
    }
}
