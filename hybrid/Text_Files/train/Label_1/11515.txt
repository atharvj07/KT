#include<bits/stdc++.h>
using namespace std;
int main() {
  int n; cin >> n;
  vector<int> a(n + 2);
  for (int i = 1; i <= n; ++i) cin >> a[i];
  int l = 1, r = 1;
  for (int i = 2; i <= n; ++i)
    if (a[i] != a[i - 1]) { l = i - 1; break; }
  for (int i = n - 1; i >= 1; --i)
    if (a[i] != a[i + 1]) { r = i + 1; break; }
  cout << r - l << endl;
}
