#include<bits/stdc++.h>
#define mod (int)(1e9+7)
using namespace std;
int n, m, f[100005], a[100005], l[100005], r[100005], ll[100005], rr[100005];
string s, t;
int main() {
    cin >> s >> t;
    n = s.size();
    m = t.size();
    for (int i = 0; i < n; i++) {
        if (s[i] == t[0]) {
            a[i] = 1;
        }
    }
    for (int i = 1; i < n; i++) {
        a[i] += a[i - 1];
    }
    for (int i = 0; i < n; i++) {
        if (s[i] == t[m - 1]) {
            r[i] = 1;
        }
    }
    for (int i = n - 2; i >= 0; i--) {
        r[i] += r[i + 1];
    }
    ll[0] = 0;
    for (int i = 1; i < n; i++) {
        if (s[i] == t[0]) {
            ll[i] = ll[i - 1] + 1;
        } else {
            ll[i] = ll[i - 1];
        }
    }
    rr[n - 1] = 0;
    for (int i = n - 2; i >= 0; i--) {
        if (s[i] == t[m - 1]) {
            rr[i] = rr[i + 1] + 1;
        } else {
            rr[i] = rr[i + 1];
        }
    }
    f[0] = 1;
    for (int i = 0; i < n; i++) {
        if (s[i] != t[0]) {
            continue;
        }
        for (int j = i + m - 1; j < n; j++) {
            if (s[j] != t[m - 1]) {
                continue;
            }
            long long num = a[i - 1];
            num = num * r[j + 1] % mod;
            num = num * ll[i] % mod;
            num = num * rr[j] % mod;
            num = num * f[i - m] % mod;
            f[j] = (f[j] + num) % mod;
        }
    }
    cout << f[n - 1] << endl;
    return 0;
}
