from sys import stdin

s = stdin.readline().strip()
n = len(s)
dp = [[n for _ in range(10)] for _ in range(10)]
dp[0][0] = 0
for i in range(n):
    if s[i] == '0':
        for j in range(10):
            for k in range(10):
                dp[j][k] = min(dp[j][k], dp[0][0] + i)
    else:
        a = int(s[i])
        for j in range(10):
            for k in range(10):
                if j + a == k:
                    dp[j][k] = min(dp[j][k], i)
                elif k + a == j:
                    dp[j][k] = min(dp[j][k], i)
                else:
                    dp[j][k] = min(dp[j][k], i + 1)
for i in range(10):
    for j in range(10):
        if dp[i][j] == n:
            dp[i][j] = -1
for row in dp:
    print(' '.join(str(x) for x in row))
