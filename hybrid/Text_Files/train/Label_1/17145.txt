#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define endl '\n'
#define mod 1000000007
int main() {
  ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    ll n;
    cin >> n;
    int cnt = 0, flag = 0;
    while (n > 0) {
      if (n % 10 != 0) {
        if (flag == 0)
          cnt++, flag = 1;
        n = n / 10;
      } else if (flag == 0)
        n = n / 10;
      else
        break;
    }
    cout << cnt << endl;
  }
}
