n=int(input())
a=[i%n for i in range(n)]
b=[0]*n
c=[(2*i)%n for i in range(n)]
assert 1<=n<=1e5
if n%2==0:
    if n%4!=0:
        a[0], a[1] = a[1], a[0]
elif n%4==1:
    b[n//2]=n//2
elif n%4==3:
    a, c = c, a
if -1 in [a[i]+b[i]-c[i]for i in range(n)]:
    print(-1)
else:
    print(" ".join(str(j) for j in a))
    print(" ".join(str(j) for j in b))
    print(" ".join(str(j) for j in c))
