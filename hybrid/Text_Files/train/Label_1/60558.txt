n = int(input())
a = []
b = []
h = []
for i in range(n):
    x, y, z = map(int, input().split())
    a.append(x)
    b.append(y)
    h.append(z)
dp = [0] * n
dp[0] = h[0]
for i in range(1, n):
    for j in range(i):
        if b[i] > a[j] and dp[i] < dp[j] + h[i]:
            dp[i] = dp[j] + h[i]
print(max(dp))
