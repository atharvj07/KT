import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int d = sc.nextInt();
        int[] affected = new int[n + 1];
        for (int i = 0; i < m; i++) {
            affected[sc.nextInt()] = 1;
        }
        int[][] adj = new int[n + 1][n + 1];
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a][b] = 1;
            adj[b][a] = 1;
        }
        int[] dist = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            if (affected[i] == 0) {
                continue;
            }
            Queue<Integer> q = new LinkedList<>();
            q.offer(i);
            dist[i] = 0;
            while (!q.isEmpty()) {
                int u = q.poll();
                for (int v = 1; v <= n; v++) {
                    if (adj[u][v] == 0 || dist[v] != 0) {
                        continue;
                    }
                    dist[v] = dist[u] + 1;
                    if (dist[v] > d) {
                        break;
                    }
                    q.offer(v);
                }
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] == 0) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
