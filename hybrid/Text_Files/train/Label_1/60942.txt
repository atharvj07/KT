from collections import defaultdict
from heapq import heappush, heappop

def solve():
    n, m = map(int, input().split())
    x = [0] * (n + 1)
    y = [0] * (n + 1)
    for i in range(1, n + 1):
        a, b = map(int, input().split())
        x[i] = a
        y[i] = b
    edges = defaultdict(list)
    for _ in range(m):
        a, b = map(int, input().split())
        edges[a].append(b)
        edges[b].append(a)
    ans = [0] * (n + 1)
    vis = [False] * (n + 1)
    for i in range(1, n + 1):
        if not vis[i]:
            pq = []
            heappush(pq, (0, i, 0))
            while pq:
                cur, node, tp = heappop(pq)
                if vis[node]: continue
                vis[node] = True
                ans[node] += cur
                if tp == 0:
                    for u in edges[node]:
                        heappush(pq, (x[node] + y[u], u, 1))
                else:
                    for u in edges[node]:
                        heappush(pq, (x[u] + y[node], u, 0))
    for i in range(1, n + 1):
        print(ans[i])

solve()
