#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m, x, y, z, p, q;
  cin >> n >> m >> x >> y >> z >> p;
  vector<pair<int, int>> v(p);
  for (int i = 0; i < p; i++) {
    int a, b;
    cin >> a >> b;
    v[i] = {a, b};
  }
  x %= 4, y %= 2, z %= 4;
  q = x + y + z;
  q %= 4;
  for (int i = 0; i < p; i++) {
    int a = v[i].first, b = v[i].second;
    if (q == 0)
      cout << a << ' ' << b << '\n';
    else if (q == 1)
      cout << m - b + 1 << ' ' << a << '\n';
    else if (q == 2)
      cout << n - a + 1 << ' ' << m - b + 1 << '\n';
    else if (q == 3)
      cout << b << ' ' << n - a + 1 << '\n';
  }
  return 0;
}
