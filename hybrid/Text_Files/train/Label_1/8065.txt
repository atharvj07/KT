#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  for (int i = 0; i < n; ++i) {
    if (i % 2)
      b[i] = (a[i] + 1) % 2;
    else
      b[i] = a[i];
  }
  for (int i = 0; i < n; ++i) { cout << b[i] << " "; }
  return 0;
}
