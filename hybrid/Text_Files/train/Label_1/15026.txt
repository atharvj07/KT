#include <bits/stdc++.h>

using namespace std;
#define int long long

const int maxn = 3e5;

int a[maxn], b[maxn];

vector<tuple<int, int, int>> cs[maxn];
int c[maxn], d[maxn], gp[maxn];

int32_t main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
    }
    vector<int> e;
    for (int i = 1; i <= n; i++) {
        e.push_back(a[i]);
    }
    for (int i = 1; i <= n; i++) {
        e.push_back(d[i]);
    }
    sort(e.begin(), e.end());
    e.erase(unique(e.begin(), e.end()), e.end());
    int cnt = 1;
    for (int x : e) {
        c[cnt] = x;
        cnt++;
    }
    for (int i = 1; i <= n; i++) {
        a[i] = lower_bound(c + 1, c + 1 + cnt, a[i]) - c;
        d[i] = lower_bound(c + 1, c + 1 + cnt, d[i]) - c;
    }
    for (int i = 1; i <= n; i++) {
        gp[i] = d[i + 1] - d[i];
    }
    for (int i = 0; i <= cnt; i++) cs[i].clear();
    for (int i = 1; i <= n; i++) cs[d[i]].push_back(make_tuple(-gp[i], -b[i], -i));
    for (int i = 0; i <= cnt; i++) {
        sort(cs[i].begin(), cs[i].end());
    }
    int ans = 0;
    for (int i = 1; i <= cnt; i++) {
        int now = 0;
        for (auto it : cs[i]) {
            int g, p, id;
            tie(g, p, id) = it;
            now += g;
            ans = max(ans, m * (i - d[id]) + p + now);
        }
    }
    cout << ans << endl;
    return 0;
}
