#include <bits/stdc++.h>
using namespace std;
bool ok[420];
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    for (int i = 1; i < 418; i++) { ok[i] = ((i % 3 == 0 && (i / 3) % 2 == 0) || (i % 2 == 0 && (i / 2) % 3 == 0)); }
    for (int i = 0; i < n; i++) {
        string y;
        cin >> y;
        sort(y.begin(), y.end());
        if (y.size() == 1 && y[0] == '0')
            cout << "red" << endl;
        else {
            if (y.size() > 2 && (y[0] == '0' || y[y.size() - 1] == '0'))
                cout << "cyan" << endl;
            else if (ok[(int)y[y.size() - 1] - 48])
                cout << "red" << endl;
            else
                cout << "cyan" << endl;
        }
    }
}
