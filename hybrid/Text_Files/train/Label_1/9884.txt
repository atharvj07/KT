#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 5;
int n, m, d, t;
ll ans;
vector<int> adj[maxn], anss;
bool visited[maxn];
void dfs(int u) {
  if (u == 1)
    t++;
  if (t == d)
    anss.push_back(u);
  for (auto v : adj[u])
    if (!visited[v])
      visited[v] = true, dfs(v);
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n >> m >> d;
  while (m--) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  if (d > n - 1) {
    cout << "NO";
    return 0;
  }
  visited[1] = true;
  dfs(1);
  if (t != d) {
    cout << "NO";
    return 0;
  }
  cout << "YES\n";
  for (auto v : anss)
    for (auto u : adj[v])
      if (!visited[u])
        visited[u] = true, cout << v << " " << u << "\n";
  return 0;
}
