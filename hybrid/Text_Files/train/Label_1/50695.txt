import java.util.*;

public class Main {

    static class UnionFind {

        private int[] parent;
        private int[] size;

        UnionFind(int n) {
            parent = new int[n + 1];
            size = new int[n + 1];

            for (int i = 0; i <= n; i++) {
                parent[i] = i;
                size[i] = 1;
            }
        }

        int find(int x) {
            if (parent[x] == x) {
                return x;
            }

            return parent[x] = find(parent[x]);
        }

        void union(int a, int b) {
            int rootA = find(a);
            int rootB = find(b);

            if (rootA != rootB) {
                if (size[rootA] < size[rootB]) {
                    parent[rootA] = rootB;
                    size[rootB] += size[rootA];
                } else {
                    parent[rootB] = rootA;
                    size[rootA] += size[rootB];
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt();
        int m = input.nextInt();

        int[] energy = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            energy[i] = input.nextInt();
        }

        UnionFind uf = new UnionFind(n);

        for (int i = 0; i < m; i++) {
            int a = input.nextInt();
            int b = input.nextInt();

            uf.union(a, b);
        }

        int totalEnergy = 0;

        for (int i = 1; i <= n; i++) {
            int root = uf.find(i);

            if (uf.size[root] == 1) {
                totalEnergy += energy[i];
            }
        }

        System.out.println(totalEnergy);
    }
}
