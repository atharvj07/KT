#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  cin >> m;
  for (int i = 0; i < m; i++) {
    long long x, y;
    cin >> x >> y;
    long long ans = 1e18;
    for (int j = 0; j < n; j++) {
      if (a[j] < x) {
        ans = min(ans, a[j] + x - 1);
      }
    }
    long long sum = 0;
    for (int j = 0; j < n; j++) { sum += a[j]; }
    if (sum < y) {
      ans = min(ans, y - sum);
    }
    cout << ans << endl;
  }
  return 0;
}
