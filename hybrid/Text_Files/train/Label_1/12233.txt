#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<string> ss(n);
  for (int i = 0; i < n; ++i) cin >> ss[i];
  sort(ss.begin(), ss.end(), [&](const string& lhs, const string& rhs) { return lhs + rhs < rhs + lhs; });
  for (const string& s : ss) cout << s;
  return 0;
}
