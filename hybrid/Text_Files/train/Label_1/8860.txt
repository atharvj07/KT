#include <bits/stdc++.h>
using namespace std;
int main()
{
    int t, n, a[200001], sum, cnt;
    cin >> t;
    while (t--)
    {
        cin >> n;
        sum = cnt = 0;
        for (int i = 0; i < n; i++)
        {
            cin >> a[i];
            sum += a[i];
        }
        sort(a, a + n);
        for (int i = 0; i < n - 1; i++)
        {
            if (a[i] < a[i + 1])
                cnt++;
            else
                break;
        }
        cout << cnt << '\n';
    }
    return 0;
}
