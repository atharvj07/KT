import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        double[][] points = new double[n][2];
        for (int i = 0; i < n; i++) {
            points[i][0] = sc.nextDouble();
            points[i][1] = sc.nextDouble();
        }
        double[] center = {0, 0};
        double totalArea = 0;
        for (int i = 0; i < n; i++) {
            center[0] += points[i][0];
            center[1] += points[i][1];
            totalArea += (points[(i + 1) % n][0] - points[i][0]) * (points[(i + 1) % n][1] + points[i][1]);
        }
        center[0] /= n;
        center[1] /= n;
        totalArea = Math.abs(totalArea) / 2;
        double[][][] area = new double[n][n][2];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                double[] line = lineEquation(points[i], points[j]);
                double[] intersection1 = intersection(line, points[(i + 1) % n]);
                double[] intersection2 = intersection(line, points[(j + 1) % n]);
                area[i][j][0] = Math.abs(totalArea * (areaTriangle(points[i], points[j], center) / triangleArea(points[i], points[j], intersection1)));
                area[i][j][1] = Math.abs(totalArea * (areaTriangle(points[i], points[j], center) / triangleArea(points[i], points[j], intersection2)));
            }
        }
        double[] answer = {0, 0};
        boolean found = false;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                double[] side1 = {points[(i + 1) % n][0] - points[i][0], points[(i + 1) % n][1] - points[i][1]};
                double[] side2 = {points[(j + 1) % n][0] - points[j][0], points[(j + 1) % n][1] - points[j][1]};
                if (side1[0] * side2[1] != side1[1] * side2[0]) {
                    double[] meet = meetLine(lineEquation(points[i], points[j]), lineEquation(points[(i + 1) % n], points[(j + 1) % n]));
                    if (meet != null) {
                        if (areaTriangle(points[i], points[j], meet) == area[i][j][0] && areaTriangle(points[(i + 1) % n], points[(j + 1) % n], meet) == area[i][j][1]) {
                            answer = meet;
                            found = true;
                            break;
                        }
                    }
                }
            }
            if (found) {
                break;
            }
        }
        if (found) {
            System.out.println(answer[0] + " " + answer[1]);
        } else {
            System.out.println("NA");
        }
    }

    public static double[] lineEquation(double[] p1, double[] p2) {
        double a = p2[1] - p1[1];
        double b = p1[0] - p2[0];
        double c = p1[1] * p2[0] - p1[0] * p2[1];
        return new double[]{a, b, c};
    }

    public static double[] meetLine(double[] l1, double[] l2) {
        double a1 = l1[0];
        double b1 = l1[1];
        double c1 = l1[2];
        double a2 = l2[0];
        double b2 = l2[1];
        double c2 = l2[2];
        double det = a1 * b2 - a2 * b1;
        if (det == 0) {
            return null;
        }
        double x = (b2 * c1 - b1 * c2) / det;
        double y = (a1 * c2 - a2 * c1) / det;
        return new double[]{x, y};
    }

    public static double[] intersection(double[] line, double[] point) {
        double a = line[0];
        double b = line[1];
        double c = line[2];
        double x = point[0];
        double y = point[1];
        double det = a * b - a * y - b * x + c;
        return new double[]{x - det * b / a, y + det * a / b};
    }

    public static double areaTriangle(double[] p1, double[] p2, double[] p3) {
        return 0.5 * (p1[0] * (p2[1] - p3[1]) + p2[0] * (p3[1] - p1[1]) + p3[0] * (p1[1] - p2[1]));
    }

    public static double triangleArea(double[] a, double[] b, double[] c) {
        return Math.abs((a[0] * (b[1] - c[1]) + b[0] * (c[1] - a[1]) + c[0] * (a[1] - b[1])) / 2);
    }
}
