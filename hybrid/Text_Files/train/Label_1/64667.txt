n=int(input())
a=list(map(int,input().split()))
l=[0]*n
r=[0]*n
for i in range(n):
	l[i]=1
for i in range(n):
	for j in range(i):
		if a[i]>a[j]:
			l[i]=max(l[i],l[j]+1)
for i in range(n-1,-1,-1):
	r[i]=1
for i in range(n-1,-1,-1):
	for j in range(n-1,i,-1):
		if a[i]>a[j]:
			r[i]=max(r[i],r[j]+1)
pos=[]
for i in range(n):
	if l[i]==1 and r[i]==1:
		pos.append(3)
	elif l[i]==1:
		pos.append(2)
	elif r[i]==1:
		pos.append(1)
	else:
		pos.append(2)
print(''.join(list(map(str,pos))))
