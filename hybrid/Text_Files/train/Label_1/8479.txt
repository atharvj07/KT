#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int ans = 0;
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] > i + 1) {
            ans += a[i] - i - 1;
        } else if (a[i] < i + 1) {
            cnt++;
        }
    }
    ans += cnt / 2 + cnt % 2;
    cout << ans << endl;
    return 0;
}
