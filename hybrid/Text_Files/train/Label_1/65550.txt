g = input()
p = input()
n = int(input())
for i in range(n):
    s = input()
    good = True
    for j in range(len(s)):
        if s[j] != p[j] and p[j] != '?' and p[j] != '*':
            good = False
    if not good:
        print("NO")
        continue
    if p.count('*') == 1:
        p2 = p.split('*')
        if s.find(p2[0]+p2[1]) != -1:
            print("YES")
        else:
            print("NO")
    else:
        print("YES")
