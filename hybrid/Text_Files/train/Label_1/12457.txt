#include <bits/stdc++.h>

using i64 = long long;

const int mod = 998244353;
const int P = 1000000007;

i64 binpow(i64 a, i64 b){
    i64 res = 1;
    while(b){
        if(b % 2) res = (res * a) % mod;
        a = (a * a) % mod;
        b /= 2;
    }
    return res;
}

int main(){
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int n, k;
    std::cin >> n >> k;
    std::vector<int> h(n);
    for(int i = 0; i < n; i++)  std::cin >> h[i];

    std::vector<std::vector<i64>> dp(n + 1, std::vector<i64>(k + 1, 0));
    dp[n][0] = 1;
    for(int i = n - 1; i >= 0; i--){
        for(int j = 0; j <= k; j++){
            dp[i][j] = (dp[i][j] + dp[i + 1][j] + dp[i + 1][(j + h[i % n]) % k]) % mod;
        }
    }

    std::vector<int> fact(k + 1, 1);
    for(int i = 2; i <= k; i++)  fact[i] = (i64)i * fact[i - 1] % mod;

    std::vector<int> inv(k + 1, 1);
    for(int i = 2; i <= k; i++)  inv[i] = binpow(i, mod - 2);

    auto ncr = [&](int n, int r){
        if(n < r)  return 0ll;
        return ((i64)fact[n] * inv[n - r]) % mod * inv[r] % mod;
    };

    i64 ans = 0;
    for(int i = 0; i <= k; i++){
        for(int j = 1; j <= k; j++){
            if(i == j)  continue;
            ans = (ans + ((i64)dp[0][i] * ncr(k, i) % mod) * ((i64)dp[0][j] * ncr(k, j) % mod) % mod * ncr(k - 1, j)) % mod;
        }
    }

    std::cout << ans << "\n";
    return 0;
}
