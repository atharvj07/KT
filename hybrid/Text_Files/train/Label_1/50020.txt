import java.util.Scanner;
import java.util.LinkedList;
import java.util.Queue;

public class GameExpansion {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();
        int[] speeds = new int[p];
        for (int i = 0; i < p; i++) {
            speeds[i] = scanner.nextInt();
        }
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String row = scanner.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = row.charAt(j);
            }
        }

        int[] result = solve(grid, speeds);
        for (int count : result) {
            System.out.print(count + " ");
        }
    }

    private static int[] solve(char[][] grid, int[] speeds) {
        int n = grid.length;
        int m = grid[0].length;
        int p = speeds.length;
        int[][] playerGrid = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] != '.') {
                    playerGrid[i][j] = grid[i][j] - '0';
                }
            }
        }

        int[] result = new int[p];
        boolean[][] visited = new boolean[n][m];
        for (int player = 0; player < p; player++) {
            Queue<int[]> queue = new LinkedList<>();
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (playerGrid[i][j] == player + 1) {
                        queue.offer(new int[]{i, j});
                        visited[i][j] = true;
                    }
                }
            }

            while (!queue.isEmpty()) {
                int[] current = queue.poll();
                int x = current[0];
                int y = current[1];
                result[player]++;
                for (int dx = -speeds[player]; dx <= speeds[player]; dx++) {
                    for (int dy = -speeds[player]; dy <= speeds[player]; dy++) {
                        int nx = x + dx;
                        int ny = y + dy;
                        if (nx >= 0 && nx < n && ny >= 0 && ny < m && !visited[nx][ny] && grid[nx][ny] == '.') {
                            visited[nx][ny] = true;
                            queue.offer(new int[]{nx, ny});
                        }
                    }
                }
            }
        }

        return result;
    }
}
