n = int(input())
t = [int(x) for x in input().split()]
if n % 2 == 0:
    odd = []
    even = []
    for i in range(n, 0, -1):
        if i % 2 == 0:
            even.append(t[i - 1])
        else:
            odd.append(t[i - 1])
    odd.sort()
    even.sort()
    ans = 0
    for i in range(min(len(odd), len(even))):
        if odd[i] < even[i]:
            ans += 1
else:
    odd = []
    even = []
    for i in range(n, 0, -1):
        if i % 2 == 1:
            even.append(t[i - 1])
        else:
            odd.append(t[i - 1])
    odd.sort()
    even.sort()
    ans = 0
    for i in range(min(len(odd), len(even))):
        if odd[i] > even[i]:
            ans += 1
print(ans)
