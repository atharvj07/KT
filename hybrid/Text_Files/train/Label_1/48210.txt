import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        int lastSorted = Integer.MIN_VALUE;
        int swaps = 0;
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
            if (a[i] < lastSorted) {
                swaps++;
            }
            lastSorted = a[i];
        }
        System.out.println(swaps <= 1 ? "YES" : "NO");
    }
}
