def calculate_score():
    m = list(map(int, input().split()))
    w = list(map(int, input().split()))
    hs, hu = map(int, input().split())

    total_score = 0
    for i in range(5):
        x = 500 * (i + 1)
        score = max(0.3 * x, (1 - m[i] / 250) * x - 50 * w[i])
        total_score += score

    total_score += hs * 100 - hu * 50
    return int(total_score)

print(calculate_score())
