#include <iostream>
#include <vector>
#include <algorithm>
#include <utility>
using namespace std;
 
#define ll long long
 
int main() {
    ll t;
    cin >> t;
    while(t--) {
        ll p, f, cnt_s, cnt_w, s, w;
        cin >> p >> f >> cnt_s >> cnt_w >> s >> w;
 
        ll sum = p + f;
        ll ans = min({cnt_s + cnt_w, sum / s + (sum % s == 0 ? 0 : 1),
                (sum - s + w - 1) / w + (sum % s == 0 ? 0 : 1)});
        cout << ans << endl;
    }
    return 0;
}
