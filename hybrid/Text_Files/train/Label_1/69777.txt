n = int(input())
s = input()
beauty = list(map(int, input().split()))

from collections import Counter

letter_counts = Counter(s)
max_beauty = 0

# Check if it's possible to form an antipalindromic string
for letter, count in letter_counts.items():
    if count > n // 2:
        print(0)
        exit()

# Calculate max beauty
letter_positions = {letter: [] for letter in set(s)}
for i, letter in enumerate(s):
    letter_positions[letter].append(i)

sorted_beauties = sorted(((letter, sum(sorted(beauty[pos] for pos in positions)[-len(positions) // 2:])) for letter, positions in letter_positions.items()), key=lambda x: x[1], reverse=True)

for _, b in sorted_beauties:
    max_beauty += b

print(max_beauty)
