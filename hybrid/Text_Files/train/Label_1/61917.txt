n, x, y = map(int, input().split())
rain_amounts = list(map(int, input().split()))

for i in range(n):
    left_bound = max(0, i-x)
    right_bound = min(n, i+y+1)
    if all(rain_amounts[i] < rain_amounts[j] for j in range(left_bound, i)) and all(rain_amounts[i] < rain_amounts[j] for j in range(i+1, right_bound)):
        print(i+1)
        break
