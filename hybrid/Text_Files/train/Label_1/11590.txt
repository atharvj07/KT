#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a, b, k;
    cin >> n >> a >> b >> k;
    string s;
    cin >> s;
    int l = -1, r = -1;
    for (int i = 0; i < n; i++) {
        if (s[i] == '1') {
            if (l == -1)
                l = i;
            else
                r = i;
        }
    }
    if (l == -1) {
        cout << 1 << '\n' << 1 << '\n';
    } else {
        if (s[l - 1] == '0') {
            cout << 1 << '\n' << l << '\n';
        } else if (l - b >= 0 && s[l - b] == '0') {
            cout << 1 << '\n' << l - b + 1 << '\n';
        } else if (s[l + 1] == '0') {
            cout << 1 << '\n' << l + 2 << '\n';
        } else if (r - l + 1 <= b) {
            cout << 1 << '\n' << l + b - r << '\n';
        } else {
            cout << 2 << '\n' << l + 1 << ' ' << r + b << '\n';
        }
    }
    return 0;
}
