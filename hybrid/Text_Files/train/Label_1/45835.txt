import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        Arrays.sort(s);
        Arrays.sort(t);
        boolean possible = true;
        for (int i = 0; i < n; i++) {
            if (s[i] > t[i]) {
                possible = false;
                break;
            }
        }
        if (!possible) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        int m = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == t[i]) {
                continue;
            }
            int d = (t[i] - s[i]) / 2;
            s[i] += d;
            s[i + 1] -= d;
            System.out.println((i + 1) + " " + (i + 2) + " " + d);
            m++;
        }
        System.out.println(m);
    }
}
