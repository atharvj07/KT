#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, W;
    cin >> n >> W;
    vector<int> w(n);
    for (int i = 0; i < n; i++) { cin >> w[i]; }
    int half = (W + 1) / 2;
    int sum = 0;
    vector<int> res;
    for (int i = 0; i < n; i++) {
      if (sum + w[i] > W) continue;
      res.push_back(i + 1);
      sum += w[i];
      if (sum >= half) break;
    }
    if (sum >= half) {
      cout << res.size() << endl;
      for (int i : res) cout << i << " ";
      cout << endl;
    } else
      cout << -1 << endl;
  }
}
