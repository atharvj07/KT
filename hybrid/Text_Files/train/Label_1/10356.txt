#include <iostream>
#include <queue>
using namespace std;
int main() {
    int n, t, c, mx = 0, tm = 0, sm = 0;
    cin >> n;
    queue<int> q;
    for (int i = 0; i < n; i++) {
        cin >> t >> c;
        while (!q.empty() && sm < t) {
            sm++;
            if (!q.empty()) { q.pop(); }
        }
        if (sm < t) tm = t;
        q.push(c);
        sm += c;
        mx = max(mx, sm);
    }
    while (!q.empty()) {
        sm++;
        if (!q.empty()) { q.pop(); }
    }
    tm += sm - 1;
    cout << tm << " " << mx << endl;
}
