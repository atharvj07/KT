import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        char[][] matrix = new char[n][n / 4];

        for (int i = 0; i < n; i++) {
            matrix[i] = reader.readLine().toCharArray();
        }

        int maxCompression = 0;
        for (int x = 2; x * x <= n; x++) {
            if (n % x == 0) {
                if (checkCompression(matrix, x)) {
                    maxCompression = x;
                }
            }
        }

        System.out.println(maxCompression);
    }

    private static boolean checkCompression(char[][] matrix, int x) {
        int blockWidth = matrix[0].length * 4 / x;
        int blockHeight = matrix.length / x;

        char[][] compressedMatrix = new char[blockHeight][blockWidth];

        for (int i = 0; i < matrix.length; i++) {
            for (int j = 0; j < matrix[i].length; j++) {
                int blockI = i / x;
                int blockJ = j / x;
                int blockElementIndex = (i % x) * blockWidth + (j % x);

                if (compressedMatrix[blockI][blockJ] == 0) {
                    compressedMatrix[blockI][blockJ] = matrix[i][j];
                } else {
                    if (compressedMatrix[blockI][blockJ] != matrix[i][j]) {
                        return false;
                    }
                }
            }
        }

        return true;
    }
}
