#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 7;
vector<int> adj[N], ord;
int sz[N];

void dfs(int u, int p) {
    sz[u] = 1;
    for (int v : adj[u]) {
        if (v == p) continue;
        dfs(v, u);
        sz[u] += sz[v];
    }
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n + 7);
    unordered_map<int, int> m;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        a[i] = x;
        m[x] = i;
    }
    for (int i = 0; i < n - 1; i++) {
        if (a[i] > a[i + 1]) {
            adj[m[a[i]]].push_back(m[a[i + 1]]);
            adj[m[a[i + 1]]].push_back(m[a[i]]);
            ord.push_back(m[a[i]]);
            ord.push_back(m[a[i + 1]]);
        }
    }
    dfs(*ord.begin(), *ord.begin());
    int mx = 0, best;
    for (int x : ord) {
        if (sz[x] > mx) {
            best = x;
            mx = sz[x];
        }
    }
    cout << mx << '\n';
    return 0;
}
