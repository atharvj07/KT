import sys
readline = sys.stdin.readline

n = int(readline())
d = list(map(int, readline().split()))
d.sort()
L = []

for i in range(len(d) - 1):
    L.append(d[i + 1] - d[i])

for j in L:
    cnt = 0
    for k in L:
        if k % j == 0:
            cnt += 1
    if cnt == len(L):
        print(j)
        break
