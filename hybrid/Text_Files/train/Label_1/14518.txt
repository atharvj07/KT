#include <bits/stdc++.h>
#define ll long long
using namespace std;

ll n, x;
ll res[20];
ll ans[20];
ll bit[1 << 20];
ll lim;

void print(int l) {
    for (int i = 1; i <= l; i++) cout << ans[i] << " ";
    cout << "\n";
}

void dfs(int pos, ll now) {
    if (pos > lim) {
        if (bit[now ^ x] && bit[now]) {
            int l = 0;
            for (int i = 1; i <= lim; i++) if (bit[res[i]]) l++;
            if (l > ans[0]) {
                for (int i = 1; i <= lim; i++) ans[i] = res[i];
                ans[0] = l;
            }
        }
        return;
    }
    res[pos] = 0, bit[now] ^= 1;
    dfs(pos + 1, now);
    bit[now] ^= 1;
    res[pos] = (1 << (n - 1)) + pos - 1;
    bit[now] ^= 1;
    dfs(pos + 1, now);
    bit[now] ^= 1;
}

int main() {
    cin >> n >> x;
    lim = 1 << n;
    dfs(1, 0);
    cout << ans[0] << "\n";
    if (ans[0]) print(ans[0]);
    return 0;
}
