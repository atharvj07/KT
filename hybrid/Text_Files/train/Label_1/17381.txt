#include <bits/stdc++.h>

using namespace std;

int main() {
  long long x;
  cin >> x;
  if (x == 1) {
    cout << "1 1" << endl;
  } else {
    for (long long i = 1; i * i <= x; ++i) {
      if (x % i == 0) {
        long long a = i;
        long long b = x / i;
        cout << a << " " << b << endl;
        break;
      }
    }
  }
  return 0;
}
