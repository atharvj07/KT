import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            if (n == 1) {
                System.out.println(0);
            } else if (n % 6 == 0) {
                int moves = 0;
                while (n > 1) {
                    moves++;
                    if (n % 6 == 0) {
                        n /= 6;
                    } else {
                        n *= 2;
                    }
                }
                System.out.println(moves);
            } else {
                System.out.println(-1);
            }
        }
    }
}
