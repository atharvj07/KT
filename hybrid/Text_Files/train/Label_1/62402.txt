import sys
input=sys.stdin.readline

n,q=map(int,input().split())
s=list(input())
ans=[0]*q
for i in range(q):
    x,y=input().split()
    x=int(x)-1
    s[x]=y
    a=s.count('a')
    b=s.count('b')
    c=s.count('c')
    if a>=1 and b>=1 and c>=1:
        ans[i]=0
    else:
        ans[i]=min(a,min(b,c))
print(*ans, sep='\n')
