#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, a[101];
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int ok = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] % 2 == 0) {
        cout << 1 << endl << i + 1 << endl;
        ok = 1;
        break;
      }
    }
    if (!ok) {
      for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
          if ((a[i] + a[j]) % 2 == 0) {
            cout << 2 << endl << i + 1 << " " << j + 1 << endl;
            ok = 1;
            break;
          }
        }
        if (ok) break;
      }
    }
    if (!ok) cout << -1 << endl;
  }
  return 0;
}
