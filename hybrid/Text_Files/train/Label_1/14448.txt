#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int s, n;
    cin >> s >> n;
    vector<int> ans(n);
    for (int i = n - 1; i >= 0; i--) {
      for (int j = 10; j >= 1; j--) {
        if (s >= j) {
          s -= j;
          ans[i] = j;
          break;
        }
      }
    }
    for (auto &x : ans) cout << x << ' ';
    cout << endl;
  }
}
