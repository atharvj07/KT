#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;

    vector<pair<int, int>> eggs(n);
    for (int i = 0; i < n; i++) {
        int a, g;
        cin >> a >> g;
        eggs[i] = {a, g};
    }

    sort(eggs.begin(), eggs.end());

    int sa = 0, sg = 0;
    string ans = "";
    for (int i = 0; i < n; i++) {
        if (sa + eggs[i].first <= sg + eggs[i].second) {
            sa += eggs[i].first;
            sg += eggs[i].second;
            ans += 'A';
        } else {
            sa += eggs[i].second;
            sg += eggs[i].first;
            ans += 'G';
        }
    }

    if (abs(sa - sg) > 500) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }

    return 0;
}
