#include<bits/stdc++.h>
using namespace std;
int main(){
  int n; cin>>n;
  if(n==1||n==2){
    cout<<-1<<'\n';
    return 0;
  }
  if(n%2==1){
    cout<<0<<' '<<1<<' '<<1<<endl;
    return 0;
  }
  cout<<0<<' '<<1<<' '<<2<<endl;
  int v=n/2-1;
  for(int i=4; i<=n; i+=2){
    cout<<i-1<<' '<<i<<' '<<i-2<<endl;
    --v;
    if(v==0){
      cout<<2<<' '<<3<<' '<<0<<endl;
      break;
    }
  }
  return 0;
}
