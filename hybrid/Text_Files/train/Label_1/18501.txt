#include <bits/stdc++.h>
using namespace std;

const int maxn = 2e5+10;
int n;
char s[maxn], t[maxn];

int main() {
    cin >> n >> s;
    int ans = 0;
    for (int i = 2; i < n; ++i) {
        if (s[i] == s[i-1] && s[i] == s[i-2]) {
            t[i] = s[i] == 'R' ? 'G' : (s[i] == 'G' ? 'B' : 'R');
            ++ans;
        } else {
            t[i] = s[i];
        }
    }
    for (int i = 0; i < n; ++i) {
        if (t[i] == 0) {
            t[i] = s[i];
        }
    }
    cout << ans << '\n' << t << '\n';
    return 0;
}
