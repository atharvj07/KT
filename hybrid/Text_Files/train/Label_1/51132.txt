import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();
        int b = a[0];
        for (int i = 2; i < n; i += 2) b ^= a[i];
        for (int x : a) System.out.print((b ^ x) + " ");
    }
}
