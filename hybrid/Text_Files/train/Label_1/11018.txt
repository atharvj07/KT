#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e3 + 5;
char a[MAXN][MAXN];
int n, m, dis[MAXN][MAXN];

int bfs() {
  queue<pair<int, int>> q;
  memset(dis, -1, sizeof dis);
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (a[i][j] == 'G') q.emplace(i, j), dis[i][j] = 0;
    }
  }
  while (!q.empty()) {
    auto u = q.front();
    q.pop();
    if (a[u.first][u.second] == 'S') return dis[u.first][u.second];
    if (u.second + 1 <= m && a[u.first][u.second + 1] != '*' && dis[u.first][u.second + 1] == -1)
      q.emplace(u.first, u.second + 1), dis[u.first][u.second + 1] = dis[u.first][u.second] + 1;
  }
  return -1;
}
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
  }
  cout << bfs() << endl;
  return 0;
}
