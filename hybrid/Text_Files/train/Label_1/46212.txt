import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Integer, Integer> bestFriends = new HashMap<>();
        int[] genders = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            int bestFriend = sc.nextInt();
            int gender = sc.nextInt();
            bestFriends.put(i, bestFriend);
            genders[i] = gender;
        }

        int pairs = 0;
        int boyGirlPairs = 0;
        StringBuilder result = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            if (!bestFriends.containsKey(i)) {
                continue;
            }
            int bestFriend = bestFriends.remove(i);
            if (!bestFriends.containsKey(bestFriend)) {
                continue;
            }
            if (genders[i] != genders[bestFriend]) {
                boyGirlPairs++;
            }
            pairs++;
            result.append(i).append(" ").append(bestFriend).append("\n");
        }

        System.out.println(pairs + " " + boyGirlPairs);
        System.out.println(result);
    }
}
