#include <iostream>
#include <queue>
#include <vector>
using namespace std;
#define ll long long
pair<ll, ll> dir[4] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll n, m, r, c;
    cin >> n >> m >> r >> c;
    vector<vector<ll>> dist(n + 1, vector<ll>(m + 1, -1)), vis(n + 1, vector<ll>(m + 1, 0));
    queue<pair<ll, ll>> q;
    q.push({r, c});
    dist[r][c] = 0;
    vis[r][c] = 1;
    while (!q.empty()) {
      auto u = q.front();
      q.pop();
      for (ll j = 0; j < 4; j++) {
        ll x = u.first + dir[j].first, y = u.second + dir[j].second;
        if (x >= 1 && x <= n && y >= 1 && y <= m && vis[x][y] == 0) {
          vis[x][y] = 1;
          dist[x][y] = dist[u.first][u.second] + 1;
          q.push({x, y});
        }
      }
    }
    ll ans = 0;
    for (ll i = 1; i <= n; i++) {
      for (ll j = 1; j <= m; j++) {
        ans = max(ans, dist[i][j]);
      }
    }
    cout << ans << "\n";
  }
}
