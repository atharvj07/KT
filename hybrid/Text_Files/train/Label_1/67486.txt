from collections import deque
def iff(c):
    return c != 'D'
def main():
    n,m = map(int,input().split())
    d = [list(input()) for i in range(n)]
    mx = 0
    for i in range(n):
        for j in range(m):
            if d[i][j] == 'D':
                q = deque()
                q.append((i,j))
                c=4
                while q:
                    x,y = q.popleft()
                    if x == 0 or y == 0 or x == n-1 or y == m-1:
                        c=10
                    elif d[x][y] == 'D':
                        c = 4
                    elif d[x][y] != 'I':
                        c = 10
                    else:
                        if (x,y+1) not in q and not iff(d[x][y+1]): q.append((x,y+1))
                        if (x+1,y) not in q and not iff(d[x+1][y]): q.append((x+1,y))
                        if (x,y-1) not in q and not iff(d[x][y-1]): q.append((x,y-1))
                        if (x-1,y) not in q and not iff(d[x-1][y]): q.append((x-1,y))
                mx = max(mx,c)
    if mx == 10: print("Poor Dima!")
    elif mx == 4: print("Poor Inna!")
    else: print(mx//4)
main()
