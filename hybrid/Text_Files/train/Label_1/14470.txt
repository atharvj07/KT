#include <bits/stdc++.h>
#define MAX 100001
using namespace std;

map<int, int> m;
vector<int> v;

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    if (m[x] == 0) v.push_back(x);
    m[x]++;
  }
  int cnt = 1;
  for (int i = 0; i < v.size(); i++) {
    while (m[cnt] != 0) cnt++;
    for (int j = 1; j <= m[v[i]]; j++) cout << cnt << ' ';
    cnt++;
  }
  cout << endl;
  return 0;
}
