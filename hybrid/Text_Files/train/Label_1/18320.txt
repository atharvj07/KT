#include<bits/stdc++.h>
using namespace std;
const int N = 1e6, LG = 17;
int pf[N + 1][21];
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    for (int i = 0; i < n; ++i) {
        for (int j = 0;; ++j) {
            int nx = a[i];
            while (nx % a[j] == 0) nx /= a[j], pf[i][j] |= (1 << j);
            if (nx == 1) break;
        }
    }
    int ans = 1;
    for (int i = 0; i < n; ++i) {
        int msk = pf[i][0];
        for (int j = 1; j < LG; ++j) msk |= pf[i][j];
        int cnt = 1;
        for (int j = i + 1; j < n; ++j) {
            if ((pf[j][0] | pf[j][1] | pf[j][2]) & msk) {
                cnt = 2;
                break;
            }
        }
        ans = max(ans, cnt);
    }
    cout << ans << "\n";
}
