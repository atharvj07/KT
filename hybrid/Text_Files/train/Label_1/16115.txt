#include <iostream>
#include <vector>
#include <array>
#include <algorithm>

using namespace std;

int main() {
    int n, m, k, s, a[200001], b[200001];
    cin >> n >> m >> k >> s;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];

    constexpr int inf = 1e9 + 10;
    vector<array<long long, 2>> c(m);
    vector<int> order(m), ans_day(m, -1);
    for (int i = 0; i < m; i++) {
        int t, w;
        cin >> t >> w;
        c[i] = {w, t};
        order[i] = i;
    }
    sort(order.begin(), order.end(),
         [&](auto& i, auto& j) { return c[i][0] < c[j][0]; });

    long long money = s;
    int ans = -1;
    for (auto& i : order) {
        int t = c[i][1];
        long long w = c[i][0];
        int day = -1;
        if (t == 1) {
            for (int i = 0; i < n; i++) {
                if (w * a[i] <= money) {
                    day = i + 1;
                    break;
                }
            }
        } else {
            for (int i = 0; i < n; i++) {
                if (w * b[i] <= money) {
                    day = i + 1;
                    break;
                }
            }
        }
        if (day == -1 || ans_day[i] != -1) continue;
        money -= w * ((t == 1) ? a[day - 1] : b[day - 1]);
        ans_day[i] = day;
        if (++ans == k - 1) {
            cout << day << '\n';
            for (int j = 0; j < k; j++) {
                int day_ = -1;
                for (int p = 0; p < m; p++) {
                    if (ans_day[p] != -1 &&
                        (day_ == -1 || ans_day[p] < day_)) {
                        day_ = ans_day[p];
                        i = p;
                    }
                }
                cout << (i + 1) << ' ' << day_ << '\n';
            }
            break;
        }
    }
    if (ans != k - 1) cout << -1 << '\n';
}
