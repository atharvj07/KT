def smallest_possible_starting_number(X_2):
    def is_prime(n):
        if n <= 1:
            return False
        for i in range(2, int(n**0.5) + 1):
            if n % i == 0:
                return False
        return True

    for X_0 in range(3, X_2 + 1):
        possible = False
        for X_1 in range(X_0, X_2 + 1):
            if X_2 % X_1 == 0:
                continue
            prime_found = False
            for p in range(2, X_1):
                if is_prime(p) and X_1 % p == 0:
                    prime_found = True
                    break
            if prime_found:
                for p in range(2, X_0):
                    if is_prime(p) and (X_1 == X_0 or X_1 % p == 0):
                        possible = True
                        break
            if possible:
                break
        if possible:
            return X_0
    return X_2

X_2 = int(input())
print(smallest_possible_starting_number(X_2))
