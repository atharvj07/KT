#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin>>s;
    stack<int> st;
    int ans[s.size()+1];
    int ptr=0;
    for(int i=0;i<s.size();i++)
    {
        if(s[i]=='r')
        {
            if(!st.empty())
            {
                ans[ptr]=st.top();
                ptr++;
                st.pop();
            }
            st.push(i+1);
        }
        else
        {
            st.push(i+1);
        }
    }
    while(!st.empty())
    {
        ans[ptr]=st.top();
        ptr++;
        st.pop();
    }
    for(int i=0;i<s.size();i++)
        cout<<ans[i]<<endl;
    return 0;
}
