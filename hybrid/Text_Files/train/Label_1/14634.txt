#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int buses = 0, sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i];
    if (sum > m) {
      buses++;
      sum = a[i];
    }
  }
  if (sum > 0) buses++;
  cout << buses << endl;
  return 0;
}
