from sys import stdin
input = stdin.readline
import math
n, k = map(int, input().split())
a = list(map(int, input().split()))
def gcd(a, b):
    if b == 0: return a
    return gcd(b, a % b)
def is_possible():
    for i in range(1, n+1):
        if i % gcd(i, n) * 1000 in a: return True
    return False
if not is_possible(): print(-1)
else:
    num = math.ceil(n * 1000 / gcd(n, 1000))
    den = 1000 // gcd(n, 1000)
    print(num // den)
