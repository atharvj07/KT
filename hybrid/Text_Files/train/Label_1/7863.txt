#include <bits/stdc++.h>
#define pb push_back
#define vi vector<int>
#define vl vector<long long>
#define pi pair<int, int>
#define pl pair<long long, long long>
#define all(x) (x).begin(), (x).end()
#define rall(x) (x).rbegin(), (x).rend()
#define fi first
#define se second
using namespace std;
const int mod = 1e9 + 7;
const int inf = 1e9;
int n;
vl x(2005), y(2005), c(2005), k(2005);
vl p(2005, -1);
vector<pi> edges;
long long dist(int a, int b) {
    return abs(x[a] - x[b]) + abs(y[a] - y[b]);
}
int find(int a) {
    if (p[a] < 0) return a;
    return p[a] = find(p[a]);
}
void join(int a, int b) {
    a = find(a);
    b = find(b);
    if (a != b) {
        if (p[a] < p[b]) swap(a, b);
        p[b] += p[a];
        p[a] = b;
    }
}
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> x[i] >> y[i];
    }
    for (int i = 1; i <= n; i++) cin >> c[i];
    for (int i = 1; i <= n; i++) cin >> k[i];
    long long ans = 0;
    for (int i = 1; i <= n; i++) {
        long long d = inf;
        for (int j = 1; j <= n; j++) {
            if (j == i) continue;
            d = min(d, k[i] + k[j] + dist(i, j));
        }
        ans += min(c[i], d);
    }
    cout << ans << '\n';
    int cnt = 0;
    vl power;
    for (int i = 1; i <= n; i++) {
        if (ans == c[i]) {
            cnt++;
            power.pb(i);
            ans -= c[i];
        }
    }
    cout << cnt << '\n';
    for (int u : power) cout << u << ' ';
    cout << '\n' << edges.size() << '\n';
    for (auto [u, v] : edges) cout << u << ' ' << v << '\n';
    return 0;
}
