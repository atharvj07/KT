#include <iostream>
#include <map>
#include <string>
#include <vector>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<map<char, int>> freq(n);
  vector<int> cnt(n);
  for (int i = 0; i < n; i++) {
    if (s[i] != '?') {
      freq[i][s[i]]++;
      cnt[i]++;
    }
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == '?') {
      for (auto &p : freq[i]) {
        cnt[i] *= p.second;
      }
    }
  }
  long long ans = 1;
  for (int i = 0; i < n; i++) {
    if (s[i] == '?') {
      ans *= 10 - cnt[i];
    }
  }
  cout << ans << endl;
  return 0;
}
