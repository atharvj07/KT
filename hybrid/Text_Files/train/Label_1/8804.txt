#include<bits/stdc++.h>
using namespace std;
const int maxn=100005;
int main(){
    int a[maxn],b[maxn],i,j,n,ans=0,mx=0;
    scanf("%d",&n);
    b[0]=0;
    for(i=1;i<=n;i++){
        scanf("%d",&a[i]);
        if(a[i]-a[i-1]==1)b[i]=b[i-1]+1;
        else b[i]=1;
    }
    for(i=1;i<=n;i++){
        if(b[i]>mx){
            mx=b[i];
            ans=a[i];
        }
    }
    cout<<mx<<endl;
    return 0;
}
