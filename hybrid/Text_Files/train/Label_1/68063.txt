from bisect import bisect_left

n=int(input())
a=list(map(int,input().split()))
a.sort()
l=len(a)
def check(k):
    lo,hi=-1e9,1e9
    for i in range(k):
        lo=max(lo,a[k-i-1]-k+i)
        hi=min(hi,a[k-i-1]+k-i)
    if hi<=lo:return True
    pos=bisect_left(a,hi+1)
    rem=k-pos
    if rem<=0:return True
    lo,hi=-1e9,1e9
    for i in range(k):
        lo=max(lo,a[rem-i-1]-rem+i)
        hi=min(hi,a[rem-i-1]+rem-i)
    if hi<=lo:return True
    return False

l,r=0,n-1
ans=n//2+1
idx=[]
while l<=r:
    m=(l+r)//2
    if check(m):
        ans=m
        idx=[]
        for i in range(m):
            idx.append(a[i])
        r=m-1
    else:l=m+1
idx.sort()
print(ans)
if ans:
    print(*idx)
