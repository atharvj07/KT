import sys

n,d=map(int,sys.stdin.readline().split())
a=[int(x) for x in sys.stdin.readline().split()]
chk,days=0,0
flag=0
for e in a:
    if e>0:chk+=e
    elif e<0:
        if chk+e<0:flag=1;break
        chk+=e
    else:
        if chk<0:days+=1
        if chk+d>=0:chk=0
if flag or chk<0:
    print(-1)
else:print(days)
