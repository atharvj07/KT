import java.util.*;
public class Main {
    static int[][] a, b;
    static int n, m;
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        n = input.nextInt();
        m = input.nextInt();
        a = new int[n][m];
        b = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = input.nextInt();
                b[i][j] = a[i][j];
            }
        }
        for (int i = 0; i < n; i++) {
            TreeSet<Integer> set = new TreeSet<>();
            for (int j = 0; j < m; j++) {
                set.add(a[i][j]);
            }
            int idx = 1;
            for (int j = 0; j < m; j++) {
                b[i][j] = idx;
                idx += set.headSet(a[i][j]).size() + 1;
            }
        }
        for (int j = 0; j < m; j++) {
            TreeSet<Integer> set = new TreeSet<>();
            for (int i = 0; i < n; i++) {
                set.add(a[i][j]);
            }
            int idx = 1;
            for (int i = 0; i < n; i++) {
                b[i][j] = idx;
                idx += set.headSet(a[i][j]).size() + 1;
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(b[i][j] + " ");
            }
            System.out.println();
        }
    }
}
