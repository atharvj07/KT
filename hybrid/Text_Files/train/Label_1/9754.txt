#include <bits/stdc++.h>
using namespace std;
int n, a[100], b[100], x[100], y[100];
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    sort(a, a + n);
    sort(b, b + n);
    for (int i = 0; i < n; i++) x[i] = a[i], y[i] = b[i];
    do {
      bool ok = true;
      for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++) {
          if (i == j) continue;
          if (x[i] + y[i] == x[j] + y[j]) ok = false;
        }
      if (ok) {
        for (int i = 0; i < n; i++) cout << x[i] << " ";
        cout << endl;
        for (int i = 0; i < n; i++) cout << y[i] << " ";
        cout << endl;
        break;
      }
    } while (next_permutation(x, x + n));
  }
}
