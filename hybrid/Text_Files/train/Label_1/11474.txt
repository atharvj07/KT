#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int t;
    cin >> t;
    while (t--) {
        int n, s;
        cin >> n >> s;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        int ans = 0, sum = 0;
        for (int i = 0; i < n; i++) {
            if (sum + a[i] <= s)
                sum += a[i];
            else
                break;
            ans = i + 1;
        }
        cout << ans << '\n';
    }
    return 0;
}
