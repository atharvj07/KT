#include <bits/stdc++.h>
using namespace std;
const int mod = 1000000007;
int n, m, pos[1010], dp[1010][1010];
int calc(int l, int r) {
  if (l == r) return 1;
  if (l > r) return 0;
  if (dp[l][r] != -1) return dp[l][r];
  int &ret = dp[l][r] = 0;
  if (pos[l] != pos[r]) {
    ret += calc(l, r - 1);
    ret %= mod;
    ret += calc(l + 1, r);
    ret %= mod;
  } else {
    int mid = (l + r) / 2;
    ret += calc(l, mid);
    ret %= mod;
    ret += calc(mid + 1, r);
    ret %= mod;
    ret += mod - 1;
    ret %= mod;
  }
  return ret;
}
int main() {
  cin >> n >> m;
  for (int i = 1; i <= m; i++) cin >> pos[i];
  memset(dp, -1, sizeof dp);
  cout << calc(1, m) << endl;
  return 0;
}
