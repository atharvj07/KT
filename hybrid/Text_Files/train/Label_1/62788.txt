import sys

def main():
    n, k = map(int, sys.stdin.readline().split())
    d = list(map(int, sys.stdin.readline().split()))

    if d[0] != 0:
        print(-1)
        return

    adj = [[] for _ in range(n+1)]
    vis = [False for _ in range(n+1)]

    for i in range(1, n+1):
        if d[i] == 0:
            continue

        for j in range(1, k+1):
            if i+j <= n and not vis[i+j] and d[i]+j == d[i+j]:
                adj[i].append(i+j)
                adj[i+j].append(i)
                vis[i+j] = True

    m = 0
    for i in range(1, n+1):
        for j in adj[i]:
            if i < j:
                m += 1
                print(i, j)

    if m == n-1:
        return
    else:
        print(-1)

main()
