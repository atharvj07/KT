#include <bits/stdc++.h>

using namespace std;

int minimumChanges(string s) {
    int n = s.size();
    int last[26];
    memset(last, -1, sizeof(last));

    int ans = 0;

    for (int i = 0; i < n; i++) {
        if (last[s[i] - 'a'] != -1) {
            ans = max(ans, last[s[i] - 'a'] + 1);
        }
        last[s[i] - 'a'] = i;
    }

    if (ans == n) return -1;

    return ans;
}

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;

    cout << minimumChanges(s) << endl;

    return 0;
}
