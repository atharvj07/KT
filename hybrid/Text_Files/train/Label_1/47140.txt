import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] arr = new int[n + 1][4];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < 4; j++) {
                arr[i][j] = sc.nextInt();
            }
        }
        int[] ans = new int[361];
        for (int i = 1; i <= n; i++) {
            int m = arr[i][0];
            int d = arr[i][1];
            int v = arr[i][2];
            int s = arr[i][3];
            for (int j = d; j < m * 30 + d + v; j++) {
                if (j > 360) {
                    j -= 360;
                }
                ans[j] = Math.max(ans[j], s - Math.min(d - j, m * 30 + d + v - j));
            }
        }
        int min = Integer.MAX_VALUE;
        for (int i = 0; i < 361; i++) {
            min = Math.min(min, ans[i]);
        }
        System.out.println(min);
    }
}
