import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        if (!input.hasNext()) {
            return;
        }

        int n = input.nextInt();
        int h = input.nextInt();
        int w = input.nextInt();

        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = input.nextInt();
        }

        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) {
                ans += w * h - x[i] * h;
            } else {
                ans += x[i] * h;
            }
        }
        ans += (n / 2) * h * w;
        System.out.println(ans);
    }
}
