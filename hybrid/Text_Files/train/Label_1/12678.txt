#include <bits/stdc++.h>
using namespace std;
#define ll long long

bool vis[110];
ll tele[110][2];

bool ok(ll l){
    if(l<=0) return false;
    if(vis[l]) return true;
    vis[l]=1;
    for(ll i=0;i<110;i++){
        if(tele[i][0]<=l && tele[i][1]>=l)
            if(ok(i)) return true;
    }
    return false;
}

int main(){
    ll n,m;
    cin>>n>>m;
    for(ll i=0;i<n;i++) cin>>tele[i][0]>>tele[i][1];
    cout<<(ok(m)?"YES":"NO")<<endl;
}
