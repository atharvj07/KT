import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = a[i - 1] + scanner.nextInt();
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i] * 2 >= a[n]) {
                ans = i;
                break;
            }
        }
        System.out.println(ans);
    }
}
