#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main(){
    int n,m,a,d;
    cin>>n>>m>>a>>d;
    vector<ll> c(m);
    for(auto &x : c)cin>>x;
    int cnt=0,p=c[0]/a,ans=0;
    for(int i=0;i<m;++i){
        ll r=c[i]/a;
        if(p==r)continue;
        if(p+1==r){
            if(d>=c[i]%a)++ans;
            else ++cnt;
        }
        else{
            ++ans;
            ++cnt;
        }
        p=r;
    }
    cout<<ans+cnt<<endl;
    return 0;
}
