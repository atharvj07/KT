#include <iostream>
#include <vector>

using namespace std;

int n, a[1000001], p[1000001], q[1000001], h;
vector<int> g[1000001];

int main() {
  cin >> h;
  for (int i = 0; i <= h; i++) cin >> a[i];
  int r = 1;
  while (p[r]) r++;
  for (int i = 2; i <= h; i++) p[a[i]]++, g[a[i]].push_back(i);
  bool found = 0;
  for (int i = 2; i <= h; i++) {
    if (a[i] != p[a[i]]) {
      if (found) {
        cout << "ambiguous\n";
        for (int i = 1; i <= h; i++) cout << p[i] << " ";
        cout << "\n";
        for (int i = 1; i <= h; i++) cout << p[i] - 1 << " ";
        cout << "\n";
        return 0;
      }
      found = 1;
      for (auto j : g[a[i]]) {
        q[j] = q[a[i]] + 1;
        if (q[j] > p[a[i]]) {
          cout << "ambiguous\n";
          for (int i = 1; i <= h; i++) cout << p[i] << " ";
          cout << "\n";
          for (int i = 1; i <= h; i++) cout << q[i] << " ";
          cout << "\n";
          return 0;
        }
      }
    }
  }
  cout << "perfect\n";
}
