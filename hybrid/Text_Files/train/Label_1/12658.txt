#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, a, b;
    cin >> n >> m >> a >> b;
    if (a * n != b * m) {
      cout << "NO\n";
      continue;
    }
    if (a > b) { swap(a, b); swap(n, m); }
    vector<vector<int>> ans(n, vector<int>(m, 0));
    int i = 0, j = 0, cnt = 0;
    while (cnt < a * n) {
      int f = 0;
      if ((j + 1 < m) && (ans[i][j + 1] == 0)) {
        ans[i][j++] = 1;
        cnt++;
      } else if ((i + 1 < n) && (ans[i + 1][j] == 0)) {
        ans[i++][j] = 1;
        cnt++;
      } else if ((j > 0) && (ans[i][j - 1] == 0)) {
        ans[i][j--] = 1;
        cnt++;
      } else if ((i > 0) && (ans[i - 1][j] == 0)) {
        ans[i--][j] = 1;
        cnt++;
      } else if ((i > 0) && (j > 0) && (ans[i - 1][j - 1] == 0)) {
        ans[i - 1][j - 1] = 1;
        cnt++;
      } else if ((i + 1 < n) && (j > 0) && (ans[i + 1][j - 1] == 0)) {
        ans[i + 1][j - 1] = 1;
        cnt++;
      } else if ((i + 1 < n) && (j + 1 < m) && (ans[i + 1][j + 1] == 0)) {
        ans[i + 1][j + 1] = 1;
        cnt++;
      } else if ((i > 0) && (j + 1 < m) && (ans[i - 1][j + 1] == 0)) {
        ans[i - 1][j + 1] = 1;
        cnt++;
      } else {
        f = 1;
      }
      if (f) break;
    }
    if (cnt == a * n) {
      cout << "YES\n";
      for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) cout << ans[i][j];
        cout << "\n";
      }
    } else
      cout << "NO\n";
  }
}
