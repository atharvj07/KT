#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<int> dp(n, 0);  // dp[i]: the length of the longest increasing subsequence ending with a[i]
  vector<int> pre(n, -1);  // pre[i]: the index of the previous element in the longest increasing subsequence ending with a[i]
  dp[0] = 1;
  for (int i = 1; i < n; i++) {
    for (int j = i - 1; j >= 0; j--) {
      if (a[i] > a[j] && dp[i] < dp[j] + 1) {
        dp[i] = dp[j] + 1;
        pre[i] = j;
      }
    }
  }
  int idx = max_element(dp.begin(), dp.end()) - dp.begin();
  cout << dp[idx] << endl;
  string s = "";
  while (idx != -1) {
    if (idx == 0)
      s = 'L' + s;
    else if (idx == n - 1)
      s = 'R' + s;
    else if (pre[idx] == idx - 1)
      s = 'L' + s;
    else
      s = 'R' + s;
    idx = pre[idx];
  }
  cout << s << endl;
  return 0;
}
