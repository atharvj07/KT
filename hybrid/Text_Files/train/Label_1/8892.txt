#include <bits/stdc++.h>
using namespace std;
const int mxN = 2e5 + 5;
int n, k, a[mxN], b[mxN];
int main() {
    cin >> n >> k;
    for (int i = 1; i <= k; i++) cin >> a[i];
    priority_queue<int> pq;
    bool fl = 1;
    for (int i = k; i >= 1; i--) {
        if (!pq.empty() && pq.top() < a[i]) fl = false;
        pq.push(a[i]);
    }
    if (!fl) {
        cout << -1 << endl;
        return 0;
    }
    int pos = k + 1;
    for (int i = 0; i < n; i++) {
        if (i == a[k]) continue;
        pq.push(i);
        b[pos++] = i;
    }
    for (int i = k - 1; i >= 1; i--) {
        while (a[i] != b[pos - 1]) {
            pos--;
            pq.pop();
        }
        pos--;
        pq.pop();
    }
    while (!pq.empty()) {
        b[pos++] = pq.top();
        pq.pop();
    }
    for (int i = 1; i <= n; i++) a[i] = b[i];
    for (int i = 1; i <= n; i++) cout << a[i] << " ";
    cout << endl;
    return 0;
}
