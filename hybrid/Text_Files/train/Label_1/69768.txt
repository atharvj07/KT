import sys
t= int(input())
def f(w, h, x1, y1, x2, y2, w1, h1):
    if(x2-x1>=w1 and y2-y1>=h1):
        return 0
    if(x1<=0 and x2>=w and y1<=0 and y2>=h):
        return -1
    if(h1>=h and ( (y2>=h and x2>=w and x1<=0) or (y1>=h and x1<=0 and x2>=w) or (y1>=h and y2<=h and x1<=0) or (y1>=h and y2<=h and x2>=w))):
        return -1
    if(w1>=w and ( (x2>=w and y2>=h and y1<=0) or (x1>=w and y1<=0 and y2>=h) or (x1>=w and x2<=w and y1<=0) or (x1>=w and x2<=w and y2>=h))):
        return -1
    if(y2>=h):
        return y2-h
    if(x2>=w):
        return x2-w
    if(y1<=0):
        return y1
    if(x1<=0):
        return x1
    return 0
for _ in range(t):
    w, h= map(int, input().split())
    x1, y1, x2, y2= map(int, input().split())
    w1, h1= map(int, input().split())
    print(f(w, h, x1, y1, x2, y2, w1, h1))
