#include <bits/stdc++.h>
using namespace std;
int w, p, arr[100009], m, b[100009], c[100009];
char a[100009];
int main() {
  scanf("%d", &w);
  for (int i = 1; i <= w; i++) { scanf(" %c", &a[i]); }
  if (a[1] == '-') {
    puts("NO");
    return 0;
  }
  arr[0] = m = -1;
  for (int i = 1; i <= w; i++) {
    if (a[i] == '+') {
      arr[++m] = ++p;
    } else {
      scanf(" -%d", &b[m]);
      c[m] = 1;
      for (int j = m; j >= 0; j--) {
        if (a[j] == '+' && b[m] < arr[j]) {
          c[j] = 0;
          c[m] = 1;
        }
      }
    }
    if (!c[m] || (m > 0 && c[m - 1] == 1)) {
      puts("NO");
      return 0;
    }
  }
  puts("YES");
  for (int i = m; i >= 0; i--) { printf("%d ", arr[i]); }
  return 0;
}
