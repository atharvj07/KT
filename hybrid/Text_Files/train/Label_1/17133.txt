#include <iostream>
#include <vector>

using namespace std;

int main() {
  vector<string> rows(6);
  for (int i = 0; i < 6; i++) {
    cin >> rows[i];
  }

  int mx = 0;
  int mx_i = -1, mx_j = -1;
  for (int i = 0; i < 6; i++) {
    for (int j = 0; j < 8; j += 4) {
      if (rows[i][j] == '.' && rows[i][j + 2] == '.' && (j == 0 || i == 5)) {
        int cur = 3 - (i % 3);
        if (cur > mx) {
          mx = cur;
          mx_i = i;
          mx_j = j;
        }
      }
      if (rows[i][j] == '.' && rows[i][j + 2] == '.') {
        int cur = 2;
        if (cur > mx) {
          mx = cur;
          mx_i = i;
          mx_j = j;
        }
      }
      if (rows[i][j] == '.' || rows[i][j + 2] == '.') {
        int cur = 1;
        if (cur > mx) {
          mx = cur;
          mx_i = i;
          mx_j = j;
        }
      }
    }
  }

  rows[mx_i][mx_j] = 'P';
  for (int i = 0; i < 6; i++) {
    cout << rows[i] << endl;
  }

  return 0;
}
