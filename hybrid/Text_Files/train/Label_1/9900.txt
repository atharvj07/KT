#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b;
    cin >> a >> b;
    string s;
    cin >> s;
    int n = s.size();
    int ans = 0;
    for (int i = 0; i < n;) {
      if (s[i] == '0') {
        i++;
        continue;
      }
      if (a <= b) {
        ans += a;
        i++;
        while (i < n && s[i] == '1') i++;
      } else {
        int j = i + 1;
        while (j < n && s[j] == '1') j++;
        if (j < n) {
          ans += b + min(a, b);
          i = j + 1;
        } else {
          ans += b;
          i = j;
        }
      }
    }
    cout << ans << endl;
  }
}
