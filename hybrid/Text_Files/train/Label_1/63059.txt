import sys
input = sys.stdin.readline

def solve():
    n, m = map(int, input().split())
    a = [list(map(int, input().split())) for _ in range(n)]
    b = [[0] * m for _ in range(n)]
    ans = []
    for i in range(n - 1):
        for j in range(m - 1):
            if a[i][j] != a[i + 1][j + 1]:
                b[i][j] = 1
                b[i][j + 1] = 1
                b[i + 1][j] = 1
                b[i + 1][j + 1] = 1
                ans.append([i + 1, j + 1])
    for i in range(n):
        for j in range(m):
            if a[i][j] != b[i][j]:
                return -1
    return ans

ans = solve()
if ans == -1:
    print(ans)
else:
    print(len(ans))
    for x, y in ans:
        print(x, y)
