#include<bits/stdc++.h>
using namespace std;
int main()
{
	int t;
	cin>>t;
	while(t--)
	{
		int n;
		cin>>n;
		string s;
		cin>>s;
		int l=0,r=n-1;
		while(l<=r)
		{
			if(s[l]!=s[r])break;
			l++;
			r--;
		}
		if(l>r)cout<<"-1 -1"<<endl;
		else
		{
			int a=l,b=l;
			while(a>=0 && s[a]==s[l])a--;
			a++;
			while(b<n && s[b]==s[l])b++;
			b--;
			cout<<a+1<<" "<<b+1<<endl;
		}
	}
	return 0;
}
