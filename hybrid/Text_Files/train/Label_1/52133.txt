import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (in.hasNext()) {
            int n = in.nextInt();
            boolean[] p = new boolean[n + 1];
            for (int i = 2; i <= n; i++)
                p[i] = true;
            int ans = 0;
            for (int i = 2; i <= n; i++) {
                if (p[i]) {
                    ans++;
                    for (int j = i + i; j <= n; j += i)
                        p[j] = false;
                }
            }
            System.out.println(ans);
        }
    }
}
