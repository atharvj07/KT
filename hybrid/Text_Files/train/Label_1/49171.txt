import java.util.Arrays;
import java.util.Scanner;

public class Main {

    private static int getMaxProfit(int n, int m, int[] l, int[] s, int[] c) {
        int[][] dp = new int[n + 1][m + 1];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(dp[i], -5001);
        }
        dp[0][0] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i - 1][Math.max(j, l[i - 1])] + c[l[i - 1]]);
            }
        }
        int maxProfit = 0;
        for (int i = 0; i <= m; i++) {
            maxProfit = Math.max(maxProfit, dp[n][i] - s[n - 1]);
        }
        return maxProfit;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] l = new int[n];
        int[] s = new int[n];
        int[] c = new int[m + 1];
        for (int i = 0; i < n; i++) {
            l[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            s[i] = in.nextInt();
        }
        for (int i = 1; i <= m; i++) {
            c[i] = in.nextInt();
        }
        System.out.println(getMaxProfit(n, m, l, s, c));
        in.close();
    }
}
