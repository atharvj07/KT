import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int ans = 0;
        while (n > 0) {
            if (n % 10 == 4 || n % 10 == 7) {
                ans++;
            }
            n /= 10;
        }
        System.out.println(ans);
    }
}
