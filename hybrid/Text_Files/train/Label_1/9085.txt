#include <iostream>
#include <unordered_map>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::unordered_map<int, int> likes;
  for (int i = 0; i < n; i++) {
    int id;
    std::cin >> id;
    likes[id]++;
  }
  int maxLikes = 0;
  int winner = -1;
  for (auto &entry : likes) {
    if (entry.second > maxLikes) {
      maxLikes = entry.second;
      winner = entry.first;
    }
  }
  std::cout << winner << std::endl;
  return 0;
}
