from collections import deque
import sys

si = sys.stdin.readline
n, m = map(int, si().split())
x, k, y = map(int, si().split())
a = list(map(int, si().split()))
b = list(map(int, si().split()))

ans = 0
a.reverse()
b.reverse()
q = deque()
for i in range(n):
    if len(q) >= 2 and q[-1] > q[-2]:
        while len(q) >= 2 and q[-1] > q[-2]:
            q.pop()
            ans += y
        if q[-1] < b[i]:
            ans += y
        q.append(b[i])
    else:
        q.append(b[i])
    if len(q) >= k and q[0] == a[i]:
        q.popleft()
        ans += x
if q == []:
    print(ans)
else:
    print(-1)
