#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> s(n);
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    cin >> s[i];
    for (int j = 0; j < s[i].size() - 1; j++)
      if (s[i][j] == 's' && s[i][j + 1] == 'h') cnt++;
  }
  int ans = cnt;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < s[i].size() - 1; j++) {
      if (s[i][j] == 's' && s[i][j + 1] == 'h') cnt--;
      else if (s[i][j] == 'h' && s[i][j + 1] == 's') cnt++;
      ans = max(ans, cnt);
    }
  }
  cout << ans << endl;
  return 0;
}
