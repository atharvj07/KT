#include <iostream>
#include <set>
#include <vector>
using namespace std;
int main() {
  ios::sync_with_stdio(false), cin.tie(nullptr);
  int n, x;
  cin >> n;
  set<int> s;
  vector<int> a(n, 0);
  for (auto& x : a) cin >> x;
  for (int i = n - 1; ~i; --i) a[i] |= a[i + 1];
  for (auto x : a) s.insert(x);
  cout << s.size();
}
