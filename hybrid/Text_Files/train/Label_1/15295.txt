#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll caff[200005];
int main(){
    ll n, m, sum = 0;
    cin>>n>>m;
    for(int i = 0; i<n; i++){
        cin>>caff[i];
        sum += caff[i];
    }
    if(sum < m)    cout<<-1<<endl;
    else{
        sort(caff, caff+n, greater<ll>());
        sum = caff[0];
        ll ans = 1, cur = 0;
        for(int i = 1; i<n; i++){
            ans++;
            sum = sum + caff[i] - ans + 1;
            if(sum >= m){
                cout<<ans<<endl;
                return 0;
            }
        }
        cout<<-1<<endl;
    }
    return 0;
}
