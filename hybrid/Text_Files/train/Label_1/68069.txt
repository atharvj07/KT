from collections import Counter

n, k = map(int, input().split())
a = []
for i in range(n):
    b = [int(x) for x in input().split('.')]
    a.append(b)
c = Counter()
for i in range(32):
    x = 0
    for j in a:
        x |= (1 << i & j[i // 8])
    c[x] += 1
if len(c) == k:
    print('.'.join([str(int(bin(i)[2:])) for i in sorted(c.keys())]))
else:
    print(-1)
