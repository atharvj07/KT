import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    String[] line = br.readLine().split(" ");
    int[] colors = new int[n];
    for (int i = 0; i < n; i++) {
      colors[i] = Integer.parseInt(line[i]);
    }
    List<List<Integer>> adjList = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      adjList.add(new ArrayList<>());
    }
    for (int i = 0; i < n - 1; i++) {
      line = br.readLine().split(" ");
      int u = Integer.parseInt(line[0]) - 1;
      int v = Integer.parseInt(line[1]) - 1;
      adjList.get(u).add(v);
      adjList.get(v).add(u);
    }
    int[] res = new int[n];
    dfs(0, -1, colors, adjList, res);
    for (int i = 0; i < n; i++) {
      System.out.print(res[i] + " ");
    }
  }

  private static void dfs(int u, int p, int[] colors, List<List<Integer>> adjList, int[] res) {
    int white = 0;
    int black = 0;
    if (colors[u] == 0) {
      black++;
    } else {
      white++;
    }
    for (int v : adjList.get(u)) {
      if (v != p) {
        dfs(v, u, colors, adjList, res);
        white += Math.max(res[v], 0);
        black -= Math.min(res[v], 0);
      }
    }
    res[u] = white - black;
  }
}
