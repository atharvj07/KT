#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define pb push_back

int n, a, b;
string s;
int dp[5005];
int main() {
  cin >> n >> a >> b >> s;
  for (int i = 1; i <= n; i++) {
    dp[i] = a;
    for (int j = 1; j < i; j++) {
      if (s.substr(i - j, j) == s.substr(i - 1, j)) dp[i] = min(dp[i], dp[i - j] + b);
    }
  }
  cout << dp[n] << endl;
  return 0;
}
