def UnionFind(n):
    par = list(range(n))
    rank = [0] * n
    def findpar(x):
        if par[x] == x:
            return x
        return findpar(par[x])
    def unite(a, b):
        a = findpar(a)
        b = findpar(b)
        if a != b:
            if rank[a] < rank[b]:
                par[a] = b
            else:
                par[b] = a
                if rank[a] == rank[b]:
                    rank[a] += 1
    return unite, findpar

n, m = map(int, input().split())
cities = [[] for i in range(n+1)]
for i in range(m):
    a, b = map(int, input().split())
    cities[a].append(b)
    cities[b].append(a)
ans = 0
uf = UnionFind(n+1)
for i in range(1, n+1):
    for j in cities[i]:
        if uf.findpar(i) != uf.findpar(j):
            uf.unite(i, j)
            ans += 1
print(ans)
