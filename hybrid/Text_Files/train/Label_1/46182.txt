import java.util.*;
import java.lang.*;
import java.io.*;

public class Main {

	static class Pair implements Comparable<Pair> {
		long a, b, k;
		double rate;

		Pair(long a, long b, long k) {
			this.a = a;
			this.b = b;
			this.k = k;
			rate = (double)a / b;
		}

		public int compareTo(Pair o) {
			return Double.compare(rate, o.rate);
		}
	}

	public static void main(String[] args) throws IOException {
		BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(in.readLine());
		Pair[] offers = new Pair[n];
		for (int i = 0; i < n; i++) {
			String[] s = in.readLine().split(" ");
			offers[i] = new Pair(Long.parseLong(s[0]), Long.parseLong(s[1]), Long.parseLong(s[2]));
		}

		Arrays.sort(offers);
		long money = 0;
		for (Pair offer : offers) {
			if (money >= offer.b * offer.k) {
				money += offer.a;
			} else {
				long months = (offer.b * offer.k - money + offer.b - 1) / offer.b;
				money += offer.a - months * offer.b;
				if (money >= offer.b) {
					money += offer.a;
				}
			}
		}

		System.out.println(money);
	}
}
