#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> A(n, vector<int>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      cin >> A[i][j];
    }
  }
  vector<bool> good(n, true);
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (A[i][j] != 0) {
        good[i] = false;
        good[j] = false;
      }
    }
  }
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (good[i]) {
      cnt++;
    }
  }
  cout << cnt << endl;
  for (int i = 0; i < n; i++) {
    if (good[i]) {
      cout << i + 1 << " ";
    }
  }
  cout << endl;
  return 0;
}
