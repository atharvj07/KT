s = input()
p = input()
n, m = len(s), len(p)
ans = [0] * (n + 1)
L = [0] * m
for i in range(n - 1, -1, -1):
    L[len(p) - 1] = i if s[i] == p[len(p) - 1] else L[len(p) - 1]
    for j in range(len(p) - 2, -1, -1):
        L[j] = L[j + 1] if s[i] == p[j] else i
    ans[i] = max(ans[i + 1], ans[L[0] + 1] + 1) if L[0] != -1 else ans[i + 1]
print(*ans)
