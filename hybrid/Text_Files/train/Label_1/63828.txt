import sys

n, m = map(int, input().split())
J = [list(input().split()) for _ in range(n)]
C = [int(input()) for _ in range(m)]
J = [(i[0], int(i[1])) for i in J]
J.sort(key=lambda x: (x[1], x[0]))

ans, j = 0, 0
C.sort(reverse=True)
for c in C:
    while j < n and c < J[j][1]:
        j += 1
    if j == n:
        break
    if J[j][0] == "ATK":
        ans += max(0, c - J[j][1])
    j += 1
print(ans)
