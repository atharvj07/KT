#include <iostream>
#include <vector>
using namespace std;
int main() {
  int T;
  cin >> T;
  for (int t = 0; t < T; t++) {
    int n, bs;
    cin >> n >> bs;
    vector<int> ps;
    string s;
    cin >> s;
    int c = 0, ans = -1;
    for (char ch : s) {
      c += ch - '0';
      ps.push_back(c);
    }
    for (int i = 0, j = 0; i < n; i++) {
      while (j < n && ps[j] - (i ? ps[i - 1] : 0) < bs) { j++; }
      int len = j + 1;
      if (i) len -= i;
      if (ps[j] - (i ? ps[i - 1] : 0) == bs) {
        ans = 1LL * len * (1LL * T + t + 1) * n + ans;
      }
    }
    if (ans == -1)
      cout << ans << endl;
    else if (ans >= int(2e9))
      cout << "-1\n";
    else
      cout << ans << endl;
  }
}
