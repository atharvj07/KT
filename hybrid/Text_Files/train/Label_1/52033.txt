import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String input = scanner.nextLine();
        String output = input;
        if (input.charAt(input.length() - 1) == 's') {
            output += "es";
        } else {
            output += "s";
        }
        System.out.println(output);
    }
}
