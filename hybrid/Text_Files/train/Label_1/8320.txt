#include <iostream>
#include <cmath>
using namespace std;
int main() {
    int r, d, n;
    cin >> r >> d, cin >> n;
    int ans = 0;
    while (n--) {
        int x, y, rr;
        cin >> x >> y >> rr;
        if (sqrt(x * x + y * y) + rr <= r - d || sqrt(x * x + y * y) - rr >= r)
            ans++;
    }
    cout << ans << endl;
    return 0;
}
