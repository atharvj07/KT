#include <bits/stdc++.h>

const int oo = 1e9;

using namespace std;

int n, m;
char a[2005][2005];
int lf[2005][2005], rg[2005][2005], up[2005][2005], dw[2005][2005];

int32_t main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
            if (a[i][j] == 'w') {
                lf[i][j] = j;
                rg[i][j] = j;
                up[i][j] = i;
                dw[i][j] = i;
            } else {
                lf[i][j] = oo;
                rg[i][j] = -1;
                up[i][j] = oo;
                dw[i][j] = -1;
            }
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 1; j < m; j++) {
            lf[i][j] = min(lf[i][j], lf[i][j - 1]);
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = m - 2; j >= 0; j--) {
            rg[i][j] = max(rg[i][j], rg[i][j + 1]);
        }
    }
    for (int j = 0; j < m; j++) {
        for (int i = 1; i < n; i++) {
            up[i][j] = min(up[i][j], up[i - 1][j]);
        }
    }
    for (int j = 0; j < m; j++) {
        for (int i = n - 2; i >= 0; i--) {
            dw[i][j] = max(dw[i][j], dw[i + 1][j]);
        }
    }
    int res = oo;
    int lx, rx, uy, dy;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (a[i][j] == 'w') {
                int nw = i;
                while (nw > 0 && a[nw - 1][j] == 'w') {
                    nw--;
                }
                int ed = i;
                while (ed + 1 < n && a[ed + 1][j] == 'w') {
                    ed++;
                }
                int ww = j;
                while (ww > 0 && a[i][ww - 1] == 'w') {
                    ww--;
                }
                int ne = j;
                while (ne + 1 < m && a[i][ne + 1] == 'w') {
                    ne++;
                }
                int cur = max(ne - ww, ed - nw) + 1;
                if (cur < res) {
                    res = cur;
                    lx = ww;
                    rx = ne;
                    uy = nw;
                    dy = ed;
                }
            }
        }
    }
    if (res == oo) {
        cout << "-1";
        return 0;
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (lf[i][j] >= lx && rg[i][j] <= rx && up[i][j] >= uy && dw[i][j] <= dy) {
                cout << '+';
            } else if (a[i][j] == 'w') {
                cout << 'w';
            } else {
                cout << '.';
            }
        }
        cout << '\n';
    }
    return 0;
}
