import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int t = in.nextInt();
        String s = in.next();
        char[] a = s.toCharArray();
        for (int i = 0; i < t; i++) {
            for (int j = 0; j < n - 1; j++) {
                if (a[j] == 'B' && a[j + 1] == 'G') {
                    a[j] = 'G';
                    a[j + 1] = 'B';
                    j++;
                }
            }
        }
        System.out.println(new String(a));
    }
}
