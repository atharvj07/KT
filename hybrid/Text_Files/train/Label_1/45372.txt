import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        input.nextLine();
        Set<String> titles = new HashSet<>();
        for (int i = 0; i < n; i++) {
            titles.add(input.nextLine());
        }
        StringBuilder title = new StringBuilder();
        for (int i = 1; ; i++) {
            String s = Character.toString((char) ('a' + i - 1));
            if (!titles.contains(s)) {
                title = new StringBuilder(s);
                break;
            }
            for (int j = i; ; j++) {
                String t = s + Character.toString((char) ('a' + j - 1));
                if (!titles.contains(t)) {
                    title = new StringBuilder(t);
                    break;
                }
                for (int k = j; ; k++) {
                    String u = t + Character.toString((char) ('a' + k - 1));
                    if (!titles.contains(u)) {
                        title = new StringBuilder(u);
                        break;
                    }
                    for (int l = k; ; l++) {
                        String v = u + Character.toString((char) ('a' + l - 1));
                        if (!titles.contains(v)) {
                            title = new StringBuilder(v);
                            break;
                        }
                        for (int m = l; ; m++) {
                            String w = v + Character.toString((char) ('a' + m - 1));
                            if (!titles.contains(w)) {
                                title = new StringBuilder(w);
                                break;
                            }
                            for (int o = m; ; o++) {
                                String x = w + Character.toString((char) ('a' + o - 1));
                                if (!titles.contains(x)) {
                                    title = new StringBuilder(x);
                                    break;
                                }
                                for (int p = o; ; p++) {
                                    String y = x + Character.toString((char) ('a' + p - 1));
                                    if (!titles.contains(y)) {
                                        title = new StringBuilder(y);
                                        break;
                                    }
                                    for (int q = p; ; q++) {
                                        String z = y + Character.toString((char) ('a' + q - 1));
                                        if (!titles.contains(z)) {
                                            title = new StringBuilder(z);
                                            break;
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
        System.out.println(title);
    }
}
