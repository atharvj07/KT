#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    int freq[n + 1];
    for (int i = 0; i <= n; i++) freq[i] = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      freq[a[i]]++;
    }
    int x = 0;
    for (int i = 1; i <= n; i++) {
      if (freq[i] >= 2) {
        x = i;
        break;
      }
    }
    if (x == 0) {
      cout << "0" << endl;
      continue;
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] == x) continue;
      ans++;
      while (i < n && a[i] != x) i++;
    }
    cout << ans << endl;
  }
  return 0;
}
