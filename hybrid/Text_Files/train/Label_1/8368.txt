#include <iostream>
using namespace std;

int main() {
  int n, a, b, k;
  cin >> n >> a >> b >> k;
  int ans = 0;
  long long rem = 0;
  int arr[n];
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
    rem += arr[i];
  }
  rem -= n * a;
  while (rem > 0 && k > 0) {
    ans++;
    rem -= b;
    k--;
  }
  if (rem <= 0)
    ans++;
  cout << ans << endl;
  return 0;
}
