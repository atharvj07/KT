#include <algorithm>
#include <vector>
#include <iostream>

using namespace std;

int main() {
    unsigned int n, f;
    cin >> n >> f;
    vector<pair<int, int>> clients;
    for (unsigned int i = 0; i < n; ++i) {
        int p, c;
        cin >> p >> c;
        clients.emplace_back(p, c);
    }
    sort(clients.begin(), clients.end(), [](pair<int, int>& a, pair<int, int>& b) {
        return a.first * b.second < a.second * b.first;
    });
    int answer = 0;
    for (unsigned int i = 0; i < n; ++i) {
        if (f > 0)
            clients[i].second *= 2, --f;
        answer += min(clients[i].first, clients[i].second);
    }
    cout << answer << '\n';
}
