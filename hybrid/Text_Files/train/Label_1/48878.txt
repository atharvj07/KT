import java.util.*;

public class Main {

    static int n, m, s, t, ans;
    static ArrayList<Integer>[] adj;
    static boolean[] vis;
    static int[] dis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        s = sc.nextInt() - 1;
        t = sc.nextInt() - 1;
        adj = new ArrayList[n];
        vis = new boolean[n];
        dis = new int[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            adj[u].add(v);
            adj[v].add(u);
        }
        Queue<Integer> q = new LinkedList<>();
        q.add(s);
        vis[s] = true;
        dis[s] = 0;
        ans = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : adj[u]) {
                if (vis[v]) continue;
                if (dis[u] + 1 < dis[v]) ans++;
                dis[v] = dis[u] + 1;
                q.add(v);
                vis[v] = true;
            }
        }
        System.out.println(ans / 2);
    }
}
