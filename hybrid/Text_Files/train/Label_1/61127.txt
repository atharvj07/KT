def solve(n, k):
    if n == 1:
        return 1
    if k & 1:
        return -1
    k //= 2
    if k > n * (n - 1) // 2:
        return -1
    segs = [0] * (n + 1)
    for i in range(1, n + 1):
        segs[i] = (k + i - 1) // i
    k = (k - 1) % segs[n] + 1
    ret = [1] * n
    for i in range(n - 1, 0, -1):
        if segs[i] < k:
            ret[i] = ret[i - 1] + (k - segs[i]) // segs[i - 1]
            k %= segs[i - 1]
    return ' '.join(map(str, ret))

for _ in range(int(input())):
    n, k = map(int, input().split())
    print(solve(n, k))
