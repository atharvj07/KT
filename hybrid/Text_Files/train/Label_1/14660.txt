#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    string arr = "";
    for (int i = 1; i <= min(min(n, m), 2); i++) {
        if (i & 1) {
            arr += 'G';
            m--;
        } else {
            arr += 'B';
            n--;
        }
    }
    while (n > 0) {
        arr += 'B';
        n--;
    }
    while (m > 0) {
        arr += 'G';
        m--;
    }
    cout << arr << endl;
    return 0;
}
