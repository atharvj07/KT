n,m=map(int,input().split());a=[input()for i in range(n)];res=[]
for x in range(n):
 for y in range(m):
  if a[x][y]=='*':
   for s in range(min(n,m)//2):
    ok=True
    for j in range(1,s+1):
     if (0<=x-j<n and 0<=y-j<m and a[x-j][y-j]=='.' and 0<=x+j<n and 0<=y-j<m and a[x+j][y-j]=='.' and 0<=x-j<n and 0<=y+j<m and a[x-j][y+j]=='.' and 0<=x+j<n and 0<=y+j<m and a[x+j][y+j]=='.'):continue
     else:ok=False;break
    if ok:res.append([x+1,y+1,s]);[a[x-j][y-j],a[x+j][y-j],a[x-j][y+j],a[x+j][y+j]]=['*']*4;break
print(len(res))
for x in res:print(*x)
