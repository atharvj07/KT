import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[] p = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < m; i++) {
            String action = sc.next();
            int team = sc.nextInt();
            if (action.equals("p")) {
                p[team - 1]++;
            } else {
                b[team - 1]++;
            }
        }
        int[] t1 = new int[n];
        int[] t2 = new int[n];
        for (int i = 0; i < n; i++) {
            if (p[0] > p[1]) {
                t1[i] = s[i];
                p[0]--;
            } else if (p[0] < p[1]) {
                t2[i] = s[i];
                p[1]--;
            } else {
                if (s[i] > s[n - i - 1]) {
                    t1[i] = s[i];
                    t2[n - i - 1] = s[n - i - 1];
                } else {
                    t1[n - i - 1] = s[i];
                    t2[i] = s[n - i - 1];
                }
                p[0]--;
                p[1]--;
            }
        }
        int sum1 = 0;
        int sum2 = 0;
        for (int i = 0; i < n; i++) {
            if (t1[i] > 0) {
                sum1 += t1[i];
            }
            if (t2[i] > 0) {
                sum2 += t2[i];
            }
        }
        System.out.println(sum1 - sum2);
    }
}
