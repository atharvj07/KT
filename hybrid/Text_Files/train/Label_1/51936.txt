import java.util.*;

class Main {
    static int n;
    static int k;
    static char[] arr;
    static int[] dp;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        k = scanner.nextInt();
        String str = scanner.next();
        arr = str.toCharArray();
        dp = new int[n];
        Arrays.fill(dp, Integer.MAX_VALUE);
        int ans = dfs(0);
        System.out.println(ans == Integer.MAX_VALUE ? -1 : ans);
    }

    static int dfs(int pos) {
        if (pos == k) {
            int ans = 0;
            for (int i = 0; i < k; i++) {
                ans += (arr[i] - 'a' + 1);
            }
            return ans;
        }
        if (pos > k || dp[pos] != Integer.MAX_VALUE) {
            return dp[pos];
        }
        int ans = Integer.MAX_VALUE;
        for (int i = pos; i < n; i++) {
            if (i != pos && (arr[pos] + 2) < arr[i]) {
                ans = Math.min(ans, dfs(pos + 1));
            }
            if (arr[pos] + 1 == arr[i]) {
                ans = Math.min(ans, dfs(i + 1));
            }
        }
        return dp[pos] = ans;
    }
}
