n=int(input())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
aa=[abs(x) for x in a]
bb=[abs(x) for x in b]
aa.sort(reverse=True)
bb.sort(reverse=True)
s=0
for i in range(n):
    if aa[i]==0 or bb[i]==0:
        s+=1
        continue
    if aa[i]>=bb[i]:
        s+=1
        aa[i]-=bb[i]
    else:
        bb[i]-=aa[i]
        aa[i]=0
print(s)
