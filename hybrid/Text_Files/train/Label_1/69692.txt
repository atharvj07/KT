import sys
n, m = map(int, input().split())
know = [set() for _ in range(n)]
for i in range(n):
    know[i].update(map(int, input().split()[1:]))
dp = [min(m, len(know[i])) for i in range(n)]
for i in range(n):
    for j in know[i]:
        for k in range(n):
            if i != k and j not in know[k]:
                dp[k] = min(dp[k], dp[i] + 1)
print(sum(dp))
