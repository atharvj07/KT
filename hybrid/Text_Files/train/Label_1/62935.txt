import sys
import os
from collections import defaultdict

fin = sys.stdin
fout = sys.stdout

for line in fin:
    n, m = map(int, line.split())
    break

graph = defaultdict(int)

for line in fin:
    u, v, d = map(int, line.split())
    graph[(u, v)] += d
    graph[(v, u)] += -d

tmp = []
for u in graph.keys():
    tmp.append([u[0], u[1], graph[u]])

tmp.sort(key=lambda x: -x[2])

small = 0
for i in range(m):
    u, v, d = tmp[i]
    if d <= 0:
        break
    small += d

ans = []

for i in range(m):
    u, v, d = tmp[i]
    if d <= 0:
        break
    ans.append([u, v, d])
    graph[(u, v)] = 0
    graph[(v, u)] = 0

for u in graph.keys():
    if graph[u] != 0:
        ans.append([u[0], u[1], -graph[u]])

fout.write(f'{len(ans)}\n')
for edge in ans:
    fout.write(f'{edge[0]} {edge[1]} {edge[2]}\n')
