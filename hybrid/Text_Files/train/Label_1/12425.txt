#include <bits/stdc++.h>
using namespace std;
const int dx[] = {-1, 0, 1, 0}, dy[] = {0, 1, 0, -1};
int n, m, k, stx, sty, tx, ty, cnt, ans;
bool vis[505][505], used[505][505];
char a[505][505];
bool valid(int x, int y) { return x >= 1 && y >= 1 && x <= n && y <= m && a[x][y] == '.'; }
void dfs(int x, int y) {
  if (!valid(x, y)) return;
  if (vis[x][y]) return;
  vis[x][y] = true;
  for (int i = 0; i < 4; i++) dfs(x + dx[i], y + dy[i]);
}
void find(int x, int y) {
  if (!valid(x, y)) return;
  if (used[x][y]) return;
  used[x][y] = true;
  cnt++;
  for (int i = 0; i < 4; i++) find(x + dx[i], y + dy[i]);
}
int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL);
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) {
      cin >> a[i][j];
      if (a[i][j] == '.' && !vis[i][j]) {
        dfs(i, j);
        if (++ans == 2) stx = i, sty = j, tx = i, ty = j;
      }
    }
  cnt = 0;
  find(stx, sty);
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (!used[i][j]) {
        if (a[i][j] == '.') {
          a[i][j] = 'X';
          k--;
        } else
          a[i][j] = '#';
        if (!k) break;
      }
    }
    if (!k) break;
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) cout << a[i][j];
    cout << endl;
  }
}
