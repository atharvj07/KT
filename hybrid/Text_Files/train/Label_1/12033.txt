#include<bits/stdc++.h>
using namespace std;
#define i64 long long
#define input(x) cin >> x;
#define print(x) cout << x << endl;
#define what_is(x) cout << #x << " is " << x << endl;
#define mod 1000000007
#define MAX 1000001
int main(){
    int n, m;
    input(n);
    input(m);
    vector<map<int, int>> v(n + 1);
    for(int i = 1; i <= n; i++){
        int sz;
        input(sz);
        for(int j = 1; j <= sz; j++){
            int x; input(x);
            v[i][x]++;
        }
    }
    i64 ans = 1;
    for(int i = 1; i <= m; i++){
        map<int, int> mp;
        for(int j = 1; j <= n; j++){
            mp[v[j][i]]++;
        }
        ans = (ans * mp.size()) % mod;
    }
    print(ans);

    return 0;
}
