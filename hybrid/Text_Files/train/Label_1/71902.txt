n = int(input().strip())
a = list(map(int, input().strip().split()))
b = list(map(int, input().strip().split()))

position_diff_counts = {}

for i in range(n):
    current_diff = (a.index(b[i]) - i) % n
    if current_diff in position_diff_counts:
        position_diff_counts[current_diff] += 1
    else:
        position_diff_counts[current_diff] = 1

print(max(position_diff_counts.values()))
