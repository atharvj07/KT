#include<bits/stdc++.h>
using namespace std;
int m,ans;
map<string,int> mp;
int main()
{
    string s;
    cin>>m;
    while(m--)
    {
        cin>>s;
        mp[s]++;
    }
    map<string,int>::iterator it;
    for(it=mp.begin();it!=mp.end();it++)
    {
        int cnt=0;
        for(int i=0;i<6;i++)
        {
            for(char c='0';c<='9';c++)
            {
                if(c!=(it->first)[i])
                {
                    string ss=it->first;
                    ss[i]=c;
                    if(mp.find(ss)!=mp.end())
                        cnt++;
                }
            }
        }
        ans=max(ans,cnt);
    }
    cout<<ans<<endl;
    return 0;
}
