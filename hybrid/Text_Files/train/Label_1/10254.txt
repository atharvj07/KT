#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
#define endl "\n"

void run_case() {
    int n, k;
    cin >> n >> k;
    vector<ll> a(n);
    for (auto &x : a) { cin >> x; }
    sort(a.begin(), a.end(), greater<ll>());
    ll cur = 0;
    int ans = 0;
    for (auto &x : a) {
        if (cur + x > k) {
            ans++;
            cur -= ans;
        }
        cur += x;
    }
    cout << ans << endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin >> t;
    while (t--) run_case();
    return 0;
}
