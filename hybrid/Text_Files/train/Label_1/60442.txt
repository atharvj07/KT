def solve(n, m, comments):
    graph = [[] for _ in range(n)]
    for i, j, c in comments:
        i, j = i - 1, j - 1
        if c == 'imposter':
            graph[i].append((j, 1))
            graph[j].append((i, 1))
        else:
            graph[i].append((j, 0))
            graph[j].append((i, 0))
    
    color = [-1] * n
    contradictions = False
    
    def dfs(v, c):
        nonlocal contradictions
        if color[v] != -1:
            return color[v] == c
        color[v] = c
        for u, is_imposter in graph[v]:
            if not dfs(u, c ^ is_imposter):
                contradictions = True
                return False
        return True
    
    for i in range(n):
        if color[i] == -1 and not dfs(i, 0):
            contradictions = True
            break
    
    if contradictions:
        return -1
    
    answer = 0
    for c in color:
        if c == 1:
            answer += 1
    return max(answer, n - answer)

t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    comments = [input().split() for _ in range(m)]
    for i in range(m):
        comments[i][0], comments[i][1] = int(comments[i][0]), int(comments[i][1])
    print(solve(n, m, comments))
