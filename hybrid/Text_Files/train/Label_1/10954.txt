#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, q;
    string s;
    cin >> n >> s >> q;
    vector<vector<int>> cnt(26, vector<int>(n + 1));
    for (int i = 0; i < n; i++) cnt[s[i] - 'a'][i + 1]++;
    for (int i = 0; i < 26; i++)
        for (int j = 1; j <= n; j++) cnt[i][j] += cnt[i][j - 1];
    while (q--) {
        int m;
        char c;
        cin >> m >> c;
        int l = 0, r = n;
        while (l < r) {
            int mid = (l + r + 1) >> 1;
            int num = cnt[c - 'a'][mid];
            if (mid - num <= m) l = mid;
            else r = mid - 1;
        }
        cout << l << "\n";
    }
}
