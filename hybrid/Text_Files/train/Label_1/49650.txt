import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    long[] ink = new long[n];
    for (int i = 0; i < n; i++) {
      ink[i] = scanner.nextLong();
    }
    Arrays.sort(ink);
    long days = 0;
    for (long i : ink) {
      days += (i + 6) / 7;
    }
    System.out.println(ink.length - (int) (days % ink.length));
  }
}
