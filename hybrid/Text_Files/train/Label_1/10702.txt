#include<bits/stdc++.h>
using namespace std;
int main(){
  int n,Sereja=0,Dima=0;
  cin>>n;
  vector<int>v(n);
  for(int i=0;i<n;i++) cin>>v[i];
  int l=0,r=n-1;
  while(l<=r){
    if(v[l]>v[r]){
      if(Sereja%2){ Dima+=v[l]; l++;}
      else { Sereja+=v[l]; l++;}
    }
    else{
      if(Sereja%2){ Dima+=v[r]; r--;}
      else { Sereja+=v[r]; r--;}
    }
  }
  cout<<Sereja<<" "<<Dima;
  return 0;
}
