#include <iostream>
#include <cstdio>
#include <vector>

int count = 0, len1, len2, i, j, isin;
std::string str1, str2;
std::vector<int> prefix;

void prefix_function(std::string p) {
    prefix.resize(p.length());
    for (i = 1; i < p.length(); i++) {
        j = prefix[i - 1];
        while (j > 0 && p[i] != p[j]) j = prefix[j - 1];
        if (p[i] == p[j]) j++;
        prefix[i] = j;
    }
}

int main() {
    std::cin >> str1 >> str2;
    len1 = str1.length();
    len2 = str2.length();
    isin = 0;
    prefix_function(str2);
    for (i = 0; i < len1; i++) {
        while (j > 0 && str1[i] != str2[j]) j = prefix[j - 1];
        if (str1[i] == str2[j]) j++, count++;
        if (j == len2) count++, j = prefix[j - 1];
        if (count >= 1) {
            isin = 1;
            break;
        }
    }
    if (!isin) printf("0\n");
    else printf("%d\n", count);
    return 0;
}
