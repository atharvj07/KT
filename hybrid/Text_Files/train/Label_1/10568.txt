#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> points(n);
  for (pair<int, int>& p : points) { cin >> p.first >> p.second; }
  if (n == 1) {
    cout << -1 << endl;
    return 0;
  }
  int min_x = points[0].first, max_x = points[0].first, min_y = points[0].second,
      max_y = points[0].second;
  for (int i = 1; i < n; i++) {
    min_x = min(min_x, points[i].first);
    max_x = max(max_x, points[i].first);
    min_y = min(min_y, points[i].second);
    max_y = max(max_y, points[i].second);
  }
  cout << (max_x - min_x) * (max_y - min_y) << endl;
  return 0;
}
