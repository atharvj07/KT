#include <iostream>
#include <map>
#include <vector>
using namespace std;

int main() {
  string s;
  int k;
  cin >> s >> k;
  map<char, int> mp;
  vector<bool> taken(26, false);
  for (auto &c : s) {
    mp[c]++;
    taken[c - 'a'] = true;
  }
  int cnt = 0;
  for (int i = 0; i < 26; i++) {
    if (!taken[i] || mp[i + 'a'] >= 2) {
      cnt++;
    } else {
      cnt += mp[i + 'a'] - 1;
    }
  }
  if (cnt <= k) {
    cout << 0 << endl;
  } else {
    cout << "impossible" << endl;
  }
}
