#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, l, r, k;
        cin >> n >> l >> r >> k;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (l <= a[i] && a[i] <= r && a[i] <= k) {
                k -= a[i];
                ans++;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
