from collections import defaultdict
n=int(input())
a=list(map(int,input().split()))
d=defaultdict(lambda:[])
for i in range(n):
    d[a[i]].append(i)
res=[]
ans=[]
for i in d:
    ans.append(i)
    if len(d[i])==1:
        res.append([i,0])
        continue
    l=d[i]
    ll=len(l)
    dv=l[1]-l[0]
    flag=True
    for j in range(1,ll):
        if l[j]-l[j-1]!=dv:
            flag=False
            break
    if flag:
        res.append([i,dv])
res.sort()
print(len(res))
for i in res:
    print(i[0],i[1])
