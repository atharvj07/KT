import sys
input = sys.stdin.readline
n,a,b = map(int,input().split())
x = []
for i in range(n):
    xi,Vxi,Vyi = map(int,input().split())
    x.append([xi,(Vxi-Vyi*a)/Vyi])
x.sort(key = lambda x:x[0])
v = []
ans = 0
for i in range(n-1):
    if x[i+1][1] <= x[i][1]: continue
    x1 = x[i][0]
    x2 = (x[i][1]*a+b-x[i+1][0])/(x[i+1][1]-x[i][1])
    v.append([x1,x2])
for i in range(len(v)):
    x1,x2 = v[i]
    if i==0: ans+=x1*x1
    else: ans+=(x1-p)*x1
    p = x2
ans+=len(v)*len(v)
print(ans)
