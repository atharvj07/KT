import java.util.Scanner;

public class RedundantTVSet {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] l = new int[n];
        int[] r = new int[n];

        for (int i = 0; i < n; i++) {
            l[i] = scanner.nextInt();
            r[i] = scanner.nextInt();
        }

        int redundantIndex = -1;

        for (int i = 0; i < n; i++) {
            boolean isRedundant = true;

            for (int j = 0; j < n; j++) {
                if (i != j && l[i] <= r[j] && l[j] <= r[i]) {
                    isRedundant = false;
                    break;
                }
            }

            if (isRedundant) {
                redundantIndex = i + 1;
                break;
            }
        }

        System.out.println(redundantIndex);
    }
}
