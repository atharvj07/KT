import java.util.*;

public class Main {
    static int[] dp;
    static int n;
    static int w;
    static int[][] items;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        w = sc.nextInt();
        dp = new int[w + 1];
        items = new int[n][2];
        for (int i = 0; i < n; i++) {
            items[i][0] = sc.nextInt();
            items[i][1] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            for (int j = w; j >= items[i][1]; j--) {
                dp[j] = Math.max(dp[j], dp[j - items[i][1]] + items[i][0]);
            }
        }
        System.out.println(dp[w]);
    }
}
