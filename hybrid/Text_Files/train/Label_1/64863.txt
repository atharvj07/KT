import sys
input = sys.stdin.readline

n = int(input())
a = []
b = []
c = []
for _ in range(n):
    ai, bi, ci = map(int, input().split())
    a.append(ai)
    b.append(bi)
    c.append(ci)

adj = [[] for _ in range(n)]
for _ in range(n - 1):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

# dp[i][0/1] = minimum cost to make subtree rooted at i have all digits equal to c[i], with/without flipping digit at i
dp = [[-1] * 2 for _ in range(n)]

def dfs(u, p):
    if dp[u][b[u]] != -1:
        return dp[u][b[u]]
    dp[u][b[u]] = a[u]
    for v in adj[u]:
        if v == p:
            continue
        dp[u][b[u]] += dfs(v, u)

    if dp[u][1 - b[u]] != -1:
        return dp[u][1 - b[u]]
    dp[u][1 - b[u]] = 0
    for v in adj[u]:
        if v == p:
            continue
        dp[u][1 - b[u]] += min(dfs(v, u), dfs(v, u) + a[u])

    return dp[u][b[u]]

if b[0] != c[0]:
    print(-1)
else:
    print(dfs(0, -1))
