#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (int)(n); i++)

using namespace std;
using ll = long long;

int main() {
  int n;
  cin >> n;
  map<int, int> c;
  int x;
  rep(i, n) {
    cin >> x;
    c[x]++;
  }
  int a[n + 1], ans = -1;
  rep(i, n + 1) { a[i] = -1; }
  rep(i, n + 1) {
    if (a[i] != -1) continue;
    int j = i, t = 1;
    while (a[j] == -1) {
      a[j] = i;
      j = c[j];
      t++;
    }
    if (a[j] == i && t > 1) {
      if (ans == -1 || t < ans) { ans = t; }
    }
  }
  cout << ans << '\n';
  return 0;
}
