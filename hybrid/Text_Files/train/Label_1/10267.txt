#include<bits/stdc++.h>
using namespace std;
char c[105][105];
bool vis[105][105];
int dx[4]={0,1,0,-1};
int dy[4]={1,0,-1,0};
int n,cnt,sum;
bool valid(int x,int y)
{
    if(x>=1 && x<=n && y>=1 && y<=n)
    {
        if(vis[x][y]==0 && c[x][y]=='#')
        return 1;
    }
    return 0;
}
void dfs(int x,int y)
{
    if(!valid(x,y))
    return;
    vis[x][y]=1;
    cnt++;
    for(int i=0;i<4;i++)
    {
        int X=x+dx[i];
        int Y=y+dy[i];
        dfs(X,Y);
    }
}
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int x,y;
    cin>>n;
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        {
            cin>>c[i][j];
            if(c[i][j]=='#')
            {
                x=i;
                y=j;
                sum++;
            }
        }
    }
    if(sum%5!=0)
    {
        cout<<"NO"<<"\n";
        return 0;
    }
    dfs(x,y);
    if(cnt==5)
    cout<<"YES"<<"\n";
    else
    cout<<"NO"<<"\n";
    return 0;
}
