#include <iostream>
#include <vector>

int main(){
    long long n, p;
    std::cin >> n >> p;

    if(p >= 1 and p < n){
        std::cout << -1;
        return 0;
    }

    std::vector<long long> ans;

    if(p < 0){
        long long sum = 0;
        while(sum + p < n){
            ans.push_back(n - sum);
            sum += n - sum;
        }

        ans.push_back(p + (n - sum));
    } else {
        long long pow2 = 1;
        while(pow2 <= n){
            pow2 *= 2;
        }
        pow2 /= 2;

        while(pow2 >= 1){
            if(n >= pow2){
                n -= pow2;
                ans.push_back(pow2 + p);
            }
            pow2 /= 2;
        }
    }

    if(n > 0){
        std::cout << -1;
    } else {
        std::cout << ans.size();
    }
    
    return 0;
}
