import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            int[] arr = new int[k];
            for (int j = 0; j < k; j++) arr[j] = sc.nextInt();
            Arrays.sort(arr);
            ans[i] = arr[0] == arr[k - 1] ? 1 : 2;
        }
        for (int i = 0; i < n; i++) System.out.println(ans[i]);
    }
}
