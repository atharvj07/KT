def calc(a):
    sz = len(a)
    while sz > 1:
        b = []
        for i in range(0, sz, 2):
            if i + 1 == sz:
                b.append(a[i])
            else:
                b.append(a[i] | a[i + 1])
        a = b
        sz = len(a)
        b = []
        for i in range(0, sz, 2):
            if i + 1 == sz:
                b.append(a[i])
            else:
                b.append(a[i] ^ a[i + 1])
        a = b
        sz = len(a)
    return a[0]

n, m = map(int, input().split())
a = list(map(int, input().split()))
for i in range(m):
    p, b = map(int, input().split())
    p -= 1
    a[p] = b
    print(calc(a))
