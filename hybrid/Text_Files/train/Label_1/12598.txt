#include <iostream>

using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;

ll n, a, b, c, d, ans;

int main() {
    cin >> n >> a >> b >> c >> d;
    ans = (((a * b) % mod * c) % mod * d) % mod;
    n -= 4;
    c = d;
    while (n) {
        if (n & 1) ans = (((ans * c) % mod) * d) % mod;
        d = (((d * d) % mod) * c) % mod;
        c = d;
        n >>= 1;
    }
    cout << ans;
    return 0;
}
