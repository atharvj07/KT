import java.util.*;
class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    String s = sc.next();
    int k = sc.nextInt();
    int rem = Integer.parseInt(s) % (1 << k);
    for (int i = k; i < s.length(); i++) {
      rem = (rem * 10 + s.charAt(i) - '0') % (1 << k);
    }
    System.out.print((int)Math.ceil((1 << k) - rem));
  }
}
