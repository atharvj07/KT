#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'R') {
      ans++;
      int j = i + 1;
      while (j < n && s[j] == 'B') {
        j++;
      }
      if (j < n) {
        s[j] = 'R';
      }
    }
  }
  cout << ans << endl;
  return 0;
}
