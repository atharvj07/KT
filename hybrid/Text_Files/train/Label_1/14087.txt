#include <vector>
#include <utility>
#include <iostream>
#include <algorithm>

using namespace std;

typedef long long ll;

const ll INF = 1e18;

int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
    }

    vector<pair<ll, ll>> dp(n, {INF, -1});
    dp[0] = {a[0], -1};

    for (ll i = 1; i < n; i++) {
        ll val = dp[i - 1].first;
        for (ll j = i - 1; j >= 0; j--) {
            if (val < a[j]) {
                if (dp[j].second != -1) {
                    dp[i] = {a[i], dp[j].second};
                    break;
                }
            }
            val = dp[j].first;
        }
    }

    ll cur = -1;
    vector<ll> ans;
    for (ll i = n - 1; i >= 0; i--) {
        if (dp[i].second == cur) {
            ans.push_back(a[i]);
        }
        cur = i;
    }

    cout << ans.size() << endl;
    for (auto i : ans) {
        cout << i << " ";
    }
    cout << endl;

    return 0;
}
