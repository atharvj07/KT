#include <bits/stdc++.h>
using namespace std;
int32_t main() {
    int n, a, dp[300001], diff[300001];
    long long ans = 0, sum = 0;
    cin >> n >> a;
    for (int i = 1; i <= n; i++) {
        int p, q;
        cin >> p >> q;
        diff[i] = p - q;
        sum += diff[i];
        dp[i] = a * i - sum;
    }
    for (int i = 2; i <= n; i++) { dp[i] = max(dp[i], dp[i - 1]); }
    long long cur = dp[n];
    for (int i = n - 1; i > 0; i--) {
        cur += a;
        sum -= diff[i];
        cur -= sum;
        ans = max(ans, cur);
    }
    cout << ans << endl;
    return 0;
}
