#include <bits/stdc++.h>
using namespace std;
int n,m,q,k,a[500001],ans[500001];
map<int,int> p;
int main(){
	cin>>n>>m>>q;
	for(int i=1;i<=n;i++) cin>>a[i];
	for(int i=1;i<=q;i++){
		cin>>k;
		if(k<=n) { cout<<a[k]<<endl; continue; }
		int cnt=1e9;
		for(int i=1;i<=n;i++) if(p[a[i]]<cnt) cnt=p[a[i]];
		for(int i=1;i<=n;i++) if(p[a[i]]==cnt){
			ans[++cnt]=a[i];
			p[a[i]]=cnt;
		}
		cout<<ans[(k-n)%cnt+1]<<endl;
	}
	return 0;
}
