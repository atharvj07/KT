#include <bits/stdc++.h>
using namespace std;
const int N=55;
int n,m,a[N][N],ans;
map<int,set<int>> mp;
int main(){
  ios::sync_with_stdio(false);cin.tie(nullptr);
  cin>>n>>m;
  for(int i=1;i<=n;i++){
    for(int j=1;j<=m;j++){
      cin>>a[i][j];
    }
  }
  for(int i=1;i<=n;i++){
    for(int j=1;j<=m;j++){
      if(a[i][j]!=a[i][j-1]) mp[j].insert(i);
      if(a[i][j]!=a[i-1][j]) mp[i].insert(j);
    }
  }
  for(auto &t:mp){
    ans+=int(t.second.size())*(int(t.second.size())+1)/2;
  }
  cout<<ans<<endl;
}
