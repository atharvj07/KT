T = int(input())
for _ in range(T):
    n = int(input())
    prev_plays, prev_clears = 0, 0
    valid = True
    for i in range(n):
        plays, clears = map(int, input().split())
        if plays < prev_plays or clears < prev_clears or clears > plays or (clears - prev_clears) > (plays - prev_plays):
            valid = False
            break
        prev_plays, prev_clears = plays, clears
    if valid:
        print("YES")
    else:
        print("NO")
