#include<bits/stdc++.h>
using namespace std;
int main()
{
	int ans = 0, cnt = 0;
	string s;
	cin >> s;
	for(int i = 0; i < s.length(); i++) {
		if(i + 4 < s.length() && s[i] == 'h' && s[i + 1] == 'e' && s[i + 2] == 'a' && s[i + 3] == 'v' && s[i + 4] == 'y') {
			ans += cnt;
			cnt++;
		}
		else if(s[i] == 'm' && s[i + 1] == 'e' && s[i + 2] == 't' && s[i + 3] == 'a' && s[i + 4] == 'l')
			cnt++;
	}
	cout << ans + cnt << endl;
}
