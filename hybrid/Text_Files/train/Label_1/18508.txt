#include <iostream>
#include <sstream>
#include <string>
#include <vector>
using namespace std;
int main() {
  string input;
  getline(cin, input);
  vector<int> v1;
  stringstream ss(input);
  int num;
  while (ss >> num) { v1.push_back(num); }
  getline(cin, input);
  vector<int> v2;
  stringstream ss2(input);
  while (ss2 >> num) { v2.push_back(num); }
  int h1 = v1[0], m1 = v1[1], h2 = v2[0], m2 = v2[1];
  h1 += h2;
  m1 += m2;
  if (m1 >= 60) {
    h1++;
    m1 -= 60;
  }
  cout << h1 / 10 << h1 % 10 << ':' << m1 / 10 << m1 % 10 << '\n';
  return 0;
}
