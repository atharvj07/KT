#include<bits/stdc++.h>
#define ll long long int
#define lll long long long int
using namespace std;

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    ll n, m;
    cin >> n >> m;
    vector<pair<ll, ll>> seg(n, {0, 0});
    vector<ll> len(m);
    for (int i = 0; i < n; i++) {
        cin >> seg[i].first >> seg[i].second;
        seg[i].second++;
    }
    for (int i = 0; i < m; i++) cin >> len[i];
    sort(len.begin(), len.end());
    sort(seg.begin(), seg.end());
    for (int i = 0; i < n - 1; i++) {
        ll cur = seg[i].second - seg[i].first - 1;
        auto pos = lower_bound(len.begin(), len.end(), cur);
        if (pos == len.end() || *pos != cur) {
            cout << "No" << '\n';
            return 0;
        }
        len.erase(pos);
    }
    cout << "Yes" << '\n';
    for (int i = 0; i < n - 1; i++) {
        cout << lower_bound(len.begin(), len.end(), seg[i].second - seg[i].first - 1) - len.begin() + 1 << " ";
    }
    cout << '\n';
}
