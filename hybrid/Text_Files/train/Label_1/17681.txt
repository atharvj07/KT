#include <iostream>
#include <vector>
using namespace std;

vector<vector<char>> arr;
int x, y;

bool check(int i, int j, int dx, int dy) {
    while (i < x && j < y && arr[i][j] != '.') {
        i += dx;
        j += dy;
    }
    if (i >= x || j >= y) return false;
    return true;
}

int main() {
    int h, w;
    cin >> h >> w;
    x = h, y = w;
    arr.resize(x, vector<char>(y));
    for (int i = 0; i < x; i++) {
        for (int j = 0; j < y; j++) {
            cin >> arr[i][j];
            if (arr[i][j] == '*') {
                if (i == x / 2 && j == y / 2)
                    continue;
                else if (i == x / 2 || j == y / 2)
                    check(i, j, 0, 1) && check(i, j, 0, -1) && check(i, j, 1, 0) && check(i, j, -1, 0);
                else
                    return cout << "NO", 0;
            }
        }
    }
    cout << "YES";
}
