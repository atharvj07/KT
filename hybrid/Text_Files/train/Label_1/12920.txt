#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 1010;

int main() {
  int s, x1, x2, t1, t2, p, d;
  cin >> s >> x1 >> x2 >> t1 >> t2 >> p >> d;
  if (x1 > x2) swap(x1, x2);
  ll time = 0;
  if (x1 < p && p < x2) {
    if (d == 1) {
      time += (p - x1) * t2;
      p = s;
      d = -1;
    } else {
      time += (x2 - p) * t2;
      p = 0;
      d = 1;
    }
  }
  while (p + d * t1 != x2) {
    time += t1;
    p += d * t1;
    if (p == 0) d = 1;
    if (p == s) d = -1;
  }
  if (x1 != p) time += abs(x1 - p) * t2;
  cout << time << endl;
  return 0;
}
