#include<bits/stdc++.h>
using namespace std;
const int N = 3e5 + 5;
int par[N], bit[N], l[N], r[N], ans[N], T, m, n, a[N], q, x, y;
vector<int> v[N];
void upd(int x, int y) {
	for (; x <= n; x += x & -x) {
		bit[x] = max(bit[x], y);
	}
}
int get(int x) {
	int ans = 0;
	for (; x > 0; x -= x & -x) {
		ans = max(ans, bit[x]);
	}
	return ans;
}
int main() {
	cin >> n >> m;
	for (int i = 1; i <= n; i++) {
		cin >> a[i];
		v[a[i]].push_back(i);
		l[i] = r[i] = i;
	}
	for (int i = n; i >= 1; i--) {
		int j = lower_bound(v[a[i]].begin(), v[a[i]].end(), l[i]) - v[a[i]].begin();
		if (j != v[a[i]].size())l[i] = v[a[i]][j];
		j = upper_bound(v[a[i]].begin(), v[a[i]].end(), r[i]) - v[a[i]].begin() - 1;
		if (j >= 0)r[i] = v[a[i]][j];
	}
	for (int i = 1; i <= n; i++) {
		int t = get(r[i]);
		par[i] = (t == 0 ? 0 : par[t]);
		upd(l[i], i);
	}
	for (int i = 1; i <= m; i++) {
		cin >> x >> y;
		ans[i] = (par[x] == 0 ? 0 : par[x] >= y);
	}
	for (int i = 1; i <= m; i++)cout << (ans[i] ? "Shi\n" : "Fou\n");
}
