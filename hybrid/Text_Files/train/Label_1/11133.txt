#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k;
    cin >> n >> k;
    if (k % 2 == n && n == 1)
      cout << "NO" << endl;
    else {
      cout << "YES ";
      if (k & 1) {
        cout << (n - 1) / 2 << endl;
      } else {
        cout << (n - 1) / 2 - 1 << endl;
      }
    }
  }
  return 0;
}
