#include <bits/stdc++.h>
using namespace std;
int g[1005][1005], a[1005], tot, flag;
void dfs(int u, int v) {
  if (g[u][v] != -1) return;
  if (a[u] == 0) a[u] = 1;
  if (a[v] == 0) a[v] = 1;
  tot++;
  g[u][v] = 1, g[v][u] = 1;
  for (int i = 1; i <= g[0][0]; i++)
    if (!g[v][i] && i != u) dfs(v, i);
}
int main() {
  int n, aa, bb;
  cin >> n >> aa >> bb;
  if (aa < 2 || bb < 2) {
    cout << "NO" << endl;
    return 0;
  }
  for (int i = 1; i <= n; i++) {
    a[i] = 0;
    for (int j = 1; j <= n; j++) g[i][j] = -1;
  }
  tot = 0;
  while (tot < aa - 1) {
    int u = rand() % n + 1, v = rand() % n + 1;
    if (u != v && g[u][v] == -1) dfs(u, v);
  }
  g[0][0] = n;
  flag = 0;
  for (int i = 1; i <= n; i++)
    if (i == 1)
      cout << "YES\n";
    else if (a[i] == 0) {
      if (flag == 0)
        cout << g[i][1], flag = 1;
      else
        for (int j = 1; j <= n; j++) {
          if (j == 1 || a[j] != 0) continue;
          dfs(i, j);
          for (int k = 1; k <= g[0][0]; k++) cout << g[i][k] << " ";
          cout << endl;
          return 0;
        }
    } else
      for (int j = 1; j <= n; j++) {
        if (j == 1 || a[j] != 0) continue;
        dfs(i, j);
        for (int k = 1; k <= g[0][0]; k++) cout << g[i][k] << " ";
        cout << endl;
        return 0;
      }
  return 0;
}
