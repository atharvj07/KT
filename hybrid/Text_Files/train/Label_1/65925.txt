t = int(input())
for _ in range(t):
    q, d = map(int, input().split())
    a = list(map(int, input().split()))
    for num in a:
        if num >= d * 10:
            print("YES")
        else:
            temp = num
            while temp > 0:
                if temp % 10 == d:
                    print("YES")
                    break
                temp -= d
            else:
                print("NO")
