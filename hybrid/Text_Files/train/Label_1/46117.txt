import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++) {
            board[i] = sc.next().toCharArray();
        }
        boolean[][] visited = new boolean[n][n];
        int free = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.') {
                    free++;
                }
            }
        }
        if (free % 5 != 0) {
            System.out.println("NO");
            return;
        }
        int[][][] pieces = {
            { { 0, 0 }, { 1, 0 }, { 2, 0 }, { 1, 1 }, { 1, 2 } },
            { { 0, 0 }, { 1, 0 }, { 2, 0 }, { 2, 1 }, { 2, 2 } },
            { { 0, 0 }, { 1, 0 }, { 2, 0 }, { 0, 1 }, { 0, 2 } },
            { { 0, 0 }, { 0, 1 }, { 0, 2 }, { 1, 1 }, { 2, 1 } }
        };
        int[][] dirs = { { 1, 2 }, { 1, 0 }, { 0, 0 }, { 1, 0 }, { 0, 0 } };
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.' && !visited[i][j]) {
                    for (int[][] piece : pieces) {
                        boolean valid = true;
                        for (int[] point : piece) {
                            int x = i + point[0];
                            int y = j + point[1];
                            if (x < 0 || x >= n || y < 0 || y >= n || board[x][y] == '#') {
                                valid = false;
                                break;
                            }
                        }
                        if (valid) {
                            for (int[] point : piece) {
                                int x = i + point[0];
                                int y = j + point[1];
                                board[x][y] = '#';
                                visited[x][y] = true;
                            }
                            free -= 5;
                            break;
                        }
                    }
                }
            }
        }
        if (free == 0) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
