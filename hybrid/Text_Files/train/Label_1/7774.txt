#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int a, d, m, n;
vector<ll> clients;

ll get(ll x) {
    return (x + a - 1) / a;
}
int bs(ll left, ll right) {
    while (left < right) {
        ll mid = (left + right) / 2;
        ll tmp = get(mid);
        if (tmp < (ll)clients.size()) {
            if (clients[tmp - 1] < mid)
                left = mid + 1;
            else
                right = mid;
        } else
            right = mid;
    }
    return left;
}

int main() {
    cin >> n >> m >> a >> d;
    clients.resize(m);
    for (int i = 0; i < m; ++i) cin >> clients[i];
    cout << bs(1, clients[m - 1] + d);
}
