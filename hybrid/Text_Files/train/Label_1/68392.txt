n, k = map(int, input().split())
marks = list(map(int, input().split()))

current_sum = sum(marks)
current_avg = current_sum / n

additional_marks = 0
while True:
    if round((current_sum + k * additional_marks) / (n + additional_marks)) >= k:
        break
    additional_marks += 1

print(additional_marks)
