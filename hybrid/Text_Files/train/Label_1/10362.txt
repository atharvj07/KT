#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<string> s(n);
  for (int i = 0; i < n; i++) { cin >> s[i]; }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int l = -1, r = -1, cnt = 0;
    for (int j = 0; j < m; j++) {
      if (s[i][j] == '1') {
        if (l == -1)
          l = j;
        r = j;
        cnt++;
      }
      if (cnt > k) {
        ans += r - l + 1;
        l = j, cnt = 1;
      }
    }
    if (cnt > 0) { ans += r - l + 1; }
  }
  cout << ans << endl;
  return 0;
}
