#include<bits/stdc++.h>
using namespace std;
int main() {
  int N;
  cin >> N;
  vector<int> T(N);
  int X;
  int a[2] = {0};
  int b[2] = {0};
  for (int i = 0; i < N; i++) {
    cin >> T[i];
    a[T[i] - 1]++;
  }
  int ans = 0;
  for (int i = 0; i < N; i++) {
    int j = i;
    while (j < N && T[i] == T[j]) j++;
    int L = j - i;
    ans = max(ans,min(b[T[i] - 1], a[T[i] - 1] - b[T[i] - 1]) * 2 + L);
    while (i < j) b[T[i++] - 1]++;
  }
  cout << ans << endl;
  return 0;
}
