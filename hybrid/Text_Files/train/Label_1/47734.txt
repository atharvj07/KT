import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n], b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            long x = (a[i] + b[i] - 1) / b[i];
            ans += x;
        }
        System.out.println(ans);
    }
}
