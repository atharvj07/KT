import sys
readline = sys.stdin.readline

n, a, b, t = map(int, readline().strip().split())
ori = readline().strip()
last = -1
tot = (t - b) // a
left = ans = 0
for i, c in enumerate(ori):
    if i == 0 or c != ori[i - 1]:
        ans += 1
    if c == ori[last]:
        left += 1
    else:
        left = 0
        last = i
    tot -= min(left, tot)
    if tot <= 0:
        break
print(ans)
