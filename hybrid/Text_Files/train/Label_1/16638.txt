#include<bits/stdc++.h>
using namespace std;
#define MOD 1000000007LL
typedef long long ll;
int main(){
    string a;
    ll k;
    cin>>a>>k;
    ll n=a.size()*k,ans=0;
    vector<pair<ll,ll>>dp(n,{0,0});
    dp[0].first=(a[0]-'0')%5==0;
    dp[0].second=(a[0]-'0')%5==0;
    for(ll i=1;i<n;++i){
        ll x=(a[i]-'0')%5;
        dp[i].first=(dp[i-1].first*10 + x)%5;
        dp[i].second=(dp[i-1].second*10 + x)%5;
    }
    vector<pair<ll,ll>>pw(n);
    pw[0]={1,1};
    for(ll i=1;i<n;++i){
        pw[i].first=(pw[i-1].first*10)%5;
        pw[i].second=(pw[i-1].second*10)%5;
    }
    ans=pw[n-1].first;
    ll pre=0,suf=0;
    for(ll i=n-2;i>=0;--i){
        ll sum=(pw[i].first*pre + pw[n-1-i].second*suf)%5;
        ans=(ans + sum)%MOD;
        pre=(pre + pw[i].second*dp[i].first)%5;
        suf=(suf + pw[n-1-i].first*dp[i].second)%5;
    }
    cout<<(ans+MOD)%MOD<<endl;
    return 0;
}
