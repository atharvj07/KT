const char nl = '\n';
const int mod = 998244353;
const int maxn = 2e5 + 50;

int n, cnt[26];
string s;

int main() {
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);

    cin >> n >> s;
    F0R(i, n) cnt[s[i] - 'a']++;
    int eq = 0, mx = 0;
    F0R(i, 26) {
        eq += (cnt[i] == 1);
        mx = max(mx, cnt[i]);
    }
    eq = (eq > 1);
    if (eq) {
        cout << (1ll * mx * (mx - 1)) % mod << nl;
    } else {
        cout << (1ll * mx * (mx - 1) * (mx - 2)) % mod << nl;
    }

    return 0;
}
