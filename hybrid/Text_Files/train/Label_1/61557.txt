import sys

n, k = [int(i) for i in sys.stdin.readline().split()]
a = [int(i) for i in sys.stdin.readline().split()] + [0]

cnt = {}
for x in a:
    if x not in cnt:
        cnt[x] = 0
    cnt[x] += 1

stress = {}
for x in cnt:
    stress[x] = n - cnt[x]

genre = -1
min_stress = n
for x in cnt:
    if stress[x] < min_stress:
        min_stress = stress[x]
        genre = x

print(genre)
