#include<bits/stdc++.h>
using namespace std;
int main() {
    int n; cin >> n;
    vector<int> a(n);
    for(int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    bool f = 0;
    int k = -1, ans = -1, sum = 0;
    for(int i = n - 1; i >= 0; i--) sum += a[i];
    for(int i = n - 1; i >= 0; i--) {
        int x = ceil((double)sum / (n + 3));
        int y = floor((double)sum / (n + 1));
        int z = sum - a[i] - (n - 1) * x;
        if(z >= x && z <= y) {
            f = 1;
            k = i;
            ans = z;
            break;
        }
        sum -= a[i];
    }
    if(f) {
        cout << "YES" << endl;
        for(int i = n - 1; i > k; i--) cout << a[i] << endl;
        for(int i = k; i >= 0; i--) cout << ans << endl;
    }
    else cout << "NO" << endl;
}
