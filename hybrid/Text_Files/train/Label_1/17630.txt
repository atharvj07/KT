struct DSU {
    vector<int> parent, sz; 
    void init(int n) { parent.resize(n);iota(all(parent), 0); sz.assign(n,1); }
    int find(int a) { return parent[a] == a ? a : parent[a] = find(parent[a]); } 
    int merge(int a, int b) { 
        if((a=find(a))!=(b=find(b))) { 
            if(sz[a] < sz[b]) swap(a, b); 
            parent[b] = a; sz[a] += sz[b]; return 1; } 
        return 0; }
    int size(int a) { return sz[find(a)];}
    bool same(int a, int b) { return find(a) == find(b);}
}; 
int32_t main(){
    ios_base::sync_with_stdio(false); cin.tie(0); cout.tie(0);

    while(t--) {
        int n, l, r; cin >> n >> l >> r;
        r -= n-1; l -= n-1;
        r %= (n-1)*(n);
        l %= (n-1)*(n);
        if(r < l) r += (n-1)*(n);
        FOR(i, 0, n) cout << ((i-1+l+n-1)%n+1) << " ";
        cout << '\n';
    }
    return 0;
}
