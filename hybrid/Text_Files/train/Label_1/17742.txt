#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll a[1005], b[1005], dp[1005][1005];
int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) cin >> b[i];
    for (int i = 1; i <= n; i++) {
        a[i] |= a[i - 1];
        b[i] |= b[i - 1];
    }
    for (int l = 1; l <= n; l++) {
        dp[l][l] = a[l] + b[l];
    }
    for (int len = 2; len <= n; len++) {
        for (int l = 1; l <= n - len + 1; l++) {
            int r = l + len - 1;
            dp[l][r] = dp[l][r - 1] + a[r];
            dp[l][r] = max(dp[l][r], dp[l + 1][r] + b[l]);
        }
    }
    cout << dp[1][n] << endl;
}
