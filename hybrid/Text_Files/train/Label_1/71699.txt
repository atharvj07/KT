def check_substrings(s):
    if "AB" in s:
        index_ab = s.find("AB")
        if "BA" in s[:index_ab] or "BA" in s[index_ab+2:]:
            return "YES"
    if "BA" in s:
        index_ba = s.find("BA")
        if "AB" in s[:index_ba] or "AB" in s[index_ba+2:]:
            return "YES"
    return "NO"

s = input()
print(check_substrings(s))
