#include<bits/stdc++.h>
using namespace std;
int main() {
    int t; cin >> t;
    while(t--) {
        vector<int>a; int n; cin >> n; int q[105]={0};
        for(int i=0; i<n; i++) {
            int x; cin >> x; q[x]++;
        }
        int x=0;
        while(q[x]) x++;
        int y=0;
        for(int i=x; i<105; i++) {
            if(q[i]) {
                y=i; break;
            }
        }
        cout << x+y << endl;
    }
}
