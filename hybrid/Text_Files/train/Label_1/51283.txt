import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        java.util.Scanner in = new java.util.Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            b[i] = in.nextInt();
        }
        Arrays.sort(a);
        Arrays.sort(b);
        int min = 0;
        for (int i = 0; i < n; i++) {
            min = Math.max(min, (b[i] - a[i] + m) % m);
        }
        System.out.println(min);
    }
}
