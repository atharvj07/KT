#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> x(n), c(n);
    for (int i = 0; i < n; i++) cin >> x[i];
    int a;
    cin >> a;
    for (int i = 0; i < n; i++) cin >> c[i];
    vector<int> dp(n + 1, 1e9);
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
        if (dp[i - 1] + c[i - 1] < dp[i]) dp[i] = dp[i - 1] + c[i - 1];
        if (dp[max(0, i - 2)] + a + c[i - 1] < dp[i]) dp[i] = dp[max(0, i - 2)] + a + c[i - 1];
        if (x[i - 1] <= k) dp[i] = 0;
        k = max(k, x[i - 1]);
    }
    if (dp[n] == 1e9) dp[n] = -1;
    cout << dp[n];
    return 0;
}
