#include <iostream>

using namespace std;

int main() {
    string s;
    cin >> s;
    int m;
    cin >> m;
    if (s[0] == '1' && m % 2 == 0) {
        cout << "YES\n";
        for (int i = 0; i < m / 2; i++) {
            cout << 1 << " " << 10 << " ";
        }
    } else if (s[9] == '1' && m % 2 == 1) {
        cout << "YES\n";
        cout << 10 << " ";
        for (int i = 0; i < m / 2; i++) {
            cout << 1 << " " << 10 << " ";
        }
    } else {
        cout << "NO";
    }
    return 0;
}
