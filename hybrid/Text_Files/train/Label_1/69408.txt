for t in range(int(input())):
    n, k1, k2 = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    assert len(a) == k1
    assert len(b) == k2
    assert k1 + k2 == n
    a.sort()
    b.sort(reverse=True)
    ok = False
    for aa in a:
        if ok: break
        for bb in b:
            if aa < bb:
                ok = True
                break
            if aa == bb:
                continue
            b.remove(bb)
    print("YES" if ok else "NO")
