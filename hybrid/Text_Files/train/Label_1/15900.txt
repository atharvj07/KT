#include<iostream>
#include<queue>
#include<vector>
using namespace std;
#define ll long long
const int maxn=20000;

int main()
{
	int n,a[110],b[110],cnt=0,ans=0;
	queue<int>q; ll dis[100000];
	bool f[100000];
	vector<pair<int,int> >op[100000];
	cin>>n;
	for (int i=1; i<=n; i++) cin>>a[i]>>b[i];
	q.push(0); f[0]=1; dis[0]=0;
	while (!q.empty())
	{
		int now=q.front(); q.pop();
		if (now==cnt)
		{
			ans=dis[now];
			break;
		}
		for (int i=1; i<=n; i++)
		{
			int step=a[i];
			if (now>=step && f[now-step]==0)
			{
				q.push(now-step);
				dis[now-step]=dis[now]+1;
				f[now-step]=1;
				op[now-step].push_back(make_pair(i,-1));
			}
			step=b[i];
			if (now+step<=99999 && f[now+step]==0)
			{
				q.push(now+step);
				dis[now+step]=dis[now]+1;
				f[now+step]=1;
				op[now+step].push_back(make_pair(i,1));
			}
		}
	}
	if (ans==0) cout<<-1<<endl;
	else
	{
		cout<<ans<<endl;
		while (cnt!=0)
		{
			cout<<op[cnt][0].first<<' '<<op[cnt][0].second<<endl;
			cnt=cnt-op[cnt][0].second*b[op[cnt][0].first];
		}
	}
	return 0;
}
