n=int(input())
a=[]
b=[]
for i in range(n):
    q,w=map(int,input().split())
    a.append(q)
    b.append(w)
ans=[]
l=[]
for i in range(2*n+1):
    l.append(0)
for i in range(n):
    l[a[i]]+=1
    l[b[i]]+=1
for i in range(1,2*n+1):
    if(l[i]==0):
        continue
    ans.append(i)
    if(l[i]>=2):
        ans.append(i)
for i in ans:
    print(ans.index(i)+1,end=" ")
