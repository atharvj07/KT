n = int(input())
s = list(map(int, input().split()))
c = list(map(int, input().split()))
t = sorted([i for i in zip(s, c)])
ans = 1e9
for i in range(n):
    for j in range(i + 1, n):
        for k in range(j + 1, n):
            if t[i][0] < t[j][0] < t[k][0]:
                ans = min(ans, t[i][1] + t[j][1] + t[k][1])
print(ans if ans != 1e9 else -1)
