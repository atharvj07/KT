#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main(){
    int n,m;
    cin>>n>>m;
    map<string,ll> M;
    vector<string> v;
    for(int i=0;i<n;i++){
        string s1,s2;
        char op;
        cin>>s1;
        cin>>s2>>op;
        if(op == '='){
            ll x;
            cin>>x;
            M[s1] = x;
        }
        else{
            string s3;
            cin>>s2>>s3;
            if(s2 == "?") v.push_back(s1+" "+s3+" "+op);
            else if(s3 == "?") v.push_back(s1+" "+s2+" "+op);
            else{
                ll x = M[s2];
                ll y = M[s3];
                if(op == 'A') M[s1] = x&y;
                else if(op == 'O') M[s1] = x|y;
                else M[s1] = x^y;
            }
        }
    }
    ll mn = LLONG_MAX, mx = LLONG_MIN;
    for(int i=0;i<(1<<v.size());i++){
        map<string,ll> MP = M;
        ll sum = 0;
        for(int j=0;j<v.size();j++){
            if(i & (1<<j)){
                string s1 = v[j].substr(0,v[j].find(' '));
                string s2 = v[j].substr(v[j].find(' ')+1,v[j].find(' ')-v[j].find(' ')-1);
                string s3 = v[j].substr(v[j].find(' ')+1,v[j].size()-v[j].find(' ')-1);
                ll x = MP[s2];
                ll y = MP[s3];
                if(v[j][v[j].size()-1] == 'A') MP[s1] = x&y;
                else if(v[j][v[j].size()-1] == 'O') MP[s1] = x|y;
                else MP[s1] = x^y;
            }
        }
        for(auto it:MP) sum += it.second;
        mn = min(mn,sum);
        mx = max(mx,sum);
    }
    bitset<1005> a(mn),b(mx);
    string x = a.to_string().substr(a.to_string().size()-m);
    string y = b.to_string().substr(b.to_string().size()-m);
    cout<<x<<endl<<y<<endl;
}
