#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> p(2 * n);
    for (int i = 0; i < 2 * n; ++i) {
        cin >> p[i];
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        if (p[i] != i + 1) {
            int j = -1;
            for (int k = n; k < 2 * n; ++k) {
                if (p[k] == i + 1) {
                    j = k;
                }
            }
            if (j == -1) {
                cout << -1 << endl;
                return 0;
            }
            ++ans;
            swap(p[i], p[j]);
        }
    }
    cout << ans << endl;
    return 0;
}
