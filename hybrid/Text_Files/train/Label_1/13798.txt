#include<bits/stdc++.h>
using namespace std;
bool isSubsequence(string a, string b) {
    int pos = a.size() - 1;
    for (int i = b.size() - 1; i >= 0; i--) {
        if (b[i] == a[pos] && pos >= 0) {
            pos--;
        }
    }
    return pos == -1;
}
int main() {
    int n;
    cin >> n;
    vector<string> s(n);
    for (int i = 0; i < n; i++) {
        cin >> s[i];
    }
    vector<string> res;
    res.push_back(s[0]);
    for (int i = 1; i < n; i++) {
        bool ok = false;
        for (int j = 0; j < res.size(); j++) {
            if (isSubsequence(res[j], s[i])) {
                ok = true;
                res.insert(res.begin() + j + 1, s[i]);
                break;
            }
        }
        if (!ok) {
            cout << "NO" << endl;
            return 0;
        }
    }
    cout << "YES" << endl;
    for (string x : res) {
        cout << x << endl;
    }
    return 0;
}
