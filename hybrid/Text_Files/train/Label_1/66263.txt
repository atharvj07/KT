s = input()
cur = 1
ans = 0
for i in range(0, len(s)):
    for j in range(i + 1, len(s) + 1):
        tmp = s[i: j]
        flag = True
        for k in range(0, len(tmp)):
            if tmp[k] != tmp[len(tmp) - 1 - k]:
                flag = False
        if flag == False:
            cur = max(cur, len(tmp))
    ans = max(ans, cur)
print(ans)
