#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, k, ans = 0, End = (1ll << 50) - 1, a[55];
  cin >> n >> k;
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n, greater<long long>());
  if (k == n) {
    for (int i = 0; i < k; i++) ans &= a[i];
  } else {
    ans = a[0] & a[1];
    for (int i = 2; i < k; i++) {
      long long cur = a[i];
      if (cur > 0) ans &= cur;
    }
  }
  cout << ans << endl;
  return 0;
}
