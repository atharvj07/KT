import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), k = input.nextInt();
        int[] a = new int[]{0, 0, 0, n+1};
        for (int i = 0; i < n; i++) {
            int x = input.nextInt(), r = input.nextInt(), q = input.nextInt();
            a[0] = Math.max(a[0], x - r);
            a[1] = Math.max(a[1], x + r);
            a[2] = Math.min(a[2], x - r);
            a[3] = Math.min(a[3], x + r);
        }
        int ans = 0;
        for (int i = 0; i < 3; i++) {
            if (a[i+1] > a[i]) {
                ans += a[i+1] - a[i] - (a[i+1] - a[i]) % (k + 1) / (k + 1);
            }
        }
        System.out.print(ans);
    }
}
