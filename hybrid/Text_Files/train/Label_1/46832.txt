import java.util.Scanner;

public class Main {

    static int minimumLength(int n, String s) {
        int countZeros = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '0') {
                countZeros++;
            }
        }
        return Math.max(0, countZeros - 1);
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String s = scanner.next();
        System.out.println(minimumLength(n, s));
    }
}
