for _ in range(int(input())):
    n = int(input())
    a = []
    b = []
    for _ in range(n):
        l,r = map(int,input().split())
        a.append(l)
        b.append(r)
    q = []
    for j in range(n):
        if q and q[0] == a[j]:
            q.pop(0)
        q.append(b[j])
        if not q:
            print(a[j])
        else:
            print(q[0])
