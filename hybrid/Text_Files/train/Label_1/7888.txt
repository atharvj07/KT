#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> c(n * k), f(n), h(k);
  for (int &i : c) cin >> i;
  for (int &i : f) cin >> i;
  for (int &i : h) cin >> i;
  vector<vector<int>> freq(n);
  vector<int> ans(k);
  for (int i = 0; i < n * k; ++i) freq[c[i] % n].push_back(c[i]);
  for (int i = 1; i < k; ++i)
    h[i] -= h[i - 1];
  for (int i = 0; i < n; ++i) {
    int idx = f[i] % n;
    for (int j = 0; j < k; ++j) {
      while (freq[idx].size() && freq[idx][freq[idx].size() - 1] != f[i]) freq[idx].pop_back();
      if constexpr (false) cout << i << ' ' << j << ' ' << freq[idx].size() << endl;
      if (freq[idx].size()) {
        freq[idx].pop_back();
        ans[j] += h[k - j - 1];
      } else
        break;
      if (freq[idx].size()) idx = freq[idx][freq[idx].size() - 1] % n;
    }
  }
  cout << accumulate(ans.begin(), ans.begin() + k, 0) << endl;
}
