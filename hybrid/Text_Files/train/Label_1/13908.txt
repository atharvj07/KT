#include <bits/stdc++.h>
using namespace std;
int n, c0, c1, h;
int main() {
  int t;
  cin >> t;
  for (; t--;) {
    cin >> n >> c0 >> c1 >> h;
    string s;
    cin >> s;
    int ans = 0;
    for (int i = 0; i < n; ++i) {
      if (s[i] == '0') {
        ans += c0;
      }
      if (s[i] == '1') {
        ans += min(c1, c0 + h);
      }
    }
    cout << ans << '\n';
  }
}
