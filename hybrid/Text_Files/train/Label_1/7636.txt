#include <iostream>
using namespace std;
typedef long long ll;

ll f(ll x) {
    //calculate ranges
    ll cnt = 0;
    for (ll r = 1; r <= x; r *= 10) {
        ll l = max(1ll, (x + r) / 10), rr = (x + r - 1) / 10;
        if (l <= rr) cnt += rr / 2 - (l - 1) / 2;
    }
    return x - cnt;
}

int main() {
    ll n;
    cin >> n;
    cout << f(n) << endl;
    return 0;
}
