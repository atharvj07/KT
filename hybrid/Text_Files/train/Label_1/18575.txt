#include <iostream>
#include <string>
using namespace std;

string solve(int n, int m, string s) {
    while (m--) {
        string t = s;
        for (int i = 0; i < n; i++) {
            int cnt = 0;
            if (i - 1 >= 0) cnt += s[i - 1] == '1';
            if (i + 1 < n) cnt += s[i + 1] == '1';
            if (s[i] == '0' && cnt == 1) t[i] = '1';
        }
        s = t;
    }
    return s;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        string s;
        cin >> n >> m >> s;
        cout << solve(n, m, s) << "\n";
    }
    return 0;
}
