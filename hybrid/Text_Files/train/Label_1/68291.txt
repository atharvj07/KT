n, x, y = map(int, input().split())

for _ in range(n):
    a = int(input())
    vanya_hits = (a * x) / (x + y)
    vova_hits = (a * y) / (x + y)
    if vanya_hits.is_integer() and vova_hits.is_integer():
        print("Both")
    elif vanya_hits > int(vanya_hits):
        print("Vova")
    else:
        print("Vanya")
