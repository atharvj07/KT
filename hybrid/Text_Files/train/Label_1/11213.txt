#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    long long cnt = 0;
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        sum += x;
        cnt += sum >= 0;
        sum = max(sum, 0LL);
    }
    cout << cnt << '\n';
    return 0;
}
