import sys

n, m = map(int, input().split())

t = [0] * m
l = [0] * m
r = [0] * m
d = [0] * m
mi = [0] * m

for i in range(m):
	t[i], l[i], r[i], *p = map(int, input().split())
	if t[i] == 1:
		d[i] = p[0]
	else:
		mi[i] = p[0]

a = [0] * n

for i in range(m):
	if t[i] == 1:
		for j in range(l[i] - 1, r[i]):
			a[j] += d[i]
	else:
		ma = -10000000000
		for j in range(l[i] - 1, r[i]):
			ma = max(ma, a[j])
		if ma < mi[i]:
			print('NO')
			sys.exit()

print('YES')
for i in range(n):
	print(a[i], end=' ')
