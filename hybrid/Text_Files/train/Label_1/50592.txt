import java.util.Scanner;

public class Time {
    static class Point {
        int x, y;

        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int d = scanner.nextInt();
        int[] a = new int[n];
        a[0] = scanner.nextInt();
        Point[] points = new Point[n];
        points[0] = new Point(0, 0);
        for (int i = 1; i < n; i++) {
            points[i] = new Point(scanner.nextInt(), scanner.nextInt());
            a[i] = scanner.nextInt();
        }

        int money = 0;
        long time = 0;
        for (int i = 0; i < n - 1; i++) {
            long distance = Math.abs(points[i].x - points[i + 1].x) + Math.abs(points[i].y - points[i + 1].y);
            long timeNeeded = d * distance;
            if (timeNeeded > time) {
                long timeToBuy = timeNeeded - time;
                money += (timeToBuy + a[i + 1] - 1) / a[i + 1];
                time += timeToBuy;
            }
            time += a[i + 1];
        }

        System.out.println(money);
    }
}
