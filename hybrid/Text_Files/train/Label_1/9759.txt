#include <bits/stdc++.h>
using namespace std;
struct operation {
    int type, l, r, d, m;
};
const int MAXN = 5005;
long long a[MAXN], ans[MAXN], b[MAXN];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n, m;
    cin >> n >> m;
    vector<operation> opr(m);
    for (int i = 0; i < m; i++) {
        cin >> opr[i].type;
        if (opr[i].type == 1) cin >> opr[i].l >> opr[i].r >> opr[i].d;
        else cin >> opr[i].l >> opr[i].r >> opr[i].m;
    }
    bool flag = 1;
    multiset<long long> t;
    for (int i = m - 1; i >= 0; i--) {
        if (opr[i].type == 1) {
            for (int j = opr[i].l; j <= opr[i].r; j++) t.erase(t.find(a[j]));
            for (int j = opr[i].l; j <= opr[i].r; j++) { a[j] += opr[i].d; }
            for (int j = opr[i].l; j <= opr[i].r; j++) t.insert(a[j]);
        } else {
            for (int j = opr[i].l; j <= opr[i].r; j++) t.erase(t.find(a[j]));
            for (int j = opr[i].l; j <= opr[i].r; j++) { a[j] = opr[i].m; }
            for (int j = opr[i].l; j <= opr[i].r; j++) t.insert(a[j]);
            if (*t.begin() != opr[i].m) { flag = 0; }
        }
    }
    if (flag) {
        cout << "YES" << endl;
        for (int i = 1; i <= n; i++) cout << a[i] << " ";
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
