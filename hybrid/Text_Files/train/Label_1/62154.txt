n = int(input())
s = input()
s = s[::-1]+s
dp = [0 for i in range(len(s))]
pow2 = [1 for i in range(len(s))]
for i in range(1, len(s)):
    pow2[i] = pow2[i-1]*2
dp[len(s)-1] = pow2[len(s)-1]
for i in range(len(s)-2, -1, -1):
    if s[i] == '1':
        dp[i] = pow2[i]
        if dp[i+1] > 0:
            dp[i] += dp[i+1]
    if i != 0:
        dp[i] += dp[i-1]
for i in range(len(s)-len(s)//2, -1, -1):
    if s[i] == '1':
        dp[i] = pow2[i]
        if dp[(i+1)%len(s)] > 0:
            dp[i] += dp[(i+1)%len(s)]
    if i != 0:
        dp[i] += dp[i-1]
print(dp[0])
