#include <bits/stdc++.h>
using namespace std;
int n;
string s;
int main() {
  cin >> s >> n;
  int f[26] = {};
  for (int i = 0; i < s.size(); i++) f[s[i] - 'a']++;
  int cnt = 0;
  for (int i = 0; i < 26; i++)
    if (f[i]) cnt++;
  cnt = (cnt + n - 1) / n;
  if (cnt * n < s.size())
    cout << -1 << endl;
  else {
    cout << cnt << endl;
    string t = "";
    for (int i = 0; i < 26; i++)
      for (int j = 0; j < f[i] / cnt; j++) t += i + 'a';
    for (int i = 0; i % cnt != f[t[i % cnt] - 'a'] / cnt; i++) cout << t[i % cnt];
  }
  return 0;
}
