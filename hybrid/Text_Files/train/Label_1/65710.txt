def solve(n, p):
    mn = -1
    for i in range(1, n):
        if p[i - 1] > p[i]:
            mn = i - 1
            break
    if mn == -1:
        return p
    idx = -1
    for i in range(mn + 1, n):
        if p[i] < p[mn]:
            idx = i
    p[mn], p[idx] = p[idx], p[mn]
    p[mn + 1:] = p[mn + 1:][::-1]
    return p

for _ in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    print(*solve(n, p))
