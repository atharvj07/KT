#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
#define nl "\n"
#define all(x) x.begin(), x.end()

const int mxN = 2e5 + 500;
vector<int> adj[mxN], ord;
int low[mxN], in[mxN], timer = 0, comp[mxN], sz[mxN];
bool vis[mxN];

void dfs(int u) {
    vis[u] = 1;
    in[u] = low[u] = timer++;
    ord.push_back(u);
    for (auto v : adj[u]) {
        if (vis[v] == 0) {
            dfs(v);
            low[u] = min(low[u], low[v]);
        } else if (in[v] < in[u])
            low[u] = min(low[u], in[v]);
    }

    if (in[u] == low[u]) {
        int v;
        comp[u] = u;
        do {
            v = ord.back();
            ord.pop_back();
            comp[v] = u;
            sz[u]++;
        } while (v != u);
    }
}

void solve() {
    int n, m, q;
    cin >> n >> m >> q;
    vector<int> v(n), a(m), p[n + 5];
    unordered_map<int, int> mp;
    for (int i = 0; i < n; i++) {
        cin >> v[i];
        mp[v[i]] = i;
    }
    for (int i = 0; i < m; i++) {
        cin >> a[i];
        p[mp[a[i]]].push_back(i);
    }

    while (q--) {
        int l, r;
        cin >> l >> r;
        l--, r--;
        int c1 = -1;
        for (int i = 0; i < n; i++) {
            int in1 = lower_bound(all(p[i]), l) - p[i].begin();
            int in2 = upper_bound(all(p[i]), r) - p[i].begin() - 1;
            if (in1 <= in2) {
                c1 = i;
                break;
            }
        }
        int ans = 0;
        if (c1 != -1) { ans |= (int)(p[c1].size() >= n); }
        cout << ans;
    }
}

int main() {
    solve();
    return 0;
}
