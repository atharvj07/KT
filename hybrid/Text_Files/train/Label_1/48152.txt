import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++)
            a[i] = in.nextLong();
        Arrays.sort(a);
        PriorityQueue<Long> pq = new PriorityQueue<>();
        for (int i = 0; i < m; i++)
            pq.offer(1L);
        long ans = 0;
        for (int i = n - 1; i >= 0; i--) {
            long t = a[i];
            if (!pq.isEmpty()) {
                long val = pq.poll();
                t /= val;
                ans += t;
                pq.offer(val * 2);
            } else {
                ans += t;
            }
        }
        System.out.println(ans);
    }
}
