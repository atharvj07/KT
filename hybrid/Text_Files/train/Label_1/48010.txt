import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int q = scanner.nextInt();

    for (int i = 0; i < q; i++) {
      int n = scanner.nextInt();
      int[] a = new int[n];

      for (int j = 0; j < n; j++) {
        a[j] = scanner.nextInt();
      }

      int minSum = 0;
      for (int j = 0; j < n; j++) {
        minSum += a[j];
      }

      int equalPrice = (minSum + n - 1) / n;
      System.out.println(equalPrice);
    }
  }
}
