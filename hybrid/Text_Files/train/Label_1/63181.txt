for _ in range(int(input())):
    n, s = list(map(int, input().split()))
    x = str(n)
    x1 = list(x)
    ans = 0
    sum1 = 0
    for j in x1:
        sum1 += int(j)
    while sum1 > s:
        ch = x[-1]
        if int(ch) != 0:
            x[-1] = str(int(ch) - 1)
            sum1 -= 1
            ans += 1
        else:
            x.pop()
            ans += 1
    print(ans)
