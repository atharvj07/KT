import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int[][] student = new int[N][2];
        int[][] checkpoint = new int[M][2];
        for (int i = 0; i < N; i++) {
            student[i][0] = sc.nextInt();
            student[i][1] = sc.nextInt();
        }
        for (int i = 0; i < M; i++) {
            checkpoint[i][0] = sc.nextInt();
            checkpoint[i][1] = sc.nextInt();
        }
        for (int i = 0; i < N; i++) {
            int minDist = Integer.MAX_VALUE;
            int minIdx = -1;
            for (int j = 0; j < M; j++) {
                int dist = Math.abs(student[i][0] - checkpoint[j][0]) + Math.abs(student[i][1] - checkpoint[j][1]);
                if (dist < minDist) {
                    minDist = dist;
                    minIdx = j;
                } else if (dist == minDist) {
                    minIdx = Math.min(minIdx, j);
                }
            }
            System.out.println(minIdx + 1);
        }
    }
}
