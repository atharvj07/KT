n, m = map(int, input().split())
answers = [input() for _ in range(n)]
points = list(map(int, input().split()))

total_score = 0
for i in range(m):
    answer_count = [0]*5
    for j in range(n):
        answer_count[ord(answers[j][i])-65] += 1
    total_score += max(answer_count) * points[i]

print(total_score)
