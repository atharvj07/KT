import sys

input = sys.stdin.readline

T = int(input())
for _ in range(T):
    x, y = map(int, input().split())
    c = list(map(int, input().split()))
    x = abs(x)
    y = abs(y)
    if x < y:
        x, y = y, x
    if x % 2 == 0:
        ans = c[0] * x + c[5] * y
    else:
        ans = c[0] * x - c[5] * y
    print(ans // 6 + min(ans % 6, c[3] + c[4]))
