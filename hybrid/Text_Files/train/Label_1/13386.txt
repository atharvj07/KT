#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int N=1e3+5;
int n;
ll a[N], b[N], ans[N];

void solve(){
    for(int i=1; i<=n; i++)
        for(int j=1; j<=n; j++)
            if(i!=j)
                a[i]=__gcd(a[i], b[j]);
    for(int i=1; i<=n; i++)
        ans[i]=b[i]/a[i];
}

int main(){
    cin>>n;
    for(int i=1; i<=n; i++)
        for(int j=1; j<=n; j++)
            cin>>b[j];
    solve();
    for(int i=1; i<=n; i++)
        cout<<ans[i]<<" ";
    cout<<endl;
}
