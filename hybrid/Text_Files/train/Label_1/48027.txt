import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int len = sc.nextInt();
        int[] arr = new int[len + 2];
        for (int i = 1; i <= len; i++)
            arr[i] = sc.nextInt();
        int times = 0;
        for (;;) {
            int stable = 1;
            for (int i = 2; i <= len; i++) {
                int m = (arr[i - 1] + arr[i] + arr[i + 1]) / 2;
                if (arr[i] != m) {
                    arr[i] = m;
                    stable = 0;
                }
            }
            times++;
            if (stable != 0) {
                break;
            }
        }
        if (times == 1) {
            System.out.println(1);
        } else {
            System.out.println(times - 1);
            for (int i = 1; i <= len; i++)
                System.out.print(arr[i] + " ");
        }
    }
}
