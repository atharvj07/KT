#include <vector>
#include <iostream>
#include <tuple>
#include <algorithm>
#include <map>

using namespace std;

class Point {
public:
    Point(int x_ = 0, int y_ = 0) : x(x_), y(y_) {}
    
    int x, y;
};

typedef tuple<int, int, int, int> Ray;

int main() {
    pair<int, int> room_size;
    int n_sensors;
    cin >> room_size.first >> room_size.second >> n_sensors;
    
    vector<Point> sensors(n_sensors);
    for (int i = 0; i < n_sensors; ++i) {
        cin >> sensors[i].x >> sensors[i].y;
    }
    
    int N = room_size.first * room_size.second;
    vector<Ray> rays(N);
    for (int i = 0; i < N; ++i) {
        rays[i] = make_tuple(0, 0, -1, -1);
    }
    
    int next_ray = 1;
    rays[0] = make_tuple(0, 0, 1, 1);
    
    while (next_ray < N) {
        Ray ray = rays[next_ray - 1];
        int new_x = get<0>(ray) + get<2>(ray);
        int new_y = get<1>(ray) + get<3>(ray);
        
        if (new_x < 0 || new_x >= room_size.first || new_y < 0 || new_y >= room_size.second) {
            get<2>(rays[next_ray - 1]) *= -1;
            get<3>(rays[next_ray - 1]) *= -1;
        } else {
            for (size_t i = 0; i < sensors.size(); ++i) {
                if (sensors[i].x == new_x && sensors[i].y == new_y) {
                    cout << get<1>(ray) << endl;
                    sensors[i].x = -1;
                    break;
                }
            }
        }
        
        rays[next_ray++] = make_tuple(new_x, new_y, get<2>(ray), get<3>(ray));
    }
    
    for (size_t i = 0; i < sensors.size(); ++i) {
        if (sensors[i].x != -1) {
            cout << "-1" << endl;
        }
    }
    
    return 0;
}
