def solve(s):
    stk = []
    ans = []
    need = 0
    for c in s:
        if c in "L)R":
            if c == "L":
                if stk:
                    stk.pop()
            else:
                need += 1
                stk.append(need)
        else:
            stk.append(0)
        ans.append(-1 if stk else max(stk))
    return ans
print(*solve(input()))
