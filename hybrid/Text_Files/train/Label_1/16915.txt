#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  string s;
  cin >> s;
  vector<int> dp(n + 1, INT_MAX);
  dp[0] = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < i; j++) {
      string cur = s.substr(j, i - j);
      if (j > 0)
        dp[i] = min(dp[i], dp[j] + b);
      else
        dp[i] = min(dp[i], a);
      for (int k = j + 1; k < i; k++) {
        if (s.substr(k, i - k) == cur) {
          dp[i] = min(dp[i], dp[k] + b);
          break;
        }
      }
    }
  }
  cout << dp[n] << endl;
  return 0;
}
