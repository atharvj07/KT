def max_dominoes(n, m, edges):
    # Using a graph coloring approach to maximize the number of dominoes
    color = [-1] * n
    adj = [[False] * n for _ in range(n)]
    for edge in edges:
        adj[edge[0] - 1][edge[1] - 1] = True
        adj[edge[1] - 1][edge[0] - 1] = True

    def dfs(v):
        for i in range(n):
            if adj[v][i]:
                if color[i] == -1:
                    color[i] = 1 - color[v]
                    if not dfs(i):
                        return False
                elif color[i] == color[v]:
                    return False
        return True

    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            if not dfs(i):
                return m // 2  # If the graph is not bipartite, return half of the edges rounded down

    # Count the number of vertices in each set
    count = [0, 0]
    for c in color:
        if c != -1:
            count[c] += 1

    # The maximum number of dominoes is the minimum number of edges from vertices in one set to another
    ans = min(count[0] * count[1], m)
    return ans

n, m = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(m)]
print(max_dominoes(n, m, edges))
