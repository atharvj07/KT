import sys
input = sys.stdin.readline

for q in range(int(input())):
    n = int(input())
    s = input()[:-1]
    t = input()[:-1]
    if s==t:
        print("YES")
    else:
        if s[0]!=t[0] or s[-1]!=t[-1]:
            print("NO")
            continue
        f=0
        for x in range(1,n):
            if s[x]!=s[x-1] and s[x]!=t[x] and t[x]!=t[x-1]:
                f=1
        if f: print("YES")
        else: print("NO")
