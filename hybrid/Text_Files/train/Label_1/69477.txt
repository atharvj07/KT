import sys
input = sys.stdin.readline
n,m = map(int,input().split())
d = {}
for i in range(m):
    a,b = map(int,input().split())
    if a not in d: d[a] = set()
    if b not in d: d[b] = set()
    d[a].add(b)
    d[b].add(a)
ans = 0
for i in d:
    for j in d[i]:
        for k in d[i]&d[j]:
            if j in d[k] and k in d[j]:
                ans += 1
print(ans//2)
