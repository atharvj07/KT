#include <iostream>
using namespace std;

int n, k;
string s;

int main() {
  cin >> n >> k >> s;
  for (int i = 0; i < n && k > 0 && s[i] != '1'; ++i) {
    s[i] = '0';
    --k;
  }
  for (int i = n - 1; i >= 0 && k > 0; --i) {
    if (s[i] > '0') {
      s[i] = '0';
      --k;
    }
  }
  cout << s << endl;
  return 0;
}
