import sys

input = sys.stdin.readline
MOD = 998244353

n, m = map(int, input().split())
arr = []
for _ in range(n):
    arr.append(input().strip())

dp = [[0] * m for _ in range(n)]
dp[0][0] = 1
for i in range(n):
    for j in range(m):
        if arr[i][j] == '*':
            continue
        if i - 1 >= 0:
            dp[i][j] = (dp[i][j] + dp[i - 1][j]) % MOD
        if j - 1 >= 0:
            dp[i][j] = (dp[i][j] + dp[i][j - 1]) % MOD

ans = ((dp[n - 1][m - 1]) % MOD) * (2**(n * m - arr.count('*'))) % MOD
print(ans)
