#include <iostream>
using namespace std;

int main() {
    int n; cin >> n;
    int a[n], dp[n]; cin >> a[0]; dp[0] = a[0];
    for (int i = 1; i < n; i++) {
        cin >> a[i];
        dp[i] = dp[i-1] + a[i] - a[i-1];
    }
    int ans = dp[n-1];
    for (int i = 2; i < n-1; i++) {
        ans = min(ans, dp[i-1] + dp[n-1] - dp[i] + a[i-1] - a[i]);
    }
    cout << ans << endl;
}
