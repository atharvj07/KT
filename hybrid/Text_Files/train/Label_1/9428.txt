#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const ll INF=2e9+5;
int main() {
    int t,n;
    cin>>t;
    for(int cs=1;cs<=t;cs++) {
        cin>>n;
        vector<ll>a(n),b(n);
        for(int i=0;i<n;i++) cin>>a[i];
        b[0]=a[0];
        for(int i=1;i<n;i++) {
            b[i]=a[i];
            if(b[i]*b[i-1]<0) b[i]+=b[i-1];
        }
        ll ans=-INF;
        for(ll x:b) ans=max(ans,x);
        cout<<ans<<endl;
    }
}
