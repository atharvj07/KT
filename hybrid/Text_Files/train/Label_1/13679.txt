#include <bits/stdc++.h>
using namespace std;
int main(){
    int n; cin >> n;
    vector<int> adj[n];
    vector<int> v(n);
    for(int i = 0; i < n - 1; i++){
        int a, b; cin >> a >> b;
        a--; b--;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    for(int i = 0; i < n; i++)
        cin >> v[i];
    int ans = 0;
    function<int(int, int)> dfs = [&](int node, int par){
        int sum = v[node];
        for(int to: adj[node]){
            if(to == par) continue;
            sum += dfs(to, node);
        }
        if(sum < 0) ans -= sum, sum = 0;
        else if(sum > 0) ans += sum, sum = 0;
        return sum;
    };
    dfs(0, -1);
    cout << ans << '\n';
    return 0;
}
