import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long[] a = new long[N];
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextLong();
        }
        long xor = 0;
        for (long ai : a) {
            xor ^= ai;
        }
        for (int i = 0; i < N; i++) {
            System.out.print((xor ^ a[i]) + " ");
        }
        System.out.println();
    }
}
