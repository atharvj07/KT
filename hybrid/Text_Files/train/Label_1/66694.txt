import sys
input = sys.stdin.readline
mod = pow(2, 63) - 1

def max_beauty(nums, n, start=0, ans=0, max_num=0):
    if n == 0: return nums[start]
    mx = 0
    for i in range(4):
        cur_max = 0
        for j in range(start+i*n//4, start+(i+1)*n//4):
            cur_max = max(cur_max, nums[j])
        mx = max(mx, cur_max)
    ans = (ans + mx) % mod
    for i in range(4):
        ans = (ans + max_beauty(nums, n-1, start+i*n//4, ans, mx)) % mod
    return ans

n = int(input())
nums = list(map(int, input().split()))
print(max_beauty(nums, n))
