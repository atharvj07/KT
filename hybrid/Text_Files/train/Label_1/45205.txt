import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] s = sc.nextLine().toCharArray();

        int n = s.length;
        int i = 0;
        while (i < n - 1) {
            if (s[i] == 'y' && s[i + 1] == 'x') {
                char temp = s[i];
                s[i] = s[i + 1];
                s[i + 1] = temp;
            } else if (s[i] == 'x' && s[i + 1] == 'y') {
                s[i] = s[i + 1];
                n--;
            } else {
                i++;
            }
        }

        for (i = 0; i < n; i++) {
            System.out.print(s[i]);
        }
        System.out.println();
    }
}
