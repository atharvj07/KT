#include <iostream>
#include <cstdio>
#include <cstring>
#include <set>
#include <vector>
using namespace std;
const int MAXN = 1e3;
const int MAXK = 15;
const int D[4][2] = {{1,0},{-1,0},{0,1},{0,-1}};
int H[MAXN][MAXN];
vector<string> path;
char ans[MAXN];
set<vector<int>> vis;
bool dfs(int n, int k, int x, int y) {
    if (n == 1) {
        ans[n - 1] = 'E';
        return true;
    }
    for (int i = 0; i < 4; ++i) {
        int dx = x + D[i][0] * n;
        int dy = y + D[i][1] * n;
        if (dx <= 0 || dy <= 0 || dx > 2 * k + 1 || dy > 2) continue;
        if (H[dx][dy] == 0) {
            ans[n - 1] = "ESRLUD"[i];
            H[x][y] = H[dx][dy];
            H[dx][dy] = 0;
            bool OK = dfs(n - 1, k, dx, dy);
            H[dx][dy] = H[x][y];
            H[x][y] = 0;
            if (OK) return true;
            ans[n - 1] = 0;
        }
    }
    return false;
}
string convert(int n, int k) {
    string result;
    for (int i = n - 1; i >= 0; i--) result += ans[i];
    return result;
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        int k;
        cin >> k;
        memset(H, 0, sizeof(H));
        path.clear();
        for (int i = 1; i <= 2; i++) {
            for (int j = 1; j <= 2 * k + 1; j++) {
                int x;
                if (i == 1) {
                    cin >> x;
                } else if (i == 2) {
                    char c;
                    cin >> c;
                    if (c == 'E') {
                        x = 0;
                    } else {
                        x = c - '0' + 2 * k + 2;
                    }
                }
                H[j][i] = x;
            }
        }
        bool OK = dfs(4 * k + 1, k, 0, 0);
        if (!OK) {
            cout << "SURGERY FAILED" << endl;
            continue;
        }
        cout << "SURGERY COMPLETE" << endl;
        string res = convert(4 * k + 1, k);
        int last = -1;
        path.push_back("");
        for (int i = 0; i < res.size(); i++) {
            if (res[i] == 'E') {
                last++;
                path.push_back("");
            }
            path[last].push_back(res[i]);
        }
        string rep = "LDRU";
        string tmp = "";
        for (auto &s : path) {
            if (s.size() <= 2) {
                tmp += s;
            } else {
                char c = s[0];
                int cnt = 0;
                for (auto &ch : s) {
                    cnt += (ch == c);
                }
                if (cnt == s.size()) {
                    tmp += c;
                    tmp += to_string(cnt);
                } else {
                    tmp += c;
                    for (auto &ch : s) {
                        if (ch != c) tmp += ch;
                    }
                }
            }
        }
        last = -1;
        path.clear();
        path.push_back("");
        for (int i = 0; i < tmp.size(); i++) {
            if (rep.find(tmp[i]) != string::npos) {
                last++;
                path.push_back("");
            }
            path[last].push_back(tmp[i]);
        }
        set<string> shortcuts;
        for (auto &s : path) {
            if (s.size() >= 3) {
                shortcuts.insert(s);
            }
        }
        for (auto &s : shortcuts) {
            cout << (char)(shortcuts.size() + 64) << " " << s << endl;
        }
        cout << "DONE" << endl;
    }
    return 0;
}
