import sys
n, m = map(int, input().split())
edges = [[] for i in range(n + 1)]
for i in range(m):
    u, v = map(int, input().split())
    edges[u].append(v)
    edges[v].append(u)
vis = [False for i in range(n + 1)]
ans = []
def dfs(node):
    vis[node] = True
    ans.append(node)
    for u in edges[node]:
        if not vis[u]:
            dfs(u)
dfs(1)
print(len(ans) - 1)
for i in range(len(ans) - 1):
    print(ans[i], end = ' ')
