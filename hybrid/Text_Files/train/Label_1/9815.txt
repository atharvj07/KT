#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int n, x, y, a[N], b[N], f[N], g[N], ans;
int main() {
    cin >> n >> x >> y;
    for (int i = 1; i <= n; i++) cin >> a[i];
    b[0] = -1;
    for (int i = 1; i <= n; i++) {
        while (b[f[i - 1]] >= a[i]) f[i] = f[f[i - 1]];
        b[f[i]] = a[i];
    }
    for (int i = n; i > 0; i--) {
        while (g[g[i + 1]] >= a[i]) g[i] = g[g[i + 1]];
        g[g[i]] = a[i];
    }
    for (int i = 1; i <= n; i++) {
        if (f[i - x] >= 0 && g[i + y] >= 0) {
            if (!ans || a[ans] > a[i]) {
                ans = i;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
