#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    unordered_map<int, int> freq;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      int a;
      cin >> a;
      if (freq.find(a) == freq.end()) {
        freq[a] = 1;
      } else {
        freq[a]++;
      }
    }
    for (auto it : freq) {
      if (it.second % 2 != 0) {
        ans += it.second - 1;
      }
    }
    cout << ans / 2 << endl;
  }
  return 0;
}
