#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  for (int &x : a) cin >> x;
  int ans = 0, last = -1;
  for (int i = 0; i < n; i++) {
    if ((a[i] == 1 || a[i] == 3) && a[i] == last) continue;
    if ((a[i] == 2 || a[i] == 3) && i > 0 && a[i] == a[i - 1]) continue;
    ans++;
    last = a[i];
  }
  cout << ans << "\n";
}
