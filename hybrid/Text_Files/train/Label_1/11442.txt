#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, m, deg[N];
int main() {
    cin >> n >> m;
    while (m--) {
        int u, v;
        cin >> u >> v;
        deg[u]++;
        deg[v]++;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (deg[i] > 1) {
            ans += deg[i] / 2 - 1;
        }
    }
    cout << ans << endl;
}
