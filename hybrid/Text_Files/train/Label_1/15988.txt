#include <iostream>
using namespace std;
int main() {
  long long n;
  cin >> n;
  for (long long i = n + 1; i <= 1000000000000000000; i++) {
    int cnt = 0;
    long long tmp = i;
    while (tmp != 0) {
      if (tmp % 10 != 0) cnt++;
      tmp /= 10;
    }
    if (cnt <= 1) {
      cout << i - n;
      return 0;
    }
  }
  return 0;
}
