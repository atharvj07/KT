import java.util.*;
import java.io.*;

class Main {

    private static final int MOD = 10;

    static class Pair {
        int i, j;

        public Pair(int i, int j) {
            this.i = i;
            this.j = j;
        }
    }

    private static int dfs(char[][] grid, int i, int j, int size) {
        if (i < 0 || i >= grid.length || j < 0 || j >= grid[0].length || grid[i][j] != '.') {
            return 0;
        }

        grid[i][j] = (char) ((size % MOD) + '0');
        return size + dfs(grid, i - 1, j, size) + dfs(grid, i + 1, j, size) + dfs(grid, i, j - 1, size) + dfs(grid, i, j + 1, size);
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String row = br.readLine();
            for (int j = 0; j < m; j++) {
                grid[i][j] = row.charAt(j);
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '*') {
                    grid[i][j] = '.';
                    dfs(grid, i, j, 1);
                    grid[i][j] = '*';
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                sb.append(grid[i][j]);
            }
        }
        System.out.println(sb);
    }
}
