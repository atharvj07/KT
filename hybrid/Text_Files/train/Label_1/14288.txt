#include<bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;
int n, k, ans[N], call = 0;
vector<int>a;

void merge(int l, int m, int r) {
  vector<int>L, R;
  for (int i = l; i < m; ++i) L.emplace_back(a[i]);
  for (int i = m; i < r; ++i) R.emplace_back(a[i]);
  L.emplace_back(2e9);
  R.emplace_back(2e9);
  int p1 = 0, p2 = 0;
  for (int i = l; i < r; ++i) {
    if (L[p1] < R[p2]) {
      a[i] = L[p1];
      p1++;
    }
    else {
      a[i] = R[p2];
      p2++;
      call++;
    }
  }
}

void mergesort(int l, int r) {
  if (l >= r) return;
  if (r - l == 1) {
    call++;
    return;
  }
  int m = (l + r) / 2;
  mergesort(l, m);
  mergesort(m, r);
  merge(l, m, r);
}

int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; ++i) a.push_back(i);
  mergesort(0, n);
  if (call < k) { cout << -1 << endl; return 0; }
  for (int i = 0; i < n; ++i) {
    cout << a[i] << ' ';
  }
  return 0;
}
