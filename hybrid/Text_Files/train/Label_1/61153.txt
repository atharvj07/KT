from sys import stdin

for _ in range(int(stdin.readline())):
    n, p, k = map(int, stdin.readline().split())
    a = sorted([int(x) for x in stdin.readline().split()])
    i = 0
    cnt = 0
    while p >= a[0] and i < n:
        if p >= a[i]:
            p -= a[i]
            i += 1
            cnt += 1
        else:
            if i + k <= n and p >= a[i + k - 1]:
                p -= a[i + k - 1]
                i += k
                cnt += k
            else:
                break
    print(cnt)
