import java.util.Scanner;

public class Main {
    static class Domino implements Comparable<Domino> {
        int x, y;

        Domino(int x, int y) {
            this.x = x;
            this.y = y;
        }

        public int compareTo(Domino o) {
            return this.x - o.x;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Domino[] dominoes = new Domino[n];
        for (int i = 0; i < n; i++) {
            dominoes[i] = new Domino(sc.nextInt(), sc.nextInt());
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt() - 1;
            int y = sc.nextInt() - 1;
            int cost = 0;
            for (int j = x + 1; j < y; j++) {
                if (dominoes[j].x < dominoes[x].x + dominoes[j].y) {
                    cost++;
                }
            }
            System.out.println(cost);
        }
    }
}
