#include<cstdio>
#include<cstring>
#include<vector>
#define mid (l + r >> 1)
const int N = 1e5 + 10;

int n, da, db, a, b, par[N], dist[N], ans[N], up[N][18], dp[N][18];
std::vector<int> g[N];

void dfs(int u, int p = 0) {
    up[u][0] = p;
    for(int i = 1; i < 18; i++) up[u][i] = up[up[u][i - 1]][i - 1];
    for(int v : g[u]) if(v != p) {
        par[v] = u; dist[v] = dist[u] + 1; dfs(v, u);
    }
}

int lca(int u, int v) {
    if(dist[u] > dist[v]) std::swap(u, v);
    int diff = dist[v] - dist[u];
    for(int i = 0; i < 18; i++) if((diff >> i) & 1) v = up[v][i];
    if(u == v) return u;
    for(int i = 17; i >= 0; i--) if(up[u][i] != up[v][i]) {
        u = up[u][i]; v = up[v][i];
    }
    return par[u];
}

void pre() {
    for(int i = 0; i < 18; i++) {
        dp[0][i] = -1;
        for(int u = 1; u <= n; u++) {
            dp[u][i] = up[u][i];
            if(i) dp[u][i] = dp[dp[u][i]][i - 1];
        }
    }
}

int jump(int s, int t) {
    if(s == t) return 0; if(s == -1) return dist[t];
    int l = lca(s, t), d1 = dist[s] - dist[l], d2 = dist[t] - dist[l];
    if(d1 <= da && d2 <= db) return 1;
    if(d1 <= da) return 1 + dp[s][0];
    if(d2 <= db) return dp[t][0] + 1;
    for(int i = 17; i >= 0; i--) {
        if(jump(s, dp[t][i]) + dp[t][i] + 1 <= da + 1) {
            s = dp[t][i];
        }
        if(jump(dp[s][i], t) + 1 + dp[s][i] <= db + 1) {
            t = dp[s][i];
        }
    }
    return dist[s] + dist[t] - 2 * dist[l] + 2;
}

int main() {
    int tc, cs = 0; scanf("%d", &tc);
    while(tc--) {
        for(int u = 0; u <= n; u++) g[u].clear();
        memset(up, 0, sizeof up); memset(par, 0, sizeof par);
        memset(dp, 0, sizeof dp); memset(ans, 0, sizeof ans);
        scanf("%d %d %d %d %d", &n, &a, &b, &da, &db);
        for(int i = 1; i < n; i++) {
            int a, b;
            scanf("%d %d", &a, &b);
            g[a].push_back(b);
            g[b].push_back(a);
        }
        dfs(1); pre();
        ans[a] = jump(a, b);
        ans[b] = jump(b, a);
        puts((ans[a] <= ans[b] && ans[a] <= da) ? "Alice" : "Bob");
    }
    return 0;
}
