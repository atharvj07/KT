#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> l(n);
  for (int i = 0; i < n; i++) cin >> l[i];
  sort(l.begin(), l.end());
  long long sum = 0;
  for (int i = 1; i < n - 1; i++) sum += l[i];
  if (sum > l[n - 1]) {
    cout << 1 << endl;
  } else {
    cout << l[n - 1] - sum + 1 << endl;
  }
  return 0;
}
