n = int(input())
p = list(map(int, input().split()))

# Calculate initial positions of each number
positions = [0] * n
for i in range(n):
    positions[p[i] - 1] = i

# Store the number of moves required to get subsegment with values 1, 2, ..., k
moves = [0] * n

for k in range(1, n + 1):
    # Calculate moves for current k
    current_positions = positions[:k]
    current_positions.sort()
    median_pos = current_positions[k // 2]
    moves[k - 1] = sum(abs(pos - median_pos) for pos in current_positions) - sum(range(k // 2)) - sum(range((k + 1) // 2))
    
print(*moves)
