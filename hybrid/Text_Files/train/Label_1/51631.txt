import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), p = input.nextInt();
        char[] s = input.next().toCharArray();
        boolean found = false;
        for (int i = n - 1; i >= 0; i--) {
            for (char j = (char) (s[i] + 1); j <= (char) ('a' + p - 1); j++) {
                if (isTolerable(s, i, j)) {
                    s[i] = j;
                    found = true;
                    break;
                }
            }
            if (found) {
                break;
            }
        }
        if (found) {
            System.out.println(s);
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isTolerable(char[] s, int i, char c) {
        return (i == 0 || s[i - 1] != c) && (i == s.length - 1 || s[i + 1] != c);
    }
}
