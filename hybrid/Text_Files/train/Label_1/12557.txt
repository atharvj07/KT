#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int n, v, p[110];
    cin >> n >> v;
    for (int i = 1; i <= n; i++) { cin >> p[i]; }
    int cur = 0, ans = 0;
    for (int i = 1; i <= n; i++) {
        if (cur + v >= i) {
            ans += (i - cur) * p[i];
            cur = i;
        } else
            break;
    }
    if (cur < n) ans += -1;
    cout << ans << endl;
}
