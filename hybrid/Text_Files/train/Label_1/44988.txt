import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), h = sc.nextInt();
        int[] ai = new int[n];
        for (int i = 0; i < n; i++) ai[i] = sc.nextInt();
        System.out.println(Arrays.stream(ai).map(x -> x <= h ? 1 : 2).sum());
    }
}
