import sys,math

n,k=map(int,input().split())
a=[list(map(int,sys.stdin.readline().split())) for _ in range(n)]
from fractions import Fraction as fr

def area(x):
    return fr(x[2]-x[0])*(x[3]-x[1])

a.sort(key=lambda x:x[2]-x[0])
u=a[0];a=a[1:]+[u];z={}

for i in range(n):
    for j in range(i+1,n):
        if i==j:continue
        X=fr(a[i][2]-a[j][2])/(a[i][2]-a[j][0])
        dX=fr(a[i][0]-a[j][0])/(a[i][2]-a[j][0])
        Y=fr(a[i][3]-a[j][3])/(a[i][3]-a[j][1])
        dY=fr(a[i][1]-a[j][1])/(a[i][3]-a[j][1])
        if (X,Y) in z:z[(X,Y)].append((i,j,dX,dY))
        else:z[(X,Y)]=[(i,j,dX,dY)]

if k==0:
    ans=area(a[0])
    for X,Y in z:
        u=z[(X,Y)][0]
        ans=min(ans,area(a[u[0]])+area(a[u[1]])-fr(a[u[0]][2]-a[u[0]][0])/fr(a[u[1]][2]-a[u[1]][0])*(area(a[u[1]])-area(a[u[0]])))

if k==1:
    ans=1e20
    for X,Y in z:
        u=z[(X,Y)]
        ans=min(ans,min(area(a[x[0]])+area(a[x[1]])-fr(a[x[0]][2]-a[x[0]][0])/fr(a[x[1]][2]-a[x[1]][0])*(area(a[x[1]])-area(a[x[0]])) for x in u))
    for i in a:
        ans=min(ans,area(i))

if n-k!=1:
    ans=1e20
    for X,Y in z:
        u=z[(X,Y)]
        ans=min(ans,min(area(a[x[0]])+area(a[x[1]]) for x in u if x[2]!=0 or x[3]!=0))

print(ans)
