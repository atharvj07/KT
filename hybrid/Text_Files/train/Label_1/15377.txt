#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> ii;
#define int long long

const int maxn = 31, maxk = 31;

int32_t main() {
  ios::sync_with_stdio(0); cin.tie(0);
  int t; cin >> t;
  while (t--) {
    int n, k; cin >> n >> k;
    int a[maxn], b[maxn];
    for (int i = 1; i <= n; i++) cin >> a[i], b[i] = a[i];
    sort(b + 1, b + 1 + n);
    for (int i = 1; i <= min(k, n); i++) {
      for (int j = 1; j <= n; j++) {
        if (a[j] < b[n - i + 1]) {
          a[j] = b[n - i + 1];
        }
      }
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) ans += a[i];
    cout << ans << '\n';
  }
  return 0;
}
