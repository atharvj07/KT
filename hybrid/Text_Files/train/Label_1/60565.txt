mod = int(1e9) + 7
n, q = map(int, input().split())
s = list(input())
x = [0] * n
for i in range(n):
    x[i] = int(s[i])
pre = [0] * n
for i in range(n):
    pre[i] = x[i] if i == 0 else pre[i - 1] + x[i]
for t in range(q):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    ans = pre[r]
    if l > 0: ans -= pre[l - 1]
    print(ans % mod)
