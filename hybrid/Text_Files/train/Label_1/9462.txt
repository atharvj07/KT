#include <iostream>
#include <vector>

using namespace std;

int main() {
    string a, b;
    cin >> a >> b;

    int n = a.size(), m = b.size(), ans = 0;
    vector<int> cnt(n);
    for (int i = 0; i < n - m + 1; i++) {
        int dif = 0;
        for (int j = i; j < i + m; j++) {
            dif += (a[j] != b[j - i]);
        }
        cnt[dif]++;
    }
    for (int i = 0; i < n; i += 2) {
        ans += cnt[i];
    }
    cout << ans << endl;
}
