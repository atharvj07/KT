#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;

  vector<vector<int>> segments(n, vector<int>(7, 0));
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < 7; j++) {
      segments[i][j] = s[j] - '0';
    }
  }

  int max_num = -1;
  int cnt = k;
  for (int i = 0; i < n; i++) {
    int digit = -1;
    for (int j = 0; j < 7; j++) {
      if (segments[i][j] == 0 && cnt > 0) {
        cnt--;
        if (digit == -1) digit = 0;
        digit |= (1 << j);
      }
    }
    if (digit == -1) break;
    max_num = max_num * 10 + digit;
  }

  if (cnt == 0) {
    cout << max_num << endl;
  } else {
    cout << -1 << endl;
  }

  return 0;
}
