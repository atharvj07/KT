for t in range(int(input())):
    n = int(input())
    s = input()
    b = []
    for i in range(n):
        if s[i] != '?':
            b.append(s[i])
    for i in range(n):
        if s[i] == '?':
            if len(b) == 0:
                b.append('R')
            elif b[-1] == 'B':
                b.append('R')
            else:
                b.append('B')
    print(''.join(b))
