#include<bits/stdc++.h>
using namespace std;
const int N = 510;
char s[N];
int n, f[N][N][2], g[N][N][2];
int main() {
    cin >> n;
    cin >> s + 1;
    for (int i = n; i; --i) {
        for (int j = i + 1; j <= n; ++j) {
            for (int k = 0; k < 2; ++k) {
                int &x = f[i][j][k];
                for (int l = i; l < j; ++l) {
                    for (int m = j + 1; m <= n; ++m) {
                        if (k == 0) x = max(x, max(f[i][l][1] + f[m][j][0], f[i][l][0] + f[m][j][1]));
                        else x = max(x, max(g[i][l][1] + g[m][j][0], g[i][l][0] + g[m][j][1]));
                    }
                }
                if (s[i] == ')' || k == 0) x = max(x, 1 + f[i + 1][j][0]);
                if (s[j] == '(' || k == 0) x = max(x, 1 + f[i][j - 1][0]);
                if (s[i] == '(' || k == 1) x = max(x, 1 + g[i + 1][j][1]);
                if (s[j] == ')' || k == 1) x = max(x, 1 + g[i][j - 1][1]);
            }
        }
    }
    cout << f[1][n][0] << endl;
    int l = 1, r = n;
    for (int i = 1; i <= n; ++i) {
        for (int j = i + 1; j <= n; ++j) {
            if (f[i][j][0] == f[1][n][0]) {
                l = i, r = j;
                break;
            }
            if (f[i][j][1] == f[1][n][0]) {
                l = i, r = j;
                break;
            }
        }
    }
    cout << l << " " << r << endl;
}
