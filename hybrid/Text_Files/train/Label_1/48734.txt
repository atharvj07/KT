import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }

        boolean found = false;
        for (int d = -1000; d <= 1000; d++) {
            if (d == 0) {
                continue;
            }
            int positiveCount = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] * d > 0) {
                    positiveCount++;
                }
            }
            if (positiveCount >= (n + 1) / 2) {
                System.out.println(d);
                found = true;
                break;
            }
        }

        if (!found) {
            System.out.println(0);
        }
    }
}
