n, b, a = map(int, input().split())
s = list(map(int, input().split()))
cur_batt, cur_acc = b, a
ans = 0
for light in s:
    if cur_batt > 0:
        ans += 1
        cur_batt -= 1
        if light: cur_acc = min(cur_acc + 1, a)
    elif cur_acc > 0:
        ans += 1
        cur_acc -= 1
    else: break
print(ans)
