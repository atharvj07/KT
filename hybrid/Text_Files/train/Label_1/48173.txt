import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long d = in.nextLong();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        long ans = 0;
        long cur = 0;
        for (long b : a) {
            cur += b;
            if (cur < 0) {
                cur = 0;
                ans++;
            } else if (cur > d) {
                System.out.println(-1);
                return;
            }
        }
        System.out.println(ans);
    }
}
