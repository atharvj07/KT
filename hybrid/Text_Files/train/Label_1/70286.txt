from collections import defaultdict
from fractions import Fraction
import sys
input = sys.stdin.readline

n = int(input())
p = [0] + list(map(int, input().split()))
children = defaultdict(list)
for i in range(2, n + 1):
    children[p[i]].append(i)

dp = [0] * (n + 1)
dp[1] = 1
for i in range(2, n + 1):
    s = Fraction(0)
    for j in children[i]:
        s += dp[j]
    dp[i] = s + 1

print(*dp[1:], sep=" ")
