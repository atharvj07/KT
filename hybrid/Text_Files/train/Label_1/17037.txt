#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    string ans(n, '0');
    for (int i = 0; i < n; i++) {
      if (s[2 * i] == s[2 * i + 1])
        ans[i] = s[2 * i];
    }
    for (int i = 0; i < n; i++) {
      if (ans[i] == '0') ans[i] = s[2 * i];
    }
    cout << ans << endl;
  }
}
