t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    b = sorted(a)
    if a == b:
        print("YES")
    else:
        can_sort = True
        for i in range(n):
            if a[i] != b[i] and (i < x and (n - i - 1) < x):
                can_sort = False
                break
        print("YES" if can_sort else "NO")
