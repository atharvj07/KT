import sys
n = int(sys.stdin.readline())
mines = {(int(x), int(y)): True for x, y in (sys.stdin.readline().split() for i in range(n))}
xmax, ymax = max(x for x, y in mines), max(y for x, y in mines)
xmin, ymin = min(x for x, y in mines), min(y for x, y in mines)
print((max(xmax-xmin, ymax-ymin) + 1)**2)
