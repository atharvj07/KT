#include <iostream>
#include <vector>
using namespace std;

typedef unsigned long long ull;
const ull B = 1000000000000000000;

int main() {
    ull p, x, cur = 1, ans = 0;
    cin >> p >> x;
    for (ull i = (ull)(p - 1); i > 0; i--) {
        ull f = 0;
        for (ull j = 0; j <= 9; j++) {
            if (cur * (j + 1) % x == 0) {
                f = 1;
                ans = ans * 10 + j;
                cur = cur * (j + 1) / x;
                break;
            }
        }
        if (!f) {
            cout << "Impossible" << endl;
            return 0;
        }
    }
    cout << ans << endl;
    return 0;
}
