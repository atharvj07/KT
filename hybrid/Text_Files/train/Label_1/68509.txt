t = int(input().strip())
for _ in range(t):
    n, p = map(int, input().strip().split())
    edges = 2*n + p
    
    for i in range(1, n):
        for j in range(i+1, n+1):
            if edges == 0:
                break
            print(i, j)
            edges -= 1
        if edges == 0:
            break
