#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    unordered_set<int> s;
    for (int i = 0; i < n; i++) s.insert(a[i]);
    int m = s.size();
    vector<int> p(m, -1);
    int ans = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (p[a[i] - 1] != -1) ans = max(ans, p[a[i] - 1] + 1);
        p[a[i]] = i;
    }
    cout << n - ans << endl;
    return 0;
}
