#include<bits/stdc++.h>

#define ll long long
#define ull unsigned long long
#define ld long double
#define pb push_back
#define ff first
#define ss second
#define mp make_pair
#define f(i,n) for(int i=0;i<n;i++)
#define all(a) a.begin(),a.end()
using namespace std;
typedef vector<int> vi;
typedef pair<int,int> pi;
const ll mod=1e9+7;
ll pw(ll x,ll n){
    ll ans=1;
    while(n){
        if(n%2)ans=(ans*x)%mod;
        x=(x*x)%mod;
        n/=2;
    }
    return ans;
}
const ll N=2e5+10;
ll dp[N][2][2],n;
pi a[N];
ll fun(ll i,bool last,bool added){
    if(i==n){
        if(last)return 1;
        else return 0;
    }
    ll &x=dp[i][last][added];
    if(x!=-1)return x;
    x=0;
    if(!added){
        x=(x+fun(i+1,last,added))%mod;
    }
    if(!last){
        x=(x+fun(i+1,1,added))%mod;
        if(a[i].ff<=a[i].ss){
            x=(x+fun(i+1,last,1))%mod;
        }
    }
    return x;
}
void solve(){
    memset(dp,-1,sizeof(dp));
    cin>>n;
    f(i,n)cin>>a[i].ff>>a[i].ss;
    cout<<fun(0,0,0)%mod<<endl;
}
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    solve();
    return 0;
}
