import re

n = int(input())
for i in range(n):
    s = input()
    s = re.sub("::", "0000:0000:0000:0000:0000:0000:0000:0000:", s, count=1)
    blocks = s.split(':')
    for i in range(len(blocks)):
        if not blocks[i]:
            blocks[i] = '0000'
        if len(blocks[i]) == 1:
            blocks[i] = '000' + blocks[i]
        elif len(blocks[i]) == 2:
            blocks[i] = '00' + blocks[i]
        elif len(blocks[i]) == 3:
            blocks[i] = '0' + blocks[i]
    print(':'.join(blocks))
