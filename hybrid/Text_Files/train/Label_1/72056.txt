t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    s = input()
    operations = []
    for i in range(k-1):
        # Find the first ')' to make a regular prefix
        p_close = s.find(')', i*2)
        if p_close != i*2:
            s = s[:i*2] + ')' + s[i*2:p_close] + s[p_close+1:]
            operations.append((i*2+1, p_close+1))

        # Find the first '(' to complete the pair
        p_open = s.find('(', i*2+1)
        if p_open != i*2+1:
            s = s[:i*2+1] + '(' + s[i*2+1:p_open] + s[p_open+1:]
            operations.append((i*2+2, p_open+1))

    # Adjust the rest of the string to be a regular sequence
    for i in range(n//2-k+1):
        if s[2*k-2+i] == ')':
            p_open = s.find('(', 2*k-2+i)
            s = s[:2*k-2+i] + '(' + s[2*k-2+i:p_open] + s[p_open+1:]
            operations.append((2*k-1+i, p_open+1))

    print(len(operations))
    for op in operations:
        print(*op)
