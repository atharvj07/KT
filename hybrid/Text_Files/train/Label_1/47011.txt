import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int x = in.nextInt();
            int y = in.nextInt();
            int z = in.nextInt();
            if (x == y && y == z)
                System.out.println("YES\n" + x + " " + y + " " + z);
            else if (x >= y && x >= z) {
                if (z == y)
                    System.out.println("YES\n" + x + " " + y + " " + z);
                else
                    System.out.println("NO");
            } else if (y >= x && y >= z) {
                if (z == x)
                    System.out.println("YES\n" + x + " " + y + " " + z);
                else
                    System.out.println("NO");
            } else if (z >= x && z >= y) {
                if (y == x)
                    System.out.println("YES\n" + x + " " + y + " " + z);
                else
                    System.out.println("NO");
            }
        }
    }
}
