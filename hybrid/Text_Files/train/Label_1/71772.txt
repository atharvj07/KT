import sys

input = sys.stdin.readline

MOD = int(1e9) + 7

n, k = map(int, input().split())
a = list(map(int, input().split()))
ans = 0
if a[0] % 2 == 1: ans += k * 2
else: ans += 1
if a[n] % 2 == 1: ans -= k * 2
else: ans -= 1
for i in range(1, n):
    if a[i] % 2 == 1: ans += k * 4
    else: ans += 2
print(ans % MOD)
