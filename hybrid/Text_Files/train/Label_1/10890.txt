#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct item {
    int idx;
    int cost;
    bool operator < (const item& rhs) const {
        return cost < rhs.cost;
    }
};

int main() {
    int n, s, a;
    cin >> n >> s;
    vector<int> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    sort(arr.begin(), arr.end());
    vector<item> items;
    for (int i = 0; i < n; i++) {
        items.push_back({i, arr[i] + (i + 1) * 1});
    }
    sort(items.begin(), items.end());
    int sum = 0, cnt = 0, ans = n;
    for (int i = 0; i < n; i++) {
        if (sum + items[i].cost > s) {
            ans = cnt;
            break;
        }
        sum += items[i].cost;
        cnt++;
    }
    cout << ans << ' ' << sum << endl;
    return 0;
}
