#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 1e9 + 7;
const int maxn = 100005;
int n, x, d[maxn], dp[maxn][105];
int dfs(int u, int dist) {
  if (~dp[u][dist]) return dp[u][dist];
  dp[u][dist] = 1;
  for (int i = 1; i <= n; i++)
    if (dist + d[i] <= x) dp[u][dist] = (dp[u][dist] + dfs(i, dist + d[i])) % mod;
  return dp[u][dist];
}
int32_t main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cin >> n >> x;
  for (int i = 1; i <= n; i++) cin >> d[i];
  memset(dp, -1, sizeof dp);
  cout << dfs(0, 0) << endl;
}
