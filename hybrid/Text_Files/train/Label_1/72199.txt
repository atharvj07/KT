import sys
N = int(sys.stdin.readline())
W = list(map(int, sys.stdin.readline().split()))
P = [0] * N
G = [[] for _ in range(N)]
for i in range(N - 1):
    A, B, C = map(int, sys.stdin.readline().split())
    A -= 1
    B -= 1
    G[A].append((B, C))
    G[B].append((A, C))

def dfs(v, p):
    for u, w in G[v]:
        if u != p:
            dfs(u, v)
            P[v] = max(P[v], W[u] + P[u] - w)
for i in range(N):
    dfs(i, -1)
    W[i] += P[i]
print(max(W))
