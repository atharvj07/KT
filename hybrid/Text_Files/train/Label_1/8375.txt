#include <bits/stdc++.h>

using namespace std;

vector<int> adj[101];
bool vis[101], mark[101];
int dis[101], ans=0, df[101];

struct node
{
    int x,a,f;
};

void bfs(int src) {
    vis[src]=1;
    queue<node> q;
    q.push({src,0,0});
    while(!q.empty()) {
        node p=q.front();
        q.pop();
        for(int v: adj[p.x]) {
            if(!vis[v]) {
                vis[v]=1;
                dis[v]=p.f+1;
                q.push({v,p.f+1,0});
            }
        }
    }
}

int main()
{
    int n,m,s,b,k,h,x,a,f,d;
    cin>>n>>m;
    for(int i=1;i<=m;i++) {
        cin>>x>>a;
        adj[x].push_back(a);
        adj[a].push_back(x);
    }
    cin>>s>>b>>k>>h;
    for(int i=1;i<=s;i++) {
        cin>>x>>a>>f;
        if(!mark[x]) {
            df[x]++;
            mark[x]=1;
        }
    }
    for(int i=1;i<=b;i++) {
        cin>>x>>d;
        ans+=k;
    }
    bfs(1);
    for(int i=1;i<=n;i++) {
        if(df[i]>0) {
            for(int j=1;j<=n;j++) {
                if(dis[j] && mark[j]) {
                    if(a>=df[j] && f>=dis[j]) {
                        ans-=k;
                        break;
                    }
                }
            }
        }
    }
    while(b-df[1]) {
        ans+=h;
        df[1]++;
    }
    cout<<ans<<endl;
}
