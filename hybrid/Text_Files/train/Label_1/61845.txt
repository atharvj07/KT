n = int(input())
a = list(map(int, input().split()))

if n <= 2:
    print(0)
else:
    dp = [0] * n
    dp[1] = 0
    for i in range(2, n):
        dp[i] = min(a[i - 1], a[i - 2]) + dp[i - 2]
        if i > 2:
            dp[i] = max(dp[i], min(a[i - 1], a[i - 3]) + dp[i - 3])
    print(max(dp[-1], dp[-2]))
