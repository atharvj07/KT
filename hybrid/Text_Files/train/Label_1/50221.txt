import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class DimaAndInna {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }

        List<Integer> w = new ArrayList<>();
        int k = 0;

        for (int i = 0; i < n; i++) {
            int event = scanner.nextInt();
            if (event == -1) {
                while (k < m && a[k] <= w.size()) {
                    w.remove(w.size() - a[k]);
                    k++;
                }
            } else {
                w.add(event);
            }
        }

        if (w.isEmpty()) {
            System.out.println("Poor stack!");
        } else {
            for (int i : w) {
                System.out.print(i);
            }
        }

        scanner.close();
    }
}
