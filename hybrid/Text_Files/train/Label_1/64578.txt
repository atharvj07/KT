def find_xor_sum_indices(basket, x):
    n = len(basket)
    dp = [None for _ in range(1 << n)]
    dp[0] = []
    for mask in range(1, 1 << n):
        for j in range(n):
            if mask & (1 << j):
                prev_mask = mask ^ (1 << j)
                if dp[prev_mask] is not None:
                    dp[mask] = dp[prev_mask] + [j]
                    break
    for mask in range(1, 1 << n):
        if dp[mask] is not None:
            xor_sum = 0
            for idx in dp[mask]:
                xor_sum ^= basket[idx]
            if xor_sum == x:
                return dp[mask]
    return None

m = int(input().strip())
basket = []
for _ in range(m):
    x = int(input().strip())
    indices = find_xor_sum_indices(basket, x)
    if indices is None:
        print(0)
    else:
        print(len(indices), *indices)
    basket.append(x)
