from sys import stdin
mod = 10**9 + 7

n, m = map(int, stdin.readline().split())
board = [list(stdin.readline().strip()) for i in range(n)]
dp = [[0]*m for i in range(n)]
for i in range(n-1,-1,-1):
    for j in range(m-1,-1,-1):
        if board[i][j]=='.':
            if i+1<n and j+1<m: dp[i][j]=(dp[i][j+1]+dp[i+1][j])%mod
            elif i+1<n: dp[i][j]=dp[i+1][j]
            else: dp[i][j]=dp[i][j+1]
        else:
            if i+1<n and board[i+1][j]=='.':
                dp[i][j]=dp[i+1][j]
            elif j+1<m and board[i][j+1]=='.':
                dp[i][j]=dp[i][j+1]
            else:
                dp[i][j]=0
print(dp[0][0])
