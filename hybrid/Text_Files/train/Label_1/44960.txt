import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int A = in.nextInt();
        int B = in.nextInt();
        int max = Math.max(A, B);
        int len = 1;
        while (max >= 10) {
            max /= 10;
            len++;
        }
        System.out.println(len);
    }
}
