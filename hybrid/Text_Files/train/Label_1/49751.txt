import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String pw = sc.nextLine().toLowerCase();
        int n = sc.nextInt();
        boolean[] exist = new boolean[26 * 26];
        for (int i = 0; i < n; i++) {
            String word = sc.next().toLowerCase();
            int idx = (word.charAt(0) - 'a') * 26 + (word.charAt(1) - 'a');
            exist[idx] = true;
        }
        int sidx = (pw.charAt(0) - 'a') * 26 + (pw.charAt(1) - 'a');
        for (int i = 0; i < 26 * 26; i++) {
            if (exist[i] && exist[sidx]) System.out.println("YES");
        }
        System.out.println("NO");
    }
}
