#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_map<int, int> m;
  vector<int>v(n);
  for (int i = 0; i < n; ++i) {
    cin >> v[i];
    ++m[v[i]];
  }
  int ans = INT_MAX;
  for (auto& it : m) {
    if (it.second > 1)
      ans = min(ans, min(it.second, n - it.second));
  }
  cout << (ans == INT_MAX ? 0 : ans);
  return 0;
}
