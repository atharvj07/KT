import java.util.Arrays;
import java.util.Scanner;

public class VovaAndTests {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int p = scanner.nextInt();
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int[] marks = new int[n];
        for (int i = 0; i < k; i++) {
            marks[i] = scanner.nextInt();
        }

        Arrays.sort(marks, 0, k);

        int remainingMarks = n - k;
        int remainingSum = x - Arrays.stream(marks, 0, k).sum();
        int remainingMedian = y - marks[k / 2];

        if (remainingSum < 0 || remainingMedian < 0) {
            System.out.println("-1");
            return;
        }

        int[] remainingMarksArray = new int[remainingMarks];
        for (int i = 0; i < remainingMarks; i++) {
            remainingMarksArray[i] = Math.min(remainingSum / remainingMarks, remainingMedian);
            remainingSum -= remainingMarksArray[i];
            remainingMedian--;
        }

        for (int mark : remainingMarksArray) {
            System.out.print(mark + " ");
        }
    }
}
