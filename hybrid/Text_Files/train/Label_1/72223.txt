n = int(input())
a = list(input())
ans = []
for i in range(n):
    if a[i] == '*':
        l, r = i, i
        while l >= 0 and a[l] == '*':
            l -= 1
        while r < n and a[r] == '*':
            r += 1
        if l < 0:
            ans.append(r)
        else:
            ans.append((r - l + 1) // 2)
print(max(ans))
