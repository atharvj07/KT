#include <bits/stdc++.h>
using namespace std;
int main() {
  int t; cin >> t;
  while (t--) {
    string s; cin >> s;
    int l = s.size();
    int ans = 0;
    for (int i = 0; i < l; i++) {
      if (s[i] - '0' > ans) ans = s[i] - '0';
      else if (s[i] - '0' == ans) ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
