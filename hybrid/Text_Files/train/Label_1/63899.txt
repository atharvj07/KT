import sys

def rotate(grid):
  new_grid = [['', '', '', ''], ['', '', '', '']]
  for i in range(len(grid)):
    for j in range(len(grid)):
      new_grid[j][len(grid) - i - 1] = grid[i][j]
  return new_grid

def same(grid1, grid2):
  for i in range(len(grid1)):
    for j in range(len(grid1)):
      if grid1[i][j] != grid2[i][j]:
        return False
  return True

def bfs(grid1, grid2, q):
  while len(q) > 0:
    grid, num_moves = q.pop(0)
    if same(grid, grid2):
      return num_moves
    # slide up
    new_grid = rotate(rotate(grid))
    if grid[0][0] == 'X':
      q.append((new_grid, num_moves + 1))
    # slide left
    new_grid = rotate(grid)
    if grid[0][1] == 'X':
      q.append((new_grid, num_moves + 1))
    # slide down
    if grid[1][0] == 'X':
      q.append((grid, num_moves + 1))
    # slide right
    if grid[0][2] == 'X':
      q.append((rotate(rotate(rotate(grid))), num_moves + 1))
  return -1

grid1 = [list(sys.stdin.readline().strip()) for i in range(2)]
grid2 = [list(sys.stdin.readline().strip()) for i in range(2)]
q = [(grid1, 0)]
result = bfs(grid1, grid2, q)
if result == -1:
  print('NO')
else:
  print('YES')
