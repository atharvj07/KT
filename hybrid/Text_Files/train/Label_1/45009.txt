import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] T = new int[N];
        for (int i = 0; i < N; i++) {
            T[i] = sc.nextInt();
        }
        int M = sc.nextInt();
        int[][] P = new int[M][2];
        for (int i = 0; i < M; i++) {
            for (int j = 0; j < 2; j++) {
                P[i][j] = sc.nextInt();
            }
        }
        for (int[] p : P) {
            int sum = 0;
            for (int i = 0; i < N; i++) {
                sum += (i + 1 == p[0]) ? p[1] : T[i];
            }
            System.out.println(sum);
        }
    }
}
