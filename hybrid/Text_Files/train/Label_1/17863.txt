#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e6 + 10;
long long a[MAXN];
int main() {
    long long v, temp;
    cin >> v >> temp;
    for (int i = 1; i < 10; ++i) {
        cin >> a[i];
        if (a[i] > temp) {
            cout << -1 << endl;
            return 0;
        }
        temp -= a[i];
    }
    long long res = 0;
    while (temp >= a[9]) {
        res = res * 10 + 9;
        temp -= a[9];
    }
    for (int i = 8; i >= 1; i--) {
        while (temp >= a[i]) {
            res = res * 10 + i;
            temp -= a[i];
        }
    }
    cout << res << endl;
    return 0;
}
