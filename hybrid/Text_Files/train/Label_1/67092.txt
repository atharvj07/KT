import sys
import numpy as np

s = sys.stdin.readline().strip()
t = sys.stdin.readline().strip()

cnt = np.zeros(len(t), dtype=int)
for i in range(len(s)):
    cnt[i%len(t)] += s[i]=='1'

ans = s
for i in range(len(t)):
    best = -1
    idx = -1
    for j in range(len(s)):
        if s[j]=='0' and cnt[(j+i)%len(t)]<len(t):
            if best < cnt[(j+i+1)%len(t)]:
                best = cnt[(j+i+1)%len(t)]
                idx = j
    if best!=-1:
        ans = ans[:idx]+'1'+ans[idx+1:]
        cnt[(idx+i)%len(t)] += 1

print(ans)
