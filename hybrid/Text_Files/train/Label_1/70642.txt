def max_value_d(n, k, heights):
    def is_valid(d):
        cuts = 0
        for height in heights:
            cuts += (height - 1) // d
        return cuts <= k

    left, right = 1, max(heights)
    while left < right:
        mid = (left + right + 1) // 2
        if is_valid(mid):
            left = mid
        else:
            right = mid - 1
    return left

n, k = map(int, input().split())
heights = list(map(int, input().split()))
print(max_value_d(n, k, heights))
