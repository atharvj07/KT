#include <bits/stdc++.h>

using namespace std;

int main() {
    long long n, k;
    cin >> n >> k;
    if (n % k) {
        cout << -1 << endl;
        return 0;
    }
    long long g = n / k;
    for (long long i = 0; i < k - 1; i++) {
        cout << i * g + 1 << " ";
    }
    cout << n - (k - 1) * g << endl;
    return 0;
}
