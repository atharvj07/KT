import sys
s, t = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(s)]
b = [[0] * t for _ in range(s)]
flag = False
for i in range(s):
    for j in range(t):
        if a[i][j] != b[i][j]:
            if a[i][j]:
                for z in range(2):
                    for w in range(2):
                        b[i + z][j + w] = 1
            else:
                flag = True
if flag:
    print(-1)
    exit()
print(s * t // 4)
for i in range(s - 1):
    for j in range(t - 1):
        print(i + 1, j + 1)
