#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int idx[26];
  for (int i = 0; i < 26; i++) { idx[i] = n; }
  for (int i = 0; i < n; i++) { idx[s[i] - 'a'] = i; }
  string str = "";
  for (int x = 0; x < k; x++) {
    int cur = 1000000;
    for (int i = 0; i < 26; i++) { cur = min(cur, idx[i]); }
    str += s[cur];
    idx[s[cur] - 'a'] = n;
  }
  cout << str;
  return 0;
}
