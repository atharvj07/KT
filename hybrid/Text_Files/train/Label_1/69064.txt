n, k, M = map(int, input().split())
t = list(map(int, input().split()))
t.sort()

# Total time to complete one task
complete_task_time = sum(t)

max_points = 0

for complete_tasks in range(n + 1):
    time_left = M - complete_task_time * complete_tasks
    if time_left < 0:
        break
    points = (k + 1) * complete_tasks
    for i in range(k):
        additional_tasks = min(time_left // t[i], n - complete_tasks)
        time_left -= t[i] * additional_tasks
        points += additional_tasks
    max_points = max(max_points, points)

print(max_points)
