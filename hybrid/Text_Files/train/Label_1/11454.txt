#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<long long> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }
  long long min_diff = (long long)1e18;
  int ans = -1;
  for (int i = 0; i < n; ++i) {
    for (int j = i + 1; j < n; ++j) {
      long long diff = a[j] - a[i];
      if (diff < 0) diff = -diff;
      if (diff < min_diff) {
        min_diff = diff;
        ans = i;
      }
    }
  }
  std::cout << ans + 1 << std::endl;
  return 0;
}
