#include<bits/stdc++.h>
using namespace std;
int main()
{
    string a,b;
    getline(cin,a);
    int x=a[0];
    int y=a[1];
    for(int i=0;i<5;i++)
    {
        getline(cin,b);
        if(b[0]==x||b[1]==y)
        {
            cout<<"YES";
            return 0;
        }
    }
    cout<<"NO";
    return 0;
}
