#include <bits/stdc++.h>

using namespace std;

using ll = long long;

ll solve() {
  ll ans(0), n;
  cin >> n;
  vector<ll> cnt(n + 1);
  for (ll i = 0, x; i < n; i++) {
    cin >> x;
    cnt[x]++;
  }
  for (ll i = 1; i <= n; i++) {
    ans += cnt[i] * (cnt[i] - 1) / 2;
  }
  return ans;
}

int main() {
  ll t;
  cin >> t;
  while (t--) {
    cout << solve() << endl;
  }
  return 0;
}
