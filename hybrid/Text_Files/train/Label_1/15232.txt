#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 100;
int n, adj[MAXN][MAXN], dp[MAXN][MAXN];

int solve(int u, int p) {
    dp[u][0] = 0;
    dp[u][u] = 0;
    for (int v = 0; v < n; v++) {
        if (v == u || v == p || adj[u][v] == 0) continue;
        solve(v, u);
        for (int i = 0; i < n; i++) {
            dp[u][i] = max(dp[u][i], dp[v][i] + adj[u][v]);
        }
    }
}

int main() {
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        int u, v, c;
        cin >> u >> v >> c;
        adj[u][v] = c;
        adj[v][u] = c;
    }
    solve(0, 0);
    int ans = 0;
    for (int i = 0; i < n; i++) {
        ans = max(ans, dp[0][i]);
    }
    cout << ans << endl;
    return 0;
}
