BOARD_SIZE = 8

white = [
    {'name': 'Q', 'weight': 9},
    {'name': 'R', 'weight': 5},
    {'name': 'B', 'weight': 3},
    {'name': 'N', 'weight': 3},
    {'name': 'P', 'weight': 1},
]

black = [
    {'name': 'q', 'weight': 9},
    {'name': 'r', 'weight': 5},
    {'name': 'b', 'weight': 3},
    {'name': 'n', 'weight': 3},
    {'name': 'p', 'weight': 1},
]

board = []
for _ in range(BOARD_SIZE):
    board.append(list(input()))

white_weight = 0
black_weight = 0

for i in range(BOARD_SIZE):
    for j in range(BOARD_SIZE):
        piece = board[i][j]
        if piece in [piece['name'] for piece in white]:
            white_weight += next(piece for piece in white if piece['name'] == piece)['weight']
        elif piece in [piece['name'] for piece in black]:
            black_weight += next(piece for piece in black if piece['name'] == piece)['weight']

if white_weight > black_weight:
    print("White")
elif white_weight == black_weight:
    print("Draw")
else:
    print("Black")
