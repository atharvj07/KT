#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

typedef long long ll;

vector<ll> dec_to_anybase(ll n, ll b) {
  vector<ll> ans;
  while (n > 0) {
    ans.push_back(n % b);
    n /= b;
  }
  reverse(ans.begin(), ans.end());
  return ans;
}

int main() {
  ll n;
  cin >> n;
  while (n--) {
    ll p, q, b;
    cin >> p >> q >> b;
    ll bg = 0;
    while (q > 0) {
      q /= b;
      bg++;
    }
    vector<ll> p_any = dec_to_anybase(p, b);
    bool finite = true;
    for (ll i = 0; i < p_any.size(); i++) {
      if (p_any[i] > 0) {
        if (i > bg) {
          finite = false;
          break;
        }
      }
    }
    cout << (finite ? "Finite" : "Infinite") << '\n';
  }
}
