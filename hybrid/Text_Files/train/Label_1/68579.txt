import sys
input = sys.stdin.readline
n, r = map(int, input().split())
pts = [list(map(int, input().split())) for _ in range(n)]
pts.sort()
ans = 0
l, r = -1000000000, -1000000000
for x, y in pts:
    if abs(x - l) + abs(y - r) <= r:
        ans += 1
    else:
        ans = 1
        l, r = x, y
print(ans)
