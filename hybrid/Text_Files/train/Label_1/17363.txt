#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  vector<int> x(n);
  for (int i = 0; i < n; i++) cin >> x[i];
  sort(x.begin(), x.end());
  for (int i = 0; i < q; i++) {
    int m;
    cin >> m;
    int l = 0, r = n - 1;
    while (l < r) {
      int mid = (l + r) / 2;
      if (x[mid] <= m) l = mid + 1;
      else r = mid;
    }
    cout << (x[r] <= m ? r + 1 : 0) << endl;
  }
  return 0;
}
