n, m = map(int, input().split())
sx, sy = map(int, input().split())
lx, rx = map(int, input().split())
grid = [input() for i in range(n)]
vis = [[False] * m for i in range(n)]

def dfs(x, y, lx, rx):
    if x < 0 or x >= n or y < 0 or y >= m or grid[x][y] == '*' or vis[x][y]:
        return
    if lx < 0 and y < sy or rx < 0 and y > sy:
        return
    vis[x][y] = True
    dfs(x - 1, y, lx - 1, rx)
    dfs(x + 1, y, lx, rx - 1)
    dfs(x, y - 1, lx, rx)
    dfs(x, y + 1, lx, rx)

dfs(sx - 1, sy - 1, lx, rx)
print(sum(sum(row) for row in vis))
