#include<bits/stdc++.h>
# define rep(i,n) for(int i=0;i<(n);i++)
using namespace std;
const int inf=1<<30;
int cost[5][150010];
int mp[5][150010];
int main(){
  int n[4],a[4][150010],m[4],x[4][200010],y[4][200010];
  rep(i,4)cin>>n[i];
  rep(i,4)rep(j,n[i])cin>>a[i][j];
  rep(i,4)cin>>m[i];
  rep(i,4)rep(j,m[i])cin>>x[i][j]>>y[i][j];
  rep(i,4)rep(j,n[i])cost[i][j]=inf;
  rep(i,n[0])cost[0][i]=a[0][i];
  rep(k,3){
    int p[]={-1,1};
    rep(i,n[k])rep(j,n[k+1])if(!mp[k][i] && !mp[k+1][j] && (x[k][i]!=j || y[k][j]!=i)){
      rep(l,2)if(cost[k+1][j+p[l]]+a[k][i]<cost[k][i]){
        cost[k][i]=cost[k+1][j+p[l]]+a[k][i];
        mp[k][i]=j+p[l];
      }
    }
  }
  int ans=inf;
  rep(i,n[3])if(cost[3][i]!=inf && cost[3][i]+a[3][i]<ans)ans=cost[3][i]+a[3][i];
  cout<<(ans==inf?-1:ans)<<endl;
}
