def find_peaks(n, p):
    peaks = []
    for i in range(1, n - 1):
        if p[i] > p[i - 1] and p[i] > p[i + 1]:
            peaks.append(i)
    return peaks

def find_valley_lengths(n, p, peaks):
    valley_lengths = []
    for peak in peaks:
        left = peak - 1
        right = peak + 1
        length = 1
        while left >= 0 and p[left] < p[left + 1]:
            left -= 1
            length += 1
        while right < n and p[right] < p[right - 1]:
            right += 1
            length += 1
        valley_lengths.append(length)
    return valley_lengths

def solve(n, p):
    peaks = find_peaks(n, p)
    valley_lengths = find_valley_lengths(n, p, peaks)
    valley_length_counts = {}
    for length in valley_lengths:
        if length not in valley_length_counts:
            valley_length_counts[length] = 0
        valley_length_counts[length] += 1
    winning_positions = 0
    for length, count in valley_length_counts.items():
        if count == 1 and ((length - 1) & length) == 0:
            winning_positions += 1
    return 1 if winning_positions == 1 else 0

n = int(input().strip())
p = list(map(int, input().strip().split()))
print(solve(n, p))
