import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long l = in.nextLong(), r = in.nextLong(), k = in.nextLong();
        System.out.println(l ^ (l + 1));
        System.out.println(Math.min(k, r - l));
        for (long i = l; i <= l + k - 1; i++)
            System.out.print(i + " ");
        System.out.println();
    }
}
