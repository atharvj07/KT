import sys

t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    h = list(map(int, input().split()))
    ans = 'YES'
    if h[0] != 0 or h[n - 1] != 0:
        ans = 'NO'
    for i in range(1, n - 1):
        if h[i] > h[i - 1] + k - 1:
            ans = 'NO'
        if h[i] < h[i - 1] and h[i] + k - 1 < h[i + 1]:
            ans = 'NO'
    sys.stdout.write(ans + '\n')
