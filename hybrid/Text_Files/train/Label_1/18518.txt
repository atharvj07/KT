#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (auto &ai : a) cin >> ai;
  long long sm = 0;
  sort(rbegin(a), rend(a));
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (sm + a[i] + 1 < a[0])
      sm += a[i] + 1, ans++;
    else {
      sm -= a[i] + 1;
      ans++;
      sm += a[0];
    }
  }
  cout << ans << endl;
}
