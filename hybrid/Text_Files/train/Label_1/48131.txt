import java.util.Scanner;

public class RomaGrind {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();

        int[] weaponAttack = new int[n];
        int[] weaponCost = new int[n];
        int[] armorDefense = new int[m];
        int[] armorCost = new int[m];
        int[][] monsterDefenseAttackCoins = new int[p][3];

        for (int i = 0; i < n; i++) {
            weaponAttack[i] = scanner.nextInt();
            weaponCost[i] = scanner.nextInt();
        }

        for (int j = 0; j < m; j++) {
            armorDefense[j] = scanner.nextInt();
            armorCost[j] = scanner.nextInt();
        }

        for (int k = 0; k < p; k++) {
            monsterDefenseAttackCoins[k][0] = scanner.nextInt();
            monsterDefenseAttackCoins[k][1] = scanner.nextInt();
            monsterDefenseAttackCoins[k][2] = scanner.nextInt();
        }

        long maxProfit = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                long profit = 0;
                for (int k = 0; k < p; k++) {
                    if (weaponAttack[i] > monsterDefenseAttackCoins[k][0] && armorDefense[j] > monsterDefenseAttackCoins[k][1]) {
                        profit += monsterDefenseAttackCoins[k][2];
                    }
                }
                profit -= (weaponCost[i] + armorCost[j]);
                maxProfit = Math.max(maxProfit, profit);
            }
        }

        System.out.println(maxProfit);
    }
}
