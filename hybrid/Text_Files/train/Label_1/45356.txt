import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[n];
        long gcd = 0;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            gcd = gcd(gcd, a[i]);
        }
        for (int i = 1; i * i <= m; i++) {
            if (gcd % i == 0) {
                if (m / i <= (long) gcd / i) {
                    ans += 1;
                } else {
                    ans += 2;
                }
            }
        }
        System.out.println(ans);
    }

    public static long gcd(long a, long b) {
        while (b > 0) {
            long temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
