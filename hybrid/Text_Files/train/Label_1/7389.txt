#include<bits/stdc++.h>
using namespace std;
int n,k,a,b,c,d,arr[100005];
int main(){
    cin>>n>>k;
    for(int i=1;i<=k;i++){
        cin>>a;
        for(int j=1;j<=a;j++){
            cin>>b;
            arr[b]=a;
        }
    }
    for(int i=2;i<=n;i++)
        c+=arr[i-1]-1;
    cout<<c<<endl;
}
