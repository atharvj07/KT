def solve(N, K, A, D):
    prefix_sum_A = [0] * (N + 1)
    prefix_sum_D = [0] * (N + 1)
    for i in range(1, N + 1):
        prefix_sum_A[i] = prefix_sum_A[i - 1] + A[i - 1]
        prefix_sum_D[i] = prefix_sum_D[i - 1] + D[i - 1]

    dp = [[0] * (K + 2) for _ in range(N + 1)]
    for i in range(1, N + 1):
        for j in range(1, min(i, K + 1) + 1):
            dp[i][j] = dp[i - 1][j]
            for k in range(1, i):
                dp[i][j] = max(dp[i][j], dp[k][j - 1] + prefix_sum_A[i] - prefix_sum_A[k] - D[k])

    return max(dp[N])

N, K = map(int, input().split())
A = list(map(int, input().split()))
D = list(map(int, input().split()))

print(solve(N, K, A, D))
