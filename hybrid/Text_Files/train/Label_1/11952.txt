#include <bits/stdc++.h>
using namespace std;
const int maxn = 3 * 1e5 + 5;
char s[maxn];
int main() {
  int n;
  cin >> n;
  cin >> s;
  int l = 0, r = n - 1;
  while (l <= r) {
    if (s[l] == s[r]) {
      if (s[l] != '?') {
        if (s[l] == ')')
          break;
      } else if (l != r) {
        s[l] = '(', s[r] = ')';
      } else
        s[l] = '(';
    } else
      break;
    l++, r--;
  }
  while (l <= r) {
    if (s[l] != '?' && s[r] != '?') {
      if (s[l] == s[r]) {
        if (s[l] == ')')
          break;
      } else
        break;
    } else if (s[l] != '?') {
      if (s[l] == ')')
        break;
      s[r] = s[l];
    } else if (s[r] != '?') {
      if (s[r] == ')')
        break;
      s[l] = s[r];
    } else {
      s[l] = '(', s[r] = ')';
    }
    l++, r--;
  }
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(') cnt++;
    if (s[i] == ')') cnt--;
    if (cnt < 0) break;
  }
  if (cnt != 0 || l <= r)
    cout << ":(" << endl;
  else
    cout << s << endl;
  return 0;
}
