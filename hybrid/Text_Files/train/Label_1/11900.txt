#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b;
    cin >> a >> b;
    int ans = 0;
    for (long long i = 1; i <= a; i++) {
      for (long long j = 1; j <= b; j++) {
        string s1 = to_string(i);
        string s2 = to_string(j);
        string conc = s1 + s2;
        long long val = stol(conc);
        long long eq = i * j + i + j;
        if (eq == val) ans++;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
