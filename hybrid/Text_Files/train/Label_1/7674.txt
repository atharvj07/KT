#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> piles(n);
    for (auto &pile : piles) { cin >> pile; }
    sort(piles.begin(), piles.end());
    cout << (piles[0] == piles.back() || piles[0] + 1 == piles.back() ? "First" : "Second") << endl;
  }
  return 0;
}
