#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, i, j;
  string a[30], b = "";
  for (i = 0; i < 30; i++) a[i] = "0";
  cin >> n;
  for (i = 0; i < n; i++) {
    cin >> a[i];
    for (j = 0; j < a[i].length(); j++) b += a[i][j];
  }
  sort(b.begin(), b.end());
  char z = 'a';
  for (i = 0; i < b.size(); i++) {
    if (b[i] != z) {
      cout << z << endl;
      return 0;
    }
    z++;
  }
  cout << z << endl;
}
