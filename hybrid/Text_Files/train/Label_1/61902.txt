a = input().lstrip('0')
b = input().lstrip('0')
if sum(c=='1' for c in a)!=sum(c=='1' for c in b): print('NO'); exit()
pa = a==''; pb = b==''; i = 0; j = 0
while a!='':
    if i<len(a) and a[i]=='1':
        pa ^= 1
        i += 1; continue
    if j<len(b) and b[j]=='1':
        pb ^= 1
        j += 1
    if pa==pb and a[i:]==b[j:]: break
    if a=='': print('NO'); break
    a = a[1:]; pa ^= 1
if a=='': print('YES')
else: print('NO')
