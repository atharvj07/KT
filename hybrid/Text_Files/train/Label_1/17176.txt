#include <bits/stdc++.h>
#define rep(i, a, b) for (size_t i = (a), _##i = (b); i < _##i; ++i)

template<typename T>
bool get_bit(const T& x, int idx) { return (x & (1 << idx)) != 0; }

template<typename T>
T set_bit(const T& x, int idx) { return x | (1 << idx); }

template<typename T>
T unset_bit(const T& x, int idx) { return x & ~(1 << idx); }

using namespace std;

int main() {
    int t;
    cin >> t;
    rep(_, 0, t) {
        string x, y;
        cin >> x >> y;
        int n = x.size() - 1, m = y.size() - 1, k = -1;
        reverse(x.begin(), x.end());
        reverse(y.begin(), y.end());
        rep(i, 0, m + 1) {
            int j = i;
            bool ok = true;
            while (j < n && get_bit(x[j], 0) == get_bit(y[i], 0)) {
                j++;
            }
            if (j == n) {
                if (get_bit(x[j], 0) == 0 && get_bit(y[i], 0) == 1) {
                    ok = false;
                }
            } else if (get_bit(x[j], 0) < get_bit(y[i], 0)) ok = false;
            if (ok) k = i;
        }
        cout << k << '\n';
    }
    return 0;
}
