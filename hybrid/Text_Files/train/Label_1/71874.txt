import sys
input = sys.stdin.readline

n = int(input())
points = [tuple(map(int, input().split())) for _ in range(n)]
count = 0

for p1 in points:
    for p2 in points:
        for p3 in points:
            if p1 == p2 or p1 == p3 or p2 == p3:
                continue
            x1, y1 = p1
            x2, y2 = p2
            x3, y3 = p3
            if (x1 - x2) * (y3 - y1) == (x3 - x1) * (y2 - y1):
                count += 1

print(count // 4)
