#include <iostream>
#include <string>
using namespace std;
int main() {
  int n, a, b, ans = 0, cnt = 0;
  string s;
  cin >> n >> a >> b >> s;
  for (int i = 0; i < n; i++)
    if (s[i] == '*')
      cnt++;
    else if (s[i] == '.' && cnt > 0) {
      if (a + b - ans > cnt) {
        int tmp = min(a, cnt / 2);
        ans += tmp;
        a -= tmp;
      }
      cnt = 0;
    }
  cout << ans << endl;
  return 0;
}
