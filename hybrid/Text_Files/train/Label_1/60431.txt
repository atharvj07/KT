t = int(input().strip())
for _ in range(t):
    path = input().strip()
    visited = set()
    time = 0
    x, y = 0, 0
    for move in path:
        nx, ny = x, y
        if move == 'N':
            ny += 1
        elif move == 'S':
            ny -= 1
        elif move == 'E':
            nx += 1
        elif move == 'W':
            nx -= 1
        if (x, y, nx, ny) in visited or (nx, ny, x, y) in visited:
            time += 1
        else:
            time += 5
            visited.add((x, y, nx, ny))
        x, y = nx, ny
    print(time)
