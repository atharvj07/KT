import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int count = 0;
        while (w >= b) {
            while (h >= a) {
                h -= a;
                count++;
            }
            w -= b;
        }
        System.out.println((h * w) + ((w / b) + (h / a)) * a * b);
    }
}
