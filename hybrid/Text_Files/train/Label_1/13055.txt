#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, t;
    while (cin >> n >> t) {
        vector<int> a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        int sum = a[0], ans = 1;
        for (int i = 1; i < n; i++) {
            if (sum + a[i] <= t) {
                sum += a[i];
                ans++;
            } else {
                break;
            }
        }
        cout << ans << endl;
    }
}
