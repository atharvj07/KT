#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int MAXN = 1005;

int n, d;
array<int, MAXN> deg;
vector<int> g[MAXN];

int main() {
  cin >> n >> d;
  for (int i = 0; i < d; ++i) {
    int x, y;
    cin >> x >> y;
    g[x].push_back(y);
    g[y].push_back(x);
  }
  for (int i = 1; i <= n; ++i) {
    deg[i] = g[i].size();
  }
  for (int i = 0; i < d; ++i) {
    int v;
    cin >> v;
    for (int u : g[v]) {
      deg[u]--;
      deg[v]--;
    }
    int mx = *max_element(deg.begin() + 1, deg.begin() + n + 1);
    cout << mx << " \n"[i == d - 1];
  }
  return 0;
}
