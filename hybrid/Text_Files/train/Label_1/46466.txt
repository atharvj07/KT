import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Deque<Integer> dq = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            dq.add(a[i]);
        }
        while (q-- > 0) {
            long m = sc.nextLong();
            int j = 0;
            for (int x : dq) {
                if (j++ >= m % dq.size()) {
                    int y = dq.removeFirst();
                    System.out.println(y + " " + x);
                    dq.addLast(y);
                    break;
                }
                m %= dq.size();
            }
        }
    }
}
