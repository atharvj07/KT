import sys
mod = 10 ** 9 + 7

def power(x, y):
    ret = 1
    while y > 0:
        if y % 2 == 0:
            x = x * x
            x = x % mod
            y = y / 2
        else:
            ret = ret * x
            ret = ret % mod
            y = y - 1
    return ret

def inverse_mod(a):
    return power(a, mod - 2)

n, k = map(int, sys.stdin.readline().split())
ans = []
fact = [1] * (n + 2)
for i in range(2, n + 1):
    fact[i] = fact[i - 1] * i
    fact[i] = fact[i] % mod
inv_fact = [1] * (n + 2)
for i in range(1, n + 1):
    inv_fact[i] = inverse_mod(fact[i])
for j in range(1, k + 1):
    x = fact[n] * inv_fact[n - j]
    x = x % mod
    y = inv_fact[j]
    y = y % mod
    ans.append((x * y) % mod)
print(*ans)
