n = int(input().strip())
a = list(map(int, input().strip().split()))

indices = {}
for i, x in enumerate(a):
    if x not in indices:
        indices[x] = []
    indices[x].append(i)

valid_x = {}
for x, positions in indices.items():
    if len(positions) == 1:
        valid_x[x] = 0
    else:
        diff = positions[1] - positions[0]
        is_arithmetic = all(positions[i] - positions[i-1] == diff for i in range(2, len(positions)))
        if is_arithmetic:
            valid_x[x] = diff

print(len(valid_x))
for x, p_x in sorted(valid_x.items()):
    print(x, p_x)
