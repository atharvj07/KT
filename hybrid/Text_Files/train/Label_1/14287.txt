#include <bits/stdc++.h>

int n, k;

int main() {
  std::cin >> n >> k;
  if (n * log2(n) < k) return std::puts("-1"), 0;
  if (n == 1) return std::cout << 1 << std::endl, 0;
  long long diff = (k / n + 1) % 2;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    a[i] = (i % 2 == 0) == diff ? i + 2 : i + 1;
  }
  if (n % 2 == 0) { a[n - 1] = n; }
  std::reverse(a.begin() + n / 2, a.end());
  for (auto& x : a) std::cout << x << ' '; std::cout << std::endl;
  return 0;
}
