def generate_vectors(k):
    if k == 0:
        return ['+']
    prev_vectors = generate_vectors(k - 1)
    new_vectors = []
    for v in prev_vectors:
        new_vectors.append(v + v)
        new_vectors.append(v + ''.join(['*' if c == '+' else '+' for c in v]))
    return new_vectors

k = int(input().strip())
vectors = generate_vectors(k)
for vector in vectors:
    print(vector.replace('+', '+').replace('*', '*'))
