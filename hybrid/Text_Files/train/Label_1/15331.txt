#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n], b[n], x[n];
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        a[i]--;
        x[a[i]] = i;
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i];
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == i) continue;
        if (b[i] ^ b[x[i]]) ans++;
        swap(a[i], a[x[i]]);
        swap(b[i], b[x[i]]);
        i--;
    }
    cout << ans << endl;
}
