#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        if (k <= n) {
            cout << "YES " << n - k << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
