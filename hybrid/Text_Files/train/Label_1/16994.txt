#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 5;
int dp[N], n, x, c[N];

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> x;
    for (int i = 0; i < n; i++) cin >> c[i];
    for (int i = 0; i <= x; i++) dp[i] = INT_MAX;
    dp[x] = 0;
    for (int i = 1; i <= x; i++) {
        for (int j = 0; j < n; j++) {
            if (i >= c[j]) {
                dp[i] = min(dp[i], dp[i - c[j]] + c[j]);
            }
        }
    }
    cout << dp[x] << '\n';
}
