import java.util.*;
public class Main {
    public static void main(String[] args)  {
        Scanner sc = new Scanner(System.in);
        long d = sc.nextLong();
        long s = sc.nextLong();
        long sm = 0;
        for (long i = 0; i <= 1000000000000000000L; i += d) {
            long t = i;
            sm = 0;
            while (t > 0) {
                sm += t % 10;
                t /= 10;
            }
            if (sm == s) {
                System.out.println(i);
                return;
            }
        }
        System.out.println(-1);
    }
}
