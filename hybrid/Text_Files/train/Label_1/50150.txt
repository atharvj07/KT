import java.io.*;
import java.util.*;

class Main {
    static class Edge {
        int to, w;

        Edge(int to, int w) {
            this.to = to;
            this.w = w;
        }
    }

    static int n, m, ans, ansDist;
    static int[] dist;
    static List<Edge>[] graph;
    static boolean[] attacked;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        graph = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(in.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            graph[a].add(new Edge(b, 1));
            graph[b].add(new Edge(a, 1));
        }
        st = new StringTokenizer(in.readLine());
        attacked = new boolean[n + 1];
        for (int i = 0; i < m; i++) {
            attacked[Integer.parseInt(st.nextToken())] = true;
        }
        bfs(1);
        bfs(ans);
        System.out.println(ans + "\n" + ansDist);
    }

    static void bfs(int start) {
        Queue<Integer> q = new LinkedList<>();
        q.add(start);
        dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[start] = 0;
        while (!q.isEmpty()) {
            int curr = q.poll();
            for (Edge next : graph[curr]) {
                if (dist[next.to] > dist[curr] + next.w) {
                    dist[next.to] = dist[curr] + next.w;
                    q.add(next.to);
                    if (attacked[next.to] && dist[next.to] > ansDist) {
                        ans = next.to;
                        ansDist = dist[next.to];
                    }
                }
            }
        }
    }
}
