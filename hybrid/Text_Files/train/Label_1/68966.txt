import sys
input=sys.stdin.readline
n,m=map(int,input().split())
similar=[[]for _ in range(n+1)]
for _ in range(m):
 u,v=map(int,input().split())
 similar[u].append(v),similar[v].append(u)
def check(a,b):
    if not a and not b:return 1
    if any(a)and any(b):
       for x in a:
           if any(y in b for y in similar[x]):return 0
       for y in b:
           if any(x in a for x in similar[y]):return 0
    return sum(check(a,b-set([x]))for x in b)
print(check(set(range(1,n+1)),set()))
