#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, l, r, x;
    cin >> n >> l >> r >> x;
    vector<int> c(n);
    for (int i = 0; i < n; i++) cin >> c[i];
    sort(c.begin(), c.end());
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (l <= c[i] + c[j] && c[i] + c[j] <= r && c[j] - c[i] >= x) ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
