import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long sum = 0;
        for (long v : a) {
            sum += v;
        }
        if (sum % n != 0) {
            System.out.println(-1);
            return;
        }
        long target = sum / n;
        for (int i = 0; i < 3; i++) {
            long curSum = 0;
            int l = 0, r = 0;
            for (int j = 0; j < n; j++) {
                curSum += a[j];
                if (curSum >= target * (j + 1)) {
                    r = j;
                    break;
                }
            }
            for (int j = l; j <= r; j++) {
                a[j] -= target * (j + 1) - curSum;
            }
            System.out.println((l + 1) + " " + (r + 1));
            for (int j = l; j <= r; j++) {
                System.out.print((target * (j + 1) - curSum) / (r - l + 1) + " ");
            }
            System.out.println();
        }
        sc.close();
    }
}
