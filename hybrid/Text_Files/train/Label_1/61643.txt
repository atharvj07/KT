for _ in range(int(input())):
    n, m = map(int, input().split())
    a = []
    for i in range(2*n):
        a.append(list(map(int, input().split())))
    for x in range(n):
        for i in range(m):
            for j in range(m):
                if a[2*x][0] != a[2*x+1][j] or a[2*x][1] != a[2*x+1][m-1-j]:
                    print('NO')
                    break
            else:
                continue
            break
        else:
            print('YES')
            break
    else:
        print('NO')
