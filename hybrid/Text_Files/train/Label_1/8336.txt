#include <algorithm>
#include <cstring>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  string s;
  getline(cin, s);
  int n = s.size();
  vector<string> v;
  for (int i = 0, j; i < n; i = j + 1) {
    if (s[i] == '"') {
      for (j = i + 1; j < n && s[j] != '"'; j++) {
      }
      if (j < n) {
        v.push_back(s.substr(i + 1, j - 1 - i));
      }
    } else {
      for (j = i + 1; j < n; j++) {
        if (s[j] == ' ') {
          break;
        }
      }
      v.push_back(s.substr(i, j - i));
    }
  }
  for (auto &x : v) {
    cout << "<" << x << ">" << endl;
  }
  return 0;
}
