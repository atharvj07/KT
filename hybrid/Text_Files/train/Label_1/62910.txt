def check(n,grid):
	for x in range(n):
		for y in range(n):
			if grid[x][y]:
				if (x==0 or grid[x-1][y]>0) and (y==0 or grid[x][y-1]>0):
					if x+1<n and grid[x+1][y]>0:
						while x+2<=n and grid[x+1][y]==grid[x][y]:x+=1
					if y+1<n and grid[x][y+1]>0:
						while y+2<=n and grid[x][y+1]==grid[x][y]:y+=1
					if x+1<=n and y+1<=n and not (grid[x+1][y]==grid[x][y] and grid[x][y+1]==grid[x][y]):return "No"
				else:return "No"
	return "Yes"

n=int(input())
grid=[list(map(int,input()))for i in range(n)]
print(check(n,grid))
