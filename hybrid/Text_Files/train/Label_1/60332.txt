a, b, c = map(int, input().split())
m = int(input())
usb_mouses = []
ps2_mouses = []
for _ in range(m):
    price, port = input().split()
    price = int(price)
    if port == 'USB':
        usb_mouses.append(price)
    else:
        ps2_mouses.append(price)

usb_mouses.sort()
ps2_mouses.sort()

total_cost = 0
equipped_computers = 0

for mouse_list, available_ports in [(usb_mouses, a), (ps2_mouses, b)]:
    while mouse_list and available_ports > 0:
        total_cost += mouse_list.pop(0)
        equipped_computers += 1
        available_ports -= 1

both_ports_mouses = usb_mouses + ps2_mouses
both_ports_mouses.sort()

while both_ports_mouses and c > 0:
    total_cost += both_ports_mouses.pop(0)
    equipped_computers += 1
    c -= 1

print(equipped_computers, total_cost)
