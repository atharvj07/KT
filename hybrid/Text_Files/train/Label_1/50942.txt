import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int[][] jobs = new int[N][2];
        for (int i = 0; i < N; i++) {
            jobs[i][0] = sc.nextInt();
            jobs[i][1] = sc.nextInt();
        }

        Arrays.sort(jobs, (a, b) -> Integer.compare(a[0], b[0]));

        int[] dp = new int[M + 1];
        for (int i = 0; i < N; i++) {
            int day = jobs[i][0];
            int reward = jobs[i][1];
            for (int j = M; j >= day; j--) {
                dp[j] = Math.max(dp[j], dp[j - day] + reward);
            }
        }

        System.out.println(dp[M]);
    }
}
