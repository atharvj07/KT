import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            long a = in.nextLong();
            long b = in.nextLong();
            int count = 0;
            while (a < b) {
                count++;
                if (b % 8 == 0) {
                    b /= 8;
                } else if (b % 4 == 0) {
                    b /= 4;
                } else if (b % 2 == 0) {
                    b /= 2;
                } else {
                    count = -1;
                    break;
                }
            }
            if (a != b) {
                count = -1;
            }
            System.out.println(count);
        }
        in.close();
    }
}
