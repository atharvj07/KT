#include <iostream>
#include <vector>
using namespace std;
int main() {
  long long n, p, m = 0;
  cin >> n >> p;
  vector<string> buyers(n);
  for (string &k : buyers) cin >> k;
  for (int i = 0; i < n; ++i)
    m += ((buyers[i] == "halfplus") ? p + p / 2 : p / 2);
  cout << m << endl;
  return 0;
}
