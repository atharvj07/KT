import sys

q,m=map(int,input().split())
damage,mana=list(),list()
for i in range(q):
 a,b,c=map(int,input().split())
 if c==2:
  if len(damage)==0:print("NO")
  else:
   x,y=damage[-1],mana[-1]
   if x*a+y*b<=m and a<=x*a+y*b:print("YES")
   else:print("NO")
 if c==1:
  damage.append((a+i+1)%1000001)
  mana.append((b+i+1)%1000001)
