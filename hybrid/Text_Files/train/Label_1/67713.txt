n, l, x, y = map(int,(input().split()))
a = list(map(int,input().split()))
x,a = sorted([x]+a),a+[l]
ans = 0
for i in range(1,len(x)):
    if x[i] == x[0]:
        ans += 1
    elif x[i]-x[i-1] < y:
        ans += y-(x[i]-x[i-1])
print(ans)
