import java.util.Arrays;
import java.util.Scanner;

public class Main {
	
	static int[] alice, bob;
	static int[][] dp;
	static boolean[] done;

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int T = sc.nextInt();
		int[] wins = new int[T];
		dp = new int[T][1 << 16];
		done = new boolean[T];
		for (int t = 0; t < T; t++) {
			int f = sc.nextInt();
			alice = new int[8];
			bob = new int[8];
			for (int i = 0; i < 8; i++)
				alice[i] = sc.nextInt();
			for (int i = 0; i < 8; i++)
				bob[i] = sc.nextInt();
			wins[t] = f == 0 ? dp(0, 0, t) : dp(1, 0, t);
		}
		for (int t = 0; t < T; t++) {
			if (wins[t] == 0)
				System.out.println("Alice");
			else if (wins[t] == 1)
				System.out.println("Bob");
			else if (wins[t] == -1)
				System.out.println("Deal");
		}
		sc.close();
	}

	static int dp(int turn, int mask, int t) {
		if (done[t])
			return dp[t][mask];
		done[t] = true;
		int res = -1;
		boolean isAlice = turn == 0;
		for (int i = 0; i < 8; i++) {
			boolean works = !isAlice ? bob[i] * alice[i] != 0 : alice[i] * bob[i] != 0;
			if (!works)
				continue;
			int al = alice[i], bo = bob[i];
			alice[i] = (alice[i] + bob[i]) % 5;
			bob[i] = isAlice ? al : bo;
			if (turn > 0) {
				int op = turn;
				while (op < 8 && alice[op] == 0 && bob[op] == 0)
					op++;
				if (op == 8) {
					int cnt = 0;
					for (int j = 0; j < 8; j++) {
						if (isAlice)
							cnt += alice[j] == 0 ? 1 : 0;
						else
							cnt += bob[j] == 0 ? 1 : 0;
					}
					if (cnt == 8)
						return dp[t][mask] = res = 0;
					else
						return dp[t][mask] = res = 2;
				}
				if (isAlice)
					res = Math.max(res, dp(1 - turn, mask | (1 << op), t));
				else
					res = Math.min(res, dp(1 - turn, mask | (1 << op), t));
				bob[i] = bo;
				alice[i] = al;
			} else {
				for (int j = 0; j < 8; j++) {
					boolean works2 = !isAlice ? bob[j] * alice[j] != 0 : alice[j] * bob[j] != 0;
					if (!works2)
						continue;
					int op = isAlice ? j : i;
					int al2 = alice[j], bo2 = bob[j];
					alice[j] = (alice[j] + bob[j]) % 5;
					bob[j] = isAlice ? al2 : bo2;
					if (isAlice)
						res = Math.max(res, dp(1 - turn, mask | (1 << op), t));
					else
						res = Math.min(res, dp(1 - turn, mask | (1 << op), t));
					bob[j] = bo2;
					alice[j] = al2;
				}
			}
		}
		return dp[t][mask] = res;
	}
}
