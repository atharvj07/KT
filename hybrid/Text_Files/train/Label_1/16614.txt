#include <bits/stdc++.h>
using namespace std;

const int MXN = 150000;

int main() {
  int n, k, q; cin >> n >> k >> q;
  vector<long long> v(n);
  for (auto& x : v) cin >> x;
  set<long long> online;
  while (q--) {
    int type, id; cin >> type >> id;
    if (type == 1) {
      if (online.size() < (unsigned long long)k) {
        online.insert(v[id - 1]);
      } else {
        auto it = online.begin();
        if (v[id - 1] > *it) {
          online.erase(it);
          online.insert(v[id - 1]);
        }
      }
    } else {
      if (online.count(v[id - 1])) cout << "YES\n";
      else cout << "NO\n";
    }
  }
}
