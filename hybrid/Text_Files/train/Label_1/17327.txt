#include<bits/stdc++.h>
using namespace std;
int n,m;
int p[505][505],cnt;
int mod=1e9+7;
int dp(int x,int y)
{
    if(x==1){
        if(y<=m)
        return 1;
        else return 0;
    }
    if(p[x][y]!=-1)
    return p[x][y];
    int ans=0;
    for(int j=y;j*1LL*y<=m;j++)
    {
        ans+=dp(x-1,j);
        ans%=mod;
    }
    return p[x][y]=ans;
}
int main()
{
    cin>>n>>m;
    memset(p,-1,sizeof(p));
    cnt=dp(n,1);
    cout<<cnt<<endl;
    return 0;
}
