#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    bool ok = true;
    int sum_a = 0, sum_b = 0;
    for (int i = 0; i < n; i++) {
      sum_a += a[i];
      sum_b += b[i];
      if (i && b[i] < b[i - 1]) ok = false;
    }
    if (sum_a != sum_b || !ok)
      cout << "NO\n";
    else
      cout << "YES\n";
  }
  return 0;
}
