#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
const int maxn = 1005;
int dp[maxn][maxn];

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    sort(a.begin() + 1, a.end());
    memset(dp, 0, sizeof(dp));
    dp[0][0] = 1;
    for (int i = 1; i <= n; i++) {
      dp[i][0] = 1;
      for (int j = 1; j <= k; j++) {
        dp[i][j] = dp[i - 1][j];
        if (a[i] <= j) { dp[i][j] = (dp[i][j] + dp[i - 1][j - a[i]]) % mod; }
      }
    }
    cout << dp[n][k] << "\n";
  }
}
