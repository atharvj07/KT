#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pii pair<int, int>
const int MAXN = 2e5 + 5;
const int inf = 1e9;
struct Edge{
    int u, v, w;
    bool is_modify;
    bool operator<(const Edge&e) const{
        return w < e.w;
    }
};
vector<Edge> edges;
int par[MAXN];
int n, m;
int ans;
int find_par(int u){
    if(u == par[u]) return u;
    return par[u] = find_par(par[u]);
}
void init(){
    for(int i = 0;i <= n;++i) par[i] = i;
}
void unite(int a, int b){
    a = find_par(a);
    b = find_par(b);
    if(a != b) par[a] = b;
}
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr); cout.tie(nullptr);
    cin >> n >> m;
    for(int i = 0;i < m;++i){
        int u, v, w;
        cin >> u >> v >> w;
        edges.push_back({u, v, w, false});
    }
    sort(edges.begin(), edges.end());
    init();
    for(auto p : edges){
        if(find_par(p.u) != find_par(p.v)){
            unite(p.u, p.v);
        } else if(!p.is_modify){
            p.is_modify = true;
            ++ans;
        }
    }
    cout << ans << '\n';
}
