#include <iostream>
#include <vector>
#include <queue>
#include <tuple>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<tuple<int, int, int, int>> chips(k);
  for (int i = 0; i < k; i++) {
    int sx, sy, fx, fy;
    cin >> sx >> sy >> fx >> fy;
    --sx, --sy, --fx, --fy;
    chips[i] = {sx, sy, fx, fy};
  }
  int ans = 0;
  int dx[] = {0, 0, 1, -1};
  int dy[] = {-1, 1, 0, 0};
  char dir[] = {'L', 'R', 'D', 'U'};
  queue<tuple<int, int, int>> q;
  vector<vector<bool>> vis(n, vector<bool>(m, false));
  for (int i = 0; i < k; i++) {
    int sx, sy, fx, fy;
    tie(sx, sy, fx, fy) = chips[i];
    if (sx == fx && sy == fy) continue;
    q.push({sx, sy, 0});
    vis[sx][sy] = true;
    while (!q.empty()) {
      int x, y, steps;
      tie(x, y, steps) = q.front();
      q.pop();
      if (x == fx && y == fy) {
        ans += steps;
        break;
      }
      for (int j = 0; j < 4; j++) {
        int nx = x + dx[j];
        int ny = y + dy[j];
        if (nx >= 0 && nx < n && ny >= 0 && ny < m && !vis[nx][ny]) {
          q.push({nx, ny, steps + 1});
          vis[nx][ny] = true;
        }
      }
    }
  }
  if (ans <= 2 * n * m) {
    cout << ans << endl;
    for (int i = 0; i < k; i++) {
      int sx, sy, fx, fy;
      tie(sx, sy, fx, fy) = chips[i];
      if (sx == fx && sy == fy) continue;
      q.push({sx, sy, 0});
      vis[sx][sy] = true;
      while (!q.empty()) {
        int x, y, steps;
        tie(x, y, steps) = q.front();
        q.pop();
        if (x == fx && y == fy) {
          for (int j = 0; j < steps; j++) cout << dir[j];
          break;
        }
        for (int j = 0; j < 4; j++) {
          int nx = x + dx[j];
          int ny = y + dy[j];
          if (nx >= 0 && nx < n && ny >= 0 && ny < m && !vis[nx][ny]) {
            q.push({nx, ny, steps + 1});
            vis[nx][ny] = true;
          }
        }
      }
    }
    cout << endl;
  } else {
    cout << -1 << endl;
  }
  return 0;
}
