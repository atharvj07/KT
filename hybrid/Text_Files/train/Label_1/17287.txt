#include <bits/stdc++.h>
using namespace std;
int main() {
  string s, t;
  cin >> s >> t;
  int n = s.length();
  bool flag = 1;
  for (int i = 0; i < n; i++) {
    if (s[i] == t[i]) continue;
    if (s[i] == '0' && t[i] == '1') {
      s[i] = t[i];
    } else {
      flag = 0;
      break;
    }
  }
  if (flag)
    cout << s;
  else
    cout << "impossible";
  return 0;
}
