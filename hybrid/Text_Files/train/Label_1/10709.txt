#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> f(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> f[i];
    }
    int m = n + 1;
    vector<int> g(n + 1), h(m + 1);
    for (int i = 1; i <= n; i++) {
        if (g[f[i]]) {
            h[g[f[i]]] = i;
        } else {
            g[f[i]] = i;
        }
    }
    for (int i = 1; i <= n; i++) {
        if (g[h[g[f[i]]]] != f[i] || h[g[f[i]]] != i) {
            m = -1;
            break;
        }
    }
    cout << m << endl;
    if (m != -1) {
        for (int i = 1; i <= n; i++) {
            cout << g[i] << ' ';
        }
        cout << endl;
        for (int i = 1; i <= m; i++) {
            cout << h[i] << ' ';
        }
        cout << endl;
    }
    return 0;
}
