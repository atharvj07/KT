import java.io.PrintWriter;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static class Pair {
        int v, w;
        public Pair(int v, int w) {
            this.v = v;
            this.w = w;
        }
    }

    static int[][] dis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        int n = sc.nextInt();
        int M = (int)1e9;
        dis = new int[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dis[i], M);
            dis[i][i] = 0;
        }
        int diff = 0;
        for (int i = 2; i < n; i++) {
            for (int j = 0; j < i; j++) {
                dis[i][j] = diff;
                dis[j][i] = diff;
                diff += (int)1e9 / 2;
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                out.printf("%d ", dis[i][j]);
            }
            out.println();
        }
        out.flush();
    }
}
