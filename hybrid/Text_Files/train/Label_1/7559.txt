#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    stack<int> st;
    int ans = 0;
    for (int i = n - 1; i >= 0; i--) {
      while (!st.empty() && a[st.top()] >= a[i]) st.pop();
      if (!st.empty()) ans++;
      st.push(i);
    }
    cout << ans << endl;
  }
  return 0;
}
