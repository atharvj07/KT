#include <bits/stdc++.h>
using namespace std;

char board[8][8], vis[8][8];
int dx[] = {-2, -2, -2, 2, 2, 2}, dy[] = {-2, 2, 2, -2, 2, -2};
bool met = 0;

void dfs(int x, int y, int d) {
    if (board[x][y] == 'K' && vis[x][y]) met = 1;
    if (met || vis[x][y] || board[x][y] == '#') return;
    vis[x][y] = d;
    for (int i = 0; i < 6; i++) dfs(x + dx[i], y + dy[i], d + 1);
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int sx, sy, ex, ey;
        for (int i = 0; i < 8; i++) {
            for (int j = 0; j < 8; j++) {
                cin >> board[i][j];
                if (board[i][j] == 'K') {
                    sx = i, sy = j;
                }
            }
        }
        memset(vis, 0, sizeof(vis));
        dfs(sx, sy, 1);
        cout << (met ? "YES\n" : "NO\n");
        met = 0;
    }
    return 0;
}
