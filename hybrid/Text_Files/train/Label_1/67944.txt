from array import array
import sys

input = sys.stdin.readline

n = int(input())
is_min = array('i', [False]*(n+1))
for x in map(int, input().split()):
    if x:
        is_min[x] = True

par = array('i', [0]*(n+1))
par[1] = 1
for i, x in enumerate(map(int, input().split()), 2):
    par[i] = x

g = [[] for _ in range(n+1)]
for i in range(2, n+1):
    g[par[i]].append(i)
    
use = [0]*(n+1)
def dfs(x):
    if not g[x]:
        use[x] = 1
        return x
    
    if is_min[x]:
        val = 1e18
        for y in g[x]:
            val = min(val, dfs(y))
        use[x] = val
        return val
    else:
        val = -1e18
        for y in g[x]:
            val = max(val, dfs(y))
        use[x] = val
        return val
    
dfs(1)
print(use[1])
