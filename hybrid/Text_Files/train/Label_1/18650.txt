#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> ans;
  if (n % 2 == 0) {
    for (int i = 2; i <= n; i += 2) { ans.push_back(i); }
    for (int i = 1; i <= n; i += 2) { ans.push_back(i); }
  } else {
    for (int i = 1; i <= n; i += 2) { ans.push_back(i); }
    for (int i = 2; i <= n; i += 2) { ans.push_back(i); }
  }
  std::cout << ans.size() << std::endl;
  for (int x : ans) { std::cout << x << " "; }
  std::cout << std::endl;
  return 0;
}
