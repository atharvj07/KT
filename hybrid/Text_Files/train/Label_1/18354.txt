#include <iostream>

using namespace std;

const int MOD = 1e9+7;

typedef long long ll;

int main() {
  ll l, r;
  cin >> l >> r;
  ll ans = 0;
  ll sum = 0;
  ll k = 1;
  while (l <= r) {
    if (k % 2 == 0) {
      ans = (ans + sum * (r / k - (l - 1) / k)) % MOD;
    } else {
      ans = (ans + sum * ((3 * r - l + 2) % MOD) * ((r - l + 2) % MOD)) % MOD;
    }
    sum = (sum * 2 + (k % 2 == 0)) % MOD;
    k++;
  }
  cout << ans << endl;
  return 0;
}
