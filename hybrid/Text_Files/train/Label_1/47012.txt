import java.util.*;
import java.io.*;
public class Main {
    static PrintWriter out = new PrintWriter(System.out);
    static int cnt = 0;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int[] a = new int[3];
            for (int i = 0; i < 3; i++) {
                a[i] = in.nextInt();
            }
            Arrays.sort(a);
            if (a[1] == a[2]) {
                out.println("YES");
                out.println(a[2] + " " + a[0] + " " + a[1]);
            } else if (a[0] + a[1] == a[2]) {
                out.println("YES");
                out.println(a[2] + " " + a[1] + " " + a[0]);
            } else {
                out.println("NO");
            }
        }
        out.close();
    }
}
