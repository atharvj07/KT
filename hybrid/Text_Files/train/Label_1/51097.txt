import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);

    String s = in.nextLine();
    int n = s.length();
    int[] pos = new int[n];
    int[] p = new int[n];
    int[] next = new int[n];
    int[] prev = new int[n];
    int[] ans = new int[n];
    char[] arr = s.toCharArray();

    for (int i = 0; i < n; i++) {
      pos[i] = -1;
    }

    int cur_pos = 0;
    for (int i = 0; i < n; i++) {
      if (arr[i] == 'R') {
        cur_pos++;
      } else {
        cur_pos--;
      }
      pos[i] = cur_pos;
    }

    for (int i = 0; i < n; i++) {
      if (pos[i] < 0) {
        prev[i] = -1;
      } else {
        prev[i] = next[pos[i]];
        next[pos[i]] = i;
      }
    }

    int best_pos = -1;
    int best_cnt = n;
    for (int i = 0; i < n; i++) {
      int cnt = 0;
      for (int j = i; j != -1; j = prev[j]) {
        cnt++;
      }
      if (cnt < best_cnt) {
        best_cnt = cnt;
        best_pos = i;
      }
    }

    int cur = best_pos;
    for (int i = 0; i < n; i++) {
      p[i] = cur;
      ans[i] = best_cnt - 1;
      cur = next[cur];
    }

    best_cnt--;
    System.out.println(best_cnt);
    for (int i = 0; i < n; i++) {
      System.out.print((p[i] + 1) + " ");
    }
    System.out.println();
  }
}
