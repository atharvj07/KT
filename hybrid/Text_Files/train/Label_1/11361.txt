#include <iostream>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  if (m % n != 0)
    cout << -1 << endl;
  else {
    int ans = 0;
    while (n < m) {
      if (m % 3 == 0) {
        m /= 3;
        ans++;
      } else if (m % 2 == 0) {
        m /= 2;
        ans++;
      } else
        break;
    }
    if (n == m)
      cout << ans << endl;
    else
      cout << -1 << endl;
  }
  return 0;
}
