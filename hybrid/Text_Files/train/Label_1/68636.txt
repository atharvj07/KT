n, v = map(int, input().split())
a = [[int(x) for x in input().split()] for i in range(n)]
ans = 0
ans2 = []
for i in range(n):
    b = []
    for j in range(1, a[i][0]+1):
        if a[i][j] <= v:
            b.append(j)
    if len(b) != 0:
        ans+=1
        ans2.append(i+1)
print(ans)
print(*ans2)
