#include <iostream>
#include <string>
using namespace std;

int main() {
  int n;
  string s;
  cin >> n >> s;
  int ans = n;
  for (int i = 0; i < n - 3; i++) {
    int cnt = 0;
    for (int j = i; j < i + 4; j++) {
      if (s[j] != "ACTG"[(j - i) % 4]) {
        cnt++;
      }
    }
    ans = min(ans, cnt);
  }
  cout << ans << endl;
  return 0;
}
