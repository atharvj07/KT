import java.util.*;

class Main {
  public static void main(String[] args)  {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    String S = sc.next();
    for (int i = 0; i < S.length(); i++) {
      char c = S.charAt(i);
      int idx = c - 'A';
      idx = (idx + N) % 26;
      System.out.print((char)('A' + idx));
    }
    System.out.println();
  }
}
