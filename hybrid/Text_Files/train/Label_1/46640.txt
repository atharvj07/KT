import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    boolean[] bulbs = new boolean[m + 1];
    for (int i = 1; i <= n; i++) {
      int k = sc.nextInt();
      for (int j = 1; j <= k; j++) {
        int y = sc.nextInt();
        bulbs[y] = true;
      }
    }
    boolean allOn = true;
    for (int i = 1; i <= m; i++) {
      if (!bulbs[i]) {
        allOn = false;
        break;
      }
    }
    System.out.println(allOn ? "YES" : "NO");
  }
}
