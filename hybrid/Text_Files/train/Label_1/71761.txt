import sys
import gc

input = sys.stdin.readline

def solve():
    s = input().rstrip()
    if len(s) % 2 == 1:
        return len(s) - 1
    s = s + s
    L = len(s)
    n = len(s) // 2
    l = [0] * L
    r = [0] * L
    for i in range(n - 1, -1, -1):
        l[i] = (l[i + 1] + int(s[i]) != int(s[i + 1]))
    for i in range(n, L):
        r[i] = (r[i - 1] + int(s[i]) != int(s[i - 1]))
    mi = L
    for i in range(n):
        mi = min(mi, r[i] + l[i + n])
    gc.collect()
    return mi

for _ in range(int(input())):
    print(solve())
