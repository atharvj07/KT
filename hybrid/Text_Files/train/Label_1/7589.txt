#include <bits/stdc++.h>
using namespace std;
#define int long long
#define mod 998244353
vector<int> fact, invfact;
int quickpow(int base, int exp) {
    if (!exp) return 1;
    int half = quickpow(base, exp / 2);
    half = (half * half) % mod;
    if (exp % 2) half = (half * base) % mod;
    return half;
}
int comb(int n, int k) {
    return ((fact[n] * invfact[n - k]) % mod * invfact[k]) % mod;
}
int32_t main() {
    int n;
    cin >> n;
    fact.assign(n + 5, 1);
    invfact.assign(n + 5, 1);
    for (int i = 1; i <= n; i++) {
        fact[i] = (fact[i - 1] * i) % mod;
        invfact[i] = quickpow(fact[i], mod - 2);
    }
    int ans = (n * (fact[n] - 1) % mod) % mod;
    cout << ans << '\n';
}
