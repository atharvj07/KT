#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> s(n);
        for (int i = 0; i < n; i++)
            cin >> s[i];
        sort(s.begin(), s.end());
        int ans = s[n - 1] - s[0];
        for (int i = 1; i < n; i++)
            ans = min(ans, s[i] - s[i - 1]);
        cout << ans << endl;
    }
    return 0;
}
