import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.util.*;

public class Main {
  static int[] tree;
  static int[] minval;
  static int[] nums;
  static int maxlen = 0;
  static int maxb;
  static int t;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
    StringTokenizer st = new StringTokenizer(br.readLine());
    int n = Integer.parseInt(st.nextToken());
    int k = Integer.parseInt(st.nextToken());
    maxb = Integer.parseInt(st.nextToken());
    t = Integer.parseInt(st.nextToken());
    tree = new int[(maxb + 1) * t];
    minval = new int[(maxb + 1) * t];
    nums = new int[n * t];
    for (int tc = 0; tc < k; tc++) {
      st = new StringTokenizer(br.readLine());
      for (int i = 0; i < n; i++) {
        int curnum = Integer.parseInt(st.nextToken());
        nums[i * t] = curnum;
        for (int j = 1; j < t; j++) {
          nums[i * t + j] = (nums[i * t + j - 1] + curnum) % maxb;
        }
      }
      int result = getLIS(nums, n * t);
      bw.write(result + "\n");
    }
    bw.flush();
    bw.close();
  }

  public static int init(int start, int end, int node) {
    if (start == end) {
      minval[node] = nums[start];
      return minval[node];
    }
    int mid = (start + end) / 2;
    int leftmin = init(start, mid, node * 2);
    int rightmin = init(mid + 1, end, node * 2 + 1);
    minval[node] = Math.min(leftmin, rightmin);
    return minval[node];
  }

  public static void update(int start, int end, int node, int index, int newnum) {
    if (index < start || index > end) {
      return;
    }
    tree[node] = 0;
    if (start == end) {
      minval[node] = newnum;
    } else {
      int mid = (start + end) / 2;
      update(start, mid, node * 2, index, newnum);
      update(mid + 1, end, node * 2 + 1, index, newnum);
      minval[node] = Math.min(minval[node * 2], minval[node * 2 + 1]);
    }
  }

  public static int getLIS(int[] nums, int n) {
    init(0, n - 1, 1);
    int answer = 0;
    for (int i = 0; i < n; i++) {
      int left = 0;
      int right = maxlen;
      int low = 0;
      while (left <= right) {
        int mid = (left + right) / 2;
        if (getmin(0, n - 1, 1, nums[i], mid) < nums[i]) {
          low = mid;
          left = mid + 1;
        } else {
          right = mid - 1;
        }
      }
      tree[1] = 0;
      update(0, n - 1, 1, low, nums[i]);
      answer = Math.max(answer, low + 1);
    }
    maxlen = answer;
    return answer;
  }

  public static int getmin(int start, int end, int node, int leftnum, int cnt) {
    if (cnt == 0) {
      return Integer.MAX_VALUE;
    }
    if (start == end) {
      return minval[node];
    }
    int mid = (start + end) / 2;
    if (tree[node * 2] == cnt) {
      return getmin(start, mid, node * 2, leftnum, cnt - 1);
    } else {
      return getmin(mid + 1, end, node * 2 + 1, leftnum, cnt);
    }
  }
}
