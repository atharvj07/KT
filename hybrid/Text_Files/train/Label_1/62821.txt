import queue

n, m = map(int, input().split())

a = input().split()
b = []
lst = [0] * (n + 1)
d = [0] * (n + 1)

for i in range(m):
    x, y = map(int, input().split())
    b.append((x, y))
    lst[x] += 1
    lst[y] += 1

for i in range(1, n + 1):
    if lst[i] == 1:
        q = queue.Queue()
        q.put(i)
        d[i] = 1
        while not q.empty():
            x = q.get()
            for j in range(len(b)):
                if b[j][0] == x:
                    if d[b[j][1]] == 0:
                        d[b[j][1]] = d[x] + 1
                        q.put(b[j][1])
                elif b[j][1] == x:
                    if d[b[j][0]] == 0:
                        d[b[j][0]] = d[x] + 1
                        q.put(b[j][0])

ans = 0
for i in range(1, n + 1):
    ans += int(a[i - 1]) * d[i]

print(ans)
