#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      cnt += a[i] % 3 == 0;
    }
    sort(a.rbegin(), a.rend());
    int sum = cnt;
    for (int i = 0; i < n; i++) {
      if (i + 1 >= a.size()) break;
      if (a[i] > a[i + 1] + a[i + 1]) {
        sum = max(sum, cnt + 1);
        cnt++;
      }
    }
    cout << sum << endl;
  }
  return 0;
}
