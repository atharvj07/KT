from collections import Counter

n = int(input())
arr = list(map(int, input().split()))
c = Counter(arr)

l = list(c)
s = arr.copy()
l.sort()
while len(l) >= 2:
    val = l[0]
    if c[val] >= 2:
        s.remove(val)
        s[s.index(val)] += val
        c[val] -= 2
        l = list(Counter(s))
        l.sort()
    else:
        l.pop(0)

print(len(s))
print(*s)
