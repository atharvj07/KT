#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<long long> p(n);
    for (int i = 0; i < n; i++) cin >> p[i];
    long long sum = 0;
    for (int i = 1; i < n; i++) {
      long long c = (k * (p[0] + sum)) / 100 - p[i];
      if (c > 0) {
        sum += c;
        p[0] += c;
      }
    }
    cout << sum << endl;
  }
}
