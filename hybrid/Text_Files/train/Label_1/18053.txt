#include <bits/stdc++.h>
using namespace std;
int c[20], a, b, n;
int f(int l, int r, int b) {
  if (l > r)
    return 0;
  if (c[l] == c[r])
    return f(l + 1, r - 1, b);
  if (c[l] == 2)
    if (b == 0)
      return f(l + 1, r, b);
    else
      return min(f(l + 1, r, 0) + a, f(l + 1, r, 1) + b);
  if (c[r] == 2)
    if (b == 0)
      return f(l, r - 1, b);
    else
      return min(f(l, r - 1, 0) + a, f(l, r - 1, 1) + b);
  return -1;
}
int main() {
  cin >> n >> a >> b;
  for (int i = 0; i < n; i++) {
    cin >> c[i];
  }
  cout << f(0, n - 1, 0) << endl;
}
