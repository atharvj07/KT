#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mxN=2e5+1;
ll fr[mxN],a[mxN];
int main()
{
    ll n,k,x;
    cin>>n>>k;
    for(ll i=0;i<n;i++)
    {
        cin>>x;
        fr[x]++;
    }
    vector<ll>v;
    for(ll i=1;i<=2*mxN;i++)
    {
        vector<ll>temp;
        for(ll j=i;j<=2*mxN;j+=i)
        {
            while(fr[j]--)
                temp.push_back(j);
        }
        if(temp.size()>=k)
        {
            sort(temp.begin(),temp.end());
            for(ll l=0;l<k;l++)
                cout<<temp[l]<<" ";
            return 0;
        }
    }
}
