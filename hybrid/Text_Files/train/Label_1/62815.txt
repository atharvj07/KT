t=int(input())
for i in range(t):
    a,b,p=map(int,input().split())
    s=input()
    st=list(s)
    l=[0]*len(s)
    r=[0]*len(s)
    for j in range(len(s)-1,-1,-1):
        if s[j]=='A':
            l[j]=1
            if j+1<len(s):
                l[j]+=l[j+1]
    for j in range(1,len(s)):
        if s[j-1]=='B':
            r[j]=1
            if j-1>=0:
                r[j]+=r[j-1]
    ans=len(s)
    for j in range(1,len(s)):
        if l[j]>0 and r[j]>0:
            if b*r[j]+a*l[j]<=p:
                ans=min(ans,j)
    for j in range(1,len(s)):
        if l[j]==0 and r[j]>0:
            if b*r[j]<=p:
                ans=min(ans,j)
    for j in range(1,len(s)):
        if l[j]>0 and r[j]==0:
            if a*l[j]<=p:
                ans=min(ans,j)
    print(ans)
