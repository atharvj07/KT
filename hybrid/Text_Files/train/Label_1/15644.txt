#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main()
{
    int k;
    cin >> k;
    while (k--)
    {
        int n;
        cin >> n;
        string s, t;
        vector<int> swaps;
        cin >> s >> t;
        for (int i = 0; i < n; i++)
        {
            if (s[i] != t[i])
            {
                swaps.push_back(i);
                if (swaps.size() > 2)
                {
                    cout << "No" << endl;
                    goto skip;
                }
            }
        }
        if (swaps.empty() || swaps.size() == 2)
            cout << "Yes" << endl;
        else
            cout << "No" << endl;
    skip:;
    }
}
