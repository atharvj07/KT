#include <iostream>
#include <string>
using namespace std;

int main() {
    string s;
    cin >> s;
    for (int i = 1; i < s.size(); i++) {
        if (s[i] < s[i - 1]) {
            swap(s[i], s[i - 1]);
            if (i > 1 && s[i - 2] > s[i]) {
                swap(s[i - 2], s[i]);
            }
        }
    }
    cout << s << endl;
    return 0;
}
