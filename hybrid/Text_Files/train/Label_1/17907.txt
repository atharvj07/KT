#include <bits/stdc++.h>
using namespace std;
#define PB push_back
#define endl "\n"

struct p { int a, i; };
vector<p> v;
int b[300005];

int main() {
    int n, m, q, x, y, ans;
    cin >> n >> m;
    while (n--) {
        cin >> x;
        v.PB({x, 1});
    }
    while (m--) {
        cin >> x;
        v.PB({x, 0});
    }
    sort(v.begin(), v.end(), [](p &a, p &b) { return a.a > b.a; });
    for (int i = 0; i < v.size(); i++) {
        if (v[i].i == 0)
            b[v[i].a]++, ans = v[i].a;
        else
            if (b[v[i].a] > 0) b[v[i].a]--, ans = v[i].a;
            else break;
    }
    cout << (ans ? ans : -1) << endl;
    return 0;
}
