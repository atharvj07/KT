#include<iostream>
#include<vector>
using namespace std;
int main() {
    int t;
    cin >> t;
    for (int p = 0; p < t; p++) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n);
        long long s = 0, x;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            s += a[i];
        }
        vector<int> ans(m);
        for (int i = 0; i < m; i++) {
            cin >> x;
            if (s < x) ans[i] = -1;
            else {
                ans[i] = (x - s % x) % x;
            }
        }
        for (int i = 0; i < m; i++) cout << ans[i] << " ";
        cout << endl;
    }
}
