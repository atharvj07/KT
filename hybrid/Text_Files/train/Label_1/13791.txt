#include <iostream>
#include <string>

int main() {
  int n, m;
  std::cin >> n >> m;
  int r, c;
  std::cin >> r >> c;
  int x, y;
  std::cin >> x >> y;
  std::string labyrinth[n];
  for (int i = 0; i < n; i++) {
    std::cin >> labyrinth[i];
  }
  int dx[] = {0, 0, 1, -1};
  int dy[] = {1, -1, 0, 0};
  bool visited[n][m];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      visited[i][j] = false;
    }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (labyrinth[i][j] == '*' || visited[i][j]) {
        continue;
      }
      visited[i][j] = true;
      int left = x;
      int right = y;
      int curx = i;
      int cury = j;
      bool possible = true;
      while (possible) {
        possible = false;
        for (int k = 0; k < 4; k++) {
          int newx = curx + dx[k];
          int newy = cury + dy[k];
          if (newx >= 0 && newx < n && newy >= 0 && newy < m && labyrinth[newx][newy] != '*' && !visited[newx][newy]) {
            if (k == 1 && right == 0) {
              continue;
            }
            if (k == 3 && left == 0) {
              continue;
            }
            visited[newx][newy] = true;
            if (k == 0) {
              right--;
            } else if (k == 1) {
              left--;
            }
            curx = newx;
            cury = newy;
            possible = true;
            break;
          }
        }
      }
      ans++;
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
