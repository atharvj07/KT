import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {

    // the size of heap
    private static int H;

    // the array to store elements in the heap
    private static int[] A;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());

        // read the size of heap
        H = Integer.parseInt(st.nextToken());

        // read the elements in the heap
        A = new int[H + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= H; i++) {
            A[i] = Integer.parseInt(st.nextToken());
        }

        // build max-heap
        buildMaxHeap(A);

        // print values of nodes in the max-heap
        for (int i = 1; i <= H; i++) {
            System.out.print(A[i] + " ");
        }
        System.out.println();
    }

    private static void buildMaxHeap(int[] A) {
        // perform maxHeapify in a bottom-up manner
        for (int i = H / 2; i >= 1; i--) {
            maxHeapify(A, i);
        }
    }

    // move the value of A[i] down to leaves to make a sub-tree of node i a max-heap
    private static void maxHeapify(int[] A, int i) {
        // get the left and right child indices
        int l = left(i);
        int r = right(i);

        // select the node which has the maximum value
        int largest = i;
        if (l <= H && A[l] > A[i]) {
            largest = l;
        }
        if (r <= H && A[r] > A[largest]) {
            largest = r;
        }

        // if value of children is larger than that of i, swap A[i] and A[largest] and call maxHeapify recursively

        if (largest != i) {
            int temp = A[i];
            A[i] = A[largest];
            A[largest] = temp;
            maxHeapify(A, largest);
        }
    }

    // get the left child index
    private static int left(int i) {
        return 2 * i;
    }

    // get the right child index
    private static int right(int i) {
        return 2 * i + 1;
    }


}
