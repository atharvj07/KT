for _ in range(int(input())):
    s = input()
    l, r = s.count('('), s.count(')')
    if l != r:
        print('NO')
        continue
    bal = 0
    for ch in s:
        if bal < 0 or (ch == ')' and bal == 0):
            break
        bal += 1 if ch == '(' else -1
    print(['NO', 'YES'][bal == 0])
