import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        char[] line = in.next().toCharArray();
        int steps = 0;
        for (int i = 0; i < n / 2; i++) {
            if(line[i] == 'X') steps++;
        }
        System.out.println(steps);
        for (int i = 0; i < n / 2; i++) {
            if(line[i] == 'X') {
                steps--;
                line[i] = 'x';
                line[n - 1 - i] = 'X';
            } else {
                if(steps > 0) {
                    steps--;
                    line[i] = 'X';
                    line[n - 1 - i] = 'x';
                }
            }
        }
        System.out.println(new String(line));
    }
}
