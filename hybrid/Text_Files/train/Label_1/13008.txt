#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n;
        std::cin >> n;
        std::vector<int> a(n);
        for (int i = 0; i < n; ++i) { std::cin >> a[i]; }
        int ans = 0;
        bool sorted = false;
        while (!sorted) {
            sorted = true;
            for (int j = ans % 2; j < n - 1; j += 2) {
                if (a[j] > a[j + 1]) {
                    std::swap(a[j], a[j + 1]);
                    sorted = false;
                }
            }
            ++ans;
        }
        std::cout << ans - 1 << std::endl;
    }
    return 0;
}
