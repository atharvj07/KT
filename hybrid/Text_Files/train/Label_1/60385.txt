for _ in range(int(input())):
    c=[int(i) for i in input().split()]
    d=[int(i) for i in input().split()]
    a=[]
    b=[]
    for i in range(3):
        for j in range(c[i]): a+=[i]
        for j in range(d[i]): b+=[i]
    a.sort()
    b.sort()
    ans=0
    for i in range(len(a)):
        ans+=(a[i]-b[i])*(a[i]>b[i])
    print(ans)
