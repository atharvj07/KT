import math

for nt in range(int(input())):
    n, k = map(int, input().split())
    taps = list(map(int, input().split()))
    # first two seconds tap watering
    all_wet = [2] * n
    for tap in taps:
        all_wet[max(0, tap - 2):tap + 2] = [1] * (tap - max(0, tap - 2) + 1 + 1)
    # every following second
    i = 3
    while 0 in all_wet:
        for j, w in enumerate(all_wet):
            if w == 0:
                all_wet[max(0, j - i + 1):j + i] = [1] * (j - max(0, j - i + 1) + i + 1)
        i += 1
    print(i - 1)
