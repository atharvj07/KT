#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, a, b, c, d;
  cin >> n >> a >> b >> c >> d;
  int sum = (a + b + c + d) / 4;
  if (sum * 4 != (a + b + c + d)) {
    cout << 0 << endl;
  } else {
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      if (i == b || i == d) continue;
      for (int j = 1; j <= n; j++) {
        if (j == a || j == c) continue;
        if (i != j && i + j <= sum) ans++;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
