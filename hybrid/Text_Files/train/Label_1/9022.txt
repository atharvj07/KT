#include <bits/stdc++.h>

using namespace std;

int main() {
  int q;
  cin >> q;

  while (q--) {
    int l1, r1, l2, r2;
    cin >> l1 >> r1 >> l2 >> r2;

    // check if segments intersect
    if (l1 <= l2 && l2 <= r1) {
      cout << l2 << " " << r2 << endl;
    } else if (l2 <= l1 && l1 <= r2) {
      cout << l1 << " " << r1 << endl;
    } else if (max(l1, r1) > max(l2, r2)) {
      cout << l2 << " " << r2 << endl;
    } else if (max(l2, r2) > max(l1, r1)) {
      cout << l1 << " " << r1 << endl;
    }
  }

  return 0;
}
