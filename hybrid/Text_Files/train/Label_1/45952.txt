import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int l = sc.nextInt();
            int r = sc.nextInt();
            if (l == r) {
                System.out.print(1);
            } else {
                System.out.print("1 ");
                for (int i = l + 1; i <= r; i++) {
                    System.out.print(i % n + 1 + " ");
                }
            }
            System.out.println();
        }
    }
}
