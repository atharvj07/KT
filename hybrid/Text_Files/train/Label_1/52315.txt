import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        Map<String, Integer> actions = new HashMap<>();
        for (int i = 0; i < m; i++) {
            String action = sc.next();
            int team = sc.nextInt();
            actions.put(team + " " + action, i + 1);
        }

        Arrays.sort(s);
        int first = 0, second = 0;
        boolean[] picked = new boolean[n];
        for (int i = 0; i < m; i++) {
            if (actions.get("1 p") <= actions.get("1 b") && actions.get("2 p") <= actions.get("2 b")) {
                int max = s[n - 1];
                int index = n - 1;
                while (picked[index]) {
                    index--;
                    max = s[index];
                }
                if (i % 2 == 0) {
                    first += max;
                } else {
                    second += max;
                }
                picked[index] = true;
            } else if (actions.get("1 p") <= actions.get("1 b") && actions.get("2 p") > actions.get("2 b")) {
                int max = s[n - 1];
                int index = n - 1;
                while (picked[index]) {
                    index--;
                    max = s[index];
                }
                if (i % 2 == 0) {
                    first += max;
                } else {
                    second -= max;
                }
                picked[index] = true;
            } else if (actions.get("1 p") > actions.get("1 b") && actions.get("2 p") <= actions.get("2 b")) {
                int min = s[0];
                int index = 0;
                while (picked[index]) {
                    index++;
                    min = s[index];
                }
                if (i % 2 == 0) {
                    first -= min;
                } else {
                    second += min;
                }
                picked[index] = true;
            } else {
                int min = s[0];
                int index = 0;
                while (picked[index]) {
                    index++;
                    min = s[index];
                }
                if (i % 2 == 0) {
                    first -= min;
                } else {
                    second -= min;
                }
                picked[index] = true;
            }
        }

        System.out.println(first - second);
    }
}
