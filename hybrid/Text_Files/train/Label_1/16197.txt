#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  int x, y;
  set<int> levelX, levelY;
  cin >> x;
  for (int i = 0; i < x; i++) {
    int temp;
    cin >> temp;
    levelX.insert(temp);
  }
  cin >> y;
  for (int i = 0; i < y; i++) {
    int temp;
    cin >> temp;
    levelY.insert(temp);
  }
  set<int> total;
  total.insert(levelX.begin(), levelX.end());
  total.insert(levelY.begin(), levelY.end());
  if (total.size() < n) {
    cout << "Oh, my keyboard!" << endl;
  } else {
    cout << "I become the guy." << endl;
  }
  return 0;
}
