#include <cstdio>
#include <iostream>
#include <cstdlib>
using namespace std;

typedef long long ll;

ll pm(ll x,ll p) {
    if(!p) return 1LL;
    if(p==1LL) return x;
    x=pm(x,p/2LL);
    x*=x;
    if(p%2LL) x*=2LL;
    return x;
}

ll gcd(ll a,ll b) {
    if(!a) return b;
    return gcd(b%a,a);
}

int main() {
    ll n;
    cin >> n;
    for(ll i=2LL;;i+=2LL) {
        ll p=-1,cur=1;
        for(ll j=2LL;j*j<=i;j++) {
            if(!(i%j)) { p=j; break; }
        }
        if(p==-1LL) {
            if(n%i==0LL) {
                while(n%i==0LL) { n/=i; }
            } else {
                if(n%pm(i,2LL)) { puts("-1"); return 0; }
                while(n%pm(i,cur)) { cur++; }
                if(n%pm(i,cur+1LL)) { puts("-1"); return 0; }
                while(n%pm(i,cur+1LL)) { n/=pm(i,cur+1LL); }
            }
        } else {
            if(n%p) { puts("-1"); return 0; }
            while(n%p==0LL) { n/=p; p*=p; }
        }
        if(n==1LL) {
            cout << pm(i,cur) << endl;
            return 0;
        }
    }
    return 0;
}
