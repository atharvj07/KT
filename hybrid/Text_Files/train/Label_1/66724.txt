n, k = map(int, input().split())
a = []
for i in range(n+1):
    ai = input()
    if ai == '?':
        a.append(None)
    else:
        a.append(int(ai))
if n == 1:
    print('Yes')
    exit()
if k == 0 and a[0] != 0:
    print('No')
    exit()
if k == 0:
    print('Yes')
    exit()
cnt = a.count(None)
if cnt == 0:
    print('No')
    exit()
for i in range(1, n+1):
    if a[i] == None:
        if n % 2 == 1:
            a[i] = (k - a[0]) / 2
        else:
            a[i] = (k - a[0]) / 2 % 1
        cnt -= 1
        break
if cnt == 0:
    for i in range(1, n+1):
        if a[i] == None:
            if n % 2 == 1:
                a[i] = (k - a[0]) / 2
            else:
                a[i] = (k - a[0]) / 2 % 1
            cnt -= 1
            break
if cnt == 0:
    sum = 0
    for i in range(0, n+1):
        sum += a[i]
    if sum % k == 0:
        print('No')
    else:
        print('Yes')
else:
    print('Yes')
