#include <cstdio>
#include <vector>
using namespace std;

const int maxn = 1205;
int st[maxn], tp = 0;
int l[maxn], r[maxn];
int n, ans[maxn];

int main() {
    scanf("%d", &n);
    for (int i = 0; i < n; ++i) {
        scanf("%d%d", &l[i], &r[i]);
    }

    for (int i = 0; i < n; ++i) {
        if (!tp || l[i] == r[i]) {
            st[++tp] = i;
            ans[i] = ans[i + r[i]] = 1;
            i += r[i];
        } else {
            if (st[tp] != i - l[i])
                return puts("IMPOSSIBLE"), 0;
            --tp, ans[i] = ans[i + r[i]] = 0;
        }
    }

    if (tp) return puts("IMPOSSIBLE"), 0;

    for (int i = 0; i < 2 * n; ++i) {
        putchar(ans[i] ? '(' : ')');
    }
    puts("");
    return 0;
}
