#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    int ans = 0;
    for (int i = 0; i < s.length(); ++i) {
        if (s[i] == 'n') {
            bool flag = true;
            for (int j = i + 1, k = 0; j < i + 7 && k < 7; ++j, ++k) {
                if (s[j] != "ineteen"[k]) flag = false;
            }
            ans += (flag);
        }
    }
    cout << ans << endl;
}
