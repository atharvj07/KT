import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n + 1];
        int[] a = new int[n + 1];
        int[] d = new int[n + 1];
        int[] s = new int[n + 1];
        long res = 0;
        for (int i = 0; i <= n; i++) {
            h[i] = sc.nextInt();
            a[i] = sc.nextInt();
            d[i] = sc.nextInt();
            s[i] = sc.nextInt();
        }
        boolean[] vis = new boolean[n + 1];
        PriorityQueue<Integer> pq = new PriorityQueue<>((x, y) -> s[y] - s[x]);
        pq.add(0);
        while (!pq.isEmpty()) {
            int cur = pq.poll();
            if (vis[cur])
                continue;
            vis[cur] = true;
            if (cur == 0) {
                res += Math.max(a[0] - d[0], 0);
                h[0] -= Math.max(a[0] - d[0], 0);
                if (h[0] <= 0) {
                    System.out.println(-1);
                    return;
                }
            } else {
                res += Math.max(a[cur] - d[0], 0);
                h[0] -= Math.max(a[cur] - d[0], 0);
                if (h[0] <= 0) {
                    System.out.println(res);
                    return;
                }
            }
            for (int i = 1; i <= n; i++) {
                if (!vis[i] && s[i] < s[cur] && h[i] > 0)
                    pq.add(i);
            }
        }
    }
}
