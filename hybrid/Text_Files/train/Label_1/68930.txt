import heapq

n = int(input())
items = []
saved = []
for i in range(n):
    t, d, p = map(int, input().split())
    if t <= d:
        heapq.heappush(items, (-p, t))

ans, tm = 0, 0
while items and tm <= 2e3:
    p, t = heapq.heappop(items)
    if tm + t <= 2e3:
        ans -= p
        tm += t
        heapq.heappush(saved, -p)
print(ans)
print(len(saved))
print(*sorted(map(lambda x: -x, saved)))
