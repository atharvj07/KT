import math

for _ in range(int(input())):
    n = int(input())
    times = [list(map(int, input().split())) for _ in range(n)]
    tm = list(map(int, input().split()))
    cur = 0
    for a, b in times:
        cur = max(cur, b)
        cur += math.ceil((b - a) / 2) + tm[times.index([a, b])]
    print(cur)
