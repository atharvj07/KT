def solve():
    n, s = map(int, input().split())
    left, right = [], []
    for _ in range(n):
        x, v, t = map(int, input().split())
        if t == 1:
            left.append((x, v))
        else:
            right.append((10**6 - x, v))
    
    def find_time(people, s):
        people.sort(reverse=True)
        max_time = 0
        for i, (dist, speed) in enumerate(people):
            time_before_accel = dist / (s + speed)
            max_time = max(max_time, time_before_accel + i/s)
        return max_time

    left_time = find_time(left, s)
    right_time = find_time(right, s)
    
    print(f"{max(left_time, right_time):.24f}")

solve()
