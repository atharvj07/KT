#include <algorithm>
#include <cmath>
#include <cstdio>
#include <iostream>
#include <vector>
typedef long long ll;
const int N = 100000 + 5, B = 500;
int n;
std::vector<int> a, L[N], R[N], val;
int ind[N];
int ans[N];
bool done[N], used[N];
long long gcd(long long a, long long b) {
  return !b ? a : gcd(b, a % b);
}
void sieve() {
  for (int i = 1; i < N; i++)
    for (int j = i; j < N; j += i) val.push_back(j);
  std::sort(val.begin(), val.end());
  val.resize(unique(val.begin(), val.end()) - val.begin());
}
int g(int x) {
  for (int y : val)
    if (x % y == 0) return y;
  return -1;
}
void add_R(int x) {
  int v;
  for (int i = 0, j = 0; i < n; i++, j++) {
    if (done[i]) {
      v = ans[i];
      j--;
    } else {
      v = a[i];
      if (!used[v]) L[v].push_back(x), R[x].push_back(v), used[v] = 1, j--;
    }
    if (!j) ans[x] = std::max(ans[x], v);
  }
}
int main() {
  sieve();
  scanf("%d", &n);
  a.resize(n);
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  std::sort(a.begin(), a.end());
  std::vector<int> p;
  for (int x : a)
    if (!done[ind[x]] && x != 1) {
      int y = g(x);
      if (y != -1) {
        p.push_back(x);
        p.push_back(y);
        done[ind[x]] = done[ind[y]] = 1;
      }
    }
  std::sort(p.begin(), p.end());
  int lst = -1;
  for (int x : p) {
    if (lst != x) add_R(x), lst = x;
  }
  int qnt = 0;
  for (int i = 0; i < n; i++)
    if (!done[i]) qnt++;
  add_R(1);
  for (int i = 1; i < N; i++)
    if (R[i].size() && !done[ind[i]]) done[ind[i]] = 1, qnt--;
  printf("%d\n", qnt);
  return 0;
}
