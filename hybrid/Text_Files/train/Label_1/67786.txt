import sys

input = sys.stdin.readline

t = int(input())

for _ in range(t):
    n = int(input())
    a = [int(x) for x in input().split()]
    inc = [1] * n
    dec = [1] * n
    for i in range(1, n):
        if a[i] > a[i - 1]:
            inc[i] = inc[i - 1] + 1
    for i in range(n - 2, -1, -1):
        if a[i] < a[i + 1]:
            dec[i] = dec[i + 1] + 1
    ans = False
    for i in range(n):
        if inc[i] == dec[i]:
            ans = True
    if ans: print("YES")
    else: print("NO")
