import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }
        boolean[] vis = new boolean[n];
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (vis[arr[i]]) {
                System.out.println("No");
                return;
            }
            vis[arr[i]] = true;
            cur = arr[i];
            if (cur != i) {
                System.out.println("No");
                return;
            }
        }
        System.out.println("Yes");
    }
}
