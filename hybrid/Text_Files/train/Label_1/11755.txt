#include <bits/stdc++.h>
using namespace std;
const int maxn=(1e6)+10;
bool np[maxn];
int ans=1e9;
int main(){
    int a,b,k;
    cin>>a>>b>>k;
    np[0]=np[1]=true;
    for(int i=2;i<=b;i++){
        if(!np[i]){
            for(int j=i+i;j<=b;j+=i){np[j]=true;}
        }
    }
    vector<int>v={0};
    v.resize(b+1);
    for(int i=2;i<=b;i++){
        if(!np[i]){v[i]++;}
    }
    for(int i=a;i<=b-k+1;i++){
        int cnt=0;
        for(int j=0;j<k;j++){cnt+=v[i+j];}
        if(cnt>=k) ans=min(ans,k);
    }
     for(int i=a+1;i<=b-k+1;i++){
        ans=min(ans,k-(v[i-1]-v[i-k]));
    }
    if(ans==1e9) ans=-1;
    cout<<ans;
}
