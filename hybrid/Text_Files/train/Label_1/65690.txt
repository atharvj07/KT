n, k = map(int, input().split())
a = list(map(int, input().split()))
sm = sum(a)
assert sm >= k
a.sort(reverse=True)
ans = 0
for x in a:
  rem = k
  # The remaining carrots of the previous length
  while rem:
    t = min(rem, x)
    k -= t
    rem -= t
    ans += t*t
if k:
  print(-1)
else:
  print(ans)
