#include <cassert>
#include <iostream>
#include <string>
using namespace std;
int main() {
  int const mod = 1e9 + 7;
  int n;
  string s1, s2;
  cin >> n >> s1 >> s2;
  int cmp = 0;
  for (int i = 0; i < n; i++) {
    cmp |= (s1[i] != '?' && s2[i] != '?' && s1[i] != s2[i]);
  }
  if (cmp) {
    cout << 0 << endl;
  } else {
    long long ans = 1;
    for (int i = 0; i < n; i++) {
      if (s1[i] == '?' || s2[i] == '?') ans = ans * 10 % mod;
      if (s1[i] == s2[i] && s1[i] == '?') ans = ans * 9 % mod;
    }
    cout << ans << endl;
  }
  return 0;
}
