#pragma GCC optimize("O3")
#include<bits/stdc++.h>
using namespace std;typedef long long ll;typedef pair<ll,ll>pi;const ll MOD = 1e9+7;

deque<ll>a,b;
ll ans=LLONG_MIN;
void out() {
    if(a.empty())cout<<-1;
    else cout<<a.front();
    cout<<endl;
}
int main() {
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
    ll n,m,q;cin>>n>>m;
    for(ll i=0,x;i<n;i++)cin>>x,a.push_back(x);
    for(ll i=0,x;i<m;i++)cin>>x,b.push_back(x);
    
    sort(a.begin(),a.end(),greater<ll>());
    sort(b.begin(),b.end());
    cin>>q;

    while(q--) {
        ll x,y;cin>>x;
        if(x==1) {
            cin>>x>>y;
            a[x-1]=y;
        }
        else {
            cin>>x>>y;
            b[x-1]=y;
        }
        
        ans = -1;
        while(!a.empty() && !b.empty()) {
            if(b.front()<a.front())b.pop_front();
            else {
                b.pop_front();
                a.pop_front();
                ans = max(ans,a.front());
            }
        }
        out();
    }
}
