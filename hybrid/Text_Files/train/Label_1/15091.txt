#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n, q;
    cin >> n >> q;
    struct P {
        int l, r;
    };
    P a[q];
    for (int i = 0; i < q; i++) cin >> a[i].l >> a[i].r;
    sort(a, a + q, [](const P &a, const P &b) { return a.r < b.r; });
    int ans = 0, R = 0;
    for (int i = 0; i < q - 2; i++) {
        if (a[i].r > R) {
            ans++;
            R = a[i].r;
        }
    }
    cout << ans * n << endl;
}
