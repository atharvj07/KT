#include <bits/stdc++.h>
using namespace std;
const int maxn=1e5+10;
bool vis[maxn];
int m,ans=0;
struct node{
    int v,L;
};
vector<node>v[maxn];
void dfs(int x,int f,int len){
    vis[x]=true;
    for (int i=0;i<v[x].size();i++)
        if (!vis[v[x][i].v]){
            if (v[x][i].v==f)
                dfs(v[x][i].v,x,v[x][i].L);
            else
                dfs(v[x][i].v,x,len+v[x][i].L);
        }
    ans=max(ans,len);
}
int main()
{
    int n,k,x,y; cin>>n>>k;
    for (int i=1;i<=k;i++){
        cin>>m;
        for (int j=1;j<=m;j++){
            cin>>x;
            if (j>1) v[y].push_back((node){x,j-1}), v[x].push_back((node){y,j-1});
            y=x;
        }
    }
    for (int i=1;i<=n;i++)
        if (!vis[i])
            dfs(i,-1,0);
    cout<<ans<<endl;
    return 0;
}
