from sys import stdin

n = int(stdin.readline())
p = list(zip(map(int, stdin.readline().split()), map(int, stdin.readline().split())))

p.sort(key=lambda x: x[0])

def bin(l, r, x):
    if l >= r:
        return int(l == r and p[l][0] == x)
    m = (l + r) // 2
    if p[m][0] == x:
        return 1 + int(p[m][1] > x) + bin(l, m - 1, x) + bin(m + 1, r, x)
    if p[m][0] < x:
        return bin(m + 1, r, x)
    return bin(l, m - 1, x)

ans = 0

for i in range(n):
    x1, y1 = p[i]
    for j in range(i + 1, n):
        x2, y2 = p[j]
        if y1 > y2:
            y1, y2 = y2, y1
        a = (y2 - y1) / (x2 - x1)
        b = y1 - a * x1
        ans += bin(0, n - 1, (a * -10000 - b) / (a + 1e-9))

print(ans)
