#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 998244353;
const int MAXN = 1e5 + 5;
char s[MAXN];
ll a[MAXN], dp[MAXN][2];
int main() {
	int n; cin >> n;
	cin >> (s + 1);
	for(int i = 1; i <= n; i++) cin >> a[i];
	memset(dp, -1, sizeof dp);
	dp[n][0] = dp[n][1] = 0;
	for(int i = n - 1; i; i--) {
		dp[i][1] = dp[i + 1][1];
		if(s[i] == 'a' || s[i] == 'r' || s[i] == 'd') dp[i][1] = min(dp[i][1], dp[i + 1][0]);
		if(s[i] == 'h') dp[i][1] = min(dp[i][1], dp[i + 1][1] + a[i]);
		if(s[i] == 'r') dp[i][1] = min(dp[i][1], dp[i + 1][1] + a[i]);
		if(dp[i][1] >= mod) dp[i][1] -= mod;
		dp[i][0] = dp[i + 1][0];
		if(s[i] == 'e') dp[i][0] = min(dp[i][0], dp[i + 1][0]);
		if(s[i] == 's') dp[i][0] = min(dp[i][0], dp[i + 1][1] + a[i]);
		if(dp[i][0] >= mod) dp[i][0] -= mod;
	}
	cout << dp[1][0] << endl;
	return 0;
}
