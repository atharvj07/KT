#include<bits/stdc++.h>
using namespace std;
int p[55],r[55];
int find(int x){
    if(p[x]!=x){
        p[x]=find(p[x]);
    }
    return p[x];
}
int main(){
    int n;cin>>n;
    int a[n];
    for(int i=0;i<n;++i){
        p[i]=i,r[i]=1,cin>>a[i];
    }
    for(int i=0;i<n-1;++i){
        for(int j=i+1;j<n;++j){
            if(a[j]==a[i]){
                int ri=find(i),rj=find(j);
                if(ri!=rj){
                    p[ri]=rj;
                }
            }
        }
    }
    map<int,vector<int>>v;
    for(int i=0;i<n;++i){
        v[find(i)].push_back(i);
    }
    int ans=0;
    for(auto &x:v){
        ans+=x.second.size()>0;
    }
    cout<<ans<<endl;
    for(auto &x:v){
        if(x.second.size()>0){
            cout<<(x.second)[0]+1<<" "<<(x.second).back()+1<<endl;
        }
    }
    return 0;
}
