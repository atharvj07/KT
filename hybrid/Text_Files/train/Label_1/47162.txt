import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String str = br.readLine();
        int len = str.length();
        if (len < 3 || len > 5000) {
            System.out.println(-1);
        } else {
            int[] firstCount = new int[26];
            int[] secondCount = new int[26];
            for (int i = 0; i < len; i++) {
                char c = str.charAt(i);
                if (i == 0) {
                    firstCount[c - 'a']++;
                } else {
                    secondCount[c - 'a']++;
                }
            }
            int firstMax = 0;
            int firstMaxCount = 0;
            int secondMax = 0;
            int secondMaxCount = 0;
            for (int i = 0; i < 26; i++) {
                if (firstCount[i] > firstMax) {
                    firstMax = firstCount[i];
                    firstMaxCount = 1;
                } else if (firstCount[i] == firstMax) {
                    firstMaxCount++;
                }
                if (secondCount[i] > secondMax) {
                    secondMax = secondCount[i];
                    secondMaxCount = 1;
                } else if (secondCount[i] == secondMax) {
                    secondMaxCount++;
                }
            }
            if (firstMaxCount == 1 && secondMaxCount == 1) {
                System.out.println(1.0);
            } else if (firstMaxCount == 1 && secondMaxCount > 1) {
                System.out.println(1.0 / secondMaxCount);
            } else {
                System.out.println(0.0);
            }
        }
    }
}
