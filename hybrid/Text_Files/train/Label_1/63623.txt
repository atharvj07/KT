import sys

n = int(input())
l = []
for i in range(n):
    a, b = map(int, input().split())
    l.append([a, b])

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

wcd = 0
for a, b in l:
    if wcd == 0:
        wcd = gcd(a, b)
    else:
        wcd = gcd(wcd, gcd(a, b))

if wcd > 1:
    print(wcd)
else:
    print(-1)
