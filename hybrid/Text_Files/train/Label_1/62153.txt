def count_cyclical_strings(n, s):
    mod = 10**9 + 7
    len_s = len(s)
    overlap = [0] * (len_s + 1)
    for i in range(1, len_s):
        j = overlap[i - 1]
        while j > 0 and s[i] != s[j]:
            j = overlap[j - 1]
        if s[i] == s[j]:
            j += 1
        overlap[i] = j

    dp = [[0] * (len_s + 1) for _ in range(n + 1)]
    dp[0][0] = 1
    for i in range(n):
        for j in range(len_s + 1):
            for k in '01':
                if j == len_s:
                    dp[i + 1][j] += dp[i][j]
                else:
                    nj = j
                    while nj > 0 and s[nj] != k:
                        nj = overlap[nj - 1]
                    if s[nj] == k:
                        nj += 1
                    dp[i + 1][nj] += dp[i][j]
    result = sum(dp[n][len(s):]) % mod
    return result

n = int(input().strip())
s = input().strip()

print(count_cyclical_strings(n, s))
