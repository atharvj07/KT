#include <iostream>
#include <set>
#include <vector>
using namespace std;
int n, m, d, c[1001], a[1001];
bool ok() {
  set<int> p;
  for (int i = 1; i <= n; i++) {
    if (!p.empty() && *p.rbegin() + d >= i) continue;
    bool f = 0;
    for (int j = 1; j <= m; j++) {
      if (i + c[j] - 1 > n) continue;
      bool g = 1;
      for (int k = i; k <= i + c[j] - 1; k++) g &= !p.count(k);
      if (g) p.insert(i + c[j] - 1), f = 1;
    }
    if (!f) return 0;
  }
  int v = 0;
  for (int i = 1; i <= n; i++) {
    if (p.count(i))
      a[i] = ++v;
    else
      a[i] = 0;
  }
  return 1;
}
int main() {
  cin >> n >> m >> d;
  for (int i = 1; i <= m; i++) cin >> c[i];
  cout << (ok() ? "YES\n" : "NO\n");
  for (int i = 1; i <= n; i++) cout << a[i] << ' ';
}
