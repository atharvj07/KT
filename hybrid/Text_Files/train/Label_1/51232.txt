import java.util.*;
import java.io.*;

public class BananaMicrowave {
    static class Operation {
        int type;
        double x;
        int y;

        public Operation(int type, double x, int y) {
            this.type = type;
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        List<Operation> operations = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            double x = Double.parseDouble(st.nextToken()) / 100000;
            int y = Integer.parseInt(st.nextToken());
            operations.add(new Operation(type, x, y));
        }

        int[] earliestTime = new int[m + 1];
        Arrays.fill(earliestTime, -1);
        earliestTime[0] = 0;

        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[0]));
        pq.offer(new int[]{0, 0});

        for (Operation operation : operations) {
            int k = 0;
            while (!pq.isEmpty() && pq.peek()[0] <= k) {
                int[] curr = pq.poll();
                k = curr[1];
            }

            if (operation.type == 1) {
                for (int a = 0; a <= operation.y; a++) {
                    int newK = (int) Math.ceil(k + operation.x * a);
                    if (newK <= m && earliestTime[newK] == -1) {
                        earliestTime[newK] = k + 1;
                        pq.offer(new int[]{k + 1, newK});
                    }
                }
            } else {
                for (int a = 0; a <= operation.y; a++) {
                    int newK = (int) Math.ceil(k * Math.pow(operation.x, a));
                    if (newK <= m && earliestTime[newK] == -1) {
                        earliestTime[newK] = k + 1;
                        pq.offer(new int[]{k + 1, newK});
                    }
                }
            }
        }

        for (int i = 1; i <= m; i++) {
            System.out.print(earliestTime[i] + " ");
        }
    }
}
