#include <iostream>
using namespace std;

int main() {
    int x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2;
    int x, y;
    cin >> x >> y;
    bool possible = false;
    if ((y2 - y1) % y == 0 && (x2 - x1) % x == 0) {
        if ((x2 - x1) / x == (y2 - y1) / y) possible = true;
    }
    cout << (possible ? "YES" : "NO") << endl;
}
