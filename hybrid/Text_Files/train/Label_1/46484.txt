import java.util.Scanner;

public class Main {
    static int[][] matrix = new int[5][5];

    static boolean isProperNet() {
        int[] freq = new int[7];
        for (int i = 0; i < 5; i++) {
            for (int j = 0; j < 5; j++) {
                int num = matrix[i][j];
                if (num < 0 || num > 6) return false;
                freq[num]++;
            }
        }
        for (int i = 1; i < 7; i++) {
            if (freq[i] != 2) return false;
        }

        for (int i = 0; i < 5; i++) {
            if (matrix[i][0] + matrix[i][4] != 7) return false;
        }
        for (int i = 0; i < 5; i++) {
            if (matrix[0][i] + matrix[4][i] != 7) return false;
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    matrix[j][k] = scanner.nextInt();
                }
            }
            System.out.println(isProperNet() ? "true" : "false");
        }
    }
}
