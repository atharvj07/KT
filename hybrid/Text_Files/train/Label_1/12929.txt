#include <iostream>

using namespace std;

int main() {
    int a, b, c;
    cin >> a >> b >> c;
    if (a == 1 || b == 1 || c == 1 || (a + b + c) % 2)
        cout << "Impossible";
    else if (a > 1 && b > 1) {
        if (c % 2 == 0)
            cout << 1 << " " << b - 1 << " " << c / 2;
        else
            cout << 1 << " " << b - 2 << " " << c / 2 + 1;
    } else if (a > 1 && c > 1) {
        if (b % 2 == 0)
            cout << a - 1 << " " << 1 << " " << b / 2;
        else
            cout << a - 2 << " " << 1 << " " << b / 2 + 1;
    } else if (b > 1 && c > 1) {
        if (a % 2 == 0)
            cout << a / 2 << " " << 1 << " " << c - 1;
        else
            cout << a / 2 + 1 << " " << 1 << " " << c - 2;
    }
    return 0;
}
