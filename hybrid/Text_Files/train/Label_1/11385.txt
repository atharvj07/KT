#include <algorithm>
#include <iostream>

int main() {
  int q;
  std::cin >> q;
  while (q--) {
    long long n;
    std::cin >> n;
    if (n == 1)
      std::cout << 0 << std::endl;
    else if (n < 10)
      std::cout << -1 << std::endl;
    else {
      int ans = 0;
      while (n > 1) {
        if (n % 2 == 0 && n % 3 == 0 && n % 5 == 0)
          n /= 3, ans++;
        else if (n % 5 == 0)
          n /= 5, ans++;
        else if (n % 2 == 0)
          n /= 2, ans++;
        else if (n % 3 == 0)
          n /= 3, ans++;
        else
          break;
      }
      if (n > 1)
        std::cout << -1 << std::endl;
      else
        std::cout << ans << std::endl;
    }
  }
  return 0;
}
