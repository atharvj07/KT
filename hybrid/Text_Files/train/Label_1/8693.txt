#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
#define _GLIBCXX_DEBUG
int main()
{
    int n;
    cin >> n;
    string s;
    cin >> s;
    int cnt[3] = {};
    rep(i, n) cnt[s[i] - '0']++;
    if (cnt[0] == cnt[1] && cnt[1] == cnt[2])
    {
        cout << s << endl;
        return 0;
    }
    int m = min({cnt[0], cnt[1], cnt[2]});
    cnt[0] -= m;
    cnt[1] -= m;
    cnt[2] -= m;
    s = "";
    rep(i, m) s += '2';
    while (cnt[1] > 0 && cnt[2] > 0)
    {
        s += '1';
        s += '2';
        cnt[1]--;
        cnt[2]--;
    }
    while (cnt[0] > 0 && cnt[1] > 0)
    {
        s += '0';
        s += '1';
        cnt[0]--;
        cnt[1]--;
    }
    while (cnt[0] > 0 && cnt[2] > 0)
    {
        s += '0';
        s += '2';
        cnt[0]--;
        cnt[2]--;
    }
    cout << s << endl;
    return 0;
}
