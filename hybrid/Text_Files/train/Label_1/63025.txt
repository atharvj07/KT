def min_moves_to_rearrange_stones(n, initial_positions, diagonals_colors):
    from collections import deque
    
    # Convert initial positions from 1-indexed to 0-indexed
    initial_positions = [p - 1 for p in initial_positions]
    target_positions = {0, 1, 2}
    visited = set()
    visited.add(tuple(sorted(initial_positions)))
    
    # BFS queue: (current positions of stones, moves count, path taken)
    queue = deque([(initial_positions, 0, [])])
    
    while queue:
        current_positions, moves, path = queue.popleft()
        
        # Check if current positions match target positions
        if set(current_positions) == target_positions:
            return (moves, path)
        
        # Try moving each stone to all other positions
        for i, pos in enumerate(current_positions):
            for next_pos in range(n):
                if next_pos not in current_positions:  # Ensure it's a free position
                    # Check if move is valid by color matching
                    if all(diagonals_colors[pos][next_pos] == diagonals_colors[current_positions[j]][current_positions[(j + 1) % 3]]
                           for j in range(3) if j != i):
                        new_positions = current_positions[:i] + [next_pos] + current_positions[i+1:]
                        new_positions_sorted = tuple(sorted(new_positions))
                        if new_positions_sorted not in visited:
                            visited.add(new_positions_sorted)
                            queue.append((new_positions, moves + 1, path + [(pos + 1, next_pos + 1)]))
    
    return -1, []

n = int(input().strip())
initial_positions = list(map(int, input().split()))
diagonals_colors = [input().strip() for _ in range(n)]

moves, path = min_moves_to_rearrange_stones(n, initial_positions, diagonals_colors)

if moves == -1:
    print(-1)
else:
    print(moves)
    for move in path:
        print(*move)
