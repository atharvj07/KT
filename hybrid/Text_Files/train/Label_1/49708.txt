import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        int[] oldPath = new int[m + 1];
        for (int i = 0; i <= m; i++) {
            oldPath[i] = sc.nextInt();
        }

        List<Integer>[] graph = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            int a = oldPath[i];
            int b = oldPath[i + 1];
            graph[a].add(b);
            graph[b].add(a);
        }

        int[] newPath = new int[m + 1];
        boolean found = false;

        for (int i = 0; i < m; i++) {
            int a = oldPath[i];
            int b = oldPath[i + 1];

            if (graph[a].size() == 1) {
                newPath[i] = a;
                newPath[i + 1] = b;
            } else if (graph[b].size() == 1) {
                newPath[i] = b;
                newPath[i + 1] = a;
            } else {
                found = true;
                break;
            }
        }

        if (found) {
            System.out.println("No solution");
        } else {
            for (int i : newPath) {
                System.out.print(i + " ");
            }
        }
    }
}
