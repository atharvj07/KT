#include <bits/stdc++.h>
using namespace std;

int main() {
  long long n, m, x, y, vx, vy;
  cin >> n >> m >> x >> y >> vx >> vy;
  if (vx == 0) {
    if (x == n)
      cout << n << " " << y << endl;
    else
      cout << -1 << endl;
  } else if (vy == 0) {
    if (y == m)
      cout << x << " " << m << endl;
    else
      cout << -1 << endl;
  } else if (vx > 0) {
    if ((x == n && vy > 0) || (y == m && vy < 0)) {
      cout << n << " " << m << endl;
    } else if ((y == 0 && vy > 0) || (x == 0 && vy < 0)) {
      cout << 0 << " " << 0 << endl;
    } else {
      cout << -1 << endl;
    }
  } else {
    if ((x == 0 && vy > 0) || (y == m && vy < 0)) {
      cout << 0 << " " << m << endl;
    } else if ((y == 0 && vy > 0) || (x == n && vy < 0)) {
      cout << n << " " << 0 << endl;
    } else {
      cout << -1 << endl;
    }
  }
}
