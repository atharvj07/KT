import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            int bi = sc.nextInt();
            long li = sc.nextLong();
            long ri = sc.nextLong();
            long ans = 0;
            for (long i = li; i <= ri; i++) {
                if (isMagic(i, bi))
                    ans++;
            }
            System.out.println(ans);
        }
    }

    public static boolean isMagic(long num, int bi) {
        long[] freq = new long[bi];
        while (num != 0) {
            long rem = num % bi;
            freq[(int) rem]++;
            num /= bi;
        }
        for (long i = 0; i < bi; i++) {
            if (freq[(int) i] % 2 == 1)
                return false;
        }
        return true;
    }
}
