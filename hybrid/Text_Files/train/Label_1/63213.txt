n, m, k = map(int, input().split())
s = input()
t = input()
ans = 0
for i in range(n - k + 1):
    for j in range(m - k + 1):
        ok = True
        for l in range(k):
            if s[i + l] != t[j + l]:
                ok = False
        if ok:
            ans = max(ans, i + k + m - j - k)
print(ans)
