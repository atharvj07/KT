n = int(input().strip())
s = input().strip()

count = {'D': 0, 'R': 0}
for char in s:
    count[char] += 1

idx = 0
while count['D'] > 0 and count['R'] > 0:
    if s[idx] == 'D':
        if count['R'] > 0:
            count['R'] -= 1
            s += 'D'
    elif s[idx] == 'R':
        if count['D'] > 0:
            count['D'] -= 1
            s += 'R'
    idx += 1

print('D' if count['D'] > 0 else 'R')
