#include <bits/stdc++.h>
using namespace std;
const int maxn=2e5+5;
vector<int>adj[maxn],adj2[maxn];
int par[maxn],a[maxn],w[maxn];
int32_t main(){
	int n,m,x,y,z;
	cin>>n>>m;
	for(int i=1;i<=n;++i)cin>>a[i],par[i]=i;
	while(m--){
		cin>>x>>y>>z;
		w[x]=w[y]=max(w[x],z);
		adj[x].push_back(y),adj[y].push_back(x);
	}
	function<int(int)>find=[&](int x){
		if(x==par[x])return x;
		return par[x]=find(par[x]);
	};
	auto uni=[&](int x,int y){
		x=find(x),y=find(y);
		if(x!=y)par[x]=y;
	};
	for(int i=1;i<=n;++i)
		for(auto x:adj[i])
			if(find(i)!=find(x))uni(i,x);
	priority_queue<pair<int,int>,vector<pair<int,int>>,greater<pair<int,int>>>pq;
	for(int i=1;i<=n;++i)
		if(adj[i].empty())pq.push({a[i],i});
	long long ans=0;
	while(!pq.empty()){
		x=pq.top().second,pq.pop();
		for(auto y:adj2[x])
			if(find(x)!=find(y)){
				ans+=max(a[x],a[y])+w[x]+w[y];
				uni(x,y);
				for(auto z:adj[y])
					if(find(y)!=find(z))pq.push({a[z],z});
				break;
			}
	}
	cout<<ans<<'\n';
}
