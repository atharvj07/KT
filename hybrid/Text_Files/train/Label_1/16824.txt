#include <bits/stdc++.h>
using namespace std;
#define FOR(i,a,b) for(int i=a;i<b;++i)
#define REP(i,a,b) for(int i=a;i<=b;++i)
#define f first
#define s second
typedef pair<int,int> pi;
typedef long long ll;
bool mp[1005][1005];
const int mod=1e9+7;
int32_t main(){
  string a,b;
  cin>>a>>b;
  int n=a.size(),m=b.size();
  REP(i,0,m){
    int j=i;
    for(;j<n+i;++j) mp[0][j-i]=a[j-i]==b[j];
  }
  REP(i,1,n){
    REP(j,0,m){
      if(j-i>=0&&mp[i-1][j-i]==1){mp[i][j]=mp[i][j]|(mp[i][j-1]||mp[i-1][j]);}
      else{mp[i][j]=mp[i][j-1];}
    }
  }
  int ans=0;
  REP(i,0,m){
    if(mp[n][i]){ans++;}
  }
  cout<<ans<<'\n';
}
