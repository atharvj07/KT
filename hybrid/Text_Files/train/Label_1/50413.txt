import java.util.*;
import java.math.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x = in.nextInt();
        int y = in.nextInt();
        long[] ai = new long[n];
        for (int i = 0; i < n; i++) {
            ai[i] = in.nextLong();
        }
        for (int i = 0; i < n; i++) {
            long a = ai[i];
            double t = a / x;
            double s = a / y;
            if (t == s) {
                System.out.println("Both");
            } else if (t < s) {
                System.out.println("Vanya");
            } else {
                System.out.println("Vova");
            }
        }
    }
}
