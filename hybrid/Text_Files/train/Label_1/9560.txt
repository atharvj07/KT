#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> d(n);
  for (int i = 0; i < n; i++) {
    cin >> d[i];
  }

  vector<long long> pref(n + 1, 0);
  for (int i = 0; i < n; i++) {
    pref[i + 1] = pref[i] + d[i];
  }

  long long ans = 0;
  for (int i = 1; i < n - 1; i++) {
    long long left = pref[i];
    long long right = pref[n] - pref[i + 1];
    if (left == right) {
      ans = max(ans, left);
    }
  }

  cout << ans << endl;

  return 0;
}
