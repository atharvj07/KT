import java.io.*;
import java.util.*;

public class Main {
  static int[] subtreeCount;
  static int[] size;
  static ArrayList<Integer>[] adj;
  static int res = 0, ans = 0;

  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    subtreeCount = new int[n + 1];
    size = new int[n + 1];
    adj = new ArrayList[n + 1];
    for (int i = 0; i <= n; i++) {
      adj[i] = new ArrayList<>();
    }
    for (int i = 0; i < n - 1; i++) {
      StringTokenizer st = new StringTokenizer(br.readLine());
      int u = Integer.parseInt(st.nextToken());
      int v = Integer.parseInt(st.nextToken());
      adj[u].add(v);
      adj[v].add(u);
    }
    dfs1(1, 0);
    dfs2(1, 0);
    System.out.println(ans);
  }

  static void dfs1(int u, int p) {
    subtreeCount[u] = 1;
    size[u] = 1;
    for (int v : adj[u]) {
      if (v != p) {
        dfs1(v, u);
        subtreeCount[u] += subtreeCount[v];
        size[u] += size[v];
      }
    }
  }

  static void dfs2(int u, int p) {
    for (int v : adj[u]) {
      if (v != p) {
        ans = Math.max(ans, subtreeCount[v] * (size[u] - subtreeCount[v]));
        dfs2(v, u);
      }
    }
    ans = Math.max(ans, subtreeCount[u] * (size[u] - subtreeCount[u]));
  }
}
