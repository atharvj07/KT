#include <bits/stdc++.h>
using namespace std;
#define fast ios::sync_with_stdio(0), cin.tie(0), cout.tie(0)
typedef long long ll;
const int N = 55;
int n, arr[N];


int solve() {
    sort(arr, arr + n);
    int ans = INT_MAX;
    for (int i = 1; i < n; i++) {
        int mx = arr[i - 1], mn = arr[i];
        ans = min(ans, mn - mx);
    }
    return ans;
}

int32_t main() {
    fast;
    int tc;
    cin >> tc;
    while (tc--) {
        cin >> n;
        for (int i = 0; i < n; i++) cin >> arr[i];
        cout << solve() << "\n";
    }
    return 0;
}
