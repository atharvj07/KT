#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k, m;
    cin >> n >> k >> m;
    vector<int> t(k);
    for (int i = 0; i < k; ++i) cin >> t[i];
    sort(t.begin(), t.end());
    int ans = 0;
    for (int i = 0; i < k && m >= t[i]; ++i) {
        m -= t[i];
        ++ans;
    }
    ans += m / t[0];
    cout << ans * (m % t[0] ? 1 : 2) << endl;
    return 0;
}
