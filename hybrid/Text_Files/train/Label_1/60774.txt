import sys,math
n=int(sys.stdin.readline().strip())
for i in range(n):
    a=input().strip()
    l=len(a)
    L=[]
    for k in range(l):
        a1=a[:k]+a[k+1:]
        if int(a1)%2==0:
           L.append(k+1)
    if len(L)>0:
        print(min(L)+1)
    else:
        lstDig=a[l-1]
        a=a[0:l-1]
        if (int(a)%2==0 and int(lstDig)%2==0):
            print("1")
        else:
            print("2")
