#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  if (n % 4 == 2 || n % 4 == 3)
    cout << "NO\n";
  else {
    cout << "YES\n";
    for (int i = 1; i <= n; i++) {
      if (i % 2) {
        cout << n << " " << i << " ";
      } else {
        cout << i - 1 << " " << n + i << " ";
      }
    }
  }
}
