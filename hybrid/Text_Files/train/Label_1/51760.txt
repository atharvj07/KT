import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();
        int hh = sc.nextInt();
        int mm = sc.nextInt();

        int ans = (hh * 60 + mm) % x;
        if (ans == 0) ans = x;
        while (true) {
            if (ans % 10 == 7 || ans / 10 % 10 == 7 || ans / 60 % 10 == 7 || ans / 60 / 10 % 10 == 7) break;
            ans += x;
        }
        System.out.println((ans - (hh * 60 + mm)) / x);
    }
}
