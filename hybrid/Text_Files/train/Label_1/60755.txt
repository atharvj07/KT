n, m = map(int, input().split())
prices = list(map(int, input().split()))
auctions = list(map(int, input().split()))

# Convert auctions indices to zero-based and sort prices of auction questions in descending order
auction_prices = sorted([prices[i - 1] for i in auctions], reverse=True)
# Sort all question prices in ascending order for optimal play (non-auction questions first)
prices.sort()

points = sum(prices[:-m])  # Sum of non-auction question prices
current_points = points

# Play auction questions optimally
for price in auction_prices:
    if current_points > price:
        # Increase the auction question price to current points if possible
        points += current_points
    else:
        # Otherwise, add the original price
        points += price
    current_points = points  # Update current points after each auction question

print(points)
