#include<bits/stdc++.h>
#define ll long long
using namespace std;

ll n, k;
string chips, new_chips;

int main(){
	cin >> n >> k;
	cin >> chips;
	new_chips = chips;
	while(k--){
		for(ll j = 1; j <= n; j++){
			ll i = (j + n - 1)%n + 1, l = (j - 1 + n)%n + 1;
			ll white = 0;
			if(chips[i - 1] == 'W')white++;
			if(chips[j - 1] == 'W')white++;
			if(chips[l - 1] == 'W')white++;
			if(white > 1)new_chips[j - 1] = 'W';
			else new_chips[j - 1] = 'B';
		}
		chips = new_chips;
	}
	cout << new_chips << endl;
	return 0;
}
