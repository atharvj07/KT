#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, p, k;
    cin >> n >> p >> k;
    vector<long long> a(n);
    for (auto &i : a) cin >> i;
    sort(a.begin(), a.end());
    long long ans = 0;
    while (p >= a[ans] && k > 0) {
      k = k - 1;
      p = p - a[ans];
      ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
