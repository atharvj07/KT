n, m, k = map(int, input().split())
blocked = list(map(int, input().split())) + [n]
costs = list(map(int, input().split()))

# Calculate the maximum stretch of unblocked street
max_unblocked = max(blocked[0]+1, *(blocked[i+1] - blocked[i] - 1 for i in range(m)))
if max_unblocked == 0:
    print(-1)
else:
    # Find the minimum cost to cover the max_unblocked length
    min_cost = -1
    for power, cost in enumerate(costs, 1):
        if power >= max_unblocked:
            if min_cost == -1 or cost < min_cost:
                min_cost = cost
    if min_cost == -1:
        print(-1)
    else:
        # Calculate the number of lamps needed
        lamps_needed = (n + power - 1) // power
        print(lamps_needed * min_cost)
