import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            long n = scanner.nextLong();
            long k = scanner.nextLong();
            long b = (k - 1) % (2 * n - 2);
            if (b >= n) {
                b = b - n + 1;
            }
            System.out.println(b);
        }
    }
}
