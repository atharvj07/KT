n,m,k=map(int,input().split())
a=[list(map(int,input().split())) for i in range(n)]
ans=[[0] * (m+1) for i in range(n+1)]
for i in range(n-1,-1,-1):
    for j in range(m):
        if a[i][j]:
            ans[i][j]=1+min(ans[i+1][j],ans[i+1][j+1])
z=ans[0][0]
for j in range(m):
    for i in range(n):
        if not a[i][j] and z<1+min(ans[i+1][j],ans[i+1][j+1]):
            z=1+min(ans[i+1][j],ans[i+1][j+1])
            ans=[[-1] * (m+1) for i in range(n+1)]
            ans[0][0]=z
            a[i][j]=1
            break
print(z,z-ans[0][0])
