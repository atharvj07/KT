import math
PI = math.acos(-1)
n = int(input())
v = []; nn = 0
for i in range(n):
    x, y = map(int, input().split())
    if x != 0:
        ang = math.atan(y/x)
        if x < 0: ang += PI
        if ang < 0: ang += 2*PI
        v.append((ang, i+1))
        nn += 1
        if nn == 1: pans = ang
    elif y > 0:
        v.append((PI/2, i+1))
        nn += 1
        if nn == 1: pans = PI/2
v.sort()
mn = 2*PI
for i in range(nn):
    ang1 = v[i][0]
    ang2 = ang1 + PI
    if ang2 > 2*PI: ang2 -= 2*PI
    for j in range(i+1, nn):
        ang3 = v[j][0]
        if ang1 <= ang3 < ang1+PI or ang2 <= ang3 < ang2+PI:
            if ang2-ang1 < ang3-ang1: t1 = ang2
            else: t1 = ang3
            if t1-ang1 < mn:
                mn = t1 - ang1
                ind = [v[i][1], v[j][1]]
print(*ind)
