#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, x;
    cin >> n;
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        cin >> x;
        if (x == 1) sum += 1;
        if (x == 2) sum += 2;
        if (x == 3) sum += 1;
    }
    if (sum != 0)
        cout << sum << endl;
    else
        cout << -1 << endl;
    return 0;
}
