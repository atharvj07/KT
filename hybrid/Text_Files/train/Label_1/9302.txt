#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<int> dp(n + 1);
    vector<int> take(n + 1);
    for (int i = n - 1; i >= 0; i--) {
        for (int j = k; j >= 1; j--) {
            if (dp[j] < dp[j - 1] + a[i]) {
                dp[j] = dp[j - 1] + a[i];
                take[j] = i;
            }
        }
    }
    cout << dp[k] << endl;
    int j = k;
    while (j > 0) {
        int i = take[j];
        cout << i - take[j] + 1 << " ";
        j--;
    }
    cout << endl;
    return 0;
}
