import sys
n,m=map(int,input().split())
e=[[] for _ in range(n+1)]
for _ in range(m):
    x,y=map(int,input().split())
    e[x]+=[y]
    e[y]+=[x]
used=[False]*(n+1)
def dfs(cur,pre,dist):
    used[cur]=True
    mxdist[0]=max(mxdist[0],dist)
    nodes.append(cur)
    for u in e[cur]:
        if u!=pre and not used[u]:
            dfs(u,cur,dist+1)
