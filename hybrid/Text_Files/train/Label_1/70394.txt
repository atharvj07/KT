n, q = map(int, input().split())
word = input()
religions = ["" for _ in range(3)]
for _ in range(q):
    cmd, *args = input().split()
    if cmd == "+":
        i, c = args
        i = int(i)
        religions[i - 1] += c
    else:
        i = int(i)
        religions[i - 1] = religions[i - 1][:-1]
    print("YES" if all(r in word for r in religions) and sum(map(len, religions)) == len(word) else "NO")
