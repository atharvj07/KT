import java.util.Scanner;

public class ArithmeticProgression {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read input
        int N = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();
        int M = scanner.nextInt();

        int[] A = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = a + i * d;
        }

        // Process statements
        for (int i = 0; i < M; i++) {
            int xi = scanner.nextInt();
            int yi = scanner.nextInt() - 1;
            int zi = scanner.nextInt() - 1;

            if (xi == 0) {
                reverse(A, yi, zi);
            } else if (xi == 1) {
                increment(A, yi, zi);
            } else {
                halve(A, yi, zi);
            }
        }

        // Read K and output the K-th item
        int K = scanner.nextInt() - 1;
        System.out.println(A[K]);
    }

    // Reverse the order of values in the interval [yi, zi]
    private static void reverse(int[] A, int yi, int zi) {
        for (int i = yi, j = zi; i < j; i++, j--) {
            int temp = A[i];
            A[i] = A[j];
            A[j] = temp;
        }
    }

    // Increment each value by 1 in the interval [yi, zi]
    private static void increment(int[] A, int yi, int zi) {
        for (int i = yi; i <= zi; i++) {
            A[i]++;
        }
    }

    // Halve each value in the interval [yi, zi] (rounded down)
    private static void halve(int[] A, int yi, int zi) {
        for (int i = yi; i <= zi; i++) {
            A[i] /= 2;
        }
    }
}
