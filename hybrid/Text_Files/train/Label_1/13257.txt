#include <iostream>
#include <vector>
using namespace std;

int main() {
  int a, b, x;
  cin >> a >> b >> x;
  vector<int> v;
  for (int i = 0, first = 1; i < b && first; i++) {
    if (i == x) {
      if (v.empty()) {
        v.push_back(1);
        v.push_back(0);
      } else {
        v.insert(v.begin(), 1);
      }
      first = 0;
      continue;
    }
    v.push_back(1);
  }
  for (int i = 0; i < a; i++) {
    if (i == x) {
      if (v.size() == 1) {
        v.push_back(0);
        v.push_back(1);
      } else {
        v.insert(v.end() - 1, 0);
      }
      continue;
    }
    v.push_back(0);
  }
  for (int x : v) { cout << x; }
  return 0;
}
