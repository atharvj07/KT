#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int sz[maxn], ans[maxn], par[maxn], n, k;
vector<int> gr[maxn];
bool vis[maxn];

void dfs(int u){
    sz[u] = 1;
    for(int v : gr[u]){
        if(v == par[u] or vis[v])   continue;
        par[v] = u;
        dfs(v);
        sz[u] += sz[v];
    }
}

void dfs2(int u, int y){
    vis[u] = 1;
    ans[u] = y;
    int mx = 0, id = -1;
    for(int v : gr[u]){
        if(v == par[u])    continue;
        if(sz[v] > mx)  mx = sz[v], id = v;
    }
    if(id == -1){
        return;
    }
    dfs2(id, y);
    for(int v : gr[u]){
        if(v == par[u] or v == id)    continue;
        dfs2(v, y + 1);
    }
}

int main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> k;
    for(int i = 0, x, y; i < n - 1; i++){
        cin >> x >> y;
        gr[x].push_back(y);
        gr[y].push_back(x);
    }
    dfs(1);
    dfs2(1, 0);
    int cnt = 0;
    for(int i = 2; i <= n; i++){
        if(ans[i] != ans[par[i]])    cnt++;
    }
    if(cnt > k) cout << -1 << endl;
    else{
        cnt = 1;
        for(int i = 1; i <= n; i++){
            if(ans[i] == 0)   continue;
            if(ans[i] == ans[par[i]])  continue;
            cnt++;
        }
        cout << cnt << endl;
        for(int i = 1; i <= n - 1; i++)
            cout << ans[i] + 1 << ' ';
        cout << endl;
    }
    return 0;
}
