#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
ll pw[100001];

int main() {
    pw[0] = 1;
    for(int i = 1; i <= 100000; i++) pw[i] = (pw[i - 1] * 2) % mod;
    int q;
    cin >> q;
    while(q--) {
        int n, f;
        cin >> n >> f;
        cout << pw[n - f] << '\n';
    }
    return 0;
}
