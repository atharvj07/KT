#include <iostream>
#include <vector>
#include <climits>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<long long int> a(n), b(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];

  long long int ans = LLONG_MAX;
  for (int i = 0; i < n; i++) {
    ans = min(ans, k / a[i] * b[i]);
  }
  cout << ans << endl;

  return 0;
}
