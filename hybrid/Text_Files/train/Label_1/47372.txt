import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), m = in.nextInt();
    int[][] bans = new int[m][2];
    for (int i = 0; i < m; ++i) {
      bans[i][0] = in.nextInt();
      bans[i][1] = in.nextInt();
    }
    Arrays.sort(bans, (a, b) -> {
      if (a[0] != b[0]) return a[0] - b[0];
      return a[1] - b[1];
    });
    int ans = 0, r = 0, c = 0;
    for (int i = 0, j = 0; i < n; ++i) {
      while (j < m && bans[j][0] == i + 1) ++j;
      int min = 1, max = n;
      if (c > 1) min = c - 1;
      if (r + 1 < n) max = r + 1;
      if (j == m || bans[j][1] < min || bans[j][1] > max) {
        if (min <= r && max >= r) ++ans;
      }
      if (r + 1 < n) {
        ++r;
        c = max;
      } else {
        ++c;
        r = max;
      }
    }
    System.out.println(ans);
  }
}
