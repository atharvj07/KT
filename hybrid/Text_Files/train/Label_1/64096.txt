def manhattan_distance(x1, y1, x2, y2):
    return abs(x1 - x2) + abs(y1 - y2)

def find_optimal_restaurant(N, M, hotels, restaurants):
    optimal_distance = float('inf')
    optimal_restaurant = None
    for i, (rx, ry) in enumerate(restaurants, start=1):
        max_distance = max(manhattan_distance(hx, hy, rx, ry) for hx, hy in hotels)
        if max_distance < optimal_distance:
            optimal_distance = max_distance
            optimal_restaurant = i
    return optimal_distance, optimal_restaurant

N, M = map(int, input().split())
C = int(input())
hotels = [tuple(map(int, input().split())) for _ in range(C)]
H = int(input())
restaurants = [tuple(map(int, input().split())) for _ in range(H)]

distance, restaurant = find_optimal_restaurant(N, M, hotels, restaurants)
print(distance)
print(restaurant)
