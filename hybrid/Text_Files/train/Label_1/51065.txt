import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for(int i=0; i<n; i++){
            a[i] = sc.nextInt();
        }
        sc.close();
        Arrays.sort(a);
        int ans = 0;
        for(int i=0; i<n; i++){
            if(a[i] == 0){
                continue;
            }
            int j = i + 1;
            int k = i + 2;
            while(j < n && k < n){
                if(a[i] > 0 && a[j] > 0 && a[k] > 0){
                    a[i]--;
                    a[j]--;
                    a[k]--;
                    ans++;
                } else {
                    break;
                }
                j++;
                k++;
            }
        }
        System.out.println(ans);
    }
}
