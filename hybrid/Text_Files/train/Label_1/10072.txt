#include <cstdio>
#include <iostream>
#include <cstring>
int main() {
  int t, n, i, j;
  std::cin >> t;
  while (t--) {
    std::cin >> n;
    if (n == 1) {
      printf("0\n");
      continue;
    }
    j = 1;
    for (i = 2; i * i <= n; i++) {
      if (n % i == 0) {
        j = 0;
        j += 1;
        while (n % i == 0) n /= i;
      }
    }
    if (n > 1) j++;
    printf("%d\n", j);
  }
  return 0;
}
