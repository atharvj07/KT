#include <iostream>
#include <string>
#include <set>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n;
  cin >> n;
  set<char> st;
  for (char c : s)
    st.insert(c);
  while (n--) {
    int type;
    cin >> type;
    if (type == 1) {
      int pos;
      char c;
      cin >> pos >> c;
      if (s[pos - 1] != c) {
        st.erase(s[pos - 1]);
        st.insert(c);
        s[pos - 1] = c;
      }
    } else {
      int l, r;
      cin >> l >> r;
      r = min(r, (int)s.size());
      cout << st.size() << endl;
    }
  }
  return 0;
}
