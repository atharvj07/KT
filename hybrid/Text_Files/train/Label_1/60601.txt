n, h, l, r = map(int, input().split())
a = list(map(int, input().split()))

# Dynamic Programming table
dp = [[-1 for _ in range(h)] for _ in range(n + 1)]
dp[0][0] = 0

for i in range(1, n + 1):
    for j in range(h):
        for k in range(2):  # 0 for sleeping after a[i] hours, 1 for sleeping after a[i] - 1 hours
            # Calculate the new time after sleeping
            nj = (j + a[i - 1] - k) % h
            if dp[i - 1][j] != -1:
                dp[i][nj] = max(dp[i][nj], dp[i - 1][j] + (l <= nj <= r))

# Find the maximum number of good sleeping times
print(max(dp[n]))
