import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[] a = new int[n];
    int[] b = new int[m];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    for (int i = 0; i < m; i++) {
      b[i] = sc.nextInt();
    }
    if (!check(a, b, n, m)) {
      System.out.println("NO");
      return;
    }
    System.out.println("YES");
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        System.out.print((a[i] ^ b[j]) + " ");
      }
      System.out.println();
    }
  }

  private static boolean check(int[] a, int[] b, int n, int m) {
    int[][] matrix = new int[n][m];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        matrix[i][j] = a[i] ^ b[j];
      }
    }
    for (int i = 0; i < n; i++) {
      int rowXor = 0;
      for (int j = 0; j < m; j++) {
        rowXor ^= matrix[i][j];
      }
      if (rowXor != a[i]) {
        return false;
      }
    }
    for (int j = 0; j < m; j++) {
      int colXor = 0;
      for (int i = 0; i < n; i++) {
        colXor ^= matrix[i][j];
      }
      if (colXor != b[j]) {
        return false;
      }
    }
    return true;
  }
}
