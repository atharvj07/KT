import sys

n = int(sys.stdin.readline())
board = [list(sys.stdin.readline().rstrip()) for _ in range(n)]

def valid(x, y):
    return 0 <= x < n and 0 <= y < n and board[x][y] == '#'

def dfs(x, y, visited):
    if not valid(x, y) or visited[x][y]:
        return 0
    visited[x][y] = True
    cnt = 1
    for dx, dy in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
        cnt += dfs(x + dx, y + dy, visited)
    return cnt

for i in range(n):
    for j in range(n):
        if board[i][j] == '#':
            visited = [[False] * n for _ in range(n)]
            if dfs(i, j, visited) != 5:
                print('NO')
                exit()
print('YES')
