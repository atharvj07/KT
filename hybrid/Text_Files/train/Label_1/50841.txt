import java.util.*;
import java.math.*;
import java.io.*;

class Main {
    public static int MOD = 998244353;

    public static long powMod(long n, long p) {
        if (p == 0)
            return 1;
        if (p > 0 && p % 2 == 0) {
            long halfPow = powMod(n, p / 2);
            return (halfPow * halfPow) % MOD;
        }
        return (n * powMod(n, p - 1)) % MOD;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        String[] aStr = br.readLine().split(" ");
        long sumA = 0;
        long[] a = new long[n + 1];
        for (int i = 0; i < n; i++) {
            a[i + 1] = Long.parseLong(aStr[i]);
            sumA += a[i + 1];
        }

        long[] pref = new long[n + 2];
        for (int i = 1; i <= n; i++) {
            pref[i] = pref[i - 1] + a[i];
        }

        long bigNum = sumA * (sumA + 1) / 2 % MOD;
        for (int i = 1; i <= n; i++) {
            bigNum = (bigNum - ((pref[i - 1] * (sumA - pref[i - 1])) % MOD) % MOD) % MOD;
        }
        System.out.println(bigNum);
    }
}
