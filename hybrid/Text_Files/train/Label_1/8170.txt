#include <bits/stdc++.h>

using namespace std;

int main() {
  long long n, m, w;
  cin >> n >> m >> w;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  long long ans = a[0];
  for (int i = 0; i < min(m, n); i++) {
    ans = max(a[i] + m - i, ans);
    i += w - 1;
  }
  cout << ans << endl;
  return 0;
}
