import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n + 1];
        int[] a = new int[n + 1];
        int[] d = new int[n + 1];
        int[] s = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            h[i] = sc.nextInt();
            a[i] = sc.nextInt();
            d[i] = sc.nextInt();
            s[i] = sc.nextInt();
        }
        sc.close();

        // Sort enemies by agility in descending order
        Integer[] enemyIndices = new Integer[n];
        for (int i = 0; i < n; i++) {
            enemyIndices[i] = i + 1;
        }
        Arrays.sort(enemyIndices, new Comparator<Integer>() {
            @Override
            public int compare(Integer o1, Integer o2) {
                return s[o2] - s[o1];
            }
        });

        // Initialize hero's total damage taken
        int totalDamage = 0;

        // Start the battle
        boolean heroIncapacitated = false;
        for (int i = 0; i < n; i++) {
            int enemyIndex = enemyIndices[i];

            // Enemy attacks hero
            int damage = Math.max(a[enemyIndex] - d[0], 0);
            h[0] -= damage;
            totalDamage += damage;

            // Check if hero is incapacitated
            if (h[0] <= 0) {
                heroIncapacitated = true;
                break;
            }

            // Hero attacks enemy
            int targetEnemyIndex = -1;
            for (int j = 0; j < n; j++) {
                if (enemyIndices[j] != enemyIndex && h[enemyIndices[j]] > 0) {
                    if (targetEnemyIndex == -1 || s[targetEnemyIndex] < s[enemyIndices[j]]) {
                        targetEnemyIndex = enemyIndices[j];
                    }
                }
            }

            if (targetEnemyIndex != -1) {
                damage = Math.max(a[0] - d[targetEnemyIndex], 0);
                h[targetEnemyIndex] -= damage;
                totalDamage += damage;
            }
        }

        // Output the result
        if (heroIncapacitated) {
            System.out.println(-1);
        } else {
            System.out.println(totalDamage);
        }
    }
}
