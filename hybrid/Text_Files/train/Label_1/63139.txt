from collections import deque
def check(dl, dr, N, x):
    for i in range(N):
        c = 0
        for j in range(N):
            if x[i][j] == 'x':
                c += 1
                if c > dr:
                    return False
            elif x[i][j] == 'o':
                c = 0
            if c > dl:
                return False
        c = 0
        for j in range(N):
            if x[j][i] == 'x':
                c += 1
                if c > dr:
                    return False
            elif x[j][i] == 'o':
                c = 0
            if c > dl:
                return False
    return True
def bfs(Q, dl, dr, N, x, ans, d):
    while Q:
        u, v = Q.popleft()
        for p, q in d:
            x1 = u + p
            y1 = v + q
            if 0 <= x1 < N and 0 <= y1 < N and x[x1][y1] == '.':
                if x1 == N // 2 and y1 == N // 2:
                    if check(dl, dr, N, x):
                        ans.append('YES')
                        ans.append(x)
                    return ans
                x[x1][y1] = 'x'
                Q.append((x1, y1))
    ans.append('NO')
    return ans
n = int(input())
x = []
for i in range(n):
    s = input()
    l = []
    for j in s:
        l.append(j)
    x.append(l)
dl, dr = -1, -1
for i in range(n):
    for j in range(n):
        if x[i][j] == 'o':
            if dl == -1:
                dl = 0
            dr = 0
        elif x[i][j] == 'x':
            dr += 1
Q = deque([(n // 2, n // 2)])
ans = []
d = [(1, 0), (0, 1), (-1, 0), (0, -1)]
ans = bfs(Q, dl, dr, n, x, ans, d)
for i in ans:
    print(i)
