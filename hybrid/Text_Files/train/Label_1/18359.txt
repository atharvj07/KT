#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    map<int, int> cnt;
    for (auto &x : a) cnt[x]++;
    int even = 0, odd = 0, ev = n;
    vector<pair<int, int>> ans;
    while (even < n / 2 || odd < n / 2) {
        int x = INT_MAX, y;
        for (auto &p : cnt) {
            if (p.second & 1 && p.first < x) {
                x = p.first;
                y = p.second;
            }
        }
        if (x == INT_MAX) {
            cout << -1 << endl;
            exit(0);
        }
        ev++;
        for (int i = 0; i < n; i++) {
            if (a[i] == x) {
                a[i] = ev;
                if (y == 1) cnt.erase(x);
                else
                    cnt[x]--, cnt[ev]++;
                ans.push_back({x, ev});
                break;
            }
        }
        if (ev % 2 == 0)
            even++;
        else
            odd++;
    }
    cout << ans.size() << endl;
    for (auto &x : a) cout << x << ' ';
    cout << endl;
}
