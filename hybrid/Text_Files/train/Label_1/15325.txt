#include <iostream>
#include <vector>
using namespace std;

int main() {
  vector<int> a;
  int n, x;
  cin >> n >> x;
  int c = 0;
  for (int i = 0; i < n; i++) {
    int y;
    cin >> y;
    if (y % x == 0) c++;
    a.push_back(y);
  }
  if (c > 0)
    cout << n - c << endl;
  else {
    int ans = -1;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        int y = a[i] & a[j];
        int d = __builtin_popcount(x ^ y);
        if (d == 1) ans = 1;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
