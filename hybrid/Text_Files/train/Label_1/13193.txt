#include<bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr ll inf = numeric_limits<ll>::max() / 2;
struct animal { ll x, y; };
ll dist(const animal &a, const animal &b) { return ll(a.x - b.x) * ll(a.x - b.x) + ll(a.y - b.y) * ll(a.y - b.y); }
ll minR(const ll xCnt, const ll yCnt, const animal &a, const animal &b, ll bound = inf) {
    if (xCnt * xCnt + yCnt * yCnt >= bound) { return 0; }
    if (xCnt == 0 || yCnt == 0) { return inf; }
    ll dx = b.x - a.x, dy = b.y - a.y, md = dx * dx + dy * dy;
    return min(minR(abs(yCnt), abs(xCnt), a, b, md), minR(abs(xCnt), abs(yCnt), a, b, md) + 1);
}
signed main() {
    ll n; cin >> n;
    vector<animal> v(n);
    for (ll i = 0; i < n; i++) { cin >> v[i].x >> v[i].y; }
    sort(v.begin(), v.end(), [](const animal &a, const animal &b) { return a.x * a.x + a.y * a.y < b.x * b.x + b.y * b.y; });
    ll res = minR(v[0].x, v[0].y, v[0], v[0]);
    if (res == 0) { cout << -1 << endl; }
    else { cout << fixed << setprecision(9) << sqrt(res) << endl; }
}
