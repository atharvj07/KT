import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] strengths = new int[n];
        for (int i = 0; i < n; i++) {
            strengths[i] = scanner.nextInt();
        }
        int m = scanner.nextInt();
        int[] picks = new int[m];
        int[] bans = new int[m];
        for (int i = 0; i < m; i++) {
            String action = scanner.next();
            int team = scanner.nextInt();
            if (action.equals("p")) {
                picks[i] = team;
            } else {
                bans[i] = team;
            }
        }
        Arrays.sort(strengths);
        int firstTeamStrength = 0;
        int secondTeamStrength = 0;
        boolean[] picked = new boolean[n];
        for (int i = 0; i < m; i++) {
            int pick = picks[i];
            int ban = bans[i];
            if (pick == 1) {
                for (int j = n - 1; j >= 0; j--) {
                    if (!picked[j] && j != ban - 1) {
                        firstTeamStrength += strengths[j];
                        picked[j] = true;
                        break;
                    }
                }
            } else {
                for (int j = n - 1; j >= 0; j--) {
                    if (!picked[j] && j != pick - 1) {
                        secondTeamStrength += strengths[j];
                        picked[j] = true;
                        break;
                    }
                }
            }
        }
        System.out.println(firstTeamStrength - secondTeamStrength);
    }
}
