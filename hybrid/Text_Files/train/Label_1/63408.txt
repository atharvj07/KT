import sys
input = sys.stdin.readline

n = int(input())
x = [tuple(map(int, input().split())) for _ in range(n)]
adj = [[] for _ in range(n)]
deg = [0] * n
for u, xor in x:
    deg[u] += u
    for i in range(u):
        if xor >> i & 1:
            adj[u].append(i)
            adj[i].append(u)
            deg[u] -= 1
            deg[i] -= 1
print(sum(deg) // 2)
for u in range(n):
    for v in adj[u]:
        if u < v: print(u, v)
