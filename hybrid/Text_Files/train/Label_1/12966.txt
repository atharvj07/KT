#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    vector<pair<int, int>> ans;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (a[i] > a[j]) {
                ans.push_back({i + 1, j + 1});
                swap(a[i], a[j]);
            }
        }
    }
    if (is_sorted(a.begin(), a.end())) {
        cout << ans.size() << endl;
        for (auto &x : ans) cout << x.first << " " << x.second << endl;
    } else {
        cout << "-1" << endl;
    }
}
