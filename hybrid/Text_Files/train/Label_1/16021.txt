#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  vector<int> b(m);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < m; i++) {
    cin >> b[i];
  }
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  int ans = 1;
  while (true) {
    bool found_a = false;
    bool found_b = false;
    int tmp = ans;
    while (tmp > 0) {
      int digit = tmp % 10;
      if (binary_search(a.begin(), a.end(), digit)) {
        found_a = true;
      }
      if (binary_search(b.begin(), b.end(), digit)) {
        found_b = true;
      }
      tmp /= 10;
    }
    if (found_a && found_b) {
      cout << ans << endl;
      return 0;
    }
    ans++;
  }
  return 0;
}
