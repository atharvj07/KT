#include <bits/stdc++.h>

using namespace std;

#define ll long long

const ll mod = 1e9 + 7;

int main() {
    ll n, m;
    cin >> n >> m;

    vector<ll> a(n);
    for (ll i = 0; i < n; i++) { cin >> a[i]; }

    sort(a.rbegin(), a.rend());

    ll ans = 0;
    ll sum = 0;

    for (ll i = 0; i < n; i++) {
        if (sum + a[i] - i - 1 >= m) {
            ans = i + 1;
            break;
        } else {
            sum += a[i] - i - 1;
        }
    }

    if (sum < m) {
        ans = -1;
    }

    cout << ans << "\n";

    return 0;
}
