#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    if (k <= 13 * n) {
        for (int i = 0; i < n; i++) {
            cout << char((s[i] - 'a' + k) % 26 + 'a');
        }
        cout << endl;
    } else
        cout << "-1" << endl;
    return 0;
}
