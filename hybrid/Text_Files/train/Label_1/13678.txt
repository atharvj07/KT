#include<bits/stdc++.h>
using namespace std;

int n, v[100001], p[100001];
vector<int> g[100001];

int main() {
    int x, y, sum = 0;
    cin >> n;
    for (int i = 1; i < n; i++) {
        cin >> x >> y;
        g[x].push_back(y);
        g[y].push_back(x);
    }

    for (int i = 1; i <= n; i++) {
        cin >> v[i];
        sum += abs(v[i]);
    }

    if (!sum) {
        cout << 0 << endl;
        return 0;
    }

    cout << (sum + 1) / 2 << endl;
    return 0;
}
