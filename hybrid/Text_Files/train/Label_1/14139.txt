#include <iostream>
#include <string>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int l = 0, r = 0;
  int ans = 1;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'L')
      l++;
    else
      r++;
  }
  ans = 1 + min(l, r) + min(l, r) * (min(l, r) + 1) / 2;
  cout << ans << endl;
  return 0;
}
