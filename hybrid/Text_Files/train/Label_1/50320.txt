import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];

            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }

            boolean noVacancy = true;

            for (int j = 0; j < n; j++) {
                int newRoom = (j + a[j % n] + n) % n;

                if (newRoom != j) {
                    noVacancy = false;
                    break;
                }
            }

            System.out.println(noVacancy ? "YES" : "NO");
        }
    }
}
