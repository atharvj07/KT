import java.util.Scanner;

public class Main {
  public static void main(String[] args) throws Exception {
    Scanner sc = new Scanner(System.in);

    int n = sc.nextInt();
    int h = sc.nextInt();
    int w = sc.nextInt();

    int[] x = new int[n];
    for (int i = 0; i < n; i++) {
      x[i] = sc.nextInt();
    }

    int ans = 0;
    for (int i = 1; i < n; i += 2) {
      ans += h * Math.max(w - x[i] - x[i - 1], 0);
    }

    System.out.println(ans);
  }
}
