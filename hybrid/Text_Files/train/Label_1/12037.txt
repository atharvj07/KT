#include <bits/stdc++.h>
using namespace std;
const int N = 100000;
int a[N];
long long hsh[N << 1];
int main() {
  int h;
  cin >> h;
  long long c = 0;
  for (int i = 0; i < h + 1; i++) {
    cin >> a[i];
    c += a[i];
    hsh[i] = i >= 1 ? hsh[i - 1] : 0;
  }
  for (int i = 0; i <= h; i++) hsh[i] = (hsh[i] * 31 + a[i]) % c;
  for (int i = 0; i < h + 1; i++)
    if (i != h && hsh[i] == hsh[h] && a[i] == a[h]) goto bad;
  cout << "perfect" << endl;
  return 0;
bad:
  cout << "ambiguous" << endl;
  a[h] ^= 1;
  for (int i = 0; i <= h; i++) { hsh[i] = i >= 1 ? hsh[i - 1] : 0; }
  for (int i = 0; i <= h; i++) hsh[i] = (hsh[i] * 31 + a[i]) % c;
  for (int i = h - 1; i >= 0; i--)
    if (hsh[i] == hsh[h]) goto done;
done:
  for (int i = 0; i <= h; i++) { cout << a[i] << ' '; }
  cout << endl;
  for (int i = 0; i <= h; i++) { cout << (i ? (h - i) : 0) << ' '; }
  cout << endl;
}
