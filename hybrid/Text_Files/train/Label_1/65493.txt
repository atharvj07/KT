import sys

n, k = [int(x) for x in sys.stdin.readline().split()]
a = [0] * (n+1)
e = [0] * (n+1)

for i in range(n):
    p, ef = [int(x) for x in sys.stdin.readline().split()]
    a[i+1] = p
    e[i+1] = ef

a.sort(reverse=True)
e.sort()

ans = 0
for i in range(1, n+1):
    if a[i] < k:
        if ans + e[i] >= a[k-1]:
            ans += e[i] - (a[k-1] - a[i])
            break
        else:
            ans += e[i]
    else:
        break

if ans + e[i] < a[k-1]:
    print(-1)
else:
    print(ans)
