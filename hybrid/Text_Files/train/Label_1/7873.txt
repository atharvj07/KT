#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  long long a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  long long diff = 0;
  int p = 0, q = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] < a[p]) p = i;
    if (a[i] > a[q]) q = i;
    diff += a[q] - a[p];
  }
  cout << ++diff << ' ' << a[q] - a[p] << '\n';
  return 0;
}
