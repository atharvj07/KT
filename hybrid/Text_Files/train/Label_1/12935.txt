#include <bits/stdc++.h>
using namespace std;
vector<int> a;
long long b[100005];
int main()
{
	long long i,j,k,n,m,x,o,l,r,u;
	cin>>n>>k>>m;
	a.resize(n);
	for (i=0;i<n;i++)
	{	cin>>x;
		a[x%m]++;
	}
	if (a[0]==0 and k!=0)
	{	cout<<"No\n";
		return 0;
	}
	cout<<"Yes\n";
	for (i=0;i<m;i++)
	{	if (a[i]%2==0 and a[i]>=k)
		{	for (j=0;j<k;j++)
				b[j]=i;
			break;
		}
		else
		{	if ((a[i]%2==1 and k-1<=a[i]) or a[i]==0)
			{	x=0;
				b[x]=i;
				x++;
				k--;
				if (a[i]>0)
					a[i]--;
				else
					a[(i+1)%m]--;
			}
			else
			{	cout<<"No\n";
				return 0;
			}
		}
	}
	l=m;
	r=2*m-1;
	for (;k>0;k--)
	{	if (l<=r)
		{	o=0;
			for (i=l;i<=r;i++)
			{	if (a[i]>0)
				{	b[x]=i;
					o++;
					x++;
					k--;
					a[i]--;
					if (o==2)
						break;
				}
			}
			l=i+1;
		}
		else
		{	cout<<"No\n";
			return 0;
		}
	}
	for (i=0;i<k;i++)
		cout<<b[i]<<" ";
	return 0;
}
