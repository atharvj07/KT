import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int max = Integer.MIN_VALUE;
        for (int i = 0; i < n; i++) {
            int a = sc.nextInt();
            int low = 0, high = 46340;
            while (low <= high) {
                int mid = (low + high) / 2;
                if (mid * mid == a) {
                    low = mid + 1;
                } else {
                    high = mid - 1;
                }
            }
            if (low * low != a) {
                max = Math.max(max, a);
            }
        }
        System.out.println(max);
    }
}
