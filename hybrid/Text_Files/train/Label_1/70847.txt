import sys

m, k = map(int, sys.stdin.readline().split())
friends = {}
for _ in range(m):
    a, b = map(int, sys.stdin.readline().split())
    friends.setdefault(a, set()).add(b)
    friends.setdefault(b, set()).add(a)

for id in sorted(friends):
    common = set()
    for friend in friends[id]:
        common |= friends[friend]
    friends[id] -= {id}
    common -= {id}
    if len(friends[id]) * k // 100 > len(common):
        print(f"{id}: {len(common)} {' '.join(map(str, sorted(common)))}")
