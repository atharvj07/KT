#include <bits/stdc++.h>
using namespace std;
#define all(v) (v).begin(), (v).end()
typedef long long ll;

const ll mod = 1e9 + 7;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<ll>> a(n, vector<ll>(m));
    for (auto& vi : a) {
      for (auto& v : vi) cin >> v;
      sort(all(vi));
    }
    for (int j = 0; j < m; ++j) {
      int idx = 0;
      for (int i = 1; i < n; ++i) {
        if (a[idx][j] < a[i][j]) idx = i;
      }
      for (int i = 0; i < idx; ++i) swap(a[i][j], a[i + 1][j]);
    }
    ll ans = 0;
    for (int i = 0; i < n; ++i) ans += a[i][m - 1];
    cout << ans << '\n';
  }
  return 0;
}
