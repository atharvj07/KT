import java.util.*;

public class AntonAndArrayStudying {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] b = new int[n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        int[] a = new int[n];
        boolean possible = true;
        for (int i = 0; i < n; i++) {
            int maxBitwiseAnd = 0;
            for (int j = i; j < n; j++) {
                maxBitwiseAnd = Math.max(maxBitwiseAnd, b[j] & c[j]);
            }
            a[i] = maxBitwiseAnd;
            if (a[i] != (b[i] | c[i])) {
                possible = false;
                break;
            }
        }

        if (!possible) {
            System.out.println(-1);
        } else {
            for (int i = 0; i < n; i++) {
                System.out.print(a[i] + " ");
            }
        }
    }
}
