import java.util.Scanner;

public class MagnetGroups {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();

        int groups = 1;
        char prevPolarity = scanner.nextLine().charAt(1);

        for (int i = 1; i < n; i++) {
            String polarity = scanner.nextLine();
            if (polarity.charAt(0) != prevPolarity) {
                groups++;
            }
            prevPolarity = polarity.charAt(1);
        }

        System.out.println(groups);
    }
}
