#include<bits/stdc++.h>
using namespace std;
const int maxn = 55, maxm = 123456789;
int n, k, a[maxn], b[maxn], used[maxm];

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        b[i] = a[i];
    }
    sort(a + 1, a + 1 + n);
    for (int i = 1; i <= k; i++) {
        cout << 1 << ' ' << a[i] << '\n';
        used[a[i]] = 1;
    }
    for (int i = 1; i <= n; i++) {
        if (used[b[i]]) { continue; }
        int cnt = 0;
        while (used[a[++cnt]]) { }
        cout << 1 << ' ' << a[cnt] << '\n';
        used[a[cnt]] = 1;
    }
    for (int i = 1; i <= n; i++) {
        if (used[b[i]]) { continue; }
        int cnt = 0;
        while (used[a[++cnt]]) { }
        cout << 2 << ' ' << a[cnt] << ' ' << b[i] << '\n';
        used[a[cnt]] = 1;
        used[b[i]] = 1;
    }
    return 0;
}
