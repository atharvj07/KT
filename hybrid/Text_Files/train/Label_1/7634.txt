#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  long long cnt = 0;
  for (long long i = 1; i <= n; i++) {
    bool ok = true;
    for (long long j = 2; j <= 10 && ok; j++) {
      if (i % j == 0) {
        ok = false;
      }
    }
    if (ok) cnt++;
  }
  cout << cnt << endl;
  return 0;
}
