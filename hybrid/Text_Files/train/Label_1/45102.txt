import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    int h1 = scan.nextInt(), a1 = scan.nextInt(), c1 = scan.nextInt();
    int h2 = scan.nextInt(), a2 = scan.nextInt();
    int n = (h2 - 1 + a1 - 1) / a1;
    System.out.println(n);
    for (int i = 0; i < n; i++) {
      if (h1 > a2) System.out.println("STRIKE");
      else System.out.println("HEAL");
      h1 = Math.min(h1 + c1 - a2, h1);
    }
  }
}
