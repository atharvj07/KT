import java.util.*;
import java.lang.*;
import java.io.*;
import java.math.*;

class Main {

    public static void main(String[] args) {
        try {
            BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
            PrintWriter out = new PrintWriter(System.out);
            final int MAXN = 100000 + 5;
            int[] dp = new int[MAXN];
            int[] pre = new int[MAXN];
            int n = Integer.parseInt(br.readLine());
            String[] str = br.readLine().split(" ");
            int[] arr = new int[n];
            int[] brr = new int[n];
            int[] ord = new int[n];
            for (int i = 0; i < n; i++) {
                arr[i] = Integer.parseInt(str[i]);
            }

            for (int j = 0; j < n; j++) {
                int val = arr[j];
                int p = j;
                while (p > 0 && val < arr[p - 1]) {
                    arr[p] = arr[p - 1];
                    p--;
                }
                arr[p] = val;
                ord[j] = p;
            }

            for (int i = 0; i < n; i++) {
                int val = arr[i];
                int p = i;
                while (p > 0 && val > arr[p - 1]) {
                    arr[p] = arr[p - 1];
                    p--;
                }
                arr[p] = val;
                pre[i] = p;
            }

            for (int i = 0; i < n; i++) {
                if (pre[i] != ord[i]) {
                    out.print("No");
                    out.flush();
                    return;
                }
                brr[i] = arr[i] ^ (i > 0 ? brr[i - 1] : 0);
            }

            for (int i = 0; i < n; i++) {
                arr[i] = brr[ord[i]];
            }

            out.print("Yes");
            out.println();
            for (int x : arr) {
                out.print(x + " ");
            }
            out.flush();
            out.close();
            br.close();
        } catch (Exception ignored) {
        }
    }
}
