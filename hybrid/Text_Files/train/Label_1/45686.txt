import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
    }
    Arrays.sort(a);
    int k = 0;
    int sum = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (sum <= n * a[i]) {
        k++;
        sum += n - a[i];
      }
    }
    System.out.println(k);
  }
}
