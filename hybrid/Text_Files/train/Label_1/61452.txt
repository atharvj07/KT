import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    b = []
    cur = 0
    for i in range(n):
        if not b or b[-1] != a[i]:
            b.append(a[i])
            cur += a[i]
    if cur:
        print('YES')
        print(*b)
    else:
        print('NO')
