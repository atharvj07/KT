import java.util.Scanner;

public class Solution {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int str = scanner.nextInt();
            int intl = scanner.nextInt();
            int exp = scanner.nextInt();

            // Calculate the maximum possible strength and intelligence.
            int maxStr = str + exp;
            int maxIntl = intl + exp;

            // Count the number of possible builds.
            int count = 0;

            // Iterate over all possible strengths.
            for (int s = str; s <= maxStr; s++) {
                // Check if the resulting intelligence is less than the strength.
                if (s > maxIntl - (maxStr - s)) {
                    count++;
                }
            }

            // Print the result.
            System.out.println(count);
        }
    }
}
