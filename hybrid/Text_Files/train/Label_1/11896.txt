#include <bits/stdc++.h>
using namespace std;
const int inf=2e9;
int n,m,c[110][110],ans=inf;
int main() {
    cin>>n>>m;
    for(int i=1; i<=n; ++i)
        for(int j=1; j<=m; ++j) cin>>c[i][j];
    for(int i=1; i<=n; ++i) {
        int mx=-inf;
        for(int j=1; j<=m; ++j) mx=max(mx,c[i][j]);
        ans=min(ans,mx);
    }
    cout<<ans<<endl;
}
