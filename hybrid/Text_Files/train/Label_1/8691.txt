#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt0 = 0;
  int cnt1 = 0;
  for (char ch : s) {
    if (ch == '0')
      ++cnt0;
    else if (ch == '1')
      ++cnt1;
  }
  int extra0 = max(0, cnt0 - (n / 3));
  int extra1 = max(0, cnt1 - (n / 3));
  for (char &ch : s) {
    if (extra0 > 0) {
      if (ch == '1') {
        ch = '0';
        --extra0;
      } else if (ch == '2') {
        ch = '0';
        --extra0;
        --extra1;
      }
    } else if (extra1 > 0) {
      if (ch == '2') {
        ch = '1';
        --extra1;
      } else if (ch == '0') {
        ch = '1';
        --extra1;
        --extra0;
      }
    }
  }
  cout << s << '\n';
  return 0;
}
