t = int(input().strip())
for _ in range(t):
    n, s = map(int, input().split())
    transactions = list(map(int, input().split()))

    max_len = 0
    solution = (-1, -1)
    for start in range(n):
        balance = s
        for end in range(start, n):
            balance += transactions[end]
            if balance < 0:
                break
            if end - start + 1 > max_len:
                max_len = end - start + 1
                solution = (start + 1, end + 1)
                
    if solution == (-1, -1):
        print(-1)
    else:
        print(solution[0], solution[1])
