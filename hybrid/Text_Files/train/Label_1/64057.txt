from collections import defaultdict
import sys
readline = sys.stdin.readline
MOD = 1000000007

def fast_pow(base, exp):
    if exp == 0: return 1
    if exp == 1: return base
    val = fast_pow(base, exp//2)
    val = val * val
    val %= MOD
    if exp % 2 == 1:
        val = val * base
        val %= MOD
    return val

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve():
    n = int(readline())
    arr = list(map(int, readline().split()))

    max_val = max(arr)
    freq = defaultdict(int)
    for el in arr:
        freq[el] += 1

    ans = 0
    for i in range(1, max_val + 1):
        cur = 0
        for j in range(i, max_val + 1, i):
            cur += freq[j]
        if cur > 1:
            ans += fast_pow(cur, 2) * i
            ans %= MOD
    return ans

print(solve())
