def solve():
    t = int(input())
    while t != 0:
        t -= 1
        n = int(input())
        matrix = [[0] * n for _ in range(n)]
        if n % 2 == 1:
            k = 1
            for i in range(0, n):
                for j in range(0, n):
                    matrix[i][j] = k
                    k += 2
        else:
            if n == 2:
                print(-1)
                continue
            k = 1
            for i in range(0, n):
                for j in range(0, n):
                    if j % 2 == 0:
                        matrix[i][j] = k
                        k += 2
                    else:
                        matrix[i][j] = k + 1
                        k += 2
        for row in matrix:
            print(*row)

solve()
