import sys

input = sys.stdin.readline

MOD = 998244353

n, k = map(int, input().split())
lamps = []
for i in range(n):
    l, r = map(int, input().split())
    lamps.append((l, r))

lamps.sort(key=lambda x: x[0])

dp = [[0] * (k + 1) for _ in range(n + 1)]
dp[0][0] = 1
for i in range(1, n + 1):
    l, r = lamps[i - 1]
    for j in range(k + 1):
        dp[i][j] = dp[i - 1][j]
        if j > 0 and l <= dp[i - 1][j - 1] <= r:
            dp[i][j] = (dp[i][j] + dp[i - 1][j - 1]) % MOD

print(dp[n][k])
