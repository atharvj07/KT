import java.util.*;
class Main {
    static int n;
    static int[] a;
    static int[][] b;
    static int max;
    static int[] ans;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        a = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            a[i] = sc.nextInt();
        }
        b = new int[2 * n][2 * n];
        for (int i = 0; i < 2 * n; i++) {
            for (int j = 0; j < 2 * n; j++) {
                if (i == j) {
                    b[i][j] = 0;
                } else {
                    int aa = a[i] / 10;
                    int bb = a[i] % 10;
                    int cc = a[j] / 10;
                    int dd = a[j] % 10;
                    aa *= 10;
                    cc *= 10;
                    int res = aa + bb + cc + dd;
                    int cnt = 0;
                    char[] s = String.valueOf(res).toCharArray();
                    Arrays.sort(s);
                    for (int k = 0; k < s.length - 1; k++) {
                        if (s[k] == s[k + 1]) {
                            cnt++;
                        }
                    }
                    b[i][j] = cnt;
                }
            }
        }
        max = 0;
        ans = new int[2 * n];
        dfs(0, new boolean[2 * n], 0, 0);
        System.out.println(max);
        for (int i = 0; i < 2 * n; i++) {
            System.out.print(ans[i] + " ");
        }
    }
    static void dfs(int idx, boolean[] vis, int cnt, int sum) {
        if (idx == 2 * n) {
            if (cnt > max) {
                max = cnt;
                for (int i = 0; i < 2 * n; i++) {
                    ans[i] = vis[i] ? 1 : 2;
                }
            }
            return;
        }
        vis[idx] = true;
        for (int i = idx + 1; i < 2 * n; i++) {
            if (!vis[i]) {
                if (sum + b[idx][i] < max) {
                    vis[idx] = false;
                    continue;
                }
                dfs(idx + 1, vis, cnt + 1, sum + b[idx][i]);
            }
        }
        vis[idx] = false;
        dfs(idx + 1, vis, cnt, sum);
    }
}
