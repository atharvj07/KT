#include<bits/stdc++.h>
using namespace std;

int main() {
  int n, ans1 = 0, ans2 = 0, cur = 0;
  cin >> n;
  string s;
  cin >> s;

  for (int i = 0; i < n; i++) {
    if (s[i] == '(') {
      ans1 = max(ans1, cur);
      cur = 0;
    } else if (s[i] == ')') {
      ans2++;
      continue;
    } else {
      cur++;
    }
  }
  ans1 = max(ans1, cur);
  cout << ans1 << ' ' << ans2 << '\n';
}
