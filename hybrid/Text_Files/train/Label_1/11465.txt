#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n), dp(n, 1), lis(n, 1);
  for (int i = 0; i < n; ++i) cin >> a[i];
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < i; ++j)
      if (a[i] > a[j]) {
        if (dp[i] < dp[j] + 1) {
          dp[i] = dp[j] + 1;
          lis[i] = j;
        }
      }
  vector<int> ans(n, 3);
  for (int i = n - 1, cnt = 0; i >= 0; --i)
    if (cnt++ == dp[i] - 1) {
      ans[i] = 1;
      i = lis[i];
    }
  for (int i = 0; i < n; ++i) cout << ans[i];
  cout << '\n';
}
