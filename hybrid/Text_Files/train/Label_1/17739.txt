#include <bits/stdc++.h>
using namespace std;

const int MAXN = 15, MAXM = 100;
int dist[MAXN][MAXM+2], d[MAXN][MAXM+2];
bool vis[MAXN][MAXM+2];
int n, m;
string s[MAXN];
queue<pair<int, int>> q;

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; ++i) cin >> s[i];
    
    for (int i = 0; i < MAXN; ++i) {
        for (int j = 0; j <= MAXM+1; ++j) {
            dist[i][j] = INT_MAX;
        }
    }
    
    dist[n][0] = 0;
    q.push({n, 0});
    
    while (!q.empty()) {
        pair<int, int> u = q.front();
        q.pop();
        if (vis[u.first][u.second]) continue;
        vis[u.first][u.second] = true;
        
        // turn off lights
        if (u.second > 0 && u.second < m+1 && s[u.first][u.second] == '1') {
            d[u.first][u.second] = dist[u.first][u.second];
        }
        
        // go up and down
        if (u.first > 0) {
            dist[u.first-1][u.second] = min(dist[u.first-1][u.second], dist[u.first][u.second] + 1);
            q.push({u.first-1, u.second});
        }
        if (u.first < n-1) {
            dist[u.first+1][u.second] = min(dist[u.first+1][u.second], dist[u.first][u.second] + 1);
            q.push({u.first+1, u.second});
        }
        
        // go left and right
        if (u.second > 0) {
            dist[u.first][u.second-1] = min(dist[u.first][u.second-1], dist[u.first][u.second] + 1);
            q.push({u.first, u.second-1});
        }
        if (u.second < m+1) {
            dist[u.first][u.second+1] = min(dist[u.first][u.second+1], dist[u.first][u.second] + 1);
            q.push({u.first, u.second+1});
        }
    }
    
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j <= m+1; ++j) {
            if (d[i][j] != 0) ans = max(ans, d[i][j]);
        }
    }
    
    cout << ans << '\n';
    return 0;
}
