import sys
input = sys.stdin.readline
print = sys.stdout.write

n, m = map(int, input().split())
vs = [int(input()) - 1 for _ in range(n)]
vs = sorted(list(set(vs)))
hs = []
for _ in range(m):
    a, b, c = map(int, input().split())
    a -= 1
    b -= 1
    c -= 1
    hs.append((a, b, c))
hs.sort()
v = [1] * len(vs)
h = [1] * len(hs)
for i in range(1, len(v)):
    if vs[i] == vs[i-1]: v[i] = v[i-1]
for i in range(1, len(h)):
    if hs[i] == hs[i-1]: h[i] = h[i-1]
ans = 0
i = 0
j = 0
while i < len(v) and j < len(h):
    if v[i] == h[j]:
        ans += 1
        i += 1
        j += 1
    elif v[i] < h[j]:
        i += 1
    else:
        j += 1
print(str(ans))
