import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        scanner.close();
        if (isGeometricProgression(a)) {
            System.out.println(0);
        } else {
            int m = -1;
            for (int i = 0; i < n; i++) {
                int[] b = new int[n - 1];
                for (int j = 0; j < i; j++) {
                    b[j] = a[j];
                }
                for (int j = i + 1; j < n; j++) {
                    b[j - 1] = a[j];
                }
                if (isGeometricProgression(b)) {
                    m = i;
                    break;
                }
            }
            if (m == -1) {
                System.out.println(2);
            } else {
                System.out.println(1);
            }
        }
    }

    public static boolean isGeometricProgression(int[] a) {
        if (a.length < 2) {
            return true;
        }
        double r = (double) a[1] / a[0];
        for (int i = 1; i < a.length - 1; i++) {
            if ((double) a[i + 1] / a[i] != r) {
                return false;
            }
        }
        return true;
    }
}
