t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    if a[0] == 1:
        print(n+1, end=' ')
        for i in range(1,n+1):
            print(i, end=' ')
        print()
    elif a[-1] == 0:
        for i in range(1,n+2):
            print(i, end=' ')
        print()
    else:
        for i in range(1, n+2):
            if i == n+1:
                print(n+1, end=' ')
            elif a[i] == 1:
                print(n+1, end=' ')
                for j in range(i+1, n+1):
                    print(j, end=' ')
                print(i, end=' ')
                for k in range(1, i):
                    print(k, end=' ')
                break
            else:
                print(i, end=' ')
        print()
