from collections import defaultdict
from heapq import heappop, heappush

m = int(input())
essay = input().replace('R', 'r').split()
n = int(input())
syn = defaultdict(list)
for _ in range(n):
    a, b = input().split()
    syn[a.lower()].append(b.lower())
    syn[b.lower()].append(a.lower())

mp = {}
for w in essay:
    w = w.lower()
    if w not in mp:
        q = [(len(w), w)]
        while q:
            l, cw = heappop(q)
            if cw not in mp:
                mp[cw] = l
                for nw in syn[cw]:
                    heappush(q, (len(nw), nw))

ans = []
for w in essay:
    w = w.lower()
    ans.append(mp[w])

print(ans.count(min(ans)), sum(ans))
