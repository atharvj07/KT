#include<bits/stdc++.h>
using namespace std;
#define pb push_back
int vis[101],a[101];
vector<int>adj[101];
int main() {
    int n, m, l;
    cin >> n >> m;
    while(m--) {
        cin >> l;
        vis[l] = true;
        if(a[l] == 0)
            a[l] = 1;
    }
    int cnt = 1;
    for(int i = 1; i <= n; i++) {
        if(!vis[i]) {
            int l = 1;
            while(a[i] <= n) {
                if(!vis[i]) {
                   a[i] = cnt++;
                   vis[i] = true;
                }
                i = (i + l - 1) % n + 1;
                l++;
            }
        }
    }
    bool all = true;
    for(int i = 1; i <= n; i++)
        all &= vis[i];
    if(!all)
        cout << -1 << endl;
    else {
        for(int i = 1; i <= n; i++)
            cout << a[i] << ' ';
        cout << endl;
    }
    return 0;
}
