#include <iostream>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[5005], q = 0, l = 1, r = 1, ans = 1;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) {
    q += a[i];
    while (q > 100 * l) { q -= a[l++]; }
    if (r - l + 1 > ans) { ans = r - l + 1; }
    r++;
  }
  cout << ans << endl;
}
