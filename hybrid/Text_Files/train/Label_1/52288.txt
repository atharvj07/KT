import java.util.Scanner;
import java.util.HashSet;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int p = sc.nextInt();
        HashSet<Integer> x = new HashSet<>();
        while (p-- > 0) x.add(sc.nextInt());
        int q = sc.nextInt();
        while (q-- > 0) x.add(sc.nextInt());
        System.out.println(x.size() == n ? "I become the guy." : "Oh, my keyboard!");
    }
}
