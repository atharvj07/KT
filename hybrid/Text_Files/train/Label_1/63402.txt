def possible(c, nodes):
  if nodes == 1:
    return c == 1
  if c % 2 == 0:
    return True
  if c < 2 * nodes - 1:
    return False
  return possible(c - nodes + 1, nodes - 1)

n = int(stdin.readline())
c = list(map(int, stdin.readline().split()))
ans = "YES" if all(possible(cc, 1) for cc in c) else "NO"
