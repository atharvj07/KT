import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.Map;

public class Main {
    private static class City implements Comparable<City> {
        int x;
        char type;

        public City(int x, char type) {
            this.x = x;
            this.type = type;
        }

        @Override
        public int compareTo(City other) {
            return Integer.compare(this.x, other.x);
        }
    }

    private static class UnionFind {
        private int[] parent;
        private int[] rank;

        public UnionFind(int n) {
            parent = new int[n];
            rank = new int[n];
            for (int i = 0; i < n; i++) {
                parent[i] = i;
                rank[i] = 0;
            }
        }

        public int find(int x) {
            if (parent[x] == x) {
                return x;
            }
            return parent[x] = find(parent[x]);
        }

        public void union(int x, int y) {
            int xRoot = find(x);
            int yRoot = find(y);
            if (xRoot == yRoot) {
                return;
            }
            if (rank[xRoot] < rank[yRoot]) {
                parent[xRoot] = yRoot;
            } else if (rank[xRoot] > rank[yRoot]) {
                parent[yRoot] = xRoot;
            } else {
                parent[xRoot] = yRoot;
                rank[yRoot]++;
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        City[] cities = new City[n];
        Map<Character, LinkedList<Integer>> cityMap = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String[] input = reader.readLine().split(" ");
            int x = Integer.parseInt(input[0]);
            char type = input[1].charAt(0);
            cities[i] = new City(x, type);
            if (!cityMap.containsKey(type)) {
                cityMap.put(type, new LinkedList<>());
            }
            cityMap.get(type).addLast(i);
        }
        Arrays.sort(cities);
        UnionFind byteland = new UnionFind(n);
        UnionFind berland = new UnionFind(n);
        long answer = 0;
        char[] types = {'B', 'R', 'P'};
        for (char type : types) {
            LinkedList<Integer> cityList = cityMap.getOrDefault(type, null);
            if (cityList == null) {
                continue;
            }
            int previous = cityList.removeFirst();
            for (int current : cityList) {
                if (type == 'P') {
                    byteland.union(previous, current);
                    berland.union(previous, current);
                } else if (type == 'B') {
                    byteland.union(previous, current);
                } else {
                    berland.union(previous, current);
                }
                answer += cities[current].x - cities[previous].x;
                previous = current;
            }
        }
        System.out.println(answer);
    }
}
