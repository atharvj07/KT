import java.util.*;

public class Main {
    static final int ALPHABET_SIZE = 26;

    static char lucky(int idx) {
        return (char)('a' + ((idx - 1) / 3) % ALPHABET_SIZE);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        char[] ans = new char[n];
        int idx = 0;
        while (idx < n) {
            for (int i = idx + 1; i < n; i++) {
                if (ans[i] == 0 || (i - idx) % 3 != 0) continue;
                if (Math.abs(ans[i] - lucky(i)) != 4 && Math.abs(ans[i] - lucky(i)) != 7) break;
            }
            ans[idx] = lucky(idx);
            idx++;
        }
        System.out.println(ans);
    }
}
