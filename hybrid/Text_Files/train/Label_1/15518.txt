#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const ll MAXN = 1e5 + 5;
ll a[MAXN][3];
vector<ll> adj[MAXN];

int main() {
  ll n;
  cin >> n;
  for (ll i = 1; i <= n; i++) {
    cin >> a[i][0] >> a[i][1] >> a[i][2];
    sort(a[i], a[i] + 3);
  }
  for (ll i = 1; i <= n; i++) {
    for (ll j = i + 1; j <= n; j++) {
      bool ok = true;
      for (ll k = 0; k < 3; k++) {
        if (a[i][k] != a[j][k]) ok = false;
      }
      if (ok) {
        adj[i].push_back(j);
        adj[j].push_back(i);
      }
    }
  }
  vector<ll> vis(n + 1, 0);
  vector<ll> ans;
  function<void(ll)> dfs = [&](ll u) {
    vis[u] = 1;
    ans.push_back(u);
    for (ll v : adj[u]) {
      if (!vis[v]) dfs(v);
    }
  };
  for (ll i = 1; i <= n; i++) {
    if (!vis[i]) {
      dfs(i);
      if (ans.size() <= 2) break;
      ans.pop_back();
    }
  }
  cout << ans.size() << '\n';
  for (ll u : ans) cout << u << ' ';
  cout << '\n';
  return 0;
}
