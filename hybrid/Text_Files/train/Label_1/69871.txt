from collections import Counter

def solve_array(arr):
    count = Counter(arr)
    while any(v > 1 for v in count.values()):
        x = min(k for k, v in count.items() if v > 1)
        count[x] -= 2
        if count[x] <= 0:
            del count[x]
        count[2*x] = count.get(2*x, 0) + 1
    result = []
    for k, v in sorted(count.items()):
        result.extend([k]*v)
    return result

n = int(input().strip())
arr = list(map(int, input().split()))
result = solve_array(arr)
print(len(result))
print(' '.join(map(str, result)))
