n, A, B, C, T = map(int, input().split())
t = list(map(int, input().split()))
money = [A] * n
for i in range(n): t[i] -= 1
t.sort()
unread = 0
ans = 0
for i in range(T):
    while unread < n and t[unread] == i:
        money[unread] -= B
        unread += 1
    ans += unread * C
    for j in range(unread, n):
        ans += max(money[j], 0)
print(ans)
