n = int(input().strip())
a = list(map(int, input().strip().split()))

min_val = min(a)
min_indices = [i for i, x in enumerate(a) if x == min_val]

min_distance = float('inf')
for i in range(1, len(min_indices)):
    min_distance = min(min_distance, min_indices[i] - min_indices[i-1])

print(min_distance)
