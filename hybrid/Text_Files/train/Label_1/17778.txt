#include <iostream>
#include <vector>
#include <cmath>
using namespace std;
int main() {
    int n; cin >> n; vector<double> p(n); 
    for (int i = 0; i < n; ++i) cin >> p[i];
    double res = 1;
    for (int i = 0; i < n; ++i) res -= (1 - p[i]) * (1 - p[i]);
    cout.precision(12); cout << fixed << res << endl;
}
