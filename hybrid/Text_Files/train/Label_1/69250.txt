n, m = map(int, input().split())

# Initialize friendships and vulnerability status
friendships = [set() for _ in range(n + 1)]
vulnerable = [False] * (n + 1)

# Read initial friendships
for _ in range(m):
    u, v = map(int, input().split())
    friendships[u].add(v)
    friendships[v].add(u)
    # Update vulnerability status
    if u > v:
        vulnerable[u] = True
    if v > u:
        vulnerable[v] = True

q = int(input())

for _ in range(q):
    query = list(map(int, input().split()))
    
    if query[0] == 1:
        u, v = query[1], query[2]
        friendships[u].add(v)
        friendships[v].add(u)
        # Update vulnerability status
        if u > v:
            vulnerable[u] = True
        if v > u:
            vulnerable[v] = True
    
    elif query[0] == 2:
        u, v = query[1], query[2]
        friendships[u].remove(v)
        friendships[v].remove(u)
        # Need to re-evaluate vulnerability
        vulnerable[u] = any(u < friend for friend in friendships[u])
        vulnerable[v] = any(v < friend for friend in friendships[v])
    
    elif query[0] == 3:
        alive = set(range(1, n + 1))
        while True:
            killed = set()
            for noble in alive:
                if vulnerable[noble] and all(friend in alive for friend in friendships[noble]):
                    killed.add(noble)
            if not killed:
                break
            alive -= killed
            for noble in killed:
                vulnerable[noble] = False
        print(len(alive))
