import java.util.Scanner;
import java.util.Stack;

public class DimaAndInna {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }

        Stack<Integer> stack = new Stack<>();

        for (int i = 0; i < n; i++) {
            int event = scanner.nextInt();

            if (event == -1) {
                // Dima's hit
                int k = 0;
                while (!stack.isEmpty() && k < m && a[k] <= stack.size()) {
                    stack.pop();
                    k++;
                }
            } else {
                // Adding a number to the sequence
                stack.push(event);
            }
        }

        if (stack.isEmpty()) {
            System.out.println("Poor stack!");
        } else {
            StringBuilder sb = new StringBuilder();
            while (!stack.isEmpty()) {
                sb.append(stack.pop());
            }
            System.out.println(sb.reverse().toString());
        }
    }
}
