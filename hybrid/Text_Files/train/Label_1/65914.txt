n = int(input())
events = []
for _ in range(n):
    l, r = map(int, input().split())
    events.append((l, "L"))
    events.append((r + 1, "R"))

events.sort()

cnt = [0] * (n + 1)
covered = 0
last_position = events[0][0]

for event in events:
    position, type = event
    cnt[covered] += position - last_position
    if type == "L":
        covered += 1
    else:
        covered -= 1
    last_position = position

print(*cnt[1:])
