#include <iostream>
#include <vector>
#include <cmath>
using namespace std;
typedef long long ll;
const ll maxn = 100005;
ll a[maxn], pref[maxn], ans = 2;

int main() {
  ll n; cin >> n;
  for (ll i = 1; i <= n; ++i) { cin >> a[i]; pref[i] = pref[i - 1] + a[i]; }
  for (ll i = 1; i < n; ++i) {
    ll left = i, right = n;
    while (left < right) {
      ll mid = (left + right) >> 1;
      double mean = (double)(pref[mid] - pref[i - 1]) / (mid - i + 1);
      if (mean < (double)(pref[n] - pref[mid]) / (n - mid)) left = mid + 1;
      else right = mid;
    }
    ans = max(ans, right - i + 1);
  }
  cout << ans << '\n';
  return 0;
}
