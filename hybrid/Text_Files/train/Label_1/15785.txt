#include <iostream>
#include <string>

int main() {
  int n;
  std::cin >> n;
  std::string s;
  std::cin >> s;
  for (int i = 0; i < n; i++) {
    if (s[i] == '1' && (i == 0 || s[i - 1] == '0') && (i == n - 1 || s[i + 1] == '0')) {
      std::cout << "Yes" << std::endl;
      return 0;
    }
  }
  std::cout << "No" << std::endl;
  return 0;
}
