#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(m), ans;
  unordered_map<int, int> l;
  for (int i = 0; i < n; ++i) { cin >> a[i]; l[a[i]] = i; }
  for (int i = 0; i < m; ++i) cin >> b[i];
  for (int i = 0; i < m; ++i) {
    int j = b[i];
    if (l.count(j) == 0) continue;
    while (ans.size() > 0 && ans.back() > j) ans.pop_back();
    while (ans.size() > 0 && l[ans.back()] > l[j]) ans.pop_back();
    ans.push_back(j);
  }
  for (int j : ans) cout << j << " ";
}
