import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int k = sc.nextInt();
        long ans = (n + 9) / 10;
        if (ans % k == 0) {
            System.out.println(0);
        } else {
            System.out.println(k - ans % k);
        }
    }
}
