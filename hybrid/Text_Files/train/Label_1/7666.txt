#include <iostream>
#include <vector>
using namespace std;

bool inRange(int l, int r, int i) {
    return l <= i && i <= r;
}

int f(const string& s, int l, int r) {
    int n = r - l + 1;
    if (n == 1) return 0;
    vector<int> v;
    for (int i = l; i <= r; i++) {
        v.push_back(s[i - 1] - '0');
    }
    int ans = 0;
    while (v.size() > 1) {
        vector<int> t;
        for (int i = 0; i < v.size(); i += 2) {
            int sum = v[i] + v[i + 1];
            ans += (sum >= 10);
            t.push_back(sum % 10);
        }
        v = t;
    }
    return ans;
}

int main() {
    int n; cin >> n;
    string s; cin >> s;
    int q; cin >> q;
    while (q--) {
        int l, r; cin >> l >> r;
        cout << f(s, l, r) << endl;
    }
    return 0;
}
