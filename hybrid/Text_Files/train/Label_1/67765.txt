import heapq

n = int(input())
r = list(map(int, input().split()))
s = [0] * n
heapq.heapify(r)

mx = r[0]
for i in r:
    mx = max(i, mx)

ans = []

while r[0] != mx:
    t = []
    cnt = 5
    while cnt != 0:
        if s[0] == 0 and r[0] < mx:
            t.append(0)
            s[0] = 1
        else:
            t.append(1)
            heapq.heappop(r)
        cnt -= 1
    ans.append(t)
    for i in range(n):
        if t[i] == 1:
            s[i] = 0

print(mx)
print(len(ans))
for a in ans:
    print(''.join(map(str, a)))
