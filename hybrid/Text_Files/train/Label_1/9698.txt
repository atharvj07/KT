#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> b(n + 1, 0);
  for (int i = 1; i <= n; ++i) std::cin >> b[i];
  b.insert(b.begin(), 0);

  std::vector<int> dp(n + 1, 0);
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= i; ++j) {
      if (b[i] - b[j] == i - j) {
        dp[i] = std::max(dp[i], dp[j] + b[i]);
      }
    }
  }

  int ans = 0;
  for (int i = 1; i <= n; ++i) ans = std::max(ans, dp[i]);

  std::cout << ans << "\n";
  return 0;
}
