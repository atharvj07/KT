import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[][] a = new int[n][m];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    a[i][j] = sc.nextInt();
                }
            }
            boolean[][] vis = new boolean[n][m];
            boolean ashishWins = true;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (a[i][j] == 0) {
                        vis[i][j] = true;
                        if (isWinning(a, vis, i, j, n, m)) {
                            ashishWins = false;
                            break;
                        }
                        vis[i][j] = false;
                    }
                }
            }
            System.out.println(ashishWins ? "Ashish" : "Vivek");
        }
    }

    static boolean isWinning(int[][] a, boolean[][] vis, int x, int y, int n, int m) {
        for (int i = 0; i < n; i++) {
            if (a[i][y] == 0 && !vis[i][y]) {
                vis[i][y] = true;
                if (!isWinning(a, vis, i, y, n, m)) {
                    vis[i][y] = false;
                    return true;
                }
                vis[i][y] = false;
            }
        }
        for (int j = 0; j < m; j++) {
            if (a[x][j] == 0 && !vis[x][j]) {
                vis[x][j] = true;
                if (!isWinning(a, vis, x, j, n, m)) {
                    vis[x][j] = false;
                    return true;
                }
                vis[x][j] = false;
            }
        }
        return false;
    }
}
