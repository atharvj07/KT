#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll dist[100001];
vector<pair<ll, ll>> adj[100001];
int main() {
  ll n, m, a, b, ans = 0, mn = LLONG_MAX;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> dist[i];
  for (int i = 1; i <= m; i++) {
    cin >> a >> b;
    adj[a].push_back({b, 0});
    adj[b].push_back({a, 0});
  }
  priority_queue<pair<ll, ll>, vector<pair<ll, ll>>, greater<pair<ll, ll>>> pq;
  pq.push({0, 1});
  while (!pq.empty()) {
    a = pq.top().second;
    b = pq.top().first;
    pq.pop();
    if (dist[a] < b) continue;
    dist[a] = b;
    for (auto c : adj[a]) {
      if (dist[c.first] > b + c.second) pq.push({b + c.second, c.first});
    }
  }
  for (int i = 1; i <= n; i++) ans += dist[i];
  cout << ans << '\n';
}
