n, b, a = map(int, input().split())
s = list(map(int, input().split()))
if s[0] == 0:
    b -= 1
else:
    a += 1
ans = 0
for i in range(n - 1):
    if s[i + 1] == 0:
        if b > 0:
            b -= 1
            ans += 1
        else:
            break
    else:
        if a > 0:
            a -= 1
            ans += 1
        else:
            break
print(ans)
