#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
  int n;
  cin >> n;
  vector<ll> a(n), c(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; c[i] = a[i]; }
  sort(c.begin(), c.end());
  for (int i = 0; i < n; i++) c[i + 1] += c[i];
  int m;
  cin >> m;
  while (m--) {
    int t, l, r;
    cin >> t >> l >> r;
    if (t == 1)
      cout << a[l - 1] + a[r - 1] << '\n';
    else
      cout << c[r - 1] - c[l - 2] << '\n';
  }
}
