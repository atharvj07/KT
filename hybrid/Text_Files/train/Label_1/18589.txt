#include <cstdio>

int n, m, a[51];
int cnt[51][51][51]={};
int f[51][51]={};

int calc(int pos, int last, int prv){
    if(pos == n+1)
        return 1;
    int &ret = cnt[pos][last][prv];
    if(ret+1)
        return ret;
    ret = 0;
    for(int i = last; i <= n; i++){
        if(i != last && prv >= f[i-last+1][i]){
            continue;
        }
        ret += calc(pos+1, i, prv+f[last+1][i]);
        if(ret >= m)
            return ret;
    }
    return ret;
}

int main(){
    scanf("%d%d", &n, &m);
    for(int l = 1; l <= n; l++){
        for(int r = l; r <= n; r++){
            for(int i = l; i <= r; i++)
                f[l][r] += i;
        }
    }
    calc(1, 0, 0);
    for(int l = 1; l <= n; l++){
        for(int i = l; i <= n; i++){
            if(i != l && m > cnt[i][l][f[l+1][i]])
                m -= cnt[i][l][f[l+1][i]];
            else{
                printf("%d ", i);
                return 0;
            }
        }
    }
}
