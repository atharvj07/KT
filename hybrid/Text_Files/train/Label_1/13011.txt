#include<bits/stdc++.h>
using namespace std;
int main() {
  string s;
  getline(cin, s);
  string l = s.substr(0, s.find('|'));
  string r = s.substr(s.find('|') + 1);
  map<char, int> m;
  for(char c : l) m[c]++;
  for(char c : r) m[c]++;
  string t;
  for(auto c : m) t += string(c.second, c.first);
  if(t.size() % 2 == 0 && t.substr(0, t.size() / 2) == t.substr(t.size() / 2)) {
    cout << t.substr(0, t.size() / 2) + '|' + t.substr(t.size() / 2);
  }
  else cout << "Impossible";
}
