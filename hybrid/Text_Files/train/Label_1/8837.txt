#include <bits/stdc++.h>
using namespace std;
const int MXN = 55;
vector<int> adj[MXN];
int h, q, lg[MXN], ans[MXN][MXN];
int main() {
    cin >> h >> q;
    for (int i = 1; i < MXN; ++i) lg[i] = log2(i);
    for (int i = 0, a, b; i < q; ++i) {
        cin >> a >> b >> lg[ans[a][b] = -1];
    }
    for (int i = 1; i < h + 1; i++) {
        for (int x = 2, y = 2; y < (1 << i) - 1; ++x, ++y) {
            adj[x].push_back(y);
            adj[y].push_back(x);
        }
    }
    int curr = 0;
    for (int i = h; i >= 1; i--) {
        if (lg[ans[i][2 << i - 1]] == -1)
            curr = 0;
        else if (ans[i][2 << i - 1] && curr)
            curr = 2 << i - 1;
        else if (!ans[i][2 << i - 1] && !curr)
            curr = 2 << i - 1;
        else {
            cout << "Game cheated!" << endl;
            return 0;
        }
    }
    if (curr)
        cout << curr << endl;
    else
        cout << "Data not sufficient!" << endl;
    return 0;
}
