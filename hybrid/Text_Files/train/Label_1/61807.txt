n = int(input())
s = list(map(int, input().split()))
q = int(input())
l = []
r = []
for i in range(q):
    a, b = map(int, input().split())
    l.append(a)
    r.append(b)
for i in range(n):
    s[i] += l[0]
for j in range(q):
    t = set()
    for i in range(n):
        t.add(s[i] + l[j] + r[j])
    print(len(t))
