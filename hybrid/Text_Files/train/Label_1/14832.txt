#include <bits/stdc++.h>

using namespace std;

long long d, n, m;
vector<pair<long long, long long>> v;
const long long INF = 1e18;

int main() {
  cin >> d >> n >> m;
  for (int i = 0; i < m; i++) {
    long long x, p;
    cin >> x >> p;
    v.push_back({x, p});
  }
  sort(v.begin(), v.end());
  long long ans = 0, j = 0;
  priority_queue<long long, vector<long long>, greater<long long>> pq;
  for (long long i = 0; i < d;) {
    while (j < m && v[j].first <= i) {
      pq.push(v[j].second);
      j++;
    }
    if (!pq.empty()) {
      long long cur = pq.top();
      pq.pop();
      if (i + n >= d) {
        ans += (d - i) * cur;
        break;
      }
      if (cur == -1) {
        ans = -1;
        break;
      }
      ans += n * cur;
      i += n;
    } else {
      ans = -1;
      break;
    }
  }
  cout << ans;
  return 0;
}
