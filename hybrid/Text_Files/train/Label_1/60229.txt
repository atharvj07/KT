import math
for t in range(int(input())):
    n,k=map(int,input().split())
    arr=[0]*n
    for i in range(n):
        arr[i]=int(input())
    ans=[]
    for j in arr:
        if j%k==0:
            ans.append(0)
        else:
            m=math.ceil(j/k)
            ans.append((m*k-j))
    print(sum(ans))
