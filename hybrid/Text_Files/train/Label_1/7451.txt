#include <bits/stdc++.h>
using namespace std;

int main() {
    int m, d, a, b;
    cin >> m >> d >> a >> b;
    long long base = 1;
    long long base_10 = 10;
    long long ans = 0;
    for (int i = a; i < b + 1; ++i) {
        string s = to_string(i);
        int lastDPos = -1;
        bool ok = true;
        for (int j = 0; j < s.length(); ++j) {
            if (s[j] == '0' + d) {
                if (j % 2 == 0) {
                    if (lastDPos != -1) {
                        ok = false;
                        break;
                    }
                    lastDPos = j;
                } else {
                    ok = false;
                    break;
                }
            }
        }
        if (ok && i % m == 0) {
            if (lastDPos == -1) lastDPos = s.length();
            long long cnt = (b - a + 1) / base_10 / m;
            if (lastDPos % 2 == 0) {
                long long k = lastDPos / 2 - 1;
                if (k > 0) {
                    cnt -= ((k - 1) * base_10 + 1) / m;
                }
            } else {
                long long k = lastDPos / 2;
                if (k > 0) {
                    cnt -= ((k - 1) * base_10 + 1) / m;
                }
            }
            ans = (ans + cnt) % (int)(1e9 + 7);
        }
        base *= 10;
    }
    cout << ans << endl;
    return 0;
}
