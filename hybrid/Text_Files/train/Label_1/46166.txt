import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            long a = input.nextInt(), b = input.nextInt();
            long res = Math.min(a / 2, b);
            b -= res;
            res += Math.min(a - 2 * res, 2 * b);
            System.out.println(res);
        }
        input.close();
    }
}
