import sys

n, m = [int(x) for x in input().split()]
a = [[int(x) for x in input()] for _ in range(n)]

ans = -1
for i in range(n):
    for j in range(m):
        cnt = 0
        for x in range(i, n):
            for y in range(j, m):
                cnt += (x - i + 1) * (y - j + 1) % 2 * a[x][y]
        if cnt % 2 == a[i][j]:
            ans = 0
            break
    if ans == 0:
        break

print(ans)
