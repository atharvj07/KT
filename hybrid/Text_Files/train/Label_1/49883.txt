import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), i = 0, ans = 0;
        int[] a = new int[n], f = new int[n + 1];
        for (; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (; i > 0; i--) {
            f[a[i - 1]]++;
            if (f[a[i - 1]] == 1) ans++;
        }
        System.out.println(Math.min(ans, k));
    }
}
