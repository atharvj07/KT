import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        long[] a = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextLong();
        }
        long max = 0;
        for (int i = 0; i < k; i++) {
            max += a[i];
        }
        if (max % 2 != 0) {
            System.out.println(-1);
            return;
        }
        long min = 2;
        for (int i = 0; i < k; i++) {
            long groups = max / a[i];
            min = Math.max(min, groups * a[i]);
        }
        min++;
        System.out.println(min + " " + max);
    }
}
