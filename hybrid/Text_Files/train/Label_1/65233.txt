import sys
input = sys.stdin.readline
n,m=map(int,input().split())
a=[0]*n
b=[0]*n
r=[0]*m
for i in range(m):
 r[i]=input().rstrip()
if m>n-1:
 print("NO")
else:
 print("YES")
 b[0]=1
 c=0
 for i in range(1,n):
  if c==m:break
  if r[c]=="greater":
   if a[i-1]==0:
    a[i-1]=i+1
    a[i]=i
    b[i]=b[i-1]
    c+=1
   elif b[i-1]==0:
    a[i]=i
    b[i]=b[i-1]
    c+=1
   else:
    c+=1
  elif r[c]=="less":
   if a[i-1]==0:
    a[i-1]=i

    b[i]=b[i-1]
    c+=1
   elif b[i-1]==0:
    a[i]=i+1
    b[i]=b[i-1]
    c+=1
   else:
    c+=1
  else:
   if a[i-1]==0:
    a[i-1]=i
    a[i]=i
    b[i]=i+1
    c+=1
   elif b[i-1]==0:
    a[i]=i
    b[i]=i+1
    c+=1
   else:
    c+=1
 if c==m:
