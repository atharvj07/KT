n = int(input())
triples = [list(map(int, input().split())) for _ in range(n-2)]

count = [0] * (n + 1)
adj = [[] for _ in range(n + 1)]

for a, b, c in triples:
    count[a] += 1
    count[b] += 1
    count[c] += 1
    adj[a].append(b)
    adj[a].append(c)
    adj[b].append(a)
    adj[b].append(c)
    adj[c].append(a)
    adj[c].append(b)

p = [0] * n
for i in range(1, n+1):
    if count[i] == 1:
        p[0] = i
        break

for i in range(1, n+1):
    if count[i] == 2:
        if p[0] in adj[i]:
            p[1] = i
            break

for i in range(2, n):
    a, b = p[i - 2], p[i - 1]
    for next in adj[b]:
        if next != a and next != b:
            p[i] = next
            break

print(*p)
