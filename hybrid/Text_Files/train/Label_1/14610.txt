#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  set<long long> s;
  int ans = 0;
  while (n--) {
    long long v;
    char ch;
    cin >> ch >> v;
    if (ch == '+')
      s.insert(v);
    else
      s.erase(v);
    ans = max(ans, (int)s.size());
  }
  cout << ans << endl;
  return 0;
}
