import java.util.Arrays;
import java.util.Scanner;

public class Championship {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
            }

            int[] playersWithNonZeroProbability = findPlayersWithNonZeroProbability(a);
            System.out.println(playersWithNonZeroProbability.length);
            Arrays.sort(playersWithNonZeroProbability);
            for (int player : playersWithNonZeroProbability) {
                System.out.print(player + " ");
            }
            System.out.println();
        }
    }

    static int[] findPlayersWithNonZeroProbability(int[] a) {
        int n = a.length;
        int[] playersWithNonZeroProbability = new int[n];
        int index = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] > 0) {
                playersWithNonZeroProbability[index++] = i + 1;
            }
        }
        return Arrays.copyOf(playersWithNonZeroProbability, index);
    }
}
