#include <iostream>
#include <set>
#include <vector>

using namespace std;

int main() {
  int k;
  cin >> k;
  vector<int> r(k);
  for (int i = 0; i < k; i++) cin >> r[i];
  set<int> s;
  for (int i = 1; i <= 25; i++) s.insert(i);
  for (int i = 0; i < k; i++) s.erase(r[i]);
  cout << *s.begin() - 1 << endl;
  return 0;
}
