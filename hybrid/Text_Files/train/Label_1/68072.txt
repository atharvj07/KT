n,k=map(int,input().split())
ip=[list(map(int,i.split('.')))for i in(input()for _ in range(n))]
ans=['.'*4]
a=0
for i in range(32,-1,-1):
  b=[0]*4
  for j in range(n):
    b[j//8]|=(ip[j][j//8]&(1<<i))>>(i%8)
  cost=n*(i+1)-sum(map(lambda x:bin(x).count('1'),b))
  if cost<=k:
    k-=cost
    for j in range(n):
      ip[j][j//8]|=(1<<i)>>(i%8)
    a|=1<<i
    ans=[str(b[0]),str(b[1]),str(b[2]),str(b[3])]
  else:
    for j in range(n):
      ip[j][j//8]&=~(1<<i)>>(i%8)
if a>0:print('.'.join(ans))
else:print(-1)
