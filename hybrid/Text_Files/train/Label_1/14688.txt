#include<bits/stdc++.h>
using namespace std;
typedef unsigned long long ull;
typedef long long ll;
typedef long double ld;
typedef pair<ull,ull> p64;
typedef pair<int,int> p32;

#define deb(x) cout<<#x<<"="<<x<<endl
#define FIO ios::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL)

const ull MOD = 1000000007;
const ull MAXN = 1e8+1;
ull gcd(ull a,ull b){
    return (b==0)?a:gcd(b,a%b);
}
int main(){
   FIO;
   ull n,m,k;
   cin>>n>>m>>k;
   ull a[k];
   for(ull i=0;i<k;i++)cin>>a[i];
   ull g=a[0];
   for(ull i=1;i<k;i++)g=gcd(g,a[i]);
   bool f=0;
   for(ull i=1;i<=n;i++){
       if(i%g!=0)continue;
       ull j=(i/g);
       if(j+k-1>m)continue;
       bool f1=1;
       for(ull ii=0;ii<k;ii++)if(g*j+ii+1!=a[ii])f1=0;
       if(f1){f=1;break;}
    }
   cout<<(f?"YES":"NO")<<endl;
   return 0;
}
