#include <iostream>
#include <vector>
#include <unordered_map>
#include <algorithm>
using namespace std;
int mod = 1e9 + 7;
int main() {
    int h, w;
    cin >> h >> w;
    vector<int> r(h), c(w);
    for (int i = 0; i < h; i++) { cin >> r[i]; }
    for (int j = 0; j < w; j++) { cin >> c[j]; }
    vector<unordered_map<int, int>> dp(h + 1, unordered_map<int, int>());
    dp[0][0] = 1;
    for (int i = 0; i < h; i++) {
        for (int j = 0; j <= w; j++) {
            for (int k = 0; k <= r[i]; k++) {
                dp[i + 1][k + c[i + 1]] = (dp[i + 1][k + c[i + 1]] + dp[i][k]) % mod;
            }
        }
    }
    cout << dp[h][w] << endl;
    return 0;
}
