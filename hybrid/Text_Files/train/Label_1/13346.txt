#include <iostream>
#include <vector>
using namespace std;
int main()
{
    int n;
    cin >> n;
    vector<int> b(n + 1), a(n + 1), x(n), k(n);
    for (int i = 1; i <= n; i++) cin >> b[i];
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 0; i < n - 1; i++) { cin >> x[i] >> k[i]; }
    for (int i = 1; i <= n; i++)
        if (a[i] <= b[i] + a[x[i]] * k[i]) { cout << "YES" << endl; return 0; }
    cout << "NO" << endl;
    return 0;
}
