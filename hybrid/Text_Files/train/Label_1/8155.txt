#include <algorithm>
#include <iostream>
#include <vector>

int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }
int lcm(int a, int b) { return a * b / gcd(a, b); }

int main() {
  long long a, b;
  std::cin >> a >> b;
  for (int i = 0; i < 1000000; i++) {
    if (lcm(a + i, b + i) < lcm(a + i + 1, b + i + 1)) {
      std::cout << i << std::endl;
      return 0;
    }
  }
  std::cout << std::min(a, b) << std::endl;
  return 0;
}
