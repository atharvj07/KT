#include <bits/stdc++.h>

using namespace std;

#define ll long long
#define vi vector<int>

int main() {
    int m, n, k, t, ans = 0;
    cin >> m >> n >> k >> t;
    vi a(m + 1), pos(2e5 + 5), danger(2e5 + 5), dp(2e5 + 5, -1e9);
    for (int i = 1; i <= m; i++) cin >> a[i];
    for (int i = 1; i <= k; i++) {
        int l, r, d;
        cin >> l >> r >> d;
        pos[l] = i;
        danger[l] = d;
    }
    dp[0] = 0;
    for (int i = 1; i <= n + 1; i++) {
        dp[i] = max(dp[i], dp[i - 1]);
        if (i == t) continue;
        if (pos[i]) {
            dp[i] = max(dp[i], dp[pos[i]]);
        }
        for (int j = 1; j <= m; j++) {
            if (a[j] >= danger[i]) { dp[i + 1] = max(dp[i + 1], dp[i] + 1); }
        }
    }
    cout << dp[n + 1];
    return 0;
}
