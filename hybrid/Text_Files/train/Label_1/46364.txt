import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] lights = new int[m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                if (s.charAt(j) == '1') {
                    lights[j]++;
                }
            }
        }
        boolean ans = false;
        for (int i = 0; i < m; i++) {
            if (lights[i] == n - 1) {
                ans = true;
                break;
            }
        }
        System.out.println(ans ? "YES" : "NO");
    }
}
