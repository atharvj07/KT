#include <iostream>
#include <cmath>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int h, c, tt;
        cin >> h >> c >> tt;
        if (h == tt) {
            cout << 1 << endl;
        } else if (c == tt) {
            cout << -1 << endl;
        } else {
            cout << ceil((double)(h - tt) / (h - c)) + 1 << endl;
        }
    }
    return 0;
}
