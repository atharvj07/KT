#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> a(n);
    vector<pair<int, int>> b(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i].first >> a[i].second;
        b[i].first = a[i].second;
        b[i].second = i;
    }
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    int l = 0, r = n - 1;
    vector<int> ans;
    while (l < r) {
        if (a[l].first < b[l].first) {
            ans.push_back(b[l].second + 1);
            l++;
        } else {
            ans.push_back(b[r].second + 1);
            r--;
        }
    }
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
