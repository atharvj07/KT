def max_video_cards_power(n, powers):
    powers.sort()
    max_total_power = 0
    for i in range(n):
        leading_power = powers[i]
        total_power = leading_power
        for j in range(i + 1, n):
            if powers[j] % leading_power == 0:
                total_power += powers[j]
        max_total_power = max(max_total_power, total_power)
    return max_total_power

n = int(input())
powers = list(map(int, input().split()))
print(max_video_cards_power(n, powers))
