#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, p;
  cin >> n >> p;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> suf(n);
  suf[n - 1] = a[n - 1];
  for (int i = n - 2; i >= 0; i--) suf[i] = (a[i] + suf[i + 1]) % p;
  vector<int> pre(n);
  pre[0] = a[0];
  for (int i = 1; i < n; i++) pre[i] = (a[i] + pre[i - 1]) % p;
  int ans = 0;
  for (int i = 0; i < n - 1; i++) ans = max(ans, (suf[i + 1] + pre[i]) % p);
  cout << ans << endl;
  return 0;
}
