n = int(input())
d = int(input())
e = int(input())

min_rubles = n
for i in range(0, n // (5 * e) + 1):
    euros_to_rubles = i * 5 * e
    dollars_left = (n - euros_to_rubles) // d * d
    rubles_left = n - euros_to_rubles - dollars_left
    min_rubles = min(min_rubles, rubles_left)

print(min_rubles)
