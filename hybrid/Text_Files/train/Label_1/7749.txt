#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
typedef long long ll;

int main() {
	ll n, t;
	cin >> n >> t;
	vector<int> a(n);
	for (int i = 0; i < n - 1; i++) cin >> a[i];
	vector<bool> vis(n + 3, false);
	queue<ll> q;
	vis[1] = true;
	q.push(1);
	while (!q.empty()) {
		ll u = q.front();
		q.pop();
		if (u == t) {
			cout << "YES" << endl;
			return 0;
		}
		for (ll i = u; i < u + a[u - 1] && i <= n; i++) {
			if (!vis[i]) {
				vis[i] = true;
				q.push(i);
			}
		}
	}
	cout << "NO" << endl;
	return 0;
}
