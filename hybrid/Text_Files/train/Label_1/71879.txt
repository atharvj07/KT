import bisect
import sys
input = sys.stdin.readline
n,m,k = map(int,input().split())
a = list(map(int,input().split()))
b = list(map(int,input().split()))
b.sort()
a.sort()
j=0
ans = [0 for i in range(k+1)]
for i in range(k+1):
    while j<m and b[j] <= i:
        j+=1
    ans[i] = j
for i in range(k,0,-1):
    ans[i] = min(ans[i],ans[i-1])
mn = 10**9
for i in range(k):
    if ans[i] == m:
        mn = min(mn,a[i])
if mn==10**9: print(-1)
else: print(mn)
