import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        Point[] points = new Point[n];

        for (int i = 0; i < n; i++) {
            points[i] = new Point(sc.nextInt(), sc.nextInt(), sc.nextInt());
        }

        Arrays.sort(points, (a, b) -> a.x - b.x);

        int ans = 0;

        for (int i = 0; i < n; i++) {
            int l = i - 1;
            int r = i + 1;
            int maxR = points[i].x + points[i].radius;
            int minR = points[i].x - points[i].radius;

            while (l >= 0 && points[l].x >= minR) {
                if (Math.abs(points[i].iq - points[l].iq) <= k) {
                    ans++;
                }
                l--;
            }

            while (r < n && points[r].x <= maxR) {
                if (Math.abs(points[i].iq - points[r].iq) <= k) {
                    ans++;
                }
                r++;
            }
        }

        System.out.println(ans);
    }
}

class Point {
    int x, radius, iq;

    public Point(int x, int radius, int iq) {
        this.x = x;
        this.radius = radius;
        this.iq = iq;
    }
}
