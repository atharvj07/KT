n = int(input())
way = input()
dist = list(map(int, input().split()))
used = [False for _ in range(n)]
last = 0
while not used[last]:
    used[last] = True
    if last == 0 and way[0] == "<":
        break
    if last == n - 1 and way[n - 1] == ">":
        break
    if way[last] == ">":
        last -= dist[last]
        last = (last + n) % n
    else:
        last += dist[last]
        last %= n
print("INFINITE" if not used.count(True) == n else "FINITE")
