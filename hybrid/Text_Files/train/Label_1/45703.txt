import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        sc.nextLine();
        String s = sc.nextLine();
        int[] a = new int[26];
        for (int i = 0; i < n; i++) {
            a[s.charAt(i) - 'A']++;
        }
        int cnt = 0;
        for (int i = 0; i < 26; i++) {
            if (a[i] > 0) {
                cnt++;
            }
        }
        if (cnt > k) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
