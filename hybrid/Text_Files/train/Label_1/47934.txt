import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int[] h = new int[n];
    for (int i = 0; i < n; i++) {
      h[i] = scanner.nextInt();
    }
    int[] dp = new int[n];
    Arrays.fill(dp, Integer.MAX_VALUE);
    dp[n - 1] = 0;
    for (int i = n - 2; i >= 0; i--) {
      for (int j = i + 1; j < n; j++) {
        if (h[i] > h[j] || h[i] == h[j]) {
          if (dp[j] != Integer.MAX_VALUE) {
            dp[i] = Math.min(dp[i], dp[j] + 1);
          }
        }
        if (h[j] > h[i] || h[i] == h[j]) {
          if (dp[j] != Integer.MAX_VALUE) {
            dp[i] = Math.min(dp[i], dp[j] + 1);
          }
        }
      }
    }
    System.out.println(dp[0]);
  }
}
