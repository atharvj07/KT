#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        string s;
        cin >> s;
        int cnt = 0;
        for (int i = k; i < n; i++) {
            if (!cnt && s[i] == '0' && s[i - k] == '0') {
                cnt++;
            } else if (cnt && s[i] == '0' && s[i - k] == '1') {
                cnt = 0;
            }
        }
        cout << cnt << endl;
    }
    return 0;
}
