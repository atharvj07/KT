#include <bits/stdc++.h>
using namespace std;

int main() {
    int q; cin >> q;
    while (q--) {
        long long x1, x2, y1, y2;
        cin >> x1 >> x2 >> y1 >> y2;
        if (x1 != y1) {
            cout << x1 << " " << y1 << "\n";
        } else {
            cout << x1 << " " << y2 << "\n";
        }
    }
    return 0;
}
