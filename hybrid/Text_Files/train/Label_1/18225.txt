#include <bits/stdc++.h>
using namespace std;

bool comm(int a, int b, int c, int d) {
    vector<int> v(10, 0);
    v[a] = v[b] = v[c] = v[d] = 1;
    int tmp = 0;
    for (int i = 1; i <= 9; ++i)
        tmp += v[i];
    return tmp == 2;
}

int main() {
    int n, m, x;
    cin >> n >> m;
    bool a = 0, b = 0;
    for (int i = 0; i < n; ++i) {
        cin >> x;
        if (x <= 4) a = 1;
        if (x > 4) b = 1;
    }
    for (int i = 0; i < m; ++i) {
        cin >> x;
        if (x <= 4) a = 1;
        if (x > 4) b = 1;
    }
    if (!a) {
        cout << "1\n";
        return 0;
    }
    if (!b) {
        cout << "4\n";
        return 0;
    }
    for (int i = 0; i < n; ++i) {
        int aa, bb;
        cin >> aa >> bb;
        for (int j = 0; j < m; ++j) {
            int cc, dd;
            cin >> cc >> dd;
            if (comm(aa, bb, cc, dd)) {
                if (comm(aa, bb, dd, cc))
                    continue;
                cout << "0\n";
                return 0;
            }
        }
    }
    cout << "-1\n";
}
