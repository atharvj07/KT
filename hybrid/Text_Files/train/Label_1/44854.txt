import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int q = sc.nextInt();
    char[] s = sc.next().toCharArray();
    StringBuilder[] ss = new StringBuilder[q];
    for (int i = 0; i < q; i++) {
      ss[i] = new StringBuilder();
    }
    int[] cp = new int[q];
    int[][] dp = new int[q][2];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < q; j++) {
        if (cp[j] >= ss[j].length()) {
          continue;
        }
        while (cp[j] < ss[j].length() && ss[j].charAt(cp[j]) != s[i]) {
          cp[j]--;
          dp[j][0] *= -1;
          dp[j][1] *= -1;
        }
        if (cp[j] >= ss[j].length()) {
          continue;
        }
        char cc = ss[j].charAt(cp[j]);
        if (cc >= '0' && cc <= '9') {
          ss[j].replace(cp[j], cp[j] + 1, "" + (cc - '0' - 1));
          cp[j] += dp[j][1];
        } else if (cc == '<') {
          dp[j][1] = -1;
          cp[j] += dp[j][1];
        } else {
          dp[j][1] = 1;
          cp[j] += dp[j][1];
        }
      }
      s[i] = (s[i] <= '0') ? s[i] : (char) (s[i] - 1);
    }
    for (StringBuilder sss : ss) {
      int[] cnt = new int[10];
      Arrays.fill(cnt, 0);
      for (int i = 0; i < sss.length(); i++) {
        if (sss.charAt(i) >= '0' && sss.charAt(i) <= '9') {
          cnt[sss.charAt(i) - '0']++;
        }
      }
      for (int x : cnt) {
        System.out.print(x + " ");
      }
      System.out.println();
    }
  }
}
