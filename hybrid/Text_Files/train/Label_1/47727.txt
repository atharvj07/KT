import java.util.*;
import java.math.BigDecimal;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        double[] a = new double[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        double[] b = new double[n];
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }

        double[][] dp = new double[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            Arrays.fill(dp[i], Double.MAX_VALUE);
        }

        dp[1][0] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                if (dp[i][j] == Double.MAX_VALUE) {
                    continue;
                }

                // Try to lift off from planet i
                if (i < n) {
                    double fuelNeeded = (m + dp[i][j]) / a[i];
                    if (j + fuelNeeded <= m) {
                        dp[i + 1][j + (int) fuelNeeded] = Math.min(dp[i + 1][j + (int) fuelNeeded], dp[i][j] + fuelNeeded);
                    }
                }

                // Try to land on planet i
                if (i > 1) {
                    double fuelNeeded = (m + dp[i][j]) / b[i - 1];
                    if (j + fuelNeeded <= m) {
                        dp[i - 1][j + (int) fuelNeeded] = Math.min(dp[i - 1][j + (int) fuelNeeded], dp[i][j] + fuelNeeded);
                    }
                }
            }
        }

        double minFuel = Double.MAX_VALUE;
        for (int j = 0; j <= m; j++) {
            if (dp[n][j] != Double.MAX_VALUE) {
                minFuel = Math.min(minFuel, dp[n][j]);
            }
        }

        if (minFuel == Double.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.printf("%.10f", minFuel);
        }
    }
}
