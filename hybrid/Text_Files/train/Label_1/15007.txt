#include <bits/stdc++.h>
using namespace std;
const int maxn=1e2+5;
int n;
long long a[maxn];
int main()
{
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1, greater<int>());
    long long ans = 0;
    for (int i = 2; i <= n; i++)
        ans += a[i];
    cout << ans << endl;
    return 0;
}
