#include <bits/stdc++.h>
using namespace std;
#define ll long long

ll binPow(ll a, ll b) {
    ll res = 1;
    while (b > 0) {
        if (b & 1)
            res = res * a;
        a = a * a;
        b >>= 1;
    }
    return res;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin >> t;
    while (t--) {
        string x, y;
        cin >> x >> y;
        ll p = 0, n = x.size();
        for (ll i = n - 1; i >= 0; i--) {
            if (x[i] != y[i]) {
                p = i + 1;
                break;
            }
        }
        cout << p << endl;
    }
    return 0;
}
