import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        long[] arr = fastInputLong();
        long ind = arr[1];
        long s = 0;
        for (long i = 1; i <= arr[0]; i++) {
            s += i;
            if (ind <= s) {
                System.out.println(arr[0] - i + 1);
                return;
            }
        }
    }

    public static long[] fastInputLong() throws IOException {
        String[] strs = fastInputString().split(" ");
        long[] arr = new long[strs.length];
        for (int i = 0; i < arr.length; i++) {
            arr[i] = Long.parseLong(strs[i]);
        }
        return arr;
    }

    public static String fastInputString() throws IOException {
        try (InputStream inputStream = System.in;
             InputStreamReader inputStreamReader = new InputStreamReader(inputStream);
             BufferedReader bufferedReader = new BufferedReader(inputStreamReader)) {
            return bufferedReader.readLine();
        }
    }
}
