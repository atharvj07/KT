#include <iostream>
#include <vector>

int main() {
  int n, c;
  std::cin >> n >> c;
  std::vector<int> x(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> x[i];
  }
  int max_profit = 0;
  for (int i = 0; i < n - 1; ++i) {
    int profit = x[i] - c;  // Profit from selling the barrel on day i
    int remaining = profit;  // Remaining raspberries after selling the barrel
    for (int j = i + 1; j < n; ++j) {
      if (remaining >= x[j]) {
        remaining -= x[j];  // Buy a new barrel on day j if possible
        profit += x[j];    // Increase profit by the cost of the new barrel
      } else {
        break;
      }
    }
    max_profit = std::max(max_profit, profit);
  }
  std::cout << max_profit << '\n';
}
