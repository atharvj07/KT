import sys
input = sys.stdin.readline

n, d = map(int, input().split())
conns = [[-1] * (n + 1) for _ in range(d + 1)]
for i in range(1, d + 1):
    a, b = map(int, input().split())
    conns[i][a] = b
    conns[i][b] = a

dp = [[0] * (n + 1) for _ in range(d + 1)]
result = [0] * (d + 1)
for i in range(1, d + 1):
    for j in range(1, n + 1):
        if j not in conns[i]:
            dp[i][j] = dp[i - 1][j] + 1
        else:
            dp[i][j] = max(dp[i - 1][j], dp[i - 1][conns[i][j]])
    result[i] = max(dp[i])

for r in result[1:]:
    print(r)
