from collections import defaultdict
from heapq import heappush, heappop

def dijkstra(graph, start):
    distance = defaultdict(lambda: float('inf'))
    distance[start] = 0
    queue = [(0, start)]
    while queue:
        dist, current = heappop(queue)
        if dist > distance[current]:
            continue
        for neighbor, cost in graph[current].items():
            new_dist = dist + cost
            if new_dist < distance[neighbor]:
                distance[neighbor] = new_dist
                heappush(queue, (new_dist, neighbor))
    return distance

def solve(n, m, a, b, edges):
    graph = defaultdict(dict)
    for u, v, c in edges:
        if v in graph[u]:
            graph[u][v] = graph[v][u] = min(graph[u][v], c)
        else:
            graph[u][v] = graph[v][u] = c
            
    mst = defaultdict(dict)
    for i in range(1, n + 1):
        distances = dijkstra(graph, i)
        for j in range(i + 1, n + 1):
            if distances[j] == a or distances[j] == b:
                mst[i][j] = mst[j][i] = distances[j]
    
    result = [0] * n
    for i in range(1, n + 1):
        distances = dijkstra(mst, i)
        for j in range(1, n + 1):
            result[j-1] = distances[j]
    return result

n, m, a, b = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(m)]
print(*solve(n, m, a, b, edges))
