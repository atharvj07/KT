#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int n, a[N];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        int sum = 0, k;
        for (int i = 1; i <= n; i++) cin >> a[i], sum += a[i];
        cin >> k;
        if (sum % n != k)
            puts("no");
        else
            puts("yes");
    }
}
