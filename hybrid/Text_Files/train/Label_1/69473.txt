import sys

input = sys.stdin.readline

n, m = map(int, input().split())
graph = [[] for _ in range(n + 1)]

for _ in range(m):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

count = 0

for i in range(1, n + 1):
    for j in range(i + 1, n + 1):
        if len(graph[i]) > 1:
            for k in graph[i]:
                if len(graph[k]) > 1:
                    for l in graph[k]:
                        if len(graph[l]) > 1:
                            if i != k and i != l and j != k and j != l:
                                if j in graph[l]:
                                    count += 1

print(count)
