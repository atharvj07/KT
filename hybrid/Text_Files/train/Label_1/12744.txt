#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2e5 + 5;

int n;
pair<int, int> shows[MAXN];

int main()
{
    cin >> n;
    for (int i = 0; i < n; i++)
    {
        cin >> shows[i].first >> shows[i].second;
    }

    sort(shows, shows + n);

    int j = 0;
    bool can = true;
    for (int i = 0; i < n; i++)
    {
        int l = shows[i].first;
        int r = shows[i].second;
        if (l >= shows[j].second)
        {
            j = i;
        }
        else
        {
            can = false;
            break;
        }
    }

    cout << (can ? "YES" : "NO") << endl;

    return 0;
}
