n = int(input())
colors_seen = [input().strip() for _ in range(n)]

color_to_gem = {
    'purple': 'Power',
    'green': 'Time',
    'blue': 'Space',
    'orange': 'Soul',
    'red': 'Reality',
    'yellow': 'Mind'
}

gems_absent = [color_to_gem[color] for color in color_to_gem if color not in colors_seen]

print(len(gems_absent))
for gem in gems_absent:
    print(gem)
