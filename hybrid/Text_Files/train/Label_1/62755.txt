import sys

n,x=map(int,sys.stdin.readline().split())
a=sorted(list(map(int,sys.stdin.readline().split())))
p=[-1,-1]
for v in a:
    if v<x:
        p[0]=v
        break
for v in a:
    if (v&x)!=v:
        p[1]=v
        break
if p[0]==-1 or p[1]==-1:
    print(-1)
else:
    print(a.count(p[1])==0)
