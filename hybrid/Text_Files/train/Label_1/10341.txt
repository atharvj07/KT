#include <iostream>
#include <cmath>
using namespace std;
int main()
{
    long long m, n;
    cin >> m >> n;
    double p = 1.0 / m;
    double sum = 0;
    for (long long i = 1; i <= m; ++i)
        sum += i * pow(p, i) * pow(1 - p, m - i);
    sum *= (1 - pow(1 - p, n)) / p;
    cout.precision(10);
    cout << sum << endl;
}
