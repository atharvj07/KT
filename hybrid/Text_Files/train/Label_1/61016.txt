import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    if sum(a) <= 1:
        print(-1)
        continue
    ans = [1]
    for i in range(n):
        if a[i] == 1:
            ans.append(n+1)
        else:
            ans.append(i+2)
    print(*ans)
