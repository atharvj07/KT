n, x = map(int, input().split())
ans = []
if x != 1:
    for i in range(n):
        if (x >> i) & 1 == 1:
            ans.append(2**i)
if ans:
    print(len(ans))
    print(*ans)
    exit(0)
for i in range(n):
    if (x >> i) & 1 == 0:
        ans.append(2**i)
if ans:
    print(len(ans))
    print(*ans)
    exit(0)
print(0)
