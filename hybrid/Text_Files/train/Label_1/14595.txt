#include <bits/stdc++.h>
using namespace std;
const int MX = 1e5+5;
int n, a[MX];

void solve() {
    cin >> n;
    for(int i=1; i<=n; i++) {
        cin >> a[i];
    }
    bool f = 0;
    for(int i=1; i<=n-1; i++) {
        if(a[i] > a[i+1]) f = 1;
    }
    if(f) cout << "NO" << endl;
    else cout << "YES" << endl;
}

int main() {
    int t; cin >> t;
    while(t--) {
        solve();
    }
    return 0;
}
