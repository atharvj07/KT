#include <iostream>
#include <unordered_map>

using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_map<string, int> m;
  while (n--) {
    string s;
    cin >> s;
    m[s]++;
  }
  cout << m["Tetrahedron"] * 4 + m["Cube"] * 6 + m["Octahedron"] * 8 +
            m["Dodecahedron"] * 12 + m["Icosahedron"] * 20;
  return 0;
}
