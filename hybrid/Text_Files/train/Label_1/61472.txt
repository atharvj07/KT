import sys

mod=int(1e9)+7
n,k,d=list(map(int,sys.stdin.readline().split()))
f=[[[-1]*k for _ in range(n+1)] for _ in range(d+1)]
def dp(i,rem,bef):
  if i==n:
    if rem==0:
      return 1
    return 0
  if f[i][rem][bef]!=-1:
    return f[i][rem][bef]
  ret=0
  for j in range(k):
    if j>=bef:
      ret=(ret+dp(i+1,rem-1-j,j))%mod
    else:
      ret=(ret+dp(i+1,rem-1,j))%mod
  f[i][rem][bef]=ret
  return ret
print(dp(0,n-1,d))
