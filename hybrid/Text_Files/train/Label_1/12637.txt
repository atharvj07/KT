#include <bits/stdc++.h>
using namespace std;
int a[110];
int main(){
    int n,ans=0;
    cin>>n;
    string s;
    cin>>s;
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    for(int i=0;i<n;i++){
        if(s[i]!=s[i+1]){
            ans=max(ans,a[i]);
        }
    }
    cout<<ans<<endl;
    return 0;
}
