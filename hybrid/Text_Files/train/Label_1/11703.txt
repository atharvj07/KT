#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k, d;
        cin >> n >> k >> d;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        map<int, bool> m;
        int ans = k;
        for (int i = 0; i < n; i++) {
            if (i < d - 1) m[a[i]] = true;
            else {
                m[a[i]] = true;
                if (m.size() >= d) {
                    ans = min(ans, d);
                    for (int j = i - (d - 1); j <= i; j++) m[a[j]] = false;
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
