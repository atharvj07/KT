def find_cycles(p):
    cycles = []
    visited = [False] * len(p)
    for i in range(len(p)):
        if not visited[i]:
            cycle_size = 0
            j = i
            while not visited[j]:
                visited[j] = True
                j = p[j] - 1
                cycle_size += 1
            cycles.append(cycle_size)
    return cycles

def max_convenience(n, p):
    cycles = find_cycles(p)
    cycles.sort(reverse=True)
    if len(cycles) > 1:
        cycles[0] += cycles[1]
    return sum(cycle * cycle for cycle in cycles)

n = int(input().strip())
p = list(map(int, input().strip().split()))
print(max_convenience(n, p))
