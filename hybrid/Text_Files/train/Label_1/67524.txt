import sys
input = sys.stdin.readline

n = int(input())
a = [0]+[int(i) for i in input().split()]

l = []
for i in range(1,n+1):
    x = a[i] - a[i-1]
    ok = True
    for j in range(i+1,n+1):
        if a[j] - a[j-i] != x:
            ok = False
            break
    if ok: l.append(i)
print(len(l))
print(*l)
