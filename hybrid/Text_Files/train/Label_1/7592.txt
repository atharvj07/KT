#include <iostream>
using namespace std;

const int MOD = 998244353;

long long pow2(long long a, long long b){
    long long ret = 1;
    while(b){
        if(b&1) ret = ret * a % MOD;
        a = a * a % MOD;
        b >>= 1;
    }
    return ret;
}

int main(){
    long long n; cin >> n;
    cout << ((pow2(n, 2) + n) % MOD * pow2(n + 1, MOD - 2)) % MOD << endl;
    return 0;
}
