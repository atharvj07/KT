import bisect

n,k = list(map(int, input().split()))
a = list(map(int, input().split()))
b = sorted(range(len(a)), key=lambda x:a[x])
for i in range(0, len(a)-k+1):
    l = b[i]
    r = b[i+k-1]
    if (r-l+1)&1:
        print((a[l+(r-l)//2]+a[(l+(r-l+1)//2)])//2)
    else:
        print(a[(l+(r-l)//2)])
