import sys

n=int(input())

if n%2==1: print("NO\nNO")
else:
    print("YES")
    print(2,1,n,n-1,3,end=' ')
    for i in range(4,n-1,2):
        print(i+1,i,end=' ')
    print(n,n-2)
    print("YES")
    print(2,1,n,3,n-1,end=' ')
    for i in range(4,n-1,2):
        print(i,i+1,end=' ')
    print(n-2,n)
    sys.exit(0)
