#include <bits/stdc++.h>

using namespace std;
using ll = long long;
using vi = vector<int>;
using vvi = vector<vi>;
using ii = pair<int,int>;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, d;
        cin >> n >> d;
        vi a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        int sum = 0;
        for (int i = 1; i < n; i++) {
            sum += min(d, a[i]);
            a[0] += min(d, a[i]);
            a[i] = 0;
        }
        cout << a[0] << endl;
    }
    return 0;
}
