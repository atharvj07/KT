t = int(input().strip())
for _ in range(t):
    n, m, k = map(int, input().split())
    max_jokers_per_player = min(m, n // k)
    remaining_jokers = max(0, m - max_jokers_per_player)
    max_jokers_next_player = remaining_jokers // (k - 1) + (1 if remaining_jokers % (k - 1) > 0 else 0)
    print(max_jokers_per_player - max_jokers_next_player)
