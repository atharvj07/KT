#include <iostream>
#include <vector>
using namespace std;

bool f(const string& s, int n, vector<int>& ans) {
    if (n == 1) return false;
    ans.push_back(s[0]);
    int last = s[0] - '0';
    for (int i = 1; i < n; i++) {
        int cur = s[i] - '0';
        if (cur <= last) return false;
        last = cur;
        ans.push_back(s[i]);
    }
    return true;
}

int main() {
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        vector<int> ans;
        bool ok = f(s, n, ans);
        if (!ok) {
            cout << "NO" << endl;
            continue;
        }
        cout << "YES" << endl;
        int m = ans.size();
        cout << m << endl;
        bool first = true;
        for (int i = 0; i < m; i++) {
            if (!first) cout << " ";
            first = false;
            cout << ans[i];
        }
        cout << endl;
    }
    return 0;
}
