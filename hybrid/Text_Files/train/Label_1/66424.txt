def can_transform(n, m, A, B):
    for i in range(n-1):
        for j in range(m-1):
            if A[i][j] != B[i][j]:
                A[i][j] ^= 1
                A[i][j+1] ^= 1
                A[i+1][j] ^= 1
                A[i+1][j+1] ^= 1
    if A == B:
        return "Yes"
    else:
        return "No"

n, m = map(int, input().split())
A = [list(map(int, input().split())) for _ in range(n)]
B = [list(map(int, input().split())) for _ in range(n)]

print(can_transform(n, m, A, B))
