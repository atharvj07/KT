int n, m, b[55][55];

bool isConvex(int x, int y, int type) {
    if(x < 0 || x >= n || y < 0 || y >= m || b[x][y]==0 || (x == n - 1 && y == m - 1)) return true;
    b[x][y] = 0;
    bool f = true;

        int nx = x + dx[dir];
        int ny = y + dy[dir];

            f &= isConvex(nx, ny, 1);
        } else {

                f &= isConvex(nx, ny, type);
            }
        }
    }
    return f;
}
