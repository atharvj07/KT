def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

t = int(input())
for _ in range(t):
    a, b = map(int, input().split())
    if a == b:
        print("0 0")
    else:
        max_excitement = gcd(a, b)
        moves = min(abs(a - b) - a % abs(a - b), a % abs(a - b))
        print(max_excitement, moves)
