#include <iostream>
#include <vector>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int k;
  cin >> k;
  vector<char> ans;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '?' || s[i] == '*') {
      if (ans.empty()) {
        cout << "Impossible" << endl;
        return 0;
      }
      if (s[i] == '?') {
        ans.pop_back();
      }
    } else {
      ans.push_back(s[i]);
    }
  }
  if (ans.size() == k) {
    for (int i = 0; i < k; i++) {
      cout << ans[i];
    }
    cout << endl;
  } else {
    cout << "Impossible" << endl;
  }
  return 0;
}
