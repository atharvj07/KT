#include<bits/stdc++.h>
#define int long long
using namespace std;
int32_t main(){
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);
	int n,k,mn=1e18;
	cin>>n>>k;
	string s;
	cin>>s;
	map<string,int> mp;
	vector<int> dp(n+1,-1);dp[0]=0;
	for(int i=1;i<=n;i++){
		string x="";
		for(int j=i;j<=n;j++){
			x+=s[j-1];
			if(!mp[x]){
				mp[x]=1;
				if(mp.size()<k) dp[j]=i-1;
				else{
					int pre=1e18;
					for(int p=i;p>=1;p--){
						if(dp[p-1]!=-1){
							pre=p-1;
							break;
						}
					}
					dp[j]=dp[pre]+(i-1-pre);
					mn=min(mn,dp[j]);
				}
			}
		}
	}
	if(mn==1e18) cout<<-1;
	else cout<<mn;
	return 0;
}
