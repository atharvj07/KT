import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class EdgeAddition {

    static class Edge {
        int from;
        int to;

        public Edge(int from, int to) {
            this.from = from;
            this.to = to;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            edges.add(new Edge(a, b));
        }

        int maxEdges = n - 1; // Initial maximum edges is n-1 for a tree

        // Check for connected components
        int[] visited = new int[n + 1];
        int connectedComponents = 0;
        for (int i = 1; i <= n; i++) {
            if (visited[i] == 0) {
                connectedComponents++;
                dfs(i, visited, edges);
            }
        }

        // Update maxEdges based on connected components
        maxEdges += connectedComponents - 1;

        System.out.println(maxEdges);
    }

    static void dfs(int node, int[] visited, List<Edge> edges) {
        visited[node] = 1;
        for (Edge edge : edges) {
            if (edge.from == node && visited[edge.to] == 0) {
                dfs(edge.to, visited, edges);
            } else if (edge.to == node && visited[edge.from] == 0) {
                dfs(edge.from, visited, edges);
            }
        }
    }
}
