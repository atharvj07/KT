import java.util.*;

public class Main {
    static boolean[][] vis;
    static int[][] dis;
    static int[][] dir = {{-1, 0}, {0, -1}, {0, 1}, {1, 0}};

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        vis = new boolean[n + 1][m + 1];
        dis = new int[n + 1][m + 1];
        int k = in.nextInt();
        Queue<Integer> qx = new LinkedList<>();
        Queue<Integer> qy = new LinkedList<>();
        for (int i = 0; i < k; i++) {
            int x = in.nextInt();
            int y = in.nextInt();
            vis[x][y] = true;
            dis[x][y] = 0;
            qx.add(x);
            qy.add(y);
        }
        int ansx = -1;
        int ansy = -1;
        int ansdis = -1;
        while (!qx.isEmpty()) {
            int x = qx.poll();
            int y = qy.poll();
            if (dis[x][y] > ansdis) {
                ansdis = dis[x][y];
                ansx = x;
                ansy = y;
            }
            for (int[] d : dir) {
                int nx = x + d[0];
                int ny = y + d[1];
                if (nx >= 1 && nx <= n && ny >= 1 && ny <= m && !vis[nx][ny]) {
                    vis[nx][ny] = true;
                    dis[nx][ny] = dis[x][y] + 1;
                    qx.add(nx);
                    qy.add(ny);
                }
            }
        }
        System.out.println(ansx + " " + ansy);
    }
}
