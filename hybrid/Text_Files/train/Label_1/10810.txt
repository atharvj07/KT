#include <bits/stdc++.h>
using namespace std;
const int mxN = 2e5;
int psum[mxN + 1];
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k, m;
        cin >> n >> k >> m;
        vector<int> b(m);
        for (auto &x : b) cin >> x;
        psum[0] = 0;
        for (int i = 1; i <= n; i++) psum[i] = psum[i - 1] + i;
        int sumB = psum[b[m - 1]] - psum[b[0] - 1];
        int mn = (k + 1) / 2, mx = n - k / 2, idx = b[m / 2] - 1;
        if (mn <= idx && idx <= mx && psum[idx] - psum[idx - mn] == sumB) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
    return 0;
}
