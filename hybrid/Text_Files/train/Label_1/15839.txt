#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<vector<int>> grid(n, vector<int>(m, 0));
  while (k--) {
    int x, y;
    cin >> x >> y;
    --x;
    --y;
    ++grid[x][y];
    if (x + 1 < n && y + 1 < m && grid[x][y] && grid[x + 1][y] && grid[x][y + 1] && grid[x + 1][y + 1]) {
      cout << k + 1 << endl;
      return 0;
    }
  }
  cout << 0 << endl;
}
