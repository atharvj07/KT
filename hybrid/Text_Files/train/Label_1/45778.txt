import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int best = scanner.nextInt();
    int worst = best;
    int count = 0;
    while (--n > 0) {
      int score = scanner.nextInt();
      if (score > best) {
        best = score;
        count++;
      } else if (score < worst) {
        worst = score;
        count++;
      }
    }
    System.out.println(count);
  }
}
