for _ in range(int(input())):
    n,m=map(int,input().split())
    a=[list(map(int,input().split())) for _ in range(n)]
    x=0
    for i in range(1,n):
        for j in range(1,m):
            if a[i][j]!=a[i-1][j]!=a[i][j-1]:
                x+=1
            elif a[i][j]!=a[i-1][j]:
                x+=1
            elif a[i][j]!=a[i][j-1]:
                x+=1
    print(x)
