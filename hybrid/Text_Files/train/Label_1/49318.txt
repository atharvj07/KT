import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int w = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }

        int[] score = new int[n];
        for (int i = 0; i < n; i++) {
            int l = Math.max(0, i - w);
            int r = Math.min(n - 1, i + w);
            int cnt = 0;
            for (int j = l; j <= r; j++) {
                if (a[j] > 0) cnt++;
            }
            if (cnt % 2 == 1) {
                score[i] += b[i];
            } else {
                score[i] += a[i];
            }
        }

        int maxScore = 0;
        int maxScoreIdx = -1;
        for (int i = 0; i < n; i++) {
            if (score[i] > maxScore) {
                maxScore = score[i];
                maxScoreIdx = i;
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (i == maxScoreIdx) {
                sb.append('1');
            } else {
                sb.append('0');
            }
        }

        System.out.println(sb);
    }
}
