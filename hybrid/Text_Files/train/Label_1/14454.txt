#include <bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    string s; cin >> s;
    vector<int> a(n);
    for (auto &v : a) cin >> v;
    int cur = 0, mx = 0, x = 0;
    for (int i = 0; i < n; i++) {
        if (i < n - 1 && s[i] == s[i + 1])
            x++;
        else
            x = 0;
        cur += a[x];
        mx = max(mx, cur);
    }
    cout << mx << endl;
}
