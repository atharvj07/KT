from sys import stdin
input = stdin.readline

for _ in range(int(input())):
    n,k = [int(x) for x in input().split()]
    a = [int(x) for x in input().split()]
    b = [0]*n
    for i in range(1,n):
        if a[i]>a[i-1]:
            b[i] = b[i-1] + 1
    c = []
    for i in range(n-k+1):
        if i==0:
            x = b[k-1]
        else:
            x = b[i+k-1] - b[i-1]
        c.append(x)
    print(max(c)+1,c.index(max(c))+1)
