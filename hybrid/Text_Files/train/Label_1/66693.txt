n=int(input())
a=list(map(int,input().split()))
# sort array
a.sort()
# maximum beauty of submatrix of size 2**n x 2**n
def solve(l,r,size):
    if size==1:return a[r]
    # maximum beauty submatrix size 2**(n-1) x 2**(n-1)
    max1=solve(l,l+size//2-1,size//2)
    # maximum beauty submatrix size 2**(n-1) x 2**(n-1)
    max2=solve(l+size//2,r,size//2)
    mx=max(a[l+size//2-1],a[r])
    return mx+max1+max2
print(solve(0,len(a)-1,1<<n))
