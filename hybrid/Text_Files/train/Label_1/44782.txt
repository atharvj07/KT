import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.LinkedList;
import java.util.Queue;

public class Jaber {

    static class City {
        int row;
        int col;
        int color;

        City(int row, int col, int color) {
            this.row = row;
            this.col = col;
            this.color = color;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        int k = Integer.parseInt(line[2]);

        int[][] grid = new int[n][m];
        for (int i = 0; i < n; i++) {
            line = br.readLine().split(" ");
            for (int j = 0; j < m; j++) {
                grid[i][j] = Integer.parseInt(line[j]);
            }
        }

        int q = Integer.parseInt(br.readLine());
        for (int i = 0; i < q; i++) {
            line = br.readLine().split(" ");
            int r1 = Integer.parseInt(line[0]) - 1;
            int c1 = Integer.parseInt(line[1]) - 1;
            int r2 = Integer.parseInt(line[2]) - 1;
            int c2 = Integer.parseInt(line[3]) - 1;

            System.out.println(bfs(grid, r1, c1, r2, c2));
        }
    }

    static int bfs(int[][] grid, int r1, int c1, int r2, int c2) {
        int n = grid.length;
        int m = grid[0].length;
        boolean[][] visited = new boolean[n][m];
        Queue<City> queue = new LinkedList<>();
        queue.add(new City(r1, c1, grid[r1][c1]));
        visited[r1][c1] = true;

        int steps = 0;
        while (!queue.isEmpty()) {
            int size = queue.size();
            for (int i = 0; i < size; i++) {
                City current = queue.poll();
                if (current.row == r2 && current.col == c2) {
                    return steps;
                }

                int[] dx = {-1, 1, 0, 0};
                int[] dy = {0, 0, -1, 1};
                for (int j = 0; j < 4; j++) {
                    int newRow = current.row + dx[j];
                    int newCol = current.col + dy[j];
                    if (newRow >= 0 && newRow < n && newCol >= 0 && newCol < m && !visited[newRow][newCol] && (grid[newRow][newCol] == grid[current.row][current.col] || grid[newRow][newCol] == grid[r2][c2])) {
                        queue.add(new City(newRow, newCol, grid[newRow][newCol]));
                        visited[newRow][newCol] = true;
                    }
                }
            }
            steps++;
        }

        return -1;
    }
}
