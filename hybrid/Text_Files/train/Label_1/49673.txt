import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt(), n = sc.nextInt();
        Map<Character, Integer> hash = new HashMap<>();
        int[] freq = new int[k];
        char[] s = new char[n];

        for (int i = 0; i < k; i++) {
            char[] t = sc.next().toCharArray();
            for (char c : t) {
                hash.merge(c, 1, Integer::sum);
            }
            for (int j = 0; j < n; j++) {
                s[j] ^= t[j];
                if (s[j] != '\0') freq[i]++;
            }
        }

        if ((n & 1) == 0 && (hash.size() & 1) == 1) {
            System.out.println("-1");
            return;
        }
        for (char c : hash.keySet()) {
            if (hash.get(c) % 2 == 1) {
                if (n % 2 == 1) n--;
                for (int i = 0; i < n; i += 2) s[i] = c;
                break;
            }
        }

        int idx = 0, count = 0;
        for (char c : hash.keySet()) {
            if (count == k - 1) {
                for (int i = idx; i < n; i += 2) s[i] = c;
                break;
            }
            if (hash.get(c) % k != 0) {
                idx = 0;
                count = 0;
            }
            for (int i = idx; i < n; i += 2) {
                if (freq[count] == k) continue;
                if (s[i] == 0) s[i] = c;
                idx = i + 1;
                count++;
            }
        }
        System.out.println(s);
    }
}
