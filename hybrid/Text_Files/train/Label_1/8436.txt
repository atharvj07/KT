#include <iostream>
#include <vector>
using namespace std;

int main() {
    int w, l;
    cin >> w >> l;
    vector<int> a(w - 1);
    for (int i = 0; i < w - 1; i++) { cin >> a[i]; }
    int ans = 0, can = 0;
    for (int i = 0; i < w - l; i++) {
        ans += can;
        can = 0;
        for (int j = 1; j <= l; j++) { can += a[i + j - 1]; }
        can = min(can, 1);
    }
    cout << ans << endl;
    return 0;
}
