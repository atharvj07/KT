#include <iostream>
#include <string>
#include <vector>

int main() {
  std::string s;
  std::cin >> s;
  std::vector<bool> vis(s.size(), false);
  for (int i = 0; i < (int)s.size(); i++) {
    for (int j = i + 1; j < (int)s.size(); j++) {
      if (s[i] == s[j]) {
        vis[i] = vis[j] = true;
      }
    }
  }
  bool ok = true;
  for (int i = 0; i < 10; i++) {
    if (s[i] != "CODEFORCES"[i]) {
      ok = ok && vis[i];
    }
  }
  std::cout << ((ok) ? "YES" : "NO") << "\n";
  return 0;
}
