N, D = map(int, input().split())
S = list(map(int, input().split()))
P = list(map(int, input().split()))

# Calculate the new score for the chosen astronaut
new_score = S[D - 1] + P[0]

# Count how many astronauts will have a score higher than the new score of the chosen astronaut
better_rank = sum(1 for score in S if score + P[-1] > new_score)

# The best possible ranking is one place better than the number of astronauts with a better score
print(better_rank + 1)
