import sys

def dist(x,y):
    return (x*x+y*y)**0.5

ax,ay,bx,by,tx,ty = map(int, input().split())
n = int(input())
ans = 1e9
for i in range (n):
    x,y = map(int, input().split())
    ans = min(ans, dist(x-ax,y-ay)+dist(x-bx,y-by)+dist(x-tx,y-ty))
    ans = min(ans, dist(x-ax,y-by)+dist(x-bx,y-ay)+dist(x-tx,y-ty))
    ans = min(ans, dist(x-bx,y-ax)+dist(x-ax,y-by)+dist(x-tx,y-ty))
    ans = min(ans, dist(x-bx,y-ay)+dist(x-ax,y-bx)+dist(x-tx,y-ty))
print(ans)
