#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
#define endl "\n"
ll big_mod(ll a,ll n){ll res=1;while(n){if(n%2)res=((res%mod)*(a%mod))%mod;a=((a%mod)*(a%mod))%mod;n>>=1;}return res%mod;}
ll mod_inv(ll base, ll p){return big_mod(base, p - 2);}

int main()
{
	int n;
	cin >> n;
	int arr[n];
	for(int i = 0; i < n; i++)
	{
		cin >> arr[i];
	}
	sort(arr, arr + n);
	cout << arr[n - 1] << endl;
}
