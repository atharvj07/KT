import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));

        int t = Integer.parseInt(in.readLine());
        while (t-- > 0) {
            int n = Integer.parseInt(in.readLine());
            int[][] a = new int[n + 1][n + 1];
            int[] min = new int[n + 1];
            int[] max = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                min[i] = i;
                max[i] = i;
            }

            for (int i = 1; i < n; i++) {
                int l = 1;
                int r = n;
                int indexl = -1;
                int indexr = -1;
                while (l <= r) {
                    int m = (l + r) / 2;
                    out.println("? " + min[m] + " " + min[i]);
                    out.flush();
                    String s = in.readLine();
                    if (s.equals("<")) {
                        r = m - 1;
                        indexr = m;
                    } else if (s.equals("=")) {
                        indexl = indexr = m;
                        break;
                    } else {
                        l = m + 1;
                        indexl = m;
                    }
                }
                if (indexl == -1) {
                    indexl = l;
                }
                if (indexr == -1) {
                    indexr = r + 1;
                }
                min[i] = min[indexl];
                min[indexl] = min[i];

                l = 1;
                r = n;
                indexl = -1;
                indexr = -1;
                while (l <= r) {
                    int m = (l + r) / 2;
                    out.println("? " + max[m] + " " + max[i]);
                    out.flush();
                    String s = in.readLine();
                    if (s.equals("<")) {
                        indexl = m;
                    } else if (s.equals("=")) {
                        indexl = indexr = m;
                        break;
                    } else {
                        r = m - 1;
                        indexr = m;
                    }
                }
                if (indexl == -1) {
                    indexl = l;
                }
                if (indexr == -1) {
                    indexr = r + 1;
                }
                max[i] = max[indexl];
                max[indexl] = max[i];
            }

            out.println("! " + min[1] + " " + max[1]);
            out.flush();
        }
        out.close();
    }
}
