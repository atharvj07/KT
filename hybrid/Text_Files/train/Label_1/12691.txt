#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
#define rep(i,n) for(int i=0;i<n;i++)
#define rep1(i,n) for(int i=1;i<=n;i++)

const int N=2e5+5;
vector<int> g[N];
int n,m,k;
int dis[N];

void bfs(int s){
    queue<int> q;
    q.push(s);
    dis[s]=0;

    while(!q.empty()){
        int u=q.front();
        q.pop();

        for(auto v:g[u]){
            if(dis[v]!=-1) continue;
            dis[v]=dis[u]+1;
            q.push(v);
        }
    }
}

int main(){
    cin>>n>>m>>k;
    vector<int> a(k+1);
    rep1(i,k) cin>>a[i];

    rep(i,m){
        int u,v;
        cin>>u>>v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    int mx=0;

    rep1(i,k){
        memset(dis,-1,sizeof dis);
        bfs(a[i]);
        mx=max(mx,dis[n]);
    }
    
    cout<<mx<<endl;

    return 0;
}
