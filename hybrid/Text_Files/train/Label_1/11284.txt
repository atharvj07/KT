#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct Item {
    int t, d, p;
};

bool cmp(const Item& a, const Item& b) {
    return a.d > b.d;
}

int main() {
    int n;
    cin >> n;
    vector<Item> items(n);
    for (int i = 0; i < n; i++) {
        cin >> items[i].t >> items[i].d >> items[i].p;
    }
    sort(items.begin(), items.end(), cmp);
    int dp[n + 1][2001];
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= 2000; j++) {
            dp[i][j] = 0;
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 2000; j >= items[i].t; j--) {
            dp[i + 1][j] = max(dp[i + 1][j], dp[i][j - items[i].t] + items[i].p);
        }
    }
    int max_value = 0;
    int max_idx = -1;
    for (int j = 0; j <= 2000; j++) {
        if (dp[n][j] > max_value) {
            max_value = dp[n][j];
            max_idx = j;
        }
    }
    cout << max_value << endl;
    int idx = n;
    vector<int> ans;
    while (idx > 0) {
        for (int i = idx - 1; i >= 0; i--) {
            if (dp[i][max_idx] + items[i].p == dp[idx][max_idx]) {
                ans.push_back(i + 1);
                max_idx -= items[i].t;
                idx = i;
                break;
            }
        }
    }
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
