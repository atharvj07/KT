import math

x1, y1, x2, y2 = map(int, input().split())
vmax, t = map(int, input().split())
v1x, v1y = map(int, input().split())
v2x, v2y = map(int, input().split())

v1 = math.sqrt(v1x**2 + v1y**2)
v2 = math.sqrt(v2x**2 + v2y**2)

if v1 > vmax or v2 > vmax:
    print(-1.)
    exit()

num = (x2 - x1)*v1y - (y2 - y1)*v1x
den = (v1x*v2y - v1y*v2x)

if den == 0:
    print(-1.)
    exit()

t1 = num/den

if t1 < 0 or t1 > t:
    print(-1.)
    exit()

x3 = (x1 + v1x*t1)
y3 = (y1 + v1y*t1)

dist = math.sqrt((x2 - x3)**2 + (y2 - y3)**2) / v2

time = t1 + (dist/v2)

print(time)
