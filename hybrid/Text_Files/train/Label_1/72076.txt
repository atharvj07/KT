import sys

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

if a[0] == 0:
    print(-1)
    sys.exit()

jump = [n] * (n + 1)
jump[0] = 0
for i in range(0, n):
    for j in range(i, n):
        x = j - i
        if x <= a[i]:
            jump[j] = min(jump[j], jump[i] + 1)
        else:
            break

if jump[n] == n:
    print(-1)
else:
    print(jump[n])
    k = n
    res = []
    while k > 0:
        for i in range(k - 1, -1, -1):
            x = k - i
            if x <= a[i] and jump[k] == jump[i] + 1:
                res.append(k - b[i])
                k = i
                break
    res.reverse()
    print(*res)
