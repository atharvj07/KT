#include <iostream>
#include <string>

int main() {
    std::string s;
    std::cin >> s;
    int ans = 0;
    for (int i = 0; i < s.length(); ++i) {
        if (i + 5 <= s.length() && s.substr(i, 5) == "heavy" && (i == 0 || s[i - 1] == ' ') && (i + 5 == s.length() || s[i + 5] == ' ')) {
            ++ans;
        }
        if (s[i] == 'm' && (i == 0 || s[i - 1] == ' ') && (i + 5 <= s.length() && s.substr(i + 1, 5) == "etal" || i + 6 <= s.length() && s.substr(i + 1, 6) == "etalis")) {
            ++ans;
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
