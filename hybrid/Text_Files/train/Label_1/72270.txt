s = input()
g, k = input(), int(input())
n = len(s)
pref = [[0] * 26 for _ in range(n + 1)]
for i in range(n - 1, -1, -1):
    for j in range(26):
        pref[i][j] = pref[i + 1][j]
    pref[i][ord(s[i]) - ord('a')] += 1
suf = [[0] * 26 for _ in range(n + 1)]
for i in range(n):
    for j in range(26):
        suf[i + 1][j] = suf[i][j]
    suf[i][ord(s[i]) - ord('a')] += 1
ans = 0
for i in range(n):
    for j in range(i, n):
        bad = 0
        for l in range(26):
            bad += pref[i][l] + suf[j + 1][l] - pref[j + 1][l] - suf[i][l]
            if bad > k:
                break
        else:
            ans += 1
print(ans)
