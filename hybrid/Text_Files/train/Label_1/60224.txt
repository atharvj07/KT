from collections import defaultdict
from queue import Queue
for _ in range(int(input())):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    q = Queue()
    vis = defaultdict(bool)
    vis[x] = True
    q.put(x)
    if x in a:
        ans = 0
    else:
        ans = 1
    while not q.empty():
        cur = q.get()
        for i in a:
            if i != cur and not vis[i]:
                ans += 1
                vis[i] = True
                q.put(i)
    print(ans)
