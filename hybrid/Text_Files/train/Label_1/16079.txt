#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5+5;
int main(){
    int n, l, a, t[maxn], x[maxn];
    cin >> n >> l >> a;
    for(int i=0;i<n;++i) cin >> t[i] >> x[i];
    int sum = 0, ans = 0;
    for(int i=0;i<n;++i){
        sum += x[i];
        if(sum > l) break;
        if(i+1==n){
            ans += (l-sum+a-1)/a;
            break;
        }
        if(t[i+1]-sum-a>=0){
            ans++;
            sum+=a;
        }
    }
    cout << ans << endl;
}
