#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    vector<int>v(n);
    for(int i=0;i<n;i++)
    cin>>v[i];
    vector<int>::iterator it1,it2;
    it1 = max_element(v.begin(),v.end());
    it2 = min_element(v.begin(),v.end());
    int pos1=it1-v.begin(),pos2=it2-v.begin();
    if(pos1==pos2)
    {cout<<"YES"; return 0;}
    else
    {
        if(pos1<pos2)
        {
            for(int i=0;i<pos1;i++)
            {
                if(v[i]>v[i+1])
                {cout<<"NO"; return 0;}
            }
            for(int i=pos1;i<pos2;i++)
            {
                if(v[i]!=v[i+1])
                {cout<<"NO"; return 0;}
            }
            for(int i=pos2;i<n-1;i++)
            {
                if(v[i]>v[i+1])
                {cout<<"NO"; return 0;}
            }
        }
        else
        {
            for(int i=0;i<pos2;i++)
            {
                if(v[i]>v[i+1])
                {cout<<"NO"; return 0;}
            }
            for(int i=pos2;i<pos1;i++)
            {
                if(v[i]!=v[i+1])
                {cout<<"NO"; return 0;}
            }
            for(int i=pos1;i<n-1;i++)
            {
                if(v[i]>v[i+1])
                {cout<<"NO"; return 0;}
            }
        }
    }
    cout<<"YES";
    return 0;
}
