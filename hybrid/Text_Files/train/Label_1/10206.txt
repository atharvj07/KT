#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 5;
int N, D, S[MAXN], P[MAXN];
pair<int, int> pts[MAXN];

int solve() {
    for (int i = 0; i < N; i++) {
        pts[i] = {S[i] + P[i], i + 1};
    }
    sort(pts, pts + N);
    int cnt = 1;
    for (int i = 0; i < N; i++) {
        if (pts[i].second != D) cnt++;
        else break;
    }
    return cnt;
}

int main() {
    cin >> N >> D;
    for (int i = 0; i < N; i++) cin >> S[i];
    for (int i = 0; i < N; i++) cin >> P[i];
    cout << solve() << "\n";
    return 0;
}
