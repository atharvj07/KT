#include<bits/stdc++.h>
using namespace std;
const int MAXN = 1005;
int n, s[MAXN], d[MAXN];
int main() {
    cin >> n;
    for(int i = 1; i <= n; i++) {
        cin >> s[i] >> d[i];
    }
    int ans = 0;
    for(int i = 1; i <= n; i++) {
        ans = max(ans, (s[i] - 1 + d[i] - 1) / d[i] * d[i] + s[i]);
    }
    cout << ans << endl;
    return 0;
}
