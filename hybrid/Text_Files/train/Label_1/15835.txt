#include <iostream>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  bool grid[n + 1][m + 1] = {false};
  int moves[4] = {0, 1, 0, -1};
  int move_num = 0;
  for (int i = 0; i < k; i++) {
    move_num++;
    int x, y;
    cin >> x >> y;
    grid[x][y] = true;
    for (int j = 0; j < 4; j++) {
      int nx = x + moves[j];
      int ny = y + moves[j + 1];
      if (nx >= 1 && nx <= n && ny >= 1 && ny <= m && grid[nx][ny] && grid[nx][ny + moves[j + 1]] && grid[nx + moves[j]][ny] &&
          grid[nx + moves[j]][ny + moves[j + 1]]) {
        cout << move_num << endl;
        return 0;
      }
    }
  }
  cout << 0 << endl;
  return 0;
}
