#include <iostream>
using namespace std;
int main() {
    int n;
    cin >> n;
    int arr[100][100] = {0};
    while (n--) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        for (int i = y1 - 1; i < y2; i++) {
            for (int j = x1 - 1; j < x2; j++) { arr[i][j]++; }
        }
    }
    int ans = 0;
    for (int i = 0; i < 100; i++) {
        for (int j = 0; j < 100; j++) { ans += arr[i][j]; }
    }
    cout << ans << endl;
}
