from queue import Queue

n = int(input())
p = [int(i) for i in input().split()]
a = [[int(i) for i in input()] for _ in range(n)]
vis = [0] * (n + 1)

ans = []
q = Queue()
q.put([1])
while not q.empty():
    p = q.get()
    ans = p.copy()
    for i in p:
        for j in range(n):
            if not vis[j + 1] and a[i - 1][j] and p[-1] < j + 1:
                next = p.copy()
                next.append(j + 1)
                q.put(next)
                vis[j + 1] = 1

print(*ans)
