import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.next();
        int n = s.length();
        int open = 0;
        int need = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '(') {
                open++;
            } else if (s.charAt(i) == ')') {
                open--;
            } else {
                need++;
            }
        }
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '#') {
                if (need == 0 || open < 0) {
                    System.out.println(-1);
                    return;
                }
                open++;
                need--;
                System.out.println(1);
            } else if (s.charAt(i) == '(') {
                open++;
            } else {
                open--;
            }
        }
        if (open != 0 || need != 0) {
            System.out.println(-1);
        }
    }
}
