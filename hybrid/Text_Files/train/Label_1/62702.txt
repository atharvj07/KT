n, m = map(int, input().split())
food_packages = list(map(int, input().split()))

food_counts = {}
for food in food_packages:
    if food in food_counts:
        food_counts[food] += 1
    else:
        food_counts[food] = 1

days = float('inf')
for count in food_counts.values():
    days = min(days, count // n)

if days == float('inf'):
    print(0)
else:
    print(days)
