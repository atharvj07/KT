#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 1024, mod = 1e9 + 7;

int n, k;

bool solve() {
    if((1LL * n * (n - 1) / 2) % k) { return 0; }
    for(int i = 1; i <= n; ++i)
        for(int j = 1; j <= n; ++j)
            if(i != j and --k != 0)
                printf("%d %d\n", i, j);
    return 1;
}

int main() {
    cin >> n >> k;
    if(!solve())
        puts("-1");
    return 0;
}
