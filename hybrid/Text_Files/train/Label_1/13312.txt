#include <iostream>
#include <vector>

using namespace std;

vector<int> perfect;

int main() {
  int k;
  cin >> k;
  int sum = 0;
  for (int n = 1; perfect.size() < k; ++n) {
    sum = 0;
    int t = n;
    while (t > 0) {
      sum += t % 10;
      t /= 10;
    }
    if (sum == 10) perfect.push_back(n);
  }
  cout << perfect[k - 1];
}
