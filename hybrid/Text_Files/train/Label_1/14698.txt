#include <iostream>
#include <cstring>
#include <vector>

using namespace std;

typedef long long ll;

const ll maxn = 1005;
ll n;
char s[maxn];
vector<ll> ans;

void solve() {
  ll l = 0, r = 0;
  for (ll i = 0; i < n; i++) {
    if (s[i] == '(') l++;
    else r++;
    if (r > l) {
      ans.push_back(i + 1);
      l = 0, r = 0;
    }
  }
  if (r != 0) ans.push_back(n);
}

int main() {
  cin >> s;
  n = strlen(s);
  solve();
  cout << ans.size() << endl;
  for (ll i = 0; i < ans.size(); i += 2) {
    cout << ans[i] - ans[i + 1] + 1 << endl;
    for (ll j = ans[i + 1]; j <= ans[i]; j++) cout << j << " ";
    cout << endl;
  }
  return 0;
}
