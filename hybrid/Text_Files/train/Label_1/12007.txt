#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define mp make_pair
#define pb push_back
#define ll long long
const int inf = 1e9 + 7;
int n, m;
vector<int> a, t;
priority_queue<pair<int, int>> pq;
int main() {
    cin >> n >> m;
    a.resize(m + 1);
    t.resize(n + 1);
    for (int i = 1; i <= n; i++) cin >> t[i];
    for (int i = 1; i <= m; i++) { cin >> a[i]; pq.push(mp(-a[i], i)); }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (t[i]) {
            int res = -pq.top().first, j = pq.top().second;
            pq.pop();
            res += a[j];
            if (pq.empty() || res > i) return cout << -1 << endl, 0;
            pq.push(mp(-(res + a[pq.top().second]), pq.top().second));
        }
    }
    cout << ans << endl;
    return 0;
}
