import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[in.nextInt()] = i;
        }
        int[] p = new int[n + 1];
        int[] s = new int[n + 1];
        int[] t = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = a[i] - i;
        }
        System.out.println((n - 1) / 2);
        for (int i = 1; i <= n; i++) {
            if (p[i] == 0) {
                continue;
            }
            if (p[i] > 0) {
                s[p[i]]++;
            } else {
                t[-p[i]]++;
            }
        }
        for (int i = 1; i <= n; i++) {
            if (t[i] > 0) {
                System.out.println(2 + " " + s[i] + " " + i + " " + t[i]);
                for (int j = 1; j <= s[i]; j++) {
                    p[a[j]] = 0;
                }
                for (int j = 1; j <= t[i]; j++) {
                    p[a[n - j + 1]] = 0;
                }
            }
        }
    }
}
