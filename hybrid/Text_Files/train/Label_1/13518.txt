#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll mod = 998244353;
const ll N = 3e5 + 5;
vector<int>adj[N];
vector<bool>vis(N, 0);
vector<int>p(N, 0);
int d[N];
ll dfs(int u) {
  vis[u] = 1;
  ll ans = 1;
  for (auto v : adj[u]) {
    if (!vis[v])ans += dfs(v),ans %= mod;
  }
  return d[u] = ans;
}
void solve(){
    int n, m;
    cin >> n >> m;
    for (int i = 1;i <= n;i++) {
      int x;cin >> x;
      p[x] = i;
    }
    for (int i = 1;i <= m;i++) {
      int a, b;
      cin >> a >> b;
      adj[p[a]].push_back(p[b]);
      adj[p[b]].push_back(p[a]);
    }
    ll ans = 0;
    for (int i = 1;i <= n;i++)
      if (!vis[i])ans += dfs(i), ans %= mod;
    cout << ans << "\n";
}
int main() {
  solve();
  return 0;
}
