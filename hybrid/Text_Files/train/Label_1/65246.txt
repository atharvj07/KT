def count_ways(cards):
    n = len(cards)
    ways = []
    for a in range(1, n + 1):
        if n % a == 0:
            b = n // a
            for col in range(b):
                if all(cards[row * b + col] == 'X' for row in range(a)):
                    ways.append(f"{a}x{b}")
                    break
    return ways

t = int(input())
for _ in range(t):
    cards = input().strip()
    ways = count_ways(cards)
    print(len(ways), " ".join(ways))
