#include<bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;
const int maxn = 2e5 + 5;
int32_t main() {
    long long n, k;
    cin >> n >> k;
    long long ans = 0;
    for(int i = 0; i < n; ++i) {
        long long a;
        cin >> a;
        ans = (ans + (abs(a) <= k && a % 2)) % mod;
    }
    cout << ans << '\n';
    return 0;
}
