#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int t, m, c;
int main() {
  cin >> t >> m;
  vector<vector<int>> v(t, vector<int>(m, 0));
  for (int i = 0; i < t; ++i) {
    for (int j = 0; j < m; ++j) {
      cin >> v[i][j];
      if (v[i][j] && i > 0 && i < t - 1 && j > 0 && j < m - 1) c++;
    }
  }
  cout << (c + 3) / 4 << '\n';
  return 0;
}
