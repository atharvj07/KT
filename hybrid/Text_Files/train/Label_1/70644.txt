n, m = map(int, input().split())
memory = [input().strip() for _ in range(n)]
memory_set = set(memory)

for _ in range(m):
    query = input().strip()
    found = False
    for i in range(len(query)):
        for c in 'abc':
            if c != query[i]:
                possible_match = query[:i] + c + query[i+1:]
                if possible_match in memory_set:
                    print("YES")
                    found = True
                    break
        if found:
            break
    if not found:
        print("NO")
