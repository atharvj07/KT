import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] queries = new int[q][2];
        for (int i = 0; i < q; i++) {
            queries[i][0] = sc.nextInt();
            queries[i][1] = sc.nextInt();
        }
        for (int[] query : queries) {
            int l = query[0];
            int r = query[1];
            int count = 0;
            for (int i = 0; i < n; i++) {
                if (l <= a[i] && a[i] <= r) {
                    count++;
                }
            }
            System.out.println(count);
        }
    }
}
