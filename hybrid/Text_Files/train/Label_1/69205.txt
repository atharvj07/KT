import sys

for t in range(int(input())):
    n = int(input())
    f = [[] for i in range(n+1)]
    s = [0 for i in range(n+1)]
    for i in range(n):
        k = int(input());
        f[i+1] = [int(j) for j in input().split(' ')[1:]]
        for j in f[i+1]:
            s[j] = 1
    ans = False
    for i in range(n,0,-1):
        if s[i] == 0:
            for j in range(1,n+1):
                if i not in f[j]:
                    f[j].append(i)
                    ans = True
                    print("IMPROVE")
                    print(j, i)
                    break
        if ans: break
    if not ans: print("OPTIMAL")
    sys.stdout.flush()
