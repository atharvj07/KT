    cin >> a[i];
  }
  sort(a + 1, a + n + 1);
  if (a[k] != a[k + 1]) {
    cout << "YES\n";
    for (int i = 1; i <= k; ++i) {
      cout << lower_bound(a + 1, a + n + 1, a[i]) - a << ' ';
    }

  }

}
