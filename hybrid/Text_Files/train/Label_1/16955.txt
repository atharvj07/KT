#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n), b(n), c(n);
  for (int i = 0; i < n; i++) cin >> a[i] >> b[i] >> c[i];
  vector<long long> dp(n, -1);
  dp[0] = max({a[0], b[0], c[0]});
  for (int i = 1; i < n - 1; i++)
    dp[i] = max({a[i] + dp[i - 1], b[i] + max(dp[i - 1], dp[i - 2]), c[i] + dp[i - 2]});
  dp[n - 1] = max({a[n - 1] + dp[n - 2], b[n - 1] + max(dp[n - 1], dp[n - 2]), c[n - 1] + dp[n - 2]});
  cout << dp[n - 1] << endl;
}
