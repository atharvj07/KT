import java.util.Scanner;

public class JabberChecker {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String jabberId = scanner.nextLine();

        if (isValidJabberId(jabberId)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isValidJabberId(String jabberId) {
        // Check if jabberId is null or empty
        if (jabberId == null || jabberId.isEmpty()) {
            return false;
        }

        // Split jabberId into username, hostname, and resource
        String[] parts = jabberId.split("@");
        if (parts.length != 2) {
            return false;
        }
        String username = parts[0];
        String hostnameAndResource = parts[1];

        // Validate username
        if (!isValidUsername(username)) {
            return false;
        }

        // Split hostname and resource
        parts = hostnameAndResource.split("/");
        String hostname = parts[0];
        String resource = parts.length == 1 ? null : parts[1];

        // Validate hostname
        if (!isValidHostname(hostname)) {
            return false;
        }

        // Validate resource (optional)
        return resource == null || isValidResource(resource);
    }

    private static boolean isValidUsername(String username) {
        return username.matches("^[a-zA-Z0-9_]+$") && username.length() >= 1 && username.length() <= 16;
    }

    private static boolean isValidHostname(String hostname) {
        String[] words = hostname.split("\\.");
        if (words.length < 1 || words.length > 32) {
            return false;
        }
        for (String word : words) {
            if (!isValidUsername(word)) {
                return false;
            }
        }
        return true;
    }

    private static boolean isValidResource(String resource) {
        return resource.matches("^[a-zA-Z0-9_]+$") && resource.length() >= 1 && resource.length() <= 16;
    }
}
