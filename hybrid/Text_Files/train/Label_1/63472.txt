import sys

n = int(input())
a = list(map(int, input().split()))
if any(ai > n-1 or ai%2 != n%2 for ai in a):
    print('Impossible')
    sys.exit()
b = [0]*n
for i in range(n):
    b[i] = n - a[i] - 1
    a[i] -= (a[i] // 2) * 2
    for j in range(i+1, n):
        a[j] -= min(a[j], b[i])
        b[i] -= min(a[j], b[i])
if any(ai>0 or bi<0 for ai, bi in zip(a,b)):
    print('Impossible')
    sys.exit()
print('Possible')
print(*b)
