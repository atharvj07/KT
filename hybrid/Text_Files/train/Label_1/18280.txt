#include <iostream>
using namespace std;
int main() {
  string s, t;
  cin >> s >> t;
  int ls = s.length(), lt = t.length();
  if (ls != lt) {
    cout << "No";
    return 0;
  }
  bool v[1000], c[1000];
  for (int i = 0; i < ls; i++) {
    if (s[i] == t[i]) continue;
    if (s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' || s[i] == 'u' ||
        t[i] == 'a' || t[i] == 'e' || t[i] == 'i' || t[i] == 'o' || t[i] == 'u')
      v[i] = 1;
    else
      c[i] = 1;
  }
  int cnt = 0;
  for (int i = 0; i < ls; i++) { cnt += v[i] + c[i]; }
  if (cnt == ls)
    cout << "Yes";
  else
    cout << "No";
  return 0;
}
