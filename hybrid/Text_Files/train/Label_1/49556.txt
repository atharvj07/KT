import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        Arrays.sort(a);
        int m = 1;
        int i = 0;
        int j = 0;
        while (i < n && j < n) {
            if ((a[j] - a[i]) % m == 0) {
                j++;
            } else {
                i++;
                m++;
                j = i;
            }
        }

        if (j - i != n - k) {
            m++;
        }

        System.out.println(m);
    }
}
