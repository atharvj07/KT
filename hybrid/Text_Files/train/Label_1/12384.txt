#include <iostream>
#include <vector>
using namespace std;

vector<vector<int>> g;
vector<int> sub_tree;
int ans = 0;
int n, m;

void dfs(int u, int p) {
  sub_tree[u] = 1;
  for (int v : g[u]) {
    if (v == p) continue;
    dfs(v, u);
    sub_tree[u] += sub_tree[v];
  }
  ans = max(ans, sub_tree[u]);
}

int main() {
  int q;
  cin >> q;
  while (q--) {
    ans = 0;
    cin >> n;
    g.clear();
    g.resize(n + 1);
    sub_tree.clear();
    sub_tree.resize(n + 1);
    for (int i = 0; i < n - 1; i++) {
      int u, v;
      cin >> u >> v;
      g[u].push_back(v);
      g[v].push_back(u);
    }
    dfs(1, 1);
    cout << ans << "\n";
  }
}
