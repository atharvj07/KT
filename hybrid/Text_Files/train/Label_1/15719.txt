#include<bits/stdc++.h>
using namespace std ;
int main() {
    int n, ans = 0 ;
    bool ok[105] ;
    cin >> n ;
    memset(ok, 1, sizeof ok) ;
    for( int i = 0 ; i < n ; i++ ) {
        for( int j = 0 ; j < n ; j++ ) {
            int a ; cin >> a ;
            if(a&&a!=3) ok[i] = ok[j] = 0 ;
        }
    }
    for( int i = 0 ; i < n ; i++ ) if(ok[i]) ans++ ;
    cout << ans << endl ;
    for( int i = 0 ; i < n ; i++ ) if(ok[i]) cout << i+1 << ' ' ;
}
