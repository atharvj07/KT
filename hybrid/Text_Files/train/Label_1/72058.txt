n = int(input())
a = sorted([(int(i), k) for k, i in enumerate(input().split())][::-1])
ans = 0
cur = []
used = 0
for v, k in a:
	if len(cur) == 0:
		cur.append((v, 1, k))
		ans += 1
		used += 1
		continue
	last = cur[len(cur) - 1]
	if last[0] == v:
		last = (last[0], last[1] + 1, last[2])
		cur[len(cur) - 1] = last
		used += 1
	elif used == len(cur):
		cur.append((v, 1, k))
		ans += 1
		used += 1
	else:
		ans += 1
		used = 1
		cur = [(v, 1, k)]
print(ans)
