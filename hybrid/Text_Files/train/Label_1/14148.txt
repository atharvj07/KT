#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 1e5 + 5;
const ll inf = 1e18;
ll dp[maxn][2];
int main() {
    int n, m;
    cin >> n >> m;
    vector<pair<ll, ll>> a(n);
    for (int i = 0; i < n; i++) {
        ll x, y;
        cin >> x >> y;
        a[i] = {x, y};
    }
    sort(a.begin(), a.end(), [&](pair<ll, ll> &p1, pair<ll, ll> &p2) { return p1.second - p1.first > p2.second - p2.first; });
    dp[n][0] = dp[n][1] = inf;
    dp[n - 1][1] = a[n - 1].second;
    dp[n - 1][0] = a[n - 1].first;
    for (int i = n - 2; i >= 0; i--) {
        dp[i][1] = min(dp[i + 1][1], dp[i + 1][0] + a[i].second);
        dp[i][0] = min(dp[i + 1][1] + a[i].first - a[i].second, dp[i + 1][0] + a[i].first);
    }
    if (dp[0][1] <= m)
        cout << 0;
    else if (dp[0][0] <= m)
        cout << 1;
    else
        cout << -1;
    return 0;
}
