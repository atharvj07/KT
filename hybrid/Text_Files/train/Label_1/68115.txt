import sys
input = sys.stdin.readline

n, m = [int(_) for _ in input().split()]
coupons = []
d = {}
for _ in range(m):
  qi, wi = [int(_) for _ in input().split()]
  coupons.append((qi, wi))
  d[qi] = wi

coupons.sort()
a = 0
for qi, wi in coupons:
  if qi+1 in d:
    a = max(a, wi+d[qi+1])
  elif qi-1 in d:
    a = max(a, wi+d[qi-1])
  else:
    a = max(a, wi)
print(a)
