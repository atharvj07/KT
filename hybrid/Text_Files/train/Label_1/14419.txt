#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, q, k;
  cin >> n >> q >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  while (q--) {
    int l, r;
    cin >> l >> r;
    l--, r--;
    if (a[r] - a[l] > 1) cout << k - 1 << '\n';
    else cout << k - 2 << '\n';
  }
  return 0;
}
