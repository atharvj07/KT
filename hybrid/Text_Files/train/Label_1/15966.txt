#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const int N = 205, INF = 1e9+7;

int n, k, a[N], sub[N], par[N], mark[N];
vector<int> adj[N];

void dfs1(int u, int p) {
    par[u] = p;
    sub[u] = 1;
    for (int v: adj[u]) {
        if (v == p) continue;
        dfs1(v, u);
        sub[u] += sub[v];
    }
}

void dfs2(int u, int p, int keep) {
    for (int v: adj[u]) {
        if (v == p || mark[v]) continue;
        if (sub[v] > n/2) dfs2(v, u, 1);
        else dfs2(v, u, 0);
    }
    mark[u] = keep;
}

int get_centroid(int u) {
    dfs1(u, -1);
    int cen = -1, sz = -1;
    function<void(int,int)> dfs = [&](int u, int p) {
        int mx = n - sub[u];
        for (int v: adj[u]) {
            if (v == p || mark[v]) continue;
            dfs(v, u);
            mx = max(mx, sub[v]);
        }
        if (sz == -1 || mx < sz) {
            sz = mx;
            cen = u;
        }
    };
    dfs(u, -1);
    return cen;
}

ll solve(int u) {
    u = get_centroid(u);
    mark[u] = 1;
    ll ans = a[u];
    for (int v: adj[u]) {
        if (!mark[v]) ans += solve(v);
    }
    return ans;
}

int main() {
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    for (int i = 1; i < n; ++i) {
        int u, v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    cout << solve(1) << endl;
    return 0;
}
