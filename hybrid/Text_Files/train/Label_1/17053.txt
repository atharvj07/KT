#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, b, d;
  cin >> n >> b >> d;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  int cnt = 0, cur = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] > b)
      continue;
    cur += a[i];
    if (cur > d) {
      cnt++;
      cur = 0;
    }
  }

  cout << cnt << endl;
  return 0;
}
