import sys

n, m = map(int, sys.stdin.readline().split())
ans = [i for i in range(1, n + 1)]
arr = [i for i in range(1, n + 1)]

for i in range(n - 1):
    if m == 1:
        break
    if m > n - i - 1:
        m -= n - i - 1
        ans.append(arr.pop(0))
    else:
        ans.append(arr.pop(m - 1))
        m = 1

print(*ans)
