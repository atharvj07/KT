n = int(input())
a = list(map(lambda x: list(map(int, list(str(x)))), input().split()))

def f(x, y):
    z = []
    for i in range(max(len(x), len(y))):
        if len(x) > len(y):
            y.append(0)
        elif len(x) < len(y):
            x.append(0)
        z.append(y[i])
        z.append(x[i])
    return z

s = 0
for i in range(n):
    for j in range(n):
        z = f(a[i], a[j])
        x = 0
        for k in range(len(z)):
            x = (x * 10 + z[k]) % 998244353
        s += x

print(s % 998244353)
