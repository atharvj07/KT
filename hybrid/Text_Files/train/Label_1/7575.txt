#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const int N = 1e5 + 5;

int32_t main() {
  int n;
  cin >> n;
  unordered_map<int, int> mp;
  for (int i = 1; i <= n; ++i) {
    int x, y;
    cin >> x >> y;
    mp[x]++;
    mp[y]++;
  }
  for (int i = 1; i <= n; ++i) {
    int x, y;
    cin >> x >> y;
    cout << n - mp[x] << ' ' << n - mp[y] << endl;
  }
  return 0;
}
