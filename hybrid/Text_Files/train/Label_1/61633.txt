import sys

def min_digit(x):
    res = x % 10
    x //= 10
    while x > 0:
        res = min(x % 10, res)
        x //= 10
    return res


def max_digit(x):
    res = x % 10
    x //= 10
    while x > 0:
        res = max(x % 10, res)
        x //= 10
    return res


for _ in range(int(sys.stdin.readline())):
    n, k = map(int, sys.stdin.readline().split())
    while k > 0:
        mn = min_digit(n)
        mx = max_digit(n)
        n += mn * mx
        k -= 1
    print(n)
