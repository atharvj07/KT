#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  int min = 1e9;
  int max = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    min=min<a[i]?min:a[i];
    max=max>a[i]?max:a[i];
  }
  if ((max - min) % 2 == 0)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
