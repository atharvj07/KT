import java.util.Scanner;

public class ColorfulRibbon {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] colors = new int[n + 1];
        long[] colorfulness = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            colors[i] = i;
        }
        for (int i = 0; i < m; i++) {
            int type = scanner.nextInt();
            if (type == 1) {
                int l = scanner.nextInt();
                int r = scanner.nextInt();
                int x = scanner.nextInt();
                paint(l, r, x, colors, colorfulness);
            } else {
                int l = scanner.nextInt();
                int r = scanner.nextInt();
                System.out.println(query(l, r, colors, colorfulness));
            }
        }
    }

    private static void paint(int l, int r, int x, int[] colors, long[] colorfulness) {
        for (int i = l; i <= r; i++) {
            colorfulness[i] += Math.abs(x - colors[i]);
            colors[i] = x;
        }
    }

    private static long query(int l, int r, int[] colors, long[] colorfulness) {
        long sum = 0;
        for (int i = l; i <= r; i++) {
            sum += colorfulness[i];
        }
        return sum;
    }
}
