def f(x, y):
    x, y = str(x), str(y)
    result = []
    len_x, len_y = len(x), len(y)
    min_len = min(len_x, len_y)
    for i in range(1, min_len + 1):
        result.append(x[-i])
        result.append(y[-i])
    if len_x > len_y:
        result.extend(reversed(x[:len_x-len_y]))
    else:
        result.extend(reversed(y[:len_y-len_x]))
    return int(''.join(reversed(result)))

n = int(input())
a = list(map(int, input().split()))

mod = 998244353
total = 0
for i in range(n):
    for j in range(n):
        total = (total + f(a[i], a[j])) % mod

print(total)
