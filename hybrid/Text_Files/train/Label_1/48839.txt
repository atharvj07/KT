import java.util.Scanner;

public class LongestUncommonSubsequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String a = scanner.nextLine();
        String b = scanner.nextLine();

        // If the strings are equal, there is no uncommon subsequence.
        if (a.equals(b)) {
            System.out.println("-1");
            return;
        }

        // Find the length of the longest common subsequence (LCS).
        int[][] lcs = new int[a.length() + 1][b.length() + 1];
        for (int i = 1; i <= a.length(); i++) {
            for (int j = 1; j <= b.length(); j++) {
                if (a.charAt(i - 1) == b.charAt(j - 1)) {
                    lcs[i][j] = lcs[i - 1][j - 1] + 1;
                } else {
                    lcs[i][j] = Math.max(lcs[i - 1][j], lcs[i][j - 1]);
                }
            }
        }

        // The length of the longest uncommon subsequence is the length of the longest string minus the length of the LCS.
        int longestUncommonSubsequenceLength = Math.max(a.length(), b.length()) - lcs[a.length()][b.length()];

        System.out.println(longestUncommonSubsequenceLength);
    }
}
