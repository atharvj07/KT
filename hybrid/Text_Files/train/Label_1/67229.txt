import sys
input = sys.stdin.readline

def fun(n: int, seg: list[list[int]]) -> list[int]:
    seg.sort()
    x1, y1, x2, y2 = seg[0]
    for i in range(1, n):
        a, b, c, d = seg[i]
        if a>x2 or b<y1:
            x2 = max(x2, c)
            y1 = min(y1, d)
        else:
            return [-1]
    ans = [1]*n
    for i in range(n):
        a, b, c, d = seg[i]
        if x1<=a<=x2 and y1<=b<=y2:
            ans[i] = 2
    return ans

for _ in range(int(input())):
    n = int(input())
    seg = []
    for _ in range(n):
        a, b = map(int, input().split())
        seg.append([a, b, a, b])
    print(*fun(n, seg))
