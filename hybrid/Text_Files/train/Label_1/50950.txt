import java.util.Scanner;

public class Combinatorics {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        long[][] dp = new long[n + 1][k + 1];
        dp[0][0] = 1;

        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                // Connect the current vertex to 0, 1, 2, or 3 other vertices.
                for (int connected = 0; connected <= 3; connected++) {
                    if (i == 1 && connected > 2) {
                        continue; // Vertex 1 can only be connected to 0, 1, or 2 vertices.
                    }
                    if (j - connected < 0) {
                        continue; // Not enough edges for the matching.
                    }
                    dp[i][j] = (dp[i][j] + dp[i - 1][j - connected]) % MOD;
                }
            }
        }

        System.out.println(dp[n][k]);
    }
}
