import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int y = sc.nextInt();
            double max = 0;
            int bank = 0;
            for (int i = 1; i <= n; i++) {
                int b = sc.nextInt();
                int r = sc.nextInt();
                int t = sc.nextInt();
                double p = 0;
                if (t == 1) {
                    p = 100 * (1 + (r * y) / 100.0);
                } else {
                    p = 100 * Math.pow(1 + (r / 100.0), y);
                }
                if (p > max) {
                    max = p;
                    bank = b;
                }
            }
            System.out.println(bank);
        }
        sc.close();
    }
}
