#include <iostream>
#include <cstring>
using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.length();
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        bool flag = true;
        for (int j = 0; j < i; j++) {
            if (s[j] > s[j + 1]) {
                flag = false;
            }
        }
        if (flag) cnt++;
    }
    cout << cnt << endl;
    return 0;
}
