from collections import defaultdict
import sys

n, m = map(int, input().split())
g = defaultdict(list)
for _ in range(m):
    x, y = map(int, input().split())
    g[x].append(y)
    g[y].append(x)

vis = [False] * (n + 1)
par = [0] * (n + 1)

def dfs(u, p):
    vis[u] = True
    par[u] = p
    mx = -1
    Farthest = -1
    for v in g[u]:
        if not vis[v]:
            dfs(v, u)
            if mx < 1 + dfs(v, u):
                mx = 1 + dfs(v, u)
                Farthest = v
    D[u] = mx
    return mx

D = [0] * (n + 1)
mx = 0
Farthest1, Farthest2 = -1, -1
dfs(1, 1)
for i in range(1, n + 1):
    if D[i] > mx:
        mx = D[i]
        Farthest1 = i
dfs(Farthest1, Farthest1)
for i in range(1, n + 1):
    if D[i] > mx:
        mx = D[i]
        Farthest2 = i

Path = []
cur = Farthest2
while cur != Farthest1:
    Path.append((cur, par[cur]))
    cur = par[cur]

ans = mx
print(ans)
if mx != 1:
    for x, y in Path[::-1]:
        print(x, y)
