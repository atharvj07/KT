#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int l[n + 1], r[n + 1], pre[n + 1];
  memset(l, 0, sizeof(l));
  memset(r, 0, sizeof(r));
  memset(pre, 0, sizeof(pre));
  l[n] = n;
  r[0] = -1;
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == s[i + 1]) {
      l[i] = l[i + 1];
    } else {
      l[i] = i;
    }
  }
  for (int i = 1; i <= n; i++) {
    if (s[i] == s[i - 1]) {
      r[i] = r[i - 1];
    } else {
      r[i] = i;
    }
  }
  for (int i = 0; i < n; i++) {
    pre[i + 1] = pre[i];
    if (s[i] != s[i + 1]) {
      pre[i + 1]++;
    }
  }
  int ans = 0, j;
  for (int i = 0; i <= n; i++) {
    j = l[i] - 1;
    if (i - j > ans) {
      while (i + j <= n && pre[i + j] - pre[i] <= k) {
        j++;
      }
      ans = max(ans, 2 * j - i);
    }
    j = r[i] + 1;
    if (j - i > ans) {
      while (i - j >= 0 && pre[i] - pre[i - j] <= k) {
        j++;
      }
      ans = max(ans, 2 * j - i);
    }
  }
  cout << ans << endl;
  return 0;
}
