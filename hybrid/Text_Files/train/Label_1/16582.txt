#include <bits/stdc++.h>
 
using namespace std;
 
const int N = 1e5+5;
 
int n, q;
int a[N], b[N];
 
int l[N], r[N];
int ans[N]={};
bool exist[N];
 
 
int main() {
    cin >> n >> q;
    for(int i=1;i<=n;i++) cin >> a[i];
    for(int i=1;i<=n;i++) cin >> b[i];
 
    for(int i=1;i<=q;i++) {
        cin >> l[i] >> r[i];
        exist[l[i]]=1, exist[r[i]+1]=1;
    }
    int cnt=0;
    for(int i=1;i<=n;i++) {
        if(exist[i]) {
            if(a[i-1]!=b[i-1]) cnt++;
            if(cnt%2) ans[i]=-1;
            else ans[i]=cnt/2;
        }
    }
    for(int i=1;i<=q;i++) cout << ans[l[i]] << endl;
    return 0;
}
