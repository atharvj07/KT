import java.util.*;

public class Main {

    static class Player {
        int team;
        int n;
        int x;
        int y;

        public Player(int team, int n, int x, int y) {
            this.team = team;
            this.n = n;
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<Player> players = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            players.add(new Player(sc.nextInt(), sc.nextInt(), sc.nextInt(), sc.nextInt()));
        }
        sc.close();

        Map<Integer, List<Player>> teamPlayers = new HashMap<>();
        for (Player player : players) {
            teamPlayers.computeIfAbsent(player.team, k -> new ArrayList<>()).add(player);
        }

        double maxA = 0;
        int timeA = -1;
        int nA = -1;
        double maxB = 0;
        int timeB = -1;
        int nB = -1;

        for (List<Player> team : teamPlayers.values()) {
            for (int i = 0; i < team.size(); i++) {
                Player player = team.get(i);
                for (int j = i + 1; j < team.size(); j++) {
                    Player next = team.get(j);
                    if (player.n + 1 == next.n) {
                        double distance = Math.sqrt((player.x - next.x) * (player.x - next.x) + (player.y - next.y) * (player.y - next.y));
                        int time = next.n - player.n;
                        if (player.team == 0) {
                            if (distance > maxA || (distance == maxA && time < timeA)) {
                                maxA = distance;
                                timeA = time;
                                nA = player.n;
                            }
                        } else {
                            if (distance > maxB || (distance == maxB && time < timeB)) {
                                maxB = distance;
                                timeB = time;
                                nB = player.n;
                            }
                        }
                    }
                }
            }
        }

        if (nA == -1) {
            System.out.println(-1 + " " + -1);
        } else {
            System.out.println(maxA + " " + (double) timeA / 60);
        }
        if (nB == -1) {
            System.out.println(-1 + " " + -1);
        } else {
            System.out.println(maxB + " " + (double) timeB / 60);
        }
    }
}
