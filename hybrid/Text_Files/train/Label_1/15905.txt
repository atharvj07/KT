#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl '\n'
const int maxN = 1e5 + 5;
ll a[maxN];

int main() {
    ios_base::sync_with_stdio(0); 
    cin.tie(0); cout.tie(0);
    int n; cin >> n;
    ll dp[maxN], Max = 0;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        dp[i] = a[i];
        if (i - 1 >= 1) {
            dp[i] = max(dp[i - 1] + a[i], a[i]);
        }
        Max = max(Max, dp[i]);
    }
    ll sum = 0;
    int pos = 0;
    for (int i = n; i >= 1; i--) {
        sum += a[i];
        if (dp[i] == Max) {
            pos = i;
            break;
        }
    }
    int cnt = 0;
    for (int i = pos; i <= n; i++) {
        sum -= a[i];
        cnt++;
        if (sum * (i - pos + 1) == Max * cnt) {
            cout << i - pos + 1 << endl;
            break;
        }
    }
    return 0;
}
