#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    string a, b;
    cin >> a >> b;
    if (a.size() < b.size()) swap(a, b);
    int lena = a.size(), lenb = b.size(), carry = 0;
    vector<int> v(lena + 1, 0);
    reverse(a.begin(), a.end());
    reverse(b.begin(), b.end());
    for (int i = 0; i < lena; ++i) {
        v[i] += a[i] - '0' + carry;
        carry = v[i] / 10;
        v[i] %= 10;
    }
    if (carry) v[lena] = carry;
    for (int i = lenb; i < lena; ++i) {
        v[i] += b[i - lenb] - '0' + carry;
        carry = v[i] / 10;
        v[i] %= 10;
    }
    if (carry) v[lena] = carry;
    while (lena > 1 && v[lena] == 0) lena--;
    if (lena > lenb) cout << '>';
    else if (lena == lenb) {
        carry = 0;
        for (int i = 0; i < lena; ++i) {
            carry = carry || (v[i] != b[i] - '0');
            if (v[i] > b[i] - '0') break;
        }
        if (carry) cout << '>';
        else if (!carry) cout << '=';
        else cout << '<';
    } else cout << '<';
    cout << endl;
    return 0;
}
