#include<bits/stdc++.h>
using namespace std;
const int N=200;
char c[N+5];
int f[N+5][4][4]; //f[i][x][y]=true if we take this suffix and the last move is x, and the second last move is y.
int32_t main()
{
    int n;
    cin>>n;
    for(int i=1;i<=n;++i) cin>>c[i];
    int dx[]={0,0,1,-1},dy[]={1,-1,0,0};
    for(int i=n;i;--i)
    {
        for(int x=0;x<4;++x)
        {
            for(int y=0;y<4;++y)
            {
                int j=(i+1),nx=x,ny=y;
                if(c[i]=='U') nx=0,ny=1;
                else if(c[i]=='D') nx=2,ny=3;
                else if(c[i]=='L') nx=3,ny=2;
                else if(c[i]=='R') nx=1,ny=0;
                if(nx==ny) continue;
                f[i][x][y]|=f[j][nx][ny];
            }
        }
        for(int x=0;x<4;++x)
        {
            for(int y=0;y<4;++y)
            {
                f[i][x][y]|=(x==y&&f[i+1][x][y]);
            }
        }
    }
    int ans=0;
    for(int x=0;x<4;++x) for(int y=0;y<4;++y) ans+=f[1][x][y];
    cout<<ans<<endl;
}
