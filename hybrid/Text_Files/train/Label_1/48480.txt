import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        char[] A = new StringBuilder(in.next()).reverse().toString().toCharArray();
        char[] B = new StringBuilder(in.next()).reverse().toString().toCharArray();
        int max = Math.max(A.length, B.length);
        int[] C = new int[max + 1];
        for (int i = 0; i < max; i++) {
            if (i < A.length) C[i] += A[i] - '0';
            if (i < B.length) C[i] += B[i] - '0';
            C[i + 1] += C[i] / 10;
            C[i] %= 10;
        }
        StringBuilder str = new StringBuilder();
        for (int i = max; i >= 0; i--)
            if (C[i] != 0 || str.length() != 0) str.append(C[i]);
        System.out.println(str.length() == 0 ? "0" : str);
    }
}
