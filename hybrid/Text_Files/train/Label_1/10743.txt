#include <bits/stdc++.h>
using namespace std;

int main() {
  int da, db;
  cin >> da >> db;
  if (da == db) {
    cout << static_cast<long long>(da) * 100 + 99 << ' ' << static_cast<long long>(da) * 100 + 100
         << endl;
  } else if (da < db) {
    cout << static_cast<long long>(da) * 100 + 9 << ' ' << static_cast<long long>(da) * 100 + 10
         << endl;
  } else {
    cout << -1 << endl;
  }
  return 0;
}
