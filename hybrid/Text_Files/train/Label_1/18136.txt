#include <iostream>
#include <cstdio>
#include <vector>

using namespace std;

int n, m;
vector<vector<int>> a;
int ans = 1e9;

void dfs(int left, int right, int up, int down, int cnt) {
    if (left > right || up > down) return;
    if (left == right && up == down) {
        ans = min(ans, cnt);
        return;
    }
    for (int i = left; i <= right; ++i) {
        for (int j = up; j <= down; ++j) {
            if (a[i][j]) {
                dfs(left, right, j + 1, down, cnt + 1);
                dfs(i + 1, right, up, j - 1, cnt + 1);
                dfs(left, i - 1, up, j - 1, cnt + 1);
                dfs(i + 1, right, j + 1, down, cnt + 1);
            }
        }
    }
}

int main() {
    cin >> n >> m;
    a.resize(n + 1, vector<int>(m + 1));
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            cin >> a[i][j];
        }
    }
    dfs(1, n, 1, m, 0);
    cout << ans << endl;
    return 0;
}
