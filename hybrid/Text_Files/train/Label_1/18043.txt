#include <bits/stdc++.h>
using namespace std;

int main() {

  int n;
  cin >> n;
  map<int, set<int>> m;

  for (int i = 0; i < n; ++i) {
    int a, b;
    cin >> a;
    for (int j = 0; j < a; ++j) {
      cin >> b;
      m[i].insert(b);
    }
  }

  set<int> res;
  for (auto p : m) {
    for (auto s : res) {
      if (p.second.find(s) == p.second.end()) continue;
      for (auto t : p.second) {
        if (t == s) continue;
        res.insert(t);
      }
    }
    if (res.empty()) {
      res = p.second;
    }
  }

  for (auto x : res) { cout << x << ' '; }
  cout << '\n';
  return 0;
}
