import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        long[] a=new long[n+1];
        for(int i=1;i<=n;i++) a[i]=sc.nextLong();
        int ans=0;
        for(int i=1;i<=n;i++){
            int l=i+1, r=n;
            while(l<=r){
                int mid=(l+r)/2;
                if(a[mid]>=a[i]){
                    l=mid+1;
                }
                else{
                    ans+=mid-i;
                    r=mid-1;
                }
            }
        }
        System.out.println(ans);
    }
}
