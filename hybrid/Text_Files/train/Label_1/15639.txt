#include<bits/stdc++.h>
using namespace std;
#define ll long long
int n,k;
vector<int>a;
ll ans,mod=1e9+7,x;
void solve(int l,int r,int cnt){
    if(l>=r){
        if(is_sorted(a.begin(),a.end())) x++;
        return;
    }
    for(int i=l;i<=r;i++){
        for(int j=i+1;j<=r;j++){
            if(a[i]>a[j]){
                swap(a[i],a[j]);
                solve(l,r-1,cnt+1);
                swap(a[i],a[j]);
            }
        }
    }
}
int main(){
    int i;
    cin>>n>>k;
    a.resize(n);
    for(i=0;i<n;i++) cin>>a[i];
    solve(0,n-1,0);
    if(!x) cout<<0<<endl;
    else cout<<x%mod<<" "<<ans%mod<<" "<<endl;
}
