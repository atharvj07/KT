#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int depth = 0;
    int extra = (k - n) / 2;
    string ans;
    for (char x : s) {
        if (x == ')') {
            ans += x;
            depth--;
        }
        if (extra > 0) {
            ans += '(';
            depth++;
            extra--;
        }
        if (x == '(' && depth + 1 <= k / 2) {
            ans += x;
            depth++;
        }
    }
    cout << ans << endl;
    return 0;
}
