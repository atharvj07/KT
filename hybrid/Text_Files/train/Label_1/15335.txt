#include <iostream>
#include <vector>
using namespace std;
struct pair_i { int x, y; };
int main() {
  int n, k;
  cin >> n >> k;
  vector<pair_i> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i].x; }
  for (int i = 0; i < n; ++i) { cin >> a[i].y; }
  int res = 0;
  for (int i = 0; i < n; ++i) {
    int sum_x = a[i].x, sum_y = a[i].y;
    if (sum_x * k > sum_y) {
      res = max(res, sum_x);
      continue;
    }
    int j = i + 1;
    while (j < n && (sum_x * k <= sum_y || sum_y == 0)) {
      sum_x += a[j].x;
      sum_y += a[j].y;
      res = max(res, sum_x);
      ++j;
    }
  }
  if (res == 0) res = -1;
  cout << res << endl;
  return 0;
}
