import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    a = [int(x) for x in input().split()]
    a *= n
    l = [0] * len(a)
    ans = 0
    for i in range(len(a)):
        p = -1
        for j in range(i):
            if a[j] < a[i]:
                p = j
        l[i] = 1 if p == -1 else l[p] + 1
        ans = max(ans, l[i])
    return ans

t = int(input())
for _ in range(t):
    print(solve())
