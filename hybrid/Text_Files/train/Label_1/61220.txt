n=int(input())
s=[input()for i in range(n)]
check1=s[n//2][n//2]
flag=1
for i in range(n):
    if s[i][i]!=check1 or s[i][n-1-i]!=check1:
        flag=0
for i in range(n):
    for j in range(n):
        if i!=n//2 and j!=n//2 and s[i][j]!=s[n//2][n//2]:
            flag=0
if flag:
    print("YES")
else:
    print("NO")
