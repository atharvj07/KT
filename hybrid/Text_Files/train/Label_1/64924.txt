n, m = map(int, input().split())
arr = sorted([list(map(int, input().split())) for i in range(n)], key=lambda x: x[1])
i, ans, pos, neg = 0, 0, 0, 0
while i < n:
    j = i
    p, s = 0, arr[i][1]
    while j < n and arr[j][1] == s:
        if arr[j][0] != pos:
            neg = 1
        if arr[j][0] != neg:
            p = 1
        if not p:
            pos = arr[j][0]
        j += 1
    if neg and ans < 0:
        ans += sum(x[1] for x in arr[i:j])
    i = j
print(ans if ans >= 0 else 0)
