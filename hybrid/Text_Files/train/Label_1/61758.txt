n, k1, k2 = list(map(int, input().split()))
a = list(map(int, input().split()))
b = list(map(int, input().split()))
da =  sorted([(abs(a[i] - b[i]), i) for i in range(n)])
db =  sorted([(abs(b[i] - a[i]), i) for i in range(n)])
sum = 0
for i in range(min(k1,k2)):
    x, y = da[i]
    sum += x
    a[y] += 1 if a[y] < b[y] else -1
    x, y = db[i]
    sum += x
    b[y] += 1 if b[y] < a[y] else -1
for i in range(min(k1, len(da) - k2)):
    x, y = da[i + k2]
    sum += x
    a[y] += 1 if a[y] < b[y] else -1
for i in range(min(k2, len(db) - k1)):
    x, y = db[i + k1]
    sum += x
    b[y] += 1 if b[y] < a[y] else -1
print(sum)
