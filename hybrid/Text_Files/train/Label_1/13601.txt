#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  long long x[3], y[3], t, r;
  cin >> s >> x[0] >> x[1] >> x[2] >> y[0] >> y[1] >> y[2] >> r;
  for(int i = 0; i < 3; i++) {
    if(r < y[i]) r = -1;
    t = min(x[i], r / y[i]);
    r -= t * y[i];
    x[i] -= t;
  }
  y[0] = min(x[0], min(x[1], x[2]));
  cout << y[0] << '\n';
  return 0;
}
