#include <iostream>
#include <vector>

using namespace std;

const int maxn = 110;

double dp[maxn][maxn];
double p[maxn][maxn];
double ans;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> P(n);
    for (int i = 0; i < n; ++i) {
        cin >> P[i]; --P[i];
    }
    for (int len = 3; len <= n; ++len) {
        for (int l = 0; l + len - 1 < n; ++l) {
            for (int i = l; i < l + len - 1; ++i) {
                for (int j = i + 1; j < l + len; ++j) {
                    if (P[i] > P[j]) ++dp[l][l + len - 1];
                }
            }
        }
    }
    for (int len = 3; len <= n; ++len) {
        for (int l = 0; l + len - 1 < n; ++l) {
            p[l][l + len - 1] = dp[l][l + len - 1] / (len * (len - 1) / 2);
        }
    }
    for (int i = n - 1; i >= 1; --i) {
        for (int l = 0; l < n - i; ++l) {
            p[l][l + i] += p[l][l + i + 1] * (i + 1) / i;
        }
    }
    for (int i = 0; i < n; ++i) {
        ans += p[i][i] * (k + 1);
    }
    cout.precision(10);
    cout << fixed << ans << endl;
    return 0;
}
