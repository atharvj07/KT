import java.util.*;

class Main {
  static boolean[] vis;
  static List<Integer>[] G;
  static int n, m;

  static void dfs(int u) {
    if (vis[u]) {
      return;
    }
    vis[u] = true;
    for (int v : G[u]) {
      dfs(v);
    }
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    n = in.nextInt();
    m = in.nextInt();
    G = new ArrayList[n + 1];
    vis = new boolean[n + 1];
    for (int i = 1; i <= n; i++) {
      G[i] = new ArrayList<>();
    }
    for (int i = 1; i <= n; i++) {
      int k = in.nextInt();
      for (int j = 0; j < k; j++) {
        int v = in.nextInt();
        G[i].add(v);
      }
    }
    int s = in.nextInt();
    dfs(s);
    int v = -1;
    for (int i = 1; i <= n; i++) {
      if (!vis[i]) {
        v = i;
        break;
      }
    }
    if (v == -1) {
      System.out.println("Draw");
    } else {
      System.out.println("Lose");
    }
  }
}
