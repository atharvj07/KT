#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  cin >> T;
  while (T--) {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    bool flag = false;
    for (int i = 1; i < n; i++) {
      if (v[i] < v[i - 1]) flag = true;
    }
    if (!flag)
      cout << 0 << endl;
    else
      cout << 1 << endl;
  }
  return 0;
}
