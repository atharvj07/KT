def is_fair(n):
    for digit in str(n):
        if digit == '0':
            continue
        if n % int(digit) != 0:
            return False
    return True

def find_fair_number(n):
    while not is_fair(n):
        n += 1
    return n

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    print(find_fair_number(n))
