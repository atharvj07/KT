import java.util.*;
public class Main {
    static final int INF = 10000;
    static int[] x1 = new int[1000];
    static int[] y1 = new int[1000];
    static int[] x2 = new int[1000];
    static int[] y2 = new int[1000];
    static int[] t = new int[1000];
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int mx = -INF;
        int my = -INF;
        int Mx = INF;
        int My = INF;
        for (int i = 1; i <= n; i++) {
            int x = cin.nextInt();
            int y = cin.nextInt();
            int r = cin.nextInt();
            mx = Math.max(mx, x - r);
            my = Math.max(my, y - r);
            Mx = Math.min(Mx, x + r);
            My = Math.min(My, y + r);
            if (t[i] == 0) {
                x1[i] = x - r;
                y1[i] = y - r;
                x2[i] = x + r;
                y2[i] = y + r;
                t[i] = 1;
            }
        }
        ArrayList<Integer> v = new ArrayList<Integer>();
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (i == j) continue;
                long dx = x1[i] - x1[j];
                long dy = y1[i] - y1[j];
                long r = (long)x1[i] * x1[i] + (long)y1[i] * y1[i] - (long)x1[j] * x1[j] - (long)y1[j] * y1[j];
                for (int k = 0; k < 4; k++) {
                    int vx = x1[i] + (k & 1) * (x2[i] - x1[i]);
                    int vy = y1[i] + ((k & 2) >> 1) * (y2[i] - y1[i]);
                    if ((long)vx*vx + (long)vy*vy >= r) v.add((int) (dx * vy - dy * vx));
                }
            }
        }
        Collections.sort(v);
        int ans = 2;
        for (int i = 1; i < v.size(); i++) {
            if (!v.get(i).equals(v.get(i - 1))) ans++;
        }
        System.out.println(ans);
    }
}
