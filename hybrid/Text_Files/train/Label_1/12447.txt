#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[200005], ans = 1;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 2; i <= n; i++) {
        if (a[i] > a[ans]) ans = i;
    }
    cout << ans << endl;
    for (int i = ans; i >= 1; i--) {
        if (i == 1 || a[i - 1] < a[i]) cout << "L";
        else
            cout << "R";
    }
    cout << endl;
}
