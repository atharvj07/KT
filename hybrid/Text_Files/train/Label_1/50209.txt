import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.BitSet;
import java.util.StringTokenizer;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken()), m = Integer.parseInt(st.nextToken()), k = Integer.parseInt(st.nextToken()), q = Integer.parseInt(st.nextToken());
        int[][] rooks = new int[k][2];
        for (int i = 0; i < k; i++) {
            st = new StringTokenizer(br.readLine());
            rooks[i][0] = Integer.parseInt(st.nextToken()) - 1;
            rooks[i][1] = Integer.parseInt(st.nextToken()) - 1;
        }
        int[] area = new int[q];
        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int x1 = Integer.parseInt(st.nextToken()) - 1, y1 = Integer.parseInt(st.nextToken()) - 1, x2 = Integer.parseInt(st.nextToken()) - 1, y2 = Integer.parseInt(st.nextToken()) - 1;
            area[i] = (x1 << 20) | (y1 << 10) | (x2 << 8) | y2;
        }
        boolean[] covered = new boolean[n * m];
        for (int i = 0; i < m; i++) {
            Arrays.fill(covered, false);
            for (int[] rook : rooks) {
                covered[rook[0] * m + i] = true;
            }
            for (int[] rook : rooks) {
                for (int j = rook[0] + 1; j < n; j++) {
                    covered[j * m + i] |= covered[(j - 1) * m + i];
                }
            }
            for (int[] rook : rooks) {
                for (int j = rook[0] - 1; j >= 0; j--) {
                    covered[j * m + i] |= covered[(j + 1) * m + i];
                }
            }
        }
        ArrayList<Integer> b = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            Arrays.fill(covered, false);
            for (int[] rook : rooks) {
                covered[rook[1] * m + i] = true;
            }
            for (int[] rook : rooks) {
                for (int j = rook[1] + 1; j < m; j++) {
                    covered[rook[1] * m + j] |= covered[rook[1] * m + (j - 1)];
                }
            }
            for (int[] rook : rooks) {
                for (int j = rook[1] - 1; j >= 0; j--) {
                    covered[rook[1] * m + j] |= covered[rook[1] * m + (j + 1)];
                }
            }
            BitSet bs = new BitSet();
            for (int j = 0; j < n * m; j++) {
                if (covered[j]) {
                    bs.set(j / m);
                }
            }
            b.add(bs.cardinality());
        }
        for (int i = 0; i < q; i++) {
            int x1 = area[i] >> 20, y1 = (area[i] >> 10) & 1023, x2 = area[i] >> 8 & 1023, y2 = area[i] & 1023;
            System.out.println(b.get(y1) >= x2 - x1 + 1 && b.get(y2) >= x2 - x1 + 1 ? "YES" : "NO");
        }
    }
}
