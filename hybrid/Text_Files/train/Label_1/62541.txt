from bisect import bisect_left

n = int(input())
b = list(map(int, input().split()))
a = sorted(b)
for i in range(1, len(a)):
    if b[i] != a[i]:
        if i < len(b) - 1 and a[i - 1] + a[i + 1] == 2 * b[i]:
            print(i + 1)
            exit()
        else:
            print(i)
            exit()
print(-1)
