#include <bits/stdc++.h>
using namespace std;
int main() {
  int t, n, k;
  cin >> t;
  while (t--) {
    cin >> n >> k;
    if (n * (n + 1) / 2 < k) {
      cout << -1 << endl;
      continue;
    }
    vector<int> p(n + 1);
    iota(p.begin(), p.end(), 1);
    for (int i = n - 1; i >= 1; i--) {
      if (k > i) {
        swap(p[i], p[k - i]);
        k -= i;
      }
    }
    if (k == 1) swap(p[n - 1], p[n]);
    for (int i = 1; i <= n; i++) cout << p[i] << ' ';
    cout << endl;
  }
}
