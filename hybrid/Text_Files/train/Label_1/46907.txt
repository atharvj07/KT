import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n;
    while ((n = scanner.nextInt()) != 0) {
      int[] a = new int[n];
      int count = 0;
      for (int i = 0; i < n; i++) {
        a[i] = scanner.nextInt();
      }
      for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
          if (a[i] > a[j]) {
            int tmp = a[i];
            a[i] = a[j];
            a[j] = tmp;
            count++;
          }
        }
      }
      System.out.println(count);
    }
  }
}
