#include <bits/stdc++.h>
#define endl "\n"
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int ans = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] - a[i - 1] > 1) {
      ans = max(ans, i - 1);
      break;
    }
  }
  for (int i = n - 2; i >= 0; i--) {
    if (a[i + 1] - a[i] > 1) {
      ans = max(ans, n - i - 1);
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
