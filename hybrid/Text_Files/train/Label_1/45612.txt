import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

class Main {

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        int n = scan.nextInt();
        int m = scan.nextInt();
        Map<Integer, List<Integer>> roads = new HashMap<>();
        for (int i = 0; i < m; i++) {
            int a = scan.nextInt() - 1;
            int b = scan.nextInt() - 1;
            roads.computeIfAbsent(a, k -> new ArrayList<>()).add(b);
            roads.computeIfAbsent(b, k -> new ArrayList<>()).add(a);
        }

        System.out.println(solve(roads));
    }

    private static int solve(Map<Integer, List<Integer>> roads) {
        int[] state = new int[roads.size()];
        for (int i = 0; i < state.length; i++) {
            state[i] = -1;
        }

        int minRoads = Integer.MAX_VALUE;
        for (int i = 0; i < state.length; i++) {
            if (state[i] == -1) {
                minRoads = Math.min(minRoads, dfs(i, 0, state, roads));
            }
        }
        if (minRoads == Integer.MAX_VALUE) {
            return -1;
        }
        return minRoads;
    }

    private static int dfs(int cur, int prevState, int[] state, Map<Integer, List<Integer>> roads) {
        if (state[cur] != -1) {
            return state[cur] == prevState ? 0 : Integer.MAX_VALUE;
        }

        int minRoads = Integer.MAX_VALUE;
        state[cur] = 1 - prevState;
        minRoads = Math.min(minRoads, 1 + dfs(roads.get(cur).get(0), state[cur], state, roads));
        minRoads = Math.min(minRoads, 1 + dfs(roads.get(cur).get(1), state[cur], state, roads));
        state[cur] = -1;
        return minRoads;
    }
}
