import sys

input = sys.stdin.readline

n = int(input().strip())
l = []
r = []
for i in range(n):
    a, b = map(int, input().strip().split())
    l.append(a)
    r.append(b)

l.sort()
r.sort()
cnt = [0] * (n + 1)
cur = 0
j = 0
for i in range(n):
    while j < n and l[j] <= r[i]:
        cur += 1
        j += 1
    cnt[cur] += r[i] - l[i] + 1
    cur -= 1
for i in range(1, n + 1):
    print(cnt[i], end=' ')
