#include <iostream>
#include <vector>
#include <list>
#include <cmath>

using namespace std;

typedef long long LL;

int Solve(const LL& n) {
    LL ans = n;
    vector<bool> used(10, false);
    for (LL i = 2; i * i <= ans && ans > n; ++i) {
        if (ans % i == 0) {
            while (ans % i == 0) {
                ans /= i;
                used[i] = true;
            }
            if (ans % i != 0) {
                ans = ans * i;
                LL fre = 1;
                while (ans % i == 0) {
                    ans /= i;
                    fre++;
                }
                LL val = (pow(i, fre) - 1) / (i - 1);
                if (used[i] == false)
                    ans *= val;
            }
        }
    }
    if (ans != 1)
        ans *= ans;
    return ans;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        LL n;
        cin >> n;
        cout << Solve(n) << endl;
    }
    return 0;
}
