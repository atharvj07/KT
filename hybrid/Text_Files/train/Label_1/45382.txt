import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while(true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if(n == 0 && m == 0) break;
            int[][] a = new int[n][m];
            for(int i = 0; i < n; i++) {
                String s = sc.next();
                for(int j = 0; j < m; j++)
                    a[i][j] = s.charAt(j) - '0';
            }
            int[] b = new int[m];
            Arrays.fill(b, 0);
            int ans = 0;
            for(int i = 0; i < n; i++) {
                int cnt = 0;
                for(int j = 0; j < m; j++) {
                    if(a[i][j] == 1) {
                        b[j]++;
                        cnt++;
                    }
                }
                if(cnt % 2 == 0) ans++;
                else {
                    boolean flag = false;
                    for(int j = 0; j < m; j++) {
                        if(a[i][j] == 1 && b[j] % 2 == 0) {
                            b[j]++;
                            flag = true;
                            break;
                        }
                    }
                    if(flag) ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
