#include <iostream>
#include <vector>
using namespace std;

int main() {
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<int> p(n);
        for (int i = 0; i < n; i++) {
            cin >> p[i];
            p[i]--;
        }
        vector<int> ans(n);
        for (int i = 0; i < n; i++) {
            int j = i;
            int cnt = 0;
            while (true) {
                j = p[j];
                cnt++;
                if (j == i) {
                    ans[i] = cnt;
                    break;
                }
            }
        }
        for (auto x : ans) cout << x << " ";
        cout << endl;
    }
    return 0;
}
