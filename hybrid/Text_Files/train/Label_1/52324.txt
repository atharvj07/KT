import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        int[][] bc = new int[q][2];
        for (int i = 0; i < q; i++) {
            bc[i][0] = sc.nextInt();
            bc[i][1] = sc.nextInt();
        }
        long[] s = new long[q];
        for (int i = 0; i < n; i++) {
            s[0] += a[i];
        }
        for (int i = 1; i < q; i++) {
            for (int j = 0; j < n; j++) {
                if (a[j] == bc[i - 1][0]) {
                    s[i] -= bc[i - 1][0];
                    s[i] += bc[i - 1][1];
                }
            }
            s[i] += s[i - 1];
        }
        for (int i = 0; i < q; i++) {
            System.out.println(s[i]);
        }
    }
}
