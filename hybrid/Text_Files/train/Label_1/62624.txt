import sys
input = sys.stdin.readline

n, m = map(int, input().split())
vs = set()
hs = []
Max = 0
for i in range(n):
    vs.add(int(input()))
for i in range(m):
    x, y, z = map(int, input().split())
    hs.append((x-1, y, z))
    Max = max(Max, z)
hs.sort()
ans = 0
for x1, y, x2 in hs:
    if x2 == Max:
        break
    if x1 not in vs and (x1+1, y, x2) not in hs:
        ans += 1
print(ans)
