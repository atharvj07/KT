n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
a.sort()
b.sort()
a_s = []
b_s = []
a_i = 0
b_i = 0
for j in range(n):
    if a_i < n and a[a_i] > b[b_i]:
        a_s.append(a[a_i])
        a_i += 1
    elif b_i < n and b[b_i] > a[a_i]:
        b_s.append(b[b_i])
        b_i += 1
    else:
        a_i += 1
        b_i += 1
print(sum(a_s) - sum(b_s))
