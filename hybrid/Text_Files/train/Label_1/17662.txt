#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    int n = s.length();
    int cntV = 0, ans = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'K') {
            ans += cntV;
        }
        if (s[i] == 'V') {
            cntV++;
        }
    }
    ans = max(ans, cntV - 1);
    cout << ans << endl;
    return 0;
}
