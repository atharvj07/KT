#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<bool> present(n + 1, false);
  int leader_count = 0;
  vector<int> leaders;
  for (int i = 0; i < m; i++) {
    string message;
    int id;
    cin >> message >> id;
    if (message[0] == '+') {
      present[id] = true;
      if (leader_count == 0) leaders.push_back(id);
    } else {
      present[id] = false;
      if (leader_count == 0) {
        for (int j = 1; j <= n; j++) {
          if (present[j]) {
            leaders.push_back(j);
            break;
          }
        }
      }
    }
    bool is_leader = true;
    for (int j = 1; j <= n; j++) {
      if (!present[j]) {
        is_leader = false;
        break;
      }
    }
    if (is_leader) {
      leader_count++;
      if (leader_count > 1) leaders.clear();
    }
  }
  cout << leaders.size() << '\n';
  for (int leader : leaders) {
    cout << leader << ' ';
  }
  cout << '\n';
  return 0;
}
