n, m = map(int, input().split())
v = list(map(int, input().split()))
e = [[] for _ in range(n+1)]
for i in range(m):
    x, y = map(int, input().split())
    e[x]+=[y]; e[y]+=[x]
q = [i for i in range(1,n+1) if len(e[i])==1]
while q:
    i = q.pop(0)
    j = e[i].pop()
    v[i-1]+=v[j-1]; v[j-1]=0
    e[j].remove(i)
    if len(e[j])==1: q+=[j]
print(sum(v))
