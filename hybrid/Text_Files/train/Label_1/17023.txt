#include <iostream>
#include <cstdio>
#include <algorithm>

using namespace std;

int n, w[102], ans;

int main() {
    cin >> n;
    for (int i = 1; i <= 2 * n; i++) cin >> w[i];
    sort(w + 1, w + 2 * n + 1);
    for (int i = 1; i <= n - 1; i++) ans += w[i + n] - w[i];
    cout << ans << endl;
    return 0;
}
