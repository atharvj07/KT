from collections import deque, defaultdict

n, m, k = map(int, input().split())
field = [list(input()) for _ in range(n)]

spiders = deque()
for i in range(n):
    for j in range(m):
        if field[i][j]!='.':
            spiders.append((i,j,field[i][j]))

ans = [-1] * m

dx = [-1, 0, 1, 0]
dy = [0, 1, 0, -1]
while spiders:
    i, j, d = spiders.popleft()
    if i==n-1:
        ans[j]+=1
        continue
    x, y = i+dx[ord(d)-ord('L')], j+dy[ord(d)-ord('L')]
    if 0<=x<n and 0<=y<m:
        if field[x][y]=='.':
            spiders.append((x, y, d))
            field[x][y]=d
        elif field[x][y]==d:
            spiders.append((x, y, d))
            field[x][y]='.'
            spiders.append((i, j, d))

for x in ans:
    print(x, end=' ')
