#include <bits/stdc++.h>
using namespace std;

#define endl "\n"
#define ll long long
#define all(x) x.begin(), x.end()

bool valid(vector<vector<char>> &grid, int x, int y, int n, int m) {
    for (int i = x - 1; i <= x + 1; i++) {
        for (int j = y - 1; j <= y + 1; j++) {
            if (i >= 0 && i < n && j >= 0 && j < m && (i != x || j != y)) {
                if (grid[i][j] == grid[x][y]) return false;
            }
        }
    }
    return true;
}

void solve(vector<vector<char>> &grid, int n, int m, bool &found) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (grid[i][j] == '.') {
                grid[i][j] = 'R';
                if (valid(grid, i, j, n, m)) {
                    solve(grid, n, m, found);
                    if (!found) grid[i][j] = '.';
                }
                grid[i][j] = 'W';
                if (valid(grid, i, j, n, m)) {
                    solve(grid, n, m, found);
                    if (!found) grid[i][j] = '.';
                }
            }
        }
    }
    if (found) return;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (grid[i][j] == '.') return;
        }
    }
    found = true;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<char>> grid(n, vector<char>(m));
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                cin >> grid[i][j];
            }
        }
        bool found = false;
        solve(grid, n, m, found);
        if (!found) cout << "NO" << endl;
        else {
            cout << "YES" << endl;
            for (vector<char> &row : grid) {
                for (char &ch : row) cout << ch;
                cout << endl;
            }
        }
    }
    return 0;
}
