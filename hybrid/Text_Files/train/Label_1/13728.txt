#include <bits/stdc++.h>
using namespace std;
#define fast ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
int main() {
    fast;
    int n, m, a[1002], tmp[1002];
    string s[1002];
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> s[i];
    for (int i = 0; i < m; i++) cin >> a[i];
    int ans = 0;
    for (int i = 0; i < m; i++) {
        int cnt[5] = {0};
        for (int j = 0; j < n; j++) cnt[s[j][i] - 'A']++;
        int mx = 0;
        for (int j = 0; j < 5; j++) mx = max(mx, cnt[j]);
        for (int j = 0; j < 5; j++)
            if (cnt[j] == mx) tmp[j] = a[i];
        ans += mx * tmp[max_element(tmp, tmp + 5) - tmp];
    }
    cout << ans;
}
