#include <cstdio>
#include <vector>
#include <algorithm>

int main() {
  int n;
  scanf("%d", &n);
  std::vector<int> a(n), b(n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &a[i]);
    b[i] = a[i];
  }
  std::sort(b.begin(), b.end());
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    if (a[i] != b[i]) {
      ++ans;
    }
  }
  printf("%d\n", ans);
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      if (a[i] == b[j]) {
        b[j] = -1;
        printf("%d ", i + 1);
        break;
      }
    }
  }
  puts("");
  return 0;
}
