#include <bits/stdc++.h>
using namespace std;
string p[1005], s;
map<string, set<string> > mp;
int main() {
  int n, sz;
  cin >> n;
  while (n--) {
    cin >> s;
    cin >> sz;
    while (sz--) {
      string num;
      cin >> num;
      if (mp[s].size() == 0) {
        mp[s].insert(num);
      } else {
        string u = num;
        int x, y;
        while (true) {
          x = 0, y = 0;
          for (int i = 0; i < u.size(); i++) x = x * 10 + u[i] - '0';
          for (int i = 0; i < mp[s].begin()->size(); i++) y = y * 10 + mp[s].begin()->at(i) - '0';
          if (x <= y) {
            mp[s].erase(mp[s].begin());
            mp[s].insert(num);
            u = num;
          } else {
            break;
          }
        }
      }
    }
  }
  cout << mp.size() << endl;
  for (map<string, set<string> >::iterator it = mp.begin(); it != mp.end(); it++) {
    cout << it->first << " " << it->second.size() << " ";
    for (set<string>::iterator it2 = it->second.begin(); it2 != it->second.end(); it2++) cout << *it2 << " ";
    cout << endl;
  }
  return 0;
}
