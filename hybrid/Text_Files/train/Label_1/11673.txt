#include<bits/stdc++.h>
using namespace std;
using ll = long long;

const int N = 505;
ll dist[N][N], dp[N];
int n, vis[N];
vector<vector<int>> g;

int main() {
	ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >> n;
	g.resize(n);
	for(int i = 0; i < n; i++){
		for(int j = 0; j < n; j++){
			int x; cin >> x;
			g[i].push_back(x);
			dist[i][j] = 1e18;
		}
	}
	for(int i = 0; i < n; i++){
		for(int j = 0; j < n; j++){
			if(dist[i][j] == 1e18){
				dist[i][j] = dist[j][i] = g[i][j];
			}
		}
		dist[i][i] = 0;
	}
	for(int k = 0; k < n; k++) {
		for(int i = 0; i < n; i++){
			for(int j = 0; j < n; j++) {
				dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);
			}
		}
	}
	vector<int> seq(n);
	for(int i = 0; i < n; i++){
		cin >> seq[i];
	}
	for(int i = 0; i < n; i++){
		for(int j = 0; j < n; j++){
			if(i == j) continue;
			dp[i] += dist[seq[i]][seq[j]];
		}
		cout << dp[i] << ' ';
		vis[seq[i]] = 1;
		for(int j = 0; j < n; j++){
			for(int k = 0; k < n; k++){
				if(vis[j] || vis[k]) continue;
				dist[j][k] = min(dist[j][k], dist[j][seq[i]] + dist[seq[i]][k]);
			}
		}
	}
	cout << endl;
	return 0;
}
