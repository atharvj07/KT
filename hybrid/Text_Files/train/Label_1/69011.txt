from math import gcd

def lcm(x, y):
    return x * y // gcd(x, y)

n, m = map(int, input().split())
a = list(map(int, input().split()))

answer_lcm = 1
answer_length = 0
answer_indices = []

for i in range(1, m + 1):
    indices = []
    current_lcm = 1
    for j in range(n):
        if a[j] % i == 0:
            temp_lcm = lcm(current_lcm, a[j])
            if temp_lcm <= m:
                current_lcm = temp_lcm
                indices.append(j + 1)
    if len(indices) > answer_length:
        answer_lcm = current_lcm
        answer_length = len(indices)
        answer_indices = indices

print(answer_lcm, answer_length)
print(*answer_indices)
