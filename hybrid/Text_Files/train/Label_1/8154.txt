#include <iostream>
using namespace std;
long long gcd(long long a, long long b) {
    return b ? gcd(b, a % b) : a;
}
int main() {
    long long a, b;
    cin >> a >> b;
    long long aa = a, bb = b;
    while (1) {
        long long k = b / a * a - a;
        if (!k)
            break;
        long long g = gcd(a + k, b + k);
        if (g < gcd(aa, bb)) {
            aa = a + k;
            bb = b + k;
        }
        a += k;
    }
    cout << aa - a << endl;
    return 0;
}
