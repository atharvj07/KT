#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    long long n, k;
    cin >> n >> k;
    vector<long long> a(n), sorted_a(n);
    for (long long i = 0; i < n; i++) {
        cin >> a[i];
        sorted_a[i] = a[i];
    }
    sort(sorted_a.begin(), sorted_a.end());
    long long ans = 0;
    long long sum = 0;
    for (long long i = 0; i < n; i++) {
        sum += sorted_a[i];
        if (sum > k) break;
        ans++;
    }
    long long l = 0, r = ans;
    while (l < r) {
        long long mid = (l + r + 1) / 2;
        sum = 0;
        for (long long i = 0; i < mid; i++) {
            sum += a[i];
            if (sum > k) {
                r = mid - 1;
                break;
            }
        }
        if (sum <= k) l = mid;
    }
    cout << l << endl;
    return 0;
}
