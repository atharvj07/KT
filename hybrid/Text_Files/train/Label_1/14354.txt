#include <bits/stdc++.h>
using namespace std;

int main() {
    long long a, b, op = 0;
    cin >> a >> b;
    while (a != b) {
        if (a > b) {
            if (a % 2 == 0) {
                op++;
                a /= 2;
            } else if (a % 3 == 0) {
                op++;
                a = (a * 2) / 3;
            } else if (a % 5 == 0) {
                op++;
                a = (a * 4) / 5;
            } else {
                op = -1;
                break;
            }
        } else {
            if (b % 2 == 0) {
                op++;
                b /= 2;
            } else if (b % 3 == 0) {
                op++;
                b = (b * 2) / 3;
            } else if (b % 5 == 0) {
                op++;
                b = (b * 4) / 5;
            } else {
                op = -1;
                break;
            }
        }
    }
    cout << op << endl;
    return 0;
}
