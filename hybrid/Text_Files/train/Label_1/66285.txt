import sys
input = sys.stdin.readline

n, k = map(int, input().split())
a = list(map(int, input().split()))
freq = {}
l = r = ans_l = ans_r = 0
max_len = 0
for value in a:
    freq[value] = freq.get(value, 0) + 1
    while len(freq) > k:
        freq[a[l]] -= 1
        if freq[a[l]] == 0:
            del freq[a[l]]
        l += 1
    if r - l + 1 > max_len:
        max_len = r - l + 1
        ans_l = l
        ans_r = r
    r += 1
print(ans_l + 1, ans_r + 1)
