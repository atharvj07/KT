#include <bits/stdc++.h>
#include <map>
#define ll long long
#define pb push_back
#define mp make_pair

using namespace std;

int k,n[200005],i1,i2,l[200005],r[200005],sum[200005];
ll a[200005];
map <ll, pair<int,int> > M;
int main()
{
    cin>>k;
    for (int i=1;i<=k;i++)
    {
        cin>>n[i];
        for (int j=1;j<=n[i];j++)
        {
            cin>>a[j];
            sum[i]+=a[j];
        }
        l[i]=M[sum[i]].first;
        r[i]=M[sum[i]].second;
        M[sum[i]]=mp(i,i);
    }
    for (int i=k;i>=1;i--)
    {
        M[sum[i]]=mp(l[i],r[i]);
        if (r[i]-l[i]>0)
        {
            if (i1!=0 && i1!=i)
            {
                cout<<"YES\n"<<i1<<" "<<n[i1]-r[i1]+l[i1]<<" "<<i<<" "<<n[i]-r[i]+l[i]<<endl;
                return 0;
            }
            if (i1==0) i1=i;
        }
        if (i1!=0 && i1!=i)
        {
            cout<<"YES\n"<<i1<<" "<<n[i1]-r[i1]+l[i1]<<" "<<i<<" "<<n[i]-r[i]+l[i]<<endl;
            return 0;
        }
    }
    cout<<"NO";

    return 0;
}
