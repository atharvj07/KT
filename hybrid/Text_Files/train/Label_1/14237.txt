#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n;
        cin>>n;
        int a[2*n];
        for(int i=0;i<2*n;i++) cin>>a[i];
        bool f=1;
        for(int i=1;i<2*n;i++)
        {
            if(a[i]<a[i-1]) f=0;
        }
        if(f) cout<<"NO"<<endl;
        else cout<<"YES"<<endl;
    }
}
