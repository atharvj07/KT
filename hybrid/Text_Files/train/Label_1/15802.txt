#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  int ans = 360;
  for (int i = 0; i < n / 2; i++) {
    ans = min(ans, a[i + n / 2] - a[i]);
  }
  cout << ans << endl;
  return 0;
}
