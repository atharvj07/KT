import java.util.Scanner;

public class JabberIDChecker {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String jabberID = scanner.nextLine();

        if (isValidJabberID(jabberID)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isValidJabberID(String jabberID) {
        // Split the Jabber ID into its components.
        String[] components = jabberID.split("@|/|\\.");

        // Check if the Jabber ID has the correct number of components.
        if (components.length < 2 || components.length > 3) {
            return false;
        }

        // Check the validity of each component.
        String username = components[0];
        String hostname = components[1];
        String resource = components.length == 3 ? components[2] : null;

        if (!isValidUsername(username)) {
            return false;
        }

        if (!isValidHostname(hostname)) {
            return false;
        }

        if (resource != null && !isValidResource(resource)) {
            return false;
        }

        return true;
    }

    private static boolean isValidUsername(String username) {
        return username.matches("^[a-zA-Z0-9_]+$") && username.length() >= 1 && username.length() <= 16;
    }

    private static boolean isValidHostname(String hostname) {
        return hostname.matches("^[a-zA-Z0-9_\\.]+$") && hostname.length() >= 1 && hostname.length() <= 32;
    }

    private static boolean isValidResource(String resource) {
        return resource.matches("^[a-zA-Z0-9_]+$") && resource.length() >= 1 && resource.length() <= 16;
    }
}
