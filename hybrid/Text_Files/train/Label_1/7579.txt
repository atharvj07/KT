#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> b(n);
    for (int i = 0; i < n; i++) cin >> b[i];
    map<char, vector<int>> pos;
    for (int i = 0; i < n; i++) pos[s[i]].push_back(i);
    vector<char> t(n);
    for (auto p : pos) {
        vector<int> v = p.second;
        for (int i = 0; i < v.size() / 2; i++) {
            t[v[i]] = p.first;
            t[v[v.size() - 1 - i]] = p.first;
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++)
        if (s[i] == t[i]) ans += b[i];
    cout << ans << endl;
    return 0;
}
