#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll n,h,m,l[51],r[51],x[51],dp[51][51][51];
int main() {
    cin>>n>>h>>m;
    for(ll i=1;i<=m;i++){
        cin>>l[i]>>r[i]>>x[i];
    }
    for(ll i=1;i<=n;i++){
        for(ll j=0;j<=h;j++){
            for(ll k=0;k<=m;k++){
                dp[i][j][k]=dp[i-1][j][k];
                if(i>=l[k] && i<=r[k]){
                    dp[i][j][k]=max(dp[i][j][k],dp[i-1][min(j,x[k])][k]+(j*j));
                }
                else dp[i][j][k]=max(dp[i][j][k],dp[i-1][j][k]+(j*j));
            }
        }
    }
    cout<<dp[n][h][0]<<endl;
    return 0;
}
