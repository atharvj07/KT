#include <bits/stdc++.h>
using namespace std;
const int dir[4][2] = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
char a[55][55];
bool used[55][55][4];
bool dfs(int x, int y, int d, char color, int dep) {
  if (dep == 4 && (a[x][y] == color || d == 0)) return true;
  if (used[x][y][d] || a[x][y] != color || dep > 4) return false;
  used[x][y][d] = 1;
  int nx = x + dir[d][0];
  int ny = y + dir[d][1];
  if (nx >= 1 && nx <= 50 && ny >= 1 && ny <= 50) {
    if (dfs(nx, ny, d, color, dep + 1)) return true;
  }
  nx = x + dir[(d + 3) % 4][0];
  ny = y + dir[(d + 3) % 4][1];
  if (nx >= 1 && nx <= 50 && ny >= 1 && ny <= 50) {
    if (dfs(nx, ny, (d + 3) % 4, color, dep + 1)) return true;
  }
  return false;
}
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) cin >> a[i][j];
  }
  bool ans = false;
  for (int i = 1; i <= n && !ans; ++i) {
    for (int j = 1; j <= m && !ans; ++j) {
      for (int k = 0; k < 4 && !ans; ++k) {
        if (dfs(i, j, k, a[i][j], 0)) ans = true;
      }
    }
  }
  if (ans)
    cout << "Yes" << endl;
  else
    cout << "No" << endl;
  return 0;
}
