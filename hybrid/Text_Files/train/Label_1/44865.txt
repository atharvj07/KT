import java.util.ArrayList;
import java.util.Collections;
import java.util.Scanner;

class Main {
  private static final int MAX_K = 5;
  private static final int GRID_SIZE = 4;
  private static final int MAX_PANEL_VALUE = 9;
  private static final char MARKER_DOT = '.';

  private static class Panel implements Comparable<Panel> {
    int row;
    int col;
    int value;

    public Panel(int row, int col, int value) {
      this.row = row;
      this.col = col;
      this.value = value;
    }

    @Override
    public int compareTo(Panel other) {
      return Integer.compare(this.value, other.value);
    }
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int k = scanner.nextInt();
    char[][] grid = new char[GRID_SIZE][GRID_SIZE];
    ArrayList<Panel> panels = new ArrayList<>();

    for (int i = 0; i < GRID_SIZE; i++) {
      String row = scanner.next();
      for (int j = 0; j < GRID_SIZE; j++) {
        grid[i][j] = row.charAt(j);
        if (grid[i][j] != MARKER_DOT) {
          panels.add(new Panel(i, j, grid[i][j] - '0'));
        }
      }
    }

    Collections.sort(panels);

    System.out.println(canPressAllPanels(k, panels) ? "YES" : "NO");
  }

  private static boolean canPressAllPanels(int k, ArrayList<Panel> panels) {
    int numPanels = panels.size();
    int[] timeSlots = new int[MAX_PANEL_VALUE + 1];

    for (int i = 0; i < numPanels; i++) {
      Panel panel = panels.get(i);
      int value = panel.value;
      if (timeSlots[value] >= k) {
        return false;
      }
      timeSlots[value]++;
    }

    return true;
  }
}
