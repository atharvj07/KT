#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int sum = 0, mul = 1;
        vector<int> a(n);
        for (int &i : a) {
            cin >> i;
            sum += i;
            mul *= i;
        }
        int ans = 0;
        if (sum == 0) ans++;
        if (mul == 0) ans++;
        cout << ans << "\n";
    }
    return 0;
}
