import java.util.Scanner;

public class LiegesOfLegendre {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] piles = new int[n];
        for (int i = 0; i < n; i++) {
            piles[i] = scanner.nextInt();
        }

        // Check if any pile has an even size greater than 2
        boolean hasEvenPileGreaterThanTwo = false;
        for (int pile : piles) {
            if (pile % 2 == 0 && pile > 2) {
                hasEvenPileGreaterThanTwo = true;
                break;
            }
        }

        // Determine the winner based on the pile sizes
        if (hasEvenPileGreaterThanTwo) {
            System.out.println("Kevin");
        } else {
            System.out.println("Nicky");
        }
    }
}
