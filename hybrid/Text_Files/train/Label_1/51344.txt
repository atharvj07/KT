import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int w = 1;
            int h = 1;
            int[] left = new int[n];
            int[] up = new int[n];
            for (int i = 1; i < n; i++) {
                int p = sc.nextInt();
                int d = sc.nextInt();
                if (d == 0) {
                    left[i] = left[p] - 1;
                    up[i] = up[p];
                } else if (d == 1) {
                    left[i] = left[p];
                    up[i] = up[p] + 1;
                    h++;
                } else if (d == 2) {
                    left[i] = left[p] + 1;
                    up[i] = up[p];
                    w++;
                } else if (d == 3) {
                    left[i] = left[p];
                    up[i] = up[p] - 1;
                }
            }
            System.out.println(w + " " + h);
        }
        sc.close();
    }
}
