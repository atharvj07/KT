import sys

q, m = map(int, input().split())
spells = []
monsters = []

for _ in range(q):
    k, a, b = map(int, input().split())
    if k == 1:
        spells.append((a, b))
    if k == 2:
        monsters.append((a, b))

for t, h in monsters:
    for x, y in spells:
        if y <= m and t >= (h + x - 1) // x:
            print('YES')
            break
    else:
        print('NO')
