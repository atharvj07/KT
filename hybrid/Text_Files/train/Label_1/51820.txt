import java.util.Scanner;
import java.util.ArrayList;
import java.util.List;

@SuppressWarnings("unused")
public class Main {

    public static void main(String[] args) {
        try (Scanner sc = new Scanner(System.in)) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            int d = sc.nextInt();
            List<Integer> path1 = new ArrayList<>();
            List<Integer> path2 = new ArrayList<>();
            if (k % 2 != 0 || k > 2 * n - 2) {
                System.out.println(-1);
                return;
            }
            path1.add(a);
            path1.add(b);
            path2.add(c);
            path2.add(d);
            int steps1 = n - 1;
            int steps2 = n - 2;
            int steps = (k - 2 * n + 2) / 2;
            for (int i = n; i >= 1; i--) {
                if (i != a && i != b && steps1 > 0) {
                    path1.add(i);
                    steps1--;
                    if (steps1 == 0) {
                        for (int j = n; j >= 1; j--) {
                            if (j != a && j != b && j != path1.get(path1.size() - 1)) {
                                path1.add(j);
                            }
                        }
                    }
                }
                if (i != c && i != d && steps2 > 0) {
                    path2.add(i);
                    steps2--;
                    if (steps2 == 0) {
                        for (int j = n; j >= 1; j--) {
                            if (j != c && j != d && j != path2.get(path2.size() - 1)) {
                                path2.add(j);
                            }
                        }
                    }
                }
            }
            for (int i : path1) {
                System.out.print(i + " ");
            }
            System.out.println();
            for (int i : path2) {
                System.out.print(i + " ");
            }
            System.out.println();
        }
    }
}
