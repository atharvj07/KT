#include <bits/stdc++.h>
using namespace std;

using ll = long long;
const ll B = 31;
const ll X = 419;
const ll MOD = 1e9 + 9;

ll pw[500001], invPw[500001];

ll pwTimes(ll a, ll b){
    if(b == 0) return 1;
    if(b == 1) return a;
    ll res = pwTimes(a, b/2);
    res = (res * res) % MOD;
    if(b&1) res = (res * a) % MOD;
    return res;
}

void pre(){
    pw[0] = invPw[0] = 1;
    for(ll i = 1; i <= 500000; i++){
        pw[i] = (pw[i-1] * B) % MOD;
        invPw[i] = pwTimes(pw[i], MOD - 2);
    }
}

struct Hash{
    vector<ll> h, rh;
    void build(string s){
        h.assign(s.length() + 1, 0); rh.assign(s.length() + 1, 0);
        for(ll i = 0; i < s.length(); i++){
            h[i+1] = (h[i] * B + s[i] - 'a' + 1) % MOD;
            rh[i+1] = (rh[i] * B + s[s.length() - i - 1] - 'a' + 1) % MOD;
        }
    }
    ll get(ll l, ll r){
        return (h[r] - (h[l-1] * pw[r-l+1]) % MOD + MOD) % MOD;
    }
    ll getRev(ll l, ll r){
        return (rh[l] - (rh[r+1] * pw[r-l+1]) % MOD + MOD) % MOD;
    }
};

int main(){
    pre();
    string s; cin >> s;
    Hash hs, hsRev; hs.build(s);
    ll n = s.length();
    vector<pair<ll, ll>> palin;
    for(ll l = 1; l<=n; l++){
        for(ll r = l; r<=n; r++){
            if(hs.get(l, r) == hs.getRev(l, r)) palin.push_back({l, r});
        }
    }
    sort(palin.begin(), palin.end());
    vector<pair<ll, ll>> sameEnds;
    for(ll i = 0; i < palin.size();){
        ll j = i; while(j < palin.size() && palin[j].second == palin[i].second) j++;
        if(j - i >= 2) sameEnds.push_back(palin[i]);
        i = j;
    }
    ll m = palin.size() + sameEnds.size();
    cout << m << '\n';
    for(auto x: palin) cout << s.substr(x.first-1, x.second-x.first+1) << " ";
    for(auto x: sameEnds) cout << s.substr(x.first-1, x.second-x.first+1) << " ";
    cout << '\n';
}
