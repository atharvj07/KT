import java.util.Arrays;
import java.util.Scanner;

public class GoodVertexWeightedRootedBinaryTrees {

    private static final int MOD = 998244353;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        Arrays.sort(c);

        int[][] dp = new int[m + 1][n + 1];
        dp[0][0] = 1;
        for (int w = 1; w <= m; w++) {
            for (int j = 1; j <= n; j++) {
                for (int i = j - 1; i >= 0 && w - c[i] >= 0; i--) {
                    dp[w][j] = (dp[w][j] + dp[w - c[i]][i]) % MOD;
                }
            }
        }

        for (int w = 1; w <= m; w++) {
            int ans = 0;
            for (int j = 1; j <= n; j++) {
                ans = (ans + dp[w][j]) % MOD;
            }
            System.out.println(ans);
        }
    }
}
