from sys import stdin

n = int(stdin.readline())
s = [[int(x) for x in stdin.readline().split()] for i in range(n)]
m = max([max(ss) for ss in s])
cnt = [[0 for j in range(m + 1)] for i in range(n)]
ans = 0

for i in range(n):
	for v in s[i]:
		for j in range(v, m + 1):
			cnt[i][j] += 1

for i in range(n):
	for j in range(m + 1):
		cnt[i][j] += cnt[i - 1][j] if i else 0

for i in range(n):
	l = len(s[i])
	for v in s[i]:
		for j in range(i + 1, n):
			if v < s[j][0]:
				ans += cnt[i][v - 1] * cnt[j][s[j][0] - 1]
			elif v == s[j][0]:
				ans += cnt[i][v - 1] * (cnt[j][s[j][0] - 1] - 1)

print(ans)
