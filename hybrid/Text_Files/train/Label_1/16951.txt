#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int N = (int)3e5 + 5;
int x[N], y[N];
vector<vector<pair<int, int>>> adj(N);
int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int t;
    cin >> t;
    while (t--) {
        int n, m, k;
        cin >> n >> m >> k;
        for (int i = 0; i < n; i++) { cin >> x[i]; }
        for (int i = 0; i < m; i++) { cin >> y[i]; }
        for (int i = 0; i < k; i++) {
            int a, b;
            cin >> a >> b;
            int idx_x = upper_bound(x, x + n, a) - x - 1;
            int idx_y = upper_bound(y, y + m, b) - y - 1;
            // cout << idx_x << " " << idx_y << endl;
            adj[idx_x].push_back({idx_y, abs(x[idx_x] - a) + abs(y[idx_y] - b)});
            adj[idx_y].push_back({idx_x, abs(x[idx_x] - a) + abs(y[idx_y] - b)});
        }
        priority_queue<pair<int, pair<int, int>>> pq;
        vector<int> dist(N, INT_MAX);
        pq.push({0, {0, 0}});
        dist[0] = 0;
        while (!pq.empty()) {
            int u = pq.top().second.first;
            int len = -pq.top().first;
            int idx = pq.top().second.second;
            pq.pop();
            if (dist[u] != len) continue;
            for (auto v : adj[u]) {
                if (idx && v.first < idx - 1) continue;
                if (v.second + len < dist[v.first]) {
                    dist[v.first] = v.second + len;
                    pq.push({-v.second - len, {v.first, idx}});
                }
            }
        }
        ll ans = 0;
        for (int i = 0; i < N; i++) {
            for (auto v : adj[i]) { ans += (ll)(dist[i] + v.second > abs(x[i] - x[v.first]) + abs(y[i] - y[v.first])); }
        }
        cout << ans / 2 << endl;
        for (int i = 0; i < N; i++) { adj[i].clear(); }
    }
    return 0;
}
