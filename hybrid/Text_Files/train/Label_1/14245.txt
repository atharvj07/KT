#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 3001;
int dp[MAXN][MAXN];

int main() {
  int n, m;
  cin >> n >> m;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    int p, c;
    cin >> p >> c;
    v[i] = {c, p};
  }
  sort(v.begin(), v.end());
  for (int i = 0; i <= n; i++) {
    for (int j = 0; j <= m; j++) dp[i][j] = 1e9;
  }
  dp[0][0] = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j <= m; j++) {
      dp[i + 1][j] = min(dp[i + 1][j], dp[i][j]);
      dp[i + 1][v[i].second] = min(dp[i + 1][v[i].second], dp[i][j] + v[i].first);
    }
  }
  int ans = 1e9;
  for (int i = 2; i <= m; i++) ans = min(ans, dp[n][i]);
  cout << (ans == 1e9 ? -1 : ans) << endl;
}
