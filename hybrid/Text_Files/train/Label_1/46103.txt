import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < n; j++) {
                board[i][j] = line.charAt(j);
            }
        }
        boolean[][] used = new boolean[n][n];
        boolean[][][] visited = new boolean[n][n][4];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.' && !used[i][j]) {
                    used[i][j] = true;
                    if (dfs(board, used, visited, i, j, 0)) {
                        System.out.println("YES");
                        return;
                    } else {
                        used[i][j] = false;
                    }
                }
            }
        }
        System.out.println("NO");
    }

    private static boolean dfs(char[][] board, boolean[][] used, boolean[][][] visited, int x, int y, int dir) {
        if (x < 0 || y < 0 || x >= board.length || y >= board.length || board[x][y] == '#') {
            return false;
        }
        if (visited[x][y][dir]) {
            return false;
        }
        if (dir == 4) {
            return true;
        }
        visited[x][y][dir] = true;
        boolean result = false;
        switch (dir) {
            case 0:
                // right
                result = dfs(board, used, visited, x, y + 1, 0) || dfs(board, used, visited, x + 1, y, 1);
                break;
            case 1:
                // down
                result = dfs(board, used, visited, x + 1, y, 1) || dfs(board, used, visited, x, y - 1, 2);
                break;
            case 2:
                // left
                result = dfs(board, used, visited, x, y - 1, 2) || dfs(board, used, visited, x - 1, y, 3);
                break;
            case 3:
                // up
                result = dfs(board, used, visited, x - 1, y, 3) || dfs(board, used, visited, x, y + 1, 0);
                break;
        }
        if (!result) {
            visited[x][y][dir] = false;
        }
        return result;
    }
}
