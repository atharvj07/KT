#include <bits/stdc++.h>
using namespace std;

bitset<1001> a[1001];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < 7; j++) { a[i][j] = (s[j] == '1'); }
  }
  int ans = 0;
  for (int i = 0; i < 7; i++) {
    int cnt = 0;
    for (int j = 1; j <= n; j++) { cnt += a[j][i]; }
    ans = max(ans, cnt);
  }
  cout << ans << endl;
}
