#include <bits/stdc++.h>
using namespace std;

int main() {
    int q;
    cin >> q;
    while(q--) {
        int n, m, ans = 0;
        cin >> n >> m;
        vector<bool> row(n, 0), col(m, 0);
        for(int i = 0; i < n; i++) {
            string s;
            cin >> s;
            for(int j = 0; j < m; j++) {
                if(s[j] == '*') {
                    row[i] = 1;
                    col[j] = 1;
                }
            }
        }
        for(int i = 0; i < n; i++)
            ans += (row[i] == 0);
        for(int i = 0; i < m; i++)
            ans += (col[i] == 0);
        cout << ans << '\n';
    }
}
