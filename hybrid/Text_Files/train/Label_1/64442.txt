n, T = map(int, input().split())
a = list(map(int, input().split()))
t = list(map(int, input().split()))
 
below = []
above = []
 
for i in range(n):
    if t[i] < T:
        below.append((t[i], a[i]))
    elif t[i] > T:
        above.append((t[i], a[i]))
    else:
        below.append((t[i], a[i]))
        above.append((t[i], a[i]))
 
below.sort(reverse=True)
above.sort()
 
sum_below = sum(a[i] for i, _ in enumerate(below))
sum_above = sum(a[i] for i, _ in enumerate(above))
 
def calc(temp, T, sum_other):
    water = 0
    for t, a in temp:
        if t == T:
            water += a
        else:
            req = sum_other * abs(T - t)
            if a * abs(T - t) <= req:
                water += a
                sum_other -= a * abs(T - t) / abs(T - t)
            else:
                water += req / abs(T - t)
                break
    return water
 
result = 0
if sum_below < sum_above:
    result = calc(above, T, sum_below)
else:
    result = calc(below, T, sum_above)
 
print(f'{result:.15f}')
