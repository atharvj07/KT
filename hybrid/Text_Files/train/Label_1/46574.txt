import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        int sum = 0;
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            sum += a[i];
        }
        int count = 0;
        int index = 0;
        for (int i = 0; i < n; i++) {
            count += a[i];
            if (count * 2 > sum) {
                index = i + 1;
                break;
            }
        }
        System.out.println(index);
    }
}
