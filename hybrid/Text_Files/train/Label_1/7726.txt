#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    ios::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
    int n, m;
    cin >> n >> m;
    vector<int> need(n), when(n, 2e9);
    for (int &x : need) cin >> x;
    for (int i = 0; i < m; i++) {
        int x, y;
        cin >> x >> y;
        when[y - 1] = min(when[y - 1], x);
    }
    int ans = 0;
    for (int i = 0; i < n; i++)
        ans = max(ans, (need[i] + when[i] - 1) / when[i]);
    cout << ans;
    return 0;
}
