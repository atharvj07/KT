import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int y0 = sc.nextInt();
        int y1 = sc.nextInt();
        int[] x1 = new int[n];
        for (int i = 0; i < n; i++) {
            x1[i] = sc.nextInt();
        }
        int[] x2 = new int[m];
        for (int i = 0; i < m; i++) {
            x2[i] = sc.nextInt();
        }
        int[] dy = new int[n];
        for (int i = 0; i < n; i++) {
            dy[i] = Math.abs(y1 - y0) + Math.abs(x1[i] - x2[0]);
            for (int j = 1; j < m; j++) {
                int cur = Math.abs(y1 - y0) + Math.abs(x1[i] - x2[j]);
                if (cur < dy[i]) {
                    dy[i] = cur;
                }
            }
        }
        Arrays.sort(dy);
        int ans = n;
        for (int i = 0; i < m; i++) {
            if (dy[i] > dy[m - 1]) {
                break;
            }
            ans--;
        }
        System.out.println(ans);
    }
}
