#include <bits/stdc++.h>

using namespace std;

const int N = 105;
int n, ans[N], deg[N];
string s[N];
vector < int > g[N];
bool mark[N];

int main() {
  ios_base::sync_with_stdio(false); cin.tie(nullptr); cout.tie(nullptr);
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> s[i];
    for (int j = 1; j <= n; j++) {
      if (i == j) continue;
      if (s[i] == s[j]) {
        g[i].push_back(j);
        g[j].push_back(i);
        deg[i]++; deg[j]++;
        continue;
      }
      bool ok = false;
      for (int x = 0; x < (int)s[j].size() - (int)s[i].size() + 1; x++) {
        if (s[i] == s[j].substr(x, (int)s[i].size())) {
          ok = true;
          g[i].push_back(j);
          deg[i]++; deg[j]--;
          break;
        }
      }
      if (ok) {
        reverse(s[i].begin(), s[i].end());
        for (int x = 0; x < (int)s[j].size() - (int)s[i].size() + 1; x++) {
          if (s[i] == s[j].substr(x, (int)s[i].size())) {
            g[j].push_back(i);
            deg[j]++; deg[i]--;
            break;
          }
        }
        reverse(s[i].begin(), s[i].end());
      }
    }
  }
  for (int i = 1; i <= n; i++)
    if (!deg[i]) {
      queue < int > q;
      q.push(i);
      while (!q.empty()) {
        int u = q.front(); q.pop();
        mark[u] = true;
        for (int j = 0; j < (int)g[u].size(); j++) {
          int v = g[u][j];
          deg[v]--;
          if (deg[v] == 0) q.push(v);
        }
      }
    }
  bool f = false;
  for (int i = 1; i <= n; i++) f |= mark[i];
  if (!f) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  for (int i = 1; i <= n; i++)
    if (mark[i]) {
      cout << s[i] << endl;
      ans[i] = 1;
    }
  return 0;
}
