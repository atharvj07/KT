#include <iostream>
#include <unordered_map>
#include <string>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    string s[n], q[m];
    for (int i = 0; i < n; i++) { cin >> s[i]; }
    for (int i = 0; i < m; i++) { cin >> q[i]; }
    unordered_map<int, int> cnt;
    for (int i = 0; i < n; i++) {
        int x = 0;
        for (int j = 0; j < s[i].size(); j++) { x |= (1 << (s[i][j] - 'a')); }
        cnt[x]++;
    }
    for (int i = 0; i < m; i++) {
        int x = 0;
        for (int j = 0; j < q[i].size(); j++) { x |= (1 << (q[i][j] - 'a')); }
        for (int j = 0; j < 26; j++) {
            if (cnt[x ^ (1 << j)]) {
                cout << "YES" << endl;
                goto ok;
            }
        }
        cout << "NO" << endl;
    ok:;
    }
    return 0;
}
