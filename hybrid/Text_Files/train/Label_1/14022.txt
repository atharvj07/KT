#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, w;
  cin >> n >> w;
  vector<int> a(2 * n);
  for (int i = 0; i < 2 * n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  int l = 0, r = 2 * n - 1;
  double ans = 0;
  while (l < r) {
    if ((w - a[l] - a[r]) >= 0) {
      ans += (a[l] + a[r]) / 3.0;
      l++;
      r--;
      w -= (a[l] + a[r]);
    } else {
      r--;
    }
  }
  cout << ans << endl;
  return 0;
}
