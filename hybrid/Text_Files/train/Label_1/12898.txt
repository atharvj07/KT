#include<bits/stdc++.h>

using namespace std;
#define vi(x) vector<x>

typedef pair<int, int> pi;

const pi invalid = {-1, -1};

vi(int) rep(int n, int start = 0) { return vi(int)(start, start + n); }

int get(int i, vi(int) &arr) {
  if (i < (int)arr.size()) return arr[i];
  return -1;
}

int main() {
  int n, d, h;
  cin >> n >> d >> h;
  if (h == 1) {
    cout << "-1" << endl;
    return 0;
  }
  if (d == h) {
    cout << "-1" << endl;
    return 0;
  }
  if ((d - h) % 2 != 0) {
    cout << "-1" << endl;
    return 0;
  }
  int a = (d - h) / 2;
  int b = h;
  if (b > a + 1) {
    cout << "-1" << endl;
    return 0;
  }
  int sz = n - 1;
  vi(int) adj[sz + 5];
  for (int i = 0; i < sz; i++) {
    adj[i].push_back(i + 1);
    adj[i + 1].push_back(i);
  }
  int c = b;
  b = a;
  a = c;
  while (b > 0) {
    adj[0].push_back(sz + 1);
    adj[sz + 1].push_back(0);
    b--;
    sz++;
  }
  while (a > 0) {
    adj[0].push_back(sz + 1);
    adj[sz + 1].push_back(0);
    adj[sz + 1].push_back(sz + 2);
    adj[sz + 2].push_back(sz + 1);
    a--;
    sz += 2;
  }
  for (int i = 0; i < sz; i++) {
    for (auto u : adj[i]) {
      if (u > i) cout << i + 1 << " " << u + 1 << endl;
    }
  }
  return 0;
}
