import sys
from typing import List, Tuple

input = sys.stdin.readline

n, m, k = map(int, input().split())
w = list(map(int, input().split()))
match = []

for _ in range(m):
    a, b = map(int, input().split())
    a -= 1
    b -= 1
    match.append((a, b))

ans = [0] * m
s = [[0] * k for _ in range(n)]

for l, (a, b) in enumerate(match):
    if abs(max(s[a]) - min(s[a])) > 2 or abs(max(s[b]) - min(s[b])) > 2:
        continue
    max_stadiums = []
    min_stadiums = []
    for i in range(k):
        if s[a][i] == max(s[a]):
            max_stadiums.append(i)
        if s[b][i] == max(s[b]):
            max_stadiums.append(i)
        if s[a][i] == min(s[a]):
            min_stadiums.append(i)
        if s[b][i] == min(s[b]):
            min_stadiums.append(i)

    common = set(max_stadiums) & set(min_stadiums)
    if len(common) > 0:
        ans[l] = list(common)[0] + 1
        s[a][ans[l] - 1] += 1
        s[b][ans[l] - 1] += 1
    else:
        if len(max_stadiums) > len(min_stadiums):
            min_s = min_stadiums[0]
            s[a][min_s] += 1
            s[b][min_s] += 1
            ans[l] = min_s + 1
        else:
            max_s = max_stadiums[0]
            s[a][max_s] += 1
            s[b][max_s] += 1
            ans[l] = max_s + 1

print(*ans)
