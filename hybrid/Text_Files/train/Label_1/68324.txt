T = int(input())
for _ in range(T):
    n, g, b = map(int, input().split())
    high_quality_needed = (n + 1) // 2
    full_cycles_needed = high_quality_needed // g
    last_cycle_good_days = high_quality_needed % g
    
    if last_cycle_good_days == 0:
        days_needed = full_cycles_needed * (g + b) - b
    else:
        days_needed = full_cycles_needed * (g + b) + last_cycle_good_days
        
    print(max(n, days_needed))
