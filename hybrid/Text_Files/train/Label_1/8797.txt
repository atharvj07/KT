#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<ll> a(n);
        for (ll &i : a) cin >> i;
        sort(a.begin(), a.end());
        ll sum = 0, cnt = 0;
        for (ll i = n - 1; i >= 0; i--) {
            sum += a[i];
            if (sum >= x) {
                cnt += n - i;
                break;
            }
        }
        cout << cnt << '\n';
    }
    return 0;
}
