for _ in range(int(input())):
    n, k, d = map(int, input().split())
    a = list(map(int, input().split()))
    st = set()
    pref = [0] * (n + 1)
    for i in range(n):
        st.add(a[i])
        pref[i + 1] = pref[i]
        if len(st) >= d:
            pref[i + 1] -= 1
        st.remove(a[i])
    print(k - min(pref[len(a) - d + 1 : len(a) + 1]))
