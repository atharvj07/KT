#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll N = 2010;

ll n, l[N], r[N];
ll a[N], b[N];
bool used[N];

int main() {
    cin >> n;
    for(ll i = 1; i <= n; i++) cin >> l[i];
    for(ll i = 1; i <= n; i++) cin >> r[i];

    for(ll i = 1; i <= n; i++) {
        if(l[i] + r[i] + 1 > n) {
            cout << "NO";
            return 0;
        }
    }

    cout << "YES\n";

    for(ll i = n; i >= 1; i--) {
        ll sum = n - l[i] - r[i] - 1;
        while(b[sum] && sum > 0) sum--;
        a[i] = sum;
        b[sum] = 1;
    }

    for(ll i = 1; i <= n; i++) cout << a[i] << " ";

    return 0;
}
