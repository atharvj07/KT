#include <vector>
#include <cstdio>
#include <algorithm>

using namespace std;

int main(){
    int t;
    scanf("%d",&t);
    for(int i = 0; i < t; i++){
        int m,k;
        scanf("%d%d",&m,&k);
        vector<int> a(k+1),b(k+1);
        for(int j = 1; j <= k; j++)
            scanf("%d",&a[j]);
        for(int j = 1; j < m; j++){
            int x, y;
            scanf("%d%d",&x,&y);
            if(y == 0){
                b[x]++;
            }
            else if(a[x] - b[x] <= 0)
                a[0]++;
        }
        for(int j = 1; j <= k; j++){
            if(a[j] <= b[j])
                printf("Y");
            else
                printf("N");
        }
        printf("\n");
    }
    return 0;
}
