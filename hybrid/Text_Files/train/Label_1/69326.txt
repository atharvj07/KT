import datetime

n = int(input())
max_d = 0
events = []
for i in range(n):
    m, d, p, t = map(int, input().split())
    b = datetime.datetime(2013, m, d) - datetime.timedelta(days=t)
    e = datetime.datetime(2013, m, d)
    max_d = max(max_d, e.timetuple().tm_yday)
    events.append((b, e, p))

days = [0] * max_d
for b, e, p in events:
    for j in range(b.timetuple().tm_yday, e.timetuple().tm_yday):
        days[j] += p

print(max(days))
