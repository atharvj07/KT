#include <bits/stdc++.h>

using namespace std;

const int N = 2e5 + 5;

int n, r[N], c[N];
long long ans;

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> r[i] >> c[i];
  }
  for (int i = 0; i < n; i++) {
    ans += r[i] - 1;
    ans += c[i] - 1;
  }
  cout << ans << endl;
  return 0;
}
