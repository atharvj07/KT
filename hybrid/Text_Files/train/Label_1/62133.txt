import sys
input = sys.stdin.readline

mod = 10**9+7

def mpow(x,n):
    ret=1
    while n>0:
        if n%2==1: ret=ret*x%mod
        x=x*x%mod
        n>>=1
    return ret

k=int(input())
n=int(input())
clrs={'white':1,'yellow':2,'green':3,'blue':4,'red':5,'orange':6}
a=[0]*(2**k)
for _ in range(n):
    v,c=input().split()
    v=int(v)
    a[v]=clrs[c]

def dfs(v,prv):
    if v>=(1<<(k-1)): return [1]*(clrs['white']+1)
    ret=[0]*(clrs['white']+1)
    for i in range((a[v]==0 and clrs['white']) or a[v],(a[v]==0 and clrs['white']+1) or a[v]+1):
        if prv==0 or i!=prv and (abs(prv-i)!=1 or (v>>1)&1):
            nxt=[0]*(clrs['white']+1)
            for j in range(1,clrs['white']+1):
                if prv==0 or j!=prv and (abs(j-i)!=1 or (v>>1)&1):
                    tmp=dfs(v*2,i)
                    for c in range(clrs['white']+1):
                        nxt[(c+j)%7]=(nxt[(c+j)%7]+tmp[c])%mod
            for j in range(clrs['white']+1):
                ret[j]=(ret[j]+nxt[j])%mod
    return ret

ans=dfs(1,0)[0]
print(ans)
