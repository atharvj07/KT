from collections import Counter

n, m = map(int, input().split())
g = []
for i in range(n):
    _, *g_i = map(int, input().split())
    g.append(g_i)

c = [Counter(x) for x in g]
f = [0] * m
for d in c:
    for j in d:
        f[j-1] += d[j]
ans = 1
for j in f:
    ans = (ans * j) % (10**9 + 7)

print(ans)
