import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int w = sc.nextInt();
        int b = sc.nextInt();
        System.out.printf("%.9f", (w * 1.0 / (w + b)) * (w * 1.0 / (w + b - 1)) * (w * 1.0 / (w + b - 2)));
    }
}
