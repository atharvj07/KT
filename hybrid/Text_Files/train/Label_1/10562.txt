#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define ull unsigned long long
int main() {
    ll n,m,q; cin>>n>>m>>q;
    map<ll,ll>vis;
    ll cur=1;
    for(ll i=1;i<=n;++i) {
        ll x; cin>>x;
        vis[x]++;
        cur=x;
    }
    for(ll i=1;i<=q;++i) {
        ll x; cin>>x;
        while(vis[cur]==((x-n)/m+1)) {
            cur++;
            if(cur>m) cur=1;
        }
        cout<<cur<<endl;
    }
}
