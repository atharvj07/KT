from math import ceil, log2

n, m = map(int, input().split())

height = []
for _ in range(n):
    a = list(map(int, input().split()))
    height.append(a)

x = [[0] * m for _ in range(n)]

for i in range(n):
    a = height[i]
    for j in range(m):
        l = a[j]
        r = []
        for k in range(i + 1, n):
            if a[j] > height[k][j]:
                r.append(height[k][j])
        for k in range(j + 1, m):
            if a[j] > height[i][k]:
                r.append(height[i][k])
        x[i][j] = 1
        for h in r:
            t = h - l + 1
            while t > 1:
                t = max(ceil(t / 2), 1)
                x[i][j] += 1
        x[i][j] = max(x[i][j], log2(l) + 1)

for i in x:
    print(*i)
