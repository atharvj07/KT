#include <iostream>
#include <vector>
#include <tuple>
#include <algorithm>

using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<tuple<int, int>> scores(n);
    for (int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        scores[i] = make_tuple(a, b);
    }
    int mx = 0;
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        int a, b;
        tie(a, b) = scores[i];
        if (a == b)
            cnt++;
        else
            cnt = 0;
        mx = max(mx, cnt);
    }
    cout << mx + 1 << endl;
    return 0;
}
