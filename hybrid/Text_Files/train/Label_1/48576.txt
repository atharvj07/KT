import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = sc.nextInt();
        }
        sc.close();
        if (n != m) {
            System.out.println(-1);
            return;
        }
        long sumA = 0, sumB = 0;
        for (int i = 0; i < n; i++) {
            sumA += a[i];
            sumB += b[i];
        }
        if (sumA != sumB) {
            System.out.println(-1);
            return;
        }
        int i = 0, j = 0, l = 0;
        while (i < n && j < m) {
            if (a[i] == b[j]) {
                l++;
                i++;
                j++;
            } else {
                l = 0;
                i++;
            }
        }
        System.out.println(l);
    }
}
