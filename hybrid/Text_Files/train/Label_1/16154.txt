#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<bool> used(n, false);
    string res;
    int cnt = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (k == 0) break;
        if (s[i] == ')' && !used[i]) {
            res += ')';
            used[i] = true;
            k--;
        } else if (s[i] == '(' && cnt < k / 2) {
            res = '(' + res;
            used[i] = true;
            k--;
            cnt++;
        }
    }
    for (int i = 0; i < k; i++) res = '(' + res + ')';
    cout << res;
    return 0;
}
