def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def lcm(a, b):
    return a * b // gcd(a, b)

n, m, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

cycle = lcm(n, m)
diffs = 0
for i in range(1, cycle + 1):
    if a[(i-1) % n] != b[(i-1) % m]:
        diffs += 1
    if diffs == k:
        print(i)
        break
else:
    full_cycles = (k - 1) // diffs
    remaining_diffs = (k - 1) % diffs + 1
    found_diffs = 0
    for i in range(1, cycle + 1):
        if a[(i-1) % n] != b[(i-1) % m]:
            found_diffs += 1
            if found_diffs == remaining_diffs:
                print(full_cycles * cycle + i)
                break
