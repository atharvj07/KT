#include <bits/stdc++.h>
using namespace std;
#define int long long

int32_t main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> dif(n - 1);
    for (int i = 0; i < n - 1; i++) dif[i] = a[i + 1] - a[i];
    sort(dif.rbegin(), dif.rend());
    int ans = 0;
    for (int i = 0; i < k - 1; i++) ans += dif[i];
    cout << ans << '\n';
}
