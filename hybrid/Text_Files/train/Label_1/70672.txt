import sys
input=sys.stdin.readline
for _ in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    b=[0]*n
    for i in range(1,n+1):
        b[a[i-1]-1]=i
    j=0
    ans=[0]*n
    for i in range(1,n+1):
        for k in range(j,n):
            if b[k]==i:
                ans[k]=i-j
                j+=1
                break
    print(max(ans))
