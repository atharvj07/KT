import java.util.*;
import java.io.*;

class Main {
    static class Base {
        int x, d, g;
    }
    static class Spaceship {
        int x, a, f, p;
    }

    static int[] dist;
    static int[] vis;
    static List<Integer>[] adj;

    static int bfs(int n, int s, int[] dist) {
        Queue<Integer> q = new LinkedList<>();
        q.add(s);
        vis[s] = 1;
        dist[s] = 0;
        while (!q.isEmpty()) {
            int u =q.poll();
            for (int v : adj[u]) {
                if (vis[v] == 0) {
                    vis[v] = 1;
                    dist[v] = dist[u] + 1;
                    q.add(v);
                }
            }
        }
        int mx = 0;
        for (int i = 1; i <= n; i++) {
            if (dist[i] > mx) {
                mx = dist[i];
            }
        }
        return mx;
    }

    public static void main(String[] args) throws Exception {
        int n = scan.nextInt();
        int m = scan.nextInt();
        adj = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        while (m-- > 0) {
            int u = scan.nextInt();
            int v = scan.nextInt();
            adj[u].add(v);
            adj[v].add(u);
        }
        int s = scan.nextInt();
        int b = scan.nextInt();
        int k = scan.nextInt();
        Spaceship[] spaceships = new Spaceship[s + 1];
        for (int i = 1; i <= s; i++) {
            spaceships[i] = new Spaceship();
            spaceships[i].x = scan.nextInt();
            spaceships[i].a = scan.nextInt();
            spaceships[i].f = scan.nextInt();
            spaceships[i].p = scan.nextInt();
        }
        Base[] bases = new Base[b + 1];
        for (int i = 1; i <= b; i++) {
            bases[i] = new Base();
            bases[i].x = scan.nextInt();
            bases[i].d = scan.nextInt();
            bases[i].g = scan.nextInt();
        }
        boolean[] dep = new boolean[s + 1];
        for (int i = 0; i < k; i++) {
            int x = scan.nextInt();
            int y = scan.nextInt();
            dep[x] = true;
            if (y > x) {
                dep[y] = true;
            }
        }
        long ans = 0;
        for (int i = 1; i <= s; i++) {
            if (dep[i] == false) {
                dist = new int[n + 1];
                vis = new int[n + 1];
                int mx = bfs(n, spaceships[i].x, dist);
                if (mx <= spaceships[i].f) {
                    ans += bases[spaceships[i].x].g - spaceships[i].p;
                }
            }
        }
        out.println(ans);
    }

    static Scanner scan = new Scanner(System.in);
    static PrintStream out = new PrintStream(System.out);
}
