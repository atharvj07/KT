#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
string s, t; int n;
int main() {
    cin >> n;
    getline(cin, s);
    getline(cin, s);
    if (n == 1) { cout << s.size(); return 0; }
    int siz = s.size();
    for (int i = 0; i < siz - 1; i++) {
        int j = i + 1;
        while (j < siz && s[j] == s[j - 1]) j++;
        if (j - i > 2) t += s[i], i = j - 1;
        else t += s[i];
    }
    t += s[siz - 1];
    cout << t.size();
    return 0;
}
