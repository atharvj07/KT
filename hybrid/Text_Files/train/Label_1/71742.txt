def checkGood(x, powers):
    for pow in powers:
        if pow <= x:
            x -= pow
        else:
            return x == 0

def rec(n):
    if checkGood(n, powers):
        return n
    a = 0
    while 3**a <= n:
        a += 1
    a -= 1
    return rec(n - 3**a) + 3**a

powers = [3**i for i in range(31)]

for _ in range(int(input())):
    print(rec(int(input())))
