#include <bits/stdc++.h>
using namespace std;

const int MOD = 1e9 + 7;
const int MAX_N = 200001;
long long dp[201][201];

long long fastpow(long long base, long long pow) {
    if (pow == 0) return 1;
    long long x = fastpow(base, pow / 2);
    x = x * x % MOD;
    if (pow % 2) x = x * base % MOD;
    return x;
}

int main() {
    int n; cin >> n;
    vector<int> primes(n);
    for (int i = 0; i < n; i++) cin >> primes[i];
    dp[0][0] = 1;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= i; j++) {
            dp[i][j] = dp[i - 1][j];
            if (j) dp[i][j] = (dp[i][j] + dp[i - 1][j - 1] * primes[i - 1]) % MOD;
        }
    }
    cout << dp[n][n] << endl;
    return 0;
}
