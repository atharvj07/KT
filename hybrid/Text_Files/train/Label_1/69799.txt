import sys
from collections import defaultdict

input = sys.stdin.readline

n, s = map(int, input().split())
book = defaultdict(list)

for _ in range(n):
    d, p, q = input().split()
    book[d].append((int(p), int(q)))

for d in ['S', 'B']:
    for p, q in sorted(book[d], reverse=(d == 'S')):
        print(d, p, q)
        s -= 1
        if not s:
            break
