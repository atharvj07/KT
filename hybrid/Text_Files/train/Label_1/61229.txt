n, T, c = map(float, input().split())
a = list(map(int, input().split()))
m = int(input())
p = list(map(int, input().split()))

def real_mean(a, T, p_j):
    return sum(a[p_j-int(T):p_j]) / T

mean = 0.0
means = []
for t in range(1, int(n)+1):
    mean = (mean + a[t-1] / T) / c
    if t in p:
        means.append(mean)

for p_j in p:
    real = real_mean(a, T, p_j)
    approx = means.pop(0)
    error = abs(approx-real) / real
    print(f"{real:.5f} {approx:.5f} {error:.5f}")
