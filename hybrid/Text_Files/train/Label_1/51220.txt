import java.util.*;

class Main {
    static int n;
    static int arr[];
    static int dp[];
    static boolean visit[];
    static int solve(int i) {
        if (visit[i])
            return dp[i];
        visit[i] = true;
        int ans = Integer.MAX_VALUE;
        for (int j = 1; j <= n; j++) {
            if (i == j) {
                ans = Math.min(ans, arr[i]);
            } else {
                ans = Math.min(ans, arr[i] + solve(j));
            }
        }
        dp[i] = ans;
        return dp[i];
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        arr = new int[n + 2];
        dp = new int[n + 2];
        visit = new boolean[n + 2];
        for (int i = 1; i <= n; i++) {
            arr[i] = in.nextInt();
        }
        for (int i = 1; i <= n; i++) {
            arr[in.nextInt()] = i;
        }
        int ans = Integer.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            ans = Math.min(ans, solve(i));
        }
        System.out.println(ans);
    }
}
