import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[2 * n];
        int[] b = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < 2 * n; i++) {
            b[i] = sc.nextInt();
        }
        int[][] dp = new int[2 * n + 1][2];
        for (int i = 2 * n - 1; i >= 0; i--) {
            dp[i][0] = Math.max(dp[i + 1][0] + a[i], dp[i + 1][1] + b[i]);
            dp[i][1] = Math.max(dp[i + 1][0] + b[i], dp[i + 1][1] + a[i]);
        }
        while (q-- > 0) {
            int p = sc.nextInt() - 1;
            int x = sc.nextInt();
            int y = sc.nextInt();
            dp[p][0] = Math.max(dp[p + 1][0] + x, dp[p + 1][1] + y);
            dp[p][1] = Math.max(dp[p + 1][0] + y, dp[p + 1][1] + x);
            System.out.println(dp[0][0]);
        }
    }
}
