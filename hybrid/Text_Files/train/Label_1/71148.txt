import sys
s = sys.stdin.readline()[:-1]
if 'X' in s:
    xcnt = s.count('X')
    for i in range(10):
        ns = s.replace('X', str(i))
        if ns[0] == '0' or (len(ns) > 1 and ns[-2:] != '00' and int(ns[-2:]) % 25 != 0):
            xcnt -= 1
    print(9 ** xcnt)
else:
    print(int(s) % 25 == 0)
