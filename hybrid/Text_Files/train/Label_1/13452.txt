#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, l, r, s;
    cin >> n >> l >> r >> s;
    vector<int> p(n);
    for (int i = l; i <= r; i++) p[i - 1] = i;
    int sum = s - (r - l + 1);
    int lpos = l - 1, rpos = r;
    for (int i = 0; i < l - 1; i++) {
      if (sum >= 2) {
        p[i] = sum / 2;
        sum -= sum / 2;
      }
    }
    if (sum) {
      p[rpos--] = sum;
      sum = 0;
    }
    for (int i = n - 1; i >= r + 1; i--) {
      if (sum >= 2) {
        p[i] = sum / 2;
        sum -= sum / 2;
      }
    }
    if (sum) {
      p[lpos++] = sum;
    }
    bool ok = true;
    for (int i = 0; i < n - 1; i++) {
      if (p[i] >= p[i + 1]) {
        ok = false;
        break;
      }
    }
    if (ok) {
      for (int i = 0; i < n; i++) cout << p[i] << " ";
      cout << endl;
    } else {
      cout << -1 << endl;
    }
  }
  return 0;
}
