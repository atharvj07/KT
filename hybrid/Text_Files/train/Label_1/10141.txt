#include<iostream>
#include<queue>
using namespace std;
int n, m, k, h[110], vis[110][110];
bool check(int x) {
    for(int i = 1; i <= n; i++) {
        if(h[i] - x > m)return false;
    }
    return true;
}
int main() {
    int t; cin >> t;
    while(t--) {
        cin >> n >> m >> k;
        for(int i = 1; i <= n; i++) {
            cin >> h[i];
        }
        if(check(k)) {cout << "YES" << endl; continue;}
        queue < pair<int, int> > q; q.push(make_pair(1, m)); vis[1][m] = 1;
        while(!q.empty()) {
            int u = q.front().first, w = q.front().second; q.pop();
            if(u == n) {cout << "YES" << endl; goto end;}
            if(w > 0) {
                if(!vis[u+1][w+h[u+1]-h[u]]) {
                    vis[u+1][w+h[u+1]-h[u]] = 1;
                    q.push(make_pair(u+1, w+h[u+1]-h[u]));
                }
                if(!vis[u][w+h[u]-h[u-1]]) {
                    vis[u][w+h[u]-h[u-1]] = 1;
                    q.push(make_pair(u, w+h[u]-h[u-1]));
                }
                if(!vis[u][w-1]) {
                    vis[u][w-1] = 1;
                    q.push(make_pair(u, w-1));
                }
            }
        }
        cout << "NO" << endl;
        end:;
    }
    return 0;
}
