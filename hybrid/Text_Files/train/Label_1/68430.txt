n, m = map(int, input().split())
files = [input() for _ in range(n)]
to_delete = [int(input()) - 1 for _ in range(m)]

# Check lengths of files to delete
len_check = len(files[to_delete[0]])
if any(len(files[i]) != len_check for i in to_delete):
    print("No")
else:
    pattern = list(files[to_delete[0]])
    for i in to_delete:
        for j in range(len_check):
            if files[i][j] != pattern[j]:
                pattern[j] = '?'
    
    # Check if pattern matches only files to delete
    matches_only_delete = True
    for i in range(n):
        if i in to_delete:
            continue
        if all(c1 == c2 or c2 == '?' for c1, c2 in zip(files[i], pattern)):
            matches_only_delete = False
            break
            
    if matches_only_delete:
        print("Yes")
        print(''.join(pattern))
    else:
        print("No")
