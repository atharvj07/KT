import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int p = sc.nextInt();
        int[] t = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            t[i] = sc.nextInt();
        }
        int[] ans = new int[n + 1];
        int cur = 0, head = 1;
        for (int i = 1; i <= n; i++) {
            if (t[i] < cur) continue;
            while (head < i && t[head] <= cur) head++;
            ans[i] = cur = t[head] + p;
        }
        for (int i = 1; i <= n; i++) {
            System.out.println(ans[i]);
        }
    }
}
