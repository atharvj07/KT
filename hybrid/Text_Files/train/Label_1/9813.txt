#include <iostream>
#include <vector>
#include <algorithm>
int main() {
    int n, x, y;
    std::cin >> n >> x >> y;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    std::vector<int> l(n), r(n);
    int cur = -1 << 30;
    for (int i = 0; i < n; i++) {
        l[i] = cur;
        cur = a[i];
    }
    cur = 1 << 30;
    for (int i = n - 1; i >= 0; i--) {
        r[i] = cur;
        cur = a[i];
    }
    for (int i = x; i < n - y; i++) {
        if (a[i] < l[i] && a[i] < r[i]) {
            std::cout << i + 1 << std::endl;
            return 0;
        }
    }
    return 0;
}
