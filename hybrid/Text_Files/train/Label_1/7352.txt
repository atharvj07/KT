#include <bits/stdc++.h>
using namespace std;

bool cmp(const pair<int, int> &a, const pair<int, int> &b) 
{
    return a.first < b.first || (a.first == b.first && a.second < b.second);
}

void solve() 
{
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (auto &x : v)
        cin >> x.first >> x.second;
    sort(v.begin(), v.end(), cmp);
    string ans;
    int x = 0, y = 0;
    for (auto &p : v) {
        while (x < p.first) {
            ans += 'R';
            x++;
        }
        while (y < p.second) {
            ans += 'U';
            y++;
        }
    }
    if (x == v.back().first && y == v.back().second)
        cout << "YES\n" << ans << '\n';
    else
        cout << "NO\n";
}
int main() 
{
    ios::sync_with_stdio(0);
    cin.tie(0);
    int t;
    cin >> t;
    while (t--)
        solve();
}
