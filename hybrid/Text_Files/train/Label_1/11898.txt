#include <iostream>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        long long ans = 0LL, a, b;
        std::cin >> a >> b;
        for (long long i = 1, j = 1; i < a + 1; i = 10 * i + 1, j = 10 * j) {
            for (long long k = 1; k < b + 1; k++) {
                long long ab = i * k + i + k;
                long long c = i * j + k;
                if (ab == c) ans++;
            }
        }
        std::cout << ans << '\n';
    }
    return 0;
}
