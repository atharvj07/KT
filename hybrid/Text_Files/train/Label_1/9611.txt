#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int n, s;
vector<vector<int>> g;
vector<bool> used;
vector<int> dist, p;

void dfs(int v, int pr, int d) {
    dist[v] = d;
    p[v] = pr;
    used[v] = true;
    for (int u : g[v]) {
        if (used[u]) continue;
        dfs(u, v, d + 1);
    }
}

int main() {
    cin >> n >> s;
    g.resize(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        g[a].push_back(b);
        g[b].push_back(a);
    }

    used.resize(n + 1, false);
    dist.resize(n + 1);
    p.resize(n + 1);

    dfs(1, 0, 0);

    int v1 = -1;
    for (int i = 1; i <= n; i++) {
        if (dist[i] > dist[v1]) v1 = i;
    }

    dfs(v1, 0, 0);

    int v2 = -1;
    for (int i = 1; i <= n; i++) {
        if (dist[i] > dist[v2]) v2 = i;
    }

    int diam = dist[v2];
    double ans = double(s) / (diam - 1);
    cout << ans << endl;
    return 0;
}
