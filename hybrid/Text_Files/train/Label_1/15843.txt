#include <iostream>
#include <vector>
using namespace std;

int main() {
  vector<vector<int>> lights(4);
  for (auto &road : lights) {
    int l, s, r, p;
    cin >> l >> s >> r >> p;
    road = {l, s, r, p};
  }
  cout << ((lights[0][3] && (lights[0][1] || lights[1][2])) ||
           (lights[1][3] && (lights[1][0] || lights[2][1])) ||
           (lights[2][3] && (lights[2][0] || lights[3][1])) ||
           (lights[3][3] && (lights[3][2] || lights[0][1])) ? "YES" : "NO")
       << endl;
}
