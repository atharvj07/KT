#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> p(n + 1);
    vector<bool> v(n + 1);
    for (int i = 1; i <= n; i++) { p[i] = i; }
    for (int i = 1; i < n; i++) {
        int x, y;
        cin >> x >> y;
        int rootx = x, rooty = y;
        while (p[x] != x) { x = p[x]; }
        while (p[y] != y) { y = p[y]; }
        if (rootx != rooty) {
            p[rooty] = rootx;
        }
    }
    for (int i = 1; i <= n; i++) {
        int root = i;
        while (p[root] != root) { root = p[root]; }
        v[root] = true;
    }
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (v[i]) {
            cnt++;
            cout << i << ' ';
        }
    }
    for (int i = 1; i <= n; i++) {
        if (!v[i]) {
            cnt++;
            cout << i << ' ';
        }
    }
    return 0;
}
