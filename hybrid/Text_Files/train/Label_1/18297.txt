#include <iostream>
#include <string>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  string s[n];
  for (int i = 0; i < n; i++) {
    cin >> s[i];
  }
  int ans = 0;
  for (int i = 0; i < n - 1; i++) {
    for (int j = 0; j < m - 1; j++) {
      if (s[i][j] == 'f' && s[i][j + 1] == 'a' && s[i + 1][j] == 'c' && s[i + 1][j + 1] == 'e') {
        ans++;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
