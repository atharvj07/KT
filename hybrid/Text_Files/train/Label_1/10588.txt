#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll n, x, f, ans = 0;
    cin >> n;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    cin >> x >> f;
    for (int i = 0; i < n; i++) {
        if (a[i] <= x) continue;
        ll k = (a[i] % x + f - 1) / f;
        ans += k * f;
        a[i] -= k * x;
    }
    cout << ans << endl;
}
