import java.util.Scanner;

public class LiegesOfLegendre {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] cows = new int[n];
        for (int i = 0; i < n; i++) {
            cows[i] = scanner.nextInt();
        }

        boolean kevinWins = isKevinWin(cows, k);

        System.out.println(kevinWins ? "Kevin" : "Nicky");
    }

    static boolean isKevinWin(int[] cows, int k) {
        for (int cow : cows) {
            if (cow % 2 == 1) {
                return true;
            }
        }

        return k > 1;
    }
}
