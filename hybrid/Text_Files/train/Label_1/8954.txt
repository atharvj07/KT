#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < n; i++)
using namespace std;
const int maxn = 2000 + 10;
const int inf = 2e9;
int c[maxn][4], dp[maxn][4];
char s[maxn];
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n, k;
    cin >> n >> k;
    cin >> s;
    memset(c, 0, sizeof c);
    memset(dp, 0x7f, sizeof dp);
    int rg = (k + 1) / 2, br = k / 2, gb = k / 2, rb = (k - 1) / 2;
    rep(i, n) {
      if (i) {
        memcpy(c[i], c[i - 1], sizeof c[0]);
      }
      if (s[i] == 'R') {
        rg = min(rg, i + k - 1);
      }
      if (s[i] == 'G') {
        br = min(br, i + k - 1);
        gb = min(gb, i + k - 1);
      }
      if (s[i] == 'B') {
        rb = min(rb, i + k - 1);
        gb = min(gb, i + k - 1);
      }
      rep(j, 4) { dp[i][j] = dp[i][0]; }
      if (i) { dp[i][1] = min(dp[i][1], dp[i - 1][0] + (s[i] != 'R')); }
      if (i) { dp[i][2] = min(dp[i][2], dp[i - 1][0] + (s[i] != 'G')); }
      if (i) { dp[i][3] = min(dp[i][3], dp[i - 1][0] + (s[i] != 'B')); }
      if (i + 1 >= k) {
        dp[i][1] = min(dp[i][1], dp[i - (k + 1)][1] + (s[i] != 'R' || rg > i));
        dp[i][2] = min(dp[i][2], dp[i - k][2] + (s[i] != 'G' || br > i));
        dp[i][3] = min(dp[i][3], dp[i - k][3] + (s[i] != 'B' || gb > i));
        dp[i][0] = min(dp[i][0], dp[i - k + 1][1] + (s[i] != 'R' || rb > i));
        dp[i][0] = min(dp[i][0], dp[i - k + 1][2] + (s[i] != 'G' || rb > i));
        dp[i][0] = min(dp[i][0], dp[i - k + 1][3] + (s[i] != 'B' || rb > i));
      }
    }
    int ans = inf;
    rep(i, 4) { ans = min(ans, dp[n - 1][i]); }
    cout << ans << '\n';
  }
}
