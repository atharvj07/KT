import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] dp = new int[n + 1];
        dp[n] = 1;
        for (int i = n - 1; i >= 0; i--) {
            int cnt = 0;
            for (int j = i; j < n; j++) {
                if ((long) a[i] * a[j] % 2 == 0) {
                    cnt++;
                }
            }
            if (cnt > 0) {
                dp[i] = dp[i + cnt];
            } else {
                dp[i] = 1;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(dp[i] + " ");
        }
        System.out.println();
    }
}
