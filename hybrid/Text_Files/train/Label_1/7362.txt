#include <bits/stdc++.h>

using namespace std;

#define int long long

const int mod = 1e9 + 7;
const int N = 2001;
int dp[N][N][2];

int solve(vector<int> &a, int l, int r, int b) {
	int &ans = dp[l][r][b];
	if (ans != -1) {
		return ans;
	}
	if (l == r) {
		if (a[l] == b) {
			ans = 1;
		} else {
			ans = 0;
		}
	} else {
		if (a[l] == b) {
			ans = solve(a, l + 1, r, b);
		} else {
			ans = solve(a, l + 1, r, b) + solve(a, l, r, a[l]);
			ans %= mod;
		}
		if (a[r] == b) {
			ans += solve(a, l, r - 1, b);
			ans %= mod;
		} else {
			ans += solve(a, l, r - 1, a[r]);
			ans %= mod;
		}
	}
	return ans;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(0);

	int n, h;
	cin >> n >> h;
	vector<int> a(n);
	for (int i = 0; i < n; i++) {
		cin >> a[i];
	}
	memset(dp, -1, sizeof(dp));
	cout << solve(a, 0, n - 1, h) % mod << "\n";

	return 0;
}
