import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        int max = -1;
        for (int x : a) {
            int sq = (int) Math.sqrt(x);
            if (sq * sq != x)
                max = Math.max(max, x);
        }
        System.out.println(max);
    }
}
