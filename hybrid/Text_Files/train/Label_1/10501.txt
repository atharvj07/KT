
#include <bits/stdc++.h>
#define int long long
using namespace std;
int n, a[4], sum = 0;
bool check(int x) {
  a[n] = x;
  sort(a, a + 4);
  int am = (a[0] + a[1] + a[2] + a[3]) / 4;
  int md = (a[1] + a[2]) / 2;
  return am == md && md == a[3] - a[0];
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n;
  if (n == 4) {
    cout << "NO";
    return 0;
  }
  for (int i = 0; i < n; i++) { cin >> a[i], sum += a[i]; }
  bool chk = false;
  for (int i = 1; i <= 1000000; i++) {
    if (check(i)) {
      chk = true;
      cout << "YES" << endl;
      for (int j = 0; j < 4 - n; j++) { cout << i << endl; }
      break;
    }
  }
  if (!chk) cout << "NO";
  return 0;
}
