#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int w = 0, ans = 1e9;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (a[i] > a[j]) {
        if (++w >= k) { ans = a[i]; break; }
      } else {
        w = 0;
      }
    }
    if (ans != 1e9) break;
  }
  cout << ans;
  return 0;
}
