#include<iostream>

using namespace std;

int main() {
	int n , m , mod ; cin >> n >> m >> mod ;
	long long ans = 1 ;
    for (int i = 0 ; i < n - 2 * m ; i++) ans = ans * 2 % mod ;
    for (int i = 0 ; i < m ; i++) {
		for (int j = i + 1 ; j < m ; j++) {
			int cnt = 0 ;
			for (int k = 0 ; k < n ; k++) cnt += (i == j && k < i) ? 1 : 0 ;
			ans = ans * (8 - cnt) % mod ;
		}
	}
	cout << ans << "\n" ;
}
