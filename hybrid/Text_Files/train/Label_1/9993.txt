#include <bits/stdc++.h>
using namespace std;
#define N 1000010
int a[N], t[N], deg[N];
vector<int> g[N];
queue<int> q;
int main() {
    int n;
    cin >> n;
    for (int i = 2; i <= n; i++) {
        string s;
        cin >> s;
        if (s == "IN") {
            cin >> a[i];
            t[i] = 1;
            deg[i] = 1;
            q.push(i);
        } else {
            int x, y;
            cin >> x >> y;
            deg[i] = 2;
            g[x].push_back(i);
            g[y].push_back(i);
        }
    }
    while (q.size()) {
        int u = q.front();
        q.pop();
        bool fl = 1;
        for (auto v : g[u]) {
            if (t[v]) fl = 0;
        }
        for (auto v : g[u]) {
            t[v] = t[v] || (!fl && t[u]);
        }
        if (u == 1) {
            for (auto v : g[u]) {
                t[v] = (!fl && t[u]);
            }
        }
    }
    for (int i = 2; i <= n; i++) {
        if (t[i] == 1) cout << 1;
        else
            cout << 0;
    }
    return 0;
}
