import java.util.*;

public class AutoDoor {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int a = scanner.nextInt();
        int d = scanner.nextInt();

        int[] clients = new int[m];
        for (int i = 0; i < m; i++) {
            clients[i] = scanner.nextInt();
        }

        int openCount = 0;
        int lastOpenTime = -1;
        int clientIndex = 0;

        for (int i = 1; i <= n; i++) {
            int employeeTime = i * a;

            while (clientIndex < m && clients[clientIndex] < employeeTime) {
                if (clients[clientIndex] > lastOpenTime + d) {
                    openCount++;
                    lastOpenTime = clients[clientIndex];
                }
                clientIndex++;
            }

            if (employeeTime > lastOpenTime + d) {
                openCount++;
                lastOpenTime = employeeTime;
            }
        }

        System.out.println(openCount);
    }
}
