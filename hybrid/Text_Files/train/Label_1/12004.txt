#include<bits/stdc++.h>
#define ll long long
using namespace std;
bool vis[100005];
vector<pair<int,int>>gr[100005];
void dfs(int s){
    vis[s]=1;
    for(auto x:gr[s]){
        if(!vis[x.first]){
            dfs(x.first);
        }
    }
}
int main()
{
    int n,m,u,v;
    cin>>n>>m;
    while(m--){
        cin>>u>>v;
        gr[u].push_back({v,1});
        gr[v].push_back({u,1});
    }
    int ans=0;
    for(int i=1;i<=n;i++){
        if(!vis[i]){
            dfs(i);
            ans++;
        }
    }
    cout<<ans-1<<endl;
    return 0;
}
