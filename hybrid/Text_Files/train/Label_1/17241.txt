#include <iostream>
#include <vector>

int main() {
  long long b1, q, l, m;
  std::cin >> b1 >> q >> l >> m;
  std::vector<long long> a(m);
  for (int i = 0; i < m; i++) std::cin >> a[i];
  long long b = b1;
  int ans = 0;
  while (std::abs(b) <= l) {
    bool ok = 1;
    for (int j = 0; j < m; j++) {
      if (b == a[j]) {
        ok = 0;
        break;
      }
    }
    if (ok) ans++;
    b = b * q;
  }
  if (ans < 1e9) std::cout << ans << '\n';
  else
    std::cout << "inf" << '\n';
  return 0;
}
