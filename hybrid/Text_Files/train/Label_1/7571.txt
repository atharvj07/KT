#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> ans(m);
  for (int i = 0; i < 2 * n; i++) {
    for (int j = 2 * (i + 1) - 1; j <= m; j += 2 * (i + 1)) {
      ans[j - 1] = j;
    }
  }
  for (int i = 2 * n; i <= m; i += 2 * (2 * n - 1)) {
    for (int j = 2 * (i - 2 * n + 1) - 1; j <= m; j += 2 * (i - 2 * n + 1)) {
      ans[j - 1] = j;
    }
  }
  for (int i = 0; i < m; i++) {
    cout << ans[i] << ' ';
  }
  return 0;
}
