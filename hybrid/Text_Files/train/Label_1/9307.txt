#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, p;
  cin >> n >> p;
  string s;
  cin >> s;
  bool flag = true;
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] != 'a' + p - 1) {
      flag = false;
      break;
    }
  }
  if (flag) {
    cout << "NO" << endl;
    return 0;
  }
  int pos = -1;
  for (int i = 0; i < n - 1; i++) {
    if (s[i] == s[i + 1]) {
      pos = i;
      break;
    }
  }
  if (pos == -1) {
    for (int i = 0; i < n - 1; i++) {
      if (s[i] < s[i + 1]) {
        pos = i;
        break;
      }
    }
  }
  s[pos] = s[pos] + 1;
  for (int i = pos + 1; i < n; i++) {
    s[i] = 'a';
  }
  cout << s << endl;
  return 0;
}
