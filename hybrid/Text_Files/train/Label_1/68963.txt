import sys
input = sys.stdin.readline
n, m = map(int, input().split())
a = []
for i in range(m):  a += map(int, input().split())
s = [1] * (n + 1)
for i in range(0, len(a), 2):
    s[a[i]] = -s[a[i]]
    s[a[i + 1]] = -s[a[i + 1]]
for i in range(1, n + 1):
    s[i] += s[i - 1]
if max([s[i] for i in range(n + 1)]) > 0: print(1)
else: print(0)
