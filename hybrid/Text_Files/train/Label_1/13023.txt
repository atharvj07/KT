#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, t;
    cin >> n >> t;
    vector<pair<int, int>> bus(n);
    for (int i = 0; i < n; i++) {
        cin >> bus[i].first >> bus[i].second;
    }
    int ans = -1;
    for (int i = 0; i < n; i++) {
        int cur = t;
        while (cur >= bus[i].first) {
            if (cur == bus[i].first) {
                ans = i + 1;
                break;
            }
            cur -= bus[i].second;
        }
        if (ans != -1) break;
    }
    cout << ans << endl;
}
