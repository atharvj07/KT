#include <iostream>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    long long n;
    std::cin >> n;
    if (n == 1) {
      std::cout << "0" << std::endl;
      continue;
    }
    if (n % 3 != 0) {
      std::cout << -1 << std::endl;
      continue;
    }
    int ans = 0;
    while (n % 3 == 0) {
      ans++;
      n /= 3;
    }
    while (n % 2 == 0) {
      ans++;
      n /= 2;
    }
    if (n == 1)
      std::cout << ans << std::endl;
    else
      std::cout << -1 << std::endl;
  }
  return 0;
}
