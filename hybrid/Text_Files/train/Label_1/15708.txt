#include <bits/stdc++.h>
using namespace std;
int main(){ int t; cin >> t; while(t--) { int n, k; cin >> n >> k; string s = to_string(n), st = s; bool f = 0, d[12]; memset(d, 0, sizeof(d)); for(int i = 0; i < s.size(); ++i) { if((s[i] - '0') > k) { cout << s << endl; f = 1; break; } if(!d[s[i] - '0']) { d[s[i] - '0'] = 1; st += s[i]; } } if(f) continue; for(int i = 0; i < s.size(); ++i) { for(int j = st[i] - '0'; j <= k; ++j) { if(!d[j]) { s[i] = j + '0'; d[j] = 1; break; } } } string ans = ""; for(int i = 0; i < s.size(); ++i) { if(s[i] != '0') { ans += s[i]; break; } } for(int i = ans.size(); i < s.size(); ++i) { ans += s[i]; } cout << ans << endl; } return 0; }
