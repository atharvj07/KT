import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] rectangles = new int[6][2];

        for (int i = 0; i < rectangles.length; i++) {
            rectangles[i][0] = scanner.nextInt();
            rectangles[i][1] = scanner.nextInt();
        }

        Arrays.sort(rectangles, (a, b) -> (a[0] - b[0]));

        for (int i = 0; i < rectangles.length; i++) {
            for (int j = i + 1; j < rectangles.length; j++) {
                for (int k = j + 1; k < rectangles.length; k++) {
                    if (rectangles[i][0] == rectangles[j][0] && rectangles[i][1] == rectangles[k][1] &&
                            rectangles[j][0] == rectangles[k][0] && rectangles[j][1] == rectangles[i][1]) {
                        System.out.println("yes");
                        return;
                    }
                }
            }
        }

        System.out.println("no");
    }
}
