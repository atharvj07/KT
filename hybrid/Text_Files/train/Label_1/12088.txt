#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;

    vector<int> l(m);
    for (int i = 0; i < m; i++) {
        cin >> l[i];
    }

    vector<bool> painted(n + 1, false);
    vector<int> color(n + 1, 0);

    int last_color = 0;
    for (int i = 0; i < m; i++) {
        bool found = false;
        for (int p = 1; p <= n - l[i] + 1; p++) {
            if (!painted[p]) {
                found = true;
                for (int j = p; j <= p + l[i] - 1; j++) {
                    painted[j] = true;
                    color[j] = last_color + 1;
                }
                last_color++;
                cout << p << " ";
                break;
            }
        }
        if (!found) {
            cout << "-1" << endl;
            return 0;
        }
    }

    cout << endl;

    return 0;
}
