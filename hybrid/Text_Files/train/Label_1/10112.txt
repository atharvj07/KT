#include <bits/stdc++.h>
#define ll long long
#define mod 1000000007
using namespace std;

ll power(ll x, ll n) {
    ll res = 1;
    while (n > 0) {
        if (n % 2 == 1) {
            res = (res * x) % mod;
        }
        x = (x * x) % mod;
        n /= 2;
    }
    return res;
}

ll get_count(ll n, ll k) {
    ll count = 0;
    for (ll i = 0; i <= k; i++) {
        ll low = 0;
        ll high = n >> i;
        while (low <= high) {
            ll mid = (low + high) / 2;
            if (mid + (1ll << i) - 1 <= n) {
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }
        count += high;
    }
    return count;
}
