import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt() + 1;
            int zeroes = Integer.bitCount(n);
            System.out.println((1 << zeroes));
        }
    }
}
