import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

public class Main {

    static class Edge {
        int u, v;
        Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static ArrayList<Integer>[] adj;
    static int[] color;
    static int n, m;
    static ArrayList<Edge> edges;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        adj = new ArrayList[n + 1];
        color = new int[n + 1];
        Arrays.fill(color, -1);
        edges = new ArrayList<>();

        for (int i = 0; i < n + 1; i++) {
            adj[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            String s = scanner.next();
            edges.add(new Edge(u, v));
            if (s.equals("M")) {
                color[u] = color[v] = 0;
            } else {
                color[u] = color[v] = 1;
            }
            adj[u].add(v);
            adj[v].add(u);
        }

        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            if (color[i] != -1) {
                cnt++;
            }
        }

        if (cnt % 2 == 1 || cnt == 0) {
            System.out.println(-1);
            return;
        }

        int[] path = new int[cnt + 1];
        int idx = 0;
        for (int i = 1; i <= n; i++) {
            if (color[i] != -1) {
                path[idx++] = i;
            }
        }

        boolean[] vis = new boolean[n + 1];
        dfs(path[0], path[0], vis, 0);

        int cntEdge = 0;
        for (Edge edge : edges) {
            if (color[edge.u] != color[edge.v]) {
                cntEdge++;
            }
        }

        if (cntEdge != cnt / 2) {
            System.out.println(-1);
            return;
        }

        System.out.println(cntEdge);
        for (Edge edge : edges) {
            if (color[edge.u] != color[edge.v]) {
                System.out.print(edge.u + " " + edge.v + " ");
            }
        }
    }

    static void dfs(int u, int pre, boolean[] vis, int col) {
        vis[u] = true;
        color[u] = col;
        for (int v : adj[u]) {
            if (v != pre && !vis[v]) {
                dfs(v, u, vis, col ^ 1);
            }
        }
    }
}
