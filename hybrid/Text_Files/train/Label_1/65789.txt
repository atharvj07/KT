import heapq
n,m=map(int,input().split())
p=[0]+list(map(int,input().split()))
e=[[]for i in range(n+1)]
for i in range(m):
	a,b,c=map(int,input().split())
	e[a].append((b,c))
	e[b].append((a,c))
ans=0
g,q=[],[]
for i in range(1,n+1):
	if not e[i]:g.append((p[i],i))
	else:heapq.heappush(q,p[i])
while g:
	mn=heapq.heappop(g)
	while e[mn[1]]:
		t=e[mn[1]].pop()
		if t[0]==mn[0]or t[0]==mn[1]:continue
		if t[1]>=mn[0]:heapq.heappush(q,mn[0])
		else:heapq.heappush(g,(t[1],t[0]))
		ans+=t[1]
		mn=t
		break			
while q:
	ans+=heapq.heappop(q)
print(ans)
