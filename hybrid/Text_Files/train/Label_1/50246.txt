import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        long[][] sequences = new long[k][];
        for (int i = 0; i < k; i++) {
            int n = sc.nextInt();
            sequences[i] = new long[n];
            for (int j = 0; j < n; j++) {
                sequences[i][j] = sc.nextInt();
            }
        }

        boolean possible = false;
        int i = -1, x = -1, j = -1, y = -1;
        for (int a = 0; a < k; a++) {
            long sumA = 0;
            for (long num : sequences[a]) {
                sumA += num;
            }
            for (int b = 0; b < k; b++) {
                if (a == b) {
                    continue;
                }
                long sumB = 0;
                for (long num : sequences[b]) {
                    sumB += num;
                }
                for (int aa = 0; aa < sequences[a].length; aa++) {
                    for (int bb = 0; bb < sequences[b].length; bb++) {
                        if (sumA - sequences[a][aa] == sumB - sequences[b][bb]) {
                            possible = true;
                            i = a + 1;
                            x = aa + 1;
                            j = b + 1;
                            y = bb + 1;
                            break;
                        }
                    }
                }
            }
        }

        if (possible) {
            System.out.println("YES");
            System.out.println(i + " " + x);
            System.out.println(j + " " + y);
        } else {
            System.out.println("NO");
        }
    }
}
