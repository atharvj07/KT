#include <bits/stdc++.h>
#define endl "\n"
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.size();
        vector<bool> vis(26, 0);
        vector<char> ans;
        for (int i = 0; i < n; i++) {
            if (i > 0 && s[i] == s[i - 1]) {
                cout << "NO" << endl;
                goto out;
            }
            if (!vis[s[i] - 'a']) {
                ans.push_back(s[i]);
                vis[s[i] - 'a'] = 1;
            }
        }
        for (char c = 'a'; c <= 'z'; c++) {
            if (!vis[c - 'a']) {
                ans.push_back(c);
            }
        }
        cout << "YES" << endl;
        for (char c : ans) {
            cout << c;
        }
        cout << endl;
    out:;
    }
    return 0;
}
