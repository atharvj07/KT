#include <bits/stdc++.h>

using namespace std;

typedef long long int lli;

int main() {
    lli n, x;
    cin >> n >> x;
    vector<lli> v(n, 0);
    for (lli i = 0; i < n; i++) cin >> v[i];
    sort(v.rbegin(), v.rend());
    lli sum = 0;
    for (lli i = 0; i < n; i++) {
        sum += (v[i] * max(x - i, 1LL));
    }
    cout << sum << endl;
    return 0;
}
