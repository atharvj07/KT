#include <bits/stdc++.h>
using namespace std;
int n,a[105],b[105];
int main(){
    cin>>n;
    for(int i=0;i<n;++i){cin>>a[i];b[i]=a[i];}
    sort(a,a+n);
    int j=n-1;
    for(int i=0;i<n/2;++i,j--){
        b[i]=a[i];
        b[j]=a[n-1-i];
    }
    for(int i=0;i<n;++i)cout<<b[i]<<" ";
    cout<<endl;
    return 0;
}
