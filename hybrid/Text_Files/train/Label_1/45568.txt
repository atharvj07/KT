import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int f = sc.nextInt();
        int k = sc.nextInt();
        int dist = a * 2;
        if (b < dist || f <= 0 || f >= a) {
            System.out.println(-1);
            return;
        }
        int needRefuel = k * dist / b;
        if (k * dist % b > 0) {
            needRefuel++;
        }
        if (f <= b) {
            needRefuel = Math.max(needRefuel, 1);
        } else if (f >= a - b && f <= a) {
            needRefuel = Math.max(needRefuel, 1);
            if (needRefuel % 2 == 0) {
                needRefuel++;
            }
        } else {
            int dist1 = f - b;
            int dist2 = a - f - b;
            if (dist1 % b == 0) {
                needRefuel = Math.max(needRefuel, dist1 / b + 1);
            } else {
                needRefuel = Math.max(needRefuel, dist1 / b + 2);
            }
            if (dist2 % b == 0) {
                needRefuel = Math.max(needRefuel, dist2 / b + 1);
            } else {
                needRefuel = Math.max(needRefuel, dist2 / b + 2);
            }
        }
        System.out.println(needRefuel);
    }
}
