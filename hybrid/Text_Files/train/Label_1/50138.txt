import java.util.*;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();

        List<Integer>[] adj = new List[N + 1];
        for (int i = 1; i <= N; i++) {
            adj[i] = new ArrayList<>();
        }

        for (int i = 0; i < M; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }

        int[] visited = new int[N + 1];
        Arrays.fill(visited, -1);

        int count = 0;
        for (int i = 1; i <= N; i++) {
            if (visited[i] == -1) {
                count++;
                dfs(adj, visited, i);
            }
        }

        System.out.println(count - 1);
    }

    static void dfs(List<Integer>[] adj, int[] visited, int node) {
        visited[node] = 1;
        for (int neighbor : adj[node]) {
            if (visited[neighbor] == -1) {
                dfs(adj, visited, neighbor);
            }
        }
    }
}
