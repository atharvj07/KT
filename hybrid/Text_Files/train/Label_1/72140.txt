def max_circular_value(n, arr):
    if n % 2 == 0:
        return 0  # This case should not happen as n is always odd
    if n == 1:
        return arr[0]
    sum_all = sum(arr)
    dp = [0] * n
    for i in range(n):
        dp[i] = arr[i - 1] + arr[(i + 1) % n]
    for j in range(3, n + 1, 2):
        new_dp = [0] * n
        for i in range(n):
            new_dp[i] = max(dp[i] - arr[i] - arr[(i + j - 1) % n] + arr[(i + j - 2) % n] + arr[(i - 1) % n], dp[(i + 1) % n] - arr[(i + 1) % n] - arr[(i + j - 1) % n] + arr[i] + arr[(i + j - 2) % n])
        dp = new_dp
    return max(dp)

n = int(input().strip())
arr = list(map(int, input().strip().split()))
print(max_circular_value(n, arr))
