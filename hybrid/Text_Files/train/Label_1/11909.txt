#include <iostream>
#include <vector>
#include <map>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<tuple<int, int, int>> points(n);
  for (int i = 0; i < n; i++) {
    int x, y, z;
    cin >> x >> y >> z;
    points[i] = {x, y, z};
  }
  sort(points.begin(), points.end());
  map<tuple<int, int, int>, int> idx;
  for (int i = 0; i < n; i++) {
    idx[points[i]] = i;
  }
  vector<pair<int, int>> ans;
  for (int i = 0; i < n / 2; i++) {
    auto p1 = points[i];
    auto p2 = points[n - i - 1];
    ans.push_back({idx[p1] + 1, idx[p2] + 1});
  }
  for (auto p : ans) {
    cout << p.first << " " << p.second << "\n";
  }
  return 0;
}
