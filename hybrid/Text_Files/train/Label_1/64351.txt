import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
b = [0] * n
b[0] = a[0]
for i in range(1, n):
    b[i] = b[i - 1] ^ a[i]
c = [0] * n
c[-1] = a[-1]
for i in range(n - 2, -1, -1):
    c[i] = c[i + 1] ^ a[i]
ans = 0
for i in range(n):
    for j in range(i + 1, n):
        if (j - i + 1) % 2 == 0 and b[j] == c[i]:
            ans += 1
print(ans)
