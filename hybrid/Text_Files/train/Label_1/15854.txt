#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    ll x, k, ans1 = 0, ans2 = 0;
    cin >> x >> k;
    set<ll>st;
    for (int i = 0; i < k; i++) {
        int tp;
        cin >> tp;
        if (tp == 1) {
            ll a, b;
            cin >> a >> b;
            st.insert(b - 1);
            ans1++;
        } else {
            ll a;
            cin >> a;
            st.insert(a);
            ans2++;
        }
    }
    cout << (x - ans2 - *st.begin()) << " ";
    cout << (x - k - *st.begin()) << endl;
    return 0;
}
