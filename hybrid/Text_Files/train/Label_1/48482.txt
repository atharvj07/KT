import java.util.*;
import java.lang.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int x = in.nextInt();
    int y = in.nextInt();

    double percent = (double) y / 100;
    double numPeople = n * percent;
    double numClones = numPeople - x;

    System.out.println((int) Math.ceil(numClones));
  }
}
