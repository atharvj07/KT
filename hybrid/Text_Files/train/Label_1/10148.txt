#include<bits/stdc++.h>
using namespace std;
int n, k;
map<int, int>mp;
int main(){
scanf("%d%d",&k,&n);
for(int i=1;i<=k;i++){
string s;
cin>>s;
for(int j=0;j<n-1;j++){
if(s[j]!=s[j+1])
++mp[s[j]-'a'];
}
}
string ans;
for(auto it :mp){
if(it.second&1)
ans+='a'+it.first;
}
int c=0;
for(auto it :ans)
c++;
reverse(ans.begin(),ans.end());
if(c<=1)
ans="";
if(ans.size()>0){
for(int i=0;i<k-1;i++){
ans+=ans;
}
}
if(ans.size()==0)
cout<<-1<<endl;
else cout<<ans<<endl;
}
