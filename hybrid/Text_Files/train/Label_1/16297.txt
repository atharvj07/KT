#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;
typedef long double ld;

int main() {
  ll n, T;
  cin >> n >> T;
  vector<ll> a(n), t(n);
  for (ll i = 0; i < n; ++i) cin >> a[i];
  for (ll i = 0; i < n; ++i) cin >> t[i];
  ll lo = 0, hi = 1e18;
  while (hi - lo > 1) {
    ll mid = (lo + hi) / 2;
    ld sum_a = 0, sum_ta = 0;
    for (ll i = 0; i < n; ++i) {
      if (mid >= a[i]) {
        sum_a += a[i];
        sum_ta += a[i] * t[i];
      } else {
        sum_a += mid;
        sum_ta += mid * t[i];
        break;
      }
    }
    if (sum_ta / sum_a >= (ld)T)
      hi = mid;
    else
      lo = mid;
  }
  ld sum_a = 0, sum_ta = 0;
  for (ll i = 0; i < n; ++i) {
    if (lo >= a[i]) {
      sum_a += a[i];
      sum_ta += a[i] * t[i];
    } else {
      sum_a += lo;
      sum_ta += lo * t[i];
      break;
    }
  }
  if (sum_ta / sum_a != (ld)T)
    cout << 0 << endl;
  else
    cout << sum_a << endl;
  return 0;
}
