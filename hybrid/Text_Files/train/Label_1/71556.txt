from sys import stdin
input=stdin.readline

n=int(input())
p=[0 for i in range(n)]
c=[0 for j in range(n)]
for i in range(n):
    p[i],c[i]=map(str,input().split())
    p[i]=int(p[i])

r=[]
g=[]
b=[]
for i in range(n):
    if c[i]=='R':
        r.append(p[i])
    elif c[i]=='G':
        g.append(p[i])
    else:
        b.append(p[i])

ans=0
gr=[]
if len(r)>0:
    for i in range(len(g)):
        gr.append(abs(g[i]-r[0]))

    for j in range(len(b)):
        gr.append(abs(b[j]-r[0]))

    gr.sort()
    for i in range(len(gr)-1):
        ans+=gr[i]

gb=[] 
if len(b)>0:
    for i in range(len(g)):
        gb.append(abs(b[0]-g[i]))

    for j in range(len(r)):
        gb.append(abs(r[j]-b[0]))

    gb.sort()
    for i in range(len(gb)-1):
        ans+=gb[i]
print(ans)
