#include <bits/stdc++.h>
using namespace std;
int main() {
  double x1, y1, r1, x2, y2, r2;
  cin >> x1 >> y1 >> r1 >> x2 >> y2 >> r2;
  double dx = x2 - x1, dy = y2 - y1;
  double dist = sqrt(dx * dx + dy * dy);
  if (dist > r1 + r2) {
    cout << "0.00000000000000000000" << endl;
    return 0;
  }
  if (dist + r1 < r2 || dist + r2 < r1) {
    double minr = r1;
    if (r1 > r2) swap(r1, r2);
    cout << minr * minr * acos(-1.0) << endl;
    return 0;
  }
  double a = (r1 * r1 - r2 * r2 + dist * dist) / (2 * dist);
  double h = sqrt(r1 * r1 - a * a);
  cout << a * h * acos(-1.0) << endl;
  return 0;
}
