MOD = 10**9 + 7

n = int(input())
h = list(map(int, input().split()))

ways = 0
prev_diff = 0

for i in range(1, n):
    diff = abs(h[i] - h[i-1])
    if diff == 0:
        ways += (2 * prev_diff + 1) % MOD
    elif diff == 1:
        ways += (2 * prev_diff + 2) % MOD
    else:
        ways += (2 * prev_diff + diff + 1) % MOD
    prev_diff = diff

print(ways % MOD)
