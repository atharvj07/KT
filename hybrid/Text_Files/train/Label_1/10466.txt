#include <iostream>
#include <vector>

int main() {
    int n; std::cin >> n;
    std::string s; std::cin >> s;
    std::vector<int> pref(n, 0);
    for (int i = 1; i < n; ++i) {
        pref[i] = pref[i - 1] + (s[i] == '<' ? -1 : 1);
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        if (pref[i] == 0) ans++;
    }
    std::cout << ans << std::endl;
}
