#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int a, b, k;
        cin >> a >> b >> k;
        vector<pair<int, int>> p;
        while (k--) {
            int x, y;
            cin >> x >> y;
            p.push_back({x, y});
        }
        int ans = 0;
        for (int i = 0; i < p.size(); i++) {
            for (int j = i + 1; j < p.size(); j++) {
                if (p[i].first != p[j].first && p[i].second != p[j].second) {
                    ans++;
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
