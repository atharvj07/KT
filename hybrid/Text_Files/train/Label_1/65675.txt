MOD = 10**9 + 7

def binomial_coeff(n, k):
    if k == 0 or k == n:
        return 1
    if k > n - k:
        k = n - k
    result = 1
    for i in range(k):
        result = result * (n - i) // (i + 1)
    return result

def max_matching(n):
    return binomial_coeff(2 * n, n) // (n + 1) % MOD

n = int(input())
print(max_matching(n))
