#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> h(n + 2);
  for (int i = 1; i <= n; i++) cin >> h[i];
  int ans = 0;
  for (int i = 2; i <= n; i++) {
    if (h[i] > h[i - 1] || h[i] < h[i + 1]) ans++;
  }
  cout << ans << endl;
  return 0;
}
