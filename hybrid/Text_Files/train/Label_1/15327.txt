#include<bits/stdc++.h>
using namespace std;
int main(){
	int n,x,ans=-1,check=0,p=0;
	cin>>n>>x;
	int a[n+2];
	for(int i=0; i<n; i++){ cin>>a[i]; if(a[i]==x) check=1; }
	sort(a,a+n);
	for(int i=0; i<n; i++){
		if(a[i]==a[i+1]) p=1;
		if(a[i]&x==a[i+1]) p=1;
	}
	if(check==1) ans=0;
	if(p==1) ans=1;
	cout<<ans<<endl;
}
