#include <iostream>
#include <vector>
#include <algorithm>

using std::cin;
using std::cout;

int main() {
    int n, s;
    cin >> n >> s;
    std::vector<std::pair<int, int>> passengers(n);
    for (auto& p : passengers) {
        cin >> p.first >> p.second;
    }
    std::sort(passengers.begin(), passengers.end(),
              [](const std::pair<int, int>& a, const std::pair<int, int>& b) {
                  return a.second < b.second;
              });
    int time = 0;
    for (const auto& p : passengers) {
        time = std::max(time, p.second);
        time += s - p.first;
    }
    cout << time << std::endl;
    return 0;
}
