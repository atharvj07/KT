n, m = map(int, input().split())
p = list(map(int, input().split()))

# Convert permutation to positions
pos = [0] * (n + 1)
for i, val in enumerate(p):
    pos[val] = i

# Initialize foe pairs
foes = [n] * (n + 1)

for _ in range(m):
    a, b = map(int, input().split())
    # Store the minimum position of a foe for each element
    foes[a] = min(foes[a], pos[b])
    foes[b] = min(foes[b], pos[a])

# Count intervals
count = 0
right = 0
for left in range(1, n + 1):
    right = max(right, left - 1)
    while right < n and right <= foes[p[left - 1]]:
        right += 1
    count += right - left + 1

print(count)
