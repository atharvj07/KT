import java.util.*;

class Main {
    static int n, k, a[];

    static int max() {
        int ans = 0, x = (1 << n) - 1;
        for (int i = 1; i <= k; i++) ans += Integer.bitCount(x & a[i]) + 1;
        return ans;
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        n = input.nextInt();
        k = input.nextInt();
        a = new int[k + 1];
        for (int i = 1; i <= k; i++) a[i] = input.nextInt();
        System.out.println(max());
    }
}
