#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 2e9;
  for (int i = 0; i < n - 3; i++) {
    int cnt = 0;
    for (int j = i; j < i + 4; j++) {
      char c = s[j];
      if (c != "ACTG"[j - i]) cnt += min(c - "ACTG"[j - i], 'Z' - c + "ACTG"[j - i] - 'A');
      else cnt += 0;
    }
    ans = min(ans, cnt);
  }
  cout << ans << endl;
  return 0;
}
