from collections import defaultdict, deque

def find_order(n, directed_edges, undirected_edges):
    graph = defaultdict(list)
    in_degree = defaultdict(int)
    for u, v in directed_edges:
        graph[u].append(v)
        in_degree[v] += 1
    queue = deque([i for i in range(1, n+1) if in_degree[i] == 0])
    order = {}
    pos = 1
    while queue:
        node = queue.popleft()
        order[node] = pos
        pos += 1
        for neighbour in graph[node]:
            in_degree[neighbour] -= 1
            if in_degree[neighbour] == 0:
                queue.append(neighbour)
    if len(order) != n:
        return False, []
    result_edges = directed_edges[:]
    for u, v in undirected_edges:
        if order[u] < order[v]:
            result_edges.append((u, v))
        else:
            result_edges.append((v, u))
    return True, result_edges

t = int(input().strip())
for _ in range(t):
    n, m = map(int, input().split())
    directed_edges = []
    undirected_edges = []
    for _ in range(m):
        t_i, x_i, y_i = map(int, input().split())
        if t_i == 1:
            directed_edges.append((x_i, y_i))
        else:
            undirected_edges.append((x_i, y_i))
    possible, result_edges = find_order(n, directed_edges, undirected_edges)
    if possible:
        print("YES")
        for u, v in result_edges:
            print(u, v)
    else:
        print("NO")
