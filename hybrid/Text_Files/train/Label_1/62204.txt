input()
colors = list(map(int, input().split()))
cnt = [0] * 300001
last = [0] * 300001
for i in range(len(colors)):
    if cnt[colors[i]]:
        last[colors[i]] = i
    cnt[colors[i]] = i
ans = 0
for i in range(1, len(colors) + 1):
    if not cnt[i]: continue
    ans = max(ans, i - 1, len(colors) - last[i])
print(ans)
