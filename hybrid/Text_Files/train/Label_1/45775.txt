import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner cin = new Scanner(System.in);
    int n = cin.nextInt();
    int[] pts = new int[n];
    for (int i = 0; i < n; i++) {
      pts[i] = cin.nextInt();
    }
    int max = pts[0], min = pts[0], cnt = 0;
    for (int i = 1; i < n; i++) {
      if (pts[i] > max) {
        cnt++;
        max = pts[i];
      }
      if (pts[i] < min) {
        cnt++;
        min = pts[i];
      }
    }
    System.out.println(cnt);
  }
}
