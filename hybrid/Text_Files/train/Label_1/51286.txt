import java.util.*;
class Main {
  static void solve(int n, int m, int[] pos, int q, int[] loss) {
    PriorityQueue<Integer> pq = new PriorityQueue<>();
    for (int p : pos) {
      pq.add(p);
    }
    for (int l : loss) {
      int res = -1;
      int cur = 0;
      int cnt = 0;
      while (!pq.isEmpty()) {
        int p = pq.poll();
        cnt++;
        if (p > cur) {
          cur = p;
          if (cnt > l) {
            break;
          }
        } else if (cur + 1 < n) {
          res = cur + 1;
          pq.add(cur + 1);
          cnt--;
        }
      }
      if (cnt <= l) {
        res = cur + 1;
      }
      System.out.println(res);
    }
  }
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    String[] nm = sc.nextLine().split(" ");
    int n = Integer.parseInt(nm[0]);
    int m = Integer.parseInt(nm[1]);
    int[] pos = new int[m];
    for (int i = 0; i < m; i++) {
      pos[i] = sc.nextInt();
    }
    int q = sc.nextInt();
    int[] loss = new int[q];
    for (int i = 0; i < q; i++) {
      loss[i] = sc.nextInt();
    }
    solve(n, m, pos, q, loss);
  }
}
