import sys
input = sys.stdin.readline

for _ in range(int(input())):
  n = int(input())
  l = [0] * n
  r = [0] * n
  p = [0] * (2 * n)
  for i in range(n):
    l[i], r[i] = map(int, input().split())
    p[i] = (l[i], 1)
    p[i + n] = (r[i] + 1, -1)
  p.sort()
  ans = [0] * n
  cnt = 0
  for i in range(2 * n):
    cnt += p[i][1]
    if not cnt:
      for j in range(n):
        if l[j] == p[i][0]:
          ans[j] = 1
      for j in range(n):
        if r[j] + 1 == p[i][0]:
          ans[j] = 2
  if 0 in ans:
    print(-1)
  else:
    print(*ans)
