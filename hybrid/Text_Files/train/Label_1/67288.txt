t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().strip().split())
    s = input().strip()

    # Count of wins
    wins = s.count('W')

    if wins + k >= n:  # If we can turn all to wins
        print(2 * n - 1)
        continue

    if wins == 0:  # If no wins at all
        if k == 0:
            print(0)
        else:
            print(2 * k - 1)
        continue

    # Count scores without changes
    score = 1 if s[0] == 'W' else 0
    for i in range(1, n):
        if s[i] == 'W':
            if s[i-1] == 'W':
                score += 2
            else:
                score += 1

    # Find all L streaks between Ws
    streaks = []
    prev = None
    count = 0
    for char in s:
        if char == 'L':
            if prev == 'W':
                count = 1
            elif prev == 'L':
                count += 1
        else:
            if prev == 'L' and count > 0:
                streaks.append(count)
            count = 0
        prev = char

    # Sort the streaks to fill smaller gaps first
    streaks.sort()

    # Fill the gaps with K
    for streak in streaks:
        if k >= streak:
            k -= streak
            score += 2 * streak + 1  # Convert L streak to W and add points

    # Add remaining K to the score
    score += 2 * k

    print(score)
