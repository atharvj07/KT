#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
#define ll long long
vector<int>adj[200001];
int color[200001];
int t;

void dfs(int node,int c){
    if(color[node]!=-1)return;
    color[node]=c;
    for(int child:adj[node]){
        dfs(child,c);
    }
}

int main(){
    int n;
    cin>>n;
    map<pair<int,int>,int>mp;
    map<int,int>freq;
    for(int i=0;i<n;i++){
        int x,y;
        cin>>x>>y;
        if(mp.find({x,y})==mp.end()){
            mp[{x,y}]=++t;
        }
        int u=mp[{x,y}];
        if(freq.find(x)==freq.end()){
            freq[x]=++t;
        }
        if(freq.find(y)==freq.end()){
            freq[y]=++t;
        }
        int v1=freq[x],v2=freq[y];
        adj[u].push_back(v1);
        adj[u].push_back(v2);
        adj[v1].push_back(u);
        adj[v2].push_back(u);
    }
    memset(color,-1,sizeof(color));
    ll ans=1;
    for(int i=1;i<=t;i++){
        if(color[i]==-1){
            dfs(i,i);
            ans=(ans*3)%mod;
        }
    }
    cout<<ans;
    return 0;
}
