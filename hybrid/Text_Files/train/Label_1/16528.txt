#include <bits/stdc++.h>

using namespace std;

int main() {
  cin.tie(0), ios_base::sync_with_stdio(0);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  int xor_sum = 0;
  for (const int &x : a) xor_sum ^= x;
  if (xor_sum == 0) {
    cout << -1;
    return 0;
  }
  vector<int> seg;
  int cur = 0;
  for (int i = 0; i < n; i++) {
    xor_sum ^= a[i];
    if (xor_sum == 0 || i == (n - 1)) {
      if (i != 0) seg.push_back(i - cur);
      cur = i + 1;
    }
  }
  cout << seg.size();
  return 0;
}
