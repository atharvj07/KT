#include <bits/stdc++.h>
using namespace std;
#define mod 1000000007LL
int main() {
  long long f, w, h;
  cin >> f >> w >> h;
  if (f == 0) {
    if (w <= h) {
      cout << 0;
      return 0;
    }
    long long ans = 2;
    for (long long i = 2; i <= w - h; i++) {
      ans = ((ans % mod) * (i % mod)) % mod;
    }
    cout << ans;
  } else {
    if (w <= h) {
      cout << 0;
      return 0;
    }
    long long ans = 1;
    for (long long i = 2; i <= f - 1; i++) {
      ans = ((ans % mod) * (i % mod)) % mod;
    }
    long long ans2 = 1;
    for (long long i = 2; i <= w - h; i++) {
      ans2 = ((ans2 % mod) * (i % mod)) % mod;
    }
    ans = ((ans % mod) * (ans2 % mod)) % mod;
    long long ans3 = 1;
    for (long long i = 2; i <= f + w - 1; i++) {
      ans3 = ((ans3 % mod) * (i % mod)) % mod;
    }
    long long inv = 1;
    long long pw = mod - 2;
    while (pw > 0) {
      if (pw % 2 == 1) {
        inv = ((inv % mod) * (ans % mod)) % mod;
      }
      ans = ((ans % mod) * (ans % mod)) % mod;
      pw = pw / 2;
    }
    ans = ((ans % mod) * (ans3 % mod)) % mod;
    cout << ans;
  }
}
