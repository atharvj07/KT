for _ in range(int(input())):
    n, q = map(int, input().split())
    a = [1 if x=="+" else -1 for x in input()]
    b = [0]*n
    for i in range(n):
        b[i] = a[i]
        if i>0: b[i] += b[i-1]
    for _ in range(q):
        l, r = map(int, input().split())
        l -= 1
        if l: print(abs(b[r-1] - b[l-1]))
        else: print(b[r-1])
