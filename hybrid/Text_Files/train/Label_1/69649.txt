n, m = map(int, input().split())
leaders = list(map(int, input().split()))

a = [0] * n
possible = True

for i in range(1, m):
    current_leader = leaders[i-1] - 1
    next_leader = leaders[i] - 1
    diff = (next_leader - current_leader) % n

    if diff == 0:
        diff = n

    if a[current_leader] == 0 or a[current_leader] == diff:
        a[current_leader] = diff
    else:
        possible = False
        break

if possible:
    used_numbers = set(a)
    for i in range(n):
        if a[i] == 0:
            for j in range(1, n+1):
                if j not in used_numbers:
                    a[i] = j
                    used_numbers.add(j)
                    break
    if len(set(a)) != n:
        print(-1)
    else:
        print(" ".join(map(str, a)))
else:
    print(-1)
