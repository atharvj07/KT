def merge_sequences(k, n, m, a, b):
    result = []
    current_length = k
    index_a, index_b = 0, 0

    while index_a < n or index_b < m:
        action_performed = False

        if index_a < n and (a[index_a] == 0 or a[index_a] <= current_length):
            if a[index_a] == 0:
                current_length += 1
            result.append(a[index_a])
            index_a += 1
            action_performed = True
        elif index_b < m and (b[index_b] == 0 or b[index_b] <= current_length):
            if b[index_b] == 0:
                current_length += 1
            result.append(b[index_b])
            index_b += 1
            action_performed = True

        if not action_performed:
            return [-1]

    return result

t = int(input().strip())
for _ in range(t):
    input()  # Read and ignore the empty line before each test case
    k, n, m = map(int, input().strip().split())
    a = list(map(int, input().strip().split()))
    b = list(map(int, input().strip().split()))
    result = merge_sequences(k, n, m, a, b)
    print(' '.join(map(str, result)))
