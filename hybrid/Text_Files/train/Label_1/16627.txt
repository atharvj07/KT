#include <iostream>
#include <vector>
#include <set>
#include <algorithm>
using namespace std;

int main() {
  long a, b, c;
  cin >> a >> b >> c;
  long t = (a * 1000000) / b;
  string ans = to_string(t);
  long pos = ans.find(c + '0');
  if (pos == -1) cout << "-1\n";
  else {
    pos = pos - (a * 1000000) / b + 1;
    cout << pos << endl;
  }
  return 0;
}
