import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] a = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = in.nextInt();
        }
        String s = in.next();
        int sum = 0;
        for (int i = 0; i < s.length(); i++) {
            sum += a[s.charAt(i) - '1'];
        }
        System.out.println(sum);
    }
}
