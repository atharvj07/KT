#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    int n, m;
    string s;
    cin >> n >> s >> m;
    vector<int> q;
    for (int i = 0; i < n; i++) {
        if (s[i] == '?') q.push_back(i);
    }
    int ans = q.size() / 2;
    if (q.size() % 2 == 1) ans++;
    cout << ans << endl;
    return 0;
}
