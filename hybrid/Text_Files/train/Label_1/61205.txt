n,q=map(int,input().split())
a=list(map(int,input().split()))
for i in range(q):
    s,x=input().split()
    if s=='>':
        for j in range(n):
            if a[j]>int(x):
                a[j]=-a[j]
    else:
        for j in range(n):
            if a[j]<int(x):
                a[j]=-a[j]
print(*a)
