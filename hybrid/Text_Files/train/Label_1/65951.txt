t = int(input())
for _ in range(t):
    n = int(input())
    s = input()
    if s[0] == '0' or s[-1] == '0' or s.count('1') % 2 != 0:
        print("NO")
    else:
        print("YES")
        a, b = "", ""
        open_a, open_b = 0, 0
        flip = False
        for i in s:
            if i == '1':
                if not flip:
                    a += "("
                    b += "("
                    open_a += 1
                    open_b += 1
                else:
                    a += ")"
                    b += ")"
                    open_a -= 1
                    open_b -= 1
                if open_a == 0:
                    flip = not flip
            else:
                if open_a <= open_b:
                    a += "("
                    b += ")"
                    open_a += 1
                    open_b -= 1
                else:
                    a += ")"
                    b += "("
                    open_a -= 1
                    open_b += 1
        print(a)
        print(b)
