import java.util.*;
import java.io.*;

public class Main {
    static int mod = 1000000007;
    static int[] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] deg = new int[n + 1];
        ArrayList<ArrayList<Integer>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++)
            adj.add(new ArrayList<>());
        for (int i = 1; i < n; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj.get(u).add(v);
            adj.get(v).add(u);
            deg[u]++;
            deg[v]++;
        }
        int k = sc.nextInt();
        int x = sc.nextInt();
        dp = new int[n + 1];
        Arrays.fill(dp, -1);
        System.out.println(dfs(1, 0, adj, deg, k, x, m));
    }

    static int dfs(int node, int par, ArrayList<ArrayList<Integer>> adj, int[] deg, int k, int x, int m) {
        if (dp[node] != -1)
            return dp[node];
        long ways = 1;
        if (deg[node] == 1) {
            if (node != 1)
                ways = x;
        } else {
            for (int nbr : adj.get(node)) {
                if (nbr == par)
                    continue;
                ways *= dfs(nbr, node, adj, deg, k, x, m);
                ways %= mod;
            }
        }
        if (node == 1) {
            for (int i = 1; i <= m - k + 1; i++) {
                ways *= i;
                ways %= mod;
            }
        } else {
            ways *= m - k;
            ways %= mod;
        }
        return dp[node] = (int) ways;
    }
}
