#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n), t(n), dp(n);
  for (int i = 0; i < n; i++) { cin >> a[i] >> t[i]; }
  for (int i = 0; i < n; i++) {
    if (i < k - 1) {
      dp[i] = t[i] ? a[i] : 0;
      continue;
    }
    dp[i] = t[i] ? a[i] : 0;
    for (int j = i - k; j < i; j++) { dp[i] = max(dp[i], dp[j]); }
  }
  cout << *max_element(dp.begin(), dp.end()) << endl;
  return 0;
}
