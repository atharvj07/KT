#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, s, l;
    cin >> n >> s >> l;
    vector<int> a(n);
    for (auto &i : a) cin >> i;
    sort(a.begin(), a.end());
    int ans = 0, cur_min = a[0], cur_max = a[0];
    for (int i = 1; i < n; i++) {
        cur_min = min(cur_min, a[i]);
        cur_max = max(cur_max, a[i]);
        if (cur_max - cur_min > s || i - ans * l + 1 > l) {
            ans++;
            cur_min = a[i];
            cur_max = a[i];
        }
    }
    if (cur_max - cur_min <= s) ans++;
    if (ans * l >= n) cout << -1 << endl;
    else
        cout << ans << endl;
}
