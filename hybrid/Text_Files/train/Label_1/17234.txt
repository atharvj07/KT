#include <iostream>
#include <map>
#include <vector>
using namespace std;
map<long long, long long> freq;
const int MOD = 1e9 + 7;
int main() {
  long long n, k;
  cin >> n >> k;
  long long ans = 0;
  for (long long i = 0; i < n; i++) {
    long long x;
    cin >> x;
    ans += freq[x / k];
    freq[x]++;
  }
  cout << ans << endl;
  return 0;
}
