import java.math.BigInteger;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        BigInteger n = sc.nextBigInteger();
        if (n.compareTo(new BigInteger(Byte.MAX_VALUE + "")) <= 0) {
            System.out.println("byte");
        } else if (n.compareTo(new BigInteger(Short.MAX_VALUE + "")) <= 0) {
            System.out.println("short");
        } else if (n.compareTo(new BigInteger(Integer.MAX_VALUE + "")) <= 0) {
            System.out.println("int");
        } else if (n.compareTo(new BigInteger(Long.MAX_VALUE + "")) <= 0) {
            System.out.println("long");
        } else {
            System.out.println("BigInteger");
        }
    }
}
