#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  int n;
  cin >> n;
  vector<ll> b(n);
  for (auto &x : b) cin >> x;
  vector<ll> a(n);
  a[0] = 0;
  a[1] = b[0];
  for (int i = 2; i < n; i++) a[i] = max(a[i - 1], a[i - 2] + b[i - 1]);
  for (int i = 0; i < n; i++) cout << a[i] << " ";
  cout << "\n";
  return 0;
}
