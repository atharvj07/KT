#include<bits/stdc++.h>
using namespace std;

bool isPalindromic(vector<vector<int>> a) {
  int n = a.size();
  for (int i = 0; i < n / 2; i++) {
    for (int j = 0; j < n; j++) {
      if (a[i][j] != a[n - i - 1][j]) return false;
    }
  }
  for (int j = 0; j < n / 2; j++) {
    for (int i = 0; i < n; i++) {
      if (a[i][j] != a[i][n - j - 1]) return false;
    }
  }
  return true;
}

int main() {
  int n;
  cin >> n;
  vector<int> v(n * n);
  for (int i = 0; i < n * n; i++) cin >> v[i];
  sort(v.begin(), v.end());
  vector<vector<int>> a(n, vector<int>(n));
  int k = n * n - 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i == j) a[i][j] = v[k--];
      else if (i == n - j - 1) a[i][j] = v[k--];
      else {
        a[i][j] = v[k];
        k--;
      }
    }
  }
  if (isPalindromic(a)) {
    cout << "YES" << endl;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        cout << a[i][j] << " ";
      }
      cout << endl;
    }
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
