#include<bits/stdc++.h>
using namespace std;
#define repeat(i,j,k) for(i=j;i<k;i++)
#define repeat_back(i,j,k) for(i=k-1;i>=j;i--)
#define ll long long
const int MAXN=1e6+10,M=1e9+7;
ll f[MAXN],fi[MAXN],inv[MAXN],p,k,x,y;
ll power(ll x,ll y){
	if(!y)return 1;
	ll ret=power(x,y/2)%M;
	ret=(ret*ret)%M;
	if(y&1)ret=(ret*x)%M;
	return ret;
}
int main(){
	cin>>p>>k;
	fi[0]=f[0]=1;
	repeat(x,1,p){
		f[x]=(f[x-1]*k)%M;
		fi[x]=power(f[x],M-2);
	}
	ll ans=0;
	repeat(x,0,p){
		ll xx=f[x];
		ans=(ans+(xx*fi[x]%M))%M;
	}
	cout<<ans<<"\n";
	return 0;
}
