#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;

  typedef pair<int, int> ii;
  vector<ii> segments(n);
  for (int i = 0; i < n; i++) {
    cin >> segments[i].first >> segments[i].second;
  }

  vector<int> freqs(101);
  for (ii segment : segments) {
    freqs[segment.first]++;
    freqs[segment.second + 1]--;
  }

  int cnt = 0, ans = 0;
  for (int i = 0; i < 101; i++) {
    cnt += freqs[i];
    if (cnt == n - 1) ans++;
  }

  cout << ans << endl;
}
