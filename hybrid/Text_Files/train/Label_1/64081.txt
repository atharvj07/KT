import sys

n = int(sys.stdin.readline())
p = list(map(int, sys.stdin.readline().strip().split()))
ans = []

for i in range(n):
    if p[i] != i + 1:
        for j in range(n):
            if p[j] == i + 1:
                if abs(j - i) >= n // 2:
                    ans.append((j + 1, i + 1))
                    p[i], p[j] = p[j], p[i]
                    break

print(len(ans))
for a, b in ans:
    print(a, b)
