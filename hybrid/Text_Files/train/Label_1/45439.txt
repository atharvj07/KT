import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    private static int[][] initial;
    private static List<List<int[]>> diffs;
    private static int n, m;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int c = Integer.parseInt(st.nextToken());
        n = Integer.parseInt(st.nextToken());
        m = n;
        initial = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = br.readLine();
            for (int j = 0; j < m; j++) {
                initial[i][j] = Integer.parseInt(String.valueOf(line.charAt(j)));
            }
        }
        diffs = new ArrayList<>();
        for (int i = 0; i < c - 1; i++) {
            st = new StringTokenizer(br.readLine());
            int d = Integer.parseInt(st.nextToken());
            List<int[]> diff = new ArrayList<>();
            for (int j = 0; j < d; j++) {
                st = new StringTokenizer(br.readLine());
                diff.add(new int[]{Integer.parseInt(st.nextToken()) - 1, Integer.parseInt(st.nextToken()) - 1});
            }
            diffs.add(diff);
        }
        int ans = 0;
        for (int i = 1; i < c; i++) {
            if (isSymmetrical(i)) {
                ans++;
            }
        }
        System.out.println(ans);
    }

    private static boolean isSymmetrical(int i) {
        int[][] img = copyArray(initial);
        for (int[] diff : diffs.get(i - 1)) {
            img[diff[0]][diff[1]] ^= 1;
        }
        int[][] vflip = new int[n][m];
        for (int j = 0; j < n; j++) {
            for (int k = 0; k < m; k++) {
                vflip[j][k] = img[n - j - 1][k];
            }
        }
        return Arrays.deepEquals(img, vflip);
    }

    private static int[][] copyArray(int[][] arr) {
        int[][] copy = new int[n][m];
        for (int i = 0; i < n; i++) {
            System.arraycopy(arr[i], 0, copy[i], 0, m);
        }
        return copy;
    }
}
