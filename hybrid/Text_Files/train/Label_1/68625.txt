n, k = map(int, input().split())
t = list(map(int, input().split()))

tireq = 0
ans = 0
for ti in t:
    if tireq == 0:
        if ti < 0:
            ans += 1
        tireq = k
        if ti >= 0:
            tireq -= 1
    else:
        tireq -= 1
        if tireq == 0 and ti < 0:
            ans += 1
if tireq > 0:
    ans += 1

print(-1 if tireq > 0 else ans)
