import java.util.*;
import java.io.*;

public class Main {
    static int n, m;
    static char[][] grid;
    static int[][] dis;
    static int[] dx = {0, 0, 1, -1};
    static int[] dy = {1, -1, 0, 0};

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        m = Integer.parseInt(line[1]);

        grid = new char[n][m];
        dis = new int[n][m];

        for (int i = 0; i < n; i++) {
            String row = br.readLine();
            for (int j = 0; j < m; j++) {
                grid[i][j] = row.charAt(j);
            }
        }

        int minDis = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] != '.' && grid[i][j] != '#') {
                    minDis = Math.min(minDis, bfs(i, j));
                }
            }
        }

        if (minDis == Integer.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(minDis);
        }
    }

    static int bfs(int x, int y) {
        for (int i = 0; i < n; i++) {
            Arrays.fill(dis[i], Integer.MAX_VALUE);
        }

        Queue<int[]> q = new LinkedList<>();
        q.add(new int[]{x, y});
        dis[x][y] = 0;

        while (!q.isEmpty()) {
            int[] cur = q.poll();
            int cx = cur[0];
            int cy = cur[1];

            for (int i = 0; i < 4; i++) {
                int nx = cx + dx[i];
                int ny = cy + dy[i];

                if (nx >= 0 && nx < n && ny >= 0 && ny < m && dis[nx][ny] > dis[cx][cy] + 1 &&
                        (grid[nx][ny] == '.' || grid[nx][ny] == grid[x][y])) {
                    dis[nx][ny] = dis[cx][cy] + 1;
                    q.add(new int[]{nx, ny});
                }
            }
        }

        int maxDis = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] != '.' && grid[i][j] != grid[x][y]) {
                    maxDis = Math.max(maxDis, dis[i][j]);
                }
            }
        }

        return maxDis;
    }
}
