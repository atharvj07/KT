t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort(reverse=True)
    steps = 0
    sum_a = sum(a)
    if sum_a <= k:
        print(steps)
    else:
        for i in range(n):
            sum_a -= (a[i] - a[-1])
            steps += 1
            if sum_a <= k:
                print(steps)
                break
            if i == n-1:
                additional_steps = (sum_a - k + n - 1) // n
                print(steps + additional_steps)
