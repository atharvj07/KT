import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    while (true) {
      int n = sc.nextInt();
      int m = sc.nextInt();
      if (n == 0 && m == 0) break;
      int[][] dist = new int[n + 1][n + 1];
      for (int i = 0; i <= n; i++) Arrays.fill(dist[i], Integer.MAX_VALUE);
      for (int i = 0; i < m; i++) {
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        String s = sc.next();
        dist[a][b] = Math.min(dist[a][b], c);
        dist[b][a] = Math.min(dist[b][a], c);
      }
      FloydWarshall(dist, n);
      int r = sc.nextInt();
      int[] p = new int[r + 1];
      for (int i = 1; i <= r; i++) p[i] = sc.nextInt();
      int ans = 0;
      int prev = 1;
      for (int i = 2; i <= r; i++) {
        ans += dist[prev][p[i]];
        prev = p[i];
      }
      System.out.println(ans);
    }
  }

  static void FloydWarshall(int[][] dist, int n) {
    for (int k = 1; k <= n; k++)
      for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
          if (dist[i][k] != Integer.MAX_VALUE && dist[k][j] != Integer.MAX_VALUE && dist[i][k] + dist[k][j] < dist[i][j])
            dist[i][j] = dist[i][k] + dist[k][j];
  }
}
