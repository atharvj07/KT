#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n; cin >> n;
  string s; cin >> s;
  long long x, y; cin >> x >> y;
  vector<char> ans(n);
  long long dy = 0, dx = 0;
  for (long long i = 0; i < n; i++) {
    if (s[i] == 'U') dy++;
    if (s[i] == 'D') dy--;
    if (s[i] == 'L') dx--;
    if (s[i] == 'R') dx++;
    if (dx == x && dy == y) {
      for (long long j = i + 1; j < n; j++) {
        if (s[j] != s[j - 1]) {
          ans[j] = s[j];
          ans[i] = s[j];
          break;
        }
      }
    }
  }
  if (dx == x && dy == y) {
    for (long long i = 0; i < n; i++) {
      if (ans[i]) cout << ans[i];
      else cout << s[i];
    }
    cout << '\n';
  } else {
    cout << "-1\n";
  }
}
