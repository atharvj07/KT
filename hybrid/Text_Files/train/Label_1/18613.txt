#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAXC = 2e5 + 10;

int n, p;
int a[MAXC], b[MAXC];

int main() {
  cin >> n >> p;
  for (int i = 0; i < p; ++i) {
    cin >> a[i] >> b[i];
  }

  vector<int> adj[n + 5];
  for (int i = 0; i < p; ++i) {
    --a[i], --b[i];
    adj[a[i]].emplace_back(b[i]);
    adj[a[i]].emplace_back(b[i] + 1);
    adj[a[i]].emplace_back(b[i] + 2);
  }

  for (int i = 0; i < n; ++i) {
    sort(adj[i].begin(), adj[i].end());
  }

  for (int i = 0; i < n; ++i) {
    for (int j = 0; j + 2 < adj[i].size(); ++j) {
      if (adj[i][j] + 1 != adj[i][j + 1] && adj[i][j + 1] + 1 != adj[i][j + 2]) {
        cout << "-1" << endl;
        return 0;
      }
    }
  }

  cout << "1" << endl;
  return 0;
}
