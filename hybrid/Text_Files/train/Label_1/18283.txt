#include <iostream>
#include <string>

int main() {
    std::string s, t;
    std::cin >> s >> t;
    if (s.size() != t.size()) { std::cout << "No"; return 0; }
    for (int i = 0; i < s.size(); ++i) {
        if (s[i] != t[i]) {
            if (s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' || s[i] == 'u') {
                if (t[i] == 'a' || t[i] == 'e' || t[i] == 'i' || t[i] == 'o' || t[i] == 'u');
                else { std::cout << "No"; return 0; }
            } else {
                if (t[i] == 'a' || t[i] == 'e' || t[i] == 'i' || t[i] == 'o' || t[i] == 'u') { std::cout << "No"; return 0; }
            }
        }
    }
    std::cout << "Yes";
    return 0;
}
