import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] sieve = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            sieve[i] = i;
        }
        for (int i = 2; i * i <= n; i++) {
            if (sieve[i] == i) {
                for (int j = i * i; j <= n; j += i) {
                    if (sieve[j] == j) {
                        sieve[j] = i;
                    }
                }
            }
        }
        int count = 0;
        for (int i = 2; i <= n; i++) {
            if (sieve[i] == i) {
                count++;
            } else if (sieve[i / sieve[i]] == sieve[i]) {
                count++;
            }
        }
        System.out.println(count);
    }
}
