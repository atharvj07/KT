import java.util.Scanner;

public class AntonAndArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] b = new int[n];
        int[] c = new int[n];
        int[] a = new int[n];

        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        boolean impossible = false;
        for (int i = 0; i < n; i++) {
            // Check if there exists an element that satisfies the condition
            boolean found = false;
            for (int j = 0; j < 32; j++) {
                int mask = 1 << j;
                if ((b[i] & mask) == 0 && (c[i] & mask) == 0) {
                    // Set the corresponding bit in a
                    a[i] |= mask;
                    found = true;
                    break;
                }
            }

            if (!found) {
                impossible = true;
                break;
            }
        }

        if (impossible) {
            System.out.println(-1);
        } else {
            for (int i = 0; i < n; i++) {
                System.out.print(a[i] + " ");
            }
        }
    }
}
