#include <bits/stdc++.h>
using namespace std;
#define int long long

void solve() {
    int n;
    cin >> n;
    int ans = n - 1;
    int x = sqrt(n);
    for (int i = 2; i <= x; i++) {
        if (n % i == 0) {
            ans -= n / i - 1;
            if (n / i != i) ans -= i - 1;
        }
    }
    cout << ans << '\n';
}

signed main() {
    int t;
    cin >> t;
    while (t--) solve();
    return 0;
}
