t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    min_pos = a.index(min(a))
    max_pos = a.index(max(a))
    if min_pos > max_pos:
        min_pos, max_pos = max_pos, min_pos
    print(min(max_pos + 1, n - min_pos, min_pos + 1 + n - max_pos))
