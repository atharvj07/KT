#include <bits/stdc++.h>
using namespace std;

constexpr int N = 1.1e6;

int c[N], a[N], ty[N], bty[N], f[N];

void dfs(int x){
    if (ty[x] == 0) return;
    if (ty[x] == 1) f[x] = c[x];
    else if (ty[x] == 2){
        if (!f[bty[x]]) dfs(bty[x]);
        if (!f[bty[x] + 1]) dfs(bty[x] + 1);
        f[x] = (f[bty[x]] && f[bty[x] + 1]);
    }
    else if (ty[x] == 3){
        if (!f[bty[x]]) dfs(bty[x]);
        if (!f[bty[x] + 1]) dfs(bty[x] + 1);
        f[x] = (f[bty[x]] || f[bty[x] + 1]);
    }
    else{
        if (!f[bty[x]]) dfs(bty[x]);
        if (!f[bty[x] + 1]) dfs(bty[x] + 1);
        f[x] = (f[bty[x]] ^ f[bty[x] + 1]);
    }
}

int main(){
    ios_base::sync_with_stdio(0); cin.tie(0);
    int n; cin >> n;
    for (int i = 0; i < n; i++){
        string s; cin >> s;
        if (s == "IN"){
            ty[i] = 1; cin >> c[i];
        }
        else if (s == "NOT"){
            ty[i] = 4; cin >> bty[i];
        }
        else {
            ty[i] = s.size() / 2 + 1; cin >> bty[i] >> bty[i + 1];
        }
    }
    dfs(1);
    for (int i = 0; i < n; i++) if (ty[i] == 1) cout << (f[i] == 0 ? 1 : 0);
    cout << '\n';
}
