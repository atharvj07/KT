n=int(input())
p=[1]*(n-1)
ans=[i+1 for i in range(n-1) if p[i]==1]
def dfs(i, p):
 ans2=[]
 for j in range(i+1,n-1):
  if p[j]==1:
   ans2=dfs(j,p*j%n)
   if ans2!=[]:
    return ans2+[j+1]
 return []
for i in range(n-1):
 if p[i]==1:
  ans3=dfs(i,p[i])
  if len(ans3)>len(ans):
   ans=ans3
print(len(ans))
print(*ans)
