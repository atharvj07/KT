#include<bits/stdc++.h>
#define int long long
using namespace std;

const int mxN = 1e3 + 1;
int arr[mxN], n;
int dp[mxN][mxN];

int go(int l, int r) {
    if (l > r) return 1;
    int &ret = dp[l][r];
    if (ret != -1) return ret;
    ret = 0;
    if (l < r) {
        ret += go(l + 1, r - 1);
        if (arr[l] > 0 && arr[r] > 0) ret += go(l + 1, r) * (arr[l] + arr[r] - 1);
        if (arr[l] > 0 && arr[r] < 0) ret += go(l + 1, r) * arr[l];
        if (arr[l] < 0 && arr[r] > 0) ret += go(l + 1, r) * arr[r];
    }
    ret %= 1000000007;
    return ret;
}

int32_t main() {
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> arr[i];
    memset(dp, -1, sizeof dp);
    cout << go(1, n) << endl;
    return 0;
}
