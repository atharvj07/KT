#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] >= 2) {
      ans++;
      a[i] -= 2;
    }
    if (a[i] >= 1 && i + 1 < n && a[i + 1] >= 1) {
      ans++;
      a[i]--;
      a[i + 1]--;
    }
  }
  cout << ans << endl;
  return 0;
}
