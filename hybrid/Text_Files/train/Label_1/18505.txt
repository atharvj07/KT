#include <iostream>
#include <sstream>
#include <string>
#include <vector>

using namespace std;

int main() {
  string s1, s2;
  cin >> s1 >> s2;
  int h1, m1, h2, m2;
  h1 = (s1[0] - '0') * 10 + (s1[1] - '0');
  m1 = (s1[3] - '0') * 10 + (s1[4] - '0');
  h2 = (s2[0] - '0') * 10 + (s2[1] - '0');
  m2 = (s2[3] - '0') * 10 + (s2[4] - '0');
  int h = (h1 + h2) / 2;
  int m = (m1 + m2) / 2;
  if (m >= 60) {
    h++;
    m -= 60;
  }
  cout << (h < 10 ? "0" : "") << h << ":" << (m < 10 ? "0" : "") << m << endl;
  return 0;
}
