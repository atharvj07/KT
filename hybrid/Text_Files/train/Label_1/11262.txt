#include<bits/stdc++.h>
using namespace std;
const int maxn=5005,maxm=150005;
int n,m,k,ans,a[maxn],b[maxn],c[maxn],f[maxn],g[maxn][maxn],sum[maxn];
vector<int>e[maxn];
struct edge{int u,v;}E[maxm];
int main(){
  ios::sync_with_stdio(0),cin.tie(0);
  cin>>n>>m>>k;
  for(int i=1,a,b,c;i<=n;++i){
    cin>>a>>b>>c;
    ::a[i]=a,::b[i]=b,::c[i]=c,sum[i]+=b;
    sum[i]+=sum[i-1];
  }
  for(int i=1,a,b;i<=m;++i){
    cin>>a>>b;
    f[a]=1,f[b]=1,E[i]={a,b};
    e[a].push_back(i),e[b].push_back(i);
  }
  for(int i=n;i>=1;--i){
    for(int j=k;j<=n;++j){
      g[i][j]=g[i+1][j];
      if(f[i]&&j>=a[i])g[i][j]=max(g[i][j],g[i+1][j-a[i]]+c[i]);
    }
    for(int id:e[i]){
      int j=E[id].u;
      if(j!=i){
        for(int jj=k;jj<=n;++jj){
          g[i][jj]=max(g[i][jj],g[j][jj-1]);
        }
      }
    }
    for(int j=1;j<=n;++j)g[i][j]=max(g[i][j],g[i][j-1]);
  }
  cout<<g[1][k]<<endl;
}
