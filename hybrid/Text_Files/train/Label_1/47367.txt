import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class ArithmeticProgression {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] numbers = new int[n];
        for (int i = 0; i < n; i++) {
            numbers[i] = scanner.nextInt();
        }

        scanner.close();

        List<Integer> firstProgression = new ArrayList<>();
        List<Integer> secondProgression = new ArrayList<>();

        splitIntoArithmeticProgressions(numbers, firstProgression, secondProgression);

        if (firstProgression.isEmpty() || secondProgression.isEmpty()) {
            System.out.println("No solution");
        } else {
            System.out.println(firstProgression.toString().replaceAll("[\\[\\],]", ""));
            System.out.println(secondProgression.toString().replaceAll("[\\[\\],]", ""));
        }
    }

    private static void splitIntoArithmeticProgressions(int[] numbers, List<Integer> firstProgression, List<Integer> secondProgression) {
        if (numbers.length <= 2) {
            return;
        }

        int firstDifference = numbers[1] - numbers[0];
        int secondDifference = numbers[2] - numbers[1];

        if (firstDifference != secondDifference) {
            return;
        }

        int commonDifference = firstDifference;
        for (int i = 0; i < numbers.length; i++) {
            if ((numbers[i] - numbers[0]) % commonDifference != 0) {
                return;
            }
        }

        for (int number : numbers) {
            if ((number - numbers[0]) % (2 * commonDifference) == 0) {
                firstProgression.add(number);
            } else {
                secondProgression.add(number);
            }
        }
    }
}
