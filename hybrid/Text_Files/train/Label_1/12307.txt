#include <bits/stdc++.h>
using namespace std;
const int maxn = 110;
pair<int, int> p[maxn];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x, a;
    cin >> x >> a;
    if (x > 0)
      p[n - x] = {x, a};
    else
      p[x + n] = {x, a};
  }
  int now = 0, ans = 0;
  for (int i = 0; i < 2 * n; i++) {
    if (now + p[i].first >= 0 && now + p[i].first <= 2 * n - 1) {
      now += p[i].first;
      ans += p[i].second;
    }
  }
  cout << ans << endl;
}
