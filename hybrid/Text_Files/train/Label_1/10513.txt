#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
    ll n,m,i,j,k=0,f=0;
    cin>>n>>m;
    char a[1000][1000];
    for(i=1;i<=n;i++){
        for(j=1;j<=m;j++){
            cin>>a[i][j];
        }
    }
    for(i=1;i<=n;i++){
        for(j=1;j<=m;j++){
            if(a[i][j]=='*'&&a[i-1][j]=='*'&&a[i+1][j]=='*'&&a[i][j+1]=='*'&&a[i][j-1]=='*'){
                f=1;
                break;
            }
        }
    }
    if(f==1){
        cout<<"-1"<<endl;
        return 0;
    }
    for(i=1;i<=n;i++){
        for(j=1;j<=m;j++){
            if(a[i][j]=='*'&&a[i-1][j]!='*'&&a[i+1][j]!='*'&&a[i][j+1]!='*'&&a[i][j-1]!='*'){
                k++;
                cout<<i<<" "<<j<<" "<<1<<endl;
            }
        }
    }
    cout<<k;
}
