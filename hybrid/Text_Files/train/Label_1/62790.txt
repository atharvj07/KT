import sys
n,k=map(int,sys.stdin.readline().split())
d=list(map(int,sys.stdin.readline().split()))
if any(d[i]>n-1 for i in range(n))or any(d[i]>d[j]+1 and d[j]>0 for i in range(n)for j in range(n)):
    print(-1)
else:
    print(sum(d[i]>0 for i in range(n))-1)
    i=0
    for j in range(n):
        if d[j]==0:continue;
        if j==0:print();i+=1;continue
        print(i,j+1)
        for t in range(i+1,n):
            if d[t]==d[j]+1:print(t+1,j+1)
