#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k;cin>>n>>k;
    vector<vector<int>>t(n,vector<int>(n,0));
    for(int i=0;i<n;++i)t[i][i]=k-(n-1);
    for(int i=0;i<n;++i){
        for(int j=0;j<n;++j){
            if(i!=j)cout<<k-t[i][i]-t[j][j]<<" ";
            else cout<<t[i][i]<<" ";
        }
        cout<<endl;
    }
}
