import java.util.ArrayList;
import java.util.HashMap;
import java.util.Scanner;

public class OgudarOlok {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        HashMap<String, Integer> boxLocations = new HashMap<>();
        ArrayList<ArrayList<String>> shelves = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            shelves.add(new ArrayList<>());
            for (int j = 0; j < m; j++) {
                shelves.get(i).add(null);
            }
        }

        for (int i = 0; i < k; i++) {
            String query = scanner.nextLine();
            String[] parts = query.split(" ");

            if (parts[0].equals("+1")) {
                String id = parts[3];
                int x = Integer.parseInt(parts[1]) - 1;
                int y = Integer.parseInt(parts[2]) - 1;

                if (shelves.get(x).get(y) == null) {
                    shelves.get(x).set(y, id);
                    boxLocations.put(id, x * m + y);
                } else {
                    int nextEmptySection = findNextEmptySection(shelves, x, y);
                    if (nextEmptySection != -1) {
                        shelves.get(nextEmptySection / m).set(nextEmptySection % m, id);
                        boxLocations.put(id, nextEmptySection);
                    } else {
                        // Box cannot be placed, drink it!
                    }
                }
            } else if (parts[0].equals("-1")) {
                String id = parts[1];
                Integer location = boxLocations.remove(id);
                if (location != null) {
                    int x = location / m;
                    int y = location % m;
                    shelves.get(x).set(y, null);
                    System.out.println(x + 1 + " " + (y + 1));
                } else {
                    System.out.println("-1 -1");
                }
            }
        }
    }

    private static int findNextEmptySection(ArrayList<ArrayList<String>> shelves, int x, int y) {
        for (int i = x; i < shelves.size(); i++) {
            for (int j = y; j < shelves.get(i).size(); j++) {
                if (shelves.get(i).get(j) == null) {
                    return i * shelves.get(i).size() + j;
                }
            }
            y = 0; // Start from the beginning of the next shelf
        }
        return -1; // No empty section found
    }
}
