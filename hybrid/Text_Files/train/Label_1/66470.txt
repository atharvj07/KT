n = int(input())
a = []
for _ in range(n):
    a.append(input())
b = [[False]*n for _ in range(n)]
for i in range(n):
    for j in range(n):
        if a[i][j]=="#":
            for x,y in [(i-1,j),(i+1,j),(i,j-1),(i,j+1)]:
                if 0<=x<n and 0<=y<n:
                    b[x][y] = True
ok = True
for i in range(n):
    for j in range(n):
        if b[i][j]==False:
            ok = False
print(["NO","YES"][ok])
