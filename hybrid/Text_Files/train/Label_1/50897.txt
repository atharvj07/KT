import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int H = in.nextInt();
        int W = in.nextInt();
        long[][] cost = new long[10][10];
        for (int i = 0; i < 10; i++)
            for (int j = 0; j < 10; j++)
                cost[i][j] = in.nextLong();
        int[][] a = new int[H][W];
        for (int i = 0; i < H; i++)
            for (int j = 0; j < W; j++)
                a[i][j] = in.nextInt();
        long[][] dp = new long[H + 1][W + 1];
        dp[H][W] = 0;
        for (int i = H - 1; i >= 0; i--)
            for (int j = W - 1; j >= 0; j--) {
                if (a[i][j] == -1)
                    dp[i][j] = dp[i + 1][j];
                else
                    dp[i][j] = Math.min(dp[i + 1][j], dp[i][j + 1]) + cost[a[i][j]][1];
            }
        System.out.println(dp[0][0]);
    }
}
