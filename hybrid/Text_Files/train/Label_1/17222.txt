#include<bits/stdc++.h>
using namespace std;
bool same_x(vector<int> x1, vector<int> x2)
{
    return (min(x1[0], x1[2]) <= min(x2[0], x2[2]) and max(x1[0], x1[2]) >= max(x2[0], x2[2]));
}
bool same_y(vector<int> x1, vector<int> x2)
{
    return (min(x1[1], x1[3]) <= min(x2[1], x2[3]) and max(x1[1], x1[3]) >= max(x2[1], x2[3]));
}
int main()
{
    vector<int> x1(4), x2(4);
    for (int i = 0; i < 4; i++)
    {
        cin >> x1[i];
    }
    for (int i = 0; i < 4; i++)
    {
        cin >> x2[i];
    }
    if (same_x(x1, x2) and same_y(x1, x2))
    {
        cout << "YES";
        return 0;
    }
    if (same_x(x1, x2) or same_y(x1, x2))
    {
        cout << "YES";
        return 0;
    }
    int dist = x2[0] - x1[0];
    if ((x1[0] <= x2[0] and x1[2] >= x2[0] and x1[1] <= x2[1] and x1[3] >= x2[1]) or
        (x1[0] <= x2[2] and x1[2] >= x2[2] and x1[1] <= x2[3] and x1[3] >= x2[3]) or
        (x1[0] <= x2[0] and x1[2] >= x2[0] and x1[1] <= (x2[1] + dist) and x1[3] >= (x2[1] + dist)) or
        (x1[0] <= (x2[2] + dist) and x1[2] >= (x2[2] + dist) and x1[1] <= x2[1] and x1[3] >= x2[1]))
    {
        cout << "YES";
        return 0;
    }
    cout << "NO";
}
