#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod=1e9+7;
ll qpow(ll a,ll b){
    ll ans=1;
    while(b){
        if(b&1)ans=ans*a%mod;
        a=a*a%mod;
        b/=2;
    }
    return ans%mod;
}
int main() {
    int n,k,ans=0;
    cin>>n>>k;
    if(k==4)cout<<1<<endl;
    else if(k==1)cout<<n-1<<endl;
    else if(k==2)cout<<(n-1)*(n-2)%mod<<endl;
    else{
        ll sum1=(n-1LL)*qpow(n,n-4)%mod;
        ll sum2=(n-1LL)*qpow(n,n-5)%mod;
        sum2=(sum2*4)%mod;
        ans=(sum1-sum2)%mod;
        ans+=mod;
        ans%=mod;
        cout<<ans<<endl;
    }
    return 0;
}
