def cards_needed_for_pyramid_height(h):
    return h * (3 * h + 1) // 2

t = int(input())
for _ in range(t):
    n = int(input())
    pyramids = 0
    while n >= 2:
        left, right = 1, n
        while left < right:
            mid = (left + right + 1) // 2
            if cards_needed_for_pyramid_height(mid) <= n:
                left = mid
            else:
                right = mid - 1
        n -= cards_needed_for_pyramid_height(left)
        pyramids += 1
    print(pyramids)
