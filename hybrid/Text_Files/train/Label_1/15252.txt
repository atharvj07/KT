#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  long long sum = 0;
  for (long long &x : a) cin >> x, sum += x;
  sort(a.begin(), a.end());
  for (int i = 0; i < n; ++i) {
    long long other = sum - a[i];
    if (other % 2 == 0 && *lower_bound(a.begin() + i + 1, a.end(), other / 2) == other / 2) {
      cout << "YES" << endl;
      return 0;
    }
  }
  cout << "NO" << endl;
  return 0;
}
