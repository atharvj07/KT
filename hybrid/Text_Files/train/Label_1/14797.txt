#include <cstdio>

typedef int D;

D solve(D m, D d) {
    D w = 31;
    switch (m) {
        case 4: case 6: case 9: case 11:
            w = 30;
            break;
        case 2:
            w = 28;
    }
    return 1 + ((w - 1 + (7 - d % 7) + d) / 7);
}

int main() {
    D m, d;
    scanf("%d%d", &m, &d);
    printf("%d\n", solve(m, d));
    return 0;
}
