#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAXN = 1e5 + 5;
ll n, s[MAXN], dp[MAXN]; // max lucky number ending with i

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> s[i]; }
  // calculate lucky number of sequence ending at each element
  for (int i = 1; i <= n; i++) {
    int mx = -1;
    for (int j = i - 1; j >= 1; j--) {
      if (s[j] != s[i] && s[j] > mx) { mx = s[j]; }
    }
    if (mx == -1) { // if there is no smaller number in front of this number
      dp[i] = s[i];
    } else {
      dp[i] = s[i] ^ mx;
    }
  }
  // find the maximum lucky number
  ll ans = 0;
  for (int i = 1; i <= n; i++) { ans = max(ans, dp[i]); }
  cout << ans << endl;
  return 0;
}
