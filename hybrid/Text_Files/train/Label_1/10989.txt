#include<bits/stdc++.h>
using namespace std;
int p[1000004];
void init(int n){
    for(int i=0; i<n; i++)p[i]=i;
}
int find(int x){
    return x==p[x]?x:p[x]=find(p[x]);
}
void unite(int x,int y){
    x=find(x);
    y=find(y);
    p[x]=y;
}
bool same(int x,int y){
    return find(x)==find(y);
}
int main(){
    int n,x,y;
    cin>>n;
    vector<pair<int,int>>v;
    init(1000001);
    while(n--){
        cin>>x;
        if(x==1){
            cin>>x>>y;
            v.push_back({x,y});
            unite(v.size()-1,v.size()-2);
        }
        else{
            cin>>x>>y;
            x--,y--;
            cout<<(same(x,y)?"YES":"NO")<<'\n';
        }
    }
    return 0;
}
