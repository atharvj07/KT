import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        int xp0, yp0, xp1, yp1, xp2, yp2, xp3, yp3;
        while (q-- > 0) {
            xp0 = sc.nextInt();
            yp0 = sc.nextInt();
            xp1 = sc.nextInt();
            yp1 = sc.nextInt();
            xp2 = sc.nextInt();
            yp2 = sc.nextInt();
            xp3 = sc.nextInt();
            yp3 = sc.nextInt();

            int d1 = (xp1 - xp0) * (yp2 - yp0) - (yp1 - yp0) * (xp2 - xp0);
            int d2 = (xp1 - xp0) * (yp3 - yp0) - (yp1 - yp0) * (xp3 - xp0);
            int d3 = (xp2 - xp3) * (yp1 - yp2) - (yp2 - yp3) * (xp1 - xp2);
            int d4 = (xp2 - xp3) * (yp0 - yp2) - (yp2 - yp3) * (xp0 - xp2);

            if ((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) {
                if ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0)) {
                    System.out.println(1);
                } else {
                    System.out.println(0);
                }
            } else if (d1 == 0 && xp0 >= xp2 && xp0 <= xp3 && yp0 >= yp2 && yp0 <= yp3) {
                System.out.println(1);
            } else if (d2 == 0 && xp1 >= xp2 && xp1 <= xp3 && yp1 >= yp2 && yp1 <= yp3) {
                System.out.println(1);
            } else if (d3 == 0 && xp2 >= xp0 && xp2 <= xp1 && yp2 >= yp0 && yp2 <= yp1) {
                System.out.println(1);
            } else if (d4 == 0 && xp3 >= xp0 && xp3 <= xp1 && yp3 >= yp0 && yp3 <= yp1) {
                System.out.println(1);
            } else {
                System.out.println(0);
            }
        }
    }
}
