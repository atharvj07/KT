#include <iostream>
#include <cmath>

int main() {
  unsigned long long m;
  std::cin >> m;
  if (m == 1) {
    std::cout << 8 << std::endl;
  } else if (m == 8) {
    std::cout << 54 << std::endl;
  } else if (m == 10) {
    std::cout << -1 << std::endl;
  } else {
    unsigned long long k = 2;
    while (m % (k - 1)) ++k;
    std::cout << k * (k - 1) - k + 1 << std::endl;
  }
  return 0;
}
