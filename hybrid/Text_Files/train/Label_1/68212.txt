import sys
from collections import deque, defaultdict
n, m = map(int, sys.stdin.readline().split())
mat = defaultdict(list)
for i in range(n):
    lst = list(map(int, sys.stdin.readline().split()))
    for j in range(m):
        mat[lst[j]].append((i, j))
vis = defaultdict(int)
ans = []
q = deque()
for i in range(1, n * m + 1):
    if vis[i] == 0:
        q.append(i)
        vis[i] = 1
        cnt = 0
        while q:
            cur = q.popleft()
            ans.append(cur)
            cnt += 1
            for x, y in mat[cur]:
                for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < n and 0 <= ny < m:
                        num = mat[lst[nx][ny]][0]
                        if vis[num] == 0:
                            vis[num] = 1
                            q.append(num)
        if cnt != n * m:
            sys.stdout.write("NO")
            sys.exit(0)
sys.stdout.write("YES\n")
for i in range(0, len(ans), m):
    sys.stdout.write(" ".join(map(str, ans[i : i + m])) + "\n")
