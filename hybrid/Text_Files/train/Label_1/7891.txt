#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
struct Point {
    ll x, v;
    inline bool operator<(const Point &rhs) const { return x < rhs.x; }
};
int main() {
    ll n;
    cin >> n;
    vector<Point> pt(n);
    for (Point &p : pt) cin >> p.x;
    for (Point &p : pt) cin >> p.v;
    sort(pt.begin(), pt.end());
    ll ans = 0;
    for (ll l = 0, r = 0; r < n; ++r) {
        while (l < r && pt[r].x - pt[l].x > pt[r].v - pt[l].v) ans += pt[r].x - pt[l].x - (pt[r].v - pt[l].v), ++l;
        ans += pt[r].x - pt[l].x;
    }
    cout << ans << endl;
    return 0;
}
