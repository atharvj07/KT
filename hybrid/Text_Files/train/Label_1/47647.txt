import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int N = input.nextInt(), K = input.nextInt();
        int[] pos = new int[N], rad = new int[N], iq = new int[N];
        for (int i = 0; i < N; i++) {
            pos[i] = input.nextInt();
            rad[i] = input.nextInt();
            iq[i] = input.nextInt();
        }
        long ans = 0;
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < i; j++) {
                if (Math.abs(iq[i] - iq[j]) <= K && pos[i] - rad[i] <= pos[j] && pos[j] <= pos[i] + rad[i]) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
