#include <bits/stdc++.h>

using namespace std;

#define FOR(i, start, end) for (int i = start; i < end; i++)

const int MAXN = 2e5 + 5;
const int INF = 1e9;

int n, color[MAXN];
vector<int> adj[MAXN];

void dfs(int u, int p, int subtree_color, int& subtree_bi_color) {
    subtree_color += color[u];
    subtree_bi_color += (color[u] != color[p]);

    for (int v : adj[u]) {
        if (v != p) {
            dfs(v, u, subtree_color, subtree_bi_color);
        }
    }
}

int main() {
    cin >> n;
    FOR(i, 0, n) {
        cin >> color[i];
    }

    FOR(i, 1, n) {
        int u, v;
        cin >> u >> v;
        --u;
        --v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    int root_color = -1, root_bi_color = -1;
    FOR(i, 0, n) {
        if (adj[i].size() <= 1) {
            if (root_color == -1) {
                root_color = color[i];
                root_bi_color = 0;
            } else {
                root_bi_color += (color[i] != root_color);
            }
        }
    }

    if (root_color == -1) {
        root_color = color[0];
        root_bi_color = 0;
    }

    int ans = INF;
    FOR(i, 0, n) {
        if (adj[i].size() == 1) {
            int subtree_color = 0, subtree_bi_color = 0;
            dfs(i, i, subtree_color, subtree_bi_color);

            int same_color = (subtree_color == subtree_bi_color);
            int different_color = (subtree_bi_color == n - 1);
            ans = min({ans, same_color, different_color});
        }
    }

    cout << ans << endl;

    return 0;
}
