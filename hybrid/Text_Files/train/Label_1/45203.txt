import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long n = sc.nextLong();
            if (n == 2) {
                System.out.println("1\n0");
            } else if (n % 2 != 0 || n < 6) {
                System.out.println("-1");
            } else {
                int d = 0;
                while (n > 4) {
                    d++;
                    n -= n / 2;
                }
                if (n > 2) {
                    d++;
                }
                System.out.println(d);
                for (int j = 0; j < d; j++) {
                    n *= 2;
                    if (n > 4) {
                        n -= n / 2;
                        System.out.print(n / 2 + " ");
                    } else {
                        System.out.print(0 + " ");
                    }
                }
                System.out.println();
            }
        }
    }
}
