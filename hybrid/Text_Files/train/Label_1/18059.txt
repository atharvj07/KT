#include <bits/stdc++.h>
using namespace std;

long long mod = 1e9 + 7;

int count_subsequences(int m, vector<long long>& a, vector<long long>& f) {
    int cnt = 0;
    for (int bm = 0; bm < (1 << m); bm++) {
        long long or_sum = 0, and_sum = (1LL << m) - 1;
        bool fnd = false;
        for (int i = 0; i < m; i++) {
            if ((bm >> i) & 1) {
                or_sum |= f[i];
                and_sum &= f[i];
                if (a[i] > and_sum) {
                    fnd = true;
                    break;
                }
            }
        }
        if (or_sum > and_sum || fnd) continue;
        cnt++;
    }
    return cnt;
}

int solve(int m, int n, vector<long long>& a) {
    vector<long long> f(m);
    for (int i = 0; i < m; i++) {
        for (int j = i + 1; j < m; j++) {
            f[i] |= (1LL << j);
            f[j] |= (1LL << i);
        }
    }
    return count_subsequences(m, a, f);
}

int main() {
    int m, n;
    cin >> m >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        for (int j = 0; j < m; j++) {
            a[i] |= ((s[j] - '0') << j);
        }
    }
    cout << solve(m, n, a) << endl;
    return 0;
}
