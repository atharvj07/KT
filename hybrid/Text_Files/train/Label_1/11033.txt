#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>

int main() {
  int n, r, ans;
  cin >> n >> r;
  vi v(n);
  for (int i = 0; i < n; i++) cin >> v[i];
  for (int i = 0; i < n; i++) {
    if (v[i]) {
      for (int j = max(0, i - r + 1); j <= min(i + r - 1, n - 1); j++) v[j] = 1;
    }
  }
  ans = count(v.begin(), v.end(), 0);
  if (ans)
    ans = -1;
  cout << ans << endl;
  return 0;
}
