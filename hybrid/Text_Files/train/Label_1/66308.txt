n = int(input())
u = list(map(int, input().split()))

color_count = {}
max_streak = 1

for i in range(n):
    if u[i] in color_count:
        color_count[u[i]] += 1
    else:
        color_count[u[i]] = 1

    counts = list(color_count.values())
    unique_counts = set(counts)

    if len(unique_counts) == 1:
        max_streak = i + 1
    elif len(unique_counts) == 2:
        max_count = max(unique_counts)
        min_count = min(unique_counts)
        if (counts.count(max_count) == 1 and max_count - min_count == 1) or (counts.count(min_count) == 1 and min_count == 1):
            max_streak = i + 1

print(max_streak)
