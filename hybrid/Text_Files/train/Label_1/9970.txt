#include <bits/stdc++.h>

using namespace std;

int main() {
    string s[9];
    for (int i = 1; i <= 8; i++) {
        cin >> s[i];
        if (i == 1 && s[i][1] == 'W') {
            s[i].insert(s[i].begin(), 'B');
            s[i] = s[i].substr(0, 8);
        }
        if (s[i] == s[i - 1]) {
            return cout << "NO", 0;
        }
    }
    return cout << "YES", 0;
}
