#include<bits/stdc++.h>
using namespace std;
#define ll long long int
int main()
{
	ll a,b,k;cin>>a>>b>>k;
    ll m=sqrt(b);
    vector<ll>ar(m+1);
    for(ll i=2;i<=m;++i)
    {
    	if(!ar[i])
    	{
    		for(ll j=i*i;j<=m;++j+=i)
    		{
    			ar[j]=1;
			}
		}
	}
	vector<ll>isPrime;
	for(ll i=2;i<=b;++i)
	{
		if(!ar[i])
			isPrime.push_back(i);
	}
	ll l=b-a+1;
	for(ll i=a;i<=b;++i)
	{
		ll cnt=0,j=i;
		while(j<=b)
		{
			if(binary_search(isPrime.begin(),isPrime.end(),j))
			{
				cnt++;
			}
			if(cnt>=k){l=min(l,j-i+1);break;}
			j++;
		}
	}
	if(l==b-a+1)
		l=-1;
	cout<<l<<endl;
    return 0;
}
