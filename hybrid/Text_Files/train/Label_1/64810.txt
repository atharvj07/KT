from collections import defaultdict
n=int(input())
a=list(map(int,input().split()))
ans=[0]*n
for i in range(n):
 d=defaultdict(int)
 for j in range(i,n):
  if a[j]>a[i]:
   if (j-i)%a[i]==0 and d[(j-i)//a[i]]==0:
    ans[j]=1-ans[i]
    d[(j-i)//a[i]]=1
print(''.join(['A' if x else 'B' for x in ans]))
