import math

for _ in range(int(input())):
    n, w = map(int, input().split())
    widths = list(map(int, input().split()))
    widths.sort(reverse=True)
    mxp = 0
    total = 0
    h = 1
    for wi in widths:
        if total + wi <= w:
            total += wi
        else:
            total = wi
            h += 1
        mxp = max(mxp, int(math.log2(wi)))
    print(max(h, mxp))
