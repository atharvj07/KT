import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    static final int INF = (int) 1e9;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), m = scanner.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            String s = scanner.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j) - '0';
            }
        }
        int[] dp = new int[m];
        Arrays.fill(dp, INF);
        for (int i = 0; i < m; i++) {
            int[] cnt = new int[2];
            for (int j = 0; j < n; j++) {
                cnt[a[j][i]]++;
            }
            if (cnt[0] == n) {
                continue;
            }
            dp[i] = Math.min(cnt[0], cnt[1]);
        }
        for (int i = 1; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (a[j][i] == 1 && a[j][i - 1] == 0) {
                    dp[i] = Math.min(dp[i], dp[i - 1] + 1);
                }
            }
        }
        int ans = INF;
        for (int value : dp) {
            ans = Math.min(ans, value);
        }
        if (ans == INF) {
            ans = -1;
        }
        System.out.println(ans);
    }
}
