import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        long[] x = new long[n + 1];
        long[] y = new long[m + 1];

        for (int i = 1; i <= n; i++) {
            x[i] = scanner.nextLong();
        }
        for (int i = 1; i <= m; i++) {
            y[i] = scanner.nextLong();
        }

        long answer = 0;
        for (int i = 1; i < n; i++) {
            for (int j = i + 1; j <= n; j++) {
                for (int k = 1; k < m; k++) {
                    for (int l = k + 1; l <= m; l++) {
                        answer += (x[j] - x[i]) * (y[l] - y[k]);
                    }
                }
            }
        }

        System.out.println(answer % (long) (1e9 + 7));
    }
}
