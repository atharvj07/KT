#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    for (auto &el : a) cin >> el;
    sort(a.begin(), a.end());
    int ans = 0;
    for (ans = 1; ans <= 100; ans++) {
      bool ok = true;
      for (int i = 1; i <= ans; i++) {
        ok &= (lower_bound(a.begin(), a.end(), i) != a.end()) ||
              (a.back() + x >= i);
      }
      if (!ok) break;
    }
    cout << ans - 1 << endl;
  }
}
