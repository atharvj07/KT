#include <iostream>
#include <cmath>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  vector<int> is_removed(n, 0);
  if (n <= 2) cout << n << endl;
  else {
    int res = 2;
    int l = 0, r = 1, mx = 1;
    for (int r = 1; r < n; r++) {
      if (a[r] <= a[r - 1]) {
        if (r - l + 1 > mx) {
          mx = r - l + 1;
          is_removed[l] = 1;
        }
        l = r;
      }
    }
    if (r - l + 1 > mx) {
      mx = r - l + 1;
      is_removed[l] = 1;
    }
    l = 0;
    int ll = 0;
    for (int r = 1; r < n; ++r) {
      if (is_removed[r - 1]) ll = r;
      if (a[r] <= a[r - 1]) {
        if (r - ll + 1 > res) { res = r - ll + 1; }
        ll = r;
      }
    }
    if (r - ll + 1 > res) { res = r - ll + 1; }
    cout << res << endl;
  }
  return 0;
}
