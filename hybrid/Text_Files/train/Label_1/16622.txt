#include <bits/stdc++.h>
#define ll long long
#define f first
#define s second
#define fi(a, b) for (int i = a; i < b; i++)
#define fj(a, b) for (int j = a; j < b; j++)
#define fk(a, b) for (int k = a; k < b; k++)
using namespace std;
const int N = 1e5 + 1e2;
ll n, a[N], b[N], L, ans, ch[N];
bool walkable[N], swimable[N];
int main() {
    cin >> n;
    fi(0, n) cin >> a[i];
    string x;
    cin >> x;
    fi(0, n) b[i] = (x[i] == 'W'), walkable[i] = (x[i] == 'G'), swimable[i] = (x[i] == 'W');
    for (L = 0; L <= n; L++) ch[L] = LLONG_MAX;
    ans = LLONG_MAX;
    priority_queue<pair<ll, ll>, vector<pair<ll, ll>>, greater<pair<ll, ll>>> q;
    q.push({0, 0});
    while (!q.empty()) {
        ll d = q.top().f;
        ll pos = q.top().s;
        q.pop();
        if (pos == n) {
            ans = d;
            break;
        };
        if (pos + 1 <= n) {
            if (!swimable[pos])
                if (d + 5 < ch[pos + 1]) {
                    q.push({d + 5, pos + 1});
                    ch[pos + 1] = d + 5;
                }
            if (!walkable[pos])
                if (d + 3 < ch[pos + 1]) {
                    q.push({d + 3, pos + 1});
                    ch[pos + 1] = d + 3;
                }
            if (pos != 0)
                if (d + 1 < ch[pos + 1]) {
                    q.push({d + 1, pos + 1});
                    ch[pos + 1] = d + 1;
                }
        }
        if (pos - 1 >= 0) {
            if (!swimable[pos - 1])
                if (d + 5 < ch[pos - 1]) {
                    q.push({d + 5, pos - 1});
                    ch[pos - 1] = d + 5;
                }
            if (!walkable[pos - 1])
                if (d + 3 < ch[pos - 1]) {
                    q.push({d + 3, pos - 1});
                    ch[pos - 1] = d + 3;
                }
            if (pos != 0)
                if (d + 1 < ch[pos - 1]) {
                    q.push({d + 1, pos - 1});
                    ch[pos - 1] = d + 1;
                }
        }
    }
    cout << ans << '\n';
    return 0;
}
