import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        int ans = 0;
        for (int i = 0; i < 1 << n; i++) {
            int prod = 1;
            boolean ok = true;
            for (int j = 0; j < n; j++) {
                if ((i >> j & 1) != 0) {
                    prod *= a[j] + 1;
                    if (a[j] + 1 > 100) ok = false;
                } else {
                    prod *= a[j] - 1;
                    if (a[j] - 1 < 1) ok = false;
                }
            }
            if (ok && prod % 2 == 0) ans++;
        }
        System.out.println(ans);
    }
}
