#include <bits/stdc++.h>
using namespace std;

int main() {
    string s;
    cin >> s;
    long long n = stoll(s.substr(0, s.size() - 1));
    char seat = s[s.size() - 1];
    long long ans = 0;
    ans += (n - 1) * 4 + 3;
    if (seat == 'a' || seat == 'f')
        ans--;
    else if (seat == 'd' || seat == 'e')
        ans += 2;
    cout << ans << endl;
    return 0;
}
