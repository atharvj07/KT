import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int xv = scanner.nextInt();
        int yv = scanner.nextInt();
        int xp = scanner.nextInt();
        int yp = scanner.nextInt();
        int xw1 = scanner.nextInt();
        int yw1 = scanner.nextInt();
        int xw2 = scanner.nextInt();
        int yw2 = scanner.nextInt();
        int xm1 = scanner.nextInt();
        int ym1 = scanner.nextInt();
        int xm2 = scanner.nextInt();
        int ym2 = scanner.nextInt();
        boolean canSee = false;
        if (xv == xp && yv == yp) {
            canSee = true;
        } else if (xw1 <= xv && xv <= xw2 && yw1 <= yv && yv <= yw2) {
            canSee = false;
        } else if (xm1 <= xv && xv <= xm2 && ym1 <= yv && yv <= ym2) {
            int dx = xv - xp;
            int dy = yv - yp;
            int nx = xm1 - xm2;
            int ny = ym1 - ym2;
            int dotProduct = dx * nx + dy * ny;
            if (dotProduct == 0) {
                canSee = true;
            } else if (dotProduct > 0) {
                if ((xv - xm1) * (ym2 - ym1) - (yv - ym1) * (xm2 - xm1) >= 0) {
                    canSee = true;
                }
            } else {
                if ((xv - xm2) * (ym1 - ym2) - (yv - ym2) * (xm1 - xm2) <= 0) {
                    canSee = true;
                }
            }
        } else {
            canSee = true;
        }
        System.out.println(canSee ? "YES" : "NO");
    }
}
