#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const ll INF = 1e18;

int main() {
    ll n, m;
    cin >> n >> m;
    vector<pair<ll, ll>> a(n);
    for (ll i = 0; i < n; i++) {
        ll x, y;
        cin >> x >> y;
        a[i] = {x, y};
    }
    vector<pair<ll, ll>> b(m);
    for (ll i = 0; i < m; i++) {
        ll x, y;
        cin >> x >> y;
        x--, y--;
        b[i] = {x, y};
    }
    vector<vector<ll>> g(n);
    for (auto [x, y] : b) {
        g[x].push_back(y);
        g[y].push_back(x);
    }
    vector<ll> dist(n, INF);
    for (ll i = 0; i < n; i++) {
        if (dist[i] == INF) {
            queue<ll> q;
            dist[i] = 0;
            q.push(i);
            while (!q.empty()) {
                ll v = q.front();
                q.pop();
                for (auto u : g[v]) {
                    if (dist[u] == INF) {
                        dist[u] = dist[v] + 1;
                        q.push(u);
                    }
                }
            }
        }
    }
    vector<ll> ans(n);
    for (ll i = 0; i < n; i++) {
        for (ll j = i + 1; j < n; j++) {
            if (dist[i] % 2 != dist[j] % 2) {
                if (a[i].first < a[j].first) {
                    ans[i] += a[i].first;
                    ans[j] += a[i].second;
                } else {
                    ans[i] += a[i].second;
                    ans[j] += a[i].first;
                }
            } else {
                if (a[i].second < a[j].second) {
                    ans[i] += a[i].first;
                    ans[j] += a[i].second;
                } else {
                    ans[i] += a[i].second;
                    ans[j] += a[i].first;
                }
            }
        }
    }
    for (auto &x : ans) {
        cout << x << ' ';
    }
    cout << '\n';
    return 0;
}
