#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    ll n, m;
    cin >> n >> m;
    if (m <= n - 1)
      cout << 0 << "\n";
    else if (n == 2)
      cout << m - 1 << "\n";
    else
      cout << m - n + 1 << "\n";
  }
  return 0;
}
