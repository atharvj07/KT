#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxN = 2e5 + 10;
int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int curx = 0, cury = 0;
        int mx = 0, my = 0, mn = 0;
        for (int i = 0; i < s.size(); ++i) {
            if (s[i] == 'D') ++curx;
            else if (s[i] == 'W') --curx;
            else if (s[i] == 'A') ++cury;
            else if (s[i] == 'S') --cury;
            mx = max(mx, curx);
            mn = min(mn, curx);
            my = max(my, cury);
        }
        int ans = (max(0, mx - mn) + 1) * (max(0, my - 0) + 1);
        cout << ans << '\n';
    }
    return 0;
}
