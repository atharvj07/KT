#include <bits/stdc++.h>
using namespace std;

const int maxn = 2e5 + 10;
int a[maxn], L[maxn], R[maxn];

int main() {
    int n, k; cin >> n >> k;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    a[0] = a[n + 1] = 1e9;
    int ans = 1e9;
    for (int i = 1; i <= n; ++i) {
        if (i % 2) L[i] = a[i];
        else L[i] = min(L[i - 1], a[i]);
    }
    for (int i = n; i >= 1; --i) {
        if (i % 2 == 0) R[i] = a[i];
        else R[i] = min(R[i + 1], a[i]);
    }
    for (int i = 1; i + k - 1 <= n; ++i) {
        if (k % 2) ans = min(ans, min(L[i + k - 1], R[i]));
        else ans = min(ans, min(L[i + k - 1], R[i + 1]));
    }
    cout << ans << endl;
}
