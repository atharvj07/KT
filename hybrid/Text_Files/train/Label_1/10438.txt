#include<bits/stdc++.h>
using namespace std;
int n,a[505],b[505];
int main(){
	cin>>n;
	for(int i=1;i<=n;i++)
		cin>>a[i],b[a[i]]=1;
	for(int i=1;i<=n;i++)
		if(!b[i]){
			cout<<-1;
			return 0;
		}
	for(int i=1;i<=n;i++){
		for(int j=1;j<=n;j++)
			if(i==j)
				cout<<i<<" ";
			else
				cout<<"0 ";
		cout<<endl;
	}
	return 0;
}
