import java.util.Scanner;

public class IvanAndUniversity {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] timetable = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                timetable[i][j] = line.charAt(j) - '0';
            }
        }

        int[][] dp = new int[n + 1][k + 1];
        for (int i = 0; i <= k; i++) {
            dp[n][i] = 0;
        }

        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = Integer.MAX_VALUE;
                for (int skip = 0; skip <= j && skip <= m; skip++) {
                    int hoursSpent = 0;
                    for (int hour = 0; hour < m; hour++) {
                        if (timetable[i][hour] == 1 && hour >= skip) {
                            hoursSpent++;
                        }
                    }
                    dp[i][j] = Math.min(dp[i][j], hoursSpent + dp[i + 1][j - skip]);
                }
            }
        }

        System.out.println(dp[0][k]);
    }
}
