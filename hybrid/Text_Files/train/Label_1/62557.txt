import heapq

n, m = map(int, input().split())
graph = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b, w = map(int, input().split())
    graph[a].append((b, w))
    graph[b].append((a, w))
a = list(map(int, input().split()))
dist = [float('inf')] * (n + 1)

def dijkstra(start):
    q = [(0, start)]
    dist[start] = 0
    while q:
        d, node = heapq.heappop(q)
        if d > dist[node]:
            continue
        for u, w in graph[node]:
            if d + w < dist[u]:
                dist[u] = d + w
                heapq.heappush(q, (d + w, u))

for i in range(1, n + 1):
    dijkstra(i)
    print(min(2 * dist[j] + a[j - 1] for j in range(1, n + 1)))
