from collections import Counter
s = input()
n = len(s)
if len(s) % 2 == 0: x = n // 2
else: x = n // 2 + 1
ans = []
cnt = Counter()
for e in s: cnt[e]+=1
for i in sorted(cnt.items()):
	for j in range(i[1]//2):
		if len(ans) == x: break
		ans.append(i[0]*2)
if len(ans) < x:
	for i in sorted(cnt.items()):
		if len(ans) == x: break
		ans.append(i[0])
print(len(ans))
print(*ans)
