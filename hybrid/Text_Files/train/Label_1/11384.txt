#include<bits/stdc++.h>
using namespace std;
int main(){
    int q;
    cin >> q;
    while(q--){
        int n;
        cin >> n;
        if(n == 1){
            cout << 0 << endl;
        }
        else{
            bitset<3> bs(n);
            if(bs == bitset<3>(0)){
                cout << -1 << endl;
            }
            else{
                int cnt = 0;
                while(n > 1){
                    cnt++;
                    if(n % 5 == 0){
                        n /= 5;
                    }
                    else if(n % 3 == 0){
                        n /= 3;
                    }
                    else if(n % 2 == 0){
                        n /= 2;
                    }
                    else{
                        cnt = -1;
                        break;
                    }
                }
                cout << cnt << endl;
            }
        }
    }
    return 0;
}
