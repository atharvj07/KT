#include<bits/stdc++.h>
using namespace std;
const int maxn = 1005;
int n, m, a[maxn], s[maxn];
int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> a[i];
    int sum = accumulate(a, a + n, 0);
    sort(a, a + n);
    cout << min(a[n - 1] + m, sum + m) << " " << a[n - 1] + m << endl;
    return 0;
}
