#include <bits/stdc++.h>
using namespace std;
#define int long long
int n, a[200005], freq[200005];

int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i], freq[a[i]]++; }
    int ans = -1;
    for (int i = 1; i <= n; i++) {
      if (freq[i] == 1) {
        if (ans == -1)
          ans = i;
        else
          ans = min(ans, i);
      }
    }
    cout << ans << endl;
    for (int i = 1; i <= n; i++) { freq[a[i]] = 0; }
  }
}
