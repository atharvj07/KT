import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine(), t = in.nextLine();
        int n = s.length(), m = t.length();
        if (n != m || !isPermutation(s, t)) {
            System.out.println("need tree");
            return;
        }
        boolean flag = false;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) != t.charAt(i)) {
                if (flag) {
                    System.out.println("both");
                    return;
                }
                flag = true;
            }
        }
        System.out.println(flag ? "automaton" : "array");
    }

    private static boolean isPermutation(String s, String t) {
        int[] cnt = new int[26];
        for (int i = 0; i < s.length(); i++) {
            cnt[s.charAt(i) - 'a']++;
            cnt[t.charAt(i) - 'a']--;
        }
        for (int i = 0; i < 26; i++) {
            if (cnt[i] != 0) {
                return false;
            }
        }
        return true;
    }
}
