import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] birthday1 = readBirthday(scanner);
        int[] birthday2 = readBirthday(scanner);
        scanner.close();

        int maxAgeDiffernce = 0;
        for (int year = Math.min(birthday1[0], birthday2[0]); year <= 3000; year++) {
            int ageDifference = getAgeDifference(birthday1, birthday2, year);
            maxAgeDiffernce = Math.max(maxAgeDiffernce, ageDifference);
        }

        System.out.println(maxAgeDiffernce);
    }

    private static int[] readBirthday(Scanner scanner) {
        int[] birthday = new int[3];
        for (int i = 0; i < 3; i++) {
            birthday[i] = scanner.nextInt();
        }
        return birthday;
    }

    private static int getAgeDifference(int[] birthday1, int[] birthday2, int year) {
        int age1 = year - birthday1[0];
        int age2 = year - birthday2[0];

        if (birthday1[1] > birthday2[1]) {
            age1--;
        } else if (birthday1[1] == birthday2[1]) {
            if (birthday1[2] > birthday2[2]) {
                age1--;
            }
        }

        return Math.abs(age1 - age2);
    }
}
