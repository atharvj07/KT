
import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            long p = in.nextLong();
            long q = in.nextLong();
            while (p % q == 0) p /= q;
            System.out.println(p);
        }
    }
}
