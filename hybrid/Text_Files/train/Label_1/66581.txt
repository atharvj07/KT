n, m, k = map(int, input().split())
x = [[[0] * k for j in range(m)] for i in range(n)]
for i in range(n):
    input()
    for j in range(m):
        x[i][j] = [int(l) for l in input()]
cnt = 0
for i in range(n):
    for j in range(m):
        for p in range(k):
            if x[i][j][p] == 1:
                l = 0
                r = 0
                for q in range(i + 1, n):
                    if x[q][j][p] == 1:
                        l += 1
                    else:
                        break
                for q in range(i - 1, -1, -1):
                    if x[q][j][p] == 1:
                        r += 1
                    else:
                        break
                b = f = 0
                for q in range(j + 1, m):
                    if x[i][q][p] == 1:
                        b += 1
                    else:
                        break
                for q in range(j - 1, -1, -1):
                    if x[i][q][p] == 1:
                        f += 1
                    else:
                        break
                c = d = 0
                for q in range(p + 1, k):
                    if x[i][j][q] == 1:
                        c += 1
                    else:
                        break
                for q in range(p - 1, -1, -1):
                    if x[i][j][q] == 1:
                        d += 1
                    else:
                        break
                if l * r * b * f * d * c == 0:
                    cnt += 1
print(cnt)
