import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            long x = 0;
            for (int j = 0; j < s.length(); j++) {
                if (s.charAt(j) != '?') {
                    x = x * 10 + s.charAt(j) - '0';
                }
            }
            a[i] = x;
        }
        boolean[] used = new boolean[10];
        for (int i = 0; i < n; i++) {
            if (a[i] == 0) {
                System.out.println("NO");
                return;
            }
            for (int j = 0; j < 10; j++) {
                if (used[j]) {
                    continue;
                }
                long x = a[i];
                long y = 0;
                boolean ok = false;
                int k = 0;
                while (x > 0 || y > 0) {
                    int d = (int) (x % 10);
                    int e = (int) (y % 10);
                    x /= 10;
                    y /= 10;
                    if (d == 0 && e != 0) {
                        ok = false;
                        break;
                    }
                    if (d == 0) {
                        d = j;
                    }
                    if (d > e) {
                        ok = true;
                        break;
                    }
                    if (d == e) {
                        k++;
                    }
                    y = y * 10 + d;
                }
                if (ok && k == i) {
                    used[j] = true;
                    a[i] = y;
                    break;
                }
            }
        }
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            if (a[i] == 0 || (i > 0 && a[i] <= a[i - 1])) {
                ok = false;
                break;
            }
        }
        if (!ok) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            System.out.println(a[i]);
        }
    }
}
