import java.util.Scanner;

public class Main {
    static int n, m;
    static int[][] table;
    static int max;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        table = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                table[i][j] = in.nextInt();
            }
        }
        dp(0, 0, 0, 0);
        System.out.println(max);
    }

    static void dp(int row, int col, int comb, int sum) {
        if (row == n) {
            if (isComb(comb)) {
                max = Math.max(max, sum);
            }
            return;
        }
        if (col == m) {
            dp(row + 1, 0, 0, 0);
            return;
        }
        dp(row, col + 1, comb, sum);
        dp(row, col + 1, comb + 1, sum + table[row][col]);
    }

    static boolean isComb(int comb) {
        int last = 1;
        for (int i = 2; i <= comb; i++) {
            if (last % 2 == i % 2) {
                return false;
            }
            last = i;
        }
        return true;
    }
}
