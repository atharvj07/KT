import java.util.*;

public class Main {
    static class Domino implements Comparable<Domino> {
        int x, y;

        public Domino(int x, int y) {
            this.x = x;
            this.y = y;
        }

        @Override
        public int compareTo(Domino o) {
            return this.x - o.x;
        }
    }

    static class Point implements Comparable<Point> {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }

        @Override
        public int compareTo(Point o) {
            if (this.x != o.x) {
                return this.x - o.x;
            }
            return this.y - o.y;
        }
    }

    static int[] getMinCost(Domino[] dominoes, Point[] points) {
        TreeSet<Point> set = new TreeSet<>();
        int[] ans = new int[points.length];
        for (Point p : points) {
            Point r = new Point(p.x, p.y + 1);
            NavigableSet<Point> subSet = set.subSet(p, true, r, false);
            if (subSet.isEmpty()) {
                set.add(p);
                continue;
            }
            int minCost = Integer.MAX_VALUE;
            for (Point pt : subSet) {
                Domino d = dominoes[pt.x - 1];
                int y = Math.max(pt.y, d.y);
                minCost = Math.min(minCost, y - p.y);
            }
            ans[p.x - 1] = minCost;
            set.add(p);
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Domino[] dominoes = new Domino[n];
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            dominoes[i] = new Domino(x, y);
        }
        int q = sc.nextInt();
        Point[] points = new Point[q];
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            points[i] = new Point(x, y);
        }
        int[] ans = getMinCost(dominoes, points);
        for (int i = 0; i < q; i++) {
            System.out.println(ans[i]);
        }
    }
}
