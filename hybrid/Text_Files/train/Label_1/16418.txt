#include <cstdio>
#include <set>

const int MAXN = 10;

int main() {
    int n;
    scanf("%d", &n);

    std::set<int> x, y;
    for (int i = 0; i < 4 * n + 1; ++i) {
        int xi, yi;
        scanf("%d %d", &xi, &yi);
        if (!x.count(xi))
            x.insert(xi);
        else
            x.erase(xi);
        if (!y.count(yi))
            y.insert(yi);
        else
            y.erase(yi);
    }

    for (int xi : x) {
        for (int yi : y) {
            printf("%d %d\n", xi, yi);
            return 0;
        }
    }

    return 0;
}
