#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const ll mod = 998244353;
const ll inv2 =  499122177; // 1 / 2 (mod mod)

ll qpow(ll a, ll p) {
    ll ans = 1;
    while (p) {
        if (p & 1) {
            ans = 1LL * ans * a % mod;
        }
        a = 1LL * a * a % mod;
        p >>= 1;
    }
    return ans;
}

int main() {
    ll n;
    cin >> n;
    cout << qpow(inv2, n) << endl;
    return 0;
}
