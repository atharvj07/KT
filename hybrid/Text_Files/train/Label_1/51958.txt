import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] strangeness = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            int start = i, end = i, maxDist = 0;
            while (start >= 0 && end < n && a[start] == a[end]) {
                maxDist = Math.max(maxDist, (end - start + 1) / 2);
                start--;
                end++;
            }
            start = i;
            end = i + 1;
            while (start >= 0 && end < n) {
                maxDist = Math.max(maxDist, (end - start) / 2);
                start--;
                end++;
            }
            strangeness[i] = maxDist;
        }
        for (int i = 0; i < n; i++) {
            System.out.print(strangeness[i] + " ");
        }
    }
}
