#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
int main() {
    int n, m, a, b, g0, x, y, z;
    cin >> n >> m >> a >> b >> g0 >> x >> y >> z;
    vector<ll> pref(n * m);
    vector<ll> row(n * m);
    int cur = g0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            int idx = i * m + j;
            pref[idx] = cur;
            row[idx] = cur;
            cur = (1LL * cur * x + y) % z;
        }
    }
    for (int i = 1; i < n; ++i) {
        for (int j = 0; j < m; ++j) pref[i * m + j] += pref[(i - 1) * m + j];
    }
    for (int j = 1; j < m; ++j) {
        for (int i = 0; i < n; ++i) row[i * m + j] += row[i * m + j - 1];
    }
    ll ans = 0;
    for (int i = 0; i < n - a + 1; ++i) {
        for (int j = 0; j < m - b + 1; ++j) {
            ll lb = (i > 0) ? pref[(i - 1) * m + j] : 0;
            ll rb = (j > 0) ? row[i * m + (j - 1)] : 0;
            ll p = pref[i * m + j];
            ans += p - lb - rb;
        }
    }
    cout << ans << '\n';
    return 0;
}
