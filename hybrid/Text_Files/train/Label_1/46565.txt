import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long[] a = new long[4];
        for (int i = 0; i < 4; i++) {
            a[i] = in.nextLong();
        }
        if (a[1] > a[0] + a[2] || a[2] > a[0] + a[3] || a[0] * a[3] != a[1] * a[2]) {
            System.out.println("Impossible");
        } else {
            StringBuilder s = new StringBuilder();
            for (int i = 0; i < a[0]; i++) {
                s.append("0");
            }
            for (int i = 0; i < a[1]; i++) {
                s.append("10");
            }
            for (int i = 0; i < a[2]; i++) {
                s.append("01");
            }
            for (int i = 0; i < a[3]; i++) {
                s.append("1");
            }
            System.out.println(s);
        }
    }
}
