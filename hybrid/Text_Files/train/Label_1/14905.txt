#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mp make_pair
#define pb push_back
#define pll pair<ll,ll>
#define pii pair<int,int>
#define vll vector<ll>
#define vii vector<int>
#define vpi vector<pii>
#define vpl vector<pll>
#define fi first
#define se second
#define all(v) v.begin(),v.end()
#define sz(v) (int)(v.size())
#define inf (ll)1e18

ll n,k;
vll t,b;

ll f(ll l,ll r){
	vector<pll> arr;
	for(int i=l;i<=r;i++){
		arr.pb({t[i],b[i]});
	}
	
	sort(all(arr));
	ll ans=arr[0].fi*arr[0].se;
	return ans;
}
int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(0),cout.tie(0);
	cin>>n>>k;
	t.resize(n+1);
	b.resize(n+1);
	
	for(int i=1;i<=n;i++){
		cin>>t[i]>>b[i];
	}
	
	if(k==n){
		ll ans=0;
		for(int i=1;i<=n;i++){
			ans+=t[i];
		}
		ll bb=*min_element(all(b));
		ans*=bb;
		cout<<ans<<endl;
		return 0;
	}
	
	ll ans=0;
	for(int i=1;i<=n-k+1;i++){
		ans=max(ans,f(i,i+k-1));
	}
	cout<<ans<<endl;
	return 0;
}
