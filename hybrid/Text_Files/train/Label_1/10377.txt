#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

const int N = 81;
int dp[N][N], a[N], k;

int main() {
    scanf("%d%d", &k, &k);
    for (int i = 0; i < k; ++i) scanf("%d", a + i);
    dp[0][0] = 0;
    for (int i = 0; i < k; ++i) {
        for (int j = 0; j <= i; ++j) dp[i+1][j] = dp[i][j];
        for (int j = 0; j < i; ++j)
            dp[i+1][j+1] = max(dp[i+1][j+1], dp[i][j] + (a[i] != a[j]));
    }
    int mx = 0;
    for (int i = 0; i <= k; ++i) mx = max(dp[k][i], mx);
    printf("%d", k - mx);
    return 0;
}
