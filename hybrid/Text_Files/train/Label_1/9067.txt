#include <bits/stdc++.h>
using namespace std;
int main(){
    int tc;
    cin>>tc;
    while(tc--){
        int n,k;
        cin>>n>>k;
        string s;
        cin>>s;
        int cnt[26]{};
        for(int i=0;i<n;++i) ++cnt[s[i]-'a'];
        int ans=0;
        for(int i=0;i<26;++i){
            ans+=min(cnt[i],cnt[i]*(k+1)/k);
        }
        cout<<ans<<'\n';
    }
}
