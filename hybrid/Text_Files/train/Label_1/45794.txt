import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long n = input.nextInt();
        int m = input.nextInt();
        int[] p = new int[m];
        for (int i = 0; i < m; i++) {
            p[i] = input.nextInt();
        }
        Arrays.sort(p);
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < m && p[j] <= i; j++) {
                ans++;
                j++;
            }
        }
        System.out.println(ans);
    }
}
