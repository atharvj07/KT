import java.util.HashMap;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        HashMap<String, String> hm = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String w = sc.next();
            String wp = sc.next();
            hm.put(w, wp);
        }
        String s = null;
        String t = null;
        boolean ok = true;
        for (String w : hm.keySet()) {
            String wp = hm.get(w);
            if (s == null) {
                s = w;
                t = wp;
                continue;
            }
            if (!s.equals(w) || !t.equals(wp)) {
                int idx = s.indexOf(t);
                if (idx == -1 || !w.substring(idx, idx + t.length()).equals(t)) {
                    ok = false;
                }
            }
        }
        if (ok) {
            System.out.println("YES");
            System.out.println(s);
            System.out.println(t);
        } else {
            System.out.println("NO");
        }
    }
}
