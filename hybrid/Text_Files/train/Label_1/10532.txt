#include <cstdio>
#include <cstring>
using namespace std;

int n,p,cnt;
char a[1005][5];

bool check(){
	for (int i=1;i<=p;i++)
		if (a[i][0]=='O' && a[i][1]=='O') return true;
	return false;
}

void solve(){
	bool flag=false;
	for (int i=1;i<=n;i++){
		if (!flag && a[i][0]=='O' && a[i][1]=='O'){
			a[i][0]=a[i][1]='+';
			flag=true;
		}
		printf("%s\n",a[i]);
	}
}

int main(){
	scanf("%d",&n);
	p=0;
	for (int i=1;i<=n;i++){
		scanf("%s",&a[i]);
		if (check()) p=i;
	}
	if (p){
		printf("YES\n");
		solve();
	}
	else printf("NO\n");
	return 0;
}
