from collections import Counter

for _ in range(int(input())):
    a = Counter(map(int, input().split()))
    for i in a:
        if a[i] % 2 == 1:
            for j in a:
                if i * j in a:
                    a[i * j] -= 1
            break
    ans = 1
    for i in a:
        ans *= i**a[i]
    print([ans, -1][ans in a])
