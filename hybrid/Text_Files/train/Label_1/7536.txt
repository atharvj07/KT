#include <iostream>
using namespace std;
int main() {
  long long int n, k, a, b;
  cin >> n >> k >> a >> b;
  long long int ans = 1e18;
  for (long long int div = 1; div <= n; div *= k) {
    long long int cnt1 = (n - div + k - 1) / k;
    long long int cnt2 = (n - 1) - cnt1;
    ans = min(ans, cnt1 * a + cnt2 * b);
  }
  cout << ans << endl;
  return 0;
}
