#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[n],b[n];
    for(int i=0;i<n;i++) cin>>a[i];
    sort(a,a+n);
    for(int i=0;i<n;i++) b[i]=a[n-1-i];

    int ans=0;
    for(int i=0;i<n;i++)
    {
        if(i+1<n&&a[i]==a[i+1]) continue;
        int j=i+1;
        while(j<n&&b[i]==b[j]) j++;
        if(j==n) continue;
        ans++;
        i=j-1;
    }
    cout<<ans<<endl;
    return 0;
}
