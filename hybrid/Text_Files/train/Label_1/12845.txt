#include<bits/stdc++.h>
#define gc getchar_unlocked
using namespace std;
#define vi vector<int>
#define vvi vector<vector<int>>
#define pi pair<int,int>
#define vpi vector<pair<int,int>>
#define pb push_back
#define mp make_pair
#define fi first
#define se second
#define fin(a) scanf("%d",&a)
#define fout(a) printf("%d\n",a)
#define ll long long
#define sz(x) ((int)(x).size())
#define all(x) (x).begin(),(x).end()
#define rep(i,a,n) for(ll i=a;i<n;++i)
#define re(i,a,n) for(ll i=a;i<=n;++i)
const ll mod=1e9+7;
const ll inf=1e18;
void get(int &x)
{    register int c = gc();
    x = 0;
    int neg = 0;
    for(;((c<48 || c>57) && c != '-');c = gc());
    if(c=='-') {neg=1;c=gc();}
    for(;c>47 && c<58;c = gc()) {x = (x<<1) + (x<<3) + c - 48;}
    if(neg) x=-x;
}
void solve(){
    int n, m, no;
    get(n);
    get(m);
    vi c(n), d(n, 0);
    rep(i,0,n)get(c[i]);
    vvi adj(n);
    rep(i,0,m){
        get(no);
        get(no);
        no--;
        no--;
        adj[no].pb(no+1);
        adj[no+1].pb(no);
    }
    vi level(n, n), ans;
    queue<pi> q;
    q.push(mp(0,0));
    level[0] = 0;
    while(!q.empty()){
        pi p = q.front();
        q.pop();
        ans.pb(p.fi);
        for(const auto &nbr: adj[p.fi]){
            if(level[nbr] > p.se+1){
                level[nbr] = p.se+1;
                q.push(mp(nbr, p.se+1));
            }
        }
    }
    sort(all(ans));
    ll sum=0;
    rep(i,0,n)sum+=c[ans[i]];
    cout<<sum<<endl;
}
int main(int argc, char** argv) {
	solve();
	return 0;
}
