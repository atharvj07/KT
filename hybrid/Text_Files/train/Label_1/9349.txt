#include <bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, k;
    cin >> n >> k;
    vector<int> t(n);
    for (auto &i : t) cin >> i;
    int cnt = 0;
    bool sw = false;
    for (auto &i : t) {
        if ((i < 0) && !sw) cnt++, sw = true;
        if ((i >= 0) && sw) cnt++;
    }
    if (cnt > k) cout << -1;
    else
        cout << cnt / 2;
    return 0;
}
