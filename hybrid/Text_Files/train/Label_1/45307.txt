import java.util.*;

class Main {

    static int gcd(int a, int b) {
        if (a % b == 0) {
            return b;
        } else {
            return gcd(b, a % b);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int A[] = new int[N];
        HashSet<Integer> primes = new HashSet<>();
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextInt();
            for (int prime : primes) {
                if (gcd(A[i], prime) > 1) {
                    System.out.println("setwise coprime");
                    return;
                }
            }
            HashSet<Integer> currentPrimes = new HashSet<>();
            for (int j = 2; j * j <= A[i]; j++) {
                if (A[i] % j == 0) {
                    currentPrimes.add(j);
                    while (A[i] % j == 0) {
                        A[i] /= j;
                    }
                }
            }
            if (A[i] > 1) {
                currentPrimes.add(A[i]);
            }
            for (int prime : currentPrimes) {
                primes.add(prime);
            }
        }
        int GCD = A[0];
        for (int i = 1; i < N; i++) {
            GCD = gcd(GCD, A[i]);
        }
        if (GCD == 1) {
            System.out.println("setwise coprime");
        } else {
            System.out.println("not coprime");
        }
    }
}
