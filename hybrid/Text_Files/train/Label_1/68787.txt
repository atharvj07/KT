from collections import Counter
n, m = map(int, input().split())
arr = list(map(int, input().split()))
ans = []
for i in range(m):
    l, r = map(int, input().split())
    l, r = l - 1, r - 1
    sum = arr[l:r + 1]
    if Counter(sum)[0] % 2 == 0 and Counter(sum)[1] % 2 == 0:
        ans.append(1)
    else:
        ans.append(0)
for x in ans:
    print(x)
