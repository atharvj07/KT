n, k = map(int, input().split())
if k == 0:
    print(n)
    print(*range(1, n + 1))
else:
    actions = (n + 2*k) // (2*k + 1)
    if n % (2*k + 1) == 0:
        start = k + 1
    else:
        start = n % (2*k + 1) // 2 + 1
    print(actions)
    for i in range(actions):
        print(start + i*(2*k + 1), end=' ')
