import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String homeTeam = sc.nextLine();
        String awayTeam = sc.nextLine();
        int n = sc.nextInt();
        Map<String, Map<Integer, Integer>> players = new HashMap<>();
        for (int i = 0; i < n; i++) {
            int minute = sc.nextInt();
            String team = sc.next();
            int playerNumber = sc.nextInt();
            String card = sc.next();
            String player = team.equals("h") ? homeTeam : awayTeam;
            Map<Integer, Integer> cards = players.computeIfAbsent(player, k -> new HashMap<>());
            cards.put(playerNumber, cards.getOrDefault(playerNumber, 0) + (card.equals("y") ? 1 : 2));
            if (cards.get(playerNumber) >= 2) {
                System.out.println(player + " " + playerNumber + " " + minute);
            }
        }
    }
}
