g = []
for i in range(3):
  g.append(list(map(int,input().split())))
ans = [['0']*3 for i in range(3)]
for i in range(3):
  for j in range(3):
    if g[i][j]%2==1:
      ans[i][j] = '1'
      if i: ans[i-1][j] = str(1-int(ans[i-1][j]))
      if j: ans[i][j-1] = str(1-int(ans[i][j-1]))
      if i<2: ans[i+1][j] = str(1-int(ans[i+1][j]))
      if j<2: ans[i][j+1] = str(1-int(ans[i][j+1]))
for i in ans:
  print(''.join(i))
