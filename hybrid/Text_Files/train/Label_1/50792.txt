import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), x = sc.nextInt();
        if (x % 2 == 1) {
            System.out.println("No");
            return;
        } else {
            System.out.println("Yes");
        }
        int[][] p = new int[N + 1][(1 << N)];
        for (int i = 1; i <= N; i++) {
            for (int j = 0; j < (1 << (N - i)); j++) {
                p[i][j] = (p[i - 1][j] << 1);
                p[i][(1 << (N - i)) + j] = (p[i - 1][j] << 1) | 1;
            }
        }
        int[] ans = new int[(1 << N) - 1];
        ans[(1 << N) - 2] = x / 2;
        for (int i = N - 1; i >= 1; i--) {
            for (int j = 0; j < (1 << N); j++) {
                int l = (j + (1 << (N - i))) % (1 << N);
                int r = (j + (1 << (N - i - 1))) % (1 << N);
                int m = (j + (1 << (N - i)) / 2) % (1 << N);
                if (m != 0 && l != ans[m - 1] && r != ans[m - 1])
                    ans[j] = ans[m - 1];
                else if (l != 0 && r != ans[l - 1])
                    ans[j] = ans[l - 1];
                else if (r != 0 && l != ans[r - 1])
                    ans[j] = ans[r - 1];
            }
        }
        for (int i = 0; i < (1 << N) - 1; i++)
            System.out.println(p[N][i] + ans[i] + 1);
    }
}
