#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    while(n--){
        string s;
        cin>>s;
        string xxx="0000000000000000000000000000000000000000000000";
        bool ok=0;
        int pos=-1;
        for(int i=0;i<s.size();i++){
            xxx[i]=s[i];
            if(s[i]=='0'&&!ok&&i!=s.size()-1){
                pos=i;
            }else{
                ok=1;
            }
        }
        if(s==xxx){
            cout<<"red\n";
            continue;
        }
        for(int i=pos+1;i<s.size();i++){
            if(s[i]>xxx[i]){
                swap(s[i],xxx[i]);
            }
        }
        if(xxx==s){
            cout<<"red\n";
            continue;
        }
        for(int i=pos+1;i<s.size();i++){
            if(s[i]<xxx[i]){
                swap(s[i],xxx[i]);
                break;
            }
        }
        if(xxx==s){
            cout<<"red\n";
        }else{
            cout<<"cyan\n";
        }
    }
    return 0;
}
