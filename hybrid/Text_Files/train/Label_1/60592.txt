n = int(input())
a = list(map(int, input().split()))

a.sort()
dp = [float('inf')] * (n + 1)
prev = [-1] * (n + 1)
dp[0] = 0

for i in range(3, n + 1):
    for j in range(i):
        cost = a[i - 1] - a[j]
        if dp[j] + cost < dp[i]:
            dp[i] = dp[j] + cost
            prev[i] = j

res = dp[n]
k = 0
teams = [0] * n
i = n
while i > 0:
    k += 1
    for j in range(prev[i], i):
        teams[j] = k
    i = prev[i]

print(res, k)
print(' '.join(map(str, teams)))
