#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod=1e9+7;
ll n,m;
ll dp[1005][1005][2];
ll cal[1005][1005];
ll f(ll x,ll y,ll z){
    if(x==n&&y==m&&z){
        return cal[x][y];
    }
    if(x==n&&y==m&&!z){
        return 0;
    }
    if(dp[x][y][z]!=-1){
        return dp[x][y][z]%mod;
    }
    ll ans=cal[x][y]%mod;
    if(z){
        ans+=f(x+1,y,z)%mod;
        ans+=f(x,y+1,z)%mod;
    }
    else{
        ans+=f(x,y+1,z)%mod;
        ans+=f(x-1,y,z)%mod;
    }
    dp[x][y][z]=ans%mod;
    return dp[x][y][z];
}
int main() {
    cin>>n>>m;
    for(ll i=1;i<=n;++i){
        for(ll j=1;j<=m;++j){
            cin>>cal[i][j];
        }
    }
    memset(dp,-1,sizeof(dp));
    ll x=f(1,1,1);
    cout<<(x%mod+mod)%mod<<endl;
    return 0;
}
