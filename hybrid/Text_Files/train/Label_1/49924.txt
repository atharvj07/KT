import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int[] colors = new int[n];
        for (int i = 0; i < n; i++) {
            colors[i] = input.nextInt();
        }
        ArrayList<ArrayList<Integer>> adjList = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            adjList.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int u = input.nextInt() - 1;
            int v = input.nextInt() - 1;
            adjList.get(u).add(v);
            adjList.get(v).add(u);
        }
        int maxDiversity = 0;
        int bestColor = -1;
        HashMap<Integer, HashSet<Integer>> neighborColors = new HashMap<>();
        for (int i = 0; i < n; i++) {
            neighborColors.put(colors[i], new HashSet<>());
        }
        for (int i = 0; i < n; i++) {
            for (int neighbor : adjList.get(i)) {
                neighborColors.get(colors[i]).add(colors[neighbor]);
            }
        }
        for (int color : neighborColors.keySet()) {
            int diversity = neighborColors.get(color).size();
            if (diversity > maxDiversity) {
                maxDiversity = diversity;
                bestColor = color;
            }
        }
        System.out.println(bestColor);
    }
}
