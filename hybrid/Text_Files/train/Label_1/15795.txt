#include <iostream>

long long fastpow(long long x, long long p) {
    long long res = 1;
    for (; p; p >>= 1, (x *= x) %= 10) {
        if (p & 1) {
            (res *= x) %= 10;
        }
    }
    return res;
}

long long last(long long x) {
    if (x == 0) {
        return 1;
    }
    if (x % 4 == 1) {
        return 6;
    }
    if (x % 4 == 2) {
        return 2;
    }
    if (x % 4 == 3) {
        return 8;
    }
    return 4;
}

int main() {
    long long a, b;
    std::cin >> a >> b;
    std::cout << last(b) * fastpow(last(a - 1), b - a + 1) % 10 << std::endl;
    return 0;
}
