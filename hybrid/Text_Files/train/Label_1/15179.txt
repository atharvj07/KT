#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int64_t> a(n);
    vector<int64_t> c(n);
    int ans = 0;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        c[i] = a[i];
    }
    sort(c.begin(), c.end());
    for (int i = 0; i < n; i++) {
        if (a[i] != c[i])
            ans++;
    }
    cout << ans << '\n';
    return 0;
}
