#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, v;
  cin >> n >> v;
  vector<vector<int>> items(n);
  for (int i = 0; i < n; i++) {
    int k;
    cin >> k;
    items[i].resize(k);
    for (int j = 0; j < k; j++) {
      cin >> items[i][j];
    }
    sort(items[i].begin(), items[i].end());
  }
  vector<int> ans;
  for (int i = 0; i < n; i++) {
    if (items[i][0] <= v) {
      ans.push_back(i + 1);
    }
  }
  cout << ans.size() << '\n';
  for (auto x : ans) {
    cout << x << ' ';
  }
  cout << '\n';
  return 0;
}
