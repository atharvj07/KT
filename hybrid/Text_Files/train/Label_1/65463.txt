import sys
sys.setrecursionlimit(10**6)

n, b = map(int, input().split(' '))

queries = []
for i in range(n):
    queries.append(tuple(map(int, input().split(' '))))

queries.sort()

ans = []

t = 0
q = []

for query in queries:
    arrival, duration = query
    while t < arrival and q:
        t += q[0][1]
        q.pop(0)
    if len(q) < b:
        q.append((arrival, duration))
    else:
        ans.append(-1)
        continue

    if t < arrival:
        t = arrival

    t += duration
    ans.append(t)

for a in ans:
    print(a)
