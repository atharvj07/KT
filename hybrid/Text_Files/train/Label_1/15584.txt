#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll a, b, x, y, z, ans = 0;
int main() {
  cin >> a >> b >> x >> y >> z;
  ans += max(0ll, 2 * x - a);
  if (y >= b) {
    ans += a;
  } else {
    ans += max(0ll, b - y);
  }
  ans += max(0ll, 3 * z - b);
  cout << ans << endl;
  return 0;
}
