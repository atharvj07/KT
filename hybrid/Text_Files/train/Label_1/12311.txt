#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int ans = 2;
    for (int i = 1; i < n; i++) {
        if (a[i] - a[i - 1] > 1) {
            ans = max(ans, i + 1);
        }
    }
    cout << ans << endl;
    return 0;
}
