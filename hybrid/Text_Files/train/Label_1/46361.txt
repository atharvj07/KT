import java.util.Arrays;
import java.util.Scanner;

public class DolphinChemical {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int M_a = scanner.nextInt();
        int M_b = scanner.nextInt();

        Package[] packages = new Package[N];
        for (int i = 0; i < N; i++) {
            int a_i = scanner.nextInt();
            int b_i = scanner.nextInt();
            int c_i = scanner.nextInt();
            packages[i] = new Package(a_i, b_i, c_i);
        }

        // Sort packages by the ratio of a_i / b_i.
        Arrays.sort(packages, (p1, p2) -> Double.compare(p1.a_i * 1.0 / p1.b_i, p2.a_i * 1.0 / p2.b_i));

        int minCost = -1;
        int targetA = M_a, targetB = M_b;

        for (Package package_i : packages) {
            int a_i = package_i.a_i;
            int b_i = package_i.b_i;
            int c_i = package_i.c_i;

            // Check if the current package can contribute to the target ratio.
            if (a_i * targetB == b_i * targetA) {
                targetA -= a_i;
                targetB -= b_i;
                minCost += c_i;
            } else {
                break;
            }
        }

        System.out.println(targetA == 0 && targetB == 0 ? minCost : -1);
    }

    static class Package {
        int a_i;
        int b_i;
        int c_i;

        public Package(int a_i, int b_i, int c_i) {
            this.a_i = a_i;
            this.b_i = b_i;
            this.c_i = c_i;
        }
    }
}
