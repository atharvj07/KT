#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

typedef long long ll;
using namespace std;

int main() {
  ll n;
  cin >> n;
  vector<ll> a(n);
  ll sum = 0;
  for (ll i = 0; i < n; i++) {
    cin >> a[i];
    sum += a[i];
  }
  sort(a.begin(), a.end());

  ll mn = sum;

  for (ll i = 0; i < n - 1; i++) {
    ll g = __gcd(a[i], a[i + 1]);
    mn = min(mn, sum - 2 * g + a[i] / g * a[i + 1]);
  }

  cout << mn << endl;
  return 0;
}
