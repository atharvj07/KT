#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k, i, cnt = 0;
    cin >> n >> k;
    vector<int> ans;
    for (i = n; i >= 1; i--) {
        if (cnt == k) break;
        if (i != n - k) ans.push_back(i);
        else cnt++;
    }
    ans.push_back(n - k);
    for (int x : ans) cout << x << " ";
    cout << "\n";
}
