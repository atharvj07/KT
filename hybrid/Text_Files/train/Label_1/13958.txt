#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define FOR(i, a, b) for(int i = a; i <= b; i++)

const int maxn = 1e3 + 5;
bool flipped[maxn];

int main() {
    int n, k, ans = 0; cin >> n >> k;
    vector<int> v;
    FOR(i, 1, n) {
        if(flipped[i]) continue;
        v.pb(i); ans++;
        FOR(j, i + 1, min(n, i + k)) flipped[j] = true;
        FOR(j, max(1, i - k), i - 1) flipped[j] = true;
    }
    cout << ans << endl;
    for(int x : v) cout << x << " ";
}
