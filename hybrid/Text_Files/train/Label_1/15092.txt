#include <bits/stdc++.h>
using namespace std;
const int MAXN = 5000;
int a[MAXN], n, m, ans;
int main() {
    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        int l, r;
        cin >> l >> r;
        for (int j = l; j <= r; j++) a[j]++;
    }
    for (int i = 1; i <= n; i++) ans += (bool)a[i];
    cout << ans << endl;
    return 0;
}
