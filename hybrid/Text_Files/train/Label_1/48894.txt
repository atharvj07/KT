import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    long[] A = new long[N];
    long sum = 0;
    long mod = 1000000007;
    for (int i = 0; i < N; i++) {
      A[i] = sc.nextLong();
      sum = (sum + A[i]) % mod;
    }
    long ans = 0;
    for (int i = 0; i < N - 1; i++) {
      ans = (ans + A[i] * (sum - A[i])) % mod;
    }
    System.out.println(ans);
  }
}
