import sys
from collections import defaultdict, deque

def bfs(graph, start):
    visited = [False] * (len(graph) + 1)
    q = deque()
    q.append(start)
    visited[start] = True
    while q:
        cur = q.popleft()
        yield cur
        for u in graph[cur]:
            if not visited[u]:
                q.append(u)
                visited[u] = True

def solve():
    n, m = map(int, input().split())
    graph = defaultdict(list)
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        graph[b].append(a)

    for node in graph:
        if len(list(bfs(graph, node))) >= 3:
            return "0"

    return "-1"

print(solve())
