import java.util.*;
import java.io.*;

class Main {
    private static void countChars(Map<Character, Integer> map, String title) {
        for (int i = 0; i < title.length(); i++) {
            char ch = title.charAt(i);
            if (map.containsKey(ch)) {
                map.put(ch, map.get(ch) + 1);
            } else {
                map.put(ch, 1);
            }
        }
    }

    public static void main(String[] args) throws IOException {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            int n = Integer.parseInt(br.readLine());
            String[] titles = new String[n];
            for (int i = 0; i < n; i++) {
                titles[i] = br.readLine();
            }
            Map<Character, Integer> allChars = new HashMap<>();
            boolean[] used = new boolean[26];
            for (String title : titles) {
                countChars(allChars, title);
                for (char ch : title.toCharArray()) {
                    used[ch - 'a'] = true;
                }
            }
            int minLen = Integer.MAX_VALUE;
            for (int i = 0; i < 26; i++) {
                if (!used[i]) {
                    minLen = 1;
                    System.out.println((char) (i + 'a'));
                    return;
                }
            }
            for (Map.Entry<Character, Integer> entry : allChars.entrySet()) {
                if (entry.getValue() < n) {
                    minLen = Math.min(minLen, entry.getKey() - 'a' + 1);
                }
            }
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < 26; i++) {
                if (minLen == i + 1) {
                    sb.append((char) (i + 'a'));
                }
            }
            System.out.println(sb.toString());
        }
    }
}
