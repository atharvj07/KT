import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int I = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        Arrays.sort(a);
        long sum = 0;
        for (int i = 0; i < n; i++) sum += a[i];
        long low = 0, high = sum, mid = 0;
        while (low <= high) {
            mid = (low + high) / 2;
            long left = 0, right = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] < mid) left += mid - a[i];
                else if (a[i] > mid) right += a[i] - mid;
            }
            if (left <= I && right <= I) high = mid - 1;
            else low = mid + 1;
        }
        System.out.println(low);
    }
}
