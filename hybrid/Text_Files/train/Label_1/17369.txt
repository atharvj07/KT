#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define all(x) x.begin(), x.end()

int main() {
    int n, m;
    cin >> n >> m;
    vi y1(n), y2(m);
    for (auto &v : y1) cin >> v;
    for (auto &v : y2) cin >> v;
    sort(all(y1)), sort(all(y2));
    int ans = n + m;
    for (auto v : y1) {
        if (binary_search(all(y2), -v)) ans -= 2;
    }
    cout << ans << endl;
}
