#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, s;
    cin >> n >> s;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    if (a[s - 1] && (b[s] || s == 1)) { cout << "YES" << endl; }
    else { cout << "NO" << endl; }
    return 0;
}
