#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, l[301], c[301], d[3000001], ll, cc;
  cin >> n;
  bool b = false;
  for (int i = 0; i < n; i++) { cin >> l[i]; }
  for (int i = 0; i < n; i++) { cin >> c[i]; }
  for (int i = 0; i < 3000001; i++) { d[i] = 1e9; }
  d[0] = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 3000001; j++) {
      if (d[j] != 1e9) {
        if (j - l[i] >= 0) d[j - l[i]] = min(d[j - l[i]], d[j] + c[i]);
        if (j + l[i] < 3000001) d[j + l[i]] = min(d[j + l[i]], d[j] + c[i]);
      }
    }
  }
  for (int i = 1; i < 3000001; i++) {
    if (d[i] == 1e9) b = true;
  }
  if (b)
    cout << -1 << endl;
  else {
    cc = 1e9;
    for (int i = 0; i < 3000001; i++) { cc = min(cc, d[i]); }
    cout << cc << endl;
  }
  return 0;
}
