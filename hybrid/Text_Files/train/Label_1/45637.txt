import java.util.Scanner;

public class SquareFormation {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int x1 = scanner.nextInt();
            int y1 = scanner.nextInt();
            int x2 = scanner.nextInt();
            int y2 = scanner.nextInt();
            int x3 = scanner.nextInt();
            int y3 = scanner.nextInt();
            int x4 = scanner.nextInt();
            int y4 = scanner.nextInt();

            int minX = Math.min(Math.min(x1, x2), Math.min(x3, x4));
            int maxX = Math.max(Math.max(x1, x2), Math.max(x3, x4));
            int minY = Math.min(Math.min(y1, y2), Math.min(y3, y4));
            int maxY = Math.max(Math.max(y1, y2), Math.max(y3, y4));

            int sideLength = maxX - minX;
            int steps = 0;

            if (y1 != minY) {
                steps += Math.abs(y1 - minY);
            }
            if (y2 != maxY) {
                steps += Math.abs(y2 - maxY);
            }
            if (y3 != minY) {
                steps += Math.abs(y3 - minY);
            }
            if (y4 != maxY) {
                steps += Math.abs(y4 - maxY);
            }

            if (x1 != minX) {
                steps += Math.abs(x1 - minX);
            }
            if (x2 != maxX) {
                steps += Math.abs(x2 - maxX);
            }
            if (x3 != minX) {
                steps += Math.abs(x3 - minX);
            }
            if (x4 != maxX) {
                steps += Math.abs(x4 - maxX);
            }

            System.out.println(steps);
        }
    }
}
