from sys import stdin
readline = stdin.readline

def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)

for _ in range(int(readline())):
    s, t = readline().strip(), readline().strip()
    if len(s) < len(t):
        s, t = t, s
    if s == t:
        print(s)
    elif len(s) % len(t) == 0 and s[:len(t)] == t:
        print(s)
    else:
        l = len(s) // gcd(len(s), len(t))
        if s[:l] == t[:l]:
            print(s[:l] * (len(s) // l))
        else:
            print(-1)
