def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

t = int(input().strip())
for _ in range(t):
    n, e = map(int, input().strip().split())
    a = list(map(int, input().strip().split()))
    count = 0
    for i in range(n):
        for k in range((n-i)//e):
            product = 1
            for j in range(k+1):
                product *= a[i + j*e]
            if is_prime(product):
                count += 1
    print(count)
