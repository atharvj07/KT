import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] a = new int[12];
        for (int i = 0; i < 12; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[6];
        for (int i = 0; i < 12; i++) {
            for (int j = i + 1; j < 12; j++) {
                for (int k = j + 1; k < 12; k++) {
                    if (a[i] + a[j] + a[k] <= a[11] - a[i] - a[j] - a[k]) {
                        b[0] = a[i];
                        b[1] = a[j];
                        b[2] = a[k];
                        b[3] = a[11] - a[i] - a[j] - a[k];
                        for (int l = 0; l < 12; l++) {
                            if (l != i && l != j && l != k) {
                                b[4] = a[l];
                                for (int m = l + 1; m < 12; m++) {
                                    if (m != i && m != j && m != k) {
                                        b[5] = a[m];
                                        if (b[0] + b[1] + b[2] == b[3] + b[4] + b[5]) {
                                            System.out.println("yes");
                                            return;
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
        System.out.println("no");
    }
}
