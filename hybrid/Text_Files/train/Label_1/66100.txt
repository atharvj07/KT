import sys

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
vis = [False] * n
ans = 0

for i in range(n):
    if vis[i]:
        continue
    j = i
    summ = 0
    while not vis[j]:
        summ += a[j]
        vis[j] = True
        if b[j] != -1:
            j = b[j] - 1
    ans += summ

print(ans)

for i in range(n):
    if vis[i]:
        print(i + 1, end=" ")
