#include <algorithm>
#include <vector>
#include <iostream>
using namespace std;
int main() {
  int t; cin >> t;
  while (t--) {
    vector<int> b(7);
    for (int i = 0; i < 7; i++) { cin >> b[i]; }
    vector<int> a(3);
    a[0] = b[3] - b[0];
    a[1] = b[4] - b[0];
    a[2] = b[6] - b[4];
    for (int i = 0; i < 3; i++) { cout << a[i] << ' '; }
    cout << '\n';
  }
  return 0;
}
