#include <bits/stdc++.h>
using namespace std;
int n,k,a[505],vis[505]={0},cnt=0;
int main() {
    cin>>n>>k;
    for (int i=1;i<=n;++i) cin>>a[i];
    while (cnt<k) {
        int mx=0,t=0;
        for (int i=1;i<=n;++i) {
            if (vis[i]) continue;
            if (a[i]>mx) {
                mx=a[i];
                t=i;
            }
        }
        vis[t]=1;
        ++cnt;
        if (cnt==k) {
            cout<<a[t]<<endl;
            return 0;
        }
        mx=0,t=0;
        for (int i=1;i<=n;++i) {
            if (vis[i]) continue;
            if (a[i]>mx) {
                mx=a[i];
                t=i;
            }
        }
        vis[t]=0;
    }
}
