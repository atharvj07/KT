from collections import Counter, defaultdict

n = int(input())
a = list(map(int, input().split()))
c = Counter(a)
ap = defaultdict(list)
for i, x in enumerate(a):
    ap[x].append(i)
t = 0
for x, pi in ap.items():
    if len(pi) > 1:
        d = pi[1] - pi[0]
        for i in range(1, len(pi)):
            if pi[i] - pi[i - 1] != d:
                break
        else:
            t += 1
            print(x, d)
    else:
        t += 1
        print(x, 0)
print(t)
