import java.util.*;
import java.io.*;

public class Main {
    static long[] b;
    static long[] ans;
    static boolean[] used;
    static int n;
    static long xor;
    static long prexor;

    static boolean dfs(int x) {
        if (x > n) {
            return xor < prexor;
        }
        for (int i = 1; i <= n; ++i) {
            if (used[i]) {
                continue;
            }
            xor ^= b[i];
            ans[x] = b[i];
            used[i] = true;
            if (dfs(x + 1)) {
                return true;
            }
            xor ^= b[i];
            used[i] = false;
        }
        return false;
    }

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        b = new long[n + 1];
        ans = new long[n + 1];
        used = new boolean[n + 1];
        for (int i = 1; i <= n; ++i) {
            b[i] = sc.nextInt();
        }
        xor = 0;
        prexor = 0;
        for (int i = 2; i <= n; ++i) {
            prexor ^= b[i];
        }
        if (dfs(1)) {
            System.out.println("Yes");
            for (int i = 1; i <= n; ++i) {
                System.out.print(ans[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("No");
        }
    }
}
