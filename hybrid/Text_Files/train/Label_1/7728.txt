#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int MAXN = 2e5 + 5;

int n, m, k[MAXN], d[MAXN], t[MAXN], c[MAXN];

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; ++i) cin >> k[i];
    for (int i = 0; i < m; ++i) {
        cin >> d[i] >> t[i];
        c[t[i]] = max(c[t[i]], d[i]);
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) ans += (k[i] + 1) / 2;
    for (int i = 0; i < n; ++i) {
        if (k[i] % 2 == 0) continue;
        if (c[i + 1] > 0 && c[i + 1] <= ans) --ans;
    }
    cout << ans << endl;
    return 0;
}
