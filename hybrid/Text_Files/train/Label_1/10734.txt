#include <bits/stdc++.h>
using namespace std;
struct ARR {
    int x, c;
    bool operator<(const ARR& rhs) const { return c < rhs.c; }
};
int main() {
    int n, k;
    cin >> n >> k;
    unordered_map<int, int> mp;
    for (int i = 0; i < n; ++i) {
        int x;
        cin >> x;
        mp[x]++;
    }
    vector<ARR> arr;
    for (auto& x : mp) arr.emplace_back(ARR({x.first, x.second}));
    sort(arr.begin(), arr.end());
    vector<int> ans;
    int idx = 0;
    while (ans.size() < k) {
        if (idx >= arr.size()) break;
        if (arr[idx].c <= 0) {
            ++idx;
            continue;
        }
        ans.emplace_back(arr[idx].x);
        arr[idx].c--;
    }
    for (auto& x : ans) cout << x << ' ';
    return 0;
}
