import java.util.*;
import java.io.*;

class Main {
    static int n, m, k;
    static int[] dist;
    static ArrayList<Integer>[] towns;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        towns = new ArrayList[n + 5];
        dist = new int[n + 1];
        for (int i = 0; i < n + 5; i++) {
            towns[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int w = sc.nextInt();
            towns[u].add(v);
            towns[v].add(u);
        }
        int[] malls = new int[k + 1];
        for (int i = 0; i < k; i++) {
            malls[i] = sc.nextInt();
        }
        boolean stopped = false;
        bfs(1, malls, stopped);
        int mx = 0;
        for (int i = 1; i <= n; i++) {
            mx = Math.max(dist[i], mx);
        }
        System.out.println(mx);
    }

    public static void bfs(int startingPoint, int[] malls, boolean stopped) {
        boolean[] visited = new boolean[n + 1];
        visited[startingPoint] = true;
        Queue<Integer> q = new LinkedList<>();
        q.add(startingPoint);
        while (!q.isEmpty()) {
            int current = q.poll();
            for (int v : towns[current]) {
                if (!visited[v]) {
                    visited[v] = true;
                    q.add(v);
                }
                if (dist[v] < dist[current] + 1) {
                    dist[v] = dist[current] + 1;
                }
                for (int j = 0; j < k; j++) {
                    if (v == malls[j]) {
                        stopped = true;
                        break;
                    }
                }
            }
        }
    }
}
