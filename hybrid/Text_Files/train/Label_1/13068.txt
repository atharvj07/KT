#include <iostream>
#include <map>
using std::cin;
using std::cout;
using std::map;

int main() {
  int n;
  cin >> n;
  map<int, int> m;
  int max_=0;
  for (int i = 0; i < n; i++) {
    int ai;
    cin >> ai;
    if (!m[ai]) { m[ai]=1; }
    else { m[ai]++; }
  }
  for (auto &i : m) { max_+=i.second; }
  cout << max_;
}
