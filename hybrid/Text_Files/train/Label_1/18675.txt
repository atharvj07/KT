#include <bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int &ai : a) cin >> ai;
        int ans = *max_element(a.begin(), a.end());
        for (int i = 1; i <= ans; i++) {
            if (ans % i == 0) {
                bool ok = true;
                for (int ai : a) {
                    if (ai % i != 0) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    ans = i;
                    break;
                }
            }
        }
        cout << ans << "\n";
    }
    return 0;
}
