#include <iostream>
#include <map>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  map<string, int> m;
  for (int i = 0; i < n - 1; i++) {
    string t = s.substr(i, 2);
    m[t]++;
  }
  int mx = 0;
  string ans = "";
  for (auto it : m) {
    if (it.second > mx) {
      mx = it.second;
      ans = it.first;
    }
  }
  cout << ans << "\n";
}
