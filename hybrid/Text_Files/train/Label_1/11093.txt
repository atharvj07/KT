#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;

const ll MOD = 998244353;

int main() {
  ll n, k, a, m, ans = -1;
  cin >> n >> k >> a >> m;
  if (k * a != n) {
    cin >> n >> k >> a; // Ignore invalid input
    return cout << "-1\n", 0;
  }
  vector<bool> shot(n + 1, false);
  for (ll i = 0; i < m;) {
    ll x;
    cin >> x;
    if (!shot[x]) {
      if (k == 1) {
        ans = i + 1;
        break;
      }
      k--;
      shot[x] = shot[x + a] = true;
      i += 2;
    } else {
      i++;
    }
  }
  cout << ans << '\n';
  return 0;
}
