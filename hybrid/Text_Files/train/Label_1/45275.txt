import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        boolean[][] obstacle = new boolean[n][m];
        for (int i = 0; i < k; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            obstacle[x][y] = true;
        }
        int[][] dp = new int[n][m];
        for (int i = 1; i < n; i++) {
            for (int j = 1; j < m; j++) {
                if (obstacle[i][j]) {
                    dp[i][j] = -1;
                    continue;
                }
                if (dp[i - 1][j] == -1 || dp[i][j - 1] == -1) {
                    dp[i][j] = -1;
                    continue;
                }
                dp[i][j] = Math.min(dp[i - 1][j], dp[i][j - 1]) + 1;
            }
        }
        System.out.println(dp[n - 1][m - 1]);
    }
}
