lst = []
d = {}
for _ in range(int(input())):
    s, t, x = map(int, input().split())
    if s == 1:
        lst.append((t, x, 1))
        if not (x in d): d[x] = 0
        d[x] += 1
    elif s == 2:
        lst.append((t, x, -1))
        d[x] -= 1
    else:
        ans = 0
        for j, k, c in lst:
            if j <= t:
                ans += c * k
        print(d.get(x, 0) if x in d else 0 + ans)
