#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  int n, k;
  cin>>n>>k;
  map<int, int>mp;
  for(int i=0; i<n; i++) {
    int x;
    cin>>x;
    mp[x]++;
  }
  int ans = 0, sum = 0;
  for(auto x: mp) {
    int need = (x.second+1)/2;
    if(need<=x.second) {
      ans += x.second - need;
      sum += need;
    }
  }
  cout << ans + min(sum, (n+1)/2) << endl;
  return 0;
}
