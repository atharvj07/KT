#include <bits/stdc++.h>

using namespace std;

int grid[5][5];

int main() {
  int k;
  cin >> k;
  for (int i = 1; i <= 4; i++) {
    string s;
    cin >> s;
    for (int j = 1; j <= 4; j++) {
      if (s[j - 1] >= '1' && s[j - 1] <= '9')
        grid[s[j - 1] - '0'][i] = 1;
    }
  }
  for (int i = 1; i <= 4; i++) {
    int cnt = 0;
    for (int j = 1; j <= 4; j++)
      cnt += grid[i][j];
    if (cnt > k) {
      cout << "NO" << endl;
      return 0;
    }
  }
  for (int i = 1; i <= 4; i++) {
    int cnt = 0;
    for (int j = 1; j <= 4; j++)
      cnt += grid[j][i];
    if (cnt > k) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
