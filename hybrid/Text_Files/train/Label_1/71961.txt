a, k = map(int, input().split())
s = str(a)
l = []
for i in range(1, len(s)):
    if s[i - 1] < s[i]:
        l.append(i - 1)
if k == 0 or len(l) == 0:
    print(s)
else:
    if l[0] == 0:
        l.pop(0)
    if k % 2 == 1:
        i = l[-1]
        s1 = list(s)
        s1[i], s1[i + 1] = s1[i + 1], s1[i]
        print(''.join(s1))
    else:
        i = l[0]
        s1 = list(s)
        s1[i], s1[i + 1] = s1[i + 1], s1[i]
        s2 = list(s)
        s2[len(s2) - 1], s2[len(s2) - 2] = s2[len(s2) - 2], s2[len(s2) - 1]
        if int(''.join(s1)) > int(''.join(s2)):
            print(''.join(s1))
        else:
            print(''.join(s2))
