n = int(input())
matrix = [list(map(int, input().split())) for _ in range(n)]

good_cars = []

for i in range(n):
    is_good = True
    for j in range(n):
        if matrix[i][j] == 1 or matrix[i][j] == 3:
            is_good = False
            break
    if is_good:
        good_cars.append(i+1)

print(len(good_cars))
print(' '.join(map(str, good_cars)))
