import sys
mod = 1000000007

s = list(sys.stdin.readline().strip())
t = list(sys.stdin.readline().strip())
n = len(s)
m = len(t)
pos = [-1] * (n + 1)
for i in range(n - 1, -1, -1):
    if s[i] == t[0]:
        pos[i] = pos[i + 1]
    pos[i] = i
lst = [0] * n
for i in range(n - m + 1):
    if pos[i] != -1:
        lst[i] = lst[pos[i] + 1] + 1
dp = [[0] * (n + 1) for _ in range(n + 1)]
cnt = 0
for j in range(n - 1, -1, -1):
    for i in range(j):
        dp[i][j] = (dp[i][j + 1] + dp[i + 1][j] - dp[i + 1][j + 1] + mod) % mod
        if pos[i] != -1 and pos[i] <= j:
            dp[i][j] = (dp[i][j] + lst[pos[i]] - lst[j + 1]) % mod
        if s[i] == t[0] and (i == 0 or s[i - 1] != t[0]):
            cnt += dp[i][j]
            cnt %= mod
print(cnt)
