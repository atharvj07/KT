for _ in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    ans = [0] * n
    for i in range(n):
        cycle = set()
        j = i
        while j not in cycle:
            cycle.add(j)
            j = p[j] - 1
        c = len(cycle)
        ans[i] = (ans[i] + c) % c + 1
    print(*ans)
