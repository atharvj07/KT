#include <cstdio>
int main() {
  int x, y, sum = 0;
  char op;
  scanf("%d %c %d", &x, &op, &y);
  if (op == '+') sum = x + y;
  else sum = x - y;
  while (scanf(" %c %d", &op, &y) == 2) {
    if (op == '+') sum += y;
    else sum -= y;
  }
  printf("%d\n", ((sum % 256) + 256) % 256);
  return 0;
}
