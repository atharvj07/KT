#include <iostream>
#include <vector>
int main() {
    int n;
    std::cin >> n;
    std::vector<int> l(n), r(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> l[i] >> r[i];
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        if (l[i] != l[0]) ++ans;
        if (r[i] != r[0]) ++ans;
    }
    std::cout << ans << std::endl;
    return 0;
}
