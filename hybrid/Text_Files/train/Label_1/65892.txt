def solve_traveler_problem(t, test_cases):
    results = []
    for i in range(t):
        n, s = test_cases[i]
        result = [1] * (n + 1)
        for direction in ['L', 'R']:
            cnt = 0
            for j in range(n + 1):
                if j < n and s[j] == direction:
                    cnt += 1
                else:
                    if direction == 'L':
                        result[j] += cnt
                    else:
                        result[j - cnt - 1] += cnt
                    cnt = 0
        results.append(result)
    return results

# Read input
t = int(input().strip())
test_cases = []
for _ in range(t):
    n = int(input().strip())
    s = input().strip()
    test_cases.append((n, s))

# Solve
results = solve_traveler_problem(t, test_cases)

# Print output
for result in results:
    print(*result)
