#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    if (n % 2 == 0) {
      n /= 2;
      cout << "YES 2 ";
    } else {
      cout << "NO\n";
      continue;
    }
    if (n % 3 == 0) {
      n /= 3;
      cout << "3 ";
    } else {
      cout << "NO\n";
      continue;
    }
    if (n > 1) {
      cout << n << endl;
    } else {
      cout << "NO\n";
    }
  }
  return 0;
}
