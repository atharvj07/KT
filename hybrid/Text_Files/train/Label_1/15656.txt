#include <bits/stdc++.h>
#define ll long long 
#define vi vector<ll>
#define pb push_back
using namespace std; 
const ll mod = 1e9+7; 
int main() {
    ll n , m , k; 
    cin >> n >> m >> k; 
    vi v(n); 
    for (ll i = 0 ; i < n ; i++) cin >> v[i];
    sort(v.rbegin(),v.rend());
    vi suf(n);
    for (ll i = 0 ; i < n ; i++) {
        if (i) suf[i] = suf[i-1];
        suf[i] += v[i];
    }
    vector<array<ll,3>> ans; 
    ll cur = 0 , sum = 0; 
    for (ll i = 0; i < k-1; i++) {
        ll rem = k -1 - i;
        ll l = 0 , r = n-1 , ind = -1; 
        while (l <= r) {
          ll mid = (l+r)/2;
          if (suf[mid] - (i ? suf[i-1] : 0) >=  (m*rem)) ind = mid , r = mid -1;
          else l = mid + 1;
        }
        assert(ind != -1);
        sum += suf[ind] - (i ? suf[i-1] : 0);
        ans.pb({i+1,ind+1,suf[ind] - (i ? suf[i-1] : 0)});
        cur = ind+1;
    }
    sum += suf[n-1] - (cur ? suf[cur-1] : 0);
    cout << sum << "\n";
    for (auto &it: ans) 
      cout << it[1] << " ";
    cout << "\n";
    return 0; 
}
