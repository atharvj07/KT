for _ in range(int(input())):
    n, m, a, b = map(int, input().split())
    if a * m != b * n:
        print('NO')
    else:
        print('YES')
        for i in range(n):
            s = ['0'] * m
            for j in range(a):
                s[j * m // a + i % m // (m // a)] = '1'
            print(''.join(s))
