#include <bits/stdc++.h>
using namespace std;

void dfs(int v, vector<vector<int>> &g, vector<int> &order, vector<int> &vis)
{
    vis[v] = 1;
    for (auto u : g[v])
        if (!vis[u])
            dfs(u, g, order, vis);
    order.push_back(v);
}

int main()
{
    int n, m;
    cin >> n >> m;
    vector<vector<int>> g(n + 1);
    for (int i = 0; i < m; i++)
    {
        int v, u;
        cin >> v >> u;
        g[v].push_back(u);
    }

    vector<int> order, vis(n + 1);
    for (int i = 1; i <= n; i++)
        if (!vis[i])
            dfs(i, g, order, vis);

    reverse(order.begin(), order.end());
    for (auto v : order)
        cout << v << " ";
    return 0;
}
