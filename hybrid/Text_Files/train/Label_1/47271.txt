import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        int count = 0;
        char[][] image = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = sc.next();
            for (int j = 0; j < m; j++) {
                image[i][j] = line.charAt(j);
                if (i >= 1 && j >= 1 && image[i][j] == 'e' && image[i - 1][j] == 'c' && image[i][j - 1] == 'f' && image[i - 1][j - 1] == 'a') {
                    count++;
                }
            }
        }
        System.out.println(count);
    }
}
