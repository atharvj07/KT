#include <bits/stdc++.h>

using namespace std;

struct coor {int y; int ind;};

bool comp(coor a, coor b) { return a.y < b.y; }

int main() {
  int n;
  cin >> n;

  vector<coor> arr(n);
  for (int i = 0; i < n; ++i) {
    cin >> arr[i].y;
    arr[i].ind = i + 1;
  }

  sort(arr.begin(), arr.end(), comp);

  bool can_connect = false;
  for (int i = 0; i + 1 < n; ++i) {
    if (arr[i].ind % 2 == 0 && arr[i + 1].ind % 2 == 1 &&
        arr[i].y == arr[i + 1].y) {
      can_connect = true;
      cout << "Yes";
      return 0;
    }
  }

  cout << "No";
  return 0;
}
