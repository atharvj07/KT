t = int(input())
for _ in range(t):
    n, c0, c1, h = map(int, input().split())
    s = input()
    count0 = s.count('0')
    count1 = n - count0
    cost_without_changes = count0*c0 + count1*c1
    cost_change_to_0 = count1*h + n*c0
    cost_change_to_1 = count0*h + n*c1
    print(min(cost_without_changes, cost_change_to_0, cost_change_to_1))
