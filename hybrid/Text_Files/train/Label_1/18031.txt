#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> s(m + 1);
    for (int i = 0; i < n; i++) {
        int l, r;
        cin >> l >> r;
        s[l].push_back(r);
    }
    vector<bool> covered(m + 1, false);
    for (int i = 1; i <= m; i++) {
        for (int j : s[i]) {
            covered[j] = true;
        }
    }
    int cnt = 0;
    for (int i = 1; i <= m; i++) {
        if (!covered[i]) cnt++;
    }
    cout << cnt << endl;
    for (int i = 1; i <= m; i++) {
        if (!covered[i]) cout << i << ' ';
    }
    cout << endl;
}
