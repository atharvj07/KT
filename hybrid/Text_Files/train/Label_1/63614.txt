import sys

def is_happy(a, k):
    n = len(a)
    b = []
    for i in range(n - k + 1):
        b.append(min(a[i:i + k]))
    return sorted(b) == list(range(1, n - k + 2))

t = int(sys.stdin.readline())
for _ in range(t):
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    print(''.join(['1' if is_happy(a, k) else '0' for k in range(1, n + 1)]))
