#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<string> s(n);
    for (auto &i : s) cin >> i;
    int ans = 0;
    map<string, int> mp;
    for (int i = 0; i < n; i++) {
      if (mp.count(s[i])) {
        int pos = -1;
        for (int j = 0; j < 4; j++) {
          if (s[i][j] == s[mp[s[i]]][j]) continue;
          pos = j;
          break;
        }
        s[mp[s[i]]][pos] = s[mp[s[i]]][pos] ^ (s[i][pos] ^ (s[mp[s[i]]][pos]));
        s[i][pos] = s[mp[s[i]]][pos];
        ans++;
        mp.clear();
      }
      mp[s[i]] = i;
    }
    cout << ans << endl;
    for (auto i : s) cout << i << endl;
  }
  return 0;
}
