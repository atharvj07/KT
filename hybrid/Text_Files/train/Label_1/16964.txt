#include <iostream>
#include <algorithm>

using namespace std;

struct subseq {
    int sum, pos, val[101];

    bool operator<(const subseq &other) const {
        if (sum != other.sum) return sum > other.sum;
        for (int i = 1; i <= pos; i++)
            if (val[i] != other.val[i]) return val[i] < other.val[i];
        return false;
    }
};

int main() {
    int n, a[101], m, k, pos;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    cin >> m;
    while (m--) {
        cin >> k >> pos;
        subseq dp[101][101];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                if (j == 0) dp[i][j].sum = 0, dp[i][j].pos = 0;
                else {
                    dp[i][j] = dp[i - 1][j];
                    if (dp[i][j].sum < dp[i - 1][j - 1].sum + a[i]) {
                        dp[i][j].sum = dp[i - 1][j - 1].sum + a[i];
                        dp[i][j].pos = j;
                        dp[i][j].val[j] = a[i];
                        for (int l = j - 1; l >= 1; l--)
                            dp[i][j].val[l] = dp[i - 1][j - 1].val[l];
                    }
                }
            }
        }
        cout << dp[n][k].val[pos] << endl;
    }
    return 0;
}
