import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.StringTokenizer;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String s = br.readLine();
        int[] a = new int[n - 1];
        int[] b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            a[i] = Integer.parseInt(st.nextToken());
            b[i] = Integer.parseInt(st.nextToken());
        }
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            graph.get(a[i] - 1).add(b[i] - 1);
            graph.get(b[i] - 1).add(a[i] - 1);
        }
        int[] dist = new int[n];
        Arrays.fill(dist, -1);
        int[] count = new int[n];
        dfs(0, -1, s, graph, dist, count);
        int maxCount = 0;
        int maxDist = 0;
        for (int i = 0; i < n; i++) {
            if (count[i] > maxCount) {
                maxCount = count[i];
                maxDist = dist[i];
            }
        }
        int[] distToMax = new int[n];
        Arrays.fill(distToMax, -1);
        dfs2(maxDist, -1, 0, graph, distToMax);
        int minDist = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '1') {
                minDist = Math.min(minDist, distToMax[i]);
            }
        }
        if (minDist == Integer.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(minDist - 1);
        }
    }

    private static void dfs(int u, int p, String s, List<List<Integer>> graph, int[] dist, int[] count) {
        dist[u] = 0;
        count[u] = s.charAt(u) - '0';
        for (int v : graph.get(u)) {
            if (v != p) {
                dfs(v, u, s, graph, dist, count);
                dist[u] = Math.max(dist[u], dist[v] + 1);
                count[u] += count[v];
            }
        }
    }

    private static void dfs2(int u, int p, int dist, List<List<Integer>> graph, int[] distToMax) {
        distToMax[u] = dist;
        for (int v : graph.get(u)) {
            if (v != p) {
                dfs2(v, u, dist + 1, graph, distToMax);
            }
        }
    }
}
