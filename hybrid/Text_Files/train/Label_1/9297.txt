#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int s;
    cin >> s;
    if (s == 1)
      cout << 1 << endl;
    else if (s == 2)
      cout << 2 << endl;
    else if (s == 3)
      cout << 3 << endl;
    else if (s % 3 == 0)
      cout << s / 3 << endl;
    else
      cout << s / 3 + 1 << endl;
  }
  return 0;
}
