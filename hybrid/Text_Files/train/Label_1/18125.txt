#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<vector<int>> monsters(n);
    int max_armor = 0;
    for (int i = 0; i < n; i++) {
      int k;
      cin >> k;
      monsters[i].resize(k);
      for (int j = 0; j < k; j++) {
        cin >> monsters[i][j];
        max_armor = max(max_armor, monsters[i][j]);
      }
    }
    sort(monsters.begin(), monsters.end());
    int power = max_armor + 1;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < monsters[i].size(); j++) {
        power = max(power, monsters[i][j] + j + 1);
      }
    }
    cout << power << endl;
  }
  return 0;
}
