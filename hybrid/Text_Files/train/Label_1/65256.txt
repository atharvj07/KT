def solve():
    x = int(input())
    t = 0
    li = [0]
    count = []
    i = 0
    while x != 2 ** (i + 1) - 1:
        add = (2 ** i) - 1
        if t + 2 > 39:
            break
        if ((x ^ add) != (x + 1)):
            x ^= add
            if i not in count:
                li.append(i)
                count.append(i)
            t += 1
        else:
            x += 1
            t += 1
        i += 1
    if t <= 40:
        print(t)
        for i in li:
            print(i, end=' ')
    else:
        print(-1)

solve()
