import java.util.Scanner;

public class FloodRisk {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int w = scanner.nextInt();
        int d = scanner.nextInt();
        int n = scanner.nextInt();

        int[][] grid = new int[w + 1][d + 1];
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            int z = scanner.nextInt();
            grid[x][y] = z;
        }

        if (!isValid(grid)) {
            System.out.println("No");
            return;
        }

        int totalAltitude = 0;
        for (int i = 1; i <= w; i++) {
            for (int j = 1; j <= d; j++) {
                if (grid[i][j] == 0) {
                    grid[i][j] = getAltitude(grid, i, j);
                }
                totalAltitude += grid[i][j];
            }
        }

        System.out.println(totalAltitude);
    }

    private static boolean isValid(int[][] grid) {
        for (int i = 1; i < grid.length; i++) {
            for (int j = 1; j < grid[0].length; j++) {
                if (grid[i][j] != 0) {
                    if (Math.abs(grid[i][j] - grid[i - 1][j]) > 1 ||
                            Math.abs(grid[i][j] - grid[i][j - 1]) > 1) {
                        return false;
                    }
                }
            }
        }
        return true;
    }

    private static int getAltitude(int[][] grid, int x, int y) {
        int minAltitude = Integer.MAX_VALUE;
        if (grid[x - 1][y] != 0) {
            minAltitude = Math.min(minAltitude, grid[x - 1][y] + 1);
        }
        if (grid[x][y - 1] != 0) {
            minAltitude = Math.min(minAltitude, grid[x][y - 1] + 1);
        }
        if (grid[x + 1][y] != 0) {
            minAltitude = Math.min(minAltitude, grid[x + 1][y] - 1);
        }
        if (grid[x][y + 1] != 0) {
            minAltitude = Math.min(minAltitude, grid[x][y + 1] - 1);
        }
        return minAltitude;
    }
}
