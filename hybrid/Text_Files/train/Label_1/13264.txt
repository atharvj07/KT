#include <iostream>
#include <cstdlib>

int main() {
  int n, k;
  std::cin >> n >> k;
  int a[n];
  for (int i = 0; i < n; i++) std::cin >> a[i];
  int ans = 0;
  for (int i = 0; i < n; i++)
    for (int j = i + 1; j < n; j++)
      if (__builtin_popcount(a[i] ^ a[j]) == k) ans++;
  std::cout << ans << '\n';
  return 0;
}
