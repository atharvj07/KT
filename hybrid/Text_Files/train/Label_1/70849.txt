import sys
input = sys.stdin.readline
m, k = map(int, input().split())
edges = {}
for _ in range(m):
    a, b = map(int, input().split())
    if a not in edges:
        edges[a] = [b];
    else:
        edges[a].append(b)
    if b not in edges:
        edges[b] = [a];
    else:
        edges[b].append(a)

for id in sorted(edges.keys()):
    candidates = []
    for adj in edges[id]:
        count = 0
        for friend in edges[adj]:
            if friend in edges[id]:
                count += 1
        if count / len(edges[adj]) * 100 >= k:
            candidates.append(adj)
    candidates.sort()
    print(f'{id}: {len(candidates)} {" ".join(map(str, candidates))}')
