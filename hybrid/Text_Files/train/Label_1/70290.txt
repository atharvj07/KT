import array
def get_dice_probabilities(max_probabilities, min_probabilities):
    n = len(max_probabilities)
    dice_probabilities = [[0] * n for _ in range(2)]
    for i1 in range(n):
        for i2 in range(n):
            for j1 in range(n):
                for j2 in range(n):
                    if i1 > j1 or (i1 == j1 and i2 >= j2):
                        dice_probabilities[0][i1] += min_probabilities[j1] * max_probabilities[i2] / n
                        dice_probabilities[1][i2] += min_probabilities[j1] * max_probabilities[i2] / n
    return dice_probabilities
n = int(input())
max_probabilities = list(map(float, input().split()))
min_probabilities = list(map(float, input().split()))
dice_probabilities = get_dice_probabilities(max_probabilities, min_probabilities)
for probabilities in dice_probabilities:
    print(*probabilities)
