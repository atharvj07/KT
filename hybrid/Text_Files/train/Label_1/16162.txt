#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 0; i < n - 4; i++) {
    if (s[i] == '*' && s[i + 1] == '*' && s[i + 2] == '*' && s[i + 3] == '*' &&
        s[i + 4] == '*')
      ans = 1;
  }
  if (ans == 1)
    cout << "yes" << endl;
  else
    cout << "no" << endl;
  return 0;
}
