import java.util.*;
import java.math.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        String S = sc.next();
        String T = sc.next();

        int[] periodS = new int[N];
        int[] periodT = new int[M];
        int[] prefixS = new int[N + 1];
        int[] prefixT = new int[M + 1];

        int gcd = BigInteger.valueOf(N).gcd(BigInteger.valueOf(M)).intValue();

        int pS = -1;
        int pT = -1;
        for (int i = 0; i < N; i++) {
            if (S.charAt(i) != S.charAt((i + gcd) % N)) {
                pS = i;
                break;
            }
        }
        for (int i = 0; i < N; i++) {
            if ((i + 1) % gcd != 0) {
                periodS[i] = periodS[(i + gcd) % N];
            } else {
                periodS[i] = pS;
            }
        }
        for (int i = 0; i < M; i++) {
            if (T.charAt(i) != T.charAt((i + gcd) % M)) {
                pT = i;
                break;
            }
        }
        for (int i = 0; i < M; i++) {
            if ((i + 1) % gcd != 0) {
                periodT[i] = periodT[(i + gcd) % M];
            } else {
                periodT[i] = pT;
            }
        }
        for (int i = 0; i < N; i++) {
            prefixS[i + 1] = prefixS[i];
            if (periodS[i] != -1) {
                prefixS[i + 1] += S.charAt(i) == S.charAt(periodS[i]) ? 1 : -1;
            }
        }
        for (int i = 0; i < M; i++) {
            prefixT[i + 1] = prefixT[i];
            if (periodT[i] != -1) {
                prefixT[i + 1] += T.charAt(i) == T.charAt(periodT[i]) ? 1 : -1;
            }
        }
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < M; j++) {
                if (prefixS[i + 1] == 0 && prefixT[j + 1] == 0) {
                    if (((i + 1) / gcd * (j + 1) / gcd) * gcd == N * M) {
                        int len = (i + 1) * (j + 1) / gcd;
                        System.out.println(len);
                        return;
                    }
                }
            }
        }
        System.out.println(-1);
    }
}
