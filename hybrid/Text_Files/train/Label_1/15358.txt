#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> dp(n, 0);
  int ans = 0;
  for (int i = 3; i < n; ++i) {
    if (s[i] == s[i - 1] && s[i] == s[i - 2] && s[i] == s[i - 3]) {
      dp[i] = dp[i - 4] + 1;
    }
    ans = max(ans, dp[i]);
  }
  cout << ans << endl;
  return 0;
}
