#include <iostream>
#include <string>
using namespace std;

int main() {
    int n, q;
    cin >> n >> q;
    string s;
    cin >> s;
    while (q--) {
        int l, r;
        cin >> l >> r;
        int len = r - l + 1;
        for (int i = 0; i < len - 1; i++) {
            len += s[l + i] - 96;
        }
        cout << len << endl;
    }
    return 0;
}
