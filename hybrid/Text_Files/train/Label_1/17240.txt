#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long b1,q,l,bad[100005],m,i,bad_index=0,mark=0;
    string ans;
    cin>>b1>>q>>l>>m;
    for(i=1;i<=m;i++)
    {
        cin>>bad[i];
    }
    long long f=b1,cnt=0;
    while(1){
        if(f==b1 && mark==1){
            ans="inf";
            break;
        }
        if(f==0)
            mark=1;
        if(f>l||f<-l){
            break;
        }
        if(f==bad[bad_index])
            bad_index++;
        if(bad_index>m)
            bad_index=m;
        f+=q;
        cnt++;
    }
    if(ans!="inf")
        cout<<cnt<<endl;
    else
        cout<<ans<<endl;
    return 0;
}
