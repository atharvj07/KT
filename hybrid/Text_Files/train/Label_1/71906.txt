n,m=map(int,input().split())
a=[]
b=[]
for i in range(n):
    a.append(input())
for i in range(m):
    for j in range(n):
        b.append(a[j][i])
c=""
d=""
for i in a:
    c+=i[0]
for i in b:
    d+=i[0]
if c.count('R')==c.count('G')==c.count('B')==n//3 and d.count('R')==d.count('G')==d.count('B')==m//3:
    print("YES")
else: print("NO")
