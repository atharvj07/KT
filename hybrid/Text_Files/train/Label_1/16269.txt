#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int y, b, r;
  std::cin >> y >> b >> r;

  int min_y = std::min(y, b - 1);
  int min_b = std::min(b, r - 1);
  int min_r = std::min(r, b + 1);

  int max_ornament = 3 * std::min(min_y, std::min(min_b, min_r));
  std::cout << max_ornament << std::endl;

  return 0;
}
