#include<bits/stdc++.h>
using namespace std;
const int M=1e9+7;
int f[230][3][1<<3];
int n,t;
int a[20][2];
int dp(int rem, int lst, int msk){
    int &ans=f[rem][lst][msk];
    if(~ans) return ans;
    ans=0;
    if(rem==0) return ans=1;
    for(int i=0; i<n; i++){
        if(!(msk&(1<<a[i][1])) && a[i][0]<=rem){
            if(a[i][1]==lst){
                ans=(ans+dp(rem-a[i][0],lst,msk))%M;
            }
            else{
                ans=(ans+dp(rem-a[i][0],a[i][1],msk|(1<<a[i][1])))%M;
            }
        }
    }
    return ans;
}
int main(){
    cin>>n>>t;
    for(int i=0; i<n; i++){
        cin>>a[i][0]>>a[i][1];
    }
    memset(f,-1,sizeof f);
    cout<<dp(t,0,0)<<endl;
}
