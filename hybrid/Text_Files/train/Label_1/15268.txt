#include <bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;
int main() {
  long long n, p, k, a1, a2, cur = 0;
  cin >> n >> p >> k;
  vector<int> cnt(p, 0), fre(p, 0);
  for (int i = 0; i < n; i++) {
    cin >> a1;
    if (fre[a1] == 0) cur += fre[(p - a1) % p];
    cnt[a1]++;
    fre[a1]++;
  }
  a1 = a2 = 0;
  for (int i = 0; i < p; i++) {
    if (fre[i]) {
      a1 = (a1 + 1ll * fre[i] * (fre[i] - 1)) % mod;
      a2 = (a2 + 1ll * fre[i] * fre[(k - i) % p]) % mod;
    }
  }
  cout << ((cur + (a1 - a2 + mod)) % mod) << endl;
}
