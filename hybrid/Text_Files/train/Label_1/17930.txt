#include<bits/stdc++.h>
#define pb push_back
#define vi vector<int>
using namespace std;
int n,a,b;
int main(){
  cin>>n>>a>>b;
  if(a>b/2){ cout<<-1; return 0; }
  if(b==a*2){
    for(int i=0;i<a;i++){
      for(int j=0;j<b;j++){
        if((i+j)%2==1&&n>0) cout<<n--, j--;
        else cout<<0<<" ";
      }
      cout<<endl;
    }
    return 0;
  }
  for(int i=0;i<a;i++){
    for(int j=0;j<b;j++){
      if((i+j)%2==0&&n>0) cout<<n--, j--;
      else cout<<0<<" ";
    }
    cout<<endl;
  }
}
