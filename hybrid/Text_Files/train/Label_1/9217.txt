#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    std::string s;
    std::cin >> s;
    std::vector<int> ans(s.length());
    int al = 0, bl = 0;
    int n = s.length();
    for (int i = 0; i < n; i++) {
        if (s[i] == 'a') {
            al++;
        } else {
            bl++;
        }
        if (bl > al) {
            ans[i] = 1;
        }
    }
    for (int i = 0; i < n; i++) {
        std::cout << ans[i] << ' ';
    }
    std::cout << std::endl;
    return 0;
}
