#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int x[8], y[8];
  for (int i = 0; i < 8; i++) {
    std::cin >> x[i] >> y[i];
  }
  std::sort(x, x + 8);
  std::sort(y, y + 8);
  for (int i = 0; i < 3; i++) {
    for (int j = 0; j < 3; j++) {
      if (i == 1 && j == 1) {
        continue;
      }
      int cnt = 0;
      for (int k = 0; k < 8; k++) {
        if (x[k] == x[i] && y[k] == y[j]) {
          cnt++;
        }
      }
      if (cnt != 1) {
        std::cout << "ugly\n";
        return 0;
      }
    }
  }
  std::cout << "respectable\n";
}
