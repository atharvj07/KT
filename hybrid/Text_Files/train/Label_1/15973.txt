#include<iostream>
using namespace std;
long long a[100005], b[100005];
int main(){
    int n, x1, x2, ans = 0;
    cin >> n >> x1 >> x2;
    for(int i = 1; i <= n; i++) cin >> a[i] >> b[i];
    for(int i = 1; i <= n; i++)
        for(int j = i + 1; j <= n; j++)
            if((a[i] - a[j]) * (x1 - b[i]) > (b[i] - b[j]) * (x1 - a[i]) && (a[i] - a[j]) * (x2 - b[i]) < (b[i] - b[j]) * (x2 - a[i])) {
                cout << "YES";
                return 0;
            }
    cout << "NO";
    return 0;
}
