import math

def gcd(x, y):
    if y == 0:
        return x
    return gcd(y, x % y)

def get_lcm(x, y):
    lcm = (x * y) // gcd(x, y)
    return math.floor(lcm)

for _ in range(int(input())):
    l, r = map(int, input().split())
    
    if get_lcm(l, l + 1) <= r:
        print(l, l + 1)
        continue
    
    if get_lcm(l + 1, l + 2) <= r:
        print(l + 1, l + 2)
        continue
    
    if l + 3 <= r:
        print(l + 2, l + 3)
        continue
    
    print(-1, -1)
