import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int xv = in.nextInt();
        int yv = in.nextInt();
        int xp = in.nextInt();
        int yp = in.nextInt();
        int xw1 = in.nextInt();
        int yw1 = in.nextInt();
        int xw2 = in.nextInt();
        int yw2 = in.nextInt();
        int xm1 = in.nextInt();
        int ym1 = in.nextInt();
        int xm2 = in.nextInt();
        int ym2 = in.nextInt();
        System.out.println(canSee(xv, yv, xp, yp, xw1, yw1, xw2, yw2, xm1, ym1, xm2, ym2) ? "YES" : "NO");
    }

    // Check if two points can see each other with the given obstacles
    // Obstacles: wall - a segment joining points (xw1, yw1) and (xw2, yw2), mirror - a segment joining points (xm1, ym1) and (xm2, ym2)
    public static boolean canSee(int xv, int yv, int xp, int yp, int xw1, int yw1, int xw2, int yw2, int xm1, int ym1, int xm2, int ym2) {
        // Check if the line of sight is parallel to the mirror
        boolean parallel = (xm1 == xm2);

        // Check if the line of sight intersects with the wall
        boolean intersectsWall = intersects(xv, yv, xp, yp, xw1, yw1, xw2, yw2);

        // Check if the line of sight intersects with the mirror
        boolean intersectsMirror = intersects(xv, yv, xp, yp, xm1, ym1, xm2, ym2);

        // If the line of sight is parallel to the mirror or intersects with the wall, then Victor and Peter cannot see each other
        if (parallel || intersectsWall) {
            return false;
        }

        // If the line of sight intersects with the mirror and Victor and Peter are on the same side of the mirror, then they can see each other
        if (intersectsMirror && onSameSide(xv, yv, xp, yp, xm1, ym1, xm2, ym2)) {
            return true;
        }

        // Otherwise, Victor and Peter cannot see each other
        return false;
    }

    // Check if the line segment from (x1, y1) to (x2, y2) intersects with the line segment from (x3, y3) to (x4, y4)
    public static boolean intersects(int x1, int y1, int x2, int y2, int x3, int y3, int x4, int y4) {
        // Check if the two line segments are colinear
        boolean colinear = (x1 * (y3 - y4) + y1 * (x4 - x3) + x2 * (y4 - y3) == 0);

        // If the line segments are not colinear, then they cannot intersect
        if (!colinear) {
            return false;
        }

        // Check if the two line segments overlap
        boolean overlapX = (Math.min(x1, x2) <= Math.max(x3, x4) && Math.max(x1, x2) >= Math.min(x3, x4));
        boolean overlapY = (Math.min(y1, y2) <= Math.max(y3, y4) && Math.max(y1, y2) >= Math.min(y3, y4));

        // If the line segments overlap, then they intersect
        return overlapX && overlapY;
    }

    // Check if the points (x1, y1) and (x2, y2) are on the same side of the line segment from (x3, y3) to (x4, y4)
    public static boolean onSameSide(int x1, int y1, int x2, int y2, int x3, int y3, int x4, int y4) {
        // Check if the two points are on the same side of the line y = mx + b
        boolean sameSide = ((y1 - y3) * (x4 - x3) - (x1 - x3) * (y4 - y3)) >= 0;

        // Check if the two points are on different sides of the line segment
        boolean differentSides = ((x1 - x3) * (y4 - y3) - (y1 - y3) * (x4 - x3)) != 0;

        // Return true if the two points are on the same side of the line segment, and false otherwise
        return sameSide && !differentSides;
    }
}
