n = int(input())
arr = [[int(x) for x in input().split()] for _ in range(n)]
dist = [[1000000000] * n for _ in range(n)]
for i in range(n):
    dist[i][i] = 0
    for j in range(i, n):
        dist[i][j] = arr[i][j]
        dist[j][i] = arr[i][j]
for k in range(n):
    for i in range(n):
        for j in range(n):
            dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])
mx = 0
for i in range(n):
    for j in range(n):
        if i != j:
            mx = max(mx, dist[i][j])
print(mx)
