#include <iostream>
#include <vector>
#include <map>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    string s, t;
    cin >> s >> t;

    vector<int> ans;
    map<char, int> freq;
    for (int i = 0; i < n; i++) {
        freq[s[i]]++;
    }

    for (int i = 0; i < m; i++) {
        freq[t[i]]--;
        if (freq[t[i]] < 0) {
            ans.push_back(i + 1);
        }
    }

    cout << ans.size() << endl;
    for (int i : ans) {
        cout << i << " ";
    }
    cout << endl;

    return 0;
}
