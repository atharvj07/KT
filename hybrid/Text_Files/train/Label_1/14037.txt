#include <cstdio>
#include <vector>
#include <string>
#include <iostream>
using namespace std;
int main() {
  int v, ulim, pd[10000], ucnt = 0;
  scanf("%d", &v);
  for (; v--;) {
    for (int i = 0; i < 10000; i++) pd[i] = 0;
    scanf("%d", &ulim);
    int sum = 0;
    for (int i = 0; i < 250; i++) {
      int num;
      scanf("%d", &num);
      sum += num;
      pd[num]++;
      if (num <= 2 * ulim) ucnt++;
    }
    if (ucnt > 250 || sum > 250 * ulim)
      printf("poisson\n");
    else
      printf("uniform\n");
    ucnt = 0;
  }
  return 0;
}
