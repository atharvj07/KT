import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] heights = new int[n];
        for (int i = 0; i < n; i++) {
            heights[i] = input.nextInt();
        }
        int jumps = 0;
        int max = heights[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            if (heights[i] >= max) {
                jumps++;
            } else {
                max = heights[i];
            }
        }
        System.out.println(jumps);
    }
}
