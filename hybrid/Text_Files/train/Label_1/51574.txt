import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        long n = scan.nextLong(), k = scan.nextLong(), p = scan.nextLong(), ans = 0;
        PriorityQueue<Long> people = new PriorityQueue<>(), keys = new PriorityQueue<>();
        for (long i = 0; i < n; i++) people.add(scan.nextLong());
        for (long i = 0; i < k; i++) keys.add(scan.nextLong());
        while (!people.isEmpty() && !keys.isEmpty()) {
            if (keys.peek() < people.peek()) {
                keys.poll();
                ans++;
            } else {
                people.poll();
                ans += Math.abs(p - keys.poll());
            }
        }
        while (!people.isEmpty()) {
            people.poll();
            ans += p;
        }
        System.out.println(ans);
    }
}
