n = int(input())
opening = {}
closing = {}
balanced = 0
for _ in range(n):
    s = input()
    balance = 0
    min_balance = 0
    for c in s:
        if c == '(':
            balance += 1
        else:
            balance -= 1
        min_balance = min(min_balance, balance)
    if balance == 0 and min_balance == 0:
        balanced += 1
    elif balance > 0 and min_balance == 0:
        if balance not in opening:
            opening[balance] = 0
        opening[balance] += 1
    elif balance < 0 and min_balance == balance:
        if -balance not in closing:
            closing[-balance] = 0
        closing[-balance] += 1

result = balanced * balanced
for k in opening.keys():
    if k in closing:
        result += opening[k] * closing[k]
print(result)
