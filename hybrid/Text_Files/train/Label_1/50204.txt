import java.util.Scanner;

public class CandyShop {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        long[] allowances = new long[n];
        for (int i = 0; i < n; i++) {
            allowances[i] = scanner.nextLong();
        }

        long totalCandies = 0;
        long minAllowance = allowances[0];
        for (int i = 1; i < n; i++) {
            minAllowance = Math.min(minAllowance, allowances[i]);
            totalCandies += minAllowance;
        }

        long maxCandies = Math.min(totalCandies, m * (long) n);
        System.out.println(maxCandies);
    }
}
