import sys

n = int(input())
x, y = map(int, input().split())
pieces = []
for _ in range(n):
    t, a, b = input().split()
    pieces.append((t, int(a), int(b)))

def check(a, b):
    for t, x, y in pieces:
        if t == 'B':
            if abs(a - x) == abs(b - y):
                return True
        elif t == 'R':
            if a == x or b == y:
                return True
        elif t == 'Q':
            if abs(a - x) == abs(b - y) or a == x or b == y:
                return True
    return False

print("YES" if check(x, y) else "NO")
