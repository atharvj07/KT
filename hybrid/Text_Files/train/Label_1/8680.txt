#include <bits/stdc++.h>
using namespace std;
const int maxN = 1e5 + 5;
vector<int> g[maxN];
int vis[maxN];
int dfs(int node) {
    vis[node] = 1;
    int ret = 1;
    for (int nebr : g[node]) {
        if (vis[nebr] == 0) {
            ret += dfs(nebr);
        }
    }
    return ret;
}
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        g[a].push_back(b);
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (vis[i] == 0) {
            int sz = dfs(i);
            ans = max(ans, sz);
        }
    }
    cout << ans - 1 << '\n';
}
