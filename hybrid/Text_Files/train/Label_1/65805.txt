import sys

n, m = map(int, input().split())
edges = [[] for i in range(n + 1)]
for i in range(m):
    a, b, c = input().split()
    a = int(a)
    b = int(b)
    edges[a].append((b, c))
    edges[b].append((a, c))

vis = [False] * (n + 1)
ans = []

def dfs(node, col):
    if vis[node] == True:
        return
    vis[node] = True
    ans.append(node)
    for u, c in edges[node]:
        if c != col:
            dfs(u, col)

for i in range(1, n + 1):
    if vis[i] == False:
        dfs(i, 'B')

        if len(ans) == n:
            print(len(ans) - 1)
            for i in ans[1:]:
                print(i)
            sys.exit()

print(-1)
