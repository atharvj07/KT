#include <iostream>
using namespace std;
typedef long long ll;
const int maxn = 100001;
ll a[maxn], b[maxn];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n + m; i++)
        cin >> a[i];
    for (int i = 1; i <= n + m; i++)
        cin >> b[i];
    for (int i = 1; i <= n + m; i++) {
        if (b[i]) {
            ll cnt = 0;
            for (int j = 1; j <= n + m; j++) {
                if (!b[j] && a[j] < a[i])
                    cnt++;
            }
            cout << cnt + 1 << ' ';
        }
    }
    cout << endl;
    return 0;
}
