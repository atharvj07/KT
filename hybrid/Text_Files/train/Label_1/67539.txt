for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    ans = 0
    for i in range(n-2):
        l, r = i+1, n-1
        while l < r:
            if a[r]-a[l]<=2:
                ans += r-l
                l += 1
            else: r -= 1
    print(ans)
