from collections import defaultdict

n = int(input())
edges = defaultdict(list)
for _ in range(n - 1):
    a, b = map(int, input().split())
    edges[a].append(b)
    edges[b].append(a)

vis = [False for _ in range(n + 1)]

def dfs(cur, pre):
    vis[cur] = True
    for adj in edges[cur]:
        if not vis[adj]:
            dfs(adj, cur)
    return

ans = []

def dfs2(cur, pre, mx):
    ans.append((pre, cur))
    for adj in edges[cur]:
        if adj != pre:
            if adj > mx:
                mx = adj
            dfs2(adj, cur, mx)


if len(edges[1]) > 2:
    print("NO")
else:
    dfs(1, 1)
    for i in range(1, n + 1):
        if not vis[i]:
            print("NO")
            exit()
    dfs2(1, 1, 0)
    print("YES")
    for edge in ans:
        print(*edge)
