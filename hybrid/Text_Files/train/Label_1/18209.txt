#include <iostream>
int main() {
    int h1, a1, c1, h2, a2;
    std::cin >> h1 >> a1 >> c1 >> h2 >> a2;
    std::cout << (h2 + a1 - 1) / a1 << '\n';
    while (h1 > 0 && h2 > 0) {
        if (h1 > h2) {
            h2 -= a1;
            std::cout << "STRIKE\n";
        } else {
            h1 += c1 - a2;
            std::cout << "HEAL\n";
        }
    }
}
