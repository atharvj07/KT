recipe = input()
n_b, n_s, n_c = map(int, input().split())
p_b, p_s, p_c = map(int, input().split())
r = int(input())

num_b = recipe.count('B')
num_s = recipe.count('S')
num_c = recipe.count('C')

low = 0
high = 10**13
while low < high:
    mid = (low + high + 1) // 2
    total_cost = max(0, (mid * num_b - n_b) * p_b) + max(0, (mid * num_s - n_s) * p_s) + max(0, (mid * num_c - n_c) * p_c)
    if total_cost <= r:
        low = mid
    else:
        high = mid - 1

print(low)
