import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i] * (i + 1);
        }
        long max = sum;
        for (int i = 0; i < n - 1; i++) {
            sum = sum + a[i + 1] - a[i] * (n - i);
            max = Math.max(max, sum);
        }
        System.out.println(max);
    }
}
