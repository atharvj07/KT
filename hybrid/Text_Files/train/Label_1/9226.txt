#include <iostream>
using namespace std;

int main() {
  char m[4][4];
  for (int i = 0; i < 4; i++) {
    for (int j = 0; j < 4; j++) { cin >> m[i][j]; }
  }
  for (int i = 0; i < 4; i++) {
    for (int j = 0; j < 4; j++) {
      if (m[i][j] == '.') {
        m[i][j] = 'x';
        int cntr = 0, cntc = 0, cntd1 = 0, cntd2 = 0;
        for (int u = 0; u < 4; u++) {
          cntr += (m[i][u] == 'x');
          cntc += (m[u][j] == 'x');
          cntd1 += (m[u][u] == 'x');
          cntd2 += (m[u][3 - u] == 'x');
        }
        if (cntr == 3 || cntc == 3 || cntd1 == 3 || cntd2 == 3) {
          cout << "YES" << endl;
          return 0;
        }
        m[i][j] = '.';
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
