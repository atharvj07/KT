import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        int n = Integer.parseInt(tokenizer.nextToken());
        int m = Integer.parseInt(tokenizer.nextToken());
        int[] array = new int[n];
        tokenizer = new StringTokenizer(reader.readLine());
        for (int i = 0; i < n; i++) {
            array[i] = Integer.parseInt(tokenizer.nextToken());
        }
        List<Integer> queries = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int operation = Integer.parseInt(tokenizer.nextToken());
            switch (operation) {
                case 1:
                    array[Integer.parseInt(tokenizer.nextToken()) - 1] = Integer.parseInt(tokenizer.nextToken());
                    break;
                case 2:
                    int increment = Integer.parseInt(tokenizer.nextToken());
                    for (int j = 0; j < n; j++) {
                        array[j] += increment;
                    }
                    break;
                case 3:
                    queries.add(Integer.parseInt(tokenizer.nextToken()) - 1);
                    break;
            }
        }
        for (int i : queries) {
            System.out.println(array[i]);
        }
    }
}
