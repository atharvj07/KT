import sys
input = sys.stdin.readline

for t in range(int(input())):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort()
    ans = 0
    i = 0
    j = n - 1
    while i <= j:
        if a[i] * (j - i + 1) >= x:
            ans += 1
            i += 1
        else:
            j -= 1
    print(ans)
