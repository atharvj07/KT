import sys
sys.setrecursionlimit(1000001)

N = int(input())
dist = [0] * (N + 1)
a = list(map(int, input().split()))
sad = [0] * (N + 1)

p = [0] * (N + 1)
c = [0] * (N + 1)
for i in range(N - 1):
    x, y = map(int, input().split())
    p[i + 2] = x
    c[i + 2] = y

mem = [0] * (N + 1)

def dfs(v):
    if mem[v]: return mem[v]

    mem[v] = 1
    for u in range(2, N + 1):
        if p[u] == v:
            mem[v] += dfs(u)

    if a[v - 1] < dist[v]: sad[v] = 1

    return mem[v]

def dfs_dist(v, d):
    if mem[v] == 1: return

    mem[v] = 1
    dist[v] = d

    for u in range(2, N + 1):
        if p[u] == v: dfs_dist(u, d + c[u])

for i in range(2, N + 1): dfs_dist(i, c[i])
for i in range(N - 1, 0, -1): dfs(p[i])

print(sum(sad))
