#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 360;
  for (int i = 0; i < n; i++) {
    int cur = 0;
    for (int j = i; j < n; j++) { cur += a[j]; }
    for (int j = 0; j < i; j++) { cur += a[j]; }
    ans = min(ans, abs(cur - 360 + cur));
  }
  cout << ans << endl;
  return 0;
}
