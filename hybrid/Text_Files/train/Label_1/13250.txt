#include <bits/stdc++.h>
using namespace std;
#define MAXN 305
int n, a[MAXN][MAXN], vis[MAXN], p[MAXN];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> p[i];
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++) {
            char c;
            cin >> c;
            a[i][j] = c - '0';
        }
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (a[i][j] && p[i] > p[j]) {
                swap(p[i], p[j]);
                cout << i << ' ' << j << endl;
            }
        }
    }
    return 0;
}
