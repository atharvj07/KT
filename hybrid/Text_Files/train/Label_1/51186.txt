import java.util.Scanner;
import java.util.HashMap;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        HashMap<Character, Integer> cnt = new HashMap<>();
        for (int i = 0; i < n; i++) {
            if (!cnt.containsKey(s.charAt(i))) {
                cnt.put(s.charAt(i), 0);
            }
            cnt.put(s.charAt(i), cnt.get(s.charAt(i)) + 1);
        }
        if (n == 1) {
            System.out.println(s.charAt(0));
            return;
        }
        int[] arr = new int[3];
        arr[0] = cnt.getOrDefault('B', 0);
        arr[1] = cnt.getOrDefault('G', 0);
        arr[2] = cnt.getOrDefault('R', 0);
        Arrays.sort(arr);
        if (arr[0] == 0) {
            System.out.println(arr[1] == 0 ? String.valueOf(arr[2]) : String.valueOf(arr[1]) + String.valueOf(arr[2]));
        } else {
            System.out.println(arr[0] == arr[1] ? String.valueOf(arr[0]) + String.valueOf(arr[2]) : String.valueOf(arr[0]) + String.valueOf(arr[1]));
        }
    }
}
