n, m, h = map(int, input().split())
front_view = list(map(int, input().split()))
left_view = list(map(int, input().split()))
top_view = [list(map(int, input().split())) for _ in range(n)]

for i in range(n):
    for j in range(m):
        if top_view[i][j] == 1:
            print(min(left_view[i], front_view[j]), end=' ')
        else:
            print(0, end=' ')
    print()
