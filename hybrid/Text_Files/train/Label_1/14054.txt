#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    unordered_map<string, bool> mp;
    while (t--) {
        string s;
        cin >> s;
        int n = s.size();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (mp[s.substr(0, i + 1)])
                ans++;
            else {
                mp[s.substr(0, i + 1)] = 1;
                ans += 5;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
