#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 1;

vector<int> id;
int ans[maxn];

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; ++i) id.push_back(i);
    for (int i = 0, j = n; i < j; ++i, --j) {
        ans[i] = id[i];
        ans[j] = id[j - 1];
    }
    for (int i = 0; i < n; ++i) cout << ans[i] << ' ';
    return 0;
}
