#include <bits/stdc++.h>
using namespace std;
int main() {
    int q;
    cin >> q;
    while (q--) {
        string s, t;
        cin >> s >> t;
        int g = __gcd(s.size(), t.size());
        string lcm = "";
        for (int i = 0; i < g; i++) { lcm += s[i]; }
        if (s.size() % g != 0) {
            bool ok = 1;
            for (int i = g; i < s.size(); i += g) {
                if (s.substr(i, g) != lcm) ok = 0;
            }
            if (!ok) {
                lcm = "-1";
            }
        }
        if (t.size() % g != 0) {
            bool ok = 1;
            for (int i = g; i < t.size(); i += g) {
                if (t.substr(i, g) != lcm) ok = 0;
            }
            if (!ok) {
                lcm = "-1";
            }
        }
        cout << lcm << endl;
    }
}
