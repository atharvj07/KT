#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
struct point {
    long long pos, h;
    int flag;
    bool operator<(const point& p) const {
        return pos < p.pos;
    }
};
int main() {
    int n; cin >> n;
    vector<point> p(n);
    for (int i = 0; i < n; i++) {
        cin >> p[i].pos >> p[i].h;
        p[i].flag = 0;
    }
    sort(p.begin(), p.end());
    int last = -1000000007, cnt = 0;
    for (int i = 0; i < n; i++) {
        if (last <= p[i].pos - p[i].h) {
            cnt++;
            last = p[i].pos + p[i].h;
        }
        else if (last <= p[i].pos + p[i].h) {
            last = p[i].pos + p[i].h;
            p[i].flag = 1;
        }
    }
    for (int i = n - 1; i >= 0; i--) {
        if (!p[i].flag && last >= p[i].pos - p[i].h) {
            cnt++;
            last = p[i].pos - p[i].h;
        }
    }
    cout << cnt << endl;
}
