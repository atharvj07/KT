#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-6;
const int maxn = 1e3 + 5;
struct Point {
    double x, y;
    Point() {}
    Point(double _x, double _y): x(_x), y(_y) {}
    double dist(const Point& p) {return hypot(x - p.x, y - p.y);}
};
Point p;
int n;
struct Taxi {
    Point pos;
    double v;
    Taxi() {}
    Taxi(const Point& _pos, double _v): pos(_pos), v(_v) {}
    double get_time() {return p.dist(pos) / v;}
} t[maxn];
int main() {
    int x, y; cin >> x >> y; p = Point(x, y);
    cin >> n;
    for (int i = 0; i < n; i++) {
        int x, y, v;
        cin >> x >> y >> v;
        t[i] = Taxi(Point(x, y), v);
    }
    double ans = 1e9;
    for (int i = 0; i < n; i++) {
        ans = min(ans, t[i].get_time());
    }
    cout << fixed << setprecision(12) << ans << endl;
    return 0;
}
