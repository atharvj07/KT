import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[][] uniforms = new int[n][2];

        // Read the uniforms of each team
        for (int i = 0; i < n; i++) {
            uniforms[i][0] = input.nextInt();
            uniforms[i][1] = input.nextInt();
        }

        // Count the number of games where the host team will wear the guest uniform
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (uniforms[i][0] == uniforms[j][1]) {
                    count++;
                }
            }
        }

        // Print the count
        System.out.println(count);
    }
}
