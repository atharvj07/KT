import java.util.Scanner;

public class Main {
  static boolean isKeyence(String s) {
    if (s.length() == 7) {
      return true;
    }
    if (s.length() < 7 || s.length() > 100) {
      return false;
    }

    int n = s.length();
    for (int i = 1; i < n; i++) {
      String t = s.substring(0, i) + s.substring(i + 1);
      if (t.equals("keyence")) {
        return true;
      }
    }

    return false;
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    String s = sc.nextLine();
    System.out.println(isKeyence(s) ? "YES" : "NO");
  }
}
