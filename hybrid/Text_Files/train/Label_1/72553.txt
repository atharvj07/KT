n = int(input())
a = list(map(int, input().split()))
for i in range(n - 2):
    if a[i] != a[i + 1] - 1 + a[i + 2] - 2:
        a[i + 2] = a[i + 1] - 1 + a[i + 2] - 2
ans = 0
l = -1
for i in range(n):
    if a[i] != i + 1:
        l = i
        break
r = -1
for i in range(n - 1, -1, -1):
    if a[i] != i + 1:
        r = i
        break
print(max(r - l + 1, 2))
