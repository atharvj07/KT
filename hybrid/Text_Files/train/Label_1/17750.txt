#include<bits/stdc++.h>
using namespace std;
int main() {
    long long n, d;
    cin >> n >> d;
    int m;
    cin >> m;
    vector<long long> a(m);
    for (int i = 0; i < m; i++) cin >> a[i];
    long long den = 1, num = a[m - 1];
    for (int i = m - 2; i >= 0; i--) {
        long long temp = num; num = (den * a[i]) + num; den = temp;
    }
    if (n * den == d * num) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
