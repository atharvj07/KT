import java.util.Scanner;

public class NeuralNetworkCountry {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int L = scanner.nextInt();
        int M = scanner.nextInt();

        int[] entryCosts = new int[N];
        int[] interLayerCosts = new int[N];
        int[] exitCosts = new int[N];

        for (int i = 0; i < N; i++) {
            entryCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < N; i++) {
            interLayerCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < N; i++) {
            exitCosts[i] = scanner.nextInt();
        }

        long[][] dp = new long[L + 1][N];
        dp[0][0] = 1;

        for (int layer = 1; layer <= L; layer++) {
            for (int city = 0; city < N; city++) {
                for (int prevCity = 0; prevCity < N; prevCity++) {
                    long cost = entryCosts[prevCity] + (layer - 1) * interLayerCosts[prevCity] + exitCosts[city];
                    if (cost % M == 0) {
                        dp[layer][city] = (dp[layer][city] + dp[layer - 1][prevCity]) % MOD;
                    }
                }
            }
        }

        long totalPaths = 0;
        for (int city = 0; city < N; city++) {
            totalPaths = (totalPaths + dp[L][city]) % MOD;
        }

        System.out.println(totalPaths);
    }
}
