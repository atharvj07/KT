#include <bits/stdc++.h>
using namespace std;
struct Edge {
    int v, w;
};
vector<Edge> G[200005];
long long ans = LLONG_MAX;
int n;
long long a[200005];
int dp[200005][2]; // 0 -> not flip, 1 -> flipped
void dfs(int node, int par) {
    dp[node][0] = a[node];
    dp[node][1] = LLONG_MAX;
    for (Edge e : G[node]) {
        if (e.v == par) continue;
        dfs(e.v, node);
        dp[node][0] += min(dp[e.v][0], dp[e.v][1] + e.w);
        dp[node][1] += min(dp[e.v][0] + e.w, dp[e.v][1]);
    }
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int u, v, w;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        int b, c;
        cin >> b >> c;
        if (b != c) a[i] = LLONG_MAX;
    }
    for (int i = 1; i < n; i++) {
        cin >> u >> v;
        cin >> w;
        G[u].push_back({v, w});
        G[v].push_back({u, w});
    }
    dfs(1, 1);
    cout << (dp[1][0] == LLONG_MAX ? -1 : dp[1][0]);
}
