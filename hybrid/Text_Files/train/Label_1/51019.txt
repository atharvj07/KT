import java.util.*;

public class Main {
    static int n, m;
    static int[] a;
    static int[][] b;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        a = new int[n + 1];
        for (int i = 1; i <= n; i++) a[i] = sc.nextInt();
        b = new int[m][2];
        for (int i = 0; i < m; i++) {
            b[i][0] = sc.nextInt();
            b[i][1] = sc.nextInt();
        }
        int ans = 0;
        for (int[] p : b) {
            int x = a[p[0]], y = a[p[1]];
            for (int i = 2; i * i <= x; i++) {
                while (x % i == 0 && y % i == 0) {
                    x /= i;
                    y /= i;
                    ans++;
                }
            }
            if (x > 1 && y > 1) ans++;
        }
        System.out.println(ans);
    }
}
