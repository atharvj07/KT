n, k = map(int, input().split())
books = [tuple(map(int, input().split())) for _ in range(n)]

both_likes = []
alice_likes = []
bob_likes = []
neither_likes = []

for t, a, b in books:
    if a and b:
        both_likes.append(t)
    elif a:
        alice_likes.append(t)
    elif b:
        bob_likes.append(t)
    else:
        neither_likes.append(t)

both_likes.sort()
alice_likes.sort()
bob_likes.sort()
neither_likes.sort()

alice_count, bob_count = 0, 0
total_time = 0
i = 0
while i < len(both_likes) and alice_count < k and bob_count < k:
    total_time += both_likes[i]
    alice_count += 1
    bob_count += 1
    i += 1

if alice_count < k:
    diff = k - alice_count
    if len(alice_likes) >= diff:
        total_time += sum(alice_likes[:diff])
        alice_count = k
    else:
        print(-1)
        exit()

if bob_count < k:
    diff = k - bob_count
    if len(bob_likes) >= diff:
        total_time += sum(bob_likes[:diff])
        bob_count = k
    else:
        print(-1)
        exit()

if alice_count < k or bob_count < k:
    print(-1)
else:
    additional_books = both_likes[i:] + alice_likes[k-alice_count:] + bob_likes[k-bob_count:] + neither_likes
    additional_books.sort()
    additional_required = max(0, k - alice_count, k - bob_count)
    total_time += sum(additional_books[:additional_required])
    print(total_time)
