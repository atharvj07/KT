n = int(input())
f = [int(x) for x in input().split()]
c = [0] * (n + 1)
for i in f:
    c[i] += 1
for i in range(1, n + 1):
    if c[i] == 0:
        for j in range(1, n + 1):
            if i != j and f[j] == 0:
                f[j] = i
                break
    elif c[i] > 1:
        for j in range(1, n + 1):
            if f[j] == i:
                for k in range(1, n + 1):
                    if j != k and f[k] == 0:
                        f[k] = i
                        break
                break
print(*f)
