import java.util.*;

class Main {
    public static void main(String[] args) {
        class Pair implements Comparable<Pair> {
            private String a;
            private int b;

            public Pair(String a, int b) {
                this.a = a;
                this.b = b;
            }

            public int compareTo(Pair other) {
                return this.b - other.b;
            }

            public String toString() {
                return "(" + a + ", " + b + ")";
            }
        }

        Scanner scn = new Scanner(System.in);
        int n = scn.nextInt();
        Map<String, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String str = scn.next();
            map.put(str, scn.nextInt());
        }
        int m = scn.nextInt();
        UnionFind uf = new UnionFind();
        for (int i = 0; i < m; i++) {
            String a = scn.next();
            String b = scn.next();
            uf.unionSet(a, b);
        }
        List<Pair> list = new ArrayList<>();
        for (String key : map.keySet()) {
            list.add(new Pair(uf.findSet(key), map.get(key)));
        }
        Collections.sort(list);
        int ans = 0;
        for (Pair p : list) {
            if (!uf.isSameSet(p.a, "super")) {
                ans += p.b;
                uf.unionSet(p.a, "super");
            }
        }
        System.out.println(ans);
    }

    static class UnionFind {
        private Map<String, String> parent;

        public UnionFind() {
            this.parent = new HashMap<>();
        }

        public void unionSet(String a, String b) {
            parent.put(findSet(a), findSet(b));
        }

        public String findSet(String a) {
            if (!parent.containsKey(a)) {
                parent.put(a, a);
                return a;
            }
            if (!parent.get(a).equals(a)) {
                parent.put(a, findSet(parent.get(a)));
            }
            return parent.get(a);
        }

        public boolean isSameSet(String a, String b) {
            return findSet(a).equals(findSet(b));
        }
    }
}
