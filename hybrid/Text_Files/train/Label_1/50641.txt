import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++)
            x[i] = sc.nextInt();
        int sum = 0;
        for (int i = 0; i < n / 2; i++)
            sum += ((i * 2 + 1) * x[i * 2] + i * 2 * x[i * 2 + 1]) * w;
        System.out.println(sum * h);
    }
}
