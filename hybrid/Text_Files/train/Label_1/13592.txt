#include <iostream>
#include <vector>
#include <set>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        set<int> s(a.begin(), a.end());
        bool flag = true;
        for (auto it = s.begin(); it != s.end(); it++) {
            if (*it < 0) {
                flag = false;
                break;
            }
            if (*it % 2 == 1) {
                flag = false;
                break;
            }
        }
        if (flag) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
