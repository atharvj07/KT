#include <iostream>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, d;
        cin >> n >> d;
        int arr[n + 1], b[n + 1];
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        for (int i = 1; i <= n; i++) { b[i] = arr[i]; }
        for (int i = 1; i <= d; i++) {
            for (int j = 1; j <= n; j++) {
                if (j == 1 and arr[j] > 0) { ans++; }
                if (j > 1 and j < n and arr[j] > 0) {
                    if (arr[j - 1] < arr[j] || b[j - 1] == b[j]) {
                        ans++;
                    }
                }
                if (j == n and arr[j] > 0) { ans++; }
                b[j] = arr[j];
            }
            for (int j = 1; j <= n; j++) {
                if (j == 1 and arr[j] > 0) { arr[j]--; }
                if (j > 1 and j < n and arr[j] > 0) {
                    if (arr[j - 1] < arr[j] or b[j - 1] == b[j]) {
                        arr[j]--;
                    }
                }
                if (j == n and arr[j] > 0) { arr[j]--; }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
