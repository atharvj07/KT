#include<bits/stdc++.h>
using namespace std;
int isPrime(long long int x){
    if(x<=1)  return 0;
    if(x<=3)  return 1;
    if(x%2==0||x%3==0)  return 0;
    for(int i=5;i*i<=x;i+=6)
        if(x%i==0||x%(i+2)==0)
            return 0;
    return 1;
}
int main(){
    int n;
    cin>>n;
    for(int m=1;;m++){
        long long int x=1LL*n*m+1;
        if(!isPrime(x)){
            cout<<m<<endl;
            return 0;
        }
    }
    return 0;
}
