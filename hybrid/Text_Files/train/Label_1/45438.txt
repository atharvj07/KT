import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int C = sc.nextInt();
        int N = sc.nextInt();

        // First coaster
        int[][] coaster = new int[N][N];
        for (int i = 0; i < N; i++) {
            String line = sc.next();
            for (int j = 0; j < N; j++) {
                coaster[i][j] = Integer.parseInt(line.charAt(j) + "");
            }
        }

        int answer = 0;
        for (int c = 1; c < C; c++) {
            int D = sc.nextInt();
            boolean isVertical = true;
            boolean isHorizontal = true;
            for (int d = 0; d < D; d++) {
                int r = sc.nextInt() - 1;
                int c1 = sc.nextInt() - 1;
                coaster[r][c1] ^= 1;
                isVertical &= coaster[r][c1] == coaster[r][N - c1 - 1];
                isHorizontal &= coaster[r][c1] == coaster[N - r - 1][c1];
            }
            if (isVertical || isHorizontal) {
                answer++;
            }
        }
        System.out.println(answer);
    }
}
