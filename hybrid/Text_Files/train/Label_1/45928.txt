import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashSet;
import java.util.List;
import java.util.Scanner;

public class Solution {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] segments = new int[n][2];
        for (int i = 0; i < n; i++) {
            segments[i][0] = scanner.nextInt();
            segments[i][1] = scanner.nextInt();
        }

        if (isTree(segments)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isTree(int[][] segments) {
        int n = segments.length;
        // Sort segments by left endpoint
        Arrays.sort(segments, (a, b) -> Integer.compare(a[0], b[0]));

        // Create a list of adjacent segments for each segment
        List<Integer>[] adjacencyList = new List[n];
        for (int i = 0; i < n; i++) {
            adjacencyList[i] = new ArrayList<>();
        }
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (segments[i][1] > segments[j][0] && segments[i][1] < segments[j][1]) {
                    adjacencyList[i].add(j);
                    adjacencyList[j].add(i);
                }
            }
        }

        // Perform a depth-first search to check if the graph is a tree
        boolean[] visited = new boolean[n];
        int[] parent = new int[n];
        Arrays.fill(parent, -1);
        if (!dfs(0, -1, visited, parent, adjacencyList)) {
            return false;
        }

        // Check if there are any cycles in the graph
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                return false;
            }
        }

        return true;
    }

    private static boolean dfs(int node, int parent, boolean[] visited, int[] parentArray, List<Integer>[] adjacencyList) {
        visited[node] = true;
        for (int neighbor : adjacencyList[node]) {
            if (neighbor == parent) {
                continue;
            }
            if (visited[neighbor]) {
                return false;
            }
            parentArray[neighbor] = node;
            if (!dfs(neighbor, node, visited, parentArray, adjacencyList)) {
                return false;
            }
        }
        return true;
    }
}
