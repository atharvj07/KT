import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextLong();
            long k = sc.nextLong();
            long aSpot = (k - 1) % (2 * n - 1) + 1;
            long bSpot = (k - 1 + n) % (2 * n - 1) + 1;
            if (aSpot == bSpot) bSpot++;
            System.out.println(bSpot);
        }
    }
}
