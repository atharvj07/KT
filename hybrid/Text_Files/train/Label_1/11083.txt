#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  long long n, ans1, ans2 = 1;
  cin >> n;
  vector<long long> a(n);
  for (long long i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  ans1 = a.back() - a[0];
  for (long long i = 1; i < n; i++) {
    if (a[i] == a[i - 1]) ans2++;
  }
  cout << ans1 << " " << ans2 << "\n";
  return 0;
}
