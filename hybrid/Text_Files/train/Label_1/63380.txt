from collections import defaultdict, deque

def solve(n, m, roads):
    graph = defaultdict(list)
    for road in roads:
        x, y, z = road
        graph[x].append((y, z))
        graph[y].append((x, z))

    def bfs():
        queue = deque([(1, 0)])
        visited = [False] * (n + 1)
        parents = [-1] * (n + 1)
        while queue:
            node, dist = queue.popleft()
            if visited[node]:
                continue
            visited[node] = True
            if node == n:
                return dist, parents
            for neighbor, state in graph[node]:
                if not visited[neighbor]:
                    parents[neighbor] = (node, state)
                    queue.append((neighbor, dist + 1))
        return float('inf'), parents

    dist, parents = bfs()
    path = []
    current = n
    while parents[current] != -1:
        path.append((current, parents[current][0], parents[current][1]))
        current = parents[current][0]
    path.reverse()

    affected_roads = set()
    for road in roads:
        x, y, z = road
        if (x, y, z) in path or (y, x, z) in path:
            if z == 0: # Repair needed
                affected_roads.add((min(x, y), max(x, y), 1))
        else:
            if z == 1: # Blow up
                affected_roads.add((min(x, y), max(x, y), 0))

    return affected_roads

n, m = map(int, input().split())
roads = [list(map(int, input().split())) for _ in range(m)]
affected_roads = solve(n, m, roads)
print(len(affected_roads))
for road in affected_roads:
    print(*road)
