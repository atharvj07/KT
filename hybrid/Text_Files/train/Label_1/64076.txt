import sys

mod = 998244353

n, m = map(int, input().split())
likes = [int(x) for x in input().split()]
weights = [int(x) for x in input().split()]
like_cnt = sum(likes)

for i in range(n):
    if likes[i] == 0:
        weights[i] -= m
    else:
        weights[i] += m

    p, q = 1, weights[i]
    for j in range(2, min(p, q) + 1):
        while p % j == 0 and q % j == 0:
            p //= j
            q //= j

    print((q * pow(p, mod - 2, mod)) % mod)
