import queue
from collections import defaultdict, deque
x = input().split()
n, m = map(int, x)
l1 = list(map(int, input().split()))
l2 = [deque() for i in range(n+1)]
in_deg = [0]*(n+1)
for i in range(1, n+1):
    x = input().split()
    l2[i].extend(list(map(int, x[1:])))
    for j in x[1:]:
        in_deg[j]+=1
q = queue.Queue()
for i in range(1, n+1):
    if in_deg[i] == 0:
        q.put(i)
ans = []
count = 0
while not q.empty():
    x = q.get()
    count+=1
    ans.append(x)
    while l2[x]:
        y = l2[x].popleft()
        in_deg[y]-=1
        if in_deg[y] == 0:
            q.put(y)
if count < n:
    print(-1)
else:
    print(m)
    print(*ans)
