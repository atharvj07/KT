def transform_image(w, h, image):
    # Rotate 90 degrees clockwise
    rotated_image = [''.join([image[h-1-row][col] for row in range(h)]) for col in range(w)]
    
    # Flip horizontally
    flipped_image = [row[::-1] for row in rotated_image]
    
    # Zoom in twice
    zoomed_image = []
    for row in flipped_image:
        zoomed_row = ''.join([char*2 for char in row])
        zoomed_image.append(zoomed_row)
        zoomed_image.append(zoomed_row)
    
    return zoomed_image

# Read input
w, h = map(int, input().split())
image = [input() for _ in range(h)]

result = transform_image(w, h, image)

# Output result
for row in result:
    print(row)
