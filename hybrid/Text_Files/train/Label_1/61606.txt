import sys
n,k=map(int,input().split())
a=[[0]*n for _ in range(n)]
if n*n-1<k:
    print(-1)
else:
    for i in range(n):
        for j in range(n):
            if i>=j:
                if k>0:
                    a[i][j]=1
                    k-=1
                else:
                    a[i][j]=0
    for i in range(n):
        for j in range(n):
            print(a[i][j],end=' ')
        print()
