#include<bits/stdc++.h>
using namespace std;
long long power(int x,int y)
{
    if(y==0){ return 1; }
    if(y==1){ return x; }
    if(y%2==0)
    {
        long long p=power(x,y/2);
        return (p*p)%5;
    }
    else
    {
        long long p=power(x,y/2);
        return (x*((p*p)%5))%5;
    }
}
int main()
{
    int n;
    cin>>n;
    cout<<(((power(1,n)+power(2,n)+power(3,n)+power(4,n))%5))<<endl;
    return 0;
}
