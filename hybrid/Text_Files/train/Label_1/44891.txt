import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        long ans = 0;
        for (int i = 0; i < n - k; i++) {
            ans += a[i];
        }
        for (int i = n - k; i < n; i++) {
            ans += Math.min(a[i], 0);
        }
        System.out.println(ans);
    }
}
