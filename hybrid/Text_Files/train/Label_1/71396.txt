n = int(input())
contests = [list(map(int, input().split())) for _ in range(n)]

rating_lower_bound = -float('inf')
rating_upper_bound = float('inf')

current_rating = 0

for c_i, d_i in contests:
    if d_i == 1:
        rating_lower_bound = max(rating_lower_bound, 1900 - current_rating)
    else:
        rating_upper_bound = min(rating_upper_bound, 1900 - current_rating)
    current_rating += c_i

    if rating_lower_bound >= rating_upper_bound:
        print("Impossible")
        break
else:
    if rating_upper_bound == float('inf'):
        print("Infinity")
    else:
        print(current_rating + rating_lower_bound)
