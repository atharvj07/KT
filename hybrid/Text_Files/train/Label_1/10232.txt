#include <bits/stdc++.h>
using namespace std;
#define pb push_back
const int N = 1001;
vector<int> g[N * 2];
vector<pair<int, int>> points;
int main() {
    int n;
    cin >> n;
    map<pair<int, int>, int> idx;
    for (int i = 0; i < n; ++i) {
        int x, y;
        cin >> x >> y;
        points.pb({x, y});
        if (!idx.count({x, y})) {
            idx[{x, y}] = i;
            g[i].pb(N + i);
            g[N + i].pb(i);
        }
        for (auto j : idx) {
            int u = j.second;
            auto p = points[u];
            if (p.first == x || p.second == y) {
                g[i].pb(u);
                g[u].pb(i);
                g[i].pb(N + u);
                g[N + u].pb(i);
                g[u].pb(N + i);
                g[N + i].pb(u);
            }
        }
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        for (auto j : g[i]) {
            for (auto k : g[j]) {
                if (j < k && i < j && i < k) ans++;
            }
        }
    }
    cout << ans << '\n';
    return 0;
}
