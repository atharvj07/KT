#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<long long> a(n);
    for (int i = 0; i < n; i++)
        cin >> a[i];
    sort(a.begin(), a.end());
    int l = (k + 1) / 2, r = k;
    if (k % 2 == 0)
        cout << a[r - 1];
    else
        cout << a[l - 1];
    return 0;
}
