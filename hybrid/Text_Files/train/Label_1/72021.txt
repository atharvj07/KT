from collections import Counter
for _ in range(int(input())):
    n, k = map(int,input().split())
    A = Counter(map(int,input().split()))
    mx = Max = max(A.keys())
    for i in range(k):
        mn = next(x for x in A if not A[x])
        mx = (mn + Max + 1) // 2
        A[mx] += 1
        Max = max(Max, mx)
    print(len(A))
