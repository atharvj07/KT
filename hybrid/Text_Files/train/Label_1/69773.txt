import sys
input = sys.stdin.readline

def check(t):
    if t[0]+w<=x1 or t[0]>=x2 or t[1]+h<=y1 or t[1]>=y2:
        return 1
    return 0

t = int(input())
while(t>0):
    w,h = map(int, input().split())
    x1,y1,x2,y2 = map(int, input().split())
    x,y = map(int, input().split())
    l,r = 0, w-x
    ans = -1
    while(l<=r):
        mid = (l+r)//2
        if check([mid,y]):
            r = mid-1
            ans = mid
        else:
            l = mid+1
    print(ans)
    t-=1
