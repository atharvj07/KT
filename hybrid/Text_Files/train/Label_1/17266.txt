#include <bits/stdc++.h>
using namespace std;
const int maxn = 45001;
int g[maxn], sum[2], sz[2];
vector<int> v[2];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> g[i];
  sum[0] = sum[1] = 0;
  sz[0] = sz[1] = 0;
  for (int i = 1; i <= n; i++) {
    if (g[i] % 2 + sum[0] % 2 == sum[1] % 2) {
      v[0].push_back(g[i]);
      sum[0] += g[i];
      sz[0]++;
    } else {
      v[1].push_back(g[i]);
      sum[1] += g[i];
      sz[1]++;
    }
  }
  if (sum[0] == 0 || sum[1] == 0 || __gcd(sum[0], sum[1]) == 1) {
    cout << "No" << endl;
  } else {
    cout << "Yes" << endl;
    cout << sz[0] << ' ';
    for (int x : v[0]) cout << x << ' ';
    cout << endl;
    cout << sz[1] << ' ';
    for (int x : v[1]) cout << x << ' ';
    cout << endl;
  }
  return 0;
}
