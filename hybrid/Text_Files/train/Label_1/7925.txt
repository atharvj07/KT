#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<vector<int>> m(n, vector<int>(5));
    bool flag = true;
    for (int i = 0; i < n; i++) {
      int sum = 0;
      for (int j = 0; j < 5; j++) {
        cin >> m[i][j];
        sum += m[i][j];
      }
      if (sum == 0) flag = false;
    }
    if (n % 2 == 0 && flag) cout << "YES" << endl;
    else cout << "NO" << endl;
  }
  return 0;
}
