#include <iostream>
#include <vector>

using namespace std;

struct node {
  int data;
  int l;
  int r;
};

int main() {
  int n;
  cin >> n;
  vector<node> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i].data >> v[i].l >> v[i].r;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    bool flag = false;
    int cur = i;
    while (cur != -1) {
      if (v[cur].data == v[i].data) {
        flag = true;
        break;
      } else if (v[cur].data < v[i].data) {
        cur = v[cur].r;
      } else {
        cur = v[cur].l;
      }
    }
    if (!flag) ans++;
  }
  cout << ans << endl;
  return 0;
}
