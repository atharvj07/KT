import java.util.Scanner;

public class InnopolisHills {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] heights = new int[n];

        for (int i = 0; i < n; i++) {
            heights[i] = scanner.nextInt();
        }

        int[] minHours = new int[n / 2 + 1];

        for (int k = 1; k <= n / 2; k++) {
            int minHeight = 0;
            int count = 0;

            for (int i = 0; i < n; i++) {
                if (heights[i] > minHeight) {
                    count++;
                    minHeight = heights[i];
                }
            }

            if (count < k) {
                minHeight++;
                count = 0;
                for (int i = 0; i < n; i++) {
                    if (heights[i] > minHeight) {
                        count++;
                        minHeight = heights[i];
                    }
                }
            }

            int hours = 0;
            for (int i = 0; i < n; i++) {
                if (heights[i] > minHeight) {
                    hours += heights[i] - minHeight;
                }
            }

            minHours[k - 1] = hours;
        }

        for (int hours : minHours) {
            System.out.print(hours + " ");
        }
    }
}
