import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int x = input.nextInt();
    int[] a = new int[n];
    int[] b = new int[105];
    for (int i = 0; i < n; i++) {
      a[i] = input.nextInt();
      b[a[i]]++;
    }
    int l = 0;
    int r = 0;
    int s = 0;
    for (int i = 0; i < x; i++) { s += b[i]; }
    int ans = n;
    while (r < 105) {
      while (l < r && s >= (n + r - l) / 2) { s -= b[l++]; }
      ans = Math.min(ans, n + r - s);
      s += b[r++];
    }
    System.out.println(ans);
  }
}
