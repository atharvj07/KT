def expected_value(n, bid_ranges):
    max_bid = 0
    second_max_bid = 0
    for L, R in bid_ranges:
        max_bid = max(max_bid, R)
    
    probabilities = [0] * (max_bid + 1)
    for L, R in bid_ranges:
        for bid in range(L, R + 1):
            probabilities[bid] += 1 / (R - L + 1)
    
    expected_cost = 0
    for second_bid in range(1, max_bid + 1):
        prob_second_bid = 0
        prob_less_than_second = 0
        for L, R in bid_ranges:
            if L <= second_bid <= R:
                prob_second_bid += probabilities[second_bid] / (R - L + 1)
            prob_less_than_second += max(0, min(second_bid - 1, R) - L + 1) / (R - L + 1)
            
        prob_exact_second = prob_second_bid * (1 - prob_less_than_second) ** (n - 1)
        for L, R in bid_ranges:
            if L <= second_bid <= R:
                prob_exact_second += probabilities[second_bid] / (R - L + 1) * (prob_less_than_second - max(0, min(second_bid - 1, R) - L + 1) / (R - L + 1)) * (1 - prob_less_than_second) ** (n - 2)
        
        expected_cost += second_bid * prob_exact_second
    
    return expected_cost

n = int(input().strip())
bid_ranges = [list(map(int, input().strip().split())) for _ in range(n)]

print(f"{expected_value(n, bid_ranges):.10f}")
