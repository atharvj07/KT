import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        long n = scanner.nextLong();
        int k = scanner.nextInt();

        int w = 0;
        while (n % (long) Math.pow(10, k) != 0) {
            n /= 10;
            w++;
        }

        System.out.println(w);
    }
}
