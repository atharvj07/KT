#include <bits/stdc++.h>
using namespace std;
template<typename T>
void read(T& x) {
    x = 0;
    char c = getchar();
    int sign = 1;
    while (!isdigit(c)) {
        if (c == '-') sign = -1;
        c = getchar();
    }
    while (isdigit(c)) x = x * 10 + c - '0', c = getchar();
    x *= sign;
}
const int N = 105;
int n, m, a[N][N], ans;
int main() {
    int t;
    read(t);
    while (t--) {
        read(n), read(m), ans = 0;
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++) read(a[j][i]);
        for (int i = 1; i <= n; i++) {
            int mx = 0;
            for (int j = 1; j <= m; j++) mx = max(mx, a[j][i]);
            ans += mx;
        }
        cout << ans << "\n";
    }
    return 0;
}
