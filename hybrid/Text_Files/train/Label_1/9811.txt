#include<bits/stdc++.h>
using namespace std;
const int N=1e5+7;
int t,n,x,y,a[N],ans;
int main(){
    cin>>t>>x>>y;
    for(int i=1;i<=t;++i) cin>>a[i];
    sort(a+1,a+t+1);
    int sum=a[1];
    for(int i=1;i<=t-max(x,y);++i){
        if(a[i+max(x,y)]-sum>a[i+max(x,y)-1]-sum){
            sum=a[i+max(x,y)];
            ans=i;
        }
    }
    cout<<ans<<endl;
}
