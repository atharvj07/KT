def min_cost_to_paint(n, m, rectangles):
    events = []
    for x1, y1, x2, y2 in rectangles:
        events.append((y1, x1, x2, 1))
        events.append((y2+1, x1, x2, -1))
    events.sort()

    # Sweep line algorithm
    active_intervals = []
    last_y = cost = 0
    for y, x1, x2, delta in events:
        # Calculate cost for the current segment
        segments = 0
        last_x = -1
        for start, end in active_intervals:
            last_x = max(last_x, start)
            if end > last_x:
                segments += end - last_x
                last_x = end
        cost += (y - last_y) * segments

        if delta == 1:
            active_intervals.append((x1, x2))
            active_intervals.sort()
        else:
            active_intervals.remove((x1, x2))
        last_y = y

    return cost

# Read input
n, m = map(int, input().split())
rectangles = [list(map(int, input().split())) for _ in range(m)]

# Calculate and print the result
print(min_cost_to_paint(n, m, rectangles))
