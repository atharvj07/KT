#include <iostream>
using namespace std;
int main() {
  int l, r, x, y;
  cin >> l >> r >> x >> y;
  if (y % x != 0) {
    cout << 0 << endl;
    return 0;
  }
  int res = (r / y - (l - 1) / y) * (r / x - (l - 1) / x);
  if (l % x != 0) { res -= (r / y) % 2; }
  if (l % x == 0 && l / x <= r / y) { res++; }
  cout << res << endl;
  return 0;
}
