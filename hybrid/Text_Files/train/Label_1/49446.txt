import java.util.Arrays;
import java.util.Scanner;

public class Main {
    private static String s;
    private static int[] pre, suf, ans;

    private static void pre() {
        pre[0] = s.indexOf("one");
        pre[1] = s.indexOf("two");
        for (int i = 2; i < s.length(); i++) {
            pre[i] = Math.min(s.indexOf("one", i), s.indexOf("two", i));
        }
    }

    private static void suf() {
        suf[s.length() - 1] = s.lastIndexOf("one");
        suf[s.length() - 2] = s.lastIndexOf("two");
        for (int i = s.length() - 3; i >= 0; i--) {
            suf[i] = Math.max(s.lastIndexOf("one", i), s.lastIndexOf("two", i));
        }
    }

    private static int solve() {
        int res = Integer.MAX_VALUE, r = 0;
        for (int i = 0; i < s.length(); i++) {
            int j = pre[i], k = suf[i];
            if (j > k) continue;
            if (j != -1) r = Math.max(r, j + 2);
            if (k != -1) r = Math.max(r, s.length() - k);
        }
        if (r == 0) return 0;
        ans = new int[r];
        int idx = 0;
        for (int i = 0; i < s.length(); i++) {
            int j = pre[i], k = suf[i];
            if (j > k) continue;
            if (j != -1 && s.length() - j >= r) {
                ans[idx++] = j + 1;
                ans[idx++] = j + 2;
            }
            if (k != -1 && k - j + 1 >= r) {
                ans[idx++] = k + 1;
            }
        }
        res = idx;
        return res;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            s = sc.next();
            pre = new int[s.length()];
            Arrays.fill(pre, -1);
            suf = new int[s.length()];
            Arrays.fill(suf, -1);
            pre();
            suf();
            int r = solve();
            System.out.println(r);
            for (int i = 0; i < r; i++) {
                System.out.print(ans[i] + " ");
            }
            System.out.println();
        }
    }
}
