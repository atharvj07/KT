import java.util.Scanner;

public class Main {
    private static final long INF = Long.MAX_VALUE;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextLong();
            long[][] roses = new long[(int) n][3];
            long minX = INF, maxX = -INF, minY = INF, maxY = -INF, minZ = INF, maxZ = -INF;
            for (int i = 0; i < n; i++) {
                roses[i][0] = sc.nextLong();
                minX = Math.min(minX, roses[i][0]);
                maxX = Math.max(maxX, roses[i][0]);
                roses[i][1] = sc.nextLong();
                minY = Math.min(minY, roses[i][1]);
                maxY = Math.max(maxY, roses[i][1]);
                roses[i][2] = sc.nextLong();
                minZ = Math.min(minZ, roses[i][2]);
                maxZ = Math.max(maxZ, roses[i][2]);
            }
            long maxDist = INF;
            long bestX = 0, bestY = 0, bestZ = 0;
            for (long i = minX; i <= maxX; i++) {
                for (long j = minY; j <= maxY; j++) {
                    for (long k = minZ; k <= maxZ; k++) {
                        long maxCurr = 0;
                        for (int l = 0; l < n; l++) {
                            maxCurr = Math.max(maxCurr, Math.abs(roses[l][0] - i) + Math.abs(roses[l][1] - j) + Math.abs(roses[l][2] - k));
                        }
                        if (maxCurr < maxDist) {
                            maxDist = maxCurr;
                            bestX = i;
                            bestY = j;
                            bestZ = k;
                        }
                    }
                }
            }
            System.out.println(bestX + " " + bestY + " " + bestZ);
        }
    }
}
