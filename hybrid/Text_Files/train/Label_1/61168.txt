for _ in range(int(input())):
    n,q=map(int,input().split())
    a=list(map(int,input().split()))
    for _ in range(q):
        s=int(input())
        low,high=0,1<<30
        while low<=high:
            mid=(low+high)//2
            cnt1,cnt2=0,0
            for val in a:
                if val<=mid:cnt1+=1
                else:cnt2+=1
            sm=cnt1*(cnt1-1)//2+cnt2*(cnt2+1)//2
            if sm<s:low=mid+1
            else:high=mid-1
        print(['No','Yes'][s<=high])
