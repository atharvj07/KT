#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);

    int n;
    cin >> n;
    vector<int> h(n), b(n);
    for (int i = 0; i < n; i++) {
        cin >> h[i] >> b[i];
    }

    vector<pair<int, int>> buildings;
    for (int i = 0; i < n; i++) {
        buildings.push_back({h[i], b[i]});
    }

    sort(buildings.begin(), buildings.end());

    int max_beauty = 0;
    int cur_beauty = 0;
    int last_h = -1;
    for (auto building : buildings) {
        int h = building.first;
        int b = building.second;
        if (h == last_h) {
            cur_beauty = max(cur_beauty, b);
        } else {
            max_beauty += cur_beauty;
            cur_beauty = b;
            last_h = h;
        }
    }
    max_beauty += cur_beauty;

    cout << max_beauty << endl;

    return 0;
}
