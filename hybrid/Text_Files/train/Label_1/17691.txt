#include <cstdio>
#include <vector>
int main() {
  int n;
  scanf("%d", &n);
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  int ans = -1000000000;
  for (int i = 0; i < n; i++) {
    int sum = 0;
    for (int j = i; j < n; j++) {
      sum += a[j];
      if (sum % 2 == 1) ans = std::max(ans, sum);
    }
  }
  printf("%d", ans);
  return 0;
}
