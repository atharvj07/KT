n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]
r1, c1 = map(int, input().split())
r2, c2 = map(int, input().split())
vis = [[False] * m for _ in range(n)]
q = [(r1, c1)]
while q:
    r, c = q.pop(0)
    if r == r2 and c == c2:
        print('YES')
        exit()
    if vis[r - 1][c] == False and grid[r - 1][c] == '.' and 1 <= r - 1 <= n:
        q.append((r - 1, c))
        vis[r - 1][c] = True
    if vis[r][c - 1] == False and grid[r][c - 1] == '.' and 1 <= c - 1 <= m:
        q.append((r, c - 1))
        vis[r][c - 1] = True
    if vis[r + 1][c] == False and grid[r + 1][c] == '.' and 1 <= r + 1 <= n:
        q.append((r + 1, c))
        vis[r + 1][c] = True
    if vis[r][c + 1] == False and grid[r][c + 1] == '.' and 1 <= c + 1 <= m:
        q.append((r, c + 1))
        vis[r][c + 1] = True
print('NO')
