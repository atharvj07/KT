#include <iostream>
using namespace std;
typedef long long ll;
int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n;
        cin >> n;
        ll ans = 0;
        for (ll i = 1; i * i <= n; i++) {
            for (ll j = i; j <= n; j++) {
                ll c = i * i - j;
                if (c > 0 && c <= n && c * c + j * j == i * i) ans++;
            }
        }
        cout << ans << "\n";
    }
    return 0;
}
