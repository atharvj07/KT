from collections import Counter

def solve():
    n, m = map(int, input().split())
    arr = list(map(int, input().split()))

    # Count occurrences of each number
    cnt = Counter(arr)

    # Check if it's possible to make the number of even and odd equal
    if (sum(cnt[x] for x in cnt if x % 2 == 0) - sum(cnt[x] for x in cnt if x % 2 == 1)) % 2 != 0:
        return -1

    # Sort the card numbers in ascending order
    arr.sort()

    # Find the cards that need to be exchanged
    exchanges = []
    odd_cnt = 0
    even_cnt = 0
    for x in arr:
        if x % 2 == 1:
            if odd_cnt < n // 2:
                odd_cnt += 1
            else:
                exchanges.append(x)
        else:
            if even_cnt < n // 2:
                even_cnt += 1
            else:
                exchanges.append(x)

    # Exchange the cards
    for i, x in enumerate(exchanges):
        arr[i] = m - x + 1

    # Print the result
    print(len(exchanges))
    print(*arr)

solve()
