import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int d = sc.nextInt();
        if (d > k || k > n) {
            System.out.println(-1);
            return;
        }
        for (int i = 0; i < d; i++) {
            for (int j = 0; j < n; j++) {
                System.out.print((j % k) + 1 + " ");
            }
            System.out.println();
        }
    }
}
