#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 1, last = 0;
    for (int i = 1; i < n; i++) {
        if (a[i] <= a[last] * 2) {
            ans++;
            last = i;
        }
    }
    cout << ans << endl;
}
