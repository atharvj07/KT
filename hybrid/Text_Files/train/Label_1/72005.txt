import math
for t in range(int(input())):
    s, n = map(int, input().split())
    ma = 0
    lis = []
    for i in range(0, min(66, n)):
        if 2**i <= s:
            s -= 2**i
            n -= 1
            lis.append(str(2**i))
            ma = i
    if n > 0:
        lis.append(str(s))
    print(' '.join(lis))
