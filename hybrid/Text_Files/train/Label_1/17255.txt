#include <iostream>
using namespace std;

int main() {
  string n;
  cin >> n;
  bool contains1 = false, contains14 = false, contains144 = false;
  for (int i = 0; i < n.size(); i++) {
    if (n[i] == '1')
      contains1 = true;
    else if (n[i] == '4') {
      if (i + 1 < n.size() && n[i + 1] == '1')
        contains14 = true;
      if (i + 2 < n.size() && n[i + 1] == '1' && n[i + 2] == '4')
        contains144 = true;
    }
  }
  cout << (contains1 && contains14 && contains144 ? "YES" : "NO") << endl;
  return 0;
}
