t = int(input())
for _ in range(t):
    n, k, l = map(int, input().split())
    d = list(map(int, input().split()))
    p = [i for i in range(k)] + [i for i in range(k, 0, -1)]
    
    can_reach = True
    tide = 0
    for depth in d:
        max_safe_depth = l - depth
        if max_safe_depth < 0:
            can_reach = False
            break
        tide_incr = p.index(max_safe_depth) if max_safe_depth in p else k-1
        tide = min(tide + 1, tide_incr)
        if tide == k-1 and p[tide] > max_safe_depth:
            can_reach = False
            break
    print("Yes" if can_reach else "No")
