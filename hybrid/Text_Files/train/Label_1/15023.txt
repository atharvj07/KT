#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct Task {
    int d, c;
};

int main() {
    int n, a;
    cin >> n >> a;
    vector<Task> tasks(n);
    for (int i = 0; i < n; i++) {
        cin >> tasks[i].d >> tasks[i].c;
    }
    vector<long long> dp(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        dp[i] = max(dp[i - 1], (i > 1 ? dp[i - 2] : 0) + a - tasks[i - 1].c - (i > 1 ? (tasks[i - 1].d - tasks[i - 2].d) * (tasks[i - 1].d - tasks[i - 2].d) : 0));
    }
    cout << dp[n] << endl;
    return 0;
}
