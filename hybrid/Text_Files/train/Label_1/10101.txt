#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, m;
  cin >> n >> m;
  vector<int> f(n);
  for (int i = 0; i < n; i++) cin >> f[i];
  vector<int> b(m);
  for (int i = 0; i < m; i++) cin >> b[i];
  map<int, vector<int>> mf;
  for (int i = 0; i < m; i++) mf[b[i]].push_back(i);
  map<int, int> mcnt;
  for (auto& [_, v] : mf) { mcnt[v[0]]++; }
  bool amb = false;
  int cnt = 0, k;
  for (auto& [pos, v] : mf) {
    if (v.size() == 1) cnt++;
    else if (v.size() > 1) {
      if (!amb) k = pos;
      amb = true;
    }
  }
  if (mcnt[0] > 0 && cnt == m - mcnt[0]) {
    for (auto& [pos, v] : mf) {
      if (v.size() == 1) v[0] = 0;
    }
    cout << "Possible\n";
    for (auto& [_, v] : mf) {
      for (auto x : v) cout << x + 1 << " ";
    }
    return 0;
  }
  if (cnt == m && !amb) {
    cout << "Possible\n";
    for (auto& [_, v] : mf) {
      for (auto x : v) cout << x + 1 << " ";
    }
    return 0;
  }
  if (amb && cnt == m - 1 && mcnt[0] == 1) {
    int x = 0;
    for (auto& [_, v] : mf) {
      if (v.size() > 1) x = v[0];
    }
    mf[k].push_back(x);
    cout << "Possible\n";
    for (auto& [_, v] : mf) {
      for (auto x : v) cout << x + 1 << " ";
    }
    return 0;
  }
  cout << (amb ? "Ambiguity" : "Impossible") << '\n';
  return 0;
}
