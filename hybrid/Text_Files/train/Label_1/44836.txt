import java.util.*;

public class Main {

    private static class Castle {
        int capture, recruit, defend;
    }

    private static class Portal implements Comparable<Portal> {
        int from, to;

        @Override
        public int compareTo(Portal p) {
            return to - p.to;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        Castle[] castles = new Castle[n + 1];
        Portal[] portals = new Portal[m];
        for (int i = 1; i <= n; i++) {
            castles[i] = new Castle();
            castles[i].capture = sc.nextInt();
            castles[i].recruit = sc.nextInt();
            castles[i].defend = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            portals[i] = new Portal();
            portals[i].from = sc.nextInt();
            portals[i].to = sc.nextInt();
        }
        Arrays.sort(portals);
        int maxDefend = 0;
        boolean[] defend = new boolean[n + 1];
        int army = k;
        for (int i = 1; i <= n; i++) {
            if (army < castles[i].capture) {
                maxDefend = -1;
                break;
            }
            army -= castles[i].capture;
            army += castles[i].recruit;
            defend[i] = true;
            maxDefend += castles[i].defend;
            for (Portal p : portals) {
                if (p.from == i && !defend[p.to]) {
                    defend[p.to] = true;
                    maxDefend += castles[p.to].defend;
                }
            }
        }
        System.out.println(maxDefend);
    }
}
