#include <cstdio>
#include <unordered_map>
#include <vector>
using namespace std;

int main() {
  int n;
  scanf("%d", &n);
  unordered_map<int, int> mp;
  for (int i = 0, id; i < n; i++) {
    scanf("%d", &id);
    if (id == 0)
      continue;
    if (mp.find(id) != mp.end()) {
      printf("-1");
      return 0;
    }
    mp[id] = i + 1;
  }
  int ans = 0;
  for (auto it : mp) ans += mp.find(-it.first) != mp.end();
  printf("%d", ans / 2);
  return 0;
}
