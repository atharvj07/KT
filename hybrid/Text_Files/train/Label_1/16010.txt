#include<bits/stdc++.h>
using namespace std;
const int N = 1e5+9, mod = 1e9+7;
int n, k, l, a[N], f[N];

main() {
    cin >> n >> k >> l;
    for (int i = 1; i <= n*k; i++) { cin >> a[i]; }
    sort(a+1, a+1+n*k);
    for (int i = n+1; i <= n*k; i++) {
        int idx = upper_bound(a+1, a+i, a[i]-l) - a;
        idx--;
        if (idx >= n) { continue; }
        f[i] = max(f[i], f[idx]+a[i]);
    }
    cout << *max_element(f+1, f+1+n*k) << "\n";
}
