import heapq
def prim(g: dict, n: int, x: int, y: int) -> int:
    h = [[0, next(iter(g))]]
    vis = [False for _ in range(n+1)]
    ans = 0
    edges = 0
    while edges < n-1:
        cost, u = heapq.heappop(h)
        if vis[u]:
            continue
        vis[u] = True
        ans += cost
        edges += 1
        for v in g[u]:
            if not vis[v]:
                heapq.heappush(h, [y if (u, v) in g or (v, u) in g else x, v])
    return ans
n, x, y = map(int, input().split())
g = {}
for i in range(n):
    g[i+1] = set()
for i in range(n-1):
    u, v = map(int, input().split())
    g[u].add(v)
    g[v].add(u)
print(prim(g, n, x, y))
