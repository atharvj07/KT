def find_max_x(S):
    max_S = max(S)
    x = 0
    while (1 << x) <= max_S:
        x += 1
    for i in range(x, -1, -1):
        found = True
        for j in range(1, 1 << i):
            if (j ^ (j - 1)) not in S:
                found = False
                break
        if found:
            return i
    return 0

def generate_permutation(x):
    if x == 0:
        return [0]
    permutation = [0, 1]
    for i in range(1, x):
        n = len(permutation)
        for j in range(n):
            permutation.append(permutation[n-j-1] ^ (1 << i))
    return permutation

n = int(input().strip())
S = set(map(int, input().strip().split()))

x = find_max_x(S)
print(x)
print(*generate_permutation(x))
