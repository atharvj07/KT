import sys

m, k = map(int, input().split())
graph = {}
for i in range(m):
    a, b = map(int, input().split())
    if not a in graph:
        graph[a] = set()
    if not b in graph:
        graph[b] = set()
    graph[a].add(b)
    graph[b].add(a)

for i in sorted(graph.keys()):
    arr = {}
    for j in graph[i]:
        for t in graph[j]:
            if t != i and not t in graph[i]:
                if not t in arr:
                    arr[t] = 1
                else:
                    arr[t] += 1
    candidates = []
    for t in arr.keys():
        if arr[t] > k * (len(graph[i]) - 1) / 100:
            candidates.append(t)
    print(str(i) + ': ' + str(len(candidates)) + ' ' + ' '.join(map(str, candidates)))
