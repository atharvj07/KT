a1, b1 = map(int, input().split())
a2, b2 = map(int, input().split())
a3, b3 = map(int, input().split())
a4, b4 = map(int, input().split())

def win(a1, b1, a2, b2):
    return a1 > b2 and b1 > a2

# Permutation of arrange for Team 1
if win(a1, b1, a3, b3):
    print("Team 1")
elif win(a2, b2, a3, b3):
    print("Team 2")
elif win(a1, b1, a4, b4):
    print("Team 1")
elif win(a2, b2, a4, b4):
    print("Team 2")
else:
    print("Draw")
