#define INF (int)1e9
#define ll long long
#define mp make_pair
#define pii pair<int, int>
#define pb push_back
#define eps 1e-9

#include <iostream>
#include <vector>
#include <algorithm>
#include <cstdio>
#include <cstring>
#include <string>
#include <set>
#include <map>
#include <stack>
#include <queue>

using namespace std;

const int MAXN = 300010;

int n, a[MAXN];

int main() {
    cin >> n;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    vector<pii> ans;
    int st = 0, en = 0;
    while (st < n) {
        int pos = -1;
        for (int j = st; j < n; ++j) {
            if (a[j] == a[st]) {
                pos = j;
                break;
            }
        }
        if (pos == -1) break;
        en = pos;
        while (en < n && a[en] == a[st]) ++en;
        ans.pb(mp(st + 1, en));
        st = en;
    }
    if (ans.size() == 0) {
        cout << -1 << endl;
        return 0;
    }
    cout << ans.size() << endl;
    for (int i = 0; i < int(ans.size()); ++i) {
        cout << ans[i].first << " " << ans[i].second << endl;
    }
    return 0;
}
