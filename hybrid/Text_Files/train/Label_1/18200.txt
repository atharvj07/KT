#include<bits/stdc++.h>
#define mod 1000000007
using namespace std;
int main() {
    int k;
    cin >> k;
    long long ans[k + 1];
    ans[0] = 1;
    ans[1] = 6;
    for (int i = 2; i <= k; ++i) {
        ans[i] = ans[i - 1] * ans[i - 1] % mod * 3 % mod;
    }
    cout << ans[k] << endl;
    return 0;
}
