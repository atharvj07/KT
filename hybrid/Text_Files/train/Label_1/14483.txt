#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
#define mp make_pair
#define f first
#define s second
ll a[100005];
ll f[100005];
ll n;
int main()
{
    cin>>n;
    for(ll i=1;i<=n;i++)cin>>a[i];
    for(ll i=1;i<=n;i++)
    {
        if(f[a[i]]==0)
        {
            ll l=a[i]-1,r=a[i]+1;
            while(l>=1||r<=n)
            {
                if(l>=1)f[l]+=f[a[i]];
                if(r<=n)f[r]+=f[a[i]];
                l--;
                r++;
            }
        }
        f[a[i]]++;
    }
    ll ans=0;
    for(ll i=1;i<=n;i++)ans=max(ans,f[a[i]]);
    cout<<ans<<endl;
    return 0;
}
