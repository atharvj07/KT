import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int n = s.length();
        int ans = n;
        for (int i = 1; i <= n / 2; i++) {
            if (n % i == 0) {
                int cnt = n / i;
                String t = s.substring(0, i);
                int len = t.length();
                boolean ok = true;
                for (int j = i; j < n; j += i) {
                    if (!s.substring(j, j + len).equals(t)) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    ans = Math.min(ans, len + (int)Math.log10(cnt) + 1);
                }
            }
        }
        System.out.println(ans);
    }
}
