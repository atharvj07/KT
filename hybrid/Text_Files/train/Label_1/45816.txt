import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] freq = new int[26];
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            freq[i] = x;
        }
        int ans = 0;
        List<Character> necklace = new ArrayList<>();
        for (int i = 0; i < 26; i++) {
            while (freq[i] > 0) {
                if (necklace.isEmpty() || necklace.get(necklace.size() - 1) == (char) ('a' + i)) {
                    necklace.add((char) ('a' + i));
                    freq[i]--;
                    if (isPalindrome(necklace)) {
                        ans++;
                    }
                } else {
                    break;
                }
            }
        }
        System.out.println(ans);
        for (char c : necklace) {
            System.out.print(c);
        }
    }

    public static boolean isPalindrome(List<Character> necklace) {
        int n = necklace.size();
        for (int i = 0; i < n / 2; i++) {
            if (necklace.get(i) != necklace.get(n - i - 1)) {
                return false;
            }
        }
        return true;
    }
}
