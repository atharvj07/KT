for _ in range(int(input())):
    a, b = map(int, input().split())
    m = input()
    ans = 0
    i = 0
    while i < len(m):
        if m[i] == '1':
            cost = m.count('1', i, i + 4)
            if cost > 1:
                ans += min(cost * a, b + (cost - 1) * a)
            i += cost
        else:
            i += 1
    print(ans)
