#include<bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
    ios::sync_with_stdio(false), cin.tie(nullptr);
    ll t;
    cin >> t;
    while(t--) {
        ll n, m;
        cin >> n >> m;
        vector<ll> v(m);
        for(ll &x : v) cin >> x;
        ll ans = 0;
        for(ll i = 0; i < m; i++) {
            if(v[i] <= n) {
                while(v[i] * 2 <= n) {
                    v[i] *= 2;
                    ans++;
                }
                n -= v[i];
            }
        }
        if(n == 0) cout << ans << endl;
        else cout << "-1" << endl;
    }
}
