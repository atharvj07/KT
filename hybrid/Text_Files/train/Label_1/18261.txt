#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
void solve(){
	int n; cin>>n;
	vector<int> v(n);
	for(int &x: v) cin>>x;
	int ans=0;
	string s; cin>>s;
	for(char c: s){
		ans+=(c=='a'||c=='e'||c=='i'||c=='o'||c=='u'||c=='y');
	}
	if(v[0]==ans){
		for(int i=1; i<n; i++){
			cin>>s;
			int sum=0;
			for(char c: s){
				sum+=(c=='a'||c=='e'||c=='i'||c=='o'||c=='u'||c=='y');
			}
			if(sum!=v[i]){
				cout<<"NO"<<endl;
				return ;
			}
		}
		cout<<"YES"<<endl;
	}else{
		cout<<"NO"<<endl;
	}	
}
int main(){
	int t=1;
	while(t--) solve();
}
