import java.util.Scanner;

public class Ksenia {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        long[][] dp = new long[n + 1][k + 1];
        dp[0][0] = 1;

        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                // Add edges to the tree, connecting the current vertex (i) to other vertices.
                for (int e = 1; e <= 3 && i - e >= 0; e++) {
                    // Ensure that the maximum matching size is not exceeded.
                    if (j - (e == 1 ? 0 : 1) >= 0) {
                        dp[i][j] += dp[i - e][j - (e == 1 ? 0 : 1)] * (e == 1 ? i - 1 : i - 2);
                        dp[i][j] %= MOD;
                    }
                }
            }
        }

        System.out.println(dp[n][k]);
    }
}
