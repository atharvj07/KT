import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNext()) {
            int n = sc.nextInt();
            long cnt = 0;
            for (int i = 0; i <= 1000; i++) {
                for (int j = 0; j <= 1000; j++) {
                    if (i + j > n) {
                        break;
                    }
                    for (int k = 0; k <= 1000; k++) {
                        if (i + j + k > n) {
                            break;
                        }
                        int d = n - i - j - k;
                        if (d >= 0 && d <= 1000) {
                            cnt++;
                        }
                    }
                }
            }
            System.out.println(cnt);
        }
        sc.close();
    }
}
