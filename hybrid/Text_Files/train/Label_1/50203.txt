import java.util.Scanner;

public class CandyShop {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        long[] allowances = new long[n];
        for (int i = 0; i < n; i++) {
            allowances[i] = scanner.nextLong();
        }

        long maxCandies = 0;
        long currentSum = 0;
        int currentKid = 0;

        for (int i = 1; i <= m; i++) {
            currentSum += i;

            while (currentKid < n && currentSum > allowances[currentKid]) {
                currentSum -= i - 1;
                currentKid++;
            }

            if (currentKid == n) {
                break;
            }

            maxCandies = Math.max(maxCandies, currentSum);
        }

        System.out.println(maxCandies);
    }
}
