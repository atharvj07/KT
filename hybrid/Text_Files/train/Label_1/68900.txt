n, p = map(int, input().split())
s = list(input())

def next_tolerable_string(n, p, s):
    for i in range(n - 1, -1, -1):
        while ord(s[i]) < ord('a') + p:
            s[i] = chr(ord(s[i]) + 1)
            if (i >= 1 and s[i] == s[i - 1]) or (i >= 2 and s[i] == s[i - 2]):
                continue
            if i < n - 1:
                for j in range(i + 1, n):
                    for k in range(ord('a'), ord('a') + p):
                        s[j] = chr(k)
                        if (j >= 1 and s[j] == s[j - 1]) or (j >= 2 and s[j] == s[j - 2]):
                            continue
                        break
                    else:
                        break
                else:
                    return ''.join(s)
            else:
                return ''.join(s)
            break
        else:
            s[i] = 'a'
    return "NO"

print(next_tolerable_string(n, p, s))
