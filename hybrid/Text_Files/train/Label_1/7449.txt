#include <iostream>
#include <vector>
using namespace std;

int main() {
  vector<int> a(6);
  for (int &x : a) cin >> x;
  for (int x : a) if (x % 2 == 0) x = x / 2 - 1;
  else x = x / 2;
  int ans = a[0] * a[1] + a[1] * a[2] + a[2] * a[3] + a[3] * a[4] + a[4] * a[5] + a[5] * a[0];
  cout << ans << endl;
  return 0;
}
