#include <bits/stdc++.h>
#define int long long
using namespace std;
main() {
    int n, m, i, j, ans = 0, a[100005], b[100005];
    cin >> n >> m;
    for (i = 0; i < n; i++) { cin >> a[i]; }
    for (i = 0; i < m; i++) { cin >> b[i]; }
    i = 0, j = m - 1;
    sort(a, a + n);
    sort(b, b + m);
    while (a[i] < b[j]) { i++, j--, ans++; }
    cout << ans;
}
