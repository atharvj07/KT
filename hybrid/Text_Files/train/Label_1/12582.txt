#include<bits/stdc++.h>
using namespace std;

int gcd(int a, int b){
    if(b==0)
        return a;
    return gcd(b, a%b);
}

int lcm(int a, int b){
    return a * b / gcd(a, b);
}

int main(){
    int n;
    cin>>n;
    
    vector<int> a(n);
    for(int i=0;i<n;i++)
        cin>>a[i];
    
    int ans=1e9;
    int ind1=0, ind2=0;
    for(int i=0;i<n-1;i++){
        for(int j=i+1;j<n;j++){
            if(lcm(a[i], a[j])<ans){
                ans=lcm(a[i], a[j]);
                ind1=i+1;
                ind2=j+1;
            }
        }
    }
    cout<<ind1<<" "<<ind2<<endl;
    
    return 0;
}
