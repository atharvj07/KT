import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))

a.sort()

res = a[-1] - a[0]
k = 1
for i in range(1, n):
    if a[i] - a[i - 1] > res:
        res = a[i] - a[i - 1]
        k = 1
    elif a[i] - a[i - 1] == res:
        k += 1

print(res, k)
t = [k] * n
for i in range(n - 1, -1, -1):
    if i - 2 >= 0 and a[i] - a[i - 2] == res:
        t[i] = t[i - 1] - 1
    elif i - 1 >= 0 and a[i] - a[i - 1] == res:
        t[i] = t[i - 1]
print(*t)
