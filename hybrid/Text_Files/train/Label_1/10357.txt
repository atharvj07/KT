#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  long long time, messages, last_send_time = 0, max_queue = 0, queue = 0;
  while (n--) {
    cin >> time >> messages;
    long long gap = time - last_send_time;
    queue = max(0LL, queue - gap);
    queue += messages;
    max_queue = max(max_queue, queue);
    last_send_time = time + 1;
  }
  cout << last_send_time << ' ' << max_queue << endl;
  return 0;
}
