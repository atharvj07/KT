#include <iostream>
#include <cstring>
#include <vector>
using namespace std;

string s;
vector<int> a[26];
int n;

int main() {
  cin >> s;
  n = int(s.size());
  for (int i = 0; i < n; ++i)
    a[s[i] - 'a'].push_back(i);
  int ans = 0;
  while (1) {
    int p = 0;
    bool ok = 0;
    for (int i = 0; i < 26; ++i) {
      int m = a[i].size();
      for (int j = 0; j < m; ++j) {
        if (p != 0 && j != 0 && a[i][j] - a[i][j - 1] > 1) {
          ok = 1;
          break;
        }
        if (j != m - 1 && a[i][j + 1] - a[i][j] > 1) {
          ok = 1;
          break;
        }
        p = a[i][j] + 1;
      }
    }
    if (!ok)
      break;
    ++ans;
    for (int i = 0; i < 26; ++i) {
      vector<int> t;
      int m = a[i].size();
      for (int j = 0; j < m; ++j) {
        if (j == 0 || a[i][j] - a[i][j - 1] <= 1)
          t.push_back(a[i][j]);
        else
          --j;
      }
      swap(a[i], t);
    }
  }
  cout << ans << endl;
  return 0;
}
