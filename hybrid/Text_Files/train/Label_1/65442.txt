from array import array
import sys

def solve(n: int, x: int, d: array, h: array):
    ans = 10**18
    for i in range(1<<n):
        cur = x
        cnt = 0
        for j in range(n):
            if i & (1<<j):
                cur -= d[j]
                if cur <= 0:
                    cur = 0
                    cnt += 1
                else: cur += h[j]
        if cur <= 0: ans = min(ans, cnt)
    return ans

t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    d = array('L', [])
    h = array('L', [])
    for i in range(n):
        di, hi = map(int, input().split())
        d.append(di)
        h.append(hi)
    ans = solve(n, x, d, h)
    print(-1 if ans == 10**18 else ans)
