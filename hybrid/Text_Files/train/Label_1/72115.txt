l = int(input())
commands = [input() for _ in range(l)]

stack = []
x = 0
current_multiplier = 1

for command in commands:
    if command.startswith("for"):
        n = int(command.split()[1])
        stack.append((current_multiplier, n))
        current_multiplier *= n
        if current_multiplier > 2**32 - 1:
            break
    elif command == "end":
        prev_multiplier, n = stack.pop()
        current_multiplier = prev_multiplier
    else:  # "add" command
        x += current_multiplier
        if x > 2**32 - 1:
            break

if x > 2**32 - 1:
    print("OVERFLOW!!!")
else:
    print(x)
