#include <bits/stdc++.h>
using namespace std;

int main() {
    int m;
    cin >> m;
    set<int> s;
    for (int i = 0; i < m; i++) {
        int x;
        cin >> x;
        s.insert(x);
    }
    if (m == 1) {
        cout << -1 << "\n";
        return 0;
    }
    vector<int> ans;
    int cur = *s.begin();
    ans.push_back(cur);
    s.erase(s.begin());
    while (!s.empty()) {
        bool found = false;
        for (auto it = s.begin(); it != s.end(); it++) {
            if (cur % *it == 0) {
                cur = *it;
                ans.push_back(cur);
                s.erase(it);
                found = true;
                break;
            }
        }
        if (!found) {
            cout << -1 << "\n";
            return 0;
        }
    }
    cout << ans.size() << "\n";
    for (auto x : ans) cout << x << " ";
    cout << "\n";
    return 0;
}
