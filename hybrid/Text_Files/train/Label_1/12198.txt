#include<iostream>
#include<algorithm>
using namespace std;
int main(){
    int n,a[100005],b[100005];
    cin>>n;
    long long sum1=0,sum2=0,ans=0;
    for(int i=0;i<n;i++){
        cin>>a[i]>>b[i];
        sum1+=a[i];
        sum2+=b[i];
    }
    sort(a,a+n);
    sort(b,b+n);
    for(int i=0;i<n;i++){
        ans+=a[i]*(sum1-a[i]) + b[i]*(sum2-b[i]);
        sum1-=a[i];
        sum2-=b[i];
    }
    cout<<ans<<endl;
}
