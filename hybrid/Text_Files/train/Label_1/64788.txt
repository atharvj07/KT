from collections import defaultdict
import sys

def ccw(p1, p2, p3):
  return (p2[0]-p1[0]) * (p3[1]-p1[1]) - (p3[0]-p1[0]) * (p2[1]-p1[1])

def graham(s):
  n = len(s)
  p = sorted(s, key=lambda x: (x[0], x[1]))
  st = []

  for p_ in p:
    while len(st) >= 2 and ccw(st[-2], st[-1], p_) < 0:
      st.pop()

    st.append(p_)

  for p_ in p[::-1]:
    while len(st) >= 2 and ccw(st[-2], st[-1], p_) < 0:
      st.pop()
    st.append(p_)

  st.pop()
  return st[1:]

def check():
  for x1, y1 in P:
    ans = True
    for x2, y2 in S:
      if x1 == x2 and y1 == y2:
        continue
      for (x3, y3), (x4, y4) in zip(hull, hull[1:]+hull[:1]):
        if not ((x3 <= x1 <= x4) or (x4 <= x1 <= x3)) and ccw((x1,y1), (x2,y2), (x4,y4)) * ccw((x1,y1), (x2,y2), (x3,y3)) > 0:
          ans = False
          break
      if not ans:
        break
    if ans:
      return True

  return False

R, B = map(int, input().split())
P = [list(map(int, input().split())) for _ in range(R)]
S = [list(map(int, input().split())) for _ in range(B)]
hull = graham(P)
print('Yes' if check() else 'No')
