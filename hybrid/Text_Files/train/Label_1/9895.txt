#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n, t, last_time = 0, ans = 0;
  std::vector<int> times;
  std::cin >> n;
  times.reserve(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> t;
    times.push_back(t);
  }
  std::sort(times.begin(), times.end());
  std::cin >> t;
  for (int time : times) {
    if (time > last_time + t) break;
    ++last_time;
    ++ans;
  }
  std::cout << ans << '\n';
  return 0;
}
