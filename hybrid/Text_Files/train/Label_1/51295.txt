import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] left = new int[n];
        int[] right = new int[n];
        int[] cnt = new int[n + 1];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            cnt[a[i]]++;
            if (i > 0) {
                left[i] = left[i - 1];
                if (cnt[a[i]] == 2) {
                    left[i] = Math.max(left[i], i - right[i - 1]);
                }
            }
        }
        Arrays.fill(cnt, 0);
        for (int i = n - 1; i >= 0; i--) {
            cnt[a[i]]++;
            if (i < n - 1) {
                right[i] = right[i + 1];
                if (cnt[a[i]] == 2) {
                    right[i] = Math.max(right[i], left[i + 1] - i);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            ans = Math.max(ans, right[i] + left[i] + 1);
        }
        System.out.println(ans);
    }
}
