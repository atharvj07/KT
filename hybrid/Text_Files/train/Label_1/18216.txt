#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 1e3 + 5;
char s[maxn];
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        cin >> s + 1;
        bool ok = 1;
        for (int i = 1; i <= n / 2; i++) {
            ok &= (s[i] == s[n - i + 1]);
        }
        if (ok == 1)
            cout << "BOB\n";
        else
            cout << "ALICE\n";
    }
    return 0;
}
