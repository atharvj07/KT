import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        boolean neg = false;
        for (int i = 0; i < n; i++) {
            if (a[i] < 0) {
                neg = true;
            }
        }
        if (neg) {
            int max = Integer.MIN_VALUE;
            for (int i = 0; i < n; i++) {
                if (a[i] > max) {
                    max = a[i];
                }
            }
            for (int i = 0; i < n; i++) {
                if (a[i] > 0) {
                    System.out.print(a[i] + " ");
                }
            }
            for (int i = 0; i < n; i++) {
                if (a[i] == max) {
                    System.out.print(a[i] + " ");
                }
            }
        } else {
            int min = Integer.MAX_VALUE;
            for (int i = 0; i < n; i++) {
                if (a[i] < min) {
                    min = a[i];
                }
            }
            for (int i = 0; i < n; i++) {
                if (a[i] == min) {
                    System.out.print(a[i] + " ");
                }
            }
        }
        System.out.println();
    }
}
