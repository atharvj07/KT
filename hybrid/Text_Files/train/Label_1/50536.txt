import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int t = sc.nextInt();
		while (t-- > 0) {
			int n = sc.nextInt();
			int m = sc.nextInt();
			int x = sc.nextInt();
			int y = sc.nextInt();
			char[][] grid = new char[n][m];
			for (int i = 0; i < n; i++) {
				String line = sc.next();
				grid[i] = line.toCharArray();
			}
			long ans = 0;
			for (int i = 0; i < n; i++) {
				for (int j = 0; j < m; j++) {
					if (grid[i][j] == '.') {
						if (j + 1 < m && grid[i][j + 1] == '.') {
							ans += y;
							j++;
						} else {
							ans += x;
						}
					}
				}
			}
			System.out.println(ans);
		}
		sc.close();
	}
}
