import java.util.*;

public class Main {
    static int[][] grid;
    static int[][] cnt;
    static int w, h, n;
    static int[] a;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        h = sc.nextInt();
        w = sc.nextInt();
        n = sc.nextInt();
        grid = new int[h][w];
        cnt = new int[101][101];
        a = new int[n]; for (int i=0; i<n; i++) a[i] = sc.nextInt();
        dfs(1, 0, 0);
        for (int i=0; i<h; i++){
            for (int j=0; j<w; j++) System.out.print(grid[i][j]+" ");
            System.out.println();
        }
    }

    static void dfs(int id, int x, int y) {
        if (x >= h) { x = 0; y++; }
        if (y >= w) return;
        for (int i=1; i<=n; i++) {
            int nx = x+h, ny = y+w;
            if (cnt[i][a[i-1]]<a[i-1]&&grid[x][y]==0&&(x==0||grid[x-1][y]==i)&&(y==0||grid[x][y-1]==i)){
                grid[x][y] = i;
                cnt[i][a[i-1]]++;
                dfs(id, nx, ny);
                return;
            }
        }
        dfs(id, x+1, y);
    }
}
