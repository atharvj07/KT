#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
char s[N];
int n, freq[26];
ll ans;
int main() {
    cin >> n >> (s + 1);
    for (int i = 1; i <= n; i++) { freq[s[i] - 'a']++; }
    ans = 1ll * n * (n - 1);
    int dif = 0;
    for (int i = 0; i < 26; i++) {
        if (freq[i] > 0) {
            if (dif == 0) {
                dif = i;
            } else if (dif != i) {
                ans -= 1ll * freq[i] * freq[dif] * 2;
            }
        }
    }
    cout << (ans + 998244353) % 998244353;
    return 0;
}
