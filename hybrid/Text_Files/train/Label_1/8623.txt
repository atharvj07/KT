#include <iostream>
#include <vector>
using namespace std;
int main() {
  long long int H, n, ans = -1;
  cin >> H >> n;
  vector<long long int> d(n);
  for (auto &t : d) { cin >> t; }
  long long int HP = H;
  for (long long int i = 0; i < n; i++) {
    HP += d[i];
    if (HP <= 0) {
      ans = i + 1;
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
