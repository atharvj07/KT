#include <iostream>
#include <vector>
#include <cmath>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<double> a(n);
    double sum = 0;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        sum += a[i];
    }
    for (int i = 0; i < n; i++) {
        a[i] -= sum / n;
        if (abs(a[i]) < 1) a[i] = 0;
        cout << round(a[i]) << endl;
    }
    return 0;
}
