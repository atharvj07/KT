#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=15;
int a[N],b,L,ans;

int main(){
    for(int i=1;i<=14;++i)cin>>a[i];
    for(int i=1;i<=14;++i){
        if(a[i]){
            b=a[i];a[i]=0;
            L=i;ans=0;
            while(b--){
                L=(L+1)%15;
                a[L]++;
                if(a[L]==2){
                    a[L]=0;
                    ans+=2;
                }
            }
        }
    }
    cout<<ans<<'\n';
    return 0;
}
