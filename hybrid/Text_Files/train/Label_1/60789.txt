import sys
input = sys.stdin.readline

n,m=map(int,input().split())
a=[]
for i in range(n):
    a.append([int(c) for c in input().strip()])

ans=0
for i in range(1,n-1):
    for j in range(m-2):
        if ((a[i-1][j]+a[i-1][j+1]+a[i][j]+a[i][j+1]+a[i][j+2]+a[i+1][j]+a[i+1][j+1]+a[i+1][j+2])%2==0):
            ans+=1
            a[i][j+1]=1-a[i][j+1]

print(ans)
