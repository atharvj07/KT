import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] f = new int[n];
        int[] ans = new int[n];
        int max = 0;
        int cur = 0;
        int l = 0;
        int r = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                cur++;
                if (cur > max) {
                    max = cur;
                    l = i - cur + 1;
                    r = i;
                }
            } else {
                cur = 0;
            }
            f[i] = cur;
        }
        for (int i = l; i <= r; i++) {
            ans[i] = 1;
        }
        for (int i = 0; i < l; i++) {
            if (k > 0 && a[i] == 0 && f[i] + f[l - 1] + 1 > max) {
                ans[i] = 1;
                k--;
            }
        }
        for (int i = r + 1; i < n; i++) {
            if (k > 0 && a[i] == 0 && f[i] + f[r + 1] - 1 > max) {
                ans[i] = 1;
                k--;
            }
        }
        System.out.println(max + k);
        for (int i : ans) {
            System.out.print(i + " ");
        }
    }
}
