#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, s;
    cin >> n >> s;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int sum = 0, ans = -1;
    for (int i = 0; i < n; i++) {
      if (sum + a[i] <= s) {
        sum += a[i];
      } else {
        ans = i;
        break;
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
