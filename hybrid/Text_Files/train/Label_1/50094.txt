import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Stack;
import java.util.StringTokenizer;

public class Main {

    private static List<Integer> stack = new LinkedList<>();
    private static Queue<Integer> queue = new LinkedList<>();
    private static List<Integer> deck = new LinkedList<>();

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());

        for (int i = 0; i < n; i++) {
            int number = Integer.parseInt(br.readLine());
            if (number > 0) {
                add(number);
            } else {
                extract();
            }
        }
    }

    private static void add(int number) {
        int max = 0;
        String container = "";
        if (!stack.isEmpty() && stack.get(stack.size() - 1) > max) {
            max = stack.get(stack.size() - 1);
            container = "pushStack";
        }
        if (!queue.isEmpty() && queue.peek() > max) {
            max = queue.peek();
            container = "pushQueue";
        }
        if (!deck.isEmpty() && deck.get(0) > max) {
            max = deck.get(0);
            container = "pushFront";
        }
        if (!deck.isEmpty() && deck.get(deck.size() - 1) > max) {
            max = deck.get(deck.size() - 1);
            container = "pushBack";
        }

        switch (container) {
            case "pushStack":
                stack.add(number);
                System.out.println("pushStack");
                break;
            case "pushQueue":
                queue.add(number);
                System.out.println("pushQueue");
                break;
            case "pushFront":
                deck.add(0, number);
                System.out.println("pushFront");
                break;
            case "pushBack":
                deck.add(number);
                System.out.println("pushBack");
                break;
        }
    }

    private static void extract() {
        int k = 0;
        List<String> operations = new ArrayList<>();

        if (!stack.isEmpty()) {
            k++;
            operations.add("popStack");
        }
        if (!queue.isEmpty()) {
            k++;
            operations.add("popQueue");
        }
        if (!deck.isEmpty()) {
            k++;
            operations.add("popFront");
            operations.add("popBack");
        }

        System.out.println(k + " " + String.join(" ", operations));

        stack.clear();
        queue.clear();
        deck.clear();
    }
}
