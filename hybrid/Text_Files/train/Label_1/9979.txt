#include <iostream>
using namespace std;
int main() {
    int x, h, m;
    cin >> x >> h >> m;
    int ans = 0;
    int me = h * 60 + m;
    for (int i = 1; i <= 24 * 60 / x + 1; i++) {
        if (i * x >= me) {
            int t = i * x;
            if (to_string(t).find("7") != string::npos) {
                ans = i - 1;
                break;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
