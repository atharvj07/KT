#include <iostream>
#include <algorithm>
using namespace std;
pair<int, int> a[100];
int n;
int main() {
    cin >> n;
    for(int i = 1; i <= n; i++) cin >> a[i].first, a[i].second = i;
    sort(a + 1, a + 1 + n);
    for(int i = 1, j; i <= n; i++) {
        j = i;
        while(a[j].second != i) j++;
        if(i == j) continue;
        cout << i << " " << j << "\n";
        swap(a[i], a[j]);
    }
    return 0;
}
