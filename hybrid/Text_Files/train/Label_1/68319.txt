from typing import List

def can_be_made_equal(arr: List[int]) -> bool:
    arr.sort()
    return arr[0] == arr[len(arr)-1]

def get_maximum_k(arr: List[int]) -> int:
    max_diff = 0
    for i in range(len(arr)):
        if max_diff < arr[i]:
            max_diff = arr[i]
    
    min_diff = 1000000000
    for i in range(1, len(arr)):
        if min_diff > abs(arr[i] - arr[i-1]):
            min_diff = abs(arr[i] - arr[i-1])

    if min_diff == 0:
        return -1
    
    return max_diff//min_diff

for i in range(int(input())):
    n = int(input())
    arr = [int(x) for x in input().split()]

    if can_be_made_equal(arr):
        print(get_maximum_k(arr))
    else:
        print(-1)
