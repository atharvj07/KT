import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] sticks = new int[12];
        for (int i = 0; i < 12; i++) {
            sticks[i] = sc.nextInt();
        }
        Arrays.sort(sticks);
        int[] sides = {sticks[0] + sticks[1] + sticks[2], sticks[3] + sticks[4] + sticks[5], sticks[6] + sticks[7] + sticks[8]};
        Arrays.sort(sides);
        System.out.println(Arrays.equals(sides, new int[]{sticks[9] + sticks[10] + sticks[11], sticks[0] + sticks[2] + sticks[3], sticks[0] + sticks[4] + sticks[5]}) ? "yes" : "no");
    }
}
