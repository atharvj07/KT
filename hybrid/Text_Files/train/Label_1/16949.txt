#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, k;
    cin >> n >> m >> k;
    vector<int> x(n);
    vector<int> y(m);
    for (int i = 0; i < n; i++) { cin >> x[i]; }
    for (int j = 0; j < m; j++) { cin >> y[j]; }
    vector<pair<int, int>> p(k);
    for (int i = 0; i < k; i++) { cin >> p[i].first >> p[i].second; }
    long long ans = 0;
    for (int i = 0; i < k; i++) {
      for (int j = i + 1; j < k; j++) {
        long long dist1 = abs(p[i].first - p[j].first);
        long long dist2 = abs(p[i].second - p[j].second);
        long long dist3 = dist1 + dist2;
        long long dist4;
        if (p[i].first == p[j].first) dist4 = 0;
        else
          dist4 = abs(upper_bound(x.begin(), x.end(), p[i].first) -
                      lower_bound(x.begin(), x.end(), p[j].first)) +
                 abs(upper_bound(y.begin(), y.end(), p[i].second) -
                      lower_bound(y.begin(), y.end(), p[j].second));
        if (p[i].second == p[j].second) dist4 = 0;
        else
          dist4 = abs(upper_bound(x.begin(), x.end(), p[i].first) -
                      lower_bound(x.begin(), x.end(), p[j].first)) +
                 abs(upper_bound(y.begin(), y.end(), p[i].second) -
                      lower_bound(y.begin(), y.end(), p[j].second));
        if (dist4 > dist3) ans++;
      }
    }
    cout << ans << "\n";
  }
}
