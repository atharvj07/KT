#include<iostream>
#include <vector>
#include <algorithm>
#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
 
typedef long long ll;
 
int main() {
    ll n, k;
    cin >> n >> k;
    vector<ll> a(n);
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    sort(a.begin(), a.end());
    ll l = 0, r = a[n - 1] - a[0], ans = LLONG_MAX;
    while (l <= r) {
        ll mid = (l + r) >> 1;
        ll cnt = 0, now = a[0] + mid;
        for (ll i = 0; i < n; ++i) {
            if (a[i] > now || i == n - 1) {
                now = a[i] + mid;
                ++cnt;
            }
        }
        if (cnt >= k) {
            ans = mid;
            r = mid - 1;
        } else
            l = mid + 1;
    }
    cout << ans << endl;
    return 0;
}
