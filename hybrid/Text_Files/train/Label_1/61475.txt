from queue import Queue
n,m = map(int,input().split())
adj = [[] for i in range(n+1)]
for i in range(m):
 adj[int(input().split()[0])].append(int(input().split()[1]))
 adj[int(input().split()[1])].append(int(input().split()[0]))
vis1 = [0]*(n+1)
vis2 = [0]*(n+1)
def bfs(s,vis,l):
 dis = [0]*(n+1)
 dis[s]=1
 q = Queue()
 q.put(s)
 while not q.empty():
  u = q.get()
  for v in adj[u]:
   if vis[v]==0 and dis[v]<=l:
    dis[v]=dis[u]+1
    q.put(v)
  vis[u]=1
s1,t1,l1 = map(int,input().split())
s2,t2,l2 = map(int,input().split())
bfs(s1,vis1,l1)
bfs(s2,vis2,l2)
if vis1[t1]==1 and vis2[t2]==1:
    print(0)
else:
    print(-1)
