#include <iostream>
#include <map>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n;
  cin >> n;
  map<char, int> count;
  for (char c : s) { count[c]++; }
  int sheets = 0;
  for (auto it = count.begin(); it != count.end(); it++) {
    sheets += (it->second + n - 1) / n;
  }
  if (sheets * n < s.size()) {
    cout << -1 << endl;
    return 0;
  }
  string sheet;
  for (auto it = count.begin(); it != count.end(); it++) {
    for (int i = 0; i < (it->second + n - 1) / n; i++) { sheet += it->first; }
  }
  sheet = sheet.substr(0, n);
  cout << sheets << endl;
  cout << sheet << endl;
  return 0;
}
