import java.util.Scanner;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[][] d = new long[n][m];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                d[i][j] = sc.nextInt();
        BigInteger ans = BigInteger.ZERO;
        for (int i = 0; i < n; i++) {
            BigInteger cnt = BigInteger.valueOf(1);
            for (int j = 0; j < m; j++)
                cnt = cnt.multiply(BigInteger.valueOf(d[i][j]));
            ans = ans.add(cnt);
        }
        System.out.println(ans.remainder(new BigInteger("998244353")));
    }
}
