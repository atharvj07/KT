n, m = map(int, input().split())
l = list(map(int, input().split()))

if sum(l) < n:
    print(-1)
else:
    positions = []
    current_pos = 1
    for i in range(m):
        if current_pos + l[i] - 1 > n:
            print(-1)
            break
        positions.append(current_pos)
        current_pos += 1
    else:
        # Adjust positions if necessary to ensure all cells are colored
        for i in range(m - 1, -1, -1):
            if positions[i] + l[i] - 1 < n:
                positions[i] = n - l[i] + 1
                n -= l[i]
            else:
                break
        print(*positions)
