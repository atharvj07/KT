#include <iostream>
#include <vector>
#include <tuple>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    int j = 0;
    for (int &x : a) {
        if (j == 0 || j == 1)
            j++;
        else if (j >= 2 && x != a[j - 2] + 2)
            break;
        j++;
    }
    cout << j << endl;
}
