#include <iostream>

int main()
{
    int n, m;
    std::cin >> n >> m;
    bool ok;
    int ans = 0;
    char c;
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            ok = 1;
            std::cin >> c;
            if (i + 1 < n && j + 1 < m)
            {
                if (c == 'f')
                    if (std::cin >> c, c == 'a')
                        if (std::cin >> c, c == 'c')
                            if (std::cin >> c, c == 'e')
                                ans++;
                std::cin >> c;
            }
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
