#include <bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin >> t;
    while(t--){
        int n, k;
        cin >> n >> k;
        vector<long long> v(n);
        for(long long &x : v) cin >> x;
        sort(v.begin(), v.end());
        long long ans = 0;
        int j = n - 1;
        for(int i = 0; i < k; i++){
            if(v[i] == v[j]){
                ans += v[j];
                v.pop_back();
                j--;
            }
            else{
                ans += v[j] - v[i];
                v[j]--;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
