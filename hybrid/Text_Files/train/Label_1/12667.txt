#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 5e6 + 10;
ll a[N];
int main() {
    ll n2, n3, n5, n6;
    cin >> n2 >> n3 >> n5 >> n6;
    ll ans = 0;
    while (n2 >= 3) {
        ans += 256;
        n2 -= 3;
    }
    if (n2 > 0) {
        ans += 32;
        n2 = 0;
    }
    ans += 32 * n3;
    ans += 16 * n5;
    ans += 8 * n6;
    cout << ans << endl;
    return 0;
}
