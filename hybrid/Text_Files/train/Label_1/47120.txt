import java.util.Scanner;

public class Main {
    static boolean[] open;
    static int[][] graph;
    static int n, m;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        open = new boolean[n + 1];
        graph = new int[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            open[a] = true;
            open[b] = true;
            graph[a][b] = 1;
            graph[b][a] = 1;
        }
        int[] path = bfs();
        if (path[n] == -1) {
            System.out.println(-1);
        } else {
            System.out.println(path[n] - 1);
            StringBuilder sb = new StringBuilder();
            int cur = n;
            while (cur != 1) {
                sb.append(cur + " ");
                cur = path[cur];
            }
            sb.append(1);
            System.out.println(sb);
        }
    }

    static int[] bfs() {
        int[] path = new int[n + 1];
        int[] dist = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            path[i] = -1;
            dist[i] = -1;
        }
        dist[1] = 0;
        int[] queue = new int[n + 1];
        int head = 0;
        int tail = 0;
        queue[tail++] = 1;
        while (head < tail) {
            int cur = queue[head++];
            for (int i = 1; i <= n; i++) {
                if (graph[cur][i] == 1 && (open[cur] && !open[i] || !open[cur] && open[i])) {
                    if (dist[i] == -1) {
                        dist[i] = dist[cur] + 1;
                        path[i] = cur;
                        queue[tail++] = i;
                    }
                }
            }
        }
        return path;
    }
}
