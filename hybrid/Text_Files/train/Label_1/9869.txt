#include <iostream>
#include <vector>

using namespace std;

bool check(vector<long long>& a) {
  long long n = a.size();
  for (long long i = 0; i < n; i++) {
    if (a[i] <= a[(i + 1) % n]) return false;
  }
  return true;
}

int main() {
  long long n;
  cin >> n;

  vector<long long> b(n);
  for (long long i = 0; i < n; i++) cin >> b[i];

  vector<long long> a(n);
  a[0] = 1;
  for (long long i = 1; i < n; i++) {
    a[i] = max(a[i - 1], b[i - 1] + 1);
  }

  if (check(a)) {
    cout << "YES" << endl;
    for (long long i = 0; i < n; i++) cout << a[i] << " ";
    cout << endl;
  } else {
    cout << "NO" << endl;
  }
}
