#include <bits/stdc++.h>
using namespace std;
typedef pair<int,int> pii;
pii p[4] = {{0,0},{0,1},{1,0},{1,1}};
const int dx[4] = {0,1,0,-1};
const int dy[4] = {1,0,-1,0};
int n, m, a[55][55], ans = 1e9, vis[55][55];
void dfs(int x, int y, int px, int py, int cnt) {
    if(x < 1 || x > n || y < 1 || y > m || vis[x][y]) return;
    vis[x][y] = 1;
    if(a[x][y]) {
        ans = min(ans, cnt);
        return;
    }
    for(int i = 0; i < 4; i++) {
        dfs(px+dx[i], py+dy[i], x, y, cnt+1);
        dfs(x, y, px+dx[i], py+dy[i], cnt+1);
    }
}
int main() {
    cin >> n >> m;
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) cin >> a[i][j];
    }
    for(int i = 2; i < n; i++) {
        for(int j = 2; j < m; j++) {
            if(a[i][j]) {
                for(int k = 0; k < 4; k++) {
                    int x = i - p[k].first, y = j - p[k].second;
                    if(!a[x][y]) {
                        memset(vis, 0, sizeof(vis));
                        vis[i][j] = 1;
                        dfs(x, y, i, j, 1);
                    }
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
