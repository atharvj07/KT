t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    total_blocks = sum(a)
    max_blocks = max(a)
    if total_blocks % (n - 1) == 0:
        possible_equal = total_blocks // (n - 1)
    else:
        possible_equal = (total_blocks // (n - 1)) + 1
    if max_blocks > possible_equal:
        print((max_blocks * (n - 1)) - total_blocks)
    else:
        print((possible_equal * (n - 1)) - total_blocks)
