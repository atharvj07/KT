#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAXN = 150005;

int n;
int a[MAXN];
ll ans, sum;

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    for (int i = 1; i + 1 <= n; i += 2) {
        if (a[i] > a[i + 1]) sum += a[i] - a[i + 1];
    }
    for (int i = 2; i <= n; i += 2) {
        if (a[i] < a[i - 1]) ans += min(sum, 1ll * a[i - 1] - a[i]);
    }
    cout << ans << endl;
    return 0;
}
