n, k = map(int, input().split())
requests = list(map(int, input().split()))

served = set()
queue = []
count = 0

for req in requests:
    if req not in served:
        if len(queue) >= k:
            removed = queue.pop(0)
            served.remove(removed)
        queue.append(req)
        served.add(req)
        count += 1

print(count)
