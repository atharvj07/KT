t = int(input())
for _ in range(t):
    n = int(input())
    commands = [list(map(int, input().split())) for _ in range(n)]
    commands.append([float('inf'), 0])  # Add a dummy command for simplicity
    
    successful = 0
    position = 0  # Current position of the robot
    target = 0  # Current target position of the robot
    startTime = 0  # When we started moving towards the current target
    for i in range(n):
        t_i, x_i = commands[i]
        t_next, x_next = commands[i + 1]
        
        # Calculate where the robot will be at time t_i
        if t_i >= startTime:
            moveTime = min(t_i - startTime, abs(target - position))
            if target >= position:
                position += moveTime
            else:
                position -= moveTime
        
        # Check if we need to update the target
        if position == target:
            target = x_i
            startTime = t_i
        
        # Check if the command is successful
        # Calculate where the robot will be at time t_next
        endTime = min(t_next, startTime + abs(target - position))
        if (position <= x_i <= target or position >= x_i >= target) and startTime <= t_i < endTime:
            successful += 1
    
    print(successful)
