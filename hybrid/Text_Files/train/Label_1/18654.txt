#include <bits/stdc++.h>
#include <iostream>
using namespace std;

int main() {
  string s;
  cin >> s;
  int lr = 0, ud = 0;
  for (char c : s) {
    switch (c) {
      case 'L':
        lr--;
        break;
      case 'R':
        lr++;
        break;
      case 'U':
        ud++;
        break;
      case 'D':
        ud--;
        break;
    }
  }
  if (lr == 0 && ud == 0)
    cout << 0;
  else if (lr + ud == 0 || lr - ud == 0)
    cout << 1;
  else
    cout << -1;
  return 0;
}
