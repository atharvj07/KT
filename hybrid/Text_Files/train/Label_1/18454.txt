#include <bits/stdc++.h>

using namespace std;
const int N = 1e5 + 5;

vector<int> G[N];
vector<int> order;

int main() {
    int n;
    cin >> n;
    vector<int> a(n + 1);
    cin >> a[1];
    for (int i = 2; i <= n; i++) {
        cin >> a[i];
        if (a[i] < a[i - 1]) G[a[i]].push_back(a[i - 1]), G[a[i - 1]].push_back(a[i]);
    }
    for (int i = 1; i <= n; i++) {
        if (a[i] == i) order.push_back(i);
    }
    for (int i = 1; i <= n; i++) {
        for (int j : G[i]) {
            if (binary_search(order.begin(), order.end(), j)) continue;
            order.push_back(i);
            order.push_back(j);
            break;
        }
    }
    cout << (int)order.size() << "\n";
    return 0;
}
