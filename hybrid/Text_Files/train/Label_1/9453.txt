#include<bits/stdc++.h>
using namespace std;
char a[502][502];
int b[502][502];
int dx[4]={1,-1,0,0};
int dy[4]={0,0,1,-1};
int n,m;
int main() {
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;i++) {
		scanf("%s",a[i]+1);
	}
	for(int i=1;i<=n;i++) {
		for(int j=1;j<=m;j++) {
			if(a[i][j]=='X') {
				for(int k=0;k<4;k++) {
					int x=i+dx[k],y=j+dy[k];
					if(x>0&&x<=n&&y>0&&y<=m&&!b[x][y]) {
						b[x][y]=1;
					}
				}
			}
		}
	}
	int x=1,y=1;
	while(x<=n&&a[x][y]!='X') {
		x++;
	}
	if(x>n) {
		printf("NO\n");
		return 0;
	}
	while(y<=m&&a[x][y]!='X') {
		y++;
	}
	if(y>m) {
		printf("NO\n");
		return 0;
	}
	if(b[x][y]) {
		printf("YES\n");
	} else {
		printf("NO\n");
	}
	return 0;
}
