def solve_tea_problem(n, k, a, b):
    if a < b:
        tea1, tea2, char1, char2 = b, a, 'B', 'G'
    else:
        tea1, tea2, char1, char2 = a, b, 'G', 'B'

    if tea1 > (tea2 + 1) * k:
        return "NO"

    result = []
    while tea1 > 0 or tea2 > 0:
        limit = min(k, tea1)
        result.append(char1 * limit)
        tea1 -= limit

        if tea2 > 0:
            result.append(char2)
            tea2 -= 1

    return ''.join(result)

n, k, a, b = map(int, input().split())
print(solve_tea_problem(n, k, a, b))
