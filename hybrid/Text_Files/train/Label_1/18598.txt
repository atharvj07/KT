#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> x(n), y(n);
    for (int i = 0; i < n; ++i) { cin >> x[i]; }
    for (int i = 0; i < n; ++i) { cin >> y[i]; }
    sort(x.begin(), x.end());
    sort(y.begin(), y.end());
    int ans = 0;
    while (y.empty() == false) {
      int low = 0, high = k - 1;
      while (low < high) {
        int mid = (low + high) / 2;
        if (y[0] + mid < y[y.size() - 1]) {
          low = mid + 1;
        } else {
          high = mid;
        }
      }
      int j = y.size() - 1;
      while (j >= 0 && y[j] >= y[0] + low) { --j; }
      ans += y.size() - j - 1;
      for (int i = 0; i < n; ++i) {
        if (x[i] >= x[0] + low && x[i] <= x[n - 1] - low) { --y[i]; }
      }
      vector<int> z;
      for (int i = 0; i < y.size(); ++i) {
        if (y[i] > y[0] + low) { z.push_back(y[i]); }
      }
      y = z;
    }
    cout << ans << endl;
  }
  return 0;
}
