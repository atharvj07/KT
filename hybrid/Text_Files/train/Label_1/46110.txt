import java.util.*;

public class Main {
    private static final int[][] PIECES = {{0, 1, 2, 3, 4}, {0, 1, 2, 5, 6}, {0, 3, 4, 5, 6}, {1, 2, 5, 8, 9},
            {1, 2, 5, 7, 8}, {1, 2, 4, 7, 8}, {1, 3, 4, 7, 9}, {1, 3, 4, 5, 9}, {2, 5, 7, 8, 9},
            {2, 4, 5, 7, 9}, {2, 3, 5, 7, 8}, {2, 3, 4, 5, 6}, {3, 4, 5, 8, 9}, {3, 4, 5, 6, 7},
            {3, 1, 5, 6, 7}, {3, 1, 5, 7, 8}, {4, 5, 8, 9, 10}, {4, 5, 7, 9, 10}, {4, 2, 5, 7, 9},
            {4, 2, 3, 5, 6}, {4, 1, 5, 7, 8}, {5, 7, 8, 9, 10}, {5, 7, 8, 3, 4}, {5, 2, 7, 8, 9},
            {5, 1, 2, 7, 8}, {5, 1, 3, 7, 9}};

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < n; j++) {
                board[i][j] = line.charAt(j);
            }
        }
        boolean[][] used = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.') {
                    if (canTile(board, used, i, j)) {
                        System.out.println("YES");
                        return;
                    }
                }
            }
        }
        System.out.println("NO");
    }

    private static boolean canTile(char[][] board, boolean[][] used, int x, int y) {
        if (x < 0 || x >= board.length || y < 0 || y >= board[0].length || used[x][y] || board[x][y] == '#') {
            return false;
        }
        used[x][y] = true;
        for (int[] piece : PIECES) {
            boolean valid = true;
            for (int i = 0; i < 5; i++) {
                int nx = x + piece[i] / board.length;
                int ny = y + piece[i] % board.length;
                valid &= canTile(board, used, nx, ny);
            }
            if (valid) {
                return true;
            }
        }
        used[x][y] = false;
        return false;
    }
}
