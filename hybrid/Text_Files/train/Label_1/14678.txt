#include <bits/stdc++.h>
using namespace std;

#define INF 1e9+7

int n, m, a[503][503], dp[503][503][2], mx = 0;
bool pos = 1;
struct Coor{
    int x, y;
};

bool cmp(Coor a, Coor b){
    if(a.x == b.x) return a.y < b.y;
    return a.x < b.x;
}

int func(int x, int y, int prv){
    if(dp[x][y][prv] != -1) return dp[x][y][prv];

    int &ans = dp[x][y][prv];
    ans = 0; mx = max(mx, a[x][y]);
    
    if(x+1 <= n && y+1 <= m && a[x][y] != 0 && a[x+1][y] != 0 && a[x][y+1] != 0){
        if(a[x][y] < a[x+1][y] && a[x][y] < a[x][y+1]){
            pos = 0;
            return ans = -INF;
        }
    }

    if(x-1 >= 1 && y+1 <= m && a[x][y] != 0 && a[x-1][y] != 0 && a[x][y+1] != 0){
        if(a[x][y] < a[x-1][y] && a[x][y] < a[x][y+1]){
            pos = 0;
            return ans = -INF;
        }
    }

    if(x+1 <= n && y-1 >= 1 && a[x][y] != 0 && a[x+1][y] != 0 && a[x][y-1] != 0){
        if(a[x][y] < a[x+1][y] && a[x][y] < a[x][y-1]){
            pos = 0;
            return ans = -INF;
        }
    }

    if(x-1 >= 1 && y-1 >= 1 && a[x][y] != 0 && a[x-1][y] != 0 && a[x][y-1] != 0){
        if(a[x][y] < a[x-1][y] && a[x][y] < a[x][y-1]){
            pos = 0;
            return ans = -INF;
        }
    }

    int ans1 = 0, ans2 = 0, ans3 = 0, ans4 = 0;
    if(x+1 <= n){
        if(prv == 0) ans1 = func(x+1, y, 0) + a[x+1][y];
        else if(a[x][y] < a[x+1][y]) ans1 = func(x+1, y, 0) + a[x+1][y];
        else{
            pos = 0;
            return ans = -INF;
        }
    }
    if(y+1 <= m){
        if(prv == 0) ans2 = func(x, y+1, 0) + a[x][y+1];
        else if(a[x][y] < a[x][y+1]) ans2 = func(x, y+1, 0) + a[x][y+1];
        else{
            pos = 0;
            return ans = -INF;
        }
    }
    if(x-1 >= 1){
        if(prv == 0) ans3 = func(x-1, y, 0) + a[x-1][y];
        else if(a[x][y] < a[x-1][y]) ans3 = func(x-1, y, 0) + a[x-1][y];
        else{
            pos = 0;
            return ans = -INF;
        }
    }
    if(y-1 >= 1){
        if(prv == 0) ans4 = func(x, y-1, 0) + a[x][y-1];
        else if(a[x][y] < a[x][y-1]) ans4 = func(x, y-1, 0) + a[x][y-1];
        else{
            pos = 0;
            return ans = -INF;
        }
    }

    return ans = max(ans, max({ans1, ans2, ans3, ans4}));
}

int main(){
    cin >> n >> m;
    for(int i = 1; i <= n; ++i){
        for(int j = 1; j <= m; ++j){
            cin >> a[i][j];
            dp[i][j][0] = dp[i][j][1] = -1;
        }
    }
    func(1, 1, 1);
    if(pos) cout << mx << "\n";
    else cout << -1 << "\n";
    return 0;
}
