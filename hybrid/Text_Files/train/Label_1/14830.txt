#include <bits/stdc++.h>
using namespace std;
const int MAXN = 200005;
typedef long long ll;

struct gasStation {
    ll pos, price;
};
bool operator<(const gasStation &a, const gasStation &b) { return a.price < b.price; }

gasStation stations[MAXN];
ll dis[MAXN];

int main() {
    ll d, n, m;
    cin >> d >> n >> m;
    for (ll i = 0; i < m; i++) {
        cin >> stations[i].pos >> stations[i].price;
    }
    sort(stations, stations + m);
    ll pos = 0, ans = 0, tank = n;
    for (ll i = 0; i < m; i++) {
        if (tank >= dis[i]) {
            tank -= dis[i];
            dis[i] = 0;
            ans += tank * stations[i].price;
            tank += n - (stations[i].pos - pos);
            tank = min(tank, n);
            pos = stations[i].pos;
        } else {
            ans = -1;
            break;
        }
    }
    if (tank < dis[m]) {
        ans = -1;
    }
    cout << ans << endl;
    return 0;
}
