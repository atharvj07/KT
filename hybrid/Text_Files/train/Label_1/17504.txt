#include <iostream>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  long long a[n], cur = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    cur += a[i];
    if (cur > m)
      cur = a[i], ans++;
  }
  for (int i = 0; i < n; i++) cout << (cur = cur - a[i] + a[i] > m) << endl;
  return 0;
}
