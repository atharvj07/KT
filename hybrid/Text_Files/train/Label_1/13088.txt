#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;

ll f[2][10000010];
ll g[2][10000010];

int main() {
  int a, b, n;
  cin >> a >> b >> n;
  f[0][0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j <= 1; j++) {
      for (int k = a + j; k <= b; k++) {
        f[j][i + 1] = (f[j][i + 1] + f[j][i]) % mod;
      }
    }
    for (int j = 0; j <= 1; j++) {
      for (int k = 0; k <= 9; k++) {
        g[j][i + 1] = (g[j][i + 1] + g[j][i]) % mod;
      }
    }
    for (int j = 0; j <= 1; j++) {
      for (int k = a; k <= b; k++) {
        g[j][i + 1] = (g[j][i + 1] + f[j][i]) % mod;
      }
    }
  }
  ll ans = f[0][n] - g[1][n];
  ans = (ans + mod) % mod;
  cout << ans << endl;
  return 0;
}
