#include <bits/stdc++.h>

using namespace std;

struct string_pref{
    int first;
    string str;
};

vector<string> vec;
vector<int> vec2;
int b[20][20];

int main(){
    int n,m;
    string s;
    int mn=INT_MAX;
    cin>>n>>m;
    for(int i=0;i<n;i++){
        cin>>s;
        vec.push_back(s);
    }
    for(int i=0;i<n;i++){
        cin>>s;
        vec2.push_back(stoi(s));
    }
    for(int p=0;p<m;p++){
        map<string, int> mp;
        vector<string_pref> v;
        for(int i=0;i<n;i++){
            string t = vec[i];
            if(mp.find(t.substr(0, p))==mp.end()){
                v.push_back(string_pref{i, t.substr(0, p)});
                mp[t.substr(0, p)] = i;
            }
            t.erase(p, 1);
            mp[t] = i;
        }
        int ans = INT_MAX;
        for(int i=0;i<v.size();i++){
            if(mp.find(v[i].str)!=mp.end()){
                ans=min(ans, vec2[v[i].first]+vec2[mp[v[i].str]]);
            }
            if(p+1 < m){
                for(int j=0;j<26;j++){
                    string t = v[i].str;
                    t.push_back(char(j+'a'));
                    if(mp.find(t)!=mp.end()){
                        ans = min(ans, vec2[v[i].first] + vec2[mp[t]]);
                    }
                }
            }
        }
        mn=min(mn, ans);
    }
    cout<<mn<<endl;
    return 0;
}
