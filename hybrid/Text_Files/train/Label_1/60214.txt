A = input().split('.')[1]
B = input().split('.')[1]
C = input().split('.')[1]
D = input().split('.')[1]
choices = [A, B, C, D]
min_len = min(len(choice) for choice in choices)
max_len = max(len(choice) for choice in choices)
if len([choice for choice in choices if len(choice) == min_len]) == 1:
    shortest = [choice for choice in choices if len(choice) == min_len][0]
    print('A' if choices.index(shortest) == 0 else 'B' if choices.index(shortest) == 1 else 'C' if choices.index(shortest) == 2 else 'D')
elif len([choice for choice in choices if len(choice) == max_len]) == 1:
    longest = [choice for choice in choices if len(choice) == max_len][0]
    print('A' if choices.index(longest) == 0 else 'B' if choices.index(longest) == 1 else 'C' if choices.index(longest) == 2 else 'D')
else:
    print('C')
