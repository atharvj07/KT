import java.util.Scanner;

public class RegularBracketSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        for (int i = 0; i < t; i++) {
            String s = scanner.nextLine();
            System.out.println(maxMoves(s));
        }
    }

    private static int maxMoves(String s) {
        int n = s.length();
        int open = 0;
        int close = 0;
        int maxMoves = 0;

        for (int i = 0; i < n; i++) {
            char c = s.charAt(i);
            if (c == '(') {
                open++;
            } else if (c == ')') {
                close++;
            } else if (c == '[') {
                open++;
            } else if (c == ']') {
                close++;
            }

            if (open == close) {
                maxMoves++;
                open = 0;
                close = 0;
            }
        }

        return maxMoves;
    }
}
