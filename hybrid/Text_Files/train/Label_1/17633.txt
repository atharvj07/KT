#include <bits/stdc++.h>
#define ll long long
#define rep(i, x, y) for(ll i = x; i <= y; i++)
#define rep_(i, x, y) for(ll i = x; i >= y; i--)
using namespace std;

int main() {
    ll T;
    cin >> T;
    while(T--) {
        ll n, l, r;
        cin >> n >> l >> r;
        ll len = l - 1;
        ll st = len / (n - 1), en = len % (n - 1);
        st++, en++;
        cout << st << " ";
        ll i = st;
        while(i <= n) {
            if(i != en) cout << i << " ";
            i++;
        }
        en--;
        i = en;
        while(i >= 1) {
            if(i != st) cout << i << " ";
            i--;
        }
        cout << st << endl;
    }
    return 0;
}
