import java.util.Scanner;

public class ChoutiSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] evenElements = new int[n / 2];
        for (int i = 0; i < n / 2; i++) {
            evenElements[i] = scanner.nextInt();
        }

        long[] oddElements = new long[n / 2];
        oddElements[0] = (long) Math.sqrt(evenElements[0]);
        for (int i = 1; i < n / 2; i++) {
            oddElements[i] = (long) Math.sqrt(evenElements[i] - evenElements[i - 1]) - oddElements[i - 1];
        }

        if (isPerfectSquare(evenElements[n / 2 - 1] + oddElements[n / 2 - 1])) {
            System.out.println("Yes");
            for (int i = 0; i < n; i++) {
                System.out.print((i % 2 == 0 ? evenElements[i / 2] : oddElements[i / 2]) + " ");
            }
        } else {
            System.out.println("No");
        }
    }

    private static boolean isPerfectSquare(long num) {
        return Math.sqrt(num) == (int) Math.sqrt(num);
    }
}
