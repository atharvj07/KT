import sys

n = int(input())
adj = [[] for _ in range(n + 1)]
vis = [False for _ in range(n + 1)]

for _ in range(n - 1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

paths = []

def dfs(u, p, path):
    vis[u] = True
    path.append(u)
    for v in adj[u]:
        if not vis[v]:
            dfs(v, u, path)
        elif v != p:
            paths.append(path[path.index(v):])

dfs(1, 1, [])

if len(paths) == 0:
    sys.stdout.write("No")
else:
    sys.stdout.write("Yes")
    sys.stdout.write("\n")
    sys.stdout.write(str(len(paths)))
    sys.stdout.write("\n")
    for path in paths:
        sys.stdout.write(str(path[0]) + " " + str(path[-1]))
        sys.stdout.write("\n")
