#include <iostream>
#include <utility>
#include <vector>

using namespace std;

typedef long long ll;
typedef vector<ll> vi;
const ll MOD = 1e9 + 7;

ll fastpow(ll a, ll n) {
  ll ret = 1;
  while (n > 0) {
    if (n % 2 == 1) { ret = (ret * a) % MOD; }
    a = (a * a) % MOD;
    n /= 2;
  }
  return ret;
}

int main() {
  ll n, k;
  cin >> n >> k;
  vi a(n / k), b(n / k);
  for (auto& x : a) { cin >> x; }
  for (auto& x : b) { cin >> x; }
  vi dp(1 << k, -1);
  ll ans = 0;
  dp[0] = 1;
  for (ll i = 0; i < n / k; ++i) {
    vi ndp(1 << k, 0);
    for (ll j = 0; j < (1 << k); ++j) {
      if (dp[j] == -1) continue;
      for (ll d = 0; d < 10; ++d) {
        if (d == b[i]) continue;
        ll nk = j;
        if (d > 0) nk |= (1 << (k - 1));
        ll num = 0;
        for (ll t = 0; t < k; ++t) {
          if (j >> t & 1) num += (d * fastpow(10, k - 1 - t)) % MOD;
        }
        num %= a[i];
        if (num == 0) nk |= 1;
        ndp[nk] += dp[j];
        ndp[nk] %= MOD;
      }
    }
    dp = ndp;
  }
  for (ll i = 0; i < (1 << k); ++i) {
    ans += dp[i];
    ans %= MOD;
  }
  cout << ans << endl;
  return 0;
}
