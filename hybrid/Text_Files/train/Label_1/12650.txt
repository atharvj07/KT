#include <iostream>
#include <vector>
using namespace std;

vector<int> z, l, r;

int main() {
  string s;
  cin >> s;
  int n = s.size(), idx = 0, lmax = 0;
  z.resize(n, 0);
  for (int i = 1, l = 0, r = 0; i < n; i++) {
    if (i <= r)
      z[i] = min(r - i + 1, z[i - l]);
    while (i + z[i] < n && s[z[i]] == s[i + z[i]]) z[i]++;
    if (i + z[i] - 1 > r)
      l = i, r = i + z[i] - 1;
  }
  vector<int> cnt(2);
  cnt[0] = cnt[1] = 0;
  l.resize(n, n), r.resize(n, -1);
  for (int i = 0; i < n; i++) {
    if (s[i] == 'a') {
      cnt[0]++;
      if (!cnt[1]) {
        l[i] = 0, r[i] = n - 1;
      } else {
        l[i] = r[i - 1] + 1;
        r[i] = min(n - 1, r[i - 1] + z[r[i - 1] + 1]);
      }
    }
    if (s[i] == 'b') { cnt[1]++; }
  }
  long long ans = 0, mod = 1e9 + 7;
  lmax = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'a') {
      ans += (lmax - l[i] + 1);
      ans %= mod;
      lmax = r[i];
    }
  }
  cout << ans << endl;
  return 0;
}
