import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String homeTeam = input.nextLine();
        String awayTeam = input.nextLine();

        Map<Integer, String> homeTeamPlayers = new HashMap<>();
        Map<Integer, String> awayTeamPlayers = new HashMap<>();

        int n = input.nextInt();
        for (int i = 0; i < n; i++) {
            int minute = input.nextInt();
            String team = input.next();
            int playerNumber = input.nextInt();
            String card = input.next();

            if (team.equals("h")) {
                if (card.equals("y")) {
                    homeTeamPlayers.put(playerNumber, homeTeam);
                } else if (card.equals("r")) {
                    System.out.println(homeTeam + " " + playerNumber + " " + minute);
                }
            } else if (team.equals("a")) {
                if (card.equals("y")) {
                    awayTeamPlayers.put(playerNumber, awayTeam);
                } else if (card.equals("r")) {
                    System.out.println(awayTeam + " " + playerNumber + " " + minute);
                }
            }

            if (homeTeamPlayers.containsKey(playerNumber) && awayTeamPlayers.containsKey(playerNumber)) {
                System.out.println(homeTeam + " " + playerNumber + " " + minute);
            }
        }
    }
}
