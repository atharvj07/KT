from sys import stdin
input = stdin.readline
mod = 998244353

n, k = map(int, input().split())
a = list(map(int, input().split()))
odd = [0] * n
for i in range(1, n):
	if (i+1)&1 and a[i] != -1 and a[i] == a[i-1]:
		odd[i] = odd[i-1] + 1
	else:
		odd[i] = odd[i-1]
odd.reverse()
suf = odd[:]
a.reverse()
suf[n-1] = 0
for i in range(n-2, -1, -1):
	if (n-i)&1 and a[i] != -1 and a[i] == a[i+1]:
		suf[i] = suf[i+1] + 1
	else:
		suf[i] = suf[i+1]
dp = [(1, 1)] * n
for i in range(n-2, -1, -1):
	if a[i] != -1:
		dp[i] = (dp[i+1][0], dp[i+1][1])
	else:
		if (i+1)&1:
			if odd[i] + suf[i+1] == 0:
				dp[i] = (dp[i+1][0] * k % mod + dp[i+1][1], dp[i+1][0])
			else:
				dp[i] = (dp[i+1][0], dp[i+1][0] * k % mod)
		else:
			dp[i] = (dp[i+1][0], dp[i+1][0] * k % mod + dp[i+1][1] * k % mod)
print(dp[0][0])
