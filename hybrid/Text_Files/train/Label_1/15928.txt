#include<bits/stdc++.h>
using namespace std;
int main(){
    long long t; cin>>t; while(t--){
        long long n; cin>>n; vector<pair<long long,long long>>segs;
        while(n--){ long long a,b; cin>>a>>b; segs.push_back({a,b}); }
        sort(segs.begin(),segs.end());
        int ans=0;
        for(int i=0; i<segs.size()-1; ){
            int j=i+1;
            while(j<segs.size() && segs[j].second<=segs[i].second) j++;
            if(j==segs.size()){ ans+=segs.size()-i-1; break; }
            i=j;
        }
        cout<<ans<<endl;
    }
}
