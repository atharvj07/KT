#include <bits/stdc++.h>

#define ll long long

using namespace std;

const int max_N = 2e5 + 5;

struct segment {
    ll st, en;
    bool operator<(const segment &seg) const {
        return st < seg.st;
    }
};

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        segment seg[max_N];
        for (int i = 0; i < n; i++) {
            cin >> seg[i].st >> seg[i].en;
        }
        sort(seg, seg + n);
        ll ans = 0, en = seg[0].en;
        for (int i = 1; i < n; i++) {
            if (seg[i].st > en) {
                en = seg[i].en;
                ans++;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
