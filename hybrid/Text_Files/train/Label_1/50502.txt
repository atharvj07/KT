import java.util.*;
import java.math.BigInteger;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long[][] dp = new long[n][n];
        for (int i = 0; i < n; i++) {
            dp[i][i] = 1;
        }
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                dp[i][j] = dp[i][j - 1];
                if (BigInteger.valueOf(a[i]).multiply(BigInteger.valueOf(a[j])).isProbablePrime(1)) {
                    dp[i][j] += dp[i + 1][j - 1];
                }
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(dp[i][n - 1] + " ");
        }
        System.out.println();
    }
}
