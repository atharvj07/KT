#include <bits/stdc++.h>

using namespace std;

const int M = 2e5 + 5;
const long long MOD = 1e9 + 7;

vector<pair<int, int>> adj[M];
int sz[M], dist[M], ans[M];

void dfs(int u, int p) {
    sz[u] = 1;
    for (auto v : adj[u]) {
        if (v.first != p) {
            dist[v.first] = dist[u] + v.second;
            dfs(v.first, u);
            sz[u] += sz[v.first];
        }
    }
}

void solve(int u, int p) {
    for (auto v : adj[u]) {
        if (v.first != p) {
            ans[u] += ans[v.first] + sz[v.first] * (sz[u] - sz[v.first]);
            solve(v.first, u);
        }
    }
}

int main() {
    int n, k;
    cin >> n >> k;

    for (int i = 1; i < n; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }

    dfs(1, 1);

    for (int i = 1; i <= n; i++) {
        for (auto v : adj[i]) {
            int u = v.first, w = v.second;
            if (dist[u] > dist[i]) {
                if (dist[u] - dist[i] <= k) {
                    ans[i]++;
                }
            }
        }
    }

    solve(1, 1);

    long long sum = 0;
    for (int i = 1; i <= n; i++) {
        sum += ans[i];
    }

    cout << sum % MOD << endl;

    return 0;
}
