#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  set<int> s;
  for (int l = 0; l < n; l++) {
    int cur = 0;
    for (int r = l; r < n; r++) {
      cur |= a[r];
      s.insert(cur);
    }
  }
  cout << s.size() << endl;
  return 0;
}
