t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    count_map = {}
    result = 0
    for i, value in enumerate(a):
        if value in count_map:
            result += count_map[value] * (n - i)
            count_map[value] += i + 1
        else:
            count_map[value] = i + 1
    print(result)
