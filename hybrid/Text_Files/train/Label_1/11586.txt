#include <cstdio>
#include <cstring>
using namespace std;
int n, k, v[200005], ans;
int main() {
  for (int t, i; scanf("%d", &t) != EOF;)
    while (t--) {
      scanf("%d%d%s", &n, &k, v);
      for (ans = 0, i = k + 1; i <= n; i++)
        if (!v[i - k] && !v[i]) {
          ans++;
          v[i] = 1;
        }
      printf("%d\n", ans);
    }
  return 0;
}
