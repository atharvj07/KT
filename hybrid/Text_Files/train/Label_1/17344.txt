#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> v(n * 2), p(n * 2, -1);
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (!i || s[i - 1] == '0') {
        int j = i;
        while (j < n * 2 && v[j] == 0) {
          ans++;
          v[j] = 1;
          j = v[j + n] == 0 ? j + n : j + 1;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
