import collections
for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = collections.Counter(a)
    ans = 0
    for i in range(101):
        if b[i] == 0:
            ans += 1
            b[i] += 1
            if b[i] % 2 == 0:
                b[i] -= 1
            else:
                if b[i] != 0:
                    ans += 1
                    b[i] -= 1
    print(ans)
