#include <iostream>
#include <stack>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int max_len = 0, num_words = 0;
    stack<char> st;
    int cur_len = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '_') {
            max_len = max(max_len, cur_len);
        } else if (s[i] == '(') {
            st.push('(');
            cur_len = 0;
        } else if (s[i] == ')') {
            st.pop();
            num_words++;
        } else {
            cur_len++;
        }
    }
    max_len = max(max_len, cur_len);
    if (st.empty()) {
        cout << max_len << " " << num_words << endl;
    } else {
        cout << 0 << " " << 0 << endl;
    }
    return 0;
}
