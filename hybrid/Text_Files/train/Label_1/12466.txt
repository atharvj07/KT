#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> x(n), y(n), t(k), f(k);
  for (int i = 0; i < n; i++) cin >> x[i];
  for (int i = 0; i < n; i++) cin >> y[i];
  int tt = 0, ff = 0;
  for (int i = 0; i < n; i++) {
    if (x[i] > k) tt = -1;
    if (y[i] > k) ff = -1;
    t[(tt + x[i]) % k]++;
    f[(ff + y[i]) % k]++;
    if (t[(tt + x[i]) % k] > 1 || f[(ff + y[i]) % k] > 1) { tt = -1; ff = -1; }
  }
  if (tt == 0 && ff == 0)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
}
