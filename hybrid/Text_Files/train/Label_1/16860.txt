#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = (int)1e9 + 7;
ll f(ll x, ll y) {
    for (ll p : {2, 3, 5, 7})
        while (x % p == 0 && (x /= p) % p == 0) y /= p;
    return y % mod;
}
int main() {
    ll x, n;
    cin >> x >> n;
    ll ans = 1;
    while (n) {
        if (n & 1) ans = ans * f(x, n) % mod;
        x *= x;
        x %= mod;
        n >>= 1;
    }
    cout << ans << "\n";
    return 0;
}
