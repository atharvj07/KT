import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int q = sc.nextInt();
            String s = sc.next();

            for (int j = 0; j < q; j++) {
                int l = sc.nextInt();
                int r = sc.nextInt();

                boolean found = false;
                for (int k = l - 1; k < r - 1; k++) {
                    if (s.charAt(k) == s.charAt(r - 1)) {
                        found = true;
                        break;
                    }
                }

                System.out.println(found ? "YES" : "NO");
            }
        }
    }
}
