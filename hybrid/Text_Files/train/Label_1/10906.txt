#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    unordered_map<int, int> cnt;
    int win = -1;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      cnt[x]++;
      if (cnt[x] == 1) win = x;
    }
    for (auto &p : cnt) {
      if (p.first < win && p.second == 1) win = p.first;
    }
    cout << win << endl;
  }
  return 0;
}
