import sys
sys.setrecursionlimit(2500)
dr = [0, 0, -1, 1]
dc = [-1, 1, 0, 0]
def go(r, c, o, t):
    if r>=n or r<0 or c>=m or c<0 or o[r][c]==-1:
        return n*m+1
    if t==0:
        return 0
    if o[r][c]!=-2:
        ans = o[r][c]
    else:
        o[r][c] = -1
        ans = n*m+1
        for i in range(4):
            ans = min(ans, 1+go(r+dr[i], c+dc[i], o, t-1))
        o[r][c] = ans
    return ans

n, m, k = map(int, input().split())
g = []
for i in range(n):
    a = []
    s = input()
    for j in range(m):
        if s[j]==".":
            a += [0]
        else:
            dd = "LRUD".index(s[j])
            if (i + j) % 2==dd or (i + dd) % 2==j:
                a += [-2]
            else:
                a += [0]
    g += [a]
ans = []
for j in range(m):
    ans += [go(0, j, g, n)]
print(*ans)
