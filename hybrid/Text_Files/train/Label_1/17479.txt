#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<long long, long long>> companies(n);
    long long ans = 0;
    for (auto &company : companies) {
        int m;
        cin >> m;
        long long sum = 0, mx = 0;
        for (int i = 0; i < m; i++) {
            long long x;
            cin >> x;
            sum += x;
            mx = max(mx, x);
        }
        ans += max(0LL, mx - sum / m);
        company = {sum / m, mx};
    }
    sort(companies.begin(), companies.end());
    for (int i = 1; i < n; i++) {
        if (companies[i].first != companies[i - 1].first) {
            ans += companies[i].first - companies[i - 1].first;
        }
    }
    cout << ans << endl;
    return 0;
}
