import java.util.*;
 
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
 
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
 
        int m = sc.nextInt();
        int[] ans = new int[m];
 
        for (int i = 0; i < m; i++) {
            int w = sc.nextInt();
            int h = sc.nextInt();
            ans[i] = arr[w - 1] + h;
            for (int j = 0; j < w - 1; j++) {
                ans[i] = Math.max(ans[i], arr[j] + h);
            }
        }
 
        for (int x: ans) {
            System.out.println(x);
        }
    }
}
