import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        long[] c = new long[k];
        for (int i = 0; i < k; i++) {
            c[i] = in.nextInt();
        }
        long[] f = new long[k];
        long mod = 1000000007;
        f[k - 1] = 1;
        for (int i = k - 2; i >= 0; i--) {
            f[i] = f[i + 1] * c[i] % mod;
        }
        long ans = 0;
        for (int i = 0; i < k; i++) {
            ans = (ans + f[i]) % mod;
        }
        System.out.println(ans);
    }
}
