#include <iostream>
#include <cmath>
using namespace std;
bool isPT(long long n, long long &mp) {
	double p = sqrt(n * n * 1.0 / 2);
	if (p != int(p)) return false;
	mp = int(p);
	return true;
}
int main() {
	long long n, mp;
	cin >> n;
	long long lim = sqrt(2) * n;
	for (long long m = 1; m < lim; m++) {
		if (!isPT(n, mp)) continue;
		cout << m << " " << mp << "\n";
		return 0;
	}
	cout << -1 << "\n";
	return 0;
}
