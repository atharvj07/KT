#include <vector>
#include <iostream>

using namespace std;

int main()
{
    int n;
    cin >> n;

    vector<int> nums(n), freq(n + 1, 0);
    for (int i = 0; i < n; i++)
    {
        cin >> nums[i];
        freq[nums[i]]++;
    }

    vector<int> ans(n + 1, 0);

    for (int i = 0; i < n; i++)
        ans[nums[i]] += n - i;

    for (int i = 1; i <= n; i++)
    {
        if (i > 1)
            ans[i] += ans[i - 1];
        ans[i] -= freq[i] * (freq[i] - 1) / 2;
        cout << ans[i] << " ";
    }

    return 0;
}
