#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int& ai : a) cin >> ai;
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i];
        if (sum * 2 >= sum) {
            cout << i + 1 << endl;
            return 0;
        }
    }
    return 0;
}
