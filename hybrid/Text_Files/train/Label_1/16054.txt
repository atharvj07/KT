#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int mod = 998244353;
const int maxn = 2e5 + 10;

ll n, ans;
ll a[maxn], b[maxn], c[maxn];
vector<ll> adj[maxn];

bool bfs(ll s) {
    queue<ll> q;
    q.push(s);
    c[s] = 1;
    while (!q.empty()) {
        ll u = q.front();
        q.pop();
        for (auto v : adj[u]) {
            if (c[v] == 0) {
                q.push(v);
                c[v] = 3 - c[u];
            } else if (c[v] == c[u]) return false;
        }
    }
    return true;
}

int main() {
    cin >> n;
    for (ll i = 0; i < n; i++) {
        cin >> b[i];
        a[i] = b[i];
    }
    sort(a, a + n);
    for (ll i = 0; i < n; i++) {
        for (ll j = i + 1; j < n; j++) {
            if (a[j] - a[i] > a[n - 1]) break;
            adj[a[i]].push_back(a[j] - a[i]);
            adj[a[j] - a[i]].push_back(a[i]);
        }
    }
    for (ll i = 0; i < n; i++) {
        if (c[b[i]] == 0) {
            if (bfs(b[i]) == false) { ans++; c[b[i]] = -1; }
        }
    }
    cout << ans << endl;
    for (ll i = 0; i < n; i++) {
        if (c[b[i]] == -1) cout << b[i] << " ";
    }
    return 0;
}
