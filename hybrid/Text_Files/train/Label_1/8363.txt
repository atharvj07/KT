#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll a[55][2],b[55][2],c[55],d[55];
ll ans;
int main(){
    ll p,q,l,r;
    cin>>p>>q>>l>>r;
    for(ll i=1;i<=p;i++)cin>>a[i][0]>>a[i][1];
    for(ll i=1;i<=q;i++)cin>>b[i][0]>>b[i][1];
    for(ll i=1;i<=q;i++){
        c[i]=b[i][0];
        d[i]=b[i][1];
    }
    for(ll i=l;i<=r;i++){
        for(ll j=1;j<=p;j++){
            for(ll k=1;k<=q;k++){
                if(i>=a[j][0] && i<=a[j][1] && i+c[k]>=b[k][0] && i+c[k]<=b[k][1]){
                    ans++;
                }
            }
        }
    }
    cout<<ans<<endl;
}
