import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        int n = Integer.parseInt(s[0]), m = Integer.parseInt(s[1]);
        int[][] a = new int[n][m], b = new int[n][m], ans = new int[2501][2];
        for (int i = 0; i < n; i++) {
            s = br.readLine().split(" ");
            for (int j = 0; j < m; j++) {
                a[i][j] = Integer.parseInt(s[j]);
            }
        }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == 0) continue;
                if (b[i][j] == 0) {
                    b[i][j] = 1;
                    ans[cnt][0] = i + 1;
                    ans[cnt++][1] = j + 1;
                }
                if (i < n - 1 && b[i + 1][j] == 0) {
                    b[i + 1][j] = 1;
                    ans[cnt][0] = i + 2;
                    ans[cnt++][1] = j + 1;
                }
                if (j < m - 1 && b[i][j + 1] == 0) {
                    b[i][j + 1] = 1;
                    ans[cnt][0] = i + 1;
                    ans[cnt++][1] = j + 2;
                }
                if (i < n - 1 && j < m - 1 && b[i + 1][j + 1] == 0) {
                    b[i + 1][j + 1] = 1;
                    ans[cnt][0] = i + 2;
                    ans[cnt++][1] = j + 2;
                }
            }
        }
        boolean flag = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] != b[i][j]) {
                    flag = false;
                    break;
                }
            }
            if (!flag) break;
        }
        if (flag) {
            System.out.println(cnt);
            for (int i = 0; i < cnt; i++) {
                System.out.println(ans[i][0] + " " + ans[i][1]);
            }
        } else {
            System.out.println(-1);
        }
    }
}
