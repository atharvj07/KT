#include <cstdio>
int main() {
    int t;
    scanf("%d\n", &t);
    while (t--) {
        long long a, b, c, d;
        scanf("%lld %lld %lld\n", &a, &b, &c);
        if (a / 2 == b / 2 && b / 2 == c / 2)
            printf("%lld\n", 2 * b / a * c);
        else
            printf("-1\n");
    }
    return 0;
}
