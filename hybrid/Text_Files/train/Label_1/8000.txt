#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<vector<int>> ips(n, vector<int>(4));
    map<vector<int>, int> cnts;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 4; j++) {
            cin >> ips[i][j];
        }
        sort(ips[i].begin(), ips[i].end());
        cnts[ips[i]]++;
    }
    if (cnts.size() == k) {
        int ans = 0;
        int minCnt = n;
        for (auto p : cnts) {
            int mask = 0;
            int cnt = p.second;
            if (cnt > minCnt) {
                continue;
            } else if (cnt == minCnt) {
                int cur = 0;
                for (int i = 0; i < 4; i++) {
                    if (p.first[i] == 255) {
                        cur += 8;
                    } else {
                        break;
                    }
                }
                if (cur > ans) {
                    continue;
                }
            }
            for (int i = 0; i < 4; i++) {
                if (p.first[i] == 255) {
                    mask |= (1 << (32 - i * 8 - 1)) - 1;
                } else {
                    mask |= (1 << (32 - i * 8 - 1)) - 1 -
                            ((p.first[i] << (32 - i * 8 - 1)) - 1);
                    break;
                }
            }
            if (cnt == minCnt && ans == mask) {
                continue;
            }
            ans = mask;
            minCnt = cnt;
        }
        if (ans == 0) {
            ans = -1;
        }
        cout << ans << endl;
    } else {
        cout << -1 << endl;
    }
    return 0;
}
