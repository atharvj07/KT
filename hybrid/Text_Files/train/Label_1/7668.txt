#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int deg = 0;
    int center = -1;
    for (int i = 1; i <= n; i++) {
        deg = max(deg, (int)adj[i].size());
        if ((int)adj[i].size() > 2) { center = -1; }
        if ((int)adj[i].size() == 1) { center = i; }
    }
    if (deg == 2) { cout << "bus topology" << endl; }
    else if (center != -1) { cout << "star topology" << endl; }
    else if (deg == n - 1) { cout << "ring topology" << endl; }
    else { cout << "unknown topology" << endl; }
    return 0;
}
