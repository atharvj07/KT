#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; ++i) cin >> v[i];
    int ans = v[0];
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (abs(v[i] - v[j]) > 1) {
                ans = v[i];
                return cout << ans << endl, 0;
            }
        }
    }
    return 0;
}
