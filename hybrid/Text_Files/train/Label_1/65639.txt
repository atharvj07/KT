from sys import stdin
input = stdin.readline

n = int(input())
MOD = 1000000007

add_prices = []
best = [0, 308983066]

for i in range(n):
    action, price = input().split()
    if action=='ADD':
        add_prices.append(int(price))
    else:
        price = int(price)
        best[int(price==best[0])] = price

add_prices.sort()

# number of ways to assign direction to add_prices
ans = 1
for price in add_prices:
    if price > best[0]:
        break
    if price < best[1]:
        ans = 0
        break
    ans = 2 * ans % MOD

print(ans)
