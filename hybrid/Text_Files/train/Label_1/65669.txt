import sys
from collections import defaultdict
input=sys.stdin.readline
n,m=map(int,input().split())
e=[[]for i in range(n+1)]
for i in range(m):
    l=[int(x)for x in input().split()]
    e[l[0]].append(l[1])
k=1
col=[0]*m
while k<=4:
    ok=True
    c=0
    for i in range(1,n+1):
        if col[i]==0:
            q=[]
            q.append([i,k])
            while len(q):
                now=q[0][0]
                col[now]=q[0][1]
                del q[0]
                for j in e[now]:
                    if col[j]==0:
                        q.append([j,k])
                    elif col[j]==col[now]:
                        ok=False
                        break
            c=c+1
    if ok:break
    k=k+1
print(k)
for i in range(1,n+1):print(col[i],end=' ')
print()
