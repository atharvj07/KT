from collections import defaultdict

n, x0, y0 = map(int, input().split())
d = defaultdict(list)

for x, y in (map(int, input().split()) for _ in range(n)):
    if x == x0:
        d[x0].append(y)
    else:
        a = (y - y0) / (x - x0)
        b = y0 - a * x0
        d[(a, b)].append(x)

ans = len(d)

for key in d:
    if type(key) == tuple:
        x, c = key
        d[key] = sorted(d[key])
        ans += 1 - (len(d[key]) & 1)
        if c * x0 + y0 in d[x0]:
            ans -= 1

print(ans)
