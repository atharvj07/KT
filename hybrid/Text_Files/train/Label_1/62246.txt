import sys
input = sys.stdin.readline

n, k = map(int, input().split())
a = list(map(int, input().split()))
l, r = 0, 0
ans = 0
while r < n:
    if a[r] == 0:
        if k > 0:
            k -= 1
            ans = max(ans, r - l + 1)
        else:
            while a[l] != 0:
                l += 1
            l += 1
    else:
        ans = max(ans, r - l + 1)
    r += 1
print(ans)
for i in range(l, r):
    a[i] = 1
print(*a)
