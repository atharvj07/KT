#include<bits/stdc++.h>
#define ll long long
using namespace std;
map<ll,ll>m,p,mp;
int main() {
    ll n,i,j,x,y,k,ans[100005],mx=0;
    cin>>n>>k;
    for(i=1; i<=n; i++) {
        cin>>x;
        m[x]=i,p[i]=x;
    }
    while(k--) {
        cin>>x>>y;
        mp[min(x,y)]++,mp[max(x,y)]--;
    }
    for(auto it=mp.begin(); it!=mp.end(); it++) {
        auto x=it->first,y=it->second;
        if(y>0) {
            while(y--) {
                swap(m[p[x]],m[p[x+1]]);
                swap(p[x],p[x+1]);
                x++;
            }
        }
        else y*=-1;
        if(y>0) {
            while(y--) {
                swap(m[p[x]],m[p[x-1]]);
                swap(p[x],p[x-1]);
                x--;
            }
        }
    }
    for(i=1; i<=n; i++) ans[m[i]]=p[i];
    for(i=1; i<=n; i++) cout<<ans[i]<<" ";
    cout<<endl;
}
