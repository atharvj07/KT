#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
int n, m, res[105][105];
int a[105][105];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < m; j++) { cin >> a[i][j]; }
        }
        for (int i = 1; i <= n; i++) {
            sort(a[i], a[i] + m);
            for (int j = 0; j < m; j++) { res[i][j + 1] = a[i][j]; }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) { cout << res[i][j] << ' '; }
            cout << endl;
        }
    }
    return 0;
}
