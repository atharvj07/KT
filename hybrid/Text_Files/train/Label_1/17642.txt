#include<bits/stdc++.h>
using namespace std;
const int maxn=2e5+1;
int s[maxn],f[maxn],mxr[maxn],mnr[maxn];bool fl[maxn];
int main()
{
    int n,ans=0;cin>>n;
    for(int i=1;i<=n;i++) cin>>s[i];
    fill(f,f+maxn,1);mnr[n]=1e9;
    for(int i=n;i>=1;i--)
    {
        fl[i]=fl[i+1]||f[i]==n;
        int rpos=lower_bound(s+fl[i]+1,s+n+1,2*s[i])-s-1;
        if(rpos<n) ans+=fl[rpos+1]||rpos==i;
        mxr[i]=rpos;
    }
    memset(fl,0,sizeof(fl));
    for(int i=1;i<=n;i++)
    {
        fl[i]=fl[i-1]||f[i]==n;
        int lpos=upper_bound(s+fl[i],s+n+1,s[i]/2)-s-1;
        if(lpos>0) ans+=fl[lpos-1]||lpos==i;
        mnr[i]=lpos;
    }
    int l=1,r=n;
    while(l<=r)
    {
        if(mxr[l]>=mnr[r])
        {
            ans--,l++,r--;
            if(l<=r) ans++;
        }
        else l++;
    }
    cout<<ans;
}
