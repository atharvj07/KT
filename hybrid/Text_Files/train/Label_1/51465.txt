import java.util.HashSet;
import java.util.Scanner;

public class SergeyAndTheForbiddenSubstrings {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        String s = scanner.nextLine();
        int k = scanner.nextInt();
        scanner.nextLine();

        HashSet<String> forbidden = new HashSet<>();
        for (int i = 0; i < k; i++) {
            String pair = scanner.nextLine();
            forbidden.add(pair);
        }

        int deletions = 0;
        int n = s.length();
        for (int i = 0; i < n - 1; i++) {
            String pair = s.substring(i, i + 2);
            if (forbidden.contains(pair)) {
                deletions++;
                // Remove either the first or the second character of the pair
                if (i + 2 < n && !forbidden.contains(s.substring(i + 1, i + 3))) {
                    s = s.substring(0, i + 1) + s.substring(i + 2);
                } else {
                    s = s.substring(0, i) + s.substring(i + 1);
                }
                i--;
                n--;
            }
        }

        System.out.println(deletions);
    }
}
