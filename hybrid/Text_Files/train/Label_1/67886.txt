from math import gcd, ceil
def solve():
    t = int(input())
    for i in range(t):
        n = int(input())
        a = [int(x) for x in input().split()]
        a.sort()
        if n == 2: print(0)
        else: b = a[1]-a[0]
        for j in range(2,n): b = gcd(b,a[j]-a[j-1])
        if b <= 1: print(0)
        else: print(b)
solve()
