import java.util.HashSet;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;
import java.util.Set;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int q = scanner.nextInt();
        Set<Integer> has = new HashSet<>();
        for (int i = 0; i < q; i++) {
            int r = scanner.nextInt() - 1;
            int c = scanner.nextInt() - 1;
            has.add(r * m + c);
        }
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!has.contains(i * m + j)) {
                    queue.offer(i * m + j);
                }
            }
        }
        int ans = 0;
        while (!queue.isEmpty()) {
            Set<Integer> next = new HashSet<>();
            while (!queue.isEmpty()) {
                int cur = queue.poll();
                int r = cur / m;
                int c = cur % m;
                if (r > 0 && !has.contains((r - 1) * m + c)) {
                    next.add((r - 1) * m + c);
                }
                if (r < n - 1 && !has.contains((r + 1) * m + c)) {
                    next.add((r + 1) * m + c);
                }
                if (c > 0 && !has.contains(r * m + c - 1)) {
                    next.add(r * m + c - 1);
                }
                if (c < m - 1 && !has.contains(r * m + c + 1)) {
                    next.add(r * m + c + 1);
                }
            }
            ans++;
            queue.addAll(next);
            has.addAll(next);
        }
        System.out.println(ans);
    }
}
