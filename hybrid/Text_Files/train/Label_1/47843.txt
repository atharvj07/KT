import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] rows = new int[n + 1];
        int[] cols = new int[n + 1];
        int[][] rectangles = new int[q][4];
        for (int i = 0; i < q; i++) {
            for (int j = 0; j < 4; j++) {
                rectangles[i][j] = sc.nextInt();
            }
        }
        for (int[] rectangle : rectangles) {
            for (int i = rectangle[0]; i <= rectangle[2]; i++) {
                rows[i] = 1;
            }
            for (int i = rectangle[1]; i <= rectangle[3]; i++) {
                cols[i] = 1;
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            ans += rows[i] == 0 ? 1 : 0;
        }
        for (int i = 1; i <= n; i++) {
            ans += cols[i] == 0 ? 1 : 0;
        }
        System.out.println(ans);
    }
}
