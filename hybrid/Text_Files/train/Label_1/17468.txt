#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    int n; cin >> n;
    vector<ll>v(n), dp(n);
    for(int i = 0; i < n; i++) {
        cin >> v[i];
    }
    dp[0] = 1;
    for(int i = 1; i < n; i++) {
        int lo = 0, hi = i - 1, idx = i;
        while(lo <= hi) {
            int mid = (lo + hi) / 2;
            if(2 * v[mid] >= v[i]) {
                idx = mid;
                hi = mid - 1;
            } else {
                lo = mid + 1;
            }
        }
        dp[i] = dp[idx] + 1;
    }
    cout << *max_element(dp.begin(), dp.end()) << '\n';
}
