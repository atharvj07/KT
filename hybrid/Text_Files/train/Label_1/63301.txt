import sys
input = sys.stdin.readline

def dist(a, b):
    x1, y1 = a
    x2, y2 = b
    return ((x1 - x2)**2 + (y1 - y2)**2)**0.5

for t in range(int(input())):
    n = int(input())
    m = []
    for _ in range(2 * n):
        x, y = map(int, input().split())
        m.append((x,y))
    x, y = [], []
    for p in m:
        if p[0] == 0: y.append(p)
        if p[1] == 0: x.append(p)
    x.sort()
    y.sort()
    ans = 0
    for i in range(n):
        ans += dist(x[i], y[i])
    print(ans)
