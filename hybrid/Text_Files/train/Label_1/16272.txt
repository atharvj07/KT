#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, l, r;
    cin >> n;
    vector<pair<int, int>> pts(n * 2);
    for (int i = 0; i < n; i++) {
        cin >> l >> r;
        pts[i] = { l, 1 };
        pts[i + n] = { r + 1, -1 };
    }
    sort(pts.begin(), pts.end());
    int cnt = 0, ans[n + 1] = { 0 };
    for (auto seg : pts) {
        cnt += seg.second;
        ans[cnt]++;
    }
    for (int i = 1; i <= n; i++) cout << ans[i] << ' ';
}
