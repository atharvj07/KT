import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int[] dp = new int[n + 1];
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            dp[i + 1] = dp[i];
            ans[i] = a[i];
            if (a[i] == 0) {
                if (k > 0) {
                    k--;
                    dp[i + 1]++;
                    ans[i] = 1;
                }
            } else {
                dp[i + 1]++;
            }
        }
        int max = 0;
        int idx = -1;
        for (int i = 0; i <= n; i++) {
            if (dp[i] > max) {
                max = dp[i];
                idx = i;
            }
        }
        System.out.println(max);
        for (int i = 0; i < idx; i++) {
            System.out.print(ans[i] + " ");
        }
        for (int i = idx; i < n; i++) {
            System.out.print("1 ");
        }
        System.out.println();
    }
}
