n = int(input())
al = []
for i in range(n):
    al.append(list(map(int, input().split())))
upper, lower = 0, 0
for a in al:
    upper += a[0]
    lower += a[1]
if (upper + lower) % 2 == 1:
    print(-1)
else:
    odd_upper = 0
    odd_lower = 0
    for a in al:
        if a[0] % 2 == 1:
            odd_upper += 1
        if a[1] % 2 == 1:
            odd_lower += 1
    print(min(odd_upper, odd_lower, n - odd_upper, n - odd_lower))
