import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read input
        int n = scanner.nextInt();
        int w = scanner.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
        }

        // Initialize dp table
        int[][] dp = new int[n + 1][2];
        dp[0][0] = 0;
        dp[0][1] = 0;

        // Calculate dp table
        for (int i = 1; i <= n; i++) {
            // Calculate score for i-th bit being 0
            int score0 = dp[i - 1][0];
            if (i - w - 1 >= 0) {
                score0 += a[i - w - 1];
            }

            // Calculate score for i-th bit being 1
            int score1 = dp[i - 1][1];
            int oddCount = 0;
            for (int j = i - 1; j >= 0 && j >= i - w; j--) {
                oddCount += a[j];
            }
            if (oddCount % 2 == 1) {
                score1 += b[i - 1];
            }

            // Update dp table
            dp[i][0] = Math.max(score0, score1);
            dp[i][1] = Math.max(score0, score1 + a[i - 1]);
        }

        // Construct bit string
        StringBuilder bitString = new StringBuilder();
        int i = n;
        int bit = 1;
        while (i > 0) {
            if (dp[i][bit] == dp[i - 1][0] + a[i - 1]) {
                bit = 0;
            }
            bitString.append(bit);
            i--;
        }

        // Print bit string
        System.out.println(bitString.reverse());
    }
}
