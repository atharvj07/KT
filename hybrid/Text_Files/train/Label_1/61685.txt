import math

def gcd(a, b):
    if a == 0:
        return b
    return gcd(b % a, a)

def solve(a1, b1, a2, b2):
    g = gcd(a1, b1)
    if a2 % g == 0 and b2 % g == 0:
        m = 0
        while a1 % 3 == 0 and b1 % 3 == 0:
            a1 //= 3
            b1 //= 3
            m += 1
        while a1 % 2 == 0 and b1 % 2 == 0:
            a1 //= 2
            b1 //= 2
            m += 1
        if a1 == 1 and b1 == 1:
            return m
    if a1 % 2 == 0 and b2 % 2 == 0 and gcd(a1 // 2, b2) == g:
        m = 0
        while a1 % 3 == 0 and b2 % 3 == 0:
            a1 //= 3
            b2 //= 3
            m += 1
        while a1 % 2 == 0 and b2 % 2 == 0:
            a1 //= 2
            b2 //= 2
            m += 1
        if a1 == 1 and b2 == 1:
            return m
    if b1 % 2 == 0 and a2 % 2 == 0 and gcd(b1 // 2, a2) == g:
        m = 0
        while b1 % 3 == 0 and a2 % 3 == 0:
            b1 //= 3
            a2 //= 3
            m += 1
        while b1 % 2 == 0 and a2 % 2 == 0:
            b1 //= 2
            a2 //= 2
            m += 1
        if b1 == 1 and a2 == 1:
            return m
    return -1

a1, b1 = map(int, input().split())
a2, b2 = map(int, input().split())
ans = solve(a1, b1, a2, b2)
if ans == -1:
    print(ans)
else:
    print(ans)
    print(a1, b1)
    print(a2, b2)
