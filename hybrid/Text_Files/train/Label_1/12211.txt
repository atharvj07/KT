#include <iostream>
#include <vector>
using namespace std;
const int mod = 1e9 + 7;
long long n, m, ans;
string s, t;
vector<vector<int>> nxt;
int main() {
  cin >> n >> s >> m >> t;
  nxt.resize(26);
  for (int i = n - 1; i >= 0; i--) {
    nxt[s[i] - 'a'].push_back(i);
  }
  for (int i = 0; i < m / 2; i++) {
    int cur = i & 1;
    if (nxt[t[i * 2] - 'a'].empty()) {
      ans++;
      continue;
    }
    ans += abs(nxt[t[i * 2] - 'a'].back() - nxt[t[i * 2 + 1] - 'a'].back());
    nxt[t[i * 2] - 'a'].pop_back();
    nxt[t[i * 2 + 1] - 'a'].pop_back();
  }
  cout << ans << '\n';
  return 0;
}
