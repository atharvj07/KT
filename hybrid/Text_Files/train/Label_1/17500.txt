#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  int win = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (win == k - 1) {
      cout << a[i] << endl;
      return 0;
    }
    win++;
  }
  return 0;
}
