#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &ai : a) cin >> ai;
    int ans = 1;
    for (int i = 0, j; i < n; i = j) {
        for (j = i + 1; j < n && a[j] > a[j - 1]; j++)
            ;
        ans = max(ans, j - i);
    }
    cout << ans << endl;
}
