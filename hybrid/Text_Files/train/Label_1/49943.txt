import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        int count = 0;
        for (int i = 100; i >= 0; i--) {
            while (n >= Math.pow(10, i)) {
                n -= Math.pow(10, i);
                count++;
            }
        }
        System.out.println(count);
    }
}
