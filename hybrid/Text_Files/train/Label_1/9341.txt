#include <bits/stdc++.h>
using namespace std;
string a,b;
int n,m,f[100010];
int main(){
  cin>>a>>b;
  n=a.size();
  m=b.size();
  for(int i=0; i<n; i++)
    if(i&&a[i]==a[i-1]) f[i]=f[i-1];
    else {
      int k=i-1;
      while(k>=0&&a[k]!=a[i]) k=f[k]-1;
      if(k==-1) f[i]=0;
      else f[i]=k+1;
    }
  int ans=0,j=0;
  for(int i=0; i<n; i++){
    while(j&&b[j]!=a[i]) j=f[j]-1;
    if(b[j]==a[i]) j++;
    if(j==m) {
      ans++;
      j=f[j]-1;
    }
  }
  cout<<ans<<endl;
  return 0;
}
