#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];

  vector<int> da(n, 0), db(m, 0);
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      if (a[i][j] == '>')
        da[i] = max(da[i], db[j] + 1);
      else if (a[i][j] == '<')
        db[j] = max(db[j], da[i] + 1);
      else if (a[i][j] == '=')
        da[i] = db[j] = max(da[i], db[j]) + 1;
    }

  bool ok = true;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      if (a[i][j] == '>')
        ok &= da[i] > db[j];
      else if (a[i][j] == '<')
        ok &= da[i] < db[j];
      else if (a[i][j] == '=')
        ok &= da[i] == db[j];
    }

  if (!ok)
    cout << "No" << endl;
  else {
    cout << "Yes" << endl;
    for (int i = 0; i < n; i++) cout << da[i] << ' ';
    cout << endl;
    for (int j = 0; j < m; j++) cout << db[j] << ' ';
    cout << endl;
  }

  return 0;
}
