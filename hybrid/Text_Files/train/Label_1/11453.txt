#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
 
const int MAXN = 2e5 + 7;
 
int a[MAXN], Sorted[MAXN];
 
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i <= n; i++) {
    Sorted[i] = a[i];
  }
  sort(Sorted + 1, Sorted + 1 + n);
  int pos = -1;
  for (int i = 1; i <= n; i++) {
    int d1 = 0, d2 = 0;
    if (i > 1) {
      d1 = abs(Sorted[i] - Sorted[i-1]);
    }
    if (i < n) {
      d2 = abs(Sorted[i+1] - Sorted[i]);
    }
    if (d1 == d2) {
      pos = i;
    }
  }
  for (int i = 1; i <= n; i++) {
    int d1 = 0, d2 = 0;
    if (i > 1) {
      d1 = abs(a[i] - a[i-1]);
    }
    if (i < n) {
      d2 = abs(a[i+1] - a[i]);
    }
    if (d1 == d2) {
      pos = i;
    }
  }
  cout << pos << endl;
  return 0;
}
