def can_draw_ticks(n, m, k, field):
    for i in range(n-1, -1, -1):
        for j in range(m):
            if field[i][j] == '*':
                # Find the maximum possible size of a tick with the center at (i, j)
                tick_size = 0
                while (i - tick_size - 1 >= 0 and j - tick_size - 1 >= 0 and j + tick_size + 1 < m and
                       field[i - tick_size - 1][j - tick_size - 1] == '*' and
                       field[i - tick_size - 1][j + tick_size + 1] == '*'):
                    tick_size += 1
                
                # If the tick size is less than k, return False
                if tick_size < k:
                    return "NO"
                
                # Mark all cells of the current tick as '.', except the center
                for h in range(1, tick_size + 1):
                    field[i - h][j - h] = '.'
                    field[i - h][j + h] = '.'
    
    return "YES"

# Read the number of test cases
t = int(input())

# Process each test case
for _ in range(t):
    n, m, k = map(int, input().split())  # Read field size and minimum tick size
    field = [list(input().strip()) for _ in range(n)]  # Read the field
    print(can_draw_ticks(n, m, k, field))
