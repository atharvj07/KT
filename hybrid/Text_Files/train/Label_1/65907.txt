import sys

n, m, h = map(int, input().split())
front = list(map(int, input().split()))
left = list(map(int, input().split()))
top = [list(map(int, input().split())) for i in range(n)]

result = [[' ' for j in range(m)] for i in range(n)]

for i in range(n):
  for j in range(m):
    if front[j] > 0:
      h = front[j]
      result[i][j] = h
      front[j] -= h
    if left[i] > 0:
      h = left[i]
      result[i][j] = h
      left[i] -= h
    if top[i][j] > 0:
      result[i][j] = h

for i in range(n):
  for j in range(m):
    if result[i][j] == ' ':
      result[i][j] = 0

for i in range(n):
  print(' '.join(map(str, result[i])))
