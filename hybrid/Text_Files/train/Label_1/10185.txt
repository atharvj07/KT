#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e6+10;
vector<int> f(MAXN);

void pre() {
    for(int i = 2; i < MAXN; ++i) {
        if(!f[i]) {
            for(int j = i; j < MAXN; j+=i)
                if(!f[j])
                    f[j] = i;
        }
    }
}

int main() {
    pre();
    int t;
    cin >> t;
    while(t--) {
        int n, k;
        cin >> n >> k;
        while(k--) {
            n += f[n];
        }
        cout << n << endl;
    }
    return 0;
}
