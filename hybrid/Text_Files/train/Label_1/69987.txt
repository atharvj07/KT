import sys

n = int(sys.stdin.readline())
a = list(map(int, sys.stdin.readline().split()))
b = list(map(int, sys.stdin.readline().split()))
a.sort(reverse=True)
b.sort(reverse=True)
threshold = (sum(a[i] for i in range(n)) + sum(a[i] for i in range(n // 2, n)) - sum(a[i] for i in range(n, n * 3 // 2))) * 1000 // (sum(b[i] for i in range(n)) + sum(b[i] for i in range(n // 2, n)))
print(threshold + 1)
