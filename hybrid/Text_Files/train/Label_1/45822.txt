import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n1 = input.nextInt(), n2 = input.nextInt(), k1 = input.nextInt(), k2 = input.nextInt(), dp[][][][] = new int[n1 + 3][n2 + 3][k1 + 3][k2 + 3];
        dp[0][0][0][0] = 1;
        for (int i = 0; i <= n1; i++) {
            for (int j = 0; j <= n2; j++) {
                for (int x = 0; x <= k1; x++) {
                    for (int y = 0; y <= k2; y++) {
                        if (i > 0) dp[i][j][x][y] = (dp[i][j][x][y] + dp[i - 1][j][Math.min(k1, x + 1)][y]) % 100000000;
                        if (j > 0) dp[i][j][x][y] = (dp[i][j][x][y] + dp[i][j - 1][x][Math.min(k2, y + 1)]) % 100000000;
                    }
                }
            }
        }
        System.out.println(dp[n1][n2][0][0]);
    }
}
