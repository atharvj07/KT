#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
#define fi first
#define se second
typedef pair<int,int>pi;
vector<pi>adj[N];
int r[N], vis[N], sub[N], ans[N];
void dfs(int u, int p) {
  vis[u]=1;
  sub[u]=1;
  for(auto [v,i]:adj[u]) {
    if(v==p) continue;
    if(!vis[v]) {
      dfs(v,u);
      sub[u]+=sub[v];
    }
    ans[i]+=sub[v];
  }
}
int main() {
  int n,k;
  cin>>n>>k;
  for(int i=1; i<=n; i++) { cin>>r[i]; }
  for(int i=1; i<=k; i++) {
    int x,y;
    cin>>x>>y;
    adj[x].emplace_back(y,i);
    adj[y].emplace_back(x,i);
  }
  for(int i=1; i<=n; i++) {
    if(!vis[i]) dfs(i,0);
  }
  for(int i=1; i<=n; i++) { cout<<ans[i]<<" "; } cout<<endl;
  return 0;
}
