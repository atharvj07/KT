#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin>>s;
    int n = s.size(),l,r;
    for(int i=1;i<n;++i){
        if(s[i]<s[i-1]){
            l=i-1;
            for(int j=i+1;j<n;++j){
                if(s[j]>=s[i-1]){
                    r=j-1;
                    break;
                }
            }
            if(r>=0){
                reverse(s.begin()+l,s.begin()+r+1);
                cout<<"YES\n"<<l+1<<" "<<r+1<<endl;
                return 0;
            }
        }
    }
    cout<<"NO\n";
    return 0;
}
