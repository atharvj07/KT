import sys
s=input()
b=input()
k=int(input())
ans=0;cur=0;bad=0;ls=[[-1]*26 for i in range(len(s)+1)]
for i in range(len(s)):
 ls[i+1]=ls[i].copy()
 if b[ord(s[i])-ord('a')]=='0':
  bad+=1
 ls[i+1][ord(s[i])-ord('a')]=i
 ans+=cur
 if ls[i][ord(s[i])-ord('a')]!=-1:
  cur-=i-ls[i][ord(s[i])-ord('a')]
  if bad-cur<=k:
   ans+=i-ls[i][ord(s[i])-ord('a')]
  else:
   bad-=1
 if bad<=k:
  cur+=1
print(ans)
