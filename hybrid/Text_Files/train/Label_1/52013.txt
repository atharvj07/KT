import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        scanner.close();
        for (int i = 1; i < k; i++) {
            for (int j = 1; j < k; j++) {
                System.out.printf("%d ", i * j);
            }
            System.out.println();
        }
    }
}
