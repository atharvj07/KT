from collections import Counter

s = input()
t = input()
n = len(s)
if s == t:
    print(0)
else:
    diff = []
    for i in range(n):
        if s[i] != t[i]:
            diff.append(i)
    if len(diff) % 2 == 1:
        print(-1)
    else:
        print(len(diff) // 2)
        for i in range(0, len(diff), 2):
            print(s[diff[i]], s[diff[i + 1]])
