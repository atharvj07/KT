import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] dp = new int[n][2];
        dp[0][0] = (a[0] % 2 == 0) ? 1 : 0;
        dp[0][1] = (a[0] % 2 == 1) ? 1 : 0;
        for (int i = 1; i < n; i++) {
            if (a[i] % 2 == 0) {
                dp[i][0] = dp[i - 1][0] + dp[i - 1][1];
                dp[i][1] = dp[i - 1][1];
            } else {
                dp[i][0] = dp[i - 1][0];
                dp[i][1] = dp[i - 1][0] + dp[i - 1][1];
            }
        }
        System.out.println(dp[n - 1][0]);
    }
}
