import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.stream.IntStream;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        while (t-- > 0) {
            int n = Integer.parseInt(br.readLine());
            int[] a1 = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            int[] a2 = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            boolean[] ok = new boolean[n + 1];
            for (int v : a1) ok[v] = true;
            boolean found = true;
            for (int i = 1; i <= n; i++) found &= ok[i];
            if (!found) {
                System.out.println(-1);
                continue;
            }
            Map<Integer, Integer> cnt1 = new HashMap<>();
            Map<Integer, Integer> cnt2 = new HashMap<>();
            for (int v : a1) cnt1.put(v, cnt1.getOrDefault(v, 0) + 1);
            for (int v : a2) cnt2.put(v, cnt2.getOrDefault(v, 0) + 1);
            int ans = 0;
            for (int i = 1; i <= n; i++)
                if (cnt1.getOrDefault(i, 0) != cnt2.getOrDefault(i, 0)) {
                    ans++;
                    cnt1.put(i, cnt1.getOrDefault(i, 0) - 1);
                    cnt2.put(i, cnt2.getOrDefault(i, 0) + 1);
                }
            System.out.println(ans);
            int[] res = IntStream.range(1, n + 1).filter(x -> cnt1.getOrDefault(x, 0) != cnt2.getOrDefault(x, 0)).toArray();
            for (int v : res) System.out.print(v + " ");
            System.out.println();
        }
    }
}
