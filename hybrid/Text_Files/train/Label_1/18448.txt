#include <iostream>
#include <vector>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n, q;
        std::cin >> n >> q;
        std::string s;
        std::cin >> s;
        std::vector<long long> sum(n);
        for (int i = 0; i < n; i++) {
            sum[i] = (s[i] == '+' ? 1 : -1);
            if (i > 0) sum[i] += sum[i - 1];
        }
        while (q--) {
            int l, r;
            std::cin >> l >> r;
            l--, r--;
            std::cout << (sum[r] - (l > 0 ? sum[l - 1] : 0) == 0 ? 0 : 1) << '\n';
        }
    }
    return 0;
}
