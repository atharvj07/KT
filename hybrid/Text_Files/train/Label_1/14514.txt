#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> x(n), y(n);
    unordered_set<int> us;
    for (int i = 0; i < n; i++) {
        cin >> x[i];
        us.insert(x[i]);
    }
    for (int i = 0; i < n; i++) {
        cin >> y[i];
        us.insert(y[i]);
    }
    cout << (us.size() & 1 ? "Koyomi" : "Karen") << endl;
    return 0;
}
