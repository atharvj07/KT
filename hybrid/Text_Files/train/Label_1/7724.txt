#include<iostream>
#include<vector>
#include<set>
using namespace std;

int main() {
  int n;
  cin >> n;
  int cnt = 0;
  vector<pair<int, int>> p(n+1);
  for(int i = 0; i <= n; i++) cin >> p[i].first >> p[i].second;
  set<pair<int, int>> s;
  s.insert(make_pair(p[0].first, p[0].second));
  for(int i = 1; i < n; i++) {
    int x1 = p[i].first, y1 = p[i].second, x2 = p[i+1].first, y2 = p[i+1].second;
    if(x1 == x2) {
      if(s.find(make_pair(x1-1, y1)) == s.end() && s.find(make_pair(x1+1, y1)) == s.end()) cnt++;
    } else {
      if(s.find(make_pair(x1, y1-1)) == s.end() && s.find(make_pair(x1, y1+1)) == s.end()) cnt++;
    }
    s.insert(make_pair(x1, y1));
  }
  cout << cnt << endl;
  return 0;
}
