#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> ans;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') {
      ans.push_back(0);
    } else {
      if (!ans.empty() && ans.back() == 0) {
        continue;
      } else {
        ans.push_back(1);
      }
    }
  }
  if (ans.empty()) ans.push_back(0);
  for (int i = 0; i < ans.size(); i++) {
    cout << ans[i];
  }
  cout << endl;
  return 0;
}
