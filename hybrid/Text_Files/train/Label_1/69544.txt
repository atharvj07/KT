import sys

N,Q = map(int,input().split())
A = list(map(int,input().split()))
B = []
for _ in range(Q): B.append(list(map(int,input().split())))

A.sort(reverse=True)
B.sort(key = lambda x:x[1]-x[0],reverse=True)

ans = 0
idx = 0

for l,r in B:
    while idx<N and A[idx]>0 and l<=r:
        if A[idx]<=r-l+1:
            ans += A[idx]**2
            r -= A[idx]
            idx += 1
        else:
            ans += (r-l+1)**2
            A[idx] -= r-l+1
            break
print(ans)
