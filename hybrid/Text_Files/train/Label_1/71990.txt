import sys

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
sum_b = [0] * n
sum_b[0] = b[0]
for i in range(1, n):
    sum_b[i] = sum_b[i - 1] + b[i]
b_sort = sorted(enumerate(b), key=lambda x: x[1])
ans = 0
for i in range(n):
    ans += a[i] * sum_b[b_sort[i][0]]
print(ans % 998244353)
