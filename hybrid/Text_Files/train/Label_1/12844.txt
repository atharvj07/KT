#include<bits/stdc++.h>
using namespace std;

const int MAXN = 1e5 + 1;

int par[MAXN];
int sz[MAXN];

int getp(int x) {
	return par[x] = (par[x] == x ? x : getp(par[x]));
}

void unite(int x, int y) {
	x = getp(x);
	y = getp(y);
	if (x == y) return;
	if (sz[x] < sz[y]) swap(x, y);
	sz[x] += sz[y];
	par[y] = x;
}

int main() {
	int n, m;
	cin >> n >> m;
	for (int i = 1; i <= n; i++) cin >> par[i], par[i] = i, sz[i] = 1;
	vector<pair<int, int>> edgelist;
	while (m--) {
		int x, y;
		cin >> x >> y;
		edgelist.push_back({x, y});
	}
	long long ans = 0;
	for (auto e : edgelist) {
		int x = e.first, y = e.second;
		if (getp(x) != getp(y)) {
			ans += min(par[getp(x)], par[getp(y)]);
			unite(x, y);
		}
	}
	cout << ans << '\n';
}
