#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> v(m + 1);
  for (int i = 1; i <= m; i++) v[i] = i;
  for (int i = 1; i < n; i++) {
    for (int j = i * 4, k = i * 2; j >= k; j--, k++) swap(v[j], v[k]);
  }
  for (int i = 1; i <= m; i++) cout << v[i] << ' ';
}
