import java.util.*;
import java.lang.*;

public class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    String s = scan.next();
    if (s.substring(s.length() - 1).equals("s")) {
      s += "es";
    } else {
      s += "s";
    }
    System.out.println(s);
  }
}
