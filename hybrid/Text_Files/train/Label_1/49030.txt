import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long max = -1;
        for (int i = 0; i < n; i++) {
            long a = in.nextInt();
            if (Math.sqrt(a) * Math.sqrt(a) != a) {
                max = Math.max(max, a);
            }
        }
        System.out.println(max);
    }
}
