#include<bits/stdc++.h>
using namespace std;
#define ll long long
int n;
ll a[100010], ans, pw2[33];
unordered_map<ll, ll> cnt;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        cnt[a[i]]++;
    }
    pw2[0] = 1;
    for (int i = 1; i <= 32; i++) pw2[i] = pw2[i - 1] * 2;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= 32; j++) {
            ll x = pw2[j] - a[i];
            if (x >= 0 && cnt[x]) {
                if (x == a[i]) ans += cnt[x] - 1;
                else ans += cnt[x];
            }
        }
    }
    cout << ans / 2 << endl;
    return 0;
}
