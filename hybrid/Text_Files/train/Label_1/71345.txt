import sys
input = sys.stdin.readline
d, n, m = map(int, input().split())
gas = [(0, 0)]
for _ in range(m):
    x, p = map(int, input().split())
    gas.append((x, p))
gas.append((d, 0))
if gas[1][0] <= n:
    ans = 0
    i = 0
    while i + 1 < len(gas):
        if gas[i][0] <= n:
            j = i + 1
            while j + 1 < len(gas) and gas[j + 1][0] <= n: j += 1
            ans += (gas[j][0] - gas[i][0]) * gas[i][1]
            n -= gas[j][0] - gas[i][0]
            i = j
        else: break
    if i == len(gas) - 1: print(ans)
    else: print(-1)
else: print(-1)
