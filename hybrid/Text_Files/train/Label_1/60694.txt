from sys import stdin
E=stdin.readline().strip()
P,M=map(int,stdin.readline().split())
def f(e,p,m,op):
 if e=='?': return 0,0
 if e.isdigit(): return int(e),0
 a,b=0,0
 for i in range(1,len(e),2):
  if e[i]=='+': a+=f(e[i-1:i+2],p-1,m,True)[0]
  else:b+=f(e[i-1:i+2],p,m-1,False)[0]
 b=max(b,1)
 if op:
  if p: return a+b,b
  else: return a,b
 else:
  if m: return a-b,a
  else: return a,a
print(f(E,P,M,True)[0])
