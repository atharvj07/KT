import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[a[i] - 1] = i;
        }
        int op = 0;
        for (int i = 0; i < n; i++) {
            if (i == b[i]) {
                continue;
            }
            op++;
            System.out.println(i + 1 + " " + (b[i] + 1));
            int t = a[i];
            a[i] = a[b[i]];
            a[b[i]] = t;
            b[t - 1] = b[i];
            b[a[i] - 1] = i;
        }
        System.out.println(op);
    }
}
