#include <bits/stdc++.h>
using namespace std;
#define int long long

int32_t main() {
  int n, m, q; cin >> n >> m >> q;
  string s, t; cin >> s >> t;
  vector<int> p(n + 5, -1);
  for (int i = m - 1, j = 0; i >= 0; i--) {
    if (t[i] == t[j]) p[i] = j, j++;
  }
  vector<int> cnt(n + 5);
  int occ = 0;
  for (int i = 0, j = 0; i < n; i++) {
    while (j >= 0 && s[i] != t[j]) j = p[j];
    if (++j == m) {
      occ--;
      j = p[j];
    }
    cnt[i] = occ;
    occ++;
  }
  while (q--) {
    int l, r; cin >> l >> r;
    cout << cnt[r] - (l > 0 ? cnt[l - 1] : 0) << '\n';
  }
  return 0;
}
