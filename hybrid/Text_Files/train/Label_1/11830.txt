#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, a, b, da, db;
    cin >> n >> a >> b >> da >> db;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; i++) {
      int u, v;
      cin >> u >> v;
      adj[u].push_back(v);
      adj[v].push_back(u);
    }
    int dist[n + 1];
    for (int i = 0; i <= n; i++) dist[i] = -1;
    dist[a] = 0;
    vector<int> q;
    q.push_back(a);
    while (!q.empty()) {
      int u = q.back();
      q.pop_back();
      for (int v : adj[u]) {
        if (dist[v] == -1) {
          dist[v] = dist[u] + 1;
          if (dist[v] <= da) q.push_back(v);
        }
      }
    }
    if (dist[b] != -1) {
      cout << "Alice" << endl;
      continue;
    }
    for (int i = 0; i <= n; i++) dist[i] = -1;
    dist[b] = 0;
    q.push_back(b);
    while (!q.empty()) {
      int u = q.back();
      q.pop_back();
      for (int v : adj[u]) {
        if (dist[v] == -1) {
          dist[v] = dist[u] + 1;
          if (dist[v] <= db) q.push_back(v);
        }
      }
    }
    if (dist[a] != -1) {
      cout << "Bob" << endl;
      continue;
    }
    cout << "Alice" << endl;
  }
  return 0;
}
