import sys

n, m = [int(x) for x in sys.stdin.readline().split()]
a = [int(x) for x in sys.stdin.readline().split()]
b = [int(x) for x in sys.stdin.readline().split()]

if any(x > 100 for x in a or b):
    print(-1)
    exit()

a.sort()
b.sort()

# Start with the minimum possible TL that satisfies conditions 1, 3, and 4.
# This is the maximum running time of the wrong solutions.
tl = max(b)

# Check if this TL satisfies condition 2.
for x in a:
    if 2 * x <= tl:
        print(tl)
        exit()

# If it doesn't, increase TL until it does.
for x in a[1:]:
    tl = x
    if 2 * x <= tl:
        print(tl)
        exit()

# If no TL can be found that satisfies all the conditions, print -1.
print(-1)
