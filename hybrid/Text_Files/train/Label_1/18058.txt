#include <algorithm>
#include <iostream>
using namespace std;
const int mxN = 1e5;
int n, a[mxN], b[mxN], ans;

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a, a + n);
    for (int i = 0; i < n; i++) {
        if (i && a[i] == a[i - 1]) { b[i] = b[i - 1]; }
        else { b[i] = i; }
    }
    for (int i = n - 1; ~i; i--) {
        if (i != n - 1 && a[i] == a[i + 1]) { continue; }
        ans += n - b[i];
    }
    cout << ans << '\n';
}
