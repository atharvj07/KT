import sys

n, m = map(int, sys.stdin.readline().split())
g = [[] for _ in range(n + 1)]
for _ in range(m):
    x, y = map(int, sys.stdin.readline().split())
    g[x].append(y)
    g[y].append(x)

vis = [0] * (n + 1)

def dfs(u, p):
    sz = 1
    mx = 0
    for v in g[u]:
        if v == p:
            continue
        t = dfs(v, u)
        sz += t
        mx = max(mx, t)
    return sz, mx

x, y = dfs(1, 1)

vis[1] = 1
ans = []
for v in g[1]:
    sz, mx = dfs(v, 1)
    if sz > n - ans - 1:
        ans.append(v)
    elif mx > n - ans - 1:
        ans.append(v)

print(max(vis) - 1)
print(*ans)
