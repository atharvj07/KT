#include<bits/stdc++.h>
#define ll long long
#define db double
#define pi acos(-1)
#define pb push_back
#define sz(x) (int)x.size()
#define all(x) x.begin(),x.end()
#define pii pair<int,int>
#define pll pair<ll,ll>
#define fast ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);
using namespace std;
int deg[200005],xorSum[200005];
vector<int>adj[200005];
int main()
{
    fast;
    int n; cin>>n;
    for(int i=0;i<n;i++)
    {
        int a,b; cin>>a>>b;
        deg[i]=a;
        xorSum[i]=b;
    }
    for(int i=0;i<n;i++)
    {
        for(int j=0;j<n;j++)
        {
            if(i==j) continue;
            if((deg[i]==1&&xorSum[i]==xorSum[j])||(deg[i]==2&&xorSum[i]==j)||(deg[i]>=3&&deg[j]==0&&xorSum[i]==xorSum[j]))
            {
                adj[i].pb(j);
                adj[j].pb(i);
                deg[i]--;
                deg[j]--;
                break;
            }
        }
    }
    int cnt=0;
    for(int i=0;i<n;i++)
        cnt+=(deg[i]>0);
    cout<<cnt/2<<endl;
    for(int i=0;i<n;i++)
    {
        for(auto x:adj[i])
        {
            if(i>x)
                continue;
            cout<<i<<" "<<x<<endl;
        }
    }
    return 0;
}
