#include <iostream>
#include <string>

int main() {
  int k;
  std::cin >> k;
  int num_digits = 1;
  int start = 1;
  while (k > num_digits * 9 * start) {
    k -= num_digits * 9 * start;
    num_digits++;
    start *= 10;
  }
  int num = start + (k - 1) / num_digits;
  int digit = (num - (start / 10)) % 10;
  std::cout << digit << std::endl;
  return 0;
}
