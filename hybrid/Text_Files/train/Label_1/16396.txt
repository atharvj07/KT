#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int k;
    cin >> k;
    string s;
    cin >> s;
    vector<bool> vis(k);
    for (int i = 0; i < k; i++) {
      if (s[i] == '3' || s[i] == '5' || s[i] == '7') {
        vis[i] = 1;
      }
    }
    int ans = 0;
    string t;
    for (int i = 0; i < k; i++) {
      if (!vis[i]) {
        ans++;
        if (t == "") {
          t += s[i];
        } else {
          t = s[i] + t;
        }
      }
    }
    cout << ans << endl;
    cout << t << endl;
  }
  return 0;
}
