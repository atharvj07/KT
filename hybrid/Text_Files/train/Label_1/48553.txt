import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] sticks = new int[12];
        for (int i = 0; i < sticks.length; i++) {
            sticks[i] = sc.nextInt();
        }
        Arrays.sort(sticks);
        boolean ok = true;
        for (int i = 0; i < 6; i++) {
            if (sticks[i] != sticks[6]) {
                ok = false;
            }
        }
        for (int i = 6; i < 12; i++) {
            if (sticks[i] != sticks[8]) {
                ok = false;
            }
        }
        System.out.println(ok ? "yes" : "no");
    }
}
