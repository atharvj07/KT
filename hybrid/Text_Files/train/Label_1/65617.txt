n, k = map(int, input().split())
arr = [list(input()) for _ in range(n)]
ans = 0
x, y = 0, 0

for i in range(n):
    for j in range(n):
        cnt1, cnt2 = 0, 0
        if arr[i][j] == '.':
            for p in range(k):
                if i + p < n and arr[i + p][j] != '#':
                    cnt1 += 1
                if j + p < n and arr[i][j + p] != '#':
                    cnt2 += 1
            if cnt1 == k or cnt2 == k:
                cnt = 0
                for p in range(k):
                    if i + p < n and arr[i + p][j] != '#':
                        cnt += 1
                    if j + p < n and arr[i][j + p] != '#':
                        cnt += 1
                if cnt > ans:
                    ans = cnt
                    x, y = i + 1, j + 1

print(x, y)
