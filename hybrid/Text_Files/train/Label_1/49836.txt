import java.io.*;
import java.util.*;

public class Main {
    static long mod = (long)1e15;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while(t-- > 0) {
            int n = sc.nextInt();
            int[] a = new int[n];
            int max = 0;
            for(int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
                max = Math.max(max, a[i]);
            }
            int[][] dp = new int[max + 1][n];
            for(int i = 0; i <= max; i++) {
                for(int j = 0; j < n; j++) {
                    dp[i][j] = -1;
                }
            }
            for(int i = 0; i < n; i++) {
                dp[a[i]][i] = 1;
            }
            int cnt = 0;
            for(int i = max - 1; i >= 1; i--) {
                for(int j = 0; j < n; j++) {
                    dp[i][j] = 0;
                    for(int k = 0; k < n; k++) {
                        if(a[k] > i) {
                            dp[i][j] = (dp[i][j] + dp[i + 1][k]) % (int)mod;
                        }
                    }
                    if(dp[i][j] > 0) {
                        cnt++;
                    }
                }
            }
            System.out.println(cnt);
        }
    }
}
