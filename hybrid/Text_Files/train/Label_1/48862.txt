import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
            }
            boolean ok = true;
            for (int i = 0; i < n - 1; i++) {
                if (a[i] != a[i + 1]) {
                    ok = false;
                    break;
                }
            }
            if (n == 2) ok = true;
            System.out.println(ok ? "YES" : "NO");
        }
    }
}
