#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<pair<int, vector<int>>> v(n);
        for (int i = 0; i < n; i++) {
            int a, b;
            cin >> a;
            v[i].first = a;
            while (a--) { cin >> b; v[i].second.push_back(b); }
        }
        sort(v.begin(), v.end());
        int ans = 0;
        for (auto x : v) {
            int mx = 0;
            for (auto y : x.second) mx = max(mx, y);
            ans = max(ans, mx + 1);
        }
        cout << ans << "\n";
    }
    return 0;
}
