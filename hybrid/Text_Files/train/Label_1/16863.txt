#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    sort(a.begin(), a.end());
    int l = 0, r = n - 1, ans = 0;
    while (l <= r) {
      int mid = (l + r) / 2;
      int sum = accumulate(a.begin(), a.begin() + mid, 0ll);
      if (sum * 2 <= mid * a[mid]) {
        ans = mid;
        l = mid + 1;
      } else
        r = mid - 1;
    }
    cout << n - ans << '\n';
  }
}
