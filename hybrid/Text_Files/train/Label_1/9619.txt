#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, s;
    cin >> n >> s;
    vector<pair<int, int>> p(n);
    for (auto &x : p) cin >> x.first >> x.second;
    sort(p.begin(), p.end());
    int ans = 0;
    for (auto x : p) {
        if (ans < x.second) ans = x.second;
        ans += s - x.first;
    }
    cout << ans << endl;
}
