import sys
def gcd(a, b):
    if a == 0:
        return b
    return gcd(b % a, a)
def play(x, y):
    if x < y:
        x, y = y, x
    if x % y != 0:
        return "Impossible"
    ans = ""
    while y != 1:
        while x % y != 0:
            x, y = x - y, y
        if x == y == 1:
            break
        ans += "B" * (x / y)
        x //= (x / y)
    ans += "A" * x
    return ans
x, y = map(int, sys.stdin.readline().strip().split())
ans = play(x, y)
if ans == "Impossible":
    print("Impossible")
else:
    print(ans)
