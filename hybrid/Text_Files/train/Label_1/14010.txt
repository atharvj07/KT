#include <iostream>
#include <algorithm>
#include <vector>
#include <unordered_map>

using namespace std;
using ll = long long;

const ll INF = 1e18;

struct Point {
  ll x, y;
  Point() : x(0), y(0) {}
  Point(ll x, ll y) : x(x), y(y) {}
  bool operator <(const Point& p) const {
    return make_pair(x, y) < make_pair(p.x, p.y);
  }
};

struct Rect {
  Point p1, p2;
  int id;
  Rect() : p1(Point()), p2(Point()), id(0) {}
  Rect(Point p1, Point p2, int id) : p1(p1), p2(p2), id(id) {}
  bool operator <(const Rect& r) const {
    return make_tuple(p1, p2) < make_tuple(r.p1, r.p2);
  }
};

typedef pair<Point, Point> Interval;

struct Event {
  Point p;
  bool is_start;
  int rect_id;
  Event() : p(Point()), is_start(false), rect_id(0) {}
  Event(Point p, bool is_start, int rect_id) : p(p), is_start(is_start), rect_id(rect_id) {}
  bool operator <(const Event& e) const {
    return make_pair(p.x, is_start) < make_pair(e.p.x, e.is_start);
  }
};

unordered_map<int, vector<Interval>> intervals;
vector<int> res;

int main() {
  int n;
  cin >> n;
  vector<Rect> rects(n);
  for (int i = 0; i < n; i++) {
    ll x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2;
    rects[i] = Rect(Point(x1, y1), Point(x2, y2), i);
  }
  sort(rects.begin(), rects.end());
  for (const auto& rect : rects) {
    intervals[rect.id].push_back(Interval(rect.p1, rect.p2));
    intervals[rect.id].push_back(Interval(Point(rect.p2.x, rect.p1.y), Point(rect.p1.x, rect.p2.y)));
  }
  for (auto& [id, intervals] : intervals) {
    sort(intervals.begin(), intervals.end());
    int cnt = 0;
    for (const auto& interval : intervals) {
      if (cnt % 2 == 0) {
        res.push_back(1);
      } else {
        res.push_back(2);
      }
      cnt++;
    }
  }
  cout << "YES" << endl;
  for (int i = 0; i < n; i++) {
    cout << res[i] << endl;
  }
  return 0;
}
