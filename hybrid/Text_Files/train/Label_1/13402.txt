#include <iostream>
#include <vector>
#include <string>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<std::string> p(n);
  for (auto &pp : p) {
    std::cin >> pp;
  }
  std::string t;
  std::cin >> t;
  int best = 0, worst = 0;
  for (int i = 0;; i++) {
    if (p[i] == t) {
      best = i + 1;
      break;
    }
    ++worst;
    if (worst > k) {
      worst += 5;
    }
  }
  for (int i = n - 1;; i--) {
    if (p[i] == t) {
      worst += n - 1 - i;
      break;
    }
    ++worst;
    if (worst > k) {
      worst += 5;
    }
  }
  std::cout << best << ' ' << worst << std::endl;
  return 0;
}
