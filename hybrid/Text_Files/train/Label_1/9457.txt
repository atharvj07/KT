#include <bits/stdc++.h>
using namespace std;
int main() {
  int n; cin >> n;
  if (n < 3) { cout << -1 << endl; return 0; }
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    int x, y; cin >> x >> y;
    v[i] = {x, y};
  }
  sort(v.begin(), v.end());
  int ans = 1e9;
  for (int i = 0; i < n; i++) {
    int l = lower_bound(v.begin(), v.end(), make_pair(v[i].first + 1, 0)) - v.begin();
    if (l < n) {
      int r = lower_bound(v.begin(), v.end(), make_pair(v[l].first + 1, 0)) - v.begin();
      if (r < n) ans = min(ans, v[i].second + v[l].second + v[r].second);
    }
  }
  if (ans != 1e9)
    cout << ans << endl;
  else
    cout << -1 << endl;
}
