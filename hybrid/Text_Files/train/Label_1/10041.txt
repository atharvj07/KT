#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for(int i = 0; i < n; i++)
#define all(x) x.begin(), x.end()
using ll = long long;
#define int ll
const ll INF = 1e18;
const ll mod = 1e9 + 7;
const int maxn = 1e5 + 5;

int32_t main() {
    int n, d; cin >> n >> d;
    map<int, int> dp;
    map<int, int> fac;
    dp[INF] = 0;
    int ans = 0;
    rep(i, n) {
        int m, s; cin >> m >> s;
        if (dp[m + d] + m >= dp[m]) {
            dp[m] = dp[m + d] + m;
            fac[m] = fac[m + d] + s;
        }
        ans = max(ans, fac[m]);
    }
    cout << ans << '\n';
}
