#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> dp(n + 1, 0);
  for (int i = n; i >= 2; i--) {
    for (int j = i; j <= n; j++) {
      if ((i ^ j) <= n) { dp[i ^ j]++; }
    }
  }
  for (int i = n; i >= 3; i--) {
    for (int j = 1; j <= n; j++) { dp[i] += dp[j]; }
  }
  cout << dp[n] / 6 << endl;
  return 0;
}
