#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string a, s;
    cin >> a >> s;
    int n = a.length(), m = s.length();
    if (n == m) {
      bool ok = true;
      for (int i = 0; i < n; i++) {
        int x = a[i] - '0', y = s[i] - '0';
        if (x + y > 9) {
          ok = false;
          break;
        }
      }
      if (ok) {
        cout << s << endl;
      } else {
        cout << -1 << endl;
      }
    } else {
      cout << -1 << endl;
    }
  }
  return 0;
}
