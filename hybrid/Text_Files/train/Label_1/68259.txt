def can_prepare_contest(h, m, s, t1, t2):
    # Convert all times to minutes to have a common scale
    # Hour hand moves 0.5 degrees per minute (360 deg / 12 hours / 60 minutes)
    # Minute hand moves 6 degrees per minute (360 deg / 60 minutes)
    # Second hand moves 6 degrees per second (360 deg / 60 seconds)
    hour_pos = (h % 12) * 30 + m * 0.5 + s * (0.5 / 60)
    minute_pos = m * 6 + s * 0.1
    second_pos = s * 6

    # Convert t1 and t2 to degrees on the clock
    t1_pos = (t1 % 12) * 30
    t2_pos = (t2 % 12) * 30

    # Sort the positions of the hands and the start/end points
    points = sorted([hour_pos, minute_pos, second_pos, t1_pos, t2_pos])

    # Find index of t1 and t2 after sorting
    t1_index = points.index(t1_pos)
    t2_index = points.index(t2_pos)

    # Check if t1 and t2 are adjacent in the sorted list, considering wrap-around
    if abs(t1_index - t2_index) == 1 or abs(t1_index - t2_index) == 4:
        return "YES"
    else:
        return "NO"

# Read input
h, m, s, t1, t2 = map(int, input().split())

# Print output
print(can_prepare_contest(h, m, s, t1, t2))
