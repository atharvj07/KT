#include <iostream>
#include <vector>
using namespace std;

bool are_overlapping (int d1, int m1, int d2, int m2) {
    return (m1 == m2 && d1 <= d2) || (m1 < m2);
}

int main() {
    int n, days[12] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

    cin >> n;

    vector< pair<int, int> > ends(n, {0, 0});
    vector< pair<int, int> > starts(n, {0, 0});

    for (int i = 0; i < n; i++) {
        int d, m, p, t;
        cin >> m >> d >> p >> t;

        starts[i].first = m, starts[i].second = d;
        ends[i].first = m, ends[i].second = d + t - 1;

        if (m == 2 && d + t - 1 > days[m - 1]) {
            ends[i].second -= days[m - 1];
            ends[i].first++;
        } else if (d + t - 1 > days[m - 1]) {
            ends[i].second -= days[m - 1];
            ends[i].first++;

            int j = m;
            while (d + t - 1 > days[j - 1]) {
                d -= days[j - 1];
                j++;
                ends[i].first++;
            }
        }

        days[m - 1] = d;
    }

    vector<int> needed(366, 0);

    for (int i = 0; i < n; i++) {
        for (int j = starts[i].first * 31 + starts[i].second; j <= ends[i].first * 31 + ends[i].second; j++) {
            needed[j] += 1;
        }
    }

    int ans = 0;

    for (int j = 0; j < 366; j++) {
        ans = max(ans, needed[j]);
    }

    cout << ans;
    return 0;
}
