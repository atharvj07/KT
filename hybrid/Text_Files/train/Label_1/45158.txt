import java.util.Arrays;
import java.util.Scanner;

class Main {
    static int[][] forest;
    static int[][] moves;
    static int r, c, minBattles, ex, ey;

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        String[] rc = scan.nextLine().split(" ");
        r = Integer.parseInt(rc[0]);
        c = Integer.parseInt(rc[1]);

        forest = new int[r][c];
        moves = new int[r][c];

        for (int i = 0; i < r; i++) {
            String row = scan.nextLine();
            for (int j = 0; j < c; j++) {
                forest[i][j] = row.charAt(j) == 'T' ? -1 : row.charAt(j) - '0';
                if (forest[i][j] == 1) {
                    moves[i][j] = -2;
                } else if (forest[i][j] > 0) {
                    moves[i][j] = -1;
                } else if (row.charAt(j) == 'S') {
                    moves[i][j] = 0;
                } else if (row.charAt(j) == 'E') {
                    moves[i][j] = 1;
                    ex = i;
                    ey = j;
                }
            }
        }

        minBattles = Integer.MAX_VALUE;
        floodFill(0, 0, 0);
        System.out.println(minBattles);

        scan.close();
    }

    static int[] dx = {0, 1, 0, -1};
    static int[] dy = {1, 0, -1, 0};

    static void floodFill(int x, int y, int battles) {
        if (x == ex && y == ey) {
            minBattles = Math.min(minBattles, battles);
            return;
        }

        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i];
            int ny = y + dy[i];

            if (nx < 0 || nx >= r || ny < 0 || ny >= c || moves[nx][ny] == -2 || (moves[nx][ny] >= 0 && moves[nx][ny] > moves[x][y])) {
                continue;
            }

            int newBattles = battles;
            if (moves[nx][ny] == -1) {
                newBattles++;
            }

            moves[nx][ny] = moves[x][y] + 1;
            floodFill(nx, ny, newBattles);
            moves[nx][ny] = moves[x][y] - 1;
        }
    }
}
