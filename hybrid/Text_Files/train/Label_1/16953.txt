#include <bits/stdc++.h>
using namespace std;
const int N = 2e5+5;
int n,m,k,x[N],y[N],X[N],Y[N],ans,vis[N];

int main(){
  ios_base::sync_with_stdio(0), cin.tie(0);
  int t;
  cin >> t;
  while(t--){
    cin >> n >> m >> k;
    for(int i=0;i<=n;i++) X[i]=0, vis[i]=0;
    for(int i=0;i<=m;i++) Y[i]=0;
    for(int i=0;i<n;i++){
      cin >> x[i];
      X[i+1]=x[i];
    }
    for(int i=0;i<m;i++){
      cin >> y[i];
      Y[i+1]=y[i];
    }
    for(int i=0;i<k;i++){
      cin >> x[i] >> y[i];
      int j=-1,jj=-1;
      for(int l=1;l<=n;l++){
        if(x[i]==X[l]){
          if(j!=-1) j=l-1;
          else j=l;
        }
      }
      for(int l=1;l<=m;l++){
        if(y[i]==Y[l]){
          if(jj!=-1) jj=l-1;
          else jj=l;
        }
      }
      vis[j]+=!vis[jj];
      vis[jj]+=!vis[j];
    }
    ans=0;
    for(int i=0;i<k;i++) if(vis[i]%2) ans+=vis[i]-1;
    ans/2;
    cout << ans << endl;
  }
}
