import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[2*n - 1];
        for (int i = 0; i < 2*n - 1; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += Math.abs(a[i]);
        }
        for (int i = 2*n - 2; i >= n; i--) {
            sum += a[i];
        }
        System.out.println(sum);
    }
}
