#include <iostream>

int main() {
  long long int p, n;
  int counter = 0;
  std::cin >> n;

  for (p = 2; p * p <= n; p++) {
    if (n % p == 0) {
      while (n % p == 0) {
        n = n / p;
        counter++;
      }
    }
  }
  if (n > 1) { counter++; }

  std::cout << counter << std::endl;
  return 0;
}
