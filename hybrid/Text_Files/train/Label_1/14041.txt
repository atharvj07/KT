#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<double> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  double sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i];
  }

  if (sum > 0.000001 || sum < -0.000001) {
    cout << "ERROR: Sum of numbers is not equal to 0" << endl;
    return 0;
  }

  vector<int> b(n);
  for (int i = 0; i < n; i++) {
    if (a[i] >= 0) {
      b[i] = ceil(a[i]);
    } else {
      b[i] = floor(a[i]);
    }
  }

  for (int i = 0; i < n; i++) {
    cout << b[i] << endl;
  }

  return 0;
}
