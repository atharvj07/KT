#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, l;
    cin >> n >> l;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    double d = max(a[0], l - a[n - 1]);
    for (int i = 1; i < n; i++) {
        d = max(d, (a[i] - a[i - 1]) / 2.0);
    }
    cout.precision(9);
    cout << d << endl;
}
