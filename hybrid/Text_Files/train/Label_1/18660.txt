#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int t; cin >> t;
  while (t--) {
    int n; cin >> n;
    vector<int> a(n), b(n);
    for (int &x : a) cin >> x;
    for (int &x : b) cin >> x;
    if (a == b) cout << "YES" << endl;
    else {
      sort(a.rbegin(), a.rend());
      sort(b.rbegin(), b.rend());
      int d = b.front() - a.front();
      if (d <= 0) cout << "NO" << endl;
      else {
        int cnt = 0;
        for (int x : a) { if (x == a.front()) cnt++; }
        if (cnt >= d) cout << "YES" << endl;
        else cout << "NO" << endl;
      }
    }
  }
}
