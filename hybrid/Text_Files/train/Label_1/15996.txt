#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  int ans = 0;
  vector<string> result;

  int ones = 0;
  string cur;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') {
      if (ones > 0) {
        cur += '0';
      } else {
        if (!cur.empty()) {
          result.push_back(cur);
          ans++;
        }
        cur = "0";
        ones = 1;
      }
    } else {
      if (ones == 0) {
        if (!cur.empty()) {
          result.push_back(cur);
          ans++;
        }
        cur = "1";
        ones = -1;
      } else {
        cur += '1';
      }
    }
  }
  if (!cur.empty()) {
    result.push_back(cur);
    ans++;
  }

  cout << ans << endl;
  for (int i = 0; i < ans; i++) {
    cout << result[i] << " ";
  }
  cout << endl;

  return 0;
}
