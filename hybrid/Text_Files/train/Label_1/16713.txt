#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 998244353;
const int maxn = 1e6 + 10;
vector<int> fac(maxn), invfac(maxn);
vector<int> arr;
int n;
int quickpow(int a,int b){
    if(b == 0) return 1;
    int res = quickpow(a,b/2);
    res = res * res % mod;
    if(b % 2) res = res * a % mod;
    return res;
}
int nCr(int n,int r){
    if(n < r || n < 0 || r < 0) return 0;
    return 1ll * fac[n] * invfac[n - r] % mod * invfac[r] % mod;
}
int nPr(int n,int r){
    if(n < r || n < 0 || r < 0) return 0;
    return 1ll * fac[n] * invfac[n - r] % mod;
}
void preprocess(){
    fac[0] = 1;
    for(int i = 1;i < maxn;++i) fac[i] = fac[i - 1] * i % mod;
    invfac[0] = 1;
    for(int i = 1;i < maxn;++i) invfac[i] = quickpow(fac[i],mod - 2);
}
void solve(){
    cin >> n;
    int sum = (1ll * n * (n + 1) / 2) % mod;
    arr.resize(n * n);
    for(int i = 1;i <= n * n;++i){
        arr[i - 1] = i;
    } 
    do{
        vector<int> now;
        int cnt = 0;
        for(int j = 0;j < n * n;++j){
            int add = 0;
            if(cnt + n - 1 >= j) add = 1;
            if(add) now.push_back(arr[j]);
            cnt += add;
            if(cnt == n){
                if(accumulate(now.begin(),now.end(),0ll) == sum){
                    cout << nCr(n * n - n,n - 1) % mod << "\n";
                    return;
                }
                now.clear();
                cnt = 0;
            }
        }
    }while(next_permutation(arr.begin(),arr.end()));
}
signed main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    preprocess();
    solve();
}
