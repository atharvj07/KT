int merge_func(int arr[],int l,int mid,int r,int& ans)
{
    int i,j,k,inv_count = 0;
    int n1 = mid - l + 1;
    int n2 = r - mid;

    int L[n1],R[n2];
    for(i = 0;i<n1;i++)
        L[i] = arr[l+i];
    for(j = 0;j<n2;j++)
        R[j] = arr[mid+1+j];

    i = 0;j=0;k=l;
    while(i<n1 && j<n2)
    {
        if(L[i]<=R[j])
        {
            arr[k] = L[i];
            i++;
        }
        else
        {
            arr[k] = R[j];
            j++;
            inv_count+=n1-i; // If current element of right side is smaller than current element of left side than all the remaining elements of left side is greater than current element of right side.
        }
        k++;
    }
    while(i<n1){
        arr[k] = L[i];
        k++;i++;
    }
    while(j<n2){
        arr[k] = R[j];
        k++;j++;
    }
    ans+=inv_count;
    return inv_count;
}

int merge_sort(int arr[],int l,int r,int& ans)
{
    int mid,i,j,k,inv_count=0;
    if(l<r)
    {
        mid = l + (r-l)/2;
        ans+=merge_sort(arr,l,mid,inv_count); 
        ans+=merge_sort(arr,mid+1,r,inv_count);
        ans+=merge_func(arr,l,mid,r,inv_count);
    }
    return inv_count;
}
int main()
{
    int n,k,i,j,ans=0;
    cin>>n>>k;
    int arr[n];
    for(i=0;i<n;i++)
    {
        arr[i]=i+1;
    }
    k=k+1;
    if(k%2!=0)
    {
        cout<<"-1"<<endl;
        return 0;
    }
    ans+=merge_sort(arr,0,n-1,ans); 
    if(ans==k)
    {
        for(i=0;i<n;i++)
        {
            cout<<arr[i]<<" ";
        }
        cout<<endl;
    }
    else
        cout<<"-1"<<endl;

    return 0;
}
