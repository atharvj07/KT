#include <bits/stdc++.h>

using namespace std;

int n;
int a[300005], b[300005];
vector<int> ans;
bool vis[300005];

void dfs(int x) {
  if (x == 0) {
    reverse(ans.begin(), ans.end());
    cout << ans.size() << "\n";
    for (int i : ans) cout << i << " ";
    exit(0);
  }
  vis[x] = 1;
  for (int i = x - 1; i >= max({0, x - b[x], x - a[x]}); i--) {
    if (!vis[i]) {
      ans.push_back(i);
      dfs(i);
      ans.pop_back();
    }
  }
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> b[i];
  dfs(n);
  cout << "-1";
  return 0;
}
