import math

n, k = map(int, input().split())
if n > k * (k + 1) / 2:
    print("NO")
else:
    print("YES")
    arr = [1] * k
    rem = n - 1
    i = k - 1
    while rem > 0:
        if rem <= i:
            arr[i] += rem
            rem = 0
        else:
            rem -= i
            arr[i] = i
            i -= 1
    print(*arr)
