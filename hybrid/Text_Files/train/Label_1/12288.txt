#include<iostream>
#include<bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll mod=1e9+7;
const ll maxInt=1e18;
const ll minInt=-1e18;

int main(){
	int l, ans=0;
	cin>>l;
	while(l--){
		string s;
		cin>>s;
		if(s=="add") ans++;
		else if(s!="end"){
			int num=0;
			for(int i=3; i<s.size(); i++) num=num*10+(s[i]-'0');
			ans += (num-1)*ans;
			if(ans>=mod) ans%=mod;
		}
		if(ans==0) ans=1;
	}
	cout<<(ans==0?"OVERFLOW!!!":to_string(ans));
	return 0;
}
