MOD = int(1e9)+7
def count_occurrences(s, t):
    n = len(s)
    m = len(t)

    occurrences = [0] * n

    i = 0
    j = 0
    while i < n:
        if s[i] == t[j]:
            j += 1
            if j == m:
                occurrences[i - m + 1] = 1
                j -= 1
        i += 1
    for i in range(1, n):
        occurrences[i] += occurrences[i - 1]
    return occurrences

def count_subsequences(s, t, occurrences):
    n = len(s)
    m = len(t)

    dp = [[0] * n for _ in range(n + 1)]

    for i in range(n - 1, -1, -1):
        for j in range(i):
            dp[i][j] = dp[i + 1][j]
            if occurrences[i] > occurrences[j]:
                dp[i][j] += dp[i + 1][occurrences[i]]
                dp[i][j] %= MOD
    return dp[0][0]

def solve(s, t):
    occurrences = count_occurrences(s, t)
    subsequences = count_subsequences(s, t, occurrences)
    return subsequences

s = input()
t = input()
print(solve(s, t))
