#include <iostream>
#include <cmath>
using namespace std;
int gcd(int a, int b) { if (b == 0) { return a; } return gcd(b, a % b); }
int main() {
    int x, y, n; cin >> x >> y >> n;
    int a = 0, b = 1;
    for (int i = 1; i <= n; i++) {
        int num = (long long)i * x / y;
        if (abs(num * y - x * i) < abs(a * y - b * x) || (abs(num * y - x * i) == abs(a * y - b * x) && (i < b || (i == b && num < a)))) {
            a = num;
            b = i;
        }
    }
    cout << a << "/" << b << endl;
    return 0;
}
