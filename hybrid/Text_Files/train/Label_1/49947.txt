import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Integer, Integer> bitLen = new HashMap<>();
        for (int i = 0; i < n; i++) {
            int bitmask = 0;
            String s = sc.next();
            for (int j = 0; j < s.length(); j++) {
                bitmask |= (1 << (s.charAt(j) - '0'));
            }
            if (!bitLen.containsKey(bitmask)) {
                bitLen.put(bitmask, s.length());
            } else if (bitLen.get(bitmask) != s.length()) {
                System.out.println(Math.min(bitLen.get(bitmask), s.length()));
                return;
            }
        }
        System.out.println(0);
    }
}
