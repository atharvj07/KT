#include <iostream>
#include <vector>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin >> n;
    vector<vector<int>> v(101, vector<int>(101, 0));
    for (int i = 0; i < n; i++) {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        for (int j = x1; j <= x2; j++) {
            v[j][y1]++;
            v[j][y2 + 1]--;
        }
    }
    int sum = 0;
    for (int i = 1; i <= 100; i++) {
        for (int j = 1; j <= 100; j++) {
            v[i][j] += v[i - 1][j];
            sum += v[i][j];
        }
    }
    cout << sum << '\n';
}
