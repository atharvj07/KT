import java.util.*;

public class Main {
    static int n, m, k;
    static int[] a, b, c;
    // dp[i][j][l] means the maximum score when at castle i, have j warriors, and l is the current defended
    // castle (0 means not defended).
    static int[][][] dp;
    static ArrayList<Integer>[] portals;
    static boolean[][] portalUsed;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        a = new int[n + 1];
        b = new int[n + 1];
        c = new int[n + 1];
        portals = new ArrayList[n + 1];
        portalUsed = new boolean[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
            c[i] = sc.nextInt();
            portals[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            portals[u].add(v);
            portalUsed[u][v] = true;
        }
        dp = new int[n + 1][k + 1][n + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                Arrays.fill(dp[i][j], -1);
            }
        }
        System.out.println(dfs(1, k, 0));
    }

    static int dfs(int i, int warriors, int defended) {
        if (i > n) {
            // All castles have been captured
            return 0;
        }
        if (dp[i][warriors][defended] != -1) {
            return dp[i][warriors][defended];
        }
        int ans = 0;
        // Case 1: Capture the current castle
        if (warriors >= a[i]) {
            // Defend the current castle
            if (defended != i) {
                ans = Math.max(ans, dfs(i + 1, warriors - 1, i) + c[i]);
            }
            // Don't defend the current castle
            ans = Math.max(ans, dfs(i + 1, warriors - 1, defended));
            // Use portals to defend other castles
            for (int j : portals[i]) {
                if (!portalUsed[i][j] && warriors > 1) {
                    ans = Math.max(ans, dfs(i + 1, warriors - 2, j) + c[j]);
                    portalUsed[i][j] = true;
                }
            }
        }
        // Case 2: Don't capture the current castle
        ans = Math.max(ans, dfs(i + 1, warriors + b[i], defended));
        dp[i][warriors][defended] = ans;
        return ans;
    }
}
