#include <iostream>
#include <vector>
#include <algorithm>

#define ll long long

const int mod = 1e9 + 7;
const int maxn = 1e6 + 10;

int n, m, a[maxn];
std::vector<ll> l, r;
std::vector<std::pair<ll, int>> s;

ll fastpow(ll x, ll y) {
    if (y == 0) return 1;
    ll res = fastpow(x, y >> 1);
    res = (res * res) % mod;
    if (y & 1) res = (res * x) % mod;
    return res;
}

int main() {
    ll x, y, l1, r1;
    std::cin >> x >> y >> l1 >> r1;
    for (ll p = 2; p * p <= x; p++) {
        if (x % p) continue;
        int cnt = 0;
        while (x % p == 0) {
            x /= p;
            cnt++;
        }
        l.push_back(fastpow(p, cnt));
        r.push_back(fastpow(p, cnt + 1) - 1);
    }
    if (x > 1) {
        l.push_back(x);
        r.push_back(x * 2 - 1);
    }
    for (ll p = 2; p * p <= y; p++) {
        if (y % p) continue;
        int cnt = 0;
        while (y % p == 0) {
            y /= p;
            cnt++;
        }
        l.push_back(fastpow(p, cnt));
        r.push_back(fastpow(p, cnt + 1) - 1);
    }
    if (y > 1) {
        l.push_back(y);
        r.push_back(y * 2 - 1);
    }
    for (int i = 0; i < l.size(); i++) {
        s.push_back(std::make_pair(r[i], i));
    }
    sort(s.begin(), s.end());
    int ans = 0, las = -1;
    for (int i = 0; i < s.size(); i++) {
        if (s[i].first < l1) {
            continue;
        }
        if (s[i].second == las) {
            continue;
        }
        las = s[i].second;
        if (r1 >= s[i].first) {
            ans++;
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
