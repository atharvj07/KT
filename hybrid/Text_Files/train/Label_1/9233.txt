#include <bits/stdc++.h>
using namespace std;
int a[100005], b[100005], c[100005];
int main() {
  int n, m;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  cin >> m;
  for (int i = 1; i <= m; i++) {
    int x, y;
    cin >> x >> y;
    c[max(x, b[x])] = max(c[max(x, b[x])], y);
    b[x] = x;
  }
  for (int i = 1; i <= n; i++)
    cout << max(a[i], c[i]) << " ";
  return 0;
}
