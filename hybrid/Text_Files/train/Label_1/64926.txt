n, m = map(int, input().split())
d = [[] for i in range(m+1)]
for i in range(n):
    s, r = map(int, input().split())
    d[s].append(r)
ans = 0
for i in range(1, m+1):
    if len(d[i]) > 0:
        d[i].sort()
        k = len(d[i])//2
        ans += sum(d[i][k:])
if ans > 0:
    print(ans)
else:
    print(0)
