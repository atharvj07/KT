#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m, d;
    cin >> n >> m >> d;
    vector<pair<int, int>> physics, informatics;
    for (int i = 0; i < n; i++) {
        int c, w;
        cin >> c >> w;
        physics.push_back({c, w});
    }
    for (int i = 0; i < m; i++) {
        int c, w;
        cin >> c >> w;
        informatics.push_back({c, w});
    }
    sort(physics.begin(), physics.end());
    sort(informatics.begin(), informatics.end());
    int ans = 0, width = 0;
    bool hasPhysics = false, hasInformatics = false;
    for (int i = 0, j = 0; i < n || j < m;) {
        if ((j == m || (i < n && physics[i].first >= informatics[j].first)) && !hasPhysics) {
            if (width + physics[i].second > d) break;
            ans += physics[i].first;
            width += physics[i].second;
            hasPhysics = true;
            i++;
        } else if ((i == n || (j < m && informatics[j].first >= physics[i].first)) && !hasInformatics) {
            if (width + informatics[j].second > d) break;
            ans += informatics[j].first;
            width += informatics[j].second;
            hasInformatics = true;
            j++;
        } else break;
    }
    cout << ans << endl;
    return 0;
}
