import java.util.*;
import java.io.*;

class Main {

    // number of people
    private static int n;

    // total weight capacity of the boat
    private static int k;

    // to store the weights of the people
    private static int[] w;

    // to store the minimum number of times the boat needs to cross the river
    private static int minRides;

    // to store the number of ways to transport everybody to the other side in the minimum number of boat rides
    private static int numWays;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = br.readLine().split(" ");
        n = Integer.parseInt(parts[0]);
        k = Integer.parseInt(parts[1]);

        w = new int[n];
        String[] weights = br.readLine().split(" ");
        for (int i = 0; i < n; ++i) {
            w[i] = Integer.parseInt(weights[i]);
        }

        // if the sum of the weights of the people is greater than the total weight capacity of the boat, then it is impossible to transport everybody to the other side
        int sum = 0;
        for (int weight : w) {
            sum += weight;
        }
        if (sum > k) {
            System.out.println(-1);
            System.out.println(0);
            return;
        }

        // sort the weights in ascending order
        Arrays.sort(w);

        // calculate the minimum number of rides required
        minRides = 0;
        int weight = 0;
        for (int i = n - 1; i >= 0; --i) {
            if (weight + w[i] <= k) {
                weight += w[i];
            } else {
                ++minRides;
                weight = w[i];
            }
        }

        // calculate the number of ways to transport everybody to the other side in the minimum number of boat rides
        numWays = 1;
        weight = 0;
        int cnt50 = 0;
        for (int i = n - 1; i >= 0; --i) {
            if (weight + w[i] <= k) {
                weight += w[i];
                if (w[i] == 50) {
                    ++cnt50;
                }
            } else {
                numWays = (numWays * (cnt50 + 1)) % 1000000007;
                cnt50 = 0;
                weight = w[i];
            }
        }

        System.out.println(minRides);
        System.out.println(numWays);
    }
}
