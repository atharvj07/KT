#include <algorithm>
#include <cstdio>
#include <vector>
using namespace std;
int main() {
  int n;
  scanf("%d", &n);
  vector<long long> a(n);
  for (long long &x : a) { scanf("%lld", &x); }
  long long Sum = 0;
  for (const long long &x : a) { Sum += x; }
  vector<long long> nice;
  for (int i = 0; i < n; i++) {
    if (Sum - a[i] == a[i]) { nice.push_back(i + 1); }
  }
  printf("%d\n", nice.size());
  for (const long long &x : nice) { printf("%lld ", x); }
  return 0;
}
