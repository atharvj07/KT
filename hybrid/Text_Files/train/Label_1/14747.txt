#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int a[105][105], vert[105][105], horz[105][105];
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  for (int i = 0; i < n; i++) {
    cin >> s;
    for (int j = 0; j < n; j++) a[i][j] = s[j] == '#' ? -1 : 0;
  }
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = 0; j < n; j++) {
      if (a[i][j] >= 0) { vert[i][j] = ++cnt; }
      if (a[i][j] != 0) { cnt = 0; }
      if (a[j][i] >= 0) { horz[j][i] = ++cnt; }
      if (a[j][i] != 0) { cnt = 0; }
    }
  }
  int ansr = 0, ansc = 0, ansv = 0, ansh = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (k <= vert[i][j] && k <= horz[i][j] &&
          vert[i][j] + horz[i][j] - 1 > ansv + ansh) {
        ansr = i + 1;
        ansc = j + 1;
        ansh = horz[i][j];
        ansv = vert[i][j];
      }
    }
  }
  cout << ansr << " " << ansc << endl;
}
