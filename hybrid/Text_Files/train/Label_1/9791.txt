#include<bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (s[i] != s[i - 1]) ans++, i++;
        }
        if (s[n - 1] != s[0]) ans++;
        cout << ans << endl;
    }
}
