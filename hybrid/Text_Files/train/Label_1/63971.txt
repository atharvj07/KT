from collections import Counter
n = int(input())
d = [Counter(input()) for _ in range(n)]
k = 0
while True:
    k += 1
    for i in d:
        for j in d:
            if sum((i+Counter(j)).values()) <= 6+2*k:
                break
        else:
            continue
        break
    else:
        break
print(k)
