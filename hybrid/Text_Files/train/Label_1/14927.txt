#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> c(n), a(m);
  for (int i = 0; i < n; i++) cin >> c[i];
  for (int i = 0; i < m; i++) cin >> a[i];
  int ans = 0, j = 0;
  for (int i = 0; i < n; i++) {
    while (j < m && a[j] < c[i]) j++;
    if (j < m) ans++, j++;
  }
  cout << ans << endl;
  return 0;
}
