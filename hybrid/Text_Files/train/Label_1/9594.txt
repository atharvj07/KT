#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define ull unsigned long long
#define IOS ios_base::sync_with_stdio(false);cin.tie(NULL);cout.tie(NULL);
#define endl "\n"
#define pb push_back
#define mod 1000000007
ll power(ll x,  ll y){
    if (y == 0) return 1;
    if(y == 1)  return x;
    ll ans = power(x,y/2);
    ans*=ans;
    if(y % 2 == 1) ans*=x;
    return ans % mod;
}
int main(){
    IOS
    ll n,ans;
    cin>>n;
    vector<pair<ll,ll>>a;
    map<pair<ll,ll>,ll>mp;
    for(ll i=0;i<n;i++){
        ll x,y;
        cin>>x>>y;
        a.pb({x,y});
        mp[{x,y}]++;
    }
    ans=0;
    for(ll i=0;i<n;i++){
        for(ll j=i+1;j<n;j++){
            ll x1=a[i].first,y1=a[i].second,x2=a[j].first,y2=a[j].second;
            ll arx=(x2-x1),ary=(y2-y1);
            ans+=mp[{x1+ary,y1-arx}]*mp[{x2-ary,y2+arx}]*1LL;
        }
    }
    cout<<(ans/2)<<endl;
    return 0;
}
