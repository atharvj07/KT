#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, cnt = 0;
  cin >> n;
  vector<int> freq(8);
  while (n--) {
    string s;
    cin >> s;
    for (int i = 0; i < 7; i++) {
      if (s[i] == '1') freq[i + 1]++;
    }
  }
  for (int i = 1; i <= 7; i++) {
    if (freq[i] > 0) cnt++;
  }
  cout << cnt;
}
