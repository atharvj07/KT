#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const int max_n = 2e5 + 6;
vector<ll> pfx(max_n), a(max_n);

int main() {
  ll n, k, ans = 0, sum = 0;
  cin >> n >> k;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) {
    sum += a[i];
    pfx[i] = sum;
  }
  while (ans < n && pfx[ans] < k) {
    sum = 0;
    for (int i = ans + 1; i < n; i++) {
      sum += a[i];
      pfx[i] = pfx[ans] + sum;
    }
    ans++;
  }
  cout << ans << endl;
  return 0;
}
