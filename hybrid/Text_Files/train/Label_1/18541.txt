#include <iostream>
using namespace std;
const int days[12]={31,28,31,30,31,30,31,31,30,31,30,31};
bool leap(int x){return !(x%400)||(!(x%4)&&x%100);}
int main(){
    int y1,m1,d1,y2,m2,d2,ans=0;
    cin>>y1>>m1>>d1>>y2>>m2>>d2;
    if(y2<y1){int t=y1;y1=y2;y2=t;t=m1;m1=m2;m2=t;t=d1;d1=d2;d2=t;}
    if(m2<m1||(m2==m1&&d2<d1)){y1++;m1+=12;m2+=12;}
    for(int i=y1;i<y2;++i){if(leap(i))ans+=366;else ans+=365;}
    for(int i=m1;i<m2;++i)ans-=days[i-1];
    ans+=d2-d1;
    cout<<(ans+1)%7<<endl;
    return 0;
}
