import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    static ArrayList<Integer>[] g;
    static boolean[] vis;
    static int[] ans;

    static int dfs(int u) {
        vis[u] = true;
        int res = 1;
        for (int v : g[u]) {
            if (!vis[v])
                res += dfs(v);
        }
        ans[u] = res;
        return res;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        g = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++)
            g[i] = new ArrayList<>();
        vis = new boolean[n + 1];
        ans = new int[n + 1];
        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            g[a].add(b);
        }
        for (int i = 1; i <= n; i++) {
            if (!vis[i]) {
                dfs(i);
            }
        }
        int sum = 0;
        for (int i = 1; i <= n; i++) {
            if (ans[i] > 1)
                sum++;
        }
        System.out.println(sum);
    }
}
