def swap_at_most_one(s, c):
    n = len(s)
    m = len(c)
    for i in range(min(n, m)):
        if s[i] > c[i]:
            for j in range(i + 1, min(n, m)):
                if s[j] < c[i]:
                    s = s[:i] + s[j] + s[i + 1:j] + s[i] + s[j + 1:]
                    return s
            return "---"
    return s
t = int(input())
for _ in range(t):
    s, c = input().split()
    print(swap_at_most_one(s, c))
