s=input()
l=len(s)
t=[[1]*l for i in range(l)]
for i in range(l-1,-1,-1):
    for j in range(l):
        k=i+j+1
        if k>=l:
            break
        if s[i]==s[k]:
            t[i][k]+=t[i+1][k-1]
m=0
for i in range(l):
    for j in range(l):
        if i>j:
            m=max(m,t[i][j])
        else:
            m=max(m,t[i][j]-1)
print(m)
