#include <iostream>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, x, y;
        cin >> n >> m >> x >> y;
        int wh = 0;
        char a[101][1001];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                cin >> a[i][j];
                if (a[i][j] == '.') ++wh;
            }
        }
        if (wh <= 1) { cout << x << endl; continue; }
        cout << (wh / 2 + (wh % 2)) * min(x, 2 * y) << endl;
    }
    return 0;
}
