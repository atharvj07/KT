#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  bool ok = true;
  for (int i = 1; i < n; i++) {
    if (i % 2 == 0) {
      ok &= (a[i] >= a[i - 1]);
    } else {
      ok &= (a[i] <= a[i - 1]);
    }
  }
  if (!ok) {
    cout << "Impossible" << endl;
  } else {
    for (int i = 0; i < n; i++) {
      cout << a[i] << " ";
    }
    cout << endl;
  }
  return 0;
}
