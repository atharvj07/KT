a,b=map(int,input().split())
if(a==b):
    print("YES\n0")
    exit()
l=[a]
while(True):
    if(a==b):
        break
    if(a*2<=b):
        a*=2
        l.append(a)
    elif((a*10+1)<=b):
        a=a*10+1
        l.append(a)
    else:
        break
if(a==b):
    print("YES")
    print(len(l))
    for i in l:
        print(i)
else:
    print("NO")
