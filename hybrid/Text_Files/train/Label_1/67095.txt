n, m = map(int, input().split())
ss = []
for _ in range(n):
    s = input()
    t = s[::-1]
    ss.append(s)
    ss.append(t)
ss.sort()
for i in range(n + n - 1):
    j = 0
    while j < m and ss[i][j] == ss[i + 1][j]:
        j += 1
    if j == m:
        break
else:
    i = -1
if i == -1:
    print(0)
else:
    print(2 * j)
    print(ss[i][:j] + ss[i + 1][j:])
