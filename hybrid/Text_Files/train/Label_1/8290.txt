#include <bits/stdc++.h>

using namespace std;
#define pii pair<int, int>
#define fi first
#define se second
#define FOR(i, a, b) for(int i=(a); i<(b); ++i)
#define REP(i, b) FOR(i, 0, b)

typedef long long ll;
const int MAXN = 200000;
const int oo = (1 << 30);
const char nl = '\n';

int n, k;
string S;
int ans[MAXN];

int main() {
    int c = oo;
    ios::sync_with_stdio(0);
    cin.tie(0);

    cin >> n >> k;
    cin >> S;

    S = "0" + S;
    int j = n;
    int cnt_change = 0;

    FOR(i, 1, n + 1) {
        int a = S[i] - '0';
        FOR(b, a, 10) {
            if(b == 0 && i == 1) continue;
            if(cnt_change == k) {
                c = min(c, j);
                break;
            }

            if(j > i) {
                ans[j] = b;
                cnt_change++;
                j--;
            } else {
                if(a != b) cnt_change++;
                c = min(c, j);
                break;
            }
        }
        j = n;
        cnt_change = 0;
    }

    ans[n] = 0;
    FOR(i, c, n + 1) {
        cout << ans[i];
    }

    return 0;
}
