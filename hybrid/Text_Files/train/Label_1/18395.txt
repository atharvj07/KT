#include<bits/stdc++.h>
#define ll long long
using namespace std;
const ll maxn=1000005,inf=2e18;
ll n,m;
ll a[maxn];
ll d[maxn];
bool v[maxn];
ll f(ll l,ll r){
    if(l>r)swap(l,r);
    if(l==r)return a[l];
    if(v[r]==true)return d[r];
    ll x=f(l,r-1);
    ll y=f(l+1,r);
    d[r]=__gcd(x,y);
    v[r]=true;
    return d[r];
}
int main(){
    cin>>n>>m;
    for(ll i=1;i<=n;i++){
        cin>>a[i];
    }
    ll l=1,r=n,ans=0,pos=0,maxl=0;
    while(l<=r){
        ll mid=(l+r)/2;
        ll val=f(1,mid);
        if(val<=m){
            if(mid>maxl){
                pos=mid;
                maxl=mid;
                ans=val;
            }
            l=mid+1;
        }
        else r=mid-1;
    }
    cout<<ans<<" "<<maxl<<endl;
    for(ll i=1;i<=maxl;i++){
        cout<<f(i,pos)<<" ";
    }
    return 0;
}
