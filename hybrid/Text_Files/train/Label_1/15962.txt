#include <bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, p, k;
        cin >> n >> p >> k;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        sort(a.begin(), a.end());
        int ans = 0;
        while (p >= a[ans] && ans < n) {
            if (ans < n - 1 && p >= a[ans] + a[ans + 1]) {
                p -= a[ans] + a[ans + 1];
                ans += 2;
            } else {
                p -= a[ans];
                ans++;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
