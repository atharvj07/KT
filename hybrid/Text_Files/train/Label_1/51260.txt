import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[][] a = new int[n][m], b = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        int cnt = 0;
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < m - 1; j++) {
                if (a[i][j] + a[i + 1][j] + a[i][j + 1] + a[i + 1][j + 1] < 2) {
                    System.out.println(-1);
                    return;
                }
                if (a[i][j] == 0) {
                    if (a[i + 1][j] + a[i][j + 1] + a[i + 1][j + 1] > 0) {
                        cnt++;
                        b[i][j] = b[i][j + 1] = b[i + 1][j] = b[i + 1][j + 1] = 1;
                    }
                } else if (a[i][j] == 1) {
                    if (a[i + 1][j] + a[i][j + 1] + a[i + 1][j + 1] != 3) {
                        cnt++;
                        b[i][j] = b[i][j + 1] = b[i + 1][j] = b[i + 1][j + 1] = 1;
                    }
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] != b[i][j]) {
                    System.out.println(-1);
                    return;
                }
            }
        }
        System.out.println(cnt);
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < m - 1; j++) {
                if (b[i][j] == 1) {
                    System.out.println((i + 1) + " " + (j + 1));
                }
            }
        }
    }
}
