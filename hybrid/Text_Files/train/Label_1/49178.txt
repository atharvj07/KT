import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<String> s = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            s.add(sc.next());
        }
        int m = sc.nextInt();
        List<String> t = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            t.add(sc.next());
        }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (s.get(i).equals(t.get(j))) cnt++;
            }
        }
        System.out.println(cnt);
    }
}
