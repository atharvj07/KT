#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    vector<int> dp(n, 0);
    int mx = 0;
    for (int i = 1; i < n; ++i) {
        dp[i] = dp[i - 1] + a[i];
        if (dp[i] > mx) mx = dp[i];
    }
    for (int i = 0; i < n; ++i) {
        for (int j = i; j < n; ++j) {
            int sum = dp[j] - (i > 0 ? dp[i - 1] : 0);
            if (mx < sum) mx = sum;
        }
    }
    cout << mx << endl;
    return 0;
}
