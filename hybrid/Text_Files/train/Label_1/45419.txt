import java.util.*;
import java.io.*;

class Main {
    final static int MAX_N = 300005;
    static Map<Integer, Integer> cnt = new HashMap<>();
    static int[] dp = new int[MAX_N];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();

        for (int i = 0; i < k; i++) {
            int size = sc.nextInt();
            for (int j = 0; j < size; j++) {
                int x = sc.nextInt();
                cnt.put(x, cnt.getOrDefault(x, 0) + 1);
            }
        }

        for (int i = 1; i <= n; i++) {
            if (s.charAt(i - 1) == '1') {
                dp[i] = dp[i - 1];
            } else {
                dp[i] = dp[i - 1];
                if (cnt.containsKey(i)) {
                    dp[i] = Math.min(dp[i], dp[i - cnt.get(i)] + 1);
                }
            }
        }

        for (int i = 1; i <= n; i++) {
            System.out.println(dp[i]);
        }
    }
}
