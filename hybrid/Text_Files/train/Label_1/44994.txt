import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), h = sc.nextInt();
        int width = 0;
        for (int i = 0; i < n; i++) {
            int a = sc.nextInt();
            width += (a > h ? 2 : 1);
        }
        System.out.println(width);
    }
}
