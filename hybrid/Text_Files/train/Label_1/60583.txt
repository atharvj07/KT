import sys
def dpoint(x, y):
    return [x, y]
def cross(p, q, r):
    v = (q[0] - p[0]) * (r[1] - p[1]) - (q[1] - p[1]) * (r[0] - p[0])
    return v
def seg_intersect(p, q, r, s):
    if cross(r, s, p) * cross(r, s, q) != 0 and cross(p, q, r) * cross(p, q, s) != 0:
        return True
    if p == r or p == s or q == r or q == s:
        return True
    if cross(r, p, q) == 0 and min(r[0], s[0]) <= p[0] <= max(r[0], s[0]) and min(r[1], s[1]) <= p[1] <= max(r[1], s[1]):
        return True
    if cross(p, q, r) == 0 and min(p[0], q[0]) <= r[0] <= max(p[0], q[0]) and min(p[1], q[1]) <= r[1] <= max(p[1], q[1]):
        return True
    return False
def point_line_dist(p, q, r, s):
    if r == s:
        return float('inf')
    if r == p or r == q:
        return 0.
    if s == p or s == q:
        return 0.
    if cross(s, r, p) * cross(s, r, q) < 0 or cross(r, s, p) * cross(r, s, q) < 0:
        dist_p = ((r[0] - p[0]) * (r[0] - p[0]) + (r[1] - p[1]) * (r[1] - p[1])) ** 0.5
        dist_q = ((s[0] - q[0]) * (s[0] - q[0]) + (s[1] - q[1]) * (s[1] - q[1])) ** 0.5
        return min(dist_p, dist_q)
    a = cross(s, r, p) / ((s[0] - r[0]) * (r[1] - p[1]) - (s[1] - r[1]) * (r[0] - p[0]))
    return abs(a)
