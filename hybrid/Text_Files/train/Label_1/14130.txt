#include<bits/stdc++.h>
using namespace std;
const int b = 1000000007;
int main() {
  long long a, d;
  cin >> a >> d;
  if (d == 0) { cout << -1 << endl; return 0; }
  long long x = 1, y = d;
  for (int i = 1; i <= 62; i++) {
    if (d == 1) { x = 1; y = 1; break; }
    if (1ll * x * d <= 1ll * y * (d - 1)) { y = x * d; x *= d; }
    else { x *= d - 1; }
    //cout << x << " " << y << endl;
  }
  long long r = 0;
  int w = 0;
  for (int i = 62; i >= 0; i--) {
    if (x > 1) { w += 1; r += x; x /= d; }
    else if (1ll * r + 1ll * w * d <= a) {w += 1; r += x;}
    else break;
  }
  cout << w << endl;
  for (int i = 0; i < w; i++) { cout << y << " "; y /= d; }
  cout << endl;
}
