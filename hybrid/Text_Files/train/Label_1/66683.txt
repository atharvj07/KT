import sys
from fractions import Fraction

S = []
Q = int(input())
for _ in range(Q):
    query = sys.stdin.readline().split()
    if query[0] == '1':
        S.append(int(query[1]))
    else:
        ma, sum = S[0], S[0]
        for x in S[1:]:
            sum += x
            ma = max(ma, x)
        print(Fraction(ma, 1) - Fraction(sum, len(S)))
