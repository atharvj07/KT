import java.util.*;
import java.io.BufferedReader;
import java.io.InputStreamReader;
 
class Main
{
   public static void main(String[] args) throws Exception
   {
      BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
      String[] parts = br.readLine().split(" ");
      int n = Integer.parseInt(parts[0]), m = Integer.parseInt(parts[1]);
      long[] hi = new long[n];
      parts = br.readLine().split(" ");
      for (int i = 0; i < n; i++) hi[i] = Long.parseLong(parts[i]);
      long[] pi = new long[m];
      parts = br.readLine().split(" ");
      for (int i = 0; i < m; i++) pi[i] = Long.parseLong(parts[i]);
      long ans = 0;
      for (int j = 0; j < n; j++) ans = Math.max(ans, Math.abs(hi[j] - pi[0]));
      for (int i = 0; i < m - 1; i++) ans = Math.max(ans, (pi[i + 1] - pi[i]) / 2);
      System.out.println(ans);
   }
}
