#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        int x, y;
        cin >> x >> y;
        v[i] = {x, y};
    }
    vector<int> c(n, -1);  // direction of movement (1 or -1)
    double p = 0;  // current position
    for (int i = 0; i < n; i++) {
        double p_next = p + sqrt(v[i].first * v[i].first + v[i].second * v[i].second);
        if (p_next > 1.5e6)  // if moving in this direction would make |p| > 1.5e6
            c[i] = -c[i];  // change direction
        p = p + v[i].first * c[i] + v[i].second * c[i];  // move in the chosen direction
    }
    for (int i = 0; i < n; i++)
        cout << c[i] << " ";
    cout << endl;
    return 0;
}
