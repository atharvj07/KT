#include <cstdio>
#include <algorithm>

int main() {
    int n, k;
    scanf("%d%d", &n, &k);
    int a[n+1];
    for (int i = 1; i<=n; i++)
        a[i] = i;
    int r = n-1;
    for (int t = 1; t<=k; t++) {
        int u = 1+t*t%r;
        int v = n-t*t%r;
        std::swap(a[u], a[v]);
        r -= std::min(u, v);
    }
    int ans = 0;
    for (int i = 2; i<=n; i++)
        if (a[i] < a[i-1]) ans += i-1;
    printf("%d\n", ans);
    return 0;
}
