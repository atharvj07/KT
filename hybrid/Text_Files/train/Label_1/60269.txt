def count_digits_base7(x):
    if x == 0:
        return 1
    digits = 0
    while x:
        digits += 1
        x //= 7
    return digits

def calculate_combinations(n, m):
    digits_hours = count_digits_base7(n-1)
    digits_minutes = count_digits_base7(m-1)
    if digits_hours + digits_minutes > 7:
        return 0
    total = 1
    for i in range(7 - digits_hours - digits_minutes + 1, 8):
        total *= i
    for i in range(1, digits_hours + 1):
        total //= i
    for i in range(1, digits_minutes + 1):
        total //= i
    return total

n, m = map(int, input().split())
print(calculate_combinations(n, m))
