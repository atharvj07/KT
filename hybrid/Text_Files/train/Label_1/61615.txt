def solve(n, b):
    used = [False] * (2*n + 1)
    a = []
    for x in b:
        used[x] = True
        a.append(x)
        next_min = None
        for i in range(x + 1, 2*n + 1):
            if not used[i]:
                next_min = i
                used[i] = True
                break
        if next_min is None:
            return [-1]
        a.append(next_min)
    return a

t = int(input())
for _ in range(t):
    n = int(input())
    b = list(map(int, input().split()))
    result = solve(n, b)
    print(" ".join(map(str, result)))
