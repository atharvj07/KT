import java.util.*;
import java.io.*;
class Main{
   static int t, n, m;
   static int[][] a;
   static boolean[] vis;
   static Queue<Integer> q;
   static void dfs(int v){
      if(vis[v]) return;
      vis[v] = true;
      if(a[v/m][v%m] == 0) q.add(v);
      else{
         if(v/m > 0) dfs(v-m);
         if(v%m > 0) dfs(v-1);
         if(v/m < n-1) dfs(v+m);
         if(v%m < m-1) dfs(v+1);
      }
   }
   static boolean win(){
      vis = new boolean[n*m];
      dfs(n*m-1);
      for(int i=0; i<n*m; i++)
         if(!vis[i] && a[i/m][i%m] > 0) return true;
      return false;
   }
   public static void main(String[] args){
      Scanner in = new Scanner(System.in);
      t = in.nextInt();
      while(t-- > 0){
         n = in.nextInt();
         m = in.nextInt();
         a = new int[n][m];
         for(int i=0; i<n; i++)
            for(int j=0; j<m; j++)
               a[i][j] = in.nextInt();
         q = new LinkedList<>();
         boolean game = win();
         if(!game){
            while(!q.isEmpty()){
               int now = q.poll();
               int x = now/m, y = now%m;
               int temp = a[x][y] - 1;
               if(temp > 0){
                  a[x][y] = temp;
                  q.add(now-1);
                  q.add(now+1);
                  q.add(now+m);
                  q.add(now-m);
               }
            }
            game = win();
         }
         if(game) System.out.println("Ashish");
         else System.out.println("Jeel");
      }
   }
}
