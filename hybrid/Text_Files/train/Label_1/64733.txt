import math
import sys

x1, y1, x2, y2 = map(int, sys.stdin.readline().split())
VMAX, t = map(int, sys.stdin.readline().split())

v = list(map(int, sys.stdin.readline().split())) + list(map(int, sys.stdin.readline().split()))
w = list(map(int, sys.stdin.readline().split()))

def dist(a, b):
    return math.sqrt((a[0] - b[0])**2 + (a[1] - b[1])**2)

def time(a, b, v):
    return dist(a, b) / v

ans = [time((x1, y1), (x2, y2), VMAX)]
for i in range(t):
    a = (x1, y1)
    b = (x1 + v[2*i], y1 + v[2*i + 1])
    c = (x2, y2)
    ans.append(min(ans[-1] + time(a, b, VMAX), ans[-1] + time(a, c, VMAX) + time(c, b, VMAX)))
for i in range(t, len(v), 2):
    a = (x1, y1)
    b = (x1 + v[i], y1 + v[i + 1])
    ans.append(min(ans[-1] + time(a, b, VMAX), ans[-1] + time(a, c, VMAX) + time(c, b, VMAX)))
print(min(ans))
