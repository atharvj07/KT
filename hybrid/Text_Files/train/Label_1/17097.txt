#include <bits/stdc++.h>

using namespace std;

long long get_max_area(string s1, string s2) {
  vector<int> fr(26, 0);
  for (char ch : s1) fr[ch - 'a']++;
  vector<int> ans;
  for (char ch : s2) {
    if (fr[ch - 'a'] > 0) {
      fr[ch - 'a']--;
      ans.push_back(1);
    } else {
      ans.push_back(0);
    }
  }
  if (ans.size() < s2.size())
    return -1;
  else {
    sort(ans.rbegin(), ans.rend());
    long long sum = 0;
    for (int i = 0; i < s2.size(); i++) {
      sum += ans[i];
    }
    return sum;
  }
}

int main() {
  string s1, s2;
  cin >> s1 >> s2;
  cout << get_max_area(s1, s2) << endl;
  return 0;
}
