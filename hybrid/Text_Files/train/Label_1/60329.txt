import math

def countDivisors(n):
    cnt = 0
    for i in range(1, int(math.sqrt(n)) + 1):
        if n % i == 0:
            cnt += 1
            if i != n // i:
                cnt += 1
    return cnt

for _ in range(int(input())):
    d = int(input())
    a = d + 1
    while countDivisors(a) < 4:
        a += 1
    print(a)
