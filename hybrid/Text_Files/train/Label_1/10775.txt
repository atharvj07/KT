#include<bits/stdc++.h>
using namespace std;
const int N = 105;
int partition(vector<int>& a) {
	int x = a[0];
	int i = -1;
	for (int j = 0; j < a.size(); j++)
		if (a[j] == x) {
			i++;
			swap(a[i], a[j]);
		}
	int j = a.size();
	for (int k = i + 1; k < a.size(); k++)
		if (a[k] < x)
			swap(a[i + 1], a[k]), i++;
	return i + 1;
}
int main() {
	int n;
	cin >> n;
	vector<int> a(n / 2);
	for (int i = 0; i < a.size(); i++)
		cin >> a[i];
	int x = partition(a);
	int cnt = 0;
	for (int i = 0; i < n / 2; i++)
		if (i % 2 == 0 && a[i] > n / 2) cnt++;
	cnt += n / 4;
	cout << cnt << endl;
	return 0;

}
