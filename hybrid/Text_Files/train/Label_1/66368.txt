n, p = map(int, input().split())
ranges = [list(map(int, input().split())) for _ in range(n)]

def probability_divisible_by_p(l, r, p):
    return (r // p) - ((l - 1) // p)

total_expected_value = 0.0

for i in range(n):
    l1, r1 = ranges[i]
    l2, r2 = ranges[(i + 1) % n]
    
    prob1 = probability_divisible_by_p(l1, r1, p) / (r1 - l1 + 1)
    prob2 = probability_divisible_by_p(l2, r2, p) / (r2 - l2 + 1)
    
    prob_neither = (1 - prob1) * (1 - prob2)
    total_expected_value += (1 - prob_neither) * 2000

print(total_expected_value)
