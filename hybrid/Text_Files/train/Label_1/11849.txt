#include <bits/stdc++.h>

using namespace std;

const int MOD = 1e9 + 7;

int main() {
    string s;
    int n;
    cin >> s >> n;
    vector<pair<char, string>> queries(n);
    for (int i = 0; i < n; i++) { cin >> queries[i].first >> queries[i].second; }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < s.size(); j++) {
            if (s[j] == queries[i].first) {
                s.erase(j, 1);
                s.insert(j, queries[i].second);
                j += queries[i].second.size() - 1;
            }
        }
    }
    while (s.size() > 1 && s[0] == '0') s.erase(0, 1);
    long long ans = 0;
    for (int i = 0; i < s.size(); i++) { ans = (ans * 10 + s[i] - '0') % MOD; }
    cout << ans << endl;
}
