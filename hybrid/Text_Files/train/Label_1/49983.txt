import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long x = sc.nextLong();
        long y = sc.nextLong();
        long t = Math.min(2 * n * x, x + (n - 1) * y);
        System.out.println(t);
    }
}
