import sys
input = sys.stdin.readline

n, m = map(int, input().split())
grid = [input() for _ in range(n)]
sx, sy = 0, 0
for i in range(n):
    for j in range(m):
        if grid[i][j] == 'S':
            sx, sy = i, j
        elif grid[i][j] == 'E':
            ex, ey = i, j
dir_map = [(1, 0), (-1, 0), (0, 1), (0, -1)]
ans = 0
def dfs(x, y, z, dirs):
    global ans
    if x == ex and y == ey:
        ans += 1
        return
    c = grid[x][y]
    for i in range(4):
        nx, ny = x + dir_map[i][0], y + dir_map[i][1]
        nd = (z + i) % 4
        if 0 <= nx < n and 0 <= ny < m and grid[nx][ny] != '#' and dirs[nd] == 0:
            dirs[nd] = 1
            dfs(nx, ny, nd, dirs)
            dirs[nd] = 0
s = input()
dfs(sx, sy, 0, [0] * 4)
print(ans)
