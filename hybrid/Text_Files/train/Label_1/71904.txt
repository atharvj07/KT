def check_flag(n, m, field):
    if n % 3 == 0:
        third_n = n // 3
        if all(field[i] == field[0] for i in range(third_n)) and all(field[i] == field[third_n] for i in range(third_n, 2*third_n)) and all(field[i] == field[2*third_n] for i in range(2*third_n, n)):
            if field[0][0] != field[third_n][0] and field[third_n][0] != field[2*third_n][0] and field[0][0] != field[2*third_n][0]:
                return "YES"
    if m % 3 == 0:
        third_m = m // 3
        if all(row[0:third_m] == row[third_m:2*third_m] == row[2*third_m:3*third_m] for row in field):
            chars = set(row[0] for row in field)
            if len(chars) == 3:
                return "YES"
    return "NO"

n, m = map(int, input().split())
field = [input() for _ in range(n)]
print(check_flag(n, m, field))
