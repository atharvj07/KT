import sys
import math

n = int(sys.stdin.readline())
a = list(map(int, sys.stdin.readline().split()))
k = 0
for i in range(n-1):
    if a[i] == i+1:
        continue
    j = i+1
    while j < n and (j - i + 1) % 2 == 0:
        j += 1
    if j == n:
        j = i+1
        while j < n and (j - i + 1) % 3 == 0:
            j += 1
    if j == n:
        break
    k += 1
    print(i+1, j+1)
    a[i], a[j] = a[j], a[i]
print(k)
