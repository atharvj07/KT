#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::vector<int> a(100002);
  std::vector<std::vector<int>> ind(100002);
  std::cin >> n;
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
    ind[a[i]].push_back(i);
  }
  int ans = 0;
  for (int i = 1; i <= 100000; ++i) {
    if (ind[i].size() < 2) continue;
    int d = ind[i][1] - ind[i][0];
    bool f = true;
    for (int j = 2; j < ind[i].size(); ++j) {
      if (ind[i][j] - ind[i][j - 1] != d) {
        f = false;
        break;
      }
    }
    if (f) {
      ++ans;
      std::cout << i << " " << d << "\n";
    }
  }
  std::cout << ans << "\n";
  return 0;
}
