import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    static char[][] board;
    static int n;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        board = new char[n][];
        for (int i = 0; i < n; i++) {
            board[i] = br.readLine().toCharArray();
        }

        System.out.println(checkPossible() ? "YES" : "NO");
    }

    private static boolean checkPossible() {
        List<Point> freeCells = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (board[i][j] == '.') {
                    freeCells.add(new Point(i, j));
                }
            }
        }

        for (Point cell : freeCells) {
            if (canFitPiece(cell)) {
                return true;
            }
        }

        return false;
    }

    private static boolean canFitPiece(Point cell) {
        // Check if the top left cell is free
        if (board[cell.x][cell.y] != '.') {
            return false;
        }

        // Check if the bottom right cell is free
        if (cell.x + 2 >= n || cell.y + 2 >= n || board[cell.x + 2][cell.y + 2] != '.') {
            return false;
        }

        // Check if the middle cell is free
        if (cell.x + 1 >= n || cell.y + 1 >= n || board[cell.x + 1][cell.y + 1] != '.') {
            return false;
        }

        // Check if the right cell is free
        if (cell.y + 1 >= n || board[cell.x + 1][cell.y + 2] != '.') {
            return false;
        }

        // Check if the bottom cell is free
        if (cell.x + 2 >= n || board[cell.x + 2][cell.y] != '.') {
            return false;
        }

        return true;
    }

    private static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
