import math
n, x1, y1, x2, y2 = map(int, input().split())
flowers = [list(map(int, input().split())) for i in range(n)]
xmin = min(x1, x2)
xmax = max(x1, x2)
ymin = min(y1, y2)
ymax = max(y1, y2)
ans = 0
for x, y in flowers:
    ans = max(ans, min((x - xmin)**2 + (y - ymin)**2, (x - xmax)**2 + (y - ymax)**2, (xmin - x)**2 + (ymin - y)**2, (xmax - x)**2 + (ymax - y)**2))
print(ans)
