#include <algorithm>
#include <iostream>
#include <string>

int main() {
  std::string n;
  std::cin >> n;

  int len = n.size();
  int last_odd = -1;
  for (int i = len - 1; i >= 0; i--) {
    int digit = n[i] - '0';
    if (digit % 2 == 0) {
      last_odd = i;
      break;
    }
  }

  if (last_odd == -1) {
    std::cout << -1 << std::endl;
    return 0;
  }

  int max_even = -1;
  int max_even_idx = -1;
  for (int i = last_odd - 1; i >= 0; i--) {
    int digit = n[i] - '0';
    if (digit % 2 == 1 && digit > max_even) {
      max_even = digit;
      max_even_idx = i;
    }
  }

  if (max_even == -1) {
    std::cout << -1 << std::endl;
    return 0;
  }

  std::swap(n[last_odd], n[max_even_idx]);
  std::cout << n << std::endl;

  return 0;
}
