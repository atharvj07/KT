import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int[] kefa = new int[n];
        int[] sasha = new int[n];
        for (int i = 0; i < n; i++) {
            kefa[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            sasha[i] = scanner.nextInt();
        }
        Arrays.sort(kefa);
        Arrays.sort(sasha);
        boolean same = true;
        for (int i = 0; i < n; i++) {
            if (kefa[i] != sasha[i]) {
                same = false;
                break;
            }
        }
        System.out.println(same ? "YES" : "NO");
    }
}
