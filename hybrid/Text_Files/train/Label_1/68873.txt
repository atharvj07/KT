from collections import deque
from sys import stdin,setrecursionlimit
setrecursionlimit(10**6)
n,d,m=map(int,stdin.readline().strip().split(' '))
a=deque([[0,x] for x in map(int,stdin.readline().strip().split(' '))])
a.rotate(-m % len(a))  # a starts from the element no greater than m
ans=deque()
while a:
    l=a.popleft()
    ans.append(l)
    while a and l[1] < m:
        ans.append(a.popleft())
    if len(ans) > d:
        ans.popleft()
    a.rotate(-d)
print(sum([x[1] for x in ans]))
