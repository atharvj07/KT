#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
  int n;
  cin >> n;
  vector<ll> a(n);
  ll sum = 0;
  for (ll &x : a) {
    cin >> x;
    sum += x;
  }
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    if (sum - a[i] == a[i]) ++ans;
  }
  cout << ans << '\n';
  for (int i = 0; i < n; ++i) {
    if (sum - a[i] == a[i]) cout << i + 1 << ' ';
  }
}
