#include <cstdio>
#include <vector>
using namespace std;
const int basex[] = {-1, -1, 0, 1, 1, 1, 0, -1};
const int basey[] = {0, 1, 1, 1, 0, -1, -1, -1};
const int maxn = 1000;
char grid[maxn][maxn];
bool vis[maxn][maxn];
int n, m;
void dfs(int x, int y, int z) {
  if (x < 1 || x > n || y < 1 || y > m || grid[x][y] == '.' || vis[x][y])
    return;
  vis[x][y] = true;
  for (int i = 0; i < 8; ++i) {
    int nx = x + basex[i] * z;
    int ny = y + basey[i] * z;
    if (grid[nx][ny] == '*' && !vis[nx][ny]) { dfs(nx, ny, z); }
  }
}
int main() {
  scanf("%d%d", &n, &m);
  vector<int> x, y, z;
  for (int i = 1; i <= n; ++i) {
    scanf("%s", grid[i] + 1);
  }
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      if (grid[i][j] == '*' && !vis[i][j]) {
        int c = 0;
        for (int p = 0; p < 8; p++) {
          int nx = i + basex[p];
          int ny = j + basey[p];
          if (grid[nx][ny] == '*') ++c;
        }
        if (c == 1) {
          puts("-1");
          return 0;
        }
        for (int p = 1;; ++p) {
          bool ok = false;
          for (int q = 0; q < 8; q++) {
            int nx = i + basex[q] * p;
            int ny = j + basey[q] * p;
            if (nx >= 1 && nx <= n && ny >= 1 && ny <= m && grid[nx][ny] ==
                '.') {
              ok = true;
              break;
            }
          }
          if (!ok) {
            dfs(i, j, p);
            x.push_back(i);
            y.push_back(j);
            z.push_back(p);
            break;
          }
        }
      }
    }
  }
  printf("%d\n", (int)x.size());
  for (int i = 0; i < (int)x.size(); ++i) {
    printf("%d %d %d\n", x[i], y[i], z[i]);
  }
  return 0;
}
