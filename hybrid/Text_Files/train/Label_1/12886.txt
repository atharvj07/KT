#include <bits/stdc++.h>

using namespace std;

typedef long long LL;

LL gcd(LL a, LL b) {
  return b ? gcd(b, a % b) : a;
}

int main() {
  LL n, m, ans = 0;
  cin >> n >> m;
  if (m == 1) {
    ans = n * n;
  } else {
    for (int i = 1; i <= n; ++i) {
      for (int j = 1; j <= n; ++j) {
        ans += (i * i + j * j) % m == 0;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
