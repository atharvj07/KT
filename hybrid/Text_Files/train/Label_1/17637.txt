#include <bits/stdc++.h>
using namespace std;

const int mxN = 101;
int n, t, a[mxN], b[mxN];
int c[mxN];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i] >> b[i];

    for (int i = 1; i <= n; i++) {
        c[i] = b[i] - a[i];
        if (c[i] < 0) c[i] = 0;
    }

    sort(c + 1, c + n + 1, greater<int>());

    int ans1 = 0, ans2 = 0, sum = 0;
    for (int i = 1; i <= n; i++) {
        if (sum + c[i] >= a[i]) {
            sum += c[i] - a[i];
            ans1++;
        } else {
            ans1++;
            ans2 += a[i] - sum;
            sum = c[i];
        }
    }

    cout << ans1 << " " << ans2;

    return 0;
}
