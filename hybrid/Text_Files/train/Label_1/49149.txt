import java.util.Scanner;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        BigInteger l = new BigInteger(sc.next());
        BigInteger r = new BigInteger(sc.next());

        BigInteger twoK = BigInteger.valueOf(2).pow(k);
        BigInteger rSubL = r.subtract(l).add(BigInteger.ONE);
        BigInteger mod = BigInteger.valueOf(1000000007);

        BigInteger res = ((twoK.multiply(rSubL.modPow(twoK.subtract(BigInteger.ONE), mod))
                .multiply(l.modPow(twoK, mod))).multiply(BigInteger.valueOf(2).modPow(twoK, mod)))
                .mod(mod);
        System.out.println(res);
    }
}
