import java.util.*;

class Main {
    static int n;
    static int k;
    static int[] a;

    static boolean can(int m) {
        int[][] d = new int[m + 1][n + 1];
        for (int i = 0; i <= m; i++) {
            for (int j = 0; j <= n; j++) {
                d[i][j] = -1;
            }
        }
        d[0][0] = 0;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j <= n; j++) {
                if (d[i][j] == -1) {
                    continue;
                }
                for (int l = 0; l <= k && j + l <= n; l++) {
                    d[i + 1][j + l] = Math.max(d[i + 1][j + l], d[i][j] + a[j + l]);
                }
            }
        }
        return d[m][n] == a[n];
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            n = sc.nextInt();
            k = sc.nextInt();
            a = new int[n + 1];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            int l = 1;
            int r = n;
            int ans = -1;
            while (l <= r) {
                int m = (l + r) / 2;
                if (can(m)) {
                    ans = m;
                    r = m - 1;
                } else {
                    l = m + 1;
                }
            }
            System.out.println(ans);
        }
    }
}
