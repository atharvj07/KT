from copy import deepcopy
n,m,h=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
t=[]
for i in range(n):t.append(list(map(int,input().split())))
H=[]
for i in range(n):
    L=[0]*m
    for j in range(m):
        if a[j]==0:L[j]=0
        else:L[j]=max(L[j],min(b[i],h))
    H.append(L)
L=deepcopy(H)
for i in range(1,n):
    for j in range(m):
        L[i][j]-=L[i-1][j]
for i in range(n):
    for j in range(m):
        if t[i][j]==0:H[i][j]=0
        else:H[i][j]=max(H[i][j],L[i][j]);H[i][j]=min(H[i][j],h)
for i in range(n):
    for j in range(m):
        if j>0:H[i][j]+=H[i][j-1]
for i in range(n):
    for j in range(m):print(H[i][j],end=' ')
    print()
