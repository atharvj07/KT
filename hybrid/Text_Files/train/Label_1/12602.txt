#include <iostream>
using namespace std;

int main() {
  int t; cin >> t;
  while (t--) {
    int n; cin >> n;
    string s; cin >> s;
    bool ok = true;
    int cntT = 0, cntM = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == 'T') cntT++;
      if (s[i] == 'M') cntM++;
      if (cntT < cntM) ok = false;
    }
    if ((cntT + cntM) % 3 > 0) ok = false;
    if (!ok) cout << "NO" << endl;
    else cout << "YES" << endl;
  }
}
