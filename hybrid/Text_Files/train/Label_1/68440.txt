n, k, p = list(map(int,input().split()))
a = list(map(int,input().split()))
b = list(map(int,input().split()))
a.sort()
b.sort()
ans = 10**9
for x in b:
    l, r = -1, len(a)
    while l+1<r:
        m = (l+r)//2
        if a[m]<x:
            l=m
        else:
            r=m
    if l!=-1: ans=min(ans,abs(x-a[l])+abs(x-p))
    if r!=len(a): ans=min(ans,abs(x-a[r])+abs(x-p))
print(ans)
