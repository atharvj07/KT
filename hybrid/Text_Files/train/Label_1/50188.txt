import java.util.*;
import java.io.*;

class Main {
    static long[] a;
    static long[] sum;
    public static void main(String[] args) throws IOException {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        a = new long[n];
        sum = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextLong();
        }
        Arrays.sort(a);
        sum[0] = a[0];
        for (int i = 1; i < n; i++) {
            sum[i] = sum[i - 1] + a[i];
        }
        long ans = Long.MAX_VALUE;
        for (int i = 1; i <= n; i += 2) {
            ans = Math.min(ans, sum[n - 1] - 2 * sum[n - i]);
        }
        System.out.println(ans * ans);
    }
}
