#include<bits/stdc++.h>
using namespace std;
typedef pair<int, int> pi;
int n, m, p[5005], c[5005], a[5005], b[5005], ans;
vector<pi> v;
int main() {
    scanf("%d %d", &n, &m);
    for(int i = 1;i <= n;i++) scanf("%d", &c[i]), p[c[i]]++;
    for(int i = 1;i <= m;i++) if(p[i] % 2 == 1) v.push_back({p[i], i});
    for(int i = 0;i < v.size();i++, ans++) {
        for(int j = 1;j <= n;j++) if(a[j] == 0 && c[j] == v[i].second) a[j] = v[i].first % 2 + 1;
        for(int j = 1;j <= n;j++) if(b[j] == 0 && c[j] == v[i].second) b[j] = v[i].first % 2 + 2;
    }
    for(int i = 1;i <= m;i++) {
        if(p[i] > 0 && p[i] % 2 == 0) {
            for(int j = 1;j <= n;j++) if(c[j] == i && a[j] == 0) a[j] = b[j] = (p[i] / 2 % 2 + 1);
            for(int j = 1;j <= n;j++) if(c[j] == i && a[j] == 0) a[j] = b[j] = (p[i] / 2 % 2 + 2);
        }
    }
    printf("%d\n", ans);
    for(int i = 1;i <= n;i++) printf("%d %d\n", a[i], b[i]);
    return 0;
}
