t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    s = input()
    changes = 0
    for i in range(k // 2):
        char_count = [0] * 26
        for j in range(i, n, k):
            char_count[ord(s[j]) - ord('a')] += 1
            char_count[ord(s[k - 1 - i + j // k * k]) - ord('a')] += 1
        changes += 2 * n // k - max(char_count)
    if k % 2:
        char_count = [0] * 26
        for i in range(k // 2, n, k):
            char_count[ord(s[i]) - ord('a')] += 1
        changes += n // k - max(char_count)
    print(changes)
