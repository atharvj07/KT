import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        int[] c = new int[m];
        int[] d = new int[m];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            c[i] = sc.nextInt();
            d[i] = sc.nextInt();
        }
        sc.close();
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            int min = Integer.MAX_VALUE;
            int idx = -1;
            for (int j = 0; j < m; j++) {
                int dist = Math.abs(a[i] - c[j]) + Math.abs(b[i] - d[j]);
                if (dist < min) {
                    min = dist;
                    idx = j + 1;
                } else if (dist == min) {
                    idx = Math.min(idx, j + 1);
                }
            }
            ans[i] = idx;
        }
        for (int i = 0; i < n; i++) {
            System.out.println(ans[i]);
        }
    }
}
