#include <cstdio>
#include <vector>
using namespace std;

typedef long long ll;

const int MAXN = 12;

int n, t;
bool A[MAXN][MAXN];

int main() {
    scanf("%d%d", &n, &t);
    for (int i = 1; i <= n; i++) for (int j = 1; j <= i; j++) {
        A[i][j] = false;
    }
    ll d = t / n;
    ll r = t % n;
    A[n][1] = true;
    for (int i = 0; i < d; i++) {
        for (int j = 1; j <= n; j++) {
            if (A[j][1]) {
                A[j + 1][1] = true;
                A[j][2] = true;
            }
        }
    }
    for (int i = 0; i < r; i++) {
        for (int j = 1; j <= n; j++) {
            if (A[j][1]) {
                A[j][i + 2] = true;
            }
        }
    }
    ll sum = 0;
    for (int i = 1; i <= n; i++) for (int j = 1; j <= i; j++) {
        if (A[i][j]) sum++;
    }
    printf("%lld", sum);
    return 0;
}
