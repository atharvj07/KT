#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const ll INF = LLONG_MAX / 3;

int n, m, k;

struct edge {
    int from, to, weight;
};

vector<edge> edges;

vector<vector<ll>> d;

void dijkstra() {
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll, int>>> pq;
    d[1][1] = 0;
    pq.emplace(0, 1);
    while (!pq.empty()) {
        int v = pq.top().second;
        ll dist = pq.top().first;
        pq.pop();
        if (d[1][v] < dist) {
            continue;
        }
        for (edge e : edges) {
            int u = e.from, w = e.weight;
            if (u == v) {
                u = e.to;
            }
            if (d[1][u] > d[1][v] + w) {
                d[1][u] = d[1][v] + w;
                pq.emplace(d[1][u], u);
            }
        }
    }
}

int main() {
    cin >> n >> m >> k;
    edges.resize(m);
    d.assign(n + 1, vector<ll>(n + 1, INF));
    for (int i = 0; i < m; i++) {
        cin >> edges[i].from >> edges[i].to >> edges[i].weight;
    }
    dijkstra();
    vector<ll> ans;
    for (int i = 2; i <= n; i++) {
        if (i != 1) {
            ans.push_back(d[1][i]);
        }
    }
    sort(ans.begin(), ans.end());
    cout << ans[k - 1] << endl;
    return 0;
}
