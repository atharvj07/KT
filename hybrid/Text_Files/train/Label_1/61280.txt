def min_moves_to_transform(s, t):
    if sorted(s) != sorted(t):
        return -1
    moves = 0
    for i in range(len(s)):
        if s[i] != t[i]:
            index = s.index(t[i], i)
            if index < i:
                moves += i - index
                s = s[:index] + s[index+1:i+1] + s[index] + s[i+1:]
            else:
                moves += index - i
                s = s[:i] + s[index] + s[i:index] + s[index+1:]
    return moves

q = int(input().strip())
for _ in range(q):
    n = int(input().strip())
    s = input().strip()
    t = input().strip()
    print(min_moves_to_transform(s, t))
