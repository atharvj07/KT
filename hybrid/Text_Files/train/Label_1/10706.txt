#include <cstdio>
#include <vector>
using namespace std;

int n, k;
vector<int> v;

int gcd(int a, int b) {
    int r;
    while (b != 0) {
        r = a % b;
        a = b;
        b = r;
    }
    return a;
}

int main() {
    scanf("%d%d", &n, &k);
    if (k * (k + 1) / 2 > n) {
        puts("-1");
        return 0;
    }
    int g = 1;
    for (int j = k; j >= 1; j--) {
        int t = min(n - g, j);
        n -= t;
        g = gcd(g, t);
        v.push_back(t);
    }
    for (int i = 0; i < k; i++) {
        printf("%d ", v[i]);
    }
    puts("");
    return 0;
}
