#include<bits/stdc++.h>
using namespace std;
int main() {
    long long int n;
    cin >> n;
    if(n < 3) cout << "No";
    else{
        cout << "Yes\n";
        cout << n/2 << " ";
        for(int i = 1; i <= n/2; i+=2) cout << i << " ";
        cout << "\n" << n/2 << " ";
        for(int i = 2; i <= n/2; i+=2) cout << i << " ";
    }
    return 0;
}
