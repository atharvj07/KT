import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int ans = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (Character.isLowerCase(s.charAt(i))) {
                cur++;
            } else {
                ans = Math.max(ans, cur);
                cur = 0;
            }
        }
        ans = Math.max(ans, cur);
        System.out.println(ans);
    }
}
