#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;

  while (t--) {
    int n;
    cin >> n;

    vector<int> a(n);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
    }

    int bad = 0;
    int max = a[n - 1];
    for (int i = n - 2; i >= 0; i--) {
      if (a[i] > max) {
        bad++;
      } else {
        max = a[i];
      }
    }

    cout << bad << '\n';
  }

  return 0;
}
