#include <bits/stdc++.h>
using namespace std;
int a[200005],n,ans;
int main(){
    cin>>n;
    string s;
    cin>>s;
    for(int i=0;i<n;++i) a[i]=(s[i]-'0');
    for(int i=0;i<n;++i)
        for(int j=i+1;j<n;++j){
            int l=(i-j+n)%n;
            if(a[i]!=a[l]) ++ans;
        }
    cout<<ans<<endl;
}
