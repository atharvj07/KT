import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] a = new int[n];
        int ones = 0;
        for (int i = 0; i < n; i++) {
            a[i] = scan.nextInt();
            ones += Integer.bitCount(a[i]);
        }
        if (ones % n != 0) {
            System.out.println(-1);
            return;
        }
        int x = ones / n;
        System.out.println(x);
    }
}
