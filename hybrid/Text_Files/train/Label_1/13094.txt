#include <bits/stdc++.h>

using namespace std;

#define ll long long

const int maxn = 10001;

ll dp[maxn];

ll go(ll n, ll m) {
    if (n == m) return 0;
    if (dp[n] != -1) return dp[n];
    ll ans = 1e9;
    if (n < m) ans = min(ans, 1 + go(n * 2, m));
    else ans = min(ans, 1 + go(n - 1, m));
    return dp[n] = ans;
}

int main() {
    memset(dp, -1, sizeof dp);
    ll n, m;
    cin >> n >> m;
    cout << go(n, m) << endl;
    return 0;
}
