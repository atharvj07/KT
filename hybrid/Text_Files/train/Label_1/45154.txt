import java.util.*;

class Main {
    static int r, c, minBattle;
    static char[][] map;
    static boolean[][] visited;
    static int[][] moves = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}}; // down, up, right, left

    static void bfs(int x, int y) {
        Queue<Integer> queue = new LinkedList<>();
        queue.add(x * c + y);
        visited[x][y] = true;
        while (!queue.isEmpty()) {
            int cur = queue.poll();
            int cx = cur / c;
            int cy = cur % c;
            if (map[cx][cy] == 'E') {
                minBattle = Math.min(minBattle, bfsHelper(cx, cy));
                return;
            }
            for (int[] move : moves) {
                int dx = cx + move[0];
                int dy = cy + move[1];
                if (dx >= 0 && dx < r && dy >= 0 && dy < c && map[dx][dy] != 'T' && !visited[dx][dy]) {
                    queue.add(dx * c + dy);
                    visited[dx][dy] = true;
                }
            }
        }
    }

    static int bfsHelper(int x, int y) {
        Queue<Integer> queue = new LinkedList<>();
        queue.add(x * c + y);
        int battle = 0;
        while (!queue.isEmpty()) {
            int cur = queue.poll();
            int cx = cur / c;
            int cy = cur % c;
            if (map[cx][cy] != 'E') battle += Character.getNumericValue(map[cx][cy]);
            for (int[] move : moves) {
                int dx = cx + move[0];
                int dy = cy + move[1];
                if (dx >= 0 && dx < r && dy >= 0 && dy < c && map[dx][dy] != 'T' && !visited[dx][dy]) {
                    queue.add(dx * c + dy);
                    visited[dx][dy] = true;
                }
            }
        }
        return battle;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        r = sc.nextInt();
        c = sc.nextInt();
        map = new char[r][c];
        visited = new boolean[r][c];
        minBattle = Integer.MAX_VALUE;
        for (int i = 0; i < r; i++) {
            String line = sc.next();
            for (int j = 0; j < c; j++) {
                map[i][j] = line.charAt(j);
                if (map[i][j] == 'S') {
                    bfs(i, j);
                }
            }
        }
        System.out.println(minBattle == Integer.MAX_VALUE ? -1 : minBattle);
    }
}
