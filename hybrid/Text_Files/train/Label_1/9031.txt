#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> f(10);
  for (int i = 0; i < 9; i++) { cin >> f[i + 1]; }
  vector<int> dp(n);
  dp[n - 1] = f[s[n - 1] - '0'];
  for (int i = n - 2; i >= 0; i--) {
    if (f[s[i + 1] - '0'] >= s[i] - '0')
      dp[i] = dp[i + 1];
    else
      dp[i] = f[s[i] - '0'] * 10 + dp[i + 1];
  }
  cout << dp[0] << endl;
}
