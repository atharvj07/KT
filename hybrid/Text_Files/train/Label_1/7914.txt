#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const ll mod = 1e9 + 7;
const int N = 1e5 + 5;

int n, m;
string s;
ll pw[N], inv2[N];
ll dp[N][26];

int main()
{
    cin >> n >> m;
    cin >> s;

    pw[0] = 1;
    for (int i = 1; i <= n; i++)
    {
        pw[i] = pw[i - 1] * m % mod;
    }

    inv2[0] = 1;
    inv2[1] = 500000004;
    for (int i = 2; i <= n; i++)
    {
        inv2[i] = inv2[i - 1] * inv2[1] % mod;
    }

    dp[0][s[0] - 'a'] = 1;
    for (int i = 1; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            if (s[i] - 'a' == j)
            {
                dp[i][j] = (dp[i][j] + dp[i - 1][j] * inv2[1]) % mod;
            }
            else
            {
                dp[i][j] = (dp[i][j] + dp[i - 1][j] * (pw[i] - 2 + mod) % mod) % mod;
                dp[i][j] = (dp[i][j] + dp[i - 1][s[i] - 'a'] * inv2[i]) % mod;
            }
        }
    }
    cout << (dp[n - 1][0] + mod) % mod << '\n';
    return 0;
}
