#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  vector<int> dp(n + 1, 1e9);
  dp[0] = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = i; j <= n; j++) {
      dp[j] = min(dp[j], dp[j - i] + a[i - 1] * j);
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << dp[i] << " ";
  }
  cout << endl;
  return 0;
}
