import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int caseNum = 1;
    while (true) {
      int n = sc.nextInt();
      int m = sc.nextInt();
      if (n == 0 && m == 0) {
        break;
      }
      int turns = (m * (m + 1)) / 2;
      turns = (int) Math.ceil((double) turns / n);
      System.out.println("Case " + (caseNum++) + ": " + turns);
    }
  }
}
