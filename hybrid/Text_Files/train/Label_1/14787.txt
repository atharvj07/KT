#include <bits/stdc++.h>
using namespace std;

int main() {
  int t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    vector<int> a(n);
    for (int &i : a) { cin >> i; }
    bool ans = true;
    for (int i = 0; i < n; i++) {
      if (a[i] % (i + 2) == 0) { ans = false; }
    }
    cout << (ans ? "YES\n" : "NO\n");
  }
  return 0;
}
