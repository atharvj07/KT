import sys
input = sys.stdin.readline

n1, n2, n3, n4 = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = list(map(int, input().split()))
d = list(map(int, input().split()))
m1 = int(input())
ban1 = []
for i in range(m1):
    x, y = map(int, input().split())
    ban1.append((x - 1, y - 1))
m2 = int(input())
ban2 = []
for i in range(m2):
    x, y = map(int, input().split())
    ban2.append((x - 1, y - 1))
m3 = int(input())
ban3 = []
for i in range(m3):
    x, y = map(int, input().split())
    ban3.append((x - 1, y - 1))
ans = 1e9
for i in range(n1):
    for j in range(n2):
        if (i, j) in ban1:
            continue
        for k in range(n3):
            if (j, k) in ban2:
                continue
            for l in range(n4):
                if (k, l) in ban3:
                    continue
                ans = min(ans, a[i] + b[j] + c[k] + d[l])
if ans == 1e9:
    print(-1)
else:
    print(ans)
