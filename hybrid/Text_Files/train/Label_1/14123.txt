#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int w, h, n;
    cin >> w >> h >> n;
    if (w % 2 == 0 && h % 2 == 0)
      cout << ((w / 2) * (h / 2) >= n ? "YES" : "NO") << endl;
    else if (w % 2 == 0)
      cout << ((w / 2) * h >= n ? "YES" : "NO") << endl;
    else if (h % 2 == 0)
      cout << (w * (h / 2) >= n ? "YES" : "NO") << endl;
    else
      cout << (w * h >= n ? "YES" : "NO") << endl;
  }
  return 0;
}
