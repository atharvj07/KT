import string
s=input()
assert 1<=len(s)<=50000
al=list(string.ascii_uppercase)
if len(s)<26:
    print(-1)
else:
    l=[]
    for c in s:
        if c in al:
            l.append(c)
        else:
            l.append('A')
    cnt=[0]*26
    for c in l:
        cnt[ord(c)-ord('A')]+=1
    ans=''
    for c in s:
        if c in al:
            ans+=c
        else:
            for i in range(26):
                if cnt[i]==0:
                    cnt[i]+=1
                    ans+=al[i]
                    break
    if set(l)==set(string.ascii_uppercase):
        print(ans)
    else:
        print(-1)
