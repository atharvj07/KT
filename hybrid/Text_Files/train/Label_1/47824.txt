import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

public class Main {

    static final int MOD = 1000000007;

    static long[][] dp;

    // n: number of cards
    // k: number of shuffles
    // c: target card
    // t: time limit
    static long solve(int n, int k, int c, int t, int[][] shuffles) {
        dp = new long[n + 1][t + 1];
        for (int i = 1; i <= n; i++) {
            Arrays.fill(dp[i], -1);
        }
        return dfs(1, t, c, k, shuffles);
    }

    static long dfs(int top, int t, int c, int k, int[][] shuffles) {
        if (top == c && t == 0) {
            return 1;
        }
        if (t < 0 || top > c) {
            return 0;
        }
        if (dp[top][t] != -1) {
            return dp[top][t];
        }
        long ans = 0;
        for (int i = 0; i < k; i++) {
            int[] shuffle = shuffles[i];
            int a = shuffle[0];
            int b = shuffle[1];
            int d = shuffle[2];
            if (a <= top && top <= a + b - 1) {
                ans = (ans + dfs(a + b - top, t - d, c, k, shuffles)) % MOD;
            } else {
                ans = (ans + dfs(top, t - d, c, k, shuffles)) % MOD;
            }
        }
        return dp[top][t] = ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        int c = Integer.parseInt(st.nextToken());
        int t = Integer.parseInt(st.nextToken());
        int[][] shuffles = new int[k][3];
        for (int i = 0; i < k; i++) {
            st = new StringTokenizer(br.readLine());
            shuffles[i][0] = Integer.parseInt(st.nextToken());
            shuffles[i][1] = Integer.parseInt(st.nextToken());
            shuffles[i][2] = Integer.parseInt(st.nextToken());
        }
        System.out.println(solve(n, k, c, t, shuffles));
    }
}
