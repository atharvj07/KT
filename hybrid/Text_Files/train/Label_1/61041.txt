def minimum_replacements(ai_name, phone_name):
    n = len(ai_name)
    m = len(phone_name)
    replacements = 0

    i = 0
    while i <= n - m:
        if ai_name[i:i+m] == phone_name:
            replacements += 1
            i += m
        else:
            i += 1
    return replacements

ai_name = input().strip()
phone_name = input().strip()

print(minimum_replacements(ai_name, phone_name))
