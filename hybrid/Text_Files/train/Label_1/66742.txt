import math

px, py, vx, vy, a, b, c, d = map(int, input().split())

# Normalize the vector
length = math.sqrt(vx**2 + vy**2)
vx, vy = vx / length, vy / length

# Calculate the coordinates of the triangle's tip
tx, ty = px + vx * b, py + vy * b

# Calculate the vector perpendicular to (vx, vy)
px_v, py_v = -vy, vx

# Calculate the points at the base of the triangle
bx1, by1 = px + px_v * (a / 2), py + py_v * (a / 2)
bx2, by2 = px - px_v * (a / 2), py - py_v * (a / 2)

# Calculate the points at the ends of the rectangle
rx1, ry1 = px + px_v * (c / 2) - vx * d, py + py_v * (c / 2) - vy * d
rx2, ry2 = px - px_v * (c / 2) - vx * d, py - py_v * (c / 2) - vy * d

print(f"{tx:.9f} {ty:.9f}")
print(f"{bx1:.9f} {by1:.9f}")
print(f"{rx1:.9f} {ry1:.9f}")
print(f"{rx2:.9f} {ry2:.9f}")
print(f"{bx2:.9f} {by2:.9f}")
