#include <iostream>
#include <vector>
const int MAXN = 2e5 + 5;
int n, t, m, k, a[MAXN], l[MAXN], r[MAXN], d[MAXN], dp[MAXN];
std::vector<int> loc[MAXN];
int main() {
  std::cin >> m >> n >> k >> t;
  for (int i = 1; i <= m; i++) { std::cin >> a[i]; }
  for (int i = 0; i <= n + 1; i++) { dp[i] = -1; }
  for (int i = 1; i <= k; i++) {
    std::cin >> l[i] >> r[i] >> d[i];
    loc[l[i]].emplace_back(i);
  }
  dp[0] = 0;
  for (int i = 0; i <= n + 1; i++) {
    if (dp[i] == -1) continue;
    for (int j = 0; j + 1 <= n + 1; j++) {
      dp[j + 1] = std::max(dp[j + 1], dp[i]);
    }
    for (int index : loc[i]) {
      if (d[index] <= a[dp[i] + 1]) { continue; }
      for (int j = i + 1; j <= n + 1; j += 2) {
        dp[j] = std::max(dp[j], dp[i] + 1);
      }
    }
  }
  std::cout << dp[n + 1] << std::endl;
  return 0;
}
