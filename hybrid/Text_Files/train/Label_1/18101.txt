#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 10;
const ll inf = 1e16;
int n, m;
ll a[maxn];
int t[maxn], r[maxn];
bool vis[maxn];
priority_queue<ll, vector<ll>, greater<ll>> pq;
int main() {
    cin >> n >> m;
    for(int i = 1; i <= n; i++) cin >> a[i];
    for(int i = 1; i <= m; i++) cin >> t[i] >> r[i];
    for(int i = n; i >= 1; i--) {
        if(i == n) pq.push(a[i]);
        else if(vis[i + 1] && r[i] == 1) pq.push(a[i]);
        else if(vis[i + 1] && r[i] == 2) pq.push(-a[i]);
        else if(!vis[i + 1] && r[i] == 1) {
            int j = i + r[i];
            while(j <= n) {
                pq.push(a[j]);
                j++;
            }
            pq.push(a[i]);
            j = i + r[i];
            while(j <= n) {
                vis[j] = 1;
                j++;
            }
        }
        else if(!vis[i + 1] && r[i] == 2) {
            int j = i + r[i];
            while(j <= n) {
                pq.push(-a[j]);
                j++;
            }
            pq.push(-a[i]);
            j = i + r[i];
            while(j <= n) {
                vis[j] = 1;
                j++;
            }
        }
    }
    while(!pq.empty()) {
        ll x = pq.top(); pq.pop();
        if(x < 0) cout << -x << ' ';
        else cout << x << ' ';
    }
    cout << endl;
    return 0;
}
