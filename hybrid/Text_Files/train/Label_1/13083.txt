#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<int> dp(n, 0);
    for (int i = 1; i < n - 1; i++) {
        dp[i] = max({dp[i - 1], a[i], dp[i - 2] + a[i], dp[i - 1] + min(a[i - 1], a[i + 1])});
    }
    cout << *max_element(dp.begin(), dp.end());
}
