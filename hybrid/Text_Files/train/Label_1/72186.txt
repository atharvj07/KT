import sys
input = sys.stdin.readline

def dfs(u, p):
    vis[u] = True
    if u == 1: return 1
    if dist[u][0] == -1:
        t = 0
        for v in adj[u]:
            if v == p:
                continue
            if not vis[v]:
                if dfs(v, u):
                    if dist[u][0] == -1:
                        dist[u][0], dist[u][1], dist[u][2] = 2, 0, 0
                    else:
                        dist[u][0] += 1
            else:
                if dist[u][0] == -1:
                    if dist[v][0] != -1:
                        t = 1
        if t:
            dist[u][0] = -1
    return dist[u][0] != -1

for _ in range(int(input())):
    input()
    n, m = map(int, input().split())
    adj = [[] for i in range(n+1)]
    for i in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
    vis = [False] * (n+1)
    dist = [[-1] * 3 for i in range(n+1)]
    print(*[dfs(i, -1) for i in range(1, n+1)])
