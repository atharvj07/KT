import sys

t = int(sys.stdin.readline())
for _ in range(t):
    n, k = map(int, sys.stdin.readline().split())
    h = list(map(int, sys.stdin.readline().split()))
    if h[0] != 0 or h[-1] != 0:
        print("NO")
    else:
        m = max(h)
        if m <= k - 1:
            print("YES")
        else:
            print("NO")
