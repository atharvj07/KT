#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, a, b;
    cin >> n;
    priority_queue<long long> pa, pb;
    for (long long i = 0; i < n; i++) {
        cin >> a;
        pa.push(a);
    }
    for (long long i = 0; i < n; i++) {
        cin >> b;
        pb.push(b);
    }
    long long sum_a = 0, sum_b = 0;
    while (!pa.empty() && !pb.empty()) {
        if (pa.top() > pb.top()) {
            sum_a += pa.top();
            pa.pop();
            if (!pa.empty()) pa.pop();
        } else {
            sum_b += pb.top();
            pb.pop();
            if (!pb.empty()) pb.pop();
        }
    }
    while (!pa.empty()) {
        sum_a += pa.top();
        pa.pop();
    }
    while (!pb.empty()) {
        sum_b += pb.top();
        pb.pop();
    }
    cout << sum_a - sum_b << '\n';
    return 0;
}
