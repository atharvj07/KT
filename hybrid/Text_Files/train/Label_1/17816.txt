#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  priority_queue<long long> pq;
  for (int i = 0;i < n;++i) {
    long long a;
    cin >> a;
    pq.push(a);
  }
  vector<long long> ans;
  long long sum = 0;
  while (!pq.empty()) {
    long long t = pq.top();
    pq.pop();
    sum += t;
    ans.push_back(t);
    if (sum >= ans.size()) break;
  }
  for (long long x : ans) cout << x << ' ';
  cout << endl;
}
