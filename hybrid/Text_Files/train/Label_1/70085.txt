from datetime import datetime, timedelta

# Reading input
s = input().strip()
t = input().strip()

# Converting string inputs to datetime objects
current_time = datetime.strptime(s, "%H:%M")
sleep_duration = timedelta(hours=int(t[:2]), minutes=int(t[3:]))

# Calculating the time George went to bed
bed_time = current_time - sleep_duration

# Formatting and printing the output
print(bed_time.strftime("%H:%M"))
