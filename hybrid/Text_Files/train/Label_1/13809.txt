#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, sum = 0;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    sum += a[i];
  }
  double avg = sum / (double)n;
  int ans = 0;
  if (avg < 5.0) {
    while ((int)round(avg) < 5) {
      ans++;
      avg = (5 + avg) / (1 + (double)ans / n);
    }
  }
  cout << ans << endl;
}
