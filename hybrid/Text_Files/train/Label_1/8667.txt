#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
  ll n, a, dp[100005][3], ans, now = 0;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a;
    dp[i][0] = dp[i - 1][1] + a;
    if (dp[i][0] < now) dp[i][0] = now;
    dp[i][1] = dp[i - 1][0] + a;
    if (dp[i][1] > now) now = dp[i][1];
    dp[i][2] = min(dp[i - 1][2] + a, now);
  }
  ans = dp[n][2];
  for (int i = n - 1; i > 0; i--) ans = min(ans, dp[i][0] + dp[n][2] - dp[i][1]);
  cout << ans - dp[n][2] << endl;
}
