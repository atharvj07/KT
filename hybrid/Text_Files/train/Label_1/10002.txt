#include <bits/stdc++.h>
using namespace std;
int f[1005], p[1005];
int getp(int p) {
    if (p == f[p]) return p;
    return f[p] = getp(f[p]);
}
bool uni(int a, int b) {
    a = getp(a);
    b = getp(b);
    if (a == b) return true;
    if (rand() % 2) f[a] = b;
    else f[b] = a;
    return false;
}
int main() {
    int n, a, b;
    cin >> n >> a >> b;
    if (a > n || b > n || a + b > n) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    for (int i = 1; i <= n; i++) f[i] = i, p[i] = i;
    int k = 0;
    for (int i = 1; i <= a + b; i++) {
        if (k == a) {
            int s = getp(1);
            for (int j = 1; j <= n; j++)
                if (getp(j) == s) p[j] = i;
            continue;
        }
        if (i <= a) p[i] = i;
        int s = getp(1);
        for (int j = 2; j <= n; j++)
            if (getp(j) != s) {
                uni(1, j);
                p[j] = i;
                break;
            }
        k++;
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (p[i] == p[j] && i != j) cout << "1";
            else cout << "0";
        }
        cout << endl;
    }
    return 0;
}
