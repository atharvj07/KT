#include <iostream>

using namespace std;

int main() {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    double misha = max(3*a/10.0, a - 0.004 * a * c);
    double vasya = max(3*b/10.0, b - 0.004 * b * d);
    if (misha > vasya) cout << "Misha" << endl;
    else if (vasya > misha) cout << "Vasya" << endl;
    else cout << "Tie" << endl;
    return 0;
}
