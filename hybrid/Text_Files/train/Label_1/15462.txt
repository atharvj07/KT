#include <bits/stdc++.h>
using namespace std;

const int maxn = 1e5+5;
int a[maxn], n;

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    long long ans = a[n-1] - 2;
    for (int i = 1; i < n-1; i++) {
        ans = min(ans, 1ll * (a[n-1] - a[i]) / 2);
    }
    cout << ans << endl;
    return 0;
}
