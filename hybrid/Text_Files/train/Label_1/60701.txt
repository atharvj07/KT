n, m = map(int, input().split())
votes = [[] for _ in range(m)]
for _ in range(n):
    p, c = map(int, input().split())
    votes[p-1].append(c)

for v in votes:
    v.sort()

ans = float('inf')
for i in range(1, n+1):
    cost = 0
    my_votes = len(votes[0])
    needed_votes = i - my_votes
    temp = []
    for v in votes[1:]:
        if len(v) >= i:
            cost += sum(v[:len(v)-i+1])
            temp += v[len(v)-i+1:]
        else:
            temp += v
    temp.sort()
    if needed_votes > 0:
        cost += sum(temp[:needed_votes])
    ans = min(ans, cost)

print(ans)
