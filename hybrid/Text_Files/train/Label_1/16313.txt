#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, s;
    cin >> n >> s;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
      cin >> v[i].first >> v[i].second;
    }
    sort(v.begin(), v.end(), [&](const auto& a, const auto& b) {
      return a.second < b.second;
    });
    int med = v[(n - 1) / 2].second;
    long long sum = 0;
    for (auto& p : v) {
      if (sum + p.second <= s)
        sum += p.second;
      else {
        long long remain = s - sum;
        med = p.first + remain;
        break;
      }
    }
    cout << med << endl;
  }
  return 0;
}
