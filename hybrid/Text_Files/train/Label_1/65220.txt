import sys
from math import sqrt

t = int(input())
for _ in range(t):
    n = int(input())
    if n % 2 == 0:
        if n > 3:
            print('YES', n-3, 2, 1, n-2)
        else:
            print('NO')
    else:
        if n > 5:
            print('YES', n-5, 2, 2, n-3)
        else:
            print('NO')
