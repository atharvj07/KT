#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<vector<int>> states(n, vector<int>(2));
  int tot = 0;
  for (int i = 0; i < n; i++) {
    int l, r;
    cin >> l >> r;
    states[i] = {l, r};
    tot += (l == 0) + (r == 0);
  }
  tot = (tot + 1) / 2;
  cout << tot << endl;
}
