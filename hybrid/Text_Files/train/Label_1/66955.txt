x, y, n = map(int, input().split())
a, b, ans = 0, 1, (1e9, 0, 0)
for i in range(1, n+1):
    cur_a = x//y * i
    cur_b = i
    if abs(x/y - cur_a/cur_b) < ans[0] or abs(x/y - cur_a/cur_b) == ans[0] and cur_a < ans[1]:
        ans = (abs(x/y - cur_a/cur_b), cur_a, cur_b)
print(f'{ans[1]}/{ans[2]}')
