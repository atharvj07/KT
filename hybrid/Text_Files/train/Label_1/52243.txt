import java.util.Scanner;

public class Main {

    static int n, m, a[][], b[][], mb[];
    static boolean vis[][];

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        a = new int[n][n];
        b = new int[n][n];
        mb = new int[n];
        vis = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < n; j++) {
                a[i][j] = s.charAt(j) - '0';
            }
        }
        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < n; j++) {
                b[i][j] = s.charAt(j) - '0';
            }
        }
        String s = in.next();
        for (int i = 0; i < n; i++) {
            mb[i] = s.charAt(i) - '0';
        }
        in.close();
        if (!canTransform()) {
            System.out.println(-1);
            return;
        }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (a[i][j] != b[i][j] && !vis[i][j]) {
                    if (i + 1 < n && a[i + 1][j] != b[i + 1][j] && !vis[i + 1][j]) {
                        System.out.println(cnt++);
                        System.out.println("row " + i);
                        vis[i][j] = vis[i + 1][j] = true;
                    } else {
                        System.out.println(cnt++);
                        System.out.println("col " + j);
                        vis[i][j] = true;
                    }
                }
            }
        }
        System.out.println(cnt);
    }

    static boolean canTransform() {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (a[i][j] != b[i][j]) {
                    if (i + 1 < n && a[i + 1][j] != b[i + 1][j]) {
                        return true;
                    }
                }
            }
        }
        return false;
    }
}
