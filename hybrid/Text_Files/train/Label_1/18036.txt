#include <algorithm>
#include <bits/stdc++.h>
#include <cstdio>
#include <vector>

using namespace std;
const int MAXN = 2e5 + 7;
const int INF = 2e9 + 7;
int n, m, k;
vector<int> a;
vector<pair<int, int>> sp;
vector<int> dp;
void input() {
  cin >> n >> m >> k;
  dp.assign(k + 2, INF);
  dp[0] = 0;
  a.resize(n + 1);
  for (int i = 0; i < n; i++) cin >> a[i];
  sp.resize(m);
  sort(a.begin(), a.end());
  for (int i = 0; i < m; i++) cin >> sp[i].first >> sp[i].second;
}
void f(int mask) {
  for (int i = 1; i <= k; i++) {
    dp[i] = min(dp[i], dp[i - 1] + a[mask]);
  }
  for (auto& x : sp) {
    if (!(1 << mask & x.first)) continue;
    dp[x.second] = min(dp[x.second], dp[0] + a[mask]);
  }
}
int main() {
  input();
  for (int i = 0; i < n; i++) f(i);
  cout << dp[k] << "\n";
  return 0;
}
