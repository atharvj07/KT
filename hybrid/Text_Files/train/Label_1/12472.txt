#include <bits/stdc++.h>
using namespace std;

int main() {
    string n;
    cin >> n;
    int len = n.length();
    int last = n[len - 1] - '0';
    if (last % 8) {
        cout << "NO";
        return 0;
    }
    while (len > 1 && n[len - 1] == '0') len--;
    int sum = last;
    bool flag = false;
    for (int i = len - 2; i >= 0; i--) {
        int cur = n[i] - '0';
        if (!flag) {
            if (cur % 8) {
                if (cur * 10 + sum % 8) {
                    cout << "NO";
                    return 0;
                }
                flag = true;
                n[i + 1] = '0';
            }
        } else {
            sum += cur;
            if (sum % 8) {
                cout << "NO";
                return 0;
            }
        }
    }
    cout << "YES\n";
    int cnt = 0;
    for (int i = 0; i < len; i++) {
        if (n[i] != '0') {
            n[cnt] = n[i];
            cnt++;
        }
    }
    n.resize(cnt);
    cout << n << '\n';
    return 0;
}
