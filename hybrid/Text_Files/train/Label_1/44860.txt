import java.util.ArrayList;
import java.util.Arrays;
import java.util.BitSet;
import java.util.List;
import java.util.Scanner;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        BitSet isPrime = new BitSet(100001);
        for (int i = 2; i < 100001; i++) {
            if (isPrime.get(i)) {
                continue;
            }
            for (int j = i + i; j < 100001; j += i) {
                isPrime.set(j);
            }
        }
        List<Integer> primes = new ArrayList<>();
        for (int i = 2; i < 100001; i++) {
            if (!isPrime.get(i)) {
                primes.add(i);
            }
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (isPrime.get(a[i] + a[j])) {
                    b[i]++;
                    b[j]++;
                }
            }
        }
        if (Arrays.stream(b).anyMatch(x -> x < 2)) {
            System.out.println("Impossible");
            return;
        }
        int[][] c = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (isPrime.get(a[i] + a[j])) {
                    c[i][j] = 1;
                    c[j][i] = 1;
                }
            }
        }
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (c[i][j] == 1) {
                    if (d[i] == 0) {
                        d[i] = j;
                    } else if (d[j] == 0) {
                        d[j] = i;
                    }
                }
            }
        }
        List<Integer> e = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (d[i] == 0) {
                e.add(i);
            }
        }
        List<List<Integer>> f = new ArrayList<>();
        for (int i = 0; i < e.size(); i++) {
            List<Integer> g = new ArrayList<>();
            g.add(e.get(i));
            int h = d[e.get(i)];
            while (h != e.get(i)) {
                g.add(h);
                h = d[h];
            }
            f.add(g);
        }
        System.out.println(f.size());
        for (List<Integer> g : f) {
            System.out.println(g.size() + " " + g.stream().map(x -> x + 1).collect(Collectors.toList()));
        }
    }
}
