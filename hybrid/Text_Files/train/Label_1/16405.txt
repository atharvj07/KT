#include <bits/stdc++.h>
using namespace std;
#define endl '\n'

int main() {
    int q; cin >> q;
    while (q--) {
        int n, k; cin >> n >> k;
        vector<int> v(n);
        for (int i = 0; i < n; i++) cin >> v[i];
        sort(v.begin(), v.end());
        if ((v[0] < k || v.back() > v[0] + 2 * k) && (v[0] + k < v.back() || v.back() > v[0] + 2 * k)) {
            cout << -1 << endl;
            continue;
        }
        int ans = v[0] + k;
        cout << ans << endl;
    }
    return 0;
}
