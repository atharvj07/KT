#include <bits/stdc++.h>
using namespace std;

const int MXN = 1e2 + 5;
vector<int> dp(MXN, INT_MAX);

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int t;
  cin >> t;
  dp[0] = 0;
  for (int i = 1; i < MXN; i++)
    for (auto& j : {1, 2, 3}) dp[i] = min(dp[i], 1 + dp[i - j]);
  while (t--) {
    int n;
    cin >> n;
    int ans = INT_MAX;
    for (int i = 0; i < n; i++) {
      int a;
      cin >> a;
      if (a < MXN) ans = min(ans, dp[a]);
    }
    cout << ans << '\n';
  }
  return 0;
}
