#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> h(n);
  for (auto &hi : h) { std::cin >> hi; }
  int cnt = 0;
  while (!h.empty()) {
    std::vector<int> v;
    for (auto hi : h) {
      if (hi > 1) { v.push_back(hi - 1); }
    }
    h = v;
    ++cnt;
  }
  std::cout << cnt << std::endl;
}
