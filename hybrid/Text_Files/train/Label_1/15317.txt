#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  bool in_parentheses = false;
  int longest_word_outside = 0;
  int num_words_inside = 0;
  int curr_word_length = 0;
  for (char c : s) {
    if (c == '(') {
      in_parentheses = true;
    } else if (c == ')') {
      in_parentheses = false;
    } else if (isalpha(c)) {
      curr_word_length++;
    } else {
      longest_word_outside = max(longest_word_outside, curr_word_length);
      if (in_parentheses) num_words_inside++;
      curr_word_length = 0;
    }
  }
  longest_word_outside = max(longest_word_outside, curr_word_length);
  cout << longest_word_outside << " " << num_words_inside << endl;
  return 0;
}
