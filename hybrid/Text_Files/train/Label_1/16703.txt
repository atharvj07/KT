#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
  int n, k;
  cin >> n >> k;
  vector<bitset<7>> v(n);
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    v[i] = bitset<7>(s);
  }
  int mn = 7 * n, cnt = 0;
  for (int i = 0; i < n; i++) {
    int c1 = 0;
    for (int j = 0; j < 7; j++) {
      c1 += v[i][j];
    }
    int c2 = 7 - c1;
    cnt += min(c1, c2);
    mn = min(mn, c1);
    mn = min(mn, c2);
  }
  if (cnt < k || (mn < k && k % 2 == mn % 2)) {
    cout << -1;
    return 0;
  }
  vector<int> d;
  for (int i = 9; i >= 0; i--) {
    if (mn > 0) {
      d.push_back(i);
      mn--;
    }
  }
  int x = 0;
  for (int i = 0; i < n; i++) {
    int n1 = 0;
    for (int j = 0; j < 7; j++) {
      if (!v[i][j]) n1++;
    }
    if ((k - cnt) % 2 == n1 % 2) {
      for (int j = 0; j < n1; j++) {
        if (cnt < k) {
          v[i][d[x]] = 1;
          x++;
          cnt++;
        }
      }
    }
  }
  for (int i = n - 1; i >= 0; i--) {
    for (int j = 6; j >= 0; j--) {
      if (v[i][j]) cout << '1';
      else
        cout << '0';
    }
  }
  cout << endl;
  return 0;
}
