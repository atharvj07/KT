#include <bits/stdc++.h>

using namespace std;

const int mod = 1e9 + 7;
const int maxn = 22;
int n, m, k, a[maxn][maxn];
int64_t dp[maxn][maxn][2][200];

int64_t solve(int x, int y, int prv_l, int px) {
  if (x == n && y == m) {
    if (px == k) return 1;
    return 0;
  }
  int64_t &ans = dp[x][y][prv_l][px];
  if (~ans) return ans;
  ans = 0;
  if (x < n) {
    if (!prv_l) ans = (ans + solve(x + 1, y, prv_l, px ^ a[x + 1][y])) % mod;
    ans = (ans + solve(x + 1, y, 0, px ^ a[x + 1][y])) % mod;
  }
  if (y < m) {
    if (!prv_l) ans = (ans + solve(x, y + 1, prv_l, px ^ a[x][y + 1])) % mod;
    ans = (ans + solve(x, y + 1, 0, px ^ a[x][y + 1])) % mod;
  }
  return ans;
}

int main() {
  memset(dp, -1, sizeof dp);
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) cin >> a[i][j];
  }
  cout << solve(1, 1, 1, a[1][1]) << '\n';
  return 0;
}
