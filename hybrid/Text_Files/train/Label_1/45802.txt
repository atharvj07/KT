import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    int[] b = new int[n];
    int minA = 100001, maxB = 0;
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
      b[i] = sc.nextInt();
      minA = Math.min(minA, a[i]);
      maxB = Math.max(maxB, b[i]);
    }
    int ans = 0;
    for (int i = minA; i <= maxB; i++) {
      int cnt = 0;
      for (int j = 0; j < n; j++) {
        if (a[j] <= i && i <= b[j]) {
          cnt++;
        }
      }
      ans = Math.max(ans, cnt);
    }
    System.out.println(ans);
  }
}
