t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    positions = list(map(int, input().split()))
    directions = input().split()

    # Convert positions and directions to a list of tuples
    robots = [(positions[i], directions[i]) for i in range(n)]

    # Sort robots by their starting positions
    robots.sort(key=lambda x: x[0])

    # Use a stack to simulate collisions
    stack = []
    results = [-1] * n

    for pos, dir in robots:
        if dir == 'L':
            if stack and stack[-1][1] == 'R':
                # Calculate explosion time
                time = (pos - stack[-1][0]) // 2
                # Mark both robots as exploded
                results[stack[-1][2]] = time
                results[robots.index((pos, dir))] = time
                stack.pop()
            else:
                stack.append((pos, dir, robots.index((pos, dir))))
        else:
            stack.append((pos, dir, robots.index((pos, dir))))

    # Print results
    print(' '.join(map(str, results)))
