import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        String x = sc.next();
        String y = sc.next();
        System.out.println(Math.min(n, m) * x.chars().mapToLong(a -> a != y.charAt(0) ? 1 : 0).sum() + Math.max(n, m) * x.length() - Math.min(n, m) * x.length());
    }
}
