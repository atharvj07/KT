def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def make_coprime_array(n, arr):
    added_elements = []
    for i in range(n - 1):
        if gcd(arr[i], arr[i + 1]) != 1:
            added_elements.append(1)
            arr.insert(i + 1, 1)
    print(len(added_elements))
    print(' '.join(map(str, arr)))

n = int(input().strip())
arr = list(map(int, input().strip().split()))
make_coprime_array(n, arr)
