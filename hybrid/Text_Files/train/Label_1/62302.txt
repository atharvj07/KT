def dfs(graph, node, parent, dp):
    for child, cost in graph[node]:
        if child != parent:
            dfs(graph, child, node, dp)
            dp[node] = max(dp[node], dp[child] + cost)

n = int(input())
graph = [[] for _ in range(n)]
dp = [0] * n

for _ in range(n-1):
    u, v, c = map(int, input().split())
    graph[u].append((v, c))
    graph[v].append((u, c))

dfs(graph, 0, -1, dp)
print(dp[0])
