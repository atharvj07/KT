def solve_chess_problem():
    n = int(input().strip())
    board = [input().strip() for _ in range(n)]

    # Initialize the move pattern with all unvisited (.) states
    move_pattern = [['.'] * (2*n - 1) for _ in range(2*n - 1)]

    # Mark the center of the move_pattern as 'o'
    move_pattern[n-1][n-1] = 'o'

    for i in range(n):
        for j in range(n):
            if board[i][j] == 'o':
                for x in range(n):
                    for y in range(n):
                        if board[x][y] in 'x.':
                            dx = x - i
                            dy = y - j
                            # If it's an attack position, mark as 'x', else check if contradicting information exists
                            if board[x][y] == 'x':
                                move_pattern[dx + n-1][dy + n-1] = 'x'
                            elif move_pattern[dx + n-1][dy + n-1] == 'x':
                                print('NO')
                                return

    # Check if any attacked cell isn't covered by the move_pattern
    for i in range(n):
        for j in range(n):
            if board[i][j] == 'x':
                is_covered = False
                for dx in range(-n+1, n):
                    for dy in range(-n+1, n):
                        if 0 <= i+dx < n and 0 <= j+dy < n and board[i+dx][j+dy] == 'o':
                            if move_pattern[dx+n-1][dy+n-1] == 'x':
                                is_covered = True
                                break
                    if is_covered:
                        break
                if not is_covered:
                    print('NO')
                    return

    print('YES')
    for row in move_pattern:
        print(''.join(row))

solve_chess_problem()
