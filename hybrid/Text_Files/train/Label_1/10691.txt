#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> words(n);
  for (int i = 0; i < n; i++) {
    cin >> words[i];
  }
  
  sort(words.begin(), words.end(), [](string a, string b) {
    return a.size() > b.size();
  });
  
  int ans = 0;
  int used[26];
  fill(used, used + 26, 0);
  
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = 0; j < words[i].size(); j++) {
      int idx = words[i][j] - 'a';
      if (used[idx] == 0) {
        cnt++;
        used[idx] = 1;
        if (cnt > 2) {
          break;
        }
      }
    }
    
    if (cnt <= 2) {
      ans += words[i].size();
    } else {
      fill(used, used + 26, 0);
    }
  }
  
  cout << ans << endl;
  
  return 0;
}
