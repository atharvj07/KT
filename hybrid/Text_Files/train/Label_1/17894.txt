#include <iostream>
#include <vector>
using namespace std;
vector<int> v[101], a[101];
int main() {
  int n, k;
  cin >> n >> k;
  int sum = k * (n + n - 1), mn = sum / n / n;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) v[i].push_back(mn), a[i].push_back(k - 2 * mn);
  for (int i = 1; i <= n; i++)
    v[i][i] += sum % n;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) cout << v[i][j] << ' ';
    cout << endl;
  }
}
