import sys
import math

n = int(input())
a = [0] * (2 * n + 1)
b = [0] * (2 * n + 1)
for i in range(n):
    x, y = map(int, input().split())
    a[x] = y
    b[y] = x

ans = []
for i in range(1, 2 * n + 1):
    if i % 2 == 1:
        if a[i] % 2 == 1:
            ans.append([1, 2])
        else:
            ans.append([2, 1])
    else:
        if a[i] % 2 == 1:
            ans.append([2, 1])
        else:
            ans.append([1, 2])

for i in range(n):
    print(*ans[i])
