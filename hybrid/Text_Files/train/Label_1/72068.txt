q = int(input())
costs = {}

def update_cost(v, u, w):
    while v != u:
        if v < u:
            u, v = v, u
        if v not in costs:
            costs[v] = 0
        costs[v] += w
        v //= 2

def query_cost(v, u):
    cost = 0
    path_v, path_u = {}, {}
    while v:
        path_v[v] = True
        v //= 2
    while u:
        if u in path_v:
            break
        u //= 2
    common_ancestor = u
    v, u = path_v, path_u
    for node in v:
        if node in costs:
            cost += costs[node]
        if node == common_ancestor:
            break
    for node in path_u:
        if node in costs:
            cost += costs[node]
        if node == common_ancestor:
            break
    return cost

for _ in range(q):
    event = list(map(int, input().split()))
    if event[0] == 1:
        _, v, u, w = event
        update_cost(v, u, w)
    else:
        _, v, u = event
        print(query_cost(v, u))
