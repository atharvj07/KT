import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        int[] factors = new int[10];
        int cnt = 0;
        for (long i = 3; i * i <= n; i += 4) {
            if (n % i == 0) {
                factors[cnt++] = (int) i;
                while (n % i == 0) {
                    n /= i;
                }
            }
        }
        if (n > 1) {
            factors[cnt++] = (int) n;
        }
        System.out.printf("! %d", cnt);
        for (int i = 0; i < cnt; i++) {
            System.out.printf(" %d", factors[i]);
        }
        System.out.println();
        in.close();
    }
}
