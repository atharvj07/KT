import java.util.Scanner;

public class Main
{
    public static void main(String[] args)
    {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int m = scan.nextInt();
        int ans = (n - 1) % m;
        if (ans == 0)
            System.out.println(n - 1);
        else
            System.out.println(n - 1 + (m - ans));
    }
}
