import sys

n = int(input())
a = list(map(int, input().split()))

maxa = [0] * n
mina = [0] * n

maxa[0] = a[0]
mina[0] = a[0]

for i in range(1, n):
    maxa[i] = max(maxa[i - 1], a[i])

mina[n - 1] = a[n - 1]
for i in range (n - 2, -1, -1):
    mina[i] = min(mina[i + 1], a[i])

res = []

for i in range(n):
    if i == 0 or maxa[i - 1] < 2 * a[i]:
        res.append(-1)
    else:
        cnt = 1
        for j in range(i, n):
            if a[j] < maxa[i - 1] / 2:
                break
            cnt += 1
        res.append(cnt)

print(*res)
