#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, d;
    cin >> n >> d;
    vector<int> a(n + 2);
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= d; i++) {
      a[1] += a[n];
      a[n] = 0;
      for (int i = 2; i < n; i++) {
        a[i] = (a[i] + a[i - 1]) / 2;
        a[i - 1] /= 2;
      }
    }
    cout << a[1] << endl;
  }
}
