#include <bits/stdc++.h>
using namespace std;
int cal(int x, int n, int d[], int h[]){
  int l=0,r=1e9,mid,ans=-1;
  while(l<=r){
    mid=(l+r)>>1;
    int now=x;
    for(int i=0;i<n;i++){
      now-=min(mid,now);
      if(now<=0) break;
      now+=1LL*h[i]*(mid/d[i]);
    }
    if(now>0) l=mid+1;
    else r=mid-1,ans=mid;
  }
  return ans;
}
int main(){
  int t;
  cin>>t;
  while(t--){
    int n,x;
    cin>>n>>x;
    int d[n],h[n];
    for(int i=0;i<n;i++){ cin>>d[i]>>h[i];  }
    cout<<cal(x,n,d,h)<<endl;
  }
}
