#include <bits/stdc++.h>
using namespace std;
#define int long long

using ll = long long;
const ll INF = 1e18;
const int maxn = 2e5;
const int maxk = 100;

struct task {
    int p, a, t;
    bool operator<(task const &other) const {
        return t < other.t;
    }
};

int32_t main() {
    int n, T;
    cin >> n >> T;
    vector<task> tasks(n);
    for (task &task : tasks) cin >> task.a >> task.t;
    sort(tasks.begin(), tasks.end());
    vector<int> dp(T + 1, -INF);
    dp[0] = 0;
    for (task const &task : tasks) {
        for (int t = T; t >= task.t; --t) {
            dp[t] = max(dp[t], min((ll)task.a, 1ll * t / task.t + dp[t % task.t]));
        }
    }
    int mxs = *max_element(dp.begin(), dp.end());
    cout << mxs << endl;
    vector<int> ans;
    int t = T;
    for (int k = n - 1; k >= 0; --k) {
        if (tasks[k].a <= mxs) {
            while (t >= tasks[k].t && dp[t] == dp[t - tasks[k].t] + tasks[k].a) {
                ans.push_back(k);
                t -= tasks[k].t;
            }
        }
    }
    cout << ans.size() << endl;
    for (int k : ans) cout << k + 1 << ' ';
    return 0;
}
