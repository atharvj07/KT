import java.util.Scanner;

public class StaircaseBoxes {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] staircase = new int[n];
        for (int i = 0; i < n; i++) {
            staircase[i] = scanner.nextInt();
        }

        int m = scanner.nextInt();
        int[] boxHeights = new int[m];
        for (int i = 0; i < m; i++) {
            int width = scanner.nextInt();
            int height = scanner.nextInt();

            int landingHeight = findLandingHeight(staircase, boxHeights, width, height);
            boxHeights[i] = landingHeight;
            System.out.println(landingHeight);
        }
    }

    private static int findLandingHeight(int[] staircase, int[] boxHeights, int boxWidth, int boxHeight) {
        int landingHeight = 0;
        for (int i = 0; i < boxWidth; i++) {
            int stairHeight = staircase[i];
            int boxHeightAbove = (i < boxHeights.length) ? boxHeights[i] : 0;
            landingHeight = Math.max(landingHeight, Math.max(stairHeight, boxHeightAbove));
        }
        return landingHeight + boxHeight;
    }
}
