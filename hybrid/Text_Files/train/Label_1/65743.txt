n = int(input())
a = list(map(int, input().split()))
pref = [0]
for i in a:
    pref.append(pref[-1]+i)
worm_pos = [(0, 0)]
for i in range(0, n):
    for j in range(a[i]):
        worm_pos.append((i+1, pref[i]+j+1))
m = int(input())
for q in list(map(int, input().split())):
    for i, p in worm_pos:
        if p == q:
            print(i)
