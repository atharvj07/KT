#include <bits/stdc++.h>
using namespace std;

int main() {
    int a, b;
    cin >> a >> b;
    int ans = 0;
    for (int i = a; i <= b; i++) {
        string t = bitset<63>(i).to_string();
        int cnt = 0;
        for (int j = 0; j < t.size(); j++) cnt += (t[j] == '0');
        if (cnt == 1) ans++;
    }
    cout << ans << "\n";
    return 0;
}
