#include <bits/stdc++.h>
using namespace std;
int n, h, m;
struct data {
    int l, r, x;
} a[51];
int f[51][51];
int main() {
    cin >> n >> h >> m;
    for (int i = 1; i <= m; i++) {
        cin >> a[i].l >> a[i].r >> a[i].x;
    }
    memset(f, -1, sizeof f);
    f[1][0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= h; j++) {
            if (f[i][j] == -1) continue;
            f[i + 1][j] = max(f[i + 1][j], f[i][j]);
            for (int k = i; k <= n; k++) {
                bool ok = true;
                for (int l = 1; l <= m; l++) {
                    if (i <= a[l].l && a[l].r <= k) {
                        if (j > a[l].x) {
                            ok = false;
                        }
                    }
                }
                if (ok) {
                    f[k + 1][min(j + 1, h)] = max(f[k + 1][min(j + 1, h)], f[i][j] + j * j);
                } else {
                    break;
                }
            }
        }
    }
    cout << f[n + 1][0] << endl;
    return 0;
}
