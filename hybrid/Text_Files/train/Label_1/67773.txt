import sys

input = sys.stdin.readline

for t in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    s = input()
    q = [0] * n
    a, b = [], []
    for i in range(n):
        if s[i] == '0':
            a.append((p[i], i))
        else:
            b.append((p[i], i))
    a.sort()
    b.sort()
    for i in range(n):
        if i < len(a):
            q[a[i][1]] = i + 1
        else:
            q[b[i - len(a)][1]] = i + 1
    print(*q)
