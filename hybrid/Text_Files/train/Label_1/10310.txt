#include <bits/stdc++.h>
using namespace std;
const int maxn = 101;

int main() {
    int n, k;
    cin >> n >> k;
    int a[maxn];
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a, a + n, greater<int>());
    k++;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        while (k > a[i]) {
            k -= a[i];
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
