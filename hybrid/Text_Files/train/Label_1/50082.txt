import java.util.Scanner;

public class Takahashi {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextLong();
        }

        long total = 0;
        for (long value : a) {
            total += value;
        }

        long coins = 0;
        long ten = 10;
        while (total > 0) {
            if (total >= ten) {
                coins++;
                total -= ten;
            }
            ten *= 10;
        }

        System.out.println(coins);
    }
}
