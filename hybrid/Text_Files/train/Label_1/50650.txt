import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int h = sc.nextInt();
    int w = sc.nextInt();
    int[] x = new int[n];
    for (int i = 0; i < n; i++) {
      x[i] = sc.nextInt();
    }
    long ans = 0;
    for (int i = 0; i < n; i++) {
      if (i % 2 == 0) {
        ans += (long) h * w;
        if (i + 1 < n) {
          ans -= (long) h * Math.min(x[i + 1], w);
        }
      } else {
        ans += (long) h * w;
        if (i - 1 >= 0) {
          ans -= (long) h * Math.min(x[i - 1], w);
        }
      }
    }
    System.out.println(ans);
  }
}
