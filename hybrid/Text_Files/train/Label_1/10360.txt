#include <cstdio>
#include <vector>
using namespace std;
typedef long long ll;
const int maxn = 505;
vector<int> v[maxn];

int main(){
    int n, m, k;
    scanf("%d%d%d", &n, &m, &k);
    for (int i = 0; i < n; i++){
        for (int j = 0; j < m; j++){
            int x;
            scanf("%i", &x);
            v[i].push_back(x);
        }
    }
    int ans = n * m;
    for (int i = 0; i < n; i++){
        int l = -1, r = -1, sum = 0;
        for (int j = 0; j < m; j++){
            if (v[i][j]){
                if (l == -1) l = j;
                r = j;
                sum++;
            }
        }
        if (sum > k) ans -= min(sum - k, r - l + 1);
    }
    printf("%d\n", ans);
}
