#include<bits/stdc++.h>
using namespace std;
int main(){
    int n; cin >> n;
    string s1, s2; cin >> s1 >> s2;
    vector<pair<int, int>> v1(n, {0, 0}), v2(n, {0, 0}), p1(n, {0, 0}), p2(n, {0, 0});
    for(int i = 0; i < n - 1; ++i){
        if(s1[i] == 'N') v1[i + 1] = {v1[i].first, v1[i].second + 1};
        else if(s1[i] == 'E') v1[i + 1] = {v1[i].first + 1, v1[i].second};
        else if(s1[i] == 'S') v1[i + 1] = {v1[i].first, v1[i].second - 1};
        else if(s1[i] == 'W') v1[i + 1] = {v1[i].first - 1, v1[i].second};
        
        if(s2[i] == 'N') v2[i + 1] = {v2[i].first, v2[i].second + 1};
        else if(s2[i] == 'E') v2[i + 1] = {v2[i].first + 1, v2[i].second};
        else if(s2[i] == 'S') v2[i + 1] = {v2[i].first, v2[i].second - 1};
        else if(s2[i] == 'W') v2[i + 1] = {v2[i].first - 1, v2[i].second};
    }
    p1[0] = v1[0]; p2[0] = v2[0];
    for(int i = 1; i < n; ++i){
        if(p1[i - 1] == v1[i]){
            p1[i] = {p1[i - 1].first, p1[i - 1].second};
        }
        else{
            p1[i] = v1[i];
        }
        
        if(p2[i - 1] == v2[i]){
            p2[i] = {p2[i - 1].first, p2[i - 1].second};
        }
        else{
            p2[i] = v2[i];
        }
    }
    if(p1[n - 1] == v1[n - 1] && p2[n - 1] == v2[n - 1]){
        cout << "YES";
    }
    else{
        cout << "NO";
    }
}
