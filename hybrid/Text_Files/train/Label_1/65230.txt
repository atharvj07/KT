for l in sys.stdin.readlines():
    e=[int(v)-1 for v in l.strip().split()]
    if not e in d:  d[e]=[]
    else:  print('NO'); exit(0)
E=[v for v in d]
for f in E:

        if all(d[v]==d[f] for v in E if v!=s):
            print('YES')
            print(' '.join([str(v+1) for v in [x for x in f]]))
