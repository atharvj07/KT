#include <iostream>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long a, b, c, d;
        cin >> a >> b >> c >> d;
        long long ans = (a - b) * (c / d) + a;
        if (ans <= 0)
            cout << -1 << endl;
        else
            cout << ans << endl;
    }
    return 0;
}
