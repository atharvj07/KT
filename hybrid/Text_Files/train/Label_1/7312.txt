#include<bits/stdc++.h>
using namespace std;
struct node{
    int u,v;
};
int dsu[100001]={0},rk[100001]={0},ans=-1;
vector<node>adj[100001];
void initialize(int n){
    for(int i=0;i<=n;i++){
        dsu[i]=i;
        rk[i]=0;
    }
}
int find(int x){
    if(x==dsu[x]){
        return x;
    }
    return dsu[x]=find(dsu[x]);
}
void MAunion(int a,int b){
    int u=find(a);
    int v=find(b);
    if(rk[u]>rk[v]){
        dsu[v]=u;
    }
    else{
        dsu[u]=v;
        if(rk[u]==rk[v]){
            rk[v]++;
        }
    }
}
bool checkcycle(node temp,int cnt){
    if(cnt==adj[temp.u].size()){
        return false;
    }
    int x=find(temp.u);
    int y=find(temp.v);
    if(x!=y){
        MAunion(x,y);
        return 1;
    }
    else{
        if(cnt==0){
            ans=1;
        }
        return 0;
    }
}
int main(){
    int n,m;
    cin>>n>>m;
    initialize(n);
    while(m--){
        int x,y;
        cin>>x>>y;
        adj[x].push_back({x,y});
        int cnt=checkcycle({x,y},adj[x].size()-1);
        if(ans==1){
            break;
        }
    }
    cout<<ans<<'\n';
}
