import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class DimaAndInna {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] chosenNumbers = new int[m];
        for (int i = 0; i < m; i++) {
            chosenNumbers[i] = scanner.nextInt();
        }

        List<Integer> sequence = new ArrayList<>();
        int lastIndex = 0;
        for (int i = 0; i < n; i++) {
            int event = scanner.nextInt();
            if (event == -1) {
                int k = 0;
                while (k < m && chosenNumbers[k] <= sequence.size()) {
                    k++;
                }
                if (k > 0) {
                    sequence = sequence.subList(k, sequence.size());
                }
                lastIndex = 0;
            } else {
                sequence.add(event);
                lastIndex++;
            }
        }

        if (sequence.isEmpty()) {
            System.out.println("Poor stack!");
        } else {
            StringBuilder result = new StringBuilder();
            for (int i = 0; i < lastIndex; i++) {
                result.append(sequence.get(i));
            }
            System.out.println(result);
        }
    }
}
