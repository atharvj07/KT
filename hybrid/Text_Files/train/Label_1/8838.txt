#include <bits/stdc++.h>
#define si(a) scanf("%d",&a)
#define sll(a) scanf("%lld",&a)
#define ll long long
#define maxn 51
#define yes 1
#define no 0
using namespace std;
int n,q,x,l,r,ans;
int tree[maxn][maxn][2];
int  solve(int lft,int rgt){
    if(lft>rgt)return no;
    int mid=(lft+rgt)/2;
    if(lft==rgt)return mid;
    int kk=no;
    for(int i=1;i<=n;i++){
         if(kk==yes)break;
         if(mid<=tree[i][l][1]&&tree[i][l][0]<=mid){
            if(tree[i][l][0]==tree[i][r][0]&&tree[i][l][1]==tree[i][r][1]){
                kk=yes;
            }
         }
    }
    if(!kk)return no;
    for(int i=1;i<=n;i++){
        for(int j=l;j<=lft;j++){
            x=tree[i][j][0];
            if(x>=lft&&x<=rgt){
                 if(x<=mid)tree[i][j][1]=min(tree[i][j][1],mid);
                 else tree[i][j][0]=max(tree[i][j][0],mid);
            }
        }
    }
    int p=solve(lft,mid-1);
    int q=solve(mid+1,rgt);
    if(p!=-1)return p;
    if(q!=-1)return q;
    return no;
}
int main(){
    si(n);
    si(q);
    memset(tree,-1,sizeof tree);
    for(int i=1;i<=q;i++){
      si(x);si(l);si(r);si(ans);
      tree[x][l][0]=max(tree[x][l][0],l);
      tree[x][l][1]=min(tree[x][l][1],r);
      if(ans==no){
          tree[x][l][0]=-1;
          tree[x][l][1]=-1;
      }
    }
    int ans=solve(1,(1<<n)-1);
    if(ans==-1)printf("Data not sufficient!\n");
    else printf("%d\n",ans);
    return 0;
}
