#include <cstdio>
#include <algorithm>
#define MAXN 1001
#define MAXCHAR 26
using namespace std;
int main() {
    char s1[MAXN], s2[MAXN];
    int count[MAXCHAR] = {0};
    scanf(" %s %s", s1, s2);
    int n = 0, m = 0;
    for (; s1[n] != '\0'; n++) {
        count[s1[n] - 'a']++;
    }
    for (; s2[m] != '\0'; m++) {
        count[s2[m] - 'a']--;
        if (count[s2[m] - 'a'] < 0) {
            printf("-1\n");
            return 0;
        }
    }
    int ans = 0;
    for (int i = 0; i < MAXCHAR; i++) {
        ans += min(count[i], 0);
    }
    printf("%d\n", ans);
    return 0;
}
