n,m = map(int,input().split())
a = list(map(int,input().split()))
while m:
    m -= 1
    x = input().split()
    i,j = map(int,x)
    b = a[i-1:j]
    if len(b) == 1:
        print('Yes')
    elif b == sorted(b):
        print('Yes')
    elif b == sorted(b,reverse=True):
        print('Yes')
    else:
        print('No')
