import java.util.*;
import java.io.*;

public class Main {
    static int n, m, ans = 10000000;
    static char[][] lab;

    static void dfs(int x, int y, boolean[][] used, int cnt) {
        if (x < 0 || y < 0 || x >= n || y >= m || used[x][y] || lab[x][y] == '.') {
            ans = Math.min(ans, cnt);
            return;
        }
        used[x][y] = true;
        dfs(x, y + 1, used, cnt + 1);
        dfs(x + 1, y, used, cnt + 1);
        dfs(x - 1, y, used, cnt + 1);
        dfs(x, y - 1, used, cnt + 1);
        used[x][y] = false;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);

        n = sc.nextInt();
        m = sc.nextInt();
        lab = new char[n][m];

        for (int i = 0; i < n; i++) {
            lab[i] = sc.next().toCharArray();
        }

        boolean[][] used = new boolean[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!used[i][j] && lab[i][j] == '#') {
                    dfs(i, j, used, 0);
                }
            }
        }

        if (ans != 10000000) {
            out.println(ans);
        } else {
            out.println(-1);
        }

        out.close();
    }
}
