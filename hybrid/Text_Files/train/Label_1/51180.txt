import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt();
        String colors = input.next();

        int[] freq = new int[26];
        Arrays.fill(freq, 0);
        for (char color : colors.toCharArray()) {
            freq[color - 'A'] += 1;
        }

        ArrayList<Integer> uniqueColors = new ArrayList<>();
        for (int i = 0; i < 26; i++) {
            if (freq[i] > 0) {
                uniqueColors.add(i);
            }
        }

        String result = "";
        for (Integer color : uniqueColors) {
            if (freq[color] % 2 == 1) {
                result += (char) (color + 'A');
            }
        }

        if (result.length() == 0) {
            result = "B";
        }

        Collections.sort(uniqueColors);
        for (Integer color : uniqueColors) {
            if (freq[color] % 2 == 0) {
                result += (char) (color + 'A');
            }
        }

        System.out.println(result);
    }
}
