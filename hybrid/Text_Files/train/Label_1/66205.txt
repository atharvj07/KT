from sys import stdin

a, n, m = map(int, stdin.readline().strip().split())
seg = [[int(x) for x in stdin.readline().strip().split()] for i in range(n)]
umbrellas = sorted((int(x), int(y)) for x, y in (stdin.readline().strip().split() for i in range(m)))
inf = 1 << 30
dp = [[inf] * 2 for i in range(a + 5)]
dp[0][1] = 0
for position in range(a):
    for took in range(2):
        if took == 0:
            dp[position + 1][1] = min(dp[position + 1][1], dp[position][1])
        for x, w in umbrellas:
            if position < x and x < position + 2 and took == 0:
                dp[position + 1][1] = min(dp[position + 1][1], dp[position][0] + w)
            elif position <= x:
                dp[position + 1][1] = min(dp[position + 1][1], dp[position][1] + w)
                if took == 1:
                    dp[position + 1][0] = min(dp[position + 1][0], dp[position][1])
for i in seg:
    l, r = i
    dp[r][1] = inf
if dp[a][1] != inf: print(dp[a][1])
else: print(-1)
