import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] a = new int[4];
        int[] b = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int[] d = new int[2];
        int[] at = new int[2];
        for (int i = 0; i < 2; i++) {
            if (a[i] > a[i + 2]) {
                d[i] = a[i];
                at[i] = b[i + 2];
            } else {
                d[i] = a[i + 2];
                at[i] = b[i];
            }
            if (a[i] > d[i ^ 1] && b[i] > at[i ^ 1]) {
                System.out.println("Team " + (i + 1));
                return;
            }
        }
        System.out.println("Draw");
    }
}
