from collections import defaultdict

n = int(input())
rect = defaultdict(list)
xs, ys = {}, {}
for i in range(n):
    x1, y1, x2, y2 = map(int, input().split())
    xs[(x1, i)] = xs.get((x1, i), []) + [(x2, i)]
    xs[(x2, i)] = xs.get((x2, i), []) + [(x1, i)]
    ys[(y1, i)] = ys.get((y1, i), []) + [(y2, i)]
    ys[(y2, i)] = ys.get((y2, i), []) + [(y1, i)]
    rect[i].extend([(x1, y1), (x2, y2)])
for x in xs:
    xs[x].sort()
for y in ys:
    ys[y].sort()
col = {}
def is_cross():
    for x in xs.values():
        for lo, hi in x:
            if lo in col and col[lo] == col[hi]:
                return True
    for y in ys.values():
        for lo, hi in y:
            if lo in col and col[lo] == col[hi]:
                return True
    return False
curcol = 1
for x in xs.keys():
    if x not in col:
        col[x] = curcol
        curcol = (curcol + 1) % 4 + 1
for y in ys.keys():
    if y not in col:
        col[y] = curcol
        curcol = (curcol + 1) % 4 + 1
if is_cross():
    print("NO")
else:
    print("YES")
    for i in range(n):
        print(col[rect[i][0]])
