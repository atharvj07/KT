#include <vector>
#include <string>
#include <iostream>

using namespace std;

int main() {
    string s, t;
    getline(cin, s);
    getline(cin, t);
    int n = s.size(), m = t.size(), ans = 0;
    vector<int> f(m, -1);
    for (int i = 1, j = -1; i < m; i++) {
        while (j != -1 && t[i] != t[j + 1]) j = f[j];
        if (t[i] == t[j + 1]) j++;
        f[i] = j;
    }
    for (int i = 0, j = -1; i < n; i++) {
        while (j != -1 && s[i] != t[j + 1]) j = f[j];
        if (s[i] == t[j + 1]) j++;
        if (j == m - 1) {
            ans++;
            j = f[j];
        }
    }
    cout << ans << endl;
    return 0;
}
