from sys import stdin
from math import ceil

n,p,q = map(int, stdin.readline().split())
a_b = [list(map(int, stdin.readline().split())) for _ in range(n)]
a_b.sort(key = lambda x:x[1]/x[0], reverse = True)
ans = 0
for ai, bi in a_b:
    if p <= q/bi:
        ans += p/ai
        q -= p*bi/ai
        break
    else:
        ans += q/bi
        p -= q*ai/bi
        q = 0

print(ans)
