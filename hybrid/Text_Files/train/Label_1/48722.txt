import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        String s = sc.next();
        String t = sc.next();
        sc.close();
        int l = lcm(n, m);
        if (l == -1) {
            System.out.println(-1);
        } else {
            int ns = l / n;
            int mt = l / m;
            String ss = "";
            for (int i = 0; i < ns; i++) {
                ss += s;
            }
            String tt = "";
            for (int i = 0; i < mt; i++) {
                tt += t;
            }
            if (ss.equals(tt)) {
                System.out.println(l);
            } else {
                System.out.println(-1);
            }
        }
    }

    public static int lcm(int a, int b) {
        if (a == 0 || b == 0) {
            return 0;
        }
        return a * b / gcd(a, b);
    }

    public static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
