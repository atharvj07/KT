import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<int[]> polygons = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            int k = Integer.parseInt(br.readLine());
            int[] polygon = new int[k];
            for (int j = 0; j < k; j++) {
                String[] line = br.readLine().split(" ");
                int x = Integer.parseInt(line[0]);
                int y = Integer.parseInt(line[1]);
                polygon[j] = x + y * 1000000000;
            }
            polygons.add(polygon);
        }
        int q = Integer.parseInt(br.readLine());
        int[] results = new int[q];
        for (int i = 0; i < q; i++) {
            String[] line = br.readLine().split(" ");
            int l = Integer.parseInt(line[0]) - 1;
            int r = Integer.parseInt(line[1]) - 1;
            int[] sum = polygons.get(l);
            for (int j = l + 1; j <= r; j++) {
                sum = merge(sum, polygons.get(j));
            }
            results[i] = sum.length;
        }
        for (int result : results) {
            System.out.println(result);
        }
    }

    private static int[] merge(int[] p1, int[] p2) {
        int[] result = new int[p1.length + p2.length];
        int i = 0, j = 0, k = 0;
        while (i < p1.length && j < p2.length) {
            if (p1[i] <= p2[j]) {
                result[k] = p1[i];
                i++;
            } else {
                result[k] = p2[j];
                j++;
            }
            k++;
        }
        while (i < p1.length) {
            result[k] = p1[i];
            i++;
            k++;
        }
        while (j < p2.length) {
            result[k] = p2[j];
            j++;
            k++;
        }
        return result;
    }
}
