#include <bits/stdc++.h>
using namespace std;
#define mx 1000006
vector<int>adj[mx];
int ara[mx],sum[mx];

int main() {
    memset(sum,0,sizeof sum);
    int n,i,j,x,y,z;
    cin>>n;
    for(i=1; i<=n; i++) {
        cin>>ara[i];
        sum[ara[i]]++;
    }
    for(i=1; i<=n; i++) {
        for(j=i; j<=n; j++) {
            x=ara[i];
            y=ara[j];
            z=n-x-y;
            if( (x^y^z)==0 && (i<j) && (j<z) && (x+y>z) && (x+z>y) && (y+z>x) && z<=n ) {
                if( x==y && y==z ) {
                    if(sum[x]>=3) {
                        sum[x]-=3;
                    } else {
                        sum[x]-=1;
                    }
                } else if( x==y ) {
                    if(sum[x]>=2 && sum[z]>=1) {
                        sum[x]-=2;
                        sum[z]--;
                    } else {
                        sum[x]-=2;
                    }
                } else if( y==z ) {
                    if(sum[y]>=2 && sum[x]>=1) {
                        sum[y]-=2;
                        sum[x]--;
                    } else {
                        sum[y]-=2;
                    }
                } else {
                    if(sum[x]>=1 && sum[y]>=1 && sum[z]>=1) {
                        sum[x]--;
                        sum[y]--;
                        sum[z]--;
                    }
                }
            }
        }
    }
    long long int ans=0;
    for(i=1; i<=n; i++) {
        ans+=sum[i];
    }
    cout<<ans<<endl;
    return 0;
}
