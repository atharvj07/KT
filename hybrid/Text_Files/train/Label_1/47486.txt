import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    int q = Integer.parseInt(in.readLine());
    for (int i = 0; i < q; i++) {
      int n = Integer.parseInt(in.readLine());
      String[] arr = in.readLine().split(" ");
      boolean clockwise = true;
      boolean counterClockwise = true;
      for (int j = 0; j < n; j++) {
        int num = Integer.parseInt(arr[j]);
        if (j == 0 && num != 1) clockwise = false;
        if (j > 0 && num != j + 1) clockwise = false;
        if (j > 0 && num != j) counterClockwise = false;
      }
      if (clockwise || counterClockwise) System.out.println("YES");
      else System.out.println("NO");
    }
  }
}
