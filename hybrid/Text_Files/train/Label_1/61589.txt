n = int(input())
a = list(map(int, input().split()))
flag = 0
for i in range(1, n - 1):
    if i % 2 == 1 and a[i] > a[i - 1]:
        flag = 1
        break
    elif i % 2 == 0 and a[i] < a[i - 1]:
        flag = 1
        break
if flag or a[0] > a[1] or a[n - 2] > a[n - 1]:
    print("Impossible")
else:
    a.sort()
    print(*a)
