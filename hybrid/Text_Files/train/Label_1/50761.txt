import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    private static final int INF = 1_000_000_000;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] data = br.readLine().split(" ");
        int n = Integer.parseInt(data[0]);
        int m = Integer.parseInt(data[1]);
        Map<Integer, List<Integer>> adjList = new HashMap<>();
        for (int i = 0; i < m; i++) {
            data = br.readLine().split(" ");
            int u = Integer.parseInt(data[0]);
            int v = Integer.parseInt(data[1]);
            List<Integer> neighbors = adjList.getOrDefault(u, new ArrayList<>());
            neighbors.add(v);
            adjList.put(u, neighbors);
        }

        int[][] dist = new int[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            Arrays.fill(dist[i], INF);
        }

        for (int i = 1; i <= n; i++) {
            dfs(i, i, 0, dist, adjList);
        }

        int maxDist = 0;
        int[] path = new int[4];
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                for (int k = j + 1; k <= n; k++) {
                    for (int l = k + 1; l <= n; l++) {
                        int dist1 = dist[i][j];
                        int dist2 = dist[j][k];
                        int dist3 = dist[k][l];
                        int dist4 = dist[l][i];
                        int totalDist = dist1 + dist2 + dist3 + dist4;
                        if (totalDist > maxDist) {
                            maxDist = totalDist;
                            path[0] = i;
                            path[1] = j;
                            path[2] = k;
                            path[3] = l;
                        }
                    }
                }
            }
        }

        for (int city : path) {
            System.out.print(city + " ");
        }
    }

    private static void dfs(int u, int start, int distToStart, int[][] dist, Map<Integer, List<Integer>> adjList) {
        if (distToStart > dist[start][u]) {
            return;
        }
        dist[start][u] = distToStart;
        List<Integer> neighbors = adjList.getOrDefault(u, new ArrayList<>());
        for (int v : neighbors) {
            dfs(v, start, distToStart + 1, dist, adjList);
        }
    }
}
