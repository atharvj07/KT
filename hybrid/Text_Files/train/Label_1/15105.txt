#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
#define int ll

#define FOR(i, a, b) for (ll i = a; i < b; i++)
#define FORR(i, b, a) for (ll i = b; i >= a; i--)
#define REP(i, a, b) FOR(i, a, b + 1)
#define REPR(i, b, a) FORR(i, b, a - 1)
#define all(x) x.begin(), x.end()

int32_t main()
{

    int t;
    cin >> t;
    while (t--)
    {
        int a, b, k;
        cin >> a >> b >> k;
        unordered_map<int, int> x, y;
        int ans = 0;
        FOR(i, 0, k)
        {
            int p, q;
            cin >> p >> q;
            if (!x[p])
                x[p] = q;
            if (!y[q])

    }

}
