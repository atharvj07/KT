m = int(input())
s = input()

# Initialize a list to keep track of selected characters
selected_chars = []

# Loop through each segment of length m
for i in range(len(s)-m+1):
    # Find the minimum character in the current segment and its index
    min_char = min(s[i:i+m])
    selected_chars.append(min_char)

# Sort the selected characters to get the lexicographically smallest string
selected_chars.sort()

# Join and print the sorted characters
print("".join(selected_chars))
