import sys

n,m = map(int,input().split())

d = [[] for i in range(0,m+1)]
for i in range(0,n):
    a = input().split()
    d[int(a[1])].append((a[0],int(a[2])))
flag = False
for i in range(1,m+1):
    l = len(d[i])
    if l<2:
        flag = True
        print("?")
    else:
        d[i].sort(key = lambda x: -x[1])
        if l>2:
            if d[i][0][1]!=d[i][1][1] or d[i][1][1]!=d[i][2][1]:
                print(d[i][0][0],d[i][1][0])
            else:
                flag = True
                print("?")
        else:
            print(d[i][0][0],d[i][1][0])
    if flag:
        break
