n, m = map(int, input().split())

regions = {}
for _ in range(n):
    name, region, score = input().split()
    region, score = int(region), int(score)
    if region not in regions:
        regions[region] = []
    regions[region].append((score, name))

for region in range(1, m + 1):
    regions[region].sort(reverse=True)
    if len(regions[region]) > 2 and regions[region][1][0] == regions[region][2][0]:
        print("?")
    else:
        print(regions[region][0][1], regions[region][1][1])
