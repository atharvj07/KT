import java.util.*;
import java.io.*;
public class Main {
    static int[] c;
    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), p = sc.nextInt(), q = sc.nextInt();
        c = new int[n + 1];
        long max = 0, now = 0;
        for (int i = 1; i <= n; i++) {
            c[i] = sc.nextInt();
            now = Math.max((-p) * q + c[i], p * q + c[i]);
            max = Math.max(now, max);
            q = (now >= 0) ? 1 : -1;
        }
        System.out.println(max);
    }
}
