import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long I = sc.nextLong();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        long[] cnt = new long[n + 1];
        for (int i = 0; i < n; i++) {
            cnt[i + 1] = cnt[i] + (long)a[i];
        }
        long ans = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            int l = i + 1;
            int r = n - 1;
            int idx = -1;
            while (l <= r) {
                int mid = (l + r) / 2;
                long sum = cnt[mid] - cnt[i];
                long bytes = (mid - i) * (long)a[i] + sum;
                if (bytes <= I * 8) {
                    idx = mid;
                    r = mid - 1;
                } else {
                    l = mid + 1;
                }
            }
            if (idx != -1) {
                long sum = cnt[idx] - cnt[i];
                long bytes = (idx - i) * (long)a[i] + sum;
                long changed = 0;
                for (int j = i; j <= idx; j++) {
                    changed += Math.max(a[j] - a[i], 0);
                }
                for (int j = idx + 1; j < n; j++) {
                    changed += Math.max(a[j] - a[idx], 0);
                }
                ans = Math.min(ans, changed);
            }
        }
        System.out.println(ans);
    }
}
