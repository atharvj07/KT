n = int(input())
u = list(map(int, input().split()))
f = [0] * 100001
for x in u:
    f[x] ^= 1
for i in range(1, 100001):
    f[i] += f[i - 1]
ans = -1
for i in range(n):
    for j in range(n):
        if f[u[j]] - (0 if i == 0 else f[u[i - 1]]) == f[u[j]] - (0 if j == 0 else f[u[j - 1]]):
            ans = max(ans, j - i + 1)
print(ans)
