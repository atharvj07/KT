#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
const int MAX_N=1e5+1;
int n;
ll a[MAX_N];
unordered_map<ll,int> freq;

int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>a[i];
        freq[a[i]]=1;
    }
    ll ans=0;
    int cnt=0;
    vector<ll> anss;
    for(int i=0;i<n;i++){
        ll gg=a[i];
        if(freq[gg]>0){
            freq[gg]=-1;
            ll cur=gg;
            int cntt=0;
            while(cur%2==0){
                cur/=2;
                cntt++;
            }
            if(cntt>ans){
                ans=cntt;
                cnt=1;
                anss.clear();
                anss.push_back(gg);
            }else if(cntt==ans){
                cnt++;
                anss.push_back(gg);
            }
        }
    }
    cout<<cnt<<endl;
    for(int i=0;i<anss.size();i++){
        cout<<anss[i]<<' ';
    }
    cout<<endl;
}
