from collections import Counter
n, k = map(int, input().split())
if k*(n-1) % n:
    print(-1)
else:
    a = []
    for i in range(1, n+1):
        for _ in range(k):
            for j in range(1, n+1):
                if i!=j:
                    a.append([j, i])
        b = Counter(a)
        if n-1-k not in b or b[n-1-k]%k:
            print(-1)
            exit()
    print(len(a))
    for i in a: print(*i)
