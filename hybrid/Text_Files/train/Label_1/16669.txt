#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long n, p, sum = 0;
  cin >> n >> p;
  vector<long long> v(n);
  for (long long i = 0; i < n; i++) {
    cin >> v[i];
    sum += v[i];
  }
  long long ans = 0, cur = 0;
  for (long long i = 0; i < n; i++) {
    cur = (cur + v[i]) % p;
    ans = max(ans, cur + sum - cur);
  }
  cout << ans << endl;
  return 0;
}
