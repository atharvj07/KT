import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        System.out.println(s.charAt(0) == 'A' && s.indexOf('C') >= 2 && s.lastIndexOf('C') < s.length() - 1 && s.substring(1, s.indexOf('C')).matches("[a-z]+") && s.substring(s.lastIndexOf('C') + 1).matches("[a-z]+") ? "AC" : "WA");
    }
}
