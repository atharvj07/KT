import java.util.ArrayList;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int d = sc.nextInt();
    int k = sc.nextInt();
    if (n - 1 > k * (k - 1) / 2) {
      System.out.println("NO");
    } else if (n - 1 != k * (k - 1) / 2 && d > 2) {
      System.out.println("NO");
    } else {
      System.out.println("YES");
      ArrayList<Integer>[] g = new ArrayList[n + 1];
      for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
      if (k > 1) {
        for (int i = 2; i <= k; i++) g[1].add(i);
        for (int i = k + 2; i <= n; i++) {
          g[i].add(i - 1);
          g[i].add(i - 2);
        }
      } else {
        for (int i = 2; i <= n; i++) g[1].add(i);
      }
      for (int i = 1; i <= n; i++) {
        for (int j : g[i]) System.out.println(i + " " + j);
      }
    }
  }
}
