#include <iostream>
#include <vector>
using namespace std;
int main() {
  int q, n;
  cin >> q;
  while (q--) {
    cin >> n;
    vector<int> p(n);
    for (int i = 0; i < n; i++) cin >> p[i];
    vector<int> ans(n);
    for (int day = 1; day <= n; day++) {
      vector<int> next(n);
      for (int i = 0; i < n; i++) next[p[i] - 1] = p[i];
      bool ok = true;
      for (int i = 0; i < n; i++) {
        if (p[i] != next[i]) ok = false;
      }
      if (ok) {
        for (int i = 0; i < n; i++) { ans[i] = day; }
        break;
      }
      p = next;
    }
    for (int i = 0; i < n; i++) cout << ans[i] << ' ';
    cout << endl;
  }
  return 0;
}
