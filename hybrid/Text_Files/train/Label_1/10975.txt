#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define int long long
typedef long long ll;

int32_t main() {
  int n; cin >> n;
  int arr[n];
  for (int i = 0; i < n; i++) cin >> arr[i];
  int mx = 0;
  vector<int> ans(n);
  for (int i = n - 1; i >= 0; i--) {
    if (arr[i] > mx) ans[i] = 0;
    else ans[i] = mx - arr[i] + 1;
    mx = max(arr[i], mx);
  }
  for (int i = 0; i < n; i++) {
    cout << ans[i] << ' ';
  }
  cout << endl;
  return 0;
}
