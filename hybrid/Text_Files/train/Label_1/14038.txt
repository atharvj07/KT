#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin >> n;
    double ans[n];
    double sum = 0;
    for (int i = 0; i < n; i++)
    {
        cin >> ans[i];
        sum += ans[i];
    }
    for (int i = 0; i < n; i++)
    {
        double x = ans[i] - floor(ans[i]);
        if (x >= 0.5)
            ans[i] = ceil(ans[i]);
        else
            ans[i] = floor(ans[i]);
    }
    for (int i = 0; i < n; i++)
        cout << ans[i] << endl;
    return 0;
}
