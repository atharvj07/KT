#include <bits/stdc++.h>

using namespace std;
using ll = long long;
const ll mod = 998244353;

ll fpow(ll x, ll p) {
  ll ret = 1;
  while (p > 0) {
    if (p % 2 == 0) {
      x = x * x;
      x %= mod;
      p /= 2;
    } else {
      ret = ret * x;
      ret %= mod;
      p--;
    }
  }
  return ret;
}

int main() {
  ll n, m, cnt;
  cin >> n >> m >> cnt;
  vector<int> arr(m);
  for (int i = 0; i < m; i++) { cin >> arr[i]; }
  int p = 2, ans = 0;
  for (int i = 0; i <= m; i++) {
    if (i == m) ans = (ans + fpow(2, cnt)) % mod;
    else {
      ans = (ans + fpow(2, cnt - arr[i] + 1)) % mod;
      p++;
      cnt = cnt - 2 * arr[i] + 1;
    }
  }
  cout << ans << endl;
  return 0;
}
