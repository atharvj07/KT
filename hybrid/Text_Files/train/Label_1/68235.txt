import sys
input=sys.stdin.readline
for _ in range(int(input())):
    n,k,l=map(int,input().split())
    a=list(map(int,input().split()))
    p=[0,1]
    while len(p)<2*k:p.append(k-p[-1])
    t=-1
    for i in range(n+1):
        if i:
            while t+1<2*k and a[i-1]+p[t+1]<=l:t+=1
            if t==2*k-1:break
        while t+1<2*k and a[i]+p[t+1]<=l:t+=1
        if t==2*k-1:break
    print(['No','Yes'][t==2*k-1])
