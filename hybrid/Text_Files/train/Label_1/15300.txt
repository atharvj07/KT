#include<bits/stdc++.h>
using namespace std;

int main(){
	int n;
	cin>>n;
	vector<int>a(n+1),b(n+1),f(n+1);
	for(int i=1;i<=n;i++){
		cin>>a[i];
		f[a[i]]=i;
	}
	for(int i=1;i<=n;i++){
		cin>>b[i];
	}
	int k=n,ans=0;
	for(int i=1;i<=n;i++){
		if(f[b[i]]<k)	ans++;
		else	k=f[b[i]];
	}
	cout<<ans<<endl;
	return 0;
}
