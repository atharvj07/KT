#include <iostream>

int main() {
    int q;
    std::cin >> q;
    while (q--) {
        int n;
        std::cin >> n;
        std::string s, t;
        std::cin >> s >> t;
        int ans = -1;
        for (int i = 0; i < n; ++i) {
            if (s[i] != t[(i + (n >> 1)) % n]) {
                ans = (ans == -1) ? 1 : 2;
            }
        }
        std::cout << ans << std::endl;
    }
}
