#include <bits/stdc++.h>

using namespace std;

string a, f, ans;

int main() {
  int n; cin >> n;
  cin >> a; cin >> f;
  for (char c : a) {
    ans += f[c - '1'];
  }
  int len = ans.size(), cnt[len], tmp;
  for (int i = 0, j; i < len; i++) {
    tmp = 0;
    for (j = i; j < len; j++) {
      if (ans[j] != ans[i]) break;
      tmp++;
    }
    cnt[i] = tmp;
    i = j - 1;
  }
  for (int i = 1; i < len; i++) {
    cnt[i] = max(cnt[i], cnt[i - 1]);
  }
  for (int i = 0; i < len; i++) {
    for (int j = i + 1; j < len; j++) {
      if (ans[i] != ans[j]) break;
      tmp = max(tmp, cnt[i] + cnt[j] - 1);
    }
  }
  tmp = max(tmp, cnt[len - 1]);
  cout << ans.substr(0, tmp) << '\n';
  return 0;
}
