#include <iostream>
#include <vector>
using namespace std;
int solve(long long int n) {
  vector<int> v;
  while (n > 0) {
    v.push_back(n % 10);
    n /= 10;
  }
  int ans = 0;
  for (int i = v.size() - 1; i >= 0; i--) {
    if (v[i] != 0) {
      ans++;
      n -= v[i];
    }
  }
  return ans;
}
int main() {
  long long int n;
  cin >> n;
  cout << solve(n) << endl;
  return 0;
}
