#include <iostream>

int main() {
  long long n;
  char s;
  std::cin >> n >> s;
  long long r, c, ans;
  if (s == 'a' || s == 'f')
    c = 6;
  else if (s == 'b' || s == 'e')
    c = 5;
  else if (s == 'c' || s == 'd')
    c = 4;
  r = (n + 3) / 4;
  ans = (r - 1) * 6 + c;
  std::cout << ans << std::endl;
  return 0;
}
