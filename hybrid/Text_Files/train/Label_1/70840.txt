import sys

ln = sys.stdin.readline
n = int(ln())
s = ln().split()
t = [0]*n
for i in range(n):
    t[i] = [int(s[i]),i]
t.sort()
a = []
ans = 0
for x in t:
    if a and a[-1][0] == x[0]:
        a.pop()
        ans += 1
    else:
        a += [x]
if len(a) & 1:
    print(-1)
else:
    print(ans)
    for i in range(0,len(a),2):
        print(a[i][1]+1,a[i+1][1]+1)
