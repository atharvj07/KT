#include <bits/stdc++.h>
using namespace std;
int n, a[10];
string s;
int get(int x) {
  int ans = 0;
  while (x != 0) {
    if (x % 2) ans++;
    x /= 2;
  }
  return ans;
}
bool check() {
  int ans = get(a[s[0] - '0']);
  for (int i = 1; i < n; i++) {
    int cur = get(a[s[i] - '0']);
    if (cur != ans) return 0;
  }
  return 1;
}
int main() {
  cin >> n >> s;
  for (int i = 0; i < 10; i++) cin >> a[i];
  cout << (check() ? "YES" : "NO");
}
