#include <iostream>

using namespace std;

int main() {
  long long la, ra, ta, lb, rb, tb, ans = 0;
  cin >> la >> ra >> ta >> lb >> rb >> tb;
  lb -= la; rb -= ra; ta -= la; tb -= la;
  ans = max(ans, min(ra, rb) - max(la, lb));
  if (ta == tb) {
    ans = max(ans, ra - lb);
    ans = ans / ta * ta;
  } else {
    long long m = 1e18;
    if (ta > 1) m = min(m, (lb + tb + ta - 1) / ta);
    if (tb > 1) m = min(m, (lb + tb + tb - 1) / tb);
    ans = max(ans, (m - 1) * ta + ra - lb);
  }
  cout << ans << endl;
  return 0;
}
