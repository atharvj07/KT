import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int A = scanner.nextInt();
        int B = scanner.nextInt();
        int H = scanner.nextInt();
        int M = scanner.nextInt();
        scanner.close();

        double angleM = (double) M / 60 * 2 * Math.PI;
        double angleH = (double) H / 12 * 2 * Math.PI + (double) M / 60 / 12 * 2 * Math.PI;
        double angleDiff = Math.abs(angleM - angleH);
        double distance = Math.sqrt(Math.pow(A, 2) + Math.pow(B, 2) - 2 * A * B * Math.cos(angleDiff));

        System.out.println(distance);
    }
}
