k = int(input())
if k == 1:
    print("2\nNY\nYN")
else:
    n = 4
    while (n-3) * 2**(n-4) < k:
        n += 1
    print(n)
    paths = (n-3) * 2**(n-4)
    extra = k - 2**(n-4)
    matrix = [['N']*n for _ in range(n)]
    for i in range(2, n-1):
        matrix[0][i] = matrix[i][0] = 'Y'
        matrix[1][i] = matrix[i][1] = 'Y'
        for j in range(2, i):
            matrix[i][j] = matrix[j][i] = 'Y'
    for i in range(2**(n-4)):
        if i < extra:
            matrix[n-2][i+2] = matrix[i+2][n-2] = 'Y'
        else:
            matrix[n-1][i+2] = matrix[i+2][n-1] = 'Y'
    matrix[0][n-2] = matrix[n-2][0] = 'Y'
    matrix[1][n-2] = matrix[n-2][1] = 'Y'
    for row in matrix:
        print(''.join(row))
