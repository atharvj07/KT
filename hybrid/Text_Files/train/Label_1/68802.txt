n = int(input().strip())
swaps = [list(map(int, input().split())) for _ in range(n)]

# Unique positions involved in swaps
positions = set()
for a, b in swaps:
    positions.add(a)
    positions.add(b)

# Map original positions to a new contiguous range
pos_to_contig = {pos: i for i, pos in enumerate(sorted(positions))}

# Initialize list to track the final positions
final_positions = [pos_to_contig[pos] for pos in sorted(positions)]

# Apply swaps on the contiguous range
for a, b in swaps:
    i, j = pos_to_contig[a], pos_to_contig[b]
    final_positions[i], final_positions[j] = final_positions[j], final_positions[i]

# Count inversions using a Fenwick Tree (Binary Indexed Tree)
class FenwickTree:
    def __init__(self, size):
        self.size = size
        self.tree = [0] * (size + 1)

    def update(self, i, delta):
        while i <= self.size:
            self.tree[i] += delta
            i += i & -i

    def query(self, i):
        sum = 0
        while i > 0:
            sum += self.tree[i]
            i -= i & -i
        return sum

# Count inversions
inversions = 0
fenwick = FenwickTree(len(final_positions))
for i in range(len(final_positions) - 1, -1, -1):
    inversions += fenwick.query(final_positions[i])
    fenwick.update(final_positions[i] + 1, 1)

print(inversions)
