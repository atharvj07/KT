#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
#define all(x) (x).begin(), (x).end()
using ll = long long;
const int INF = 1e9;

int n;
vector<pair<int, int>> v;

int main() {
  cin >> n;
  v.resize(n);
  rep(i, n) cin >> v[i].first >> v[i].second;
  sort(all(v));
  int ans = 0;
  int now = -INF;
  for (auto p : v) {
    if (p.first >= now + p.second) {
      ans++;
      now = p.first;
    }
  }
  cout << ans << endl;
}
