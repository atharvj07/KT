import java.util.*;
import java.lang.*;

class Main {
  static final int Impossible = 0;
  static final int Possible = 1;
  static final int Jump = 1;

  public static void main(String...args) {
    Scanner input = new Scanner(System.in);

    int N = input.nextInt();
    int V = input.nextInt();
    int[] arrX = new int[N];
    for (int i = 0; i < N; i++) { arrX[i] = input.nextInt(); }
    int[] dp = new int[N];
    int MV = 200010;

    Arrays.fill(dp, Impossible);
    dp[0] = Possible;

    for (int i = 0; i < N; i++) {
      if (dp[i] == Impossible) { continue; }
      long storedWater = V;
      long minOfJumps = Math.max(1, V / 2);
      for (int j = i + 1; j < N; j++) {
        long diff = arrX[j] - arrX[i];
        if (diff > storedWater) {
          // walk
          storedWater -= diff;
          if (storedWater < 0) { break; }
        } else if (diff > 0) {
          // lump
          storedWater = storedWater / 2;
        }
        if (storedWater >= minOfJumps) { dp[j] = Possible; }
      }
    }

    for (int i = 0; i < N; i++) {
      if (dp[i] == Possible) { System.out.println("Possible"); }
      else { System.out.println("Impossible"); }
    }
  }
}
