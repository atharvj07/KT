import java.util.*;
import java.lang.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws java.lang.Exception {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++)
            arr[i] = scan.nextInt();
        Arrays.sort(arr);
        double maxNum = 0;
        for (int i = n - 1; i >= 0; i--) {
            double num = Math.sqrt(arr[i]);
            if (Math.floor(num) != Math.ceil(num)) {
                maxNum = arr[i];
                break;
            }
        }
        System.out.println((int) maxNum);
    }
}
