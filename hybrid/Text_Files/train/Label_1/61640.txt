import sys
input = sys.stdin.readline

n,k=map(int,input().split())
a=[0 for i in range(n+1)]
b=[0 for i in range(n+1)]
t=[0 for i in range(n+1)]
ans=-1
for i in range(n):
	ti,ai,bi=map(int,input().split())
	t[i+1]=ti
	a[i+1]=ai
	b[i+1]=bi

for mask1 in range(1<<n):
	books1=0
	books2=0
	total=0
	for i in range(n):
		if (1<<i)&mask1:
			total+=t[i+1]
			books1+=a[i+1]
			books2+=b[i+1]
	if books1>=k and books2>=k:
		if ans==-1 or ans>total:
			ans=total

print(ans)
