n=int(input())
s=list(map(int,input().split()))
q=int(input())
L=[]
for i in range(q):
    l,r=map(int,input().split())
    L.append([(l-1)//2,(r-1)//2])
d=[]
for i in L:
    t=0
    while(len(s[i[0]:i[1]+1])>1):
        ch=0
        for j in range(i[0],i[1],2):
            if(s[j]+s[j+1]>=10):ch+=1
            s[j]=(s[j]+s[j+1])%10
        if(ch>0):t+=1
        i[0]=i[0]//2
        i[1]=i[1]//2
    d.append(t)
for i in d:
    print(i)
