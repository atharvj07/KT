n, q = map(int, input().split())
blocked = set()

for _ in range(q):
    r, c = map(int, input().split())
    if (r, c) in blocked:
        blocked.remove((r, c))
    else:
        blocked.add((r, c))
    
    possible = True
    if len({c for r, c in blocked if r == 1}) > 1 or len({c for r, c in blocked if r == 2}) > 1:
        possible = False
    
    if possible:
        for c in range(1, n):
            if (1, c) in blocked and (2, c+1) in blocked:
                possible = False
                break
            if (2, c) in blocked and (1, c+1) in blocked:
                possible = False
                break
    
    print("Yes" if possible else "No")
