import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long a = sc.nextLong();
        long b = sc.nextLong();
        long[] h = new long[(int) n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextLong();
        }
        long ans = 0;
        for (long hi : h) {
            ans += (hi + b - 1) / a;
        }
        System.out.println(ans);
    }
}
