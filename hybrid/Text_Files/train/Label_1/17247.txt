#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(false);
    long long n, m, cnt = 0;
    cin >> n >> m;
    set<long long> s;
    for (long long x = 0; x < n; x++) {
        long long a;
        cin >> a;
        s.insert(a);
    }
    vector<long long> v;
    for (long long i = 1;; i++) {
        auto it = s.find(i);
        if (it == s.end()) {
            v.push_back(i);
            m -= i;
            cnt++;
            s.insert(i);
        }
        if (m < i || cnt == 100000) break;
    }
    cout << cnt << endl;
    for (auto x : v) cout << x << " ";
    return 0;
}
