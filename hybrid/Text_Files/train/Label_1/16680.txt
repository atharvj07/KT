#include<bits/stdc++.h>
#define ll long long
using namespace std;
int n,a[200005],b[200005];
int main()
{
	cin>>n;
	for(int i=1;i<n;i++)
	{
		cin>>a[i];
		b[i]=b[i-1]+a[i];
	}
	if(b[n-1]<=0)
	{
		cout<<-1<<endl;
		return 0;
	}
	for(int i=2;i<=b[n-1];i++)
	{
		vector<int> v1,v2;
		int x=i;
		for(int j=1;j<n;j++)
		{
			if(b[j]-b[j-1]!=0)
			{
				v1.push_back(x);
				x+=a[j];
				v2.push_back(a[j]);
			}
		}
		if(v1[v1.size()-1]==b[n-1])
		{
			v1.push_back(x);
			for(int j=v2.size()-1;j>=0;j--)
			{
				v1.push_back(v1[v1.size()-1]-v2[j]);
			}
			for(int j=0;j<v1.size();j++)
			{
				cout<<v1[j]<<" ";
			}
			cout<<endl;
			return 0;
		}
	}
	cout<<-1<<endl;
	return 0;
}
