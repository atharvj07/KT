import math

def intersection_area(w, h, alpha):
    if alpha > 90:
        alpha = 180 - alpha
    alpha = math.radians(alpha)
    if alpha == 0:
        return w * h
    cos_a, sin_a = math.cos(alpha), math.sin(alpha)
    tan_a = sin_a / cos_a
    if h * tan_a <= w and w * tan_a <= h:
        return w * h - (w * h * tan_a * math.sin(2 * alpha)) / 2
    elif h * tan_a > w:
        return (w ** 2) / (math.sin(alpha) * math.cos(alpha))
    else:
        return (h ** 2) / (math.sin(alpha) * math.cos(alpha))

w, h, alpha = map(int, input().split())
print(intersection_area(w, h, alpha))
