#include <iostream>
#include <queue>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<int> cnt(n + 1, 0);
  for (int i = 1; i <= n; i++) { cin >> cnt[i]; }

  priority_queue<pair<int, int>> pq;
  for (int i = 0; i < m; i++) {
    int x, y;
    cin >> x >> y;
    if (x > y) swap(x, y);
    pq.push({x, y});
  }

  vector<int> res;
  while (!pq.empty()) {
    auto p = pq.top();
    pq.pop();
    int x = p.first, y = p.second;
    if (cnt[x] == 0 && cnt[y] == 0) {
      cout << "DEAD" << endl;
      return 0;
    }
    if (cnt[x] > 0) {
      res.push_back(x);
      cnt[x]--;
    }
    if (cnt[y] > 0) {
      res.push_back(y);
      cnt[y]--;
    }
  }

  cout << "ALIVE" << endl;
  for (int i = 0; i < res.size(); i++) { cout << res[i] << " "; }
  cout << endl;
  return 0;
}
