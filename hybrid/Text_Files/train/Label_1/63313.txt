weights = {'q': 9, 'r': 5, 'b': 3, 'n': 3, 'p': 1, 'Q': 9, 'R': 5, 'B': 3, 'N': 3, 'P': 1}
white_weight, black_weight = 0, 0

for _ in range(8):
    line = input().strip()
    for char in line:
        if char.isupper():
            white_weight += weights.get(char, 0)
        elif char.islower():
            black_weight += weights.get(char, 0)

if white_weight > black_weight:
    print("White")
elif black_weight > white_weight:
    print("Black")
else:
    print("Draw")
