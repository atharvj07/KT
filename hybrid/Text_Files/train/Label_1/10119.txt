#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  long long sum = 0;
  for (int i = 0; i < n; ++i) { cin >> a[i]; sum += a[i]; }
  sort(a.begin(), a.end());
  long long lo = a[0], hi = *max_element(a.begin(), a.end());
  long long res = (hi - lo + 1) / 2;
  long long ex = sum - res * n;
  res += (ex + hi - 1) / hi;
  cout << res;
}
