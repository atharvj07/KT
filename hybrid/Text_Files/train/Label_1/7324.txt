#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        string s;
        cin>>s;
        bool ok=1;
        int bal=0,qcnt=0;
        for(int i=0;i<n;++i){
            if((bal==0 && s[i]=='1') || (bal<0))ok=0;
            if(s[i]=='1')++bal,++qcnt;else --bal;
        }
        if(bal!=0 || qcnt%2)ok=0;
        if (ok)cout<<"YES\n",cout<<"(",cout<<")";else cout<<"NO";
        cout<<endl;
    }
    return 0;
}
