#include<bits/stdc++.h>
using namespace std;
long long int dp[300005];
long long int arr[300005];
int main() {
    long long int n,k;
    cin>>n>>k;
    for(long long int i=0;i<n;i++) {
        cin>>arr[i];
    }
    long long int pos=k-1;
    long long int sum=arr[0];
    for(long long int i=1;i<n;i++) {
        sum+=arr[i];
        dp[i]=dp[i-1]+arr[i]*pos;
        if(i>=k) {
            dp[i]=max(dp[i],dp[i-k]+sum);
            sum-=arr[i-k];
        }
        pos--;
    }
    cout<<dp[n-1]<<endl;
    return 0;
}
