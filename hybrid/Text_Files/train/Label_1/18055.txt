#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int gcd(int a, int b) {
  if (b == 0)
    return a;
  return gcd(b, a % b);
}

int main() {
  int n;
  cin >> n;
  vector<int> arr(n);
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int g = arr[i];
    for (int j = i + 1; j < n; j++) {
      g = gcd(g, arr[j]);
      if (g == 1)
        break;
    }
    if (g != 1)
      ans++;
  }
  cout << ans << endl;
  return 0;
}
