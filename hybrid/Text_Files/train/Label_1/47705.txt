import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int a = 0;
        char b = 'A';
        if (x % 4 == 1) {
            b = 'A';
        } else if (x % 4 == 3) {
            b = 'B';
            a = 1;
        } else if (x % 4 == 2) {
            b = 'C';
            a = 2;
        } else {
            b = 'D';
            a = 3;
        }
        System.out.println(a + " " + b);
    }
}
