#include<bits/stdc++.h>
using namespace std;
int main(){
  int n,m,a[100005],ind[100005]={0};
  cin>>n>>m;
  for(int i=0;i<m;i++){
    int l,r;
    cin>>l>>r;
    ind[l]++;
    ind[r]--;
  }
  for(int i=1;i<=n;i++){
    ind[i]+=ind[i-1];
    if(ind[i]<0){
      cout<<-1;
      return 0;
    }
  }
  int ans[100005],as=0;
  for(int i=1;i<=n;i++){
    if(ind[i]==0)
      a[as++]=i;
  }
  for(int i=1;i<=n;i++){
    if(ind[i]>0)
      a[as++]=i;
  }
  cout<<a[0];
  for(int i=1;i<n;i++)
    cout<<" "<<a[i];
  cout<<"\n";
  return 0;
}
