#include <bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;
vector<int> adj[N], ar[N];
char c[N];

int main() {
  int n, m;
  cin >> n >> m;
  while (m--) {
    int a, b;
    char t;
    cin >> t;
    if (t == '+') {
      cin >> a >> b >> c[m];
      adj[a].push_back(b);
      ar[b].push_back(m);
    } else if (t == '-') {
      cin >> a >> b;
      for (int &x : ar[b])
        if (adj[a][x] == b) {
          adj[a].erase(adj[a].begin() + x);
          break;
        }
    } else {
      cin >> a;
      bool ok = 0;
      for (int &x : adj[a]) {
        if (x == a) {
          ok = 1;
          break;
        }
        int len = adj[x].size();
        bool same = 1;
        for (int j = 0; j < len; j++)
          if (adj[x][j] == a && c[ar[x][j]] != c[ar[x][(j + 1) % len]]) same = 0;
        ok |= same;
      }
      cout << (ok ? "YES\n" : "NO\n");
    }
  }
}
