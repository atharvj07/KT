import java.util.Scanner;

class Main {

    static int n, k, p, a[], sum;

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        a = new int[1024];
        n = scan.nextInt();
        k = scan.nextInt();
        p = scan.nextInt();
        for (int i = 0; i < p; ++i) {
            int t = scan.nextInt() - 1;
            if (++a[t] == 1) ++sum;
            if (sum == k) break;
        }
        for (int i = 0; i < n; ++i) {
            if (a[i] > 0) System.out.print('X');
            else System.out.print('.');
        }
        System.out.println();
    }
}
