#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n);
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        sum += a[i];
    }

    if (sum % n != 0) {
        cout << -1 << endl;
        return 0;
    }

    long long avg = sum / n;

    map<long long, vector<pair<int, int>>> intervals;
    for (int i = 0; i < n; i++) {
        intervals[a[i] - avg].push_back({i, i});
    }

    vector<pair<int, int>> operations;
    for (auto it = intervals.begin(); it != intervals.end(); it++) {
        vector<pair<int, int>>& interval = it->second;
        for (int i = 0; i < interval.size(); i++) {
            operations.push_back({interval[i].first, interval[i].second});
        }
    }

    int op = 0;
    for (int i = 0; i < 3; i++) {
        cout << operations[op].first + 1 << " " << operations[op].second + 1 << endl;
        long long b = a[operations[op].first] - avg;
        cout << b << " ";
        for (int j = operations[op].first + 1; j <= operations[op].second; j++) {
            cout << b << " ";
        }
        cout << endl;
        op++;
    }

    return 0;
}
