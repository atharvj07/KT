import sys
input = sys.stdin.readline

n = int(input())
f = list(map(int, input().split()))

for i in range(n):
    if f[i] == 0:
        for j in range(1, n+1):
            if i+1 != j and j not in f:
                f[i] = j
                break
for j in range(1, n+1):
    if j not in f:
        for i in range(n):
            if i+1 != j and f[i] == 0:
                f[i] = j
                break
for i in range(n):
    print(f[i], end=" ")
