import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int t = 0;
        int l = 1;
        int r = 106;
        for (int i = 0; i < n; i++) {
            if (Math.abs(l - a[i]) < Math.abs(r - a[i])) {
                t += Math.abs(l - a[i]);
                l = a[i];
            } else {
                t += Math.abs(r - a[i]);
                r = a[i];
            }
        }
        System.out.println(t);
    }
}
