def valid(grid, n, top, bottom, left, right):
    x1, x2 = left, right
    for i in range(top, bottom + 1):
        for j in range(left, right + 1):
            if i == top and grid[i][j] == 0 or i == bottom and grid[i][j] == 0 or j == left and grid[i][j] == 0 or j == right and grid[i][j] == 0:
                return False
            elif i != top and i != bottom and j != left and j != right and grid[i][j] == 0:
                return False
            elif i == top or i == bottom:
                if grid[i][j] not in (2, 3):
                    return False
            elif j == left or j == right:
                if grid[i][j] not in (2, 3):
                    return False
            elif 1 <= i < top or bottom < i <= n or 1 <= j < left or right < j <= n:
                if grid[i][j] == 4:
                    return False
    return True

n = int(input())
grid = [input() for i in range(n)]
ans = 'No'
for top in range(n):
    for bottom in range(top, n):
        for left in range(n):
            for right in range(left, n):
                if valid(grid, n, top, bottom, left, right):
                    ans = 'Yes'
                    break
            if ans == 'Yes':
                break
        if ans == 'Yes':
            break
print(ans)
