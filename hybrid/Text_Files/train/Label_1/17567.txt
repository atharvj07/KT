#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    long long t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        while (true) {
            string str = to_string(n);
            long long sum = 0;
            for (auto c : str) sum += c - '0';
            if (sum != 0 && __gcd(n, sum) != 1) break;
            n++;
        }
        cout << n << '\n';
    }
}
