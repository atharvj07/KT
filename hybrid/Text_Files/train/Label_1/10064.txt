#include<bits/stdc++.h>
using namespace std ;
int main(){
    int n ; cin >>n ;
    vector<int> v(n) ;
    for(int i = 0 ; i<n ; i++) cin >> v[i];
    int mx = 0 , idx = 0 ;
    for(int i = 0 ; i<n ; i++){
        if(v[i] > mx) mx = v[i] , idx = i ;
    }
    cout << idx + 1 << " " ;
    sort(v.begin(), v.end());
    cout << v[n - 2] << endl;
}
