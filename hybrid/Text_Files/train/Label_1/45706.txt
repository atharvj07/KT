import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();
        sc.close();
        int[] cnt = new int[26];
        boolean ans = false;
        for (char c : s.toCharArray()) {
			if (++cnt[c - 'A'] > k) {
				ans = true;
				break;
			}
        }
        System.out.println(ans ? "YES" : "NO");
    }
}
