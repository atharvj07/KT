#include <iostream>
#include <vector>

using namespace std;

int main() {
  long int n, a, b;
  cin >> n;

  // Create an adjacency list to store the graph.
  vector<vector<long int>> adj(n);
  for (long int i = 0; i < n - 1; i++) {
    cin >> a >> b;
    adj[a - 1].push_back(b - 1);
    adj[b - 1].push_back(a - 1);
  }

  // Count the number of lifelines.
  long int count = 0;
  for (long int i = 0; i < n; i++) {
    if (adj[i].size() == 1) {
      count++;
    }
  }

  // Print the number of lifelines.
  cout << count << endl;

  return 0;
}
