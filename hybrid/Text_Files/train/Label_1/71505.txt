n = int(input())
s = input()

# Initialize variables to keep track of Vasya's potential drinks
drinks = 0

# Iterate through the actions, starting from the fourth turn
for i in range(3, len(s)):
    # Check if the last three actions are the same and if it's Vasya's turn
    if s[i - 1] == s[i - 2] == s[i - 3] and i % n == 0:
        # Increment the count of potential drinks
        drinks += 1

# Output the number of possible drinks for Vasya
print(drinks)
