#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int maxn = 3e5 + 5;
int n, m;
vector<int> foe[maxn];
ll cnt[maxn];
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        foe[x].push_back(i);
    }
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        foe[u].push_back(foe[v][0]);
        foe[v].push_back(foe[u][0]);
    }
    for (int i = 1; i <= n; i++) {
        sort(foe[i].begin(), foe[i].end());
    }
    for (int i = 1; i <= n; i++) {
        int l = 0, r = foe[i].size() - 1;
        while (l < r) {
            int mid = (l + r) >> 1;
            if (foe[i][mid] < i)
                l = mid + 1;
            else
                r = mid;
        }
        cnt[i] = foe[i].size() - l;
    }
    ll ans = 0;
    for (int i = 1; i <= n; i++) { ans += (cnt[i] * (cnt[i] + 1)) / 2; }
    cout << ans << endl;
    return 0;
}
