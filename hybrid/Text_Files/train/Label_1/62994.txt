def solve(n, k, a):
    def mex(s):
        i = 0
        while i in s:
            i += 1
        return i

    def sg(x):
        if x == 0:
            return 0
        if x == 1:
            return 1
        if x == 2:
            return 2
        if x % 2:
            return 0
        if k % 2 == 0:
            return 1 if x == 4 else 2
        if x == 4:
            return 2
        return 1 if (x // 2) % 2 == 0 else 0

    g = 0
    for x in a:
        g ^= sg(x)
    return "Kevin" if g else "Nicky"

n, k = map(int, input().split())
a = list(map(int, input().split()))
print(solve(n, k, a))
