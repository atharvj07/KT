#include<bits/stdc++.h>
using namespace std;
#define F first
#define S second
#define ll long long
const int N=1e6+5;
ll x,n,m,d[N],w[N],l[N],r[N];
vector<pair<ll,ll> >a,b;
int main()
{
    cin>>x>>n>>m;
    for(int i=0;i<n;i++)
    {
        cin>>l[i]>>r[i];
        a.push_back({l[i],-1});
        a.push_back({r[i]+1,1});
    }
    for(int i=0;i<m;i++)
    {
        cin>>w[i]>>d[i];
        b.push_back({d[i],w[i]});
    }
    sort(b.begin(),b.end());
    sort(a.begin(),a.end());
    ll j=0,ans=0;
    for(auto t:a)
    {
        while(j<m&&b[j].F<=t.F)
        {
            ans+=b[j].S;
            j++;
        }
        if(t.S==1) ans-=b[j-1].S;
        if(ans<0)
        {
            cout<<-1;
            return 0;
        }
    }
    cout<<ans;
    return 0;
}
