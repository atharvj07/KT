#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i];
        if (sum * 2 >= (i + 1) * sum) {
            std::cout << i + 1 << "\n";
            return 0;
        }
    }
    return 0;
}
