#include <cstdio>
#include <algorithm>

int main() {
    int t;
    scanf("%d", &t);
    while (t--) {
        int n;
        scanf("%d", &n);
        int a[n], mx[n][2];
        for (int i = 0; i < n; ++i) {
            scanf("%d", a + i);
            mx[i][0] = mx[i][1] = a[i];
        }
        for (int i = 1; i < n; ++i) {
            for (int j = 0; j < i; ++j) {
                if (a[i] > mx[j][0]) {
                    mx[i][1] = std::max(mx[i][1], mx[j][0]);
                } else if (a[i] >= mx[j][1]) {
                    mx[i][0] = std::max(mx[i][0], mx[j][1]);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; ++i) {
            ans = std::max({ans, mx[i][0], mx[i][1]});
        }
        printf("%d\n", ans);
    }
    return 0;
}
