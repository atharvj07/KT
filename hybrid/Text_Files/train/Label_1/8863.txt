#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, sum = 0, ans = 0;
    cin >> n;
    vector<int> v(n);
    for (auto &i : v) { cin >> i, sum += i; }
    int mean = sum / n;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        int rem = sum - v[i] - v[j];
        int rlen = n - 2;
        int rmean = rem / rlen;
        ans += (rmean == mean);
      }
    }
    cout << ans << endl;
  }
  return 0;
}
