#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5, K = 2e5 + 5, M = 2e5 + 5;
int32_t n, m, k;
int32_t par[N], sz[N];

int32_t find(int32_t u) {
    if (u == par[u]) return u;
    return par[u] = find(par[u]);
}
void merge(int32_t u, int32_t v) {
    int32_t x = find(u), y = find(v);
    if (x != y) {
        if (sz[x] > sz[y]) {
            par[y] = x;
            sz[x] += sz[y];
        } else {
            par[x] = y;
            sz[y] += sz[x];
        }
    }
}
int main() {
    cin >> n >> m >> k;
    iota(par, par + N, 0);
    fill(sz, sz + N, 1);
    for (int i = 0; i < m; i++) {
        int32_t u, v;
        cin >> u >> v;
        merge(u, v);
    }
    vector<int32_t> ans(m, 0);
    for (int i = m - 1; i >= 0; i--) {
        int32_t u = find(1);
        ans[i] = sz[u] > k ? sz[u] : 0;
        merge(u, (u + 1) % n + 1);
    }
    for (int32_t i = 0; i < m; i++) { cout << ans[i] << '\n'; }
    return 0;
}
