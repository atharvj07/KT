import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt(), D = in.nextInt() - 1;
        int[] S = new int[N], P = new int[N];
        for (int i = 0; i < N; i++) {
            S[i] = in.nextInt();
        }
        for (int i = 0; i < N; i++) {
            P[i] = in.nextInt();
        }
        Map<Integer, Integer> ranking = new HashMap<>();
        int score = S[D] + P[0];
        for (int i = 1; i < N; i++) {
            if (!ranking.containsKey(S[i] + P[i])) {
                ranking.put(S[i] + P[i], i + 1);
            }
        }
        for (int i = 1; i < N; i++) {
            if (!ranking.containsKey(S[i] + P[i])) {
                ranking.put(S[i] + P[i], i + 1);
            }
            if (S[D] + P[i] == S[i] + P[i]) {
                score = S[i] + P[i];
            }
        }
        System.out.println(ranking.get(score));
    }
}
