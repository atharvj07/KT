import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int max = Math.max(a, b);
        int len = 0;
        while (max > 0) {
            len++;
            max /= 10;
        }
        System.out.println(len);
    }
}
