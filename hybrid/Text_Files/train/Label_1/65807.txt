n, m = map(int, input().split())
graph = [[] for i in range(n+1)]
for i in range(m):
    a, b, color = input().split()
    a, b = int(a), int(b)
    if color=='R': graph[a].append(b)
    else: graph[b].append(a)

color = [0]*(n+1)

def dfs(node, col, pre):
    if color[node]!=0:
        return color[node]==col
    color[node]=col
    ok=True
    for u in graph[node]:
        if u==pre: continue
        ok&=dfs(u, col^3, node)
    return ok

for i in range(1, n+1):
    if color[i]==0:
        if not dfs(i, 1, 0) and not dfs(i, 2, 0):
            print(-1); exit()

ans=[]
def dfs2(node, col, pre):
    if color[node]==col:
        ans.append(node)
        color[node]=0
        for u in graph[node]:
            if u==pre: continue
            dfs2(u, col, node)

for i in range(1, n+1):
    if color[i]!=0:
        dfs2(i, color[i], 0)
        break

print(len(ans))
print(*ans)
