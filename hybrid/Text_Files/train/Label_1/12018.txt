#include <bits/stdc++.h>
using namespace std;

const int maxn = 55, maxh = 55, maxm = 55;

int n, h, m, l[maxm], r[maxm], x[maxm], dp[maxn][maxh], mark[maxm][maxh];
bool vis[maxm][maxh];

int go(int pos, int mxh) {
    if (pos > n) return 0;
    int &ans = dp[pos][mxh];
    if (vis[pos][mxh]) return ans;
    vis[pos][mxh] = 1;

    for (int i = 0; i <= min(h, mxh); i++) {
        bool ok = 1;
        for (int j = 0; j < m; j++) {
            if (l[j] <= pos && pos <= r[j]) {
                if (x[j] < i) ok = 0;
            }
        }
        if (ok) ans = max(ans, i * i + go(pos + 1, i));
    }

    for (int p = 0; p < m; p++) {
        if (pos <= r[p]) {
            if (mark[p][mxh] == 0) {
                mark[p][mxh] = 1;
                ans = max(ans, go(pos, x[p]));
            }
        }
    }
    return ans;
}

int main() {
    cin >> n >> h >> m;
    for (int i = 0; i < m; i++) { cin >> l[i] >> r[i] >> x[i]; }
    cout << go(1, h) << '\n';
    return 0;
}
