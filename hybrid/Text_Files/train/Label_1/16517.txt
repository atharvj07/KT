#include <bits/stdc++.h>
#define int long long

using namespace std;

const int N = 2e5+5;
int a[N], b[N];
int c[N], l[N], r[N];
int p[N], x[N], y[N];
int ans[N];
int need[N], sumneed[N];
int ansneed[N];
int32_t main() {
    int n, m, s;
    cin >> n >> m >> s;
    for (int i = 0; i < m; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    for (int i = 0; i < n; i++) cin >> c[i];
    sort(c, c + n);
    reverse(c, c + n);
    int cnt = 0;
    for (int i = 0; i < n; i++)
        if (c[i]) {
            x[cnt] = c[i];
            l[cnt] = -1;
            r[cnt] = i;
            cnt++;
        }
    for (int i = 0; i < m; i++) {
        y[i] = a[i];
        p[i] = i;
    }
    sort(x, x + cnt);
    sort(y, y + m);
    int totneed = 0;
    for (int i = 0; i < cnt; i++) {
        totneed += x[i];
    }
    if (s < totneed) {
        cout << "NO";
        return 0;
    }
    cout << "YES" << '\n';
    int j = 0;
    for (int i = 0; i < cnt; i++) {
        while (j < m && y[j] < x[i]) j++;
        for (int k = l[i]+1; k <= r[i]; k++) {
            need[k] = x[i];
            sumneed[k] += x[i];
        }
        need[j] = x[i];
        sumneed[j] += x[i];
        ans[p[j]] = j+1;
        j++;
        if (sumneed[j] > s) break;
    }
    for (int i = 0; i < m; i++) {
        if (!ans[i]) {
            ans[i] = j+1;
        }
    }
    for (int i = 0; i < m; i++) {
        ansneed[ans[i]] += need[i];
        if (ansneed[ans[i]] > s) {
            cout << "NO";
            return 0;
        }
    }
    for (int i = 0; i < m; i++) {
        cout << ans[i] << ' ';
    }
    return 0;
}
