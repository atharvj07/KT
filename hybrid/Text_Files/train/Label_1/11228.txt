#include <bits/stdc++.h>
using namespace std;
#define ll long long int
int main() {
  int n;
  cin >> n;
  ll s = 0, a = 0;
  double avg;
  while (n--) {
    int t;
    cin >> t;
    if (t == 1) {
      ll aa, x;
      cin >> aa >> x;
      s += (min(a, aa) * x);
    }
    if (t == 2) {
      ll k;
      cin >> k;
      s += k;
      a++;
    }
    if (t == 3) {
      s -= a;
      a--;
    }
    avg = 1. * s / a;
    printf("%.6f\n", avg);
  }
  return 0;
}
