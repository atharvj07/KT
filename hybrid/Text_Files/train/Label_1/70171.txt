for _ in range(int(input())):
    n = int(input())
    s = input()
    is_palindrome = True
    for i in range(n // 2):
        if s[i] != s[n - i - 1]:
            if s[i] == chr(ord('z')):
                s = s[:i] + chr(ord('a')) + s[i + 1:]
            else:
                s = s[:i] + chr(ord(s[i]) + 1) + s[i + 1:]

            if s[i] != s[n - i - 1]:
                is_palindrome = False
                break
    print(['NO', 'YES'][is_palindrome])
