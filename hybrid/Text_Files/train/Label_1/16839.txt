#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int mod = 998244353;

int get(int n, vector<ll>& pre) {
  int last = n - 1;
  while (last >= 0 && pre[last] == 0) last--;
  return last;
}

int main() {
  int n;
  cin >> n;

  map<ll, int> cnt;
  vector<ll> pre(n);
  for (int i = 0; i < n; i++) {
    ll x;
    cin >> x;
    if (i > 0) pre[i] = pre[i - 1];
    pre[i] += cnt[x] == 0;
    cnt[x]++;
  }

  ll ans = 1;
  int last = get(n, pre);
  for (int i = last + 1; i < n; i++) {
    if (pre[i] == pre[i - 1]) ans = (ans * 2) % mod;
  }

  cout << ans << endl;
  return 0;
}
