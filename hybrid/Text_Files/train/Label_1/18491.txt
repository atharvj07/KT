#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 2e5+1, mod = 1e9+7;
int n, par[N], sub[N], ans, deg[N];
bool gone, vis[N];
vector<int> v[N];
void dfs1(int x, int p) {
    sub[x] = 1;
    for(auto y:v[x]) {
        if(y^p) {
            dfs1(y, x);
            sub[x] += sub[y];
        }
    }
}
void dfs3(int x, int type, int p) {
    if(type == 1) ans += sub[x]*(n-sub[x]);
    for(auto y:v[x]) {
        if(y^p) {
            gone = false;
            dfs3(y, type^1, x);
            if(gone == false) deg[x]^=1;
        }
    }
}
void dfs2(int x, int p, int state) {
    if(v[x].size() == 1 && x != 1) {
        ans += n-deg[x]-vis[x];
        return;
    }
    for(auto y:v[x]) {
        if(y^p) {
            gone = true;
            if(state) vis[x] = true;
            dfs2(y, x, state^1);
            if(gone == false) deg[x]^=1;
        }
    }
}
signed main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cin >> n;
    for(int i = 0;i<n-1;i++) {
        int x, y, z;
        cin >> x >> y >> z;
        v[x].push_back(y);
        v[y].push_back(x);
        par[y] = x;
    }
    dfs1(1, 1);
    dfs3(1, 1, 1);
    dfs2(1, 1, 0);
    cout << ans << endl;
    return 0;
}
