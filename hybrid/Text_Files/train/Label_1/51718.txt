import java.util.Arrays;
import java.util.Scanner;

public class Main {

    private static int[] bitCount;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long i = scanner.nextLong();
        int[] a = new int[n];
        for (int j = 0; j < n; j++) {
            a[j] = scanner.nextInt();
        }
        scanner.close();

        Arrays.sort(a);
        bitCount = new int[40];
        for (int v : a) {
            int b = 32;
            while (v > 0) {
                if ((v & 1) == 1) {
                    bitCount[b]++;
                }
                v >>= 1;
                b--;
            }
        }

        long lo = 1;
        long hi = 1000000000;
        while (lo <= hi) {
            long mi = (lo + hi) / 2;
            if (check(mi, a, i)) {
                hi = mi - 1;
            } else {
                lo = mi + 1;
            }
        }

        int ans = 0;
        for (int j = 0; j < 40; j++) {
            ans += Math.min(bitCount[j], lo - 1);
        }

        System.out.println(ans);
    }

    private static boolean check(long threshold, int[] a, long i) {
        long bits = 0;
        for (int v : a) {
            int b = 32;
            long count = 0;
            while (v > 0) {
                if ((v & 1) == 1) {
                    if (count < threshold) {
                        bits++;
                    }
                }
                v >>= 1;
                b--;
                count++;
            }
        }

        return bits <= 8 * i;
    }
}
