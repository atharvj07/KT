#include <fstream>
#include <iostream>
#include <queue>
#include <vector>

using namespace std;

const int MAXN = 10;
int d[MAXN][MAXN];

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      cin >> d[i][j];
    }
  }
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    int vis[MAXN] = {};
    queue<int> q;
    vis[i] = 1;
    q.push(i);
    while (!q.empty()) {
      int now = q.front();
      q.pop();
      for (int j = 0; j < n; ++j) {
        if (i == j) continue;
        if (vis[j]) continue;
        if (d[now][j] == 0) continue;
        vis[j] = 1;
        q.push(j);
        d[i][j] = d[i][now] + d[now][j];
        ans = max(ans, d[i][j]);
      }
    }
  }
  cout << ans << endl;
  return 0;
}
