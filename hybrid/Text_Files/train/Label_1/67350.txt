import sys

def solve():
    n = int(input())
    a = list(map(int, input().split()))
    inc = []
    dec = []
    for i in range(n):
        if i == 0 or a[i] > a[i - 1]:
            inc.append(a[i])
        elif a[i] < a[i - 1]:
            dec.append(a[i])
        else:
            return "NO"
    if len(inc) == 1 and len(dec) == 1:
        return "NO"
    return "YES\n" + " ".join(['0'] * len(inc) + ['1'] * len(dec))

print(solve())
