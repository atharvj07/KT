#include <iostream>
#include <queue>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];

  bool bought[n + 1] = {};
  queue<int> q;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (!bought[a[i]]) {
      if (q.size() < k) {
        q.push(a[i]);
        bought[a[i]] = true;
        ans++;
      } else {
        int j = q.front();
        q.pop();
        bought[j] = false;
        q.push(a[i]);
        bought[a[i]] = true;
      }
    }
  }

  cout << ans << endl;

  return 0;
}
