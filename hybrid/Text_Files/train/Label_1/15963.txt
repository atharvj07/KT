#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, p, k;
        cin >> n >> p >> k;
        vector<int> a(n);
        for(int i = 0; i < n; i++) {
            cin >> a[i];
        }
        sort(a.begin(), a.end());
        n = unique(a.begin(), a.end()) - a.begin();
        int i=0, x=p;
        while (i < n) {
            if (x - a[i] < 0) {
                break;
            }
            if (i + 1 == n || x - a[i + 1] < 0) {
                p -= a[i];
                ++i;
            } else {
                p -= a[i] + a[i + 1];
                i += 2;
            }
        }
        cout << i << endl;
    }
    return 0;
}
