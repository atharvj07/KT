import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
if a[0] == 0:
    print(-1)
else:
    d, ans = n, []
    while d > 0:
        j = -1
        for i in range(d):
            if a[i] >= d - i and (j == -1 or a[i] > a[j] or (a[i] == a[j] and b[i] < b[j])):
                j = i
        if j == -1:
            print(-1)
            exit()
        d -= a[j] + 1
        ans.append(d)
    print(len(ans))
    print(*ans)
