#include <iostream>
#include <stack>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    stack<char> st;
    vector<int> pref(s.length());
    for (int i = 0; i < s.length(); i++) {
      if (s[i] == '[' or s[i] == '(')
        st.push(s[i]);
      else if (s[i] == ')' and (!st.empty() and st.top() == '('))
        pref[i] = 1, st.pop();
      else if (s[i] == ']' and (!st.empty() and st.top() == '['))
        pref[i] = 1, st.pop();
    }
    int ans = 0;
    for (int i = 1; i < s.length(); i++) pref[i] += pref[i - 1];
    for (int i = 0; i < s.length();) {
      if (pref[i] < pref[s.length() - 1])
        i++;
      else {
        int j = i;
        while (j < s.length() and pref[j] == pref[s.length() - 1]) j++;
        ans++;
        i = j;
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
