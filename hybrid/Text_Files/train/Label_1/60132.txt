import sys

n, p, q = map(int, sys.stdin.readline().split())
a = [0] * n
b = [0] * n
for i in range(n):
    a[i], b[i] = map(int, sys.stdin.readline().split())

a.sort(reverse=True)
b.sort(reverse=True)

ans = 0
exp = 0
money = 0

for i in range(n):
    if money < q:
        if exp >= p:
            break
        exp += a[i]
        money += b[i]
        ans += 1
    if exp >= p and money >= q:
        break

if exp < p:
    ans = -1

print(ans)
