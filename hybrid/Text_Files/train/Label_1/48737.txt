import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int pos = 0, neg = 0;

        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            if (a[i] > 0)
                pos++;
            else if (a[i] < 0)
                neg++;
        }

        if (pos >= (n + 1) / 2) {
            System.out.println(1);
        } else if (neg >= (n + 1) / 2) {
            System.out.println(-1);
        } else {
            System.out.println(0);
        }
    }
}
