import java.util.Scanner;

public class Main {
    static int solve(int n1, int n2, int k1, int k2) {
        int mod = 100000000;
        long[][][] dp = new long[n1 + n2 + 1][n1 + 1][n2 + 1];
        dp[0][0][0] = 1;
        for (int i = 1; i <= n1 + n2; ++i) {
            for (int a = 0; a <= n1; ++a) {
                for (int b = 0; b <= n2; ++b) {
                    int foot = a + (i <= n1 ? 1 : 0);
                    int horse = b + (i <= n2 ? 1 : 0);
                    long sum = (dp[i - 1][a][b] + ((a > 0 && foot - 1 <= k1) ? dp[i - 1][a - 1][b] : 0) + ((b > 0 && horse - 1 <= k2) ? dp[i - 1][a][b - 1] : 0)) % mod;
                    dp[i][a][b] = sum;
                }
            }
        }
        return (int) (dp[n1 + n2][n1][n2] % mod);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n1 = sc.nextInt();
        int n2 = sc.nextInt();
        int k1 = sc.nextInt();
        int k2 = sc.nextInt();
        System.out.println(solve(n1, n2, k1, k2));
    }
}
