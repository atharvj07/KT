#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    multiset<int> a;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        a.insert(x);
    }
    if (n % k != 0) {
        cout << "No" << endl;
        return 0;
    }
    int target = 0;
    for (auto x : a) target += x;
    target /= k;
    vector<int> ans;
    int cur = 0;
    for (auto x : a) {
        cur += x;
        if (cur == target) {
            ans.push_back(ans.size() + 1);
            cur = 0;
        }
    }
    if (ans.size() != k) {
        cout << "No" << endl;
        return 0;
    }
    cout << "Yes" << endl;
    for (auto x : ans) cout << x << " ";
    cout << endl;
}
