import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n];
        long[] maxRight = new long[n];
        long[] minRight = new long[n];
        long[] maxLeft = new long[n];
        long[] minLeft = new long[n];
        long[] dp = new long[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }
        maxRight[n - 1] = h[n - 1];
        minRight[n - 1] = h[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            maxRight[i] = Math.max(maxRight[i + 1], h[i]);
            minRight[i] = Math.min(minRight[i + 1], h[i]);
        }
        maxLeft[0] = h[0];
        minLeft[0] = h[0];
        for (int i = 1; i < n; i++) {
            maxLeft[i] = Math.max(maxLeft[i - 1], h[i]);
            minLeft[i] = Math.min(minLeft[i - 1], h[i]);
        }
        dp[0] = 0;
        for (int i = 1; i < n; i++) {
            if (minLeft[i - 1] < h[i] && h[i] < maxRight[i]) {
                dp[i] = dp[i - 1];
            } else {
                dp[i] = dp[i - 1] + 1;
            }
        }
        System.out.println(dp[n - 1]);
    }
}
