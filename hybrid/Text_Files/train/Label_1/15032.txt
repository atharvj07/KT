#include <iostream>
#include <algorithm>
using namespace std;

int a[500000], n, m, d, i, j, ans, cur, p;

int main() {
  cin >> n >> m >> d;
  for (i = 0; i < n; ++i) { cin >> a[i]; }
  sort(a, a + n);
  i = 0;
  while (i < n) {
    p = a[i], cur = 0;
    while (i < n && a[i] <= p + d) ++i, ++cur;
    if (cur >= m) ++ans;
  }
  if (ans >= 1)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
