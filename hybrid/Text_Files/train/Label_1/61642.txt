for _ in range(int(input())):
    n, m = map(int, input().split())
    d = {}
    for i in range(n):
        a, b, c, d = map(int, input().split())
        if a != b and a != d and b != c and c != d:
            d[a, b] = 1
            d[c, d] = 1
        if a == c and b == d and a != b:
            d[a, b] = 1
    if len(d) * 2 >= m * m:
        print("YES")
    else:
        print("NO")
