import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt(), K = in.nextInt();
        int[] X = new int[N], R = new int[N], Q = new int[N];
        for (int i = 0; i < N; i++) {
            X[i] = in.nextInt();
            R[i] = in.nextInt();
            Q[i] = in.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < N; i++) {
            for (int j = i + 1; j < N; j++) {
                if (Math.abs(Q[i] - Q[j]) > K) continue;
                if (X[i] - R[i] <= X[j] && X[j] <= X[i] + R[i]) ans++;
                else if (X[j] - R[j] <= X[i] && X[i] <= X[j] + R[j]) ans++;
            }
        }
        System.out.println(ans);
    }
}
