import sys
input = sys.stdin.readline

n,k = map(int,input().split())
a = list(map(int,input().split()))
g = [[] for i in range(n+1)]
for i in range(n-1):
    u,v = map(int,input().split())
    g[u]+=[v]
    g[v]+=[u]

used = [False for i in range(n+1)]
size = [0 for i in range(n+1)]
def dfs(u,p,dis):
    used[u] = True
    size[u] = 1
    ans = a[u-1]
    for v in g[u]:
        if v==p or used[v]: continue
        dis+=1
        ans+=dfs(v,u,dis)
    if dis>k:
        return 0
    return ans

mx = 0
for i in range(1,n+1):
    if used[i]: continue
    mx = max(mx,dfs(i,-1,0))
print(mx)
