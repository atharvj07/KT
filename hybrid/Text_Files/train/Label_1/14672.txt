#include <bits/stdc++.h>
using namespace std;

const int MAXN = 300005;

int main() {
  long long n, x;
  char c;
  vector<pair<long long, char>> v;
  cin >> n;
  long long ans = 0;
  map<char, long long> last;
  for (long long i = 0; i < n; i++) {
    cin >> x >> c;
    v.push_back(make_pair(x, c));
    last[c] = x;
  }
  sort(v.begin(), v.end());
  for (long long i = 0; i < n; i++) {
    c = v[i].second;
    if (c == 'B') {
      ans += last['G'] - v[i].first;
      last.erase('G');
    } else if (c == 'R') {
      ans += last['G'] - v[i].first;
      last.erase('B');
    }
  }
  cout << ans;
  return 0;
}
