import sys
input = sys.stdin.readline

n = int(input())
a = [input() for _ in range(n)]
b = [input() for _ in range(n)]
ans = 0
for i in range(n):
    if a[i] != b[i]:
        if a[i] != 'M' and a[i][-1] != b[i][-1]:
            ans += 1
        else:
            ans += 0 if len(a[i]) == len(b[i]) else 1
print(ans)
