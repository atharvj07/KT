from collections import defaultdict

def dfs(node, graph, leaf_count):
    if len(graph[node]) == 0:
        leaf_count[node] = 1
        return 1
    total = 0
    for child in graph[node]:
        total += dfs(child, graph, leaf_count)
    leaf_count[node] = total
    return total

def solve(n, parents):
    graph = defaultdict(list)
    for i, p in enumerate(parents, 2):
        graph[p].append(i)
    
    leaf_count = [0] * (n + 1)
    dfs(1, graph, leaf_count)
    
    leaf_counts = list(sorted(leaf_count[1:], reverse=True))
    answers = []
    for k in range(1, n + 1):
        sum_leaves = 0
        for i, count in enumerate(leaf_counts):
            sum_leaves += count
            if i + 1 >= k:
                break
        answers.append(sum_leaves)
    return answers

n = int(input().strip())
parents = list(map(int, input().split()))
print(*solve(n, parents))
