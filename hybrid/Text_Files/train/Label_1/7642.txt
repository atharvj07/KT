#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int cnt = 0, ans = 0;
    for (int i = 0; i < n; i++) {
      if (i > 0 && s[i] != s[i - 1]) cnt = 0;
      if (s[i] == '1') ans = max(ans, ++cnt);
    }
    cout << ans << '\n';
  }
  return 0;
}
