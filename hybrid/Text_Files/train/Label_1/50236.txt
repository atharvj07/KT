import java.util.*;

class Main {
  private static class Event {
    public final int a;
    public final int b;
    public final int c;

    public Event(int a, int b, int c) {
      this.a = a;
      this.b = b;
      this.c = c;
    }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    int M = sc.nextInt();
    int E = sc.nextInt();
    int S = sc.nextInt();
    int T = sc.nextInt();
    int R = sc.nextInt();

    Map<Integer, List<Integer>> graph = new HashMap<>();
    for (int i = 0; i < N; i++) {
      graph.put(i, new ArrayList<>());
    }

    for (int i = 0; i < M; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      graph.get(a).add(b);
      graph.get(b).add(a);
    }

    Event[] events = new Event[E];
    for (int i = 0; i < E; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      int c = sc.nextInt();
      events[i] = new Event(a, b, c);
    }

    int[][] dist = new int[N][N];
    for (int i = 0; i < N; i++) {
      Arrays.fill(dist[i], -1);
    }
    Queue<int[]> queue = new LinkedList<>();
    queue.offer(new int[] {S, 0});
    while (!queue.isEmpty()) {
      int[] curr = queue.poll();
      int u = curr[0];
      int d = curr[1];
      for (int v : graph.get(u)) {
        if (dist[u][v] == -1) {
          dist[u][v] = d + 1;
          queue.offer(new int[] {v, d + 1});
        }
      }
    }

    int[][] dp = new int[E + 1][N];
    for (int i = 0; i <= E; i++) {
      Arrays.fill(dp[i], -1);
    }
    dp[E][S] = 0;
    for (int i = E; i >= 0; i--) {
      for (int j = 0; j < N; j++) {
        if (dp[i][j] == -1) {
          continue;
        }
        if (i < E) {
          dp[i][events[i].c] = Math.min(dp[i][events[i].c], dp[i][j] + dist[j][events[i].c]);
        }
        for (int k = 0; k < N; k++) {
          if (dist[j][k] != -1 && dp[i][k] == -1) {
            dp[i][k] = dp[i][j] + dist[j][k];
          } else if (dist[j][k] != -1) {
            dp[i][k] = Math.min(dp[i][k], dp[i][j] + dist[j][k]);
          }
        }
      }
    }

    int ans = Integer.MAX_VALUE;
    for (int i = 0; i <= E; i++) {
      if (dp[i][T] != -1 && dp[i][T] <= R) {
        ans = Math.min(ans, dp[i][T] + i);
      }
    }

    System.out.println(ans == Integer.MAX_VALUE ? -1 : ans);
  }
}
