n = int(input())
sequence = [0]
total_sum = 0

for _ in range(n):
    op = list(map(int, input().split()))
    
    if op[0] == 1:
        a, x = op[1], op[2]
        total_sum += a * x
        sequence[0:a] = [sequence[i] + x for i in range(a)]
    elif op[0] == 2:
        k = op[1]
        sequence.append(k)
        total_sum += k
    else:
        total_sum -= sequence[-1]
        sequence.pop()
    
    print("{:.6f}".format(total_sum / len(sequence)))
