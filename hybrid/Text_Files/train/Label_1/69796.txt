from collections import defaultdict
import sys
input = sys.stdin.readline
n, s = map(int, input().split())
orders = defaultdict(list)
for _ in range(n):
    d, p, q = input().split()
    orders[d+' '+str(p)].append(int(q))
buy, sell = [], []
for p in sorted(orders.keys(), key=lambda x: x[:1]+str(-int(x[2:])), reverse=True)[:s]:
    buy.append(f"{'B' if p[:1]=='B' else 'S'} {p[2:]} {sum(orders[p])}")
for p in sorted(orders.keys(), key=lambda x: x[:1]+str(int(x[2:])), reverse=True)[:s]:
    sell.append(f"{'B' if p[:1]=='B' else 'S'} {p[2:]} {sum(orders[p])}")
print('\n'.join(buy+sell))
