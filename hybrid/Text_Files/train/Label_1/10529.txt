#include <bits/stdc++.h>
using namespace std;
char s[1001][6];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> s[i];
  }
  int p = -1;
  for (int i = 0; i < n; i++) {
    if (s[i][0] == 'O' && s[i][1] == 'O') p = i;
    if (s[i][3] == 'O' && s[i][4] == 'O') p = i;
  }
  if (p == -1) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  for (int i = 0; i < n; i++) {
    if (i == p) s[i][0] = s[i][1] = '+';
    if (i == p) s[i][3] = s[i][4] = '+';
    cout << s[i] << endl;
  }
}
