import sys

t = int(input())
for tc in range(t):
    n, k, l = map(int, input().split())
    d = list(map(int, input().split()))
    p = [0, 1, 2, 3, 4, 5, 4, 3, 2, 1] * (k // 5 + 1)
    a = []
    for i in range(n):
        for j in range(2 * k):
            a.append(d[i] + p[j])
    a.sort()
    ans = 'Yes' if a[n-1] <= l else 'No'
    print(ans)
