#include <bits/stdc++.h>
using namespace std;

struct st {
    int b, e, tp;
    bool operator<(const st& p) {
        return b < p.b;
    }
};

int main() {
    int n;
    cin >> n;
    vector<st> a(2 * n), p;
    for (int i = 0, j = 0; i < n; i++) {
        int a1, b1, a2, b2;
        cin >> a1 >> b1 >> a2 >> b2;
        a[j] = {a1, b1, 0};
        a[j + 1] = {a2, b2, 1};
        j += 2;
    }
    sort(a.begin(), a.end());
    int cur = 0;
    for (st x : a) {
        if (x.tp == 0) {
            p.push_back(x);
            cur++;
        } else if (cur > 0) {
            cur--;
            p.pop_back();
        }
    }
    cout << (cur == 0 ? "YES" : "NO");
    return 0;
}
