import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        char[][] board = new char[4][4];
        for (int i = 0; i < 4; i++) {
            board[i] = sc.next().toCharArray();
        }
        int[] a = new int[10];
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                if (board[i][j] != '.') {
                    a[board[i][j] - '0']++;
                }
            }
        }
        boolean ok = true;
        for (int i = 1; i <= 9; i++) {
            if ((a[i] % 2 == 1 && k % 2 == 0) || (a[i] % 2 == 0 && k % 2 == 1) || a[i] > 2 * k) {
                ok = false;
                break;
            }
        }
        System.out.println(ok ? "YES" : "NO");
    }
}
