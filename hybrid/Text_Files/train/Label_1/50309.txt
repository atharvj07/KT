import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long k = in.nextLong(), d = in.nextLong(), t = in.nextLong();
        long n = t / k + (t % k + d - 1) / d;
        System.out.println(n * t / 2.0);
    }
}
