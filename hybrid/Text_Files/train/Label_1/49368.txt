import java.util.*;

class Main {
    private static final int MAX_N = 1000;
    private static final int UNVISITED = -1;
    private static final int STATE1 = 1;
    private static final int STATE2 = 2;
    private static final int STATE3 = 3;
    private static final int ROAD = 0;
    private static final int WALL = -2;

    private static int[][] map;
    private static int n, m;
    private static int[] dx = {0, 0, -1, 1};
    private static int[] dy = {1, -1, 0, 0};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        map = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = sc.next();
            for (int j = 0; j < m; j++) {
                char c = line.charAt(j);
                if (c == '.') {
                    map[i][j] = ROAD;
                } else if (c == '#') {
                    map[i][j] = WALL;
                } else {
                    map[i][j] = c - '0';
                }
            }
        }

        int minCost = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] == ROAD) {
                    int cost = bfs(i, j);
                    if (cost != -1) {
                        minCost = Math.min(minCost, cost);
                    }
                }
            }
        }

        if (minCost == Integer.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(minCost);
        }
    }

    private static int bfs(int x, int y) {
        Queue<Integer> q = new LinkedList<>();
        q.add(x * m + y);
        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = UNVISITED;
            }
        }
        dist[x][y] = 0;

        int cost = 0;
        while (!q.isEmpty()) {
            int cur = q.poll();
            int curX = cur / m;
            int curY = cur % m;
            if (map[curX][curY] != ROAD) {
                cost++;
            }
            for (int i = 0; i < 4; i++) {
                int nextX = curX + dx[i];
                int nextY = curY + dy[i];
                if (nextX >= 0 && nextX < n && nextY >= 0 && nextY < m && dist[nextX][nextY] == UNVISITED && map[nextX][nextY] != WALL) {
                    q.add(nextX * m + nextY);
                    dist[nextX][nextY] = dist[curX][curY] + 1;
                }
            }
        }

        boolean connected = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] != WALL && dist[i][j] == UNVISITED) {
                    connected = false;
                }
            }
        }

        return connected ? cost : -1;
    }
}
