#include <cstdio>

int H, M, S;
int T1, T2;

bool e(int t) { return t >= 0 && t <= H; }

int main() {
    scanf("%d %d %d %d %d", &H, &M, &S, &T1, &T2);
    int T = H * 60 + M + S / 60;
    T1--;
    T2--;
    if (T1 == T2) {
        puts("YES");
        return 0;
    }
    printf("%s\n", !e(T1) || !e(T1 - 1) || !e(T1 + 1) || !e(T2) || !e(T2 - 1) || !e(T2 + 1) || T2 - T1 != 1 && T2 - T1 != H - 1 ? "NO" : "YES");
}
