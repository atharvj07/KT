import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        int[] y = new int[n];
        int sum = 0, max = 0;
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            y[i] = x[i];
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == 0 || j == 0) {
                    sum = x[i] + y[j];
                } else {
                    sum = x[i] + y[j] + max;
                }
                max = Math.max(sum, max);
            }
            x[i] = max;
            max = 0;
        }
        System.out.println(max);
    }
}
