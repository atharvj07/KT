#include <bits/stdc++.h>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> t(n);
  for (int i = 0; i < n; i++) { std::cin >> t[i]; }
  int ans = 0, cnt = 0;
  for (int i = 0; i < n; i++) {
    if (t[i] < 0) {
      if (cnt == k) {
        ans++;
        cnt = 0;
      }
      cnt++;
    } else {
      cnt = 0;
    }
  }
  if (cnt > 0) ans++;
  if (k == 0) ans = -1;
  std::cout << ans << std::endl;
  return 0;
}
