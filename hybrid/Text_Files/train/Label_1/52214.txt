import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        int[] p = new int[n + 1];
        for (int i = 0; i < n; i++)
            p[a[i]] = i + 1;
        int q = 0;
        for (int i = 1; i <= n; i++)
            if (p[i] != i) {
                System.out.println(++q);
                System.out.println(i + " " + p[i]);
                int t = a[i - 1];
                a[i - 1] = a[p[i] - 1];
                a[p[i] - 1] = t;
                p[a[i - 1]] = i;
                p[a[p[i] - 1]] = p[i];
                p[i] = i;
            }
        System.out.println(q);
    }
}
