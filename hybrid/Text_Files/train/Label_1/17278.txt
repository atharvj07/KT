#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
    int t;
    cin >> t;
    while (t--) {
        ll x0, n;
        cin >> x0 >> n;
        if (n % 2 == 0)
            cout << x0 - n / 2 << endl;
        else
            cout << x0 + (n + 1) / 2 << endl;
    }
    return 0;
}
