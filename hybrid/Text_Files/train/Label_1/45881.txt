import java.util.Scanner;

public class BubbleCupFinals {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[][] bubbleField = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                bubbleField[i][j] = scanner.nextInt();
            }
        }

        int maxBubbles = 0;
        for (int i = 0; i < n; i++) {
            maxBubbles = Math.max(maxBubbles, calculateRowSum(bubbleField, i));
        }
        for (int j = 0; j < m; j++) {
            maxBubbles = Math.max(maxBubbles, calculateColumnSum(bubbleField, j));
        }

        System.out.println(maxBubbles);
    }

    private static int calculateRowSum(int[][] bubbleField, int row) {
        int sum = 0;
        for (int j = 0; j < bubbleField[row].length; j++) {
            sum += bubbleField[row][j];
        }
        return sum;
    }

    private static int calculateColumnSum(int[][] bubbleField, int column) {
        int sum = 0;
        for (int i = 0; i < bubbleField.length; i++) {
            sum += bubbleField[i][column];
        }
        return sum;
    }
}
