n, b, a = map(int, input().split())
s = list(map(int, input().split()))

max_a = a
segments_passed = 0

for i in range(n):
    if s[i] == 1:
        if a < max_a and b > 0:
            b -= 1
            a += 1
        elif a > 0:
            a -= 1
        else:
            break
    else:
        if a > 0:
            a -= 1
        elif b > 0:
            b -= 1
        else:
            break
    segments_passed += 1

print(segments_passed)
