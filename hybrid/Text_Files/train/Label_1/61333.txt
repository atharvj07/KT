import sys
input = sys.stdin.readline
n, m, a = map(int, input().split())
b = list(map(int, input().split()))
p = list(map(int, input().split()))
p.sort()
b.sort(reverse = True)
ans = 0
sum = 0
i = 0
while i < n and i < m and a > 0:
    if a >= p[i] + b[i]:
        a -= p[i] + b[i]
        sum += b[i]
        ans += 1
    elif a >= p[i]:
        a -= p[i]
        ans += 1
    i += 1
print(ans, sum)
