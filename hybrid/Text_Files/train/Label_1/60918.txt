import sys

n, k = map(int, input().split())
c = []
for i in range(n):
    l, r = map(int, input().split())
    c.append((l, r))
c.sort(key=lambda x: x[1])
ans = 0
arr = []
for i in range(k):
    l, r = c[i]
    if i == 0 or l > arr[-1]:
        ans += r - l + 1
        arr.append(r)
print(ans)
print(" ".join(map(str, range(1, k + 1))))
