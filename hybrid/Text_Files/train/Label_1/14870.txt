#include <iostream>
#include <queue>
#include <vector>
using namespace std;
typedef long long ll;
const ll INF = 1e9;
const int MAXN = 1e3 + 5, MAXM = 2e3 + 5;
int n, m, p[MAXN], ind[MAXN], deg[MAXN], ans;
ll val[MAXN];
vector<vector<int>> adj;

int main() {
    cin >> n >> m;
    adj.resize(n + 1);
    for (int i = 1; i <= n; ++i) cin >> val[i];
    for (int x, y, i = 0; i < m; ++i) {
        cin >> x >> y;
        adj[x].push_back(y);
        adj[y].push_back(x);
        ind[x]++;
        ind[y]++;
    }
    priority_queue<int, vector<int>, greater<int>> pq;
    for (int i = 1; i <= n; ++i) {
        if (ind[i] == 1) {
            pq.emplace(i);
            deg[i] = 1;
        }
    }
    while (!pq.empty()) {
        int u = pq.top();
        pq.pop();
        ans += val[u];
        for (auto& v : adj[u]) {
            ind[v]--;
            deg[v]++;
            if (ind[v] == 1) pq.emplace(v);
        }
    }
    cout << ans << '\n';
    return 0;
}
