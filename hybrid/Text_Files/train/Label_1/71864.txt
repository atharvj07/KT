import sys
input = sys.stdin.readline

n, k = map(int, input().split())
c = list(map(int, input().split()))
b = 0
s = 0
ans = 0
for i in range(n):
    b += max(c[i], 0)
    ans += max(c[i] + b, 0)
    if b < 0 and k > 0:
        s += b
        k -= 1
        b = 0
for i in range(n - 1, -1, -1):
    b += max(c[i], 0)
    s += max(c[i] + b, 0)
    if b < 0 and k > 0:
        s += b
        k -= 1
        b = 0
print(ans + s)
