#include<bits/stdc++.h>
using namespace std;
#define INF (int)1e9
int n,a[1005];
int main(){
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    vector<int>dp(n+1,INF);
    dp[1]=0;
    for(int i=2;i<=n;i++){
        int t=a[i];
        dp[i]=dp[i-1]+1;
        for(int j=i-1;j>=1;j--){
            t-=a[j];
            if(t<0)break;
            dp[i]=min(dp[i],dp[j]+1);
        }
    }
    cout<<dp[n]<<endl;
}
