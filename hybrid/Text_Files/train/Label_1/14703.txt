#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int N = 505;
vector<int> g[N];

struct node {
    int deg, id;
};

bool operator<(node a, node b) { return a.deg < b.deg; }

int main() {
    int n;
    cin >> n;
    vector<node> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i].deg, v[i].id = i + 1; }
    sort(v.begin(), v.end());
    if (v[0].deg == 1) return cout << "NO", 0;
    int diam = 0, sz = 0;
    for (int i = 0; i < n - 1; i++) {
        if (v[i].deg == v[i + 1].deg) sz++;
        diam++;
    }
    if (sz > 1) return cout << "NO", 0;
    cout << "YES " << diam << '\n' << n - 1 << '\n';
    for (int i = 1; i < n; i++) { g[v[i].id].push_back(v[i - 1].id), g[v[i - 1].id].push_back(v[i].id); }
    for (int i = 1; i <= n; i++) {
        for (int j : g[i]) { if (i < j) cout << i << " " << j << '\n'; }
    }
    return 0;
}
