#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int maxn = 2e5, mod = 998244353;
vector<vector<int>> adj(maxn + 5);
vector<int> ord, sub_sz(maxn + 5);
int n, q, u, k;
void dfs(int s, int p) {
    ord.push_back(s);
    sub_sz[s] = 1;
    for (int v : adj[s]) {
        if (v != p) {
            dfs(v, s);
            sub_sz[s] += sub_sz[v];
        }
    }
}
int main() {
    cin >> n >> q;
    int p;
    for (int i = 2; i <= n; i++) {
        cin >> p;
        adj[i].push_back(p);
        adj[p].push_back(i);
    }
    dfs(1, 1);
    for (int i = 0; i < q; i++) {
        cin >> u >> k;
        if (sub_sz[u] < k)
            cout << -1 << '\n';
        else
            cout << ord[k - 1] << '\n';
    }
}
