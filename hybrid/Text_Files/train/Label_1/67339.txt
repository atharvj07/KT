n, m = map(int, input().split())
s = input()
t = input()
if '*' in s:
    i = s.index('*')
    if i == 0 or i == n - 1:
        print("YES" if t == s.replace('*', t) else "NO")
    else:
        print("YES" if t[1:m - 1] == s[i - 1:i + 1] and t[0] == s[0] and t[m - 1] == s[n - 1] else "NO")
else:
    print("NO" if s != t else "YES")
