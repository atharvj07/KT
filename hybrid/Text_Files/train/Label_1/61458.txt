import math
def get_beauty(s):
  ans = 0
  last = s[0]
  cur = 1
  for i in range(1, len(s)):
    if s[i] == last:
      cur += 1
    else:
      ans = max(ans, cur)
      cur = 1
      last = s[i]
  ans = max(ans, cur)
  return ans
n = int(input())
s = input()
for i in range(1, n):
  s = s + input()
print(get_beauty(s))
