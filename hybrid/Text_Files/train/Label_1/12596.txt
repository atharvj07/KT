#include<bits/stdc++.h>
using namespace std;
#define MOD 1000000007
#define ll long long
ll power(ll x, ll y){
    ll ans = 1;
    while(y>0){
        if(y%2){
            ans = ans * x;
            ans = ans % MOD;
        }
        x = x * x;
        x = x % MOD;
        y /= 2;
    }
    return ans;
}
int main(){
    ll n, a, b, c, d;
    cin >> n >> a >> b >> c >> d;
    ll ans = ((power(d, 2*n-6)%MOD) * (a * b))%MOD;
    ans = (ans * c) % MOD;
    cout << ans << "\n";
    return 0;
}
