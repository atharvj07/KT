import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String S = scanner.next();
        int price = 700;
        for (char c : S.toCharArray()) {
            if (c == 'o') {
                price += 100;
            }
        }
        System.out.println(price);
    }
}
