#include <bits/stdc++.h>
using namespace std;

bool solve() {
  long long n;
  cin >> n;
  vector<long long> a(n);
  for (auto &aa : a) { cin >> aa; }
  for (long long i = 1; i < n - 1; i++) {
    if (a[i] > a[i + 1] && a[i] > a[i - 1]) {
      cout << i + 1 << endl;
      return true;
    }
  }
  if (a[0] > a[1]) {
    cout << 1 << endl;
    return true;
  }
  if (a[n - 1] > a[n - 2]) {
    cout << n << endl;
    return true;
  }
  cout << -1 << endl;
  return false;
}

int main() {
  long long t;
  cin >> t;
  while (t--) { solve(); }
  return 0;
}
