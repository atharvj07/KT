#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    long long n;
    std::cin >> n;
    std::vector<long long> a(n), b(n);
    for (auto& x : a) std::cin >> x;
    for (auto& x : b) std::cin >> x;
    long long ans = 0;
    bool flag = false;
    for (int i = 0; i < n; i++) {
        if (a[i] + b[i] & 1) {
            flag = true;
            break;
        }
        ans += std::min(a[i] / 2, b[i] / 2) * std::min(a[i] / 2, b[i] / 2);
    }
    std::cout << (flag ? -1 : ans) << '\n';
    return 0;
}
