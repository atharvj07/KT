import sys
rl=sys.stdin.readline

n, x = map(int, rl().split())
g = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, rl().split())
    g[a].append(b)
    g[b].append(a)

d = [-1] * (n+1)
d[x] = 0
q = [x]
while q:
    v = q.pop(0)
    for u in g[v]:
        if d[u] == -1:
            d[u] = d[v] + 1
            q.append(u)

furthest = max(d)
idx = d.index(furthest)
d = [-1] * (n+1)
d[idx] = 0
q = [idx]
while q:
    v = q.pop(0)
    for u in g[v]:
        if d[u] == -1:
            d[u] = d[v] + 1
            q.append(u)

furthest = max(d)

print(furthest)
