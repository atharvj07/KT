#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, c0, c1, h;
    cin >> n >> c0 >> c1 >> h;
    string s;
    cin >> s;
    int cost = 0;
    for (int i = 0; i < n; i++) {
      int buy = min(s[i] == '0' ? c0 : c1, h + min(c0, c1));
      cost += buy;
    }
    cout << cost << endl;
  }
  return 0;
}
