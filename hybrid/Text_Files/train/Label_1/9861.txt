#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, j, mx1 = 0, mx2 = 0, ans;
    map<int, int> a, b;
    cin >> n;
    for(int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        a[x]++;
    }
    cin >> m;
    for(int i = 1; i <= m; i++) {
        int x, y;
        cin >> x >> y;
        b[x]++;
        b[y]++;
    }
    for(auto i: b) {
        if(i.second > mx1) {
            mx1 = i.second;
            ans = i.first;
        }
    }
    for(auto i: b) {
        if(i.second == mx1 and i.first != ans) {
            mx2 = max(mx2, i.second);
        }
    }
    if(mx2) {
        for(auto i: b) {
            if(i.second == mx2)
                cout << i.first << endl;
        }
    } else
        cout << ans << endl;
}
