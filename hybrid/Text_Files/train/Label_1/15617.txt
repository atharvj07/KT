#include <iostream>
#include <string>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    while (n--) {
        getline(cin, s);
        if (s[0] == 'm' and s[s.length() - 1] == 'a') {
            cout << "OMG>.< I don't know!\n";
            continue;
        }
        bool isFreda = false;
        bool isRainbow = false;
        if (s[s.length() - 5] == 'l' and s[s.length() - 4] == 'a' and
            s[s.length() - 3] == 'l' and s[s.length() - 2] == 'a') {
            isFreda = true;
        }
        if (s[0] == 'm' and s[1] == 'i' and s[2] == 'a' and s[3] == 'o') {
            isRainbow = true;
        }
        if (isFreda != isRainbow) {
            if (isFreda) {
                cout << "Freda's\n";
            } else {
                cout << "Rainbow's\n";
            }
        } else {
            cout << "OMG>.< I don't know!\n";
        }
    }
}
