#include <algorithm>
#include <cstdio>
#include <vector>

using namespace std;

struct Leg {
  int l, d;
};

bool compare(const Leg& a, const Leg& b) {
  return a.l < b.l;
}

int main() {
  int n;
  scanf("%d", &n);
  vector<Leg> l(n);
  for (int i = 0; i < n; i++) {
    scanf("%d", &l[i].l);
  }
  for (int i = 0; i < n; i++) {
    scanf("%d", &l[i].d);
  }
  sort(l.begin(), l.end(), compare);
  int ans = 0;
  int max_len = l[n - 1].l;
  int cnt_max_len = 1;
  for (int i = n - 2; i >= 0 && n - i > n / 2; i--) {
    if (l[i].l == max_len) {
      cnt_max_len++;
    } else {
      ans += min(l[i].d, l[i + 1].d) * cnt_max_len;
      max_len = l[i].l;
      cnt_max_len = 1;
    }
  }
  printf("%d\n", ans);
  return 0;
}
