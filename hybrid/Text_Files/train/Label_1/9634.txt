#include <iostream>
#include <string>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    int dp[n + 1][3];
    dp[n][0] = dp[n][1] = dp[n][2] = 0;
    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j < 3; j++) {
            int d = ((s[i] - '0') + j) % 3;
            dp[i][j] = max(dp[i + 1][j], dp[i + 1][d] + 1);
        }
    }
    cout << dp[0][0] << endl;
    return 0;
}
