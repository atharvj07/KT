import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            String s = sc.next();
            String t = sc.next();
            boolean isPossible = true;
            for (int j = 0; j < n; j++) {
                if (s.charAt(j) != t.charAt(j)) {
                    isPossible = false;
                    break;
                }
            }
            System.out.println(isPossible ? "YES" : "NO");
        }
    }
}
