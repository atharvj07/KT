import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int q = input.nextInt();
        for (int i = 0; i < q; i++) {
            int n = input.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = input.nextInt();
            }
            Arrays.sort(a);
            long sum = 0;
            for (int j = 0; j < n; j++) {
                sum += a[j];
            }
            long minPrice = (sum + n - 1) / n;
            System.out.println(minPrice);
        }
    }
}
