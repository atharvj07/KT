#include <bits/stdc++.h>
#define N 100000
int n, a[N], ans;
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  for (int i = 0; i < n; i++) ans += (i == a[i]);
  int mx = 0;
  for (int i = 0; i < n; i++)
    if (i != a[i]) {
      mx = std::max(mx, 1 + (a[a[i]] == i));
    }
  printf("%d\n", ans + mx);
  return 0;
}
