n = int(input())
a = list(map(int, input().split()))

for i in range(n):
    a[i] -= i

entrance = 0
min_time = float('inf')
for i in range(n):
    if a[i] >= 0:
        time = (a[i] // n) + (1 if a[i] % n != 0 else 0)
        if time < min_time:
            min_time = time
            entrance = i + 1

print(entrance)
