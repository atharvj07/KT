import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        int sum1 = 0, sum2 = 0;
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            sum1 += a[i] * (i + 1);
            sum2 += a[i] * ((n - 1) - i);
        }
        System.out.println(Math.max(sum1, sum2));
    }
}
