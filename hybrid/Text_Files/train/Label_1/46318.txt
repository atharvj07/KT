import java.io.*;
import java.util.*;
class Main {
    static ArrayList<ArrayList<Integer>> g;
    static int n, w[], max;
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        StringTokenizer st = new StringTokenizer(br.readLine());
        g = new ArrayList<>();
        w = new int[n + 1];
        for (int i = 0; i <= n; i++) g.add(new ArrayList<>());
        for (int i = 1; i <= n; i++) w[i] = Integer.parseInt(st.nextToken());
        for (int i = 1; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken()), y = Integer.parseInt(st.nextToken()), c = Integer.parseInt(st.nextToken());
            g.get(x).add(y);
            g.get(y).add(x);
        }
        dfs(1, 1, w[1]);
        System.out.println(max);
    }
    public static void dfs(int u, int p, int f) {
        max = Math.max(max, f);
        for (int v : g.get(u)) {
            if (v == p) continue;
            dfs(v, u, Math.min(f, w[v] + f));
        }
    }
}
