n, m = map(int, input().split())
knows = [[] for x in range(n)]
for idx in range(n):
  knows[idx] = set(map(int, input().split()[1:]))
tolearn = [set(range(1, m + 1)) - knows[x] for x in range(n)]
cnt = 0
vis = [False] * (m + 1)
q = [1]
vis[1] = True
while len(q) > 0:
  cur = q.pop(0)
  for x in tolearn:
    if cur in x:
      x.clear()
      if not vis[cur]:
        vis[cur] = True
        q.append(cur)
  cnt += 1
print(cnt - 1)
