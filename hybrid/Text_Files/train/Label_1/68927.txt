import sys

n = int(input())
v = [list(map(int,input().split())) for _ in range(n)]
v.sort(key = lambda x: max(x[0],x[1]), reverse = True)

ans, cur, res = 0, 0, []
for t, d, p in v:
    if t <= d and cur+t <= d:
        ans += p
        res.append(v.index(v)+1)
        cur += t

print(ans)
print(len(res))
print(*res)
