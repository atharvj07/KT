#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<pair<int, int>> t(n);
    for (int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        t[i] = {b, a};
    }
    sort(t.begin(), t.end());
    int cur = 0, ans = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '1' && cur < t[i].first) ans++;
        cur = t[i].second ? cur + t[i].first : t[i].first;
    }
    cout << ans << endl;
    return 0;
}
