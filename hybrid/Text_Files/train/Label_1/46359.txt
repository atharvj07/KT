import java.util.Scanner;

public class Dolphin {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int ma = scanner.nextInt();
        int mb = scanner.nextInt();

        int[][] chemicals = new int[n][3];
        for (int i = 0; i < n; i++) {
            chemicals[i][0] = scanner.nextInt(); // a_i
            chemicals[i][1] = scanner.nextInt(); // b_i
            chemicals[i][2] = scanner.nextInt(); // c_i
        }

        int minCost = solve(chemicals, ma, mb);
        System.out.println(minCost == Integer.MAX_VALUE ? -1 : minCost);
    }

    private static int solve(int[][] chemicals, int ma, int mb) {
        int gcd = gcd(ma, mb);
        ma /= gcd;
        mb /= gcd;

        int[][] dp = new int[ma + 1][mb + 1];
        for (int i = 0; i <= ma; i++) {
            for (int j = 0; j <= mb; j++) {
                dp[i][j] = Integer.MAX_VALUE;
            }
        }

        for (int[] chemical : chemicals) {
            int a = chemical[0];
            int b = chemical[1];
            int c = chemical[2];

            for (int i = ma; i >= a; i--) {
                for (int j = mb; j >= b; j--) {
                    dp[i][j] = Math.min(dp[i][j], dp[i - a][j - b] + c);
                }
            }
        }

        return dp[ma][mb];
    }

    private static int gcd(int a, int b) {
        return b == 0 ? a : gcd(b, a % b);
    }
}
