#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
typedef long long ll;
ll fact[505], ifact[505];
ll fastpow(ll a, ll b){
    ll ans = 1;
    while (b) {
        if (b & 1)
            ans = (ans * a) % mod;
        a = (a * a) % mod;
        b >>= 1;
    }
    return ans % mod;
}
ll ncr(ll n, ll r){
    ll ret = (fact[n] * ifact[n - r]) % mod;
    return (ret * ifact[r]) % mod;
}
ll npr(ll n, ll r){
    ll ret = (fact[n] * ifact[n - r]) % mod;
    return ret;
}
int main(){
    fact[0] = 1;
    ifact[0] = 1;
    for(ll i = 1; i <= 500; i++){
        fact[i] = (i * fact[i - 1]) % mod;
        ifact[i] = fastpow(fact[i], mod-2);
    }
    int n;
    cin >> n;
    vector<vector<int>> adj(n + 1);
    for(int i = 0; i < n; i++){
        for(int j = 0; j < n; j++){
            int x;
            cin >> x;
            if(x){
                adj[i + 1].push_back(j + 1);
                adj[j + 1].push_back(i + 1);
            }
        }
    }
    vector<vector<bool>> vis(n + 1, vector<bool>(n + 1));
    ll ans = 0;
    function<void(int)> dfs = [&](int nod){
        for(auto x: adj[nod]){
            if(!vis[nod][x]){
                vis[nod][x] = vis[x][nod] = true;
                dfs(x);
                ans = (ans + ncr(n - 2, n - 3)) % mod;
            }
        }
    };
    dfs(1);
    cout << ans << endl;
}
