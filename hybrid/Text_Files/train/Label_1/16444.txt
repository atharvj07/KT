#include<bits/stdc++.h>
using namespace std;
#define ll long long int
#define mod 998244353
ll n,arr[200005],cnt[200005],fact[200005],invfact[200005],pows[200005];
ll modpow(ll a,ll b){
  if(b==0LL)  return 1LL;
  if(b==1LL)  return a%mod;
  ll tmp=modpow(a,b/2)%mod;
  tmp=(tmp*tmp)%mod;
  if(b%2)    return (tmp*a)%mod;
  else return tmp%mod;
}
int main()
{
  cin>>n;
  for(ll i=1;i<=n;i++){
    cin>>arr[i];
    if(arr[i]==-1LL)  cnt[i]=1LL;
  }
  fact[0]=fact[1]=invfact[0]=invfact[1]=1LL;
  for(ll i=2;i<=n;i++){
    fact[i]=(fact[i-1]%mod*i%mod)%mod;
    invfact[i]=modpow(fact[i],mod-2);
  }
  pows[0]=1LL;
  for(ll i=1;i<=n;i++){
    pows[i]=(pows[i-1]%mod*2LL%mod)%mod;
  }
  ll ans=0LL;
  ll sum=0LL;
  for(ll i=n;i>=1;i--){
    if(cnt[i]==0LL)  continue;
    sum+=1LL;
    ll mul=1LL;
    for(ll j=n;j>=1;j--){
      if(cnt[j]==0LL)  continue;
      if(j<i)  mul=(mul%mod*pows[j]%mod)%mod;
      else mul=(mul%mod*pows[j-1]%mod)%mod;
    }
    ans=(ans%mod+(mul%mod*sum%mod)%mod*invfact[sum]%mod)%mod;
  }
  ans=(ans%mod*invfact[n]%mod)%mod;
  cout<<(ans+mod)%mod<<endl;
  return 0;
}
