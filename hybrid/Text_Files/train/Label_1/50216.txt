import java.util.*;
import java.io.*;
import java.lang.*;

// Standard Input Format
class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int t = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[t];
        for (int i = 0; i < t; i++) {
            a[i] = scanner.nextInt();
        }
        if (!check(n, t, k, a)) {
            System.out.println(-1);
            return;
        }
        List<int[]> roads = buildRoads(n, t, k, a);
        System.out.println(n);
        for (int[] road : roads) {
            System.out.println(road[0] + " " + road[1]);
        }
    }

    private static boolean check(int n, int t, int k, int[] a) {
        int sum = 0;
        for (int i = 0; i < t; i++) {
            sum += a[i];
        }
        return sum == n - 1 && a[t - 1] >= k;
    }

    private static List<int[]> buildRoads(int n, int t, int k, int[] a) {
        List<int[]> roads = new ArrayList<>();
        int[] deg = new int[n + 1];
        for (int i = 0; i < t; i++) {
            for (int j = 0; j < a[i]; j++) {
                int u = i + 2;  // city id
                int v = i + 1;  // parent city id
                roads.add(new int[] {u, v});
                deg[u]++;
                deg[v]++;
            }
        }
        int[] deadEnds = new int[k];
        int idx = 0;
        for (int i = 2; i <= n; i++) {
            if (deg[i] == 1) {
                deadEnds[idx++] = i;
            }
        }
        for (int i = 0; i < k; i++) {
            int u = deadEnds[i];
            int v = i + 2;
            roads.add(new int[] {u, v});
            deg[u]++;
            deg[v]++;
        }
        return roads;
    }
}
