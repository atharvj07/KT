n=int(input())
a=list(map(int,input().split()))
b=[[False]*n for i in range(n)]
for p in range(n):
	for q in range(n):
		if(p==q):
			continue
		for i in range(n):
			if(a[i]!=a[p]):
				continue
			for j in range(n):
				if(a[j]!=a[q]):
					continue
				if(i==j):
					continue
				b[p][q]=b[p][q] or (i&1)!=(j&1)
for x in b:
	if(not any(x)):
		print(-1)
		exit()
ans=[0]*n
for p in range(n):
	for q in range(n):
		if(b[p][q]):
			ans[p]=a[q]
print(*ans)
