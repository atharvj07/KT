#include <iostream>
#include <vector>
using namespace std;
const int INF = 2e9;
int main() {
  int n;
  cin >> n;
  vector<int> v(n+1), dp(n+1);
  for (int i = 1; i <= n; i++) cin >> v[i];
  for (int i = n; i >= 1; i--) {
    dp[i] = INF;
    if (i == n) dp[i] = 0;
    if (i != n) {
      if (v[i + 1] < v[i]) dp[i] = min(dp[i], dp[i + 1]);
      if (v[n] < v[i]) dp[i] = min(dp[i], dp[n]);
    }
    if (i + 1 != n) {
      if (v[i + 1] > v[i]) dp[i] = min(dp[i], 1 + dp[i + 1]);
      if (v[n] > v[i]) dp[i] = min(dp[i], 1 + dp[n]);
    }
  }
  cout << dp[1];
  return 0;
}
