k, p = input().split()
k = int(k)
p = int(p)
ans = 0
for i in range(1,10):
    for j in range(i+1,10):
        ans += ((i*10+j)%p*pow(10,(k-2)//2))%p
        if (k-2)%2 == 1:
            for l in range(j+1,10):
                ans += ((i*10+j*100+l)%p*pow(10,(k-3)//2))%p
    ans += ((i*100+i)%p*pow(10,(k-2)//2))%p
    if (k-2)%2 == 1:
        for l in range(i+1,10):
            ans += ((i*100+l*10+i)%p*pow(10,(k-3)//2))%p
ans = ans%p
print(ans)
