#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, x, pos;
        bool ok = true;
        vector<int> a;
        cin >> n >> m;
        for (int i = 0; i < n; i++) {
            cin >> x;
            a.push_back(x);
        }
        sort(a.begin(), a.end());
        vector<int> p;
        for (int i = 0; i < m; i++) {
            cin >> pos;
            p.push_back(pos);
        }
        sort(p.begin(), p.end());
        for (int i = 0, j = 0; i < n; i++) {
            if (j < m and p[j] == i) {
                swap(a[i], a[i + 1]);
                j++;
            }
            if (a[i] != i + 1) ok = false;
        }
        if (ok)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
