#include <cstdio>
#include <cstring>
using namespace std;

int a[55][55];
bool f, ok;

int main() {
  int n, i, j;
  while (scanf("%d", &n) > 0) {
    f = 1;
    for (i = 1; i <= n; i++) {
      char c[55];
      scanf("%s", c + 1);
      for (j = 1; j <= n; j++)
        if ((a[i][j] = c[j] - '0') > 0) {
          f = 0;
          if (a[i][j] == 1 || a[i][j] > 3) ok = 0;
          ok = 1;
        }
    }
    for (i = 1; i <= n && ok; i++)
      for (j = 1; j <= n; j++) {
        int sum = 0;
        if (a[i][j]) sum++;
        if (a[i - 1][j]) sum++;
        if (a[i][j - 1]) sum++;
        if (a[i - 1][j - 1]) sum++;
        if (sum != a[i][j]) ok = 0;
      }
    printf("%s\n", ok ? "Yes" : "No");
  }
  return 0;
}
