#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int& ai : a) { cin >> ai; }
  long long s = 0;
  for (int i = 0; i < k; i++) { s += a[i]; }
  long long mini = s;
  int ans = 0;
  for (int i = k; i < n; i++) {
    s += a[i] - a[i - k];
    if (s < mini) {
      mini = s;
      ans = i - k + 1;
    }
  }
  cout << ans + 1 << endl;
}
