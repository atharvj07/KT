import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        char[] left = scanner.next().toCharArray();
        char[] right = scanner.next().toCharArray();
        boolean[][] dp = new boolean[n + 1][2];
        dp[1][0] = true;
        for (int i = 1; i <= n; i++) {
            if (!dp[i][0]) {
                continue;
            }
            if (i + k <= n && right[i + k - 1] != 'X') {
                dp[i + k][1] = true;
            }
            if (i - 1 > 0 && left[i - 1] != 'X') {
                dp[i - 1][0] = true;
            }
            if (i + 1 <= n && right[i] != 'X') {
                dp[i + 1][0] = true;
            }
        }
        System.out.println(dp[n][1] ? "YES" : "NO");
    }
}
