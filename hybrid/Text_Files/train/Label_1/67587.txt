m, s = map(int, input().split())
if s > 9 * m or s == 0 and m > 1:
    print('-1 -1')
else:
    mn = [0] * m
    for i in range(m - 1, -1, -1):
        mn[i] = min(9, s)
        s -= mn[i]
    if s > 0:
        mn[0] = s
    mx = [0] * m
    for i in range(m - 1, -1, -1):
        mx[i] = min(9, s)
        s -= mx[i]
    if s > 0:
        mx[0] += s
    print(''.join(map(str, mn)), ''.join(map(str, mx)))
