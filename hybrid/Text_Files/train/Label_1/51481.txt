import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextInt(), m = sc.nextInt();
        int[][] a = new int[(int)n][(int)m];
        long odd = 0, even = 0;
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = Integer.parseInt(s.charAt(j) + "");
                if (i % 2 == j % 2) {
                    if (a[i][j] == 0) {
                        odd++;
                    }
                } else {
                    if (a[i][j] == 1) {
                        odd++;
                    }
                }
            }
        }
        even = n * m - odd;
        if (odd % 2 == 0 && even % 2 == 0) {
            System.out.println("0");
        } else if (odd % 2 == 1 && even % 2 == 1) {
            System.out.println("-1");
        } else {
            System.out.println("1");
        }
    }
}
