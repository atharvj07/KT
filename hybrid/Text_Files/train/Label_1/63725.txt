n, k = map(int, input().split())
passwords = [input() for _ in range(n)]
correct_password = passwords[-1]
passwords.sort(key=len)

time_best = time_worst = penalty_count = 0

for password in passwords:
    if len(password) < len(correct_password):
        time_best += 1
        time_worst += 1
        penalty_count += 1
        if penalty_count == k:
            time_best += 5
            time_worst += 5
            penalty_count = 0
    elif len(password) == len(correct_password):
        time_best += 1
        break

penalty_count = 0
for password in passwords:
    if password != correct_password:
        time_worst += 1
        penalty_count += 1
        if penalty_count == k:
            time_worst += 5
            penalty_count = 0
    else:
        break

print(time_best, time_worst)
