import java.util.Arrays;
import java.util.Scanner;

public class RyoukosMemoryNote {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }

        Arrays.sort(a);

        long totalPagesTurned = 0;
        int currentPage = a[0];
        for (int i = 1; i < m; i++) {
            totalPagesTurned += Math.abs(currentPage - a[i]);
            currentPage = a[i];
        }

        int bestMergePage = -1;
        long minPagesTurned = totalPagesTurned;
        for (int i = 1; i < n; i++) {
            long pagesTurnedIfMerged = 0;
            int leftPage = currentPage;
            for (int j = i; j < m; j++) {
                if (a[j] > i) {
                    pagesTurnedIfMerged += Math.abs(leftPage - a[j]);
                    leftPage = a[j];
                }
            }

            if (pagesTurnedIfMerged < minPagesTurned) {
                minPagesTurned = pagesTurnedIfMerged;
                bestMergePage = i;
            }
        }

        if (bestMergePage != -1) {
            totalPagesTurned -= minPagesTurned;
        }

        System.out.println(totalPagesTurned);
    }
}
