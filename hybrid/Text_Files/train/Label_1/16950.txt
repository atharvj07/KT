#include <iostream>
#include <vector>
#include <set>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, k;
    cin >> n >> m >> k;
    vector<long long> x(n + 1);
    for (int i = 0; i < n; i++) { cin >> x[i]; }
    x.back() = 1e6;
    vector<long long> y(m + 1);
    for (int i = 0; i < m; i++) { cin >> y[i]; }
    y.back() = 1e6;
    vector<pair<long long, long long>> p(k);
    for (int i = 0; i < k; i++) { cin >> p[i].first >> p[i].second; }

    long long ans = 0;
    for (int i = 0; i < k; i++) {
      long long dist_x = 1e18, dist_y = 1e18;
      for (int j = 1; j <= n; j++) {
        if (p[i].first == x[j]) {
          dist_x = 0;
          break;
        }
        dist_x = min(dist_x, min(abs(p[i].first - x[j]), abs(p[i].first - x[j - 1])));
      }
      for (int j = 1; j <= m; j++) {
        if (p[i].second == y[j]) {
          dist_y = 0;
          break;
        }
        dist_y = min(dist_y, min(abs(p[i].second - y[j]), abs(p[i].second - y[j - 1])));
      }

      ans += (p[i].first + p[i].second - dist_x - dist_y) > 0;
    }
    cout << ans / 2 << endl;
  }
  return 0;
}
