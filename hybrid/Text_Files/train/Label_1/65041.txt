n,h,m=map(int,input().split())
a=[[0]*4 for i in' '*m]
for i in range(m):a[i]=[int(j) for j in input().split()]
for i in range(m):a[i][3]=max(a[i][3],h)
a.sort(key=lambda x:(x[2],x[1]))
dp=[[[-1]*(h+1) for j in' '*6] for i in' '*n]
def fun(pos,lst,cur):
 if pos==n:return 0
 if dp[pos][lst][cur]!=-1:return dp[pos][lst][cur]
 ans=0
 for i in range(cur,h+1):
  flag=1
  for j in range(m):
   if a[j][0]<=pos<=a[j][1] and i>a[j][2]:flag=0
  if flag:ans=max(ans,i*i+fun(pos+1,i,i))
 dp[pos][lst][cur]=ans
 return ans
print(fun(0,0,0))
