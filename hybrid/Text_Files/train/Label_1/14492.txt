#include <iostream>
#include <vector>

using namespace std;

vector<vector<int>> g;
vector<int> col;

void dfs(int u, int c) {
  if(col[u] != -1) {
    return;
  }
  col[u] = c;
  for(int v : g[u]) {
    dfs(v, c^1);
  }
}

int main() {
  int t;
  cin >> t;
  while(t--) {
    int n;
    cin >> n;
    g = vector<vector<int>>(n);
    col = vector<int>(n, -1);
    for(int i = 0; i < n; i++) {
      int x;
      cin >> x;
      x--;
      g[i].push_back(x);
      g[x].push_back(i);
    }
    dfs(0, 0);
    bool ok = false;
    for(int i = 0; i < n; i++) {
      if(col[i] == 1) {
        for(int j : g[i]) {
          if(col[j] == 0) {
            ok = true;
            break;
          }
        }
      }
    }
    if(ok) {
      cout << "YES\n";
    } else {
      cout << "NO\n";
    }
  }
  return 0;
}
