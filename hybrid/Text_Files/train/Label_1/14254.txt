#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        string n;
        int s;
        cin >> n >> s;
        int ans = 0;
        int sum = 0;
        for (int i = n.size() - 1; i >= 0; i--) {
            int d = n[i] - '0';
            if (sum + d <= s) {
                break;
            } else {
                int need = s - sum;
                ans += (9 - d) * i;
                n[i] = (d + need) + '0';
                sum += need;
            }
        }
        cout << ans << endl;
    }
}
