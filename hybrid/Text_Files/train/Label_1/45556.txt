import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        int f = in.nextInt();
        int k = in.nextInt();
        int ans = 0;
        int c = b;
        if (k == 1)
            ans = 0;
        else if ((b - f) * 2 < a)
            ans = -1;
        else if ((b - f) * 2 >= a) {
            ans = k / 2;
            if (k % 2 == 1)
                ans++;
        }
        System.out.println(ans);
    }
}
