import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        long[][] a = new long[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        long[][] f = new long[n][m];
        long[][] g = new long[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                f[i][j] = a[i][j];
                if (i > 0) {
                    f[i][j] = Math.max(f[i - 1][j], f[i][j]);
                }
                if (j > 0) {
                    f[i][j] = Math.max(f[i][j - 1], f[i][j]);
                }
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            for (int j = m - 1; j >= 0; j--) {
                g[i][j] = a[i][j];
                if (i < n - 1) {
                    g[i][j] = Math.max(g[i + 1][j], g[i][j]);
                }
                if (j < m - 1) {
                    g[i][j] = Math.max(g[i][j + 1], g[i][j]);
                }
            }
        }
        long ans = 0;
        int x = 0;
        int y = 0;
        for (int i = k - 1; i < n - k + 1; i++) {
            for (int j = k - 1; j < m - k + 1; j++) {
                if (ans < Math.max(Math.max(f[i][j], g[i][j]), a[i][j])) {
                    ans = Math.max(Math.max(f[i][j], g[i][j]), a[i][j]);
                    x = i + 1;
                    y = j + 1;
                }
            }
        }
        System.out.println(x + " " + y);
    }
}
