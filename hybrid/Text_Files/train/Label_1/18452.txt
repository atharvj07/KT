#include<bits/stdc++.h>

using namespace std;

int n, freq[26], done[26];
string s;

int main() {
    cin >> n >> s;
    bool f = 1;
    for (int i = 0; i < n; i++) {
        freq[s[i] - 'a']++;
        if (freq[s[i] - 'a'] * 2 > n) {
            f = 0;
            break;
        }
    }
    if (f == 0) {
        cout << "NO";
        return 0;
    }
    cout << "YES\n";
    for (int i = 0; i < n; i++) {
        if (freq[s[i] - 'a'] * 2 <= n && done[s[i] - 'a'] == 0) {
            cout << s[i];
            done[s[i] - 'a'] = 1;
        }
    }
    return 0;
}
