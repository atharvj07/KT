import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner scanner = new Scanner(System.in);
		String s = scanner.next();
		scanner.close();
		int toppings = 0;
		if (s.charAt(0) == 'o') {
			toppings++;
		}
		if (s.charAt(1) == 'o') {
			toppings++;
		}
		if (s.charAt(2) == 'o') {
			toppings++;
		}
		System.out.println(700 + 100 * toppings);
	}
}
