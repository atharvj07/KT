#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, ans = 0;
    cin >> n >> m;
    vector<int> c(n), a(m);
    for (int i = 0; i < n; i++) cin >> c[i];
    for (int i = 0; i < m; i++) cin >> a[i];
    for (int i = 0; i < n; i++) {
        if (i < m && c[i] <= a[i]) ans++, a.erase(a.begin());
    }
    cout << ans << endl;
}
