#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    int a;
    getline(cin, s);
    cin >> a;
    int h = stoi(s.substr(0, 2));
    int m = stoi(s.substr(3, 2));
    m += a;
    h += m / 60;
    m %= 60;
    if (h >= 24)
        h -= 24;
    cout << setfill('0') << setw(2) << h << ':' << setfill('0') << setw(2) << m << '\n';
    return 0;
}
