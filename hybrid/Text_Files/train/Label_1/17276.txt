#include <cmath>
#include <iostream>
using namespace std;

int main() {
    long long t, x0, n, ans;
    cin >> t;
    while (t--) {
        cin >> x0 >> n;
        if (n == 0) ans = x0;
        else if (n & 1 and x0 % 2 == 0) ans = x0 - (n + 1) * n / 2;
        else if (n & 1 and x0 % 2) ans = x0 + (n + 1) * n / 2;
        else if (!(n & 1) and x0 % 2 == 0) ans = x0 + (n + 1) * n / 2;
        else ans = x0 - (n + 1) * n / 2;
        cout << ans << endl;
    }
    return 0;
}
