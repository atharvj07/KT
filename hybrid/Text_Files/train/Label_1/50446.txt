import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int num = scanner.nextInt();
        int[] maxjumps = new int[num];
        for (int i = 0; i < num; i++) {
            maxjumps[i] = scanner.nextInt();
        }
        scanner.close();
        int pos = 0, reach = 0;
        while (pos < num - 1) {
            reach = Math.max(reach, pos + maxjumps[pos]);
            if (reach >= num - 1) {
                System.out.println("yes");
                return;
            }
            if (reach <= pos) {
                System.out.println("no");
                return;
            }
            pos++;
        }
        pos = num - 1;
        reach = num - 1;
        while (pos > 0) {
            reach = Math.min(reach, pos - maxjumps[pos]);
            if (reach <= 0) {
                System.out.println("no");
                return;
            }
            pos--;
        }
        System.out.println("yes");
    }
}
