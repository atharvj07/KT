#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  sort(a.begin(), a.end());
  if (k >= n) {
    cout << *max_element(a.begin(), a.end()) << endl;
  } else if (k == 0) {
    cout << a[n - 1] + 1 << endl;
  } else {
    cout << a[k - 1] << endl;
  }
}
