#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    string s;
    cin >> s;
    int n = s.size(), l = 0, k = 0;
    vector<vector<int>> ans;
    for (int r = 0; r < n; r++) {
        if (s[r] == '(') l++;
        else {
            if (l > 0) {
                k++;
                l--;
                ans.push_back({2, r - l, r});
            }
        }
    }
    cout << k << '\n';
    for (auto &op : ans)
        cout << op[0] << '\n' << op[1] << ' ' << op[2] << '\n';
}
