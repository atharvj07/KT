def restore_ipv6(n, addresses):
    restored_addresses = []
    for address in addresses:
        parts = address.split(':')
        idx_double_colon = None  # Index of the "::" if exists
        
        # Check if there's a "::"
        if '' in parts:
            idx_double_colon = parts.index('')
            # Handle edge cases with "::" at the start or end
            if idx_double_colon + 1 < len(parts) and parts[idx_double_colon + 1] == '':
                parts.remove('')
            parts.remove('')
        
        # Add leading zeros to each part except for the "::"
        parts = [p.zfill(4) for p in parts]
        
        # If there was a "::", we need to insert zeros for it
        if idx_double_colon is not None:
            num_zeros_to_insert = 8 - len(parts)
            for _ in range(num_zeros_to_insert):
                parts.insert(idx_double_colon, '0000')
        
        restored_addresses.append(':'.join(parts))
    
    return restored_addresses

n = int(input().strip())
addresses = [input().strip() for _ in range(n)]

restored_addresses = restore_ipv6(n, addresses)
for addr in restored_addresses:
    print(addr)
