import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] range = new int[n][2];
        for (int i = 0; i < n; i++) {
            range[i][0] = sc.nextInt();
            range[i][1] = sc.nextInt();
        }
        int[] bridges = new int[m];
        for (int i = 0; i < m; i++) {
            bridges[i] = sc.nextInt();
        }
        Arrays.sort(bridges);
        Arrays.sort(range, (a, b) -> a[0] - b[0]);
        int j = 0;
        boolean possible = true;
        int[] ans = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            while (j < m && bridges[j] < range[i + 1][0] - range[i][0]) {
                j++;
            }
            if (j >= m || bridges[j] > range[i + 1][1] - range[i][0]) {
                possible = false;
                break;
            }
            ans[i] = j + 1;
            j++;
        }
        if (possible) {
            System.out.println("Yes");
            System.out.println(Arrays.toString(ans).replaceAll("[\\[\\],]", ""));
        } else {
            System.out.println("No");
        }
        sc.close();
    }
}
