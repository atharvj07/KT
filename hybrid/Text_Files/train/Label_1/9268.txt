#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    if (n == 0 || k == 300000) {
        cout << "0\n";
        for (int i = 1; i <= n; i++) {
            cout << i << " ";
        }
        cout << endl;
        return 0;
    }
    vector<pair<int, int>> flights;
    for (int i = 0; i < n; i++) {
        int c;
        cin >> c;
        flights.push_back({c, i + 1});
    }
    sort(flights.begin(), flights.end());
    int total_cost = 0;
    vector<int> order;
    for (int i = 0; i < n; i++) {
        int c = flights[i].first;
        int j = flights[i].second;
        total_cost += c * (k + i + 1 - j);
        order.push_back(k + i + 1);
    }
    cout << total_cost << endl;
    for (int i = 0; i < n; i++) {
        cout << order[i] << " ";
    }
    cout << endl;
    return 0;
}
