t = int(input())
for _ in range(t):
    n = int(input())
    b = list(map(int, input().split()))

    left = b[0] - 1
    right = 2*n - b[-1]

    possible_xs = [0] * (n + 1)
    possible_xs[left] = 1
    possible_xs[right] = 1

    j = 0
    for i in range(1, n):
        if b[i] - b[i-1] == 1:
            j += 1
        else:
            possible_xs[j] = 1
            j = 0
    possible_xs[j] = 1

    print(sum(possible_xs))
