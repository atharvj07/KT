#include <bits/stdc++.h>
using namespace std;
#define ll long long

const ll mod = 1e9 + 7;
const int maxN = 5005;

int n;
char c[maxN];

ll f[maxN][maxN], dp[maxN];

ll power(ll x, ll y) {
    ll ans = 1;
    while (y > 0) {
        if (y % 2 == 1) { ans = ans * x % mod; }
        x = x * x % mod;
        y /= 2;
    }
    return ans;
}

ll inverse(ll x) { return power(x, mod - 2); }

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> c[i];
    }

    f[1][1] = dp[1] = 1;
    for (int i = 2; i <= n; i++) {
        if (c[i] == 's') {
            for (int j = 1; j <= n; j++) {
                for (int k = 1; k < j; k++) {
                    (f[i][j] += (f[i - 1][k] * dp[j - k]) % mod) %= mod;
                }
                (dp[j] = (dp[j] + f[i][j]) % mod) %= mod;
            }
        } else {
            for (int j = 2; j <= n; j++) {
                (f[i][j] = f[i - 1][j - 1]) %= mod;
            }
        }
    }

    ll ans = 0;
    for (int i = 1; i <= n; i++) { (ans += dp[i]) %= mod; }
    cout << ans << endl;

    return 0;
}
