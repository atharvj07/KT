from collections import deque

n = int(input())
w = list(map(int, input().split()))
t = input()
a = deque()
b = []
for i in t:
    if i == '0':
        a.append(w.index(min(w)) + 1)
        w[w.index(min(w))] = 10 ** 9 + 1
    else:
        b.append(w.index(max(w)) + 1)
        w[w.index(max(w))] = 0
print(*a, *b)
