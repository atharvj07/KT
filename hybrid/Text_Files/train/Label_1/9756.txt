#include <cstdio>
#include <vector>
#include <set>
#include <cassert>
#define pb push_back
using namespace std;
#define ll long long
#define maxn 5010
ll n, m, arr[maxn], op, l, r, d, x, t[maxn], t1[maxn], a[maxn];
vector<pair<ll, ll> > range[maxn];
vector<ll> v;
set<ll> st;
int main() {
  scanf("%lld %lld", &n, &m);
  for (ll i = 1; i <= m; i++) {
    scanf("%lld", &op);
    if (op == 1) {
      scanf("%lld %lld %lld", &l, &r, &d);
      st.insert(l);
      st.insert(r);
      range[l].pb(make_pair(r, d));
    } else {
      scanf("%lld %lld %lld", &l, &r, &x);
      t[l] += x;
      t[r + 1] -= x;
      t1[l] += x;
      t1[r + 1] += x;
    }
  }
  ll sum = 0;
  for (auto it : st) {
    ll l = it;
    for (auto p : range[it]) sum = max(sum, p.second + t[l]), l = p.first;
    v.pb(sum);
    sum = 0;
    for (ll i = l + 1; i <= n; i++) sum = max(sum, t[i]);
    v.pb(sum);
    sum = 0;
  }
  for (auto it : v) a[it]++;
  for (ll i = 1; i <= n; i++) {
    if (a[i] == 0) { puts("NO"); return 0; }
  }
  ll sum1 = 0;
  for (ll i = 1; i <= n; i++) {
    sum1 = max(sum1, t1[i]);
    if (sum1 != v[i]) {
      puts("NO");
      return 0;
    }
  }
  puts("YES");
  for (ll i = 1; i <= n; i++) printf("%lld ", v[i]);
}
