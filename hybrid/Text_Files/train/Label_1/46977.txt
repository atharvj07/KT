import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int free = 0;
        int untreated = 0;
        for (int i = 0; i < n; i++) {
            int x = in.nextInt();
            if (x == -1) {
                if (free == 0) {
                    untreated++;
                } else {
                    free--;
                }
            } else {
                free += x;
            }
        }
        System.out.println(untreated);
    }
}
