import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int t = in.nextInt();
    while (t-- > 0) {
      int n = in.nextInt();
      int[] a = new int[n];
      Set<Integer> s = new HashSet<>();
      for (int i = 0; i < n; i++) {
        int x = 1;
        while (s.contains(x)) x++;
        a[i] = x;
        s.add(x);
      }
      for (int i = 0; i < n; i++) {
        System.out.print(a[i] + " ");
      }
      System.out.println();
    }
  }
}
