#include <bits/stdc++.h>
using namespace std;
long long n, m, S[105], dp[105];
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> S[i];
  }
  for (int i = 1; i <= n; i++) {
    dp[i] = S[i];
    for (int j = 1; j < i; j++) {
      if (j != i - 1) dp[i] = max(dp[i], dp[j] + min(S[i], (long long)i - j));
      else
        dp[i] = max(dp[i], dp[j] + S[i]);
    }
  }
  cout << dp[n];
}
