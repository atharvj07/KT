import sys

n, m = map(int, input().split())

grid = [list(input()) for _ in range(n)]

for i in range(1, n - 1):
    for j in range(1, m - 1):
        if grid[i][j] == '.':
            if grid[i - 1][j] == grid[i + 1][j] == grid[i][j - 1] == grid[i][j + 1] == '#':
                grid[i][j] = '#'
if grid[1][1] == grid[1][m - 2] == grid[n - 2][1] == grid[n - 2][m - 2] == '#':
    print("YES")
else:
    print("NO")
