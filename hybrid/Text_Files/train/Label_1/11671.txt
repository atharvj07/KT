#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 505, M = 2000005;
int n, m, tot, d[N][N], x[N], vis[N], ans[N];
vector<pair<int, int>>g[N];
struct node {
    int now, dis;
    bool operator<(const node&b) const {
        return dis > b.dis;
    }
}nd;
void dijkstra(int st) {
    memset(d[st], 0x3f, sizeof(d[st]));
    memset(vis, 0, sizeof(vis));
    priority_queue<node>q;
    q.push({st, d[st][st] = 0});
    while(!q.empty()) {
        node now = q.top();q.pop();
        int x = now.now;
        if(vis[x]) continue;
        vis[x] = 1;
        for(auto i : g[x]) {
            int v = i.first, w = i.second;
            if(!vis[v] && d[st][v] > d[st][x] + w) {
                d[st][v] = d[st][x] + w;
                q.push({v, d[st][v]});
            }
        }
    }
}
signed main() {
    cin >> n;
    for(int i = 1; i <= n; i++)
        for(int j = 1; j <= n; j++) {
            int x; cin >> x;
            if(i == j) continue;
            g[i].push_back({j, x});
            g[j].push_back({i, x});
            m += x;
        }
    for(int i = 1; i <= n; i++) cin >> x[i];
    for(int i = 1; i <= n; i++) dijkstra(i);
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= n; j++)
            if(i != j) ans[i] += d[i][j] + d[j][i];
        ans[i] *= i - 1;
    }
    for(int i = 1; i <= n; i++) {
        int pos = x[i];
        for(auto it : g[pos]) {
            int v = it.first, w = it.second;
            ans[i] -= ((i - 1) * (i - 1) * w + (w * (w + 1) / 2) * (n - i));
            if(v > i) ans[i] -= (w * (i - 1));
        }
    }
    for(int i = 1; i <= n; i++) cout << ans[i] << ' ';
    cout << endl;
}
