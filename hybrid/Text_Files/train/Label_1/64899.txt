import sys

n = int(sys.stdin.readline())
l = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]
l=sorted(l,key=lambda x:x[1])
ans=0
for i in range(n):
    ans=max(ans,l[i][0]+1)
    l[i][0]+=l[i][1]
for i in range(1,n):
    l[i][0]=max(l[i][0],l[i-1][0])
    ans=max(ans,l[i][0]+1)
print(ans)
