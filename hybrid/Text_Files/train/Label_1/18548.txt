#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 998244353;
int dp[2005][2005][3][2];
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  dp[0][0][0][0] = dp[0][0][1][0] = 1;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      for (int f = 0; f < 2; ++f) {
        for (int p = 0; p < 2; ++p) {
          if (i == 0 && p && j == k) continue;
          int &res = dp[i + 1][j][f][p];
          (res += ll(dp[i][j][f][p]) * m % mod) %= mod;
          (res += ll(dp[i][j][f][p]) * (m - 1) % mod) %= mod;
          if (f == 0) (res += ll(dp[i][j][1][p]) * (m - 1) % mod) %= mod;
          if (p && k > 0) {
            (res += ll(dp[i][j][0][0]) * j % mod) %= mod;
            (res += ll(dp[i][j][1][0]) * j % mod) %= mod;
          }
        }
      }
    }
  }
  cout << dp[n][m][0][1] << '\n';
}
