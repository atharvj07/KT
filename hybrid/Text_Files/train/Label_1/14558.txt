#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    long long sum = 0;
    for (int &x : a) {
      cin >> x;
      sum += x;
    }
    sort(a.begin(), a.end());
    long long ans = -1;
    for (int i = 0; i < n; i++) {
      if (sum - a[i] <= (long long)n * a[i]) {
        ans = i;
        break;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
