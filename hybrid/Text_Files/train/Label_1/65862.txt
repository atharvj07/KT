def solve():
    m={(1):4,(2):12,(3):28,(4):48,(5):72,(6):100,(7):132}
    n, b, x = map(int, input().split())
    a = list(map(int, input().split()))
    b = [i-1 for i in a]
    if x == 1 or (x == 2 and n % 2 == 0):
        print(1)
    else:
        ans = 0
        for i in b:
            ans += m[i % x]
            ans %= 998244353
        print(ans)
solve()
