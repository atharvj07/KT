import java.util.Scanner;

public class CaracalVsMonster {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long monsterHealth = scanner.nextLong();

        long attacks = 0;
        while (monsterHealth > 0) {
            attacks++;
            // Find the number of monsters
            long monsters = 1;
            while (monsterHealth > 1) {
                monsters *= 2;
                monsterHealth /= 2;
            }
            // Update monster health after attacking
            while (monsters > 0) {
                monsterHealth -= Math.min(monsterHealth, 1);
                monsters--;
            }
        }

        System.out.println(attacks);
    }
}
