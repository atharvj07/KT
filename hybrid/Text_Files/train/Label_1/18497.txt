#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int i = 0;
  while (i < s.size()) {
    if (s[i] == 'y' && s[i + 1] == 'x') {
      swap(s[i], s[i + 1]);
      i += 2;
    } else if (s[i] == 'x' && s[i + 1] == 'y') {
      s.erase(i, 2);
    } else {
      i++;
    }
  }
  cout << s << endl;
  return 0;
}
