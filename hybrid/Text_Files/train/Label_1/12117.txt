#include <bits/stdc++.h>
using namespace std;
const int MX = 2e5 + 5;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  if (n == k) {
    cout << 0 << endl;
    return 0;
  }
  k = n - k;
  int ans = 1e9 + 7;
  for (int i = 0; i < n; i++) {
    if (i == 0 || a[i] != a[i - 1]) {
      int cnt = 0;
      int mn = 1e9 + 7;
      for (int j = i; j < n; j++) {
        if (a[j] == a[i]) {
          cnt++;
        } else {
          mn = min(mn, a[j]);
        }
      }
      if (cnt > k) {
        ans = min(ans, 0);
      } else {
        ans = min(ans, mn - a[i]);
      }
    }
  }
  cout << ans << endl;
  return 0;
}
