MOD = 10**9 + 7

def solve(n, m, labyrinth):
    dp = [[0] * m for _ in range(n)]
    right = [[0] * m for _ in range(n)]
    down = [[0] * m for _ in range(n)]
    dp[0][0] = 1
    
    for i in range(n):
        for j in range(m):
            if i > 0:
                dp[i][j] += down[i-1][j]
                dp[i][j] %= MOD
            if j > 0:
                dp[i][j] += right[i][j-1]
                dp[i][j] %= MOD
                
            if labyrinth[i][j] == '.':
                if i < n - 1:
                    down[i][j] = dp[i][j]
                if j < m - 1:
                    right[i][j] = dp[i][j]
            else:
                if i < n - 1:
                    down[i+1][j] = down[i][j]
                if j < m - 1:
                    right[i][j+1] = right[i][j]
                    
    return dp[n-1][m-1]

n, m = map(int, input().split())
labyrinth = [input() for _ in range(n)]
print(solve(n, m, labyrinth))
