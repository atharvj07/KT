#include <bits/stdc++.h>
using namespace std;
int n, k, a[50];
int main() {
    cin >> n >> k;
    a[0] = 1;
    for (int i = 1; i < n; i++) {
        a[i] = a[i - 1] * 2;
        int j = 1;
        while (j < a[i]) {
            if (!binary_search(a, a + i, j)) {
                a[i]++;
                break;
            }
            j++;
        }
    }
    cout << a[k - 1] << endl;
    return 0;
}
