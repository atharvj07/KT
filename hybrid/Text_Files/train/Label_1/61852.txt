t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    odd_count = len([num for num in a if num % 2 != 0])
    even_count = n - odd_count
    if odd_count == 0:
        print("No")
    elif even_count == 0 and x % 2 == 0:
        print("No")
    elif odd_count % 2 == 0 and even_count == 0 and x == n:
        print("No")
    else:
        print("Yes")
