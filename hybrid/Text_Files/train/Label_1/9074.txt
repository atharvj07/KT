#include <bits/stdc++.h>

using namespace std;

const int MAXN = 105;
const int MAXK = 1e4 + 5;

int a[MAXN];

int main() {
  int t;
  cin >> t;

  while (t--) {
    int n, k;
    cin >> n >> k;

    for (int i = 1; i <= n; ++i) { cin >> a[i]; }

    sort(a + 1, a + 1 + n);

    for (int i = 2; i <= n && k; ++i) {
      int take = min(k, a[i - 1]);
      a[i - 1] -= take;
      a[i] += take;
      k -= take;
    }

    for (int i = 1; i <= n; ++i) { cout << a[i] << ' '; }
    cout << '\n';
  }

  return 0;
}
