#include<bits/stdc++.h>

using namespace std;

int main(){
    int n; cin >> n;
    vector<long long> b(n / 2);
    for(int i = 0; i < n / 2; ++i) cin >> b[i];
    vector<long long> ans(n);
    for(int i = 0; i < n / 2; ++i){
        ans[i] = b[i] / 2;
        ans[n - i - 1] = b[i] - ans[i];
    }
    for(int x : ans) cout << x << ' ';
}
