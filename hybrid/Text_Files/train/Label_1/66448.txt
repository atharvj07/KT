import sys

mod = 10**9 + 7

n = int(input())
arr = []

for _ in range(n):
    out, inn = map(int, input().split())
    arr.append((out, inn))

arr.sort()

dp = [0] * n

dp[0] = arr[0][0]

mx = 0
ans = 0

for i in range(1, n):
    out, inn = arr[i]
    idx = mx
    while idx >= 0 and out > arr[idx][1]:
        idx -= 1
    if idx == -1:
        dp[i] = out
        mx += 1
    else:
        dp[i] = dp[idx] + out - arr[idx][1]

    if dp[i] == dp[mx]:
        ans += 1
    elif dp[i] > dp[mx]:
        mx = i
        ans = 1

print(ans)
