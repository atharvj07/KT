#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN=1e5+1;
ll n,m,a[MAXN],b[MAXN];
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++)cin>>a[i];
    for(int i=1;i<=m;i++)cin>>b[i];
    sort(a+1,a+n+1);
    sort(b+1,b+m+1);
    ll ans=0;
    for(int i=1;i<=n;i++){
        ans+=max(a[i],b[m-i+1]);
    }
    if(a[n]>b[1]){cout<<"-1\n";return 0;}
    cout<<ans<<endl;
    return 0;
}
