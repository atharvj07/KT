#include <bits/stdc++.h>
#include <cmath>
#define debug(x) cerr << #x << ": " << x << endl;
using namespace std;
#define rep(i, n) for (int i = 0; i < n; i++)

int main() {
  int n;
  cin >> n;
  pair<double, double> x = make_pair(0.0, 0.0);
  pair<double, double> ans = make_pair(0.0, 0.0);
  rep(i, n) {
    int a;
    int b;
    cin >> a >> b;
    pair<double, double> vec = make_pair(a - x.first, b - x.second);
    pair<double, double> vecc = make_pair(vec.second, -vec.first);
    if (vec.first >= 0 && vecc.second >= 0) {
      ans = vec;
    } else if (vec.first >= 0 && vecc.second < 0) {
      ans = vecc;
    } else if (vec.first < 0 && vecc.second < 0) {
      ans = vec;
    } else if (vec.first < 0 && vecc.second >= 0) {
      ans = vecc;
    }
  }
  double ansr = acos(ans.first / sqrt(ans.first * ans.first + ans.second * ans.second));
  ansr *= 180.0 / M_PI;
  if (ans.second < 0) {
    ansr = 180.0 - ansr;
  }
  cout << setprecision(10) << fixed << ansr << endl;
}
