#include <iostream>
#include <vector>
#include <cstring>
using namespace std;
typedef uint64_t ll;

const int mod = 1e9 + 7;
const int mxN = 1e3 + 5;
ll dp[mxN][mxN], n, m, r[mxN], c[mxN];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) cin >> r[i];
    for (int i = 1; i <= m; ++i) cin >> c[i];
    dp[1][0] = 1;
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j < m; ++j)
            if (j + 1 <= c[j + 1])
                dp[i][j + 1] = (dp[i][j + 1] + dp[i][j]) % mod;
        for (int j = 1; j <= m; ++j) {
            if (j + 1 <= c[j + 1] && i + 1 <= r[i + 1])
                dp[i + 1][j + 1] = (dp[i + 1][j + 1] + dp[i][j + 1]) % mod;
        }
    }
    cout << dp[n][m] << endl;
}
