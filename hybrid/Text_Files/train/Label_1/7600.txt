#include <bits/stdc++.h>
 
using namespace std;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int q;
    cin >> q;
    while (q--) {
        int n, k;
        cin >> n >> k;
        vector<int> a(n);
        for (int &ai : a) {
            cin >> ai;
            ai %= 2;
        }
        if (n % 2 != k % 2) {
            cout << "NO\n";
        } else {
            cout << "YES\n";
            for (int i = 0; i < k - 1; i++) {
                cout << (i + 1) << ' ';
            }
            cout << n << '\n';
        }
    }
    return 0;
}
