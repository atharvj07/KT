import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int b = scanner.nextInt();

        int answer = 0;
        StringBuilder cards = new StringBuilder();

        if (a % 2 != b % 2) {
            answer = -1;
            if (a % 2 == 1) {
                cards.append("o");
                a--;
            } else {
                cards.append("x");
                b--;
            }
        }

        while (a > 0 && b > 0) {
            if (a >= 2) {
                answer += 4;
                cards.append("oo");
                a -= 2;
            } else {
                answer -= 4;
                cards.append("xx");
                b -= 2;
            }
        }

        while (a > 0) {
            answer += a * a;
            cards.append("o".repeat(a));
            a = 0;
        }

        while (b > 0) {
            answer -= b * b;
            cards.append("x".repeat(b));
            b = 0;
        }

        System.out.println(answer);
        System.out.println(cards);
    }
}
