import java.util.*;
class Main {
    static int solve() {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), a = in.nextInt(), b = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = in.nextInt();
        long sum = 0;
        for (int x: arr) sum += (long)x;
        int GCD = 0;
        for (int x: arr) GCD = HCF(GCD, x);
        if (GCD > 1) return 0;
        for (int x: arr) {
            if (x == GCD && GCD != 1) return b;
            if (x % GCD == 0) sum -= (long)x;
        }
        if (sum <= (long)b) return (int)sum;
        if (a + GCD <= b) return a + GCD;
        return -1;
    }
    static int HCF(int a, int b) {
        if (b == 0) return a;
        return HCF(b, a % b);
    }
    public static void main(String[] args) {
        System.out.println(solve());
    }
}
