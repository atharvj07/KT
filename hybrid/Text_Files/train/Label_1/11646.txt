#include <iostream>
#include <algorithm>

// Do not submit this line
int main(int argc, char const *argv[]) {
    int n; std::cin >> n;
    int p[n + 1];
    for (int i = 0; i < n; ++i) {
        std::cin >> p[i + 1];
    }
    int mx = 0, seg[n + 1];
    for (int i = 1; i < n + 1; ++i) {
        if (p[i] > mx) {
            seg[i] = 1;
            mx = p[i];
        } else seg[i] = 0;
    }
    mx = 0, seg[n + 1] = 0;
    for (int i = n; i >= 1; --i) {
        if (p[i] > mx) {
            seg[i] += seg[i + 1];
            mx = p[i];
        }
    }
    long long ans = 0;
    for (int i = 1; i < n + 1; ++i) {
        ans += 1LL * seg[i] * (seg[i] - 1) / 2;
    }
    std::cout << ans << std::endl;
    return 0;
}
