import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int ct = sc.nextInt();
        while (ct-- > 0) {
            List<Integer> lst = new ArrayList<>();
            int n = sc.nextInt();
            for (int i = 1; i <= n / 2; i++) {
                lst.add(i);
                lst.add(n - i);
            }
            if (n % 2 == 1) lst.add(n / 2);
            for (int i = 0; i < lst.size(); i++) System.out.print(lst.get(i) + " ");
            System.out.println();
        }
    }
}
