n = int(input())
words = [input() for _ in range(n)]
encoded_message = input()

def check_message(words, encoded_message):
    pattern = "<3" + "<3".join(words) + "<3"
    i = 0
    for char in pattern:
        i = encoded_message.find(char, i) + 1
        if i == 0:
            return "no"
    return "yes"

print(check_message(words, encoded_message))
