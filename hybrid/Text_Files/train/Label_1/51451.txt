import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String weights = scanner.nextLine();
        int m = scanner.nextInt();
        boolean[] available = new boolean[11];
        boolean possible = true;
        int[] solution = new int[m];
        for (int i = 1; i <= 10; i++) {
            available[i] = weights.charAt(i - 1) == '1';
        }
        int left = 0, right = 0;
        for (int i = 0; i < m; i++) {
            for (int j = 10; j > 0; j--) {
                if (available[j] && (left <= right || j != solution[i - 1])) {
                    solution[i] = j;
                    if (left <= right) {
                        left += j;
                    } else {
                        right += j;
                    }
                    available[j] = false;
                    break;
                }
            }
            if (solution[i] == 0) {
                possible = false;
                break;
            }
        }
        if (possible) {
            System.out.println("YES");
            for (int weight : solution) {
                System.out.print(weight + " ");
            }
        } else {
            System.out.println("NO");
        }
    }
}
