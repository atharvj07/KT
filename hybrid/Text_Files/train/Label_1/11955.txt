#include <iostream>
#include <cstring>

int main() {
  int n;
  std::cin >> n;
  char s[n + 1];
  std::cin >> s;
  int cnt = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] != '(' && !cnt) {
      std::cout << ':';
      std::cout << '(';
      s[i] = '(';
      cnt = 1;
    }
    if (s[i] == '(') cnt--;
    if (s[i] == '?') s[i] = ')';
  }
  if (cnt)
    std::cout << ':';
  else
    std::cout << s;
  std::cout << '(';
  return 0;
}
