n, k = map(int, input().split())

if k > n * (n - 1) / 2:
    print(-1)
else:
    a = [[0] * n for _ in range(n)]
    i, j = 0, n - 1
    while k > 0:
        a[i][j] = 1
        k -= 1
        i += 1
        j -= 1
        if i == n:
            i = 0
        if j < 0:
            j = n - 1
    for row in a:
        print(*row)
