#include <bits/stdc++.h>
using namespace std;

pair<string, int> p[100010];
int r[100010];

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; ++i) {
    string s;
    cin >> s;
    int a, b;
    cin >> a >> b;
    p[i] = make_pair(s, b);
    r[i] = a;
  }
  for (int i = 1; i <= m; ++i) {
    sort(p, p + n);
    int cnt = 0;
    string t1 = "", t2 = "";
    bool q = false;
    for (int j = 0; j < n; ++j) {
      if (r[j] == i) {
        cnt++;
        if (cnt == 1) {
          t1 = p[j].first;
        }
        if (cnt == 2) {
          t2 = p[j].first;
        }
        if (cnt >= 3) {
          q = true;
          break;
        }
      }
    }
    if (q) {
      cout << "?" << endl;
    } else if (t1 != "" && t2 != "") {
      cout << t1 << " " << t2 << endl;
    } else {
      cout << "?" << endl;
    }
  }
  return 0;
}
