#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n;
  scanf("%d", &n);
  vector<int>x(n);
  for (int i = 0; i < n; i++) scanf("%d", &x[i]);
  sort(x.begin(), x.end());
  vector<int>l(n, 1), r(n, 1);
  for (int i = 1; i < n; i++) {
    if (x[i] >= x[i - 1]) l[i] = l[i - 1] + 1;
    if (x[n - i - 1] >= x[n - i]) r[n - i - 1] = r[n - i] + 1;
  }
  int ans = 0;
  for (int i = 0; i < n; i++)
    ans = max(ans, max(l[i], r[i]));
  printf("%d\n", ans);
}
