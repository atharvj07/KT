import java.util.*;
import java.io.*;

public class Main {

    static class Swap {
        int a;
        int b;
        public Swap(int a, int b) {
            this.a = a;
            this.b = b;
        }
    }

    static int n, m, k;
    static int count0, count1;
    static String initial, ideal;
    static Swap[] swaps;
    static int[][] dp;
    static int[] sum0, sum1;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        n = Integer.parseInt(s[0]);
        m = Integer.parseInt(s[1]);
        k = Integer.parseInt(s[2]);
        initial = br.readLine();
        ideal = br.readLine();

        swaps = new Swap[n];
        for (int i = 0; i < n; i++) {
            s = br.readLine().split(" ");
            swaps[i] = new Swap(Integer.parseInt(s[0]), Integer.parseInt(s[1]));
        }

        initialize();

        System.out.println(dp[n][m]);
        int minL = 1;
        int minR = m;
        for (int i = m; i <= n; i++) {
            for (int j = 1; j <= n - i + 1; j++) {
                if (dp[i][j] > dp[minR][minL]) {
                    minR = i;
                    minL = j;
                }
            }
        }
        System.out.println(minL + " " + minR);
    }

    static void initialize() {
        count0 = 0;
        count1 = 0;
        for (char c : initial.toCharArray()) {
            if (c == '0') count0++;
            else count1++;
        }

        sum0 = new int[n + 1];
        sum1 = new int[n + 1];
        for (int i = 0; i < n; i++) {
            int j = i + 1;
            int a = swaps[i].a;
            int b = swaps[i].b;
            int v0 = (initial.charAt(a - 1) == '0' ? -1 : 1);
            int v1 = (initial.charAt(b - 1) == '1' ? -1 : 1);
            sum0[j] = sum0[i] + v0;
            sum1[j] = sum1[i] + v1;
        }

        dp = new int[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                dp[i][j] = dp[i - 1][j];
                if (i >= j) {
                    int a = swaps[i - j].a;
                    int b = swaps[i - j].b;
                    int c0 = (ideal.charAt(a - 1) == '0' ? 1 : -1);
                    int c1 = (ideal.charAt(b - 1) == '1' ? 1 : -1);
                    int t0 = dp[i - 1][j] + c0 * (sum0[i] - sum0[i - j]);
                    int t1 = dp[i - 1][j] + c1 * (sum1[i] - sum1[i - j]);
                    dp[i][j] = Math.max(dp[i][j], Math.max(t0, t1));
                }
            }
        }
    }
}
