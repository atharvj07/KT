def max_score(test_cases):
    results = []

    for n, C, T, problems in test_cases:
        problems.sort(reverse=True, key=lambda x: x[1])  # Sort problems by score in descending order

        max_score = 0
        for start in range(n):
            time_left = T
            s = 1.0 + C * time_left  # Initially assume all time is used for training
            score = 0
            for i in range(start, n):
                ai, pi = problems[i]
                watch_time = 10
                solve_time = ai / s
                if time_left >= watch_time + solve_time:
                    score += pi
                    time_left -= (watch_time + solve_time)
                    s *= 0.9  # Decrease skill after solving a problem
                else:
                    break
            max_score = max(max_score, score)

        results.append(max_score)

    return results

# Example usage
tc = int(input().strip())
test_cases = []
for _ in range(tc):
    n = int(input().strip())
    C, T = map(float, input().strip().split())
    problems = [list(map(int, input().strip().split())) for _ in range(n)]
    test_cases.append((n, C, T, problems))

results = max_score(test_cases)
for result in results:
    print(result)
