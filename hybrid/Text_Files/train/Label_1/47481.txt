import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), c = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int maxProfit = 0;
        for (int i = 1; i < n; i++) {
            int profit = x[i] - x[i - 1];
            if (profit > c) {
                profit -= c;
                maxProfit += profit;
            }
        }
        System.out.println(maxProfit);
    }
}
