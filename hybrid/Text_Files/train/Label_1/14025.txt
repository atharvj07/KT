#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
  ll n, w;
  cin >> n >> w;
  vector<ll> a(2 * n);
  for (ll i = 0; i < 2 * n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end(), greater<ll>());
  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    if (a[i] <= a[i + n] / 2) {
      ans += a[i];
    } else {
      ans += a[i + n] / 2;
    }
  }
  cout << ans << endl;
  return 0;
}
