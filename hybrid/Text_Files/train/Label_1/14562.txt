#include <bits/stdc++.h>
#define ll long long

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    ll sum1 = 0, sum2 = 0;
    for (int i = 0; i < n / 2; i++) { sum1 += a[i]; }
    for (int i = n / 2; i < n; i++) { sum2 += a[i]; }
    if (sum1 == sum2) {
      cout << 0 << endl;
    } else if (a[n / 2 - 1] == a[n / 2]) {
      cout << 1 << endl;
    } else {
      cout << -1 << endl;
    }
  }
  return 0;
}
