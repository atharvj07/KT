import sys

n, m, k = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
offers = []
for i in range(m):
    offers.append(tuple(map(int, sys.stdin.readline().split())))
dp = [float('inf')] * (k + 1)
dp[0] = 0
a.sort()
for i in range(1, k + 1):
    for j in range(n):
        if i - a[j] >= 0:
            dp[i] = min(dp[i], dp[i - a[j]] + a[j])
    for j in range(m):
        x, y = offers[j][0], offers[j][1]
        if i - a[y-1] >= 0:
            dp[i] = min(dp[i], dp[i - a[y-1]] + a[y])
print(dp[-1])
