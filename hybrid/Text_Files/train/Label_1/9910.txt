#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e3;
vector<int> g[maxn];
int vis[maxn];

void dfs(int u) {
  vis[u] = 1;
  for (int v : g[u]) {
    if (!vis[v]) dfs(v);
  }
}

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  dfs(1);
  bool valid = true;
  for (int i = 1; i <= n; i++) {
    if (!vis[i]) {
      valid = false;
    }
  }
  cout << (valid ? "yes" : "no") << endl;
  return 0;
}
