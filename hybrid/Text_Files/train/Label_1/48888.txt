import java.util.*;
import java.lang.*;

class Main {
    static final int INF = 1000000000;
    static int n, m, s, t;
    static ArrayList<ArrayList<Integer>> adj = new ArrayList<>();
    static int[] dist;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        s = sc.nextInt() - 1;
        t = sc.nextInt() - 1;
        for (int i = 0; i < n; i++) {
            adj.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            adj.get(a).add(b);
            adj.get(b).add(a);
        }
        dist = new int[n];
        Arrays.fill(dist, INF);
        bfs(s);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (dist[i] != INF && dist[j] != INF && dist[i] + 1 == dist[j]) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }

    static void bfs(int start) {
        Queue<Integer> q = new LinkedList<>();
        q.offer(start);
        dist[start] = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : adj.get(u)) {
                if (dist[v] == INF) {
                    dist[v] = dist[u] + 1;
                    q.offer(v);
                }
            }
        }
    }
}
