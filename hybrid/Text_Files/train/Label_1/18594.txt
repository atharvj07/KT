#include <iostream>
const long long mod = 1e9 + 7;

int main() {
  long long n;
  std::cin >> n;
  std::string s, t;
  std::cin >> s >> t;
  long long ans = 0;
  for (long long i = 0; i < n; i++) {
    if (s[i] == t[i]) {
      if (s[i] != '?' && t[i] != '?') continue;
      ans = (ans + 9) % mod;
    } else {
      if (s[i] != '?' && t[i] != '?') break;
      ans = (ans + 99) % mod;
    }
  }
  std::cout << (ans + 1) % mod << '\n';
}
