for _ in range(int(input())):
    n, k = map(int, input().split())
    if k > (n - 1) * (n - 2) // 2 or n - 1 - k < 0:
        print(-1)
    else:
        arr = [i for i in range(1, n + 1)]
        for j in range(0, k):
            arr[2 * j], arr[2 * j + 1] = arr[2 * j + 1], arr[2 * j]
        print(*arr)
