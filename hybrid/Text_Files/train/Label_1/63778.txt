import sys
n = int(input())
a = [0] * (2 * n + 1)
b = [0] * (2 * n + 1)
for _ in range(n):
    x, y = map(int, input().split())
    a[x], b[y] = y, x
if not all(a[i] <= a[i + 1] for i in range(1, 2 * n)) or not all(
        b[i] >= b[i + 1] for i in range(1, 2 * n)):
    print(-1)
    sys.exit()
print((2 * n - len({*a[1:2 * n]} & {*b[1:2 * n]}) + 1) // 2)
