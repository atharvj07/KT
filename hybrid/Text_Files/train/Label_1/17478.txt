#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, vector<int>>> companies(n);
    for (int i = 0; i < n; i++) {
        int m;
        cin >> m;
        vector<int> salaries(m);
        for (int j = 0; j < m; j++) {
            cin >> salaries[j];
        }
        companies[i] = {m, salaries};
    }
    sort(companies.begin(), companies.end());
    int total_increase = 0;
    for (int i = 1; i < n; i++) {
        if (companies[i].second[0] != companies[i - 1].second[0]) {
            total_increase += companies[i].second[0] - companies[i - 1].second[0];
        }
    }
    cout << total_increase << endl;
    return 0;
}
