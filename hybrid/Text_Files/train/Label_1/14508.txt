#include <bits/stdc++.h>
using namespace std;
#define int long long

int32_t main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 0, cur = -2e9;
    for (int i = 0; i < n; i++) {
      if (a[i] >= cur)
        cur = a[i];
      else
        ans++;
      a[i] = cur;
    }
    cout << ans << '\n';
  }
  return 0;
}
