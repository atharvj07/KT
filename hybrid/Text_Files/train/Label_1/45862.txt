import java.util.Scanner;

public class StrongPassword {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String[] dangerousPasswords = new String[n];
        for (int i = 0; i < n; i++) {
            dangerousPasswords[i] = scanner.next();
        }

        Trie trie = new Trie();
        for (String dangerousPassword : dangerousPasswords) {
            trie.insert(dangerousPassword);
        }

        String password = "";
        for (char c = 'a'; c <= 'z'; c++) {
            if (!trie.isPrefix(password + c)) {
                password += c;
                break;
            }
        }

        System.out.println(password);
    }

    static class Trie {
        TrieNode root;

        public Trie() {
            root = new TrieNode();
        }

        public void insert(String word) {
            TrieNode node = root;
            for (char c : word.toCharArray()) {
                if (node.children[c - 'a'] == null) {
                    node.children[c - 'a'] = new TrieNode();
                }
                node = node.children[c - 'a'];
            }
            node.isWord = true;
        }

        public boolean isPrefix(String prefix) {
            TrieNode node = root;
            for (char c : prefix.toCharArray()) {
                if (node.children[c - 'a'] == null) {
                    return false;
                }
                node = node.children[c - 'a'];
            }
            return true;
        }
    }

    static class TrieNode {
        TrieNode[] children;
        boolean isWord;

        public TrieNode() {
            children = new TrieNode[26];
        }
    }
}
