#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  long long n;
  n = stoll(s);
  int len = s.length();
  if ((n % 100 == 0) || ((n % 10) % 4 == 0 && s[len - 2] % 2 == 0)) {
    cout << 0 << endl;
    return 0;
  }
  if (((n % 4 == 0) && (n % 2 == 0)) || (n % 10 == 0) || (n % 10 == 5)) {
    cout << 1 << endl;
    return 0;
  }
  int ind = (len) % 2;
  for (int i = len - 1; i >= 0; i -= 2) {
    long long x = 0, y = 0;
    if (ind == 0) {
      x = (n / 10), y = (n % 10);
    } else {
      x = (n % 10), y = (n / 10);
    }
    if (x % 4 == 0 || (x % 2 == 0 && y % 5 == 0)) {
      cout << i / 2 << endl;
      return 0;
    }
    ind ^= 1;
    n = x;
  }
  cout << -1 << endl;
  return 0;
}
