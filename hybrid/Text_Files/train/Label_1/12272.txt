#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
#define mod (1000000007LL)

int n, dp[1010], sum[1010];

int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	cin >> n;
	dp[0] = 1;
	dp[1] = 1;
	sum[0] = 1;
	sum[1] = 1;
	for (int i = 2; i <= n; ++i) {
		for (int j = 0; j < i; ++j) {
			dp[i] = (dp[i] + (1LL * dp[j] * sum[i - j - 1]) % mod) % mod;
		}
		sum[i] = (sum[i - 1] + dp[i]) % mod;
	}
	cout << dp[n] << endl;
	return 0;
}
