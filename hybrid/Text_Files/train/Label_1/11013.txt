#include <iostream>
#include <algorithm>
#include <vector>

int main() {
    long long n, k;
    std::cin >> n >> k;
    std::vector<long long> a(n);
    for (long long i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    std::sort(a.begin(), a.end());
    if (k > a[k - 1]) {
        std::cout << "-1" << std::endl;
        return 0;
    }
    std::cout << a[k - 1] << std::endl;
    return 0;
}
