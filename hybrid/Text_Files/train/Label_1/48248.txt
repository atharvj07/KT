import java.util.Scanner;
import java.math.BigInteger;

class Main {
    private static final long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int x = sc.nextInt();
        long ans = ((BigInteger.TWO.modPow(BigInteger.valueOf(m), BigInteger.valueOf(MOD)).subtract(BigInteger.TWO.modPow(BigInteger.valueOf(x), BigInteger.valueOf(MOD)))).modPow(BigInteger.valueOf(n - 1), BigInteger.valueOf(MOD))).longValue();
        System.out.println(ans);
    }
}
