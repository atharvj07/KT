#include <iostream>
#include <vector>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        int x = 0;
        for (int i = 0; i < n; i++) x ^= a[i];
        if (x)
            cout << "T\n";
        else
            cout << "HL\n";
    }
    return 0;
}
