#include <bits/stdc++.h>
using namespace std;
const int mxn=2e5+1;
typedef long long ll;
int main(){
    ll n, ans=0, a[mxn], deg[mxn]={0};
    cin>>n;
    for(int i=1;i<=n;++i){
        cin>>a[i];
        deg[i]+=a[i];
        deg[a[i]+1]-=1;
    }
    for(int i=1;i<=n;++i){
        ans+=deg[i];
        if(ans<0){
            cout<<"Impossible";
            return 0;
        }
    }
    cout<<"Possible\n";
    vector<ll> ord;
    for(int i=1;i<=n;++i){
        if(deg[i]==1) ord.push_back(i);
    }
    for(int i=1;i<=n;++i){
        if(a[i]==0) ord.push_back(i);
    }
    for(int i=1;i<=n;++i){
        if(deg[i]==3) ord.push_back(i);
    }
    for(int i=0;i<ord.size();++i){
        cout<<ord[i]<<" ";
    }
    return 0;
}
