n, k = map(int, input().split())
a = sorted(map(int, input().split()))
ans = -1
for i in range(k):
    tot = a[i] * 1000
    req = (tot * n) // 1000
    need = (req + 999) // 1000
    j = i + 1
    while j < k and need > 0:
        tot += a[j] * 1000
        req = (tot * n) // 1000
        need = (req + 999) // 1000
        j += 1
    if need <= 0:
        if ans == -1 or ans > need:
            ans = need
print(ans)
