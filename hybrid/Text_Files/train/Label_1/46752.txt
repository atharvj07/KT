import java.util.Scanner;
import java.util.LinkedList;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        LinkedList<Integer>[] chains = new LinkedList[k];
        for (int i = 0; i < k; i++) {
            chains[i] = new LinkedList<>();
        }
        boolean[] visited = new boolean[n + 1];
        for (int i = 0; i < k; i++) {
            int m = sc.nextInt();
            for (int j = 0; j < m; j++) {
                int x = sc.nextInt();
                chains[i].add(x);
                visited[x] = true;
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                ans++;
            }
        }
        for (LinkedList<Integer> chain : chains) {
            ans += chain.size() - 1;
        }
        System.out.println(ans);
    }
}
