#include <iostream>

std::pair<int,int> mp[4];
int n,m,x,y,vx,vy;

bool check(int & p) {
    for (int i=0;i<4;++i) {
        if (mp[i].first==x&&mp[i].second==y) {
            p=i;
            return 1;
        }
    }
    return 0;
}

int main() {
    std::cin >> n >> m >> x >> y >> vx >> vy;
    mp[0].first=0, mp[0].second=0;
    mp[1].first=n, mp[1].second=0;
    mp[2].first=0, mp[2].second=m;
    mp[3].first=n, mp[3].second=m;
    int p;
    if (check(p)) {std::cout << mp[p].first << ' ' << mp[p].second << std::endl; return 0;}
    if (vx!=0) {
        if (vy!=0) {
            int t=std::min((n-x)/(-vx),(m-y)/(-vy));
            x-=t*vx, y-=t*vy;
            if (check(p)) {std::cout << mp[p].first << ' ' << mp[p].second << std::endl; return 0;}
            vx=-vx, vy=-vy;
        } else {
            if ((-vx>0&&x==n)||(-vx<0&&x==0)) {std::cout << -1 << std::endl; return 0;}
        }
    } else {
        if (vy!=0) {
            if ((-vy>0&&y==m)||(-vy<0&&y==0)) {std::cout << -1 << std::endl; return 0;}
        }
    }
    int w1=x*(-vy), w2=y*(-vx);
    if (w1>w2) {
        if (((-vy>0&&y==m)||(-vy<0&&y==0))&&(x!=n||vx!=0)) {std::cout << -1 << std::endl; return 0;}
        if ((-vx<0&&x==0)||(-vx>0&&x==n)) {std::cout << mp[y/(-vy)].first << ' ' << mp[y/(-vy)].second << std::endl; return 0;}
        int t=(n-x)/(-vx);
        if (t*(m-y)>=std::max(y*(n-x),x*(m-y))) {std::cout << mp[y/(-vy)].first << ' ' << mp[y/(-vy)].second << std::endl; return 0;}
        std::cout << mp[n/(-vx)].first << ' ' << mp[n/(-vx)].second << std::endl;
        return 0;
    } else {
        if (((-vx>0&&x==n)||(-vx<0&&x==0))&&(y!=m||vy!=0)) {std::cout << -1 << std::endl; return 0;}
        if ((-vy<0&&y==0)||(-vy>0&&y==m)) {std::cout << mp[x/(-vx)].first << ' ' << mp[x/(-vx)].second << std::endl; return 0;}
        int t=(m-y)/(-vy);
        if (t*(n-x)>=std::max(y*(n-x),x*(m-y))) {std::cout << mp[x/(-vx)].first << ' ' << mp[x/(-vx)].second << std::endl; return 0;}
        std::cout << mp[m/(-vy)].first << ' ' << mp[m/(-vy)].second << std::endl;
        return 0;
    }
    return 0;
}
