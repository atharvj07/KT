def fact(n):
    if n == 1:
        return 1
    else:
        return n * fact(n - 1)

n = int(input())
a = list(map(int, input().split()))
p = int(input())

a.sort()

cnt = 0
for i in range(n):
    if p < a[i]:
        break
    cnt += 1
    p -= a[i]

ans = cnt * fact(n) / fact(n - cnt)

print(ans)
