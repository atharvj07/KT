import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int p = sc.nextInt();
        long pw = 2;
        int ans = 0;
        while (pw <= n) {
            if (pw >= p) ans++;
            pw *= 2;
            ans++;
        }
        if (pw <= n + p) ans++;
        if (p < 0 && n % -p == 0) ans--;
        if (p == 0 && n % 2 == 1) ans = -1;
        System.out.println(ans == 0 ? -1 : ans);
    }
}
