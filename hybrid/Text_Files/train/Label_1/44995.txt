import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), h = input.nextInt();
        int width = 0;
        for (int i = 0; i < n; i++) {
            width += (input.nextInt() > h ? 2 : 1);
        }
        System.out.println(width);
    }
}
