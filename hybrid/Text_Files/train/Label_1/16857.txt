#include<bits/stdc++.h>
using namespace std;
int main() {
    long long x, n, ans = 1, mod = 1000000007, ans2, pow2;
    bool check;
    cin >> x >> n;
    for (long long i = 2; i <= n; i++) {
        ans2 = 1, check = true, pow2 = 1;
        for (long long j = 2; j * j <= x; j++) {
            if (x % j == 0) {
                while (x % j == 0) {
                    x /= j;
                    pow2 *= j;
                }
                while (pow2 % i == 0) {
                    ans2 *= i;
                    pow2 /= i;
                    check = false;
                }
            }
        }
        if (x > 1) {
            if (x % i == 0) {
                while (x % i == 0) {
                    x /= i;
                    pow2 *= i;
                }
                while (pow2 % i == 0) {
                    ans2 *= i;
                    pow2 /= i;
                    check = false;
                }
            }
        }
        if (check) ans2 *= i;
        ans = (ans * ans2) % mod;
    }
    cout << ans << endl;
}
