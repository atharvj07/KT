import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] arg) throws Exception {
    Scanner sc = new Scanner(System.in);
    while (sc.hasNext()) {
      int n = sc.nextInt();
      if (n == 0) break;
      int[] a = new int[n];
      int tot = 0;
      for (int i = 0; i < n; i++) {
        a[i] = sc.nextInt();
        tot += a[i];
      }
      if (tot == n) {
        System.out.println(1);
        continue;
      }
      if (tot != 2 * n) {
        System.out.println("NA");
        continue;
      }
      Arrays.sort(a);
      int sum = a[n - 1], cnt = 0;
      for (int i = n - 2; i >= 0; i--) {
        sum += a[i];
        cnt++;
        if (sum >= n) {
          System.out.println(cnt);
          break;
        }
      }
    }
    sc.close();
  }
}
