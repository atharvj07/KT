#include<bits/stdc++.h>
using namespace std;
const int N=5e4+5;
int a[N],ans[N];
int main(){
    int t; cin>>t;
    while(t--){
        int n; cin>>n;
        for(int i=1;i<=n;++i) cin>>a[i];
        int l=1,r=n;
        bool fl=false;
        while(l+1<r){
            int mid=(l+r)>>1;
            if(a[mid]+a[mid+1]<=a[n]){
                l=mid;
            }
            else r=mid-1;
        }
        if(r==1){
            cout<<-1<<endl;
            continue;
        }
        cout<<l<<' '<<l+1<<' '<<n<<endl;
    }
    return 0;
}
