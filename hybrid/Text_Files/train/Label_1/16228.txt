#include <iostream>
#include <vector>
#include <queue>
#include <cstring>

using namespace std;

const int offset[4][2] = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}};

int main() {
  int t;
  cin >> t;
  while (t--) {
    vector<vector<int>> grid(1001, vector<int>(1001, 0));
    int Ax, Ay, Bx, By, Fx, Fy;
    cin >> Ax >> Ay >> Bx >> By >> Fx >> Fy;
    grid[Fx][Fy] = -1;
    queue<pair<int, int>> q;
    q.push({Ax, Ay});
    grid[Ax][Ay] = 1;
    while (!q.empty()) {
      pair<int, int> cur = q.front();
      q.pop();
      if (cur.first == Bx && cur.second == By) break;
      for (int i = 0; i < 4; i++) {
        int nx = cur.first + offset[i][0];
        int ny = cur.second + offset[i][1];
        if (nx >= 1 && nx <= 1000 && ny >= 1 && ny <= 1000 && grid[nx][ny] != -1 &&
            grid[nx][ny] == 0) {
          grid[nx][ny] = grid[cur.first][cur.second] + 1;
          q.push({nx, ny});
        }
      }
    }
    if (grid[Bx][By] != 0)
      cout << grid[Bx][By] - 1 << "\n";
    else
      cout << -1 << "\n";
  }
}
