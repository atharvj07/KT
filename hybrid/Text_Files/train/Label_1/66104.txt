import sys
input = sys.stdin.readline

n, k = map(int, input().split())
cards = [input().strip() for _ in range(n)]

def get_sets(cards, k):
    cnt = 0
    for i in range(n):
        for j in range(i+1, n):
            for p in range(j+1, n):
                isSame = True
                isDiff = True
                for l in range(k):
                    if cards[i][l] == cards[j][l] and cards[i][l] == cards[p][l]:
                        continue
                    if cards[i][l] == cards[j][l] or cards[i][l] == cards[p][l] or cards[j][l] == cards[p][l]:
                        isDiff = False
                    else:
                        isSame = False
                if isSame or isDiff:
                    cnt += 1
    return cnt

print(get_sets(cards, k))
