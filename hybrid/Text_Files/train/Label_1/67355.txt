n = int(input())
a = list(map(int, input().split()))

L = [0] * n
R = [0] * n
L[0] = 1
R[n-1] = 1

for i in range(1, n):
    if a[i] > a[i-1]:
        L[i] = L[i-1] + 1

for i in range(n-2, -1, -1):
    if a[i] > a[i+1]:
        R[i] = R[i+1] + 1

for i in range(1, n-1):
    if L[i] == i and R[i] == n-i-1:
        print("YES")
        print(" ".join(map(str, L)))
        exit()

print("NO")
