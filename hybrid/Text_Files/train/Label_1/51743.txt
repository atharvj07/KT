import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long m = sc.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    long[] t = new long[n];
    long p = 0;
    for (int i = 0; i < n; i++) {
      if (p + a[i] <= m) {
        t[i] = p;
        p += a[i];
      } else {
        t[i] = p;
        p = a[i];
      }
    }
    for (long x : t) {
      System.out.println(x);
    }
  }
}
