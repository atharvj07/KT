#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int n;
int main() {
    scanf("%d", &n);
    vector<pair<int, int> > v(n);
    for (int i = 0; i < n; ++i) {
        int a, c;
        scanf("%d %d", &a, &c);
        v[i].first = a;
        v[i].second = c;
    }
    stable_sort(v.begin(), v.end(), [](const pair<int, int> &a, const pair<int, int> &b) -> bool { return a.first > b.second; });
    for (int i = 0; i < n; ++i)
        v[i].first += v[i].second;
    nth_element(v.begin(), v.begin() + n / 2, v.end(), [](const pair<int, int> &a, const pair<int, int> &b) -> bool { return a.first > b.first; });
    sort(v.rbegin(), v.rend());
    int sum1 = 0, sum2 = 0;
    for (int i = 0; i < n; ++i) {
        if (abs(sum1 - sum2) > 500) {
            printf("-1");
            return 0;
        }
        if (sum1 <= sum2) {
            sum1 += v[i].first;
            printf("G");
        } else {
            sum2 += v[i].first;
            printf("A");
        }
    }
    return 0;
}
