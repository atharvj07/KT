#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod=1e9+7;
ll mul(ll x,ll y){
    return (1ll*x*y)%mod;
}
const ll N = 5005;
const int lg=18;
int a[N];
int sub[N][lg+1];
ll pre[N];
int32_t main(){
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n; cin >> n;
    for(int i=1;i<=n;++i) cin >> a[i];

    for(int i=1;i<=n;++i){
        pre[i]=pre[i-1];
        if(a[i]>0) pre[i]=pre[i]+1;
        else pre[i]=pre[i]-1;
    }

    for(int i=1;i<=n;++i){
        sub[i][0]=a[i];
    }
    for(int lg_val=1;lg_val<=lg;++lg_val){
        for(int i=1;i+(1<<lg_val)-1<=n;++i){
            int low=i,high=i+(1<<lg_val)-1;
            if(sub[low][lg_val-1]==sub[high][lg_val-1]) sub[i][lg_val]=sub[low][lg_val-1];
            else if(!sub[low][lg_val-1] || !sub[high][lg_val-1]) sub[i][lg_val]=1;
            else sub[i][lg_val]=-1;
        }
    }

    for(int l=1;l<=n;++l){
        for(int r=l;r<=n;++r){
            ll sm=pre[r]-pre[l-1];
            if(sm){
                if(sm>0) cout << 0 << ' ';
                else cout << 1 << ' ';
                continue;
            }
            int val=0;
            bool flag=1;
            for(int lg_val=lg;lg_val>=0;--lg_val){
                if((1<<lg_val)<=r-l+1){
                    if(sub[l][lg_val]==0) continue;
                    if(sub[l][lg_val]==-1) flag=0;
                    else val=sub[l][lg_val];
                    l+=(1<<lg_val);
                }
            }
            if(!flag) cout << 0 << ' ';
            else cout << (val>0) << ' ';
        }
        cout << '\n';
    }
    return 0;
}
