n = int(input())
a = list(map(int, input().split()))

used = set()
result = []
free = []

for i in range(1, n + 1):
    if i not in a:
        free.append(i)

for num in a:
    if num not in used and 1 <= num <= n:
        result.append(num)
        used.add(num)
    else:
        result.append(free.pop(0))
        used.add(result[-1])

print(" ".join(map(str, result)))
