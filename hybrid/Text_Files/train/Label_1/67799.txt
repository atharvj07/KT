import sys

n, m, d = map(int, sys.stdin.readline().split())
if d > n - 1:
    print("NO")
else:
    print("YES")
    g = [[]]
    for i in range(n): g.append([])
    for i in range(m):
        v, u = map(int, sys.stdin.readline().split())
        g[v].append(u)
        g[u].append(v)
    vis = [False] * (n + 1)
    used = []
    def dfs(v):
        if vis[v]: return
        vis[v] = True
        for u in g[v]:
            if not vis[u]:
                used.append((v, u))
                dfs(u)
    dfs(1)
    cur = len(used)
    while cur < n - 1:
        v, u = used[-1]
        g[u].remove(v)
        g[v].remove(u)
        del used[-1]
        for u in g[v]:
            if not vis[u]:
                used.append((v, u))
                dfs(u)
                break
        cur = len(used)
    for v, u in used:
        print(v, u)
