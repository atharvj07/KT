#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> adj(n);
    for (int i = 0; i < m; i++) {
      int u, v;
      cin >> u >> v;
      u--;
      v--;
      adj[u].push_back(v);
      adj[v].push_back(u);
    }
    vector<int> in(n), out(n);
    for (int i = 0; i < n; i++) {
      for (int u : adj[i]) {
        in[u]++;
        out[i]++;
      }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (in[i] == out[i]) {
        ans++;
      }
    }
    cout << ans << '\n';
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < adj[i].size(); j++) {
        if (in[i] == out[i]) {
          cout << i + 1 << ' ' << adj[i][j] + 1 << '\n';
          in[i]--;
          out[adj[i][j]]--;
          adj[i].erase(adj[i].begin() + j);
          j--;
        }
      }
    }
  }
  return 0;
}
