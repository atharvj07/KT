#include <bits/stdc++.h>
using namespace std;
#define N 200006
int n,k;
struct Book{int t,a,b;};
Book v[N];
int main() {
    cin >> n >> k;
    for(int i=1; i<=n; i++) {
        cin >> v[i].t >> v[i].a >> v[i].b;
    }
    sort(v+1, v+n+1, [](const Book &x, const Book &y) -> bool {
        return x.t < y.t;
    });
    int l1=0, l2=0, r1=0, r2=0, ans=0, ansa=0, ansb=0;
    for(int i=1; i<=n; i++) {
        l1+=v[i].a; r1+=v[i].a;
        l2+=v[i].b; r2+=v[i].b;
        if(l1>=k && l2>=k) {
            if(ans==0 || ans>v[i].t) {
                ans=v[i].t; ansa=l1; ansb=l2;
            }
        }
        if(r1<k || r2<k) {
            l1=r1=0; l2=r2=0;
        }
        if(l1>=k && r2>=k) {
            if(ans==0 || ans>v[i].t) {
                ans=v[i].t; ansa=l1; ansb=l2;
            }
        }
        if(r1<k || r2<k) {
            l1=r1=0; l2=r2=0;
        }
    }
    if(ans) {
        for(int i=1; i<=n; i++) {
            if(v[i].a && ansa>0) {
                ans+=v[i].t; ansa--;
            }
            if(v[i].b && ansb>0) {
                ans+=v[i].t; ansb--;
            }
            if(!ansa && !ansb) break;
        }
        cout << ans << endl;
    }
    else cout << "-1" << endl;
}
