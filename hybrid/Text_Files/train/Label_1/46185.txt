import java.util.*;

public class Main {
    static int A, B, C;
    static double eps = 1e-9;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        A = sc.nextInt();
        B = sc.nextInt();
        C = sc.nextInt();
        if (Math.abs(A) < eps) {
            if (Math.abs(B) < eps) {
                if (Math.abs(C) < eps)
                    System.out.println(-1);
                else
                    System.out.println(0);
            } else {
                System.out.println(1);
                System.out.println(String.format("%.9f", -C / (double) B));
            }
        } else {
            double D = (double) B * B - 4 * A * C;
            if (D < -eps)
                System.out.println(0);
            else {
                int count = 0;
                double sqrtD = Math.sqrt(D);
                double r1 = (-B - sqrtD) / (2 * A);
                double r2 = (-B + sqrtD) / (2 * A);
                if (Math.abs(r1 - r2) > eps)
                    count = 2;
                else
                    count = 1;
                System.out.println(count);
                if (r1 > r2) {
                    double t = r1;
                    r1 = r2;
                    r2 = t;
                }
                System.out.println(String.format("%.9f", r1));
                if (count == 2)
                    System.out.println(String.format("%.9f", r2));
            }
        }
    }
}
