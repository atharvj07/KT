#include <bits/stdc++.h>
using namespace std;

string solve(string s) {
    for (int i = 0; i < (int)s.size(); ++i) {
        if (s[i] != '?' && (i > 0 && s[i] == s[i - 1])) {
            return "-1";
        }
        if (s[i] == '?') {
            if (i > 0) {
                if (i > 1 && s[i - 1] == s[i - 2]) {
                    s[i] = 6 - s[i - 2] - s[i - 1];
                } else {
                    s[i] = 6 - s[i - 1];
                }
            } else {
                s[i] = 'a';
            }
        }
    }
    return s;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        cout << solve(s) << endl;
    }
    return 0;
}
