t = int(input())
for _ in range(t):
    n = int(input())
    l = list(map(int, input().split()))
    d = {}
    for i, x in enumerate(l):
        if x in d:
            d[x].append(i + 1)
        else:
            d[x] = [i + 1]
    unique = [x for x in d if len(d[x]) == 1]
    if unique:
        print(min(unique))
    else:
        print(-1)
