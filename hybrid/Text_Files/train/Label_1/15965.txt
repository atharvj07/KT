#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int N = 200;
const int INF = 1e9;

vector<int> g[N];
int a[N], dp[N][2];

void dfs(int v, int pr) {
    dp[v][0] = a[v];
    dp[v][1] = 0;
    for (auto u : g[v]) {
        if (u == pr) continue;
        dfs(u, v);
        dp[v][0] += dp[u][1];
        dp[v][1] += max(dp[u][0], dp[u][1]);
    }
}

int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 1);
    cout << max(dp[1][0], dp[1][1]) << "\n";
    return 0;
}
