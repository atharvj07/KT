import sys

s = sys.stdin.readline().strip()
n = len(s)
dp = [[0] * (n + 1) for _ in range(n + 1)]
for i in range(n):
    dp[i][i] = 1
    if i + 1 < n and s[i] == s[i + 1]:
        dp[i][i + 1] = 1
for k in range(3, n + 1):
    for i in range(n - k + 1):
        j = i + k - 1
        if s[i] == s[j] and dp[i + 1][j - 1] == 1:
            dp[i][j] = 1
ans = [0] * n
for i in range(n):
    for j in range(i, n):
        if dp[i][j] == 1:
            ans[j - i] += 1
print(*ans)
