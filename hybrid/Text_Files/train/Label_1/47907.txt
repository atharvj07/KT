import java.util.Scanner;

public class Main {

    private static final int MAX_N = 22;
    private static int[][] map;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int N = sc.nextInt();
            int M = sc.nextInt();
            if (N == 0 && M == 0) {
                break;
            }
            map = new int[MAX_N][MAX_N];
            for (int i = 0; i < N; i++) {
                for (int j = 0; j < M; j++) {
                    map[i][j] = sc.nextInt();
                }
            }
            int maxCapacity = 0;
            for (int i = 1; i < N - 1; i++) {
                for (int j = 1; j < M - 1; j++) {
                    int minOuter = Integer.MAX_VALUE;
                    for (int k = -1; k <= 1; k++) {
                        for (int l = -1; l <= 1; l++) {
                            if (k == 0 && l == 0) {
                                continue;
                            }
                            minOuter = Math.min(minOuter, map[i + k][j + l]);
                        }
                    }
                    maxCapacity = Math.max(maxCapacity, minOuter - map[i][j]);
                }
            }
            System.out.println(maxCapacity);
        }
        sc.close();
    }
}
