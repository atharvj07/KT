n = int(input())
a = []
for i in range(n):
    a.append(list(map(int, input().split())))
u, d = 0, 0
for i in range(n):
    u += a[i][0]
    d += a[i][1]
if u % 2 == d % 2:
    print(0)
elif (u % 2 == 0 and d % 2 != 0) and u > d or (d % 2 == 0 and u % 2 != 0) and d > u:
    print(1)
else:
    print(-1)
