#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, t, ans = 0;
    vector<int> v;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> t;
        v.push_back(t);
    }
    cin >> t;
    sort(v.begin(), v.end());
    for (int i = 0; i < n; i++) {
        if (i + 1 < n && v[i + 1] <= v[i] + t) continue;
        ans++;
    }
    cout << ans << endl;
    return 0;
}
