#include <iostream>
#include <vector>

using namespace std;

#define MOD 1000000007

long long power(long long x, long long y) {  
    long long ret = 1;  
    while (y > 0) {  
        if (y % 2 == 0) {
            x = x * x % MOD;  
            y /= 2;  
        } else {  
            ret = ret * x % MOD;  
            y--;  
        }  
    }  
    return ret % MOD;  
}

long long f(long long x, long long y) {
    vector<long long> primes;
    for (long long i = 2; i * i <= x; i++) {
        if (x % i == 0) {
            primes.push_back(i);
            while (x % i == 0) {
                x /= i;
            }
        }
    }
    if (x > 1) {
        primes.push_back(x);
    }
    long long ans = 1;
    for (auto p : primes) {
        long long res = 0;
        while (y % p == 0) {
            res++;
            y /= p;
        }
        ans = ans * power(p, res) % MOD;
    }
    return ans;
}

int main() {
    long long x, n;
    cin >> x >> n;
    long long ans = 1;
    for (long long i = 1; i <= n; i++) {
        ans = ans * f(x, i) % MOD;
    }
    cout << ans << endl;
    return 0;
}
