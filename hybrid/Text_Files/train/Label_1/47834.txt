import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] min = new long[n];
        long[] max = new long[n];
        for (int i = 0; i < n; i++) {
            String s = scanner.next();
            long x = scanner.nextLong();
            char ans = scanner.next().charAt(0);
            if (s.charAt(0) == '>') {
                if (ans == 'Y') {
                    min[i] = x + 1;
                } else {
                    max[i] = x;
                }
            } else if (s.charAt(0) == '<') {
                if (ans == 'Y') {
                    max[i] = x - 1;
                } else {
                    min[i] = x;
                }
            } else if (s.charAt(0) == '=') {
                if (ans == 'Y') {
                    min[i] = x;
                    max[i] = x;
                }
            }
        }
        long start = Long.MAX_VALUE;
        long end = Long.MIN_VALUE;
        boolean possible = true;
        for (int i = 0; i < n; i++) {
            start = Math.min(start, min[i]);
            end = Math.max(end, max[i]);
            if (min[i] > max[i]) {
                possible = false;
            }
        }
        if (possible) {
            System.out.println(start);
        } else {
            System.out.println("Impossible");
        }
    }
}
