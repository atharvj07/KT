n = int(input())
a = list(map(int, input().split()))

total_odd = sum(a[0::2])
total_even = sum(a[1::2])

current_odd, current_even = 0, 0
good_candies = 0

for i in range(n):
    if i % 2 == 0:
        total_odd -= a[i]
    else:
        total_even -= a[i]
    
    if current_odd + total_even == current_even + total_odd:
        good_candies += 1
    
    if i % 2 == 0:
        current_odd += a[i]
    else:
        current_even += a[i]

print(good_candies)
