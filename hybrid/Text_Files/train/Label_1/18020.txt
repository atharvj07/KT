#include <iostream>
#include <string>

int main() {
    std::string s;
    std::cin >> s;
    bool a = false, b = false, c = false;
    for (char ch : s) {
        if (ch == 'a')
            a = true;
        else if (ch == 'b')
            b = true;
        else if (ch == 'c')
            c = true;
    }
    std::cout << ((a && b && c) ? "YES" : "NO") << std::endl;
    return 0;
}
