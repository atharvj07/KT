MOD = 1000000007

def solve(n, k):
    # For Moamen to win, all bits in positions 0 to k-1 must be the same in all numbers
    # since XOR of any bit position with different values will be 1, 
    # and AND of any bit position with different values will be 0.
    # Therefore, for any bit position, if one number has 0 and another has 1, Moamen can't win.
    # Moamen can only win if all numbers are the same or if all numbers have 0 in a particular bit position.
    # So, we can choose any number from 0 to 2^k-1 for each position, 
    # but once a number has a 1 in any bit position, all numbers must have 1 in that position.
    # Thus, there are 2^k possibilities for each of the n numbers, 
    # but we must subtract the cases where all numbers are different except for 0.
    # As all numbers can be chosen independently, the answer is (2^k)^n.
    # However, since we are interested in Moamen winning, we don't actually need to subtract any cases.
    # So, the formula simplifies to (2^k)^n modulo 1000000007.
    return pow(2, n*k, MOD)

t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().strip().split())
    print(solve(n, k))
