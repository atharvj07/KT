#include <cstdio>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

int n;
vector<string> v;
int p[400];

int get(int a) {
    if (p[a] == a) {
        return a;
    }
    return p[a] = get(p[a]);
}

void uni(int a, int b) {
    p[get(a)] = get(b);
}

int main() {
    scanf("%d", &n);
    v.resize(n);
    for (int i = 0; i < n; i++) {
        char a[25];
        scanf("%s", a);
        string b = a;
        for (int j = 0; j < b.size(); j++) {
            if (b[j] == 'u') {
                b[j] = 'o';
            }
        }
        for (int j = 0; j < b.size(); j++) {
            if (b[j] == 'h' || b[j] == 'k') {
                b[j] = 'x';
            }
        }
        v[i] = b;
    }
    for (int i = 0; i < n; i++) {
        p[i] = i;
    }
    for (int i = 0; i < n - 1; i++) {
        for (int j = i + 1; j < n; j++) {
            if (v[i] == v[j]) {
                uni(i, j);
            }
        }
    }
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (get(i) == i) {
            cnt++;
        }
    }
    printf("%d", cnt);
    return 0;
}
