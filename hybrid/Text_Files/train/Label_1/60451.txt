n, a, b = map(int, input().split())
creatures = []
for i in range(n): creatures.append(list(map(int, input().split())))
creatures.sort(key=lambda x: x[0])
damage = 0
for i in range(n):
    if a > 0 and creatures[i][0] < creatures[i][1]:
        creatures[i][0] *= 2
        a -= 1
    damage += max(creatures[i][0], creatures[i][1])
    if b > 0 and creatures[i][0] > creatures[i][1]:
        creatures[i][1] = creatures[i][0]
        b -= 1
print(damage)
