#include<bits/stdc++.h>
using namespace std;
int main() {
    int n; cin >> n;
    vector<int> b(n - 1), c(n - 1);
    for (int i = 0; i < n - 1; i++) cin >> b[i];
    for (int i = 0; i < n - 1; i++) cin >> c[i];
    for (int i = 0; i < n - 1; i++) {
        if (b[i] > c[i]) swap(b[i], c[i]);
    }
    auto bb = b, cc = c;
    sort(bb.begin(), bb.end()); sort(cc.begin(), cc.end());
    for (int i = 0; i < n - 1; i++) {
        if (bb[i] != b[i] || cc[i] != c[i]) {
            cout << "-1" << endl; return 0;
        }
    }
    for (int j = 0; j < n - 1; j++) {
        b[j] = cc[j], c[j] = bb[j];
    }
    set<int> A;
    for (int i = 0; i < n - 1; i++) {
        A.insert(b[i]), A.insert(c[i]);
    }
    vector<int> a;
    for (auto it : A) a.push_back(it);
    a.push_back(!a.empty() ? *a.rbegin() + 1 : 1);
    for (int i = 0; i < n - 1; i++) {
        if (a[i] > a[i + 1]) swap(a[i], a[i + 1]);
        if (b[i] == a[i] && c[i] == a[i + 1]) continue;
        if (b[i] == a[i + 1] && c[i] == a[i]) continue;
        cout << "-1" << endl; return 0;
    }
    for (auto &x : a) cout << x << " ";
    cout << endl;
    return 0;
}
