#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = m - k;
  sort(a.begin(), a.end(), greater<int>());
  for (int l : a) {
    if (ans <= 0) break;
    ans -= l - 1;
  }
  if (ans <= 0)
    cout << n << endl;
  else
    cout << -1 << endl;
}
