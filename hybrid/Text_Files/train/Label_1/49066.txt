import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int h = sc.nextInt();
            if (n == 0 && h == 0) {
                break;
            }
            boolean[][][] cube = new boolean[n][n][n];
            for (int i = 0; i < h; i++) {
                String plane = sc.next();
                int a = sc.nextInt() - 1;
                int b = sc.nextInt() - 1;
                if (plane.equals("xy")) {
                    for (int k = 0; k < n; k++) {
                        cube[a][b][k] = true;
                    }
                } else if (plane.equals("xz")) {
                    for (int j = 0; j < n; j++) {
                        cube[a][j][b] = true;
                    }
                } else if (plane.equals("yz")) {
                    for (int k = 0; k < n; k++) {
                        cube[k][a][b] = true;
                    }
                }
            }
            int count = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    for (int k = 0; k < n; k++) {
                        if (!cube[i][j][k]) {
                            count++;
                        }
                    }
                }
            }
            System.out.println(count);
        }
    }
}
