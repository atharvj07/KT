#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 1e5 + 5, mod = 1e9 + 7;
int main() {
    int n;
    ll ans = 0;
    cin >> n;
    for (int i = 0; i < n; i++) {
        ll x, y;
        cin >> x >> y;
        if (abs(x) + abs(y) > 2e6) {
            if (abs(x) == abs(y)) {
                ans += y, x = 0;
            } else if (abs(x) > abs(y)) {
                ans += y, x -= y;
            } else {
                ans -= x, y -= x;
            }
        }
        if (x >= 0 && y >= 0) ans += y, x -= y;
        else if (x >= 0 && y < 0) ans -= x, y += x;
        else if (x < 0 && y < 0) ans -= y, x += y;
        else if (x < 0 && y >= 0) ans += x, y -= x;
        cout << (x > 0 || y > 0) << ' ';
    }
    return 0;
}
