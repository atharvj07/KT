n = int(input())
A = [int(i)-1 for i in input().split()]
par = [-1] * n
L = [[]] * n
R = [[]] * n
cur = 0
for i in range(n):
    L[cur].append(i)
    while par[i] != -1:
        i = par[i]
        L[cur].append(i)
    j = i
    while R[cur] and R[cur][-1] > j:
        R[cur].pop()
    R[cur].append(j)
    cur+=1
ans = 0
del L[cur:]
del R[cur:]
for i in range(len(L)):
    l = L[i][0]
    r = R[i][-1]
    lenL = len(L[i])
    lenR = len(R[i])
    ans += lenL * lenR
    if i < len(L)-1 and r == R[i+1][-1]:
        ans -= lenL * (len(R[i+1])-lenR)
    if i > 0 and l == L[i-1][0]:
        ans -= lenR * (len(L[i-1])-lenL)
    par[A[l]] = A[r]
print(ans)
