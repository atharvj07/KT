#include <iostream>
#include <vector>
using namespace std;

struct Square {
  vector<pair<int, int>> points;  // clockwise or anti-clockwise
  int minX, minY, maxX, maxY;
  int C1, C2, D1, D2;           // ax + by = c1, ax - by = c2
  bool inSquare(int x, int y) {
    bool x_in = minX <= x && x <= maxX;
    bool y_in = minY <= y && y <= maxY;
    bool ab_in = (C1 - x) * (D2 - y) == (C2 - x) * (D1 - y);
    return x_in && y_in && ab_in;
  }
  void init() {
    C1 = points[0].first + points[1].first;
    C2 = points[0].first - points[1].first;
    D1 = points[0].second + points[1].second;
    D2 = points[0].second - points[1].second;
    minX = points[0].first, minY = points[0].second;
    maxX = points[0].first, maxY = points[0].second;
    for (int i = 1; i < 4; i++) {
      minX = min(minX, points[i].first);
      minY = min(minY, points[i].second);
      maxX = max(maxX, points[i].first);
      maxY = max(maxY, points[i].second);
    }
  }
};

int main() {
  Square sq[2];
  for (int i = 0; i < 2; i++) {
    for (int j = 0; j < 4; j++) {
      int x, y;
      cin >> x >> y;
      sq[i].points.emplace_back(x, y);
    }
    sq[i].init();
  }

  bool flag = false;
  for (int i = 0; i < 4; i++) {
    if (sq[0].inSquare(sq[1].points[i].first, sq[1].points[i].second)) {
      flag = true;
      break;
    }
  }
  if (!flag) {
    for (int i = 0; i < 4; i++) {
      if (sq[1].inSquare(sq[0].points[i].first, sq[0].points[i].second)) {
        flag = true;
        break;
      }
    }
  }

  if (flag)
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
