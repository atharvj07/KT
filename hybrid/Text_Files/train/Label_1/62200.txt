import sys
input = sys.stdin.readline

n,r=map(int,input().split())
x=[0]+list(map(float,input().split()))
y=[1e100]*n
for i in range(n):
    for j in range(i):
        if x[i]==x[j]:
            continue
        if (x[i]-x[j])**2+(y[j]-y[i])**2<=4*r**2:
            y[i]=max(y[i],(4*r**2-(x[i]-x[j])**2)**0.5+y[j])

    y[i]=max(0,(y[i]-(y[i]**2-4*r**2)**0.5)/2)

print(*y[1:],sep=' ')
