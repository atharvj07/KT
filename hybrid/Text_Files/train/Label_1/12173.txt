#include <bits/stdc++.h>
using namespace std;
const int MXN = 2e5 + 5;
int n, a[MXN], ans;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    sort(a + 1, a + 1 + n);
    ans = a[n];
    for (int i = n - 1; i >= 1; i--) {
        if (a[n] % a[i] == 0)
            ans += a[i];
    }
    cout << ans << endl;
    return 0;
}
