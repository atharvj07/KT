#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> freq(2 * n + 1);
    int ans = 0;
    for (int i = 0; i < n; i++) {
      ans += freq[n - a[i] + 1];
      freq[a[i]]++;
    }
    cout << ans << endl;
  }
}
