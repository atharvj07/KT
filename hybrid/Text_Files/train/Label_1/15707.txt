#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k;
    cin >> n >> k;
    string s = to_string(n);
    int cnt[10] = {};

    int pre = -1, j = 0;
    for (int i = 0; i < s.length(); i++) {
      if (cnt[s[i] - '0']++ == 0) j++;

      if (j > k) {
        while (j > k) {
          cnt[s[pre] - '0']--;
          pre++;
          j--;
        }
        s = s.substr(pre, i - pre);
        i = pre;
      }
    }
    while (j < k) {
      if (cnt[9 - j] == 0) {
        s.push_back('0' + 9 - j);
        cnt[9 - j]++;
        j++;
      } else
        break;
    }

    long long ans = stoll(s);
    cout << ans << "\n";
  }
  return 0;
}
