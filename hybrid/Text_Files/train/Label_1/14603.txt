#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll mod = 1e9+7;
ll binpow(ll a, ll b){
    ll ans = 1;
    while(b > 0){
        if(b&1) ans = ans*a % mod;
        a = a*a % mod;
        b >>= 1;
    }
    return ans;
}
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    ll k;
    cin >> k;
    ll ans = binpow(2, k-1);
    ll den = binpow(2, k);
    ll g = __gcd(ans, den);
    ans /= g;
    den /= g;
    cout << ans << "/" << den << endl;
}
