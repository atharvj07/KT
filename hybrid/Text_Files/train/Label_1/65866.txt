def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def find_prime_graph(n):
    # For n=3, the graph cannot be formed as per the conditions.
    if n < 3:
        return -1
    
    # Find the smallest prime number greater than or equal to n
    m = n
    while not is_prime(m):
        m += 1
    
    edges = []
    # Connect all vertices in a cycle
    for i in range(1, n):
        edges.append((i, i+1))
    edges.append((n, 1))  # Complete the cycle
    
    # Add extra edges from vertex 1 to others until the total number of edges is prime
    extra_edges = m - n
    for i in range(2, extra_edges + 2):
        edges.append((1, i))
    
    return edges

n = int(input().strip())
result = find_prime_graph(n)
if result == -1:
    print(-1)
else:
    print(len(result))
    for edge in result:
        print(edge[0], edge[1])
