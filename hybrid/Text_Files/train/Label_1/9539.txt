#include<bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string a;
    cin >> a;
    stack<char> st;
    map<char, bool> vis;
    int n = a.size();
    for (int i = n - 1; i >= 0; i--) {
      if (!vis[a[i]]) {
        while (!st.empty() && st.top() < a[i]) {
          char c = st.top();
          st.pop();
          vis[c] = false;
        }
        st.push(a[i]);
        vis[a[i]] = true;
      }
    }
    string b = "";
    while (!st.empty()) {
      b.push_back(st.top());
      st.pop();
    }
    reverse(b.begin(), b.end());
    cout << b << "\n";
  }
  return 0;
}
