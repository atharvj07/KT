#include <bits/stdc++.h>

#define ll long long

ll gcd(ll a, ll b) {
  return !b ? a : gcd(b, a % b);
}

ll solve(ll a, ll b, ll x) {
  ll gg = gcd(a, b);
  if(x % gg != 0) {
    return 0;
  }

  return 1;
}

int main() {
  std::ios::sync_with_stdio(false);
  std::cin.tie(nullptr);

  int t;
  
  std::cin >> t;
  
  while(t--) {
    ll a, b, x;
    std::cin >> a >> b >> x;
    std::cout << (solve(a, b, x) ? "YES" : "NO")  << std::endl;
  }

  return 0;
}
