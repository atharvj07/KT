#include <iostream>
#include <map>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    char cmd[n + 1];
    cin >> cmd;
    int ans = 0;
    map<vector<int>, int> mp;
    vector<int> pos = {0, 0};
    mp[pos] = 1;
    for (int i = 0; i < n; i++) {
        if (cmd[i] == 'U')
            pos[0]--;
        else if (cmd[i] == 'L')
            pos[1]--;
        else if (cmd[i] == 'R')
            pos[1]++;
        else
            pos[0]++;
        if (mp[pos] > 0)
            ans += mp[pos];
        mp[pos]++;
    }
    cout << ans << endl;
    return 0;
}
