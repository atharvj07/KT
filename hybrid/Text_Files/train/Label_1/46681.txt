import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    static class Query {
        int type;
        int x;
        int y;

        public Query(int type, int x, int y) {
            this.type = type;
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int q = Integer.parseInt(line[1]);
        SegmentTree st = new SegmentTree(n);
        List<Query> queries = new ArrayList<>();
        for (int i = 0; i < q; i++) {
            line = br.readLine().split(" ");
            int type = Integer.parseInt(line[0]);
            int x = Integer.parseInt(line[1]);
            int y = Integer.parseInt(line[2]);
            queries.add(new Query(type, x, y));
        }
        for (Query query : queries) {
            if (query.type == 0) {
                st.update(query.x, query.y);
            } else {
                System.out.println(st.query(query.x, query.y));
            }
        }
    }

    static class SegmentTree {
        int[] tree;
        int n;

        public SegmentTree(int n) {
            this.n = n;
            tree = new int[4 * n];
            for (int i = 0; i < 4 * n; i++) {
                tree[i] = Integer.MAX_VALUE;
            }
        }

        public void update(int index, int value) {
            update(index, value, 1, 0, n - 1);
        }

        private void update(int index, int value, int node, int start, int end) {
            if (index < start || index > end) {
                return;
            }
            if (start == end) {
                tree[node] = value;
            } else {
                int mid = (start + end) / 2;
                update(index, value, 2 * node, start, mid);
                update(index, value, 2 * node + 1, mid + 1, end);
                tree[node] = Math.min(tree[2 * node], tree[2 * node + 1]);
            }
        }

        public int query(int l, int r) {
            return query(l, r, 1, 0, n - 1);
        }

        private int query(int l, int r, int node, int start, int end) {
            if (l > end || r < start) {
                return Integer.MAX_VALUE;
            }
            if (l <= start && r >= end) {
                return tree[node];
            }
            int mid = (start + end) / 2;
            return Math.min(query(l, r, 2 * node, start, mid), query(l, r, 2 * node + 1, mid + 1, end));
        }
    }
}
