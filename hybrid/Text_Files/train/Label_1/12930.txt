#include <iostream>
using namespace std;

int main() {
    int a, b, c;
    cin >> a >> b >> c;
    if ((a + b + c) % 2 == 1) {
        cout << "Impossible" << endl;
    } else {
        if (a % 2 == 1) {
            cout << a / 2 << " " << b + c - a / 2 << " " << c - a / 2 << endl;
        } else if (b % 2 == 1) {
            cout << a + c - b / 2 << " " << b / 2 << " " << c - b / 2 << endl;
        } else {
            cout << a + b - c / 2 << " " << b - c / 2 << " " << c / 2 << endl;
        }
    }
    return 0;
}
