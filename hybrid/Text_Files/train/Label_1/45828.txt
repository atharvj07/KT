import java.util.*;
public class Main {
    public static void main(String[] arg) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int res = 1378;
        for (int i = 1; i < n; i++) {
            res *= 1378;
            res %= 10;
        }
        System.out.println(res);
    }
}
