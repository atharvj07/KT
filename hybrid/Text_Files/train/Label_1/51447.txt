import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        char[] str = in.next().toCharArray();
        for (int i = 0; i < m; i++) {
            int l = in.nextInt() - 1;
            int r = in.nextInt() - 1;
            char c1 = in.next().charAt(0);
            char c2 = in.next().charAt(0);
            for (int j = l; j <= r; j++) {
                if (str[j] == c1) {
                    str[j] = c2;
                }
            }
        }
        System.out.println(new String(str));
    }
}
