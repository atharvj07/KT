#include <iostream>

using namespace std;

const int mod = 1e9 + 7;

int main() {
  int n;
  cin >> n;

  int a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  int ans = (1LL << n) - 2;  // exclude empty set and set of all roads
  for (int i = 0; i < n; i++) {
    if (a[i] == i + 1) {
      ans = (1LL * ans * 2) % mod;
    }
  }

  cout << ans << endl;

  return 0;
}
