#include <bits/stdc++.h>
using namespace std;
const int maxn = 505;
char G[maxn][maxn];
struct pos{
    int x, y;
    pos(int a, int b): x(a), y(b) {}
};
int main() {
    int R, C;
    cin >> R >> C;
    for (int x = 0; x < R; x++) {
        for (int y = 0; y < C; y++) {
            cin >> G[x + 1][y + 1];
        }
    }
    bool flag = true;
    for (int x = 1; x <= R; x++) {
        for (int y = 1; y <= C; y++) {
            if (G[x][y] == 'S') {
                if (G[x - 1][y] == 'W') flag = false;
                if (G[x + 1][y] == 'W') flag = false;
                if (G[x][y - 1] == 'W') flag = false;
                if (G[x][y + 1] == 'W') flag = false;
            }
        }
    }
    if (!flag) {
        cout << "No" << endl;
        return 0;
    } else {
        cout << "Yes" << endl;
        for (int x = 1; x <= R; x++) {
            for (int y = 1; y <= C; y++) {
                if (G[x][y] != 'W') {
                    G[x][y] = '.';
                }
            }
        }
        for (int x = 1; x <= R; x++) {
            for (int y = 1; y <= C; y++) {
                if (G[x][y] == 'S') {
                    G[x][y] = 'D';
                } else if (G[x][y] == '.') {
                    if (G[x - 1][y] == 'S') G[x][y] = 'D';
                    if (G[x + 1][y] == 'S') G[x][y] = 'D';
                    if (G[x][y - 1] == 'S') G[x][y] = 'D';
                    if (G[x][y + 1] == 'S') G[x][y] = 'D';
                }
            }
        }
        for (int x = 1; x <= R; x++) {
            for (int y = 1; y <= C; y++) {
                cout << G[x][y];
            }
            cout << endl;
        }
    }
    return 0;
}
