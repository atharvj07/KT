import java.util.*;

class Main {
  static long[] pre;
  static int n, q;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    q = sc.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) a[i] = sc.nextInt();
    long[] k = new long[q];
    for (int i = 0; i < q; i++) k[i] = sc.nextInt();
    long[] ans = new long[q];
    pre = new long[n];
    pre[0] = a[0];
    for (int i = 1; i < n; i++) pre[i] = pre[i - 1] + a[i];
    for (int i = 0; i < q; i++) {
      if (k[i] <= pre[n - 1]) {
        ans[i] = n;
      } else {
        long lo = 0, hi = n - 1;
        while (lo < hi) {
          long mid = (lo + hi) / 2;
          if (k[i] <= pre[(int) mid]) hi = mid;
          else lo = mid + 1;
        }
        ans[i] = n - lo;
      }
    }
    for (long l : ans) System.out.println(l);
  }
}
