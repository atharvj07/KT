#include <bits/stdc++.h>

using namespace std;
typedef long long LL;
const LL MOD = 1e9 + 7;
const int N = 1e5 + 10;
int n;
int a[N];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  sort(a + 1, a + 1 + n);
  int sum = 0, mx = 0;
  for (int i = 1; i <= n; i++) {
    sum += a[i];
    mx = max(a[i], mx);
  }
  vector<int> party;
  if (a[n] * 2 <= sum) {
    party.push_back(1);
    for (int i = 2; i <= n; i++) {
      if (a[n] >= a[i] * 2) { party.push_back(i); }
    }
    cout << party.size() << endl;
    for (auto &x : party) { cout << x << " "; }
    cout << endl;
  } else {
    cout << 0 << endl;
  }
  return 0;
}
