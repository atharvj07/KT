#include <iostream>

using namespace std;

int main() {
    int c[5];
    cin >> c[0] >> c[1] >> c[2] >> c[3] >> c[4];
    int sum = 0, min = 101;
    for (int i = 0; i < 5; i++) {
        sum += c[i];
        if (c[i] > 0) min = min < c[i] ? min : c[i];
    }
    if (sum % 4 != 0 || min % 2 == 0) cout << -1;
    else cout << min;
    return 0;
}
