#include <bits/stdc++.h>
using namespace std;

const int MN = 21;
int a[MN][MN], l[MN], r[MN], v[MN], ans[MN], sum[MN], ord[MN];

bool cantor(const int &i, const int &j) {
  return l[i] == l[j] ? r[i] < r[j] : l[i] < l[j];
}

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) cin >> a[i][j], --a[i][j];
    l[i] = *min_element(a[i], a[i] + m), r[i] = *max_element(a[i], a[i] + m);
  }
  for (int i = 0; i < m; ++i) v[i] = i;
  sort(v, v + m, cantor);
  for (int i = 0; i < m; ++i) ord[v[i]] = i;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) ans[ord[a[i][j]]] += i + 1;
  }
  for (int i = 0; i < m; ++i) ord[i] = 0;
  sort(v, v + m);
  for (int i = 0; i < m; ++i) ord[v[i]] = i;
  for (int i = 0; i < m; ++i) {
    for (int j = 0; j < n; ++j) ord[a[j][i]] += j + 1;
    sort(a[i], a[i] + n);
    for (int j = 0; j < n; ++j) ord[a[j][i]] -= j + 1;
  }
  for (int i = 0; i < m; ++i) ans[i] += ord[i];
  for (int i = 0; i < m; ++i) sum[ans[i]]++;
  for (int i = 0; i <= n + 1; ++i) sum[i] = bool(sum[i]);
  if (accumulate(sum, sum + n + 1, 0) == m)
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
