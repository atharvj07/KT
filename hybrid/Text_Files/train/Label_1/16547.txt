#include <bits/stdc++.h>
using namespace std;
const int MAXN = 210;

int n;
char s[MAXN], t[] = "trygub";

int main() {
    int nt;
    cin >> nt;
    while (nt--) {
        cin >> n >> s;
        bool found = 0;
        for (int i = 0; i < 6 && !found; i++) {
            for (int j = i + 1; j < n; j++) {
                if (s[j] != t[i]) continue;
                for (int k = j + 1; k < n; k++) {
                    if (s[k] != t[i]) continue;
                    for (int l = k + 1; l < n; l++) {
                        if (s[l] != t[i]) continue;
                        for (int m = l + 1; m < n; m++) {
                            if (s[m] != t[i]) continue;
                            for (int x = 0; x < n; x++) {
                                if (x == j || x == k || x == l || x == m) continue;
                                if (s[x] != t[i]) cout << s[x];
                            }
                            cout << s[j] << s[k] << s[l] << s[m] << endl;
                            found = 1;
                            break;
                        }
                    }
                }
            }
        }
        if (!found) {
            for (int i = 0; i < n; i++) {
                if (s[i] == t[0]) continue;
                cout << s[i];
            }
            for (int i = 0; i < 6; i++) cout << t[i];
            cout << endl;
        }
    }
    return 0;
}
