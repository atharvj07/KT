def find_permutation(n, k):
    def generate_permutation(l, r, k, permutation):
        if k < 0: return False
        if l == r: return k == 0
        if k == 1: return False  # Can't make any more recursive calls
        mid = (l + r) // 2
        permutation[mid], permutation[r] = permutation[r], permutation[mid]
        return generate_permutation(l, mid, k - 1, permutation) or generate_permutation(mid + 1, r, k - 1, permutation)
    
    permutation = list(range(1, n + 1))
    if generate_permutation(0, n - 1, k, permutation):
        return permutation
    else:
        return [-1]

n, k = map(int, input().split())
result = find_permutation(n, k - 1)  # Adjust k since initial call is not counted
print(' '.join(map(str, result)))
