#include <bits/stdc++.h>
using namespace std;
#define int long long
#define maxn 1000005
int power(int x, int y) {
    if (y == 0)
        return 1;
    if (y == 1)
        return x;
    int ans = power(x, y / 2);
    ans = (ans * ans) % 1000000007;
    if (y % 2)
        return (ans * x) % 1000000007;
    else
        return ans % 1000000007;
}
int32_t main() {
    int n, x;
    cin >> n >> x;
    int ans = 0;
    int xs = 1;
    vector<int> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
        xs = (xs * power(x, v[i])) % 1000000007;
    }
    for (auto p : v) {
        ans = (ans + power(x, p)) % 1000000007;
    }
    int gcd = __gcd(ans, xs);
    cout << gcd << endl;
}
