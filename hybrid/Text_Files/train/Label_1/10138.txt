#include <iostream>
#include <vector>
using namespace std;
#define int long long
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, k;
        cin >> n >> m >> k;
        vector<int> h(n);
        for (int i = 0; i < n; i++) cin >> h[i];
        bool ok = 0;
        if (m == 0)
            ok |= h[0] <= k;
        for (int i = 0; i < n - 1; i++) {
            if (h[i] < h[i + 1]) {
                if (h[i + 1] - h[i] <= k) ok = 1;
                h[i + 1] = h[i];
            } else {
                if (h[i] - h[i + 1] > k) {
                    m -= h[i] - h[i + 1] - k;
                    h[i] = h[i + 1] + k;
                }
                if (m < 0) ok = 0;
            }
        }
        cout << (ok ? "YES" : "NO") << endl;
    }
}
