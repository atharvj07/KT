#include<bits/stdc++.h>
using namespace std;
#define ll long long int
#define lld long double
#define mod 1000000007

ll x, d, ans[10005];

void solve() {
  if(d == 0) {
    if(x > 1) {
      cout << -1 << endl;
      return;
    }
    cout << 1 << endl << x << endl;
    return;
  }
  ll n = (x+1)/2;
  ll y = 1, z = d+1;
  if(n % 2 == 0) {
    y = d+1;
    z = 1;
  }
  for(ll i = 1; i <= n; i++) {
    ans[i] = y;
    y *= 2;
    z *= 2;
    if(y > 1000000000000000000) {
      y = 1;
      z = d+1;
    }
  }
  cout << n << endl;
  for(ll i = 1; i <= n; i++)
    cout << ans[i] << " ";
  cout << endl;
}
int main() {
  cin >> x >> d;
  solve();
  return 0;
}
