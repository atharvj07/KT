#include<iostream>
using namespace std;
int main() {
    int k, i, j, ti, a[10], f = 0;
    char str[5];
    cin >> k;
    for (i = 0; i < 4; i++) {
        f = 0;
        cin >> str;
        for (j = 0; j < 4; j++) {
            if (str[j] != '.') {
                ti = str[j] - '0';
                if (a[ti] == 0) { a[ti] = 1; } else { f = 1; }
            }
        }
        if (f == 1) { break; }
        for (j = 1; j <= 9; j++) { a[j] = 0; }
    }
    if (f == 1) { cout << "NO" << endl; } else { cout << "YES" << endl; }
    return 0;
}
