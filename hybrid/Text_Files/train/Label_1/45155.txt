import java.util.*;

class Main {
    static int r, c, res;
    static char[][] g;
    static boolean[][] vis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        r = sc.nextInt();
        c = sc.nextInt();
        g = new char[r][c];
        vis = new boolean[r][c];
        for (int i = 0; i < r; i++) {
            String s = sc.next();
            for (int j = 0; j < c; j++) {
                g[i][j] = s.charAt(j);
            }
        }
        bfs();
        System.out.println(res);
    }

    static void bfs() {
        Queue<Pos> q = new LinkedList<>();
        for (int i = 0; i < r; i++) {
            for (int j = 0; j < c; j++) {
                if (g[i][j] == 'S') {
                    q.add(new Pos(i, j, 0));
                    vis[i][j] = true;
                }
            }
        }
        while (!q.isEmpty()) {
            Pos cur = q.poll();
            int x = cur.x;
            int y = cur.y;
            int cnt = cur.cnt;
            if (g[x][y] == 'E') {
                res = cnt;
                return;
            }
            for (int[] d : new int[][]{{0, 1}, {0, -1}, {1, 0}, {-1, 0}}) {
                int nx = x + d[0];
                int ny = y + d[1];
                if (nx >= 0 && nx < r && ny >= 0 && ny < c && !vis[nx][ny] && g[nx][ny] != 'T') {
                    if (g[nx][ny] >= '1' && g[nx][ny] <= '9') {
                        q.add(new Pos(nx, ny, cnt + g[nx][ny] - '0'));
                    } else {
                        q.add(new Pos(nx, ny, cnt));
                    }
                    vis[nx][ny] = true;
                }
            }
        }
    }

    static class Pos {
        int x;
        int y;
        int cnt;

        public Pos(int x, int y, int cnt) {
            this.x = x;
            this.y = y;
            this.cnt = cnt;
        }
    }
}
