#include<bits/stdc++.h>
using namespace std;
const int N=1e5+10;
int n,k;
string a[N];
string b;
map<string,int>mp;
int check(string s,int x){
    int ans=0;
    int l=s.size();
    for(int i=0;i<n;i++){
        if(a[i].size()>=l && s==a[i].substr(0,l))ans+=(!mp[a[i]]);
    }
    if(x==0 && ans>0)return 1;
    if(x==1 && ans==0)return 1;
    return 0;
}
int main(){
    cin>>n>>k;
    for(int i=0;i<n;i++){
        cin>>a[i];
        mp[a[i]]++;
    }
    int c=check(b,0);
    if(k==1){
        cout<<(c?"First":"Second")<<endl;
    }
    else{
        cout<<(c?"Second":"First")<<endl;
    }
}
