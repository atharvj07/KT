import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = s.charAt(i) == 'R' ? 1 : -1;
        }

        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            if (sum == 0) {
                k--;
                if (k < 0) {
                    break;
                }
                sum = a[i];
            }
        }

        int ans = Math.abs(sum);
        if (k % 2 == 1) {
            ans = n - ans;
        }

        System.out.println(ans);
    }
}
