#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        string s;
        for (int i = 0; i < k; i++) {
            s += (char)('a' + i);
        }
        int len = s.size();
        for (int i = 0; i < n - len; i++) {
            s += s[i % len];
        }
        cout << s << "\n";
    }
    return 0;
}
