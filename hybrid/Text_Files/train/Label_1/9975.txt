#include <iostream>
using namespace std;

int main() {
    int s, v1, v2, t1, t2;
    cin >> s >> v1 >> v2 >> t1 >> t2;
    int t = max(t1 + s * v1, t2 + s * v2);
    if (t == t1 + s * v1)
        cout << "First" << endl;
    else if (t == t2 + s * v2)
        cout << "Second" << endl;
    else
        cout << "Friendship" << endl;
    return 0;
}
