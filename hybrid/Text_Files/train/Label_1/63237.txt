n, p, s = map(int, input().strip().split())
a = list(map(int, input().strip().split()))
b = list(map(int, input().strip().split()))
ind = [i for i in range(n)]
ind.sort(key=lambda x: a[x]-b[x], reverse=True)
spa = 0
spb = 0
for i in range(p):
    spa += a[ind[i]]
for i in range(s):
    spb += b[ind[p+i]]
print(spa+spb)
print(*[x+1 for x in ind[:p]])
print(*[x+1 for x in ind[p:]])
