#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> t(n);
  for (int i = 0; i < n; i++) {
    cin >> t[i];
  }
  int i, j, k, l = 0, r = n - 1, cnt1 = 0, cnt2 = 0;
  for (i = 0; i < n; i++) {
    if (t[i] == 1) {
      cnt1++;
    } else {
      cnt2++;
    }
  }
  int ans = 0;
  for (i = 0, j = n - 1; i <= j;) {
    if (t[i] == 1) {
      cnt1--;
    } else {
      cnt2--;
    }
    if (t[j] == 1) {
      cnt1--;
    } else {
      cnt2--;
    }
    if (cnt1 == cnt2) {
      ans = max(ans, j - i + 1);
    }
    if (cnt1 < cnt2) {
      i++;
    } else {
      j--;
    }
  }
  cout << ans << endl;
  return 0;
}
