#include <cmath>
#include <cstdio>
#include <cstring>

using namespace std;

int n;
char grid[51][52];
bool vis[51][52];

bool valid(int x1, int y1) {
  return (x1 >= 0 && y1 >= 0 && x1 < n && y1 < n && grid[x1][y1] == '.' &&
          !vis[x1][y1]);
}

bool check(int x1, int y1) {
  if (!valid(x1, y1)) return false;
  vis[x1][y1] = true;
  if (check(x1 + 1, y1) && check(x1 - 1, y1) && check(x1, y1 + 1) &&
      check(x1, y1 - 1))
    return true;
  vis[x1][y1] = false;
  return false;
}

bool solve() {
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (grid[i][j] == '.' && check(i, j)) {
        return true;
      }
    }
  }
  return false;
}

int main() {
  while (scanf("%d", &n) == 1) {
    memset(grid, 0, sizeof(grid));
    memset(vis, 0, sizeof(vis));
    for (int i = 0; i < n; i++) {
      scanf("%s", grid[i]);
    }
    puts(solve() ? "YES" : "NO");
  }
  return 0;
}
