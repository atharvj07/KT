#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, m, k, l, r, ans = -1, a[1000001];
  cin >> n >> m >> k;
  for (int i = 0; i < m; i++) {
    cin >> l;
    ++a[l];
  }
  for (int i = 1; i <= k; i++) {
    cin >> l;
    bool tr = false;
    for (int j = 0; j <= n; j += i) {
      if (a[j] > 0) {
        tr = true;
        break;
      }
    }
    if (tr == false) {
      if (ans == -1) {
        ans = l;
      } else {
        ans = min(ans, l);
      }
    }
  }
  cout << ans << endl;
  return 0;
}
