import heapq
import sys

n, m, a, b = map(int, input().split())
adj = [[] for i in range(n + 1)]
for i in range(m):
    u, v, c = map(int, input().split())
    adj[u].append((c, v))
    adj[v].append((c, u))

def dijkstra(st, en):
    dist = [float('inf') for i in range(n + 1)]
    dist[st] = 0
    pq = [(0, st)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]: continue
        for w, v in adj[u]:
            if d + w < dist[v]:
                dist[v] = d + w
                heapq.heappush(pq, (d + w, v))
    return dist[en]

ans = []
for i in range(1, n + 1):
    ans.append(dijkstra(1, i))
print(*ans)
