n = int(input())
a = [[0 for j in range(5)] for i in range(n)]
for i in range(n):
    a[i] = input()
f = False
for i in range(n):
    for j in range(4):
        if i+1<n and j+1<4 and a[i+1][j+1] == 'O' and a[i][j] == 'O':
            a[i][j] = '+'
            a[i+1][j+1] = '+'
            f = True
            break
    if f: break
if not f:
    print('NO')
else:
    print('YES')
    for i in range(n):
        print(*a[i], sep='')
