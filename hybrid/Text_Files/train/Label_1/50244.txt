import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        long[] sums = new long[k];
        List<List<Integer>> seqs = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            int n = sc.nextInt();
            int[] seq = new int[n];
            for (int j = 0; j < n; j++) {
                seq[j] = sc.nextInt();
            }
            sums[i] = Arrays.stream(seq).sum();
            seqs.add(new ArrayList<>());
            for (int j = 0; j < n; j++) {
                seqs.get(i).add(seq[j]);
            }
        }

        for (int i = 0; i < k; i++) {
            for (int j = i + 1; j < k; j++) {
                long sum1 = sums[i], sum2 = sums[j];
                for (int x = 0; x < seqs.get(i).size(); x++) {
                    for (int y = 0; y < seqs.get(j).size(); y++) {
                        if (sum1 - seqs.get(i).get(x) == sum2 - seqs.get(j).get(y)) {
                            System.out.println("YES");
                            System.out.println((i + 1) + " " + (x + 1));
                            System.out.println((j + 1) + " " + (y + 1));
                            return;
                        }
                    }
                }
            }
        }
        System.out.println("NO");
    }
}
