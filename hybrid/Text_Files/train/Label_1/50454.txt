import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        sc.close();

        int pos = 0;
        int max = d[0];
        for (int i = 0; i < n; i++) {
            if (pos + d[i] >= n - 1) {
                pos += d[i];
                break;
            }
            pos += d[i];
            max = Math.max(max, pos);
            if (pos >= n - 1) {
                System.out.println("yes");
                return;
            }
            pos -= d[i];
            if (pos < i) {
                System.out.println("no");
                return;
            }
            if (max <= i) {
                System.out.println("no");
                return;
            }
        }

        for (int i = n - 2; i >= 0; i--) {
            if (pos - d[i] <= 0) {
                pos -= d[i];
                break;
            }
            pos -= d[i];
            max = Math.max(max, pos);
            if (pos <= 0) {
                System.out.println("yes");
                return;
            }
            pos += d[i];
            if (pos > i) {
                System.out.println("no");
                return;
            }
            if (max <= i) {
                System.out.println("no");
                return;
            }
        }

        System.out.println("no");
    }
}
