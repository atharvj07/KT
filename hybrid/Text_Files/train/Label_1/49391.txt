import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = scanner.nextInt();
        }
        int s = 0;
        int min = 1000000000;
        for (int i = 0; i < n; i++) {
            if (x[i] < min) {
                min = x[i];
            } else {
                s += min;
            }
        }
        System.out.println(s);
    }
}
