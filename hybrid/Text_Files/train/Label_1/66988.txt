from sys import stdin
input = stdin.readline

for _ in range(int(input())):
    n, t = map(int, input().split())
    a = list(map(int, input().split()))
    b = []
    for v in a:
        if v + v <= t: b.append(v * 2)
        else: b.append(v)
    b.sort(reverse=True)
    ans = [0] * n
    i, j = 0, n - 1
    for v in b:
        if i >= j: break
        if v <= t:
            ans[i] = 1
            ans[j] = 1
            i += 1
            j -= 1
        else:
            ans[i] = 1
            i += 1
    print(' '.join(map(str, ans)))
