#include <vector>
#include <iostream>

int main() {
    int t; std::cin >> t;
    for (int T = 0; T < t; ++T) {
        int x, y; std::cin >> x >> y;
        std::vector<int> ans;
        while (x != 0 || y != 0) {
            if (x == 0) {
                while (y > 0) ans.push_back(1), --y;
                while (y < 0) ans.push_back(3), ++y;
            } else if (y == 0) {
                while (x > 0) ans.push_back(2), --x;
                while (x < 0) ans.push_back(4), ++x;
            } else {
                int dx = std::min(abs(x - 1), abs(x + 1));
                int dy = std::min(abs(y - 1), abs(y + 1));
                if (dx < dy) {
                    if (x < 0) {
                        ans.push_back(4), ++x;
                        if (dy < x + 1) {
                            ans.push_back(1), ++y;
                            if (x + 1 < y) ans.push_back(3);
                            else if (x + 1 > y) ans.push_back(1);
                        } else {
                            ans.push_back(2), --x;
                            if (y < x - 1) ans.push_back(1);
                            else if (y > x - 1) ans.push_back(3);
                        }
                    } else {
                        ans.push_back(2), --x;
                        if (dy < y - x) {
                            ans.push_back(3), --y;
                            if (y - x < -1) ans.push_back(4);
                            else if (y - x > -1) ans.push_back(2);
                        } else {
                            ans.push_back(1), ++y;
                            if (y - x < 1) ans.push_back(4);
                            else if (y - x > 1) ans.push_back(2);
                        }
                    }
                } else {
                    if (y < 0) {
                        ans.push_back(3), ++y;
                        if (dx < y + 1) {
                            ans.push_back(2), ++x;
                            if (y + 1 < x) ans.push_back(4);
                            else if (y + 1 > x) ans.push_back(2);
                        } else {
                            ans.push_back(1), --y;
                            if (x < y - 1) ans.push_back(2);
                            else if (x > y - 1) ans.push_back(4);
                        }
                    } else {
                        ans.push_back(1), ++y;
                        if (dx < x - y) {
                            ans.push_back(4), --x;
                            if (x - y < -1) ans.push_back(3);
                            else if (x - y > -1) ans.push_back(1);
                        } else {
                            ans.push_back(2), --x;
                            if (x - y < 1) ans.push_back(3);
                            else if (x - y > 1) ans.push_back(1);
                        }
                    }
                }
            }
        }
        std::cout << ans.size() << "\n";
    }
    return 0;
}
