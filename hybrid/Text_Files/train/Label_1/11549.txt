#include <bits/stdc++.h>
using namespace std;
int n, k;
vector<int> p, q, s;

bool solve(vector<int>& per) {
	vector<int> cur = per;
	for (int i = 0; i < k; i++) {
		int ch;
		cin >> ch;
		if (ch == 1) {
			for (int j = 0; j < n; j++) cur[j] = per[p[j] - 1];
		} else {
			for (int j = 0; j < n; j++) cur[p[j] - 1] = per[j];
		}
	}
	return (cur == s);
}
int main() {
	cin >> n >> k;
	p.resize(n);
	for (int i = 0; i < n; i++) cin >> p[i];
	q.resize(n);
	for (int i = 0; i < n; i++) cin >> q[i];
	s.resize(n);
	for (int i = 0; i < n; i++) cin >> s[i];
	cout << (solve(q) ? "YES" : "NO") << endl;
	return 0;
}
