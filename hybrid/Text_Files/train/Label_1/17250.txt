#include <bits/stdc++.h>
using namespace std;
long long a[1000001];
int main(){
    long long n,m,b;
    cin>>n>>m;
    for(long long i = 0; i < n; i++){
        cin>>a[i];
        b |= 1LL<<a[i];
    }
    long long ans = 0,sum = 0;
    for(long long i = 1; i <= 1000000; i++){
        if((b & (1LL<<i)) == 0 && i + sum <= m){
            ans++;
            sum += i;
        }
    }
    cout<<ans<<'\n';
    sum = 0;
    for(long long i = 1; i <= 1000000; i++){
        if((b & (1LL<<i)) == 0 && i + sum <= m){
            cout<<i<<' ';
            sum += i;
        }
    }
    cout<<'\n';
    return 0;
}
