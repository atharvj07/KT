import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int xor = 0;
        for (int i = 1; i < n; i++) {
            xor ^= a[i] ^ a[0];
        }
        if (xor == 0) {
            System.out.println(a[0]);
        } else {
            System.out.println(-1);
        }
    }
}
