#include <bits/stdc++.h>
using namespace std;

const long long INF = 1e18;

int main() {
  int n, k;
  cin >> n >> k;
  vector<pair<int, int>> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i].first >> a[i].second;
  }
  sort(a.begin(), a.end(), [](const pair<int, int>& p1, const pair<int, int>& p2) {
    return -p1.first + p1.second > -p2.first + p2.second;
  });
  long long ans = INF;
  for (int i = 0; i < n; i++) {
    if (n - i < k) break;
    long long cur = 0;
    for (int j = 0; j < n; j++) {
      if (i == j) cur += a[j].second;
      else cur -= a[j].first;
    }
    ans = min(ans, cur);
  }
  cout << (ans == INF ? -1 : ans) << endl;
  return 0;
}
