#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) { cin >> x; }
  int ans = 0;
  for (int i = 0; i < n; i++) { ans += (a[i] == i); }
  unordered_map<int, int> cnt;
  for (int i = 0; i < n; i++) {
    if (a[i] == i) { continue; }
    if (a[a[i]] == i) { ans++; }
    cnt[a[i]]++;
    cnt[i]++;
  }
  for (auto p : cnt) { ans = max(ans, 1 + p.second); }
  cout << ans << '\n';
  return 0;
}
