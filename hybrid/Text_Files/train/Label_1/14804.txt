#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
const ll N = 150005;
const ll NN = 410000;
const ll inf = 1e18;
vector<ll> g[4][N];
ll a[N], b[N], c[N], d[N];
vector<ll> dist(NN, inf);
ll n1, n2, n3, n4, m1, m2, m3;
ll food[4][N], ans = inf;
queue<ll> q;
int main() {
  cin >> n1 >> n2 >> n3 >> n4;
  for (ll i = 1; i <= n1; i++) cin >> a[i];
  for (ll i = 1; i <= n2; i++) cin >> b[i];
  for (ll i = 1; i <= n3; i++) cin >> c[i];
  for (ll i = 1; i <= n4; i++) cin >> d[i];
  cin >> m1;
  while (m1--) {
    ll x, y;
    cin >> x >> y;
    g[0][x].push_back(n1 + y);
    g[1][y].push_back(x);
  }
  cin >> m2;
  while (m2--) {
    ll x, y;
    cin >> x >> y;
    g[1][x].push_back(n1 + n2 + y);
    g[2][y].push_back(n1 + x);
  }
  cin >> m3;
  while (m3--) {
    ll x, y;
    cin >> x >> y;
    g[2][x].push_back(n1 + n2 + n3 + y);
    g[3][y].push_back(n1 + n2 + x);
  }
  for (ll i = 0; i <= 3; i++) {
    for (ll j = 1; j <= n1 + n2 + n3 + n4; j++) {
      food[i][j] = i * (n1 + n2 + n3 + n4) + j;
      q.push(food[i][j]);
      dist[food[i][j]] = 0;
    }
  }
  while (!q.empty()) {
    ll cur = q.front();
    q.pop();
    ll i = (cur - 1) / (n1 + n2 + n3 + n4);
    ll j = cur - i * (n1 + n2 + n3 + n4);
    if (i == 0 && j >= 1 && j <= n1) ans = min(ans, dist[cur] + a[j]);
    if (i == 1 && j >= n1 + 1 && j <= n1 + n2) ans = min(ans, dist[cur] + b[j - n1]);
    if (i == 2 && j >= n1 + n2 + 1 && j <= n1 + n2 + n3)
      ans = min(ans, dist[cur] + c[j - n1 - n2]);
    if (i == 3 && j >= n1 + n2 + n3 + 1 && j <= n1 + n2 + n3 + n4)
      ans = min(ans, dist[cur] + d[j - n1 - n2 - n3]);
    for (ll nxt : g[i][j]) {
      if (dist[nxt] > dist[cur] + 1) {
        dist[nxt] = dist[cur] + 1;
        q.push(nxt);
      }
    }
  }
  if (ans < inf)
    cout << ans << endl;
  else
    cout << -1 << endl;
  return 0;
}
