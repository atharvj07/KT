import java.util.*;
import java.io.*;

public class Main {
    static int n, k;
    static int[] a;
    static List<Integer> ans = new ArrayList<>();
    static int solve(int i, int sum, int count) {
        if (i == n) {
            if (sum <= k) {
                if (count > ans.size()) {
                    ans.clear();
                    for (int j = 0; j < n; j++)
                        if (a[j] <= k - sum)
                            ans.add(j + 1);
                }
            }
            return count;
        }
        int x = solve(i + 1, sum + a[i], count + 1);
        int y = solve(i + 1, sum, count);
        return Math.max(x, y);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        k = Integer.parseInt(line[1]);
        a = new int[n];
        line = br.readLine().split(" ");
        for (int i = 0; i < n; i++)
            a[i] = Integer.parseInt(line[i]);
        int count = solve(0, 0, 0);
        System.out.println(count);
        for (int x : ans)
            System.out.print(x + " ");
        System.out.println();
    }
}
