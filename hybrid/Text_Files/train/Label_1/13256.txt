#include <bits/stdc++.h>
using namespace std;
const int N = 105;
int n, m, arr[N][N], vis[205];
vector<string> ans;

void col(int x) {
    ans.push_back("col " + to_string(x));
    for (int i = 1; i <= n; i++) {
        arr[i][x]++;
    }
    vis[x + 100] = 1;
}

void row(int x) {
    ans.push_back("row " + to_string(x));
    for (int i = 1; i <= m; i++) {
        arr[x][i]++;
    }
    vis[x] = 1;
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> arr[i][j];
        }
    }
    int ok = 1;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (arr[i][j]) {
                vis[i] += arr[i][j];
                vis[j + 100] += arr[i][j];
                if (vis[i] > 1 || vis[j + 100] > 1) ok = 0;
            }
        }
    }
    int cnt_row = 0, cnt_col = 0;
    for (int i = 1; i <= n; i++) cnt_row += (vis[i] == 0);
    for (int i = 1; i <= m; i++) cnt_col += (vis[i + 100] == 0);
    if (cnt_row + cnt_col != 1) ok = 0;
    if (!ok)
        cout << -1 << endl;
    else {
        if (cnt_row == 1) {
            for (int i = 1; i <= n; i++) {
                if (vis[i] == 0) { row(i); break; }
            }
            for (int i = 1; i <= m; i++) {
                if (vis[i + 100] == 0) { col(i); break; }
            }
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= m; j++) {
                    if (arr[i][j] == 0) { row(i); col(j); }
                }
            }
        } else {
            for (int i = 1; i <= m; i++) {
                if (vis[i + 100] == 0) { col(i); break; }
            }
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= m; j++) {
                    if (arr[i][j] == 0) { row(i); col(j); }
                }
            }
        }
        cout << ans.size() << endl;
        for (string x : ans) cout << x << endl;
    }
}
