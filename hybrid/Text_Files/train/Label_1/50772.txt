import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        BigInteger x = input.nextBigInteger();
        int k = x.bitLength() - 1;
        if ((x.testBit(k)) || (x.equals(BigInteger.ONE))) System.out.println("1 " + x);
        else System.out.println("2 " + x);
    }
}
