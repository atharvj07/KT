import sys

def solve():
    t = int(input())
    for _ in range(t):
        n, k = map(int, input().split())
        p = list(map(int, input().split()))
        ans = 0
        for i in range(1, n):
            cur = (p[i] * 100) / (sum(p[:i]) + p[i])
            if cur > k:
                ans += p[i] - (sum(p[:i]) * k) // 100
        print(ans)

solve()
