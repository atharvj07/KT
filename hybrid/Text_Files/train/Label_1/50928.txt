import java.util.*;
import java.io.*;

class Main {
    static int[] a;
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), k = input.nextInt();
        a = new int[n + 1];
        for(int i = 1; i <= n; ++i)
            a[i] = input.nextInt();
        ArrayDeque<Integer> q = new ArrayDeque<>();
        for(int i = 1; i <= n; ++i) {
            while(!q.isEmpty() && a[q.peekLast()] >= a[i])
                q.pollLast();
            q.add(i);
            while(q.size() > k)
                q.poll();
            if(q.size() == k)
                System.out.print(a[q.peek()] + " ");
            else
                System.out.print(-1 + " ");
        }
    }
}
