n,m=map(int,input().split())
a=len(str(n-1))
b=len(str(m-1))
d=[0]*max(a,b)
e=[]
for i in str(n-1):
    if int(i) in d:
        e.append(1)
        break
    else:    
        d[int(i)]=1
for i in str(m-1):
    if int(i) in d:
        e.append(1)
        break
    else:
        d[int(i)]=1
if len(e)==0:print(n*m)
else:print(0)
