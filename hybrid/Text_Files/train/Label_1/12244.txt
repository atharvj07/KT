#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> h(n);
  for (int i = 0; i < n; ++i) cin >> h[i];
  sort(h.begin(), h.end());
  int ans = 0;
  long long sum = 0;
  for (int i = n - 1; i >= 0; --i) {
    if (sum + h[i] - h[0] <= k) {
      sum += h[i] - h[0];
      ans++;
    } else {
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
