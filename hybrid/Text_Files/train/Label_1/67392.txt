from sys import stdin

mod = 1000000007

def get_adjacent_bombs(field, i):
    adjacent_bombs = 0
    if i > 0 and field[i - 1] == '*':
        adjacent_bombs += 1
    if i < len(field) - 1 and field[i + 1] == '*':
        adjacent_bombs += 1
    return adjacent_bombs

def count_ways(field):
    dp = [0] * len(field)
    if field[0] == '?':
        dp[0] = 2
    else:
        dp[0] = 1
    for i in range(1, len(field)):
        if field[i] == '*':
            dp[i] = 0
        elif field[i] == '?':
            if get_adjacent_bombs(field, i) == int(field[i - 1]):
                dp[i] = dp[i - 1]
            else:
                dp[i] = (2 * dp[i - 1]) % mod
        else:
            if get_adjacent_bombs(field, i) == int(field[i - 1]):
                dp[i] = dp[i - 1]
            else:
                dp[i] = 0
    return dp[-1]

field = stdin.readline().strip()
print(count_ways(field))
