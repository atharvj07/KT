from collections import defaultdict, deque

def find_order(words, n):
    # Create a graph and in-degree array
    graph = defaultdict(list)
    in_degree = {chr(ord('a')+i): 0 for i in range(26)}

    # Function to add edge to the graph
    def add_edge(u, v):
        graph[u].append(v)
        in_degree[v] += 1

    # Build the graph
    for i in range(n-1):
        word1, word2 = words[i], words[i+1]
        length = min(len(word1), len(word2))
        for j in range(length):
            if word1[j] != word2[j]:
                add_edge(word1[j], word2[j])
                break
        else: # Check if a prefix is longer than the word following it
            if len(word1) > len(word2):
                return "Impossible"

    # Topological sort using Kahn's algorithm
    queue = deque([node for node in in_degree if in_degree[node] == 0])
    order = ''
    while queue:
        current = queue.popleft()
        order += current
        for neighbor in graph[current]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    if len(order) < 26:
        return "Impossible"
    return order

# Read input
n = int(input().strip())
words = [input().strip() for _ in range(n)]

# Find and print the order
print(find_order(words, n))
