#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long a, b;
        cin >> a >> b;
        if (a == 0 && b == 0)
            cout << "YES" << endl;
        else if (a == b)
            cout << "YES" << endl;
        else if (a > b) {
            if ((a - b) % 3 == 0)
                cout << "YES" << endl;
            else
                cout << "NO" << endl;
        } else {
            if ((b - a) % 3 == 0)
                cout << "YES" << endl;
            else
                cout << "NO" << endl;
        }
    }
    return 0;
}
