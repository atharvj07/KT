#include <bits/stdc++.h>
using namespace std;

int a[2005],f[2005][2005],n;

int main(){
    cin>>n;
    for(int i=1;i<=n;++i)cin>>a[i];
    for(int i=1;i<=n;++i){
        if(i>1){
            for(int j=i-1;j>=1;--j){
                if(a[i]>=a[j])f[i][f[j][i-1]+1]=max(f[i][f[j][i-1]+1],f[j][i-1]+1);
                else f[i][f[i][i-1]]=max(f[i][f[i][i-1]],f[i][i-1]);
            }
        }
        else f[i][1]=1;
    }
    int ans=0;
    for(int i=1;i<=n;++i)ans=max(ans,f[i][i]);
    cout<<ans<<endl;
    return 0;
}
