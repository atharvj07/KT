import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int red = 0;
        int blue = 0;
        for (char c : s.toCharArray()) {
            if (c == 'R') {
                red++;
            } else {
                blue++;
            }
        }
        System.out.println(red > blue ? "Yes" : "No");
    }
}
