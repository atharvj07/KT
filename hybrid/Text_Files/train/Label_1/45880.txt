import java.util.Scanner;

public class BubbleCupFinals {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] field = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                field[i][j] = scanner.nextInt();
            }
        }

        int maxBubbles = 0;
        for (int i = 0; i < n; i++) {
            int rowSum = 0;
            for (int j = 0; j < m; j++) {
                rowSum += field[i][j];
            }
            maxBubbles = Math.max(maxBubbles, rowSum);
        }

        for (int j = 0; j < m; j++) {
            int colSum = 0;
            for (int i = 0; i < n; i++) {
                colSum += field[i][j];
            }
            maxBubbles = Math.max(maxBubbles, colSum);
        }

        System.out.println(maxBubbles);
    }
}
