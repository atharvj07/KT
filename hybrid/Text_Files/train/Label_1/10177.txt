#include <bits/stdc++.h>
using namespace std;
int main() {
    string str;
    cin >> str;
    int n = str.length();
    char arr[100005];
    for (int i = 0; i < n; i++) { arr[i] = str[i]; }
    sort(arr, arr + n);
    bool isPosible = false;
    for (int i = n - 2; i >= 0; i--) {
        if (arr[i] != arr[i + 1]) {
            int a = i;
            int b = i + 1;
            char temp = arr[a];
            arr[a] = arr[b];
            arr[b] = temp;
            isPosible = true;
            break;
        }
    }
    if (!isPosible) {
        cout << "-1" << endl;
        return 0;
    }
    string ans = "";
    ans += arr[n - 1];
    for (int i = n - 2; i >= 0; i--) { ans += arr[i]; }
    int x = stoi(ans);
    if (x % 2 == 0) {
        cout << x << endl;
    } else {
        cout << "-1" << endl;
    }
    return 0;
}
