#include <bits/stdc++.h>

using namespace std;
string s;
int n;
int main() {
  cin >> n >> s;
  int p = n;
  while (p > 0 && s[p - 1] == '0') p--;
  if (p == 0) cout << 0 << endl;
  else {
    for (int i = 0; i < p - 1; i++) {
      if (s[i] == '0' && s[i + 1] == '1') {
        s[i] = '1', s[i + 1] = '0';
        break;
      }
    }
    s.resize(p);
    cout << s << endl;
  }
  return 0;
}
