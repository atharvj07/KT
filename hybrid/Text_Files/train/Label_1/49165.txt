import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

class Main {
    public static void main(String[] args) {
        try (BufferedReader bf = new BufferedReader(new InputStreamReader(System.in))) {
            int[] line = createLine(bf.readLine().split(" "));
            int limit = 0;
            for (int number: line) {
                if (number < 0) {
                    limit = 1;
                }
            }
            if (line[0] == 1 && line[1] == 1) {
                System.out.println(-1);
            } else if (line[1] == 0) {
                int count = 0;
                line[0] = (int) Math.pow(2, floorLog2(line[0]));
                while (line[0] > 0) {
                    count++;
                    line[0] /= 2;
                }
                System.out.println(count);
            } else {
                int count = 1;
                for (int i = floorLog2(line[0]) + limit; i >= limit; i--) {
                    if ((int) Math.pow(2, i) + line[1] <= line[0]) {
                        count++;
                        line[0] -= (int) Math.pow(2, i) + line[1];
                        i = floorLog2(line[0]) + limit + 1;
                    }
                }
                if (line[0] > 0) {
                    System.out.println(-1);
                } else {
                    System.out.println(count);
                }
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private static int floorLog2(int n) {
        int result = (int) (Math.log(n) / Math.log(2));
        return (int) Math.pow(2, result) == n ? result : result - 1;
    }

    private static int[] createLine(String[] arr) {
        int[] result = new int[arr.length];

        for (int i = 0; i < arr.length; i++) {
            result[i] = Integer.parseInt(arr[i]);
        }

        return result;
    }
}
