import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int m = in.nextInt();
            int a = in.nextInt();
            int b = in.nextInt();
            if (n * a != m * b) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < m; j++) {
                        if (i < a && j < b) {
                            System.out.print(1);
                        } else {
                            System.out.print(0);
                        }
                    }
                    System.out.println();
                }
            }
        }
        in.close();
    }
}
