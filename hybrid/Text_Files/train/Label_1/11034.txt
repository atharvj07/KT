#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, r;
  cin >> n >> r;
  vector<int> arr(n);
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  int ans = 0, cur = 0;
  for (int i = 0; i < n; i++) {
    if (arr[i] == 1) {
      cur = max(cur, i + r - 1);
    }
    if (cur < i) {
      ans = -1;
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
