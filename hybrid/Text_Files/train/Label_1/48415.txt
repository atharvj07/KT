import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt(), m = in.nextInt(), a[] = new int[k];
        for (int i = 0; i < k; ++i) { a[i] = in.nextInt(); }
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        for (int i = 0; i < k; ++i) { pq.add(a[i]); }
        int ans = 0;
        while (!pq.isEmpty()) {
            int x = pq.poll();
            if (x <= m) {
                ++ans;
                m -= x;
            } else { break; }
        }
        ans += m / a[0];
        System.out.println(ans);
    }
}
