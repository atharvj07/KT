n=int(input())
p=[int(x) for x in input().split()]
q=[int(x) for x in input().split()]
r=[0 for _ in range(n)]
cnt=[0 for _ in range(n+1)]
for x in p:
    cnt[x]+=1
for x in q:
    cnt[x]+=1
for x in range(n-1,-1,-1):
    t=0
    for y in range(n):
        if(cnt[y]>0):
            cnt[y]-=1
            t=y
            break
    for y in range(n):
        if(cnt[y]==0):
            r[x]=y
            break
    cnt[t]+=1
print(*r)
