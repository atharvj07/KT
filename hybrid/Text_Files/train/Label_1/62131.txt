n, m, s, f = map(int, input().split())
watch_steps = [list(map(int, input().split())) for _ in range(m)]

def move_spy(n, m, s, f, watch_steps):
    current_step = 1
    current_spy = s
    actions = []
    watch_index = 0

    while current_spy != f:
        if watch_index < m and watch_steps[watch_index][0] == current_step:
            if watch_steps[watch_index][1] <= current_spy <= watch_steps[watch_index][2] or \
               ((f > current_spy and watch_steps[watch_index][1] <= current_spy + 1 <= watch_steps[watch_index][2]) or
               (f < current_spy and watch_steps[watch_index][1] <= current_spy - 1 <= watch_steps[watch_index][2])):
                actions.append('X')
            else:
                if f > current_spy:
                    current_spy += 1
                    actions.append('R')
                else:
                    current_spy -= 1
                    actions.append('L')
            watch_index += 1
        else:
            if f > current_spy:
                current_spy += 1
                actions.append('R')
            else:
                current_spy -= 1
                actions.append('L')
        current_step += 1
    return ''.join(actions)

print(move_spy(n, m, s, f, watch_steps))
