#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MAXN = 2e5+5;
ll f[MAXN];
ll n;
bool vis[MAXN];
ll p[MAXN];
int main() {
    cin >> n;
    for (ll i = 1; i <= n; i++) {
        cin >> f[i];
    }
    for (ll i = 1; i <= n; i++) {
        if (f[i]) {
            p[i] = f[i];
            vis[f[i]] = true;
        }
    }
    for (ll i = 1; i <= n; i++) {
        if (f[i] == 0) {
            for (ll j = 1; j <= n; j++) {
                if (!vis[j] && j != i) {
                    p[i] = j;
                    vis[j] = true;
                    break;
                }
            }
        }
    }
    for (ll i = 1; i <= n; i++) {
        cout << p[i] << " ";
    }
    cout << endl;
    return 0;
}
