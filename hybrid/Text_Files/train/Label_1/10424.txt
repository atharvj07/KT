#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (auto &[a, b] : v) {
    cin >> a >> b;
    a += 1000 - b;
  }
  sort(v.begin(), v.end());
  int sum = 0;
  for (int i = 0; i < (n + 1) / 2; i++) sum += v[i].first;
  if (sum - 500 > v[n / 2].first || sum + 500 < v[n / 2].first) return puts("-1"), 0;
  for (auto [a, b] : v) {
    if (sum <= 500) {
      cout << "A";
      sum += a;
    } else {
      cout << "G";
      sum -= b;
    }
  }
  return 0;
}
