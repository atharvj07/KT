#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> adj(n + 1);
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  queue<int> q;
  vector<bool> vis(n + 1, false);
  q.push(1);
  int ans = 0;
  while (!q.empty()) {
    int cur = q.front();
    q.pop();
    vis[cur] = true;
    int cnt = 0;
    for (int u : adj[cur]) {
      if (!vis[u]) {
        q.push(u);
        cnt++;
        if (cnt > 1) ans++;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
