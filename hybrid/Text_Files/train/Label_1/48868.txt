import java.util.*;
import java.lang.Character;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        String str = s.nextLine();
        int[] freq = new int[26];
        for (int i = 0; i < str.length(); i++) {
            freq[str.charAt(i) - 'A']++;
        }
        int[] freq_odd = new int[26];
        int odd_count = 0;
        for (int i = 0; i < 26; i++) {
            if (freq[i] % 2 == 1) {
                freq_odd[i]++;
                odd_count++;
            }
        }
        if (odd_count > 4) {
            System.out.println("Impossible");
            return;
        }
        int[] row1 = new int[13];
        int[] row2 = new int[13];
        int idx = 0;
        for (int i = 0; i < 26; i++) {
            if (freq[i] <= 1) continue;
            if (freq[i] % 2 == 0) {
                while (freq[i] > 0) {
                    row1[idx++] = i;
                    row2[idx++] = i;
                    freq[i] -= 2;
                }
            } else {
                freq[i]--;
                row1[idx++] = i;
            }
        }
        for (int i = 0; i < 26; i++) {
            while (freq_odd[i] > 0) {
                row2[idx++] = i;
                freq_odd[i]--;
            }
        }
        for (int i = 0; i < 26; i++) {
            if (freq[i] > 0) {
                row1[idx++] = i;
            }
        }
        char[] ans1 = new char[13];
        char[] ans2 = new char[13];
        for (int i = 0; i < 13; i++) {
            ans1[i] = (char) (row1[i] + 'A');
            ans2[i] = (char) (row2[i] + 'A');
        }
        System.out.println(new String(ans1));
        System.out.println(new String(ans2));
    }
}
