import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++)
                a[i] = in.nextInt();
            int min = Integer.MAX_VALUE, max = Integer.MIN_VALUE, minIdx = -1, maxIdx = -1;
            for (int i = 0; i < n; i++) {
                if (a[i] < min) {
                    min = a[i];
                    minIdx = i;
                }
                if (a[i] > max) {
                    max = a[i];
                    maxIdx = i;
                }
            }
            if (max - min >= n)
                System.out.println("YES\n" + (minIdx + 1) + " " + (maxIdx + 1));
            else
                System.out.println("NO");
        }
    }
}
