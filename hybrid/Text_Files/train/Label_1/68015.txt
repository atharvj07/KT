import sys

t = int(sys.stdin.readline())
for _ in range(t):
    n = int(sys.stdin.readline())
    arr = [int(x) for x in sys.stdin.readline().split()]
    if len(set(arr)) == 1:
        print("YES")
    elif len(set(arr)) == 2:
        if arr.count(arr[0]) == 1 or arr.count(arr[0]) == n-1:
            print("YES")
        else:
            print("NO")
    else:
        print("NO")
