def is_power_of_two(n):
    return (n & (n - 1) == 0) and n != 0

def min_elements_to_remove(n, arr):
    max_val = 2 ** 31
    power_of_twos = [2 ** i for i in range(31)]
    count = {x: 0 for x in arr}
    for x in arr:
        count[x] += 1
    arr.sort()
    to_remove = n
    
    for i in range(n):
        for p in power_of_twos:
            needed = p - arr[i]
            if needed in count and (needed != arr[i] or count[needed] > 1):
                to_remove = min(to_remove, n - count[arr[i]] - count[needed])
                break
    
    return to_remove

n = int(input().strip())
arr = list(map(int, input().strip().split()))
print(min_elements_to_remove(n, arr))
