#include <bits/stdc++.h>
using namespace std;
int b[100][100], v[100], e, n;
int f(int k) {
    if (k == n) {
        for (int i = 1; i <= e; i++) {
            if (!v[i]) {
                return 0;
            }
        }
        return 1;
    }
    int ans = 0;
    ans |= f(k + 1);
    for (int j = 1; j <= b[k][0]; j++) {
        v[b[k][j]] = 1;
    }
    ans |= f(k + 1);
    for (int j = 1; j <= b[k][0]; j++) {
        v[b[k][j]] = 0;
    }
    return ans;
}
int main() {
    cin >> n >> e;
    for (int i = 1; i <= n; i++) {
        cin >> b[i][0];
        for (int j = 1; j <= b[i][0]; j++) {
            cin >> b[i][j];
        }
    }
    for (int i = 1; i <= e; i++) {
        v[i] = 0;
    }
    if (f(1))
        cout << "YES";
    else
        cout << "NO";
}
