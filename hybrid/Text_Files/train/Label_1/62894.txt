t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().strip().split())
    h = list(map(int, input().strip().split()))
    min_h, max_h = h[0], h[0] + k
    possible = True
    for i in range(1, n):
        new_min_h = max(h[i], min_h - k + 1)
        new_max_h = min(h[i] + k, max_h + k - 1)
        if new_min_h > new_max_h:
            possible = False
            break
        min_h, max_h = new_min_h, new_max_h
    if possible and max_h >= h[-1] and min_h <= h[-1] + k - 1:
        print("YES")
    else:
        print("NO")
