def solve_sweet_matrix(n, m, k):
    print(k)  # The minimum penalty is always equal to k
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if (i + j - 2) < k:
                path = []
                for x in range(1, i + 1):
                    path.append(f"({x},1)")
                for y in range(2, j + 1):
                    path.append(f"({i},{y})")
                print(" ".join(path))
            else:
                return

n, m, k = map(int, input().split())
solve_sweet_matrix(n, m, k)
