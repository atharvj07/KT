#include <iostream>
typedef long long ll;
ll n, k;
int main() {
    int t;
    std::cin >> t;
    while (t--) {
        std::cin >> n >> k;
        ll div = 0;
        while (n % k == 0) { n /= k; div++; }
        if (n == 0) std::cout << div << std::endl;
        else if (n == 1) std::cout << div + 1 << std::endl;
        else if (n < k) std::cout << div + n - 1 << std::endl;
        else if (n > k) std::cout << div + n / k + n % k - 1 << std::endl;
    }
    return 0;
}
