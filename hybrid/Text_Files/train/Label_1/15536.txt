#include <iostream>
using namespace std;
int main() {
    long long n, ans;
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        while (n % 2 == 0) n /= 2;
        while (n % 3 == 0) n /= 3;
        if (n != 1) ans = 0;
        else ans = n;
        cout << ans << endl;
    }
    return 0;
}
