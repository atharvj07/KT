n = int(input())
a = list(map(int, input().split()))

last_seen = {}
segments = []
start = 0

for i in range(n):
    if a[i] in last_seen:
        if start < last_seen[a[i]]:
            start = last_seen[a[i]]
        segments.append((start + 1, i + 1))
        start = i + 1
    last_seen[a[i]] = i

if not segments:
    print(-1)
else:
    # Adjust the last segment to include all remaining pearls
    if segments[-1][1] != n:
        segments[-1] = (segments[-1][0], n)
    print(len(segments))
    for segment in segments:
        print(segment[0], segment[1])
