#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    int a, b;
    cin >> a >> b;
    v[i] = {a, b};
  }
  int sum = 0, ans = -1;
  for (int i = 0; i < n; i++) {
    if (i != 0) {
      if (abs(v[i - 1].first - v[i].first) > 1) {
        ans = -1;
        break;
      }
    }
    if (abs(v[i].first - v[i].second) <= 1) {
      v[i].second = 0;
    } else {
      v[i].second = v[i].second - abs(v[i].first - v[i].second) + 1;
      sum += v[i].second;
    }
    v[i].first = v[i].first + v[i].second;
  }
  if (ans == -1) {
    cout << -1 << endl;
    return 0;
  }
  cout << sum << endl;
  for (int i = 0; i < n; i++) { cout << v[i].first << " "; }
  cout << endl;
  return 0;
}
