import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int d = sc.nextInt();
        int m = sc.nextInt();

        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = a + i * d;
        }

        for (int i = 0; i < m; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt() - 1;
            int z = sc.nextInt() - 1;
            if (x == 0) {
                reverse(arr, y, z);
            } else if (x == 1) {
                increment(arr, y, z);
            } else {
                halve(arr, y, z);
            }
        }

        int k = sc.nextInt() - 1;
        System.out.println(arr[k]);
    }

    static void reverse(int[] arr, int y, int z) {
        for (int i = y, j = z; i < j; i++, j--) {
            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }
    }

    static void increment(int[] arr, int y, int z) {
        for (int i = y; i <= z; i++) {
            arr[i]++;
        }
    }

    static void halve(int[] arr, int y, int z) {
        for (int i = y; i <= z; i++) {
            arr[i] /= 2;
        }
    }
}
