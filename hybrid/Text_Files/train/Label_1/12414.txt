#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < n; i++)
typedef long long int ll;
typedef unsigned long long int ull;
#define pb push_back
typedef vector<int> vi;
typedef pair<int, int> pi;
typedef map<int, int> mi;
#define vvi vector<vi>
typedef vector<ll> vll;
typedef pair<ll, ll> pll;
typedef map<ll, ll> mll;

int n, used[1001], mat[21][21];
vi v;

bool check() {
    rep(i, n) {
        for (int j = 0; j < n / 2; j++)
            if (mat[i][j] != mat[i][n - 1 - j]) return false;
    }
    rep(i, n) {
        for (int j = 0; j < n / 2; j++)
            if (mat[j][i] != mat[n - 1 - j][i]) return false;
    }
    return true;
}

void f(int x) {
    if (x == v.size()) {
        if (check()) {
            cout << "YES" << endl;
            rep(i, n) {
                rep(j, n) { cout << mat[i][j] << " "; }
                cout << endl;
            }
            exit(0);
        }
        return;
    }
    rep(i, n) {
        rep(j, n) {
            if (i == n / 2 && j == n / 2) continue;
            if (i + n / 2 < n && i == j) continue;
            if (j + n / 2 < n && i == n - 1 - j) continue;
            if (used[v[x]]) continue;
            used[v[x]] = 1;
            mat[i][j] = v[x];
            f(x + 1);
            mat[i][j] = 0;
            used[v[x]] = 0;
        }
    }
}

int main() {
    cin >> n;
    rep(i, n * n) {
        int x;
        cin >> x;
        v.pb(x);
    }
    f(0);
    cout << "NO" << endl;
    return 0;
}
