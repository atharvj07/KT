import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long a = sc.nextLong();
            long b = sc.nextLong();
            int k = sc.nextInt();
            if (a == b) {
                System.out.println("YES");
            } else {
                if (a > b) {
                    if (b * k < a)
                        System.out.println("NO");
                    else
                        System.out.println("YES");
                } else {
                    if (a * k < b)
                        System.out.println("NO");
                    else
                        System.out.println("YES");
                }
            }
        }
        sc.close();
    }
}
