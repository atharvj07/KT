import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n - 1];
        int[] b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = in.nextInt();
            b[i] = in.nextInt();
        }
        int[] t = new int[n];
        t[0] = a[0];
        for (int i = 1; i < n - 1; i++) {
            t[i] = (a[i] | t[i - 1]) & (b[i] | t[i - 1]);
        }
        t[n - 1] = (a[n - 2] | t[n - 2]) & (b[n - 2] | t[n - 2]);
        if (t[n - 1] == a[n - 2] && t[n - 1] == b[n - 2]) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                System.out.print(t[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("NO");
        }
    }
}
