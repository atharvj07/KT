def is_integer_number(word):
    return word.isdigit() and (len(word) == 1 or word[0] != '0')

s = input()

words = s.replace(';', ',').split(',')
a, b = [], []

for word in words:
    if is_integer_number(word):
        a.append(word)
    else:
        b.append(word)

a_string = ",".join(a) if a else "-"
b_string = ",".join(b) if b else "-"

print(f'"{a_string}"')
print(f'"{b_string}"')
