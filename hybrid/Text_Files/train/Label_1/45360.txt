import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }

        long g = a[0];
        for (int i = 1; i < n; i++) {
            g = gcd(g, a[i]);
        }

        long ans = 0;
        for (long i = 1; i <= m; i++) {
            if (i % g == 0) {
                ans++;
            }
        }

        System.out.println(ans);
    }

    private static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }

        return gcd(b, a % b);
    }
}
