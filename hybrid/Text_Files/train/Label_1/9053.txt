#include<bits/stdc++.h>
using namespace std ;
#define int long long
const int mod = 998244353 ;
const int N = 1e5 + 10 ;
int solve(vector<int>& a, int n, int k, int len){
    int dp[2][N][k + 1] ; // dp[isLastEqual][idx][last]
    memset(dp, 0, sizeof dp) ;
    dp[0][1][0] = 1 ;
    for(int i=1; i<=n; i++){
        for(int j=0; j<k+1; j++){
            dp[0][i][j] = (dp[0][i - 1][j] + dp[1][i - 1][j]) % mod ;
            if(a[i - 1] != -1){
                dp[0][i][a[i - 1]] = (dp[0][i - 1][j] + dp[1][i - 1][j]) % mod ;
                dp[1][i][a[i - 1]] = dp[0][i][a[i - 1]] ;
            }
            else{
                for(int l=1; l<=k; l++){
                    if(l != j && ((i - len <= 0) || (a[i - len - 1] != l))){
                        dp[1][i][l] = (dp[0][i - 1][l] + dp[1][i - 1][l] + dp[1][i - 1][j]) % mod ;
                    }
                }
            }
        }
    }
    return dp[1][n][0] ;
}
int32_t main(){
    int n, k, len ;
    cin >> n >> k >> len ;
    vector<int> a(n) ;
    for(int i=0; i<n; i++) cin >> a[i] ;
    cout << solve(a, n, k, len) << endl ;
    return 0 ;
}
