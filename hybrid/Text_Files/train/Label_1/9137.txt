#include<bits/stdc++.h>
using namespace std;
int main() {
    long long n,t,sum=0;
    cin>>n>>t;
    vector<long long> v;
    for(long long i=0; i<n; i++) {
        long long x; cin>>x; v.push_back(x);
    }
    for(long long i=0; i<n; i++) {
        if(t>=v[i]) { sum++; t-=v[i]; }
        else break;
    }
    cout<<sum<<endl;
    return 0;
}
