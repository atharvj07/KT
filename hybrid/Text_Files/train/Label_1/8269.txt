#include <bits/stdc++.h>
#define int long long
using namespace std;

const int MAXN = 1e6 + 10, MAXM = 2020;
char v[MAXM][MAXM];
int dx[] = {0, 0, 1, -1}, dy[] = {1, -1, 0, 0};
int n, m, a[MAXN];
stack<pair<int, int>> s;

int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= m; j++) { cin >> v[i][j]; }
    }
    memset(a, 0, sizeof(a));
    char cur = 'a';
    bool ok = 1;
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= m; j++) {
        if (!v[i][j]) continue;
        if (v[i][j] != cur) ok = 0;
        char prv = v[i][j];
        int sum = 0;
        for (int k = 0; k < 4; k++) {
          int ni = i + dx[k], nj = j + dy[k];
          if (ni >= 1 && ni <= n && nj >= 1 && nj <= m && v[ni][nj] && v[ni][nj] == prv)
            sum++;
        }
        a[v[i][j] - 'a'] = sum;
      }
      cur++;
    }
    if (!ok)
      cout << "NO\n";
    else {
      cout << "YES\n";
      int last = 'z';
      for (int i = 'a'; i <= 'z'; i++) {
        int ans = a[i - 'a'];
        while (ans--) {
          if (last == 'z') {
            cout << 1 << " " << 1 << " " << 1 << " " << 1 << "\n";
            s.push({1, 1});
            last = i;
          } else {
            auto p = s.top();
            s.pop();
            int x = p.first, y = p.second;
            if (v[x][y] == i) {
              if (x + 1 <= n && v[x + 1][y] == i) {
                cout << x << " " << y << " " << x + 1 << " " << y << "\n";
                s.push({x + 1, y});
              } else {
                cout << x << " " << y << " " << x << " " << y + 1 << "\n";
                s.push({x, y + 1});
              }
              last = i;
            } else {
              if (v[x + 1][y] == last) {
                cout << x << " " << y << " " << x + 1 << " " << y << "\n";
                s.push({x + 1, y});
              } else {
                cout << x << " " << y << " " << x << " " << y + 1 << "\n";
                s.push({x, y + 1});
              }
            }
          }
        }
      }
    }
  }
  return 0;
}
