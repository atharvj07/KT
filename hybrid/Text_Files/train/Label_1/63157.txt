import sys
input = sys.stdin.readline

t = int(input())

for _ in range(t):
    n, m, k = map(int, input().split())
    X = list(map(int, input().split()))
    Y = list(map(int, input().split()))
    
    P = sorted([(x, 'X') for x in X] + [(y, 'Y') for y in Y])
    ans = 0
    
    for p in P:
        x, px = p
        if px == 'X':
            i, j = X.index(x), -1
            while j + 1 < m and Y[j + 1] <= x:
                j += 1
            ans += j + (1 + i) * (k - 1 - i - j)
        else:
            i, j = -1, Y.index(x)
            while i + 1 < n and X[i + 1] <= x:
                i += 1
            ans += i + (1 + j) * (k - 1 - i - j)
    print(ans)
