import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static class Senator implements Comparable<Senator> {
        int level;
        int loyalty;

        public Senator(int level, int loyalty) {
            this.level = level;
            this.loyalty = loyalty;
        }

        @Override
        public int compareTo(Senator o) {
            return Integer.compare(this.level, o.level);
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int A = scanner.nextInt();
        Senator[] senators = new Senator[n];
        for (int i = 0; i < n; i++) {
            int level = scanner.nextInt();
            int loyalty = scanner.nextInt();
            senators[i] = new Senator(level, loyalty);
        }
        Arrays.sort(senators);
        double probability = 0;
        double loyalty = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (loyalty + senators[i].loyalty * 1.0 / 100.0 > 50) {
                probability = 1.0;
                break;
            }
            if (k > 0) {
                k--;
                loyalty += 10;
            }
        }
        if (probability < 1.0) {
            double B = 0;
            for (int i = 0; i < n; i++) {
                if (senators[i].loyalty < 50) {
                    B += senators[i].level;
                }
            }
            probability = A * 1.0 / (A + B);
        }
        System.out.printf("%.10f", probability);
    }
}
