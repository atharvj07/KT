n, m, k = map(int, input().strip().split())
sensors = [[int(j) for j in input().strip().split()] for i in range(k)]
lasty = [0] * (n + 1)
ans = [-1] * k
queue = []
queue.append((0, 0, 0))
lasty[0] = m + 1
while queue:
    x, y, t = queue.pop(0)
    for i in range(k):
        if x == sensors[i][0] and y == sensors[i][1]:
            ans[i] = t + 1
    if lasty[x + 1] >= y + 1:
        queue.append((x + 1, y, t + 1))
        lasty[x + 1] = y
    if x + 1 == n and y + 1 <= m:
        queue.append((x, y + 1, t + 1))
        lasty[x] = y + 1
        y = y + 1
    if lasty[x - 1] >= y + 1:
        queue.append((x - 1, y, t + 1))
        lasty[x - 1] = y
    if x - 1 == 0 and y + 1 <= m:
        queue.append((x, y + 1, t + 1))
        lasty[x] = y + 1
        y = y + 1
print(*ans)
