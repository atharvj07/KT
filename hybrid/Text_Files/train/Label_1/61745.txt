n = int(input())
a = list(map(int, input().split()))

from collections import Counter

topics = Counter(a)
max_problems = 0

for start in topics.values():
    contests = 0
    problems = start
    while problems:
        contests += problems
        problems //= 2
        if problems not in topics or topics[problems] < problems:
            break
    max_problems = max(max_problems, contests)

print(max_problems)
