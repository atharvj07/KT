#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, a, b;
  cin >> n >> a >> b;
  int one = 0, two = 0;
  int ans = 0;
  while (n--) {
    int t;
    cin >> t;
    if (t == 1) {
      if (a)
        --a;
      else if (two)
        --two;
      else
        ++ans;
    } else {
      if (b)
        --b;
      else

    }
  }
  cout << 2 * ans << endl;
  return 0;
}
