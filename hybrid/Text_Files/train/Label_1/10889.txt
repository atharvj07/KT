#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;

ll n, s, a[100005];
vector<pair<ll, ll>> v;

int main() {
  cin >> n >> s;
  for (ll i = 1; i <= n; i++) { cin >> a[i]; v.push_back({a[i], i}); }
  sort(v.begin(), v.end());
  ll ans = 0, cost = 0;
  for (auto x : v) {
    if (cost + x.second * x.first <= s) {
      cost += x.second * x.first;
      ans++;
    } else {
      break;
    }
  }
  cout << ans << ' ' << cost << endl;
  return 0;
}
