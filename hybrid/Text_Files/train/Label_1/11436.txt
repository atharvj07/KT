#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll f[1000000];
int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n;
        cin >> n;
        ll s = 1;
        ll ans = n;
        while (ans > s) {
            s *= 2;
            ans /= 2;
        }
        if (s - ans == ans) {
            cout << ans << endl;
        } else {
            cout << ans * 2 << endl;
        }
    }
    return 0;
}
