def mod(x):
    return x % 1000000007

n = int(input())
h = sorted([int(x) for x in input().split()], reverse=True)
cnt = 1
ans = 0
lst = 0
for x in h:
    if x != lst:
        ans = mod(ans + mod(cnt * (cnt + 1) // 2))
        cnt = 1
        lst = x
    else:
        cnt += 1
ans = mod(ans + mod(cnt * (cnt + 1) // 2))
print(ans)
