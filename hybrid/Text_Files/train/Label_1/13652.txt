#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int md = (int)1e9 + 7;
int main() {
    ll n, k;
    cin >> n >> k;
    vector<pair<ll, ll>> v(n);
    for (ll i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    sort(v.begin(), v.end());
    ll ansl = -1e9, ansr = -1e9, ans = 0;
    for (ll i = 0; i < k; i++) {
        if (v[i].first > ansr) {
            ansl = v[i].first;
            ansr = v[i].second;
            ans++;
        } else
            ansr = max(ansr, v[i].second);
    }
    cout << ans << '\n';
    for (ll i = 0; i < k; i++) {
        if (v[i].first <= ansl && v[i].second >= ansr) cout << i + 1 << ' ';
    }
    cout << '\n';
    return 0;
}
