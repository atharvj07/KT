#include <bits/stdc++.h>

using namespace std;

using ll = long long;
using ld = long double;

int main() {
  ll n;
  cin >> n;
  vector<array<ll, 5>> a(n);
  for (auto& aa : a) {
    for (auto& aaa : aa) { cin >> aaa; }
  }
  vector<ll> ans;
  for (ll i = 0; i < n; i++) {
    bool f = false;
    for (ll j = 0; j < n; j++) {
      if (i == j) { continue; }
      ld cos;
      ld mn = 0;
      for (ll k = 0; k < 5; k++) {
        mn += a[i][k] * a[j][k];
      }
      ld l1 = 0, l2 = 0;
      for (ll k = 0; k < 5; k++) {
        l1 += a[i][k] * a[i][k];
        l2 += a[j][k] * a[j][k];
      }
      cos = mn / (sqrt(l1) * sqrt(l2));
      if (cos > 0) {
        f = true;
        break;
      }
    }
    if (!f) { ans.push_back(i + 1); }
  }
  cout << ans.size() << endl;
  for (auto& aa : ans) { cout << aa << endl; }
}
