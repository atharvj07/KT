import java.util.*;

class Main {

    static class DSU {
        int[] parent;
        int[] rank;

        DSU(int n) {
            parent = new int[n + 1];
            rank = new int[n + 1];
            for (int i = 0; i <= n; i++) {
                parent[i] = i;
            }
        }

        int find(int x) {
            if (parent[x] != x) {
                parent[x] = find(parent[x]);
            }
            return parent[x];
        }

        void union(int a, int b) {
            int rootA = find(a);
            int rootB = find(b);
            if (rootA != rootB) {
                if (rank[rootA] > rank[rootB]) {
                    parent[rootB] = rootA;
                } else if (rank[rootA] < rank[rootB]) {
                    parent[rootA] = rootB;
                } else {
                    parent[rootB] = rootA;
                    rank[rootA]++;
                }
            }
        }
    }

    static class Edge {
        int a;
        int b;
        int weight;

        Edge(int a, int b, int weight) {
            this.a = a;
            this.b = b;
            this.weight = weight;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int[] special = new int[k];
        for (int i = 0; i < k; i++) {
            special[i] = sc.nextInt();
        }
        DSU dsu = new DSU(n);
        Set<Integer> specialSet = new HashSet<>();
        for (int i : special) {
            specialSet.add(i);
        }
        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            edges.add(new Edge(a, b, 1));
            if (specialSet.contains(a) && specialSet.contains(b)) {
                dsu.union(a, b);
            }
        }
        int maxDist = 0;
        for (Edge edge : edges) {
            int rootA = dsu.find(edge.a);
            int rootB = dsu.find(edge.b);
            if (rootA != rootB) {
                dsu.union(rootA, rootB);
                maxDist = Math.max(maxDist, edge.weight);
            }
        }
        System.out.println(maxDist);
    }
}
