s, b = map(int, input().split())
a = list(map(int, input().split()))
L = []
for _ in range(b):
    d, g = map(int, input().split())
    L.append((d, g))
L.sort(key=lambda x: x[0])
a.sort()
ans = []
taken = [False for _ in range(b)]
for x in a:
    ans.append(0)
    for i in range(b):
        if not taken[i] and L[i][0] <= x:
            ans[-1] += L[i][1]
            taken[i] = True
print(' '.join(map(str, ans)))
