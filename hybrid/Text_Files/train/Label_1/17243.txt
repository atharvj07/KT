#include <iostream>
#include <vector>
using namespace std;
#define ll long long
const ll INF = 1e18;
int main() {
  ll b1, q, l, m;
  cin >> b1 >> q >> l >> m;
  vector<ll> a(m);
  for (int i = 0; i < m; i++) cin >> a[i];
  ll ans = 0, lst = b1;
  while (abs(lst) <= l) {
    bool ok = 1;
    for (auto ai : a) if (ai == lst) ok = 0;
    if (ok) ans++;
    lst = lst * q;
  }
  if (ans == INF)
    cout << "inf" << endl;
  else
    cout << ans << endl;
  return 0;
}
