import java.util.Scanner;

public class ArtemAndMasha {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int maxPresents = 1;
        int currentCount = 1;

        for (int i = 2; i <= n; i++) {
            if (i % 2 == 0) {
                currentCount++;
            } else {
                currentCount = 1;
            }
            maxPresents = Math.max(maxPresents, currentCount);
        }

        System.out.println(maxPresents);
    }
}
