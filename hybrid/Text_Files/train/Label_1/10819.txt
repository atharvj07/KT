#include <bits/stdc++.h>
using namespace std;
struct node { int x, y; };
int main() {
    int n, m, cnt, u, v, w[100006], w2[100006];
    vector<node> food;
    queue<int> q;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { cin >> w[i]; w2[i] = w[i]; }
    for (int i = 1; i <= m; i++) {
        cin >> u >> v;
        for (int j = 0; j < food.size(); j++) {
            if (u == food[j].x && v == food[j].y) { cnt--; } else if (u == food[j].y && v == food[j].x)
                cnt--;
        }
        node p;
        p.x = u;
        p.y = v;
        food.push_back(p);
    }
    cnt = food.size();
    for (int i = 0; i < food.size(); i++) {
        u = food[i].x;
        v = food[i].y;
        if (w[u] == 0 || w[v] == 0) { cnt--; continue; }
        w[u]--; w[v]--;
        q.push(u); q.push(v);
        if (w[u] == 0) { q.push(u); }
        if (w[v] == 0) q.push(v);
    }
    if (cnt > 0) { cout << "DEAD" << endl; } else {
        cout << "ALIVE" << endl;
        while (!q.empty()) {
            cout << q.front() << ' ';
            q.pop();
        }
        for (int i = 1; i <= n; i++)
            if (w2[i] > 0) { cout << i << ' '; }
        cout << endl;
    }
    return 0;
}
