n, m, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
ans, cur = 0, 0
for x in b:
    i = next((j for j in range(n) if a[j] == x), -1)
    if i > cur:
        for j in range(cur // k, i // k):
            ans += k
        cur = i
    else:
        ans -= 1
    if i > 0:
        a[i], a[i - 1] = a[i - 1], a[i]
    ans += 1
print(ans)
