#include <iostream>
using namespace std;

int main() {
  string s;
  cin >> s;
  int h = 0, e = 0, i = 0, d = 0;
  for (char c : s) {
    if (c == 'h') h++;
    if (c == 'e' && h > 0) e++;
    if (c == 'i' && e > 0) i++;
    if (c == 'd' && i > 0) d++;
  }
  if (d > 0)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
}
