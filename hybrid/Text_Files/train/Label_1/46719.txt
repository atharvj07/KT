import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    while (t-- > 0) {
      int n = sc.nextInt(), x = sc.nextInt();
      int[] a = new int[n];
      for (int i = 0; i < n; i++) {
        a[i] = sc.nextInt();
      }
      Arrays.sort(a);
      int ans = 0;
      int[] c = new int[n];
      for (int i = 0; i < n; i++) {
        if (a[i] == x) {
          ans++;
          c[i] = 1;
        } else {
          c[i] = -1;
        }
      }
      for (int i = n - 2; i >= 0; i--) {
        c[i] += c[i + 1];
      }
      for (int i = 0; i < n; i++) {
        if (c[i] == ans) {
          ans++;
          break;
        }
      }
      System.out.println(ans);
    }
  }
}
