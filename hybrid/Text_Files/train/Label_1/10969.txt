#include <bits/stdc++.h>
using namespace std;

struct Person {
    int x, y;
    vector<int> adj;
};

int main() {
    int n, m;
    cin >> n >> m;
    vector<Person> people(n);
    for (auto &p : people) { cin >> p.x >> p.y; }
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        --u, --v;
        people[u].adj.push_back(v);
        people[v].adj.push_back(u);
    }
    vector<bool> vis(n, false);
    vector<int> low(n), num(n), par(n, -1);
    vector<vector<int>> components;
    int timer = 0;
    function<void(int)> dfs = [&](int u) -> void {
        vis[u] = true;
        num[u] = low[u] = timer++;
        for (auto v : people[u].adj) {
            if (!vis[v]) {
                par[v] = u;
                dfs(v);
                low[u] = min(low[u], low[v]);
            } else if (v != par[u])
                low[u] = min(low[u], num[v]);
        }
        if (num[u] == low[u]) {
            components.push_back(vector<int>());
            int v = u;
            while (~v) {
                components.back().push_back(v);
                v = par[v];
            }
        }
    };
    for (int i = 0; i < n; ++i)
        if (!vis[i]) dfs(i);
    vector<int> sum(n, 0);
    for (auto &c : components) {
        int minx = INT_MAX, miny = INT_MAX;
        for (auto i : c) {
            minx = min(minx, people[i].x);
            miny = min(miny, people[i].y);
        }
        for (auto i : c) sum[i] += minx + miny;
    }
    for (auto s : sum) cout << s << " ";
}
