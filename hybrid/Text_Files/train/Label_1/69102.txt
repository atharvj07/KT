t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    b.reverse()
    answer = True
    for i in range(n):
        if a[i] + b[i] > x:
            answer = False
            break
    print("Yes" if answer else "No")
