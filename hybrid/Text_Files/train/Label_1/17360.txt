#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &ai : a) cin >> ai;
  vector<int> b(n + 1);
  for (int ai : a) b[ai]++;
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (b[i]) continue;
    ans++;
    for (int j = 1; j <= n; j++) {
      if (b[j] > 1) {
        a[find(a.begin(), a.end(), j) - a.begin()] = i;
        b[j]--;
        b[i]++;
        break;
      }
    }
  }
  cout << ans << '\n';
  for (int ai : a) cout << ai << ' ';
  cout << '\n';
  return 0;
}
