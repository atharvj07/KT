import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), t = sc.nextInt(), k = 0;
        char[] ch = sc.next().toCharArray();
        for (char c : ch)
            if (c == 'H')
                k++;
        if (n - k > t || k > t)
            System.out.println(-1);
        else
            System.out.println(k - (t - (n - k)));
    }
}
