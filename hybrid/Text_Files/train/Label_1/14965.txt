#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        long long int n,m,a,b;
        cin>>n>>m>>a>>b;
        vector<long long int> v(m);
        for(int i=0;i<m;i++)
        {
            cin>>v[i];
        }
        sort(v.begin(),v.end());
        long long int ans=0;
        long long int dif=abs(a-b);
        for(auto i:v)
        {
            if(i<=dif)
            {
                ans++;
                dif-=i;
            }
            else
            {
                break;
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
