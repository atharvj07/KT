n, m = map(int, input().split())
songs = [list(map(int, input().split())) for _ in range(n)]
moments = list(map(int, input().split()))

cumulative_durations = []
duration_so_far = 0

for c, t in songs:
    duration_so_far += c * t
    cumulative_durations.append(duration_so_far)

for moment in moments:
    low, high = 0, len(cumulative_durations) - 1
    while low < high:
        mid = (low + high) // 2
        if cumulative_durations[mid] < moment:
            low = mid + 1
        else:
            high = mid
    print(low + 1)
