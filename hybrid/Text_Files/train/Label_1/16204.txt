#include <iostream>
using namespace std;

typedef long long ll;

int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll l, v, a, b;
    cin >> l >> v >> a >> b;
    if (a <= v && b >= v)
      cout << 0 << endl;
    else if (a <= v)
      cout << (l - (b - v)) / v + (a < v) << endl;
    else
      cout << (l - (a - 1)) / v << endl;
  }
}
