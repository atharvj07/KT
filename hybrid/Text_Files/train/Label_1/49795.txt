import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int[] cnt = new int[3];
        for (char c : s.toCharArray()) {
            cnt[c - 'a']++;
        }
        if (cnt[0] % 2 == 1 || cnt[1] % 2 == 1 || cnt[2] % 2 == 1) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
        }
    }
}
