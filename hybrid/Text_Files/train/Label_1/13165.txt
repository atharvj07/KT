#include<bits/stdc++.h>
typedef long long ll;
using namespace std;

int main() {
    int T;
    cin >> T;
    while (T--) {
        int n, x;
        cin >> n >> x;
        string s;
        cin >> s;
        vector<ll> pref(n, s[0] - 48);
        for (int i = 1; i < n; i++) pref[i] = pref[i - 1] + s[i] - 48;
        ll add = (x + pref[n - 1]) / n;
        ll ans = 0;
        for (int i = 0; i < n; i++) {
            ll balance = pref[i] + add * ((n - i + n - 1) / n);
            if (balance >= x) ans++;
        }
        if (ans >= ll(1e10)) cout << -1 << endl;
        else cout << ans << endl;
    }
}
