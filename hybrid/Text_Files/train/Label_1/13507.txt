#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

typedef long long int lli;

int main() {
    int n;
    cin >> n;
    lli sum = 0, jiro = 0;
    for (int i = 0; i < n; i++) {
        int sz;
        cin >> sz;
        vector<lli> v;
        lli t;
        for (int j = 0; j < sz; j++) {
            cin >> t;
            v.push_back(t);
        }
        sort(v.begin(), v.end());
        for (int j = 0; j < (sz + 1) / 2; j++) {
            sum += v[j];
            if (sz % 2 == 1 && j == (sz + 1) / 2 - 1) continue;
            jiro += v[sz - j - 1];
        }
    }
    cout << sum << " " << jiro << endl;
    return 0;
}
