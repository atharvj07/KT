import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[][] arr = new int[n][n];
        for(int i = 0; i<n; i++){
            String[] input = br.readLine().split(" ");
            for(int j = 0; j<n; j++){
                arr[i][j] = Integer.parseInt(input[j]);
            }
        }
        int[] ans = new int[n];
        boolean[] vis = new boolean[n+1];
        for(int i = 0; i<n; i++){
            int idx = 1;
            while(vis[idx]) idx++;
            for(int j = 0; j<n; j++){
                if(arr[i][j] == 0) continue;
                if(arr[i][j] == arr[j][i]){
                    idx = Math.max(idx, arr[i][j]);
                }
            }
            vis[idx] = true;
            ans[i] = idx;
        }
        for(int i: ans){
            System.out.print(i+" ");
        }
        System.out.println();
    }
}
