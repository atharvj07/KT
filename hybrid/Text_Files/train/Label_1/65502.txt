n = int(input())
s = [int(x) for x in input().split()]
c = [int(x) for x in input().split()]
t = [0] * n
for i in range(n - 1):
    if s[i] < s[i + 1]: t[i + 1] = i + 1
for i in range(n - 1, 0, -1):
    if t[i] != 0: t[t[i] - 1] = t[i]
ans = []
for i in range(n):
    if t[i] != 0 and (len(ans) < 3 or t[i] > ans[-1]):
        ans.append(t[i])
        if len(ans) == 3: break
if len(ans) == 3: print(sum([c[i] for i in ans]))
else: print(-1)
