import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(reader.readLine());
        Point[] pts = new Point[n];

        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            int x = Integer.parseInt(line[0]);
            int y = Integer.parseInt(line[1]);
            pts[i] = new Point(x, y);
        }

        int numVisible = 0;
        for (int i = 1; i < n - 1; i++) {
            boolean visible = true;

            for (int j = 0; j < n; j++) {
                if (j == i || j == i + 1)
                    continue;

                if (!isLineSegmentVisible(pts[i], pts[i + 1], pts[j])) {
                    visible = false;
                    break;
                }
            }

            if (visible) { numVisible++; }
        }

        System.out.println(numVisible);
    }

    static boolean isLineSegmentVisible(Point p1, Point p2, Point p3) {
        long crossProduct = (p2.x - p1.x) * (p3.y - p1.y) - (p2.y - p1.y) * (p3.x - p1.x);

        if (crossProduct == 0)
            return false;

        double r = (double) (p2.x - p1.x) / (p3.x - p1.x);
        if (r < 0 || r > 1)
            return false;

        double s = (double) (p2.y - p1.y) / (p3.y - p1.y);
        if (s < 0 || s > 1)
            return false;

        return true;
    }

    static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
