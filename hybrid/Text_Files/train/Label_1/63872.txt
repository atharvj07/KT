import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
cores = [[] for i in range(n)]
for i in range(n):
    cores[i] = [int(x) for x in input().split()]
locked = [False for i in range(n)]
locked_cell = [False for i in range(k+1)]
locked_cycle = [-1 for i in range(n)]
for cycle in range(m):
    for i in range(n):
        if locked[i]:
            continue
        cell = cores[i][cycle]
        if locked_cell[cell]:
            locked[i] = True
            locked_cycle[i] = cycle
        else:
            locked_cell[cell] = True
for i in range(n):
    print(locked_cycle[i])
