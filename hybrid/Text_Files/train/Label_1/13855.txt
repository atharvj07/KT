#include <cstdio>
#include <vector>
#include <algorithm>

int main() {
    int n;
    scanf("%d", &n);
    std::vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        scanf("%d", &a[i]);
    }
    std::sort(a.begin(), a.end());
    std::vector<int> d(n);
    for (int i = 1; i < n; ++i) {
        d[i] = a[i] - a[i - 1];
    }
    int ans = 0;
    std::vector<int> s(n);
    for (int i = 0; i < n; ++i) {
        for (int j = i; j < n - 1; ++j) {
            s[j + 1] = d[j];
            s[j] = -d[j];
        }
        std::sort(s.begin(), s.end());
        int tmp = 0;
        for (int j = 0; j <= i; ++j) {
            tmp -= s[j];
        }
        ans = std::max(ans, tmp);
    }
    for (int k = 1; k <= (n + 1) / 2; ++k) {
        printf("%d ", ans);
    }
    puts("");
    return 0;
}
