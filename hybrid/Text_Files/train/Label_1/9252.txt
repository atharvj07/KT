#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, a;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) {
        cin >> a;
        v[i] = a;
    }
    int m = (v[0] == 1 || (v[1] == 1 && v[2] == 2)) ? 3 : 7;
    if (n % m == 0) {
        cout << "Alex" << endl;
    } else {
        cout << "Petr" << endl;
    }
    return 0;
}
