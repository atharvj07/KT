def read_puzzle():
    return input().strip() + input().strip()

def can_be_same(puzzle1, puzzle2):
    for i in range(4):
        if puzzle1 == puzzle2:
            return "YES"
        puzzle1 = puzzle1[1:] + puzzle1[0]
    return "NO"

puzzle1 = read_puzzle()
puzzle2 = read_puzzle()
puzzle1 = puzzle1.replace('X', '')
puzzle2 = puzzle2.replace('X', '')
print(can_be_same(puzzle1, puzzle2))
