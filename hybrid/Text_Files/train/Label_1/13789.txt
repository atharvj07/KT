#include<bits/stdc++.h>
using namespace std;
const int N=2005,OO=1e9;
int go[4][2]={{-1,0},{1,0},{0,1},{0,-1}};
bool visited[N][N],valid[N][N];
char grid[N][N];
int n,m,r,c,x,y;
bool valid_cell(int i,int j)
{
    return i>=1&&i<=n&&j>=1&&j<=m&&grid[i][j]!='*'&&!visited[i][j];
}
void dfs(int i,int j,int left,int right)
{
    if(visited[i][j])return;
    visited[i][j]=1;
    valid[i][j]=1;
    for(int k=0;k<4;++k)
    {
        int ni=i+go[k][0],nj=j+go[k][1];
        if(valid_cell(ni,nj))
        {
            dfs(ni,nj,left,right);
        }
        if(k==1&&left>0)
        {
            dfs(i,j-1,left-1,right);
        }
        if(k==3&&right>0)
        {
            dfs(i,j+1,left,right-1);
        }
    }
}
int solve()
{
    dfs(r,c,x,y);
    int cnt=0;
    for(int i=1;i<=n;++i)
    {
        for(int j=1;j<=m;++j)
        {
            cnt+=valid[i][j];
        }
    }
    return cnt;
}
int main()
{
    cin>>n>>m;
    cin>>r>>c;
    cin>>x>>y;
    for(int i=1;i<=n;++i)
    {
        for(int j=1;j<=m;++j)
        {
            cin>>grid[i][j];
        }
    }
    cout<<solve()<<'\n';
    return 0;
}
