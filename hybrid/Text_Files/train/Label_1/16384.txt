#include <bits/stdc++.h>
#define mod 1000000007
using namespace std;
long long power(long long a, long long b) {
    if (b == 0) { return 1; }
    long long t = power(a, b / 2) % mod;
    t = (t * t) % mod;
    if (b & 1) { return (t * a) % mod; }
    return t;
}
int main() {
    long long x, k;
    cin >> x >> k;
    cout << (x * power(2, k) % mod + mod) % mod << endl;
    return 0;
}
