#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int d[n];
    for (int i = 0; i < n; i++) cin >> d[i];

    int l = 0, r = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '<') l += d[i];
        else r += d[i];
    }

    if (l > r) cout << "INFINITE" << endl;
    else cout << "FINITE" << endl;

    return 0;
}
