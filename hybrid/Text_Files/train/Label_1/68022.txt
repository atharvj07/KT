from collections import Counter

def solve_tanya_problem(s, t):
    s_count = Counter(s)
    t_count = Counter(t)
    yay = 0
    whoops = 0

    for char in s_count:
        common = min(s_count[char], t_count[char])
        yay += common
        s_count[char] -= common
        t_count[char] -= common

    for char in s_count:
        if char.islower():
            common = min(s_count[char], t_count[char.upper()])
        else:
            common = min(s_count[char], t_count[char.lower()])
        whoops += common
        s_count[char] -= common
        t_count[char] -= common

    return yay, whoops

s = input()
t = input()
result = solve_tanya_problem(s, t)
print(result[0], result[1])
