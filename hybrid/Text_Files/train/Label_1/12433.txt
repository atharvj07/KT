#include <iostream>
#include <cstdio>
#include <vector>
#include <algorithm>
#define ll long long
using namespace std;
const int N = 4e5 + 1;
vector< vector<int> >g(N);
ll a[N], ans[N], sz[N];
int dep[N], par[N], cnt;

void dfs(int u) {
	sz[u] = 1;
	int mx = -1;
	for (auto v : g[u]) {
		if (v == par[u])continue;
		par[v] = u;
		dep[v] = dep[u] + 1;
		dfs(v);
		sz[u] += sz[v];
		if (sz[v] > mx)mx = v;
	}
	if (mx != -1)swap(g[u][0], g[u][g[u].size() - 1]);
}

void modify(int u, ll p) {
	ans[u] += p;
	for (auto v : g[u]) {
		if (v == par[u])continue;
		ans[v] += -p;
		modify(v, -p);
	}
}

int main() {
	int n, m;
	cin >> n >> m;
	for (int i = 1; i <= n; i++)cin >> a[i];
	for (int i = 1; i<n; i++) {
		int u, v; cin >> u >> v;
		g[u].push_back(v);
		g[v].push_back(u);
	}
	dfs(1);
	for (int i = 0; i<m; i++) {
		int tp, x; ll p;
		cin >> tp >> x;
		if (tp == 1) {
			cin >> p;
			modify(x, p);
		}
		else {
			cout << ans[x] + a[x] << '\n';
		}
	}
	return 0;
}
