def mex(s):
    for i in range(len(s) + 1):
        if i not in s:
            return i

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    nums = list(map(int, input().strip().split()))
    nums.sort()
    A = set()
    B = set()
    for num in nums:
        if num not in A:
            A.add(num)
        else:
            B.add(num)
    print(mex(A) + mex(B))
