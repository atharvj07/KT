import java.util.*;

public class Main {

    static int[] root;
    static int[] rank;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] u = new int[m];
        int[] v = new int[m];
        int[] w = new int[m];
        for (int i = 0; i < m; i++) {
            u[i] = in.nextInt();
            v[i] = in.nextInt();
            w[i] = in.nextInt();
        }
        int q = in.nextInt();
        root = new int[n + 1];
        rank = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            root[i] = i;
            rank[i] = 0;
        }
        int[] ans = new int[q];
        for (int i = 0; i < q; i++) {
            int k = in.nextInt();
            Set<Integer> set = new HashSet<>();
            for (int j = 0; j < k; j++) {
                int e = in.nextInt();
                set.add(e);
            }
            int cnt = 0;
            for (int e : set) {
                int ru = find(u[e - 1]);
                int rv = find(v[e - 1]);
                if (ru != rv) {
                    cnt++;
                    union(ru, rv);
                }
            }
            ans[i] = (cnt == n - 1 && set.size() == k) ? 1 : 0;
        }
        for (int i = 0; i < q; i++) {
            System.out.println(ans[i] == 1 ? "YES" : "NO");
        }
    }

    static int find(int x) {
        if (root[x] == x) {
            return x;
        }
        return root[x] = find(root[x]);
    }

    static void union(int x, int y) {
        int rx = find(x);
        int ry = find(y);
        if (rx != ry) {
            if (rank[rx] < rank[ry]) {
                root[rx] = ry;
            } else if (rank[rx] > rank[ry]) {
                root[ry] = rx;
            } else {
                root[ry] = rx;
                rank[rx]++;
            }
        }
    }
}
