s = input().strip()
letter_counts = {}
for char in s:
    if char in letter_counts:
        letter_counts[char] += 1
    else:
        letter_counts[char] = 1

odd_counts = sum(1 for count in letter_counts.values() if count % 2 != 0)

if odd_counts <= 1 or odd_counts % 2 != 0:
    print("First")
else:
    print("Second")
