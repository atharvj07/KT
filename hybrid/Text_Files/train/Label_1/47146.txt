import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = Integer.parseInt(sc.nextLine());
        int[][] a = new int[N][4];
        for (int i = 0; i < N; i++) {
            String[] s = sc.nextLine().split(" ");
            a[i][0] = Integer.parseInt(s[0]) - 1;
            a[i][1] = Integer.parseInt(s[1]) - 1;
            a[i][2] = Integer.parseInt(s[2]);
            a[i][3] = Integer.parseInt(s[3]);
        }
        int[] d = new int[366];
        for (int[] e : a) {
            int st = e[0] * 30 + e[1];
            for (int i = 0; i < e[2]; i++) {
                d[st++] += e[3];
            }
        }
        int mi = Integer.MAX_VALUE;
        for (int i = 0; i < 366; i++) {
            mi = Math.min(mi, d[i]);
        }
        System.out.println(mi);
    }
}
