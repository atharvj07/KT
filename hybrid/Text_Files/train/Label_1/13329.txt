#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[100005][5];
int main() {
  ll n, m, x, y, z, p;
  cin >> n >> m >> x >> y >> z >> p;
  x %= 4, y %= 2, z %= 4;
  while (p--) {
    ll a1, b1;
    cin >> a1 >> b1;
    ll na, nb;
    na = a1, nb = b1;
    if (x) {
      swap(na, nb);
      na = n + 1 - na, nb = m + 1 - nb;
      if (x > 1) {
        na = n + 1 - na, nb = m + 1 - nb;
        if (x > 2) na = n + 1 - na, nb = m + 1 - nb;
      }
    }
    if (y) {
      swap(na, nb);
      na = n + 1 - na;
    }
    if (z) {
      swap(na, nb);
      na = n + 1 - na, nb = m + 1 - nb;
      if (z > 1) {
        swap(na, nb);
        na = n + 1 - na, nb = m + 1 - nb;
        if (z > 2) na = n + 1 - na, nb = m + 1 - nb;
      }
    }
    cout << na << " " << nb << endl;
  }
  return 0;
}
