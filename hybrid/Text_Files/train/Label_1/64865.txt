from sys import stdin
n = int(stdin.readline())
a = [0]*n
b = [0]*n
c = [0]*n
adj = [[]]
for i in range(n):
    a[i],b[i],c[i] = map(int, stdin.readline().split())
    adj.append([])
for i in range(n-1):
    u,v = map(int, stdin.readline().split())
    adj[u].append(v)
    adj[v].append(u)
def dfs(u, p):
    if b[u] == c[u]:
        return 0
    ans = []
    for v in adj[u]:
        if v == p:
            continue
        tmp = dfs(v, u)
        if tmp < 0:
            return -1
        ans.append(tmp)
    if b[u] == 0:
        ans.append(0)
    if len(ans) < 2:
        return -1
    ans.sort(reverse=True)
    x = ans[0]
    y = ans[1]
    return x+y+a[u]
print(dfs(1, 0))
