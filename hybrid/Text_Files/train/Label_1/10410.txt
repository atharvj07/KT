#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 105;
typedef pair<int, int> pi;

int n;
pi arr[N];

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i)
        cin >> arr[i].first >> arr[i].second;
    sort(arr + 1, arr + n + 1);
    int ans = 0;
    for (int i = n; i >= 1; --i) {
        if (arr[i].second > ans)
            ans += arr[i].first;
    }
    cout << ans << "\n";
}
