mod = 998244353

n = int(input())
a = list(map(int, input().split()))

dp = [[0] * 2 for _ in range(n + 1)]
dp[n][0] = dp[n][1] = 1

for i in range(n - 1, -1, -1):
    dp[i][0] = ((dp[i][0] + dp[i + 1][0] + dp[i + 1][1]) % mod) * a[i] % mod
    dp[i][1] = (dp[i][1] + dp[i + 1][1]) % mod

print((dp[0][0] - dp[0][1] + mod) % mod)
