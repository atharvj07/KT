#include <bits/stdc++.h>
#define eps 1e-9
using namespace std;
typedef long long ll;
int n;
int ar[305][2];
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> ar[i][0] >> ar[i][1];
  double ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      for (int k = j + 1; k < n; k++) {
        for (int l = k + 1; l < n; l++) {
          double ar1 =
              0.5 * ((ar[i][0] * ar[j][1] + ar[j][0] * ar[k][1] + ar[k][0] * ar[l][1] +
                      ar[l][0] * ar[i][1]) -
                     (ar[j][0] * ar[i][1] + ar[k][0] * ar[j][1] + ar[l][0] * ar[k][1] +
                      ar[i][0] * ar[l][1]));
          if (ar1 >= eps) ans = max(ans, ar1);
        }
      }
    }
  }
  cout << ans << "\n";
  return 0;
}
