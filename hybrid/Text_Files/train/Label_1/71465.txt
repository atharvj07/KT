import sys

def is_prime(num):
    if num <= 1:
        return False
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            return False
    return True

def solve():
    n, e = map(int, input().split())
    a = list(map(int, input().split()))
    ans = 0
    for i in range(n):
        prod = a[i]
        for j in range(i + e, n, e):
            prod *= a[j]
            if is_prime(prod):
                ans += 1
    return ans

t = int(input())
for _ in range(t):
    print(solve())
