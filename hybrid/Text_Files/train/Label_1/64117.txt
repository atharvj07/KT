from collections import defaultdict
from heapq import heappush, heappop

n = int(input())
ps = []
for i in range(n):
    ps.append(tuple(map(int, input().split())))
ps.sort()

hps = []
for i, p in enumerate(ps):
    for j in range(i + 1, n):
        q = ps[j];
        minx = min(p[0], q[0]);
        miny = min(p[1], q[1]);
        minz = min(p[2], q[2]);
        maxx = max(p[0], q[0]);
        maxy = max(p[1], q[1]);
        maxz = max(p[2], q[2]);
        cnt = 0
        for k in range(n):
            if k != i and k != j:
                r = ps[k]
                if minx <= r[0] <= maxx and miny <= r[1] <= maxy and minz <= r[2] <= maxz:
                    cnt = 1
        if not cnt:
            heappush(hps, (p + q, i, j))

hs = defaultdict(bool)

ans = []
while hps:
    t, a, b = heappop(hps)
    if not hs[a] and not hs[b]:
        ans.append((a + 1, b + 1))
        hs[a] = hs[b] = True

print("\n".join([" ".join(map(str, x)) for x in ans]))
