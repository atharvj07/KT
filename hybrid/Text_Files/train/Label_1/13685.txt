#include<bits/stdc++.h>
using namespace std;
int main(){
    int k,a,b,v,ans=0,tmp=0;
    cin>>k>>a>>b>>v;
    while(1){
        if(v-tmp>0 && tmp+v<=k){
            tmp+=v;
            ans++;
        }
        else if(v-tmp<=0){
            break;
        }
        else if(v-tmp>0 && tmp+v>k){
            ans++;
            tmp=v;
        }
    }
    if((k-tmp-v)%v==0) ans+=k/v;
    else ans+=(k/v)+1;
    cout<<ans<<endl;
}
