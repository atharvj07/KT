n, p, q, r = map(int, input().split())
a = list(map(int, input().split()))

max_p, max_pq, max_pqr = -float('inf'), -float('inf'), -float('inf')

for i in range(n):
    max_p = max(max_p, p * a[i])
    max_pq = max(max_pq, max_p + q * a[i])
    max_pqr = max(max_pqr, max_pq + r * a[i])

print(max_pqr)
