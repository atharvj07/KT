#include<bits/stdc++.h>
using namespace std;

const int maxn=26;
int num[maxn];
int main()
{
	string a, b, t;
	cin >> t >> b;
	a = t;
	for (int i = 0; i < a.size(); i++) num[a[i] - 'A']++, a.pop_back();
	for (int i = 0; i < b.size(); i++) num[b[i] - 'A']--;
	int sum = 0;
	for (int i = 0; i < maxn; i++) sum += num[i];
	if (sum) { cout << "Impossible"; return 0; }
	for (int i = 0; i < b.size(); i++) a += b[i];
	cout << a << endl;
    return 0;
}
