#include <bits/stdc++.h>
using namespace std;
#define fastio ios::sync_with_stdio(0), cin.tie(0), cout.tie(0)
#define int long long
#define endl "\n"
#define fr(i, a, b) for (int i = a; i <= b; i++)
#define rf(i, a, b) for (int i = a; i >= b; i--)

int32_t main() {
  fastio;
  int n, k, p;
  cin >> n >> k >> p;
  int a[n];
  fr(i, 0, n - 1) { cin >> a[i]; }
  int b[k];
  fr(i, 0, k - 1) { cin >> b[i]; }
  sort(a, a + n);
  sort(b, b + k);
  int j = 0, ans = 0;
  fr(i, 0, n - 1) {
    while (j < k && b[j] < a[i]) { j++; }
    if (j < k) {
      ans += abs(p - (a[i] - b[j]));
      j++;
    } else
      ans += abs(p - (a[i]));
  }
  cout << ans << endl;
}
