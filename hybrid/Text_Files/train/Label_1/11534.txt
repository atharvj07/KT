#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b, c, ans = 0;
    cin >> a >> b >> c;
    while (a >= 1 && b >= 2) {
      ans += 3;
      a -= 1;
      b -= 2;
    }
    while (b >= 1 && c >= 2) {
      ans += 3;
      b -= 1;
      c -= 2;
    }
    cout << ans << endl;
  }
  return 0;
}
