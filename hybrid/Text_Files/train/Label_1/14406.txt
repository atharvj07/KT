#include<bits/stdc++.h>
using namespace std ;
const int N = 2e5 + 10 , M = 400 , inf = 1e9 ;
int t[N][M] ;
int main() {
  int n , m , k , edg = 0 ;
  cin >> n >> m >> k ;
  for(int i = 1 ; i <= n ; i++) {
    for(int j = 1 ; j <= k ; j++) {
      t[i][j] = inf ;
    }
  }
  for(int i = 0 ; i < m ; i++) {
    int a , b , c ;
    cin >> a >> b >> c ;
    t[a][++edg] = c ;
    t[b][++edg] = c ;
  }
  for(int i = 1 ; i <= n ; i++) {
    sort(t[i] + 1 , t[i] + k + 1) ;
  }
  cout << t[1][k] ;
  return 0 ;
}
