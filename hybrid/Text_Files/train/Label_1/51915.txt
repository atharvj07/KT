import java.util.*;
import java.io.*;
import java.math.*;

class Main {
  static int n;
  static List<Line> lines = new ArrayList<>();
  static List<Triple> triples = new ArrayList<>();

  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    n = scan.nextInt();
    for (int i = 0; i < n; i++) {
      lines.add(new Line(scan.nextInt(), scan.nextInt(), scan.nextInt()));
    }
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        for (int k = j + 1; k < n; k++) {
          if (convex(i, j, k)) {
            triples.add(new Triple(i, j, k));
          }
        }
      }
    }
    int result = 0;
    for (Triple t : triples) {
      result += t.isOriginal();
    }
    System.out.println(result);
  }

  static boolean convex(int i, int j, int k) {
    Line li = lines.get(i), lj = lines.get(j), lk = lines.get(k);
    BigInteger det = li.det(lj).multiply(lk.det(li));
    return det.compareTo(BigInteger.ZERO) != 0;
  }

  static class Line {
    BigInteger a, b, c;

    Line(int a, int b, int c) {
      this.a = BigInteger.valueOf(a);
      this.b = BigInteger.valueOf(b);
      this.c = BigInteger.valueOf(c);
    }

    BigInteger det(Line l) {
      return a.multiply(l.b).subtract(b.multiply(l.a));
    }
  }

  static class Triple {
    int i, j, k;

    Triple(int i, int j, int k) {
      this.i = i;
      this.j = j;
      this.k = k;
    }

    int isOriginal() {
      Line li = lines.get(i), lj = lines.get(j), lk = lines.get(k);
      BigInteger zero = BigInteger.ZERO;
      BigInteger a = li.det(lj);
      BigInteger b = lj.det(lk);
      BigInteger c = lk.det(li);
      return (a.multiply(b).multiply(c).compareTo(zero) == 0) ? 1 : 0;
    }
  }
}
