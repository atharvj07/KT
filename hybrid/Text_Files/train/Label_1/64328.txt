for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    f = 0
    for i in range(1,n):
        if a[i]>a[i-1]:
            f = i
    if not f:
        print(-1)
    else:
        for i in range(1,n):
            if i!=f and a[i]>a[f]:
                f = 0
                break
        print(f)
