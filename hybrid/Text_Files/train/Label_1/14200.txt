#include <bits/stdc++.h>
#define all(x) (x).begin(), (x).end()
using namespace std;
typedef long long ll;
const char nl = '\n';

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int k;
    cin >> k;
    string s, t;
    cin >> s >> t;
    vector<int> cnt(26);
    for (int i = 0; i < k; i++) {
        cnt[s[i] - 'a']++;
        cnt[t[i] - 'a']--;
    }
    vector<int> sum(26, 0);
    for (int i = 0; i < 26; i++) {
        sum[i] = sum[i - 1] + cnt[i];
    }
    int mid = (k + 1) / 2;
    int ch = 0;
    while (mid > sum[ch]) {
        ch++;
    }
    string ans(k, 'a' + ch);
    int rem = k - 1;
    for (int l = 0, r = 0; l < k; l++) {
        while (r < k && sum[t[r] - 'a'] - sum[s[l] - 'a'] <= rem)
            r++;
        ans[l] = t[r - 1];
        rem--;
    }
    cout << ans << nl;
    return 0;
}
