import java.util.*;

public class Main {
    static int[] p;
    static int[] d;
    static int s, t;
    static int minD;
    static final int MAX_D = 1 << 20;

    static void union(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) {
            return;
        }
        if (d[a] > d[b]) {
            p[b] = a;
        } else if (d[a] < d[b]) {
            p[a] = b;
        } else {
            p[a] = b;
            d[b]++;
        }
    }

    static int find(int a) {
        if (p[a] == a) {
            return a;
        }
        return p[a] = find(p[a]);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        s = sc.nextInt() - 1;
        t = sc.nextInt() - 1;
        p = new int[n];
        d = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = i;
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            union(u, v);
        }
        int rootS = find(s);
        int minDst = MAX_D;
        minD = MAX_D;
        for (int i = 0; i < n; i++) {
            if (p[i] == rootS) {
                minDst = Math.min(minDst, d[i]);
            } else {
                minD = Math.min(minD, d[i]);
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int rootI = find(i);
            if (rootI == rootS && d[i] == minDst) {
                continue;
            }
            if (rootI != rootS && d[i] == minD) {
                continue;
            }
            ans++;
        }
        System.out.println(ans / 2);
    }
}
