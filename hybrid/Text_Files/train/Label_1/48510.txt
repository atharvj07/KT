import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        boolean isBerlanese = true;
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == 'a' || c == 'o' || c == 'u' || c == 'i' || c == 'e') {
                continue;
            } else if (c == 'n') {
                continue;
            } else if (i == s.length() - 1) {
                isBerlanese = false;
                break;
            } else if (s.charAt(i + 1) != 'a' && s.charAt(i + 1) != 'o' && s.charAt(i + 1) != 'u' && s.charAt(i + 1) != 'i' && s.charAt(i + 1) != 'e') {
                isBerlanese = false;
                break;
            }
        }
        System.out.println(isBerlanese ? "YES" : "NO");
    }
}
