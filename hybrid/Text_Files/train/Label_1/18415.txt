#include<bits/stdc++.h>
using namespace std;

int32_t main() {
    int t;
    cin >> t;
    while(t--) {
        int k;
        cin >> k;
        int r = 1, c = 1;
        long long int num = 1;
        while(num < k) {
            if(c < r) {
                num += c;
                c++;
            }
            else {
                num += r;
                r++;
            }
        }
        cout << r << " " << c << endl;
    }
    return 0;
}
