import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Scanner;
import java.util.Set;

public class SupersonicRocket {

    static class Point {
        double x;
        double y;

        public Point(double x, double y) {
            this.x = x;
            this.y = y;
        }

        public Point rotate(double theta) {
            double newX = x * Math.cos(theta) - y * Math.sin(theta);
            double newY = x * Math.sin(theta) + y * Math.cos(theta);
            return new Point(newX, newY);
        }

        public Point translate(double a, double b) {
            return new Point(x + a, y + b);
        }

        @Override
        public String toString() {
            return "(" + x + ", " + y + ")";
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Point> engine1 = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            engine1.add(new Point(scanner.nextInt(), scanner.nextInt()));
        }

        List<Point> engine2 = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            engine2.add(new Point(scanner.nextInt(), scanner.nextInt()));
        }

        if (isSafe(engine1, engine2)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isSafe(List<Point> engine1, List<Point> engine2) {
        Set<Point> powerField = new HashSet<>();
        for (Point p1 : engine1) {
            for (Point p2 : engine2) {
                for (double k = 0; k <= 1; k += 0.01) {
                    double x = k * p1.x + (1 - k) * p2.x;
                    double y = k * p1.y + (1 - k) * p2.y;
                    powerField.add(new Point(x, y));
                }
            }
        }

        for (Point p1 : engine1) {
            Set<Point> newPowerField = new HashSet<>();
            for (Point p2 : engine2) {
                for (double k = 0; k <= 1; k += 0.01) {
                    double x = k * p1.x + (1 - k) * p2.x;
                    double y = k * p1.y + (1 - k) * p2.y;
                    newPowerField.add(new Point(x, y));
                }
            }
            if (!powerField.equals(newPowerField)) {
                return false;
            }
        }

        for (Point p2 : engine2) {
            Set<Point> newPowerField = new HashSet<>();
            for (Point p1 : engine1) {
                for (double k = 0; k <= 1; k += 0.01) {
                    double x = k * p1.x + (1 - k) * p2.x;
                    double y = k * p1.y + (1 - k) * p2.y;
                    newPowerField.add(new Point(x, y));
                }
            }
            if (!powerField.equals(newPowerField)) {
                return false;
            }
        }

        return true;
    }
}
