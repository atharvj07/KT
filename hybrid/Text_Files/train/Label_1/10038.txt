#include <bits/stdc++.h>
using namespace std;

struct Node{
    long long mon, fri;
};

bool compare(const Node &a, const Node &b){
    return a.mon < b.mon;
}

int main(){

    long long n, d, x, y, ans, sum;
    priority_queue<long long> pq;
    cin >> n >> d;
    Node v[n+1];
    ans = 0;
    sum = 0;

    for(int i = 0; i < n; i++){
        cin >> x >> y;
        sum += y;
        v[i].mon = x;
        v[i].fri = y;
        pq.push(y);
    }

    sort(v, v+n, compare);
    while(!pq.empty()){
        x = pq.top();
        pq.pop();
        if(sum - x >= d){
            ans += x;
            sum -= x;
        }
    }

    cout << ans << endl;

    return 0;
}
