import java.util.Scanner;

public class MartianDollar {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int b = scanner.nextInt();

        int[] prices = new int[n];
        for (int i = 0; i < n; i++) {
            prices[i] = scanner.nextInt();
        }

        int[] maxProfit = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            maxProfit[i] = maxProfit[i - 1]; // No transaction on day i
            for (int j = 0; j < i; j++) {
                maxProfit[i] = Math.max(maxProfit[i], maxProfit[j] + (prices[i - 1] - prices[j]) * (i - j));
            }
        }

        System.out.println(maxProfit[n] + b);
    }
}
