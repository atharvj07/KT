#include <algorithm>
#include <string>
#include <vector>
#include <iostream>
using namespace std;

int main() {
    vector<char> used(26, 0);
    int k; string q; cin >> k >> q;
    if(q.length()<k) cout << "NO", exit(0);
    vector<string> ans; string pre; 
    for(char c:q) {
        if(used[c-'a']==0) {
            used[c-'a']=1;
            ans.push_back(pre);
            pre = string(1,c);
            k--;
            if(k==0) break;
        } else {
            pre += c;
        }
    }
    if(k>0) cout << "NO";
    else {
        cout << "YES" << endl;
        for(string s:ans) cout << s << endl;
    }
}
