#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define INF LLONG_MAX
const int maxn = 1e5 + 5;
struct edge {
    ll to, w;
};
vector<edge> e[maxn];
ll ans, n, m, d[maxn];
int main() {
    cin >> n >> m;
    while (m--) {
        ll u, v, w;
        cin >> u >> v >> w, d[u] -= w, d[v] += w;
    }
    for (int i = 1; i <= n; i++)
        if (d[i] != 0) e[n + 1].emplace_back((edge){i, d[i]});
    for (int i = 1; i <= n; i++)
        if (d[i] != 0)
            e[i].emplace_back((edge){n + 1, -d[i]}), ans += abs(d[i]);
    ll s = n + 2;
    d[s] = INF;
    priority_queue<ll, vector<ll>, greater<ll>> q;
    q.push(s);
    while (!q.empty()) {
        ll u = q.top();
        q.pop();
        if (d[u] == INF) continue;
        for (edge i : e[u])
            if (d[i.to] > i.w)
                q.push(i.to), d[i.to] = i.w, ans += d[u] + i.w;
    }
    cout << e[n + 1].size() << endl;
    for (edge i : e[n + 1]) cout << i.to << " " << n + 1 << " " << abs(i.w) << endl;
}
