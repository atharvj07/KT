#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int n,a[105],r1,c1,r2,c2,ans;
vector<int>pos[105];
int main() {
    cin>>n;
    for(int i=1;i<=n;i++) {
        cin>>a[i];
        for(int j=1;j<=a[i];j++) pos[i].push_back(j);
    }
    cin>>r1>>c1>>r2>>c2;
    if(r1==r2) ans=abs(c1-c2)-1;
    else ans=abs(r1-r2)+abs(c1-c2)-1;
    cout<<ans;
    return 0;
}
