#include <bits/stdc++.h>
using namespace std;
#define PI 3.14159265358979323846
#define EPS 1e-9

int main() {
    int n;
    cin >> n;
    vector<pair<double, double>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    double minAngle = PI;
    pair<int, int> ans;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            double angle = acos(
                (v[i].first * v[j].first + v[i].second * v[j].second) /
                (hypot(v[i].first, v[i].second) * hypot(v[j].first, v[j].second)));
            if (angle < minAngle) {
                minAngle = angle;
                ans = {i + 1, j + 1};
            }
        }
    }
    cout << ans.first << " " << ans.second << endl;
    return 0;
}
