def min_pieces(n, s, l, arr):
    dp = [float('inf')] * (n + 1)
    dp[0] = 0
    for i in range(1, n + 1):
        max_num, min_num = arr[i - 1], arr[i - 1]
        for j in range(i, max(0, i - l), -1):
            max_num = max(max_num, arr[j - 1])
            min_num = min(min_num, arr[j - 1])
            if max_num - min_num <= s:
                dp[i] = min(dp[i], dp[j - 1] + 1)
            else:
                break
    return dp[n] if dp[n] != float('inf') else -1

n, s, l = map(int, input().split())
arr = list(map(int, input().split()))

print(min_pieces(n, s, l, arr))
