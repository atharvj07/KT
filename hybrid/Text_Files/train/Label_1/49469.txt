import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    private static int maxLength = 0;
    private static Map<Character, List<String>> firstLetterMap = new HashMap<>();

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());

        for (int i = 0; i < n; i++) {
            String name = br.readLine();
            char firstLetter = name.charAt(0);
            char lastLetter = name.charAt(name.length() - 1);

            if (!firstLetterMap.containsKey(firstLetter)) {
                firstLetterMap.put(firstLetter, new ArrayList<>());
            }

            firstLetterMap.get(firstLetter).add(name);
        }

        Set<Character> keySet = firstLetterMap.keySet();
        for (Character c : keySet) {
            List<String> list = firstLetterMap.get(c);
            Collections.sort(list);
            for (int i = 0; i < list.size(); i++) {
                char curLast = list.get(i).charAt(list.get(i).length() - 1);
                if (c == curLast) {
                    boolean flag = true;
                    for (int j = i + 1; j < list.size(); j++) {
                        char nextFirst = list.get(j).charAt(0);
                        if (nextFirst != curLast) {
                            flag = false;
                            break;
                        }
                        curLast = list.get(j).charAt(list.get(j).length() - 1);
                    }
                    if (flag) {
                        maxLength = Math.max(maxLength, list.get(i).length() * list.size());
                    }
                }
            }
        }

        System.out.println(maxLength);
    }
}
