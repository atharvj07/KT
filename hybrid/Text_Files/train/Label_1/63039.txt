from collections import Counter

n = int(input())
friends = list(map(int, input().split()))
sum_friends = sum(friends)
counts = Counter(friends)
possibilities = []
if sum_friends % 2:
    possibilities.append(sum_friends // 2 + 1)
else:
    possibilities += [sum_friends // 2, sum_friends // 2 + 1]
    if counts[sum_friends // 2] > 1:
        possibilities.append(sum_friends // 2 - 1)
if possibilities:
    print(*possibilities)
else:
    print(-1)
