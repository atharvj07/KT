import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();

        boolean isPrime = true;
        for(int i = 2; i * i <= m; i++){
            if(m % i == 0){
                isPrime = false;
                break;
            }
        }

        if(n + 1 == m && isPrime){
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
