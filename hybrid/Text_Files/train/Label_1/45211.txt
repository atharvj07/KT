import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt();
        int x = sc.nextInt();
        int y = sc.nextInt();
        int x2 = sc.nextInt();
        int y2 = sc.nextInt();
        double dist = Math.sqrt((x2 - x) * (x2 - x) + (y2 - y) * (y2 - y));
        int ans = 0;
        if (dist > 2 * r) {
            ans = 2;
        }
        System.out.println(ans);
    }
}
