import java.util.Scanner;

public class ElectricShocks {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline after the integer

        int shocks = 0;
        boolean[] possibleLetters = new boolean[26];
        for (int i = 0; i < 26; i++) {
            possibleLetters[i] = true;
        }

        for (int i = 0; i < n; i++) {
            String action = scanner.nextLine();
            char firstChar = action.charAt(0);

            if (firstChar == '.') {
                String word = action.substring(2);
                for (char c : word.toCharArray()) {
                    possibleLetters[c - 'a'] = false;
                }
            } else if (firstChar == '!') {
                String word = action.substring(2);
                shocks++;
                for (char c : word.toCharArray()) {
                    if (possibleLetters[c - 'a']) {
                        possibleLetters[c - 'a'] = false;
                    } else {
                        shocks++; // Excessive shock
                    }
                }
            } else { // '?'
                char guess = action.charAt(2);
                if (!possibleLetters[guess - 'a']) {
                    shocks++; // Excessive shock
                } else {
                    for (int j = 0; j < 26; j++) {
                        if (j != guess - 'a' && possibleLetters[j]) {
                            shocks++; // Excessive shock
                            possibleLetters[j] = false;
                        }
                    }
                    break; // We found the letter, stop processing
                }
            }
        }

        System.out.println(shocks);
    }
}
