#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=2e5+5;
struct node
{
    ll val;
    int dep,len;
    node(ll _val,int _dep){val=_val;dep=_dep;len=dep*2;}
    bool operator < (const node &o)const
    {
        return val<o.val;
    }
};
char s[maxn];
int main()
{
    ll k,p,ans=0;
    cin>>k>>p;
    priority_queue<node> pq;
    pq.push(node(11,1));
    for(k++;k--;)
    {
        node now=pq.top();
        pq.pop();
        ans+=now.val;
        if(now.dep<maxn-1)
        {
            pq.push(node(now.val*10+1,now.dep+1));
        }
        if(now.dep)
        {
            pq.push(node(now.val*10,now.dep+1));
        }
    }
    cout<<ans%p<<endl;
    return 0;
}
