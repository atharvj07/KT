import java.util.Scanner;
import java.util.LinkedList;
import java.util.Queue;

public class GameExpansion {

    static class Cell {
        int x, y;

        Cell(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();
        int[] speeds = new int[p];
        for (int i = 0; i < p; i++) {
            speeds[i] = scanner.nextInt();
        }
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = line.charAt(j);
            }
        }

        int[] result = new int[p];
        for (int i = 0; i < p; i++) {
            result[i] = solve(grid, speeds[i], i);
        }

        for (int i : result) {
            System.out.print(i + " ");
        }
    }

    static int solve(char[][] grid, int speed, int playerId) {
        int n = grid.length;
        int m = grid[0].length;
        boolean[][] visited = new boolean[n][m];
        Queue<Cell> queue = new LinkedList<>();

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == playerId + '1') {
                    visited[i][j] = true;
                    queue.add(new Cell(i, j));
                }
            }
        }

        int count = 0;
        while (!queue.isEmpty()) {
            Cell cell = queue.poll();
            count++;

            for (int dx = -speed; dx <= speed; dx++) {
                for (int dy = -speed; dy <= speed; dy++) {
                    int nx = cell.x + dx;
                    int ny = cell.y + dy;
                    if (nx >= 0 && nx < n && ny >= 0 && ny < m && !visited[nx][ny] && grid[nx][ny] == '.') {
                        visited[nx][ny] = true;
                        queue.add(new Cell(nx, ny));
                    }
                }
            }
        }

        return count;
    }
}
