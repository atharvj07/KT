#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, ans = 0;
    cin >> n;
    vector<int> a(n + 1), b(n + 1), f(1001, 0);
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
        f[a[i]] = 1;
    }
    for (int i = 1; i <= n; i++) {
        ans += !f[b[i]];
    }
    cout << ans;
}
