#include <algorithm>
#include <cstdio>

int main(void) {
  int n;
  scanf("%d", &n);
  long long arr[n];
  for (int i = 0; i < n; i++) {
    scanf("%lld", &arr[i]);
  }
  std::sort(arr, arr + n);
  long long max_count = 1;
  long long cur = arr[0];
  for (int i = 1; i < n; i++) {
    if (cur < arr[i]) {
      max_count++;
      cur = arr[i];
    }
  }
  printf("%lld\n", max_count);
  return 0;
}
