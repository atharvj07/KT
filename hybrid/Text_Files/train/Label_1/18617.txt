#include <bits/stdc++.h>
using namespace std;

const int mod = 998244353;
int dp[1005][1005][2];
int n, k;

int get(int pos, int rem, bool last) {
  if (pos == n) {
    if (rem == 0)
      return 1;
    else
      return 0;
  }
  int &ret = dp[pos][rem][last];
  if (ret != -1) return ret;

  ret = 0;
  if (last == 0) {
    ret = (ret + get(pos + 1, rem, 1)) % mod;
  }
  ret = (ret + get(pos + 1, rem, 0)) % mod;
  if (rem > 0) {
    ret = (ret + get(pos + 1, rem - 1, 1)) % mod;
  }
  return ret;
}

int main() {
  cin >> n >> k;
  memset(dp, -1, sizeof(dp));
  cout << get(0, k, 0) << "\n";
  return 0;
}
