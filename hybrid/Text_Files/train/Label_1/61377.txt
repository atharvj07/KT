from collections import Counter

n,k=list(map(int,input().split()))
a=list(map(int,input().split()))
if len(set(a))>k or k==1:
    print("NO")
else:
    print("YES")
    b=[1]*n
    cnt=1
    for i in range(len(a)):
        if b[i]==1:
            x=a[i]
            for j in range(i,len(a)):
                if a[j]==x and b[j]==1:
                    b[j]=cnt
            cnt+=1
    for i in range(len(a)):
        print(b[i],end=" ")
    print()
