#include <bits/stdc++.h>
using namespace std;
int a[100001],ans[501],b[501],c[501],k,p,n,sum[100001],sumc[100001],odd,even,l1,l2;
void fun()
{
    int i;
    for(i=1; i<=p; i++)
        ans[i]=b[i]=c[i]=0;
    odd=even=l1=l2=0;
    sort(sum,sum+n+1);
}
int main()
{
    int i;
    cin>>n>>k>>p;
    for(i=1; i<=n; i++)
        cin>>a[i],sum[i]=sum[i-1]+a[i];
    if(p==k)
    {
        if(n%2==1)
            cout<<"NO";
        else
        {
            cout<<"YES\n";
            for(i=1; i<=n; i++)
                cout<<"1 "<<a[i]<<endl;
        }
    }
    else if(p==k-1)
    {
        if(n%2==0)
            cout<<"NO";
        else
        {
            cout<<"YES\n";
            for(i=1; i<=n; i++)
            {
                if(a[i]%2==0)
                {
                    if(l1==0)
                    {
                        l1++;
                        cout<<"1 "<<a[i]<<endl;
                    }
                    else if(l2==0)
                    {
                        l2++;
                        cout<<"1 "<<a[i]<<endl;
                    }
                }
                else
                {
                    if(l1==0)
                    {
                        l1++;
                        cout<<"1 "<<a[i]<<endl;
                    }
                    else
                    {
                        odd++;
                        cout<<"1 "<<a[i]<<endl;
                    }
                }
            }
        }
    }
    else if(k-p==1)
    {
        fun();
        for(i=1; i<=n; i++)
        {
            sumc[i]=sumc[i-1]+a[i];
            if(sumc[i]%2==1)
                c[i]=1;
        }
        odd=n/2;
        if((n/2)%2==0)
            even=odd-1;
        else
            even=odd;
        if(sumc[n]%2==1&&even<p)
        {
            even++;
            odd--;
        }
        if(sumc[n]%2==0&&odd<p)
        {
            even++;
            odd--;
        }
        if(even>=p)
        {
            l1=0;
            l2=0;
            cout<<"YES\n";
            for(i=n; i>=1; i--)
            {
                if(c[i]==1&&odd)
                {
                    b[++l2]=a[i];
                    odd--;
                }
                else if(c[i]==0&&even)
                {
                    b[++l1]=a[i];
                    even--;
                }
            }
            if(l1!=0)
            {
                cout<<l1<<" ";
                for(i=1; i<=l1; i++)
                    cout<<b[i]<<" ";
                cout<<endl;
            }
            if(l2!=0)
            {
                cout<<l2<<" ";
                for(i=1; i<=l2; i++)
                    cout<<b[i]<<" ";
                cout<<endl;
            }
            if(n-l1-l2!=0)
            {
                cout<<n-l1-l2<<" ";
                for(i=1; i<=n; i++)
                {
                    if(c[i]==0&&l1==0)
                        continue;
                    if(c[i]==1&&l2==0)
                        continue;
                    cout<<a[i]<<" ";
                }
                cout<<endl;
            }
        }
        else
            cout<<"NO";
    }
    else if(k-p>1)
        cout<<"NO";
    return 0;
}
