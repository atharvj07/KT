def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def find_min_points_to_add(n, coordinates):
    coordinates.sort()
    diffs = [coordinates[i+1] - coordinates[i] for i in range(n-1)]
    gcd_val = diffs[0]
    for diff in diffs[1:]:
        gcd_val = gcd(gcd_val, diff)
    total_points_needed = sum(diff // gcd_val for diff in diffs) - (n - 1)
    return total_points_needed

n = int(input().strip())
coordinates = list(map(int, input().strip().split()))
print(find_min_points_to_add(n, coordinates))
