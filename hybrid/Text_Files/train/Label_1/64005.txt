def domino(h,w,g,q):
 for i in range(h):
  for j in range(w-1):
   if g[i][j]=='.' and g[i][j+1]=='.': g[i][j:j+2]='#'
 for j in range(w):
  for i in range(h-1):
   if g[i][j]=='.' and g[i+1][j]=='.': g[i:i+2][j]='#'
 for i in range(q):
  r1,c1,r2,c2=map(int,input().split())
  r1-=1;c1-=1
  ans=0
  for j in range(c1,c2): ans+=g[r1][j]=='.' and g[r1+1][j]=='.';g[r1][j:j+2]='#'
  for j in range(c1,c2-1): ans+=g[r2][j]=='.' and g[r2][j+1]=='.';g[r2][j:j+2]='#'
  for i in range(r1+1,r2): ans+=g[i][c2]=='.' and g[i+1][c2]=='.';g[i:i+2][c2]='#'
  for i in range(r1+1,r2-1): ans+=g[i][c1]=='.' and g[i][c1+1]=='.';g[i][c1:c1+2]='#'
  print(ans)
h,w=map(int,input().split())
g=[list(input())for i in range(h)]
q=int(input())
domino(h,w,g,q)
