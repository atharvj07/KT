import java.util.Scanner;

public class CompetitionMoney {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int money = 0;
        if (x <= 3) {
            money += 100000 * (4 - x);
        }
        if (y <= 3) {
            money += 100000 * (4 - y);
        }
        if (x == 1 && y == 1) {
            money += 400000;
        }
        System.out.println(money);
    }
}
