import sys
mod = 10**9+7
def mpow(x, y):
  if y == 0: return 1
  if y == 1: return x
  t = mpow(x, y//2)
  t = (t*t)%mod
  if y&1:
    return (t*x)%mod
  return t
def solve(n, k):
  ans = 0
  l = len(n)
  for i in range(l):
    if n[i] == '0':
      for j in range(i+1,l):
        if n[j] == '1':
          x = j - i
          if x <= k:
            ans += mpow(2,k-x)
          ans %= mod
  return ans
n = input()
k = int(input())
print(solve(n,k))
