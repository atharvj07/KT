#include<bits/stdc++.h>
#define int long long
using namespace std;
const int mod=998244353;
int a[100005];
int dp[100005][3][2];
string s;
int n;
int solve(int i,int j,int t){
    if(i==n){
        if(j==0)return 0;
        else return 1e18;
    }
    int &ans=dp[i][j][t];
    if(~ans)return ans;
    ans=solve(i+1,j,t);
    if(t==0)ans=min(ans,a[i]+solve(i+1,j+1,1));
    else{
        if(j==0 or (j==1 and s[i]==s[i-1]))ans=min(ans,a[i]+solve(i+1,j+1,1));
        if(j==1 or (j==2 and s[i]==s[i-1]))ans=min(ans,a[i]+solve(i+1,j+1,1));
        if(j==0 or (j==1 and s[i]!=s[i-1]))ans=min(ans,solve(i+1,j,t));
        if(j==1 or (j==2 and s[i]!=s[i-1]))ans=min(ans,solve(i+1,j,t));
    }
    return ans;
}
int32_t main(){
    cin>>n;
    cin>>s;
    for(int i=0;i<n;i++)cin>>a[i];
    memset(dp,-1,sizeof(dp));
    cout<<solve(0,0,0)<<endl;
}
