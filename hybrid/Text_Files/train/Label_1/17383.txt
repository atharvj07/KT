#include <bits/stdc++.h>
#define ll long long
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    ll a[n + 5], b[n + 5], tm[n + 5];
    for (int i = 1; i <= n; i++) { cin >> a[i] >> b[i]; }
    for (int i = 1; i <= n; i++) { cin >> tm[i]; }
    ll ans = 0;
    for (int i = 1; i < n; i++) {
      ll diff = b[i] - a[i] + tm[i];
      ans += max(0ll, (diff + 1) / 2);
      ans += diff / 2;
      ans = max(ans, b[i]);
    }
    cout << ans << endl;
  }
  return 0;
}
