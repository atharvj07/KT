#include <bits/stdc++.h>
using namespace std;

const int maxn = 100005;
vector<int> adj[maxn];
int vis[maxn];
int ans;
int dfs(int u) {
    if (vis[u] == 1) return 0;
    if (vis[u] == 2) return ans;
    vis[u] = 1;
    int p = 0;
    for (auto &x : adj[u]) {
        p = max(abs(x), dfs(abs(x)));
    }
    vis[u] = 2;
    return ans = max(p, ans);
}
int32_t main() {
    int n;
    cin >> n;
    for (int i = 2; i <= n; i++) {
        for (int j = i * i; j <= n; j += i) {
            adj[i].push_back(j);
            adj[j].push_back(i);
        }
    }
    cout << dfs(2) << "\n";
}
