from math import gcd

def read_ints():
    return map(int, input().strip().split())

def solve(l_a, r_a, t_a, l_b, r_b, t_b):
    g = gcd(t_a, t_b)
    shift = (l_b - l_a) % g
    overlap1 = min(r_a, r_b - shift) - l_a + 1
    overlap2 = min(r_a + g, r_b - shift + t_b) - (l_a + g) + 1
    return max(0, max(overlap1, overlap2))

l_a, r_a, t_a = read_ints()
l_b, r_b, t_b = read_ints()

print(solve(l_a, r_a, t_a, l_b, r_b, t_b))
