#include <bits/stdc++.h>
using namespace std;
char s[1002][6];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> s[i];
  for (int i = 0; i < n; i++) {
    if (s[i][0] == 'O' && s[i][1] == 'O') {
      s[i][0] = '+';
      s[i][1] = '+';
      cout << "YES\n";
      for (int j = 0; j < n; j++) cout << s[j] << '\n';
      return 0;
    } else if (s[i][3] == 'O' && s[i][4] == 'O') {
      s[i][3] = '+';
      s[i][4] = '+';
      cout << "YES\n";
      for (int j = 0; j < n; j++) cout << s[j] << '\n';
      return 0;
    }
  }
  cout << "NO\n";
}
