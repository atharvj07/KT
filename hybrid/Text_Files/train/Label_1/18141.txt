#include <bits/stdc++.h>
using namespace std;
#define int long long

vector<int> a;
int32_t main() {
    int n, k;
    cin >> n >> k;
    a.resize(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    set<int> ban;
    for (int i = 0; i < k; i++) {
        int t;
        cin >> t;
        ban.insert(t - 1);
    }
    ban.insert(n);
    int ans = 0;
    int last = 1e9;
    for (int i = n - 1; i >= 0; i--) {
        if (ban.find(i) != ban.end()) {
            if (last <= a[i]) {
                ans = -1;
                break;
            }
        } else {
            if (a[i] >= last) {
                last = a[i];
            } else {
                ans += last - a[i];
                a[i] = last;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
