import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), m = in.nextInt(), d = in.nextInt();
    int[] a = new int[n];
    int[] b = new int[n];
    int[] p = new int[m];
    for (int i = 0; i < m; i++) p[i] = in.nextInt();
    for (int i = 0; i < n - 1; i++) {
      a[i] = in.nextInt() - 1;
      b[i] = in.nextInt() - 1;
    }
    int[] d1 = new int[n], d2 = new int[n];
    Arrays.fill(d1, -1);
    Arrays.fill(d2, -1);
    Queue<Integer> q = new ArrayDeque<>();
    q.add(0);
    d1[0] = 0;
    while (!q.isEmpty()) {
      int x = q.poll();
      for (int i = 0; i < n - 1; i++) {
        if (x == a[i] && d2[b[i]] == -1) {
          d2[b[i]] = d1[x] + 1;
          q.add(b[i]);
        }
        if (x == b[i] && d2[a[i]] == -1) {
          d2[a[i]] = d1[x] + 1;
          q.add(a[i]);
        }
      }
    }
    q = new ArrayDeque<>();
    q.add(n - 1);
    d2[n - 1] = 0;
    while (!q.isEmpty()) {
      int x = q.poll();
      for (int i = 0; i < n - 1; i++) {
        if (x == a[i] && d1[b[i]] == -1) {
          d1[b[i]] = d2[x] + 1;
          q.add(b[i]);
        }
        if (x == b[i] && d1[a[i]] == -1) {
          d1[a[i]] = d2[x] + 1;
          q.add(a[i]);
        }
      }
    }
    int cnt = 0;
    for (int i = 0; i < m; i++) if (d1[p[i] - 1] <= d || d2[p[i] - 1] <= d) cnt++;
    System.out.println(cnt);
  }
}
