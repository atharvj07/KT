#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N=150005;
int n,a[N],b[N],c[N],cc,d;
int main()
{
  cin>>n;
  for(int i=0;i<n;i++)
   {
      cin>>a[i];
      b[a[i]]++;
   }
  for(int i=0;i<n;i++)
   {
      cc=b[i];
      d=cc-(i&1);
      cc-=d;
      c[i]=cc;
   }
  ll ans=0;
  for(int i=1;i<N;i++)
   {
      ans+=1LL*c[i]*c[i-1];
      c[i]+=c[i-1];
   }
  cout<<ans<<endl;
  return 0;
}
