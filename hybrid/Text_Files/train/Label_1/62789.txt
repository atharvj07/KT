n, k = map(int, input().split())
d = list(map(int, input().split()))

if d.count(0) != 1:
    print(-1)
    exit()

edges = []
level = {}
for i in range(n):
    if d[i] in level:
        level[d[i]].append(i+1)
    else:
        level[d[i]] = [i+1]

if 0 not in level or len(level[0]) != 1:
    print(-1)
    exit()

for i in range(max(d)):
    if i not in level or i+1 not in level or len(level[i])*k < len(level[i+1]):
        print(-1)
        exit()
    for j in range(len(level[i+1])):
        edges.append((level[i][j//k], level[i+1][j]))

print(len(edges))
for edge in edges:
    print(edge[0], edge[1])
