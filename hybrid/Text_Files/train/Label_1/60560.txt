class Ring:
    def __init__(self, a, b, h):
        self.a = a
        self.b = b
        self.h = h

    def __repr__(self):
        return f"Ring(a={self.a}, b={self.b}, h={self.h})"

    def can_be_placed_on(self, other):
        return self.a < other.b and self.b > other.a


def main():
    n = int(input())
    rings = [Ring(*map(int, input().split())) for _ in range(n)]
    rings.sort(key=lambda ring: (ring.b, ring.a))

    dp = [0] * n
    for i in range(n):
        for j in range(i):
            if rings[i].can_be_placed_on(rings[j]):
                dp[i] = max(dp[i], dp[j] + rings[i].h)

    return max(dp)


print(main())
