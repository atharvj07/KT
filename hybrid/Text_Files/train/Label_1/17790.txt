#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long int r, g, b, w;
        cin >> r >> g >> b >> w;
        if (r == 0 && g == 0 && b == 0 && w >= 3)
            cout << "Yes" << endl;
        else if (r % 2 == g % 2 && g % 2 == b % 2)
            cout << "Yes" << endl;
        else
            cout << "No" << endl;
    }
    return 0;
}
