import java.util.Scanner;

public class Main {

    private static final int MOD = 1000000009;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int q = scanner.nextInt();

        long[] dp = new long[n + 1];
        long[] orders = new long[n + 1];
        long[] prefixSum = new long[n + 1];

        for (int i = 0; i < q; i++) {
            int type = scanner.nextInt();
            int day = scanner.nextInt();

            if (type == 1) {
                int count = scanner.nextInt();
                orders[day] += count;
            } else {
                computeDp(n, k, a, b, orders, prefixSum, dp);
                System.out.println(dp[day] % MOD);
            }
        }
    }

    private static void computeDp(int n, int k, int a, int b, long[] orders, long[] prefixSum, long[] dp) {
        prefixSum[0] = orders[0];
        for (int i = 1; i <= n; i++) {
            prefixSum[i] = prefixSum[i - 1] + orders[i];
        }

        dp[0] = prefixSum[0];
        dp[1] = Math.min(prefixSum[1], b);
        for (int i = 2; i <= n; i++) {
            dp[i] = Math.min(dp[i - 1] + b, prefixSum[i]);
            if (i >= k) {
                dp[i] = Math.max(dp[i], dp[i - k] + prefixSum[i] - prefixSum[i - k]);
            }
        }
    }
}
