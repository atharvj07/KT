import java.util.Arrays;
import java.util.Scanner;

public class Main {
  static int n, k;
  static int[] arr;
  static int mod = 998244353;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    k = sc.nextInt();
    Integer[] a = new Integer[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    Arrays.sort(a);
    arr = Arrays.stream(a).mapToInt(Integer::intValue).toArray();
    System.out.println(solve(0, 0, 0));
  }

  static int solve(int l, int r, int mask) {
    if (r == n) {
      if (l == n)
        return 1;
      else
        return 0;
    }
    int ans = solve(l + 1, r + 1, mask);
    if ((mask & (1 << arr[r])) == 0 && cnt(mask, arr[r]) == 0)
      ans = (ans + solve(l, r + 1, mask | (1 << arr[r]))) % mod;
    return ans;
  }

  static int cnt(int mask, int x) {
    int cnt = 0;
    while (x > 0) {
      if ((x & 1) == 1)
        cnt++;
      x >>= 1;
    }
    return cnt;
  }
}
