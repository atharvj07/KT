import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        char[] positions = scanner.next().toCharArray();

        int standing = 0;
        for (int i = 0; i < n; i++) {
            if (positions[i] == 'X') standing++;
        }

        int steps = Math.abs(standing - (n / 2));
        System.out.println(steps);

        int sitting = 0;
        for (int i = 0; i < n; i++) {
            if (positions[i] == 'X' && sitting < n / 2) {
                positions[i] = 'x';
                sitting++;
            } else if (positions[i] == 'x' && standing > n / 2) {
                positions[i] = 'X';
                standing--;
            }
        }

        System.out.println(positions);
    }
}
