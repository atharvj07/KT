#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    long long n, k;
    cin >> n >> k;
    if (k > (n + 1) / 2) {
        cout << "-1\n";
        return 0;
    }
    string s(n, 'a');
    for (int i = 0; i < k - 1; i += 2) {
        s[i] = 'a' + i / 2;
    }
    if (k % 2 == 0) {
        s[k - 1] = 'a' + k / 2 - 1;
    }
    cout << s << "\n";
    return 0;
}
