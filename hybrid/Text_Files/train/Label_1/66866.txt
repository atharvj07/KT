import sys
T = int(sys.stdin.readline())
for _ in range(T):
    n, m = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    if m < n - 1:
        print(-1)
    else:
        d, e, i = [], [], 0
        for j in range(n):
            for k in range(j + 1, n):
                i += 1
                d.append(a[j] + a[k])
                e.append([j + 1, k + 1])
        d, e = zip(*sorted(zip(d, e)))
        cost = sum(d[:m])
        print(cost)
        for j in range(m):
            print(*e[j])
