for _ in range(int(input())):
    input()
    p=[int(x)-1 for x in input().split()]
    ans=[0]*len(p)
    vis=[False]*len(p)
    for i in range(len(p)):
        if(vis[i]==False):
            j=i
            cnt=1
            while(p[j]!=i):
                j=p[j]
                cnt+=1
            for k in range(i,p[j]+1):
                vis[k]=True
                ans[k]=cnt
    print(*ans)
