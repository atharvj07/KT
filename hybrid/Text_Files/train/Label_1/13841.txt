#include <bits/stdc++.h>

using namespace std;

bool check_valid(int n) {
  n %= 100;
  if (n % 4 == 0) return true;
  n %= 10;
  if (n == 2 || n == 5 || n == 6 || n == 9) return true;
  return false;
}

int main() {
  int t;
  cin >> t;
  string s;
  cin >> s;
  int ans = 0;
  int len = s.size();
  for(int i = 0; i < len; i++)
    if (s[i] - '0' == t || (s[i] == '6' && t == 9) || (s[i] == '9' && t == 6) ||
        (s[i] == '5' && t == 2) || (s[i] == '2' && t == 5)) {
      if (check_valid(10 * t + (s[i] - '0'))) ans++;
    }
  cout << ans << endl;
  return 0;
}
