import java.util.Scanner;

public class CodeforcesContest {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Input
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int c = scanner.nextInt();
        int d = scanner.nextInt();

        // Calculate points for Misha
        int mishaPoints = calculatePoints(a, c);

        // Calculate points for Vasya
        int vasyaPoints = calculatePoints(b, d);

        // Determine the winner
        if (mishaPoints > vasyaPoints) {
            System.out.println("Misha");
        } else if (vasyaPoints > mishaPoints) {
            System.out.println("Vasya");
        } else {
            System.out.println("Tie");
        }
    }

    // Calculate points based on problem cost and submission time
    private static int calculatePoints(int cost, int submissionTime) {
        return cost - (submissionTime / 250) * cost / 10;
    }
}
