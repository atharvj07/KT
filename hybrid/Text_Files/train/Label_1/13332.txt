#include<bits/stdc++.h>
using namespace std;
int main(){
    long long n, m, x, y, z, p;
    cin >> n >> m >> x >> y >> z >> p;
    while (x--) { swap(n, m); }
    while (y--){}
    while (z--) { swap(n, m); }
    while (p--) {
        long long a, b;
        cin >> a >> b;
        swap(a, b);
        a = n + 1 - a;
        cout << a << ' ' << b << '\n';
    }
}
