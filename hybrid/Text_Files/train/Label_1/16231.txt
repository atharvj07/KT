typedef int64_t ll;

const ll inf=1e18;
ll power(ll x, ll n){
    if(n==0){
        return 1;
    }
    ll ans=power(x,n/2);
    if(n%2==1){
        return ((ans%mod)*(ans%mod))%mod*(x%mod);
    }
    else{
        return ((ans%mod)*(ans%mod))%mod;
    }
}

ll fact(ll x){
    ll ans=1;
    for(ll i=1;i<=x;i++){
        ans=(ans%mod)*(i%mod);
    }
    return ans;
}
