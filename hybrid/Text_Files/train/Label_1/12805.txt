#include <iostream>
#include <algorithm>

using namespace std;

bool check(int n, int m, int A[500][500], int B[500][500]) {
  for (int i = 1; i <= n - 1; i++) {
    for (int j = 1; j <= m - 1; j++) {
      if (A[i][j] == A[i + 1][j] && A[i][j] == A[i][j + 1] && A[i][j] == A[i + 1][j + 1]) {
        if (A[i][j] == B[i][j]) continue;
        return false;
      }
      if (A[i][j] != A[i + 1][j] && A[i][j] != A[i][j + 1] && A[i][j] != A[i + 1][j + 1]) {
        if (A[i][j] == B[i][j]) continue;
        return false;
      }
    }
  }
  return true;
}

int main() {
  int n, m;
  cin >> n >> m;
  int A[500][500], B[500][500];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cin >> A[i][j];
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cin >> B[i][j];
    }
  }
  if (check(n, m, A, B)) cout << "Yes" << endl;
  else cout << "No" << endl;
  return 0;
}
