#include <cstdio>
#include <vector>
#include <map>
using namespace std;

void solve(vector<pair<char, int>> &s) {
    map<char, int> cnt;
    for (auto e : s) cnt[e.first] += e.second;
    int k = 0;
    for (auto e : cnt) k += min(e.second, 1);
    printf("%d\n", k);
}

int main() {
    int T;
    scanf("%d", &T);
    while (T--) {
        char buf[55];
        scanf("%s", buf);
        vector<pair<char, int>> s;
        for (int i = 0; buf[i] != '\0'; i += 2) {
            s.emplace_back(buf[i], buf[i + 1] - '0');
        }
        solve(s);
    }
    return 0;
}
