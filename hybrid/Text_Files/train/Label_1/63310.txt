board = [input() for _ in range(8)]
white, black = 0, 0
for row in board:
    for piece in row:
        if piece.isupper():
            if piece == 'Q':
                white += 9
            elif piece == 'R':
                white += 5
            elif piece in 'BN':
                white += 3
            elif piece == 'P':
                white += 1
        elif piece.islower():
            if piece == 'q':
                black += 9
            elif piece == 'r':
                black += 5
            elif piece in 'bn':
                black += 3
            elif piece == 'p':
                black += 1
if white > black:
    print("White")
elif black > white:
    print("Black")
else:
    print("Draw")
