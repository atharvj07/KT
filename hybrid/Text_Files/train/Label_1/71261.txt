n, m = map(int, input().split())
connections = [set() for _ in range(n + 1)]
for _ in range(m):
    a, b = map(int, input().split())
    connections[a].add(b)
    connections[b].add(a)

min_sum = float('inf')
for i in range(1, n + 1):
    for j in connections[i]:
        for k in connections[j]:
            if i != k and i in connections[k] and j in connections[k]:
                curr_sum = (len(connections[i]) - 2) + (len(connections[j]) - 2) + (len(connections[k]) - 2)
                min_sum = min(min_sum, curr_sum)

if min_sum == float('inf'):
    print(-1)
else:
    print(min_sum)
