s=input()
lf,rt=0,[(0,0)for _ in s]
for i in range(len(s)):
    if s[i]=="?": lf+=1
    else: lf=1; rt[i]=(s[i],s[i])
for c in range(ord('A'),ord('J')+1):
    if s.count(chr(c))>0: lf*=(lf+int(s.count(chr(c))>1))
    for i in range(len(s)):
        if s[i]==chr(c):
            f=1
            for j in range(len(s)):
                if s[j]==chr(c): f*=rt[j][0]
                else: f*=rt[j][1]
            lf*=f
            for j in range(len(s)):
                if s[j]==chr(c):
                    rt[j]=(rt[j][0],rt[j][0]+rt[j][1])
                else: rt[j]=(rt[j][0],rt[j][1])
print(lf)
