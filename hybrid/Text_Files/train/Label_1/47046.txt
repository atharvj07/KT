import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt(), m = cin.nextInt(), ans = 0;
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = cin.nextInt();
                ans += a[i][j];
            }
        }
        for (int i = 0; i < n; i++) {
            int lst = 0, sum = 0;
            for (int j = 0; j < m; j++) {
                if (a[i][j] > 0) {
                    sum += a[i][j];
                    if (j % 2 == lst) {
                        sum -= 2 * a[i][j];
                    }
                    lst = j % 2;
                }
            }
            ans -= sum;
        }
        System.out.println(ans);
    }
}
