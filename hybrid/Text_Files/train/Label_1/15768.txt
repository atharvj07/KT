#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    set<char> st;
    for (int i = 0; i < n; i++) {
      if (i > 0 and s[i] == s[i - 1]) {
        st.erase(s[i]);
      } else {
        st.insert(s[i]);
      }
    }
    for (char c : st) cout << c;
    cout << '\n';
  }
}
