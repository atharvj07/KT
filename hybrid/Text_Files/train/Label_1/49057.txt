import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long totalFileSize = input.nextLong();
        long totalTime = input.nextLong();
        long regularSpeed = input.nextLong();
        long firstPackageLimit = input.nextLong();
        long firstPackageSpeed = input.nextLong();
        long secondPackageLimit = input.nextLong();
        long secondPackageSpeed = input.nextLong();
        long firstPackageCost = input.nextLong();
        long secondPackageCost = input.nextLong();

        long totalCost = 0;
        long totalTimeUsed = 0;
        long totalDownloaded = 0;
        int packagesToUse = 0;

        // Calculate the time it takes to download the entire file using the regular speed.
        long timeForRegularSpeed = totalFileSize / regularSpeed;

        // Calculate the maximum number of packages that Polycarp can use to download the file in time.
        long maxPackages = (totalTime - timeForRegularSpeed) / firstPackageSpeed;

        // Iterate over the possible number of packages to use.
        for (int i = 0; i <= maxPackages; i++) {
            // Calculate the total cost of using the current number of packages.
            long cost = (i * firstPackageCost) + (packagesToUse * secondPackageCost);

            // Calculate the total time it takes to download the file using the current number of packages.
            long time = timeForRegularSpeed + (i * firstPackageLimit / firstPackageSpeed) + (packagesToUse * secondPackageLimit / secondPackageSpeed);

            // If the total time is less than the total time allowed, and the total cost is less than the current minimum cost, then update the minimum cost and the total time used.
            if (time <= totalTime && cost < totalCost) {
                totalCost = cost;
                totalTimeUsed = time;
                packagesToUse = i;
            }
        }

        // If no solution is found, then print -1.
        if (totalCost == 0) {
            System.out.println(-1);
        } else {
            // Otherwise, print the minimum cost.
            System.out.println(totalCost);
        }
    }
}
