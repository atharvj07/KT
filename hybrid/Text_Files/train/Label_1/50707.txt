import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        long[] c = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        long minA = Arrays.stream(a).min().getAsLong();
        long maxA = Arrays.stream(a).max().getAsLong();
        long minB = Arrays.stream(b).min().getAsLong();
        long maxB = Arrays.stream(b).max().getAsLong();
        int ans = 0;
        if (maxA * minB + minA * maxB == 0) {
            ans = n;
        } else {
            for (int i = 0; i < n; i++) {
                c[i] = a[i] * minB + b[i] * maxA;
                ans += c[i] == 0 ? 1 : 0;
            }
        }
        System.out.println(ans);
    }
}
