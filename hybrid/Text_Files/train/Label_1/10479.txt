#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
int main(){
    int t, r, c;
    cin>>t;
    while(t--){
        cin>>r>>c;
        vector<vector<int>> v(r, vector<int>(c));
        for(int i=0;i<r;++i){
            string s; cin>>s;
            for(int j=0;j<c;++j)
                v[i][j]=s[j]-'A';
        }
        int cnt=0;
        for(int i=0;i<r;++i){
            for(int j=0;j<c;++j){
                bool can=false;
                for(int k=j+1;k<c;++k){
                    if(!v[i][k]){
                        can=true;
                        break;
                    }
                }
                if(can) ++cnt;
            }
        }
        for(int i=0;i<c;++i){
            for(int j=0;j<r;++j){
                bool can=false;
                for(int k=j+1;k<r;++k){
                    if(!v[k][i]){
                        can=true;
                        break;
                    }
                }
                if(can) ++cnt;
            }
        }
        
        cout<<(cnt?"MORTAL":to_string(cnt))<<endl;
    }
}
