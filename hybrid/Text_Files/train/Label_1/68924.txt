from collections import deque
n=int(input())
s=[""]*n
for i in range(n):
 s[i]=input()
used=[0]*n
q=deque([0])
while q:
 i=q.popleft()
 if used[i]:continue
 used[i]=1
 for j in range(n):
  if s[j].find(s[i])!=-1:
   q.append(j)
   used[j]=1
print(["NO","YES"][all(used)])
for i in range(n):
 if used[i]:print(s[i])
