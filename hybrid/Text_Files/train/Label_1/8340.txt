#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  string s;
  cin >> s;
  int k;
  cin >> k;
  vector<int> w(26);
  for (int& wi : w) {
    cin >> wi;
  }
  vector<int> cnt(26);
  for (const char& c : s) {
    cnt[c - 'a']++;
  }
  sort(cnt.rbegin(), cnt.rend());
  sort(w.rbegin(), w.rend());
  int ans = 0;
  for (int i = 0; i < 26 && cnt[i] > 0; i++) {
    ans += cnt[i] * w[i];
    k -= cnt[i];
  }
  for (int i = 0; i < 26 && k > 0; i++) {
    ans += k * w[i];
    k -= 26;
  }
  cout << ans << endl;
  return 0;
}
