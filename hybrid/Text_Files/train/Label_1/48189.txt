import java.util.Scanner;

class Main {
    static int n, y;
    static Bank[] banks = new Bank[51];

    static class Bank {
        int b, r, t;
        double a;

        public Bank(int b, int r, int t) {
            this.b = b;
            this.r = r;
            this.t = t;
        }

        public void calc() {
            if (t == 1) {
                a = (double) r / 100 * y + 1;
            } else {
                a = Math.pow(1 + (double) r / 100, y);
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            n = sc.nextInt();
            if (n == 0) {
                return;
            }
            y = sc.nextInt();
            for (int i = 0; i < n; i++) {
                banks[i] = new Bank(sc.nextInt(), sc.nextInt(), sc.nextInt());
            }
            for (int i = 0; i < n; i++) {
                banks[i].calc();
            }
            int idx = 0;
            double max = 0;
            for (int i = 0; i < n; i++) {
                if (banks[i].a > max) {
                    max = banks[i].a;
                    idx = banks[i].b;
                }
            }
            System.out.println(idx);
        }
    }
}
