#include <bits/stdc++.h>
using namespace std;

int main() {
  int t, n, a[200001], freq[200001];
  cin >> t;
  while (t--) {
    cin >> n;
    memset(freq, 0, sizeof(freq));
    for (int i = 1; i <= n; i++) { cin >> a[i]; freq[a[i]]++; }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      if (freq[i] >= 2) ans = max(ans, freq[i] / 2);
    }
    int diff = 0;
    for (int i = 1; i <= n; i++) {
      if (freq[i] == 1) diff++;
    }
    ans = max(ans, diff / 2);
    cout << ans << "\n";
  }
  return 0;
}
