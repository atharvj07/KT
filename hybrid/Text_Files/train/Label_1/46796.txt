import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), l = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        Integer[][] dp = new Integer[n][l + 1];
        System.out.println(dfs(a, 0, l, dp));
    }

    private static int dfs(int[] a, int idx, int l, Integer[][] dp) {
        if (idx == a.length) {
            return l == 0 ? 1 : 0;
        }
        if (dp[idx][l] != null) {
            return dp[idx][l];
        }
        int res = dfs(a, idx + 1, l, dp);
        if (l > 0) {
            res += dfs(a, idx + 1, l - 1, dp);
            if (a[idx] == 1 || l > 1) {
                res += dfs(a, idx + 1, l - 1, dp);
            }
        }
        return dp[idx][l] = res;
    }
}
