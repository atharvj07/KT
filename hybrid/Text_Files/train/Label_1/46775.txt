import java.util.*;
public class Main {
public static void main(String[] args) {
Scanner scan = new Scanner(System.in);
int n = scan.nextInt(), m = scan.nextInt();
int[] a = new int[n];
for (int i = 0; i < n; i++) a[i] = scan.nextInt();
int[][] sub = new int[m][2];
for (int i = 0; i < m; i++) { sub[i][0] = scan.nextInt() - 1; sub[i][1] = scan.nextInt() - 1; }
int[] cnt = new int[n];
for (int[] p : sub) { for (int i = p[0]; i <= p[1]; i++) cnt[i]++; }
long ans = 0;
for (int i = 0; i < n; i++) ans += (long) a[i] * cnt[i];
System.out.println(ans);
}}
