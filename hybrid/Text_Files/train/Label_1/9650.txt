#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  for (int i = 0; i < n; ++i) {
    b[i] = (a[i] + m - 1) / m;
  }
  int ans = 0, sum = 0;
  for (int i = 0; i < n; ++i) {
    ans = max(ans, sum + b[i]);
    sum += b[i];
  }
  cout << ans << endl;
  return 0;
}
