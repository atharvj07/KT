import java.util.*;
import java.io.*;
import java.math.BigInteger;

public class Main {
  static int n, m, L, s, t;
  static List<Edge> edges;
  static long[] dist;
  static int[] p;
  static boolean flag;

  static class Edge {
    int from, to, cost;

    public Edge(int from, int to, int cost) {
      this.from = from;
      this.to = to;
      this.cost = cost;
    }
  }

  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(in.readLine());
    n = Integer.parseInt(st.nextToken());
    m = Integer.parseInt(st.nextToken());
    L = Integer.parseInt(st.nextToken());
    s = Integer.parseInt(st.nextToken());
    t = Integer.parseInt(st.nextToken());

    dist = new long[n];
    p = new int[n];
    edges = new LinkedList<>();

    for (int i = 0; i < m; i++) {
      st = new StringTokenizer(in.readLine());
      int from = Integer.parseInt(st.nextToken());
      int to = Integer.parseInt(st.nextToken());
      int cost = Integer.parseInt(st.nextToken());
      if (cost == 0)
        edges.add(new Edge(from, to, -1));
      else
        edges.add(new Edge(from, to, cost));
    }

    Arrays.fill(dist, Long.MAX_VALUE);
    dist[s] = 0;
    for (int i = 0; i < n - 1; i++)
      for (Edge e : edges)
        if (dist[e.from] < Long.MAX_VALUE && dist[e.from] + e.cost < dist[e.to]) {
          dist[e.to] = dist[e.from] + e.cost;
          p[e.to] = e.from;
        }

    if (dist[t] == L) {
      System.out.println("YES");
      for (Edge e : edges)
        if (e.cost == -1) {
          System.out.println(e.from + " " + e.to + " " + (L - dist[e.from] - dist[e.to]));
        } else {
          System.out.println(e.from + " " + e.to + " " + e.cost);
        }
    } else {
      flag = false;
      for (Edge e : edges)
        if (e.cost == -1) {
          dfs(e.from, e.to, 0, L - dist[e.from] - dist[e.to]);
        }
      if (flag) {
        System.out.println("YES");
        for (Edge e : edges)
          if (e.cost == -1) {
            System.out.println(e.from + " " + e.to + " " + p[e.to]);
          } else {
            System.out.println(e.from + " " + e.to + " " + e.cost);
          }
      } else {
        System.out.println("NO");
      }
    }
  }

  static void dfs(int u, int v, long d, long w) {
    if (u == v) {
      for (Edge e : edges)
        if (e.from == u && e.to == v) {
          e.cost = (int) w;
          return;
        }
    }
    for (Edge e : edges)
      if (p[u] != v && e.to == u && dist[e.from] + e.cost + d == dist[v]) {
        dfs(e.from, v, d + e.cost, w);
        return;
      }
    flag = false;
  }
}
