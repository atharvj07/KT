#include<bits/stdc++.h>
using namespace std;

#define ll long long int
#define ull unsigned long long int
#define mod 1000000007

int main(){
    ll n;
    cin >> n;
    ll sum = 0;
    vector<ll>q(n-1);
    for(ll i = 0;i<n-1;i++){
        cin >> q[i];
        sum += q[i];
    }
    if(sum != 0){
        cout << "-1" << endl;
        return 0;
    }
    ll ans = 0;
    for(ll i = 0;i<n-1;i++){
        ans+=q[i];
        cout << ans << " ";
    }
    cout << ans+1 << endl;
    return 0;
}
