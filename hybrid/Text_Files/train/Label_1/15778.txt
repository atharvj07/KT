#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;string s;
    cin>>n>>s;
    char last=s[0];
    int times=1;
    string ans="";
    ans+=s[0];
    for(int i=1;i<n;i++){
        if(s[i]!=last){
            last=s[i];
            times=1;
            ans+=last;
        }
        else times++;
    }
    cout<<ans<<endl;
    return 0;
}
