import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] nm = reader.readLine().split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);
        int[][] matrix = new int[n][2 * m];
        for (int i = 0; i < n; i++) {
            String[] dominoes = reader.readLine().split(" ");
            for (int j = 0; j < m; j++) {
                matrix[i][2 * j] = Integer.parseInt(dominoes[j].substring(0, 1));
                matrix[i][2 * j + 1] = Integer.parseInt(dominoes[j].substring(1));
            }
        }
        int[][] reversedMatrix = new int[n][2 * m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 2 * m; j++) {
                reversedMatrix[i][j] = matrix[i][2 * m - j - 1];
            }
        }
        if (compare(matrix, reversedMatrix)) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < 2 * m; j++) {
                    System.out.print(matrix[i][j]);
                }
                System.out.println();
            }
        } else {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < 2 * m; j++) {
                    System.out.print(reversedMatrix[i][j]);
                }
                System.out.println();
            }
        }
    }

    private static boolean compare(int[][] matrix1, int[][] matrix2) {
        int[] sum1 = new int[matrix1[0].length];
        int[] sum2 = new int[matrix2[0].length];
        for (int i = 0; i < matrix1.length; i++) {
            for (int j = 0; j < matrix1[0].length; j++) {
                sum1[j] += matrix1[i][j];
                sum2[j] += matrix2[i][j];
            }
        }
        Arrays.sort(sum1);
        Arrays.sort(sum2);
        return Arrays.equals(sum1, sum2);
    }
}
