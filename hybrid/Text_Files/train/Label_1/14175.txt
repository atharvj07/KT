#include<vector>
#include<iostream>
using namespace std;
int n,ans;
vector<int>g[1001];
bool mrk[1001];
void dfs(int u,int p)
{
    mrk[u]=true;
    int cc=0;
    for(auto x:g[u])
    {
        if(x==p)continue;
        cc++;
        if(!mrk[x])dfs(x,u);
    }
    if(cc==1)ans++;
}
int main()
{
    cin>>n;
    for(int i=1;i<n;i++)
    {
        int u,v;
        cin>>u>>v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1,1);
    cout<<ans<<endl;
    return 0;
}
