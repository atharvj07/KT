import sys
input = sys.stdin.readline

n, k = map(int, input().split())
books = [(int(x), int(y), int(z)) for x, y, z in (input().split() for _ in range(n))]
books.sort()

ans = -1
cur = [0, 0, 0]
for i in range(n):
    t, a, b = books[i]
    if a: cur[0] += 1
    if b: cur[1] += 1
    cur[2] += t
    if cur[0] >= k and cur[1] >= k:
        ans = cur[2]
        break

print(ans)
