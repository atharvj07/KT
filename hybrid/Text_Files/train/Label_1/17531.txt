#include <bits/stdc++.h>
using namespace std;
const int N = 2e3 + 5;
int dp[N][N], w[N], x[N], n, m, a;
vector<pair<int, int>>v;
int main() {
  cin >> a >> n >> m;
  for (int i = 0; i < n; i++) {
    int l, r;
    cin >> l >> r;
    v.push_back({l, r});
  }
  for (int i = 0; i < m; i++) cin >> x[i] >> w[i];
  memset(dp, 0x3f, sizeof(dp));
  dp[0][0] = 0;
  sort(v.begin(), v.end());
  sort(x, x + m);
  for (int i = 0; i < a; i++) {
    for (int j = 0; j <= m; j++) {
      if (dp[i][j] == 0x3f3f3f3f) continue;
      for (int k = 0; k < n; k++) {
        if (v[k].first <= i && i + 1 <= v[k].second) {
          if (j > 0) dp[i + 1][j] = min(dp[i + 1][j], dp[i][j]);
          dp[i + 1][j + 1] = min(dp[i + 1][j + 1], dp[i][j] + w[x[j]]);
        }
      }
    }
  }
  int ans = 0x3f3f3f3f;
  for (int i = 0; i <= m; i++) ans = min(ans, dp[a][i]);
  if (ans == 0x3f3f3f3f) ans = -1;
  cout << ans << endl;
  return 0;
}
