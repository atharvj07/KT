import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        char[] s = in.readLine().toCharArray();
        int[][] subsets = new int[k][];
        for (int i = 0; i < k; i++) {
            st = new StringTokenizer(in.readLine());
            int c = Integer.parseInt(st.nextToken());
            subsets[i] = new int[c];
            st = new StringTokenizer(in.readLine());
            for (int j = 0; j < c; j++) {
                subsets[i][j] = Integer.parseInt(st.nextToken()) - 1;
            }
        }
        int[] minOperations = new int[n];
        int[][] dp = new int[n][2];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], -1);
        }
        for (int i = 0; i < n; i++) {
            minOperations[i] = dfs(i, s, subsets, dp);
        }
        for (int i = 0; i < n; i++) {
            System.out.println(minOperations[i]);
        }
    }

    public static int dfs(int i, char[] s, int[][] subsets, int[][] dp) {
        if (i == 0) {
            return 0;
        }
        if (dp[i][s[i] - '0'] != -1) {
            return dp[i][s[i] - '0'];
        }
        int min = Integer.MAX_VALUE;
        for (int[] subset : subsets) {
            if (i + 1 == subset.length) {
                int count = 0;
                for (int j = 0; j < subset.length; j++) {
                    if (s[subset[j]] == '0') {
                        count++;
                    }
                }
                min = Math.min(min, count);
            } else if (i + 1 < subset.length) {
                int count = 0;
                for (int j = 0; j < subset.length; j++) {
                    if (subset[j] <= i) {
                        if (s[subset[j]] == '0') {
                            count++;
                        }
                    } else {
                        break;
                    }
                }
                int next = dfs(i - count, s, subsets, dp);
                if (next != Integer.MAX_VALUE) {
                    min = Math.min(min, count + next);
                }
            }
        }
        dp[i][s[i] - '0'] = min;
        return min;
    }
}
