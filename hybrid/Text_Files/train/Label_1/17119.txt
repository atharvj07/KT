#include <iostream>
#include <cstring>
#include <vector>
using namespace std;
const int maxn=105;
vector<string> ans;
bool vis[26];
int main(){
  int k; cin>>k; string s; cin>>s;
  int n=s.size();
  for (int i=0;i<n;i++){
    if (!vis[s[i]-'a']){
      ans.push_back(string(1,s[i]));
      vis[s[i]-'a']=1;
      if (ans.size()==k) goto ok;
    }
  }
  cout<<"NO"<<endl;
  return 0;
  ok:;
  cout<<"YES"<<endl;
  for (string i:ans) cout<<i<<endl;
}
