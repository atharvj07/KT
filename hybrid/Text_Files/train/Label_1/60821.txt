n, t = map(int, input().split())
s1 = input()
s2 = input()

def diff(s1, s2):
    return sum(1 for c1, c2 in zip(s1, s2) if c1 != c2)

if diff(s1, s2) % 2 == t % 2 and diff(s1, s2) + t <= 2 * n:
    s3 = [' ' for _ in range(n)]
    for i in range(n):
        if s1[i] == s2[i]:
            s3[i] = s1[i]
        else:
            s3[i] = 'a'
    if diff(s1, s3) == t and diff(s2, s3) == t:
        print(''.join(s3))
        exit(0)
print(-1)
