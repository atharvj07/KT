import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int C = sc.nextInt();
        int N = sc.nextInt();
        int[][] first = new int[N][N];
        for (int i = 0; i < N; i++) {
            String line = sc.next();
            for (int j = 0; j < N; j++) {
                first[i][j] = line.charAt(j) - '0';
            }
        }
        int cnt = 0;
        int[][] prev = first;
        for (int i = 1; i < C; i++) {
            int D = sc.nextInt();
            int[][] curr = new int[N][N];
            for (int j = 0; j < N; j++) {
                System.arraycopy(prev[j], 0, curr[j], 0, N);
            }
            for (int j = 0; j < D; j++) {
                int r = sc.nextInt() - 1;
                int c = sc.nextInt() - 1;
                curr[r][c] = 1 - curr[r][c];
            }
            if (isVerticalSymmetric(curr) && isSymmetric(curr)) {
                cnt++;
            }
            prev = curr;
        }
        System.out.println(cnt);
    }

    public static boolean isVerticalSymmetric(int[][] img) {
        for (int i = 0; i < img.length; i++) {
            for (int j = 0; j < img[i].length / 2; j++) {
                if (img[i][j] != img[i][img[i].length - j - 1]) {
                    return false;
                }
            }
        }
        return true;
    }

    public static boolean isSymmetric(int[][] img) {
        for (int i = 0; i < img.length / 2; i++) {
            for (int j = 0; j < img[i].length; j++) {
                if (img[i][j] != img[img.length - i - 1][j]) {
                    return false;
                }
            }
        }
        return true;
    }
}
