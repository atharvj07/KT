for TC in range(int(input())):
    n, k, l = map(int, input().split())
    d = list(map(int, input().split()))
    p = [0, 1]
    for i in range(2, 2*k+1):
        p.append(2*k-i)
    f = True
    for i in range(n):
        if d[i]+p[(i+1)%2*k] > l:
            f=False
    if f:
        print('Yes')
    else:
        print('No')
