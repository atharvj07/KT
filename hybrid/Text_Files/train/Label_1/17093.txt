#include <bits/stdc++.h>
using namespace std;
const int N=1e6+10;
int a[N],vis[N],pri[N],cnt=0,pos[N],ans[N];
bool co[N][2];
void sieve(){
    for(int i=1;i<N;i++) vis[i]=i;
    for(int i=2;i<N;i++){
        if(vis[i]==i){
            pri[cnt++]=i;
            co[i][0]=co[i][1]=1;
            for(int j=i+i;j<N;j+=i){
                if(vis[j]==j){
                    vis[j]=i;
                    co[i][(j/i)&1]=0;
                }
            }
        }
    }
}
int main(){
    sieve();
    int n,k=0;
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>a[i];
        int x=a[i],y=0;
        while(x>1){
            if(vis[x]==x){
                co[x][y]=1;
                y^=1;
                x/=x;
            }
            else{
                int p=vis[x];
                co[p][y]=1;
                x/=p;
                y^=1;
            }
        }
    }
    bool f=0;
    for(int i=0;i<n-1;i++){
        int x=a[i],y=0;
        while(x>1){
            int p=vis[x];
            if(!co[p][y^1]){
                ans[k++]=a[i];
                pos[k]=i;
                f=1;
                break;
            }
            x/=p;
            y^=1;
        }
        if(f) break;
    }
    cout<<k<<endl;
    for(int i=0;i<n;i++) cout<<a[i]<<" ";
    for(int i=0;i<k;i++){
        int p=pos[i+1];
        a[p]=ans[i];
        for(int j=p;j<n;j++) cout<<" "<<a[j];
        cout<<endl;
    }
}
