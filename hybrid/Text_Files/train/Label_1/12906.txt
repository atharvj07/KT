#include <bits/stdc++.h>
using namespace std;

const int MM = 1e5 + 5;

int n, m, u, v, tot, d[MM];
vector<int> adj[MM];

bool ok() {
    memset(d, -1, sizeof d);
    queue<int> q;
    for (int i = 1; i <= n; i++) {
        if (adj[i].empty()) { continue; }
        if (d[i] != -1) { continue; }
        d[i] = 0;
        q.push(i);
        while (!q.empty()) {
            int p = q.front();
            q.pop();
            for (int x : adj[p]) {
                if (d[x] == -1) {
                    d[x] = d[p] + 1;
                    q.push(x);
                } else if (d[x] == d[p]) {
                    return 1;
                }
            }
        }
    }
    return 0;
}

int main() {
    cin >> n >> m;
    while (m--) {
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    if (!ok()) {
        cout << 0 << ' ' << 1 << '\n';
        return 0;
    }
    for (int i = 1; i <= n; i++) { tot += adj[i].size() % 2; }
    cout << tot / 2 << ' ' << (1 << (tot / 2)) << '\n';
    return 0;
}
