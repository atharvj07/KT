#include <bits/stdc++.h>
using namespace std;

struct D { int x, y; };

const int MAXN = 1e6 + 5;

int n, m, k;
int h[MAXN], u[MAXN], v[MAXN];
bool vis[MAXN];

int main() {
    cin >> n >> m >> k;
    for (int i = 0; i < m; ++i) cin >> h[i];
    for (int i = 0; i < k; ++i) cin >> u[i] >> v[i];

    int hole = -1, pos = 1;
    for(int i = 0; i < k; ++i){
        if(hole != -1) continue;
        if(u[i] == pos) swap(pos, v[i]), hole = v[i];
        else if(v[i] == pos) swap(pos, u[i]), hole = u[i];
        else if(!vis[u[i]]) vis[u[i]] = vis[v[i]] = 1;
    }

    if(hole == -1) hole = pos;

    for (int i = 0; i < m; ++i) vis[h[i]] ? --hole : vis[h[i]] = 1;

    cout << hole << '\n';
    return 0;
}
