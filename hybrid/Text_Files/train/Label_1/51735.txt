import java.util.Scanner;

public class PerfectMatchings {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        int[][] probabilities = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                probabilities[i][j] = scanner.nextInt();
            }
        }

        long[][] dp = new long[1 << n][n + 1];
        for (int i = 0; i < (1 << n); i++) {
            for (int j = 0; j <= n; j++) {
                dp[i][j] = -1;
            }
        }

        long answer = solve(probabilities, 0, 0, 0, dp);
        long inverse = modInverse(pow(2, n - 1, MOD), MOD);
        System.out.println((answer * inverse) % MOD);
    }

    private static long solve(int[][] probabilities, int mask, int i, int j, long[][] dp) {
        if (i == probabilities.length) {
            return j == probabilities.length ? 1 : 0;
        }

        if (dp[mask][j] != -1) {
            return dp[mask][j];
        }

        long result = 0;
        for (int k = 0; k < probabilities.length; k++) {
            if (((mask >> k) & 1) == 0 && probabilities[i][k] > 0) {
                result = (result + solve(probabilities, mask | (1 << k), i + 1, j + 1, dp) * probabilities[i][k]) % MOD;
            }
        }

        return dp[mask][j] = result;
    }

    private static long pow(long a, long b, long mod) {
        long result = 1;
        while (b > 0) {
            if ((b & 1) == 1) {
                result = (result * a) % mod;
            }
            a = (a * a) % mod;
            b >>= 1;
        }
        return result;
    }

    private static long modInverse(long a, long mod) {
        return pow(a, mod - 2, mod);
    }
}
