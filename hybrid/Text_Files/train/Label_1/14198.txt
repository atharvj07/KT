#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const ll mod=1e9+7;
const int maxn=1e5+1;
string s[maxn];
ll n,m,skip[maxn][2],l,r,i,j,k,fl;
int main(){
    cin>>n>>m;
    for(i=1;i<=n;i++) cin>>s[i];
    l=r=0;
    skip[n][0]=skip[n][1]=0;
    for(i=n-1;i>0;i--){
        skip[i][0]=skip[i+1][0];
        skip[i][1]=skip[i+1][1];
        if(s[i+1][0]=='+') skip[i][0]++;
        if(s[i+1][m-1]=='+') skip[i][1]++;
        if(s[i][1]=='+') skip[i][1]++;
    }
    if(s[1][0]=='+') l++;
    if(s[1][m-1]=='+') r++;
    for(i=1;;i++){
        if(l==0){
            if(s[i][0]=='+') l++;
            else if(s[i][1]=='+'){
                l++; fl=1;
            }
            else{
                if(fl) r--;
                else l--;
            }
        }
        else if(r==0){
            if(s[i][m-1]=='+') r++;
            else if(s[i][0]=='+'){
                r++; fl=0;
            }
            else{
                if(fl) l--;
                else r--;
            }
        }
        if(i==n) return cout<<i<<endl,0;
        if(l<0||r<0) return cout<<"Never"<<endl,0;
    }
    return cout<<"Never"<<endl,0;
}
