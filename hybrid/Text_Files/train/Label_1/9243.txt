#include <iostream>
#include <map>

int main() {
  int n;
  std::cin >> n;
  std::map<int, int> last;
  bool ok = true;
  for (int i = 0; i < n; i++) {
    int x, k;
    std::cin >> x >> k;
    ok &= (last[k] <= x);
    last[k] = x + 1;
  }
  std::cout << (ok ? "YES" : "NO") << std::endl;
  return 0;
}
