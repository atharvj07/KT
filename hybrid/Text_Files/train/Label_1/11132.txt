#include<bits/stdc++.h>
using namespace std;
int main() {
  int t; cin >> t;
  while (t--) {
    long long n, k; cin >> n >> k;
    if (n >= 64) cout << "YES " << 63 << endl;
    else if ((k & 1) || (n > 62) || (n != (n & k))) cout << "NO" << endl;
    else cout << "YES " << n - 1 << endl;
  }
  return 0;
}
