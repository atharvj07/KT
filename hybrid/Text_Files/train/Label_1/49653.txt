import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    long[] a = new long[n + 1];
    for (int i = 1; i <= n; i++) {
      a[i] = scanner.nextInt();
    }
    long min = Long.MAX_VALUE;
    for (int i = 1; i <= n; i++) {
      min = Math.min(min, a[i] / i);
    }
    for (int i = 1; i <= n; i++) {
      if (a[i] / i == min) {
        System.out.println(i);
        break;
      }
    }
  }
}
