import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int q = in.nextInt();
        for (int i = 0; i < q; i++) {
            long l = in.nextLong();
            long r = in.nextLong();
            long d = in.nextLong();
            if (l % d == 0) l++;
            if (l <= r) System.out.println(l);
            else System.out.println((r / d + 1) * d);
        }
    }
}
