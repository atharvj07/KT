import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 0) {
                ans++;
            }
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += Math.abs(a[i]);
        }
        for (int i = 0; i < n; i++) {
            if (a[i] != 0) {
                sum -= Math.abs(a[i]);
                if (sum <= Math.abs(b[i])) {
                    ans++;
                }
                sum += Math.abs(a[i]);
            }
        }
        System.out.println(ans);
    }
}
