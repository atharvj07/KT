#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  std::vector<long long> p(t), q(t);
  for (int i = 0; i < t; i++) { std::cin >> p[i] >> q[i]; }
  for (int i = 0; i < t; i++) {
    long long ans = 1;
    while (p[i] % ans == 0 && ans % q[i] == 0) { ans++; }
    std::cout << ans << '\n';
  }
  return 0;
}
