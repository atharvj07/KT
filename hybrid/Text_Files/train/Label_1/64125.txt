import sys

n = int(input())
tot = 0.
cnt = 1
for _ in range(n):
    cmd, *lst = map(int, input().split())
    if cmd == 1:
        x = lst[1]
        tot += x * min(lst[0], cnt)
    elif cmd == 2:
        tot += lst[0]
        cnt += 1
    else:
        tot -= lst[0]
        cnt -= 1
    print(tot / cnt)
