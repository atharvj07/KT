import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int m = Integer.parseInt(line[0]);
        int d = Integer.parseInt(line[1]);
        List<Map<Integer, Set<Integer>>> subspaces = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int k = Integer.parseInt(br.readLine());
            Map<Integer, Set<Integer>> subspace = new HashMap<>();
            for (int j = 0; j < k; j++) {
                line = br.readLine().split(" ");
                int[] v = new int[d];
                for (int l = 0; l < d; l++) {
                    v[l] = Integer.parseInt(line[l]);
                }
                int norm = 0;
                for (int val : v) {
                    norm += val * val;
                }
                if (!subspace.containsKey(norm)) {
                    subspace.put(norm, new HashSet<>());
                }
                for (int l = 0; l < d; l++) {
                    if (v[l] != 0) {
                        subspace.get(norm).add(l);
                    }
                }
            }
            subspaces.add(subspace);
        }
        int[] groups = new int[m];
        Arrays.fill(groups, -1);
        int curGroup = 0;
        for (int i = 0; i < m; i++) {
            if (groups[i] == -1) {
                groups[i] = curGroup++;
                for (int j = i + 1; j < m; j++) {
                    if (subspaces.get(i).equals(subspaces.get(j))) {
                        groups[j] = groups[i];
                    }
                }
            }
        }
        for (int group : groups) {
            System.out.print(group + 1 + " ");
        }
        System.out.println();
    }
}
