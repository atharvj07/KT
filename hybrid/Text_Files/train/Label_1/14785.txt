#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<bool> divisible(n + 2, 0);
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      if (x % (i + 2) == 0) {
        divisible[i + 2] = 1;
      }
    }
    bool possible = 1;
    for (int i = 2; i <= n + 1; i++) {
      possible &= (divisible[i] == 0);
    }
    cout << (possible ? "YES" : "NO") << endl;
  }
  return 0;
}
