s=input()
m=int(input())
qs=[]
for i in range(m):
    x,y=map(int,input().split())
    qs.append([x,y])
dict={}
for i in range(1,len(s)):
    if s[i]==s[i-1]:
        if s[i] not in dict:
            dict[s[i]]=[i]
        else:
            dict[s[i]].append(i)
ans=[]
for l,r in qs:
    if s[l-1]==s[r-1]:
        if not s[l-1] in dict or r-1 not in dict[s[l-1]]:
            ans.append(0)
        else:
            pos=dict[s[l-1]].index(r-1)
            ans.append(len(dict[s[l-1]])-pos)
    else:
        ans.append(0)
print(*ans)
