n = int(input())
a = list(map(int, input().split()))
m = int(input())
p = [list(map(int, input().split())) for _ in range(m)]

t = 0
for x in a:
    while t + x > p[-1][1]:
        if len(p) <= 1:
            print(-1)
            exit()
        p.pop(0)
    t += x
    for i in p:
        if i[0] <= t <= i[1]:
            break
    else:
        print(-1)
        exit()

print(t)
