#include<bits/stdc++.h>
using namespace std;
int main() {
	int n;
	cin >> n;
	int a[n];
	for (int i = 0; i < n; i++) {
		cin >> a[i];
	}
	int xr = 0;
	set<int> s;
	s.insert(0);
	for (int i = 0; i < n; i++) {
		xr ^= a[i];
		s.insert(xr);
		for (int j = i + 1; j < n; j++) {
			xr ^= a[j];
			s.insert(xr);
			xr ^= a[j];
		}
	}
	cout << s.size() << endl;
	for (auto it : s) {
		cout << it << " ";
	}
	cout << "\n";
	return 0;
}
