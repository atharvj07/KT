n, a, b = map(int, input().split())
if a + b > n * (n - 1) / 2:
    print("NO")
else:
    print("YES")
    adj = [[0] * n for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            if i != j:
                if a > 0:
                    a -= 1
                    adj[i][j] = 1
                    adj[j][i] = 1
                elif b > 0:
                    b -= 1
                    pass
                else:
                    adj[i][j] = 1
                    adj[j][i] = 1
    for i in range(n):
        print(' '.join(map(str, adj[i])))
