#include <bits/stdc++.h>
using namespace std;
const int N = 2e5+1;
int a[N], b[N], pre[N];
vector<int> pos[N];
int main() {
    int n, m, x, k, y;
    cin >> n >> m >> x >> k >> y;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        pre[i] = pre[i-1] + (a[i] == a[i-1]);
        pos[a[i]].push_back(i);
    }
    for (int i = 1; i <= m; i++) cin >> b[i];
    bool ok = 1;
    long long ans = 0;
    for (int i = 1, j = 1; i <= m; i++) {
        while (j <= n && a[j] != b[i]) {
            ans += 1LL*y;
            if (pre[j] != pre[j-1]) ok = 0;
            j++;
        }
        if (j > n || a[j] != b[i]) { ok = 0; break; }
        if (j - k >= 1 && a[j-k] == b[i-1]) {
            ans += 1LL*x;
            j = j - k + 1;
            continue;
        }
        if (j + 1 <= n && a[j+1] == b[i-1]) {
            ans += 1LL*x;
            j += 2;
            continue;
        }
        if (pos[b[i]].size() == 1) { ok = 0; break; }
        ans += 1LL*y;
        j = *lower_bound(pos[b[i]].begin(), pos[b[i]].end(), j) + 1;
    }
    cout << (ok ? ans : -1) << '\n';
    return 0;
}
