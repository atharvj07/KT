n = int(input())
d = {'North': 0, 'South': 0, 'West': 0, 'East': 0}
for _ in range(n):
    x, dir = input().split()
    x = int(x)
    if (dir == 'North' and d['South'] > 0) or (dir == 'South' and d['North'] > 0):
        print('NO')
        exit()
    if (dir == 'South' and (d['South'] + x > 20000 or d['North'] > 0)) or (dir == 'North' and (d['North'] + x > 20000 or d['South'] > 0)):
        print('NO')
        exit()
    if dir == 'North':
        d['North'] += x
    elif dir == 'South':
        d['South'] += x
print('YES' if d['North'] > 0 and d['South'] == 0 and d['North'] == 20000 else 'NO')
