import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        long l = sc.nextLong();
        long r = sc.nextLong();
        long mod = 1_000_000_007;
        long ans = (r - l + 1) * (r - l) % mod / 2;
        ans = (ans * k % mod) * k % mod;
        System.out.println(ans);
    }
}
