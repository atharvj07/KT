import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        scanner.close();

        long count = 0;
        for (long i = 1; i <= n; i++) {
            if (isValid(i)) {
                count++;
            }
        }
        System.out.println(count);
    }

    private static boolean isValid(long n) {
        while (n > 0) {
            if (n % 10 != 0 && n % 10 != 1) {
                return false;
            }
            n /= 10;
        }
        return true;
    }
}
