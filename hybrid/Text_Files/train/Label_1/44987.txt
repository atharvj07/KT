import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), h = sc.nextInt();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            ans += Math.min(sc.nextInt(), h) == h ? 1 : 2;
        }
        System.out.println(ans);
    }
}
