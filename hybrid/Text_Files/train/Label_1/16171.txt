#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 2e6 + 5;
int32_t main() {
	ios_base::sync_with_stdio(0);
	cin.tie(0);

	int n; cin >> n;
	unordered_map<ll, int> mp;
	while (n--) {
		int x1, y1, x2, y2; cin >> x1 >> y1 >> x2 >> y2;
		if (x1 > x2) swap(x1, x2);
		if (y1 > y2) swap(y1, y2);
		if (x1 == x2) {
			for (int y = y1; y <= y2; y++) {
				mp[1LL * x1 * N + y]++;
			}
		}
		else if (y1 == y2) {
			for (int x = x1; x <= x2; x++) {
				mp[1LL * x * N + y1]++;
			}
		}
		else {
			int dx = x2 - x1, dy = y2 - y1, g = __gcd(dx, dy);
			dx /= g, dy /= g;
			for (int x = x1, y = y1; x <= x2; x += dx, y += dy) {
				mp[1LL * x * N + y]++;
			}
		}
	}
	cout << mp.size() << endl;

	return 0;
}
