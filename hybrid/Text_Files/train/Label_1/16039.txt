#include <bits/stdc++.h>
using namespace std;
bool v[200005];
int main() {
    long long n, x, y;
    cin >> n >> y;
    for (int i = 0; i < n; i++) {
        x >>= 1LL;
        x += (1LL << (n - 1)) * (y & 1);
        v[x] = 1;
        y >>= 1LL;
    }
    long long ans = 0;
    for (long long i = 0; i < (1LL << n); i++)
        if (v[i] && v[i ^ x]) ans++;
    cout << ans;
    return 0;
}
