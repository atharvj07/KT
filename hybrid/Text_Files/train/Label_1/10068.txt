#include<bits/stdc++.h>
#define int long long
using namespace std;
int32_t main(){
    ios_base::sync_with_stdio(0);
    cin.tie(nullptr); cout.tie(nullptr);
    int t; cin >> t; while(t--){
        int ans = 0, n; cin >> n;
        for (int i = 2; i*i <= n; i++){
            if (n % i == 0){
                ans += (n / i -1);
                while(n % i == 0) n/= i;
            }
        }
        if (n > 1) ans += (n - 1);
        cout << ans << endl;
    }
}
