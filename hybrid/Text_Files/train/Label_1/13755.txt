#include<bits/stdc++.h>
using namespace std;
int main() {
    string a, b, c, d;
    getline(cin, a, '\n');
    getline(cin, b, '\n');
    getline(cin, c, '\n');
    getline(cin, d, '\n');
    int l1 = a.size() - 3, l2 = b.size() - 3, l3 = c.size() - 3, l4 = d.size() - 3;
    if(l1 >= 2 * l2 && l1 >= 2 * l3 && l1 >= 2 * l4) cout << 'A';
    else if(l2 >= 2 * l1 && l2 >= 2 * l3 && l2 >= 2 * l4) cout << 'B';
    else if(l3 >= 2 * l1 && l3 >= 2 * l2 && l3 >= 2 * l4) cout << 'C';
    else if(l4 >= 2 * l1 && l4 >= 2 * l2 && l4 >= 2 * l3) cout << 'D';
    else cout << 'C';
    return 0;
}
