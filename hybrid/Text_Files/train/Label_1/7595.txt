#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int &x : v) cin >> x;
    int ans = abs(v[0] - v[1]);
    bool f = false;
    for (int i = 0; i < n - 1; i += 2) {
      int x = v[i] - v[i + 1];
      if (!f) {
        f = true;
        ans = x;
      } else if (ans != x) {
        ans = -1;
        break;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
