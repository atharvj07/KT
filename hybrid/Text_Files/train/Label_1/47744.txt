import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] l = new int[n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            l[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        sc.close();

        int[] dp = new int[1000001];
        Arrays.fill(dp, -1);
        dp[0] = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 1000000; j >= l[i]; j--) {
                if (dp[j - l[i]] != -1) {
                    dp[j] = Math.min(dp[j], dp[j - l[i]] + c[i]);
                }
            }
        }

        System.out.println(dp[1000000] == -1 ? -1 : dp[1000000]);
    }
}
