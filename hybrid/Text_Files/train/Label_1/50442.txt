import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] d = new int[N + 2];
        int[] f = new int[N + 2];
        int[] b = new int[N + 2];
        for (int i = 1; i <= N; i++) {
            d[i] = sc.nextInt();
        }
        f[1] = d[1];
        for (int i = 2; i <= N; i++) {
            f[i] = Math.max(f[i - 1], d[i] + d[i - 1]);
        }
        b[N] = d[N];
        for (int i = N - 1; i >= 1; i--) {
            b[i] = Math.max(b[i + 1], d[i] + d[i + 1]);
        }
        boolean ok = true;
        for (int i = 2; i < N; i++) {
            if (f[i - 1] < i - 1 || b[i + 1] < N - i) {
                ok = false;
            }
        }
        System.out.println(ok ? "yes" : "no");
    }
}
