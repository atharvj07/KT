#include <iostream>
#include <string>
#include <map>

int main() {
  std::string l1, l2, s;
  std::map<char, char> m;
  std::cin >> l1 >> l2 >> s;
  for (int i = 0; i < 26; i++) {
    m[l1[i]] = l2[i];
  }
  for (auto& c : s) {
    if (std::islower(c)) {
      std::cout << m[c];
    } else {
      std::cout << c;
    }
  }
  return 0;
}
