#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> g(n + 1);  // Graph of debts
    vector<int> balance(n + 1, 0);  // Balance of debts for each person

    for (int i = 0; i < m; i++) {
        int a, b, c;
        cin >> a >> b >> c;
        g[a].push_back(c);
        g[b].push_back(-c);
        balance[a] -= c;
        balance[b] += c;
    }

    int sum = 0;  // Total sum of debts to be minimized

    while (true) {
        bool found = false;
        for (int i = 1; i <= n; i++) {
            if (balance[i] > 0) {  // Person i owes money
                for (int j = 1; j <= n; j++) {
                    if (balance[j] < 0 && g[i][j - 1] > 0) {  // Person j is owed money
                        int debt = min(balance[i], -balance[j]);
                        g[i][j - 1] -= debt;
                        g[j][i - 1] += debt;
                        balance[i] -= debt;
                        balance[j] += debt;
                        sum += debt;
                        found = true;
                        break;
                    }
                }
            }
        }
        if (!found) break;
    }

    cout << sum << endl;
    return 0;
}
