def ext_gcd(a, b):
    if a == 0:
        return b, 0, 1
    g, x, y = ext_gcd(b % a, a)
    return g, y - (b // a) * x, x

def solve(u, v):
    if u > v or (u & 1) != (v & 1):
        return -1
    g, vx, vy = ext_gcd(u, v)
    if g != 1:
        return -1
    x = (vx * v) % u
    n = (v - x) // u
    ans = [x]
    for i in range(n):
        ans.append(u)
    return len(ans), ans

u, v = map(int, input().split())
ans = solve(u, v)
if ans == -1:
    print(-1)
else:
    print(ans[0])
    print(*ans[1])
