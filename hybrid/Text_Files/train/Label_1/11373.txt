#include <bits/stdc++.h>

using namespace std;

const int MX = 1e5 + 5, LG = 20;
int n, sz[MX], ans;
vector<int> adj[MX];

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        cin >> n;

        for (int i = 1; i <= n; i++) {
            cin >> sz[i];
            adj[i].clear();
        }

        stack<pair<int, int>> st;
        for (int i = n; i >= 1; i--) {
            while (!st.empty() && st.top().first >= sz[i]) { st.pop(); }
            if (!st.empty()) { adj[st.top().second].push_back(i); }
            st.push({sz[i], i});
        }

        st = stack<pair<int, int>>();
        for (int i = 1; i <= n; i++) {
            while (!st.empty() && st.top().first <= sz[i]) { st.pop(); }
            if (!st.empty()) { adj[i].push_back(st.top().second); }
            st.push({sz[i], i});
        }

        ans = 1;
        function<void(int, int)> dfs = [&](int u, int p) {
            ans = max(ans, 1 + (p != -1));
            for (auto v : adj[u]) {
                if (v != p) { dfs(v, u); }
            }
        };

        for (int i = 1; i <= n; i++) { dfs(i, -1); }

        cout << ans << '\n';
    }
}
