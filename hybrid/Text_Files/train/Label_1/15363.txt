#include <bits/stdc++.h>
using namespace std;
#define mod 1000000007
int main() {
  int n, m;
  cin >> n >> m;
  string s;
  cin >> s;
  vector<int> dp(m + 1, 0);
  dp[m] = 1;
  for (int i = m - 1; i >= 0; i--) {
    if (s[i] == '(') {
      dp[i] = 0;
      if (i + 1 < m && dp[i + 1] > 0 && s[i + 1] == ')') { dp[i] = (dp[i] + dp[i + 2]) % mod; }
      if (i + 2 < m && dp[i + 2] > 0) { dp[i] = (dp[i] + dp[i + 2]) % mod; }
    } else {
      dp[i] = dp[i + 1];
      if (i + 1 < m && dp[i + 1] > 0 && s[i + 1] == ')') { dp[i] = (dp[i] + dp[i + 2]) % mod; }
    }
  }
  cout << dp[0] << endl;
}
