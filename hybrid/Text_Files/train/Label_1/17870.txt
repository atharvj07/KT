#include <bits/stdc++.h>
#define int long long
using namespace std;
const int maxn = 1e5 + 5;
int n, m, k, a[maxn], b[maxn], pos[maxn], cur[maxn], p[maxn], ans, cnt;
signed main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        pos[a[i]] = i;
    }
    for (int i = 1; i <= m; i++) {
        cin >> b[i];
    }
    for (int i = 1; i <= n; i++) {
        cur[i] = pos[i];
    }
    for (int i = 1; i <= m; i++) {
        ans += (cur[b[i]] - 1) / k;
        if (cur[b[i]] - 1 % k) ans++;
        int t = a[cur[b[i]] - 1];
        if (cur[b[i]] > 1) {
            cnt--;
            p[cnt] = t;
        }
        cur[b[i]]--;
        if (cur[b[i]] == 0) {
            cur[b[i]] = n;
            cnt++;
            p[cnt] = b[i];
        }
        int tt = n;
        for (int j = cnt; j >= 1; j--) {
            int x = p[j];
            pos[x] = tt;
            a[tt] = x;
            tt--;
        }
        for (int j = 1; j <= n; j++) {
            cur[a[j]] = j;
        }
    }
    cout << ans << endl;
    return 0;
}
