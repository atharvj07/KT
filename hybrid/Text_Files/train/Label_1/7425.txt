#include <vector>
#include <iostream>
using namespace std;

void find_primes(int n, vector<int> &primes)
{
    int i = 2;
    while (i * i <= n)
    {
        if (n % i == 0)
        {
            primes.push_back(i);
            while (n % i == 0)
            {
                n /= i;
            }
        }
        i += 1;
    }
    if (n > 1)
    {
        primes.push_back(n);
    }
}

int main()
{
    int n;
    cin >> n;

    vector<int> primes;
    find_primes(n, primes);

    cout << primes.size() << endl;
    for (int prime : primes)
    {
        cout << prime << " ";
    }
    cout << endl;

    return 0;
}
