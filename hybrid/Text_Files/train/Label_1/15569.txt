#include <iostream>
#include <vector>
#include <cmath>

const int mod = 1e9+7;

int main() {
    long long n, ans = 1, f = 1;
    std::cin >> n;
    std::vector<int> v(n);
    for (long long i = 0; i < n; ++i) {
        char c; std::cin >> c;
        v[i] = (c == 'f');
    }
    for (long long i = 0; i < n; ++i) {
        if (v[i]) {
            ans = ans * f % mod;
            f = f * 2 % mod;
        }
        else f--;
    }
    std::cout << ans << std::endl;
    return 0;
}
