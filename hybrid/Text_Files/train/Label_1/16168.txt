#include <bits/stdc++.h>
using namespace std;

struct Point {
    int x, y;
    bool operator<(const Point& p) const { return tie(x, y) < tie(p.x, p.y); }
};

int main() {
    int n;
    cin >> n;
    vector<Point> p(2 * n);
    for (int i = 0; i < 2 * n; i += 2) {
        cin >> p[i].x >> p[i].y >> p[i + 1].x >> p[i + 1].y;
    }
    sort(p.begin(), p.end());
    set<Point> st;
    for (int i = 0; i < 2 * n; i += 2) {
        int ax = p[i].x, ay = p[i].y, bx = p[i + 1].x, by = p[i + 1].y;
        if (ax > bx) swap(ax, bx);
        if (ay > by) swap(ay, by);
        if (ax == bx) {
            for (int y = ay; y <= by; y++) st.insert({ax, y});
        } else {
            for (int x = ax; x <= bx; x++) st.insert({x, ay});
        }
    }
    cout << (int)st.size() << endl;
    return 0;
}
