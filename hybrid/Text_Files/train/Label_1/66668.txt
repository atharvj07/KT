import sys
input = sys.stdin.readline

n = int(input())
a = [int(x) for x in input().split()]
m = int(input())
b = [int(x) for x in input().split()]

sa = [0] * n
sb = [0] * m
sa[0] = a[0]
sb[0] = b[0]
for i in range(1, n):
    sa[i] = sa[i - 1] + a[i]
for i in range(1, m):
    sb[i] = sb[i - 1] + b[i]

if sa[n - 1] != sb[m - 1]:
    print(-1)
else:
    i = j = ans = 0
    while i < n and j < m:
        if sa[i] == sb[j]:
            i += 1
            j += 1
            ans += 1
        elif sa[i] < sb[j]:
            i += 1
        else:
            j += 1
    print(ans)
