import java.util.*;

public class Main {
    static int n, sum, ans;
    static int[] deg, r;
    static ArrayList<Integer>[] a;

    static boolean dfs(int u, int p) {
        sum += r[u];
        ans = Math.max(ans, sum);
        int mx = 0;
        for (int v : a[u]) {
            if (v == p)
                continue;
            if (r[v] > r[u]) {
                if (dfs(v, u))
                    return true;
            } else {
                mx = Math.max(mx, deg[v]);
            }
        }
        if (mx <= 1)
            return true;
        return false;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        r = new int[n];
        a = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            a[i] = new ArrayList<>();
        }
        deg = new int[n];
        for (int i = 0; i < n; i++)
            r[i] = sc.nextInt();
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            a[u].add(v);
            a[v].add(u);
            deg[u]++;
            deg[v]++;
        }
        for (int i = 0; i < n; i++) {
            if (deg[i] == 1) {
                sum = 0;
                if (dfs(i, -1))
                    break;
            }
        }
        System.out.println(ans);
    }
}
