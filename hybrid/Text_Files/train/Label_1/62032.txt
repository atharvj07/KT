def min_paint(n, m, k, colors, paint):
    # Calculate initially formed groups and uncolored trees
    groups = 1 if colors[0] != 0 else 0
    uncolored = 0
    for i in range(1, n):
        if colors[i] == 0:
            uncolored += 1
        elif colors[i] != colors[i-1]:
            groups += 1
    
    # If already more groups than k or impossible to achieve k, return -1
    if groups > k or groups + uncolored < k:
        return -1
    
    # If no uncolored trees and groups match k, return 0
    if uncolored == 0 and groups == k:
        return 0
    
    # DP[i][j] = minimum paint needed to color first i trees with j groups formed
    dp = [[float('inf')] * (k + 1) for _ in range(uncolored + 1)]
    dp[0][groups] = 0
    
    # Indexes of uncolored trees
    uncolored_indexes = [i for i, c in enumerate(colors) if c == 0]
    
    for i in range(1, uncolored + 1):
        for j in range(1, k + 1):
            for color in range(1, m + 1):
                cost = paint[uncolored_indexes[i-1]][color-1]
                if i == 1 or (colors[uncolored_indexes[i-2]] != color and i-2 >= 0):
                    # If coloring this tree forms a new group
                    dp[i][j] = min(dp[i][j], dp[i-1][j-1] + cost)
                else:
                    # If coloring this tree continues the last group
                    dp[i][j] = min(dp[i][j], dp[i-1][j] + cost)
    
    # The answer is the minimum paint needed to color all uncolored trees forming exactly k groups
    answer = min(dp[uncolored])
    return -1 if answer == float('inf') else answer

# Input reading
n, m, k = map(int, input().split())
colors = list(map(int, input().split()))
paint = [list(map(int, input().split())) for _ in range(n)]

print(min_paint(n, m, k, colors, paint))
