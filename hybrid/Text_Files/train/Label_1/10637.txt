#include<bits/stdc++.h>
using namespace std;
int32_t main(){
    int t; cin>>t;
    while(t--){
        int a,b,n;
        cin>>a>>b>>n;
        int ans=0;
        while(a<=n&&b<=n){
            if(a<b){
                a+=b;
                ans++;
            }
            else{
                b+=a;
                ans++;
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
