s=input()
for i in range(len(s)):
    if s[i]!='.':
        if i-1>=0:
            if s[i-1]!=s[i] and s[i-1]!='.':
                print('Yes')
                exit()
        if i+1<len(s):
            if s[i+1]!=s[i] and s[i+1]!='.':
                print('Yes')
                exit()
print('No')
