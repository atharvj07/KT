#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
LL a[1000001], l[1000001], r[1000001], c[1000001];
int main() {
  int n, i;
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%lld", &a[i]);
    l[a[i]]++;
  }
  for (i = 1000000; i >= 1; i--) l[i] += l[i + 1];
  for (i = 1; i <= n; i++) {
    c[a[i]]++;
    if (l[a[i]] > c[a[i]]) r[a[i]]++;
  }
  LL ans = 0;
  for (i = 1; i <= n; i++) ans += r[a[i]];
  cout << ans << endl;
  return 0;
}
