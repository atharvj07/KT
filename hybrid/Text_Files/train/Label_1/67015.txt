n, k = map(int, input().split())

if k == 2:
    print(n - 1)
    colors = []
    for i in range(1, n):
        for j in range(i + 1, n + 1):
            colors.append((i - 1) % (n - 1) + 1)
    print(*colors)
else:
    print(2)
    colors = []
    for i in range(1, n):
        for j in range(i + 1, n + 1):
            if j == i + 1:
                colors.append(1)
            else:
                colors.append(2)
    print(*colors)
