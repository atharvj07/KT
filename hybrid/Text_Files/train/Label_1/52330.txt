import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] m = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            int x = (k + 1) / 2;
            int y = (k + 1) / 2 - m[i] + 1;
            if (y <= 0 || y + m[i] - 1 > k) {
                System.out.println(-1);
                continue;
            }
            System.out.println(x + " " + y + " " + (y + m[i] - 1));
        }
    }
}
