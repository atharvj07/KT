from collections import defaultdict

n, k = map(int, input().split())
a = list(map(int, input().split()))
count = defaultdict(int)
count[a[0]] += 1
l = r = mx = 0
for i in range(1, n):
    count[a[i]] += 1
    while len(count) > k:
        count[a[l]] -= 1
        if count[a[l]] == 0:
            del count[a[l]]
        l += 1
    if i - l + 1 > mx:
        mx = i - l + 1
        r = i
print(l + 1, r + 1)
