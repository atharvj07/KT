import sys

n, m = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
count = {}
for i in a:
    if i not in count:
        count[i] = 0
    count[i] += 1
for i in range(m):
    l = int(sys.stdin.readline())
    ans = 0
    for j in a[l - 1:]:
        if j not in count:
            break
        ans += count[j] > 0
        count[j] -= 1
    print(ans)
