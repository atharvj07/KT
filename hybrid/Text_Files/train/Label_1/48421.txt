import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }
        scanner.close();
        int[] freq = new int[n + 1];
        for (int i = 0; i < m; i++) {
            freq[a[i]]++;
        }
        int min_pages = n;
        for (int i = 1; i <= n; i++) {
            if (freq[i] > 1) {
                min_pages = Math.min(min_pages, m - freq[i]);
            }
        }
        System.out.println(min_pages);
    }
}
