#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> dances(m, vector<int>(3));
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < 3; j++) {
      cin >> dances[i][j];
    }
  }
  vector<int> colors(n, 0);
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < 3; j++) {
      if (colors[dances[i][j] - 1] == 0) {
        colors[dances[i][j] - 1] = j + 1;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    cout << colors[i] << " ";
  }
  cout << endl;
}
