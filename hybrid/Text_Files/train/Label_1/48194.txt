import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] left = new int[n], right = new int[n];
        for (int i = 0; i < n; i++) {
            left[i] = sc.nextInt() - 1;
            right[i] = sc.nextInt();
        }
        int[] ans = new int[m];
        for (int d = 1; d <= m; d++) {
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                if (left[i] <= d && d <= right[i]) {
                    cnt++;
                }
            }
            ans[d - 1] = cnt;
        }
        for (int d : ans) {
            System.out.println(d);
        }
    }
}
