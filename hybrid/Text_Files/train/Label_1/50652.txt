import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long ans = n * (n - 1) * (n - 2) / 6 - m * (m - 1) * (m - 2) / 6;
        System.out.println(ans);
    }
}
