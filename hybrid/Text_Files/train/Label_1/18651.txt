#include <bits/stdc++.h>
using namespace std;

#define endl "\n"

int main() {
  string s;
  cin >> s;

  int n = s.size();

  int x = 0, y = 0;
  int ans = 0;

  for (int i = 0; i < n; i++) {
    if (s[i] == 'U') {
      y += 1;
    } else if (s[i] == 'D') {
      y -= 1;
    } else if (s[i] == 'L') {
      x -= 1;
    } else {
      x += 1;
    }

    if (x == 0 && y == 0) {
      ans = i;
      break;
    }
  }

  if (x != 0 || y != 0) {
    ans = -1;
  }

  cout << ans << endl;

  return 0;
}
