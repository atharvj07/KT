#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[25], ch = 0, bi = 0, ba = 0, m = 0;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) {
    if (i % 3 == 1)
      ch += a[i];
    else if (i % 3 == 2)
      bi += a[i];
    else
      ba += a[i];
  }
  if (ch >= bi && ch >= ba)
    cout << "chest\n";
  else if (bi >= ch && bi >= ba)
    cout << "biceps\n";
  else
    cout << "back\n";
  return 0;
}
