import sys

t = int(input())

for _ in range(t):
    n = int(input())
    a = [int(x) for x in input().split()]
    d = {0: 0, 1: 0, 2: 0}
    for i in a:
        d[i % 3] += 1

    ans = 0
    rem = (3 - max(d.values()) % 3) % 3
    for key, val in d.items():
        if key == rem:
            continue
        ans += abs(d[key] - d[rem])
        d[key] = d[rem]

    print(ans)
