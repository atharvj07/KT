#include<bits/stdc++.h>
using namespace std;
int main() {
  long long n, k;
  cin >> n >> k;
  long long a[k];
  for(int i=0;i<k;i++) { cin >> a[i]; }
  long long mx = 0, ans = -1;
  for(int i=0;i<k;i++) {
    long long x = n/a[i];
    if(x*a[i] == n && x>mx) {
      mx = x;
      ans = i+1;
    }
  }
  cout << ans << ' ' << mx << endl;
  return 0;
}
