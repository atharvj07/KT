#include <bits/stdc++.h>
#include <algorithm>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s, ans;
    cin >> n >> s;
    vector<int> pos[26];
    for (int i = 0; i < n; i++) { pos[s[i] - 'a'].push_back(i); }
    for (char ch : {'t', 'r', 'y', 'g', 'u', 'b'}) {
      if (!pos[ch - 'a'].empty()) {
        ans = s;
        for (int j = (int)pos[ch - 'a'].size() - 1; j >= 0; j--) {
          ans[pos[ch - 'a'][j]] = ch;
        }
        break;
      }
    }
    for (int i = 0; i < n; i++) {
      if (ans[i] == '\0') ans[i] = s[i];
    }
    cout << ans << endl;
  }
  return 0;
}
