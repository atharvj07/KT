import java.util.stream.IntStream;
import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int[] caps = Arrays.stream(input.nextLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] vols = Arrays.stream(input.nextLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] remain = Arrays.copyOf(vols, caps.length);
        int ret = IntStream.range(0, caps.length).parallel().map(i -> {
            int cur = vols[i];
            caps[i] -= cur;
            for (int j = i + 1; j < caps.length; j++) {
                if (caps[j] > 0) {
                    if (cur > 0) {
                        int minus = Math.min(caps[j], cur);
                        cur -= minus;
                        caps[j] -= minus;
                    } else {
                        break;
                    }
                }
            }
            return vols[i] - cur;
        }).peek(i -> System.arraycopy(remain, 0, vols, 0, caps.length)).sum();
        System.out.println(ret == 0 ? "YES" : "NO");
    }
}
