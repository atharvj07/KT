#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);cout.tie(NULL);
    int n,s=0,ans=0;
    cin>>n;
    vector<int> v(n);
    for(int i=0;i<n;i++)
    {
        cin>>v[i];
        s+=v[i];
        if(s>=0)
        ans++;
        else
        s=0;
    }
    cout<<ans<<endl;
}
