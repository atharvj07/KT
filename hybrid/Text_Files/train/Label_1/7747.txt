#include <bits/stdc++.h>
using namespace std;
const int N=3e4+9;
int a[N];
bool vis[N];
int main(){
    int n,t;cin>>n>>t;
    for(int i=0;i<n-1;++i)cin>>a[i];
    vis[1]=1;
    for(int i=1;;){
        if(i+a[i-1]==t){
            cout<<"YES";
            exit(0);
        }
        if(vis[i+a[i-1]])break;
        i+=a[i-1];
        vis[i]=1;
    }
    cout<<"NO";
    return 0;
}
