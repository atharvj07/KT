for _ in[0]*int(input()):
    n,w=map(int,input().split())
    l=list(map(int,input().split()))
    l=[0]+l
    l.sort()
    dp=[[0 for i in range(w+1)]for j in range(n+1)]
    for i in range(1,n+1):
        for j in range(1,w+1):
            if j<l[i]:
                dp[i][j]=dp[i-1][j]
            else:
                dp[i][j]=max(dp[i-1][j],dp[i-1][j-l[i]]+1)
    print(dp[n][w])
