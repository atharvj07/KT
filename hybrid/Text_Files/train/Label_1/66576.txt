
import sys
n, m, k = map(int, sys.stdin.readline().strip().split())

blocks = sys.stdin.read().strip().split("\n\n") # read the blocks of characters
blocks = [list(map(list, l.split("\n"))) for l in blocks] # convert the blocks to lists of lists of characters

def get_front(x, y, z): # get the front neighbor of a CPU
    if x < n - 1:
        return x + 1, y, z
    else:
        return False

def get_up(x, y, z): # get the up neighbor of a CPU
    if y < m - 1:
        return x, y + 1, z
    else:
        return False

def get_down(x, y, z): # get the down neighbor of a CPU
    if z < k - 1:
        return x, y, z + 1
    else:
        return False


def check_critical(x, y, z): # check if a CPU is critical
    critical = False
    for neighbor in [get_front(x, y, z), get_up(x, y, z), get_down(x, y, z)]:
        if neighbor and blocks[neighbor[0]][neighbor[1]][neighbor[2]] == '1':
            for neighbor2 in [get_front(neighbor[0], neighbor[1], neighbor[2]), get_up(neighbor[0], neighbor[1], neighbor[2]), get_down(neighbor[0], neighbor[1], neighbor[2])]:
                if neighbor2 and blocks[neighbor2[0]][neighbor2[1]][neighbor2[2]] == '1':
                    if blocks[x][y][z] == '0':
                        critical = True
                    else:
                        blocks[neighbor2[0]][neighbor2[1]][neighbor2[2]] = '0' # turn off the neighbor2 CPU
    return critical

critical_count = 0
for i in range(n):
    for j in range(m):
        for l in range(k):
            if blocks[i][j][l] == '1' and check_critical(i, j, l):
                critical_count += 1

print(critical_count)
