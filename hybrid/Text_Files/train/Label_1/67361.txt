import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))

maxa = [0] * n
maxa[0] = a[0]
for i in range(1, n):
  maxa[i] = max(maxa[i - 1], a[i])

ans = [-1] * n
for i in range(n):
  x = 0
  for j in range(i, n):
    x = max(x, a[j])
    if a[j] < x // 2:
      ans[i] = j - i + 1
      break
  if ans[i] == -1:
    for j in range(0, i):
      x = max(x, a[j])
      if a[j] < x // 2:
        ans[i] = j - i + n + 1
        break

print(*ans)
