#include <string>
#include <vector>

std::vector<int> dist[600];
int l[600];
int r[600];
bool vis[600];
int ord[600];

int main() {
  int n;
  while (scanf("%d", &n) != EOF) {
    for (int i = 0; i < 2 * n; ++i) {
      dist[i].clear();
      vis[i] = false;
    }
    for (int i = 0; i < n; ++i) {
      scanf("%d%d", &l[i], &r[i]);
      --l[i];
      --r[i];
      dist[l[i]].push_back(r[i]);
    }
    int seq = 0;
    for (int i = 0; i < 2 * n; ++i) {
      while (vis[i]) ++i;
      if (i == 2 * n) break;
      ord[seq++] = i;
      vis[i] = true;
      int to = -1;
      for (int j = 0; j < dist[i].size(); ++j) {
        if (dist[i][j] > i && !vis[dist[i][j]]) {
          to = dist[i][j];
          break;
        }
      }
      if (to == -1) {
        seq = -1;
        break;
      }
      ord[seq++] = to;
      vis[to] = true;
    }
    if (seq == -1)
      puts("IMPOSSIBLE");
    else {
      std::string ans;
      for (int i = 0; i < 2 * n; ++i) {
        if (ord[i] < ord[i + 1]) {
          ans.push_back('(');
        } else {
          ans.push_back(')');
        }
      }
      puts(ans.c_str());
    }
  }
  return 0;
}
