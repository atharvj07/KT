s = input().strip()
n = len(s)

# Initialize a 2D list to keep track of palindromes (1-palindrome and k-palindrome)
dp = [[0] * n for _ in range(n)]
pal = [[False] * n for _ in range(n)]

# Check for 1-palindrome and fill in the pal[][] table
for i in range(n):
    for j in range(i, n):
        if s[i] == s[j] and (j - i <= 2 or pal[i+1][j-1]):
            pal[i][j] = True
            dp[i][j] = 1

# Check for k-palindromes where k > 1
for k in range(2, n+1):
    for i in range(n-k+1):
        j = i + k - 1
        if pal[i][j] and i + 1 <= j - 1:
            dp[i//2][j//2] += 1

# Calculate and print the palindromic characteristics
result = [0] * n
for i in range(n):
    for j in range(i, n):
        if dp[i][j]:
            result[dp[i][j]-1] += 1

print(*result)
