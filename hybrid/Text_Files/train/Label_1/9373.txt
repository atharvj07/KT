#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
int main() {
    ll n, t, c = 0, l = -1, tmp;
    bool cont = true;
    vector<ll> ans;
    cin >> n;
    while (n--) {
        cin >> t;
        if (c == 0) {
            l = t;
            c++;
        } else if (t != l) {
            ans.push_back(l);
            c = 1;
            l = t;
        } else {
            c++;
            l = t;
        }
    }
    if (c > 0) ans.push_back(l);
    cout << ans.size() << endl;
    for (ll x : ans) cout << x << " ";
    cout << endl;
}
