#include <algorithm>
#include <iostream>
#include <string>
#include <vector>

int main() {
	int t;
	std::cin >> t;
	while (t--) {
		long long n;
		int k;
		std::cin >> n >> k;
		std::vector<bool> seen(11);
		std::string s = std::to_string(n);
		for (auto c : s) seen[c - '0'] = true;
		bool f = false;
		long long ans = n;
		while (!f) {
			f = true;
			for (long long i = ans;; i++) {
				bool found = false;
				for (int j = 0; j < 11; j++) {
					int cnt = 0;
					if (!seen[j]) continue;
					std::string c = std::to_string(i);
					for (auto it : c) {
						cnt += (it - '0') == j;
					}
					if (cnt > k) {
						found = true;
						seen[j] = false;
						std::cout << i << '\n';
						ans = i;
						f = false;
						break;
					}
				}
				if (!found) {
					ans = i;
					break;
				}
			}
		}
	}
}
