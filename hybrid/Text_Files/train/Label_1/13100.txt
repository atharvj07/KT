#include<bits/stdc++.h>
#define REP(i,s,b) for(int i=s;i<b;i++)
using namespace std;
const double eps=1e-6;
class point{
	public:
		double x,y;
		point(){};
		point(double _x,double _y){
			x=_x,y=_y;
		}
		double operator-(point a){
			return sqrt((x-a.x)*(x-a.x)+(y-a.y)*(y-a.y));
		}
		point operator-(double a){
			return point(x-a,y-a);
		}
		point operator+(double a){
			return point(x+a,y+a);
		}
}p[1005];
double max_dis(int n){
	double mx=-1;
	REP(i,1,n+1){
		REP(j,i+1,n+1){
			double tmp=p[i]-p[j];
			if(tmp>mx){
				mx=tmp;
			}
		}
	}
	return mx;
}
bool check(int n,double r){
	int sum=0;
	REP(i,1,n+1){
		if(p[i]-p[0]<=r){
			sum+=p[i].x+p[i].y;
		}
	}
	if(sum>=1e6){
		return 1;
	}
	return 0;
}
int main(){
	int n,t;
	cin>>n>>t;
	REP(i,1,n+1){
		double x,y;
		cin>>x>>y;
		p[i]=point(x,y);
	}
	double l=0,r=max_dis(n);
	while((r-l)>eps){
		double mid=(l+r)/2;
		if(check(n,mid)){
			r=mid;
		}
		else{
			l=mid;
		}
	}
	if(r==max_dis(n))	cout<<-1<<endl;
	else	cout<<fixed<<setprecision(9)<<l<<endl;
	return 0;
}
