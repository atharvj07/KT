import java.util.*;

class Main {
  static final int INF = (int) 1e9;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int V = sc.nextInt();
    int E = sc.nextInt();
    int r = sc.nextInt();

    int[][] adj = new int[V][V];
    for (int i = 0; i < V; i++) {
      for (int j = 0; j < V; j++) {
        adj[i][j] = INF;
      }
    }

    for (int i = 0; i < E; i++) {
      int s = sc.nextInt();
      int t = sc.nextInt();
      int w = sc.nextInt();
      adj[s][t] = w;
    }

    int[] dist = new int[V];
    Arrays.fill(dist, INF);

    dist[r] = 0;

    boolean[] visited = new boolean[V];

    for (int i = 0; i < V; i++) {
      int minDist = INF;
      int minIdx = -1;

      for (int j = 0; j < V; j++) {
        if (!visited[j] && dist[j] < minDist) {
          minDist = dist[j];
          minIdx = j;
        }
      }

      if (minIdx == -1) {
        break;
      }

      visited[minIdx] = true;

      for (int j = 0; j < V; j++) {
        if (!visited[j] && adj[minIdx][j] != INF) {
          if (dist[minIdx] + adj[minIdx][j] < dist[j]) {
            dist[j] = dist[minIdx] + adj[minIdx][j];
          }
        }
      }
    }

    int sum = 0;
    for (int i = 0; i < V; i++) {
      if (dist[i] != INF) {
        sum += dist[i];
      }
    }

    System.out.println(sum);
  }
}
