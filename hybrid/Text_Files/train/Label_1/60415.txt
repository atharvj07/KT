import sys
input = sys.stdin.readline

n = int(input())
e = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    e[a].append(b)
    e[b].append(a)
q = []
for i in range(2,n+1):
    if len(e[i]) == 1:
        q.append(i)

cnt = 0
while len(q):
    now = q.pop()
    cnt += 1
    e[e[now][0]].remove(now)
    if len(e[e[now][0]]) == 1:
        q.append(e[now][0])

ps = [(-1, 0) for _ in range(cnt+1)]

def dfs(now, pr):
    mx = -1
    mn = n+1
    for nxt in e[now]:
        if nxt == pr:
            continue
        dfs(nxt, now)
        mx = max(mx, ps[nxt][0])
        mn = min(mn, ps[nxt][1])
    if mx == -1:
        ps[now] = (1, 1)
    else:
        ps[now] = (mx+1, mn+1)
    
dfs(1,-1)
print(ps[1][0], ps[1][1])
