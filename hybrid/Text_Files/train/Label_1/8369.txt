#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, a, b, k;
    cin >> n >> a >> b >> k;
    vector<long long> h(n);
    for (long long i = 0; i < n; i++) cin >> h[i];
    vector<long long> dp(n + 1, 0);
    for (long long i = n - 1; i >= 0; i--) {
        dp[i] = dp[i + 1];
        if (h[i] <= a) dp[i] = max(dp[i], dp[i + 1] + 1);
        if (k > 0 && h[i] > a && h[i] <= a + b) {
            dp[i] = max(dp[i], dp[i + 1] + 1);
            k--;
        }
    }
    cout << dp[0] << endl;
    return 0;
}
