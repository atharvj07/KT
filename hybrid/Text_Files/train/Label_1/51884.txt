import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        Map<String, String> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String w = sc.nextLine();
            String w2 = sc.nextLine();
            if (!w.equals(w2)) {
                map.put(w, w2);
            }
        }
        sc.close();
        if (map.size() == 0) {
            System.out.println("YES");
            System.out.println("a");
            System.out.println("a");
        } else if (map.size() == 1) {
            Map.Entry<String, String> entry = map.entrySet().iterator().next();
            String s = entry.getKey();
            String t = entry.getValue();
            if (s.length() > 5000 || t.length() > 5000) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
                System.out.println(s);
                System.out.println(t);
            }
        } else {
            List<Map.Entry<String, String>> list = new ArrayList<>(map.entrySet());
            Map.Entry<String, String> entry1 = list.get(0);
            String s1 = entry1.getKey();
            String t1 = entry1.getValue();
            Map.Entry<String, String> entry2 = list.get(1);
            String s2 = entry2.getKey();
            String t2 = entry2.getValue();
            if (!s1.equals(s2) || !t1.equals(t2)) {
                System.out.println("NO");
            } else {
                if (s1.length() > 5000 || t1.length() > 5000) {
                    System.out.println("NO");
                } else {
                    System.out.println("YES");
                    System.out.println(s1);
                    System.out.println(t1);
                }
            }
        }
    }
}
