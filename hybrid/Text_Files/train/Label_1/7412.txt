#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const double eps = 1e-9;

struct Edge {
  int u, v, w;
  bool operator<(const Edge &other) const { return w < other.w; }
};

struct DSU {
  vector<int> par, sz;
  DSU(int n) : par(n), sz(n, 1) { iota(par.begin(), par.end(), 0); }
  int find(int u) { return par[u] = (u == par[u] ? u : find(par[u])); }
  bool join(int a, int b) {
    a = find(a), b = find(b);
    if (a != b) {
      if (sz[a] > sz[b])
        swap(a, b);
      par[a] = b;
      sz[b] += sz[a];
      return true;
    }
    return false;
  }
};

int main() {
  int n, m;
  cin >> n >> m;
  vector<ll> val(n);
  for (ll &x : val) cin >> x;
  vector<Edge> edges(m);
  for (Edge &e : edges) { cin >> e.u >> e.v >> e.w; }
  sort(edges.begin(), edges.end());
  double ans = 0;
  ll sum_v = accumulate(val.begin(), val.end(), 0LL);
  ll sum_e = 0;
  DSU dsu(n);
  for (Edge &e : edges) {
    if (dsu.join(e.u - 1, e.v - 1)) {
      ans = max(ans, (double)(sum_v) / (sum_e + e.w));
      sum_e += e.w;
    }
  }
  if (sum_e == 0)
    ans = 0;
  cout << fixed << setprecision(15) << ans << endl;
  return 0;
}
