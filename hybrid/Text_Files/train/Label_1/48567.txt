import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        for (int i = 0; i < n / 2; i++) {
            int diff = (int) Math.sqrt(a[i]) * (int) Math.sqrt(a[i]) - a[i];
            if (diff != 0) {
                ans += diff;
                a[i] += diff;
            }
        }
        for (int i = n / 2; i < n; i++) {
            int diff = a[i] - (int) Math.sqrt(a[i]) * (int) Math.sqrt(a[i]);
            if (diff != 0) {
                ans += diff;
                a[i] -= diff;
            }
        }
        System.out.println(ans);
    }
}
