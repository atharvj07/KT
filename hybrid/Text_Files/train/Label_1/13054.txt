#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<vector<int>> p(n);
    vector<int> d(n + 1);
    for (int i = 0; i < n; ++i) {
      int m;
      cin >> m;
      while (m--) {
        int x;
        cin >> x;
        p[i].push_back(x);
        d[x]++;
      }
    }
    int ans = count(d.begin() + 1, d.end(), 0);
    if (ans)
      cout << "IMPROVE\n" << n << " " << find(d.begin() + 1, d.end(), 0) - d.begin() << "\n";
    else
      cout << "OPTIMAL\n";
  }
}
