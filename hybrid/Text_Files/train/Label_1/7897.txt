#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> v(n), t(n);
  for (int i = 0; i < n; i++) cin >> v[i];
  for (int i = 0; i < n; i++) cin >> t[i];
  vector<long long> ans(n);
  for (int i = 0; i < n; i++) {
    if (v[i] <= t[i]) {
      ans[i] = v[i];
      v[i] = 0;
    } else {
      ans[i] = t[i];
      v[i] -= t[i];
    }
    if (i != n - 1) {
      if (v[i + 1] <= t[i + 1]) {
        v[i + 1] = 0;
      } else {
        v[i + 1] -= t[i + 1];
      }
    }
  }
  for (int i = 0; i < n; i++) { cout << ans[i] << " "; }
  return 0;
}
