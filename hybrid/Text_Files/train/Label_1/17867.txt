#include<bits/stdc++.h>
using namespace std;

int main() {
  int n, m, k,a[100001],p[100001],ans=0;
  cin>>n>>m>>k;
  for(int i=1;i<=n;i++) {
    cin>>a[i];
    p[a[i]]=i;
  }
  for(int i=1;i<=m;i++) {
    int x;
    cin>>x;
    ans+=p[x]/k+1;
    for(int j=p[x]-1;j>=1;j--) {
      swap(a[j],a[j+1]);
      p[a[j]]--;
      p[a[j+1]]++;
    }
  }
  cout<<ans<<endl;
  return 0;
}
