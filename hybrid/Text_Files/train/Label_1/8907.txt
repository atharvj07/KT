#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string level;
    cin >> level;
    int cnt = 0, ans = 0;
    for (int i = 0; i < n; i++) {
      if (level[i] == '*') {
        ans += cnt / 2;
        cnt++;
      } else {
        cnt = 0;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
