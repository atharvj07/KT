n, m = map(int, input().split())
grid = [input() for _ in range(n)]

# Pre-compute the number of empty cells to the left, right, up, and down of each cell
left = [[0]*m for _ in range(n)]
right = [[0]*m for _ in range(n)]
up = [[0]*m for _ in range(n)]
down = [[0]*m for _ in range(n)]

for i in range(n):
    for j in range(m):
        if grid[i][j] == '.':
            left[i][j] = (left[i][j-1] if j > 0 else 0) + 1
            up[i][j] = (up[i-1][j] if i > 0 else 0) + 1

for i in range(n-1, -1, -1):
    for j in range(m-1, -1, -1):
        if grid[i][j] == '.':
            right[i][j] = (right[i][j+1] if j < m-1 else 0) + 1
            down[i][j] = (down[i+1][j] if i < n-1 else 0) + 1

# Count the number of ways to create a pipe
ways = 0

for i in range(n):
    for j in range(m):
        if grid[i][j] == '.':
            if i > 0 and i < n-1:
                ways += (up[i][j] - 1) * (down[i][j] - 1)
            if j > 0 and j < m-1:
                ways += (left[i][j] - 1) * (right[i][j] - 1)

print(ways)
