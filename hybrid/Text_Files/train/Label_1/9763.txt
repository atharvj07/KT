#include<bits/stdc++.h>
using namespace std;
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int n,m;
    cin>>n>>m;
    char s[n+1],x[m+1],y[m+1];
    cin>>s;
    for(int i=0;i<m;i++){
       cin>>x[i]>>y[i];
    }
    int a[200001];
    for(int i=0;i<200001;i++)a[i]=i;
    for(int i=0;i<m;i++){
      int temp=a[x[i]-'a'];
      a[x[i]-'a']=a[y[i]-'a'];
      a[y[i]-'a']=temp;
    }
    for(int i=0;i<n;i++)cout<<(char)(a[s[i]-'a']+'a');
    cout<<'\n';
    return 0;
}
