import java.util.*;
import java.math.*;

public class Main {
    static BigInteger ksm(BigInteger a, long b) {
        if(b == 0) return BigInteger.ONE;
        BigInteger tmp = ksm(a, b / 2);
        tmp = tmp.multiply(tmp);
        if(b % 2 == 1) tmp = tmp.multiply(a);
        return tmp;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int d = in.nextInt();
        int s = in.nextInt();
        if(s % d != 0) {
            System.out.println(-1);
            return;
        }
        if(s / d > 9) {
            System.out.println(-1);
            return;
        }
        BigInteger res = BigInteger.ZERO;
        for(int i = 1; i * d <= s; i++) {
            int j = s / d - i;
            res = res.add(ksm(new BigInteger(Integer.toString(i)), j)).add(ksm(new BigInteger(Integer.toString(j)), i));
        }
        System.out.println(res);
    }
}
