def solve(n, a):
    if a == sorted(a):
        return 0
    for i in range(1, n+1):
        if i % 2 == 1:
            for j in range(1, n, 2):
                if a[j-1] > a[j]:
                    a[j-1], a[j] = a[j], a[j-1]
        else:
            for j in range(2, n, 2):
                if a[j-1] > a[j]:
                    a[j-1], a[j] = a[j], a[j-1]
        if a == sorted(a):
            return i

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    print(solve(n, a))
