#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m, d;
  cin >> n >> m >> d;
  vector<int> c(m);
  for (auto &x : c) cin >> x;
  for (int i = 0; i + 1 < c.size(); i++) {
    if (c[i] > c[i + 1]) swap(c[i], c[i + 1]);
  }
  vector<int> a(n + 1);
  int cur = 1, cnt = 0, p = 0;
  while (cur <= n) {
    if (cnt == c[p]) {
      p++;
      cnt = 0;
    }
    if (cur + d >= n) break;
    a[cur++] = 0;
    cnt++;
  }
  if (cur <= n) {
    cout << "NO\n";
    return 0;
  }
  cout << "YES\n";
  int cnt_ = 0;
  for (int i = 1; i <= n; i++) {
    if (a[i] == 0) {
      a[i] = p + 1;
      cnt_++;
      if (cnt_ == c[p]) {
        cnt_ = 0;
        p++;
      }
    }
    cout << a[i] << ' ';
  }
  cout << '\n';
}
