#include <bits/stdc++.h>
#define pb push_back
#define vi vector<int>
#define ll long long
using namespace std;
const int maxn = 2005;
struct segtree
{
    ll tr[maxn * 4];
    void init() { memset(tr, 0, sizeof tr); }
    void build(int l, int r, int pos, vi& a)
    {
        if (l == r)
        {
            tr[pos] = a[l];
            return;
        }
        int mid = (l + r) / 2;
        build(l, mid, pos * 2, a);
        build(mid + 1, r, pos * 2 + 1, a);
        tr[pos] = min(tr[pos * 2], tr[pos * 2 + 1]);
    }
    void update(int l, int r, int pos, int val, int idx)
    {
        if (l > idx || r < idx)
            return;
        if (l == r)
        {
            tr[pos] = val;
            return;
        }
        int mid = (l + r) / 2;
        update(l, mid, pos * 2, val, idx);
        update(mid + 1, r, pos * 2 + 1, val, idx);
        tr[pos] = min(tr[pos * 2], tr[pos * 2 + 1]);
    }
    ll get(int l, int r, int pos, int L, int R)
    {
        if (l > R || r < L) return 1e18;
        if (l >= L && r <= R) return tr[pos];
        int mid = (l + r) / 2;
        return min(get(l, mid, pos * 2, L, R), get(mid + 1, r, pos * 2 + 1, L, R));
    }
};
vi a;
int main()
{
    int n, k;
    cin >> n >> k;
    a.pb(1);
    for (int i = 0; i < n; i++)
    {
        int x;
        cin >> x;
        a.pb(x);
    }
    a.pb(1);
    sort(a.begin(), a.end());
    segtree s;
    s.init();
    s.build(1, a.size() - 1, 1, a);
    ll ans = 0;
    for (int i = 2; i < a.size(); i++)
    {
        ll x = s.get(1, a.size() - 1, 1, 1, a[i] - 1);
        s.update(1, a.size() - 1, 1, x + a[i] - a[i - 1], a[i]);
        ans = ans + x + a[i] - a[i - 1];
    }
    cout << ans << endl;
    return 0;
}
