import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n, T = map(int, input().split())
    a = [int(x) for x in input().split()]
    b = sorted([x for x in a if x <= T // 2])
    ans = [0] * n
    for x in a:
        if not b or x > T // 2 or x + b[0] < T:
            ans[a.index(x)] = 1
        else:
            ans[a.index(x)] = 0
            b.pop(0)
    print(*ans)
