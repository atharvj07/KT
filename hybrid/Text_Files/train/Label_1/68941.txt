n = int(input())
board = [input() for _ in range(n)]
ans = "YES"
for i in range(n):
    for j in range(n):
        cnt = 0
        if i > 0: cnt += board[i - 1][j] == 'o'
        if i < n - 1: cnt += board[i + 1][j] == 'o'
        if j > 0: cnt += board[i][j - 1] == 'o'
        if j < n - 1: cnt += board[i][j + 1] == 'o'
        if cnt % 2 == 1: ans = "NO"
print(ans)
