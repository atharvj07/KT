#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    bool found = false;
    do {
      bool ok = true;
      int sum = 0;
      for (int x : a) {
        sum += x;
        if (sum == 0) ok = false;
      }
      if (ok) {
        cout << "YES\n";
        for (int x : a) cout << x << " ";
        cout << "\n";
        found = true;
        break;
      }
    } while (next_permutation(a.begin(), a.end()));
    if (!found) cout << "NO\n";
  }
  return 0;
}
