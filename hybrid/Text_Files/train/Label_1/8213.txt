#include <cstdio>
#include <cstring>
using namespace std;
int n, k, v[400005], p[400005], id[200005], ans;
int main() {
  scanf("%d %d", &n, &k);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &v[i]);
    if (!id[v[i]]) p[++id[v[i]]] = i;
  }
  for (int i = 1; i <= n; i++) {
    if (p[v[i]] == i) ans++;
    p[v[i]]++;
    if (p[v[i] - k] == i) ans--;
  }
  printf("%d\n", ans);
  return 0;
}
