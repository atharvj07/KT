#include<bits/stdc++.h>
#define sq(x) (x)*(x)
using namespace std;
typedef long long ll;
vector<ll> X,Y,Z;
bool valid(){
    X.clear();Y.clear();Z.clear();
    ll n;cin>>n;
    for(ll i=0;i<n;i++){
            ll a,b,c;
            cin>>a>>b>>c;
            X.push_back(a);
            Y.push_back(b);
            Z.push_back(c);
    }
    ll m;cin>>m;
    for(ll i=0;i<m;i++){
            ll a,b,c;
            cin>>a>>b>>c;
            X.push_back(a);
            Y.push_back(b);
            Z.push_back(c);
    }
    n+=m;
    vector<ll> polyx(n),polyy(n),polyz(n);
    for(ll i=0;i<n;i++){
        polyx[i]=X[(i+1)%n]+abs(X[(i+1)%n]-X[i]);
        polyy[i]=Y[(i+1)%n]+abs(Y[(i+1)%n]-Y[i]);
        polyz[i]=Z[(i+1)%n]+abs(Z[(i+1)%n]-Z[i]);
    }

    ll negx=0,negy=0,negz=0;
    for(ll i=0;i<n;i++){
        negx+=(polyx[i]*(polyy[i]+polyy[(i+1)%n]))/2;
        negy+=(polyy[i]*(polyz[i]+polyz[(i+1)%n]))/2;
        negz+=(polyz[i]*(polyx[i]+polyx[(i+1)%n]))/2;
    }
    return (negx*negy*negz>0)?true:false;
}

int main(){
    string str=(valid())?"YES":"NO";
    cout<<str<<'\n';
    return 0;
}
