#include <iostream>
#include <vector>
using namespace std;

const int mxN = 3e5+5;
int n;
char s[mxN];
bool vis[mxN];

int main() {
    cin >> n;
    cin >> s;

    vector<int> l(n), r(n);
    for (int i = 0, j = n-1; i < n; ++i, --j) {
        if (s[i] == s[j]) {
            l[i] = l[i+1];
            r[j] = r[j-1];
        }
        else {
            l[i] = i;
            r[j] = j;
        }
    }

    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        if (!vis[i]) {
            int j = i;
            while (j < n && l[j] <= i) {
                vis[j++] = true;
            }
            ans += 1LL * (j-i) * (j-i+1) / 2;
        }
    }

    for (int i = n-1; i >= 0; --i) {
        if (!vis[i]) {
            int j = i;
            while (j >= 0 && r[j] >= i) {
                vis[j--] = true;
            }
            ans += 1LL * (j-i+1) * (j-i+2) / 2;
        }
    }

    cout << ans << '\n';
    return 0;
}
