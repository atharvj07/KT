n = int(input())
points = [tuple(map(int, input().split())) for _ in range(n)]
w = list(map(int, input().split()))

points.sort(key=lambda x: (x[0] + x[1], x[0]))

special_values = [y - x for x, y in points]

if special_values == w:
    print("YES")
    for p in points:
        print(*p)
else:
    print("NO")
