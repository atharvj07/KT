n, m, k = map(int, input().split())
a = [['0'] * (k + 2) for _ in range(m + 2)]
for i in range(1, n + 1):
    input()
    for j in range(1, m + 1):
        a[j][1:k + 1] = list(input())
b = [[[0] * (k + 2) for _ in range(m + 2)] for _ in range(n + 2)]
for x in range(1, n + 1):
    for y in range(1, m + 1):
        for z in range(1, k + 1):
            if a[y][z] == '1':
                b[x][y][z] = b[x - 1][y][z] or b[x][y - 1][z] or b[x][y][z - 1] or (x == 1 and y == 1 and z == 1)
                if b[x][y][z]:
                    b[x + 1][y][z] = 1
                    b[x][y + 1][z] = 1
                    b[x][y][z + 1] = 1
ans = 0
for x in range(1, n + 1):
    for y in range(1, m + 1):
        for z in range(1, k + 1):
            if b[x - 1][y][z] and not b[x][y][z]:
                ans += a[y][z] == '1'
            if b[x][y - 1][z] and not b[x][y][z]:
                ans += a[y][z] == '1'
            if b[x][y][z - 1] and not b[x][y][z]:
                ans += a[y][z] == '1'
print(ans)
