#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 505, maxk = 15;
struct data {
    int c, f;
} a[maxn * maxk];
int n, k, f[maxn], cnt[maxn * maxk], h[maxk];
ll dp[maxk][maxn * maxk];
int main() {
    cin >> n >> k;
    for (int i = 0; i < n * k; i++) cin >> a[i].c;
    for (int i = 0; i < n; i++) cin >> a[i].f;
    for (int i = 0; i < k; i++) cin >> h[i];
    for (int i = 0; i < n * k; i++) cnt[a[i].c]++;
    for (int i = 0; i < k; i++)
        for (int j = 0; j <= n * k; j++) {
            if (j - i > h[i]) continue;
            if (!i)
                dp[i][j] = max(dp[i][j], 0LL);
            else {
                for (int l = 1; l <= 100000; l++) {
                    if (j + l > n * k) break;
                    if (h[i] < h[i - 1] && cnt[l] <= j) continue;
                    dp[i][j] = max(dp[i][j], dp[i - 1][j - l] + 1LL * h[i] * cnt[l]);
                }
            }
        }
    cout << dp[k - 1][n * k] << endl;
    return 0;
}
