n, m = map(int, input().split())
coupons = sorted([list(map(int, input().split())) for _ in range(m)], key=lambda x: x[1], reverse=True)

# In a beautiful array, the maximum number of distinct elements it can have is 2.
# Because for each pair of distinct elements x, y in a beautiful array, there must exist a position j
# such that either a[j] = x and a[j+1] = y, or a[j] = y and a[j+1] = x.
# This implies that if there are more than 2 distinct elements, it wouldn't be possible to satisfy the conditions
# for all pairs of distinct elements.

# Thus, Sereja can pay for at most 2 distinct numbers to be used in constructing the array.
# To maximize the amount of money he can pay, we select the two coupons with the highest values.

max_payment = sum(coupon[1] for coupon in coupons[:2]) if n > 1 else coupons[0][1]
print(max_payment)
