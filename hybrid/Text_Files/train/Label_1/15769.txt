#include <iostream>
#include <string>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    string res = "";
    int cnt[26] = {};
    for (int i = 0; i < n; i++) {
      cnt[s[i] - 'a']++;
    }
    for (int i = 0; i < 26; i++) {
      if (cnt[i] == 1) res += char(i + 'a');
    }
    if (res.empty())
      cout << endl;
    else
      cout << res << endl;
  }
}
