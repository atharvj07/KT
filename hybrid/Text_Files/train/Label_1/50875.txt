import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    while (t-- > 0) {
      int n = sc.nextInt();
      int v = 1;
      int m = Math.min(n / 2, 1000);
      for (int i = 0; i < n; i++, v += 2) {
        if (i < m) {
          System.out.print(v + " ");
        } else {
          System.out.print("1 ");
        }
      }
      System.out.println();
    }
    sc.close();
  }

}
