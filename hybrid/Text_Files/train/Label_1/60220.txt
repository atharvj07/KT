def max_sequence_factors(n):
    x = 2
    k = 0
    while x * x <= n:
        if n % x:
            x += 1
        else:
            n //= x
            k += 1
    if n > 1:
        k += 1
    return k, [x for x in range(2, k + 2) if not n % x]

t = int(input())
for i in range(t):
    n = int(input())
    k, seq = max_sequence_factors(n)
    print(k)
    print(*seq)
