from collections import defaultdict
from ipaddress import IPv4Address

info = list(map(IPv4Address, input().strip().split()[2:]))
dict = defaultdict(list)

for ip in info:
    for i in range(32):
        dict[ip.network_address.packed[i]][i] = True

for i in range(32):
    if i in dict and len(dict[i]) == len(info):
        print('.'.join(map(str, i.to_bytes(4, 'big'))))
        exit()
print('-1')
