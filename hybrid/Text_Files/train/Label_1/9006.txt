#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n), used(n), ans[k];
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    int x = 0;
    for (int i = 0; i < k; i++) {
        int y = a[x++];
        while (x < n && y == a[x]) { x++; }
        if (x == n) {
            x = 0;
            y = a[x++];
            while (x < n && y == a[x]) { x++; }
        }
        ans[i].push_back(1);
        ans[i].push_back(y);
        used[x - 1] = 1;
        for (int j = 0; j < n; j++) {
            if (!used[j]) {
                y = a[j];
                while (j + 1 < n && y == a[j + 1]) { j++; }
                ans[i].push_back(j - x + 2);
                for (int t = x; t <= j; t++) { used[t] = 1; }
            }
        }
        sort(ans[i].begin() + 1, ans[i].end());
    }
    for (int i = 0; i < k; i++) {
        cout << ans[i][0] << ' ';
        for (int j = 1; j < ans[i].size(); j++) { cout << ans[i][j] << ' '; }
        cout << '\n';
    }
}
