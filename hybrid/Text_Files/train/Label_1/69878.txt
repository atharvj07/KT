n = int(input())
s = input()
ans = 0
for i in range(1, n+1):
    if s[i - 1] == '>':
        while i < n and s[i] == '>':
            i += 1
        i -= 1
        ans += (i != n)
    else:
        while i > 1 and s[i - 1] == '<':
            i -= 1
        i += 1
        ans += (i != 1)
print(ans)
