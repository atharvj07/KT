import sys
def solve(n, m, c):
    L = [0] * (m + 1)
    R = [0] * (m + 1)
    for x in c:
        if not L[x]:
            L[x] = 1
        elif not R[x]:
            R[x] = 1

    ans = 0
    for i in range(1, m + 1):
        if L[i] and R[i]:
            ans += 1
    print(ans)
    for i in range(n):
        a = c[i]
        for j in range(1, m + 1):
            if L[j] and not R[a]:
                print(j, a)
                R[a] = 1
                L[j] = 0
                break
            elif R[j] and not L[a]:
                print(a, j)
                L[a] = 1
                R[j] = 0
                break

n, m = map(int, sys.stdin.readline().split())
c = list(map(int, sys.stdin.readline().split()))
solve(n, m, c)
