#include <cstdio>
int n, m;
double p[5005], ans;
int main() {
    scanf("%d %d", &n, &m);
    for (int i = 0; i < n; i++) {
        int a, b;
        scanf("%d %d", &a, &b);
        p[i] = a * 1.0 / b;
    }
    ans = p[0];
    for (int i = 1; i < n; i++) ans = ans > p[i] ? p[i] : ans;
    printf("%.8lf\n", ans * m);
    return 0;
}
