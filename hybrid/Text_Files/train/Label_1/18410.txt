#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, i = 0;
    cin >> n >> k;
    vector<int> h(n);
    for (auto &x : h) cin >> x;
    while (k--) {
      if (++i == n) {
        i = -1;
        break;
      }
      if (h[i] < h[i + 1]) h[i]++;
    }
    cout << (i == -1 ? -1 : i + 1) << "\n";
  }
  return 0;
}
