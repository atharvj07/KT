import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if(n == 0) break;
            String[] words = new String[n];
            for (int i = 0; i < n; i++) {
                words[i] = sc.next();
            }
            int start = 0;
            loop:
            for (int i = 0; i < n; i++) {
                int len = 0;
                for (int j = i; j < n; j++) {
                    len += words[j].length();
                    if(len == 5) {
                        start = i;
                        break loop;
                    }
                }
            }
            System.out.println(start + 1);
        }
    }
}
