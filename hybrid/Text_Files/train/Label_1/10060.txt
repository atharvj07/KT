#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int main() {
  int n, k;
  cin >> n >> k;
  vector<ll> c(n);
  for (int i = 0; i < n; i++)
    cin >> c[i];
  vector<bool> cap(n);
  for (int i = 0; i < k; i++) {
    int x;
    cin >> x;
    x--;
    cap[x] = true;
  }
  ll ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (cap[i] || cap[j])
        ans += c[i] * c[j];
    }
  }
  cout << ans << '\n';
  return 0;
}
