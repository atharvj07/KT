#include <bits/stdc++.h>
using namespace std;

int main() {
	int n; cin >> n;
	vector<int> a(n);
	for (int &x : a) cin >> x;
	unordered_map<int, vector<int>> p;
	for (int i = 0; i < n; i++) {
		if (!p.count(a[i])) p[a[i]] = vector<int>();
		p[a[i]].push_back(i);
	}
	vector<int> ans;
	for (int i = 0; i < n; i++) {
		for (int j = i + 1; j < n; j++) {
			int cur = a[i] + a[j];
			if (cur == a[i] && cur == a[j]) {
				if (p[cur].size() >= 4) {
					for (int l = 0; l < 2; l++) {
						for (int m = 0; m < 2; m++) {
							ans = {p[cur][l], p[cur][1 + l],
								p[cur][2 + m], p[cur][3 + m]};
						}
					}
				}
			} else if (p.count(cur) && p[cur].size() >= 2) {
				for (int l = 0; l < 2; l++) {
					for (int m = 0; m < p[cur].size(); m++) {
						ans = {i, j, p[cur][l], p[cur][m]};
					}
				}
			}
		}
	}
	if (ans.size()) {
		cout << "YES\n";
		for (int x : ans) cout << x + 1 << " ";
		cout << "\n";
	} else
		cout << "NO\n";
}
