#include <algorithm>
#include <cstdio>
#include <vector>
using namespace std;

const int MAXN=5003;

int n,m;
vector<vector<int>> adj;
int clr[MAXN];
bool vis[MAXN],vis2[MAXN];
bool has_cycle;

void dfs(int u,int c,int fa){
	vis[u]=true;
	clr[u]=c;
	for(int i:adj[u]){
		if(!vis[i]){
			dfs(i,3-c,u);
		}else if(clr[i]==clr[u]&&i!=fa){
			has_cycle=true;
		}
	}
}

int main(){
	scanf("%d%d",&n,&m);
	adj.resize(n+1);
	for(int i=1;i<=m;++i){
		int u,v;
		scanf("%d%d",&u,&v);
		adj[u].push_back(v);
	}
	dfs(1,1,1);
	if(has_cycle){
		for(int i=1;i<=n;++i){
			for(int j:adj[i]){
				if(clr[i]==clr[j]){
					clr[j]=clr[i]=3;
				}
			}
		}
		printf("2\n");
		for(int i=1;i<=m;++i){
			int u,v;
			scanf("%d%d",&u,&v);
			printf("%d ",clr[u]);
		}
		printf("\n");
	}else{
		printf("1\n");
		for(int i=1;i<=m;++i){
			int u,v;
			scanf("%d%d",&u,&v);
			printf("1 ");
		}
		printf("\n");
	}
	return 0;
}
