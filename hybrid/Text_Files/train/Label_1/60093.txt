def valid(grid,n,m,i,j):
    if i < 0 or i >= n or j < 0 or j >= m or grid[i][j] != '.': return True
    ni, nj = i-1, j
    if 0 <= ni < n and 0 <= nj < m and grid[ni][nj] == grid[i][j]: return False
    ni, nj = i+1, j
    if 0 <= ni < n and 0 <= nj < m and grid[ni][nj] == grid[i][j]: return False
    ni, nj = i, j-1
    if 0 <= ni < n and 0 <= nj < m and grid[ni][nj] == grid[i][j]: return False
    ni, nj = i, j+1
    if 0 <= ni < n and 0 <= nj < m and grid[ni][nj] == grid[i][j]: return False
    return True

def color(grid,n,m,i,j):
    if i == n:
        return True
    if j == m:
        return color(grid,n,m,i+1,0)
    if grid[i][j] != '.':
        return color(grid,n,m,i,j+1)
    grid[i][j] = 'R'
    if valid(grid,n,m,i,j) and color(grid,n,m,i,j+1):
        return True
    grid[i][j] = 'W'
    if valid(grid,n,m,i,j) and color(grid,n,m,i,j+1):
        return True
    grid[i][j] = '.'
    return False

t = int(input())
for _ in range(t):
    n,m = map(int,input().split())
    grid = [list(input()) for i in range(n)]
    if color(grid,n,m,0,0):
        print('YES')
        for row in grid:
            print(''.join(row))
    else:
        print('NO')
