import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[4];
        for (int i = 0; i < 4; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        boolean flag = true;
        for (int i = 0; i < 4; i++) {
            if (arr[i] != "1974".charAt(i) - '0') {
                flag = false;
            }
        }
        System.out.println(flag ? "YES" : "NO");
    }
}
