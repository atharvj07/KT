#include <bits/stdc++.h>
using namespace std;

const int mxN = 501, mod = 1e9 + 7;
vector<bool> isPrime(mxN);
int n, m;

bool isPrimeMatrix(const vector<vector<int>>& a) {
    for (int i = 0; i < n; ++i) {
        bool allPrime = true;
        for (int j = 0; j < m; ++j) {
            allPrime &= isPrime[a[i][j]];
        }
        if (allPrime) {
            return true;
        }
    }
    for (int j = 0; j < m; ++j) {
        bool allPrime = true;
        for (int i = 0; i < n; ++i) {
            allPrime &= isPrime[a[i][j]];
        }
        if (allPrime) {
            return true;
        }
    }
    return false;
}

void sieve() {
    isPrime[1] = false;
    isPrime[2] = true;
    for (int p = 3; p * p <= mxN; p += 2) {
        if (isPrime[p]) {
            for (int i = p * p; i < mxN; i += p) {
                isPrime[i] = false;
            }
        }
    }
}

int main() {
    cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m, 0));
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            cin >> a[i][j];
        }
    }
    sieve();
    int ans = mxN;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            for (int k = a[i][j]; ; ++k) {
                if (isPrime[k]) {
                    ans = min(ans, k - a[i][j]);
                    break;
                }
            }
        }
    }
    if (isPrimeMatrix(a)) {
        ans = 0;
    }
    cout << ans << "\n";
    return 0;
}
