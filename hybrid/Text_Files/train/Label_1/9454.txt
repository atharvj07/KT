#include <cstdio>
#define rep(i, j) for(int i=0;i<j;++i)
#define REP(i, l, r) for(int i=(l);i<(r);++i)
using namespace std;
char s[505][505], t[505][505];
bool isrt[510][510];
int cnt;
void dfs(int x, int y){
	if(x<0||y<0||x>=502||y>=502||isrt[x][y]||s[x][y]=='.') return;
	isrt[x][y]=1, ++cnt;
	dfs(x+1, y), dfs(x-1, y), dfs(x, y+1), dfs(x, y-1);
}
int main(){
	int n, m, cnt1=0;
	scanf("%d%d", &n, &m);
	rep(i, n) rep(j, m){
		scanf(" %c", &s[i+1][j+1]);
		if(s[i+1][j+1]=='X') ++cnt1;
	} rep(i, 502) rep(j, 502) t[i][j] = s[i][j];
	bool r = 0;
	rep(i, 502) rep(j, 502){
		if(!isrt[i][j]&&s[i][j]=='X'){
			cnt=0, dfs(i, j); r+=(cnt+cnt1==n*m);
		}
	}
	rep(i, 502) rep(j, 502){
		if(!isrt[i][j]) r = 0;
	} puts(r?"YES":"NO");
	return 0;
}
