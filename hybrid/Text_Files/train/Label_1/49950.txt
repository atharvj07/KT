import java.util.*;

class Main {
    static class Node {
        BitSet suffix;
        BitSet all;
        boolean ambig;

        Node() {
            this.suffix = new BitSet();
            this.all = new BitSet();
        }
    }

    public static void main(String[] arg) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Node[] nodes = new Node[n];
        for (int i = 0; i < n; i++) {
            nodes[i] = new Node();
        }

        for (int i = 0; i < n; i++) {
            String str = sc.next();
            BitSet bs = new BitSet();
            for (int k = str.length() - 1; k >= 0; k--) {
                if (str.charAt(k) == '1') {
                    bs.set(str.length() - 1 - k);
                }
            }
            nodes[i].all.or(bs);
            nodes[i].suffix.or(bs);
        }

        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                boolean all = nodes[i].all.equals(nodes[j].all);
                BitSet suffix = (BitSet) nodes[i].suffix.clone();
                suffix.and(nodes[j].all);
                boolean suff = suffix.cardinality() > 0 && !all;
                if (all || suff) {
                    nodes[i].ambig = true;
                    nodes[j].ambig = true;
                }
            }
        }

        boolean found = false;
        int min = 1000;
        for (int i = 0; i < n; i++) {
            if (nodes[i].ambig) {
                found = true;
                int card = nodes[i].all.cardinality();
                if (card < min) { min = card; }
            }
        }

        if (found) {
            System.out.println(min);
        } else {
            System.out.println(0);
        }
    }
}
