#include <iostream>
#include <string>

using namespace std;

int main() {
  int n;
  string s;
  cin >> n >> s;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'e' || s[i] == 'o') {
      if (i + 1 < n && s[i] == s[i + 1] && i + 2 < n && s[i] == s[i + 2]) {
        i += 2;
        cout << s[i];
      } else {
        cout << s[i];
      }
    } else {
      cout << s[i];
    }
  }
  cout << endl;
  return 0;
}
