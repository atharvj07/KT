#include <iostream>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '?') {
      if (cnt == 0) {
        s[i] = '(';
      } else {
        s[i] = ')';
      }
      cnt++;
    } else if (s[i] == '(') {
      cnt++;
    } else {
      cnt--;
    }
    if (cnt < 0) {
      cout << ":(";
      return 0;
    }
  }
  if (cnt == 0) {
    cout << s;
  } else {
    cout << ":(";
  }
  return 0;
}
