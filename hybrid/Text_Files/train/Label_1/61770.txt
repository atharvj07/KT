mod = 10**9 + 7
n, t = map(int, input().split())
songs = []
for _ in range(n):
    ti, gi = map(int, input().split())
    songs.append((ti, gi))
dp = [[[0] * 4 for _ in range(4)] for _ in range(t + 1)]
for i in range(t + 1):
    for last in range(4):
        for use in range(4):
            if i != 0:
                dp[i][last][use] = dp[i - 1][last][use]
            if i >= songs[use - 1][0] and last != songs[use - 1][1]:
                dp[i][songs[use - 1][1]][use] = (dp[i][songs[use - 1][1]][use] + dp[i - songs[use - 1][0]][last][use]) % mod
            if use == 0:
                dp[i][last][use] = (dp[i][last][use] + 1) % mod
print(dp[t][0][0])
