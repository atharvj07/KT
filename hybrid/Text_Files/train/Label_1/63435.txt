import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
spt = list(map(int, input().split()))
g = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

vis = [False] * (n + 1)
d = [0] * (n + 1)

def dfs(u):
    vis[u] = True
    for v in g[u]:
        if not vis[v]:
            d[v] = d[u] + 1
            dfs(v)

ans = 0

for x in spt:
    dfs(x)
    if max(d) > ans:
        ans = max(d)
        a = x
    d = [0] * (n + 1)
    vis = [False] * (n + 1)

for x in spt:
    dfs(x)
    if max(d) > ans:
        ans = max(d)
        b = x
    d = [0] * (n + 1)
    vis = [False] * (n + 1)

g[a].append(b)
g[b].append(a)
dfs(1)
print(max(d))
