import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n], y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
        }

        int ans = 1;
        for (int i = 0; i < n; i++) {
            int j = (i + 1) % n;
            int k = (i + 2) % n;
            if (((x[j] - x[i]) * (y[k] - y[j]) - (x[k] - x[j]) * (y[j] - y[i])) < 0) {
                ans = 0;
            }
        }

        System.out.println(ans);
    }
}
