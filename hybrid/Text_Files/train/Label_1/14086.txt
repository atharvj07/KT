#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }
  int m = a[0];
  int ans = 1;
  for (int i = 1; i < n; ++i) {
    if (a[i] > m) {
      m = a[i];
      ++ans;
    } else if (a[i] == m) {
      continue;
    } else {
      break;
    }
  }
  m = a[n - 1];
  for (int i = n - 2; i >= 0; --i) {
    if (a[i] < m) {
      m = a[i];
      ++ans;
    } else if (a[i] == m) {
      continue;
    } else {
      break;
    }
  }
  std::cout << ans << std::endl;
  for (int i = 0; i < ans; ++i) {
    std::cout << a[i] << ' ';
  }
  for (int i = n - 1; i >= ans; --i) {
    std::cout << a[i] << ' ';
  }
  std::cout << std::endl;
  return 0;
}
