#include <bits/stdc++.h>
using namespace std;
int main() {
  int t, m = 0, sm;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s[2], x;
    m = 0, sm = 0;
    for (int i = 0; i < 2; i++) { cin >> x, s[i] = " " + x; }
    for (int i = 1; i <= n; i++)
      if (s[0][i] == s[1][i])
        m += (s[0][i] == '0'), sm += (s[0][i] == '1');
      else
        sm += 2;
    cout << m + sm << "\n";
  }
}
