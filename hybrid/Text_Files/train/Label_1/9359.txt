#include <cstdio>
#include <vector>
typedef long long ll;
const ll N = 1e5 + 10;
std::vector<ll> g[N], ans;
ll n, m, v, u, ok, vis[N];
void dfs(ll x) {
  if (vis[x] == 2)
    return;
  if (vis[x] == 1) {
    ok = 0;
    return;
  }
  vis[x] = 1;
  for (ll i : g[x])
    dfs(i);
  ans.push_back(x);
  vis[x] = 2;
}
int main() {
  scanf("%lld%lld", &n, &m);
  for (ll i = 0; i < n; i++) {
    for (ll j = 1; j <= m; j++) {
      scanf("%lld", &v);
      if (i > 0)
        g[v].push_back((i - 1) * m + j);
      if (j > 1)
        g[v].push_back(i * m + j - 1);
    }
  }
  ok = 1;
  for (ll i = 1; i <= n * m; i++)
    if (!vis[i])
      dfs(i);
  if (!ok)
    printf("NO");
  else {
    printf("YES\n");
    for (ll i = ans.size() - 1; i >= 0; i--) {
      printf("%lld ", ans[i]);
      if (i % m == 0)
        printf("\n");
    }
  }
  return 0;
}
