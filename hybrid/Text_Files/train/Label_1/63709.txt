for _ in range(int(input())):
    x, y, z = map(int, input().split())
    if x == z and z == y:
        print("YES")
        print(x, y, z)
    elif x <= y and y <= z:
        a = x
        b = max(y, z)
        c = min(y, z)
        print("YES")
        print(a, b, c)
    else:
        print("NO")
