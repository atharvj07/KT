#include <bits/stdc++.h>
using namespace std;
#define int long long
#define endl '\n'
#define for0(i, n) for(int i=0; i<n; i++)
#define for1(i, n) for(int i=1; i<=n; i++)
#define clr(a, n) memset(a, n, sizeof(a))
#define all(v) (v).begin(), (v).end()
int mod = 1e9+7;

const int maxn = 3e5+5;
vector<int> low(maxn), high(maxn), bit(maxn), mp(maxn);
int n, k;
void upd(int pos, int val) {
    while(pos <= n) {
        bit[pos] = (bit[pos] + val)%mod;
        pos += pos&-pos;
    }
}

int get(int pos) {
    int ans = 0;
    while(pos >= 1) {
        ans = (ans + bit[pos])%mod;
        pos -= pos&-pos;
    }

    return ans;
}

int32_t main() {
#ifndef ONLINE_JUDGE
    freopen("input.txt", "r", stdin);
#endif
    cin >> n >> k;
    for0(i, n) {
        cin >> low[i] >> high[i];
        mp[low[i]]++; mp[high[i]+1]--;
    }

    for0(i, n) mp[i+1] += mp[i];
    for0(i, n) {
        if(!mp[i]) continue;
        upd(i, mp[i]); upd(high[i]+1, mod-mp[i]);
    }

    int ans = 0;
    for0(i, n) {
        int tmp = get(i+1);
        if(tmp == k) ans++;
    }

    cout << ans << endl;

    return 0;
}
