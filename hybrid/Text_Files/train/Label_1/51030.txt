import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        long[] a = new long[2 * n];
        long[] b = new long[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            a[i] = sc.nextLong();
        }
        for (int i = 0; i < 2 * n; i++) {
            b[i] = sc.nextLong();
        }
        long[][] dp = new long[2 * n][2];
        dp[2 * n - 1][0] = 0;
        dp[2 * n - 1][1] = b[2 * n - 1];
        for (int i = 2 * n - 2; i >= 0; i--) {
            dp[i][0] = Math.max(dp[i + 1][0] + a[i], dp[i + 1][1] + b[i]);
            dp[i][1] = Math.max(dp[i + 1][1] + a[i], dp[i + 1][0] + b[i]);
        }
        for (int i = 0; i < q; i++) {
            int p = sc.nextInt() - 1;
            long x = sc.nextLong();
            long y = sc.nextLong();
            a[p] = x;
            b[p] = y;
            long val = dp[0][0];
            for (int j = 1; j < 2 * n; j++) {
                val = Math.max(val, dp[j][0] + dp[j][1] - b[j]);
            }
            System.out.println(val);
        }
    }
}
