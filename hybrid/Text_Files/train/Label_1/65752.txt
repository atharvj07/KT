import sys
input = sys.stdin.readline
for _ in range(int(input())):
    s = input().strip()
    a=[0]*3
    for c in s: a[int(c)]+=1
    ans = 1e10
    for lft in range(3):
        ans = min(ans, a[lft]+(a[(lft+1)%3]==0)+(a[(lft+2)%3]==0))
    for rgt in range(3):
        ans = min(ans, a[rgt]+(a[(rgt+1)%3]==0)+(a[(rgt+2)%3]==0))
    print(ans)
