#include<bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> p(n + 1), ans;
    for (int i = 1; i <= n; i++) { cin >> p[i]; }
    for (int i = 1; i <= n; i++)
      if (ans.empty() || ans.back() != p[i]) ans.push_back(p[i]);
    int sz = ans.size();
    cout << sz << '\n';
    for (int x : ans) cout << x << ' ';
    cout << '\n';
  }
  return 0;
}
