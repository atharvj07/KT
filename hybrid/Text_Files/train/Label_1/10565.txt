#include <iostream>
#include <set>

using namespace std;

int main() {
    int n;
    cin >> n;
    if (n == 1) {
        cout << -1 << endl;
        return 0;
    }
    set<pair<int, int>> s;
    for (int i = 0; i < n; i++) {
        int x, y;
        cin >> x >> y;
        s.insert(make_pair(x, y));
    }
    int x1 = (*s.begin()).first;
    int y1 = (*s.begin()).second;
    s.erase(s.begin());
    int x2 = (*s.begin()).first;
    int y2 = (*s.begin()).second;
    s.erase(s.begin());
    int x3 = (*s.begin()).first;
    int y3 = (*s.begin()).second;
    s.erase(s.begin());
    int area = abs((x2 - x1) * (y3 - y1) - (x3 - x1) * (y2 - y1));
    cout << area << endl;
    return 0;
}
