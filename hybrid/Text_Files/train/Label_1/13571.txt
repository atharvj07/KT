#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;

    std::vector<int> m(n);
    for (int i = 0; i < n; i++) {
        std::cin >> m[i];
    }

    int sum = 0;
    int min = 0;
    for (int i = 0; i < n; i++) {
        min = std::min(min, m[i]);
        sum += min;
    }

    std::cout << sum << std::endl;
    return 0;
}
