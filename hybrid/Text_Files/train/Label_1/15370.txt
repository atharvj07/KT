#include <bits/stdc++.h>
using namespace std;
long long x, y, ans = 1e18, sumx, sumy;
int main() {
  cin >> x >> y;
  string s;
  cin >> s;
  for (int i = 0; i < s.length(); i++) {
    if (s[i] == 'U')
      sumy++;
    if (s[i] == 'D')
      sumy--;
    if (s[i] == 'L')
      sumx--;
    if (s[i] == 'R')
      sumx++;
  }
  if (x != sumx || y != sumy)
    cout << -1 << endl;
  else
    cout << 0 << endl;
  return 0;
}
