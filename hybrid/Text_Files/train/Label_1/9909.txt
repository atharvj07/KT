#include <vector>
#include <cstring>
#include <iostream>
#include <algorithm>
#include <queue>
using namespace std;

int n, m;
vector<bool> visited;
vector<int> graph[1001];
queue<int> q;
bool checkRedundancy;

bool valid(int src) {
  checkRedundancy = false;
  visited.assign(n+1, false);
  q.push(src);
  visited[src] = true;
  while (!q.empty()) {
    int cur = q.front(); q.pop();
    for (int next : graph[cur]) {
      if (!visited[next]) {
        q.push(next); visited[next] = true;
      } else {
        checkRedundancy = true;
      }
    }
  }
  return !checkRedundancy;
}

bool solvable() {
  for (int i=1; i<=n; i++) {
    if (!valid(i)) return false;
  }
  return true;
}

int main() {
  freopen("zombies.in","r",stdin);
  cin >> n >> m;
  for (int i=0; i<m; i++) {
    int a, b;
    cin >> a >> b;
    graph[a].push_back(b); graph[b].push_back(a);
  }
  cout << (solvable() ? "yes" : "no") << endl;
}
