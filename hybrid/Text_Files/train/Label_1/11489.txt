#include <iostream>
#include <vector>

using namespace std;

int main() {
  string s;
  cin >> s;
  int k;
  cin >> k;
  int n = s.length();
  vector<int> dp(n + 1, -1);
  dp[n] = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == '?') {
      dp[i] = max(dp[i + 1], dp[i + 2]);
    } else if (s[i] == '*') {
      if (i + 1 < n) dp[i] = max(dp[i + 1], dp[i + 2] + 1);
      else dp[i] = -1;
    } else {
      if (dp[i + 1] != -1) dp[i] = max(dp[i + 1], dp[i + 1] + 1);
      else dp[i] = -1;
    }
  }
  if (dp[0] == -1 || dp[0] < k)
    cout << "Impossible" << endl;
  else {
    string ans = "";
    int i = 0;
    while (i < n) {
      if (s[i] == '?') {
        if (dp[i + 1] >= k - ans.length()) {
          i++;
        } else {
          ans += s[i - 1];
          i += 2;
        }
      } else if (s[i] == '*') {
        ans += s[i - 1];
        i++;
      } else {
        ans += s[i];
        i++;
      }
    }
    while (ans.length() < k) ans += ans.back();
    cout << ans << endl;
  }
  return 0;
}
