n=int(input());a=list(map(int,input().split()))
for i in range(n<<1):
    for j in range(n<<1):
        if i!=j and a[i]+a[j]!=a[(i+n-1)%(n<<1)]+a[(j+n-1)%(n<<1)]:
            print(*[a[k]for k in range(i,(i+n)%26)]+[a[k]for k in range(j,(j+n)%26)]+[a[k]for k in range(i,(i+n)%26)]+[a[k]for k in range(j,(j+n)%26)])
            exit()
print(-1)
