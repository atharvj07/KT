import sys
import math

n = int(input())

def is_prime(n):
    if n < 2:
        return False
    elif n % 2 == 0 and n != 2:
        return False
    else:
        for i in range(3, int(math.sqrt(n)) + 1, 2):
            if n % i == 0:
                return False
        return True

i = 3
while i * i <= n:
    if is_prime(i) and is_prime(n - i):
        print(2)
        print(i, n - i)
        sys.exit()
    i += 2

i = 3
while i * i * i <= n:
    if is_prime(i) and is_prime(n - i - i):
        print(3)
        print(i, i, n - i - i)
        sys.exit()
    i += 2
