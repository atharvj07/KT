import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
        }
        int answer = -1;
        for (int i = 1; i <= n; i++) {
            if (a[i] % (n + 1) == 0) {
                answer = i;
                break;
            }
        }
        System.out.println(answer);
    }
}
