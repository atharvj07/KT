n = int(input())
legs = list(map(int, input().split()))
energy = list(map(int, input().split()))

legs_energy = list(zip(legs, energy))
legs_energy.sort(key=lambda x: x[0])
legs_sorted = [leg for leg, _ in legs_energy]

unique_legs = list(set(legs_sorted))
unique_legs.sort()

costs = [0] * len(unique_legs)
for i, leg_length in enumerate(unique_legs):
    costs[i] = sum(en for l, en in legs_energy if l < leg_length)

total_energy = sum(energy)
min_energy = total_energy
for i, leg_length in enumerate(unique_legs):
    count = legs_sorted.count(leg_length)
    current_energy = (total_energy - sum(en for l, en in legs_energy if l == leg_length))
    current_cost = costs[i] if i > 0 else 0
    if n - count < count:
        min_energy = min(min_energy, current_cost)
    else:
        required = count - (n - count) + 1
        if required > 0:
            leg_energy = sorted([en for l, en in legs_energy if l == leg_length], reverse=True)
            additional_cost = sum(leg_energy[:-required])
            min_energy = min(min_energy, current_cost + additional_cost)

print(min_energy)
