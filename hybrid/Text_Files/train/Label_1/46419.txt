import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        Arrays.sort(x);

        // Find the maximum power of two that divides any pair of points
        int maxPower = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int diff = x[j] - x[i];
                if ((diff & (diff + 1)) == 0) {
                    int power = 0;
                    while ((diff >>= 1) != 0) {
                        power++;
                    }
                    maxPower = Math.max(maxPower, power);
                }
            }
        }

        // Find the subset of points that satisfy the condition
        List<Integer> subset = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            boolean found = false;
            for (int j = 0; j < subset.size(); j++) {
                int diff = x[i] - subset.get(j);
                if ((diff & (diff + 1)) != 0) {
                    found = true;
                    break;
                }
            }
            if (!found) {
                subset.add(x[i]);
            }
        }

        // Print the size of the subset and the coordinates of the points in the subset
        System.out.println(subset.size());
        for (int point : subset) {
            System.out.print(point + " ");
        }
    }
}
