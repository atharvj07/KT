#include <algorithm>
#include <iostream>
#include <vector>
#define MOD 1000000007

int n, ans = 1;
std::vector<int> p;
int main() {
    std::cin >> n;
    p.resize(n + 1);
    for (int i = 0; i < n; i++) {
        int x;
        std::cin >> x;
        if (x != -1) p[x] = i + 1;
    }
    for (int i = 1; i <= n; i++) {
        if (p[i]) ans = (1LL * ans * (i - 1)) % MOD;
    }
    std::cout << ans << std::endl;
    return 0;
}
