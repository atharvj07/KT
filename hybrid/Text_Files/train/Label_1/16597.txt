#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 1e5 + 1;
const int MOD = 998244353;

int n, m;
vector<long long> a;
long long dp[MAXN][2][2];

int main() {
    cin >> n >> m;
    a.resize(n + 1);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }

    dp[n][0][0] = dp[n][0][1] = dp[n][1][0] = dp[n][1][1] = 1;
    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j < 2; j++) {
            for (int k = 0; k < 2; k++) {
                dp[i][j][k] = dp[i + 1][j][k];
                if (j == 0 && a[i + 1] % 2 == 0) dp[i][j || 0][k] = (dp[i][j || 0][k] + dp[i + 1][0][k]) % MOD;
                if (k == 0 && a[i + 1] % 2 == 1) dp[i][j || 1][k] = (dp[i][j || 1][k] + dp[i + 1][j][0]) % MOD;
                if (j == 1 && k == 1) dp[i][j][k] = (dp[i][j][k] + dp[i + 1][j][k]) % MOD;
                if (j == 0) dp[i][j][k] = (dp[i][j][k] + dp[i + 1][j | k][k]) % MOD;
                if (k == 0) dp[i][j][k] = (dp[i][j][k] + dp[i + 1][j][j | k]) % MOD;
            }
        }
    }

    cout << dp[0][1][1] << endl;

    return 0;
}
