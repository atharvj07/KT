#include <bits/stdc++.h>
using namespace std;
const int maxn=1005;
vector<int>G[maxn];
bool vis[maxn];
int main()
{
    int n;
    cin>>n;
    for(int i=1;i<=n;++i)
    {
        int a,b;
        cin>>a>>b;
        G[a].push_back(b);
    }
    int cnt=0;
    for(int i=1;i<=n;++i)
    {
        if(vis[i]==0)
        {
            bool flag=0;
            queue<int>q;
            q.push(i);
            while(!q.empty())
            {
                int t=q.front();
                q.pop();
                if(vis[t]) continue;
                vis[t]=1;
                for(auto j:G[t])
                {
                    if(j==i)
                        flag=1;
                    q.push(j);
                }
            }
            if(!flag) cnt++;
        }
    }
    cout<<cnt<<endl;
}
