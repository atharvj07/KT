n = int(input())
bombs = [list(map(int, input().split())) for _ in range(n)]

# Sort bombs by their distance from the origin to ensure we pick the closest bombs first
bombs.sort(key=lambda x: abs(x[0]) + abs(x[1]))

# The minimum number of operations needed is 6 operations per bomb: 
# Move to bomb, pick it up, return to origin, drop it off, destroy it, return to origin
operations = []
for x, y in bombs:
    if x > 0:
        operations.append(f"1 {x} R")
    elif x < 0:
        operations.append(f"1 {-x} L")
    if y > 0:
        operations.append(f"1 {y} U")
    elif y < 0:
        operations.append(f"1 {-y} D")
    operations.append("2")
    if y > 0:
        operations.append(f"1 {y} D")
    elif y < 0:
        operations.append(f"1 {-y} U")
    if x > 0:
        operations.append(f"1 {x} L")
    elif x < 0:
        operations.append(f"1 {-x} R")
    operations.append("3")

print(len(operations))
print("\n".join(operations))
