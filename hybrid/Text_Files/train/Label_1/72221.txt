d = int(input())
n, m = map(int, input().split())
sofas = [list(map(int, input().split())) for _ in range(d)]
cnt_l, cnt_r, cnt_t, cnt_b = map(int, input().split())

# Initialize counters
left_counts = [0] * d
right_counts = [0] * d
top_counts = [0] * d
bottom_counts = [0] * d

# Populate counters
for i in range(d):
    for j in range(d):
        if i != j:
            # Check left/right
            if sofas[i][0] > sofas[j][0] or sofas[i][2] > sofas[j][2]:
                left_counts[i] += 1
            elif sofas[i][0] < sofas[j][0] or sofas[i][2] < sofas[j][2]:
                right_counts[i] += 1
            # Check top/bottom
            if sofas[i][1] > sofas[j][1] or sofas[i][3] > sofas[j][3]:
                top_counts[i] += 1
            elif sofas[i][1] < sofas[j][1] or sofas[i][3] < sofas[j][3]:
                bottom_counts[i] += 1

# Identify Grandpa Maks's sofa
for i in range(d):
    if left_counts[i] == cnt_l and right_counts[i] == cnt_r and top_counts[i] == cnt_t and bottom_counts[i] == cnt_b:
        print(i + 1)
        break
else:
    print(-1)
