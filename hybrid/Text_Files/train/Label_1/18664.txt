#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    long long n, m, k, ans = 0, cur = 0, last = -1;
    cin >> n >> m >> k;
    vector <long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    reverse(a.begin(), a.end());
    for (int i = 0; i < n; i++) {
        if (a[i] == last) cur++;
        else cur = 0;
        if (cur >= k) continue;
        long long canuse = min(k - cur, m);
        ans += canuse * a[i];
        m -= canuse;
        last = a[i];
    }
    cout << ans << endl;
}
