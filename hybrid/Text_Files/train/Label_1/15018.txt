#include<bits/stdc++.h>
using namespace std;
int main(){
  int n; cin>>n;
  map<int,int>a,b;
  for(int i=0,x;i<n;i++) cin>>x, a[x]++;
  for(int i=0,x;i<n;i++) cin>>x, b[x]++;
  int ans = 0;
  for(auto x: a){
    if(a[x.first] != b[x.first]){
      ans += abs(a[x.first] - b[x.first]);
    }
  }
  if(ans&1){ cout << -1; return 0; }
  cout << ans/2;
}
