import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    Map<String, Integer> cur = new HashMap<>();
    Map<String, Integer> prev = new HashMap<>();

    for (int i = 0; i < n; i++) {
      String shirt = sc.next();
      prev.put(shirt, prev.getOrDefault(shirt, 0) + 1);
    }

    for (int i = 0; i < n; i++) {
      String shirt = sc.next();
      cur.put(shirt, cur.getOrDefault(shirt, 0) + 1);
    }

    int total = 0;
    for (String key : cur.keySet()) {
      if (!prev.containsKey(key)) {
        total += cur.get(key);
      } else if (!prev.get(key).equals(cur.get(key))) {
        total += Math.abs(cur.get(key) - prev.get(key));
      }
    }

    System.out.println(total);
  }
}
