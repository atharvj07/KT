import sys
input=sys.stdin.readline
n,k=map(int,input().split())
s=input()
ans=-1
ST=sorted(range(26))
for l in s:
    if (i:=ST.index(ord(l)-ord('a')))>=k:break
    ans=l
if ans=='-1':print(-1)
else:
    ST=ST[i+2:]
    for j in ST:
        x=chr(j+ord('a'))
        if s.find(x)>=0 and (i==j or ST.index(j)>j-i-2):
            ans+=x
    print(sum(ord(x)-ord('a')+1 for x in ans))
