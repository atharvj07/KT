n = int(input())
votes = list(map(int, input().split()))

max_votes_for_elodreip = max(votes)
total_votes_for_elodreip = sum(votes)
k = max_votes_for_elodreip

while True:
    total_votes_for_awruk = n*k - total_votes_for_elodreip
    if total_votes_for_awruk > total_votes_for_elodreip:
        print(k)
        break
    k += 1
