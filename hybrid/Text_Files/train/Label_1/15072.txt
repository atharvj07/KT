#include<bits/stdc++.h>
#define ll long long
#define mod 1000000007
using namespace std;
const int mxN = 1e5+5;
vector<int>adj[mxN], radj[mxN];
int dist[mxN], cost[mxN], par[mxN];
bool vis[mxN];
void dfs(int u, int p){
    vis[u]=1;
    par[u] = p;
    for(int i=0; i<adj[u].size(); i++){
        int v= adj[u][i];
        if(vis[v]==0){
            dfs(v, u);
        }
    }
}
int f(int u){
    vis[u]=1;
    int ans =0;
    if(par[u]!=-1) ans = cost[par[u]];
    for(int i=0; i<radj[u].size(); i++){
        int v = radj[u][i];
        if(vis[v]==0) ans+= f(v);
    }
    return ans;
}
int main(){
    int n; cin>>n;
    for(int i=1; i<=n; i++) cin>>cost[i];
    int m; cin>>m;
    while(m--){
        int u, v; cin>>u>>v;
        adj[u].push_back(v);
        radj[v].push_back(u);
    }
    dfs(1,-1);
    memset(vis, 0, sizeof(vis));
    ll ans = LLONG_MAX;
    int cnt =0;
    for(int i=1; i<=n; i++){
        if(vis[i]==0){
            ll x = f(i);
            if(x < ans){
                ans = x;
                cnt=1;
            }
            else if(x==ans) cnt++;
        }
    }
    cout<<ans<<" "<<cnt<<endl;
}
