#include <bits/stdc++.h>
using namespace std;
#define ll long long
template <typename T> using min_heap = priority_queue<T, vector<T>, greater<T>>;
int n, m;
char a[2005][2005];
int dp[2005][2005][2][2];
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) cin >> a[i][j];
  }
  dp[1][1][0][1] = dp[1][1][1][0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (a[i][j] == '#') continue;
      if (i == 1 || j == 1 || i == n || j == m) {
        if (i + 1 <= n && a[i + 1][j] != '#') dp[i + 1][j][0][1] = (dp[i + 1][j][0][1] + dp[i][j][0][1]) % 998244353;
        if (j + 1 <= m && a[i][j + 1] != '#') dp[i][j + 1][1][0] = (dp[i][j + 1][1][0] + dp[i][j][1][0]) % 998244353;
      } else {
        if (a[i - 1][j] != '#' && a[i][j + 1] != '#') dp[i][j + 1][0][1] = (dp[i][j + 1][0][1] + dp[i][j][0][1] + dp[i][j][0][0]) % 998244353;
        if (a[i][j + 1] != '#' && a[i + 1][j] != '#') dp[i + 1][j][1][0] = (dp[i + 1][j][1][0] + dp[i][j][1][0] + dp[i][j][0][0]) % 998244353;
        if (a[i][j - 1] != '#' && a[i + 1][j] != '#') dp[i + 1][j][0][0] = (dp[i + 1][j][0][0] + dp[i][j][0][1] + dp[i][j][1][0]) % 998244353;
        if (a[i - 1][j] != '#' && a[i][j + 1] != '#') dp[i][j + 1][1][1] = (dp[i][j + 1][1][1] + dp[i][j][1][0] + dp[i][j][0][1]) % 998244353;
        if (a[i - 1][j] != '#') dp[i][j][0][0] = (dp[i][j][0][0] + dp[i - 1][j][0][1]) % 998244353;
        if (a[i][j + 1] != '#') dp[i][j][1][0] = (dp[i][j][1][0] + dp[i][j + 1][1][0]) % 998244353;
        if (a[i + 1][j] != '#') dp[i][j][0][1] = (dp[i][j][0][1] + dp[i + 1][j][0][1]) % 998244353;
        if (a[i][j - 1] != '#') dp[i][j][1][1] = (dp[i][j][1][1] + dp[i][j - 1][1][1]) % 998244353;
      }
    }
  }
  cout << dp[n][m][0][1] << "\n";
  return 0;
}
