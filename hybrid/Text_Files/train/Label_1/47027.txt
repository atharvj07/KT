import java.util.Scanner;

public class Lemmings {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] shields = new int[n];
        for (int i = 0; i < n; i++) {
            shields[i] = scanner.nextInt();
        }

        int[][] dp = new int[n + 1][(n * (n - 1)) / 2 + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= (n * (n - 1)) / 2; j++) {
                dp[i][j] = -1;
            }
        }

        System.out.println(solve(shields, 0, 0, dp));
    }

    private static int solve(int[] shields, int idx, int orders, int[][] dp) {
        if (idx == shields.length) {
            return orders;
        }

        if (dp[idx][orders] != -1) {
            return dp[idx][orders];
        }

        int ans = solve(shields, idx + 1, orders, dp);

        if (idx > 0 && shields[idx - 1] == 0 && shields[idx] == 1) {
            ans = Math.max(ans, solve(shields, idx + 1, orders + 1, dp));
        }

        if (idx < shields.length - 1 && shields[idx + 1] == 0 && shields[idx] == 1) {
            ans = Math.max(ans, solve(shields, idx + 1, orders + 1, dp));
        }

        dp[idx][orders] = ans;
        return ans;
    }
}
