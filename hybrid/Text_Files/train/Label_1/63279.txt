def normalize_name(name):
    name = name.replace("oo", "u")
    name = name.replace("kh", "h")
    return name

n = int(input().strip())
names = [input().strip() for _ in range(n)]
normalized_names = set()

for name in names:
    normalized_names.add(normalize_name(name))

print(len(normalized_names))
