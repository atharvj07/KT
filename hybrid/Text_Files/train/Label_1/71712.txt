n = int(input())
board = [list(input()) for _ in range(n)]

def can_tile(board):
    for i in range(n-2):
        for j in range(n-2):
            if board[i][j] == '.' and board[i+1][j] == '.' and board[i+1][j-1] == '.' and board[i+1][j+1] == '.' and board[i+2][j] == '.':
                board[i][j] = board[i+1][j] = board[i+1][j-1] = board[i+1][j+1] = board[i+2][j] = '#'
    for row in board:
        if '.' in row:
            return "NO"
    return "YES"

print(can_tile(board))
