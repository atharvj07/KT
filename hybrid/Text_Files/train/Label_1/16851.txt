#include <iostream>
using namespace std;
int main(){
  int n,m,i,j,b=0,k,f=1;
  cin>>n>>m;
  char c[n+1][m+1];
  for(i=1;i<=n;++i){
    for(j=1;j<=m;++j){
      cin>>c[i][j];
      if(c[i][j]=='*') ++b;
    }
  }
  for(i=1;i<=n;++i){
    for(j=1;j<=m;++j){
      if(c[i][j]=='.'){
        k=0;
        if(c[i+1][j]=='*') ++k;
        if(c[i-1][j]=='*') ++k;
        if(c[i][j+1]=='*') ++k;
        if(c[i][j-1]=='*') ++k;
        if(c[i+1][j+1]=='*') ++k;
        if(c[i-1][j-1]=='*') ++k;
        if(c[i+1][j-1]=='*') ++k;
        if(c[i-1][j+1]=='*') ++k;
        if(k) f=0;
      }
      else if(c[i][j]!='*'){
        k=c[i][j]-'0';
        if(c[i+1][j]=='*') --k;
        if(c[i-1][j]=='*') --k;
        if(c[i][j+1]=='*') --k;
        if(c[i][j-1]=='*') --k;
        if(c[i+1][j+1]=='*') --k;
        if(c[i-1][j-1]=='*') --k;
        if(c[i+1][j-1]=='*') --k;
        if(c[i-1][j+1]=='*') --k;
        if(k) f=0;
      }
    }
  }
  if(f&&b==n*m/4) cout<<"YES";
  else cout<<"NO";
  return 0;
}
