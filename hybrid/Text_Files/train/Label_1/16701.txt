#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n, k, sd = 0;
    cin >> n >> k;
    vector<pair<long long, long long>> v(n);
    for (auto &p : v) {
        cin >> p.first >> p.second;
        sd += p.second - p.first + 1;
    }
    if (sd % k == 0) {
        cout << 0;
        return 0;
    }
    sort(v.begin(), v.end());
    long long need = (k - sd % k) % k, mmin = 1e18;
    for (int i = 0; i < n; i++) {
        long long l = max(v[i].first, (v[i].second + need + k - 1) / k * k),
                   r = (v[i].second + 1);
        if (r < l) continue;
        if (r - l + 1 < mmin) {
            mmin = r - l + 1;
        }
    }
    cout << mmin;
    return 0;
}
