from collections import Counter

n, k = map(int, input().split())

x = Counter()

for i in range(n):
    x[int(input())] += 1

ans = 0
deltas = sorted([x[i + 1] - x[i] for i in range(k - 1)])

for d in deltas:
    if d <= 0: continue
    if d <= ans: ans += d // 2
    else: break

print(2 * ans + (x[1] + k // 2) // k)
