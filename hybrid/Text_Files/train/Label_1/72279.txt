a, b = map(int, input().split())
res = []
while a < b:
    if not b % 2:
        res.append(b)
        b //= 2
    elif b % 10 == 1:
        res.append(b)
        b = (b - 1) // 10
    else:
        break
if a != b:
    res = ["NO"]
else:
    res.reverse()
    res[0] = "YES"
    res.insert(1, str(len(res) - 1))
print(*res, sep="\n")
