#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    multiset<ll> ms;
    for (int i = 0; i < n << 1; i++) {
      ll x;
      cin >> x;
      ms.insert(x);
    }
    if (ms.size() == n && *ms.begin() >= 0)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
