import sys
input = sys.stdin.readline

n, k = map(int, input().split())
a = []; b = []; t = []
for i in range(n):
    ti, ai, bi = map(int, input().split())
    a.append(ai); b.append(bi); t.append(ti)
dp = [[[-1] * 3 for _ in range(k + 1)] for _ in range(n + 1)]
dp[n][k][2] = 0
for i in range(n - 1, -1, -1):
    for j in range(k + 1):
        dp[i][j][0] = dp[i + 1][j][0]
        dp[i][j][1] = dp[i + 1][j][1]
        dp[i][j][2] = dp[i + 1][j][2]
        if j > 0:
            if a[i] == 1:
                if dp[i + 1][j - 1][0] == -1:
                    dp[i + 1][j - 1][0] = dp[i][j][0] + t[i]
                else:
                    dp[i + 1][j - 1][0] = min(dp[i][j][0] + t[i], dp[i + 1][j - 1][0])
            if b[i] == 1:
                if dp[i + 1][j - 1][1] == -1:
                    dp[i + 1][j - 1][1] = dp[i][j][1] + t[i]
                else:
                    dp[i + 1][j - 1][1] = min(dp[i][j][1] + t[i], dp[i + 1][j - 1][1])
            if a[i] == 1 and b[i] == 1:
                if dp[i + 1][j - 1][2] == -1:
                    dp[i + 1][j - 1][2] = dp[i][j][2] + t[i]
                else:
                    dp[i + 1][j - 1][2] = min(dp[i][j][2] + t[i], dp[i + 1][j - 1][2])
print(dp[0][k][2])
