#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll inf = 1e18;

int main() {
    int n;
    cin >> n;
    vector<ll> a(n), b(n), c(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    for (int i = 0; i < n; i++) cin >> c[i];
    vector<vector<ll>> dp(n + 1, vector<ll>(3, -inf));
    dp[0][0] = 0;  // n hares, fed = false
    dp[0][1] = 0;  // n hares, fed = true
    for (int i = 0; i < n; i++) {
        for (int fed = 0; fed <= 1; fed++) {
            if (dp[i][fed] == -inf)
                continue;
            // feed i-th hare
            if (i != n - 1)
                dp[i + 1][true] = max(dp[i][fed], dp[i][fed] + (fed ? b[i] : c[i]));
            if (i)
                dp[i + 1][false] = max(dp[i][fed], dp[i][fed] + (fed ? c[i] : a[i]));
            // don't feed i-th hare
            dp[i + 1][fed] = max(dp[i + 1][fed], dp[i][fed]);
        }
    }
    cout << dp[n][true] << endl;
    return 0;
}
