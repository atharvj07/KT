import math
s = input()
l = len(s)
s = sorted(s)
res = []
i = 0
while i < l:
    j = i
    t = [s[i]]
    while j + 1 < l and s[j] == s[j + 1]:
        j += 1
        t += s[j]
    res += [''.join(t)]
    i = j + 1
k = math.ceil(l / len(res[0]))
print(k)
for i in range(k - 1):
    print(res[0] * (l // k), end = ' ')
print(res[0] * (l % k))
