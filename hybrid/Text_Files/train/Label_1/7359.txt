#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
    ll n, sx, sy, dx, dy, t; cin >> n >> sx >> sy >> dx >> dy >> t;
    ll x = sx, y = sy;
    ll tt = t % 2;
    t /= 2;
    dx = ((dx % n) + n) % n, dy = ((dy % n) + n) % n;
    x = (x + t * dx + (tt * dx)) % n, y = (y + t * dy + (tt * dy)) % n;
    if (x == 0) x = n;
    if (y == 0) y = n;
    cout << x << " " << y << endl;
    return 0;
}
