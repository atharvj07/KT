#include<bits/stdc++.h>
#define ll long long int
using namespace std;
int main()
{
    ll a,b,x,y,sum=0,g;
    cin>>a>>b>>x>>y;
    g=__gcd(x,y);
    x/=g;
    y/=g;
    for(ll i=1;i*x<=a;i++)
    {
        ll j=i*y;
        if(j<=b)
        {
            sum++;
        }
    }
    cout<<sum<<endl;
    return 0;
}
