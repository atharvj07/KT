#include <vector>
#include <iostream>
#define int long long
using namespace std;

int getMEX(vector<int> &v) {
    int mex = 0;
    while (true) {
        bool found = false;
        for (int x : v) {
            if (x == mex)
                found = true;
        }
        if (!found)
            return mex;
        mex++;
    }
}

int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> v(n);
        for (int i = 0; i < n; i++) { cin >> v[i]; }

        int ans = getMEX(v);
        cout << ans << '\n';
    }
    return 0;
}
