#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, ans = 0, type = 0;
    cin >> n;
    vector<int> have(3, 0), cost(3, INT_MAX);
    for (int i = 0; i < n; i++) {
        string s;
        int c;
        cin >> c >> s;
        for (char ch : s) have[ch - 'A'] = 1;
        if (s != "") cost[s.size() - 1] = min(cost[s.size() - 1], c);
    }
    bool flag = 1;
    for (int i = 0; i < 3; i++) {
        type += have[i];
        ans += cost[i];
        flag &= have[i];
    }
    if (!flag)
        ans = -1;
    else if (type == 2)
        ans -= cost[0];
    cout << ans << endl;
    return 0;
}
