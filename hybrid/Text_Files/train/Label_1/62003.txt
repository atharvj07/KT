n = int(input())
a = set(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
c = list(map(int, input().split()))
cnt_a = [0] * 1000001
cnt_b = [0] * 1000001
max_a = 0
max_b = 0
ans = 0
for x in a:
    cnt_a[x] += 1
for i in range(m):
    cnt_b[b[i]] += 1
    cnt_b[c[i]] += 1
    if cnt_b[b[i]] > max_b:
        max_b = cnt_b[b[i]]
        ans = i + 1
    if cnt_b[c[i]] > max_b:
        max_b = cnt_b[c[i]]
        ans = i + 1
for x in a:
    if cnt_a[x] > max_a:
        max_a = cnt_a[x]
if max_a > max_b:
    print(0)
else:
    print(ans)
