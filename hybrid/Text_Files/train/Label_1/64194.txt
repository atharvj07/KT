n, m = [int(x) for x in input().split()]
arr = list(map(int, input().split()))
ops = []
for i in range(m):
    ops.append([int(x) for x in input().split()])

for op in ops:
    if op[0] == 1:
        arr[op[1]-1] = op[2]
    elif op[0] == 2:
        for i in range(n):
            arr[i] += op[1]
    else:
        print(arr[op[1]-1])
