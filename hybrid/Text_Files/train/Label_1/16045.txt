#include <bits/stdc++.h>
using namespace std;
#define all(v) v.begin(), v.end()
#define int long long
const int mod = 1e9 + 7;
const long long inf = 1e18;
template <typename T>
istream &operator>>(istream &is, vector<T> &v) {
  int n;
  is >> n;
  v.resize(n);
  for (int &it : v) is >> it;
  return is;
}
void solve() {
  int n;
  cin >> n;
  vector<int> p(n), s(n);
  map<int, int> mp;
  for (int &it : p) cin >> it;
  for (int i = 0; i < n; i++) {
    cin >> s[i];
    mp[s[i]] = i;
  }
  int ans = 0, cnt = 0;
  for (int i = 0; i < n; i++) {
    if (p[i] != s[i]) {
      ans += i - mp[p[i]];
      mp[p[i]] = mp[s[i]];
      swap(p[i], p[mp[p[i]]]);
      cnt++;
    }
  }
  cout << ans << endl;
  cout << cnt << endl;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (s[j] < s[i]) {
        cout << j + 1 << " " << i + 1 << endl;
        swap(p[j], p[i]);
        swap(s[j], s[i]);
      }
    }
  }
}
signed main() {
  solve();
  return 0;
}
