#include<bits/stdc++.h>
using namespace std;
const int maxn=2e5+5;
int a[maxn],b[maxn];
int n,tmp[maxn];
int main()
{
 cin>>n;
 for (int i=1; i<=n; i++) cin>>a[i];
 for (int i=1; i<=n; i++) b[a[i]]=i;
 for (int i=1; i<=n; i++) tmp[i]=b[i];
 for (int i=1; i<=n; i++) cin>>a[i];
 for (int i=1; i<=n; i++)
 {
  while (tmp[a[i]]!=i) a[i]=b[a[i]];
  cout<<a[i]<<" ";
  tmp[a[i]]=i;
 }
 return 0;
}
