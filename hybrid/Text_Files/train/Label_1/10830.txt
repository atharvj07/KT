#include<bits/stdc++.h>

using namespace std;

const int maxn = 3e5 + 5;
int a[maxn], pre[maxn], nxt[maxn], ans[maxn];
int32_t main() {
    int n, k, l = maxn, r = 0, cnt = 0;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        if (a[i]) {
            cnt++;
            if (i > r) {
                r = i;
                if (l == maxn) { l = i; }
            }
        }
    }
    //cout << l << ' ' << r << ' ' << cnt << endl;
    if (cnt + k >= n) {
        cout << n << '\n';
        for (int i = 1; i <= n; i++) { cout << 1; }
        cout << '\n';
        return 0;
    }
    for (int i = 1; i <= n; i++) {
        if (a[i]) { pre[i] = max(pre[i - 1], i - l); }
        else { pre[i] = pre[i - 1]; }
    }
    for (int i = n; i >= 1; i--) {
        if (a[i]) { nxt[i] = min(nxt[i + 1], r - i); }
        else { nxt[i] = nxt[i + 1]; }
    }
    for (int i = 1; i <= n; i++) {
        if (!a[i]) {
            int op_l = pre[i - 1];
            int op_r = nxt[i + 1];
            ans[i] = 1;
            if (op_l + op_r + 1 > cnt + k) { ans[i] = 0; }
            else if (op_l + op_r + 1 == cnt + k) {
                if (op_l <= k) { ans[i] = 1; }
                else { ans[i] = 0; }
            }
        }
        cout << ans[i];
    }
    cout << '\n';
}
