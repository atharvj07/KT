n, k = map(int, input().split())
segments = [tuple(map(int, input().split())) + (i + 1,) for i in range(n)]

def solve(n, k, segments):
    points = {}
    for l, r, i in segments:
        for point in range(l, r + 1):
            if point not in points:
                points[point] = []
            points[point].append(i)

    removed_segments = set()
    while True:
        max_cover = 0
        for point, covering in points.items():
            if len(covering) > k:
                if len(covering) > max_cover:
                    max_cover = len(covering)
                    max_point = point

        if max_cover == 0:
            break

        remove_seg = points[max_point][-1]
        removed_segments.add(remove_seg)
        for point in list(points.keys()):
            if remove_seg in points[point]:
                points[point].remove(remove_seg)
                if len(points[point]) == 0:
                    del points[point]

    return len(removed_segments), sorted(removed_segments)

m, removed = solve(n, k, segments)
print(m)
print(' '.join(map(str, [seg for seg in removed])))
