#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    char arr[n];
    int ch = 'a';
    int l = 0, h = k-1;
    while( l <= h ){
        for(int i=l; i<=h; i++){
            arr[i] = ch++;
        }
        l++,h--, ch = 'a';
    }
    for(int i=0; i<n; i++){
        cout << arr[i];
    }
    cout << '\n';
}
