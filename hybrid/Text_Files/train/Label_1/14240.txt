#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n,k;
        cin>>n>>k;
        int ans[n+1],cnt=0;
        for(int i=1; i<=n; i++)
        {
            if(i!=k)
            {
                ans[cnt++]=i;
            }
        }
        cout<<cnt<<endl;
        for(int i=0; i<cnt; i++)
        {
            cout<<ans[i]<<" ";
        }
        cout<<endl;
    }
    return 0;
}
