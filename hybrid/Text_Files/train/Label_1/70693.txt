for _ in range(int(input())):
    n, m = map(int, input().split())
    s = input()
    p = [int(i) for i in input().split()]
    ans = [0] * 26
    for i in range(n):
        for j in range(m):
            if p[j] <= i:
                ans[ord(s[i]) - ord('a')] += 1
        ans[ord(s[i]) - ord('a')] += 1
    print(*ans)
