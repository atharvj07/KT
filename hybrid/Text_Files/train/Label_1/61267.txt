from collections import Counter

def is_vowelly(word):
    n = len(word)
    m = n // 5
    if n % 5 != 0:
        return False
    for i in range(m):
        for j in range(5):
            if word[i * 5 + j] not in 'aeiou':
                return False
    for i in range(5):
        for j in range(m):
            if word[i + j * 5] not in 'aeiou':
                return False
    return True

k = int(input())
if k % 5 != 0 or k < 10:
    print(-1)
else:
    ans = 'aeiou' * (k // 5)
    if is_vowelly(ans):
        print(ans)
    else:
        print(-1)
