#include <bits/stdc++.h>
using namespace std;

bool SWAP(int &a, int &b)
{
    if (a > b)
    {
        a = a ^ b;
        b = a ^ b;
        a = a ^ b;
        return 1;
    }
    return 0;
}

int check(int a[8][3])
{
    int vis[999] = {0};
    for (int i = 0; i < 8; i++)
    {
        int sum = 0;
        for (int j = 0; j < 3; j++)
        {
            vis[a[i][j]]++;
            sum += a[i][j];
        }
        if (sum != 0 && sum != 3) return 1;
        if (vis[a[i][0]] > 1 || vis[a[i][1]] > 1 || vis[a[i][2]] > 1) return 1;
    }
    return 0;
}

int main()
{
    int a[8][3];
    for (int i = 0; i < 8; i++)
        cin >> a[i][0] >> a[i][1] >> a[i][2];
    int fl = check(a);
    for (int i = 0; i < 8; i++)
        for (int j = 0; j < 3; j++)
        {
            for (int k = j + 1; k < 3; k++)
            {
                if ((SWAP(a[i][j], a[i][k])) == 1) fl = check(a);
            }
        }
    if (fl == 1)
        cout << "NO";
    else
    {
        cout << "YES" << endl;
        for (int i = 0; i < 8; i++)
        {
            cout << a[i][0] << ' ' << a[i][1] << ' ' << a[i][2] << endl;
        }
    }
}
