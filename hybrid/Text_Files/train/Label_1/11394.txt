#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 1e5 + 10;
int num[maxn], nl[maxn], nr[maxn], dp[maxn];
int main() {
    string s;
    cin >> s;
    int n = s.size();
    for (int i = 0; i < n; ++i) {
        num[s[i]]++;
        nl[s[i]] = i;
    }
    for (int i = n - 1; i >= 0; --i) {
        nr[s[i]] = i;
    }
    int ans = 0;
    for (char c = 'a'; c <= 'z'; ++c) {
        dp[nl[c]] = 1;
        for (int i = nl[c] + 1; i < n; ++i) {
            if (nr[c] > i && dp[i - 1] > 0) { dp[i] = dp[i - 1] + 1; }
            ans = max(dp[i], ans);
        }
    }
    cout << ans << endl;
    return 0;
}
