n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]

def check(i, j, k, l):
    for a in range(i, k+1):
        for b in range(j, l+1):
            if a>=0 and a<n and b>=0 and b<m and grid[a][b]=='X':
                return False
    return True

ans = 'NO'
for i in range(n):
    for j in range(m):
        if grid[i][j]=='X':
            for k in range(i, n):
                for l in range(j, m):
                    if grid[k][l]=='X':
                        if check(i, j, k, l) and check(i+1, j+1, k-1, l-1):
                            ans = 'YES'
                            break
                if ans=='YES':
                    break
            if ans=='YES':
                break
print(ans)
