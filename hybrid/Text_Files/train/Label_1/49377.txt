import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        for (int t = 0; t < T; t++) {
            int n = sc.nextInt();
            int maxIndex = 1;
            int minIndex = 1;
            for (int i = 2; i <= n; i++) {
                System.out.println("? " + maxIndex + " " + i);
                System.out.flush();
                String result = sc.next();
                if (result.equals("<")) {
                    maxIndex = i;
                }
                System.out.println("? " + minIndex + " " + i);
                System.out.flush();
                result = sc.next();
                if (result.equals(">")) {
                    minIndex = i;
                }
            }
            System.out.println("! " + minIndex + " " + maxIndex);
            System.out.flush();
        }
    }
}
