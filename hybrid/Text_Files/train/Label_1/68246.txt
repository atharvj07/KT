n, k = map(int, input().split())
arr = list(map(int, input().split()))

def count_subsegments(n, k, arr):
    count = 0
    for i in range(n):
        product = 1
        summ = 0
        for j in range(i, n):
            product *= arr[j]
            summ += arr[j]
            if product == k * summ:
                count += 1
            if product > k * summ or product % k != 0:
                break
    return count

print(count_subsegments(n, k, arr))
