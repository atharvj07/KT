import sys
input=sys.stdin.readline

def dfs(u, p):
    dp[u]=1
    for v in g[u]:
        if v==p: continue
        dfs(v, u)
        if d[u][x]!=-1 and d[v][y]!=-1 and d[u][y]>d[v][x]: dp[u]+=dp[v]

n, x, y = map(int, input().split())
g=[[]]
for i in range(n): g.append([])

d=[[0]*3 for i in range(n+1)]
for i in range(1, n):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

def dist(u, p, k, depth):
    if d[u][k]!=-1: return d[u][k]
    d[u][k]=depth
    for v in g[u]:
        if v==p: continue
        d[u][k]=min(d[u][k], dist(v, u, k, depth+1))
    return d[u][k]

dist(x, -1, x, 0)
dist(y, -1, y, 0)

dp=[0]*(n+1)
dfs(1, -1)

print(dp[1])
