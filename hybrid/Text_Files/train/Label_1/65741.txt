n = int(input())
a = list(map(int, input().split()))
m = int(input())
q = list(map(int, input().split()))

# Calculate the cumulative sum to determine the range of each pile
cumulative_a = [0] * n
current_sum = 0
for i in range(n):
    current_sum += a[i]
    cumulative_a[i] = current_sum

for worm in q:
    # Binary search to find the pile containing the worm
    left, right = 0, n - 1
    while left <= right:
        mid = (left + right) // 2
        if cumulative_a[mid] < worm:
            left = mid + 1
        else:
            right = mid - 1
    print(left + 1)
