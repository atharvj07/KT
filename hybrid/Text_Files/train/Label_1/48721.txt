import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    sc.nextLine();
    String s = sc.nextLine();
    String t = sc.nextLine();
    int ans = -1;
    for (int l = Math.max(n, m); l <= 100000; l++) {
      if (l % n == 0 && l % m == 0) {
        boolean ok = true;
        for (int i = 0; i < n; i++) {
          for (int j = 0; j < l / n; j++) {
            if (s.charAt(i) != t.charAt((j + i) % m)) {
              ok = false;
              break;
            }
          }
          if (!ok) {
            break;
          }
        }
        if (ok) {
          ans = l;
          break;
        }
      }
    }
    System.out.println(ans);
  }
}
