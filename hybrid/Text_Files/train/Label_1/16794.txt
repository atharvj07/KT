#include<bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;
int32_t main() {
    int n, k;
    cin >> n >> k;
    vector<int> p(n);
    for(int i = 0; i < n; i++) {
        cin >> p[i];
    }
    long long ans = 1;
    for(int i = 1; i < n; i++) {
        if(p[i - 1] > p[i]) {
            ans = (ans * k) % MOD;
        }
    }
    cout << ans << endl;
    return 0;
}
