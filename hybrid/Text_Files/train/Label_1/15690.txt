#include <bits/stdc++.h>
using namespace std;

const int INF = 1e9 + 7;
const int MAXN = 200005;
vector<int> G[MAXN];
int d[MAXN];

int dfs(int u, int p, int sum) {
    for (int v : G[u]) {
        if (v == p) continue;
        sum = dfs(v, u, sum + 1);
    }
    return d[u] = sum;
}

int main() {
    int n, x, y; cin >> n >> x >> y;
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        G[u].push_back(v);
        G[v].push_back(u);
    }
    int sum = dfs(1, 1, 0);
    int mx = 0, id = -1;
    for (int i = 1; i <= n; i++) {
        if (d[i] > mx) {
            mx = d[i];
            id = i;
        }
    }
    sum = dfs(id, id, 0);
    int ans = 1LL * (sum / 2) * x + 1LL * (sum - sum / 2) * y;
    cout << ans << endl;
}
