import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    while (true) {
      int an = sc.nextInt(), bn = sc.nextInt(), r = sc.nextInt();
      if (an == 0 && bn == 0) {
        break;
      }

      int[][] a = new int[an][2], b = new int[bn][2];
      for (int i = 0; i < an; i++) {
        a[i][0] = sc.nextInt();
        a[i][1] = sc.nextInt();
      }

      for (int i = 0; i < bn; i++) {
        b[i][0] = sc.nextInt();
        b[i][1] = sc.nextInt();
      }

      int cnt = 0;
      for (int[] ai : a) {
        for (int[] bi : b) {
          if (Math.pow(ai[0] - bi[0], 2) + Math.pow(ai[1] - bi[1], 2) <= Math.pow(2 * r, 2)) {
            cnt++;
          }
        }
      }

      System.out.println(cnt);
    }
  }
}
