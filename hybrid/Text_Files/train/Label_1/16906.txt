#include<bits/stdc++.h>

using namespace std;

const int MAXN = 45;
const long long INF = 2e18;

long long t[MAXN][MAXN];
long long dp[MAXN][MAXN][MAXN];

long long solve(int n, int src, int dest) {
    if(n == 0) {
        return 0;
    }
    if(dp[n][src][dest] != -1) {
        return dp[n][src][dest];
    }
    long long ans = INF;
    for(int i = 1; i <= 3; i++) {
        if(i != src && i != dest) {
            ans = min(ans, t[src][i] + solve(n - 1, i, dest));
        }
    }
    dp[n][src][dest] = ans;
    return ans;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    for(int i = 1; i <= 3; i++) {
        for(int j = 1; j <= 3; j++) {
            cin >> t[i][j];
        }
    }
    cin >> n;
    memset(dp, -1, sizeof dp);
    cout << solve(n, 1, 3) << endl;
    return 0;
}
