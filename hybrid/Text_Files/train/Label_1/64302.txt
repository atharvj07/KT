n = int(input())
a = list(map(int,input().split()))
a_sort = sorted(a)
f = 0
for i in range(n-1):
    if(a[i]>a[i+1]):
        f = 1
        break
inc = []
dec = []
if(f==0):
    inc = a
else:
    l = []
    r = []
    i = 0
    j = n-1
    while(i<j):
        while(a_sort[i]<=a_sort[j]):
            i+=1
        while(a_sort[j]>=a_sort[i]):
            j-=1
        if(i<j):
            l.append(a_sort[i])
            r.append(a_sort[j])
            i+=1
            j-=1
    if(len(l)>len(r)):
        l,r = r,l
    for el in r:
        inc.append(el)
    for el in l:
        dec.append(el)
    if(a!=sorted(inc+dec)):
        print("NO")
        exit()
print("YES")
print(len(inc))
for el in inc:
    print(el,end=" ")
print()
print(len(dec))
for el in dec:
    print(el,end=" ")
print()
