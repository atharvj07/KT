from queue import PriorityQueue

n, k = map(int, input().split())
c = [list(input()) for _ in range(n)]
t = [[c[i][j] for j in range(n)] for i in range(n)]
q = PriorityQueue()
for i in range(n):
    for j in range(n):
        q.put((chr(min(ord(c[i][j]), ord(t[i][j]))) + chr(max(ord(c[i][j]), ord(t[i][j]))), (i, j), 0))
while k > 0 and not q.empty():
    v, (x, y), d = q.get()
    if d == 0:
        t[x][y] = v[0]
    else:
        t[x][y] = v[1]
    k -= 1
    if x + 1 < n:
        q.put((v + t[x + 1][y], (x + 1, y), d + 1))
    if y + 1 < n:
        q.put((v + t[x][y + 1], (x, y + 1), d + 1))
print(''.join([c for r in t for c in r]))
