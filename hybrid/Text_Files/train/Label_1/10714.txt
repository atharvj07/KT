#include<bits/stdc++.h>
using namespace std;
int vis[105][105];
struct d{ int x,y; };
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n,m;
        cin>>n>>m;
        queue<d>q;
        for(int i=1;i<=n;i++)
        {
            for(int j=1;j<=m;j++)
            {
                int x;
                cin>>x;
                if(x) q.push({i,j});
            }
        }
        int f=0;
        while(q.size())
        {
            int x=q.front().x;
            int y=q.front().y;
            q.pop();
            if(vis[x][y]) continue;
            if((x==1)&&(y==1))
            {
                f=1;
                break;
            }
            vis[x][y]=1;
            if(x>1) q.push({x-1,y});
            if(y>1) q.push({x,y-1});
        }
        if(f) cout<<"Ashish"<<endl;
        else cout<<"Jeel"<<endl;
    }
    return 0;
}
