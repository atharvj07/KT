from sys import stdin
input = stdin.readline

def solve():
    n=int(input())
    a=list(map(int,input().split()))
    r=[i for i in range(1,n+1,2)]
    r2=r[-1::-1]
    b=[0]*n
    for i in a:
        b[i-1]=1
    if b[n-1]==0 or b[0]==1:
        print(-1)
        return
    i=n-1
    while b[i]!=0:
        i-=1
    for j in reversed(r):
        if i<j:
            break
        print(1,j,end=' ')
        for k in range(0,j//2):
            a[i-k],a[i-j+1+k]=a[i-j+1+k],a[i-k]
        i-=j
    for j in r2:
        if i<j:
            break
        print(1,j,end=' ')
        for k in range(0,j//2):
            a[i-k],a[i-j+1+k]=a[i-j+1+k],a[i-k]
        i-=j
    print()

for i in range(int(input())):
    solve()
