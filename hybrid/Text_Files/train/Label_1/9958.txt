#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    map<int,char>mp;
    for(int i=1; i<=n; i++)
    {
        char ch;
        cin>>ch;
        mp[i]=ch;
    }
    for(int i=1; i<=n/2; i++)
    {
        int x=i+n/2;
        if(mp[i]!=mp[x])
            cout<<i<<' '<<x<<endl;
        else
        {
            x++;
            if(x>n)
                x=n/2+1;
            cout<<x<<' '<<i<<endl;
        }
    }
    return 0;
}
