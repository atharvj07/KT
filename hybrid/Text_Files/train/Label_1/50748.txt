import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Warehouse {

    private static class Section {
        int shelf;
        int index;

        Section(int shelf, int index) {
            this.shelf = shelf;
            this.index = index;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int shelves = scanner.nextInt();
        int sections = scanner.nextInt();
        int operations = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        Map<String, Section> boxes = new HashMap<>();

        for (int i = 0; i < operations; i++) {
            String line = scanner.nextLine();
            String[] parts = line.split(" ");
            if (parts[0].equals("+1")) {
                String id = parts[3];
                int shelf = Integer.parseInt(parts[1]) - 1;
                int section = Integer.parseInt(parts[2]) - 1;
                boxes.put(id, new Section(shelf, section));
            } else if (parts[0].equals("-1")) {
                String id = parts[1];
                Section section = boxes.remove(id);
                if (section != null) {
                    System.out.println(section.shelf + 1 + " " + (section.index + 1));
                } else {
                    System.out.println("-1 -1");
                }
            }
        }
    }
}
