for t in range(int(input())):
    n,l,r=map(int,input().split())
    if n>4 or ((n-1) * n < r-l+1):
        print(1)
    else:
        p=[]
        for i in range(1,n+1):
            if l>1:
                p.append(i)
                l-=1
            else:
                d=1
                if r==n*(n-1):
                    d=n
                if r%n==0:
                    r=r//n
                else:
                    r=r//n+1
                if r<d:
                    r=n
                while d<=r and p[-1]!=d:
                    p.append(d)
                    d+=1
                l=d+1
                r=r-n+1
        print(*p)
