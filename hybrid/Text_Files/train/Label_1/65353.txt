grid = []
for i in range(3):
    grid.append([int(x) for x in input().split()])

for i in range(3):
    for j in range(3):
        if (i == 0 or grid[i - 1][j] % 2 == 1) ^ (i == 2 or grid[i + 1][j] % 2 == 1) ^ (j == 0 or grid[i][j - 1] % 2 == 1) ^ (j == 2 or grid[i][j + 1] % 2 == 1):
            print('1', end='')
        else:
            print('0', end='')
    print()
