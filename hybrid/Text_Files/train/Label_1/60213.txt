import sys
line=[sys.stdin.readline().strip('X.') for i in range(4)]
mini=min(len(l.split('_')) for l in line)
maxi=max(len(l.split('_')) for l in line)
if len(list(filter(lambda l:len(l.split('_'))==mini,line)))+len(list(filter(lambda l:len(l.split('_'))==maxi,line)))==1:print(''.join(filter(lambda l:len(l.split('_')) in [mini,maxi],line))[0][0])
else:print('C')
