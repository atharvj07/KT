from sys import stdin

n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]
stars = []
for i in range(1, n + 1):
    for j in range(1, m + 1):
        if grid[i - 1][j - 1] == '*' and not any(
            [
                (i - a <= s <= i + a) and (j - b <= s <= j + b)
                for a, b, s in stars
            ]
        ):
            stars.append((i, j, 1))
            for x in range(i - 1, -1, -1):
                if grid[x][j - 1] == '.':
                    break
                stars[-1] = (x + 1, j, stars[-1][2] + 1)
            for x in range(i + 1, n + 1):
                if grid[x][j - 1] == '.':
                    break
                stars[-1] = (x - 1, j, stars[-1][2] + 1)
            for y in range(j - 1, -1, -1):
                if grid[i - 1][y] == '.':
                    break
                stars[-1] = (i, y + 1, stars[-1][2] + 1)
            for y in range(j + 1, m + 1):
                if grid[i - 1][y] == '.':
                    break
                stars[-1] = (i, y - 1, stars[-1][2] + 1)
            if any(
                [
                    (i - a <= s <= i + a) and (j - b <= s <= j + b)
                    for a, b, s in stars[:-1]
                ]
            ):
                stars.pop()

if all([grid[i - 1][j - 1] == '.' for i, j in sum(stars, ())]):
    print(len(stars))
    for x, y, s in stars:
        print(x, y, s)
else:
    print(-1)
