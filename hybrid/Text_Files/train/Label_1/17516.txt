#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<int>> grid(n, vector<int>(m));
        for (int i = 0; i < n; i++) {
            string s;
            cin >> s;
            for (int j = 0; j < m; j++) {
                grid[i][j] = (s[j] == '*');
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j]) {
                    int h = 1;
                    while (i + h - 1 < n && j - h + 1 >= 0 && j + h - 1 < m && grid[i + h - 1][j - h + 1] && grid[i + h - 1][j + h - 1]) {
                        h++;
                    }
                    ans += h - 1;
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
