import sys
T = int(input())
for _ in range(T):
    n = int(input())
    A = []
    B = []
    for i in range(n):
        a, b = map(int, sys.stdin.readline().split())
        A += [a, b]
        B += [b, a]
    A.sort()
    B.sort()
    C = []
    for i in range(n):
        C += [B[i], A[i]]
    C.sort()
    ans = float('inf')
    for i in range(n-1):
        ans = min(ans, C[i+1]-C[i])
    print(ans)
