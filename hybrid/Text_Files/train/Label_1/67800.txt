from collections import deque, defaultdict

def bfs(graph, start):
    visited = {start}
    queue = deque([start])
    parent = {start: None}

    while queue:
        vertex = queue.popleft()
        for neighbour in graph[vertex]:
            if neighbour not in visited:
                visited.add(neighbour)
                parent[neighbour] = vertex
                queue.append(neighbour)
    return parent

def solve(n, m, D, edges):
    graph = defaultdict(list)
    degree = defaultdict(int)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
        if u == 1 or v == 1:
            degree[1] += 1

    if degree[1] < D or len(graph[1]) < D:
        print("NO")
        return

    parent = bfs(graph, 1)
    if len(parent) != n:
        print("NO")
        return

    answer_edges = []
    used_edges = set()
    for child, par in parent.items():
        if par is not None:
            answer_edges.append((par, child))
            used_edges.add((par, child))
            used_edges.add((child, par))

    # Check if we can adjust the degree of vertex 1 to D by adding/removing edges directly connected to it
    directly_connected = [edge for edge in edges if 1 in edge and edge not in used_edges]
    if len(answer_edges) - degree[1] + len(directly_connected) < D:
        print("NO")
        return

    while degree[1] > D:
        for edge in answer_edges:
            if 1 in edge:
                answer_edges.remove(edge)
                degree[1] -= 1
                break

    while degree[1] < D:
        for edge in directly_connected:
            if edge not in answer_edges and (edge[1], edge[0]) not in answer_edges:
                answer_edges.append(edge)
                degree[1] += 1
                break

    print("YES")
    for u, v in answer_edges:
        print(u, v)

n, m, D = map(int, input().split())
edges = [tuple(map(int, input().split())) for _ in range(m)]
solve(n, m, D, edges)
