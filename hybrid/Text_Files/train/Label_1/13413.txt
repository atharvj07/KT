#include <iostream>
#include <vector>

using namespace std;

typedef unsigned long long ull;

const int mod = 1e9 + 7;

const int N = 40000;

bool vis[N];
int num[N];
vector<int> prime;

void Euler() {
  for (int i = 2; i < N; i++) {
    if (!vis[i]) {
      prime.push_back(i);
      num[i] = 1;
    }
    for (int j = 0; j < prime.size(); j++) {
      if (i * prime[j] >= N) break;
      vis[i * prime[j]] = 1;
      if (i % prime[j] == 0) {
        num[i * prime[j]] = num[i] + 1;
        break;
      } else {
        num[i * prime[j]] = 1;
      }
    }
  }
}

ull pow_mod(ull a, ull p) {
  if (p == 0) return 1;
  if (p == 1) return a;
  if (p % 2) return a * pow_mod(a * a % mod, p / 2) % mod;
  return pow_mod(a * a % mod, p / 2) % mod;
}

int main() {
  Euler();
  ull t, a, b;
  cin >> t >> a >> b;
  if (t == 1) {
    if (b == a)
      cout << 2 << endl;
    else
      cout << "inf" << endl;
  } else {
    ull ans = num[b];
    a %= mod;
    ull tmp = a;
    for (int i = 0; i < prime.size(); i++) {
      if (tmp % prime[i] == 0) {
        ans -= num[prime[i]];
        while (tmp % prime[i] == 0) tmp /= prime[i];
      }
    }
    if (tmp != 1) ans--;
    cout << pow_mod(2, ans) << endl;
  }
  return 0;
}
