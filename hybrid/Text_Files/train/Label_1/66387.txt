import sys

for t in range(int(sys.stdin.readline())):
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    b = [a[i] % 2 == (a[(i + 1) % n] % 2) for i in range(n)]
    if any(b):
        print(-1)
        continue
    pos = n
    for i in range(n):
        if a[i] % 2 == 1:
            pos = i
            break
    swaps = ((pos - 1) // 2 + (n - 1 - pos) // 2) % 2
    print(swaps)
