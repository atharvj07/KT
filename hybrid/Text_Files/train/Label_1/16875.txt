#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,ans=0;
    cin>>n;
    int a[n];
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    while(1){
        bool c=1;
        for(int i=0;i<n;i++){
            if(a[i]==0)continue;
            if(i==0 || i==n-1 || a[i-1]==0 || a[i+1]==0){
                c=0;
                a[i]--;
            }
        }
        if(c)break;
        ans++;
    }
    cout<<ans<<'\n';
}
