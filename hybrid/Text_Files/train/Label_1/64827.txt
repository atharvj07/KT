def f(a, b):
    c, d = list(str(a)), list(str(b))
    ans = []
    while c or d:
        if c:
            ans.append(c.pop(0))
        if d:
            ans.append(d.pop(0))
    return int(''.join(ans))

n = int(input())
a = [int(i) for i in input().split()]

ans = 0
for i in range(n):
    for j in range(n):
        ans = (ans + f(a[i], a[j])) % 998244353

print(ans)
