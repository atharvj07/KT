k = int(input())

if k % 2 == 0:
    print("NO")
else:
    print("YES")
    n = 2 * (k + 1)
    m = k * (k + 1)
    print(n, m)
    for i in range(1, k + 2):
        for j in range(1, k + 1):
            if i == j:
                continue
            if i < k + 1:
                print(i, k + 1 + ((i + j - 2) % k) + 1)
            else:
                if j <= k // 2:
                    print(i, j)
