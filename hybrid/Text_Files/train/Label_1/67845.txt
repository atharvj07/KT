n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

b.sort()
possible_x = []

for i in range(n):
    initial_x = (b[i] - a[0] + m) % m
    temp_a = [(x + initial_x) % m for x in a]
    temp_a.sort()
    if temp_a == b:
        possible_x.append(initial_x)

print(min(possible_x))
