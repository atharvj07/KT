#include <iostream>

using namespace std;

const int L = 200000;
char p[L+1];
int st[L+1];
int x[L+1];

int main() {
    int n, j;
    cin >> n;
    cin >> p;
    for (j = 1; j <= n; ++j) {
        if (p[j] == '(') {
            ++st[0];
            x[st[0]] = j;
        }
        else {
            if (!st[0]) {
                x[++st[0]] = j;
            }
            else {
                --st[0];
            }
        }
    }
    if (!st[0]) {
        cout << "Yes" << endl;
    }
    else if (st[0] == 1) {
        if (st[1] != 1 && x[st[1]] - x[st[1]-1] >= 2) {
            cout << "Yes" << endl;
        }
        else {
            cout << "No" << endl;
        }
    }
    else {
        cout << "No" << endl;
    }
    return 0;
}
