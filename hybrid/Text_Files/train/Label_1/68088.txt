import sys

def solve():
    n, m, q = map(int, sys.stdin.readline().split())
    samples = set()
    for _ in range(q):
        samples.add(tuple(map(int, sys.stdin.readline().split())))
    missing = n * m - q
    if missing % 2 == 0 or missing % 4 == 1:
        return missing // 2
    return missing // 2 + 1

print(solve())
