#include <iostream>
#include <vector>
using namespace std;
const int mod = 1e9;
int main() {
  long long n, q, k;
  cin >> n >> q >> k;
  vector<long long> a(n);
  for (long long i = 0; i < n; i++) {
    cin >> a[i];
  }
  while (q--) {
    long long l, r;
    cin >> l >> r;
    l--;
    r--;
    long long sum = ((a[r] * (a[r] + 1)) / 2) % mod;
    if (l > 0) {
      sum -= ((a[l - 1] * (a[l - 1] + 1)) / 2) % mod;
    }
    sum = ((sum % mod) + mod) % mod;
    long long ans = (sum * k) % mod;
    sum = ((a[l] * (a[l] + 1)) / 2) % mod;
    if (r > l) {
      sum -= ((a[r - 1] * (a[r - 1] + 1)) / 2) % mod;
    }
    ans = ((ans - sum) % mod + mod) % mod;
    cout << ans << "\n";
  }
  return 0;
}
