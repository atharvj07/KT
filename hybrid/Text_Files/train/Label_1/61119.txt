import sys
input = sys.stdin.readline

n, k = map(int, input().split())
li = [[] for _ in range(n+1)]

for _ in range(n-1):
    a, b = map(int, input().split())
    li[a].append(b)
    li[b].append(a)

cen = 0
for i in range(1, n+1):
    if len(li[i]) > 3:
        cen = i
        break

que = [cen]
deg = [0] * (n+1)
deg[cen] = 3

exist = False

while que:
    x = que.pop()
    if len(li[x]) == 1:
        exist = True
    for i in li[x]:
        if deg[i] == 0:
            que.append(i)
            deg[i] = 1

if cen == 0 or not exist or k > len(li[cen]):
    print("No")
else:
    print("Yes")
