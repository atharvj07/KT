#include <bits/stdc++.h>
using namespace std;

using ll = long long;
const ll MAXN = 300005;

ll n;
ll a[MAXN], b[MAXN];
vector<ll> divs[MAXN];
vector<pair<ll, ll>> ans;
ll max_len = 0;

int main()
{
    cin >> n;
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
        b[i] = a[i];
    }
    sort(b, b + n);
    for (ll i = 0; i < n; i++) {
        for (ll j = 2; j * j <= b[i]; j++) {
            if (b[i] % j == 0) {
                while (b[i] % j == 0) {
                    b[i] /= j;
                }
                divs[i].push_back(j);
            }
        }
        if (b[i] != 1) {
            divs[i].push_back(b[i]);
        }
    }
    for (ll i = 0; i < n; i++) {
        for (auto& d : divs[i]) {
            if (a[i] % d == 0) {
                for (ll j = i + 1; j < n; j++) {
                    if (a[j] % d == 0) {
                        if (j - i > max_len) {
                            max_len = j - i;
                            ans.clear();
                            ans.push_back({i, j});
                        } else if (j - i == max_len) {
                            ans.push_back({i, j});
                        }
                    }
                }
            }
        }
    }
    cout << ans.size() << ' ' << max_len << '\n';
    for (auto& el : ans) {
        cout << el.first + 1 << ' ';
    }
    cout << '\n';
}
