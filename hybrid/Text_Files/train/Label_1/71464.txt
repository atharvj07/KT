import sys

def isprime(n):
    if n < 2:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

for _ in range(int(input())):
    n, e = map(int, input().split())
    a = list(map(int, input().split()))
    ans = 0
    for i in range(n):
        p = 1
        for j in range(i, n, e):
            p *= a[j]
            if not isprime(p):
                break
            ans += 1
    print(ans)
