n = int(input())
last_time = 0
max_queue = 0
current_queue = 0

for _ in range(n):
    t, c = map(int, input().split())
    current_queue = max(0, current_queue - (t - last_time))
    current_queue += c
    max_queue = max(max_queue, current_queue)
    last_time = t

last_time += current_queue
print(last_time, max_queue)
