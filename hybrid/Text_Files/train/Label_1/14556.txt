#include <cstring>
#include <vector>
#include <iostream>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    if (n <= 2 || k > n / 3) {
        cout << "-1" << endl;
    } else {
        vector<int> assign(n + 1);
        for (int i = 1; i <= k; i++) {
            int start = i + (k - i) % 3;
            for (int j = start; j <= n; j += k) {
                assign[j] = i;
            }
        }
        for (int i = 1; i <= n; i++) {
            cout << assign[i] << (i == n ? '\n' : ' ');
        }
    }
    return 0;
}
