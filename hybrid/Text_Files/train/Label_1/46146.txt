import java.util.*;

public class Main {
    private static int n, m, d;
    private static int[] affected;
    private static boolean[] visited;
    private static List<List<Integer>> adj;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        d = sc.nextInt();
        affected = new int[m];
        for (int i = 0; i < m; i++) {
            affected[i] = sc.nextInt();
        }
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj.get(a).add(b);
            adj.get(b).add(a);
        }
        visited = new boolean[n + 1];
        int count = 0;
        for (int a : affected) {
            if (!visited[a]) {
                if (dfs(a, -1, 0)) {
                    count++;
                }
            }
        }
        System.out.println(count);
    }

    private static boolean dfs(int a, int p, int dist) {
        visited[a] = true;
        if (dist > d) {
            return false;
        }
        boolean ret = false;
        for (int b : adj.get(a)) {
            if (b != p) {
                ret |= dfs(b, a, dist + 1);
            }
        }
        return ret;
    }
}
