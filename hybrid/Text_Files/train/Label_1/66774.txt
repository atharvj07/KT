import math
import sys
import random

def satisfied(p, c):
    for i in range(len(p)):
        for j in range(i+1, len(p)):
            if abs(p[i][0] - p[j][0]) + abs(p[i][1] - p[j][1]) < abs(c[i] - c[j]):
                return False
    return True

def minSize(c):
    l, r = 1, 10000
    while l < r:
        m = (l+r) // 2
        p = [[random.randint(1, m), random.randint(1, m)] for _ in range(len(c))]
        if satisfied(p, c):
            r = m
        else:
            l = m + 1
    return l

n = int(input())
c = list(range(1, n+1))
m = minSize(c)
print(m)
for i in range(n):
    print("{} {}".format(*[random.randint(1, m) for _ in range(2)]))
