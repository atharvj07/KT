#include<iostream>
#include<cstdio>
using namespace std;
int main()
{
    int n, m, i, j, mn = 1<<30, mx = -1<<30, ans = 0;
    cin>>n>>m;
    int a[n], b[m];
    for(i=0; i<n; i++)
    {
        cin>>a[i];
        mn = min(mn, a[i]);
    }
    for(i=0; i<m; i++)
    {
        cin>>b[i];
        mx = max(mx, b[i]);
    }
    if(mx >= mn)
        cout<<0<<endl;
    else
        cout<<mn-mx<<endl;
}
