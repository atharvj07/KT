#include <iostream>
#include <string>

int main() {
	int t; std::cin >> t;
	while (t--) {
		int n; std::cin >> n;
		std::string s; std::cin >> s;
		bool flag = true;
		for (int i = 0; i < n; i += 3) {
			if (s[i] != 'T') flag = false;
			if (s[i + 1] != 'M') flag = false;
			if (s[i + 2] != 'T') flag = false;
		}
		std::cout << (flag ? "YES" : "NO") << '\n';
	}
	return 0;
}
