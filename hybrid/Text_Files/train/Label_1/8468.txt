#include <iostream>
#include <vector>
#include <string>
#include <sstream>

using namespace std;

int main() {
  string s;
  getline(cin, s);

  vector<string> a, b;
  bool has_num = false, has_non_num = false;
  stringstream ss(s);
  string word;
  while (getline(ss, word, ';')) {
    if (word == "") {
      continue;
    }
    bool is_num = true;
    for (char c : word) {
      if (!isdigit(c)) {
        is_num = false;
        break;
      }
    }
    if (is_num) {
      if (word[0] == '0' && word.size() > 1) {
        is_num = false;
      }
    }
    if (is_num) {
      a.push_back(word);
      has_num = true;
    } else {
      b.push_back(word);
      has_non_num = true;
    }
  }

  string res_a, res_b;
  if (!has_num) {
    res_a = "-";
  } else {
    for (int i = 0; i < a.size(); i++) {
      res_a += a[i];
      if (i < a.size() - 1) {
        res_a += ",";
      }
    }
  }
  if (!has_non_num) {
    res_b = "-";
  } else {
    for (int i = 0; i < b.size(); i++) {
      res_b += b[i];
      if (i < b.size() - 1) {
        res_b += ",";
      }
    }
  }

  cout << "\"" << res_a << "\"" << endl;
  cout << "\"" << res_b << "\"" << endl;

  return 0;
}
