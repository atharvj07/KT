import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        String weights = sc.nextLine();
        int m = sc.nextInt();
        if(weights.charAt(0) == '0' && m % 2 != 0) System.out.println("NO");
        else if(weights.substring(1).contains("11")) System.out.println("NO");
        else{
            StringBuilder sb = new StringBuilder("YES\n");
            int sum = 0;
            for(int i = 1; i <= 10; i++){
                if(weights.charAt(i - 1) == '1'){
                    while(sum + i <= m){
                        sb.append(i + " ");
                        sum += i;
                    }
                }
            }
            System.out.println(sb);
        }
    }
}
