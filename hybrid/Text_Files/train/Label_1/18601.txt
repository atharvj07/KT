#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  vector<pair<int, int>> queries(m);
  for (auto &q : queries) { cin >> q.first >> q.second; }
  vector<int> ans(m);
  for (int i = 0; i < m; ++i) {
    int sum = 0;
    for (int j = queries[i].first - 1; j <= queries[i].second - 1; ++j) {
      sum += a[j];
    }
    ans[i] = abs(sum) % 2;
  }
  for (auto x : ans) { cout << x << endl; }
  return 0;
}
