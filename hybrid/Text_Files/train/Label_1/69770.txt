import sys

EPS = 1e-9

def solve(w, h, x1, y1, x2, y2, ww, hh):
    is_in_horizontal = y1 - hh >= y2 or y2 >= h
    is_in_vertical   = x1 - ww >= x2 or x2 >= w

    if is_in_horizontal + is_in_vertical > 0:
        return -1

    if is_in_horizontal > 0:
        return max(0, y2 - h)
    else:
        return max(0, x2 - w)

for t in range(int(input())):
    w, h = map(int, input().split())
    x1, y1, x2, y2 = map(int, input().split())
    ww, hh = map(int, input().split())
    d = solve(w, h, x1, y1, x2, y2, ww, hh)
    print(d if d != -1 else '%.8f' % -1.0)
