import java.util.*;

public class Main {

    private static final long INF = 1 << 62;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long e = sc.nextLong();
        long t = sc.nextLong();
        long[] x = new long[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextLong();
        }
        long ans = INF;
        for (int i = 0; i < n; i++) {
            // Time to reach i-th bear
            long t1 = x[i];
            // Time to pick up coin from i-th bear
            long t2 = t1 + t;
            // Time to reach exit after picking up coin from i-th bear
            long t3 = t2 + e - x[i];
            // Update answer
            ans = Math.min(ans, t1 + t2 + t3);
        }
        System.out.println(ans);
    }
}
