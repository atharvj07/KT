import java.util.Scanner;
import java.math.BigInteger;
import java.math.BigDecimal;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    long n = input.nextLong();
    long[] q = new long[(int)n];
    for (int i = 0; i < n; i++) q[i] = input.nextLong();
    long ans = 1;
    for (int i = 0; i < n; i++) ans = (ans * (q[i] - 1)) % 1000000007L;
    BigInteger prime = new BigInteger("1000000007");
    BigInteger res = new BigInteger(ans + "");
    res = res.modPow(new BigInteger("2"), prime);
    System.out.println(res);
  }
}
