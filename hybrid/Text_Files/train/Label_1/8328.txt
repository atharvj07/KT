#include <iostream>

using namespace std;

int main() {
  int n, k, a, b;
  cin >> n >> k >> a >> b;

  if (abs(a - b) > 1) {
    cout << "NO" << endl;
    return 0;
  }

  if (a == b) {
    for (int i = 0; i < n; i++) {
      if (i % (k + 1) < k) {
        cout << "G";
      } else {
        cout << "B";
      }
    }
  } else if (a > b) {
    for (int i = 0; i < n; i++) {
      if (i % (k + 1) == k) {
        cout << "B";
      } else {
        cout << "G";
      }
    }
  } else {  // a < b
    for (int i = 0; i < n; i++) {
      if (i % (k + 1) == 0) {
        cout << "G";
      } else {
        cout << "B";
      }
    }
  }

  cout << endl;

  return 0;
}
