import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int x1[] = new int[n], y1[] = new int[n], z1[] = new int[n];
        int x2[] = new int[n], y2[] = new int[n], z2[] = new int[n];
        for (int i = 0; i < n; i++) {
            x1[i] = sc.nextInt();
            y1[i] = sc.nextInt();
            z1[i] = sc.nextInt();
            x2[i] = sc.nextInt();
            y2[i] = sc.nextInt();
            z2[i] = sc.nextInt();
        }
        int minX = x1[0], minY = y1[0], minZ = z1[0];
        int maxX = x2[0], maxY = y2[0], maxZ = z2[0];
        for (int i = 1; i < n; i++) {
            minX = Math.min(minX, x1[i]);
            minY = Math.min(minY, y1[i]);
            minZ = Math.min(minZ, z1[i]);
            maxX = Math.max(maxX, x2[i]);
            maxY = Math.max(maxY, y2[i]);
            maxZ = Math.max(maxZ, z2[i]);
        }
        int cnt[][][] = new int[maxX - minX + 1][maxY - minY + 1][maxZ - minZ + 1];
        for (int i = 0; i < n; i++) {
            for (int x = x1[i] - minX; x <= x2[i] - minX; x++) {
                for (int y = y1[i] - minY; y <= y2[i] - minY; y++) {
                    for (int z = z1[i] - minZ; z <= z2[i] - minZ; z++) {
                        cnt[x][y][z]++;
                    }
                }
            }
        }
        long ans = 0;
        for (int x = 0; x < maxX - minX + 1; x++) {
            for (int y = 0; y < maxY - minY + 1; y++) {
                for (int z = 0; z < maxZ - minZ + 1; z++) {
                    if (cnt[x][y][z] >= k) {
                        ans++;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
