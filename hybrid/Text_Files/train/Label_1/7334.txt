#include <iostream>
using namespace std;
int main() {
  int a, b, ans = 0;
  cin >> a >> b;
  for (int i = a; i <= b; i++) {
    int num = i, cnt = 0;
    while (num > 0) {
      cnt += num % 10 == 1 || num % 10 == 4 ? 2 : num % 10 == 2 || num % 10 == 3
                                              || num % 10 == 5 || num % 10 == 7 ? 5 : 3;
      num /= 10;
    }
    ans += cnt;
  }
  cout << ans << '\n';
  return 0;
}
