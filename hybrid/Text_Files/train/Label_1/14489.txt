#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin>>s;
    int n=s.size();
    int l=0,r=n-1;
    int pref[n],suff[n];
    pref[0]=(s[0]=='a');
    for(int i=1;i<n;i++){
        pref[i]=pref[i-1]+(s[i]=='a');
    }
    suff[n-1]=(s[n-1]=='a');
    for(int i=n-2;i>=0;i--){
        suff[i]=suff[i+1]+(s[i]=='a');
    }
    int ans=0;
    for(int i=0;i<n;i++){
        for(int j=i;j<n;j++){
            int cnt;
            if(i!=0){
                cnt=suff[i]-1;
            }
            else{
                cnt=suff[i];
            }
            if(j!=n-1){
                cnt+=pref[j+1];
            }
            else{
                cnt+=pref[j];
            }
            if(cnt>(j-i+1)/2){
                ans=max(ans,(j-i+1));
            }
        }
    }
    cout<<ans<<endl;
    return 0;
}
