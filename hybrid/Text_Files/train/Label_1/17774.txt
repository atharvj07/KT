#include <bits/stdc++.h>
using namespace std;

int main() {
    long long A, B, n, l, t, m;
    cin >> A >> B >> n;
    while (n--) {
        cin >> l >> t >> m;
        double x = (sqrt((1LL * t * m * B - 1LL * A) * 8 + 4LL * t * m * A) - 1LL * B) /
                   (2LL * t * m);
        if (x > 0LL && x - floor(x) == 0LL) {
            x += l - 1;
            if (x > 0LL)
                cout << (long long)x << "\n";
            else
                cout << "-1\n";
        } else
            cout << "-1\n";
    }
}
