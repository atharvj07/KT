#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> y1(n), y2(m);
    for (int i = 0; i < n; ++i) {
        cin >> y1[i];
    }
    for (int i = 0; i < m; ++i) {
        cin >> y2[i];
    }
    sort(y1.begin(), y1.end());
    sort(y2.begin(), y2.end());
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        auto it = lower_bound(y2.begin(), y2.end(), y1[i]);
        if (it != y2.end()) {
            ans++;
            y2.erase(it);
        }
    }
    cout << ans << endl;
    return 0;
}
