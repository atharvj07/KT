import java.io.*;
import java.math.BigInteger;

public class Main {
    static int mod = (int) (1e9 + 7);

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int[] in = getInts(br.readLine());
        int n = in[0], m = in[1];
        BigInteger ans = BigInteger.valueOf(2).pow(BigInteger.valueOf(n).multiply(BigInteger.valueOf(m)).intValue()).mod(BigInteger.valueOf(mod));
        System.out.println(ans);
    }

    private static int[] getInts(String line) {
        String[] f = line.split(" ");
        int[] ret = new int[f.length];
        for (int i = 0; i < f.length; i++) {
            ret[i] = Integer.parseInt(f[i]);
        }
        return ret;
    }
}
