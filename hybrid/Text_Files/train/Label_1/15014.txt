#include <algorithm>
#include <cstdio>
#include <vector>

int main() {
  long long n, m, ans = 1LL << 60;
  int sx, sy, fx, fy;
  std::vector<std::pair<int, int>> loc;
  scanf("%lld%lld", &n, &m);
  scanf("%d%d%d%d", &sx, &sy, &fx, &fy);
  for (int i = 0; i < m; ++i) {
    int x, y;
    scanf("%d%d", &x, &y);
    if (x == sx) ans = std::min(ans, (long long)std::abs(sy - y) +
                                         std::abs(fy - y) + 1);
    if (x == fx) ans = std::min(ans, (long long)std::abs(sy - y) +
                                         std::abs(fy - y) + 1);
    if (y == sx) ans = std::min(ans, (long long)std::abs(sy - x) +
                                         std::abs(fy - x) + 1);
    if (y == fx) ans = std::min(ans, (long long)std::abs(sy - x) +
                                         std::abs(fy - x) + 1);
    loc.push_back({x, y});
  }
  for (auto &e : loc) {
    ans = std::min(ans, (long long)std::abs(sx - e.first) + std::abs(sy - e.second) +
                              std::abs(fx - e.first) + std::abs(fy - e.second));
  }
  std::printf("%lld\n", ans);
  return 0;
}
