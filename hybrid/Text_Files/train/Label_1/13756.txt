#include<bits/stdc++.h>

using namespace std;

int main()
{
    int l[4],mi=INT_MAX,ma=0,ind=-1;
    char c;
    for(int i=0;i<4;i++)
    {
        cin>>c;
        string s;
        getline(cin,s);
        l[i]=s.size();
        if(s.size()<mi)
        {
            mi=s.size();
            ind=i;
        }
        if(s.size()>ma)
        {
            ma=s.size();
            ind=i;
        }
    }
    if(ma>2*mi||mi>2*ma)
        cout<<(char)(ind+'A');
    else
        cout<<"C";
}
