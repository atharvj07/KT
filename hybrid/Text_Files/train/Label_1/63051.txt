import sys
input = sys.stdin.readline

nqs = []
for i in range(int(input())):
    n = int(input())
    a = input()
    b = input()
    nqs.append([n, a, b])

for n, a, b in nqs:
    if (a[0] == '3' or a[0] == '4') and (b[0] == '3' or b[0] == '4'):
        print('NO')
        continue
    if a[0] == '6':
        a = '1' + a[1:]
    if b[0] == '6':
        b = '1' + b[1:]
    if a[n - 1] == '5':
        a = a[:n - 1] + '2'
    if b[n - 1] == '5':
        b = b[:n - 1] + '2'
    if a != b:
        print('NO')
    else:
        print('YES')
