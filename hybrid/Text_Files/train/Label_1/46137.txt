import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int H = sc.nextInt();
        int W = sc.nextInt();
        char[][] image = new char[H][W];
        for (int i = 0; i < H; i++) {
            String line = sc.next();
            for (int j = 0; j < W; j++) {
                image[i][j] = line.charAt(j);
            }
        }

        for (int i = 0; i < 2 * H; i++) {
            for (int j = 0; j < W; j++) {
                System.out.print(image[i / 2][j]);
            }
            System.out.println();
        }
    }
}
