from sys import stdin, stdout
t=int(stdin.readline().strip())
for _ in range(t):
    n=int(stdin.readline().strip())
    print(n**2//4)
    print(' '.join([str(i+1) for i in range(n)]))
    qs=n**2//4
    print(qs)
    for i in range(n-1,0,-1):
        for j in range(i):
            qs-=1
            print(i+1,j+1)
            if qs==0: break
        if qs==0: break
