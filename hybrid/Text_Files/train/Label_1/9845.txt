#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define mp make_pair
#define forn(i, n) for (int i = 0; i < n; i++)
#define nrick(i, n) for (int i = n - 1; i >= 0; i--)
#define forr(i, l, r) for (int i = l; i <= r; i++)
#define nranc(a, b) ((rand() % ((b) - (a))) + (a))
#define maxn 102
#define mod 1000000007
typedef pair<int, int> pii;
typedef vector<int> vi;
typedef vector<pii> vpi;
typedef vector<vi> vvi;
int n, m, k;
int c[maxn], p[maxn][maxn];
int dp[maxn][maxn][maxn];

int f(int x, int y, int clr) {
    if (y == k) {
        return 0;
    }
    if (x == n + 1) {
        return INT_MAX;
    }
    if (dp[x][y][clr] != -1) {
        return dp[x][y][clr];
    }
    int ret = INT_MAX;
    if (c[x]) {
        ret = min(ret, f(x + 1, y, clr));
    } else {
        if (clr == c[x]) {
            ret = min(ret, f(x + 1, y, clr));
        } else {
            ret = min(ret, f(x, y + 1, c[x]));
        }
        forn(j, m) {
            ret = min(ret, p[x][j] + f(x + 1, y, j + 1));
        }
    }
    return dp[x][y][clr] = ret;
}

int main() {
    cin >> n >> m >> k;
    forn(i, n) cin >> c[i + 1];
    forn(i, n) {
        forn(j, m) { cin >> p[i + 1][j]; }
    }
    memset(dp, -1, sizeof(dp));
    int ans = f(1, 0, 0);
    if (ans == INT_MAX) {
        cout << -1 << endl;
        return 0;
    }
    cout << ans << endl;
}
