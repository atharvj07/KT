import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] a = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        int[][] dpH = new int[n][n];
        int[][] dpV = new int[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dpH[i], Integer.MIN_VALUE);
            Arrays.fill(dpV[i], Integer.MIN_VALUE);
        }
        int max = Integer.MIN_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                dpH[i][j] = a[i][j];
                dpV[i][j] = a[i][j];
            }
        }
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < n; j++) {
                dpH[i][j] = Math.max(a[i][j], dpH[i - 1][j] + a[i][j]);
            }
        }
        for (int j = 1; j < n; j++) {
            for (int i = 0; i < n; i++) {
                dpV[i][j] = Math.max(a[i][j], dpV[i][j - 1] + a[i][j]);
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                max = Math.max(max, dpH[i][j]);
                max = Math.max(max, dpV[i][j]);
            }
        }
        System.out.println(max);
    }
}
