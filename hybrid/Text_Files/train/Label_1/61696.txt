import sys

lines = sys.stdin.readlines()
n = int(lines[0])

dic = {}
for card in lines[1].strip().split():
    color, value = card
    dic[(color, value)] = dic.get((color, value), 0) + 1

color_need = list()
value_need = list()

for color, value in dic:
    if dic[(color, value)] > 1:
        color_need.append(color)
        value_need.append(value)

print(max(len(set(color_need)), len(set(value_need))))
