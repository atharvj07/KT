#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> p(n);
    for (int i = 0; i < n; i++) {
        cin >> p[i];
    }
    int ans_id = 0, ans = n * (n + 1) / 2;
    for (int id = 0; id < n; id++) {
        int cur = 0;
        for (int i = 0; i < n; i++) {
            cur += abs(p[i] - (i + id + 1));
        }
        if (cur < ans) {
            ans = cur;
            ans_id = id;
        }
    }
    cout << ans << ' ' << ans_id << endl;
    return 0;
}
