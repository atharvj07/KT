#include <bits/stdc++.h>
using namespace std;

const int N = 1e3 + 10, INF = 1e8;
int n, l, r, x, dp[N][N][2];
vector<int> a;

int solve(int i, int sum, bool first) {
  if (i > n) {
    if (sum >= l && sum <= r) return 1;
    return 0;
  }
  int &res = dp[i][sum][first];
  if (~res) return res;
  res = 0;
  if (first)
    res = solve(i + 1, sum, first);
  else
    res = solve(i + 1, sum, first) + solve(i + 1, sum + a[i], false);
  return res;
}

int main() {
  cin >> n >> l >> r >> x;
  a.resize(n);
  for (int &x : a) cin >> x;
  sort(a.begin(), a.end());
  memset(dp, -1, sizeof dp);
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (i > 0 && (a[i] - a[i - 1] < x)) continue;
    ans += solve(i + 1, a[i], true);
  }
  cout << ans << "\n";
  return 0;
}
