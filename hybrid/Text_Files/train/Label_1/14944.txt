#include <vector>
#include <iostream>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int64_t> a(n), delta(n + 1), ans(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < m; i++) {
    int l, r, d;
    cin >> l >> r >> d;
    delta[l - 1] += d;
    delta[r] -= d;
  }

  ans[0] = delta[0];
  for (int i = 1; i < n; i++) { ans[i] = ans[i - 1] + delta[i]; }
  for (int i = 0; i < n; i++) { ans[i] += a[i]; }

  for (int i = 0; i < k; i++) {
    int x, y;
    cin >> x >> y;
    cout << ans[y] - (x - 1 >= 0 ? ans[x - 1] : 0) << ' ';
  }
  cout << '\n';
}
