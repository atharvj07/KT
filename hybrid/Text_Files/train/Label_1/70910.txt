t = int(input().strip())
for _ in range(t):
    a = input().strip()
    # Let's say:
    # A transforms to '(' and C transforms to ')', then B must be either '(' or ')'
    # but if A and C are both '(' or both ')', then it's impossible to form a regular bracket sequence.
    # So, we can use this logic to simplify the problem to checking the positions of each character.
    if a.count('A') + a.count('C') == len(a) or (a[0] == 'B' and a[-1] == 'B') or (a[0] != 'B' and a[-1] != 'B' and a[0] == a[-1]):
        print("NO")
    else:
        print("YES")
