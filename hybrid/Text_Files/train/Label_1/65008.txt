import sys
n,m,k = map(int,input().split())
p = list(map(int,input().split()))
s = list(map(int,input().split()))
c = list(map(int,input().split()))
d = {}
for i in range(n):
    if s[i] not in d: d[s[i]] = []
    d[s[i]].append((p[i],i))
e = []
for i in d:
    d[i].sort(reverse=True)
    e.append(d[i])
e.sort(reverse=True)
j = 0
ans = 0
for i in e:
    if len(i)>1:
        if j+2<=k:
            ans += 1
            j += 2
        else:
            j += 1
    else:
        j += 1
    if j==k: break;
if j<k: ans += k-j
print(ans)
