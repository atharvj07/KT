def solve():
    t = int(input())
    for _ in range(t):
        n, m = [int(x) for x in input().split()]
        grid = [list(input()) for _ in range(n)]
        if not valid(grid, n, m):
            print("NO")
            continue
        print("YES")
        for i in range(n):
            for j in range(m):
                if grid[i][j] == ".":
                    if (i == 0 or grid[i - 1][j] == "R") and (i == n - 1 or grid[i + 1][j] == "W") and (j == 0 or grid[i][j - 1] == "R") and (j == m - 1 or grid[i][j + 1] == "W"):
                        grid[i][j] = "R"
                    else:
                        grid[i][j] = "W"
        for row in grid:
            print("".join(row))

def valid(grid, n, m):
    for i in range(n):
        for j in range(m):
            if grid[i][j] == ".":
                cnt_r = 0
                cnt_w = 0
                if i != 0 and grid[i - 1][j] == "R":
                    cnt_r += 1
                if i != n - 1 and grid[i + 1][j] == "R":
                    cnt_r += 1
                if j != 0 and grid[i][j - 1] == "R":
                    cnt_r += 1
                if j != m - 1 and grid[i][j + 1] == "R":
                    cnt_r += 1
                if i != 0 and grid[i - 1][j] == "W":
                    cnt_w += 1
                if i != n - 1 and grid[i + 1][j] == "W":
                    cnt_w += 1
                if j != 0 and grid[i][j - 1] == "W":
                    cnt_w += 1
                if j != m - 1 and grid[i][j + 1] == "W":
                    cnt_w += 1
                if cnt_r > 0 and cnt_w > 0:
                    return False
    return True

solve()
