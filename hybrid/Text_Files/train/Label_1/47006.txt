import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            int x = input.nextInt();
            int y = input.nextInt();
            int z = input.nextInt();
            if (x == y && y == z) {
                System.out.println("YES");
                System.out.println(x + " " + x + " " + x);
            } else if (x == y || x == z) {
                System.out.println("YES");
                System.out.println(x + " " + y + " " + z);
            } else if (y == z) {
                System.out.println("YES");
                System.out.println(x + " " + y + " " + z);
            } else {
                System.out.println("NO");
            }
        }
    }
}
