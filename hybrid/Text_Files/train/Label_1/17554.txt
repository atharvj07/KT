#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
typedef long long ll;
int n, m;
ll dp[100005][2][2][2], pow2[100005];

ll get(int id, int l0, int l1, int r0, int r1) {
  if(id == m) {
    if(l0 + r0 <= 1 && l1 + r1 <= 1) return 1;
    return 0;
  }
  if(~dp[id][l0][l1][r0]) return dp[id][l0][l1][r0];
  ll& ret = dp[id][l0][l1][r0] = 0;
  if(r1 == 1) ret = get(id + 1, l0, l1, l0, r1);
  if(r1 == 0) ret = (ret + get(id + 1, l0, l1, l0, 1)) % mod;
  if(l1 == 0) ret = (ret + get(id + 1, l0, 1, r0, r1)) % mod;
  if(l1 == 1) ret = (ret + get(id + 1, l0, 0, r0, r1)) % mod;
  ret = (ret + get(id + 1, l0, 1, r0, 0)) % mod;
  ret = (ret + get(id + 1, l0, 0, r0, 1)) % mod;
  return ret;
}

int main() {
  memset(dp, -1, sizeof dp);
  cin >> n >> m, pow2[0] = 1;
  for(int i = 1; i <= m; i++) pow2[i] = (2 * pow2[i - 1]) % mod;
  cout << get(0, 0, 0, 0, 0) << endl;
  return 0;
}
