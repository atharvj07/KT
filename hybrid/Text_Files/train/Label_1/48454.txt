import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] teams = new int[k];
        for (int i = 0; i < k; i++) {
            teams[i] = scanner.nextInt();
        }
        int[][] dp = new int[n][(1 << k)];
        for (int i = 0; i < (1 << k); i++) {
            dp[0][i] = Integer.bitCount(i);
        }
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < (1 << k); j++) {
                for (int l = 0; l < (1 << k); l++) {
                    if ((j & l) == 0) {
                        dp[i][j | l] = Math.max(dp[i][j | l], dp[i - 1][j] + dp[i - 1][l]);
                    }
                }
            }
        }
        int maxGames = 0;
        for (int i = 0; i < (1 << k); i++) {
            maxGames = Math.max(maxGames, dp[n - 1][i]);
        }
        System.out.println(maxGames);
    }
}
