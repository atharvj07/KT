#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> ans(k, n / k);
    n %= k;
    for (int i = 0; i < n; ++i) {
      ans[i]++;
    }
    for (const int& a : ans) {
      cout << a << ' ';
    }
    cout << '\n';
  }
  return 0;
}
