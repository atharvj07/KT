#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.length();
        int ans = 0;
        for (int i = 0; i < n - 2; i++) {
            if (s[i] == s[i + 1] && s[i + 1] == s[i + 2]) {
                ans++;
                s[i + 1] = '0' + '1' - s[i + 1];
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
