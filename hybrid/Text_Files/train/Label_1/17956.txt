#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

typedef pair<int,int> pi;
#define vi(x) vector<x>

int main() {
    int n,k,m,a;
    cin>>n>>k>>m>>a;
    vi(pi) v(n);
    for(int i=0; i<n; i++) {
        v[i].first=0;
        v[i].second=i+1;
    }
    for(int i=0; i<a; i++) {
        int x;
        cin>>x;
        x--;
        v[x].first++;
    }
    sort(v.rbegin(), v.rend());
    for(int i=0; i<n; i++) {
        if(i+1<=k) cout<<1<<' ';
        else if(v[i].first) cout<<2<<' ';
        else cout<<3<<' ';
    } cout<<endl;
    return 0;
}
