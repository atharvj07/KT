#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 9;
typedef long long ll;
ll fastpow(ll a, ll b) {
    ll ans = 1;
    while (b) {
        if (b & 1) ans = ans * a % mod;
        b >>= 1;
        a = a * a % mod;
    }
    return ans;
}
int main() {
    ll n, a, b, k;
    cin >> n >> a >> b >> k;
    string s;
    cin >> s;
    int ans = 0;
    for (int i = 0; i < min(n + 1, (ll)s.size()); i++) {
        if (s[i] == '+') ans = (ans + fastpow(a, n - i) * fastpow(b, i) % mod) % mod;
        else ans = (ans - fastpow(a, n - i) * fastpow(b, i) % mod + mod) % mod;
    }
    cout << ans << endl;
    return 0;
}
