#include <bits/stdc++.h>
using namespace std;
typedef long long LL;

const int maxN = 2e5 + 10;
const LL mod = 1e9 + 7;
const int oo = 0x3f3f3f3f;

int n, k;
pair<LL, LL> p[maxN];
LL ans[maxN];

void solve() {
  cin >> n >> k;
  for (int i = 0; i < n; ++i) cin >> p[i].first;
  for (int i = 0; i < n; ++i) cin >> p[i].second;

  sort(p, p + n);

  int l = 0, r = 0;
  ans[0] = 0;
  for (int i = 0; i < n; ++i) {
    LL start = p[r].first + k;
    while (l <= r && p[l].first < start) ++l;
    if (l <= r) ans[i + 1] = ans[l - 1] + (r - l + 1);
    else ans[i + 1] = (i + 1);
    if (p[i].first <= start) ++r;
  }

  cout << ans[n] << '\n';
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  int t;
  cin >> t;
  while (t--) solve();

  return 0;
}
