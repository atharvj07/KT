n = int(input())
a = list(map(int, input().split()))

# Prefix XOR array
prefix_xor = [0] * (n+1)
for i in range(1, n+1):
    prefix_xor[i] = prefix_xor[i-1] ^ a[i-1]

# Counting funny pairs
funny_pairs = 0
xor_map = {0: [1, 0]}  # Initial XOR value and its counts for even and odd indices
for i in range(1, n+1):
    parity = i % 2
    if prefix_xor[i] in xor_map:
        funny_pairs += xor_map[prefix_xor[i]][1 - parity]
        xor_map[prefix_xor[i]][parity] += 1
    else:
        xor_map[prefix_xor[i]] = [0, 0]
        xor_map[prefix_xor[i]][parity] = 1

print(funny_pairs)
