import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    static int m, d;
    static List<int[]>[] subspaces;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] line = in.readLine().split(" ");
        m = Integer.parseInt(line[0]);
        d = Integer.parseInt(line[1]);
        subspaces = new List[m];
        for (int i = 0; i < m; i++) {
            line = in.readLine().split(" ");
            int n = Integer.parseInt(line[0]);
            subspaces[i] = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                line = in.readLine().split(" ");
                int[] v = new int[d];
                for (int k = 0; k < d; k++) {
                    v[k] = Integer.parseInt(line[k]);
                }
                subspaces[i].add(v);
            }
        }
        Map<String, Integer> map = new HashMap<>();
        int group = 1;
        for (int i = 0; i < m; i++) {
            String repr = Arrays.toString(subspaces[i].toArray());
            if (!map.containsKey(repr)) {
                map.put(repr, group++);
            }
            System.out.print(map.get(repr) + " ");
        }
        System.out.println();
    }
}
