#include <iostream>
#include <vector>
using namespace std;
vector<int> v;

int main() {
    int n;
    cin >> n;
    v.resize(n + 1, 0);
    int sum = 0;
    for (int i = 1; i <= n; i++) {
        cin >> v[i];
        sum += v[i];
    }
    int ans = sum;
    for (int i = 1; i <= n; i++) {
        int sum1 = 0;
        int sum2 = 0;
        for (int j = 1; j < i; j++) {
            sum1 += (i - j) * v[j];
        }
        for (int j = i + 1; j <= n; j++) {
            sum2 += (j - i) * v[j];
        }
        ans = min(ans, sum1 + sum2);
    }
    cout << ans << endl;
    return 0;
}
