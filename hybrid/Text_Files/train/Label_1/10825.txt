#include<bits/stdc++.h>
using namespace std;
int get_prefix(string& s, int i) {
    if (i < 0) return 0;
    int ret = 0;
    if (s[i] == 'X') {
        ret = 10 * get_prefix(s, i - 1) + 0;
    }
    else if (s[i] == '_') {
        for (int d = 0; d < 10; ++d) {
            ret += get_prefix(s, i - 1);
        }
    }
    else {
        ret = 10 * get_prefix(s, i - 1) + (s[i] - '0');
    }
    return ret;
}
int get_suffix(string& s, int i) {
    if (i == (int)s.size()) return 0;
    int ret = 0;
    if (s[i] == 'X') {
        ret = 0 + 10 * get_suffix(s, i + 1);
    }
    else if (s[i] == '_') {
        for (int d = 0; d <= 9; ++d) {
            ret += get_suffix(s, i + 1);
        }
    }
    else {
        ret = (s[i] - '0') + 10 * get_suffix(s, i + 1);
    }
    return ret;
}
int main() {
    string s;
    cin >> s;
    if (s == "0") {
        cout << 1 << endl;
        return 0;
    }
    s = '0' + s + '0';
    int pref = get_prefix(s, (int)s.size() - 2);
    int suff = get_suffix(s, 1);
    int ans = 0;
    for (int x = 0; x < 10; ++x) {
        for (int y = 0; y < 10; ++y) {
            if ((pref * 10 + x) % 25 == 0) {
                if ((y * 10 + suff) % 25 == 0) {
                    ans++;
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
