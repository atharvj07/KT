n, m = map(int, input().split())
logs = [input().split() for _ in range(m)]

logged_in = set()
possible_leaders = set(range(1, n + 1))

for action, id in logs:
    id = int(id)
    if action == '+':
        if id not in logged_in:
            for person in possible_leaders.copy():
                if person not in logged_in:
                    possible_leaders.remove(person)
        logged_in.add(id)
    else:
        logged_in.remove(id)
        if id not in possible_leaders:
            for person in logged_in:
                possible_leaders.discard(person)

for action, id in reversed(logs):
    id = int(id)
    if action == '-':
        logged_in.add(id)
    else:
        if logged_in:
            possible_leaders.discard(id)
        logged_in.remove(id)

print(len(possible_leaders))
print(*sorted(possible_leaders))
