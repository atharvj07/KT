#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const ll MOD = 998244353;

int main() {
  ll n, m, l, r;
  cin >> n >> m >> l >> r;
  if (l > r - 1) {
    cout << 0 << endl;
    return 0;
  }
  ll ans = (r - l + 1) % MOD;
  ans = (ans * ((r - l + 1) % MOD)) % MOD;
  ans = (ans * ((r - l) % MOD)) % MOD;
  ans = (ans * n % MOD) * (m % MOD);
  cout << ans << endl;
  return 0;
}
