import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt();
        int[] f = new int[n];
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            f[i] = sc.nextInt();
            t[i] = sc.nextInt();
        }

        Arrays.sort(f);
        Arrays.sort(t);

        int ans = 0;
        int j = 0;
        for (int i = 0; i < n; i++) {
            if (j < n && f[j] <= t[i]) {
                ans += t[i] - f[j];
                j++;
            } else {
                ans = -1;
                break;
            }
        }

        System.out.println(ans);
    }
}
