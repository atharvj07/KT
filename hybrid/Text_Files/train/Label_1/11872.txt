#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    int a, b;
    cin >> a >> b;
    v[i] = {a, b};
  }
  int ans = 0;
  int mx = 0;
  for (int i = 0; i < n; i++) {
    int diff = abs(v[i].first - v[i].second);
    if (diff > mx) {
      mx = diff;
      ans = i;
    }
  }
  cout << ans << '\n';
  return 0;
}
