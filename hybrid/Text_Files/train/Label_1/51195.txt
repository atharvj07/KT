import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        int n = Integer.parseInt(tokenizer.nextToken());
        int m = Integer.parseInt(tokenizer.nextToken());
        int[][] graph = new int[n+1][n+1];
        List<Integer> edgesToCover = new ArrayList<>();

        for (int i = 0; i < m; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int u = Integer.parseInt(tokenizer.nextToken());
            int v = Integer.parseInt(tokenizer.nextToken());
            graph[u][v] = 1;
            graph[v][u] = 1;
            edgesToCover.add(u);
            edgesToCover.add(v);
        }


        List<Integer> pari = new ArrayList<>();
        List<Integer> arya = new ArrayList<>();
        List<Integer> independent = new ArrayList<>();
        boolean[] visited = new boolean[n+1];

        for (int edge : edgesToCover) {
            if (!visited[edge]) {
                pari.clear();
                arya.clear();
                independent.clear();
                if (!dfs(edge, graph, visited, pari, arya, independent)) {
                    System.out.println(-1);
                    return;
                }
                if (pari.size() >= arya.size()) {
                    System.out.println(pari.size());
                    for (int vertex : pari) {
                        System.out.print(vertex + " ");
                    }
                    System.out.println();
                    System.out.println(arya.size());
                    for (int vertex : arya) {
                        System.out.print(vertex + " ");
                    }
                } else {
                    System.out.println(arya.size());
                    for (int vertex : arya) {
                        System.out.print(vertex + " ");
                    }
                    System.out.println();
                    System.out.println(pari.size());
                    for (int vertex : pari) {
                        System.out.print(vertex + " ");
                    }
                }
                return;
            }
        }
    }

    private static boolean dfs(int vertex, int[][] graph, boolean[] visited, List<Integer> pari, List<Integer> arya, List<Integer> independent) {
        visited[vertex] = true;
        pari.add(vertex);
        boolean canCover = false;
        for (int i = 1; i < graph.length; i++) {
            if (graph[vertex][i] == 1 && !visited[i]) {
                if (dfs(i, graph, visited, pari, arya, independent)) {
                    canCover = true;
                }
            }
        }
        if (!canCover) {
            pari.remove(pari.size() - 1);
            arya.add(vertex);
            canCover = false;
            for (int i = 1; i < graph.length; i++) {
                if (graph[vertex][i] == 1 && !visited[i]) {
                    if (dfs(i, graph, visited, pari, arya, independent)) {
                        canCover = true;
                    }
                }
            }
            if (!canCover) {
                arya.remove(arya.size() - 1);
                independent.add(vertex);
                return false;
            }
        }
        return true;
    }
}
