n = int(input())
trees = []
for i in range(n):
    trees.append(tuple(map(int, input().split())))
trees.sort()
left = [trees[0][0] - trees[0][1]]
right = [trees[0][0] + trees[0][1]]
for i in range(1, n):
    if trees[i][0] - trees[i][1] > left[-1]:
        left.append(trees[i][0] - trees[i][1])
    if trees[i][0] + trees[i][1] < right[-1]:
        right.append(trees[i][0] + trees[i][1])
print(len(left) + len(right) - 1)
