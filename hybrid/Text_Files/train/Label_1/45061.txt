import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();

        int[][] display = new int[n][n];
        for (int i = 0; i < n; i++) {
            String line = in.next();
            for (int j = 0; j < n; j++) {
                display[i][j] = line.charAt(j) - '0';
            }
        }

        int[][] commands = new int[2][n * n];
        int commandIdx = 0;

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (display[i][j] == 1) {
                    commands[0][commandIdx] = i + 1;
                    commands[1][commandIdx] = j + 1;
                    commandIdx++;
                }
            }
        }

        System.out.println(commandIdx);
    }
}
