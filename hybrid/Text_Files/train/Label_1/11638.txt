#include <bits/stdc++.h>
using namespace std;

const int N = 2e5;

int n, h[N], w[N];

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < n; ++i) {
            cin >> h[i] >> w[i];
        }
        for (int i = 0; i < n; ++i) {
            int ans = -1;
            for (int j = 0; j < n; ++j) {
                if (i == j)
                    continue;
                if ((h[i] > h[j] && w[i] > w[j]) || (w[i] > h[j] && h[i] > w[j])) {
                    ans = j + 1;
                    break;
                }
            }
            cout << ans << " ";
        }
        cout << "\n";
    }
}
