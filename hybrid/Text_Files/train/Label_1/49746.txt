import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        List<Frog> frogs = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            line = reader.readLine().split(" ");
            frogs.add(new Frog(Integer.parseInt(line[0]), Integer.parseInt(line[1])));
        }

        List<Mosquito> mosquitoes = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            mosquitoes.add(new Mosquito(Integer.parseInt(line[0]), Integer.parseInt(line[1])));
        }

        Collections.sort(frogs);
        Collections.sort(mosquitoes);

        int[] eatenMosquitoes = new int[n];
        int[] tongueLengths = new int[n];
        for (Mosquito mosquito : mosquitoes) {
            int frogIndex = -1;
            for (int i = 0; i < n; i++) {
                if (frogs.get(i).canEat(mosquito)) {
                    frogIndex = i;
                    break;
                }
            }

            if (frogIndex != -1) {
                eatenMosquitoes[frogIndex]++;
                frogs.get(frogIndex).eat(mosquito);
                tongueLengths[frogIndex] = frogs.get(frogIndex).getTongueLength();
            }
        }

        for (int i = 0; i < n; i++) {
            System.out.println(eatenMosquitoes[i] + " " + tongueLengths[i]);
        }
    }

    private static class Frog implements Comparable<Frog> {
        private int position;
        private int tongueLength;

        public Frog(int position, int tongueLength) {
            this.position = position;
            this.tongueLength = tongueLength;
        }

        public boolean canEat(Mosquito mosquito) {
            return mosquito.getPosition() >= position && mosquito.getPosition() <= position + tongueLength;
        }

        public void eat(Mosquito mosquito) {
            tongueLength += mosquito.getSize();
        }

        public int getTongueLength() {
            return tongueLength;
        }

        @Override
        public int compareTo(Frog other) {
            return Integer.compare(position, other.position);
        }
    }

    private static class Mosquito implements Comparable<Mosquito> {
        private int position;
        private int size;

        public Mosquito(int position, int size) {
            this.position = position;
            this.size = size;
        }

        public int getPosition() {
            return position;
        }

        public int getSize() {
            return size;
        }

        @Override
        public int compareTo(Mosquito other) {
            return Integer.compare(position, other.position);
        }
    }
}
