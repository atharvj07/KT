#include <bits/stdc++.h>
using namespace std;
const int M = 105;
vector<int> g[M];
bool vis[M];
int col[M], c, n, a[M];

void dfs(int u){
  vis[u]=1;
  col[u]=c;
  for(int v:g[u]){
    if(!vis[v]) dfs(v);
  }
}
int main(){
  cin >> n;
  for(int i=1;i<=n;i++){
    cin >> a[i];
    g[a[i]].push_back(i);
  }
  for(int i=1;i<=n;i++){
    if(!vis[i]){
      dfs(i);
      c++;
    }
  }
  cout << c << endl;
}
