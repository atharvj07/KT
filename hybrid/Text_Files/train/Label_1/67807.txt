MOD = 10**9 + 7

def modinv(a, m=MOD):
    return pow(a, m-2, m)

def solve(n, k, a):
    if sorted(a) != a:
        return 0
    ones = a.count(1)
    zeros = n - ones
    dp = [[0 for _ in range(ones + 1)] for _ in range(zeros + 1)]
    dp[0][0] = 1
    for i in range(zeros + 1):
        for j in range(ones + 1):
            if i > 0:
                dp[i][j] = (dp[i][j] + dp[i-1][j] * i) % MOD
            if j > 0:
                dp[i][j] = (dp[i][j] + dp[i][j-1] * j) % MOD
    P = dp[zeros][ones]
    Q = pow(n * (n - 1) // 2, k, MOD)
    return (P * modinv(Q)) % MOD

n, k = map(int, input().split())
a = list(map(int, input().split()))
print(solve(n, k, a))
