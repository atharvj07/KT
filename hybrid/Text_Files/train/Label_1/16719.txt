#include<bits/stdc++.h>
using namespace std;
char grid[10][10];
bool vis[10][10][10][10];
bool valid(int x,int y)
{
    return (x>=0 && x<8) && (y>=0 && y<8) && (grid[x][y]!='#');
}
struct point
{
    int x1,y1,x2,y2;
};
bool reachable(point p)
{
    int dx[]={-2,-2,2,2};
    int dy[]={-2,2,-2,2};
    for(int k=0;k<4;k++)
    {
        if(valid(p.x1+dx[k],p.y1+dy[k]) && valid(p.x2+dx[k],p.y2+dy[k]))
        {
            int x1=p.x1+dx[k];
            int y1=p.y1+dy[k];
            int x2=p.x2+dx[k];
            int y2=p.y2+dy[k];
            if(x1==x2 && y1==y2) return true;
            if(vis[x1][y1][x2][y2] || vis[x2][y2][x1][y1]) continue;
            vis[x1][y1][x2][y2]=vis[x2][y2][x1][y1]=1;
            point p1={x1,y1,x2,y2};
            if(reachable(p1)) return true;
        }
    }
    return false;
}
int main()
{
    int t;
    cin>>t;
    cin.ignore();
    while(t--)
    {
        memset(vis,false,sizeof(vis));
        for(int i=0;i<8;i++) cin>>grid[i];
        point p;
        for(int i=0;i<8;i++)
            for(int j=0;j<8;j++)
            {
                if(grid[i][j]=='K')
                {
                    if(p.x1==0)
                    {
                        p.x1=i;
                        p.y1=j;
                    }
                    else
                    {
                        p.x2=i;
                        p.y2=j;
                    }
                }
            }
        if(reachable(p)) cout<<"YES\n";
        else cout<<"NO\n";
        cin.ignore();
    }
    return 0;
}
