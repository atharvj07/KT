#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;

  while (t--) {
    string s;
    cin >> s;

    vector<int> counts(12);
    for (int i = 0; i < 12; i++) {
      counts[i] = s[i] == 'X' ? 1 : 0;
    }

    vector<pair<int, int>> pairs;
    for (int i = 1; i <= 12; i++) {
      if (12 % i == 0) {
        pairs.push_back({i, 12 / i});
      }
    }

    bool found = false;
    for (auto pair : pairs) {
      int a = pair.first;
      int b = pair.second;
      bool valid = true;
      for (int i = 0; i < a; i++) {
        int sum = 0;
        for (int j = 0; j < b; j++) {
          sum += counts[i * b + j];
        }
        if (sum == 0) {
          valid = false;
          break;
        }
      }

      if (valid) {
        if (!found) {
          cout << pairs.size() << " ";
          found = true;
        }
        cout << a << "x" << b << " ";
      }
    }

    if (!found) {
      cout << 0;
    }

    cout << endl;
  }

  return 0;
}
