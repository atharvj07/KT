import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextLong();
            b[i] = scanner.nextLong();
        }
        long lcm = b[0];
        for (long v : b) {
            lcm = lcm * v / gcd(lcm, v);
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += (lcm / b[i]) * a[i] % lcm;
        }
        System.out.println((sum + lcm - 1) / lcm);
    }

    static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
