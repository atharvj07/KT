def get_reversed(arr):
    rev = []
    for i in range (len(arr) - 1, -1, -1):
        rev.append(arr[i])

    return rev

def get_fingerprint(arr):
    finger = []

    for i in range(0, len(arr) - 1):
        finger.append(arr[i] + arr[i + 1])

    return sorted(finger)

def get_next_permutation(arr):
    n = len(arr)

    i = n - 1

    while(i > 0 and arr[i] <= arr[i - 1]):
        i -= 1

    if(i > 0):
        j = n - 1

        while (j >= i and arr[j] <= arr[i - 1]):
            j -= 1
        
        arr[i - 1], arr[j] = arr[j], arr[i - 1]
    
    arr[i:n] = get_reversed(arr[i:n])

    return arr

for _ in range(int(input())):
    n = int(input())
    arr = list(map(int, input().split()))

    if(get_fingerprint(arr) == get_fingerprint(get_next_permutation(arr))):
        print(*get_next_permutation(arr))
    else:
        print(*arr)
