n, k = map(int, input().split())
p = list(map(int, input().split()))
p.sort(reverse = True)

mx = 0
ans = 0
m = 998244353
suff = [0] * (n + 1)
for i in range(n - 1, -1, -1):
  suff[i] = max(suff[i + 1], p[i])

for i in range(0, n - k + 1):
  mx = max(mx, p[i] + suff[i + k])
ans = 1
for i in range(0, n - k + 1):
  if mx == p[i] + suff[i + k]:
    ans = (ans * (i + 1)) % m
print(mx, ans)
