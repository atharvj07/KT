#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e6 + 5;
int freq[MAXN];

int main() {
    int n, x;
    cin >> n >> x;
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        freq[a]++;
    }
    int suf[MAXN];
    suf[x + 1] = 0;
    for (int i = x; i >= 1; i--) {
        suf[i] = suf[i + 1] + freq[i];
    }
    long long ans = 0;
    for (int i = 1; i <= x; i++) {
        ans += 1ll * suf[i] * (suf[i] + 1) / 2;
    }
    cout << ans << endl;
    return 0;
}
