#include <iostream>
#include <vector>
 
using namespace std;
 
int main() {
    int n;
    cin >> n;
    vector<int> a(n + 1);
    int mx = 0;
    for (int i = 2; i <= n; i++) {
        if (a[i] == 0) {
            for (int j = i; j <= n; j += i) {
                if (a[j] == 0) {
                    a[j] = mx + 1;
                    mx++;
                }
            }
        }
    }
    for (int i = 2; i <= n; i++) {
        cout << a[i] << " ";
    }
    cout << endl;
    return 0;
}
