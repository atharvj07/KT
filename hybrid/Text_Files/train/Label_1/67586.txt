m, s = map(int, input().split())

if s == 0:
    if m == 1:
        print("0 0")
    else:
        print("-1 -1")
else:
    if s > 9*m:
        print("-1 -1")
    else:
        # For minimum number
        min_num = [0]*m
        sum_left = s - 1
        for i in range(m-1, 0, -1):
            if sum_left > 9:
                min_num[i] = 9
                sum_left -= 9
            else:
                min_num[i] = sum_left
                sum_left = 0
        min_num[0] = sum_left + 1
        min_str = ''.join(map(str, min_num))

        # For maximum number
        max_num = [0]*m
        sum_left = s
        for i in range(m):
            if sum_left > 9:
                max_num[i] = 9
                sum_left -= 9
            else:
                max_num[i] = sum_left
                sum_left = 0
        max_str = ''.join(map(str, max_num))

        print(f"{min_str} {max_str}")
