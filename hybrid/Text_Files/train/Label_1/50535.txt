import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int x = sc.nextInt();
            int y = sc.nextInt();
            String[] grid = new String[n];
            for (int j = 0; j < n; j++) {
                grid[j] = sc.next();
            }
            int cost = 0;
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < m; k++) {
                    if (grid[j].charAt(k) == '.') {
                        if (k + 1 < m && grid[j].charAt(k + 1) == '.') {
                            cost += y;
                            k++;
                        } else {
                            cost += x;
                        }
                    }
                }
            }
            System.out.println(cost);
        }
    }
}
