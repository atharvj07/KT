#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

long long pow10[200005]{1LL};
int main() {
    string s;
    cin >> s;
    int m;
    cin >> m;
    pow10[0] = 1LL;
    for (int i=1;i<=int(s.length());i++) {
        pow10[i] = pow10[i-1] * 10LL;
        pow10[i] %= m;
    }
    vector<int> hash(int(s.length())+1,0);
    for (int i=int(s.length())-1;i>=0;i--) {
        hash[i] = (hash[i+1] * 10LL + s[i] - '0') % m;
    }
    int ans = hash[0];
    for (int len=2;len<=int(s.length());len++) {
        for (int i=0;i+len-1<int(s.length());i++) {
            int curHash = (hash[i] + m - ((hash[i+len] - hash[i]) * pow10[i] % m) + m) % m;
            ans = min(ans, curHash);
        }
    }
    cout << ans << "\n";
}
