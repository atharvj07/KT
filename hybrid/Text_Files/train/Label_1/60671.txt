import sys

def solve(n, a):
    a.sort()
    ans = 0
    for i in range(n):
        if i + 1 < n and a[i + 1] % a[i] != 0:
            ans = a[i] + a[i + 1]
        if i + 2 < n and a[i + 2] % a[i] != 0 and a[i + 2] % a[i + 1] != 0:
            ans = a[i] + a[i + 1] + a[i + 2]
    return ans

q = int(sys.stdin.readline())
for _ in range(q):
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    print(solve(n, a))
