#include <bits/stdc++.h>

using namespace std;

const int INF = (int)1e9;

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  vector<int> c(n);
  for (int i = 0; i < n; ++i) {
    cin >> c[i];
  }
  vector<int> dp(1 << n, INF);
  dp[0] = 0;
  for (int mask = 0; mask < (1 << n); ++mask) {
    vector<int> cnt(2);
    for (int i = 0; i < n; ++i) {
      if ((mask >> i) & 1) {
        cnt[c[i]]++;
      }
    }
    if (cnt[0] == cnt[1] || cnt[0] + 1 == cnt[1] || cnt[1] + 1 == cnt[0]) {
      dp[mask] = min(dp[mask], dp[mask ^ ((1 << n) - 1 - mask)]);
    } else {
      continue;
    }
    for (int i = 0; i < n; ++i) {
      if (((mask >> i) & 1) == 0 && c[i] != 2) {
        dp[mask] = min(dp[mask], dp[mask | (1 << i)] + a);
      } else if (((mask >> i) & 1) == 0 && c[i] == 2) {
        dp[mask] = min(dp[mask], dp[mask | (1 << i)] + min(a, b));
      }
    }
  }
  if (dp[(1 << n) - 1] == INF) {
    cout << -1 << endl;
  } else {
    cout << dp[(1 << n) - 1] << endl;
  }
  return 0;
}
