#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<string> v(n);
    for (int i = 0; i < n; ++i) cin >> v[i];
    long long ans = 0;
    for (int mask = 0; mask < (1 << k); ++mask) {
        map<string, int> m;
        for (int i = 0; i < n; ++i) {
            string t;
            for (int j = 0; j < k; ++j) {
                t += (mask & (1 << j)) ? v[i][k - 1 - j] : v[i][j];
            }
            m[t]++;
        }
        for (auto &p : m) {
            ans += p.second * (p.second - 1) * (p.second - 2) / 6;
        }
    }
    cout << ans << endl;
    return 0;
}
