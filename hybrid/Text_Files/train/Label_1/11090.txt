#include <cstdio>
#include <vector>
#define MAX_N 200001
int p[MAX_N];
int main() {
    int n, k, a, m, x, cnt = 0;
    scanf("%d %d %d", &n, &k, &a);
    scanf("%d", &m);
    for (int i = 1; i <= m; i++) {
        scanf("%d", &x);
        if (x + a - 1 <= n && ++p[x] == 1) cnt++;
        if (cnt >= k) { printf("%d\n", i); return 0; }
    }
    printf("-1\n");
    return 0;
}
