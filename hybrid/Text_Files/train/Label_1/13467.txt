#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll dp[20],p[107],d[107],t[107],m[107],ans=0;
int main(){
    ll n,i,j,l,r,c=1e9,a=0,b=0,k;
    cin>>n;
    for(i=1;i<=n;i++)
        cin>>m[i]>>d[i]>>p[i]>>t[i];
    dp[1]=1;
    for(i=1;i<=n;i++){
        ans+=p[i];
        for(j=i-1;j>=1;j--){
            if(m[j]<m[i]){
                if(d[i]>=d[j]&&d[i]-d[j]>=t[j]){
                    if(dp[i]<dp[j]+p[i])
                        dp[i]=dp[j]+p[i];
                }
            }
            else if(m[j]==m[i]){
                if(d[i]>=d[j]&&d[i]-d[j]>=t[j]+1){
                    if(dp[i]<dp[j]+p[i])
                        dp[i]=dp[j]+p[i];
                }
            }
        }
    }
    for(i=1;i<=n;i++)
        ans=min(dp[i],ans);
    cout<<ans<<'\n';
}
