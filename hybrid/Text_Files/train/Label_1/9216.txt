#include <bits/stdc++.h>

using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<int> ans(n);
    for (int i = 1; i <= n; i++) {
        bool reversed = false;
        for (int j = 0; j < i / 2; j++) {
            if (s[j] > s[i - j - 1]) {
                reversed = true;
                break;
            } else if (s[j] < s[i - j - 1]) {
                break;
            }
        }
        ans[i - 1] = reversed;
    }
    for (int a : ans) {
        cout << a << " ";
    }
    return 0;
}
