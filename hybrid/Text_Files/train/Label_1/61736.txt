import sys

input = sys.stdin.readline

n = int(input())
g = [[] for i in range(n)]
for i in range(n - 1):
    a, b = map(int, input().split())
    a -= 1
    b -= 1
    g[a].append(b)
    g[b].append(a)

vis = [0] * n
lab = [0] * (n - 1)

def dfs(u, p, l):
    vis[u] = 1
    for v in g[u]:
        if vis[v] == 0:
            while l in lab:
                l += 1
            lab[p] = l
            dfs(v, p + 1, l + 1)

dfs(0, 0, 0)
for i in lab:
    print(i)
