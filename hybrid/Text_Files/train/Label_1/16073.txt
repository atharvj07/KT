#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n;
  int a[n + 2], b[n + 2];
  set<int> s;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  cin >> m;
  for (int i = 0; i < m; i++) { cin >> b[i]; }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (s.count(a[i] + b[j]) == 0) {
        cout << a[i] << " " << b[j] << endl;
        return 0;
      }
    }
  }
}
