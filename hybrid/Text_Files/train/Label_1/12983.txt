#include <bits/stdc++.h>
using namespace std;
int n, m, k, a[205][205], b[205][205], f[205][205], x[205];
vector<char> ans;
int dx[] = {0, 0, 1, -1};
int dy[] = {-1, 1, 0, 0};
int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= k; i++) {
    int x, y;
    cin >> x >> y;
    a[x][y] = i, b[x][y] = 1;
  }
  for (int i = 1; i <= k; i++) {
    int x, y;
    cin >> x >> y;
    f[x][y] = i, b[x][y] = -1;
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (!a[i][j]) continue;
      for (int d = 0; d < 4; d++) {
        int u = i + dx[d], v = j + dy[d];
        if (1 <= u && u <= n && 1 <= v && v <= m && b[u][v] != -1) {
          ans.push_back('L' + d);
          swap(a[i][j], a[u][v]);
          swap(b[i][j], b[u][v]);
          x[a[i][j]] ^= d;
          break;
        }
      }
    }
  }
  if ((int)ans.size() <= 2 * n * m) {
    cout << ans.size() << endl;
    for (char v : ans) cout << v;
    cout << endl;
  } else {
    cout << -1 << endl;
  }
  return 0;
}
