import queue
n,m,t=map(int,input().split())
g=[[] for _ in range(n+1)]
for _ in range(m):
	a,b,c=map(int,input().split())
	g[a].append((c,b))
q=queue.Queue()
q.put((0,1))
d=[1e9]*(n+1)
d[1]=0
r=[1]
while not q.empty():
	x,y=q.get()
	r.append(y)
	for w,z in g[y]:
		if x+w<=t and d[z]>x+w:
			d[z]=x+w
			q.put((x+w,z))
print(len(set(r)))
for x in set(r): print(x,end=' ')
