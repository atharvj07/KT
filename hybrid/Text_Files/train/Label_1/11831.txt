#include <bits/stdc++.h>

using namespace std;
using ll = long long;

#define endl "\n"

const int MAX_V = 1e5 + 5;

vector<int> adj[MAX_V];

int dis[MAX_V];

void dfs(int u, int p, int dist, int &d) {
    dis[u] = dist;
    d = max(d, dist);
    for (int v : adj[u]) {
        if (v != p) {
            dfs(v, u, dist + 1, d);
        }
    }
}

void solve() {
    int n, a, b, da, db;
    cin >> n >> a >> b >> da >> db;
    for (int i = 1; i <= n; i++) { adj[i].clear(); }
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    int mx = 0;
    dfs(a, -1, 0, mx);
    int ansA = mx;

    mx = 0;
    dfs(b, -1, 0, mx);
    int ansB = mx;

    string winner = (ansA <= ansB) ? "Alice" : "Bob";
    cout << winner << endl;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int t;
    cin >> t;
    while (t--) {
        solve();
    }
}
