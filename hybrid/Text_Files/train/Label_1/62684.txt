import sys

l, r = map(int, sys.stdin.readline().split())

if (r - l + 1) % 2 == 1:
    print("NO")
else:
    print("YES")
    arr = []
    i = l
    while i <= r:
        if i % 2 == 1:
            arr.append(i)
        i += 1
    i = r
    while i >= l:
        if i % 2 == 0:
            arr.append(i)
        i -= 1
    for i in range(0, len(arr), 2):
        print(arr[i], arr[i + 1])
