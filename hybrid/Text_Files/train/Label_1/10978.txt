#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> h(n + 1);
  for (int i = 1; i <= n; i++) cin >> h[i];
  vector<long long> mx(n + 1, 0);
  for (int i = n - 1; i >= 1; i--) mx[i] = max(mx[i + 1], h[i + 1]);
  for (int i = 1; i <= n; i++) cout << max(0LL, mx[i] - h[i]) << " ";
  cout << endl;
}
