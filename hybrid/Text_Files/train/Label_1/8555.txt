#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    sort(arr.begin(), arr.end());
    vector<int> ans;
    ans.push_back(0);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < ans.size(); j++) {
            int tmp = ans[j] ^ arr[i];
            if (tmp > arr[i]) {
                break;
            }
            if (find(ans.begin(), ans.end(), tmp) == ans.end()) {
                ans.push_back(tmp);
            }
        }
    }
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
