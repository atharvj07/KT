#include <iostream>
#include <string>
#include <vector>

using std::cin;
using std::cout;
using std::string;
using std::vector;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> house(n);
  for (int i = 0; i < n; i++) {
    cin >> house[i];
  }

  int seconds = 0;
  int floor = n - 1;
  int col = 0;
  int dir = 1;
  while (floor > 0 || col < m - 1) {
    if (house[floor][col] == '.') {
      if (floor == 0 && col == m - 1) {
        break;
      }
      if (house[floor - 1][col] == '.') {
        floor--;
      } else {
        col += dir;
        dir = -dir;
      }
    } else {
      col += dir;
      dir = -dir;
    }
    seconds++;
  }

  if (floor == 0 && col == m - 1) {
    cout << seconds << '\n';
  } else {
    cout << "Never\n";
  }

  return 0;
}
