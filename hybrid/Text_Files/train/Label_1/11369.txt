#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    vector<int> color(n + 1, -1);
    color[1] = 0;
    set<int> q;
    q.insert(1);
    int ans = 0;
    while (!q.empty()) {
        int u = *q.begin();
        q.erase(q.begin());
        for (int v : adj[u]) {
            if (color[v] == -1) {
                color[v] = 1 - color[u];
                q.insert(v);
            } else if (color[v] == color[u]) {
                ans++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
