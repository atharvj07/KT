#include <bits/stdc++.h>
using namespace std;
#define maxn 300500
#define ll long long
int fr[maxn], p, ctr = 0, n, g[maxn];
vector <int> ans;
int main() {
    cin >> n >> p;
    ll a[n + 5];
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        a[i] %= p;
        fr[a[i]]++;
        ctr += (fr[a[i]] >= 2);
    }
    for (int i = p - 1; i >= 0; i--) {
        fr[i] += fr[i + p];
        if (fr[i] % 3 == 1) g[ctr]++, ctr--;
        else if (fr[i] % 3 == 2) g[ctr + 1]++, ctr++;
    }
    ll sum = 0;
    for (int i = 0; i <= ctr; i++) {
        sum += g[i];
        if (sum % 3 == 2) ans.push_back(i);
    }
    cout << ans.size() << endl;
    for (int i : ans) {
        cout << i << endl;
    }
}
