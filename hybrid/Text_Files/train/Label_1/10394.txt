#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
int n, p, ca[maxn], cb[maxn], ca2[30], cb2[30];
int main() {
    string sa, sb, ans = "01234567890123456789";
    cin >> n >> p;
    cin >> sa >> sb;
    reverse(sa.begin(), sa.end());
    for (int i = 1; i <= n; ++i) {
        ca[i] = (sa[i - 1] != sb[i - 1]);
        cb[i] = (sa[i - 1] != sb[n - i]);
    }
    for (int i = 1; i <= n; ++i) {
        ca2[sa[i - 1] - 'a'] += ca[i];
        cb2[sb[i - 1] - 'a'] += cb[i];
    }
    int mn = n;
    for (int i = 0; i < 30; ++i) {
        mn = min(mn, ca2[i] + cb2[i] - min(ca2[i], cb2[i]));
    }
    cout << mn << endl;
    return 0;
}
