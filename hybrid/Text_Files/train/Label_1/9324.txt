#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<string> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<string> ans(n, string(m, '-'));
  for (int i = 0; i < n; i += 2)
    for (int j = 0; j < m; j += 2) if (a[i][j] == '.') ans[i][j] = (j / 2) % 2 ? 'W' : 'B';
  for (int i = 0; i < n; i += 2)
    for (int j = 1; j < m; j += 2) if (a[i][j] == '.') ans[i][j] = (j / 2) % 2 ? 'B' : 'W';
  for (int i = 1; i < n; i += 2)
    for (int j = 0; j < m; j += 2) if (a[i][j] == '.') ans[i][j] = (j / 2) % 2 ? 'B' : 'W';
  for (int i = 1; i < n; i += 2)
    for (int j = 1; j < m; j += 2) if (a[i][j] == '.') ans[i][j] = (j / 2) % 2 ? 'W' : 'B';
  for (auto &x : ans) cout << x << '\n';
}
