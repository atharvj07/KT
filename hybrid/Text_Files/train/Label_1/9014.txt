#include<iostream>
using namespace std;
long long getMex(string s){
	int n=s.length();
	long long ans=0;
	for(int i=0;i<n;i++){
		int bit=s[i]-'0';
		if(bit==1){
			ans+=1LL<<(n-1-i);
		}
		else{
			break;
		}
	}
	return ans;
}
string getMax(string s){
	int n=s.size(),i=0;
	long long mex=getMex(s);
	for(i=0;i<n;i++){
		int bit=s[i]-'0';
		if(bit){
			long long tmp=mex;
			tmp+=1LL<<(n-1-i);
			if(tmp>mex){
				s.erase(i,1);
				break;
			}
		}
		else{
			break;
		}
	}
	int j=n-1;
	while(s[j]=='0'){
		j--;
	}
	string ans;
	for(int k=0;k<=j;k++){
		ans+=s[k];
	}
	return ans;
}
int main(){
	string s;
	cin>>s;
	cout<<getMax(s)<<endl;
}
