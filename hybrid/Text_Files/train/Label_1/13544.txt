#include <bits/stdc++.h>

const int N = 2e5 + 5;

int a[N], l[N], r[N], x[N], n, m, p[N];

int main() {
  std::cin >> n >> m;
  for (int i = 0; i < n; i++) { std::cin >> a[i]; }
  for (int i = 0; i < m; i++) { std::cin >> l[i] >> r[i] >> x[i]; }
  for (int i = 0; i < m; i++) {
    if (l[i] == r[i]) {
      if (a[l[i] - 1] != x[i]) p[i] = l[i];
      else p[i] = -1;
    } else {
      if (a[l[i] - 1] != x[i]) p[i] = l[i];
      else if (a[r[i] - 1] != x[i]) p[i] = r[i];
      else
        for (int j = l[i]; j < r[i]; j++)
          if (a[j] != x[i]) { p[i] = j + 1; break; }
      if (!p[i]) p[i] = -1;
    }
    std::cout << p[i] << std::endl;
  }
  return 0;
}
