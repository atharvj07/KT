def earliest_login(first_name, last_name):
    login = first_name[0]  # starting with the first letter of the first name
    for i in range(1, min(len(first_name), len(last_name)) + 1):
        # Compare prefixes of the same length
        prefix_first = first_name[:i]
        prefix_last = last_name[:i]
        # If prefix of first name is lexicographically smaller or equal, take it
        if prefix_first <= prefix_last:
            login = prefix_first
        else:
            login += last_name[0]
            break
    else:
        # If we exited the loop normally, append the first letter of the last name
        login += last_name[0]
    return login

first_name, last_name = input().split()
print(earliest_login(first_name, last_name))
