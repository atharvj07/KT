import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        long mod = 998244353;

        int[][] range = new int[n][2];
        for (int i = 0; i < n; i++) {
            line = br.readLine().split(" ");
            range[i][0] = Integer.parseInt(line[0]);
            range[i][1] = Integer.parseInt(line[1]);
        }

        int[][] hate = new int[m][2];
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            hate[i][0] = Integer.parseInt(line[0]) - 1;
            hate[i][1] = Integer.parseInt(line[1]) - 1;
        }

        List<Integer>[] graph = new List[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int[] edge : hate) {
            int u = edge[0];
            int v = edge[1];
            graph[u].add(v);
            graph[v].add(u);
        }

        long[] dp = new long[1 << n];
        dp[0] = 1;
        for (int mask = 1; mask < (1 << n); mask++) {
            long count = 0;
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) != 0) {
                    boolean valid = true;
                    int size = Integer.bitCount(mask);
                    if (size < range[i][0] || size > range[i][1]) {
                        valid = false;
                    }
                    for (int j = 0; j < graph[i].size(); j++) {
                        int v = graph[i].get(j);
                        if ((mask & (1 << v)) != 0) {
                            valid = false;
                        }
                    }
                    if (valid) {
                        count += dp[mask ^ (1 << i)];
                    }
                }
            }
            dp[mask] = count % mod;
        }

        long result = dp[(1 << n) - 1];
        System.out.println(result);
    }
}
