n = input().strip()

found = False
# Directly check if the number itself is divisible by 8
if int(n) % 8 == 0:
    print("YES")
    print(n)
    found = True
else:
    # Check all subsets of size 1, 2 and 3
    for i in range(len(n)):
        if found:
            break
        for j in range(i+1, len(n)):
            if found:
                break
            for k in range(j+1, len(n)):
                if int(n[i]+n[j]+n[k]) % 8 == 0:
                    print("YES")
                    print(n[i]+n[j]+n[k])
                    found = True
                    break
                if int(n[i]+n[k]) % 8 == 0:
                    print("YES")
                    print(n[i]+n[k])
                    found = True
                    break
            if int(n[i]+n[j]) % 8 == 0:
                print("YES")
                print(n[i]+n[j])
                found = True
                break
        if int(n[i]) % 8 == 0:
            print("YES")
            print(n[i])
            found = True
            break

if not found:
    print("NO")
