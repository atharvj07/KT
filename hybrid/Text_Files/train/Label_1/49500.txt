import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = br.readLine().split(" ");
        int d = Integer.parseInt(line1[0]);
        int n = Integer.parseInt(line1[1]);
        int m = Integer.parseInt(line1[2]);
        long[] dists = new long[m + 2];
        long[] prices = new long[m + 2];
        for (int i = 0; i < m; i++) {
            String[] line2 = br.readLine().split(" ");
            dists[i] = Integer.parseInt(line2[0]);
            prices[i] = Integer.parseInt(line2[1]);
        }
        dists[m] = d;
        prices[m] = 0;
        Arrays.sort(dists);
        Arrays.sort(prices);
        long ans = 0;
        long need = 0;
        for (int i = 0; i < m + 1; i++) {
            long dist = dists[i + 1] - dists[i];
            if (need > dist * prices[i]) {
                ans = -1;
                break;
            } else {
                ans += need * prices[i];
                need = Math.max(0, dist - need);
            }
        }
        System.out.println(ans);
    }
}
