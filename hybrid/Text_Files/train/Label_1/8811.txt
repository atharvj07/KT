#include <bits/stdc++.h>
using namespace std;

void solve() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  int inc = 0, dec = 0;
  for (int i = 0; i + 1 < n; ++i) {
    if (a[i] < a[i + 1]) {
      ++dec;
    } else if (a[i] > a[i + 1]) {
      ++inc;
    }
  }
  cout << min({inc, dec, inc - 1, dec - 1}) << "\n";
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) { solve(); }
}
