#include<bits/stdc++.h>
#define ll long long
#define fi first
#define se second
#define pb push_back
#define mp make_pair
using namespace std;
const int maxn = 1e5 + 5;
int pri[maxn], vis[maxn], tot = 0, cnt = 0, a[maxn], b[maxn];
void dfs(int x) {
	vis[x] = 1;
	a[++ cnt] = x;
	for (int i = 1; i <= tot; i ++) {
		if (!vis[pri[i]] && x % pri[i]) {
			b[++ cnt] = pri[i];
			dfs(pri[i]);
		}
	}
}
int main() {
	int n, m, u, v; cin >> n >> m;
	for (int i = 2; i <= n; i ++) {
		if (!vis[i]) {
			pri[++ tot] = i;
			dfs(i);
		}
	}
	if (cnt < n) {cout << "Impossible" << endl; return 0;};
	cout << "Possible" << endl;
	for (int i = 1; i < m; i ++) {
		cout << a[i] << ' ' << a[i + 1] << endl;
	}
	cout << a[m] << ' ' << b[m - 1] << endl;
	return 0;
}
