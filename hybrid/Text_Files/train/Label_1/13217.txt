#include <iostream>
#include <string>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        string s = "";
        int i = 0;
        while (s.size() < n) {
            s += 'a' + i;
            i++;
            if (i == k) i = 0;
        }
        cout << s << endl;
    }
    return 0;
}
