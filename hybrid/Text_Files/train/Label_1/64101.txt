t = int(input())
for _ in range(t):
    n = int(input())
    ti = [0]
    xi = [0]
    for _ in range(n):
        t_, x_ = map(int, input().split())
        if t_ >= ti[-1]:
            ti.append(t_)
            xi.append(x_)
    ans = 0
    for i in range(len(ti)):
        if ti[i] <= xi[i] <= (ti[i+1] if i+1 < len(ti) else float('inf')):
            ans += 1
    print(ans)
