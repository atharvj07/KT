#include <bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin >> n;
    vector<int>a(n);
    int mx = 0;
    for(int i = 0;i < n;i++){
        cin >> a[i];
        mx = max(a[i], mx);
    }
    sort(a.begin(), a.end());
    if(a[n - 1] - a[0] <= 2){
        cout << "YES" << "\n";
        return 0;
    }
    else{
        if(a[n - 1] - a[1] <= 2 || a[n - 2] - a[0] <= 2){
            cout << "YES" << "\n";
            return 0;
        }
        else cout << "NO" << "\n";
    }
}
