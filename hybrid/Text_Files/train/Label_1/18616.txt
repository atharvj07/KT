#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll n, k;
int main() {
  cin >> n >> k;
  vector<bool> a(n + 1, 0);
  for (int i = 0; i < k; i++) {
    ll u, v;
    cin >> u >> v;
    a[u] = 1;
    if (v > 1) a[u] = 2;
  }
  set<ll> s;
  for (int i = 1; i <= n; i++) {
    if (a[i] == 2) s.insert(i);
  }
  if (s.size() > 0) {
    cout << 1 << endl;
    return 0;
  }
  for (int i = 1; i <= n; i++) {
    if (a[i] == 1) {
      if ((n - 1) % 2 == 1) {
        if (s.empty()) {
          cout << 1 << endl;
          return 0;
        }
      } else if (i == n) {
        cout << 1 << endl;
        return 0;
      }
    }
  }
  cout << -1 << endl;
  return 0;
}
