n, k = map(int, input().split())
costs = list(map(int, input().split()))

delayed_flights = sorted(((cost, i) for i, cost in enumerate(costs, start=1)), reverse=True)
departure_times = [0] * n
heap = []

for minute in range(k + 1, k + n + 1):
    while delayed_flights and delayed_flights[-1][1] <= minute:
        heapq.heappush(heap, delayed_flights.pop()[0])
    departure_times[minute - k - 1] = heapq.heappop(heap)

total_cost = sum(cost * (minute - i) for i, (cost, minute) in enumerate(sorted((cost, minute + k) for minute, cost in enumerate(departure_times, start=1)), start=1))
print(total_cost)
print(*[minute + k for minute in departure_times])
