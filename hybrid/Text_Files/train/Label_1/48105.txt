import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] arg = sc.nextLine().split(" ");
        long n = Long.parseLong(arg[0]);
        int k = Integer.parseInt(arg[1]);
        int ans = 0, need = 0;
        while (n > 0) {
            need = (int)(n % 10);
            n /= 10;
            ans++;
            if (need % k == 0 || (need == 0 && ans == 1)) {
                break;
            }
        }
        System.out.println(ans);
    }
}
