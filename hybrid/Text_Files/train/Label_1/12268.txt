#include<iostream>
#include<queue>
#include<vector>
#define ll long long
using namespace std;
vector<pair<ll,ll>>adj[100001];
vector<ll>dist(100001,1e18);
int main(){
    ll n,m,k,x=0,a,b,c;
    cin>>n>>m>>k;
    for(int i=0;i<m;++i){
        cin>>a>>b>>c;
        adj[a].push_back({b,c});
        adj[b].push_back({a,c});
    }
    if(k){cin>>x;}
    vector<ll>s(k,0);
    for(int i=0;i<k;++i){
        cin>>a;
        s[i]=a;
    }
    priority_queue<pair<ll,ll>,vector<pair<ll,ll>>,greater<pair<ll,ll>>>q;
    for(int i=0;i<k;++i){
        q.push({0,s[i]});
    }
    while(!q.empty()){
        auto u=q.top();
        q.pop();
        if(dist[u.second]<1e18){
            continue;
        }
        dist[u.second]=u.first;
        for(auto p:adj[u.second]){
            q.push({u.first+p.second,p.first});
        }
    }
    ll ans=1e18;
    for(int i=1;i<=n;++i){
        if(dist[i]==1e18){
            continue;
        }
        for(int j=0;j<k;++j){
            if(s[j]==i){
                continue;
            }
            ans=min(ans,dist[i]+dist[s[j]]);
        }
    }
    if(ans==1e18){
        cout<<-1<<endl;
    }
    else{
        cout<<ans<<endl;
    }
    return 0;
}
