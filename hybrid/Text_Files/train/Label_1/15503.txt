#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, a[1001], b[1001], c[1001] = {0}, d[1001] = {0}, p[1001], skip = 0;
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  for (int i = 0; i < n; i++) {
    c[a[i]]++;
    d[b[i]]++;
    if (c[a[i]] > 1 || d[b[i]] > 1) { skip = 1; }
  }
  if (skip == 1) {
    for (int i = 1; i <= n; i++) {
      if (c[i] == 0) { p[1] = i; }
      if (d[i] == 0) { p[2] = i; }
    }
    for (int i = 3; i <= n; i++) {
      p[i] = n - i + 3;
    }
  } else {
    for (int i = 1; i <= n; i++) {
      if (a[i] != b[i]) {
        p[i] = a[i];
        p[i + 1] = b[i];
        for (int j = i + 2; j <= n; j++) {
          if (a[j] == a[i]) {
            p[j] = b[i];
          } else if (a[j] == b[i]) {
            p[j] = a[i];
          } else {
            p[j] = a[j];
          }
        }
        break;
      }
    }
  }
  for (int i = 1; i <= n; i++) { cout << p[i] << " "; }
  cout << endl;
  return 0;
}
