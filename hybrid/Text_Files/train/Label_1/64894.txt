import sys

input = sys.stdin.readline

t = int(input())

for _ in range(t):
    n, m = map(int, input().split())
    k = list(map(int, input().split()))
    c = list(map(int, input().split()))

    dp = [1e18] * (m + 1)
    dp[0] = 0

    for i in range(1, m + 1):
        dp[i] = min(dp[i], c[i - 1])

    for i in k:
        for j in range(i, m + 1):
            dp[j] = min(dp[j], dp[j - i] + c[i - 1])

    print(dp[m])
