#include <bits/stdc++.h>
using namespace std;

int main() {
  int t; cin >> t; while (t--) {
    string s; cin >> s;
    int fr[26] = {0};
    for (char c : s) fr[c - 'a']++;
    bool odd = 0;
    for (int i = 0; i < 26; ++i) odd |= fr[i] & 1;
    if (odd) cout << s; else cout << -1;
    cout << '\n';
  }
}
