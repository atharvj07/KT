#include <bits/stdc++.h>
using namespace std;
int main(){
  int n,a,ans=2e9;
  cin>>n>>a;
  vector<int>x(n+2);
  for(int i=1;i<=n;++i)cin>>x[i];
  sort(x.begin()+1,x.end());
  x[0]=-2e9,x[n+1]=2e9;
  for(int i=1;i<=n;++i){
    int l=max(a,x[i-1]),r=min(a,x[i+1]);
    ans=min({ans,abs(l-x[i])+abs(a-x[i]),abs(r-x[i])+abs(a-x[i])});
  }
  cout<<ans<<endl;
}
