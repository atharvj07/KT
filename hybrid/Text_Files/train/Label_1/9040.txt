#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<string> grid(n);
  for (int i = 0; i < n; i++) { cin >> grid[i]; }
  vector<int> ans(m);
  for (int j = 0; j < m; j++) {
    vector<vector<int>> dp(n, vector<int>(m));
    int cnt = 0;
    for (int i = n - 2; i >= 0; i--) {
      for (int t = 0; t < m; t++) {
        if (i == n - 1)
          dp[i][t] = 0;
        else {
          if (j > 0) dp[i][t] = max(dp[i][t], dp[i + 1][t - 1]);
          if (j < m - 1) dp[i][t] = max(dp[i][t], dp[i + 1][t + 1]);
          dp[i][t] = max(dp[i][t], dp[i + 1][t]);
          if (grid[i][t] != '.' && (grid[i][t] == 'L' || grid[i][t] == 'U'))
            cnt++;
          dp[i][t] += cnt;
        }
      }
    }
    ans[j] = dp[0][j];
  }
  for (int x : ans) { cout << x << " "; }
  cout << endl;
}
