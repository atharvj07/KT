#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main()
{
    int n, ans = 0;
    cin >> n;
    vector<vector<ll>> a(n);
    for (int i = 0; i < n; i++)
    {
        int m;
        cin >> m;
        a[i].resize(m);
        for (ll &x : a[i])
            cin >> x;
    }
    for (auto &v : a)
    {
        sort(v.begin(), v.end());
        ans += v.back() - v[0];
    }
    cout << ans << '\n';
}
