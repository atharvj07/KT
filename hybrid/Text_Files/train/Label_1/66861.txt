n,q = map(int,input().split())
s=[x for x in input()]
for l,r in [list(map(int,input().split()))for _ in range(q)]:
  c=[0]*10
  i=l
  while i<=r:
    if s[i] in '<>':
      if s[i:i+1]<=s[i+1:i+2]:
        i+=1
      i+=1
    else:
      c[int(s[i])]+=1
      s[i] = str(max(0,int(s[i]) - 1))
      if s[i]=='0':
        s=s[:i]+s[i+1:]
      i+=1
  print(*c)
