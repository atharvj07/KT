#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, k, x;
    cin >> n >> k;
    vector<int> team[k];
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < k; ++j) {
            cin >> x;
            team[j].push_back(x);
        }
    }
    bool ok = true;
    for (int i = 0; i < n; ++i) {
        int cnt = 0;
        for (int j = 0; j < k; ++j) {
            cnt += team[j][i];
        }
        if (cnt * 2 >= k) {
            ok = false;
            break;
        }
    }
    cout << (ok ? "YES" : "NO") << endl;
    return 0;
}
