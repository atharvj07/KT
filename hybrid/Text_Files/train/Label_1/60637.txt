password = input()
n = int(input())
words = [input() for _ in range(n)]

can_unlock = "NO"
for word in words:
    if password[0] in word:
        for next_word in words:
            if password[1] in next_word and (word != next_word or word.count(password[0]) > 1):
                can_unlock = "YES"
                break
    if can_unlock == "YES":
        break

print(can_unlock)
