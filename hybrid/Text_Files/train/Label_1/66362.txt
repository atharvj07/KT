def count_occurrences(n, m, t, s):
    # Expanding only the first and last blocks if needed, and keeping the middle blocks as is
    expanded_t = []
    expanded_s = []
    
    # Expand t
    for i in range(n):
        l, c = t[i]
        if i == 0 or i == n-1:
            expanded_t.append((l, c))
        else:
            if len(expanded_t) > 0 and expanded_t[-1][1] == c:
                prev_l, _ = expanded_t.pop()
                expanded_t.append((prev_l + l, c))
            else:
                expanded_t.append((l, c))
    
    # Expand s
    for i in range(m):
        l, c = s[i]
        if i == 0 or i == m-1:
            expanded_s.append((l, c))
        else:
            if len(expanded_s) > 0 and expanded_s[-1][1] == c:
                prev_l, _ = expanded_s.pop()
                expanded_s.append((prev_l + l, c))
            else:
                expanded_s.append((l, c))
    
    # Count occurrences
    count = 0
    for i in range(len(expanded_t) - len(expanded_s) + 1):
        match = True
        for j in range(len(expanded_s)):
            if expanded_s[j][1] != expanded_t[i+j][1]: # Check character
                match = False
                break
            if j == 0 or j == len(expanded_s) - 1: # First or last block
                if expanded_s[j][0] > expanded_t[i+j][0]: # Check length strictly for first and last
                    match = False
                    break
            else: # Middle blocks
                if expanded_s[j][0] != expanded_t[i+j][0]: # Check length exactly for middle
                    match = False
                    break
        if match:
            count += 1
    
    return count

# Input handling
n, m = map(int, input().split())
t = [tuple(input().split('-')) for _ in range(n)]
t = [(int(l), c) for l, c in t]
s = [tuple(input().split('-')) for _ in range(m)]
s = [(int(l), c) for l, c in s]

# Output
print(count_occurrences(n, m, t, s))
