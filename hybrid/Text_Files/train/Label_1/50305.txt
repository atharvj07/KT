import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt();
        HashMap<Long, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            long coin = sc.nextLong();
            map.put(coin, map.getOrDefault(coin, 0) + 1);
        }
        for (int i = 0; i < q; i++) {
            long goal = sc.nextLong();
            long ans = -1;
            for (long coin : map.keySet()) {
                while (goal >= coin) {
                    if (ans == -1) ans = 0;
                    ans += map.get(coin);
                    goal -= coin;
                }
            }
            System.out.println(goal == 0 ? ans : -1);
        }
    }
}
