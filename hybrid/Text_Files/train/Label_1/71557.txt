n = int(input())
points = [tuple(input().split()) for _ in range(n)]
points = [(int(p), c) for p, c in points]

def min_cost(points):
    # Initialize variables to store the minimum cost and previous points for each color
    cost, last_r, last_g, last_b = 0, None, None, None
    
    for pos, color in points:
        if color == 'R':
            if last_g is not None: cost += pos - last_g
            if last_b is not None: cost += pos - last_b
            last_r = pos
        elif color == 'G':
            if last_r is not None: cost += pos - last_r
            if last_b is not None: cost += pos - last_b
            last_g = pos
        else: # color == 'B'
            if last_r is not None: cost += pos - last_r
            if last_g is not None: cost += pos - last_g
            last_b = pos
    
    # Initialize the minimum additional cost to connect the green points directly to infinity
    min_additional_cost = float('inf')
    first_g, last_g = None, None
    
    for pos, color in points:
        if color == 'G':
            if first_g is None:
                first_g = pos
            last_g = pos
    
    if first_g is not None:
        for i in range(1, len(points)):
            if points[i-1][1] != 'G' and points[i][1] != 'G':
                min_additional_cost = min(min_additional_cost, points[i][0] - points[i-1][0])
        cost += min_additional_cost - max(0, last_g - first_g)
    
    return cost

print(min_cost(points))
