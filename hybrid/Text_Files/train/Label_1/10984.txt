#include <bits/stdc++.h>
using namespace std;
int main() {
    int k;
    cin >> k;
    vector<vector<int>> v(4, vector<int>(4, 0));
    for (int i = 0; i < 4; i++) {
        string s;
        cin >> s;
        for (int j = 0; j < 4; j++) {
            if (s[j] == '.')
                continue;
            v[i][j] = s[j] - '0';
        }
    }
    int cnt = 0;
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            if (v[i][j] == 0)
                continue;
            cnt++;
            if (cnt > k) {
                cout << "NO" << endl;
                return 0;
            }
            if (v[i][j] > 2) {
                for (int h = 0; h < 4; h++) {
                    if (v[i][h] != 0 && v[i][h] == v[i][j]) {
                        cnt++;
                        if (cnt > k) {
                            cout << "NO" << endl;
                            return 0;
                        }
                    }
                }
                for (int w = 0; w < 4; w++) {
                    if (v[w][j] != 0 && v[w][j] == v[i][j]) {
                        cnt++;
                        if (cnt > k) {
                            cout << "NO" << endl;
                            return 0;
                        }
                    }
                }
            }
            if (v[i][j] == 2) {
                int cnt2 = 0;
                for (int h = 0; h < 4; h++) {
                    if (v[i][h] != 0) {
                        cnt2++;
                        if (cnt2 > k) {
                            cout << "NO" << endl;
                            return 0;
                        }
                    }
                }
                for (int w = 0; w < 4; w++) {
                    if (v[w][j] != 0) {
                        cnt2++;
                        if (cnt2 > k) {
                            cout << "NO" << endl;
                            return 0;
                        }
                    }
                }
            }
        }
    }
    cout << "YES" << endl;
    return 0;
}
