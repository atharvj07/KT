import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long s = sc.nextLong();
        long b = 2;
        while (b <= s) {
            long cur = 0;
            long t = n;
            while (t > 0) {
                cur += t % b;
                t /= b;
            }
            if (cur == s) {
                System.out.println(b);
                return;
            }
            b++;
        }
        System.out.println(-1);
    }
}
