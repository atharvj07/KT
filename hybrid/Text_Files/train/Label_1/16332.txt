#include <iostream>
#include <string>

int main() {
    std::string a, b, ans = "";
    std::cin >> a >> b;
    for (int i = 0, j = 0; i < b.size(); i++) {
        if (b[i] == a[j]) {
            ans += b[i];
            j++;
        }
    }
    if (ans.empty()) ans = "-";
    std::cout << ans << std::endl;
    return 0;
}
