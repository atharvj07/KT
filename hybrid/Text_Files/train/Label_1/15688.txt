#include <bits/stdc++.h>
using namespace std;

#define endl '\n'
typedef long long ll;
typedef pair<int, int> pii;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0); cout.tie(0);

    int n, x, y; cin >> n >> x >> y;
    vector<vector<int>> g(n+1);
    int deg[n+1] = {};
    for (int i = 1; i < n; i++) {
        int u, v; cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
        deg[u]++; deg[v]++;
    }

    int cen = 0;
    for (int i = 1; i <= n; i++) {
        if (deg[i] == 1) {
            cen = i; break;
        }
    }

    vector<int> dist(n+1, 1e9);
    function<void(int, int, int)> dfs1 = [&](int u, int p, int d) {
        dist[u] = d;
        for (int v : g[u]) {
            if (v == p) continue;
            dfs1(v, u, d+1);
        }
    };
    dfs1(cen, cen, 0);

    int mx = 0;
    int far = 0;
    for (int i = 1; i <= n; i++) {
        if (dist[i] > mx) {
            mx = dist[i];
            far = i;
        }
    }

    fill(dist.begin(), dist.end(), 1e9);
    dfs1(far, far, 0);

    mx = 0;
    int diam = 0;
    for (int i = 1; i <= n; i++) {
        if (dist[i] > mx) {
            mx = dist[i];
            diam = i;
        }
    }

    vector<int> path;
    function<void(int, int, int)> dfs2 = [&](int u, int p, int d) {
        path.push_back(u);
        if (u == diam) return;
        for (int v : g[u]) {
            if (v == p) continue;
            dfs2(v, u, d+1);
            path.pop_back();
        }
    };
    dfs2(diam, diam, 0);

    ll ans = 1LL * x * (path.size() - 1) + 1LL * y * mx;
    cout << ans << endl;

    return 0;
}
