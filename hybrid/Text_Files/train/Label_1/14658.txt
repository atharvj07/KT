#include <iostream>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n + m; i++) {
        if (i % 2 == 0) {
            if (n > 0) {
                cout << 'B';
                n--;
            } else {
                cout << 'G';
                m--;
            }
        } else {
            if (m > 0) {
                cout << 'G';
                m--;
            } else {
                cout << 'B';
                n--;
            }
        }
    }
    return 0;
}
