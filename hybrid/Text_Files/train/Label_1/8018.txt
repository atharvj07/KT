#include <algorithm>
#include <climits>
#include <iostream>
#include <vector>

using namespace std;

struct Seal {
  int x, y;
  bool swapped;
};

int main() {
  int n, a, b;
  cin >> n >> a >> b;

  vector<Seal> seals(n);
  for (int i = 0; i < n; i++) {
    cin >> seals[i].x >> seals[i].y;
    seals[i].swapped = false;
  }

  int maxArea = 0;

  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      int area1 = seals[i].x * seals[i].y;
      int area2 = seals[j].x * seals[j].y;

      if (!seals[i].swapped && !seals[j].swapped) {
        if (seals[i].x <= a && seals[i].y <= b && seals[j].x <= a && seals[j].y <= b) {
          maxArea = max(maxArea, area1 + area2);
        }
        if (seals[i].y <= a && seals[i].x <= b && seals[j].x <= a && seals[j].y <= b) {
          maxArea = max(maxArea, area1 + area2);
        }
      } else if (seals[i].swapped && !seals[j].swapped) {
        if (seals[i].y <= a && seals[i].x <= b && seals[j].x <= a && seals[j].y <= b) {
          maxArea = max(maxArea, area1 + area2);
        }
        if (seals[i].x <= a && seals[i].y <= b && seals[j].y <= a && seals[j].x <= b) {
          maxArea = max(maxArea, area1 + area2);
        }
      } else if (!seals[i].swapped && seals[j].swapped) {
        if (seals[i].x <= a && seals[i].y <= b && seals[j].y <= a && seals[j].x <= b) {
          maxArea = max(maxArea, area1 + area2);
        }
        if (seals[i].y <= a && seals[i].x <= b && seals[j].x <= a && seals[j].y <= b) {
          maxArea = max(maxArea, area1 + area2);
        }
      } else {
        if (seals[i].y <= a && seals[i].x <= b && seals[j].y <= a && seals[j].x <= b) {
          maxArea = max(maxArea, area1 + area2);
        }
        if (seals[i].x <= a && seals[i].y <= b && seals[j].x <= a && seals[j].y <= b) {
          maxArea = max(maxArea, area1 + area2);
        }
      }
    }
  }

  cout << maxArea << endl;

  return 0;
}
