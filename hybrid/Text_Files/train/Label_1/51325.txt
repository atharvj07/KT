import java.util.Scanner;

public class Main {
    private static boolean isDivisibleBy(int num, int d) {
        return num % d == 0;
    }

    private static int getSumOfDigits(int num) {
        int sum = 0;
        while (num > 0) {
            sum += num % 10;
            num /= 10;
        }
        return sum;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int d = scanner.nextInt();
        int s = scanner.nextInt();
        int n = d;
        while (getSumOfDigits(n) != s) {
            n += d;
        }
        System.out.println(isDivisibleBy(n, d) ? n : -1);
    }
}
