#include <bits/stdc++.h>

using namespace std;

int main() {
    int n; cin >> n;
    vector<unordered_set<int>> divs(n + 2);
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j += i) {
            divs[i].insert(j);
            divs[j].insert(i);
        }
    }
    vector<int> ans(n + 1), col(n + 1);
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (col[i]) continue;
        cnt++;
        queue<int> q; q.push(i); col[i] = cnt;
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : divs[u]) {
                if (!col[v]) {
                    q.push(v);
                    col[v] = cnt;
                }
            }
        }
    }
    cout << cnt << endl;
    for (int i = 1; i <= n; i++) {
        cout << col[i] << " ";
    } cout << endl;
    return 0;
}
