#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int INF=0x3f3f3f3f;
const int maxn=100010;
char s[maxn];
int main(){
    ll n,k;
    cin>>n>>k;
    cin>>s;
    if(n%k!=0){
        cout<<"tokitsukaze\n";
    }else{
        ll cnt=0;
        for(ll i=0;i<n;i++){
            if(s[i]=='0')cnt++;
        }
        if(cnt%k==0){
            cout<<"once again\n";
        }else{
            if(cnt&1){
                cout<<"quailty\n";
            }else{
                cout<<"tokitsukaze\n";
            }
        }
    }
    return 0;
}
