#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  bool a = 0, b = 0, c = 0;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == 'C' and !c) c = 1;
    else if (s[i] == 'D' and c and !a) a = 1;
    else if (s[i] == 'E' and c and a and !b) b = 1;
  }
  cout << ((c and a and b) ? "YES" : "NO") << endl;
  return 0;
}
