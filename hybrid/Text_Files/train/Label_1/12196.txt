#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e5 + 5;
long long N, K, A[MAXN], D[MAXN], E[MAXN], dp[2][MAXN];

int main() {
    cin >> N >> K;
    for (long long i = 1; i <= N; i++) cin >> A[i];
    for (long long i = 1; i <= N; i++) cin >> D[i];

    for (long long i = 1; i <= K; i++) E[i] = i + 1;

    for (long long i = K + 1; i <= N; i++) {
        long long j = i - K;
        dp[1][i] = dp[0][j];
        for (long long k = j + 1; k < i; k++) {
            dp[1][i] = max(dp[1][i], dp[0][k] + A[i] - D[i]);
        }
        swap(dp[0], dp[1]);
    }

    cout << dp[0][N] << endl;
    return 0;
}
