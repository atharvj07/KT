#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int mx=1e8;
int a[mx+2],b[mx+2];
int32_t main() {
    ll n,k,x;
    cin>>n>>k>>x;
    for(int i=0; i<n; i++) {
        cin>>a[i];
    }
    sort(a,a+n);
    ll ans=0,l=0,r=0,cnt=0;
    while(r<n) {
        while(r<n && a[r]-a[l]<=x) {
            r++;
            cnt++;
        }
        if(cnt>0){
            ans++;
            cnt--;
            l=r;
        }
        else l++,r++;
    }
    if(cnt>0) ans++;
    if(k>=ans-1) cout<<1<<endl;
    else cout<<ans-k<<endl;
    return 0;
}
