#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int N = 1e5 + 5;

ll a[N], dp[N];

int main() {
  int n, c;
  cin >> n >> c;
  for (int i = 0; i < n; i++) cin >> a[i];
  priority_queue<ll> v;
  for (int i = 0; i < c; i++) v.push(-a[i]);
  dp[0] = c * (-v.top());
  for (int i = 1; i < n; i++) {
    v.push(-a[i]);
    v.pop();
    dp[i] = dp[i - 1] + c * (-v.top());
  }
  ll ans = 1e18;
  for (int i = 0; i < n; i++) ans = min(ans, dp[i] + c * v.top());
  cout << ans << endl;
  return 0;
}
