import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static int[] dx = new int[]{-1, 0, 1, 0};
    static int[] dy = new int[]{0, 1, 0, -1};

    static int bfs(int[][] g, int x, int y) {
        int m = g.length, n = g[0].length;
        int[][] d = new int[m][n];
        for (int[] e : d) Arrays.fill(e, -1);
        d[x][y] = 0;
        Queue<int[]> q = new LinkedList<>();
        q.add(new int[]{x, y});
        while (!q.isEmpty()) {
            int[] p = q.poll();
            for (int i = 0; i < 4; i++) {
                int nx = p[0] + dx[i], ny = p[1] + dy[i];
                if (0 <= nx && nx < m && 0 <= ny && ny < n && g[nx][ny] > 0 && d[nx][ny] == -1) {
                    d[nx][ny] = d[p[0]][p[1]] + 1;
                    q.add(new int[]{nx, ny});
                }
            }
        }
        int max = 0;
        for (int[] e : d) for (int f : e) max = Math.max(max, f);
        return max;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int m = in.nextInt(), n = in.nextInt();
        int[][] g = new int[m][n];
        for (int i = 0; i < m; i++) {
            String s = in.next();
            for (int j = 0; j < n; j++) g[i][j] = s.charAt(j) == '#' ? 1 : 0;
        }
        int ans = -1;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (g[i][j] > 0) {
                    g[i][j] = 0;
                    int cur = bfs(g, i, j);
                    ans = ans == -1 ? cur : Math.min(ans, cur);
                    g[i][j] = 1;
                }
            }
        }
        System.out.println(ans);
    }
}
