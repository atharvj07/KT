import sys
input = sys.stdin.readline

q = int(input())

for _ in range(q):
    n, k = map(int, input().split())
    s = input().strip()
    if k == n:
        print(0)
        continue
    if k == 1:
        print(1)
        continue
    if s[0] != s[k - 1]:
        print(1)
        continue
    if s[0] == 'R':
        if 'G' in s[1: k - 1] or 'B' in s[1: k - 1]:
            print(1)
            continue
    elif s[0] == 'G':
        if 'R' in s[1: k - 1] or 'B' in s[1: k - 1]:
            print(1)
            continue
    else:
        if 'R' in s[1: k - 1] or 'G' in s[1: k - 1]:
            print(1)
            continue
    print(2)
