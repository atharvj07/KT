import sys

t = int(input())
for _ in range(t):
    n = int(input())
    l, r = [], []
    for _ in range(n):
        a, b = map(int, input().split())
        l.append(a)
        r.append(b)
    l.sort()
    r.sort()
    ans = [-1] * n
    if l[0] >= r[n - 2]:
        for i in range(n - 1):
            ans[i] = 1
        ans[n - 1] = 2
    elif l[1] >= r[n - 1]:
        ans[0] = 1
        for i in range(1, n):
            ans[i] = 2
    else:
        print(-1)
        continue
    for i in ans:
        print(i, end=' ')
    print()
