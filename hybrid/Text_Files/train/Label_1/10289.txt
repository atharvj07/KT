#include<bits/stdc++.h>
using namespace std;
int id1[256],id2[256];
int main(){
	string s1,s2,s,t;
	cin>>s1>>s2>>t;
	for(int i=0;i<26;++i)id1[s1[i]]=i,id2[s2[i]]=i;
	for(int i=0;i<t.size();++i){
		char c=t[i];
		if(c>='a'&&c<='z'){
			c=s2[id1[c]];
		}
		s.push_back(c);
	}
	cout<<s<<'\n';
	return 0;
}
