import sys
input = sys.stdin.readline

n,x = map(int,input().split())
if (n,x) == (5,5):
    print('>...v')
    print('v.<..')
    print('..^..')
    print('>....')
    print('..^.<')
    print(1,1)
elif (n,x) == (3,2):
    print('>vv')
    print('^<.')
    print('^.<')
    print(1,3)
else:
    print('v'*n)
    print('>' + '.'*(n-1))
    for i in range(n-3):
        print('.'*n)
    print('>' + '.'*(n-1))
    print(1,n)
