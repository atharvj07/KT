import bisect

n = int(input())
multiset = dict()
times = []
for i in range(n):
    a, t, x = map(int, input().split())
    if a == 1:
        multiset[x] = multiset.get(x, []) + [t]
        times.append(t)
    elif a == 2:
        ind = bisect.bisect_left(multiset[x], t)
        if ind < len(multiset[x]):
            del multiset[x][ind]
    elif a == 3:
        print(bisect.bisect_left(multiset.get(x, []), t))
