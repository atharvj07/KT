#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<pair<int, int>> debts[n + 1];
  for (int i = 0; i < m; i++) {
    int a, b, c;
    cin >> a >> b >> c;
    debts[a].emplace_back(b, c);
    debts[b].emplace_back(a, -c);
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    int sum = 0;
    for (auto& debt : debts[i]) {
      sum += debt.second;
    }
    ans += abs(sum);
  }
  ans /= 2;
  cout << ans << endl;
  return 0;
}
