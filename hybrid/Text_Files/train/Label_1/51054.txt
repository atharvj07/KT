import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        System.out.println(isConsistent(n, a) ? pow(2, n - 1, 1000000007) : 0);
    }

    static boolean isConsistent(int n, int[] a) {
        int[] l = new int[n], r = new int[n];
        for (int i = 0; i < n; i++) {
            if (i - a[i] < 0 || i + a[i] >= n) return false;
            l[i - a[i]]++;
            r[i + a[i]]++;
        }
        for (int i = 0; i < n; i++) if (l[i] != 1 || r[i] != 1) return false;
        return true;
    }

    static long pow(long base, long exp, long mod) {
        if (exp == 0) return 1;
        if (exp == 1) return base % mod;
        long half = pow(base, exp / 2, mod);
        if (exp % 2 == 0) return (half * half) % mod;
        return ((half * half) % mod * base) % mod;
    }
}
