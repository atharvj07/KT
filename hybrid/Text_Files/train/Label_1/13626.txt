#include <bits/stdc++.h>
using namespace std;
#define MAXN 500111
#define MOD 1000000007
#define int LL

typedef long long LL;
char s[MAXN], t[MAXN];
char ans[MAXN];
LL mch[MAXN][3],rev[MAXN][3],sLen,tLen;
void pre()
{
    LL cnt[3]={0,0,0};
    mch[0][0]=mch[0][1]=mch[0][2]=0;
    for(LL i=0;i<sLen;++i)
    {
        cnt[s[i]-'A']++;
        mch[i+1][0]=cnt[0];mch[i+1][1]=cnt[1];mch[i+1][2]=cnt[2];
    }
    memset(cnt,0,sizeof(cnt));
    for(LL i=sLen-1;i>=0;--i)
    {
        cnt[s[i]-'A']++;
        rev[i+1][0]=cnt[0];rev[i+1][1]=cnt[1];rev[i+1][2]=cnt[2];
    }
}
int32_t main() {
    cin>>s>>t;
    sLen=strlen(s),tLen=strlen(t);
    pre();
    LL q,a,b,c,d;
    cin>>q;
    while(q--)
    {
        cin>>a>>b>>c>>d;
        c--;
        LL cnt=mch[b][0]-mch[a-1][0]-rev[c][0]+rev[a-1+tLen-d][0];
        if(cnt&1)ans[q]='0';
        else
       {
            cnt=mch[b][1]-mch[a-1][1]-rev[c][1]+rev[a-1+tLen-d][1];
            if(cnt&1)ans[q]='0';
            else
            {
                cnt=mch[b][2]-mch[a-1][2]-rev[c][2]+rev[a-1+tLen-d][2];
                if(cnt&1)ans[q]='0';
                else ans[q]='1';
            }
        }
    }
    ans[q]='\0';
    cout<<ans<<endl;
    return 0;
}
