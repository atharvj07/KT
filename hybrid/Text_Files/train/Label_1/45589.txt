import java.util.Arrays;
import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);

		int n = sc.nextInt();
		int[] a = new int[n];
		for (int i = 0; i < n; i++) {
			a[i] = sc.nextInt();
		}

		boolean[] odd = new boolean[n];
		int sumOdd = 0;
		for (int i = 0; i < n; i++) {
			odd[i] = (a[i] % 2 == 1);
			sumOdd += odd[i] ? 1 : 0;
		}

		Arrays.sort(a);

		if (sumOdd % 2 == 0) {
			System.out.println("YES");
			return;
		}

		boolean hasZero = false;
		for (int i = 0; i < n; i++) {
			if (a[i] == 0) {
				hasZero = true;
			}
		}

		if (hasZero && sumOdd == 1) {
			System.out.println("YES");
			return;
		}

		int minOdd = Integer.MAX_VALUE;
		for (int i = 0; i < n; i++) {
			if (odd[i] && a[i] < minOdd) {
				minOdd = a[i];
			}
		}

		if (minOdd % 2 == 0) {
			System.out.println("YES");
			return;
		}

		System.out.println("NO");
	}
}
