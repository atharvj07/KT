#include <iostream>
#include <string>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    string w(n, '0');
    for (int i = 1, j = 0; i <= n; i++, j += 2) {
      w[i - 1] = s[j] == '0' ? '0' : '1';
    }
    cout << w << endl;
  }
  return 0;
}
