#include <iostream>
#include <map>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int x = 0, y = 0;
    map<pair<int, int>, int> m;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'R')
            x++;
        if (s[i] == 'L')
            x--;
        if (s[i] == 'U')
            y++;
        if (s[i] == 'D')
            y--;
        if (!m[{x, y}]) ans++;
        m[{x, y}]++;
    }
    cout << ans;
    return 0;
}
