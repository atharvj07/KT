import sys
input = sys.stdin.readline

n, k = map(int, input().split())
a, b, c, d = map(int, input().split())
if k > 2 * n - 4:
    print(-1)
    exit()
if k < n:
    print(-1)
    exit()
print('%d %d' % (b, a), end=' ')
for i in range(1, n + 1):
    if i != a and i != b: print(i, end=' ')
print()
print('%d %d' % (d, c), end=' ')
for i in range(n, 0, -1):
    if i != c and i != d: print(i, end=' ')
