#include <bits/stdc++.h>
using namespace std;
int a[1005][1005], b[1005][1005], n, m;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> m;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) cin >> a[i][j];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      b[i][j] = a[i][j];
      if (j > 1 && a[i][j] > a[i][j - 1]) b[i][j] = max(b[i][j], a[i][j - 1] + 1);
      if (i > 1 && a[i][j] > a[i - 1][j]) b[i][j] = max(b[i][j], a[i - 1][j] + 1);
    }
  }
  for (int j = 1; j <= m; j++) {
    for (int i = n; i >= 1; i--) {
      if (j > 1 && b[i][j] > b[i][j - 1]) b[i][j] = max(b[i][j], b[i][j - 1] + 1);
      if (i > 1 && b[i][j] > b[i - 1][j]) b[i][j] = max(b[i][j], b[i - 1][j] + 1);
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) cout << b[i][j] << " ";
    cout << "\n";
  }
}
