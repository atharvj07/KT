#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin>>s;
    int length=s.length();
    int open=0,ans=0;
    for(int i=0;i<length;i++){
        if(s[i]=='[')    open++;
        else if(s[i]==']')   open--;
        else if(open!=0) ans+=2;
    }
    if(open!=0)   cout<<-1<<endl;
    else cout<<ans<<endl;
    return 0;
}
