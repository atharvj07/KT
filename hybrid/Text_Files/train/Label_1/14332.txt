#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        if (n % 2 != k % 2 || n < k) {
            cout << "NO\n";
            continue;
        }
        cout << "YES\n";
        if (k == 1) {
            cout << n << "\n";
            continue;
        }
        // k >= 2
        long long x = n / k;
        if (n % k == 0) {
            for (int i = 0; i < k; i++) cout << x << " ";
        } else {
            cout << (n % k) << " ";
            x++;
            for (int i = 1; i < k; i++) cout << x << " ";
        }
        cout << "\n";
    }
}
