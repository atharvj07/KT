import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] arr = new int[n][2 * m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 2 * m; j += 2) {
                String s = sc.next();
                arr[i][j] = s.charAt(0) - '0';
                arr[i][j + 1] = s.charAt(1) - '0';
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 2 * m; j++) {
                if (arr[i][j] == 0) {
                    arr[i][j] = 1;
                    arr[i][j + 1] = 0;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 2 * m; j++) {
                System.out.print(arr[i][j] + "" + arr[i][j + 1] + " ");
                j++;
            }
            System.out.println();
        }
    }
}
