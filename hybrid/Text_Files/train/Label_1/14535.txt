#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back

int main(){
    ll n, x, h, ans = 0;
    vector<pair<ll,ll>> v;
    cin >> n;
    for(ll i=0;i<n;i++){
        cin >> x >> h;
        v.pb({x-h,1});
        v.pb({x+h,-1});
    }
    sort(v.begin(),v.end());
    ll cnt = 0;
    for(auto c: v){
        cnt += c.second;
        ans += (cnt==0);
    }
    cout << ans << endl;
}
