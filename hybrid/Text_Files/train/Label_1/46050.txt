import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        String S = sc.next();
        List<Integer>[] G = new List[N];
        for (int i = 0; i < N; i++) {
            G[i] = new ArrayList<>();
        }
        for (int i = 0; i < N - 1; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            G[a].add(b);
            G[b].add(a);
        }

        int[] cnt = new int[N];
        for (int i = 0; i < N; i++) {
            cnt[i] = S.charAt(i) - '0';
        }

        int ans = -1;
        Queue<Integer> Q = new LinkedList<>();
        for (int i = 0; i < N; i++) {
            if (cnt[i] > 0) {
                Q.add(i);
            }
        }

        while (Q.size() > 1) {
            int u = Q.poll();
            int v = Q.poll();
            ans++;
            cnt[u]--;
            cnt[v]--;
            if (cnt[u] > 0) {
                Q.add(u);
            }
            if (cnt[v] > 0) {
                Q.add(v);
            }
        }

        System.out.println(ans);
    }
}
