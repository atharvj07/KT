#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll N=2e5+9;
ll n,a[N],b[N];
int main(){
    cin>>n;
    for(ll i=0;i<n;i++)
      cin>>a[i];
    for(ll i=0;i<n;i++)
      cin>>b[i];
    sort(b,b+n);
    ll cnt=0;
    for(ll i=0;i<n;i++){
      if(b[i]>0)
       break;
      if(a[i]==0)
       cnt++;
      else
       break;
    }
    for(ll i=n-1;i>=0;i--){
      if(b[i]<0)
       break;
      if(a[i]==0)
       cnt++;
      else
       break;
    }
    cout<<cnt<<endl;
}
