n=int(input())
a=[ list(map(int,input().split())) for i in range(n)]
k=int(input())
b=list(map(int,input().split()))
a.sort(key=lambda x:x[1]/x[0],reverse=True)
b.sort()
sum=0
ans=[]
for x,y in a:
    for i in range(k):
        if b[i]>=x:
            sum+=y
            b[i]-=x
            ans.append([x,i+1])
            break
    if b[k-1]<x:
        break
print(len(ans),sum)
for x in ans:
    print(*x)
