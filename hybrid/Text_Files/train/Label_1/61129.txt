def solve():
    t = int(input())
    for _ in range(t):
        n, k = map(int, input().split())
        k -= 1
        if k >= (n-1)*n//2:
            print(-1)
            continue
        p = list(range(1, n+1))
        for i in range(k):
            j = k - i - 1
            if p[j] < p[j+1]:
                p[j], p[j+1] = p[j+1], p[j]
                break
        print(*p)

solve()
