#include <bits/stdc++.h>
using namespace std;
const int mod=1e9+7;
int main() {
    int t,n;
    cin>>t;
    while(t--){
        cin>>n;
        vector<int> a(n);
        map<int,int> cnt;
        for(int &x:a) cin>>x,cnt[x]++;
        int ans=n/2;
        for(int i=0;i<n;i++){
            for(int j=0;j<n;j++){
                if(a[i]==a[j]) continue;
                if(cnt.find(a[i]%a[j])==cnt.end()){
                    cout<<a[i]<<" "<<a[j]<<'\n';
                    if(--ans==0) return 0;
                }
            }
        }
    }
    return 0;
}
