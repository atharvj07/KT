#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
const ll mod=1e9+7;
int n;
ll a[300005];
ll b[300005];
vector<ll>bul;
int main(){
	cin>>n;
	for(int i=0;i<n;i++){
		cin>>a[i];
		b[a[i]]=i;
	}
	for(int i=0;i<n;i++){
		cin>>a[i];
		bul.pb(a[i]);
	}
	sort(bul.begin(),bul.end());
	ll ans=0;
	for(auto &x:bul){
		ans+=x;
		x=b[x];
	}
	cout<<ans<<endl;
	return 0;
}
