#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int v;
    cin >> v;
    vector<int> a(9);
    for (int i = 0; i < 9; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 8; i >= 0; i--) {
        if (v >= a[i]) {
            v -= a[i];
            ans = ans * 10 + (i + 1);
        }
    }
    if (ans == 0) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }
    return 0;
}
