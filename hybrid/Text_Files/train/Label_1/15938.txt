#include <bits/stdc++.h>
using namespace std;
const int maxn = 1 << 20;
int a[maxn], nx[maxn], c[119];
int main() {
    int n, x;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> x;
        int t = 0;
        for (int j = 0; j < 119; j++) {
            int p = x >> j & 1;
            if (c[j] && nx[c[j]] < i) {
                cout << 1 << ' ' << c[j] << endl;
                return 0;
            }
            if (!c[j] || p) {
                t = j;
            }
        }
        if (!c[t]) {
            cout << 0 << endl;
        } else {
            nx[c[t]] = i;
            cout << 1 << ' ' << c[t] << endl;
        }
        c[t] = i;
    }
    return 0;
}
