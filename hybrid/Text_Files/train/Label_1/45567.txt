import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int a = input.nextInt();
        int b = input.nextInt();
        int f = input.nextInt();
        int k = input.nextInt();

        int requiredRefuels = (int) Math.ceil((double) (2 * a * k) / b);

        if (requiredRefuels * b < 2 * a * k) {
            System.out.println(-1);
        } else {
            if (f < a - f) {
                System.out.println(requiredRefuels - (requiredRefuels % 2));
            } else {
                System.out.println(requiredRefuels - (requiredRefuels - 1) % 2);
            }
        }
    }
}
