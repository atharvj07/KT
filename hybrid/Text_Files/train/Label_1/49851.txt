import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextLong();
        }
        Arrays.sort(arr);
        long min = arr[0];
        long max = arr[n - 1];
        long ans = 0;
        if (min == max) {
            ans = n * (n + 1) / 2;
        }
        System.out.println(ans);
    }
}
