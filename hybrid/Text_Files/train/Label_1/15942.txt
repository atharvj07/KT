#include <bits/stdc++.h>
using namespace std;

int main() {
  long long n, k, x;
  cin >> n >> k >> x;
  vector<long long> a(n);
  for (auto &x : a) cin >> x;
  sort(a.begin(), a.end(), greater<long long>());
  for (int i = 0; i < k; i++) {
    a[i] *= x;
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    ans |= a[i];
  }
  cout << ans << endl;
  return 0;
}
