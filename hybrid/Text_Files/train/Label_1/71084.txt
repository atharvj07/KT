from collections import deque

s = input()
n = len(s)
if s[0] == ')' or s[-1] == '(':
	print(':(')
	exit()
l = 0
r = 0
for i in s:
	if i == '(':
		l += 1
	elif i == ')':
		r += 1
	else:
		l += 1
		r += 1
if r < l:
	print(':(')
	exit()
ans = [i for i in s]
for i in range(n):
	if ans[i] == '?' and l > 0 and r > 0:
		if l > r:
			ans[i] = ')'
			r -= 1
		else:
			ans[i] = '('
			l -= 1
		continue
	if ans[i] == '(':
		l -= 1
	elif ans[i] == ')':
		r -= 1
if r == 0 and l == 0:
	print(''.join(ans))
else:
	print(':(')
