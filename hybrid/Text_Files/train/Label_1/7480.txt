#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int mx = 4e5 + 10;
vector<int> vec[25];
int cnt[25], lft[mx], rgt[mx];
int n, a[mx];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i], cnt[a[i]]++; }
  for (int i = 1; i <= n; i++) {
    vec[a[i]].push_back(i);
  }
  int op = 0;
  for (int i = 1; i <= 20; i++) {
    int lst = vec[i].size() - 1;
    if (lst < 0) continue;
    lft[vec[i][0]] = 1;
    rgt[vec[i][lst]] = n;
    for (int j = 0; j < lst; j++) {
      int l = vec[i][j], r = vec[i][j + 1];
      if (l + 1 == r) continue;
      op += min({r - l + 1 - cnt[i], (lft[l] - l + 1), (r + 1 - rgt[r])});
      lft[r] = lft[l];
      rgt[l] = rgt[r];
    }
  }
  cout << op << endl;
  return 0;
}
