import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class GameGrid {

    static class Cell {
        int x, y;

        Cell(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static class Player {
        int speed;
        int castles;

        Player(int speed, int castles) {
            this.speed = speed;
            this.castles = castles;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();

        Player[] players = new Player[p];
        for (int i = 0; i < p; i++) {
            players[i] = new Player(scanner.nextInt(), 0);
        }

        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = line.charAt(j);
                if (grid[i][j] != '.' && grid[i][j] != '#') {
                    players[grid[i][j] - '1'].castles++;
                }
            }
        }

        Queue<Cell> queue = new LinkedList<>();
        boolean[][] visited = new boolean[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] != '.' && grid[i][j] != '#') {
                    queue.add(new Cell(i, j));
                    visited[i][j] = true;
                }
            }
        }

        while (!queue.isEmpty()) {
            Cell current = queue.poll();

            for (int dx = -1; dx <= 1; dx++) {
                for (int dy = -1; dy <= 1; dy++) {
                    if (Math.abs(dx) == Math.abs(dy)) continue;

                    int nx = current.x + dx;
                    int ny = current.y + dy;

                    if (nx >= 0 && nx < n && ny >= 0 && ny < m && !visited[nx][ny]) {
                        boolean canExpand = true;
                        for (int k = 0; k <= players[grid[current.x][current.y] - '1'].speed; k++) {
                            int x = current.x + k * dx;
                            int y = current.y + k * dy;

                            if (x < 0 || x >= n || y < 0 || y >= m || grid[x][y] == '#') {
                                canExpand = false;
                                break;
                            }
                        }

                        if (canExpand) {
                            grid[nx][ny] = grid[current.x][current.y];
                            queue.add(new Cell(nx, ny));
                            visited[nx][ny] = true;
                            players[grid[nx][ny] - '1'].castles++;
                        }
                    }
                }
            }
        }

        for (int i = 0; i < p; i++) {
            System.out.print(players[i].castles + " ");
        }
    }
}
