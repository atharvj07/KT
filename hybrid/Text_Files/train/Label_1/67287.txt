n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

chosen_jobs = {}
for i in range(n):
    job = a[i]
    time = b[i]
    if job in chosen_jobs:
        chosen_jobs[job].append(time)
    else:
        chosen_jobs[job] = [time]

for job in chosen_jobs:
    chosen_jobs[job].sort(reverse=True)

unassigned_jobs = k - len(chosen_jobs)
persuasion_times = []

for job in chosen_jobs:
    persuasion_times.extend(chosen_jobs[job][1:])

persuasion_times.sort()

print(sum(persuasion_times[:unassigned_jobs]))
