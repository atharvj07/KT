n=int(input())
a=[0]*n
b=[0]*n
for i in range(n):
    a[i],b[i]=map(int,input().split())
for i in range(1,n):
    for j in range(i):
        if b[i]>b[j]:
            b[i],a[i],b[j],a[j]=b[j],a[j],b[i],a[i]
tot=0
for i in range(n):
    if tot>=a[i]:
        continue
    x=b[i]
    tmp=(a[i]-tot)//x
    if (a[i]-tot)%x:
        tmp+=1
    tot+=x*tmp
    tot-=a[i]
print(tot*2)
