import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long n = input.nextLong(); // number of crosswords
        long t = input.nextLong(); // time Adilbek has to spend
        long[] times = new long[(int)n]; // time it takes to solve each crossword
        for (int i = 0; i < n; i++)
            times[i] = input.nextLong();
        long p = 0;
        long q = 1;
        for (long time : times) {
            p = p * 2 + q;
            q = q * 2;
            if (t >= time) {
                t -= time;
            } else
                break;
        }
        long gcd = gcd(p, q);
        System.out.println((p / gcd) * modPow(q / gcd, 1000000006, 1000000007) % 1000000007);
    }

    static long gcd(long a, long b) {
        if (b == 0)
            return a;
        return gcd(b, a % b);
    }

    static long modPow(long base, long exp, long mod) {
        base %= mod;
        long result = 1;
        while (exp > 0) {
            if ((exp & 1) == 1)
                result = (result * base) % mod;
            base = (base * base) % mod;
            exp >>= 1;
        }
        return result;
    }
}
