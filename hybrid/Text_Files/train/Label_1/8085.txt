#include <iostream>
#include <cstdio>
#include <vector>
#include <algorithm>

using namespace std;

int main()
{
    long long a1, b1, a2, b2;
    cin >> a1 >> b1 >> a2 >> b2;
    vector<pair<long long, long long> > ans;
    long long sq1 = a1 * b1;
    long long sq2 = a2 * b2;
    if ((sq1 & sq2) || (sq1 & (sq2 >> 1)) || (sq2 & (sq1 >> 1)))
    {
        long long x, y;
        sq1 >>= 1;
        x = sq1 % 2;
        sq1 >>= 1;
        y = sq1 % 2;
        ans.push_back(make_pair(sq1, 2 * sq1 + x + y));
    }
    else if ((sq2 & (sq1 << 1)) || (sq1 & (sq2 << 1)))
    {
        long long x, y;
        sq2 >>= 1;
        x = sq2 % 2;
        sq2 >>= 1;
        y = sq2 % 2;
        ans.push_back(make_pair(2 * sq2 + x + y, sq2));
    }
    if ((sq1 % 3) == 0)
    {
        long long x, y;
        sq1 /= 3;
        x = sq1 % 3;
        sq1 /= 3;
        y = sq1 % 3;
        ans.push_back(make_pair(sq1, 3 * sq1 + x + y));
    }
    else if ((sq2 % 3) == 0)
    {
        long long x, y;
        sq2 /= 3;
        x = sq2 % 3;
        sq2 /= 3;
        y = sq2 % 3;
        ans.push_back(make_pair(3 * sq2 + x + y, sq2));
    }
    if ((sq1 & (sq2 >> 2)) || (sq2 & (sq1 >> 2)))
    {
        long long x, y;
        sq1 >>= 2;
        x = sq1 % 2;
        sq1 >>= 1;
        y = sq1 % 2;
        ans.push_back(make_pair(sq1, 4 * sq1 + x + y));
    }
    else if ((sq2 & (sq1 << 2)) || (sq1 & (sq2 << 2)))
    {
        long long x, y;
        sq2 >>= 2;
        x = sq2 % 2;
        sq2 >>= 1;
        y = sq2 % 2;
        ans.push_back(make_pair(4 * sq2 + x + y, sq2));
    }
    if (ans.size() > 0)
    {
        cout << ans.size() << endl;
        for (long long i = 0; i < ans.size(); i++)
        {
            cout << ans[i].first << ' ' << ans[i].second << endl;
        }
    }
    else
    {
        cout << -1 << endl;
    }
    return 0;
}
