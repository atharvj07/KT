#include <iostream>
#include <unordered_map>

int main()
{
    int n, m, i, j, c = 0;
    std::string a[300000], b;
    std::unordered_map<std::string, int> p, q;
    std::cin >> n >> m;
    for (i = 0; i < n; i++)
    {
        std::cin >> a[i];
        b = a[i];
        for (j = 0; j < b.size(); j++)
            p[b.substr(0, j) + b.substr(j + 1)] = 1;
    }
    for (i = 0; i < m; i++)
    {
        std::cin >> b;
        q[b]++;
        if ((c = p[b]) != 0)
        {
            if (q[b] == c)
                std::cout << "YES" << std::endl;
            else
                std::cout << "NO" << std::endl;
        }
        else
            std::cout << "NO" << std::endl;
    }
    return 0;
}
