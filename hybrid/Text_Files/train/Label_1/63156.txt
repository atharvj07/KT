import sys

def solve():
    n, m, k = map(int, input().split())

    streetsV = list(map(int, input().split()))
    streetsH = list(map(int, input().split()))
    
    x = [0] * k
    y = [0] * k
    
    for i in range(k):
        a, b = map(int, input().split())
        x[i], y[i] = a, b
    
    x.sort()
    y.sort()
    
    ans = 0
    for i in range(k):
        for j in range(i + 1, k):
            a1 = streetsV.index(x[i]) if x[i] in streetsV else -1
            b1 = streetsH.index(y[i]) if y[i] in streetsH else -1
            a2 = streetsV.index(x[j]) if x[j] in streetsV else -1
            b2 = streetsH.index(y[j]) if y[j] in streetsH else -1

            da = abs(a2 - a1) if a1 != -1 and a2 != -1 else 10**9
            db = abs(b2 - b1) if b1 != -1 and b2 != -1 else 10**9
            dm = abs(x[i] - x[j]) + abs(y[i] - y[j])
            
            if dm < da and dm < db:
                ans += 1
    
    print(ans)

for _ in range(int(input())):
    solve()
