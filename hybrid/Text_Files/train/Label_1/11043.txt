#include <bits/stdc++.h>
using namespace std;
#define endl '\n'

typedef long long ll;

int main() {
  int n;
  cin >> n;
  ll arr[n];
  unordered_map<ll, vector<int>> m;
  for (int i = 0; i < n; i++) {
    ll x;
    cin >> x;
    arr[i] = x;
    m[x].push_back(i);
  }
  if (m.size() == 1) {
    cout << "YES" << endl;
    cout << "0" << endl;
  } else if (m.size() == 2) {
    ll x = m.begin()->first;
    ll y = (++m.begin())->first;
    auto &xv = m[x], &yv = m[y];
    if (xv.size() > 1 && yv.size() > 1) {
      cout << "YES" << endl;
      cout << "1" << endl;
      cout << xv[0] + 1 << " " << xv[1] + 1 << " " << yv[1] + 1 << endl;
    } else {
      cout << "NO" << endl;
    }
  } else {
    cout << "NO" << endl;
  }
}
