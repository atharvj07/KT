def MOD(x): return x % (10 ** 9 + 7)
def calc(state, idx, dep, dist):
    if dep == 0: return 1
    if state & 1 << idx: return 0
    if idx + 1 < dep:
        if state & 1 << idx + 1 and (state & 1 << idx + 1) != (state & 1 << idx):
            return 0
    if idx - 1 >= 0:
        if state & 1 << idx - 1 and (state & 1 << idx - 1) != (state & 1 << idx):
            return 0
    return (
        calc(MOD(state + 2 << idx), idx + 1, dep - 1, dist + 'r') +
        calc(MOD(state + 2 << idx), idx + 1, dep - 1, dist + 'b') +
        calc(MOD(state + 2 << idx), idx + 1, dep - 1, dist + 'g') +
        calc(MOD(state + 2 << idx), idx + 1, dep - 1, dist + 'y') +
        calc(MOD(state + 2 << idx), idx + 1, dep - 1, dist + 'o') +
        calc(MOD(state + 2 << idx), idx + 1, dep - 1, dist + 'w')
    )
def main():
    k = int(input())
    print(calc(0, 0, k, ''))
main()
