n, w, h = map(int, input().split())
dancers = []
for _ in range(n):
    g, p, t = map(int, input().split())
    # Normalize positions: All dancers now appear to start from the bottom-left corner
    # and move diagonally to the top-right. This is achieved by adjusting their start
    # positions by their wait times, thus aligning their effective start times.
    if g == 1:
        dancers.append((p - t, 0, g, p))
    else:
        dancers.append((0, p - t, g, p))

# Sort dancers based on their normalized start positions
dancers.sort()

# Since dancers now effectively move diagonally, their final positions can be determined
# directly by their order and the dimensions of the stage. The key insight is that
# collisions only swap their directions, not their order in the movement line.
for i, (_, _, g, p) in enumerate(dancers):
    if g == 1:
        # Vertical movers end on the right or top edge
        print(f"{min(w, p + i)} {min(h, h - (p + i - w)) if p + i > w else h}")
    else:
        # Horizontal movers end on the top or right edge
        print(f"{min(w, w - (p + i - h)) if p + i > h else w} {min(h, p + i)}")
