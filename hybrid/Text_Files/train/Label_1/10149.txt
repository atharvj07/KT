#include<cstdio>
#include<vector>
#include<cstring>
using namespace std;
int n, k, a[5050][5050], b[5050];
vector<int> u[5050];
char s[5050], t[5050];
int main(){
    scanf("%d%d", &k, &n);
    for (int i = 0; i < k; i++)
        scanf("%s", s + 1),
        a[i][0] = n,
        a[i][n + 1] = -1;
    for (int i = 1; i <= n; i++)
        for (int j = 0; j < k; j++)
            if (s[i] == s[i + 1]){
                a[j][i] = a[j][i + 1];
                a[j][a[j][i]] = i + 1;
            }
            else{
                a[j][i] = b[j];
                b[j] = i;
            }
    if (b[k - 1]) puts("-1");
    else{
        for (int j = 0; j < k; j++){
            int i = a[j][0];
            while (i){
                t[i] = s[i];
                i = a[j][i];
            }
        }
        puts(t + 1);
    }
}
