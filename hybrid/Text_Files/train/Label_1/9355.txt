#include <iostream>
#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, c, d, beauty = 0;
  cin >> n >> c >> d;
  priority_queue<pair<int, char>> coins;
  priority_queue<pair<int, char>> diamonds;
  while (n--) {
    int b, p;
    char type;
    cin >> b >> p >> type;
    if (type == 'C') coins.push({b - p, type});
    if (type == 'D') diamonds.push({b - p, type});
  }
  while (c > 0 && !coins.empty()) {
    c -= coins.top().first;
    beauty += coins.top().second == 'C' ? coins.top().second - coins.top().first : coins.top().second;
    coins.pop();
  }
  while (d > 0 && !diamonds.empty()) {
    d -= diamonds.top().first;
    beauty += diamonds.top().second == 'D' ? diamonds.top().second - diamonds.top().first : diamonds.top().second;
    diamonds.pop();
  }
  if (beauty > 0) cout << beauty << endl;
  else cout << 0 << endl;
}
