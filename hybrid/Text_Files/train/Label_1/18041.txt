#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    vector<unordered_set<int>>stops(n);
    for(auto &x:stops){
        int cnt;
        cin>>cnt;
        for(int i=0;i<cnt;i++){
            int line;
            cin>>line;
            x.insert(line);
        }
    }
    unordered_set<int>ans;
    for(auto x:stops)
        for(auto y:x)
            ans.insert(y);
    for(auto &x:ans)
        cout<<x<<" ";
    return 0;
}
