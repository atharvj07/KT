s = input()

count_0 = s.count('0')
count_1 = s.count('1')
count_2 = s.count('2')

# Since we can move 1s past 0s and 2s past 1s, but not 2s past 0s,
# we can put all the 0s at the beginning and all the 2s at the end.
# The 1s will then naturally fall in the middle.
# This way, the string is lexicographically smallest.

result = '0' * count_0 + '1' * count_1 + '2' * count_2
print(result)
