#include <bits/stdc++.h>
using namespace std;
const int maxn=105;
int n,a[maxn];
int main(){
    cin>>n;
    for(int i=1;i<=n;++i)cin>>a[i];
    int mx=*max_element(a+1,a+n+1);
    cout<<mx<<'\n';
    int ans=0;
    for(int i=1;i<=n;++i)
        if(a[i]<mx)++ans;
    cout<<ans<<'\n';
    for(int i=1;i<=ans;++i){
        for(int j=1;j<=n;++j)
            if(a[j]<mx)cout<<1,a[j]++;
            else cout<<0;
        cout<<'\n';
    }
    return 0;
}
