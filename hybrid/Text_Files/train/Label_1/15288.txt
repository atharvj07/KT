#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2e5;

int main() {
    long long a[MAXN];
    long long dp[MAXN];
    long long l[MAXN];
    long long r[MAXN];
    int n, m;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    for (int i = 0; i < n; i++) {
        dp[i] = a[i];
    }
    l[0] = 1;
    r[n - 1] = n;
    for (int i = 1; i < n; i++) {
        if (a[i] >= a[i - 1]) {
            l[i] = max(l[i], l[i - 1] + 1);
        }
    }
    for (int i = n - 2; i >= 0; i--) {
        if (a[i] <= a[i + 1]) {
            r[i] = min(r[i], r[i + 1] - 1);
        }
    }
    cin >> m;
    for (int i = 0; i < m; i++) {
        int k, x;
        cin >> k >> x;
        k--;
        x--;
        long long res;
        if (x < l[k]) {
            res = a[x];
        } else if (x > r[k]) {
            res = dp[n - 1 - x + k];
        } else {
            res = dp[x - k];
        }
        cout << res << endl;
    }
    return 0;
}
