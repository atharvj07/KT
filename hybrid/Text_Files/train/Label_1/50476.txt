import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] ans = new int[1000001];
        for (int i = 1; i <= 1000000; i++) {
            int tmp = i, sum = 0;
            while (tmp > 0) {
                sum += tmp % 10;
                tmp /= 10;
            }
            for (int j = 1;; j++) {
                int x = i + j;
                tmp = x;
                sum = 0;
                while (tmp > 0) {
                    sum += tmp % 10;
                    tmp /= 10;
                }
                if (gcd(x, sum) > 1) {
                    ans[i] = x;
                    break;
                }
            }
        }
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int x = sc.nextInt();
            System.out.println(ans[x]);
        }
    }
    static int gcd(int a, int b) {
        if (b == 0)
            return a;
        return gcd(b, a % b);
    }
}
