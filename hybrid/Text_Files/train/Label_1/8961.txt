#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 10;

int n, ans;
int a[MAXN], b[MAXN], sz[MAXN];

int main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> sz[i];
    for (int j = 1; j <= sz[i]; ++j) { cin >> a[j]; }
    for (int j = 1; j <= sz[i]; ++j) { cin >> b[j]; }
    for (int j = 1; j <= sz[i]; ++j) {
      for (int k = 1; k <= sz[i]; ++k) {
        if (a[j] < b[k]) ++ans;
        if (b[j] < a[k]) ++ans;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
