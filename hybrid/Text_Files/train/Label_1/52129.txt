import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        String t = sc.next();

        int diff = 0;
        int[] pos = new int[n];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) != t.charAt(i)) {
                diff++;
                pos[diff - 1] = i;
            }
        }

        if (diff % 2 == 1) {
            System.out.println(-1);
            return;
        }

        System.out.println(diff / 2);
        for (int i = 0; i < diff / 2; i++) {
            System.out.println((pos[i] + 1) + " " + (pos[diff - i - 1] + 1));
        }
    }
}
