#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> v(n);
        for (int i = 0; i < n; i++) cin >> v[i];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (v[i] > 1) ans++;
            while (v[i] > 1) v[i] = (v[i] + 1) / 2;
        }
        cout << ans << endl;
    }
    return 0;
}
