n = int(input())
a = list(map(int, input().split()))
block_sums = {}
for l in range(n):
    current_sum = 0
    for r in range(l, n):
        current_sum += a[r]
        if current_sum not in block_sums:
            block_sums[current_sum] = []
        block_sums[current_sum].append((l+1, r+1))

max_blocks = []
for _, blocks in block_sums.items():
    blocks.sort(key=lambda x: x[1])
    non_intersecting_blocks = [blocks[0]]
    for current in blocks[1:]:
        if current[0] > non_intersecting_blocks[-1][1]:
            non_intersecting_blocks.append(current)
    if len(non_intersecting_blocks) > len(max_blocks):
        max_blocks = non_intersecting_blocks

print(len(max_blocks))
for block in max_blocks:
    print(block[0], block[1])
