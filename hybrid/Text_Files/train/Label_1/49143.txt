import java.util.Scanner;

public class Main {
    static int mod = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        long[][] dp = new long[n][202];
        dp[0][arr[0] == -1 ? 0 : arr[0]] = 1;
        for (int i = 1; i < n; i++) {
            for (int j = 1; j <= 200; j++) {
                if (arr[i] == -1) {
                    for (int k = j; k <= 200; k++) {
                        dp[i][j] = (dp[i][j] + dp[i - 1][k]) % mod;
                    }
                } else if (arr[i] != -1) {
                    if (arr[i] != 1) {
                        for (int k = j; k < arr[i]; k++) {
                            dp[i][j] = (dp[i][j] + dp[i - 1][k]) % mod;
                        }
                    }
                    dp[i][j] = (dp[i][j] + dp[i - 1][arr[i]]) % mod;
                    if (arr[i] != 200) {
                        for (int k = arr[i] + 1; k <= 200; k++) {
                            dp[i][j] = (dp[i][j] + dp[i - 1][k]) % mod;
                        }
                    }
                }
            }
        }
        System.out.println(dp[n - 1][1]);
    }
}
