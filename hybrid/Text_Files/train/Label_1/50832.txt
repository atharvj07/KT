import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] degree = new int[n + 1];
        ArrayList<Integer>[] edges = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            edges[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            edges[u].add(v);
            edges[v].add(u);
            degree[u]++;
            degree[v]++;
        }
        int[] res = new int[n];
        int idx = 0;
        Queue<Integer> q = new LinkedList<>();
        for (int i = 1; i <= n; i++) {
            if (degree[i] == 1) {
                q.add(i);
            }
        }
        while (!q.isEmpty()) {
            int u = q.poll();
            res[idx++] = u;
            for (int v : edges[u]) {
                degree[v]--;
                if (degree[v] == 1) {
                    q.add(v);
                }
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(res[i] + " ");
        }
        System.out.println();
    }
}
