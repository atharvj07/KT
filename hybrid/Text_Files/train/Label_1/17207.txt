#include <iostream>
#include <cmath>
#include <vector>
#define MOD 100

int main() {
  long long n;
  std::cin >> n;

  long long ans = 1;
  n %= MOD - 1;
  for (long long i = 1; i <= n; i++) {
    ans = ans * 5 % MOD;
  }
  
  if (ans < 10) std::cout << '0';
  std::cout << ans << '\n';
  return 0;
}
