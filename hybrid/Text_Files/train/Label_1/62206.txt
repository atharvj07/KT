n = int(input())
colors = list(map(int, input().split()))

if colors[0] != colors[-1]:
    print(n-1)
else:
    for i in range(n):
        if colors[i] != colors[0]:
            left_most = i
            break
    for i in range(n-1, -1, -1):
        if colors[i] != colors[0]:
            right_most = i
            break
    print(max(right_most, n-1-left_most))
