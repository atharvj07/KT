#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main(){
    int n, d;
    cin >> n >> d;
    vector<int> p(n);
    for(int i = 0; i < n; ++i) cin >> p[i];
    sort(p.begin(), p.end());
    int ans = 0;
    for(int i = 1; i < n; ++i){
        if(p[i] - p[i - 1] > d){
            ++ans;
            i += d - 1;
        }
    }
    cout << ans << endl;
    return 0;
}
