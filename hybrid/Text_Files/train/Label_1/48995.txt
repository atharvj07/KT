import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNextInt()) {
            int n = sc.nextInt();
            System.out.println(calc(n));
        }
    }

    public static int calc(int n) {
        long ans = 0;
        for (int i = 0; i <= 1000; i++) {
            for (int j = 0; j <= 1000; j++) {
                for (int k = 0; k <= 1000; k++) {
                    if (i + j + k <= n) {
                        ans += 1;
                    }
                }
            }
        }
        return (int) ans;
    }
}
