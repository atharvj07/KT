#include <iostream>
#include <vector>

int main() {
  int n, x;
  std::cin >> n >> x;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) { std::cin >> a[i]; }
  int sum = 0;
  for (int i = 0; i < n; i++) { sum += a[i]; }
  if (sum == x) { std::cout << "YES" << std::endl; } else { std::cout << "NO" << std::endl; }
  return 0;
}
