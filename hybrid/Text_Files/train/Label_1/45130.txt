import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.List;
import java.util.Map;
import java.util.Queue;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]), m = Integer.parseInt(input[1]);
        int[] w = new int[n];
        String[] wInput = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            w[i] = Integer.parseInt(wInput[i]);
        }
        boolean[][] friends = new boolean[m][n];
        for (int i = 0; i < m; i++) {
            input = br.readLine().split(" ");
            int x = Integer.parseInt(input[0]) - 1, y = Integer.parseInt(input[1]) - 1;
            friends[i][x] = true;
            friends[i][y] = true;
        }
        br.close();
        Map<Integer, List<Integer>> graph = new HashMap<>();
        for (int i = 0; i < m; i++) {
            graph.put(i, new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            for (int j = i + 1; j < m; j++) {
                if (hasCommonFood(friends[i], friends[j]) && canEat(w, friends[i]) && canEat(w, friends[j])) {
                    graph.get(i).add(j);
                    graph.get(j).add(i);
                }
            }
        }
        int[] order = new int[m];
        if (bfs(graph, order)) {
            System.out.println("ALIVE");
            for (int i = 0; i < m; i++) {
                System.out.print(order[i] + 1 + " ");
            }
        } else {
            System.out.println("DEAD");
        }
    }

    private static boolean hasCommonFood(boolean[] a, boolean[] b) {
        for (int i = 0; i < a.length; i++) {
            if (a[i] && b[i]) {
                return true;
            }
        }
        return false;
    }

    private static boolean canEat(int[] w, boolean[] f) {
        for (int i = 0; i < f.length; i++) {
            if (f[i] && w[i] == 0) {
                return false;
            }
        }
        return true;
    }

    private static boolean bfs(Map<Integer, List<Integer>> graph, int[] order) {
        Queue<Integer> queue = new LinkedList<>();
        boolean[] visited = new boolean[graph.size()];
        queue.offer(0);
        visited[0] = true;
        int index = 0;
        while (!queue.isEmpty()) {
            int current = queue.poll();
            order[index++] = current;
            for (int neighbor : graph.get(current)) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    queue.offer(neighbor);
                }
            }
        }
        return index == graph.size();
    }
}
