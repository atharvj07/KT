#include<bits/stdc++.h>
using namespace std;
int main()
{
    string s;
    int a[10]={0};
    cin>>s;
    int l=s.length();
    int c=0;
    for(int i=0;i<l;i++)
    {
        int x=s[i]-'0';
        if(x==0&&i==0)
        {
            c=-1;
            break;
        }
        a[x]++;
    }
    if(c==-1)
    cout<<0<<endl;
    else
    {
        int ans=1;
        for(int i=1;i<=9;i++)
        {
            if(a[i]>0)
            {
                ans=ans*a[i];
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
