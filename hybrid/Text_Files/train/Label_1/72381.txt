from collections import defaultdict

n, q = map(int, input().split())
busy = set()
ans = []
tasks = sorted(zip(*[map(int, input().split()) for _ in range(q)]))
for t, k, d in tasks:
    se = set(i for i in range(1, n+1) if i not in busy)
    if len(se) >= k:
        ans.append(sum(se[:k]))
        busy.update(se[:k])
    else:
        ans.append(-1)
    busy.difference_update(range(t, t+d))
print(*ans, sep="\n")
