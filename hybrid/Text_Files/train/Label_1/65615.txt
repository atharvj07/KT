n, k = map(int, input().split())
a = [[0] * n for _ in range(n)]
s = []
for i in range(n):
    t = input()
    for j in range(n):
        if t[j] == '#': a[i][j] = -1
        else: s.append((i, j))
ans = 0
for i, j in s:
    t = 0
    for x in range(-k, k):
        if 0 <= i+x < n and a[i+x][j] != -1: t += 1
        if 0 <= j+x < n and a[i][j+x] != -1: t += 1
    if t > ans:
        ans = t
        ans_i, ans_j = i+1, j+1
print(ans_i, ans_j)
