#include <bits/stdc++.h>

#define ll long long
using namespace std;
typedef pair<ll, ll> pi;

#define F first
#define S second

const ll MX = 2e5 + 5;
const ll inf = 1e18;

ll n;
ll arr[MX];

void solve() {
    cin >> n;
    for (ll i = 1; i <= n; i++) cin >> arr[i];

    ll ans = 0;
    for (ll i = 1; i <= n; i++) ans += (arr[i] != i);
    ans /= 2;
    cout << ans << '\n';
}

int main() {
    ll t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
