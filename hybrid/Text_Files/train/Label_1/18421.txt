#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod=1000000007;
int main(){
    ll n,m;
    cin>>n>>m;
    bitset<1005> p,q;
    for(int i=0;i<m;++i){
        int x;
        cin>>x;
        p[x]=1;
    }
    q[1]=1;
    for(int i=2;i<=n;++i){
        if(!p[i]){
            ll k=(q[i-1]+q[i-2])%mod;
            q[i]=k;
        }
    }
    cout<<q[n]<<endl;
    return 0;
}
