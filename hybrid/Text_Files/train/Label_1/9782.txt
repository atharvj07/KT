#include <iostream>
#include <vector>

struct seg {
    long long l, r;
};

using namespace std;

int main() {
    long long n, m, i, j, cur, ans = 0;
    long long minu = 1e9, maxu = -1e9;
    vector<seg> v;
    cin >> n >> m;
    vector<long long> l(n);
    for (i = 0; i < n; i++) {
        cin >> l[i];
        minu = min(minu, l[i]);
        maxu = max(maxu, l[i]);
    }
    for (i = 0; i < m; i++) {
        seg x;
        cin >> x.l >> x.r;
        v.push_back(x);
    }
    for (i = 0; i < m; i++) {
        cur = l[v[i].l - 1];
        for (j = v[i].l; j <= v[i].r; j++) {
            cur = min(cur, l[j - 1]);
            ans = max(ans, l[j - 1] - cur);
        }
    }
    cout << ans << endl;
    cout << 0 << endl;
    return 0;
}
