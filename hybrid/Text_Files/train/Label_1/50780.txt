import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt(), k = cin.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = cin.nextInt();
        System.out.println(f(a, k) ? "YES" : "NO");
        if (f(a, k)) {
            for (int[] ai : ans) System.out.printf("%d %d\n", ai[0], ai[1]);
        }
    }

    static List<int[]> ans = new LinkedList<>();

    static boolean f(int[] a, int k) {
        int n = a.length;
        int[] b = Arrays.copyOf(a, n);
        Arrays.sort(b);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < i; j++) {
                int x = b[j], y = b[i];
                if (f(a, k, x, y)) return true;
            }
        }
        return false;
    }

    static boolean f(int[] a, int k, int x, int y) {
        for (int i = 0; i < a.length; i++) {
            for (int j = 0; j < i; j++) {
                if (a[i] == x && a[j] == y) {
                    a[i] = f(x, y);
                    a[j] = -1;
                    ans.add(new int[] { x, y });
                    return true;
                }
            }
        }
        return false;
    }

    static int f(int x, int y) {
        return x % 3 == 0 || y % 3 == 0 ? x + y : x + y - x % 3;
    }
}
