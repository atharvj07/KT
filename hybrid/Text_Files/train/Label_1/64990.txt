from sys import stdin

def solve(a, q):
    pos = [[]] * (1<<17)
    for i in range(len(a)):
        pos[a[i]].append(i)
    for x,y in q:
        x-=1;y-=1
        ans = 'Fou'
        for j in range(17):
            if not x&1<<j or not y&1<<j:
                continue
            k = x|1<<j
            idx = pos[k].binary_search(y,key=lambda z:z|k==k)
            if idx and idx>=0:
                ans = 'Shi'
                break
        print(ans)

n,q = map(int,stdin.readline().split())
a = list(map(int,stdin.readline().split()))
q = [list(map(int,stdin.readline().split())) for _ in range(q)]

solve(a,q)
