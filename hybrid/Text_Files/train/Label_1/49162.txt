import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long n = input.nextLong();
        long p = input.nextLong();
        if (p < 0) {
            long bits = -1L, t = p;
            while (t < 0) {
                bits++;
                t <<= 1;
            }
            if ((n & (1L << bits)) != 0) {
                System.out.println(-1);
                return;
            }
            n ^= (1L << bits);
            p = -p;
        }
        long ans = 0;
        while (n > 0) {
            long pw = 1L;
            while (p * pw < n) {
                pw <<= 1;
            }
            n -= p * pw;
            ans++;
        }
        System.out.println(ans);
    }
}
