#include <iostream>
#include <vector>
using namespace std;

int main() {
    long long n;
    cin >> n;

    long long ans = 0;
    for (long long a = 1; a <= n; a++) {
        long long max_b = a;
        long long min_c = a + 1;
        while (max_b * max_b + a * a <= n * n) {
            max_b++;
        }
        while (min_c * min_c < max_b * max_b + a * a) {
            min_c++;
        }
        ans += max_b - a;
    }
    cout << ans << endl;
}
