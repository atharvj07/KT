#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, t;
  cin >> n >> t;
  vector<pair<int, int>> buses;
  for (int i = 0; i < n; i++) {
    int s, d;
    cin >> s >> d;
    int next_bus = s;
    while (next_bus <= t) {
      buses.push_back({next_bus, i + 1});
      next_bus += d;
    }
  }
  sort(buses.begin(), buses.end());
  if (buses.size() > 0) {
    cout << buses[0].second << endl;
  }
  return 0;
}
