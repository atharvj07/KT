def day_of_week_to_number(day):
    days = ["monday", "tuesday", "wednesday", "thursday", "friday", "saturday", "sunday"]
    return days.index(day)

first_day = input().strip()
second_day = input().strip()

first_day_num = day_of_week_to_number(first_day)
second_day_num = day_of_week_to_number(second_day)

# Days difference in a non-leap year for months in sequence
possible_days_diff = [0, 3, 2, 3, 2, 3, 3, 2, 3, 2, 3]

possible = "NO"
for days_diff in possible_days_diff:
    if (first_day_num + days_diff) % 7 == second_day_num:
        possible = "YES"
        break

print(possible)
