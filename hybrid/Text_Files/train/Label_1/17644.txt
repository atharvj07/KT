#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  sort(a.rbegin(), a.rend());
  int ans = 0, x = 1;
  for (int v : a) {
    ans += (x++) * v;
  }
  cout << ans << endl;
  for (int i = 0; i < n; i++) cout << i + 1 << " ";
  cout << endl;
  return 0;
}
