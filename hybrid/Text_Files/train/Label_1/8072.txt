#include <bits/stdc++.h>
using namespace std;

int main() {
  int n; cin >> n;
  vector<int> p(n), v(n);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
    string s; cin >> s;
    for (char c : s) {
      if (c == 'A') v[i] |= 1;
      else if (c == 'B') v[i] |= 2;
      else v[i] |= 4;
    }
  }
  int ans = 1e9;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      for (int k = j + 1; k < n; k++) {
        if ((v[i] | v[j] | v[k]) == 7) {
          ans = min(ans, p[i] + p[j] + p[k]);
        }
      }
    }
  }
  if (ans == 1e9) cout << -1 << "\n";
  else cout << ans << "\n";
  return 0;
}
