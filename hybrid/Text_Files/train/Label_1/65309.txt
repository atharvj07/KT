import sys

n, s = input(), list(map(int, input().split()))
q = int(input())

for l, r in zip(*[iter(map(int, sys.stdin.readline().split())) for _ in range(q)]*2):
    cur, p = s[l-1:r], 0
    for i in range(1<<len(cur)):
      if i&1: p += cur[i>>1&1] + cur[i>>1] >= 10
      else: cur[i>>1] = (cur[i>>1] + cur[i>>1&1]) % 10
    print(p)
