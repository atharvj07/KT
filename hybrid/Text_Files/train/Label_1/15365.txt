#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<ll> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i]; }
    if (n % 2 == 1 && k % 2 == 0) {
      cout << "no\n";
      continue;
    }
    ll sum = 0;
    for (int i = 0; i < n; i++) { sum += v[i]; }
    if (sum % 2 == 1 && k % 2 == 0) {
      cout << "no\n";
      continue;
    }
    if (sum % n != k) {
      cout << "no\n";
      continue;
    }
    cout << "yes\n";
  }
  return 0;
}
