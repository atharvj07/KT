#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct elem {
	int x, y, v, idx;
	bool operator<(const elem &o) const {
		return v < o.v;
	}
};
vector<elem> A;

int main() {
	int n;
	cin >> n;
	A.resize(n);
	for (int i = 0; i < n; i++) {
		int x;
		cin >> x;
		A[i] = { i / 300, i % 300, x, i };
	}
	sort(A.begin(), A.end());
	int ans = 0;
	vector<int> X;
	for (int i = 0; i < n; i++) {
		int nx = A[i].x;
		int ny = A[i].y;
		if (i && nx == A[i - 1].x && ny == A[i - 1].y) continue;
		int cnt = 0;
		for (int j = 0; j < n; j++) {
			if (nx == A[j].x && ny == A[j].y) cnt++;
		}
		if (cnt > ans) {
			X.clear();
			ans = cnt;
			for (int j = 0; j < n; j++) {
				if (nx == A[j].x && ny == A[j].y) X.push_back(A[j].idx);
			}
		}
	}
	cout << ans << endl;
	cout << (int)X.size() / ans << ' ' << ans << endl;
	while (X.size()) {
		for (int i = 0; i < ans; i++) {
			cout << X.back() + 1 << ' ';
			X.pop_back();
		}
		cout << endl;
	}
}
