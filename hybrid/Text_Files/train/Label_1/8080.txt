#include<bits/stdc++.h>
#define forr(i,a,b) for(int i=(a);i<(b);++i)
using namespace std;
const int alph=26;
const int maxN=2005;
const int mod=1e9+7;
int n,m,f[maxN][maxN],suf[maxN],psuf[maxN];
int main(){
    string s,p;
    cin>>s>>p;
    n=s.size();
    m=p.size();
    forr(i,0,n){
        int t=i?f[i-1][m]:0;
        suf[i]=((s[i]==p[m-1])?t+1:t);
    }
    reverse(p.begin(),p.end());
    forr(i,0,n){
        int t=(i?psuf[i-1]:0);
        psuf[i]=((s[i]==p[m-1])?t+1:t);
    }
    f[n][m]=0;
    for(int i=n-1;~i;--i){
        f[i][m]=f[i+1][m];
        forr(j,0,m-1){
            if(s[i]==p[j])
                f[i][m]=max(f[i][m],suf[i+m-j-1]+1);
        }
    }
    for(int j=m-1;~j;--j){
        f[n][j]=f[n][j+1];
        forr(i,0,n){
            if(s[i]==p[j])
                f[n][j]=max(f[n][j],f[i+1][j+1]);
        }
    }
    forr(i,0,n+1){
        int res=0;
        forr(j,0,m)
            res=max(res,f[i][j]);
        cout<<res<<' ';
    }
    return 0;
}
