#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
const int MAXN = 105;
vector<int> a, b;
int main() {
    int n, m;
    cin >> n >> m;
    a.resize(n); b.resize(m);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < m; i++) cin >> b[i];
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    int ans = -1;
    for (int i = 1; i <= 100; i++) {
        bool ok = true;
        for (int j = 0; j < n; j++)
            if (a[j] <= i)
                ok = false;
        for (int j = 0; j < m; j++)
            if (b[j] > i)
                ok = false;
        if (ok) {
            ans = i;
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
