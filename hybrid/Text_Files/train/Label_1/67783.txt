from sys import stdin
input = stdin.readline

def lis(a):
    n = len(a)
    b = [1] * n
    inc_sub = [-1] * n
    for i in range(n):
        for j in range(i):
            if a[i] > a[j] and b[i] < b[j] + 1:
                b[i] = b[j] + 1
                inc_sub[i] = j
    return max(b), inc_sub

def solve():
    n = int(input())
    a = list(map(int, input().split()))
    l, inc = lis(a)
    if l%2 == 0:
        print('YES')
    else:
        flag = False
        for i in range(n):
            if inc[i] != -1 and (l-1)%2 == b[inc[i]]%2:
                flag = True
        if flag: print('YES')
        else: print('NO')
    print()

for _ in range(int(input())):
    solve()
