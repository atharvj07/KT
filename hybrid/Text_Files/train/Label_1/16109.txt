#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll n, m, c_l, c_e, v, a[10000005], b[10000005], l[10000005], r[10000005], ans[10000005];
int main() {
  cin >> n >> m >> c_l >> c_e >> v;
  for (ll i = 1; i <= n; i++) {
    a[i] = 1000000000000;
    b[i] = 1000000000000;
  }
  for (ll i = 1; i <= c_l; i++) {
    cin >> l[i];
  }
  for (ll i = 1; i <= c_e; i++) {
    cin >> r[i];
  }
  for (ll i = 1; i <= c_l; i++) {
    ll p = l[i];
    for (ll j = p; j <= n; j += v) {
      a[j] = min(a[j], p - 1);
    }
  }
  for (ll i = 1; i <= c_e; i++) {
    ll p = r[i];
    for (ll j = p; j <= n; j += v) {
      b[j] = min(b[j], p - 1);
    }
    for (ll j = p; j >= 1; j -= v) {
      b[j] = min(b[j], p - 1);
    }
  }
  for (ll i = 2; i <= n; i++) {
    a[i] = min(a[i], a[i - 1] + 1);
    b[i] = min(b[i], b[i - 1] + 1);
  }
  ll q;
  cin >> q;
  for (ll i = 1; i <= q; i++) {
    ll x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2;
    if (b[x2] != 1000000000000) {
      ans[i] += b[x2] + 1 + (abs(y2 - r[1]));
    } else {
      ans[i] += a[x2] + 1 + (abs(y2 - l[1]));
    }
    if (a[x1] != 1000000000000) {
      ans[i] += a[x1] + 1 + (abs(y1 - l[1]));
    } else {
      ans[i] += b[x1] + 1 + (abs(y1 - r[1]));
    }
    ans[i] -= 1;
    cout << ans[i] << '\n';
  }
}
