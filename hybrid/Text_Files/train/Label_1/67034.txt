MOD = 10**9 + 7

def solve(n, m, segments):
    # Bitwise OR of all segments
    all_or = 0
    for _, _, x in segments:
        all_or |= x
        
    # Calculate coziness
    coziness = all_or * pow(2, n-1, MOD)
    return coziness % MOD

t = int(input().strip())
for _ in range(t):
    n, m = map(int, input().strip().split())
    segments = [list(map(int, input().strip().split())) for _ in range(m)]
    print(solve(n, m, segments))
