#include <iostream>
#include <set>
#include <vector>

using namespace std;

typedef long long LL;

int main() {
  LL n, t;
  cin >> n >> t;
  vector<pair<LL, LL>> route(n);
  for (LL i = 0; i < n; ++i) { cin >> route[i].first >> route[i].second; }

  set<LL> t_arr;
  for (LL i = 0; i < n; ++i) {
    for (LL t_tmp = route[i].first; t_tmp <= t; t_tmp += route[i].second) {
      t_arr.insert(t_tmp);
    }
  }

  for (auto t_tmp : t_arr) {
    if (t <= t_tmp) {
      for (LL i = 0; i < n; ++i) {
        if (route[i].first <= t_tmp && (t_tmp - route[i].first) % route[i].second ==
                                           0) {
          cout << i + 1 << endl;
          return 0;
        }
      }
    }
  }

  return 0;
}
