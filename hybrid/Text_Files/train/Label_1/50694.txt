import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static long[] energy;
    static long[][] graph;
    static long[][] dis;
    static int[] visited;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        energy = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            energy[i] = sc.nextInt();
        }
        graph = new long[n + 1][n + 1];
        dis = new long[n + 1][n + 1];
        visited = new int[n + 1];
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            graph[a][b] = graph[b][a] = 1;
        }
        Arrays.fill(dis[1], Long.MAX_VALUE);
        dis[1][1] = 0;
        dp(1);
        long ans = Long.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            ans = Math.min(ans, dis[1][i]);
        }
        System.out.println(ans);
    }

    public static void dp(int now) {
        visited[now] = 1;
        for (int i = 1; i <= energy.length - 1; i++) {
            if (graph[now][i] != 0) {
                if (visited[i] == 0) {
                    dis[1][i] = Math.min(dis[1][i], dis[1][now] + energy[now]);
                    dp(i);
                }
                dis[1][i] = Math.min(dis[1][i], dis[1][now]);
            }
        }
        visited[now] = 0;
    }
}
