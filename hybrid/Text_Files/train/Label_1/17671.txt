#include <iostream>
#include <unordered_map>
#include <unordered_set>
using namespace std;

int main() {
  int N;
  cin >> N;

  unordered_map<string, unordered_set<string>> types;
  while (N--) {
    string name, type;
    cin >> name >> type;
    types[name].insert(type);
  }

  int count = 0;
  for (auto &[_, typeset] : types) {
    count += typeset.size() == 1;
  }
  cout << count << endl;
  return 0;
}
