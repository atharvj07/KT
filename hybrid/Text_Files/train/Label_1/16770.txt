#include <bits/stdc++.h>
using namespace std;
void solve() {
  int l, r;
  cin >> l >> r;
  int x = l;
  while (1) {
    bool ok = true;
    set<int> st;
    while (x > 0) {
      int d = x % 10;
      if (st.find(d) != st.end()) ok = false;
      st.insert(d);
      x /= 10;
    }
    x = l;
    if (ok) break;
    l++;
  }
  if (l <= r)
    cout << l;
  else
    cout << -1;
}
int main() {
  solve();
  return 0;
}
