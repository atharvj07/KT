n = int(input())
l = []
for _ in range(n):
  l.append(list(map(int, input().split()[1:])))

c = 0

for i in range(n):
  for j in range(n):
    if i == j:
      for a in range(len(l[i])):
        for b in range(a+1, len(l[i])):
          if l[i][a] < l[i][b]:
            c += 1
    else:
      for a in range(len(l[i])):
        for b in range(len(l[j])):
          if l[i][a] < l[j][b]:
            c += 1

print(c)
