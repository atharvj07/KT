#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    vector<int> ans;
    for (int i = 0; i + 2 < s.size(); i++) {
      if ((s[i] == 'o' && s[i + 1] == 'n' && s[i + 2] == 'e') ||
          (s[i] == 't' && s[i + 1] == 'w' && s[i + 2] == 'o'))
        ans.push_back(i + 2);
    }
    cout << ans.size() << '\n';
    for (int p : ans) cout << p << ' ';
    cout << '\n';
  }
  return 0;
}
