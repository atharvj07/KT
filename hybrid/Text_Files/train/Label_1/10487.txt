#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  string s;
  cin >> n >> s;
  int cnt[10] = {};
  for (int i = 0; i < n; ++i) cnt[s[i] - '0']++;
  int ans = min({cnt[8], cnt[0], cnt[1], cnt[2], cnt[3], cnt[4], cnt[5], cnt[6], cnt[7], cnt[9]});
  cout << ans << endl;
  return 0;
}
