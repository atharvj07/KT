import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, k = map(int, input().split())
    s = input()
    l = [0] * n
    for i in range(k, n):
        if s[i] == '1':
            l[i] = i - k - 1
        else:
            l[i] = l[i - 1]
    r = [n] * n
    for i in range(n - k - 1, -1, -1):
        if s[i] == '1':
            r[i] = i + k + 1
        else:
            r[i] = r[i + 1]
    ans = 0
    for i in range(n):
        if s[i] == '0' and (i < r[i] or i > l[i]):
            ans += 1
    print(ans)
