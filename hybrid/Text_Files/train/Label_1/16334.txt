#include <iostream>
#include <string>

using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  int n = a.size(), m = b.size(), l1 = 0, l2 = 0;
  while (l1 < n && l2 < m) {
    if (a[l1] == b[l2]) l2++;
    l1++;
  }
  if (l2 == m) {
    for (int i = 0; i < l1; i++) cout << a[i];
  } else {
    cout << "-";
  }
  cout << endl;
  return 0;
}
