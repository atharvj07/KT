#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> grid(n, vector<int>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) { cin >> grid[i][j]; }
  }
  bool good = true;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (grid[i][j] != 1) {
        bool found = false;
        for (int k = 0; k < n; k++) {
          if (k != j && grid[i][j] == grid[i][k] + grid[j][j]) {
            found = true;
            break;
          }
        }
        if (!found) { good = false; }
      }
    }
  }
  cout << (good ? "Yes" : "No") << endl;
}
