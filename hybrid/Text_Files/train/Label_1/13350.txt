#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    bool isPossible=true;
    vector<pair<long long,long long>> adj[n+1];
    for(int i=1;i<=n;i++){
        long long c,r;
        cin>>c>>r;
        if(c>r){
            isPossible=false;
        }
    }
    for(int i=1;i<n;i++){
        int x,y;
        cin>>x>>y;
        x--;
        y--;
        adj[y].push_back({x,1});
        adj[x].push_back({y,y});
    }
    if(isPossible){
        queue<pair<long long,long long>> Q;
        Q.push({1,1});
        vector<long long> vis(n+1,0);
        while(!Q.empty()){
            pair<long long,long long> p=Q.front();
            Q.pop();
            vis[p.first]=1;
            for(auto u:adj[p.first]){
                if(!vis[u.first]){
                    Q.push({u.first,p.second*u.second});
                }
            }
        }
        for(int i=1;i<=n;i++){
            if(vis[i]==0){
                isPossible=false;
            }
        }
    }
    if(isPossible)cout<<"YES"<<endl;
    else cout<<"NO"<<endl;
}
