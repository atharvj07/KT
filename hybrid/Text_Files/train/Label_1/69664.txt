import sys

input = []
for line in sys.stdin:
    input.append(line[:-1])

q = int(input[0])
queries = []
for i in range(1, q + 1):
    queries.append([int(x) for x in input[i].split()])

for query in queries:
    n, m, k = query
    if k < abs(n) + abs(m):
        print(-1)
    else:
        k -= abs(n) + abs(m)
        if k % 2 == 0:
            print(abs(n) + abs(m) + min(n, m) + min(k // 2, max(n, m) - min(n, m)))
        else:
            print(-1)
