#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, x;
    cin >> n >> x;
    int a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    unordered_set<int> st;
    for (int i = 0; i < n; i++) {
        int tmp = a[i] & x;
        if (st.count(tmp)) {
            cout << "0\n";
            return 0;
        }
        st.insert(a[i]);
    }
    cout << "-1\n";
    return 0;
}
