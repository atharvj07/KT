#include <bits/stdc++.h>
#define ll long long
using namespace std;

int n;
ll a[7001], b[7001];
map<ll, vector<int>> aa;

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) {
        cin >> b[i];
        aa[a[i]].push_back(i);
    }
    ll ans = 0;
    for (auto bt = aa.begin(); bt != aa.end(); bt++) {
        vector<int>& bb = bt->second;
        if (bb.size() >= 2) {
            ll s = 0;
            sort(bb.begin(), bb.end(), [&](int x, int y) { return b[x] > b[y]; });
            for (int i = 0; i < 2; i++) s += b[bb[i]];
            ans = max(ans, s);
        }
    }
    cout << ans << endl;
    return 0;
}
