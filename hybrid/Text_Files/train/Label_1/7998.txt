#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll a[100005], b[100005];
int main() {
  ll n, k, x;
  cin >> n >> k;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < 32; i++) {
    ll c = 0;
    for (int j = 0; j < n; j++) {
      ll z = a[j] >> (32 - i - 1);
      if (z % 2) c++;
    }
    if (c == k) {
      for (int j = 0; j < i; j++) cout << "1.";
      for (int j = i; j < 32; j++) cout << "0.";
      return 0;
    }
  }
  cout << -1 << endl;
  return 0;
}
