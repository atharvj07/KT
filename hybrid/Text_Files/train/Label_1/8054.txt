#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main()
{
    int T;
    cin >> T;
    while (T--)
    {
        int n;
        cin >> n;
        vector<int> a(n);
        int last;
        int cnt = 0;
        int max_dist = 0;
        for (int i = 0; i < n; i++)
        {
            cin >> a[i];
            if (i > 0)
            {
                if (a[i] == last)
                    cnt++;
                else
                {
                    max_dist = max(max_dist, cnt);
                    cnt = 1;
                }
            }
            last = a[i];
        }
        max_dist = max(max_dist, cnt);
        cout << max_dist << endl;
    }

    return 0;
}
