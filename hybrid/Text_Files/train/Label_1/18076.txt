#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> ans;
        for (int i = 2; i * i <= n; i++) {
            while (n % i == 0) {
                ans.push_back(i);
                n /= i;
            }
        }
        if (n > 1) ans.push_back(n);
        cout << ans.size() << endl;
        for (int x : ans) { cout << x << ' '; }
        cout << endl;
    }
    return 0;
}
