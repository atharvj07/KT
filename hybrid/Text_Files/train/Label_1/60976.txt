n = int(input())
a = list(map(int, input().split()))
b = sorted(a)
if a == b:
    print(0)
else:
    ans = []
    for i in range(n):
        for j in range(i + 1, n):
            if a[i] > a[j] and b[i] < b[j]:
                ans.append((i + 1, j + 1))
    print(len(ans))
    for p in ans:
        print(*p)
