def min_operations_to_convert_z_to_t(s, t):
    # Check if it's possible to form t from s
    for char in t:
        if char not in s:
            return -1
    # Initialize variables to track current positions in s and t, and the number of operations
    s_index, t_index, operations = 0, 0, 1
    while t_index < len(t):
        # If we've reached the end of s, but not t, increase operations and reset s_index
        if s_index == len(s):
            operations += 1
            s_index = 0
        # If the current character in s matches the current character in t, move to the next character in t
        if s[s_index] == t[t_index]:
            t_index += 1
        # Move to the next character in s
        s_index += 1
    return operations

T = int(input().strip())
for _ in range(T):
    s = input().strip()
    t = input().strip()
    print(min_operations_to_convert_z_to_t(s, t))
