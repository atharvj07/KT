#include <algorithm>
#include <cstdio>
#include <vector>

using namespace std;

struct Edge {
    int to, weight;
    int special;
};

const int MAXN = 2e5 + 5, MAXM = 2e5 + 5;

vector<Edge> adj[MAXN];
int n, m, k, a[MAXN], d[MAXN], special[MAXN];

void dijkstra(int start, int *d, int special[]) {
    for (int i = 1; i <= n; i++) { d[i] = 1e9; special[i] = 0; }
    d[start] = 0;

    for (int x = 1; x <= n; x++) {
        int v = -1;
        for (int i = 1; i <= n; i++) {
            if (!special[i] && (v == -1 || d[i] < d[v])) { v = i; }
        }
        if (v == -1) { break; }
        special[v] = 1;

        for (int i = 0; i < adj[v].size(); i++) {
            Edge &e = adj[v][i];
            if (d[e.to] > d[v] + e.weight) {
                d[e.to] = d[v] + e.weight;
            }
        }
    }
}

int main() {
    scanf("%d %d %d", &n, &m, &k);
    for (int i = 1; i <= k; i++) {
        scanf("%d", &a[i]);
    }
    while (m--) {
        int x, y;
        scanf("%d %d", &x, &y);
        adj[x].push_back({y, 1});
        adj[y].push_back({x, 1});
    }

    dijkstra(1, d, special);
    int ans = 0;
    for (int i = 2; i < k; i++) {
        ans = max(ans, d[a[i - 1]] + d[a[i + 1]]);
    }
    ans = max(ans, d[a[1]] + d[a[k]]);

    printf("%d\n", ans);
    return 0;
}
