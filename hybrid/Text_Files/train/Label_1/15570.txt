#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, i, m, j, a[105];
    cin >> n;
    for (i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1);
    for (i = n, m = 0; i > 0; i--) {
        if (a[i] == 0)
            continue;
        for (j = i - 1; j > 0; j--) {
            if ((a[j] > a[i] || j == i - 1) && a[j]) m++;
        }
        break;
    }
    cout << m + 1 << endl;
    return 0;
}
