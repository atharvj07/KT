def readBits(m):
    b = bin(int(input(), 2))[2:].zfill(m)
    return ('0' * m) if m == len(b) else b


n, m = map(int, input().split())
var = {}
for _ in range(n):
    v, s, o, w = input().split()
    if o == '?': o = readBits(m)
    var[v] = {'o': o, 't': v == w}

mn = mx = int('0b' + ('0' * m), 2)
for i in range(2 ** m):
    b = bin(i)[2:].zfill(m)
    t = 0; d = 1
    for v, c in var.items():
        if c['t']: t |= int(b, 2)
        elif c['o'] == b: t |= d
        elif c['o'] == '?': d = 0
    if d:
        mn = min(mn, t)
        mx = max(mx, t)

print(bin(mn)[2:].zfill(m))
print(bin(mx)[2:].zfill(m))
