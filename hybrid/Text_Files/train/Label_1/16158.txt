#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int j = 0;
    vector<int> v;
    for (int i = 0; i < n; i++) {
        if (s[i] == '*') {
            j++;
        } else {
            if (j >= 5) v.push_back(j);
            j = 0;
        }
    }
    if (j >= 5) v.push_back(j);
    cout << (v.size() >= 1 ? "yes" : "no") << endl;
}
