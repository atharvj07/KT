#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n;
        std::cin >> n;
        std::vector<long long> a(n);
        for (int i = 0; i < n; i++) {
            std::cin >> a[i];
        }
        long long ans = 0, sum = 0;
        bool neg = false;
        for (int i = 0; i < n; i++) {
            if ((neg && a[i] > 0) || (!neg && a[i] < 0)) {
                sum += a[i];
                neg = !neg;
            } else {
                ans = std::max(ans, sum);
                sum = a[i];
                neg = a[i] < 0;
            }
        }
        ans = std::max(ans, sum);
        std::cout << ans << std::endl;
    }
    return 0;
}
