n=int(input())
l=list(map(int,input().split()))
l.sort()
ans=0
for i in range(n):
    ans+=l.count(l[i])
    for j in range(i+1,n):
        if(l[j]==l[i]+1):
            ans+=l.count(l[i]+1)
        else:
            break
    for j in range(i-1,-1,-1):
        if(l[j]==l[i]-1):
            ans+=l.count(l[i]-1)
        else:
            break
print(ans)
