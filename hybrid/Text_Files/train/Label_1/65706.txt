from sys import stdin

for _ in range(int(stdin.readline())):
    n = int(stdin.readline())
    p = list(map(int, stdin.readline().split()))
    for i in range(n - 1):
        if p[i] > p[i + 1]:
            print(' '.join(map(str, p[: i + 2][::-1] + p[i + 2 : ])))
            break
    else:
        print(' '.join(map(str, p)))
