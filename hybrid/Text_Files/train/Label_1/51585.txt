import java.util.Scanner;

public class DISCO {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int prizeCoding = 0;
        if (x <= 3) {
            prizeCoding = (4 - x) * 100000;
        }

        int prizeRobot = 0;
        if (y <= 3) {
            prizeRobot = (4 - y) * 100000;
        }

        int bonus = 0;
        if (x == 1 && y == 1) {
            bonus = 400000;
        }

        int total = prizeCoding + prizeRobot + bonus;
        System.out.println(total);
    }
}
