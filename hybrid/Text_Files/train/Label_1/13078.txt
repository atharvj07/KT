#include <bits/stdc++.h>
using namespace std;
char s[210000],t[210000];
int n,x,a,b,d[26][26];
int main() {
    cin>>n>>s+1>>t+1;for(int i=1;i<=25;++i)for(int j=1;j<=25;++j)d[i][j]=i!=j?1e9:0;
    for(int i=1;i<=n;++i){if(s[i]!=t[i])a=s[i]-'a',b=t[i]-'a',d[a][b]=d[b][a]=1;}
    for(int i=0;i<=25;++i)for(int j=0;j<=25;++j)for(int k=0;k<=25;++k)d[i][j]=min(d[i][j],d[i][k]+d[k][j]);
    for(int i=1;i<=n;++i)for(int j=1;j<=n;++j)if(s[i]!=t[j])x=min(x,d[s[i]-'a'][t[j]-'a']);
    cout<<x<<endl;if(x<1e9)for(int i=1;i<=n;++i)for(int j=1;j<=n;++j)if(x==d[s[i]-'a'][t[j]-'a']){cout<<i<<' '<<j<<endl;return 0;}
    cout<<"-1 -1\n";return 0;}
