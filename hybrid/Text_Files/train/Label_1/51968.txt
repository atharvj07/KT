import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            int l = i, r = i;
            while (l >= 0 && r < n && a[l] == a[r]) {
                l--;
                r++;
            }
            l++;
            r--;
            int mid = (l + r) / 2;
            ans[i] = Math.max(i - mid, mid - i);
        }
        for (int i : ans) {
            System.out.print(i + " ");
        }
    }
}
