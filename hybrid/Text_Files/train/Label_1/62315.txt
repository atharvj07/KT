n = int(input())
a = list(map(int, input().split()))

def partition_days(events):
    in_office = set()
    partitions = []
    count = 0

    for event in events:
        if event > 0:
            if event in in_office:
                return -1
            in_office.add(event)
        else:
            if -event not in in_office:
                return -1
            in_office.remove(-event)

        count += 1

        if not in_office:
            partitions.append(count)
            count = 0

    if in_office:
        return -1

    return partitions

partitions = partition_days(a)
if partitions == -1:
    print(-1)
else:
    print(len(partitions))
    print(' '.join(map(str, partitions)))
