#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    string s;
    cin >> s;
    int ans = 0;
    reverse(s.begin(), s.end());
    for (int i = 0; i < s.size(); ++i) {
        if (s[i] == '0') {
            ++ans;
        } else {
            ans += (i != 0);
        }
    }
    cout << ans << endl;
    return 0;
}
