#include <bits/stdc++.h>
#define ll long long
const ll MOD = 1e9 + 7;
using namespace std;
const int ms = 3e5 + 10;
int main() {
  ll n, mx, ans = 0;
  cin >> n;
  vector<vector<ll>> a(n, vector<ll>(6));
  vector<vector<vector<ll>>> sum(n + 1, vector<vector<ll>>(6, vector<ll>(121)));
  for (ll i = 1; i <= n; i++) {
    for (ll j = 1; j <= 5; j++) {
      cin >> a[i - 1][j];
      a[i - 1][j]++;
      sum[i][j][a[i - 1][j]]++;
    }
    if ((a[i - 1][1] != -1) && (a[1][1] != -1)) {
      for (ll k = 1; k <= 5; k++) sum[i][k][a[i - 1][k]]--;
    }
    for (ll j = 1; j <= 5; j++)
      for (ll k = 119; k >= 0; k--) sum[i][j][k] += sum[i][j][k + 1];
  }
  for (ll i = 1; i <= n; i++) {
    ll t = 0;
    for (ll j = 1; j <= 5; j++) t += sum[i][j][a[0][j]];
    if (t <= (n - 1) / 2) mx = 500;
    else if (t <= (n - 1) / 4)
      mx = 1000;
    else if (t <= (n - 1) / 8)
      mx = 1500;
    else if (t <= (n - 1) / 16)
      mx = 2000;
    else if (t <= (n - 1) / 32)
      mx = 2500;
    else
      mx = 3000;
    for (ll j = 1; j <= 5; j++) {
      ll p = 250;
      ll add = mx - ((mx * (119 - a[i - 1][j])) / p);
      add = max(add, 0ll);
      if ((a[i - 1][1] != -1) && (a[1][1] != -1)) add = 0;
      if (mx < 500)
        add = 0;
      if (a[i - 1][j] == -1)
        ans += add;
      else
        ans = max(ans, add);
    }
  }
  if (ans < 0)
    ans = -1;
  cout << ans << endl;
}
