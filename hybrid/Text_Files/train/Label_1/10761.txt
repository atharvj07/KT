#include <bits/stdc++.h>
using namespace std;
#define MAXN 100001
vector<int> pos[MAXN];
int main() {
  int n;
  cin >> n;
  int a[MAXN];
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    pos[a[i]].push_back(i);
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    int sz = pos[i].size();
    if (sz >= 2 && pos[i][1] - pos[i][0] == pos[i][2] - pos[i][1]) ans++;
    if (sz == 1) ans++;
  }
  cout << ans << endl;
  for (int i = 1; i <= n; i++) {
    int sz = pos[i].size();
    if (sz >= 2 && pos[i][1] - pos[i][0] == pos[i][2] - pos[i][1])
      cout << i << ' ' << pos[i][1] - pos[i][0] << endl;
    if (sz == 1) cout << i << ' ' << 0 << endl;
  }
  return 0;
}
