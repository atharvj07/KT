import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = in.nextInt();
        int[] b = new int[n];
        b[n - 1] = a[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            int max = b[i + 1];
            for (int j = i + 1; j < n; j++)
                max = Math.max(max, a[j]);
            b[i] = Math.max(a[i], max);
        }
        for (int i : b)
            System.out.println(i);
    }
}
