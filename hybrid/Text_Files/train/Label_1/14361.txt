#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int& x : a) cin >> x;
  sort(a.begin(), a.end());
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    ans += a[i];
    for (int j = i + 1; j < n; j++) {
      int g = __gcd(a[i], a[j]);
      if (g > 1) {
        ans -= g;
        a[i] /= g;
        a[j] *= g;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
