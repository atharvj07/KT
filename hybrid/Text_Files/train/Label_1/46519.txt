import java.util.*;

public class Main {
    static final int MOD = 1000000007;
    static final long INF = Long.MAX_VALUE;

    static int n, m;
    static int[] a;
    static int[][] g;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        n = sc.nextInt();
        m = sc.nextInt();
        a = new int[n + 1];
        g = new int[n + 1][n + 1];

        for (int i = 0; i < m; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            char c = sc.next().charAt(0);
            if (c == 'S') {
                a[x]++;
                a[y]++;
            }
            g[x][y] = 1;
            g[y][x] = 1;
        }

        int odd = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i] % 2 == 1) {
                odd++;
            }
        }

        if (odd > 2) {
            System.out.println("-1");
            return;
        }

        List<Integer> ans = new ArrayList<>();

        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (g[i][j] == 1) {
                    if (a[i] % 2 == 1 && a[j] % 2 == 1) {
                        ans.add(i);
                        ans.add(j);
                        g[i][j] = 0;
                        g[j][i] = 0;
                    }
                }
            }
        }

        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (g[i][j] == 1) {
                    ans.add(i);
                    ans.add(j);
                    g[i][j] = 0;
                    g[j][i] = 0;
                }
            }
        }

        System.out.println(ans.size());
        for (int x : ans) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
