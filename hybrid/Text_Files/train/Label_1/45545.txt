import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();

        int[] testing = new int[k + 2];
        int time = 0, ans = 0;
        for (int i = 0; i < n; i++) {
            time = Math.max(time, testing[k]);
            testing[k] += a[i];
            for (int j = k; j > 0; j--)
                testing[j - 1] = Math.min(testing[j - 1] + 1, testing[j]);
            if (testing[0] % (n - i) == 0 && testing[0] / (n - i) > 0)
                ans++;
        }
        System.out.println(ans);
    }
}
