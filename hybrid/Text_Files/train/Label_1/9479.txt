#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length(), even = 0, odd = 0;
  vector<int> dif;
  for (int i = 1; i < n; i++) {
    if (s[i] != s[i - 1]) { dif.push_back(i - 1); }
  }
  if (dif.size() % 2 == 1) { dif.push_back(n); }
  if (dif.size() == 0) {
    even += (n / 2);
    odd += (n - (n / 2));
  } else {
    for (int i = 0; i < dif.size(); i += 2) {
      int l = dif[i], r = dif[i + 1];
      even += (r - l) / 2;
      odd += (r - l + 1) / 2;
    }
  }
  cout << even << " " << odd << endl;
  return 0;
}
