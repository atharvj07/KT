#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        long long ans1 = 0, ans2 = 0, pre = 0, suf = 0;
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) ans1 += a[i];
            else ans2 += a[i];
        }
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) pre += a[i];
            else pre -= a[i];
            ans1 = max(ans1, pre + ans2);
        }
        pre = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (i % 2 == 0) suf += a[i];
            else suf -= a[i];
            ans2 = max(ans2, suf + ans1);
        }
        cout << ans1 << endl;
    }
    return 0;
}
