#include <cstdio>
#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
#include <cstring>

const int N = 75;

int M, a[N][N], d[N][N];
bool vis[N];
char path[N];
std::vector<int> g[N];

void Modify(int u, int v, int w) {
    a[u][v] = a[v][u] = w;
    g[u].push_back(v);
}

bool Match(int u, int from) {
    vis[u] = true;
    if (u == 3) {
        return true;
    }
    int sz = g[u].size();
    for (int i = 0; i < sz; i++) {
        int v = g[u][i];
        if (!vis[v] && d[from][v] == d[from][u] + 1 && a[u][v] == a[from][u]) {
            if (Match(v, u)) {
                path[u] = v;
                return true;
            }
        }
    }
    return false;
}

int main() {
    int n;
    std::cin >> n;
    int s[4];
    for (int i = 1; i <= 3; i++) {
        std::cin >> s[i];
    }
    char c;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            std::cin >> c;
            if (c == '*') {
                a[i][j] = 0;
            } else {
                a[i][j] = c - 'a' + 1;
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        std::queue<int> q;
        memset(d, 0x3f, sizeof(d));
        d[i][i] = 0;
        q.push(i);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int j = 1; j <= n; j++) {
                if (a[u][j] && d[i][j] > d[i][u] + 1) {
                    d[i][j] = d[i][u] + 1;
                    q.push(j);
                }
            }
        }
        for (int j = 1; j <= n; j++) {
            if (d[i][j] == d[i][s[1]] + d[s[1]][j]) {
                Modify(i, j, a[i][j]);
            }
        }
    }
    bool find = false;
    for (int i = 1; i <= n; i++) {
        if (d[s[2]][i] == d[s[2]][s[1]] + d[s[1]][i] && a[s[2]][i] == a[s[1]][i]) {
            path[s[2]] = i;
            find = true;
        }
    }
    if (!find) {
        std::cout << "-1" << std::endl;
        return 0;
    }
    for (int i = 1; i <= n; i++) {
        if (d[s[3]][i] == d[s[3]][s[2]] + d[s[2]][i] && a[s[3]][i] == a[s[2]][i]) {
            path[s[3]] = i;
        }
    }
    int ans = 0;
    if (path[s[3]] == s[1] && path[s[2]] == s[1]) {
        path[s[2]] = 0;
        path[s[3]] = 0;
        ans++;
    } else {
        for (int i = s[3]; i; i = path[i]) {
            path[i] = 0;
            ans++;
        }
    }
    std::cout << ans << std::endl;
    for (int i = s[3]; i; i = path[i]) {
        if (path[i]) {
            std::cout << i << " " << path[i] << std::endl;
        }
    }
    for (int i = s[2]; i; i = path[i]) {
        if (path[i]) {
            std::cout << i << " " << path[i] << std::endl;
        }
    }
    return 0;
}
