#include<bits/stdc++.h>
#define int long long
using namespace std;
const int mod=1000000007;
int32_t main(){
    int n;
    cin>>n;
    vector<int>adj[100005];
    int a[n];
    int freq[100005]={0};
    for(int i=0;i<n;i++){
        cin>>a[i];
        freq[a[i]]++;
    }
    for(int i=2;i<=100000;i++){
        for(int j=i+i;j<=100000;j+=i){
            adj[j].push_back(i);
        }
    }
    int ans=1;
    for(int i=100000;i>=2;i--){
        int mx=0;
        for(auto p:adj[i]){
            mx=max(mx,freq[p]);
        }
        if(mx>0)
        ans=(ans*(freq[i]-mx)%mod)%mod;
        else ans=(ans*freq[i])%mod;
    }
    cout<<ans<<endl;
}
