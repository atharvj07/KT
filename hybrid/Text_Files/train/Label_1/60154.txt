n, m, k = map(int, input().split())
a = [0] + list(map(int, input().split()))
b = [0] + list(map(int, input().split()))
c = [0] + list(map(int, input().split()))
e = [[0] * (n + 1) for i in range(n + 1)]
for i in range(1, n + 1):
    for j in range(1, i + 1):
        e[i][j] = e[j][i] = 10 ** 10
for i in range(m):
    u, v = map(int, input().split())
    e[u][v] = 1
d = [[[0] * 3 for t in range(n + 1)] for s in range(k + 1)]
for s in range(k, 0, -1):
    for t in range(n + 1):
        d[s][t][0] = d[s][t][1] = -10 ** 10
        d[s][t][2] = c[t]
for s in range(k, 0, -1):
    for t in range(n + 1):
        if s + b[t] <= k and e[t][0] <= s and d[s][t][0] < 0:
            d[s + b[t]][0][0] = max(d[s + b[t]][0][0], d[s][t][0] + c[t])
            d[s + b[t]][0][1] = t
            d[s + b[t]][0][2] = max(d[s + b[t]][0][2], d[s][t][0] + c[t])
        for u in range(1, n + 1):
            if s + b[t] <= k and e[t][u] <= s and d[s][t][0] < 0:
                d[s + b[t]][0][0] = max(d[s + b[t]][0][0], d[s][t][0] + c[u])
                d[s + b[t]][0][1] = u
                d[s + b[t]][0][2] = max(d[s + b[t]][0][2], d[s][t][0] + c[u])
            if e[t][u] <= s and d[s][t][1] < 0:
                d[s][u][1] = t
                d[s][u][2] = max(d[s][u][2], d[s][t][1] + c[u])
            if e[t][u] <= s and d[s][t][2] >= 0:
                d[s][u][0] = max(d[s][u][0], d[s][t][2] + c[t])
                d[s][u][1] = t
                d[s][u][2] = max(d[s][u][2], d[s][t][2] + c[t])
        if s + 1 <= k and d[s + 1][t][0] < 0:
            d[s + 1][t][0] = max(d[s + 1][t][0], d[s][t][0] - a[t])
            d[s + 1][t][1] = t
            d[s + 1][t][2] = max(d[s + 1][t][2], d[s][t][0] - a[t])
print(max(max(d[0][i]) for i in range(n + 1)) if d[0][0][0] >= 0 else -1)
