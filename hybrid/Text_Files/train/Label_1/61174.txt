n = int(input())
a = list(map(int, input().split()))

# Create a dictionary to store the positions of each tier size
positions = {i: [] for i in range(1, n+1)}

# Fill the positions dictionary
for i, size in enumerate(a):
    positions[size].append(i)

# Initialize variables to store the total distance and the current positions of Sasha and Dima
total_distance = 0
sasha_pos = 0
dima_pos = 0

# Iterate through each tier size
for size in range(1, n+1):
    # Calculate the distances to the next tier for Sasha and Dima
    sasha_dist = abs(sasha_pos - positions[size][0])
    dima_dist = abs(dima_pos - positions[size][1])
    
    # Update the total distance and the positions of Sasha and Dima
    total_distance += sasha_dist + dima_dist
    sasha_pos = positions[size][0]
    dima_pos = positions[size][1]

print(total_distance)
