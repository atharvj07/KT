import java.util.Scanner;

public class JabberIDValidator {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String jabberId = scanner.nextLine();

        if (isValidJabberID(jabberId)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isValidJabberID(String jabberId) {
        // Check username
        if (!jabberId.matches("^[a-zA-Z0-9_]{1,16}@")) {
            return false;
        }

        // Check hostname
        String[] hostnameParts = jabberId.substring(jabberId.indexOf('@') + 1).split("/");
        String hostname = hostnameParts[0];
        if (!hostname.matches("^[a-zA-Z0-9_]{1,16}(\\.[a-zA-Z0-9_]{1,16})*$")) {
            return false;
        }

        // Check resource (optional)
        if (hostnameParts.length > 1) {
            String resource = hostnameParts[1];
            if (!resource.matches("^[a-zA-Z0-9_]{1,16}$")) {
                return false;
            }
        }

        return true;
    }
}
