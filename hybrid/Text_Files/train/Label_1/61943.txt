import sys

n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]
q = int(input())
for x1, x2 in (map(int, input().split()) for _ in range(q)):
    ok = True
    for i in range(n):
        if grid[i][x1-1] == '.' or grid[i][x2] == '.':
            for j in range(x1, x2):
                if grid[i][j] == 'X':
                    ok = False
                    break
            if not ok: break
    print('YES' if ok else 'NO')
