import sys

def dijkstra(n, m, k, graph):
    dist = [sys.maxsize] * (n + 1)
    vis = [False] * (n + 1)
    dist[1] = 0
    for _ in range(n - 1):
        u = -1
        for i in range(1, n + 1):
            if not vis[i] and (u == -1 or dist[i] < dist[u]):
                u = i
        vis[u] = True
        for edge in graph[u]:
            if not vis[edge[0]]:
                dist[edge[0]] = min(dist[edge[0]], dist[u] + edge[1])
    ans = []
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            if i != j:
                ans.append(dist[i] + dist[j])
    ans.sort()
    return ans[k - 1]

n, m, k = map(int, input().split())
graph = [[]]
for _ in range(n):
    graph.append([])
for _ in range(m):
    a, b, w = map(int, input().split())
    graph[a].append((b, w))
    graph[b].append((a, w))
print(dijkstra(n, m, k, graph))
