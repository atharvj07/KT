#include <iostream>
#include <vector>
using namespace std;
typedef long long LL;

int main() {
  int t;
  cin >> t;
  while (t--) {
    LL x, y, a, b;
    cin >> x >> y >> a >> b;
    if (x == y) {
      cout << 0 << endl;
    } else if (x < y) {
      if ((y - x) % 2 == 0) {
        cout << (y - x) / 2 * a << endl;
      } else {
        cout << (y - x) / 2 * a + b << endl;
      }
    } else {
      if ((x - y) % 2 == 0) {
        cout << (x - y) / 2 * a << endl;
      } else {
        cout << (x - y) / 2 * a + b << endl;
      }
    }
  }
}
