import math

steps = []
for _ in range(int(input())):
    n, d = map(int, input().split())
    a = list(map(int, input().split()))
    b = [0] * n
    ans = -1
    while True:
        for i in range(n):
            b[(i + d) % n] &= a[i]
        if a == b:
            ans = steps[-1] if steps else 0
            break
        a = b.copy()
        b = [0] * n
        steps.append(steps[-1] + 1 if steps else 1)
    print(ans)
