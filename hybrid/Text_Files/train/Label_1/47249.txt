import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }
        int min = p[0];
        long maxProfit = 0;
        for (int i = 1; i < n; i++) {
            if (p[i] < min) {
                min = p[i];
            } else if (p[i] - min > maxProfit) {
                maxProfit = p[i] - min;
            }
        }
        System.out.println(maxProfit);
    }
}
