#include <cstdio>
#include <vector>

int main() {
  int n, k;
  scanf("%d%d", &n, &k);
  std::vector<int> coef(n + 1), pos;
  for (int i = 0; i <= n; ++i) {
    char c;
    scanf(" %c", &c);
    if (c == '?') {
      coef[i] = -1;
      pos.push_back(i);
    } else {
      scanf("%d", &coef[i]);
    }
  }
  bool first = true;
  for (auto& j : pos) {
    if (first) {
      first = false;
      coef[j] = k;
    } else {
      coef[j] = -coef[j];
    }
  }
  if (coef[n] == 0) {
    puts("No");
    return 0;
  }
  puts("Yes");
}
