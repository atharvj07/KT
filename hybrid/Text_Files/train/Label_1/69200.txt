n = int(input())
votes = list(map(int, input().split()))

bribes = 0
while True:
    max_votes = max(votes[1:])
    if votes[0] > max_votes:
        break
    for i in range(1, n):
        if votes[i] >= votes[0]:
            votes[i] -= 1
            votes[0] += 1
            bribes += 1
            break

print(bribes)
