#include <iostream>
#include <vector>
#include <string>
using namespace std;

const vector<string> eeveelutions = {"vaporeon", "jolteon", "flareon", "espeon", "umbreon", "leafeon", "glaceon", "sylveon"};
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  for (auto &eeveelution : eeveelutions) {
    if (eeveelution.size() == n) {
      bool flag = true;
      for (int i = 0; i < n; i++) {
        if (s[i] != '.' && s[i] != eeveelution[i]) {
          flag = false;
        }
      }
      if (flag) {
        cout << eeveelution << endl;
        return 0;
      }
    }
  }
  cout << "none" << endl;
  return 0;
}
