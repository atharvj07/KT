#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[101], cnt = 0;
  cin >> n;
  for (int i = 9; i >= 1; i--) {
    while (n >= i) {
      a[cnt++] = i;
      n -= i;
    }
  }
  cout << cnt << endl;
  for (int i = 0; i < cnt; i++) { cout << a[i] << " "; }
  cout << endl;
  return 0;
}
