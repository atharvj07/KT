#include<bits/stdc++.h>
#define clr(ar) memset(ar, 0, sizeof(ar))
#define read() freopen("input.txt", "r", stdin)
#define write() freopen("output.txt", "w", stdout)
using namespace std;

typedef long long ll;
const int inf = 0x3f3f3f3f;
const ll INF = 0x3f3f3f3f3f3f3f3fLL;
const double pi = acos(-1.0);
const double eps = 1e-9;

const int mx = 200005;
vector<int> adj[mx];
int color[mx];

int paint(int u, int p){
    int cnt_black = 0, cnt_white = 0;

    for (int v: adj[u]){
        if (v == p) continue;
        if (color[v] == 0) cnt_white++;
        else cnt_black++;
    }

    if (cnt_black > cnt_white){
        color[u] = 1;
        return cnt_white;
    }
    else{
        color[u] = 0;
        return cnt_black;
    }
}

int main(){
    int n, ans = inf;
    cin >> n;

    for (int i = 1; i <= n; i++){
        cin >> color[i];
    }

    for (int i = 1; i < n; i++){
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    for (int i = 1; i <= n; i++){
        ans = min(ans, paint(i, i) + paint(i, i));
    }

    cout << ans << endl;
    return 0;
}
