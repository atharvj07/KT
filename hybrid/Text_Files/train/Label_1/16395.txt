#include<vector>
#include<iostream>

using namespace std;

int main() {
    long long t, k, n, ans, pow10;
    vector<long long> v;

    cin >> t;
    while (t--) {
        cin >> k >> n;
        while (n != 0) {
            pow10 = 1;
            for (long long i = 1; i < k; i++) pow10 *= 10;
            v.push_back(n / pow10);
            n %= pow10;
            k--;
        }
        cout << v.size() << endl;
        for (long long p : v) cout << p;
        cout << endl;
        v.clear();
    }
    return 0;
}
