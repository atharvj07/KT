#include <iostream>
using namespace std;

int main() {
  int n, k, a, m;
  cin >> n >> k >> a;
  cin >> m;
  int x[m];
  for (int i = 0; i < m; ++i) cin >> x[i];
  for (int i = 1; i <= m; ++i) {
    int l = x[i - 1] - a + 1;
    int r = x[i - 1];
    if (l > 0 && l <= n && (l - 1) % a == 0) {
      cout << i << endl;
      return 0;
    }
    if (r > 0 && r <= n && r % a == 0) {
      cout << i << endl;
      return 0;
    }
  }
  cout << -1 << endl;
  return 0;
}
