#include<bits/stdc++.h>
using namespace std;
long long a[100001], b[100001];
int main()
{
    long long n, sum = 0;
    cin >> n;
    for(int i=1; i<=n; i++)
        cin >> a[i];
    for(int i=1; i<=n; i++)
        cin >> b[i];
    for(int i=1; i<=n; i++){
        long long tmp = min(a[i], b[i] - a[i]);
        a[i] -= tmp;
        b[i] -= tmp;
        sum += a[i] * b[i];
    }
    cout << sum << endl;
    return 0;
}
