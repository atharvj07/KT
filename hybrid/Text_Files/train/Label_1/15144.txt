#include <iostream>
#include <string>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s[n];
  for (int i = 0; i < n; i++) cin >> s[i];
  int f = s[(n - 1) / 2][(n - 1) / 2];
  for (int i = 0; i < n; i++) {
    if (s[i][i] != f || s[i][n - i - 1] != f) {
      cout << "NO" << endl;
      return 0;
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i != (n - 1) / 2 && j != (n - 1) / 2 && s[i][j] != f) {
        cout << "NO" << endl;
        return 0;
      }
    }
  }
  cout << "YES" << endl;
  return 0;
}
