#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  string s;
  cin >> s;
  int p = s.find('.');
  int e = 0;
  if (p != -1) {
    s.erase(p, 1);
    e = p - 1;
  }
  while (s[0] == '0') {
    s.erase(0, 1);
    e++;
  }
  if (s[0] != '.') s.insert(1, ".");
  while (s.size() > 1 && s.back() == '0') s.pop_back();
  if (s.back() == '.') s.pop_back();
  if (e == 0) cout << s << '\n'; else cout << s << "E" << e << '\n';
  return 0;
}
