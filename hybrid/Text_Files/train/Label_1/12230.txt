#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n, m, d, x, h, mx = -1e9;
    cin >> n >> m;
    while (m--) {
        cin >> d >> h;
        if ((mx < 0) || (d == 1) || (h >= mx - 1) || (x < h)) mx = h, x = h;
        else return cout << "IMPOSSIBLE", 0;
    }
    cout << mx;
    return 0;
}
