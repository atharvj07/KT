import java.util.Scanner;

public class VasyaAndPainting {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine();

        char[][] sheet = new char[n][m];
        for (int i = 0; i < n; i++) {
            String row = scanner.nextLine();
            for (int j = 0; j < m; j++) {
                sheet[i][j] = row.charAt(j);
            }
        }

        boolean hasFrames = false;
        int[] firstFrame = new int[4];
        int[] secondFrame = new int[4];

        // Check for frames
        for (int i = 0; i < n - 2; i++) {
            for (int j = 0; j < m - 2; j++) {
                if (sheet[i][j] == '#' && sheet[i][j + 1] == '#' && sheet[i][j + 2] == '#' &&
                        sheet[i + 1][j] == '#' && sheet[i + 1][j + 2] == '#' && sheet[i + 2][j] == '#' &&
                        sheet[i + 2][j + 1] == '#' && sheet[i + 2][j + 2] == '#') {
                    if (!hasFrames) {
                        firstFrame[0] = i;
                        firstFrame[1] = j;
                        firstFrame[2] = i + 2;
                        firstFrame[3] = j + 2;
                        hasFrames = true;
                    } else {
                        secondFrame[0] = i;
                        secondFrame[1] = j;
                        secondFrame[2] = i + 2;
                        secondFrame[3] = j + 2;
                        break;
                    }
                }
            }
            if (hasFrames) {
                break;
            }
        }

        // Check if there are no other painted squares
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (sheet[i][j] == '#') {
                    if ((i < firstFrame[0] || i > firstFrame[2]) && (j < firstFrame[1] || j > firstFrame[3]) &&
                            (i < secondFrame[0] || i > secondFrame[2]) && (j < secondFrame[1] || j > secondFrame[3])) {
                        hasFrames = false;
                        break;
                    }
                }
            }
        }

        // Print the result
        if (hasFrames) {
            System.out.println("YES");
            System.out.println(firstFrame[0] + " " + firstFrame[1] + " " + firstFrame[2] + " " + firstFrame[3]);
            System.out.println(secondFrame[0] + " " + secondFrame[1] + " " + secondFrame[2] + " " + secondFrame[3]);
        } else {
            System.out.println("NO");
        }
    }
}
