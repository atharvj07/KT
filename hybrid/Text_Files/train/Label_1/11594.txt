#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k, s;
  cin >> n >> k >> s;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<vector<int>> dp(n + 1, vector<int>(k + 1, 0));
  dp[0][0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j <= k; j++) {
      dp[i + 1][j] += dp[i][j];
      if (j > 0) {
        dp[i + 1][j] += dp[i][j - 1] * a[i];
        if (dp[i + 1][j] >= 1000000007) dp[i + 1][j] -= 1000000007;
      }
    }
  }
  cout << dp[n][k] << endl;
  return 0;
}
