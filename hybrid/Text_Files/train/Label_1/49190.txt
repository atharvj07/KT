import java.util.*;

class Main {

    static int n, q;
    static char[] s;
    static int[] lcp;
    static int[][] dp;
    static int[] ans;

    public static void solve(Scanner in) {
        n = in.nextInt();
        q = in.nextInt();
        s = in.next().toCharArray();
        lcp = new int[n];
        dp = new int[2][n];
        ans = new int[q + 1];
        lcp[0] = 0;
        for (int i = 1; i < n; i++) {
            int j = lcp[i - 1];
            while (j > 0 && s[i] != s[j])
                j = lcp[j - 1];
            lcp[i] = j + (s[i] == s[j] ? 1 : 0);
        }
        for (int i = 0; i < n; i++) {
            if (i > 0) {
                dp[0][i] = dp[0][i - 1];
                dp[1][i] = Math.max(dp[1][i - 1], dp[0][i - 1] + (s[i] != s[i - 1] ? 1 : 2));
            }
            int j = lcp[i];
            if (j > 0) {
                dp[0][i] = Math.max(dp[0][i], dp[1][j - 1]);
                dp[1][i] = Math.max(dp[1][i], dp[0][j - 1] + 1);
            }
        }
        ans[0] = dp[1][n - 1];
        for (int i = 1; i <= q; i++) {
            int p = in.nextInt() - 1;
            s[p] = in.next().charAt(0);
            for (int j = p; j < n; j++) {
                if (j > 0) {
                    dp[0][j] = dp[0][j - 1];
                    dp[1][j] = Math.max(dp[1][j - 1], dp[0][j - 1] + (s[j] != s[j - 1] ? 1 : 2));
                }
                int k = lcp[j];
                if (k > 0) {
                    dp[0][j] = Math.max(dp[0][j], dp[1][k - 1]);
                    dp[1][j] = Math.max(dp[1][j], dp[0][k - 1] + 1);
                }
            }
            ans[i] = dp[1][n - 1];
        }
        for (int i = 0; i <= q; i++)
            System.out.println(ans[i]);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        solve(in);
    }
}
