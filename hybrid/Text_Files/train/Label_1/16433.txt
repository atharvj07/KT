#include<bits/stdc++.h>
using namespace std;
const int N = 320000;
int a[N];
int main() {
	int n;
	cin >> n;
	int ans = 0;
	int i = 0, j = 0, cnt = 0;
	while (i < n) {
		while (j < n && a[j] >= a[i]) {
			cnt++;
			j++;
		}
		if (cnt == 1) ans++;
		else ans += cnt / 2;
		cnt = 0;
		i = j;
	}
	cout << ans << endl;
	return 0;
}
