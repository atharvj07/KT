#include <bits/stdc++.h>
using namespace std;
int a[500001], n, k, d;
int main() {
  cin >> n >> k >> d;
  for (int i = 1; i <= n; i++) cin >> a[i];
  sort(a + 1, a + 1 + n);
  for (int i = 1; i <= n;) {
    int j = i;
    while (j <= n && a[j] - a[i] <= d) j++;
    if (j - i < k) {
      cout << "NO";
      return 0;
    }
    i = j;
  }
  cout << "YES";
  return 0;
}
