#include <iostream>
#include <vector>
using namespace std;

#define ll long long

ll const maxn = 4e4 + 1;
ll n, m, k;
vector<ll> a(maxn), b(maxn), pref1_a(maxn), pref1_b(maxn);

int main() {
  cin >> n >> m >> k;
  for (ll i = 0; i < n; i++) cin >> a[i];
  for (ll j = 0; j < m; j++) cin >> b[j];
  for (ll i = 0; i < n; i++) pref1_a[i + 1] = pref1_a[i] + a[i];
  for (ll j = 0; j < m; j++) pref1_b[j + 1] = pref1_b[j] + b[j];

  ll ans = 0;
  for (ll l1 = 0; l1 < n; l1++) {
    for (ll r1 = l1; r1 < n; r1++) {
      ll tot_a = pref1_a[r1 + 1] - pref1_a[l1];
      if (tot_a > k) break;
      ll l2 = 0, r2 = -1;
      while (l2 < m) {
        while (r2 + 1 < m and pref1_b[r2 + 2] - pref1_b[l2] + tot_a <= k) r2++;
        if (pref1_b[r2 + 1] - pref1_b[l2] + tot_a == k) ans++;
        l2++;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
