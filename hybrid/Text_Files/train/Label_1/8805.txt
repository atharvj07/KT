#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll gcd(ll a, ll b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
int main() {
  ll n, ans = 0, l = 0, r = 0;
  cin >> n;
  vector<ll> ar(n);
  for (auto &it : ar) cin >> it;
  while (r < n) {
    while (r < n && (ar[r] - ar[l] <= 1 || gcd(ar[r], ar[r - 1]) == 1))
      r++;
    ans = max(ans, r - l);
    l++;
  }
  cout << ans << endl;
  return 0;
}
