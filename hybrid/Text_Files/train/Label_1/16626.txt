#include <iostream>

int main() {
  long a, b, c;
  std::cin >> a >> b >> c;
  a *= 10;
  long pos = -1;
  for (long i = 1;; i++) {
    if (a >= b) {
      a %= b;
      a *= 10;
    }
    if (a % b == a) {
      break;
    }
    if (a % b * 10 / b == c) {
      pos = i;
      break;
    }
  }
  std::cout << pos << std::endl;
  return 0;
}
