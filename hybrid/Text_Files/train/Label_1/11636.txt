#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

struct data {
    long long x, y, d;
};

data pt[100005];
long long dist[100005];

bool cmp(const data &a, const data &b) {
    return a.d < b.d;
}

int main() {
    int n, m;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) scanf("%lld%lld", &pt[i].x, &pt[i].y);
    scanf("%d", &m);
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) scanf("%lld", &dist[j]);
        sort(dist, dist + n);
        sort(pt, pt + n, cmp);
        int k = 0;
        vector<pair<long long, long long>> v;
        for (int j = 0; j < n; j++) {
            for (int l = j + 1; l < n; l++) {
                long long dx = pt[j].x - pt[l].x;
                long long dy = pt[j].y - pt[l].y;
                long long d = dx * dx + dy * dy;
                if (d == dist[j]) {
                    long long x1 = pt[l].x + (pt[j].x - pt[l].x) * (dist[j] - d) / d;
                    long long y1 = pt[l].y + (pt[j].y - pt[l].y) * (dist[j] - d) / d;
                    long long x2 = pt[j].x + (pt[l].x - pt[j].x) * (dist[j] - d) / d;
                    long long y2 = pt[j].y + (pt[l].y - pt[j].y) * (dist[j] - d) / d;
                    if (x1 == x2 && y1 == y2) {
                        v.push_back({x1, y1});
                        k++;
                    }
                }
            }
        }
        sort(v.begin(), v.end());
        printf("%d", k);
        for (auto p : v) printf(" %lld %lld", p.first, p.second);
        printf("\n");
    }
    return 0;
}
