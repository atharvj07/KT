import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n, m, turns;
        int[] arr;
        for(int t = 1; ; t++) {
            n = sc.nextInt();
            m = sc.nextInt();
            if(n == 0 && m == 0) break;
            arr = new int[n];
            turns = 0;
            while(true) {
                turns++;
                for(int i = 0; i < n; i++) {
                    for(int j = 0; j < n; j++) {
                        if(arr[i] == arr[j] && arr[i] >= 0) {
                            if(Math.random() > 0.5) arr[i]++;
                            else arr[j]++;
                        }
                    }
                }
                int max = 0;
                for(int i = 0; i < n; i++) {
                    if(arr[i] > max) max = arr[i];
                }
                if(max >= m) break;
            }
            System.out.println("Case " + t + ": " + turns);
        }
    }
}
