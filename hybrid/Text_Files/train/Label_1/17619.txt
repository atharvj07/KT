#include <iostream>
using namespace std;

int main() {
    int n, k, last = 2, pos = 1;
    cin >> n >> k;
    for (int i = 2; i <= n; i++) {
        pos += last;
        if (pos >= k) {
            cout << last << endl;
            return 0;
        }
        last++;
    }
    cout << last << endl;
    return 0;
}
