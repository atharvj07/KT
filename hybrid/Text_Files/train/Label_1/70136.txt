def dfs(u, p):
    g[u] = 1
    for v in graph[u]:
        if v == p:
            continue
        dfs(v, u)
        if g[v] > k:
            g[u] += g[v]
    return g[u]

n, k = list(map(int, input().split()))
a = list(map(int, input().split()))
graph = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    u, v = list(map(int, input().split()))
    graph[u].append(v)
    graph[v].append(u)

g = [0] * (n + 1)
dfs(1, 1)
print(max(a) * (g[1] > k))
