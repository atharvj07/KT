n, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
if a[0] >= n:
    print(-1)
else:
    ans = 0
    cur = 0
    for i in range(k - 1, -1, -1):
        ans += 1
        cur += a[i]
        if cur >= n:
            break
    if cur == n:
        print(ans)
    else:
        print(-1)
