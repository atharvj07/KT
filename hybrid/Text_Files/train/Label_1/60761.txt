import sys
input = sys.stdin.readline

n = int(input())
l = [list(map(int, input().split())) for i in range(n)]

l.sort()
ans = 0
for i in range(n):
    x1, y1 = l[i]
    for j in range(i+1, n):
        x2, y2 = l[j]
        for k in range(j+1, n):
            x3, y3 = l[k]
            if (x2-x1) * (y3-y2) == (x3-x2) * (y2-y1):
                ans += 1

print(ans)
