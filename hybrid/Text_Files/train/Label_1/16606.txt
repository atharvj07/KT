#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int n;
    cin >> n;
    vector<pair<ll, ll> > v(n);
    for (int i = 0; i < n; i++) {
        ll a, b;
        cin >> a >> b;
        v[i] = {a, b};
    }
    for (int i = 0; i < n - 1; i++) {
        if (v[i].first > v[i + 1].first && v[i].second > v[i + 1].second) {
            cout << "NO" << endl;
            return 0;
        }
        swap(v[i].first, v[i].second);
    }
    cout << "YES" << endl;
    return 0;
}
