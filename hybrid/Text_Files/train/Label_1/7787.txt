#include <bits/stdc++.h>
using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int q;
  cin >> q;
  while (q--) {
    string s, t, p;
    cin >> s >> t >> p;
    sort(s.begin(), s.end());
    sort(t.begin(), t.end());
    sort(p.begin(), p.end());
    if (s == t)
      cout << "YES\n";
    else
      cout << "NO\n";
  }
}
