#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n, time_last = 0, mx_size = 0;
    cin >> n;
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    while ( n-- )
    {
        int t, c;
        cin >> t >> c;
        while ( !pq.empty() && t >= pq.top().first )
        {
            time_last = pq.top().first;
            pq.pop();
        }
        pq.push({t + c, c});
        mx_size = max(mx_size, (int)pq.size());
    }
    while ( !pq.empty() )
    {
        time_last = pq.top().first;
        pq.pop();
    }
    cout << time_last << ' ' << mx_size << '\n';
    return 0;
}
