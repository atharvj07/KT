x, y, n = map(int, input().split())

best_diff = float("inf")
best_a, best_b = 0, 1

for b in range(1, n + 1):
    a = round(x * b / y)
    diff = abs(x/y - a/b)
    if diff < best_diff:
        best_diff = diff
        best_a, best_b = a, b
    elif diff == best_diff and b < best_b:
        best_a, best_b = a, b

print(f"{best_a}/{best_b}")
