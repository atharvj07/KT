#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int N = 2e5;

int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n, m;
  cin >> n >> m;
  vector<ll> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  vector<pair<int, int>> op(m);
  for (int i = 0; i < m; ++i) {
    cin >> op[i].first >> op[i].second;
    op[i].second--;
  }
  reverse(op.begin(), op.end());
  for (const auto &p : op) {
    if (p.first == 1)
      sort(a.begin(), a.begin() + p.second + 1);
    else
      sort(a.begin(), a.begin() + p.second + 1, greater<ll>());
  }
  for (const auto &it : a) cout << it << " ";
  cout << "\n";
  return 0;
}
