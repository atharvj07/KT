#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  sort(a.rbegin(), a.rend());
  int ans = 0;
  for (int x : a) {
    if (k <= 0) break;
    x = min(100, x / 10 * 10 + 10);
    if (x > 100) break;
    ans += x / 10;
    k--;
  }
  cout << ans << endl;
}
