#include<bits/stdc++.h>
using namespace std;
int main(){
    long long n,k;
    cin>>n>>k;
    long long a[k];
    for(int i=0;i<k;i++)cin>>a[i];
    long long ans=0,ind=0;
    for(int i=0;i<k;i++){
        if(a[i]>ans){
            ans=a[i];
            ind=i;
        }
    }
    cout<<ind+1<<" "<<n/ans<<endl;
    return 0;
}
