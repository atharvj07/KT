#include <bits/stdc++.h>
using namespace std;
int l[100005], r[100005];
string s, t;
int main() {
  cin >> s >> t;
  stack<char> st;
  for (int i = 0; i < s.length(); i++) {
    st.push(s[i]);
    while (st.size() >= 3) {
      char a = st.top();
      st.pop();
      char b = st.top();
      st.pop();
      char c = st.top();
      st.pop();
      if ((a == 'A' && b == 'A' && c == 'A') || (a == 'B' && b == 'B' && c == 'B') ||
          (a == 'C' && b == 'C' && c == 'C')) {
        continue;
      } else if (a == 'A') {
        st.push('B');
        st.push('C');
      } else if (a == 'B') {
        st.push('A');
        st.push('C');
      } else if (a == 'C') {
        st.push('A');
        st.push('B');
      }
    }
    r[i] = st.size();
  }
  stack<char> st1;
  for (int j = t.length() - 1; j >= 0; j--) {
    st1.push(t[j]);
    while (st1.size() >= 3) {
      char a = st1.top();
      st1.pop();
      char b = st1.top();
      st1.pop();
      char c = st1.top();
      st1.pop();
      if ((a == 'A' && b == 'A' && c == 'A') || (a == 'B' && b == 'B' && c == 'B') ||
          (a == 'C' && b == 'C' && c == 'C')) {
        continue;
      } else if (a == 'A') {
        st1.push('B');
        st1.push('C');
      } else if (a == 'B') {
        st1.push('A');
        st1.push('C');
      } else if (a == 'C') {
        st1.push('A');
        st1.push('B');
      }
    }
    l[j] = st1.size();
  }
  int q;
  cin >> q;
  while (q--) {
    int aa, bb, cc, dd;
    cin >> aa >> bb >> cc >> dd;
    if (r[bb] <= l[dd] && r[bb - 1] >= l[cc - 1]) {
      cout << '1';
    } else {
      cout << '0';
    }
  }
  cout << "\n";
  return 0;
}
