import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long p = sc.nextLong();
        int k = sc.nextInt();
        if (p % k == 0) {
            System.out.println(-1);
        } else {
            int cnt = 0;
            long tmp = p;
            while (tmp > 0) {
                tmp /= k;
                cnt++;
            }
            System.out.println(cnt);
            for (int i = cnt - 1; i >= 0; i--) {
                p %= k;
                System.out.print(p + " ");
                tmp = p;
                for (int j = 0; j < i; j++) {
                    tmp *= k;
                }
                p += tmp;
            }
            System.out.println();
        }
    }
}
