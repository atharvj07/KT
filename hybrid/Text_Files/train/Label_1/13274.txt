#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++)
    {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    vector<int> x(n);
    for (int i = 0; i < n; i++)
    {
        if (i % 2 == 0)
        {
            x[i] = a[i / 2];
        }
        else
        {
            x[i] = a[n - 1 - i / 2];
        }
    }
    for (int i = 0; i < n; i++)
    {
        cout << x[i] << " ";
    }
    cout << endl;
    return 0;
}
