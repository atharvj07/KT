#include <iostream>
using namespace std;
#define ll long long
#define mod 1000000007

int main() {
    ll n, m;
    cin >> n >> m;
    ll on[n + 1], dp[n + 1][2];
    for (ll i = 1; i <= m; i++) { cin >> on[i]; }
    dp[1][1] = 1, dp[1][0] = 0;
    for (ll i = 2; i <= n; i++) {
        dp[i][0] = dp[i - 1][0];
        if (dp[i - 1][1] > 0) {
            dp[i][0] = (dp[i][0] + dp[i - 1][1]) % mod;
        }
        if (dp[i - 1][0] > 0) {
            dp[i][1] = (dp[i][1] + dp[i - 1][0]) % mod;
        }
        for (ll j = 1; j <= m; j++) {
            if (on[j] == i) { dp[i][1] = (dp[i][1] + dp[i - 1][0]) % mod; }
        }
    }
    cout << dp[n][1] << endl;
    return 0;
}
