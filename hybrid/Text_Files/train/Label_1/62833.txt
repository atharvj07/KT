n = int(input())
a = list(map(int, input().split()))

fixed_points = 0
for i in range(n):
    if a[i] == i:
        fixed_points += 1

for i in range(n):
    if a[i] != i and a[a[i]] == i:
        fixed_points += 1
        break

if fixed_points < n:
    print(fixed_points + 1)
else:
    print(fixed_points)
