import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) throws Exception {
        Scanner scan = new Scanner(System.in);
        long n = scan.nextLong();
        long k = scan.nextLong();
        long[] arr = new long[(int)n];
        for (int i = 0; i < n; i++)
            arr[i] = scan.nextLong();
        Arrays.sort(arr);
        long ans = arr[(int)(n - 1 - k)];
        if (ans == 0)
            ans++;
        while (true) {
            boolean flag = true;
            for (int i = 0; i < n - 1 - k; i++)
                if (arr[i] % ans != 0) {
                    flag = false;
                    break;
                }
            if (flag)
                break;
            ans++;
        }
        System.out.println(ans);
    }
}
