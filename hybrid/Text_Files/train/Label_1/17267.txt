#include <bits/stdc++.h>
using namespace std;
const int N = 45005;
int n;
int a[N];
int main() {
  cin >> n;
  int sum = 0;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i], sum += a[i];
    if (sum > a[i]) swap(sum, a[i]);
  }
  if (sum > 1)
    cout << "Yes\n" << 1 << ' ' << a[1] << '\n' << n - 1 << ' ';
  else
    cout << "No";
  for (int i = 2; i <= n; ++i) cout << a[i] << ' ';
  cout << '\n';
}
