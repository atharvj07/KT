import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt();
        int k = sc.nextInt();

        if (d > 2 * (k - 1)) {
            System.out.println("NO");
        } else if (n == 1) {
            System.out.println("YES");
        } else if (d >= 3) {
            System.out.println("YES");
            int[] p = new int[n];
            int[] c = new int[n];
            for (int i = 1; i < n; i++) {
                p[i] = i + 1;
                c[i] = 1;
            }
            int u = 1;
            int v = 2;
            int w = 3;
            while (d > 2) {
                System.out.println(u + " " + v);
                System.out.println(v + " " + w);
                p[v] = u;
                p[w] = v;
                c[u]++;
                c[v]++;
                d -= 2;
                if (c[u] >= k) {
                    u = v;
                }
                if (c[v] >= k) {
                    v = w;
                }
                if (c[w] >= k) {
                    w++;
                }
            }
            if (d == 1) {
                System.out.println(u + " " + v);
            }
            for (int i = 1; i < n; i++) {
                if (p[i] != 0) {
                    System.out.println(i + " " + p[i]);
                }
            }
        } else {
            if (k == 2) {
                if (n == 2) {
                    System.out.println("YES");
                    System.out.println(1 + " " + 2);
                } else if (n == 3) {
                    System.out.println("NO");
                } else {
                    System.out.println("YES");
                    for (int i = 2; i <= n; i++) {
                        System.out.println(i + " " + 1);
                    }
                }
            } else {
                if (n == 2) {
                    System.out.println("YES");
                    System.out.println(1 + " " + 2);
                } else {
                    System.out.println("NO");
                }
            }
        }
    }
}
