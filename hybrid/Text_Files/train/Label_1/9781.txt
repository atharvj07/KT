#include <iostream>
#include <algorithm>
#include <vector>
#define int long long
using namespace std;
const int N = 3e2+3;
main(){
    int n,m,a[N],l[N],r[N],b[N],ans[N],ansl=0;
    cin>>n>>m;
    for(int i=1;i<=n;i++)cin>>a[i];
    for(int i=1;i<=m;i++){
        cin>>l[i]>>r[i];
    }
    for(int i=1;i<=m;i++){
        b[i]=a[r[i]]-a[l[i]-1];
    }
    sort(b+1,b+m+1);
    for(int i=m;i>=1;i--){
        if(ansl+b[i]>0){
            ans[++ansl]=i;
        }
    }
    cout<<a[n]-a[0]<<endl;
    cout<<ansl<<endl;
    for(int i=1;i<=ansl;i++){
        cout<<ans[i]<<" ";
    }
    cout<<endl;
}
