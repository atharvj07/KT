#include <iostream>
#include <vector>

using namespace std;

const int M = 4;
const int N = 5005;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> dp[n + 1][M];
  for (int i = n - 1; i >= 0; i--) {
    for (int j = 0; j < M; j++) {
      if (j == 0) {
        if (s[i] == 'b') {
          dp[i][j].push_back(i);
        }
        continue;
      }
      for (int k = i + 1; k < n; k++) {
        for (auto &idx : dp[k][j - 1]) {
          if (s[k] == "eabr"[j]) {
            dp[i][j].push_back(k);
          }
        }
      }
    }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < M; j++) ans += dp[i][j].size();
  }
  cout << ans << "\n";
  return 0;
}
