#include <bits/stdc++.h>

#define REP(i, a, b) for (int i = int(a); i < int(b); ++i)

using namespace std;

int main() {
  int n;
  cin >> n;
  string a, b;
  cin >> a >> b;
  int cnt = 0;
  REP(i, 0, n) {
    cnt += (a[i] != b[i]);
  }
  int ans = cnt / 2;
  if (cnt % 2) ans++;
  cout << ans << '\n';
  return 0;
}
