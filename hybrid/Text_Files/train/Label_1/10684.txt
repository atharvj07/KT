#include <bits/stdc++.h>

using namespace std;

const int mxN = 3e5 + 5;
int n, k, m, a[mxN], b[mxN], g, ans;
unordered_map<int, bool> mp;

int gcd(int a, int b) {
    if (!b) return a;
    return gcd(b, a % b);
}

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) { cin >> m; mp[m] = true; }
    for (auto i = mp.begin(); i != mp.end(); ++i) b[++g] = i->first;
    for (int i = 1; i <= g; ++i) {
        for (int j = 1; j <= g; ++j) {
            m = gcd(b[i], b[j]);
            if (m > ans && m - 1 <= k) ans = m;
        }
    }
    cout << ans << '\n';
    return 0;
}
