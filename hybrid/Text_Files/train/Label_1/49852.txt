import java.util.*;
import java.lang.*;

class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k1 = sc.nextInt();
        int k2 = sc.nextInt();
        int[] A = new int[n];
        int[] B = new int[n];
        for (int i = 0; i < n; i++) {
            A[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            B[i] = sc.nextInt();
        }
        Arrays.sort(A);
        Arrays.sort(B);
        int i = 0, j = 0;
        long sum = 0;
        while (i < n && k1 > 0) {
            sum += (long) (A[i] - B[j]) * (A[i] - B[j]);
            i++;
            j++;
            k1--;
        }
        while (j < n && k2 > 0) {
            sum += (long) (A[i - 1] - B[j]) * (A[i - 1] - B[j]);
            j++;
            k2--;
        }
        System.out.println(sum);
    }
}
