import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    arr = list(map(int, input().split()))
    dp = [0] * n
    dp[n - 1] = 1
    ans = 1
    for i in range(n - 2, -1, -1):
        mx = 0
        for j in range(i + 1, n):
            if arr[i] == arr[j]:
                mx = max(mx, dp[j])
        if mx > 0:
            dp[i] = mx + 1
            ans = max(ans, dp[i])
        else:
            dp[i] = 1
    print(ans)
