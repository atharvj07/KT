from collections import deque

def solve_game(n, rick_set, morty_set):
    dp = [[-1]*n for _ in range(2)]
    dp[0][0] = dp[1][0] = 0
    q = deque([(0, 0), (1, 0)])
    while q:
        player, pos = q.popleft()
        opponent = 1 - player
        steps = rick_set if player == 0 else morty_set
        for step in steps:
            prev_pos = (pos - step + n) % n
            if dp[opponent][prev_pos] == -1:
                if dp[player][pos] == 0:
                    dp[opponent][prev_pos] = 1
                    q.append((opponent, prev_pos))
                else:
                    is_loop = True
                    for step_opp in (rick_set if opponent == 0 else morty_set):
                        if dp[opponent][(prev_pos - step_opp + n) % n] != 1:
                            is_loop = False
                            break
                    if is_loop:
                        dp[opponent][prev_pos] = 0
                        q.append((opponent, prev_pos))
    return dp

def print_result(dp, n):
    for player in range(2):
        for pos in range(1, n):
            if dp[player][pos] == -1:
                print("Loop", end=" ")
            elif dp[player][pos] == 1:
                print("Win", end=" ")
            else:
                print("Lose", end=" ")
        print()

n = int(input().strip())
rick_set = list(map(int, input().strip().split()))[1:]
morty_set = list(map(int, input().strip().split()))[1:]

dp = solve_game(n, rick_set, morty_set)
print_result(dp, n)
