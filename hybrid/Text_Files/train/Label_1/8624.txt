#include <iostream>

using namespace std;

int main() {
  long long H, n;
  cin >> H >> n;
  long long d[n];
  for (int i = 0; i < n; i++) cin >> d[i];
  long long res = -1;
  long long hp = H;
  for (int i = 0; i < n; i++) {
    hp += d[i];
    if (hp <= 0) { res = i + 1; break; }
  }
  cout << res << '\n';
  return 0;
}
