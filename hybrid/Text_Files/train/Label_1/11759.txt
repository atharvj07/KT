#include <iostream>
#include <string>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int a = 0, c = 0, g = 0, t = 0, q = 0;
  for (char ch : s) {
    if (ch == '?')
      q++;
    else if (ch == 'A')
      a++;
    else if (ch == 'C')
      c++;
    else if (ch == 'G')
      g++;
    else if (ch == 'T')
      t++;
  }
  int m = (a + c + g + t + q) / 4;
  if (a > m || c > m || g > m || t > m) {
    cout << "===" << endl;
  } else {
    for (int i = 0; i < n; i++) {
      if (s[i] == '?') {
        if (a < m) {
          s[i] = 'A';
          a++;
        } else if (c < m) {
          s[i] = 'C';
          c++;
        } else if (g < m) {
          s[i] = 'G';
          g++;
        } else if (t < m) {
          s[i] = 'T';
          t++;
        }
      }
    }
    cout << s << endl;
  }
  return 0;
}
