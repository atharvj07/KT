#include <iostream>
#define ll long long
using namespace std;
bool f[64][100005];
int main() {
    int n, i, j, k, ans = 0;
    cin >> n;
    ll a[n + 5], b[n + 5], c, d;
    for (i = 0; i < n; i++) {
        cin >> a[i];
        b[i] = a[i];
    }
    for (i = 0; i < 63; i++) {
        for (j = 0; j < n; j++) {
            if (j == 0)
                f[i][j] = (a[j] & (1LL << i));
            else
                f[i][j] = (f[i][j - 1] | (a[j] & (1LL << i)));
        }
    }
    for (i = 0; i < 63; i++) {
        for (j = 0; j < n; j++) {
            if (j == 0)
                f[i][j + n] = (b[j] & (1LL << i));
            else
                f[i][j + n] = (f[i][j - 1 + n] | (b[j] & (1LL << i)));
        }
    }
    for (i = 1; i <= n; i++) {
        for (j = i; j <= n; j++) {
            c = 0;
            d = 0;
            for (k = 0; k < 63; k++) {
                c |= f[k][j - 1];
                if (i - 1 >= 0) d |= f[k][i - 2];
            }
            c = c ^ d;
            if (c > ans) ans = c;
        }
    }
    cout << ans << endl;
    return 0;
}
