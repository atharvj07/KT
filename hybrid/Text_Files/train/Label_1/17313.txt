#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll mul(ll a,ll b,ll mod) {
	return (((a%mod)*(b%mod))%mod);
}
ll bigmod(ll a,ll b,ll mod) {
	a%=mod;
	ll res=1;
	while(b>0) {
		if(b&1) res=mul(res,a,mod);
		a=mul(a,a,mod);
		b>>=1;
	}
	return (res%mod);
}
int main() {
	ll a,b;
	cin>>a>>b;
	if(a==b) {
		cout<<"0"<<endl;
		return 0;
	}
	ll inv=bigmod((b-a),1000000005,1000000007);
	ll add=mul(a,inv,1000000007);
	ll ans=mul(add,(1+a*(a-1)/2)%1000000007,1000000007);
	cout<<ans<<endl;
	return 0;
}
