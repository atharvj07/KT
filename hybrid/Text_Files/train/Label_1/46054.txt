import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        String S = sc.next();
        List<Integer>[] adj = new ArrayList[N + 1];
        for (int i = 0; i <= N; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < N - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        int[] cnt = new int[N + 1];
        for (int i = 0; i < N; i++) cnt[i] = S.charAt(i) - '0';
        boolean ok = true;
        for (int i = 1; i <= N; i++) {
            if (cnt[i] > 0) {
                if (ok) {
                    ok = false;
                } else {
                    System.out.println(-1);
                    return;
                }
            }
        }
        int ans = 0;
        for (int i = 1; i <= N; i++) {
            if (cnt[i] > 0) {
                int dist = bfs(adj, cnt, i);
                if (dist == -1) {
                    System.out.println(-1);
                    return;
                }
                ans += dist - 1;
            }
        }
        System.out.println(ans);
    }

    private static int bfs(List<Integer>[] adj, int[] cnt, int start) {
        int[] dist = new int[cnt.length];
        Arrays.fill(dist, -1);
        dist[start] = 0;
        Queue<Integer> q = new LinkedList<>();
        q.offer(start);
        while (!q.isEmpty()) {
            int cur = q.poll();
            if (cnt[cur] > 0) return dist[cur];
            for (int next : adj[cur]) {
                if (dist[next] == -1) {
                    dist[next] = dist[cur] + 1;
                    q.offer(next);
                }
            }
        }
        return -1;
    }
}
