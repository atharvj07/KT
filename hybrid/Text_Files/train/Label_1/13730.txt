#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> ans(m, vector<int>(5, 0));
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        for (int j = 0; j < m; j++) {
            ans[j][s[j] - 'A']++;
        }
    }
    vector<int> pts(m, 0);
    for (int i = 0; i < m; i++) {
        int max_a = 0, idx;
        for (int j = 0; j < 5; j++) {
            if (ans[i][j] > max_a) {
                max_a = ans[i][j];
                idx = j;
            }
        }
        pts[i] = idx;
    }
    int max_ans = 0;
    vector<int> a(m);
    for (int i = 0; i < m; i++) {
        cin >> a[i];
        max_ans += (pts[i] == 0 ? a[i] : 0);
    }
    cout << max_ans << endl;
    return 0;
}
