#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int k, n;
  cin >> k >> n;
  vector<int> a(k), b(n);
  for (int i = 0; i < k; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  sort(b.begin(), b.end());
  int ans = 0;
  for (int i = 0; i < b.size(); ++i) {
    int sum = b[i];
    for (int j = i + 1; j < b.size(); ++j) {
      if (sum == b[j]) continue;
      for (int x = -2000; x <= 2000; ++x) {
        if (sum + a[0] == x && sum + a[0] + a[1] == b[j]) {
          ++ans;
          break;
        }
      }
    }
  }
  cout << ans << '\n';
  return 0;
}
