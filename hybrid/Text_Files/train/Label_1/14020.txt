#include <bits/stdc++.h>
using namespace std;
#define int long long
vector<int> adj[100001];
int Center, dist[100001], deg[100001];
int32_t main() {
  int n, k;
  cin >> n >> k;
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
    deg[u]++;
    deg[v]++;
  }
  bool center = false;
  for (int i = 1; i <= n; i++) {
    if (deg[i] >= 3) {
      if (!center) {
        Center = i;
        center = true;
      } else {
        cout << "No";
        return 0;
      }
    }
  }
  if (!center) {
    cout << "No";
    return 0;
  }
  queue<int> Q;
  Q.push(Center);
  while (!Q.empty()) {
    int u = Q.front();
    Q.pop();
    for (int v : adj[u]) {
      if (dist[v] == 0) {
        dist[v] = dist[u] + 1;
        Q.push(v);
      } else if (dist[v] == dist[u]) {
        cout << "No";
        return 0;
      }
    }
  }
  cout << "Yes";
  return 0;
}
