#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n + 1);
  vector<int> in_degree(n + 1);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    in_degree[v]++;
  }
  int cnt = 0;
  vector<int> q;
  for (int i = 1; i <= n; i++) {
    if (in_degree[i] == 0) q.push_back(i);
  }
  while (!q.empty()) {
    int u = q.back();
    q.pop_back();
    cnt++;
    for (int v : adj[u]) {
      in_degree[v]--;
      if (in_degree[v] == 0) q.push_back(v);
    }
  }
  if (cnt == n)
    cout << cnt << endl;
  else
    cout << -1 << endl;
  return 0;
}
