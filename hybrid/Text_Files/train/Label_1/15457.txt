#include<iostream>
using namespace std;
int main(){
    long long t;
    cin>>t;
    while(t--){
        long long str,intg,exp;
        cin>>str>>intg>>exp;
        if(exp < intg || str + exp < intg){
            cout<<0<<endl;
            continue;
        }
        cout<<exp - intg + 1<<endl;
    }
}
