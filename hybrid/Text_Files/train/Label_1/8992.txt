#include <iostream>

using namespace std;

int p, x, r = 1;
bool possible;

int main() {
    cin >> p >> x;
    while (p--) {
        r *= x;
        if (r > 1e8) { possible = false; break; }
    }
    if (!possible) cout << "Impossible";
    else {
        r /= x;
        while (r % 10 == 0) r /= 10;
        cout << r;
    }
    cout << endl;
}
