#include <iostream>
#include <vector>

using namespace std;

int n, m;
vector<int> a, sub;
vector<vector<int>> g;

void dfs(int u, int p) {
  for (int v : g[u]) {
    if (v == p) continue;
    dfs(v, u);
    a[u] += a[v];
  }
}

int main() {
  cin >> n >> m;
  a.resize(n);
  sub.resize(n);
  g.resize(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    u -= 1, v -= 1;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  dfs(0, -1);
  while (m--) {
    int t, x;
    cin >> t >> x;
    x -= 1;
    if (t == 1) {
      int val;
      cin >> val;
      a[x] += val;
      for (int v : g[x]) sub[v] -= val;
    } else {
      cout << a[x] + sub[x] << "\n";
    }
  }
  return 0;
}
