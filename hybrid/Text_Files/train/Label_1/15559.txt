#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (auto &x : a) cin >> x;
        int l = 1, r = 1, mx = 1;
        for (int i = 1; i < n; i++) {
            if (a[i] == a[i - 1])
                l++;
            else {
                l = 1;
                if (a[i] <= a[mx])
                    l = r + 1;
            }
            r = max(r, l);
            mx = max(mx, l);
        }
        cout << r << endl;
    }
    return 0;
}
