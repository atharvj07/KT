import sys
input = sys.stdin.readline

n = int(input())
s1 = list(map(int, input().split()))
s2 = list(map(int, input().split()))
ans1 = []
ans2 = []
for i in range(1, n):
    win1 = set()
    win2 = set()
    lose1 = set()
    lose2 = set()
    loop1 = set()
    loop2 = set()
    win1.add(i)
    win2.add(i)
    for j in s1:
        win1.add((i + j - 1) % n + 1)
        lose1.add((i + j) % n + 1)
    for j in s2:
        win2.add((i + j - 1) % n + 1)
        lose2.add((i + j) % n + 1)
    for j in win1:
        if j in lose1:
            loop1.add(j)
    for j in win2:
        if j in lose2:
            loop2.add(j)
    lose1 -= loop1
    lose2 -= loop2
    if 1 in win1:
        ans1.append("Win")
    elif 1 in lose1:
        ans1.append("Lose")
    else:
        ans1.append("Loop")
    if 1 in win2:
        ans2.append("Win")
    elif 1 in lose2:
        ans2.append("Lose")
    else:
        ans2.append("Loop")
print(*ans1)
print(*ans2)
