#include <iostream>
using namespace std;

const int MOD = 1e9 + 7;

int fastpow(int x, int y) {
    int ret = 1;
    while(y) {
        if(y & 1) {
            ret = 1LL * ret * x % MOD;
        }
        x = 1LL * x * x % MOD;
        y >>= 1;
    }
    return ret;
}

int main() {
    int n; cin >> n;
    int arr[n + 1];
    for(int i = 0; i < n; i++) {
        cin >> arr[i + 1];
    }
    int dp[n + 1][3]; // dp[i][j] = number of ways to arrange i balls, where the ith ball has participated in j throws
    dp[0][0] = 1;
    for(int i = 1; i <= n; i++) {
        for(int j = 0; j < 3; j++) {
            if(j == 0) {
                dp[i][j] = (1LL * dp[i - 1][j] * i) % MOD;
            } else {
                dp[i][j] = (1LL * dp[i - 1][j - 1] * (i - 1)) % MOD;
            }
            if(arr[i] > j) {
                dp[i][j] = (dp[i][j] + dp[i - 1][j]) % MOD;
            }
        }
    }
    cout << dp[n][2] << '\n';
    return 0;
}
