#include <bits/stdc++.h>
using namespace std;

const int INF = 1e9;

int main() {
    long long d, n, m; cin >> d >> n >> m;
    vector<pair<long long, long long>> gs(m);
    for (auto& [x, p] : gs) { cin >> x >> p; }
    sort(gs.begin(), gs.end());

    long long ans = 0, cur = 0, tank = n;
    for (auto [x, p] : gs) {
        if (tank < x - cur) {
            if (ans == 0) ans = INF;
            ans += (x - cur) * gs[0].second;
            tank = n - (x - cur);
            cur = x;
        }
        ans += (x - cur) * p;
        tank -= (x - cur);
        cur = x;
    }
    if (tank < d - cur) ans = INF;
    cout << (ans == INF ? -1 : ans) << endl;
}
