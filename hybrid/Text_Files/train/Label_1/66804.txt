import sys
from collections import defaultdict

reader = (int(i) for i in sys.stdin)

n = next(reader)
a = [next(reader) for _ in range(n)]

G = defaultdict(list)
for x, y in zip(reader, reader):
    G[x].append(y)
    G[y].append(x)

def dfs(cur, par, res):
    res[cur - 1] = a[cur - 1]
    for u in G[cur]:
        if u != par:
            res = dfs(u, cur, res)
    for i in range(cur, n, cur):
        res[i] = min(res[i], a[cur - 1])
    return res

res = dfs(1, 0, [1] * n)
print(*res)
