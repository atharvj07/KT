#include <cstdio>
#include <vector>
using namespace std;

int main() {
    unsigned long long x;
    scanf("%llu", &x);
    vector<pair<long long, long long>> ans;
    for (long long i = 1; i * i <= x; ++i) {
        if (x % i == 0) {
            long long n, m;
            if (i > x / i)
                n = x / i, m = i;
            else
                n = i, m = x / i;
            long long t = n * m - (n + m - 1) / 2;
            if (t == x)
                ans.push_back({n, m});
        }
    }
    printf("%lu\n", ans.size());
    for (auto x : ans) printf("%lld %lld\n", x.first, x.second);
}
