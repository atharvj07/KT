n, m, k = map(int, input().split())
a = [[c == '1' for c in input()] for i in range(n)]
ans = m * n
for mask in range(1 << n):
    spent = 0
    skip = 0
    for i in range(n):
        if mask >> i & 1:
            skip += 1
            continue
        for j in range(m):
            if a[i][j]:
                spent += 1
                break
    if skip <= k:
        ans = min(ans, spent)
print(ans)
