import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextInt();
        long m = sc.nextInt();
        System.out.println(beautifulSequences(n, m));
    }

    private static long beautifulSequences(long n, long m) {
        long mod = 1000000007;
        long[][] dp = new long[(int) (n + 1)][(int) (m + 1)];
        for (int i = 1; i <= n; i++) {
            dp[i][0] = 1;
            dp[i][1] = 1;
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 2; j <= m; j++) {
                dp[i][j] = (dp[i - 1][j] + dp[i - 1][j - 1] * (i - j + 1)) % mod;
            }
        }
        return dp[(int) n][(int) m];
    }
}
