#include <iostream>
#include <vector>
#include <limits>
#include <algorithm>
using namespace std;

int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }

int main() {
    int q;
    cin >> q;
    while (q--) {
        int a;
        cin >> a;
        if (a & 1)
            cout << 1 << endl;
        else {
            int g = 0;
            for (int d = 2; d < a; d++) {
                if (a % d == 0) {
                    g = gcd(g, d);
                    if (g == 1) {
                        break;
                    }
                }
            }
            cout << g << endl;
        }
    }
}
