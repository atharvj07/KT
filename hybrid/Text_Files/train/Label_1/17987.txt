#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i].first >> v[i].second;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int delta = min(v[i].first, v[i].second);
    ans += delta;
    v[i].first += delta;
    v[i].second -= delta;
  }
  for (int i = 1; i < n; i++) {
    if (abs(v[i].first - v[i - 1].first) > 1) {
      cout << -1 << endl;
      return 0;
    }
  }
  cout << ans << endl;
  for (int i = 0; i < n; i++) {
    cout << v[i].first << " ";
  }
  cout << endl;
  return 0;
}
