#include <bits/stdc++.h>
#define ll long long
#define pb push_back
#define fi first
#define se second
#define sz(x) ((int)(x).size())
using namespace std;

const int N = 1e5 + 10;
ll a[N], b[N];

int main() {
    int n, q;
    cin >> n >> q;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    for (int i = 2; i <= n; i++) {
        b[i] = b[i - 1] + max(0LL, abs(a[i] - a[i - 1]));
    }
    while (q--) {
        int l, r;
        cin >> l >> r;
        cout << b[r] - b[l - 1] << '\n';
    }
    return 0;
}
