#include <iostream>
using namespace std;

int main() {
    long long a1, b1, a2, b2, L, R;
    cin >> a1 >> b1 >> a2 >> b2 >> L >> R;
    long long delta1 = a2 - a1, delta2 = b2 - b1;
    if (delta1 == 0 && delta2 == 0) {
        cout << R - L + 1;
        return 0;
    }
    if (delta1 == 0) {
        if (b1 >= L && b1 <= R)
            cout << 1;
        else
            cout << 0;
        return 0;
    }
    if (delta2 == 0) {
        if (b1 - L % delta1 == 0 && b1 - L >= 0 && b1 - L <= delta1 * (R - L)) {
            cout << 1;
            return 0;
        }
        cout << 0;
        return 0;
    }
    long long ans = 0;
    long long L1 = (L - b1 + delta1 - 1) / delta1;
    long long L2 = (L - b2 + delta2 - 1) / delta2;
    long long R1 = (R - b1) / delta1;
    long long R2 = (R - b2) / delta2;
    if (delta1 * L2 <= delta2 * L1) {
        ans += max(0LL, min(R1, (delta2 * R1 - delta1 * L2 + delta1 - 1) / delta1));
    }
    if (delta1 * R2 >= delta2 * L1) {
        ans += max(0LL, min(R2, (delta2 * R2 - delta1 * L1 + delta1 - 1) / delta1));
    }
    cout << ans;
    return 0;
}
