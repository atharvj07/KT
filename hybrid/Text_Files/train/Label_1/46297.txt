import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        if(k>2*(n-1)){
            System.out.println(-1);
            return;
        }
        int[] a = new int[n];
        int cnt = k / 2;
        int num = 2;
        for(int i=0; i<cnt; i++){
            a[i] = num;
            a[n-i-1] = num+1;
            num += 2;
        }
        if(k % 2 == 1){
            a[cnt] = 1;
        }
        for(int i=0; i<n; i++){
            System.out.print(a[i]+" ");
        }
    }
}
