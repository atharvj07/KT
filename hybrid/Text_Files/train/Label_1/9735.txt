#include <bits/stdc++.h>
using namespace std;
char grid[60][60];
int n, m;
bool row[55], col[55];
int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++) cin >> grid[i];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == '#') {
        row[i] = col[j] = true;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == '#') cnt++;
    }
    if (cnt == m && row[i] == false) {
      cout << "No" << endl;
      return 0;
    }
  }
  for (int j = 0; j < m; j++) {
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (grid[i][j] == '#') cnt++;
    }
    if (cnt == n && col[j] == false) {
      cout << "No" << endl;
      return 0;
    }
  }
  cout << "Yes" << endl;
  return 0;
}
