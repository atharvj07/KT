#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
const int maxn = 2e3 + 5;
string s, t;
int dp[maxn][maxn], occ[maxn][maxn];
int main() {
  cin >> s >> t;
  int n = s.length();
  int m = t.length();
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (s[i - 1] == t[j - 1]) {
        dp[i][j] = dp[i - 1][j - 1] + 1;
        occ[i][j] = occ[i - 1][j - 1] + 1;
      } else {
        dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
        occ[i][j] = max(occ[i - 1][j], occ[i][j - 1]);
      }
    }
  }
  for (int i = 0; i <= n; i++) {
    cout << occ[n][m] - occ[n - i][m] << " ";
  }
}
