#include <bits/stdc++.h>
using namespace std;
const int N = 100 + 5;
int n, m, k, p[N], a[N][N], pre[N], t;
int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= k; i++) {
        cin >> p[i];
        pre[p[i]] = i;
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) { t += pre[a[i][j]]; }
    }
    cout << t << endl;
}
