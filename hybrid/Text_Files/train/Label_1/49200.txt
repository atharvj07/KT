import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[n];
        int b_idx = 0;
        for (int i = 0; i < n; i++) {
            if (i + 1 < n && a[i] == a[i + 1]) {
                b[b_idx++] = a[i] + 1;
                i++;
            } else {
                b[b_idx++] = a[i];
            }
        }
        System.out.println(b_idx);
        for (int i = 0; i < b_idx; i++) {
            System.out.print(b[i] + " ");
        }
    }
}
