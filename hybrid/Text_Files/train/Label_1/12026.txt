#include <bits/stdc++.h>
#define ii pair<long long, long long>
using namespace std;
const long long INF = 1e18;

int main() {
    int n;
    cin >> n;
    vector<ii> a(n);
    for (int i = 0; i < n; i++) cin >> a[i].first >> a[i].second;
    sort(a.begin(), a.end(), [&](ii l, ii r) { return l.first < r.first; });
    long long ans = 0;
    long long mx = -INF;
    for (int i = 0; i < n; i++) {
        ans += max(a[i].second, mx - a[i].first);
        mx = max(mx, a[i].first + a[i].second);
    }
    cout << ans << endl;
    return 0;
}
