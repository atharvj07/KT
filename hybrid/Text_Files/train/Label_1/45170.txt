import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[m];
        int[] b = new int[m];
        int[] deg = new int[n + 1];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
            deg[a[i]]++;
            deg[b[i]]--;
        }
        int sum = 0;
        for (int i = 1; i <= n; i++) {
            sum += Math.max(0, deg[i]);
        }
        System.out.println(sum);
    }
}
