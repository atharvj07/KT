import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        long[] numbers = new long[k];
        for (int i = 0; i < k; i++) {
            numbers[i] = scanner.nextLong();
        }
        Arrays.sort(numbers);
        List<Long> chosenNumbers = new ArrayList<>();
        int count = 0;
        for (long number : numbers) {
            if (containsZero(number)) {
                chosenNumbers.add(number);
                count++;
            }
        }
        System.out.println(count);
        for (long number : chosenNumbers) {
            System.out.print(number + " ");
        }
        System.out.println();
    }

    public static boolean containsZero(long number) {
        while (number > 0) {
            if (number % 10 == 0) {
                return true;
            }
            number /= 10;
        }
        return false;
    }
}
