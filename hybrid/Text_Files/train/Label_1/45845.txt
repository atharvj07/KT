import java.util.*;

public class Main {
  static class Pair {
    int u, w;
    Pair(int u, int w) {
      this.u = u;
      this.w = w;
    }
  }

  static void bfs(int n, int k, ArrayList<ArrayList<Pair>> adj, int[] ans) {
    boolean[] vis = new boolean[n + 1];
    Queue<Pair> q = new LinkedList<>();
    q.add(new Pair(1, n));
    int cnt = 0;
    while (!q.isEmpty() && cnt < k) {
      Pair p = q.poll();
      if (vis[p.u]) continue;
      vis[p.u] = true;
      ans[cnt++] = p.u;
      for (Pair i : adj.get(p.u)) {
        if (!vis[i.u] && i.w == p.w) {
          q.add(new Pair(i.u, i.w));
        }
      }
    }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    ArrayList<ArrayList<Pair>> adj = new ArrayList<>();
    for (int i = 0; i <= n; i++) adj.add(new ArrayList<>());
    for (int i = 0; i < n - 1; i++) {
      int u = sc.nextInt();
      int v = sc.nextInt();
      adj.get(u).add(new Pair(v, u));
      adj.get(v).add(new Pair(u, v));
    }
    int[] ans = new int[k];
    bfs(n, k, adj, ans);
    for (int i : ans) System.out.print(i + " ");
  }
}
