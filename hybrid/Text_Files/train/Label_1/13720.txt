#include <vector>
#include <string>
int dx[] = {0, 1, 0, -1}, dy[] = {1, 0, -1, 0};
int main() {
    int n, m;
    scanf("%d%d", &n, &m);
    std::vector<std::string> ans(n, std::string(m, '.'));
    std::vector<std::vector<int>> a(n, std::vector<int>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            char ch;
            scanf(" %c", &ch);
            ans[i][j] = ch;
            a[i][j] = (ch == '*') ? -1 : 0;
        }
    }
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (a[i][j] < 0) {
                a[i][j] = ++cnt;
                for (int k = 0; k < 4; k++) {
                    int x = i + dx[k], y = j + dy[k];
                    if (x >= 0 and x < n and y >= 0 and y < m and a[x][y] == 0) {
                        a[x][y] = cnt;
                    }
                }
            }
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (a[i][j] < 0) {
                ans[i][j] = '*';
            } else if (a[i][j] > 0) {
                ans[i][j] = ((a[i][j] % 10) + '0');
            }
        }
    }
    for (auto &i: ans) {
        printf("%s\n", i.c_str());
    }
    return 0;
}
