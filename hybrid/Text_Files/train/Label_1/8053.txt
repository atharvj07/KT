#include <iostream>
#include <vector>

using std::vector;

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n;
        std::cin >> n;
        vector<int> v(n);
        vector<int> last(n, -1);
        for (int i = 0; i < n; ++i) {
            std::cin >> v[i];
            v[i]--;
        }
        int ans = 0;
        for (int i = 0; i < n; ++i) {
            if (last[v[i]] == -1) {
                last[v[i]] = i;
            } else {
                ans = std::max(ans, i - last[v[i]]);
                last[v[i]] = i;
            }
        }
        std::cout << ans << "\n";
    }
    return 0;
}
