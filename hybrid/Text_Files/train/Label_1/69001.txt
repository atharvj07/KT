from collections import defaultdict

n, m = map(int, input().split())
graph = defaultdict(list)
for _ in range(m): a, b = map(int, input().split()); graph[a].append(b); graph[b].append(a)
q = int(input())

ans = [0]
bad = set()
for who in map(int, [input() for _ in range(q)]):
    salary = [who]
    for b in graph[who]:
        if b in bad or salary[-1] < b: graph[b].remove(who); bad.add(who)
        elif salary[-1] == b: salary.append(b)
    if len(salary) >= 3: salary.sort(reverse = True)
    ans.append(len([1 for b in graph[salary[0]] if b in graph[salary[1]] if salary[1] in graph[b]]))
    bad.clear()

print(*ans)
