#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define pii pair<int,int>
#define vii vector<pii>
#define rep(i,a,b) for(int i=a ; i<b ; i++)
#define ff first
#define ss second
#define setBits(x) __builtin_popcount(x)

int n,m,p;
int grid[1002][1002];
bool vis[1002][1002];
int dx[4]={1,0,-1,0};
int dy[4]={0,1,0,-1};

bool isValid(int x,int y){
    if(x<0 || y<0 || x>=n || y>=m || grid[x][y]==-1 || vis[x][y]==true){
        return false;
    }
    return true;
}

int bfs(int x,int y,int s){
    queue<pii> q;
    q.push({x,y});
    vis[x][y]=true;
    int cnt=0;

    while(!q.empty()){
        int curX = q.front().ff;
        int curY = q.front().ss;
        q.pop();
        cnt++;

        for(int i=0 ; i<4 ; i++){
            int newX = curX + dx[i];
            int newY = curY + dy[i];

            if(isValid(newX,newY) && cnt<=s){
                q.push({newX,newY});
                vis[newX][newY]=true;
            }
        }
    }

    return cnt;
}

int main(){
    cin>>n>>m>>p;
    int speed[p+1];

    for(int i=1 ; i<=p ; i++) cin>>speed[i];

    for(int i=0 ; i<n ; i++){
        for(int j=0 ; j<m ; j++){
            char ch; cin>>ch;
            if(ch=='.') grid[i][j]=0;
            else if(ch=='#') grid[i][j]=-1;
            else grid[i][j]=(int)(ch-'0');
        }
    }

    int ans[p+1];
    memset(ans,0,sizeof(ans));

    for(int i=0 ; i<n ; i++){
        for(int j=0 ; j<m ; j++){
            if(grid[i][j]!=0 && vis[i][j]==false){
                int player = grid[i][j];
                ans[player]+=bfs(i,j,speed[player]);
            }
        }
    }

    for(int i=1 ; i<=p ; i++){
        cout<<ans[i]<<" ";
    }
    cout<<"\n";

    return 0;
}
