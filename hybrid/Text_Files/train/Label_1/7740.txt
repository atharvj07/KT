#include<bits/stdc++.h>
using namespace std;
const int maxn=100001;
char str[maxn];
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    int n,i,swapped=-1,ans=0;
    cin>>n>>str;
    char lst;
    for(i=0;i<n;i++){
        if(i==0){
            lst=str[i];
            continue;
        }
        if(str[i]!=lst){
            swapped=i-1;
            lst=str[i];
        }
    }
    for(i=swapped+1;i<n;i++)
        if(str[i]==str[i-1]) ans++;
    cout<<ans<<endl;
    return 0;
}
