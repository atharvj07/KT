#include <iostream>
#include <vector>
#include <algorithm>
#include <set>

using namespace std;

vector<vector<int> > v;

int main() {
    int m;
    cin >> m;
    set<int> s;
    for (int i = 0; i < m; ++i) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int j = 0; j < n; ++j) {
            cin >> a[j];
            s.insert(a[j]);
        }
        v.push_back(a);
    }
    int n;
    cin >> n;
    vector<int> b(n);
    for (int i = 0; i < n; ++i) {
        cin >> b[i];
        s.insert(b[i]);
    }
    vector<int> c(s.size());
    int j = 0;
    for (auto it = s.begin(); it != s.end(); it++) {
        c[j++] = *it;
    }
    int ans = n * *c.begin();
    for (auto it = v.begin(); it != v.end(); it++) {
        sort(it->begin(), it->end());
        int p = it->size();
        int x = *c.begin();
        for (int i = 0; i < min(p, 2); ++i) {
            x = min(x, it->at(i));
        }
        int l = lower_bound(c.begin(), c.end(), x) - c.begin();
        ans = min(ans, (*it->begin()) * (n / p) + n % p * c[l]);
    }
    cout << ans << endl;
    return 0;
}
