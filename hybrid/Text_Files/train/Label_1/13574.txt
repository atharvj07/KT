#include <bits/stdc++.h>
using namespace std;

int main() {
  int n; cin >> n;
  vector<int> a(n);
  for (int& i: a) cin >> i;
  int l = 0, r = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i]) r++;
    else l++;
    if (l == r) ans++;
  }
  cout << ans + (a[0] == 1 && a[n - 1] == 1) << endl;
}
