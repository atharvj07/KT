#include <bits/stdc++.h>
using namespace std;
#define ll long long
int n, ans = 0;
char s[100001];
int main() {
    cin >> n >> s;
    map<int, int> freq;
    int l = 0, r = 0;
    while (r <= n) {
        while (r <= n && freq[s[r]] == 0) freq[s[r++]]++;
        freq[s[r - 1]]--;
        ans = max(ans, r - l - 1);
        l = r;
        while (r <= n && freq[s[r]] != 0) freq[s[r++]]++;
        freq[s[r - 1]]--;
        ans = max(ans, r - l - 1);
        l = r;
    }
    cout << ans << endl;
    return 0;
}
