import java.util.Scanner;

public class SummerJob {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int f = scanner.nextInt();

        int[][] days = new int[n][2];
        for (int i = 0; i < n; i++) {
            days[i][0] = scanner.nextInt();
            days[i][1] = scanner.nextInt();
        }

        // Sort days by the difference between clients and products (descending order)
        sortDays(days);

        int maxProducts = 0;
        for (int i = 0; i < f; i++) {
            maxProducts += days[i][0] * 2; // Double products on chosen days
        }

        for (int i = f; i < n; i++) {
            maxProducts += days[i][0]; // Add products from remaining days
        }

        System.out.println(maxProducts);
    }

    private static void sortDays(int[][] days) {
        for (int i = 0; i < days.length - 1; i++) {
            for (int j = i + 1; j < days.length; j++) {
                if (days[j][1] - days[j][0] > days[i][1] - days[i][0]) {
                    swap(days, i, j);
                }
            }
        }
    }

    private static void swap(int[][] days, int i, int j) {
        int[] temp = days[i];
        days[i] = days[j];
        days[j] = temp;
    }
}
