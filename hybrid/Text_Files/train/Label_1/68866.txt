from collections import Counter
n=int(input())
a=[int(x) for x in input().split()]
c=Counter(a)
ans=[]
while len(ans)<n:
    b=[]
    for x in sorted(list(c.keys())):
        while c[x] and not (b and x==b[-1]):
            b.append(x)
            c[x]-=1
    while c[b[-1]]:
        b.pop()
    if len(ans)<len(b): ans=b
print(len(ans)**2)
print(len(ans),len(ans))
for x in ans:
    for y in range(len(ans)): print(x,end=' ')
    print()
