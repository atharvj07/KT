#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    stack<int> s;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      while (!s.empty() && x < s.top()) {
        s.pop();
        ans++;
      }
      s.push(x);
    }
    cout << ans << endl;
  }
  return 0;
}
