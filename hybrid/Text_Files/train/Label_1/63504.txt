from math import factorial

def count_permutations(n, k):
    def C(n, k):
        return factorial(n) // (factorial(k) * factorial(n - k))
    
    def D(n):
        if n == 0:
            return 1
        elif n == 1:
            return 0
        elif n == 2:
            return 1
        else:
            return (n - 1) * (D(n - 1) + D(n - 2))
    
    result = 0
    for i in range(k + 1):
        result += C(n, i) * D(i)
    return result

n, k = map(int, input().split())
print(count_permutations(n, k))
