t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    s = input().strip()
    if '?' in s:
        for i in range(n):
            if s[i] == '?':
                if i > 0 and s[i-1] != '?':
                    s = s[:i] + ('B' if s[i-1] == 'R' else 'R') + s[i+1:]
                elif i < n-1 and s[i+1] != '?':
                    s = s[:i] + ('B' if s[i+1] == 'R' else 'R') + s[i+1:]
        if '?' in s:  # If still '?' exists, means all were '?' or isolated '?'
            s = s.replace('?', 'B', 1)  # Replace the first '?' to start
            for i in range(1, n):  # After setting a start, we can alternate
                s = s[:i] + ('B' if s[i-1] == 'R' else 'R') if s[i] == '?' else s[:i+1]
    print(s)
