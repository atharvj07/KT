#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, d;
    cin >> n >> d;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    do {
      b = a;
      for (int i = 0; i < n; i++) { a[i] = b[(i + n - d) % n] & b[i]; }
    } while (a != b);
    int ans = -1;
    for (int i : a) {
      if (i) {
        ans = -1;
        break;
      }
      ans++;
    }
    cout << ans << '\n';
  }
  return 0;
}
