#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mxN=1e5+5;
char str[mxN];
bool is_abc[mxN];
void build(const int &n){
    memset(is_abc,0,n);
    for(int i=1;i<=n;++i){
        if(i>=3 && str[i]=='a' && str[i-1]=='b' && str[i-2]=='c')is_abc[i]=1;
    }
}
void go(const int &n,const int &idx,const char &ch){
    str[idx]=ch;
    is_abc[idx]=0;
    if(idx>=3){
        if(str[idx]=='a' && str[idx-1]=='b' && str[idx-2]=='c')is_abc[idx]=1;
        if(is_abc[idx-1]||is_abc[idx-2]){
            is_abc[idx]=1;
        }
    }
}
int main(){
    ios_base::sync_with_stdio(false);cin.tie(NULL);
    int n,q;
    cin >> n >> q;
    cin >> str+1;
    build(n);
    while(q--){
        int pos;
        char c;
        cin >> pos >> c;
        go(n,pos,c);
        int ans=0;
        for(int i=1;i<=n;++i){
            if(is_abc[i] && str[i]==str[i-1] && str[i]==str[i-2])++ans;
            else if(is_abc[i])++ans;
        }
        cout << ans << endl;
    }
    return 0;
}
