#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> p(n), b(n), mp(n), p2(n);
  for (int &x : p) cin >> x, mp[x] = 1;
  for (int i = 0; i < n; i++) p2[i] = mp[p[i]];
  for (int &x : b) cin >> x;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int x = -1;
    if (b[i])
      x = p[i];
    else
      for (int j = p2[i] - 1; j >= 0; j--) {
        if (p[j] == p[i]) break;
        if (p[j] < p[i]) { x = p[j]; break; }
      }
    if (x != -1) {
      ans++;
      p[p2[i] - 1] = x;
      p2[x] = p2[i];
      p2[i] = 0;
    }
  }
  cout << ans << endl;
}
