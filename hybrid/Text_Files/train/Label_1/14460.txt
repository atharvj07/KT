#include <bits/stdc++.h>
using namespace std;
string Solve(const string& s, int k) {
    string t(k, 'a');
    int idx = 0;
    for (int i = 0; i < (int)s.size() && idx < k; i++) {
        if (s[i] >= t[idx]) { t[idx++] = s[i]; }
    }
    return t;
}
int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    cout << Solve(s, k) << '\n';
    return 0;
}
