#include <iostream>
using namespace std;
int main() {
    long long n, p, cur = 1;
    cin >> n >> p;
    int ans = 0;
    while (n >= 0) {
        if (n & 1) ans++;
        if (p - (n & -n) < 0) break;
        n -= p - (n & -n);
        cur <<= 1;
    }
    if (n) ans = -1;
    cout << ans << endl;
}
