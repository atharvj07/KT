import java.util.*;
import java.io.*;

public class Main {
    static int[] dsu;

    public static void merge(int a, int b) {
        int da = dsu[a], db = dsu[b];
        if (da == db)
            return;
        if (da < db) {
            dsu[db] = da;
        } else {
            dsu[da] = db;
        }
    }

    public static int find(int a) {
        if (dsu[a] == a)
            return a;
        return dsu[a] = find(dsu[a]);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        for (int i = 0; i < t; i++) {
            String[] s = br.readLine().split(" ");
            int n = Integer.parseInt(s[0]), m = Integer.parseInt(s[1]);
            dsu = new int[n + 1];
            for (int j = 0; j <= n; j++)
                dsu[j] = j;
            String[] arr = new String[m];
            for (int j = 0; j < m; j++)
                arr[j] = br.readLine();
            boolean flag = true;
            for (int j = 0; j < m; j++) {
                int firstIndx = -1, lastIndx = -1;
                for (int k = 0; k < n; k++) {
                    if (arr[j].charAt(k) == '1' && lastIndx != -1) {
                        flag = false;
                        break;
                    }
                    if (arr[j].charAt(k) == '1') {
                        if (firstIndx == -1)
                            firstIndx = k;
                        lastIndx = k;
                    }
                }
                if (flag == false)
                    break;
                for (int k = firstIndx; k <= lastIndx; k++) {
                    if (arr[j].charAt(k) == '1') {
                        merge(k + 1, lastIndx + 1);
                    }
                }
            }
            if (!flag)
                System.out.println("NO");
            else {
                System.out.println("YES");
                Set<Integer> done = new HashSet<>();
                for (int j = 1; j <= n; j++) {
                    if (!done.contains(j)) {
                        int fa = find(j);
                        for (int k = 1; k <= n; k++) {
                            if (k != j && fa == find(k)) {
                                done.add(k);
                                System.out.println(j + " " + k);
                            }
                        }
                    }
                }
            }
        }
    }
}
