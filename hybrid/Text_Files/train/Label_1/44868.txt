import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    int k = Integer.parseInt(in.readLine());
    String[][] table = new String[4][4];
    for (int i = 0; i < 4; i++) {
      String line = in.readLine();
      for (int j = 0; j < 4; j++) {
        table[i][j] = line.substring(j, j + 1);
      }
    }
    boolean possible = true;
    for (int i = 0; i < 4; i++) {
      for (int j = 0; j < 4; j++) {
        if (table[i][j].equals(".")) {
          continue;
        }
        int time = Integer.parseInt(table[i][j]);
        if (time <= k) {
          possible = false;
          break;
        }
        if (i > 0 && table[i - 1][j].equals(".")) {
          table[i - 1][j] = "" + (time - 1);
        }
        if (i < 3 && table[i + 1][j].equals(".")) {
          table[i + 1][j] = "" + (time - 1);
        }
        if (j > 0 && table[i][j - 1].equals(".")) {
          table[i][j - 1] = "" + (time - 1);
        }
        if (j < 3 && table[i][j + 1].equals(".")) {
          table[i][j + 1] = "" + (time - 1);
        }
      }
    }
    System.out.println(possible ? "YES" : "NO");
  }
}
