import sys

def solve():
    cases = int(input())
    arr = list(map(int, input().split()))

    psum = []
    psum.append(arr[0])
    for i in range(1, len(arr)):
        psum.append(psum[i - 1] + arr[i])

    dp = {}

    def play(i, j):
        if (i, j) not in dp:
            if i == j:
                dp[(i, j)] = arr[i]
            else:
                dp[(i, j)] = max(arr[i] - play(i + 1, j), psum[j] - psum[i - 1] if i != 0 else 0 - play(i + 1, j))
        return dp[(i, j)]

    return play(0, len(arr) - 1)

print(solve())
