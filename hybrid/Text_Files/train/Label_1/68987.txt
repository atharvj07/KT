ch = list(input())
ans = False
for i in range(len(ch)):
    if ans: break
    if ch[i] == 'H':
        ans = True
        if i+1<len(ch):
            if ch[i+1] == 'Q':
                if i+2<len(ch):
                    if ch[i+2] == '9':
                        ans = False
                else: ans = False
        else: ans = False
print('Yes' if ans else 'No')
