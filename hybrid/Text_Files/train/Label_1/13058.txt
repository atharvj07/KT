#include<bits/stdc++.h>
#define LL long long int
using namespace std;
int main() {
    int n, t; cin >> n >> t;
    vector<int>v(n);
    for (int i = 0; i < n; i++)cin >> v[i];
    LL lo = 0, hi = 1e12, md;
    while (lo < hi) {
        md = (lo + hi + 1) / 2;
        LL now = 0;
        int cnt = 0;
        for (int i = 0; i < n && now + v[i] <= md; i++) {
            now += v[i];
            cnt++;
        }
        if (now + v[0] > md)cnt--;
        if (cnt >= t)lo = md;
        else hi = md - 1;
    }
    cout << lo << endl;
}
