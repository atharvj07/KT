import sys
input = sys.stdin.readline

n = int(input())
l, b = [], []
for _ in range(n):
  x1, y1, x2, y2 = map(int, input().split())
  l.append([x1, x2])
  b.append([y1, y2])

x = []
for i in range(n):
  for j in range(i + 1, n):
    if l[i][0] <= l[j][0] <= l[i][1] and b[i][0] <= b[j][0] <= b[i][1]:
      x.append([l[j][0], b[j][0]])

# print(x)

cnt = 0
ans = []

for c in x:
  t = []
  for i in range(n):
    if l[i][0] <= c[0] <= l[i][1] and b[i][0] <= c[1] <= b[i][1]:
      t.append(i)
  if len(set(t)) > cnt:
    cnt = len(set(t))
    ans = c

print(*ans)
