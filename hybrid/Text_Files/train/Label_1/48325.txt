import java.util.Scanner;
import java.util.TreeSet;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt();
        long[] power = new long[k];
        TreeSet<Long> powerSet = new TreeSet<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < k; j++) {
                power[j] = in.nextLong();
            }
            for (int j = 0; j < k; j++) {
                for (long powerValue : powerSet) {
                    long winnerPower = Math.max(powerValue, power[j]);
                    powerSet.add(winnerPower);
                }
                powerSet.add(power[j]);
            }
            System.out.println(powerSet.size());
            powerSet.clear();
        }
    }
}
