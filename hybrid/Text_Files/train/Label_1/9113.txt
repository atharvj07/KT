#include<iostream>
#include<string>
#include<vector>
using namespace std;

int main() {
	string s;
	cin >> s;
	vector<int> v(s.size() + 1, -1);
	int cnt = 0;
	for (int i = 0; i < s.size(); i++) {
		if (v[i] == -1) {
			int j = i;
			while (j < s.size() && s[j] == s[i]) j++;
			j--;
			for (int k = i; k <= j; k++)
				v[k] = cnt;
			cnt++;
			i = j;
		}
	}
	cout << cnt - 1 << endl;
	return 0;
}
