#include <iostream>
#include <vector>
#include <map>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  map<int, vector<pair<string, int>>> regions;
  for (int i = 0; i < n; ++i) {
    string surname;
    int region, score;
    cin >> surname >> region >> score;
    regions[region].emplace_back(surname, score);
  }

  for (int region = 1; region <= m; ++region) {
    auto& region_participants = regions[region];
    sort(region_participants.begin(), region_participants.end(),
         [](const auto& p1, const auto& p2) { return p1.second > p2.second; });

    if (region_participants.size() >= 2) {
      cout << region_participants[0].first << " " << region_participants[1].first << endl;
    } else {
      cout << '?' << endl;
    }
  }

  return 0;
}
