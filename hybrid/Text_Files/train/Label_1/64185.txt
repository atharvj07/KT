from collections import deque

def solve(n, edges):
    def bfs(start):
        dist = [-1] * (n + 1)
        q = deque([start])
        dist[start] = 0
        while q:
            node = q.popleft()
            for nei in graph[node]:
                if dist[nei] == -1:
                    dist[nei] = dist[node] + 1
                    q.append(nei)
        return dist

    graph = [[] for _ in range(n + 1)]
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    # Finding the two farthest nodes (ends of the diameter)
    dist1 = bfs(1)
    node1 = dist1.index(max(dist1))
    dist2 = bfs(node1)
    node2 = dist2.index(max(dist2))

    # Creating a relabeling
    label = [0] * (n + 1)
    for i in range(1, n + 1):
        if dist2[i] % 2 == 0:
            label[i] = 2 * dist2[i] + 1
        else:
            label[i] = 2 * (n - dist2[i])

    # Assigning unique labels
    unique_labels = sorted(range(1, n + 1), key=lambda x: label[x])
    relabeling = [0] * n
    for i, node in enumerate(unique_labels):
        relabeling[node - 1] = i + 1

    return relabeling

t = int(input())
for _ in range(t):
    n = int(input())
    edges = [tuple(map(int, input().split())) for _ in range(n - 1)]
    result = solve(n, edges)
    print(*result)
