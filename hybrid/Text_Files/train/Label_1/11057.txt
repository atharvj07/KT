#include <iostream>
#include <string>
#include <algorithm>

using namespace std;

int main() {
    string s;
    int k;
    cin >> s >> k;
    k %= 26;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] >= 'a' && s[i] <= 'z') {
            s[i] = (s[i] - 'a' + k) % 26 + 'a';
        } else if (s[i] >= 'A' && s[i] <= 'Z') {
            s[i] = (s[i] - 'A' + k) % 26 + 'A';
        }
    }
    cout << s;
    return 0;
}
