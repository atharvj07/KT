#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> r(n);
    for (int i = 0; i < n; i++) cin >> r[i];
    int sum1 = 0, sum2 = 0;
    for (int i = 0; i < n; i++) {
      if (r[i] == 1 || (r[i] == 3 && sum1 <= sum2)) sum1++;
      if (r[i] == 2 || (r[i] == 3 && sum1 > sum2)) sum2++;
    }
    cout << sum1 + sum2 << endl;
  }
}
