import sys

s = sys.stdin.readline().strip()
t = sys.stdin.readline().strip()

if not t in s:
    print(0)
else:
    i = 0
    ans = ''
    while i < len(s):
        if s[i] == t[0]:
            if s[i:i+len(t)] == t:
                ans += t
                i += len(t)
            else:
                ans += s[i]
                i += 1
        else:
            ans += s[i]
            i += 1
    print(ans)
