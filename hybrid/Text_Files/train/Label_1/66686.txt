import sys

n, m, k = map(int, input().split())
w = list(map(int, input().split()))

ab = []
for _ in range(m):
    a, b = map(int, input().split())
    a -= 1
    b -= 1
    ab.append((w[a] + w[b], a, b))

ab.sort(reverse=True)

s = [[0] * k for _ in range(n)]
T = [0] * m

for i, (w, a, b) in enumerate(ab):
    st = []
    for j in range(k):
        st.append((s[a][j] - s[b][j], j))

    st.sort()
    l, r = st[0][1], st[-1][1]

    if s[a][r] - s[a][l] > 2 or s[b][r] - s[b][l] > 2:
        continue

    T[i] = st[0][1] + 1
    s[a][st[0][1]] += 1
    s[b][st[0][1]] += 1

print(' '.join(map(str, T)))
