import java.util.*;

class Main {
    static int[] p;
    static int[] parent;
    static int[] rank;

    static void initialize(int n) {
        p = new int[n + 1];
        parent = new int[n + 1];
        rank = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = i;
            parent[i] = i;
            rank[i] = 0;
        }
    }

    static int find(int x) {
        if (x == parent[x])
            return x;
        return parent[x] = find(parent[x]);
    }

    static void union(int a, int b) {
        int pA = find(a);
        int pB = find(b);

        if (pA == pB)
            return;
        if (rank[pA] < rank[pB]) {
            parent[pA] = pB;
        } else if (rank[pB] < rank[pA]) {
            parent[pB] = pA;
        } else {
            parent[pB] = pA;
            rank[pA] += 1;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            arr[i] = scanner.nextInt();
        }
        int count = 1;
        initialize(n);
        for (int i = 0; i < n - 1; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();

            union(a, b);
        }

        HashMap<Integer, Integer> map = new HashMap<>();

        for (int i = 1; i <= n; i++) {
            int par = find(i);
            map.put(par, map.getOrDefault(par, 0) + 1);
        }

        for (Map.Entry<Integer, Integer> entry : map.entrySet()) {
            count = Math.max(count, entry.getValue());
        }

        System.out.println(count);
    }
}
