import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n], b = new int[n];
        for (int i = 0; i < n; i++) {
            String[] s = br.readLine().split(" ");
            a[i] = Integer.parseInt(s[0]);
            b[i] = Integer.parseInt(s[1]);
        }
        int[] p = new int[n];
        int[] order = new int[n];
        boolean unique = true, ok = true;
        for (int i = 0; i < n; i++) {
            if (p[a[i]-1] > 0) {
                unique = false;
            } else {
                p[a[i]-1] = 1;
            }
            if (p[b[i]-1] < 0) {
                unique = false;
            } else {
                p[b[i]-1] = -1;
            }
        }
        for (int i = 0; i < n; i++) {
            if (p[i] == 0) {
                ok = false;
                break;
            }
        }
        if (unique) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                for (int j = a[i]-1; j <= b[i]-1; j++) {
                    if (order[j] == 0) {
                        order[j] = i+1;
                        break;
                    }
                }
            }
            for (int i = 0; i < n; i++) {
                System.out.print(order[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("NO");
            for (int i = 0; i < n; i++) {
                for (int j = a[i]-1; j <= b[i]-1; j++) {
                    if (order[j] == 0) {
                        order[j] = i+1;
                        break;
                    }
                }
            }
            for (int i = 0; i < n; i++) {
                System.out.print(order[i] + " ");
            }
            System.out.println();
            for (int i = 0; i < n; i++) {
                order[i] = 0;
            }
            for (int i = 0; i < n; i++) {
                for (int j = b[i]-1; j >= a[i]-1; j--) {
                    if (order[j] == 0) {
                        order[j] = i+1;
                        break;
                    }
                }
            }
            for (int i = 0; i < n; i++) {
                System.out.print(order[i] + " ");
            }
            System.out.println();
        }
    }
}
