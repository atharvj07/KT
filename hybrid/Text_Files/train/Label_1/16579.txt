#include <bits/stdc++.h>
using namespace std;

int t[100005][22];
int lg[100005];

void build(int n, int a[]) {
  for (int i = 0; i < n; i++) { t[i][0] = a[i]; }

  for (int j = 1; j <= 20; j++) {
    for (int i = 0; i + (1 << j) <= n; i++) {
      t[i][j] = max(t[i][j - 1], t[i + (1 << (j - 1))][j - 1]);
    }
  }

  for (int i = 2; i <= n; i++) { lg[i] = lg[i / 2] + 1; }
}

int query(int l, int r) {
  int j = lg[r - l + 1];
  return max(t[l][j], t[r - (1 << j) + 1][j]);
}

int main() {
  int n;
  cin >> n;
  int a[n + 5];
  for (int i = 0; i < n; i++) cin >> a[i];

  build(n, a);
  long long ans = 0;
  for (int l = 1; l <= n; l++) {
    for (int r = l; r <= n; r++) {
      if (query(l - 1, r - 1) >= l && query(l - 1, r - 1) <= r) ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
