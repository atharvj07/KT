import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] strengths = new int[n];
        for (int i = 0; i < n; i++) {
            strengths[i] = in.nextInt();
        }
        int m = in.nextInt();
        boolean[] picked = new boolean[n];
        int[] team1 = new int[m / 2];
        int[] team2 = new int[m / 2];
        int j = 0;
        int k = 0;
        for (int i = 0; i < m; i++) {
            String action = in.next();
            int team = in.nextInt() - 1;
            if (action.equals("p")) {
                if (team == 0) {
                    for (int l = 0; l < n; l++) {
                        if (!picked[l]) {
                            team1[j++] = strengths[l];
                            picked[l] = true;
                            break;
                        }
                    }
                } else {
                    for (int l = 0; l < n; l++) {
                        if (!picked[l]) {
                            team2[k++] = strengths[l];
                            picked[l] = true;
                            break;
                        }
                    }
                }
            }
        }
        int sum1 = 0;
        for (int i = 0; i < team1.length; i++) {
            sum1 += team1[i];
        }
        int sum2 = 0;
        for (int i = 0; i < team2.length; i++) {
            sum2 += team2[i];
        }
        System.out.println(sum1 - sum2);
    }
}
