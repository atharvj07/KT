from collections import defaultdict
from sys import stdin
read = stdin.readline
N, M = map(int, read().split())
T = list(map(int, read().split()))
E = [list(map(int, read().split())) for _ in range(M)]
G = defaultdict(set)
for l, r in E:
    G[l].add(r)
    G[r].add(l)
cnt = defaultdict(lambda: 0)
for v in G:
    for u in G[v]:
        if T[v - 1] != T[u - 1]:
            cnt[T[v - 1]] += 1
res = []
num = -1
for k, c in cnt.items():
    if c > num:
        num = c
        res = []
        res.append(k)
    elif c == num:
        res.append(k)
print(min(res))
