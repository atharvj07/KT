import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        long sum = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            sum += a[i] * i;
        }
        long max = sum;
        for (int i = 0; i < n; i++) {
            max = Math.max(max, sum + a[i] * (i - n));
        }
        System.out.println(max);
    }
}
