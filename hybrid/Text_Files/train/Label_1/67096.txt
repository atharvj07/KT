n, m = map(int, input().split())
strings = [input().strip() for _ in range(n)]

palindrome_parts = []
middle_part = ''
used = [False] * n

for i in range(n):
    if used[i]:
        continue
    for j in range(i + 1, n):
        if strings[i] == strings[j][::-1]:
            palindrome_parts.append(strings[i])
            used[i] = True
            used[j] = True
            break

for i in range(n):
    if not used[i] and strings[i] == strings[i][::-1]:
        middle_part = strings[i]
        break

palindrome = ''.join(palindrome_parts) + middle_part + ''.join(palindrome_parts[::-1])
print(len(palindrome))
print(palindrome)
