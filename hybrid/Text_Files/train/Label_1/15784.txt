#include<iostream>

using namespace std;

int main() {
    int n;
    bool vacant[1001];
    char c;
    bool maximal;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> c;
        vacant[i] = c == '0';
    }
    maximal = vacant[1] && vacant[n];
    for (int i = 2; i < n; i++)
        maximal &= vacant[i] == (vacant[i - 1] || vacant[i + 1]);
    cout << (maximal ? "Yes" : "No") << endl;
    return 0;
}
