import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Scanner;

public class RocketSafety {

    private static class Point {
        double x;
        double y;

        public Point(double x, double y) {
            this.x = x;
            this.y = y;
        }

        public Point rotate(double theta) {
            double newX = x * Math.cos(theta) - y * Math.sin(theta);
            double newY = x * Math.sin(theta) + y * Math.cos(theta);
            return new Point(newX, newY);
        }

        public Point shift(double a, double b) {
            return new Point(x + a, y + b);
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Point> engine1 = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            engine1.add(new Point(scanner.nextInt(), scanner.nextInt()));
        }

        List<Point> engine2 = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            engine2.add(new Point(scanner.nextInt(), scanner.nextInt()));
        }

        if (isSafe(engine1, engine2)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isSafe(List<Point> engine1, List<Point> engine2) {
        HashSet<Point> powerField = generatePowerField(engine1, engine2);
        for (Point point : engine1) {
            if (!powerField.equals(generatePowerField(removePoint(engine1, point), engine2))) {
                return false;
            }
        }
        for (Point point : engine2) {
            if (!powerField.equals(generatePowerField(engine1, removePoint(engine2, point)))) {
                return false;
            }
        }
        return true;
    }

    private static HashSet<Point> generatePowerField(List<Point> engine1, List<Point> engine2) {
        HashSet<Point> powerField = new HashSet<>();
        for (Point point1 : engine1) {
            for (Point point2 : engine2) {
                for (double k = 0; k <= 1; k += 0.01) {
                    powerField.add(new Point(k * point1.x + (1 - k) * point2.x, k * point1.y + (1 - k) * point2.y));
                }
            }
        }
        return powerField;
    }

    private static List<Point> removePoint(List<Point> engine, Point point) {
        List<Point> newEngine = new ArrayList<>(engine);
        newEngine.remove(point);
        return newEngine;
    }
}
