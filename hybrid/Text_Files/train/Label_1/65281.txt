n, a, b = map(int, input().split())
ghosts = [tuple(map(int, input().split())) for _ in range(n)]

relative_velocities = {}

for i in range(n):
    for j in range(i + 1, n):
        dx = ghosts[j][0] - ghosts[i][0]
        dvx = ghosts[j][1] - ghosts[i][1]
        dvy = ghosts[j][2] - ghosts[i][2] - a * dvx
        if dvx == 0 and dvy == 0:
            continue
        if dvx == 0:
            relative_velocities[('inf', dvy)] = relative_velocities.get(('inf', dvy), 0) + 1
        else:
            slope = dvy / dvx
            relative_velocities[(dvx, slope)] = relative_velocities.get((dvx, slope), 0) + 1

GX = sum(v * (v + 1) for v in relative_velocities.values())
print(GX)
