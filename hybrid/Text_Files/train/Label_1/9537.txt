#include <bits/stdc++.h>
using namespace std;
struct ret { string s; int mx;};
ret f(string s) {
  vector<int> cnt(26, 0);
  int lax = -1;
  for (char c : s) cnt[c - 'a']++;
  for (int i = 0; i < 26; i++)
    if (cnt[i] >= 2) lax = max(lax, i + 'a');
  if (lax != -1) {
    string t;
    for (char c : s)
      if (c != lax) t.push_back(c);
    ret l = f(t);
    l.s.push_back(lax);
    return l;
  }
  ret r; r.s = s; r.mx = 0;
  return r;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    ret ans = f(s);
    cout << ans.s << "\n";
  }
  return 0;
}
