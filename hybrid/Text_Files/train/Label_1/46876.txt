import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int K = sc.nextInt();
        int[] A = new int[M];
        int[] B = new int[M];
        int[] C = new int[K];
        int[] D = new int[K];
        for (int i = 0; i < M; i++) {
            A[i] = sc.nextInt();
            B[i] = sc.nextInt();
        }
        for (int i = 0; i < K; i++) {
            C[i] = sc.nextInt();
            D[i] = sc.nextInt();
        }
        int[] cnt = new int[N + 1];
        for (int ai : A) cnt[ai]++;
        for (int bi : B) cnt[bi]++;
        for (int ci : C) cnt[ci]++;
        for (int di : D) cnt[di]++;
        for (int i = 1; i <= N; i++) {
            int c = 0;
            for (int j = 0; j < M; j++) {
                if (A[j] == i || B[j] == i) {
                    c |= (1 << j);
                }
            }
            for (int j = 0; j < K; j++) {
                if (C[j] == i || D[j] == i) {
                    c |= (1 << j);
                }
            }
            if (cnt[i] == 0) {
                System.out.print(N - 1 + " ");
            } else if (c == (1 << (M + K)) - 1) {
                System.out.print(0 + " ");
            } else {
                System.out.print(N - cnt[i] - 1 + " ");
            }
        }
    }
}
