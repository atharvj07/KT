import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), c = in.nextInt(), x[] = new int[n], dp[] = new int[n];
        for (int i = 0; i < n; i++) x[i] = in.nextInt();
        for (int i = 1; i < n; i++) {
            dp[i] = dp[i - 1];
            if (x[i] > x[i - 1] + c) dp[i] = Math.max(dp[i], x[i] - x[i - 1] - c);
            if (x[i] > x[0] + c*(i)) dp[i] = Math.max(dp[i], x[i] - x[0] - c*i);
        }
        System.out.println(dp[n - 1]);
    }
}
