#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> ans(n, 2);
    for (int i = 1; i < n; i++) {
      if (s[i] == s[i - 1]) ans[i] = 1;
    }
    if (s[0] == s[n - 1]) ans[0] = 1;
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (ans[i] == 2) cnt++;
    }
    cout << cnt / 2 << endl;
  }
  return 0;
}
