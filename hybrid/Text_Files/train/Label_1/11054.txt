#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    int a[n], b[n], c[n], d[n];
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    b[0] = a[0];
    c[n - 1] = a[n - 1];
    for (int i = 1; i < n; ++i) b[i] = a[i] + b[i - 1];
    for (int i = n - 2; i >= 0; --i) c[i] = a[i] + c[i + 1];
    d[0] = b[0];
    for (int i = 1; i < n; ++i) d[i] = min(b[i], c[i]);
    int flag = 0;
    for (int i = 0; i < n; ++i) {
        if (a[i] > d[i]) flag = 1;
    }
    if (flag) cout << "NO" << endl;
    else
        cout << "YES" << endl;
    return 0;
}
