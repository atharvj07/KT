#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    string t;
    cin >> t;
    string s;
    while (!t.empty()) {
        if (t.front() < t.back()) {
            s += t.back();
            t.pop_back();
        } else {
            s = t.front() + s;
            t.erase(t.begin());
        }
    }
    cout << s << endl;
    return 0;
}
