#include <bits/stdc++.h>
#define m first
#define mx second
using namespace std;
const int NN = 1<<10;
const int mod = 1e9+7;
typedef long long ll;
typedef pair<int, int> pi;
int n,v[NN],mn[NN],mx[NN];
void md(int x, int &a, int &b, bool){
    a += (1<<(x-mn[x]));
    b += (1<<(x-mx[x]));
}
int main(){
    ios::sync_with_stdio(0), cin.tie(0);
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>v[i];
        mn[v[i]] = max(mn[v[i]], i);
        mx[v[i]] = i;
    }
    int a = 0, b = 0;
    for(int i=1;i<=1000;i++){
        if(mn[i] != mx[i]){
            md(i, a, b, true);
        }
    }
    int ans = __builtin_popcount(a|b);
    cout<<ans<<endl;
}
