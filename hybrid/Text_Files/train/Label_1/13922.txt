#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> r(n);
  for (int i = 0; i < n; i++) { cin >> r[i]; }
  sort(r.begin(), r.end());
  int R = r[0];
  int cnt = 0;
  for (int i = 0; i < n; i++) { cnt += max(0, R - r[i]); }
  cout << R << endl;
  cout << cnt << endl;
  for (int i = 0; i < cnt; i++) {
    for (int j = 0; j < n; j++) {
      cout << (r[j] == R ? 0 : 1);
      r[j] = max(0, r[j] - 1);
    }
    cout << endl;
  }
  return 0;
}
