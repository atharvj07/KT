n, m, k = map(int, input().split())
matrix = [list(map(int, input().split())) for _ in range(n)]
notifications = [0] * n

for _ in range(k):
    x, y = map(int, input().split())
    for i in range(n):
        if matrix[i][y - 1] == 1:
            if i != x - 1:
                notifications[i] += 1

print(' '.join(map(str, notifications)))
