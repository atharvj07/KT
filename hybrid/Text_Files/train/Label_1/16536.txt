#include<iostream>
#include<queue>
using namespace std;
using ll = long long;

int main() {
    ios_base::sync_with_stdio(0), cin.tie(0);
    int t, x, y;
    cin >> t;
    while (t--) {
        cin >> x >> y;
        int ans = max(x, y) * 2 + (x + y) % 2, a0 = ans >> 1;
        if (x <= a0 && y >= a0) --ans, ++x, --y;
        if ((x - a0) % 2 != (y - a0) % 2) { --ans; ++x; }
        cout << ans << '\n';
    }
}
