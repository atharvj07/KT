import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Scanner;
import java.util.Set;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Point> points = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            points.add(new Point(x, y));
        }
        Set<Point> goodPoints = new HashSet<>();
        for (Point point : points) {
            if (goodPoints.isEmpty()) {
                goodPoints.add(point);
                continue;
            }
            boolean isGood = false;
            for (Point goodPoint : goodPoints) {
                if (point.x == goodPoint.x || point.y == goodPoint.y || isRectangleGood(point, goodPoint, points)) {
                    isGood = true;
                    break;
                }
            }
            if (!isGood) {
                goodPoints.add(point);
            }
        }
        System.out.println(goodPoints.size());
        for (Point point : goodPoints) {
            System.out.println(point.x + " " + point.y);
        }
    }

    private static boolean isRectangleGood(Point point1, Point point2, List<Point> points) {
        int minX = Math.min(point1.x, point2.x);
        int maxX = Math.max(point1.x, point2.x);
        int minY = Math.min(point1.y, point2.y);
        int maxY = Math.max(point1.y, point2.y);
        for (Point point : points) {
            if (point.x > minX && point.x < maxX && point.y > minY && point.y < maxY) {
                return true;
            }
        }
        return false;
    }

    private static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
