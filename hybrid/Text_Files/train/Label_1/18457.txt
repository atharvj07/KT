#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
int main(){
    int n; cin >> n;
    unordered_map<int, int> umap;
    for(int i = 1; i <= n; i++){
        int x; cin >> x;
        umap[x] = i;
    }
    vector<int> v;
    for(int i = 1; i <= n; i++){
        v.push_back(umap[i]);
    }
    int lis = 0;
    vector<int> dp(n + 1, 0);
    for(int i = 1; i <= n; i++){
        dp[i] = 1;
        for(int j = 1; j < i; j++){
            if(v[i] > v[j] && dp[i] < dp[j] + 1){
                dp[i] = dp[j] + 1;
            }
        }
        lis = max(lis, dp[i]);
    }
    cout << lis << endl;
    return 0;
}
