#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int mx = 1;
    int l = 0, r = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'G') {
            r++;
        } else {
            r = 0;
        }
        mx = max(mx, r);
    }
    l = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == 'G') {
            l++;
        } else {
            l = 0;
        }
        mx = max(mx, l);
    }
    for (int i = 0; i < n; i++) {
        if (s[i] == 'S') {
            int cntl = 0, cntr = 0;
            for (int j = i - 1; j >= 0; j--) {
                if (s[j] == 'G') {
                    cntl++;
                } else {
                    break;
                }
            }
            for (int j = i + 1; j < n; j++) {
                if (s[j] == 'G') {
                    cntr++;
                } else {
                    break;
                }
            }
            mx = max(mx, cntl + cntr + 1);
        }
    }
    cout << mx << endl;
    return 0;
}
