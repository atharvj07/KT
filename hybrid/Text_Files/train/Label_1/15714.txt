#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    if (n - 1 < k || n / 2 < k)
      cout << -1 << endl;
    else {
      vector<int> a(n);
      for (int i = 0; i < n - k - 1; i++) a[i] = i + 1;
      int r = k + (n - k - 1) % 2;
      for (int i = n - k - 1; i < n; i++) a[i] = r++;
      for (int x : a) cout << x << ' ';
      cout << endl;
    }
  }
  return 0;
}
