n=int(input())
s=list(map(int,input().split()))
c=list(map(int,input().split()))
a=[0]*n
d=[0]*n
for i in range(1,n):
    if s[i]>s[i-1]:
        a[i]=a[i-1]+1
for i in range(n-2,-1,-1):
    if s[i]<s[i+1]:
        d[i]=d[i+1]+1
ans=1e9
for i in range(n):
    if a[i]>0 and d[i]>0:
        ans=min(ans,c[i-a[i]]+c[i]+c[i+d[i]])
print(-1 if ans==1e9 else ans)
