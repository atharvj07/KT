import java.util.Scanner;
import java.util.LinkedList;
import java.util.Queue;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String string = scanner.nextLine();
        Queue<Integer> queue = new LinkedList<>();
        if (string.charAt(9) == '0') queue.add(-1);
        for (int i = 0; i < 10; i++) {
            if (string.charAt(i) == '1') queue.add(i + 1);
        }
        if (queue.size() < 2) {
            System.out.println("NO");
            return;
        }
        int m = scanner.nextInt();
        int[] array = new int[m];
        int cnt = 0;
        int sum1 = 0, sum2 = 0;
        for (int i = 0; i < m; i++) {
            int weight = queue.poll();
            if (weight == -1) {
                System.out.println("NO");
                return;
            }
            array[i] = weight;
            if (cnt % 2 == 0) {
                sum1 += weight;
                if (sum1 < sum2) {
                    System.out.println("NO");
                    return;
                }
            } else {
                sum2 += weight;
                if (sum2 < sum1) {
                    System.out.println("NO");
                    return;
                }
            }
            cnt++;
            Integer next = queue.poll();
            if (next != null && next != weight) {
                queue.add(next);
            }
        }
        System.out.println("YES");
        for (int a : array) {
            System.out.print(a + " ");
        }
    }
}
