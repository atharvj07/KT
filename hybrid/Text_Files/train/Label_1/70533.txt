def min_power(n, k, A, B, avengers):
    avengers.sort()
    def destroy(l, r, avengers):
        if not avengers:
            return A
        if l == r:
            return B * len(avengers) * (r-l+1)
        mid = (l + r) // 2
        left_avengers = [a for a in avengers if a <= mid]
        right_avengers = [a for a in avengers if a > mid]
        return min(B * len(avengers) * (r-l+1), destroy(l, mid, left_avengers) + destroy(mid+1, r, right_avengers))
    return destroy(1, 2**n, avengers)

n, k, A, B = map(int, input().split())
avengers = list(map(int, input().split()))
print(min_power(n, k, A, B, avengers))
