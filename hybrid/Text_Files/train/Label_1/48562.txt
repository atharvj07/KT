import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[12];
        for (int i = 0; i < 12; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        int[] sides = new int[6];
        sides[0] = arr[0] * arr[1];
        sides[1] = arr[2] * arr[3];
        sides[2] = arr[4] * arr[5];
        sides[3] = arr[1] * arr[6];
        sides[4] = arr[3] * arr[5];
        sides[5] = arr[6] * arr[7];
        Arrays.sort(sides);
        if (sides[0] != sides[5]) {
            System.out.println("no");
        } else {
            System.out.println("yes");
        }
    }
}
