def can_form_palindrome(s):
    char_count = [0] * 26
    for char in s:
        char_count[ord(char) - ord('a')] += 1
    odd_count = sum(count % 2 for count in char_count)
    return odd_count <= 1

def concat_and_check(a, b):
    return can_form_palindrome(a + b)

def count_palindrome_pairs(words):
    count = 0
    for i in range(len(words)):
        for j in range(i + 1, len(words)):
            if concat_and_check(words[i], words[j]) or concat_and_check(words[j], words[i]):
                count += 1
    return count

N = int(input().strip())
words = [input().strip() for _ in range(N)]
print(count_palindrome_pairs(words))
