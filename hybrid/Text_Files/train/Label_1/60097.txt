import sys

def solve():
    t = int(input())
    for _ in range(t):
        n, m = map(int, input().split())
        grid = [list(input()) for _ in range(n)]
        if not valid(grid):
            print("NO")
        else:
            print("YES")
            for row in grid:
                for cell in row:
                    if cell == '.':
                        if row.count('R') > row.count('W'):
                            print('R', end='')
                        else:
                            print('W', end='')
                    else:
                        print(cell, end='')
                print()

def valid(grid):
    n = len(grid)
    m = len(grid[0])
    for i in range(n):
        for j in range(m):
            if grid[i][j] == '.':
                continue
            cntR = 0
            cntW = 0
            if i > 0 and grid[i-1][j] == 'R':
                cntR += 1
            if i < n-1 and grid[i+1][j] == 'R':
                cntR += 1
            if j > 0 and grid[i][j-1] == 'R':
                cntR += 1
            if j < m-1 and grid[i][j+1] == 'R':
                cntR += 1
            if i > 0 and grid[i-1][j] == 'W':
                cntW += 1
            if i < n-1 and grid[i+1][j] == 'W':
                cntW += 1
            if j > 0 and grid[i][j-1] == 'W':
                cntW += 1
            if j < m-1 and grid[i][j+1] == 'W':
                cntW += 1
            if grid[i][j] == 'R' and cntW > 0:
                return False
            if grid[i][j] == 'W' and cntR > 0:
                return False
    return True

solve()
