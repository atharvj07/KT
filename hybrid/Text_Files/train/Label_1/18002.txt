#include <bits/stdc++.h>
using namespace std;

#define all(x)          (x).begin(),(x).end()
#define Unique(x)       sort(all(x)), x.erase(unique(all(x)), x.end())
#define el              "\n"
#define debug(x)        cout << #x << " is " << x << el;

const long long  N = 5000, OO = 0x3f3f3f3f, MOD = 1e9 + 7;
const double PI = acos(-1), EPS = 1e-9;

int main() {
  ios::sync_with_stdio(false), cin.tie(nullptr), cout.tie(nullptr);

  int n; cin >> n;
  map<vector<int>, int> mp;
  vector<int> days[N];

  while (n--) {
    char gender; int l, r;
    cin >> gender >> l >> r;
    vector<int> range;
    if (gender == 'M')
      range = {l, -r, 1};
    else
      range = {l, r, -1};
    days[l].push_back(range[1]);
    days[r + 1].push_back(range[1]);
    mp[range]++;
  }

  int ans = 0;
  for (int i = 1; i <= 366; i++) {
    int sum = 0;
    for (auto &period : days[i])
      sum += period;

    ans = max(ans, abs(sum));
  }

  cout << ans / 2 << el;
  return 0;
}
