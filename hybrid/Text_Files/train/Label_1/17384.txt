#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll INF = 1e18;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<pair<ll, ll>> T(n);
    vector<ll> TM(n);
    for (int i = 0; i < n; i++) cin >> T[i].first >> T[i].second;
    for (ll &x : TM) cin >> x;
    ll cur = 0;
    for (int i = 0; i < n; i++) {
      cur = max(cur, T[i].first);
      cur += TM[i];
      cur += (T[i].second - T[i].first + 1) / 2;
    }
    cout << cur << '\n';
  }
}
