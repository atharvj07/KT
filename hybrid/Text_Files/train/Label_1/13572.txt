#include<iostream>
using namespace std;
int n,m[100005],d[100005];
int main(){
    cin >> n;
    for(int i = 1;i <= n;i++){
        cin >> m[i];
    }
    d[n + 1] = 0;
    for(int i = n;i >= 1;i--){
        d[i] = d[i + 1] + 1;
        d[i] = min(d[i],i - m[i] - 1);
    }
    long long ans = 0;
    for(int i = 1;i <= n;i++)ans += d[i];
    cout << ans << endl;
    return 0;
}
