import sys

for t in range(int(input())):
    n, m, k = map(int, input().split())
    grid = [list(input()) for _ in range(n)]

    for i in range(n):
        for j in range(m):
            if grid[i][j] == '*' and (i < k or j < k or i >= n - k or j >= m - k or grid[i - k][j - k] != '*' or grid[i - k][j + k] != '*'):
                print("NO")
                sys.exit()

    print("YES")
