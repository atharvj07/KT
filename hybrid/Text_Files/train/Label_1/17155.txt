#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    int m;
    cin >> n >> m;
    vector<long long> a(m);
    for (int i = 0; i < m; i++) {
      cin >> a[i];
    }
    sort(a.begin(), a.end());
    long long ans = 0;
    for (int i = 0; i < m; i++) {
      long long sum = a[i];
      int j = i + 1;
      while (j < m && sum * 2 <= n) {
        sum *= 2;
        j++;
      }
      if (sum <= n) {
        ans = 0;
        break;
      }
      long long div = n;
      while (div > a[i]) {
        ans++;
        div /= 2;
      }
      i = j - 1;
    }
    cout << (ans == LLONG_MAX ? -1 : ans) << '\n';
  }
  return 0;
}
