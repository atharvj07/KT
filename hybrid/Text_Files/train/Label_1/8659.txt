#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int ans = 0, last = 0;
    for (int i = 1; i < n; i++) {
      if (a[i] < a[i - 1]) {
        if (last == 0)
          last = i;
      } else if (last != 0) {
        ans++;
        last = 0;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
