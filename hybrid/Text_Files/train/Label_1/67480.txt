import sys
input = sys.stdin.readline

n, k = map(int, input().split())
r = list(map(int, input().split()))
a = dict()
for _ in range(k):
    x, y = map(int, input().split())
    if x not in a: a[x] = set()
    if y not in a: a[y] = set()
    a[x].add(y)
    a[y].add(x)
ans = [0] * n
for i in range(1, n):
    for j in range(i):
        if r[i] > r[j] and i not in a.get(j, set()):
            ans[i] += 1
print(*ans)
