import sys
n, k = map(int, input().split())
a = []
for i in range(n):
    a.append(input())
cnt = 0
for i in range(n):
    for j in range(i+1, n):
        for k in range(j+1, n):
            f = 0
            for l in range(len(a[i])):
                if a[i][l] == a[j][l] and a[j][l] == a[k][l]:
                    f += 1
                elif a[i][l] != a[j][l] and a[j][l] != a[k][l] and a[i][l] != a[k][l]:
                    f += 1
            if f == k:
                cnt += 1
print(cnt)
