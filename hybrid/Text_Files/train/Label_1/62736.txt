def min_insertions(s):
    def dist(a, b, x, y):
        # Calculate the distance from a to b using steps of x and y
        for d in range(10):
            if (a + d * x) % 10 == b or (a + d * y) % 10 == b:
                return d
        return float('inf')  # Return infinity if not reachable

    n = len(s)
    s = [int(c) for c in s]

    # Initialize the result matrix with infinity
    result = [[float('inf')] * 10 for _ in range(10)]

    # For each pair of x and y
    for x in range(10):
        for y in range(10):
            dp = [0] * n
            for i in range(1, n):
                dp[i] = dp[i-1] + dist(s[i-1], s[i], x, y)
                # If no path, break and set to -1
                if dp[i] == float('inf'):
                    dp[n-1] = -1
                    break
            if dp[n-1] != -1:
                dp[n-1] -= (n-1)  # Subtract to get the number of insertions
            result[x][y] = dp[n-1]

    # Convert result to the desired format
    output = ""
    for row in result:
        output += " ".join(str(-1 if x == float('inf') else x) for x in row) + "\n"
    return output.strip()

# Reading input and printing output
s = input().strip()
print(min_insertions(s))
