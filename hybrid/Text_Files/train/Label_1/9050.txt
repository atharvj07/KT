#include <bits/stdc++.h>

using namespace std;

const long long MOD = 998244353;

long long mod(long long x) {
	return (x % MOD + MOD) % MOD;
}

int n, k, len;
vector<int> a;
vector<vector<long long>> dp;

long long solve(int l, int r) {
	if (l == r) {
		return a[l] == -1 ? k : 1;
	}
	if (dp[l][r] != -1) {
		return dp[l][r];
	}
	long long ans = 0;
	if (a[l] == -1) {
		for (int i = 1; i <= k; i++) {
			if (i == a[l + 1] and i == a[r]) {
				continue;
			}
			ans = mod(ans + solve(l + 1, r));
		}
	} else {
		if (a[l] == a[l + 1] and a[l] == a[r]) {
			dp[l][r] = 0;
		} else {
			ans = solve(l + 1, r);
		}
	}
	return dp[l][r] = ans;
}

int main() {
	cin >> n >> k >> len;
	a.resize(n);
	dp.assign(n, vector<long long>(n, -1));
	for (int i = 0; i < n; i++) {
		cin >> a[i];
	}
	cout << solve(0, n - 1) << endl;
	return 0;
}
