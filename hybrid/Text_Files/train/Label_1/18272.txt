#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> pre(n + 1);
    for (int i = 0; i < n; ++i) { pre[i + 1] = pre[i] + s[i] - '0'; }
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
      for (int j = i + 1; j <= n; ++j) {
        if (pre[j] - pre[i] == j - i) { ++ans; }
      }
    }
    cout << ans << "\n";
  }
}
