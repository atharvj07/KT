n, m = map(int, input().split(' '))
a = [[c == '.' for c in input()] for _ in range(n)]
for i in range(n):
    for j in range(m):
        if a[i][j]: continue
        if all(a[i][k] for k in range(m)) or all(a[k][j] for k in range(n)):
            print('YES')
            print(i + 1, j + 1)
            exit()
print('NO')
