n = int(input())
s = input()
mod = 10 ** 9 + 7
l = []
for i in range(n):
    if s[i] == '?':
        l.append(i)
ans = 3 ** len(l)
for i in range(len(l)):
    for j in range(i + 1, len(l)):
        for k in range(j + 1, len(l)):
            if s[l[i] - 1] == s[l[j] + 1] == s[l[k] - 1] == 'a':
                ans -= 1
            elif s[l[i] - 1] == s[l[j] + 1] == s[l[k] - 1] == 'c':
                ans -= 1
            elif s[l[i] - 1] == s[l[j] + 1] == s[l[k] - 1] == 'b':
                ans -= 1
            if s[l[i] + 1] == s[l[j] - 1] == s[l[k] + 1] == 'a':
                ans -= 1
            elif s[l[i] + 1] == s[l[j] - 1] == s[l[k] + 1] == 'c':
                ans -= 1
            elif s[l[i] + 1] == s[l[j] - 1] == s[l[k] + 1] == 'b':
                ans -= 1
            if ans < 0:
                ans += mod
print(ans)
