def find_replication_order(n, replication_rules):
    from collections import defaultdict, deque

    # Counts for each server
    current_load = defaultdict(int, {i: 4 for i in range(1, n + 1)})
    increase_on_apply = defaultdict(int)
    decrease_on_apply = defaultdict(int)

    for i, (a, b, c) in enumerate(replication_rules, start=1):
        increase_on_apply[b] += 1
        increase_on_apply[c] += 1 if b != c else 0
        decrease_on_apply[a] += 1

    # Directed graph and indegrees
    graph = defaultdict(list)
    indegree = defaultdict(int)
    for i, (a, b, c) in enumerate(replication_rules, start=1):
        graph[a].append((b, c, i))
        indegree[b] += 1
        indegree[c] += 1 if b != c else 0

    # Queue initialization with servers having zero indegree
    queue = deque([i for i in range(1, n + 1) if current_load[i] + increase_on_apply[i] - decrease_on_apply[i] <= 9])

    order = []
    while queue:
        a = queue.popleft()
        for b, c, rule_idx in graph[a]:
            order.append(rule_idx)
            current_load[b] += 1
            current_load[c] += 1 if b != c else 0
            indegree[b] -= 1
            indegree[c] -= 1 if b != c else 0

            # Check if new processes can be added without exceeding the limit
            if current_load[b] + increase_on_apply[b] - decrease_on_apply[b] <= 9 and indegree[b] == 0:
                queue.append(b)
            if b != c and current_load[c] + increase_on_apply[c] - decrease_on_apply[c] <= 9 and indegree[c] == 0:
                queue.append(c)

            decrease_on_apply[a] -= 1
            increase_on_apply[b] -= 1
            increase_on_apply[c] -= 1 if b != c else 0

    if len(order) != 4 * n:
        return "NO"
    else:
        return "YES\n" + " ".join(map(str, order))


n = int(input())
replication_rules = [tuple(map(int, input().split())) for _ in range(4 * n)]
result = find_replication_order(n, replication_rules)
print(result)
