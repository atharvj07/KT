#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
using ll = long long;

const ll LINF = 1e18;
const int MINSUF = 0, MINPRS = 1, SRS = 2;

struct seg {
    int l, r, ty;
    ll val;
} ss[602];

int main() {
    int n, m, p=1, st[302], ed[302], ty[302];
    ll ans=LINF, mx=LINF, mn=-LINF;
    cin >> n >> m;
    if(!m){
        cout << 0 << endl << 0 << endl;
        return 0;
    }
    vector<ll> v(n+1);
    for(int i=1; i<=n; i++){
        cin >> v[i];
        if(i==1 || v[i-1] < v[i]) mx = max(mx,v[i]), st[p]=i, ty[p++]=MINSUF;
        if(i==n || v[i] < v[i+1]) mn = min(mn,v[i]), ed[p]=i, ty[p++]=MINPRS;
    }
    for(int i=1; i<=n; i++){
        v[i] -= mx + mn;
        if(v[i] == 0) ss[p++] = {i,i,SRS,-1};
    }
    for(int i=1, j=p; i<j; i++){
        if(ss[i].ty==SRS && ss[i+1].ty==SRS){
            ss[i].r = ss[i+1].r;
            ss[i+1].r = -1;
        }
    }
    for(int i=0, j=0; i<p; i++){
        if(ss[i].l==-1) continue;
        while(j<m && ss[j].r<ss[i].l) j++;
        if(j==m){
            break;
        }
        if(ss[j].l<=ss[i].l && ss[j].r>=ss[i].r && ss[i].ty==SRS){
            ss[i].val = ss[j].val = mx - v[ss[i].l];
            ans = max(ans, ss[i].val);
        }else if(ss[j].l<=ss[i].l && ss[j].r<=ss[i].r && (ty[i]==MINSUF || ty[i]==MINPRS)){
            ss[i].val = ss[j].val = max(v[ss[i].l] - ss[j].r + 1, v[ss[i].r] - ss[j].l + 1);
            ans = max(ans, ss[i].val);
        }else if(ss[j].l>=ss[i].l && ss[j].r>=ss[i].r && (ty[i]==MINSUF || ty[i]==MINPRS)){
            ss[i].val = ss[j].val = max(ss[j].l - v[ss[i].l], ss[j].r - v[ss[i].r]);
            ans = max(ans, ss[i].val);
        }
    }
    cout << ans << endl;
    int cnt=0;
    for(int i=0; i<p; i++){
        if(ss[i].val!=0){
            cnt++;
        }
    }
    cout << cnt << endl;
    for(int i=0; i<p; i++){
        if(ss[i].val!=0){
            cout << i+1 << " ";
        }
    }
    cout << endl;
    return 0;
}
