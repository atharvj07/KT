// Read vectors from standard input

// Print vectors to standard output
template<typename T>
vi& wrVec(vi& v) {for (auto& x: v) cout << x << " "; return v;}


// Solution 
void code() {
    ll t;
    cin >> t;
    while (t--) {
        ll L, v, l, r;
        cin >> L >> v >> l >> r;
        ll p = L / v;
        if ((p - (l - 1) / v) > 0) {
            cout << p - (l - 1) / v - (r <= L / v && v * (r - 1) >= l);
        } else {
            cout << 0;
        }
        cout << endl;
    }
}

int main() {
    fastIO;
    ll TT = 1;
    // cin >> TT;
    for (int tt = 1; tt <= TT; tt++) {
        code();
    }
    return 0;
}
