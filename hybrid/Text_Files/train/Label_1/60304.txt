s = input()
if s[0] != '[' or s[-1] != ']':
    print(-1)
    exit()
s = s[1:-1]
stack = []
ans = 0
for c in s:
    if c == '[':
        stack.append('[')
    elif c == ']':
        if stack:
            stack.pop()
            ans += 2
    elif c == ':':
        if stack and stack[-1] == '[':
            stack.append(':')
            ans += 1
    elif c == '|':
        if stack and stack[-1] == ':':
            ans += 1
if stack:
    print(-1)
else:
    print(ans)
