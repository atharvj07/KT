import sys

input = sys.stdin.readline

t = int(input())

for _ in range(t):
    n = int(input())
    a = [0] * (n + 1)
    b = [0] * (n + 1)

    for i in range(n):
        ai, bi = map(int, input().split())
        a[ai] += 1
        b[bi] += 1

    ans = 0

    for i in range(1, n + 1):
        ans += a[i] * (a[i] - 1) * (a[i] - 2) // 6

    for i in range(1, n + 1):
        ans += b[i] * (b[i] - 1) * (b[i] - 2) // 6

    ans = ans * n * (n - 1) * (n - 2) // 6

    print(ans)
