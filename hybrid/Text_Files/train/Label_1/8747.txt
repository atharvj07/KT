#include <iostream>
#include <map>
#include <algorithm>

using namespace std;

int main() {
    int n, k, m;
    cin >> n >> k >> m;
    map<int, int> cityCount;
    for (int i = 0; i < n; i++) {
        int city;
        cin >> city;
        cityCount[city]++;
    }
    int ans = n * m;
    for (auto it = cityCount.begin(); it != cityCount.end(); it++) {
        ans -= (it->second / k) * k;
    }
    cout << ans << endl;
    return 0;
}
