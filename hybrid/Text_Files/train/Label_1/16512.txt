#include <iostream>
#include <vector>
#include <algorithm>
#define ll long long
using namespace std;
const int mod = 1e9 + 7;

int main() {
    int n, k;
    cin >> n >> k;
    vector<ll> a(n);
    for (int i = 0; i < n; i++)
        cin >> a[i];
    vector<ll> pre(n, 1), suf(n, 1);
    for (int i = 1; i < n; i++)
        pre[i] = pre[i - 1] * a[i - 1];
    for (int i = n - 2; i >= 0; i--)
        suf[i] = suf[i + 1] * a[i + 1];
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        ll p = pre[i] * suf[i];
        ll s = 0;
        for (int j = i; j < n; j++) {
            s += a[j];
            if (p * k == s)
                ans++;
            if (p * k < s)
                break;
        }
        p = suf[i];
        s = a[i];
        for (int j = i - 1; j >= 0; j--) {
            s += a[j];
            if (p * k == s)
                ans++;
            if (p * k < s)
                break;
            p = pre[j] * p;
        }
    }
    cout << ans << "\n";
}
