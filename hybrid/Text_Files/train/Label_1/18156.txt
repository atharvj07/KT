#include <iostream>
int main() {
  long long x, ans = 0;
  std::cin >> x;
  for (long long i = 1; i * i <= x; ++i)
    for (long long j = i; i * j <= x && j <= i * i; ++j)
      if (i * j == x) ++ans;
  std::cout << ans << '\n';
  for (long long i = 1; i * i <= x; ++i)
    for (long long j = i; i * j <= x && j <= i * i; ++j)
      if (i * j == x) std::cout << i << ' ' << j << '\n';
  return 0;
}
