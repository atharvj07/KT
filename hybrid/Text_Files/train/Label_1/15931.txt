#include<bits/stdc++.h>
using namespace std;
#define endl "\n"
#define test_case(t) for(int total_test_case=t;total_test_case--;)
int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0);
    int t;
    cin>>t;
    test_case(t)
    {
        int n;
        cin>>n;
        string s,t;
        cin>>s>>t;
        int l=0,r=0,ans=-1;
        bool one=0,two=0;
        for(int i=0;i<n;i++)
        {
            if(s[i]!=t[i])
            {
                if(s[i]==t[n-1-i])
                {
                    r++;
                }
                else if(s[n-1-i]==t[i])
                {
                    l++;
                }
                else
                {
                    one=1;
                }
            }
        }
        if(!one)
        {
            ans=max(l,r);
        }
        l=0,r=0;
        for(int i=0;i<n;i++)
        {
            if(s[i]!=t[n-1-i])
            {
                if(s[i]==t[i])
                {
                    r++;
                }
                else if(s[n-1-i]==t[n-1-i])
                {
                    l++;
                }
                else
                {
                    two=1;
                }
            }
        }
        if(!two)
        {
            if(ans==-1)
            {
                ans=max(l,r);
            }
            else
            {
                ans=min(ans,max(l,r));
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
