#include<bits/stdc++.h>
using namespace std;
int main(){
 string s,t;
 getline(cin,s);
 getline(cin,t);
 int m=0,w=0;
 for(int i=0;i<s.size();i++){
  if(s[i]=='a'||s[i]=='e'||s[i]=='i'||s[i]=='o'||s[i]=='u'){
   m+=(s[i]!=t[i]);
  }
  else w+=(s[i]!=t[i]);
 }
 if(m>0&&w>0)puts("No");
 else puts("Yes");
}
