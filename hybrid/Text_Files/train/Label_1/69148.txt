import sys

input = sys.stdin.readline

n = int(input())
nums = []
for i in range(0, n):
    x1, y1, x2, y2 = map(int, input().split())
    nums.append((x1, y1, x2, y2))
nums.sort()
ans = [0]*n
flag = True
for i in range(0, n):
    for j in range(i+1, n):
        x1, y1, x2, y2 = nums[i]
        x3, y3, x4, y4 = nums[j]
        if (x2 == x3 or y2 == y3) and (x1 <= x3 <= x2 or x3 <= x1 <= x4):
            if ans[i] == ans[j]:
                flag = False
                break
            if ans[i] == 0:
                ans[i] = 4
            if ans[j] == 0:
                ans[j] = 3
            if ans[i] == ans[j]:
                flag = False
                break
        elif (x2 == x3 or y2 == y3) and (y1 <= y3 <= y2 or y3 <= y1 <= y4):
            if ans[i] == ans[j]:
                flag = False
                break
            if ans[i] == 0:
                ans[i] = 3
            if ans[j] == 0:
                ans[j] = 4
            if ans[i] == ans[j]:
                flag = False
                break

for i in range(0, n):
    if ans[i] == 0:
        ans[i] = 1
if not flag:
    print("NO")
else:
    print("YES")
    for i in range(0, n):
        print(ans[i])
