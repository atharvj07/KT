import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] cubes = new int[n][6];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 6; j++) {
                cubes[i][j] = in.nextInt();
            }
        }
        int result = 0;
        for (int i = 1; i <= 1000000; i++) {
            boolean canMake = true;
            int[] digits = new int[10];
            String str = String.valueOf(i);
            for (int j = 0; j < str.length(); j++) {
                int digit = str.charAt(j) - '0';
                boolean found = false;
                for (int k = 0; k < n; k++) {
                    for (int l = 0; l < 6; l++) {
                        if (cubes[k][l] == digit) {
                            digits[digit]++;
                            found = true;
                            break;
                        }
                    }
                }
                if (!found) {
                    canMake = false;
                    break;
                }
            }
            if (canMake) {
                result = i;
            } else {
                break;
            }
        }
        System.out.println(result);
    }
}
