import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        int maxb = sc.nextInt();
        int t = sc.nextInt();
        int[] ans = new int[k];
        for (int ki = 0; ki < k; ki++) {
            int[] b = new int[n];
            for (int ni = 0; ni < n; ni++) {
                b[ni] = sc.nextInt();
            }
            int[] a = new int[n * t];
            for (int ai = 0; ai < n * t; ai++) {
                a[ai] = b[ai / t] % (ai % t + 1);
            }
            int[] dp = new int[n * t];
            int maxLen = 0;
            for (int ai = 0; ai < n * t; ai++) {
                dp[ai] = 1;
                for (int bj = 0; bj < ai; bj++) {
                    if (a[bj] < a[ai]) {
                        dp[ai] = Math.max(dp[ai], dp[bj] + 1);
                    }
                }
                maxLen = Math.max(maxLen, dp[ai]);
            }
            ans[ki] = maxLen;
        }
        for (int ki = 0; ki < k; ki++) {
            System.out.println(ans[ki]);
        }
    }
}
