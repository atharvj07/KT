def Board(n, B):
  F = []
  for x in range(n):
    F.append([])
    for y in range(n):
      F[x].append(B[x * n + y])
  return F

def inside(n, x, y):
  return x >= 0 and x < n and y >= 0 and y < n

def move(n, B, x, y, dx, dy):
  if not inside(n, x, y):
    return False
  if B[x * n + y] == 'o':
    return True
  if F[x][y] > 0:
    return False
  F[x][y] = 1
  if move(n, B, x + dx, y + dy, dx, dy):
    return True
  F[x][y] = 0
  return False

def solve(n, B):
  for x in range(n):
    for y in range(n):
      if B[x * n + y] == 'o':
        for dx in range(-n, n + 1):
          for dy in range(-n, n + 1):
            F = Board(n, B)
            if dx == 0 and dy == 0:
              continue
            if move(n, B, x, y, dx, dy):
              return F
  return None

n = int(input())
B = input()
F = solve(n, B)
if F is None:
  print('NO')
else:
  print('YES')
  for row in F:
    print(''.join(row))
