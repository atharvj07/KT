#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  stack<int> s;
  vector<int> ans(n, 0);
  for (int i = 0; i < n; i++) {
    int h;
    cin >> h;
    while (!s.empty() && s.top() <= h) {
      s.pop();
    }
    if (!s.empty()) {
      ans[i] = h - s.top();
    }
    s.push(h);
  }
  for (int i = 0; i < n; i++) { cout << ans[i] << " "; }
}
