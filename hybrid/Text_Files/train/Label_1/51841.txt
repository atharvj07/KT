import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();

        int count = 0;
        for (long k = 2; k * k <= n; k++) {
            if (n % k == 0) {
                count++;
                while (n % k == 0) {
                    n /= k;
                }
            }
        }
        if (n > 1) {
            count++;
        }

        System.out.println(count);
    }
}
