#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  int cnt = 0;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '+')
      cnt++;
    else
      cnt--;
    if (cnt < 0) {
      cout << "No";
      return 0;
    }
  }
  cout << "Yes";
  return 0;
}
