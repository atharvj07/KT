n, A, m, q = int(input()), list(map(int, input().split())), int(input()), list(map(int, input().split()))
A.sort(); q.sort(); j=0
for i in range(m):
    ans=0
    for k in A[n-q[i]:]:
        ans+=k
        n-=1
    for k in A[j:n-q[i]]:
        j+=1
        if j==n-q[i]: break
        ans+=k
    print(ans)
