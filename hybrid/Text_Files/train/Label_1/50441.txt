import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();

        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = in.nextInt();
        }

        int[] right = new int[n];
        int[] left = new int[n];

        right[n - 1] = d[n - 1];
        left[0] = d[0];

        for (int i = n - 2; i >= 0; i--) {
            right[i] = Math.min(d[i], right[i + 1]);
        }

        for (int i = 1; i < n; i++) {
            left[i] = Math.min(d[i], left[i - 1]);
        }

        boolean can = true;
        for (int i = 0; i < n; i++) {
            if (left[i] < right[i]) {
                can = false;
                break;
            }
        }

        System.out.println(can ? "yes" : "no");
    }
}
