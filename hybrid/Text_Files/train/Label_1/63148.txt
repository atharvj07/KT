import math

def solve(n, b):
    a = [0 for i in range(n)]
    if b[n - 1] == 0:
        return "NO"

    a[0] = math.pow(2, int(math.log2(b[n - 1])))

    for i in range(1, n):
        a[i] = math.pow(2, int(math.log2(b[i - 1])) + 1)

    if a[n - 1] % a[0] != b[n - 1]:
        return "NO"

    return "YES\n" + " ".join(str(i) for i in a)

n = int(input())
b = [int(i) for i in input().split()]
print(solve(n, b))
