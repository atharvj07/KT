def count_good_integers(n):
    # Mapping of each digit to the digits it can become if some segments are broken
    transition = {
        '0': ['0', '8'],
        '1': ['0', '1', '3', '4', '7', '8', '9'],
        '2': ['2', '8'],
        '3': ['3', '8', '9'],
        '4': ['4', '8', '9'],
        '5': ['5', '6', '8', '9'],
        '6': ['6', '8'],
        '7': ['0', '3', '7', '8', '9'],
        '8': ['8'],
        '9': ['8', '9']
    }
    
    n = str(n).zfill(2)  # Ensure n is a string with at least 2 digits
    first_digit_options = transition[n[0]]
    second_digit_options = transition[n[1]]
    
    return len(first_digit_options) * len(second_digit_options)

n = input().strip()
print(count_good_integers(n))
