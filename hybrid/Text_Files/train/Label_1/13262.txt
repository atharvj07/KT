#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            int dif = __builtin_popcount(a[i] ^ a[j]);
            if (dif == k) ++ans;
        }
    }
    cout << ans << '\n';
}
