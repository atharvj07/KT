#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.size();
        vector<int> z(n);
        int l = 0, r = 0;
        for (int i = 1; i < n; i++) {
            if (i <= r) {
                z[i] = min(r - i + 1, z[i - l]);
            }
            while (i + z[i] < n && s[z[i]] == s[i + z[i]]) {
                z[i]++;
            }
            if (i + z[i] - 1 > r) {
                l = i;
                r = i + z[i] - 1;
            }
        }
        int p = n;
        for (int i = 1; i < n; i++) {
            if (z[i] == i && (n - i) % i == 0) {
                p = min(p, i);
            }
        }
        for (int i = 0; i < p; i++) {
            cout << s[i];
        }
        cout << endl;
    }
    return 0;
}
