#include<bits/stdc++.h>
using namespace std;
int main(){
    long long t;
    cin>>t;
    while(t--){
        long long n;
        cin>>n;
        long long a[n],b[n];
        for(int i=0;i<n;i++){
            cin>>a[i]>>b[i];
        }
        if(a[0]!=b[0]){
            cout<<"NO\n";
        }
        else{
            long long x=0,y=0;
            for(int i=0;i<n;i++){
                if(a[i]+y!=b[i]){
                    cout<<"NO\n";
                    return 0;
                }
                if(b[i]+x!=a[i]){
                    cout<<"NO\n";
                    return 0;
                }
                x=a[i]-b[i];
                y=b[i]-a[i];
            }
            cout<<"YES\n";
        }
    }
}
