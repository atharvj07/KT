n, k = map(int, input().split())

ans = 1
used = {1}

for i in range(n-1):
    if k <= ans:
        break
    # append last element to the end
    ans += 1
    used.add(ans)
    # append current sequence to the end
    ans += i+1
    # insert unused minimum to the middle
    for j in range(2, ans+1):
        if j not in used:
            used.add(j)
            if k <= ans:
                break
            ans += 1

print(ans-k+1)
