#include <bits/stdc++.h>
#define pb push_back
#define eb emplace_back
#define all(x) (x).begin(), (x).end()
using namespace std;
const int _ = 1 << 18;
int n, m, a[_], b[_], maxb, ans;
vector<int> Q[_];
void solve(int x, int l, int r, int t) {
    int u = [&] {
        if (t > 0) {
            return a[Q[t][lower_bound(all(Q[t]), r) - lower_bound(all(Q[t]), l)]];
        } else {
            return maxb;
        }
    }();
    if (t == 0) {
        ans = max(ans, u);
    } else {
        solve(x | 1, l, (l + r) >> 1, t - 1);
        solve(x, (l + r) >> 1, r, t - 1);
        solve(x | 2, l, (l + r) >> 1, t - 1);
        solve(x, (l + r) >> 1, r, t - 1);
    }
    ans += u;
}
int main() {
    ios_base::sync_with_stdio(0), cin.tie(0);
    cin >> n;
    for (int i = 0; i < (1 << n); ++i) {
        cin >> a[i];
    }
    for (int i = 0; i < (1 << n); ++i) {
        for (int j = 0; j < n; ++j) {
            if (i & (1 << j)) {
                b[i] |= (1 << j);
            }
        }
        Q[b[i]].pb(i);
    }
    for (int i = 1; i < (1 << n); ++i) {
        maxb = max(maxb, b[i]);
    }
    solve(0, 0, (1 << n) - 1, n - 1);
    cout << ans << '\n';
}
