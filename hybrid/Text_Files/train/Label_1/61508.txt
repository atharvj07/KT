start, end = input().split()
n = int(input())
if end == start:
    if n % 4 == 0:
        print("cw")
    elif n % 2 == 0:
        print("ccw")
    else:
        print("undefined")
elif end == ">" and start == "v":
    print("cw" if n % 4 < 2 else "ccw")
elif end == "<" and start == "^":
    print("cw" if n % 4 < 2 else "ccw")
elif end == "^" and start == "<":
    print("cw" if n % 4 < 2 else "ccw")
elif end == "v" and start == ">":
    print("cw" if n % 4 < 2 else "ccw")
else:
    print("undefined")
