#include<bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    for(int i = 1; i < s.length(); i++) {
        for(int j = i+1; j < s.length(); j++) {
            int a = stoi(s.substr(0, i)), b = stoi(s.substr(i, j-i)), c = stoi(s.substr(j));
            if(a+b==c) {
                cout << s.substr(0, i) << '+' << s.substr(i, j-i) << '=' << s.substr(j) << '\n';
                return 0;
            }
        }
    }
}
