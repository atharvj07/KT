#include <cstdio>
using i64 = long long;
int main() {
    i64 n, m;
    scanf("%lld%lld", &n, &m);
    i64 a[n];
    for (i64 i = 0; i < n; ++i) scanf("%lld", &a[i]);
    i64 currentPage = 0, pagesTurned = 0;
    for (i64 i = 0; i < n; ++i) {
        currentPage += a[i];
        pagesTurned += currentPage / m;
        currentPage %= m;
        printf("%lld ", pagesTurned);
    }
    return 0;
}
