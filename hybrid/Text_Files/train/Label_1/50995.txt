import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            int n = input.nextInt();
            int[] a = new int[n];
            int max = -1, min = Integer.MAX_VALUE, l = -1, r = -1;
            for (int i = 0; i < n; i++) {
                a[i] = input.nextInt();
                if (a[i] > max) {
                    max = a[i];
                    r = i;
                }
                if (a[i] <= min) {
                    min = a[i];
                    l = i;
                }
            }
            if (max - min >= l - r + 1)
                System.out.println("YES\n" + (l + 1) + " " + (r + 1));
            else
                System.out.println("NO");
        }
    }
}
