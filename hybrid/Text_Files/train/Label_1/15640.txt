#include <iostream>
#include <cstring>

using namespace std;

int main() {
    int k;
    cin >> k;
    while (k--) {
        int n;
        cin >> n;
        string s, t;
        cin >> s >> t;
        bool flag = false;
        int ind1 = -1, ind2 = -1;
        int d = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] != t[i]) {
                if (d == 0) {
                    ind1 = i;
                    d++;
                } else if (d == 1) {
                    ind2 = i;
                    d++;
                }
            }
            if (d > 2)
                break;
        }
        if (d == 2 && s[ind1] == t[ind2] && s[ind2] == t[ind1])
            flag = true;
        if (flag)
            cout << "Yes\n";
        else
            cout << "No\n";
    }
    return 0;
}
