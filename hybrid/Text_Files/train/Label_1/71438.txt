def determine_winner(xp, yp, xv, yv):
    # Polycarp can win in max(xp, yp) moves by moving straight down or straight left
    # Vasiliy can win in max(min(xv, yv) + abs(xv-yv), max(xv, yv)) moves. The first part is the diagonal moves plus the straight moves to (0, 0)
    if max(xp, yp) <= max(min(xv, yv) + abs(xv - yv), max(xv, yv)):
        return "Polycarp"
    else:
        return "Vasiliy"

# Reading from standard input
xp, yp, xv, yv = map(int, input().split())
print(determine_winner(xp, yp, xv, yv))
