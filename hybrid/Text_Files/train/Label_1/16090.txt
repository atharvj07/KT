#include <iostream>
#include <string>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;

    int ans = 1;
    int cnt = 1;
    for (int i = 1; i < n; i++) {
        if (s[i] != s[i - 1]) {
            cnt++;
            ans = max(ans, cnt);
        } else {
            cnt = 1;
        }
    }

    cout << ans + 1 << endl;

    return 0;
}
