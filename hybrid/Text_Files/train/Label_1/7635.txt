#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=1e9+7;

ll n, ans;

int main() {
    cin >> n;
    for(int i=1; i<=n; i++) {
        bool t=0;
        for(int j=2; j<=10; j++) {
            if(i%j==0) {
                t=1;
                break;
            }
        }
        if(!t) ans++;
    }
    cout << ans << "\n";
    return 0;
}
