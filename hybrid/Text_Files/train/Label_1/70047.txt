n, k, m = map(int, input().split())
a = list(map(int, input().split()))

remainder_counts = {}
for num in a:
    remainder = num % m
    if remainder not in remainder_counts:
        remainder_counts[remainder] = []
    remainder_counts[remainder].append(num)

for remainder, nums in remainder_counts.items():
    if len(nums) >= k:
        print("Yes")
        print(' '.join(map(str, nums[:k])))
        break
else:
    print("No")
