import java.util.*;
import java.io.*;

class Main {

    private static void solve(int n, int m, int b, int[][] fr) {
        int[] need = new int[n];
        for (int i = 0; i < n; i++) {
            Arrays.sort(fr[(i << 1) + 1]);
            int c = 0;
            for (int j = 0; j < m; j++) {
                if (Arrays.binarySearch(fr[(i << 1) + 1], j + 1) >= 0) need[i] |= 1 << j;
            }
        }
        BitSet b1 = new BitSet();
        BitSet b2 = new BitSet();
        int ans = Integer.MAX_VALUE;
        for (int msk = 0; msk < (1 << m); msk++) {
            b1.clear();
            b2.clear();
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                if ((need[i] & msk) == need[i]) {
                    b1.or(b2);
                    b2.clear();
                    b2.set(i);
                    cnt++;
                }
            }
            if (b1.cardinality() == m) {
                int sum = 0;
                for (int i = b2.nextSetBit(0); i >= 0; i = b2.nextSetBit(i + 1)) {
                    sum += fr[i << 1][0];
                }
                sum += cnt * b;
                ans = Math.min(ans, sum);
            }
        }
        System.out.println(ans == Integer.MAX_VALUE ? -1 : ans);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = in.readLine().split(" ");
        int n = Integer.parseInt(parts[0]), m = Integer.parseInt(parts[1]), b = Integer.parseInt(parts[2]);
        int[][] fr = new int[n << 1][];
        for (int i = 0; i < n; i++) {
            parts = in.readLine().split(" ");
            fr[i << 1] = new int[]{Integer.parseInt(parts[0]), Integer.parseInt(parts[1]), Integer.parseInt(parts[2])};
            fr[(i << 1) + 1] = new int[fr[i << 1][2]];
            parts = in.readLine().split(" ");
            for (int j = 0; j < fr[i << 1][2]; j++) {
                fr[(i << 1) + 1][j] = Integer.parseInt(parts[j]);
            }
        }
        solve(n, m, b, fr);
    }
}
