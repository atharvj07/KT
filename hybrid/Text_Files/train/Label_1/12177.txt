#include <bits/stdc++.h>

using namespace std;

const int mod = 1e9 + 7;

int n, t;
vector<pair<int, int>> songs;
vector<vector<vector<int>>> dp;

int solve(int i, int rem, int last) {
  if (i == n) {
    if (rem == 0)
      return 1;
    return 0;
  }
  int &res = dp[i][rem][last];
  if (res != -1)
    return res;
  res = 0;
  if (rem >= songs[i].first && last != songs[i].second)
    res = (res + solve(i + 1, rem - songs[i].first, songs[i].second)) % mod;
  res = (res + solve(i + 1, rem, last)) % mod;
  return res;
}

int main() {
  cin >> n >> t;
  songs.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> songs[i].first >> songs[i].second;
  }
  dp.assign(n,
             vector<vector<int>>(t + 1, vector<int>(4, -1)));  // dp[i][rem][last]
  cout << solve(0, t, 0) << endl;
  return 0;
}
