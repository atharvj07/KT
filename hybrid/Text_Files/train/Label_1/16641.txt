#include <bits/stdc++.h>
using namespace std;

const int MOD = 1e9 + 7;

int main() {
    string a;
    int k;
    cin >> a >> k;
    int n = a.size() * k;
    vector<int> dp(n + 1);
    dp[0] = 1;
    for (int i = 1; i <= n; i++) {
        int sum = 0;
        for (int j = i; j <= n; j++) {
            sum = (sum + (a[j % a.size()] - '0') * dp[i - j]) % MOD;
            if (j != i) sum = (sum + MOD) % MOD;
            if (sum % 5 == 0) dp[j] = sum;
        }
    }
    cout << dp[n] << '\n';
    return 0;
}
