#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
int const N = 2e5 + 5;
int n, ans, s[N], t[N];
int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> s[i] >> t[i];
    int p1 = *min_element(s, s + n), p2 = *max_element(s, s + n);
    if (p2 - p1 > 1) { cout << -1 << endl; return 0; }
    for (int i = 0; i < n; i++) {
        if (t[i] == 0) continue;
        if (s[i] < p2) {
            int dt = min(t[i], p2 - s[i]);
            ans += dt;
            s[i] += dt;
            t[i] -= dt;
        }
        if (s[i] > p1) {
            int dt = min(t[i], s[i] - p1);
            ans += dt;
            s[i] -= dt;
            t[i] -= dt;
        }
        if (s[i] < p2) {
            int dt = min(t[i], p2 - s[i]);
            ans += dt;
            s[i] += dt;
            t[i] -= dt;
        }
    }
    cout << ans << endl;
    for (int i = 0; i < n; i++)
        s[i] += t[i];
    for (int i = 0; i < n; i++)
        cout << s[i] << ' ';
    cout << endl;
}
