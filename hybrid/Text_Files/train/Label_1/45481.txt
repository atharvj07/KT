import java.util.*;

public class TrailConnection {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        // Create a graph to represent the trails
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            graph.get(u).add(v);
            graph.get(v).add(u);
        }

        // Find the connected components of the graph
        int[] components = new int[n + 1];
        int componentCount = 0;
        for (int i = 1; i <= n; i++) {
            if (components[i] == 0) {
                componentCount++;
                dfs(graph, components, componentCount, i);
            }
        }

        int trailsToAdd = componentCount - 1;

        // Output the result
        System.out.println(trailsToAdd);
    }

    private static void dfs(List<List<Integer>> graph, int[] components, int componentCount, int node) {
        components[node] = componentCount;
        for (int neighbor : graph.get(node)) {
            if (components[neighbor] == 0) {
                dfs(graph, components, componentCount, neighbor);
            }
        }
    }
}
