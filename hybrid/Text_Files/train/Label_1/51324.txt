import java.util.*;
import java.lang.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int d = input.nextInt();
        int s = input.nextInt();

        int smallest = 1;
        while (smallest % d != 0) {
            smallest += d;
        }

        int sum = 0;
        int n = smallest;
        while (sum < s) {
            sum += n % 10;
            n /= 10;
        }

        while (sum > s) {
            if (n % 10 == 0) {
                n /= 10;
                n += d;
            } else {
                n++;
            }
            sum = 0;
            while (n > 0) {
                sum += n % 10;
                n /= 10;
            }
        }

        if (sum == s) {
            System.out.println(smallest);
        } else {
            System.out.println(-1);
        }
    }
}
