#include <bits/stdc++.h>
using namespace std;

int n, t[3][3];
long long dp[41][3][3][3];

long long solve(int x, int a, int b, int c) {
    if (dp[x][a][b][c] != -1) return dp[x][a][b][c];
    if (x == 0) return dp[x][a][b][c] = 0;
    long long add, p = t[a][b];
    add = solve(x - 1, a, c, b) + p;
    p = t[a][c];
    add = min(add, solve(x - 1, a, b, c) + p);
    p = t[b][c];
    add = min(add, solve(x - 1, b, a, c) + p);
    p = t[b][a];
    add = min(add, solve(x - 1, b, c, a) + p);
    p = t[c][b];
    add = min(add, solve(x - 1, c, a, b) + p);
    p = t[c][a];
    add = min(add, solve(x - 1, c, b, a) + p);
    return dp[x][a][b][c] = add;
}

int main() {
    memset(dp, -1, sizeof dp);
    for (int i = 0; i < 3; i++)
        for (int j = 0; j < 3; j++) cin >> t[i][j];
    cin >> n;
    cout << solve(n, 0, 1, 2) << endl;
    return 0;
}
