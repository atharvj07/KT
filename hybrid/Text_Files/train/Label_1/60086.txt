n=int(input())
l=list(map(int,input().split()))
s=input()
t=0
stamina=0
for i in range(n):
 ll=l[i]
 if s[i]=="L":
  if stamina>=2:
   stamina-=2
   t+=ll
  else:
   need=2-stamina
   stamina=need+1
   t+=need*5
   ll-=need
   if ll==0:
    continue
   if ll<stamina:
    t+=(stamina-ll)*5
    stamina=0
   else:
    stamina-=ll
    t+=ll
 elif s[i]=="W":
  if stamina==0:
   stamina=1
   t+=5
   ll-=1
   if ll==0:
    continue
  if ll<stamina:
   t+=(stamina-ll)*5
   stamina=0
  else:
   stamina-=ll
   t+=ll*3
 else:
  t+=ll
  if s[i]=="W":
   stamina+=ll
  elif s[i]=="G":
   stamina+=ll
print(t)
