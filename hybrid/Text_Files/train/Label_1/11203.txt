#include <bits/stdc++.h>
using namespace std;
typedef pair<int,int> pii;
const int maxn = 5e5+5;
int n;
pii v[maxn];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> v[i].first >> v[i].second;
    sort(v+1,v+1+n);
    int r = 0;
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (v[i].first >= r) {
            r = v[i].second;
            ans++;
        }
    }
    cout << ans << endl;
}
