#include <cstdio>
const int limit = 1LL << 32;
int main() {
    long long x = 0;
    int l;
    scanf("%d", &l);
    while (l--) {
        char cmd[5];
        int n;
        scanf("%s", cmd);
        if (cmd[0] == 'f') {
            scanf("%d", &n);
            l -= n;
        } else if (cmd[0] == 'a') {
            if (x < limit) ++x;
            else {
                puts("OVERFLOW!!!");
                return 0;
            }
        }
    }
    printf("%lld\n", x);
}
