import sys
input=lambda: sys.stdin.readline().strip()
mod=10**9+7
for _ in range(int(input())):
 n,m,k=map(int,input().split())
 a=list(map(int,input().split()))
 b=[[0]*k for i in range(n+1)]
 cnt=[0]*(n+1)
 for i in a: cnt[i]+=1
 for i in range(k):
  for j in range(1,n+1):
   b[j][i]=cnt[j]
   if j-i-1>=1: b[j][i]+=b[j-i-1][i]
   b[j][i]%=mod
 for i in range(1,n+1):
  for j in range(k):
   b[i][j]+=b[i-1][j]
   b[i][j]%=mod
 ans=0
 for i in range(m,n+1):
  ans+=b[i][k-1]
  ans%=mod
 print(ans)
