n = int(input())
s = input()
a = s.count('A')
c = s.count('C')
g = s.count('G')
t = s.count('T')
q = s.count('?')
if max((a, c, g, t) + (q,)) - min((a, c, g, t) + (q,)) > 1 or q < max((a, c, g, t) + (q,)) - min((a, c, g, t) + (q,)):
    print('===')
else:
    for i in range(n):
        if s[i] == '?':
            if a < max((a, c, g, t) + (q,)):
                s = s[:i] + 'A' + s[i+1:]
                a += 1
            elif c < max((a, c, g, t) + (q,)):
                s = s[:i] + 'C' + s[i+1:]
                c += 1
            elif g < max((a, c, g, t) + (q,)):
                s = s[:i] + 'G' + s[i+1:]
                g += 1
            else:
                s = s[:i] + 'T' + s[i+1:]
                t += 1
    print(s)
