#include <bits/stdc++.h>

using namespace std;

const int mod = 1e9 + 7, maxn = 5e4 + 5;

int n, b, k, x, pw[maxn], a[maxn], dp[maxn][105][2];

int solve(int, int, int);

int main() {
    cin >> n >> b >> k >> x;
    pw[0] = 1;
    for (int i = 1; i <= n; i++) {
        pw[i] = 1LL * pw[i - 1] * 10 % x;
    }
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    cout << solve(0, 0, 0) << endl;
    return 0;
}

int solve(int pos, int sum, int prv) {
    if (pos == n) {
        return sum == k;
    }
    int &ans = dp[pos][sum][prv];
    if (~ans) return ans;
    ans = 0;
    if (prv) ans = (ans + solve(pos + 1, sum, prv)) % mod;
    for (int i = 0; i < a[pos]; i++) {
        ans = (ans + solve(pos + 1, (sum + (1LL * pw[n - pos - 1] * (i + 1)) % x) % x, 1)) % mod;
    }
    return ans;
}
