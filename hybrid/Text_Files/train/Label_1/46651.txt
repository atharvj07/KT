import java.util.Scanner;

public class MagnetGroups {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();

        int groups = 1;
        char previousPole = scanner.nextLine().charAt(1);

        for (int i = 1; i < n; i++) {
            char currentPole = scanner.nextLine().charAt(0);
            if (currentPole != previousPole) {
                groups++;
            }
            previousPole = currentPole;
        }

        System.out.println(groups);
    }
}
