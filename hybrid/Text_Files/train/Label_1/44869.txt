import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int x = 0, y = 0;
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            char c = s.charAt(i);
            if (c >= 'a' && c <= 'm') {
                x++;
            } else if (c >= 'n' && c <= 'z') {
                x--;
            } else if (c >= 'A' && c <= 'M') {
                y++;
            } else if (c >= 'N' && c <= 'Z') {
                y--;
            }
        }
        if (x == 0 && y == 0) {
            System.out.println(0);
        } else {
            if (x > 0) {
                for (int i = 0; i < x; i++) {
                    sb.append('a');
                }
            } else {
                for (int i = 0; i < -x; i++) {
                    sb.append('n');
                }
            }
            if (y > 0) {
                for (int i = 0; i < y; i++) {
                    sb.append('A');
                }
            } else {
                for (int i = 0; i < -y; i++) {
                    sb.append('N');
                }
            }
            System.out.println(sb.length());
            System.out.println(sb);
        }
    }
}
