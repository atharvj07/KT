#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, a, b;
  cin >> n >> a >> b;
  if (a + b > n || a * b == 0) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    vector<vector<int>> adj(n, vector<int>(n, 0));
    for (int i = 0; i < a; i++) {
      for (int j = i + 1; j < n; j++) {
        adj[i][j] = adj[j][i] = 1;
      }
    }
    for (int i = a; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if (b > 0) {
          adj[i][j] = adj[j][i] = 1;
          b--;
        }
      }
    }
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        cout << adj[i][j];
      }
      cout << endl;
    }
  }
  return 0;
}
