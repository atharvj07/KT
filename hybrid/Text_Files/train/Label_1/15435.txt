#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    ll a[3][2];
    for (int i = 0; i < 3; i++) { cin >> a[i][0] >> a[i][1]; }
    ll x1 = a[1][0] - a[0][0], y1 = a[1][1] - a[0][1];
    ll x2 = a[2][0] - a[1][0], y2 = a[2][1] - a[1][1];
    ll e = x1 * y2 - y1 * x2;
    if (e == 0) {
        cout << "No" << endl;
        return 0;
    }
    cout << "Yes" << endl;
    return 0;
}
