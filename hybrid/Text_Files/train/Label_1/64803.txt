import sys

n, k, m = map(int, input().split())
a = list(map(int, input().split()))

q = []
ans = 0

for elem in a:
    q.append((elem, k))
    ans += elem

q.sort(key = lambda x: x[0])

while m > 0:
    top = q[0][0] + 1
    if top > q[0][1]:
        q.pop(0)
    else:
        q[0] = (top, q[0][1])
        ans += 1
    m -= 1

if len(q) > 0:
    ans -= q[0][0] * len(q)

print(ans / n)
