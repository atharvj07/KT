import sys
def perm_differs():
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    if n == 2: return print(*a[::-1])
    for i in range(n-1):
        b = [a[i+1]] + [x for x in a if x!=a[i+1]]
        for j in range(i+1, n-1):
            if b[j] == b[i]: return print(-1)
        return print(*b)
    return print(-1)

perm_differs()
