n = int(input())
a = list(map(int, input().split()))
dp = [[0] * (n + 1) for i in range(n + 1)]
for i in range(n - 1, -1, -1):
    for j in range(i + 1):
        for k in range(2, n - i + 1):
            dp[i][j] = max(dp[i][j], dp[i + k][j + k] + sum(a[i:i + k]))
        dp[i][j + 1] = max(dp[i][j + 1], dp[i + 1][j + 1])
print(dp[0][0] - dp[1][0])
