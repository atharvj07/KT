import sys

n = int(sys.stdin.readline())
intervals = []
for i in range(n):
    l, r = map(int, sys.stdin.readline().split())
    intervals.append((l, r))

intervals.sort(key=lambda x: x[1])

count = 1
last = intervals[0][1]
for i in range(1, n):
    if intervals[i][0] >= last:
        count += 1
        last = intervals[i][1]

print(count)
