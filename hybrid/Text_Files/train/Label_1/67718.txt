import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    inv = 0
    for i in range(n):
        for j in range(i + 1, n):
            if a[j] < a[i]:
                inv += 1
    print("YES" if inv <= (n * (n - 1)) // 2 - 1 else "NO")
