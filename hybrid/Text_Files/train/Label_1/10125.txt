#include <bits/stdc++.h>
using namespace std;
const int mod = 998244353;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int n, k;
  cin >> n >> k;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) cin >> v[i].first >> v[i].second;
  sort(v.begin(), v.end());
  vector<long long> dp(1000001, 0);
  for (int i = 0; i < n; i++) {
    int b = v[i].first, e = v[i].second;
    dp[b]++, dp[e + 1]--;
  }
  for (int i = 1; i < 1000001; i++) dp[i] += dp[i - 1];
  int lm = v[0].first, rm = v[0].second;
  for (int i = 1; i < n; i++) {
    if (rm < v[i].first) {
      lm = v[i].first, rm = v[i].second;
    } else {
      rm = max(rm, v[i].second);
    }
  }
  dp[rm]--, dp[lm]++;
  for (int i = 1; i < 1000001; i++) dp[i] += dp[i - 1];
  long long ans = 1;
  for (int i = 0; i < k; i++) ans = ans * dp[i] % mod;
  cout << ans << endl;
}
