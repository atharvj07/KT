for _ in range(int(input())):
    n, m = list(map(int, input().split()))
    s = input()
    s = ' ' + s + ' '
    for i in range(m):
        t = ''
        for j in range(1, n + 1):
            x = 1 if s[j-1] == '1' else 0
            y = 1 if s[j+1] == '1' else 0
            z = x + y
            if s[j] == '0' and z == 1:
                t += '1'
            else:
                t += s[j]
        s = t
    print(s[1:n+1])
