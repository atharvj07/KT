import sys
input = sys.stdin.readline

n = int(input())

g = [[]] * (n + 1)
for i in range(n - 1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

depth = [-1] * (n + 1)
par = [-1] * (n + 1)
ans = 0

def dfs(u, p, d):
    global ans
    depth[u] = d
    for v in g[u]:
        if v != p:
            par[v] = u
            dfs(v, u, d + 1)
    if len(g[u]) == 1:
        ans += d

dfs(1, -1, 0)

print(ans)

leaves = []
for i in range(1, n + 1):
    if len(g[i]) == 1:
        leaves.append(i)

for l in leaves:
    print(l, par[l], l)
