#include <bits/stdc++.h>
using namespace std;
#define mod (int)(1e9 + 7)
int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  unordered_map<int, int> odd, cnt;
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      int val = (a[i] * a[j] % mod + mod) % mod;
      int sq = (int)(sqrt(val) + 0.5);
      if (sq * sq != val) continue;
      if (!(sq & 1)) {
        if (cnt[sq])
          odd[sq] = (odd[sq] + cnt[sq] % mod) % mod;
        else
          odd[sq] = 1;
      }
      cnt[val]++;
    }
  }
  int ans = 0;
  for (auto it : odd) {
    ans = (ans + ((it.second * (it.second - 1)) % mod)) % mod;
  }
  cout << ans;
  return 0;
}
