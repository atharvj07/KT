from fractions import Fraction
def gcd(a, b):
    if a == 0:
        return b
    return gcd(b % a, a)
def solve(a1, b1, a2, b2):
    g = gcd(a1, b1)
    a1 //= g
    b1 //= g
    g = gcd(a2, b2)
    a2 //= g
    b2 //= g
    if a1 * b2 != a2 * b1:
        return -1
    ans = 0
    while a1 != a2 or b1 != b2:
        if a1 < a2 or (a1 == a2 and b1 < b2):
            a1 *= 2
            b1 //= 2
            ans += 1
        else:
            a2 *= 2
            b2 //= 2
            ans += 1
    return ans, a1, b1
a1, b1 = map(int, input().split())
a2, b2 = map(int, input().split())
ans = solve(a1, b1, a2, b2)
print(ans[0])
if ans[0] != -1:
    print(*ans[1:])
    print(*ans[1:])
