#include <iostream>
#include <vector>
using namespace std;

int main() {
  int p, q, l, r;
  cin >> p >> q >> l >> r;
  vector<pair<int, int>> z, x;
  for (int i = 0; i < p; i++) {
    int a, b;
    cin >> a >> b;
    z.push_back({a, b});
  }
  for (int i = 0; i < q; i++) {
    int c, d;
    cin >> c >> d;
    x.push_back({c, d});
  }
  int ans = 0;
  for (int t = l; t <= r; t++) {
    bool ok = false;
    for (auto& [a, b] : z) {
      for (auto& [c, d] : x) {
        if (a <= c + t && b >= c + t) {
          ok = true;
          break;
        }
      }
      if (ok) break;
    }
    if (ok) ans++;
  }
  cout << ans << endl;
}
