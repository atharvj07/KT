n, m = map(int, input().split())
a = [list(map(int, input().split())) for i in range(n)]
s = [[[] for j in range(m)] for i in range(n)]
for i in range(n):
    for j in range(m):
        if a[i][j] == 1:
            if i > 0 and a[i - 1][j] == 1:
                s[i][j] += s[i - 1][j]
            if j > 0 and a[i][j - 1] == 1:
                s[i][j] += s[i][j - 1]
        s[i][j].append((i, j))
print(sum(len(x) for x in s if x))
