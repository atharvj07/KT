#include<bits/stdc++.h>
using namespace std;
int main(){
  long long n, x, total = 0, distressed = 0;
  cin >> n >> x;
  while (n--) {
    char c;
    long long t;
    cin >> c >> t;
    if (c == '-') {
      if (t > x)
        distressed++;
      else {
        x -= t;
        total++;
      }
    } else
      x += t;
  }
  cout << x << " " << distressed << endl;
}
