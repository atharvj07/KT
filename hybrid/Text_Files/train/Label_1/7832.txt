#include <bits/stdc++.h>
using namespace std;

bool vis[23][23];
int n, m;
int ans[23];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= m; ++i) {
        int u, v;
        cin >> u >> v;
        vis[u][v] = vis[v][u] = true;
    }
    int c = 0;
    for (int i = 1; i <= n; ++i) {
        bool ok = true;
        for (int j = 1; j <= n; ++j) {
            if (i == j) continue;
            if (!vis[i][j]) {
                vis[i][j] = vis[j][i] = true;
                ans[++c] = i;
                ok = false;
                break;
            }
        }
        if (ok) break;
    }
    cout << c << endl;
    for (int i = 1; i <= c; ++i) { cout << ans[i] << " "; }
    cout << endl;
    return 0;
}
