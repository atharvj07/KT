#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;

    vector<bool> read(n);
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        read[i] = (x == 1);
    }

    int unread = 0;
    for (bool r : read) {
        if (!r) {
            unread++;
        }
    }

    int min_ops = unread;
    int cur = 0;
    while (unread > 0) {
        if (!read[cur]) {
            unread--;
        }
        read[cur] = true;
        if (unread > 0) {
            min_ops++;
        }
        cur = (cur + 1) % n;
    }

    cout << min_ops << endl;

    return 0;
}
