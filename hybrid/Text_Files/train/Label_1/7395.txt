#include <bits/stdc++.h>
using namespace std;
long long solve() {
  int n;
  cin >> n;
  vector<pair<long long, long long>> v(n);
  for (int i = 0; i < n; ++i) { cin >> v[i].first >> v[i].second; }
  long long ans = LLONG_MAX;
  long long tot = 0;
  for (int i = 0; i < n; ++i) {
    long long now = 0;
    for (int j = 0; j < n; ++j) { now += max(v[(i + j) % n].first - j * v[(i + j) % n].second, 0LL); }
    if (ans > now) ans = now;
    tot += v[i].first;
  }
  cout << ans << "\n\n";
  return 0;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) solve();
  return 0;
}
