import math

n = int(input())
a = list(map(int, input().split()))

mn = a[0]
idx = 0
for i in range(1, n):
    if a[i] < mn:
        mn = a[i]
        idx = i

ans = []

for i in range(n):
    if i == idx:
        continue
    if math.gcd(a[i], mn) == 1:
        ans = [i+1, idx+1]
        break

if not ans:
    ans = [idx+1, 1]

print(*ans)
