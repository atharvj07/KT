#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
int w[505][505];
int main(){
    int t;
    cin>>t;
    while(t--)
    {
        int n,m,ans = 0;
        cin>>n>>m;
        for(int i=1;i<=n;i++)
        {
            string s;
            cin>>s;
            for(int j=1;j<=m;j++)
            {
                if(s[j-1]=='*'){
                    w[i][j] = min(w[i][j-1],min(w[i-1][j],w[i-1][j-1]))+1;
                    ans+=w[i][j];
                }
                else w[i][j] = 0;
            }
        }
        cout<<ans<<endl;
    }
}
