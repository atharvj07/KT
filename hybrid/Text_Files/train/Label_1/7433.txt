#include <cstdio>
#include <cstring>
int a[3][6],dp[100],x,ans;
int main(){
    scanf("%d",&x);
    for(int i=0;i<x;++i){
        for(int j=0;j<6;++j){
            scanf("%d",&a[i][j]);
        }
    }
    dp[0]=1;
    for(int i=1;i<=98;++i){
        for(int j=0;j<x;++j){
            int tmp=0;
            for(int k=0;k<6;++k){
                if(dp[i-a[j][k]]){
                    ++tmp;
                }
            }
            if(tmp==6)dp[i]=1;
        }
    }
    for(int i=98;i>=1;--i){
        if(dp[i]){
            printf("%d",i);
            return 0;
        }
    }
    printf("0");
    return 0;
}
