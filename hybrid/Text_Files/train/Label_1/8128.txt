#include <iostream>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  long long a[n], b[n];
  for (int i = 0; i < n; ++i) std::cin >> a[i];
  std::copy(a, a + n, b);
  std::sort(b, b + n);
  bool ok = true;
  for (int i = 1; i < n; ++i) {
    if (a[i] == a[i - 1] && b[i] != b[i - 1]) {
      ok = false;
      break;
    }
  }
  std::cout << (ok ? "Yes" : "No") << '\n';
  return 0;
}
