#include <iostream>
#include <string>

using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  if (a == b) {
    cout << "YES";
    return 0;
  }
  int n = a.size();
  int i;
  for (i = 0; i < n / 2; i++) {
    if (a[i] != b[n - 1 - i]) break;
  }
  if (i == n / 2) {
    for (i = n / 2; i < n; i++) {
      if (a[i] != b[i - n / 2]) break;
    }
  }
  if (i == n) {
    cout << "YES";
    return 0;
  }
  for (i = 0; i < n / 2; i++) {
    if (a[i] != b[i]) break;
  }
  if (i == n / 2) {
    for (i = n / 2; i < n; i++) {
      if (a[i] != b[n - 1 - i + n / 2]) break;
    }
  }
  if (i == n) {
    cout << "YES";
    return 0;
  }
  cout << "NO";
  return 0;
}
