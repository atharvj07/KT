#include <iostream>
#include <cstring>
#include <cctype>

struct State {
  int x, y, dir;
  int vars[4];
  bool is_ok;

  State(int ix, int iy, int idir) : x(ix), y(iy), dir(idir), is_ok(false) {
    memset(vars, 0, sizeof(vars));
  }
};

void read_instructions(const char* name, char* instructions) {
  int name_len = strlen(name);
  int plen = 0;
  for (int i = 0; i < name_len; ++i) {
    char c = name[i];
    if (c == '-' || c == '_') {
      instructions[plen++] = '>';
    } else if (c == '\\' || c == '|') {
      instructions[plen++] = 'v';
    } else if (c == '/' || c == 'y') {
      instructions[plen++] = '^';
    } else if (c == '>' || c == 'n') {
      instructions[plen++] = '<';
    }
  }
  instructions[plen] = '\0';
}

void run_program(const char* instructions, State& state) {
  int len = strlen(instructions);
  for (int i = 0; i < len; ++i) {
    char c = instructions[i];
    if (c == '<' || c == '>' || c == 'v' || c == '^') {
      state.dir = (state.dir + (c == '<' || c == 'v' ? -1 : 1) + 4) % 4;
    } else if (c == '=') {
      state.vars[0] = state.vars[1];
    } else if (c == '\\') {
      state.vars[1] = state.vars[0];
    } else if (isdigit(c)) {
      state.vars[state.dir] = c - '0';
      state.is_ok = true;
    }
    if (state.is_ok) {
      int nx = state.x + (state.dir == 0 ? 1 : state.dir == 2 ? -1 : 0);
      int ny = state.y + (state.dir == 1 ? 1 : state.dir == 3 ? -1 : 0);
      state.x = nx;
      state.y = ny;
    }
  }
}

int main() {
  char name[1024];
  std::cin >> name;
  char instructions[1024];
  State state(0, 0, 0);
  read_instructions(name, instructions);
  run_program(instructions, state);
  if (!state.is_ok) {
    std::cout << "false" << std::endl;
  } else {
    for (int i = 0; i < 4; ++i) {
      std::cout << state.vars[i];
    }
    std::cout << std::endl;
  }
  return 0;
}
