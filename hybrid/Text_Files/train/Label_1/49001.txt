import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    String n = scanner.nextLine();
    System.out.println((isPalindrome(n) ? "Yes" : "No"));
  }

  private static boolean isPalindrome(String n) {
    int l = 0;
    int r = n.length() - 1;
    while (l < r) {
      if (n.charAt(l) != n.charAt(r)) {
        return false;
      }
      l++;
      r--;
    }
    return true;
  }
}
