#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  vector<pair<int, int>> c(n);
  for (int i = 0; i < n; i++) {
    c[i] = {a[i] + b[i], i};
  }
  sort(c.begin(), c.end());
  for (int i = 0; i < n; i++) {
    b[c[i].second] = i;
  }
  for (int i = 0; i < n; i++) {
    cout << b[i] << " ";
  }
  return 0;
}
