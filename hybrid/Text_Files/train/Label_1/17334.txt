#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, a, x;
  cin >> n >> m;
  vector<int> A(n), cnt(m + 1, 0);
  for (int i = 0; i < n; ++i) { cin >> A[i]; }
  x = m;
  for (int i = 0; i < n; ++i) {
    if (A[i] <= m) cnt[A[i]]++;
    else
      x--;
  }
  cout << x << ' ' << n - x << '\n';
  for (int i = 0; i < n; ++i) {
    if (A[i] <= m) {
      cout << A[i] << ' ';
      cnt[A[i]]--;
      if (!cnt[A[i]]) x++;
    } else
      cout << x << ' ';
  }
}
