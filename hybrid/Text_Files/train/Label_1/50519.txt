import java.util.Scanner;

public class JoystickCharger {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a1 = scanner.nextInt();
        int a2 = scanner.nextInt();

        int minutes = 0;
        while (a1 > 0 && a2 > 0) {
            minutes++;
            if (a1 <= 1) {
                a1 = 100;
            } else {
                a1 -= 2;
            }
            if (a2 <= 1) {
                a2 = 100;
            } else {
                a2 -= 2;
            }
        }
        System.out.println(minutes);
    }
}
