#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v(n + 2, 1e9);
  for (int i = 1; i <= n; i++) {
    cin >> v[i];
  }
  for (int i = n; i > 0; i--) {
    v[i] = min(v[i], min(v[i + 1], v[i - 1]));
    if (v[i] == 1e9) v[i] = 0;
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    ans += v[i];
  }
  cout << ans << endl;
  return 0;
}
