def max_coins(n):
    if n == 1:
        return 1
    elif n % 2 == 0:
        if n % 4 == 0 or n == 4:
            return n // 2
        else:
            return max_coins(n // 2) + (n // 2)
    else:
        return max_coins(n - 1) + 1

T = int(input().strip())
for _ in range(T):
    N = int(input().strip())
    print(max_coins(N))
