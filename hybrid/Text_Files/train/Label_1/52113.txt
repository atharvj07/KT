import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        List<int[]> intervals = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] parts = reader.readLine().split(" ");
            int[] interval = new int[2];
            interval[0] = Integer.parseInt(parts[0]);
            interval[1] = Integer.parseInt(parts[1]);
            intervals.add(interval);
        }
        int k = 0;
        List<Integer> possibleGroups = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int[] interval = intervals.get(i);
            boolean intersects = false;
            for (int j = 0; j < n; j++) {
                if (i == j) continue;
                int[] otherInterval = intervals.get(j);
                if (interval[0] < otherInterval[1] && interval[1] > otherInterval[0]) {
                    intersects = true;
                    break;
                }
            }
            if (!intersects) {
                k++;
                possibleGroups.add(i + 1);
            }
        }
        System.out.println(k);
        for (int group : possibleGroups) {
            System.out.print(group + " ");
        }
    }
}
