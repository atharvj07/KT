#include<iostream>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n,m,a[101][101],flag=0;
        cin>>n>>m;
        for(int i=1;i<=n;i++){
            for(int j=1;j<=m;j++){
                cin>>a[i][j];
            }
        }
        for(int i=1;i<=n;i++){
            for(int j=1;j<=m;j++){
                if(a[i][j]>0){flag=1;break;}
            }
        }
        !flag?cout<<"Jeel":cout<<"Ashish";
        cout<<endl;
    }
    return 0;
}
