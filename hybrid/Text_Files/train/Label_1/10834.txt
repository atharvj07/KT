#include <bits/stdc++.h>
using namespace std;
#define max_n 300005
#define ll long long
ll n, k, a[max_n], ans[max_n];
int main()
{
    cin >> n >> k;
    for (ll i = 0; i < n; i++)
        cin >> a[i];
    ll l = 0, r = 0, cnt = 0, pre = 0, mx = 0;
    while (r < n)
    {
        cnt += (a[r] == 0);
        while (cnt > k)
        {
            cnt -= (a[l] == 0);
            l++;
        }
        if (r - l + 1 > mx)
        {
            mx = r - l + 1;
            pre = l;
        }
        r++;
    }
    cout << mx << endl;
    for (ll i = 0; i < n; i++)
        cout << (i >= pre && i < pre + mx ? 1 : a[i]) << " ";
    cout << endl;
    return 0;
}
