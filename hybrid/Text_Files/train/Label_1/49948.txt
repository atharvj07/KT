import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] words = new String[n];
        for (int i = 0; i < n; i++) {
            words[i] = sc.next();
        }
        Set<String> seen = new HashSet<>();
        int ans = 0;
        for (String word : words) {
            if (!seen.add(word)) {
                ans = Math.max(ans, word.length());
            }
        }
        System.out.println(ans);
    }
}
