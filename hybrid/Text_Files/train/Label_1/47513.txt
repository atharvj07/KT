import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    long[] a = new long[n];
    long sum = 0;
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
      sum += a[i] * i;
    }
    long max = sum;
    for (int i = 0; i < n; i++) {
      max = Math.max(max, sum - a[i] * i + a[i] * n);
    }
    System.out.println(max);
  }
}
