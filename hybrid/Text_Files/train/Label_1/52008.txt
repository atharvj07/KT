import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nm = br.readLine().split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);
        char[][] grid = new char[n][m];
        boolean[] hasSouthRow = new boolean[n];
        boolean[] hasSouthCol = new boolean[m];
        int numBlack = 0;
        for (int i = 0; i < n; i++) {
            String line = br.readLine();
            for (int j = 0; j < m; j++) {
                grid[i][j] = line.charAt(j);
                if (grid[i][j] == '#') numBlack++;
                if (grid[i][j] == '.' || (grid[i][j] == '#' && (i == 0 || j == 0))) {
                    hasSouthRow[i] = true;
                    hasSouthCol[j] = true;
                }
            }
        }
        boolean possible = true;
        for (boolean b : hasSouthRow) possible &= b;
        for (boolean b : hasSouthCol) possible &= b;
        if (!possible) {
            System.out.println(-1);
            return;
        }
        System.out.println(numBlack);
    }
}
