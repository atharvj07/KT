import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int N = input.nextInt();
        Set<Integer> setX = new HashSet<>();
        int P = input.nextInt();
        for (int i = 0; i < P; i++) { 
            setX.add(input.nextInt());
        }
        int Q = input.nextInt();
        Set<Integer> setY = new HashSet<>();
        for (int i = 0; i < Q; i++) { 
            setY.add(input.nextInt());
        }
        boolean flag = true;
        for (int i = 1; i <= N; i++) {
            if (!setX.contains(i) && !setY.contains(i)) {
                flag = false;
                break;
            }
        }
        System.out.println(flag ? "I become the guy." : "Oh, my keyboard!");
    }
}
