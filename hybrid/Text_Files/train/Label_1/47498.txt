import java.util.*;

public class Main {
  static long[] a;
  static int n;

  static long solve() {
    Arrays.sort(a);
    long sum = 0;
    for (int i = 0; i < n; i++) sum += (long) (a[i] * i);
    long maxMove = a[0];
    for (int i = 1; i < n; i++) {
      maxMove = Math.max(maxMove, a[i] * (n + 1 >> 1) - a[i - 1] * (n >> 1));
    }
    return sum + maxMove;
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    n = in.nextInt();
    a = new long[n];
    for (int i = 0; i < n; i++) a[i] = in.nextLong();

    System.out.println(solve());
  }
}
