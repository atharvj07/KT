n, m = map(int, input().split())
a = list(map(int, input().split()))

# Sort the coffee cups in descending order of caffeine content
a.sort(reverse=True)

# Binary search on the number of days
left, right = 1, n + 1
while left < right:
    mid = (left + right) // 2
    total_pages_written = sum(max(0, a[i] - i // mid) for i in range(n))
    if total_pages_written >= m:
        right = mid
    else:
        left = mid + 1

if left > n:
    print(-1)
else:
    print(left)
