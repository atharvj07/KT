#include <iostream>
#include <vector>
const int MAXN = 3e5+5;
int x[MAXN], f[MAXN];
long long sum[MAXN];
int main() {
  int n; std::cin >> n;
  for (int i = 0; i < n; ++i) std::cin >> x[i];
  for (int i = 0; i < n; ++i) f[i] = x[i + 1] - x[i - 1];
  for (int i = 0; i < n; ++i) sum[0] += f[i];
  for (int i = 1; i < n; ++i) {
    sum[i] = sum[i - 1];
    sum[i] = (sum[i] + f[i] - f[i - 1]) % (int)(1e9 + 7);
  }
  std::cout << sum[n - 1] % (int)(1e9 + 7) << '\n';
  return 0;
}
