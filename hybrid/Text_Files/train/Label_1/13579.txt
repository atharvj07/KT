#include <bits/stdc++.h>

using namespace std;

int main() {
  long long ts, tf, t, n;
  cin >> ts >> tf >> t >> n;
  vector<long long> arr(n);
  for (int i = 0; i < n; i++) cin >> arr[i];
  long long lo = 0, hi = tf - 1;
  while (lo < hi) {
    long long mid = (lo + hi) / 2;
    int s = 0;
    if (mid < arr[0]) {
      s += 1;
    }
    for (long long i = 0; i < n - 1; i++) {
      if (arr[i + 1] <= mid && mid < arr[i + 1] + t) {
        s += 1;
      }
    }
    if (s <= n)
      hi = mid;
    else
      lo = mid + 1;
  }
  cout << lo << endl;
}
