#include <cstdio>

using namespace std;

int main() {
    int n, m, k;
    scanf("%d%d%d", &n, &m, &k);
    char grid[n + 1][m + 1];
    for (int i = 1; i <= n; i++) {
        scanf("%s", grid[i] + 1);
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j + k - 1 <= m; j++) {
            bool ok = true;
            for (int l = 0; l < k; l++) {
                ok &= grid[i][j + l] == '.';
            }
            if (ok) ans++;
        }
    }
    for (int i = 1; i + k - 1 <= n; i++) {
        for (int j = 1; j <= m; j++) {
            bool ok = true;
            for (int l = 0; l < k; l++) {
                ok &= grid[i + l][j] == '.';
            }
            if (ok) ans++;
        }
    }
    printf("%d\n", ans);
    return 0;
}
