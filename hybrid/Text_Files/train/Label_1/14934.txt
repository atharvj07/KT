#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n; cin >> n;
  string s; cin >> s;
  vector<int> lst(26, -1);
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int cur = s[i] - 'a';
    if (lst[cur] != -1 && lst[cur] >= i - n) ans++;
    lst[cur] = i;
  }
  if (ans == n) cout << -1 << '\n';
  else cout << ans << '\n';
}
