#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    int sum = 0, p = 1;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      sum += a[i];
      p *= a[i];
    }
    if (sum == 0 || p == 0)
      cout << 1 << endl;
    else
      cout << 0 << endl;
  }
}
