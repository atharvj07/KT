#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b;
    cin >> a >> b;
    if (a == 0 || b == 0)
      cout << 0 << " " << 0 << endl;
    else {
      long long ans = __gcd(a, b);
      long long x = a / ans, y = b / ans;
      if (x + y > 1)
        cout << ans << " " << 0 << endl;
      else
        cout << ans << " " << 1 << endl;
    }
  }
}
