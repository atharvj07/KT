#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k; scanf("%d", &n);
    scanf("%d", &k);
    vector<int> a(n);
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    vector<int> cnt(1000010);
    int r = 0, l = 0, ans = 0, l1 = 0, r1 = 0;
    while (r < n) {
        cnt[a[r]]++;
        while (cnt[a[r]] > k) {
            cnt[a[l]]--;
            l++;
        }
        if (r - l + 1 > ans) {
            ans = r - l + 1;
            l1 = l, r1 = r;
        }
        r++;
    }
    printf("%d %d", l1 + 1, r1 + 1);
    return 0;
}
