#include <iostream>

using namespace std;

int main() {
  long long n, ans = 0;
  cin >> n;
  for (long long i = 4; i <= n; ++i) {
    bool lucky = true;
    for (long long j = i; j > 0; j /= 10) {
      if (j % 10 != 4 && j % 10 != 7) {
        lucky = false;
        break;
      }
    }
    if (lucky) {
      ++ans;
    }
  }
  cout << ans << endl;

  return 0;
}
