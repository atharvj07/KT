#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 300001, maxk = 300001;
int n, x[maxn], y[maxn], f[maxk];
ll ans[maxk];

struct pt {
    ll x, y;
    pt(ll _x = 0, ll _y = 0) : x(_x), y(_y) {}
    bool operator<(const pt& b) const { return x == b.x ? y > b.y : x < b.x; }
    pt operator-(const pt& b) const { return pt(x - b.x, y - b.y); }
    ll cross(const pt& b) const { return x * b.y - y * b.x; }
    ll cross(const pt& a, const pt& b) const { return (a - *this).cross(b - *this); }
} p[maxn];

ll dist(const pt& a, const pt& b) { return abs(a.x - b.x) + abs(a.y - b.y); }

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> p[i].x >> p[i].y, ans[3] += dist(p[i], p[i % n + 1]);
    sort(p + 1, p + n + 1);
    for (int k = 4; k <= n; k++) {
        f[k] = f[k - 1];
        pt st;
        for (int i = n - k + 3; i <= n; i++) {
            pt nb = p[i];
            while (f[k - 2] != f[k] && (p[f[k - 2] - 1] - p[f[k - 2]])
                                           .cross(nb - p[f[k - 2]], p[f[k - 1]] - p[f[k - 2]]) > 0)
                f[k] = f[k - 2];
            if ((st - p[f[k]]).cross(nb - p[f[k]], p[f[k] + 1] - p[f[k]]) < 0) st = p[f[k]], f[k] = i;
        }
        ans[k] = ans[k - 1] - dist(p[f[k]], p[f[k] - 1]) - dist(p[f[k]], p[f[k] + 1]) + dist(p[f[k - 1]], p[f[k] + 1]);
    }
    for (int i = 3; i <= n; i++) cout << ans[i] << ' ';
    return 0;
}
