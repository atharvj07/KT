import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int or = a[i];
                for (int k = i + 1; k <= j; k++) {
                    or |= a[k];
                }
                if (or > a[j]) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
