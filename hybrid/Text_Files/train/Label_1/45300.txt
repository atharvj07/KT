import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n - 1], b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
        }
        String s = scanner.next();

        System.out.println(solve(n, a, b, s));
    }

    private static long solve(int n, int[] a, int[] b, String s) {
        long m = 1;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '1') {
                m *= 3;
            }
        }
        return m;
    }
}
