import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, u, v = map(int, input().split())
    a = list(map(int, input().split()))
    ans = 0
    a.sort()
    for i in range(n):
        if a[i] > 1:
            ans += min(u, v)
        elif a[i] == 1:
            ans += u
    print(ans)
