n = int(input().strip())
psychos = list(map(int, input().strip().split()))

def process(psychos):
    steps = 0
    while True:
        kills = [i for i in range(len(psychos)-1) if psychos[i] > psychos[i+1]]
        if not kills:
            break
        psychos = [psychos[i] for i in range(len(psychos)) if i not in kills or i-1 in kills]
        steps += 1
    return steps

print(process(psychos))
