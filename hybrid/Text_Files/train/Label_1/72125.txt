from math import gcd
from functools import reduce

def lcm(x, y):
    return x * y // gcd(x, y)

n = int(input().strip())
a = list(map(int, input().strip().split()))

gcd_result = lcm(a[0], a[1])
for i in range(n):
    for j in range(i+1, n):
        gcd_result = gcd(gcd_result, lcm(a[i], a[j]))

print(gcd_result)
