import java.io.*;
import java.util.*;

public class Main {
    /*
    * n - the number of cups on the royal table
    * a - 1st cup
    * b - 2nd cup
    * max - max possible volume in a cup
    */
    static int n, a, b, max;
    final static int MINUS_ONE = -1;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        n = Integer.parseInt(br.readLine());
        ArrayList<Integer> l = new ArrayList<Integer>(n);
        for (int i = 0; i < n; ++i) {
            l.add(Integer.parseInt(br.readLine()));
        }

        main(l);
    }

    //overloaded method
    public static void main(List<Integer> l) {
        boolean ok = true;
        for (int i = 0; i < n && ok; ++i) {
            int v = l.get(i);
            if (v < 0 || v > max) {
                ok = false;
            } else {
                if (v == max) {
                    a = MINUS_ONE;
                    b = i + 1;
                } else {
                    a = i + 1;
                    b = MINUS_ONE;
                }
            }
        }

        l.sort(null);
        int diff = l.get(n - 1) - l.get(0);
        if (diff <= 0 || b < 0 || a < 0 || (a > 0 && b > 0))
            System.out.println("Unrecoverable configuration.");
        else
            System.out.println(diff + " ml. from cup #" + a + " to cup #" + b + ".");
    }

    static {
        max = 10000;
    }
}
