import sys
input = sys.stdin.readline

for t in range(int(input())):
    input()
    n, m = map(int, input().split())
    g = [[]]
    for i in range(n):
        g.append([])
    for i in range(m):
        a, b = map(int, input().split())
        g[a].append(b)
    vis = [0] * (n + 1)
    ans = [2] * (n + 1)
    def dfs(node):
        vis[node] = 1
        ans[node] = 0
        for u in g[node]:
            if vis[u] == 0:
                if dfs(u) == -1:
                    ans[node] = -1
            elif vis[u] == 1:
                ans[node] = -1
        vis[node] = 2
        return ans[node]
    for i in range(1, n + 1):
        if vis[i] == 0:
            if dfs(i) == -1:
                ans = [-1] * (n + 1)
                break
    print(*ans[1:])
