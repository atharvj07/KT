n = int(input().strip())
a = list(map(int, input().split()))
t = list(map(int, input().split()))

a_t_pairs = sorted(zip(a, t))

time = 0
current_max = 0
for ai, ti in a_t_pairs:
    if ai > current_max:
        current_max = ai
    else:
        diff = current_max - ai + 1
        time += diff * ti
        current_max += 1

print(time)
