from collections import deque

n, m = map(int, input().split())
a = [['.' for j in range(m)] for i in range(n)]
for i in range(n):
    a[i] = list(input())

r1, c1 = map(int, input().split())
r2, c2 = map(int, input().split())
r1 -= 1
c1 -= 1
r2 -= 1
c2 -= 1
a[r1][c1] = 'X'
q = deque([(r1, c1)])

while q:
    r, c = q.popleft()
    if r == r2 and c == c2:
        print('YES')
        exit()
    for nr, nc in [(r-1, c), (r+1, c), (r, c-1), (r, c+1)]:
        if 0 <= nr < n and 0 <= nc < m and a[nr][nc] == '.':
            q.append((nr, nc))
            a[nr][nc] = 'X'
print('NO')
