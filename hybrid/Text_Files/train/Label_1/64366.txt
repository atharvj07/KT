import bisect

m, k, n, s = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

if n * k > m or m - n * k < s: print(-1); exit()

cnt = [0] * 500001
for i in a: cnt[i] += 1

for x in b:
    if cnt[x] < cnt[b[0]]: b[0] = x

rem = n * k - m + s
pos = [0] * rem

for x in b:
    while cnt[x] and rem:
        i = bisect.bisect_left(a, x)
        if i < m and (i + k <= m or cnt[a[i + k]] == 0):
            rem -= 1
            pos[rem] = i + 1
            cnt[a[i]] -= 1
            a[i] = a[m - 1]
            m -= 1
        else: break

if rem == 0: print(rem); print(*pos)
else: print(-1)
