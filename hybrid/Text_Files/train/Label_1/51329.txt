import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int d = input.nextInt(), s = input.nextInt(), ans = -1;
        for (int i = 1; i <= 1000000; i++) {
            int sum = 0, num = i;
            while (num > 0) {
                sum += num % 10;
                num /= 10;
            }
            if (sum == s && i % d == 0) {
                ans = i;
                break;
            }
        }
        System.out.println(ans);
    }
}
