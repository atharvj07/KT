import sys
input = sys.stdin.readline
inf = 1 << 60

for q in range(int(input())):
    n = int(input())
    a, b = [0] * n, [0] * n
    for i in range(n):
        a[i], b[i] = map(int, input().split())
    mn, ans = inf, 0
    for i in range(n):
        if i > 0 and a[i] == a[i - 1]:
            ans += min(mn, b[i])
        mn = min(mn, b[i])
    print(ans)
