#include<iostream>
using namespace std;
int main() {
    int n, k, a[100], ans = 0;
    cin >> n >> k;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        string s = to_string(a[i]), t = "";
        for (int j = 0; j < s.size(); j++)
            if (s[j] == '4' || s[j] == '7') t += s[j];
        if (t.size() <= k) ans++;
    }
    cout << ans << endl;
    return 0;
}
