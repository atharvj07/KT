import sys
t = int(input())
for _ in range(t):
    n = int(input())
    a = [list(map(int, input().split())) for i in range(n)]
    ans, sum = 0, 0
    for h, d in a:
        if h > d:
            ans += 1
            h -= d + 1
            sum += h
        sum += d
    ans += (sum + n - 1) // n
    print(ans)
