#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define ll long long
#define fo(i,n) for(int i=0; i<n; i++)
vector<int> solve(vector<int> a) {
    map<int, int> mp;
    ll sum = 0;
    int lz = 0;
    for (int &x : a) mp[x]++;
    vector<int> b;
    for (auto it : mp) {
        while (it.second--) {
            b.push_back(it.first);
            sum += it.first;
            if (sum == 0) lz++;
        }
    }
    if (lz != 0 || b.size() != a.size()) return vector<int>();
    return b;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        fo(i, n) cin >> a[i];
        vector<int> b = solve(a);
        if (b.size())
            cout << "YES\n";
        else
            cout << "NO\n";
        for (int x : b) cout << x << ' ';
        cout << '\n';
    }
    return 0;
}
