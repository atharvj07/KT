import java.util.*;
import static java.lang.Math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long ans = 0;
        for (long i = 2; i * i <= n; i++) {
            long j = 2 * i;
            if (n % j != 0) {
                continue;
            }
            if ((long)floor((1.0 * j) / 2) == (long)(j / 2)) {
                ans = max(ans, j);
            }
            while (n % j == 0) {
                n /= j;
            }
        }
        if (n > 1) {
            ans = max(ans, n);
        }
        if (ans == 0) {
            ans = 1;
        }
        System.out.println(ans);
    }
}
