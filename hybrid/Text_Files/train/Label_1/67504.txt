n = int(input())
a = [int(i) for i in input().split()]
vis = [0 for i in range(0, n + 1)]
ans = 0
now = 1
vis[1] = 1
while now <= n:
    ans += 1
    while now <= n and vis[now] == 1:
        now += 1
    if now > n:
        break
    vis[now] = 1
    while a[now] != now:
        now = a[now]
        vis[now] = 1
    now += 1
print(ans)
