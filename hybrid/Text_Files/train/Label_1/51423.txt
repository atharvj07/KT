import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextInt();
            long k = sc.nextInt();
            System.out.println((n * k + n - 1 - 1) % n + 1);
        }
    }
}
