s = input()
t = s.split(";")
a = []
b = []
for word in t:
    t2 = word.split(",")
    for w in t2:
        if len(w) == 0:
            continue
        elif w[0] == '0' and len(w) > 1:
            b.append(word)
            break
        elif w.isdigit():
            a.append(word)
            break
        else:
            b.append(word)
print('"' + ",".join(a) + '"')
print('"' + ",".join(b) + '"' if len(b) > 0 else "-")
