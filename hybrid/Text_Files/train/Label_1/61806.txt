n = int(input())
s = list(map(int, input().split()))
q = int(input())
l = []
r = []
for i in range(q):
    a, b = map(int, input().split())
    l.append(a)
    r.append(b)
for i in range(n):
    s[i] += l[0]
ans = [s[0] for i in range(q)]
for i in range(1, q):
    for j in range(n):
        s[j] += r[i] - r[i - 1]
    ans[i] = s[0]
print(*ans)
