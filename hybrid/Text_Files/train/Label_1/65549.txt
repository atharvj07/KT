import re

good, pattern = input(), input()
bad = set(tuple(chr(ord('a') + i) for i in range(26) if chr(ord('a') + i) not in good))
queries = []
for _ in range(int(input())):
    queries.append(input())

for query in queries:
    if re.fullmatch('^'+pattern.replace('?', '[%s]' % ''.join(good)).replace('*', '[%s]*' % ''.join(bad))+'$', query):
        print('YES')
    else:
        print('NO')
