import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d1 = sc.nextInt();
        int d2 = sc.nextInt();
        int[][] a = new int[n * n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                a[i * n + j][0] = i;
                a[i * n + j][1] = j;
            }
        }
        int[][] b = new int[n * n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                b[i * n + j][0] = i;
                b[i * n + j][1] = (j + n / 2) % n;
            }
        }
        int[][] c = new int[n * n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                c[i * n + j][0] = (i + n / 2) % n;
                c[i * n + j][1] = j;
            }
        }
        int[][] d = new int[n * n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                d[i * n + j][0] = (i + n / 2) % n;
                d[i * n + j][1] = (j + n / 2) % n;
            }
        }
        int[] s = new int[n * n];
        if (d1 == d2) {
            for (int i = 0; i < n * n; i++) {
                s[i] = i;
            }
        } else if (d1 > d2) {
            for (int i = 0; i < n * n; i++) {
                int x = a[i][0] - b[i][0];
                int y = a[i][1] - b[i][1];
                s[i] = x * x + y * y == d1 ? 0 : 1;
            }
        } else {
            for (int i = 0; i < n * n; i++) {
                int x = c[i][0] - d[i][0];
                int y = c[i][1] - d[i][1];
                s[i] = x * x + y * y == d2 ? 0 : 1;
            }
        }
        for (int i = 0; i < n * n; i++) {
            if (s[i] == 0) {
                System.out.println(a[i][0] + " " + a[i][1]);
            } else {
                System.out.println(b[i][0] + " " + b[i][1]);
            }
        }
    }
}
