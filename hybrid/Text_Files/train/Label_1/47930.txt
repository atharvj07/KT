import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
        }
        int ans = 0, min = h[0], max = h[n - 1];
        for (int i = 1; i < n - 1; i++) {
            if (h[i] > min && h[i] < max) {
                ans++;
            }
            min = Math.min(min, h[i]);
        }
        System.out.println(ans);
    }
}
