import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            if (a[i] == 1) {
                b[i] = 1;
                for (int j = 2 * i; j <= n; j += i) {
                    b[j] = 1 - b[j];
                }
            }
        }
        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            if (b[i] == 1) {
                cnt++;
            }
        }
        System.out.println(cnt);
        for (int i = 1; i <= n; i++) {
            if (b[i] == 1) {
                System.out.print(i + " ");
            }
        }
        System.out.println();
    }
}
