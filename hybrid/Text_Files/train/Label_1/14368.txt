#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll x, n, k;

int main() {
  cin >> x;
  while (x--) {
    cin >> n >> k;
    if (k == 1)
      cout << n << '\n';
    else {
      ll temp = n;
      while (temp > 0) {
        if (temp % 10 != 0 && temp % 10 != k) {
          n = (n - temp % 10) + (k);
          break;
        } else
          temp /= 10;
      }
      if (temp == 0) n++;
      cout << n << '\n';
    }
  }
}
