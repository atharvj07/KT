n = int(input())
x, y = 0, 20000
for _ in range(n):
    t, d = input().split()
    t = int(t)
    if d == "North": y = min(20000, y + t)
    elif d == "South": y = max(0, y - t)
    elif d == "East": x = min(40000, x + t)
    elif d == "West": x = max(0, x - t)
    if x == 0 and y > 0 or y == 0 and x > 0 or (x == 0 and y == 0 and _ != n - 1):
        print("NO")
        break
else: print("YES")
