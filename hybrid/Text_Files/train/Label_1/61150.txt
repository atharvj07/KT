import sys
n=int(sys.stdin.readline())
x=[int(i) for i in sys.stdin.readline().split()]
y=[int(i) for i in sys.stdin.readline().split()]
if sum(x)!=sum(y):print('No');sys.exit()
a=[x[0]-y[0]]
for i in range(1,len(x)-1):a+=[x[i]-y[i],x[i]-y[i-1]]
a+=[x[-1]-y[-1]]
if sum(a)%3!=0:print('No');sys.exit()
a=[i//3 for i in a]
if min(a)<0:print('No');sys.exit()
print('Yes')
