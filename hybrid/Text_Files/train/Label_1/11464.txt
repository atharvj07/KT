#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int a[100001], l[100001], r[100001];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  int len = 1;
  for (int i = 2; i <= n; i++) {
    l[i] = 1;
    for (int j = 1; j < i; j++) {
      if (a[j] < a[i]) l[i] = max(l[i], l[j] + 1);
    }
    len = max(len, l[i]);
  }
  int len2 = 1;
  for (int i = n - 1; i >= 1; i--) {
    r[i] = 1;
    for (int j = n; j > i; j--) {
      if (a[j] < a[i]) r[i] = max(r[i], r[j] + 1);
    }
    len2 = max(len2, r[i]);
  }
  for (int i = 1; i <= n; i++) {
    if (l[i] + r[i] - 1 == len)
      cout << 3;
    else if (l[i] + r[i] - 1 == len2)
      cout << 1;
    else
      cout << 2;
  }
  cout << endl;
}
