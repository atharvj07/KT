#include <cstdio>

int main() {
    int x1, y1, z1, x2, y2, z2;
    scanf("%d:%d:%d\n%d:%d:%d", &x1, &y1, &z1, &x2, &y2, &z2);
    bool leap1 = (x1 % 400 == 0 || (x1 % 4 == 0 && x1 % 100 != 0)),
         leap2 = (x2 % 400 == 0 || (x2 % 4 == 0 && x2 % 100 != 0));
    int month[13] = {0, 31, 28 + leap1, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    long long ans = (x2 - x1) * 365 + (y2 - y1) * 30 + z2 - z1;
    for (int i = y1 + 1; i <= y2; ++i)
        ans += leap1 && i == 2 || leap2 && i == 2 ? -1 : 0;
    printf("%lld", ans);
    return 0;
}
