n, m = map(int, input().split())
a = list(map(int, input().split()))

operations = [list(map(int, input().split())) for _ in range(m)]

max_r = 0
for t, r in operations:
    max_r = max(max_r, r)

sorted_a = sorted(a[:max_r])
descending_a = sorted_a[::-1]

final_report = a
for i in range(m-1, -1, -1):
    t, r = operations[i]
    if r < max_r:
        continue
    if t == 1:
        final_report[:r] = sorted_a[:r]
    else:
        final_report[:r] = descending_a[:r]
    max_r = r - 1

print(*final_report)
