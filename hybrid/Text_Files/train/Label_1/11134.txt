#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n, k;
        cin >> n >> k;
        if (k > (1LL << n) * (1LL << n) - 1 || n == 1 && k) {
            cout << "NO\n";
            continue;
        }
        cout << "YES ";
        while (k && n) {
            cout << (1LL << (n / 2)) << ' ';
            k -= (1LL << n) << 2;
            n /= 2;
        }
        cout << '\n';
    }
    return 0;
}
