n = int(input().strip())
stations = list(map(int, input().split()))
k = int(input().strip())

# Sort stations by coordinates to minimize the sum of pairwise distances
stations_sorted = sorted((val, idx + 1) for idx, val in enumerate(stations))

# Choose k consecutive stations to minimize average commute time
selected_stations = [stations_sorted[i][1] for i in range(n - k + 1)]

# Print the indices of selected stations in any order
print(*selected_stations)
