#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  ll a[4][4], x, y, z;
  cin >> a[0][0] >> a[0][1] >> a[1][0] >> a[1][1];
  if (a[0][1] + a[1][0] - a[0][0] - a[1][1] < 0) {
    cout << "Impossible";
    return 0;
  }
  x = min(a[0][0], a[1][0]);
  y = min(a[0][0], a[0][1]);
  z = a[0][1] - x + a[1][0] - x;
  string s;
  for (ll i = 0; i < x; i++) s += "0";
  for (ll i = 0; i < y; i++) s += "1";
  for (ll i = 0; i < z; i++) s += "0";
  cout << s;
  return 0;
}
