import sys
input=sys.stdin.readline
n=int(input())
a=list(map(int,input().split()))
g=[[] for i in range(n+1)]
for i in range(n-1):
 u,v=map(int,input().split())
 g[u].append(v)
 g[v].append(u)
def dfs(u,f):
 c=a[u-1]
 for v in g[u]:
  if v==f:continue
  dfs(v,u)
  if a[v-1]>c:c=a[v-1]
 a[u-1]=c
dfs(1,-1)
print(max(a))
