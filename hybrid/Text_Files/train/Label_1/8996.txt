#include <bits/stdc++.h>
using namespace std;

const int MX = 1e5 + 5;
int n, m, deg[MX], a[MX], c[MX];
vector<int> adj[MX];

void dfs(int v, int p) {
    a[v] = c[v];
    for(int u: adj[v]) if(u != p) {
        dfs(u, v);
        if(a[u] < a[v]) swap(a[u], a[v]);
    }
}

int main() {
    cin >> n >> m;
    for(int i = 1; i <= n; i++) c[i] = i;
    while(m--) {
        int u, v; cin >> u >> v;
        adj[u].push_back(v); deg[v]++;
    }
    queue<int> q;
    for(int i = 1; i <= n; i++) if(deg[i] == 0) q.push(i);
    int idx = 1;
    while(!q.empty()) {
        int v = q.front(); q.pop();
        a[v] = idx++;
        for(int u: adj[v]) {
            c[u] = max(c[u], a[v] + 1);
            deg[u]--, q.push(u);
        }
    }
    dfs(1, 1);
    for(int i = 1; i <= n; i++) cout << a[i] << " "; cout << endl;
}
