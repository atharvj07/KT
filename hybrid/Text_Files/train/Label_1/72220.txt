import sys

d=int(input())
n,m=map(int,input().split())
a=[]
for i in range(d):
    q=[int(x) for x in input().split()]
    a.append(q)
l,r,t,b=map(int,input().split())

for i in a:
    cnt_l=cnt_r=cnt_t=cnt_b=0
    for j in a:
        if i==j:
            continue
        if i[0]>j[0]:
            cnt_l+=1
        elif i[0]<j[0]:
            cnt_r+=1
        elif i[1]>j[1]:
            cnt_t+=1
        elif i[1]<j[1]:
            cnt_b+=1

    if cnt_l==l and cnt_r==r and cnt_t==t and cnt_b==b:
        print(a.index(i)+1)
        sys.exit()
print(-1)
