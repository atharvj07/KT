#include<bits/stdc++.h>
#define ll long long int
#define vi vector<ll>
using namespace std;
bool comp(ll a, ll b){
    return a>b;
}
int main(){
    ll n,m,k,pos;
    cin>>n;
    vi a(n);
    for(ll i=0; i<n; i++){
        cin>>a[i];
    }
    sort(a.begin(),a.end(),comp);
    cin>>m;
    while(m--){
        cin>>k>>pos;
        if(pos>k){
            cout<<"-1\n";
            continue;
        }
        cout<<a[pos-1]<<'\n';
    }
    return 0;
}
