import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        long[] a = new long[k];
        long minN = 0, maxN = 0;
        for (int i = 0; i < k; i++) {
            a[i] = in.nextLong();
            minN = Math.max(minN, a[i]);
            maxN += a[i];
        }
        maxN--;
        if (minN * 2 > maxN) {
            System.out.println(-1);
            return;
        }
        System.out.println(minN + " " + maxN);
    }
}
