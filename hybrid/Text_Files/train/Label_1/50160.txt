import java.util.Scanner;

public class JOIWalk {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int T = scanner.nextInt();
        int Q = scanner.nextInt();

        int[] houses = new int[N];
        int[] directions = new int[N];
        for (int i = 0; i < N; i++) {
            houses[i] = scanner.nextInt();
            directions[i] = scanner.nextInt();
        }

        int[] importantPeople = new int[Q];
        for (int i = 0; i < Q; i++) {
            importantPeople[i] = scanner.nextInt() - 1;
        }

        int[] positions = new int[N];
        for (int i = 0; i < N; i++) {
            positions[i] = houses[i] + directions[i] * T;
        }

        for (int i = 0; i < Q; i++) {
            System.out.println(positions[importantPeople[i]]);
        }
    }
}
