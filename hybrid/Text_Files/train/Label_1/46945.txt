import java.io.*;
import java.util.*;
import java.lang.*; 

public class Main {
    static int[] tree;
    static int MAXN;
            
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        MAXN = n;
        tree = new int[n * n * 4];
        
        for (int i = 0; i < m; i++) {
            int op = sc.nextInt();
            if (op == 1) {
                int x1 = sc.nextInt();
                int y1 = sc.nextInt();
                int x2 = sc.nextInt();
                int y2 = sc.nextInt();
                System.out.println(get(1, 1, MAXN, 1, MAXN, x1, x2, y1, y2));
            } else {
                int x1 = sc.nextInt();
                int y1 = sc.nextInt();
                int x2 = sc.nextInt();
                int y2 = sc.nextInt();
                int v = sc.nextInt();
                update(1, 1, MAXN, 1, MAXN, x1, x2, y1, y2, v);
            }
        }
    }

    static int get(int node, int start, int end, int start1, int end1, int x1, int x2, int y1, int y2) {
        if (x2 < start || x1 > end || y2 < start1 || y1 > end1) return 0;
        else if (x1 <= start && end <= x2 && y1 <= start1 && end1 <= y2) return tree[node];
        else {
            int midx = start + (end - start) / 2;
            int midy = start1 + (end1 - start1) / 2;
            
            return get(node * 4, start, midx, start1, midy, x1, x2, y1, y2) ^
                   get(node * 4 + 1, midx + 1, end, start1, midy, x1, x2, y1, y2) ^
                   get(node * 4 + 2, start, midx, midy + 1, end1, x1, x2, y1, y2) ^
                   get(node * 4 + 3, midx + 1, end, midy + 1, end1, x1, x2, y1, y2);
        }
    }
    
    static void update(int node, int start, int end, int start1, int end1, int x1, int x2, int y1, int y2, int v) {
        if (x2 < start || x1 > end || y2 < start1 || y1 > end1) return;
        else if (x1 <= start && end <= x2 && y1 <= start1 && end1 <= y2) {
            tree[node] ^= v;
        } else {
            int midx = start + (end - start) / 2;
            int midy = start1 + (end1 - start1) / 2;
            
            update(node * 4, start, midx, start1, midy, x1, x2, y1, y2, v);
            update(node * 4 + 1, midx + 1, end, start1, midy, x1, x2, y1, y2, v);
            update(node * 4 + 2, start, midx, midy + 1, end1, x1, x2, y1, y2, v);
            update(node * 4 + 3, midx + 1, end, midy + 1, end1, x1, x2, y1, y2, v);
            
            tree[node] = tree[node * 4] ^ tree[node * 4 + 1] ^ tree[node * 4 + 2] ^ tree[node * 4 + 3];
        }
    }
}
