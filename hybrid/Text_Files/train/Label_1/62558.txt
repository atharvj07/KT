import heapq
import sys

n,m = map(int,input().split())
edges = [[] for i in range(n+1)]
dist = [[10**9] * (n+1) for i in range(n+1)]
for i in range(1,n+1):
    dist[i][i] = 0
for i in range(m):
    a,b,w = map(int,input().split())
    edges[a].append((b,w))
    edges[b].append((a,w))
a = list(map(int,input().split()))
for i in range(1,n+1):
    heap = [(0,i)]
    while heap:
        d,node = heapq.heappop(heap)
        if d > dist[i][node]: continue
        for t, w in edges[node]:
            nd = d + w
            if nd < dist[i][t]:
                dist[i][t] = nd
                heapq.heappush(heap,(nd,t))
ans = []
for i in range(1,n+1):
    mn = 10**9
    for j in range(1,n+1):
        mn = min(mn,dist[i][j] + dist[j][i] + a[j-1])
    ans.append(mn)
print(*ans)
