#include <iostream>
using namespace std;

const int MOD = 1e9 + 7;

int f[501][501];

int main() {
  int n;
  cin >> n;
  int a[n + 1];
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) {
    f[i][i] = 1;
    f[i - 1][i] = a[i];
  }
  for (int i = n - 1; i >= 1; i--) {
    for (int j = i; j <= n; j++) {
      (f[i - 1][j] += 1LL * f[i][j] * a[i]) %= MOD;
      (f[i][j] += 1LL * f[i - 1][j] * f[i][j] / a[i]) %= MOD;
    }
  }
  cout << f[0][n] << endl;
  return 0;
}
