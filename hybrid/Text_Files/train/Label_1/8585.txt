#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (auto &x : v)
        cin >> x;
    vector<vector<int>> increase;
    for (const auto &x : v) {
        if (increase.empty())
            increase.push_back(vector<int>{x});
        else if (increase.back().back() < x)
            increase.back().push_back(x);
        else
            increase.push_back(vector<int>{x});
    }
    for (const auto &x : increase) {
        for (const auto &y : x)
            cout << y << ' ';
        cout << endl;
    }
}
