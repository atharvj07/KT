import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long sx = input.nextLong();
        long sy = input.nextLong();
        long[] x = new long[n];
        long[] y = new long[n];
        for (int i = 0; i < n; i++) {
            x[i] = input.nextLong();
            y[i] = input.nextLong();
        }
        long minX, minY, maxX, maxY;
        minX = x[0];
        minY = y[0];
        maxX = x[0];
        maxY = y[0];
        for (int i = 1; i < n; i++) {
            minX = Math.min(minX, x[i]);
            minY = Math.min(minY, y[i]);
            maxX = Math.max(maxX, x[i]);
            maxY = Math.max(maxY, y[i]);
        }
        System.out.println(n);
        System.out.println((minX + maxX) / 2 + " " + (minY + maxY) / 2);
    }
}
