import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        Arrays.sort(d);
        int s = d[0];
        for (int i = 1; i < n; i++) {
            s = Math.max(s, Math.min(d[i] - d[i - 1], 24 - d[i] + d[i - 1]));
        }
        System.out.println(s);
    }
}
