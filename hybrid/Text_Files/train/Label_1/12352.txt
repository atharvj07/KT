#include <iostream>
#include <vector>
using namespace std;
int main() {
  int p, n;
  cin >> p >> n;
  vector<int> v(p, -1);
  int res = -1;
  for (int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    int idx = x % p;
    if (v[idx] != -1) { res = i; break; }
    v[idx] = x;
  }
  cout << res << endl;
  return 0;
}
