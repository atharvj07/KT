def solve_game(n, arr):
    dp = [0] * n
    for i in range(n - 1, -1, -1):
        moves = []
        for j in range(i + arr[i], n, arr[i]):
            moves.append(dp[j])
        if all(moves):
            dp[i] = 0
        else:
            dp[i] = 1
    return ''.join(['A' if x else 'B' for x in dp])

n = int(input().strip())
arr = list(map(int, input().strip().split()))
print(solve_game(n, arr))
