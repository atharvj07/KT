n, m, k = map(int, input().split())
instructions = [list(map(int, input().split())) for _ in range(n)]

locked_cores = set()
locked_cells = set()
core_lock_cycle = [0] * n

for cycle in range(m):
    write_attempts = {}
    for core, instruction in enumerate(instructions):
        if core in locked_cores or instruction[cycle] == 0:
            continue
        cell = instruction[cycle]
        if cell in write_attempts:
            write_attempts[cell].append(core)
        else:
            write_attempts[cell] = [core]

    for cell, cores in write_attempts.items():
        if len(cores) > 1 or cell in locked_cells:
            for core in cores:
                locked_cores.add(core)
                locked_cells.add(cell)
                if core_lock_cycle[core] == 0:
                    core_lock_cycle[core] = cycle + 1
        elif len(cores) == 1:
            core = cores[0]
            if cell not in locked_cells:
                locked_cells.add(cell)
            else:
                locked_cores.add(core)
                if core_lock_cycle[core] == 0:
                    core_lock_cycle[core] = cycle + 1

for cycle in core_lock_cycle:
    print(cycle)
