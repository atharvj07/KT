for _ in range(int(input())):
    p, f = map(int, input().split())
    s, w = map(int, input().split())
    v = list(map(int, input().split()))
    v.sort()
    ans = 0
    for g in v:
        if p >= g:
            p -= g
            ans += 1
        elif f >= g:
            f -= g
            ans += 1
        else:
            break
    print(ans)
