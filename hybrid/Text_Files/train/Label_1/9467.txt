#include <bits/stdc++.h>
using namespace std;
int32_t main() {
    int tc;
    cin >> tc;
    while (tc--) {
        int m;
        cin >> m;
        int a[2][m + 1];
        for (int i = 0; i < 2; i++) {
            for (int j = 0; j < m; j++) { cin >> a[i][j]; }
        }
        vector<int> dp(m + 1, 0);
        for (int j = 0; j < m; j++) {
            dp[j + 1] = max(dp[j + 1], dp[j] - a[0][j]);
            dp[j + 1] = max(dp[j + 1], dp[j] + a[1][j]);
        }
        cout << dp[m] << "\n";
    }
    return 0;
}
