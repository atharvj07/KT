import sys

t = int(input())

for i in range(t):
    a = sys.stdin.readline().strip()
    b = sys.stdin.readline().strip()
    c = sys.stdin.readline().strip()
    
    if a == b:
        print('YES')
    else:
        freq = {}
        for ch in a + b + c:
            if ch not in freq:
                freq[ch] = 0
            freq[ch] += 1
        
        for ch in freq:
            if freq[ch] % 3 != 0:
                print('NO')
                break
        else:
            print('YES')
