n, m, k = map(int, input().split())
classroom = [input() for _ in range(n)]

def count_ways(n, m, k, classroom):
    ways = 0

    # Horizontal check
    for row in classroom:
        count = 0
        for seat in row:
            if seat == '.':
                count += 1
                if count >= k:
                    ways += 1
            else:
                count = 0

    # Vertical check
    for col in range(m):
        count = 0
        for row in range(n):
            if classroom[row][col] == '.':
                count += 1
                if count >= k:
                    ways += 1
            else:
                count = 0

    # If k is 1, we need to subtract the overlap of horizontal and vertical counts
    if k == 1:
        ways //= 2

    return ways

print(count_ways(n, m, k, classroom))
