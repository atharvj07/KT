#include <algorithm>
#include <cstdio>
#include <vector>
typedef long long ll;

const int MAXV = 2e5 + 1;
ll cnt[MAXV];

int main() {
  int n, m, a, b;
  ll t1, t2;
  ll ans, tot;
  int test;
  scanf("%d", &test);
  for (int i = 0; i < test; i++) {
    scanf("%d%d%d%d", &n, &m, &a, &b);
    for (int j = 0; j < m; j++) scanf("%lld", &cnt[j]);
    std::sort(cnt, cnt + m);
    t1 = a - 1;
    t2 = n - a;
    ans = 0;
    tot = 0;
    for (int j = 0; j < m; j++) {
      if (t1 > t2) {
        t1 -= cnt[j];
        if (t1 >= 0) ans++;
      } else {
        t2 -= cnt[j];
        if (t2 >= 0) ans++;
      }
      if (t1 + t2 <= 0) {
        ans--;
        break;
      }
      tot += cnt[j];
      if (tot > t1 + t2) {
        ans++;
        t1 = t2 = (tot - 1) / 2;
      }
    }
    printf("%lld\n", ans);
  }
  return 0;
}
