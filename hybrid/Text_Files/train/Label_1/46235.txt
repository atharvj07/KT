import java.util.Scanner;

public class MoneyWithdrawal {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();

        int operations = 0;
        while (N > 0) {
            int powerOf6 = 0;
            while (Math.pow(6, powerOf6) <= N) {
                powerOf6++;
            }

            N -= Math.pow(6, powerOf6 - 1);
            operations++;

            if (N > 0) {
                int powerOf9 = 0;
                while (Math.pow(9, powerOf9) <= N) {
                    powerOf9++;
                }

                N -= Math.pow(9, powerOf9 - 1);
                operations++;
            }
        }

        System.out.println(operations);
    }
}
