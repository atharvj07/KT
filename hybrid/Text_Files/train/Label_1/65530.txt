import sys
n=int(sys.stdin.readline())
a=list(map(int,sys.stdin.readline().split()))
if sorted(a)==a:
    print(n)
    sys.exit()
ans=1
for i in range(n//2):
    if a[i]>a[i+1]:
        ans=i+1
        break
if ans==1:
    print(0)
else:
    print(max(Thanos(a[:ans]),Thanos(a[ans:])))
def Thanos(a):
    if sorted(a)==a:
        return len(a)
    if len(a)<=2:
        return 1
    ans1=Thanos(a[:len(a)//2])
    ans2=Thanos(a[len(a)//2:])
    return max(ans1,ans2)
