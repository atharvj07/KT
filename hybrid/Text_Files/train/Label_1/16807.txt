#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Droid {
    vector<int> details;
    int destroyed;
};

int main() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<Droid> droids(n);

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            int a;
            cin >> a;
            droids[i].details.push_back(a);
        }
    }

    // Count the number of droids that can be destroyed with each shot
    vector<int> count(m, 0);
    for (Droid& droid : droids) {
        int minDetails = *min_element(droid.details.begin(), droid.details.end());
        for (int i = 0; i < m; i++) {
            if (droid.details[i] <= minDetails) {
                count[i]++;
            }
        }
    }

    // Sort the shots in descending order of count
    vector<pair<int, int>> shots;
    for (int i = 0; i < m; i++) {
        shots.push_back({count[i], i});
    }
    sort(shots.begin(), shots.end(), greater<pair<int, int>>());

    // Destroy the maximum number of droids with the given number of shots
    vector<int> result(m, 0);
    for (int i = 0; i < min(k, (int)shots.size()); i++) {
        result[shots[i].second]++;
    }

    for (int shot : result) {
        cout << shot << " ";
    }
    cout << endl;

    return 0;
}
