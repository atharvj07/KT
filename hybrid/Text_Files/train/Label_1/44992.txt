import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), h = sc.nextInt();
        int w = 0;
        for (int i = 0; i < n; i++) {
            w += sc.nextInt() > h ? 2 : 1;
        }
        System.out.println(w);
    }
}
