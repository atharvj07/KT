import java.util.*;

public class Main {
  private static void solve() {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    while (t-- > 0) {
      int n = sc.nextInt();
      int[] perm = new int[n];
      for (int i = 0; i < n; i++)
        perm[i] = sc.nextInt();
      int count = 0;
      for (int i = 0; i < n; i++) {
        int min_idx = i;
        for (int j = i + 1; j < n; j++)
          if (perm[j] < perm[min_idx])
            min_idx = j;
        if (min_idx != i) {
          count++;
          for (int j = min_idx; j > i; j--) {
            int tmp = perm[j];
            perm[j] = perm[j - 1];
            perm[j - 1] = tmp;
          }
        }
      }
      System.out.println(count);
    }
  }

  public static void main(String[] args) {
    solve();
  }
}
