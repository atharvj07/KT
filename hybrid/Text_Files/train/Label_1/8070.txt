#include<bits/stdc++.h>
using namespace std;
const int mx=60;
int n,m,a[mx][mx],r,c,ex,ey,ans;
bool vis[mx][mx];
const int dx[]={0,0,1,-1};
const int dy[]={1,-1,0,0};
void dfs(int x,int y,string s){
	if(s.empty()){
		ans++;
		return;
	}
	for(int i=0;i<4;i++){
		int x1=x+dx[i];
		int y1=y+dy[i];
		if(x1>=0&&x1<n&&y1>=0&&y1<m&&!vis[x1][y1]&&a[x1][y1]!='#'){
			if(a[x1][y1]=='E'){
				ans++;
				return;
			}
			vis[x1][y1]=1;
			dfs(x1,y1,s.substr(1));
			vis[x1][y1]=0;
		}
	}
}
int main(){
	cin>>n>>m;
	for(int i=0;i<n;i++){
		string s;
		cin>>s;
		for(int j=0;j<m;j++){
			a[i][j]=s[j];
			if(a[i][j]=='S'){
				r=i;
				c=j;
			}
			if(a[i][j]=='E'){
				ex=i;
				ey=j;
			}
		}
	}
	string s;
	cin>>s;
	dfs(r,c,s);
	cout<<ans<<endl;
	return 0;
}
