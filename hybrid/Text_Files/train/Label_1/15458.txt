#include <iostream>

int main() {
    int t;
    std::cin >> t;
    for (int i = 0; i < t; i++) {
        int str, intl, exp;
        std::cin >> str >> intl >> exp;
        if (str > intl)
            std::cout << exp + 1 << std::endl;
        else
            std::cout << 0 << std::endl;
    }
    return 0;
}
