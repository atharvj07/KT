import sys

for t in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    if all(x % 2 == i % 2 for i, x in enumerate(a)): print(0); continue
    odd = 0
    for x in a: odd += x % 2
    even = n - odd
    if odd % 2 == 0 and even % 2 == 0: print(odd // 2)
    else: print(-1)
