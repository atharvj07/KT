import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = br.readLine().split(" ");
        int n = Integer.parseInt(nk[0]);
        int k = Integer.parseInt(nk[1]);
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(br.readLine());
        }
        int[] max_right_length = new int[n];
        int[] max_left_length = new int[n];
        max_right_length[n - 1] = 1;
        for (int i = n - 2; i >= 0; i--) {
            if (Math.abs(a[i] - a[i + 1]) <= k) {
                max_right_length[i] = max_right_length[i + 1] + 1;
            } else {
                max_right_length[i] = 1;
            }
        }
        max_left_length[0] = 1;
        for (int i = 1; i < n; i++) {
            if (Math.abs(a[i] - a[i - 1]) <= k) {
                max_left_length[i] = max_left_length[i - 1] + 1;
            } else {
                max_left_length[i] = 1;
            }
        }
        int max_length = 0;
        for (int i = 0; i < n; i++) {
            max_length = Math.max(max_length, max_left_length[i] + max_right_length[i] - 1);
        }
        System.out.println(max_length);
    }
}
