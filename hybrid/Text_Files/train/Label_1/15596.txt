#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) v[i] = 1;
  for (int i = 1; i < n; i++) {
    int j = i - 1;
    while (j >= 0 && v[j] == v[j + 1]) {
      v[j]++, v.erase(v.begin() + j + 1);
      j--;
    }
  }
  for (auto x : v) cout << x << " ";
  cout << "\n";
  return 0;
}
