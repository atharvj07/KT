#include<bits/stdc++.h>
using namespace std;
bool g[1000][1000];
int main(){
    int n,m,a,b;
    cin>>n>>m;
    for(int i=0; i<m; i++){
        cin>>a>>b;
        g[a][b]=1;
        g[b][a]=1;
    }
    for(int k=1; k<=n; k++)
        for(int i=1; i<=n; i++)
            for(int j=1; j<=n; j++)
                if(g[i][k]&&g[k][j]) g[i][j]=1;
    for(int i=1;i<=n;i++)
        for(int j=1; j<=n; j++)
            if(g[i][j]==0&&i!=j){ cout<<"no"; return 0; }
    cout<<"yes";
}
