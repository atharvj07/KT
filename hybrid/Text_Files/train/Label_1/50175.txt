import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class ChoutiAndSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] evenElements = new int[n / 2];
        for (int i = 0; i < n / 2; i++) {
            evenElements[i] = scanner.nextInt();
        }

        if (!isPossible(evenElements)) {
            System.out.println("No");
            return;
        }

        int[] restoredSequence = restoreSequence(evenElements);

        System.out.println("Yes");
        for (int element : restoredSequence) {
            System.out.print(element + " ");
        }
    }

    private static boolean isPossible(int[] evenElements) {
        int sum = 0;
        for (int element : evenElements) {
            sum += element;
        }

        return isPerfectSquare(sum);
    }

    private static int[] restoreSequence(int[] evenElements) {
        int[] restoredSequence = new int[evenElements.length * 2];
        restoredSequence[0] = (int) Math.sqrt(evenElements[0]);
        restoredSequence[1] = evenElements[0] - restoredSequence[0];

        for (int i = 1; i < evenElements.length; i++) {
            int previousSum = restoredSequence[i * 2 - 2] + restoredSequence[i * 2 - 1];
            int currentSum = previousSum + evenElements[i];
            int nextElement = (int) Math.sqrt(currentSum) - restoredSequence[i * 2 - 1];
            restoredSequence[i * 2] = nextElement;
            restoredSequence[i * 2 + 1] = evenElements[i] - nextElement;
        }

        return restoredSequence;
    }

    private static boolean isPerfectSquare(int n) {
        return Math.sqrt(n) == (int) Math.sqrt(n);
    }
}
