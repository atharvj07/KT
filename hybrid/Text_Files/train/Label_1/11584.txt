#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int l = -1e9, r = -1e9;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '1') { l = i; }
      if (i - l > k && i - r > k) { ans++; }
      if (s[i] == '1') { r = i; }
    }
    cout << ans << endl;
  }
}
