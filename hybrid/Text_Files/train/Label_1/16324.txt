#include <iostream>
#include <vector>

using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  int n = a.size(), m = b.size();
  vector<int> pref(m + 1);
  for (int i = 1; i <= m; i++) {
    pref[i] = pref[i - 1];
    if (b[i - 1] != a[0]) pref[i]++;
    if (i >= n) pref[i] -= pref[i - n];
  }
  long long ans = 0;
  for (int i = 0; i < m - n + 1; i++) { ans += pref[i + n] - pref[i]; }
  cout << ans << endl;
  return 0;
}
