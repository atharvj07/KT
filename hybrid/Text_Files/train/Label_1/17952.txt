#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
int32_t main() {
    int n; cin >> n;
    vector<int>a(n);
    for (int &x : a) cin >> x;
    int nneg = 0, npos = 0;
    int prv = 0;
    if (a[0] > 0) prv = 1, npos++;
    else if (a[0] < 0) prv = -1, nneg++;
    for (int i = 1; i < n; i++) {
        if (prv * a[i] > 0) {
            if (a[i] > 0) npos++;
            else nneg++;
        }
        else {
            prv *= -1;
            if (a[i] > 0) nneg++, npos++;
            else npos++, nneg++;
        }
    }
    cout << nneg << " " << npos << endl;
    return 0;
}
