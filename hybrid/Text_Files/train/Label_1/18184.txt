#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int solve(int x) {
    vector<int> digits;
    while (x > 0) {
        digits.push_back(x % 10);
        x /= 10;
    }
    sort(digits.begin(), digits.end());
    int sum = 0;
    for (int digit : digits) {
        sum += digit;
    }
    if (sum != x) {
        return -1;
    }
    int smallest = 0;
    for (int i = 0; i < digits.size(); i++) {
        if (i > 0) {
            smallest *= 10;
        }
        smallest += digits[i];
    }
    return smallest;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int x;
        cin >> x;
        cout << solve(x) << endl;
    }
    return 0;
}
