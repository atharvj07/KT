#include<bits/stdc++.h>
using namespace std ;
#define vi(x) vector<x>
#define pb push_back
const int mx = 1e5 + 10 ;
vi(int) adj[mx] , ans ;
bool vis[mx] ;
void dfs(int u , int p = -1) {
  vis[u] = 1 ;
  for(int v : adj[u]) {
    if(!vis[v]) {
      dfs(v , u) ;
    }
  }
  set<int> all ;
  for(int v : adj[u]) {
    if(u == p) continue ;
    for(int x : ans) {
      if(v == u) continue ;
      if(u == x) continue ;
      if(x != p) {
        all.insert(x) ;
      }
    }
  }
  int val = 0 ;
  while(all.find(val) != all.end())
    val++ ;
  if(u != p)
    ans.pb(val) ;
}
int32_t main() {
  int n ; cin >> n ;
  for(int i = 0 ; i < n - 1 ; i++) {
    int u , v ; cin >> u >> v ;
    adj[u].pb(v) , adj[v].pb(u) ;
  }
  dfs(1) ;
  for(int x : ans)
    cout << x << " " ;
  cout << endl ;
  return 0 ;
}
