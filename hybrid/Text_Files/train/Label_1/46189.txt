import java.util.Scanner;
import java.util.Arrays;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] dices = new int[n];
    for (int i = 0; i < n; i++) {
      int[] v = new int[6];
      for (int j = 0; j < 6; j++) {
        v[j] = in.nextInt();
      }
      Arrays.sort(v);
      StringBuilder sb = new StringBuilder();
      for (int j = 0; j < 6; j++) {
        sb.append(v[j]);
      }
      dices[i] = Integer.parseInt(sb.toString());
    }
    Arrays.sort(dices);
    int last = -1;
    boolean flag = true;
    for (int i : dices) {
      if (i == last) {
        flag = false;
        break;
      }
      last = i;
    }
    System.out.println(flag ? "Yes" : "No");
  }
}
