import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = reader.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int m = Integer.parseInt(line1[1]);
        int[][] trees = new int[n][4];
        for (int i = 0; i < n; i++) {
            String[] line2 = reader.readLine().split(" ");
            for (int j = 0; j < 4; j++) {
                trees[i][j] = Integer.parseInt(line2[j]);
            }
        }
        int[][] mushrooms = new int[m][2];
        for (int i = 0; i < m; i++) {
            String[] line3 = reader.readLine().split(" ");
            for (int j = 0; j < 2; j++) {
                mushrooms[i][j] = Integer.parseInt(line3[j]);
            }
        }
        Arrays.sort(trees, (a, b) -> Integer.compare(a[0], b[0]));
        Arrays.sort(mushrooms, (a, b) -> Integer.compare(a[0], b[0]));

        double ans = 0;
        for (int[] mushroom : mushrooms) {
            int pos = mushroom[0];
            int power = mushroom[1];
            int l = -1, r = -1;
            for (int[] tree : trees) {
                if (l == -1 && pos >= tree[0] - tree[1] && pos < tree[0]) {
                    l = tree[0];
                }
                if (r == -1 && pos > tree[0] && pos <= tree[0] + tree[1]) {
                    r = tree[0];
                }
            }
            if (l == -1) {
                l = Integer.MAX_VALUE;
            }
            if (r == -1) {
                r = -Integer.MAX_VALUE;
            }
            double probability = 1.0;
            for (int[] tree : trees) {
                if (tree[0] < pos) {
                    probability *= 1.0 * (100 - tree[2]) / 100;
                } else if (tree[0] > pos) {
                    probability *= 1.0 * (100 - tree[3]) / 100;
                }
            }
            if (l <= pos && pos < r) {
                ans += probability * power;
            }
        }
        System.out.printf("%.9f", ans);
    }
}
