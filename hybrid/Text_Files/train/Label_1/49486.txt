import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.stream.Collectors;

public class Main {
    static final int MOD = 998244353;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        Map<Integer, Integer> degrees = new HashMap<>();
        int[][] edges = new int[n - 1][2];
        for (int i = 0; i < n - 1; i++) {
            String[] line = br.readLine().split(" ");
            int u = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);
            edges[i] = new int[]{u, v};
            degrees.merge(u, 1, Integer::sum);
            degrees.merge(v, 1, Integer::sum);
        }
        int[] visited = new int[n + 1];
        Map<String, Long> dp = new HashMap<>();
        Arrays.fill(visited, -1);
        dfs(edges, degrees, 1, 0, dp, visited);
        System.out.println(dp.values().stream().reduce(1L, (a, b) -> (a * b) % MOD));
    }

    private static long dfs(int[][] edges, Map<Integer, Integer> degrees, int current, int parent, Map<String, Long> dp, int[] visited) {
        String key = current + "," + parent;
        if (dp.containsKey(key)) {
            return dp.get(key);
        }
        // If current vertex is a leaf
        if (degrees.get(current) == 1) {
            dp.put(key, 1L);
            return 1L;
        }
        // Calculate the number of paths if we take the edge to parent
        long pathsWithParent = 1L;
        for (int[] edge : edges) {
            int u = edge[0];
            int v = edge[1];
            if ((u == current && v == parent) || (v == current && u == parent)) {
                continue;
            }
            pathsWithParent = pathsWithParent * dfs(edges, degrees, u == current ? v : u, current, dp, visited) % MOD;
        }
        // Calculate the number of paths if we take an edge to a child
        long pathsWithChild = 0L;
        for (int[] edge : edges) {
            int u = edge[0];
            int v = edge[1];
            if ((u == current && v == parent) || (v == current && u == parent)) {
                continue;
            }
            pathsWithChild = pathsWithChild + dfs(edges, degrees, u == current ? v : u, current, dp, visited) % MOD;
        }
        // Store the result in dp
        dp.put(key, (pathsWithParent + pathsWithChild) % MOD);
        return (pathsWithParent + pathsWithChild) % MOD;
    }
}
