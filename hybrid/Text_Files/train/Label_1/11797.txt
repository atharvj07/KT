#include <iostream>
#include <algorithm>
#include <vector>
#include <map>
using namespace std;

const int maxN = 5000, maxM = 100;

int f[2][maxN];
int p[2][maxN];
int n, m;
vector<int> a, sol;
map<int,int> cnt;

int main() {
    cin >> n >> m;
    a.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        cnt[a[i]]++;
    }
    
    for (int i = 0; i < n; i++) {
        if (cnt[a[i]] > 1) {
            cnt[a[i]]--;
            sol.push_back(a[i]);
            continue;
        }
        for (int col = 1; col <= m; col++) {
            int cur = f[i%2][col] + 1;
            if (cur > f[(i+1)%2][a[i]]) {
                f[(i+1)%2][a[i]] = cur;
                p[(i+1)%2][a[i]] = col;
            }
        }
    }
    
    int ans = 0;
    for (int i = 1; i <= m; i++) {
        ans = max(ans, f[(n-1)%2][i]);
    }
    cout << ans << endl;
    int cur = ans;
    for (int i = n - 1; i >= 0; i--) {
        if (f[i%2][a[i]] == cur) {
            cur--;
            sol.push_back(p[(i+1)%2][a[i]]);
        }
    }
    reverse(sol.begin(), sol.end());
    while (sol.size() > n) sol.pop_back();
    
    for (int x: sol) {
        cout << x << ' ';
    }
    cout << endl;
    
    return 0;
}
