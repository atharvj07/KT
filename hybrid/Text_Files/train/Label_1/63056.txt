n = int(input())
L = []
d = {}
mn = 200
for i in range(n):
    S = input()
    L.append(S)
    mn = min(mn, len(S))
mn -= 1
for i in range(10):
    c = chr(ord('a')+i)
    d[c] = i
for i in range(n):
    S = L[i]
    for j in range(len(S)):
        S = S[:j]+str(d[S[j]])+S[j+1:]
    L[i] = int(S)
L.sort()
print(sum(L))
