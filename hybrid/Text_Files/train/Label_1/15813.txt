#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), cnt(n + 1, 0);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      cnt[a[i]]++;
    }
    int mx = 0, ans = -1;
    for (int i = 0; i <= n; i++) {
      if (cnt[i] > mx) {
        mx = cnt[i];
        ans = i;
      }
    }
    for (int i = 0; i < n; i++) {
      if (a[i] != ans) {
        cout << -1 << endl;
        goto here;
      }
    }
    cout << n << endl;
  here:;
  }
  return 0;
}
