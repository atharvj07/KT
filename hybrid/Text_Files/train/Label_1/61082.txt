n = int(input())
s = input()

text = []
colors = 0
results = []

cursor = 0
for command in s:
    if command == 'L':
        cursor = max(0, cursor - 1)
    elif command == 'R':
        cursor += 1
        if cursor > len(text):
            text.append(' ')
    else:
        if cursor < len(text):
            text[cursor] = command
        else:
            text.append(command)
        cursor += 1

    # Check if the text is correct and calculate colors
    balance = 0
    max_balance = 0
    is_correct = True
    for char in text:
        if char == '(':
            balance += 1
            max_balance = max(max_balance, balance)
        elif char == ')':
            balance -= 1
            if balance < 0:
                is_correct = False
                break
    if balance != 0:
        is_correct = False

    if is_correct:
        results.append(max_balance)
    else:
        results.append(-1)

print(' '.join(map(str, results)))
