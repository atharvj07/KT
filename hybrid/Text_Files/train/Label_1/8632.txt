#include <iostream>
#include <vector>

int main() {
  std::string pw;
  std::cin >> pw;
  int n;
  std::cin >> n;
  std::vector<std::string> words(n);
  for (int i = 0; i < n; i++) {
    std::cin >> words[i];
  }
  bool found = false;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (words[i] + words[j] == pw || words[j] + words[i] == pw) {
        found = true;
        break;
      }
    }
    if (found) break;
  }
  if (found)
    std::cout << "YES" << std::endl;
  else
    std::cout << "NO" << std::endl;
  return 0;
}
