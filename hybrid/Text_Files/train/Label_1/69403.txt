import sys
input=sys.stdin.readline

def dfs(v):
    vis[v]=1
    for i in adj[v]:
        if vis[i]==0:
            par[i]=v
            if dfs(i)==1:
                return 1
        else:
            if par[v]!=i:
                return 1
    return 0

n=int(input())
adj=[[]]
for i in range(n):
    adj.append([])
for i in range(n-1):
    a,b=map(int,input().split())
    adj[a].append(b)
    adj[b].append(a)
vis=[0]*(n+1)
par=[0]*(n+1)
if dfs(1)==1:
    print('No')
else:
    print('Yes')
    print(n-1)
    for i in range(2,n+1):
        print(par[i],i)
