import java.util.Deque;
import java.util.LinkedList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int K = sc.nextInt();
        int[] V = new int[N];
        for (int i = 0; i < N; i++) {
            V[i] = sc.nextInt();
        }
        int sum = 0;
        Deque<Integer> dq = new LinkedList<>();
        for (int i = 0; i < N; i++) {
            if (dq.isEmpty() || V[i] <= dq.peekLast()) {
                dq.addLast(V[i]);
            } else {
                while (!dq.isEmpty() && V[i] > dq.peekLast()) {
                    dq.pollLast();
                    K--;
                }
                if (K >= 0) {
                    dq.addLast(V[i]);
                }
            }
        }
        while (!dq.isEmpty()) {
            sum += dq.pollFirst();
        }
        System.out.println(sum);
    }
}
