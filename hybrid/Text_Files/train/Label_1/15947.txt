#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> ans(n * k);
  for (int i = 0, j = 0; i < n * k; ++i) {
    int x;
    cin >> x;
    if (i != x - 1)
      ans[(x - 1) / n] = j + 1;
    else {
      j += n;
      ans[(x - 1) / n] = i + 1;
    }
  }
  for (int i = 0; i < n * k; ++i) cout << ans[i] << ' ';
}
