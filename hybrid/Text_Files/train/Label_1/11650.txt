#include<iostream>
using namespace std;
bool poss(string s){
    int p=0,q=0;
    for(int i=0;i<s.size();i++){
        if(s[i]=='(')p++;
        else{
            p--;
            if(p<0){
                q--;
            }
        }
        if(s[i]=='?')q++;
    }
    if(p+q<=0)return 1;
    else return 0;
}
int main(){
    int n;
    cin>>n;
    while(n--){
        string s;
        cin>>s;
        if(poss(s))cout<<"YES\n";
        else cout<<"NO\n";
    }
}
