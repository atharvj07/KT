import java.util.*;

public class Main {
  static class Edge {
    int u, v;

    Edge(int u, int v) {
      this.u = u;
      this.v = v;
    }
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), m = in.nextInt();
    List<Edge> edges = new ArrayList<>();
    for (int i = 0; i < m; i++) {
      edges.add(new Edge(in.nextInt() - 1, in.nextInt() - 1));
    }
    int[] match = new int[n];
    Arrays.fill(match, -1);
    for (Edge edge : edges) {
      if (match[edge.u] == -1 && match[edge.v] == -1) {
        match[edge.u] = edge.v;
        match[edge.v] = edge.u;
      }
    }
    Set<Integer> set1 = new HashSet<>();
    Set<Integer> set2 = new HashSet<>();
    for (int i = 0; i < n; i++) {
      if (match[i] == -1) {
        set1.add(i);
      } else {
        set2.add(match[i]);
        set2.add(i);
      }
    }
    if (set1.size() == n || set2.size() == n) {
      System.out.println("-1");
    } else {
      System.out.println(set1.size());
      for (int i : set1) {
        System.out.print((i + 1) + " ");
      }
      System.out.println();
      System.out.println(set2.size());
      for (int i : set2) {
        System.out.print((i + 1) + " ");
      }
    }
  }
}
