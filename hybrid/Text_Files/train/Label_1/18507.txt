#include <bits/stdc++.h>
using namespace std;

int main() {
  int H1, M1, H2, M2;
  scanf("%d:%d\n%d:%d", &H1, &M1, &H2, &M2);
  int total = (H2 - H1) * 60 + (M2 - M1);
  int h = H1 + total / 120;
  int m = (M1 + (total % 120) / 2) % 60;
  if (m < 10) { printf("%d:0%d\n", h, m); } else { printf("%d:%d\n", h, m); }
  return 0;
}
