import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int N = Integer.parseInt(line[0]);
        int R = Integer.parseInt(line[1]);
        int L = Integer.parseInt(line[2]);

        List<Record> records = new ArrayList<>();
        for (int i = 0; i < R; i++) {
            line = br.readLine().split(" ");
            int d = Integer.parseInt(line[0]);
            int t = Integer.parseInt(line[1]);
            int x = Integer.parseInt(line[2]);
            records.add(new Record(d, t, x));
        }

        Collections.sort(records, Comparator.comparingInt(Record::getTime));

        int[] scores = new int[N + 1];
        int[] onTvTime = new int[N + 1];
        int currentTeam = 0;
        int currentScore = 0;
        int startTime = 0;

        for (Record record : records) {
            int team = record.getTeam();
            int time = record.getTime();
            int score = record.getScore();

            scores[team] += score;

            if (scores[team] > currentScore || (scores[team] == currentScore && team < currentTeam)) {
                onTvTime[currentTeam] += time - startTime;
                currentTeam = team;
                currentScore = scores[team];
                startTime = time;
            }
        }

        onTvTime[currentTeam] += L - startTime;

        int longestTeam = 0;
        int longestTime = 0;
        for (int i = 1; i <= N; i++) {
            if (onTvTime[i] > longestTime) {
                longestTeam = i;
                longestTime = onTvTime[i];
            }
        }

        System.out.println(longestTeam);
    }

    static class Record {
        private final int team;
        private final int time;
        private final int score;

        public Record(int team, int time, int score) {
            this.team = team;
            this.time = time;
            this.score = score;
        }

        public int getTeam() {
            return team;
        }

        public int getTime() {
            return time;
        }

        public int getScore() {
            return score;
        }
    }
}
