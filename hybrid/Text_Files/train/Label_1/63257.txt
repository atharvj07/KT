N = int(input())
A = list(map(int, input().split()))
D = {}
for i in range(N):
    for j in range(i + 1, N):
        v = A[i] + A[j]
        if v not in D:
            D[v] = [(i, j)]
        else:
            D[v].append((i, j))
ans = 'NO'
for k, v in D.items():
    if len(v) >= 2:
        for i, j in v:
            for x, y in v:
                if i != x and i != y and j != x and j != y:
                    ans = 'YES'
                    print(ans)
                    print(' '.join(map(str, [i + 1, j + 1, x + 1, y + 1])))
                    exit()
print(ans)
