import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long ans = a[0];
        for (int i = 1; i < n; i++) {
            if (i % 2 == 1) {
                ans += a[i];
            } else {
                ans -= a[i];
            }
        }
        System.out.println((ans % (1000000007L)) + "");
    }
}
