import sys

n, m = map(int, input().split())
adj = {}
tot = [0 for i in range(n+1)]

for i in range(m):
    a, b, c = map(int, input().split())
    tot[a] -= c
    tot[b] += c

    if a not in adj:
        adj[a] = []
    adj[a].append((b,-c))

    if b not in adj:
        adj[b] = []
    adj[b].append((a,c))

st = []
vis = []

for i in range(1, n+1):
    vis.append(0)

q = []
for i in range(1, n+1):
    if tot[i] < 0:
        q.append(i)
        vis[i] = 1

ans = 0

while len(q) > 0:
    cur = q[0]
    q.pop(0)
    for i in adj[cur]:
        ans -= i[1]
        tot[i[0]] += i[1]
        if vis[i[0]] == 0 and tot[i[0]] < 0:
            q.append(i[0])
            vis[i[0]] = 1

print(ans)
