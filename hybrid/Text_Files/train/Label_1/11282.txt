#include <bits/stdc++.h>

#define pb push_back
#define all(v) (v).begin(), (v).end()
using namespace std;
typedef long long Long;

#define rep(i, n) for (Long i = 0; i < (Long) (n); i++)
#define forn(i, a, b) for (Long i = (Long) (a); i < (Long) (b); i++)
const int N = (int) 1e5 + 7;
vector<Long> g[N];
bitset<N> vis;
Long ans = -1;
Long cnt = 0;

void dfs(Long u, Long p) {
  vis[u] = 1;
  cnt++;
  for (auto& v: g[u]) {
    if (vis[v]) continue;
    if (v == p) continue;
    dfs(v, u);
  }
}

int main() {
  Long n, m;
  cin >> n >> m;
  rep(i, m) {
    Long a, b;
    cin >> a >> b;
    a--, b--;
    g[a].pb(b);
    g[b].pb(a);
  }
  rep(i, n) {
    if (vis[i]) continue;
    cnt = 0;
    dfs(i, -1);
    ans = max(ans, cnt * (cnt - 1));
  }
  cout << ans << endl;
  return 0;
}
