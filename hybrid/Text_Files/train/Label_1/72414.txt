n, m = [int(i) for i in input().split()]
a = [[i for i in input()] for j in range(n)]
for x in range(n):
    for y in range(m):
        if a[x][y] == '.':
            ok = True
            for nx in range(n):
                if a[nx][y] == '*':
                    ok = False
                    break
            for ny in range(m):
                if a[x][ny] == '*':
                    ok = False
                    break
            if ok:
                print("YES")
                print(x + 1, y + 1)
                exit()
print("NO")
