n, m, q = map(int, input().split())
a = list(map(int, input().split()))
c = [0]*m
for x in a: c[x-1] += 1
for i in range(1, m): c[i] += c[i-1]
ans = []
for x in range(q):
    k = int(input())-1
    l, r = 0, m-1
    while l<=r:
        mid = l+r>>1
        if c[mid]==k+1: break
        elif c[mid]<k+1: l = mid+1
        else: r = mid-1
    ans.append(r+1)
print(*ans)
