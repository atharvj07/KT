import sys

n=int(input())
a=[0]*n
b=[0]*n
c=[0]*n
p=[0]*n
for i in range(n):
    a[i],b[i],c[i]=map(int,input().split())
for i in range(n-1):
    x,y=map(int,input().split())
    x-=1
    y-=1
    p[x]+=1
    p[y]+=1
    
def dfs(x,y,u):
    global ans
    if x>=0 and x<n:
        if b[x]!=c[x] and p[x]==u:
            ans+=a[x]
        for i in range(n):
            if p[i]==u+1 and x==y: dfs(i,x,u+1)
            elif p[i]==u+1 and x!=y: dfs(i,y,u+1)
    
ans=0
dfs(0,0,0)
if ans>1e18: ans=-1
print(ans)
