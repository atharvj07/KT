#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[100001];
ll n;
int main() {
    cin >> n;
    for (ll i = 0;i < n;i++) {
        cin >> a[i];
    }
    sort(a, a + n);
    ll ans = a[n - 1] - a[1];
    ll pos1 = lower_bound(a, a + n, a[1] + 1) - a;
    ll pos2 = upper_bound(a, a + n, a[1] - 1) - a;
    ll ans1 = a[n - 1] - a[pos1];
    ll ans2 = a[pos2] - a[1];
    pos1 = upper_bound(a, a + n, a[n - 1] - 1) - a;
    pos2 = lower_bound(a, a + n, a[n - 1] + 1) - a;
    ll ans3 = a[pos1] - a[1];
    ll ans4 = a[n - 1] - a[pos2];
    ans = min({ans, ans1, ans2, ans3, ans4});
    cout << ans << "\n";
    return 0;
}
