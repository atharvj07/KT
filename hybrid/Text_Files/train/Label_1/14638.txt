#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 1000000007
int main() {
    string s, t;
    cin >> s >> t;
    int n = s.size(), m = t.size();
    int fr0[n], fr1[n], lz = -1, lo = -1, cz = 0, co = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '0') {
            fr0[i] = fr0[i - 1] + 1;
            cz++;
            lz = i;
        }
        else {
            fr0[i] = fr0[i - 1];
        }
        if (s[i] == '1') {
            fr1[i] = fr1[i - 1] + 1;
            co++;
            lo = i;
        }
        else {
            fr1[i] = fr1[i - 1];
        }
    }
    if (cz == 0 || co == 0 || cz == n || co == n) {
        cout << 0;
        return 0;
    }
    int ans = 0;
    for (int i = 0; i < m; i++) {
        int z = i - fr0[lz] + cz;
        int o = m - z - co;
        if (z > 0 && o > 0 && t[i] != t[i - fr0[lz]] && t[i] != t[m - o])ans++;
    }
    cout << ans;
    return 0;
}
