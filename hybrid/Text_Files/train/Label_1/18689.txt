#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int fpow(int x, int p) {
    if (p == 0) return 1;
    if (p == 1) return x;
    int t = fpow(x, p / 2);
    t = (1LL * t * t) % mod;
    if (p % 2) t = (1LL * t * x) % mod;
    return t;
}
const int MAXN = 1e5 + 5;
vector<int> pref[MAXN], suf[MAXN];
int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string s, t;
    cin >> s >> t;
    for (int i = 0; i < s.size(); i++) {
        if (i > 0) {
            pref[i] = pref[i - 1];
            suf[i] = suf[i + 1];
        }
        if (s[i] == t[0]) pref[i].push_back(i);
        if (s[s.size() - 1 - i] == t[t.size() - 1]) suf[i].push_back(s.size() - 1 - i);
    }
    vector<int> dp(s.size() + 1, 0);
    dp[s.size()] = 1;
    for (int i = s.size() - 1; i >= 0; i--) {
        for (int j : pref[i]) {
            int rem = i + t.size() - 1 - (j + t.size() - 1);
            dp[i] = (dp[i] + 1LL * dp[j + t.size()] * suf[rem].size()) % mod;
        }
    }
    cout << dp[0] << '\n';
    return 0;
}
