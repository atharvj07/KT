#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int maxn = 2e5 + 5;
int a[maxn], L[maxn], R[maxn], dp[maxn][2];

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        for (int i = 0; i < n; i++) cin >> a[i];

        memset(dp, 0x3f, sizeof(dp));
        memset(L, 0, sizeof(L));
        memset(R, 0, sizeof(R));

        dp[n][0] = dp[n][1] = 0;
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 2; j++) {
                dp[i][j] = dp[i + 1][j];
                int pre = j ? R[i + 1] : L[i + 1];
                if (a[i] > pre) {
                    dp[i][j] = min(dp[i][j], dp[i + 1][!j] + j);
                    if (j == 0) {
                        L[i] = a[i];
                    } else {
                        R[i] = a[i];
                    }
                }
            }
        }
        cout << dp[0][0] << "\n";
    }
    return 0;
}
