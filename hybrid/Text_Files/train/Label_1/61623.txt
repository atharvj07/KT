n=int(input())
a=list(map(int,input().split()))
def f(l,r) -> int:
    ans=0
    opn=0
    for i in range(l,r+1):
        if i%2:
            if a[i]%2:opn+=a[i]
        else:
            if opn>=a[i]:
                opn-=a[i]
                ans+=1
            else:break
    return ans
ans=0
for i in range(n):
    for j in range(i,n):
        ans+=f(i,j)
print(ans)
