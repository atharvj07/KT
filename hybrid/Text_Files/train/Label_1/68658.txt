import collections

n = input()
vis = collections.defaultdict(lambda: 0)
ans = []
last = 0
cur = n[len(n) - 1]
for i in range(len(n) - 2, -1, -1):
    prv = n[i]
    last += 1
    if prv == '0':
        ans.append(0)
        continue
    if cur == '0':
        if i == 0:
            ans = [-1]
            break
        for j in range(i + 1, len(n)):
            if n[j] != '0':
                ans.append(j - i)
                cur = n[j]
                n = n[:i] + n[j] + n[i + 1:j] + n[i] + n[j + 1:]
                break
        else:
            ans = [-1]
            break
    if (int(prv) * 10 + int(cur)) % 25 == 0:
        if vis[i] != last:
            vis[i] = last
            ans.append(0)
    else:
        t = (25 - (int(prv) * 10 + int(cur)) % 25) % 25
        if vis[i] != last:
            vis[i] = last
            ans.append(t // int(prv))
        cur = str((int(cur) + t) % 10)
    assert(cur != '0')
if len(ans) > 0 and ans[0] == -1:
    print(-1)
else:
    print(len(ans))
