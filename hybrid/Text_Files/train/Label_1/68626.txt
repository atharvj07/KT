import sys
input = sys.stdin.readline

n, k = map(int, input().split())
temp = list(map(int, input().split()))
c, ans = 0, 0
need = 0
for x in temp:
    if x >= 0:
        if need:
            ans += 1
            c = 0
        c += 1
    else:
        if not need:
            ans += 1
        need = c
        c = 0
        if need > k:
            ans = -1
            break
if need:
    ans += 1
print(ans)
