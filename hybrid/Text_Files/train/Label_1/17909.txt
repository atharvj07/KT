#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (ll i = 0; i < n; ++i)
#define all(a) (a).begin(), (a).end()
using ll = long long;
const ll mod = 1000000007;
const long double EPS = 1e-15;
const long double PI = acosl(-1.0);

template <class T> inline bool chmin(T& a, T b) {if (a > b) {a = b; return true;} return false;}
template <class T> inline bool chmax(T& a, T b) {if (a < b) {a = b; return true;} return false;}

typedef pair<ll, ll> P;
struct UnionFind {
  vector<int> par, siz;
  UnionFind(int n) : par(n, -1), siz(n, 1) {}
  int root(int x) {
    if (par[x] == -1) return x;
    return par[x] = root(par[x]);
  }
  bool unite(int x, int y) {
    x = root(x), y = root(y);
    if (x == y) return false;
    if (siz[x] < siz[y]) swap(x, y);
    par[y] = x;
    siz[x] += siz[y];
    return true;
  }
  bool same(int x, int y) { return root(x) == root(y); }
  int size(int x) { return siz[root(x)]; }
};

int main() {
  cin.tie(0);
  ios::sync_with_stdio(false);
  ll n, m, q;
  cin >> n >> m >> q;
  vector<ll> a(n);
  rep(i, n) cin >> a[i];
  vector<ll> b(m);
  rep(i, m) cin >> b[i];
  UnionFind uf(n);
  multiset<ll> s;
  for (int i = 0; i < n; i++) s.insert(a[i]);
  while (q--) {
    int type;
    cin >> type;
    if (type == 1) {
      ll idx, x;
      cin >> idx >> x;
      idx--;
      uf.unite(idx, *s.begin());
      s.erase(s.begin());
      s.insert(x);
    } else {
      ll idx, x;
      cin >> idx >> x;
      idx--;
      if (uf.same(uf.root(idx), *s.begin()) && *s.begin() <= x) {
        s.erase(s.begin());
        if (!s.empty()) uf.unite(idx, *s.begin());
      }
      b[idx] = x;
    }
    if (s.empty())
      cout << -1 << endl;
    else
      cout << *s.begin() << endl;
  }
  return 0;
}
