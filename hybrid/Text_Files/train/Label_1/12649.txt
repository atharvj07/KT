#include<bits/stdc++.h>
using namespace std;
#define int long long

const int mod=1e9+7;
int b,a,cur,n,ans;
string s;

signed main(){
    cin>>s;
    n=s.size();
    for(int i=0;i<n;i++){
        if(s[i]=='b')b++,cur=0;
        else a++,cur++;
        ans=(ans+cur*b%mod)%mod;
    }
    cout<<(ans+mod)%mod;
}
