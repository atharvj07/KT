#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main(){
  string s, t;
  cin >> s >> t;
  int cnt = 0, trg = 0;
  for(int i = 0;i < s.size();i++){
    if(s[i] == t[i]) trg = i + 1;
  }
  for(int i = trg - 1;i < s.size();i++){
    if(s[i] == s[trg - 1]) cnt++;
  }
  cout << trg + cnt << '\n';
}
