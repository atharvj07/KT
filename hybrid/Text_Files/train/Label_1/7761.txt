#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<long long> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    vector<int> ans;
    for (int k = 1; k <= n; k++) {
        bool flag = true;
        for (int i = k + 1; i <= n; i++) {
            if (a[i] * k != a[i - k] * (k - 1)) {
                flag = false;
                break;
            }
        }
        if (flag) ans.push_back(k);
    }
    cout << ans.size() << endl;
    for (auto &x : ans) cout << x << ' ';
    return 0;
}
