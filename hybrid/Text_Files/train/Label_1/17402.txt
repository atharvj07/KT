#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main()
{
    int n;
    cin >> n;
    priority_queue<ll> pq;
    int c = 0, ans = 0;
    while (n--)
    {
        int x;
        cin >> x;
        if (x == -1)
        {
            if (pq.empty())
                c++;
            else
            {
                ll y = pq.top();
                pq.pop();
                if (-y > 1)
                    pq.push(y + 1);
            }
        }
        else
            pq.push(-(ll)x);
    }
    cout << c << endl;
    return 0;
}
