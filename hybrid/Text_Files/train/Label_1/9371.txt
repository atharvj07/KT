#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const int mod = 998244353;

ll power(ll a, ll b) {
    ll ans = 1;
    while (b > 0) {
        if (b % 2 == 1) ans = (ans * a) % mod;
        a = (a * a) % mod;
        b /= 2;
    }
    return ans;
}

int main() {
    int n, k;
    cin >> n >> k;
    cout << power(n, k) << endl;
    return 0;
}
