#include<bits/stdc++.h>
#define nl "\n"
using namespace std;

int main() {
    int tc; cin >> tc;
    while (tc--) {
        int n; cin >> n;
        priority_queue<int> p, c;
        while (n--) {
            int _p, _c; cin >> _p >> _c;
            p.push(_p); c.push(_c);
        }
        int f = 1;
        while (!p.empty() && !c.empty()) {
            if (p.top() == c.top()) {
                p.pop(); c.pop();
            }
            else if (p.top() > c.top()) {
                f = 0; break;
            }
            else { p.pop(); }
        }
        cout << ((f) ? "YES" : "NO") << nl;
    }
    return 0;
}
