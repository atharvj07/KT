import java.util.*;
import java.io.*;
import java.lang.*;

public class Main 
{
    public static final int mod = (int) (1e9 + 7);
    public static long pow(long a, long p) {
        long ans = 1;
        a = a % mod;
        while (p > 0) {
            if ((p & 1) > 0) {
                ans = (ans * a) % mod;
            }
            a = (a * a) % mod;
            p >>= 1;
        }
        return ans;
    }

    public static void main(String[] args) throws Exception {
        Scanner cin = new Scanner(System.in);
        long l1 = cin.nextLong();
        long r1 = cin.nextLong();
        long l2 = cin.nextLong();
        long r2 = cin.nextLong();

        long num = r1 - l1 + 1;
        long den = pow(2, num) - 1;
        den = pow(den, mod - 2);

        long len = r2 - l2 + 1;

        long ans = (num * len) % mod;
        ans = (ans * den) % mod;

        System.out.println(ans);
    }
}
