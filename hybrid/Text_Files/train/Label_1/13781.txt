#include <bits/stdc++.h>
using namespace std;
int n, m, a[55], b[55];
int main() {
    ios_base::sync_with_stdio(0), cin.tie(0);
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < m; i++) cin >> b[i];
    sort(a, a + n);
    sort(b, b + m);
    if (a[0] >= 0) {
        cout << a[0] * b[m - 1];
        return 0;
    }
    if (b[0] >= 0) {
        cout << a[n - 1] * b[0];
        return 0;
    }
    cout << a[0] * b[0];
    return 0;
}
