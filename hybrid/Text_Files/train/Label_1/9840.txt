#include <iostream>

using namespace std;

long long gcd(long long a, long long b) {
    if (b == 0) {
        return a;
    }
    return gcd(b, a % b);
}

int main() {
    long long a, b;
    cin >> a >> b;
    if (a % b == 0){
        cout << a / b;
    } else {
        long long g = gcd(a, b);
        a /= g;
        b /= g;
        long long ans = 0;
        while (a % b != 0) {
            a += g;
            ans++;
        }
        ans += a / b;
        cout << ans;
    }
    return 0;
}
