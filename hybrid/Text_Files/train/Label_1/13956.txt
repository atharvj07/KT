#include <iostream>
#include <vector>
#include <bitset>
#include <queue>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> ans;
  std::vector<std::bitset<1001>> graph(n + 1);
  for (int i = 1; i <= n; i++) {
    if (i + k <= n) graph[i][i + k] = 1;
    if (i - k >= 1) graph[i][i - k] = 1;
  }
  int cnt = 0;
  std::vector<bool> vis(n + 1, 0);
  std::queue<int> q;
  for (int i = 1; i <= n; i++) {
    if (vis[i]) continue;
    q.push(i);
    cnt++;
    while (!q.empty()) {
      int x = q.front();
      q.pop();
      if (vis[x]) continue;
      vis[x] = 1;
      for (int j = 1; j <= n; j++) {
        if (graph[x][j] && !vis[j]) q.push(j);
      }
    }
  }
  std::cout << cnt << "\n";
  for (int i = 1; i <= n; i++) {
    if (vis[i]) ans.push_back(i);
  }
  for (auto x : ans) std::cout << x << " ";
  std::cout << "\n";
  return 0;
}
