#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

const int MAXN = 1e5 + 10;

int n;
ll k;
vector<ll> a;

ll gcd(ll a, ll b) {
    while (b) {
        a %= b;
        swap(a, b);
    }
    return a;
}

int main() {
    cin >> n >> k;
    a.resize(n + 10);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }

    ll g = 0;
    for (int i = 1; i <= n; i++) { g = gcd(g, a[i]); }

    if (g != 1) { cout << (ll)n * (n + 1) / 2 << endl; } else
        cout << 0 << endl;

    return 0;
}
