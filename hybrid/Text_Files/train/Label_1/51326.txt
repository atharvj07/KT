import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int d = in.nextInt();
        int s = in.nextInt();
        in.close();

        int digit = s / 9;
        int mod = s % 9;
        long ans = 0;
        if (mod != 0) {
            ans = ans + (mod * (long)Math.pow(10, digit));
            digit++;
        }
        for (int i = 0; i < digit; i++) {
            ans = ans + (9 * (long)Math.pow(10, i));
        }
        if (ans % d == 0) {
            System.out.println(ans);
        } else {
            System.out.println(-1);
        }
    }
}
