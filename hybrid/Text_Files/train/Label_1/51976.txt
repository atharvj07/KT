import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long ans = -1;
        int[] min = new int[n];
        for (int i = 0; i < n; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            while (a % 2 == 0 && b % 2 == 0) {
                a /= 2;
                b /= 2;
            }
            while (a % 3 == 0 && b % 3 == 0) {
                a /= 3;
                b /= 3;
            }
            if (a * b != 0) min[i] = a * b;
            else min[i] = -1;
        }
        for (int i = 0; i < n; i++) {
            if (min[i] == -1) {
                ans = -1;
                break;
            }
            if (ans == -1) ans = min[i];
            else {
                while(ans % min[i] != 0) ans += min[i];
            }
        }
        System.out.println(ans % 1000000007);
    }
}
