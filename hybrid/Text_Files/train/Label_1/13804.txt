#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        deque<int> b(n);
        for (int &i : b) cin >> i;
        bool ok = true;
        deque<int> a;
        for (int i = 0; i < n; i++) {
            if (a.empty() || a.back() > b[i]) {
                a.push_back(b[i]);
            } else if (!a.empty() && a.front() > b[i]) {
                a.push_front(b[i]);
            } else {
                ok = false;
                break;
            }
        }
        if (!ok) {
            cout << "-1\n";
            continue;
        }
        int cur = 2 * n;
        while (!a.empty()) {
            cout << a.front() << " ";
            a.pop_front();
            if (!a.empty()) {
                cout << cur << " ";
                cur--;
            }
        }
        cout << "\n";
    }
}
