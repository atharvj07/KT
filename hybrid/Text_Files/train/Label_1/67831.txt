import sys
input = sys.stdin.readline

n = int(input())
x = list(map(int, input().split()))
v = list(map(int, input().split()))

ans = 0
for i in range(n):
    for j in range(i+1, n):
        if v[i] == v[j]:
            ans += abs(x[i] - x[j])
        elif v[i] > v[j]:
            delta = (x[i] - x[j]) / (v[j] - v[i])
            if 0 <= delta <= 1:
                ans += abs(x[i] - x[j] - v[j] * delta)
        else:
            delta = (x[j] - x[i]) / (v[i] - v[j])
            if 0 <= delta <= 1:
                ans += abs(x[j] - x[i] - v[i] * delta)
print(ans)
