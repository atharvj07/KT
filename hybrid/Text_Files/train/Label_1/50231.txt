import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n + 1];
        for (int i = 1; i <= n; i++)
            a[i] = in.nextLong();
        long[] dp = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            if (i == 1) dp[i] = a[i] / 2 + (a[i] + 1) / 2;
            else dp[i] = Math.min(dp[i - 1], dp[Math.max(0, i - 2)] + 1) + a[i] / 2 + (a[i] + 1) / 2;
        }
        for (int i = 1; i <= n - 1; i++)
            System.out.println(dp[i]);
    }
}
