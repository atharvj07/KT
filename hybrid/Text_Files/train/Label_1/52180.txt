import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[][] cards = new long[n][];
        for (int i = 0; i < n; i++) {
            int k = in.nextInt();
            cards[i] = new long[k * 2];
            for (int j = 0; j < k; j++) {
                cards[i][2 * j] = in.nextInt();
                cards[i][2 * j + 1] = in.nextInt();
            }
        }
        long ans = 0;
        for (long[] turn : cards) {
            Arrays.sort(turn);
            long cur = 0;
            int cnt = 0;
            for (int i = 0; i < turn.length; i += 2) {
                if (cur + turn[i] <= 3) {
                    cur += turn[i];
                    cnt++;
                    ans += turn[i + 1];
                } else {
                    break;
                }
            }
            if (cnt % 10 == 9) {
                ans += turn[turn.length - 1];
            }
        }
        System.out.println(ans);
    }
}
