#include<bits/stdc++.h>
#define pii pair<int,int>
#define mp make_pair
#define int long long int
using namespace std;
const int N=2e5+5;
const int M=2e5+5;
vector<int> g[N];
vector<int> sc(N);
int vis[N];
int d[N];
bool cmp(int a,int b){
    return sc[a]>sc[b];
}
void dfs(int u,int p){
    vis[u]=true;
    for(auto x:g[u]){
        if(x==p or vis[x])continue;
        d[x]=d[u]+1;
        dfs(x,u);
    }
}
int32_t main(){
    int n,m;
    cin>>n>>m;
    for(int i=1;i<=n;i++)cin>>sc[i];
    for(int i=1;i<=m;i++){
        int u,v;
        cin>>u>>v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    int s;
    cin>>s;
    dfs(s,-1);
    vector<int> v;
    for(int i=1;i<=n;i++){
        if(vis[i])v.push_back(i);
    }
    sort(v.begin(),v.end(),cmp);
    int ans=0;
    for(int i=0;i<v.size();i++){
        if(!vis[v[i]])continue;
        ans+=sc[v[i]];
        int x=v[i];
        while(d[x]>1){
            vis[x]=false;
            x=d[x]-1;
        }
    }
    cout<<ans<<endl;
}
