#include <stdio.h>
#include <algorithm>
#include <vector>

using namespace std;

const int maxn = 2e5 + 5;
int a[maxn];
int main() {
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) { scanf("%d", &a[i]); }
    sort(a, a + n);
    int idx = 0;
    while (idx < n && a[idx] == a[0]) { idx++; }
    int l = max(idx, n / 2 - (n - idx + 1) / 2), r = n - max(idx, n / 2 - (n - idx) / 2);
    if (n - idx - (r - l) > 0) {
        idx += n - idx - (r - l);
        while (idx < n && a[idx] == a[0]) { idx++; }
    }
    printf("%d\n", idx);
    for (int i = 0; i < idx; i++) { printf("%d ", a[i]); }
    printf("\n");
    return 0;
}
