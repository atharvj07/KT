#include<bits/stdc++.h>
using namespace std;
int s[1005],f[26],ans;
int main()
{
    int n,k,i,j;
    string str;
    cin>>str>>k;
    for(i=0;i<26;i++)   cin>>f[i];
    n=str.length();
    for(i=0;i<n;i++)
    {
        s[i+1]=s[i]+f[str[i]-'a'];
    }
    sort(f,f+26);
    for(i=n+1;i<=n+k;i++)
    {
        ans+=f[25]*(i);
    }
    cout<<ans+s[n]<<endl;
    return 0;
}
