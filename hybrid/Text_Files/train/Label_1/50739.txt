import java.util.*;

public class Main {
    static long inf = Long.MAX_VALUE / 10;
    static long[][] graph;
    static int[] a, b;
    static int v, n, e;

    static boolean solve() {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) if (graph[i][j] > 0) {
                graph[i][i] -= graph[i][j];
                graph[j][j] -= graph[i][j];
                graph[i][j] += graph[j][i];
                graph[j][i] = 0;
            }
            if (graph[i][i] < 0) return false;
        }
        for (int i = 0; i < n; i++) {
            b[i] -= a[i];
            for (int j = 0; j < n; j++) if (graph[i][j] > 0) {
                long need = Math.min(b[i], graph[i][i]);
                b[i] -= need;
                b[j] += need;
                graph[i][i] -= need;
                graph[i][j] -= need;
                graph[j][j] += need;
            }
        }
        for (int i = 0; i < n; i++) if (b[i] != 0) return false;
        return true;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        v = in.nextInt();
        e = in.nextInt();
        a = new int[n];
        b = new int[n];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();
        for (int i = 0; i < n; i++) b[i] = in.nextInt();
        graph = new long[n][n];
        for (int i = 0; i < e; i++) {
            int x = in.nextInt() - 1;
            int y = in.nextInt() - 1;
            graph[x][y]++;
        }
        if (solve()) {
            long ans = 0;
            for (int i = 0; i < n; i++)
                for (int j = 0; j < n; j++)
                    if (graph[i][j] > 0) ans += graph[i][j];
            long cnt = ans / 2;
            System.out.println(ans / 2);
            for (int i = 0; i < n; i++)
                for (int j = 0; j < n; j++)
                    if (graph[i][j] > 0)
                        for (long k = 0; k < graph[i][j] / 2; k++) {
                            System.out.println((i + 1) + " " + (j + 1) + " " + 1);
                            cnt--;
                            if (cnt == 0) return ;
                        }
        } else System.out.println("NO");
    }
}
