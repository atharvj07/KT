#include <bits/stdc++.h>
#define ll long long
#define mod 1000000007
using namespace std;

ll f(ll A, ll B, ll n, ll x) {
    if (n == 0) return x;
    if (n == 1) return (A * x + B) % mod;
    ll x1 = f(A, B, n / 2, x);
    ll x2 = (A * x1 + B) % mod;
    if (n % 2) return x2;
    else return x1;
}

int main() {
    ll A, B, n, x;
    cin >> A >> B >> n >> x;
    cout << f(A, B, n, x) << endl;
}
