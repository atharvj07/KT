#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const ll mod = 1e9 + 7;

ll qpow(ll x, ll p) {
  if (p == 0)
    return 1;
  if (p == 1)
    return x;
  ll ans = qpow(x, p / 2);
  ans = (ans * ans) % mod;
  if (p % 2 == 1)
    ans = (ans * x) % mod;
  return ans;
}

int main() {
  int n;
  cin >> n;
  vector<int> a(n + 1);
  vector<int> deg(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    if (a[i] != -1)
      deg[a[i]]++;
  }
  ll ans = 0;
  for (int i = 1; i <= n; i++) {
    if (a[i] == -1 && deg[i] == 0)
      ans++;
  }
  ans = (ans * ((ans - 1) % mod)) % mod;
  ll inv2 = qpow(2, mod - 2);
  ans = (ans * inv2) % mod;
  cout << ans << endl;
  return 0;
}
