#include <iostream>
using namespace std;
typedef long long ll;
ll gcd(ll a, ll b) { return b ? gcd(b, a % b) : a; }
int main() {
  ll a1, b1, a2, b2, a, b;
  cin >> a1 >> b1 >> a2 >> b2;
  a = a1 * b1;
  b = a2 * b2;
  if (a % b) {
    cout << -1 << endl;
    return 0;
  }
  ll n = (a / b), ans = 0;
  while (n > 1) {
    if (n % 2 == 0) {
      n /= 2;
      ans += b1;
    } else {
      if (n % 3 == 0) {
        n /= 3;
        ans += a1;
      } else {
        cout << -1 << endl;
        return 0;
      }
    }
  }
  cout << ans << endl;
  cout << (a1 * ans) / a << " " << (b1 * ans) / a << endl;
  cout << (a2 * ans) / a << " " << (b2 * ans) / a << endl;
  return 0;
}
