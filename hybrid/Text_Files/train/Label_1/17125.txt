#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, d;
  cin >> n >> d;
  vector<int> hotels(n);
  for (int i = 0; i < n; i++) cin >> hotels[i];
  sort(hotels.begin(), hotels.end());

  int ans = 0;
  for (int i = 0; i < n; i++) {
    int start = hotels[i] - d, end = hotels[i] + d;
    int l = lower_bound(hotels.begin(), hotels.end(), start) - hotels.begin();
    int r = upper_bound(hotels.begin(), hotels.end(), end) - hotels.begin();
    ans += r - l - (i >= l && i < r);
  }
  cout << ans << endl;
  return 0;
}
