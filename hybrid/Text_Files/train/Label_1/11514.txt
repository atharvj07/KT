#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  int l = 0, r = n - 1, ans = 0;
  while (l < r) {
    if (a[l] == a[r]) {
      l = l + 1;
    } else {
      ans = max(ans, r - l);
      r = r - 1;
    }
  }
  cout << ans << endl;
  return 0;
}
