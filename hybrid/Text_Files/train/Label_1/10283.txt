#include <bits/stdc++.h>
using namespace std;
const int mod=1e9+7;
const int maxn=2e5+5;
int main(){
    int n,a[maxn];
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    sort(a+1,a+1+n,greater<int>());
    long long ans=0;
    for(int i=1;i<=n;i++){
        if(i+1<=n){
            a[i]-=a[i+1];
            if(a[i]<=0)break;
            ans+=a[i];
        }
        else{
            ans+=a[i];
        }
    }
    cout<<ans<<endl;
    return 0;
}
