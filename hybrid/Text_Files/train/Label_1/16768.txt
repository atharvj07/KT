#include <bits/stdc++.h>
using namespace std;
int main() {
  int l, r;
  cin >> l >> r;
  if (l == r) {
    cout << "-1";
    return 0;
  }
  l++;
  while (true) {
    bool ok = true;
    stringstream ss;
    ss << l;
    string s;
    ss >> s;
    sort(s.begin(), s.end());
    for (int i = 0; i < s.length() - 1; i++) {
      if (s[i] == s[i + 1]) {
        ok = false;
        break;
      }
    }
    if (ok) {
      cout << l;
      return 0;
    }
    l++;
    if (l > r) {
      cout << "-1";
      return 0;
    }
  }
}
