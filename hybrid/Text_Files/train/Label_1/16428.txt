#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, d;
        cin >> n >> d;
        vector<int> a(n);
        for (int &x : a) cin >> x;
        sort(a.begin(), a.end(), greater<int>());
        int ans = a[0];
        for (int i = 1; i < n && d > 0; i++) {
            if (a[i] < a[i - 1]) break;
            ans += min(d, a[i]), d -= min(d, a[i]);
        }
        cout << ans << endl;
    }
    return 0;
}
