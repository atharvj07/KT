import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] time = new int[n][2];
        for (int i = 0; i < n; i++) {
            time[i][0] = sc.nextInt();
            time[i][1] = sc.nextInt();
        }
        Arrays.sort(time, (a, b) -> a[1] - b[1]);
        int ans = 0;
        int[] pos = new int[n];
        for (int i = 0; i < n; i++) {
            int j = 0;
            while (j < i && time[j][1] <= time[i][0]) j++;
            if (j == i) {
                ans++;
                pos[ans - 1] = i + 1;
            }
        }
        System.out.println(ans);
        for (int i = 0; i < ans; i++) System.out.print(pos[i] + " ");
        sc.close();
    }
}
