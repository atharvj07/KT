import sys
input = sys.stdin.readline

n = int(input().strip())
cur, ans = 1900, 1900
for i in range(n):
    c, d = map(int, input().split())
    if d == 1:
        cur += c
    else:
        ans = max(ans, max(0, cur + c))
if ans == 1900: print('Impossible')
else: print('Infinity' if ans > 1900 else ans)
