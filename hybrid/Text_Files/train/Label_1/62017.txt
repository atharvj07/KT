from collections import Counter
from math import factorial

def count_variants(n):
    digits = Counter(str(n))
    total_variants = factorial(sum(digits.values()))
    for count in digits.values():
        total_variants //= factorial(count)
    leading_zero_variants = 0
    if '0' in digits:
        digits['0'] -= 1
        leading_zero_variants = factorial(sum(digits.values()) - 1)
        for count in digits.values():
            leading_zero_variants //= factorial(count)
        digits['0'] += 1
    return total_variants - leading_zero_variants

n = int(input().strip())
print(count_variants(n))
