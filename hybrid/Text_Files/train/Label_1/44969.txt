import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = input.nextInt();
        }
        int[] dist = new int[n];
        for (int i = 0; i < n; i++) {
            dist[i] = n;
            for (int j = 0; j < n; j++) {
                if (a[j] == b[i]) {
                    dist[i] = Math.min(dist[i], Math.abs(i - j));
                }
            }
        }
        for (int d : dist) {
            System.out.println(d);
        }
    }
}
