import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int countRed = 0;
        for (char c : s.toCharArray()) {
            if (c == 'R') {
                countRed++;
            }
        }
        System.out.println(countRed > n - countRed ? "Yes" : "No");
    }
}
