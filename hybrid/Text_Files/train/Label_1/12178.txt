#include <cstdio>
#include <cstring>
#define MAXN 17
#define MOD 1000000007

int t, n, dp[MAXN][MAXN][MAXN][MAXN];
typedef long long ll;

int func(int i, int ls, int cnt1, int rem) {
    if (rem == 0) return cnt1 == t;
    if (dp[i][ls][cnt1][rem] != -1) return dp[i][ls][cnt1][rem];

    ll ans = 0;
    if (i < n) {
        if (ls != i + 1) ans += func(i + 1, i + 1, cnt1 + 1, rem - 1);
        ans += func(i + 1, ls, cnt1, rem);
    }

    return dp[i][ls][cnt1][rem] = ans % MOD;
}

int main() {
    memset(dp, -1, sizeof dp);
    scanf("%d%d", &n, &t);
    int ans = func(0, 0, 0, t);
    printf("%d\n", ans);
    return 0;
}
