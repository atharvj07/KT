#include <bits/stdc++.h>
#define int long long
using namespace std;
const int MAX = 2e5 + 5;

int32_t main() {
    int n, k;
    cin >> n >> k;
    vector<int> cnt(MAX, 0);
    for (int i = 0; i < n; ++i) {
        int x;
        cin >> x;
        cnt[x]++;
    }
    vector<int> ans;
    for (int i = 0; i < MAX; ++i) {
        while (cnt[i] > 0 && ans.size() < k) {
            ans.push_back(i);
            for (int j = 0; j < MAX; ++j) cnt[j] -= cnt[i];
        }
    }
    for (int i = 0; i < ans.size(); ++i) {
        cout << ans[i] << " ";
    }
    return 0;
}
