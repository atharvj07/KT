import sys
input = sys.stdin.readline

n = int(input())
a = list(range(1, n + 1))

stack = []
inc = 0
dec = n
for i in range(n):
    while stack and stack[-1] > a[i]:
        dec = min(dec, i - stack[-1])
        stack.pop()
    while stack and stack[-1] < a[i]:
        inc = max(inc, i - stack[-1])
        stack.pop()
    stack.append(a[i])
print(*a)
