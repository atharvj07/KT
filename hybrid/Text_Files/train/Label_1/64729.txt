from math import sqrt

def find_time(x1, y1, x2, y2, vmax, t, vx, vy, wx, wy):
    def time_to_reach(dx, dy, v, t):
        # Solving quadratic equation ax^2 + bx + c = 0
        a = v * v - (wx * wx + wy * wy)
        b = 2 * (dx * wx + dy * wy - v * t * sqrt(v * v - vx * vx - vy * vy))
        c = dx * dx + dy * dy - 2 * t * (dx * vx + dy * vy) + t * t * (vx * vx + vy * vy)
        D = b * b - 4 * a * c
        return (-b + sqrt(D)) / (2 * a) if D >= 0 else float('inf')

    dx, dy = x2 - x1, y2 - y1
    # Without wind change
    time_direct = sqrt(dx * dx + dy * dy) / vmax
    # With wind, before and after the change
    time_with_wind = time_to_reach(dx, dy, vmax, t)
    return min(time_direct, time_with_wind)

x1, y1, x2, y2 = map(int, input().split())
vmax, t = map(int, input().split())
vx, vy = map(int, input().split())
wx, wy = map(int, input().split())

print(find_time(x1, y1, x2, y2, vmax, t, vx, vy, wx, wy))
