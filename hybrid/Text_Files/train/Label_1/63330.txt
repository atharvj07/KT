import sys

inp = lambda: sys.stdin.readline()
n = int(inp())
a = list(map(int, inp().split()))
if sum(a) % n != 0:
    print('NO')
else:
    x = sum(a) // n
    print('YES')
    for i in range(n):
        if a[i] < x:
            print(i + 1, i + 1, i + 2)
            a[i + 1] ^= x
        else:
            print(i + 1, i + 2, i + 2)
            a[i + 2] ^= x
