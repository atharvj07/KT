import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class BerlandRoads {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] roads = new int[n - 1];
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }

        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            roads[i] = u;
            graph.get(u).add(v);
            graph.get(v).add(u);
        }

        int k = scanner.nextInt();
        int[] fools = new int[2 * k];
        for (int i = 0; i < 2 * k; i++) {
            fools[i] = scanner.nextInt() - 1;
        }

        int[] visited = new int[n];
        for (int i = 0; i < 2 * k; i += 2) {
            int start = fools[i];
            int end = fools[i + 1];
            dfs(graph, visited, start, end);
        }

        for (int i = 0; i < n - 1; i++) {
            System.out.print(visited[roads[i]] + " ");
        }
    }

    private static void dfs(List<List<Integer>> graph, int[] visited, int start, int end) {
        visited[start]++;
        if (start == end) {
            return;
        }
        for (int neighbor : graph.get(start)) {
            if (visited[neighbor] == 0) {
                dfs(graph, visited, neighbor, end);
            }
        }
    }
}
