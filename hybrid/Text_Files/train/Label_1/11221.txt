#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e6 + 5;
int d[N], a[1005];
vector<int> b;
int main() {
    int n;
    cin >> n;
    for(int i = 0; i < n; i++) cin >> a[i];
    d[1] = 1;
    for(int i = 2; i < N; i++) {
        if(d[i] == 0) {
            for(int j = i; j < N; j += i) d[j] = i;
        }
    }
    int mx = 0;
    for(int i = 0; i < n; i++) {
        set<int> s;
        for(int x = a[i]; x > 1; x /= d[x]) s.insert(d[x]);
        int f = 1;
        for(set<int>::iterator it = s.begin(); it != s.end(); it++) {
            if(s.count(*it * *it)) f = 0;
        }
        if(f) {
            b.push_back(a[i]);
            mx++;
        }
    }
    cout << mx << endl;
    for(int i = 0; i < mx; i++) cout << b[i] << " ";
    cout << endl;
}
