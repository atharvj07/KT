#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n, m, ans = 0;
    cin >> n >> m;
    vector<pair<int, int>> v(n);
    map<int, priority_queue<int> > a;
    for (int i = 0; i < n; ++i) {
        cin >> v[i].first >> v[i].second;
        a[v[i].first].push(v[i].second);
    }
    for (int i = 1; i <= m; ++i) {
        int cnt = 0, sum = 0;
        while (!a[i].empty()) {
            ++cnt;
            sum += a[i].top();
            a[i].pop();
        }
        if (cnt > 0) ans += (sum / cnt) * cnt;
    }
    cout << ans << '\n';
}
