import sys
from collections import defaultdict

input = sys.stdin.readline
n, m = map(int, input().split())
adj = defaultdict(list)
an = [0] * (n + 1)
for _ in range(m):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)
q = int(input())
ans = [0] * (q + 1)
pos = [0] * (n + 1)

for i in range(n):
    an[i + 1] = i + 1
    pos[i + 1] = i + 1

for i in range(q):
    v = int(input())
    val = n + i + 1
    ans[i + 1] = ans[i]
    for j in adj[v]:
        if pos[j] < pos[v] and an[pos[j]: pos[v]].count(pos[v]) > 0:
            ans[i + 1] += 1
    for j in range(pos[v], n + 1):
        if an[j] == v:
            an[j] = val
    pos[v] = n + 1

print(*ans)
