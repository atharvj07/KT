import sys

input = sys.stdin

t = int(input.readline())
for i in range(t):
    input.readline()
    m, k = map(int, input.readline().split())
    a = list(map(int, input.readline().split()))
    b = [True] * (k+1)
    for j in range(m-1):
        dish, dis = map(int, input.readline().split())
        if dish != 0:
            if dis == 0:
                a[dish] -= 1
            b[dish] = False
    print(''.join(['Y' if x  <=0 else 'N' for x in a]))
