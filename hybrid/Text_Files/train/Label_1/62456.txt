for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    x = list(map(int, input().split()))
    sm = 0
    ans = []
    f = 0
    for i in range(n):
        sm += a[i]
        for j in range(m):
            if sm >= x[j]:
                ans.append(i + 1)
                f = 1
                break
        if f == 1:
            break
    if f == 0:
        ans.extend([-1 for _ in range(m)])
    print(*ans)
