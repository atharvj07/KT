const ll mod1 = 1e9 + 7;
const ll mod2 = 998244353;
const ll inf = 1e18;
const ld eps = 1e-9;
const ll nax = 1e5 + 5;
ll poww(ll a, ll p){
  ll ret = 1;
  a %= mod1;
  while(p > 0){
    if(p&1) ret = ret * a % mod1;
    a = a * a % mod1;
    p >>= 1;
  }
  return ret;
}
ll add(ll x, ll y){
  x += y;
  if(x >= mod1) x -= mod1;
  if(x < 0) x += mod1;
  return x;
}
ll sub(ll x, ll y){
  x -= y;
  if(x >= mod1) x -= mod1;
  if(x < 0) x += mod1;
  return x;
}
ll mul(ll x, ll y){
  return (x * 1ll * y) % mod1;
}
ll divi(ll x, ll y){
  return mul(x, poww(y, mod1 - 2));
}
ll inv(ll x){
  return poww(x, mod1 - 2);
}
ll fact(ll x){
  ll ret = 1;
  for(ll i = 1; i <= x; i+=1){
    ret = mul(ret, i);
  }
  return ret;
}
ll ncr(ll n, ll r){
  return mul(fact(n), divi(mul(fact(r), fact(n - r)), 1));
}
ll npr(ll n, ll r){
  return mul(fact(n), divi(fact(n - r), 1));
}
const ll SIZE = 1e5 + 5;
ll bit[SIZE];
ll get(ll idx){
  ll sm = 0;
  while(idx > 0){
    sm = add(sm, bit[idx]);
    idx -= (idx & -idx);
  }
  return sm;
}
ll get(ll l, ll r){
  return sub(get(r), get(l - 1));
}
void update(ll idx, ll val){
  while(idx < SIZE){
    bit[idx] = add(bit[idx], val);
    idx += (idx & -idx);
  }
}
int main(){
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ll n;
  cin >> n;
  ll pref[n + 5];
  cin >> pref[0];
  fora(i, 1, n - 1){
    ll x;
    cin >> x;
    pref[i] = pref[i - 1] ^ x;
  }
  update(1, pref[0]);
  fora(i, 1, n - 1){
    ll x = pref[i - 1];
    ll y = pref[i];
    ll ans = 0;
    ll sm = 0;
    update(i + 1, pref[i]);
    ll mid = (i + i + 1) / 2;
    if((i + i + 1) % 2 == 0){
      if(i >= mid){
        sm = get(i + 1);
      }
      if(i > mid){
        sm = add(sm, get(mid));
      }
      if(sm == y - x){
        ans+=1;
      }
    }
    cout << ans << "\n";
  }

  return 0;
}
