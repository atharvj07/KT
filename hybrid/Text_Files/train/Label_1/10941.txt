#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ld long double
#define endl "\n"

struct node {
  int prev, next;
  node() : prev(0), next(0) {}
};
node List[101];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> List[i].prev >> List[i].next;
  }
  int tail = 0;
  for (int i = 1; i <= n; i++) {
    if (List[i].prev == 0) {
      if (tail != 0) List[tail].next = i;
      while (List[i].next != 0) i = List[i].next;
      tail = i;
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << List[i].prev << " " << List[i].next << endl;
  }
  return 0;
}
