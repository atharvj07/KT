import sys

n, p = map(int, input().split())
a, b = [0] * n, [0] * n
for i in range(n):
    a[i], b[i] = map(int, input().split())
res = 0
time = [0] * n
for i in range(n):
    if a[i] > p:
        print(0)
        sys.exit()
    if a[i] != 0:
        time[i] = (b[i] + a[i] - 1) // a[i]
    else:
        time[i] = sys.maxsize
    res = max(res, time[i])
if res == sys.maxsize:
    print(-1)
else:
    print(res)
