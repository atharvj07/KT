#include <bits/stdc++.h>
using namespace std;
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        bool ok = true;
        for (int i = 0; i < n - 1; i++) {
            if (a[i] > a[i + 1]) {
                if (i + x >= n) {
                    ok = false;
                }
            }
        }
        cout << (ok ? "YES" : "NO") << "\n";
    }
}
