#include<bits/stdc++.h>
using namespace std;
int n,x;
int a[20];
int main(){
    cin>>n>>x;
    if(x==1){
        return cout<<0<<endl,0;
    }
    if(x==3){
        for(int i=1;i<=3;i++){
            a[i]=i+2;
        }
    }
    else{
        for(int i=1;i<=4;i++){
            a[i]=i*2;
        }
    }
    cout<<4<<endl;
    for(int i=1;i<=4;i++){
        cout<<a[i]<<" ";
    }
    return 0;
}
