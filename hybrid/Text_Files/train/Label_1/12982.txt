#include <bits/stdc++.h>

using namespace std;

#define ll long long

const ll mod = 1e9+7;

ll f(ll m, ll a, ll b){
    if(a > b)
        swap(a, b);
    ll ans = (m/a + 1) * (m/b + 1);
    ans %= mod;
    return ans;
}

int main(){
    ll x,y,z;
    cin >> x >> y >> z;
    cout << f(x, y, z);
    return 0;
}
