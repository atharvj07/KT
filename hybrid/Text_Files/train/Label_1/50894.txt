import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    static final int inf = Integer.MAX_VALUE;
    static int h, w;
    static int[][] c;
    static int[][] a;
    static int[][] dp;
    static int[] dx = {0, 1, 0, -1};
    static int[] dy = {1, 0, -1, 0};

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        String[] s1 = br.readLine().split(" ");
        h = Integer.parseInt(s1[0]);
        w = Integer.parseInt(s1[1]);
        c = new int[10][10];
        for (int i = 0; i < 10; i++) {
            String[] s2 = br.readLine().split(" ");
            for (int j = 0; j < 10; j++) {
                c[i][j] = Integer.parseInt(s2[j]);
            }
        }
        a = new int[h + 2][w + 2];
        dp = new int[h + 2][w + 2];
        for (int[] row : dp) {
            Arrays.fill(row, inf);
        }
        for (int i = 1; i <= h; i++) {
            String[] s3 = br.readLine().split(" ");
            for (int j = 1; j <= w; j++) {
                a[i][j] = Integer.parseInt(s3[j - 1]);
            }
        }

        dp[1][1] = 0;
        for (int i = 1; i <= h; i++) {
            for (int j = 1; j <= w; j++) {
                if (a[i][j] == -1) {
                    continue;
                }
                for (int k = 0; k < 4; k++) {
                    int nx = i + dx[k];
                    int ny = j + dy[k];
                    if (a[nx][ny] == -1) {
                        continue;
                    }
                    dp[nx][ny] = Math.min(dp[nx][ny], dp[i][j] + c[a[i][j]][a[nx][ny]]);
                }
            }
        }

        int ans = dp[h][w];
        if (ans == inf) {
            ans = -1;
        }
        System.out.println(ans);
    }
}
