import sys

n, k, d = map(int, input().split())
arr = list(map(int, input().split())) + [0]

# build prefix sum of platform
pref = [0]
for x in arr: pref.append(pref[-1] + x)

# dp[i][j] = True if can reach cell i with platform j
dp = [[False] * (k + 1) for _ in range(n + 2)]
dp[0][0] = True

# iterate over each cell
for i in range(1, n + 1):
    # iterate over each platform
    for j in range(k + 1):
        # try to jump from previous cell
        if i > d:
            dp[i][j] |= dp[i - 1][j]
        # try to jump from previous platform
        if j > 0 and i > pref[j]:
            dp[i][j] |= dp[i - pref[j]][j - 1]

# if cannot reach last cell print NO
if not dp[n][k]:
    print('NO')
    sys.exit()

# print answer
print('YES')
res = []
i = n
j = k
while i > 0:
    # find the last platform that can reach cell i
    while j > 0 and not dp[i - pref[j]][j - 1]:
        j -= 1
    # add last platform to answer
    res.append(j)
    # move to the previous platform
    i -= pref[j]
    j -= 1

res.reverse()
print(*res)
