import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Integer[] a = new Integer[n], b = new Integer[n];
        for (int i = 0; i < n; i++)
            a[i] = scanner.nextInt();
        for (int i = 0; i < n; i++)
            b[i] = scanner.nextInt();
        Arrays.sort(a, Comparator.naturalOrder());
        Arrays.sort(b, Comparator.naturalOrder());
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++)
                if (a[i].equals(b[j])) {
                    p[i] = j + 1;
                    break;
                }
        }
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (p[i] < p[i - 1]) {
                System.out.println(++ans + " " + p[i] + " " + (p[i - 1]));
                p[i] ^= p[i - 1] ^= p[i] ^= p[i - 1];
            }
        }
        System.out.println(ans);
    }
}
