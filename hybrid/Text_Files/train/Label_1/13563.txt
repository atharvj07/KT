#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int q, x;
  cin >> q >> x;
  vector<int> a;
  int mex = 0;
  while (q--) {
    int y;
    cin >> y;
    a.push_back(y);
    while (mex < (int)a.size() && count(a.begin(), a.end(), mex) > 0) { mex++; }
    cout << mex << endl;
  }
  return 0;
}
