n = int(input())
a = list(map(int, input().split()))

# Initialize the answer
answer = 0

# Precompute the prefix sums of the number of occurrences of each number
occurrences = [0] * (n+1)
prefix_sum = [0] * (n+1)

for i in range(n):
    occurrences[a[i]] += 1

for i in range(1, n+1):
    prefix_sum[i] = prefix_sum[i-1] + occurrences[i]

# Calculate the sum of f(l, r) for all l and r
for l in range(1, n+1):
    for r in range(l, n+1):
        # Calculate the number of components for the current l and r
        components = 0
        prev = 0
        for i in range(l, r+1):
            if occurrences[i] == 0:
                components += max(0, prefix_sum[i-1] - prefix_sum[prev] - 1)
                prev = i
        components += max(0, prefix_sum[r] - prefix_sum[prev] - 1)
        answer += 1 + components

print(answer)
