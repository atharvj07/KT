#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int MAXN = 2e5 + 10;
ll n, a[MAXN], b[MAXN], ans[MAXN];
struct A {
    ll p, x;
} aa[MAXN];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    int q;
    cin >> q;
    for (int i = 1; i <= q; i++) {
        cin >> aa[i].p >> aa[i].x;
        if (aa[i].p) b[aa[i].p] = aa[i].x;
    }
    for (int i = 1; i <= n; i++) {
        ans[i] = a[i];
        if (!b[i]) continue;
        if (b[i] > ans[i]) ans[i] = b[i];
    }
    for (int i = 1; i <= q; i++) {
        if (!aa[i].p) {
            for (int j = 1; j <= n; j++) {
                if (ans[j] < aa[i].x) ans[j] = aa[i].x;
            }
        }
    }
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    cout << endl;
    return 0;
}
