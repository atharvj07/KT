#include<bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int n;
char s[maxn];
int main() {
    cin >> n;
    cin >> s;
    int r = -1;
    for (int i = 0; i < n; i++) {
        if (i + 1 < n && s[i] > s[i + 1]) {
            r = i;
            break;
        }
    }
    for (int i = 0; i < n; i++) {
        if (r != i && r != i - 1) { cout << s[i]; }
    }
    return 0;
}
