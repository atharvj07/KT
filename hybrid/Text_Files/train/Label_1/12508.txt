#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;

int main() {
    ll t, n, x, a, sum;
    cin >> t;
    for (ll p = 1; p <= t; p++) {
        cin >> n >> x;
        sum = 0;
        vector<ll> v(n);
        for (ll i = 0; i < n; i++) {
            cin >> v[i];
        }
        for (ll i = 0; i < v.size(); i++) {
            a = v[i];
            if (a % x == 0) {
                while (a > 0) {
                    v.push_back(a / x);
                    a = a / x;
                }
            } else {
                break;
            }
        }
        for (auto u : v) {
            sum += u;
        }
        cout << sum << endl;
    }
    return 0;
}
