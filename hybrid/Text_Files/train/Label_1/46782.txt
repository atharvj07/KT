import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        Set<String> users = new HashSet<>();
        int sum = 0;
        for(String cmd; (cmd = br.readLine()) != null; ) {
            if(cmd.startsWith("+")) {
                users.add(cmd.substring(1));
            } else if(cmd.startsWith("-")) {
                users.remove(cmd.substring(1));
            } else {
                char[] arr = cmd.toCharArray();
                int n = 0;
                for(char c: arr) {
                    if(!Character.isLetter(c)) ++n;
                }
                sum += n * users.size();
            }
        }
        System.out.println(sum);
    }
}
