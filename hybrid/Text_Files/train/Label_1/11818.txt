#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, d, u, v, x, p[1005], ans;
    cin >> n >> d;
    for (int i = 1; i <= n; i++) p[i] = i;
    while (d--) {
        cin >> u >> v;
        x = p[u];
        while (p[x] != x) x = p[x];
        p[u] = x;
        x = p[v];
        while (p[x] != x) x = p[x];
        p[v] = x;
        if (x != u) p[x] = u;
    }
    ans = 0;
    for (int i = 1; i <= n; i++) {
        x = p[i];
        while (p[x] != x) x = p[x];
        ans = max(ans, __builtin_popcount(x));
    }
    cout << ans << endl;
}
