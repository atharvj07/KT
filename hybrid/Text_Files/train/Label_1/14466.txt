#include <algorithm>
#include <bits/stdc++.h>
using namespace std;
int t, n, m, x;
int a[500010];
int main() {
  cin >> t;
  while (t--) {
    cin >> n >> m;
    vector<int> b(n);
    for (int i = 0; i < n; i++) {
      cin >> x;
      b[x - 1] = i;
    }
    for (int i = 0; i < n; i++) a[i] = b[(i + n - m % n) % n];
    vector<int> pos;
    for (int i = 0; i < n; i++)
      if (a[i] == i) pos.push_back(i);
    cout << pos.size() << " ";
    for (int i = 0; i < pos.size(); i++) {
      x = pos[i];
      while (x >= m % n) x -= m % n;
      cout << x << " ";
    }
    cout << "\n";
  }
}
