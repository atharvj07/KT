import java.util.Scanner;

class Main {
    static int[][] grid;
    static int[][] dirs = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int n = sc.nextInt();
        grid = new int[m][n];
        for (int i = 0; i < m; i++) {
            String line = sc.next();
            for (int j = 0; j < n; j++) {
                grid[i][j] = line.charAt(j) == '.' ? 0 : line.charAt(j) - 'A' + 1;
            }
        }
        int maxScore = 0;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 0) {
                    int[] colors = new int[26];
                    for (int[] dir : dirs) {
                        int x = i + dir[0];
                        int y = j + dir[1];
                        while (x >= 0 && x < m && y >= 0 && y < n) {
                            if (grid[x][y] != 0) {
                                colors[grid[x][y] - 1]++;
                            }
                            x += dir[0];
                            y += dir[1];
                        }
                    }
                    int score = 0;
                    for (int cnt : colors) {
                        score += cnt / 2;
                    }
                    maxScore = Math.max(maxScore, score);
                }
            }
        }
        System.out.println(maxScore);
    }
}
