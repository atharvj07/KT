N = int(input())
adj = [[] for i in range(N)]
dp = [[0]*2 for i in range(N)]
for i in range(N):
    c = input()
    for j in range(N):
        if c[j] == '?':
            adj[i].append(j)
    c2 = c[::-1]
    for j in range(N):
        if c2[j] == '?':
            adj[j].append(i)
        elif c2[j] == 'W':
            dp[i][1] = 1
            dp[j][0] = 1
dp[0][0] = 1
for i in range(N):
    for j in adj[i]:
        dp[i][0] = (dp[i][0] + dp[j][1]) % 998244353
        dp[i][1] = (dp[i][1] + dp[j][0]) % 998244353
print(dp[0][0])
