#include <iostream>

using namespace std;

const long long MOD = 1e9 + 7;

long long pwr(long long x, long long n) {
  long long ans = 1;
  while (n) {
    if (n % 2) ans = ans * x % MOD;
    x = x * x;
    x %= MOD;
    n >>= 1;
  }
  return ans % MOD;
}

int main() {
  long long t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    cout << pwr(2, n - 1) << endl;
  }
  return 0;
}
