#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int N = 2e5 + 10, M = 2e5 + 10, oo = 0x3f3f3f3f;
ll a[N], b[N], ans[N];
vector<pair<int, pair<int, int>>> qu;
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    int n, q, m;
    cin >> n >> q >> m;
    for (int i = 1; i <= n; i++) cin >> a[i], b[i] = a[i];
    while (q--) {
        int t, l, r;
        cin >> t >> l >> r;
        qu.push_back({t, {l, r}});
    }
    for (int i = 1; i <= m; i++) cin >> ans[i];
    reverse(qu.begin(), qu.end());
    for (auto it : qu) {
        int t = it.first, l = it.second.first, r = it.second.second;
        if (t == 1) {
            ll x = a[r];
            for (int i = r; i > l; i--) a[i] = a[i - 1];
            a[l] = x;
        } else {
            for (ll i = l, j = r; i < j; i++, j--) swap(a[i], a[j]);
        }
    }
    for (int i = 1; i <= m; i++) {
        cout << a[ans[i]] << " ";
    }
    cout << "\n";
    return 0;
}
