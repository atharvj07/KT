import sys
sys.setrecursionlimit(100000)

n, m, k = map(int, input().split())
a = [['.'] * (m + 2) for i in range(n + 2)]
for i in range(1, n + 1):
	s = list(input())
	for j in range(1, m + 1):
		a[i][j] = s[j - 1]

for x in range(1, n + 1):
	for y in range(1, m + 1):
		if a[x][y] == 'X':
			ans = [[x, y, 0]]
			c = 0
			while c < k:
				x, y, t = ans[-1]
				if t == 2:
					if a[x][y - 1] == '.' and [x, y - 1, 0] not in ans:
						ans.append([x, y - 1, 0])
						c += 1
					elif a[x - 1][y] == '.' and [x - 1, y, 1] not in ans:
						ans.append([x - 1, y, 1])
						c += 1
					elif a[x][y + 1] == '.' and [x, y + 1, 2] not in ans:
						ans.append([x, y + 1, 2])
						c += 1
					else:
						ans.pop()
						ans.append([x + 1, y, 3])
				elif t == 3:
					if a[x][y + 1] == '.' and [x, y + 1, 2] not in ans:
						ans.append([x, y + 1, 2])
						c += 1
					elif a[x + 1][y] == '.' and [x + 1, y, 3] not in ans:
						ans.append([x + 1, y, 3])
						c += 1
					elif a[x][y - 1] == '.' and [x, y - 1, 0] not in ans:
						ans.append([x, y - 1, 0])
						c += 1
					else:
						ans.pop()
						ans.append([x - 1, y, 1])
				elif t == 0:
					if a[x - 1][y] == '.' and [x - 1, y, 1] not in ans:
						ans.append([x - 1, y, 1])
						c += 1
					elif a[x][y + 1] == '.' and [x, y + 1, 2] not in ans:
						ans.append([x, y + 1, 2])
						c += 1
					elif a[x + 1][y] == '.' and [x + 1, y, 3] not in ans:
						ans.append([x + 1, y, 3])
						c += 1
					else:
						ans.pop()
						ans.append([x, y - 1, 0])
				elif t == 1:
					if a[x + 1][y] == '.' and [x + 1, y, 3] not in ans:
						ans.append([x + 1, y, 3])
						c += 1
					elif a[x][y - 1] == '.' and [x, y - 1, 0] not in ans:
						ans.append([x, y - 1, 0])
						c += 1
					elif a[x - 1][y] == '.' and [x - 1, y, 1] not in ans:
						ans.append([x - 1, y, 1])
						c += 1
					else:
						ans.pop()
						ans.append([x, y + 1, 2])

			ans.pop()
			if c == k:
				break

if len(ans) > 0:
	print("".join(["LRUD"[(t + 2) % 4] for _, _, t in ans]))
else:
	print("IMPOSSIBLE")
