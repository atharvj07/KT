from sys import stdin
input = stdin.readline

for _ in range(int(input())):
    n = int(input())
    edges = [[] for _ in range(n+1)]
    for _ in range(n):
        a, b = map(int, input().split())
        edges[a].append(b)
        edges[b].append(a)
    vis = [False] * (n+1)
    
    def dfs(node, cnt):
        vis[node] = True
        ret = cnt
        for next in edges[node]:
            if vis[next] == True:
                continue
            ret += dfs(next, cnt+1)
        vis[node] = False
        return ret
    
    print(dfs(1, 0))
