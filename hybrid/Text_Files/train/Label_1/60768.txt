t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    a.sort()
    b.sort()
    
    possible = True
    for i in range(n):
        if b[i] != a[i] and b[i] != a[i] + 1:
            possible = False
            break
    
    if possible:
        print("YES")
    else:
        print("NO")
