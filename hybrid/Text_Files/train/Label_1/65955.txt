for _ in range(int(input())):
  n = int(input())
  s = input()
  if s.count('0')%2: ans = 'NO'
  else:
    a = ['(']*n
    b = ['(']*n
    c = 0
    for i in range(n):
      if s[i]=='0':
        if not c%2: a[i] = b[i] = ')'
        c+=1
      else: c=0
    ans = 'YES\n'+''.join(a)+'\n'+''.join(b)
  print(ans)
