#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    string s;
    cin >> s;
    vector<int> pre(n), suf(n);
    for (int i = 1; i < n; i++) {
        pre[i] = pre[i - 1];
        if (s[i] == s[i - 1]) pre[i]++;
    }
    for (int i = n - 2; i >= 0; i--) {
        suf[i] = suf[i + 1];
        if (s[i] == s[i + 1]) suf[i]++;
    }
    while (m--) {
        int l, r;
        cin >> l >> r;
        int ans = 0;
        if (l > 1) ans += pre[l - 2];
        if (r < n) ans += suf[r];
        cout << ans << endl;
    }
    return 0;
}
