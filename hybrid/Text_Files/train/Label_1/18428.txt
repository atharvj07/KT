#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<pair<int, pair<int, int>>> v(n);
    for (int i = 0; i < n; i++) {
        int a, b, c, d;
        cin >> a >> b >> c >> d;
        v[i].first = a + b + c + d;
        v[i].second.first = i;
        v[i].second.second = a;
    }
    sort(v.rbegin(), v.rend());
    for (int i = 0; i < n; i++) {
        if (v[i].second.first == 0 && v[i].second.second == 100) {
            cout << i + 1 << endl;
            return 0;
        }
    }
}
