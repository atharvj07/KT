import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[][] players = new int[4][2];
        for (int i = 0; i < 4; i++) {
            players[i][0] = in.nextInt();
            players[i][1] = in.nextInt();
        }

        // Sort players by defence skill
        for (int i = 0; i < 3; i++) {
            for (int j = i + 1; j < 4; j++) {
                if (players[i][0] > players[j][0]) {
                    int[] temp = players[i];
                    players[i] = players[j];
                    players[j] = temp;
                }
            }
        }

        // Sort players by attack skill
        for (int i = 0; i < 3; i++) {
            for (int j = i + 1; j < 4; j++) {
                if (players[i][1] > players[j][1]) {
                    int[] temp = players[i];
                    players[i] = players[j];
                    players[j] = temp;
                }
            }
        }

        // Check if first team can win
        if (players[0][0] > players[2][1] && players[1][1] > players[3][0]) {
            System.out.println("Team 1");
            return;
        }

        // Check if second team can win
        if (players[2][0] > players[0][1] && players[3][1] > players[1][0]) {
            System.out.println("Team 2");
            return;
        }

        // Otherwise, it's a draw
        System.out.println("Draw");
    }
}
