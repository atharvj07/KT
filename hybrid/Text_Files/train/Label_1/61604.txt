n, k = map(int, input().split())

# If k is larger than the maximum number of 1s that can fit in such a matrix, print -1
if k > n * (n + 1) // 2:
    print(-1)
else:
    matrix = [[0] * n for _ in range(n)]
    for i in range(n):
        if k > i:
            matrix[i][i] = 1
            k -= (i + 1)
        else:
            for j in range(k):
                matrix[i-j][j] = 1
            break
    for row in matrix:
        print(" ".join(map(str, row)))
