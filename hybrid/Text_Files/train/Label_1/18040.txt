#include <vector>
#include <set>
#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<set<int>> stops(n);

  for (int i = 0; i < n; i++) {
    int s;
    cin >> s;

    for (int j = 0; j < s; j++) {
      int l;
      cin >> l;
      stops[i].insert(l);
    }
  }

  set<int> possible_lines;

  for (set<int> s : stops) possible_lines.insert(s.begin(), s.end());

  for (int l : possible_lines) {
    bool line_possible = true;

    for (set<int> s : stops) {
      if (s.find(l) == s.end()) {
        line_possible = false;
        break;
      }
    }

    if (line_possible) cout << l << ' ';
  }

  cout << endl;

  return 0;
}
