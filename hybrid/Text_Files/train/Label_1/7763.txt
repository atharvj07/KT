#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define FOR(i,a,b) for(int i=a;i<=b;++i)
#define FORD(i,a,b) for(int i=a;i>=b;--i)
#define REP(i,n) for(int i=0;i<n;++i)
#define REPD(i,n) for(int i=n-1;i>=0;--i)
#define fi first
#define se second
#define pb push_back
#define mp make_pair
#define endl '\n'
const ll mod=1e9+7;
int main()
{
    int n;
    cin>>n;
    vector<ll> a(n);
    map<ll,vector<int>> m;
    FOR(i,1,n)
    {
        cin>>a[i-1];
        m[a[i-1]].pb(i);
    }
    vector<int> ans;
    for(auto it:m)
    {
        vector<int> p=it.se;
        bool ok=0;
        for(int i=1;i<p.size();i++)
        {
            if(p[i]-p[i-1]==1)
            {
                ok=1;
            }
            else
            {
                ok=0;
                break;
            }
        }
        if(ok)
        {
            ans.pb(p.size());
        }
    }
    cout<<ans.size()<<endl;
    for(auto it:ans)
    {
        cout<<it<<" ";
    }
    return 0;
}
