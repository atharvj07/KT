#include<bits/stdc++.h>
using namespace std;
int n;
vector<vector<int>> a;
vector<int> p;
int main(){
    cin>>n;
    a.resize(n,vector<int>(n));
    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++) cin>>a[i][j];
        p.push_back(0);
    }
    iota(p.begin(),p.end(),1);
    for(int i=0;i<n;i++){
        vector<int> b=p;
        for(int j=0;j<n;j++){
            if(p[j]!=0){
                if(a[p[j]-1][j]==0){
                    for(int k=i+1;k<n;k++){
                        if(b[k]==0) continue;
                        swap(b[j],b[k]);
                        break;
                    }
                }
            }
        }
        if(b!=p){
            p=b;
            break;
        }
    }
    for(int x:p) cout<<x<<" ";
}
