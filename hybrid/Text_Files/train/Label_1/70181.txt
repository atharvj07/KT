n,m=map(int,input().split())
g = []
for i in range(n):
    g+=[list(map(int,input().split()))]
cnt = 0
for i in range(n):
    for j in range(m):
        for t in range(i+1):
            for y in range(j+1):
                if g[i][j]==g[t][y]==g[(i+j-t-y)%n][(i+j-t-y)%m]:
                    cnt+=1
print(cnt)
