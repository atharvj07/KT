const int N = 2e5+5;

int h,a[N],l[N],r[N];
map<pii,int> mp;
void dfs(int node,int par,int dep){
	a[dep]++;
	l[node] = mp[{dep,10*par+node}];
	r[node] = mp[{dep,10*node+par}];
	mp[{dep,10*node+par}] = node;
	mp[{dep,10*par+node}] = node;
	if(l[node]!=0){
		dfs(l[node],node,dep+1);
	}
	if(r[node]!=0){
		dfs(r[node],node,dep+1);
	}
}

int32_t main(){
	cin>>h;
	int sum = 0;
	for(int i=0;i<=h;i++){
		cin>>a[i];
		sum+=a[i];
	}
	if(sum==h){
		cout<<"perfect"<<endl;
	}
	else{
		cout<<"ambiguous"<<endl;
		for(int i=1;i<=h;i++){
			cout<<"0 ";
		}
		cout<<endl;
		int i=1;
		while(a[i]==0)i++;
		i--;
		int c = a[i];
		a[i]--;
		dfs(1,0,i);
		for(int j=1;j<=h;j++){
			cout<<l[j]<<" ";
		}
		cout<<endl;
		i++;
		a[i]++;
		dfs(1,0,i);

			cout<<l[j]<<" ";
		}

	}

}
