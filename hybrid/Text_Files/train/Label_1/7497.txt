#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int l, r;
        cin >> l >> r;
        int x = l, y = l * 2;
        if (y > r) {
            x = l + 1;
            y = l * 2;
        }
        cout << x << " " << y << endl;
    }
    return 0;
}
