n = int(input())
a = list(map(int, input().split()))
print(n - 1)
ma = max(a)
for i in range(n):
    if a[i] == ma:
        continue
    if a[i] * ma > 0:
        print(1, i + 1, a.index(ma) + 1)
        a[a.index(ma)] = a[i] * ma
        a[i] = ma
    else:
        print(2, i + 1)
        a.pop(i)
        break
