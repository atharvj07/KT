from fractions import Fraction
n, p = map(int, input().split())
l = [list(map(int, input().split())) for _ in range(n)]
ans = Fraction(0)
for i in range(n):
    for j in range(i + 1, n):
        if (l[j][0] <= l[i][1] <= l[j][1] or l[j][0] <= l[i][0] <= l[j][1]) and (l[j][0] * l[i][1] % p == 0 or l[j][1] * l[i][0] % p == 0):
            ans += 1000
print(ans / n**2)
