#include <bits/stdc++.h>
using namespace std;
const int N = 1<<20;
int sa[N], ea[N], sb[N], eb[N];
int main() {
    int n; cin >> n;
    for (int i = 1; i <= n; i++) cin >> sa[i] >> ea[i] >> sb[i] >> eb[i];
    bool flag = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (i == j) continue;
            if (sa[i] <= sb[j] && ea[i] >= sb[j]) {
                if (sb[i] >= sa[j] && eb[i] <= ea[j]) flag = 1;
            }
            if (sb[i] <= sa[j] && eb[i] >= sa[j]) {
                if (sa[i] >= sb[j] && ea[i] <= eb[j]) flag = 1;
            }
        }
    }
    cout << (flag ? "NO" : "YES") << endl;
    return 0;
}
