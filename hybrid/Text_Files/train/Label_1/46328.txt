import java.util.*;
class Main {
    static int n;
    static List<Node>[] children;
    static int[] w;
    static int[] prefix;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        children = new ArrayList[n + 1];
        w = new int[n + 1];
        prefix = new int[n + 1];
        
        for (int i = 1; i <= n; i++) {
            w[i] = sc.nextInt();
            children[i] = new ArrayList<>();
        }
        for (int i = 1; i < n; i++) {
            int u = sc.nextInt(), v = sc.nextInt(), c = sc.nextInt();
            children[u].add(new Node(v, c));
            children[v].add(new Node(u, c));
        }
        dfs1(1, 1, 0);
        int mx = 0;
        dfs2(1, 1, 0, mx);
        System.out.println(mx);
    }
    static void dfs1(int u, int p, int sum) {
        prefix[u] = sum;
        for (Node child : children[u]) {
            if (child.to != p) {
                dfs1(child.to, u, sum + child.cost);
            }
        }
    }
    static void dfs2(int u, int p, int sum, int mx) {
        mx = Math.max(mx, sum + w[u]);
        for (Node child : children[u]) {
            if (child.to != p) {
                dfs2(child.to, u, sum + child.cost - w[u], mx);
            }
        }
        mx = Math.max(mx, prefix[u]);
    }
    static class Node {
        int to, cost;
        public Node(int u, int c) {
            this.to = u;
            this.cost = c;
        }
    }
}
