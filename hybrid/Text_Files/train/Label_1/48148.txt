import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt();
        int M = in.nextInt();
        long[] A = new long[N];
        for (int i = 0; i < N; i++) {
            A[i] = in.nextInt();
        }
        long ans = Long.MAX_VALUE;
        for (int i = 0; i < (1 << M); i++) {
            long sum = 0;
            for (int j = 0; j < N; j++) {
                long cost = A[j];
                int used = 0;
                while ((i & (1 << used)) > 0) {
                    used++;
                    cost /= 2;
                }
                sum += cost;
            }
            ans = Math.min(ans, sum);
        }
        System.out.println(ans);
    }
}
