import heapq
import sys

n, m, s, t = map(int, input().split())
graph = [[] for i in range(n+1)]
inf = n * m * 10**6
for i in range(m):
    a, b, l = map(int, input().split())
    graph[a].append((b, l))

dist = [inf] * (n+1)
dist[s] = 0
heap = [(0, s)]
taken = [False] * (n+1)
while heap:
    d, u = heapq.heappop(heap)
    taken[u] = True
    if u == t:
        break
    for v, w in graph[u]:
        if d + w < dist[v]:
            dist[v] = d + w
            heapq.heappush(heap, (d+w, v))

for i in range(1, n+1):
    if taken[i]:
        print("YES")
    else:
        if dist[i] == inf:
            print("NO")
        else:
            print('CAN', dist[i] - 1)
