import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        char[] s = sc.next().toCharArray();
        for (int i = 0; i < m; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            char c1 = sc.next().charAt(0);
            char c2 = sc.next().charAt(0);
            for (int j = l; j <= r; j++) {
                if (s[j] == c1) {
                    s[j] = c2;
                }
            }
        }
        System.out.println(new String(s));
    }
}
