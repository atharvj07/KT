#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int M = 1e9 + 7;

int f, w, h;

ll power(ll x, ll y) {
  ll ans = 1;
  while (y > 0) {
    if (y % 2) { ans = (x * 1ll * ans) % M; }
    x = (x * 1ll * x) % M;
    y /= 2;
  }
  return ans;
}

int main() {
  cin >> f >> w >> h;
  ll ans = 0;
  if (h == 0) {
    ans = (f + w);
  } else {
    if (f > 0) {
      ll a = w + 1;
      ll b = (power(a, f + w) - 1 + M) % M;
      a = power(a, M - 2);
      ans = (a * 1ll * b) % M;
    }
    if (w > 0) {
      ll a = f + 1;
      ll b = (power(a, f + w) - 1 + M) % M;
      a = power(a, M - 2);
      ans = (ans + a * 1ll * b) % M;
    }
  }
  cout << ans << endl;
  return 0;
}
