#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
const int INF = 1e9;
int n, m, s[N], x[N], ans = -INF;
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) { cin >> s[i] >> x[i]; }
    for (int i = 1; i <= m; ++i) {
        int mn = INF, mx = -INF;
        for (int j = 1; j <= n; ++j) {
            if (s[j] != i) continue;
            mn = min(mn, x[j]), mx = max(mx, x[j]);
        }
        if (mn < mx) ans += mx;
    }
    if (ans == -INF) ans = 0;
    cout << ans;
}
