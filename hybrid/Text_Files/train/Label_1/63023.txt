n=int(input())
lst=[int(x) for x in input().split()]
msk=int(''.join(input()),2)
msk1=(1<<n)-1
dp=[[-1]*(msk1+1) for _ in range(n+1)]
def go(i,msk):
    if i==n: return 0
    if dp[i][msk]!=-1: return dp[i][msk]
    op1=go(i+1,msk)
    op2=lst[i]+go(i+1,msk|(1<<i))
    dp[i][msk]=max(op1,op2)
    return dp[i][msk]
print(go(0,0))
