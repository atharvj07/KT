import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    Arrays.sort(a);
    int ans = 1;
    for (int i = 0; i < n; i++) {
      if (a[i] > 0) {
        ans *= a[i];
      }
    }
    for (int i = 0; i < n; i++) {
      if (a[i] > 0) {
        System.out.print(a[i] + " ");
      } else {
        break;
      }
    }
  }
}
