#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    vector<char> a(n), b(n);
    bool f = false;
    for (int i = 0; i < n; i++) {
      if (v[i] == 1 && !f) {
        a[i] = '(';
        b[i] = ')';
      } else if (v[i] == 0) {
        if (a[i] == '(') {
          b[i] = ')';
          f = true;
        } else {
          a[i] = ')';
          b[i] = '(';
          f = false;
        }
      } else if (v[i] == 1 && f) {
        a[i] = ')';
        b[i] = '(';
      }
    }
    if (f == true) {
      cout << "YES" << endl;
      for (int i = 0; i < n; i++) { cout << a[i]; }
      cout << endl;
      for (int i = 0; i < n; i++) { cout << b[i]; }
      cout << endl;
    } else {
      cout << "NO" << endl;
    }
  }
  return 0;
}
