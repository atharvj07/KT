#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define vi vector<int>
#define pb push_back
#define MAXN 500005

int t[MAXN], a[MAXN], b[MAXN];

int main() {
    int n, k;
    cin >> n >> k;
    vector<tuple<int, int, int>> books;
    for (int i = 0; i < n; i++) {
        cin >> t[i] >> a[i] >> b[i];
        books.pb({t[i], a[i], b[i]});
    }
    sort(books.begin(), books.end());
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> aa;
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> bb;
    int ta = 0, tb = 0, cnta = 0, cntb = 0;
    ll ans = LLONG_MAX;
    for (auto [tim, a, b] : books) {
        if (!a) {
            aa.push({-tim, cnta});
            cnta++;
        }
        if (!b) {
            bb.push({-tim, cntb});
            cntb++;
        }
        ta += tim * a;
        tb += tim * b;
        while (cnta > k and !aa.empty()) {
            auto u = aa.top();
            aa.pop();
            ta -= -u.first * u.second;
            cnta -= u.second;
        }
        while (cntb > k and !bb.empty()) {
            auto u = bb.top();
            bb.pop();
            tb -= -u.first * u.second;
            cntb -= u.second;
        }
        if (cnta >= k and cntb >= k) {
            ans = min(ans, 1LL * ta + tb);
        }
    }
    if (ans == LLONG_MAX) ans = -1;
    cout << ans << '\n';
    return 0;
}
