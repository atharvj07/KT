#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
vector<int>e[maxn], ed[maxn];
int type[maxn];
int d[maxn];
int vis[maxn];
int ans;
int n, m;
void dfs(int u, bool b) {
  if(vis[u] == 1) return;
  vis[u] = 1;
  if(type[u] != b) ans++;
  for(int v : e[u]) {
    if(vis[v] == 1) return;
    dfs(v, b);
  }
  for(int v : ed[u]) dfs(v, b);
}
int main() {
  cin >> n >> m;
  queue<int>q;
  for(int i = 0; i < n; i++) {
    cin >> type[i];
    if(type[i] == 1) q.push(i);
  }
  for(int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    e[u].push_back(v);
    ed[v].push_back(u);
    d[v]++;
  }
  while(!q.empty()) {
    int u = q.front();
    q.pop();
    for(int v : e[u]) {
      d[v]--;
      if(d[v] == 0) {
        q.push(v);
      }
    }
  }
  dfs(0, 0);
  dfs(n - 1, 1);
  cout << ans << endl;
  return 0;
}
