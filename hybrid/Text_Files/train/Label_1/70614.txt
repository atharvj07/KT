import sys

def main():
    n, k, x = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    a.sort(reverse=True)
    b = [a[i] for i in range(n) if a[i] > 0]
    c = [a[i] for i in range(n) if a[i] == 0]
    for i in range(k):
        if len(b) == 0:
            break
        b[0] *= x
        b.sort(reverse=True)
    ans = 0
    for i in b:
        ans |= i
    for i in c:
        ans |= i
    print(ans)

main()
