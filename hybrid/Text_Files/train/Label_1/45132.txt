import java.util.*;
import java.math.*;

public class Main {

    static class foodList implements Comparable<foodList>
    {
        int index;
        int count;
        foodList(int a, int b)
        {
            index = a;
            count = b;
        }

        public int compareTo(foodList other)
        {
            return count - other.count;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] quantity = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            quantity[i]=scanner.nextInt();
        }
        boolean[] exists = new boolean[n + 1];
        ArrayList<ArrayList<Integer>> adj = new ArrayList<ArrayList<Integer>>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<Integer>());
        }
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            adj.get(a).add(b);
            adj.get(b).add(a);
            exists[a] = true;
            exists[b] = true;
        }
        int ansCount = 0;
        for (int i = 1; i <= n; i++) {
            if (quantity[i] > 0) {
                ansCount++;
            }
        }
        ArrayList<foodList> arr = new ArrayList<foodList>();
        for (int i = 1; i <= n; i++) {
            if (quantity[i] > 0) {
                arr.add(new foodList(i, quantity[i]));
            }
        }
        Collections.sort(arr);
        ArrayList<Integer> ans = new ArrayList<Integer>();
        while (ansCount > 0 && arr.size() > 0) {
            foodList top = arr.get(arr.size() - 1);
            arr.remove(arr.size() - 1);
            if (quantity[top.index] > 0) {
                ans.add(top.index);
                ansCount--;
                for (int i = 0; i < adj.get(top.index).size(); i++) {
                    quantity[adj.get(top.index).get(i)]--;
                    if (quantity[adj.get(top.index).get(i)] > 0) {
                        arr.add(new foodList(adj.get(top.index).get(i), quantity[adj.get(top.index).get(i)]));
                        Collections.sort(arr);
                    }
                }
            }
        }
        if (ansCount > 0) {
            System.out.println("DEAD");
        }
        else {
            Collections.reverse(ans);
            System.out.println("ALIVE");
            for (int i = 0; i < ans.size(); i++) {
                System.out.print(ans.get(i) + " ");
            }
            System.out.println();
        }
    }
}
