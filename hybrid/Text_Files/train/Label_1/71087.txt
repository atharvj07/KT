from collections import deque
s=input();st='('+s+')';L=len(s);l,c,r=deque(),0,0
for i in st:
    if i=='(':c+=1;l.append('1')
    else:
        if c==0 or l.pop()!='1':r+=1
        else:c-=1
if c or r>L-c-2:print(':(')
else:
    for i in range(1,L-1):
        if st[i]=='?':
            st=st[:i] + (l.popleft() if c else ')') + st[i+1:]
        else:
            c+=1 if st[i]=='('else -1
    print(st)
