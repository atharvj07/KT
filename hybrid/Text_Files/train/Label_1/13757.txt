#include <bits/stdc++.h>
using namespace std;
int main() {
    vector<int> len(4);
    for (int i = 0; i < 4; i++) {
        string s;
        getline(cin, s);
        len[i] = s.length() - 3;
    }
    int minl = *min_element(len.begin(), len.end());
    int maxl = *max_element(len.begin(), len.end());
    if (minl * 2 > maxl || maxl * 2 > minl) {
        cout << (char)('A' + distance(len.begin(), find(len.begin(), len.end(), minl * 2 > maxl ? minl : maxl)));
        return 0;
    }
    cout << 'C';
}
