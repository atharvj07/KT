from math import ceil
n = int(input())
a = [*map(int, input().split())]
ta = sum(a)
a.sort(reverse=True)
for i in range(len(a)):
    tb = a[i]
    c = [1]
    if tb * 2 <= ta:
        for j in range(i + 1, len(a)):
            if tb * 2 < ta - a[j]: break
            tb += a[j]
            c.append(j + 1)
    if tb * 2 > ta:
        print(len(c))
        print(*c)
        exit(0)
print(0)
