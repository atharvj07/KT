import java.util.Scanner;

public class Restaurant {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();

        int onePersonSeated = 0;
        int twoPersonSeated = 0;
        int deniedService = 0;

        for (int i = 0; i < n; i++) {
            int groupSize = scanner.nextInt();

            if (groupSize == 1) {
                if (onePersonSeated < a) {
                    onePersonSeated++;
                } else if (twoPersonSeated < b) {
                    twoPersonSeated++;
                } else if (twoPersonSeated < 2 * b) {
                    twoPersonSeated++;
                } else {
                    deniedService++;
                }
            } else { // groupSize == 2
                if (twoPersonSeated < b) {
                    twoPersonSeated++;
                } else {
                    deniedService += 2;
                }
            }
        }

        System.out.println(deniedService);
    }
}
