import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int[] arr = new int[n];
            boolean[] broken = new boolean[n];
            for (int i = 0; i < n; i++) {
                String s = sc.next();
                if (s.equals("x")) {
                    broken[i] = true;
                } else {
                    arr[i] = Integer.parseInt(s);
                }
            }
            boolean unique = true;
            int prev = -1;
            int next = -1;
            for (int i = 0; i < n; i++) {
                if (broken[i]) {
                    if (prev != -1 && next != -1) {
                        if (prev < next) {
                            arr[i] = prev + 1;
                        } else {
                            arr[i] = next - 1;
                        }
                    } else {
                        unique = false;
                    }
                } else {
                    if (i % 2 == 0 && arr[i] <= prev) {
                        unique = false;
                    } else if (i % 2 == 1 && arr[i] >= next) {
                        unique = false;
                    }
                    prev = arr[i];
                    if (i + 1 < n) {
                        next = arr[i + 1];
                    }
                }
            }
            if (!unique) {
                System.out.println("ambiguous");
            } else {
                for (int i = 0; i < n; i++) {
                    if (broken[i]) {
                        System.out.print(arr[i] + " ");
                    }
                }
                System.out.println();
            }
        }
    }
}
