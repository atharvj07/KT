#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    long long sum_L = 0, sum_R = 0, ans;
    vector<long long> vec(n);
    for (int i = 0; i < n; i++) {
      cin >> vec[i];
      sum_R += vec[i];
    }
    ans = sum_L + sum_R;
    for (int i = 0; i < n; i++) {
      long long cur = 0;
      if (i > 0) cur += vec[i - 1];
      if (i < n - 1) cur += vec[i + 1];
      sum_L += vec[i];
      sum_R -= vec[i];
      ans = min(ans, sum_L + sum_R + cur);
    }
    cout << ans << endl;
  }
  return 0;
}
