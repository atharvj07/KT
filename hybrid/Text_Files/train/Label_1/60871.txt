import sys

line = sys.stdin.readline()
n, s, k = map(int, line.split())

line = sys.stdin.readline()
r = list(map(int, line.split()))

line = sys.stdin.readline()
c = list(line)

ans = s-1      
i = s
cnt = 0
last = -1

while cnt < k:
    while i < n+1 and (cnt + r[i-1] < k or c[i-1] == last):
        i += 1
    if i == n+1: break

    cnt += r[i-1]
    ans += 1
    last = c[i-1]
    i += 1

print(ans) if cnt >= k else print('-1')
