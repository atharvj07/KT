import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = reader.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int k = Integer.parseInt(line1[1]);
        int p = Integer.parseInt(line1[2]);
        String[] line2 = reader.readLine().split(" ");
        int[] people = new int[n];
        for (int i = 0; i < n; i++) {
            people[i] = Integer.parseInt(line2[i]);
        }
        String[] line3 = reader.readLine().split(" ");
        int[] keys = new int[k];
        for (int i = 0; i < k; i++) {
            keys[i] = Integer.parseInt(line3[i]);
        }
        Arrays.sort(people);
        Arrays.sort(keys);
        int time = 0;
        int peopleIndex = 0;
        int keyIndex = 0;
        while (peopleIndex < n) {
            if (keys[keyIndex] >= people[peopleIndex]) {
                time = Math.max(time, Math.abs(keys[keyIndex] - p));
                peopleIndex++;
                keyIndex++;
            } else {
                keyIndex++;
            }
            if (keyIndex == k) {
                break;
            }
        }
        if (peopleIndex == n) {
            System.out.println(time);
        } else {
            System.out.println(-1);
        }
    }
}
