#include <bits/stdc++.h>
using namespace std;
const int maxn = 5e4 + 10;
int main() {
  int n, y[maxn];
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> y[i];
  }
  vector<int> x;
  for (int i = 0; i < n; i++) {
    int yy = y[i];
    while (yy % 2 == 0) {
      yy /= 2;
    }
    x.push_back(yy);
  }
  sort(x.begin(), x.end());
  for (int i = 0; i < n; i++) {
    cout << x[i] << " ";
  }
  cout << endl;
  return 0;
}
