import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Long, Pos> a = new HashMap<>();
        Map<Long, Pos> b = new HashMap<>();
        double maxDistA = -1;
        long maxTimeA = Integer.MAX_VALUE;
        double maxDistB = -1;
        long maxTimeB = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            long f = sc.nextLong();
            int a_ = sc.nextInt();
            int t_ = sc.nextInt();
            int x = sc.nextInt();
            int y = sc.nextInt();
            if (t_ == 0) {
                Pos prevA = a.get(f);
                if (prevA != null) {
                    double dist = Math.sqrt(Math.pow(x - prevA.x, 2) + Math.pow(y - prevA.y, 2));
                    if (dist > maxDistA) {
                        maxDistA = dist;
                        maxTimeA = f - prevA.time;
                    } else if (dist == maxDistA) {
                        maxTimeA = Math.min(maxTimeA, f - prevA.time);
                    }
                }
                a.put(f, new Pos(f, x, y));
            } else {
                Pos prevB = b.get(f);
                if (prevB != null) {
                    double dist = Math.sqrt(Math.pow(x - prevB.x, 2) + Math.pow(y - prevB.y, 2));
                    if (dist > maxDistB) {
                        maxDistB = dist;
                        maxTimeB = f - prevB.time;
                    } else if (dist == maxDistB) {
                        maxTimeB = Math.min(maxTimeB, f - prevB.time);
                    }
                }
                b.put(f, new Pos(f, x, y));
            }
        }
        sc.close();
        if (maxDistA == -1) {
            System.out.println("-1 -1");
        } else {
            System.out.println(String.format("%.10f %f", maxDistA, maxTimeA / 60.0));
        }
        if (maxDistB == -1) {
            System.out.println("-1 -1");
        } else {
            System.out.println(String.format("%.10f %f", maxDistB, maxTimeB / 60.0));
        }
    }

    static class Pos {
        long time;
        int x;
        int y;

        public Pos(long time, int x, int y) {
            this.time = time;
            this.x = x;
            this.y = y;
        }
    }
}
