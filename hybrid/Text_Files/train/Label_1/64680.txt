import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    seg = []
    for i in range(n-1):
        seg.append([int(j) for j in input().split()])
    used = set()
    p = []
    for i in range(n-1):
        for j in seg[i][1:]:
            if j not in used:
                p.append(j)
                used.add(j)
    for i in range(1,n+1):
        if i not in used:
            p.append(i)
            break
    for i in seg[n-2:]:
        for j in i[1:]:
            if j not in used:
                p.append(j)
                used.add(j)
    print(*p)
