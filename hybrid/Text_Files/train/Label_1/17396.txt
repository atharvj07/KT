#include <vector>
#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        vector<int> last(n + 1, -1);
        long long ans = 0;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            if (last[a[i]] != -1) {
                ans += i - last[a[i]];
            }
            last[a[i]] = i;
        }
        cout << ans << "\n";
    }
    return 0;
}
