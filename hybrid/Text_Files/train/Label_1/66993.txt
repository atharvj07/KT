t = int(input())
for _ in range(t):
    n = input()
    if int(n[-1]) % 2 == 0:
        print(0)
    elif int(n[0]) % 2 == 0:
        print(1)
    else:
        even_digit_found = False
        for digit in n:
            if int(digit) % 2 == 0:
                even_digit_found = True
                break
        if even_digit_found:
            print(2)
        else:
            print(-1)
