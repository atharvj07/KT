from collections import defaultdict

n = int(input())
d = defaultdict(list)
for _ in range(n - 1):
    a, b = map(int, input().split())
    d[a].append(b)
    d[b].append(a)
if any(len(v) > 2 for v in d.values()):
    print('No')
else:
    print('Yes')
    paths = []
    vis = set()
    for node in d:
        if node in vis:
            continue
        vis.update((node, dest) for dest in d[node])
        paths.append((node, *d[node]))
    print(len(paths))
    for path in paths:
        print(*path)
