n, m, q = map(int, input().split())
a = list(map(int, input().split()))

# Count the occurrences of each city hosting the olympiad before the rule was introduced
city_counts = [0] * m
for city in a:
    city_counts[city - 1] += 1

# Find the minimum count of olympiads hosted by any city before the rule
min_count = min(city_counts)

# Calculate the number of olympiads hosted by each city after the rule is introduced up to the max query
additional_olympiads = [0] * m
for i in range(m):
    additional_olympiads[i] = min_count - city_counts[i]

total_olympiads = sum(additional_olympiads) + n

for _ in range(q):
    k = int(input())
    if k <= total_olympiads:
        idx = (k - n - 1) % m
        print(idx + 1)
    else:
        k -= total_olympiads
        idx = (k - 1) % m
        print(idx + 1)
