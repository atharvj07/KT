#include<bits/stdc++.h>
using namespace std;
int main() {
	int n, a[100001], mini = INT_MAX, maxi = 0;
	cin >> n;
	for (int i = 0; i < n; ++i) {
		cin >> a[i];
		mini = min(a[i], mini);
		maxi = max(a[i], maxi);
	}
	int ans = maxi - mini;
	for (int i = 0; i < n; ++i) {
		if (a[i] == maxi || a[i] == mini) {
			ans = min(ans,  maxi - mini + 1 - (2 * (a[i] == mini)));
		}
	}
	cout << ans << endl;
	return 0;
}
