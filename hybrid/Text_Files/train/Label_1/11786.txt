int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int a, b, n;
        cin >> a >> b >> n;
        vector<pair<int, int>>mon(n);
        for (int i = 0; i < n; i++) { cin >> mon[i].first; }
        for (int i = 0; i < n; i++) { cin >> mon[i].second; }
        int mn = LLONG_MAX;
        for (int i = 0; i < n; i++) {
            if (mon[i].second % a) {
                mn = min(mn, (a - (mon[i].second % a)));
            }
        }
        if (mn == LLONG_MAX) { cout << "YES" << endl; }
        else {
            b -= mn;
            if (b <= 0) { cout << "NO" << endl; }
            else { cout << "YES" << endl; }
        }
    }
    return 0;
}
