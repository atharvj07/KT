#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,d;
    cin>>n>>d;
    long long a[n];
    for(int i=0 ; i<n ; i++)cin>>a[i];
    if(n==1){
        cout<<2<<"\n";
        return 0;
    }
    if(d==1){
        cout<<0<<"\n";
        return 0;
    }
    int x=lower_bound(a,a+n,a[0]+d)-a;
    int y=upper_bound(a,a+n,a[n-1]-d)-a;
    cout<<y-x+1<<"\n";
}
