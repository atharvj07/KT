#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const ll maxn = 1e6;
bool check(ll x, ll y, ll n, ll m, const string& s) {
    vector<vector<bool>> used(n + 1, vector<bool>(m + 1, false));
    ll ansx = 0;
    ll ansy = 0;
    ll cnt = 0;
    used[x][y] = true;
    for (auto& c : s) {
        if (c == 'L') {
            if (y > 1) {
                cnt++;
                ansx = x;
                ansy = y - 1;
                if (!used[ansx][ansy]) used[ansx][ansy] = true;
            } else {
                break;
            }
        } else if (c == 'R') {
            if (y < m) {
                cnt++;
                ansx = x;
                ansy = y + 1;
                if (!used[ansx][ansy]) used[ansx][ansy] = true;
            } else {
                break;
            }
        } else if (c == 'U') {
            if (x > 1) {
                cnt++;
                ansx = x - 1;
                ansy = y;
                if (!used[ansx][ansy]) used[ansx][ansy] = true;
            } else {
                break;
            }
        } else if (c == 'D') {
            if (x < n) {
                cnt++;
                ansx = x + 1;
                ansy = y;
                if (!used[ansx][ansy]) used[ansx][ansy] = true;
            } else {
                break;
            }
        }
    }
    if (cnt == s.size()) return true;
    return false;
}
int main() {
    ll t;
    cin >> t;
    while (t--) {
        ll n, m;
        cin >> n >> m;
        string s;
        cin >> s;
        ll ansx = 1, ansy = 1;
        for (ll i = 1; i <= n; i++) {
            for (ll j = 1; j <= m; j++) {
                if (check(i, j, n, m, s)) {
                    ansx = i;
                    ansy = j;
                }
            }
        }
        cout << ansx << " " << ansy << endl;
    }
    return 0;
}
