#include <algorithm>
#include <vector>
#include <map>
#include <set>
#include <iostream>
#include <queue>
#include <stack>
#include <string>

typedef long long ll;
typedef long double ld;

using namespace std;

struct Edge{
    int from;
    int to;
    int weight;
    Edge(int f,int t,int w):from(f),to(t),weight(w){}
    bool operator < (const Edge&e){
        if(from!=e.from) return from<e.from;
        if(to!=e.to) return to<e.to;
        return weight<e.weight;
    }
};

int main(){
    int n;cin>>n;
    vector<int> a(n);
    for(int i=0;i<n;i++) cin>>a[i];
    vector<vector<Edge>> g(n);
    vector<pair<int,int>> edges;
    for(int sum=n-1,i=0;i+1<n;i++){
        while(a[i]==0) i++;
        if(sum-a[i]<1){
            cout<<"NO"<<endl;
            return 0;
        }
        g[i].push_back(Edge(i,(i+1)%n,1));
        g[(i+1)%n].push_back(Edge((i+1)%n,i,1));
        edges.push_back(make_pair(i,(i+1)%n));
        a[i]--;
        sum--;
        for(int j=i+1;j<n;j++){
            if(a[j]==0) continue;
            g[i].push_back(Edge(i,j,1));
            g[j].push_back(Edge(j,i,1));
            edges.push_back(make_pair(i,j));
            a[j]--;
            sum--;
            if(sum-1<1){
                cout<<"NO"<<endl;
                return 0;
            }
        }
    }
    cout<<"YES "<<edges.size()<<endl;
    for(auto&e:edges) cout<<e.first+1<<" "<<e.second+1<<endl;
    return 0;
}
