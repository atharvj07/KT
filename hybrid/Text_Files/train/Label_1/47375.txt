import java.util.Scanner;
import java.util.HashSet;

public class Main {
    static int n, m;
    static HashSet<Integer> banned;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        banned = new HashSet<>();
        for (int i = 0; i < m; i++) {
            banned.add(sc.nextInt() * 1000 + sc.nextInt());
        }
        System.out.println(solve());
    }

    static int solve() {
        boolean[] col = new boolean[n + 1];
        for (int x : banned) {
            col[x / 1000] = true;
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (!col[i]) {
                ans++;
            }
        }
        return ans;
    }
}
