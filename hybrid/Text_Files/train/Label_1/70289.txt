import sys
input = sys.stdin.readline

n = int(input())
parents = [int(x) for x in input().split()]
adj = [[] for i in range(n+1)]
for i in range(2, n + 1):
    adj[parents[i-1]].append(i)

expect = [0] * (n + 1)

def dfs(u, p):
    expect[u] = 1
    for v in adj[u]:
        if v == p:
            continue
        dfs(v, u)
        expect[u] += (expect[v] + 1) / len(adj[u])

dfs(1, 0)
for i in expect[1:]:
    print(i, end=" ")
