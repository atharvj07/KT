import java.util.*;

public class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();

        int[] pre = new int[n + 2];
        pre[0] = (s.charAt(0) == 'B' ? 1 : 0);
        for (int i = 1; i <= n; i++) {
            pre[i] = pre[i - 1] + (s.charAt(i - 1) == 'B' ? 1 : 0);
        }

        int[] suf = new int[n + 2];
        suf[n + 1] = (s.charAt(n - 1) == 'W' ? 1 : 0);
        for (int i = n - 1; i >= 0; i--) {
            suf[i] = suf[i + 1] + (s.charAt(i) == 'W' ? 1 : 0);
        }

        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'X') {
                ans += pre[i] * (long) suf[i + 1] % MOD * (long) k % MOD;
                ans %= MOD;
            }
        }
        System.out.println(ans);
    }
}
