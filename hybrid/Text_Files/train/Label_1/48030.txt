import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        sc.close();
        StringBuilder sb = new StringBuilder();
        while (n > 0 && m > 0) {
            sb.append('B').append('G');
            n--;
            m--;
        }
        while (n > 0) {
            sb.append('B');
            n--;
        }
        while (m > 0) {
            sb.append('G');
            m--;
        }
        System.out.println(sb);
    }
}
