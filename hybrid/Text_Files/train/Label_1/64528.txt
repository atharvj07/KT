def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def is_palindrome(n):
    return str(n) == str(n)[::-1]

def calc_primes_palindromes_ratio(p, q):
    A = p / q
    primes_count, palindromes_count = 0, 0
    n = 1
    while True:
        if is_prime(n):
            primes_count += 1
        if is_palindrome(n):
            palindromes_count += 1
        if primes_count > A * palindromes_count:
            return n - 1
        n += 1
        if n > 100000:  # Safeguard to prevent infinite loop
            break
    return "Palindromic tree is better than splay tree"

p, q = map(int, input().split())
print(calc_primes_palindromes_ratio(p, q))
