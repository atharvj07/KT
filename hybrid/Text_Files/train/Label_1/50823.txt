import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        while (true) {
            int n = in.nextInt();
            if (n == 0) break;

            int[] arr = new int[n];
            boolean hasX = false;
            for (int i = 0; i < n; i++) {
                String s = in.next();
                if (s.equals("x")) {
                    hasX = true;
                } else {
                    arr[i] = Integer.parseInt(s);
                }
            }

            if (!hasX) {
                System.out.println("none");
                continue;
            }

            int[] sortedArr = Arrays.copyOf(arr, n);
            Arrays.sort(sortedArr);

            int prev = sortedArr[0];
            boolean ambiguous = false;
            for (int i = 1; i < n; i++) {
                if (sortedArr[i] == prev) {
                    ambiguous = true;
                    break;
                }
                prev = sortedArr[i];
            }

            if (ambiguous) {
                System.out.println("ambiguous");
            } else {
                for (int i = 0; i < n; i++) {
                    if (arr[i] == 0) {
                        System.out.print(sortedArr[i - 1] + " ");
                    }
                }
                System.out.println();
            }
        }
    }
}
