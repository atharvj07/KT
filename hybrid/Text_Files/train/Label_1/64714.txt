n = int(input())
seq = list(map(int, input().split()))

def is_nice(seq):
    for i in range(n - 1):
        if (i % 2 == 0 and seq[i] >= seq[i + 1]) or (i % 2 == 1 and seq[i] <= seq[i + 1]):
            return False
    return True

def count_swaps(seq):
    bad_pairs = []
    for i in range(n - 1):
        if (i % 2 == 0 and seq[i] >= seq[i + 1]) or (i % 2 == 1 and seq[i] <= seq[i + 1]):
            bad_pairs.append(i)

    if len(bad_pairs) > 4:
        return 0

    cnt = 0
    for i in range(n):
        for j in range(i + 1, n):
            seq[i], seq[j] = seq[j], seq[i]
            if is_nice(seq):
                cnt += 1
            seq[i], seq[j] = seq[j], seq[i]
    return cnt

print(count_swaps(seq))
