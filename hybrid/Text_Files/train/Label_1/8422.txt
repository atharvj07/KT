#include <stdio.h>

const int MOD = 5;
const int BIN_SIZE = 2000; // sizeof int is usually 32 or 64 bits

int main() {
  int a[BIN_SIZE];
  a[0] = 1;
  long long n;
  scanf("%lld", &n);

  int bin_exp = 6;
  while (n >= bin_exp) {
    bin_exp *= 2;
  }

  for (int p = 1; p + p < bin_exp; p *= 2) {
    for (int i = 0; i < BIN_SIZE; i++) {
      a[i] *= a[i];
      a[i] %= MOD;
    }
  }

  int last_pow = bin_exp / 2;
  for (; last_pow > 0; last_pow /= 2) {
    if (n >= last_pow) {
      for (int i = 0; i < BIN_SIZE; i++) {
        a[i] *= a[i];
        a[i] %= MOD;
        a[i] *= a[i];
        a[i] %= MOD;
      }
      n -= last_pow;
    }
  }

  long long sum = 0;
  for (int i = 0; i < BIN_SIZE; i++) {
    sum += a[i];
  }

  printf("%lld\n", sum % MOD);

  return 0;
}
