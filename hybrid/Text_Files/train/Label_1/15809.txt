#include <iostream>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    int bal = 0;
    int l = 0, r = 0;
    int min_len = n + 1;
    for (int i = 0; i < n; i++) {
      if (s[i] == 'L')
        bal--;
      else if (s[i] == 'R')
        bal++;
      else if (s[i] == 'U')
        bal += 2;
      else
        bal -= 2;
      if (bal == 0) {
        if (i - r + 1 < min_len) {
          l = r;
          r = i;
          min_len = i - r + 1;
        }
      }
    }
    if (min_len < n + 1)
      cout << l + 1 << " " << r + 1 << endl;
    else
      cout << -1 << endl;
  }
  return 0;
}
