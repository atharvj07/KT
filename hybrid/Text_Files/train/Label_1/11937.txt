#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

const int MOD = 1e9 + 7;
const int MAX_N = 500;
const int MAX_M = 500;
const int MAX_B = 500;

int n, m, b, mod;
vector<int> a;
vector<vector<int>> dp;

int main() {
  cin >> n >> m >> b >> mod;
  a.resize(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  dp.resize(n + 1, vector<int>(m + 1, -1));

  dp[n][0] = 1;

  for (int i = n - 1; i >= 0; i--) {
    for (int j = 0; j <= m; j++) {
      if (j + a[i] <= b) {
        dp[i][j] = (dp[i][j] + dp[i + 1][j + a[i]]) % mod;
      }
      dp[i][j] = (dp[i][j] + dp[i + 1][j]) % mod;
    }
  }

  cout << dp[0][0] << endl;

  return 0;
}
