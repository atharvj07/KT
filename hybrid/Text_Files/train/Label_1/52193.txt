import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int x2 = input.nextInt();
        int minX0 = 3;
        for (int i = 2; i * i <= x2; i++) {
            if (x2 % i == 0) {
                minX0 = Math.max(minX0, i);
            }
        }
        System.out.println(minX0);
    }
}
