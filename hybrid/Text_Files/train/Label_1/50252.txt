import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int z = scanner.nextInt();
        int lastCall = 2 * n;
        int killed = 0;
        while (lastCall <= z) {
            if (lastCall % m == 0) {
                killed++;
            }
            lastCall += n;
        }
        System.out.println(killed);
    }
}
