n = int(input())
a = list(map(int, input().split()))
sum_a = sum(a)
for i in range(n):
    for j in range(sum_a):
        if j < sum(a[0:i]):
            print(' ', end='')
        else:
            if i % 2 == 0:
                if j - sum(a[0:i]) == 0:
                    print('/', end='')
                elif j - sum(a[0:i]) == a[i] - 1:
                    print('\\', end='')
                else:
                    print(' ', end='')
            else:
                if j - sum(a[0:i]) == a[i] - 1:
                    print('/', end='')
                elif j - sum(a[0:i]) == 0:
                    print('\\', end='')
                else:
                    print(' ', end='')
    print()
