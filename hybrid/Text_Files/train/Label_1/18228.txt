#include <iostream>
#include <vector>
using namespace std;

int a[10];

int main() {
    int t1, t2;
    cin >> t1 >> t2;

    vector<vector<int>> p1(t1), p2(t2);
    for (int i = 0; i < t1; ++i) {
        int x, y;
        cin >> x >> y;
        p1[i] = {x, y};
    }
    for (int i = 0; i < t2; ++i) {
        int x, y;
        cin >> x >> y;
        p2[i] = {x, y};
    }

    bool fail = false;
    int ans;
    for (int i = 0; i < t1; ++i) {
        int x = p1[i][0], y = p1[i][1];
        if (!fail) {
            ++a[x], ++a[y];
            if (a[x] == 2) fail = true, ans = x;
            if (a[y] == 2) fail = true, ans = y;
        }
    }
    for (int i = 0; i < t2; ++i) {
        int x = p2[i][0], y = p2[i][1];
        if (!fail) {
            ++a[x], ++a[y];
            if (a[x] == 2) fail = true, ans = x;
            if (a[y] == 2) fail = true, ans = y;
        }
    }

    if (fail) cout << ans << endl;
    else {
        bool all = true;
        for (int i = 1; i <= 9; ++i) {
            if (a[i] == 1) {
                fail = true;
            } else {
                all = false;
            }
        }
        if (all) cout << 0 << endl;
        else if (fail) cout << -1 << endl;
    }
}
