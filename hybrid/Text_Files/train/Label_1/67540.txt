for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    ans = 0
    for i in range(n):
        j, k = i + 1, i + 2
        while j < n and a[j] <= a[i] + 2:
            while k < n and a[k] <= a[i] + 2:
                ans += 1
                k += 1
            j += 1
    print(ans)
