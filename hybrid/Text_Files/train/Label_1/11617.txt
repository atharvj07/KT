#include<bits/stdc++.h>
using namespace std;
int main(){
    int t; cin >> t; while(t--){
        string s; cin >> s; int n = s.size(); char a[n]; bool f = false;
        for(int i = 0; i < n; i++){
            if(s[i] == '?'){
                if(i != 0 && a[i - 1] == 'c'){ a[i] = 'b'; continue; }
                if(i != 0 && a[i - 1] == 'b'){ a[i] = 'c'; continue; }
                a[i] = 'a';
            } else a[i] = s[i];

            if((i != 0 && a[i] == a[i - 1]) || (i < n - 1 && a[i] == a[i + 1])){ f = true; break; }
        }
        if(!f){ for(int i = 0; i < n; i++) cout << a[i]; } else cout << -1; cout << endl;
    }
}
