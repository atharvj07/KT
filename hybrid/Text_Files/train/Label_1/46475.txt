import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        for (int i = 0; i < n; i++) {
            int[][] grid = new int[5][5];
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    grid[j][k] = in.nextInt();
                }
            }

            boolean valid = true;

            // Check if the sum of opposite faces is 7
            for (int j = 0; j < 5; j++) {
                if (grid[j][0] + grid[j][4] != 7) {
                    valid = false;
                    break;
                }
                if (grid[0][j] + grid[4][j] != 7) {
                    valid = false;
                    break;
                }
            }

            // Check if there are duplicate numbers
            Set<Integer> numbers = new HashSet<>();
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (grid[j][k] == 0) {
                        continue;
                    }
                    if (numbers.contains(grid[j][k])) {
                        valid = false;
                        break;
                    }
                    numbers.add(grid[j][k]);
                }
                if (!valid) {
                    break;
                }
            }

            // Check if the grid is a valid cube net
            boolean[][] visited = new boolean[5][5];
            int count = 0;
            for (int j = 0; j < 5; j++) {
                for (int k = 0; k < 5; k++) {
                    if (grid[j][k] != 0 && !visited[j][k]) {
                        count++;
                        dfs(grid, visited, j, k);
                    }
                }
            }

            if (count != 1) {
                valid = false;
            }

            System.out.println(valid ? "true" : "false");
        }
    }

    private static void dfs(int[][] grid, boolean[][] visited, int x, int y) {
        visited[x][y] = true;

        if (x > 0 && grid[x - 1][y] != 0 && !visited[x - 1][y]) {
            dfs(grid, visited, x - 1, y);
        }
        if (x < 4 && grid[x + 1][y] != 0 && !visited[x + 1][y]) {
            dfs(grid, visited, x + 1, y);
        }
        if (y > 0 && grid[x][y - 1] != 0 && !visited[x][y - 1]) {
            dfs(grid, visited, x, y - 1);
        }
        if (y < 4 && grid[x][y + 1] != 0 && !visited[x][y + 1]) {
            dfs(grid, visited, x, y + 1);
        }
    }
}
