import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt(), k = input.nextInt();
        List<List<int[]>> adjList = new ArrayList<>();
        for (int i = 0; i < n; i++) adjList.add(new ArrayList<>());
        for (int i = 0; i < m; i++) {
            int a = input.nextInt() - 1, b = input.nextInt() - 1, l = input.nextInt();
            adjList.get(a).add(new int[]{b, l});
            adjList.get(b).add(new int[]{a, l});
        }
        Set<Integer> malls = new HashSet<>();
        for (int i = 0; i < k; i++) malls.add(input.nextInt() - 1);
        int[] dist = dijkstra(0, adjList);
        int minDist = Integer.MAX_VALUE;
        for (int mall: malls) minDist = Math.min(minDist, dist[mall]);
        System.out.println(minDist == Integer.MAX_VALUE ? -1 : minDist);
    }

    public static int[] dijkstra(int s, List<List<int[]>> adjList) {
        int n = adjList.size();
        int[] dist = new int[n];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[s] = 0;
        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> a[1] - b[1]);
        pq.offer(new int[]{s, 0});
        while (!pq.isEmpty()) {
            int[] curr = pq.poll();
            int u = curr[0], d = curr[1];
            if (d > dist[u]) continue;
            for (int[] next: adjList.get(u)) {
                int v = next[0], l = next[1];
                if (dist[v] > d + l) {
                    dist[v] = d + l;
                    pq.offer(new int[]{v, dist[v]});
                }
            }
        }
        return dist;
    }
}
