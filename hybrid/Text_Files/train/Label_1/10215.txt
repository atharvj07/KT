#include <vector>
#include <iostream>
#include <limits>
using namespace std;

int main() {
    int n, k; cin >> n >> k;
    vector<int> d(n), s(n);
    for (int i = 0; i < n - 1; i++) cin >> d[i];
    for (int i = 0; i < n - 1; i++) cin >> s[i];
    vector<long long> pre(n, 0), suf(n, 0);
    pre[0] = d[0];
    for (int i = 1; i < n - 1; i++) {
        pre[i] = pre[i - 1] + d[i];
    }
    suf[n - 2] = s[n - 2];
    for (int i = n - 3; i >= 0; i--) {
        suf[i] = suf[i + 1] + s[i];
    }
    long long ans = pre[n - 2] + suf[0];
    for (int l = 0; l < n - 1; l++) {
        long long cur = pre[l];
        long long need = ((cur + d[l] - 1) / d[l]) - 1;
        if (need * k < suf[l + 1]) {
            cur += suf[l + 1] - need * k;
        }
        cur += d[l];
        cur += pre[n - 2] - pre[l];
        cur += suf[0];
        ans = min(ans, cur);
    }
    cout << ans << endl;
    return 0;
}
