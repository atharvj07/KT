from collections import deque

def solve(n, k, coke_types):
    if n in coke_types:
        return 1
    dp = [-1] * (n + 1)
    dp[0] = 0
    for i in range(n + 1):
        if dp[i] == -1:
            continue
        for c in coke_types:
            if i + c <= n and (dp[i + c] == -1 or dp[i + c] > dp[i] + 1):
                dp[i + c] = dp[i] + 1
    return dp[n]

def main():
    n, k = map(int, input().split())
    coke_types = list(map(int, input().split()))
    print(solve(n, k, coke_types))

main()
