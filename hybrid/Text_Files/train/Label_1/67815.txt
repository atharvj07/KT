from collections import defaultdict

def find_simple_cycle(n, m, k, edges):
    graph = defaultdict(list)
    for edge in edges:
        a, b = edge
        graph[a].append(b)
        graph[b].append(a)

    visited = [False] * (n + 1)
    parent = [-1] * (n + 1)
    cycle = []

    def dfs(v, p):
        visited[v] = True
        for u in graph[v]:
            if u == p:
                continue
            if visited[u]:
                if not cycle:
                    cur = v
                    cycle.append(cur)
                    while cur != u:
                        cur = parent[cur]
                        cycle.append(cur)
                continue
            if not cycle:
                parent[u] = v
                dfs(u, v)

    for node in range(1, n + 1):
        if not visited[node]:
            dfs(node, -1)
        if cycle:
            break

    print(len(cycle))
    print(' '.join(map(str, cycle)))

n, m, k = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(m)]
find_simple_cycle(n, m, k, edges)
