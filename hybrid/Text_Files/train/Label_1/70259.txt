import sys

mod=998244353

def powm(x,p):
    if p==0: return 1
    if p==1: return x
    val=powm(x,p//2)
    val=val*val%mod
    if p%2!=0: val=val*x%mod
    return val%mod

inp=sys.stdin.readlines()
n,m=map(int,inp[0].split())
dist=[[int(x) for x in l.split()] for l in inp[1:]]
ways=[0]*(n+1)
for l in dist:
    for x in l:
        ways[x]+=1

ex=0
for i in range(1,n+1):
    if ways[i]>0:
        ex+=powm(ways[i],n-1)
        ex%=mod
print(ex)
