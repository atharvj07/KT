n = int(input())
f = list(map(int, input().split()))

# To store friends who don't know whom to give gifts
unknown = [i for i in range(1, n+1) if f[i-1] == 0]

# To store friends who haven't received gifts yet
not_received = set(range(1, n+1)) - set(f)

# Matching unknown givers to receivers
for i in range(n):
    if f[i] == 0:
        if unknown[0] in not_received:
            # Ensure giver doesn't receive their own gift
            for ur in not_received:
                if ur != unknown[0]:
                    f[i] = ur
                    not_received.remove(ur)
                    break
        else:
            f[i] = not_received.pop()
        unknown.pop(0)

print(*f)
