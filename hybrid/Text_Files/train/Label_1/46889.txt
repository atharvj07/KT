import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            int n = in.nextInt(), m = in.nextInt(), k = in.nextInt();
            List<Integer>[] adj = new List[n];
            for (int j = 0; j < n; j++) adj[j] = new ArrayList<>();
            for (int j = 0; j < m; j++) {
                int u = in.nextInt() - 1, v = in.nextInt() - 1;
                adj[u].add(v);
                adj[v].add(u);
            }
            boolean flag = false;
            for (int j = 0; j < n; j++) {
                int cnt = 0;
                for (int u : adj[j]) if (u >= j) cnt++;
                if (cnt >= k) {
                    System.out.println("1 " + (cnt + 1));
                    for (int u : adj[j]) if (u >= j) System.out.print((u + 1) + " ");
                    System.out.println();
                    flag = true;
                    break;
                }
            }
            if (!flag) {
                for (int j = 0; j < n; j++) {
                    Set<Integer> set = new HashSet<>();
                    for (int u : adj[j]) if (u >= j) set.add(u);
                    if (set.size() >= k) {
                        System.out.println("2 " + set.size());
                        for (int u : set) System.out.print((u + 1) + " ");
                        System.out.println();
                        flag = true;
                        break;
                    }
                }
            }
            if (!flag) System.out.println("-1");
        }
    }
}
