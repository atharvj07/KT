import heapq
import sys

input = sys.stdin.readline

n, k = map(int, input().split())
videos = []

for i in range(n):
    s_i, m_i = map(int, input().split())
    videos.append((s_i, m_i))

servers = []

for i in range(k):
    heapq.heappush(servers, 0)

e = []

for s_i, m_i in videos:
    time = heapq.heappop(servers)
    e.append(s_i + max(0, time - s_i + m_i))
    heapq.heappush(servers, s_i + m_i)

print(*e)
