from sys import stdin
input = stdin.readline

def modpow(base, exp):
    result = 1
    while exp > 0:
        if exp % 2 == 1:
            result *= base
            result %= 1000000007
        exp //= 2
        base *= base
        base %= 1000000007
    return result

N = int(input())
desk = list(range(1,2*N+1))
for i in range(N):
    from_, to_ = map(int, input().split())
    desk[from_-1], desk[to_-1] = desk[to_-1], desk[from_-1]

result = 1
for i in range(N):
    for j in range(2*N):
        if desk[j] == i+1:
            break
    else:
        print(0)
        exit()
    while j > i:
        desk[j], desk[j-1] = desk[j-1], desk[j]
        j -= 1
        result *= j+1
        result %= 1000000007

print(result)
