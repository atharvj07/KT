import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            int c = sc.nextInt();
            edges.add(new Edge(l, r, c));
        }
        int[] dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[1] = 0;
        for (int i = 0; i < n - 1; i++) {
            for (Edge edge : edges) {
                int l = edge.l;
                int r = edge.r;
                int c = edge.c;
                if (dist[l] + c < dist[r]) {
                    dist[r] = dist[l] + c;
                }
            }
        }
        int ans = dist[n];
        if (ans == Integer.MAX_VALUE) {
            ans = -1;
        }
        System.out.println(ans);
    }

    static class Edge {
        int l;
        int r;
        int c;

        public Edge(int l, int r, int c) {
            this.l = l;
            this.r = r;
            this.c = c;
        }
    }
}
