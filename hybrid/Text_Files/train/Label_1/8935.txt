#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, s;
    cin >> n >> s;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    sort(a.begin(), a.end());
    if (a[0] <= s / 2) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
