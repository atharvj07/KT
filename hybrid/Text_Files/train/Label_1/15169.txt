#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int y;
  cin >> y;
  while (true) {
    y++;
    vector<int> digits;
    int temp = y;
    while (temp > 0) {
      digits.push_back(temp % 10);
      temp /= 10;
    }
    sort(digits.begin(), digits.end());
    if (unique(digits.begin(), digits.end()) == digits.end()) {
      cout << y << endl;
      return 0;
    }
  }
}
