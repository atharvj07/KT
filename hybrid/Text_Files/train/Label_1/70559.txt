from functools import reduce
from collections import Counter

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = []
    a.sort()
    for i in reversed(list(Counter(a).keys())):
        for j in range(Counter(a)[i]):
            b.append(i)
            a.remove(i)
    print(*b)
