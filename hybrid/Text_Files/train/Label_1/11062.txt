#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0), cout.tie(0);

  int n, q, x, y;
  cin >> n >> q;
  vector<int> a(n);
  for (int &v : a) cin >> v;
  vector<int> dp(n);
  for (int i = 0; i < n - 1; i++) {
    if ((a[i] & a[i + 1]) == 0) {
      dp[i + 1] = i + 1;
    } else if (i != 0) {
      dp[i + 1] = dp[i];
    }
  }
  while (q--) {
    cin >> x >> y;
    --x, --y;
    if (dp[y] >= x)
      cout << "Shi\n";
    else
      cout << "Fou\n";
  }
}
