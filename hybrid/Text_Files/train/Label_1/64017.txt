n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
bad = [True for _ in range(10)]
for i in range(0, k):
    if b[i] > 0:
        bad[b[i]] = False
num = 0
for x in a:
    for d in range(10):
        if bad[d]:
            continue
        if (x % 10 * pow(10, i, x) + d) % x == 0:
            num += pow(10, i, x)
            break
print(num % (10 ** 9 + 7))
