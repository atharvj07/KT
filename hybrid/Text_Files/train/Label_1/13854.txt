#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int n;
vector<int> a;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  cin >> n;
  a.resize(n);
  for (int &x : a) cin >> x;

  vector<ll> dp(n + 1, LLONG_MAX);
  dp[0] = 0;
  for (int i = 0; i < n; i++) {
    for (int j = n - 1 - i + 1; j > 0; j--) {
      dp[j] = min(dp[j], dp[j - 1] + max(a[i], a[i + j - 1]) - min(a[i], a[i + j - 1]));
    }
  }

  for (int k = 1; k <= n / 2; k++) {
    cout << dp[k] << ' ';
  }
  cout << '\n';

  return 0;
}
