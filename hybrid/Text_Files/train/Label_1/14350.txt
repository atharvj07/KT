#include <bits/stdc++.h>

#define pb push_back
#define all(x) x.begin(), x.end()
#define rall(x) x.rbegin(), x.rend()
#define clr(x, v) memset(x, v, sizeof(x))
#define sz(x) (int)(x.size())
using namespace std;
typedef vector<int> vi;
typedef pair<int, int> ii;
typedef long long ll;
typedef vector<ll> vll;

int n, m;
const int MAXN = 100000 + 5;
vi adj[MAXN];
int t[MAXN];
ii s[MAXN];
int dist[MAXN];

int f(int u, int p) {
	int c = 0;
	for (int to : adj[u]) {
		if (to != p) {
			if (s[u].first == s[to].first || s[u].second == s[to].first || s[u].first == s[to].second || s[u].second == s[to].second) {
				dist[s[u].first] ^= u;
				dist[s[u].second] ^= u;
			} else {
				dist[s[u].first] ^= u;
				dist[s[u].second] ^= u;
				s[to] = {s[u].first, s[u].second};
			}
			c += f(to, u) + t[to];
		}
	}
	return c;
}

int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	int tt;
	cin >> tt;
	while (tt--) {
		cin >> n;
		for (int i = 0; i < 2 * n; i++) {
			adj[i].clear();
			t[i] = 0;
			s[i] = {0, 0};
			dist[i] = 0;
		}
		for (int i = 0; i < 2 * n - 1; i++) {
			int u, v, c;
			cin >> u >> v >> c;
			u--, v--;
			adj[u].pb(v);
			adj[v].pb(u);
			t[v] = c;
			t[u] = c;
		}
		s[0] = {0, 1};
		int g = f(0, 0), b = g;
		int x1 = -1, x2 = -1;
		for (int i = 0; i < 2 * n; i++) {
			if (dist[i]) {
				if (x1 == -1)
					x1 = i;
				else
					x2 = i;
			}
		}
		swap(s[x1], s[x2]);
		b = f(0, 0);
		cout << g << " " << b << "\n";
	}
	return 0;
}
