#include <map>
#include <set>
#include <vector>
#include <string>
#include <iostream>
#include <algorithm>
#include <cmath>
using namespace std;

int main() {
	int h, m, s, t1, t2;
	cin >> h >> m >> s >> t1 >> t2;
	if(t1 != t2) {
		vector<int> v = {t1, t1 + 1, t1 + 5, t1 + 10};
		int mn = min(t1, t2);
		int mx = max(t1, t2);
		bool ck = 0;
		for(int i = mn;i < mx;i++) {
			if(binary_search(v.begin(), v.end(), i)) {
				ck = 1;
				break;
			}
		}
		if(ck) cout << "NO" << endl;
		else cout << "YES" << endl;
	} else {
		cout << "YES" << endl;
	}
}
