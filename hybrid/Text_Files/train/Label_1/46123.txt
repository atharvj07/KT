import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] stairs = new int[n + 1];
        for (int i = 0; i < n; i++) stairs[i] = in.nextInt();
        int m = in.nextInt();
        int[] bottom = new int[m + 1];
        for (int i = 1; i <= m; i++) {
            int w = in.nextInt();
            int h = in.nextInt();
            bottom[i] = stairs[Math.min(w, n)] + h;
            for (int j = 1; j < w; j++) stairs[j] = Math.max(stairs[j], bottom[i] - h);
        }
        for (int i = 1; i <= m; i++) System.out.println(bottom[i]);
    }
}
