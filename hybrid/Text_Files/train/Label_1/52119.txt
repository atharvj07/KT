import java.util.*;
class Main {
    static char[] s1, s2;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int len = sc.nextInt();
        s1 = sc.next().toCharArray();
        s2 = sc.next().toCharArray();
        int ans = 0;
        ArrayList<int[]> opts = new ArrayList<>();
        for (int i = 0; i < len; i++) {
            if (s1[i] != s2[i]) {
                for (int j = i + 1; j < len; j++) {
                    if (s1[i] == s2[j] && s1[j] == s2[i]) {
                        opts.add(new int[]{i + 1, j + 1});
                        s1[i] = s2[i];
                        s2[j] = s2[i];
                        ans++;
                        break;
                    }
                }
                for (int j = i + 1; j < len; j++) {
                    if (s1[i] == s2[j]) {
                        opts.add(new int[]{j + 1, j + 1});
                        char tmp = s2[i];
                        s2[i] = s2[j];
                        s2[j] = tmp;
                        ans++;
                        break;
                    }
                }
            }
        }
        for (int i = 0; i < len; i++)
            if (s1[i] != s2[i]) {
                System.out.println(-1);
                return;
            }
        System.out.println(ans);
        for (int[] opt : opts)
            System.out.println(opt[0] + " " + opt[1]);
    }
}
