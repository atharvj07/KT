import math

k, n1, n2, n3, t1, t2, t3 = map(int, input().split())

washing_machines_busy_time = math.ceil(k / n1) * t1
drying_machines_busy_time = math.ceil((k + n1 - 1) / n2) * t2
folding_machines_busy_time = math.ceil((k + n1 + n2 - 2) / n3) * t3

print(washing_machines_busy_time + drying_machines_busy_time + folding_machines_busy_time)
