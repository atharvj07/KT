from itertools import combinations
n, k = map(int, input().split())
cards = [input() for _ in range(n)]
same = [0]*k
diff = [0]*k
for card in cards:
    for i in range(k):
        if card[i] == cards[0][i]: same[i] += 1
        else: diff[i] += 1
ans = 0
for c in combinations(range(n), 3):
    f = True
    for i in range(k):
        f &= (same[i] >= 3 or diff[i] >= 3)
    ans += f
print(ans)
