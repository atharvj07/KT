#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &i : a) cin >> i;
  bool ev = 1;
  for (int i = 1; i < n; i++, ev = !ev) {
    if (ev) {
      if (a[i] < a[i - 1]) {
        swap(a[i], a[i - 1]);
      }
    } else {
      if (a[i] > a[i - 1]) {
        swap(a[i], a[i - 1]);
      }
    }
  }
  for (int i = 1; i < n; i++) {
    if (a[i] < a[i - 1]) {
      cout << "Impossible" << endl;
      return 0;
    }
  }
  for (int i = 0; i < n; i++) cout << a[i] << ' ';
  cout << endl;
  return 0;
}
