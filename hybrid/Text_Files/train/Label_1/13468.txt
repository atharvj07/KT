#include <iostream>
#include <algorithm>
using namespace std;

string a;
int k, n;
char sol[18];

int main() {
  cin >> a >> k;
  n = a.size();
  for (int i = 0; i < n and k; i++) {
    int mx = i;
    for (int j = i + 1; j < n; j++) {
      if (a[j] > a[mx]) mx = j;
    }
    if (mx > i) {
      swap(a[mx], a[i]);
      k--;
    }
  }
  for (int i = 0; i < n; i++) sol[i] = a[i];
  while (k--) {
    int i;
    for (i = n - 2; i >= 0 and a[i] >= a[i + 1]; i--)
      ;
    if (i < 0) break;
    swap(a[i], a[i + 1]);
    int l = i + 1, r = n - 1, m;
    while (l < r) {
      m = (l + r) / 2;
      if (a[m] > a[i])
        l = m + 1;
      else
        r = m - 1;
    }
    for (int j = r; j > i; j--)
      sol[n - j + i - 1] = a[j];
  }
  for (int i = 0; i < n; i++) sol[i] = a[i];
  for (int i = n - 1; i >= 0; i--) cout << sol[i];
  cout << endl;
  return 0;
}
