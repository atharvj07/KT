import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] w = new long[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.nextInt();
        }
        Arrays.sort(w);
        int ans = 0;
        long sum = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (sum + w[i] <= sum * 2) {
                sum += w[i];
            } else {
                ans++;
                sum = w[i];
            }
        }
        System.out.println(ans);
    }
}
