#include <bits/stdc++.h>

using namespace std;

int main() {
  int t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    int ans = 0;
    int last = 0, x;
    for (int i = 0; i < n; i++) {
      int t;
      cin >> t >> x;
      if (t >= last) {
        ans++;
        last = t;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
