#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<int> p(m);
    for (int i = 0; i < m; i++) { cin >> p[i]; }
    vector<bool> vis(n + 1, false);
    for (int x : p) { vis[x] = true; }
    vector<int> ans;
    for (int i = 1; i <= n; i++) {
      if (!vis[i]) { ans.push_back(a[i - 1]); }
    }
    sort(ans.begin(), ans.end());
    int j = 0;
    bool ok = true;
    for (int i = 1; i <= n; i++) {
      if (vis[i]) {
        if (a[i - 1] > ans[j]) { ok = false; }
        j++;
      }
    }
    if (ok) { cout << "YES" << endl; } else { cout << "NO" << endl; }
  }
  return 0;
}
