#include <bits/stdc++.h>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n, m; cin >> n >> m;
        vector<int> a(n), p(m);
        for (auto &x : a) cin >> x;
        for (auto &x : p) cin >> x;

        if (m == n - 1) { cout << "YES\n"; continue; }

        vector<bool> vis(n, false);
        for (auto &x : p) vis[x] = true;

        bool bad = false;
        for (int i = 1; i < n; i++) {
            if (a[i] < a[i - 1] && vis[i - 1] && !vis[i]) bad = true;
        }

        cout << (bad ? "NO\n" : "YES\n");
    }
}
