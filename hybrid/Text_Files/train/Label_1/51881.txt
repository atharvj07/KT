import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        boolean[] old = new boolean[26];
        boolean[] need = new boolean[26];
        boolean[] exist = new boolean[26];
        List<String> strings = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            int[] cnt = new int[26];
            for (int j = 0; j < s.length(); j++) {
                cnt[s.charAt(j) - 'a']++;
            }
            boolean[] cur = new boolean[26];
            String t = sc.next();
            for (int j = 0; j < t.length(); j++) {
                int c = cnt[t.charAt(j) - 'a']--;
                if (c > 0) {
                    cur[t.charAt(j) - 'a'] = true;
                }
            }
            if (Arrays.equals(cur, old)) {
                need = Arrays.copyOf(need, need.length);
                for (int j = 0; j < t.length(); j++) {
                    need[t.charAt(j) - 'a'] |= exist[s.charAt(j) - 'a'];
                }
            } else {
                if (old[0] || need[0]) {
                    System.out.println("NO");
                    return;
                }
                old = cur;
                exist = Arrays.copyOf(cur, cur.length);
                strings.add(s);
                strings.add(t);
            }
        }
        System.out.println("YES");
        System.out.println(strings.get(0));
        System.out.println(strings.get(1));
    }
}
