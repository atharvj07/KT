import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int res = 0;
        for (long i = 2; i * i <= n; i++) {
            while (n % i == 0) {
                n /= i;
                res++;
            }
        }
        if (n > 1)
            res++;
        System.out.print(res);
    }
}
