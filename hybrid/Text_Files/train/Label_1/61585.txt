import sys
n = int(input())
a = list(map(int, input().split()))
if n == 1:
    print(a[0])
    sys.exit()
if a[0] > a[1]:
    print("Impossible")
    sys.exit()
for i in range(2, n, 2):
    if a[i] < a[i - 1] or a[i] < a[i + 1]:
        print("Impossible")
        sys.exit()
for i in range(3, n, 2):
    if a[i] > a[i - 1] or a[i] > a[i + 1]:
        print("Impossible")
        sys.exit()
print(*a)
