#include <bits/stdc++.h>
using namespace std;

int main() {
  int tc;
  cin >> tc;
  while (tc--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    int mn = *min_element(a.begin(), a.end());
    int mnidx = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] == mn) {
        mnidx = i;
        break;
      }
    }

    int mx = *max_element(a.begin(), a.end());
    int mxidx = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] == mx) {
        mxidx = i;
        break;
      }
    }

    if (mnidx > mxidx) swap(mnidx, mxidx);

    int cnt = 0;
    if (mnidx != 0) cnt++;
    if (mxidx != n - 1) cnt++;

    cout << cnt << endl;
  }
  return 0;
}
