import sys
input = sys.stdin.readline
r, b = map(int, input().split())
rebels = []
bases = []
for i in range(r):
    x, y = map(int, input().split())
    rebels.append((x, y))
for i in range(b):
    x, y = map(int, input().split())
    bases.append((x, y))

def is_intersect(p1, p2, q1, q2):
    x1, y1 = p1
    x2, y2 = p2
    x3, y3 = q1
    x4, y4 = q2
    if (x1 <= x3 and x4 <= x2) or (x3 <= x1 and x2 <= x4):
        if (y1 <= y3 and y4 <= y2) or (y3 <= y1 and y2 <= y4):
            return True
    return False

def is_good():
    for i in range(r):
        for j in range(i + 1, r):
            for k in range(b):
                for l in range(k + 1, b):
                    if is_intersect(rebels[i], rebels[j], bases[k], bases[l]):
                        return False
    return True

if r != b or not is_good():
    print("No")
else:
    print("Yes")
