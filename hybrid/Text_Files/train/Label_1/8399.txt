#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
int n; ll a[N];
bool chk(ll g) {
    for(int i = 0; i < n; i++) if(a[i] % g) return false;
    return true;
}
int main() {
    int t; cin >> t;
    while(t--) {
        cin >> n;
        for(int i = 0; i < n; i++) {
            cin >> a[i];
        }
        ll g = a[0];
        for(int i = 1; i < n; i++) g = __gcd(g, a[i]);
        vector<ll> ans;
        for(ll i = 1; i * i <= g; i++) {
            if(g % i == 0) {
                if(chk(i)) ans.push_back(i);
                if(i != g / i && chk(g / i)) ans.push_back(g / i);
            }
        }
        sort(ans.begin(), ans.end());
        for(int x : ans) cout << x << " ";
        cout << "\n";
    }
}
