from bisect import bisect_left
n,x=map(int,input().split());a=list(map(int,input().split()))
b=[0 for i in range(x+1)];c=[0 for i in range(x+1)]
for i in a:b[i]+=1
for i in range(1,x+1):c[i]=c[i-1]+b[i]
ans=0
for i in range(x+1):
    l=i*(c[i]-c[i-1]);r=c[x]-c[i]+i*(c[i]-c[i-1]-1)
    ans+=max(0,r-l+1-bisect_left(c,l,i,c[x]))
print(ans)
