#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 105;

ll a[maxn], b[maxn];
double dp[maxn][maxn];
int n, k;

int main() {
    cin >> n >> k;
    for(int i = 1; i <= n; ++i) cin >> a[i];
    for(int i = 1; i <= n; ++i) cin >> b[i];
    
    for(int i = 0; i <= n; ++i) {
        dp[i][0] = 0.0;
        for(int j = 1; j <= n; ++j) {
            dp[i][j] = -1e9;
        }
    }
    
    dp[0][0] = 0.0;
    for(int i = 1; i <= n; ++i) {
        for(int j = 1; j <= i; ++j) {
            double ratio = (double)a[i] / b[i];
            dp[i][j] = max(dp[i][j], dp[i - 1][j]);
            if(j >= 2) {
                dp[i][j] = max(dp[i][j], dp[i - 1][j - 2] + ratio);
            }
            dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] + ratio);
        }
    }

    if(dp[n][1] < 0.0) cout << -1 << endl;
    else{
        double ans = dp[n][1] * k;
        cout << int(ans) << endl;
    }
    
    return 0;
}
