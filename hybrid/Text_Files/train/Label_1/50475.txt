import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[][] a = new long[3][n];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < 3; j++) {
                    a[j][i] = sc.nextLong();
                }
            }
            long[] b = new long[3];
            for (int i = 0; i < 3; i++) {
                Arrays.sort(a[i]);
                b[i] = a[i][n / 2];
            }
            System.out.println(b[0] + " " + b[1] + " " + b[2]);
        }
    }
}
