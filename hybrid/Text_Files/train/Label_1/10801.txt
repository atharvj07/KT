#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define pii pair<int,int>
const int inf = 1e9;
const int mod = 1e9+7;

int n,m;
vector<vi>adj;
vi color;

bool dfs(int x, int c){
	if(color[x] != -1){
		if(color[x] == c)return false;
		return true;
	}
	color[x] = c;
	for(auto v : adj[x]){
		if(dfs(v,c^1) == false){
			return false;
		}
	}
	return true;
}

void solve(){
	cin >> n >> m;
	adj.resize(n+5);
	for(int i=0; i<m; i++){
		int x,y;
		cin >> x >> y;
		adj[x].push_back(y);
	}
	color.assign(n+5,-1);
	int cnt = 0;
	vi ans;
	for(int i=1; i<=n; i++){
		if(color[i] == -1){
			bool ok = dfs(i,cnt);
			if(ok == false){
				cout << -1 << endl;
				return;
			}
			cnt++;
		}
	}
	cout << cnt << endl;
	for(int i=1; i<=n; i++){
		for(auto v : adj[i]){
			if(color[i] != color[v]){
				ans.push_back(1);
			}
			else ans.push_back(0);
		}
	}
	for(int i : ans)cout << i << " ";
	cout << endl;
}

int main(){
	solve();
	return 0;
}
