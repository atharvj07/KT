import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n+1];
        for (int i= 0; i < n; i++) a[i] = sc.nextInt();
        int ans = 0;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            double sq = Math.sqrt(a[i]);
            if((int)sq == sq) cnt++;
        }
        if(cnt == n/2) {
            System.out.println(0);
            return;
        }
        boolean[] vis = new boolean[n];
        while(true) {
            cnt = 0;
            for (int i = 0; i < n; i++) {
                if(vis[i]) continue;
                double sq = Math.sqrt(a[i]);
                if((int)sq == sq) cnt++;
            }
            if(cnt == n/2) break;
            int idx = -1;
            for (int i = 0; i < n; i++) {
                if(vis[i]) continue;
                double sq = Math.sqrt(a[i]);
                if((int)sq == sq) continue;
                if(idx == -1) idx = i;
                else if (a[i] < a[idx]) idx = i;
            }
            ans++;
            vis[idx] = true;
            a[idx]++;
        }
        System.out.println(ans);
    }
}
