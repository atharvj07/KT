#include <bits/stdc++.h>
using namespace std;

int n, k;
vector<int> a;

int main() {
  cin >> n >> k;
  a.push_back(0);
  for (int i = 0, x; i < n; ++i) cin >> x, a.push_back(x);
  a.push_back(0);
  sort(a.begin(), a.end());
  int ans = 0;
  for (int i = 1; i + k < a.size(); ++i) {
    ans += a[i + k - 1] - a[i];
    i += k - 1;
  }
  cout << ans << endl;
}
