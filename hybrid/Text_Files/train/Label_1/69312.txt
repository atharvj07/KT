def can_weight(w, m):
    while m > 0:
        if m % w == 0:
            m //= w
        elif (m - 1) % w == 0:
            m = (m - 1) // w
        elif (m + 1) % w == 0:
            m = (m + 1) // w
        else:
            return "NO"
    return "YES"

w, m = map(int, input().split())
print(can_weight(w, m))
