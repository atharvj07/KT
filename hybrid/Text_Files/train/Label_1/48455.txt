import java.util.Scanner;

public class Main {
    public static int n;
    public static int k;
    public static int[] a;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextInt();
        }
        System.out.println(solve());
    }

    public static int solve() {
        int ans = 0;
        for (int i = 0; i < k; i++) {
            int team = a[i];
            int pos = 1;
            int bracket = n;
            while (true) {
                if ((team & 1) == 1) {
                    if (bracket == 1) {
                        ans++;
                    } else {
                        team >>= 1;
                        bracket--;
                        pos = pos << 1 | 1;
                    }
                } else {
                    team >>= 1;
                    bracket--;
                    pos >>= 1;
                }
                if (team == 0) {
                    break;
                }
            }
        }
        return ans;
    }
}
