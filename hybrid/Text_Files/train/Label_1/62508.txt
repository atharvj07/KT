import math
t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    if n % k != 0:
        print(n % k + 1)
    else:
        total_steps = int(math.log(n, k))
        if n % (k ** total_steps) == 0:
            print(total_steps)
        else:
            print(total_steps + 1)
