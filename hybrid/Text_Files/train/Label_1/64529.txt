import sys
p, q = map(int, sys.stdin.readline().strip().split())
a = p / q
max_n = 1
pi = [True] * (max_n + 1)
for i in range(2, int(max_n ** 0.5) + 1):
    if pi[i] == True:
        for j in range(i * i, max_n + 1, i):
            pi[j] = False
prime = [i for i in range(2, max_n + 1) if pi[i]]
rub = []
for i in range(1, max_n + 1):
    num = i
    rev = 0
    while num > 0:
        rev = rev * 10 + num % 10
        num = num // 10
    if rev == i:
        rub.append(i)
while len(prime) > len(rub) * a:
    max_n *= 2
    pi = [True] * (max_n + 1)
    for i in range(2, int(max_n ** 0.5) + 1):
        if pi[i] == True:
            for j in range(i * i, max_n + 1, i):
                pi[j] = False
    prime = [i for i in range(2, max_n + 1) if pi[i]]
    rub = []
    for i in range(1, max_n + 1):
        num = i
        rev = 0
        while num > 0:
            rev = rev * 10 + num % 10
            num = num // 10
        if rev == i:
            rub.append(i)
if len(prime) <= len(rub) * a:
    print(max_n)
else:
    print("Palindromic tree is better than splay tree")
