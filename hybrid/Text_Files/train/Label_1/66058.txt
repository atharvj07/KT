import sys
from collections import defaultdict

n, m = map(int, input().split())
F = []
for _ in range(n):
    F.append(set([int(v) for v in input().split(' ')[1:]]))
P = []
for _ in range(m):
    vs = input().split(' ')
    vs[2:] = [int(v) for v in vs[2:]]
    P.append(set(vs[2:]))
    vs[2:] = [int(v) for v in vs[2:]]
    P.append(set(vs[2:]))

ans = []
max_cnt = -1
P_s = defaultdict(list)
for i in range(m*2):
    P_s[len(P[i])].append(i)

for sz in range(9, 0, -1):
    if max_cnt == n:
        break
    if len(P_s[sz]) < 2:
        continue
    for i in P_s[sz]:
        for j in P_s[sz]:
            if i==j:
                continue
            cnt = 0
            for f in F:
                if len(f&P[i]) > 0 and len(f&P[j]) > 0:
                    cnt += 1
            if cnt > max_cnt:
                max_cnt = cnt
                ans = [i, j]
            elif cnt == max_cnt:
                if ans[0] + ans[1] > i + j:
                    ans = [i, j]

print(*ans)
