#include <bits/stdc++.h>
#define ll long long
const ll mod = 1e9 + 7;
using namespace std;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<ll> v(n);
    for (auto &x : v) cin >> x;
    sort(v.begin(), v.end());
    ll ans = 0;
    for (int i = 0; i < n; i++) {
      if (v[i] == v[i + 1]) ans = max(ans, v[i] + v[i + 1]);
    }
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if (v[j] % v[i] || v[i] % v[j]) ans = max(ans, v[i] + v[j]);
      }
    }
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        for (int k = j + 1; k < n; k++) {
          if (!(v[i] % v[j] || v[j] % v[i] || v[i] % v[k] || v[k] % v[i] ||
                v[j] % v[k] || v[k] % v[j]))
            ans = max(ans, v[i] + v[j] + v[k]);
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
