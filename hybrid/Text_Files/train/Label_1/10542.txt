#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<vector<int>> dp(n + 1, vector<int>(n + 1, 0));

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            dp[i][j] = dp[i - 1][j];
            if (i - j >= 0) { dp[i][j] = max(dp[i][j], dp[i - j][j - 1] + a[i - j]); }
        }
    }
    for (int i = 1; i <= n; i++) { cout << dp[n][i] << ' '; }
    cout << '\n';
    return 0;
}
