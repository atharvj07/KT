n = int(input())
a = list(map(int, input().split()))
L = [0] * n
R = [0] * n
for i in range(1, n-1):
    L[i] = L[i-1]
    if a[i-1] > a[i]:
        L[i] += 1
for i in range(n-2, 0, -1):
    R[i] = R[i+1]
    if a[i+1] > a[i]:
        R[i] += 1
ans = 0
for i in range(n):
    if L[i] and R[i]:
        ans += L[i] * R[i]
print(ans)
