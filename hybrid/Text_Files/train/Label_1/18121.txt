#include <bits/stdc++.h>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n, ans = 0;
        std::cin >> n;
        while (n--) {
            int k;
            std::vector<int> v;
            std::cin >> k;
            for (int i = 0; i < k; i++) {
                int x;
                std::cin >> x;
                v.push_back(x);
            }
            ans = std::max(ans, *std::max_element(v.begin(), v.end()) + 1);
        }
        std::cout << ans << '\n';
    }
    return 0;
}
