#include <bits/stdc++.h>
using namespace std;

#define all(v) (v).begin(), (v).end()
#define pb push_back

const int MAXN = 100, MAXX = 1e9;
const int MAXK = 5000;

int n, m, a[MAXN], g[MAXN][MAXN], ord[MAXN];
vector<pair<int, int>> p;
map<pair<int, int>, int> mp;
vector<int> col[MAXN];

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);

    cin >> n >> m;
    for(int i = 1; i <= m; ++i) {
        int u, v; cin >> u >> v;
        g[u][v] = g[v][u] = 1;
    }
    queue<int> q; q.push(1);
    while(!q.empty()) {
        int u = q.front(); q.pop();
        for(int i = 1; i <= n; ++i) {
            if(g[u][i] && !ord[i]) {
                ord[i] = ord[u] + 1;
                q.push(i);
            }
        }
    }
    for(int i = 1; i <= n; ++i) {
        if(!ord[i]) ord[i] = -1;
        p.pb({ord[i], i});
    }
    sort(all(p));
    int k = 0;
    for(int i = 0; i < n; ++i) {
        col[p[i].second].pb(k++);
        a[p[i].second] = 1;
    }
    for(int i = 1; i <= n; ++i) {
        if(a[1] == 4) break;
        for(int j = 1; j <= n; ++j) {
            if(g[i][j] && i != j && a[i] < 4) {
                col[i].pb(k++);
                a[i]++;
            }
        }
    }
    for(int i = 1; i <= n; ++i) {
        cout << a[i] << '\n';
        for(int j = 0; j < a[i]; ++j) {
            cout << col[i][j] / 5 + 1 << ' ' << col[i][j] % 5 + 1 << '\n';
        }
        cout << '\n';
    }

    return 0;
}
