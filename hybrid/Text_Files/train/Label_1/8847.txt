#include<bits/stdc++.h>
#define ll long long
using namespace std;
string s;
int n,q;
ll p[10000005];
int main(){
	cin>>s>>q;
	s=' '+s;
	n=s.size();
	while(q--){
		int l,r,k;
		cin>>l>>r>>k;
		p[l]+=k;
		p[r+1]-=k;
	}
	ll v=0;
	for(int i=1;i<n;++i){
		v+=p[i];
		ll x=v%n;
		if(x==0) continue;
		if(x<0) x+=n;
		reverse(s.begin()+i,s.begin()+i+x);
		reverse(s.begin()+i+x,s.begin()+n);
		reverse(s.begin()+i,s.begin()+n);
	}
	cout<<s.substr(1)<<'\n';
}
