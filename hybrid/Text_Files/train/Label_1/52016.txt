import java.util.Arrays;
import java.util.Scanner;

public class Main {
    private static int[] costs;
    private static int[] c;

    private static int bs(int l, int r, int money) {
        int m;
        while (l < r) {
            m = (l + r) / 2;
            if (c[m] <= money) {
                l = m + 1;
            } else {
                r = m - 1;
            }
        }
        if (c[l] <= money)
            return l;
        else
            return l - 1;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt(), money = in.nextInt();
        in.nextLine();
        costs = Arrays.stream(in.nextLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        c = new int[n];
        c[0] = costs[0];
        for (int i = 1; i < n; i++) {
            c[i] = c[i - 1] + costs[i];
        }
        if (c[n - 1] <= money)
            System.out.println(1);
        else
            System.out.println(bs(0, n - 1, money) + 2);
    }
}
