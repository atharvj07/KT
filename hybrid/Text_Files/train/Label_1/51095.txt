import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
    private static class Node {
        int data;
        int index;
        Node left;
        Node right;

        public Node(int data, int index) {
            this.data = data;
            this.index = index;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        String s = st.nextToken();
        int n = s.length();
        char[] arr = s.toCharArray();
        Node root = new Node(0, -1);
        int min = n;
        int[] permutation = new int[n];
        for (int i = 0; i < n; i++) {
            char c = arr[i];
            add(root, c, i + 1, 0, permutation);
            min = Math.min(min, permutation[n - 1]);
        }
        StringBuilder sb = new StringBuilder();
        sb.append(min).append("\n");
        for (int i = 0; i < n; i++) {
            sb.append(permutation[i]).append(" ");
        }
        System.out.println(sb);
    }

    private static void add(Node root, char c, int index, int count, int[] permutation) {
        if (root.data != 0) {
            count = count + root.data;
        }
        if (c == 'R') {
            if (root.right == null) {
                root.right = new Node(count, index);
                permutation[index - 1] = index;
            } else {
                add(root.right, c, index, count, permutation);
            }
        } else {
            if (root.left == null) {
                root.left = new Node(-count, index);
                permutation[index - 1] = index;
            } else {
                add(root.left, c, index, count, permutation);
            }
        }
    }
}
