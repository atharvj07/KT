import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }

        int maxOps = 0;
        while (true) {
            boolean allEven = true;
            for (int i = 0; i < n; i++) {
                if (a[i] % 2 == 1) {
                    allEven = false;
                    break;
                }
            }
            if (allEven) {
                maxOps++;
                for (int i = 0; i < n; i++) {
                    a[i] /= 2;
                }
            } else {
                break;
            }
        }

        System.out.println(maxOps);
    }
}
