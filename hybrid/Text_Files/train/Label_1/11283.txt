#include <iostream>
#include <algorithm>
#include <vector>
#define ll long long
using namespace std;

struct obj {
  ll t, d, p;
};

bool cmp(obj a, obj b) {
  if (a.d != b.d) return a.d < b.d;
  return a.t < b.t;
}

int main() {
  ll n;
  cin >> n;
  vector<obj> v(n);
  for (ll i = 0; i < n; i++) cin >> v[i].t >> v[i].d >> v[i].p;
  sort(v.begin(), v.end(), cmp);
  ll ans = 0;
  vector<ll> res;
  ll t = 0;
  for (ll i = 0; i < n; i++) {
    if (t + v[i].t <= v[i].d) {
      t += v[i].t;
      ans += v[i].p;
      res.push_back(i + 1);
    }
  }
  cout << ans << endl;
  cout << res.size() << endl;
  for (ll x : res) cout << x << " ";
  cout << endl;
}
