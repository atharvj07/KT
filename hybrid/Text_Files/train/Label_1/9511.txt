#include <iostream>
#include <algorithm>
#define ll long long

using namespace std;

ll n, k, a[100000], b[100000];

int main() {
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    for (int i = 0; i < n; i++) {
        if (b[i] >= a[i]) k -= b[i] - a[i];
        else break;
    }
    int ans = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (k >= a[i] * ans + b[i]) ++ans;
        else break;
    }
    cout << ans << endl;
    return 0;
}
