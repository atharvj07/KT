#include <cstdio>

int M[52][52];

int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { scanf("%d", &M[i][j]); }
  }
  for (int i = 0; i < n - 1; i++) {
    for (int j = 0; j < m - 1; j++) {
      if (M[i][j] + M[i + 1][j] + M[i][j + 1] + M[i + 1][j + 1] == 3) {
        M[i][j] = 1;
        M[i + 1][j] = 1;
        M[i][j + 1] = 1;
        M[i + 1][j + 1] = 1;
        printf("1 %d %d\n", i + 1, j + 1);
      }
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (M[i][j]) continue;
      printf("-1");
      return 0;
    }
  }
  printf("0");
  return 0;
}
