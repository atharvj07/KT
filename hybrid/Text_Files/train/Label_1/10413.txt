#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<pair<int, int> > books;
    while (n--) {
        int t, w;
        cin >> t >> w;
        books.push_back(make_pair(w, t));
    }
    sort(books.begin(), books.end());
    int sum = 0;
    for (vector<pair<int, int> >::iterator it = books.begin(); it != books.end(); it++) {
        sum += it->second;
        if (sum >= it->first) {
            sum -= it->first;
        } else {
            cout << sum << endl;
            return 0;
        }
    }
    cout << sum << endl;
    return 0;
}
