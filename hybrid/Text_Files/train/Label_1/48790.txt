import java.util.*;
class Main {
    static class Spider {
        int r, c, dir;
        Spider(int r, int c, int dir) {
            this.r = r; this.c = c; this.dir = dir;
        }
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        char[][] p = new char[n][m];
        for (int i = 0; i < n; ++i) {
            String s = sc.next(); p[i] = s.toCharArray();
        }
        ArrayList<Spider> spiders = new ArrayList<>();
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < m; ++j) {
                if ((p[i][j] == '.')) continue;
                int dir = 0;
                if (p[i][j] == 'L') dir = 2;
                else if (p[i][j] == 'R') dir = 3;
                else if (p[i][j] == 'U') dir = 0;
                else if (p[i][j] == 'D') dir = 1;
                spiders.add(new Spider(i, j, dir));
            }
        }
        int[] ans = new int[m];
        for (int start = 0; start < m; ++start) {
            int cnt = 0;
            HashSet<String> seen = new HashSet<String>();
            for (Spider spider : spiders) {
                int r = spider.r, c = spider.c;
                if (r >= n) continue;
                while(true) {
                    if (r == n - 1 && c >= start && c < start + Math.min(n, m)) {
                        String key = r + "," + c;
                        if (!seen.contains(key)) {
                            cnt += 1;
                            seen.add(key);
                        }
                        break;
                    }
                    if (spider.dir == 0) {
                        --r;
                    } else if (spider.dir == 1) {
                        ++r;
                    } else if (spider.dir == 2) {
                        --c;
                    } else if (spider.dir == 3) {
                        ++c;
                    }
                }
            }
            ans[start] = cnt;
        }
        for (int i = 0; i < m; ++i) System.out.print(ans[i] + " ");
        System.out.println("");
    }
}
