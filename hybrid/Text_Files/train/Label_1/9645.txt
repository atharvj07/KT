#include <bits/stdc++.h>
using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int n, m;
  cin >> n >> m;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  vector<long long> pre(n), suf(n);
  for (int i = 1; i < n; i++) { pre[i] = (pre[i - 1] + (a[i] - a[i - 1])) % m; }
  for (int i = n - 2; i >= 0; i--) { suf[i] = (suf[i + 1] + (a[i + 1] - a[i])) % m; }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    ans += ((pre[i] * suf[i]) % m);
    ans %= m;
  }
  cout << ans << '\n';
}
