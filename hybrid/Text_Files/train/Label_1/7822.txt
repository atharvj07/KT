#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int n;
    cin >> n;
    map<pair<ll, ll>, int> mp;
    while (n--) {
        ll a, b;
        cin >> a >> b;
        mp[make_pair(a, b)]++;
    }
    ll ans = 0;
    for (auto it : mp) {
        int c = it.second;
        ans += (c * (c - 1)) / 2;
    }
    cout << ans << endl;
}
