#include <cstdio>

#define ll unsigned long long

const int MAXN = 2e5 + 5;
const ll M = 998244353;
int A[MAXN];
int cnt[MAXN];
ll dp[MAXN][2];

int main() {
  int n, k;
  scanf("%d%d", &n, &k);
  for (int i = 1; i <= n; ++i) scanf("%d", &A[i]);

  ++A[n];
  int l = A[n - 1];
  for (int i = n; i; --i) {
    ++cnt[A[i]];
    if (A[i] != l) {
      dp[i][0] = cnt[l] % M;
      dp[i][1] = (dp[i + 1][0] + cnt[l] * (k - cnt[l]) % M) % M;
      l = A[i];
      cnt[l] = 1;
    } else {
      dp[i][0] = dp[i + 1][0];
      dp[i][1] = (dp[i + 1][0] + dp[i + 1][1]) % M;
    }
  }
  printf("%lld\n", (dp[1][0] + dp[1][1]) % M);
  return 0;
}
