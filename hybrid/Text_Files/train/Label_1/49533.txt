import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] a = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextInt();
        }
        if (N == 2) {
            System.out.println("No");
            return;
        }
        if (a[0] % 2 == a[N - 1] % 2) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
