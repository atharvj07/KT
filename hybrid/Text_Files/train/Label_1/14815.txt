#include <bits/stdc++.h>
using namespace std;
int main() {
    int k;
    cin >> k;
    int n = 1 << k;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if ((i & j) == 0)
                cout << "+";
            else
                cout << "*";
        }
        cout << '\n';
    }
    return 0;
}
