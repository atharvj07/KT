import java.util.Scanner;
import java.util.Arrays;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int N = sc.nextInt(), M = sc.nextInt();
		long[] A = new long[N];
		for (int i = 0; i < N; i++)
			A[i] = sc.nextInt();
		sc.close();
		Arrays.sort(A);
		long l = 1, h = M, ans = 0;
		while (l <= h) {
			long m = (l + h) / 2;
			long cnt = 0;
			for (int i = 0; i < N; i++) {
				if (A[i] * 0.5 > m)
					break;
				cnt += m / A[i] - (A[i] * 0.5 > 0 ? 1 : 0);
			}
			if (cnt > 0) {
				ans = m;
				l = m + 1;
			} else
				h = m - 1;
		}
		System.out.println(ans);
	}
}
