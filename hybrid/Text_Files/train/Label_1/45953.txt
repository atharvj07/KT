import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        while (T-- > 0) {
            int n = sc.nextInt() - 1;
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            int[] ans = new int[r - l + 1];
            for (int i = 0; i < r - l + 1; i++) {
                ans[i] = n - i;
            }
            for (int i = 0; i < r - l + 1; i++) {
                System.out.print(ans[i] % (n + 1) + 1 + " ");
            }
            System.out.println();
        }
    }
}
