n = int(input())
g = [[int(c) for c in input()] for _ in range(n)]
ans = 'Yes'
for x in range(n):
    for y in range(n):
        if g[x][y] > 0:
            for Dx in range(1, n - x):
                for Dy in range(1, n - y):
                    flag = True
                    for xx in range(x, x + Dx + 1):
                        if g[xx][y] + g[xx][y + Dy] != g[xx][y + Dy]:
                            flag = False
                    for yy in range(y, y + Dy + 1):
                        if g[x][yy] + g[x + Dx][yy] != g[x + Dx][yy]:
                            flag = False
                    if flag:
                        ans = 'Yes'
                        break
            if ans == 'Yes':
                break
    if ans == 'Yes':
        break
print(ans)
