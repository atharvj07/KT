n, m = map(int, input().split())
a = list(map(int, input().split()))

ans = [0] * n
vis = [False] * (m + 1)

for i in range(n):
    if vis[a[i]]:
        ans[i] = -1
    else:
        vis[a[i]] = True

mx = 0
for i in range(1, m + 1):
    if not vis[i]:
        for j in range(n):
            if ans[j] == -1:
                ans[j] = i
                mx += 1

print(mx)
for i in ans:
    print(*divmod(i - 1, 2))
