#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e6 + 3;
pair<ll, ll> BSGS(ll n, ll m){
    map<ll, ll> mp;
    ll sqrt_n = sqrt(n);
    for(ll i = 0; i <= sqrt_n; i++){
        mp[(i * m) % n] = i;
    }
    ll a = 1;
    for(ll i = sqrt_n + 1;; i++){
        a = (a * m) % n;
        if(mp.find(a) != mp.end()){
            return {i - mp[a], i};
        }
    }
}
pair<ll, ll> solve(ll n, ll k){
    if(n == 1) return {0, 1};
    if(n <= k) return {1, 1};
    ll p = ((n - 1) * 499122177) % mod;
    ll ans = BSGS(mod, p).second;
    ll tot = (ans - 1 + mod) % mod;
    ll lcm = (tot * k) % mod;
    ll gcd = __gcd(tot, lcm);
    return {(lcm / gcd) % mod, (tot / gcd) % mod};
}
int main(){
    ll n, k; cin >> n >> k;
    pair<ll, ll> ans = solve(n, k);
    cout << ans.first << " " << ans.second << '\n';
    return 0;
}
