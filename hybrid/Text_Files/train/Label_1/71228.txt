def min_coating_pieces(s, t):
    # Helper function to find the maximal prefix of t in s or its reverse
    def max_prefix(s, t, reverse=False):
        if reverse:
            s = s[::-1]
        max_len = 0
        for i in range(1, min(len(s), len(t)) + 1):
            if s[:i] == t[:i]:
                max_len = i
        return max_len
    
    # Main logic
    pieces = []
    i = 0
    while i < len(t):
        # Try to match a prefix of the remaining t in s or its reverse
        len_direct = max_prefix(s, t[i:], False)
        len_reverse = max_prefix(s, t[i:], True)
        
        if len_direct == 0 and len_reverse == 0:
            return -1, []
        
        if len_direct >= len_reverse:
            # Use direct piece
            pieces.append((i + 1, i + len_direct))
            i += len_direct
        else:
            # Use reversed piece
            pieces.append((i + len_reverse, i + 1))
            i += len_reverse
    
    return len(pieces), pieces

# Read input
s = input().strip()
t = input().strip()

# Solve
n, pieces = min_coating_pieces(s, t)

# Print output
if n == -1:
    print(n)
else:
    print(n)
    for p in pieces:
        print(p[0], p[1])
