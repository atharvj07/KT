import java.util.*;

public class Main {
    static boolean[] vis = new boolean[1005];
    static boolean[] nonvis = new boolean[1005];
    static int[][] a = new int[1005][1005];
    static Queue<Integer> q = new LinkedList<>();

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt(), s = in.nextInt(), t = in.nextInt(), ans = 0;
        for (int i = 0; i < n; i++) {
            Arrays.fill(a[i], -1);
        }
        for (int i = 0; i < m; i++) {
            int x = in.nextInt(), y = in.nextInt();
            a[x][y] = a[y][x] = 1;
        }
        q.add(s);
        vis[s] = true;
        while (!q.isEmpty()) {
            int v = q.poll();
            nonvis[v] = true;
            for (int i = 1; i <= n; i++) {
                if (a[v][i] != -1 && !vis[i]) {
                    q.add(i);
                    vis[i] = true;
                    a[v][i] = a[i][v] = 0;
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (!nonvis[i] || !nonvis[j]) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
