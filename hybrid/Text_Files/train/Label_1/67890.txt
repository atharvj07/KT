t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    if max(a) == 1:
        print(0)
        continue
    found = False
    for d in range(2, max(a) + 1):
        red = False
        blue = False
        for i in range(n):
            if a[i] % d == 0:
                if red:
                    break
                red = True
            else:
                if blue:
                    break
                blue = True
            if i == n - 1:
                print(d)
                found = True
        if found:
            break
    if not found:
        print(0)
