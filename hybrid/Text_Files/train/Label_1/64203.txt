import sys

mod = 998244353

n, m = map(int, sys.stdin.readline().split())
c = list(map(int, sys.stdin.readline().split()))

dp = [[0] * (m + 1) for _ in range(n + 1)]
dp[0][0] = 1
for i in range(1, n + 1):
    for j in range(m + 1):
        dp[i][j] = dp[i][j - 1]
        if j >= i and c[j - i] == i:
            dp[i][j] += dp[i - 1][j - i]
print(dp[n][m] % mod)
