#include <iostream>
#include <string>

using namespace std;

int main() {
  int n, p;
  cin >> n >> p;
  string s;
  cin >> s;
  int l = p - 1, r = p - 1, ans = 0;
  while (l >= 0 || r < n) {
    if (l >= 0 && r < n && s[l] == s[r]) {
      l--;
      r++;
    } else {
      if (l >= 0 && r == n) {
        l--;
        ans++;
      } else if (r == -1) {
        r++;
        ans++;
      } else {
        if (s[l] != s[r]) {
          char x = min(s[l], s[r]), y = max(s[l], s[r]);
          if (y - x > 13) {
            x += 26;
          }
          ans += min(y - x, x - y);
          s[r] = s[l];
        }
        l--;
        r++;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
