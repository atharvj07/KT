a, b, c, d = map(int, input().split())

p_smallr = a / b
p_zanoes = c / d

# Probability that SmallR wins is the sum of probabilities that SmallR wins in the first round,
# or misses and Zanoes misses and SmallR wins in the next round, and so on.
# This forms a geometric series with the first term as p_smallr and common ratio as (1 - p_smallr) * (1 - p_zanoes)
# Sum of infinite geometric series is a / (1 - r) where a is the first term and r is the common ratio

probability_win = p_smallr / (1 - (1 - p_smallr) * (1 - p_zanoes))
print(f"{probability_win:.12f}")
