#include <iostream>
#include <vector>
using namespace std;

int n;
vector<vector<int>> board;

int main() {
    cin >> n;
    board.resize(n, vector<int>(n, 0));

    int count = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if ((i + j) % 2 == 0) {
                board[i][j] = 1;
                count++;
            }
        }
    }

    cout << count << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << (board[i][j] ? 'C' : '.');
        }
        cout << endl;
    }

    return 0;
}
