#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 1e5 + 5;
ll a[maxn], b[maxn], ans;
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i] >> b[i];
  sort(a + 1, a + n + 1);
  sort(b + 1, b + n + 1);
  for (int i = 1; i <= n; i++) {
    if (a[i] >= b[i])
      ans += b[i];
    else
      a[i + 1] += a[i] - b[i], ans += b[i];
  }
  cout << ans << endl;
  return 0;
}
