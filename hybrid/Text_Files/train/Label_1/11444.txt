#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, m, u, v, ans = 0;
    cin >> n >> m;
    vector<int> deg(n), deg2(n), adj[n];
    vector<pair<int, int>> edges;
    for (int i = 0; i < m; i++) {
        cin >> u >> v, u--, v--;
        deg[u]++, deg[v]++;
        adj[u].push_back(v), adj[v].push_back(u);
        edges.push_back({u, v});
    }
    for (int i = 0; i < n; i++) {
        deg2[deg[i]]++;
        if (!deg2[deg[i]] && deg[i] >= 2) ans++;
    }
    for (int i = 1; i <= n - 2; i++) ans += max(0, min(deg2[i], deg2[i + 1]) - 1);
    cout << ans << endl;
    return 0;
}
