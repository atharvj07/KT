#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;
vector<int> adj[100005];
int parent[100005], depth[100005], beauty[100005];
void dfs(int u, int p, int d) {
  parent[u] = p;
  depth[u] = d;
  for (auto v : adj[u]) {
    if (v != p) dfs(v, u, d + 1);
  }
}
int get_lca(int u, int v) {
  if (depth[u] < depth[v]) swap(u, v);
  while (depth[u] > depth[v]) {
    u = parent[u];
  }
  while (u != v) {
    u = parent[u];
    v = parent[v];
  }
  return u;
}
int gcd(int a, int b) {
  if (!b) return a;
  return gcd(b, a % b);
}
int32_t main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> beauty[i]; }
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs(1, 0, 0);
  int ans = 0;
  for (int i = 2; i <= n; i++) {
    ans += gcd(beauty[i], beauty[parent[i]]);
    ans %= mod;
  }
  cout << ans << '\n';
}
