import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n,s = map(int,input().split())
    L = []
    for i in range(n):
        a,b = map(int,input().split())
        L.append([a,b,b-a])
    L.sort(key=lambda x:x[2])
    ans = 0
    for a,b,_ in L:
        if s<a: break
        s-=a
        ans=b
    print(ans)
