#include<bits/stdc++.h>
using namespace std;
int n,m,ans[5050],aij[55][55],bij[55][55],x,y;
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;++i)
        for(int j=1;j<=m;++j)
            cin>>aij[i][j];
    for(int i=1;i<=n;++i)
        for(int j=1;j<=m;++j){
            int cnt=0;
            int ti=i,tj=j;
            if(ti-1<=0)   ++ti;
            if(tj-1<=0)   ++tj;
            if(ti+1>n)    --ti;
            if(tj+1>m)    --tj;
            for(int a=ti;a<=ti+1;++a)
                for(int b=tj;b<=tj+1;++b)
                    if(aij[a][b])  ++cnt;
            if(cnt&1){
                bij[i][j]=1;
                ans[++x]=i;
                ans[++y]=j;
            }
        }
    for(int i=1;i<=n;++i)
        for(int j=1;j<=m;++j)
            if(aij[i][j]!=bij[i][j]){
                cout<<-1<<endl;
                return 0;
            }
    cout<<x/2<<endl;
    for(int i=1;i<=x;++i)
        cout<<ans[i]<<' '<<ans[++i]<<endl;
    return 0;
}
