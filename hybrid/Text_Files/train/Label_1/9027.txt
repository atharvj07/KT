#include <iostream>
#include <vector>
#include <algorithm>

int main()
{
	int n, l;
	std::cin >> n >> l;
	
	std::vector<int> a(n);
	for (int i = 0; i < n; ++i) std::cin >> a[i];
	std::sort(a.begin(), a.end());

	double ans = std::max(a[0], l - a.back());
	for (int i = 1; i < n; ++i) {
		ans = std::max(ans, (a[i] - a[i - 1]) / 2.0);
	}
	std::cout << ans << '\n';
}
