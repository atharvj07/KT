#include <iostream>
#include <unordered_map>

using namespace std;

int main() {
  int p, n;
  cin >> p >> n;
  unordered_map<int, int> m;
  int ans = -1;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    if (m.count(x % p)) {
      ans = i;
      break;
    }
    m[x % p] = true;
  }
  cout << ans << endl;
  return 0;
}
