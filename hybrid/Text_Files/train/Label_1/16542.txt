#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 1; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i < n; i++) {
    long long x = 1, y = 0;
    while (true) {
      x += a[x];
      y += a[x];
      if (x <= 0 || x > n) break;
      x -= a[x];
      y += a[x];
    }
    if (x > 0 && x <= n)
      cout << y << "\n";
    else
      cout << -1 << "\n";
  }
  return 0;
}
