import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String str = scanner.next();

        if (str.charAt(0) == 'A' && str.chars().filter(ch -> ch == 'C').count() == 1 && str.substring(2, str.length() - 1).chars().filter(ch -> ch < 'a' || ch > 'z').count() == 0) {
            System.out.println("AC");
        } else {
            System.out.println("WA");
        }
    }
}
