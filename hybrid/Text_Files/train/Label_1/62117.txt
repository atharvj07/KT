import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = []; b = []; total = 0
for _ in range(n):
    ai, bi = map(int, input().split())
    a.append(ai); b.append(bi)
    total += ai
if total <= m:
    print(0)
    exit()
a.sort(reverse=True); b.sort(reverse=True)
need = total - m
ans = 0
for ai, bi in zip(a, b):
    if need > 0:
        need -= ai - bi
        ans += 1
    else:
        break
if need <= 0:
    print(ans)
else:
    print(-1)
