#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const ll maxn = 2e5 + 1;
string t, p;
ll n, a[maxn], fenwick[maxn];
map<char, ll> mp;

void update(ll x, ll val) {
  for (ll i = x; i <= n; i += i & -i) fenwick[i] += val;
}

ll get(ll x) {
  ll sum = 0;
  for (ll i = x; i > 0; i -= i & -i) sum += fenwick[i];
  return sum;
}

int main() {
  cin >> t >> p;
  cin >> n;
  for (ll i = 0; i < n; i++) { cin >> a[i]; }
  n = t.size();
  for (ll i = 0; i <= n; i++) fenwick[i] = 0;
  for (ll i = n - 1; i >= 0; i--) {
    auto it = mp.find(t[i]);
    ll cnt = 0;
    if (it != mp.end()) { cnt = it->second; }
    while (get(a[i]) - cnt > i) {
      a[i]--;
      update(a[i], 1);
    }
    mp[t[i]]++;
  }
  for (ll i = 0; i < n; i++) {
    if (get(a[i]) == i) {
      cout << i + 1 << endl;
      return 0;
    }
  }
  cout << n << endl;
  return 0;
}
