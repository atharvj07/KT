#include <bits/stdc++.h>
using namespace std;

const int maxn = 1e5 + 5;
int a[maxn], n, lim[maxn];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];

  vector<int> v;
  for (int i = n; i; i--) {
    while (v.size() && a[v.back()] >= a[i]) v.pop_back();
    lim[i] = v.empty() ? n : v.back();
    v.push_back(i);
  }

  for (int i = 1; i <= n; i++) {
    int x = 0;
    for (int j = i; j <= n; j = lim[j] + 1) x = max(x, a[j]);
    if ((x >> 1) < a[i]) cout << -1 << ' ';
    else cout << x << ' ';
  }
  cout << endl;
  return 0;
}
