import sys

t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    if len(a) == 3:
        if a[0] == a[2]:
            print("YES")
        else:
            print("NO")
        continue

    dp = [[[-1 for _ in range(4)] for _ in range(4)] for _ in range(n+1)]

    for i in range(3, n+1):
        for j in range(3):
            for k in range(4):
                dp[i][j][k] = dp[i-1][j][k]
                if k == 0 or k == 1:
                    if a[i-1] == a[i-j-1]:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-2][j-2][k])
                if k == 1 or k == 2:
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1])

    if dp[n][2][3] != -1:
        print("YES")
    else:
        print("NO")
