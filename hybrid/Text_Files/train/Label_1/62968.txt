import sys

n = int(input())
a = [int(x) for x in sys.stdin.readline().split()]
m = [[] for i in range(6)]

for x in a:
  m[x % 6 - 1].append(x)

tot = 0
for i in m:
  tot += len(i) // 6
  i.sort()

ans = n - 6 * tot
for i in range(len(m)):
  if len(m[i]) >= tot:
    continue
  ans += len(m[i])
  m[i] = []
  for j in range(i + 1, len(m)):
    if len(m[i] + m[j]) >= tot:
      m[i] = m[i] + m[j]
      ans -= len(m[j])
      m[j] = []
      break

print(ans)
