from sys import stdin
read = lambda: stdin.readline()
n, k = map(int, read().strip().split())
p = list(map(int, read().strip().split()))
pref = [0] * (n + 1)
suf = [0] * (n + 1)
for i in range(1, n + 1):
    pref[i] = max(pref[i - 1], p[i - 1])
for i in range(n - 1, -1, -1):
    suf[i] = max(suf[i + 1], p[i])
dp = [[[-1] * 3 for _ in range(k + 1)] for _ in range(n + 1)]
mod = 998244353

def rec(l, r, bal):
    # print(l, r, bal)
    if l > r:
        if bal == 0:
            return 1
        return 0
    if dp[l][r][bal] != -1:
        return dp[l][r][bal]
    ret = 0
    if bal == 0:
        ret += rec(l + 1, r, 0)
        ret %= mod
        ret += rec(l + 1, r, 1)
        ret %= mod
    elif bal == 1:
        ret += rec(l, r - 1, 0)
        ret %= mod
        ret += rec(l, r - 1, 1)
        ret %= mod
    elif bal == 2:
        ret += rec(l, r - 1, 2)
        ret %= mod
    dp[l][r][bal] = ret
    return ret

print(max(pref[i] + suf[i] for i in range(n - k + 1)), rec(0, n - 1, 0))
