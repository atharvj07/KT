from bisect import bisect_left

n, b = map(int, input().split())
t = []
d = []
for _ in range(n):
    ti, di = map(int, input().split())
    t.append(ti)
    d.append(di)

e = [-1] * n
q = []
time = 0
for i in range(n):
    if time <= t[i]:
        time = t[i]
    idx = bisect_left(q, time)
    if idx < len(q):
        time = q[idx]
    if time <= t[i]:
        time += d[i]
        if len(q) < b:
            q.append(time)
    e[i] = time


print(*e)
