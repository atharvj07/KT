#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n, m; cin >> n >> m;
    vector<int> a(n), b(m);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    for (int i = 0; i < m; ++i) { cin >> b[i]; }
    sort(a.begin(), a.end(), greater<int>());
    sort(b.begin(), b.end(), greater<int>());
    int ans = 0;
    while (!a.empty() && !b.empty()) {
        if (a.back() > b.back()) { a.pop_back(); ++ans; }
        else { b.pop_back(); ++ans; }
    }
    cout << ans << endl;
    return 0;
}
