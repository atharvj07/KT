#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(m);
  for (int i = 0; i < m; i++) cin >> a[i];

  int ans = 0;
  for (int i = 0; i < m; i++) {
    int cnt = 0;
    for (int j = 0; j < m; j++) {
      if (a[i] == a[j]) cnt++;
    }
    ans = max(ans, cnt / n);
  }

  cout << ans << endl;
  return 0;
}
