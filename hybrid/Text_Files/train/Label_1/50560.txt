import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] circles = new int[n][3];
        for (int i = 0; i < n; i++) {
            circles[i][0] = in.nextInt();
            circles[i][1] = in.nextInt();
            circles[i][2] = in.nextInt();
        }
        System.out.println(countRegions(circles));
    }

    private static int countRegions(int[][] circles) {
        int n = circles.length;
        int ans = n + 1;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int dx = circles[i][0] - circles[j][0];
                int dy = circles[i][1] - circles[j][1];
                int dr = circles[i][2] + circles[j][2];
                if (dx * dx + dy * dy <= dr * dr) {
                    ans--;
                }
            }
        }
        return ans;
    }
}
