#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    if (n % 2 == 0) {
      n /= 2;
      if (n % 3 == 0) {
        n /= 3;
        cout << "YES" << endl;
        cout << 2 << " " << 3 << " " << n << endl;
        continue;
      }
    }
    cout << "NO" << endl;
  }
  return 0;
}
