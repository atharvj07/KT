import java.util.PriorityQueue;
import java.util.Scanner;

public class BeruTaxi {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int n = scanner.nextInt();

        PriorityQueue<Taxi> taxis = new PriorityQueue<>((t1, t2) -> {
            double time1 = Math.sqrt(Math.pow(t1.x - a, 2) + Math.pow(t1.y - b, 2)) / t1.v;
            double time2 = Math.sqrt(Math.pow(t2.x - a, 2) + Math.pow(t2.y - b, 2)) / t2.v;
            return Double.compare(time1, time2);
        });

        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            int v = scanner.nextInt();
            taxis.offer(new Taxi(x, y, v));
        }

        System.out.println(taxis.peek().getTimeToReach(a, b));
    }

    private static class Taxi {
        int x;
        int y;
        int v;

        public Taxi(int x, int y, int v) {
            this.x = x;
            this.y = y;
            this.v = v;
        }

        public double getTimeToReach(int a, int b) {
            return Math.sqrt(Math.pow(x - a, 2) + Math.pow(y - b, 2)) / v;
        }
    }
}
