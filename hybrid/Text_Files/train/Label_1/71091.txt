import sys

def input(): return sys.stdin.readline().strip()
def list2int(l): return list(map(int, l.split()))

for _ in range(int(input())):
    n, k = list2int(input())
    a = list2int(input())
    if k == 1:
        print(-1 if a[n - 1] else 1)
    else:
        print(1 + (a[n - 1] + min(k - 1, a[0])) // k)
