n, m, k = map(int, input().split())

friendships = [set() for _ in range(n)]
can_go = set()

for _ in range(m):
    x, y = map(lambda z: int(z) - 1, input().split())
    friendships[x].add(y)
    friendships[y].add(x)
    
    if len(friendships[x]) >= k:
        can_go.add(x)
    if len(friendships[y]) >= k:
        can_go.add(y)
        
    to_remove = set()
    for person in can_go:
        if len(friendships[person]) < k:
            to_remove.add(person)
        else:
            for friend in friendships[person]:
                if friend not in can_go:
                    to_remove.add(person)
                    break
    can_go -= to_remove
    
    print(len(can_go))
