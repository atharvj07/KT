#include <bits/stdc++.h>
using namespace std;
int n, m;
char a[1001][1001];
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
  }
  int ans = 0;
  for (int i = 1; i <= n - 2; i++) {
    for (int j = 1; j <= m; j++) {
      for (int k = i + 2; k <= n; k++) {
        char x1 = a[i][j], x2 = a[i + 1][j], x3 = a[k][j];
        if (x1 != x2 && x2 != x3 && x1 != x3) ans++;
      }
    }
  }
  cout << ans << '\n';
  return 0;
}
