n = int(input())
s = input()
ans = 0
pref = [0] * (n + 1)
for i in range(n):
    pref[i + 1] = pref[i]
    pref[i + 1] += 1 if s[i] == '1' else -1
    if pref[i + 1] == 0:
        ans = i + 1
    else:
        for j in range(i + 1):
            if pref[i + 1] == pref[j]:
                ans = max(ans, i - j + 1)
print(ans)
