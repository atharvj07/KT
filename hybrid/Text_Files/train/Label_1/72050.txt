import sys
import collections

n, m, k = map(int, input().split())
if n == 1 or m == 1 or (n * m * 2 - 2 * n - 2 * m) % 2 != k % 2:
    print("NO")
    sys.exit()
    
g = collections.defaultdict(lambda: collections.defaultdict(int))
for i in range(1, n):
    for j in range(m):
        g[i][j] |= 1
        g[i + 1][j] |= 2
        
for i in range(n):
    for j in range(1, m):
        g[i][j] |= 4
        g[i][j + 1] |= 8

def dfs(x, y, d, steps):
    if d == k:
        return True
    if d + 2 * (n - x) + 2 * (m - y) < k:
        return False
    if (x, y, d) in vis:
        return vis[(x, y, d)]
    vis[(x, y, d)] = False
    if g[x][y] & (1 << steps[d % 4]) == 0:
        vis[(x, y, d)] = dfs(x + [1, -1, 0, 0][steps[d % 4]], y + [0, 0, 1, -1][steps[d % 4]], d + 1, steps)
    if not vis[(x, y, d)]:
        vis[(x, y, d)] = dfs(x, y, d, steps + [1, 3, 0, 2])
    return vis[(x, y, d)]

vis = dict()
steps = [0] * 500
print("YES")
print(dfs(1, 1, 0, steps))
for i in steps[:500]:
    if i == 0:
        break
    print(i, "RLDU"[i - 1])
