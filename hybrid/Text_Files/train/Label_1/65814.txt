MOD = 998244353

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(n, k):
    if k == 1:
        return n
    if n < k:
        return 0
    fact = [1] * (n + 1)
    for i in range(2, n + 1):
        fact[i] = fact[i - 1] * i % MOD
    inv_fact = [1] * (n + 1)
    inv_fact[n] = power(fact[n], MOD - 2, MOD)
    for i in range(n, 0, -1):
        inv_fact[i - 1] = inv_fact[i] * i % MOD
    def binom(n, k):
        return fact[n] * inv_fact[k] % MOD * inv_fact[n - k] % MOD
    ans = 0
    for i in range(1, n + 1):
        max_val = n // i
        if max_val < k - 1:
            break
        ans += binom(max_val - 1, k - 1)
        ans %= MOD
    return ans

n, k = map(int, input().split())
print(solve(n, k))
