n = int(input())
capacities = list(map(int, input().split()))
m = int(input())

water_in_vessels = [0] * n

for _ in range(m):
    query = list(map(int, input().split()))
    
    if query[0] == 1:
        p, x = query[1] - 1, query[2]
        while x > 0 and p < n:
            space_left = capacities[p] - water_in_vessels[p]
            if x >= space_left:
                water_in_vessels[p] += space_left
                x -= space_left
            else:
                water_in_vessels[p] += x
                x = 0
            p += 1
    else:
        k = query[1] - 1
        print(water_in_vessels[k])
