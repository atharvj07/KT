#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  char c = 'a';
  for (int i = 0; i < n; i++) {
    if (s[i] == c) {
      c++;
    }
  }
  if (c == 'z' + 1) {
    cout << s << endl;
  } else {
    cout << "-1" << endl;
  }
  return 0;
}
