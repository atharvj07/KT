#include <bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
vector<int> g[N];
bool vis[N], vis1[N];
vector<int> top_sort;

void dfs(int u) {
  vis[u] = true;
  for (auto v : g[u]) {
    if (!vis[v])
      dfs(v);
    else if (!vis1[v])
      top_sort.push_back(u);
  }
  vis1[u] = true;
}

void topo_sort() {
  for (int i = 1; i <= N; i++) {
    if (!vis[i])
      dfs(i);
  }
}

int main() {
  int n, m;
  cin >> n >> m;
  while (m--) {
    int x, y;
    cin >> x >> y;
    g[x].push_back(y);
  }
  topo_sort();
  if (top_sort.size())
    cout << "NO\n";
  else
    cout << "YES\n";
}
