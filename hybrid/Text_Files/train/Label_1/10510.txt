#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> grid(n);
  for (int i = 0; i < n; i++) { cin >> grid[i]; }

  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == '*' && (i == 0 || i == n - 1 || j == 0 || j == m - 1 ||
                                   grid[i - 1][j] == '.' || grid[i + 1][j] == '.' ||
                                   grid[i][j - 1] == '.' || grid[i][j + 1] == '.')) {
        ans++;
        cout << i + 1 << ' ' << j + 1 << ' ' << 1 << endl;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
