#include<bits/stdc++.h>
using namespace std;

int main() {
    long long r;
    cin >> r;
    long long y;
    y = (-1.0 + sqrt(1.0 + 4.0 * (r - 1.0))) / 2.0;
    long long x = (r - y * y - 2 * y - 1.0) / 2.0 / y;
    if (x * x + 2 * x * y + x + 1.0 == r && x > 0 && y > 0)
        cout << x << " " << y << endl;
    else
        cout << "NO" << endl;
    return 0;
}
