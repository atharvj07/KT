#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<vector<char>> cake(n, vector<char>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) { cin >> cake[i][j]; }
  }
  int happiness = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (cake[i][j] == '.') continue;
      for (int k = j + 1; k < n; k++) {
        if (cake[i][k] == 'C') happiness++;
      }
      for (int k = i + 1; k < n; k++) {
        if (cake[k][j] == 'C') happiness++;
      }
    }
  }
  cout << happiness << endl;
}
