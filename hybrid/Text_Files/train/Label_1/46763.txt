import java.util.Scanner;

public class InnopolisHills {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] heights = new int[n];

        for (int i = 0; i < n; i++) {
            heights[i] = scanner.nextInt();
        }

        int m = (n + 1) / 2;
        int[] minHours = new int[m];

        for (int k = 1; k <= m; k++) {
            int minTime = 0;
            int left = 0, right = n - 1;
            int leftHeight = heights[left], rightHeight = heights[right];

            while (left < right) {
                if (leftHeight > rightHeight) {
                    right--;
                    rightHeight = heights[right];
                } else {
                    left++;
                    leftHeight = heights[left];
                }

                int targetHeight = Math.max(leftHeight, rightHeight) - 1;

                if (targetHeight >= 0) {
                    minTime += leftHeight - targetHeight + rightHeight - targetHeight;
                }
            }

            minHours[k - 1] = minTime;
        }

        for (int minHour : minHours) {
            System.out.print(minHour + " ");
        }
    }
}
