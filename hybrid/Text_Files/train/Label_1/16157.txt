#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<char> s(n);
  for (int i = 0; i < n; i++) cin >> s[i];
  vector<char> t(k);
  for (int l = 0, r = 0; r < k; r++) {
    if (s[l] == '(' || r - l + 1 == n - l) t[r] = s[l++];
  }
  for (char c : t) cout << c;
  return 0;
}
