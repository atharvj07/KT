import sys
input = sys.stdin.readline

n = int(input())
a = [int(x) for x in input().split()]
swaps = []
for i in range(n):
    for j in range(i + 1, n):
        if a[i] > a[j]:
            a[i], a[j] = a[j], a[i]
            swaps.append([i, j])
print(len(swaps))
for swap in swaps:
    print(*swap)
