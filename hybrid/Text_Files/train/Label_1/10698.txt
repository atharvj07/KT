#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a, a + n);
  int sereja = 0, dima = 0;
  for (int i = n - 1; i >= 0; i -= 2) {
    sereja += a[i];
  }
  for (int i = n - 2; i >= 0; i -= 2) {
    dima += a[i];
  }
  cout << sereja << " " << dima << endl;
  return 0;
}
