#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
using namespace std;

int main() {
  string s;
  int k;
  cin >> s >> k;
  vector<int> w(26);
  for (int i = 0; i < 26; i++) cin >> w[i];
  vector<int> cnt(26);
  for (char c : s) cnt[c - 'a']++;
  sort(w.rbegin(), w.rend());
  int ans = 0;
  for (int i = 0; i < s.size(); i++) ans += w[cnt[s[i] - 'a']++] * (i + 1);
  for (int i = 0; i < k; i++) ans += w[cnt[0]++] * (s.size() + i + 1);
  cout << ans << endl;
}
