from sys import stdin
from math import sqrt

n, m = map(int, stdin.readline().split())
grid = stdin.readlines()
grid = [list(row) for row in grid]

def solve(n, m, grid):
    stars = []
    for i in range(1, n - 1):
        for j in range(1, m - 1):
            if grid[i][j] == '*' and grid[i-1][j] == grid[i+1][j] == grid[i][j-1] == grid[i][j+1] == '.':
                stars.append((i, j, 1))
                grid[i-1][j] = grid[i+1][j] = grid[i][j-1] = grid[i][j+1] = '*'
    
    for i in range(1, n - 1):
        for j in range(1, m - 1):
            if grid[i][j] == '*' and grid[i-1][j] == grid[i+1][j] == '.':
                stars.append((i, j, 2))
                grid[i-1][j] = grid[i+1][j] = '*'
    
    for i in range(1, n - 1):
        for j in range(1, m - 1):
            if grid[i][j] == '*' and grid[i][j-1] == grid[i][j+1] == '.':
                stars.append((i, j, 3))
                grid[i][j-1] = grid[i][j+1] = '*'
    
    for i in range(1, n - 1):
        for j in range(1, m - 1):
            if grid[i][j] == '*' and grid[i-1][j] == grid[i][j+1] == '.':
                stars.append((i, j, 4))
                grid[i-1][j] = grid[i][j+1] = '*'
    
    for i in range(1, n - 1):
        for j in range(1, m - 1):
            if grid[i][j] == '*':
                stars.append((i, j, 5))
    
    if len(stars) > n * m:
        return -1
    else:
        return len(stars), stars

result = solve(n, m, grid)
if result == -1:
    print(-1)
else:
    num_stars, stars = result
    print(num_stars)
    for star in stars:
        print(*star)
