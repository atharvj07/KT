#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  int k;
  cin >> s >> k;
  int l = s.size();
  if (l % k != 0) {
    cout << "NO" << endl;
    return 0;
  }
  l /= k;
  bool ok = true;
  for (int i = 0; i < l; i++) {
    int j = l - i - 1;
    for (int o = 0; o < i; o++) {
      ok &= s[l * o + i] == s[l * o + j];
    }
  }
  cout << (ok ? "YES" : "NO") << endl;
  return 0;
}
