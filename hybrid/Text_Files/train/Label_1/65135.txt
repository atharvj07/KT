s = input()
n = len(s)
even_count = 0
odd_count = 0

if n & 1 == 0:
    for i in range(0, n - 1, 2):
        if s[i] == s[i + 1]:
            even_count += 1
else:
    for i in range(1, n - 1, 2):
        if s[i] == s[i + 1]:
            even_count += 1

if n % 2 == 0:
    for i in range(0, n):
        if s[i] == s[(i + 1) % n]:
            odd_count += 1
else:
    for i in range(0, n - 1):
        if s[i] == s[i + 1]:
            odd_count += 1

print(even_count, odd_count)
