n = int(input())
a = list(map(int, input().split()))

def count_violations(k):
    violations = 0
    for i in range(1, n):
        parent_index = (i - 1) // k
        if a[i] < a[parent_index]:
            violations += 1
    return violations

for k in range(1, n):
    print(count_violations(k), end=' ')
