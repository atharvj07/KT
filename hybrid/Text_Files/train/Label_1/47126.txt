import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt(), m = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();

        int ans = 0;
        for (int i = 0; i < n; i++) {
            boolean can = true;
            for (int j = i + 1; j < n; j++) {
                if (a[j] != a[i]) break;
                if (j - i + 1 == k) { can = false; break; }
            }
            if (can) ans++;
        }

        System.out.println(ans * m);
    }
}
