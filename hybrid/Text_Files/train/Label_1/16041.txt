#include <bits/stdc++.h>
using namespace std;

#define ll long long

int n, k;
ll timer[200005];
pair<ll, ll> cord[200005];

int dx[4] = {1, -1, 0, 0};
int dy[4] = {0, 0, 1, -1};

bool vis[200005];
queue<int> q;

int main() {
    int t;
    cin >> t;
    while (t--) {
        while (!q.empty())
            q.pop();
        memset(timer, -1, sizeof timer);
        memset(vis, false, sizeof vis);
        cin >> n >> k;
        for (int i = 0; i < n; i++) {
            ll x, y, time;
            cin >> x >> y >> time;
            cord[i] = {x, y};
            timer[i] = time;
        }
        ll ans = LLONG_MAX;
        for (int i = 0; i < n; i++) {
            ll x = cord[i].first, y = cord[i].second;
            if (vis[i])
                continue;
            vis[i] = true;
            ll time = timer[i];
            q.push(i);
            while (!q.empty()) {
                int u = q.front();
                q.pop();
                for (int dir = 0; dir < 4; dir++) {
                    ll nx = x + dx[dir], ny = y + dy[dir];
                    int idx = -1;
                    for (int j = 0; j < n; j++) {
                        if (cord[j].first == nx && cord[j].second == ny) {
                            idx = j;
                            break;
                        }
                    }
                    if (idx == -1 || vis[idx])
                        continue;
                    if (timer[idx] < time + 1 || timer[idx] == -1) {
                        timer[idx] = time + 1;
                        vis[idx] = true;
                        q.push(idx);
                    }
                }
                time++;
            }
            ans = min(ans, time - 1);
        }
        cout << ans << '\n';
    }
    return 0;
}
