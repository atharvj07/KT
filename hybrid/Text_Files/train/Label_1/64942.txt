n = int(input())
a = list(map(int, input().split()))
m = 0
for i in a: m += i
for i in range(1, n + 1):
  if m - a[i - 1] > a[i - 1] and m / 2 < a[i - 1]:
    print(1)
    print("1")
    break
  for j in range(i + 1, n + 1):
    if m - a[i - 1] - a[j - 1] > a[j - 1] and m / 2 < a[j - 1]:
      print(2)
      print("1", j)
      break
    for k in range(j + 1, n + 1):
      if m - a[i - 1] - a[j - 1] - a[k - 1] > a[k - 1] and m / 2 < a[k - 1]:
        print(3)
        print("1", j, k)
        break
else: print(0)
