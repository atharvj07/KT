#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 100001;
const int MOD = 1e9 + 7;

vector<int> a(MAXN);
vector<int> b(MAXN);
vector<pair<int, int>> c(MAXN);

int main() {
  int m;
  cin >> m;
  int n;
  cin >> n;

  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }

  for (int i = 0; i < m; ++i) {
    cin >> b[i];
    if (b[i] == 1) {
      cin >> c[i].first;
    } else {
      cin >> c[i].first >> c[i].second;
    }
  }

  int cur = 0;
  int length = 0;

  for (int i = 0; i < m; ++i) {
    if (b[i] == 1) {
      ++length;
      a[length - 1] = c[i].first;
    } else {
      for (int j = cur; j < cur + c[i].first; ++j) {
        a[length++] = a[j];
      }
      cur += c[i].first;
      for (int j = 0; j < c[i].second - 1; ++j) {
        for (int k = cur; k < cur + c[i].first; ++k) {
          a[length++] = a[k];
        }
      }
    }
  }
  for(int i=0; i<n; ++i){
    cout << a[a[i] -1] << " ";
  }
  return 0;
}
