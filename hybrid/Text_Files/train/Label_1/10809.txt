#include <bits/stdc++.h>
using namespace std;

int main() {
  int t, n, a, b;
  cin >> t;
  while (t--) {
    cin >> n >> a >> b;
    if (2 * a + b > n - 2 || 2 * b + a > n - 2)
      cout << -1 << endl;
    else {
      vector<int> v(n + 1);
      if (a > 0) {
        v[1] = 1;
        v[2] = 2;
        v[3] = 3;
        int cur = 4;
        while (a--) {
          v[cur] = cur;
          v[cur + 1] = cur + 2;
          v[cur + 2] = cur + 1;
          cur += 3;
        }
        for (int i = 1; i <= n; i++) {
          if (v[i] == 0) {
            v[i] = i;
          }
        }
      }
      if (b > 0) {
        v[1] = 2;
        v[2] = 1;
        v[3] = 3;
        int cur = 4;
        while (b--) {
          v[cur] = cur;
          v[cur + 1] = cur + 1;
          v[cur + 2] = cur + 2;
          cur += 3;
        }
        for (int i = 1; i <= n; i++) {
          if (v[i] == 0) {
            v[i] = i;
          }
        }
      }
      for (int i = 1; i <= n; i++) {
        cout << v[i] << " ";
      }
      cout << endl;
    }
  }
  return 0;
}
