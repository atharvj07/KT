import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] strengths = new int[n];
        for (int i = 0; i < n; i++) {
            strengths[i] = scanner.nextInt();
        }
        int m = scanner.nextInt();
        int[] picks = new int[m];
        int[] bans = new int[m];
        for (int i = 0; i < m; i++) {
            String action = scanner.next();
            int team = scanner.nextInt();
            if (action.equals("p"))
                picks[i] = team;
            else
                bans[i] = team;
        }
        int team1Strength = solve(strengths, picks, bans, 0);
        int team2Strength = solve(strengths, picks, bans, 1);
        System.out.println(team1Strength - team2Strength);
    }

    public static int solve(int[] strengths, int[] picks, int[] bans, int team) {
        int totalStrength = 0;
        int pickCount = 0;
        int banCount = 0;
        int currentTeam = team;
        for (int i = 0; i < picks.length; i++) {
            if (picks[i] == currentTeam) {
                pickCount++;
                if (picks[i] == team) {
                    totalStrength += strengths[i];
                }
            }
            if (bans[i] == currentTeam) {
                banCount++;
            }
            if (pickCount == banCount) {
                currentTeam = 3 - currentTeam;
                pickCount = 0;
                banCount = 0;
            }
        }
        return totalStrength;
    }
}
