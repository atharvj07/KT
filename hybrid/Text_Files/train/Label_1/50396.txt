import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

class Main {
    static final long MOD = 1000000007L;
    static List<Integer>[] tree;
    static Map<Integer, Long> cache;
    static int n, k;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        tree = new List[n + 1];
        cache = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            tree[i] = new ArrayList<>();
        }
        for (int i = 1; i < n; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            tree[u].add(v);
            tree[v].add(u);
        }
        System.out.println(dfs(1, 0, new boolean[n + 1]));
    }

    static long dfs(int u, int p, boolean[] painted) {
        if (cache.containsKey(u)) {
            return cache.get(u);
        }
        long res = 1;
        if (k > 0) {
            res = 2;
            painted[u] = true;
            for (int v : tree[u]) {
                if (v == p) {
                    continue;
                }
                res = (res * (dfs(v, u, painted) + 1)) % MOD;
            }
        } else {
            for (int v : tree[u]) {
                if (v == p) {
                    continue;
                }
                res = (res * dfs(v, u, painted)) % MOD;
            }
        }
        cache.put(u, res);
        return res;
    }
}
