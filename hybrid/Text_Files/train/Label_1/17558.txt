#include <cstdio>
#include <algorithm>
using namespace std;
int main() {
    int t; scanf("%d", &t);
    while (t--) {
        int n; scanf("%d", &n);
        int p, c;
        bool yes = 1;
        while (n--) {
            scanf("%d%d", &p, &c);
            if (p < 0 || c < 0 || c > p) yes = 0;
        }
        puts(yes ? "YES" : "NO");
    }
    return 0;
}
