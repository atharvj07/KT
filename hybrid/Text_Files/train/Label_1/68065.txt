import sys

n = int(input())
a = list(map(int, input().split()))
a.sort()
lst = []
L = 0
R = n - 1
while L < R:
    if a[L] < 0 < a[R]:
        lst.append(L + 1)
        L += 1
    elif a[L] < a[R]:
        lst.append(R + 1)
        R -= 1
    else:
        L += 1

print(len(lst))
for x in lst:
    print(x, end=' ')
