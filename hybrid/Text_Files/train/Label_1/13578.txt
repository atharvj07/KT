#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<int> dp(n, 0);
  if (a[0]) dp[0] = 1;
  for (int i = 1; i < n; i++) {
    if (a[i]) dp[i] = dp[i - 1];
    dp[i] += a[i];
  }
  int ans = 0;
  for (int i = n - 1; i > 0; i--) {
    if (!a[i]) ans += dp[i - 1];
  }
  if (ans == 0) ans = 1;
  cout << ans << endl;
  return 0;
}
