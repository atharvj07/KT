def max_score(a, b):
    # If there are no "x" cards, the maximum score is simply a^2.
    if b == 0:
        return a**2, "o"*a
    # If there are no "o" cards, the maximum score is the negative of b^2.
    if a == 0:
        return -b**2, "x"*b

    # The optimal strategy is to divide the "o" cards into (b+1) blocks as evenly as possible.
    # This minimizes the score reduction from the "x" cards while maximizing the score from "o" cards.

    # Calculate the size of each "o" block and the remainder.
    per_block = a // (b + 1)
    remainder = a % (b + 1)

    # Initialize the score and the result string.
    score = 0
    result = ""

    # For each block of "o" cards...
    for i in range(b + 1):
        # Add the "o" cards for this block.
        block_size = per_block + (1 if i < remainder else 0)
        score += block_size ** 2
        result += "o" * block_size

        # Add an "x" card between the "o" blocks, except after the last one.
        if i < b:
            score -= 1
            result += "x"

    return score, result

# Read input
a, b = map(int, input().split())

# Calculate the maximum score and the corresponding arrangement.
score, arrangement = max_score(a, b)

# Print the results.
print(score)
print(arrangement)
