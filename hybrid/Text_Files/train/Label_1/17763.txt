#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
 
int main() {
    int n, k;
    string a[1501];
    cin >> n >> k;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            for (int l = j + 1; l < n; l++) {
                int cnt[30] = {0};
                for (int p = 0; p < k; p++) {
                    cnt[a[i][p] - 'S']++;
                    cnt[a[j][p] - 'S']++;
                    cnt[a[l][p] - 'S']++;
                }
                int sum = 0;
                for (int p = 0; p < 3; p++) {
                    if (cnt[p] == 1 || cnt[p] == 3) sum++;
                }
                ans += (sum == k);
            }
        }
    }
    cout << ans << endl;
    return 0;
}
