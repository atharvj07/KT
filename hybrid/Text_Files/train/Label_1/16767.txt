#include <cstdio>
#include <set>
#include <tuple>
#include <utility>
#include <vector>

using namespace std;

const int MAXN = 2010;

int n, x, ans;
int h[MAXN];
pair<int, int> dp[2][MAXN];
vector<tuple<int, int, int>> candies[MAXN];

int main() {
  scanf("%d%d", &n, &x);
  for (int i = 1; i <= n; i++) {
    int t, h, m;
    scanf("%d%d%d", &t, &h, &m);
    candies[h].emplace_back(t, h, m);
  }
  dp[0][0] = make_pair(0, 0);
  for (int i = 1; i <= n; i++) {
    int t = i % 2, nt = (i + 1) % 2;
    dp[t][0] = make_pair(0, 0);
    for (int j = 0; j < candies[i].size(); j++) {
      int type, height, mass;
      tie(type, height, mass) = candies[i][j];
      int last_type = dp[nt][max(0, height - x)].first;
      if (type != last_type) {
        dp[t][height] = make_pair(type, mass);
        int &new_ans = dp[t][i].second;
        new_ans = dp[nt][max(0, height - x)].second;
        new_ans += mass;
        ans = max(ans, new_ans);
      }
    }
  }
  printf("%d\n", ans);
  return 0;
}
