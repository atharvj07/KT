#include<bits/stdc++.h>
using namespace std;
#define fastio ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
#define int ll
#define ll long long
#define yes cout<<"First"<<endl
#define no cout<<"Second"<<endl
int32_t main(){
    fastio;
    int n,k;
    cin>>n>>k;
    string arr[n+1];
    map<string,int> mp;
    for(int i = 1; i<=n; i++){
        cin>>arr[i];
        mp[arr[i]] = 1;
    }
    string now  = "";
    int cnt = 0;
    while(cnt<k){
        cnt++;
        int fl = 0;
        for(int i = 0; i<26; i++){
            string sb = now;
            sb+=(i+'a');
            if(mp[sb]==1){
                fl = 1;
                now+=i+'a';
                break;
            }
        }
        if(fl==0){
            if(cnt%2==0){
                yes;
            }
            else
                no;
            return 0;
        }
    }
    if(cnt%2==0){
        no;
    }
    else
        yes;
    return 0;
}
