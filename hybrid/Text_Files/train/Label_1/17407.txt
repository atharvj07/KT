#include <bits/stdc++.h>
using namespace std;

const int maxn = 2e5 + 5;
int n;
long long a[maxn], ans, d[maxn];
vector<int> g[maxn];

void dfs(int u, int p) {
    for (auto v : g[u]) {
        if (v == p) continue;
        d[v] = d[u] + a[v];
        dfs(v, u);
    }
}

void dfs2(int u, int p) {
    for (auto v : g[u]) {
        if (v == p) continue;
        ans = max(ans, d[u] + d[v] - d[1]);
        dfs2(v, u);
    }
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 1);
    dfs2(1, 1);
    cout << ans << endl;
    return 0;
}
