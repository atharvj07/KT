import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int D = sc.nextInt();
        long[] A = new long[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextLong();
        }
        Arrays.sort(A);
        long sum = 0;
        for (int i = 0; i < N - 1; i++) {
            sum += (N - 1 - i) * D + A[i] + A[i + 1];
        }
        System.out.println(sum);
    }
}
