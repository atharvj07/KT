#include <iostream>

using namespace std;

int main() {
  char grid[4][4];
  for (int i = 0; i < 4; i++) {
    for (int j = 0; j < 4; j++) { cin >> grid[i][j]; }
  }
  bool win = false;
  for (int i = 0; i < 4; i++) {
    for (int j = 0; j < 4; j++) {
      if (grid[i][j] == '.') {
        grid[i][j] = 'x';
        bool row = true, column = true, diag1 = true, diag2 = true;
        for (int k = 0; k < 4; k++) {
          row &= grid[i][k] == 'x';
          column &= grid[k][j] == 'x';
          diag1 &= grid[k][k] == 'x';
          diag2 &= grid[k][3 - k] == 'x';
        }
        grid[i][j] = '.';
        win |= row || column || diag1 || diag2;
      }
    }
  }
  cout << (win ? "YES" : "NO") << endl;
  return 0;
}
