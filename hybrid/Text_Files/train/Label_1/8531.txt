#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int a[n];
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int sum = 0;
  for (int i = 0; i < n; i++) { sum += min(a[i], m); }
  int ans = sum;
  for (int i = 0; i < n; i++) {
    if (a[i] > m) { ans -= m; }
  }
  cout << ans << endl;
  return 0;
}
