#include <bits/stdc++.h>
using namespace std;
#define endl "\n"

typedef long long ll;

const int MAXN = 1e5 + 10;
ll a[MAXN], b[MAXN];

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int n, r, avg;
    cin >> n >> r >> avg;

    ll total_grade = 0, total_rewrite = 0;
    for (int i = 0; i < n; ++i) {
        cin >> a[i] >> b[i];
        total_grade += a[i];
    }

    ll need_grade = (avg * 1LL * n) - total_grade;
    if (need_grade <= 0) {
        cout << 0 << endl;
        return 0;
    }

    sort(a, a + n);
    sort(b, b + n);

    for (int i = 0; i < n; ++i) {
        if (need_grade <= 0) {
            break;
        }
        if (a[i] < r) {
            if (b[i] >= need_grade) {
                total_rewrite += need_grade;
                need_grade = 0;
            } else {
                total_rewrite += b[i];
                need_grade -= b[i];
            }
        }
    }
    if (need_grade > 0) {
        total_rewrite = -1;
    }
    cout << total_rewrite << endl;

    return 0;
}
