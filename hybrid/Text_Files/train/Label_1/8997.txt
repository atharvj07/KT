#include <iostream>
#include <queue>
#include <vector>
#include <algorithm>
using namespace std;

using Graph = vector<vector<int>>;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, m;
  cin >> n >> m;
  Graph g(n + 1);
  vector<int> deg(n + 1), ans;
  while (m--) {
    int x, y;
    cin >> x >> y;
    g[x].push_back(y);
    deg[y]++;
  }
  priority_queue<int, vector<int>, greater<int>> pq;
  for (int i = 1; i <= n; i++)
    if (!deg[i]) pq.push(i);
  while (!pq.empty()) {
    int u = pq.top();
    pq.pop();
    ans.push_back(u);
    for (int v : g[u]) {
      deg[v]--;
      if (!deg[v]) pq.push(v);
    }
  }
  if (ans.size() == n) {
    for (int v : ans) cout << v << ' ';
    cout << '\n';
  } else
    cout << -1 << '\n';
  return 0;
}
