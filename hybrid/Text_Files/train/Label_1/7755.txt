#include <bits/stdc++.h>

using namespace std;

long long mod = 1e9 + 7;

long long big_mod(long long base, long long pow) {
  if (pow == 0) return 1;
  long long ret = big_mod(base, pow / 2);
  ret = (ret * ret) % mod;
  if (pow % 2 == 1) ret = (ret * base) % mod;
  return ret;
}

long long ans_mod = big_mod(2, 64);

int main() {
  long long t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    long long ans = (n * ans_mod) % mod;
    cout << ans << endl;
  }
  return 0;
}
