import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int k1 = Integer.parseInt(line[0]);
        int k2 = Integer.parseInt(line[1]);
        int k3 = Integer.parseInt(line[2]);
        int[] a1 = new int[k1];
        int[] a2 = new int[k2];
        int[] a3 = new int[k3];

        line = reader.readLine().split(" ");
        for (int i = 0; i < k1; i++)
            a1[i] = Integer.parseInt(line[i]);

        line = reader.readLine().split(" ");
        for (int i = 0; i < k2; i++)
            a2[i] = Integer.parseInt(line[i]);

        line = reader.readLine().split(" ");
        for (int i = 0; i < k3; i++)
            a3[i] = Integer.parseInt(line[i]);

        int minMoves = 0;
        int left = k1 + k2;
        int right = k3;
        int[] cnt = new int[k1 + k2 + k3 + 1];
        for (int i = 0; i < k1; i++)
            cnt[a1[i]]++;
        for (int i = 0; i < k2; i++)
            cnt[a2[i]]++;
        for (int i = 0; i < k3; i++)
            cnt[a3[i]]--;

        for (int i = 1; i <= k1 + k2 + k3; i++) {
            left -= cnt[i];
            right += cnt[i];
            minMoves += Math.min(left, right);
        }

        System.out.println(minMoves);
    }
}
