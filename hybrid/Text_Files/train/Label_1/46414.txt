import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        Arrays.sort(x);
        Set<Integer> ans = new HashSet<>();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int diff = x[j] - x[i];
                if (diff == 0) {
                    continue;
                }
                if ((diff & (diff - 1)) == 0) {
                    ans.add(x[i]);
                    ans.add(x[j]);
                }
            }
        }
        System.out.println(ans.size());
        for (int num : ans) {
            System.out.print(num + " ");
        }
        System.out.println();
    }
}
