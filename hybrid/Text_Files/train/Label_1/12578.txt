#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 5;
const ll INF = LLONG_MAX;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<ll> b(n);
    ll ans = INF;
    for (int i = 0; i < n; i++) {
      ll cur = (k - a[i] % k) % k;
      b[i] = cur;
      ans = min(ans, cur);
    }
    sort(b.begin(), b.end());
    for (int i = 0; i < n; i++) {
      if (b[i] < ans) ans = (ans + b[i]) / 2;
    }
    cout << ans << '\n';
  }
}
