import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        for (int t = 0; t < T; t++) {
            String x = sc.next();
            String y = sc.next();
            int z = x.length() - y.length();
            int ans = y.length() - 1;
            for (int i = y.length() - 1; i >= 0; i--) {
                if (x.charAt(i + z) == '0' && y.charAt(i) == '1') {
                    ans = i - 1;
                    break;
                }
            }
            System.out.println(ans);
        }
    }
}
