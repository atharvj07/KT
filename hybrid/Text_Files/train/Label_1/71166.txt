import sys
input = sys.stdin.readline
n,m=map(int,input().split())
w=[0]+list(map(int,input().split()))
g=[[] for i in range(n+1)]
for i in range(m):
    a,b=map(int,input().split())
    g[a].append(b)
    g[b].append(a)
s=int(input())
f=[0]*(n+1)
ans=[0]
def dfs(now,fa):
    ans[0]+=w[now]
    for i in g[now]:
        if i!=fa:
            if f[i]==0:
                f[i]=1
                dfs(i,now)
dfs(s,-1)
print(ans[0])
