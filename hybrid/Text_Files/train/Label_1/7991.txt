#include <bits/stdc++.h>
using namespace std;

void solve() {
  int n, x;
  cin >> n >> x;
  vector<int> v;
  while (x) {
    v.push_back(x % n);
    x /= n;
  }
  int res = 0, base = 1;
  for (int i = v.size() - 1; i >= 0; i--) {
    res += base * v[i];
    base *= n;
  }
  cout << res;
}

int main() { solve(); }
