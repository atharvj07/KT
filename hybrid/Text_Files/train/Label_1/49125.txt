import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextLong();
        }
        boolean geo = true;
        for (int i = 1; i < n; i++) {
            if (arr[i] * arr[i - 1] < 0) {
                geo = false;
                break;
            }
        }
        if (geo) {
            System.out.println(0);
            return;
        }
        for (int i = 0; i < n; i++) {
            long[] newArr = new long[n - 1];
            for (int j = 0; j < i; j++) {
                newArr[j] = arr[j];
            }
            for (int j = i + 1; j < n; j++) {
                newArr[j - 1] = arr[j];
            }
            boolean newGeo = true;
            for (int j = 1; j < n - 1; j++) {
                if (newArr[j] * newArr[j - 1] < 0) {
                    newGeo = false;
                    break;
                }
            }
            if (newGeo) {
                System.out.println(1);
                return;
            }
        }
        System.out.println(2);
    }
}
