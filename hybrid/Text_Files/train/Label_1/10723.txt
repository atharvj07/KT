#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5+10;
int n, a[MAXN], m;
int main(){
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> a[i];
    a[n] = m;
    int ans = a[1] - 1;
    for(int i = 1; i < n; i ++){
        ans = max(ans, a[i+1] - a[i] - 1);
    }
    cout << ans << endl;
    return 0;
}
