import sys

t = int(sys.stdin.readline())
for _ in range(t):
    n, x = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    a.sort()
    minAns = sum(i // x for i in a)
    maxAns = sum(a[i] // x for i in range(0, len(a), 2)) + sum((a[i] + a[i + 1]) // x for i in range(1, len(a), 2))
    print(minAns, maxAns)
