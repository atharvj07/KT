import java.util.*;
import java.io.*;
import java.math.*;

class Main {
    static final int MAXN = 300;
    static final int MAXM = 50000;

    static int N, V, M;
    static int[] A = new int[MAXN];
    static int[] B = new int[MAXN];
    static ArrayList<Integer>[] E = new ArrayList[MAXN];

    static int[][] cap = new int[MAXN][MAXN];
    static int[][] flow = new int[MAXN][MAXN];
    static int[][] cost = new int[MAXN][MAXN];

    static int[] ans;

    static int Q = 0;
    static int[] par = new int[MAXN];
    static long[] dist = new long[MAXN];
    static boolean[] vis = new boolean[MAXN];
    static void spfa() {
        Arrays.fill(dist, Long.MAX_VALUE);
        Queue<Integer> que = new LinkedList<>();
        que.add(0);
        dist[0] = 0;
        vis[0] = true;
        while (!que.isEmpty()) {
            int u = que.poll();
            vis[u] = false;
            for (int v = 0; v < MAXN; v++) {
                if (cap[u][v] - flow[u][v] > 0 && dist[v] > dist[u] + cost[u][v]) {
                    dist[v] = dist[u] + cost[u][v];
                    par[v] = u;
                    if (!vis[v]) {
                        que.add(v);
                        vis[v] = true;
                    }
                }
            }
        }
    }

    static long maxflow() {
        long f = 0;
        while (true) {
            Arrays.fill(par, -1);
            spfa();
            if (par[MAXN - 1] == -1) {
                break;
            }
            long mf = Long.MAX_VALUE;
            for (int u = MAXN - 1; u != 0; u = par[u]) {
                mf = Math.min(mf, cap[par[u]][u] - flow[par[u]][u]);
            }
            for (int u = MAXN - 1; u != 0; u = par[u]) {
                flow[par[u]][u] += mf;
                flow[u][par[u]] -= mf;
            }
            f += mf;
        }
        return f;
    }

    static void dfs(int u) {
        if (u == 0) {
            return;
        }
        if (flow[u][0] > 0) {
            dfs(par[u]);
            ans[Q++] = -flow[u][0];
        } else if (flow[0][u] > 0) {
            dfs(par[u]);
            ans[Q++] = flow[0][u];
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        N = Integer.parseInt(st.nextToken());
        V = Integer.parseInt(st.nextToken());
        M = Integer.parseInt(st.nextToken());
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= N; i++) {
            A[i] = Integer.parseInt(st.nextToken());
        }
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= N; i++) {
            B[i] = Integer.parseInt(st.nextToken());
        }
        for (int i = 1; i <= N; i++) {
            E[i] = new ArrayList<>();
        }
        for (int i = 1; i <= M; i++) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            E[x].add(y);
            E[y].add(x);
        }
        for (int i = 1; i <= N; i++) {
            for (int j : E[i]) {
                cap[i][j] = V;
                cap[j][i] = V;
            }
        }
        for (int i = 1; i <= N; i++) {
            cap[0][i] = A[i];
        }
        for (int i = 1; i <= N; i++) {
            cap[i][MAXN - 1] = B[i];
        }
        long f = maxflow();
        if (f != V * N) {
            System.out.println("NO");
            return;
        }
        ans = new int[MAXN * MAXN * 2];
        dfs(MAXN - 1);
        System.out.println(Q);
        for (int i = 0; i < Q; i++) {
            System.out.println(ans[i]);
        }
    }
}
