n, m, k = map(int, input().split())
timetable = [input().strip() for _ in range(n)]

def min_hours(n, m, k, timetable):
    total_hours = 0

    for day in timetable:
        if '1' not in day:
            continue
        first_lesson = day.find('1')
        last_lesson = day.rfind('1')
        hours = last_lesson - first_lesson + 1
        
        # Calculate minimum hours by skipping lessons
        hours_to_skip = min(hours - 1, k)
        k -= hours_to_skip
        total_hours += hours - hours_to_skip
        
    return total_hours

print(min_hours(n, m, k, timetable))
