#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, s;
  cin >> n >> s;

  vector<int> dp(s + 1, INT32_MAX);
  dp[0] = 0;
  for (int i = 1; i <= s; i++) {
    for (int j = 1; j <= n; j++) {
      if (i - j >= 0 && dp[i - j] != INT32_MAX) {
        dp[i] = min(dp[i], dp[i - j] + 1);
      }
    }
  }

  if (dp[s] == INT32_MAX) {
    cout << -1 << endl;
  } else {
    cout << dp[s] << endl;
  }

  return 0;
}
