#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main()
{
    ll n;
    ll start = LLONG_MAX;
    ll end = LLONG_MIN;
    ll ans = 0;
    cin >> n;
    while (n--)
    {
        ll x, h;
        cin >> x >> h;
        start = min(start, x - h);
        end = max(end, x + h);
        if (start <= end)
            ans++;
    }
    cout << ans;
    return 0;
}
