#include <bits/stdc++.h>
using namespace std ;
const int size = 25 ;
int n , m , a[size][size] ;
bool f[size][size] ;

int main() {
  cin >> n >> m ;
  for (int i = 1 ; i <= n ; i++) {
    for (int j = 1 ; j <= m ; j++) {
      cin >> a[i][j] ;
    }
  }
  for (int i = 1 ; i <= n ; i++) {
    for (int j = 1 ; j <= m ; j++) {
      int x = a[i][j] ;
      if (a[i][x] == j) {
        f[i][j] = 1 ;
      }
    }
  }
  bool ok = 1 ;
  for (int i = 1 ; i <= n ; i++) {
    int cnt = 0 ;
    for (int j = 1 ; j <= m ; j++) {
      cnt += (f[i][j]) ;
    }
    if (cnt != m) {
      ok = 0 ;
      break ;
    }
  }
  if (ok) cout << "YES" << endl ;
  else cout << "NO" << endl ;
  return 0 ;
}
