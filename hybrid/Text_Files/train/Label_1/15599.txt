#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> slimes(n, 1);
  for (int i = 1; i < n; i++) {
    int slime;
    cin >> slime;
    slimes.push_back(slime);
  }
  vector<int> ans;
  for (int i = 0; i < n; i++) {
    if (ans.empty() || ans.back() != slimes[i]) ans.push_back(slimes[i]);
    else {
      ans.pop_back();
      ans.push_back(slimes[i] + 1);
    }
  }
  for (int slime : ans) cout << slime << " ";
  cout << endl;
  return 0;
}
