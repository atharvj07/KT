#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, a;
    cin >> n >> a;
    vector<int> d(n);
    for (int i = 0; i < n; i++) cin >> d[i];
    vector<int> ans(n);
    for (int i = 0; i < n; i++) {
        int sum = 0;
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            sum += d[j];
        }
        if (sum > a) {
            cout << d[i] << " ";
            continue;
        }
        int l = max(1, a - sum);
        int r = d[i] - 1;
        if (r < l) {
            cout << d[i] << " ";
            continue;
        }
        ans[i] = r - l + 1;
    }
    for (int i = 0; i < n; i++) cout << ans[i] << " ";
    cout << endl;
    return 0;
}
