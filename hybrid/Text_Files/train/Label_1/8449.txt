#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, c;
    cin >> n >> c;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    int ans = 0;
    for (int l = 0, r = 0; l < n;) {
        while (r < n && a[r] <= c) r++;
        ans = max(ans, r - l);
        while (l < r && a[l] >= c) l++;
    }

    cout << ans << '\n';
    return 0;
}
