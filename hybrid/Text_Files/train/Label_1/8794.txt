#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<long long> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a.begin(), a.end());
        long long sum = 0;
        int ans = 0;
        for (int i = n - 1; i >= 0; i--) {
            sum += a[i];
            if (sum >= x) {
                ans = n - i;
                break;
            }
        }
        cout << ans << endl;
    }
}
