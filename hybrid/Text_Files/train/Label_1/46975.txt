import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int cnt = 0, vacant = 0;
        while (n-- > 0) {
            int x = sc.nextInt();
            if (x == -1) {
                if (vacant == 0) cnt++;
                else vacant--;
            } else vacant += x;
        }
        System.out.println(cnt);
    }
}
