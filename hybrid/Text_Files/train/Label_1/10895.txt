#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n), b(m);
  for (int &x : a) cin >> x;
  for (int &x : b) cin >> x;
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  if (a == b) cout << "NO" << endl;
  else cout << "YES" << endl;
  return 0;
}
