import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

class Edge {
    int u, v, w;

    public Edge(int u, int v, int w) {
        this.u = u;
        this.v = v;
        this.w = w;
    }
}

class City {
    int id;
    List<Edge> edges;

    public City(int id) {
        this.id = id;
        this.edges = new ArrayList<>();
    }
}

public class Valerian {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        City[] cities = new City[n + 1];
        for (int i = 1; i <= n; i++) {
            cities[i] = new City(i);
        }

        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            int w = scanner.nextInt();
            cities[u].edges.add(new Edge(u, v, w));
            cities[v].edges.add(new Edge(v, u, w));
        }

        int minDistance = Integer.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            minDistance = Math.min(minDistance, findDistance(cities, i));
        }

        System.out.println(minDistance);
    }

    private static int findDistance(City[] cities, int startCity) {
        int distance = 0;
        boolean[] visited = new boolean[cities.length];
        visited[startCity] = true;

        for (Edge edge : cities[startCity].edges) {
            if (!visited[edge.v]) {
                distance += edge.w;
                distance += findDistance(cities, edge.v);
                visited[edge.v] = true;
            }
        }

        return distance;
    }
}
