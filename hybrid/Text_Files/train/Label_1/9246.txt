#include<bits/stdc++.h>
using namespace std;
#define MAX 500

int a[MAX+1];
int lps[MAX+1][MAX+1];

int main(){
	int n, i, j, k, l;
	cin>>n;
	for(i=1; i<=n; i++) cin>>a[i];
	for(i=1; i<=n; i++) lps[i][i]=1;
	for(l=2; l<=n; l++){
		for(i=1; i<=n-l+1; i++){
			j=i+l-1;
			if(l==2 && a[i]==a[j]) lps[i][j]=2;
			else if(a[i]==a[j] && lps[i+1][j-1]==l-2) lps[i][j]=l;
			else lps[i][j]=max(lps[i][j-1], lps[i+1][j]);
		}
	}
	cout<<(n-lps[1][n])/2<<endl;
}
