#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n; scanf("%d", &n);
    vector<int> v(n); for(int& x: v) scanf("%d", &x);
    sort(v.begin(), v.end());
    int sum = 0;
    for(int i = 1; i < n; ++i) sum += v[i - 1];
    printf("%d\n", sum);
    return 0;
}
