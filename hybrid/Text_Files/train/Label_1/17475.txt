#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define pb push_back
const int mxN = 1e5 + 1;
int n, b;
vi odd, even;
int main() {
    cin >> n >> b;
    int x, sum = 0;
    for (int i = 0; i < n; ++i) {
        cin >> x;
        if (x % 2) odd.pb(x); else even.pb(x);
    }
    sort(odd.begin(), odd.end());
    sort(even.begin(), even.end());
    int l = (even.size() + 1) / 2;
    for (int i = odd.size() - 1, j = even.size() - 1; i >= 0, j >= 0 && b >= 0; --i, --j) {
        sum += min(odd[i], even[j]);
        b -= min(odd[i], even[j]);
        l += 2;
    }
    if (l / 2 <= odd.size() && l / 2 <= even.size()) {
        cout << l / 2 - 1 << '\n';
    }
    else {
        cout << l / 2 << '\n';
    }
    return 0;
}
