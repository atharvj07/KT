#include <bits/stdc++.h>
#define ll long long
using namespace std;

bool cmp(pair<ll, ll> a, pair<ll, ll> b)
{
    return a.first > b.first;
}

int main()
{
    int n, k, x;
    cin >> n >> k >> x;
    vector<pair<ll, ll>> a(n);
    a[0] = {0, 0};
    for (int i = 1; i < n; i++)
    {
        cin >> a[i].first;
        while (!(a[i].first % x) && k--)
        {
            a[i].first /= x;
            a[i].second++;
        }
    }
    sort(a.begin(), a.end(), cmp);
    ll ans = 0;
    for (int i = 0; i < n; i++)
    {
        ans |= a[i].first;
        while (a[i].second--)
            ans |= ans << 1;
    }
    cout << ans;
    return 0;
}
