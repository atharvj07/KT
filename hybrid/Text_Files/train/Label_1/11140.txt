#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  string s;
  int k;
  cin >> s >> k;
  int n = s.size();
  vector<int> cnt(26, 0);
  for (char c : s) cnt[c - 'a']++;
  int ans = k;
  for (int i = 0; i < 26; ++i) ans -= cnt[i] > 0;
  cout << (ans <= 0 ? 0 : ans) << endl;
  return 0;
}
