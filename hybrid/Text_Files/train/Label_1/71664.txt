for _ in range(int(input())):
    n = int(input())
    s = input()
    a = s.count('a')
    b = s.count('b')
    if a == b:
        print(1, n)
    else:
        for l in range(n):
            for r in range(l, n):
                if s[l:r+1].count('a') == s[l:r+1].count('b'):
                    print(l+1, r+1)
                    break
            else:
                continue
            break
        else:
            print(-1, -1)
