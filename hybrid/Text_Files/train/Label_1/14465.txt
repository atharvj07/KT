#include <bits/stdc++.h>
using namespace std;

int n, m;
int a[300005];

int cic(const vector<int>& v) {
  int sz = (int)v.size();
  int ans = -1;
  for (int i = 0; i < sz; ++i) {
    if (v[i] == 1) {
      ans = i;
      break;
    }
  }
  return ans;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    vector<int> g(n + 1), v(n + 1);
    for (int i = 1; i <= n; ++i) { cin >> g[i]; }
    int now = g[1];
    bool fl = false;
    for (int j = 2; j <= n; ++j) {
      v[j] = now;
      if (g[j] != now + 1) { fl = true; }
      now = g[j];
    }
    if (!fl) {
      v[1] = now;
      int ans = cic(v);
      cout << 1 << '\n' << ans << '\n';
      continue;
    }
    v.clear();
    reverse(g.begin(), g.end());
    now = g[1];
    for (int j = 2; j <= n; ++j) {
      v[j] = now;
      if (g[j] != now + 1) { fl = true; }
      now = g[j];
    }
    if (!fl) {
      v[1] = now;
      int ans = cic(v);
      cout << 1 << '\n' << n - 1 - ans << '\n';
      continue;
    }
    set<int> s;
    for (int i = 1; i <= n; ++i) {
      int tmp = a[i] - g[i];
      if (s.count(tmp) || tmp < 0 || tmp > n - 1) {
        cout << 0 << '\n';
        return 0;
      }
      s.insert(tmp);
    }
    cout << (int)s.size() << '\n';
    for (int x : s) { cout << x << ' '; }
    cout << '\n';
  }
}
