n=int(input())
a=input()
q=[i for i in range(n) if a[i]=='P']
r=[i for i in range(n) if a[i]=='*']
l=len(q)
R=0
while l:
    L=R
    for i in q:
        if i in r:
            l-=1
            r.remove(i)
        else:
            if i+1<n and a[i+1]!='*' and i+1>L:
                L=i+1
            if i-1>=0 and a[i-1]!='*' and i-1>L:
                L=i-1
    if L==R:
        break
    R=L
    q=[i for i in q if i<=R]
print(R)
