import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        HashMap<Integer, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            map.put(a[i], i);
        }
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = map.get(b[i]);
        }
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[c[i]] = i;
        }
        int k = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (d[i] > d[j]) {
                    k++;
                    int temp = d[i];
                    d[i] = d[j];
                    d[j] = temp;
                }
            }
        }
        System.out.println(k);
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (d[i] > d[j]) {
                    System.out.println((i + 1) + " " + (j + 1));
                }
            }
        }
    }
}
