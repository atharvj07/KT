#include <bits/stdc++.h>
#define int long long
#define endl '\n'
using namespace std;

const int N = 2e5 + 1;
int arr[N], lazy[N], idx[N];

bool cmp1(int a, int b) {
    return a > b;
}

void update(int id, int l, int r, int type) {
    if(lazy[id] != 0) {
        if(lazy[id] == 1) {
            sort(arr + l, arr + r);
        } else {
            sort(arr + l, arr + r, cmp1);
        }
        lazy[id] = 0;
    }
}

void push(int id, int l, int r) {
    if(l != r) {
        lazy[id * 2] = lazy[id];
        lazy[id * 2 + 1] = lazy[id];
    }
}

void build(int id, int l, int r) {
    if(l >= r) {
        return;
    }

    int mid = (l + r) / 2;
    build(id * 2, l, mid);
    build(id * 2 + 1, mid + 1, r);
}

void RMQ(int id, int l, int r, int x, int y, int type) {
    update(id, l, r, type);
    if(x > r || y < l) return;
    if(x <= l && y >= r) {
        lazy[id] = type;
        update(id, l, r, type);
        return;
    }

    push(id, l, r);

    int mid = (l + r) / 2;
    RMQ(id * 2, l, mid, x, y, type);
    RMQ(id * 2 + 1, mid + 1, r, x, y, type);
}

int query(int id, int l, int r, int x, int y) {
    update(id, l, r, 1);
    if(x > r || y < l) return -1e9;
    if(x <= l && y >= r) {
        return arr[l];
    }

    push(id, l, r);

    int mid = (l + r) / 2;
    return max(query(id * 2, l, mid, x, y), query(id * 2 + 1, mid + 1, r, x, y));
}

int32_t main() {
    int n, m;
    cin >> n >> m;

    for(int i = 0; i < n; i++) {
        cin >> arr[i];
        idx[arr[i]] = i;
    }

    build(1, 0, n - 1);

    while(m--) {
        int type, l, r;
        cin >> type >> l;
        r = idx[l];
        RMQ(1, 0, n - 1, l, r, type);
    }

    for(int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;

    return 0;
}
