def min_time_to_process_laundry():
    k, n1, n2, n3, t1, t2, t3 = map(int, input().split())
    
    # Initialize time counters for each stage
    wash = [0] * k
    dry = [0] * k
    fold = [0] * k
    
    # Calculate time for washing stage
    for i in range(k):
        if i < n1:
            wash[i] = t1
        else:
            wash[i] = wash[i-n1] + t1
            
    # Calculate time for drying stage
    for i in range(k):
        dry[i] = max(wash[i], (dry[i-n2] if i >= n2 else 0)) + t2
    
    # Calculate time for folding stage
    for i in range(k):
        fold[i] = max(dry[i], (fold[i-n3] if i >= n3 else 0)) + t3
        
    # Last item's fold time is the total time
    return fold[-1]

print(min_time_to_process_laundry())
