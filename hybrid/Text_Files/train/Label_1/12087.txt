#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 10;
int a[N];
vector<int> g[N];
int b[N];
vector<int> c[3];

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    g[a[i]].push_back(i);
  }
  int x = 0, y = 0, z = 0;
  for (int i = 1; i <= n; ++i) {
    if (g[i].size() > 0) {
      if (x == 0) {
        x = i;
      } else if (y == 0) {
        y = i;
      } else if (z == 0) {
        z = i;
      }
      if (x != 0 && y != 0 && z != 0) {
        break;
      }
    }
  }
  if (x != 0 && y != 0 && z != 0) {
    cout << "YES" << endl;
    for (auto v : g[x]) {
      b[v] = 1;
      c[1].push_back(v);
    }
    for (auto v : g[y]) {
      if (b[v] == 1) {
        cout << -1 << endl;
        exit(0);
      }
      b[v] = 2;
      c[2].push_back(v);
    }
    for (int i = 1; i <= n; ++i) {
      if (b[i] == 0) {
        c[3].push_back(i);
        b[i] = 3;
      }
    }
    for (int i = 1; i <= n; ++i) {
      if (g[a[i]].size() % 2 == 0) {
        if (c[3].size() > 0) {
          swap(b[i], b[c[3].back()]);
          c[b[i]].push_back(c[3].back());
          c[3].pop_back();
        } else {
          if (c[2].size() > 0) {
            swap(b[i], b[c[2].back()]);
            c[b[i]].push_back(c[2].back());
            c[2].pop_back();
            swap(b[i], b[c[3].back()]);
            c[b[i]].push_back(c[3].back());
            c[3].pop_back();
          } else {
            swap(b[i], b[c[1].back()]);
            c[b[i]].push_back(c[1].back());
            c[1].pop_back();
            swap(b[i], b[c[2].back()]);
            c[b[i]].push_back(c[2].back());
            c[2].pop_back();
            swap(b[i], b[c[3].back()]);
            c[b[i]].push_back(c[3].back());
            c[3].pop_back();
          }
        }
      }
    }
    for (int i = 1; i <= n; ++i) {
      cout << b[i] << ' ';
    }
    cout << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
