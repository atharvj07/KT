#include <iostream>
#include <vector>
using namespace std;

struct FenwickTree {
    vector<int> bit;
    int n;

    FenwickTree(int n) {
        this->n = n;
        bit.resize(n + 1);
    }

    void update(int x, int val) {
        while (x <= n) {
            bit[x] = max(bit[x], val);
            x += x & (-x);
        }
    }

    int query(int x) {
        int ret = 0;
        while (x >= 1) {
            ret = max(ret, bit[x]);
            x -= x & (-x);
        }
        return ret;
    }
};

int main() {
    int n;
    cin >> n;
    vector<int> a(n + 1);
    vector<int> b(n + 1);
    FenwickTree treeA(n), treeB(n);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        treeA.update(i, a[i]);
    }
    for (int i = 1; i <= n; i++) {
        cin >> b[i];
        treeB.update(i, b[i]);
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        ans = max(ans, treeA.query(i) + treeB.query(i));
    }
    cout << ans << endl;
    return 0;
}
