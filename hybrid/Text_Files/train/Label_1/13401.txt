#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<string> pass(n + 2);
  cin >> pass[0];
  for (int i = 1; i <= n; i++) { cin >> pass[i]; }
  int pen = 0;
  int best_time = 0;
  int worst_time = 0;
  for (int i = 1; i <= n; i++) {
    if (pass[i] == pass[0]) {
      best_time = pen;
      worst_time = pen + i - 1;
      break;
    } else if (pass[i].size() == pass[0].size()) {
      pen++;
    } else {
      pen += 2;
    }
    if (pen > k) { pen = k; }
    worst_time++;
  }
  cout << best_time << " " << worst_time << endl;
}
