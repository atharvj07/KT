#include <iostream>
#include <vector>

typedef long long ll;
const int mod = 1e9 + 7;

ll fastpow(ll a, ll b) {
    ll ans = 1;
    while (b) {
        if (b % 2) ans = (ans * a) % mod;
        a = (a * a) % mod;
        b /= 2;
    }
    return ans;
}

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n, m, k;
        std::cin >> n >> m >> k;
        std::vector<int> a(n);
        for (int i = 0; i < n; ++i) std::cin >> a[i], --a[i];
        std::vector<std::vector<int>> cnt(n, std::vector<int>(n, 0));
        for (int i = 0; i < n; ++i) ++cnt[a[i]][a[i]];
        for (int i = 0; i < n; ++i)
            for (int j = 0; j < n; ++j) {
                if (i + k < n) cnt[i][j] += cnt[i + k][j];
                if (i - k >= 0) cnt[i][j] += cnt[i - k][j];
                cnt[i][j] %= mod;
            }
        ll ans = 0;
        for (int i = 0; i < n; ++i) {
            int cur = 0;
            for (int j = 0; j < n; ++j) { cur += cnt[j][i]; }
            if (i) cur -= cnt[i][i];
            if (m % 2) {
                ans += (ll)cur * cur % mod;
                ans %= mod;
            } else {
                ans += (ll)cur * (cur - 1) % mod;
                ans %= mod;
                ans += (ll)cur * (cur - 1) % mod;
                ans %= mod;
            }
        }
        ans = (ll)ans * fastpow(2, mod - 1 - m / 2) % mod;
        ans %= mod;
        std::cout << ans << std::endl;
    }
    return 0;
}
