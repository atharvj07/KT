import java.util.*;

class Main {
    boolean[][] matrix;
    boolean[][] target;
    int a, b;
    int[][] ans;
    int ans_cnt = 0;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        Main main = new Main();
        main.a = n;
        main.b = m;
        main.matrix = new boolean[n][m];
        main.target = new boolean[n][m];
        main.ans = new int[2502][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                main.matrix[i][j] = in.nextInt() == 1;
                main.target[i][j] = main.matrix[i][j];
            }
        }
        main.work(0, 0);
        if (main.ans_cnt == 0) {
            System.out.println(-1);
            return;
        }
        System.out.println(main.ans_cnt);
        for (int i = 0; i < main.ans_cnt; i++) {
            System.out.println((main.ans[i][0] + 1) + " " + (main.ans[i][1] + 1));
        }
    }

    public boolean work(int x, int y) {
        if (x == a) {
            return check();
        }
        for (int i = x; i < a; i++) {
            for (int j = (i == x ? y : 0); j < b; j++) {
                if (target[i][j] != matrix[i][j]) {
                    target[i][j] = target[i][j + 1] = target[i + 1][j] = target[i + 1][j + 1] = true;
                    ans[ans_cnt][0] = i;
                    ans[ans_cnt][1] = j;
                    ans_cnt++;
                    if (work(i + 1, j + 1)) {
                        return true;
                    }
                    ans_cnt--;
                    target[i][j] = target[i][j + 1] = target[i + 1][j] = target[i + 1][j + 1] = false;
                }
            }
        }
        return false;
    }

    public boolean check() {
        for (int i = 0; i < a; i++) {
            for (int j = 0; j < b; j++) {
                if (target[i][j] != matrix[i][j]) {
                    return false;
                }
            }
        }
        return true;
    }
}
