#include <iostream>
#include <vector>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, k;
        cin >> n >> k;
        if (n % 2 != k % 2 || n < k) {
            cout << "NO" << endl;
            continue;
        }
        cout << "YES" << endl;
        if (n % k == 0) {
            for (int i = 0; i < k - 1; i++) { cout << n / k << ' '; }
            cout << n / k << endl;
            continue;
        }
        long long z = n / k;
        if ((n % k) % 2 == 0) {
            long long x = (n % k) / 2;
            long long y = z - x;
            for (int i = 0; i < k - 1; i++) { cout << y << ' '; }
            cout << x + z << endl;
        } else {
            long long x = (n % k) / 2;
            long long y = z - x - 1;
            for (int i = 0; i < k - 1; i++) { cout << y << ' '; }
            cout << x + 1 + z << endl;
        }
    }
    return 0;
}
