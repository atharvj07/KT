#include<bits/stdc++.h>
using namespace std;

typedef long long ll;
#define endl "\n"
const ll mod = 1e9 + 7;

ll pw(ll x, ll y) {
    ll ret = 1;
    while (y) {
        if (y & 1) ret = (ret * x) % mod;
        x = (x * x) % mod;
        y >>= 1;
    }
    return ret;
}

ll fact[1005];
ll invfact[1005];

void pre() {
    fact[0] = 1;
    for (ll i = 1; i <= 1000; i++) fact[i] = (fact[i - 1] * i) % mod;
    invfact[1000] = pw(fact[1000], mod - 2) % mod;
    for (ll i = 999; i >= 0; i--) invfact[i] = (invfact[i + 1] * (i + 1)) % mod;
}

ll ncr(ll n, ll r) {
    return ((fact[n] * invfact[n - r] % mod) * invfact[r]) % mod;
}

const ll N = 1005;
ll dp[N][N][2][2]; // dp[i][pos][prv][less]

ll solve(ll pos, ll prv, bool less, bool taken) {
    if (pos == -1) {
        if (taken) return 1;
        else return 0;
    }
    ll &ret = dp[pos][prv][less][taken];
    if (~ret) return ret;

    ret = 0;
    for (ll i = 1; i <= N - 1; i++) {
        if (less && i >= prv) continue;
        if (i == prv && taken) continue;

        if (less || i < prv) {
            ret = (ret + solve(pos - 1, i, less, taken | (i == N - 1))) % mod;
        }
        else {
            if (taken) {
                ret = (ret + solve(pos - 1, i, less, taken)) % mod;
            }
            else {
                ret = (ret + ncr(N - 2 - pos, pos) * solve(pos - 1, i, 1, 1)) % mod;
            }
        }
    }
    return ret;
}

int main() {
    pre();
    ll n, x, pos; cin >> n >> x >> pos;
    memset(dp, -1, sizeof dp);
    cout << solve(n - 1, x, 0, 0) << endl;
    return 0;
}
