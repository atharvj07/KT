n, k1, k2 = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

errors = [abs(a[i] - b[i]) for i in range(n)]
k_total = k1 + k2

for _ in range(k_total):
    errors.sort()
    if errors[-1] == 0:
        errors[-1] = 1
    else:
        errors[-1] -= 1

print(sum(e**2 for e in errors))
