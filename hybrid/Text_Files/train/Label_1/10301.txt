#include<bits/stdc++.h>
#define ll long long int
#define endl "\n"
using namespace std;

int main()
{
    ll h,m,s,x,y;
    cin>>h>>m>>s>>x>>y;
    if(x==y) {
        cout<<"YES"<<endl;
        return 0;
    }
    s+=m*60;
    m=0;
    s+=h*60*60;
    h=0;
    ll t1=360LL*x-((360*s)/60)-((360*m)/3600);
    ll t2=360LL*y-((360*s)/60)-((360*m)/3600);
    t1=abs(t1);
    t2=abs(t2);
    if(t1==t2) {
        cout<<"YES"<<endl;
        return 0;
    }
    ll mx=max(t1,t2);
    ll mn=min(t1,t2);
    if((mx-mn)>180) {
        cout<<"NO"<<endl;
        return 0;
    }
    cout<<"YES"<<endl;
    return 0;
}
