#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;

const int N = 2e5 + 5;
pii st[4 * N];
int a[N];
bool ad[N];

void build(int node, int b, int e) {
  if (b == e) {
    st[node] = {a[b], b};
    return;
  }
  int mid = (b + e) / 2;
  build(node * 2, b, mid);
  build(node * 2 + 1, mid + 1, e);
  st[node] = max(st[node * 2], st[node * 2 + 1]);
}

pii query(int node, int b, int e, int l, int r) {
  if (b > r || e < l) {
    return {-1, 0};
  }
  if (b >= l && e <= r) {
    return st[node];
  }
  int mid = (b + e) / 2;
  return max(query(node * 2, b, mid, l, r), query(node * 2 + 1, mid + 1, e, l, r));
}

void update(int node, int b, int e, int pos, int val) {
  if (b > pos || e < pos) {
    return;
  }
  if (b == e) {
    st[node] = {val, b};
    return;
  }
  int mid = (b + e) / 2;
  update(node * 2, b, mid, pos, val);
  update(node * 2 + 1, mid + 1, e, pos, val);
  st[node] = max(st[node * 2], st[node * 2 + 1]);
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  build(1, 1, n);
  for (int i = 1; i <= n; i++) {
    pii ans = query(1, 1, n, i, i);
    if (i != n && !ad[ans.second]) {
      ad[ans.second] = true;
      update(1, 1, n, ans.second, 0);
    }
    ans = query(1, 1, n, i, i);
    cout << ans.first << ' ';
  }
  return 0;
}
