#include<bits/stdc++.h>
using namespace std;
int main() {
    int t, n, x, i, p, j, a[205];
    cin >> t;
    while(t--) {
        cin >> n;
        for(i = 0; i < n; i++) {
            cin >> a[i];
        }
        sort(a, a + n);
        vector<int> v;
        j = n - 1;
        for(i = 0; i < n; i++) {
            if(i & 1) {
                if(a[j] == a[j - 1]) {
                    j--;
                    continue;
                }
                v.push_back(a[j]);
                j--;
            } else {
                v.push_back(a[j - 1]);
            }
        }
        if(j != -1) {
            cout << -1 << endl;
        } else {
            for(i = 0; i < n; i++) {
                cout << v[i] << " ";
            }
            cout << endl;
        }
    }
    return 0;
}
