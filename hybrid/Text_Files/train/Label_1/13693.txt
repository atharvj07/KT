#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, q;
  cin >> n >> q;
  vector<long long> a(n), k(q);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < q; i++) { cin >> k[i]; }
  vector<long long> blood(n, 0);

  for (int i = 0; i < q; i++) {
    for (int j = 0; j < n; j++) {
      if (blood[j] <= 0) {
        if (k[i] >= a[j]) {
          k[i] -= a[j];
          blood[j] = -1;
        } else {
          a[j] -= k[i];
          blood[j] -= k[i];
          break;
        }
      }
    }
    int cnt = 0;
    for (int j = 0; j < n; j++) {
      if (blood[j] <= 0) { cnt++; }
    }
    cout << cnt << endl;
  }
  return 0;
}
