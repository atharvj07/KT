#include <bits/stdc++.h>

using namespace std;

using ll = long long;

struct edge { int to, w; };

const int N = 2e5 + 10, M = 2e5 + 10;

vector<ll> a(N);
vector<int> dp(N);
vector<edge> g[N];
int n;

ll dfs1(int v, int p) {
    ll sum = 0;
    for (auto e : g[v])
        if (e.to != p) sum += dfs1(e.to, v) + e.w;
    return dp[v] = sum;
}

ll dfs2(int v, int p) {
    ll sum = 0;
    for (auto e : g[v])
        if (e.to != p) sum += dfs2(e.to, v) + e.w * (dp[v] - dp[e.to]);
    return sum;
}

int main() {
    ios_base::sync_with_stdio(0), cin.tie(0);
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1, u, v, w; i < n; i++) {
        cin >> u >> v;
        g[u].push_back({v, 1});
        g[v].push_back({u, 1});
    }
    dfs1(1, 1);
    cout << dfs2(1, 1) + *max_element(a.begin() + 1, a.begin() + n + 1) * (ll)n << '\n';
}
