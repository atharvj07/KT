#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;
int32_t main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (auto &i : a) cin >> i;
  sort(a.begin(), a.end());
  int count = 0;
  for (int i = 1; i < n - 1; ++i) {
    if (a[i] != a[i - 1] && a[i] != a[i + 1]) ++count;
  }
  cout << count << endl;
  return 0;
}
