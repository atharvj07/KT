#include<bits/stdc++.h>

using namespace std;

int main()
{
	int a=0, b=0;
	string s[8];
	map<char, int> m {{'Q', 9}, {'R', 5}, {'B', 3}, {'N', 3}, {'P', 1},
	                {'q', 9}, {'r', 5}, {'b', 3}, {'n', 3}, {'p', 1}};

	for(int i=0; i<8; i++)
	{
		cin>>s[i];
		for(int j=0; j<8; j++)
		{
			if(s[i][j]>='A' && s[i][j]<='Z')   a+=m[s[i][j]];
			if(s[i][j]>='a' && s[i][j]<='z')   b+=m[s[i][j]];
		}
	}
	if(a>b)	cout<<"White";
	else if(b>a)	cout<<"Black";
	else	cout<<"Draw";

	return 0;
}
