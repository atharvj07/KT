def solve(k, n, strings):
    if k == 1:
        return strings[0]
    
    # Identify the positions where the strings differ
    diff_positions = []
    for i in range(n):
        if any(strings[0][i] != string[i] for string in strings[1:]):
            diff_positions.append(i)
    
    # If there are more than 4 positions that differ, it's not possible to get the original string
    if len(diff_positions) > 4:
        return -1
    
    # Check all permutations of up to two swaps in the diff positions
    for i in range(len(diff_positions)):
        for j in range(i, len(diff_positions)):
            for string in strings:
                # Swap characters at positions i and j in the string
                swapped = list(string)
                swapped[diff_positions[i]], swapped[diff_positions[j]] = swapped[diff_positions[j]], swapped[diff_positions[i]]
                swapped = "".join(swapped)
                
                # Check if the swapped string matches all other strings with up to two swaps
                if all(sum(swapped[p] != s[p] for p in range(n)) <= 2 for s in strings):
                    return swapped
    
    return -1

k, n = map(int, input().split())
strings = [input().strip() for _ in range(k)]
print(solve(k, n, strings))
