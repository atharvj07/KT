import sys
input = sys.stdin.readline
n = int(input())
A = [0] + list(map(int, input().split()))
B = list(map(int, input().split()))
MX = 10**9+7
dp = [[MX] * (n+1) for i in range(n+1)]
dp[0][0] = 0
for i in range(1, n+1):
    for j in range(i):
        dp[i][j] = min(dp[i][j], dp[j][j] + B[j])
        dp[i][i] = min(dp[i][i], dp[j][j] + A[i] * B[j])
print(dp[n][n])
