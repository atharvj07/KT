import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = input.nextInt();
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                sum += d[i] * d[j];
            }
        }
        System.out.println(sum);
    }
}
