#include <bits/stdc++.h>

#define int long long

using namespace std;

int32_t main() {
  int n;
  cin >> n;
  set<int> a;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    a.insert(x);
  }
  vector<char> ans;
  while (!a.empty()) {
    if (a.count(*a.begin()) < a.count(*a.rbegin())) {
      ans.push_back('L');
      a.erase(*a.begin());
    } else {
      ans.push_back('R');
      a.erase(*a.rbegin());
    }
  }
  cout << ans.size() << "\n";
  for (auto &ch : ans) {
    cout << ch;
  }
  cout << "\n";
  return 0;
}
