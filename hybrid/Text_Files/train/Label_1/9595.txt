#include <bits/stdc++.h>
using namespace std;
using ll=long long;
#define SZ(x) (int)x.size()
#define rep(i,a,b) for(int i=a; i<=b; ++i)
#define per(i,a,b) for(int i=a; i>=b; --i)
#define trav(a,x) for(auto& a : x)
#define all(x) x.begin(), x.end()
#define rall(x) x.rbegin(), x.rend()
typedef pair<ll,ll> pii;
int n;
vector<pii> points;
map<pii, vector<int>> slope;
ll area(int a) {
    return 1ll * points[a].first * points[a].second;
}
int main() {
    cin.sync_with_stdio(0);
    cin.tie(0);
    cin >> n;
    points.resize(n);
    rep(i,0,n-1) {
        cin >> points[i].first >> points[i].second;
        auto [x,y]=points[i];
        if(y==0) continue;
        if(x==0) slope[{0,0}].push_back(i);
        else {
            ll a=1ll * y/__gcd(y,x);
            ll b=1ll * x/__gcd(y,x);
            slope[{a,b}].push_back(i);
        }
    }
    ll sum=0;
    trav(s, slope) {
        int d=SZ(s.second);
        sum+=1ll * d * (d-1) / 2;
        rep(i,0,d-1) {
            rep(j,i+1,d-1) {
                auto [x1,y1]=points[s.second[i]];
                auto [x2,y2]=points[s.second[j]];
                ll area1=x1 * y2 - y1 * x2;
                if(area1==0) continue;
                map<pii,int> have;
                rep(k,0,n-1) {
                    auto [x3,y3]=points[k];
                    auto vec1=pii{x1-x3, y1-y3};
                    auto vec2=pii{x2-x3, y2-y3};
                    ll d1=vec1.first * vec2.second - vec2.first * vec1.second;
                    auto u=pii{d1, area1};
                    if(d1==0) u={area1, 0};
                    else if(d1<0) u={-u.first, -u.second};
                    have[u]++;
                }
                trav(u, have) {
                    if(u.second>1) sum+=1ll * u.second * (u.second-1) / 2;
                }
            }
        }
    }
    cout << sum << '\n';
    return 0;
}
