def solve_caves():
    t = int(input())
    for _ in range(t):
        n = int(input())
        caves = []
        for _ in range(n):
            k_i, *a_i = map(int, input().split())
            caves.append((max(a_i) - len(a_i) + 1, a_i))
        caves.sort(key=lambda x: x[0])
        hero_power = 0
        for req_power, monsters in caves:
            hero_power = max(hero_power, req_power) + len(monsters)
        print(hero_power)

solve_caves()
