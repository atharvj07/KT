#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll *p,*arr,ans[300005],t;
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    for(cin >> t;t--;){
        ll n,m,k=0,cnt=0;
        cin >> n >> m;
        p=new ll[n+5];arr=new ll[n+5];
        for(ll i=0;i<n;i++) {
            cin >> arr[i];
            p[arr[i]]=i;
        }
        for(ll i=0;i<n;i++){
            cnt+=(arr[i]!=i+1);
            if(arr[i]==1){
                k+=i;
                if(p[n]==i) {
                    k+=n;
                    k%=n;
                }
            }
        }
        ans[0]=k;
        if(cnt<=m && !(cnt&1)) {
            if(cnt==m){
                ans[++ans[0]]=k+n/2;
                ans[++ans[0]]=k+n;
            }
            else ans[++ans[0]]=k+n/2;
        }
        cout << ans[0] << " ";
        for(ll i=1;i<=ans[0];i++) cout << ans[i]%n << " ";
        cout << "\n";
    }
}
