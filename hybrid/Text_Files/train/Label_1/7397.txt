#include <bits/stdc++.h>
#define rep(i, l, r) for(int i = (l); i <= (r); ++i)
using namespace std;
int n, k, c[100010];
int main() {
    cin >> n >> k;
    rep(i, 1, n * k) cin >> c[i];
    rep(i, 1, n) {
        int a = i, b = i, las = 0;
        rep(j, 1, n * k) {
            if (c[j] == i && j != las + 1) {
                a = j;
            }
            if (c[j] == i) {
                b = j;
                las = j;
            }
        }
        cout << a << ' ' << b << endl;
    }
}
