#include <bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(0);
    int t;
    cin >> t;
    while(t--) {
        string s;
        cin >> s;
        int n = s.size();
        vector<pair<int,int>> v;
        for(int i=0, j=i+1, bal=0; i<n; i=j, bal=0) {
            while(j<n && (bal += (s[j++]=='1'?1:-1))>=0);
            if(j>i) v.push_back({i,j-1});
        }
        reverse(v.begin(),v.end());
        for(const auto& [l, r]: v) {
            for(int i=l, j=r; i<j; i++, j--) {
                s[i] ^= s[j]^= s[i]^= s[j];
            }
        }
        cout << s << '\n';
    }
    return 0;
}
