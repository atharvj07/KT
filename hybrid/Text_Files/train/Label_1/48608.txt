import java.util.Scanner;

public class Main {

    public static String nextLucky(int n) {
        StringBuilder sb = new StringBuilder();
        while (n > 0) {
            if (n == 1) {
                sb.append('a');
            } else if (n == 2) {
                sb.append('b');
            } else if (n == 3) {
                sb.append('c');
            } else if (n == 4) {
                sb.append('d');
            } else if (n == 5) {
                sb.append('e');
            } else if (n == 6) {
                sb.append('f');
            } else if (n == 7) {
                sb.append('g');
            } else if (n == 8) {
                sb.append('h');
            } else if (n == 9) {
                sb.append('i');
            } else if (n == 10) {
                sb.append('j');
            } else if (n == 11) {
                sb.append('k');
            } else if (n == 12) {
                sb.append('l');
            } else if (n == 13) {
                sb.append('m');
            } else if (n == 14) {
                sb.append('n');
            } else if (n == 15) {
                sb.append('o');
            } else if (n == 16) {
                sb.append('p');
            } else if (n == 17) {
                sb.append('q');
            } else if (n == 18) {
                sb.append('r');
            } else if (n == 19) {
                sb.append('s');
            } else if (n == 20) {
                sb.append('t');
            } else if (n == 21) {
                sb.append('u');
            } else if (n == 22) {
                sb.append('v');
            } else if (n == 23) {
                sb.append('w');
            } else if (n == 24) {
                sb.append('x');
            } else if (n == 25) {
                sb.append('y');
            } else if (n == 26) {
                sb.append('z');
            }
            n -= 4;
        }
        return sb.toString();
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        System.out.println(nextLucky(n));
        sc.close();
    }
}
