#include <bits/stdc++.h>
#define vi vector<int>
using namespace std;
int a[5001], b[5001], f[367][2], m[367][2], d[5001];
char g[5001];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> g[i] >> a[i] >> b[i];
    if (g[i] == 'F') f[a[i]][0]++, f[b[i] + 1][1]++;
    if (g[i] == 'M') m[a[i]][0]++, m[b[i] + 1][1]++;
  }
  for (int i = 1; i <= 366; i++) {
    f[i][0] += f[i - 1][0], f[i][1] += f[i - 1][1];
    m[i][0] += m[i - 1][0], m[i][1] += m[i - 1][1];
  }
  int ans = 0;
  for (int i = 1; i <= 366; i++)
    if (f[i][0] == m[i][0]) ans = max(ans, f[i][0] + m[i][0]);
  cout << ans << endl;
  return 0;
}
