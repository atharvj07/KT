#include <iostream>
#include <vector>
using namespace std;
const int INF = 2e9;
const int MAXN = 1e6;
int dp[MAXN][5];
int main() {
    int q; cin >> q;
    vector<vector<pair<char, pair<int, int>>>> v(26);
    while (q--) {
        int n, k;
        cin >> n >> k;
        string s;
        cin >> s;
        for (int i = 0; i < 5; i++)
            for (int j = 0; j < n; j++)
                dp[j][i] = INF;
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 5; j++) {
                dp[i][j] = min(dp[i + 1][j], dp[i + 1][(j + 1) % 5] + 1);
                if (i + k - 1 < n) {
                    if (i == 0 || s[i - 1] != s[i + k - 1]) dp[i][j] = min(dp[i][j], dp[i + k][j]);
                }
            }
        }
        cout << dp[0][0] << endl;
    }
}
