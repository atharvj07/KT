#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 5, M = 2e5 + 5;

int n, m, t;
int a[N];
map<int, int> mp;
pair<int, int> v[N];

pair<int, int> get(const string &x) {
  int cnt = 0, lst = -1, n = x.size();
  for (int i = 0; i < n; ++i) {
    if (x[i] == 'a' || x[i] == 'e' || x[i] == 'i' || x[i] == 'o' || x[i] == 'u') {
      if (cnt == 0) {
        lst = i;
      }
      ++cnt;
    }
  }
  return {cnt, lst};
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n;
  for (int i = 0; i < n; ++i) {
    string x;
    cin >> x;
    v[i] = get(x);
    ++mp[v[i].first];
  }
  for (auto &x : mp) {
    t += x.second / 2;
  }
  cout << t << "\n";
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < i; ++j) {
      if (v[i].first == v[j].first && v[i].second == v[j].second) {
        cout << i + 1 << " " << j + 1 << "\n";
        --mp[v[i].first];
        if (mp[v[i].first] == 0) {
          mp.erase(v[i].first);
        }
      }
    }
  }
  return 0;
}
