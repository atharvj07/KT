import sys
import heapq

for t in range(int(input())):
    n, s = map(int, input().split())
    a = []
    for i in range(n):
        l, r = map(int, input().split())
        a.append((l, r))
    a.sort()
    l, r = [], []
    for x, y in a:
        l.append(x)
        r.append(y)
    i, j = 0, n - 1
    ans = []
    while j >= i:
        if i == j:
            ans.append((l[i] + r[i]) // 2)
            break
        mid = (l[i] + r[i] + l[j] + r[j]) // 4
        if s >= mid * 2:
            s -= mid * 2
            ans.append(mid)
            i += 1
            j -= 1
        elif s >= l[i] + r[j]:
            s -= l[i] + r[j]
            ans.append((l[i] + r[j]) // 2)
            i += 1
            j -= 1
        else:
            if l[i] <= r[j]:
                ans.append(l[i])
                s -= l[i]
                i += 1
            else:
                ans.append(r[j])
                s -= r[j]
                j -= 1
    heapq.heapify(ans)
    print(ans[n // 2])
