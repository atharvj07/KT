#include<bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;

typedef long long ll;

ll powmod(ll x, ll y) {
    if (y == 0)
        return 1;
    if (y == 1)
        return x;
    if (y % 2 == 0)
        return powmod((x * x) % mod, y / 2);
    else
        return (x * powmod((x * x) % mod, y / 2)) % mod;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n, r;
        cin >> n >> r;
        if (n > r)
            cout << 0 << '\n';
        else
            cout << powmod(r, n) << '\n';
    }
    return 0;
}
