n = int(input())
a = list(map(int, input().split()))
m = int(input())
l = []
r = []
for i in range(m):
    ll, rr = map(int, input().split())
    l.append(ll)
    r.append(rr)

time = 0
for ai in a:
    flag = 0
    for i in range(m):
        if l[i] <= time + ai <= r[i]:
            time += ai
            l[i] = 10 ** 9
            r[i] = 0
            flag = 1
            break
    if flag == 0:
        print(-1)
        exit(0)

print(time)
