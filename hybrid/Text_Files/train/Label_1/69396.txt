from sys import stdin
input = stdin.readline

def solve():
    c, d = map(int, input().split())
    if (c + d) % 2 == 0:
        if (c + d) // 2 >= 0 and (c - d) % 2 == 0 and (c - d) // 2 >= 0:
            return 2
        else:
            return -1
    else:
        return -1

for _ in range(int(input())):
    print(solve())
