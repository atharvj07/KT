#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using vi = vector<int>;
using pi = pair<int, int>;
const int mod = 1e9+7;
const char nl = '\n';

vi adj[100005];
int dist[100005];
int dist2[100005];
int main() {
    int n, m;
    cin >> n >> m;
    map<int, int> mp;
    while(m--) {
        int x, y;
        cin >> x >> y;
        mp[x] = y;
    }
    for(int i = 1; i <= n; ++i) {
        dist[i] = mod;
        dist2[i] = mod;
    }
    queue<int> q;
    q.push(1);
    dist[1] = 0;
    while(!q.empty()) {
        int u = q.front();
        q.pop();
        if(u+1 <= n && dist[u+1] == mod) {
            q.push(u+1);
            dist[u+1] = dist[u] + 1;
        }
        if(u-1 >= 1 && dist[u-1] == mod) {
            q.push(u-1);
            dist[u-1] = dist[u] + 1;
        }
        if(mp.count(u)) {
            if(dist2[mp[u]] == mod) {
                q.push(mp[u]);
                dist2[mp[u]] = dist[u] + 1;
            }
        }
    }
    ll ans = 0;
    for(int i = 1; i <= n; ++i) {
        if(dist2[i] != mod) {
            ans += mp[i];
        }
        else if(dist[i] != mod) {
            ans += mod;
        }
    }
    cout << ans << nl;
    return 0;
}
