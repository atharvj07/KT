#include <bits/stdc++.h>

using namespace std;

#define ll long long
#define vi vector<int>

ll m;
set<ll> s;
map<ll, ll> d;
int n;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0), cout.tie(0);
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    ll l;
    cin >> l;
    s.insert(l);
    d[l] = i;
  }
  if (m == 0 || s.size() == 1 && *s.begin() == m) {
    cout << "-1";
    return 0;
  }
  if (*s.rbegin() >= m) {
    cout << "0";
    return 0;
  }
  ll ans = 0;
  for (ll x : s) {
    while (x < m) {
      ll p = *s.upper_bound(x);
      x = x + p;
      ans += d[p];
    }
    if (x == m)
      break;
    if (x > m) {
      cout << "-1";
      return 0;
    }
  }
  cout << ans;
  return 0;
}
