int gcd(int a,int b){
	if(!b)
		return a;
	return gcd(b,a%b);
}
int32_t main(){
	string s1,s2;
	cin>>s1>>s2;
	int n1=s1.size(),n2=s2.size(),i,j,k;
	int mx=0,divi=1;
	for(i=0;i<n1;++i)
		mx=mx*10+s1[i]-'0';
	for(i=0;i<n2;++i)
		mx=mx*10+s2[i]-'0';
	divi=2;
	if(s1.size()==s2.size()){
		for(i=s1.size()-1;i>=0;--i){
			if(s1[i]!=s2[i])
				break;
		}
		i++;
		while(i--){
			divi*=10;
		}
	}
	cout<<gcd(mx,divi);
	return 0;
}
