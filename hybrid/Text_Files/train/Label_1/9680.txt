#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, x;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> x; }
  cin >> a[0];
  for (int i = 1; i < n; i++) {
    cin >> x;
    a[i] += x;
  }
  x = 0;
  for (int i = 0; i < n; i++) { x = max(a[i], x); }
  cout << x << endl;
  return 0;
}
