#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int ans = 0;
  int cnt = 0, f = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0')
      cnt++;
    else
      f = 0;
    if (cnt >= k + 1)
      f = 1;
    if (f)
      ans = (i + 1) / 2;
  }
  cout << ans << endl;
  return 0;
}
