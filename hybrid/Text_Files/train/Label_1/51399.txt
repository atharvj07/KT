import java.util.Scanner;
import java.util.List;
import java.util.LinkedList;
import java.util.Queue;

public class Main {
    static int W, H, N;
    static int[] E_X = {0, 0, 1, -1};
    static int[] E_Y = {1, -1, 0, 0};
    static int[][] M;

    static class Staff {
        int x, y, T;
        String pat;
        Staff(int x, int y, int T, String pat) {
            this.x = x;
            this.y = y;
            this.T = T;
            this.pat = pat;
        }
        void move() {
            for(char c : pat.toCharArray()) {
                switch(c) {
                    case 'U': y--; break;
                    case 'D': y++; break;
                    case 'L': x--; break;
                    case 'R': x++; break;
                }
                if(x < 0) x = 0;
                if(x >= W) x = W - 1;
                if(y < 0) y = 0;
                if(y >= H) y = H - 1;
            }
            T--;
        }
    }

    static class Cable {
        int x1, y1, x2, y2;
        Cable(int x1, int y1, int x2, int y2) {
            this.x1 = x1;
            this.y1 = y1;
            this.x2 = x2;
            this.y2 = y2;
        }
    }

    static int minStep(Cable c, List<Staff> staffs) {
        int ans = 0;
        for(Staff s : staffs) {
            int sz = c.x2 - c.x1 + 1;
            int u = c.y1 + 1;
            int l = c.x1 + 1;
            int b = c.y2 - 1;
            int r = c.x2 - 1;

            Queue<int[]> q = new LinkedList<>();
            q.add(new int[]{s.x + 1, s.y + 1});
            M[s.x + 1][s.y + 1] = 1;
            while(!q.isEmpty()) {
                int[] cur = q.poll();
                for(int i = 0; i < 4; i++) {
                    int nx = cur[0] + E_X[i];
                    int ny = cur[1] + E_Y[i];
                    if(nx >= 0 && nx < W && ny >= 0 && ny < H && M[nx][ny] == 0) {
                        if(c.x1 - 1 + sz <= nx && nx <= c.x2 - 1 + sz && ny == u) continue;
                        if(nx == l &&  c.y1 - 1 <= ny && ny <= c.y2 - 1) continue;
                        if(c.x1 - 1 + sz <= nx && nx <= c.x2 - 1 + sz && ny == b) continue;
                        if(nx == r && c.y1 <= ny && ny <= c.y2) continue;

                        if(nx == s.x && ny == s.y) {
                            ans += s.T;
                        }
                        q.add(new int[]{nx, ny});
                        M[nx][ny] = 1;
                    }
                }
            }
            for(int i = 0; i < W; i++) {
                for(int j = 0; j < H; j++) {
                    M[i][j] = 0;
                }
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        W = input.nextInt(); H = input.nextInt(); N = input.nextInt();
        int x1 = input.nextInt(), y1 = input.nextInt(), x2 = input.nextInt(), y2 = input.nextInt();
        Cable c = new Cable(x1, y1, x2, y2);
        List<Staff> staffs = new LinkedList<>();
        for(int i = 0; i < N; i++) {
            int x = input.nextInt(), y = input.nextInt();
            int T = input.nextInt(); String pat = input.next();
            Staff s = new Staff(x, y, T, pat);
            staffs.add(s);
        }
        M = new int[W + 2][H + 2];
        System.out.println(minStep(c, staffs));
    }
}
