field = [list(line) for line in open("input.txt", "r")]
for y in range(10):
    for x in range(10):
        if field[y][x] == '.':
            for dx in range(-1, 2):
                for dy in range(-1, 2):
                    if abs(dx) + abs(dy) == 1:
                        count = 1
                        for i in range(1, 5):
                            nx, ny = x + dx * i, y + dy * i
                            if 0 <= nx < 10 and 0 <= ny < 10:
                                count += (field[ny][nx] == 'X')
                            else:
                                break
                        if count >= 5:
                            print("YES")
                            exit(0)
print("NO")
