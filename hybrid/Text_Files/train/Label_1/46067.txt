import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int minIdx = 0;
        int maxIdx = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                minIdx = i;
            } else if (a[i] == n) {
                maxIdx = i;
            }
        }
        System.out.println(Math.max(minIdx, n - 1 - maxIdx));
    }
}
