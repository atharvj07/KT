import heapq

def dijkstra(adj, start):
    n = len(adj)
    dist = [float('inf')] * n
    dist[start] = 0
    pq = [(0, start)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in adj[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    return dist

n, m, k = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(m)]

adj = [[] for _ in range(n)]
for x, y, w in edges:
    adj[x-1].append((y-1, w))
    adj[y-1].append((x-1, w))

all_shortest_paths = []
for i in range(n):
    dist = dijkstra(adj, i)
    all_shortest_paths.extend(dist[i+1:])

all_shortest_paths.sort()
print(all_shortest_paths[k-1])
