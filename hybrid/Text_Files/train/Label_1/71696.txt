import sys
n=int(input())
l2=list(map(int,input().split()))
l3=list(map(int,input().split()))
if l2[0]>l3[0]:
    print(-1)
    sys.exit()
a1=[]
a2=[]
for i in range(1,n):
    if l2[i]>=l2[i-1]:
        a2.append(l2[i])
        a1.append(l3[i])
    else:
        a1.append(l2[i])
        a2.append(l3[i])
    if l2[i]>l3[i]:
        print(-1)
        sys.exit()
a=[a1[0]]
for i in range(1,n-1):
    if a1[i-1]==a1[i]:
        a.append(a1[i])
    else:
        a.append(a2[i-1])
a.append(a2[n-2])
print(*a)
