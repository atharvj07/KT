#include <bits/stdc++.h>
#define endl "\n"
using namespace std;
using ll = long long;

int n, k, p;
vector<int> ans[100005];
bool vis[100005]; 
vector<int> a;

bool dfs(int node, int sum, int cnt) {
  if (cnt == k) return true;
  if (vis[node]) return false;
  vis[node] = true;
  for (auto next : a) {
    if (!vis[next]) {
      if (dfs(next, sum + next, cnt + 1)) {
        ans[cnt].push_back(next);
        return true;
      }
    }
  }
  vis[node] = false;
  for (auto next : a) {
    if (!vis[next]) {
      if (dfs(next, -sum - next, cnt + 1)) {
        ans[cnt].push_back(next);
        return true;
      }
    }
  }
  return false;
}

int main() {
  cin >> n >> k >> p;
  a.resize(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  if (n % 2 == p % 2) {
    cout << "YES\n";
    a.push_back(0); // Add 0 to make n and p have the same parity
    int cnt = 0;
    for (auto x : a) {
      if (!vis[abs(x)]) {
        if (!cnt) {
          if (x == 0) {
            ans[cnt++].push_back(x);
            continue;
          }
        }
        if (dfs(abs(x), abs(x), cnt++)) {
          ans[cnt - 1].push_back(x);
        } else {
          ans[cnt - 1].push_back(-x);
        }
      }
    }
    for (int i = 0; i < k; ++i) {
      cout << ans[i].size() << " ";
      for (auto x : ans[i]) cout << abs(x) << " ";
      cout << endl;
    }
  } else {
    cout << "NO\n";
  }
  return 0;
}
