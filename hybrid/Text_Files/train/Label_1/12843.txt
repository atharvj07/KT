#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 998244353;
const int maxn = 1e5 + 10;
int n, m, dep[maxn], vis[maxn], f[maxn];
vector<pair<int, int>> G[maxn];
void dfs(int x, int d) {
    dep[x] = d;
    vis[x] = 1;
    for (auto &e : G[x]) {
        int y = e.first, w = e.second;
        if (!vis[y]) {
            dfs(y, d + 1);
        }
    }
}
const int inf = 1e9 + 7;
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        int w;
        cin >> w;
        f[i] = w;
    }
    for (int i = 1; i <= m; i++) {
        int u, v;
        cin >> u >> v;
        G[u].push_back({v, 0});
        G[v].push_back({u, 0});
    }
    dfs(1, 1);
    int mx = -inf;
    int idx = -1;
    for (int i = 1; i <= n; i++) {
        if (dep[i] > mx) {
            mx = dep[i];
            idx = i;
        }
    }
    dfs(idx, 1);
    mx = -inf;
    idx = -1;
    for (int i = 1; i <= n; i++) {
        if (dep[i] > mx) {
            mx = dep[i];
            idx = i;
        }
    }
    memset(vis, 0, sizeof(vis));
    dfs(idx, 1);
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) {
            ans = -1;
            break;
        }
        ans = ans + f[i];
    }
    cout << ans << "\n";
    return 0;
}
