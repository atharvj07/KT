n, q = map(int, input().split())
a = list(map(int, input().split()))

reach = [[0] * n for _ in range(20)]
for i in range(n):
    pos = i
    for bit in range(20):
        if a[i] & (1 << bit):
            reach[bit][i] = i
        else:
            if i > 0:
                reach[bit][i] = reach[bit][i - 1]
                pos = min(pos, reach[bit][i])
            else:
                reach[bit][i] = -1
    if i > 0:
        for bit in range(20):
            if a[i] & (1 << bit):
                reach[bit][pos] = i

for _ in range(q):
    x, y = map(int, input().split())
    x -= 1
    y -= 1

    can_reach = False
    for bit in range(20):
        if a[y] & (1 << bit) and reach[bit][x] >= y:
            can_reach = True
            break

    print("Shi" if can_reach else "Fou")
