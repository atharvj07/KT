import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<String, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String abbr = sc.next();
            int year = 0;
            for (int j = abbr.length() - 1; j >= 0; j--) {
                if (Character.isDigit(abbr.charAt(j))) {
                    year = year * 10 + (abbr.charAt(j) - '0');
                } else {
                    break;
                }
            }
            if (!map.containsKey(abbr)) {
                map.put(abbr, year);
            } else {
                year = map.get(abbr) + 1;
                while (map.containsKey(abbr + year)) {
                    year++;
                }
                map.put(abbr + year, year);
            }
            System.out.println(year);
        }
    }
}
