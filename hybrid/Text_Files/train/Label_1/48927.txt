import java.util.Scanner;

public class IvanAndUniversity {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] timetable = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                timetable[i][j] = line.charAt(j) - '0';
            }
        }

        int minHours = solve(timetable, k);
        System.out.println(minHours);
    }

    private static int solve(int[][] timetable, int k) {
        int n = timetable.length;
        int m = timetable[0].length;

        int[][] dp = new int[n + 1][k + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = Integer.MAX_VALUE;
            }
        }

        dp[0][0] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                for (int skip = 0; skip <= k - j; skip++) {
                    int hours = 0;
                    for (int l = 1; l <= m; l++) {
                        if (timetable[i - 1][l - 1] == 1 && skip < l) {
                            hours++;
                        }
                    }
                    dp[i][j + skip] = Math.min(dp[i][j + skip], dp[i - 1][j] + hours);
                }
            }
        }

        return dp[n][k];
    }
}
