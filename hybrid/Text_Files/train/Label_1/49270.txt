import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        // Read input
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }
        scanner.close();

        // Find the maximum number of different tuples
        long maxTuples = 1;
        for (int i = 0; i < n; i++) {
            maxTuples *= p[i];
        }
        maxTuples %= 1000000007;

        // Print the result
        System.out.println(maxTuples);
    }
}
