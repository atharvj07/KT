n = int(input().strip())
for _ in range(n):
    sentence = input().strip()
    if sentence.endswith("lala.") and not sentence.startswith("miao."):
        print("Freda's")
    elif sentence.startswith("miao.") and not sentence.endswith("lala."):
        print("Rainbow's")
    else:
        print("OMG>.< I don't know!")
