#include<bits/stdc++.h>
using namespace std;
using ll=long long;
constexpr ll inf=1e9;
ll n,ans=inf;
map<ll,ll> vis;
void dfs(ll msk,ll sum){
	if((msk&7)==7){
		ans=min(ans,sum);
		return;
	}
	for(ll i=1;i<=n;i++){
		if(!(msk&(1<<(i-1))))
			dfs(msk|(1<<(i-1)),sum+vis[i]);
	}
}
int main(){
	cin>>n;
	for(ll i=1;i<=n;i++){
		ll x;string s;cin>>x>>s;
		for(char j:s)vis[i]|=(1<<(j-'A'));
	}
	dfs(0,0);
	cout<<(ans==inf?-1:ans);
}
