import math
for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    b = []
    for h in a:
        if not b or b[-1] < h:
            b.append(h)
    s = 0
    for h in b:
        if s + h <= n:
            s += h
        else:
            break
    print(s)
