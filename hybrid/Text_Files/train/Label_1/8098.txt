#include <cstdio>
#include <iostream>
using namespace std;

int main() {
  int a, b, c, d;
  cin >> a >> b >> c >> d;
  if (!cin) return 42;
  if ((b - a) * (c - b) == (a - c) * (c - b)) cout << d + c - b;
  else if (b * (c - d) == a * (d - c)) cout << d * (b - a) / (b - c);
  else cout << 42;
  return 0;
}
