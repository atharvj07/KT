import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        boolean[][][] flag = new boolean[105][105][4];
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            flag[x][y][0] = true;
        }
        int ans = 0;
        for (int i = 0; i < 105; i++) {
            for (int j = 0; j < 105; j++) {
                int pt = 0;
                for (int k = 0; k < 4; k++) {
                    if (flag[i][j][k]) pt++;
                }
                if (pt == 4) ans++;
            }
        }
        System.out.println(ans);
    }
}
