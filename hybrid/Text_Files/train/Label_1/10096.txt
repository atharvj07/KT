#include <algorithm>
#include <iostream>

using namespace std;

long long mod = 1000000007;
long long a[200001];

int main() {
  int n, l, r;
  cin >> n >> l >> r;
  for (int i = 1; i <= n; i++) {
    a[i] = l;
    while (a[i] % 3 != (r % 3 + 3 - l % 3) % 3) a[i]++;
  }
  a[n + 1] = r;
  for (int i = n; i > 0; i--) {
    a[i] = min(a[i], r - (a[i + 1] - l));
  }
  long long sum = 0, ans = 1;
  for (int i = 1; i <= n; i++) {
    long long x = (r - a[i]) / 3 - (l - a[i]) / 3 + 1;
    ans = (ans * x) % mod;
    sum += a[i];
  }
  if (sum % 3 == 0) {
    cout << ans << endl;
  } else {
    cout << "0\n";
  }
  return 0;
}
