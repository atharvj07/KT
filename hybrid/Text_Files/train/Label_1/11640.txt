#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
	int t;
	scanf("%d", &t);
	while(t--) {
		int h, n, x, y, z;
		scanf("%d%d", &n, &h);
		z = 0;
		vector <int> v;
		for(int i = 0; i < n; i++) {
			scanf("%d", &x);
			if(!i) y = x;
			v.push_back(x - y);
			z += (v[i] < h);
		}
		if(z > 1) puts("NO");
		else if(v[n - 1] > h - 1) puts("NO");
		else if(n > 2 && *max_element(v.begin(), v.end()) > h - 1) puts("NO");
		else puts("YES");
	}
}
