from math import ceil, log2
from collections import Counter
from itertools import accumulate

n, I = map(int, input().split())
sound = list(map(int, input().split()))

I *= 8
max_values = I // n
if max_values >= log2(n + 1):
    print(0)
else:
    K = 2 ** max_values
    count = Counter(sound)
    frequencies = sorted(count.values(), reverse=True)

    prefix_sum = list(accumulate(frequencies))
    min_changes = n - prefix_sum[K - 1] if K <= len(prefix_sum) else 0

    for i in range(1, len(prefix_sum) - K + 1):
        changes = prefix_sum[-1] - prefix_sum[i + K - 1] + prefix_sum[i - 1]
        min_changes = min(min_changes, changes)

    print(min_changes)
