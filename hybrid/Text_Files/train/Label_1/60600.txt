n,h,l,r=map(int,input().split())
b=[*map(int,input().split())]
b=[min(x,h-l) for x in b]
f=[[-1]*h for _ in range(n+1)]
def dp(idx,t):
    if idx==n: return 0 if l<=t<=r else -1
    if t<0 or t>h: return -1
    if f[idx][t]!=-1: return f[idx][t]
    f[idx][t]=max(1+dp(idx+1,t+b[idx]),dp(idx+1,t))
    return f[idx][t]
print(dp(0,0))
