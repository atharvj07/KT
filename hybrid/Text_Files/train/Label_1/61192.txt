import math

x, y, l, r = map(int, input().split())
a, b = math.ceil(math.log(l, x)), math.ceil(math.log(l, y))
ans = 0
for i in range(int(a), 33):
  for j in range(int(b), 33):
    low, high = int(math.ceil(l ** (1 / i))), int(math.floor(r ** (1 / j)))
    if low ** i + high ** j <= r:
      ans = max(ans, high - low + 1)
print(ans)
