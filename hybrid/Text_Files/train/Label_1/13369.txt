#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

struct Cake {
  int radius;
  int height;
  long double volume;
};

int main() {
  int n;
  cin >> n;
  vector<Cake> cakes(n);
  for (int i = 0; i < n; i++) {
    cin >> cakes[i].radius >> cakes[i].height;
    cakes[i].volume = M_PI * cakes[i].radius * cakes[i].radius * cakes[i].height;
  }
  vector<long double> dp(n);
  dp[0] = cakes[0].volume;
  long double max_volume = 0;
  for (int i = 1; i < n; i++) {
    dp[i] = cakes[i].volume;
    for (int j = 0; j < i; j++) {
      if (cakes[i].volume > cakes[j].volume) {
        dp[i] = max(dp[i], dp[j] + cakes[i].volume);
      }
    }
    max_volume = max(max_volume, dp[i]);
  }
  cout.precision(12);
  cout << fixed << max_volume << endl;
  return 0;
}
