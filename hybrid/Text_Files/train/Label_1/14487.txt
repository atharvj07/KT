#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    int n = s.size(), ans = 0;
    for (int l = 0, r = 0, cnt = 0; r < n; ++r) {
        cnt += s[r] == 'a';
        while ((r - l + 1) * 2 > cnt + (n - r)) --cnt, ++l;
        ans = max(ans, r - l + 1);
    }
    cout << ans << endl;
    return 0;
}
