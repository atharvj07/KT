#include <bits/stdc++.h>
typedef long long ll;
#define rep(i, n) for (int i = 0; i < (int)(n); ++i)
using namespace std;
const ll mod = 1e9 + 7;

int main() {
  int n;
  cin >> n;
  vector<ll> c(n);
  rep(i, n) cin >> c[i];
  int m;
  cin >> m;
  vector<vector<int>> g(n);
  int u, v;
  ll ans = 0, ans2 = 1;
  rep(i, m) {
    cin >> u >> v;
    u--, v--;
    g[u].emplace_back(v);
    g[v].emplace_back(u);
  }
  vector<int> ord(n);
  vector<int> low(n);
  int dfsn = 0, num_cc = 0;
  vector<vector<int>> cc;
  vector<ll> ans_cc(n);
  vector<int> vis(n);

  function<void(int, int)> dfs = [&](int v, int p) {
    vis[v] = true;
    ord[v] = low[v] = dfsn++;
    for (auto u : g[v]) {
      if (!vis[u]) {
        dfs(u, v);
        if (low[u] > ord[v]) {
          cc.emplace_back();
          ll cost = 0;
          while (v != u) {
            cost += c[v];
            ans += c[v];
            cc[num_cc].emplace_back(v);
            low[v] = n;
            v = p;
          }
          ans += c[v];
          cost += c[v];
          cc[num_cc].emplace_back(v);
          ans_cc[num_cc] = cost;
          num_cc++;
        } else {
          low[v] = min(low[v], low[u]);
        }
      } else {
        low[v] = min(low[v], ord[u]);
      }
    }
  };
  rep(i, n) {
    if (!vis[i]) {
      dfs(i, i);
    }
  }
  rep(i, num_cc) {
    for (int j = 1; j < cc[i].size(); ++j) {
      ans2 = ans2 * 2 % mod;
    }
  }
  cout << ans << " " << ans2 << endl;
}
