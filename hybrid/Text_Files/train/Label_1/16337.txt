#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, s, t;
  cin >> n >> s >> t;
  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
  }
  if (s == t) {
    cout << 0 << endl;
    return 0;
  }
  int pos = s;
  int cnt = 0;
  while (pos != t) {
    pos = p[pos - 1];
    cnt++;
    if (cnt > n) {
      cout << -1 << endl;
      return 0;
    }
  }
  cout << cnt << endl;
  return 0;
}
