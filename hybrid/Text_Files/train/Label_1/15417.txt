#include <bits/stdc++.h>
#define ll long long
using namespace std;

int main() {
  int m;
  cin >> m;
  vector<pair<int, int>> cubes;
  for (int i = 0; i < m; ++i) {
    int x, y;
    cin >> x >> y;
    cubes.emplace_back(x, y);
  }
  sort(cubes.begin(), cubes.end());
  ll ans = 0, base = 1;
  vector<pair<int, int>> tower;
  for (auto& cube : cubes) {
    if (cube.second == 0) {
      tower.push_back(cube);
      continue;
    }
    auto next = tower.begin();
    while (next != tower.end()) {
      if (next->second <= cube.second - 1) {
        next = tower.erase(next);
      } else
        break;
    }
    tower.push_back(cube);
    if (tower.size() > next - tower.begin()) {
      for (auto& t : tower) base = base * m % (ll)(1e9 + 9);
      tower.erase(tower.begin(), next);
      ans = (ans + base) % (ll)(1e9 + 9);
    }
  }
  cout << ans << "\n";
}
