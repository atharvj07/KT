import math

def mex(S):
    S_sorted = sorted(S)
    m = 0
    for i in S_sorted:
        if i == m:
            m += 1
        else:
            break
    return m

def solve(n, k, S):
    if k == 0:
        return len(set(S))
    a = mex(S)
    b = max(S)
    if a > b:
        return len(set(S)) + k
    new_element = math.ceil((a + b) / 2)
    S.add(new_element)
    return len(S)

t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().strip().split())
    S = set(map(int, input().strip().split()))
    print(solve(n, k, S))
