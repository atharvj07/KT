#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1005;
int n, m, d, a[N], c[N], dsu[N], rnk[N];
vector<pair<int, int>> ins;
int get(int x) { return dsu[x] == x ? x : dsu[x] = get(dsu[x]); }
bool unite(int x, int y) {
  x = get(x), y = get(y);
  if (x == y) return 0;
  if (rnk[x] > rnk[y])
    swap(x, y);
  dsu[x] = y;
  rnk[y] += rnk[y] == rnk[x];
  return 1;
}
bool ok(int x) {
  for (int i = 1; i <= n + 1; i++) dsu[i] = i, rnk[i] = 1;
  for (int i = 0; i < ins.size(); i++) {
    int l = ins[i].first + x, r = ins[i].second + x;
    if (l > r) swap(l, r);
    if (!unite(l, r)) return 0;
  }
  return get(0) == get(n + 1);
}
int main() {
  cin >> n >> m >> d;
  for (int i = 1; i <= m; i++) {
    cin >> c[i];
    ins.push_back({n - c[i] + 1, n});
  }
  int l = 0, r = n;
  while (l <= r) {
    int mid = (l + r) / 2;
    if (ok(mid))
      r = mid - 1;
    else
      l = mid + 1;
  }
  if (r == n) {
    cout << "NO\n";
    return 0;
  }
  cout << "YES\n";
  for (int i = 1, j = 0; i <= n; i++) {
    if (j == m or i < ins[j].first + l or i > ins[j].second + l)
      a[i] = 0;
    else {
      a[i] = j + 1;
      if (i == ins[j].second + l) j++;
    }
  }
  for (int i = 1; i <= n; i++) cout << a[i] << " ";
  cout << '\n';
}
