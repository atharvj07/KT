import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        String l = sc.nextLine();
        String r = sc.nextLine();
        char c1, c2;
        for (int i = 0; i < n; i++) {
            c1 = l.charAt(i);
            c2 = r.charAt(i);
            if (c1 == '1' && c2 == '0') {
                System.out.print('1');
            } else {
                System.out.print('0');
            }
        }
        System.out.println();
    }
}
