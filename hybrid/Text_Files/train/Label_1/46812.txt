import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            int n = in.nextInt();
            String d = in.next();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = d.charAt(j) - '0';
            }
            if (isBad(a, n)) {
                System.out.println('-');
                continue;
            }
            int[] ans = new int[n];
            boolean[] used = new boolean[10];
            for (int j = 0; j < n; j++) {
                if (used[a[j]] != true) {
                    ans[j] = 1;
                    used[a[j]] = true;
                } else {
                    for (int k = a[j] + 1; k < 10; k++) {
                        if (used[k] != true) {
                            ans[j] = 2;
                            used[k] = true;
                            break;
                        }
                    }
                }
                if (ans[j] == 0) {
                    System.out.println('-');
                    break;
                }
            }
            if (ans[0] != 0) {
                for (int j = 0; j < n; j++) {
                    System.out.print(ans[j]);
                }
                System.out.println();
            }
        }
        in.close();
    }

    public static boolean isBad(int[] a, int n) {
        for (int i = 1; i < n; i++) {
            if (a[i] < a[i - 1]) {
                return true;
            }
        }
        return false;
    }
}
