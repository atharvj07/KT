#include <bits/stdc++.h>
using namespace std;

struct edge {
    int u, v;
    edge(int a, int b) : u(a), v(b) {}
};

int n, m, s, t;
vector<vector<int>> adj;
vector<int> dis;
vector<edge> edges;

void dfs1(int u, int p) {
    dis[u] = dis[p] + 1;
    for (auto v : adj[u]) {
        if (v != p) {
            dfs1(v, u);
        }
    }
}

bool dfs2(int u, int p, int d) {
    if (u == t) {
        return true;
    }
    for (auto v : adj[u]) {
        if (v != p && dis[v] == d) {
            if (dfs2(v, u, d + 1)) {
                return true;
            }
        }
    }
    return false;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m >> s >> t;
    adj.resize(n + 1);
    dis.resize(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
        edges.emplace_back(u, v);
    }
    dfs1(s, 0);
    int ans = 0;
    for (auto [u, v] : edges) {
        if (dis[u] == dis[v]) {
            if (!dfs2(u, v, dis[u] + 1)) {
                ++ans;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
