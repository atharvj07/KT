#include <bits/stdc++.h>
using namespace std;
const int maxn = 100010;
int a[maxn], b[maxn];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  for (int i = 0; i < n; i++) b[i] = a[i];
  for (int i = 0; i < n - 1; i++) {
    if ((a[i] + a[i + 1]) & 1) {
      swap(a[i], a[i + 1]);
      if (a[i] < b[i]) {
        for (int j = i + 1; j < n; j++) b[j] = a[j];
      }
    }
  }
  for (int i = 0; i < n; i++) cout << b[i] << " ";
  cout << endl;
  return 0;
}
