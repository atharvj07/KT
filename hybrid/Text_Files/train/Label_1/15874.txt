#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define PB push_back
#define F first
#define S second
typedef long long ll; 
typedef pair<int, int> pii; 

int n, m, a[101], b[101], c[101], ans[101];
vector<pii> v;

bool check(int day, int id) {
    for (int i = 0; i < id; i++) {
        if (a[i] <= day && day < b[i]) return 0;
    }
    return 1;
}

void solve() {
    bool ok = 1;
    for (int i = 0; i < m; i++) {
        if (check(b[i], i) == 0) ok = 0;
    }
    if (!ok) {
        cout << -1 << endl;
        return;
    }
    for (int i = 0; i < m; i++) {
        int day = a[i];
        while (day < b[i]) {
            if (check(day, i)) {
                ans[day] = i + 1;
            } else {
                ok = 0;
                break;
            }
            day += c[i];
        }
        if (!ok) break;
    }
    for (int i = 0; i < n; i++) {
        cout << ans[i] << " ";
    }
    cout << endl;
}

int main() {
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        cin >> a[i] >> b[i] >> c[i];
    }
    solve();
    return 0;
}
