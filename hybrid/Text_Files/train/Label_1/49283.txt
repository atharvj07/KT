import java.util.*;
import java.lang.*;

class Main {

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);

        int t = cin.nextInt();
        for (int i = 0; i < t; i++) {
            int n = cin.nextInt();
            int[] arr = new int[n];
            int[] freq = new int[101];

            for (int j = 0; j < n; j++) {
                arr[j] = cin.nextInt();
                freq[arr[j]]++;
            }

            int ans = 0;
            for (int j = 0; j <= 100; j++) {
                if (freq[j] == 0) {
                    ans = j;
                    break;
                }
            }

            for (int j = ans + 1; j <= 100; j++) {
                freq[j] -= freq[j - ans - 1];
                if (freq[j] == 0) {
                    ans = j;
                    break;
                }
            }

            System.out.println(ans);
        }
    }
}
