import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(input.readLine());
        Set<String> titles = new HashSet<>();
        for (int i = 0; i < n; i++) {
            titles.add(input.readLine());
        }
        List<String> originalTitles = new ArrayList<>();
        for (int i = 0; i < 26; i++) {
            String title = String.valueOf((char) ('a' + i));
            if (!titles.contains(title)) {
                originalTitles.add(title);
            }
        }
        for (int len = 2; len <= 20; len++) {
            for (int i = 0; i < 26; i++) {
                for (int j = i + 1; j < 26; j++) {
                    String title = String.valueOf((char) ('a' + i)) + String.valueOf((char) ('a' + j));
                    if (!titles.contains(title)) {
                        originalTitles.add(title);
                    }
                }
            }
        }
        Collections.sort(originalTitles);
        System.out.println(originalTitles.get(0));
    }
}
