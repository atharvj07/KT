import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int k = input.nextInt();
        int[] col = new int[4];
        for (int i = 0; i < 4; i++) {
            String s = input.next();
            for (int j = 0; j < 4; j++) {
                if (s.charAt(j) != '.') {
                    col[j]++;
                }
            }
        }
        Arrays.sort(col);
        int max = 0;
        for (int c : col) {
            max += c;
        }
        System.out.println(max <= 2 * k ? "YES" : "NO");
    }
}
