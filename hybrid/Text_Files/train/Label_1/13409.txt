#include <bits/stdc++.h>
using namespace std;
int a[1000010], n, k;
long long d;
int main() {
  cin >> n >> k;
  cin >> a[1];
  for (int i = 2; i <= k; i++) { cin >> a[i]; }
  sort(a + 1, a + k + 1);
  for (int i = 1; i <= k; i++) { d += (long long)a[i] * 1000; }
  int ans = 0;
  for (int i = k; i >= 1; i--) {
    if (a[i] * 1000 <= d - a[i] * 1000) {
      ans += (d - a[i] * 1000) / (a[i] * 1000);
      d -= (ans * (a[i] * 1000));
      if (d == a[i] * 1000) { ans++; }
    } else {
      cout << -1 << endl;
      return 0;
    }
  }
  cout << ans << endl;
  return 0;
}
