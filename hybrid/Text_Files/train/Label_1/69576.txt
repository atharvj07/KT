n, m = input().split()
n, m = int(n), int(m)
p = list(map(int, input().split()))
for _ in range(m):
    l, r, x = input().split()
    l, r, x = int(l), int(r), int(x)
    lx, rx = p[l - 1], p[r - 1]
    p[l - 1:r] = sorted(p[l - 1:r])
    print(['No', 'Yes'][p[x - 1] not in [lx, rx]])
    p[l - 1:r] = sorted(p[l - 1:r])
