from collections import defaultdict, deque

inf = int(1e9)

n, m = map(int, input().split())
g = defaultdict(list)
for i in range(m):
    u, v, w = map(int, input().split())
    g[u].append((v, w))
    g[v].append((u, w))

vis = [False] * (n + 1)
par = [0] * (n + 1)

def dfs(v, p):
    vis[v] = True
    par[v] = p
    for u, w in g[v]:
        if not vis[u]:
            if dfs(u, v):
                return True
        elif u != p:
            return True
    return False

if dfs(1, 0):
    print(-1)
else:
    q = deque()
    r = []
    for i in range(1, n + 1):
        if par[i] != 0:
            q.append((i, par[i]))
    cnt = 0
    while q:
        u, v = q.popleft()
        cnt += 1
        r.append(v)
        g[u].remove((v, w))
        g[v].remove((u, w))
    print(cnt, len(r))
    print(*r)
