import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), d = 0, r = 1, mod = 1000000007;
        int[] a = new int[10];
        for (int i = 0; i < 10; i++) {
            a[i] = in.nextInt();
            d = Math.max(d, a[i]);
        }
        for (int i = 0; i <= 9; i++) {
            for (int j = 0; j < a[i]; j++) {
                r = (int) ((long) r * (n - j) % mod);
            }
        }
        System.out.println(r);
    }
}
