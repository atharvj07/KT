#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll N,x[505],g[505][505],d[505][505];
void floyd(){
  for(ll i=1;i<=N;i++){
    for(ll j=1;j<=N;j++){
      for(ll k=1;k<=N;k++){
        if(d[j][k]>d[j][i]+d[i][k]){
          d[j][k]=d[j][i]+d[i][k];
        }
      }
    }
  }
}
int main(){
  cin>>N;
  for(ll i=1;i<=N;i++){
    for(ll j=1;j<=N;j++){
      cin>>g[i][j];
      if(g[i][j]==0&&i!=j){
        g[i][j]=100000000;
      }
      d[i][j]=g[i][j];
    }
  }
  floyd();
  for(ll i=1;i<=N;i++){
    cin>>x[i];
  }
  for(ll i=1;i<=N;i++){
    ll ans=0;
    for(ll j=1;j<=N;j++){
      for(ll k=1;k<=N;k++){
        if(j!=k){
          ans+=d[j][k];
        }
      }
    }
    cout<<ans<<" ";
    for(ll j=1;j<=N;j++){
      for(ll k=1;k<=N;k++){
        if(j!=x[i]&&k!=x[i]){
          if(d[j][k]>d[j][x[i]]+d[x[i]][k]){
            d[j][k]=d[j][x[i]]+d[x[i]][k];
          }
        }
        else if(j==x[i]){
          for(ll l=1;l<=N;l++){
            if(d[k][l]>d[k][x[i]]+d[x[i]][l]){
              d[k][l]=d[k][x[i]]+d[x[i]][l];
            }
          }
        }
        else if(k==x[i]){
          for(ll l=1;l<=N;l++){
            if(d[j][l]>d[j][x[i]]+d[x[i]][l]){
              d[j][l]=d[j][x[i]]+d[x[i]][l];
            }
          }
        }
      }
    }
  }
  return 0;
}
