#include <bits/stdc++.h>
using namespace std;
int n,a[55];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a, a+n);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                if (a[i] == a[j]) ans += j-i;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
