#include<iostream>
#include<vector>
using namespace std;

const int maxn = 52;
int s[maxn];

int main() {
    int n, ans;
    string str;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> s[i];
    }
    ans = 0;
    str = "";
    for (int i = 0; i < n; i++) {
        while (s[i] > 0) {
            str += i + 'a';
            s[i] --;
        }
    }
    int len = str.length();
    for (int i = 0; i < len / 2; i++) {
        if (str[i] == str[len - 1 - i]) {
            ans ++;
        }
    }
    cout << ans << endl;
    cout << str << endl;
    return 0;
}
