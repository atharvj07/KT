#include <iostream>
#include <iomanip>
#include <cmath>

int main() {
    double a, d;
    std::cin >> a >> d;
    int n;
    std::cin >> n;

    for (int i = 1; i <= n; i++) {
        double x, y;
        if (i % 4 == 1) {
            x = d * i;
            y = 0;
        } else if (i % 4 == 2) {
            x = a;
            y = d * (i - 1);
        } else if (i % 4 == 3) {
            x = a - d * (i - 2);
            y = a;
        } else {
            x = 0;
            y = a - d * (i - 3);
        }
        std::cout << std::fixed << std::setprecision(10) << x << " " << y << std::endl;
    }
    return 0;
}
