#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  string s, t;
  int n;
  cin >> s >> n;
  vector<int> cnt(26);
  for (char c : s) { cnt[c - 'a']++; }
  sort(cnt.rbegin(), cnt.rend());
  int sheets = 0;
  for (int i = 0; i < 26; i++) {
    if (cnt[i] == 0) { break; }
    sheets += (cnt[i] + n - 1) / n;
    for (int j = 0; j < cnt[i]; j += n) { t += char(i + 'a'); }
  }
  if (t.size() < s.size()) {
    cout << -1 << endl;
  } else {
    cout << sheets << endl;
    cout << t.substr(0, n) << endl;
  }
  return 0;
}
