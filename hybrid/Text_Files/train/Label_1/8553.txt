#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, cnt = 0;
  cin >> n;
  vector<int> a(n);
  set<int> ans;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    ans.insert(a[i]); // Insert non-xor values
    for (auto it : ans) ans.insert(it ^ a[i]); // Xor all values with new value
  }
  for (int i = 0; i < n; i++) {
    for (auto it : ans)
      if (ans.find(a[i] ^ it) != ans.end()) cnt++;
  }
  cout << ans.size() << '\n';
  for (auto it : ans) cout << it << ' ';
  cout << '\n';
}
