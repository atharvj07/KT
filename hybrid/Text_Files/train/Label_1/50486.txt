import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt(), x = input.nextInt();
        char[][] keys = new char[n][m];
        boolean[][] shifts = new boolean[n][m];
        Point[][] points = new Point[n][m];
        for (int i = 0; i < n; i++) {
            String line = input.next();
            for (int j = 0; j < m; j++) {
                keys[i][j] = line.charAt(j);
                shifts[i][j] = keys[i][j] == 'S';
                points[i][j] = new Point(i, j);
            }
        }
        int q = input.nextInt();
        String text = input.next();

        int dist[][] = new int[n][m];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                dist[i][j] = Integer.MAX_VALUE;

        LinkedList<Point> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (shifts[i][j]) {
                    queue.add(points[i][j]);
                    dist[i][j] = 0;
                }
            }
        }

        while (!queue.isEmpty()) {
            Point p = queue.poll();
            int i = p.x, j = p.y;
            for (int dx = -x; dx <= x; dx++) {
                for (int dy = -x; dy <= x; dy++) {
                    int nx = i + dx, ny = j + dy;
                    if (0 <= nx && nx < n && 0 <= ny && ny < m && dist[i][j] + 1 < dist[nx][ny]) {
                        dist[nx][ny] = dist[i][j] + 1;
                        queue.add(points[nx][ny]);
                    }
                }
            }
        }

        int ans = 0;
        for (char c : text.toCharArray()) {
            boolean found = false;
            char key = Character.toLowerCase(c);
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (keys[i][j] == key && dist[i][j] <= x) {
                        found = true;
                        break;
                    }
                }
            }
            if (!found) {
                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < m; j++) {
                        if (shifts[i][j] && dist[i][j] <= x) {
                            ans++;
                            found = true;
                            break;
                        }
                    }
                }
            }
            if (!found) {
                System.out.println(-1);
                return;
            }
        }

        System.out.println(ans);
    }

    static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
