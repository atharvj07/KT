#include <algorithm>
#include <cstdio>
#include <vector>

int main() {
    int n, i, j;
    scanf("%d", &n);
    for (i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            std::vector<int> v;
            while (n % i == 0) {
                v.push_back(i);
                n /= i;
            }
            if (n > 0) v.push_back(n);
            printf("%d\n", (int)v.size());
            for (j = 0; j < (int)v.size(); j++) printf("%d ", v[j]);
            return 0;
        }
    }
    printf("1\n%d\n", n);
    return 0;
}
