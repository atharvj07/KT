#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<vector<char>> grid(n, vector<char>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      cin >> grid[i][j];
    }
  }
  int maxCnt = 0;
  pair<int, int> ans;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      bool valid = true;
      for (int x = 0; x < k && valid; x++) {
        if (i + x >= n || grid[i + x][j] == '#') valid = false;
      }
      for (int x = 0; x < k && valid; x++) {
        if (j + x >= n || grid[i][j + x] == '#') valid = false;
      }
      if (valid) {
        int cnt = 0;
        for (int x = 0; x < k; x++) {
          if (i + x < n && grid[i + x][j] != '#') cnt++;
          if (j + x < n && grid[i][j + x] != '#') cnt++;
        }
        if (cnt > maxCnt) {
          maxCnt = cnt;
          ans = {i + 1, j + 1};
        }
      }
    }
  }
  cout << ans.first << " " << ans.second << endl;
  return 0;
}
