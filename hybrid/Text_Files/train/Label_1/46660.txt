import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        String s = input.next();

        int[] pos = new int[m];
        int[] c = new int[m];

        for (int i = 0; i < n; i++) {
            c[s.charAt(i) - 'a']++;
        }

        Arrays.fill(pos, -1);

        int p = 0;
        for (int i = 0; i < m; i++) {
            if (c[i] > 0) {
                pos[i] = p;
                p++;
            }
        }

        int ans = 0;
        for (int i = 1; i < n; i++) {
            ans += Math.abs(pos[s.charAt(i) - 'a'] - pos[s.charAt(i - 1) - 'a']);
        }

        System.out.println(ans);
    }
}
