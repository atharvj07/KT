n, z = map(int, input().split())
x = sorted(int(i) for i in input().split())
matched = [False for _ in range(n)]
ans = 0
for i in range(n):
    if not matched[i]:
        pair = i + 1
        for j in range(i + 1, n):
            if not matched[j] and abs(x[j] - x[i]) >= z:
                pair = j + 1
        if pair != i + 1:
            matched[pair - 1] = True
            ans += 1
print(ans)
