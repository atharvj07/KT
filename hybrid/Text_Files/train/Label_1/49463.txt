import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Scanner;

public class Main {

    private static class Rect {
        public long a;
        public long b;
        public long c;

        public Rect(long a, long b, long c) {
            this.a = a;
            this.b = b;
            this.c = c;
        }
    }

    private static class Pair {
        public Rect r1;
        public Rect r2;
        public long maxVol;

        public Pair(Rect r1, Rect r2, long maxVol) {
            this.r1 = r1;
            this.r2 = r2;
            this.maxVol = maxVol;
        }
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt();

        List<Rect> rects = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            long a = input.nextLong();
            long b = input.nextLong();
            long c = input.nextLong();
            rects.add(new Rect(a, b, c));
        }

        Pair maxPair = null;

        for (Rect r1 : rects) {
            for (Rect r2 : rects) {
                if (r1 != r2) {
                    long maxVol = Math.max(r1.a, r2.a) * Math.max(r1.b, r2.b) * Math.max(r1.c, r2.c);
                    if (maxPair == null || maxPair.maxVol < maxVol) {
                        maxPair = new Pair(r1, r2, maxVol);
                    }
                }
            }
        }

        if (maxPair == null) {
            System.out.println(1);
            System.out.println(1);
        } else {
            System.out.println(2);
            System.out.println(rects.indexOf(maxPair.r1) + 1 + " " + rects.indexOf(maxPair.r2) + 1);
        }
    }
}
