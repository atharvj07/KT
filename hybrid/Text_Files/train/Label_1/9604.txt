#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const int MOD = 998244353;

int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for (ll &i : a) cin >> i;

    vector<ll> pref(n);
    for (ll i = 0; i < n; i++) {
        pref[i] = 1;
        if (i > 0) pref[i] = (pref[i] + pref[i - 1]) % MOD;
    }

    ll ans = pref[a[n - 1] - 1];

    for (ll i = n - 2; i >= 0; i--) {
        ll rem = a[i] - a[i + 1] + 1;
        ans = (ans * pref[rem - 1]) % MOD;
    }

    cout << ans << "\n";
}
