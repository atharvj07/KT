import sys

def solve():
    t = int(input())
    for i in range(t):
        n, q = map(int, input().split())
        s = input()
        s = [1 if c == '+' else -1 for c in s]
        pref = [0] * (n + 1)
        for j in range(n):
            pref[j + 1] = pref[j] + s[j]
        for j in range(q):
            l, r = map(int, input().split())
            print(max(0, (pref[r] - pref[l - 1] + 1) // 2))

solve()
