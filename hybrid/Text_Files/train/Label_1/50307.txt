import java.util.*;
import java.io.*;

public class Main {

    static long[] coins;
    static int[] dp;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int q = input.nextInt();

        coins = new long[n];
        for (int i = 0; i < n; i++) {
            coins[i] = input.nextInt();
        }

        dp = new int[1000001];
        Arrays.fill(dp, -1);
        dp[0] = 0;
        for (long coin : coins) {
            for (int i = 1000000; i >= 0; i--) {
                if (dp[i] != -1) {
                    dp[i + (int) coin] = 0;
                }
            }
        }

        for (int i = 0; i < q; i++) {
            int b = input.nextInt();
            System.out.println(dp[b] == -1 ? -1 : dp[b]);
        }
    }
}
