#include <bits/stdc++.h>

using namespace std;

const int maxn = 10001;

struct rec {
    string t;
    string s;
    int pos;
};

bool operator<(const rec &a, const rec &b) {
    if (a.t != b.t)
        return a.t < b.t;
    return a.s < b.s;
}

int main() {
    freopen("in.txt", "r", stdin);
    int n, m;
    cin >> n >> m;
    string s, t, tmp;
    vector<rec> v;
    while (getline(cin, tmp)) {
        stringstream ss(tmp);
        getline(ss, t, ' ');
        getline(ss, s);
        v.push_back(rec{t, s, v.size()});
    }
    sort(v.begin(), v.end());
    vector<int> cnt(v.size(), 0);
    for (int i = 0; i < v.size(); i++) {
        if (i - n >= 0)
            cnt[i - n] = 0;
        cnt[i]++;
    }
    int ans = -1;
    for (int i = 0; i < v.size(); i++)
        if (cnt[i] >= m) {
            ans = i;
            break;
        }
    if (ans == -1)
        cout << "-1\n";
    else
        cout << v[ans].t << '\n';
    return 0;
}
