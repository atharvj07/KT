#include <iostream>
#include <vector>
using namespace std;
long long ans = 0;
vector<int> close;
vector<int> open;
int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<vector<int>> dp(n + 1, vector<int>(n + 1));
    vector<int> l(n + 1);
    for (int i = 0; i < n; i++) {
        s[i] == '(' ? open.push_back(i) : close.push_back(i);
    }
    dp[n][n - 1] = 1;
    l[n - 1] = n - 1;
    for (int j = n - 2; j >= 0; j--) {
        for (int i = j; i >= 0; i--) {
            if (s[j] == '(') {
                if (j + 1 <= n - 1 && dp[j + 1][j + 1] > 0) {
                    if (l[j + 1] < i) {
                        dp[i][j] += dp[j + 1][j + 1];
                        l[i] = l[j + 1];
                    } else if (l[j + 1] == i) {
                        dp[i][j] += dp[j + 1][j + 1] - 1;
                    }
                }
                for (int k = j + 1; k < n; k++) {
                    if (dp[k + 1][j] > 0) {
                        if (l[k + 1] < i) {
                            dp[i][j] += dp[k + 1][j];
                        }
                    }
                }
            } else {
                for (int k = 0; k < j; k++) {
                    if (dp[j][k + 1] > 0) {
                        if (l[j] < i) {
                            dp[i][j] += dp[j][k + 1];
                            l[i] = l[j];
                        } else if (l[j] == i) {
                            dp[i][j] += dp[j][k + 1] - 1;
                        }
                    }
                }
            }
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = i; j < n; j++) {
            if (s[i] == '(' && s[j] == ')') { ans += dp[i + 1][j - 1]; }
            if (s[i] == '?' && s[j] == ')') { ans += dp[i + 1][j - 1]; }
            if (s[i] == '(' && s[j] == '?') { ans += dp[i + 1][j - 1]; }
            if (s[i] == '?' && s[j] == '?') { ans += dp[i + 1][j - 1]; }
        }
    }
    cout << ans;
    return 0;
}
