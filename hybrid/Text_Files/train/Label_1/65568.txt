MOD = 10**9 + 7

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(x, y):
    if y % x != 0:
        return 0
    y //= x
    ans = 1
    for i in range(2, int(y**0.5) + 1):
        if y % i == 0:
            cnt = 0
            while y % i == 0:
                cnt += 1
                y //= i
            ans = ans * (cnt + 1) % MOD
    if y > 1:
        ans = ans * 2 % MOD
    return power(2, ans - 1, MOD)

x, y = map(int, input().split())
print(solve(x, y))
