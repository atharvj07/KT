n, m = [int(x) for x in input().split()]
a = [input() for _ in range(n)]

ans = -1
for i in range(n):
    for j in range(m):
        if a[i][j] == 'B':
            for k in range(1, min(n - i, m - j) + 1):
                if all(a[ii][jj] == 'B' for ii in range(i, i + k) for jj in range(j, j + k)):
                    ans = max(ans, k * k)
print(ans)
