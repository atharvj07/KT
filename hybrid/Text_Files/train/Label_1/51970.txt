import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        boolean flag = false;
        for (int i = 0; i < s.length() - 1; i++) {
            if (s.charAt(i) != s.charAt(i + 1)) {
                if (flag) {
                    System.out.println("NO");
                    return;
                } else {
                    flag = true;
                }
            }
        }
        System.out.println("YES");
    }
}
