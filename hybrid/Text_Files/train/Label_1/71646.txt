import sys
is_ok = False
for l in sys.stdin.readlines()[0:8]:
    u, v, w = map(int, l.split())
    d = [u, v, w]
    d.sort()

    if d[0]!=0:
        continue
    u=d[1]; v=d[2]
    if u*v>0:
        continue
    if abs(u)<abs(v):
        u,v=v,u
    if u*u-v*v>0:
        continue
    is_ok = True

    print("YES")
    for l in sys.stdin.readlines()[0:8]:
        u, v, w = map(int, l.split())
        print(u*v//v, u*u//v, u)

if not is_ok:
    print("NO")
