#include <iostream>
#include <vector>
#include <algorithm>
#include <queue>
int main() {
  using namespace std;
  int q;
  cin >> q;
  while (q--) {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int ans = 0;
    for (int i = 0; i < k; i++)
      if (s[i] != "RGBRGBRGB"[i % 6]) ans++;
    for (int i = k; i < n; i++) {
      if (s[i] != "RGBRGBRGB"[i % 6]) ans++;
      if (s[i - k] != "RGBRGBRGB"[i % 6]) ans--;
    }
    cout << ans << endl;
  }
}
