#include <cstdio>
#include <cstring>
#include <vector>
#define FOR(i, s, e) for (int i = s; i < e; ++i)

int t, n, m, ans;
char c, ch;
int a[2010][2010];
std::vector<std::pair<std::pair<int, int>, std::pair<int, int>>> res;

int main() {
    scanf("%d", &t);
    FOR(tt, 1, t + 1) {
        res.clear();
        memset(a, 0, sizeof(a));
        scanf("%d %d", &n, &m);
        FOR(i, 0, n) {
            FOR(j, 0, m) {
                scanf(" %c", &c);
                if (c != '.') {
                    if (!a[i][j])
                        a[i][j] = ++ans;
                    if (i > 0 && a[i - 1][j] != a[i][j]) {
                        res.push_back(std::make_pair(std::make_pair(i, j), std::make_pair(i - 1, j)));
                    } else if (j > 0 && a[i][j - 1] != a[i][j]) {
                        res.push_back(std::make_pair(std::make_pair(i, j), std::make_pair(i, j - 1)));
                    }
                }
            }
        }
        FOR(i, 0, n) {
            FOR(j, 0, m) {
                scanf(" %c", &ch);
                if (ch != '.') {
                    if (a[i][j] == 0) {
                        puts("NO");
                        goto done;
                    } else if (c != ch) {
                        puts("NO");
                        goto done;
                    }
                }
            }
        }
        printf("YES\n");
        printf("%d\n", ans);
        for (auto &e : res) {
            printf("%d %d %d %d\n", e.first.first + 1, e.first.second + 1, e.second.first + 1, e.second.second + 1);
        }
    done:
        continue;
    }
    return 0;
}
