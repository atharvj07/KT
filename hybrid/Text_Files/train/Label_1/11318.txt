#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  map<string, int> m;
  for (int i = 0; i < n - 1; i++) { m[s.substr(i, 2)]++; }
  string max_str = "";
  int max_freq = -1;
  for (auto &p : m) {
    if (p.second > max_freq) {
      max_freq = p.second;
      max_str = p.first;
    }
  }
  cout << max_str << endl;
}
