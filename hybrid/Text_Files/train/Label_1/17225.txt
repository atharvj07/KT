#include <cstdio>
#include <vector>
using namespace std;
struct pt
{
    int x,y;
    pt(){x=y=0;}
    pt(int _x,int _y){x=_x;y=_y;}
};
pt operator-(pt a,pt b){return pt(a.x-b.x,a.y-b.y);}
int cross(pt a,pt b){return a.x*b.y-a.y*b.x;}
int area(vector<pt> &v)
{
    int ans=0;
    for(int i=1;i<v.size();i++)
    {
        ans-=cross(v[i-1],v[i]);
    }
    ans+=cross(v[v.size()-1],v[0]);
    return ans/2;
}
int area(pt a,pt b,pt c,pt d)
{
    vector<pt> v;
    v.push_back(a);
    v.push_back(b);
    v.push_back(c);
    v.push_back(d);
    return area(v);
}
int main()
{
    int x1,y1,x2,y2,x3,y3,x4,y4,x5,y5,x6,y6,x7,y7,x8,y8;
    scanf("%d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d",&x1,&y1,&x2,&y2,&x3,&y3,&x4,&y4,&x5,&y5,&x6,&y6,&x7,&y7,&x8,&y8);
    int ans1=area(pt(x1,y1),pt(x2,y2),pt(x3,y3),pt(x4,y4));
    int ans2=area(pt(x5,y5),pt(x6,y6),pt(x7,y7),pt(x8,y8));
    int ans3=area(pt(x1,y1),pt(x2,y2),pt(x7,y7),pt(x8,y8));
    int ans4=area(pt(x5,y5),pt(x6,y6),pt(x3,y3),pt(x4,y4));
    if((ans1+ans2)==ans3||(ans1+ans2)==ans4)puts("YES");
    else puts("NO");
    return 0;
}
