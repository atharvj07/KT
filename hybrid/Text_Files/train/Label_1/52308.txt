import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long k = sc.nextLong();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int[] freq = new int[200001];
        for (int i = 0; i < n; i++) {
            freq[arr[i]]++;
        }

        PriorityQueue<Integer> pq = new PriorityQueue<>(Collections.reverseOrder());
        for (int i = 1; i <= 200000; i++) {
            if (freq[i] > 0) {
                pq.add(freq[i] * i);
            }
        }
        long ans = 0;
        long cnt = 0;
        long prev = 0;
        while (!pq.isEmpty()) {
            long cur = pq.poll();
            if (cur == prev) {
                cnt++;
                continue;
            }
            cnt = 1;
            prev = cur;
            ans += cur;
            if (ans >= k) {
                break;
            }
        }
        if (ans < k) {
            ans = -1;
        }
        System.out.println(ans);
    }
}
