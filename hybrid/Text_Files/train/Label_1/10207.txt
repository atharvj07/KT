#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int N, D;
  cin >> N >> D;
  vector<int> S(N), P(N);
  for (int i = 0; i < N; i++) cin >> S[i];
  for (int i = 0; i < N; i++) cin >> P[i];
  int sum = S[D - 1] + P[0];
  int ans = 1;
  for (int i = 1; i < N; i++) {
    if (S[i - 1] + P[i] == sum)
      ans++;
    else
      break;
  }
  cout << ans << '\n';
  return 0;
}
