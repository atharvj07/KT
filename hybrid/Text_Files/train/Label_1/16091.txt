#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  vector<int> dp(n, 1);
  int last = -1;
  for (int i = 1; i < n; i++) {
    if (s[i] != s[i - 1]) {
      dp[i] = dp[i - 1] + 1;
      last = i;
    }
  }

  int ans = dp[n - 1];
  for (int i = 0; i < n; i++) {
    if (i != last && s[i] == s[i + 1]) {
      ans = max(ans, dp[i] + (i + 1 < n ? dp[i + 1] : 0) + 1);
    }
  }

  cout << ans << endl;

  return 0;
}
