import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            List<Integer> l = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
                if (l.isEmpty() || l.get(l.size() - 1) == a[j]) {
                    l.add(a[j]);
                } else {
                    l.add(0, a[j]);
                }
            }
            for (int j = 0; j < l.size(); j++) {
                System.out.print(l.get(j) + " ");
            }
            System.out.println();
        }
    }
}
