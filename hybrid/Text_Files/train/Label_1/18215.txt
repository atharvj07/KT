#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    string rev(s);
    reverse(rev.begin(), rev.end());
    int ans = n - 1;
    int rev_ans = INT_MAX;
    for (int i = 0; i < n; i++) {
      if (s[i] != s[n - i - 1]) {
        ans = min(ans, max(i, n - i - 1));
      }
    }
    for (int i = 0; i < n; i++) {
      if (rev[i] != rev[n - i - 1]) {
        rev_ans = min(rev_ans, max(i, n - i - 1));
      }
    }
    ans = min(ans, rev_ans);
    if (rev_ans == 0)
      ans = 0;
    if (ans % 2 == 1)
      cout << "BOB\n";
    else if (ans == 0)
      cout << "DRAW\n";
    else
      cout << "ALICE\n";
  }
  return 0;
}
