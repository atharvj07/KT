import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static int n, m;
    static char[][] grid;
    static boolean[][] visited;
    static int[] dx = {0, 0, -1, 1};
    static int[] dy = {1, -1, 0, 0};
    static int min_dist = 99999999;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        grid = new char[n + 2][m + 2];
        visited = new boolean[n + 2][m + 2];
        for (int i = 1; i <= n; i++) {
            String s = sc.next();
            for (int j = 1; j <= m; j++) {
                grid[i][j] = s.charAt(j - 1);
            }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (grid[i][j] != '.' && grid[i][j] != '#') {
                    bfs(i, j, grid[i][j]);
                }
            }
        }
        System.out.println(min_dist == 99999999 ? -1 : min_dist);
    }

    private static void bfs(int x, int y, char c) {
        Queue<Pair> q = new LinkedList<>();
        visited[x][y] = true;
        q.add(new Pair(x, y, 0));
        while (!q.isEmpty()) {
            Pair p = q.poll();
            if (grid[p.x][p.y] != '.' && grid[p.x][p.y] != c) {
                min_dist = Math.min(min_dist, p.dist);
            }
            for (int i = 0; i < 4; i++) {
                int nx = p.x + dx[i];
                int ny = p.y + dy[i];
                if (nx >= 1 && nx <= n && ny >= 1 && ny <= m && !visited[nx][ny] && grid[nx][ny] != '#') {
                    visited[nx][ny] = true;
                    q.add(new Pair(nx, ny, p.dist + 1));
                }
            }
        }
    }

    private static class Pair {
        int x, y, dist;

        public Pair(int x, int y, int dist) {
            this.x = x;
            this.y = y;
            this.dist = dist;
        }
    }
}
