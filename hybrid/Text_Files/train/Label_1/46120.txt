import java.util.Scanner;

public class Main {
    static int n;
    static char[][] grid;
    static boolean[][][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        grid = new char[n][n];
        dp = new boolean[n][n][256];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < n; j++) {
                grid[i][j] = s.charAt(j);
            }
        }
        int mask = 0;
        boolean can = dfs(0, 0, mask);
        System.out.println(can ? "YES" : "NO");
    }

    private static boolean dfs(int x, int y, int mask) {
        if (x == n) {
            x = 0;
            y++;
        }
        if (y == n) {
            return true;
        }
        if (dp[x][y][mask]) {
            return false;
        }
        dp[x][y][mask] = true;
        if (grid[x][y] == '#') {
            return dfs(x + 1, y, mask);
        }
        for (int i = 0; i < 32; i++) {
            if ((mask & (1 << i)) != 0) {
                continue;
            }
            if (x + 2 <= n && y + 1 <= n && grid[x][y] == grid[x + 1][y] && grid[x][y] == grid[x + 2][y] && grid[x][y] == grid[x][y + 1] && grid[x][y] != '.') {
                mask |= (1 << i);
                boolean can = dfs(x + 2, y + 1, mask);
                mask ^= (1 << i);
                if (can) {
                    return true;
                }
            }
            if (x + 1 <= n && y + 2 <= n && grid[x][y] == grid[x + 1][y] && grid[x][y] == grid[x][y + 1] && grid[x][y] == grid[x][y + 2] && grid[x][y] != '.') {
                mask |= (1 << i);
                boolean can = dfs(x + 1, y + 2, mask);
                mask ^= (1 << i);
                if (can) {
                    return true;
                }
            }
            if (x + 2 <= n && y + 2 <= n && grid[x][y] == grid[x + 1][y] && grid[x][y] == grid[x + 2][y] && grid[x][y] == grid[x][y + 1] && grid[x][y] == grid[x + 1][y + 1] && grid[x][y] != '.') {
                mask |= (1 << i);
                boolean can = dfs(x + 2, y + 2, mask);
                mask ^= (1 << i);
                if (can) {
                    return true;
                }
            }
            if (x + 2 <= n && y + 2 <= n && grid[x][y] == grid[x + 2][y] && grid[x][y] == grid[x][y + 1] && grid[x][y] == grid[x + 1][y + 1] && grid[x][y] == grid[x + 1][y + 2] && grid[x][y] != '.') {
                mask |= (1 << i);
                boolean can = dfs(x + 2, y + 2, mask);
                mask ^= (1 << i);
                if (can) {
                    return true;
                }
            }
        }
        return false;
    }
}
