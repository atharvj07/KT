#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 62;
int t, n, a[N], dp[N][N][1001];
ll s;
int rec(int x, int y, ll sum) {
    if (y == n)
        return sum == s - 1;
    int &ret = dp[x][y][sum];
    if (ret != -1)
        return ret;
    ret = rec(x + 1, y + 1, sum + a[y]);
    ret += rec(x + 1, y + 1, sum);
    return ret;
}
int main() {
    cin >> t;
    while (t--) {
        cin >> n;
        s = 0;
        for (int i = 0; i < n; i++) { cin >> a[i]; s += a[i]; }
        memset(dp, -1, sizeof dp);
        cout << rec(0, 0, 0) << '\n';
    }
}
