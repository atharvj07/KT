import sys

t = int(input())

for _ in range(t):
    n, m, x, y = map(int, input().split())
    tiles = [input() for _ in range(n)]
    dp = [[x*m for _ in range(m+1)] for _ in range(n+1)]

    for i in range(n-1,-1,-1):
        for j in range(m-1,-1,-1):
            if tiles[i][j] == '.':
                dp[i][j] = min(dp[i][j], dp[i][j+1]+x)
                if j+1 < m: dp[i][j] = min(dp[i][j], dp[i][j+2]+y)
                dp[i][j] = min(dp[i][j], dp[i+1][j]+x)
                if i+1 < n: dp[i][j] = min(dp[i][j], dp[i+1][j+1]+y)

    print(dp[0][0])
