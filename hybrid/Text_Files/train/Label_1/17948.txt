#include <iostream>
using namespace std;
int main() {
  cin.tie(nullptr);
  ios::sync_with_stdio(false);
  long long n, s;
  cin >> n >> s;
  long long ans = n;
  long long tmp = n;
  while (tmp > 0) {
    ans -= tmp % 10;
    tmp /= 10;
  }
  if (ans < s) cout << 0 << endl;
  else
    cout << ans - s + 1 << endl;
  return 0;
}
