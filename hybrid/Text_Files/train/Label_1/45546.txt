import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] count = new int[m + 1];
        for (int i : a) {
            count[i]++;
        }
        Arrays.sort(count);
        int max = 0;
        for (int j = m; j >= 1; j--) {
            if (count[j] == 0) {
                break;
            }
            if (count[j] >= 3) {
                max += count[j] / 3;
                count[j] %= 3;
            }
            if (j > 1 && count[j] > 0 && count[j - 1] > 0) {
                max++;
                count[j]--;
                count[j - 1]--;
            }
        }
        System.out.println(max);
    }
}
