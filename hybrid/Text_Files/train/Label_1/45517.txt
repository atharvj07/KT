import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int xv = sc.nextInt();
        int yv = sc.nextInt();
        int xp = sc.nextInt();
        int yp = sc.nextInt();
        int[] w = {sc.nextInt(), sc.nextInt(), sc.nextInt(), sc.nextInt()};
        int[] m = {sc.nextInt(), sc.nextInt(), sc.nextInt(), sc.nextInt()};

        if (xv == xp && yv == yp) {
            System.out.println("NO");
            return;
        }

        boolean blocked = false;
        if (Collinear(xv, yv, xp, yp, w[0], w[1]) && OnSegment(xv, yv, w[0], w[1])) {
            blocked = true;
        }
        if (Collinear(xv, yv, xp, yp, w[2], w[3]) && OnSegment(xv, yv, w[2], w[3])) {
            blocked = true;
        }
        if (Collinear(xv, yv, xp, yp, m[0], m[1]) && (OnSegment(xv, yv, m[0], m[1]) ||
                OnSegment(xv, yv, m[2], m[3]) ||
                (xv == m[0] && yv != m[1] && yv != m[3]) ||
                (xv == m[2] && yv != m[1] && yv != m[3]))) {
            blocked = false;
        }
        if (Collinear(xv, yv, xp, yp, m[2], m[3]) && (OnSegment(xv, yv, m[0], m[1]) ||
                OnSegment(xv, yv, m[2], m[3]) ||
                (xv == m[2] && yv != m[1] && yv != m[3]) ||
                (xv == m[0] && yv != m[1] && yv != m[3]))) {
            blocked = false;
        }

        System.out.println(blocked ? "NO" : "YES");
    }
    static boolean Collinear(int x1, int y1, int x2, int y2, int x3, int y3) {
        return (y3 - y2) * (x2 - x1) == (y2 - y1) * (x3 - x2);
    }

    static boolean OnSegment(int x1, int y1, int x2, int y2) {
        return (x1 <= x2 && x2 <= x1) && (y1 <= y2 && y2 <= y1);
    }
}
