#include<bits/stdc++.h>
using namespace std;
#define endl "\n"
#define int long long
#define fastIO ios_base::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL)

int32_t main(){
	fastIO;
	int n;
	cin>>n;
	vector<int> a(n);
	int sum=0;
	for(int i=0;i<n;i++){
		cin>>a[i];
		sum+=a[i];
	}
	int i=0,s=0,c=0;
	while(1){
		c++;
		s+=a[i];
		if(c==sum-s){
			cout<<i+1<<endl;
			break;
		}
		i++;
		i%=n;
	}
}
