import java.util.Scanner;

public class Frames {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        char[][] paper = new char[n][m];
        for (int i = 0; i < n; i++) {
            String row = scanner.next();
            for (int j = 0; j < m; j++) {
                paper[i][j] = row.charAt(j);
            }
        }

        if (hasTwoFrames(paper)) {
            System.out.println("YES");
            int[] firstFrame = findFrame(paper);
            System.out.println(firstFrame[0] + " " + firstFrame[1] + " " + firstFrame[2] + " " + firstFrame[3]);
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (paper[i][j] == '#') {
                        paper[i][j] = '.';
                    }
                }
            }
            int[] secondFrame = findFrame(paper);
            System.out.println(secondFrame[0] + " " + secondFrame[1] + " " + secondFrame[2] + " " + secondFrame[3]);
        } else {
            System.out.println("NO");
        }
    }

    static boolean hasTwoFrames(char[][] paper) {
        int n = paper.length;
        int m = paper[0].length;
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (paper[i][j] == '#') {
                    if (i > 0 && paper[i - 1][j] == '#' && j > 0 && paper[i][j - 1] == '#') {
                        continue;
                    } else {
                        count++;
                    }
                }
            }
        }
        return count == 2;
    }

    static int[] findFrame(char[][] paper) {
        int n = paper.length;
        int m = paper[0].length;
        int minX = n, maxX = 0, minY = m, maxY = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (paper[i][j] == '#') {
                    minX = Math.min(minX, i);
                    maxX = Math.max(maxX, i);
                    minY = Math.min(minY, j);
                    maxY = Math.max(maxY, j);
                }
            }
        }
        return new int[]{minX, minY, maxX, maxY};
    }
}
