#include<bits/stdc++.h>
int a[10],x;
main(){
    int n,i,ans=0;
    scanf("%d",&n);
    for(i=0;n&&i<10;i++){
        a[i]=n%10;
        n/=10;
    }
    for(i=0;i<10;i++){
        x=i;
        for(int j=0;j<10;j++)
            x+=a[j];
        if(x==n){
            printf("%d\n",i);
            ans++;
        }
    }
    if(!ans)printf("0\n");
}
