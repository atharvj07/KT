#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> v[m + 1];
  for (int i = 0; i < n; i++) {
    int cnt;
    cin >> cnt;
    while (cnt--) {
      int x;
      cin >> x;
      v[x].push_back(i);
    }
  }
  int vis[n + 1];
  fill(vis + 1, vis + n + 1, 0);
  for (int i = 1; i <= m; i++) {
    bool flag = false;
    for (auto u : v[i]) {
      if (!vis[u]) {
        vis[u] = 1;
        flag = true;
        break;
      }
    }
    if (!flag) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
}
