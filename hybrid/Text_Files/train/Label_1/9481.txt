#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  string s;
  cin >> s;
  int n = s.length();
  int ansOdd = 0, ansEven = 0;
  for (int i = 0; i < n; i++) {
    int l = i, r = i;
    while (l >= 0 && r < n && s[l] == s[r]) {
      l--;
      r++;
    }
    l++;
    r--;
    if (l <= r) {
      if ((r - l + 1) % 2 == 1)
        ansOdd++;
      else
        ansEven++;
    }
  }
  for (int i = 0; i < n - 1; i++) {
    int l = i, r = i + 1;
    while (l >= 0 && r < n && s[l] == s[r]) {
      l--;
      r++;
    }
    l++;
    r--;
    if (l <= r) {
      if ((r - l + 1) % 2 == 1)
        ansOdd++;
      else
        ansEven++;
    }
  }
  cout << ansEven << " " << ansOdd << endl;
  return 0;
}
