#include<bits/stdc++.h>
using namespace std;

int main() {
    int n, s;
    cin >> n >> s;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    int h = 0, m = 0;
    for (int i = 0; i < n; i++) {
        if (v[i].first * 60 + v[i].second + s < (m + 60 * h)) {
            h = v[i].first;
            m = v[i].second + s;
            if (m >= 60) {
                m -= 60;
                h++;
            }
        } else {
            break;
        }
    }
    cout << h << " " << m << endl;
    return 0;
}
