#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(2 * n);
        for (int i = 0; i < 2 * n; i++) cin >> a[i];
        sort(a.begin(), a.end());
        vector<int> b(2 * n);
        int j = 0, k = 2 * n - 1;
        for (int i = 0; i < 2 * n; i += 2) {
            b[i] = a[j];
            j++;
            b[i + 1] = a[k];
            k--;
        }
        for (int i : b) cout << i << " ";
        cout << endl;
    }
    return 0;
}
