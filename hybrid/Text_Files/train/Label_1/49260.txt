import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        int d = sc.nextInt();
        int[] k = new int[d];
        for (int i = 0; i < d; i++) {
            k[i] = sc.nextInt();
        }
        int[] cnt = new int[m + 1];
        for (int i = 0; i < n; i++) {
            cnt[c[i]]++;
        }
        for (int i = 0; i < d; i++) {
            cnt[c[k[i] - 1]]--;
            int mex = 0;
            while (cnt[mex] > 0) {
                mex++;
            }
            System.out.println(mex);
        }
    }
}
