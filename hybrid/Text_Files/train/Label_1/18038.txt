#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define pii pair<int,int>
#define vb vector<bool>
const int N=2e5+2,M=1e9+7;

int n,a[N],b[N],indeg[N];
vector<int>adj[N];

void kahn(){
	queue<int>Q;
	for(int i=1;i<=n;i++)if(!indeg[i])Q.push(i);
	while(Q.size()){
		int u=Q.front();Q.pop();
		cout<<u<<' ';
		for(int v:adj[u])if(--indeg[v]==0)Q.push(v);			
	}
}

int main(){
	cin>>n;
	for(int i=1;i<=n;i++){
		int x,y;cin>>x>>y;
		a[i]=x;b[i]=y;
		if(x)adj[x].push_back(i),indeg[i]++;
		if(y)adj[i].push_back(y),indeg[y]++;
	}
	kahn();
	return 0;
}
