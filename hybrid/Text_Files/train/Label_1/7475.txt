#include <bits/stdc++.h>
using namespace std;
int n, a[5001], b[5001];
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i], b[a[i]] = 1; }
  int ans = 0;
  for (int i = 0; i <= 5000; i++)
    if (b[i]) {
      int j;
      for (j = 0; b[i + j]; j++) ans ^= i + j;
      i = j;
    }
  cout << ans << endl;
  return 0;
}
