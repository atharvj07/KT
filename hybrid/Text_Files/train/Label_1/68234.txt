n = int(input())
s = input()
q = int(input())
for _ in range(q):
    m, c = input().split()
    m = int(m)
    a = [0] * 26
    i = 0
    j = 0
    ans = 0
    cnt = 0
    while i < n:
        if cnt < m:
            if a[ord(s[i]) - ord('a')] == 0:
                cnt += 1
            a[ord(s[i]) - ord('a')] += 1
            if s[i] == c:
                ans = max(ans, i - j + 1)
            i += 1
        else:
            a[ord(s[j]) - ord('a')] -= 1
            if a[ord(s[j]) - ord('a')] == 0:
                cnt -= 1
            j += 1
    print(ans)
