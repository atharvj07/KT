#include <cstdio>
#include <iostream>

using namespace std;

int main() {
  int q;
  cin >> q;
  for (int i = 0; i < q; ++i) {
    long long k, n, a, b;
    cin >> k >> n >> a >> b;
    if (k < n * b)
      cout << -1 << endl;
    else
      cout << (k - n * b) / (a - b) + 1 << endl;
  }
  return 0;
}
