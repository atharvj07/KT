#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  if (n <= 2)
    cout << n << endl << n << endl;
  else if (n == 3)
    cout << 3 << endl << 1 << ' ' << 2 << ' ' << 3 << endl;
  else {
    int k = n / 2;
    int m = n - k * 2;
    cout << k + m << endl;
    for (int i = 0; i < k; i++) cout << 2 << ' ';
    for (int i = 0; i < m; i++) cout << 1 << ' ';
    cout << endl;
  }
  return 0;
}
