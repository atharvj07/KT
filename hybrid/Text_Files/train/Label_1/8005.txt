#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100 + 5;
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    if (k == 1) {
      if (a[n - 1] > 0)
        cout << "-1\n";
      else
        cout << "1\n";
    } else {
      int ans = (a[n - 1] + k - 1) / k;
      cout << ans << '\n';
    }
  }
  return 0;
}
