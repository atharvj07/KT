MOD = 10**9 + 7

def count_substrings(s, t):
    n, m = len(s), len(t)
    dp = [[0] * (n + 1) for _ in range(2)]
    count = [0] * (n + 1)

    for i in range(1, n + 1):
        dp[0][i] = dp[0][i-1]
        if s[i-1] == t[0]:
            dp[0][i] += 1
        for j in range(1, min(i, m)):
            if s[i-1] == t[j]:
                dp[j % 2][i] = dp[(j-1) % 2][i-1]
            else:
                dp[j % 2][i] = 0
            if j == m-1:
                count[i] = (count[i-1] + dp[j % 2][i]) % MOD
        if m == 1:
            count[i] = (count[i-1] + dp[0][i]) % MOD

    total = 0
    for i in range(m, n + 1):
        total += count[i] - count[i-1]
        total %= MOD

    result = 0
    for i in range(m, n + 1):
        result += total * (count[i] - count[i-1])
        result %= MOD
        total -= (count[i] - count[i-1])
        total %= MOD

    return result

s = input().strip()
t = input().strip()
print(count_substrings(s, t))
