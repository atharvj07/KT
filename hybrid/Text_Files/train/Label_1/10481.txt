#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int r, c;
    cin >> r >> c;
    vector<string> grid(r);
    for (int i = 0; i < r; ++i) { cin >> grid[i]; }
    if (grid[0][0] == 'P') {
      cout << "MORTAL\n";
    } else {
      int count = 0;
      for (int i = 0; i < r; ++i) {
        for (int j = 0; j < c; ++j) {
          if (grid[i][j] == 'P') {
            if (i == 0 || grid[i - 1][j] == 'A') {
              ++count;
              if (i < r - 1) {
                for (int k = i + 1; k < r && grid[k][j] == 'P'; ++k) {
                  grid[k][j] = 'A';
                }
              } else if (j < c - 1) {
                for (int k = j + 1; k < c && grid[i][k] == 'P'; ++k) {
                  grid[i][k] = 'A';
                }
              }
            }
          }
        }
      }
      cout << count << "\n";
    }
  }
  return 0;
}
