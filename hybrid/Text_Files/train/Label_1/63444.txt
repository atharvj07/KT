for _ in range(int(input())):
    n = int(input())
    a = sorted([int(x) for x in input().split()])
    pos, neg = [], []
    for x in a:
        if x > 0:
            pos.append(x)
        elif x < 0:
            neg.append(-x)
    ans = 0
    while pos and neg:
        if pos[0] > neg[0]:
            ans += pos[0] - neg[0]
            pos[0] -= neg[0]
            neg.pop(0)
        elif pos[0] < neg[0]:
            ans += pos[0] - neg[0]
            neg[0] -= pos[0]
            pos.pop(0)
        else:
            pos.pop(0)
            neg.pop(0)
    print(ans)
