from math import factorial

MOD = 10**9 + 7

def mod_factorial(n):
    return factorial(n) % MOD

n, m = map(int, input().split())
type_counts = [0] * m

for _ in range(n):
    gym = list(map(int, input().split()))[1:]
    unique_types = set(gym)
    for ut in unique_types:
        type_counts[ut-1] += 1

distinct_counts = {}
for count in type_counts:
    if count > 0:
        if count in distinct_counts:
            distinct_counts[count] += 1
        else:
            distinct_counts[count] = 1

result = 1
for key in distinct_counts:
    result = (result * mod_factorial(distinct_counts[key])) % MOD

print(result)
