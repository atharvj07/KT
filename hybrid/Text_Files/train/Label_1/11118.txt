#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, a;
  cin >> n >> a;
  vector<int> v(n + 1), ans(n + 1);
  for (int i = 1; i <= n; i++) {
    int tmp;
    cin >> tmp;
    v[i] = tmp;
    ans[i + tmp]++;
    ans[i - tmp]--;
  }
  int now = 0;
  for (int i = 1; i <= n; i++) {
    now += ans[i];
    if (i != a && now > 0) ans[i]--;
  }
  cout << ans[a] << '\n';
}
