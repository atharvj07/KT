#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> l, r;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'L') l.push_back(i + 1);
        else r.push_back(i + 1);
    }
    for (int i = 0; i < l.size(); i++) cout << l[i] << ' ' << r[i] << endl;
    return 0;
}
