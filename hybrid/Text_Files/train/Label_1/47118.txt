import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static int n, m;
    static ArrayList<Integer>[] adj;
    static int[] dist;
    static boolean[] vis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        dist = new int[n + 1];
        Arrays.fill(dist, -1);
        vis = new boolean[n + 1];
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj[u].add(v);
            adj[v].add(u);
        }
        bfs();
        if (dist[n] == -1) {
            System.out.println(-1);
        } else {
            System.out.println(dist[n]);
            LinkedList<Integer> path = new LinkedList<>();
            int curr = n;
            while (curr != 1) {
                path.addFirst(curr);
                for (int next : adj[curr]) {
                    if (dist[next] == dist[curr] - 1) {
                        curr = next;
                        break;
                    }
                }
            }
            path.addFirst(1);
            for (int node : path) {
                System.out.print(node + " ");
            }
            System.out.println();
        }
    }

    static void bfs() {
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        dist[1] = 0;
        while (!q.isEmpty()) {
            int curr = q.poll();
            vis[curr] = true;
            for (int next : adj[curr]) {
                if (!vis[next]) {
                    q.add(next);
                    dist[next] = dist[curr] + 1;
                }
            }
        }
    }
}
