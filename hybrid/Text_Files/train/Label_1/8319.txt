#include <bits/stdc++.h>
using namespace std ;
const int MAXN = 1e5 + 10 ;
long long r, d ;
int n ;
long long x[MAXN], y[MAXN], R[MAXN] ;
int main() {
  cin >> r >> d ;
  cin >> n ;
  for(int i = 1 ; i <= n ; i++ ) {
    cin >> x[i] >> y[i] >> R[i] ;
  }
  int ans = 0 ;
  for(int i = 1 ; i <= n ; i++ ) {
    if(R[i] * R[i] <= (r - d) * (r - d) ) {
      ans ++ ;
      continue ;
    }
    if(R[i] * R[i] <= r * r && hypot(x[i], y[i]) <= r - d ) {
      ans ++ ;
    }
  }
  cout << ans << endl ;
  return 0 ;
}
