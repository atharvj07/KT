#include <iostream>
#include <unordered_set>
using namespace std;

int main() {
    int n;
    cin >> n;
    unordered_set<int> watched;
    for (int i = 1; i < n; i++) {
        int watched_ep;
        cin >> watched_ep;
        watched.insert(watched_ep);
    }
    for (int i = 1; i <= n; i++) {
        if (watched.find(i) == watched.end()) {
            cout << i << endl;
            return 0;
        }
    }
    return 0;
}
