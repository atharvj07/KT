#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ll n, s, l;
    cin >> n >> s >> l;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    ll ans = 0, curmin = -1e9, curmax = -1e9 + s, i = 0;
    while (i != n) {
        if (curmin <= a[i] && a[i] <= curmax)
            i++;
        else if (a[i] > curmax) {
            ans++;
            curmin = a[i];
            curmax = a[i] + s;
        } else {
            curmin = -1e9;
            curmax = -1e9 + s;
            ans = -1;
        }
    }
    cout << ans;
    return 0;
}
