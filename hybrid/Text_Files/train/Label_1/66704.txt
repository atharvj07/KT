n, m, k = map(int, input().split())

if k < n:
    print(k + 1, 1)
else:
    k -= n
    row = n - k // (m - 1) * 2
    if k % (m - 1) == 0:
        col = 1 if (k // (m - 1)) % 2 == 1 else m
    else:
        row -= 1
        if (k // (m - 1)) % 2 == 1:
            col = m - k % (m - 1)
        else:
            col = 2 + k % (m - 1)
    print(row, col)
