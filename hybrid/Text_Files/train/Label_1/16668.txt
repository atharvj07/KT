#include<bits/stdc++.h>
using namespace std;
const int mod=1e9+7;
int n,p,a[1000000],s[10000000];
int main()
{
    cin>>n>>p;
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
        a[i]%=p;
        s[i+1]=s[i]+a[i];
    }
    int ans=0;
    for (int l=0;l<n;l++)
    {
        for (int r=l;r<n;r++)
            ans=max(ans,((s[l]+s[n]-s[r])%p)%p);
    }
    cout<<ans;
    return 0;
}
