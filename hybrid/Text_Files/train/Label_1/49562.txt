import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] t = new int[3][3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                t[i][j] = sc.nextInt();
            }
        }
        int n = sc.nextInt();
        int[][] dp = new int[n + 1][3];
        for (int i = 0; i < 3; i++) {
            dp[1][i] = t[0][i];
        }
        for (int i = 2; i <= n; i++) {
            for (int j = 0; j < 3; j++) {
                dp[i][j] = Integer.MAX_VALUE;
                for (int k = 0; k < 3; k++) {
                    if (k != j) {
                        dp[i][j] = Math.min(dp[i][j], dp[i - 1][k] + t[k][j]);
                    }
                }
            }
        }
        System.out.println(dp[n][2]);
    }
}
