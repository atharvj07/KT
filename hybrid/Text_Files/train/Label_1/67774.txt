import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    s = input().strip()

    q = [None] * n
    ql, qr = [], []
    for i in range(n):
        if s[i] == '0':
            ql.append(p[i])
        else:
            qr.append(p[i])
    ql.sort()
    qr.sort(reverse=True)
    i, j = 0, 0
    for k in range(n):
        if i < len(ql):
            if j < len(qr) and ql[i] > qr[j]:
                q[k] = qr[j]
                j += 1
            else:
                q[k] = ql[i]
                i += 1
        else:
            q[k] = qr[j]
            j += 1
    print(*q)
