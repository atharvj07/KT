#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  for (int i = 0; i < t; i++) {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int j = 0; j < n; j++) { std::cin >> a[j]; }
    int l = -1, r = -1;
    for (int j = 0; j < n; j++) {
      if (a[j]) {
        if (l == -1) l = j;
        r = j;
      }
    }
    int ans = 0;
    if (l > 0) ans += l;
    if (r < n - 1) ans += n - 1 - r;
    std::cout << ans << std::endl;
  }
  return 0;
}
