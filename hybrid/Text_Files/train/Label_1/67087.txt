n = int(input().strip())
seating = input().strip()

def is_maximal(seating):
    for i in range(len(seating)):
        if seating[i] == '0':
            if (i == 0 or seating[i-1] == '0') and (i == len(seating)-1 or seating[i+1] == '0'):
                return "No"
        else:
            if i > 0 and seating[i-1] == '1':
                return "No"
            if i < len(seating) - 1 and seating[i+1] == '1':
                return "No"
    return "Yes"

print(is_maximal(seating))
