#include<bits/stdc++.h>
#include<bits/extc++.h>
using namespace std;
using namespace __gnu_pbds;
typedef tree<int, null_type, less<int>, rb_tree_tag, tree_order_statistics_node_update> ord_set;
vector<int> a;
int main() {
    int n; cin >> n;
    a.resize(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    vector<int> b = a;
    rotate(b.begin(), b.begin() + 1, b.end());
    for (int i = 2; i < n; i++) {
        a.push_back(b[i]);
    }
    int i = min_element(a.begin(), a.end()) - a.begin();
    int dev = i - 1;
    while (true) {
        if (a[i + 1] == a[i] + 1) {
            dev += a[i] - 1;
            i = i + 1;
        } else {
            dev += a[i] + n - i;
            break;
        }
    }
    cout << dev << ' ' << i - 1 << endl;
}
