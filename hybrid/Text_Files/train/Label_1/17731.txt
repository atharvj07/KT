#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<vector<unsigned long long>> dp(n + 1, vector<unsigned long long>(m + 1, 0ULL));
    vector<vector<unsigned long long>> grid(n + 1, vector<unsigned long long>(m + 1, 0ULL));
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            cin >> grid[i][j];
        }
    }
    dp[1][1] = grid[1][1];
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            dp[i][j] = dp[i - 1][j] ^ dp[i][j - 1] ^ grid[i][j];
        }
    }
    cout << (dp[n][m] == k ? 1 : 0) << '\n';
    return 0;
}
