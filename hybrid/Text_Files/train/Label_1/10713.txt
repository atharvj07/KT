#include <iostream>
#include <vector>
#include <queue>
using namespace std;
typedef long long ll;
const int MOD = 1e9 + 7;
ll dp[101][101][1001];
ll a[101][101];
int main() {
    int T;
    cin >> T;
    while (T--) {
        int n, m;
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                cin >> a[i][j];
            }
        }
        for (int v = 0; v <= 1000; v++) {
            for (int i = n; i >= 1; i--) {
                for (int j = m; j >= 1; j--) {
                    if (a[i][j] == 0) continue;
                    dp[i][j][v] = 0;
                    if (i == 1 && j == 1) {
                        dp[i][j][v] = 1;
                        continue;
                    }
                    if (a[i][j] - v <= 0) {
                        if (dp[i][j - 1][0] || dp[i - 1][j][0])
                            dp[i][j][v] = 1;
                        continue;
                    }
                    if (dp[i][j - 1][v - a[i][j] + 2] ||
                            dp[i - 1][j][v - a[i][j] + 2])
                        dp[i][j][v] = 1;
                    if (dp[i][j - 1][v - a[i][j]] ||
                            dp[i - 1][j][v - a[i][j]])
                        dp[i][j][v] = 1;
                    if (dp[i][j - 1][v] || dp[i - 1][j][v]) dp[i][j][v] = 1;
                }
            }
        }
        bool yes = false;
        for (int j = 1; j <= m; j++) {
            if (dp[1][j][0]) yes = true;
            if (dp[n][j][0]) yes = true;
        }
        for (int i = 1; i <= n; i++) {
            if (dp[i][1][0]) yes = true;
            if (dp[i][m][0]) yes = true;
        }
        if (yes)
            cout << "Ashish" << endl;
        else
            cout << "Jeel" << endl;
    }
}
