#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
ll n, L[2100], R[2100], x, y, f;
string s, t, a, b;
int32_t main() {
    cin >> a >> b;
    n = b.length();
    for (ll i = 0; i < n; i++) {
        if (a == b.substr(i, a.length()))
            L[f] = i, R[f++] = i + a.length() - 1;
        reverse(a.begin(), a.end());
        if (a == b.substr(i, a.length()))
            L[f] = i, R[f++] = i + a.length() - 1;
        reverse(a.begin(), a.end());
    }
    cout << (f ? f : -1) << endl;
    for (ll i = 0; i < f; i++) cout << L[i] + 1 << ' ' << R[i] + 1 << endl;
}
