#include<bits/stdc++.h>
using namespace std;
int main(){
    long long int x,n,i,j,sum=0,flag=0;cin>>n>>x;
    long long int a[n+5];
    for(i=0;i<n;i++){
        cin>>a[i];sum+=a[i];
    }
    if(sum==x)flag=1;
    else{
        for(i=0;i<n;i++){
            if(a[i]>x){flag=1;break;}
        }
    }
    if(flag==1)cout<<"NO"<<endl;
    else cout<<"YES"<<endl;
}
