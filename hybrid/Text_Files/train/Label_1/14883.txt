#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;
    vector<int>v(n,0);
    for(int i =0;i<m;i++){
        int x;
        cin>>x;
        x--;
        v[x]++;
    }
    int ans =0;
    while(1){
        int ok =1;
        for(int i =0;i<n;i++){
            if(v[i]==0){
                ok=0;
                break;
            }
        }
        if(ok){
            for(int i =0;i<n;i++){
                v[i]--;
            }
            ans++;
        }
        else{
            break;
        }
    }
    cout<<ans<<endl;
    return 0;
}
