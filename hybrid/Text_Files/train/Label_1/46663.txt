import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        String s = sc.next();
        int[] pos = new int[m];
        for (int i = 0; i < m; i++) {
            pos[s.charAt(i) - 'a'] = i;
        }
        int minSlowness = Integer.MAX_VALUE;
        for (int i = 0; i < m; i++) {
            int slowness = 0;
            for (int j = 1; j < n; j++) {
                slowness += Math.abs(pos[s.charAt(j) - 'a'] - pos[s.charAt(j - 1) - 'a']);
            }
            minSlowness = Math.min(minSlowness, slowness);
        }
        System.out.println(minSlowness);
    }
}
