import java.util.Scanner;

public class Scale {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int L = sc.nextInt();
        int N = sc.nextInt();

        int[] x = new int[N];
        int[] w = new int[N];

        for (int i = 0; i < N; i++) {
            x[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }

        int sumX = 0;
        int sumW = 0;

        for (int i = 0; i < N; i++) {
            sumX += x[i] * w[i];
            sumW += w[i];
        }

        int x0 = -sumX / sumW;

        System.out.println("1");
        System.out.println(x0 + " " + sumW);
    }
}
