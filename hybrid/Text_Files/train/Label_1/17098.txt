#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;
    string a,b;
    cin>>a>>b;
    unordered_map<char,int>mp;
    for(int i=0;i<n;i++) mp[a[i]]++;
    for(int i=0;i<m;i++){
        if(!mp[b[i]]) return cout<<-1,0;
        mp[b[i]]--;
    }
    cout<<m;
    return 0;
}
