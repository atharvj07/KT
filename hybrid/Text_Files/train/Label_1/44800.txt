import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        StringBuilder sb = new StringBuilder();
        char prev = s.charAt(0);
        sb.append(prev);
        for (int i = 1; i < n; i++) {
            char cur = s.charAt(i);
            if ((prev == 'e' || prev == 'o') && cur == prev && i < n - 1 && s.charAt(i + 1) == prev) {
                prev = cur;
                continue;
            }
            if (cur != prev) {
                prev = cur;
                sb.append(cur);
            }
        }
        System.out.println(sb);
    }
}
