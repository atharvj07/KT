n = int(input())
x_formed = True
diagonal_char = None
off_diagonal_char = None

for i in range(n):
    row = input()
    if diagonal_char is None:
        diagonal_char = row[0]
        off_diagonal_char = row[1]
    if diagonal_char == off_diagonal_char or row[i] != diagonal_char or row[n-i-1] != diagonal_char:
        x_formed = False
        break
    for j in range(n):
        if j != i and j != n-i-1 and row[j] != off_diagonal_char:
            x_formed = False
            break
    if not x_formed:
        break

if x_formed:
    print("YES")
else:
    print("NO")
