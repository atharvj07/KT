import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        String x = sc.next();
        String y = sc.next();
        long l = (n * x.length()) / gcd(x.length(), n);
        long t = (m * y.length()) / gcd(y.length(), m);
        long ans = (l / x.length()) * (t / y.length()) * (x.equals(y) ? 0 : 1) * x.length();
        System.out.println(ans);
    }

    public static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
