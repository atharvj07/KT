#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=2e5;
ll a[maxn+5],tree[maxn+5];
void upd(int idx, ll val){
    while(idx<=maxn){
        tree[idx]+=val;
        idx+=(idx&-idx);
    }
}
ll query(int idx){
    ll sum=0;
    while(idx>0){
        sum+=tree[idx];
        idx-=(idx&-idx);
    }
    return sum;
}
void build(int n){
  for(int i=1;i<=n;i++){
      upd(i,a[i]);
  }
}
int main(){
    int n,m; cin>>n;
    for(int i=1;i<=n;i++) cin>>a[i];
    build(n);
    cin>>m;
    while(m--){
        int l,r; cin>>l>>r;
        cout<<query(r)-query(l-1)<<'\n';
    }
}
