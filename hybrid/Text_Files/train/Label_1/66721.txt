def can_human_win(n, k, coefficients):
    unknowns = coefficients.count('?')
    if unknowns % 2 == 0:
        # If all coefficients are determined
        if unknowns == 0:
            # Calculate P(k) to check divisibility by Q(x) = x - k
            result = 0
            for i, coef in enumerate(coefficients):
                result += coef * (k ** i)
            return "Yes" if result == 0 else "No"
        else:
            # If even number of moves left, computer can counter any play
            return "No"
    else:
        # If odd number of moves left, human can always adjust his last move
        return "Yes"

# Input reading
n, k = map(int, input().split())
coefficients = []
for _ in range(n + 1):
    value = input().strip()
    coefficients.append(int(value) if value != '?' else '?')

# Output result
print(can_human_win(n, k, coefficients))
