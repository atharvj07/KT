#include <bits/stdc++.h>
using namespace std;

const int MAX_COLOR = 5000;

struct DSU {
    int n;
    vector<int> color, group_size;
    void init(int _n) { n = _n; color.resize(n), group_size.resize(n, 1); }
    int find(int a) { return color[a] == a ? a : color[a] = find(color[a]); }
    void merge(int a, int b) {
        a = find(a), b = find(b);
        if (a == b) return;
        if (group_size[a] < group_size[b]) swap(a, b);
        color[b] = a;
        group_size[a] += group_size[b];
    }
};

int main() {
    int n, ans = 0;
    cin >> n;
    DSU dsu;
    dsu.init(MAX_COLOR);
    for (int i = 0, x; i < n; i++) {
        cin >> x;
        x = dsu.find(x);
        if (i > 0 && dsu.find(x) != dsu.find(x - 1)) ans++;
    }
    cout << ans << endl;
    return 0;
}
