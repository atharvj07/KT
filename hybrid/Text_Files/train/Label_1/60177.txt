n,s,l=map(int,input().split())
a=list(map(int,input().split()))
mx=[0]*n
mn=[0]*n
mx[0]=a[0]
mn[0]=a[0]
for i in range(1,n):
    mx[i]=max(mx[i-1],a[i])
    mn[i]=min(mn[i-1],a[i])
mn[0]-=s
mn[-1]+=s
ans=float('inf')
for i in range(n-l+1):
    if mx[i+l-1]<=mn[i]:
        ans=min(ans,(i+l)//l+(n-i-1)//l+1)
    else:ans=-1
print(ans)
