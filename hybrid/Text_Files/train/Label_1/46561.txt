import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        int k = in.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }

        int l = 0;
        int r = 0;
        int maxLen = 0;
        int maxL = 0;
        int maxR = 0;

        while (r < n) {
            if (a[r] == 0) {
                k--;
            }
            while (k < 0) {
                if (a[l] == 0) {
                    k++;
                }
                l++;
            }

            if (r - l + 1 > maxLen) {
                maxLen = r - l + 1;
                maxL = l;
                maxR = r;
            }

            r++;
        }

        System.out.println(maxLen);
        for (int i = 0; i < n; i++) {
            if (i >= maxL && i <= maxR) {
                System.out.print(1 + " ");
            } else {
                System.out.print(a[i] + " ");
            }
        }
        System.out.println();
    }
}
