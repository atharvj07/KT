import java.util.*;
class Main {
    static int[] ri;
    static ArrayList<Integer>[] g;
    static int dfs(int u, int p, int dep) {
        int maxl = 1;
        for (int v : g[u]) {
            if (v != p) {
                maxl = Math.max(maxl, 1 + dfs(v, u, ri[u] > ri[p] ? dep + 1 : 0));
            }
        }
        return maxl;
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        ri = new int[n + 1];
        g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            ri[i] = sc.nextInt();
            g[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            g[a].add(b);
            g[b].add(a);
        }
        System.out.println(dfs(1, 1, 0));
    }
}
