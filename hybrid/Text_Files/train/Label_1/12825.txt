#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> ans;
  for (int i = 2; i * i <= n; i++) {
    while (n % i == 0) {
      ans.push_back(i);
      n /= i;
    }
  }
  if (n > 1) ans.push_back(n);
  std::cout << ans.size() << '\n';
  for (int i = 0; i < ans.size(); i++) std::cout << ans[i] << ' ';
  std::cout << '\n';
}
