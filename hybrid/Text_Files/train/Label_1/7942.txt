#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 5;
const int INF = 1e9;

int n;
int p[N], c[N];
int sub[N], del[N];
vector<int> adj[N];

void dfs1(int u, int par, int &cur) {
  sub[u] = 1;
  for (int v : adj[u]) {
    if (v == par) continue;
    dfs1(v, u, cur);
    sub[u] += sub[v];
  }
  if (u != 1 && c[u] && !c[p[u]] && sub[u] == 1) del[cur++] = u;
}

void dfs2(int u, int par, bool keep) {
  for (int v : adj[u]) {
    if (v == par || (del[v] && !keep)) continue;
    dfs2(v, u, keep);
  }
  if (del[u]) dfs2(p[u], u, 1);
}

int main() {
  cin >> n;
  for (int i = 2; i <= n; ++i) {
    cin >> p[i] >> c[i];
    adj[p[i]].push_back(i);
  }
  int cur = 0;
  dfs1(1, 1, cur);
  for (int i = 1; i < cur; ++i) {
    dfs2(del[i - 1], p[del[i - 1]], 0);
  }
  bool ok = 0;
  for (int i = 1; i < cur; ++i) {
    if (del[i]) {
      if (ok) cout << " ";
      cout << del[i];
      ok = 1;
    }
  }
  if (!ok) cout << -1;
  cout << endl;
  return 0;
}
