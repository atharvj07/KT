#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.length();
        vector<int> pref(n, 0);
        if (s[0] == 'b') pref[0] = 1;
        for (int i = 1; i < n; i++) {
            pref[i] = pref[i - 1];
            if (s[i] == 'b') pref[i]++;
        }
        int ans = INT_MAX;
        string res;
        for (int i = 0; i < n; i++) {
            int cnt1 = pref[i];
            int cnt2 = (i + 1 < n - i ? pref[n - i - 1] - cnt1 : 0);
            int tot = n - i - 1;
            if (tot - (cnt1 + cnt2) < ans) {
                res = s;
                for (int j = i + 1; j < n - i; j++) {
                    if (res[j] == 'a' && cnt1 < cnt2) {
                        res[j] = 'b';
                        cnt1++;
                    } else if (res[j] == 'b' && cnt1 > cnt2) {
                        res[j] = 'a';
                        cnt2++;
                    }
                }
                ans = tot - (cnt1 + cnt2);
            }
        }
        cout << res << endl;
    }
}
