import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int x = 0;
        int y = 0;
        int count = 0;
        for (int i = 1; i <= 109; i++) {
            for (int j = 1; j <= 109; j++) {
                for (int l = 0; l < n; l++) {
                    if ((i == 0 || j == 0) || (i <= a[l] && j <= a[l])) {
                        count++;
                    }
                }
                if (count == k) {
                    x = i;
                    y = j;
                    break;
                }
                count = 0;
            }
            if (count == k) {
                break;
            }
        }
        if (count == k) {
            System.out.println(x + " " + y);
        } else {
            System.out.println("-1");
        }
    }
}
