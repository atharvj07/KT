import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        StringBuilder brackets = new StringBuilder(scanner.nextLine());
        int count = 0;
        for (int i = 0; i < brackets.length(); i++) {
            if (brackets.charAt(i) == ')') {
                count--;
            } else {
                count++;
            }
            if (count < 0) {
                brackets.insert(i, '(');
                count += 2;
            }
        }
        while (count > 0) {
            brackets.insert(brackets.length(), ')');
            count--;
        }
        System.out.println(brackets);
    }
}
