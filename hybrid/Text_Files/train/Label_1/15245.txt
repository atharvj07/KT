#include <bits/stdc++.h>
using ll = long long;
ll n, a[200001], b[200001], ans[200001];
int main() {
  std::cin >> n;
  for (ll i = 1; i <= n; i++) { std::cin >> b[i]; }
  if (b[1] >= 0) {
    a[1] = b[1];
    for (ll i = 2; i <= n; i++) { a[i] = std::max(a[i - 1], b[i] + a[i - 1]); }
  } else {
    for (ll i = 1; i <= n; i++) { a[i] = b[i]; }
  }
  for (ll i = 1; i <= n; i++) { std::cout << a[i] << " "; }
  return 0;
}
