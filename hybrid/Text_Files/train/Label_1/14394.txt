#include<bits/stdc++.h>
using namespace std;

int main()
{
    int n,a,b,sum=0;
    cin>>n>>a>>b;
    for(int i=1;n>1;i++)
    {
        if(a<=n && b<=n)
        {
            if(a==b)
            {
                sum=i;
                break;
            }
            if(a%2==1) a=(a+1)/2;
            else a/=2;
            if(b%2==1) b=(b+1)/2;
            else b/=2;
        }
        n/=2;
    }
    if(sum==0) cout<<"Final!"<<endl;
    else cout<<sum<<endl;

    return 0;
}
