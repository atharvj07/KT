MOD = 10**9 + 7

def power(x, y, p):
    res = 1 
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(N, k):
    ans = 0
    for i in range(1, k+2):
        ans += power(i, k, MOD) * power(N-i+1, MOD-2, MOD)
        ans %= MOD
    ans *= power(2, N-1, MOD)
    ans %= MOD
    return ans

N, k = map(int, input().split())
print(solve(N, k))
