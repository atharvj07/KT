n,w = map(int,input().split())
a=list(map(int,input().split()))
a.sort(reverse=True)
ans = [-1]*n
p=0
for i in range(n):
    if p>=w: break
    if a[i]*2 > w: continue
    if w-p>=a[i]*2:
        ans[i]=a[i]
        p+=a[i]
    else:
        ans[i]=(w-p)//2
        p=w
if p!=w: ans=[-1]
print(*ans)
