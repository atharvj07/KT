#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = k;
  for (int i = 0; i < n; i++) {
    if (k % a[i] == 0) ans = min(ans, k / a[i]);
  }
  cout << ans << endl;
  return 0;
}
