import java.util.*;

class Main {
    static int n, k, A;
    static Senator[] ss;

    static class Senator implements Comparable<Senator> {
        int l, c;

        Senator(int l, int c) {
            this.l = l;
            this.c = c;
        }

        @Override
        public int compareTo(Senator s) {
            return s.l - this.l;
        }
    }

    static double solve(double d, int idx, Senator[] ss) {
        if (idx == n || k == 0) return d;
        return Math.max(solve(d, idx + 1, ss), solve(d * ss[idx].c / (double)(ss[idx].c + A), idx + 1, ss));
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        A = sc.nextInt();
        ss = new Senator[n];
        for (int i = 0; i < n; i++)
            ss[i] = new Senator(sc.nextInt(), sc.nextInt());
        Arrays.sort(ss);
        double p = solve(1, 0, ss);
        System.out.printf("%.9f%n", p);
        sc.close();
    }
}
