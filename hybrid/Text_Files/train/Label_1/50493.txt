import java.util.*;

class Main {

    static Scanner sc = new Scanner(System.in);
    static int[][] dir = new int[][] {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

    static class Block {
        char c;
        boolean shift;

        public Block(char c, boolean shift) {
            this.c = c;
            this.shift = shift;
        }
    }

    public static void main(String[] args) {
        int n = sc.nextInt();
        int m = sc.nextInt();
        int x = sc.nextInt();
        sc.nextLine();
        Block[][] keyboard = new Block[n][m];
        for (int i = 0; i < n; i++) {
            String line = sc.nextLine();
            for (int j = 0; j < m; j++) {
                char c = line.charAt(j);
                keyboard[i][j] = new Block(c, c == 'S');
            }
        }
        int q = sc.nextInt();
        sc.nextLine();
        String text = sc.nextLine();

        int[][] dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dist[i], -1);
        }
        Queue<int[]> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (keyboard[i][j].shift) {
                    queue.offer(new int[]{i, j});
                    dist[i][j] = 0;
                }
            }
        }
        while (!queue.isEmpty()) {
            int[] p = queue.poll();
            for (int[] d : dir) {
                int r = p[0] + d[0];
                int c = p[1] + d[1];
                if (r < 0 || r >= n || c < 0 || c >= m) continue;
                if (dist[r][c] != -1) continue;
                dist[r][c] = dist[p[0]][p[1]] + 1;
                queue.offer(new int[]{r, c});
            }
        }

        int ans = 0;
        for (char c : text.toCharArray()) {
            int minDist = Integer.MAX_VALUE;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (keyboard[i][j].c == c || (keyboard[i][j].shift && Character.toLowerCase(c) == keyboard[i][j].c)) {
                        minDist = Math.min(minDist, dist[i][j]);
                    }
                }
            }
            if (minDist == Integer.MAX_VALUE) {
                ans = -1;
                break;
            } else {
                ans += (minDist > x ? 1 : 0);
            }
        }
        System.out.println(ans);
    }
}
