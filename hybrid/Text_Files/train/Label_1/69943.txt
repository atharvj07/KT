n, h, a, b, k = map(int, input().split())
for _ in range(k):
    ta, fa, tb, fb = map(int, input().split())
    if ta == tb:
        print(abs(fa - fb))
    else:
        if fa < a:
            print(a - fa + abs(tb - ta) + abs(a - fb))
        elif fa > b:
            print(fa - b + abs(tb - ta) + abs(b - fb))
        else:
            print(abs(tb - ta) + abs(fa - fb))
