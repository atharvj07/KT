#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mp make_pair
const int mod = 1e9 + 7;
const int inf = 1e9;

int main() {
    int n; cin >> n;
    vector<int> x(n);
    for (int i = 0; i < n; i++) {
        cin >> x[i];
    }
    sort(x.begin(), x.end());
    for (int i = 1; i < n; i++) {
        if (x[i] - x[i - 1] < 0) {
            cout << "yes" << endl;
            return 0;
        }
    }
    cout << "no" << endl;
    return 0;
}
