#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll mod = 1e9 + 7;
ll pw[21], n;
int main() {
    // freopen("in.txt", "r", stdin);
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> pw[i];
    ll ans = 1;
    for (int i = 1; i <= n; i++) {
        ll tmp = 1;
        for (int j = 1; pow(pw[i], j) <= pw[n]; j++)
            tmp = (tmp * pw[i]) % mod;
        ans = (ans * tmp) % mod;
    }
    cout << ans << endl;
}
