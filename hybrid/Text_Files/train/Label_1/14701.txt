#include <iostream>
#include <vector>
#include <array>

using namespace std;

struct Op {
    int m;
    vector<int> Idx;
};

int main() {
    int n; cin >> n;
    vector<int> s(n);
    for (auto& e: s) {
        char ch; cin >> ch;
        e = (ch == '(');
    }

    const auto& pre = [&] (int x) { return (x + n - 1) % n; };
    const auto& ispref = [&] (int x) { return x == n / 2; };
    const auto& issu = [&] (int x) { return x == n - 1; };
    const auto& isfront = [&] (int x) { return x < n / 2; };

    bool finsh = false;
    int K = -1;
    vector<Op> Opt;

    int open = 0, close = 0;
    for (int i = 0; i < n; i++) {
        if (s[i]) open++;
        else close++;
    }

    int rem = open > close ? open - close : 0; 
    
    while (!finsh) {
        bool chg = false;
        K++;
        Op o;
        for (int i = 0; i < n; i++) {
            if (s[i]) {
                open--;
                if (isfront(pre(i)) && ispref(open) && issu(close)) {
                    o.m++;
                    o.Idx.push_back(pre(i));
                    s[pre(i)] = false;
                    chg = true;
                }
            } 
            else {
                close--;
                if (isfront(pre(i)) && ispref(close) && issu(open)) {
                    o.m++;
                    o.Idx.push_back(pre(i));
                    s[pre(i)] = true;
                    chg = true;
                }
            }
        }
        if (open == rem && close == 0) finsh = true;
        if (chg) Opt.push_back(o);
    }

    cout << K << endl;
    for (auto& e: Opt) {
        cout << e.m << endl;
        for (auto& j: e.Idx) {
            cout << j + 1 << ' ';
        }
        cout << endl;
    }
    return 0;
}
