import java.util.Scanner;

public class AntonAndArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] b = new int[n];
        int[] c = new int[n];
        int[] a = new int[n];

        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            int min = Math.min(b[i], c[i]);
            int max = Math.max(b[i], c[i]);
            if ((min & max) != b[i] || (min | max) != c[i]) {
                System.out.println(-1);
                return;
            }
            a[i] = min;
        }

        for (int i : a) {
            System.out.print(i + " ");
        }
    }
}
