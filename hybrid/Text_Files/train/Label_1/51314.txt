import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] loc = new int[n][2];
        for (int i = 0; i < n; i++) {
            loc[i] = new int[]{sc.nextInt(), sc.nextInt()};
        }
        Arrays.sort(loc, (a, b) -> {
            return a[0] == b[0] ? a[1] - b[1] : a[0] - b[0];
        });
        int[] dis = new int[n];
        Queue<Integer> q = new LinkedList<>();
        q.add(0);
        dis[0] = 1;
        while (!q.isEmpty()) {
            int p = q.poll();
            for (int i = 0; i < n; i++) {
                if (dis[i] == 0 && Math.abs(loc[p][0] - loc[i][0]) + Math.abs(loc[p][1] - loc[i][1]) == 1) {
                    q.add(i);
                    dis[i] = 1;
                }
            }
        }
        int cnt = 0;
        for (int x : dis) cnt += (x == 0 ? 1 : 0);
        System.out.print(cnt);
    }
}
