#include <iostream>
#include <vector>
int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::vector<char> S(n + 1);
    std::cin >> &S[0];
    int cntl = 0;
    for (int i = n - 1; i >= 0 && S[i] == ')'; i--) { cntl++; }
    std::cout << (cntl > n - cntl ? "Yes" : "No") << "\n";
  }
}
