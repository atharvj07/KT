#include <bits/stdc++.h>
using namespace std;

vector<int> g[100005];
int arr[100005];
int vis[100005];
int sub[100005];
int xor_val;

void dfs1(int u, int p){
  vis[u]=1;
  sub[u]=arr[u];
  for(int x : g[u]){
    if(x==p) continue;
    dfs1(x,u);
    sub[u]^=sub[x];
  }
}

bool dfs2(int u, int p, int xx){
  vis[u]=1;
  int cx=0;
  for(int x : g[u]){
    if(x==p) continue;
    if(dfs2(x,u,xx)) cx++;
  }

  xor_val^=sub[u];
  if(cx>1) return true;
  xor_val^=sub[u];
  if(xor_val==xx && cx>=1) return true;

  return false;
}

int main(){
  int t;
  cin>>t;
  while(t--){
    int n,k;
    cin>>n>>k;
    for(int i=1; i<=n; i++){
      cin>>arr[i];
    }
    for(int i=1; i<=n; i++) g[i].clear(), vis[i]=0;

    for(int i=1; i<n; i++){
      int u,v;
      cin>>u>>v;
      g[u].push_back(v);
      g[v].push_back(u);
    }

    dfs1(1,1);
    xor_val=sub[1];
    if(dfs2(1,1,xor_val)) cout<<"NO"<<endl;
    else cout<<"YES"<<endl;
  }
}
