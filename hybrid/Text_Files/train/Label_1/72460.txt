import sys

def find_end(node, next):
    while next[node] != 0:
        node = next[node]
    return node

n = int(input())
node = [0] * (n + 1)
next_ = [0] * (n + 1)
prev = [0] * (n + 1)
start = end = 0 
for i in range(1, n + 1):
    node[i] = i
    prev[i], next_[i] = map(int, input().split())
    if prev[i] == 0:
        if start == 0:
            start = i
        else:
            end = find_end(end, next_)
            next_[end] = i
            prev[i] = end
            end = i

for i in range(1, n + 1):
    print(' '.join(map(str, [prev[node[i]], next_[node[i]]])))
