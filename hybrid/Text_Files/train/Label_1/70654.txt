def find_min_x(a, h, w):
    def gcd(x, y):
        while y:
            x, y = y, x % y
        return x
    
    g1 = gcd(h - a, a)
    g2 = gcd(w - a, a)
    
    if (h - a) // g1 % 2 == 0 or (w - a) // g2 % 2 == 0:
        return -1
    
    x1 = (h - a) / ((h - a) // g1 - 1)
    x2 = (w - a) / ((w - a) // g2 - 1)
    
    return min(x1, x2)

a, h, w = map(int, input().split())
print(find_min_x(a, h, w))
