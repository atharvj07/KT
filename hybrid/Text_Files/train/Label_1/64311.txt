n = int(input())
a = list(map(int, input().split()))
mod = int(1e9) + 7
p2 = [1]
for i in range(1, n + 1):
    p2.append(p2[-1] * 2 % mod)
for i in range(n, 0, -1):
    a[i - 1] = a[i - 1] * p2[n - i] % mod
msk = [0] * (n + 1)
for i in range(1, n + 1):
    m = min(20, i)
    for j in range(1, m + 1):
        if a[i - 1] % j == 0:
            msk[i] |= (1 << (j - 1))
print((p2[n] - 1 - msk[0]) % mod)
