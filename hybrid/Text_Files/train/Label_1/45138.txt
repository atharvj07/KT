import java.util.Arrays;
import java.util.Scanner;
import java.util.HashSet;
import java.util.Set;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        Set<Integer> s = new HashSet<>();
        s.add(0);
        for (int i = 0; i < n; i++) {
            Set<Integer> t = new HashSet<>();
            for (int x : s) {
                if (x + a[i] <= k) {
                    t.add(x + a[i]);
                }
                if (x - a[i] >= 0) {
                    t.add(x - a[i]);
                }
            }
            s.addAll(t);
        }
        System.out.println(s.contains(k) ? "POSSIBLE" : "IMPOSSIBLE");
    }
}
