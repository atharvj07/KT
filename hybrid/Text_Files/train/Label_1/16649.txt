#include <bits/stdc++.h>
using namespace std;
const int N = 1e5+10;
vector<int> adj[N];
vector<pair<int, int>> edg;
vector<int> ans;
int n, m;
void dfs(int u, int p) {
    for (int v : adj[u]) {
        if (v == p) continue;
        if (adj[v].size() == 1) {
            dfs(v, u);
            ans.push_back(edg[v - 1].first);
            adj[u].erase(find(adj[u].begin(), adj[u].end(), v));
        }
    }
}
int32_t main() {
    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        int u, v, c;
        cin >> u >> v >> c;
        edg.push_back({u, v});
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (adj[i].size() > 1) cnt++;
    }
    cout << cnt << ' ' << ans.size() << endl;
    for (int i = ans.size() - 1; i >= 0; i--) {
        cout << ans[i] << ' ';
    }
    return 0;
}
