def is_palindrome(s):
    return s == s[::-1]

def get_min_cuts(s):
    if is_palindrome(s):  # Check if the string is already a palindrome
        return "Impossible"

    # Create a 2D array to store the minimum cuts needed to make a palindrome
    dp = [[0] * len(s) for _ in range(len(s))]
    # Fill the diagonal elements with 0, as no cuts are needed for a single character palindrome
    for i in range(len(s)):
        dp[i][i] = 0

    # Iterate over all substrings of length 2 to len(s)
    for l in range(2, len(s) + 1):
        for i in range(len(s) - l + 1):
            j = i + l - 1
            # Check if the substring is a palindrome
            if s[i] == s[j] and (l == 2 or dp[i + 1][j - 1] == 0):
                dp[i][j] = 0
            else:
                # Find the minimum cuts needed for the substring
                dp[i][j] = 1 + min(dp[i][k] + dp[k + 1][j] for k in range(i, j))

    # Check if a palindrome can be formed with a minimum of 1 cut
    if dp[0][len(s) - 1] == 1:
        return 1

    # Otherwise, return "Impossible"
    return "Impossible"


s = input()
print(get_min_cuts(s))
