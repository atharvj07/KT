n, S = map(int, input().split())
T = []
for _ in range(n):
    T.append(list(map(int, input().split())))
T.sort(key=lambda x: (x[2] - x[1]) / x[0], reverse=True)
t = 0
ans = 0
for a, b, c in T:
    cnt = (c - t + S - 1) // S
    ans += cnt * a * S
    t += cnt * S
    if t >= c:
        break
print(ans)
