def decrypt_message(encoded, k):
    letters = 0
    min_len = 0
    max_len = 0
    for i in range(len(encoded)):
        if encoded[i].isalpha():
            letters += 1
            min_len += 1
            max_len += 1
        elif encoded[i] == '?':
            min_len -= 1
        elif encoded[i] == '*':
            max_len = 200  # Assuming max length according to the problem statement

    if not (min_len <= k <= max_len):
        return "Impossible"

    result = []
    extra = k - letters
    for i in range(len(encoded)):
        if encoded[i].isalpha():
            result.append(encoded[i])
        elif encoded[i] == '?' and extra < 0:
            result.pop()
            extra += 1
        elif encoded[i] == '*' and extra != 0:
            result.append(encoded[i-1] * extra)
            extra = 0

    return ''.join(result)

encoded, k = input(), int(input())
print(decrypt_message(encoded, k))
