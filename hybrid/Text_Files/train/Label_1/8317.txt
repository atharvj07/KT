#include <iostream>
#include <cmath>
using namespace std;
using ll = long long;

const double EPS = 1e-9;

    ll n, r;
    cin >> n >> r;
    double a = 2 * acos(0.0);
    a = (2.0 * M_PI - a) / n;
    double ans = n / 2.0 * pow(r, 2) * sin(a);
    printf("%.10f", ans);
    return 0;
}
