A = int(input().strip())
B = int(input().strip())
N = A + B + 1

# Initial approach: Start with ascending order, then make adjustments
stops = list(range(1, N + 1))

# Adjust for the correct number of uphills and downhills
if A < B:
    stops[A], stops[A + 1] = stops[A + 1], stops[A]
else:
    stops[-(B+2)], stops[-(B+1)] = stops[-(B+1)], stops[-(B+2)]

print(" ".join(map(str, stops)))
