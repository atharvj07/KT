#include <iostream>

using namespace std;

typedef long long ll;

int main() {
    int n;
    cin >> n;
    ll arr[n];
    for (int i = 0; i < n; ++i) {
        cin >> arr[i];
    }
    ll s = 0;
    for (int i = 0; i < n; ++i) {
        s += arr[i];
    }
    ll x = 1;
    while (x * n < s) {
        x++;
    }
    int idx = 0;
    for (int i = 0; i < n; ++i) {
        if ((x * (i + 1) - arr[i]) >= 0) {
            idx = i + 1;
            break;
        }
    }
    cout << idx << endl;
    return 0;
}
