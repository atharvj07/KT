field = [input() for _ in range(4)]
ans = 'NO'
for i in range(4):
    for j in range(4):
        if field[i][j] == '.':
            for dx, dy in ((0, 1), (1, 0), (1, 1), (1, -1)):
                ok = True
                cnt = 1
                for k in range(1, 4):
                    nx, ny = i + k * dx, j + k * dy
                    if 0 <= nx < 4 and 0 <= ny < 4:
                        cnt += (field[nx][ny] == 'x')
                        ok &= field[nx][ny] == '.' or field[nx][ny] == 'x'
                if ok and cnt == 3:
                    ans = 'YES'
print(ans)
