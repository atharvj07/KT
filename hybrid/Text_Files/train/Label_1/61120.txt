import sys

n, k = map(int, input().split())
g = [[] for _ in range(n+1)]
for _ in range(n-1):
    x, y = map(int, input().split())
    g[x].append(y)
    g[y].append(x)

def dfs(u, p):
    cnt = 0
    for v in g[u]:
        if v == p: continue
        cnt += dfs(v, u)
    if cnt > 1: return -1
    return 1

flag = dfs(1, 1)
if flag == -1: print('No')
else: print('Yes')
