#include <bits/stdc++.h>
using namespace std;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int x = k / 4, y = k % 4;
    int l = (y == 0 ? x - 1 : x);
    int ans = 0;
    for (int i = 0; i < l; i++) {
      if (s[i] != "RGBR"[i % 4]) ans++;
    }
    for (int i = 0; i < y; i++) {
      if (s[l + i] != "RGBR"[i]) ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
