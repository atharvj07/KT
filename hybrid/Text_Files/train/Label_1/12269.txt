#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
int n,dp[1<<11][11],a[11][11];
int solve(int mask,int pos)
{
    if(pos==2*n)return 1;
    int &ret=dp[mask][pos];
    if(ret!=-1)return ret;
    ret=0;
    if((mask&(1<<pos))==0)
    {
        ret=(ret+solve(mask|(1<<pos),pos+1))%mod;
    }
    if(pos+1<2*n && (mask&(1<<pos))==0 && (mask&(1<<(pos+1)))==0)
    {
       ret=(ret+solve(mask|(1<<pos)|(1<<(pos+1)),pos+2))%mod;
    }
    return ret;
}
int main()
{
    int n;
    cin>>n;
    memset(dp,-1,sizeof dp);
    cout<<solve(0,0)<<endl;
}
