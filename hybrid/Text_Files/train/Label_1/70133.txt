import sys
input=sys.stdin.readline
s=input()
a=[0]*len(s)
x=y=z=0
for i in range(len(s)):
    if s[i]=='x':x+=1
    if s[i]=='y':y+=1
    if s[i]=='z':z+=1
    if s[i-1]!=s[i] or i==0:a[i]=x+y+z-1
lst=[]
for i in range(int(input())):
    l,r=map(int,input().split())
    l-=1;lst.append(a[r-1]-a[l-1])
for i in lst:print(['YES','NO'][(i==0)])
