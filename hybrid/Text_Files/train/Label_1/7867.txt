#include <iostream>
using namespace std;
int a[101][101];
int main() {
  int n, m, ans = -1, l1 = 101, l2 = 101, r1 = 0, r2 = 0, u1 = 101, u2 = 101,
      d1 = 0, d2 = 0;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (a[i][j] == 'B') {
        if (j < l1) l1 = j;
        if (i > r1) r1 = i;
        if (j > l2) l2 = j;
        if (i < r2) r2 = i;
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    if (i < u1) u1 = i;
    if (i > d1) d1 = i;
  }
  for (int i = 1; i <= m; i++) {
    if (i < u2) u2 = i;
    if (i > d2) d2 = i;
  }
  if (l1 < 101) {
    ans = (r1 - l1 + 1) * (r2 - l2 + 1);
    int ans2 = (d1 - u1 + 1) * (d2 - u2 + 1);
    if (ans > ans2) ans = ans2;
  }
  cout << ans << endl;
}
