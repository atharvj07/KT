#include <iostream>
#include <array>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    array<int, 200005> arr;
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    sort(arr.begin(), arr.begin() + n);
    long long ans = 0;
    int neg = 0;
    for (int i = 0; i < n; i++) {
        if (arr[i] > 0) break;
        neg++;
    }
    for (int i = 0; i < n; i++) {
        if (i % 2 == 0) ans += arr[i];
        else ans -= arr[i];
    }
    int p = (n + 1) / 2 - neg, g = n / 2;
    if (p == n) p--;
    if (ans < 0 && g > 0) {
        ans -= arr[p + g];
        g--;
    }
    cout << ans << endl;
    return 0;
}
