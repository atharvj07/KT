def can_measure(a, d, l):
    for mark in a:
        if mark + d in a or mark - d in a or mark + d <= l:
            return True
    return False

def find_mark(a, x, y, l):
    for mark in a:
        if mark + x <= l and (mark + x not in a):
            return mark + x
        if mark + y <= l and (mark + y not in a):
            return mark + y
        if mark - x >= 0 and (mark - x not in a):
            return mark - x
        if mark - y >= 0 and (mark - y not in a):
            return mark - y
    return -1

n, l, x, y = map(int, input().split())
a = list(map(int, input().split()))

x_found = can_measure(a, x, l)
y_found = can_measure(a, y, l)

if x_found and y_found:
    print(0)
elif x_found or y_found:
    print(1)
    if not x_found:
        print(x)
    else:
        print(y)
else:
    # Try to find a mark that can be used to measure both x and y
    mark = find_mark(a, x, y, l)
    if mark != -1:
        print(1)
        print(mark)
    else:
        print(2)
        print(x, y)
