import java.util.*;
import java.math.*;

class Main{
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int[] a = new int[n];
        for(int i = 0; i < n; i++)a[i] = s.nextInt();
        int sum = 0;
        for(int ai:a)sum+=ai;
        System.out.println(BigInteger.valueOf(sum*(long)(sum-1)).multiply(BigInteger.valueOf(2)).mod(BigInteger.valueOf(998244353)));
    }
}
