#include <bits/stdc++.h>
using namespace std;
int main(){
    long long h1,a1,c1,h2,a2;
    cin>>h1>>a1>>c1>>h2>>a2;
    long long need=h2/a1,rem=h2%a1;
    if(rem) need++;
    if(h1<=need*a2) need++;
    cout<<need<<endl;
    while(need--){
        if(h1+a2>h2) cout<<"STRIKE"<<endl;
        else {
            h1+=c1-a2;
            cout<<"HEAL"<<endl;
        }
    }
    return 0;
}
