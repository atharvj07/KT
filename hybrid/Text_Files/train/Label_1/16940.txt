#include<iostream>
using namespace std;

int main(){
    bool a[7],b;
    for(int i=0;i<24;++i){
        cin>>a[i%6+1];
        a[i%6+1]|=(i<6);
    }
    b=(a[1]&a[2]&a[4])|(a[3]&a[5]&a[6]);
    b&=(a[1]&a[2]&a[3])|(a[4]&a[5]&a[6]);
    b&=(a[1]&a[4]&a[5])|(a[2]&a[3]&a[6]);
    cout<<(b?"YES":"NO")<<'\n';
    return 0;
}
