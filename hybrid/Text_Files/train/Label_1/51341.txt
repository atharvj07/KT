import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            if (n == 0)
                break;

            int[][] a = new int[n][2];
            int[] p = new int[n];
            int[] s = new int[n];

            for (int i = 0; i < n; i++) {
                a[i][0] = scanner.nextInt();
                a[i][1] = scanner.nextInt();
                p[i] = a[i][0];
                s[i] = a[i][1];
            }

            Arrays.sort(p);
            Arrays.sort(s);

            int width = p[n - 1] - p[0] + 1;
            int height = s[n - 1] - s[0] + 1;

            System.out.println(width + " " + height);
        }
    }
}
