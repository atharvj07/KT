#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<int> cnt(26, 0);
    for (int i = 0; i < n; i++) {
        cnt[s[i] - 'A']++;
    }
    sort(cnt.begin(), cnt.end());
    int ans = 0;
    for (int i = 26 - k; i < 26; i++) {
        ans += cnt[i];
    }
    cout << ans << endl;
    return 0;
}
