s=input()
t=input()
d=dict()
for i in s:
 if i!='?':
  if i in d:d[i]+=1
  else:d[i]=1
l=list(t)
for i in l:
 if i in d:d[i]-=1
flag=True
for i in d.keys():
 if d[i]>0:flag=False
if flag:print(s)
else:
 o=[[i,d[i]]for i in d.keys()]
 o.sort(key=lambda x:x[1])
 idx=[-1]*len(s)
l=[0]*len(t)
for i in range(len(s)):
 if s[i]=='?':
  for j in range(len(t)):
   if l[j]<o[j][1]:
    idx[i]=j
    l[j]+=1
    break
for i in range(len(s)):
 if idx[i]==-1:
  for j in range(len(t)):
   if o[j][1]>0:
    idx[i]=j
    o[j][1]-=1
    break
for i in s:
 if i!='?':print(i,end='')
 else:print(t[idx[s.index(i)]],end='')
