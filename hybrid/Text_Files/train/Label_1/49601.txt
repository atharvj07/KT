import java.util.*;

class Main{
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int k = input.nextInt();

        int[] power = new int[n];
        int[] school = new int[n];
        for (int i = 0; i < n; i++)
            power[i] = input.nextInt();

        for (int i = 0; i < n; i++)
            school[i] = input.nextInt();

        int[] schoolCount = new int[m];
        int[] chosenCount = new int[m];
        for (int i = 0; i < n; i++)
            schoolCount[school[i] - 1]++;
        for (int i = 0; i < k; i++) {
            int chosen = input.nextInt();
            chosenCount[school[chosen - 1] - 1]++;
        }

        int ans = 0;
        for (int i = 0; i < m; i++)
            if (chosenCount[i] > 1)
                ans += chosenCount[i] - 1;

        System.out.println(ans);
    }
}
