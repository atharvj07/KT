#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> s(n + 2, 1);
  int ans = 0;
  for (int i = n; i > 0; i--) {
    int x = i, moves = n;
    if (s[i]) {
      while (s[x] && moves) {
        if (x >= k) { moves--; x--; }
        else { moves--; x++; }
      }
      ans += moves;
      if (s[x]) { s[i] = 0; s[x] = 0; }
      while (k > 1 && s[k]) { moves++; k--; }
      ans += moves;
    }
  }
  cout << ans << endl;
  return 0;
}
