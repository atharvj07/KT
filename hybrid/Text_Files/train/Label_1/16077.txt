#include <cstdio>
#include <vector>
using namespace std;
typedef long long LL;
int main() {
    int n, L, a;
    scanf("%d%d%d", &n, &L, &a);
    vector<LL> s(n + 2), f(n + 2);
    for (int i = 0; i < n; i++) {
        int t, l;
        scanf("%d%d", &t, &l);
        s[i + 1] = t;
        f[i + 1] = t + l;
    }
    s[n + 1] = L;
    f[0] = 0;
    LL ans = 0, cur = 0;
    for (int i = 1; i <= n + 1; i++) {
        cur = max(f[i - 1], cur + a);
        ans += (s[i] - cur) / a;
        cur = s[i];
    }
    printf("%lld\n", ans);
    return 0;
}
