#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, m;
    cin >> n >> k >> m;
    set<int> s;
    for (int i = 1; i <= n; i++) s.insert(i);
    for (int i = 0; i < m; i++) {
      int x;
      cin >> x;
      s.erase(x);
    }
    cout << (s.size() <= (n - m + k - 1) / k ? "YES" : "NO") << endl;
  }
}
