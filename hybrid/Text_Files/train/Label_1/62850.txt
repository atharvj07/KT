def check_yes_no(n: int) -> str:
    w = [input() for _ in range(n)]
    s = input()
    i = 0
    for x in s:
        if x == '<':
            j = i + 1
            while True:
                if s[j] == '3':
                    if i == 0 or ' '.join(w[i-1:]) == s[i+1:j]:
                        i = j+1
                        break
                    else:
                        return 'no'
                j += 1
    if i == len(s):
        return 'yes'
    else:
        return 'no'

n = int(input())
print(check_yes_no(n))
