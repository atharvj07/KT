import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextLong();
        }
        Arrays.sort(arr);
        long ans = -1;
        for (int i = n - 1; i >= 0; i--) {
            if (!isPerfectSquare(arr[i])) {
                ans = arr[i];
                break;
            }
        }
        System.out.println(ans);
    }

    static boolean isPerfectSquare(long x) {
        long sq = (long) Math.sqrt(x);
        return sq * sq == x;
    }
}
