#include <bits/stdc++.h>
using namespace std;
int main() {
  long long u, v;
  cin >> u >> v;
  if ((v & 1) != (u & 1)) {
    cout << "-1" << endl;
    return 0;
  }
  v /= 2;
  vector<int> ans = {v, u ^ v};
  while (ans.size() > 1 && ans.back() == 0) ans.pop_back();
  cout << ans.size() << endl;
  for (auto &x : ans) cout << x << " ";
  cout << endl;
  return 0;
}
