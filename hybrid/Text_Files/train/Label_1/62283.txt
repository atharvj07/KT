s = input()
pref = [-1] * len(s)
for i in range(1, len(s)):
    j = pref[i - 1]
    while j != -1 and s[j + 1] != s[i]:
        j = pref[j]
    pref[i] = j + 1
ans = []
cur = len(s) - 1
while cur != -1:
    ans.append([cur, s[: cur + 1]])
    cur = pref[cur]
ans.reverse()
for l, r in ans:
    print(len(r), r[:5] if len(r) > 10 else r + '..' if len(r) > 5 else r)
