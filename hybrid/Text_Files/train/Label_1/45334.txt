import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int[] p = new int[n + 1];
		for (int i = 1; i <= n; i++) {
			p[sc.nextInt()] = i;
		}
		int ans = 0;
		for (int i = 1; i <= n; i++) {
			if (p[i] == i) continue;
			ans++;
			swap(p, i, p[i]);
		}
		System.out.println(ans);
	}

	static void swap(int[] p, int i, int j) {
		int tmp = p[i];
		p[i] = p[j];
		p[j] = tmp;
	}
}
