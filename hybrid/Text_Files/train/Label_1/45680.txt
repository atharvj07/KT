import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long k = in.nextInt();
        int m = in.nextInt();
        long[] f = new long[60];
        f[1] = 1;//a
        f[2] = 1;//b
        for (int i = 3; i <= 58; i++) {
            f[i] = f[i - 1] + f[i - 2];
        }
        String b = "";
        for (int i = 0; i < k; i++) {
            b += (i % 2 + 1);
        }
        while (m-- > 0) {
            String s = in.next();
            long ans = 0;
            int l = b.length();
            int ls = s.length();
            for (int i = 0; i + ls - 1 < l; i++) {
                if (b.substring(i, i + ls).equals(s)) {
                    ans += f[i + ls - 1];
                }
            }
            System.out.println(ans % 1000000007);
        }
        in.close();
    }
}
