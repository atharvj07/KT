#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, m;
    cin >> n >> m;
    long long sum = 0;
    long long a[n];
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      sum += a[i];
    }
    sort(a, a + n);
    long long ans = 0;
    for (int i = 0; i < n; i++) {
      ans += (sum - a[i]) / (i + 1);
      sum -= a[i];
    }
    if (ans == m)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
