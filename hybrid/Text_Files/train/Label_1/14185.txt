#include<bits/stdc++.h>
using namespace std;
const int MAXN = 100010;
int arr[MAXN];
int gcd(int a,int b){
    if(a%b==0)
        return b;
    return gcd(b,a%b);
}
int main(){
    int n,i,j;
    scanf("%d",&n);
    for(i=0;i<n;++i)
        scanf("%d",&arr[i]);
    int ans = arr[0];
    for(i=1;i<n;++i){
        ans = gcd(ans,arr[i]);
        if(ans==1)
            return printf("1"),0;
    }
    printf("%d",ans);
    return 0;
}
