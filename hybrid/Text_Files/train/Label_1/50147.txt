import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

class Main {
    static int[] distance;
    static boolean[] visited;
    static int[] attack;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        distance = new int[123457];
        visited = new boolean[123457];
        Queue<Integer> q = new LinkedList<>();

        int n = sc.nextInt();
        int m = sc.nextInt();

        for (int i = 0; i < n - 1; i++) {
            int s = sc.nextInt();
            int e = sc.nextInt();
            if (s > e) {
                int temp = s;
                s = e;
                e = temp;
            }
        }

        attack = new int[m];
        for (int i = 0; i < m; i++) {
            attack[i] = sc.nextInt();
        }

        int ans = 0;

        for (int j : attack) {
            ans += distance[j];
        }

        q.offer(1);
        visited[1] = true;

        while (!q.isEmpty()) {
            int v = q.poll();
            for (int j : attack) {
                if (distance[v] + 1 < distance[j] || distance[j] == 0) {
                    distance[j] = distance[v] + 1;
                    q.offer(j);
                }
            }
        }

        int ans1 = 1;

        for (int j : attack) {
            if (distance[j] > distance[ans1]) {
                ans1 = j;
            }
        }

        System.out.println(ans1);
        System.out.println(ans);
    }
}
