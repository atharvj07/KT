import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            int n = sc.nextInt();
            int r = sc.nextInt();
            int[] x = new int[n];
            for (int i = 0; i < n; i++) {
                x[i] = sc.nextInt();
            }
            Arrays.sort(x);
            int ans = 0;
            for (int i = 0; i < n; i++) {
                if (x[i] <= r) {
                    ans++;
                } else {
                    break;
                }
            }
            System.out.println(ans);
        }
    }
}
