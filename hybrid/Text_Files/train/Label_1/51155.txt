import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            int n = in.nextInt();
            int[] b = new int[n];
            for (int j = 0; j < n; j++) {
                b[j] = in.nextInt();
            }
            int[] a = new int[n];
            a[0] = b[0];
            a[n - 1] = b[n - 1];
            int idx = 1;
            for (int j = n - 2; j >= 0; j--) {
                if (idx % 2 == 0) {
                    a[idx] = b[j];
                } else {
                    a[n - idx - 1] = b[j];
                }
                idx++;
            }
            for (int j = 0; j < n; j++) {
                System.out.print(a[j] + " ");
            }
            System.out.println();
        }
    }
}
