n, m = map(int, input().split())
s = input().strip()
t = input().strip()

left = [0] * m
right = [0] * m

j = 0
for i in range(n):
    if j < m and s[i] == t[j]:
        left[j] = i
        j += 1

j = m - 1
for i in range(n - 1, -1, -1):
    if j >= 0 and s[i] == t[j]:
        right[j] = i
        j -= 1

max_width = 0
for i in range(m - 1):
    max_width = max(max_width, right[i + 1] - left[i])

print(max_width)
