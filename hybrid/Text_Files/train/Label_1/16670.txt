#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, p;
    cin >> n >> p;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    vector<int> pre(n + 1);
    for (int i = 0; i < n; i++) pre[i + 1] = (pre[i] + a[i]) % p;
    int ans = 0;
    for (int i = 1; i < n; i++) ans = max(ans, (pre[n] - pre[i] + pre[i - 1]) % p);
    cout << ans << endl;
}
