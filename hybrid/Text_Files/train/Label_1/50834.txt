import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int na = sc.nextInt();
        String[] cardsA = new String[na];
        for (int i = 0; i < na; i++)
            cardsA[i] = sc.next();
        int nb = sc.nextInt();
        String[] cardsB = new String[nb];
        for (int i = 0; i < nb; i++)
            cardsB[i] = sc.next();
        sc.close();
        Robot robotA = new Robot(cardsA, na);
        Robot robotB = new Robot(cardsB, nb);
        while (robotA.cardsMoved < na && robotB.cardsMoved < nb) {
            robotA.move();
            robotB.move();
        }
        if (robotA.time == robotB.time)
            System.out.println(robotA.lastCard.compareTo(robotB.lastCard) > 0 ? "A wins." : "B wins.");
        else
            System.out.println(robotA.time < robotB.time ? "A wins." : "B wins.");
    }

    private static class Robot {
        private static final int DRAW_TIME = 300;
        private static final int MOVE_RIGHT_TIME = 500;
        private static final int MOVE_LEFT_TIME = 700;
        private static final int CANCEL_TIME = 500;
        private String[] cards;
        private int cardsMoved;
        private int numCards;
        private int time;
        private String lastCard;
        private int lastTable;

        public Robot(String[] cards, int numCards) {
            this.cards = cards;
            this.numCards = numCards;
            this.cardsMoved = 0;
            this.time = 0;
            this.lastCard = null;
            this.lastTable = -1;
        }

        public void move() {
            int timeLeft = MOVE_LEFT_TIME;
            int lastMoveTime = time;
            int startTable = lastTable;
            String startCard = lastCard;
            if (cardsMoved < numCards) {
                if (lastTable == -1 || lastCard != null) {
                    if (lastCard != null)
                        time += CANCEL_TIME;
                    lastTable = 1;
                    lastCard = cards[cardsMoved++];
                    time += DRAW_TIME;
                    timeLeft -= DRAW_TIME;
                }
                if (timeLeft > 0) {
                    if (time + MOVE_RIGHT_TIME <= lastMoveTime + timeLeft) {
                        time += MOVE_RIGHT_TIME;
                        startTable = 0;
                        lastCard = null;
                    }
                    else if (time + MOVE_LEFT_TIME <= lastMoveTime + timeLeft) {
                        time += MOVE_LEFT_TIME;
                        startTable = 1;
                    }
                }
            }
            if (startTable != lastTable) {
                if (startTable == -1 || startCard != null)
                    time += CANCEL_TIME;
                if (startTable == 0)
                    time += MOVE_RIGHT_TIME;
                else
                    time += MOVE_LEFT_TIME;
            }
        }
    }
}
