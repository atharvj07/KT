#include <bits/stdc++.h>
using namespace std;
#define lol long long
int n;
vector<lol> a;
int main(){
    cin >> n;
    a.resize(n + 3);
    for(int i = 1; i <= n; i++) cin >> a[i];
    int mx = 1;
    for(int i = 1; i <= n; i++){
        if(i + 2 <= n && a[i] == a[i + 1] - 1 && a[i + 1] == a[i + 2] - 1){
            int j = i;
            while(j + 1 <= n && a[j + 2] == a[j + 1] - 1) j++;
            mx = max(mx, j - i + 1);
            i = j;
        }
    }
    cout << mx;
}
