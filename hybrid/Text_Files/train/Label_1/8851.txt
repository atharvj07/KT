#include <iostream>
#include <vector>
using namespace std;
const int maxn = 1e5 + 5;
vector<int> g[maxn];
int fa[maxn], ans[maxn];
int find(int x){if(x == fa[x]) return x; return fa[x] = find(fa[x]);}
int merge(int x, int y){int fx = find(x), fy = find(y); if(fx != fy){fa[fx] = fy;}}
int main(){
    int n, m; cin >> n >> m;
    for(int i = 1; i <= n; ++i) fa[i] = i;
    for(int t, x, y, i = 1; i <= m; ++i){
        cin >> t;
        if(t == 1){cin >> x >> y; merge(x, y);}
        else if(t == 2){cin >> x; ans[x] = 1;}
        else {cin >> x >> y; if(find(x) == find(ans[y])) cout << "YES" << endl; else cout << "NO" << endl;}
    }
}
