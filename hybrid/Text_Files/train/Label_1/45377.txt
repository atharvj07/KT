import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d1 = sc.nextInt();
        int d2 = sc.nextInt();
        int[][] points = new int[n * n][2];
        int k = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                points[k][0] = i;
                points[k++][1] = j;
            }
        }
        if (d1 == d2) {
            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {
                    if (points[i][0] == points[j][0] || points[i][1] == points[j][1]) {
                        points[i] = points[k - 1];
                        k--;
                        break;
                    }
                }
            }
        }
        for (int i = 0; i < k; i++) {
            System.out.println(points[i][0] + " " + points[i][1]);
        }
    }
}
