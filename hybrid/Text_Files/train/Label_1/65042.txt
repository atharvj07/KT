import sys
input = sys.stdin.readline

n, h, m = map(int, input().split())
restrictions = [(l, r, x) for l, r, x in (map(int, input().split()) for _ in range(m))]
dp = [[[-1 for _ in range(h + 1)] for _ in range(n + 1)] for _ in range(m + 1)]
def cal(i, j, k):
    if i == m: return 0
    if dp[i][j][k] != -1: return dp[i][j][k]
    l, r, x = restrictions[i]
    res = cal(i + 1, j, k)
    if j >= l and j <= r and k < x:
        res = max(res, cal(i + 1, j + 1, k + 1) + (k + 1) ** 2)
    if j < n:
        res = max(res, cal(i, j + 1, k))
    dp[i][j][k] = res
    return res
print(cal(0, 0, 0))
