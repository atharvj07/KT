import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long c = sc.nextLong();
        long hr = sc.nextLong();
        long hb = sc.nextLong();
        long wr = sc.nextLong();
        long wb = sc.nextLong();
        if (wr > wb) {
            long tmp = wr;
            wr = wb;
            wb = tmp;
            tmp = hr;
            hr = hb;
            hb = tmp;
        }
        long ans = 0;
        while (c >= wr) {
            c -= wr;
            ans += hr;
        }
        while (c > 0) {
            if (c >= wb) {
                c -= wb;
                ans += hb;
            } else {
                break;
            }
        }
        System.out.println(ans);
    }
}
