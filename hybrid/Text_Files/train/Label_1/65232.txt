import sys

n, m = map(int, input().split())
d = {}
for _ in range(m):
    x, y = map(int, input().split())
    d[(x, y)] = d[(y, x)] = 1
i = 0
lst = []
for x in range(1, n + 1):
    if (x, x) in d:
        lst += [x, x]
        i += 1
    else:
        lst += [x]
        i += 1
    if i == 2:
        break
if i != 2:
    print('NO')
    sys.exit()
for x in range(1, n + 1):
    if x not in lst:
        lst += [x]
        break
print('YES')
print(*lst)
print(*lst[:i] + [lst[-1]] + lst[i + 1:])
