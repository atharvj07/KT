import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();

        int count = 0;
        for (int k = 2; k <= n; k++) {
            long x = n;
            while (x >= k) {
                if (x % k == 0) {
                    x /= k;
                } else {
                    x -= k;
                }
            }
            if (x == 1) {
                count++;
            }
        }

        System.out.println(count);
    }
}
