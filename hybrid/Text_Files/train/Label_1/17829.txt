#include <bits/stdc++.h>

using namespace std;

const int N = 4e6 + 7;

vector<bool> is_prime;
vector<bool> is_palindrome;

void init()
{
  is_prime.assign(N, 1);
  is_palindrome.assign(N, 1);
  is_prime[0] = is_prime[1] = 0;
  for (int i = 2; i * i < N; i++)
    if (is_prime[i])
      for (int j = i * i; j < N; j += i)
        is_prime[j] = 0;

  for (int i = 1; i < N; i++)
  {
    string tmp = to_string(i);
    string rev(tmp.rbegin(), tmp.rend());
    if (tmp != rev)
      is_palindrome[i] = 0;
  }
}

int solve(int p, int q)
{
  long double ratio = (long double)p / q;

  int ans = 0;
  long long sum = 0;
  int cnt = 0;
  for (int i = 1; i < N; i++)
  {
    sum += is_prime[i];
    if (is_palindrome[i])
      cnt++;
    while (ratio * cnt > sum)
    {
      sum -= is_prime[ans];
      if (is_palindrome[ans])
        cnt--;
      ans++;
    }
    if (ratio * cnt > sum)
      break;
  }
  return ans;
}

int main()
{
  init();
  int p, q;
  while (cin >> p >> q)
  {
    int res = solve(p, q);
    if (!res)
      cout << "Palindromic tree is better than splay tree\n";
    else
      cout << res << "\n";
  }
  return 0;
}
