#include<bits/stdc++.h>
using namespace std;
long long n,x,a,t,d;
long long b[100001];
int main(){
    cin>>n>>x;
    for(int i=0;i<n;i++){
        cin>>a;
        if(a<100)b[a/10]++;
    }
    t=a=0;
    while(x>0&&a<n){
        while(x>0&&b[a]>=1){
            t++;
            b[a]--;
            x--;
        }
        a++;
    }
    cout<<(t)<<endl;
}
