for _ in range(int(input())):
    n, k = map(int, input().split())
    n = list(str(n))
    d = {}
    for i in n:
        if i not in d:
            d[i] = 0
        d[i] += 1
    cnt = len(d)
    if cnt <= k:
        print(''.join(n))
    else:
        if cnt == k + 1:
            for i in range(len(n)):
                if d[n[i]] == 1:
                    n.pop(i)
                    break
        rem = k - cnt + 1
        for i in n:
            if rem == 0:
                break
            if d[i] > 1:
                x, y, z = d[i], rem, int(i)
                d[i] -= min(x, y)
                rem -= min(x, y)
                n[i] = str(z - min(x, y))
        m = ''
        for i in n:
            m += i
        print(m)
