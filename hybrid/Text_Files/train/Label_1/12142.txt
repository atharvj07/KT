#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;

    vector<pair<int, int>> guests(n);
    for (int i = 0; i < n; i++) {
        cin >> guests[i].first >> guests[i].second;
    }

    sort(guests.begin(), guests.end());

    int ans = 0, l = 0, r = 0;
    for (int i = 0; i < n; i++) {
        ans += max(0, guests[i].first - l);
        ans += max(0, guests[i].second - r);
        l = guests[i].first;
        r = guests[i].second;
    }

    cout << ans << endl;
    return 0;
}
