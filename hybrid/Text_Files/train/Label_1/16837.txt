#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll MOD = 998244353;
int a[200001], cnt[1000001];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i], cnt[a[i]]++;
  int ans = 1;
  for (int i = 1; i <= 1000000; i++) {
    ans = 1ll * ans * (cnt[i] + 1) % MOD;
  }
  cout << ans << endl;
  return 0;
}
