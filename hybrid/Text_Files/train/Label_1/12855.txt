#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  ll n, k;
  cin >> n >> k;
  if (n < k || k > n / 2)
    cout << "-1";
  else {
    string s = "";
    ll lst = 'a' - 1;
    for (ll i = 1; i <= k; i++) {
      char c = lst + 1;
      while (s.size() < n && c == s.back()) c++;
      s += c;
      lst = c;
    }
    cout << s;
  }
}
