#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e5 + 10, M = 5;
ll a[N][M], f[N][M][M], l[N][M], r[N][M], dp[N];
int main() {
    ll n, m, k; cin >> n >> m >> k;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
            dp[i] += a[i][j];
        }
    }
    for (int j = 0; j < m; j++) {
        for (int i = n - 1; i >= 0; i--) {
            if (i == n - 1)
                r[i][j] = dp[i];
            else
                r[i][j] = r[i + 1][j] - a[i][j];
        }
        for (int i = 0; i < n; i++) {
            if (i == 0)
                l[i][j] = dp[i];
            else
                l[i][j] = l[i - 1][j] - a[i][j];
        }
        for (int i = 0; i < n; i++) {
            for (int kk = 0; kk < m; kk++) {
                f[i][j][kk] = 1e18;
            }
        }
        f[0][j][j] = 0;
        for (int i = 0; i < n; i++) {
            for (int j1 = 0; j1 < m; j1++) {
                for (int j2 = 0; j2 < m; j2++) {
                    f[i + 1][j2][j1] = min({
                        f[i + 1][j2][j1],
                        f[i][j2][j1] + a[i][j],
                        f[i][j2][j] + l[i + 1][j1],
                        f[i][j][j2] + r[i][j1]
                    });
                }
            }
        }
        ll ans = 1e18;
        int u = -1;
        for (int j1 = 0; j1 < m; j1++) {
            ans = min(ans, f[n][j1][j]);
        }
        if (ans <= k)
            u = j;
        cout << (u == -1 ? 0 : 1) << " ";
    }
}
