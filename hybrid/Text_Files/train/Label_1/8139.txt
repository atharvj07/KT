#include <iostream>
using namespace std;

int s[100005], ans[28];

int main() {
  string a;
  cin >> a;
  int n = a.size();
  for (int i = 0; i < n; i++) s[a[i] - 'a']++;
  int k = 0;
  for (int i = 0; i < 26; i++) {
    for (int j = 0; j < s[i]; j++) { ans[k] = 'a' + i; k++; }
  }
  if (k == 26) {
    for (int i = 0; i < 26; i++) cout << char(ans[i]);
  } else {
    cout << "-1";
  }
  return 0;
}
