import java.util.*;
import java.io.*;

public class Main {
    static boolean[] vis;
    static int[] match;
    static int m;
    static boolean dfs(int v) {
        if (vis[v]) return false;
        vis[v] = true;
        for (int w = 0; w < m; w++) {
            if (match[w] == -1 || dfs(match[w])) {
                match[w] = v;
                return true;
            }
        }
        return false;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        StringTokenizer st = new StringTokenizer(in.readLine());
        int[] p = new int[n];
        for (int i = 1; i <= n; i++) p[i - 1] = Integer.parseInt(st.nextToken()) - 1;
        int[] invp = new int[n];
        for (int i = 0; i < n; i++) invp[p[i]] = i;
        boolean ans = true;
        for (int i = 0; i < n; i++) ans &= invp[i] == p[p[i]];
        if (!ans) {
            System.out.println(-1);
            return;
        }
        m = n;
        match = new int[m];
        Arrays.fill(match, -1);
        for (int i = 0; i < n; i++) {
            vis = new boolean[n];
            if (!dfs(i)) {
                match[p[i]] = -1;
                m--;
            }
        }
        System.out.println(m);
        for (int i = 0; i < n; i++) {
            if (match[p[i]] == -1) System.out.print((match[i] + 1) + " ");
            else if (match[p[i]] == i) System.out.print(-1 + " ");
            else System.out.print(0 + " ");
        }
        System.out.println();
        for (int i = 0; i < m; i++) System.out.print((match[i] + 1) + " ");
    }
}
