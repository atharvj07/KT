#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while(t--)
    {
        int r, c;
        cin >> r >> c;
        vector<string>v(r);
        for(int i = 0; i < r; i++){
            cin >> v[i];
        }
        bool f = false;
        for(int i = 0; i < r; i++){
            for(int j = 0; j < c; j++){
                if(v[i][j] == 'A'){
                    f = true;
                    break;
                }
            }
            if(f) break;
        }
        if(!f){
            cout << "MORTAL" << endl;
            continue;
        }
        int ans = 0;
        for(int i = 0; i < r; i++){
            int x = 0;
            for(int j = 0; j < c; j++){
                if(v[i][j] == 'P'){
                    x++;
                }
                else x = 0;
                if(x >= c){
                    ans++;
                    x = 0;
                    for(int k = 0; k < c; k++) v[i][k] = 'A';
                }
            }
        }
        for(int j = 0; j < c; j++){
            int x = 0;
            for(int i = 0; i < r; i++){
                if(v[i][j] == 'P'){
                    x++;
                }
                else x = 0;
                if(x >= r){
                    ans++;
                    x = 0;
                    for(int k = 0; k < r; k++) v[k][j] = 'A';
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
