#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> ans;
  char prv = s[0];
  int cnt = 1;
  for (int i = 1; i < n; i++) {
    if (s[i] == prv)
      cnt++;
    else {
      ans.push_back(cnt);
      cnt = 1;
      prv = s[i];
    }
  }
  ans.push_back(cnt);
  cout << ans.size() << endl;
  for (int x : ans) cout << x << " ";
  cout << endl;
  return 0;
}
