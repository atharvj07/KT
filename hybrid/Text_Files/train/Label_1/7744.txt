#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  vector<char> v;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (v.empty() || v.back() != s[i]) {
      v.push_back(s[i]);
    } else {
      v.pop_back();
      ans++;
    }
  }

  if (v.size() % 2 == 1) {
    ans++;
  }

  cout << ans << endl;
}
