#include<bits/stdc++.h>
#define pii pair<int, int>
using namespace std;

int main(int argc, char const *argv[])
{
    int n, deg[200005], s[200005], ans = 0;
    map<int, vector<int>> Xor;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> deg[i] >> s[i];
        Xor[s[i]].push_back(i);
    }
    
    for (auto x : Xor) {
        if (x.second.size() <= 1) continue;
        ans += x.second.size() - 1;
        for (int i = 1; i < x.second.size(); ++i) {
            cout << x.second[i] << " " << x.second[i - 1] << endl;
        }
    }
    
    cout << ans << endl;
    return 0;
}
