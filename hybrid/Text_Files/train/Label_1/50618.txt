import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int h = input.nextInt();
        int n = input.nextInt();
        int ans = 1;
        int l = 1, r = 2;
        while (n > r) {
            ans += 2;
            n = (n + 1) / 2;
            l *= 2;
            r *= 2;
            r++;
        }
        System.out.println(ans);
    }
}
