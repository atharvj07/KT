import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		String s = sc.next();
		System.out.println(s.charAt(0) == 'A' && s.indexOf('C') >= 2 && s.indexOf('C') < s.length() - 2 && s.replaceAll("[AC]", "").matches("[a-z]*") ? "AC" : "WA");
	}
}
