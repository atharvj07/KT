mod = int(1e9) + 7
def powMod(x, p):
    if p == 0: return 1
    if p == 1: return x
    val = powMod(x, int(p / 2))
    val = (val * val) % mod
    if p % 2 != 0:
        val = (val * x) % mod
    return val

s = input()
n = len(s)
# dp[i][l][r][prv] = # ways to fill [i, n] with bombs and numbers
# given that [0, i - 1] is correct andprv is the value of the cell before i
dp = [[[[0] * 3 for k in range(3)] for j in range(n + 1)] for i in range(n + 1)]
dp[n][n][0][0] = dp[n][n][0][1] = dp[n][n][0][2] = 1
for i in range(n - 1, -1, -1):
    for l in range(n):
        for r in range(n):
            for prv in range(3):
                # put a bomb
                dp[i][l][r][prv] = (dp[i][l][r][prv] + dp[i + 1][l][r + 1][0]) % mod
                # put a number
                if s[i] == '?' or s[i] == str(prv):
                    dp[i][l][r][prv] = (dp[i][l][r][prv] + dp[i + 1][l][r][prv + 1]) % mod
                # leave a question mark
                if s[i] == '?':
                    dp[i][l][r][prv] = (dp[i][l][r][prv] + dp[i + 1][l][r][prv]) % mod
                # check if the current configuration is valid
                if l > 0 and s[i - 1] != '?' and int(s[i - 1]) < prv:
                    dp[i][l][r][prv] = 0
                if r > 0 and s[i + 1] != '?' and int(s[i + 1]) < prv:
                    dp[i][l][r][prv] = 0
print(dp[0][0][0][0])
