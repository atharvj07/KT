#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
typedef long long ll;

int main() {
    ll n, ans=0;
    cin >> n;
    vector<ll> a(n);
    for (int i=0; i<n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    ll l=0, r=n-1, num=1e18, mx=0;
    while (l<=r) {
        ll mid=(l+r)/2;
        ll sum=a[mid]*2+a[mid+1];
        if (sum>=num) r=mid-1;
        else {
            mx=max(mx, sum);
            l=mid+1;
        }
    }
    num=mx;
    l=0, r=n-1;
    while (l<=r) {
        ll mid=(l+r)/2;
        ll sum=a[mid]*2+a[mid+1];
        if (sum<=num) l=mid+1;
        else r=mid-1;
    }
    if (l>=n) ans=-1;
    else {
        num=a[l]*2+a[l+1];
        if (num<=mx) ans=l+1;
        else ans=l+2;
    }
    cout << ans << endl;
}
