import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        boolean[] inv = new boolean[(n * (n - 1)) / 2];
        int[][] invs = new int[n][n];
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[i] > a[j]) {
                    inv[cnt] = true;
                    invs[i][j] = cnt;
                    cnt++;
                }
            }
        }
        int[] swaps = new int[cnt];
        for (int i = 0; i < cnt; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (invs[j][k] == i) {
                        swaps[i] = j + 1;
                        swaps[i + 1] = k + 1;
                        i++;
                        break;
                    }
                }
            }
        }
        boolean flag = true;
        for (int i = 0; i < cnt; i++) {
            if (!inv[i]) {
                flag = false;
                break;
            }
        }
        if (!flag) {
            System.out.println(-1);
        } else {
            System.out.println(cnt / 2);
            for (int i = 0; i < cnt; i += 2) {
                System.out.println(swaps[i] + " " + swaps[i + 1]);
            }
        }
    }
}
