#include <cstdio>
#include <cmath>
int main() {
    double y1, y2, yw, xb, yb, r;
    scanf("%lf %lf %lf %lf %lf %lf", &y1, &y2, &yw, &xb, &yb, &r);
    double xw;
    xw = (xb + xb * sqrt((yw - yb) * (yw - yb) / (xb - (yb - y1) * (yw - yb) / (y2 - y1)) - 1));
    if (xb + r < xw)
        printf("%.12f\n", xw);
    else
        puts("-1");
    return 0;
}
