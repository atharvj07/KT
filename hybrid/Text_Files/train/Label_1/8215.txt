#include <bits/stdc++.h>
using ll=long long;
using namespace std;
int n,k,a[400001],L[400001],R[400001],f[400001];
int main(){
  cin>>n>>k;++k;
  for(int i=1;i<=n;++i)cin>>a[i];
  for(int i=1;i<=n;++i)
    if((i<=k)||(a[i]>a[i-1]))L[i]=i-1;else L[i]=L[i-1];
  for(int i=n;i>=1;--i)
    if((i>k)||(a[i]<a[i+1]))R[i]=i+1;else R[i]=R[i+1];
  for(int i=1;i<=n;++i)
    f[i]=max(max(f[L[i]],f[R[i]]),1+(i<=k));
  cout<<*max_element(f+1,f+n+1);
  return 0;
}
