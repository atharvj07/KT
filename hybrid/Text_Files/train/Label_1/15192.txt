#include <cstdio>
#include <cstdlib>
#include <string>
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
const int N = 26;
int main() {
    int n;
    cin >> n;
    vector<string> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
    }
    sort(v.begin(), v.end());
    int perm[N];
    for (int i = 0; i < N; i++) {
        perm[i] = i;
    }
    for (int i = 0; i < n - 1; i++) {
        bool found = false;
        string &s = v[i];
        string &t = v[i + 1];
        for (int j = 0; j < min(s.size(), t.size()); j++) {
            if (s[j] != t[j]) {
                found = true;
                for (int k = 0; ; k++) {
                    if (s[j] == 'a' + k) {
                        for (int l = 0; l < N; l++) {
                            if (perm[l] == 'a' + k) {
                                perm[l] = 'a' + perm[j];
                                perm[j] = 'a' + k;
                                break;
                            }
                        }
                        break;
                    }
                }
                break;
            }
        }
        if (!found) {
            cout << "Impossible" << endl;
            return 0;
        }
    }
    for (int i = 0; i < N; i++) {
        cout << static_cast<char>(perm[i]);
    }
    cout << endl;
    return 0;
}
