#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  int cnt_a = 0, cnt_b = 0, cnt_c = 0;
  for (char c : s) {
    if (c == 'a') cnt_a++;
    if (c == 'b') cnt_b++;
    if (c == 'c') cnt_c++;
  }
  if (cnt_a > 0 && cnt_b > 0 && cnt_c == min(cnt_a, cnt_b)) cout << "YES" << endl;
  else cout << "NO" << endl;
  return 0;
}
