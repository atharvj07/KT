positions = ['^', '>', 'v', '<']
input_positions = input().split()
n = int(input())

start_pos = positions.index(input_positions[0])
end_pos = positions.index(input_positions[1])

cw_distance = (end_pos - start_pos) % 4
ccw_distance = (start_pos - end_pos) % 4

if cw_distance == n % 4 and ccw_distance == n % 4:
    print("undefined")
elif cw_distance == n % 4:
    print("cw")
else:
    print("ccw")
