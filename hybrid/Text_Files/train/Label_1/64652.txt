n, y1 = map(int, input().split())
a = sorted(list(map(int, input().split())))
m, y2 = map(int, input().split())
b = sorted(list(map(int, input().split())))
up0 = [None] * len(a)
up1 = [None] * len(a)
for i in range(len(a)):
    up0[i] = (a[i]*2 - y1) / (y2 - y1) if i%2 == 0 else (a[i]*2 - y1 + 10**9) / (y2 - y1)
    up1[i] = (a[i]*2 + y1) / (y2 - y1) if i%2 == 1 else (a[i]*2 + y1 + 10**9) / (y2 - y1)
down0 = [None] * len(b)
down1 = [None] * len(b)
for i in range(len(b)):
    down0[i] = (b[i]*2 - y1) / (y2 - y1) if i%2 == 0 else (b[i]*2 - y1 + 10**9) / (y2 - y1)
    down1[i] = (b[i]*2 + y1) / (y2 - y1) if i%2 == 1 else (b[i]*2 + y1 + 10**9) / (y2 - y1)
#print(up0)
#print(up1)
#print(down0)
#print(down1)
ans = 0
for ua in up0:
    for ua2 in up1:
        ub = max(*down0)
        lb = min(*down1)
        ans = max(ans, (ua <= ub) + (ua <= lb) + (ua2 <= ub) + (ua2 <= lb))
print(ans)
