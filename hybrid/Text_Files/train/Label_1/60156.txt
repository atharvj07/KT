def smallest_number_with_sum_x(x):
    if x > 45:  # The sum of the digits 1-9 is 45, so anything larger is not possible.
        return -1
    number = ''
    for digit in range(9, 0, -1):
        if x >= digit:
            number = str(digit) + number
            x -= digit
    return int(number)

t = int(input())
for _ in range(t):
    x = int(input())
    print(smallest_number_with_sum_x(x))
