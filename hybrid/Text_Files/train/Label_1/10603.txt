#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; ++i) {
        int w, h;
        cin >> w >> h;
        v[i] = {w, h};
    }
    int mx = 0, sum = 0;
    for (auto &x : v) {
        mx = max(mx, x.second);
        sum += x.first;
    }
    for (auto &x : v) {
        cout << sum + mx - x.first - x.second << ' ';
    }
    return 0;
}
