import java.util.*;

public class Main {
    static long P = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        boolean[][] dp = new boolean[n][n];
        long[][] count = new long[n][n];
        for (int i = 0; i < n; i++) {
            dp[i][i] = s.charAt(i) == '1';
            count[i][i] = 1;
        }
        for (int l = 2; l <= n; l++) {
            for (int i = 0; i <= n - l; i++) {
                int j = i + l - 1;
                if (dp[i + 1][j - 1] && s.charAt(i) == '0' && s.charAt(j) == '1') {
                    long sum = 0;
                    int size = 1;
                    for (int k = i + 1; k < j; k++) {
                        sum += count[i + 1][k] * count[k + 1][j - 1] % P;
                        sum %= P;
                        size++;
                    }
                    if (size % 2 == 0 && sum == (size - 1) / 2 && dp[i + 1][j - 1]) {
                        dp[i][j] = true;
                        count[i][j] = sum + 1;
                    }
                }
            }
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                ans += count[i][j];
                ans %= P;
            }
        }
        System.out.println(ans);
    }
}
