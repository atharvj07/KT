#include <iostream>
#include <unordered_map>
#include <string>
#include <algorithm>
using namespace std;

long long mod = 1000000007;

long long pow(long long x, long long p) {
    if (p == 0) {
        return 1;
    }
    if (p == 1) {
        return x;
    }
    long long val = pow(x, p / 2);
    val = (val * val) % mod;
    if (p % 2 != 0) {
        val = (val * x) % mod;
    }
    return val;
}

int main() {
    string s;
    cin >> s;
    int n = s.length();
    unordered_map<char, int> mp;
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '?') {
            cnt++;
        } else if (s[i] >= 'A' && s[i] <= 'J') {
            mp[s[i]]++;
        }
    }
    long long ans = pow(10, cnt);
    for (auto x : mp) {
        ans = (ans * pow(10, x.second)) % mod;
    }
    cout << ans << endl;
    return 0;
}
