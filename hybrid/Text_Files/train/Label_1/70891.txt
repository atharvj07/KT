t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    last_pos = {}
    max_dist = {}
    for i, num in enumerate(a):
        if num in last_pos:
            dist = i - last_pos[num]
            if num in max_dist:
                max_dist[num] = max(max_dist[num], dist)
            else:
                max_dist[num] = dist
        else:
            max_dist[num] = i + 1
        last_pos[num] = i
    for num in last_pos:
        max_dist[num] = max(max_dist[num], n - last_pos[num])
    result = [-1] * n
    for num, dist in max_dist.items():
        if result[dist-1] == -1 or result[dist-1] > num:
            result[dist-1] = num
    for i in range(1, n):
        if result[i] == -1 or (result[i-1] != -1 and result[i] > result[i-1]):
            result[i] = result[i-1]
    print(' '.join(map(str, result)))
