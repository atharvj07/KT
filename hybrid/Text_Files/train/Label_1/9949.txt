#include <iostream>
#include <map>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  map<string, int> f;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    f[s]++;
  }
  vector<int> v(m);
  for (int i = 0; i < m; i++) {
    cin >> v[i];
  }
  string ans;
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = 0; j < m; j++) {
      if (v[j] == i + 1) {
        cnt++;
      }
    }
    if (cnt == 0) {
      continue;
    }
    string s;
    cin >> s;
    if (!ans.empty()) {
      for (int j = 0; j < ans.size(); j++) {
        if (ans[j] != s[j] && ans[j] != '?') {
          goto next;
        }
      }
    }
    ans = s;
  next:;
  }
  if (ans.empty()) {
    cout << "No" << endl;
  } else {
    cout << "Yes" << endl;
    for (int i = 0; i < ans.size(); i++) {
      if (ans[i] == '?') {
        ans[i] = 'a';
      }
    }
    cout << ans << endl;
  }
  return 0;
}
