#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> want(n), sales(2e5 + 1, 0), ans(n, 2e9);
  for (int &x : want) cin >> x;
  for (int i = 0; i < m; i++) {
    int d, t;
    cin >> d >> t;
    t--;
    sales[d] |= (1 << t);
  }
  sales[0] = (1 << n) - 1;
  for (int sum = 0, day = 0; day <= 2e5; day++) {
    if ((sum |= sales[day]) == (1 << n) - 1) {
      for (int i = 0; i < n; i++)
        if (ans[i] > day) ans[i] = day;
      int mx = 0;
      for (int i = 0; i < n; i++) mx = max(mx, ans[i] + want[i] - 1);
      cout << mx + 1;
      return 0;
    }
  }
  cout << -1;
}
