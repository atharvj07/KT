n=int(input());a=[list(map(int,i.split()))for i in[input()for j in range(n)]]
good=[]
for i in range(n):
    b=True
    for j in range(n):
        if i==j:continue
        if a[i][j]==2 or a[i][j]==3:
            b=False
        if a[j][i]==1 or a[j][i]==3:
            b=False
    if b:
        good.append(i+1)
print(len(good))
print(*good)
