#include <cstdio>
#include <vector>
using namespace std;
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n;
    scanf("%d", &n);
    vector<int> a(n);
    int sum = 0;
    for (int i = 0; i < n; i++) scanf("%d", &a[i]), sum += a[i] % 2;
    if (sum)
      printf("-1\n");
    else
      printf("1\n1\n");
  }
}
