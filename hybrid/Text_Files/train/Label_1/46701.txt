import java.util.*;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] binary = sc.next().split("");
        BigInteger mod = new BigInteger("1000000007");
        BigInteger[][] dp = new BigInteger[n + 1][n + 1];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = i; j < n; j++) {
                int l = i >= 0 ? Integer.valueOf(binary[i]) : 0;
                int r = j < n ? Integer.valueOf(binary[j]) : 0;
                int k = j - i + 1;
                BigInteger sum = BigInteger.ZERO;
                for (int m = k - 1; m >= 1; m--) {
                    if (m + l <= j - r && m >= l) {
                        sum = sum.add(dp[i][i + m - 1].multiply(dp[i + m][j]));
                    }
                }
                dp[i][j] = sum.mod(mod);
                if (i == 0 && j == n - 1) {
                    dp[i][j] = dp[i][j].add(BigInteger.ONE);
                }
            }
        }
        BigInteger sum = BigInteger.ZERO;
        for (int i = 1; i <= n; i++) {
            sum = sum.add(dp[0][n - 1].pow(i));
        }
        System.out.println(sum.mod(mod));
    }
}
