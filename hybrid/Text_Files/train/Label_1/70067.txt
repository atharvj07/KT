MOD = 10**9 + 7

def solve(n, arr):
    coeff = 1
    result = 0
    for i in range(n):
        result = (result + arr[i] * coeff) % MOD
        coeff = coeff * (n - i - 1) * pow(i + 1, MOD - 2, MOD) % MOD * ((-1) ** (i % 2)) % MOD
    return result

n = int(input().strip())
arr = list(map(int, input().strip().split()))
print(solve(n, arr))
