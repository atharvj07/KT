#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
#define rep(i, a, b) for(ll i = a; i < b; i++)
#define pb push_back
int main(){
ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    ll n, q;
    cin >> n >> q;
    vector<ll> block(n + 1), hindrance(n + 1);
    rep(i, 0, q){
        ll r, c;
        cin >> r >> c;
        if(!block[c]){
            hindrance[c] = !hindrance[c];
        }
        else{
            hindrance[c] = false;
            hindrance[c - 1] = false;
            hindrance[c + 1] = false;
        }
        block[c] = !block[c];
        if(hindrance[1] || block[n] || (block[1] && !hindrance[2])){
            cout << "No\n";
        }
        else{
            cout << "Yes\n";
        }
    }
    return 0;
}
