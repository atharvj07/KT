import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] x = new int[4 * n + 1];
        int[] y = new int[4 * n + 1];
        for (int i = 0; i < 4 * n + 1; i++) {
            x[i] = scanner.nextInt();
            y[i] = scanner.nextInt();
        }
        int[] cx = new int[51];
        int[] cy = new int[51];
        for (int i = 0; i < 4 * n + 1; i++) {
            cx[x[i]]++;
            cy[y[i]]++;
        }
        int ox = 0, oy = 0;
        for (int i = 0; i < 51; i++) {
            if (cx[i] == n) {
                ox = i;
            }
            if (cy[i] == n) {
                oy = i;
            }
        }
        for (int i = 0; i < 4 * n + 1; i++) {
            if (x[i] != ox && y[i] != oy) {
                System.out.println(x[i] + " " + y[i]);
                return;
            }
        }
    }
}
