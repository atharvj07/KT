#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int n;
  string s;
  cin >> n >> s;
  vector<int> cnt(4);
  for (char c : s) {
    if (c != '?') cnt[c - 'A']++;
  }
  int avg = (n + 3) / 4;
  bool ok = true;
  for (int i = 0; i < 4; i++) {
    if (cnt[i] > avg) ok = false;
    cnt[i] = avg - cnt[i];
  }
  if (!ok) {
    cout << "===" << endl;
    return 0;
  }
  for (int i = 0, j = 0; i < n; i++) {
    if (s[i] == '?') {
      while (!cnt[j]) j++;
      cnt[j]--;
      s[i] = 'A' + j;
    }
  }
  cout << s << endl;
  return 0;
}
