import bisect
import heapq

n,m,k = map(int,input().split())
a = [*map(int,input().split())]
a.sort()
b = []
for i in range(m):
    x,y = map(int,input().split())
    b.append((y,x))
b.sort()
s = []
ans = 2e9
for x in a:
    now = -x
    while s and -s[0][1]<=now:
        heapq.heappop(s)
    heapq.heappush(s,(now,-len(s)))
    if len(s)>=k:
        ans = min(ans,s[0][1]*(-x))
    idx = bisect.bisect_left(b,(len(s),))
    for y,x in b[idx:]:
        if len(s)+x<=k:
            ans = min(ans,s[0][1]*(-x)+a[len(s)+x-1])
        else:
            break
print(ans)
