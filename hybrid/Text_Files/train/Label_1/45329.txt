import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            if (n == 0) {
                break;
            }
            int count = 0;
            for (int i = 1; i <= n / 2; i++) {
                int sum = i;
                int j = i + 1;
                while (sum < n) {
                    sum += j;
                    j++;
                }
                if (sum == n) {
                    count++;
                }
            }
            System.out.println(count);
        }
    }
}
