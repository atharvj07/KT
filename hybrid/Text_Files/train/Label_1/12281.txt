#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define mod 998244353
#define pb push_back
#define F first
#define S second
#define all(v) v.begin(), v.end()
#define pr(v)             \
    for (auto x : v)      \
        cout << x << " "; \
    cout << endl;
#define endl "\n"
int n;
vector<ll> a;
vector<ll> b;
int main()
{
    cin >> n;
    a.resize(n), b.resize(n);
    for (int i = 0; i < n; i++)
        cin >> a[i];
    for (int i = 0; i < n; i++)
        cin >> b[i];
    sort(all(b));
    ll ans = 0;
    for (int i = 0; i < n; i++)
        ans = (ans + (i + 1ll) * a[i] % mod * b[i] % mod) % mod;
    cout << ans << endl;
    return 0;
}
