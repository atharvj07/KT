#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, M;
    cin >> n >> M;
    vector<int> t(n);
    priority_queue<int> q;
    int ans[105];
    for (int i = 0; i < n; i++) {
        cin >> t[i];
    }
    int cur = 0;
    for (int i = 0; i < n; i++) {
        cur += t[i];
        while (q.size() && cur > M) {
            cur -= q.top();
            q.pop();
        }
        q.push(t[i]);
        ans[i] = q.size() - 1;
    }
    for (int i = 0; i < n; i++) {
        cout << ans[i] << " ";
    }
    return 0;
}
