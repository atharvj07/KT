n = int(input().strip())
s = input().strip()

def max_alternating_subsequence_after_flip(s):
    # Initial counts for the alternating sequence which starts and ends with either 0 or 1.
    count_0 = 1
    count_1 = 1

    for i in range(1, len(s)):
        if s[i] != s[i-1]:
            count_0 += 1
            count_1 += 1
        else:
            # If we encounter two same characters consecutively, we can consider flipping this part to make it alternating.
            # Hence, we only add to the count for the sequence that does not match the current character, simulating a flip.
            if s[i] == '0':
                count_1 += 1
            else:
                count_0 += 1
    
    # The maximum possible length after a flip would be the maximum of these two counts.
    return max(count_0, count_1)

print(max_alternating_subsequence_after_flip(s))
