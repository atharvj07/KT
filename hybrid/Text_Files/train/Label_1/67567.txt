import sys

n = int(sys.stdin.readline())
x = []
y = []
for i in range(4 * n):
    xi, yi = map(int, sys.stdin.readline().split())
    x.append(xi)
    y.append(yi)

x.sort()
y.sort()

ans_x = -1
ans_y = -1
for i in range(1, n):
    if x[i] != x[i - 1]:
        ans_x = x[i - 1]
    if y[i] != y[i - 1]:
        ans_y = y[i - 1]

print(ans_x, ans_y)
