n = int(input())
st = []
st.append(0)
f = 0
for i in range(n): st.append(1)
ans = []
for i in range(2*n):
    s = input()
    if s[0]=='+': st.append(1)
    else:
        x = int(s[2:])
        if x<=st[-1]:
            ans.append(x)
            while st[-1]>x: st.pop()
        else: f = 1
if f==1: print("NO")
else:
    print("YES")
    st.sort()
    for i in st[1:]:
        if i==ans[-1]: ans.pop()
        print(i, end=' ')
