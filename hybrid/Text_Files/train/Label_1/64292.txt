def can_forge_signature(n, m, grid):
    forged_grid = [['.' for _ in range(m)] for _ in range(n)]
    for i in range(1, n-1):
        for j in range(1, m-1):
            if grid[i-1][j-1] == '#' and grid[i-1][j] == '#' and grid[i-1][j+1] == '#' and grid[i][j-1] == '#' and grid[i][j+1] == '#' and grid[i+1][j-1] == '#' and grid[i+1][j] == '#' and grid[i+1][j+1] == '#':
                forged_grid[i-1][j-1] = forged_grid[i-1][j] = forged_grid[i-1][j+1] = forged_grid[i][j-1] = forged_grid[i][j+1] = forged_grid[i+1][j-1] = forged_grid[i+1][j] = forged_grid[i+1][j+1] = '#'
    return "YES" if forged_grid == grid else "NO"

n, m = map(int, input().split())
grid = [list(input().strip()) for _ in range(n)]
print(can_forge_signature(n, m, grid))
