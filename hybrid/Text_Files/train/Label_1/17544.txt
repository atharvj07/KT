#include<bits/stdc++.h>
#define mod 1000000007
using namespace std;
int adj[505][505],deg[505],vis[505],subt[505],nxt[505],par[505];
void dfs2(int u){
	vis[u] = 1;
	for(int i = 0; i < deg[u]; ++i){
		int v = adj[u][i];
		if(!vis[v]) dfs2(v);
	}
}
void dfs1(int u, int p){
	vis[u] = 1; par[u] = p; subt[u] = 1;
	for(int i = 0; i < deg[u]; ++i){
		int v = adj[u][i];
		if(v == p) continue;
		if(!vis[v]){
			dfs1(v,u);
			subt[u] += subt[v];
		}
		else nxt[u] = v;
	}
}
int main(){
	int i,j,n,ans = 1;
	cin >> n;
	for(i = 1; i <= n; ++i){
		for(j = 1; j <= n; ++j){
			cin >> adj[i][j];
		}
		deg[i] = 0;
	}
	int tot = 0;
	for(i = 1; i <= n; ++i){
		for(j = i + 1; j <= n; ++j){
			if(adj[i][j]){
				tot++;
				deg[i]++;
				deg[j]++;
			}
		}
	}
	memset(vis,0,sizeof(vis));
	dfs2(1);
	int cnt = 0;
	for(i = 1; i <= n; ++i){
		if(!vis[i]) cnt++;
	}
	memset(vis,0,sizeof(vis));
	for(i = 1; i <= n; ++i){
		subt[i] = nxt[i] = 0;
	}
	dfs1(1,0);
	for(i = 1; i <= n; ++i){
		if(nxt[i]){
			ans = ans * 1LL * subt[nxt[i]] * tot % mod;
			ans %= mod;
			tot--;
		}
	}
	cout << ans << endl;
	return 0;
}
