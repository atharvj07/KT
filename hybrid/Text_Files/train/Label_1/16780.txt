#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ull unsigned long long
int main() {
  string a;
  cin >> a;
  ll m, mn = LLONG_MAX;
  cin >> m;
  int n = a.size();
  a += a;
  for (int i = 1; i < n; i++) {
    ll cur = 0;
    for (int j = i; j < n + i; j++) { cur = cur * 10 + a[j % n] - '0'; }
    mn = min(mn, cur % m);
  }
  cout << mn << endl;
  return 0;
}
