n,m=map(int,input().split())
a=[list(map(int,input().split())) for i in range(n)]
c=0
for i in range(n):
    for j in range(m):
        if a[i][j]==0:
            for x in range(j+1,m):
                if a[i][x]==1:c+=1
            for x in range(i+1,n):
                if a[x][j]==1:c+=1
print(c)
