n,h=map(int,input().split());a=list(map(int,input().split()))
mod=10**9+7;dp={tuple(a):1}
for i in range(n):
    for j in range(n):
        if i<=j:continue
        b=[x+1 for x in a]
        for k in range(i,j+1):b[k]=h
        t=tuple(b)
        if t not in dp:dp[t]=1
        else:dp[t]=(dp[t]+dp[tuple(a)])%mod
print(dp[tuple([h]*n)])
