import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long f = input.nextLong();
        long t = input.nextLong();
        long t0 = input.nextLong();
        long a1 = input.nextLong();
        long t1 = input.nextLong();
        long p1 = input.nextLong();
        long a2 = input.nextLong();
        long t2 = input.nextLong();
        long p2 = input.nextLong();
        long minCost = -1;
        for (long i = 0; i * t0 <= t; i++) {
            long rem = f - i * t0;
            if (rem % a1 == 0) {
                long cost = i * 0 + (rem / a1) * p1;
                if (minCost == -1 || cost < minCost) {
                    minCost = cost;
                }
            }
            if (rem % a2 == 0) {
                long cost = i * 0 + (rem / a2) * p2;
                if (minCost == -1 || cost < minCost) {
                    minCost = cost;
                }
            }
        }
        for (long j = 0; j * t1 <= t; j++) {
            for (long k = 0; k * t2 <= t; k++) {
                long rem = f - j * a1 - k * a2;
                if (rem <= 0) {
                    long cost = j * p1 + k * p2;
                    if (minCost == -1 || cost < minCost) {
                        minCost = cost;
                    }
                } else if (rem % t0 == 0) {
                    long cost = j * p1 + k * p2 + (rem / t0) * 0;
                    if (minCost == -1 || cost < minCost) {
                        minCost = cost;
                    }
                }
            }
        }
        System.out.println(minCost);
    }
}
