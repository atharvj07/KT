import sys
input=sys.stdin.readline

n=int(input())
x,y=[[],[]]
for i in range(n):
 a,b=map(int,input().split())
 x.append(a)
 y.append(b)
    
X0=sum(x)/n
Y0=sum(y)/n
   
mu_x=sum((xi-X0)**2 for xi in x)
mu_y=sum((yi-Y0)**2 for yi in y)
mu_xy=sum((xi-X0)*(yi-Y0) for xi,yi in zip(x,y))

ang=0
if mu_x==0 and mu_y==0: print(0)
elif mu_x+mu_y==0: print(90)
elif mu_x+mu_y>0: print(format(180*mu_xy/(mu_x+mu_y)**(1/2)/3.14159265358979323846,'.12f'))
else: print(format(180*mu_xy/(mu_x+mu_y)**(1/2)/3.14159265358979323846+180,'.12f'))
