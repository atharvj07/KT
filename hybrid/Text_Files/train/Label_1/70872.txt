t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    c = [0, 0, 0]
    for num in a:
        c[num % 3] += 1
    moves = 0
    avg = n // 3
    for _ in range(2): # Iterate twice to balance out the counts
        for i in range(3):
            if c[i] > avg:
                moves += c[i] - avg
                c[(i + 1) % 3] += c[i] - avg
                c[i] = avg
    print(moves)
