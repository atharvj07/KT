#include <bits/stdc++.h>
using namespace std;

int main() {
  int tt;
  cin >> tt;
  while (tt--) {
    int n, k;
    cin >> n >> k;
    vector<int64_t> h(n);
    for (int i = 0; i < n; i++) { cin >> h[i]; }
    int ans = 1;
    if (h[0] > 0 || h[n - 1] > 0) ans = 0;
    for (int i = 1; i < n; i++)
      if (h[i] > h[i - 1] + 1 || h[i] < h[i - 1] - k + 1) ans = 0;
    if (ans)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
