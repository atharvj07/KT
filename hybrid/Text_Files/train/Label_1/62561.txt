import heapq

def dijkstra(graph, start):
    distances = {vertex: float('infinity') for vertex in graph}
    distances[start] = 0
    priority_queue = [(0, start)]

    while priority_queue:
        current_distance, current_vertex = heapq.heappop(priority_queue)

        if current_distance > distances[current_vertex]:
            continue

        for neighbor, weight in graph[current_vertex].items():
            distance = current_distance + weight

            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(priority_queue, (distance, neighbor))
    
    return distances

def min_cost_to_attend_concert(n, m, routes, concert_prices):
    graph = {i: {} for i in range(1, n+1)}
    for v, u, w in routes:
        graph[v][u] = w
        graph[u][v] = w

    min_costs = []
    for i in range(1, n+1):
        distances = dijkstra(graph, i)
        min_cost = float('infinity')
        for j in range(1, n+1):
            if distances[j] != float('infinity'):
                cost = 2 * distances[j] + concert_prices[j-1]
                min_cost = min(min_cost, cost)
        min_costs.append(min_cost)

    return min_costs

# Reading input
n, m = map(int, input().split())
routes = [list(map(int, input().split())) for _ in range(m)]
concert_prices = list(map(int, input().split()))

# Calculating minimum costs
min_costs = min_cost_to_attend_concert(n, m, routes, concert_prices)

# Printing output
print(*min_costs)
