#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int g = 0, t = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'G') g = i;
        if (s[i] == 'T') t = i;
    }
    if (abs(g - t) % k == 0)
        cout << "YES" << endl;
    else
        cout << "NO" << endl;
    return 0;
}
