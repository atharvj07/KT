import sys
input = sys.stdin.readline

m = int(input())
essay = [x.lower() for x in input().split()]
n = int(input())
repl = {}
for _ in range(n):
    x, y = input().split()
    x = x.lower()
    y = y.lower()
    repl[x] = repl.get(x, [])+[y]
    repl[y] = repl.get(y, [])+[x]

ans = [essay[0]]
mnr = essay[0].count('r')
mnlen = len(essay[0])
for word in essay[1:]:
    word = word.lower()
    if word in repl:
        for syn in repl[word]:
            if syn.count('r') < mnr or (syn.count('r') == mnr and len(syn) < mnlen):
                ans = [syn]
                mnr = syn.count('r')
                mnlen = len(syn)
            elif syn.count('r') == mnr and len(syn) == mnlen:
                ans.append(syn)
    else:
        if word.count('r') < mnr or (word.count('r') == mnr and len(word) < mnlen):
            ans = [word]
            mnr = word.count('r')
            mnlen = len(word)
        elif word.count('r') == mnr and len(word) == mnlen:
            ans.append(word)
print(mnr, sum(map(len, ans)))
