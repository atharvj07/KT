#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, a[3];
    cin >> n, fill_n(a, 3, 0);
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      a[x % 3]++;
    }
    if (a[0] == a[1] && a[1] == a[2])
      cout << 0;
    else {
      int mn = *min_element(a, a + 3);
      cout << (a[0] + a[1] + a[2] - 3 * mn + 2) / 3;
    }
    cout << '\n';
  }
  return 0;
}
