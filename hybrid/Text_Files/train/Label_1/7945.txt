#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
vector<int> adj[N], order;
int dfs(int u, int p, int& c) {
  for (auto v : adj[u]) {
    if (v == p) continue;
    c += dfs(v, u, c);
  }
  if (!c) order.push_back(u);
  return c;
}
void solve() {
  int n;
  cin >> n;
  vector<int> par(n + 1), match(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> par[i] >> match[i];
    if (par[i] != -1) adj[par[i]].push_back(i);
  }
  int c = 0;
  dfs(1, 1, c);
  reverse(order.begin(), order.end());
  if (order.empty())
    cout << -1 << '\n';
  else
    for (auto x : order) cout << x << ' ';
}
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  solve();
  return 0;
}
