#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

typedef long long ll;

const ll INF = 1e9;

struct pt {
    ll x, y, z;
    ll id;
    pt() {}
    pt(ll x, ll y, ll z) : x(x), y(y), z(z) {}
    pt(ll x, ll y, ll z, ll id) : x(x), y(y), z(z), id(id) {}
    bool operator==(const pt& p) const { return id == p.id; }
};

struct line {
    pt p1, p2;
    line() {}
    line(pt p1, pt p2) : p1(p1), p2(p2) {}
};

pt operator-(const pt& p1, const pt& p2) {
    return pt(p1.x - p2.x, p1.y - p2.y, p1.z - p2.z);
}

ll dot(const pt& p1, const pt& p2) {
    return p1.x * p2.x + p1.y * p2.y + p1.z * p2.z;
}

ll cross(const pt& p1, const pt& p2) {
    return p1.x * p2.y - p1.y * p2.x;
}

ll area(const pt& p1, const pt& p2, const pt& p3) {
    return abs(cross(p2 - p1, p3 - p1));
}

bool onsegment(const pt& p1, const pt& p2, const pt& p3) {
    return (p2.x <= std::max(p1.x, p3.x) && p2.x >= std::min(p1.x, p3.x))
        && (p2.y <= std::max(p1.y, p3.y) && p2.y >= std::min(p1.y, p3.y))
        && (p2.z <= std::max(p1.z, p3.z) && p2.z >= std::min(p1.z, p3.z));
}

bool intersect(const line& l1, const line& l2) {
    ll d1, d2, d3, d4;
    d1 = cross(l1.p2 - l1.p1, l2.p1 - l1.p1);
    d2 = cross(l1.p2 - l1.p1, l2.p2 - l1.p1);
    d3 = cross(l2.p2 - l2.p1, l1.p1 - l2.p1);
    d4 = cross(l2.p2 - l2.p1, l1.p2 - l2.p1);
    if ((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)
        && (d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0)) {
        return true;
    }
    if (d1 == 0 && onsegment(l1.p1, l2.p1, l1.p2)) {
        return true;
    }
    if (d2 == 0 && onsegment(l1.p1, l2.p2, l1.p2)) {
        return true;
    }
    if (d3 == 0 && onsegment(l2.p1, l1.p1, l2.p2)) {
        return true;
    }
    if (d4 == 0 && onsegment(l2.p1, l1.p2, l2.p2)) {
        return true;
    }
    return false;
}

std::vector<line> getlines(const std::vector<pt>& polygon) {
    std::vector<line> lines;
    for (size_t i = 0; i < polygon.size(); ++i) {
        lines.emplace_back(polygon[i], polygon[(i + 1) % polygon.size()]);
    }
    return lines;
}

bool wellconnected(const std::vector<pt>& polygon1, const std::vector<pt>& polygon2) {
    std::vector<line> lines1 = getlines(polygon1);
    std::vector<line> lines2 = getlines(polygon2);
    int cnt1 = 0, cnt2 = 0;
    for (const line& l1 : lines1) {
        for (const line& l2 : lines2) {
            if (intersect(l1, l2)) {
                cnt1++;
            }
        }
    }
    for (const line& l1 : lines2) {
        for (const line& l2 : lines1) {
            if (intersect(l1, l2)) {
                cnt2++;
            }
        }
    }
    return cnt1 != cnt2;
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);
    int n, m;
    std::cin >> n;
    std::vector<pt> polygon1(n);
    for (pt& p : polygon1) {
        std::cin >> p.x >> p.y >> p.z;
    }
    std::cin >> m;
    std::vector<pt> polygon2(m);
    for (pt& p : polygon2) {
        std::cin >> p.x >> p.y >> p.z;
    }
    bool res = wellconnected(polygon1, polygon2);
    std::cout << (res ? "YES" : "NO") << '\n';
    return 0;
}
