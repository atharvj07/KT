MOD = 1000000007

def count_ways(n, h, a):
    dp = [[0 for _ in range(n + 1)] for _ in range(n + 1)]
    dp[0][0] = 1

    for i in range(1, n + 1):
        for j in range(n + 1):
            # Case 1: a[i-1] == h - 1
            if a[i - 1] == h - 1:
                # 1.1: Continue the previous segment
                if j > 0:
                    dp[i][j] += dp[i - 1][j - 1] * j
                # 1.2: Start a new segment or don't use the i-th element
                dp[i][j] += dp[i - 1][j]
            # Case 2: a[i-1] == h
            elif a[i - 1] == h:
                # 2.1: Continue the previous segment
                if j > 0:
                    dp[i][j] += dp[i - 1][j - 1] * j
                # 2.2: Close the previous segment
                dp[i][j] += dp[i - 1][j + 1] * (j + 1)
                # 2.3: Don't use the i-th element
                dp[i][j] += dp[i - 1][j]
            dp[i][j] %= MOD

    return dp[n][0]

n, h = map(int, input().strip().split())
a = list(map(int, input().strip().split()))

print(count_ways(n, h, a))
