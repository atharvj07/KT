#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<int> bot, left;
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            bot.push_back(x);
        }
        for (int i = 0; i < m; i++) {
            int x;
            cin >> x;
            left.push_back(x);
        }
        int ans = 0;
        int i = 0, j = 0;
        while (i < n && j < m) {
            if (bot[i] == left[j]) {
                ans++;
                i++;
                j++;
            } else if (bot[i] < left[j]) {
                i++;
            } else {
                j++;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
