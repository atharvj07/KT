#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, b, d, cnt = 0, yr = 0, ans = 0;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    cin >> b >> d;
    v[i] = {b, d};
  }
  sort(v.begin(), v.end());
  for (auto p : v) {
    if (p.first <= yr)
      cnt++;
    else {
      if (cnt > ans) {
        ans = cnt;
        yr = p.first;
      }
      cnt = 1;
      yr = p.first;
    }
  }
  if (cnt > ans) {
    ans = cnt;
    yr = v.back().first;
  }
  cout << yr << " " << ans << endl;
  return 0;
}
