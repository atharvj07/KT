import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        boolean[] happy = new boolean[m];
        Map<String, Integer> handles = new HashMap<>();
        int current = 1;
        for (int i = 0; i < n; i++) {
            int type = sc.nextInt();
            if (type == 1) {
                current++;
            } else {
                String handle = sc.next();
                if (!handles.containsKey(handle)) {
                    handles.put(handle, current);
                }
                if (handles.get(handle) == current) {
                    happy[type - 2] = true;
                }
            }
        }
        int count = 0;
        for (boolean b : happy) {
            if (b) {
                count++;
            }
        }
        System.out.println(count);
    }
}
