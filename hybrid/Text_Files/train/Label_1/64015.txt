MOD = 10**9 + 7

def count_good_numbers(n, k, a, b):
    total_good_numbers = 1
    range_of_numbers = 10**k
    
    for i in range(n // k):
        count_divisible = range_of_numbers // a[i]
        start_digit = b[i] * (10**(k-1))
        end_digit = (b[i]+1) * (10**(k-1))
        
        count_divisible_with_b = (max(0, end_digit - start_digit) + a[i] - 1) // a[i]
        
        good_numbers_for_block = (count_divisible - count_divisible_with_b) % MOD
        total_good_numbers = (total_good_numbers * good_numbers_for_block) % MOD
    
    return total_good_numbers

n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

print(count_good_numbers(n, k, a, b))
