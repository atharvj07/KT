#include <bits/stdc++.h>
using namespace std;
const int mod = 1000000007;
const int N = 100005;
const int64_t inf = 1e18;
int n, m, last[N], ans, pre[N];
vector<int> adj[N];
vector<int64_t> d;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> n;
  d.resize(n + 1, inf);
  for (int i = 1; i <= n; i++) { cin >> pre[i]; }
  cin >> m;
  for (int i = 1; i <= m; i++) {
    int u, v;
    cin >> u >> v;
    adj[v].push_back(u);
  }
  d[1] = 0;
  queue<int> q;
  q.push(1);
  last[1] = 1;
  while (q.size()) {
    int v = q.front();
    q.pop();
    for (auto u : adj[v]) {
      if (d[v] + pre[v] < d[u]) {
        d[u] = d[v] + pre[v];
        last[u] = v;
        q.push(u);
      } else if (d[v] + pre[v] == d[u] && last[v] != u) { ans++; }
    }
  }
  cout << d[n] << " " << ans + 1 << "\n";
}
