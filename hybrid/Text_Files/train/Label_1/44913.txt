import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int r = in.nextInt();
        int[][] ducks = new int[n][2];
        for (int i = 0; i < n; i++) {
            ducks[i][0] = in.nextInt();
            ducks[i][1] = in.nextInt();
        }
        Arrays.sort(ducks, (a, b) -> a[1] - b[1]);
        int ans = 0, last = 0;
        for (int[] duck : ducks) {
            if (duck[0] >= last) {
                ans++;
                last = duck[1] + r;
            }
        }
        System.out.println(ans);
    }
}
