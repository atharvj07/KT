#include<bits/stdc++.h>
using namespace std;

int dp[105][105][105];

string lcs(string a, string b, string c, int n, int m, int o) {
	for (int i = n - 1; i >= 0; i--) {
		for (int j = m - 1; j >= 0; j--) {
			for (int k = o - 1; k >= 0; k--) {
				if (a[i] == b[j] && b[j] == c[k]) {
					dp[i][j][k] = 1 + dp[i + 1][j + 1][k + 1];
				}
				else {
					int x, y, z;

					x = dp[i][j][k + 1];
					y = dp[i][j + 1][k];
					z = dp[i + 1][j][k];

					dp[i][j][k] = max({x, y, z});
				}
			}
		}
	}

	string res;
	int i = 0, j = 0, k = 0;

	while (i < n && j < m && k < o) {
		if (a[i] == b[j] && b[j] == c[k]) {
			res.push_back(a[i]);
			i++;
			j++;
			k++;
		}
		else {
			int x, y, z;

			x = dp[i][j][k + 1];
			y = dp[i][j + 1][k];
			z = dp[i + 1][j][k];

			if (x >= y && x >= z) {
				k++;
			}
			else if (y >= x && y >= z) {
				j++;
			}
			else {
				i++;
			}
		}
	}
	if (res.size() == 0) {
		return "0";
	}
	return res;
}

int main() {
	string a, b, c;

	cin >> a >> b >> c;

	int n = a.size();
	int m = b.size();
	int o = c.size();

	cout << lcs(a, b, c, n, m, o) << endl;
}
