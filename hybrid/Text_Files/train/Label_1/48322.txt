import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            int[] a = new int[k];
            for (int j = 0; j < k; j++) {
                a[j] = sc.nextInt();
            }
            Arrays.sort(a);
            if (i > 0) {
                ans[i] = ans[i - 1];
            }
            for (int j = 0; j < k; j++) {
                if (a[j] == a[k - 1]) {
                    ans[i]++;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.println(ans[i]);
        }
    }
}
