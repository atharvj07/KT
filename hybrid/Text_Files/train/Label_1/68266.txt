n,m,k,s=map(int,input().split())
a=[list(map(int,input().split())) for i in range(n)]
q=list(map(int,input().split()))
ans=0
for i in range(s):
    for j in range(m):
        if(a[0][j]==q[i]):
            ans=max(ans,j)
for i in range(s-1):
    for j in range(n):
        for l in range(m):
            if(a[j][l]==q[i] and a[j][l]==q[i+1]):
                ans=max(ans,abs(j-1)+abs(l-ans))
                
print(ans)
