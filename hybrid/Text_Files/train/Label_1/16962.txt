#include <cstdio>
#include <cstring>
#include <iostream>
using namespace std;
int main()
{
    int n;
    while(cin>>n)
    {
        int k = 0, d[7] = {0};
        for(int i = 0; i < 7; i++)
        {
            d[i] = n%2;
            n /= 2;
            k += d[i];
        }
        if(k == 0 || k == 1)
        {
            cout<<100<<endl;
            continue;
        }
        int ans = 0;
        if(d[0] == 0) ans+=4;
        if(d[1] == 0) ans+=2;
        if(d[4] == 0) ans+=4;
        if(d[5] == 0) ans+=2;
        if(d[1] == 0 && d[4] == 0) ans-=2;
        cout<<ans<<endl;
    }
    return 0;
}
