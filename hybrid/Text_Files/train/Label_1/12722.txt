#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    partial_sum(a.begin(), a.end(), a.begin());
    int half = a[n - 1] / 2;
    for (int i = 0; i < n; i++) {
        if (a[i] >= half) {
            cout << i + 1 << endl;
            return 0;
        }
    }
    return 0;
}
