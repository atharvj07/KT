#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 3e5+1;
ll n,x[maxn],y[maxn],ans[maxn];
#define dbg(x) cout << #x << " = " << x << endl
int main() {
    cin >> n;
    for(int i  =0 ; i < n ; i++){
        cin >> x[i] >> y[i];
    }
    ll sumx = 0, sumy = 0;
    for(int i = 0 ; i < n ; i++){
        sumx += abs(x[i] - x[(i+1)%n]);
        sumy += abs(y[i] - y[(i+1)%n]);
    }
    // dbg(sumx);dbg(sumy);
    for(int k = 3 ; k <= n ; k++){
        // dbg(sumx);dbg(sumy);
        ans[k] = ans[k-1] + sumx + sumy;
        sumx -= abs(x[k-1] - x[(k-2)%n]);
        sumx -= abs(x[k-1] - x[(k)%n]);
        sumx += abs(x[(k-2)%n] - x[(k)%n]);
        sumy -= abs(y[k-1] - y[(k-2)%n]);
        sumy -= abs(y[k-1] - y[(k)%n]);
        sumy += abs(y[(k-2)%n] - y[(k)%n]);
    }
    // for(int i = 1 ; i <= n ; i++){
    //     cout << ans[i] << ' ';
    // }
    // cout << endl;
    for(int i = 3 ; i <= n ; i++){
        cout << ans[i] << ' ';
    }
    cout << endl;
    return 0;
}
