n, m, mod = map(int, input().split())
a, b = [[0] * (n + 1) for i in range(m + 1)], [[0] * m for i in range(n + 1)]
for i in range(1, m + 1):
    a[i][1:] = list(input())
for i in range(1, m + 1):
    for j in range(1, n + 1):
        b[j][i - 1] += int(a[i][j])
for j in range(1, n + 1):
    if b[j].count(2) != m:
        print(0)
        exit()
c = [0] * (m + 1)
for i in range(1, n + 1):
    for j in range(1, m + 1):
        a[j][i] = int(a[j][i])
        c[j] += a[j][i]
for j in range(1, m + 1):
    if c[j] != 2:
        print(0)
        exit()
ans = 1
for i in range(1, n + 1):
    for j in range(1, m + 1):
        if a[j][i] == 0:
            ans = ans * (j - 1) % mod
            ans = ans * (n - i) % mod
print(ans)
