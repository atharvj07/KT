import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int h = in.nextInt();
        System.out.println(solve(n, h));
    }

    private static long solve(int n, int h) {
        long[][] dp = new long[n + 1][h + 1];
        // Initialize base case
        dp[0][0] = 1;
        // Iterate through the number of nodes
        for (int i = 1; i <= n; i++) {
            // Iterate through the height of the tree
            for (int j = 0; j <= h; j++) {
                // If the height is less than the maximum height of the tree, we can add a left child
                if (j < h) {
                    dp[i][j] += dp[i - 1][j + 1];
                }
                // If the height is greater than 0, we can add a right child
                if (j > 0) {
                    dp[i][j] += dp[i - 1][j - 1];
                }
            }
        }
        // Return the number of ways to construct a tree with n nodes and height h
        return dp[n][h];
    }
}
