#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

#pragma GCC optimize "Ofast"

typedef long long ll;
constexpr int MAXN = 3e5, INF = 1e9;
struct song {
    ll t, b;
} a[MAXN];
struct cmp {
    bool operator()(const song& a, const song& b) const {
        return a.b > b.b;
    }
};
int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 0; i < n; ++i) cin >> a[i].t >> a[i].b;
    sort(a, a + n, cmp());
    int l = 0, r = k - 1;
    ll sum = 0;
    while (l < k) sum += a[l++].t;
    ll ans = sum * a[k - 1].b;
    while (r < n - 1) {
        sum -= a[l - 1].t;
        sum += a[++r].t;
        ans = max(ans, sum * a[r].b);
    }
    cout << ans << '\n';
    return 0;
}
