def calculate_candies():
    input_data = input().split()
    x, period = int(input_data[0]), input_data[2]

    if period == "week":
        # In a leap year, there are 52 weeks and one or two extra days.
        # If the day is Monday (1) or Sunday (7), there are 53 occurrences because 2016 is a leap year.
        if x == 6 or x == 7:
            return 53
        else:
            return 52
    else:  # period == "month"
        # All months have at least 28 days.
        # February in a leap year has 29 days.
        # April, June, September, and November have 30 days.
        # January, March, May, July, August, October, and December have 31 days.
        if x > 29:
            return 11 if x == 30 else 7
        else:
            return 12

print(calculate_candies())
