#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef vector<int> vi;
typedef vector<ll> vll;
#define eb emplace_back
inline ll isprime(ll n) {
  if (n <= 1) return 0;
  for (int i = 2; i * i <= n; ++i)
    if (n % i == 0) return 0;
  return 1;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, tmp;
  cin >> n;
  vi a(n + 1), pos(n + 1);
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    pos[a[i]] = i;
  }
  ll ans = 0;
  for (int i = 1; i <= n; ++i) {
    if (pos[i] == i) continue;
    if (isprime(pos[i] - i)) {
      cout << i << ' ' << pos[i] << '\n';
      swap(a[i], a[pos[i]]), swap(pos[a[i]], pos[a[pos[i]]]);
      ans++;
    } else {
      int j = i + 1;
      while (j <= n && !isprime(j - i)) ++j;
      if (j == n + 1) {
        for (j = i; j <= n; ++j)
          if (isprime(j - i)) break;
        cout << i << ' ' << j << '\n';
        swap(a[i], a[j]), swap(pos[a[i]], pos[a[j]]);
        ans++;
      } else {
        cout << i << ' ' << j << '\n';
        cout << j << ' ' << pos[i] << '\n';
        swap(a[i], a[j]), swap(pos[a[i]], pos[a[pos[i]]]);
        swap(a[j], a[pos[i]]), swap(pos[a[j]], pos[a[pos[j]]]);
        ans += 2;
      }
    }
  }
  cout << ans << '\n';
}
