#include<bits/stdc++.h>
using namespace std;
long long mod(long long x){
   if(x<0)x*=-1;
   return x;
}
bool comp(pair<long long,long long>p1,pair<long long,long long>p2){
   return p1.first>p2.first;
}
int main() {
   long long t,n,s,a[200005],b[200005],i,j,k,l,ans,tot,tot1;
   cin>>t;
   for(l=1;l<=t;l++){
      cin>>n>>s;
      for(i=0;i<n;i++){
         cin>>a[i]>>b[i];
      }
      vector<pair<long long,long long>>vec;
      for(i=0;i<n;i++){
         vec.push_back({a[i],b[i]});
      }
      sort(vec.begin(),vec.end(),comp);
      long long lol=0;
      for(i=0;i<n;i++){
         if(s>=vec[i].first){
            lol++;
            s-=vec[i].first;
         }
         else
            break;
      }
      ans=vec[(lol/2)].second;
      cout<<ans<<endl;
   }
   return 0;
}
