import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int q = input.nextInt();
        for (int i = 0; i < q; i++) {
            int xp0 = input.nextInt(), yp0 = input.nextInt(), xp1 = input.nextInt(), yp1 = input.nextInt();
            int xp2 = input.nextInt(), yp2 = input.nextInt(), xp3 = input.nextInt(), yp3 = input.nextInt();
            long s1 = (long) (yp1 - yp0) * (xp0 - xp2) - (long) (yp0 - yp2) * (xp1 - xp0);
            long s2 = (long) (yp3 - yp2) * (xp1 - xp2) - (long) (yp2 - yp0) * (xp3 - xp2);
            System.out.println((s1 * s2 <= 0) ? "1" : "0");
        }
    }
}
