#include <bits/stdc++.h>
using namespace std;

int a[24];
int b[7];
int main() {
    for (int i = 0; i < 24; i++) { cin >> a[i]; }
    for (int i = 1; i <= 6; i++) {
        for (int j = 0; j < 4; j++) {
            b[i] |= 1 << a[(i - 1) * 4 + j];
        }
    }
    int t = 0;
    for (int i = 1; i <= 6; i++) { t |= b[i]; }
    cout << ((t == (1 << 7) - 1) ? "YES" : "NO") << endl;
    return 0;
}
