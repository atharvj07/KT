def run_program(sequence):
    counts = [0] * 10
    sequence = list(sequence)
    cp = 0
    dp = 1

    while 0 <= cp < len(sequence):
        if sequence[cp] in "<>":
            old_cp = cp
            dp = -1 if sequence[cp] == "<" else 1
            cp += dp
            if 0 <= cp < len(sequence) and sequence[cp] in "<>":
                sequence.pop(old_cp)
                cp -= 1 if dp < 0 else 0
        else:
            digit = int(sequence[cp])
            counts[digit] += 1
            if digit == 0:
                sequence.pop(cp)
                cp -= 1 if dp < 0 else 0
            else:
                sequence[cp] = str(digit - 1)
            cp += dp

    return counts

n, q = map(int, input().split())
s = input()

for _ in range(q):
    l, r = map(int, input().split())
    subsequence = s[l-1:r]
    result = run_program(subsequence)
    print(*result)
