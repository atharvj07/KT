from collections import defaultdict
n=int(input())
a=list(map(int,input().split()))
d=defaultdict(int)
for v in a:d[v]+=1
ans=0
l=0
cnt=0
for r in range(n):
    if d[a[r]]==1:cnt+=1
    d[a[r]]-=1
    while cnt>1:
        d[a[l]]+=1
        if d[a[l]]==1:cnt-=1
        l+=1
    ans=max(ans,r-l+1)
print(ans)
