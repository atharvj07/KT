import java.util.Arrays;
import java.util.Scanner;

public class MultisetOperations {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int q = scanner.nextInt();

        int[] multiset = new int[n];
        for (int i = 0; i < n; i++) {
            multiset[i] = scanner.nextInt();
        }

        Arrays.sort(multiset);

        for (int i = 0; i < q; i++) {
            int k = scanner.nextInt();
            if (k > 0) {
                add(multiset, k);
            } else {
                remove(multiset, -k);
            }
        }

        if (multiset.length == 0) {
            System.out.println(0);
        } else {
            System.out.println(multiset[0]);
        }
    }

    static void add(int[] multiset, int k) {
        int index = Arrays.binarySearch(multiset, k);
        if (index < 0) {
            index = -index - 1;
        }
        multiset = Arrays.copyOf(multiset, multiset.length + 1);
        System.arraycopy(multiset, index, multiset, index + 1, multiset.length - index - 1);
        multiset[index] = k;
    }

    static void remove(int[] multiset, int k) {
        int index = Arrays.binarySearch(multiset, k);
        if (index >= 0) {
            multiset = Arrays.copyOf(multiset, multiset.length - 1);
            System.arraycopy(multiset, index + 1, multiset, index, multiset.length - index - 1);
        }
    }
}
