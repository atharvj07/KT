#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  int wmax = 0, hmax = 0;
  for (int i = 0; i < n; i++) {
    int w, h;
    cin >> w >> h;
    v[i] = {w, h};
    wmax += w;
    hmax = max(hmax, h);
  }
  for (int i = 0; i < n; i++) {
    cout << wmax - v[i].first + (v[i].second == hmax ? 0 : v[i].second) << " ";
  }
  cout << endl;
  return 0;
}
