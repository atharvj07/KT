#include <iostream>

using namespace std;

int main() {
  int a, b, c, d;
  cin >> a >> b >> c >> d;
  if (abs(a + b - c - d) > 1)
    cout << "NO";
  else {
    cout << "YES\n";
    for (int i = 0; i < a; i++) cout << 0 << ' ';
    for (int i = 0; i < b; i++) cout << 1 << ' ';
    for (int i = 0; i < c; i++) cout << 2 << ' ';
    for (int i = 0; i < d; i++) cout << 3 << ' ';
    cout << '\n';
  }
  return 0;
}
