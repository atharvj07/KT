def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)

n, a, b, k = map(int, input().split())
grid = input()
hits = []
for i in range(n):
    if grid[i] == '1':
        hits.append(i + 1)
if a == 1:
    print(n - k)
    for i in range(1, n + 1):
        if i not in hits:
            print(i, end=' ')
else:
    print(a)
    for i in range(1, n + 1):
        if i not in hits and (i - 1) % b not in hits:
            print(i, end=' ')
