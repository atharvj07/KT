import sys

n = int(input())
x0, y0 = map(int, input().split())
black = []
for _ in range(n):
    t, x, y = input().split()
    black.append((t, int(x), int(y)))

def is_check(x, y):
    for t, bx, by in black:
        if t == 'R' and (x == bx or y == by):
            return True
        if t == 'B' and abs(x - bx) == abs(y - by):
            return True
        if t == 'Q' and ((x == bx) or (y == by) or (abs(x - bx) == abs(y - by))):
            return True
    return False

print("YES" if is_check(x0, y0) else "NO")
