#include <bits/stdc++.h>
using namespace std;
int a[200005];
int main()
{
    int q; cin >> q;
    while(q--)
    {
        int n, k; cin >> n >> k;
        string s; cin >> s;
        int ans = 0;
        for(int i = 0; i < n; i++) a[i % k] += (s[i] - 'R' + 3) % 3 != (i / k + s[i] - 'R' + 3) % 3;
        for(int i = 0; i < k; i++) ans += min(a[i], 3 - a[i]);
        cout << ans << '\n';
        for(int i = 0; i < k; i++) a[i] = 0;
    }
}
