for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    for i in range(n - 1):
        if a[i] > a[i + 1]:
            for j in range(i, -1, -1):
                if a[j] >= a[i + 1]:
                    a[j], a[i + 1] = a[i + 1], a[j]
                    break
    print(*a)
