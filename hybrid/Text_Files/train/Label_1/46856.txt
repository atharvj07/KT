import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long a = scanner.nextLong();
        long b = scanner.nextLong();
        long p = scanner.nextLong();
        long x = scanner.nextLong();

        long[] pows = new long[(int) (Math.log(x) / Math.log(a)) + 1];
        pows[0] = 1;
        long invA = pow(a, p - 2, p);
        for (int i = 1; i < pows.length; i++) {
            pows[i] = pows[i - 1] * a % p;
        }

        long ans = 0;
        for (int i = 0; i < pows.length && pows[i] <= x; i++) {
            long pow = pow(pows[i], x - pows[i] + 1, p);
            long cur = (b * invA % p * pow % p) % p;
            if (cur == 0) {
                ans++;
            }
        }
        System.out.println(ans);
    }

    static long pow(long a, long b, long mod) {
        long res = 1;
        while (b > 0) {
            if ((b & 1) == 1) {
                res = res * a % mod;
            }
            b >>= 1;
            a = a * a % mod;
        }
        return res;
    }
}
