#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll M = 1e6;
ll n,m,k,s[M],c[M],dp[M];

ll ceil(ll x){
    if(x%k==0) return x/k;
    else return x/k+1;
}

int main(){
    cin>>n>>m>>k;
    for(int i=1;i<=m;++i) cin>>s[i];
    for(int i=1;i<=k;++i) cin>>c[i];

    for(int i=n;i>=0;--i){
        dp[i]=1e18;
        if(i==n) dp[i]=0;
        else dp[i]=dp[i+1];

        int id=lower_bound(s+1,s+m+1,i+k)-s-1;
        if(i+k-s[id]<=0) continue;
        dp[i]=min(dp[i],ceil(s[id]-i-k+1)*c[k]+dp[s[id]]);
    }

    if(dp[0]==1e18) cout<<-1<<endl;
    else cout<<dp[0]<<endl;

}
