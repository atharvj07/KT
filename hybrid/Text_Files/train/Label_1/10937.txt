#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.size();
        int ab = 0, ba = 0;
        for (int i = 0; i < n - 1; i++) {
            if (s[i] == 'a' && s[i + 1] == 'b') ab++;
            if (s[i] == 'b' && s[i + 1] == 'a') ba++;
        }
        int ans = abs(ab - ba);
        ans = (ans + 1) / 2;
        int cnt = 0;
        if (ab > ba) {
            for (int i = 0; i < n - 1; i++) {
                if (s[i] == 'a' && s[i + 1] == 'b') {
                    if (cnt == ans) {
                        s[i] = 'b';
                        break;
                    }
                    cnt++;
                }
            }
        } else if (ba > ab) {
            for (int i = 0; i < n - 1; i++) {
                if (s[i] == 'b' && s[i + 1] == 'a') {
                    if (cnt == ans) {
                        s[i] = 'a';
                        break;
                    }
                    cnt++;
                }
            }
        }
        cout << s << endl;
    }
}
