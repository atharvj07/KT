#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<string> a(n);
  for (string &x : a) { cin >> x; }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int p = 0, c = 0;
    for (int j = 0; j < a[i].size(); j++) { p += a[i][j] == 's'; }
    for (int j = 0; j < a[i].size(); j += 2) { c += a[i][j] == 's' && a[i][j + 1] == 'h'; }
    ans += c;
    for (int j = 0; j + i + 1 < n; j++) {
      c += a[j + i + 1].back() == 'h' && p;
      c -= a[j + i + 1][0] == 's' && a[i][0] == 'h';
      p -= a[j + i + 1][0] == 's';
      p += a[j + i + 1].back() == 's';
      ans = max(ans, c);
    }
  }
  cout << ans;
}
