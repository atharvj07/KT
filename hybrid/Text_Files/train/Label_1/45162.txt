import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String n = scanner.next();
        System.out.println(String.format("%s\n%s", findMaxZeros(n), findMaxZeros(n)));
    }

    public static String findMaxZeros(String n) {
        int[] count = new int[10];
        for (char c : n.toCharArray()) {
            count[c - '0']++;
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 9; i >= 0; i--) {
            while (count[i] > 0) {
                sb.append(i);
                count[i]--;
            }
        }
        return sb.toString();
    }
}
