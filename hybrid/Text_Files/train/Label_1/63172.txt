def modInverse(b, m):
    g = gcd(b, m)
    if (g != 1):
        return -1
    else:
        return pow(b, m - 2, m)

def gcd(a, b):
    if (a == 0):
        return b
    return gcd(b % a, a)

def expectedDays(n, a):
    MOD = 1000000007
    active_startups = sum([1 for i in a if i == -1])
    if active_startups == 1:
        return 0
    result = ((pow(2, active_startups, MOD) - 2) * modInverse(active_startups * (active_startups - 1) // 2, MOD)) % MOD
    return result

n = int(input().strip())
a = list(map(int, input().strip().split()))
print(expectedDays(n, a))
