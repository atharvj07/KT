t = int(input())
for _ in range(t):
    n = int(input())
    monsters = list(map(int, input().split()))
    m = int(input())
    heroes = [list(map(int, input().split())) for _ in range(m)]
    
    hero_dict = {}
    for p, s in heroes:
        if p in hero_dict:
            hero_dict[p] = max(hero_dict[p], s)
        else:
            hero_dict[p] = s
    
    max_s = 0
    for p in sorted(hero_dict.keys(), reverse=True):
        max_s = max(max_s, hero_dict[p])
        hero_dict[p] = max_s
    
    days = 0
    i = 0
    while i < n:
        max_power = monsters[i]
        s = 0
        while i + s < n and s < hero_dict.get(max_power, 0):
            s += 1
            if i + s < n:
                max_power = max(max_power, monsters[i+s])
                if max_power not in hero_dict or s >= hero_dict[max_power]:
                    break
        if s == 0:
            days = -1
            break
        days += 1
        i += s
    
    print(days)
