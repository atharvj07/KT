import sys

R, B = map(int, input().split())
sp_x = [int(x) for x in input().split()]
sp_y = [int(y) for y in input().split()]
bs_x = [int(x) for x in input().split()]
bs_y = [int(y) for y in input().split()]

sp = [(x1, y1) for x1, y1 in zip(sp_x, sp_y)]
bs = [(x2, y2) for x2, y2 in zip(bs_x, bs_y)]

def cross(o, a, b):
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

def is_intersect(a, b, c, d):
    return cross(a, b, c) * cross(a, b, d) <= 0 and cross(c, d, a) * cross(c, d, b) <= 0

ok = True

for i in range(R):
    for j in range(B):
        for ii in range(i + 1, R):
            for jj in range(j + 1, B):
                if is_intersect(sp[i], bs[j], sp[ii], bs[jj]):
                    ok = False

print('Yes' if ok else 'No')
