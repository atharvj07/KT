#include <iostream>
#include <string>
#include <vector>
using namespace std;
 
int main() {
  string s;
  cin >> s;
  while (true) {
    bool f1 = false, f2 = false;
    for (int i = 0; i < s.size() - 1; i++) {
      if (s[i] == 'y' && s[i + 1] == 'x') {
        swap(s[i], s[i + 1]);
        f1 = true;
        break;
      } else if (s[i] == 'x' && s[i + 1] == 'y') {
        s.erase(s.begin() + i, s.begin() + i + 2);
        f2 = true;
        break;
      }
    }
    if (!f1 && !f2) {
      cout << s << endl;
      return 0;
    }
  }
  return 0;
}
