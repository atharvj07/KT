import java.util.Arrays;
import java.util.Scanner;

public class Jigsaw {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int h = scanner.nextInt();
        int[][] pieces = new int[n][4];
        for (int i = 0; i < n; i++) {
            pieces[i][0] = scanner.nextInt(); // A_i
            pieces[i][1] = scanner.nextInt(); // B_i
            pieces[i][2] = scanner.nextInt(); // C_i
            pieces[i][3] = scanner.nextInt(); // D_i
        }

        // Sort pieces by height in descending order
        Arrays.sort(pieces, (a, b) -> Integer.compare(b[0] + b[1], a[0] + a[1]));

        // Check if arrangement is possible
        boolean possible = true;
        int currentHeight = 0;
        for (int[] piece : pieces) {
            int a = piece[0], b = piece[1], c = piece[2], d = piece[3];
            if (currentHeight + a + b > h) {
                possible = false;
                break;
            }
            currentHeight += a + b;
            if (c > 0) {
                currentHeight -= c;
            }
            if (d > 0) {
                currentHeight -= d;
            }
        }

        // Print result
        System.out.println(possible ? "YES" : "NO");
    }
}
