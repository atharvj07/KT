#include<bits/stdc++.h>
using namespace std;
int main() {
  int x;
  cin >> x;
  vector<int> v;
  while (x) {
    v.push_back(x % 10);
    x /= 10;
  }
  sort(v.rbegin(), v.rend());
  int ans = 0;
  for (int i = 0; i < v.size(); i++) {
    ans *= 10;
    ans += v[i];
  }
  cout << ans << endl;
  return 0;
}
