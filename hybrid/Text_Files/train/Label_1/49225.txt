import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int w = in.nextInt();
        int b = in.nextInt();
        System.out.printf("%.9f", (w + b) / Math.pow(w + b + 1, b) + (w * Math.pow(w + b, b) / Math.pow(w + b + 1, b + 1)));
    }
}
