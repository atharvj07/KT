import math

n = int(input())
a = list(map(int, input().split()))

if sum(a) == 0:
    print(-1)
    exit()

g = 0
for i in a:
    g = math.gcd(g, i)

if g != 1:
    print(0)
    exit()

ans = 0
for i in range(1, n):
    if a[i] & 1:
        a[i - 1] += 1
        ans += 1
    a[i] //= 2

print(ans)
