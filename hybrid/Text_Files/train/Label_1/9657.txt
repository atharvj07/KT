#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=3e5+1,mod=1e9+7,inf=2e9;
int main(){
	int n;
	cin>>n;
	string s;
	cin>>s;
	int l=-1;
	for(int i=0;i<n;i++){
		if(i>0&&s[i]<s[i-1])l=i-1;
	}
	if(l==-1){
		cout<<"NO";
		return 0;
	}
	cout<<"YES\n";
	for(int i=l;i>=0&&s[i]==s[l];i--)l=i;
	l++;
	int r=l;
	for(int i=l+1;i<n;i++){
		if(s[i]<s[i-1])r=i-1;
	}
	cout<<l+1<<" "<<r+1<<endl;
	return 0;
}
