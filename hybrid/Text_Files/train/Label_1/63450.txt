from collections import defaultdict

def dfs(graph, u, a, dist):
    if dist > a[u]:
        return 0
    leaves_removed = 0
    is_leaf = True
    for v, w in graph[u]:
        # Process child, update maximum distance from root to current node
        leaves_removed += dfs(graph, v, a, max(dist + w, 0))
        is_leaf = False
    return leaves_removed if not is_leaf else 0

def solve(n, a, edges):
    graph = defaultdict(list)
    for i, (p, c) in enumerate(edges, 2):
        graph[p].append((i, c))
    return dfs(graph, 1, a, 0)

n = int(input().strip())
a = [0] + list(map(int, input().strip().split()))
edges = [tuple(map(int, input().strip().split())) for _ in range(n - 1)]

print(solve(n, a, edges))
