#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 1e9;
ll solve(int a, int b, int c, int x) {
    int sum = 0, p = x;
    while (x) {
   	    sum += x % 10; x /= 10;
    }
    if (sum * b + c == p) return p;
    if (sum * b + c > p || p > maxn) return -1;
    return solve(a, b, c, p * 10 + a);
}
int main() {
    ll a, b, c; cin >> a >> b >> c;
    int sum = 0, p = 0;
    for (int i = 1; i < 10; i++) {
        p = solve(a, b, c, i);
        if (p > 0) {
   	        sum++;
   	        cout << p << " ";
        }
    }
    if (sum == 0) cout << "0";
    cout << "\n";
}
