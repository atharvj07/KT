#include <iostream>
using namespace std;

int main() {
    int n, m, c, ans = 0, h[1001] = {};
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        cin >> c;
        h[c]++;
        if (h[c] == 1) ans++;
        if (h[1] == h[2] && h[2] == h[3]) {
            for (int i = 1; i <= n; i++) h[i]--;
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
