mod=int(1e9)+7
def pp(a,b):
    p=1
    while b>0:
        if b%2:p=(p*a)%mod
        b//=2
        a=(a*a)%mod
    return p
n,k=map(int,input().split())
a=[int(i)for i in input().split()]
if n==1:print(1)
else:
    a=[i==0 for i in a]
    pr=1;x=a[0]
    for i in a[1:]:
        if x>i:break
        x=i
        pr=(pr*2)%mod
    if pr==1:print(0)
    else:print(pp(pr,k-1))
