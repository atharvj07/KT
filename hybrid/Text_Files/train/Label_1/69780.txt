import sys

n, m = map(int, input().split())

a = [[c for c in input()] for _ in range(n)]
b = [[c for c in input()] for _ in range(m)]

a_trans = [[b[j][i] for j in range(m)] for i in range(n)]

for i in range(n - m + 1):
    for j in range(n - m + 1):
        if a[i][j:j + m] == a_trans[j][i:i + m]:
            print(i + 1, j + 1)
