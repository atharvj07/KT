import bisect

n = int(input())
schedule = []

for _ in range(n):
    s, d = map(int, input().split())
    
    idx = bisect.bisect_left(schedule, s)
    
    if idx < len(schedule) and schedule[idx] <= s + d - 1:
        idx = bisect.bisect_right(schedule, s + d - 1)
    
    schedule.insert(idx, s + d - 1)
    print(s, s + d - 1)
