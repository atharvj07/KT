#include <bits/stdc++.h>
 
using namespace std;
 
int main() {
  int n;
  cin >> n;
 
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
 
  int ans = 0;
  for (int i = 0; i < n; i++) ans |= a[i];
  if (ans != 0) {
    cout << 1 << endl << ans << endl;
  } else {
    cout << 2 << endl << a[n - 1] << " " << a[n - 2] << endl;
  }
 
  return 0;
}
