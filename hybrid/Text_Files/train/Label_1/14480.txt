#include <vector>
#include <iostream>
#include <cmath>
using namespace std;
int main() {
    int t; cin >> t;
    while (t--) {
        int n, k, l; cin >> n >> k >> l;
        vector<int> d(n);
        for (int& di : d) cin >> di;
        int ans = 0;
        for (int q = 0; q < n; q++) {
            ans += d[q] >= l;
        }
        if (ans > 0) { cout << "No" << endl; continue; }
        int p[2 * k];
        for (int i = 0; i < 2 * k; i++) {
            p[i] = min(i + 1, k);
        }
        for (int i = 2 * k - 1; i >= 0; i--) {
            p[i] = p[i] - 1;
        }
        for (int i = 0; i < n; i++) {
            d[i] += p[(i + ans) % (2 * k)];
            ans += d[i] < l;
        }
        if (ans == n) { cout << "Yes" << endl; }
        else { cout << "No" << endl; }
    }
}
