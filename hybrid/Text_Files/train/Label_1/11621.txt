#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, A, B, C, T, t;
    cin >> n >> A >> B >> C >> T;
    multiset<int> received;
    long long ans = 0;
    for (int i = 1; i <= T; ++i) {
        ans += 1LL * C * received.size();
        while (!received.empty() && *received.begin() == i) {
            ans += 1LL * A;
            received.erase(received.begin());
        }
        cin >> t;
        received.insert(i + t);
    }
    cout << ans << endl;
    return 0;
}
