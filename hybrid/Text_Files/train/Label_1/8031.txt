#include <iostream>
#include <vector>
using namespace std;
int main() {
  long long n;
  cin >> n;
  vector<long long> a(n + 2);
  for (long long i = 1; i <= n; i++) cin >> a[i];
  long long l = 1, r = 1;
  long long ans = 1;
  for (long long i = 2; i <= n; i++) {
    if (a[i] == a[i - 1] + a[i - 2]) {
      if (i - l + 1 > ans) ans = i - l + 1, r = i;
    } else
      l = i;
  }
  cout << ans << endl;
}
