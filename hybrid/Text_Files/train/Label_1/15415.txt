#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mod = 1e9 + 9;

vector<pair<int, int>> v;
int c[100001];

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    int x, y;
    cin >> x >> y;
    v.emplace_back(x, y);
  }
  sort(v.begin(), v.end());
  for (int i = 1; i < n; ++i) {
    int l = 0, r = i - 1, ind = -1;
    while (l <= r) {
      int mid = (l + r) >> 1;
      if (v[mid].second < v[i].second)
        l = mid + 1;
      else
        r = mid - 1, ind = mid;
    }
    if (ind != -1) c[i] = c[ind];
  }
  ll ans = 0;
  for (int i = 0; i < n; ++i) c[i] = (c[i] + 1) % mod, ans = (ans + (c[i] * (n - i - 1)) % mod) % mod;
  cout << ans << endl;
}
