import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            String p = in.next();
            String h = in.next();
            char[] pArr = p.toCharArray();
            Arrays.sort(pArr);
            String s = new String(pArr);
            if (h.contains(s)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
