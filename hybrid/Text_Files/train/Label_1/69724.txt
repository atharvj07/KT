from collections import defaultdict

n = int(input())
a = []
b = []
for _ in range(n):
    x, y = map(int, input().split())
    a.append(x)
    b.append(y)

ans = 1
cnt = defaultdict(int)
cnt[(0, 0)] = 1
for x, y in zip(a, b):
    cnt[(x, y)] = cnt[(x - 1, y)] + cnt[(x, y - 1)]
    ans = max(ans, cnt[(x, y)])
print(ans)
