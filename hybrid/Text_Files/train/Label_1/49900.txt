import java.util.Scanner;

public class RockPaperScissorsTournament {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int q = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        char[] shapes = scanner.nextLine().toCharArray();
        int[] counts = new int[3]; // Counts of rock, paper, and scissors
        for (char shape : shapes) {
            counts[getShapeIndex(shape)]++;
        }

        int[] possibleChampions = new int[q + 1];
        possibleChampions[0] = calculatePossibleChampions(counts);

        for (int i = 0; i < q; i++) {
            int player = scanner.nextInt();
            char newShape = scanner.next().charAt(0);
            int oldShapeIndex = getShapeIndex(shapes[player - 1]);
            int newShapeIndex = getShapeIndex(newShape);
            counts[oldShapeIndex]--;
            counts[newShapeIndex]++;
            shapes[player - 1] = newShape;
            possibleChampions[i + 1] = calculatePossibleChampions(counts);
        }

        for (int i = 0; i <= q; i++) {
            System.out.println(possibleChampions[i]);
        }
    }

    private static int getShapeIndex(char shape) {
        switch (shape) {
            case 'R':
                return 0;
            case 'P':
                return 1;
            case 'S':
                return 2;
            default:
                throw new IllegalArgumentException("Invalid shape: " + shape);
        }
    }

    private static int calculatePossibleChampions(int[] counts) {
        // If there are no players left, there are no possible champions.
        if (counts[0] + counts[1] + counts[2] == 0) {
            return 0;
        }

        // If there are only players of one shape left, they are all possible champions.
        if (counts[0] == 0 || counts[1] == 0 || counts[2] == 0) {
            return counts[0] + counts[1] + counts[2];
        }

        // If there are players of all three shapes left, there is only one possible champion.
        return 1;
    }
}
