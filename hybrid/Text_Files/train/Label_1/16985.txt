#include<iostream>
#include<string>
using namespace std;
int main() {
    int n;
    string s;
    cin >> n >> s;
    int i = 0;
    while(i + 1 < n) {
        if(s[i] == s[i + 1] && s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' || s[i] == 'u' || s[i] == 'y')
            s.erase(i, 1);
        else
            i++;
    }
    cout << s << endl;
    return 0;
}
