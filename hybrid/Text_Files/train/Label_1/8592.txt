#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> a(n, vector<int>(5));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 5; j++) {
            cin >> a[i][j];
        }
    }
    vector<int> cnt(5);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 5; j++) {
            if (a[i][j] != -1) cnt[j]++;
        }
    }
    int ans = 0;
    for (int j = 0; j < 5; j++) {
        int x = cnt[j] * 1000000 / n;
        if (x >= 25 && x <= 50) ans += 5;
        else if (x >= 125 && x <= 25) ans += 4;
        else if (x >= 500 && x <= 125) ans += 3;
        else if (x >= 2000 && x <= 500) ans += 2;
        else if (x >= 8000 && x <= 2000) ans += 1;
    }
    if (ans == 0) cout << -1 << endl;
    else cout << ans << endl;
    return 0;
}
