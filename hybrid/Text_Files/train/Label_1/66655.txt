import sys, collections
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    next = list(map(int, input().split()))

    stack = []
    ans = []
    used = [False] * (n+1)
    for i in range(n-1, -1, -1):
        if next[i] == -1:
            while(stack and not used[stack[-1]]):
                ans.append(stack.pop())
            stack.append(i+1)
            used[i+1] = True
        else:
            while stack and next[i] < stack[-1]:
                ans.append(stack.pop())
            stack.append(i+1)
    while stack:
        ans.append(stack.pop())

    if len(ans) < n:
        print(-1)
    else:
        print(*ans)
