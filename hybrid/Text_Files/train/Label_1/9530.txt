#include <iostream>
using namespace std;
int main() {
    int n, m, c = 0;
    cin >> n >> m;
    int a[n + 1][m + 1];
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            cin >> a[i][j];
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++) {
            c += (!a[i][j]) * (a[i - 1][j] || a[i][j + 1] || a[i + 1][j] || a[i][j - 1]);
            c += (!a[i][j]) * (a[i - 1][j] || a[i][j - 1] || a[i + 1][j] || a[i][j + 1]);
        }
    cout << c << endl;
}
