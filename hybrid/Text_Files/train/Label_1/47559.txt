import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();

        long p = 1;
        for (long i = 3; i <= n; i++) {
            p = p * i % 1000000007;
        }

        long q = 1;
        for (long i = 1; i <= n - 2; i++) {
            q = q * i % 1000000007;
        }

        System.out.println((p - q + 1000000007) % 1000000007);
    }
}
