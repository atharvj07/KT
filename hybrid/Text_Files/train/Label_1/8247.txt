#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<pair<int, int>> ratings(n);
    for (int i = 0; i < n; i++) {
        cin >> ratings[i].first >> ratings[i].second;
    }
    bool rated = false, unrated = false;
    for (int i = 0; i < n; i++) {
        if (ratings[i].first != ratings[i].second) {
            rated = true;
        }
        if (i > 0 && ratings[i - 1].first > ratings[i].first) {
            unrated = true;
        }
    }
    if (rated) cout << "rated" << endl;
    else if (unrated) cout << "maybe" << endl;
    else cout << "unrated" << endl;
}
