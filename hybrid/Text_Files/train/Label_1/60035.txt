n, m, k = map(int, input().split())
a = [input() for _ in range(n)]
v = []
s0 = []
for i, j in enumerate(a):
    l = [0, 0]  # l[0] = first lesson, l[1] = last lesson
    t = 0
    for p in range(m):
        if j[p] == '1':
            t += 1
            if l[0] == 0:
                l[0] = p
            l[1] = p
    s0.append(max(1, t))
    v.append(l)
s0.sort()
t = sum(s0[:n - k])
for i in range(n - k, n):
    for j in range(m - v[i][1]):
        if a[i][j + v[i][1] + 1] == '1':
            t += 1
            break
print(t)
