#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    string ans;
    for (char c : s) {
      if (c == '?') {
        if (ans.empty() || ans.back() != 'R')
          ans += 'R';
        else
          ans += 'B';
      } else
        ans += c;
    }
    cout << ans << endl;
  }
  return 0;
}
