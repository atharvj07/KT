import java.util.*;
import java.lang.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    while (t-- > 0) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      int p = sc.nextInt();
      String s = sc.next();
      int n = s.length();
      int[] dp = new int[n + 1];
      Arrays.fill(dp, Integer.MAX_VALUE);
      dp[n] = 0;
      for (int i = n - 1; i >= 1; i--) {
        if (s.charAt(i - 1) == 'A') {
          dp[i] = Math.min(dp[i], dp[i + 1] + a);
          if (i + 1 < n && s.charAt(i) == 'A') {
            dp[i] = Math.min(dp[i], dp[i + 2] + a);
          }
        }
        if (s.charAt(i - 1) == 'B') {
          dp[i] = Math.min(dp[i], dp[i + 1] + b);
          if (i + 1 < n && s.charAt(i) == 'B') {
            dp[i] = Math.min(dp[i], dp[i + 2] + b);
          }
        }
      }
      for (int i = 1; i <= n; i++) {
        if (dp[i] <= p) {
          System.out.println(i);
          break;
        }
      }
    }
  }
}
