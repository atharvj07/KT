import java.util.Scanner;

public class PackingObjects {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] objects = new int[n];
        for (int i = 0; i < n; i++) {
            objects[i] = scanner.nextInt();
        }

        int maxPacked = 0;
        int objectsLeft = n;
        int currentBoxSize = k;

        for (int i = 0; i < n; i++) {
            if (objects[i] > currentBoxSize) {
                objectsLeft--;
                continue;
            }

            currentBoxSize -= objects[i];
            maxPacked++;

            if (currentBoxSize == 0) {
                m--;
                currentBoxSize = k;
            }

            if (m == 0) {
                break;
            }
        }

        System.out.println(maxPacked);
    }
}
