#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> k(n);
  for (int &i : k) cin >> i;
  vector<vector<int>> m(n);
  for (int i = 0; i < n; i++) {
    int p = k[i];
    for (int j = 0; j < p; j++) {
      int tmp;
      cin >> tmp;
      m[i].push_back(tmp);
    }
  }
  vector<long long> res(n);
  for (int i = 0; i < n; i++) {
    long long tmp = 0;
    for (int j = 0; j < (int)m[i].size(); j++) {
      tmp += (m[i][j] * 5);
      tmp += 15;
    }
    res[i] = tmp;
  }
  cout << *min_element(res.begin(), res.end()) << "\n";
}
