#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
const int INF = 0x3f3f3f3f;
const ll LLINF = 0x3f3f3f3f3f3f3f3f;
int n; string a, b, ans1, ans2;
bool vis[1 << 30];
vector<pair<int, int>> pos[260];
int main() {
	cin >> n >> a >> b;
	for (int i = 0; i < n; ++i) {
		pos[a[i] - '0'].push_back({ i, 1 });
		pos[b[i] - '0'].push_back({ i, -1 });
	}
	queue<int>q;
	int ans = INT_MAX;
	for (int i = 0; i < n; ++i) {
		int now = (1 << (a[i] - '0'));
		if (!vis[now]) {
			q.push(now);
			vis[now] = true;
			if (now == (1 << (b[i] - '0'))) {
				ans = 0;
				break;
			}
		}
	}
	if (ans != INT_MAX) {
		q.push((1 << 30) - 1);
	}
	while (!q.empty()) {
		int now = q.front(); q.pop();
		if (now == (1 << 30) - 1) {
			ans = 1;
			break;
		}
		int nw = now, nx;
		for (int i = 0; i < n; ++i) {
			int d = (now >> i) & 1;
			if (!d)continue;
			for (auto j : pos[i]) {
				int x = j.first; int y = j.second;
				if (i > 0 && (!d || d != ((now >> (i - 1)) & 1))) {
					nx = nw;
					nx ^= (1 << i);
					nx ^= (1 << (i - 1));
					if (!vis[nx]) {
						q.push(nx);
						vis[nx] = true;
						if (nx == (1 << 30) - 1) {
							ans = 2;
							break;
						}
						ans1.push_back(x), ans2.push_back(y);
					}
				}
				if (i < n - 1 && (!d || d != ((now >> (i + 1)) & 1))) {
					nx = nw;
					nx ^= (1 << i);
					nx ^= (1 << (i + 1));
					if (!vis[nx]) {
						q.push(nx);
						vis[nx] = true;
						if (nx == (1 << 30) - 1) {
							ans = 2;
							break;
						}
						ans1.push_back(x), ans2.push_back(y);
					}
				}
			}
			if (ans != INT_MAX)break;
		}
		if (ans != INT_MAX)break;
	}
	if (ans == INT_MAX) {
		cout << -1 << endl;
		return 0;
	}
	cout << ans << endl;
	for (int i = 0; i < ans1.size(); ++i) {
		cout << ans1[i] + 1 << " " << ans2[i] << endl;
	}
	return 0;
}
