#include <iostream>
#include <vector>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    vector<int> dp(a);
    for (int i = n - 1; i > 0; i--) {
        if (a[i] <= a[i - 1]) dp[i - 1] = a[i - 1];
    }
    for (int i = 1; i < k; i++) {
        for (int j = a.size() - i * k; j < a.size() - i * k + k; j++) {
            dp[j] = max(dp[j], dp[j + k]);
        }
    }
    cout << dp[0] << endl;
}
