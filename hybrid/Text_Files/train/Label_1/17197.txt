#include <iostream>

using namespace std;

typedef long long ll;

const ll M = 1000000007;

ll powmod(ll base, ll exp) {
  ll ret = 1;
  while (exp > 0) {
    if (exp % 2 == 1)
      ret = (ret * base) % M;
    base = (base * base) % M;
    exp /= 2;
  }
  return ret;
}

ll inv(ll x) { return powmod(x, M - 2); }

ll comb(ll n, ll k) {
  if (k > n) {
    return 0;
  }
  ll ret = 1;
  for (ll i = n; i > n - k; i--) {
    ret = (ret * i) % M;
  }
  ret = (ret * inv(k)) % M;
  return ret;
}

ll solve(ll n, ll r) {
  ll ans = 0;
  for (ll i = 1; i <= r; i++) {
    ans = (ans + comb(n + i - 1, i - 1)) % M;
  }
  return ans;
}

int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll n, r;
    cin >> n >> r;
    cout << solve(n, r) << endl;
  }
  return 0;
}
