import java.util.Scanner;

public class EasyOrNot {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();
        String s = scanner.nextLine();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int[][] dp = new int[n + 1][5];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 5; j++) {
                dp[i][j] = dp[i + 1][j];
                if (j > 0 && s.charAt(i) == "hard".charAt(j - 1)) {
                    dp[i][j] = Math.min(dp[i][j], dp[i + 1][j - 1] + a[i]);
                }
            }
        }
        System.out.println(dp[0][4]);
    }
}
