#include <bits/stdc++.h>
using namespace std;

const long long mod = 998244353LL;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> grid(n);
  for (auto &s : grid) {
    cin >> s;
  }
  long long ans = 0LL;
  for (int mask = 0; mask < (1 << n); mask++) {
    long long h = 0LL, v = 0LL;
    for (int i = 0; i < n; i++) {
      bool b = (mask & (1 << i));
      for (int j = 0; j < m; j++) {
        if (grid[i][j] == 'o') {
          if (b) h++;
          else v++;
        }
      }
    }
    h /= 2LL;
    v /= 2LL;
    ans = (ans + h * v) % mod;
  }
  cout << ans << endl;
  return 0;
}
