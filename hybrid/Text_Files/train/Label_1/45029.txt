import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < q; i++) {
            long l = sc.nextInt();
            long r = sc.nextInt();
            int ans = 0;
            for (int j = 0; j < n; j++) {
                if (l <= a[j] && a[j] <= r) {
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
