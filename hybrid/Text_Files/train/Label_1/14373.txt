#include<bits/stdc++.h>
using namespace std;
#define ll long long

const int MX = 2005;
const ll mod = 1e9+7;

int n, p;
ll a[MX];
ll dp[MX][MX];

void preprocess() {
    dp[0][0] = 1;
    for(int i=1; i<MX; i++) {
        dp[i][0] = dp[i-1][i-1];
        for(int j=1; j<=i; j++) {
            dp[i][j] = (dp[i-1][j] + dp[i-1][j-1]) % mod;
        }
    }
}

int main() {
    preprocess();
    cin >> n >> p;
    for(int i=1; i<=n; i++) cin >> a[i];

    ll x = dp[n][n];
    sort(a+1, a+1+n);

    vector<ll> ans;

    for(int j=0; j<=n; j++) {
        bool ok = 0;
        for(int i=1; i<=n; i++) {
            if(j <= a[i]) {
                ok = 1;
                break;
            }
        }
        if(!ok) ans.push_back(j);
    }

    cout << ans.size() << endl;
    for(auto v : ans) cout << v << ' ';
    cout << endl;

    return 0;
}
