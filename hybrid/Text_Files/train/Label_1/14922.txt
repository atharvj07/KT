#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> x(n), c(n);
    for (int i = 0; i < n; i++) cin >> x[i];
    int a;
    cin >> a;
    for (int i = 0; i < n; i++) cin >> c[i];
    vector<int> dp(k + 1, INT_MAX);
    dp[k] = 0;
    for (int i = 0; i < n; i++) {
        for (int w = k; w >= 0; w--) {
            if (dp[w] < INT_MAX) {
                if (w + x[i] <= k) dp[w + x[i]] = min(dp[w + x[i]], dp[w]);
                if (w + x[i] + a <= k) dp[w + x[i] + a] = min(dp[w + x[i] + a], dp[w] + c[i]);
            }
        }
    }
    cout << (dp[k] == INT_MAX ? -1 : dp[k]) << endl;
    return 0;
}
