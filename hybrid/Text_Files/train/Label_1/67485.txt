from collections import deque
inf = float('inf')
n, m = [int(x) for x in input().split()]
grid = [input() for i in range(n)]
ans = 0
for i in range(n):
    for j in range(m):
        if grid[i][j] == 'D':
            idx = 0
            di = [1, 0, -1, 0]
            dj = [0, 1, 0, -1]
            x = i
            y = j
            check = []
            count = 0
            while idx < len(di):
                ni = x + di[idx]
                nj = y + dj[idx]
                if ni >= 0 and ni < n and nj >= 0 and nj < m and (ni, nj) not in check:
                    if grid[ni][nj] == 'I':
                        x = ni
                        y = nj
                        idx += 1
                        check.append((ni, nj))
                        count += 1
                    elif grid[ni][nj] == 'M':
                        x = ni
                        y = nj
                        idx += 1
                        check.append((ni, nj))
                        count += 1
                    elif grid[ni][nj] == 'A':
                        x = ni
                        y = nj
                        idx += 1
                        check.append((ni, nj))
                        count += 1
                    elif grid[ni][nj] == 'D':
                        x = ni
                        y = nj
                        idx += 1
                        check.append((ni, nj))
                        count += 1
                else:
                    idx += 1
            if count % 4 == 0:
                ans = max(ans, count // 4)
if ans == 0:
    print("Poor Dima!")
elif ans == inf:
    print("Poor Inna!")
else:
    print(ans)
