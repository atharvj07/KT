#include <iostream>
#include <algorithm>

using namespace std;

int sequence[1000];

bool isFibonacci(int n) {
    return n == 0 || n == 1 || isFibonacci(n - 1) || isFibonacci(n - 2);
}

int longestFibonacci(int n) {
    int dp[n];
    fill(dp, dp + n, 2);
    dp[0] = 1;
    dp[1] = 1;
    for (int i = 2; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (sequence[i] == sequence[j] + sequence[i - j - 1] && dp[j] + 1 > dp[i]) {
                dp[i] = dp[j] + 1;
            }
        }
    }
    int maxLength = 0;
    for (int i = 0; i < n; i++) {
        maxLength = max(maxLength, dp[i]);
    }
    return maxLength;
}

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> sequence[i];
    }
    sort(sequence, sequence + n);
    cout << longestFibonacci(n) << endl;
    return 0;
}
