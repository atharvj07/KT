#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        string s;
        cin>>s;
        bool ans=true;
        int n=s.size();
        for(int i=1;i<n;i++){
            if(s[i-1]==s[i]||s[i]-'a'-s[i-1]-'a'==1||s[i]-'a'-s[i-1]-'a'==-1){
                ans=false;
                break;
            }
        }
        if(!ans){
            cout<<"NO Answer"<<endl;
        }
        else{
            for(char ch='a';ch<='z';ch++){
                if(ch>s[n-1]&&s.find(ch)==-1) continue;
                while(s.find(ch)!=-1){
                    if(s[n-1]<ch){
                        cout<<"NO Answer"<<endl;
                        return 0;
                    }
                    cout<<ch;
                    int l=s.find(ch);
                    swap(s[l],s[n-1]);
                    s.pop_back();
                    n--;
                }
            }
            cout<<endl;
        }
    }
    return 0;
}
