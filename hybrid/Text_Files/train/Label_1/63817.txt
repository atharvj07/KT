from collections import deque

n, m, k = [int(i) for i in input().split()]
a = [input() for i in range(n)]
x, y = None, None
for i in range(n):
    for j in range(m):
        if a[i][j] == 'X':
            x, y = i, j
            break
    if x is not None:
        break

assert x is not None and y is not None
was = set()
q = deque([(x, y, "", 0)])

while q:
    x, y, s, cnt = q.popleft()
    if cnt == k:
        print(s)
        exit()
    for i, j in ((0, -1), (0, 1), (-1, 0), (1, 0)):
        nx = x + i
        ny = y + j
        ns = s + ["L", "R", "U", "D"][i + 1] * (-i) + ["L", "R", "U", "D"][j + 1]
        if 0 <= nx < n and 0 <= ny < m and a[nx][ny] != '*' and (nx, ny) not in was:
            was.add((nx, ny))
            q.append((nx, ny, ns, cnt + 1))

print("IMPOSSIBLE")
