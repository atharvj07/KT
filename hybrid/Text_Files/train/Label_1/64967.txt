t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    counts = {}
    for num in a:
        counts[num] = counts.get(num, 0) + 1
    max_count = max(counts.values())
    if max_count > n - max_count:
        print((n - max_count) * 2 + 1)
    else:
        print(n % 2)
