#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> p(n + 1), b(n + 1);
    for (int i = 1; i <= n; i++) { cin >> p[i]; if (p[i]) b[p[i]] = i; }
    for (int i = 1; i <= n; i++) if (!p[i]) p[i] = n + i;
    for (int i = 2; i <= n; i++) if (b[i - 1] > b[i]) swap(p[i - 1], p[i]);
    int ans = 0;
    for (int i = 1; i < n; i++) if (p[i] % 2 != p[i + 1] % 2) ans++;
    cout << ans << endl;
}
