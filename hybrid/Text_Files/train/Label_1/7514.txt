#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define mp make_pair
#define pb push_back
#define ff first
#define ss second
int main(){
	ll n,k,mx=100000,dp[11][100010];
	string s;
	cin>>n>>k;
	cin>>s;
	for(int i=0;i<=k;i++){
		for(int j=0;j<=n;j++){
			dp[i][j]=mx;
		}
	}
	dp[0][0]=0;
	for(int i=1;i<=k;i++){
		for(int j=1;j<=n;j++){
			dp[i][j]=min(dp[i][j],dp[i-1][j]);
			if(s[j-1]==s[j-1-dp[i-1][j-1]]) dp[i][j]=min(dp[i][j],dp[i-1][j-1]+1);
			if(j-dp[i][j-1]-1>=0) dp[i][j]=min(dp[i][j],dp[i][j-dp[i][j-1]-1]+1);
		}
	}
	if(dp[k][n]==mx) cout<<-1<<endl;
	else cout<<dp[k][n]<<endl;
	return 0;
}
