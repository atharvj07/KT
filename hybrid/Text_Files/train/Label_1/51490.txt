import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] info = new int[n + 1][4];
        info[0][0] = sc.nextInt();
        info[0][1] = sc.nextInt();
        info[0][2] = sc.nextInt();
        info[0][3] = sc.nextInt();
        for (int i = 1; i <= n; i++) {
            info[i][0] = sc.nextInt();
            info[i][1] = sc.nextInt();
            info[i][2] = sc.nextInt();
            info[i][3] = sc.nextInt();
        }
        Arrays.sort(info, (a, b) -> b[3] - a[3]);
        long damage = 0;
        boolean alive = true;
        for (int i = 1; i <= n; i++) {
            long diff = Math.max(info[0][1] - info[i][2], 0);
            info[i][0] -= diff;
            damage += diff;
            if (info[i][0] > 0) {
                alive = false;
            }
        }
        if (!alive) {
            System.out.println(-1);
        } else {
            System.out.println(damage);
        }
    }
}
