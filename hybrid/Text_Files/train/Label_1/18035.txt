#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int MAXN = 2e5 + 5;
int dp[2005][MAXN];
int a[MAXN], x[MAXN], y[MAXN];
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= k; i++) dp[i][0] = 0x3f3f3f3f;
  for (int i = 1; i <= m; i++) cin >> x[i] >> y[i];
  sort(a + 1, a + 1 + n);
  for (int i = 1; i <= m; i++) {
    for (int j = 1; j <= k; j++) {
      dp[j][i] = dp[j][i - 1];
      if (j >= x[i]) { dp[j][i] = min(dp[j][i], dp[j - x[i]][i - 1] + a[y[i]]); }
    }
  }
  cout << dp[k][m] << endl;
  return 0;
}
