#include <bits/stdc++.h>
#define _ ios_base::sync_with_stdio(0);
using namespace std;

int main() {_
    int n, m, x, y;
    cin >> n >> m;
    long double tmp;
    long double ans = LLONG_MAX;
    while(n--) {
        cin >> x >> y;
        tmp = (long double)x / y;
        if(ans > tmp)  ans = tmp;
    }
    cout << fixed << setprecision(10) << ans *m << "\n";
}
