#include<bits/stdc++.h>
using namespace std;
int n,a[200005];
int f[200005];
int main(){
    cin>>n;
    for(int i=1; i<=n; ++i) cin>>a[i];
    for(int i=1; i<=n; ++i){
        // we can buy up to a[i] pizzas using coupons
        f[i]=min(a[i],f[i-1]+1);
    }
    for(int i=n; i>=1; --i){
        // we can buy up to f[i] pizzas using discounts
        f[i]=min(f[i],f[i+1]+a[i]);
    }
    for(int i=1; i<=n; ++i){
        if(a[i]!=f[i]){
            cout<<"NO\n";
            return 0;
        }
    }
    cout<<"YES\n";
}
