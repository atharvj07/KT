n = int(input())
a = [int(x) for x in input().split()]
pref = [a[0]]
for i in range(1, n): pref.append(pref[-1] + a[i])
aver = [a[i] / (i + 1) for i in range(n)]
for i in range(1, n): aver[i] = max(aver[i], aver[i - 1] + (pref[i] - pref[i - 1]) / (i + 1))
ans = 0
now = 0
for i in range(n):
    if i == n - 1 or aver[i] < aver[i + 1]:
        ans = max(ans, now)
        now = 0
    else:
        now += 1
ans = max(ans, now)
print(ans)
