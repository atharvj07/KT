#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int n, d, m;
    cin >> n >> d >> m;
    long long a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n, greater<int>());
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] > m) {
            ans += a[i];
            i += d - 1;
        }
    }
    cout << ans << endl;
    return 0;
}
