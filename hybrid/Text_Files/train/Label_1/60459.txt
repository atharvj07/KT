def longest_increasing_subsequence(arr):
    sub = []
    for ele in arr:
        pos = binary_search(sub, ele)
        if pos == len(sub):
            sub.append(ele)
        else:
            sub[pos] = ele
    return len(sub)

def binary_search(sub, ele):
    low, high = 0, len(sub)
    while low < high:
        mid = (low + high) // 2
        if sub[mid] < ele:
            low = mid + 1
        else:
            high = mid
    return low

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        arr = list(map(int, input().split()))
        if len(set(arr)) == n:
            print(n)
        else:
            print(longest_increasing_subsequence(arr))

solve()
