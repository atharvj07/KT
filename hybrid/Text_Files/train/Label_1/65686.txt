import sys
input = sys.stdin.readline

n,m=map(int,input().split())
fire,k,bers=map(int,input().split())
A=list(map(int,input().split()))
B=list(map(int,input().split()))
P=sorted(list(set(A)))+sorted(list(set(B)))

dp=[1e9 for _ in range(n+1)]

lA,lB=0,0
ans=1e9

for i in range(n+1):
    while lA<n and A[lA]<=P[i]:lA+=1
    while lB<m and B[lB]<=P[i]:lB+=1
    
    mana=0
    if i>0:mana+=min(bers,dp[i-1])
    if lA<=n and lB>0:
        if A[lA-1]==B[lB-1]:
            mana+=min(fire,dp[lA])
        else:mana+=fire
    if lA<n and lB<=m:
        if A[lA]==B[lB]:
            mana+=min(fire,dp[lA+1])
        else:mana+=fire
    
    dp[i]=mana

if dp[i]!=1e9:ans=dp[i]
if ans==1e9:ans=-1

print(ans)
