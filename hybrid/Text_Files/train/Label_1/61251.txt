import sys

n, m, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
offers = []
for i in range(m):
    x, y = map(int, input().split())
    offers.append((x, y))

ans = []
for x, y in offers:
    if x <= k:
        ans.append((x, max(0, x - y)))

ans.sort(key=lambda x: x[1], reverse=True)
cur = 0
cost = 0
for x, y in ans:
    if cur + x <= k:
        cost += sum(a[cur:cur + x - y])
        cur += x
    else:
        break

if cur < k:
    cost += sum(a[cur:cur + k - cur])

print(cost)
