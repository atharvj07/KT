import bisect
n1, y1 = map(int, input().split())
a = [int(x) for x in input().split()]
n2, y2 = map(int, input().split())
b = [int(x) for x in input().split()]
a.extend([-1000000000000000000, -1000000000000000000, 1000000000000000000, 1000000000000000000])
b.extend([-1000000000000000000, -1000000000000000000, 1000000000000000000, 1000000000000000000])
b_b = bisect.bisect_left(b, y1)
b_t = bisect.bisect_right(b, y1)
a_b = bisect.bisect_left(a, y2)
a_t = bisect.bisect_right(a, y2)
print(max(b_b + n1 - a_b, b_t + n1 - a_t))
