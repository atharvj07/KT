#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  string s;
  cin >> n >> s;

  vector<int> dp(n + 1);
  dp[0] = 1;
  dp[1] = (s[0] == '0');
  for (int i = 2; i <= n; ++i) {
    dp[i] = max(dp[i - 1], dp[i - 2] + (s[i - 1] != s[i - 2]));
  }
  cout << dp[n] << endl;
}
