#include <iostream>
#include <algorithm>
    using namespace std;
    int n, c, d, ans, beauty[100001], cost[100001], type[100001];
    int main() {
        cin >> n >> c >> d;
        for (int i = 0; i < n; i++) {
            cin >> beauty[i] >> cost[i] >> type[i];
            if (type[i] == 'C')
                c -= cost[i];
            else
                d -= cost[i];
        }
        sort(beauty, beauty + n, greater<int>());
        for (int i = 0; i < n; i++) {
            if (c > 0 && type[i] == 'C') {
                c -= cost[i];
                ans += beauty[i];
            } else if (d > 0 && type[i] == 'D') {
                d -= cost[i];
                ans += beauty[i];
            }
        }
        cout << ans << endl;
    }
