#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define pb push_back
int main() {
    int t;
    cin >> t;
    while(t--) {
        int n;
        cin >> n;
        vi a(n);
        for(int i = 0; i < n; i++)  cin >> a[i];
        int cnt = 0;
        for(int i = 0; i < n; i++)  cnt += (a[i] == 0);
        int ans = n - cnt;
        for(int i = 0; i < n; i++) {
            if(a[i] == 0)  continue;
            int x = a[i];
            for(int j = i + 1; j < n; j++) {
                if(a[j] != x) {
                    ans -= (a[j] > 0);
                    break;
                } else  a[j] = 0;
            }
        }
        cout << ans << '\n';
    }
}
