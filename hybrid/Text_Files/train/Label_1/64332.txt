from itertools import cycle

for _ in range(int(input())):
    n, k = map(int, input().split())
    s = [c for _, c in zip(range(n), cycle('abc'))]
    if k > 1:
        s[n // 2 - k // 2 : n // 2 + k // 2 + (k % 2)] = list('b' * k)
    elif (n & 1) and (n > 1):
        s[n // 2] = 'b'
    print(''.join(s))
