#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int N = 2e5 + 5, lg = 20;

vector<array<int, 3>> adj[N];
int tin[N], tout[N], timer = 1;
int p[N][lg + 1];
vector<int> val(N), sub(N);

void dfs(int u, int par) {
  p[u][0] = par;
  tin[u] = timer++;
  for (int i = 1; i <= lg; i++) {
    p[u][i] = p[p[u][i - 1]][i - 1];
  }
  for (auto &[v, w, c] : adj[u]) {
    if (v == par) continue;
    val[v] = val[u] | (c << w);
    sub[v] = 1;
    dfs(v, u);
    sub[u] += sub[v];
  }
  tout[u] = timer++;
}

bool is_anc(int u, int v) {
  return (tin[u] <= tin[v] && tout[v] <= tout[u]);
}

int get_lca(int u, int v) {
  if (is_anc(u, v)) return u;
  if (is_anc(v, u)) return v;
  for (int i = lg; i >= 0; i--) {
    if (!is_anc(p[u][i], v)) u = p[u][i];
  }
  return p[u][0];
}

int get_val_on_path(int u, int v) {
  if (is_anc(u, v)) return val[u];
  if (is_anc(v, u)) return val[v];
  int lca = get_lca(u, v);
  return val[u] ^ val[v] ^ val[lca] ^ (val[lca] << 1);
}

int32_t main() {
  int n;
  cin >> n;
  for (int i = 0, u, v, c; i < n - 1; i++) {
    cin >> u >> v >> c;
    adj[u].push_back({v, i, c});
    adj[v].push_back({u, i, c});
  }
  dfs(1, 1);
  ll ans = 0;
  for (int u = 1; u <= n; u++) {
    for (auto &[v, i, w] : adj[u]) {
      if (v == p[u][0]) continue;
      int pth = get_val_on_path(u, v);
      if (!(pth & (1 << w))) ans += sub[v] * (n - sub[v]);
    }
  }
  cout << ans / 2 << '\n';
  return 0;
}
