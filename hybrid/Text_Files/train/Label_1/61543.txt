def find_permutation(n):
    if n == 2 or n == 3:
        return -1
    even = [i for i in range(2, n+1, 2)]
    odd = [i for i in range(1, n+1, 2)]
    return even + odd if n % 2 == 0 else odd + even

t = int(input())
for _ in range(t):
    n = int(input())
    result = find_permutation(n)
    if result == -1:
        print(result)
    else:
        print(*result)
