t = int(input())
for _ in range(t):
    n = int(input())
    q = list(map(int, input().split()))
    
    # Lexicographically minimal permutation
    used = set()
    min_perm = []
    for i in range(n):
        if q[i] not in used:
            min_perm.append(q[i])
            used.add(q[i])
        else:
            for j in range(1, n+1):
                if j not in used:
                    min_perm.append(j)
                    used.add(j)
                    break
    
    # Lexicographically maximal permutation
    used = set()
    max_perm = []
    last_max = 0
    for i in range(n):
        if q[i] not in used:
            max_perm.append(q[i])
            used.add(q[i])
            last_max = q[i]
        else:
            for j in range(last_max-1, 0, -1):
                if j not in used:
                    max_perm.append(j)
                    used.add(j)
                    break
                    
    print(" ".join(map(str, min_perm)))
    print(" ".join(map(str, max_perm)))
