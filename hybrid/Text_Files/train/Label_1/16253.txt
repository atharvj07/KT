#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<pair<long long, long long>> pairs(n);
  for (int i = 0; i < n; i++) {
    cin >> pairs[i].first >> pairs[i].second;
  }

  vector<long long> operations(n);
  for (int i = 0; i < n; i++) {
    while (pairs[i].first > 0 && pairs[i].second > 0) {
      if (pairs[i].first < pairs[i].second) {
        operations[i]++;
        pairs[i].second -= pairs[i].first;
      } else {
        operations[i]++;
        pairs[i].first -= pairs[i].second;
      }
    }
  }

  for (int i = 0; i < n; i++) {
    cout << operations[i] << endl;
  }

  return 0;
}
