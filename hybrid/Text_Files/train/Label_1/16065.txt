#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
const int maxn = 1e3 + 5;
int n, s, v[maxn];
bool check(int x) {
  long long sum = 0;
  for (int i = 0; i < n; i++) sum += max(0, v[i] - x);
  return sum >= s;
}
int main() {
  cin >> n >> s;
  for (int i = 0; i < n; i++) cin >> v[i];
  int l = 0, r = 1e9;
  while (l < r) {
    int mid = (l + r) / 2;
    if (check(mid))
      l = mid + 1;
    else
      r = mid;
  }
  if (check(l - 1))
    cout << l - 1 << endl;
  else
    cout << -1 << endl;
  return 0;
}
