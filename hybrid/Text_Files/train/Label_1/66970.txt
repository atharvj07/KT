n, p, m = map(int, input().split())
top_ups = [list(map(int, input().split())) for _ in range(n)]

balance = 0
negative_days = 0
current_day = 1

for d, t in top_ups:
    while current_day < d:
        balance -= p
        if balance < 0:
            negative_days += 1
        current_day += 1
    balance += t

while current_day <= m:
    balance -= p
    if balance < 0:
        negative_days += 1
    current_day += 1

print(negative_days)
