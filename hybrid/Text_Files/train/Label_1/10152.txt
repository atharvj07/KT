#include <bits/stdc++.h>
#define int long long int
using namespace std;
const int mod = 1e9+7;
const int inf = 1e18;
int32_t main()
{
    int n,k;
    cin>>n>>k;
    vector<int> a(n),freq(k);
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
        freq[a[i]%k]++;
    }
    int ans=0;
    for(int i=0;i<k;i++)
    {
        ans+=freq[i]/2;
        int rem = k-i;
        if(rem==i)
            continue;
        ans+=min(freq[rem],freq[i]);
        freq[i]=0;
        freq[rem]=0;
    }
    cout<<ans*2<<'\n';
    return 0;
}
