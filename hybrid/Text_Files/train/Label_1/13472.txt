#include <bits/stdc++.h>
using namespace std;
int main() {
  long long a, k;
  cin >> a >> k;
  string s = to_string(a);
  for (int i = 0; i < s.size() && k; i++) {
    for (int j = i + 1; j < s.size(); j++) {
      if (s[j] > s[i]) {
        swap(s[i], s[j]);
        k--;
        break;
      }
    }
  }
  cout << s << "\n";
}
