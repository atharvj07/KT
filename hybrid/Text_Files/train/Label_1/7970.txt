#include <bits/stdc++.h>
using namespace std;
int n, k, ans, cnt[26];
string s;
int main() {
  cin >> n >> k;
  cin >> s;
  for (int i = 0; i < n; i++) {
    int ch = s[i] - 'A';
    cnt[ch]++;
    if (i >= k - 1) {
      int mn = cnt[0];
      for (int j = 1; j < k; j++) { mn = min(mn, cnt[j]); }
      if (mn > 0) {
        for (int j = 0; j < k; j++) { cnt[j] -= mn; }
        ans += mn;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
