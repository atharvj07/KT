#include<bits/stdc++.h>
using namespace std;
bool isposs(long long int x,long long int y){
    if(x>y){
        return false;
    }
    if((x+y)%3==0){
        return true;
    }
    return false;
}
int main(){
    long long int x,y;
    cin>>x>>y;
    if(isposs(x,y)){
        cout<<"Yes"<<"\n";
    }
    else{
        cout<<"No"<<"\n";
    }
    return 0;
}
