#include<bits/stdc++.h>
using namespace std;
const int N=105;
bool vis[N][N];
bool dp[N][N];
string str[N];
int main(){
    int n;
    cin>>n;
    for(int i=1;i<=n;i++)cin>>str[i];
    for(int i=1;i<=n;i++){
        dp[i][i]=1;
        for(int j=i;j>=1;j--){
            if(j!=i)vis[j][i]=1;
            for(int k=j;k<i;k++){
                if(dp[j][k]==1 and vis[k+1][i]==1){
                    dp[j][i]=1;vis[j][i]=1;
                }
            }
            if((j+1)==i){
                for(int k=i+1;k<=n;k++){
                    if(dp[k][k]==1 and str[i]==str[k].substr(0,str[i].size()))dp[j][i]=1,vis[j][i]=1;
                }
            }
        }
    }
    bool ok=0;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            if(i==j)continue;
            ok|=(dp[i][j]==1);
        }
    }
    if(!ok){
        cout<<"NO";
        return 0;
    }
    cout<<"YES"<<endl;
    for(int i=1;i<=n;i++)cout<<str[i]<<endl;
}
