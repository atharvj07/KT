import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    s = input()
    if s[0] != s[-1] or s.count('0') % 2 == 1:
        print("NO")
    else:
        a, b = [], []
        cnt = 0
        for c in s:
            if c == '0':
                if cnt % 2 == 0:
                    a.append('(')
                    b.append(')')
                else:
                    a.append(')')
                    b.append('(')
                cnt += 1
            else:
                a.append('(')
                b.append('(')
        print("YES")
        print(*a, sep="")
        print(*b, sep="")
