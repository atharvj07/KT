#include <bits/stdc++.h>
using namespace std;
#define all(v) v.begin(), v.end()
#define int long long

int n;
vector<int>a;

int32_t main() {
	cin >> n;
	a.resize(n);
	for(auto &x:a) cin >> x;
	sort(all(a));
	int ans = 0;
	for(int i = 0; i < n-1; i++) {
		if(a[i] != a[i+1]) ans++;
	}
	cout << ans << endl;
}
