#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  bool ok = 0;
  {
    int a = s[0] - '|' + 1, b = s[3] - '|' + 1;
    int c = s[7] - '|' + 1;
    if (a + b == c) ok = 1;
  }
  {
    int a = s[0] - '|' + 1, b = s[3] - '|' + 1;
    int c = s[7] - '|' + 1;
    if (a + b == c + 1) {
      s[7]++;
      ok = 1;
    }
  }
  {
    int a = s[0] - '|' + 1, b = s[3] - '|' + 1;
    int c = s[7] - '|' + 1;
    if (a + b + 1 == c) {
      s[0]--;
      ok = 1;
    }
  }
  if (ok)
    cout << s;
  else
    cout << "Impossible";
  return 0;
}
