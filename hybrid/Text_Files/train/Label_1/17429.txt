#include<bits/stdc++.h>
#define int long long int
#define MOD 1000000007
using namespace std;
int32_t main()
{
    int n, i, a[100001], x = 0, ans = 1;
    map<int, int> mp;
    cin >> n;
    for(i = 1; i <= n; i++)
    {
        cin >> a[i];
        x |= a[i];
        mp[a[i]]++;
    }
    if(x == 1)
    {
        int n1 = 0;
        for(i = 1; i <= n; i++)
            if(a[i] != 1) n1++;
        cout << (1 << n1);
        return 0;
    }
    for(auto u : mp)
    {
        int gc = __gcd(ans, u.first);
        ans = (ans * u.second) / gc;
        ans = ans % MOD;
    }
    cout << ans;
    return 0;
}
