import java.util.*;

class Main {
    static void solve() {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) p[i] = sc.nextInt();
        int[] prv = new int[n + 1];
        int[] nxt = new int[n + 1];
        for (int i = 1; i <= n; i++) prv[i] = i - 1;
        for (int i = 0; i + 1 < n; i++) nxt[i] = i + 1;
        List<Integer> ans = new ArrayList<>();
        for (int i = n - 1; i >= 0; i--) {
            int j = i + k;
            if (j < n && p[j] == p[i] - 1) {
                j = nxt[j];
                p[j] = p[i];
                p[i] = p[j] + 1;
                nxt[prv[j]] = nxt[j];
                prv[nxt[j]] = prv[j];
                prv[j] = i;
                nxt[i] = j;
            }
            ans.add(p[i]);
        }
        Collections.reverse(ans);
        for (int x : ans) System.out.println(x);
    }

    public static void main(String[] args) {
        solve();
    }
}
