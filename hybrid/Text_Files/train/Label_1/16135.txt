#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<string> strs(n);
  for (auto &s : strs) { cin >> s; }

  vector<vector<int>> costs(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) { cin >> costs[i][j]; }
  }

  vector<map<char, int>> char_counts(m);
  int ans = 0;

  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      auto &mp = char_counts[j];
      if (mp[strs[i][j]] == 1) { ans += costs[i][j]; }
      mp[strs[i][j]]++;
    }
  }

  cout << ans << endl;

  return 0;
}
