import sys
import math

L = [3,4,6,8,12,14,18,20,24,30,36,40,48,54,60,72,84,90,108,126]

def print_result(x1, y1, x2, y2, val):
    for dy in range(2):
        for dx in range(2):
            for j in range(val):
                nx = x1 + j*dx
                ny = y1 + j*dy
                print(nx, ny)

n = int(sys.stdin.readline())
x = 0
y = 0
flag = False

for i in range(n):
    x, y = int(math.sqrt(n*3+L[i])), y + 1
    if L[i] >= n:
        print_result(0, 0, x+1, y+1, n)
        flag = True
        break
    if L[i + 1] < n:
        print_result(0, y, x+1, y+1, n-L[i])
        flag = True
        break

if not flag:
    print_result(-1, -1, -1, -1, -1)
