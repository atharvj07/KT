#include<bits/stdc++.h>
using namespace std;
#define int long long
#define vi vector<int>
#define mod 1000000007
#define pb push_back
#define SZ 25
int a[SZ][SZ], dp[SZ][SZ][2050], n, m, k;
int go(int x, int y, int sum)
{
    if(x == n && y == m) return sum == k;
    if(dp[x][y][sum] != -1) return dp[x][y][sum];
    int &ret = dp[x][y][sum];
    ret = 0;
    if(x+1<=n) ret = (ret + go(x+1, y, sum^a[x+1][y]))%mod;
    if(y+1<=m) ret = (ret + go(x, y+1, sum^a[x][y+1]))%mod;
    return ret;
}
int32_t main()
{
    ios::sync_with_stdio(0);
    cin.tie(0);
    int t;
    cin >> n >> m >> k;
    memset(dp, -1, sizeof dp);
    for(int i=1; i<=n; i++)
    {
        for(int j=1; j<=m; j++)
        {
            cin >> a[i][j];
        }
    }
    cout << go(1, 1, a[1][1]) << "\n";
}
