n = int(input())
ls = []
rs = []

for i in range(n):
    l, r = map(int, input().split())
    ls.append(l)
    rs.append(r)

minl = min(ls)
maxr = max(rs)

ok = True
for i in range(n):
    if not (minl <= ls[i] <= rs[i] <= maxr):
        ok = False

if ok:
    print(ls.index(minl) + 1)
else:
    print(-1)
