#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,i,a[1001],mx=-1,id;
    cin >> n;
    for(i=1;i<=n;i++)
    {
        cin >> a[i];
        if(a[i]>=mx)
        {
            mx=a[i];
            id=i;
        }
    }
    mx=-1;
    for(i=1;i<=n;i++)
    {
        if(i==id) continue;
        mx=max(mx,a[i]);
    }
    cout << id << " " << mx << endl;
    return 0;
}
