#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> fav(n);
  for (int i = 0; i < n; i++) {
    int p, q;
    cin >> p;
    fav[i].push_back(p);
    for (int j = 0; j < p; j++) {
      cin >> q;
      fav[i].push_back(q);
    }
    sort(fav[i].begin() + 1, fav[i].end());
  }
  vector<vector<int>> pizza(m);
  for (int i = 0; i < m; i++) {
    int p, q;
    cin >> p;
    pizza[i].push_back(p);
    cin >> q;
    pizza[i].push_back(q);
    for (int j = 0; j < q; j++) {
      cin >> p;
      pizza[i].push_back(p);
    }
    sort(pizza[i].begin() + 2, pizza[i].end());
  }

  int ans1 = -1, ans2 = -1, cnt = 0, cur;
  for (int i = 0; i < m; i++) {
    for (int j = i + 1; j < m; j++) {
      int temp = 0;
      for (int k = 0; k < n; k++) {
        cur = lower_bound(pizza[i].begin() + 2, pizza[i].end(), fav[k][1]) - pizza[i].begin();
        if (cur + 1 <= pizza[i].size() && binary_search(pizza[i].begin() + 2, pizza[i].end(), fav[k][1]))
          temp++;
        cur = lower_bound(pizza[j].begin() + 2, pizza[j].end(), fav[k][1]) - pizza[j].begin();
        if (cur + 1 <= pizza[j].size() && binary_search(pizza[j].begin() + 2, pizza[j].end(), fav[k][1]))
          temp++;
      }
      if (temp > cnt) {
        cnt = temp;
        ans1 = i;
        ans2 = j;
      }
    }
  }
  cout << ans1 + 1 << " " << ans2 + 1 << endl;
  return 0;
}
