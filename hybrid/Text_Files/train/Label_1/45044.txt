import java.util.Scanner;

public class SecondPriceAuction {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read number of bidders
        int n = scanner.nextInt();

        // Read bids
        int[] bids = new int[n];
        for (int i = 0; i < n; i++) {
            bids[i] = scanner.nextInt();
        }

        // Find the winner and the second highest bid
        int winnerIndex = 0;
        int secondHighestBid = -1;
        for (int i = 1; i < n; i++) {
            if (bids[i] > bids[winnerIndex]) {
                secondHighestBid = bids[winnerIndex];
                winnerIndex = i;
            } else if (bids[i] > secondHighestBid && bids[i] != bids[winnerIndex]) {
                secondHighestBid = bids[i];
            }
        }

        // Print the winner and the price he will pay
        System.out.println((winnerIndex + 1) + " " + secondHighestBid);
    }
}
