#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define N 305
ll n, a[N], b[N];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i] >> b[i];

    vector<ll> dp(N, 1e18);
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = a[i]; j < N; j++) { dp[j] = min(dp[j], dp[j - a[i]] + b[i]); }
    }

    if (dp[N - 1] != 1e18) cout << dp[N - 1] << endl;
    else
        cout << -1 << endl;

    return 0;
}
