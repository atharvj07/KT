import java.util.*;

public class Main {
	static long mod = (long) 1e9 + 7;

	static long power(long x, long y) {
		long res = 1;
		while (y > 0) {
			if (y % 2 > 0)
				res = (res * x) % mod;
			x = (x * x) % mod;
			y /= 2;
		}
		return res;
	}

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		String s = sc.next();
		sc.close();
		long[] dp = new long[n + 1];
		Arrays.fill(dp, 0);
		dp[0] = dp[1] = 1;
		for (int i = 2; i <= n; i++) {
			if (s.charAt(i - 1) == '0') {
				dp[i] = dp[i - 1];
			} else {
				dp[i] = (dp[i - 1] + dp[i - 2]) % mod;
			}
		}
		long[] pref = new long[n + 1];
		pref[0] = 1;
		for (int i = 1; i <= n; i++) {
			pref[i] = (pref[i - 1] * dp[i]) % mod;
		}
		long[] suff = new long[n + 1];
		suff[n] = 1;
		for (int i = n - 1; i >= 0; i--) {
			suff[i] = (suff[i + 1] * dp[i + 1]) % mod;
		}
		long ans = 0;
		for (int k = 2; k <= n; k++) {
			ans = (ans + (suff[k] * pref[n - k]) % mod) % mod;
		}
		System.out.println(ans);
	}
}
