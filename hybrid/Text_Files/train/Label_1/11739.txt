#include <bits/stdc++.h>
#define pb push_back
using namespace std;
typedef long long ll;
typedef pair<int,int> pi;

const int MM = 200001;
int n, k, p[MM], e[MM], dp[MM];
vector<pi> v;
int main() {
    cin>>n>>k;
    for(int i = 1; i <= n; i++) {
        cin>>p[i]>>e[i];
        v.pb({p[i], e[i]});
    }
    sort(v.rbegin(), v.rend());
    int sum = 0;
    for(int i = 0; i < n; i++) {
        if(i+1 > k) dp[i] = -1;
        else dp[i] = max(dp[i], sum);
        if(dp[i] != -1) dp[i+1] = max(dp[i+1], dp[i]+v[i].second);
        sum += v[i].first;
    }
    cout<<dp[n]<<endl;
}
