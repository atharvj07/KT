import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    Line[] lines = new Line[n];

    for (int i = 0; i < n; i++) {
      String[] parts = br.readLine().split(" ");
      int a = Integer.parseInt(parts[0]), b = Integer.parseInt(parts[1]), c = Integer.parseInt(parts[2]);
      lines[i] = new Line(a, b, c);
    }

    int ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        for (int k = j + 1; k < n; k++) {
          Triangle triangle = new Triangle(lines[i], lines[j], lines[k]);
          if (triangle.isOriginal()) ans++;
        }
      }
    }

    System.out.println(ans);
  }

  static class Line {
    int a, b, c;

    Line(int a, int b, int c) {
      this.a = a;
      this.b = b;
      this.c = c;
    }
  }

  static class Triangle {
    Line a, b, c;
    int[][] mat;

    Triangle(Line a, Line b, Line c) {
      this.a = a;
      this.b = b;
      this.c = c;
      mat = new int[][] {{a.a, a.b, -a.c}, {b.a, b.b, -b.c}, {c.a, c.b, -c.c}};
    }

    boolean isOriginal() {
      solve();
      return mat[2][2] != 0;
    }

    private void solve() {
      for (int col = 0; col < 2; col++) {
        pivot(col);
        for (int row = 0; row < 3; row++) {
          if (row == col) continue;
          addMultipleOfRow(row, -mat[row][col], col);
        }
      }
    }

    private void pivot(int col) {
      int pivotRow = col;
      for (int row = col + 1; row < 3; row++) {
        if (Math.abs(mat[row][col]) > Math.abs(mat[pivotRow][col])) {
          pivotRow = row;
        }
      }
      if (pivotRow != col) {
        swapRows(pivotRow, col);
      }
    }

    private void addMultipleOfRow(int row, int multiple, int otherRow) {
      for (int col = 0; col < 3; col++) {
        mat[row][col] += multiple * mat[otherRow][col];
      }
    }

    private void swapRows(int row1, int row2) {
      int[] temp = mat[row1];
      mat[row1] = mat[row2];
      mat[row2] = temp;
    }
  }
}
