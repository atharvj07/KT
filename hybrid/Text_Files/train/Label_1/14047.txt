#include <bits/stdc++.h>
using namespace std;
const int maxn = 205, maxm = 40010;

int n, m, tp;
int in[maxn], out[maxn];
int a[maxn][maxn];
vector<int> g[maxn];

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    memset(in, 0, sizeof(in));
    memset(out, 0, sizeof(out));
    memset(a, 0, sizeof(a));
    for (int i = 1; i <= n; i++) g[i].clear();
    for (int i = 1; i <= m; i++) {
      int u, v;
      cin >> u >> v;
      a[u][v] = a[v][u] = 1;
      in[v]++;
      out[u]++;
      g[u].push_back(v);
      g[v].push_back(u);
    }
    int ans = 0, sum = 0;
    for (int i = 1; i <= n; i++) {
      if (in[i] == out[i]) ans++, sum += in[i];
    }
    cout << ans << endl;
    for (int i = 1; i <= n; i++) {
      for (int j = i + 1; j <= n; j++) {
        if (a[i][j] && in[i] > 0 && out[j] + 1 <= sum) {
          cout << i << " " << j << endl;
          in[i]--;
          out[j]++;
        } else if (a[i][j] && out[i] > 0 && in[j] + 1 <= sum) {
          cout << j << " " << i << endl;
          out[i]--;
          in[j]++;
        }
      }
    }
  }
  return 0;
}
