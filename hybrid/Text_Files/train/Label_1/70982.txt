n, a, b = map(int, input().split())
s = list(map(int, input().split()))
for i in range(n // 2):
    if s[i] != s[n - i - 1] and s[i] != 2 and s[n - i - 1] != 2:
        print(-1)
        exit()
    if s[i] == 0 or s[n - i - 1] == 0:
        a = 0
    if s[i] == 1 or s[n - i - 1] == 1:
        b = 0
print(a + b)
