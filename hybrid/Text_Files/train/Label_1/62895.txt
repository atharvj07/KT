import sys

t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    h = list(map(int, input().split()))
    if h[0] != 0 or h[n-1] != 0:
        print("NO")
        continue
    ok = True
    for i in range(1, n-1):
        if h[i] > k - 1:
            ok = False
    if ok:
        print("YES")
    else:
        print("NO")
