import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long c = sc.nextLong();
        long k = sc.nextLong();

        long t = a + b + c;
        long n = 0;
        for (long i = 0; i < k; ++i) {
            long na = n + c;
            long nb = t - na;
            long nc = t - nb;
            if (Math.abs(na - nb) > 1e18) {
                System.out.println("Unfair");
                return;
            }
            t = na + nb + nc;
            n = nb;
        }
        System.out.println(n - b);
    }
}
