for _ in range(int(input())):
    s, t = input(), input()
    if s == t:
        print(s)
        continue
    lsts, lstt, ans, lcm = 0, 0, "", ""
    while lsts <= len(s):
        lsts += len(s)
        lstt += len(t)
        if lsts == lstt:
            ans = s + t
            while ans == s + t:
                lcm += s + t
                lstt += len(t)
    if lcm:
        print(lcm[lsts - lstt : lsts])
    else:
        print(-1)
