#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f;
int n, x, l, r, ans;
char c;
int main() {
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> x >> c;
    if (c == 'B')
      r = max(r, x);
    if (c == 'R')
      l = max(l, x);
  }
  ans = r == l ? 0 : r - l;
  cout << ans << endl;
  return 0;
}
