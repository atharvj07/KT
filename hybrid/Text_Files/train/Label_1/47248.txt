import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    final int N = input.nextInt();

    int[] prices = new int[N];

    for (int i = 0; i < N; i++) {
      prices[i] = input.nextInt();
    }

    boolean[] buy = new boolean[N];

    for (int i = N - 2; i >= 0; i--) {
      buy[i] = prices[i] < prices[i + 1];
    }

    int profit = 0;
    int shares = 0;

    for (int i = 0; i < N - 1; i++) {
      if (shares > 0 && !buy[i]) {
        profit += prices[i] * shares;
        shares = 0;
      } else if (shares == 0 && buy[i]) {
        shares = -1;
      }
    }

    if (shares != 0) {
      profit += prices[N - 1] * shares;
    }

    System.out.println(profit);
  }
}
