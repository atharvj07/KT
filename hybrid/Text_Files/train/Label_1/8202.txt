#include <bits/stdc++.h>
using namespace std;

int main() {
  long long n, m, k;
  long long lst = 0;
  long long ans = 0;
  cin >> n >> m >> k;
  vector<long long> v(m);
  for (long long i = 0; i < m; i++) { cin >> v[i]; }
  sort(v.begin(), v.end());
  for (auto x : v) {
    lst += ((x - 1) / k) + 1;
    lst = min(lst, ans + 1);
    ans = lst;
  }
  cout << ans << endl;
  return 0;
}
