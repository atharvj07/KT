#include <bits/stdc++.h>
using namespace std;

const int maxn = 2e3 + 5;
int n, a[maxn];
unordered_map<int, vector<int>> mp;

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    mp.clear();
    for (int i = 1; i <= n; ++i) { cin >> a[i]; mp[a[i]].push_back(i); }
    int ans = 0;
    for (int i = 1; i <= n; ++i) {
      for (int j = i + 1; j <= n; ++j) {
        if (__gcd(a[i], 2 * a[j]) > 1) ans++;
        if (__gcd(2 * a[i], a[j]) > 1) ans++;
      }
      if (a[i] % 2 == 0) {
        int tmp = 0;
        for (auto it : mp[a[i] / 2]) { if (it > i) tmp++; }
        ans += tmp;
        tmp = 0;
        for (auto it : mp[2 * a[i]]) { if (it > i) tmp++; }
        ans += tmp;
      }
      for (int j = i + 1; j <= n; ++j) {
        if (a[j] % 2 == 0) {
          int tmp = 0;
          for (auto it : mp[a[j] / 2]) { if (it > i) tmp++; }
          ans -= tmp;
          tmp = 0;
          for (auto it : mp[2 * a[j]]) { if (it > i) tmp++; }
          ans -= tmp;
        }
      }
      for (int j = 2 * a[i]; j <= 1e5; j += 2 * a[i]) { ans += mp[j].size(); }
    }
    cout << ans << endl;
  }
  return 0;
}
