#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  long long mx_w = 0, mx_h = 0;
  while (n--) {
    char c;
    cin >> c;
    if (c == '+') {
      long long w, h;
      cin >> w >> h;
      mx_w = max({mx_w, w, h});
      mx_h = max({mx_h, w, h});
    } else {
      long long w, h;
      cin >> w >> h;
      if (w < mx_w && h < mx_h)
        cout << "NO" << endl;
      else
        cout << "YES" << endl;
    }
  }
}
