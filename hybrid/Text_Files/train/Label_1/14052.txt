#include <iostream>
int main() {
  int t, n, a[51], ans, b[51], l, r, i, st, cnt;
  std::cin >> t;
  while (t--) {
    std::cin >> n;
    cnt = 0;
    st = 0;
    for (i = 1; i <= n; i++) {
      std::cin >> a[i];
      cnt += a[i];
      if (st == 0 && a[i] > 0) st = i;
    }
    if (cnt == n) {
      std::cout << "0\n";
      continue;
    }
    ans = 500000000;
    for (l = st; l <= n; l++) {
      if (a[l] == 0) continue;
      r = l;
      while (r + 1 <= n && a[r] > 0 && a[r + 1] == 0) { r++; }
      if (r + 1 > n || a[r + 1] > 0) {
        ans = std::min(ans, l - st + n - r - 1);
        r++;
        while (r <= n && a[r] > 0) { r++; }
        st = r;
        l = r - 1;
      }
    }
    std::cout << ans << '\n';
  }
}
