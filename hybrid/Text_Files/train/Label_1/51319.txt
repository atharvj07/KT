import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        int s = sc.nextInt();
        int n = (s / 9) * d;
        while (n % d != 0 || sum(n) != s) {
            n += d;
        }
        if (n == 0) {
            System.out.println(-1);
        } else {
            System.out.println(n);
        }
    }

    private static int sum(int n) {
        int sum = 0;
        while (n > 0) {
            sum += n % 10;
            n /= 10;
        }
        return sum;
    }
}
