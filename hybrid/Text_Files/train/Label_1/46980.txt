import java.util.LinkedList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        LinkedList<Integer> police = new LinkedList<>();
        int untreated = 0;
        for (int i = 0; i < n; i++) {
            int recruit = in.nextInt();
            if (recruit == -1) {
                if (police.isEmpty()) {
                    untreated++;
                } else {
                    police.removeLast();
                }
            } else {
                for (int j = 0; j < recruit; j++) {
                    police.addFirst(i);
                }
            }
        }
        System.out.println(untreated);
    }
}
