#include <iostream>
#include <string>
#include <unordered_set>

int main() {
  std::string s;
  std::cin >> s;
  std::unordered_set<std::string> seen;
  int n = s.length();
  for (int i = 0; i < n; ++i) {
    seen.insert(s.substr(i) + s.substr(0, i));
  }
  std::cout << seen.size() << std::endl;
  return 0;
}
