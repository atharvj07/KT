#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n; cin >> n;
    vector<long long> a(n);
    for (auto &e : a) cin >> e;
    vector<long long> mx(n),mn(n),cur(n);
    long long maxe = 0,mine = 1e18,prevmax=0;
    for (int i = 0; i < n; ++i) {
        if(a[i]>maxe)maxe=a[i];
        if(a[i]<mine)mine=a[i];
        mx[i] = maxe;
        mn[i] = mine;
        cur[i] = prevmax;
        if(mx[i]==a[i])prevmax=mx[i];
    }
    for (int i = 0; i < n; ++i) { cout << (mn[i] >= (cur[i] + 1) / 2 ? -1 : n - i + ((cur[i] + 1) % 2 ? n : 0)) << " "; }
    return 0;
}
