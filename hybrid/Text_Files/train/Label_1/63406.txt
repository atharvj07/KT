n=int(input())
a=[list(map(int,input().split()))for i in range(n)]
p=[-1]*n
c=[0]*n
for i in range(n):
 if p[i]==-1:
  x=i
  l=0
  lst=[]
  while p[x]==-1:
   lst+=[x]
   p[x]=i
   x=a[x][0]^x
  for j in lst:
   c[j]=(c[j]+a[j][1]^l)%2
   l^=1
 ans=0
for i in range(n):
 if c[i]==1:ans+=a[i][0]
print(ans//2)
for i in range(n):
 if c[i]==1:
  for j in range(n):
   if i!=j and a[i][0]>0 and a[j][0]>0 and a[i][1]==a[j][1]:
    print(i,j)
    a[i][0]-=1
    a[j][0]-=1
