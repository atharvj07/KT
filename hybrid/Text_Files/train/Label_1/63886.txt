def cal(a, b, c, r):
    if a > b: a, b = b, a
    if c < a or b < c - r: return b - a
    if c - r <= a < c and c < b: return max(b - c + r, 0)
    if a <= c <= b and c + r < b: return b - (c + r)
    return 0

for _ in range(int(input())):
    a, b, c, r = map(int, input().split())
    print(cal(a, b, c, r))
