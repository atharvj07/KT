#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll n, A, cf, cm, m, a[100001];
int main() {
  cin >> n >> A >> cf >> cm >> m;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  sort(a + 1, a + n + 1);
  ll Max = A * cf, Min = a[1] * cm;
  vector<int> ans;
  for (int i = 1; i <= n; i++) {
    ll need = A - a[i];
    if (need <= m) {
      ans.push_back(A);
      Max += cf;
      m -= need;
    } else {
      ll add = m / n;
      Max += add * cf;
      Min += add;
      m -= add * n;
      ans.push_back(a[i] + add);
    }
    if (m == 0) break;
  }
  cout << Max + Min << endl;
  for (auto x : ans) cout << x << " ";
  return 0;
}
