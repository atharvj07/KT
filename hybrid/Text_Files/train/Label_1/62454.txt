n = int(input())
t = [input() for _ in range(n)]
sh, hs = [0 for _ in range(n)], [0 for _ in range(n)]
for i in range(n):
    for j in range(len(t[i])):
        if t[i][j] == 's':
            if i > 0: sh[i] = sh[i-1]
            sh[i] += 1
        if t[i][j] == 'h' and i > 0: hs[i] = max(hs[i-1], sh[i-1])
ans = 0
for i in range(n):
    if i > 0: ans += max(hs[i-1], sh[i])
print(ans)
