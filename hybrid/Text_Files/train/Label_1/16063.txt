#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;

int main() {
    ll n, s;
    cin >> n >> s;
    vector<ll> v(n);
    for (ll i = 0; i < n; i++) cin >> v[i];
    sort(v.begin(), v.end());
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
        if (s <= 0) break;
        ll t = s / n;
        if (t >= v[i]) {
            ans += v[i];
            s -= v[i];
        } else {
            ans += s;
            s = 0;
        }
    }
    if (s <= 0) cout << ans << endl;
    else cout << -1 << endl;
}
