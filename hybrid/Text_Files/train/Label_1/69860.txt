def arrange_students(n):
    if n == 1:
        return [1]
    elif n == 2 or n == 3:
        return [1, 3]
    else:
        odd = [i for i in range(1, n+1, 2)]
        even = [i for i in range(2, n+1, 2)]
        return odd + even

n = int(input())
arrangement = arrange_students(n)
print(len(arrangement))
print(' '.join(map(str, arrangement)))
