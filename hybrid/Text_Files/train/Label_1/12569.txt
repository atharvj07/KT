#include <cstdio>
#include <vector>

using namespace std;

typedef long long ll;
const int MAX_N = 5200 + 10;
#define sc(x) scanf("%lld", &x)
#define pr(x) printf("%lld", x)
#define nl() printf("\n")
constexpr ll MOD = 1e9 + 7;

ll a[MAX_N][MAX_N], p[MAX_N][MAX_N], n, ans = 1;
vector<ll> v;
bool ok(ll x) {
	ll tp = p[n / x][n / x];
	for (ll r = n / x; r < n; ++r) {
		for (ll c = n / x; c < n; ++c) {
			tp -= p[r][c];
			if (tp != a[r][c]) return false;
		}
	}
	return true;
}
int main() {
	sc(n);
	for (ll i = 0; i < n; ++i) {
		ll x;
		sc(x);
		for (ll j = 0; j <= 3; ++j) {
			ll t = x % 16ll;
			v.push_back(t);
			a[i][j + 4 * i] = t;
			x /= 16ll;
		}
	}
	for (ll j = 0; j < n; ++j) {
		ll pref = 0;
		for (ll i = 0; i < n; ++i) {
			pref += a[i][j];
			p[i + 1][j + 1] = pref;
		}
	}
	for (ll x = 1; x * x <= n; ++x) {
		if (n % x == 0) {
			if (ok(x)) ans = x;
		}
		if (n % x == 0 && n / x >= 4) {
			if (ok(n / x)) ans = max(ans, n / x);
		}
	}
	pr(ans); nl();
}
