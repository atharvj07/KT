def solve_chessboard_problem():
    n = int(input().strip())
    board = [list(map(int, input().split())) for _ in range(n)]

    # Calculate sums for all diagonals
    diag1 = [0] * (2 * n - 1)  # from top-left to bottom-right
    diag2 = [0] * (2 * n - 1)  # from top-right to bottom-left
    for i in range(n):
        for j in range(n):
            diag1[i + j] += board[i][j]
            diag2[i - j + n - 1] += board[i][j]

    # Find optimal positions
    pos1 = pos2 = (-1, -1)
    max1 = max2 = -1
    for i in range(n):
        for j in range(n):
            current_sum = diag1[i + j] + diag2[i - j + n - 1] - board[i][j]
            if (i + j) % 2 == 0:
                if current_sum > max1:
                    max1 = current_sum
                    pos1 = (i + 1, j + 1)
            else:
                if current_sum > max2:
                    max2 = current_sum
                    pos2 = (i + 1, j + 1)

    print(max1 + max2)
    print(*pos1, *pos2)

solve_chessboard_problem()
