#include <bits/stdc++.h>
using namespace std;

vector<long long> v;
int main() {
  long long n, sum = 0, a, mx = 0;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a;
    sum += a;
    v.push_back(a);
    mx = max(mx, a);
  }
  if (sum % 2) {
    cout << mx + (sum % 2) << endl;
  } else {
    cout << mx + 1 << endl;
  }
}
