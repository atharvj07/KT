#include <iostream>
#include <cstdio>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        vector<int> pre(n + 1, -1);
        for (int i = 1; i <= n; ++i) {
            if (s[i - 1] == s[i]) {
                if (i == n || s[i] != s[i + 1]) { pre[i + 1] = i; }
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; ++i) {
            ans = max(ans, (pre[i] > -1 ? pre[i] : 0) + 1);
        }
        cout << ans << endl;
    }
    return 0;
}
