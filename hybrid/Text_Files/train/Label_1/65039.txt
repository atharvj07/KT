def get_min(s, k):
    if len(s) < k:
        return "impossible"
    char_count = {}

    for i in s:
        char_count[i] = char_count.get(i, 0) + 1

    diff_count = len(char_count)
    need_count = k - diff_count

    if need_count <= 0:
        return 0

    change_count = 0

    for i in char_count:
        if char_count[i] > 1:
            change_count += char_count[i] - 1
            if change_count >= need_count:
                break

    if change_count >= need_count:
        return need_count
    else:
        return "impossible"


s = input()
k = int(input())
print(get_min(s, k))
