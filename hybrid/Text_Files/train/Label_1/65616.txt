from sys import stdin

n, k = map(int, input().split())
a = [[c == '.' for c in input()] for _ in range(n)]

ans = [-1, -1, 0]

for i in range(n):
	for j in range(n):
		if not a[i][j]: continue
		cnt = 0
		if i + k - 1 < n:
			for y in range(i, i + k):
				cnt += a[y][j]
		if j + k - 1 < n:
			for x in range(j, j + k):
				cnt += a[i][x]
		if cnt > ans[2]:
			ans = [i + 1, j + 1, cnt]

print(*ans[:2])
