#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int cnt = 1;
    set<char> ans;
    ans.insert(s[0]);
    for (int i = 1; i < s.size(); i++) {
      if (s[i] == s[i - 1]) cnt++;
      else
        cnt = 1;
      if (cnt == 1) ans.insert(s[i]);
    }
    for (char c = 'a'; c <= 'z'; c++) {
      if (ans.find(c) != ans.end()) cout << c;
    }
    cout << endl;
  }
  return 0;
}
