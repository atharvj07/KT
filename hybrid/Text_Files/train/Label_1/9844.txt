#include <bits/stdc++.h>
#define ll long long
using namespace std;

ll pre[101][101], dp[101][101][101];
const ll inf = 1e18;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    if (x)
      continue;
    for (int j = 1; j <= m; j++) {
      int y;
      cin >> y;
      pre[j][i + 1] = pre[j][i] + y;
    }
  }
  for (int i = 0; i <= n; i++)
    for (int j = 0; j <= k; j++)
      for (int x = 0; x <= m; x++)
        dp[i][j][x] = inf;
  dp[n][k][0] = 0;
  for (int i = n - 1; i >= 0; i--) {
    for (int j = 0; j <= k; j++) {
      for (int x = 0; x <= m; x++) {
        dp[i][j][x] = dp[i + 1][j][x];
        for (int y = x + 1; y <= m; y++)
          dp[i][j][x] = min(dp[i][j][x], pre[y][i + 1] - pre[x][i + 1] + dp[i + 1][j - 1][y]);
      }
    }
  }
  if (dp[0][k][0] == inf)
    cout << -1 << endl;
  else
    cout << dp[0][k][0] << endl;
  return 0;
}
