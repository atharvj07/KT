#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;

  while (t--) {
    int n;
    cin >> n;

    int a[n];
    for (int i = 0; i < n; i++) { cin >> a[i]; }

    sort(a, a + n);

    int diff = a[n - 1] - a[0];

    int i = 0, j = n - 1;
    while (diff > 0 && i < j) {
      a[i]++;
      a[j]--;
      diff -= 2;

      sort(a, a + n);
      i++;
      j--;
    }

    cout << diff << endl;
  }

  return 0;
}
