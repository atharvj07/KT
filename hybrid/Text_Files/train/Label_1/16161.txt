#include <iostream>
#include <string>

using namespace std;

int main() {
    int n;
    string s;
    cin >> n >> s;
    char prev = '.';
    int cnt = 0, ans = 0;
    for (char c : s) {
        if (c != prev) {
            if (cnt >= 4) ans = 1;
            cnt = 0;
            prev = c;
        }
        if (c == '*') cnt++;
    }
    if (cnt >= 4) ans = 1;
    cout << (ans ? "yes" : "no") << endl;
    return 0;
}
