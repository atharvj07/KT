#include <bits/stdc++.h>
using namespace std;

int main() {
    int cases, n;
    cin >> cases;
    while (cases--) {
        cin >> n;
        vector<int> a(n);
        vector<int> b(n);
        for (auto &x : a) cin >> x;
        for (auto &x : b) cin >> x;
        bool swapped = false;
        int i = 0;
        while (i < n - 1) {
            if (a[i] > a[i + 1]) {
                if (b[i] != b[i + 1]) {
                    swap(a[i], a[i + 1]);
                    swap(b[i], b[i + 1]);
                    swapped = true;
                } else {
                    cout << "No" << endl;
                    goto done;
                }
            }
            ++i;
        }
        if (!swapped) {
            cout << "Yes" << endl;
            goto done;
        }
        i = n - 2;
        while (i >= 0) {
            if (a[i] > a[i + 1]) {
                cout << "No" << endl;
                goto done;
            }
            --i;
        }
        cout << "Yes" << endl;
    done:;
    }
    return 0;
}
