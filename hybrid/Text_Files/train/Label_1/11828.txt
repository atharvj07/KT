#include<bits/stdc++.h>
using namespace std;
int n,m,u,v,l1,l2,s1,t1,s2,t2,ans;
vector<int>adj[3001];
bool vis[3001];
int dist[3001];
queue<int>q;
void bfs(int x,int lim){
    while(!q.empty())q.pop();
    memset(dist, 0, sizeof dist);
    memset(vis, 0, sizeof vis);
    vis[x]=1;
    dist[x]=0;
    q.push(x);
    while(!q.empty()){
        u=q.front(); q.pop();
        for(auto i:adj[u]){
            if(vis[i]) continue;
            q.push(i);
            vis[i]=1;
            dist[i]=dist[u]+1;
            if(dist[i]>lim) return;
        }
    }
    return;
}
int32_t main(){
    cin>>n>>m;
    for(int i=0;i<m;i++){
        cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    cin>>s1>>t1>>l1;
    cin>>s2>>t2>>l2;
    bfs(s1,l1);
    if(!vis[t1]) ans=-1;
    bfs(s2,l2);
    if(!vis[t2]) ans=-1;
    if(ans==-1) cout<<ans;
    else{
        ans=m-(vis[t1]&&vis[t2]);
        cout<<ans;
    }
    return 0;
}
