#include <iostream>
#include <vector>
using namespace std;

int main() {
  vector<string> v(6);
  for (auto &i : v) cin >> i;
  for (int i = 0; i < 6; ++i) {
    for (int j = 0; j < 8; j += 4) {
      if (v[i][j] == '.') {
        v[i][j] = 'P';
        for (auto &k : v) cout << k << endl;
        return 0;
      }
    }
  }
  return 0;
}
