import java.util.*;
import java.io.*;

public class Main {
    static int n, m;
    static char[][] grid;
    static int[][] cnt;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            grid[i] = in.next().toCharArray();
        }
        cnt = new int[n][m];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '#') {
                    if (cnt[i][j] == 0) {
                        ans++;
                        for (int k = 0; k < m; k++) {
                            cnt[i][k]++;
                        }
                        for (int k = 0; k < n; k++) {
                            cnt[k][j]++;
                        }
                    }
                }
                else if (cnt[i][j] == 0) {
                    System.out.println(-1);
                    return;
                }
            }
        }
        System.out.println(ans);
    }
}
