import sys
input = sys.stdin.readline

n, m, s = map(int, input().split())
graph = [[] for _ in range(n+1)]
in_edges = [0] * (n+1)

for _ in range(m):
    u, v = map(int, input().split())
    graph[u].append(v)
    in_edges[v] += 1

vis = [False] * (n+1)
vis[s] = True
q = [s]
while q:
    u = q.pop(0)
    for v in graph[u]:
        if vis[v]: continue
        vis[v] = True
        q.append(v)

cnt = 0
for i in range(1, n+1):
    if in_edges[i] == 0 and not vis[i]:
        cnt += 1

print(cnt)
