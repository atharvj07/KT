import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) throws Exception {
    Scanner in = new Scanner(System.in);
    int N = in.nextInt();
    long[] A = new long[N + 1];
    long[] B = new long[N + 1];
    for (int i = 1; i <= N; i++) {
      A[i] = in.nextLong();
      B[i] = in.nextLong();
    }
    Arrays.sort(A, 1, N + 1);
    Arrays.sort(B, 1, N + 1);
    long ans = 0;
    long sum = 0;
    int cnt = 0;
    for (int i = 1; i <= N; i++) {
      if (sum + A[i] <= B[i]) {
        sum += A[i];
        cnt++;
      } else {
        ans = Math.max(ans, sum + A[i]);
        sum = 0;
        cnt = 1;
      }
    }
    if (sum > 0) ans = Math.max(ans, sum);
    System.out.println(Math.round(ans * 1000.0));
  }
}
