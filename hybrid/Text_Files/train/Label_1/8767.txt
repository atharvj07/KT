#include<bits/stdc++.h>
#define int long long
#define mod 1000000007
using namespace std;
signed main(){
    int n,k;cin>>n>>k;
    k=min(k,n-k);
    int fact[n+1];
    fact[0]=1;
    for(int i=1;i<=n;i++)fact[i]=(i*fact[i-1])%mod;
    int dp[n+1][n+1]={};
    dp[0][0]=1;
    for(int i=1;i<=n;i++){
        for(int j=0;j<=k;j++){
            if(i==j){
                dp[i][j]=((fact[i]*2)%mod+mod)%mod;
                continue;
            }
            dp[i][j]=((dp[i-1][j]%mod)*(i-j)%mod+mod)%mod;
            dp[i][j]=((dp[i][j]%mod)*dp[i-1][j-1]%mod+mod)%mod;
        }
    }
    cout<<dp[n][k]<<endl;
}
