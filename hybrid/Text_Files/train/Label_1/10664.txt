#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long
const ll mod = 1e9 + 7, infLL = 1e18 + 5;
void dfs(int x, int p, int &ans, vector<vi> &adj, vi &vis, vi &dp, vi &a) {
    vis[x] = 1, dp[x] = a[x];
    for (int y : adj[x]) {
        if (y != p) {
            dfs(y, x, ans, adj, vis, dp, a);
            ans = max(ans, dp[y]);
            dp[x] = max(dp[x], dp[x] + dp[y]);
        }
    }
}
int main() {
    int n;
    cin >> n;
    vi a(n), vis(n, 0);
    for (int &x : a) cin >> x;
    vector<vi> adj(n);
    for (int i = 0; i + 1 < n; i++) {
        int x, y;
        cin >> x >> y;
        x--, y--;
        adj[x].push_back(y), adj[y].push_back(x);
    }
    int ans = 0;
    vi dp(n, 0);
    dfs(0, 0, ans, adj, vis, dp, a);
    if (ans <= 0)
        cout << "Impossible\n";
    else
        cout << ans << "\n";
    return 0;
}
