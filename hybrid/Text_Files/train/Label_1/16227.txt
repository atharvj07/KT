#include <bits/stdc++.h>
using namespace std;
struct point {
    int x, y;
    point(int x = 0, int y = 0) : x(x), y(y) {}
};
inline int dist(point u, point v) {
    return abs(u.x - v.x) + abs(u.y - v.y);
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        point A, B, F;
        cin >> A.x >> A.y >> B.x >> B.y >> F.x >> F.y;
        cout << min({dist(A, B), dist(A, F) + dist(F, B),
                      dist(A, {F.x, B.y}) + dist({F.x, B.y}, B),
                      dist(A, {B.x, F.y}) + dist({B.x, F.y}, B)})
             << endl;
    }
}
