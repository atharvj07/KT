import java.util.*;
import java.io.*;

public class Main {

	static class Video implements Comparable<Video> {
		int time, len;
		public Video(int time, int len) {
			this.time = time;
			this.len = len;
		}
		public int compareTo(Video other) {
			return time - other.time;
		}
	}

	public static void main(String[] args) throws IOException {
		InputStreamReader isr = new InputStreamReader(System.in);
		BufferedReader in = new BufferedReader(isr);
		StringTokenizer st = new StringTokenizer(in.readLine());
		int n = Integer.parseInt(st.nextToken());
		int k = Integer.parseInt(st.nextToken());
		PriorityQueue<Video> queue = new PriorityQueue<Video>();
		PriorityQueue<Integer> ends = new PriorityQueue<Integer>();
		for (int i = 0; i < n; i++) {
			st = new StringTokenizer(in.readLine());
			int time = Integer.parseInt(st.nextToken());
			int len = Integer.parseInt(st.nextToken());
			queue.add(new Video(time, len));
		}
		while (!queue.isEmpty()) {
			Video cur = queue.poll();
			while (!ends.isEmpty() && ends.peek() <= cur.time) {
				ends.poll();
			}
			if (ends.isEmpty()) {
				ends.add(cur.time + cur.len);
			} else {
				ends.add(ends.peek() + cur.len);
			}
			System.out.println(ends.peek());
		}
	}
}
