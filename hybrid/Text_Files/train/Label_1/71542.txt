import sys

recipe = input()
ing_stock = [int(_) for _ in input().split()]
ing_cost = [int(_) for _ in input().split()]
money = int(input())

ans = min(money // c for c in ing_cost)
for i in recipe:
    if ing_stock[ord(i) - ord('B')]:
        ing_stock[ord(i) - ord('B')] -= 1
    else:
        ans = min(ans, money // ing_cost[ord(i) - ord('B')])

print(ans)
