import java.util.*;

public class Main {
    private static class Point {
        int x, y;
        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
        public double angle() {
            return Math.atan2(y, x);
        }
    }

    public static void main(String[] args) {
        try (var sc = new Scanner(System.in)) {
            int n = sc.nextInt();
            Point[] points = new Point[n];
            for (int i = 0; i < n; i++) {
                points[i] = new Point(sc.nextInt(), sc.nextInt());
            }
            Arrays.sort(points, (a, b) -> {
                double angleA = a.angle();
                double angleB = b.angle();
                if (angleA != angleB) {
                    return Double.compare(angleA, angleB);
                } else {
                    return -Integer.compare(a.x, b.x);
                }
            });
            Stack<Point> convexHull = new Stack<>();
            convexHull.push(points[0]);
            for (int i = 1; i < n; i++) {
                Point top = convexHull.peek();
                while (convexHull.size() >= 2) {
                    Point secondTop = convexHull.get(convexHull.size() - 2);
                    if (isCounterClockwise(secondTop, top, points[i])) {
                        break;
                    } else {
                        convexHull.pop();
                        top = convexHull.peek();
                    }
                }
                convexHull.push(points[i]);
            }
            System.out.println(convexHull.size());
            for (Point point : convexHull) {
                System.out.println(point.x + " " + point.y);
            }
        }
    }

    private static boolean isCounterClockwise(Point a, Point b, Point c) {
        return (c.y - a.y) * (b.x - a.x) > (b.y - a.y) * (c.x - a.x);
    }
}
