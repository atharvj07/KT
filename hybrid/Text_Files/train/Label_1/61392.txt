import sys

t,sx,sy,ex,ey = map(int,input().split())
wind = ['E','S','W','N']
wind_dir = input()
cur_x = sx
cur_y = sy
for i in range(0,t):
    if wind_dir[i] == 'E':
        cur_x += 1
    if wind_dir[i] == 'S':
        cur_y -= 1
    if wind_dir[i] == 'W':
        cur_x -= 1
    if wind_dir[i] == 'N':
        cur_y += 1
    if cur_x == ex and cur_y == ey:
        print(i+1)
        sys.exit()
print(-1)
