#include <iostream>
#include <cstring>
#include <queue>
#include <vector>

using namespace std;

const int MAX = 1001;

int dx[4] = {1, -1, 0, 0};
int dy[4] = {0, 0, 1, -1};

int dis[MAX][MAX];

struct Node {
  int x;
  int y;
  int step;
};

int bfs(int sx, int sy, int bx, int by, int fx, int fy) {
  memset(dis, -1, sizeof dis);
  dis[sx][sy] = 0;
  queue<Node> q;
  q.push({sx, sy, 0});
  
  while(!q.empty()) {
    Node cur = q.front();
    q.pop();
    
    for(int i = 0; i < 4; i++) {
      int nx = cur.x + dx[i];
      int ny = cur.y + dy[i];
      int step = cur.step + 1;

      if(1 <= nx && nx <= 1000 && 1 <= ny && ny <= 1000 && dis[nx][ny] == -1) {
        if(nx == bx && ny == by) return step;
        if(nx != fx || ny != fy) {
          dis[nx][ny] = dis[cur.x][cur.y] + 1;
          q.push({nx, ny, step});
        }
      }
    }
  }
  
  return -1;
}

int main() {
  int t;
  cin >> t;
  
  while(t--) {
    int sx, sy, bx, by, fx, fy;
    cin >> sx >> sy >> bx >> by >> fx >> fy;
    cout << bfs(sx, sy, bx, by, fx, fy) << endl;
  }
}
