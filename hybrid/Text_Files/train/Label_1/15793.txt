#include<bits/stdc++.h>
using namespace std;
int main() {
    long long a, b, ans;
    cin >> a >> b;
    b -= a;
    ans = 1;
    for (int i = 1; i <= b; ++i) {
        ans = ans * (i % 10);
        if (ans % 10 == 0) ans = ans / 10;
        ans = ans % 10;
    }
    cout << ans << endl;
    return 0;
}
