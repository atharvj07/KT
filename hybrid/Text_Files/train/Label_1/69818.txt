import sys
n = int(input())
a = list(map(int, input().split()))
# Suffix array a[i] is the number of balloons when we take
# suffix balloons starting from index i
sa = [0 for j in range(n)]
sa[n - 1] = a[n - 1]
for i in range(n - 2, -1, -1):
    sa[i] = sa[i + 1] + a[i]

cnt = sum(a)
if cnt % 2 == 0:
    print(-1)
    sys.exit()
# If we sort balloons in increasing order, we can take any balloon
# from suffix and then take any subsequence of balloons before it, so that
# total balloons are odd
sa.sort()
idx = -1
# Suffix of an odd number can be in increasing order or decreasing order
# If suffix sums are odd until index i, we can take suffix starting
# from i + 1
for i in range(n - 1):
    if sa[i] < sa[i + 1]:
        idx = i + 1
        break
# If everything is sorted in non-decreasing order, we cannot have
# odd number of balloons
if idx == -1:
    print(-1)
    sys.exit()
print(n - idx)
for i in range(idx, n):
    print(i + 1, end=" ")
