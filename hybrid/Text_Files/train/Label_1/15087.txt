#include <algorithm>
#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k1, k2;
    cin >> n >> k1 >> k2;
    vector<int> a(k1), b(k2);
    for (int &x : a) cin >> x;
    for (int &x : b) cin >> x;
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    if (a[0] > b[k2 - 1])
      cout << "YES\n";
    else
      cout << "NO\n";
  }
}
