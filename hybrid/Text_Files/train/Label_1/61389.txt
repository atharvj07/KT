import sys

t, sx, sy, ex, ey = map(int, input().split())
dir = list(input())
cur_x, cur_y = sx, sy
for s in range(t):
    if dir[s] == "E": cur_x += 1
    if dir[s] == "S": cur_y -= 1
    if dir[s] == "W": cur_x -= 1
    if dir[s] == "N": cur_y += 1
    if ex == cur_x and ey == cur_y: print(s+1); sys.exit()
print(-1)
