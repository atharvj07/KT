import java.util.Scanner;

public class Main {

    private static long FUNC(long f, long T, long t0, long a1, long t1, long p1, long a2, long t2, long p2) {
        long ans = Long.MAX_VALUE;
        if (f % t0 == 0 && f / t0 <= T) {
            ans = 0;
        }
        if (f % t1 == 0 && f / t1 <= T) {
            ans = Math.min(ans, p1 * (f / t1));
        }
        if (f % t2 == 0 && f / t2 <= T) {
            ans = Math.min(ans, p2 * (f / t2));
        }
        for (long i = 0; i * t1 <= T; i++) {
            for (long j = 0; i * t1 + j * t2 <= T; j++) {
                long k = (f - i * t1 - j * t2) / t0;
                if (i * t1 + j * t2 + k * t0 == f) {
                    ans = Math.min(ans, i * p1 + j * p2 + k);
                }
            }
        }
        if (ans == Long.MAX_VALUE) {
            ans = -1;
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long f = sc.nextLong();
        long T = sc.nextLong();
        long t0 = sc.nextLong();
        long a1 = sc.nextLong();
        long t1 = sc.nextLong();
        long p1 = sc.nextLong();
        long a2 = sc.nextLong();
        long t2 = sc.nextLong();
        long p2 = sc.nextLong();
        System.out.println(FUNC(f, T, t0, a1, t1, p1, a2, t2, p2));
    }
}
