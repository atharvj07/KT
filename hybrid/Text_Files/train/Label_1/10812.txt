#include<bits/stdc++.h>
#define vi vector<int>
using namespace std;
const int inf=1e9;
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    //---------------------------------------
    int t;
    cin>>t;
    while(t--){
        int n,k,m;
        cin>>n>>k>>m;
        vi a(n),b(m);
        for(int i=0;i<n;++i) a[i]=i+1;
        for(int &i:b) cin>>i;
        if(n<m||n<k) cout<<"NO\n";
        else{
            int cur=1;
            bool flag=true;
            for(int i:b){
                int cnt=0;
                while(cur<i&&cnt<k){
                    cur++;
                    cnt++;
                }
                if(cur<i||cnt!=k||(cur+cnt-1>n)){
                    flag=false;
                    break;
                }
                cur+=cnt;
            }
            cout<<(flag?"YES":"NO")<<'\n';
        }
    }
    return 0;
}
