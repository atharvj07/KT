n = int(input())
x, y = 0, 0
flag = True
for _ in range(n):
    a, s = input().split()
    a = int(a)
    if s == "North":
        if x != 0:
            flag = False
        else:
            x += a
    elif s == "South":
        if x != 0:
            flag = False
        else:
            x -= a
    elif s == "East":
        if y != 0:
            flag = False
        else:
            y += a
    elif s == "West":
        if y != 0:
            flag = False
        else:
            y -= a
if x == y == 0 and flag:
    print("YES")
else:
    print("NO")
