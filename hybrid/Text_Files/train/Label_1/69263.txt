t = int(input())
for _ in range(t):
    n = int(input())
    candies = list(map(int, input().split()))
    moves = 0
    alice_eaten = 0
    bob_eaten = 0
    alice_turn = True
    alice_index, bob_index = 0, n - 1
    prev_eaten = 0

    while alice_index <= bob_index:
        current_eaten = 0
        if alice_turn:
            while alice_index <= bob_index and current_eaten <= prev_eaten:
                current_eaten += candies[alice_index]
                alice_index += 1
            alice_eaten += current_eaten
        else:
            while bob_index >= alice_index and current_eaten <= prev_eaten:
                current_eaten += candies[bob_index]
                bob_index -= 1
            bob_eaten += current_eaten

        prev_eaten = current_eaten
        moves += 1
        alice_turn = not alice_turn

    print(moves, alice_eaten, bob_eaten)
