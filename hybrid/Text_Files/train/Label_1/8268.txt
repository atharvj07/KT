#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<char>> grid(n, vector<char>(m));
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                cin >> grid[i][j];
        bool valid = true;
        int used_letters = 0;
        vector<pair<pair<int, int>, pair<int, int>>> snakes;
        for (int i = 0; i < n; i++) {
            int start = 0;
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '.')
                    continue;
                char letter = grid[i][j];
                if (used_letters & (1 << (letter - 'a')))
                    valid = false;
                used_letters |= (1 << (letter - 'a'));
                if (j == 0 || grid[i][j - 1] != letter) {
                    start = j;
                }
                if (j == m - 1 || grid[i][j + 1] != letter) {
                    snakes.push_back(make_pair(make_pair(i + 1, start + 1), make_pair(i + 1, j + 1)));
                }
            }
        }
        for (int i = 0; i < m; i++) {
            int start = 0;
            for (int j = 0; j < n; j++) {
                if (grid[j][i] == '.')
                    continue;
                char letter = grid[j][i];
                if (used_letters & (1 << (letter - 'a')))
                    valid = false;
                used_letters |= (1 << (letter - 'a'));
                if (j == 0 || grid[j - 1][i] != letter) {
                    start = j;
                }
                if (j == n - 1 || grid[j + 1][i] != letter) {
                    snakes.push_back(make_pair(make_pair(start + 1, i + 1), make_pair(j + 1, i + 1)));
                }
            }
        }
        if (!valid) {
            cout << "NO\n";
            continue;
        }
        cout << "YES\n";
        cout << snakes.size() << '\n';
        for (auto &snake : snakes) {
            cout << snake.first.first << ' ' << snake.first.second << ' ' << snake.second.first << ' ' << snake.second.second << '\n';
        }
    }
    return 0;
}
