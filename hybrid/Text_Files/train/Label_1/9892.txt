#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> t(n);
    for (int i = 0; i < n; i++) {
        cin >> t[i];
    }
    int T;
    cin >> T;
    sort(t.begin(), t.end());
    int ans = 0;
    int j = 0;
    for (int i = 0; i < n; i++) {
        while (j < n && t[j] - t[i] <= T) {
            j++;
        }
        ans = max(ans, j - i);
    }
    cout << ans << endl;
    return 0;
}
