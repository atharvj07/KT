#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
int n, a[maxn], b[maxn], c[maxn];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) b[i] = a[i], c[i] = 0;
    for (int i = 2; i <= n; i++) {
        while (b[i] % 2 == 0) {
            b[i] /= 2, c[i]++;
        }
        for (int j = i + 1; j <= n; j++) {
            while (b[i] % 2 == 0 && b[i] == b[j]) {
                b[i] /= 2, b[j] /= 2, c[i]++;
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        if (c[i] == 0) {
            cout << 1 << "\n" << a[i] << "\n";
            return 0;
        }
    }
    int sum = 0;
    for (int i = 1; i <= n; i++) sum += c[i];
    cout << sum << "\n";
    for (int i = 2; i <= n; i++) {
        if (c[i] > 0) cout << a[i] << " ";
    }
    cout << "\n";
    return 0;
}
