import sys
readline = sys.stdin.readline
n, k = map(int, readline().split())
slogan = list(readline())
pos = k
time = 0
while pos > 1 and slogan[pos - 1] == slogan[pos - 2]:
    pos -= 1
while pos < n and slogan[pos] == slogan[pos + 1]:
    pos += 1
while k != pos:
    if k > pos:
        time += 1
        k -= 1
        print('LEFT')
    else:
        time += 1
        k += 1
        print('RIGHT')
print('PRINT', slogan[k - 1])
