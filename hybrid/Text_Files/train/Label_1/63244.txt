import sys

input = sys.stdin.readline
n, k, x = map(int, input().strip().split())
a = [int(el) for el in input().strip().split()]

a.sort()

for i in range(n):
    if k == 0:
        break
    if a[i] % x == 0:
        continue
    if a[i] < 0:
        a[i] = (a[i] // x - 1) * x
    else:
        a[i] = (a[i] // x + 1) * x
    k -= 1

print(*a, sep=' ')
