#include <iostream>
#include <vector>
#include <string>

using namespace std;

typedef long long int lli;

int main() {
    int n;
    cin >> n;
    vector<string> strs(n + 1);
    vector<lli> pref(n + 1);
    lli p = 0;
    for (int i = 1; i <= n; i++) {
        cin >> strs[i];
        int len = strs[i].size();
        for (int j = 0; j < len; j++) {
            if (strs[i][j] == 's') p++;
            pref[i] = pref[i - 1] + (strs[i][j] == 'h' ? p : 0);
        }
    }
    lli ans = 0;
    p = 0;
    for (int i = 1; i <= n; i++) {
        int len = strs[i].size();
        for (int j = 0; j < len; j++) {
            if (strs[i][j] == 's') p++;
            ans = max(ans, pref[i - 1] + (strs[i][j] == 'h' ? p : 0) + pref[n] - pref[i]);
        }
    }
    cout << ans << endl;
    return 0;
}
