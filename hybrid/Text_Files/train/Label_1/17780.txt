#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<double> p(n);
    for (double &x : p) cin >> x;
    sort(p.begin(), p.end());
    double ans = 0;
    for (int i = 0; i < n; i++) {
        double prob = 1;
        for (int j = 0; j < n; j++) {
            if (i == j) prob *= p[j];
            else prob *= 1 - p[j];
        }
        ans = max(ans, prob);
    }
    cout << fixed << setprecision(12) << ans << endl;
}
