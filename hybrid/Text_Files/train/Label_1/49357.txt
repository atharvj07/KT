import java.util.*;
class Main {
    static class Edge {
        int other;
        char color;
        Edge(int other, char color) {
            this.other = other;
            this.color = color;
        }
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        List<Edge>[] edges = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            edges[i] = new ArrayList<>();
        }
        for (int i = 1; i <= m; i++) {
            int v = sc.nextInt(), u = sc.nextInt(); char color = sc.next().charAt(0);
            edges[v].add(new Edge(u, color));
            edges[u].add(new Edge(v, color));
        }
        boolean[] vis = new boolean[n + 1];
        int cnt = 0, CC = 0;
        char color = ' ';
        for (int i = 1; i <= n; i++) {
            if (vis[i]) continue;
            Queue<Integer> Q = new LinkedList<>();
            Q.add(i);
            CC++;
            boolean hasR = false, hasB = false;
            while (!Q.isEmpty()) {
                int u = Q.poll();
                vis[u] = true;
                cnt++;
                if (color == ' ') {
                    color = edges[u].get(0).color;
                    hasR = color == 'R'; hasB = color == 'B';
                } else {
                    if ((color == 'R') ^ (edges[u].get(0).color == 'R')) {
                        hasR = true; hasB = true;
                    }
                }
                for (Edge e : edges[u]) {
                    if (vis[e.other]) continue;
                    Q.add(e.other);
                }
            }
            if (!(hasR && hasB)) {
                System.out.println(-1);
                return;
            }
        }
        if (CC == 1) {
            System.out.println(0);
        } else {
            if (cnt % 2 == 0) {
                System.out.println(cnt / 2);
            } else {
                System.out.println(cnt / 2 + 1);
            }
        }
    }
}
