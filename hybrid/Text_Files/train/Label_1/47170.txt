import java.util.*;

class Main {
    static class Edge {
        int u, v;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        Edge[] edges = new Edge[m];
        for (int i = 0; i < m; i++) {
            Edge e = new Edge();
            e.u = sc.nextInt() - 1;
            e.v = sc.nextInt() - 1;
            edges[i] = e;
        }
        int[] deg = new int[n];
        for (Edge e : edges) {
            deg[e.u]++;
            deg[e.v]++;
        }
        int[] odd = new int[n];
        for (int i = 0; i < n; i++) {
            if (deg[i] % 2 != d[i]) {
                odd[i] = 1;
            }
        }
        int[] ans = new int[m];
        int cnt = 0;
        for (int i = 0; i < m; i++) {
            Edge e = edges[i];
            if (odd[e.u] == 1 && odd[e.v] == 1) {
                ans[cnt++] = i + 1;
                odd[e.u] = 0;
                odd[e.v] = 0;
            }
        }
        for (int i = 0; i < m; i++) {
            Edge e = edges[i];
            if (odd[e.u] == 1 || odd[e.v] == 1) {
                ans[cnt++] = i + 1;
                odd[e.u] = 0;
                odd[e.v] = 0;
            }
        }
        if (cnt == m) {
            System.out.println(-1);
        } else {
            System.out.println(cnt);
            for (int i = 0; i < cnt; i++) {
                System.out.println(ans[i]);
            }
        }
    }
}
