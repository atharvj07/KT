#include <bits/stdc++.h>

int main() {
  int k, n;
  std::cin >> k >> n;
  std::vector<std::pair<std::string, int>> v(k);
  for (int i = 0; i < k; ++i) std::cin >> v[i].first >> v[i].second;
  std::sort(v.begin(), v.end());
  std::vector<std::vector<int>> dp(2, std::vector<int>(n + 1, -1e9));
  dp[0][0] = 0;
  for (int i = 0; i < k; ++i) {
    int cur = i & 1, pre = cur ^ 1;
    for (int j = 0; j <= n; ++j) {
      if (j + int(v[i].first.size()) <= n) dp[cur][j + v[i].first.size()] =
          std::max(dp[cur][j + v[i].first.size()], dp[pre][j] + v[i].second);
      dp[cur][j] = std::max(dp[cur][j], dp[pre][j]);
    }
  }
  std::cout << dp[k & 1][n] << std::endl;
  return 0;
}
