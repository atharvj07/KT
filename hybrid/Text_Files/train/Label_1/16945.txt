int dp(int n,int sum)
{
    int dp[n+1][sum+1];
    for(int i=0;i<n+1;i++)
    {
        for(int j=0;j<sum+1;j++)
        {
            if(j==0)  dp[i][j]=0;
            else if(i==0 and j!=0)  dp[i][j]=LLONG_MAX-1;
            else if(j<i) dp[i][j]=dp[i-1][j];
            else dp[i][j]=min((1+dp[i][j-i]),dp[i-1][j]);
        }
    }
    if(dp[n][sum]==LLONG_MAX-1)    return -1;
    return dp[n][sum];
}
