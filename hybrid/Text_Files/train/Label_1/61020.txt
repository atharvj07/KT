t = int(input())
for _ in range(t):
    n = int(input())
    b = input()
    a = '1'
    for i in range(1, n):
        if b[i] == '1':
            if a[i-1] == '1':
                if b[i-1] == '1':
                    a += '0'
                else:
                    a += '1'
            else:
                a += '1'
        else:
            if a[i-1] == '1':
                a += '0'
            else:
                if b[i-1] == '0':
                    a += '1'
                else:
                    a += '0'
    print(a)
