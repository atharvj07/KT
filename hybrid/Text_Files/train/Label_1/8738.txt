#include <bits/stdc++.h>
using namespace std;
#define ll long long
pair<ll, ll> q[200005];
int main() {
    ll n, b, t, d;
    cin >> n >> b;
    ll ans[200005], fr = 1, end = 0;
    for (ll i = 0; i < n; i++)
        cin >> q[i].first >> q[i].second;
    for (ll i = 0; i < n; i++) {
        if (end <= q[i].first)
            ans[i] = q[i].first + q[i].second, end = q[i].first + q[i].second;
        else if (fr <= b)
            ans[i] = end + q[i].second, end = end + q[i].second, fr++;
        else
            ans[i] = -1;
    }
    for (ll i = 0; i < n; i++) cout << ans[i] << endl;
    return 0;
}
