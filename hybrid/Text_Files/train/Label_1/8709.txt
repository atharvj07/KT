#include<bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    int s[n], m[n];
    for (int i=0; i<n; ++i) {
        cin >> s[i];
    }
    for (int i=0; i<n; ++i) {
        cin >> m[i];
    }
    int mn = n, mx = 0;
    sort(s, s+n);
    sort(m, m+n);
    for (int i=0; i<n; ++i) {
        if (s[i] > m[i]) {
            mn = i;
            break;
        }
        if (s[i] < m[i]) {
            mx = i;
        }
    }
    cout << mn << ' ' << mx << endl;
    return 0;
}
