import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static final long mod = 998244353L;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n], b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        Arrays.sort(a);
        Arrays.sort(b);
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += a[i] * b[n - i - 1];
            ans %= mod;
        }
        System.out.println(ans);
    }
}
