#include <cstdio>
#include <iostream>
#include <vector>

using namespace std;

#define MAXN 1010
#define MAXM 1010

vector<int> g[MAXN];
vector<int> gc[MAXN];
int d[MAXN], p[MAXN];
int n, m, s, t;

void dfs(int x, int ds, int from) {
    d[x] = ds;
    p[x] = from;

    for (auto v : g[x]) {
        if (d[v] < 0 && v != from) { dfs(v, ds + 1, x); }
    }
}

int main() {
    scanf("%d %d %d %d", &n, &m, &s, &t);

    for (int i = 0; i < m; i++) {
        int u, v;
        scanf("%d %d", &u, &v);

        g[u].push_back(v);
        g[v].push_back(u);
    }

    dfs(s, 0, -1);

    for (int i = 0; i < n; i++) {
        if (d[i] == d[t]) {
            for (auto v : g[i]) {
                if (d[v] == d[i] - 1) { gc[v].push_back(i); }
            }
        }
    }

    bool check[MAXN] = {};
    int ans = 0;

    for (int i = n - 1; i > 0; i--) {
        if (check[i] == true) continue;

        if (i == t) {
            ans++;
            continue;
        }

        if (gc[i].size() == 0) {
            check[i] = true;
            ans++;
            continue;
        }

        int cnt = 0;
        for (auto v : gc[i]) { cnt += check[v]; }

        if (cnt == 0) {
            ans++;
            check[i] = true;
            continue;
        } else if (cnt == gc[i].size()) {
            continue;
        } else {
            ans++;
            check[i] = true;
        }
    }

    printf("%d\n", ans);

    return 0;
}
