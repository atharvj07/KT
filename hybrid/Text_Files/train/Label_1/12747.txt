#include <bits/stdc++.h>

#define MOD 1000000007

// compute x^y mod MOD
long long fastpow(long long x, long long y) {
  long long t = 1;
  while (y > 0) {
    if (y % 2)
      t = t * x % MOD;
    y /= 2;
    x = x * x % MOD;
  }
  return t;
}

int main() {
  int k;
  std::cin >> k;
  std::vector<int> cnt(k);
  for (int i = 0; i < k; i++) {
    std::cin >> cnt[i];
  }
  std::sort(cnt.begin(), cnt.end());
  long long ans = 1;
  for (int i = 1; i < k; i++) {
    ans = ans * fastpow(i, cnt[i] - cnt[i - 1]) % MOD;
  }
  std::cout << ans << std::endl;
  return 0;
}
