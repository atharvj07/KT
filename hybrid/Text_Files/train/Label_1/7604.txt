#include <iostream>
#include <bits/stdc++.h>
#define int long long
#define endl "\n"
using namespace std;
int32_t main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int sum = 0;
    for (int x : a) sum += x;
    int pref[n + 1] = {};
    pref[0] = 0;
    sort(a.begin(), a.end());
    for (int i = 0; i < n; i++) pref[i + 1] = pref[i] + a[i];
    int ans = 0;
    for (int i = 0; i < n;) {
        int j = i;
        while (j + 1 < n && a[j + 1] == a[i]) j++;
        ans += (j - i + 1) * (pref[j + 1] - pref[i]);
        i = j + 1;
    }
    cout << ans - sum << endl;
}
