n = int(input())
a = list(map(int, input().split()))

def find_expansion_coefficient(a):
    min_value = float('inf')
    for i in range(len(a)):
        for j in range(i+1, len(a)):
            k = min(a[i], a[j]) // abs(i-j)
            min_value = min(min_value, k)
    return min_value

print(find_expansion_coefficient(a))
