#include <bits/stdc++.h>
using namespace std;
#define endl '\n'

int main() {
    int t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        if (n == 3)
            cout << "2\n3 2\n3 2\n";
        else {
            cout << n + 2 << endl;
            for (int i = 4; i <= n; i += 2) cout << i << " " << i - 1 << endl;
            for (int i = 5; i <= n; i += 2) cout << i << " " << i - 2 << endl;
            cout << n << " " << 2 << endl;
            cout << n << " " << 2 << endl;
        }
    }
    return 0;
}
