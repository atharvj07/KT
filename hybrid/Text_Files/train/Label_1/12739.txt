#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, x, cnt = 0, sum = 0;
    cin >> n;
    unordered_map<int, int> mp;
    for (int i = 0; i < n; i++) {
        cin >> x;
        mp[x]++;
    }
    for (auto it : mp) {
        if (it.second > 1) {
            cnt += it.second - 1;
            sum += it.second - 1;
        } else {
            sum++;
        }
    }
    cout << sum - min(sum, cnt) << endl;
    return 0;
}
