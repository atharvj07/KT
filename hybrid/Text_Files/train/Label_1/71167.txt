import sys, heapq

n,m=map(int,input().split())
v=[[]for _ in range(n)]
w=list(map(int,input().split()))
for i in range(m):
    a,b=map(int,input().split())
    a-=1; b-=1
    v[a].append(b)
    v[b].append(a)
q=[(w[0],0)]
used=set()
ans=0
while q:
    now,pre=heapq.heappop(q)
    if now==0: continue
    ans+=now
    used.add(pre)
    for nxt in v[pre]:
        if nxt not in used:
            heapq.heappush(q,(w[nxt],nxt))
print(ans)
