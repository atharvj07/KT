#include <iostream>
#include <vector>

using namespace std;

vector<int> a, prefsum;
int n, k;

int main() {
    cin >> n >> k;
    a.resize(n);
    prefsum.resize(n + 1);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { prefsum[i + 1] = prefsum[i] + a[i]; }
    long long ans = (1LL << 62);
    int pos = -1;
    for (int i = 1; i <= n; i++) {
        long long cur = prefsum[(i + k - 1) % n + 1] - prefsum[i - 1];
        if (cur < ans || cur == ans && i < pos) { ans = cur; pos = i; }
    }
    cout << pos << endl;
    return 0;
}
