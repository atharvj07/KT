from bisect import bisect_left
n, m = map(int, input().split())
l = list(map(int, input().split()))
pos = [[], ] * n
for i in range(m):
    for j in range(l[i]):
        pos[j].append(i)
c = 0
ans = [-1] * m
for i in range(n):
    g = bisect_left(pos[i], c)
    if g < len(pos[i]):
        ans[pos[i][g]] = i
        c = pos[i][g] + 1
print(*ans)
