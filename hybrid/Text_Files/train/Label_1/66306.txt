import sys

input = sys.stdin.readline

n = int(input())
u = list(map(int, input().split()))
L, R, ans, cnt = [0] * 11, [0] * 11, 0, 0
for i in range(n):
    cnt -= R[u[i]]
    L[u[i]] += 1
    R[u[i]] -= 1
    while cnt < 0:
        L[u[ans]] -= 1
        cnt += 2 if u[ans] == u[i] else 1
        ans += 1
    R[u[i]] += 1
    ans += 1
print(ans-1)
