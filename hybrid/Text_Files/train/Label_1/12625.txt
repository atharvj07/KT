#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (n); ++i)

int main() {
  int t;
  std::cin >> t;
  rep(i, t) {
    long long x, y, z;
    std::cin >> x >> y >> z;
    if (x == y && y == z) {
      std::cout << "YES\n" << x << ' ' << y << ' ' << z << '\n';
      continue;
    }
    std::vector<long long> v = {x, y, z};
    std::sort(v.begin(), v.end());
    if (v[0] * 2 == v[1] + v[2]) {
      std::cout << "YES\n" << v[0] << ' ' << v[0] << ' ' << v[1] << '\n';
    } else {
      std::cout << "NO\n";
    }
  }
}
