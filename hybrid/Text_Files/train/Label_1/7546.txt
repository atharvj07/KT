#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  bool ok = true;
  for (int i = 0; i < n; i++) {
    if (a[i] >= a[(i + 1) % n] + a[(i - 1 + n) % n]) { ok = false; }
  }
  if (!ok) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  for (int i = 0; i < n; i++) { cout << a[i] << " "; }
  cout << endl;
  return 0;
}
