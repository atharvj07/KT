#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll gcd(ll a, ll b)
{
	if (!b)
		return a;
	return gcd(b, a % b);
}

ll solve(ll n)
{
	if (n & 1)
		return 1;
	return 0;
}

int main()
{
	ll n;
	cin >> n;
	if (solve(n))
		cout << "No" << endl;
	else
	{
		cout << "Yes" << endl;
		vector<ll> a, b;
		ll sum = 0;
		for (ll i = 2; i <= n; i++)
		{
			if (sum % 2 == 0)
			{
				sum += i;
				a.push_back(i);
			}
			else
			{
				sum += i;
				b.push_back(i);
			}
		}
		cout << a.size() << " ";
		for (ll i = 0; i < a.size(); i++)
			cout << a[i] << " ";
		cout << endl;
		cout << b.size() << " ";
		for (ll i = 0; i < b.size(); i++)
			cout << b[i] << " ";
		cout << endl;
	}
	return 0;
}
