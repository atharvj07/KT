#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;
const int inf = 0x3f3f3f3f;

ll gcd(ll a, ll b) { return b ? gcd(b, a % b) : a; }
ll lcm(ll a, ll b) { return (a / gcd(a, b)) * b; }

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<ll> a(n);
    ll sum = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      sum += a[i];
    }
    vector<ll> b(n);
    ll g = a[0];
    for (int i = 1; i < n; i++) {
      g = gcd(g, a[i]);
    }
    if (sum % (2 * g) == 0) {
      for (int i = 0; i < n; i++) {
        b[i] = (a[i] + sum / (2 * n)) / 2;
      }
    } else {
      for (int i = 0; i < n - 1; i++) {
        b[i] = (a[i] + a[i + 1]) / 2;
      }
      b[n - 1] = sum - accumulate(b.begin(), b.end() - 1, 0LL);
    }
    for (int i = 0; i < n; i++) { cout << b[i] << ' '; }
    cout << '\n';
  }
}
