#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const ll mod = 1e9+7;

ll dfs (int x, int y, vector<vector<char>>& grid, vector<vector<ll>>& dp) {
  if (x == (int) grid.size() || y == (int) grid[0].size() || grid[x][y] == 'R') return 0;
  if (dp[x][y] != -1) return dp[x][y];
  return dp[x][y] = (dfs(x+1, y, grid, dp) + dfs(x, y+1, grid, dp)) % mod;
}

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<char>> grid(n, vector<char>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> grid[i][j];
    }
  }
  vector<vector<ll>> dp(n+1, vector<ll>(m+1, -1));
  dp[n-1][m] = dp[n][m-1] = 1;
  cout << dfs(1,1,grid,dp) << "\n";
  return 0;
}
