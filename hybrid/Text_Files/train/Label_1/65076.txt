m = int(input())
q = list(map(int, input().split()))
n = int(input())
a = list(map(int, input().split()))

a.sort(reverse=True)
q.sort()

min_cost = 0
for qi in q:
    if qi >= n:
        min_cost = sum(a)
        break
    else:
        discount_carts = (n + qi) // (qi + 2)
        paid_items = discount_carts * qi
        free_items = n - paid_items
        if free_items <= 0:
            min_cost = sum(a)
        else:
            min_cost = sum(a[:paid_items]) + sum(a[paid_items:paid_items + free_items])
        break

print(min_cost)
