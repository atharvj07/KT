import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int t = sc.nextInt();
        String st = sc.next();
        char[] ch = st.toCharArray();
        int ans = 0;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (ch[i] == 'H') {
                cnt++;
            } else if (ch[i] == 'S') {
                ans++;
            }
        }
        if (cnt > t) {
            System.out.println("-1");
        } else {
            int rem = t - cnt;
            rem = Math.min(ans, rem);
            System.out.println(ans - rem);
        }
    }
}
