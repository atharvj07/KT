#include <bits/stdc++.h>

using namespace std;

int main() {
  long long n;
  cin >> n;
  vector<long long> lucky;
  for (long long p = 4; p * p <= n; p += 3) {
    long long u = p + 3;
    if (u * u <= n) lucky.push_back(p * u);
  }
  for (long long l = 7; l * l <= n; l += 8) {
    long long u = l + 1;
    if (u * u <= n) lucky.push_back(l * u);
    u = l - 1;
    if (u * u <= n) lucky.push_back(l * u);
  }
  sort(lucky.begin(), lucky.end());
  cout << 1 + lower_bound(lucky.begin(), lucky.end(), n) - lucky.begin() << endl;
  return 0;
}
