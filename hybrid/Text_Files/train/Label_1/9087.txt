#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_map<int, int> m;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    m[x]++;
  }
  int ans = 0, mx = 0;
  for (auto it : m) {
    if (it.second > mx) {
      mx = it.second;
      ans = it.first;
    } else if (it.second == mx) {
      ans = min(ans, it.first);
    }
  }
  cout << ans << endl;
  return 0;
}
