#include <iostream>
#include <string>
#include <vector>
int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n, m;
        std::cin >> n >> m;
        std::string arr;
        std::cin >> arr;
        std::vector<bool> is_alive(n, false);
        for (int i = 0; i < n; i++) is_alive[i] = (arr[i] == '1');
        while (m--) {
            std::vector<bool> next(n, false);
            for (int i = 0; i < n; i++) {
                int alive_neighbors = 0;
                if (i > 0 && is_alive[i - 1]) alive_neighbors++;
                if (i < n - 1 && is_alive[i + 1]) alive_neighbors++;
                next[i] = (alive_neighbors == 1) || is_alive[i];
            }
            is_alive = next;
        }
        for (int i = 0; i < n; i++) arr[i] = (is_alive[i]) ? '1' : '0';
        std::cout << arr << "\n";
    }
    return 0;
}
