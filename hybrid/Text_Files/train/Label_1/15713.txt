#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    if (k > (n - 1) / 2) {
      cout << "-1\n";
      continue;
    }
    int mid = (n + 1) / 2;
    int left = 1, right = n;
    for (int i = 1; i <= k; i++) {
      if (i % 2)
        cout << mid + i - 1 << " ";
      else
        cout << mid - i << " ";
    }
    if (n - 2 * k > 0) {
      for (int i = 1; i <= (n - 2 * k) / 2; i++) {
        cout << left << " " << right << " ";
        right--;
        left++;
      }
    }
    cout << "\n";
  }
}
