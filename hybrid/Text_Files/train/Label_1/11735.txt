#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    int ans = 0;
    for (int i = 0; i < n; ++i) ans += (a[i] == i);
    int bad = -1;
    for (int i = 0; i < n; ++i) {
        if (a[a[i]] == i) continue;
        bad = i;
        break;
    }
    if (bad != -1) ans = max(ans, ans + 2);
    cout << ans << endl;
}
