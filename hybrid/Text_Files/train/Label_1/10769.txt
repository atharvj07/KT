#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  cin >> s;
  int a = 0, b = 0;
  for (const auto& c : s) {
    if (c == 'A') {
      if (++a && b > 0) {
        cout << "YES" << endl;
        return 0;
      }
    }
    if (c == 'B') {
      if (++b && a > 0) {
        cout << "YES" << endl;
        return 0;
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
