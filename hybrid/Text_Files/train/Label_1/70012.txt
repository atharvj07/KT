T = int(input().strip())
for _ in range(T):
    n, m = map(int, input().split())
    edges = [tuple(map(int, input().split())) for _ in range(m)]
    vertex_to_edges = {}
    for i, (u, v) in enumerate(edges, 1):
        if u not in vertex_to_edges:
            vertex_to_edges[u] = []
        if v not in vertex_to_edges:
            vertex_to_edges[v] = []
        vertex_to_edges[u].append(i)
        vertex_to_edges[v].append(i)
    matching = []
    vertices_used = set()
    for u in vertex_to_edges:
        if len(matching) < n and u not in vertices_used:
            for edge_index in vertex_to_edges[u]:
                u, v = edges[edge_index-1]
                if u not in vertices_used and v not in vertices_used:
                    matching.append(edge_index)
                    vertices_used.add(u)
                    vertices_used.add(v)
                    break
    if len(matching) == n:
        print("Matching")
        print(*matching)
    elif len(vertices_used) <= 2 * n:
        ind_set = []
        for u in range(1, 3*n + 1):
            if len(ind_set) < n:
                if u not in vertices_used:
                    ind_set.append(u)
        print("IndSet")
        print(*ind_set)
    else:
        print("Impossible")
