#include <bits/stdc++.h>
using namespace std;

#define int long long

int32_t main() {
  int n, x;
  cin >> n >> x;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = 0, sum = 0, neg = 0, cur = 0;
  for (int i = 0; i < n; i++) {
    cur += a[i];
    if (sum >= 0)
      sum += a[i];
    else if (a[i] < 0)
      neg = 1, sum = a[i];
    else
      sum = a[i];
    ans = max(sum, ans);
  }
  cur = max(cur, 0ll);
  if (!neg && x < 0)
    ans = 0;
  else
    ans = max(ans * x, cur);
  cout << ans << "\n";
  return 0;
}
