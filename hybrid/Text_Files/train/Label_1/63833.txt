import sys
n, m = list(map(int, input().split(' ')))
j = [(input().split() + ['A']) for x in range(n)]
c = [int(input()) for x in range(m)]
c.sort(reverse=True)
j.sort(key=lambda x: int(x[1]))
ans = 0
i = 0
for y in j:
    if int(y[1]) <= 0:
        continue
    x = c[i]
    i += 1
    if y[2] == 'A':
        if x >= int(y[1]):
            ans += x - int(y[1])
        else:
            j.append(y)
            j[-1][1] = x - int(y[1])
    else:
        if x > int(y[1]):
            ans += int(y[1])
        else:
            j.append(y)
            j[-1][1] = x - int(y[1])
print(ans)
