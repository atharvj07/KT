#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

ll a[51];

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  ll sum = 0;
  for (int i = 1; i <= n; i++) sum += a[i];
  sum /= n;
  vector<pair<int, int>> ans;
  int l = 1;
  while (l <= n) {
    int suml = 0;
    for (int r = l; r <= n; r++) {
      suml += a[r];
      if (suml == sum * (r - l + 1)) {
        ans.push_back({l, r});
        l = r + 1;
        break;
      }
    }
  }
  cout << ans.size() << '\n';
  for (auto x : ans) cout << x.first << ' ' << x.second << '\n';
}
