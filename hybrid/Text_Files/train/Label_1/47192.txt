import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] < k) {
                ans += a[i];
                continue;
            }
            if (a[i] - k > 0) {
                ans += a[i] - k;
            }
        }
        System.out.println(ans);
    }
}
