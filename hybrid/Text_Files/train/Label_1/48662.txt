import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] l = new int[n];
        int[] r = new int[n];
        for (int i = 0; i < n; i++) {
            l[i] = sc.nextInt();
            r[i] = sc.nextInt();
        }
        Arrays.sort(l);
        Arrays.sort(r);
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            ans += Math.max(r[i], l[i + 1]) - Math.min(r[i], l[i + 1]);
        }
        System.out.println(ans);
    }
}
