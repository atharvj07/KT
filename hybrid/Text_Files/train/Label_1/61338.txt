n,m,a=map(int,input().split())
b=[int(i) for i in input().split()]
c=[int(i) for i in input().split()]
b.sort(reverse=True)
c.sort()
i=0
j=0
x=0
ans=0
while i<n and j<m and ans+c[j]<=a:
    if b[i]+c[j]<=a:
        x+=1
        ans+=c[j]
        j+=1
    else:
        ans+=a-b[i]
        x+=1
        i+=1
        j+=1
print(x,ans)
