def min_tire_changes(n, k, temperatures):
    negative_streaks = []
    cur_streak = 0
    
    for temp in temperatures:
        if temp < 0:
            cur_streak += 1
        else:
            if cur_streak > 0:
                negative_streaks.append(cur_streak)
            cur_streak = 0
    if cur_streak > 0:
        negative_streaks.append(cur_streak)
    
    if len(negative_streaks) == 0:
        return 0
    
    negative_streaks.sort()
    while k > 0 and negative_streaks:
        if k >= negative_streaks[-1]:
            k -= negative_streaks.pop()
        else:
            break
    
    if not negative_streaks:
        return 2 * len(negative_streaks) + 1
    
    if k == 0:
        return 2 * len(negative_streaks)
    
    required_days = sum(negative_streaks)
    if required_days > k:
        return -1
    else:
        return 2 * len(negative_streaks) + 1

n, k = map(int, input().split())
temperatures = list(map(int, input().split()))
print(min_tire_changes(n, k, temperatures))
