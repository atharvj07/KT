#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int x = 0, y = 0;
  set<pair<int, int>> st;
  for (int i = 0; i < n; ++i) {
    if (s[i] == 'U')
      y++;
    else if (s[i] == 'D')
      y--;
    else if (s[i] == 'L')
      x--;
    else
      x++;
    st.insert({x, y});
  }
  cout << st.size() << endl;
  return 0;
}
