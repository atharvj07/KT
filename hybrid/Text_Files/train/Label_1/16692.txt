#include <iostream>
#include <vector>

const int MOD = 1e9 + 7;

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> l(n), r(n);
  for (int i = 0; i < n; i++) {
    std::cin >> l[i] >> r[i];
  }
  std::vector<int> pref_l(n + 1), suf_r(n + 1);
  for (int i = 0; i < n; i++) {
    pref_l[i + 1] = pref_l[i];
    suf_r[i + 1] = suf_r[i];
    if (l[i] > pref_l[i]) { pref_l[i + 1] = l[i]; }
    if (r[i] < suf_r[i]) { suf_r[i + 1] = r[i]; }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int len = suf_r[i] - pref_l[i] + 1;
    if (len <= 0) continue;
    int coef = 1;
    for (int j = i + 1; j < n && j - i < k; j++) {
      if (pref_l[j] > suf_r[i]) continue;
      if (suf_r[j] < pref_l[i]) continue;
      int cur_len = suf_r[j] - pref_l[i] + 1;
      coef = (1LL * coef * cur_len) % MOD;
    }
    ans = (ans + 1LL * coef * len) % MOD;
  }
  std::cout << ans << std::endl;
  return 0;
}
