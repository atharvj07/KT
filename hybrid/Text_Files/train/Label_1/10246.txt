#include <iostream>

int main() {
    int n;
    std::cin >> n;

    for (int i = 0; i < n; i++) {
        std::string s;
        std::cin >> s;

        bool diverse = true;
        for (int j = 0; j < (int)s.size() - 1; j++) {
            if (s[j + 1] != s[j] + 1) {
                diverse = false;
                break;
            }
        }

        std::cout << (diverse ? "Yes" : "No") << '\n';
    }

    return 0;
}
