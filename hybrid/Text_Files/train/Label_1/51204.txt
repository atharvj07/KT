import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Main {
    private static int n, m, p;
    private static int[] d;
    private static List<Cat> cats;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = br.readLine().split(" ");
        n = Integer.parseInt(tokens[0]);
        m = Integer.parseInt(tokens[1]);
        p = Integer.parseInt(tokens[2]);

        tokens = br.readLine().split(" ");
        d = new int[n];
        for (int i = 1; i < n; i++) {
            d[i] = Integer.parseInt(tokens[i - 1]);
        }

        cats = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            tokens = br.readLine().split(" ");
            int hi = Integer.parseInt(tokens[0]);
            int ti = Integer.parseInt(tokens[1]);
            cats.add(new Cat(hi, ti));
        }

        Collections.sort(cats);

        int[] ans = new int[p];
        int[] time = new int[n + 1];
        for (int i = 0; i < p; i++) {
            int maxTime = -1;
            int idx = -1;
            for (Cat cat : cats) {
                int t = cat.ti + cat.hi - 1;
                int j = i;
                while (j > 0 && time[j - 1] > t) {
                    t += d[j];
                    j--;
                }
                if (t > maxTime) {
                    idx = i;
                    maxTime = t;
                }
            }
            ans[idx] = maxTime;
            for (int j = 0; j <= i; j++) {
                time[j] = maxTime;
            }
        }

        long sum = 0;
        for (Cat cat : cats) {
            int t = cat.ti + cat.hi - 1;
            int i = p - 1;
            while (i >= 0 && ans[i] > t) {
                t += d[i];
                i--;
            }
            sum += t - cat.ti;
        }

        System.out.println(sum);
    }

    private static class Cat implements Comparable<Cat> {
        int hi;
        int ti;

        public Cat(int hi, int ti) {
            this.hi = hi;
            this.ti = ti;
        }

        @Override
        public int compareTo(Cat o) {
            return this.hi - o.hi;
        }
    }
}
