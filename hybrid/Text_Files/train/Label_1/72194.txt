def longest_polyline(n, m):
    if n == 0:
        return [(0, 1), (0, m), (0, 0), (0, m - 1)]
    elif m == 0:
        return [(1, 0), (n, 0), (0, 0), (n - 1, 0)]
    else:
        return [(n, m), (0, 0), (n, 0), (0, m)]

n, m = map(int, input().split())
points = longest_polyline(n, m)
for point in points:
    print(*point)
