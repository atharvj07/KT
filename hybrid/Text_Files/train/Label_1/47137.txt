import java.util.*;

public class Main {
    static final int YEAR_DURATION = 360;
    static final int DAY_PER_MONTH = 30;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<Holiday> holidays = readHolidays(sc, n);
        int[] congestion = calculateCongestion(holidays);
        System.out.println(Arrays.stream(congestion).min().getAsInt());
    }

    private static List<Holiday> readHolidays(Scanner sc, int n) {
        List<Holiday> holidays = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int m = sc.nextInt();
            int d = sc.nextInt();
            int v = sc.nextInt();
            int s = sc.nextInt();
            holidays.add(new Holiday(m, d, v, s));
        }
        return holidays;
    }

    private static int[] calculateCongestion(List<Holiday> holidays) {
        int[] congestion = new int[YEAR_DURATION];
        for (Holiday holiday : holidays) {
            for (int d = holiday.start; d <= holiday.end; d++) {
                congestion[d - 1] = Math.max(congestion[d - 1], holiday.influence(d));
            }
        }
        return congestion;
    }

    private static class Holiday {
        int start, end, influence, days;

        Holiday(int m, int d, int days, int influence) {
            this.start = daysFromBeginningOfYear(m, d);
            this.end = start + days - 1;
            this.influence = influence;
            this.days = days;
        }

        int influence(int d) {
            if (start <= d && d <= end) {
                return influence;
            } else {
                return Math.max(0, influence - Math.min(d - end, start - d));
            }
        }
    }

    private static int daysFromBeginningOfYear(int m, int d) {
        return (m - 1) * DAY_PER_MONTH + d;
    }
}
