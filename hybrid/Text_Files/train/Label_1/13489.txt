#include<bits/stdc++.h>
#include<vector>
using namespace std;
int main() {
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int n, ans = 0; vector<int> v; cin >> n;
    for(int l = 0, r = n;; l++) {
        int c = (l + r) / 2;
        if(r - l <= 1) {
            cout << ans << "\n";
            for(auto p: v) cout << p << " "; cout << "\n";
            exit(0);
        }
        if(c + 1 - l < r - c) v.push_back(l + 1), ans++, r = c;
        else v.push_back(c + 1), ans++, l = c;
    }
}
