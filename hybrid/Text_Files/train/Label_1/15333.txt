#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
#define endl '\n'
int main() {
  int n;
  cin >> n;
  int p[n], b[n];
  bool f[n];
  memset(f, false, sizeof(f));
  int ans = 0;
  for (int i = 0; i < n; i++) {
    cin >> p[i];
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  for (int i = 0; i < n; i++) {
    p[i]--;
    if (p[i] != i) {
      ans += (!f[p[i]] && !f[i]);
      f[p[i]] = true;
      f[i] = true;
    } else if (b[i])
      ans++;
  }
  cout << ans << endl;
  return 0;
}
