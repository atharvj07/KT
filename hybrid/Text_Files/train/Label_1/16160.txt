#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> v;
  for (int i = 0; i < n; i++) {
    if (s[i] == '*') {
      v.push_back(i);
    }
  }
  if (v.size() < 5) {
    cout << "no" << endl;
  } else {
    int dist = v[1] - v[0];
    bool good = true;
    for (int i = 2; i < 5; i++) {
      if (v[i] - v[i - 1] != dist) {
        good = false;
        break;
      }
    }
    if (good) {
      cout << "yes" << endl;
    } else {
      cout << "no" << endl;
    }
  }
  return 0;
}
