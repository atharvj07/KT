#include<bits/stdc++.h>
using namespace std;
bool check(int a,int b,int x)
{
  for (int i = 0; i <= 6; i++)
  {
    int p = (1 << i);
    if ((x & p) != (a & p) && (x & p) != (b & p)) return false;
  }
  return true;
}
int main()
{
  int n,x,y,sum = 0;
  cin >> n;
  x = n / 10;
  y = n % 10;
  for (int i = 0; i <= 9; i++)
  {
    if (check(x, y, i)) sum++;
  }
  for (int i = 0; i <= 9; i++)
  {
    if (check(x, y, 10 * i + x)) sum++;
    if (check(x, y, 10 * i + y)) sum++;
  }
  cout << sum << endl;
}
