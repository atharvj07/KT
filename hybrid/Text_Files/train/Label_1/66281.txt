n,k=map(int,input().split())
a=list(map(int,input().split()))
ans=[]
mx=0
l,r=0,0
for i in range(n):
    if a[i] not in ans:
        if len(ans)<k:
            ans.append(a[i])
        else:
            while len(ans)==k:
                ans.remove(a[l])
                l+=1
            ans.append(a[i])
    r=i
    if len(ans)==k and r-l>mx:
        mx=r-l
        print(l+1,r+1)
