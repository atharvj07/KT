#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b, c, d;
    cin >> a >> b >> c >> d;
    if (a <= b) {
      cout << b << '\n';
    } else if ((b + d) % (c + d) == 0) {
      cout << (b + d) * (a / (c + d)) + b << '\n';
    } else {
      cout << -1 << '\n';
    }
  }
  return 0;
}
