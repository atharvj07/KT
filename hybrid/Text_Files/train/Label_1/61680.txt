def solve():
    n, k, l = map(int, input().split())
    a = sorted(list(map(int, input().split())))
    if n * k != len(a): return 0
    if a[n * k - 1] - a[0] > l: return 0
    ans = a[n * k - 1] - a[n * k - 2]
    if a[n * k - 1] - a[n * k - n * l - 1] <= l: return ans * n
    return 0

print(solve())
