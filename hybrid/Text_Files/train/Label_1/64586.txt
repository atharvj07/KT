n = int(input().strip())
prices = list(map(int, input().strip().split()))

prices.sort()

# Creating the optimal order
if n % 2 == 0:
    # For even n, split the sorted list into two and interleave them
    optimal_order = [0]*n
    optimal_order[::2], optimal_order[1::2] = prices[:n//2], prices[n//2:]
else:
    # For odd n, split such that the first half has one less element, then interleave
    optimal_order = [0]*n
    optimal_order[::2], optimal_order[1::2] = prices[(n//2)+1:], prices[:n//2+1]

# The maximum number of ice spheres Sage can buy is n//3 if n is even, else (n//3)+1
max_ice_spheres = n//3 if n % 2 == 0 else (n//3) + 1

print(max_ice_spheres)
print(' '.join(map(str, optimal_order)))
