import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long p = sc.nextLong(), q = sc.nextLong();
            while (p % q == 0) p /= q;
            System.out.println(p);
        }
    }
}
