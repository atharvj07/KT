import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
            }
            System.out.println(canEqual(a) ? "YES" : "NO");
        }
    }

    private static boolean canEqual(int[] a) {
        int xor = a[0];
        for (int i = 1; i < a.length; i++) {
            xor ^= a[i];
        }
        if (xor == 0) {
            return true;
        }
        for (int i = 0; i < a.length; i++) {
            if ((xor ^ a[i]) < a[i]) {
                return true;
            }
        }
        return false;
    }
}
