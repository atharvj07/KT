import sys
input = sys.stdin.readline

def solve():
    a, b = map(int, input().split())
    s = input().strip()
    n = len(s)
    dp = [[10**9] * 3 for _ in range(n+1)]
    dp[n][0] = dp[n][1] = dp[n][2] = 0
    for i in range(n-1, -1, -1):
        if s[i] == '1':
            dp[i][0] = min(dp[i+1][j] for j in range(3))
            dp[i][1] = min(dp[i+1][j] for j in range(2, 3)) + a
            dp[i][2] = min(dp[i+1][j] for j in range(1, 3)) + b
        else:
            dp[i][0] = dp[i+1][0]
            dp[i][1] = dp[i+1][1]
            dp[i][2] = dp[i+1][2]
    return dp[0][0]

for _ in range(int(input())):
    print(solve())
