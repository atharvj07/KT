import java.util.*;
import java.io.*;

class Main {
    static int N, F[], t[], D, st = 1;

    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        D = sc.nextInt();
        F = new int[N];
        t = new int[N];
        for (int i = 0; i < N; i++) {
            F[i] = sc.nextInt();
            t[i] = sc.nextInt();
            st = Math.max(st, F[i]);
        }
        Arrays.sort(F);
        Arrays.sort(t);
        if (F[0] > t[0]) {
            System.out.println(-1);
            return;
        }
        long ans = 0;
        int j = 0, rem = D;
        for (int i = 0; i < N;) {
            if (F[i] == st) {
                if (t[j] >= st) {
                    ans += st - t[j];
                    rem--;
                    j++;
                } else {
                    ans += st - t[j];
                    rem--;
                    j++;
                    rem--;
                    i++;
                }
            } else {
                int idx = F[i];
                if (t[j] >= idx) {
                    ans += idx - t[j];
                    rem--;
                    j++;
                } else {
                    ans += idx - t[j];
                    rem--;
                    j++;
                    i++;
                }
                st = idx;
            }
            if (rem == 0) {
                rem = D;
                st++;
            }
        }
        System.out.println(ans);
    }
}
