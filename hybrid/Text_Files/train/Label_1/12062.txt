#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;

ll qPow(ll base, ll exp) {
	if (exp == 0) return 1;
	if (exp == 1) return base;
	ll half = qPow(base, exp / 2);
	if (exp % 2 == 0) return (half * half) % mod;
	return ((half * half) % mod * base) % mod;
}

int main() {
	ll f, w, h; cin >> f >> w >> h;
	if (f == 1 && w == 1 && h == 1) {
		cout << 0 << '\n';
		return 0;
	}
	ll total = f + w;
	if (w <= h) {
		cout << 0 << '\n';
		return 0;
	}
	ll num_bad = qPow(2, total - 1 - h);
	num_bad %= mod;
	ll total_num = qPow(2, total - 1);
	total_num %= mod;
	ll ans = ((total_num - num_bad) % mod * qPow(total_num, -1)) % mod;
	cout << ans << '\n';
	return 0;
}
