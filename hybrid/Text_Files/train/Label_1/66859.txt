import sys
read = sys.stdin.readline

n, q = map(int, read().split())
s = read().strip()

l = [0] * 10
lsum = [0] * 10
for i in range(1, n + 1):
  if s[i - 1].isdigit():
    l[int(s[i - 1])] += 1
  lsum[i] = l.copy()

for i in range(q):
  a, b = map(int, read().split())
  print(*list(map(lambda x: lsum[b][x] - lsum[a - 1][x], range(10))))
