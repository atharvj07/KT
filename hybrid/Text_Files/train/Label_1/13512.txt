#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main()
{
    ll N,M,L;
    cin>>N>>M;
    vector<ll>V(M),NV(N,-1);
    for(ll i=0; i<M; i++)
        cin>>V[i],V[i]--;

    for(ll i=0,n=N; i<M; i++)
    {
        NV[V[i]]=n;
        for(ll j=1; j<V[i]; j++)
        {
            if(NV[n-j]==-1)
            {
                bool ok=1;
                for(ll k=0; k<i; k++)
                    if(V[k]==n-j)
                        ok=0;
                if(ok)
                    NV[n-j]=n;
                else
                {
                    cout<<-1<<endl;
                    return 0;
                }
            }
            else
                n=NV[n-j];
        }
        n--;
    }
    for(ll i=0; i<N; i++)
    {
        if(NV[i]==-1)
        {
            cout<<-1<<endl;
            return 0;
        }
        cout<<NV[i]+1<<" ";
    }
    cout<<endl;
    return 0;
}
