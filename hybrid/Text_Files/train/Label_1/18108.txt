#include<bits/stdc++.h>
using namespace std;
int main() {
    int n; cin >> n;
    stack<int>s;
    stack<int>ans;
    for (int i = 0; i < n; i++) {
        int x; cin >> x;
        while (!s.empty() and s.top() <= x) {
            ans.push(s.top());
            s.pop();
        }
        s.push(x);
    }
    while (!s.empty()) {
        ans.push(s.top());
        s.pop();
    }
    while (!ans.empty()) {
        cout << ans.top() << '\n';
        ans.pop();
    }
}
