#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n, a[105]={}, i, j, ans=0, cnt=0, x;
    cin >> n;
    for(i=0;i<n;i++){
        cin >> a[i];
        if(a[i]) cnt++;
    }
    sort(a, a+n);
    for(i=n-1;i>=0;i--){
        if(!a[i]) continue;
        x=a[i];
        ans++;
        for(j=i-1;j>=0;j--){
            if(a[j]>=x) ans++;
        }
        break;
    }
    cout << ans << endl;
    return 0;
}
