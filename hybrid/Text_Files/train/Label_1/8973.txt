#include <iostream>
#include <string>
#include <algorithm>
using namespace std;
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	int n;
	cin >> n;
	string s;
	cin >> s;
	int ans = 0;
	int cnt1 = 0;
	int cnt2 = 0;
	for (int i = 0; i < n; i++) {
		if (s[i] == 'a') cnt1++;
		else cnt2++;
	}
	if (cnt1 != cnt2) {
		if (cnt1 > cnt2) swap(cnt1, cnt2);
		ans = (cnt2 - cnt1) / 2;
	}
	int l = 0, r = n - 1;
	while (l < r) {
		if (s[l] == s[r]) {
			l++;
			r--;
		}
		else {
			if (s[l] == 'a') {
				s[r] = 'a';
				r--;
			}
			else {
				s[l] = 'b';
				l++;
			}
			ans++;
		}
	}
	cout << ans << '\n' << s << '\n';
}
