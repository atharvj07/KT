import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        long[][] a = new long[k][];
        long[] sum = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = new long[sc.nextInt()];
            for (int j = 0; j < a[i].length; j++) {
                a[i][j] = sc.nextInt();
                sum[i] += a[i][j];
            }
        }
        long all = 0;
        for (long l : sum) {
            all += l;
        }
        if (all % 2 != 0) {
            System.out.println("NO");
            return;
        }
        all /= 2;
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < a[i].length; j++) {
                if (all - sum[i] + a[i][j] == sum[i] - a[i][j]) {
                    System.out.println("YES");
                    System.out.println((i + 1) + " " + (j + 1));
                    System.out.println((i + 1) + " " + (j + 1));
                    return;
                }
            }
        }
        for (int i = 0; i < k; i++) {
            for (int j = i + 1; j < k; j++) {
                if (sum[i] == sum[j]) {
                    System.out.println("YES");
                    System.out.println((i + 1) + " " + (a[i].length));
                    System.out.println((j + 1) + " " + 1);
                    return;
                }
            }
        }
        System.out.println("NO");
    }
}
