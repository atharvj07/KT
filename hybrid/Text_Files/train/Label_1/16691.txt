#include <iostream>

using namespace std;

int main() {
  int n, stk = 0, ans = 0;
  string s;
  cin >> s;
  n = s.size();

  for (char c : s) {
    if (c == '[')
      stk++;
    else if (c == ']')
      stk--;
    if (c == '|')
      ans += stk == 1;
    if (stk < 0)
      return cout << "-1", 0;
  }

  if (stk == 0)
    cout << ans * 3 + 2;
  else
    cout << "-1";

  return 0;
}
