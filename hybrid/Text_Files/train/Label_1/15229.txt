#include<bits/stdc++.h>
using namespace std;
vector<pair<int,int>> adj[101];
const int oo=2e9;
vector<int> dis(101,oo);
void dfs(int node,int par)
{
	for(auto child:adj[node])
	{
		if(child.first!=par)
		{
			dis[child.first]=dis[node]+child.second;
			dfs(child.first,node);
		}
	}
}

{

	}

}
