from collections import defaultdict

def assign_companies(n, k, roads):
    tree = defaultdict(list)
    for i, (u, v) in enumerate(roads):
        tree[u].append((v, i))
        tree[v].append((u, i))
    
    visited = [False] * (n + 1)
    company = [0] * (n - 1)
    max_degree = 0
    
    def dfs(v, parent=-1):
        nonlocal max_degree
        visited[v] = True
        companies_used = set()
        for to, idx in tree[v]:
            if to == parent:
                continue
            if visited[to]:
                continue
            if len(companies_used) == max_degree:
                companies_used.clear()
            for c in range(1, max_degree + 2):
                if c not in companies_used:
                    company[idx] = c
                    companies_used.add(c)
                    break
            dfs(to, v)
    
    for v in tree:
        max_degree = max(max_degree, len(tree[v]))
    
    max_degree = min(max_degree, k + 1)
    dfs(1)
    
    return max_degree, company

n, k = map(int, input().split())
roads = [list(map(int, input().split())) for _ in range(n-1)]

r, companies = assign_companies(n, k, roads)
print(r)
print(*companies)
