import sys

n, x = map(int, input().split())
A = []
for _ in range(n):
    A.append(list(map(int, input().split())))

A.sort()
INF = 1e9
ans = INF
for i in range(n):
    for j in range(i + 1, n):
        if min(A[i][1], A[j][1]) >= max(A[i][0], A[j][0]) and A[i][1] - A[i][0] + 1 + A[j][1] - A[j][0] + 1 == x:
            ans = min(ans, A[i][2] + A[j][2])

print(-1 if ans == INF else ans)
