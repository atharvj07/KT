#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> dp(n, 0);
  dp[n - 1] = (s[n - 1] - '0') % 2 == 0;
  for (int i = n - 2; i >= 0; i--) {
    dp[i] = dp[i + 1];
    if ((s[i] - '0') % 2 == 0) { dp[i]++; }
    if (i + 1 < n && (s[i] - '0') * 10 + (s[i + 1] - '0') % 2 == 0) { dp[i]++; }
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) { ans += dp[i]; }
  cout << ans << endl;
  return 0;
}
