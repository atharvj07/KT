#include <bits/stdc++.h>
using namespace std;

int n, d, x[1005], y[1005], ans[1005], par[1005];
vector<int> g[1005];

int getpar(int x) { return (par[x] == x ? x : par[x] = getpar(par[x])); }

void merge(int x, int y) {
  x = getpar(x);
  y = getpar(y);
  if (x != y) par[x] = y;
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n >> d;
  iota(par + 1, par + n + 1, 1);
  for (int i = 1; i <= d; i++) {
    cin >> x[i] >> y[i];
    g[x[i]].push_back(y[i]);
    g[y[i]].push_back(x[i]);
  }
  for (int i = 1; i <= n; i++) {
    for (int j : g[i]) merge(i, j);
    int cnt = 0;
    for (int j = 1; j <= n; j++) {
      if (getpar(i) == getpar(j)) cnt++;
    }
    ans[i] = cnt;
  }
  for (int i = 1; i <= d; i++) cout << ans[i] << ' ';
}
