#include <bits/stdc++.h>
#define int long long
using namespace std;
const int maxn = 1e3 + 1;
bool a[maxn][maxn], vis[maxn][maxn];
struct mov {
    int x, y;
};
mov s[maxn * maxn];
int32_t main() {
    int n, m, k;
    cin >> n >> m >> k;
    for (int i = 0; i < k; i++) {
        int x, y;
        cin >> x >> y;
        s[i] = {x, y};
    }
    for (int i = 0; i <= n + 1; i++) vis[i][0] = vis[i][m + 1] = vis[0][i] = vis[n + 1][i] = true;
    for (int t = 0; t < k; t++) {
        int x = s[t].x, y = s[t].y;
        if (vis[x][y] || vis[x][y - 1] || vis[x][y + 1] || vis[x - 1][y] || vis[x + 1][y]) {
            cout << t + 1 << endl;
            return 0;
        }
        a[x][y] = vis[x][y] = vis[x][y - 1] = vis[x][y + 1] = vis[x - 1][y] = vis[x + 1][y] = true;
    }
    cout << 0 << endl;
    return 0;
}
