#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> ans;
  int cnt = 0;
  bool start = false;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') {
      if (!start) {
        start = true;
        cnt = 1;
        ans.push_back(i + 1);
      } else {
        cnt++;
      }
    } else if (start) {
      if (cnt % 2 == 0) {
        ans.push_back(i);
      }
      start = false;
    }
  }
  if (start) {
    cout << -1 << endl;
  } else {
    int k = ans.size();
    cout << k << endl;
    for (int i = 0; i < k; i++) {
      cout << (ans[i] - ans[i % 2 ? i - 1 : i]) << " ";
      for (int j = ans[i % 2 ? i - 1 : i]; j <= ans[i]; j++) {
        cout << j << " ";
      }
      cout << endl;
    }
  }
  return 0;
}
