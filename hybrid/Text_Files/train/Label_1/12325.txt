#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> l(n), r(n);
    for (int i = 0; i < n; i++) {
        cin >> l[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> r[i];
    }
    bool possible = true;
    for (int i = 0; i < n; i++) {
        if (i > l[i] || n - 1 - i > r[i]) {
            possible = false;
        }
    }
    if (!possible) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    int a[n] = {0};
    int cnt = 1;
    for (int i = 0; i < n; i++) {
        if (i > l[i]) {
            a[i + l[i]] = cnt++;
        } else {
            a[i] = cnt++;
        }
    }
    for (int i = 0; i < n; i++) {
        cout << a[i] << " ";
    }
    cout << endl;
    return 0;
}
