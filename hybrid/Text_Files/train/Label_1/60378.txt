import sys

n = int(sys.stdin.readline())
a = list(map(int, sys.stdin.readline().split()))
ans = 0
for i in range(n):
    if a[i] == 0:
        continue
    j, cnt = i + 1, 0
    while j <= n and a[j - 1] > 0:
        j += 1
        cnt += 1
    if cnt == 1:
        ans += 1
    else:
        ans += 2
    for k in range(i, j):
        a[k] -= 1
print(ans)
