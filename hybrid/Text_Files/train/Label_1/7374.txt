#include <bits/stdc++.h>

using namespace std;

#define MOD 1000000007

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int m;
  cin >> m;
  vector<int> dp(m + 5, 0);
  dp[0] = 1;
  for (int i = 1; i <= m; ++i) {
    int x;
    cin >> x;
    dp[i] = dp[i - 1];
    if (!x) dp[i] = (dp[i] + dp[i - 1]) % MOD;
    if (i >= 2 && !x && !dp[i - 2]) dp[i] = (dp[i] + dp[i - 2]) % MOD;
    if (i >= 3 && x && dp[i - 3]) dp[i] = (dp[i] + dp[i - 3]) % MOD;
  }
  for (int i = 1; i <= m; ++i) cout << dp[i] << "\n";
  return 0;
}
