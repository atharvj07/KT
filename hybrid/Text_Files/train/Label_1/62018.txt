def get_num(num):
    if num == 0:
        return 0
    multi = 1
    while num >= 10**multi:
        multi += 1
    return int(num / 10**(multi - 1))

n = int(input())
cur = n
ans = 1
last = -1
while cur > 0:
    cnt = 0
    if last < get_num(cur):
        cnt += 9
    elif last == get_num(cur):
        cnt += 1
    last = get_num(cur)
    ans *= cnt
    cur %= 10**get_num(cur)
    cur //= 10
print(ans)
