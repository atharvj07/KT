#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
const int mod = 1e9 + 7;
int main() {
    LL p, k;
    cin >> p >> k;
    LL p2 = (p - 1) / 2;
    if (p == 3) {
        cout << 3 << endl;
        return 0;
    }
    if (k == 0) {
        cout << 1 << endl;
        return 0;
    }
    LL ans = 1LL * k * k % mod * p2 % mod * p2 % mod;
    cout << ans << endl;
    return 0;
}
