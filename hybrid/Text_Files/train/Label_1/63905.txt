import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int,input().split()))
ans = []
for i in range(1,n):
    if a[i]!=a[i-1]:
        if a[i]>a[i-1]:
            ans.append([2,i,i+1] )
            a[i]-=a[i-1]
        else:
            ans.append([1,i+1,i])
            a[i]+=a[i+1]-a[i]
print(len(ans))
[print(*x) for x in ans]
