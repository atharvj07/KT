import java.util.*;

class Main {
    private static final int INF = Integer.MAX_VALUE;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<List<Edge>> adjList = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            adjList.add(new ArrayList<>());
        }
        for (int i = 0; i < n; i++) {
            int u = sc.nextInt();
            int k = sc.nextInt();
            for (int j = 0; j < k; j++) {
                int v = sc.nextInt();
                int c = sc.nextInt();
                adjList.get(u).add(new Edge(v, c));
            }
        }
        int[] dist = dijkstra(0, n, adjList);
        for (int i = 0; i < n; i++) {
            System.out.println(i + " " + (dist[i] == INF ? -1 : dist[i]));
        }
    }

    private static int[] dijkstra(int src, int n, List<List<Edge>> adjList) {
        int[] dist = new int[n];
        Arrays.fill(dist, INF);
        dist[src] = 0;
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        pq.add(new Edge(src, 0));
        while (!pq.isEmpty()) {
            Edge edge = pq.poll();
            int u = edge.v;
            int d = edge.w;
            if (d > dist[u]) {
                continue;
            }
            for (Edge e : adjList.get(u)) {
                int v = e.v;
                int w = e.w;
                if (dist[v] > d + w) {
                    dist[v] = d + w;
                    pq.add(new Edge(v, dist[v]));
                }
            }
        }
        return dist;
    }

    private static class Edge implements Comparable<Edge> {
        int v;
        int w;

        public Edge(int v, int w) {
            this.v = v;
            this.w = w;
        }

        @Override
        public int compareTo(Edge other) {
            return this.w - other.w;
        }
    }
}
