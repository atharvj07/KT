#include <bits/stdc++.h>

using namespace std;

int n, a[300005];

int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) { cin >> a[i]; }
    int ans = 0;
    for (int i = 1; i <= n; ++i) {
        int pre = 0;
        for (int j = i + 1; j <= n; ++j) {
            pre ^= a[j];
            int suf = 0;
            for (int k = j + 1; k <= n; ++k) { suf ^= a[k]; }
            if (pre == suf) ++ans;
        }
    }
    cout << ans << endl;
}
