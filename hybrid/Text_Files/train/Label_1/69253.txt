import sys
read = sys.stdin.readline
n, m = map(int, read().split())
adj = [set() for _ in range(n + 1)]
for _ in range(m):
    a, b = map(int, read().split())
    adj[a].add(b)
    adj[b].add(a)
vulnerable = [0] * (n + 1)
for i in range(2, n + 1):
    for j in adj[i]:
        if j > i: vulnerable[i] = 1
q = int(read())
for _ in range(q):
    opt, *qr = map(int, read().split())
    if opt == 1:
        a, b = qr
        adj[a].add(b)
        adj[b].add(a)
        if b > a: vulnerable[a] = 1
        if a > b: vulnerable[b] = 1
    elif opt == 2:
        a, b = qr
        adj[a].remove(b)
        adj[b].remove(a)
        if not adj[a]: vulnerable[a] = 0
        if not adj[b]: vulnerable[b] = 0
    else:
        ans = n
        while True:
            killed = False
            for i in range(n, 0, -1):
                if vulnerable[i] and all(map(lambda x: x > i, adj[i])):
                    killed = True
                    ans -= 1
                    for j in adj[i]:
                        adj[j].remove(i)
                    vulnerable[i] = 0
            if not killed: break
        print(ans)
