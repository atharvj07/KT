import math
n,m=map(int, input().split())
a=list(map(int, input().split()))
l=[0 for i in range(n+2)]
r=[0 for i in range(n+2)]
for i in range(m):
    x,y=map(int, input().split())
    if x==y:
        l[x]+=1
        r[x]+=1
    else:
        l[x]+=1
        r[y+1]-=1
ans=[0,[]]
for i in range(2,n+1):
    l[i]+=l[i-1]
    r[i]+=r[i-1]
    op=0
    for j in range(i):
        if l[j]==0 and r[j]==0:
            op=-1
        elif l[j]<r[j]:
            op+=1
        else:
            op-=1
    tmp0=[]
    if op>ans[0]:
        ans=[op,[]]
    elif op==ans[0]:
        for j in range(i):
            if r[j]==l[j]:
                tmp0.append(j)
        if tmp0 not in ans[1]:
            ans=[op,tmp0]
print(ans[0])
print(len(ans[1]))
print(*ans[1])
