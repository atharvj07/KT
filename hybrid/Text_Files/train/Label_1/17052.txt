#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 998244353, N = 2e5 + 15;
int n, x[N], y[N], st[N], dp[N];

int32_t main() {
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> x[i] >> y[i] >> st[i];
    dp[0] = 1;
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (st[i] == 0) {
            dp[i] = (dp[i - 1] + (x[i] - x[i - 1]) % mod) % mod;
            continue;
        }
        vector<int> pos = {}; // will store the prefix pos from 1 to n (index) of all portals that link to this (i-th) portal
        int temp = dp[i - 1], mn = LLONG_MAX;
        for (int j = 1; j < i; j++) {
            if (y[j] == y[i]) {
                pos.push_back(j);
                temp = (temp - dp[pos[pos.size() - 2]] + mod) % mod;
                mn = min(mn, x[i] - x[pos[pos.size() - 1]]);
            }
        }
        pos.push_back(i);
        if (pos.size() > 1) temp = (temp + dp[pos[pos.size() - 2]]) % mod;
        dp[i] = (temp + (x[i] - x[i - 1]) % mod + (mn - 1) % mod) % mod;
    }
    cout << dp[n] << "\n";
    return 0;
}
