#include <iostream>

using namespace std;
int main() {
  char str[101];
  cin >> str;
  bool consonant = true;
  for (int i = 0; str[i]; i++) {
    if (str[i] != 'n') {
      if (consonant && !(str[i] == 'a' || str[i] == 'e' || str[i] == 'i' ||
                        str[i] == 'o' || str[i] == 'u')) {
        cout << "NO";
        return 0;
      } else if (!consonant && (str[i] == 'a' || str[i] == 'e' || str[i] == 'i' ||
                               str[i] == 'o' || str[i] == 'u')) {
        cout << "NO";
        return 0;
      }
    }
    consonant = !consonant;
  }
  cout << "YES";
  return 0;
}
