#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int x1, y1, x2, y2, x3, y3;
    cin >> x1 >> y1 >> x2 >> y2 >> x3 >> y3;

    vector<int> xs = {x1, x2, x3};
    vector<int> ys = {y1, y2, y3};

    sort(xs.begin(), xs.end());
    sort(ys.begin(), ys.end());

    int ans = 2;
    if (xs[0] == xs[1] || xs[1] == xs[2]) ans--;
    if (ys[0] == ys[1] || ys[1] == ys[2]) ans--;

    cout << ans << endl;

    return 0;
}
