#include <bits/stdc++.h>

using namespace std;

#define ll long long

ll solve() {
  ll n;
  cin >> n;
  string a, b;
  cin >> a >> b;
  ll pos;
  map<char, ll> mp;
  for (ll i = 0; i < n; i++) {
    mp[a[i]] = i;
    if (a[i] != b[i]) { pos = i; }
  }
  if (b[pos] == a[pos + 1] && b[pos + 1] == a[pos]) {
    return 1;
  } else if (b[pos] == a[pos - 1] && b[pos - 1] == a[pos]) {
    return 1;
  } else if (b[pos] == a[pos - 1] && mp[b[pos]] == pos + 1) {
    return 2;
  } else if (b[pos] == a[pos + 1] && mp[b[pos]] == pos - 1) {
    return 2;
  }
  return -1;
}

int main() {
  ll t;
  cin >> t;
  while (t--) { cout << solve() << endl; }
  return 0;
}
