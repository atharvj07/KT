import sys
input = sys.stdin.readline

n, m = map(int, input().split())
g = [[] for i in range(n+1)]
for _ in range(m):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)

vis = [False] * (n+1)
ans = []
def dfs(node, p, dir):
    if vis[node] == True:
        if dir != ans[g[node].index(p)]:
            return False
        return True
    vis[node] = True
    for u in g[node]:
        if u == p: continue
        ans.append(dir)
        if not dfs(u, node, not dir):
            return False
    return True

if not dfs(1, 1, False):
    print("NO")
else:
    print("YES")
    print(''.join(map(str, ans)))
