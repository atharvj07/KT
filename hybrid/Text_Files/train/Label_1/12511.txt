#include<bits/stdc++.h>
using namespace std;

long long num_at_pos(long long pos) {
	long long num = 1;
	while (pos > num * 9 * (long long)pow(10, num - 1)) {
		pos -= num * 9 * (long long)pow(10, num - 1);
		++num;
	}
	long long current_num = (long long)pow(10, num - 1) + (long long)((pos - 1) / num);
	int index = (pos - 1) % num;
	return current_num % 10 / (long long)pow(10, index);
}

int main() {
	long long pos;
	cin >> pos;
	cout << num_at_pos(pos) << endl;
	return 0;
}
