#include <iostream>
#include <vector>

using namespace std;
typedef long long ll;

struct Edge {
    ll from, to;
};

vector<vector<ll>> adj;
vector<Edge> edges;
vector<ll> ans;
vector<bool> vis;
ll n;

void dfs(ll v, ll par) {
    vis[v] = true;
    for (ll u : adj[v]) {
        if (u == par) continue;
        ans.push_back(u);
        dfs(u, v);
    }
}

int main() {
    cin >> n;
    adj.resize(n + 1);
    vis.resize(n + 1, false);
    for (ll i = 0; i < n - 1; i++) {
        ll a, b;
        cin >> a >> b;
        edges.push_back({a, b});
        adj[a].push_back(b);
        adj[b].push_back(a);
    }

    bool ok = true;
    for (Edge e : edges) {
        if (vis[e.from] && vis[e.to]) {
            ok = false;
            break;
        }
        dfs(e.from, -1);
        dfs(e.to, -1);
        if (ans.size() != n - 1) {
            ok = false;
            break;
        }
        for (ll i = 0; i < n - 1; i++) {
            if (edges[i].from != ans[i] && edges[i].to != ans[i]) {
                ok = false;
                break;
            }
        }
        ans.clear();
        for (ll i = 0; i <= n; i++) vis[i] = false;
    }

    if (!ok) {
        cout << "NO" << endl;
    } else {
        cout << "YES" << endl;
        for (Edge e : edges) {
            cout << e.from << " " << e.to << endl;
        }
    }

    return 0;
}
