#include <iostream>
#include <algorithm>
using namespace std;

long long gcd(long long a, long long b) {
    while (b) {
        long long t = a % b;
        a = b;
        b = t;
    }
    return a;
}

int main() {
    long long a, b, c, d;
    cin >> a >> b >> c >> d;
    long long ratio = a * d / b / c;
    long long g = gcd(ratio, a * d);
    cout << (ratio / g) << "/" << (a * d / g) << endl;
}
