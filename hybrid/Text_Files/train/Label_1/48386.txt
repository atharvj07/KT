import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] ts = new int[n][2];
        for (int i = 0; i < n; i++) {
            ts[i][0] = sc.nextInt(); // price
            ts[i][1] = sc.nextInt(); // quality
        }
        int k = sc.nextInt();
        int[] bs = new int[k];
        for (int i = 0; i < k; i++) {
            bs[i] = sc.nextInt(); // budget
        }
        Arrays.sort(ts, (a, b) -> {
            if (a[1] != b[1]) return b[1] - a[1]; // quality
            return a[0] - b[0]; // price
        });
        for (int i = 0; i < k; i++) {
            int cnt = 0;
            for (int j = 0; j < n; j++) {
                if (bs[i] < ts[j][0]) break;
                cnt++;
                bs[i] -= ts[j][0];
            }
            System.out.print(cnt + " ");
        }
        System.out.println();
    }
}
