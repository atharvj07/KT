import java.util.*;
import java.math.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();

    int[][] dp = new int[n + 1][k + 1];

    // base case
    dp[0][0] = 1;
    for (int i = 1; i <= n; i++) {
      dp[i][0] = BigInteger.valueOf(m).pow(i).mod(new BigInteger("998244353")).intValue();
    }

    // dp transition
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= k; j++) {
        dp[i][j] = (dp[i - 1][j - 1] + dp[i - 1][j] * (m - 1)) % 998244353;
      }
    }

    System.out.println(dp[n][k]);
  }
}
