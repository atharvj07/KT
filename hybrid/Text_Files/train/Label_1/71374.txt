def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def find_pair(x):
    a = 1
    b = x
    for i in range(1, int(x**0.5) + 1):
        if x % i == 0:
            if gcd(i, x//i) == 1:
                a, b = i, x//i
                break
    return a, b

x = int(input())
a, b = find_pair(x)
print(a, b)
