#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  if (n == 1) {
    cout << -1 << endl;
    return 0;
  }
  long long d = a[1] - a[0];
  for (int i = 2; i < n; i++) {
    if (a[i] - a[i - 1] != d) {
      cout << 0 << endl;
      return 0;
    }
  }
  long long l = a[0] - d, r = a[n - 1] + d;
  int ans = 0;
  for (long long x = l; x <= r; x += d) {
    if (x <= 0) continue;
    bool found = false;
    for (int i = 0; i < n; i++) {
      if (x == a[i]) found = true;
    }
    if (!found) {
      ans++;
      cout << x << ' ';
    }
  }
  cout << endl;
  return 0;
}
