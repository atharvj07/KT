n,m,k=[int(i)for i in input().split()]
s=[[int(i)for i in input().split()]for i in range(k)]
dir=[[0,1],[1,0],[0,-1],[-1,0]]
chk=[[0 for j in range(m)]for i in range(n)]
for i in s:
    chk[i[0]][i[1]]=1
l=1
x,y=0,0
while 1:
    nxt=[(x+dir[l][0])%n,(y+dir[l][1])%m]
    if ( (nxt[0]==n-1 and nxt[1]==m) or 
         (nxt[0]==0 and nxt[1]==0) or 
         (nxt[0]==0 and nxt[1]==m-1) or 
         (nxt[0]==n-1 and nxt[1]==0) ):
        break
    if chk[nxt[0]][nxt[1]]==1:
        print(l)
        chk[nxt[0]][nxt[1]]=0
    x,y=nxt
    l=(l+1)%4
for i in chk:
    for j in i:
        if j!=0:
            print(-1)
