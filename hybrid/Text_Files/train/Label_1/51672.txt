import java.util.*;

public class Main {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		Target[] targets = new Target[n];
		for (int i = 0; i < n; i++) {
			targets[i] = new Target(in.nextInt(), in.nextInt());
		}
		int m = in.nextInt();
		Shot[] shots = new Shot[m];
		for (int i = 0; i < m; i++) {
			shots[i] = new Shot(in.nextInt(), in.nextInt());
		}
		int numHits = 0;
		int[] hitTimes = new int[n];
		Arrays.fill(hitTimes, -1);
		for (Shot shot : shots) {
			for (int i = 0; i < n; i++) {
				if (targets[i].isHit(shot)) {
					if (hitTimes[i] == -1) {
						numHits++;
						hitTimes[i] = shot.index;
					}
					break;
				}
			}
		}
		System.out.println(numHits);
		for (int hitTime : hitTimes) {
			System.out.print(hitTime + " ");
		}
		in.close();
	}
}

class Target {
	int x;
	int r;

	public Target(int x, int r) {
		this.x = x;
		this.r = r;
	}

	public boolean isHit(Shot shot) {
		return (x - shot.x) * (x - shot.x) + (shot.y) * (shot.y) <= r * r;
	}
}

class Shot {
	int index;
	int x;
	int y;

	public Shot(int x, int y) {
		this.index = x;
		this.x = x;
		this.y = y;
	}
}
