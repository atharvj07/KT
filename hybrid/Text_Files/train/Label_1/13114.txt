#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    int ans = 0;
    int last = *min_element(a.begin(), a.end());
    for (int i = n - 1; i >= 0; --i) {
        if (a[i] >= last) {
            last = a[i];
        } else {
            ans++;
        }
    }

    if (ans > n) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }

    return 0;
}
