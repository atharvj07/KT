#include<bits/stdc++.h>
using namespace std;
#define int long long
const int INF = 1e18;

int32_t main() {
    int n; cin >> n;
    string s; cin >> s;
    map<char, int> cnt;
    for(char ch : s) cnt[ch]++;
    vector<int> freq;
    for(auto p : cnt) freq.push_back(p.second);
    sort(freq.begin(), freq.end());
    int ans = 1;
    for(int i = 0; i < freq.size(); i++) {
        if(ans < freq[i]) ans = freq[i];
    }
    cout << ans << '\n';
    for(char ch : s) {
        if(ans == 1) cout << 1 << ' ';
        else {
            if(cnt[ch] == ans) cout << 2 << ' ';
            else cout << 1 << ' ';
            cnt[ch]--;
        }
    }
    cout << '\n';
}
