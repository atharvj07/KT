import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        Map<Integer, Integer> playerMinute = new HashMap<>();
        String home = scanner.nextLine();
        String away = scanner.nextLine();
        int n = Integer.parseInt(scanner.nextLine());
        for (int i = 0; i < n; i++) {
            int t = scanner.nextInt();
            char h = scanner.next().charAt(0);
            int m = scanner.nextInt();
            char y = scanner.next().charAt(0);
            if (y == 'r') {
                System.out.println(h == 'h' ? home : away + " " + m + " " + t);
            } else if (h == 'h') {
                if (playerMinute.computeIfAbsent(m, k -> 0) == 1) {
                    System.out.println(home + " " + m + " " + t);
                } else {
                    playerMinute.put(m, playerMinute.get(m) + 1);
                }
            } else {
                if (playerMinute.computeIfAbsent(m, k -> 0) == 1) {
                    System.out.println(away + " " + m + " " + t);
                } else {
                    playerMinute.put(m, playerMinute.get(m) + 1);
                }
            }
        }
    }
}
