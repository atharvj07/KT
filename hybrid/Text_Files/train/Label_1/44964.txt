import java.util.Scanner;

public class ArrayGame {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            int m = scanner.nextInt();
            int k = scanner.nextInt();

            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }

            int[][] dp = new int[n][k + 1];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j <= k; j++) {
                    dp[i][j] = -1;
                }
            }

            int result = solve(a, m, k, 0, dp);
            System.out.println(result);
        }
    }

    private static int solve(int[] a, int m, int k, int i, int[][] dp) {
        if (i == m) {
            return 0;
        }

        if (dp[i][k] != -1) {
            return dp[i][k];
        }

        int result = Integer.MIN_VALUE;
        if (k > 0) {
            result = Math.max(result, solve(a, m, k - 1, i + 1, dp) + a[i]);
            result = Math.max(result, solve(a, m, k - 1, i + 1, dp) + a[a.length - 1 - i]);
        } else {
            result = Math.max(result, solve(a, m, k, i + 1, dp));
        }

        dp[i][k] = result;
        return result;
    }
}
