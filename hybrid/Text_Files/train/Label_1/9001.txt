#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef long long ll;

int main() {
  int n;
  ll sum = 0;
  cin >> n;
  vector<int> nums(n);
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    int num = 0;
    int base = 1;
    for (int j = s.size() - 1; j >= 0; j--) {
      num += (s[j] - 'a') * base;
      base *= 10;
    }
    nums[i] = num;
    sum += nums[i];
  }
  sort(nums.begin(), nums.end());
  ll ans = sum;
  for (int i = 0; i < n; i++) {
    int cur = nums[i];
    int carry = 0;
    ll new_num = 0;
    while (cur > 0 || carry > 0) {
      new_num *= 10;
      new_num += (carry + cur % 10) % 10;
      carry = (carry + cur % 10) / 10;
      cur /= 10;
    }
    ans = min(ans, sum - nums[i] + new_num);
  }
  cout << ans << endl;
  return 0;
}
