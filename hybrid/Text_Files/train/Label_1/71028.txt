def expected(node, l, r, ans, bit):
    if ans and (l | (1<<bit-1)) != r:
        return 2
    elif not ans and (l | (1<<bit-1)) == r:
        return 2
    else:
        if ans:
            return node + (1<<(bit-1))
        else:
            return node 


h, q = map(int, input().split())
node = 1
bit = 1
ans = []
for i in range(q):
    tmp = list(map(int, input().split()))
    ans.append(tmp)
    tmp[3] ^= 1
    node = expected(node, tmp[1], tmp[2], tmp[3], tmp[0])
    if node == 2:
        print('Game cheated!')
        return

for tmp in ans:
    tmp[3] ^= 1
    if expected(node, tmp[1], tmp[2], tmp[3], tmp[0]) != node:
        print('Data not sufficient!')
        return
