import java.util.*;
import java.io.*;

public class Main {
    private static final Scanner scanner = new Scanner(System.in);
    private static final PrintWriter writer = new PrintWriter(System.out);

    public static void main(String[] args) {
        int[] calories = new int[4];
        Arrays.setAll(calories, i -> scanner.nextInt());
        String rectangles = scanner.nextLine();
        int totalCalories = 0;
        for (char c : rectangles.toCharArray()) {
            totalCalories += calories[Character.getNumericValue(c) - 1];
        }
        writer.println(totalCalories);
        scanner.close();
        writer.close();
    }
}
