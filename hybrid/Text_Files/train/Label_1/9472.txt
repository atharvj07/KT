#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int inc = 0, dec = 0;
    for (int i = 0; i < n; i++) {
        if (i < n - 1 && a[i] == a[i + 1]) {
            cout << "NO" << endl;
            return 0;
        }
        if (i == 0 || a[i] > a[i - 1]) {
            inc++;
        }
        if (i == 0 || a[i] < a[i - 1]) {
            dec++;
        }
    }
    cout << "YES" << endl;
    cout << inc << endl;
    for (int i = 0; i < n; i++) {
        if (i < n - 1 && a[i] == a[i + 1]) {
            cout << a[i] << " ";
        } else if (i == 0 || a[i] > a[i - 1]) {
            cout << a[i] << " ";
        }
    }
    cout << endl;
    cout << dec << endl;
    for (int i = 0; i < n; i++) {
        if (i < n - 1 && a[i] == a[i + 1]) {
            cout << a[i] << " ";
        } else if (i == 0 || a[i] < a[i - 1]) {
            cout << a[i] << " ";
        }
    }
    cout << endl;
    return 0;
}
