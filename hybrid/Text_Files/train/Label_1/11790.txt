#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ld long double
#define vi vector<int>
#define vll vector<long long>
#define pii pair<int, int>
#define pll pair<long long, long long>
#define vii vector<pair<int, int>>
#define vllv vector<vector<long long>>
#define pb push_back
#define ff first
#define ss second
#define mp make_pair
#define endl "\n"

void solve() {
    int n, m;
    cin >> n >> m;
    vi b(n), g(m);
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    for (int i = 0; i < m; i++) { cin >> g[i]; }
    sort(b.begin(), b.end());
    sort(g.rbegin(), g.rend());

    ll ans = 0;
    int i = 0, j = 0;
    while (i < n && j < m) {
        if (b[i] <= g[j]) {
            ans += b[i];
            i++;
            j++;
        } else {
            ans = -1;
            break;
        }
    }
    if (i < n || j < m) ans = -1;
    cout << ans << endl;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    // cin >> t;
    t = 1;
    while (t--) { solve(); }
}
