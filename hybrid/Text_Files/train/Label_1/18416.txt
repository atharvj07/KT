#include <iostream>
#include <cmath>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long k;
        cin >> k;
        long long row = floor(sqrt(2 * k));
        long long col = k - row * (row + 1) / 2;
        if (row % 2 == 1) {
            swap(row, col);
        }
        cout << row << " " << col << endl;
    }
    return 0;
}
