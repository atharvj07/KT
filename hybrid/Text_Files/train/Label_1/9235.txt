#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 1e5 + 5, MAXH = 1e9 + 5;
int n, m, a[MAXN];
ll b[MAXN];
int main() {
    cin >> n;
    for(int i = 0; i < n; i++) {
        cin >> a[i];
    }
    cin >> m;
    for(int i = 0; i < m; i++) {
        int x, y; cin >> x >> y;
        b[x] = max(b[x], (ll)y);
    }
    for(int i = 2; i <= n; i++) {
        b[i] = max(b[i], b[i - 1]);
    }
    for(int i = 1; i <= n; i++) {
        cout << a[i - 1] + b[i] << " ";
    }
    return 0;
}
