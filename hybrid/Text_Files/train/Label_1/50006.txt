import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        int maxb = sc.nextInt();
        int t = sc.nextInt();
        int[] res = new int[k];
        for (int i = 0; i < k; i++) {
            int[][] a = new int[n][t];
            for (int j = 0; j < n; j++) {
                int b = sc.nextInt();
                for (int l = 0; l < t; l++) {
                    a[j][l] = b % (l + 1); b /= l + 1;
                }
            }
            res[i] = lcs(a, n, t);
        }
        for (int i : res) System.out.println(i);
    }

    static int lcs(int[][] a, int n, int t) {
        int[] dp = new int[n * t + 1];
        int len = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < t; j++) {
                int p = i * t + j;
                dp[p] = 1;
                for (int k = i * t; k < p; k++) {
                    if (a[i][j] > a[k / t][k % t]) {
                        dp[p] = Math.max(dp[p], dp[k] + 1);
                    }
                }
                len = Math.max(len, dp[p]);
            }
        }
        return len;
    }
}
