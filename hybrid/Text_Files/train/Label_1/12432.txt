#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int N=2e5+5;
ll arr[N],a[N],par[N];
vector<int>adj[N];
void dfs1(int idx,int p){
  par[idx]=p;
  for(auto it:adj[idx])if(it!=p){
    dfs1(it,idx);
    a[idx]+=a[it];
  }
}
void dfs2(int idx,int p){
  for(auto it:adj[idx])if(it!=p){
    a[it]+=a[idx];
    dfs2(it,idx);
  }
}
int main(){
  int n,m;cin>>n>>m;
  for(int i=1;i<=n;++i)cin>>arr[i];
  for(int i=1;i<n;++i){
    int u,v;cin>>u>>v;
    adj[u].push_back(v);adj[v].push_back(u);
  }
  dfs1(1,0);
  for(int i=1;i<=n;++i)a[i]+=arr[i];
  while(m--){
    int t,x,val;cin>>t>>x;
    if(t==1){
      cin>>val;
      a[x]+=val;
    }
    else cout<<a[x]<<endl;
  }
}
