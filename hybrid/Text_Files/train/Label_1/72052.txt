def bashar_run(n, m, k):
    # Calculate the total number of roads in the grid
    total_roads = 4*n*m - 2*n - 2*m
    if k > total_roads:
        return "NO"
    
    steps = []
    # Move Right and Left in the first row
    if m > 1:
        steps.append(f"{m-1} R")
        steps.append(f"{m-1} L")
        k -= 2*(m-1)
    
    # Move down, right (m-1 times), up and left (m-1 times) for each row
    for row in range(1, n):
        if k == 0:
            break
        steps.append("1 D")
        k -= 1
        if k == 0:
            break
        if m > 1:
            if k >= 3*(m-1):
                steps.append(f"{m-1} RUD")
                k -= 3*(m-1)
            else:
                if k // 3 > 0:
                    steps.append(f"{k//3} RUD")
                    k %= 3
                if k == 1:
                    steps.append("1 R")
                    k -= 1
                elif k == 2:
                    steps.append("1 RU")
                    k -= 2
        if row < n-1 and k > 0:
            steps.append("1 D")
            k -= 1
            if k == 0:
                break
    
    # Move up to the first row if there are moves left
    for _ in range(n-1):
        if k > 0:
            steps.append("1 U")
            k -= 1
    
    if k > 0:
        return "NO"
    
    return f"YES\n{len(steps)}\n" + "\n".join(steps)

n, m, k = map(int, input().split())
print(bashar_run(n, m, k))
