#include <bits/stdc++.h>
using namespace std;
#define maxn 5001
char s[maxn]; int dp[maxn][maxn], ans[maxn];
int main() {
  cin >> (s+1); int n=strlen(s+1);
  for(int i=1;i<=n;++i) dp[i][i]=1;
  for(int len=2;len<=n;++len) {
    for(int l=1,r;l<=n-len+1;++l) {
      r=l+len-1;
      if(len==2) dp[l][r]=(s[l]==s[r]);
      else if(s[l]==s[r]) dp[l][r]=dp[l+1][r-1];
    }
  }
  for(int i=1;i<=n;++i) { for(int j=i;j<=n;++j) ans[dp[i][j]]++; }
  for(int i=1;i<=n;++i) cout << ans[i] << ' '; cout << endl;
  return 0;
}
