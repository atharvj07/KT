#include<bits/stdc++.h>
using namespace std;
int dp[1005][1005],ans;
int main(){
    int n,k;
    cin>>n>>k;
    vector<int>a[k];
    for(int i=0;i<k;i++){
        for(int j=0;j<n;j++){
            int x;
            cin>>x;
            a[i].push_back(x);
        }
    }
    for(int i=1;i<=n;i++){
        for(int j=0;j<k;j++){
            for(int l=j+1;l<k;l++){
                if(a[j][i-1]==a[l][i-1]){
                    dp[j][l]=dp[j][l]+1;
                    ans=max(ans,dp[j][l]);
                }
            }
        }
    }
    cout<<ans<<endl;
}
