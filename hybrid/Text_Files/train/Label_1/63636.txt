from math import log2, trunc
b1, q, l, m = map(int, input().split())
bad = set(map(int, input().split()))
cur = b1
if q == 1:
    print(0 if cur in bad else -1)
else:
    limit = trunc(log2(l / abs(q))) + 1 if q < 0 else trunc(log2(l)) + 1
    if cur not in bad:
        print(limit)
    else:
        if q < 0:
            if abs(q) ** limit > l or abs(b1) + abs(q) * limit > l:
                print(-1)
            else:
                print(limit - 1)
        else:
            print(limit - 1)
