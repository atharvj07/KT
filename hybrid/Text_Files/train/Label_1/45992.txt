import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    String X = sc.next();
    int D = sc.nextInt();
    Long ans = 0L;
    for (int i = 0; i < N; i++) {
      if (D > 0 && X.charAt(i) == '0') {
        ans |= (1L << (N - 1 - i));
        D--;
      }
    }
    System.out.println(Long.toString(ans, 2));
  }
}
