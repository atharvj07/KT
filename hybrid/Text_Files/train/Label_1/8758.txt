#include <iostream>
#include <stack>
using namespace std;

int main() {
  int n;
  cin >> n;
  int arr[n];
  stack<int> st;
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
    while (!st.empty() && arr[i] > st.top()) {
      st.pop();
    }
    st.push(arr[i]);
  }
  cout << st.size() - 1 << endl;
  return 0;
}
