def find_composite_numbers(n):
    # Since 4 and 9 are composite, we can use these as bases for finding a and b
    # We ensure a and b are composite by multiplying n by a composite number
    # and adjusting by a fixed composite number to maintain the difference as n.
    a = 4 * (n + 1)  # a is composite because 4 is composite and we are multiplying it by another integer.
    b = a - n  # b will also be composite for n > 1. For n = 1, we handle it by setting a such that b is guaranteed to be composite.
    print(a, b)

n = int(input())
find_composite_numbers(n)
