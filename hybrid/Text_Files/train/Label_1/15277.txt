#include <bits/stdc++.h>
using namespace std;
using ll = long long; using vi = vector<int>; using vll = vector<ll>;
#define sz(x) (int)x.size()
int main() {
  ll n, k;
  cin >> n >> k;
  bool done = 0;
  ll ans = 0;
  for (int i = 0; i < n; i++) {
    ll a, b, c;
    cin >> a >> b >> c;
    ll shots = (c + k - 1) / k;
    if (a + shots - 1 > b) {
      done = 1;
      break;
    };
    ans += shots;
    if (a > ans) ans = a;
  }
  cout << (done ? -1 : ans);
  return 0;
}
