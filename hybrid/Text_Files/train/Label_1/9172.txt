#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n), b(n);
  for (int &i : a) cin >> i;
  for (int &i : b) cin >> i;
  int mx = *max_element(b.begin(), b.end());
  int ans = 0;
  for (int i = 0; i < n; i++) {
    ans += min(k, max(0, a[i] - b[i])) / a[i];
    k -= min(k, max(0, a[i] - b[i]));
  }
  cout << ans << endl;
}
