#include <iostream>
#include <vector>
#include <array>
using namespace std;

int main() {
    int n, m;
    array<int, 1001> arr;
    vector<array<int, 2>> orders;
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) cin >> arr[i];
    for (int i = 1; i <= m; ++i) {
        int a, b;
        cin >> a >> b;
        orders.push_back({a, b});
    }
    double cnt = 0;

    for (int i = 1; i <= n; ++i) {
        for (int j = i + 1; j <= n; ++j) {
            double p = 1;
            for (const auto& order : orders) {
                if (i == order[0] && j == order[1]) p /= 2;
                if (j == order[0] && i == order[1]) p /= 2;
            }
            cnt += p;
        }
    }
    printf("%.9lf\n", cnt);
}
