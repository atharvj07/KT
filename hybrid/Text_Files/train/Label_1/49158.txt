import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong(), p = sc.nextLong();
        if (n == 1 && p != 1 || n - p < 0) {
            System.out.println(-1);
            return;
        }
        int ans = 0;
        while (n > 0) {
            ans++;
            n = (n - p) / 2;
        }
        System.out.println(ans);
    }
}
