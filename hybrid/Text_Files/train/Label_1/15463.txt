#include <bits/stdc++.h>
using namespace std;
using ll=long long;
#define sz(x) (int)x.size()
const int MAXN = 1e5 + 5;
ll pos[MAXN], ans;

int main() {
    int n; cin>>n;
    for (int i=0; i<n; i++) cin>>pos[i];
    for (int i=0; i<n; i++) {
        if (abs(pos[i] - 1) < abs(pos[i] - 1e6)) ans += abs(pos[i] - 1);
        else ans += abs(pos[i] - 1e6);
    }
    cout << ans << endl;
    return 0;
}
