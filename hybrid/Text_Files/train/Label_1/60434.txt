def dfs(node, parent, depth, subtree_sum, apples, tree):
    subtree_sum[node] += apples[node-1]
    for child in tree[node]:
        if child == parent: continue
        dfs(child, node, depth+1, subtree_sum, apples, tree)
        subtree_sum[node] += subtree_sum[child]

def count_swaps(n, apples, parents):
    tree = [[] for _ in range(n+1)]
    for i, p in enumerate(parents, 2):
        tree[i].append(p)
        tree[p].append(i)
    
    subtree_sum = [0] * (n+1)
    dfs(1, -1, 0, subtree_sum, apples, tree)
    
    total_sum = subtree_sum[1]
    even_odd = [0, 0]
    for i in range(1, n+1):
        even_odd[sum(subtree_sum[i] for i in tree[i]) % 2] += 1
    
    if total_sum % 2 == 0:
        if even_odd[0] > 0 and even_odd[1] == 0:
            return even_odd[0] * (even_odd[0] - 1) // 2
        if even_odd[1] > 0 and even_odd[0] == 0:
            return even_odd[1] * (even_odd[1] - 1) // 2
    return 0

n = int(input().strip())
apples = list(map(int, input().strip().split()))
parents = list(map(int, input().strip().split()))
print(count_swaps(n, apples, parents))
