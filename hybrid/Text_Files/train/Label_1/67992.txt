n = int(input().strip())
a = list(map(int, input().strip().split()))

unique_a = list(set(a))

if len(unique_a) > 3:
    print("NO")
elif len(unique_a) == 3:
    unique_a.sort()
    if unique_a[2] - unique_a[1] == unique_a[1] - unique_a[0]:
        print("YES")
    else:
        print("NO")
else:
    print("YES")
