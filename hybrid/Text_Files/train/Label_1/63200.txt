import re

def total_bill(bill):
    prices = re.findall(r'\d{1,3}(?:\.\d{3})*(?:\.\d{2})?', bill)
    total_cents = 0
    for price in prices:
        parts = price.split('.')
        dollars = int(''.join(parts[:-1])) if len(parts) > 1 else int(parts[0])
        cents = int(parts[-1]) if len(parts) > 1 else 0
        total_cents += dollars * 100 + cents
    dollars, cents = divmod(total_cents, 100)
    formatted_dollars = f"{dollars:,}".replace(',', '.')
    return f"{formatted_dollars}.{cents:02d}" if cents else f"{formatted_dollars}"

# Example usage
bill = input().strip()
print(total_bill(bill))
