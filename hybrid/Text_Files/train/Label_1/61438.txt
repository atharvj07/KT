import sys

n, m = map(int, input().split())
a = list(map(int, input().split()))
b = sorted(enumerate(a), key=lambda x: x[1])
ans = 0
for j in range(n):
    l = j
    while l < n and a[b[l][0]] >= m:
        l += 1
    for i in range(j, l):
        if b[i][1] <= m and b[l][1] >= m:
            ans += 1
print(ans)
