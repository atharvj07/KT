n, a = map(int, input().split())
tasks = [tuple(map(int, input().split())) for _ in range(n)]

def max_profit(n, a, tasks):
    prefix_sum = [0]
    for d, c in tasks:
        prefix_sum.append(prefix_sum[-1] + c)
    
    max_gap = [0] * n
    for i in range(n - 1):
        max_gap[i] = (tasks[i + 1][0] - tasks[i][0]) ** 2
        if i > 0:
            max_gap[i] = max(max_gap[i], max_gap[i - 1])
    
    max_profit = -float('inf')
    for l in range(n):
        for r in range(l, n):
            profit = (r - l + 1) * a - (prefix_sum[r + 1] - prefix_sum[l]) - (max_gap[r - 1] if r > 0 else 0)
            max_profit = max(max_profit, profit)
            
    return max_profit

print(max_profit(n, a, tasks))
