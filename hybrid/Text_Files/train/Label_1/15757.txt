#include<bits/stdc++.h>
using namespace std;
#define full 1000000000000000000
#define half 500000000000000000
typedef long long ll;
int main() {
    int n;
    cin >> n;
    vector<ll> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (arr[i] != 1)
            cnt++;
    }
    vector<ll> ans;
    int i = 0, j = 0;
    while (j < n) {
        if (i == cnt)
            break;
        if (arr[j] == 1) {
            j++;
            continue;
        }
        if (ans.empty()) {
            ans.push_back(arr[j]);
            i++;
            j++;
        }
        else {
            if (arr[j] < ans.back() * 2 || ans.back() % arr[j] != 0) {
                ans.clear();
                i = 0;
                continue;
            }
            ans.push_back(arr[j]);
            i++;
            j++;
        }
    }
    if (i != cnt) {
        ans.clear();
        i = 0;
        while (j < n) {
            if (arr[j] == 1) {
                j++;
                continue;
            }
            if (ans.empty()) {
                ans.push_back(arr[j]);
                i++;
                j++;
            }
            else {
                if (ans.back() * 2 < arr[j] || arr[j] % ans.back() != 0) {
                    ans.clear();
                    i = 0;
                    continue;
                }
                ans.push_back(arr[j]);
                i++;
                j++;
            }
        }
    }
    if (i == cnt) {
        for (int i = 0; i < n; i++) {
            if (arr[i] == 1) {
                ans.push_back(arr[i]);
            }
        }
    }
    for (int i = 0; i < n; i++) {
        cout << ans[i] << " ";
    }
}
