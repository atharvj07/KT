#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> rooms(n);
  for (int i = 0; i < n; i++) {
    cin >> rooms[i].first >> rooms[i].second;
  }
  int cnt = 0;
  for (auto &room : rooms) {
    cnt += room.second - room.first >= 2;
  }
  cout << cnt << endl;
  return 0;
}
