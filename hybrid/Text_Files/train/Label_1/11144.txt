#include <bits/stdc++.h>
using namespace std;
const int mx=1e5+10;
int n, d, dp[mx], have[mx];
vector<int> adj[mx];
int main() {
    cin>>n>>d;
    string s;
    cin>>s;
    for (int i=1; i<=n; i++) {
        have[i]=(s[i-1]-'0');
    }
    for (int i=1; i<=n; i++) {
        if (!have[i]) continue;
        for (int j=i+1; j<=i+d&&j<=n; j++) {
            if (have[j]) adj[i].push_back(j);
        }
    }
    for (int i=1; i<=n; i++) dp[i]=1e9;
    dp[1]=0;
    queue<int> q;
    q.push(1);
    while (!q.empty()) {
        int u=q.front(); q.pop();
        for (int v: adj[u]) {
            if (dp[v]>dp[u]+1) {
                dp[v]=dp[u]+1;
                q.push(v);
            }
        }
    }
    if (dp[n]==1e9) dp[n]=-1;
    cout<<dp[n]<<endl;
}
