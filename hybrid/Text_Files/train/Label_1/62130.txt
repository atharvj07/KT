from collections import deque
n,m,start,end=[int(x) for x in input().split()]
a=[]
for i in range(m):
    a.append([int(x) for x in input().split()])
a.sort()
d=[0]*n
for x,y,z in a:
    for i in range(y-1,z):
        d[i]+=1
q=deque()
q.append(start-1)
ans=[]
while q:
    now=q.popleft()
    if now==end-1: break
    if now-1>=0 and d[now-1]==0 and not (now,now+1) in a:
        q.append(now-1)
        ans.append('L')
    if now+1<n and d[now+1]==0 and not (now+1,now+2) in a:
        q.append(now+1)
        ans.append('R')
    d[now]+=1
print(''.join(ans))
