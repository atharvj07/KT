import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) d[i] = sc.nextInt();
        System.out.println(possible(d) ? "yes" : "no");
    }
    static boolean possible(int[] d) {
        int n = d.length;
        int[] left = new int[n];
        left[0] = d[0];
        for (int i = 1; i < n - 1; i++) {
            left[i] = Math.max(left[i - 1], d[i]);
        }
        int[] right = new int[n];
        right[n - 1] = d[n - 1];
        for (int i = n - 2; i > 0; i--) {
            right[i] = Math.max(right[i + 1], d[i]);
        }
        for (int i = 0; i < n - 1; i++) {
            if (left[i] < right[i + 1]) return false;
        }
        return true;
    }
}
