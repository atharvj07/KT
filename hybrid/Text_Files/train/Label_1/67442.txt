def solve(n, points):
    if n == 0:
        return []
    if n == 2:
        return [(1, 2)]
    
    # Sort points by x, then by y, then by z
    sorted_points = sorted((x, y, z, i) for i, (x, y, z) in enumerate(points, start=1))
    
    # Remove points in pairs from the start and end of the sorted list
    result = []
    for i in range(0, n, 2):
        result.append((sorted_points[i][3], sorted_points[i+1][3]))
    
    return result

def main():
    n = int(input().strip())
    points = [tuple(map(int, input().split())) for _ in range(n)]
    result = solve(n, points)
    for a, b in result:
        print(a, b)

main()
