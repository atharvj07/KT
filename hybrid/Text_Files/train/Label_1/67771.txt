from collections import Counter

for _ in range(int(input())):
    n = int(input())
    p = [int(x) for x in input().split()]
    s = input()
    q = []
    for i in range(1, n + 1):
        if s.count('1', max(q) if q else 0, i):
            q.append(i)
    for i in range(1, n + 1):
        if not i in q:
