#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MOD = 1e9 + 7;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        ll a, b, c;
        cin >> a >> b >> c;
        ll d = (c + b - a) % 2 == 0 ? (c + b - a) / 2 : -1;
        cout << d << '\n';
    }
    return 0;
}
