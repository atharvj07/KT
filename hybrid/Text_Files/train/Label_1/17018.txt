#include <iostream>
#include <cmath>

int main() {
  int n;
  std::cin >> n;
  double arr[n];
  for (int i = 0; i < n; i++) {
    std::cin >> arr[i];
  }
  bool found = false;
  for (int d = -1000; d <= 1000 && !found; d++) {
    if (d == 0) continue;
    int pos = 0;
    for (int i = 0; i < n; i++) {
      if (arr[i] / d > 0) pos++;
    }
    if (pos >= std::ceil(n / 2.0)) {
      std::cout << d << std::endl;
      found = true;
    }
  }
  if (!found) std::cout << 0 << std::endl;
  return 0;
}
