class Main {
    private static boolean checkTriangle(int[] arr) {
        if (arr.length != 4) return false;
        int[] arr2 = {0, 0, 0, 0};
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                if (arr[i] > arr[j]) {
                    arr2[i]++;
                }
            }
        }
        return arr2[0] == 3 || arr2[1] == 3 || arr2[2] == 3 || arr2[3] == 3;
    }

    private static boolean checkSegment(int[] arr) {
        int[] sortArr = {arr[0], arr[1], arr[2], arr[3]};
        int[] arr2 = {0, 0, 0, 0};
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                if (sortArr[i] > sortArr[j]) {
                    arr2[i]++;
                }
            }
        }
        for (int i = 0; i < 4; i++) {
            if (arr2[i] == 2 && arr2[i + 1] == 2) {
                return true;
            }
        }
        return false;
    }

    public static void main(String[] args) {
        java.util.Scanner scan = new java.util.Scanner(System.in);
        int[] arr = {scan.nextInt(), scan.nextInt(), scan.nextInt(), scan.nextInt()};
        if (checkTriangle(arr)) {
            System.out.println("TRIANGLE");
        } else if (checkSegment(arr)) {
            System.out.println("SEGMENT");
        } else {
            System.out.println("IMPOSSIBLE");
        }
    }
}
