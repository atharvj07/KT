import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        boolean[][] board = new boolean[n][n];
        for (int i = 0; i < q; i++) {
            int x1 = sc.nextInt() - 1;
            int y1 = sc.nextInt() - 1;
            int x2 = sc.nextInt() - 1;
            int y2 = sc.nextInt() - 1;
            for (int j = x1; j <= x2; j++) {
                for (int k = y1; k <= y2; k++) {
                    board[j][k] = true;
                }
            }
        }
        int[] rows = new int[n];
        int[] cols = new int[n];
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (!board[i][j]) {
                    if (rows[i] == 0 && cols[j] == 0) {
                        count++;
                        rows[i] = 1;
                        cols[j] = 1;
                    }
                }
            }
        }
        System.out.println(count);
    }
}
