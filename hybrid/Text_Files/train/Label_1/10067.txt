#include <iostream>

using namespace std;

int main() {
    int n;
    cin >> n;
    int prices[n];
    for (int i = 0; i < n; ++i) {
        cin >> prices[i];
    }
    int max_price = 0;
    int winner = 0;
    for (int i = 0; i < n; ++i) {
        if (prices[i] > max_price) {
            max_price = prices[i];
            winner = i;
        }
    }
    int second_max_price = 0;
    for (int i = 0; i < n; ++i) {
        if (i != winner && prices[i] > second_max_price) {
            second_max_price = prices[i];
        }
    }
    cout << winner + 1 << " " << second_max_price << endl;
    return 0;
}
