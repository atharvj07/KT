import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int f = sc.nextInt();
        int k = sc.nextInt();
        int refuelCount = 0;
        int distanceLeft = 0;
        for (int i = 0; i < k; i++) {
            if (distanceLeft <= 0) {
                distanceLeft = b;
                refuelCount++;
            }
            distanceLeft -= (a - f);
            distanceLeft -= f;
        }
        if (distanceLeft < 0) {
            System.out.println(-1);
        } else {
            System.out.println(refuelCount);
        }
    }
}
