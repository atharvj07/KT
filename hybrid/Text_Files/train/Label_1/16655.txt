#include <bits/stdc++.h>
using namespace std;
const int MX = 3005;
int a[MX], b[MX], n, v;
// dp[i][j] = max fruits which can collect with days left j and last tree harvested i.
int dp[MX][MX], pre[MX];

int go(int i, int j) {
  if (i > n || j <= 0) return 0;
  int &ret = dp[i][j];
  if (ret != -1) return ret;
  ret = go(i + 1, j);
  if (j >= 2) ret = max(ret, pre[i] + go(i + 1, j - 2));
  if (j >= 1) ret = max(ret, pre[i - 1] + go(i + 1, j - 1));
  return ret;
}

int main() {
  memset(dp, -1, sizeof dp);
  cin >> n >> v;
  for (int i = 1; i <= n; i++) {
    cin >> a[i] >> b[i];
    pre[i] = pre[i - 1] + b[i];
  }
  cout << go(1, v) << endl;
  return 0;
}
