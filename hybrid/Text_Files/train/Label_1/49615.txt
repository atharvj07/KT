import java.util.LinkedList;
import java.util.Scanner;

class Node {
    LinkedList<Integer> adj;

    public Node() {
        adj = new LinkedList<>();
    }
}

public class Main {
    static Node[] tree;
    static long[] val;
    static boolean[] vis;
    static int n, m;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        val = new long[n + 1];
        tree = new Node[n + 1];
        for (int i = 1; i <= n; i++) tree[i] = new Node();
        for (int i = 1; i <= n; i++) val[i] = in.nextInt();
        for (int i = 1; i < n; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            tree[a].adj.add(b);
            tree[b].adj.add(a);
        }
        vis = new boolean[n + 1];
        int q = in.nextInt();
        while (q-- > 0) {
            int type = in.nextInt();
            int v = in.nextInt();
            if (type == 1) {
                int x = in.nextInt();
                dfs1(v, x);
            } else {
                System.out.println(dfs2(v, 0));
            }
        }
        in.close();
    }

    static void dfs1(int u, int x) {
        vis[u] = true;
        val[u] += x;
        for (int v : tree[u].adj) {
            if (!vis[v]) dfs1(v, x);
        }
    }

    static long dfs2(int u, int p) {
        long ans = 0;
        vis[u] = true;
        for (int v : tree[u].adj) {
            if (!vis[v]) ans += dfs2(v, p);
        }
        long a = val[u];
        long b = val[p];
        long c = a - b;
        if (c <= m) {
            if (c % 2 == 0) ans++;
            c = (c - 3) / 2;
            if (c >= 0 && c < m && (m - c) % 2 == 0) ans++;
            c = (c - 2) / 2;
            if (c >= 0 && c < m && (m - c) % 2 == 0) ans++;
            c = (c - 1) / 2;
            if (c >= 0 && c < m) ans++;
        }
        return ans;
    }
}
