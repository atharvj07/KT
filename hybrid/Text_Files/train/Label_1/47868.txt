import java.util.*;
class Main {
    static int[] c, sz, res;
    static List<Integer>[] g;
    static int n;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        c = new int[n + 1];
        sz = new int[n + 1];
        g = new List[n + 1];
        res = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            c[i] = sc.nextInt();
            g[i] = new ArrayList<>();
        }
        for (int i = 1; i < n; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            g[a].add(b);
            g[b].add(a);
        }
        dfs1(1, 0, 1, 0);
        dfs2(1, 0, sz[1]);
        for (int i = 1; i <= n; i++) {
            System.out.print(res[i] + " ");
        }
    }

    static void dfs1(int v, int p, int cursz, int curc) {
        sz[v] = 1;
        res[v] = curc;
        for (int u : g[v]) {
            if (u == p) continue;
            dfs1(u, v, cursz + 1, curc + (c[v] - c[u]));
            sz[v] += sz[u];
        }
    }

    static void dfs2(int v, int p, int cursz) {
        for (int u : g[v]) {
            if (u == p) continue;
            int tmp = res[v] + cursz - 2 * sz[u] + (c[u] - c[v]);
            if (tmp > res[u]) res[u] = tmp;
            dfs2(u, v, cursz);
        }
    }
}
