#include <bits/stdc++.h>
using namespace std;

const int INF = 1e9;

int main() {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }

    vector<vector<int>> dp(n, vector<int>(n + 1, INF));
    dp[n - 1][n - b[n - 1]] = 1;
    for (int i = n - 2; i >= 0; i--) {
        for (int j = 0; j <= n - i; j++) {
            if (dp[i + 1][j + b[i + 1]] != INF) { dp[i][j] = dp[i + 1][j + b[i + 1]]; }
            for (int k = 1; k <= a[i] && k + b[i] <= j; k++) {
                if (dp[i + k][j - k - b[i]] != INF) { dp[i][j] = min(dp[i][j], dp[i + k][j - k - b[i]] + 1); }
            }
        }
    }

    if (dp[0][0] == INF) { cout << -1 << endl; }
    else {
        int i = 0, j = 0;
        vector<int> ans;
        while (i != n) {
            ans.push_back(n - i);
            for (int k = 1; k <= a[i] && k + b[i] <= j; k++) {
                if (dp[i + k][j - k - b[i]] == dp[i][j] - 1) {
                    i += k;
                    j -= k + b[i];
                    break;
                }
            }
            j += b[i];
            i++;
        }
        cout << dp[0][0] << endl;
        for (int i = ans.size() - 1; i >= 0; i--) { cout << ans[i] << " "; }
        cout << endl;
    }

    return 0;
}
