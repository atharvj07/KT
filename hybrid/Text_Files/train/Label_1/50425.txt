import java.util.Scanner;

public class TextFormatter {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String text = scanner.nextLine();

        StringBuilder formattedText = new StringBuilder();

        char prevChar = ' '; // Initialize with a space to handle the first word
        for (char ch : text.toCharArray()) {
            if (Character.isLetter(ch)) {
                formattedText.append(ch);
            } else if (ch == '.' || ch == ',' || ch == '!' || ch == '?') {
                formattedText.append(ch).append(' '); // Add a space after punctuation
            } else if (prevChar != ' ' && Character.isLetter(prevChar)) { // Add a space between words
                formattedText.append(' ');
            }
            prevChar = ch;
        }

        System.out.println(formattedText.toString().trim()); // Remove any trailing space
    }
}
