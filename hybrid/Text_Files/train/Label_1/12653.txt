#include <iostream>
typedef long long ll;
#define mod 1000000007ll

int main() {
  int len;
  char s[100000];
  std::cin >> len >> s;
  int l = -1, r = -1, ans = 0;
  for (int i = 0; i < len; i++) {
    if (s[i] == 'a') {
      if (l == -1)
        l = i;
      r = i;
    }
  }
  if (l == -1)
    ans = 0;
  else if (r == l)
    ans = 1;
  else
    ans = (r - l) * (r - l + 1) / 2;
  std::cout << ans % mod << std::endl;
  return 0;
}
