import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        double[][] points = new double[n][2];
        for (int i = 0; i < n; i++) {
            points[i][0] = in.nextInt();
            points[i][1] = in.nextInt();
        }
        double p = 0, k;
        for (int i = 0; i < n; i++) {
            p += points[i][0] * points[(i + 1) % n][1] - points[i][1] * points[(i + 1) % n][0];
        }
        p /= 2;
        for (int i = 0; i < n; i++) {
            k = (points[(i + 1) % n][0] * points[i][1] - points[(i + 1) % n][1] * points[i][0]) / p;
            if (k > 0 && k < 1) {
                System.out.printf("%.5f %.5f\n", points[i][0] + k * (points[(i + 1) % n][0] - points[i][0]),
                        points[i][1] + k * (points[(i + 1) % n][1] - points[i][1]));
                return;
            }
        }
        System.out.println("NA");
    }
}
