#include<bits/stdc++.h>
using namespace std;
#define rep(i,n) for(int i=0;i<n;++i)
#define repi(i,a,n) for(int i=a;i<n;++i)
#define repe(i,a,n) for(int i=a;i<=n;++i)
#define endl '\n'
typedef long long ll;

int main() {
    ios::sync_with_stdio(false), cin.tie(NULL);
    int n, m;
    cin >> n >> m;
    int finger[10] = {};
    rep(i, m) {
        int x;
        cin >> x;
        finger[x] = 1;
    }
    rep(i, n) {
        int x;
        cin >> x;
        if(finger[x]) cout << x << ' ';
    }
    cout << endl;
    return 0;
}
