import math
x1, y1, r1 = map(int, input().split())
x2, y2, r2 = map(int, input().split())
d = math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2)
if d >= r1 + r2:
    print(0)
elif d + min(r1, r2) <= max(r1, r2):
    print(math.pi * min(r1, r2) ** 2)
else:
    alpha = 2 * math.acos((r1 ** 2 + d ** 2 - r2 ** 2) / (2 * r1 * d))
    beta = 2 * math.acos((r2 ** 2 + d ** 2 - r1 ** 2) / (2 * r2 * d))
    print((alpha - math.sin(alpha)) * r1 ** 2 / 2 + (beta - math.sin(beta)) * r2 ** 2 / 2)
