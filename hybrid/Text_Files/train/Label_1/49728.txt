import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        int[] freq = new int[26];
        for (char c : s.toCharArray()) {
            freq[c - 'a']++;
        }
        int cnt = 0;
        for (int i : freq) {
            if (i > 0) cnt++;
        }
        System.out.println(cnt <= 26 ? "YES" : "NO");
    }
}
