#include <algorithm>
#include <vector>
#include <iostream>

int main() {
  int n, t;
  std::cin >> n >> t;
  std::vector<std::pair<int, int>> v(n);
  for (auto& p : v) {
    std::cin >> p.first >> p.second;
  }
  std::sort(v.begin(), v.end(), [&](auto a, auto b) {
    return a.second < b.second || (a.second == b.second && a.first > b.first);
  });
  int ans = 0, cur = 0;
  std::vector<int> ans_idx;
  for (auto p : v) {
    if (cur + p.second <= t) {
      ++ans;
      cur += p.second;
      ans_idx.push_back(p.first);
    }
  }
  std::cout << ans << '\n' << ans_idx.size() << '\n';
  for (auto idx : ans_idx) {
    std::cout << idx << ' ';
  }
  std::cout << '\n';
  return 0;
}
