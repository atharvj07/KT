for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    c = list(map(int, input().split()))
    p = []
    for i in range(n):
        if p and p[-1] == a[i]:
            p.append(b[i] if b[i] != a[i] else c[i])
        else:
            p.append(a[i])
    print(*p)
