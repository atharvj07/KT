#include <bits/stdc++.h>
#define rep(i, x, y) for (int i = (x); i <= (y); ++i)
const int N = 1510;
int n, s, m, k, a[N], c[N], f[N][N];
bool b[N][N];
int main() {
  scanf("%d%d%d%d", &n, &s, &m, &k);
  rep(i, 1, n) scanf("%d", &a[i]);
  rep(i, 1, s) {
    int l, r;
    scanf("%d%d", &l, &r);
    b[l][r] = 1;
  }
  rep(i, 1, n) rep(j, 1, n) {
    f[i][j] = f[i - 1][j] + b[i][j];
    c[f[i][j]]++;
  }
  rep(i, 1, m) {
    if (c[i] < k) {
      puts("-1");
      return 0;
    }
  }
  auto calc_kth = [&](int x, int y) {
    return a[f[y][x] - c[f[y][x]] + 1];
  };
  int ans = 1e9;
  rep(l, 1, n) rep(r, l, n) {
    if (f[r][l] - f[l - 1][l] >= m) ans = std::min(ans, calc_kth(l, r));
  }
  printf("%d\n", ans);
  return 0;
}
