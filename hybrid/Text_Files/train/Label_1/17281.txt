#include <bits/stdc++.h>
using namespace std;
const int MAXN=2e5+5;
const long long mod=1e9+7;
long long a[MAXN], f[MAXN], s[MAXN];
int main()
{
    int n;
    cin>>n;
    for(int i=1;i<=n;++i)
    {
        cin>>a[i];
        s[i]=s[i-1]+a[i];
    }
    f[1]=1;
    for(int i=2;i<=n;++i)
    {
        if(s[i]>s[f[i-1]]) f[i]=i;
        else f[i]=f[i-1];
    }
    long long ans=0;
    for(int i=1;i<=n;++i)
    {
        long long tmp=s[i]-s[f[i]-1];
        ans+=tmp*tmp%mod;
        ans%=mod;
    }
    cout<<ans<<endl;
    return 0;
}
