n=int(input())
a=[int(x) for x in input().split()]
b=[int(x) for x in input().split()]
i=0
fa=0
fb=0
while min(fa,fb)<n and a[fa]!=b[fb]:
    if a[fa]<b[fb]:
        b.append(b[fb])
        b.append(a[fa])
        fb+=1
        fa+=1
    else:
        a.append(a[fa])
        a.append(b[fb])
        fa+=1
        fb+=1
    i+=1
if fa==n:
    print(i,2)
elif fb==n:
    print(i,1)
else:
    print(-1)
