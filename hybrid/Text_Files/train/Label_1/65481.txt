for _ in range(int(input())):
    n, m = map(int, input().split())
    a = [[*map(int, input().split())] for _ in range(n)]
    ans = 0
    for x in range(1, n):
        for y in range(1, m):
            if x == n - 1 and y == m - 1:
                ans += a[x][y] == 0
            elif a[x][y] != a[n - x - 1][m - y - 1]:
                ans += 1
    print(ans)
