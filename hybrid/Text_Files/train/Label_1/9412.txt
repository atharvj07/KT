#include <bits/stdc++.h>
using std::cin;
using std::cout;
using std::pair;
using std::vector;
const int N = 5000 + 1;
int a[N], b[N];
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> dates;
  for (int i = 1, x, y; i <= n; i++) {
    cin >> x >> y;
    dates.emplace_back(x, 1);
    dates.emplace_back(y, -1);
  }
  sort(dates.begin(), dates.end());
  int cnt = 0, ans = 0;
  for (auto date : dates) {
    if (cnt == 0) {
      ans = date.first;
    }
    cnt += date.second;
  }
  cout << ans;
}
