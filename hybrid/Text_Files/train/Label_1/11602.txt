#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define sz(x) (int)x.size()
int a[105], b[105], f[2000005];
vector<int> v;
int main() {
  int t, n, x;
  while (cin >> t) {
    while (t--) {
      v.clear();
      memset(f, 0, sizeof(f));
      cin >> n;
      for (int i = 0; i < n; i++) { cin >> a[i]; }
      for (int j = 1; j <= 1000000; j++) {
        for (int i = 0; i < n; i++) {
          if (a[i] + j <= 1000000) {
            f[a[i] + j]++;
          }
        }
      }
      int ok = 0;
      for (int j = 1; j <= 1000000; j++) {
        if (f[j] == 0) {
          v.pb(j);
          ok = 1;
        }
        if (sz(v) == n) break;
      }
      if (ok) {
        cout << "YES\n";
        for (int i = 0; i < n; i++) { cout << v[i] << " "; }
        cout << "\n";
      } else {
        cout << "NO\n";
      }
    }
  }
  return 0;
}
