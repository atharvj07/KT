n = int(input())
a = list(map(int, input().split()))

if n == 1:
    if a[0] > 0:
        print("BitLGM")
    else:
        print("BitAryo")
elif n == 2:
    if a[0] ^ a[1]:
        print("BitLGM")
    else:
        print("BitAryo")
else:
    if (a[0] ^ a[1] ^ a[2]):
        print("BitLGM")
    else:
        print("BitAryo")
