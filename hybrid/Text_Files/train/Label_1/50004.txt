import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        int maxb = sc.nextInt();
        int t = sc.nextInt();
        int[][] bs = new int[k][n];
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < n; j++) {
                bs[i][j] = sc.nextInt();
            }
        }
        sc.close();
        int[] ans = new int[k];
        for (int i = 0; i < k; i++) {
            ans[i] = 1;
            int[] dp = new int[n];
            dp[0] = 1;
            for (int j = 1; j < n; j++) {
                dp[j] = 1;
                for (int l = 0; l < j; l++) {
                    if (bs[i][j] % bs[i][l] == 0 && dp[j] < dp[l] + 1) {
                        dp[j] = dp[l] + 1;
                    }
                }
                ans[i] = Math.max(ans[i], dp[j]);
            }
        }
        for (int i = 0; i < k; i++) {
            System.out.println(ans[i]);
        }
    }
}
