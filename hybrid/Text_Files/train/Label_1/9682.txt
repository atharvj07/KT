#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, a[N], l[N], r[N], sum[N];
int main() {
    ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    int ans = 0, tmp = 0;
    for (int i = 1; i <= n; i++) {
        tmp = max(tmp, r[i] = max(l[i], l[i - 1]));
        l[i] = tmp + 1;
        ans += min(a[i], r[i]);
    }
    cout << ans << "\n";
    return 0;
}
