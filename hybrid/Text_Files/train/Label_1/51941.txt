import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int q = in.nextInt();
        while (q-- > 0) {
            int n = in.nextInt();
            String s = in.next();
            String t = in.next();
            boolean ok = true;
            for (int i = 0; i < n; i++) {
                if (s.charAt(i) != t.charAt(i)) {
                    if (i + 1 < n && s.charAt(i) == t.charAt(i + 1) && s.charAt(i + 1) == t.charAt(i)) {
                        i++;
                    } else {
                        ok = false;
                        break;
                    }
                }
            }
            System.out.println(ok ? "YES" : "NO");
        }
    }
}
