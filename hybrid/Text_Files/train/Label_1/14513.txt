#include <iostream>
#include <set>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> x(n), y(n);
    for (auto &x_i : x) cin >> x_i;
    for (auto &y_i : y) cin >> y_i;
    set<int> s{x.begin(), x.end()};
    for (const auto &y_i : y) s.insert(y_i);
    cout << (s.size() & 1 ? "Koyomi" : "Karen") << endl;
    return 0;
}
