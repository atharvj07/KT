import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    int m = sc.nextInt();
    int[] b = new int[m];
    for (int i = 0; i < m; i++) {
      b[i] = sc.nextInt();
    }
    int sa = 0;
    int sb = 0;
    for (int i = 0; i < n; i++) {
      sa += a[i];
    }
    for (int i = 0; i < m; i++) {
      sb += b[i];
    }
    int v = Math.abs(sa - sb);
    int k = 0;
    int[][] swaps = new int[2][2];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        int na = sa - a[i] + b[j];
        int nb = sb - b[j] + a[i];
        int nv = Math.abs(na - nb);
        if (nv < v) {
          v = nv;
          k = 1;
          swaps[0][0] = i + 1;
          swaps[0][1] = j + 1;
        }
      }
    }
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        for (int p = 0; p < n; p++) {
          for (int q = 0; q < m; q++) {
            if (i == p || j == q) {
              continue;
            }
            int na = sa - a[i] - a[p] + b[j] + b[q];
            int nb = sb - b[j] - b[q] + a[i] + a[p];
            int nv = Math.abs(na - nb);
            if (nv < v) {
              v = nv;
              k = 2;
              swaps[0][0] = i + 1;
              swaps[0][1] = j + 1;
              swaps[1][0] = p + 1;
              swaps[1][1] = q + 1;
            }
          }
        }
      }
    }
    System.out.println(v);
    System.out.println(k);
    for (int i = 0; i < k; i++) {
      System.out.println(swaps[i][0] + " " + swaps[i][1]);
    }
  }
}
