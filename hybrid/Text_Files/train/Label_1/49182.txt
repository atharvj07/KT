import java.util.Scanner;

public class CountryPathCounter {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int m = scanner.nextInt();

        int[] entryCosts = new int[n];
        int[] layerCosts = new int[n];
        int[] exitCosts = new int[n];

        for (int i = 0; i < n; i++) {
            entryCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            layerCosts[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            exitCosts[i] = scanner.nextInt();
        }

        long[][] dp = new long[l + 1][n];
        dp[0][0] = 1; // Starting from the entry point

        for (int i = 1; i <= l; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if ((entryCosts[j] + layerCosts[k] * (i - 1) + exitCosts[k]) % m == 0) {
                        dp[i][j] = (dp[i][j] + dp[i - 1][k]) % MOD;
                    }
                }
            }
        }

        long totalPaths = 0;
        for (int j = 0; j < n; j++) {
            totalPaths = (totalPaths + dp[l][j]) % MOD;
        }

        System.out.println(totalPaths);
    }
}
