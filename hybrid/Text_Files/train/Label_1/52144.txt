import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[][] a = new int[n][m];
        int[] rn = new int[n], cm = new int[m];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++) {
                a[i][j] = sc.nextInt();
                rn[i] = Math.max(rn[i], a[i][j]);
                cm[j] = Math.max(cm[j], a[i][j]);
            }
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++) {
                a[i][j] = Math.min(a[i][j], Math.min(rn[i], cm[j]));
            }

        for (int[] ar : a) {
            for (int x : ar)
                System.out.print(x + " ");
            System.out.println();
        }
    }
}
