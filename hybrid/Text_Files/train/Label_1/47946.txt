import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int k = 1;
        while (k * k <= n) {
            if (n % k == 0) {
                boolean ok = true;
                for (int i = 0; i < k; i++) {
                    for (int j = 0; j < n / k; j++) {
                        if (a[i] != a[i + j * k]) {
                            ok = false;
                            break;
                        }
                    }
                    if (!ok) {
                        break;
                    }
                }
                if (ok) {
                    System.out.println(k);
                    return;
                }
            }
            k++;
        }
        System.out.println(1);
    }
}
