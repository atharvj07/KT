#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] % 2 == 0) ans += upper_bound(a.begin(), a.end(), 2 * a[i] / 3) - a.begin() - 1;
    }
    cout << ans << endl;
  }
  return 0;
}
