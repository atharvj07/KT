import java.util.*;
import java.math.BigDecimal;
import java.math.RoundingMode;

public class Main {

    static final double EPS = 1e-6;

    static double minFuel(int n, double m, int[] a, int[] b) {

        double[][] dp = new double[n + 3][n + 3];

        for (int i = 1; i <= n + 1; i++)
            dp[i][1] = i > 1 ? dp[i - 1][1] + m / a[i - 1] : m / a[0];

        for (int j = 2; j <= n + 1; j++) {
            for (int i = j - 1; i >= 1; i--) {
                double takeoff = (dp[i + 1][j] + m) / a[i];
                if (i == 1)
                    takeoff = Math.min(takeoff, m / a[0]);

                double land = (dp[i][j - 1] + m) / b[j - 1];

                dp[i][j] = Math.min(takeoff, land);
            }
        }

        if (dp[1][n + 1] >= 1 - EPS)
            return dp[1][n + 1] - 1;
        else
            return -1;
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        double m = sc.nextDouble();
        int[] a = new int[n + 1], b = new int[n + 1];

        for (int i = 1; i <= n; i++)
            a[i] = sc.nextInt();
        for (int i = 1; i <= n; i++)
            b[i] = sc.nextInt();

        double ans = minFuel(n, m, a, b);
        if (ans == -1)
            System.out.println(-1);
        else {
            ans = new BigDecimal(ans).setScale(9, RoundingMode.HALF_UP).doubleValue();
            System.out.println(ans);
        }
    }
}
