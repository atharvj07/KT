#include <bits/stdc++.h>
using namespace std;

int main() {
  string s;
  cin >> s;
  int zero = 0, one = 0;
  for (char x : s) {
    int cur = x - '0';
    if (zero == 0) zero += !cur;
    if (one == 0) one += cur;
    if (zero == one) break;
  }
  if (zero != one) {
    cout << -1 << endl;
  } else {
    cout << 1 << endl << s.size() << ' ';
    for (int i = 1; i <= s.size(); i++) cout << i << ' ';
    cout << endl;
  }
}
