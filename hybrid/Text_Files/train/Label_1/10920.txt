#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<pair<pair<int, int>, pair<int, int>>> v;
    bool ok = true;
    for (int i = 0; i < n; i++) {
      int a, b, c, d;
      cin >> a >> b >> c >> d;
      if (a > m || b > m || c > m || d > m) ok = false;
      v.push_back({{a, b}, {c, d}});
    }
    if (m % 2 == 1 && n % 2 == 0) ok = false;
    if (m % 2 == 0 && n % 2 == 1 && n != 1) ok = false;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if (v[i].first.first == v[j].second.first &&
            v[i].first.second == v[j].second.second &&
            v[i].second.first == v[j].first.second &&
            v[i].second.second == v[j].first.first)
          ok = false;
      }
    }
    cout << (ok ? "YES" : "NO") << '\n';
  }
}
