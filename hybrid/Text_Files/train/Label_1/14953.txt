#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define rep(i, l, r) for (int i = (l); i <= (r); ++i)
const int N = 2e5 + 10;
vector<int> g[N];
ll a[N];

int dfs(int x, int p, ll &ans) {
  int sz = 1;
  for (auto& y : g[x]) {
    if (y == p) continue;
    ll b = dfs(y, x, ans);
    ans += abs(a[x] - a[y]) * b * sz;
    sz += b;
  }
  return sz;
}

int main() {
  int n; cin >> n;
  rep(i, 2, n) {
    int x; cin >> x;
    g[i].push_back(x), g[x].push_back(i);
  }
  rep(i, 1, n) cin >> a[i];

  ll ans = 0;
  dfs(1, 1, ans);
  cout << ans << '\n';
  return 0;
}
