q = int(input())
for _ in range(q):
    n = int(input())
    p = list(map(int, input().split()))
    a = [0] * n
    for i in range(n):
        if a[i] == 0:
            cycle = []
            j = i
            while not a[j]:
                cycle.append(j)
                a[j] = 1
                j = p[j] - 1
            cycle_length = len(cycle)
            for j in cycle:
                a[j] = cycle_length
    print(*a)
