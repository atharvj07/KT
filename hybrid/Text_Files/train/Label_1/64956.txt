n, m = map(int, input().split())
field = [list(input()) for _ in range(n)]
for i in range(n):
    for j in range(m - 1):
        if field[i][j] == "G" and field[i][j + 1] == "S":
            print(1)
            quit()
        elif field[i][j] == "S" and field[i][j + 1] == "G":
            print(-1)
            quit()
for i in range(n):
    if field[i][m - 1] == "G":
        print(-1)
        quit()
print(2)
