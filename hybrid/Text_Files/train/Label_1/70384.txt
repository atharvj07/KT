t = int(input().strip())
for _ in range(t):
    poem = input().strip()
    changes = 0
    for i in range(1, len(poem)):
        if poem[i] == poem[i - 1]:
            changes += 1
            poem = poem[:i] + "0" + poem[i + 1:]
        elif i > 1 and poem[i] == poem[i - 2]:
            changes += 1
            poem = poem[:i] + "0" + poem[i + 1:]
    print(changes)
