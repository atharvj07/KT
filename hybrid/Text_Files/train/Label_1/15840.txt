#include <bits/stdc++.h>
using namespace std;
const int maxn = 100000 + 5;
vector<int> adj[maxn], ord;
int deg[maxn];
void dfs(int u, int p) {
  for (auto v : adj[u]) {
    if (v == p) continue;
    dfs(v, u);
    deg[u] += deg[v] + 1;
  }
  ord.push_back(u);
}
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n - 1; i++) {
    int a, b;
    cin >> a >> b;
    adj[a].push_back(b);
    adj[b].push_back(a);
    deg[a]++;
    deg[b]++;
  }
  dfs(1, 1);
  int ans = 2;
  for (auto v : ord) {
    if (deg[v] == 1) ans++;
  }
  cout << ans << endl;
}
