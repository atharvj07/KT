import sys

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = [int(input()) for _ in range(n)]
        a.sort()
        result = []
        for i in a:
            if result and result[-1][-1] == i:
                result[-1].append(i)
            else:
                result.append([i])
        for i in result:
            print('.'.join(map(str, i)))

solve()
