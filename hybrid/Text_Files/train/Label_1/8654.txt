#include <bits/stdc++.h>
using namespace std;
#define LL long long
const int MOD = 1e9 + 7;
vector<int>P;
void sieve(int n) {
    P.push_back(2);
    bool isPrime[n + 1];
    memset(isPrime, true, sizeof(isPrime));
    isPrime[0] = isPrime[1] = false;
    for (int i = 4; i <= n; i += 2) isPrime[i] = false;
    for (int i = 3; i * i <= n; i += 2) {
        if (isPrime[i]) {
            P.push_back(i);
            for (int j = i * i; j <= n; j += i) isPrime[j] = false;
        }
    }
    for (int i = 2; i * i <= n; i++) if (isPrime[i]) P.push_back(i);
}
int phi[1000001];
void Euler_Totient() {
    memset(phi, 0, sizeof(phi));
    phi[1] = 1;
    for (LL i = 2; i <= 1000000; i++) {
        if (phi[i] == 0) {
            phi[i] = i - 1;
            for (LL j = i + i; j <= 1000000; j += i) {
                phi[j] = (phi[j] * (i - 1)) / i;
            }
        }
    }
}
LL fast_pow(LL x, LL p) {
    if (p == 0) return 1;
    LL ans = fast_pow(x, p / 2) % MOD;
    ans = (ans * ans) % MOD;
    if (p % 2) ans = (ans * x) % MOD;
    return ans;
}
LL GCD(LL x, LL y) {
    if (y == 0) return x;
    return GCD(y, x % y);
}
int main() {
    int n;
    cin >> n;
    LL a[n + 1], val[1000001] = {0};
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        val[a[i]]++;
    }
    Euler_Totient();
    sieve(1000000);
    LL ans = 0;
    for (int i = 1; i <= n; i++) {
        LL x = a[i], cur = 1;
        for (int j = 0; j < P.size() && P[j] * P[j] <= x; j++) {
            if (x % P[j] == 0) {
                cur *= P[j];
                while (x % P[j] == 0) x /= P[j];
            }
        }
        if (x > 1) cur *= x;
        ans += (val[a[i]] * (val[a[i]] + 1) % MOD * (phi[cur] % MOD)) % MOD;
        ans %= MOD;
    }
    cout << ans << endl;
    return 0;
}
