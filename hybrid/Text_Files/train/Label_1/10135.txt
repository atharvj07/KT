#include<iostream>
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;
    int count=0;
    for(int i=0;i<n;i++){
        int x;
        cin>>x;
        int j=0;
        while(x>0){
            if(x%10==4 || x%10==7)j++;
            x=x/10;
        }
        if(j<=m)count++;
    }
    cout<<count<<endl;
    return 0;
}
