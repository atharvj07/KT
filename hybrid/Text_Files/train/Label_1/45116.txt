import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {

    static class Node {

        Node[] children;
        boolean isLeaf;

        public Node() {
            isLeaf = false;
            children = new Node[26];
        }

    }

    static Node buildTrie(List<String> words) {
        Node root = new Node();

        for (String word : words) {
            Node cur = root;
            for (int i = 0; i < word.length(); i++) {
                char ch = word.charAt(i);
                int index = ch - 'a';
                if (cur.children[index] == null) {
                    cur.children[index] = new Node();
                }
                cur = cur.children[index];
            }
            cur.isLeaf = true;
        }

        return root;
    }

    static long countSubstrings(Node root, String s, int l, int r, int startIndex, long[][] dp) {
        Node cur = root;
        long count = 0;
        for (int i = startIndex; i <= r; i++) {
            int index = s.charAt(i) - 'a';
            if (cur.children[index] == null) {
                break;
            }
            cur = cur.children[index];
            if (cur.isLeaf) {
                if (i == r) {
                    count++;
                } else if (dp[i + 1][r] != -1) {
                    count += dp[i + 1][r];
                } else {
                    count += countSubstrings(root, s, l, r, i + 1, dp);
                }
            }
        }

        dp[l][r] = count;
        return count;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String s = reader.readLine();
        int q = Integer.parseInt(reader.readLine());

        List<String> words = new ArrayList<>();
        long[][] dp = new long[s.length() + 1][s.length() + 1];
        for (int i = 0; i <= s.length(); i++) {
            Arrays.fill(dp[i], -1);
        }

        for (int i = 0; i < q; i++) {
            String[] query = reader.readLine().split(" ");
            int type = Integer.parseInt(query[0]);

            if (type == 1) {
                int index = Integer.parseInt(query[1]) - 1;
                s = s.substring(0, index) + query[2] + s.substring(index + 1);
            } else {
                int l = Integer.parseInt(query[1]) - 1;
                int r = Integer.parseInt(query[2]) - 1;
                String pattern = query[3];
                words.add(pattern);

                long ans;
                if (dp[l][r] != -1) {
                    ans = dp[l][r];
                } else {
                    Node root = buildTrie(words);
                    ans = countSubstrings(root, s, l, r, l, dp);
                }

                System.out.println(ans);
            }
        }
    }
}
