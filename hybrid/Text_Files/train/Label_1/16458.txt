#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll func(ll x) {  // return sum of digits of x
    ll ret = 0;
    while (x) {
        ret += x % 10;
        x /= 10;
    }
    return ret;
}

int main() {
    ll a;
    cin >> a;
    if (a == 1) {
        cout << -1 << '\n';
        return 0;
    }
    ll l = 10, r = 1e18;
    while (l <= r) {
        ll mid = (l + r) / 2;
        ll cur = mid * a;
        if (func(cur) == func(mid)) {
            cout << cur << '\n';
            return 0;
        }
        if (func(cur) < func(mid)) {
            l = mid + 1;
        } else
            r = mid - 1;
    }
    cout << -1 << '\n';
    return 0;
}
