#include<bits/stdc++.h>
using namespace std;

typedef long long ll;
const int mod=1e9+7;

bool sieve[1000009];

int32_t main() {
	int a,b,k;
	cin>>a>>b>>k;
	for(int p=2;p*p<=b;++p) {
		if(sieve[p]==0) {
			for(int i=p*p;i<=b;i+=p)
				sieve[i]=1;
		}
	}
	for(int l=1;l<=b-a+1;++l) {
		bool f=true;
		for(int x=a;x<=b-l+1;++x) {
			int cnt=0;
			for(int i=x;i<x+l;++i)
				if(sieve[i]==0) ++cnt;
			if(cnt<k) {
				f=false;
				break;
			}
		}
		if(f) {
			cout<<l<<endl;
			return 0;
		}
	}
	cout<<-1<<endl;
	return 0;
}
