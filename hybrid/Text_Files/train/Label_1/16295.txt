#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-6;

struct tap {
    long long a, t;
    double d() { return (a - 1.0) / (t - 1.0); }
};

int main() {
    ios::sync_with_stdio(false);
    int n, T;
    cin >> n >> T;
    vector<tap> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i].a >> v[i].t; }
    sort(v.begin(), v.end(), [](tap &a, tap &b) { return a.d() > b.d(); });
    double l = 0, r = 1e18;
    while (r - l > eps) {
        double mid = (l + r) / 2, sum1 = 0, sum2 = 0;
        for (int i = 0; i < n; i++) {
            if (v[i].t > T)
                continue;
            double x = min(mid * (v[i].t - 1.0), v[i].a - 1.0);
            sum1 += x;
            sum2 += (x / (v[i].t - 1.0));
            if (sum2 >= 1e18)
                break;
        }
        if (sum2 <= 0 || sum2 > 1e18)
            r = mid;
        else {
            double err = abs((sum1 / sum2) - T) / T;
            if (err <= eps) {
                l = mid + eps;
            } else
                r = mid;
        }
    }
    if (l > 1e18)
        l = 0;
    cout << setprecision(12) << l << endl;
}
