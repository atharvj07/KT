import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }
            int badDays = 0;
            int maxPrice = a[n - 1];
            for (int i = n - 2; i >= 0; i--) {
                if (a[i] > maxPrice) {
                    badDays++;
                } else {
                    maxPrice = a[i];
                }
            }
            System.out.println(badDays);
        }
    }
}
