import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int[] parent = new int[n + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 2; i <= n; i++) {
            parent[i] = Integer.parseInt(st.nextToken());
        }
        int q = Integer.parseInt(br.readLine());
        List<Map<Integer, Long>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new HashMap<>());
        }
        for (int i = 1; i <= n; i++) {
            int par = parent[i];
            adj.get(i).put(par, 0L);
            adj.get(par).put(i, 0L);
        }
        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            if (type == 1) {
                long x = Long.parseLong(st.nextToken());
                long k = Long.parseLong(st.nextToken());
                dfs1(adj, v, parent, new boolean[n + 1], 0, x, k);
            } else {
                long res = dfs2(adj, v, parent, new long[n + 1], new boolean[n + 1], 0);
                System.out.println(res % 1000000007);
            }
        }
    }

    private static void dfs1(List<Map<Integer, Long>> adj, int v, int[] parent, boolean[] vis, int depth, long x, long k) {
        vis[v] = true;
        for (Map.Entry<Integer, Long> e : adj.get(v).entrySet()) {
            int u = e.getKey();
            if (!vis[u]) {
                e.setValue(e.getValue() + x - k * depth);
                dfs1(adj, u, parent, vis, depth + 1, x, k);
            }
        }
    }

    private static long dfs2(List<Map<Integer, Long>> adj, int v, int[] parent, long[] sum, boolean[] vis, int depth) {
        vis[v] = true;
        long res = 0;
        for (Map.Entry<Integer, Long> e : adj.get(v).entrySet()) {
            int u = e.getKey();
            if (!vis[u]) {
                sum[v] += sum[u] + e.getValue() - depth * e.getValue();
                res += dfs2(adj, u, parent, sum, vis, depth + 1);
            }
        }
        return res + sum[v];
    }
}
