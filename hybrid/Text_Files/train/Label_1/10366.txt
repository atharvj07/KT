#include<bits/stdc++.h>
using namespace std;
#define pb push_back
#define int long long
#define v vector
const int mod=1e9+7,inf=1e18;
signed main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    int n,m;cin>>n>>m;
    string s;cin>>s;
    v<v<int>>g(n+1);
    int cnt[n+1]={},vis[n+1]={},f[n+1]={};
    while(m--){
        int a,b;cin>>a>>b;
        g[a].pb(b);
        cnt[s[b]-'a']++;
    }
    for(int i=0;i<n;i++)vis[i]=0,f[i]=0;
    int ans=-1;
    for(int i=0;i<n;i++){
        if(vis[i])continue;
        int mx=0,v=-1;
        queue<int>q;
        q.push(i);vis[i]=1;
        while(!q.empty()){
            int u=q.front();q.pop();
            if(cnt[s[u]-'a']>mx){mx=cnt[s[u]-'a'];v=s[u]-'a';}
            if(f[u])continue;f[u]=1;
            for(auto adj:g[u]){
                if(vis[adj])continue;
                q.push(adj);vis[adj]=1;
            }
        }
        if(v!=-1)ans=max(ans,mx);
    }
    cout<<(ans==-1?-1:ans)<<endl;
}
