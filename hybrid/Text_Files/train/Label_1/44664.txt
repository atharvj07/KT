import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class PrefixProductSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Integer> forbiddenProducts = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            forbiddenProducts.add(scanner.nextInt());
        }

        List<Integer> sequence = constructSequence(forbiddenProducts, m);

        System.out.println(sequence.size());
        for (int element : sequence) {
            System.out.print(element + " ");
        }
    }

    private static List<Integer> constructSequence(List<Integer> forbiddenProducts, int m) {
        List<Integer> sequence = new ArrayList<>();
        int currentProduct = 1;

        for (int i = 0; i < m; i++) {
            if (!forbiddenProducts.contains(currentProduct) && !sequence.contains(i)) {
                sequence.add(i);
                currentProduct = (currentProduct * i) % m;
            }
        }

        return sequence;
    }
}
