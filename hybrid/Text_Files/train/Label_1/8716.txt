#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n, ans = 0;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int m = a[0];
    for (int i = 1; i < n; i++) {
        if (a[i] > m) ans++, m = a[i];
    }
    cout << ans << endl;
}
