#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
int main()
{
    string s;
    cin >> s;
    int n = s.length();
    vector<vector<int>> dp(n, vector<int>(n));
    for( int l = 2 ; l <= n ; l++ )
    {
        for( int i = 0 ; i + l - 1 < n ; i++ )
        {
            int j = i + l - 1;
            dp[i][j] = dp[i + 1][j - 1];
            if( s[i] != s[j] ) dp[i][j]++;
        }
    }
    string ans = "Impossible";
    for( int l = 2 ; l < n ; l++ )
    {
        if( dp[0][n - 1] == dp[0][l - 1] ) ans = "1";
    }
    cout << ans << endl;
}
