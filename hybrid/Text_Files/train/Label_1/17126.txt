#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
    ll n,d;
    cin>>n>>d;
    ll x[n];
    for(int i=0;i<n;i++){
        cin>>x[i];
    }
    int ans=0;
    for(int i=1;i<n;i++){
        if(x[i]-x[i-1]>d){
            ans+=((x[i]-x[i-1]-1)/d);
        }
    }
    cout<<ans<<'\n';
    return 0;
}
