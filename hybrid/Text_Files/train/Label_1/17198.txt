#include <iostream>

using namespace std;

typedef long long ll;

ll p[1000001];

int main() {
  int t;
  cin >> t;
  p[1] = 1;
  p[2] = 2;
  for (int i = 3; i <= 1000000; i++) {
    if (i % 2 == 0)
      p[i] = p[i - 2];
    else
      p[i] = p[i - 1];
  }
  while (t--) {
    ll n, r;
    cin >> n >> r;
    if (n == 1)
      cout << 1 << endl;
    else if (r == 1)
      cout << 1 << endl;
    else
      cout << 2 * p[(n - 1) % 4 + 1] << endl;
  }
  return 0;
}
