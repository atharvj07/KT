#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
int main(){
    int a,b,k,t;
    cin>>a>>b>>k>>t;
    int dp[105][105][205];
    dp[0][0][100+a-b]=1;
    for(int i=1;i<=t;i++){
        for(int x=-k;x<=k;x++){
            for(int j=0;j<=100;j++){
                if(j+x<0 || j+x>100) continue;
                dp[i][j][x+100]+=dp[i-1][j][x+100];
                dp[i][j][x+100]+=dp[i-1][j+x][x+100];
                dp[i][j][x+100]%=mod;
            }
        }
    }
    long long ans=0;
    for(int j=0;j<=100;j++){
        ans+=dp[t][j][100+a-b];
        ans%=mod;
    }
    cout<<ans<<endl;
}
