import java.util.Scanner;

public class Main {

    static int[][] dp = new int[101][11];

    public static void main(String[] args) {
        dp[0][0] = 1;
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[10];

        for (int i = 0; i < 10; i++) {
            a[i] = scanner.nextInt();
        }

        System.out.println(calc(n, a, 0, false));
    }

    static int calc(int n, int[] a, int pos, boolean leadingZero) {
        if (pos == n) {
            boolean ok = true;
            for (int i = 0; i < 10; i++) {
                ok &= a[i] == 0;
            }
            return ok ? 1 : 0;
        }

        if (dp[pos][leadingZero ? 1 : 0] != 0) {
            return dp[pos][leadingZero ? 1 : 0];
        }

        int ret = 0;

        for (int i = (leadingZero ? 1 : 0); i < 10; i++) {
            if (a[i] > 0) {
                a[i]--;
                ret += calc(n, a, pos + 1, leadingZero || i == 0);
                a[i]++;
            }
        }

        return dp[pos][leadingZero ? 1 : 0] = ret % 1000000007;
    }
}
