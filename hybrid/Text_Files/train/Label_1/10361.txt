#include<iostream>
#define fo(i,a,b) for(int i=a;i<b;i++)
using namespace std;
int main()
{
    int n,m,k,x=0,ans=0;
    cin>>n>>m>>k;
    char a[n][m+1];
    fo(i,0,n){
        fo(j,0,m){
            cin>>a[i][j];
        }
    }
    fo(i,0,n){
        int cnt=0;
        fo(j,0,m){
            if(a[i][j]=='0'){
                cnt=0;
                x=0;
            }
            else{
                x++;
                cnt+=x;
            }
        }
        if(k){
            if(ans+cnt>ans){
                ans+=cnt-min(cnt,k);
                k-=min(cnt,k);
            }
            else{
                ans+=cnt;
            }
        }
        else{
            ans+=cnt;
        }
    }
    cout<<ans;
}
