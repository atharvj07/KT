n, d = map(int, input().split())
transactions = list(map(int, input().split()))

balance = 0
deposits = 0
min_balance = 0

for transaction in transactions:
    balance += transaction
    if transaction == 0:
        if balance < 0:
            balance = 0
            deposits += 1
    min_balance = min(min_balance, balance)
    
    if balance > d:
        print(-1)
        break
else:
    if min_balance + deposits > d:
        print(-1)
    else:
        print(deposits)
