#include <bits/stdc++.h>
using namespace std;

string s, t;
int l = 0, r = 0, mx = 0;
unordered_map<char, int> f;
unordered_map<int, vector<int>> p;

void add(int idx) {
  f[s[idx]]++;
  if(f[s[idx]] == 1) p[s[idx]].push_back(idx);
}

void rem(int idx) {
  f[s[idx]]--;
  if(f[s[idx]] == 0) p[s[idx]].erase(lower_bound(p[s[idx]].begin(), p[s[idx]].end(), idx));
}

void check() {
  if(r - l + 1 < (int)t.length()) return;
  int occ = 0;
  for(char c : t) {
    if(p[c].size() && p[c][0] >= l && p[c].back() <= r) {
      int l2;
      for(l2 = 0; l2 < (int)p[c].size(); l2++) if(p[c][l2] > r) break;
      occ += p[c].size() - l2;
    }
  }
  if(occ > mx) {
    mx = occ;
    r = r - l + 1;
  }
}

int main() {
  cin >> s >> t;
  int n = s.length();
  for(int i = 0; i < n; i++) {
    if(s[i] != '?') add(i);
    else s[i] = 'a';
  }
  for(int i = 0; i < n; i++) {
    if(s[i] == t[0]) l = i;
    while(r + 1 < n && (s[r + 1] == t.back() || s[r + 1] == '?')) r++;
    check();
    rem(l);
    l++;
  }
  for(int i = l; i <= r; i++) s[i] = t.back();
  cout << s << endl;
}
