#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a.begin(), a.end());
        long long ans = a[0];
        for (int i = 1; i < n; i++) ans = max(ans, 1LL * a[i] - a[i - 1]);
        cout << ans << '\n';
    }
    return 0;
}
