#include <cstdio>
#include <cstring>
#include <vector>

using namespace std;

int m, n, k;
vector<int> pos;
char s[1009], t[1009];
bool occur[1009];

int main() {
  scanf("%d %d", &n, &m);
  scanf("%s %s", s, t);
  for (int i = 0; i < n; i++) {
    occur[i] = 0;
  }
  for (int i = 0; i < m - n + 1; i++) {
    bool ok = 1;
    for (int j = 0; j < n; j++) {
      if (s[j] == t[i + j]) {
        occur[j] = 1;
      } else if (s[j] != '?' && t[i + j] != '?') {
        ok = 0;
      }
    }
    if (ok) {
      for (int j = 0; j < n; j++) {
        if (!occur[j]) {
          pos.push_back(j + 1);
        }
      }
      k = pos.size();
      printf("%d\n", k);
      for (int j = 0; j < k; j++) {
        printf("%d ", pos[j]);
      }
      return 0;
    }
    for (int j = 0; j < n; j++) {
      occur[j] = 0;
    }
  }
  printf("-1\n");
  return 0;
}
