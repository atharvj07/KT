import datetime as dt

n = int(input())
jobs = []; res = 0
for _ in range(n):
    m, d, p, t = map(int, input().split())
    jobs.append((dt.date(2013, m, d), p, -1, t))
jobs.sort()
for day, p, i, t in jobs:
    start = day - dt.timedelta(days=t)
    pd = {}
    for day2, p2, i2, t2 in jobs:
        if day2 >= start and i != i2:
            pd[day2] = pd.get(day2, 0) + p2
    if max(pd.values(), default=0) == 0:
        res += p
        for day2 in pd:
            jobs[pd[day2] - 1] = (day2, pd[day2] - 1, i, t2)
print(res)
