#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int n, a[maxn], ans, sum;
int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) { cin >> a[i]; }
    sort(a + 1, a + 1 + n);
    for (int i = 1; i <= n; ++i) {
        if (sum + a[i] >= i) {
            sum += a[i];
            ans = i;
        } else
            break;
    }
    cout << ans << endl;
    return 0;
}
