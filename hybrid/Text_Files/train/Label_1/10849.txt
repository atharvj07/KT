#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=10007;
ll n;
vector<ll> a;
int main(){
    cin>>n;
    a.resize(n);
    for(ll i=0;i<n;i++)cin>>a[i];
    sort(a.begin(),a.end());
    ll ans=0;
    for(ll i=0;i<n;i++)ans+=a[i]*a[n-1-i];
    cout<<ans%mod;
    return 0;
}
