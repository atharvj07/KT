#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
ll pw(ll x, ll y) {
	ll ans = 1;
	while (y) {
		if (y % 2)
			ans = (ans * x) % mod;
		x = (x * x) % mod;
		y >>= 1;
	}
	return ans;
}
int main() {
	ll n, l, r;
	cin >> n >> l >> r;
	if ((r - l + 1) % 3 != 0 || (n * (l + r)) % 3 != 0) {
		cout << 0 << endl;
		return 0;
	}
	ll k = (r + 1) / 3 - l / 3;
	if (n < k) {
		cout << 0 << endl;
		return 0;
	}
	if (k == 0) {
		cout << pw(3, n) << endl;
		return 0;
	}
	ll t = n - k;
	ll sum = (pw(3, k) * pw(2, t)) % mod;
	cout << (sum + mod) % mod << endl;
}
