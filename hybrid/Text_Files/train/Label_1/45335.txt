import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (p[i] != i) {
                ans++;
                int j = p[i];
                p[i] = p[j];
                p[j] = j;
            }
        }
        System.out.println(ans);
    }
}
