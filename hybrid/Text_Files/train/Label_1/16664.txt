#include <bits/stdc++.h>
using namespace std;
vector<int> v;
int main() {
  int n, q;
  cin >> n >> q;
  v.resize(n);
  for (int i = 0; i < n; ++i) cin >> v[i];
  deque<int> dq(v.begin(), v.end());
  while (q--) {
    long long m;
    cin >> m;
    for (; m > 0 && dq.size() > 1;) {
      int a = dq.front();
      dq.pop_front();
      int b = dq.front();
      dq.pop_front();
      if (m % 2 == 0) {
        if (a < b)
          dq.push_front(b), dq.push_back(a);
        else
          dq.push_front(a), dq.push_back(b);
      } else {
        cout << a << ' ' << b << "\n";
      }
      --m;
    }
    if (m % 2) cout << -1 << "\n";
  }
}
