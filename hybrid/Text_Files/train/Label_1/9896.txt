#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, t[105], cnt = 0, ans = 0;
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> t[i]; }
    cin >> cnt;
    sort(t, t + n);
    int l = 0, r = 0;
    while (r < n) {
        if (t[r] - t[l] <= cnt)
            r++;
        else
            l++;
        ans = max(ans, r - l);
    }
    cout << ans << endl;
    return 0;
}
