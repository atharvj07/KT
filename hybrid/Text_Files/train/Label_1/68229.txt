import bisect

n=int(input())
s=list(input())

# Create prefix sum for each alphabet
count=[[] for i in range(26)]
for c in s:
    count[ord(c)-ord('a')].append(0)
for i in range(26):
    count[i]=[0]+count[i]

for i in range(26):
    for j in range(1,len(count[i])):
        count[i][j]+=count[i][j-1]

def find(l,r,k):
    idx=bisect.bisect_right(count[k],r)
    if idx==0:
        return 0
    if count[k][idx-1]<l:
        return 0
    return idx

q=int(input())
for _ in range(q):
    m,c=input().split()
    m=int(m)
    k=ord(c)-ord('a')
    print(min(m,find(0,m-1,k)+1))
