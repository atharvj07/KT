#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) {
    int t, x, y;
    cin >> t >> x >> y;
    if (t == 1)
      a[i] = x;
    else
      b[i] = x;
  }
  int sumA = 0, sumB = 0;
  for (int i = 0; i < n; i++) {
    sumA += a[i];
    sumB += b[i];
  }
  if (sumA * 2 > n * 10)
    cout << "LIVE" << endl;
  else
    cout << "DEAD" << endl;
  if (sumB * 2 > n * 10)
    cout << "LIVE" << endl;
  else
    cout << "DEAD" << endl;
}
