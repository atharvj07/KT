#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(m);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < m; i++) { cin >> b[i]; }
  int ans = -1;
  for (int i = 0; i < 101; i++) {
    bool ok1 = true, ok2 = false;
    for (int j = 0; j < n; j++) {
      if (a[j] > i) ok1 = false;
      if (a[j] * 2 <= i) ok2 = true;
    }
    for (int j = 0; j < m; j++) {
      if (b[j] <= i) ok1 = false;
    }
    if (ok1 && ok2) { ans = i; break; }
  }
  cout << ans << endl;
  return 0;
}
