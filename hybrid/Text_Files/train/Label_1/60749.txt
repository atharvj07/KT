n, k = map(int, input().split())
s = input()
g, l = [], []
for c in s: g += [int(c)]
for j in range (0, n, k):
  for i in range (j, j + k): l += [g[i]]
  for i in range (len(l) - 1):
    if l[i] >= l[i + 1]: continue
    for i in range (j, j + k): g[i] = l[i % (k + 1)]
    break
  l = []
s = ''.join([str(i) for i in g])
print(len(s))
print(s)
