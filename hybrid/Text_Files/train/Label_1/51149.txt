import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] b = new int[n];

            for (int j = 0; j < n; j++) {
                b[j] = sc.nextInt();
            }

            int[] a = new int[n];

            for (int j = 0; j < n / 2; j++) {
                if (j % 2 == 0) {
                    a[j] = b[n - 1 - j];
                } else {
                    a[n - 1 - j] = b[j];
                }
            }

            if (n % 2 == 1) {
                a[n / 2] = b[n / 2];
            }

            for (int j = 0; j < n; j++) {
                System.out.print(a[j] + " ");
            }
            System.out.println();
        }
    }
}
