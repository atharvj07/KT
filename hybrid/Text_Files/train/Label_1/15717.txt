#include <bits/stdc++.h>
using namespace std;
int main()
{
    int t, n, k;
    cin >> t;
    while (t--)
    {
        cin >> n >> k;
        if (k > (n - 1) / 2 || (n - k - 1) % 2 == 1)
        {
            cout << -1 << endl;
        }
        else
        {
            vector<int> a(n);
            int cnt = 1;
            while (cnt <= n && k)
            {
                a[--cnt] = cnt + 1;
                k--;
            }
            while (cnt++ <= n)
                a[--cnt] = cnt;
            for (int v : a)
                cout << v << ' ';
            cout << endl;
        }
    }
}
