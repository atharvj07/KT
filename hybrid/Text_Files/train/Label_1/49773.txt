import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

class Main {

    private static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
    private static int[] p, d;
    private static int inf = 1_000_000_000;

    private static int[] dijkstra(int n, int s, int[][] g) {
        int[] d = new int[n + 1];
        Arrays.fill(d, inf);
        d[s] = 0;

        PriorityQueue<Integer> pq = new PriorityQueue<Integer>();
        pq.add(s);

        while (!pq.isEmpty()) {
            int u = pq.poll();

            for (int v : g[u]) {
                if (d[v] > d[u] + 1) {
                    d[v] = d[u] + 1;
                    pq.add(v);
                }
            }
        }
        return d;
    }

    public static void main(String[] args) throws Exception {
        String[] parts = reader.readLine().split(" ");
        int n = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);
        int k = Integer.parseInt(parts[2]);
        int q = Integer.parseInt(parts[3]);

        p = new int[n + 1];
        d = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            p[i] = i;
        }

        int[][] g = new int[n + 1][];
        for (int i = 0; i <= n; i++) {
            g[i] = new int[0];
        }
        for (int i = 0; i < m; i++) {
            parts = reader.readLine().split(" ");
            int u = Integer.parseInt(parts[0]);
            int v = Integer.parseInt(parts[1]);
            int w = Integer.parseInt(parts[2]);

            g[u] = Arrays.copyOf(g[u], g[u].length + 1);
            g[u][g[u].length - 1] = v;

            g[v] = Arrays.copyOf(g[v], g[v].length + 1);
            g[v][g[v].length - 1] = u;
        }

        for (int i = 0; i < q; i++) {
            parts = reader.readLine().split(" ");
            int a = Integer.parseInt(parts[0]);
            int b = Integer.parseInt(parts[1]);

            System.out.println(dijkstra(n, a, g)[b]);
        }
    }
}
