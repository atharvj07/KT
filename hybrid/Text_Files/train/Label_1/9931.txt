#include <iostream>
#include <vector>

int main() {
  std::string s;
  std::cin >> s;
  int n = s.length();
  std::vector<bool> vowels(n, false);
  std::vector<bool> even(n, false);
  for (int i = 0; i < n; i++) {
    if (s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' || s[i] == 'u') {
      vowels[i] = true;
    }
    if (s[i] == '0' || s[i] == '2' || s[i] == '4' || s[i] == '6' || s[i] == '8') {
      even[i] = true;
    }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (vowels[i] && !even[i]) {
      ans++;
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
