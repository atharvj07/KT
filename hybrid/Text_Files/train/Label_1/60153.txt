import sys
input = sys.stdin.readline

n, m, k = map(int, input().strip().split())
a, b, c = [list(map(int, input().strip().split())) for _ in range(n)]
mp = [{} for _ in range(n)]
vv = []
for _ in range(m):
    u, v = map(int, input().strip().split())
    u, v = u-1, v-1
    if u in mp[v]: mp[v][u] = min(mp[v][u], a[u])
    else: mp[v][u] = a[u]
    vv.append((u, v))

dp = [[-1]*5001 for _ in range(n+1)]
dp[0][k] = 0
for i in range(n):
    for j in range(5001):
        if dp[i][j] == -1: continue
        for w in mp[i]:
            if j >= mp[i][w]:
                dp[i+1][j-mp[i][w]+b[i]] = max(dp[i+1][j-mp[i][w]+b[i]], dp[i][j]+c[i])
        dp[i+1][j+b[i]] = max(dp[i+1][j+b[i]], dp[i][j])
print(-1 if max(dp[n]) == -1 else max(dp[n]))
