#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;

  for (int i = 0; i < t; i++) {
    int x;
    cin >> x;

    int ans = 0;
    while (x > 0) {
      if (ans == 0) {
        x -= 2;
      } else {
        x -= ans * 6;
      }
      ans++;
    }

    cout << ans << '\n';
  }

  return 0;
}
