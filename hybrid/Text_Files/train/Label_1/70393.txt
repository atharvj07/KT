n, q = map(int, input().split())
s = input()
a, b, c = '', '', ''
for _ in range(q):
    t = input().split()
    if len(t) == 2:
        i, ch = t
        if i == '1': a += ch
        elif i == '2': b += ch
        else: c += ch
    else:
        i = t[0]
        if i == '1': a = a[:-1]
        elif i == '2': b = b[:-1]
        else: c = c[:-1]
    print('YES' if set(s) & set(a) & set(b) & set(c) == set() else 'NO')
