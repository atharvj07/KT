#include<bits/stdc++.h>
#include<unordered_map>
using namespace std;
int main()
{
    string s;
    cin>>s;
    unordered_map<char,int> mp;
    int cnt=0;
    for(auto x:s) mp[x]++;
    if(mp['B']>0 && mp['u']>0 && mp['l']>0 && mp['b']>0 && mp['a']>0 && mp['s']>0 && mp['a']>0 && mp['u']>0 && mp['r']>0)
    {
        int mn=1e9;
        for(auto x:s)
        {
            if(x=='B' || x=='u' || x=='l' || x=='b' || x=='a' || x=='s' || x=='a' || x=='u' || x=='r')
            {
                mn=min(mn,mp[x]);
            }
        }
        cnt=mn;
    }
    cout<<cnt<<endl;
}
