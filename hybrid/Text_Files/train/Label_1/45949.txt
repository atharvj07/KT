import java.math.BigInteger;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        List<BigInteger> snukeNumbers = new ArrayList<>();
        snukeNumbers.add(BigInteger.ONE);
        for (BigInteger i = BigInteger.valueOf(2); snukeNumbers.size() < k; i = i.add(BigInteger.ONE)) {
            if (isSnukeNumber(i)) {
                snukeNumbers.add(i);
            }
        }
        Collections.sort(snukeNumbers);
        for (BigInteger snukeNumber : snukeNumbers) {
            System.out.println(snukeNumber);
        }
    }

    private static boolean isSnukeNumber(BigInteger n) {
        BigInteger sumOfDigits = getSumOfDigits(n);
        for (BigInteger m = n.add(BigInteger.ONE); ; m = m.add(BigInteger.ONE)) {
            BigInteger sumOfDigitsOfM = getSumOfDigits(m);
            if (n.multiply(sumOfDigitsOfM).compareTo(m.multiply(sumOfDigits)) > 0) {
                return false;
            } else if (n.multiply(sumOfDigitsOfM).compareTo(m.multiply(sumOfDigits)) == 0) {
                return isSnukeNumber(m);
            } else {
                return true;
            }
        }
    }

    private static BigInteger getSumOfDigits(BigInteger n) {
        BigInteger sumOfDigits = BigInteger.ZERO;
        while (n.compareTo(BigInteger.ZERO) > 0) {
            sumOfDigits = sumOfDigits.add(n.mod(BigInteger.TEN));
            n = n.divide(BigInteger.TEN);
        }
        return sumOfDigits;
    }
}
