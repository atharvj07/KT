#include <bits/stdc++.h>
using namespace std;

int r, c, k, dr[] = {-1, 0, 1, 0}, dc[] = {0, 1, 0, -1}, a[2003][2003], dp[2003][2003];
char ans[2003][2003];
queue<pair<int, int>> q;

int main() {
    cin >> r >> c >> k;
    for (int i = 1; i <= r; i++)
        for (int j = 1; j <= c; j++) {
            char x;
            cin >> x;
            switch (x) {
            case '.':
                a[i][j] = 0;
                break;
            case 'L':
                a[i][j] = 1;
                q.push({i, j});
                break;
            case 'R':
                a[i][j] = 2;
                q.push({i, j});
                break;
            case 'U':
                a[i][j] = 3;
                q.push({i, j});
                break;
            case 'D':
                a[i][j] = 4;
                q.push({i, j});
                break;
            }
        }
    while (!q.empty()) {
        int size = q.size();
        while (size--) {
            pair<int, int> p = q.front();
            q.pop();
            for (int i = 0; i < 4; i++) {
                int nr = p.first + dr[i], nc = p.second + dc[i];
                if (nr > 0 && nr <= r && nc > 0 && nc <= c) {
                    if (a[nr][nc] == 0) {
                        q.push({nr, nc});
                        ans[nr][nc] = 'L' + i;
                    }
                    a[nr][nc] = a[p.first][p.second];
                }
            }
        }
    }
    for (int i = 2; i <= r; i++)
        for (int j = 1; j <= c; j++) dp[i][j] = dp[i - 1][j] + dp[i][j - 1] - dp[i - 1][j - 1];
    for (int i = 2; i <= r; i++)
        for (int j = 1; j <= c; j++)
            if (ans[i][j] == 'U') dp[i][j]++;
    for (int i = 1; i <= c; i++) cout << dp[r][i] << " ";
    return 0;
}
