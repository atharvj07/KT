#include <iostream>
#include <vector>
#include <tuple>
#include <algorithm>

using namespace std;

int main() {
    int q;
    cin >> q;
    for (int i = 0; i < q; i++) {
        int l1, r1, l2, r2;
        cin >> l1 >> r1 >> l2 >> r2;
        if (l1 < l2) swap(l1, l2), swap(r1, r2);
        if (l1 <= r2) {
            cout << l1 << ' ' << l2 << '\n' ;
        } else {
            cout << l1 << ' ' << r2 << '\n';
        }
    }
    return 0;
}
