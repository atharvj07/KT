n1, n2, n3 = map(int, input().split())
bag1 = list(map(int, input().split()))
bag2 = list(map(int, input().split()))
bag3 = list(map(int, input().split()))

# Find total sum and minimum element in each bag
total_sum = sum(bag1) + sum(bag2) + sum(bag3)
min1 = min(bag1)
min2 = min(bag2)
min3 = min(bag3)

# Find the maximum possible value
answer = max(total_sum - 2 * min(min1, min2, min3), total_sum - 2 * (min1 + min2 + min3 - max(min1, min2, min3)))
print(answer)
