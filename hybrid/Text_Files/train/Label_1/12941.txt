#include <iostream>
#include <vector>
#define MOD (int)(1e9 + 7)
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) cin >> p[i];
  vector<int> dp(n + 1, 1);
  for (int i = 0; i < n; i++) {
    dp[i + 1] = (2 * dp[i]) % MOD;
    if (i + 1 != p[i]) { dp[p[i]] = (dp[p[i]] + dp[i]) % MOD; }
  }
  cout << dp[n] << endl;
  return 0;
}
