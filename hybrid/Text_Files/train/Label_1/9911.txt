#include <bits/stdc++.h>
using namespace std;
typedef pair<int,int> ii;
int m,n,p[1005],vis[1005];
vector<int> g[1005];
void dfs(int x){if(vis[x])return;vis[x]=1;for(vector<int>::iterator it=g[x].begin();it!=g[x].end();++it)p[*it]=p[x],dfs(*it);}
int main(){int a,b;cin>>n>>m;for(int i=0;i<m;++i){cin>>a>>b;g[a].push_back(b);g[b].push_back(a);}for(int i=1;i<=n;++i)p[i]=i;for(int i=1;i<=n;++i)if(!vis[i]){dfs(i);if(count(p+1,p+n+1,p[i])>1){cout<<"no"<<endl;return 0;}}cout<<"yes"<<endl;}
