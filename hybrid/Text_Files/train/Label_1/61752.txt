def solve_kicker_problem():
    players = [list(map(int, input().split())) for _ in range(4)]
    team1_options = [(players[0][0], players[1][1]), (players[0][1], players[1][0])]
    team2_options = [(players[2][0], players[3][1]), (players[2][1], players[3][0])]

    def can_win(team_attack_defense, opponent_options):
        return any(team_attack_defense[0] > opponent[1] and team_attack_defense[1] > opponent[0] for opponent in opponent_options)

    if any(can_win(opt, team2_options) for opt in team1_options):
        if not any(can_win(opt, team1_options) for opt in team2_options):
            return "Team 1"
    if any(can_win(opt, team1_options) for opt in team2_options):
        return "Team 2"
    return "Draw"

print(solve_kicker_problem())
