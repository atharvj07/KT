#include<bits/stdc++.h>
using namespace std;
#define maxn 1505

int n,s,m,k,a[maxn],b[maxn],p[maxn],l[maxn],r[maxn],ans=-1;
int main(){
    cin>>n>>s>>m>>k;
    for(int i=1;i<=n;i++) cin>>a[i];
    int tot=0;
    for(int i=1;i<=s;i++){
        cin>>l[i]>>r[i];
        for(int j=l[i];j<=r[i];j++) p[j]=1,b[++tot]=a[j];
    }
    sort(b+1,b+1+tot);
    tot=unique(b+1,b+1+tot)-b-1;
    if(m*k>tot) cout<<-1;
    else{
        int le=1,ri=k*m,mi;
        while(le<=ri){
            mi=(le+ri)>>1;
            l[0]=1; r[0]=tot;
            for(int i=1;i<=m;i++){
                int j=lower_bound(b+1,b+1+tot,b[l[i-1]+mi])-b;
                if(j-l[i-1]+r[i-1]-j+1>=k) l[i]=j,r[i]=r[i-1];
                else{le=mi+1; break;}
            }
            if(le!=mi+1) ri=mi-1,ans=b[l[m]+k-1];
        }
        cout<<ans;
    }
}
