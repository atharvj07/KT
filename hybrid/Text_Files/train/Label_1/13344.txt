#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  ll t, n, m, x, p = 0;
  cin >> t;
  while (t--) {
    p = 0;
    cin >> n >> m;
    vector<ll> v(n), c(m);
    for (auto &x : v) cin >> x;
    for (auto &x : c) cin >> x;
    sort(v.begin(), v.end());
    sort(c.begin(), c.end());
    for (ll i = 0; i < min(m, n); i++) {
      p += min(c[i], c[v[i]]);
    }
    cout << p << "\n";
  }
  return 0;
}
