#include<bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;
map<string, bool> mp;
int32_t main() {
    mp[""] = true;
    mp["0"] = true, mp["1"] = true;
    mp["00"] = true, mp["01"] = true, mp["10"] = true, mp["11"] = true;
    mp["000"] = true, mp["001"] = true, mp["010"] = true, mp["011"] = true;
    mp["100"] = true, mp["101"] = true, mp["110"] = false, mp["111"] = false;

    int m; cin >> m;
    string s = "";
    while(m--) {
        string x; cin >> x;
        s += x;
        int l = s.length();
        int ans = 0;
        for(int i = 0; i < l; i++) {
            string sub = "";
            for(int j = i; j < l; j++) {
                sub += s[j];
                if(mp[sub] == true) ans++;
                ans %= mod;
            }
        }
        cout << ans << endl;
    }
}
