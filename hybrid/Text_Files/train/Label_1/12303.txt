#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;cin>>n;
    stack<int>s;
    vector<int>v;
    queue<int>q;
    while(n--){
        string c;cin>>c;
        if(c=="+") s.push(n+1);
        else if(c=="-"){
            int x;cin>>x;q.push(x);
            while(!q.empty()){
                if(!s.empty()&&s.top()==q.front()){
                    v.push_back(q.front());
                    s.pop();
                    q.pop();
                }
                else break;
            }
        }
    }
    if(s.empty()&&q.empty()){
        cout<<"YES"<<endl;
        for(auto x:v) cout<<x<<" ";
    }
    else cout<<"NO";
}
