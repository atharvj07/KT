#include <iostream>
#include <vector>
#include <map>
using namespace std;
typedef long long ll;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int m, k;
        cin >> m >> k;
        vector<int> count(k), dish, pos;
        for (int i=0; i<k; i++) cin >> count[i];
        for (int j=0; j<m-1; j++) {
            int d, dis;
            cin >> d >> dis;
            dish.push_back(d);
            pos.push_back(dis);
        }
        map<int,int> cnt;
        for (int i=0; i<m-1; i++) {
            if (dish[i]>0) {
                if (cnt[dish[i]] == 0) {
                    count[dish[i]-1] -= pos[i];
                    if (count[dish[i]-1]<=0) count[dish[i]-1] = 0;
                }
            }
            cnt[dish[i]] = 1;
        }
        for (int i=0; i<k; i++) {
            if (count[i]>0) cout << 'N';
            else cout << 'Y';
        }
        cout << endl;
    }
    return 0;
}
