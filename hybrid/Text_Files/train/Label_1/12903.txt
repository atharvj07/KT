#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

vector<long long> x, y;
int n;

int main() {
  cin >> n;
  x.resize(n);
  y.resize(n);
  for (int i = 0; i < n; i++) cin >> x[i] >> y[i];
  sort(x.begin(), x.end());
  sort(y.begin(), y.end());
  if (x[n - 1] <= y[n - 1]) {
    cout << x[n - 1] * 2 << endl;
  } else {
    cout << y[n - 1] * 2 << endl;
  }
  return 0;
}
