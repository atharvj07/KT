import java.util.*;
import java.io.*;

class Main {
   public static void main(String[] args) throws IOException {
      BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
      int n = Integer.parseInt(br.readLine());
      String[] x = br.readLine().split(" ");
      int p = Integer.parseInt(x[0]);
      boolean[] bx = new boolean[n + 1];
      for (int i = 1; i <= p; i++) bx[Integer.parseInt(x[i])] = true;
      String[] y = br.readLine().split(" ");
      int q = Integer.parseInt(y[0]);
      boolean[] by = new boolean[n + 1];
      for (int i = 1; i <= q; i++) by[Integer.parseInt(y[i])] = true;
      boolean flag = true;
      for (int i = 1; i <= n; i++) flag &= bx[i] || by[i];
      System.out.println((flag ? "I become the guy." : "Oh, my keyboard!"));
   }
}
