#include <cstdio>
#include <vector>
using namespace std;

typedef long long ll;

int main() {
  int n, b, p;
  scanf("%d%d%d", &n, &b, &p);
  vector<ll> v;
  ll x, y;
  x = y = 0;
  while (n > 1) {
    x += b * (n + 1) / 2, y += p * (n + 1) / 2;
    v.push_back(n);
    n /= 2;
  }
  for (size_t i = 0; i < v.size(); ++i) {
    y += v[i] * p;
    if (i == v.size() - 1) x += v[i] * b;
  }
  printf("%lld %lld\n", x, y);
}
