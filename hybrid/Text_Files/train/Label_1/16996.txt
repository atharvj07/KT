#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long n,x,i,j,sum=0;
    cin>>n>>x;
    long long b[n];
    for(i=0;i<n;++i){
        cin>>b[i];
    }
    sort(b,b+n);
    for(i=n-1;i>=0;--i){
        sum=sum+b[i]*x;
        x--;
        x=x<1?1:x;
    }
    cout<<sum<<endl;
    return 0;
}
