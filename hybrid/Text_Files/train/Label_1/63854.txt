n = int(input())
a = [0]
b = [0]
for i in range(n):
    x, y = map(int, input().split())
    a.append(x)
    b.append(y)

s_a = [0] * (n + 1)
s_b = [0] * (n + 1)
for i in range(1, n + 1):
    s_a[i] = s_a[i - 1] + a[i]
    s_b[i] = s_b[i - 1] + b[i]

ans = 1e18
for i in range(1, n + 1):
    res = s_a[i - 1] * (n - i) + s_b[n] - s_b[i]
    ans = min(ans, res)
print(ans)
