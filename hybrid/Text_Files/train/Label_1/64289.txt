import sys

# read input
tokens = sys.stdin.readline().split()
n, m = int(tokens[0]), int(tokens[1])
caffeine = list(map(int, sys.stdin.readline().split()))

# cumulative sum of caffeine
caffeine = [0] + caffeine
for i in range(1, n + 1):
    caffeine[i] += caffeine[i - 1]

# binary search for the minimum number of days
l, r = 1, m
while l < r:
    mid = (l + r) // 2
    # check if it is possible to write the coursework in mid days
    if caffeine[n] >= mid * (mid + 1) // 2:
        r = mid
    else:
        l = mid + 1

# print the result
print(l if l <= m else -1)
