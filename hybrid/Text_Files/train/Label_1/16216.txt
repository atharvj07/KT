#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  map<string, set<string>> phones;
  for (int i = 0; i < n; i++) {
    string name, number;
    cin >> name;
    int m;
    cin >> m;
    for (int j = 0; j < m; j++) {
      cin >> number;
      string origin = number;
      bool flag = 0;
      for (auto it : phones[name]) {
        if (it.size() >= number.size() and
            it.substr(it.size() - number.size()) == number) {
          flag = 1;
          break;
        }
      }
      if (!flag) phones[name].insert(origin);
    }
  }
  cout << phones.size() << '\n';
  for (auto it : phones) {
    cout << it.first << ' ' << it.second.size() << ' ';
    for (auto k : it.second) cout << k << ' ';
    cout << '\n';
  }
  return 0;
}
