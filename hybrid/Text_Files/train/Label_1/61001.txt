import sys
from collections import deque
def bfs(s,r):
    q=deque([(s,0)])
    vis[s]=1
    while(len(q)>0):
        x,d=q.popleft()
        r[x]=d
        for y in g[x]:
            if(vis[y]==0):
                q.append((y,d^1))
                vis[y]=1

n,m=map(int,sys.stdin.readline().split())
g=[[] for _ in range(n+1)]
for _ in range(m):
    x,y=map(int,sys.stdin.readline().split())
    g[x].append(y)
    g[y].append(x)
    
vis=[0]*(n+1)
r=[0]*(n+1)
