import java.util.*;
import java.io.*;
class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter out = new BufferedWriter(new OutputStreamWriter(System.out));
        String[] parts = in.readLine().split(" ");
        int n = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);
        int[][] votes = new int[m][2];
        for (int i = 0; i < n; i++) {
            parts = in.readLine().split(" ");
            votes[Integer.parseInt(parts[0])-1][0]++;
            votes[Integer.parseInt(parts[0])-1][1] = Math.min(votes[Integer.parseInt(parts[0])-1][1], Integer.parseInt(parts[1]));
        }
        Arrays.sort(votes, new Comparator<int[]>() {
            @Override
            public int compare(int[] a, int[] b) {
                return a[0] - b[0];
            }
        });
        int ans = 0, max = 0;
        for (int i = 0; i < m; i++) {
            if (i == 0) {
                max = votes[i][0];
            } else {
                max = Math.max(max, votes[i][0]);
                max++;
            }
            if (votes[i][0] + 1 <= max) {
                ans += votes[i][1];
            }
        }
        out.write(String.valueOf(ans) + "\n");
        out.close();
    }
}
