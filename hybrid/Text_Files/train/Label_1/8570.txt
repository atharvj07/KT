#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    long x, y, n;
    cin >> n;
    vector<pair<long, long>> v;
    while (n--) {
        cin >> x >> y;
        v.push_back({x, y});
    }
    sort(v.begin(), v.end());
    long ans = 1, l = v[0].first, r = v[0].second, cnt = 0;
    for (auto i : v) {
        l = max(l, i.first);
        r = min(r, i.second);
        if (l > r) {
            l = i.first, r = i.second;
            cnt++;
        }
        if (cnt > 1) break;
    }
    cout << ans + cnt << endl;
    return 0;
}
