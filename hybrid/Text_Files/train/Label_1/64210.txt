def can_buy(k, n, S, base_costs):
    costs = [(base_costs[i] + (i + 1) * k) for i in range(n)]
    costs.sort()
    total_cost = sum(costs[:k])
    return total_cost <= S, total_cost

def solve(n, S, base_costs):
    left, right = 0, n
    max_k, min_cost = 0, 0
    while left <= right:
        mid = (left + right) // 2
        can, cost = can_buy(mid, n, S, base_costs)
        if can:
            max_k, min_cost = mid, cost
            left = mid + 1
        else:
            right = mid - 1
    return max_k, min_cost

n, S = map(int, input().split())
base_costs = list(map(int, input().split()))
print(*solve(n, S, base_costs))
