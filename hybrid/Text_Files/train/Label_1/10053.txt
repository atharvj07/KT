#include <cstdio>

typedef unsigned long long u64;

u64 pow(u64 a, u64 b) {
  u64 r = 1;
  while (b) {
    if (b & 1) r = r * a;
    a = a * a;
    b >>= 1;
  }
  return r;
}
u64 modinv(u64 a) {
  return pow(a, -1) % 998244353ull;
}
u64 n, m, l, r;

int main() {
  scanf("%lld %lld %lld %lld", &n, &m, &l, &r);
  u64 r_m1 = modinv(r - 1);
  u64 ans = (r - l + 1) * (r - l) % 998244353ull;
  ans = ans * pow(r_m1, 2) % 998244353ull;
  ans = ans * (n <= m ? n : m) % 998244353ull;
  printf("%lld\n", ans);
  return 0;
}
