#include <bits/stdc++.h>
using namespace std;
long long a[100001];
vector<long long>v[100001];
long long visited[100001], flag = 0;
long long XOR = 0;
void dfs(long long node, long long par, long long x) {
    visited[node] = 1;
    XOR = XOR ^ a[node];
    if (v[node].size() > 1)flag = 1;
    for (long long i = 0; i < v[node].size(); i++) {
        if (visited[v[node][i]] == 0)dfs(v[node][i], node, x);
    }
}
int main() {
    long long t, n, k, x, ans = 0;
    cin >> t;
    while (t--) {
        cin >> n >> k;
        for (long long i = 1; i <= n; i++) { cin >> a[i]; visited[i] = 0; }
        for (long long i = 0; i <= n; i++)v[i].clear();
        flag = 0;
        for (long long i = 0; i < n - 1; i++) {
            cin >> x;
            v[x].push_back(i + 2);
            v[i + 2].push_back(x);
        }
        dfs(1, 1, 1);
        if (flag == 0)cout << "YES" << endl;
        else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
