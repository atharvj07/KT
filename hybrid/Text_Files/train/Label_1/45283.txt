import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long x = sc.nextLong();
            if (x % 4 == 0 || x % 4 == 3) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
