#include<bits/stdc++.h>
using namespace std;
typedef unsigned long long ll;

{
    ll n;
    cin>>n;
    if(n==1)
    {
        cout<<"1"<<endl;
        return 0;
    }
    else if(n%2==0)
    {
        cout<<"1"<<endl;
        return 0;
    }
    else
    {
        cout<<"2"<<endl;
        return 0;
    }
}
