#include <bits/stdc++.h>

#define rep(i, a, b) for (int i = (a), i##_end_ = (b); i < i##_end_; ++i)
#define rer(i, a, b, step) for (int i = (a), i##_end_ = (b); i <= i##_end_; i += step)

using namespace std;

typedef long long ll;

const int N = 55;

int a[N], n, sum[N], ans[2][N], k;
map<int, vector<int>> pos;
bool mark[N][N], used[N];

int main() {
  cin >> n;
  rep(i, 1, n + 1) cin >> a[i];
  rep(i, 1, n + 1) sum[i] = sum[i - 1] + a[i];
  rep(i, 1, n + 1) pos[sum[i]].push_back(i);
  rep(i, 1, n + 1) {
    int p = 0;
    while (p < n + 1 && sum[p] + sum[n] - sum[p + i] < 0) ++p;
    if (p == n + 1) continue;
    k++;
    ans[0][k] = p + 1;
    ans[1][k] = p + i;
    used[p + 1] = used[p + i] = true;
    rep(j, p + 1, p + i + 1) mark[j][j] = true;
  }
  rep(i, 1, n + 1) if (!used[i]) {
    int tmp = pos[sum[i]][0], p = pos[sum[i]][1], st = 0;
    while (p < n + 1) {
      while (p < n + 1 && st < p && mark[st + 1][p]) ++p;
      if (p < n + 1) {
        k++;
        ans[0][k] = st + 1;
        ans[1][k] = p;
        used[st + 1] = used[p] = true;
        st = p + 1;
        p = st;
      }
      while (p < n + 1 && mark[tmp + 1][p]) ++p;
      if (p < n + 1) {
        k++;
        ans[0][k] = tmp + 1;
        ans[1][k] = p;
        used[tmp + 1] = used[p] = true;
        tmp = p;
        p = st;
      }
    }
  }
  cout << k << endl;
  rep(i, 1, k + 1) cout << ans[0][i] << ' ' << ans[1][i] << endl;
  return 0;
}
