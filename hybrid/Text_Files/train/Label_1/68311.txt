T = int(input().strip())
for _ in range(T):
    h, c, t = map(int, input().split())
    
    if t == h:
        print(1)
    elif (h + c) / 2 >= t:
        print(2)
    else:
        x = (t - c) / (2*t - h - c)
        k = int(x)
        if abs(((k+1)*h + k*c) / (2*k + 1) - t) < abs(((k)*h + (k-1)*c) / (2*k - 1) - t):
            print(2*k + 1)
        else:
            print(2*k - 1)
