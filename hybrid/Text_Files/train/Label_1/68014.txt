t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    total_xor = 0
    for num in a:
        total_xor ^= num
    if total_xor == 0:
        print("YES")
    else:
        prefix_xor = 0
        found = False
        for i in range(n):
            prefix_xor ^= a[i]
            if prefix_xor == total_xor:
                suffix_xor = 0
                for j in range(i + 1, n):
                    suffix_xor ^= a[j]
                    if suffix_xor == total_xor:
                        found = True
                        break
            if found:
                break
        print("YES" if found else "NO")
