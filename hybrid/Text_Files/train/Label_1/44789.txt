import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] ciel = new int[n];
    int[] jiro = new int[n];
    for (int i = 0; i < n; i++) {
      int size = sc.nextInt();
      for (int j = 0; j < size; j++) {
        int val = sc.nextInt();
        if (j % 2 == 0) {
          ciel[i] += val;
        } else {
          jiro[i] += val;
        }
      }
    }
    int cielSum = 0, jiroSum = 0;
    for (int i = 0; i < n; i++) {
      if (ciel[i] > jiro[i]) {
        cielSum += ciel[i];
      } else {
        jiroSum += jiro[i];
      }
    }
    System.out.println(cielSum + " " + jiroSum);
  }
}
