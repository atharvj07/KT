#include<bits/stdc++.h>
using namespace std;
#define N 15
int main() {
    int n, i, j, a[N][N], dist[N][N];
    cin >> n;
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) { cin >> a[i][j]; }
    }
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) { dist[i][j] = a[i][j]; }
    }
    int k;
    for (k = 0; k < n; k++) {
        for (i = 0; i < n; i++) {
            for (j = 0; j < n; j++) { dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]); }
        }
    }
    int mx = 0;
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) { mx = max(mx, dist[i][j]); }
    }
    cout << mx << endl;
    return 0;
}
