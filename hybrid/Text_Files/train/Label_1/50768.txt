import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        Arrays.sort(a);

        int maxNegative = 0;
        int maxPositive = 0;
        int negativeCount = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] < 0) {
                negativeCount++;
                maxNegative = Math.max(maxNegative, a[i]);
            } else {
                maxPositive = Math.max(maxPositive, a[i]);
            }
        }

        int result = 0;
        if (k % 2 == 0) {
            result += maxPositive * n;
        } else {
            if (negativeCount > 0) {
                result += maxPositive * (n - 1) + maxNegative;
            } else {
                result += maxPositive * n;
            }
        }

        System.out.println(result);
    }
}
