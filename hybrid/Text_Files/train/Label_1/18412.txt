#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k;
    cin >> n >> k;
    vector<long long> h(n);
    for (int i = 0; i < n; i++) { cin >> h[i]; }
    long long ans = 1;
    for (int i = 0; i < n - 1; i++) {
      if (k == 0) {
        break;
      }
      if (h[i] >= h[i + 1]) {
        ans++;
      } else {
        h[i]++;
      }
      k--;
    }
    if (k > 0) {
      cout << -1 << endl;
    } else {
      cout << ans << endl;
    }
  }
}
