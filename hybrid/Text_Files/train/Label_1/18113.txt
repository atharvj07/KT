#include <cstdio>
#include <vector>
#include <algorithm>
const unsigned int MM = 2e5;
int main() {
    unsigned int n, k;
    int a[MM]; char c[MM];
    std::scanf("%u %u", &n, &k);
    for (unsigned int i = 0; i < n; ++i)
        std::scanf("%d", &a[i]);
    std::scanf("%s", c);
    std::vector<int> r, o, w;
    for (unsigned int i = 0; i < n; ++i) {
        switch (c[i]) {
        case 'R':
            r.push_back(a[i]);
            break;
        case 'O':
            o.push_back(a[i]);
            break;
        case 'W':
            w.push_back(a[i]);
            break;
        }
    }
    if (k == n && w.size()) return std::puts("-1"), 0;
    std::sort(r.rbegin(), r.rend());
    std::sort(o.rbegin(), o.rend());
    std::sort(w.rbegin(), w.rend());
    int res = 0, p1 = 0, p2 = 0, p3 = 0;
    if (k & 1 && w.size()) {
        res += w[p3++];
        --k;
    }
    while (k > 0) {
        if ((k > 1) && (p1 < (int)r.size()) && (p2 < (int)o.size())) {
            if (r[p1] == o[p2]) {
                ++p1; ++p2;
            } else if (r[p1] > o[p2]) {
                res += r[p1++];
            } else {
                res += o[p2++];
            }
            k -= 2;
        } else if (p1 < (int)r.size()) {
            res += r[p1++];
            k--;
        } else if (p2 < (int)o.size()) {
            res += o[p2++];
            k--;
        } else {
            break;
        }
    }
    if (k > 1) res = -1;
    std::printf("%d", res);
    return 0;
}
