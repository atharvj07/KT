import sys
n, m, k = map(int, input().split())
arr = [list(input()) for _ in range(n)]
ans = [0] * m
for i in range(n-1, -1, -1):
    for j in range(m):
        if arr[i][j] != '.':
            ans[j] += 1
            if arr[i][j] == 'D':
                if i + 1 < n and arr[i+1][j] == '.':
                    arr[i+1][j] = 'U'
            elif arr[i][j] == 'U':
                if i - 1 >= 0 and arr[i-1][j] == '.':
                    arr[i-1][j] = 'D'
            elif arr[i][j] == 'L':
                if j + 1 < m and arr[i][j+1] == '.':
                    arr[i][j+1] = 'R'
            else:
                if j - 1 >= 0 and arr[i][j-1] == '.':
                    arr[i][j-1] = 'L'
print(*ans)
