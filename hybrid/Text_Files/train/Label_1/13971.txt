#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n, k, a, b = 0, c, d = 0;
    cin >> n >> k;
    for (int i = 0; i < k; i++) {
        cin >> a;
        if (n % a == 0) b = n / a, c = a;
        else if (n % a > d) d = n % a, c = a;
    }
    cout << c << " " << b << endl;
}
