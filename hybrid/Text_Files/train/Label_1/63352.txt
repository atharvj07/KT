import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    adj = [[] for _ in range(n+1)]
    for _ in range(n-1):
        a, b = map(int, input().split())
        adj[a].append(b)
        adj[b].append(a)

    vis = [0] * (n+1)
    ans = 0

    def dfs(node, par):
        nonlocal ans
        vis[node] = 1
        sz = 1
        for u in adj[node]:
            if u == par: continue
            sz += dfs(u, node)
        ans = max(ans, sz)
        return sz

    dfs(1, 1)
    print(ans)

for _ in range(int(input())):
    solve()
