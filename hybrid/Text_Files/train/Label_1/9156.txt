#include <iostream>
#include <algorithm>

using namespace std;

const int MAXN = 2e5 + 10;
pair<int, int> a[MAXN];
int main() {
    int n, sx, sy;
    cin >> n >> sx >> sy;
    for (int i = 1; i <= n; ++i) cin >> a[i].first >> a[i].second;
    sort(a + 1, a + n + 1);
    if (n == 1) {
        cout << 1 << '\n' << a[1].first << ' ' << a[1].second << '\n';
        return 0;
    }
    int l = 1, r = n, x, y;
    while (l <= r) {
        int mid = (l + r) >> 1;
        if (a[mid].first >= sx) {
            x = a[mid].first - 1;
            r = mid - 1;
        } else {
            l = mid + 1;
        }
    }
    l = 1, r = n;
    while (l <= r) {
        int mid = (l + r) >> 1;
        if (a[mid].second >= sy) {
            y = a[mid].second - 1;
            r = mid - 1;
        } else {
            l = mid + 1;
        }
    }
    int ans = 1;
    for (int i = 1; i <= n; ++i)
        if (a[i].first <= x && a[i].second <= y) ++ans;
    cout << ans << '\n' << x << ' ' << y << '\n';
    return 0;
}
