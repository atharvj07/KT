#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int a = 0, b = 0, c = 0;
  for (char x : s) {
    a += (x == '8');
    b += (x == '0');
    c += (x == '1' || x == '2' || x == '3' || x == '4');
  }
  if ((a == 1 && b >= 4 && c >= 6) || (a == 1 && b >= 3 && c == n))
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
