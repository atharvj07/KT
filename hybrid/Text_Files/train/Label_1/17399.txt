#include<bits/stdc++.h>
const int M=1000000007;
int a[1002],b[1002];
int main(){
    int n,k,i,j;
    scanf("%d%d",&n,&k);
    a[0]=1;b[0]=1;
    for(i=1;i<=k;i++){
        a[i]=1LL*a[i-1]*(n-i)%M;
        b[i]=1LL*b[i-1]*(n-i+1)%M;
    }
    if(n==k)printf("%d\n",a[k]);
    else printf("%d\n",1LL*a[k]*b[n-k-1]%M);
    return 0;
}
