def check(field, n, m, k):
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if field[i][j] == '*':
                if i - k + 1 <= 0 or j - k + 1 <= 0 or i + k - 1 > n or j + k - 1 > m:
                    return "NO"
                for h in range(k):
                    if field[i - h][j + h] != '*' or field[i - h][j - h] != '*':
                        return "NO"
    return "YES"

t = int(input())
for _ in range(t):
    n, m, k = map(int, input().split())
    field = [['.' for _ in range(m + 2)] for _ in range(n + 2)]
    for i in range(1, n + 1):
        field[i][1:m + 1] = list(input())
    print(check(field, n, m, k))
