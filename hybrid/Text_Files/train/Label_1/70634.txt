n, m = map(int,input().split())
a = list(map(int,input().split()))
b = list(map(int,input().split()))
c = sorted(((a[i],i) for i in range(n)),reverse=True)
b.sort()
l, r = 0, 0
for i in range(m):
    while l<n and a[c[l][1]]<b[i]:
        l+=1
    if l==n:
        break
    a[c[l][1]], b[i] = b[i], a[c[l][1]]
    r+=1
q = int(input())
for j in range(q):
    t, u, v = map(int,input().split())
    if t==1:
        for i in range(l,n):
            if a[c[i][1]]==u:
                c[i]=(v,c[i][1])
                break
    else:
        b[u-1]=v
        while r>0 and a[c[l][1]]<b[m-r]:
            l+=1
            r-=1
    print(a[c[l][1]] if l<n else -1)
