def main():
    n, m, q = map(int, input().split())
    arr = list(map(int, input().split()))
    count = [0] * m
    for i in arr:
        count[i - 1] += 1
    for i in range(q):
        k = int(input()) - n
        ans = 1
        for j in range(m):
            if count[j] < count[ans - 1] or (count[j] == count[ans - 1] and j + 1 < ans):
                ans = j + 1
        print(ans)

main()
