import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[12];
        for (int i = 0; i < 12; i++) {
            arr[i] = sc.nextInt();
        }
        int a = 0, b = 0, c = 0;
        for (int i = 0; i < 12; i++) {
            if (arr[i] == a) {
                a++;
            } else if (arr[i] == b) {
                b++;
            } else if (arr[i] == c) {
                c++;
            }
        }
        System.out.println(a >= 4 && b >= 4 && c >= 4 ? "yes" : "no");
    }
}
