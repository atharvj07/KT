import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // read the input
        long n = sc.nextLong();
        long m = sc.nextLong();
        long a = sc.nextLong();
        long b = sc.nextLong();

        // check if it is possible to distribute the boxes equally
        if (n % m != 0) {
            // it is not possible, so we need to build or demolish some boxes
            // calculate the minimum number of boxes that need to be built or demolished
            long diff = Math.abs(n - (n / m) * m);
            // calculate the minimum cost of building or demolishing boxes
            long cost = Math.min(diff * a, (n - diff) * b);
            // print the minimum cost
            System.out.println(cost);
        } else {
            // it is already possible, so no need to build or demolish boxes
            // print the minimum cost (which is 0)
            System.out.println(0);
        }
    }
}
