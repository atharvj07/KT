#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pii pair<int, int>
#define f(i, n) for (int i = 0; i < n; i++)
#define ff(i, a, b) for (int i = a; i <= b; i++)
int solve(vector<int> g[], int n, int k, vector<int> x) {
    int deg[n + 1] = {0}, par[n + 1];
    f(i, n + 1) par[i] = i;
    f(i, n - 1) {
        int a, b;
        cin >> a >> b;
        if (a == b)
            continue;
        g[a].push_back(b);
        g[b].push_back(a);
        deg[a]++;
        deg[b]++;
    }
    f(i, k) {
        deg[x[i]]--;
        if (deg[x[i]] == 0)
            return 1;
    }
    int mn = 1;
    f(i, n + 1) {
        if (deg[i] == 1) {
            if (i == 1)
                continue;
            mn = i;
        }
    }
    int ans = 0;
    f(i, k) {
        int path = mn;
        while (path != 1 && path != x[i]) {
            path = par[path];
        }
        if (path == 1) {
            ans = 1;
            break;
        }
    }
    return ans;
}
int main() {
    int t, n, k;
    cin >> t;
    while (t--) {
        cin >> n >> k;
        vector<int> x(k);
        f(i, k) { cin >> x[i]; }
        vector<int> g[n + 1];
        int ans = solve(g, n, k, x);
        if (ans)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
