#include <iostream>
int main() {
  std::string a, b;
  std::cin >> a >> b;
  if (a == b) {
    std::cout << "YES";
  } else {
    int n = a.length();
    for (int i = 0; i < n / 2; i++) {
      if ((a[i] == a[i + n / 2] && b[i] == b[i + n / 2]) ||
          (a[i] == b[i + n / 2] && b[i] == a[i + n / 2])) {
        std::cout << "YES";
        return 0;
      }
    }
    std::cout << "NO";
  }
}
