import java.util.*;
import java.lang.*;
import java.io.*;

public class Main {

    static class Edge implements Comparable<Edge> {
        int u;
        int v;
        long w;

        public Edge(int u, int v, long w) {
            this.u = u;
            this.v = v;
            this.w = w;
        }

        @Override
        public int compareTo(Edge o) {
            return Long.compare(this.w, o.w);
        }
    }

    static long[] dist;
    static int[] p;
    static ArrayList<Edge> edges;

    static void dijkstra(int s, int n) {
        dist = new long[n];
        p = new int[n];
        for (int i = 0; i < n; i++) {
            dist[i] = Long.MAX_VALUE;
            p[i] = -1;
        }
        dist[s] = 0;
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        pq.add(new Edge(s, s, 0));
        while (!pq.isEmpty()) {
            Edge e = pq.poll();
            if (dist[e.u] != e.w) continue;
            for (Edge ee : edges) {
                if (ee.u == e.u) {
                    if (dist[ee.v] > dist[ee.u] + ee.w) {
                        dist[ee.v] = dist[ee.u] + ee.w;
                        p[ee.v] = ee.u;
                        pq.add(new Edge(ee.u, ee.v, dist[ee.v]));
                    }
                } else if (ee.v == e.u) {
                    if (dist[ee.u] > dist[ee.v] + ee.w) {
                        dist[ee.u] = dist[ee.v] + ee.w;
                        p[ee.u] = ee.v;
                        pq.add(new Edge(ee.v, ee.u, dist[ee.u]));
                    }
                }
            }
        }
    }

    static long dfs(int u, int t, long w) {
        if (u == t) return w;
        long res = Long.MAX_VALUE;
        for (Edge e : edges) {
            if (e.u == u) {
                if (dist[u] + e.w + dist[e.v] == dist[t]) {
                    long nw = e.w;
                    if (e.w == 0) {
                        for (long i = 1; i <= 1000000000000000000L; i++) {
                            if (dist[u] + i + dist[e.v] == dist[t]) {
                                nw = i;
                                break;
                            }
                        }
                    }
                    res = Math.min(res, dfs(e.v, t, nw));
                }
            } else if (e.v == u) {
                if (dist[u] + e.w + dist[e.v] == dist[t]) {
                    long nw = e.w;
                    if (e.w == 0) {
                        for (long i = 1; i <= 1000000000000000000L; i++) {
                            if (dist[u] + i + dist[e.v] == dist[t]) {
                                nw = i;
                                break;
                            }
                        }
                    }
                    res = Math.min(res, dfs(e.v, t, nw));
                }
            }
        }
        return res;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        long l = in.nextInt();
        int s = in.nextInt();
        int t = in.nextInt();
        edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            long w = in.nextInt();
            edges.add(new Edge(u, v, w));
        }
        dijkstra(s, n);
        if (dist[t] != l) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        for (Edge e : edges) {
            if (e.w == 0) {
                e.w = dfs(e.u, e.v, 0);
            }
            System.out.println(e.u + " " + e.v + " " + e.w);
        }
    }
}
