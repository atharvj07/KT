#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 5;
int p[N], a[N], b[N], c[N];
int32_t main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> p[i];
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    int m;
    cin >> m;
    for (int i = 0; i < m; i++) cin >> c[i];
    int last = -1;
    for (int i = 0; i < m; i++) {
        int ans = INT_MAX;
        for (int j = 0; j < n; j++) {
            if (a[j] == c[i] || b[j] == c[i]) {
                if (p[j] < ans) {
                    ans = p[j];
                    last = j;
                }
            }
        }
        if (last == -1)
            cout << "-1 ";
        else
            cout << p[last] << " ", last = -1;
    }
    cout << endl;
}
