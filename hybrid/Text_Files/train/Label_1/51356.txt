import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long k = sc.nextLong();
        if (k < n - 1) {
            System.out.println(-1);
            return;
        }
        long ans = 0;
        for (long i = 2; i <= k; i++) {
            n -= n / i;
            ans++;
        }
        if (n > 1) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }
}
