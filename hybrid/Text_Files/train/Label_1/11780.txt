#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  long long n, k, x;
  cin >> n >> k >> x;
  vector<long long> a(n);
  for (long long i = 0; i < n; ++i) cin >> a[i];
  vector<pair<long long, long long>> f; // {x, remove}
  for (long long i = 0; i < n; ++i) if (a[i] % x != 0) {
    f.push_back({x - a[i] % x, i});
  }
  sort(f.begin(), f.end());
  for (long long i = 0; i < k && i < (long long)f.size(); ++i) a[f[i].second] += f[i].first;
  for (long long i = 0; i < n; ++i) cout << a[i] << ' ';
  return 0;
}
