n = int(input())
times = [tuple(map(int, input().split())) for _ in range(n)]

max_cashes = 1
current_cashes = 1

for i in range(1, n):
    if times[i] == times[i - 1]:
        current_cashes += 1
        max_cashes = max(max_cashes, current_cashes)
    else:
        current_cashes = 1

print(max_cashes)
