#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int ans = 0, cnt = 0;
    for (int l = 0, r = 0; r < n; r++) {
        cnt += (s[r] != s[l]);
        while (cnt > k) cnt -= (s[r] != s[l++]);
        ans = max(ans, r - l + 1);
    }
    cout << ans << endl;
    return 0;
}
