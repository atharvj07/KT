n = int(input())
s = input()
op = 0
cl = 0
mx = 0
cnt = 0
ans1 = 0
ans2 = 0
for i in s:
    if i=='(':
        op += 1
    if i == ')':
        cl+=1
        if mx < cnt:
            mx = cnt
            ans1 = mx
        cnt = 0
    if i=='_':
        cnt = 0
    else:
        cnt += 1
if op != cl:
    ans1 = 0
    ans2 = 0
if mx < cnt:
    ans1 = cnt
print(ans1, ans2)
