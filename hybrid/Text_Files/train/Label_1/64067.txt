n = int(input())
a = [int(x) for x in input().split()]
or1 = [0] * n
or1[0] = a[0]
ans = 0
for i in range(1, n):
    or1[i] = or1[i - 1] | a[i]
or2 = [0] * n
or2[n - 1] = a[n - 1]
for i in range(n - 2, -1, -1):
    or2[i] = or2[i + 1] | a[i]
for i in range(1, n):
    if max(a[i - 1], a[i]) < or1[i - 1] | or2[i]:
        ans += 1
print(ans)
