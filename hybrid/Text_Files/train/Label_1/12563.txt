#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  sort(v.begin(), v.end());
  int sum = 0;
  for (int i = 0; i < n / 2; i++) { sum += v[i]; }
  if (sum * 2 > accumulate(v.begin(), v.end(), 0)) {
    cout << "Bob" << endl;
  } else {
    cout << "Alice" << endl;
  }
  return 0;
}
