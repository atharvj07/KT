#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define rep(i, n) for (ll i = 0; i < n; ++i)

void solve() {
  int n;
  cin >> n;
  vector<pair<ll, ll>> v(n);
  rep(i, n) {
    ll l, r;
    cin >> l >> r;
    v[i] = {l, r};
  }
  sort(v.begin(), v.end());
  ll cur = 0;
  rep(i, n) {
    if (v[i].first >= cur) {
      cout << cur + 1 << ' ';
      cur++;
    } else if (v[i].second >= cur) {
      cout << 0 << ' ';
    }
  }
  cout << endl;
}

int main() {
  int t;
  cin >> t;
  while (t--) { solve(); }
  return 0;
}
