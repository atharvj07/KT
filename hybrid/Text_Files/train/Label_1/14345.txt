#include <iostream>
#include <cmath>
using namespace std;
int main() {
    long long a, b, x, y, vx, vy;
    cin >> a >> b >> x >> y >> vx >> vy;
    if (!vx) {
        if (vy > 0) cout << a << " 0\n";
        if (vy < 0) cout << "0 " << b << "\n";
        return 0;
    }
    if (!vy) {
        if (vx > 0) cout << "0 " << y << "\n";
        if (vx < 0) cout << a << " " << y << "\n";
        return 0;
    }
    long double A = a / (double)vx, B = b / (double)vy;
    long double px = (y - (A * vx)) / (double)(vy - vx * A);
    long double py = (x - (B * vy)) / (double)(vx - vy * B);
    long double Px = px, Py = py;
    px += px - x;
    py += py - y;
    if (0 <= px && px <= a && 0 <= py && py <= b) cout << px << " " << py << "\n";
    else if (0 <= Px && Px <= a && 0 <= Py && Py <= b)
        cout << Px << " " << Py << "\n";
    else
        cout << "-1\n";
    return 0;
}
