#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    string s;
    getline(cin, s);

    string t = "heidi";
    int n = t.size(), m = s.size();

    vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            dp[i][j] = max(dp[i][j - 1], dp[i - 1][j]);
            if (t[i - 1] == s[j - 1]) {
                dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] + 1);
            }
        }
    }

    if (dp[n][m] == n) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }

    return 0;
}
