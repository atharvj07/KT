#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, d;
  cin >> n >> d;
  vector<int> s(n), p(n);
  for(int& i: s) cin >> i;
  for(int& i: p) cin >> i;
  s[d - 1] += p[d - 1];
  sort(s.rbegin(), s.rend());
  int ans = unique(s.begin(), s.end()) - s.begin();
  cout << ans << endl;
  return 0;
}
