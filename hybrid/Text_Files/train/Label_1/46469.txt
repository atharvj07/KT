import java.io.*;
import java.util.*;

class Main {

    private static class Segment implements Comparable<Segment> {
        public int sum;
        public int count;

        public Segment(int sum) {
            this.sum = sum;
            this.count = 1;
        }

        public void add(Segment other) {
            this.sum += other.sum;
            this.count += other.count;
        }

        @Override
        public int compareTo(Segment other) {
            return Integer.compare(this.sum, other.sum);
        }
    }

    private static int n;
    private static long a[];
    private static long minWeakness;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(in.readLine());
        a = new long[n];
        StringTokenizer tokenizer = new StringTokenizer(in.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Long.parseLong(tokenizer.nextToken());
        }

        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }

        minWeakness = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[j] < a[i]) {
                    updateWeakness(a[j], a[i], sum);
                }
            }
        }

        System.out.println(minWeakness / 2.0);
    }

    private static void updateWeakness(long a, long b, long sum) {
        minWeakness = Math.min(minWeakness, Math.max(Math.abs(sum - 2 * a), Math.abs(2 * b - sum)));
    }
}
