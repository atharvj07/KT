#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int ans = n;
  for (int i = 0; i < n - k; i++) {
    int j = i + k;
    while (j < n && s[j] == '1') j++;
    if (j == n) continue;
    ans = min(ans, max(i, n - 1 - j));
  }
  cout << ans << endl;
  return 0;
}
