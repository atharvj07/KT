#include <bits/stdc++.h>
using namespace std;

vector<int> solve(vector<int> &a) {
  int n = a.size();
  if (n % 2 == 1 && a[(n - 1) / 2]) return {};
  vector<int> res;
  for (int i = 0; i < n - 2; ++i) {
    if (a[i] == 0) continue;
    res.push_back(i + 1);
    a[i] ^= 1;
    a[i + 1] ^= 1;
    a[i + 2] ^= 1;
  }
  if (a[n - 1] && a[n - 2]) return {};
  if (a[n - 1] && a[n - 3]) { res.push_back(n - 2); }
  if (a[n - 1]) { res.push_back(n - 1); }
  return res;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    vector<int> res = solve(a);
    if (res.empty()) {
      cout << "NO" << endl;
    } else {
      cout << "YES" << endl;
      cout << res.size() << endl;
      for (int x : res) { cout << x << " "; }
      cout << endl;
    }
  }
  return 0;
}
