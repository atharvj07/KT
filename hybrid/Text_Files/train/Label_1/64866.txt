import sys
input = sys.stdin.readline

n = int(input())
A = [list(map(int, input().split())) for _ in range(n)]
G = [[] for _ in range(n)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    a -= 1; b -= 1
    G[a].append(b); G[b].append(a)

vis = [0] * n
ans = []
def dfs(u, p):
    vis[u] = 1
    sum_a = A[u][0]
    need = [0] * 2
    need[A[u][1]] += 1
    need[A[u][2]] -= 1
    for v in G[u]:
        if v == p: continue
        if vis[v] == 0:
            dfs(v, u)
            sum_a += ans[v]
            for i in range(2):
                need[i] += ans[v][i]
    if sum_a < 0 or need[0] != 0 or need[1] != 0:
        return [-1]
    ans.append(need)
    return need

tmp = dfs(0, 0)
if tmp == [-1]: print(-1)
else:
    print(sum(map(abs, tmp[0])))
