import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String level = sc.next();
        boolean good = false;
        for (int i = 3; i < n; i++) {
            if (level.charAt(i) == '*' && level.charAt(i - 1) == '*' && level.charAt(i - 2) == '*' && level.charAt(i - 3) == '*') {
                good = true;
                break;
            }
        }
        System.out.println(good ? "yes" : "no");
    }
}
