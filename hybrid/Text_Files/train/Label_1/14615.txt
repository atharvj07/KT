#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ll n, x, y;
    cin >> n >> x >> y;
    vector<ll> a(n);
    for (auto& ai : a) cin >> ai;
    for (auto ai : a) {
        ll t1 = (ai + x - 1) / x, t2 = (ai + y - 1) / y;
        if (t1 == t2)
            cout << "Both" << endl;
        else if (t1 < t2)
            cout << "Vanya" << endl;
        else
            cout << "Vova" << endl;
    }
}
