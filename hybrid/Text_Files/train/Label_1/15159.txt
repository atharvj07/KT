#include <bits/stdc++.h>
#define pb push_back
using namespace std;

typedef long long ll;
const int N = 3000, INF = 1e9;
ll a[N], b[N];
int main() {
    ll n, p, s;
    cin >> n >> p >> s;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    vector<pair<ll, int>> skill;
    for (int i = 0; i < n; i++) skill.pb({a[i] + b[i], i});
    sort(skill.rbegin(), skill.rend());
    vector<int> team1, team2;
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        if (team1.size() == p && team2.size() == s) break;
        int idx = skill[i].second;
        if (team1.size() < p) {
            ans += a[idx];
            team1.pb(idx + 1);
        } else if (team2.size() < s) {
            ans += b[idx];
            team2.pb(idx + 1);
        }
    }
    cout << ans << endl;
    for (int i = 0; i < team1.size(); i++) cout << team1[i] << " ";
    cout << endl;
    for (int i = 0; i < team2.size(); i++) cout << team2[i] << " ";
    cout << endl;
    return 0;
}
