#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[100000];
int main() {
    ll n;
    cin >> n;
    for (ll i = 0; i < n; i++) { cin >> a[i]; }
    ll ans = 0;
    ll l = 0;
    for (ll r = 0; r < n; r++) {
        if (a[r] >= a[r - 1]) continue;
        if (r - l + 1 <= n - r) {
            l = r;
            ans += n - r;
        } else
            break;
    }
    if (l < n - 1) ans = -1;
    cout << ans << endl;
    return 0;
}
