import java.util.Scanner;

public class Main {

    static int[][] grid;
    static int[] colorCount;
    static boolean[][] visited;
    static int[][] dir = {{0, 1}, {1, 0}, {-1, 0}, {0, -1}};
    static int curColor = 0;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int n = sc.nextInt();
        colorCount = new int[n + 1];
        for (int i = 0; i < n; i++) {
            colorCount[i] = sc.nextInt();
        }
        grid = new int[h][w];
        visited = new boolean[h][w];
        dfs(0, 0);
        for (int[] row : grid) {
            for (int c : row) {
                System.out.print(c + " ");
            }
            System.out.println();
        }
    }

    private static void dfs(int row, int col) {
        visited[row][col] = true;
        grid[row][col] = ++curColor;
        for (int[] d : dir) {
            int nr = row + d[0];
            int nc = col + d[1];
            if (nr >= 0 && nr < grid.length && nc >= 0 && nc < grid[0].length && !visited[nr][nc] &&
                colorCount[curColor] > 0) {
                colorCount[curColor]--;
                dfs(nr, nc);
            }
        }
    }
}
