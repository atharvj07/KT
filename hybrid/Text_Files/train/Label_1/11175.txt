#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxN = 2e5 + 5;
pair<ll, ll> seg[maxN];
int main() {
    int n, m;
    cin >> n >> m;
    vector<int> brd(maxN);
    for (int i = 1; i <= n; ++i) { cin >> seg[i].first >> seg[i].second; }
    vector<ll> a(m);
    for (int i = 0; i < m; ++i) { cin >> a[i]; }
    sort(a.begin(), a.end());
    sort(seg + 1, seg + n + 1);
    for (int i = 2; i <= n; ++i) {
        if (seg[i].first - seg[i - 1].second > 1) {
            cout << "No"
                 << "\n";
            return 0;
        }
    }
    cout << "Yes"
         << "\n";
    for (int i = 2; i <= n; ++i) {
        int j = upper_bound(a.begin(), a.end(), seg[i].first - seg[i - 1].second) - a.begin() - 1;
        if (j == -1 || a[j] != seg[i].first - seg[i - 1].second) {
            cout << "No"
                 << "\n";
            return 0;
        } else {
            cout << j + 1 << " ";
            brd[j + 1] = 1;
        }
    }
    for (int i = 0; i < m; ++i) {
        if (!brd[i + 1]) {
            cout << "No"
                 << "\n";
            return 0;
        }
    }
    return 0;
}
