import sys

n = int(input().split()[0])
a = list(map(int, input().split()))
m = int(input().split()[0])
b = list(map(int, input().split()))

pos = -1
ans = 0
for x in b:
    if pos == -1:
        pos = x
    elif pos == x - 1:
        ans += a[x - 1] - a[pos - 1]
        pos = x
    else:
        ans = -1
        break
if pos != -1 and pos == m:
    ans += a[pos - 1] - a[0]
print(ans)
