#include <iostream>

using namespace std;

int main() {
  long long x;
  cin >> x;
  long long ans = 0;
  for (long long i = x; i > 0; i--) {
    long long sum = 0;
    long long temp = i;
    while (temp > 0) {
      sum += temp % 10;
      temp = temp / 10;
    }
    if (sum > ans) {
      ans = sum;
      x = i;
    }
  }
  cout << x << endl;
  return 0;
}
