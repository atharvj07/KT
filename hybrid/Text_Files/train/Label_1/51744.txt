import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] t = new int[n];
        int cur = 0;
        int page = 1;
        for (int i = 0; i < n; i++) {
            cur += a[i];
            if (cur > m) {
                cur = a[i];
                page++;
            }
            t[i] = page - 1;
        }
        for (int i = 0; i < n; i++) {
            System.out.println(t[i]);
        }
    }
}
