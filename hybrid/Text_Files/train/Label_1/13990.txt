#include<bits/stdc++.h>
using namespace std;
struct line {
	int a, b, c;
};
int n, x, y, p[301], t[301], s[301], ans;
line v[301];
inline int f(int a, int b) { return a < b ? a : b; }
inline int gcd(int a, int b) { return !b ? a : gcd(b, a % b); }
bool cmp(int a, int b) { return s[a] < s[b]; }
inline int id(int x, int y) {
	int res = n + 1;
	for (int i = 1; i <= n; i++) if (!v[i].a * y - v[i].b * x - v[i].c) res = i;
	return res;
}
int main() {
	cin >> x >> y >> p[1] >> t[1] >> n;
	for (int i = 1; i <= n; i++) cin >> v[i].a >> v[i].b >> v[i].c;
	for (int i = 2; i <= n + 1; i++) {
		p[i] = id(p[i - 1], t[i - 1]);
		s[i] = 0;
		if (p[i] == n + 1) continue;
		int g = gcd(abs(v[p[i]].a), abs(v[p[i]].b));
		if (v[p[i]].c < 0) s[i] = ((-v[p[i]].c % g) + g) % g;
		else s[i] = v[p[i]].c % g;
	}
	for (int i = 2; i <= n + 1; i++) if (p[i] != n + 1) {
		if (p[i] == p[i - 1] && s[i] != s[i - 1]) {
			cout << "-1";
			return 0;
		}
		if (p[i] != p[i - 1]) ans++;
	}
	cout << ans;
}
