#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    vector<int>a[101];
    for(int i=1;i<=n;++i){
        int l,r;
        cin>>l>>r;
        for(int j=l;j<r;++j){
            a[j].push_back(i);
        }
    }
    int ans=0;
    for(int i=0;i<100;++i){
        if(a[i].size()==1&&a[i][0]==1){
            ++ans;
        }
    }
    cout<<ans<<endl;
}
