n = int(input())
a = list(map(int, input().split()))
q = int(input())

a.sort()

for i in range(q):
    op, x = input().split()
    if op == '+':
        a.append(int(x))
        a.sort()
    else:
        if a.count(int(x)) == 0:
            print("NO")
        else:
            a.remove(int(x))
    cnt = [0 for i in range(100001)]
    for i in a:
        cnt[i] += 1
    if cnt[0] > 0 or cnt[1] > 0 or cnt[2] > 0 or len(a) == 0:
        print("NO")
    else:
        if cnt[a[-1]] >= 4 or cnt[a[-2]] >= 2 or (cnt[a[-1]] == 2 and cnt[a[-2]] == 1):
            print("YES")
        else:
            print("NO")
