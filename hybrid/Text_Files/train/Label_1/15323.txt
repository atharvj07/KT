#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int cnt = 0;
  bool hasCapB = false;
  for (int i = 0; i < s.length(); i++) {
    if (s[i] == 'B' && !hasCapB) {
      cnt += hasCapB = true;
    } else if (hasCapB && s[i] == 'u')
      cnt += true;
    else if (hasCapB && cnt == 2 && s[i] == 'l')
      cnt += true;
    else if (hasCapB && cnt == 3 && s[i] == 'b')
      cnt += true;
    else if (hasCapB && cnt == 4 && s[i] == 'a')
      cnt += true;
    else if (hasCapB && cnt == 5 && s[i] == 's')
      cnt += true;
    else if (hasCapB && cnt == 6 && s[i] == 'a')
      cnt += true;
    else if (hasCapB && cnt == 7 && s[i] == 'u')
      cnt += true;
    else if (hasCapB && cnt == 8 && s[i] == 'r')
      cnt += true;
    if (cnt == 9)
      hasCapB = false, cnt = 0;
  }
  cout << (cnt == 9 ? cnt / 9 : cnt);
  return 0;
}
