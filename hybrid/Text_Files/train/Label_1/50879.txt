import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int p = sc.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        sc.close();

        int[] ans = new int[n];
        int curTime = 0;
        for (int i = 0; i < n; i++) {
            if (t[i] >= curTime) {
                ans[i] = t[i] + p;
            } else {
                ans[i] = curTime + p;
            }
            curTime = ans[i];
        }
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
