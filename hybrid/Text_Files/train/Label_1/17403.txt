#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++)
    cin >> v[i];
  int p = 0, cnt = 0;
  for (int i = 0; i < n; i++) {
    if (v[i] == -1) {
      if (p)
        p--;
      else
        cnt++;
    } else
      p += v[i];
  }
  cout << cnt << endl;
}
