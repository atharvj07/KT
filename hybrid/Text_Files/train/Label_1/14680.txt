#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
const int MAXN=503, MAXM=503;
int n, m, a[MAXN][MAXM];
int dx[]={0,1,0,-1}, dy[]={1,0,-1,0};
int b[MAXN][MAXM], ans, sum, empty, id[MAXN][MAXM];
vector<int> v;
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            cin>>a[i][j];
            if(a[i][j]==0)empty++;
        }
    }
    if(empty>2*n*m-n-m-4){
        cout<<-1<<endl;
        return 0;
    }
    int tot=0;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            if(a[i][j]==0){
                tot++;
                id[i][j]=tot;
                v.push_back(0);
            }
            else{
                if(a[i-1][j]>=a[i][j]||a[i][j-1]>=a[i][j]){
                    cout<<-1<<endl;
                    return 0;
                }
            }
        }
    }
    sort(v.begin(),v.end());
    if(v.size()>n+m-2){
        cout<<-1<<endl;
        return 0;
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            if(!id[i][j])continue;
            int cnt=0, x=i, y=j;
            for(int k=0;k<4;k++){
                int nx=x+dx[k], ny=y+dy[k];
                if(nx<1||nx>n||ny<1||ny>m||!a[nx][ny])continue;
                cnt++;
                if(a[nx][ny]<v.back())ans+=v.back()-a[nx][ny];
            }
            if(cnt<2){
                cout<<-1<<endl;
                return 0;
            }
            b[i][j]=v[cnt-2];
        }
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            sum+=b[i][j];
        }
    }
    sum+=ans;
    cout<<sum<<endl;
}
