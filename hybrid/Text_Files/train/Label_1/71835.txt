import sys
from collections import deque
input = sys.stdin.readline

n = int(input())
ad = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    ad[a].append(b)
    ad[b].append(a)
v = list(map(int, input().split()))

q = deque()
q.append((1, 0))
vis = [False] * (n+1)
vis[1] = True
flag = False
ans = 0
while q:
    now, p = q.popleft()
    ans += abs(v[now-1])
    v[now-1] = 0
    if not flag and now == n and v[now-1] == 0:
        flag = True
        ans -= 1
    for nxt in ad[now]:
        if vis[nxt]: continue
        vis[nxt] = True
        q.append((nxt, now))

print(ans)
