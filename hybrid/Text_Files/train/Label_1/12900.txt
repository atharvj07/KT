#include<bits/stdc++.h>
using namespace std;
const int MAXN=1e5+5;
vector<int> adj[MAXN];
bool vis[MAXN]={false};
int par[MAXN];
struct node
{
  int x,dist;
};
bool cmp(const node &a,const node &b)
{
  return a.dist>b.dist;
}
void dfs(int u,int p,int dist)
{
  vis[u]=true;
  par[u]=p;
  for(int i=0; i<(int)adj[u].size(); i++)
  {
    if(adj[u][i]!=p)dfs(adj[u][i],u,dist+1);
  }
  return;
}
int main()
{
  int n,d,h;
  cin>>n>>d>>h;
  if(d==1)cout<<"-1"<<endl;
  else if(d==2)
  {
    if(h==1)cout<<"-1"<<endl;
    else
    {
      cout<<1<<" "<<2<<endl;
      cout<<2<<" "<<3<<endl;
      for(int i=4; i<=n; i++)cout<<2<<" "<<i<<endl;
    }
  }
  else if(d>=3)
  {
    if(h==1)
    {
      cout<<1<<" "<<2<<endl;
      for(int i=3; i<=n; i++)cout<<"2"<<" "<<i<<endl;
    }
    else if(h>=d-1)
    {
      int cnt=0;
      for(int i=1; i<=n; i++)
      {
        if(vis[i])continue;
        dfs(i,-1,0);
        cnt++;
      }
      if(cnt!=3)cout<<"-1"<<endl;
      else
      {
        int root1=-1,root2=-1,root3=-1;
        for(int i=1; i<=n; i++)
        {
          if(vis[i]==false){root1=i; break;}
        }
        for(int i=root1+1; i<=n; i++)
        {
          if(par[i]==root1){root2=i; break;}
        }
        for(int i=root2+1; i<=n; i++)
        {
          if(vis[i]==false){root3=i; break;}
        }
        dfs(root1,-1,0);
        int node1=-1,node2=-1,node3=-1,node4=-1,node5=-1;
        for(int i=1; i<=n; i++)
        {
          if(i==root1)continue;
          else if(vis[i] and par[i]==root1)node1=i;
          else if(vis[i] and par[i]==node1 and node1!=-1)node2=i;
          else if(vis[i] and par[i]==node2 and node2!=-1)node3=i;
          else if(vis[i] and par[i]==node3 and node3!=-1)node4=i;
          else if(vis[i] and par[i]==node4 and node4!=-1)node5=i;
        }
        vis[root2]=false;
        vis[root3]=false;
        if(node3==-1)
        {
          cout<<root1<<" "<<node2<<endl;
          cout<<root1<<" "<<root2<<endl;
          for(int i=3; i<=n; i++)
          {
            if(i==node2)continue;
            else cout<<node2<<" "<<i<<endl;
          }
        }
        else
        {
          cout<<root1<<" "<<node3<<endl;
          cout<<root1<<" "<<root2<<endl;
          cout<<root1<<" "<<node2<<endl;
          for(int i=4; i<=n; i++)
          {
            if(i==node2 or i==node3)continue;
            else cout<<node3<<" "<<i<<endl;
          }
        }
      }
    }
    else cout<<"-1"<<endl;
  }
}
