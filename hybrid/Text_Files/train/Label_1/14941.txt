#include <bits/stdc++.h>

using namespace std;

const int MAXN = 100;
const int MAXQ = 1e5;
const int MAX_TIME = 1e6;

struct Task {
    int t, k, d;
    Task(int _t, int _k, int _d) : t(_t), k(_k), d(_d) {}
};

int n, q;
vector<Task> tasks;
bool occupied[MAXN][MAX_TIME];

int main() {
    cin >> n >> q;
    for (int i = 0; i < q; i++) {
        int t, k, d;
        cin >> t >> k >> d;
        tasks.emplace_back(t, k, d);
    }

    for (Task &task : tasks) {
        bool found = false;
        for (int i = 0; i < task.k && i < n; i++) {
            bool can_perform = true;
            for (int j = task.t; j < task.t + task.d; j++) {
                if (occupied[i][j]) {
                    can_perform = false;
                    break;
                }
            }
            if (can_perform) {
                int sum = 0;
                for (int j = 0; j < task.k; j++) {
                    sum += j + 1;
                    for (int l = task.t; l < task.t + task.d; l++) {
                        occupied[j][l] = true;
                    }
                }
                cout << sum << endl;
                found = true;
                break;
            }
        }
        if (!found) {
            cout << -1 << endl;
        }
    }

    return 0;
}
