#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9 + 7;
int dp[2005][2005];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<pair<int, int>> v(m);
  for (int i = 0; i < m; i++) {
    int x, y;
    cin >> x >> y;
    v[i] = {x, y};
  }
  sort(v.begin(), v.end());
  sort(a.begin(), a.end());
  for (int i = 0; i <= k; i++)
    for (int j = 0; j <= m; j++) dp[i][j] = inf;
  dp[0][0] = 0;
  for (int i = 1; i <= k; i++) {
    for (int j = 0; j < m; j++) {
      if (i >= v[j].first) dp[i][j] = min(dp[i][j], dp[i - v[j].first][j] + a[i - 1]);
      if (i >= v[j].second)
        dp[i][j] = min(dp[i][j], dp[i - v[j].second][j] + a[i - v[j].second - 1]);
    }
  }
  int ans = inf;
  for (int i = 0; i < m; i++) ans = min(ans, dp[k][i]);
  cout << ans << "\n";
  return 0;
}
