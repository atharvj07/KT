import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int p = sc.nextInt();
        int s = sc.nextInt();
        int r = sc.nextInt();
        System.out.println(BigInteger.valueOf(s - r + 1).modPow(BigInteger.valueOf(p - 1), BigInteger.valueOf(998244353)).intValue());
    }
}
