#include <bits/stdc++.h>
#define ll long long
using namespace std;

int n, a[300001];
ll suf[300001];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = n - 1; i; i--)
        suf[i] = suf[i + 1] + a[i] != a[i + 1];
    int ans = 0;
    for (int i = 2; i < n; i++)
        ans += suf[i + 1] == suf[i];
    cout << ans + 1 << endl;
}
