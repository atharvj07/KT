#include <bits/stdc++.h>
using namespace std;

bool vis[505][505];
int n, m, k, nxt[505][505][4];
char path[505][505];
vector<pair<pair<int, int> , int >> ans;

bool valid(int x, int y){
    if(x < 1 || x > n || y < 1 || y > m || vis[x][y])
        return false;
    return true;
}
void dfs(int x, int y, int d){
    int dx[] = {0, 0, -1, 1};
    int dy[] = {1, -1, 0, 0};

    if(vis[x][y] || (d == k+1)){
        return;
    }

    vis[x][y] = true;
    if(d == k){
        ans.push_back({{x, y}, 0});
        ans[(int)ans.size()-1].second = ans[(int)ans.size()-2].second + 1;
        return;
    }

    for(int i = 0; i < 4; i++){
        int newX = dx[i] + x;
        int newY = dy[i] + y;

        if(valid(newX, newY) && (nxt[x][y][i] == -1 || vis[nxt[x][y][i]][(nxt[x][y][i]+1)%m])){
            ans.push_back({{x, y}, i});
            path[x][y] = i+'L';
            nxt[x][y][i] = newX;
            dfs(newX, newY, d+1);
            path[x][y] = 0;
            vis[nxt[x][y][i]][(nxt[x][y][i]+1)%m] = false;
            ans.pop_back();
        }
    }
}

int main(){
    cin >> n >> m >> k;
    memset(nxt,-1,sizeof nxt);
    memset(vis,false,sizeof vis);
    memset(path, 0, sizeof path);
    ans.clear();

    dfs(1, 1, 1);

    if(ans.empty()){
        cout << "NO\n";
        return 0;
    }
    cout << "YES\n";
    cout << ans.size() << '\n';
    for(int i = 0; i < (int)ans.size(); i++){
        if(ans[i].first.first == 1 && ans[i].first.second == 1){
            cout << ans[i].second << ' ' << path[ans[i].first.first][ans[i].first.second] << '\n';
        }
        else{
            cout << 1 << ' ' << path[ans[i].first.first][ans[i].first.second] << '\n';
        }
    }

    return 0;
}
