#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<pair<int, char>> t(n), s(m);
    for (auto &i : t) cin >> i.first >> i.second;
    for (auto &i : s) cin >> i.first >> i.second;
    int64_t ans = 0;
    int j = 0;
    for (int i = 0; i < n; i++) {
        while (j < m && t[i].second != s[j].second) j++;
        if (j == m) break;
        if (t[i].first >= s[j].first) {
            ans++;
            j++;
            t[i].first -= s[j].first;
        } else
            j = 0;
    }
    cout << ans << endl;
    return 0;
}
