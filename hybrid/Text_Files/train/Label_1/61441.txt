n, m = map(int, input().split())
p = list(map(int, input().split()))

# Convert the permutation into a list indicating whether each element is greater than or equal to m
# or less than m, using -1 for elements less than m and 1 for elements greater than or equal to m.
transformed = [1 if pi >= m else -1 for pi in p]

# Calculate prefix sums of the transformed list, which allows for efficient range sum queries later.
prefix_sums = [0]
for t in transformed:
    prefix_sums.append(prefix_sums[-1] + t)

# Use a dictionary to count occurrences of each prefix sum. This is important for calculating how many
# times a range with a specific sum (which relates to having the median at the right position) occurs.
from collections import defaultdict
prefix_count = defaultdict(int)

result = 0
for sum_val in prefix_sums:
    # If the current prefix sum is encountered again, it means there's a range that could potentially
    # have the median at our target. We add to result how many times we've seen (sum_val - 1) before,
    # as it indicates ranges where elements less than m are one less than those greater than or equal to m,
    # which is exactly the condition for our target to be the median.
    result += prefix_count[sum_val - 1]
    if sum_val == 0:
        # Additionally, if sum_val is 0, it means the range has equal numbers of elements
        # greater than or equal and less than m, so our target could be the median.
        result += 1
    prefix_count[sum_val] += 1

print(result)
