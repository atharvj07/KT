#include <iostream>
#include <vector>

using namespace std;
string sa, sb;
int n, m;
vector<vector<int>> da, db;

int main() {
  cin >> sa >> sb;
  n = sa.size(), m = n / 2;
  da.resize(n + 2, vector<int>(m + 2, -1));
  db.resize(n + 2, vector<int>(m + 2, -1));
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= m; j++) {
      for (int k = 0; k <= m; k++) {
        int ps = i - j - k;
        if (ps == 1) {
          if (sa[i - 1] == sb[i - 1]) {
            da[i][j] = 1;
            db[i][k] = 1;
          }
        } else if (ps > 1) {
          if (da[i - 1][j] >= 1 && db[i - 1][k] >= 1) {
            da[i][j] = 1;
            db[i][k] = 1;
          }
          if (da[i - 1][k] >= 1 && db[i - 1][j] >= 1) {
            da[i][j] = 1;
            db[i][k] = 1;
          }
        }
      }
    }
  }
  if (da[n][m] >= 1)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
