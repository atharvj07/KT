#include <iostream>

typedef long long ll;

using namespace std;

int main() {
  ll n, k;
  cin >> n >> k;
  ll m = n / 2;
  ll d = n / (k + 1);
  if (d + k * d <= m) {
    cout << d << " " << d * k << " " << n - d - d * k << endl;
  } else {
    cout << 0 << " " << 0 << " " << n << endl;
  }
  return 0;
}
