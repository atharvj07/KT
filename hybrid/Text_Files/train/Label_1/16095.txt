#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 3e5 + 5;
bool mark[N];

int main() {
    int n; cin >> n;
    string s; cin >> s;

    int ans = 0;
    ll pows = 0;
    for (int i = n - 1; i >= 0; i--) {
        pows = (pows * 2 + (s[i] == 'A' ? 0 : 1)) % 3;
        if (s[i] == 'B') {
            mark[pows] = 1;
            ans += mark[pows ^ 3];
        } else {
            mark[pows] = 1;
            ans += mark[(pows + 1) % 3] + mark[(pows + 2) % 3];
        }
    }
    cout << ans << '\n';
    return 0;
}
