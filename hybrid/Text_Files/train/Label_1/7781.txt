#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<ll> d(2 * n);
        for (int i = 0; i < 2 * n; i++) cin >> d[i];
        sort(d.begin(), d.end());
        bool f = true;
        for (int i = 1; i < 2 * n; i += 2)
            if (d[i] != d[i - 1]) f = false;
        for (int i = 0; i < 2 * n - 2; i += 2)
            if (d[i] + d[i + 1] != d[i + 2]) f = false;
        if (f)
            cout << "YES\n";
        else
            cout << "NO\n";
    }
}
