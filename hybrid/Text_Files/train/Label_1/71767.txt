n = int(input())
s = list(input())
a = s.count("A")
c = s.count("C")
g = s.count("G")
t = s.count("T")
q = s.count("?")
dif = abs(a - c) + abs(a - g) + abs(a - t)
if q < dif or dif % 4 != 0:
    print("===")
else:
    for i in range(len(s)):
        if s[i] == "?":
            if a == min(a, c, g, t):
                s[i] = "A"
                a += 1
            elif c == min(a, c, g, t):
                s[i] = "C"
                c += 1
            elif g == min(a, c, g, t):
                s[i] = "G"
                g += 1
            else:
                s[i] = "T"
                t += 1
    print("".join(s))
