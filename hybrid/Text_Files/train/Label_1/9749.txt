#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long n;
  cin >> n;
  vector<int> ans;
  for (int x = 1; x <= n; x++) {
    int sum = 0;
    int t = x;
    while (t) { sum += t % 10, t /= 10; }
    if (x + sum == n) ans.push_back(x);
  }
  cout << ans.size() << endl;
  for (int x : ans) cout << x << endl;
}
