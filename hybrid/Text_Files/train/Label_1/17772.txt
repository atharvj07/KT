#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    vector<int> b;
    for (int i = 0; i < n; i += 2) {
        b.push_back(a[i] + a[i + 1]);
    }
    long long ans = 0;
    for (int i = 0; i < b.size(); i++) {
        ans += pow(b[i], 2);
    }
    cout << ans << endl;
    return 0;
}
