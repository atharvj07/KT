#include<cstdio>
#include<algorithm>
using namespace std;
const int N=200000+5;
int n,k;
int p[N],e[N];
bool cmp(int x,int y){
    return p[x]<p[y];
}
int main(){
    scanf("%d%d",&n,&k);
    for(int i=1;i<=n;++i) {scanf("%d%d",&p[i],&e[i]);}
    int id[N];
    for(int i=1;i<=n;++i) id[i]=i;
    sort(id+1,id+n+1,cmp);
    long long res=0;
    for(int i=n;i>=n-k+1;--i) res+=e[id[i]];
    sort(id+1,id+n+1);
    int l=1,r=n,pos=0;
    for(int i=1;i<=n;++i){
        if(id[i]==1){
            if(p[1]==0) pos=i;
            break;
        }
    }
    while(l<=r){
        int mid=(l+r)>>1;
        int en=res,cnt=0;
        for(int i=pos-1;i>=mid;--i){
            if(e[id[i]]>0) ++cnt;
            en+=e[id[i]];
        }
        if(cnt<=k-(n-r+1)){
            l=mid+1;
        } else {
            res=en;
            r=mid-1;
        }
    }
    if(res==0) res=-1;
    printf("%lld\n",res);
    return 0;
}
