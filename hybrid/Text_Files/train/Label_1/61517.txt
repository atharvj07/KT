import sys
for t in range(int(input())):
    n = int(input())
    st = input()
    s1 = st[::-1]
    ans = st
    m = n//2
    for i in range(1,m+1):
        t1 = []
        t2 = []
        j = 0
        while(j+i-1<n):
            t1.append(st[j:j+i])
            t2.append(s1[j:j+i])
            j+=i
        t1.sort()
        t2.sort()
        t1.reverse()
        j = 0
        while(j+i-1<n):
            st = st[:j]+t1[j//i]+st[j+i:]
            s1 = s1[:j]+t2[j//i]+s1[j+i:]
            j+=i
        if st<ans:
            ans = st
    print(ans)
    print(len(ans)-len(set(ans)))
