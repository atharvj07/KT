n = int(input())
energy = list(map(int, input().split()))
strings = [input() for _ in range(n)]

# Initialize dp array to store minimum energy required for each string to keep the sequence sorted
# dp[i][0] is min energy if ith string is not reversed, dp[i][1] is min energy if ith string is reversed
dp = [[float('inf'), float('inf')] for _ in range(n)]
dp[0][0], dp[0][1] = 0, energy[0]  # No energy required to keep first string as is, energy[0] to reverse it

for i in range(1, n):
    rev_str = strings[i][::-1]  # Reverse current string
    # If current string or its reverse is lexicographically larger than previous, update dp accordingly
    if strings[i] >= strings[i-1]:  # Current string as is is correctly ordered
        dp[i][0] = min(dp[i][0], dp[i-1][0])
    if strings[i] >= strings[i-1][::-1]:  # Current string as is is correctly ordered after reversing previous
        dp[i][0] = min(dp[i][0], dp[i-1][1])
    if rev_str >= strings[i-1]:  # Reversed current string is correctly ordered
        dp[i][1] = min(dp[i][1], dp[i-1][0] + energy[i])
    if rev_str >= strings[i-1][::-1]:  # Reversed current string is correctly ordered after reversing previous
        dp[i][1] = min(dp[i][1], dp[i-1][1] + energy[i])

# The answer is the minimum energy required to keep the last string sorted, either reversed or not
ans = min(dp[n-1])
print(-1 if ans == float('inf') else ans)
