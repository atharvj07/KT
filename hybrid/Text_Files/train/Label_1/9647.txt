#include <vector>
#include <iostream>
#include <queue>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  priority_queue<pair<int, int>> q;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    q.push({-x, i + 1});
  }
  int ans = 0;
  while (!q.empty()) {
    auto u = q.top();
    q.pop();
    u.first += m;
    if (u.first >= 0) {
      ans = u.second;
    } else {
      q.push(u);
    }
  }
  cout << ans << endl;
}
