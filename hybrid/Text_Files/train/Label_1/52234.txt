import java.util.Arrays;
import java.util.Scanner;
import java.util.LinkedList;
import java.util.Queue;

class Main {
  final static int MAX_N = 6000 + 1;

  static int[] r = new int[MAX_N];
  static int[] p = new int[MAX_N];
  static boolean[] vis = new boolean[MAX_N];
  static LinkedList<Integer>[] roads = new LinkedList[MAX_N];

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();

    for (int i = 1; i <= n; i++) {
      r[i] = in.nextInt();
      roads[i] = new LinkedList<Integer>();
    }

    for (int i = 1; i < n; i++) {
      int u = in.nextInt();
      int v = in.nextInt();
      roads[u].add(v);
      roads[v].add(u);
    }

    Arrays.fill(p, -1);
    Arrays.fill(vis, false);
    dfs(1);

    int ans = 0;
    int cur = n;
    while (cur != -1) {
      ans++;
      vis[cur] = true;
      cur = p[cur];
    }

    System.out.println(ans);
  }

  private static void dfs(int u) {
    vis[u] = true;
    for (int v : roads[u]) {
      if (!vis[v]) {
        if (r[v] > r[u]) {
          p[v] = u;
          dfs(v);
        }
      }
    }
  }
}
