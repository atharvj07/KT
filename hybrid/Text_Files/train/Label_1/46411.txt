import java.util.*;

public class Main {

    static class Pair implements Comparable<Pair> {
        int a, b;

        public Pair(int a, int b) {
            this.a = a;
            this.b = b;
        }

        @Override
        public int compareTo(Pair o) {
            return b - o.b;
        }
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt(), m = input.nextInt();
        int s = input.nextInt() - 1, t = input.nextInt() - 1;
        int[] p = new int[n];
        for (int i = 0; i < n; i++) p[i] = input.nextInt();

        TreeMap<Integer, Integer>[] adj = new TreeMap[n];
        for (int i = 0; i < n; i++) adj[i] = new TreeMap<>();
        while (m-- > 0) {
            int a = input.nextInt() - 1, b = input.nextInt() - 1, w = input.nextInt();
            adj[a].put(b, Math.min(w, adj[a].getOrDefault(b, 1000000000)));
            adj[b].put(a, Math.min(w, adj[b].getOrDefault(a, 1000000000)));
        }

        long[] dis1 = new long[n];
        long[] dis2 = new long[n];
        Arrays.fill(dis1, 1000000000);
        Arrays.fill(dis2, 1000000000);

        PriorityQueue<Pair> q = new PriorityQueue<>();
        q.add(new Pair(s, 0));
        while (!q.isEmpty()) {
            Pair p1 = q.poll();
            int cur = p1.a;
            int dist = p1.b;
            if (dist > dis1[cur]) continue;
            dis1[cur] = dist;
            for (int next : adj[cur].keySet()) {
                int ndist = dist + adj[cur].get(next);
                if (ndist < dis1[next]) {
                    q.add(new Pair(next, ndist));
                }
            }
        }

        q.add(new Pair(t, 0));
        while (!q.isEmpty()) {
            Pair p1 = q.poll();
            int cur = p1.a;
            int dist = p1.b;
            if (dist > dis2[cur]) continue;
            dis2[cur] = dist;
            for (int next : adj[cur].keySet()) {
                int ndist = dist + adj[cur].get(next);
                if (ndist < dis2[next]) {
                    q.add(new Pair(next, ndist));
                }
            }
        }

        long sum1 = 0, sum2 = 0;
        for (int i = 0; i < n; i++) {
            if (dis1[i] <= 1000000000) sum1 += p[i];
            if (dis2[i] <= 1000000000) sum2 += p[i];
        }

        if (sum1 > sum2) System.out.println("Break a heart");
        else if (sum1 < sum2) System.out.println("Cry");
        else System.out.println("Flowers");
    }
}
