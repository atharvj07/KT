#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    if (n == 1 || m == 1) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    vector<vector<int>> ans(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            ans[i][j] = (i + j) % (n * m) + 1;
        }
    }
    for (auto &row : ans) {
        for (auto &x : row) {
            cout << x << " ";
        }
        cout << endl;
    }
    return 0;
}
