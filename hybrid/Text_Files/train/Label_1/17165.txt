#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> s(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> s[i];
    }
    int ans = 0;
    bool last = false;
    for (int i = 0; i < n; ++i) {
        if (s[i] == 1) {
            ans++;
            last = true;
        } else if (!last) {
            ans++;
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
