import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {
    static int n, m, x;
    static char[][] keyboard;
    static int[][] dist;
    static boolean[][] visited;
    static int[] dx = {0, 0, -1, 1};
    static int[] dy = {-1, 1, 0, 0};

    static class Point {
        int x, y;

        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        x = Integer.parseInt(st.nextToken());
        keyboard = new char[n][m];
        for (int i = 0; i < n; i++) {
            keyboard[i] = br.readLine().toCharArray();
        }
        int q = Integer.parseInt(br.readLine());
        String text = br.readLine();
        dist = new int[n][m];
        visited = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dist[i], -1);
        }
        int res = bfs(text);
        System.out.println(res == -1 ? "-1" : res);
    }

    static int bfs(String text) {
        Queue<Point> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (keyboard[i][j] == 'S') {
                    queue.add(new Point(i, j));
                    dist[i][j] = 0;
                    visited[i][j] = true;
                }
            }
        }
        while (!queue.isEmpty()) {
            Point cur = queue.poll();
            for (int i = 0; i < 4; i++) {
                int nx = cur.x + dx[i];
                int ny = cur.y + dy[i];
                if (nx < 0 || nx >= n || ny < 0 || ny >= m || visited[nx][ny]) {
                    continue;
                }
                if (Math.abs(nx - cur.x) + Math.abs(ny - cur.y) <= x) {
                    dist[nx][ny] = dist[cur.x][cur.y];
                    visited[nx][ny] = true;
                    queue.add(new Point(nx, ny));
                }
            }
        }
        int ans = 0;
        for (char c : text.toCharArray()) {
            char lc = Character.toLowerCase(c);
            boolean found = false;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (keyboard[i][j] == lc || (keyboard[i][j] == 'S' && lc == Character.toLowerCase(text.charAt(ans)))) {
                        if (dist[i][j] != -1) {
                            ans += dist[i][j];
                            found = true;
                            break;
                        }
                    }
                }
                if (found) {
                    break;
                }
            }
            if (!found) {
                return -1;
            }
        }
        return ans;
    }
}
