#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<pair<int, pair<int, int>>> books(n);
  for (int i = 0; i < n; i++) {
    int t, a, b;
    cin >> t >> a >> b;
    books[i] = {t, {a, b}};
  }
  sort(books.begin(), books.end());
  int ans = -1;
  int alice = 0, bob = 0, time = 0;
  for (int i = 0; i < n; i++) {
    time += books[i].first;
    alice += books[i].second.first;
    bob += books[i].second.second;
    if (alice >= k && bob >= k) {
      ans = time;
      break;
    }
  }
  cout << ans << '\n';
  return 0;
}
