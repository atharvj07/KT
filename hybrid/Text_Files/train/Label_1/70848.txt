m, k = map(int, input().split())
fr = {}
for _ in range(m):
    a, b = map(int, input().split())
    if a not in fr: fr[a] = set()
    if b not in fr: fr[b] = set()
    fr[a].add(b)
    fr[b].add(a)
for a in sorted(fr):
    lst = []
    for b in fr[a]:
        if len(fr[b] & fr[a]) / len(fr[a]) * 100 >= k:
            lst.append(b)
    print(f'{a}: {len(lst)} {" ".join(map(str, sorted(lst)))}')
