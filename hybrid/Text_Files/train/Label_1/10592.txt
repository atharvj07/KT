#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;

int main() {
  ll n;
  cin >> n;
  priority_queue<ll> pq;
  for (ll i = 0, x; i < n; ++i) {
    cin >> x;
    pq.push(x);
  }
  ll x, f, fee = 0;
  cin >> x >> f;
  while (pq.size() > 0) {
    ll curr = pq.top();
    pq.pop();
    if (curr > x) {
      fee += f;
      pq.push(curr - x);
    }
  }
  cout << fee << "\n";
  return 0;
}
