import java.util.*;

public class Main {
    static final int MOD = (int)1e9 + 7;
    static final int NMAX = (int)2e5 + 5;

    static long[] songs;
    static long[] sum;
    static long[] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        songs = new long[n + 1];
        for (int i = 1; i <= n; ++i)
            songs[i] = sc.nextLong();
        
        int s = sc.nextInt();
        
        sum = new long[n + 1];
        dp = new long[n + 1];

        for (int i = 1; i <= n - m + 1; ++i) {
            dp[i] = dp[i - 1] + songs[i + m - 1] - songs[i - 1];
        }
        
        for (int i = 1; i <= n - m + 1; ++i) {
            sum[i] = sum[i - 1] - (i - 1) * songs[i - 1] + (n - m + i) * songs[i - 1] - songs[i + m - 1];
        }
        
        while (s-- > 0) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            int x = sc.nextInt();

            long ans = 0;
            if (l > 1) ans = sum[l - 1] - dp[l - 1];
            if (r != n - m + 1) ans -= sum[r] - dp[r];
            ans += (dp[r] - dp[l - 1]) * x;
            
            System.out.println(ans % MOD);
        }
    }
}
