#include <bits/stdc++.h>
using namespace std;

string s, t;
int cnt[128];
string solve() {
  auto cmp = [&](char a, char b) {
    return cnt[a] == cnt[b] ? a < b : cnt[a] < cnt[b];
  };
  int n = s.size();
  fill(cnt, cnt + 128, 0);
  for (char& c : s) ++cnt[c];
  for (char& c : t) ++cnt[c];
  string ans(n, '?');
  int pos = 0, pos2 = 0;
  for (int i = 0; i < n; i++) {
    int p = -1, p2 = -1;
    for (int j = 'a'; j <= 'z'; j++) {
      if (s[pos] == j && (t[pos2] != j || cnt[j] % 2 == 1))
        p = j;
      if (t[pos2] == j && (s[pos] != j || cnt[j] % 2 == 1)) p2 = j;
    }
    if (pos == -1 || cnt[ans[pos]] != cnt[p]) ans[i] = p, pos = find(s.begin(), s.end(), p) - s.begin();
    if (pos2 == -1 || cnt[ans[pos2]] != cnt[p2]) ans[i] = p2, pos2 = find(t.begin(), t.end(), p2) - t.begin();
    cnt[ans[i]]++;
  }
  return ans;
}

int main() {
  cin >> s >> t;
  cout << solve() << "\n";
}
