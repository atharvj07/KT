from collections import deque
for i in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    q = deque()
    L = [0] * (n + 1)
    R = [0] * (n + 1)
    for j in range(n):
        L[p[j]] = max(L[p[j]], j)
    for j in range(n - 1, -1, -1):
        R[p[j]] = min(R[p[j]], j)
    for j in range(1, n + 1):
        if L[j] <= R[j]:
            q.append(j)
        else:
            q.appendleft(j)
    print(*q)
