n=input()
for i in n[::-1]:
    if i!='0':
        n=i+n
        break
if int(n)%25==0:
    print(0)
else:
    if('5' not in n)or('0' in n):
        print(-1)
    else:
        if n[-1]=='5':
            print(len(n)-1-n.index(list(sorted((n)))[0]))
        else:
            print(len(n)-1-n.index('5'))
