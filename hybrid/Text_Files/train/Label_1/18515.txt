#include<bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.rbegin(), a.rend());
  int sets = 0;
  int cur = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] > cur) {
      cur++;
      sets++;
    }
  }
  cout << sets << endl;
}
