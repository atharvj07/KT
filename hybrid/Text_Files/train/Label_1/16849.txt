#include <bits/stdc++.h>
using namespace std;
vector<int> a, b;
int main() {
    int n, m, ans = 0;
    cin >> n >> m;
    a.resize(n), b.resize(m);
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < m; ++i) cin >> b[i];
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    for (int i = 0, j = 0; i < n && j < m;) {
        if ((a[i] + b[j]) & 1)
            ++i, ++j, ++ans;
        else if (a[i] + b[j] > 1)
            --j;
        else
            ++i;
    }
    cout << ans;
}
