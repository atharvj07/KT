n, c = map(int, input().split())
p_i, t_i = [list(map(int, input().split())) for _ in range(2)]
sum_p_i = sum(p_i)
p_i.sort()
t_i.sort()
radewoosh = sum_p_i - c * sum(t_i)
limak = sum(p_i) - c * sum(reversed(t_i))
if radewoosh > limak:
    print("Radewoosh")
elif radewoosh == limak:
    print("Tie")
else:
    print("Limak")
