#include <bits/stdc++.h>

using namespace std;

typedef pair<int, int> pi;

int n, m, k, s;
vector<string> grid;
vector<bool> vis;

int di[] = {-1, 0, 1, 0};
int dj[] = {0, 1, 0, -1};

bool valid(int i, int j) {
    return i >= 0 && i < n && j >= 0 && j < m && !vis[i * m + j] && grid[i][j] != '#';
}

void dfs(int i, int j) {
    vis[i * m + j] = true;
    for (int k = 0; k < 4; k++) {
        int ni = i + di[k], nj = j + dj[k];
        if (valid(ni, nj)) dfs(ni, nj);
    }
}

int main() {
    cin >> n >> m >> k;
    grid.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> grid[i];
    }
    s = count(grid.begin(), grid.end(), string(m, '.'));
    vis.resize(n * m);
    dfs(0, 0);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (!vis[i * m + j] && grid[i][j] == '.') {
                k--;
                grid[i][j] = 'X';
                if (k == 0) break;
            }
        }
        if (k == 0) break;
    }
    for (int i = 0; i < n; i++) {
        cout << grid[i] << '\n';
    }
    return 0;
}
