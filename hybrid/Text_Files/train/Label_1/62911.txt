def solve_zombie_contamination():
    N = int(input())
    grid = [input() for _ in range(N)]

    # Finding the minimum and maximum x and y coordinates
    min_x = min_y = N
    max_x = max_y = -1

    for y in range(N):
        for x in range(len(grid[y])):
            if grid[y][x] != '0':
                min_x = min(min_x, x)
                max_x = max(max_x, x)
                min_y = min(min_y, y)
                max_y = max(max_y, y)

    # Check if the rectangle formed by (min_x, min_y) and (max_x, max_y) is valid
    for y in range(N):
        for x in range(N):
            expected = 0
            if min_x <= x <= max_x and min_y <= y <= max_y:
                if y == min_y or y == max_y:
                    expected += 1
                if x == min_x or x == max_x:
                    expected += 1
            if int(grid[y][x]) != expected:
                print("No")
                return
    print("Yes")

solve_zombie_contamination()
