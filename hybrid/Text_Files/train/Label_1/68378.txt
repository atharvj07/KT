import sys
n = int(input())
g = [[]]
for i in range(n): g.append([])
for i in range(n - 1):
  a, b = map(int, input().split())
  g[a].append(b)
  g[b].append(a)

dep = [-1] * (n + 1)
ans = 0
def dfs(v: int, p: int):
  global ans
  cnt = 0
  for u in g[v]:
    if u == p: continue
    dep[u] = dep[v] + 1
    ans += dep[v]
    cnt += 1
    dfs(u, v)
  if cnt < 2: print(0, 0, 0, end = '')
  else:
    print(*sorted([v, g[v][0], g[v][1]]), sep = ' ')

dep[1] = 0
dfs(1, 1)
