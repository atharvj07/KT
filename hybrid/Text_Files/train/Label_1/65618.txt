n, k = map(int, input().split())
field = [input() for _ in range(n)]

def count_ships(r, c):
    count = 0
    # Horizontal check
    for start in range(max(0, c - k + 1), min(n - k + 1, c + 1)):
        if all(field[r][i] == '.' for i in range(start, start + k)):
            count += 1
    # Vertical check
    for start in range(max(0, r - k + 1), min(n - k + 1, r + 1)):
        if all(field[i][c] == '.' for i in range(start, start + k)):
            count += 1
    return count

max_count = 0
best_cell = (1, 1)
for r in range(n):
    for c in range(n):
        if field[r][c] == '.':
            current_count = count_ships(r, c)
            if current_count > max_count:
                max_count = current_count
                best_cell = (r + 1, c + 1)
print(*best_cell)
