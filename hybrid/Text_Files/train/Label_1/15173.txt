#include <iostream>
#include <cmath>
using namespace std;

int main() {
  int x2;
  cin >> x2;
  for (int i = x2; i >= 4; i--) {
    int j = (int) sqrt(i);
    bool ok = true;
    for (; j > 1; j--) {
      if (i % j == 0) {
        if (i / j < x2) {
          ok = false;
          break;
        }
      }
    }
    if (ok) {
      cout << i << endl;
      return 0;
    }
  }
  return 0;
}
