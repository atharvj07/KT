using namespace std;

int n, x, y;
vector<int> adj[300001];

int dfs(int u, int prev, bool vis_x) {
    int cnt = 0;
    for (auto v : adj[u]) {
        if (v == prev) {
            continue;
        }
        if (v == x) {
            cnt += dfs(v, u, true);
        } else if (v == y) {
            if (vis_x) {
                continue;
            }
            cnt += dfs(v, u, false);
        } else {
            cnt += dfs(v, u, vis_x);
        }
    }
    return cnt + 1;
}

int main() {
    int a, b;
    cin >> n >> x >> y;
    for (int i = 0; i < n - 1; i++) {
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (i == x || i == y) {
            continue;
        }
        ans += dfs(i, i, false) * dfs(i, i, true);
    }
    cout << ans << endl;
    return 0;
}
