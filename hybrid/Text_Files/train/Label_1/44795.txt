import java.util.*;

public class Main {

  static class Event implements Comparable<Event> {
    int type;
    String handle;

    Event(int type, String handle) {
      this.type = type;
      this.handle = handle;
    }

    @Override
    public int compareTo(Event other) {
      return this.handle.compareTo(other.handle);
    }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    PriorityQueue<Event> handles = new PriorityQueue<>();
    List<Event> events = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      int type = sc.nextInt();
      String handle = null;
      if (type == 2) {
        handle = sc.next();
      }
      events.add(new Event(type, handle));
    }
    int happyFriends = 0;
    for (Event event : events) {
      if (event.type == 1) {
        handles.clear();
      } else {
        handles.add(event);
        if (handles.peek().handle.equals(event.handle)) {
          happyFriends++;
        }
      }
    }
    System.out.println(happyFriends);
  }
}
