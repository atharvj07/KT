#include<bits/stdc++.h>
using namespace std;
#define ll long long

const int N=200005;

int main()
{
	int n,k;
	cin>>n>>k;

	vector<int> v(n);

	for(int i=0;i<n;++i)
	{
		cin>>v[i];
	}

	sort(v.begin(),v.end());

	int ans=0,mn=v[0],mx=v[n-1];

	while(n-k>mn && k-1<mx)
	{
		mn++,mx--,n-=2,ans+=2;
	}

	while(n-k>mn)
	{
		mn++,n--,ans++;
	}

	while(k-1<mx)
	{
		mx--,n--,ans++;
	}

	cout<<ans<<endl;
}
