#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<string> s(n);
  for (int i = 0; i < n; ++i) { cin >> s[i]; }
  vector<vector<int>> dp(n + 1, vector<int>(k + 1, 0x3f3f3f3f));
  dp[n][0] = 0;
  for (int i = n - 1; i >= 0; --i) {
    for (int j = 0; j <= k; ++j) {
      int skip = 0x3f3f3f3f;
      if (j < k) {
        for (int t = 0; t < m; ++t) {
          if (s[i][t] == '0') { continue; }
          skip = min(skip, dp[i + 1][j + 1] + m - t - 1);
        }
      }
      int go = 0x3f3f3f3f;
      for (int t = 0; t < m; ++t) {
        if (s[i][t] == '0') { continue; }
        go = min(go, dp[i + 1][j] + t + 1);
      }
      dp[i][j] = min(dp[i][j], skip);
      dp[i][j] = min(dp[i][j], go);
    }
  }
  cout << dp[0][0] << endl;
  return 0;
}
