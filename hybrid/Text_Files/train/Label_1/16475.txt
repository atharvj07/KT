#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    unordered_map<int, int> cnt;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        cnt[x]++;
    }
    int min_deletions = 0;
    if (cnt[4] >= n / 6) {
        min_deletions = 0;
    } else {
        min_deletions = n / 6 - cnt[4];
        cnt[8] -= min_deletions;
        cnt[15] -= min_deletions;
        cnt[16] -= min_deletions;
        cnt[23] -= min_deletions;
        cnt[42] -= min_deletions;
        if (cnt[8] < 0 || cnt[15] < 0 || cnt[16] < 0 || cnt[23] < 0 || cnt[42] < 0) {
            min_deletions = -1;
        }
    }
    cout << min_deletions << "\n";
}
