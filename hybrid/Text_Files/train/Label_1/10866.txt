#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
const int N=1000005,mod=1e9+7;
const int X=1000;
int32_t main(){
    ios_base::sync_with_stdio(false);cin.tie(0);cout.tie(0);
    int n;
    cin>>n;
    int row[X+1]={},col[X+1]={};
    int r[2*X+1]={},c[2*X+1]={},a[2*X+1]={},b[2*X+1]={};
    ll ans=0;
    while(n--){
        int x,y;
        cin>>x>>y;
        if(row[x]==0&&col[y]==0){
            row[x]=++r[x+y];
            col[y]=++c[x-y+X];
            ++a[x+y];
            ++b[x-y+X];
        }
        else{
            ans+=(a[x+y]-row[x])+(c[x-y+X]-col[y]);
            --a[x+y];
            --b[x-y+X];
            row[x]=0;
            col[y]=0;
        }
    }
    cout<<ans<<endl;
}
