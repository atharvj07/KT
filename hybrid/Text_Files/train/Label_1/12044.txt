#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int sum = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] >= 0) {
            break;
        }
        sum += a[i];
    }
    int ans = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (a[i] <= 0) {
            break;
        }
        ans += a[i];
    }
    ans += sum;
    cout << ans << endl;
    return 0;
}
