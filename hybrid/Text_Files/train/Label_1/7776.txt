#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int m, k;
    cin >> m >> k;
    vector<int> a(k);
    for (int i = 0; i < k; i++) cin >> a[i];
    vector<bool> seen(k, false);
    vector<bool> taken(k, false);
    for (int i = 0; i < m - 1; i++) {
      int t, r;
      cin >> t >> r;
      if (t != 0) seen[t - 1] = true;
      if (r) taken[t - 1] = true;
    }
    for (int i = 0; i < k; i++) {
      if (seen[i] && !taken[i]) a[i]--;
    }
    for (int i = 0; i < k; i++) {
      if (a[i] < 0) cout << "Y";
      else cout << "N";
    }
    cout << endl;
  }
  return 0;
}
