#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pi;

int main() {
    ll a, h, w;
    cin >> a >> h >> w;
    if (a <= 0 || h <= 0 || w <= 0 || (h < a && w < a))
        cout << -1 << endl;
    else {
        ll x1 = (w - a) / 2;
        ll x2 = (h - a) / 2;
        ll x = min(x1, x2);
        if (x >= 0)
            cout << (double)x << endl;
        else
            cout << -1 << endl;
    }
}
