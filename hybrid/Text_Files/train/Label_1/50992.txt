import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

class Main {
    static final int MOD = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        long n = Long.parseLong(br.readLine());

        // dp[i][0] = number of zebroid sequences ending with i and ending with blue
        // dp[i][1] = number of zebroid sequences ending with i and ending with red
        long[][] dp = new long[(int) n + 1][2];
        dp[1][0] = 1;
        dp[1][1] = 1;

        for (int i = 2; i <= n; i++) {
            dp[i][0] = (dp[i - 1][0] + dp[i - 1][1]) % MOD;
            dp[i][1] = dp[i - 1][0];
        }

        System.out.println((dp[(int) n][0] + dp[(int) n][1]) % MOD);
    }
}
