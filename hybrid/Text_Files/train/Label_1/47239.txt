import java.util.Scanner;
import java.util.Stack;

public class RegularBracketSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int testCases = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        for (int i = 0; i < testCases; i++) {
            String s = scanner.nextLine();
            int maxMoves = getMaxMoves(s);
            System.out.println(maxMoves);
        }
    }

    private static int getMaxMoves(String s) {
        int maxMoves = 0;
        Stack<Character> stack = new Stack<>();

        for (char c : s.toCharArray()) {
            if (c == '(' || c == '[') {
                stack.push(c);
            } else if (c == ')' || c == ']') {
                if (!stack.isEmpty() && stack.peek() == getMatchingBracket(c)) {
                    stack.pop();
                    maxMoves++;
                } else {
                    // Invalid bracket sequence, reset stack
                    stack.clear();
                }
            }
        }

        return maxMoves;
    }

    private static char getMatchingBracket(char c) {
        switch (c) {
            case ')':
                return '(';
            case ']':
                return '[';
            default:
                throw new IllegalArgumentException("Invalid bracket: " + c);
        }
    }
}
