import sys

n = int(input())

# Store the points in a list
points = [tuple(map(int, input().split())) for _ in range(n)]

# Create a dictionary to store the number of occurrences of each point
point_counts = {}
for point in points:
    if point not in point_counts:
        point_counts[point] = 0
    point_counts[point] += 1

# Calculate the number of parallelograms
parallelograms = 0
for point in point_counts:
    parallelograms += point_counts[point] * (point_counts[point] - 1) // 2

print(parallelograms)
