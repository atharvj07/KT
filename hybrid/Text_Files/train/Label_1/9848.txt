#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k;
    cin>>n>>k;
    vector<int>v(n+5);
    for(int i=0;i<n;i++){
        cin>>v[i];
    }
    int candies=0;
    for(int i=0;i<n;i++){
        candies+=v[i];
        if(candies>=k){
            cout<<i+1<<endl;
            return 0;
        }
        candies=min(candies,8);
    }
    cout<<-1<<endl;
}
