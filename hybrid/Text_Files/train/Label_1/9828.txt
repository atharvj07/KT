#include <bits/stdc++.h>
using namespace std;
const int maxn = 100005;
int n, q, typ, p, l, r, a[maxn << 1], b[maxn << 1];
int main() {
  ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cin >> n >> q;
  while (q--) {
    cin >> typ;
    if (typ == 1) {
      cin >> p;
      for (int i = p; i < n + p; ++i) a[i] = a[i - p], b[i] = b[i - p];
      for (int i = 0; i < p; ++i) a[i] = b[i] = 0;
      n = n + p;
    } else {
      cin >> l >> r;
      int ans = 0;
      for (int i = l; i < r; ++i) ans += a[i] != b[i];
      cout << ans << '\n';
    }
  }
}
