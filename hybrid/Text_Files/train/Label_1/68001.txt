T = int(input())
for _ in range(T):
    n = int(input())
    a = list(map(int, input().split()))
    last_occ = {}
    min_len = n + 1
    for i in range(n):
        if a[i] in last_occ:
            min_len = min(min_len, i - last_occ[a[i]] + 1)
        last_occ[a[i]] = i
    if min_len == n + 1:
        print(-1)
    else:
        print(min_len)
