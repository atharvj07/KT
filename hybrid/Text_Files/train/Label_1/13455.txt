#include <cstdio>
bool g[55][55];
int main() {
    int n;
    scanf("%d", &n);
    for (int i = n; i >= 1; i--) {
        for (int j = 1; j <= n; j++) {
            int x;
            scanf("%1d", &x);
            if (x > 0) g[i][j] = 1;
        }
    }
    for (int i = n; i >= 1; i--) {
        for (int j = n; j >= 1; j--) {
            if (g[i][j]) {
                for (int ii = i; ii <= n; ii++) {
                    for (int jj = j; jj <= n; jj++) {
                        if (g[ii][jj] == 0) goto end;
                    }
                }
                printf("Yes\n");
                return 0;
            }
        }
    }
end:
    printf("No\n");
    return 0;
}
