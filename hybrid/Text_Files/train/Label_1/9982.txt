#include <iostream>
#include <algorithm>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> t(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> t[i];
  }
  int l = 0, r = 0, cnt1 = 0, cnt2 = 0, ans = 0;
  while (r < n) {
    if (t[r] == 1) {
      cnt1++;
    } else {
      cnt2++;
    }
    while (cnt1 > cnt2) {
      if (t[l] == 1) {
        cnt1--;
      } else {
        cnt2--;
      }
      l++;
    }
    ans = std::max(ans, r - l + 1);
    r++;
  }
  std::cout << ans << std::endl;
  return 0;
}
