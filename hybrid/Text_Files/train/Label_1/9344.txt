#include <bits/stdc++.h>

int main() {
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    int mn = INT_MAX;
    for (int i = 0; i < n; i++) {
        int diff = 0;
        for (int j = i; j < i + 4; j++) {
            char c = s[j % n];
            if (c == 'A')
                c = 'Z';
            else if (c == 'Z')
                c = 'A';
            diff += std::min(abs(c - 'A'), 26 - abs(c - 'A'));
        }
        mn = std::min(mn, diff);
    }
    std::cout << mn << std::endl;
}
