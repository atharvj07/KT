import sys

def solve():
    t = int(sys.stdin.readline())

    for _ in range(t):
        n, k = map(int, sys.stdin.readline().split())
        a = list(map(int, sys.stdin.readline().split()))
        ans = 0

        for i in range(n):
            if a[i] % k != 0:
                ans += 1
                if (k - a[i] % k) % k > a[i] % k:
                    a[i] += (k - a[i] % k) % k
                else:
                    a[i] += a[i] % k
        print(ans)

solve()
