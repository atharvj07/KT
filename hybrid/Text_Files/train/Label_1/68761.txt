import sys

n, a = [int(x) for x in input().split()]
d, c = [], []
for i in range(n):
    dd, cc = [int(x) for x in input().split()]
    d.append(dd)
    c.append(cc)

dp = [0] * n
for i in range(n):
    dp[i] = a - c[i]
    if i > 0:
        dp[i] = max(dp[i], dp[i - 1] + a - c[i])

for i in range(1, n):
    diff = d[i] - d[i - 1]
    dp[i] -= diff * diff
    if i > 1:
        dp[i] = max(dp[i], dp[i - 2] - diff * diff)
print(max(dp))
