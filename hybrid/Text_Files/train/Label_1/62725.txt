n = int(input())
a = list(map(int, input().split()))

if n <= 2:
    print(n)
else:
    inc = [1] * n  # Length of increasing subarray ending at each position
    dec = [1] * n  # Length of increasing subarray starting at each position

    for i in range(1, n):
        if a[i] > a[i-1]:
            inc[i] = inc[i-1] + 1

    for i in range(n-2, -1, -1):
        if a[i] < a[i+1]:
            dec[i] = dec[i+1] + 1

    max_len = max(max(inc), max(dec))

    for i in range(1, n-1):
        if a[i-1] < a[i+1]:
            max_len = max(max_len, inc[i-1] + dec[i+1])

    print(max_len)
