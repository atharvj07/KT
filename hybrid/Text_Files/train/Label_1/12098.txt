#include <algorithm>
#include <cstdio>
#include <vector>
using namespace std;
int main() {
  int n, m, p;
  scanf("%d%d%d", &n, &m, &p);
  vector<int> a(n);
  vector<int> b(m);
  for (int i = 0; i < n; ++i) scanf("%d", &a[i]);
  for (int i = 0; i < m; ++i) scanf("%d", &b[i]);
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  vector<int> res;
  for (int q = 0; ; ++q) {
    if (q + (m - 1) * p > n) break;
    if (a[q] == b[0]) {
      bool valid = true;
      for (int i = 0; i < m; ++i) {
        if (a[q + i * p] != b[i]) {
          valid = false;
          break;
        }
      }
      if (valid) res.push_back(q + 1);
    }
  }
  printf("%d\n", (int)res.size());
  for (int x : res) printf("%d ", x);
  puts("");
  return 0;
}
