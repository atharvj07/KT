#include<bits/stdc++.h>
using namespace std;
int main() {
	int n;
	cin >> n;
	vector<int> a(n + 3);
	for (int i = 1; i <= n; i++) {
		cin >> a[i];
		a[i + 1] = a[i + 1] + a[i];
	}
	int ans = 0;
	int mn = 0;
	for (int i = 1; i <= n; i++) {
		mn = min(mn, a[i]);
		ans += (a[i + 1] >= mn);
	}
	cout << ans << "\n";
}
