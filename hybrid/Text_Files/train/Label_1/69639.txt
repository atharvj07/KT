def count_replacements(s, t):
    from math import gcd

    # Count the number of zeros and ones in s
    zeros = s.count('0')
    ones = s.count('1')

    # Length of the t string
    n = len(t)

    # Calculate the greatest common divisor of the counts of zeros and ones
    g = gcd(zeros, ones)

    # Try all divisors of g as possible lengths of the repeating pattern
    result = 0
    for length in range(1, int(g**0.5) + 1):
        if g % length == 0:
            # Check if the pattern of this length is valid
            if all(t[i] == t[i+length*zeros] for i in range(length*zeros)):
                result += 1
            # Check the divisor pair
            div = g // length
            if div != length and all(t[i] == t[i+div*zeros] for i in range(div*zeros)):
                result += 1

    return result

# Read input
s = input().strip()
t = input().strip()

# Print the result
print(count_replacements(s, t))
