#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;

    vector<vector<int>> codes(n, vector<int>(6));
    for (int i = 0; i < n; i++) {
        string code;
        cin >> code;
        for (int j = 0; j < 6; j++) {
            codes[i][j] = code[j] - '0';
        }
    }

    int max_errors = 0;
    for (int i = 0; i < n; i++) {
        int min_errors = 6;
        for (int j = 0; j < n; j++) {
            int errors = 0;
            if (i != j) {
                for (int k = 0; k < 6; k++) {
                    errors += (codes[i][k] != codes[j][k]);
                }
            }
            min_errors = min(min_errors, errors);
        }
        max_errors = max(max_errors, min_errors);
    }

    cout << max_errors << endl;

    return 0;
}
