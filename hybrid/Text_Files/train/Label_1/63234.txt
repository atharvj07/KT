n, p, s = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = sorted(zip(a, b, range(n)), key=lambda x: (x[0] + x[1], x[0]))
t1 = []
t2 = []
ans = 0
for a, b, i in c:
    if len(t1) < p:
        t1.append(i + 1)
        ans += a
    elif len(t2) < s:
        t2.append(i + 1)
        ans += b
print(ans)
print(*t1)
print(*t2)
