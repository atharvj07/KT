import sys

for _ in range(int(input())):
    n = int(input())
    a = [[0] * n for _ in range(n)]
    if n % 2 == 1 or n == 2:
        print(-1)
        continue
    m = n * n
    for i in range(n):
        for j in range(n):
            a[i][j] = m
            m -= 1
    for i in range(1, n):
        for j in range(i % 2, n, 2):
            a[i][j], a[i - 1][j] = a[i - 1][j], a[i][j]
    for i in a:
        print(*i)
