#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  getline(cin, s);
  int ans = 0;
  bool neg = false;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '-') neg = true;
    else if (s[i] == '+') neg = false;
    else {
      int x = (int)(s[i] - '0');
      ans = neg ? ans - x : ans + x;
    }
  }
  cout << ans << '\n';
  return 0;
}
