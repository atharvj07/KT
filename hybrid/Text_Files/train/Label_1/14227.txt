#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    int ans = 0;
    vector<int> open(4, 0);
    for (int i = 0; i < n; i++) {
        if (s[i] == '<' || s[i] == '{' || s[i] == '[' || s[i] == '(') {
            open[s[i] - '<']++;
        }
        else {
            int j = s[i] - ')';
            if (open[j] == 0) {
                ans++;
            } else {
                open[j]--;
            }
        }
    }
    for (int i = 0; i < 4; i++) {
        ans += open[i];
    }
    if (ans > n) {
        cout << "Impossible" << endl;
    }
    else {
        cout << ans << endl;
    }
    return 0;
}
