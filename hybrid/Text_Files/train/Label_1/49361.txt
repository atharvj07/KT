
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.StringTokenizer;

class Main {
    static final int MAX_N = 100_000;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        List<Edge>[] graph = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
        while (m-- > 0) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken()), v = Integer.parseInt(st.nextToken());
            char c = st.nextToken().charAt(0);
            graph[u].add(new Edge(v, c == 'R'));
            graph[v].add(new Edge(u, c == 'R'));
        }
        int[] dist = new int[n + 1];
        for (int i = 1; i <= n; i++) dist[i] = -1;
        Queue<Integer> queue = new LinkedList<>();
        queue.add(1);
        dist[1] = 0;
        int result = -1;
        int last = 0;
        while (!queue.isEmpty()) {
            int cur = queue.poll();
            if (cur == n) {
                result = dist[cur];
                last = cur;
                break;
            }
            for (Edge next : graph[cur]) {
                if (dist[next.v] != -1) continue;
                queue.add(next.v);
                dist[next.v] = dist[cur] + 1;
            }
        }
        if (result == -1) {
            System.out.println(-1);
            return;
        }
        List<Integer> path = new ArrayList<>();
        while (last != 1) {
            path.add(last);
            for (Edge next : graph[last]) {
                if (dist[next.v] == dist[last] - 1) {
                    last = next.v;
                    break;
                }
            }
        }
        path.add(1);
        System.out.println(result);
        for (int i = path.size() - 1; i >= 0; i--) {
            System.out.print(path.get(i) + " ");
        }
    }

    static class Edge {
        int v;
        boolean color;

        public Edge(int v, boolean color) {
            this.v = v;
            this.color = color;
        }
    }
}
