#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    int ans = 0;
    for (int l = 0, r = n - 1; l < r; l++) {
        while (l < r && a[l] + a[r] > 100) { r--; }
        if (l < r) { ans++; }
    }
    cout << ans << endl;
    return 0;
}
