#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
int main()
{
    long long n,type,ans=0,l,r;
    cin>>n;
    unordered_map<long long,pair<long long,long long>> mp;
    for(long long i=1;i<=n;i++)
    {
        cin>>type;
        if(mp.count(type)==0)
        {
            mp[type].first=i;
            mp[type].second=i;
        }
        else
        {
           mp[type].second=i;
        }
    }
    for(auto it:mp)
    {
        if(it.second.first==it.second.second)
        {
            continue;
        }
        else
        {
           ans++;
           l=it.second.first;
           r=it.second.second;
           cout<<l<<" "<<r<<endl;
        }
    }
    if(ans==0)
    {
        cout<<-1;
    }
    return 0;
}
