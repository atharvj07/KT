def solve(n: int, m: int, a: int, b: int) -> str:
    if a * m != b * n:
        return 'NO'
    arr = ['0' * m for _ in range(n)]
    for i in range(n):
        for j in range(a):
            arr[i][(i + j) % m] = '1'
    return 'YES\n' + '\n'.join(arr)


for _ in range(int(input())):
    print(solve(*map(int, input().split())))
