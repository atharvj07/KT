#include<bits/stdc++.h>
using namespace std;
#define mp make_pair
#define pb push_back
#define ff first
#define ss second
pair<int,int>ar[100001];
pair<int,int>v[100001];
int32_t main() {
	int n; cin >> n;
	for (int i = 1; i <= n; i++) {
		cin >> ar[i].ff >> ar[i].ss; v[i] = ar[i];
	}
	sort(v + 1, v + n + 1);
	for (int i = 1; i <= n; i++) {
		int x; cin >> x;
		if (x != v[i].ss - v[i].ff) {
			cout << "NO\n"; return 0;
		}
	}
	cout << "YES\n";
	for (int i = 1; i <= n; i++) {
		for (int j = 1; j <= n; j++) {
			if (ar[j] == v[i]) {
				cout << ar[j].ff << " " << ar[j].ss << endl;
				ar[j] = {-1, -1};
				break;
			}
		}
	}
	return 0;
}
