#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<pair<int, int>> rooks(m);
  for (int i = 0; i < m; i++) {
    int x, y;
    cin >> x >> y;
    rooks[i] = {x, y};
  }
  vector<int> rows(n), cols(n);
  long long ans = 1LL * n * n;
  for (auto& [x, y] : rooks) {
    rows[x - 1]++;
    cols[y - 1]++;
    ans -= 2 * (rows[x - 1] - 1) + 2 * (cols[y - 1] - 1) + 1;
    cout << ans << '\n';
  }
  return 0;
}
