import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        ArrayList<Integer>[] adj = new ArrayList[N + 1];
        for (int i = 1; i <= N; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < M; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        int[] color = new int[N + 1];
        Queue<Integer> q = new LinkedList<>();
        int ans = 0;
        for (int i = 1; i <= N; i++) {
            if (color[i] != 0) continue;
            color[i] = 1;
            q.add(i);
            int cnt = 0;
            int same = 0;
            while (!q.isEmpty()) {
                int cur = q.poll();
                cnt++;
                for (int u : adj[cur]) {
                    if (color[u] == color[cur]) same++;
                    if (color[u] == 0) {
                        color[u] = -color[cur];
                        q.add(u);
                    }
                }
            }
            ans += cnt - same - 1;
        }
        System.out.println(ans);
    }
}
