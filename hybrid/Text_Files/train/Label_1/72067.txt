import sys
ds = {}
for q in range(int(input())):
    t, v, u, w = [], 0, 0, 0
    t, v, u = map(int, input().split())
    if t == 1:
        w = int(input())
        for i in range(v, u + 1):
            if i & 1:
                ds[i >> 1] = ds.get(i >> 1, []) + [(u, w)]
            else:
                ds[i >> 1] = [(v, w)] + ds.get(i >> 1, [])
    else:
        for i in range(v, u + 1):
            if i & 1:
                w += max(0, ds.get(i >> 1, [])[0][1]
                          if len(ds[i >> 1]) else 0)
            else:
                w += max(0, ds.get(i >> 1, [])[-1][1]
                          if len(ds[i >> 1]) else 0)
        print(w)
