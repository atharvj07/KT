#include<bits/stdc++.h>
#define mod 998244353
#define maxn 25
using namespace std;
long long c[maxn][maxn], f[maxn], b[maxn], a[maxn], s, t, k;
int main() {
    scanf("%d %d %lld", &t, &k, &s);
    for (int i = 1; i <= t; i++) { scanf("%lld", &a[i]); }
    c[0][0] = 1;
    for (int i = 1; i <= t; i++) {
        c[i][0] = 1;
        for (int j = 1; j <= i; j++) { c[i][j] = (c[i - 1][j - 1] + c[i - 1][j]) % mod; }
    }
    f[0] = 1;
    for (int i = 1; i <= t; i++) { f[i] = f[i - 1] * i % mod; }
    for (int i = 1; i <= t; i++) {
        long long ans = 1;
        for (int l = 1; l <= k; l++) {
            b[l] = (s - a[i] * l) / f[l];
            if ((s - a[i] * l) % f[l]) continue;
            if (b[l] >= 1 && b[l] <= t) { ans = (ans * c[b[l]][l]) % mod; }
        }
        s -= a[i];
        ans = (ans * f[k]) % mod;
        cout << ans << endl;
    }
    return 0;
}
