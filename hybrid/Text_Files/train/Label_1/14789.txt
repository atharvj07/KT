#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int n, q;
long long a[N], pay[N];
int main() {
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }
    cin >> q;
    for (int i = 1; i <= q; ++i) {
        int op;
        cin >> op;
        if (op == 1) {
            int x;
            cin >> x >> a[x];
        } else {
            long long x;
            cin >> x;
            pay[n + 1] = x;
            for (int j = n; j >= 1; --j) {
                if (a[j] < x) {
                    a[j] = x;
                } else {
                    pay[j] = x;
                    break;
                }
            }
        }
    }
    for (int i = n; i >= 1; --i) {
        pay[i] = max(pay[i], pay[i + 1]);
    }
    for (int i = 1; i <= n; ++i) {
        cout << max(a[i], pay[i]) << " ";
    }
    cout << endl;
    return 0;
}
