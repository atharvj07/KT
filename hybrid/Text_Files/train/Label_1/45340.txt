import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[] d = new long[n], c = new long[m];
        for (int i = 0; i < n; i++) d[i] = sc.nextInt();
        for (int i = 0; i < m; i++) c[i] = sc.nextInt();
        long[][] dp = new long[n + 1][m + 1];
        for (int i = 0; i <= n; i++) Arrays.fill(dp[i], Long.MAX_VALUE);
        dp[0][0] = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (dp[i][j] == Long.MAX_VALUE) continue;
                dp[i + 1][j + 1] = Math.min(dp[i + 1][j + 1], dp[i][j] + d[i] * c[j]);
                dp[i][j + 1] = Math.min(dp[i][j + 1], dp[i][j]);
            }
        }
        System.out.println(dp[n][m]);
    }
}
