#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mod = 1e9 + 7;
int p[1000005];
vector<int> divs[1000005];
void sieve() {
    p[0] = p[1] = 1;
    for(int i = 2; i * i < 1000005; i++) {
        if(p[i] == 0) {
            for(int j = i * i; j < 1000005; j += i) {
                p[j] = i;
            }
        }
    }
    for(int i = 2; i < 1000005; i++) {
        if(p[i] == 0) {
            p[i] = i;
        }
        int x = i, cnt = 0;
        while(x > 1) {
            if(p[x] == cnt) cnt++;
            else cnt = 1;
            divs[i].push_back(p[x]);
            x /= p[x];
        }
    }
}
ll fastpow(ll base, ll p) {
    if(p == 0) {
        return 1;
    }
    if(p == 1) {
        return base;
    }
    ll ret = fastpow(base, p / 2);
    ret = ret * ret % mod;
    if(p % 2) {
        ret = ret * base % mod;
    }
    return ret;
}
int main() {
    sieve();
    int n;
    cin >> n;
    vector<int> a(n);
    map<vector<int>, ll> mp;
    for(auto &x : a) {
        cin >> x;
        vector<int> cur_divs;
        for(auto y : divs[x]) {
            cur_divs.push_back(y);
        }
        sort(cur_divs.begin(), cur_divs.end());
        mp[cur_divs]++;
    }
    ll ans = 0;
    for(auto &x : mp) {
        ll cnt = x.second;
        ll cur = 1;
        for(auto y : x.first) {
            cur *= y;
            cur %= mod;
        }
        ans += cur * cnt % mod * (cnt + 1) % mod / 2;
        ans %= mod;
    }
    cout << ans << endl;
    return 0;
}
