n = int(input())
board = []; has_odd = False
for _ in range(n): board.append(list(input()))
for row in range(n):
    for col in range(n):
        if board[row][col] == 'o':
            cnt = (board[max(0, row - 1)][max(0, col - 1)] == 'o')
            cnt += (board[min(n - 1, row + 1)][max(0, col - 1)] == 'o')
            cnt += (board[max(0, row - 1)][min(n - 1, col + 1)] == 'o')
            cnt += (board[min(n - 1, row + 1)][min(n - 1, col + 1)] == 'o')
            cnt += (board[row][max(0, col - 1)] == 'o')
            cnt += (board[row][min(n - 1, col + 1)] == 'o')
            cnt += (board[max(0, row - 1)][col] == 'o')
            cnt += (board[min(n - 1, row + 1)][col] == 'o')
            if cnt % 2 == 1: has_odd = True
if has_odd: print('NO')
else: print('YES')
