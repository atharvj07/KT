#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

struct Cup {
    long long c, w;
    bool operator<(const Cup& rhs) const { return c > rhs.c; }
};

int main() {
    int n, m, d;
    cin >> n >> m >> d;
    vector<Cup> p(n), i(m);
    for (auto& x : p) cin >> x.c >> x.w;
    for (auto& x : i) cin >> x.c >> x.w;
    sort(p.begin(), p.end());
    sort(i.begin(), i.end());
    int l = 0, r = 0;
    long long ans = 0;
    while (l < n && p[l].w <= d) ans += p[l++].c;
    while (r < m && i[r].w + p[l - 1].w <= d) ans += i[r++].c;
    while (r < m && i[r].w <= d) ans += i[r++].c;
    cout << ans << endl;
}
