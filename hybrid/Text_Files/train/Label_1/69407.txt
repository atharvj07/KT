import sys

t = int(input())

for _ in range(t):
    n, k1, k2 = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    assert len(a) == k1 and len(b) == k2 and k1 + k2 == n
    a.sort()
    b.sort()
    if a[0] > b[-1]:
        print("YES")
    else:
        print("NO")
