import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long sum = 0;
        for (long v : a) {
            sum += v;
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] > 0 && sum > 0) {
                sum--;
                a[i]--;
                if (a[i] > 0 && sum > 0) {
                    sum--;
                    a[i]--;
                    if (a[i] > 0) {
                        ans++;
                    } else {
                        sum++;
                        a[i]++;
                    }
                }
                sum++;
                a[i]++;
            }
        }
        System.out.println(ans);
    }
}
