from collections import defaultdict

def read_tree(n):
    tree = defaultdict(list)
    for _ in range(n - 1):
        u, v = map(int, input().split())
        tree[u].append(v)
        tree[v].append(u)
    return tree

def find_max_depth_node(tree, n):
    def dfs(node, parent, depth):
        depths[node] = depth
        for child in tree[node]:
            if child != parent:
                dfs(child, node, depth + 1)
    depths = [0] * (n + 1)
    dfs(1, -1, 0)
    max_depth_node = depths.index(max(depths))
    return max_depth_node, depths

def find_star_transform_operations(n, tree):
    if n == 1:
        return 0
    # Find a node with maximum depth
    max_depth_node, depths = find_max_depth_node(tree, n)
    # Find the node with maximum depth from the previously found node to ensure it's on the diameter
    farthest_node, new_depths = find_max_depth_node(tree, max_depth_node)
    # The tree can be converted to a star in ceil(diameter / 2) operations
    diameter = new_depths[farthest_node]
    return (diameter + 1) // 2

n = int(input())
tree = read_tree(n)
print(find_star_transform_operations(n, tree))
