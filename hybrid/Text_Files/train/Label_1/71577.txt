import sys

def pow(x, y, MOD):
    ret = 1
    while y > 0:
        if y % 2 == 0:
            x = x * x
            x %= MOD
            y /= 2
        else:
            ret = ret * x
            ret %= MOD
            y -= 1
    return ret

n, p = map(int, input().split())
a = list(map(int, input().split()))

ans = []
for t in range(0, n+1):
    val = pow(t, n-1, p)
    good = True
    for aa in a:
        if aa % p == 0:
            good = False
            break
        if pow(aa, t, p) == val:
            good = False
            break
    if good:
        ans.append(t)

print(len(ans))
print(*ans)
