#include <cstdio>
#include <algorithm>
#include <tuple>
#include <vector>
using namespace std;
int main() {
    long long n;
    scanf("%lld", &n);
    vector<tuple<long long, long long>> v;
    for (int i = 0; i < n; i++) {
        long long a, p;
        scanf("%lld%lld", &a, &p);
        v.push_back(make_tuple(p, a));
    }
    sort(v.begin(), v.end());
    long long ans = 0, meat = 0;
    for (auto i : v) {
        long long p, a;
        tie(p, a) = i;
        if (a <= meat) {
            meat -= a;
        } else {
            ans += (a - meat) * p;
            meat = 0;
        }
    }
    printf("%lld\n", ans);
    return 0;
}
