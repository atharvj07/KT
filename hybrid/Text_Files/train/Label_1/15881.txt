#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, pos, l, r;
  cin >> n >> pos >> l >> r;
  vector<int> tabs(n + 1, n);
  for (int i = l; i <= r; ++i) tabs[i] = -1;
  int a = n, b = 1;
  for (int i = 1; i <= n; ++i) {
    a = min(a, tabs[i]);
    b = max(b, tabs[i]);
  }
  int ans = 0;
  while (a < pos) {
    ++ans;
    a = min(a + 1, b);
  }
  while (b > pos) {
    ++ans;
    b = max(a, b - 1);
  }
  cout << ans;
  return 0;
}
