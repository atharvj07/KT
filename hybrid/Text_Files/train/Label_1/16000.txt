#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const ll MAX = 1e9 + 5;

bool vis[MAX];

int main() {
  ll n;
  cin >> n;
  vector<ll> a(n);
  for (auto &it : a) cin >> it;
  sort(a.begin(), a.end());
  for (auto &it : a) vis[it] = 1;
  ll cur = 0, ans = 0;
  for (auto &it : a) {
    if (cur >= it) return cout << -1 << endl, 0;
    ll mn = MAX;
    for (ll j = 1; j * j <= it; j++) {
      if (it % j == 0) {
        if (j != it / j) {
          if (!vis[j]) mn = min(mn, j);
          if (!vis[it / j]) mn = min(mn, it / j);
        } else {
          if (!vis[j]) mn = min(mn, j);
        }
      }
    }
    if (mn == MAX) return cout << -1 << endl, 0;
    cur += mn;
    ans++;
    vis[mn] = vis[it] = 1;
  }
  cout << ans << endl;
  return 0;
}
