from collections import Counter

def trans(A):
    a = Counter(A)
    return [a[-1], a[0], a[1]]

for _ in range(int(input())):
    N = int(input())
    A = trans(list(map(int, input().split())))
    B = trans(list(map(int, input().split())))
    print(['NO', 'YES'][(A[0] == B[0] and A[1] <= B[1] and A[2] >= B[2])])
