import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] a = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextInt();
        }
        if (N == 2) {
            System.out.println("No");
            return;
        }
        int[] gcd = new int[N];
        gcd[0] = a[0];
        for (int i = 1; i < N; i++) {
            gcd[i] = gcd(gcd[i - 1], a[i]);
        }
        int[] l = new int[N];
        l[N - 1] = 1;
        for (int i = N - 2; i >= 0; i--) {
            l[i] = l[i + 1];
            if (gcd[i] % 4 == 0) {
                l[i] = Math.max(l[i], 2);
            }
        }
        int[] r = new int[N];
        r[0] = 1;
        for (int i = 1; i < N; i++) {
            r[i] = r[i - 1];
            if (gcd[i - 1] % 4 == 0) {
                r[i] = Math.max(r[i], 2);
            }
        }
        for (int i = 0; i < N; i++) {
            if (l[i] == r[i]) {
                System.out.println("Yes");
                return;
            }
        }
        System.out.println("No");
    }

    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
