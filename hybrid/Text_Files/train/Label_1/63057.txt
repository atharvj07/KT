import sys
from collections import Counter

n = int(sys.stdin.readline())
s = [sys.stdin.readline().strip() for i in range(n)]
m = len(max(s, key=len))
d = []
for i in range(m):
    c = Counter()
    for j in s:
        if i < len(j):
            c[j[i]] += 1
    d.append(c)
ans = []
for i in range(10):
    msk = -1
    for j in d:
        if j[chr(ord('a') + i)]:
            if msk == -1 or j[chr(ord('a') + i)] < j[chr(ord('a') + msk)]:
                msk = i
    if msk != -1:
        ans.append(msk)
        for j in d:
            j[chr(ord('a') + msk)] = 0
print(''.join(map(str, ans)))
