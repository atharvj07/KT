from sys import stdin

r, c = map(int, stdin.readline().strip().split())
grid = [list(stdin.readline().strip()) for _ in range(r)]
max_eaten = 0

for i in range(r):
    if 'S' not in grid[i]:
        max_eaten += c

for j in range(c):
    if 'S' not in [grid[i][j] for i in range(r)]:
        max_eaten += r

print(max_eaten)
