#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &i : a) { cin >> i; }
  sort(begin(a), end(a));
  int ans = 0;
  for (int i = 0; i + 1 < n; i += 2) {
    if (a[i] + a[i + 1] == a[i + 2] + a[i + 3]) ans++;
  }
  cout << ans + (n % 2 != 0);
  return 0;
}
