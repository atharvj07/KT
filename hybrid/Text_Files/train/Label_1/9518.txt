#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), ans(n);
  unordered_map<int, int> mp;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (!mp[a[i]]) {
      for (int j = i; j < n; j++) ans[j]++;
      mp[a[i]] = 1;
    }
  }
  for (int i = 0; i < m; i++) {
    int l;
    cin >> l;
    cout << ans[l - 1] << endl;
  }
  return 0;
}
