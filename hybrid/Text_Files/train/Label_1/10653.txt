#include<iostream>
#include<vector>
using namespace std ;
int q, n, a[105] ;
int main() {
    cin >> q ;
    while( q-- ) {
        cin >> n ;
        for( int i = 1 ; i <= n ; i++ ) {
            cin >> a[i] ;
        }
        int tg = n ;
        for( int i = n ; i >= 1 ; i-- ) {
            if( a[i] == i ) { continue ; }
            else {
                tg = i ;
                break ;
            }
        }
        if( tg == n ) {
            for( int i = 1 ; i <= n ; i++ ) {
                cout << i << ' ' ;
            }
            cout << '\n' ;
            continue ;
        }
        int pos = 0 ;
        for( int i = tg ; i <= n ; i++ ) {
            if( a[i] == tg ) { pos = i ; break ; }
        }
        for( int i = pos ; i > tg ; i-- ) {
            swap( a[i], a[i-1] ) ;
        }
        for( int i = 1 ; i <= n ; i++ ) {
            cout << a[i] << ' ' ;
        }
        cout << '\n' ;
    }
    return 0 ;
}
