#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int main() {
    int n, r, avg;
    cin >> n >> r >> avg;

    vector<pair<int, int>> exams(n);

    for (int i = 0; i < n; i++) {
        cin >> exams[i].first >> exams[i].second;
    }

    priority_queue<pair<int, int>> pq;

    for (int i = 0; i < n; i++) {
        pq.push(make_pair(exams[i].second - (r - exams[i].first + avg - 1), i));
    }

    ll essays = 0;
    while (!pq.empty()) {
        auto [diff, idx] = pq.top();
        pq.pop();
        if (diff > 0) {
            essays += diff;
            exams[idx].first++;
            pq.push(make_pair(exams[idx].second - (r - exams[idx].first + avg - 1), idx));
        }
    }

    cout << essays << endl;
    return 0;
}
