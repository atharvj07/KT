#include<iostream>
#include<vector>
using namespace std;
int main(){
  int n;
  cin>>n;
  vector<int> p(n+1),b(n+1);
  vector<int>pos(n+1),need(n+1);
  for(int i=1;i<=n;i++){
    cin>>p[i];if(pos[p[i]]==0)pos[p[i]]=i;
  }
  for(int i=1;i<=n;i++)cin>>b[i];
  int ch=0;
  for(int i=1;i<=n;i++){
    if(p[i]!=pos[i])ch++;
    need[p[i]]=1;
  }
  for(int i=1;i<=n;i++)if(!need[i])ch++;
  cout<<ch<<endl;
}
