def min_length_after_operations(n, arr):
    dp = [[0] * n for _ in range(n)]
    for length in range(1, n + 1):
        for start in range(n - length + 1):
            end = start + length - 1
            dp[start][end] = 1 + dp[start + 1][end] if start < end else 1
            for mid in range(start + 1, end + 1):
                if arr[start] == arr[mid]:
                    cost = dp[start + 1][mid - 1] + dp[mid + 1][end]
                    dp[start][end] = min(dp[start][end], cost)
    return dp[0][n - 1]

n = int(input().strip())
arr = list(map(int, input().strip().split()))
print(min_length_after_operations(n, arr))
