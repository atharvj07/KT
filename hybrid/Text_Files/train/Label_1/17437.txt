#include<bits/stdc++.h>
using namespace std;
string s, t, ans;
int fro[30], bac[30], fre[30], h[30];
int main() {
  cin >> s >> t;
  for (int i = 0; i < t.size(); i++) {
    int c = t[i] - 'a';
    fre[c]++;
  }
  int last = -1;
  for (int i = 0; i < s.size(); i++) {
    int c = s[i] - 'a';
    if (c >= 0 && c < 30) {
      ans += s[i];
      fro[c] = i;
      bac[c] = i;
      continue;
    }
    int l = 0, r = t.size() - 1;
    while (l <= r) {
      int m = (l + r) / 2;
      if (fre[t[m] - 'a'] > h[t[m] - 'a']) {
        l = m + 1;
      } else {
        r = m - 1;
      }
    }
    if (l == t.size())
      l--;
    ans += t[l];
    int num = t[l] - 'a';
    fro[num] = i;
    bac[num] = i;
    h[num]++;
  }
  int j = 0;
  for (int i = 0; i < t.size(); i++) {
    while (j < s.size() && ans[j] != t[i]) j++;
    if (j == s.size()) {
      int l = -1;
      for (int k = 0; k < 30; k++) {
        if (h[k] == 0 && fre[k] > 0) {
          if (k > l) l = k;
        }
      }
      ans += l + 'a';
      fro[l] = j;
      bac[l] = j;
      h[l]++;
    } else {
      j++;
    }
  }
  for (int i = 0; i < s.size(); i++) {
    int c = s[i] - 'a';
    if (c >= 0 && c < 30) continue;
    int cnt = 0;
    for (int j = 0; j < t.size(); j++) {
      if (fro[t[j] - 'a'] < i && bac[t[j] - 'a'] > i) cnt++;
    }
    ans[i] = t[max_element(fre, fre + 30) - fre];
  }
  cout << ans << endl;
  return 0;
}
