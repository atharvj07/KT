#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> colors(n + 1);
  for (int i = 2; i <= n + 1; i++) {
    bool found = false;
    for (int j = 2; j * j <= i; j++) {
      if (i % j == 0) {
        found = true;
        if (colors[j] != 0) {
          colors[i] = colors[j] + 1;
        } else if (colors[i / j] != 0) {
          colors[i] = colors[i / j] + 1;
        }
        break;
      }
    }
    if (!found) colors[i] = 1;
  }
  int max_color = 0;
  for (int i = 1; i <= n + 1; i++) max_color = max(max_color, colors[i]);
  cout << max_color << endl;
  for (int i = 1; i <= n; i++) cout << colors[i + 1] << " ";
  cout << endl;
}
