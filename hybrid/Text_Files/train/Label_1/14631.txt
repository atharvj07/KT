#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, l, r, k;
    cin >> n >> l >> r >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int cnt = 0;
    sort(a.begin(), a.end());
    for (int i = 0; i < n; i++) {
      if (a[i] >= l && a[i] <= r) {
        if (k >= a[i]) {
          cnt++;
          k -= a[i];
        } else
          break;
      }
    }
    cout << cnt << endl;
  }
}
