#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int buses = 0;
  int cur = 0;
  for (int i = 0; i < n; i++) {
    if (cur + a[i] <= m) {  // Can fit in current bus
      cur += a[i];
    } else {                  // Need a new bus
      buses++;
      cur = a[i];
    }
  }
  if (cur > 0) buses++;
  cout << buses << endl;
  return 0;
}
