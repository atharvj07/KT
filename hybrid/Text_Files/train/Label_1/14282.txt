#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll maxn = 3e5 + 5;
const int MAX_K = 60;
const ll inf = 2e18;
ll n, a[maxn], dp[MAX_K][maxn][2], L[MAX_K][maxn], R[MAX_K][maxn], pref[maxn], suf[maxn];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    for (ll i = 1; i <= n; i++) {
        cin >> a[i];
        pref[i] = pref[i - 1] + a[i];
    }
    for (ll i = n; i >= 1; i--) {
        suf[i] = suf[i + 1] + a[i];
    }
    for (ll k = 0; k <= 60; k++) {
        for (ll i = 0; i <= n; i++) {
            dp[k][i][0] = -inf;
            dp[k][i][1] = -inf;
        }
    }
    dp[0][n][0] = pref[n];
    dp[0][n][1] = suf[n];
    for (ll k = 1; k <= 60; k++) {
        for (ll i = 1; i <= n; i++) {
            if (dp[k - 1][i + 1][1] != -inf) {
                ll cur = suf[i] + dp[k - 1][i + 1][1];
                if (dp[k][i][0] < cur) {
                    dp[k][i][0] = cur;
                    L[k][i] = 1;
                }
            }
            if (dp[k - 1][i - 1][0] != -inf) {
                ll cur = pref[i - 1] + dp[k - 1][i - 1][0];
                if (dp[k][i][0] < cur) {
                    dp[k][i][0] = cur;
                    L[k][i] = 2;
                }
            }
            if (dp[k - 1][i - 1][1] != -inf) {
                ll cur = pref[i - 1] + dp[k - 1][i - 1][1];
                if (dp[k][i][1] < cur) {
                    dp[k][i][1] = cur;
                    R[k][i] = 1;
                }
            }
            if (dp[k - 1][i + 1][0] != -inf) {
                ll cur = suf[i] + dp[k - 1][i + 1][0];
                if (dp[k][i][1] < cur) {
                    dp[k][i][1] = cur;
                    R[k][i] = 2;
                }
            }
        }
    }
    for (ll i = 0; i <= n - 1; i++) {
        cout << dp[i][1][0] << " ";
    }
    cout << "\n";
    return 0;
}
