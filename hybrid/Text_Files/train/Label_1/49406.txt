import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    while (true) {
      int n = sc.nextInt();
      if (n == 0) break;
      int k = sc.nextInt();

      int[] a = new int[n];
      for (int i = 0; i < n; i++) {
        a[i] = sc.nextInt();
      }

      if (!check(a, k)) {
        System.out.println("NO GAME");
        continue;
      }

      int max = 0;
      for (int i = 0; i < n - k + 1; i++) {
        int p = 1;
        for (int j = 0; j < k; j++) {
          p *= a[i + j];
        }
        max = Math.max(max, p);
      }

      int min = Integer.MAX_VALUE;
      for (int i = 0; i < n - k + 1; i++) {
        int p = 1;
        for (int j = 0; j < k; j++) {
          p *= a[i + j];
        }
        min = Math.min(min, p);
      }

      System.out.println(max - min);
    }
  }

  static boolean check(int[] a, int k) {
    for (int i = 0; i < a.length - k + 1; i++) {
      int p = 1;
      for (int j = 0; j < k; j++) {
        p *= a[i + j];
      }
      int max = a[i];
      int min = a[i];
      for (int j = i + 1; j < i + k; j++) {
        max = Math.max(max, a[j]);
        min = Math.min(min, a[j]);
      }
      if (p != max * min) {
        return true;
      }
    }
    return false;
  }
}
