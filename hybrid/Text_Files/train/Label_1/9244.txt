#include <iostream>
#include <map>

using namespace std;

int main() {
  int n;
  cin >> n;
  map<int, int> last;
  bool ok = true;
  for (int i = 0; i < n; i++) {
    int x, k;
    cin >> x >> k;
    if (x < last[k]) ok = false;
    last[k] = x;
  }
  cout << (ok ? "YES" : "NO") << endl;
  return 0;
}
