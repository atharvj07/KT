import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, d = map(int, input().split())
    a = [int(x) for x in input().split()]
    b = a[n - d:] + a[:n - d]
    for i in range(n):
        a[i] &= b[i]
    if sum(a) == 0:
        print(1)
    else:
        c = a.copy()
        steps = 1
        while c != a:
            b = c[n - d:] + c[:n - d]
            for i in range(n):
                c[i] &= b[i]
            steps += 1
        if sum(a) == 0:
            print(steps)
        else:
            print(-1)
