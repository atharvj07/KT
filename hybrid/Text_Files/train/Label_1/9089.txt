#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> doors(n);
    for (int i = 0; i < n; ++i) cin >> doors[i];
    int left = 0, right = 0;
    int ans = -1;
    for (int i = 0; i < n; ++i) {
        if (doors[i] == 0)
            ++left;
        else
            ++right;
        if (left && right) {
            ans = i + 1;
            break;
        }
    }
    cout << ans << '\n';
    return 0;
}
