#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  sort(v.begin(), v.end());
  long long ans = 1e18;
  for (int i = 0; i + k - 1 < n; i++) { ans = min(ans, 1LL * (v[i + k - 1] - v[i])); }
  cout << ans << endl;
}
