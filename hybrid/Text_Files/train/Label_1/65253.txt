for t in range(int(input())):
    n = int(input())
    a = input()
    b = ''
    c = ''
    for i in range(26):
        for j in a:
            if ord(j) == ord('a') + i:
                if j == 'g':
                    c += 'g'
                elif j == 'u':
                    c += 'u'
                else:
                    b += j
    b += c
    print(b)
