#include <bits/stdc++.h>
typedef long long LL;
using namespace std;
const int maxn = 5e5 + 7;
const int mod = 1e9 + 7;
int a[maxn][2][2];
vector<int> G[maxn];
int col[maxn];
bool ok[4];
bool dfs(int x, int c) {
    col[x] = c;
    ok[c] = true;
    for (int i : G[x]) {
        for (int j = 0; j < 2; ++j) {
            for (int k = 0; k < 2; ++k) {
                if (a[x][j][k] == a[i][k][j]) {
                    if (col[i] == c)
                        return false;
                    if (!col[i] && !dfs(i, 3 - c))
                        return false;
                }
            }
        }
    }
    return true;
}
int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j < 2; ++j) {
            for (int k = 0; k < 2; ++k) {
                cin >> a[i][j][k];
            }
        }
    }
    for (int i = 1; i <= n; ++i) {
        for (int j = i + 1; j <= n; ++j) {
            bool isok = true;
            for (int k = 0; k < 2; ++k) {
                for (int l = 0; l < 2; ++l) {
                    if (a[i][k][l] == a[j][l][k])
                        isok = false;
                }
            }
            if (isok) {
                G[i].push_back(j);
                G[j].push_back(i);
            }
        }
    }
    bool flag = true;
    for (int i = 1; i <= n && flag; ++i) {
        if (!col[i]) {
            memset(ok, 0, sizeof ok);
            if (!dfs(i, 1))
                flag = false;
        }
    }
    if (!flag) {
        cout << "NO";
        return 0;
    }
    cout << "YES" << endl;
    for (int i = 1; i <= n; ++i)
        cout << col[i] << endl;
    return 0;
}
