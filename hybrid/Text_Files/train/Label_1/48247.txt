import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long m = in.nextLong();
        long x = in.nextLong();

        if (x != 1 && x != m) {
            System.out.println((n - 2) * (m - 1) * (m - 1) % 1000000007);
            return;
        }

        if (x == 1) {
            System.out.println((n - 1) * (m - 1) * (m - 2) % 1000000007);
        } else {
            System.out.println((n - 1) * (m - 2) * (m - 1) % 1000000007);
        }
    }
}
