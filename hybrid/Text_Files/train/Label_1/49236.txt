import java.util.Scanner;

public class Sellout {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int f = scanner.nextInt();

        long[][] days = new long[n][2];
        for (int i = 0; i < n; i++) {
            days[i][0] = scanner.nextInt();
            days[i][1] = scanner.nextInt();
        }

        // Sort days in descending order of clients
        sort(days);

        // Calculate maximum products sold without sellout
        long maxSoldWithoutSellout = 0;
        for (int i = 0; i < n; i++) {
            maxSoldWithoutSellout += Math.min(days[i][0], days[i][1]);
        }

        // Calculate maximum products sold with sellout
        long maxSoldWithSellout = maxSoldWithoutSellout;
        for (int i = 0; i < f; i++) {
            maxSoldWithSellout += days[i][0];
        }

        System.out.println(maxSoldWithSellout);
    }

    // Sort days in descending order of clients
    private static void sort(long[][] days) {
        for (int i = 0; i < days.length - 1; i++) {
            for (int j = i + 1; j < days.length; j++) {
                if (days[i][1] < days[j][1]) {
                    long[] temp = days[i];
                    days[i] = days[j];
                    days[j] = temp;
                }
            }
        }
    }
}
