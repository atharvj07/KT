#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
	int n, sum = 0, ans = 0;
	cin >> n;
	vector<int> a(n);
	for (int i = 0; i < n; i++) { cin >> a[i]; }
	unordered_map<int, int> m;
	for (int i = 0; i < n; i++) {
		sum += a[i];
		if (sum == 0)
			ans += i + 1;
		if (m[sum] != 0) {
			ans += i - m[sum];
		}
		m[sum] = i;
	}
	cout << ans << endl;
	return 0;
}
