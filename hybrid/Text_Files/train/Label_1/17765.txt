#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ull unsigned long long
const int maxn = 20 + 1, maxk = 30 + 1;
int n, k;
vector<int> a[maxk];
int main() {
  cin >> n >> k;
  for (int j = 0; j < k; j++) {
    string s;
    cin >> s;
    for (int i = 0; i < n; i++) a[j].push_back(s[i]);
    sort(a[j].begin(), a[j].end());
  }
  set<ull> st;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      for (int l = 0; l < n; l++) {
        if (i == j || i == l || j == l) continue;
        ull mask = 0;
        for (int p = 0; p < k; p++) {
          mask |= (1ULL << (a[p][i] == a[p][j] && a[p][i] == a[p][l]));
        }
        st.insert(mask);
      }
    }
  }
  cout << st.size() << endl;
  return 0;
}
