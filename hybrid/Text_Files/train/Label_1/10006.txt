#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, a, b;
  cin >> n >> a >> b;
  if (a == n && b == n) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  vector<bool> v(n + 1, 0);
  for (int i = 1; i <= n; i++)
    if (!v[i]) {
      for (int j = i + 1; j <= n; j++)
        if (!v[j]) {
          if (a > 1) {
            a--;
            cout << "1";
          } else
            cout << "0";
          if (b > 1) {
            b--;
            cout << "1" << endl;
          } else
            cout << "0" << endl;
          v[i] = v[j] = 1;
          goto end;
        }
  end:;
    }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
      if (i != j) cout << "0";
      else
        cout << "1";
  cout << endl;
  return 0;
}
