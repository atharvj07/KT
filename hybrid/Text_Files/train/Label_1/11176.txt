#include <bits/stdc++.h>
#define vi(x) vector<x>
#define pb push_back
using namespace std;
const long long mod = 1e9 + 7;
const char nl = '\n';
long long a[200001], l[200001], r[200001], b[200001];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { cin >> l[i] >> r[i]; }
    for (int i = 1; i <= m; i++) { cin >> a[i]; }
    sort(l + 1, l + 1 + n);
    sort(a + 1, a + 1 + m);
    for (int i = 1; i <= n - 1; i++) {
        int low = 1, high = m, ans = 0;
        while (low <= high) {
            int mid = (low + high) / 2;
            if (a[mid] == r[i] - l[i + 1]) {
                ans = mid;
                break;
            } else if (a[mid] > r[i] - l[i + 1]) {
                high = mid - 1;
            } else {
                low = mid + 1;
            }
        }
        if (ans == 0) {
            cout << "No" << nl;
            return 0;
        }
        b[i] = ans;
        l[i + 1] = r[i + 1];
    }
    cout << "Yes" << nl;
    for (int i = 1; i <= n - 1; i++) { cout << b[i] << " "; }
    cout << nl;
    return 0;
}
