t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    p = list(map(int, input().strip().split()))
    max_gap = 0
    current_max = 0
    for i in p:
        if i > current_max:
            max_gap = max(max_gap, i - current_max - 1)
            current_max = i
    if max_gap >= n:
        print("NO")
    else:
        print("YES")
