#include <iostream>
const int MOD = 998244353;
int n; long long a[200005], dp[200005][1010];
int main() {
    std::cin >> n;
    for (int i = 1; i <= n; i++) std::cin >> a[i];
    dp[0][0] = 1LL;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= a[i]; j++) {
            for (int k = 0; k <= a[i]; k++)
                if (j != k) (dp[i][k] += dp[i - 1][j]) %= MOD;
        }
    }
    std::cout << dp[n][0] << std::endl;
    return 0;
}
