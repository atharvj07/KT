#include <iostream>
#include <unordered_map>
#include <string>

int main() {
    int n;
    std::cin >> n;
    std::string s, t;
    std::cin >> s >> t;
    std::unordered_map<std::string, int> t_freq;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j)
                continue;
            std::string sub_str = s.substr(0, i) + s.substr(i + 1);
            t_freq[sub_str]++;
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j)
                continue;
            std::string sub_str = t.substr(0, i) + t.substr(i + 1);
            if (t_freq.count(sub_str) > 0) {
                ans += t_freq[sub_str];
                t_freq.erase(sub_str);
            }
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
