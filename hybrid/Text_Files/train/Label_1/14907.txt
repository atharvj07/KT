#include<bits/stdc++.h>
using namespace std;
#define PAR pair<int,int>
#define COOR pair<int,PAR>
#define F first
#define S second
int dx[] = {0,0,1,-1}, dy[] = {1,-1,0,0};
unordered_map<int,vector<COOR>> x,y;
int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    a = '0' + a;
    b = '0' + b;
    int timer = 0;
    x[0] = {COOR(0,PAR(0,0))};
    y[0] = {COOR(0,PAR(0,0))};
    for(int i = 1;i < n;i++){
        int x1 = x[i-1][0].F + dx[a[i]-'S'], y1 = x[i-1][0].S.F + dy[a[i]-'S'];
        int x2 = x[i-1][0].S.F , y2 = x[i-1][0].S.S;
        x[i] = {COOR(i,PAR(x1,y1)), COOR(i,PAR(x2,y2))};
        x1 = y[i-1][0].F + dx[b[i]-'S'], y1 = y[i-1][0].S.F + dy[b[i]-'S'];
        x2 = y[i-1][0].S.F , y2 = y[i-1][0].S.S;
        y[i] = {COOR(i,PAR(x1,y1)), COOR(i,PAR(x2,y2))};
    }
    for(int i = 0;i < x[n-1].size();i++){
        for(int j = 0;j < y[n-1].size();j++){
            if(x[n-1][i] == y[n-1][j]){
                cout << "YES" << endl;
                return 0;
            }
        }
    }
    cout << "NO" << endl;
    return 0;
}
