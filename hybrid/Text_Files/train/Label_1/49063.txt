import java.util.Scanner;

public class Main {
    private static Scanner scanner = new Scanner(System.in);

    public static void main(String[] args) {
        long file = scanner.nextLong();
        long time = scanner.nextLong();
        long regularTime = scanner.nextLong();
        int opt1Time = scanner.nextInt();
        int opt1Size = scanner.nextInt();
        int opt1Cost = scanner.nextInt();
        int opt2Time = scanner.nextInt();
        int opt2Size = scanner.nextInt();
        int opt2Cost = scanner.nextInt();

        long ans = (file + regularTime - 1) / regularTime * opt1Time > time ? -1 : opt1Cost * (file / opt1Size + ((file % opt1Size) != 0 ? 1 : 0));
        long tmp = (file + regularTime - 1) / regularTime * opt2Time;
        if (tmp <= time) ans = Math.min(ans, opt2Cost * (file / opt2Size + ((file % opt2Size) != 0 ? 1 : 0)));

        System.out.println(ans);
    }
}
