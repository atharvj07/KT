import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int Ciel = 0, Jiro = 0;
        for (int i = 0; i < n; i++) {
            int size = sc.nextInt();
            if (size % 2 == 1) {
                Ciel += sc.nextInt();
                for (int j = 1; j < size; j++) sc.nextInt();
            } else {
                int temp = 0;
                for (int j = 0; j < size; j++) temp += sc.nextInt();
                Jiro += temp;
            }
        }
        System.out.println(Ciel + " " + Jiro);
    }
}
