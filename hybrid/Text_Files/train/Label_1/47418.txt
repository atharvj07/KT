import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

class Main {
    private static class Graph {
        private int[][] adj;

        public Graph(int n) {
            adj = new int[n + 1][n + 1];
        }

        public void addEdge(int u, int v) {
            adj[u][v] = 1;
        }

        public boolean dfs(int u, int v, boolean[] visited) {
            if (u == v) return true;
            visited[u] = true;
            for (int i = 1; i < adj.length; i++) {
                if (adj[u][i] == 1 && !visited[i]) {
                    if (dfs(i, v, visited)) return true;
                }
            }
            return false;
        }

        public boolean hasPath(int u, int v) {
            boolean[] visited = new boolean[adj.length];
            return dfs(u, v, visited);
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] deployment = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                deployment[i][j] = in.nextInt();
            }
        }
        int[][] tests = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                tests[i][j] = in.nextInt();
            }
        }
        Graph g = new Graph(n);
        boolean possible = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                if (deployment[i][j] == 1 && tests[i][j] == 0) {
                    g.addEdge(1, i + 1);
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                if (deployment[i][j] == 1 && g.hasPath(1, i + 1)) {
                    possible = false;
                    break;
                }
            }
            if (!possible) break;
        }
        if (!possible) {
            System.out.println("Impossible");
        } else {
            System.out.println("Possible");
            int[] ct = new int[n + 1];
            for (int i = 0; i < n; i++) {
                if (!g.hasPath(1, i + 1)) {
                    ct[i + 1] = 1;
                }
            }
            System.out.println(Arrays.toString(ct).replace("[", "").replace("]", ""));
            int m = 0;
            for (int i = 1; i < g.adj.length; i++) {
                for (int j = 1; j < g.adj.length; j++) {
                    if (g.adj[i][j] == 1) m++;
                }
            }
            System.out.println(m);
            for (int i = 1; i < g.adj.length; i++) {
                for (int j = 1; j < g.adj.length; j++) {
                    if (g.adj[i][j] == 1) System.out.println(i + " " + j);
                }
            }
        }
    }
}
