for _ in range(int(input())):
    s = input()
    for i in range(len(s)):
        a = ord(s[i])
        for c in range(a + 1, ord('z') + 1):
            if chr(c) not in s[i + 1:]:
                s = s[:i] + chr(c) + s[i + 1:]
                break
        else:
            for c in range(ord('a'), a):
                if chr(c) not in s[i + 1:]:
                    s = s[:i] + chr(c) + s[i + 1:]
                    break
    print(s)
