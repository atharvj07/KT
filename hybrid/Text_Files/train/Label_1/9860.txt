#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 1e5;
vector<int> adj[MAXN];
vector<long long> a(MAXN), ans(MAXN), sum(MAXN), b(MAXN, -1e18);

void dfs_sum(int v, int p) {
  sum[v] = a[v];
  for (int u : adj[v]) {
    if (u == p) continue;
    dfs_sum(u, v);
    sum[v] += max(0LL, sum[u]);
  }
}

void dfs(int v, int p, long long val) {
  val += a[v];
  b[v] = max(b[v], val);
  for (int u : adj[v]) {
    if (u == p) continue;
    dfs(u, v, val);
  }
  if (b[v] > ans[v]) {
    ans[v] = b[v];
    for (int u : adj[v]) {
      if (u == p) continue;
      sum[v] -= max(0LL, sum[u]);
      dfs(u, v, val + max(0LL, sum[v] - max(0LL, sum[u])));
      sum[v] += max(0LL, sum[u]);
    }
  }
}

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 1; i < n; i++) {
    int u, w;
    cin >> u >> w;
    u--, w = -w;
    adj[u].push_back(i);
    adj[i].push_back(u);
  }
  dfs_sum(0, 0);
  dfs(0, 0, 0);
  long long res = 0;
  for (int i = 0; i < n; i++) {
    if (ans[i] > a[i]) res++;
  }
  cout << res << '\n';
  return 0;
}
