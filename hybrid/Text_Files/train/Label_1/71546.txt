def floyd_warshall(graph, n):
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if graph[i][k] + graph[k][j] < graph[i][j]:
                    graph[i][j] = graph[i][k] + graph[k][j]

def sum_of_shortest_paths(graph, n):
    sum_paths = 0
    for i in range(n):
        for j in range(n):
            sum_paths += graph[i][j]
    return sum_paths

def main():
    n = int(input())
    graph = [list(map(int, input().split())) for _ in range(n)]
    removal_order = list(map(int, input().split()))
    removal_order.reverse()

    for i in range(n):
        for j in range(n):
            if i != j and graph[i][j] == 0:
                graph[i][j] = float('inf')

    results = []
    for x in removal_order:
        floyd_warshall(graph, n)
        results.append(sum_of_shortest_paths(graph, n))
        for i in range(n):
            graph[i][x-1] = graph[x-1][i] = float('inf')

    for result in reversed(results):
        print(result, end=' ')

main()
