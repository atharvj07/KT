#include <bits/stdc++.h>
#define f first
#define s second
using namespace std;
using ll=long long;
using pi=pair<int, int>;
int n, m, q, ans=1e9;
map<pi, int> mp;
void upd(int a, int b){
    if(a<=n && b<=m)ans=min(ans, a+b);
}
int main(){
    cin>>n>>m>>q;
    for(int i=0;i<q;i++){
        int x, y; cin>>x>>y;
        mp[{x, y}]=1;
        upd(x+y, x+y+1);
        upd(x+y+1, x+y);
    }
    int p=0;
    for(int i=2;i<=n+m;i++){
        while(p+1<=i && !mp[{p+1, i-p-1}]){
            p++;
            upd(p+1, i-p-1);
            upd(i-p-1, p+1);
        }
    }
    cout<<ans;
    return 0;
}
