import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt();
        int M = in.nextInt();
        int K = in.nextInt();
        int[] shoppingMalls = new int[K];

        for (int i = 0; i < K; i++) {
            shoppingMalls[i] = in.nextInt();
        }

        Map<Integer, List<Edge>> graph = new HashMap<>();
        for (int i = 0; i < M; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            int l = in.nextInt();

            if (!graph.containsKey(a)) {
                graph.put(a, new LinkedList<>());
            }
            if (!graph.containsKey(b)) {
                graph.put(b, new LinkedList<>());
            }

            graph.get(a).add(new Edge(b, l));
            graph.get(b).add(new Edge(a, l));
        }

        int minDistance = Integer.MAX_VALUE;
        for (int i = 1; i <= N; i++) {
            int distance = dijkstra(graph, i, shoppingMalls);
            if (distance != -1) {
                minDistance = Math.min(minDistance, distance);
            }
        }

        if (minDistance == Integer.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(minDistance);
        }
    }

    private static int dijkstra(Map<Integer, List<Edge>> graph, int source, int[] destinations) {
        PriorityQueue<Node> pq = new PriorityQueue<>(Comparator.comparingInt(n -> n.distance));
        pq.add(new Node(source, 0));

        Map<Integer, Integer> distances = new HashMap<>();
        distances.put(source, 0);

        while (!pq.isEmpty()) {
            Node currentNode = pq.poll();

            if (isDestination(currentNode.vertex, destinations)) {
                return currentNode.distance;
            }

            for (Edge edge : graph.getOrDefault(currentNode.vertex, new ArrayList<>())) {
                int distance = currentNode.distance + edge.length;
                if (!distances.containsKey(edge.destination) || distance < distances.get(edge.destination)) {
                    distances.put(edge.destination, distance);
                    pq.add(new Node(edge.destination, distance));
                }
            }
        }

        return -1;
    }

    private static boolean isDestination(int vertex, int[] destinations) {
        for (int destination : destinations) {
            if (vertex == destination) {
                return true;
            }
        }

        return false;
    }

    private static class Node {
        int vertex;
        int distance;

        public Node(int vertex, int distance) {
            this.vertex = vertex;
            this.distance = distance;
        }
    }

    private static class Edge {
        int destination;
        int length;

        public Edge(int destination, int length) {
            this.destination = destination;
            this.length = length;
        }
    }
}
