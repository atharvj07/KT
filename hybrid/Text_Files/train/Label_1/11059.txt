#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  int n;
  cin >> s >> n;
  for (int i = 0; i < s.size(); i++) {
    if ((s[i] >= 'a' && s[i] + n <= 'z') || (s[i] >= 'A' && s[i] + n <= 'Z'))
      s[i] += n;
    else
      s[i] -= 26 - n;
  }
  cout << s << "\n";
}
