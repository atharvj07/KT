import java.util.*;

public class Main {
    private static final int MOD = (int)1e9 + 7;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        long[] dp = new long[n + 1];
        dp[0] = 1;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < i; j++) {
                if (a[i - 1] >= a[j]) {
                    dp[i] = (dp[i] + dp[j]) % MOD;
                }
            }
        }

        System.out.println(dp[n]);
    }
}
