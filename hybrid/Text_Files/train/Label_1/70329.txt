import sys

def check(a, x):
    n = len(a)
    for i in range(n):
        idx = min(i + x, n - 1)
        for j in range(i + 1, idx + 1):
            if a[i] > a[j]:
                return False
    return True

for _ in range(int(input())):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    print("YES" if check(a, x) else "NO")
