import sys

def check(n, m, v, t, edges):
    deg = [0] * (n + 1)
    for a, b in edges:
        deg[a] += 1
        deg[b] += 1
    if (sum(v) - sum(t)) % (n - 1):
        return "NO"
    x = (sum(v) - sum(t)) // (n - 1)
    for a, b in edges:
        if v[a] + v[b] - t[a] - t[b] != x:
            return "NO"
    if all(x == v[i] - t[i] for i in range(1, n + 1)):
        return "NO"
    if all(v[i] <= t[i] for i in range(1, n + 1)):
        if (sum([t[i] - v[i] for i in range(1, n + 1)]) - (n - deg[1])) % 2 == 0:
            return "YES"
    if all(v[i] >= t[i] for i in range(1, n + 1)):
        if (sum([v[i] - t[i] for i in range(1, n + 1)]) - (n - deg[1])) % 2 == 0:
            return "YES"
    return "NO"

input = sys.stdin.readlines()
for I in range(int(input.pop(0))):
    n, m = map(int, input.pop(0).split())
    v = list(map(int, input.pop(0).split()))
    t = list(map(int, input.pop(0).split()))
    edges = []
    for i in range(m):
        a, b = map(int, input.pop(0).split())
        edges.append([a, b])
    print(check(n, m, v, t, edges))
