#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }
  std::vector<int> b;
  for (int i = 0; i < n; i++) {
    if (b.empty() || a[i] > b.back()) {
      b.push_back(a[i]);
    }
  }
  for (int i = 0, j = 0; i < b.size(); i++) {
    for (; j < n && a[j] <= b[i]; j++) {
      std::cout << a[j] << ' ';
    }
    std::cout << '\n';
  }
  return 0;
}
