import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long ans = 0;
        for (int i = 0; i < n; i++) {
            long a = scanner.nextInt();
            if (Math.sqrt(a) * Math.sqrt(a) != a) {
                ans = Math.max(a, ans);
            }
        }
        System.out.println(ans);
    }
}
