#include<bits/stdc++.h>
using namespace std;
vector<int> adj[3005];
int par[3005],coun=0,li1=-1,li2=-1,ti1=-1,ti2=-1,si1=-1,si2=-1,vis[3005];
void dfs(int no,int p,int l)
{
    if(no==ti1 && l<=li1)
    {
        coun++;
        li1=l;
        ti1=no;
        si1=p;
    }
    if(no==ti2 && l<=li2)
    {
        coun++;
        li2=l;
        ti2=no;
        si2=p;
    }
    vis[no]=1;
    for(auto u:adj[no])
    {
        if(vis[u]==0)
        {
            dfs(u,no,l+1);
        }
    }
}

int main()
{
    int n,m;cin>>n>>m;
    for(int i=0; i<m; i++)
    {
        int x,y;cin>>x>>y;
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    cin>>si1>>ti1>>li1;
    cin>>si2>>ti2>>li2;
    dfs(si1,-1,0);
    if(coun==2)
    {
        if((si2==ti1 && li1+1<=li2) || (si1==ti2 && li2+1<=li1))
        {
            cout<<0<<endl;
        }
        else
        {
            cout<<1<<endl;
        }
    }
    else
    {
        cout<<-1<<endl;
    }
}
