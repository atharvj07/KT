#include <bits/stdc++.h>
using namespace std;
int main() {
  const int N = 1e5 + 10;
  int n, a[N];
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> ans;
  int odd_cnt = 0, i = 0;
  for (int j = 0; j < n; j++) {
    if (a[j] % 2 == 1) odd_cnt++;
    else
      break;
    i++;
  }
  reverse(a, a + i);
  for (int j = i; j < n; j++) {
    if (a[j] % 2 == 0)
      ans.push_back(a[j]);
    else
      break;
  }
  for (int j = 0; j < odd_cnt; j++) ans.push_back(a[j]);
  for (int j = i; j < n; j++) {
    if (a[j] % 2 == 1)
      ans.push_back(a[j]);
  }
  for (int j = 0; j < ans.size(); j++) cout << ans[j] << " ";
  cout << endl;
  return 0;
}
