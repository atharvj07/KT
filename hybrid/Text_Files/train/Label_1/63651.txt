MOD = 1000000007

def comb(n, r, fact, inv_fact):
    return fact[n] * inv_fact[r] * inv_fact[n-r] % MOD

def solve(n, m, lights):
    lights.sort()
    fact = [1] * (n + 1)
    inv_fact = [1] * (n + 1)
    for i in range(2, n + 1):
        fact[i] = fact[i-1] * i % MOD
    inv_fact[n] = pow(fact[n], MOD - 2, MOD)
    for i in range(n - 1, 0, -1):
        inv_fact[i] = inv_fact[i + 1] * (i + 1) % MOD
    
    ways = 1
    for i in range(1, m):
        gap = lights[i] - lights[i-1] - 1
        ways = ways * pow(2, gap - 1, MOD) % MOD if gap else ways
        ways = ways * comb(gap + 1, 1, fact, inv_fact) % MOD
    
    ways = (ways * comb(n - m + 1, m, fact, inv_fact)) % MOD
    return ways

n, m = map(int, input().split())
lights = list(map(int, input().split()))
print(solve(n, m, lights))
