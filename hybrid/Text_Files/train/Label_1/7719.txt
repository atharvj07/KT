#include <bits/stdc++.h>
using ll = long long;
using namespace std;
const int N = 5e5 + 5;
int n, m, a[N], fen[N];  // fenwick tree

void upd(int x, int v) {
  for (; x <= n; x += x & -x) fen[x] = min(fen[x], v);
}
int get(int x) {
  int ret = INT_MAX;
  for (; x; x -= x & -x) ret = min(ret, fen[x]);
  return ret;
}
int32_t main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> a[i];
  memset(fen, 0x3f, sizeof fen);
  for (int i = n; i >= 1; i--) upd(i, a[i]);
  while (m--) {
    int l, r;
    cin >> l >> r;
    int ans = get(r);
    if (ans < a[l]) ans = -1;
    cout << ans << "\n";
  }
}
