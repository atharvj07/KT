from collections import Counter

a, b = map(int, input().split())
a = min(a, 12)
b = min(b, 12)
x = Counter()
y = Counter()
for i in range(2, a+1):
    k = i
    while k % 2 == 0:
        x[2] += 1
        k //= 2
    while k % 3 == 0:
        x[3] += 1
        k //= 3
    if k > 1:
        x[k] += 1
for i in range(2, b+1):
    k = i
    while k % 2 == 0:
        y[2] += 1
        k //= 2
    while k % 3 == 0:
        y[3] += 1
        k //= 3
    if k > 1:
        y[k] += 1
res = 1
for p, v in x.items():
    res *= p**min(v, y[p])
print(res)
