import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
    static long cross(long x1, long y1, long x2, long y2) {
        return x1 * y2 - x2 * y1;
    }

    static boolean sameSide(long x1, long y1, long x2, long y2, long x3, long y3) {
        return cross(x2 - x1, y2 - y1, x3 - x1, y3 - y1) > 0;
    }

    static boolean onLine(long x, long y, long x1, long y1, long x2, long y2) {
        return cross(x2 - x1, y2 - y1, x - x1, y - y1) == 0 &&
                Math.min(x1, x2) <= x && x <= Math.max(x1, x2) &&
                Math.min(y1, y2) <= y && y <= Math.max(y1, y2);
    }

    static long flip(long x, long l1, long l2) {
        return 2 * l1 - x;
    }

    static boolean see(long[] P, long[] Q, long[] W, long[] M) {
        boolean res = false;
        // Check if a direct line of sight exists
        if (!onLine(Q[0], Q[1], P[0], P[1], W[0], W[1]) &&
                !onLine(Q[0], Q[1], P[0], P[1], W[2], W[3])) {
            res = true;
        }
        // Check if a line of sight exists through the mirror
        else if (sameSide(P[0], P[1], Q[0], Q[1], M[0], M[1]) &&
                sameSide(P[0], P[1], Q[0], Q[1], M[2], M[3])) {
            long refX = flip(Q[0], M[0], M[2]);
            long refY = flip(Q[1], M[1], M[3]);
            res = !onLine(refX, refY, P[0], P[1], W[0], W[1]) &&
                    !onLine(refX, refY, P[0], P[1], W[2], W[3]);
        }
        return res;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        long[] P = new long[]{Long.parseLong(tokenizer.nextToken()), Long.parseLong(tokenizer.nextToken())};
        tokenizer = new StringTokenizer(reader.readLine());
        long[] Q = new long[]{Long.parseLong(tokenizer.nextToken()), Long.parseLong(tokenizer.nextToken())};
        tokenizer = new StringTokenizer(reader.readLine());
        long[] W = new long[]{Long.parseLong(tokenizer.nextToken()), Long.parseLong(tokenizer.nextToken()),
                Long.parseLong(tokenizer.nextToken()), Long.parseLong(tokenizer.nextToken())};
        tokenizer = new StringTokenizer(reader.readLine());
        long[] M = new long[]{Long.parseLong(tokenizer.nextToken()), Long.parseLong(tokenizer.nextToken()),
                Long.parseLong(tokenizer.nextToken()), Long.parseLong(tokenizer.nextToken())};
        System.out.println(see(P, Q, W, M) ? "YES" : "NO");
    }
}
