#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int sieveBound = 1e6;
bitset<sieveBound> isPrime;
vector<int> prime;
void sieve() {
    isPrime.set();
    for (int i = 2; i < sieveBound; i++) {
        if (isPrime[i]) {
            prime.push_back(i);
            if ((ll)i * i < sieveBound) for (int j = i * i; j < sieveBound; j += i) isPrime[j] = 0;
        }
    }
}
ll modPow(ll X, ll P, ll MOD) {
    if (P == 0) return 1;
    if (P == 1) return X;
    X = modPow(X, P / 2, MOD);
    X *= X;
    X %= MOD;
    if (P % 2 == 1) X *= X;
    X %= MOD;
    return X;
}
int numPrimitiveRoots(int p) {
    int cnt = 0;
    for (int i = 1; i < p; i++) {
        bool flag = true;
        for (int j = 0; j < prime.size() && prime[j] * prime[j] <= i; j++) {
            if (i % prime[j] == 0) {
                while (i % prime[j] == 0) i /= prime[j];
                if (i == 1) {
                    flag = false;
                    break;
                }
            }
        }
        if (i > 1) {
            if (i % 2 == 0) {
                flag = false;
            }
            else i--;
        }
        if (flag) {
            if (modPow(i, p - 1, p) == 1) cnt++;
        }
    }
    return cnt;
}
int main() {
    sieve();
    int p;
    cin >> p;
    cout << numPrimitiveRoots(p) << endl;
    return 0;
}
