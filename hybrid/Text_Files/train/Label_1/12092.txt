#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, m; cin >> n >> m;
    vector<int> l(m), c(n + 1, 0);
    for (int &x : l) cin >> x;
    sort(l.begin(), l.end());
    int p = 0;
    for (int i = 0; i < m; i++) {
        if (p + l[i] - 1 > n) {
            cout << -1 << endl;
            return 0;
        }
        int q = 0;
        while (q + l[i] <= n && c[q + l[i]]) { q++; }
        if (q + l[i] > n) {
            cout << -1 << endl;
            return 0;
        }
        cout << q + 1 << ' ';
        for (int j = q; j < q + l[i]; j++) {
            c[j] = i + 1;
        }
        p += l[i];
    }
    cout << endl;
    return 0;
}
