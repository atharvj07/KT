n, m, k = [int(i) for i in input().split()]
grid = [list(input()) for i in range(n)]
seen = [[False] * m for i in range(n)]
dx = [0, 0, -1, 1]
dy = [-1, 1, 0, 0]
ans = [0] * m
for x in range(m):
    spiders = 0
    for i in range(n):
        for j in range(m):
            if grid[i][j] in "LRUD" and not seen[i][j]:
                nx = i + dx[grid.index(grid[i][j])]
                ny = j + dy[grid.index(grid[i][j])]
                while 0 <= nx < n and 0 <= ny < m:
                    if grid[nx][ny] in "LRUD":
                        seen[nx][ny] = True
                        nx += dx[grid.index(grid[nx][ny])]
                        ny += dy[grid.index(grid[nx][ny])]
                    else:
                        spiders += 1
                        break
    ans[x] = spiders
print(*ans)
