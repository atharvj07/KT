import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt(), count = 1, max = 0;
        int[] col = new int[n + 1], pre = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            col[i] = sc.nextInt();
            if (i > 1 && col[i] == col[i - 1]) { count++; }
            else { count = 1; }
            max = Math.max(max, count);
            pre[i] = count;
        }
        for (int i = n; i >= 1; i--) {
            int l = i + 1;
            while (l <= n && k > 0) {
                if (col[l] == col[i]) {
                    k--;
                    l++;
                    max = Math.max(max, pre[i] + 1);
                } else break;
            }
        }
        System.out.println(max);
    }
}
