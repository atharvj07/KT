#include <cstdio>
#include <vector>
#include <algorithm>

typedef long long ll;
const int inf = 1LL << 31;
const ll infLL = 1LL << 62;

using namespace std;
int main() {
	int n; scanf("%d", &n);
	pair<pair<int,int>,pair<int,int>> R[5+5];
	int maxX = -inf, maxY = -inf, minX = inf, minY = inf;
	for(int i = 1; i <= n; i++){
		scanf("%d%d%d%d",&R[i].first.first,&R[i].first.second,&R[i].second.first,&R[i].second.second);
		minX = min(minX, R[i].first.first);
		minY = min(minY, R[i].first.second);
		maxX = max(maxX, R[i].second.first); 
		maxY = max(maxY, R[i].second.second);
	}

	int side = 0;
	for(int i = 1; i <= n; i++){
		if(R[i].first.first == minX){
			for(int j = 1; j <= n; j++){
				if(R[j].first.second == minY){
					if(R[j].second.first == maxX && R[j].second.second == maxY){
						if(maxY - minY == maxX - minX){
							side = maxX - minX;
						}
					}
				}
			}
		}
	}
	if(side != 0){
		puts("YES");
	}else{
		puts("NO");
	}
	return 0;
}
