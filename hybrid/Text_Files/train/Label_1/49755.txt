import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int num = sc.nextInt();
        int[] arr = new int[num];
        for (int i = 0; i < num; i++) {
            arr[i] = sc.nextInt();
        }
        int count = 0;
        for (int i = 0; i < num - 1; i++) {
            if (arr[i] > arr[i + 1]) {
                count++;
            }
        }
        System.out.println(count);
    }
}
