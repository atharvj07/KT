#include <bits/stdc++.h>
using namespace std;

int main() {
    long long r;
    cin >> r;
    vector<pair<long long, long long>> ans;
    for (long long x = 1; x * x <= r; x++) {
        long long y = (r - x * x - x - 1) / (2 * x);
        if (y * x + x * x + x + 1 == r && y >= 1) {
            ans.push_back({x, y});
        }
    }
    sort(ans.begin(), ans.end());
    if (ans.size() == 0)
        cout << "NO" << endl;
    else
        cout << ans[0].first << " " << ans[0].second << endl;
}
