l = int(input().strip())
n = input().strip()

# Find the middle, considering even and odd lengths
mid = l // 2

# Start from the middle and adjust in case of leading zeros
left = mid
right = mid if l % 2 == 0 else mid + 1

# Adjust left and right pointers to avoid leading zeros
while left > 0 and n[left - 1] == '0':
    left -= 1
while right < l and n[right] == '0':
    right += 1

# Compare which split gives the minimum sum and output it
if left == 0 or (right < l and int(n[:right]) + int(n[right:]) < int(n[:left]) + int(n[left:])):
    print(int(n[:right]) + int(n[right:]))
else:
    print(int(n[:left]) + int(n[left:]))
