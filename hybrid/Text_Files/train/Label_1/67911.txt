from collections import deque

n, m, p = map(int, input().split())
s = list(map(int, input().split()))
grid = [list(input()) for i in range(n)]

dx = [0, 0, 1, -1]
dy = [1, -1, 0, 0]

cnt = [0] * (p + 1)

def valid(x, y):
    return 0 <= x < n and 0 <= y < m and grid[x][y] == '.'

def bfs(x, y, player):
    q = deque([(x, y)])
    while q:
        a, b = q.popleft()
        cnt[player] += 1
        for i in range(4):
            nx = a + dx[i]
            ny = b + dy[i]
            if valid(nx, ny) and not (abs(nx - x) + abs(ny - y) > s[player - 1]):
                q.append((nx, ny))
                grid[nx][ny] = str(player)

for i in range(n):
    for j in range(m):
        if grid[i][j] == '.':
            for player in range(1, p + 1):
                if abs(i - j) <= s[player - 1]:
                    bfs(i, j, player)

for i in range(1, p + 1):
    print(cnt[i], end=' ')
