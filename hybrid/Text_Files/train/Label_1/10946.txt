#include<iostream>
#include<string>
using namespace std;
int main()
{
    int n, m, i, j, f;
    string s, t;
    cin >> n >> m;
    cin >> s >> t;
    if (s[0] == '*')
        s.erase(0, 1);
    i = 0, j = 0, f = 1;
    while (i < n && j < m)
    {
        if (s[i] == '*')
        {
            i++;
            f = 0;
            int k = i;
            while (i < n && s[i] == '*')
            {
                i++;
            }
            int l = i;
            while (j < m && k < l)
            {
                if (s[k] != t[j])
                {
                    f = 1;
                    break;
                }
                k++;
                j++;
            }
        }
        else
        {
            if (s[i] == t[j])
            {
                i++;
                j++;
            }
            else
            {
                f = 1;
                break;
            }
        }
    }
    if (f == 1 || (i < n && j == m))
        cout << "NO\n";
    else
        cout << "YES\n";
    return 0;
}
