import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        ArrayList<ArrayList<Integer>> g = new ArrayList<>();
        for (int i = 0; i <= n; i++) g.add(new ArrayList<>());
        int[] deg = new int[n + 1], depth = new int[n + 1], par = new int[n + 1], vis = new int[n + 1];
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt(), v = sc.nextInt();
            g.get(u).add(v);
            g.get(v).add(u);
            deg[u]++;
            deg[v]++;
        }
        Queue<Integer> q = new LinkedList<>();
        for (int i = 1; i <= n; i++) {
            if (deg[i] == 1) {
                q.add(i);
                depth[i] = 1;
            }
        }
        int ans = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            ans = Math.max(ans, depth[u]);
            for (int v : g.get(u)) {
                if (vis[v] == 0) {
                    vis[v] = 1;
                    depth[v] = depth[u] + 1;
                    par[v] = u;
                    deg[u]--;
                    deg[v]--;
                    if (deg[u] == 1) q.add(u);
                    if (deg[v] == 1) q.add(v);
                }
            }
        }
        for (int i = 2; i <= n; i++) {
            depth[i] = depth[par[i]] + 1;
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(ans - depth[i] + " ");
        }
        System.out.println();
    }
}
