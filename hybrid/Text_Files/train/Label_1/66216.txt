import sys

input = sys.stdin.readline

n, x = map(int, input().split())
g = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

vis = [0] * (n + 1)
dep = [0] * (n + 1)

def dfs(i, p):
    vis[i] = 1
    dep[i] = dep[p] + 1
    for j in g[i]:
        if vis[j] == 0:
            dfs(j, i)

dfs(1, 0)
z = 0
for i in range(2, n + 1):
    if dep[i] > dep[z]:
        z = i

dfs(z, 0)
ans = 0
for i in range(2, n + 1):
    ans = max(ans, dep[i])

print(ans)
