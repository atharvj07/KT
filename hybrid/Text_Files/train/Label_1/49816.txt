import java.util.*;

public class Main {
   public static void main(String[] args) {
      Scanner input = new Scanner(System.in);
      int n = input.nextInt();
      int k = input.nextInt();
      input.nextLine();
      Map<Integer, Integer> pirates = new HashMap<>();
      for (int i = 0; i < k; i++) {
         String[] line = input.nextLine().split(" ");
         int pirate = Integer.parseInt(line[0]);
         int coins = Integer.parseInt(line[1]);
         pirates.put(pirate, coins);
      }
      if (pirates.size() <= 1) {
         System.out.println(-1);
         return;
      }
      boolean[] visited = new boolean[n+1];
      for (int i = 1; i <= n; i++) {
         if (!pirates.containsKey(i) || visited[i]) {
            continue;
         }
         int coins = pirates.get(i);
         visited[i] = true;
         coins -= 2;
         pirates.put(i, coins);
         if (i-1 != 0) {
            pirates.put(i-1, pirates.getOrDefault(i-1, 0) + 1);
         }
         pirates.put(i+1, pirates.getOrDefault(i+1, 0) + 1);
      }
      boolean good = false;
      for (int pirate : pirates.keySet()) {
         if (pirates.get(pirate) > 0) {
            good = true;
            break;
         }
      }
      System.out.println(good ? 1 : -1);
   }
}
