import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, k = map(int, input().split())
    s = input().strip()
    if k == n:
        print(0)
    else:
        ans = k
        for i in range(n-k+1):
            cnt = 0
            t = s[i:i+k]
            for j in range(k):
                if t[j] != 'R' and j%2==0:
                    cnt += 1
                if t[j] != 'G' and j%2==1:
                    cnt += 1
            ans = min(ans, cnt)
        print(ans)
