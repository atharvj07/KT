#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0;
  vector<char> t(n);
  for (int i = 0; i < n; i++) {
    if (i && t[i - 1] == s[i]) {
      ans++;
      t[i] = (s[i] == 'R' ? 'G' : (s[i] == 'G' ? 'B' : 'R'));
    } else
      t[i] = s[i];
  }
  cout << ans << endl;
  for (char c : t) cout << c;
  cout << endl;
  return 0;
}
