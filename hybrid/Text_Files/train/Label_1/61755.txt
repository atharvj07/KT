n, k1, k2 = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
a.sort()
b.sort()
ans = 0
for i in range(min(k1, k2)):
    if a[i] > b[i]:
        ans += a[i] - b[i]
        a[i] -= 1
    else:
        ans += b[i] - a[i]
        b[i] -= 1
for i in range(k1 - min(k1, k2)):
    ans += a[i] - 1
    a[i] -= 1
for i in range(k2 - min(k1, k2)):
    ans += b[i] - 1
    b[i] -= 1
print(ans)
