#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1010, MAXV = 1e6 + 10;
bool isPrime[MAXV];
int a[MAXN], n, m;
bool vis[MAXN], isSimple[MAXN][MAXN];
int ans[MAXN], cnt;

void sieve() {
    for (int p = 2; p < MAXV; ++p) isPrime[p] = true;
    for (int p = 2; p * p < MAXV; ++p) {
        if (!isPrime[p]) continue;
        for (int i = p * p; i < MAXV; i += p) isPrime[i] = false;
    }
}

bool check(int x, int y) {
    if (isSimple[x][y] != -1) return isSimple[x][y];
    if (isPrime[x + y])
        return isSimple[x][y] = true;
    else
        return isSimple[x][y] = false;
}

void dfs(int x) {
    if (x == n + 1) {
        if (cnt > m) {
            m = cnt;
            for (int i = 1; i <= n; ++i) if (vis[i]) ans[cnt--] = a[i];
        }
        return;
    }
    vis[x] = true;
    cnt++;
    dfs(x + 1);
    vis[x] = false;
    cnt--;
    for (int i = x + 1; i <= n; ++i)
        if (vis[i]) {
            if (check(a[x], a[i])) {
                vis[x] = true;
                cnt++;
                dfs(x + 1);
                vis[x] = false;
                cnt--;
            }
        }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);

    sieve();

    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> a[i];

    memset(isSimple, -1, sizeof(isSimple));

    dfs(1);

    cout << m << endl;
    for (int i = 1; i <= m; ++i) cout << ans[i] << " ";

    return 0;
}
