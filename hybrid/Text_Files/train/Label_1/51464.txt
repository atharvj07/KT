import java.util.*;

public class ForbiddenPairs {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.next();
        int k = scanner.nextInt();
        scanner.nextLine();
        Set<Character> forbiddenPairs = new HashSet<>();
        for (int i = 0; i < k; i++) {
            String pair = scanner.nextLine();
            forbiddenPairs.add(pair.charAt(0));
            forbiddenPairs.add(pair.charAt(1));
        }

        Stack<Character> stack = new Stack<>();
        for (char c : s.toCharArray()) {
            if (stack.isEmpty() || !forbiddenPairs.contains(stack.peek() + "" + c)) {
                stack.push(c);
            } else {
                stack.pop();
            }
        }

        System.out.println(s.length() - stack.size());
    }
}
