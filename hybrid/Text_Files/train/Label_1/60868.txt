import sys
input = sys.stdin.readline

n,s,k = map(int, input().split())
r,c = list(map(int, input().split())),list(input())
ans = s-1
candies = [0]*n
candies[s-1] = r[s-1]
for i in range(s, n):
    if r[i] > candies[i-1] and c[i] != c[i-1]:
        ans += 1
        candies[i] = r[i]
    else: break
for i in range(s-2, -1, -1):
    if r[i] > candies[i+1] and c[i] != c[i+1]:
        ans += 1
        candies[i] = r[i]
if sum(candies) < k: ans = -1
print(ans)
