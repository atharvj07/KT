MOD = 10**9 + 7

def count_subsequences_abc(s):
    a_count, b_count, c_count = 0, 0, 0
    q_count = s.count('?')
    total = 0
    
    for char in s:
        if char == 'a':
            a_count += pow(3, q_count, MOD)
        elif char == 'b':
            b_count += a_count
        elif char == 'c':
            c_count += b_count
        else:  # char == '?'
            c_count = 3*c_count + b_count
            b_count = 3*b_count + a_count
            a_count = 3*a_count + pow(3, q_count-1, MOD)
    
    return c_count % MOD
