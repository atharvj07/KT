#include<bits/stdc++.h>
using namespace std;
int n, a, b;
int main() {
    cin >> n >> a >> b;
    if (a > b) swap(a, b);
    if (((n - 1) & n) == 0) cout << "Final!\n";
    else {
        int x = 1;
        while (x * 2 <= n) x *= 2;
        if ((x <= a && a < 2 * x) && (x <= b && b < 2 * x)) cout << 1 << '\n';
        else {
            x /= 2;
            while (x * 2 <= n) x *= 2;
            cout << 2 << '\n';
        }
    }
}
