import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int v = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }

        int[] min = new int[n];
        Arrays.fill(min, Integer.MAX_VALUE);
        min[0] = 0;
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (min[j] != Integer.MAX_VALUE && x[i] - x[j] <= v) {
                    min[i] = Math.min(min[i], min[j] + x[i] - x[j]);
                }
            }
        }

        for (int i = 0; i < n; i++) {
            if (min[i] != Integer.MAX_VALUE) {
                System.out.println("Possible");
            } else {
                System.out.println("Impossible");
            }
        }
    }
}
