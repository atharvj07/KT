import java.util.Scanner;
public class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] b = new int[n][n];
        for(int i=0; i<n; i++){
            for(int j=0; j<n; j++){
                b[i][j] = sc.nextInt();
            }
        }
        int[] a = new int[n];
        a[0] = 0;
        for(int i=1; i<n; i++){
            a[i] = b[0][i];
        }
        for(int i=0; i<n; i++){
            for(int j=i+1; j<n; j++){
                a[i] = a[i] & a[j];
            }
        }
        for(int i=0; i<n; i++){
            System.out.print(a[i] + " ");
        }
        System.out.println();
    }
}
