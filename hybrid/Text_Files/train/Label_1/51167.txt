import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextInt();
        long[] a = new long[200001];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        long ans = 0;
        while (n > 1) {
            if (n % 2 == 1) {
                for (int i = 0; i < n - 1; i += 2) {
                    a[i] += a[i + 1];
                    a[i] %= 1000000007;
                }
            } else {
                for (int i = 0; i < n - 1; i += 2) {
                    a[i] -= a[i + 1];
                    a[i] = (a[i] + 1000000007) % 1000000007;
                }
            }
            n = (n + 1) / 2;
        }
        System.out.println(a[0]);
    }
}
