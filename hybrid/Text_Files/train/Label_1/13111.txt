#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    vector<int> a(n);
    for(int i=0;i<n;i++)cin>>a[i];
    int ans=-1;
    vector<pair<int,int>> ord;
    for(int i=0;i<n;i++){
        for(int j=0;j<i;j++)if(a[i]<a[j])ans=0;
        ord.emplace_back(a[i],i);
    }
    sort(ord.begin(),ord.end());
    int i=0;
    for(auto x:ord)if(x.second!=i++){
        ans=-1;
        break;
    }
    cout<<ans<<'\n';
}
