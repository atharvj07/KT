import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int p = sc.nextInt();
        int[] dp = new int[p + 1];
        dp[0] = 0;
        for (int i = 1; i <= p; i++) {
            dp[i] = dp[i - 1];
            if (i >= 500) dp[i] = Math.max(dp[i], dp[i - 500] + 500);
            if (i >= 1000) dp[i] = Math.max(dp[i], dp[i - 1000] + 1000);
            if (i >= 5000) dp[i] = Math.max(dp[i], dp[i - 5000] + 5000);
            if (i >= 10000) dp[i] = Math.max(dp[i], dp[i - 10000] + 10000);
        }
        System.out.println(dp[p]);
    }
}
