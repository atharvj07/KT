#include <iostream>

using namespace std;

int main() {
    int w, h, u1, d1, u2, d2;
    cin >> w >> h;
    cin >> u1 >> d1;
    cin >> u2 >> d2;
    while (h > 0) {
        w += h;
        if (h == d1) w -= u1;
        if (h == d2) w -= u2;
        w = max(0, w);
        h--;
    }
    cout << w << endl;
    return 0;
}
