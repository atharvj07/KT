#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const int MAXN = 1e3 + 6;
int n, h, a[MAXN];
int ans, lst;

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> h;
        ans = 0;
        lst = -1;
        for (int i = 1; i <= n; i++) cin >> a[i];
        for (int i = n; i >= 1; i--) {
            if (a[i] != lst) {
                lst = a[i];
                if (h > 0) {
                    h -= a[i];
                    ans++;
                }
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
