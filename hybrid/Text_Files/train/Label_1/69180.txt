import copy
n=int(input())
p=list(map(int,input().split()))
A=[]
for i in range(0,n):
    A.append(list(input()))
def find(l,r):
    if l==r: return l
    mid=(l+r)//2
    a=find(l,mid)
    if p[mid]<p[r]:
        return mid
    else:
        return find(mid+1,r)
ans=copy.copy(p)
for i in range(0,n):
    for j in range(i+1,n):
        if A[i][j]=='1' and p[i]>p[j]:
            ans[i]=p[j]
            ans[j]=p[i]
            idx=find(i+1,n-1)
            for k in range(idx+1,n):
                if ans[k]<ans[idx]:
                    idx=k
            tmp=ans[idx]
            ans[idx]=ans[i]
            ans[i]=tmp
print(*ans)
