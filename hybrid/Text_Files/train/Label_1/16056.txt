#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back

struct DSU {
    vector<int> p, sz;
    DSU(int n) {
        p.resize(n);
        sz.resize(n, 1);
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) { return p[x] = (p[x] == x ? x : find(p[x])); }
    void join(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return;
        if (sz[px] < sz[py]) swap(px, py);
        p[py] = px;
        sz[px] += sz[py];
    }
};
ll n, x;
vector<ll> a;
const int MAXN = 2e5+5;
bitset<MAXN> vis;
int32_t main() {
    cin >> n;
    a.resize(n);
    DSU dsu(MAXN);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        vis[a[i]]=1;
    }
    sort(a.begin(), a.end());
    for (int i = 0; i < n; i++) {
        if (vis[a[i] - 1] && vis[a[i] + 1]) {
            dsu.join(a[i] - 1, a[i] + 1);
        }
    }
    vector<ll> ans;
    for (int i = 0; i < n; i++) {
        if (dsu.find(a[i]) == dsu.find(a[i] - 1)) { ans.pb(a[i]); a.erase(a.begin() + i), i--, n--; }
    }
    cout << ans.size() << '\n';
    for (auto x : ans) cout << x << ' ';
    return 0;
}
