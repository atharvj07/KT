#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
const LL inf = 1LL << 60;

int main() {
  LL n, a[200005], q, type, x, ans;
  vector<LL> s;
  cin >> n;
  s.assign(n + 2, 0);
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
  }
  cin >> q;
  while (q--) {
    cin >> type >> x;
    ans = inf;
    if (type == 1) {
      cin >> x;
      for (LL i = x; i <= n; ++i) {
        s[i] += x;
        if (s[i] > a[i]) {
          s[i] -= a[i];
          s[i + 1] += a[i];
        }
      }
    }
    if (type == 2) ans = s[x];
    cout << ans << endl;
  }
  return 0;
}
