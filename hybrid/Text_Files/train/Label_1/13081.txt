#include <iostream>
#include <vector>
using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;

int main() {
  ll n;
  cin >> n;
  vector<ll> a(n + 2);
  for (ll i = 1; i <= n; ++i) cin >> a[i];

  vector<ll> dp(n + 2);
  for (ll i = 2; i <= n; ++i) {
    dp[i] = max(dp[i - 2] + a[i] + a[i - 1], dp[i - 1]);
  }
  cout << dp[n];
}
