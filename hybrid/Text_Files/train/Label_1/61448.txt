t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    positive = sorted([x for x in a if x > 0], reverse=True)
    negative = sorted([x for x in a if x < 0])
    if sum(positive) + sum(negative) == 0:
        print("NO")
    else:
        if sum(positive) >= abs(sum(negative)):
            print("YES")
            print(*positive, *negative, *([0]*a.count(0)))
        else:
            print("YES")
            print(*negative, *positive, *([0]*a.count(0)))
