#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n, k;
    cin >> n >> k;
    long long l = 1, r = n + 1;
    while (l <= r) {
        long long mid = (l + r) / 2;
        long long num = 0;
        for (long long i = mid; i <= n; i+=mid) {
            num += n / i;
        }
        if (num >= k) {
            r = mid - 1;
        } else {
            l = mid + 1;
        }
    }
    cout << l << endl;
}
