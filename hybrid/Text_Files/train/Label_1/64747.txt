import sys
input = sys.stdin.readline

def solve():
    mx = -10**9
    st = []
    ans = 0
    for x in map(int, input().split()):
        if x >= mx:
            st.append(x)
            mx = x
            ans = max(ans, len(st))
        else:
            while st and st[-1] > x:
                st.pop()
            if st:
                st.append(x)
    return ans

for _ in range(int(input())):
    print(solve())
