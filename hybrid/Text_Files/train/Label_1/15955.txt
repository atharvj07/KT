#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N = 1e5 + 9;
ll a[N], c, n, ans = 1e18;
int main() {
    cin >> n >> c;
    for (int i = 0; i < n; i++)cin >> a[i];
    sort(a, a + n);
    for (int i = 0; i + c - 1 < n; i++)ans = min(ans, accumulate(a + i, a + i + c - 1, 0LL));
    cout << ans << endl;
    return 0;
}
