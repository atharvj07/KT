#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int x, y;
    cin >> x >> y;
    pair<int, int> pos = {0, 0};
    for (int i = 0; i < n; i++) {
        if (s[i] == 'U') pos.second++;
        if (s[i] == 'D') pos.second--;
        if (s[i] == 'R') pos.first++;
        if (s[i] == 'L') pos.first--;
    }
    int dx = abs(x - pos.first);
    int dy = abs(y - pos.second);
    if (dx > n || dy > n) {
        cout << -1 << endl;
        return 0;
    }
    int sum = dx + dy;
    cout << sum << endl;
}
