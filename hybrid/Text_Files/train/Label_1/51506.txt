import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int f = input.nextInt();
        int[] fib = new int[1013];
        fib[0] = 0;
        fib[1] = 1;
        for (int i = 2; i < 1013; i++)
            fib[i] = (fib[i - 1] + fib[i - 2]) % 1013;
        for (int i = 0; i < 1013; i++)
            if (fib[i] == f) {
                System.out.println(i);
                return;
            }
        System.out.println(-1);
    }
}
