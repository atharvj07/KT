n = int(input())
grid = []
for _ in range(n):
    grid.append(input())
ans = 0
for i in range(n):
    for j in range(n):
        if grid[i][j] == 'C':
            for k in range(n):
                if i != k and grid[k][j] == 'C':
                    ans += 1
            for k in range(n):
                if j != k and grid[i][k] == 'C':
                    ans += 1
print(ans)
