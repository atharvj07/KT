#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, q;
  cin >> n >> q;

  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  vector<long long> k(q);
  for (int i = 0; i < q; i++) {
    cin >> k[i];
  }

  int cur = 0;
  for (int i = 0; i < q; i++) {
    while (cur < n && k[i] >= a[cur]) {
      k[i] -= a[cur];
      cur++;
    }
    if (cur < n) {
      a[cur] -= k[i];
    }
    cout << n - cur << "\n";
  }

  return 0;
}
