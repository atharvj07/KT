#include <iostream>
using namespace std;
int main() {
  int a[101];
  int n, ans = 0;
  cin >> n;
  int cnt1 = 0, cnt2 = 0;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    if (a[i] == 0 || a[i] == 2) ++cnt1;
    if (a[i] == 1 || a[i] == 3) ++cnt2;
  }
  int mx = max(cnt1, cnt2);
  ans = n - mx;
  cout << ans << endl;
  return 0;
}
