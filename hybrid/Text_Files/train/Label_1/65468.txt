n, b = map(int, input().split())
queries = [list(map(int, input().split())) for _ in range(n)]

# Initialize server state
current_time = 0
queue = []

# Process each query
for t, d in queries:
    # Remove queries from queue if they have been processed
    while queue and queue[0] <= t:
        queue.pop(0)
    
    if len(queue) < b:
        if not queue and t >= current_time:
            print(t + d, end=" ")
            current_time = t + d
        else:
            current_time = max(current_time, queue[-1] if queue else 0) + d
            print(current_time, end=" ")
            queue.append(current_time)
    else:
        print(-1, end=" ")
