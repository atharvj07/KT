n=int(input())
a=list(map(int,input().split()))
g=a[0]
for x in a:g=max(g,x)
if g%2==0:g=g//2
vis=[False]*n
ans=[]
for i in range(n):
    x=a[i]
    if vis[i]==True:continue
    vis[i]=True
    ans.append(i+1)
    for j in range(i+1,n):
        if vis[j]==True:continue
        y=a[j]
        while x>0:
            y,x=x,y%x
        if y==1:
            vis[j]=True
            ans.append(j+1)
    break
if len(ans)<n:print("NO")
else:
    print("YES")
    print(*ans)
