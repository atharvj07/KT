#include <iostream>
#include <algorithm>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }
  std::sort(a.begin(), a.end());
  int sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i];
  }
  int ans = 0;
  int pos = -1;
  for (int i = 0; i < n; i++) {
    if (sum - a[i] <= a[i]) {
      ans = 1;
      pos = i;
      break;
    }
    sum -= a[i];
  }
  std::cout << ans << "\n";
  if (ans) {
    std::cout << pos + 1 << "\n";
  }
  return 0;
}
