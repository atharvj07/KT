n=int(input())
cubes=[]
for i in range(n):
    cubes.append(list(map(int,input().split())))
vis=[0]*10

for i in range(n):
    for j in range(6):
        vis[cubes[i][j]]=1
flag=1
for i in range(1,10):
    flag=1
    for j in str(i):
        if vis[int(j)]==0:
            flag=0
            break
    if flag==1:
        print(i)
        break
if flag==0:
    print(0)
