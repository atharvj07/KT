n = int(input().strip())
s = input().strip()
a = list(map(int, input().strip().split()))

cost = [0] * 4  # Cost of removing h, a, r, d to avoid forming 'hard'

for i in range(n):
    if s[i] == 'h':
        cost[0] += a[i]
    elif s[i] == 'a':
        cost[1] = min(cost[0], cost[1] + a[i])
    elif s[i] == 'r':
        cost[2] = min(cost[1], cost[2] + a[i])
    elif s[i] == 'd':
        cost[3] = min(cost[2], cost[3] + a[i])
        
print(min(cost))
