import collections
def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        c = collections.Counter(map(int, input().split()))
        odd = []
        for num, cnt in c.items():
            if cnt&1:
                odd.append(num)
        #sort in order of smallest count of odd elements
        odd.sort(key=lambda x: c[x]&1)
        exist = [False] * 101
        a = []
        b = []
        for num in odd:
            if not exist[num]:
                if c[num]&1:
                    a.append(num)
                    exist[num] = True
                else:
                    b.append(num)
                    exist[num] = True
        for num in range(101):
            if not exist[num]:
                if len(a) > len(b):
                    b.append(num)
                    exist[num] = True
                else:
                    a.append(num)
                    exist[num] = True
        print(sum(a[0] for a in [a, b]))

solve()
