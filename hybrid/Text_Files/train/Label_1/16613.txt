#include <bits/stdc++.h>
using namespace std;

const int N = 150010;
int p[N],t[N];
int main() {
	int n,k,q; cin >> n >> k >> q;
	while(q--){
		int x; cin >> x;
		if(x==1){
			int id; cin >> id;
			p[id] = 1;
		}
		else{
			int id; cin >> id;
			priority_queue<int> pq;
			for(int i=1; i<=n; i++)
				if(p[i]) pq.push(t[i]);
			int cnt = 0;
			while(!pq.empty() && cnt<k){
				if(pq.top()==t[id]) cnt++;
				pq.pop();
			}
			cout << (cnt==0 || cnt<k ? "NO" : "YES") << endl;
		}
	}
}
