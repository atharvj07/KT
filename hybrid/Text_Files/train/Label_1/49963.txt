import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long d = sc.nextLong();
        long ans = (n - 1) * (n - 2) / 2;
        if (d == 2) ++ans;
        System.out.println(ans);
    }
}
