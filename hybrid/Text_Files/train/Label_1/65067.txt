n = int(input())
a = list(map(int, input().split()))
s = [0] * n
f = [0] * n
for i in range(n):
    if i > 0:
        s[i] = s[i - 1]
    s[i] += a[i]
for i in range(n - 1, -1, -1):
    if i < n - 1:
        f[i] = f[i + 1]
    f[i] += a[i]
ans = []
for i in range(1, n - 1):
    if a[i] == a[i - 1] and a[i] == a[i + 1]:
        ans.append(i)
ans.sort()
print(s[n - 1] - min(f[0], f[n - 1]), len(ans))
print(*ans)
