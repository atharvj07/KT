import java.io.BufferedInputStream;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    static final int MOD = 998244353;
    static long[] dp;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(new BufferedInputStream(System.in)));
        String[] s = br.readLine().split(" ");
        int n = Integer.parseInt(s[0]);
        int k = Integer.parseInt(s[1]);
        int[][] ranges = new int[k][2];
        for (int i = 0; i < k; i++) {
            s = br.readLine().split(" ");
            ranges[i][0] = Integer.parseInt(s[0]);
            ranges[i][1] = Integer.parseInt(s[1]);
        }
        Arrays.sort(ranges, (a, b) -> a[1] - b[1]);

        // dp[i] = number of ways to reach cell i
        dp = new long[n + 1];
        dp[1] = 1;
        for (int i = 2; i <= n; i++) {
            for (int[] range : ranges) {
                int l = range[0], r = range[1];
                if (l <= i - 1 && i - 1 <= r) {
                    // can reach cell i from cell i - 1
                    dp[i] += dp[i - 1];
                    dp[i] %= MOD;
                }
            }
        }

        System.out.println(dp[n]);
    }
}
