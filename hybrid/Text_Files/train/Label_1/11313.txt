#include <iostream>
#include <vector>
int main() {
  int n, x, k = 0, mx = 0;
  std::cin >> n;
  std::vector<std::vector<int>> v(n, std::vector<int>(6));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 6; j++) {
      std::cin >> v[i][j];
    }
  }
  for (int i = 0; i < 6; i++) {
    for (int j = 0; j < n; j++) {
      for (int l = 0; l < n; l++) {
        if (l == j) continue;
        x = 0;
        for (int m = 0; m < 6; m++) {
          if (v[j][m] != v[l][m]) x++;
        }
        mx = std::max(mx, x);
      }
    }
    k = std::max(k, mx);
  }
  std::cout << k << std::endl;
  return 0;
}
