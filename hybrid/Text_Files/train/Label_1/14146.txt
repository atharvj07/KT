#include <bits/stdc++.h>
using namespace std;
long long a[100001], b[100001];
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> a[i] >> b[i];
  }
  sort(b + 1, b + n + 1);
  long long sum = 0;
  int cnt = 0;
  for (int i = n; i >= 1; i--) {
    if (sum + b[i] > m)
      cnt++;
    else
      sum += b[i];
  }
  if (sum <= m)
    cout << cnt << "\n";
  else
    cout << "-1\n";
}
