import sys
n, k = map(int, input().split())
d = [*map(int, input().split())]
if n * (n - 1) // 2 < 2 * k * n:
    print(-1); sys.exit()
t = []
for i in range(n):
    for j in range(i + 1, n):
        if d[i] + d[j] == d[n - 1] and d[i] + 1 == d[j]:
            t.append((i, j))
print(len(t))
for x in t: print(*x)
