import java.util.Scanner;

public class Main {
  static int solve(int n, String s) {
    int r = 0;
    for (int i = 0; i < n; i++) {
      if (s.charAt(i) != s.charAt((i + 1) % n)) {
        r++;
      }
    }
    if (r <= 1) r = n;
    return r;
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    int[] ans = new int[t];
    for (int i = 0; i < t; i++) {
      int n = sc.nextInt();
      String s = sc.next();
      ans[i] = solve(n, s);
    }
    for (int x : ans) {
      System.out.println(x);
    }
  }
}
