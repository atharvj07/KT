#include <iostream>
#include <vector>
using namespace std;
int main() {
  int a, b, c, d;
  cin >> a >> b >> c >> d;
  size_t sum = a + b + c + d;
  vector<int> ans;
  if (sum % 2 == 0) {
    if (a <= (sum + 4) / 6 - 1) {
      cout << "YES\n";
      for (int i = 0; i < (sum + 4) / 6 - 1 - a; i++) ans.push_back(0);
      for (int i = 0; i < c; i++) ans.push_back(2);
      for (int i = 0; i < d; i++) ans.push_back(3);
      for (int i = 0; i < a; i++) ans.push_back(0);
      for (int i = 0; i < b; i++) ans.push_back(1);
      for (int num : ans) cout << num << ' ';
      return 0;
    }
  } else if (sum % 2 == 1) {
    if (d <= (sum + 2) / 6) {
      cout << "YES\n";
      for (int i = 0; i < c; i++) ans.push_back(2);
      for (int i = 0; i < d; i++) ans.push_back(3);
      for (int i = 0; i < a; i++) ans.push_back(0);
      for (int i = 0; i < b; i++) ans.push_back(1);
      for (int i = 0; i < (sum + 2) / 6 - d; i++) ans.push_back(2);
      for (int num : ans) cout << num << ' ';
      return 0;
    }
  }
  cout << "NO";
  return 0;
}
