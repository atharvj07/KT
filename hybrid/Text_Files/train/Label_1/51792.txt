import java.util.LinkedList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        LinkedList<Character> sequence = new LinkedList<>();

        Scanner input = new Scanner(System.in);
        for (char ch : input.nextLine().toCharArray()) {
            if (ch == '(') {
                sequence.addFirst(ch);
            } else {
                if (!sequence.isEmpty() && sequence.getFirst() == '(') {
                    sequence.removeFirst();
                } else {
                    sequence.addLast(ch);
                }
            }
        }

        if (sequence.isEmpty()) {
            System.out.println();
            return;
        }

        StringBuilder result = new StringBuilder();
        while (sequence.size() > 2) {
            if (sequence.getFirst() == '(') {
                result.append('(').append(sequence.removeLast()).append(')');
                sequence.removeFirst();
            } else {
                result.append(')').append('(');
                sequence.removeLast();
                sequence.removeFirst();
            }
        }

        if (sequence.size() == 1) {
            result.append('(').append(sequence.removeFirst()).append(')');
        } else {
            result.append(sequence.getFirst()).append(sequence.removeLast());
        }

        System.out.println(result);
    }
}
