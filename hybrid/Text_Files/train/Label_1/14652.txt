#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2005;

int a[MAXN], b[MAXN];
int dp[MAXN][MAXN];
int n, k;

int go(int pos, int sum) {
  if (pos == k) return sum == 0;
  int &ret = dp[pos][sum];
  if (ret != -1) return ret;
  ret = 0;
  ret |= go(pos + 1, sum + a[pos]);
  ret |= go(pos + 1, sum);
  return ret;
}

int main() {
  cin >> k >> n;
  for (int i = 0; i < k; i++) cin >> a[i];
  sort(a, a + k);
  for (int i = 0; i < n; i++) cin >> b[i];
  sort(b, b + n);
  memset(dp, -1, sizeof(dp));
  if (go(0, 0) == 0) {
    cout << "0\n";
    return 0;
  }
  int ans = 0;
  int j = 0;
  for (int i = 0; i < k; i++) {
    while (j < n && b[j] < a[i]) {
      j++;
    }
    if (j < n && b[j] == a[i]) {
      ans++;
      j++;
    }
  }
  cout << ans << "\n";
  return 0;
}
