#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int MAXN = 1e6 + 5;

vector<int> p, pos[MAXN];
int n, m;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    p.resize(n + 1);
    for (int i = 1; i <= n; ++i) {
        int x;
        cin >> x;
        p[x] = i;
        pos[i].push_back(x);
    }
    while (m--) {
        int a, b;
        cin >> a >> b;
        pos[p[a]].push_back(b);
        pos[p[b]].push_back(a);
        swap(p[a], p[b]);
    }
    for (int i = 1; i <= n; ++i) {
        sort(pos[i].begin(), pos[i].end());
    }
    vector<int> ans(n + 1);
    for (int i = 1; i <= n; ++i) {
        int cur = i;
        for (int j = 0; j < pos[i].size(); ++j) {
            if (pos[i][j] > cur) {
                ans[cur] = p[pos[i][j]];
                cur = pos[i][j];
            }
        }
        ans[cur] = p[i];
    }
    for (int i = 1; i <= n; ++i) {
        cout << ans[i] << ' ';
    }
    return 0;
}
