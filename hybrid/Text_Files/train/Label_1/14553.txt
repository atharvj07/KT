#include <iostream>
#include <vector>

int main() {
  int n, k;
  std::cin >> n >> k;
  if (k > n - 2) {
    std::cout << -1 << "\n";
    return 0;
  }
  std::vector<int> ans(n);
  for (int i = 1; i <= k; i++) {
    if (i == k) {
      for (int j = 1; j <= n; j++) {
        if (ans[j - 1] == 0) {
          ans[j - 1] = i;
        }
      }
    } else {
      int len = (n - i * 2) / k;
      for (int j = 1; j <= len; j++) {
        ans[i * 2 - 1 + j - 1] = i;
        ans[n - j] = i;
      }
    }
  }
  for (auto &x : ans) {
    std::cout << x << " ";
  }
  std::cout << "\n";
  return 0;
}
