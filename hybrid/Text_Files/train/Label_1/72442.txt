for _ in range(int(input())):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    ans = 1
    for i in range(1, n):
        if k <= 0:
            break
        if a[i] <= a[i-1]:
            ans += 1
            k -= 1
    if a[ans - 1] == a[ans]:
        ans = -1
    print(ans)
