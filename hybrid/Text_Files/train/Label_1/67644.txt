import heapq

n = int(input())
log = []
for _ in range(n):
    operation, *args = input().split()
    log.append((operation, tuple(map(int, args))))

m = n
to_insert = []
heap = []
for operation, args in log:
    if operation == 'insert':
        heapq.heappush(heap, args[0])
        to_insert.append((operation, args))
    elif operation == 'getMin':
        if not heap or args[0] != heap[0]:
            heapq.heappush(to_insert, (operation, args))
            heapq.heappush(heap, args[0])
            m += 1
        else:
            to_insert.append((operation, args))
    elif operation == 'removeMin':
        while heap and heap[0] != args[0]:
            heapq.heappush(to_insert, ('removeMin', (heap[0],)))
            heapq.heappop(heap)
            m += 1
        heapq.heappop(heap)
        to_insert.append((operation, args))

print(m)
for operation, args in to_insert:
    print(*[operation] + list(args))
