import java.util.*;
public class Main {
    public static long solve(int[] a, int n, int k) {
        long[] dp = new long[n + 1];
        Arrays.fill(dp, Long.MAX_VALUE);
        dp[0] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < i; j++) {
                if (j + 2 * (i - j) <= n) {
                    dp[i] = Math.min(dp[i], dp[j] + (a[j]));
                }
            }
        }
        return dp[k];
    }
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = cin.nextInt();
        }
        for (int k = 1; k < n; k++) {
            System.out.println(solve(a, n, k));
        }
    }
}
