import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.LinkedList;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int s = Integer.parseInt(line[1]);
        line = reader.readLine().split(" ");
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(line[i]);
        }

        List<List<Integer>> cycles = new LinkedList<>();
        int[] next = new int[n];
        for (int i = 0; i < n; i++) {
            next[i] = i + 1;
        }
        int[] prev = new int[n];
        for (int i = 0; i < n; i++) {
            prev[i] = i - 1;
        }
        prev[0] = n - 1;
        next[n - 1] = 0;

        int[] sorted = a.clone();
        Arrays.sort(sorted);
        boolean[] visited = new boolean[n];
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                int j = i;
                List<Integer> cycle = new LinkedList<>();
                while (!visited[j]) {
                    visited[j] = true;
                    cycle.add(j);
                    int t = next[j];
                    next[j] = next[t];
                    prev[next[t]] = prev[j];
                    j = next[t];
                }
                cycles.add(cycle);
                count++;
            }
        }

        if (count == 1) {
            System.out.println("0");
            return;
        }

        int sum = 0;
        for (List<Integer> cycle : cycles) {
            sum += cycle.size();
        }

        if (sum > s) {
            System.out.println("-1");
            return;
        }

        Collections.sort(cycles, (c1, c2) -> c1.size() - c2.size());
        List<List<Integer>> newCycles = new LinkedList<>();
        for (List<Integer> cycle : cycles) {
            if (cycle.size() == 1) {
                continue;
            }
            if (cycle.size() == 2) {
                newCycles.add(cycle);
                continue;
            }
            int start = cycle.get(0);
            int end = cycle.get(cycle.size() - 1);
            List<Integer> newCycle = new LinkedList<>();
            newCycle.add(start);
            newCycle.add(next[start]);
            newCycles.add(newCycle);
            int t = next[next[start]];
            prev[t] = end;
            next[end] = t;
        }
        cycles = newCycles;

        System.out.println(cycles.size());
        for (List<Integer> cycle : cycles) {
            System.out.println(cycle.size());
            for (int i : cycle) {
                System.out.print((i + 1) + " ");
            }
            System.out.println();
        }
    }
}
