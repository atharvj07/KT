#include<bits/stdc++.h>
using namespace std;
int main(){
int n, t, t1, t2,ss;
cin>>n>>t;
map<int,int>m;
priority_queue<pair<int,int>>q;
for(int i=1;i<=n;i++){
 cin>>t1>>t2;
 m[i]=t1;
 q.push({t2,i});
}
while(t>0 && !q.empty()){
 t=t-q.top().first;
 if(t>=0){
  ss++;
  t=t-m[q.top().second];
 }
 q.pop();
}
cout<<ss<<endl;
cout<<ss<<endl;
while(ss--){
 cout<<q.top().second<<" ";
 q.pop();
}
return 0;
}
