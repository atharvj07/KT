T = int(input().strip())
for _ in range(T):
    n = int(input().strip())
    sticks = list(map(int, input().strip().split()))
    stick_counts = {}
    for stick in sticks:
        stick_counts[stick] = stick_counts.get(stick, 0) + 1
    pairs = []
    for stick, count in stick_counts.items():
        if count >= 4:
            pairs.append((stick, stick))
        if count >= 2:
            pairs.append((stick, None))
    pairs.sort()
    best_ratio = float('inf')
    best_pair = (0, 0)
    for i in range(len(pairs)):
        if pairs[i][1] is not None:
            ratio = 4 + 4 * pairs[i][0] ** 2 / (pairs[i][0] ** 2)
            if ratio < best_ratio:
                best_ratio = ratio
                best_pair = (pairs[i][0], pairs[i][0])
        for j in range(i + 1, len(pairs)):
            if pairs[i][0] != pairs[j][0]:
                perimeter = 2 * (pairs[i][0] + pairs[j][0])
                area = pairs[i][0] * pairs[j][0]
                ratio = perimeter ** 2 / area
                if ratio < best_ratio:
                    best_ratio = ratio
                    best_pair = (pairs[i][0], pairs[j][0])
    print(best_pair[0], best_pair[0], best_pair[1], best_pair[1])
