#include <iostream>
#include <vector>

using namespace std;

bool possible(int p, int x) {
    if (x == 1) return false;
    if (p == 1) return true;
    int i = 1;
    while (true) {
        if (10 * i % x == 0) {
            return (10 * i / x <= p - 1);
        }
        i++;
    }
}

int main() {
    int p, x;
    cin >> p >> x;
    if (!possible(p, x)) {
        cout << "Impossible";
    } else {
        vector<int> a(p);
        int i = 1;
        while (true) {
            if (10 * i % x == 0) {
                a[p - 1] = 10 * i / x;
                break;
            }
            i++;
        }
        for (int j = p - 2; j >= 0; j--) {
            a[j] = (10 * a[j + 1]) % x;
        }
        for (int j = 0; j < p; j++) {
            cout << a[j];
        }
    }
    cout << endl;
}
