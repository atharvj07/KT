import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Map<Integer, Character> fibonacciMap = fibonacciMap(n);
        StringBuilder name = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            name.append(fibonacciMap.get(i));
        }
        System.out.println(name);
    }

    private static Map<Integer, Character> fibonacciMap(int n) {
        Map<Integer, Character> fibonacciMap = new HashMap<>();
        int f1 = 1, f2 = 1, f = 0;
        for (int i = 1; i <= n; i++) {
            if (i == 1 || i == 2) {
                fibonacciMap.put(i, 'O');
            } else {
                f = f1 + f2;
                f1 = f2;
                f2 = f;
                if (f <= n) {
                    fibonacciMap.put(f, 'O');
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (!fibonacciMap.containsKey(i)) {
                fibonacciMap.put(i, 'o');
            }
        }
        return fibonacciMap;
    }
}
