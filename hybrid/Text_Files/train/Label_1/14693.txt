#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int x1, p1, x2, p2;
        cin >> x1 >> p1 >> x2 >> p2;
        while (p1 > p2) {
            x2 *= 10;
            p2++;
        }
        while (p2 > p1) {
            x1 *= 10;
            p1++;
        }
        if (x1 == x2)
            cout << "=";
        else if (x1 < x2)
            cout << "<";
        else
            cout << ">";
    }
    return 0;
}
