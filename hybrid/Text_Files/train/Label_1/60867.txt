import sys

n, s, k = map(int, input().split())
r = list(map(int, input().split()))
c = input()
nxt = ['?', '?']
for i in range(1, n):
    if c[i] != c[i - 1]:
        nxt[int(c[i - 1]) - 1] = i
diff = [0] * 3
for rc in range(3):
    mn = sys.maxsize
    for i in range(n):
        if c[i] != chr(rc + 65):
            continue
        if rc != int(nxt[rc]):
            diff[rc] = min(diff[rc], r[i] - mn)
        mn = min(mn, r[i])
diff = list(filter(lambda x: x > 0, diff))
diff.sort()
if len(diff) < 2 or diff[0] + diff[1] < k:
    print(-1)
else:
    print(diff[0] + diff[1])
