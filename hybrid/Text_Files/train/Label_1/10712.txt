#include <bits/stdc++.h>
using namespace std;
int n;
vector<int> f, v;
int main() {
  cin >> n;
  f.resize(n + 1);
  v.resize(n + 1, 0);
  for (int i = 1; i <= n; i++) { cin >> f[i]; }
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    if (f[i] == i) cnt++;
    v[f[i]]++;
  }
  if (cnt >= 2) {
    cout << "-1\n";
    return 0;
  }
  if (cnt == 0) {
    cout << "1\n1 ";
    for (int i = 1; i <= n; i++) cout << i << " ";
    cout << "\n1\n";
    return 0;
  }
  int m = 0;
  int p;
  for (int i = 1; i <= n; i++) {
    if (v[i] >= 2) {
      m = i;
      p = v[i];
      break;
    }
  }
  if (m == 0) {
    cout << "-1\n";
    return 0;
  }
  cout << m << "\n";
  for (int i = 1; i <= n; i++) {
    if (f[i] == m)
      cout << p + 1 << " ";
    else
      cout << 1 << " ";
  }
  cout << "\n";
  for (int i = 1; i <= m; i++) {
    if (i == p + 1)
      cout << f[p + 1] << " ";
    else
      cout << i << " ";
  }
  cout << "\n";
  return 0;
}
