#include <iostream>
#include <vector>
#include <set>

using namespace std;

typedef long long LL;

int main() {
    int n, k;
    cin >> n >> k;
    set<LL> matryoshkas;
    vector<LL> chains[k];
    for (int i = 0; i < k; i++) {
        int m;
        cin >> m;
        for (int j = 0; j < m; j++) {
            LL a;
            cin >> a;
            matryoshkas.insert(a);
            chains[i].push_back(a);
        }
    }
    LL ans = 0;
    for (int i = 0; i < k; i++) {
        int size = chains[i].size();
        ans += size - 1;
        matryoshkas.erase(chains[i][size - 1]);
    }
    ans += matryoshkas.size();
    cout << ans << endl;
    return 0;
}
