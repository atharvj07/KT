import java.util.Scanner;
import java.math.BigInteger;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        BigInteger n = sc.nextBigInteger();
        BigInteger s = sc.nextBigInteger();
        BigInteger bt = BigInteger.TWO;

        if (n.compareTo(bt) < 0) {
            System.out.println("-1");
        } else {
            while (!bt.toString().equals(s.toString())) {
                bt = bt.add(BigInteger.ONE);
                String base = n.toString(bt.intValue());
                BigInteger bn = BigInteger.valueOf(0);

                for (int i = 0; i < base.length(); i++)
                    bn = bn.add(BigInteger.valueOf(Character.getNumericValue(base.charAt(i))));

                if (bn.equals(s)) break;
            }

            if (bt.compareTo(BigInteger.TWO) > 0) {
                System.out.println(bt);
            } else {
                System.out.println("-1");
            }
        }

        sc.close();
    }
}
