import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {
    private static final long MOD = 998244353;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<int[]> distancesList = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int[] distances = new int[m];
            for (int j = 0; j < m; j++) {
                distances[j] = scanner.nextInt();
            }
            distancesList.add(distances);
        }

        int[] maxDistances = new int[m];
        for (int[] distances : distancesList) {
            for (int i = 0; i < m; i++) {
                maxDistances[i] = Math.max(maxDistances[i], distances[i]);
            }
        }

        long result = 0;
        for (int i = 0; i < m; i++) {
            result += MODInverse(maxDistances[i], MOD);
            result %= MOD;
        }

        result = result * n % MOD;
        result = result * MODInverse(n, MOD) % MOD;

        System.out.println(result);
    }

    private static long MODInverse(long a, long m) {
        long m0 = m;
        long y = 0;
        long x = 1;

        while (a > 1) {
            long q = a / m;
            long t = m;

            m = a % m;
            a = t;
            t = y;

            y = x - q * y;
            x = t;
        }

        if (x < 0) {
            x += m0;
        }

        return x;
    }
}
