#include <bits/stdc++.h>
using namespace std;
string s[100005];
map<string,int> mp,mp1;
int main(){
	int n;
	cin>>n;
	for(int i=0;i<n;i++){
		cin>>s[i];
		int sz=s[i].size();
		if(mp1.count(s[i])) mp1[s[i]]++;
		else mp1[s[i]]=1;
		string t1="",t2="";
		for(int j=0;j<sz;j++){
			t1+=s[i][j];
			t2=s[i][j]+t2;
			mp[t1]++;
			mp[t2]++;
		}
	}
	int ans=0;
	for(int i=0;i<n;i++){
		int sz=s[i].size();
		string t=s[i];
		reverse(t.begin(),t.end());
		if(mp[t] and mp1[s[i]]>1){
			int cnt=mp[t]/mp1[s[i]];
			ans+=cnt-1;
		}
		else if(mp[t]){
			ans+=mp[t];
		}
	}
	int cnt=0;
	for(int i=0;i<n;i++){
		if(s[i].size()%2==0){
			string t1=s[i].substr(0,s[i].size()/2),t2=s[i].substr(s[i].size()/2,s[i].size());
			reverse(t2.begin(),t2.end());
			if(t1==t2) cnt++;
		}
	}
	ans+=cnt;
	cout<<ans/2<<endl;
	return 0;
}
