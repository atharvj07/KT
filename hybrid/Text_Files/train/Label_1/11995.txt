#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
constexpr ll MOD = 1e9 + 7;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> a(n);
  unordered_map<int, vector<int>> adj;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    --a[i];
    adj[i].push_back(a[i]);
  }
  int ans = 1LL << n;
  for (auto [x, xs] : adj) {
    int cnt = 0;
    for (int y : xs) {
      cnt += adj[y].size() == 1 && y != x;
    }
    ans -= cnt;
    ans = (ans + MOD) % MOD;
  }
  cout << ans << endl;
  return 0;
}
