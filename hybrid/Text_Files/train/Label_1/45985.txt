import java.util.Scanner;

public class BalanceScale {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int L = scanner.nextInt();
        int N = scanner.nextInt();
        int[] x = new int[N];
        int[] w = new int[N];
        for (int i = 0; i < N; i++) {
            x[i] = scanner.nextInt();
            w[i] = scanner.nextInt();
        }

        int sumX = 0;
        int sumW = 0;
        for (int i = 0; i < N; i++) {
            sumX += x[i] * w[i];
            sumW += w[i];
        }

        int xNew = -sumX / sumW;
        int wNew = sumW;

        System.out.println(1);
        System.out.println(xNew + " " + wNew);
    }
}
