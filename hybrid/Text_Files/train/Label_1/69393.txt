MOD = 998244353

def solve(n, k, len_, arr):
    dp = [[0, 0] for _ in range(n + 1)]  # dp[i][0] is the number of ways with no restriction at i, dp[i][1] is the number of ways with restriction at i
    dp[0][0] = 1
    cnt = [0] * (k + 1)  # Count of each number in the current window
    total = 0  # Total numbers in the current window
    l = 0  # Left pointer for the window

    for r in range(1, n + 1):
        if arr[r - 1] == -1:
            dp[r][0] = (dp[r - 1][0] * k) % MOD  # If it's -1, we can choose any number
            dp[r][1] = (dp[r - 1][0] + dp[r - 1][1] * (k - 1)) % MOD  # Either continue the restriction or start a new one
        else:
            dp[r][0] = dp[r - 1][0]  # We have no choice but to take the given number
            if cnt[arr[r - 1]] == 0:  # If it's the first occurrence in the window
                dp[r][1] = (dp[r - 1][0] + dp[r - 1][1]) % MOD  # Can start a new restriction or continue
            else:
                dp[r][1] = dp[r - 1][1]  # Continue the restriction
            cnt[arr[r - 1]] += 1
            total += 1

        if r - l == len_:
            if arr[l] != -1:
                cnt[arr[l]] -= 1
                total -= 1
            l += 1
        
        if total == len_:  # If the window is full and all numbers are the same, no valid arrangement
            dp[r][0] = dp[r][1] = 0

    return dp[n][0]

n, k, len_ = map(int, input().split())
arr = list(map(int, input().split()))
print(solve(n, k, len_, arr))
