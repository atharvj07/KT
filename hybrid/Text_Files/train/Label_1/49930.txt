import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] c = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            c[i] = sc.nextInt();
        }
        Map<Integer, Set<Integer>> map = new HashMap<>();
        for (int i = 0; i < m; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            map.computeIfAbsent(x, k -> new HashSet<>()).add(y);
            map.computeIfAbsent(y, k -> new HashSet<>()).add(x);
        }
        int max = 0;
        int ans = -1;
        for (int i = 1; i <= n; i++) {
            Set<Integer> set = map.get(i);
            int cnt = 0;
            for (int j : set) {
                if (j != c[i]) {
                    cnt++;
                }
            }
            if (cnt > max) {
                max = cnt;
                ans = c[i];
            }
        }
        System.out.println(ans);
    }
}
