#include<bits/stdc++.h>

using namespace std;

int32_t main()
{
    int n, i, sz, cnt, l, r, c1, c2;
    cin >> n;
    vector<int> a(n + 1, 0);
    vector<vector<int>> dis(n + 1, vector<int>(100001, 0));

    for (i = 1; i <= n; ++i) { cin >> a[i]; }
    cin >> l >> c1 >> r >> c2;

    queue<pair<int, int>> q;
    q.push(make_pair(l, c1));
    dis[l][c1] = 0;

    while (!q.empty())
    {
        l = q.front().first;
        c1 = q.front().second;
        q.pop();

        // Up (U)
        if (l > 1 && dis[l - 1][c1] == 0)
        {
            dis[l - 1][c1] = dis[l][c1] + 1;
            q.push(make_pair(l - 1, c1));
        }

        // Down (D)
        if (l < n && dis[l + 1][c1] == 0)
        {
            dis[l + 1][c1] = dis[l][c1] + 1;
            q.push(make_pair(l + 1, c1));
        }

        // Left (L)
        if (c1 > 1 && dis[l][c1 - 1] == 0)
        {
            if (c1 > a[l] + 1 || l != r)
            {
                dis[l][c1 - 1] = dis[l][c1] + 1;
                q.push(make_pair(l, c1 - 1));
            }
        }

        // Right (R)
        if (c1 < a[l] && dis[l][c1 + 1] == 0)
        {
            dis[l][c1 + 1] = dis[l][c1] + 1;
            q.push(make_pair(l, c1 + 1));
        }
    }
    cnt = dis[r][c2];
    if (cnt != 0) { cout << cnt << endl; }
    else { cout << -1 << endl; }
    return 0;
}
