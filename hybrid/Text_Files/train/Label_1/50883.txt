import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int p = input.nextInt();
        int[] times = new int[n];
        for (int i = 0; i < n; i++) {
            times[i] = input.nextInt();
        }
        PriorityQueue<Integer> queue = new PriorityQueue<>();
        int[] result = new int[n];
        int time = 0;
        for (int i = 0; i < n; i++) {
            while (!queue.isEmpty() && queue.peek() <= time) {
                queue.poll();
            }
            if (queue.isEmpty()) {
                result[i] = time;
            } else {
                result[i] = queue.peek();
            }
            time = Math.max(time, result[i] + p);
            queue.add(result[i] + times[i]);
        }
        for (int i = 0; i < n; i++) {
            System.out.println(result[i]);
        }
    }
}
