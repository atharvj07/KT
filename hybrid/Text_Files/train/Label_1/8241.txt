#include <bits/stdc++.h>
using namespace std;
const int MX = 1e5 + 5;
typedef long long ll;

struct node { ll a, b, ind; } v[MX];
bool operator<(node a, node b) {
  return (a.a - a.b) < (b.a - b.b);
}
ll n, m;
int main() {
  ll t;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
      cin >> v[i].a >> v[i].b;
      v[i].ind = i;
    }
    sort(v + 1, v + 1 + m);
    ll flowers = 0, type = 0, ans = 0;
    for (int i = m; i >= 1; i--) {
      flowers++;
      ans += v[i].a;
      if (flowers == n) break;
      ans -= v[i].b;
      type++;
      if (v[i].b > 0) break;
    }
    cout << ans << endl;
  }
}
