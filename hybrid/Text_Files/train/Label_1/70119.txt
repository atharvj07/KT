n, k = map(int, input().split())
s = input()

def max_beauty(s, k):
    max_length = 0
    for char in ['a', 'b']:
        left = 0
        count = 0
        for right in range(n):
            if s[right] != char:
                count += 1
            while count > k:
                if s[left] != char:
                    count -= 1
                left += 1
            max_length = max(max_length, right - left + 1)
    return max_length

print(max_beauty(s, k))
