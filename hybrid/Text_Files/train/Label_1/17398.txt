#include <iostream>

using namespace std;

typedef unsigned long long int ulli;
const ulli Mod = 1000000007ULL;

ulli powers[1005];

int main() {
    ios::sync_with_stdio(false);
    int n, k;
    cin >> n >> k;
    powers[0] = 1ULL;
    for (int i = 1; i <= n; ++i) {
        powers[i] = powers[i - 1] * (n - i + 1) % Mod;
    }
    if (k == 1) {
        cout << 1 << '\n';
    } else if (k == 2) {
        ulli ans = powers[n] - n - 1;
        ans %= Mod;
        ans += Mod;
        ans %= Mod;
        cout << ans << '\n';
    } else if (k == 3) {
        ulli ans = powers[n] - powers[n - 1] * (n - 3) % Mod;
        ans %= Mod;
        ans += Mod;
        ans %= Mod;
        cout << ans << '\n';
    } else if (k == 4) {
        ulli ans = powers[n] - powers[n - 1] * (n - 2) % Mod - powers[n - 2] * 2 + (n - 2) * (n - 3) % Mod;
        ans %= Mod;
        ans += Mod;
        ans %= Mod;
        cout << ans << '\n';
    }
    return 0;
}
