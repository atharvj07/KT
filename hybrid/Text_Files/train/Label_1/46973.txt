import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int crimes = 0;
        int recruits = 0;
        for (int i = 0; i < n; i++) {
            int event = sc.nextInt();
            if (event == -1) {
                if (recruits == 0) {
                    crimes++;
                } else {
                    recruits--;
                }
            } else {
                recruits += event;
            }
        }
        System.out.println(crimes);
    }
}
