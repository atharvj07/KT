#include <bits/stdc++.h>
#define MOD 1000000007
using namespace std;
typedef long long ll;
int main()
{
    ll x, k;
    cin >> x >> k;
    ll ans = x, temp = x;
    while (k--)
    {
        temp = (2LL * temp) % MOD;
        ans = (ans + temp) % MOD;
    }
    cout << ans << "\n";
    return 0;
}
