from collections import defaultdict, deque
n = int(input())
ad = defaultdict(list)
for _ in range(n-1):
	u, v = map(int, input().split())
	ad[u].append(v)
	ad[v].append(u)
vis = [-1] * (n+1)
ans = []
nn = n-1
def dfs(u, par):
	global nn
	vis[u] = 0
	for v in ad[u]:
		if vis[v] == -1:
			dfs(v, u)
			ans.append(vis[u]^vis[v])
			vis[u] = vis[v]
	if vis[u] == -1:
		vis[u] = nn
		nn -= 1
dfs(1, 0)
print(*ans)
