#include<bits/stdc++.h>
using namespace std;
int n;
int t[5001];
int l[5001],r[5001];
int main(){
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>t[i];
        l[i]=i;
        r[i]=i;
    }
    for(int i=0;i<n;i++){
        while(l[i]&&t[l[i]-1]==t[i])
            l[i]=l[l[i]-1];
    }
    for(int i=n-1;i>=0;i--){
        while(r[i]<n-1&&t[r[i]+1]==t[i])
            r[i]=r[r[i]+1];
    }
    int ans[5001];
    memset(ans,0,sizeof(ans));
    for(int i=0;i<n;i++){
        ans[t[i]]+=(i-l[i]+1)*(r[i]-i+1);
    }
    for(int i=1;i<=n;i++)
        cout<<ans[i]<<' ';
    cout<<endl;
}
