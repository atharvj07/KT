import sys
n=int(input())
a=list(map(int, input().split()))
b=[0 for i in range(2*n+1)]
c=[0 for i in range(2*n+1)]
ans=0
for x in a:
    if c[x]>0:
        for i in range(1, 2*n+1):
            if c[i]==0 and b[i]!=b[x]:
                ans+=1
                c[i]=1
                b[i]=b[x]
                break
    else:
        b[x]=ans//n
        c[x]=1
        if ans%n==n-1:
            ans+=1
            for i in range(1, 2*n+1):
                if c[i]==0:
                    c[i]=1
                    b[i]=ans//n
                    break
print(ans)
