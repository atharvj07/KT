#include <iostream>
#include <string>
using namespace std;
int main() {
  string s;
  cin >> s;
  int l = s.size();
  bool ok = true;
  for (int i = 1; i < 11; i++) {
    if (s.substr(0, i) == "CODEFORCES") {
      ok = false;
    }
  }
  for (int i = 1; i < l - 10; i++) {
    if (s.substr(i, 11) == "CODEFORCES") {
      ok = false;
    }
  }
  for (int i = l - 10; i < l; i++) {
    if (s.substr(i, 11) == "CODEFORCES") {
      ok = false;
    }
  }
  if (ok) {
    cout << "YES";
  } else {
    cout << "NO";
  }
  return 0;
}
