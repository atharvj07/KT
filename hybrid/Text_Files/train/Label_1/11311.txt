#include <iostream>
#include <string>
using namespace std;

int main() {
    int l;
    cin >> l;
    string n;
    cin >> n;
    int sum = 0;
    for (int i = 0; i < l / 2; i++) {
        sum += n[i] - '0';
    }
    for (int i = l / 2; i < l; i++) {
        sum += n[i] - '0';
    }
    cout << sum << endl;
    return 0;
}
