import java.util.*;

public class BooleanExpression {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String expression = scanner.nextLine();
        int n = scanner.nextInt();
        int[][] values = new int[n][5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                values[i][j] = scanner.nextInt();
            }
        }
        System.out.println(countValidExpressions(expression, values));
    }

    private static int countValidExpressions(String expression, int[][] values) {
        int[][][] dp = new int[expression.length() + 1][values.length][2]; // dp[i][j][k]: number of valid expressions ending at position i with j true values and k negations
        for (int i = 0; i < expression.length(); i++) {
            for (int j = 0; j < values.length; j++) {
                if (expression.charAt(i) == '?') {
                    for (int k = 0; k < 2; k++) {
                        dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i][j][0] + dp[i][j][1]) % MOD;
                    }
                } else if (expression.charAt(i) == '(') {
                    for (int k = 0; k < 2; k++) {
                        dp[i + 1][j][k] = dp[i][j][k];
                    }
                } else if (expression.charAt(i) == ')') {
                    for (int k = 0; k < 2; k++) {
                        dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i - 1][j][0] + dp[i - 1][j][1]) % MOD;
                    }
                } else if (expression.charAt(i) == '&') {
                    for (int k = 0; k < 2; k++) {
                        if (k == 0) {
                            dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i - 1][j][0] * dp[i][j][0]) % MOD;
                        } else {
                            dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i - 1][j][0] * dp[i][j][1] + dp[i - 1][j][1] * dp[i][j][0]) % MOD;
                        }
                    }
                } else if (expression.charAt(i) == '|') {
                    for (int k = 0; k < 2; k++) {
                        if (k == 0) {
                            dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i - 1][j][0] * dp[i][j][0] + dp[i - 1][j][0] * dp[i][j][1] + dp[i - 1][j][1] * dp[i][j][0]) % MOD;
                        } else {
                            dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i - 1][j][0] * dp[i][j][1] + dp[i - 1][j][1] * dp[i][j][1]) % MOD;
                        }
                    }
                } else {
                    char variable = expression.charAt(i);
                    for (int k = 0; k < 2; k++) {
                        if (k == 0) {
                            if (variable == 'A' || variable == 'a') {
                                dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i][j][0]) % MOD;
                            } else {
                                dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i][j][1]) % MOD;
                            }
                        } else {
                            if (variable == 'A' || variable == 'a') {
                                dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i][j][1]) % MOD;
                            } else {
                                dp[i + 1][j][k] = (dp[i + 1][j][k] + dp[i][j][0]) % MOD;
                            }
                        }
                    }
                }
            }
        }
        int result = 0;
        for (int j = 0; j < values.length; j++) {
            for (int k = 0; k < 2; k++) {
                if (values[j][4] == 1 && k == 0 || values[j][4] == 0 && k == 1) {
                    result = (result + dp[expression.length()][j][k]) % MOD;
                }
            }
        }
        return result;
    }
}
