import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.StringTokenizer;

class Main {
    static long mod = 1000000007;
    static int maxDeg;
    static boolean[] vis;
    static ArrayList<ArrayList<Integer>> graph = new ArrayList<>();
    static long[][] dp;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        maxDeg = 0;
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            graph.get(a).add(b);
            graph.get(b).add(a);
            maxDeg = Math.max(maxDeg, Math.max(graph.get(a).size(), graph.get(b).size()));
        }
        st = new StringTokenizer(br.readLine());
        int k = Integer.parseInt(st.nextToken());
        int x = Integer.parseInt(st.nextToken());
        dp = new long[n + 1][maxDeg + 1];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(dp[i], -1);
        }
        vis = new boolean[n + 1];
        long ans = dfs(1, 1, 0, m, k, x);
        System.out.println(ans);
    }

    static long dfs(int node, int par, int deg, int m, int k, int x) {
        if (vis[node]) {
            return dp[node][deg];
        }
        vis[node] = true;
        long ans = 0;
        if (deg == 0) {
            for (int j = 1; j <= m; j++) {
                if (j != k || x > 0) {
                    if (graph.get(node).size() == 1) {
                        ans += 1;
                        ans %= mod;
                    } else {
                        ans += dfs(graph.get(node).get(0), node, 1, m, k, x);
                        ans %= mod;
                    }
                }
            }
        } else {
            boolean adjacent[] = new boolean[k + 1];
            for (int nbr: graph.get(node)) {
                if (nbr == par) {
                    continue;
                }
                for (int i = 0; i <= k; i++) {
                    adjacent[i] |= (dp[nbr][i] != -1);
                }
            }
            long cnt = 0;
            for (int j = 1; j <= m; j++) {
                if (j != k || x > 0) {
                    if (!adjacent[j]) {
                        cnt++;
                        if (graph.get(node).size() == 1) {
                            ans += 1;
                            ans %= mod;
                        } else {
                            for (int nbr: graph.get(node)) {
                                if (nbr == par) {
                                    continue;
                                }
                                ans += (dfs(nbr, node, deg + 1, m, k, x) * cnt) % mod;
                                ans %= mod;
                            }
                        }
                    }
                }
            }
        }
        dp[node][deg] = ans;
        return ans;
    }
}
