#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5+5;
int s[maxn], f[maxn], b[maxn], ans[maxn];
inline int read()
{
  int x=0;char ch=getchar();
  while (ch<'0'||ch>'9') ch=getchar();
  while (ch>='0'&&ch<='9'){x=x*10+ch-'0';ch=getchar();}
  return x;
}
inline void read(int& x){x=read();}
int main()
{
  int n, k;
  read(n), read(k);
  for (int i = 1; i <= n; ++i) read(s[i]), ++f[s[i]];
  for (int i = 2e5; i; i--)
  {
    f[i] += f[i+1];
    if (f[i] >= k)
    {
      for (int j = 0; j < k; j++) ans[j+1] = i;
      break;
    }
  }
  for (int i = 1; i <= k; i++) cout << ans[i] << ' ';
  return 0;
}
