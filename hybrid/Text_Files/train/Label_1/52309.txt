import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        for (int i = 0; i < n; i++) {
            pq.add(sc.nextInt());
        }
        int sum = 0;
        int cnt = 0;
        int ans = -1;
        while (!pq.isEmpty()) {
            int cur = pq.poll();
            if (cnt + cur >= k) {
                ans = sum + cur / 2;
                break;
            }
            sum += cur;
            cnt++;
        }
        System.out.println(ans);
    }
}
