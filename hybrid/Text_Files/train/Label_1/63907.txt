import sys; sys.setrecursionlimit(500000)
def I(): return int(input())
def LI(): return list(map(int, input().split()))
n = I()
a = LI()
ans = []
for j in range(1, n):
    if a[j - 1] > a[j]:
        ans += [[2, j, j - 1]]
        a[j - 1] -= a[j - 1] - a[j]
    else:
        ans += [[1, j, j - 1]]
        a[j] += a[j - 1] - a[j]
print(len(ans))
for x in ans:
    print(*x)
