#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, q;
    cin >> n >> q;
    vector<int> a(n), pos(51, -1);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        pos[a[i]] = i;
    }
    for (int i = 0; i < q; i++) {
        int t;
        cin >> t;
        cout << pos[t] + 1 << endl;
        pos[a[pos[t]]] = pos[t];
        pos[t] = 0;
    }
    return 0;
}
