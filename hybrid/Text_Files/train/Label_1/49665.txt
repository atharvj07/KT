import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] pens = new int[n];
        for (int i = 0; i < n; i++) {
            pens[i] = input.nextInt();
        }
        int min = Integer.MAX_VALUE;
        int index = -1;
        for (int i = 0; i < n; i++) {
            if (pens[i] % 7 < min) {
                min = pens[i] % 7;
                index = i + 1;
            }
        }
        System.out.println(index);
    }
}
