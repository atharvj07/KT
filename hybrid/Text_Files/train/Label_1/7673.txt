#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    long long n, sum = 0;
    cin >> n;
    vector<long long> a(n);
    for (auto &i : a) cin >> i, sum += i;
    cout << (sum % 2 ? "First" : "Second") << "\n";
  }
}
