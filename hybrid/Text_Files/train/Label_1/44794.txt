import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] w = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            w[i] = in.nextInt();
        }
        Arrays.sort(w);
        int min = 0;
        for (int i = 0; i < n - 1; i++) {
            min += w[2 * i + 1] - w[2 * i];
        }
        System.out.println(min);
    }
}
