#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define mod 1000000007

ll fastpow(ll a, ll b)
{
    ll res = 1;
    while (b > 0)
    {
        if (b & 1)
            res = res * a % mod;
        b >>= 1;
        a = a * a % mod;
    }
    return res;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    ll n;
    cin >> n;
    cout << fastpow(2, n / 2 - 1) << "\n";
    return 0;
}
