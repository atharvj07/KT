n1, n2, n3, n4 = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = list(map(int, input().split()))
d = list(map(int, input().split()))
m1 = int(input())
bad1 = [[] for i in range(n1 + 1)]
for i in range(m1):
    x, y = map(int, input().split())
    bad1[x].append(y)
m2 = int(input())
bad2 = [[] for i in range(n2 + 1)]
for i in range(m2):
    x, y = map(int, input().split())
    bad2[x].append(y)
m3 = int(input())
bad3 = [[] for i in range(n3 + 1)]
for i in range(m3):
    x, y = map(int, input().split())
    bad3[x].append(y)
ans = []
for i in range(1, n1 + 1):
    for j in range(1, n2 + 1):
        if j not in bad1[i]:
            for k in range(1, n3 + 1):
                if k not in bad2[j]:
                    for l in range(1, n4 + 1):
                        if l not in bad3[k]:
                            ans.append(a[i - 1] + b[j - 1] + c[k - 1] + d[l - 1])
if ans:
    print(min(ans))
else:
    print(-1)
