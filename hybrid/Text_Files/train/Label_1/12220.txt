#include <cstdio>
#include <cmath>
#include <vector>

using namespace std;
typedef long long ll;
const ll maxLen = 1500000ll*1500000ll;

int n;
vector<pair<ll, ll>> v;
bool isexceed = false;

bool check(ll x, ll y) {
    return x*x + y*y <= maxLen;
}

void dfs(ll x, ll y, int u, vector<int>& ans) {
    if (u == n) {
        if (check(x, y)) {
            for (int i : ans) printf("%d ", i);
            printf("\n");
            isexceed = true;
        } 
        return;
    }
    if (isexceed) return;
    ans.push_back(1);
    dfs(x+v[u].first, y+v[u].second, u+1, ans);
    ans.pop_back();
    if (check(x-v[u].first, y-v[u].second)) {
        ans.push_back(-1);
        dfs(x-v[u].first, y-v[u].second, u+1, ans);
        ans.pop_back();
    }
}

int main() {
    scanf("%d", &n);
    v.resize(n);
    for (int i = 0; i < n; i++) {
        scanf("%lld %lld", &v[i].first, &v[i].second);
    }
    vector<int> ans;
    dfs(0, 0, 0, ans);
    if (!isexceed) printf("-1\n");
    return 0;
}
