#include <bits/stdc++.h>
using namespace std;
template<class T> using v=vector<T>;
int main() {
  int n,d; cin>>n;
  vector<int> a(n),c(n+1),last(n),last2(n);
  v<v<int>> g(n);
  for(int&i:a)cin>>i;
  for(int i=n-1;~i;i--) {
    int x=a[i];
    d=x;
    while(c[d]) ++d;
    if(c[d]) g[c[d]-1].push_back(i);
    c[d]=i+1;
    int  y=a[i+1];
    if(y!=x && c[y]){
      int j=c[y]-1;
      if(last[j]==i) g[j].clear();
      last[j]=i;
      g[j].push_back(i);
    }
    if(c[y]){
      int j=c[y]-1;
      if(last2[j]==i) g[j].clear();
      last2[j]=i;
      g[j].push_back(i);
    } 
  }
  for(int i=n-1;~i;i--){
    for(auto j:g[i]){
      if(a[j]==a[i]) {
        cout<<j+1<<'\n';
        return 0;
      }
    }
  }
  cout<<"-1"<<'\n';
}
