#include <bits/stdc++.h>
using namespace std;
int n;
vector<int> a;
int main() {
  cin >> n;
  a.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  int ans = 0, cur = 0, pre = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] == a[i - 1]) {
      if (cur == pre) { ans++; }
      cur++;
    } else {
      pre = cur;
      cur = 1;
    }
  }
  cout << ans << '\n';
  return 0;
}
