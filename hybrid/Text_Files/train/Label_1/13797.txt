#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
vector<long long> cnt;
int main() {
  long long n, c, al;
  cin >> n >> al;
  cnt.assign(1000001, 0);
  for (int i = 0; i < n; i++) {
    cin >> c;
    cnt[al]++;
    cnt[c]++;
  }
  bool fl = false;
  for (long long i = 1; i <= 1000000; i++) {
    if (i == al)
      continue;
    if (cnt[al] > cnt[i]) {
      fl = true;
      cout << i << endl;
      break;
    }
  }
  if (!fl)
    cout << -1 << endl;
  return 0;
}
