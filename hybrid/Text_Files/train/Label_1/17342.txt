#include <bits/stdc++.h>
using namespace std;
const int maxn = 1000 + 10;
const int inf = 0x3f3f3f3f;
int t, n, vis[maxn][maxn], a[maxn];
int32_t main() {
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 1; i <= n; i++) { cin >> a[i]; }
        int ans = 2;
        for (int i = 1; i <= n;++i) {
            if (!vis[1][i] && !vis[2][i]) {
                int cnt = 0;
                queue<pair<int, int>> q;
                q.push({1, i});
                q.push({2, i});
                vis[1][i] = vis[2][i] = 1;
                while (!q.empty()) {
                    auto x = q.front();
                    q.pop();
                    int fl = x.first, p = x.second;
                    cnt++;
                    if (p > 1 && !vis[fl][p - 1]) { q.push({fl, p - 1}); vis[fl][p - 1] = 1; }
                    if (p < n && !vis[fl][p + 1]) { q.push({fl, p + 1}); vis[fl][p + 1] = 1; }
                    if (!vis[3 - fl][p] && a[p]) { q.push({3 - fl, p}); vis[3 - fl][p] = 1; }
                }
                ans = max(ans, cnt);
            }
        }
        cout << ans << endl;
        memset(vis, 0, sizeof(vis));
    }
    return 0;
}
