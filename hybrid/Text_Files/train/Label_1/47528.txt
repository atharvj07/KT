import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long[] A = new long[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextLong();
        }
        Arrays.sort(A);
        StringBuilder sb = new StringBuilder();
        for (long a : A) {
            sb.append(a).append(' ');
        }
        System.out.println(sb.toString());
    }
}
