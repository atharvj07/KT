#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int x[4 * n + 1], y[4 * n + 1];
  int cntx[51] = {}, cnty[51] = {};
  for (int i = 0; i < 4 * n + 1; i++) {
    cin >> x[i] >> y[i];
    cntx[x[i]]++;
    cnty[y[i]]++;
  }
  for (int i = 0; i < 51; i++) {
    if (cntx[i] == n - 1) {
      for (int j = 0; j < 4 * n + 1; j++) {
        if (x[j] == i) {
          cout << x[j] << " " << y[j] << "\n";
          return 0;
        }
      }
    }
    if (cnty[i] == n - 1) {
      for (int j = 0; j < 4 * n + 1; j++) {
        if (y[j] == i) {
          cout << x[j] << " " << y[j] << "\n";
          return 0;
        }
      }
    }
  }
  return 0;
}
