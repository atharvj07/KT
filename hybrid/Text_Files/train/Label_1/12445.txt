#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), c(n), t(m);
  vector<long long> d(m);
  for (auto &x : a) cin >> x;
  for (auto &x : c) cin >> x;
  for (auto &x : t) cin >> x;
  for (auto &x : d) cin >> x;
  vector<long long> b(n);
  for (auto &x : b) x = a[x];
  priority_queue<pair<long long, int>> pq;  // {-cost, id}
  for (int i = 0; i < n; i++) pq.push({-c[i], i});
  for (int i = 0; i < m; i++) {
    long long cost = 0;
    while (d[i]--) {
      auto [co, id] = pq.top();
      co = -co;
      if (b[id] > 0) {
        b[id]--;
        cost += co;
      } else {
        pq.pop();
        if (pq.empty()) break;
        auto [co, id] = pq.top();
        co = -co;
        b[id]--;
        cost += co;
      }
    }
    cout << cost << endl;
  }
  return 0;
}
