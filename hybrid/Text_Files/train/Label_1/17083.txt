#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>
#include <set>

using namespace std;

int MOD = 1e9 + 7;

int main() {
  int n;
  cin >> n;
  vector<int> m(n);
  for (int i = 0; i < n; i++) {
    cin >> m[i];
  }
  vector<unordered_map<int, int>> dp(n);  
  dp[0][0] = 1;                                      
  for (int i = 1; i < n; i++) {
    for (const auto& [j, cnt] : dp[i - 1]) {    
      if (cnt == m[i - 1]) continue;            
      dp[i][j] = (dp[i][j] + dp[i - 1][j]) % MOD; 
      dp[i][(j + i) % n] = (dp[i][(j + i) % n] + dp[i - 1][j]) % MOD;
    }
  }
  int ans = 0;
  for (const auto& [_, cnt] : dp[n - 1]) {
    ans = (ans + cnt) % MOD;
  }
  cout << ans << endl;
  return 0;
}
