def distance_squared(x1, y1, x2, y2):
    return (x1 - x2) ** 2 + (y1 - y2) ** 2

def min_water_amount(n, x1, y1, x2, y2, flowers):
    distances_to_f1 = [distance_squared(x1, y1, x, y) for x, y in flowers]
    distances_to_f2 = [distance_squared(x2, y2, x, y) for x, y in flowers]

    # If all flowers can be watered by one fountain, return the maximum distance squared to that fountain
    if max(distances_to_f1) == 0 or max(distances_to_f2) == 0:
        return max(max(distances_to_f1), max(distances_to_f2))

    combined_distances = sorted(zip(distances_to_f1, distances_to_f2), key=lambda x: x[0])

    min_water = float('inf')
    for i in range(n + 1):
        max_distance_f1 = max(combined_distances[:i], default=(0,))[0]
        max_distance_f2 = max(combined_distances[i:], key=lambda x: x[1], default=(0, 0))[1]
        min_water = min(min_water, max_distance_f1 + max_distance_f2)

    return min_water

# Input reading
n, x1, y1, x2, y2 = map(int, input().split())
flowers = [tuple(map(int, input().split())) for _ in range(n)]

# Solve
print(min_water_amount(n, x1, y1, x2, y2, flowers))
