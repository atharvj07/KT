#include <bits/stdc++.h>

using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        string s; cin >> s;
        bool is_palin = true;
        for (int i = 0; i < n / 2; i++) {
            if (s[i] != s[n - i - 1]) {
                is_palin = false;
                break;
            }
        }
        cout << (is_palin ? "BOB" : "ALICE") << '\n';
    }
    return 0;
}
