n, k = map(int, input().split())
a = list(map(int, input().split()))

count = 0
left = {}
right = {}

for x in a:
    if x in right:
        right[x] += 1
    else:
        right[x] = 1

for i in range(n):
    right[a[i]] -= 1
    if a[i] % k == 0:
        left_val = a[i] // k
        right_val = a[i] * k
        if left_val in left and right_val in right:
            count += left[left_val] * right[right_val]
    if a[i] in left:
        left[a[i]] += 1
    else:
        left[a[i]] = 1

print(count)
