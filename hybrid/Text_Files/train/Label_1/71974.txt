n = int(input())
a = list(map(int, input().split()))
m = int(input())
s = [0] * n
for i in range(m):
    l, r = map(int, input().split())
    l, r = l - 1, r - 1
    s[l] += 1
    if r + 1 < n:
        s[r + 1] -= 1
cnt = 0
ans = []
for i in range(n):
    cnt += s[i]
    if cnt % 2 == 1:
        ans.append('odd')
    else:
        ans.append('even')
print(*ans, sep='\n')
