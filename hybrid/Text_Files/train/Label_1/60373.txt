n = int(input())
events = [input() for _ in range(n)]

current_visitors = set()
max_visitors = 0
pre_inside = 0

for event in events:
    if event[0] == '+':
        current_visitors.add(event[2:])
    else:
        if event[2:] not in current_visitors:
            pre_inside += 1
        else:
            current_visitors.remove(event[2:])
    max_visitors = max(max_visitors, len(current_visitors) + pre_inside)

print(max_visitors)
