#include <bits/stdc++.h>
using namespace std;

int main() {
    string s, t;
    cin >> s >> t;
    int ans = 0;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] != t[i]) ans++;
    }
    if (ans == 1)
        cout << "automaton";
    else if (ans > 2)
        cout << "need tree";
    else
        cout << "array";
}
