n = int(input())
beacons = [list(map(int, input().split())) for _ in range(n)]
beacons.sort()

# Initialize DP and auxiliary array
dp = [0] * (n + 1)  # dp[i] represents the minimum number of beacons destroyed when activating the ith beacon
positions = [beacons[i][0] for i in range(n)]  # Extract positions for binary search

for i in range(1, n + 1):
    # Find the beacon that won't be destroyed by the current beacon's activation
    left = 0
    right = i - 1
    while left <= right:
        mid = (left + right) // 2
        if positions[mid] < positions[i - 1] - beacons[i - 1][1]:
            left = mid + 1
        else:
            right = mid - 1

    # Calculate the number of beacons destroyed if the current beacon is activated
    dp[i] = dp[left] + (i - left - 1)

# Find the minimum number of beacons destroyed
result = n  # Assume all beacons are destroyed
for i in range(1, n + 1):
    result = min(result, dp[i] + (n - i))

print(result)
