#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;

int main() {
    int k;
    cin >> k;
    vector<ll> c(k);
    for (int i = 0; i < k; ++i) { cin >> c[i]; }
    sort(c.begin(), c.end());
    ll ans = 1;
    for (int i = 1; i < k; ++i) { ans = (ans * c[i]) % mod; }
    cout << ans << endl;
    return 0;
}
