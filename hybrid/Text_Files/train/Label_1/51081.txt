import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int[] mall = new int[k];
        boolean[] mallFlag = new boolean[n + 1];
        Arrays.fill(mallFlag, false);
        Road[] roads = new Road[m + 1];
        for (int i = 0; i < m; i++) {
            int t1 = sc.nextInt();
            int t2 = sc.nextInt();
            int l = sc.nextInt();
            roads[i] = new Road(t1, t2, l);
        }
        for (int i = 0; i < k; i++) {
            mall[i] = sc.nextInt();
            mallFlag[mall[i]] = true;
        }
        int ans = 0;
        for (int t = 1; t <= n; t++) {
            int length = solve(t, mallFlag, roads, n);
            if (t == 1 || ans < length) {
                ans = length;
            }
        }

        System.out.println(ans);
    }

    private static int solve(int start, boolean[] mallFlag, Road[] roads, int n) {
        Queue<Integer> q = new LinkedList<>();
        int[] dist = new int[n + 1];
        Arrays.fill(dist, -1);
        q.add(start);
        dist[start] = 0;
        while (!q.isEmpty()) {
            int now = q.poll();
            for (int i = 0; i < roads.length; i++) {
                if (roads[i] == null) {
                    continue;
                }
                if (roads[i].t1 == now) {
                    if (dist[roads[i].t2] == -1) {
                        dist[roads[i].t2] = dist[now] + roads[i].l;
                        q.add(roads[i].t2);
                    }
                } else if (roads[i].t2 == now) {
                    if (dist[roads[i].t1] == -1) {
                        dist[roads[i].t1] = dist[now] + roads[i].l;
                        q.add(roads[i].t1);
                    }
                }
            }
        }
        int max = 0;
        for (int i = 1; i <= n; i++) {
            if (!mallFlag[i]) {
                continue;
            }
            if (dist[i] == -1) {
                max = -1;
                break;
            }
            if (max < dist[i]) {
                max = dist[i];
            }
        }
        return max;
    }

    private static class Road {
        int t1;
        int t2;
        int l;

        Road(int t1, int t2, int l) {
            this.t1 = t1;
            this.t2 = t2;
            this.l = l;
        }
    }
}
