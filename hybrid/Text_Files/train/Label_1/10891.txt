#include <iostream>
#include <unordered_map>
using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, m, k;
  cin >> n >> m >> k;
  unordered_map<int, int> a;
  while (n--) {
    int x;
    cin >> x;
    a[x]++;
  }
  while (m--) {
    int x;
    cin >> x;
    if (a.find(x) == a.end()) {
      cout << "NO" << endl;
      return 0;
    }
    a[x]--;
    if (a[x] == 0) a.erase(x);
  }
  cout << "YES" << endl;
}
