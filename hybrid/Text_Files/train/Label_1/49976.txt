import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextInt();
        long x = in.nextInt();
        long y = in.nextInt();
        System.out.println(Math.min(n * x, y + (n - 1) * x));
    }
}
