import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    static int N;
    static int[] depth;
    static int[] siz;
    static ArrayList<Integer>[] adj;

    public static void dfs1(int u, int p) {
        siz[u] = 1;
        for (int v : adj[u]) {
            if (v != p) {
                dfs1(v, u);
                siz[u] += siz[v];
            }
        }
    }

    public static void dfs2(int u, int p, int d) {
        depth[u] = d;
        int maxSubtree = -1;
        int maxSubtreeSize = -1;
        for (int v : adj[u]) {
            if (v != p) {
                if (siz[v] > maxSubtreeSize) {
                    maxSubtree = v;
                    maxSubtreeSize = siz[v];
                }
            }
        }
        for (int v : adj[u]) {
            if (v != p && v != maxSubtree) {
                dfs2(v, u, d + 1);
            }
        }
        if (maxSubtree != -1) {
            dfs2(maxSubtree, u, d + 1);
        }
    }

    public static int solve() {
        dfs1(1, 1);
        dfs2(1, 1, 0);
        int maxDepth = 0;
        for (int i = 1; i <= N; i++) {
            maxDepth = Math.max(maxDepth, depth[i]);
        }
        return maxDepth + 1;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        adj = new ArrayList[N + 1];
        for (int i = 0; i <= N; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < N - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj[u].add(v);
            adj[v].add(u);
        }
        depth = new int[N + 1];
        siz = new int[N + 1];
        System.out.println(solve());
    }
}
