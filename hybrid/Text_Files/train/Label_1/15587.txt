#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

struct train {
    ll h, m;
    bool operator<(const train &other) const {
        if (h == other.h) return m < other.m;
        return h < other.h;
    }
};

int main() {
    ll n, h, m, k;
    cin >> n >> h >> m >> k;
    vector<train> v(n);
    for (auto &t : v) cin >> t.h >> t.m;
    sort(v.begin(), v.end());
    ll ans = 0;
    ll st = 0;
    for (auto &t : v) {
        if (st) {
            st += m / 2;
            while (st + k < t.h * m + t.m) st += m / 2;
        }
        if (st + k > t.h * m + t.m) ans++, st = 0;
    }
    ll last = h * m - 1;
    if (st) {
        st += m / 2;
        while (st + k < last) st += m / 2;
    }
    if (st + k > last) ans++;
    cout << ans << ' ' << st % (m / 2) << '\n';
    for (auto &t : v) {
        if (st + k > t.h * m + t.m) cout << t.h << ' ' << t.m << '\n';
    }
}
