#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];

  sort(a.begin(), a.end());

  int res = a[n - 1] - a[0], k = 1;
  vector<int> t(n, 1);

  for (int i = 1, j = n - 2; i < j; i++, j--) {
    if (a[j] - a[i] < a[j] - a[j - 1]) {
      t[j] = k + 1;
      res -= a[j] - a[i];
      res += a[j] - a[j - 1];
      k++;
    }
  }

  cout << res << " " << k << endl;
  for (int x : t) cout << x << " ";
  cout << endl;
  return 0;
}
