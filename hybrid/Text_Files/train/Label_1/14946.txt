#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 5, MAXQ = 1e5 + 5;
int n, m, k;
int a[MAXN], add[MAXN];
int l[MAXN], r[MAXN], d[MAXN];
int ql[MAXQ], qr[MAXQ];

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    cin >> n >> m >> k;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    for (int i = 1; i <= m; ++i) cin >> l[i] >> r[i] >> d[i];
    for (int i = 1; i <= k; ++i) cin >> ql[i] >> qr[i];

    for (int i = 1; i <= k; ++i) {
        add[l[ql[i]]] += d[ql[i]];
        add[r[qr[i]] + 1] -= d[qr[i]];
    }

    for (int i = 1; i <= n; ++i) {
        add[i] += add[i - 1];
        cout << a[i] + add[i] << ' ';
    }
    cout << '\n';

    return 0;
}
