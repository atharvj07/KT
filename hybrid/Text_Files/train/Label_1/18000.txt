#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

struct Event {
    int day, type;

    Event() : day(0), type(0) {}
    Event(int day, int type) : day(day), type(type) {}

    bool operator<(const Event& other) const {
        return day < other.day;
    }
};

int main() {
    int n;
    cin >> n;

    vector<Event> events;
    for (int i = 0; i < n; i++) {
        char gender;
        int a, b;
        cin >> gender >> a >> b;

        events.push_back(Event(a, gender == 'F' ? 1 : -1));
        events.push_back(Event(b + 1, gender == 'F' ? -1 : 1));
    }

    sort(events.begin(), events.end());

    int balance = 0;
    int ans = 0;
    for (auto& event : events) {
        balance += event.type;
        ans = max(ans, balance);
    }

    cout << ans << endl;

    return 0;
}
