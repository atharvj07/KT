import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        long m = scanner.nextLong();
        String sNumber = String.valueOf(n);
        char[] digits = sNumber.toCharArray();
        Arrays.sort(digits);
        String smallest = new String(digits);
        String smallestTrim = smallest.replaceAll("^0+", "");
        System.out.println(smallestTrim.equals(String.valueOf(m)) ? "OK" : "WRONG_ANSWER");
    }
}
