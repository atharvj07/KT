def dfs(u, par):
    node[u][0] = 1
    node[u][1] = 0
    for v in adj[u]:
        if v == par: continue
        dfs(v, u)
        node[u][0] += node[v][1]
        node[u][1] += node[v][0]

n = int(input())
adj = {}
for i in range(1, n+1):
    adj[i] = []
for i in range(1, n):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)

node = {}
for i in range(1, n+1):
    node[i] = [0, 0]
dfs(1, -1)
mx = node[1][0]
for i in range(1, n+1):
    if node[i][0] > mx:
        mx = node[i][0]
print(mx-1)
