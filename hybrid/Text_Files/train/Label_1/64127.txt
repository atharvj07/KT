import sys

input = sys.stdin.readline

k = int(input())
a = []
s = []
for _ in range(k):
    n = int(input())
    a.append([int(i) for i in input().split()])
    s.append(sum(a[-1]))
l = len(a)
for i in range(l):
    for j in range(i + 1, l):
        for x in range(len(a[i])):
            for y in range(len(a[j])):
                if s[i] - a[i][x] == s[j] - a[j][y]:
                    print("YES")
                    print(i + 1, x + 1)
                    print(j + 1, y + 1)
                    sys.exit()
print("NO")
