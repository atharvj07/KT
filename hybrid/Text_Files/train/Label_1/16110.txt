#include <cstdio>
#include <algorithm>
using namespace std;

const int MaxN = 1e8 + 10;
int n, m, c_l, c_e, v;
int l[MaxN], e[MaxN];
int cst[MaxN];

int gcd(int a,int b){
    if(a%b==0)
        return b;
    return gcd(b, a%b);
}

int main() {
    scanf("%d %d %d %d %d", &n, &m, &c_l, &c_e, &v);
    for (int i = 0; i < c_l; i++) scanf("%d", &l[i]);
    for (int i = 0; i < c_e; i++) scanf("%d", &e[i]);
    int q; scanf("%d", &q);
    for (int i = 0; i < q; i++) {
        int x1, y1, x2, y2; scanf("%d %d %d %d", &x1, &y1, &x2, &y2);
        x1--, x2--, y1--, y2--;
        int ans = v * (x2 - x1);
        if (v==n-1){
            ans = x2 - x1;
        }
        ans = min(ans, abs(y2 - y1));
        for (int j = 0; j < c_e; j++) {
            int y = e[j];
            int tt = (x2 - x1) * v;
            if (v==n-1){
                tt = x2 - x1;
            }
            tt += abs(y - y1) + abs(y - y2);
            ans = min(ans, tt);
        }
        for (int j = 0; j < c_l; j++) {
            int y = l[j];
            int dx = x2 - x1;
            if (v==n-1) dx = 1;
            int tt = dx * v;
            tt += abs(y - y1) + abs(y - y2);
            ans = min(ans, tt);
        }
        ans++;
        printf("%d\n", ans);
    }
    return 0;
}
