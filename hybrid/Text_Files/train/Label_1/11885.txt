#include <iostream>
#include <vector>
#include <map>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n + 1);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  map<pair<int, int>, int> deg;
  for (int i = 1; i <= n; i++) {
    for (int j : adj[i]) {
      if (i < j) deg[{i, j}]++;
    }
  }
  int ans = 0;
  for (auto p : deg) {
    ans += p.second / 2;
  }
  cout << ans << endl;
  return 0;
}
