#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  int32_t n, k;
  cin >> n >> k;
  vector<int32_t> a(k);
  for (auto &x : a) cin >> x;
  unordered_map<int32_t, int32_t> idx;
  for (int32_t i = 0; i < k; ++i) idx[a[i]] = i;
  int32_t j = 1;
  for (int32_t i = 0; i < k; ++i) {
    for (int32_t x : {a[i] + 1, a[i] - 1, a[i] + n, a[i] - n})
      if (idx.count(x) && idx[x] > i && j < n * k) {
        j++;
        swap(a[i], a[idx[x]]);
        idx[a[i]] = i;
        idx[x] = idx[x] - 1;
      }
  }
  for (auto x : a) cout << x << " ";
  cout << endl;
  return 0;
}
