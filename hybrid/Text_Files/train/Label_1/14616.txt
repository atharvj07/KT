#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, x, y;
  cin >> n >> x >> y;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) {
    long long time = (a[i] + 1) / x;
    if (time % (x + y) != 0) cout << "Vanya" << endl;
    else cout << "Both" << endl;
  }
  return 0;
}
