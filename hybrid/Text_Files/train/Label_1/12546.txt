#include <iostream>
#include <algorithm>

using namespace std;

int64_t gcd(int64_t a, int64_t b) {
    if (b == 0) {
        return a;
    }
    return gcd(b, a % b);
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int64_t a, b;
        cin >> a >> b;
        if (a == 0 || b == 0) {
            cout << max(a, b) << " 0\n";
            continue;
        }
        int64_t g = gcd(a, b);
        if (g == 1) {
            cout << "0 0\n";
            continue;
        }
        int64_t ans = g;
        int64_t moves = 0;
        while (a % g != 0 || b % g != 0) {
            if (a < g) {
                swap(a, b);
            }
            a -= g;
            b += g;
            moves++;
        }
        cout << ans << " " << moves << "\n";
    }
    return 0;
}
