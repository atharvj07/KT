#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<string> f(n), ord;
    map<string, int> mp;
    for (int i = 0; i < n; i++) {
        cin >> f[i];
    }
    for (int i = 0; i < m; i++) {
        int a;
        cin >> a;
        ord.push_back(f[a - 1]);
        mp[f[a - 1]] = 1;
    }
    sort(ord.begin(), ord.end());
    string ans = "?";
    for (int i = 1; i < m; i++) {
        for (int j = 0; j < min(ord[i].size(), ord[i - 1].size()); j++) {
            if (ord[i][j] != ord[i - 1][j]) {
                if (ans.size() == 1)
                    ans = ord[i - 1].substr(0, j) + '?' + ord[i].substr(j + 1);
                else
                    ans = "?";
                break;
            }
        }
    }
    for (string ff : f) {
        int ok = 1;
        if (ff.size() != ans.size())
            ok = 0;
        else {
            for (int i = 0; i < ff.size(); i++) {
                if (ans[i] != '?' && ans[i] != ff[i]) {
                    ok = 0;
                    break;
                }
            }
        }
        if (ok && !mp[ff]) {
            ans = "?";
        }
    }
    if (ans == "?")
        cout << "No" << endl;
    else
        cout << "Yes" << endl << ans << endl;
}
