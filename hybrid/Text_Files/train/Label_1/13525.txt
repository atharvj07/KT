#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<vector<int>> knows(n, vector<int>(k));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < k; j++) {
            cin >> knows[i][j];
        }
    }
    bool possible = false;
    for (int i = 0; i < (1 << n); i++) {
        int count = 0;
        for (int j = 0; j < n; j++) {
            if ((i >> j) & 1) {
                for (int l = 0; l < k; l++) {
                    count += knows[j][l];
                }
            }
        }
        if (count <= k / 2) {
            possible = true;
            break;
        }
    }
    if (possible)
        cout << "YES" << endl;
    else
        cout << "NO" << endl;
    return 0;
}
