#include <bits/stdc++.h>
#define ll long long
#define mod 1'000'000'007
using namespace std;
vector<vector<ll>> g;
ll fact[1000005];
ll power(ll x, ll n) {
    if (n == 0) return 1ll;
    ll t = power(x, n >> 1);
    t *= t, t %= mod;
    if (n % 2) t *= (x % mod), t %= mod;
    return t;
}
void pre() {
    fact[0] = 1ll;
    for (ll i = 1; i < 1000005; i++) fact[i] = fact[i - 1] * i, fact[i] %= mod;
}
int main() {
    ll n, m, i, j, u;
    cin >> n >> m;
    g.resize(n + 1);
    unordered_map<ll, ll> ct[n + 1];
    unordered_map<ll, ll> tot;
    pre();
    for (i = 1; i <= n; i++) {
        cin >> u;
        while (u--) {
            ll z;
            cin >> z;
            ct[i][z]++;
            tot[z]++;
        }
    }
    ll ans = fact[m];
    for (const auto& x : tot) {
        ans *= power(fact[x.second], m - x.second), ans %= mod;
    }
    cout << ans << '\n';
}
