for t in range(int(input())):
    n=int(input())
    s=input()
    i,j=0,0
    a=[''for x in range(n)]
    b=[''for x in range(n)]
    for x in range(n):
        if s[x]=='1':
            a[x]=b[x]='('
            i+=1
            j+=1
        else:
            if i:
                a[x]=')'
                i-=1
            else:
                b[x]=')'
                j-=1
    if i==0 and j==0:
        print('YES')
        print(''.join(a))
        print(''.join(b))
    else: print('NO')
