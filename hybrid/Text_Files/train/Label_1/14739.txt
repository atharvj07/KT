#include<cstdio>
#include<vector>
#include<set>

int main(void){
    int n;
    scanf("%d",&n);
    std::vector<int> a(n);
    for(int i=0;i<n;i++){
        scanf("%d",&a[i]);
    }
    std::set<std::pair<int,int>> ans;
    for(int t=1;t<=n;t++){
        int score[2] = {0,0};
        int st = 0;
        for(int i=0;i<n;i++){
            score[a[i]-1]++;
            if(score[a[i]-1] == t){
                st++;
                score[0] = score[1] = 0;
            }
            if(st == n/t){
                ans.insert(std::make_pair(st,t));
            }
        }
    }
    printf("%d\n",(int)ans.size());
    for(std::set<std::pair<int,int>>::iterator it=ans.begin();it!=ans.end();it++){
        printf("%d %d\n",it->first,it->second);
    }
    return 0;
}
