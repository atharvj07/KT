#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
const int mod = 1e9 + 7;
int main() {
  int n;
  cin >> n;
  while (n--) {
    ll p, q, b;
    cin >> p >> q >> b;
    if (b == 10) {
      if (p % q == 0)
        cout << "Finite" << endl;
      else
        cout << "Infinite" << endl;
    } else {
      ll rem = p % q;
      unordered_map<ll, int> mp;
      ll ans = 0, pw = 1;
      while (rem && mp.find(rem) == mp.end()) {
        mp[rem] = ++ans;
        rem *= b;
        rem %= q;
        pw *= b;
        if (pw > q) pw %= q;
      }
      if (rem)
        cout << "Infinite" << endl;
      else
        cout << "Finite" << endl;
    }
  }
  return 0;
}
