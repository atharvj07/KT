#include <bits/stdc++.h>
using namespace std;
using ll = long long;

ll solve(ll v, ll k) {
  string s = to_string(v);
  int n = s.size();
  for (int i = 0; i + 1 < n; ++i) {
    if (k == 0) { break; }
    if (s[i] < s[i + 1]) {
      swap(s[i], s[i + 1]);
      --k;
    }
  }
  for (int i = n - 2; i >= 0; --i) {
    if (k == 0) { break; }
    if (s[i] > s[i + 1]) {
      swap(s[i], s[i + 1]);
      --k;
    }
  }
  ll ans = stoll(s);
  return ans;
}

int main() {
  ll v, k;
  cin >> v >> k;
  cout << solve(v, k) << endl;
}
