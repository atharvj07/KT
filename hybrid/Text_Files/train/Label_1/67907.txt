import heapq
n, p = list(map(int, input().split()))
a, b = [], []
for _ in range(n):
    ai, bi = list(map(int, input().split()))
    a.append(ai)
    b.append(bi)

# create heap[(b/a, idx)]
heap = []
for i in range(n):
    heapq.heappush(heap, (b[i] / a[i], i))

sum = 0
ans = 0
# find time when the earliest device power runs out
while heap:
    bpi, idx = heapq.heappop(heap)
    time_needed = bpi - sum
    if p * time_needed >= a[idx]:
        ans += time_needed
        sum += p * time_needed
    else:
        ans = -1
        break

print(ans)
