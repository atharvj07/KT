import java.util.Arrays;
import java.util.Scanner;

class Main {
  static final long MOD = 1000000007;

  public static void main(String[] args) throws Exception {
    Scanner s = new Scanner(System.in);
    int n = s.nextInt(), m = s.nextInt();
    int[] broken = new int[m];
    for (int i = 0; i < m; i++) broken[i] = s.nextInt();

    long[] dp = new long[n + 1];
    Arrays.fill(dp, -1);
    System.out.println(go(n, broken, dp));
  }

  static long go(int n, int[] broken, long[] dp) {
    if (n < 0) return 0;
    if (n == 0) return 1;
    if (dp[n] != -1) return dp[n];
    long ret = go(n - 1, broken, dp) + go(n - 2, broken, dp);
    for (int b : broken) {
      if (n == b) ret = 0;
    }
    return dp[n] = ret % MOD;
  }
}
