#include<bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, m;
    cin >> n >> m;
    vector<pair<string, string>> v(n);
    int mx_idx = 0;
    for (int i = 0; i < n; i++) {
        cin >> v[i].second;
        v[i].first = "";
        for (int j = 0; j < m; j++) {
            v[i].first += (j % 2 == 0 ? v[i].second[j] : (char)('z' - v[i].second[j] + 'a'));
        }
        if (v[i].first > v[mx_idx].first) mx_idx = i;
    }
    sort(v.begin(), v.end());
    for (int i = 0; i < n; i++) {
        if (v[i].first == v[mx_idx].first) cout << mx_idx + 1 << ' ';
        else cout << find(v.begin(), v.end(), make_pair(v[i].first, v[i].second)) - v.begin() + 1 << ' ';
    }
    cout << endl;
    return 0;
}
