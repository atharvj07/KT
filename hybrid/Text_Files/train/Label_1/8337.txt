#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  string s;
  getline(cin, s);

  vector<string> v;
  int start = 0;
  bool in_quote = false;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '"' && !in_quote) {
      start = i + 1;
      in_quote = true;
      continue;
    }
    if (s[i] == '"' && in_quote) {
      v.push_back(s.substr(start, i - start));
      start = i + 1;
      in_quote = false;
      continue;
    }
    if (s[i] == ' ' && !in_quote) {
      v.push_back(s.substr(start, i - start));
      start = i + 1;
      continue;
    }
  }
  if (start < s.size()) {
    v.push_back(s.substr(start));
  }

  for (auto &x : v) {
    cout << "<" << x << ">" << endl;
  }

  return 0;
}
