#include <iostream>

long long gcd(long long a, long long b) {
  while (b) {
    a %= b;
    std::swap(a, b);
  }
  return a;
}

int main() {
  int a, b;
  std::cin >> a >> b;
  int minab = std::min(a, b);
  long long g = 1;
  for (int i = 1; i <= minab; ++i) {
    g = gcd(g, i);
  }
  std::cout << g << std::endl;
  return 0;
}
