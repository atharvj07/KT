#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main(){
    ll n,a[300005],ans=0,cur=0; cin>>n;
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int i=1;i<=n;i++){
        cur = min(cur+1, a[i]/(i-ans));
        if(i-ans-1>a[i]) ans = i-1;
    }
    cout << ans << "\n";
    return 0;
}
