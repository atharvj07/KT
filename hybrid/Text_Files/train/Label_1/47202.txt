import java.io.*;
import java.util.*;
import java.math.*;
import static java.lang.Math.*;

public class Main {

	static void solve() {
		int N = nextInt(), M = nextInt(), K = nextInt(), S = nextInt();
		int P = nextInt(), Q = nextInt();
		List<Integer>[] G = new List[N+1];
		for (int i = 0; i <= N; i++) {
			G[i] = new ArrayList<>();
		}
		for (int i = 1, x; i <= K; i++) {
			x = nextInt();
		}
		int[] D = new int[N+1];
		for (int i = 1, a, b; i <= M; i++) {
			a = nextInt();
			b = nextInt();
			G[a].add(b);
			G[b].add(a);
			D[a]++;
			D[b]++;
		}
		int[] q = new int[N+1];
		int fi = 0, la = 1;
		q[la] = 1;
		while (fi < la) {
			int v = q[++fi];
			for (int nx : G[v]) {
				if (D[nx] > S || D[nx] == 0) {
					continue;
				}
				D[nx] = 0;
				q[++la] = nx;
			}
		}
		boolean[] vis = new boolean[N+1];
		int ans = 0;
		q[fi = la = 1] = 1;
		while (fi <= la) {
			int v = q[fi++];
			vis[v] = true;
			if (v == N) {
				break;
			}
			for (int nx : G[v]) {
				if (vis[nx] || D[nx] > S) {
					continue;
				}
				if (D[nx] == 0) {
					ans += Q;
				} else {
					ans += P;
				}
				q[++la] = nx;
			}
		}
		System.out.println(ans);
	}

	static int nextInt() {
		try {
			return Integer.parseInt(nextToken());
		} catch(Exception e) {
			throw new RuntimeException(e);
		}
	}

	static String nextToken() {
		try {
			while (!st.hasMoreTokens()) {
				st = new StringTokenizer(input.readLine());
			}
		} catch(Exception e) {
			throw new RuntimeException(e);
		}
		return st.nextToken();
	}

	static StringTokenizer st;
	static BufferedReader input;

	public static void main(String[] args) throws Exception {
		st = new StringTokenizer("");
		input = new BufferedReader(new InputStreamReader(System.in));
		solve();
	}
}
