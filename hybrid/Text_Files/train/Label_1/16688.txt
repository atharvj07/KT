#include<bits/stdc++.h>
using namespace std;
#define int long long
const int mod=1e9+7;
signed main(){
    string s;
    cin>>s;
    int n=s.size();
    int L=0,R=0,ans=0;
    for(int i=0;i<s.size();i++){
        if(s[i]=='[')L++;
        if(s[i]==']')R++;
        if(L==R){
            ans=max(ans,2*R);
        }
    }
    if(L!=R)ans=-1;
    cout<<ans<<'\n';
    return 0;
}
