import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    int d = scan.nextInt();
    int s = scan.nextInt();
    int num = s / d;
    int rem = s % d;
    if (num == 0 && s != 0) {
      System.out.println(-1);
      return;
    }
    if (rem == 0) {
      for (int i = 0; i < num; i++) {
        System.out.print("9");
      }
    } else {
      System.out.print(rem);
      for (int i = 0; i < num - 1; i++) {
        System.out.print("0");
      }
    }
  }
}
