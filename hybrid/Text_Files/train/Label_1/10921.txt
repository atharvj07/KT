#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<pair<int, int>> tiles(2 * n);
        for (int i = 0; i < n; i++) {
            int x1, y1, x2, y2;
            cin >> x1 >> y1 >> x2 >> y2;
            tiles[2 * i] = {x1, x2};
            tiles[2 * i + 1] = {y1, y2};
        }
        sort(tiles.begin(), tiles.end());
        bool possible = true;
        for (int i = 0; i < m / 2; i++) {
            if (tiles[i] != tiles[m - i - 1]) {
                possible = false;
                break;
            }
        }
        cout << (possible ? "YES" : "NO") << endl;
    }
    return 0;
}
