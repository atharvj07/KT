#include <cstdio>
#include <algorithm>

using namespace std;

int main() {
    int n, m; scanf("%d%d", &n, &m);
    int g[505][505];
    for (int i = 1; i <= n; i++) {
        char c[505]; scanf("%s", c + 1);
        for (int j = 1; j <= m; j++) 
            g[i][j] = (c[j] == '.' ? 0 : -1);
    }

    for (int i = 1; i <= n; i++) for (int j = 1; j <= m; j++) {
        if (g[i][j] == -1)
            for (int k = 1; k <= n + m; k++) {
                for (int di = max(1, i - k); di <= min(n, i + k); di++) {
                    if (g[di][j] == -1 && abs(i - di) <= k) continue;
                    if (g[di][j] == 0 && abs(i - di) > k) continue;
                    int ok = 1;
                    for (int dj = max(1, j - k); dj <= min(m, j + k); dj++) {
                        if (g[di][dj] == -1 && abs(j - dj) <= k) continue;
                        if (g[di][dj] == 0 && abs(j - dj) > k) continue;
                        ok = 0;
                    }
                    if (ok) {
                        for (int di = max(1, i - k); di <= min(n, i + k); di++) {
                            if (abs(i - di) <= k) g[di][j] = 0;
                        }
                        printf("1\n3 %d %d %d\n", di, j, k);
                        return 0;
                    }
                }
            }
    }

    printf("-1\n");
    return 0;
}
