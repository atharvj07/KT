#include<bits/stdc++.h>
using namespace std;

#define int long long

int32_t main(){
    int n;
    cin >> n;
    vector<int> v(n), pref(n + 1);
    map<int, int> mp;
    for(int i = 0; i < n; i++){
        int x;
        cin >> x;
        v[i] = x;
        pref[i + 1] = pref[i] + (x == x);
    }
 
    int ans = 0;
    for(int i = 0; i < n; i++){
        int val = v[i];
        ans += (pref[n] - pref[mp[val]]);
        mp[val] = i + 1;
    }
 
    cout << ans << endl;
    return 0;
}
