#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    sort(a.begin(), a.end());
    string ans = (k > n || k >= a.size() - (a.back() - a.front())) ? "NO" : "YES";
    cout << ans << endl;
    if (ans == "NO") return 0;
    vector<int> v;
    for (int i = 0; i < k; i++) {
        v.push_back(lower_bound(a.begin(), a.end(), a.front() + i) - a.begin() + 1);
    }
    for (auto &x : v) cout << x << " ";
    cout << endl;
    return 0;
}
