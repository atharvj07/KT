def solve(n, k):
    ans = []
    while n > 0 and k > 0:
        if n % 2 == 1:
            ans.append(0)
            n -= 1
            k -= 1
        else:
            x = n.bit_length() - 1
            n -= 2**x
            ans.append(x)
            k -= 1
    if n > 0:
        return []
    return ans[::-1]

n, k = map(int, input().split())
ans = solve(n, k)
if ans == []:
    print("No")
else:
    print("Yes")
    print(*ans)
