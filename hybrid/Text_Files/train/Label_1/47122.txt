import java.util.*;

public class Main {
    static int[] dist;
    static List<Integer>[] adj;
    static Queue<Integer> q = new LinkedList<>();

    static void add(int u, int v) {
        adj[u].add(v);
    }

    static void bfs(int n) {
        Arrays.fill(dist, -1);
        dist[1] = 0;
        q.add(1);
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : adj[u]) {
                if (dist[v] == -1) {
                    dist[v] = dist[u] + 1;
                    q.add(v);
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        adj = new List[n + 1];
        for (int i = 0; i < n + 1; i++) {
            adj[i] = new ArrayList<>();
        }
        dist = new int[n + 1];
        for (int i = 0; i < m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            add(u, v);
            add(v, u);
        }
        bfs(n);
        if (dist[n] == -1) {
            System.out.println(-1);
        } else {
            System.out.println(dist[n]);
            List<Integer> path = new ArrayList<>();
            int u = n;
            while (u != 1) {
                for (int v : adj[u]) {
                    if (dist[v] == dist[u] - 1) {
                        path.add(v);
                        u = v;
                        break;
                    }
                }
            }
            path.add(1);
            Collections.reverse(path);
            for (int v : path) {
                System.out.print(v + " ");
            }
        }
    }
}
