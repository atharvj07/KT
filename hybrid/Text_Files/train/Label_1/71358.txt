import sys
n,m=map(int,sys.stdin.readline().split())
a=list(map(int,sys.stdin.readline().split()))
for i in range(n): a[i]-=1
p=list(range(n))
for _ in range(m):
 x,y=map(int,sys.stdin.readline().split())
 x-=1; y-=1
 p[x],p[y]=p[y],p[x]
inv=0
for i in range(n):
 for j in range(i+1,n):
  inv+=(a[p[i]]>a[p[j]])
print(inv/n/(n-1))
