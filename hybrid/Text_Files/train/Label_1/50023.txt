import java.util.*;

public class Game {

    static class Cell {
        int player;
        boolean visited;

        public Cell(int player) {
            this.player = player;
            this.visited = false;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int p = scanner.nextInt();

        int[] speeds = new int[p];
        for (int i = 0; i < p; i++) {
            speeds[i] = scanner.nextInt();
        }

        Cell[][] grid = new Cell[n][m];
        for (int i = 0; i < n; i++) {
            String row = scanner.next();
            for (int j = 0; j < m; j++) {
                char cell = row.charAt(j);
                if (cell == '.') {
                    grid[i][j] = new Cell(0);
                } else if (cell == '#') {
                    grid[i][j] = null;
                } else {
                    grid[i][j] = new Cell(cell - '0');
                }
            }
        }

        int[] controlledCells = new int[p];
        for (int i = 0; i < p; i++) {
            controlledCells[i] = expand(grid, speeds[i], i);
        }

        for (int controlledCell : controlledCells) {
            System.out.print(controlledCell + " ");
        }
        System.out.println();
    }

    private static int expand(Cell[][] grid, int speed, int player) {
        int controlledCells = 0;
        for (int i = 0; i < grid.length; i++) {
            for (int j = 0; j < grid[i].length; j++) {
                if (grid[i][j] != null && grid[i][j].player == player && !grid[i][j].visited) {
                    controlledCells += expandFromCell(grid, i, j, speed, player);
                }
            }
        }
        return controlledCells;
    }

    private static int expandFromCell(Cell[][] grid, int i, int j, int speed, int player) {
        grid[i][j].visited = true;
        int controlledCells = 1;

        for (int di = -speed; di <= speed; di++) {
            for (int dj = -speed; dj <= speed; dj++) {
                int ni = i + di;
                int nj = j + dj;
                if (ni >= 0 && ni < grid.length && nj >= 0 && nj < grid[ni].length && grid[ni][nj] != null && grid[ni][nj].player == 0 && !grid[ni][nj].visited) {
                    controlledCells += expandFromCell(grid, ni, nj, speed, player);
                }
            }
        }

        return controlledCells;
    }
}
