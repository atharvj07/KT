from math import gcd

def find_beautiful_permutation(n, k):
    if k == n:
        return -1
    permutation = [i for i in range(1, n + 1)]
    for i in range(1, n + 1):
        if k == 0:
            break
        if gcd(i, permutation[i - 1]) > 1:
            k -= 1
        elif i < n:
            permutation[i - 1], permutation[i] = permutation[i], permutation[i - 1]
            if gcd(i, permutation[i - 1]) > 1:
                k -= 1
    if k == 0:
        return permutation
    else:
        return -1

n, k = map(int, input().split())
result = find_beautiful_permutation(n, k)
if result == -1:
    print(-1)
else:
    print(' '.join(map(str, result)))
