#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<pair<int, int>> a(n), b(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i].first;
        }
        for (int i = 0; i < n; i++) {
            cin >> a[i].second;
        }
        b = a;
        sort(a.begin(), a.end());
        sort(b.begin(), b.end());
        vector<int> ans(n, 1);
        for (int i = 0; i < n; i++) {
            if (a[i] != b[i]) {
                ans[i] = 0;
            }
        }
        for (int i = 0; i < n; i++) {
            cout << ans[i];
        }
        cout << endl;
    }
    return 0;
}
