import sys
input=sys.stdin.readline
q=int(input())
for i in range(q):
    n=int(input())
    s=input().strip()
    t=input().strip()
    ans=n
    t=t[n//2:]+t[:n//2]
    for j in range(n):
        if s[j]!=t[j]:
            if t[j]==s[j-1]:
                ans=min(ans,j)
            if t[j]==s[(j+1)%n]:
                ans=min(ans,j+1)
    if ans==n: print(-1)
    else: print(ans)
