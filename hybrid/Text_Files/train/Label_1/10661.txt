#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 105;
ll a[maxn];
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        bool f = false;
        for (int i = 1; i <= n; i++) {
            cin >> a[i];
            if (a[i] % 2 == 1) f = true;
        }
        if (n % 2 == f) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
