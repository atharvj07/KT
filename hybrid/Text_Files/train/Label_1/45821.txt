import java.util.Scanner;

public class Main {
    static int MOD = 100000000;
    static long[][] dp = new long[101][101];

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n1 = in.nextInt();
        int n2 = in.nextInt();
        int k1 = in.nextInt();
        int k2 = in.nextInt();
        in.close();

        System.out.println(solve(n1, n2, k1, k2));
    }

    static long solve(int n1, int n2, int k1, int k2) {
        if (n1 == 0 && n2 == 0) {
            return 1;
        }
        if (n1 < 0 || n2 < 0) {
            return 0;
        }
        if (dp[n1][n2] != 0) {
            return dp[n1][n2];
        }

        long ans = 0;
        if (n1 > 0) {
            ans = (ans + solve(n1 - 1, n2, k1 - 1, k2)) % MOD;
            if (k1 > 1) {
                ans = (ans + solve(n1 - 1, n2, k1, k2)) % MOD;
            }
        }
        if (n2 > 0) {
            ans = (ans + solve(n1, n2 - 1, k1, k2 - 1)) % MOD;
            if (k2 > 1) {
                ans = (ans + solve(n1, n2 - 1, k1, k2)) % MOD;
            }
        }

        dp[n1][n2] = ans;
        return ans;
    }
}
