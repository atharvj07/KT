#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;

const int N = 100005;
ll a[N], b[N];

int main() {
    ll n, s;
    cin >> n >> s;

    vector<pair<ll, ll>> v;
    for (int i = 0; i < n; i++) {
        ll x, y, z;
        cin >> x >> y >> z;
        v.push_back(make_pair(x, y - z));
    }
    sort(v.begin(), v.end());

    ll ans = 0;
    for (int i = 0; i < n; i++) {
        if (v[i].second >= 0) {
            ans += v[i].first * v[i].second;
        } else {
            ll cnt = min(s, -v[i].first / v[i].second);
            s -= cnt;
            ans += cnt * v[i].first;
        }
    }
    cout << ans << endl;
    return 0;
}
