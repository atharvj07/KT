#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

bool cmp(const pair<long long, long long> &a, const pair<long long, long long> &b)
{
    return a.first > b.first;
}

int main()
{
    int n;
    cin >> n;
    vector<pair<long long, long long>> teams(n);
    long long t1, w1;
    cin >> t1 >> w1;
    for (int i = 0; i < n; i++)
    {
        long long t, w;
        cin >> t >> w;
        teams[i].first = t - w;
        teams[i].second = w;
    }
    sort(teams.begin(), teams.end(), cmp);
    long long sum = t1;
    int ans = 0;
    for (int i = 0; i < n; i++)
    {
        if (sum <= teams[i].second)
        {
            ans++;
            sum += w1;
        }
    }
    cout << ans + 1 << endl;
    return 0;
}
