import sys
input = sys.stdin.readline

n, m = map(int, input().split())
INF = 2e9
graph = [[] for i in range(n + 1)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

cities = list(map(int, input().split()))
distance = [INF] * (n + 1)
visited = [0] * (n + 1)

def dfs(node, dist):
    visited[node] = 1
    distance[node] = dist
    for u in graph[node]:
        if visited[u] == 0:
            dfs(u, dist + 1)

dfs(1, 0)

min_city = 1
min_dist = INF

for city in cities:
    if distance[city] < min_dist:
        min_dist = distance[city]
        min_city = city

print(min_city)
print(min_dist)
