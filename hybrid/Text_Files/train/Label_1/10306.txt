#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, d;
  cin >> n >> d;
  vector<string> schedule(d);
  for (int i = 0; i < d; i++) {
    cin >> schedule[i];
  }
  int ans = 0, cur = 0;
  for (int i = 0; i < d; i++) {
    bool ok = true;
    for (int j = 0; j < n; j++) {
      if (schedule[i][j] == '0') {
        ok = false;
      }
    }
    if (ok) {
      cur++;
    } else {
      ans = max(ans, cur);
      cur = 0;
    }
  }
  ans = max(ans, cur);
  cout << ans << endl;
  return 0;
}
