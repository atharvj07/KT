import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] weights = new int[n];
        for (int i = 0; i < n; i++) {
            weights[i] = scanner.nextInt();
        }

        int minRides = -1;
        long numWays = 0;
        for (int c1 = 0; c1 <= n && minRides == -1; c1++) {
            for (int c2 = 0; c2 <= n && minRides == -1; c2++) {
                if (c1 + c2 > n || c1 * 100 + c2 * 50 > k) continue;

                int[] used = new int[n];
                int rides = 0;
                while (true) {
                    rides++;
                    int weight = 0;
                    for (int i = 0; i < n; i++) {
                        if (used[i] == 0 && (c1 > 0 || weights[i] == 50) && weight + weights[i] <= k) {
                            used[i] = 1;
                            weight += weights[i];
                            if (c1 > 0) c1--;
                            if (weights[i] == 50) c2--;
                        }
                    }
                    if (weight == 0) break;
                }

                if (c1 == 0 && c2 == 0) {
                    minRides = rides;
                    numWays = 1;
                } else if (minRides == -1 || minRides == rides) {
                    numWays += choose(c1 + c2, c1) % 1000000007;
                    minRides = rides;
                }
            }
        }

        System.out.println(minRides);
        System.out.println(numWays % 1000000007);
    }

    public static long choose(int n, int k) {
        long result = 1;
        for (int i = 0; i < k; i++) {
            result = result * (n - i) / (i + 1);
        }
        return result;
    }
}
