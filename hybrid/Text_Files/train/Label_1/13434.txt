#include <iostream>
using namespace std;
long long ans=0;
int main(){
	long long a,b,c,l;
	cin>>a>>b>>c>>l;
	if((a+b+c)>l) cout<<ans;
	else{
		long long k=l-a-b-c+1;
		ans+=k*(k+1)*(k+2)/6;
		if(a-b+c<=l) ans+=(k+1)*(k+2)/2;
		if(a-c+b<=l) ans+=(k+1)*(k+2)/2;
		if(c-a+b<=l) ans+=(k+1)*(k+2)/2;
		cout<<ans;
	}
	return 0;
}
