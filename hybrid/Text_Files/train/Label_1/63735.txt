import sys
input = sys.stdin.readline

n, q = map(int, input().split())
c = list(map(int, input().split()))
l = [0] * (n + 1)
for i in range(q):
    x, y = map(int, input().split())
    for j in range(x, n + 1):
        if c[j] == c[x - 1]:
            l[j] = max(l[j], l[x - 1])
        else:
            break
    m = c[x - 1]
    for j in range(x, n + 1):
        if c[j] == m:
            l[j] = l[j - 1] + 1
    c[x - 1] = y
l[n] = min([i for i in l])
for i in l[1:]:
    print(i)
