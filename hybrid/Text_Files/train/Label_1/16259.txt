#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    long long ans = n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) {
        if (a[i] >= 0) {
            ans = i;
            a[i] = -1;
        }
        for (int j = i + 1; j < n; ++j) {
            if (a[j] <= 0) continue;
            if (i + a[i] >= j) {
                ans--;
                a[j] = -1;
            }
        }
    }
    cout << ans << endl;
}
