import java.util.*;

public class Main {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int m = sc.nextInt();
		int[][] a = new int[n][m];
		int[][] b = new int[n][m];
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				a[i][j] = sc.nextInt();
			}
		}
		int ans = 0;
		ArrayList<Pair> ops = new ArrayList<>();
		for (int i = 0; i < n - 1; i++) {
			for (int j = 0; j < m - 1; j++) {
				if (a[i][j] == a[i + 1][j] && a[i][j] == a[i][j + 1] && a[i][j] == a[i + 1][j + 1]) {
					if (a[i][j] == 0) {
						continue;
					} else {
						ops.add(new Pair(i + 1, j + 1));
						b[i][j] = 1;
						b[i][j + 1] = 1;
						b[i + 1][j] = 1;
						b[i + 1][j + 1] = 1;
						ans++;
					}
				} else {
					if (a[i][j] == 1 || a[i + 1][j] == 1 || a[i][j + 1] == 1 || a[i + 1][j + 1] == 1) {
						ans = -1;
						break;
					}
				}
			}
		}
		if (ans == -1) {
			System.out.println(-1);
		} else {
			System.out.println(ans);
			for (Pair op : ops) {
				System.out.println(op.x + " " + op.y);
			}
		}
		sc.close();
	}

	static class Pair {
		int x, y;

		Pair(int x, int y) {
			this.x = x;
			this.y = y;
		}
	}
}
