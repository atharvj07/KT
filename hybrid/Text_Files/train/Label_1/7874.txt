#include <bits/stdc++.h>
using namespace std;
#define int long long

vector<int> solve(vector<int> &t, int n, int M) {
  vector<int> ans(n);
  stack<pair<int, int>> s;
  for (int i = n - 1; i >= 0; i--) {
    int need = (n - i - 1) * t[i];
    while (!s.empty() && s.top().first + t[i] <= M) {
      need -= s.top().second * t[i];
      s.pop();
    }
    s.push({(M - need) / t[i], 1});
    ans[i] = s.size() - 1;
  }
  return ans;
}

int32_t main() {
  int n, M;
  cin >> n >> M;
  vector<int> t(n);
  for (auto &ti : t) cin >> ti;
  vector<int> ans = solve(t, n, M);
  for (auto &a : ans) cout << a << " ";
}
