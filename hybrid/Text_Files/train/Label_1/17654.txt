#include<bits/stdc++.h>
using namespace std;
int main() {
  cin.tie(0);
  ios_base::sync_with_stdio(0);
  string s;
  double ans = 0;
  cin >> s;
  for (int l = 0; l < s.length(); l++) {
    string name, price;
    while (l < s.length() && s[l] != '0' && s[l] != '1' && s[l] != '2' && s[l] != '3' && s[l] != '4' && s[l] != '5' && s[l] != '6' && s[l] != '7' && s[l] != '8' && s[l] != '9') {
      name += s[l];
      l++;
    }
    while (l < s.length() && (s[l] == '0' || s[l] == '1' || s[l] == '2' || s[l] == '3' || s[l] == '4' || s[l] == '5' || s[l] == '6' || s[l] == '7' || s[l] == '8' || s[l] == '9' || s[l] == '.')) {
      price += s[l];
      l++;
    }
    stringstream ss;
    double dd;
    ss << price;
    ss >> dd;
    ans += dd;
  }
  cout << fixed << setprecision(2) << ans;
}
