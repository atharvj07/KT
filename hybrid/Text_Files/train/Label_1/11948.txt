#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) cin >> b[i];
    bool valid = true;
    for (int i = 0; i < n; ++i) {
      if (a[i] != b[i]) {
        int j = i;
        while (j + 1 < n && a[j + 1] == b[i]) j++;
        if (j + 1 == n) {
          valid = false;
          break;
        }
        int k = j + 1;
        while (k < n && a[k] != b[i]) k++;
        if (k == n) {
          valid = false;
          break;
        }
        for (int l = i + 1; l <= k; ++l) {
          if (a[l] != b[(l + j - i) % n]) {
            valid = false;
            break;
          }
        }
        if (!valid) break;
        i = k;
      }
    }
    cout << (valid ? "Yes" : "No") << endl;
  }
  return 0;
}
