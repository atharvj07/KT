#include <bits/stdc++.h>
using namespace std;
#define N 1<<10

int n, m, a[N], b[N], c[N], d[N];
int main() {
  string s, t;
  cin >> s >> t;
  n = s.size(), m = t.size();
  for (int i = 0; i < n; i++) a[i] = s[i] - '0';
  for (int i = 0; i < m; i++) b[i] = t[i] - '0';
  for (int i = 0; i < n - 1; i++) {
    if (a[i] == a[i + 1]) c[i] = 2;
    else c[i] = 1, d[i + 1] = 1;
  }
  for (int i = 0; i < n - 1; i++) {
    if (c[i] + d[i] != b[i] + b[i + 1]) {
      cout << "NO";
      return 0;
    }
  }
  cout << "YES";
  return 0;
}
