#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &i : a) cin >> i;
  int ans = 0, cur = 0;
  for (int i = 0; i < n; i++) {
    if (cur < a[i]) {
      cur += (!ans);
      ans++;
    }
    cur -= a[i];
  }
  if (cur != 0) ans = -1;
  cout << ans << '\n';
}
