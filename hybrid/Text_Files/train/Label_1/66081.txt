import numpy as np

def check(x, y, a):
    if abs(x - a[0]) == 1 or abs(y - a[1]) == 1:
        return True
    return False

def solve(n):
    if n % 4 != 0:
        return -1

    a = np.arange(1, n ** 2 + 1).reshape(n, n)

    for i in range(n):
        for j in range(n):
            if i + j >= n:
                a[i, j] ^= a[j, i]

    for i in range(n):
        for j in range(n):
            if i % 2 == j % 2:
                a[i, j] ^= a[i, n - 1 - j]

    return a

t = int(input())

for _ in range(t):
    n = int(input())

    ans = solve(n)

    if type(ans) is not int:
        for i in ans:
            print(*i)
    else:
        print(-1)
