#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> v(n), t(n);
  for (int i = 0; i < n; i++) cin >> v[i];
  for (int i = 0; i < n; i++) cin >> t[i];
  for (int i = 0; i < n; i++) cout << (v[i] > t[i] ? v[i] - t[i] : 0) << ' ';
  cout << endl;
  return 0;
}
