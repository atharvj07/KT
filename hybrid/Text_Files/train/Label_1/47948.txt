import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int k = 0;
        for (int i = 1; i <= n; i++) {
            if (n % i == 0) {
                boolean f = true;
                for (int j = 0; j < n; j++) {
                    if (a[j] != a[j % i]) {
                        f = false;
                        break;
                    }
                }
                if (f) {
                    k = Math.max(k, n / i);
                }
            }
        }
        System.out.println(k);
    }
}
