#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    pair<int, int> mx = {-1, -1};
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
        mx = max(mx, {v[i].second, v[i].first});
    }
    int ans = 0, cnt = 0, cur = 1;
    for (int year = 1; year <= mx.first; year++) {
        for (int i = 0; i < n; i++) {
            if (v[i].first <= year && v[i].second >= year) cnt++;
        }
        if (cnt > ans) {
            ans = cnt;
            cur = year;
        }
        cnt = 0;
    }
    cout << cur << " " << ans << endl;
    return 0;
}
