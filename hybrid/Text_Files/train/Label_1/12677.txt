#include <bits/stdc++.h>
using namespace std;
const int M = 100;
bool t[M + 1], g = 0;
int main() {
  int n, m, a, b;
  cin >> n >> m;
  t[0] = 1;
  for (int i = 0; i < n; ++i) {
    cin >> a >> b;
    if (t[a])
      for (int j = a; j <= b; ++j) t[j] = 1;
  }
  cout << (t[m] ? "YES" : "NO") << endl;
}
