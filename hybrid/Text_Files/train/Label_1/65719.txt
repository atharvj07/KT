n, k = map(int, input().split())
s = input()

# Count occurrences of each letter
letter_counts = [0] * k
for char in s:
    letter_counts[ord(char) - ord('A')] += 1

# Find the minimum occurrence among the first k letters
min_count = min(letter_counts)

# The length of the longest good subsequence is k * min_count
print(min_count * k)
