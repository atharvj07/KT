#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int gcd(int a, int b) {
    if (b == 0) return a;
    return gcd(b, a % b);
}

int main() {
    int n, ans = 0; cin >> n;
    vector<int> v(n);
    for (int& x : v) cin >> x;
    sort(v.begin(), v.end());
    for (int l = 0, r = 0; r < n; r++) {
        if (l != r && gcd(v[l], v[r]) == 1) l++;
        ans = max(ans, r - l + 1);
    }
    cout << ans << '\n';
}
