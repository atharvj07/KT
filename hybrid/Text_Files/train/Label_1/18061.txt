#include <iostream> 
#include <vector> 
#include <algorithm> 
#include <cmath> 

using namespace std; 

int main() 
{ 
    int n, x0, y0; 
    cin >> n >> x0 >> y0; 
    vector<pair<long long, long long>> v; 
    for (int i = 0; i < n; i++) 
    { 
        int x, y; 
        cin >> x >> y; 
        v.push_back(make_pair(x - x0, y - y0)); 
    } 
    sort(v.begin(), v.end()); 
    int cnt = 0; 
    for (int i = 0; i < n; i++) 
    { 
        bool ver = true; 
        for (int j = i + 1; j < n; j++) 
        { 
            if (v[j].first * v[i].second != v[i].first * v[j].second) 
            { 
                ver = false; 
                break; 
            } 
        } 
        if (ver == true) 
        { 
            cnt++; 
            i = n; 
        } 
    } 
    cout << cnt + 1; 
}
