#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int q, d;
        cin >> q >> d;
        for (int i = 0; i < q; i++) {
            int a;
            cin >> a;
            if (to_string(a).find(d + '0') != string::npos) {
                cout << "YES\n";
            } else {
                cout << "NO\n";
            }
        }
    }
    return 0;
}
