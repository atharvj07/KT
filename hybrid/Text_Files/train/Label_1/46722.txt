import java.util.Arrays;
import java.util.Scanner;

public class AlarmClocks {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[] alarms = new int[n];
        for (int i = 0; i < n; i++) {
            alarms[i] = scanner.nextInt();
        }

        Arrays.sort(alarms);

        int minAlarmsToTurnOff = n;
        for (int i = 0; i < n - m + 1; i++) {
            int count = 0;
            for (int j = i; j < i + m; j++) {
                if (alarms[j] - alarms[i] < m) {
                    count++;
                }
            }
            minAlarmsToTurnOff = Math.min(minAlarmsToTurnOff, n - count);
        }

        System.out.println(minAlarmsToTurnOff);
    }
}
