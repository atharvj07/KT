n = int(input())
wh_pairs = [tuple(map(int, input().split())) for _ in range(n)]

def min_photo_area(n, wh_pairs):
    wh_pairs.sort(key=lambda x: -max(x))
    best_area = float('inf')
    for i in range((n + 1) // 2, n + 1):
        heights = sorted([h if idx < i else w for idx, (w, h) in enumerate(wh_pairs)], reverse=True)
        width = sum(w for w, h in wh_pairs)
        height = heights[0]
        best_area = min(best_area, width * height)
    return best_area

print(min_photo_area(n, wh_pairs))
