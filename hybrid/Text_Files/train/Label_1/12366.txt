#include<iostream>
#include<cstdio>
using namespace std;
const int mod = 1000000007;
int main()
{
    long long x, y, n;
    cin >> x >> y >> n;
    if (n & 1)
    {
        cout << (x % mod + y % mod) % mod << endl;
    }
    else
    {
        if((x + y) % 2)
        {
            cout << ((x % mod - y % mod) % mod + mod) % mod << endl;
        }
        else
        {
            cout << (x % mod - y % mod) % mod << endl;
        }
    }
    return 0;
}
