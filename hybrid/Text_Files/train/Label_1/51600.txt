import java.util.Scanner;

public class CapsLock {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String word = scanner.nextLine();

        if (isAllCaps(word) || isFirstLowerRestCaps(word)) {
            word = word.toLowerCase();
            word = Character.toUpperCase(word.charAt(0)) + word.substring(1);
        }

        System.out.println(word);
    }

    private static boolean isAllCaps(String word) {
        for (char ch : word.toCharArray()) {
            if (Character.isLowerCase(ch)) {
                return false;
            }
        }
        return true;
    }

    private static boolean isFirstLowerRestCaps(String word) {
        if (word.length() == 1) {
            return false;
        }
        return Character.isLowerCase(word.charAt(0)) && isAllCaps(word.substring(1));
    }
}
