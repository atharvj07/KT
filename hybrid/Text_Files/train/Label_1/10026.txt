#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 1e5 + 5;
vector<string> v[maxn];
unordered_map<string, int> mp;
int t, n, m, ansr = 1e9, ansl = 1e9;
string s;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> s, mp[s] = i, v[i].push_back(s);
  cin >> m;
  string a, b;
  for (int i = 0; i < m; i++) {
    cin >> a >> b;
    transform(a.begin(), a.end(), a.begin(), ::tolower);
    transform(b.begin(), b.end(), b.begin(), ::tolower);
    v[mp[a]].push_back(b), v[mp[b]].push_back(a);
  }
  function<void(int, string)> dfs = [&](int x, string s) {
    for (auto p : v[x]) {
      if (s.back() == p.back() && s.size() + p.size() - 1 >= ansl) continue;
      int r = 0;
      for (int i = 0; i < p.size(); i++) r += (p[i] == 'r' || p[i] == 'R');
      if (r < ansr || (r == ansr && s.size() + p.size() - 1 < ansl)) {
        if (p.size() + s.size() - 1 < ansl || r < ansr) ansr = r, ansl = s.size() + p.size() - 1;
        dfs(mp[p], s + p);
      }
    }
  };
  for (int i = 0; i < n; i++) dfs(i, v[i][0]);
  cout << ansr << ' ' << ansl << endl;
}
