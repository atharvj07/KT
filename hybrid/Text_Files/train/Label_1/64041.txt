t = int(input().strip())

for _ in range(t):
    n = int(input().strip())
    c = list(map(int, input().strip().split()))
    a = list(map(int, input().strip().split()))
    b = list(map(int, input().strip().split()))

    max_cycle = 0
    current_length = 0

    for i in range(1, n):
        if a[i] != b[i]:
            current_length = abs(a[i] - b[i]) + 2  # +2 for both ends of the chain
        else:
            current_length = 0  # Reset if a[i] == b[i], as we start a new potential cycle

        current_length += c[i] - 1  # Add the length of the current chain
        
        if i != 1:
            current_length += max(0, c[i-1] - max(a[i], b[i]) + min(a[i], b[i]) - 1)  # Add the length from the previous chain, if any

        max_cycle = max(max_cycle, current_length)

    print(max_cycle)
