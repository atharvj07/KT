import sys
from collections import Counter

tests = int(input())

for _ in range(tests):
    N = int(input())
    a = list(map(int, input().split()))
    c = Counter(a)
    k = len(set(a))
    if k > 1:
        if len(c) > 2:
            k -= 1
        elif list(c.values())[0] != list(c.values())[1]:
            k = -1
    print(k)
