#include <iostream>
using namespace std;
int a[105],b[105];
int n;
int main()
{
    int i,j,k=0;
    cin>>n;
    for(i=1;i<=n;++i)
    {
        cin>>a[i];
    }
    for(i=1;i<=n;++i)
    {
        b[i]=1;
        for(j=i+1;j<=n;++j)
        {
            if(a[i]<a[j])b[i]=0;
        }
        k+=b[i];
    }
    cout<<k;
    return 0;
}
