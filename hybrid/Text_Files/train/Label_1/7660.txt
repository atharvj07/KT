#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string a, s;
    cin >> a >> s;
    int n = a.size(), m = s.size(), carry = 0;
    string b(m, '0');
    for (int i = n - 1, j = m - 1; i >= 0; i--, j--) {
      int x = a[i] - '0' + b[j] - '0' + carry;
      carry = x / 10;
      b[j] = x % 10 + '0';
      if (j == 0 && carry) {
        b = char(carry + '0') + b;
      }
    }
    if (b[0] == '0' && b.size() > 1)
      b.erase(0, 1);
    if (b == s)
      cout << b << endl;
    else
      cout << "-1\n";
  }
  return 0;
}
