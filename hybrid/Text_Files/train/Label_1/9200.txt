#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        ll n;
        cin>>n;
        ll cnt=0;
        for(ll a=1;a<=n/2;a++)
        {
            ll b=n-a,x=a,y=b,c=0;
            while(x or y or c)
            {
                ll p=x%10+y%10+c;
                if(p>=10)
                {
                    c=1;
                    p-=10;
                }
                else c=0;
                x/=10;
                y/=10;
                if(c and !(x or y))
                {
                    cnt++;
                    break;
                }
            }
        }
        cnt*=2;
        cout<<cnt<<endl;
    }
}
