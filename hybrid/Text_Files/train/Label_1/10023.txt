#include <algorithm>
#include <iostream>
#include <map>
#include <set>
#include <string>
#include <vector>
using namespace std;
string toLower(string s) {for(auto& c: s) c = tolower(c); return s;}
int main() {
  int n, m;
  cin >> m;
  vector<string> v(m);
  for (string& s : v) { cin >> s; }
  cin >> n;
  map<string, set<string>> d;
  for (int i = 0; i < n; i++) {
    string x, y;
    cin >> x >> y;
    d[toLower(x)].insert(toLower(y));
  }
  int ans = 0, len = 0;
  set<string> vis;
  for (string& s : v) {
    string low = toLower(s);
    if (vis.count(low)) continue;
    vis.insert(low);
    len += s.length();
    if (d.count(low) && !s.empty()) {
      s = *d[low].begin();
      low = toLower(s);
    }
    int cnt = count(low.begin(), low.end(), 'r');
    ans += cnt;
  }
  cout << ans << " " << len << endl;
}
