#include <bits/stdc++.h>
using namespace std;
#define ll long long
typedef pair<ll, ll> p;
#define x first
#define y second
const int N = 2e5 + 10;
const ll inf = 1e17;
ll ans, n;
p a[N];
ll ccw(p a, p b, p c) { return a.x * b.y - a.y * b.x + b.x * c.y - b.y * c.x + c.x * a.y - c.y * a.x; }
int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i].x >> a[i].y;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            for (int k = j + 1; k < n; k++) {
                if (ccw(a[i], a[j], a[k])) ans++;
            }
        }
    }
    cout << ans << '\n';
}
