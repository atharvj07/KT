import sys

n = int(input())
a = list(map(int, input().split()))
stack = []
cur = 1
ans = []
for x in a:
    if not stack or stack[-1][0] == x:
        stack.append([x, cur])
    else:
        ans.append(stack.pop())
        stack.append([x, cur])
    cur += 1

if len(stack) % 2 == 0:
    ans.extend(stack[::2])
else:
    print(-1)
    sys.exit(0)

print(len(ans))
for l, r in ans:
    print(l, r)
