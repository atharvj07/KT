#include <iostream>
#include <set>
#include <vector>
using namespace std;
const int maxn = 1e5 + 5;

int main() {
	int n;
	cin >> n;
	vector<int> last(maxn, -1);
	bool ok = true;
	while (n--) {
		int x, k;
		cin >> x >> k;
		if (last[k] <= x)
			ok = false;
		last[k] = x;
	}
	cout << ((ok) ? "YES" : "NO") << '\n';
}
