#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k, pos = -1000000, l, r;
        cin >> n >> k;
        for (int i = 1; i <= k; ++i) {
            int x;
            cin >> x;
            if (pos != x - 1) {
                if (pos != -1000000) {
                    if (r - l + 1 > n - 1) {
                        k = 1;
                        pos = x;
                        break;
                    }
                }
                pos = x - 1;
            }
            pos++;
            l = min(l, pos);
            r = max(r, pos);
        }
        if (r - l + 1 > n - 1) {
            cout << 1 << endl;
            continue;
        }
        int ans = 0;
        while (r - l + 1 < n) {
            ans++;
            pos -= ans;
            r = max(r, pos);
        }
        cout << ans << endl;
    }
}
