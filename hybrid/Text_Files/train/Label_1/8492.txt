#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (a[i] != b[i] && a[j] != b[j]) {
                if (a[i] != a[j] && b[i] != b[j]) {
                    ans += 2;
                } else {
                    ans += 1;
                }
            }
        }
    }
    cout << ans << endl;
}
