n = int(input())
s = input()
a = list(map(int, input().split()))
dp = [[-1] * 27 for _ in range(n + 10)]
mod = int(1e9) + 7
def calc(pos, last):
    if pos >= n:
        if last <= 1:
            return 1
        else:
            return 0
    ret = dp[pos][last]
    if ret != -1:
        return ret
    ret = 0
    if last > 0:
        ret = calc(pos + 1, last)
    ret += calc(pos + 1, ord(s[pos]) - ord('a') + 1)
    ret %= mod
    dp[pos][last] = ret
    return ret

print(calc(0, 0))
print(max(a))
print(1)
