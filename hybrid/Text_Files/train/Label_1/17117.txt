#include <bits/stdc++.h>
using namespace std;
const int N = 5e5 + 5, M = 5e5 + 5;
char c[N];
int n, m, p, x, y, z, a[N];
int main() {
  cin >> n >> m >> p;
  cin >> (c + 1);
  while (m--) {
    cin >> c[0], x = p, y = p;
    if (c[0] == 'L')
      while (x > 1 && c[x] == ')') x--;
    if (c[0] == 'R')
      while (y <= n && c[y] == '(') y++;
    if (c[0] == 'D')
      while (x > 1 && y <= n && c[x] == ')' && c[y] == '(') { x--; y++; }
    p = min(x, y);
    if (c[0] != 'D') p++;
  }
  for (int i = 1; i <= n; i++) {
    if (a[i] == 0) a[i] = 1, putchar(c[i]);
  }
  return 0;
}
