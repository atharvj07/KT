#include <bits/stdc++.h>
using namespace std;
struct node {
    int val, mp, cnt;
};
bool cmp(const node& a, const node& b) {
    if (a.mp != b.mp) return a.mp < b.mp;
    return a.cnt > b.cnt;
}
int main() {
    int n, ans = 0;
    cin >> n;
    vector<node> sal(n);
    for (auto& x : sal) {
        cin >> x.cnt;
        x.mp = 0;
        int cur = -1e9;
        for (int j = 0; j < x.cnt; ++j) {
            int a;
            cin >> a;
            if (a > cur) cur = a;
        }
        x.mp = cur;
    }
    sort(sal.begin(), sal.end(), cmp);
    int l = 0, r = 0;
    for (auto& x : sal) {
        ans += (x.mp - x.val) * x.cnt;
        r += x.cnt;
    }
    for (auto& x : sal) {
        ans -= (x.mp - x.val) * x.cnt;
        r -= x.cnt;
        l += x.cnt;
        ans += (sal[r].mp - x.mp) * l;
    }
    cout << ans << endl;
    return 0;
}
