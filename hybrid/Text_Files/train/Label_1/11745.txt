#include <iostream>
#include <vector>

typedef long long ll;

int main() {
    ll a;
    std::cin >> a;
    std::string s;
    std::cin >> s;
    ll n = s.length();
    std::vector<ll> sum(n + 1);

    for (ll i = n - 1; i >= 0; i--) {
        sum[i] = (s[i] - '0') * (s[i] - '0') + sum[i + 1];
    }
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
        for (ll j = i; j < n; j++) {
            if (sum[i] + sum[j + 1] - sum[j + 1] == a) {
                ans++;
            }
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
