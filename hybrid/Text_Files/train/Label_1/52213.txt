import java.util.*;
import java.lang.*;

class Main {
    static int[] pos,isFlip;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        pos = new int[n+1];
        isFlip = new int[n+1];

        for(int i=1;i<=n;i++) {
            pos[sc.nextInt()] = i;
        }

        int op=0;
        for(int i=1;i<=n;i++) {
            if(pos[i]==i && isFlip[i]==0) continue;

            if(isFlip[i]==1) {
                op++;
                System.out.println(i+" "+pos[i]);
                isFlip[i]=0;
                isFlip[pos[i]]=1;
            }
            int idx = pos[i];
            op+=2;
            System.out.println(idx+" "+i);
            pos[i]=idx;
            pos[idx]=i;
            isFlip[i]=0;
            isFlip[idx]=0;
        }

        System.out.println(op);
    }
}
