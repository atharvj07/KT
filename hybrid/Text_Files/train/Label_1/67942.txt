t = int(input())
for _ in range(t):
    a, s = map(int, input().split())
    a, s = str(a), str(s)
    b = ""
    i, j = len(a) - 1, len(s) - 1
    while i >= 0 and j >= 0:
        if int(s[j]) >= int(a[i]):
            b = str(int(s[j]) - int(a[i])) + b
            i -= 1
            j -= 1
        elif j-1 >= 0 and s[j-1] == '1':
            b = str(int(s[j-1:j+1]) - int(a[i])) + b
            i -= 1
            j -= 2
        else:
            b = "-1"
            break
    if b != "-1":
        if j >= 0:
            b = s[:j+1] + b
        if int(b) + int(a) != int(s):
            b = "-1"
    print(b)
