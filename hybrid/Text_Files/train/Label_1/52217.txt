import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        int[] pos = new int[n + 1];
        for (int i = 0; i < n; i++) {
            pos[c[i]] = i + 1;
        }
        int q = 0;
        for (int i = 1; i <= n; i++) {
            if (pos[i] == i) continue;
            int j = pos[i];
            System.out.printf("%d %d\n", j, i);
            pos[c[j - 1]] = j;
            pos[c[i - 1]] = i;
            c[j - 1] = i;
            c[i - 1] = j;
            q++;
        }
        System.out.println(q);
    }
}
