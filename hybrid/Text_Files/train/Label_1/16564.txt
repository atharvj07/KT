#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, x;
  cin >> n >> x;
  vector<int> a(n);
  int mx = 0;
  for (int &ai : a) {
    cin >> ai;
    mx = max(mx, ai);
  }
  int ans = 0;
  if (mx < x) ans = x - mx;
  if (!ans) {
    for (int ai : a) {
      if (ai == x) ans++;
    }
  }
  ans += int(a.size()) - int(mx <= x);
  cout << ans << '\n';
}
