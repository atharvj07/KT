#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n - 1);
  for (int i = 0; i < n - 1; i++) cin >> a[i];
  vector<int> dp(n);
  dp[n - 1] = 0;
  for (int i = n - 2; i >= 0; i--) {
    int cur = 1;
    for (int j = i + 1; j <= i + a[i] && j < n - 1; j++) cur = max(cur, 1 + dp[j]);
    dp[i] = cur;
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) ans += dp[i];
  cout << ans << endl;
  return 0;
}
