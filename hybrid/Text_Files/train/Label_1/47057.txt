import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String t1 = sc.next();
        String t2 = sc.next();
        int p1 = 0, p2 = 0;
        for (int i = 0; i < t1.length(); i++) {
            if (t1.charAt(i) == '8') p1 += 5;
            else if (t1.charAt(i) == '<') p1 += 3;
            else if (t1.charAt(i) == ')') p1 += 2;
            if (t2.charAt(i) == '8') p2 += 5;
            else if (t2.charAt(i) == '<') p2 += 3;
            else if (t2.charAt(i) == ')') p2 += 2;
        }
        if (p1 > p2) System.out.println("TEAM 1 WINS");
        else if (p2 > p1) System.out.println("TEAM 2 WINS");
        else System.out.println("TIE");
    }
}
