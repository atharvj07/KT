#include <iostream>
#include <vector>
#include <utility>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n), b(n);
    for (long long& i : a) cin >> i;
    for (long long& i : b) cin >> i;
    sort(a.begin(), a.end(), greater<>());
    sort(b.begin(), b.end(), greater<>());
    long long scoreA = 0, scoreB = 0;
    while (a.size() || b.size()) {
        if ((a.empty() || !b.empty()) && b.back() > a.back()) {
            scoreB += b.back();
            b.pop_back();
        }
        else if (!a.empty()) {
            scoreA += a.back();
            a.pop_back();
        }
        else {
            scoreB += b.back();
            b.pop_back();
        }
    }
    cout << scoreA - scoreB;
}
