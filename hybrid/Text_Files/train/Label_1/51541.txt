import java.util.*;
class Main{
  public static void main(String[] args){
    Scanner s = new Scanner(System.in);
    int n = s.nextInt();
    long a = s.nextLong(), b = s.nextLong();
    int[] l = new int[n];
    for(int i = 0; i < n; i++){
      l[i] = s.nextInt();
    }
    long ans = 0;
    for(int i = 0; i < n; i++){
      if(l[i] < a || l[i] > b){
        ans++;
      }
    }
    System.out.println(ans);
  }
}
