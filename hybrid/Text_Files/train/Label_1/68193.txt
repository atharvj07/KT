import sys
n = int(sys.stdin.readline())
g = [[int(x) for x in sys.stdin.readline().split()]for _ in range(n)]
a = []
for i in range(n):
    x, y = 1, 1
    for j in range(n):
        if i == j: continue
        if i < j:
            x = g[i][j] // x
        else:
            y = g[j][i] // y
    for j in range(n):
        if i == j: continue
        a.append(g[i][j] // x // y)
    break
print(*a)
