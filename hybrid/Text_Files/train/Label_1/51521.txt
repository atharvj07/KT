import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int t = Integer.parseInt(br.readLine());

        for (int i = 0; i < t; i++) {
            int n = Integer.parseInt(br.readLine());
            String s = br.readLine();
            char[] dir = new char[n];
            for (int j = 0; j < n; j++) {
                dir[j] = s.charAt(j);
            }

            int[] cycleIndex = new int[n];
            int cur = 0;
            int cycle = 0;
            Arrays.fill(cycleIndex, -1);
            for (int j = 0; j < n; j++) {
                if (dir[j] == '-') {
                    if (cycleIndex[j] != -1) {
                        cur = (cur + 1) % cycle;
                        continue;
                    }
                    if (cycleIndex[j] == -1) {
                        cycleIndex[j] = cycle;
                        cycleIndex[(j + 1) % n] = cycle;
                        cur = (cur + 2) % 2;
                        cycle++;
                    }
                } else if (dir[j] == '>') {
                    if (cycleIndex[j] != -1) {
                        cur = (cur + 1) % cycle;
                        continue;
                    }
                    if (cycleIndex[j] == -1) {
                        cycleIndex[j] = cycle;
                        cycleIndex[(j - 1 + n) % n] = cycle;
                        cur = (cur + 1) % 2;
                        cycle++;
                    }
                } else {
                    if (cycleIndex[j] != -1) {
                        cur = (cur + 1) % cycle;
                        continue;
                    }
                    if (cycleIndex[j] == -1) {
                        cycleIndex[j] = cycle;
                        cycleIndex[(j + 1) % n] = cycle;
                        cycleIndex[(j - 1 + n) % n] = cycle;
                        cur = (cur + 3) % 3;
                        cycle++;
                    }
                }
            }
            int ans = 0;
            for (int j = 0; j < n; j++) {
                if (cycleIndex[j] != -1 && cycleIndex[j] == cur) {
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
