from sys import stdin

n, m = map(int, stdin.readline().split())
p = list(map(int, stdin.readline().split()))
a = list(map(int, stdin.readline().split()))
dp = [0] * (n + 1)
for x in a:
    dp[x] = p[x - 1]
for i in range(1, n + 1):
    if dp[i] == 0:
        dp[i] = max(dp[i - 1], dp[i - 2])
for x in a:
    dp[x] = max(dp[x], dp[x - 1] + p[x - 1])
print(dp[n])
