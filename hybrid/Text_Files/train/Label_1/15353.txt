#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    string s;
    for (int i = 0; i < n; i++) {
        bool can_alice_win = false;
        for (int j = 0; j < n; j++) {
            if (a[i] > a[j] && (i - j) % a[i] == 0) {
                can_alice_win = true;
            }
        }
        if (can_alice_win) {
            s += 'A';
        } else {
            s += 'B';
        }
    }
    cout << s << endl;
    return 0;
}
