#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    bool flag = true;
    for (int i = 0; i < n - 1; i++) {
      if (a[i] > a[i + 1]) {
        if (i + x >= n) {
          flag = false;
          break;
        }
      }
    }
    cout << (flag ? "YES" : "NO") << endl;
  }
  return 0;
}
