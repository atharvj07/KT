#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 1;
ll a[maxn], b[maxn], sum[maxn], pre[maxn];
int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i] >> b[i];
    for (int i = 1; i <= n; i++) sum[i] = a[i] - b[i];
    sort(sum + 1, sum + n + 1);
    for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + sum[i];
    ll cnt = 0;
    for (int i = 1; i <= n; i++) {
        int pos = upper_bound(sum + 1, sum + n + 1, sum[i]) - sum - 1;
        cnt += 1ll * i * (i - 1) / 2 + 1ll * (n - pos) * (n - pos - 1) / 2;
        cnt -= 1ll * (pre[n] - pre[pos]) * i;
    }
    cout << cnt << "\n";
    return 0;
}
