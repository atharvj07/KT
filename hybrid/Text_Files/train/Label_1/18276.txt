#include <bits/stdc++.h>
using namespace std;
int f[200001], n, k, q, l, r, a, b;
int main() {
  cin >> n >> k >> q;
  for (int i = 0; i < n; ++i) {
    cin >> l >> r;
    for (int j = l; j <= r; ++j) f[j]++;
  }
  for (int i = 1; i <= 200000; ++i) f[i] += f[i - 1];
  for (int i = 0; i < q; ++i) {
    cin >> a >> b;
    cout << f[b] - f[a - 1] << endl;
  }
  return 0;
}
