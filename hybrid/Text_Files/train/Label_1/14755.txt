#include <iostream>
#include <vector>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    string s, t;
    cin >> s >> t;
    vector<int> cnt(26);
    for (int i = 0; i < n; i++) {
      cnt[s[i] - 'a']++;
      cnt[t[i] - 'a']--;
    }
    bool ok = true;
    for (int i = 0; i < 26; i++) {
      if (cnt[i] != 0) {
        ok = false;
        break;
      }
    }
    cout << (ok ? "YES" : "NO") << endl;
  }
}
