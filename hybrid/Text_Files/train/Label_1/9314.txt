#include<bits/stdc++.h>
using namespace std;
int main(){
	int t;cin >> t;
	while(t--){
		string s;
		int m;
		cin >> s >> m;
		vector<int> b(m);
		map<char,int> mp;
		for(int i=0;i<m;i++){
			cin >> b[i];
		}
		for(int i=0;i<s.size();i++){
			mp[s[i]] = 1;
		}
		string ans;
		for(int i=0;i<m;i++){
			for(int j=25;j>=0;j--){
				if(mp.find(char(97+j))!=mp.end()){
					ans += char(97+j);
					mp.erase(char(97+j));
					break;
				}
			}
		}
		cout << ans << "\n";
	}
	return 0;
}
