#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i].first;
  }
  for (int i = 0; i < n; i++) {
    cin >> v[i].second;
  }
  sort(v.begin(), v.end(), greater<pair<int, int>>());
  int ans = 0;
  for (int i = 0; i < k; i++) {
    ans += v[i].first;
  }
  for (int i = k; i < n; i++) {
    ans += min(v[i].first, v[i].second);
  }
  cout << ans << endl;
  return 0;
}
