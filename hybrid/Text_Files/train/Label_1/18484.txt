#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const int maxn = 3e5 + 5;
vector<int> g[maxn];
ll a[maxn], mx[maxn], mn[maxn], ans = -1;

void dfs(int v, int p = -1) {
    mx[v] = mn[v] = a[v];
    for (auto u : g[v]) {
        if (u == p) continue;
        dfs(u, v);
        if (a[v] == 0) {
            mn[v] = min(mn[v], mn[u]);
            mx[v] = min(mx[v], mx[u]);
        } else {
            mn[v] = max(mn[v], mn[u]);
            mx[v] = max(mx[v], mx[u]);
        }
    }
    ans = max({ans, mx[v], mn[v]});
}

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 2; i <= n; i++) {
        int p;
        cin >> p;
        g[i].push_back(p);
        g[p].push_back(i);
    }
    dfs(1);
    cout << ans << '\n';
    return 0;
}
