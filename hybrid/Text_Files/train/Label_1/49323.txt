import java.util.*;
import java.io.*;

class Main {
    static int n;
    static int[] a, cnt;
    static long pr[], ans;

    static long binpow(long x, long p) {
        if (p == 0)
            return 1;
        if (p == 1)
            return x;
        if (p % 2 == 0) {
            long ret = binpow(x, p / 2);
            return ret * ret % pr[n];
        }
        return x * binpow(x, p - 1) % pr[n];
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(in.readLine());
        a = new int[n + 1];
        cnt = new int[n + 1];
        StringTokenizer st = new StringTokenizer(in.readLine());
        pr = new long[n + 1];
        pr[0] = 1;
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
            cnt[a[i]]++;
        }
        for (int i = 1; i <= n; i++) {
            pr[i] = pr[i - 1] * i % (1000000007);
        }
        for (int i = 1; i <= n; i++) {
            if (cnt[i] == 0) {
                System.out.println(0);
                return;
            }
        }
        long p = 1;
        for (int i = 1; i <= n; i++) {
            p = p * binpow(pr[cnt[i]], pr[n - cnt[i]]) % pr[n];
        }
        int li = 0, ri = 0;
        for (int i = 1; i <= n; i++) {
            if (cnt[i] == 0)
                continue;
            if (cnt[i] >= 2) {
                ans = ans + (pr[cnt[i] - 1] * p) % pr[n];
            }
            if (ri > 0 && a[i] == a[ri])
                continue;
            if (ri > 0) {
                int r = i - ri;
                if (cnt[a[ri]] == 1)
                    ans += pr[r];
                else
                    ans += (pr[r] * binpow(binpow(pr[cnt[a[ri]] - 1], pr[r - cnt[a[ri]]]), cnt[a[ri]] - 1))
                            % pr[n];
            }
            ri = i;
            if (li == 0)
                li = i;
        }
        int r = n - ri + 1;
        if (cnt[a[ri]] == 1)
            ans += pr[r];
        else
            ans += (pr[r] * binpow(binpow(pr[cnt[a[ri]] - 1], pr[r - cnt[a[ri]]]), cnt[a[ri]] - 1)) % pr[n];
        System.out.println(ans % pr[n]);
    }
}
