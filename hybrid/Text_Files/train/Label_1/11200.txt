#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> intervals(n);
    for (int i = 0; i < n; i++) {
        cin >> intervals[i].first >> intervals[i].second;
    }

    int curr = -1;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (intervals[i].first >= curr) {
            curr = intervals[i].second;
            ans++;
        }
    }

    cout << ans << endl;
    return 0;
}
