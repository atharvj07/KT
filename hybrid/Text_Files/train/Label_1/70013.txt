import sys

def getMatching(edges, n, m):
    matching = []
    vis = [False] * (3*n+1)
    for edge in edges:
        a, b = edge
        if vis[a] or vis[b]: continue
        matching.append(edge)
        vis[a] = vis[b] = True
    if len(matching) == n: return matching
    return []

def getIndSet(edges, n, m):
    vis = [False] * (3*n+1)
    indSet = []
    for edge in edges:
        a, b = edge
        if vis[a] or vis[b]: continue
        indSet.append(a)
        vis[a] = True
        break
    cnt = 1
    for i in range(1, 3*n+1):
        if vis[i]: continue
        indSet.append(i)
        cnt += 1
        if cnt == n: return indSet
    return []

for t in range(int(sys.stdin.readline())):
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    for i in range(m):
        a, b = map(int, sys.stdin.readline().split())
        edges.append((a, b))
    matching = getMatching(edges, n, m)
    if matching:
        print("Matching")
        print(*[edges.index(edge)+1 for edge in matching])
    else:
        indSet = getIndSet(edges, n, m)
        if indSet:
            print("IndSet")
            print(*indSet)
        else:
            print("Impossible")
