#include<bits/stdc++.h>
using namespace std;
int main()
{
    string s; cin >> s;
    int n = (int)s.size(), i, j;
    bool ok = 0;
    for (i = 0; i < n; i++)
    {
        for (j = i + 1; j < n; j++)
            if (s[i] == s[j])
                ok = 1;
    }
    if (ok == 0)
        cout << "NA";
    else
    {
        int k = n;
        for (i = n - 1; i >= 0; i--)
        {
            if (s[i] != s[k - i - 1])
                break;
        }
        if (i < 0)
        {
            char c = s[0];
            for (i = 1; i < n; i++)
                if (s[i] != c)
                    break;
            if (i == n)
            {
                for (i = 0; i < 26; i++)
                {
                    char t = 'a' + i;
                    if (t != c)
                    {
                        cout << t << s;
                        return 0;
                    }
                }
            }
            else
            {
                string ans = s.substr(i);
                reverse(ans.begin(), ans.end());
                cout << ans << s;
            }
        }
        else
        {
            string ans = s.substr(i + 1);
            reverse(ans.begin(), ans.end());
            cout << ans << s;
        }
    }
    return 0;
}
