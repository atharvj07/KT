import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            long l = sc.nextLong(), r = sc.nextLong(), d = sc.nextLong();
            long x = l / d + (l % d == 0 ? 0 : 1);
            if (x * d <= r) {
                x++;
            }
            System.out.println(x * d);
        }
        sc.close();
    }
}
