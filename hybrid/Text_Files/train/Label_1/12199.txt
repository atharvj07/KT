#include <iostream>
using namespace std;

long long n, a[100000], l[100000], r[100000], ans = 1e18;

int main() {
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> a[i] >> l[i];
  for (int i = 0; i < n; ++i) cin >> r[i];
  for (int i = 0, j = 0; i < n; ++i, ++j) {
    if (j > 0) a[j] += a[j - 1];
    ans = min(ans, j * a[j] + r[i] * (n - j - 1) + l[i]);
  }
  for (int j = n - 2, i = n - 1; i >= 0; --i, --j) {
    if (j < n - 1) r[j] += r[j + 1];
    ans = min(ans, l[i] * (n - j - 2) + r[j] * (j + 1) + a[i]);
  }
  cout << ans << '\n';
  return 0;
}
