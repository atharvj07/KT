#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  int t, n, x;
  cin >> t;
  while (t--) {
    cin >> n >> x;
    int ans = 1e9;
    for (int i = 0; i < n; i++) {
      int d, h;
      cin >> d >> h;
      if (d > x) continue;
      ans = min(ans, 1 + (x - d + h - 1) / h);
    }
    if (ans != 1e9)
      cout << ans << '\n';
    else
      cout << -1 << '\n';
  }
}
