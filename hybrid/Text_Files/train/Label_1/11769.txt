#include <bits/stdc++.h>

using namespace std;

const int MAXN = 70;

int d[MAXN][MAXN], n;
char c[MAXN][MAXN], ans[MAXN][MAXN];

int dij(int s) {
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> q;
    fill (&d[0][0], &d[n][n], -1);
    d[s][s] = 0;
    q.push({0, s});
    while (!q.empty()) {
        int v = q.top().second;
        int w = q.top().first;
        q.pop();
        if (d[v][v] == -1) continue;
        for (int i = 1; i <= n; ++i)
            if (i != v && d[v][i] == -1 && c[v][i] == c[s][v]) {
                d[v][i] = w + 1;
                q.push({w + 1, i});
                ans[v][i] = v;
            }
    }
    return d[1][2] == -1 ? -1 : d[1][2];
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    cin >> n;
    for (int i = 0; i < n; ++i) cin >> d[i][i];
    for (int i = 1; i < n; ++i) {
        string s;
        cin >> s;
        for (int j = 0; j < n; ++j) c[i][j] = s[j];
    }
    int k = dij(3);
    if (k > 0) {
        while (true) {
            int from = 3, to = 1;
            while (from != to)
                to = ans[from][to], cout << from << ' ' << to << '\n', from = to;
            if (from == 2) break;
            swap(from, to);
            while (from != to)
                to = ans[from][to], cout << from << ' ' << to << '\n', from = to;
        }
    } else
        cout << -1;
    return 0;
}
