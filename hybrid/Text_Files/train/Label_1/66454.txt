def gen(n, y):
    x = []
    for i in range(n):
        if y[i] % 2 == 1:
            y[i] -= 1
            y[i] //= 2
            x.append(y[i])
        else:
            y[i] //= 2
            x.append(y[i])
    return x

n = int(input())
y = list(map(int, input().split()))
x = gen(n, y[:])
print(*x)
