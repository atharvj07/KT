#include<bits/stdc++.h>
using namespace std;
#define edge pair< int,int >
#define vi vector<int>
const int maxn=505;
vector< edge > g[maxn];
vector<int> trk;
int vis[maxn];
bool dfs(int nd,int p){
    vis[nd]=1;
    trk.push_back(nd);
    for(auto x: g[nd]){
        int nxt=x.first;
        if(vis[nxt]==1 && nxt!=p) return true;
        if(vis[nxt]==0 && dfs(nxt,nd)) return true;
    }
    trk.pop_back();
    vis[nd]=2;
    return false;
}
int main(){
    int n,m;
    cin>>n>>m;
    while(m--){
        int a,b; cin>>a>>b;
        g[a].push_back(edge(b,0));
    }
    bool p=false;
    for(int i=1;i<=n;i++){
        if(!vis[i]){
            if(dfs(i,-1)){
                p=true;
            }
        }
    }
    if(p){
        cout<<"YES\n";
    }
    else cout<<"NO\n";
}
