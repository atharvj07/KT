#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long m,k,ans=0,i,need,last=0;
    cin>>m>>k;
    vector <long long int> d(m),s(m);
    for(i=0;i<m;i++) cin>>d[i];
    for(i=0;i<m;i++) cin>>s[i];
    for(i=0;i<m;i++)
    {
        need=(d[i]-last+s[i]-1)/s[i];
        if(need*(s[i]+k)>d[i]) last=d[i];
        else last+=need*(s[i]+k);
        ans+=need*(s[i]+k);
    }
    cout<<ans<<endl;
}
