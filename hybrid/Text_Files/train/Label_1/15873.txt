#include<bits/stdc++.h>
using namespace std;
const int maxn = 105;
int a[maxn][3];
int b[maxn][maxn];

int32_t main() {
	int n, m;
	cin >> n >> m;
	for (int i = 1; i <= m; i++) {
		cin >> a[i][0] >> a[i][1] >> a[i][2];
	}
	for (int i = 1; i <= m; i++) {
		int d = a[i][1] - a[i][0] - a[i][2];
		if (d < 0) {
			cout << -1;
			return 0;
		}
		for (int j = a[i][0] - 1; j >= a[i][0] - d; j--)b[i][j] = 1;
	}
	int pos = 0;
	for (int j = 1; j <= n; j++) {
		if (pos != 0)b[pos][j] = 2;
		bool ok = false;
		for (int i = 1; i <= m; i++)if (b[i][j] == 1) {
			pos = i;
			b[i][j] = 2;
			ok = true;
			break;
		}
		if (!ok)b[0][j] = 3;
		//cout << b[pos][j] << " ";
	}
	for (int i = 0; i <= m; i++) {
		for (int j = 1; j <= n; j++) {
			if (b[i][j] == 2)cout << i << " ";
			if (b[i][j] == 3)cout << 0 << " ";
			if (b[i][j] == 1)cout << i << " ";
		}
	}
}
