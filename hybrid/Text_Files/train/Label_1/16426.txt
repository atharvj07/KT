#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    sort(v.begin(), v.end());
    int ans = 0;
    for (int i = n - 2; i >= 0; i--) {
        ans = max(ans, v[n - 1] ^ v[i]);
    }
    cout << ans << endl;
    return 0;
}
