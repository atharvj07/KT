import sys

n, m = [ int(x) for x in sys.stdin.readline().split() ]
ops = [ [ int(x) for x in sys.stdin.readline().split() ] for i in range(m) ]

a = [0] * n
d = [0] * n

for op in ops:
    if op[0] == 1:
        l, r, di = op[1:]
        d[l-1] += di
        d[r] -= di
    else:
        l, r, mi = op[1:]
        if mi <= a[l-1]:
            sys.stdout.write("NO")
            sys.exit(0)
        a[l-1] = mi

for i in range(1, n):
    d[i] += d[i-1]
    a[i] = max(a[i], a[i-1] + d[i])

sys.stdout.write("YES\n")
for x in a:
    sys.stdout.write("%d " %x)
