import sys

n = int(input())
s = list(input())
q = int(input())

for _ in range(q):
    m, c = input().split()
    m = int(m)

    # Find the longest segment of color c
    l = 0
    r = 0
    max_len = 0
    while r < n:
        if s[r] == c or m > 0:
            if s[r] != c:
                m -= 1
            r += 1
            max_len = max(max_len, r - l)
        else:
            while s[l] != c:
                l += 1
            l += 1
            r += 1

    print(max_len)
