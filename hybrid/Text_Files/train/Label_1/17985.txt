#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>

vector<pair<int, ll>> get_prairie(ll x) {
    vector<pair<int, ll>> res;
    int k = 0;
    while ((1LL << k) - 1 < x) k++;
    res.push_back({k, (1LL << k) - 1});
    x -= (1LL << k) - 1;
    while (x) {
        int tmp = (1LL << k) - 1;
        while (tmp > 0 && tmp > x) tmp >>= 1;
        res.push_back({k, tmp});
        x -= tmp;
        k--;
    }
    return res;
}
int main() {
    int n, tmp;
    cin >> n;
    unordered_map<ll, vi> m;
    unordered_set<ll> s;
    for (int i = 0; i < n; i++) {
        cin >> tmp;
        s.insert(tmp);
        for (auto &p : get_prairie(tmp)) { m[p.second].push_back(p.first); }
    }
    vi res;
    for (auto &p : s) {
        vi vec = m[p];
        sort(vec.begin(), vec.end());
        vi new_vec;
        for (int i = 0; i < vec.size(); i++) {
            if (i + 1 < vec.size() && vec[i] == vec[i + 1]) continue;
            new_vec.push_back(vec[i]);
        }
        for (int i = 0; i < new_vec.size(); i++) { res.push_back(new_vec[i]); }
    }
    sort(res.begin(), res.end());
    res.erase(unique(res.begin(), res.end()), res.end());
    if (res.empty()) {
        cout << -1 << endl;
        return 0;
    }
    for (auto &x : res) cout << x << " ";
    cout << endl;
    return 0;
}
