#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll maxn = 10005;
vector<ll> adj[maxn];
bool vis[maxn];
void dfs(ll v){
    vis[v] = 1;
    for(ll i = 0; i < adj[v].size(); i++){
        ll x = adj[v][i];
        if(!vis[x]) dfs(x);
    }
}
int main(){
    ll n, a, b, ans = 0; cin >> n;
    for(ll i = 1; i < n; i++){
        cin >> a >> b;
        adj[a].push_back(b); adj[b].push_back(a);
    }
    dfs(1);
    for(ll i = 1; i <= n; i++){
        for(ll j = 0; j < adj[i].size(); j++){
            if(vis[adj[i][j]]) ans++;
        }
    }
    cout << ans/2 << endl;
    return 0;
}
