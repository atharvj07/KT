#include <iostream>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  int i = 0;
  char arr[100];
  while (k > 0) {
    arr[i] = 'a' + i;
    i++;
    k--;
  }
  int j = 0;
  while (i < n) {
    arr[i] = arr[j];
    i++;
    j++;
  }
  for (int i = 0; i < n; i++) { cout << arr[i]; }
  return 0;
}
