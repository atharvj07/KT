x1, y1 = list(map(int, input().split(' ')))
x2, y2 = list(map(int, input().split(' ')))
n = int(input())
s = input()
period = set()
for i in range(n):
    if i == 0:
        period.add(s[i])
    elif s[i] != s[i-1]:
        period.add(s[i])
    else:
        break
min_days = float('inf')
for p in period:
    dx, dy = 0, 0
    if p == 'U':
        dy = 1
    elif p == 'D':
        dy = -1
    elif p == 'L':
        dx = -1
    elif p == 'R':
        dx = 1
    days = abs(x2 - x1 + y2 - y1) // 2 + 1
    if (x2 - x1) % 2 != 0 or (y2 - y1) % 2 != 0:
        if s[0] != p:
            days += 1
    if (abs(x2 - x1) - abs(y2 - y1)) % 2 != 0:
        if s[0] != p:
            days += 1
    if dx * (x2 - x1) + dy * (y2 - y1) <= 0:
        if s[0] == p:
            days -= 1
    if days < min_days:
        min_days = days
if min_days == float('inf'):
    print(-1)
else:
    print(min_days)
