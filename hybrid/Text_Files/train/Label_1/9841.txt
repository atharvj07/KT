#include <bits/stdc++.h>
using namespace std;
const int MOD = 998244353;
int ad(int &a, int b) {
  a += b;
  if(a >= MOD) a -= MOD;
}
int sb(int &a, int b) {
  a -= b;
  if(a < 0) a += MOD;
}
int mm(int &a, int b) {
  a = (1ll * a * b) % MOD;
}
int fpow(int b, int e) {
  int ans = 1;
  while(e > 0) {
    if(e % 2 == 1) mm(ans, b);
    mm(b, b);
    e /= 2;
  }
  return ans;
}
const int N = 105;
int n, m, k;
vector<int> adj[N];
vector<int> col;
vector<vector<int>> cost;
vector<bool> vis;
int cur_beauty;
int trans(char c) {
  return c - 'a';
}
void dfs(int nd) {
  vis[nd] = true;
  cur_beauty++;
  for(auto u : adj[nd]) {
    if(!vis[u]) dfs(u);
  }
}
int get(int beauty) {
  if(beauty < 0) return 0;
  int ans = 2 * m * (k - beauty) + 1;
  // cout << "beauty: " << beauty << " ans: " << ans << '\n';
  return ans;
}
int32_t main() {
  cin >> n >> m >> k;
  col.resize(n + 1);
  cost.resize(n + 1, vector<int>(m + 1, 0));
  for(int i = 1; i <= n; i++) {
    int c;
    cin >> c;
    col[i] = c;
  }
  for(int i = 1; i <= n; i++) {
    for(int j = 1; j <= m; j++) {
      cin >> cost[i][j];
    }
  }
  int ans = 1e9;
  int tot = 0;
  for(int i = 1; i <= n; i++) {
    if(col[i] != 0) continue;
    int mn = 1e9;
    for(int j = 1; j <= m; j++) {
      mn = min(mn, cost[i][j]);
    }
    tot += mn;
  }
  for(int i = 1; i <= n; i++) {
    if(col[i] != 0) continue;
    vis.assign(n + 1, false);
    cur_beauty = 0;
    dfs(i);
    cur_beauty -= 1;
    if(cur_beauty == k) {
      int cans = 0;
      for(int j = 1; j <= m; j++) {
        cans += cost[i][j];
      }
      ans = min(ans, cans);
    }
  }
  if(ans == 1e9) ans = -1;
  ans += tot;
  cout << ans << '\n';
  return 0;
}
