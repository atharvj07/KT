def is_accident_possible():
    parts = [list(map(int, input().split())) for _ in range(4)]
    for i in range(4):
        if parts[i][3] == 1:  # Check if pedestrian light is green
            # Check if any car light is green in the same part
            if parts[i][0] == 1 or parts[i][1] == 1 or parts[i][2] == 1:
                return "YES"
            # Check if cars from other parts can hit the pedestrian
            if parts[(i + 1) % 4][0] == 1 or parts[(i + 2) % 4][1] == 1 or parts[(i + 3) % 4][2] == 1:
                return "YES"
    return "NO"

print(is_accident_possible())
