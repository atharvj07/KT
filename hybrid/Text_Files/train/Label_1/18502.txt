#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 2e5 + 5;

char s[MAXN];
int main() {
  int n;
  cin >> n;
  cin >> s;
  int ans = 0;
  char op = s[0];
  for (int i = 1; i < n; i++) {
    if (s[i] != op) {
      ans++;
      if (ans % 3 == 0) op = s[i];
    }
  }
  cout << ans << endl;
  for (int i = 0; i < n; i++) {
    if (ans == 0)
      cout << s[i];
    else {
      if ((ans + 1) % 3 != 0) op = s[i];
      cout << op;
      if ((ans + 1) % 3 == 0) ans--;
    }
  }
  cout << endl;
  return 0;
}
