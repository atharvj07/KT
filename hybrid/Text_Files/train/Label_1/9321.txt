#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b, c, d;
    cin >> a >> b >> c >> d;
    long double timeToFallAsleep = 0.0, timeAsleep = 0.0;
    while (timeAsleep < a) {
      if (timeToFallAsleep > 0) {
        timeAsleep += d;
        timeToFallAsleep = c;
      } else {
        timeToFallAsleep = b;
      }
      timeAsleep += timeToFallAsleep;
      timeToFallAsleep = 0;
    }
    cout << (a <= b ? b : timeAsleep) << endl;
  }
  return 0;
}
