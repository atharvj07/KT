n, c = map(int, input().split())
p = list(map(int, input().split()))
t = list(map(int, input().split()))

limak_points = 0
radewoosh_points = 0
time_l = 0
time_r = 0

for i in range(n):
    time_l += t[i]
    limak_points += max(0, p[i] - c * time_l)
    
    time_r += t[n - 1 - i]
    radewoosh_points += max(0, p[n - 1 - i] - c * time_r)

if limak_points > radewoosh_points:
    print("Limak")
elif limak_points < radewoosh_points:
    print("Radewoosh")
else:
    print("Tie")
