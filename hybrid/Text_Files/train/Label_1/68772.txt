n, m = map(int, input().split())
a = [[]for i in range(n+2)]
a[0] = [[0]*m for j in range(m+2)]
for i in range(1, n+1):
    a[i]+=[0]
    a[i]+=list(map(int, input().split()))
    a[i]+=[0]
a[n+1] = [[0]*m for j in range(m+2)]
dp = [[[-1]*(m+2) for j in range(m+2)]for i in range(n+2)]

def dfs(x, y, up):
    if dp[x][y][up] != -1:
        return dp[x][y][up]
    mx = -1e9
    for k in range(up+1):
        for dt in range(2):
            nx, ny = x+dt, y+1-dt
            v = a[nx][ny]
            if v > a[x][y] and v != 0:
                mnleft = -1e9
                if dt == 0:
                    mnleft = dfs(nx, ny, k)
                elif nx > 1 and k > 0:
                    mnleft = dfs(nx, ny, k-1)
                if mnleft != -1e9:
                    mx = max(mx, k+mnleft)
    dp[x][y][up] = mx
    return mx

ans = dfs(1, 1, 0)
print(-1 if ans < 0 else ans+(a[1][1]==0))
