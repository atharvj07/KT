from collections import defaultdict

n, m = map(int, input().split())
edges = defaultdict(list)
for _ in range(m):
    a, b = map(int, input().split())
    edges[a].append(b)
    edges[b].append(a)

ans = 0
for v in edges:
    deg = len(edges[v])
    if deg % 2 == 1:
        ans += deg // 2
    else:
        ans += deg // 2 - 1

print(ans)
