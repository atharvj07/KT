#include <iostream>
#include <string>
#include <utility>
#include <vector>

int main() {
  const int max_len = 100, max_choice = 4;
  std::vector<std::string> choice(max_choice);
  std::pair<int, int> min_max(max_len, 0);

  for (int i = 0; i < max_choice; ++i) {
    std::string line;
    std::getline(std::cin, line);
    int length = line.length() - 2;
    choice[i] = line.substr(2);
    if (length < min_max.first) min_max.first = length;
    if (length > min_max.second) min_max.second = length;
  }

  if (min_max.first * 2 >= min_max.second) {
    for (int i = 0; i < max_choice; ++i)
      if (choice[i].length() == min_max.first) {
        std::cout << char('A' + i) << std::endl;
        return 0;
      }
  } else if (min_max.second * 2 >= min_max.first) {
    for (int i = 0; i < max_choice; ++i)
      if (choice[i].length() == min_max.second) {
        std::cout << char('A' + i) << std::endl;
        return 0;
      }
  }

  std::cout << "C" << std::endl;
  return 0;
}
