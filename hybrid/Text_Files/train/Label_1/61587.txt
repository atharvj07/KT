n = int(input())
a = list(map(int, input().split()))
for i in range(1, n, 2):
    if a[i] < a[i - 1]:
        a[i], a[i - 1] = a[i - 1], a[i]
for i in range(2, n, 2):
    if a[i] > a[i - 1]:
        a[i], a[i - 1] = a[i - 1], a[i]
if all(a[i] >= a[i - 1] for i in range(1, n, 2)) and all(a[i] <= a[i - 1] for i in range(2, n, 2)):
    print(*a)
else:
    print("Impossible")
