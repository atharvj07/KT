#include <iostream>
#include <vector>

using namespace std;

int main() {
  int a;
  cin >> a;
  int sum = 0;
  vector<int> digits;
  while (a > 0) {
    digits.push_back(a % 10);
    sum += a % 10;
    a /= 10;
  }
  int remainder = sum % 4;
  int carry = 0;
  if (remainder != 0) {
    carry = 4 - remainder;
  }
  int i = digits.size() - 1;
  while (carry > 0 && i >= 0) {
    if (digits[i] + carry >= 10) {
      digits[i] = (digits[i] + carry) % 10;
      carry = 1;
    } else {
      digits[i] += carry;
      carry = 0;
    }
    i--;
  }
  if (carry > 0) {
    digits.insert(digits.begin(), carry);
  }
  for (int i = digits.size() - 1; i >= 0; i--) {
    cout << digits[i];
  }
  return 0;
}
