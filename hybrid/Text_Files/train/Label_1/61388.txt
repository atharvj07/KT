import sys
input = sys.stdin.readline

t, sx, sy, ex, ey = map(int, input().split())
wind = input().strip()

dx = [1, 0, -1, 0]
dy = [0, -1, 0, 1]
dirs = {'E': 0, 'S': 1, 'W': 2, 'N': 3}

x, y = sx, sy
for i in range(t):
    d = dirs[wind[i]]
    x += dx[d]
    y += dy[d]
    if x == ex and y == ey:
        print(i + 1)
        exit()
print(-1)
